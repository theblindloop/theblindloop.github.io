"""Independent pixel-only inverse for fraction denominator correspondence.

This module deliberately repeats only the public raster contract: the reference
card is above two fixed candidate cards, A is left and B is right, and solid
numerator marks are the colored connected components above each fraction bar.
It does not import or inspect the renderer, latent dictionaries, or manifests.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


ORACLE_VERSION = "fraction_denominator_correspondence-oracle-0.1.0"

# These are the public non-green numerator swatches.  They are intentionally
# duplicated rather than imported from the renderer.
NUMERATOR_SWATCHES = np.array(
    [
        (207, 72, 68),
        (56, 108, 184),
        (222, 137, 41),
        (145, 83, 173),
        (25, 143, 130),
        (194, 69, 133),
    ],
    dtype=np.uint8,
)
GREEN_SWATCH = np.array((42, 156, 86), dtype=np.uint8)


def _exact_color_mask(pixels: np.ndarray, swatches: np.ndarray) -> np.ndarray:
    matches = np.all(pixels[:, :, None, :] == swatches[None, None, :, :], axis=3)
    return np.any(matches, axis=2)


def _component_count(mask: np.ndarray) -> int:
    """Count separated 8-connected components without scene assumptions."""

    if mask.ndim != 2:
        raise ValueError("component mask must be two-dimensional")
    pending = mask.copy()
    height, width = pending.shape
    count = 0
    for row in range(height):
        for column in range(width):
            if not pending[row, column]:
                continue
            count += 1
            stack = [(row, column)]
            pending[row, column] = False
            while stack:
                current_row, current_column = stack.pop()
                for dr in (-1, 0, 1):
                    for dc in (-1, 0, 1):
                        if dr == 0 and dc == 0:
                            continue
                        rr = current_row + dr
                        cc = current_column + dc
                        if 0 <= rr < height and 0 <= cc < width and pending[rr, cc]:
                            pending[rr, cc] = False
                            stack.append((rr, cc))
    return count


def _count_green_reference(pixels: np.ndarray) -> int:
    crop = pixels[208:278, 260:700]
    mask = np.all(crop == GREEN_SWATCH, axis=2)
    return _component_count(mask)


def _count_candidate_numerator(pixels: np.ndarray, left: int, right: int) -> int:
    # The numerator band is above the candidate fraction bar.  The card labels
    # and borders are dark, while all possible numerator marks use a swatch.
    crop = pixels[365:442, left:right]
    return _component_count(_exact_color_mask(crop, NUMERATOR_SWATCHES))


def decision_from_image(image):
    """Return ``A`` or ``B`` from the final PNG alone."""

    pixels = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if pixels.shape[0] < 442 or pixels.shape[1] < 870:
        raise ValueError(f"unexpected raster size {pixels.shape[:2]}")
    reference_count = _count_green_reference(pixels)
    count_a = _count_candidate_numerator(pixels, 90, 430)
    count_b = _count_candidate_numerator(pixels, 530, 870)
    if reference_count < 1 or count_a < 1 or count_b < 1:
        raise ValueError("could not recover all colored mark groups")
    distance_a = abs(count_a - reference_count)
    distance_b = abs(count_b - reference_count)
    if distance_a == distance_b:
        raise ValueError("candidate correspondence is tied")
    return "A" if distance_a < distance_b else "B"
