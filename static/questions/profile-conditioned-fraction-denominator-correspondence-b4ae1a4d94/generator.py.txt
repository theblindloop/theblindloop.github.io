"""Renderer and analytic decision for the fraction correspondence world.

The visible object is a reference fraction card and two candidate fraction
cards.  A card contains solid marks above and below a fraction bar.  The
question asks for equality of the reference denominator's mark count and a
candidate numerator's mark count; all non-count presentation fields are
rendered and randomized.
"""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw


RENDERER_VERSION = "fraction_denominator_correspondence-0.1.0"
WIDTH, HEIGHT = 960, 640
BACKGROUND = (250, 248, 242)
INK = (38, 45, 56)
CARD_FILL = (255, 255, 252)
GREEN = (42, 156, 86)
NUMBER_COLORS = (
    (207, 72, 68),
    (56, 108, 184),
    (222, 137, 41),
    (145, 83, 173),
    (25, 143, 130),
    (194, 69, 133),
)
DENOMINATOR_COLORS = (
    (112, 124, 143),
    (94, 116, 126),
    (126, 111, 132),
)
TOTAL_MARKS = 10
CARD_Y = 330
CARD_W, CARD_H = 380, 260
CARD_X = {"A": 70, "B": 510}
REFERENCE_BOX = (240, 58, 720, 278)


# A tiny deterministic bitmap alphabet keeps labels independent of host fonts.
_GLYPHS = {
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "B": ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    "C": ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    "E": ("11111", "10000", "10000", "11110", "10000", "10000", "11111"),
    "F": ("11111", "10000", "10000", "11110", "10000", "10000", "10000"),
    "N": ("10001", "11001", "11001", "10101", "10011", "10011", "10001"),
    "R": ("11110", "10001", "10001", "11110", "10100", "10010", "10001"),
    "?": ("01110", "10001", "00001", "00010", "00100", "00000", "00100"),
}


def _rgb(value: Iterable[int]) -> tuple[int, int, int]:
    return tuple(int(item) for item in value)  # type: ignore[return-value]


def _draw_text(
    draw: ImageDraw.ImageDraw,
    text: str,
    xy: tuple[int, int],
    *,
    scale: int = 3,
    color: tuple[int, int, int] = INK,
    gap: int = 1,
) -> tuple[int, int]:
    """Draw uppercase bitmap text and return its pixel dimensions."""

    x, y = xy
    start_x = x
    for char in text:
        if char == " ":
            x += 4 * scale
            continue
        glyph = _GLYPHS.get(char)
        if glyph is None:
            x += 6 * scale
            continue
        for row, bits in enumerate(glyph):
            for col, bit in enumerate(bits):
                if bit == "1":
                    draw.rectangle(
                        (
                            x + col * scale,
                            y + row * scale,
                            x + (col + 1) * scale - 1,
                            y + (row + 1) * scale - 1,
                        ),
                        fill=color,
                    )
        x += 5 * scale + gap * scale
    return x - start_x, 7 * scale


def _center_text(
    draw: ImageDraw.ImageDraw,
    text: str,
    center_x: int,
    y: int,
    *,
    scale: int,
    color: tuple[int, int, int] = INK,
) -> None:
    width = sum((4 if char == " " else 6) * scale for char in text)
    _draw_text(draw, text, (int(center_x - width / 2), y), scale=scale, color=color)


def _row_x_positions(center_x: int, count: int, phase: int) -> list[int]:
    if count < 0 or count > 8:
        raise ValueError(f"mark count outside renderable range: {count}")
    if count == 0:
        return []
    spacing = 34
    offset = (int(phase) % 5 - 2) * 2
    return [
        int(round(center_x + offset + (index - (count - 1) / 2) * spacing))
        for index in range(count)
    ]


def _draw_mark_row(
    draw: ImageDraw.ImageDraw,
    *,
    center_x: int,
    y: int,
    count: int,
    radius: int,
    color: tuple[int, int, int],
    phase: int,
) -> None:
    for x in _row_x_positions(center_x, count, phase):
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)


def _draw_empty_row(
    draw: ImageDraw.ImageDraw,
    *,
    center_x: int,
    y: int,
    count: int,
    radius: int,
    phase: int,
) -> None:
    for x in _row_x_positions(center_x, count, phase):
        draw.ellipse(
            (x - radius, y - radius, x + radius, y + radius),
            fill=CARD_FILL,
            outline=INK,
            width=3,
        )


def _draw_candidate_card(
    draw: ImageDraw.ImageDraw,
    *,
    label: str,
    candidate: dict,
) -> None:
    x = CARD_X[label]
    y = CARD_Y
    draw.rounded_rectangle(
        (x, y, x + CARD_W, y + CARD_H),
        radius=18,
        fill=CARD_FILL,
        outline=INK,
        width=3,
    )
    _draw_text(draw, label, (x + 22, y + 18), scale=4)
    center_x = x + CARD_W // 2
    radius = int(candidate["mark_radius"])
    _draw_mark_row(
        draw,
        center_x=center_x,
        y=y + 72,
        count=int(candidate["numerator"]),
        radius=radius,
        color=_rgb(candidate["numerator_color"]),
        phase=int(candidate["numerator_phase"]),
    )
    draw.line((x + 62, y + 126, x + CARD_W - 62, y + 126), fill=INK, width=4)
    _draw_mark_row(
        draw,
        center_x=center_x,
        y=y + 188,
        count=int(candidate["denominator"]),
        radius=radius,
        color=_rgb(candidate["denominator_color"]),
        phase=int(candidate["denominator_phase"]),
    )


def sample_scene(seed: int) -> dict:
    """Return one deterministic, balanced correspondence scene."""

    rng = random.Random(int(seed))
    reference_count = rng.randint(3, 7)
    if reference_count == 3:
        wrong_count = 4
    elif reference_count == 7:
        wrong_count = 6
    else:
        wrong_count = reference_count + (1 if rng.random() < 0.5 else -1)

    # Seed parity counterbalances A/B while remaining invisible in the raster.
    correct_label = "A" if int(seed) % 2 == 0 else "B"
    numerator_by_label = {
        correct_label: reference_count,
        ("B" if correct_label == "A" else "A"): wrong_count,
    }
    colors = rng.sample(list(NUMBER_COLORS), 2)
    denominator_colors = [rng.choice(DENOMINATOR_COLORS), rng.choice(DENOMINATOR_COLORS)]

    candidates = {}
    for index, label in enumerate(("A", "B")):
        numerator = numerator_by_label[label]
        radius = rng.choice((12, 13, 14))
        candidates[label] = {
            "numerator": numerator,
            "denominator": TOTAL_MARKS - numerator,
            "numerator_color": list(colors[index]),
            "denominator_color": list(denominator_colors[index]),
            "mark_radius": radius,
            "numerator_phase": rng.randrange(5),
            "denominator_phase": rng.randrange(5),
        }

    return {
        "reference_denominator": {
            "count": reference_count,
            "color": list(GREEN),
            "mark_radius": rng.choice((12, 13, 14)),
            "phase": rng.randrange(5),
        },
        "candidate_a": candidates["A"],
        "candidate_b": candidates["B"],
    }


def render(scene: dict) -> Image.Image:
    """Render the scene as a deterministic RGB raster."""

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    left, top, right, bottom = REFERENCE_BOX
    draw.rounded_rectangle(
        REFERENCE_BOX,
        radius=20,
        fill=CARD_FILL,
        outline=INK,
        width=3,
    )
    _center_text(draw, "REFERENCE", (left + right) // 2, top + 18, scale=2)
    reference = scene["reference_denominator"]
    center_x = (left + right) // 2
    _draw_empty_row(
        draw,
        center_x=center_x,
        y=top + 78,
        count=3,
        radius=11,
        phase=int(reference["phase"]),
    )
    draw.line((left + 80, top + 126, right - 80, top + 126), fill=INK, width=4)
    _draw_mark_row(
        draw,
        center_x=center_x,
        y=top + 182,
        count=int(reference["count"]),
        radius=int(reference["mark_radius"]),
        color=_rgb(reference["color"]),
        phase=int(reference["phase"]),
    )

    _draw_candidate_card(draw, label="A", candidate=scene["candidate_a"])
    _draw_candidate_card(draw, label="B", candidate=scene["candidate_b"])
    return image


def analytic_gold(scene: dict) -> str:
    """Select the candidate whose numerator count matches the reference."""

    reference_count = int(scene["reference_denominator"]["count"])
    a_distance = abs(int(scene["candidate_a"]["numerator"]) - reference_count)
    b_distance = abs(int(scene["candidate_b"]["numerator"]) - reference_count)
    return "A" if a_distance < b_distance else "B"


def margin(scene: dict) -> float:
    """Signed discrete advantage of A over B; zero is the tie boundary."""

    reference_count = int(scene["reference_denominator"]["count"])
    a_distance = abs(int(scene["candidate_a"]["numerator"]) - reference_count)
    b_distance = abs(int(scene["candidate_b"]["numerator"]) - reference_count)
    return float(b_distance - a_distance)


def is_quarantined(scene: dict) -> bool:
    """Quarantine a non-unique correspondence at the visible tie boundary."""

    return abs(margin(scene)) < 0.5


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No hidden aliases: each analytic label input is visibly rendered."""

    return []
