"""Renderer, analytic gold, and sampler for reflected window-pair correspondence."""

from __future__ import annotations

import copy
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

RENDERER_VERSION = "window-pair-reflection-correspondence-0.1.0"
WIDTH = 512
HEIGHT = 512
SCALE = 4

BACKGROUND = (247, 246, 241)
BOARD = (39, 78, 92)
BOARD_EDGE = (24, 51, 64)
TILE = (214, 235, 228)
PAPER = (247, 243, 226)
TEXT = (240, 245, 238)

GRID = 5
WINDOW_SIZE = 6
TILE_WIDTH = 36
TILE_HEIGHT = 58
BOARD_WIDTH = 130
BOARD_HEIGHT = 86
TILE_X_OFFSETS = (5, 47, 89)
TILE_Y_OFFSET = 23
WINDOW_X0 = 4
WINDOW_Y0 = 8
WINDOW_DX = 6
WINDOW_DY = 8
QUARANTINE_MARGIN = 2.0 / float(GRID * GRID)


def _font(size: int):
    for name in (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
        "/usr/share/fonts/truetype/liberation2/LiberationSans-Regular.ttf",
    ):
        if Path(name).is_file():
            return ImageFont.truetype(name, size=size * SCALE)
    return ImageFont.load_default()


def _pair(value) -> tuple[tuple[int, int], ...]:
    if not isinstance(value, (list, tuple)) or len(value) != 2:
        raise ValueError("a tile must contain exactly two windows")
    points = []
    for point in value:
        if not isinstance(point, (list, tuple)) or len(point) != 2:
            raise ValueError("window coordinates must be pairs")
        x, y = int(point[0]), int(point[1])
        if not (0 <= x < GRID and 0 <= y < GRID):
            raise ValueError("window coordinate is outside the five-by-five field")
        points.append((x, y))
    if len(set(points)) != 2 or points[0][1] == points[1][1]:
        raise ValueError("windows must be distinct and occupy different rows")
    return tuple(sorted(points))


def _reflect(pair: tuple[tuple[int, int], ...] | list[list[int]]) -> tuple[tuple[int, int], ...]:
    return tuple(sorted((GRID - 1 - int(x), int(y)) for x, y in pair))


def _mismatch(left, right) -> int:
    left_set = set(_pair(left))
    right_set = set(_pair(right))
    return len(left_set ^ right_set)


def _make_pair_pool() -> tuple[tuple[tuple[int, int], ...], ...]:
    pool = []
    for y0 in range(GRID):
        for y1 in range(y0 + 1, GRID):
            for x0 in range(GRID):
                for x1 in range(GRID):
                    pair = ((x0, y0), (x1, y1))
                    if _reflect(pair) != pair:
                        pool.append(pair)
    return tuple(pool)


_PAIR_POOL = _make_pair_pool()


def _panel_map(scene: dict) -> dict[str, dict]:
    panels: dict[str, dict] = {}
    for record in scene["candidate_panels"]:
        if not isinstance(record, dict):
            raise ValueError("candidate panel records must be objects")
        slot = str(record["slot"])
        if slot not in {"A", "B"} or slot in panels:
            raise ValueError("candidate slots must be unique A and B")
        panels[slot] = record
    if set(panels) != {"A", "B"}:
        raise ValueError("candidate slots must be A and B")
    return panels


def _validate_scene(scene: dict) -> tuple[tuple[tuple[int, int], ...], dict[str, dict]]:
    reference = _pair(scene["reference_windows"])
    panels = _panel_map(scene)
    for record in panels.values():
        _pair(record["windows"])
    anchor = scene["workbench_anchor"]
    if not isinstance(anchor, (list, tuple)) or len(anchor) != 2:
        raise ValueError("workbench anchor must be an x,y pair")
    if not (0 <= int(anchor[0]) <= WIDTH - BOARD_WIDTH and 0 <= int(anchor[1]) <= HEIGHT - BOARD_HEIGHT):
        raise ValueError("workbench is outside the canvas")
    return reference, panels


def sample_scene(seed: int) -> dict:
    """Sample equal-body tiles with one exact reflected pair and one mismatch."""

    rng = random.Random(int(seed))
    reference = rng.choice(_PAIR_POOL)
    reflected = _reflect(reference)
    wrong_pool = [
        pair
        for pair in _PAIR_POOL
        if _mismatch(_reflect(pair), reference) >= 2 and pair != reflected
    ]
    near = [pair for pair in wrong_pool if _mismatch(_reflect(pair), reference) <= 6]
    wrong = rng.choice(near if near and rng.randrange(3) == 0 else wrong_pool)
    matching_slot = rng.choice(("A", "B"))
    records = [
        {"slot": "A", "windows": [list(p) for p in (reflected if matching_slot == "A" else wrong)]},
        {"slot": "B", "windows": [list(p) for p in (reflected if matching_slot == "B" else wrong)]},
    ]
    rng.shuffle(records)
    anchor = [rng.randrange(170, 221), rng.randrange(195, 246)]
    return {
        "reference_windows": [list(p) for p in reference],
        "candidate_panels": records,
        "workbench_anchor": anchor,
    }


def _errors(reference, panels: dict[str, dict]) -> dict[str, int]:
    return {slot: _mismatch(_reflect(_pair(panels[slot]["windows"])), reference) for slot in ("A", "B")}


def analytic_gold(scene: dict) -> str:
    reference, panels = _validate_scene(scene)
    errors = _errors(reference, panels)
    if errors["A"] == errors["B"]:
        raise ValueError("reflected window arrangements are tied")
    return "A" if errors["A"] < errors["B"] else "B"


def margin(scene: dict) -> float:
    reference, panels = _validate_scene(scene)
    values = sorted(_errors(reference, panels).values())
    return float(values[1] - values[0]) / float(GRID * GRID)


def is_quarantined(scene: dict) -> bool:
    try:
        reference, panels = _validate_scene(scene)
        errors = _errors(reference, panels)
        return errors["A"] == errors["B"] or margin(scene) < QUARANTINE_MARGIN
    except (KeyError, TypeError, ValueError):
        return True


def _scaled_box(box: tuple[float, float, float, float]) -> tuple[int, int, int, int]:
    return tuple(int(round(value * SCALE)) for value in box)  # type: ignore[return-value]


def _draw_text(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font, fill=TEXT) -> None:
    draw.text((int(round(xy[0] * SCALE)), int(round(xy[1] * SCALE))), text, font=font, fill=fill)


def _draw_tile(draw: ImageDraw.ImageDraw, x: float, y: float, windows, label: str, label_font) -> None:
    draw.rectangle(_scaled_box((x, y, x + TILE_WIDTH - 1, y + TILE_HEIGHT - 1)), fill=TILE)
    _draw_text(draw, (x + 14, y - 16), label, label_font)
    for gx, gy in _pair(windows):
        left = x + WINDOW_X0 + WINDOW_DX * gx
        top = y + WINDOW_Y0 + WINDOW_DY * gy
        draw.rectangle(
            _scaled_box((left, top, left + WINDOW_SIZE - 1, top + WINDOW_SIZE - 1)),
            fill=PAPER,
        )


def render(scene: dict) -> Image.Image:
    """Render one antialiased compact workbench with R, A, and B tiles."""

    reference, panels = _validate_scene(scene)
    ax, ay = (float(v) for v in scene["workbench_anchor"])
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    board_box = (ax, ay, ax + BOARD_WIDTH, ay + BOARD_HEIGHT)
    draw.rounded_rectangle(_scaled_box(board_box), radius=8 * SCALE, fill=BOARD, outline=BOARD_EDGE, width=2 * SCALE)
    label_font = _font(8)
    for index, slot in enumerate(("R", "A", "B")):
        tx = ax + TILE_X_OFFSETS[index]
        ty = ay + TILE_Y_OFFSET
        windows = reference if slot == "R" else panels[slot]["windows"]
        _draw_tile(draw, tx, ty, windows, slot, label_font)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse slot-addressed records without changing pixels or analytic gold."""

    twin = copy.deepcopy(scene)
    twin["candidate_panels"] = list(reversed(twin["candidate_panels"]))
    return [("candidate_record_order", twin)]
