"""Independent pixel-only inverse for the compact window-pair workbench.

This module intentionally duplicates only raster palette constants and sampling
geometry. It never imports the renderer, prompt, generator, verifier, or gold.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_BOARD = np.asarray((39, 78, 92), dtype=np.float32)
_TILE = np.asarray((214, 235, 228), dtype=np.float32)
_PAPER = np.asarray((247, 243, 226), dtype=np.float32)
_GRID = 5
_WINDOW_X0 = 4
_WINDOW_Y0 = 8
_WINDOW_DX = 6
_WINDOW_DY = 8
_WINDOW_SIZE = 6
_COLOR_TOL = 25.0
_MIN_BOARD_AREA = 3000
_MIN_TILE_AREA = 700
_PAPER_TOL = 24.0


def _mask(array: np.ndarray, color: np.ndarray, tolerance: float = _COLOR_TOL) -> np.ndarray:
    return np.linalg.norm(array - color, axis=2) < tolerance


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue: deque[tuple[int, int]] = deque([(y, x)])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = cy + dy, cx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            component = np.zeros_like(mask, dtype=bool)
            ys, xs = zip(*points, strict=True)
            component[np.asarray(ys), np.asarray(xs)] = True
            found.append(component)
    return found


def _bbox(component: np.ndarray) -> tuple[int, int, int, int]:
    ys, xs = np.nonzero(component)
    if len(ys) == 0:
        raise ValueError("empty component")
    return int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())


def _find_board(array: np.ndarray) -> None:
    components = _components(_mask(array, _BOARD), _MIN_BOARD_AREA)
    if len(components) != 1:
        raise ValueError("image must contain exactly one connected workbench")
    x0, y0, x1, y1 = _bbox(components[0])
    if not (120 <= x1 - x0 + 1 <= 140 and 78 <= y1 - y0 + 1 <= 96):
        raise ValueError("workbench has unexpected dimensions")


def _find_tiles(array: np.ndarray) -> list[np.ndarray]:
    components = _components(_mask(array, _TILE), _MIN_TILE_AREA)
    if len(components) != 3:
        raise ValueError("image must contain exactly three intact tiles")
    components.sort(key=lambda component: _bbox(component)[0])
    for component in components:
        x0, y0, x1, y1 = _bbox(component)
        if not (31 <= x1 - x0 + 1 <= 41 and 52 <= y1 - y0 + 1 <= 64):
            raise ValueError("tile has unexpected dimensions")
    return components


def _read_pair(array: np.ndarray, component: np.ndarray) -> tuple[tuple[int, int], ...]:
    x0, y0, _, _ = _bbox(component)
    points = []
    for gy in range(_GRID):
        for gx in range(_GRID):
            x = min(array.shape[1] - 1, x0 + _WINDOW_X0 + _WINDOW_DX * gx + _WINDOW_SIZE // 2)
            y = min(array.shape[0] - 1, y0 + _WINDOW_Y0 + _WINDOW_DY * gy + _WINDOW_SIZE // 2)
            if np.linalg.norm(array[y, x] - _PAPER) < _PAPER_TOL:
                points.append((gx, gy))
    if len(points) != 2 or points[0][1] == points[1][1]:
        raise ValueError("each tile must expose exactly two windows in distinct rows")
    return tuple(sorted(points))


def _reflect(pair: tuple[tuple[int, int], ...]) -> tuple[tuple[int, int], ...]:
    return tuple(sorted((_GRID - 1 - x, y) for x, y in pair))


def _mismatch(left, right) -> int:
    return len(set(left) ^ set(right))


def decision_from_image(image: Image.Image) -> str:
    """Return A or B from raster evidence, raising on incomplete or tied evidence."""

    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    _find_board(array)
    tiles = _find_tiles(array)
    reference = _read_pair(array, tiles[0])
    errors = {
        "A": _mismatch(_reflect(_read_pair(array, tiles[1])), reference),
        "B": _mismatch(_reflect(_read_pair(array, tiles[2])), reference),
    }
    if errors["A"] == errors["B"]:
        raise ValueError("candidate window arrangements are tied")
    return "A" if errors["A"] < errors["B"] else "B"
