"""Renderer and analytic arm for endtoend_reach_duel.

One pale instrument sheet carries exactly three solid pure-black horizontal
bars clustered inside one small neighbourhood whose absolute position is
randomised, plus grey and coloured furniture that is never pure black and is
always held well away from the cluster.  The decision binds the three black
bars together: it asks whether the single longest black bar covers more length
than the other two black bars laid end to end.  Every one of the three
disconnected bars contributes to the answer and no remote mark does.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "endtoend_reach_duel-renderer-0.1.0"

WIDTH = 720
HEIGHT = 480
PAPER = (250, 249, 245)
BLACK = (0, 0, 0)
FRAME_GREY = (206, 204, 198)
TICK_GREY = (150, 148, 142)

CLUSTER_W = 170
CLUSTER_H = 116
ROW_GAP = 34
MIN_LEN = 26
MAX_LEN = 150
THICKNESSES = (5, 6, 7, 8, 9)

#: minimum longest-versus-runner-up length gap, in pixels
LEAD_FLOOR = 9.0
#: no scene is emitted below this margin
MARGIN_FLOOR = 5.0
#: scenes at or below this margin are quarantined
QUARANTINE_MARGIN = 7.0
#: keep every non-black mark this far from the black cluster box
DECOY_CLEARANCE = 24

DECOY_COLORS = (
    (203, 86, 74),
    (78, 126, 196),
    (98, 162, 96),
    (214, 158, 66),
    (150, 106, 190),
    (176, 174, 168),
    (132, 130, 126),
    (86, 174, 186),
)


def _lengths(scene: dict) -> list[int]:
    return sorted((int(bar["length"]) for bar in scene["bars"]), reverse=True)


def analytic_gold(scene: dict) -> str:
    """`single` when the longest black bar outreaches the other two combined."""
    longest, mid, short = _lengths(scene)
    return "single" if longest > mid + short else "pair"


def margin(scene: dict) -> float:
    """Smaller of the reach surplus and the longest-versus-runner-up gap, px."""
    longest, mid, short = _lengths(scene)
    return float(min(abs(longest - (mid + short)), longest - mid))


def lead_gap(scene: dict) -> float:
    """Runner-up decision margin for the identity of the longest black bar."""
    longest, mid, _ = _lengths(scene)
    return float(longest - mid)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list:
    """No latent aliases are declared; every sampled field is drawn."""
    return []


def _split_pair(rng: random.Random, total: int, cap: int) -> tuple[int, int] | None:
    """Split `total` into two bar lengths, the larger at most `cap`."""
    low = max((total + 1) // 2, total - MAX_LEN, MIN_LEN)
    high = min(cap, total - MIN_LEN, MAX_LEN)
    if high < low:
        return None
    big = rng.randint(low, high)
    return big, total - big


def sample_scene(seed: int) -> dict:
    """Sample one legible scene; raises RuntimeError when sampling fails."""
    rng = random.Random(seed * 7919 + 13)
    target = "single" if rng.random() < 0.5 else "pair"

    bars_len: list[int] | None = None
    for _ in range(400):
        # Counterbalance: the summed black length is drawn from one common
        # range first, so total ink cannot stand in for the answer.
        grand_total = rng.randint(150, 258)
        surplus = (
            rng.randint(int(MARGIN_FLOOR), 8)
            if rng.random() < 0.28
            else rng.randint(9, 40)
        )
        if target == "single":
            longest = (grand_total + surplus) // 2
        else:
            longest = (grand_total - surplus) // 2
        total = grand_total - longest
        if longest > MAX_LEN or longest < 70 or total < 2 * MIN_LEN:
            continue
        if abs(longest - total) < MARGIN_FLOOR:
            continue
        if (longest > total) != (target == "single"):
            continue
        split = _split_pair(rng, total, int(longest - LEAD_FLOOR))
        if split is None:
            continue
        mid, short = split
        if short < MIN_LEN or mid > longest - LEAD_FLOOR:
            continue
        bars_len = [longest, mid, short]
        break
    if bars_len is None:
        raise RuntimeError("length sampling failed")

    order = [0, 1, 2]
    rng.shuffle(order)
    rows = [bars_len[index] for index in order]

    cluster_x = rng.randint(30, WIDTH - CLUSTER_W - 30)
    cluster_y = rng.randint(48, HEIGHT - CLUSTER_H - 30)

    bars = []
    for index, length in enumerate(rows):
        thickness = THICKNESSES[rng.randrange(len(THICKNESSES))]
        bars.append(
            {
                "x": cluster_x + rng.randint(0, CLUSTER_W - length),
                "y": cluster_y + 6 + index * ROW_GAP,
                "length": int(length),
                "thickness": int(thickness),
            }
        )

    cluster = {"x": cluster_x, "y": cluster_y, "w": CLUSTER_W, "h": CLUSTER_H}
    decoys = _sample_decoys(rng, cluster)
    ticks = [
        {"x": 40 + 24 * step, "h": 7 + (3 if step % 4 == 0 else 0)}
        for step in range(26)
    ]
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "bars": bars,
        "cluster": cluster,
        "decoys": decoys,
        "ticks": ticks,
    }


def _clear_of_cluster(box: tuple[int, int, int, int], cluster: dict) -> bool:
    x0, y0, x1, y1 = box
    cx0 = cluster["x"] - DECOY_CLEARANCE
    cy0 = cluster["y"] - DECOY_CLEARANCE
    cx1 = cluster["x"] + cluster["w"] + DECOY_CLEARANCE
    cy1 = cluster["y"] + cluster["h"] + DECOY_CLEARANCE
    return x1 < cx0 or x0 > cx1 or y1 < cy0 or y0 > cy1


def _sample_decoys(rng: random.Random, cluster: dict) -> list[dict]:
    decoys: list[dict] = []
    attempts = 0
    wanted = rng.randint(7, 10)
    while len(decoys) < wanted and attempts < 800:
        attempts += 1
        kind = "bar" if rng.random() < 0.62 else "blob"
        if kind == "bar":
            length = rng.randint(40, 190)
            thickness = rng.randint(5, 11)
        else:
            length = rng.randint(16, 40)
            thickness = length
        x = rng.randint(24, WIDTH - length - 24)
        y = rng.randint(44, HEIGHT - thickness - 24)
        box = (x, y, x + length - 1, y + thickness - 1)
        if not _clear_of_cluster(box, cluster):
            continue
        decoys.append(
            {
                "kind": kind,
                "x": x,
                "y": y,
                "length": int(length),
                "thickness": int(thickness),
                "color": list(DECOY_COLORS[rng.randrange(len(DECOY_COLORS))]),
            }
        )
    return decoys


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), PAPER)
    draw = ImageDraw.Draw(image)

    draw.rectangle(
        [14, 14, int(scene["width"]) - 15, int(scene["height"]) - 15],
        outline=FRAME_GREY,
        width=2,
    )
    for tick in scene["ticks"]:
        x = int(tick["x"])
        draw.rectangle([x, 22, x + 1, 22 + int(tick["h"])], fill=TICK_GREY)

    for decoy in scene["decoys"]:
        x, y = int(decoy["x"]), int(decoy["y"])
        length, thickness = int(decoy["length"]), int(decoy["thickness"])
        color = tuple(int(channel) for channel in decoy["color"])
        box = [x, y, x + length - 1, y + thickness - 1]
        if decoy["kind"] == "bar":
            draw.rectangle(box, fill=color)
        else:
            draw.ellipse(box, fill=color)

    for bar in scene["bars"]:
        x, y = int(bar["x"]), int(bar["y"])
        draw.rectangle(
            [x, y, x + int(bar["length"]) - 1, y + int(bar["thickness"]) - 1],
            fill=BLACK,
        )
    return image
