"""Pixel-only oracle for endtoend_reach_duel.

Receives the final PNG and nothing else: no latent scene, no renderer code, no
analytic gold, no metadata.  It segments the pure-black ink, recovers the three
disconnected bar components, measures each bar's own left-to-right extent, and
compares the longest extent with the sum of the other two.
"""

from __future__ import annotations

from collections import deque

from PIL import Image

ORACLE_VERSION = "endtoend_reach_duel-oracle-0.1.0"

INK_MAX_CHANNEL = 60
MIN_COMPONENT_AREA = 100
MIN_BAR_LENGTH = 20
MAX_BAR_THICKNESS = 16
MIN_LEAD_GAP = 3


def _ink_mask(image: "Image.Image") -> tuple[list[bool], int, int]:
    rgb = image.convert("RGB")
    width, height = rgb.size
    pixels = rgb.load()
    mask = [False] * (width * height)
    for y in range(height):
        row = y * width
        for x in range(width):
            r, g, b = pixels[x, y]
            if r < INK_MAX_CHANNEL and g < INK_MAX_CHANNEL and b < INK_MAX_CHANNEL:
                mask[row + x] = True
    return mask, width, height


def _components(mask: list[bool], width: int, height: int) -> list[dict]:
    seen = [False] * (width * height)
    found: list[dict] = []
    for start in range(width * height):
        if not mask[start] or seen[start]:
            continue
        seen[start] = True
        queue = deque([start])
        area = 0
        x0 = x1 = start % width
        y0 = y1 = start // width
        while queue:
            index = queue.popleft()
            area += 1
            x, y = index % width, index // width
            x0, x1 = min(x0, x), max(x1, x)
            y0, y1 = min(y0, y), max(y1, y)
            if x > 0 and mask[index - 1] and not seen[index - 1]:
                seen[index - 1] = True
                queue.append(index - 1)
            if x + 1 < width and mask[index + 1] and not seen[index + 1]:
                seen[index + 1] = True
                queue.append(index + 1)
            if y > 0 and mask[index - width] and not seen[index - width]:
                seen[index - width] = True
                queue.append(index - width)
            if y + 1 < height and mask[index + width] and not seen[index + width]:
                seen[index + width] = True
                queue.append(index + width)
        found.append(
            {
                "area": area,
                "length": x1 - x0 + 1,
                "thickness": y1 - y0 + 1,
                "cx": (x0 + x1) / 2.0,
                "cy": (y0 + y1) / 2.0,
            }
        )
    return found


def decision_from_image(image) -> str:
    """Return `single` or `pair`; raise ValueError when the image is unreadable."""
    mask, width, height = _ink_mask(image)
    bars = [
        component
        for component in _components(mask, width, height)
        if component["area"] >= MIN_COMPONENT_AREA
        and component["length"] >= MIN_BAR_LENGTH
        and component["thickness"] <= MAX_BAR_THICKNESS
    ]
    if len(bars) != 3:
        raise ValueError(f"expected three black bars, segmented {len(bars)}")
    lengths = sorted((bar["length"] for bar in bars), reverse=True)
    if lengths[0] - lengths[1] < MIN_LEAD_GAP:
        raise ValueError("the longest black bar is not resolvable")
    return "single" if lengths[0] > lengths[1] + lengths[2] else "pair"
