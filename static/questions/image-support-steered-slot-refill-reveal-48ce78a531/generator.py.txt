"""Renderer for the slot-refill reveal world.

One bordered panel carries a seven-column bar skyline drawn on a 7x7 lattice,
except that three of its columns are drawn as visibly empty slots filled with a
third, intermediate tone.  The three missing columns are drawn elsewhere on the
canvas as three separate one-column slats, widely separated from the panel and
from each other.  The declared operation is to put the loose slats back into the
empty slots, taking the slats in top-to-bottom order of their positions and the
slots in left-to-right order, and then to name the completed skyline.

The four declared skylines are built from one fixed multiset of bar heights,
{1,2,3,4,5,6,7}: the middle column is always the height-4 bar, the left triple
always carries the heights {1,2,3} and the right triple always carries the
heights {5,6,7}.  A symbol is fixed by two independent bits - whether the left
triple runs upward or downward, and whether the right triple runs upward or
downward - so the visible triple fixes one bit, the loose slats fix the other,
and neither side alone can name the symbol.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RENDERER_VERSION = "slot_refill_reveal-renderer-0.1.0"

CELLS = 7
QUARANTINE_CLEARANCE = 34.0
MIN_GLYPH_SEPARATION = 4  # lattice cells between the two closest declared symbols

# heights of the seven bars, left to right, in lattice cells
GLYPH_HEIGHTS: dict[str, tuple[int, ...]] = {
    "ramp": (1, 2, 3, 4, 5, 6, 7),
    "crest": (1, 2, 3, 4, 7, 6, 5),
    "basin": (3, 2, 1, 4, 5, 6, 7),
    "switchback": (3, 2, 1, 4, 7, 6, 5),
}
GLYPHS = tuple(sorted(GLYPH_HEIGHTS))
SLOT_SIDES = {"left": (0, 1, 2), "right": (4, 5, 6)}

_LUMA = (0.299, 0.587, 0.114)


def _luma(rgb) -> float:
    return float(rgb[0] * _LUMA[0] + rgb[1] * _LUMA[1] + rgb[2] * _LUMA[2])


def _tinted(rng: np.random.Generator, gray: float) -> list[int]:
    """A colour whose luma is (up to rounding) the requested gray level."""

    dr, dg = float(rng.integers(-9, 10)), float(rng.integers(-9, 10))
    db = -(dr * _LUMA[0] + dg * _LUMA[1]) / _LUMA[2]
    out = []
    for delta in (dr, dg, db):
        out.append(int(round(min(252.0, max(3.0, gray + delta)))))
    return out


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(int(seed) + 0x51075)

    cell_px = int(rng.integers(20, 25))
    panel_frame = int(rng.integers(4, 7))
    slat_frame = int(rng.integers(3, 6))

    panel_side = CELLS * cell_px + 2 * panel_frame
    slat_w = cell_px + 2 * slat_frame
    slat_h = CELLS * cell_px + 2 * slat_frame

    canvas_w = 760
    canvas_h = 620

    panel_left = int(rng.integers(18, 46))
    panel_top = int(rng.integers(14, canvas_h - panel_side - 14))

    # three slats, one per widely separated horizontal band on the far side
    band_lo = panel_left + panel_side + 96
    band_hi = canvas_w - slat_w - 16
    slat_span = (canvas_h - 24) // 3
    slats = []
    for index in range(3):
        top = 12 + index * slat_span + int(rng.integers(0, max(1, slat_span - slat_h)))
        left = int(rng.integers(band_lo, band_hi + 1))
        slats.append([left, top])

    ink_l = float(rng.integers(16, 54))
    paper_l = float(rng.integers(212, 249))
    frac = float(rng.uniform(0.42, 0.58))
    slot_l = ink_l + frac * (paper_l - ink_l)
    tones = {
        "ink": _tinted(rng, ink_l),
        "paper": _tinted(rng, paper_l),
        "slot": _tinted(rng, slot_l),
    }

    glyph = GLYPHS[int(rng.integers(0, len(GLYPHS)))]
    slot_side = "left" if bool(rng.integers(0, 2)) else "right"

    boxes = [(panel_left, panel_top, panel_side, panel_side)]
    boxes += [(x, y, slat_w, slat_h) for x, y in slats]
    chaff = []
    attempts = 0
    target = int(rng.integers(5, 12))
    while len(chaff) < target and attempts < 400:
        attempts += 1
        size = int(rng.integers(3, 6))
        cx = int(rng.integers(6, canvas_w - size - 6))
        cy = int(rng.integers(6, canvas_h - size - 6))
        clear = True
        for bx, by, bw, bh in boxes:
            if cx + size + 9 > bx and cx < bx + bw + 9 and cy + size + 9 > by and cy < by + bh + 9:
                clear = False
                break
        if clear:
            chaff.append([cx, cy, size])

    return {
        "canvas": [canvas_w, canvas_h],
        "cells": CELLS,
        "cell_px": cell_px,
        "panel_frame": panel_frame,
        "slat_frame": slat_frame,
        "panel_left": panel_left,
        "panel_top": panel_top,
        "slats": slats,
        "glyph": glyph,
        "slot_side": slot_side,
        "tones": tones,
        "chaff": chaff,
    }


def _slot_columns(scene: dict) -> tuple[int, ...]:
    return SLOT_SIDES[scene["slot_side"]]


def _bar_column(canvas: np.ndarray, x0: int, y0: int, cell: int, height: int, ink, paper) -> None:
    for row in range(CELLS):
        top = y0 + row * cell
        tone = ink if row >= CELLS - height else paper
        canvas[top : top + cell, x0 : x0 + cell] = tone


def render(scene: dict) -> Image.Image:
    width, height = scene["canvas"]
    cell = scene["cell_px"]
    ink = np.array(scene["tones"]["ink"], dtype=np.uint8)
    paper = np.array(scene["tones"]["paper"], dtype=np.uint8)
    slot = np.array(scene["tones"]["slot"], dtype=np.uint8)

    canvas = np.zeros((height, width, 3), dtype=np.uint8)
    canvas[:, :] = paper

    heights = GLYPH_HEIGHTS[scene["glyph"]]
    slots = _slot_columns(scene)

    # the panel: solid ink frame, seven lattice columns, three of them empty slots
    pf = scene["panel_frame"]
    side = CELLS * cell + 2 * pf
    px, py = scene["panel_left"], scene["panel_top"]
    canvas[py : py + side, px : px + side] = ink
    ix, iy = px + pf, py + pf
    for column in range(CELLS):
        x0 = ix + column * cell
        if column in slots:
            canvas[iy : iy + CELLS * cell, x0 : x0 + cell] = slot
        else:
            _bar_column(canvas, x0, iy, cell, heights[column], ink, paper)

    # the loose slats, in top-to-bottom order of the slot columns they belong to
    sf = scene["slat_frame"]
    slat_w, slat_h = cell + 2 * sf, CELLS * cell + 2 * sf
    ordered = sorted(scene["slats"], key=lambda pair: pair[1])
    for column, (sx, sy) in zip(slots, ordered):
        canvas[sy : sy + slat_h, sx : sx + slat_w] = ink
        _bar_column(canvas, sx + sf, sy + sf, cell, heights[column], ink, paper)

    for cx, cy, size in scene["chaff"]:
        canvas[cy : cy + size, cx : cx + size] = ink

    return Image.fromarray(canvas, mode="RGB")


def analytic_gold(scene: dict) -> str:
    return str(scene["glyph"])


def margin(scene: dict) -> float:
    """Half of the smallest rendered luma gap among the three lattice tones."""

    lumas = sorted(_luma(scene["tones"][key]) for key in ("ink", "slot", "paper"))
    gaps = [lumas[1] - lumas[0], lumas[2] - lumas[1]]
    return round(min(gaps) / 2.0, 6)


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_CLEARANCE)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases are declared: every scene field is rendered."""

    return []


def runner_up_cells(scene: dict) -> int:
    """Lattice cells separating the true skyline from the closest other symbol."""

    truth = GLYPH_HEIGHTS[scene["glyph"]]
    best = None
    for name, heights in GLYPH_HEIGHTS.items():
        if name == scene["glyph"]:
            continue
        distance = sum(abs(a - b) for a, b in zip(truth, heights))
        best = distance if best is None else min(best, distance)
    return int(best or 0)
