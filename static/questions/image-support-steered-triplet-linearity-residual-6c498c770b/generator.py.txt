"""Deterministic latent scene and raster renderer for paired linearity residuals."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 512
HEIGHT = 512
SCALE = 3
BACKGROUND = (250, 248, 242)
INK = (50, 56, 64)
FRAME = (184, 188, 190)
GRID = (216, 216, 210)
TEAL = (10, 145, 164)
ORANGE = (226, 112, 34)
X_CENTERS = (160, 288, 416)
TOP_CENTER = 160
BOTTOM_CENTER = 352
QUARANTINE_PX = 1.25


def _residual(values: list[int]) -> float:
    """Absolute middle residual from the outer-point linear interpolation."""
    return abs(2 * values[1] - values[0] - values[2]) / 2.0


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; seed parity fixes the requested answer class."""
    rng = random.Random(int(seed))
    wanted = "TOP" if int(seed) % 2 == 0 else "BOTTOM"
    near_boundary = int(seed) % 11 == 0
    for _ in range(20000):
        top = [rng.randint(-6, 6) for _ in range(3)]
        bottom = [rng.randint(-6, 6) for _ in range(3)]
        top_residual = _residual(top)
        bottom_residual = _residual(bottom)
        if top_residual == bottom_residual:
            continue
        decision = "TOP" if top_residual < bottom_residual else "BOTTOM"
        runner_up_gap = abs(top_residual - bottom_residual)
        if decision != wanted:
            continue
        if near_boundary and runner_up_gap <= 1.0:
            break
        if not near_boundary and runner_up_gap >= 3.0:
            break
    else:
        raise RuntimeError("could not sample a separated residual comparison")
    return {
        "top_values": top,
        "bottom_values": bottom,
        "palette_swap": bool(rng.randrange(2)),
        "dx": rng.randint(-1, 1),
        "dy": rng.randint(-1, 1),
        "radius": rng.choice([7, 8]),
    }


def render(scene: dict) -> Image.Image:
    """Render only public visual evidence; no answer-specific decoration is used."""
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)

    def box(coords: tuple[int, int, int, int], *, fill=None, outline=None, width=1) -> None:
        draw.rectangle(tuple(v * SCALE for v in coords), fill=fill, outline=outline, width=width * SCALE)

    dx = int(scene["dx"])
    dy = int(scene["dy"])
    radius = int(scene["radius"])
    top_color, bottom_color = (ORANGE, TEAL) if bool(scene["palette_swap"]) else (TEAL, ORANGE)

    # Two deliberately local plotting strips. Gray structure makes grouping and
    # equal x-spacing explicit but is ignored by the independent color oracle.
    for base, label in ((TOP_CENTER + dy, "TOP"), (BOTTOM_CENTER + dy, "BOTTOM")):
        box((118 + dx, base - 45, 454 + dx, base + 45), outline=FRAME, width=2)
        draw.line(
            ((132 + dx) * SCALE, base * SCALE, (440 + dx) * SCALE, base * SCALE),
            fill=GRID,
            width=SCALE,
        )
        for x in X_CENTERS:
            xx = x + dx
            draw.line(
                (xx * SCALE, (base - 35) * SCALE, xx * SCALE, (base + 35) * SCALE),
                fill=GRID,
                width=SCALE,
            )
        try:
            font = ImageFont.load_default(size=13 * SCALE)
        except TypeError:
            font = ImageFont.load_default()
        draw.text(((68 + dx) * SCALE, (base - 9) * SCALE), label, fill=INK, font=font)

    for values, base, color in (
        (scene["top_values"], TOP_CENTER + dy, top_color),
        (scene["bottom_values"], BOTTOM_CENTER + dy, bottom_color),
    ):
        for x, value in zip(X_CENTERS, values, strict=True):
            cx = x + dx
            cy = base - int(value)
            draw.ellipse(
                (
                    (cx - radius) * SCALE,
                    (cy - radius) * SCALE,
                    (cx + radius) * SCALE,
                    (cy + radius) * SCALE,
                ),
                fill=color,
                outline=INK,
                width=SCALE,
            )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    top_residual = _residual(scene["top_values"])
    bottom_residual = _residual(scene["bottom_values"])
    if top_residual == bottom_residual:
        raise ValueError("tied residuals have no constrained answer")
    return "TOP" if top_residual < bottom_residual else "BOTTOM"


def margin(scene: dict) -> float:
    return abs(_residual(scene["top_values"]) - _residual(scene["bottom_values"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
