"""Independent PNG-only inverse arm for paired three-point linearity residuals."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

# Public marker palette is repeated here deliberately; no renderer state is read.
_MARKER_COLORS = np.asarray(((10, 145, 164), (226, 112, 34)), dtype=np.int32)
_COLOR_RADIUS_SQ = 58 * 58
_MIN_AREA = 105
_MAX_AREA = 260


def _components(mask: np.ndarray) -> list[tuple[int, float, float, int, int]]:
    """Return (area, cx, cy, width, height) for 8-connected colored components."""
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[tuple[int, float, float, int, int]] = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for yy in range(max(0, y - 1), min(height, y + 2)):
                for xx in range(max(0, x - 1), min(width, x + 2)):
                    if mask[yy, xx] and not seen[yy, xx]:
                        seen[yy, xx] = True
                        queue.append((yy, xx))
        area = len(xs)
        if area >= 8:  # discard isolated LANCZOS ringing only
            output.append(
                (area, float(np.mean(xs)), float(np.mean(ys)), max(xs) - min(xs) + 1, max(ys) - min(ys) + 1)
            )
    return output


def _row_residual(points: list[tuple[int, float, float, int, int]]) -> float:
    ordered = sorted(points, key=lambda item: item[1])
    xs = [item[1] for item in ordered]
    ys = [item[2] for item in ordered]
    if min(np.diff(xs)) < 80 or max(np.diff(xs)) / min(np.diff(xs)) > 1.15:
        raise ValueError("three marker settings are not visibly equally spaced")
    return abs(2.0 * ys[1] - ys[0] - ys[2]) / 2.0


def decision_from_image(image: Image.Image) -> str:
    """Segment marker cores, recover six centroids, and compare row residuals."""
    array = np.asarray(image.convert("RGB"), dtype=np.int32)
    distances = ((array[:, :, None, :] - _MARKER_COLORS[None, None, :, :]) ** 2).sum(axis=3)
    mask = distances.min(axis=2) <= _COLOR_RADIUS_SQ
    components = _components(mask)
    markers = [item for item in components if _MIN_AREA <= item[0] <= _MAX_AREA]
    if len(markers) != 6:
        raise ValueError(f"expected six intact measurement dots, found {len(markers)}")
    if any(not 11 <= item[3] <= 19 or not 11 <= item[4] <= 19 for item in markers):
        raise ValueError("a colored component is not a compact measurement dot")
    ordered_y = sorted(markers, key=lambda item: item[2])
    top, bottom = ordered_y[:3], ordered_y[3:]
    if min(item[2] for item in bottom) - max(item[2] for item in top) < 100:
        raise ValueError("the two measurement rows are not separable")
    if max(item[2] for item in top) - min(item[2] for item in top) > 24:
        raise ValueError("top row exceeds its local plotting strip")
    if max(item[2] for item in bottom) - min(item[2] for item in bottom) > 24:
        raise ValueError("bottom row exceeds its local plotting strip")
    top_residual = _row_residual(top)
    bottom_residual = _row_residual(bottom)
    if abs(top_residual - bottom_residual) < 0.20:
        raise ValueError("pixel residual comparison is unresolved")
    return "TOP" if top_residual < bottom_residual else "BOTTOM"
