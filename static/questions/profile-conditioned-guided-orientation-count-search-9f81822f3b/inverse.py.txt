"""Independent pixel-only inverse for reference-relative bar counts.

This module deliberately duplicates only the public raster geometry needed to
interpret the final PNG. It receives no latent scene and imports no candidate
renderer, generator, prompt, or gold code.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

MATCH_TOLERANCE = 8.0
PANEL_DIVIDER_X = 480.0


def _blue_mask(image: Image.Image) -> np.ndarray:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    red = pixels[:, :, 0]
    green = pixels[:, :, 1]
    blue = pixels[:, :, 2]
    # The bars are the only blue-chromatic marks; gray borders and labels are
    # intentionally excluded without relying on their text or draw order.
    return ((blue - red) > 20) & ((green - red) > 10)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    visited = np.zeros(mask.shape, dtype=bool)
    components: list[np.ndarray] = []
    ys, xs = np.nonzero(mask)
    for start_y, start_x in zip(ys.tolist(), xs.tolist()):
        if visited[start_y, start_x]:
            continue
        stack = [(start_y, start_x)]
        visited[start_y, start_x] = True
        points: list[tuple[int, int]] = []
        while stack:
            y, x = stack.pop()
            points.append((y, x))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    next_y = y + dy
                    next_x = x + dx
                    if (
                        0 <= next_y < height
                        and 0 <= next_x < width
                        and mask[next_y, next_x]
                        and not visited[next_y, next_x]
                    ):
                        visited[next_y, next_x] = True
                        stack.append((next_y, next_x))
        if len(points) >= 35:
            components.append(np.asarray(points, dtype=np.int32))
    return components


def _orientation(points: np.ndarray, image: Image.Image) -> tuple[float, float, float]:
    pixels = np.asarray(image.convert("RGB"), dtype=np.float64)
    y = points[:, 0]
    x = points[:, 1]
    contrast = (pixels[y, x, 2] - pixels[y, x, 0]) + 0.5 * (
        pixels[y, x, 1] - pixels[y, x, 0]
    )
    weights = np.maximum(contrast, 1.0)
    xy = np.column_stack((x.astype(float), y.astype(float)))
    center = np.average(xy, axis=0, weights=weights)
    centered = xy - center
    covariance = (centered * weights[:, None]).T @ centered / weights.sum()
    values, vectors = np.linalg.eigh(covariance)
    direction = vectors[:, int(np.argmax(values))]
    angle = math.degrees(math.atan2(float(direction[1]), float(direction[0]))) % 180.0
    return angle, float(center[0]), float(center[1])


def _angle_gap(first: float, second: float) -> float:
    delta = abs((first - second) % 180.0)
    return min(delta, 180.0 - delta)


def decision_from_image(image: Image.Image) -> str:
    """Recover LEFT/RIGHT using only the supplied raster pixels."""

    mask = _blue_mask(image)
    components = _components(mask)
    reference_parts = []
    candidate_parts = []
    for points in components:
        y_min = int(points[:, 0].min())
        y_max = int(points[:, 0].max())
        if y_max < 140:
            reference_parts.append(points)
        elif y_min > 170:
            candidate_parts.append(points)
    if len(reference_parts) != 1:
        raise ValueError(f"expected one reference bar, found {len(reference_parts)}")
    if len(candidate_parts) != 16:
        raise ValueError(f"expected sixteen candidate bars, found {len(candidate_parts)}")
    reference_angle, _, _ = _orientation(reference_parts[0], image)
    counts = {"left": 0, "right": 0}
    for points in candidate_parts:
        angle, center_x, _ = _orientation(points, image)
        side = "left" if center_x < PANEL_DIVIDER_X else "right"
        if _angle_gap(angle, reference_angle) <= MATCH_TOLERANCE:
            counts[side] += 1
    if counts["left"] == counts["right"]:
        raise ValueError("pixel evidence has no unique count winner")
    return "left" if counts["left"] > counts["right"] else "right"
