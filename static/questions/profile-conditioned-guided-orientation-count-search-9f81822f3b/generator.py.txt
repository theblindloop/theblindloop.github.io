"""Analytic scene sampler and deterministic raster renderer for the candidate."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

CANVAS = (960, 640)
SCALE = 4
BACKGROUND = (248, 249, 246)
BAR_COLOR = (27, 76, 111)
PANEL_EDGE = (169, 178, 181)
TEXT_COLOR = (74, 83, 86)
MATCH_TOLERANCE = 8.0
BOUNDARY_CLEARANCE = 2.0
BAR_LENGTH = 66.0
BAR_WIDTH = 11.0
REFERENCE_LENGTH = 104.0
PANEL_DIVIDER_X = 480
BAR_SLOTS = (
    (88.0, 260.0),
    (188.0, 260.0),
    (288.0, 260.0),
    (388.0, 260.0),
    (88.0, 485.0),
    (188.0, 485.0),
    (288.0, 485.0),
    (388.0, 485.0),
)


def _angle_gap(first: float, second: float) -> float:
    """Return the acute gap between two undirected line orientations."""

    delta = abs((float(first) - float(second)) % 180.0)
    return min(delta, 180.0 - delta)


def _normal_angle(angle: float) -> float:
    return float(angle) % 180.0


def _distractor_angle(reference: float, rng: random.Random, *, boundary: bool = False) -> float:
    if boundary:
        offset = 9.5
    else:
        offset = rng.choice((27.0, 33.0, 43.0, 56.0, 67.0))
    if rng.randrange(2):
        offset = -offset
    return _normal_angle(reference + offset)


def _make_bars(
    reference: float,
    matching_count: int,
    rng: random.Random,
    *,
    x_offset: float = 0.0,
    near_boundary: bool = False,
) -> list[dict]:
    angles = [_normal_angle(reference + rng.uniform(-3.2, 3.2)) for _ in range(matching_count)]
    for index in range(8 - matching_count):
        angles.append(
            _distractor_angle(
                reference,
                rng,
                boundary=near_boundary and index == 0,
            )
        )
    rng.shuffle(angles)
    slots = list(BAR_SLOTS)
    rng.shuffle(slots)
    bars = []
    for (slot_x, slot_y), angle in zip(slots, angles):
        bars.append(
            {
                "x": round(slot_x + x_offset + rng.uniform(-9.0, 9.0), 3),
                "y": round(slot_y + rng.uniform(-9.0, 9.0), 3),
                "angle": round(angle, 3),
                "length": BAR_LENGTH,
                "width": BAR_WIDTH,
            }
        )
    return bars


def sample_scene(seed: int) -> dict:
    """Sample one balanced scene; even and odd seeds swap the answer side."""

    rng = random.Random(int(seed))
    reference = round(rng.uniform(5.0, 175.0), 3)
    high = rng.choice((4, 5, 6))
    gap = rng.choice((2, 3))
    low = high - gap
    answer_left = int(seed) % 2 == 0
    left_count, right_count = (high, low) if answer_left else (low, high)
    near_boundary = int(seed) % 5 == 1
    return {
        "reference_angle": reference,
        "left_bars": _make_bars(reference, left_count, rng, near_boundary=near_boundary),
        "right_bars": _make_bars(
            reference,
            right_count,
            rng,
            x_offset=480.0,
            near_boundary=near_boundary,
        ),
    }


def _bar_endpoints(bar: dict, scale: int = 1) -> tuple[tuple[int, int], tuple[int, int]]:
    angle = math.radians(float(bar["angle"]))
    half = float(bar["length"]) / 2.0
    dx = math.cos(angle) * half
    dy = math.sin(angle) * half
    center_x = float(bar["x"])
    center_y = float(bar["y"])
    return (
        (round((center_x - dx) * scale), round((center_y - dy) * scale)),
        (round((center_x + dx) * scale), round((center_y + dy) * scale)),
    )


def _draw_bar(draw: ImageDraw.ImageDraw, bar: dict, *, scale: int = SCALE) -> None:
    start, end = _bar_endpoints(bar, scale)
    width = max(1, round(float(bar["width"]) * scale))
    radius = width / 2.0
    draw.line([start, end], fill=BAR_COLOR, width=width)
    for point in (start, end):
        x, y = point
        draw.ellipse(
            [round(x - radius), round(y - radius), round(x + radius), round(y + radius)],
            fill=BAR_COLOR,
        )


def _draw_box(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], *, scale: int = SCALE) -> None:
    x0, y0, x1, y1 = box
    draw.rectangle(
        [x0 * scale, y0 * scale, x1 * scale, y1 * scale],
        outline=PANEL_EDGE,
        width=2 * scale,
    )


def render(scene: dict) -> Image.Image:
    """Render the public image from visible scene fields only."""

    image = Image.new("RGB", (CANVAS[0] * SCALE, CANVAS[1] * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_box(draw, (25, 20, 935, 130))
    _draw_box(draw, (25, 145, 935, 620))
    draw.line(
        [(PANEL_DIVIDER_X * SCALE, 145 * SCALE), (PANEL_DIVIDER_X * SCALE, 620 * SCALE)],
        fill=PANEL_EDGE,
        width=2 * SCALE,
    )
    _draw_bar(
        draw,
        {
            "x": 480.0,
            "y": 80.0,
            "angle": scene["reference_angle"],
            "length": REFERENCE_LENGTH,
            "width": BAR_WIDTH,
        },
    )
    for bar in scene["left_bars"]:
        _draw_bar(draw, bar)
    for bar in scene["right_bars"]:
        _draw_bar(draw, bar)
    image = image.resize(CANVAS, Image.Resampling.LANCZOS)
    label_draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    label_draw.text((44, 34), "REFERENCE DIRECTION", fill=TEXT_COLOR, font=font)
    label_draw.text((208, 158), "LEFT", fill=TEXT_COLOR, font=font)
    label_draw.text((688, 158), "RIGHT", fill=TEXT_COLOR, font=font)
    return image


def _matching_count(bars: Iterable[dict], reference: float) -> int:
    return sum(
        _angle_gap(float(bar["angle"]), reference) <= MATCH_TOLERANCE
        for bar in bars
    )


def analytic_gold(scene: dict) -> str:
    """Return the side with more reference-matching candidate bars."""

    reference = scene["reference_angle"]
    left = _matching_count(scene["left_bars"], reference)
    right = _matching_count(scene["right_bars"], reference)
    if left == right:
        raise ValueError("count tie is quarantined and has no unique answer")
    return "left" if left > right else "right"


def margin(scene: dict) -> float:
    reference = scene["reference_angle"]
    left = _matching_count(scene["left_bars"], reference)
    right = _matching_count(scene["right_bars"], reference)
    return float(abs(left - right))


def is_quarantined(scene: dict) -> bool:
    reference = scene["reference_angle"]
    counts = (
        _matching_count(scene["left_bars"], reference),
        _matching_count(scene["right_bars"], reference),
    )
    if abs(counts[0] - counts[1]) < 2:
        return True
    all_bars = [*scene["left_bars"], *scene["right_bars"]]
    return any(
        abs(_angle_gap(float(bar["angle"]), reference) - MATCH_TOLERANCE)
        <= BOUNDARY_CLEARANCE
        for bar in all_bars
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases: every label-bearing field is rendered visibly."""

    return []
