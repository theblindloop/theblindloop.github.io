"""Deterministic renderer and analytic LIFO-transfer replay arm."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1000, 650
TOKENS = (1, 2, 3, 4)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{name}", size)


def _replay(initial_stacks: list[list[int]], moves: list[list[int]]) -> list[list[int]]:
    stacks = [list(stack) for stack in initial_stacks]
    if len(stacks) != 3:
        raise ValueError("exactly three stacks are required")
    for move in moves:
        if len(move) != 2:
            raise ValueError("each move needs two endpoints")
        source, destination = int(move[0]), int(move[1])
        if source not in (1, 2, 3) or destination not in (1, 2, 3) or source == destination:
            raise ValueError("invalid move")
        if stacks[source - 1]:
            stacks[destination - 1].append(stacks[source - 1].pop())
    return stacks


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 130363 + 71)
    target = 1 + seed % 4
    for _ in range(30000):
        double = rng.randrange(3)
        lengths = [1, 1, 1]
        lengths[double] = 2
        order = list(TOKENS)
        rng.shuffle(order)
        initial: list[list[int]] = []
        cursor = 0
        for length in lengths:
            stack = order[cursor:cursor + length]
            cursor += length
            initial.append(stack)
        moves = []
        for _step in range(5):
            source = rng.randrange(3)
            destination = rng.randrange(2)
            if destination >= source:
                destination += 1
            moves.append([source + 1, destination + 1])
        query = rng.randrange(3) + 1
        live = [list(stack) for stack in initial]
        effective = 0
        noops = 0
        for source, destination in moves:
            if live[source - 1]:
                live[destination - 1].append(live[source - 1].pop())
                effective += 1
            else:
                noops += 1
        if live[query - 1] and live[query - 1][-1] == target and effective >= 2 and noops >= 1:
            return {
                "initial_stacks": initial,
                "moves": moves,
                "query_stack": query,
                "palette": (seed // 4) % 2,
                "offset": (-14, 0, 14)[(seed // 8) % 3],
            }
    raise RuntimeError("could not sample balanced LIFO scene")


def _dot_positions(cx: int, cy: int, count: int, spacing: int = 15) -> list[tuple[int, int]]:
    start = cx - spacing * (count - 1) // 2
    return [(start + index * spacing, cy) for index in range(count)]


def _badge(draw: ImageDraw.ImageDraw, cx: int, cy: int, count: int, fill: tuple[int, int, int], *, width: int = 58, height: int = 30) -> None:
    draw.rounded_rectangle((cx - width // 2, cy - height // 2, cx + width // 2, cy + height // 2), radius=8, fill=fill)
    for x, y in _dot_positions(cx, cy, count, 13):
        draw.ellipse((x - 4, y - 4, x + 4, y + 4), fill=(25, 31, 42))


def _token(draw: ImageDraw.ImageDraw, cx: int, cy: int, number: int, fill: tuple[int, int, int]) -> None:
    draw.rounded_rectangle((cx - 44, cy - 19, cx + 44, cy + 19), radius=9, fill=fill)
    for x, y in _dot_positions(cx, cy, number, 17):
        draw.ellipse((x - 5, y - 5, x + 5, y + 5), fill=(25, 31, 42))


def render(scene: dict) -> Image.Image:
    initial_stacks = scene["initial_stacks"]
    moves = scene["moves"]
    query_stack = int(scene["query_stack"])
    palette = int(scene["palette"])
    offset = int(scene["offset"])
    token_color = ((35, 145, 190), (112, 88, 190))[palette]
    move_color = ((239, 161, 65), (226, 125, 74))[palette]
    stack_color = ((73, 174, 119), (48, 158, 151))[palette]
    query_color = ((183, 70, 151), (211, 70, 116))[palette]
    ink = (25, 31, 42)
    muted = (92, 102, 115)
    pale = (232, 237, 242)
    card = (248, 250, 252)
    image = Image.new("RGB", (WIDTH, HEIGHT), "white")
    draw = ImageDraw.Draw(image)

    draw.text((38, 24), "STACK TRANSFER REPLAY", fill=ink, font=_font(28, True))
    draw.text((38, 63), "Move only the TOP token. Empty source = no move.", fill=muted, font=_font(18))
    draw.text((38, 105), "STARTING STACKS", fill=ink, font=_font(17, True))
    centers = [105 + offset, 245 + offset, 385 + offset]
    for index, (cx, stack) in enumerate(zip(centers, initial_stacks), 1):
        draw.line((cx - 53, 145, cx - 53, 350, cx + 53, 350, cx + 53, 145), fill=pale, width=5)
        for level, number in enumerate(stack):
            cy = 315 - 47 * level
            _token(draw, cx, cy, int(number), token_color)
        _badge(draw, cx, 390, index, stack_color)
        if index == query_stack:
            draw.rounded_rectangle((cx - 39, 367, cx + 39, 413), radius=11, outline=query_color, width=6)
            draw.text((cx, 420), "QUERY", anchor="ma", fill=query_color, font=_font(14, True))

    draw.text((505, 105), "MOVE CARDS  —  READ LEFT TO RIGHT", fill=ink, font=_font(17, True))
    for index, move in enumerate(moves):
        cx = 545 + offset + index * 90
        draw.rounded_rectangle((cx - 37, 145, cx + 37, 365), radius=12, fill=card, outline=pale, width=3)
        _badge(draw, cx, 188, int(move[0]), move_color)
        draw.line((cx, 222, cx, 273), fill=ink, width=5)
        draw.polygon([(cx, 282), (cx - 11, 264), (cx + 11, 264)], fill=ink)
        _badge(draw, cx, 315, int(move[1]), move_color)
        draw.text((cx, 378), str(index + 1), anchor="ma", fill=muted, font=_font(14, True))

    draw.rounded_rectangle((38, 500, 962, 605), radius=14, fill=(247, 249, 251), outline=pale, width=2)
    draw.text((60, 520), "CARD KEY", fill=ink, font=_font(15, True))
    _badge(draw, 180, 550, 1, move_color)
    draw.text((220, 539), "left badge = source", fill=muted, font=_font(16))
    draw.line((460, 530, 460, 571), fill=ink, width=4)
    draw.polygon([(460, 580), (450, 563), (470, 563)], fill=ink)
    _badge(draw, 570, 550, 2, move_color)
    draw.text((610, 539), "right badge = destination", fill=muted, font=_font(16))
    draw.text((60, 617), "Dots name stacks and tokens; vertical position shows each stack's order.", fill=muted, font=_font(15))
    return image


def analytic_gold(scene: dict) -> str:
    final = _replay(scene["initial_stacks"], scene["moves"])
    query = int(scene["query_stack"])
    if not final[query - 1]:
        raise ValueError("queried stack finishes empty")
    return str(final[query - 1][-1])


def margin(scene: dict) -> float:
    return 14.0


def is_quarantined(scene: dict) -> bool:
    try:
        tokens = sorted(number for stack in scene["initial_stacks"] for number in stack)
        final = _replay(scene["initial_stacks"], scene["moves"])
        query = int(scene["query_stack"])
        return tokens != list(TOKENS) or len(scene["moves"]) != 5 or not final[query - 1]
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
