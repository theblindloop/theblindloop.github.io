"""Scene sampler, raster renderer and analytic gold for ``rise_chain_reference``.

Nine gray bars of equal width stand on one black baseline.  A single black
reference bar stands on the same baseline, left of a vertical divider.  A
neighbouring pair of gray bars is a *rise* when the right bar exceeds the left
bar by strictly more than the reference height; the answer is the number of
bars in the longest run of consecutive bars linked by rises.

Every mark is an axis-aligned rectangle painted at integer pixel coordinates,
so the raster has no antialiasing and every measured extent is exact.  Bar
heights and the reference height are sampled as floats and painted at
``round(h)``: that rounding is the world's only latent alias, so the analytic
gold is defined on the quantized heights and ``latent_symmetries`` returns
sub-pixel perturbations that provably stay inside the same buckets.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "rise_chain_reference-1.0.0"

# --- fixed page geometry (integer raster, no antialiasing anywhere) -----------
WIDTH = 760
HEIGHT = 430
BASE_Y = 384              # top row of the black baseline rule
BASE_THICK = 3
TOP_LIMIT = 40            # bars never reach above this row
DIVIDER_X = 152           # vertical black divider between reference and chart
DIVIDER_W = 3
DIVIDER_TOP = 56
REF_X = 88                # left edge of the black reference bar
BAR_W = 44
BAR_GAP = 18
CHART_X0 = 184            # left edge of the first gray bar
BAR_COUNT = 9

BACKGROUND = (255, 255, 255)
INK = (20, 20, 20)        # baseline, divider, reference bar
BAR = (112, 112, 112)     # every gray chart bar
GUIDE = (231, 231, 231)   # decorative horizontal guide lines

MIN_BAR = 26              # smallest painted gray bar height, px
MAX_BAR = 300             # tallest painted gray bar height, px
REF_LO = 11               # reference-bar height range, px
REF_HI = 34
MAX_CHAIN = 5             # declared answer alphabet is 1..MAX_CHAIN

CLEAN_MARGIN = 9.0        # off-quarantine scenes keep at least this margin
QUARANTINE_PX = 5.0       # |step - reference| below this is a boundary case
NEAR_LO = 2               # sampled boundary cases live in [NEAR_LO, NEAR_HI]
NEAR_HI = 4
BOUNDARY_EVERY = 6        # one scene in six is deliberately near the boundary


# --- latent scene -------------------------------------------------------------
def _painted(values) -> list[int]:
    return [int(round(float(v))) for v in values]


def _chain_length(heights: list[int], reference: int) -> int:
    """Longest run of consecutive bars linked by rises, in bars."""
    best = 1
    run = 1
    for left, right in zip(heights, heights[1:]):
        if right - left > reference:
            run += 1
            best = max(best, run)
        else:
            run = 1
    return best


def _naive_chain_length(heights: list[int]) -> int:
    """What a reader who ignores the reference bar (treats it as zero) would say."""
    return _chain_length(heights, 0)


def _margin_px(heights: list[int], reference: int) -> float:
    return float(min(abs((r - l) - reference) for l, r in zip(heights, heights[1:])))


def _try_steps(rng: random.Random, target: int, reference: int, gap: int, tight: bool):
    """Sample eight integer steps whose only rise run has length ``target - 1``."""
    rise_count = target - 1
    slots = BAR_COUNT - 1
    start = rng.randrange(0, slots - rise_count + 1) if rise_count else 0
    plan = [start <= i < start + rise_count for i in range(slots)]
    forced = rng.randrange(slots) if tight else -1
    steps = []
    for index, is_rise in enumerate(plan):
        span = gap if index == forced else rng.randint(gap, gap + 40)
        if is_rise:
            steps.append(reference + span)
        else:
            steps.append(reference - span)
    heights = [0]
    for step in steps:
        heights.append(heights[-1] + step)
    low = min(heights)
    span = max(heights) - low
    headroom = MAX_BAR - MIN_BAR - span
    if headroom < 0:
        return None
    lift = MIN_BAR - low + rng.randint(0, headroom)
    heights = [h + lift for h in heights]
    if max(heights) > MAX_BAR or min(heights) < MIN_BAR:
        return None
    if _chain_length(heights, reference) != target:
        return None
    return heights


def sample_scene(seed: int) -> dict:
    """Deterministic latent scene for ``seed``."""
    rng = random.Random(0x5150 ^ (seed * 2654435761 % (1 << 32)))
    tight = seed % BOUNDARY_EVERY == 0
    target = 1 + seed % MAX_CHAIN
    want_naive_differs = bool(rng.getrandbits(1))

    heights = None
    reference = REF_LO
    for attempt in range(6000):
        reference = rng.randint(REF_LO, REF_HI)
        gap = rng.randint(NEAR_LO, NEAR_HI) if tight else int(CLEAN_MARGIN) + rng.randint(0, 6)
        trial = _try_steps(rng, target, reference, gap, tight)
        if trial is None:
            continue
        differs = _naive_chain_length(trial) != target
        if attempt < 3000 and differs != want_naive_differs:
            continue
        heights = trial
        break
    if heights is None:  # pragma: no cover - the constrained sampler always succeeds
        raise RuntimeError(f"scene sampling failed for seed {seed}")

    jitter = [round(rng.uniform(-0.42, 0.42), 4) for _ in range(BAR_COUNT + 1)]
    guide_count = rng.randint(3, 5)
    guides = sorted(
        rng.sample(range(TOP_LIMIT + 20, BASE_Y - 20, 14), guide_count)
    )
    return {
        "heights": [round(h + j, 4) for h, j in zip(heights, jitter)],
        "reference": round(reference + jitter[BAR_COUNT], 4),
        "guides": [float(g) for g in guides],
    }


# --- analytic consequences ----------------------------------------------------
def _quantized(scene: dict) -> tuple[list[int], int]:
    return _painted(scene["heights"]), int(round(float(scene["reference"])))


def analytic_gold(scene: dict) -> str:
    heights, reference = _quantized(scene)
    return str(_chain_length(heights, reference))


def margin(scene: dict) -> float:
    heights, reference = _quantized(scene)
    return _margin_px(heights, reference)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def naive_gold(scene: dict) -> str:
    """Diagnostic: the answer of a reader who ignores the reference bar."""
    heights, _ = _quantized(scene)
    return str(_naive_chain_length(heights))


def _runs(heights: list[int], reference: int) -> list[tuple[int, int]]:
    """All maximal chains as ``(first_bar_index, bar_count)`` pairs."""
    out: list[tuple[int, int]] = []
    start = 0
    for index in range(BAR_COUNT - 1):
        if heights[index + 1] - heights[index] <= reference:
            out.append((start, index - start + 1))
            start = index + 1
    out.append((start, BAR_COUNT - start))
    return out


def tallest_in_chain(scene: dict) -> bool:
    """Diagnostic: is the single tallest gray bar part of a longest chain?"""
    heights, reference = _quantized(scene)
    best = _chain_length(heights, reference)
    tallest = max(range(BAR_COUNT), key=lambda i: heights[i])
    return any(
        length == best and start <= tallest < start + length
        for start, length in _runs(heights, reference)
    )


def chart_third(scene: dict) -> str:
    """Diagnostic: which third of the chart holds the first bar of a longest chain."""
    heights, reference = _quantized(scene)
    best = _chain_length(heights, reference)
    if best == 1:
        return "none"
    first = next(start for start, length in _runs(heights, reference) if length == best)
    return ("left", "middle", "right")[min(2, first // 3)]


# --- raster -------------------------------------------------------------------
def _bar_x(index: int) -> int:
    return CHART_X0 + index * (BAR_W + BAR_GAP)


def render(scene: dict) -> Image.Image:
    heights, reference = _quantized(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    for guide in scene["guides"]:
        row = int(round(float(guide)))
        draw.rectangle([36, row, WIDTH - 24, row], fill=GUIDE)

    draw.rectangle([36, BASE_Y, WIDTH - 24, BASE_Y + BASE_THICK - 1], fill=INK)
    draw.rectangle([DIVIDER_X, DIVIDER_TOP, DIVIDER_X + DIVIDER_W - 1, BASE_Y - 1], fill=INK)
    draw.rectangle(
        [REF_X, BASE_Y - reference, REF_X + BAR_W - 1, BASE_Y - 1], fill=INK
    )
    for index, height in enumerate(heights):
        x0 = _bar_x(index)
        draw.rectangle([x0, BASE_Y - height, x0 + BAR_W - 1, BASE_Y - 1], fill=BAR)
    return image


# --- declared latent aliases --------------------------------------------------
def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Sub-pixel height jitter that provably stays inside the painted buckets."""
    out: list[tuple[str, dict]] = []
    for step, shift in enumerate((0.33, -0.31, 0.45, -0.43), start=1):
        twin = dict(scene)
        sign = 1 if step % 2 else -1
        twin["heights"] = [
            round(float(int(round(float(v)))) + shift * (1 if (i + step) % 2 else -1), 4)
            for i, v in enumerate(scene["heights"])
        ]
        twin["reference"] = round(
            float(int(round(float(scene["reference"])))) + shift * sign, 4
        )
        twin["guides"] = [
            round(float(int(round(float(g)))) + shift * sign * 0.5, 4) for g in scene["guides"]
        ]
        out.append((f"subpixel_jitter_{step}", twin))
    return out


def mirrored_scene(scene: dict) -> dict:
    """Declared label-transforming symmetry: left-right mirror of the chart.

    Mirroring reverses the bar order, so the longest rise chain of the mirrored
    scene is the longest *fall* chain of the original.  Used by the tests, never
    by gold.
    """
    twin = dict(scene)
    twin["heights"] = list(reversed(list(scene["heights"])))
    return twin


def descending_chain_length(scene: dict) -> int:
    heights, reference = _quantized(scene)
    return _chain_length(list(reversed(heights)), reference)
