"""Deterministic planar network renderer and independent analytic decision."""

from __future__ import annotations

import copy
from collections import defaultdict

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
COLORS = ("red", "blue", "orange")
RGB = {
    "background": (255, 255, 255),
    "black": (24, 27, 31),
    "ordinary": (199, 208, 218),
    "green": (38, 158, 73),
    "purple": (143, 68, 173),
    "red": (220, 55, 55),
    "blue": (45, 105, 210),
    "orange": (235, 139, 35),
}

# Twelve core vertices on a 4 x 3 lattice, then START and three terminal leaves.
_BASE_POSITIONS = [
    [106, 146], [206, 146], [306, 146], [406, 146],
    [106, 246], [206, 246], [306, 246], [406, 246],
    [106, 346], [206, 346], [306, 346], [406, 346],
    [306, 66], [46, 246], [206, 426], [206, 66],
]

# All terminal leaves have degree one and shortest-path distance five from START.
_BASE_EDGES = [
    [0, 1], [0, 4], [1, 5], [2, 3], [2, 6], [4, 5], [4, 8],
    [5, 6], [5, 9], [6, 7], [6, 10], [8, 9], [9, 10], [10, 11],
    [12, 2], [13, 4], [14, 9], [15, 1],
]
_START = 12
_CHECKPOINTS = [0, 8, 10]
_TERMINAL_NODES = [13, 14, 15]
_ANSWER_NODE = 15


def _transform_point(point: list[int], mode: int) -> list[int]:
    x, y = point
    transforms = (
        (x, y),
        (WIDTH - x, y),
        (x, HEIGHT - y),
        (WIDTH - x, HEIGHT - y),
        (WIDTH - y, x),
        (y, WIDTH - x),
        (y, x),
        (WIDTH - y, HEIGHT - x),
    )
    return list(transforms[mode % len(transforms)])


def sample_scene(seed: int) -> dict:
    """Return a balanced isomorphic scene for one integer seed."""
    answer_color = COLORS[seed % 3]
    remaining = [color for color in COLORS if color != answer_color]
    if (seed // 3) % 2:
        remaining.reverse()
    terminals = [
        {"node": 13, "color": remaining[0]},
        {"node": 14, "color": remaining[1]},
        {"node": _ANSWER_NODE, "color": answer_color},
    ]
    mode = (seed // 3) % 8
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "positions": [_transform_point(point, mode) for point in _BASE_POSITIONS],
        "edges": copy.deepcopy(_BASE_EDGES),
        "start": _START,
        "checkpoints": list(_CHECKPOINTS),
        "terminals": terminals,
    }


def _terminal_colors(entries: list[dict]) -> dict[int, str]:
    return {int(entry["node"]): str(entry["color"]) for entry in entries}


def _route_counts(scene: dict) -> dict[str, int]:
    # These four fields, and only these four, determine the analytic label.
    edges = scene["edges"]
    start = int(scene["start"])
    checkpoints = {int(node) for node in scene["checkpoints"]}
    terminal_colors = _terminal_colors(scene["terminals"])

    adjacency: dict[int, list[int]] = defaultdict(list)
    for pair in edges:
        a, b = int(pair[0]), int(pair[1])
        adjacency[a].append(b)
        adjacency[b].append(a)
    counts = {color: 0 for color in COLORS}

    def visit(node: int, seen: set[int]) -> None:
        if node in terminal_colors:
            if checkpoints <= seen:
                counts[terminal_colors[node]] += 1
            return
        for neighbor in adjacency[node]:
            if neighbor not in seen:
                visit(neighbor, seen | {neighbor})

    visit(start, {start})
    return counts


def analytic_gold(scene: dict) -> str:
    counts = _route_counts(scene)
    ranked = sorted(counts.items(), key=lambda item: (-item[1], item[0]))
    if ranked[0][1] <= 0 or ranked[0][1] == ranked[1][1]:
        raise ValueError(f"ambiguous or unsatisfied checkpoint routes: {counts}")
    return ranked[0][0]


def margin(scene: dict) -> float:
    values = sorted(_route_counts(scene).values(), reverse=True)
    return float(values[0] - values[1])


def is_quarantined(scene: dict) -> bool:
    counts = sorted(_route_counts(scene).values(), reverse=True)
    return counts[0] <= 0 or counts[1] != 0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Exercise the genuine alias induced by undirected edge serialization."""
    twin = copy.deepcopy(scene)
    twin["edges"] = [[b, a] for a, b in reversed(scene["edges"])]
    return [("reverse_undirected_edge_storage", twin)]


def render(scene: dict) -> Image.Image:
    """Render only observable scene fields to an antialiased RGB image."""
    width = int(scene["width"])
    height = int(scene["height"])
    positions = [[int(p[0]), int(p[1])] for p in scene["positions"]]
    edges = [[int(pair[0]), int(pair[1])] for pair in scene["edges"]]
    start = int(scene["start"])
    checkpoints = {int(node) for node in scene["checkpoints"]}
    terminal_colors = _terminal_colors(scene["terminals"])

    scale = 4
    image = Image.new("RGB", (width * scale, height * scale), RGB["background"])
    draw = ImageDraw.Draw(image)
    for a, b in sorted((min(a, b), max(a, b)) for a, b in edges):
        p1 = tuple(value * scale for value in positions[a])
        p2 = tuple(value * scale for value in positions[b])
        draw.line([p1, p2], fill=RGB["black"], width=7 * scale)

    outer_radius = 16 * scale
    inner_radius = 11 * scale
    for node, (x, y) in enumerate(positions):
        cx, cy = x * scale, y * scale
        if node == start:
            fill = RGB["green"]
        elif node in checkpoints:
            fill = RGB["purple"]
        elif node in terminal_colors:
            fill = RGB[terminal_colors[node]]
        else:
            fill = RGB["ordinary"]
        draw.ellipse(
            [cx - outer_radius, cy - outer_radius, cx + outer_radius, cy + outer_radius],
            fill=RGB["black"],
        )
        draw.ellipse(
            [cx - inner_radius, cy - inner_radius, cx + inner_radius, cy + inner_radius],
            fill=fill,
        )
    return image.resize((width, height), Image.Resampling.LANCZOS)
