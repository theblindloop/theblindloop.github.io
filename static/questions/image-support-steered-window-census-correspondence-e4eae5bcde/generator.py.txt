"""Latent sampler and Pillow renderer for Window Census Correspondence."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw


CANVAS = 960
BACKGROUND = (248, 246, 238)
DARK_TILE = (43, 47, 52)
LIGHT_TILE = (205, 211, 218)
PIP_BLUE = (42, 111, 207)
GUIDE = (151, 157, 163)
TAG_RGB = {
    "amber": (244, 151, 28),
    "cyan": (33, 186, 222),
    "violet": (166, 69, 190),
}
ANSWERS = ("amber", "cyan", "violet")
SOURCE_ORIGIN = (165, 350)
CARD_BOXES = ((550, 230, 765, 340), (550, 385, 765, 495), (550, 540, 765, 650))


def _window_counts(source: list[list[int]]) -> list[int]:
    return [
        sum(source[row + dy][column + dx] for dy in range(2) for dx in range(2))
        for row, column in ((0, 0), (0, 1), (1, 0), (1, 1))
    ]


def _valid_sources() -> list[list[list[int]]]:
    output = []
    for bits in itertools.product((0, 1), repeat=9):
        board = [list(bits[index : index + 3]) for index in range(0, 9, 3)]
        if sorted(_window_counts(board)) == [1, 2, 3, 4]:
            output.append(board)
    return output


VALID_SOURCES = _valid_sources()


def _hamming(left: list[int], right: list[int]) -> int:
    return sum(a != b for a, b in zip(left, right))


def sample_scene(seed: int) -> dict:
    seed = int(seed)
    rng = random.Random(seed ^ 0x71C0E55)
    source = [row[:] for row in VALID_SOURCES[rng.randrange(len(VALID_SOURCES))]]
    target = _window_counts(source)
    alternatives = [
        list(values)
        for values in itertools.permutations((1, 2, 3, 4))
        if list(values) != target
    ]
    near = [values for values in alternatives if _hamming(values, target) == 2]
    far = [values for values in alternatives if _hamming(values, target) >= 3]
    first = near[rng.randrange(len(near))]
    second = far[rng.randrange(len(far))]
    correct_position = (abs(seed) // 3) % 3
    proposals = [first, second]
    proposals.insert(correct_position, target)

    desired_answer = ANSWERS[abs(seed) % 3]
    remaining = [answer for answer in ANSWERS if answer != desired_answer]
    rng.shuffle(remaining)
    tag_colors = remaining
    tag_colors.insert(correct_position, desired_answer)
    return {"source": source, "proposals": proposals, "tag_colors": tag_colors}


def _matching_index(scene: dict) -> tuple[int, list[int]]:
    target = _window_counts(scene["source"])
    distances = [_hamming(target, proposal) for proposal in scene["proposals"]]
    winners = [index for index, distance in enumerate(distances) if distance == 0]
    if len(winners) != 1:
        raise ValueError("scene must contain exactly one exact census")
    return winners[0], distances


def analytic_gold(scene: dict) -> str:
    winner, _ = _matching_index(scene)
    return str(scene["tag_colors"][winner])


def margin(scene: dict) -> float:
    _, distances = _matching_index(scene)
    return float(sorted(distances)[1])


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 2.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _draw_source(draw: ImageDraw.ImageDraw, source: list[list[int]]) -> None:
    x0, y0 = SOURCE_ORIGIN
    tile = 46
    gap = 10
    for row in range(3):
        for column in range(3):
            left = x0 + column * (tile + gap)
            top = y0 + row * (tile + gap)
            color = DARK_TILE if source[row][column] else LIGHT_TILE
            draw.rectangle((left, top, left + tile - 1, top + tile - 1), fill=color)


def _pip_offsets(count: int) -> tuple[tuple[int, int], ...]:
    layouts = {
        1: ((0, 0),),
        2: ((-11, 0), (11, 0)),
        3: ((0, -10), (-11, 9), (11, 9)),
        4: ((-11, -9), (11, -9), (-11, 9), (11, 9)),
    }
    return layouts[count]


def _draw_card(
    draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], values: list[int], tag: str
) -> None:
    left, top, right, bottom = box
    draw.rectangle(box, outline=TAG_RGB[tag], width=8)
    center_x = (left + right) // 2
    center_y = (top + bottom) // 2
    draw.line((center_x, top + 12, center_x, bottom - 12), fill=GUIDE, width=2)
    draw.line((left + 12, center_y, right - 12, center_y), fill=GUIDE, width=2)
    centers = (
        ((left + center_x) // 2, (top + center_y) // 2),
        ((center_x + right) // 2, (top + center_y) // 2),
        ((left + center_x) // 2, (center_y + bottom) // 2),
        ((center_x + right) // 2, (center_y + bottom) // 2),
    )
    radius = 7
    for count, (cx, cy) in zip(values, centers):
        for dx, dy in _pip_offsets(int(count)):
            draw.ellipse(
                (cx + dx - radius, cy + dy - radius, cx + dx + radius, cy + dy + radius),
                fill=PIP_BLUE,
            )


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (CANVAS, CANVAS), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_source(draw, scene["source"])
    for box, proposal, tag in zip(CARD_BOXES, scene["proposals"], scene["tag_colors"]):
        _draw_card(draw, box, proposal, str(tag))
    return image
