"""Deterministic renderer and analytic arm for cord_peg_winding."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

BACKGROUND = (248, 246, 239)
CORD = (30, 35, 41)
ORANGE = (232, 101, 31)
BLUE = (39, 103, 196)
YELLOW = (245, 190, 45)
PEG_EDGE = (151, 104, 18)
CORD_WIDTH = 9
END_RADIUS = 8
PEG_RADIUS = 11


def _angular_sweep(path: list[list[int]], peg: list[int]) -> float:
    angles = [math.atan2(y - peg[1], x - peg[0]) for x, y in path]
    total = 0.0
    for left, right in zip(angles, angles[1:]):
        delta = right - left
        while delta <= -math.pi:
            delta += 2.0 * math.pi
        while delta > math.pi:
            delta -= 2.0 * math.pi
        total += delta
    return total


def _minimum_nonlocal_clearance(path: list[list[int]]) -> float:
    # Vertices are densely sampled, so skip nearby indices on the same stroke.
    best = float("inf")
    for index, a in enumerate(path):
        for b in path[index + 32 :]:
            best = min(best, math.dist(a, b))
    return best


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    width = height = 512
    peg = [256 + rng.randint(-17, 17), 256 + rng.randint(-17, 17)]
    phase = rng.uniform(-math.pi, math.pi)
    direction = 1.0 if int(seed) % 2 == 0 else -1.0
    # The coil is deliberately elliptical: its decisive raster support is one
    # elongated regional trace rather than a compact disk-shaped patch.
    turns = rng.uniform(1.08, 1.18)
    outer_x = rng.uniform(163.0, 174.0)
    inner_x = rng.uniform(86.0, 95.0)
    outer_y = rng.uniform(44.0, 48.0)
    inner_y = rng.uniform(28.0, 31.0)
    samples = 321
    path: list[list[int]] = []
    for index in range(samples):
        fraction = index / (samples - 1)
        theta = phase + direction * turns * 2.0 * math.pi * fraction
        # A slight radial wave defeats a fixed-radius shortcut while preserving
        # wide coil separation and a single simple tube.
        wave = math.sin(5.0 * math.pi * fraction + phase)
        radius_x = outer_x + (inner_x - outer_x) * fraction + 2.0 * wave
        radius_y = outer_y + (inner_y - outer_y) * fraction
        x = round(peg[0] + radius_x * math.cos(theta))
        y = round(peg[1] + radius_y * math.sin(theta))
        point = [x, y]
        if not path or point != path[-1]:
            path.append(point)
    return {
        "path": path,
        "orange": list(path[0]),
        "blue": list(path[-1]),
        "peg": peg,
        "width": width,
        "height": height,
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BACKGROUND)
    draw = ImageDraw.Draw(image)
    path = [tuple(map(int, point)) for point in scene["path"]]
    # Endpoint disks go down first so the black cord remains an uninterrupted
    # pixel component even when a flattened coil passes nearby.
    for key, color in (("orange", ORANGE), ("blue", BLUE)):
        x, y = map(int, scene[key])
        draw.ellipse((x-END_RADIUS, y-END_RADIUS, x+END_RADIUS, y+END_RADIUS), fill=color)
    draw.line(path, fill=CORD, width=CORD_WIDTH, joint="curve")
    joint_radius = CORD_WIDTH // 2
    for x, y in path[1:-1:8]:
        draw.ellipse((x-joint_radius, y-joint_radius, x+joint_radius, y+joint_radius), fill=CORD)
    px, py = map(int, scene["peg"])
    draw.ellipse((px-PEG_RADIUS, py-PEG_RADIUS, px+PEG_RADIUS, py+PEG_RADIUS),
                 fill=YELLOW, outline=PEG_EDGE, width=3)
    return image


def analytic_gold(scene: dict) -> str:
    sweep = _angular_sweep(scene["path"], scene["peg"])
    # Image coordinates have y increasing downwards, so positive sweep is clockwise.
    return "clockwise" if sweep > 0.0 else "counterclockwise"


def margin(scene: dict) -> float:
    return abs(_angular_sweep(scene["path"], scene["peg"]))


def is_quarantined(scene: dict) -> bool:
    try:
        path = scene["path"]
        peg = scene["peg"]
        if len(path) < 200 or list(scene["orange"]) != list(path[0]) or list(scene["blue"]) != list(path[-1]):
            return True
        if margin(scene) < 2.0 * math.pi:
            return True
        if min(math.dist(point, peg) for point in path) < PEG_RADIUS + CORD_WIDTH + 7:
            return True
        if _minimum_nonlocal_clearance(path) < CORD_WIDTH + 2:
            return True
        return False
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
