"""Independent final-PNG-only inverse arm for cord winding direction."""

from __future__ import annotations

from collections import deque
import math

import numpy as np
from PIL import Image


def _masks(image: Image.Image) -> tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    black = (array[:, :, 0] < 85) & (array[:, :, 1] < 90) & (array[:, :, 2] < 95)
    orange = (array[:, :, 0] > 185) & (array[:, :, 1] > 55) & (array[:, :, 1] < 155) & (array[:, :, 2] < 90)
    blue = (array[:, :, 0] < 100) & (array[:, :, 1] > 55) & (array[:, :, 1] < 155) & (array[:, :, 2] > 140)
    yellow = (array[:, :, 0] > 205) & (array[:, :, 1] > 145) & (array[:, :, 1] < 225) & (array[:, :, 2] < 105)
    return black, orange, blue, yellow


def _centroid(mask: np.ndarray, name: str, minimum: int) -> tuple[int, int]:
    points = np.argwhere(mask)
    if len(points) < minimum:
        raise ValueError(f"missing or damaged {name}")
    y, x = np.rint(points.mean(axis=0)).astype(int)
    return int(y), int(x)


def _nearest_ink(mask: np.ndarray, point: tuple[int, int], name: str) -> tuple[int, int]:
    points = np.argwhere(mask)
    if not len(points):
        raise ValueError("missing cord ink")
    distances = ((points - np.asarray(point, dtype=np.int64)) ** 2).sum(axis=1)
    index = int(np.argmin(distances))
    if int(distances[index]) > 18 * 18:
        raise ValueError(f"{name} is not incident to the cord")
    y, x = points[index]
    return int(y), int(x)


def _shortest_path(mask: np.ndarray, start: tuple[int, int], finish: tuple[int, int]) -> list[tuple[int, int]]:
    height, width = mask.shape
    sy, sx = start
    fy, fx = finish
    if not (0 <= sy < height and 0 <= sx < width and mask[sy, sx]):
        raise ValueError("orange endpoint is not on the cord")
    if not (0 <= fy < height and 0 <= fx < width and mask[fy, fx]):
        raise ValueError("blue endpoint is not on the cord")
    parent_y = np.full(mask.shape, -2, dtype=np.int16)
    parent_x = np.full(mask.shape, -2, dtype=np.int16)
    parent_y[sy, sx] = -1
    parent_x[sy, sx] = -1
    queue: deque[tuple[int, int]] = deque([(sy, sx)])
    while queue and parent_y[fy, fx] == -2:
        y, x = queue.popleft()
        for ny, nx in ((y-1, x), (y+1, x), (y, x-1), (y, x+1)):
            if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and parent_y[ny, nx] == -2:
                parent_y[ny, nx] = y
                parent_x[ny, nx] = x
                queue.append((ny, nx))
    if parent_y[fy, fx] == -2:
        raise ValueError("the visible cord is broken between its colored endpoints")
    path: list[tuple[int, int]] = []
    y, x = fy, fx
    while y >= 0:
        path.append((y, x))
        py, px = int(parent_y[y, x]), int(parent_x[y, x])
        y, x = py, px
    path.reverse()
    return path


def _sweep(path: list[tuple[int, int]], peg: tuple[int, int]) -> float:
    py, px = peg
    # Downsample the raster walk enough to suppress pixel-scale Manhattan jitter.
    points = path[::8]
    if points[-1] != path[-1]:
        points.append(path[-1])
    angles = [math.atan2(y - py, x - px) for y, x in points]
    total = 0.0
    for left, right in zip(angles, angles[1:]):
        delta = right - left
        while delta <= -math.pi:
            delta += 2.0 * math.pi
        while delta > math.pi:
            delta -= 2.0 * math.pi
        total += delta
    return total


def decision_from_image(image: Image.Image) -> str:
    if image.width < 300 or image.height < 300:
        raise ValueError("image is too small for winding recovery")
    black, orange_mask, blue_mask, yellow_mask = _masks(image)
    orange_mark = _centroid(orange_mask, "orange endpoint", 50)
    blue_mark = _centroid(blue_mask, "blue endpoint", 50)
    peg = _centroid(yellow_mask, "yellow peg", 180)
    route = black
    if int(route.sum()) < 4500:
        raise ValueError("too little intact cord ink")
    orange = _nearest_ink(route, orange_mark, "orange endpoint")
    blue = _nearest_ink(route, blue_mark, "blue endpoint")
    path = _shortest_path(route, orange, blue)
    if len(path) < 550:
        raise ValueError("cord path is too short or contains a visible shortcut")
    minimum_radius = min(math.hypot(y - peg[0], x - peg[1]) for y, x in path[::8])
    if minimum_radius < 25.0:
        raise ValueError("cord touches or approaches the peg ambiguously")
    sweep = _sweep(path, peg)
    if abs(sweep) < 2.0 * math.pi:
        raise ValueError("visible net winding is less than one full turn")
    return "clockwise" if sweep > 0.0 else "counterclockwise"
