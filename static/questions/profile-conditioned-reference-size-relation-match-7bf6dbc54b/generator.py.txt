"""Reference-guided within-glyph size-relation visual search world.

A dark reference glyph -- two dark filled disks side by side -- is drawn in a
box at the top of the image. Below, the image is split into four quadrants
(top-left, top-right, bottom-left, bottom-right), each holding one dark glyph
that is also composed of two side-by-side filled disks (a left disk and a right
disk).

Every glyph's *attribute* is the internal left-vs-right size relation of its
two disks, drawn from exactly three classes:

  - "LB" : the left disk is clearly larger than the right disk;
  - "RB" : the right disk is clearly larger than the left disk;
  - "EQ" : the two disks are exactly the same size.

The reference glyph exhibits one of these three relation classes. Exactly one
quadrant's glyph exhibits the SAME relation class as the reference glyph; the
other three quadrants each exhibit one of the *other* two relation classes (so
no second glyph matches the reference and a majority-vote shortcut cannot
recover the answer, because the three distractors are not required to share one
class). The decision is the quadrant containing the glyph whose left-right size
relation matches the reference glyph's relation. The answer is determined purely
by comparing each glyph's internal disk-size relation to the reference; no
colour, count, position, absolute-size, or salience cue identifies the target.

All rendering is deterministic, every disk is fully visible in the raster, and
every non-equal glyph keeps a clear size gap between its two disks so the
relation is resolvable in the pixels.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

SIZE = 480
BG = (255, 255, 255)
SEG = (20, 20, 20)
GRAY = (196, 196, 196)
BOX = (170, 170, 170)

REF_CX = SIZE // 2
REF_CY = 70
GLYPH_OFF = 40  # each disk centre is offset this many px from its glyph centre

SECTORS = ("TL", "TR", "BL", "BR")
SECTOR_CENTERS = {
    "TL": (SIZE // 4, 250),
    "TR": (3 * SIZE // 4, 250),
    "BL": (SIZE // 4, 410),
    "BR": (3 * SIZE // 4, 410),
}

RMIN = 12
RMAX = 30
# Every non-equal glyph keeps |rL - rR| >= this many px, a guaranteed clear
# left-vs-right size gap so the relation is pixel-resolvable.
MIN_GAP = 6
# Below this decision margin the relation is considered ambiguous and the scene
# is quarantined.
QUARANTINE_THRESH = 4.5

CLASSES = ("LB", "RB", "EQ")


def _relation(rl: int, rr: int) -> str:
    if rl == rr:
        return "EQ"
    return "LB" if rl > rr else "RB"


def _consistent_r_pair(rng, cls: str) -> tuple[int, int]:
    """Return (rL, rR) consistent with relation class 'cls'."""
    if cls == "EQ":
        r = int(rng.integers(RMIN, RMAX + 1))
        return r, r
    if cls == "LB":
        base = int(rng.integers(RMIN, RMAX - MIN_GAP))
        left = int(rng.integers(base + MIN_GAP, RMAX + 1))
        return left, base
    # RB
    base = int(rng.integers(RMIN, RMAX - MIN_GAP))
    right = int(rng.integers(base + MIN_GAP, RMAX + 1))
    return base, right


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(seed)

    ref_class = CLASSES[int(rng.integers(0, len(CLASSES)))]
    ref_rl, ref_rr = _consistent_r_pair(rng, ref_class)
    reference = {
        "cxl": int(REF_CX - GLYPH_OFF),
        "cyl": int(REF_CY),
        "rl": int(ref_rl),
        "cxr": int(REF_CX + GLYPH_OFF),
        "cyr": int(REF_CY),
        "rr": int(ref_rr),
        "cls": ref_class,
    }

    winner_sector = SECTORS[int(rng.integers(0, len(SECTORS)))]
    other_classes = [c for c in CLASSES if c != ref_class]

    candidates = []
    for sector in SECTORS:
        gx, gy = SECTOR_CENTERS[sector]
        if sector == winner_sector:
            cls = ref_class
        else:
            cls = other_classes[int(rng.integers(0, len(other_classes)))]
        rl, rr = _consistent_r_pair(rng, cls)
        candidates.append(
            {
                "sector": sector,
                "cxl": int(gx - GLYPH_OFF),
                "cyl": int(gy),
                "rl": int(rl),
                "cxr": int(gx + GLYPH_OFF),
                "cyr": int(gy),
                "rr": int(rr),
                "cls": cls,
            }
        )
    return {"reference": reference, "candidates": candidates}


def _stamp_disk(arr: np.ndarray, cx: int, cy: int, r: int) -> None:
    ys, xs = np.ogrid[:SIZE, :SIZE]
    mask = (xs - cx) ** 2 + (ys - cy) ** 2 <= r * r
    arr[mask] = SEG


def _stamp_glyph(arr: np.ndarray, glyph: dict) -> None:
    _stamp_disk(arr, glyph["cxl"], glyph["cyl"], glyph["rl"])
    _stamp_disk(arr, glyph["cxr"], glyph["cyr"], glyph["rr"])


def render(scene: dict) -> Image.Image:
    arr = np.empty((SIZE, SIZE, 3), dtype=np.uint8)
    arr[:] = BG

    # Reference-panel box outline (light grey, never read as dark ink).
    y0, y1 = 14, 126
    arr[y0:y1, REF_CX - 170 : REF_CX - 170 + 6] = BOX
    arr[y0:y1, REF_CX + 170 - 6 : REF_CX + 170] = BOX
    arr[y0 : y0 + 4, REF_CX - 170 : REF_CX + 170] = BOX
    arr[y1 - 4 : y1, REF_CX - 170 : REF_CX + 170] = BOX
    # Horizontal divider at y=330 and vertical divider at x=240.
    arr[329:331, :] = GRAY
    arr[:, 240:242] = GRAY

    _stamp_glyph(arr, scene["reference"])
    for cand in scene["candidates"]:
        _stamp_glyph(arr, cand)
    return Image.fromarray(arr, "RGB")


def _candidate_class_quadrant(scene: dict) -> tuple[str, str]:
    ref_class = scene["reference"]["cls"]
    for cand in scene["candidates"]:
        if cand["cls"] == ref_class:
            return cand["sector"], ref_class
    return SECTORS[0], ref_class  # pragma: no cover - guaranteed by construction


def analytic_gold(scene: dict) -> str:
    sector, _ = _candidate_class_quadrant(scene)
    return sector


def margin(scene: dict) -> float:
    """Runner-up decision margin: the minimum left-vs-right size gap over every
    non-equal glyph (reference included). All non-equal glyphs keep |rL-rR| >=
    MIN_GAP, so this is a clean lower bound on how clearly each relation (and
    hence the match) is resolvable in the raster."""
    gaps = []
    glyphs = [scene["reference"]] + scene["candidates"]
    for g in glyphs:
        if g["cls"] == "EQ":
            continue
        gaps.append(float(abs(g["rl"] - g["rr"])))
    return float(min(gaps)) if gaps else float("inf")


def is_quarantined(scene: dict) -> bool:
    m = margin(scene)
    if m == float("inf"):
        return False
    return bool(m < QUARANTINE_THRESH)


def latent_symmetries(scene: dict) -> list:
    return []
