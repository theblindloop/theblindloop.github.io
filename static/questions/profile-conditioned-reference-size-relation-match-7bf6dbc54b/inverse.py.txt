"""Independent pixel-only inverse arm for reference_size_relation_match.

Receives only a PIL image and the public question contract. It separates the
dark reference glyph (top box) from the four dark candidate glyphs (quadrants
below), measures each disk's radius as half of its dark-pixel bounding-box
width, classifies each glyph's left-vs-right size relation, and selects the
quadrant whose glyph's relation equals the reference glyph's relation.

Imports nothing from the renderer, prompts, generator, or verifier, and never
sees the latent scene.
"""

from __future__ import annotations

import numpy as np


def _radius(xs: np.ndarray) -> float | None:
    if xs.size < 8:
        return None
    return float(int(int(xs.max()) - int(xs.min())) / 2.0)


def _relation(rl: float, rr: float) -> str:
    if abs(rl - rr) < 1.5:
        return "EQ"
    return "LB" if rl > rr else "RB"


def decision_from_image(image) -> str:
    w, h = image.size
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    dark = arr.sum(axis=2) < 150

    ys, xs = np.nonzero(dark)
    if xs.size == 0:
        return "TL"

    x_split = w * 0.5
    y_ref = h * 0.32
    y_top = h * 0.32
    y_mid = h * 0.6875

    # Reference glyph (top region). Split into its left and right disks by the
    # midpoint of its dark-pixel x-extent.
    ref_sel = ys < y_ref
    if ref_sel.sum() < 16:
        return "TL"
    ref_x = xs[ref_sel]
    rs = float((ref_x.min() + ref_x.max()) / 2.0)
    rl = _radius(ref_x[ref_x < rs])
    rr = _radius(ref_x[ref_x >= rs])
    if rl is None or rr is None:
        return "TL"
    ref_class = _relation(rl, rr)

    best_sector = None
    for sector in ("TL", "TR", "BL", "BR"):
        if sector in ("TL", "BL"):
            left_mask = xs < x_split
        else:
            left_mask = xs >= x_split
        if sector in ("TL", "TR"):
            band = (ys >= y_top) & (ys < y_mid)
        else:
            band = ys >= y_mid
        sel = band & left_mask
        if sel.sum() < 16:
            continue
        sx = xs[sel]
        gs = float((sx.min() + sx.max()) / 2.0)
        rl = _radius(sx[sx < gs])
        rr = _radius(sx[sx >= gs])
        if rl is None or rr is None:
            continue
        if _relation(rl, rr) == ref_class:
            best_sector = sector
            break
    return best_sector if best_sector is not None else "TL"
