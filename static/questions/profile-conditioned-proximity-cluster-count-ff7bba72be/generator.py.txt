"""Proximity-grouping cluster count world.

A row of identical dark disks is laid out along a horizontal line. Disks close
together form a cluster; a few conspicuously larger gaps split the row into
separate clusters (groups). The decision is: how many clusters (groups of
disks) does the row split into? Answer set {2, 3, 4}.

This is a global Gestalt structure decision: the whole set organizes into
perceptual groups purely by the *proximity* (relative spacing) of its parts,
not by any single disk's color, size, or position.

The renderer owns scene sampling + rasterization + the structural analytic
gold. The independent pixel-only inverse arm (``oracle.py``) reconstructs disk
positions from the final PNG alone, measures consecutive spacings, and counts
how many clusters the large-gap boundaries divide the row into.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BASELINE_Y = HEIGHT // 2
DISK_R = 9
BACKGROUND = (255, 255, 255)
INK = (30, 30, 30)

CLUSTER_COUNTS = [2, 3, 4]

# Center-to-center disk spacing inside a cluster (edge gap 4..12 px).
WITHIN_SPACING = (22.0, 30.0)
# Center-to-center spacing across a cluster boundary (a conspicuously large gap).
BETWEEN_SPACING = (75.0, 110.0)

# Maximum disk edge-clearance so connected components always separate.
MIN_COMPONENT_CLEARANCE = 4.0
# Quarantine when the within/between gap populations are not cleanly separated.
QUARANTINE_CLEARANCE_PX = 30.0


def sample_scene(seed: int) -> dict:
    """Sample one latent scene deterministically from ``seed``."""
    rng = np.random.default_rng(seed)
    k = int(rng.integers(0, len(CLUSTER_COUNTS)))
    cluster_count = CLUSTER_COUNTS[k]
    # Total disks varies 5..8 independent of k so total ink never determines k.
    total = int(rng.integers(cluster_count + 1, 9))
    sizes = [1] * cluster_count
    remaining = total - cluster_count
    while remaining > 0:
        idx = int(rng.integers(0, cluster_count))
        if sizes[idx] < 4:
            sizes[idx] += 1
            remaining -= 1
    return {"cluster_count": cluster_count, "layout_seed": int(seed)}


def _cluster_sizes(cluster_count: int, layout_seed: int) -> list[int]:
    """Deterministic per-cluster disk counts (each 1..4) summing to 5..8 disks."""
    rng = np.random.default_rng(layout_seed)
    total = int(rng.integers(cluster_count + 1, 9))
    sizes = [1] * cluster_count
    remaining = total - cluster_count
    while remaining > 0:
        idx = int(rng.integers(0, cluster_count))
        if sizes[idx] < 4:
            sizes[idx] += 1
            remaining -= 1
    return sizes


def _layout(scene: dict) -> tuple[list[float], list[float], list[float]]:
    """Return (x-positions rebased to 0, within_gaps, between_gaps) for the row."""
    rng = np.random.default_rng(scene["layout_seed"])
    k = scene["cluster_count"]
    sizes = _cluster_sizes(k, scene["layout_seed"])
    within: list[float] = []
    between: list[float] = []
    gaps: list[float] = []
    for c in range(k):
        for j in range(sizes[c]):
            if j < sizes[c] - 1:
                g = float(rng.uniform(*WITHIN_SPACING))
                within.append(g)
                gaps.append(g)
            elif c < k - 1:
                g = float(rng.uniform(*BETWEEN_SPACING))
                between.append(g)
                gaps.append(g)
            else:
                break
    positions = [0.0]
    for g in gaps:
        positions.append(positions[-1] + g)
    return positions, within, between


def _draw_scene(scene: dict) -> Image.Image:
    positions, _, _ = _layout(scene)
    span = positions[-1] if positions else 0.0
    offset_x = (WIDTH - span) / 2.0
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    d = ImageDraw.Draw(img)
    for px in positions:
        cx = offset_x + px
        d.ellipse(
            [cx - DISK_R, BASELINE_Y - DISK_R, cx + DISK_R, BASELINE_Y + DISK_R],
            fill=INK,
        )
    return img


def render(scene: dict) -> Image.Image:
    """Rasterize the latent scene (a deterministic PNG)."""
    return _draw_scene(scene)


def analytic_gold(scene: dict) -> str:
    """Structural gold: the number of proximity clusters is the declared count."""
    return str(scene["cluster_count"])


def margin(scene: dict) -> float:
    """Commitment: clearance between the within-cluster and between-cluster gap populations.

    Larger indicates the large-gap boundaries are more decisively separated from
    the small within-cluster spacings (a cleaner global-grouping decision).
    """
    _, within, between = _layout(scene)
    if not within or not between:
        return 0.0
    return float(min(between) - max(within))


def is_quarantined(scene: dict) -> bool:
    """Quarantine rows where the gap populations are not cleanly separated."""
    m = margin(scene)
    return m < QUARANTINE_CLEARANCE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The scene is injected; no latent aliases are declared."""
    return []
