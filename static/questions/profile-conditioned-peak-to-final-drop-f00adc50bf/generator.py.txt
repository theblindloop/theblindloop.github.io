"""Deterministic renderer and analytic arm for peak_to_final_drop."""

from __future__ import annotations

import copy
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 720, 480
BASE_PALETTE = (
    ("red", (216, 55, 67)),
    ("blue", (42, 105, 205)),
    ("green", (31, 151, 88)),
    ("purple", (139, 73, 184)),
)
QUARANTINE_GAP = 12.0


def _drop(item: dict[str, Any]) -> float:
    ys = [float(point[1]) for point in item["points"]]
    # Image y increases downward, so this is peak value minus final value.
    return ys[-1] - min(ys)


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene with anti-shortcut constraints."""
    rng = random.Random(int(seed))
    wanted_winner = int(seed) % len(BASE_PALETTE)

    for _ in range(4000):
        x_start = rng.randint(112, 128)
        x_step = rng.randint(116, 122)
        horizontal_shift = rng.randint(-5, 5)
        vertical_shift = rng.randint(-8, 8)
        xs = [x_start + horizontal_shift + index * x_step for index in range(5)]
        lanes = [92 + vertical_shift, 176 + vertical_shift, 260 + vertical_shift, 344 + vertical_shift]

        raw = [[] for _ in range(4)]
        for x in xs:
            order = list(range(4))
            rng.shuffle(order)
            for raw_index, lane_index in enumerate(order):
                raw[raw_index].append([x, lanes[lane_index] + rng.randint(-8, 8)])

        drops = [points[-1][1] - min(point[1] for point in points) for points in raw]
        ranking = sorted(range(4), key=lambda index: drops[index], reverse=True)
        winner, runner = ranking[:2]
        ranges = [max(point[1] for point in points) - min(point[1] for point in points) for points in raw]
        finals = [points[-1][1] for points in raw]
        peaks = [min(point[1] for point in points) for points in raw]

        # The answer must require the peak/final composition, not endpoint,
        # highest-peak, or full-range shortcuts.
        if drops[winner] < 120 or drops[winner] - drops[runner] < 42:
            continue
        if raw[winner][-1][1] == min(point[1] for point in raw[winner]):
            continue
        if not any(finals[index] > finals[winner] for index in range(4) if index != winner):
            continue
        if not any(peaks[index] < peaks[winner] for index in range(4) if index != winner):
            continue
        if not any(ranges[index] > ranges[winner] for index in range(4) if index != winner):
            continue

        remaining_shapes = [index for index in range(4) if index != winner]
        remaining_labels = [index for index in range(4) if index != wanted_winner]
        rng.shuffle(remaining_shapes)
        assignment = {wanted_winner: winner}
        assignment.update(zip(remaining_labels, remaining_shapes))

        shade_shift = rng.randint(-7, 7)
        series = []
        for label_index, (name, base_rgb) in enumerate(BASE_PALETTE):
            rgb = [max(0, min(255, channel + shade_shift)) for channel in base_rgb]
            series.append(
                {
                    "name": name,
                    "rgb": rgb,
                    "points": copy.deepcopy(raw[assignment[label_index]]),
                }
            )
        return {"series": series}
    raise RuntimeError("could not construct a separated scene")


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (247, 248, 246))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    draw.rectangle((62, 42, 675, 412), fill=(255, 255, 255), outline=(48, 53, 58), width=2)
    all_points = [point for item in scene["series"] for point in item["points"]]
    checkpoint_xs = sorted({int(point[0]) for point in all_points})
    for number, x in enumerate(checkpoint_xs, start=1):
        for y in range(48, 407, 14):
            draw.line((x, y, x, min(y + 6, 407)), fill=(218, 220, 222), width=1)
        draw.text((x - 3, 425), str(number), fill=(45, 48, 52), font=font)
    draw.text((69, 18), "Five ordered checkpoints", fill=(35, 38, 42), font=font)
    draw.text((514, 18), "higher value", fill=(60, 64, 68), font=font)
    draw.line((659, 31, 659, 14), fill=(60, 64, 68), width=2)
    draw.polygon(((655, 18), (659, 12), (663, 18)), fill=(60, 64, 68))

    for item in scene["series"]:
        rgb = tuple(int(channel) for channel in item["rgb"])
        pale = tuple(int(round(0.22 * channel + 0.78 * 255)) for channel in rgb)
        points = [(int(point[0]), int(point[1])) for point in item["points"]]
        draw.line(points, fill=pale, width=3)
        for x, y in points:
            draw.ellipse((x - 8, y - 8, x + 8, y + 8), fill=rgb, outline=(255, 255, 255), width=2)
    return image


def analytic_gold(scene: dict) -> str:
    return str(max(scene["series"], key=_drop)["name"])


def margin(scene: dict) -> float:
    values = sorted((_drop(item) for item in scene["series"]), reverse=True)
    return float(values[0] - values[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
