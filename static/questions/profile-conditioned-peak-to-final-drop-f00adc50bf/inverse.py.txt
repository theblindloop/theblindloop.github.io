"""Independent image-only inverse arm for peak_to_final_drop."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

BASE_PALETTE = (
    ("red", (216, 55, 67)),
    ("blue", (42, 105, 205)),
    ("green", (31, 151, 88)),
    ("purple", (139, 73, 184)),
)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        pixels = []
        while queue:
            cy, cx = queue.popleft()
            pixels.append((cy, cx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = cy + dy, cx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(pixels) >= 80:
            groups.append(np.asarray(pixels, dtype=float))
    return groups


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    drops: dict[str, float] = {}
    for name, rgb in BASE_PALETTE:
        delta = arr - np.asarray(rgb, dtype=np.int32)
        mask = np.sum(delta * delta, axis=2) <= 30 * 30
        groups = _components(mask)
        if len(groups) != 5:
            raise ValueError(f"expected five isolated {name} markers, found {len(groups)}")
        centers = sorted(
            (float(group[:, 1].mean()), float(group[:, 0].mean()))
            for group in groups
        )
        if min(centers[index + 1][0] - centers[index][0] for index in range(4)) < 60:
            raise ValueError(f"{name} checkpoint markers are not horizontally separated")
        ys = [center[1] for center in centers]
        drops[name] = ys[-1] - min(ys)
    ranked = sorted(drops, key=drops.get, reverse=True)
    if drops[ranked[0]] - drops[ranked[1]] < 6.0:
        raise ValueError("pixel decision is ambiguous")
    return ranked[0]
