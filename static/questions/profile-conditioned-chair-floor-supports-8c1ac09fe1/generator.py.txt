"""Deterministic Pillow renderer for the chair-floor-support world."""

from __future__ import annotations

import copy
import random
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 512, 448

PALETTES = (
    {"bg": (250, 246, 235), "wood": (205, 132, 60), "wood_dark": (130, 73, 30),
     "blue": (36, 103, 184), "floor": (105, 108, 112), "shadow": (226, 218, 201)},
    {"bg": (248, 244, 232), "wood": (222, 153, 75), "wood_dark": (121, 70, 35),
     "blue": (30, 116, 171), "floor": (96, 102, 108), "shadow": (224, 216, 199)},
    {"bg": (252, 247, 238), "wood": (190, 119, 54), "wood_dark": (112, 64, 29),
     "blue": (45, 92, 178), "floor": (110, 110, 114), "shadow": (229, 220, 204)},
)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    count = 2 + (int(seed) % 5)
    canvas_width, canvas_height = WIDTH, HEIGHT
    seat_width = rng.randint(286, 326)
    seat_height = rng.randint(35, 43)
    chair_x = rng.randint(86, 512 - 86 - seat_width)
    seat_y = rng.randint(211, 231)
    floor_y = rng.randint(383, 398)
    leg_height = floor_y - (seat_y + seat_height)

    # Near-equal total target ink breaks the trivial "more blue pixels" cue.
    target_run_sum = rng.randint(67, 75)
    support_width = max(9, target_run_sum // count)
    usable_left = chair_x + 19
    usable_right = chair_x + seat_width - 19 - support_width
    if count == 1:
        support_xs = [(usable_left + usable_right) // 2]
    else:
        step = (usable_right - usable_left) / (count - 1)
        support_xs = [round(usable_left + i * step) for i in range(count)]

    return {
        "support_count": count,
        "support_xs": support_xs,
        "chair_x": chair_x,
        "seat_y": seat_y,
        "seat_width": seat_width,
        "seat_height": seat_height,
        "floor_y": floor_y,
        "support_width": support_width,
        "palette_id": rng.randrange(len(PALETTES)),
        "canvas_width": canvas_width,
        "canvas_height": canvas_height,
    }


def _geometry(scene: dict) -> tuple[int, int, int, int, int, int, list[int]]:
    return (
        int(scene["chair_x"]), int(scene["seat_y"]), int(scene["seat_width"]),
        int(scene["seat_height"]), int(scene["floor_y"]), int(scene["support_width"]),
        [int(x) for x in scene["support_xs"]],
    )


def render(scene: dict) -> Image.Image:
    width = int(scene["canvas_width"])
    height = int(scene["canvas_height"])
    palette = PALETTES[int(scene["palette_id"]) % len(PALETTES)]
    chair_x, seat_y, seat_w, seat_h, floor_y, leg_w, support_xs = _geometry(scene)
    count = int(scene["support_count"])
    if count != len(support_xs):
        raise ValueError("support_count must equal len(support_xs)")

    image = Image.new("RGB", (width, height), palette["bg"])
    draw = ImageDraw.Draw(image)

    # Ground shadow and explicit floor boundary.
    draw.ellipse((chair_x - 18, floor_y + 5, chair_x + seat_w + 18, floor_y + 24),
                 fill=palette["shadow"])
    draw.rectangle((52, floor_y, width - 53, floor_y + 5), fill=palette["floor"])

    # Target components. Reversing drawing order is pixel-identical because they
    # are opaque, separated, and have the same flat fill.
    seat_bottom = seat_y + seat_h
    for x in support_xs:
        draw.rectangle((x, seat_bottom - 1, x + leg_w - 1, floor_y), fill=palette["blue"])
        # Small feet remain the same connected component and make the chair legible.
        foot_extra = max(2, min(5, leg_w // 4))
        draw.rectangle((x - foot_extra, floor_y - 5, x + leg_w - 1 + foot_extra, floor_y),
                       fill=palette["blue"])

    # Seat and back are drawn after supports, leaving unambiguous blue/wood contact.
    draw.rounded_rectangle((chair_x, seat_y, chair_x + seat_w, seat_bottom), radius=8,
                           fill=palette["wood"], outline=palette["wood_dark"], width=4)
    back_left = chair_x + 20
    back_right = chair_x + seat_w - 20
    back_top = seat_y - 139
    draw.rounded_rectangle((back_left, back_top, back_right, seat_y + 4), radius=15,
                           fill=palette["wood"], outline=palette["wood_dark"], width=5)
    # Pale inset makes the back read as furniture without adding blue distractors.
    draw.rounded_rectangle((back_left + 24, back_top + 24, back_right - 24, seat_y - 23),
                           radius=11, fill=palette["bg"], outline=palette["wood_dark"], width=4)
    # A short wood brace visually separates the back opening from the seat.
    draw.rectangle((back_left + 18, seat_y - 23, back_right - 18, seat_y - 16),
                   fill=palette["wood_dark"])
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["support_count"])


def margin(scene: dict) -> float:
    _, _, _, _, _, support_width, support_xs = _geometry(scene)
    gaps = [b - (a + support_width) for a, b in zip(sorted(support_xs), sorted(support_xs)[1:])]
    separation = min(gaps) if gaps else 999.0
    return float(min(support_width, separation))


def is_quarantined(scene: dict) -> bool:
    chair_x, seat_y, seat_w, seat_h, floor_y, support_width, support_xs = _geometry(scene)
    if int(scene["support_count"]) != len(support_xs) or support_width < 8:
        return True
    if floor_y - (seat_y + seat_h) < 105:
        return True
    if min(support_xs) < chair_x or max(support_xs) + support_width > chair_x + seat_w:
        return True
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["support_xs"] = list(reversed(twin["support_xs"]))
    return [("reverse_support_drawing_order", twin)]
