"""Scene-free inverse arm using only RGB pixels from the final PNG."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image


def _blue_mask(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    r, g, b = arr[:, :, 0], arr[:, :, 1], arr[:, :, 2]
    # All renderer palettes use a saturated blue; wood, gray, shadow, and cream
    # are rejected by both channel differences.
    return (b - r > 55) & (b - g > 35) & (b > 130)


def _component_boxes(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    boxes: list[tuple[int, int, int, int, int]] = []
    for y0, x0 in zip(*np.nonzero(mask & ~seen)):
        if seen[y0, x0]:
            continue
        todo = deque([(int(y0), int(x0))])
        seen[y0, x0] = True
        minx = maxx = int(x0)
        miny = maxy = int(y0)
        area = 0
        while todo:
            y, x = todo.popleft()
            area += 1
            minx, maxx = min(minx, x), max(maxx, x)
            miny, maxy = min(miny, y), max(maxy, y)
            for yy, xx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= yy < h and 0 <= xx < w and mask[yy, xx] and not seen[yy, xx]:
                    seen[yy, xx] = True
                    todo.append((yy, xx))
        boxes.append((minx, miny, maxx, maxy, area))
    return boxes


def decision_from_image(image: Image.Image) -> str:
    boxes = _component_boxes(_blue_mask(image))
    # A qualifying component spans the seat-to-floor interval. This threshold is
    # deliberately far below the rendered 108+ px supports and far above specks.
    supports = [box for box in boxes if box[3] - box[1] + 1 >= 100 and box[4] >= 900]
    if not 2 <= len(supports) <= 6:
        raise ValueError(f"expected 2..6 separated seat-to-floor blue supports, found {len(supports)}")
    bottoms = [box[3] for box in supports]
    tops = [box[1] for box in supports]
    if max(bottoms) - min(bottoms) > 2 or max(tops) - min(tops) > 2:
        raise ValueError("blue supports do not share visible seat and floor contacts")
    return str(len(supports))
