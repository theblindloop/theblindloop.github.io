"""Independent pixel-only inverse for half-turn dot-field symmetry."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


_COLORS = ("red", "blue", "green")
_PALETTE = {
    "red": np.asarray((214, 62, 70), dtype=np.int16),
    "blue": np.asarray((52, 100, 208), dtype=np.int16),
    "green": np.asarray((44, 156, 96), dtype=np.int16),
}
_DOT_COUNT = 8


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Return connected pixel components without relying on renderer code."""

    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    components: list[np.ndarray] = []
    for row, column in zip(*np.nonzero(mask)):
        if visited[row, column]:
            continue
        stack = [(int(row), int(column))]
        visited[row, column] = True
        pixels: list[tuple[int, int]] = []
        while stack:
            current_row, current_column = stack.pop()
            pixels.append((current_row, current_column))
            for delta_row in (-1, 0, 1):
                for delta_column in (-1, 0, 1):
                    if not (delta_row or delta_column):
                        continue
                    next_row = current_row + delta_row
                    next_column = current_column + delta_column
                    if (
                        0 <= next_row < height
                        and 0 <= next_column < width
                        and mask[next_row, next_column]
                        and not visited[next_row, next_column]
                    ):
                        visited[next_row, next_column] = True
                        stack.append((next_row, next_column))
        if 35 <= len(pixels) <= 600:
            components.append(np.asarray(pixels, dtype=np.float64))
    return components


def _dot_centers(rgb: np.ndarray, target: np.ndarray) -> np.ndarray:
    distance = np.max(np.abs(rgb.astype(np.int16) - target[None, None, :]), axis=2)
    # The solid cores survive the renderer's antialiasing and avoid neutral
    # background pixels; no latent coordinates are consulted.
    components = _components(distance <= 18)
    centers = []
    for component in components:
        row, column = np.mean(component, axis=0)
        centers.append((column, row))
    return np.asarray(centers, dtype=np.float64)


def _half_turn_error(points: np.ndarray) -> float:
    if points.shape != (_DOT_COUNT, 2):
        return 1.0e6
    center = np.mean(points, axis=0)
    rotated = 2.0 * center[None, :] - points
    difference = rotated[:, None, :] - points[None, :, :]
    distances = np.sqrt(np.sum(difference * difference, axis=2))
    np.fill_diagonal(distances, np.inf)
    return float(np.mean(np.min(distances, axis=1)))


def decision_from_image(image: Image.Image) -> str:
    """Choose a color using only the supplied RGB raster."""

    if not isinstance(image, Image.Image):
        raise TypeError("decision_from_image expects a PIL image")
    rgb = np.asarray(image.convert("RGB"))
    scores = []
    for color in _COLORS:
        points = _dot_centers(rgb, _PALETTE[color])
        scores.append((_half_turn_error(points), color))
    best = min(scores, key=lambda item: (item[0], _COLORS.index(item[1])))
    if not math.isfinite(best[0]) or best[0] >= 1.0e5:
        return "ambiguous"
    return best[1]
