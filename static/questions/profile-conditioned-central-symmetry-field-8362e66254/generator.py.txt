"""Deterministic raster world for half-turn symmetry in dot fields."""

from __future__ import annotations

import copy
import math
from typing import Iterable

import numpy as np
from PIL import Image, ImageDraw


RENDERER_VERSION = "central_symmetry_field-0.1.0"
CANVAS = (640, 440)
BACKGROUND = (250, 248, 244)
PALETTE = {
    "red": (214, 62, 70),
    "blue": (52, 100, 208),
    "green": (44, 156, 96),
}
COLORS = ("red", "blue", "green")
DOT_COUNT = 8
SUPERSAMPLE = 4
QUARANTINE_MARGIN = 6.0


def _as_points(points: Iterable[Iterable[float]]) -> np.ndarray:
    values = np.asarray(list(points), dtype=float)
    if values.shape != (DOT_COUNT, 2):
        return np.empty((0, 2), dtype=float)
    return values


def half_turn_error(points: Iterable[Iterable[float]]) -> float:
    """Mean distance from each point to its nearest distinct half-turn mate."""

    values = _as_points(points)
    if len(values) != DOT_COUNT:
        return 1.0e6
    center = np.mean(values, axis=0)
    rotated = 2.0 * center[None, :] - values
    differences = rotated[:, None, :] - values[None, :, :]
    distances = np.sqrt(np.sum(differences * differences, axis=2))
    np.fill_diagonal(distances, np.inf)
    nearest = np.min(distances, axis=1)
    return float(np.mean(nearest))


def _rotate(points: np.ndarray, angle: float) -> np.ndarray:
    cosine = math.cos(float(angle))
    sine = math.sin(float(angle))
    matrix = np.asarray([[cosine, -sine], [sine, cosine]], dtype=float)
    return points @ matrix.T


def _minimum_spacing(points: np.ndarray) -> float:
    if len(points) < 2:
        return 1.0e6
    differences = points[:, None, :] - points[None, :, :]
    distances = np.sqrt(np.sum(differences * differences, axis=2))
    distances += np.eye(len(points)) * 1.0e6
    return float(np.min(distances))


def _true_local(rng: np.random.Generator) -> tuple[np.ndarray, np.ndarray]:
    """Sample four vectors and their exact opposites, with matched radii."""

    for _ in range(500):
        radii = rng.uniform(42.0, 94.0, size=4)
        angles = rng.uniform(-math.pi, math.pi, size=4)
        base = np.column_stack((radii * np.cos(angles), radii * np.sin(angles)))
        points = np.concatenate((base, -base), axis=0)
        if _minimum_spacing(points) >= 25.0 and float(np.max(np.abs(points))) <= 108.0:
            return points, np.linalg.norm(points, axis=1)
    raise RuntimeError("could not sample separated half-turn pairs")


def _decoy_local(
    radii: np.ndarray,
    rng: np.random.Generator,
    *,
    near: bool,
    template: np.ndarray,
) -> np.ndarray:
    """Create a non-symmetric field with the true field's radial-size nuisance."""

    if near:
        points = np.array(template, dtype=float, copy=True)
        index = int(rng.integers(0, DOT_COUNT))
        direction = points[index]
        length = max(float(np.linalg.norm(direction)), 1.0)
        points[index] += direction / length * float(rng.uniform(3.8, 5.2))
        points -= np.mean(points, axis=0, keepdims=True)
        if _minimum_spacing(points) >= 25.0:
            return points

    for _ in range(300):
        # A jittered octagonal scaffold guarantees separated components even
        # when all eight radii happen to be nearly equal.  The independent
        # angular jitter breaks the opposite-pair relation without introducing
        # a fixed orientation or a local answer cue.
        phase = float(rng.uniform(-math.pi, math.pi))
        angles = (
            phase
            + np.arange(DOT_COUNT, dtype=float) * (2.0 * math.pi / DOT_COUNT)
            + rng.normal(0.0, 0.24, size=DOT_COUNT)
        )
        assigned_radii = radii[rng.permutation(DOT_COUNT)]
        points = np.column_stack(
            (assigned_radii * np.cos(angles), assigned_radii * np.sin(angles))
        )
        points -= np.mean(points, axis=0, keepdims=True)
        if _minimum_spacing(points) >= 25.0 and float(np.max(np.abs(points))) <= 112.0:
            return points
    raise RuntimeError("could not sample a separated decoy field")


def sample_scene(seed: int) -> dict:
    """Sample one balanced, randomized three-field scene."""

    rng = np.random.default_rng(int(seed))
    true_index = int(rng.integers(0, len(COLORS)))
    near_boundary = int(seed) % 17 == 0
    base, radii = _true_local(rng)

    # Keep regenerating the decoys until the declared answer has a stable
    # whole-set gap.  A small deterministic subset is intentionally near the
    # boundary and is retained for explicit quarantine coverage.
    for _ in range(500):
        decoys = [
            _decoy_local(
                radii,
                rng,
                near=near_boundary and decoy_index == 0,
                template=base,
            )
            for decoy_index in range(2)
        ]
        local_by_color: list[np.ndarray] = []
        decoy_index = 0
        for color_index in range(len(COLORS)):
            if color_index == true_index:
                local_by_color.append(base.copy())
            else:
                local_by_color.append(decoys[decoy_index])
                decoy_index += 1
        errors = [half_turn_error(points) for points in local_by_color]
        ordered = sorted(range(len(errors)), key=lambda index: (errors[index], index))
        gap = errors[ordered[1]] - errors[ordered[0]]
        if ordered[0] == true_index and (near_boundary or gap >= 10.0):
            break
    else:
        raise RuntimeError("could not sample a high-margin central-symmetry scene")

    slots = np.asarray(
        [[130.0, 220.0], [320.0, 120.0], [510.0, 255.0]],
        dtype=float,
    )
    slots = slots[rng.permutation(len(slots))]
    fields = []
    for color_index, color in enumerate(COLORS):
        angle = float(rng.uniform(-math.pi, math.pi))
        center = slots[color_index] + rng.normal(0.0, 8.0, size=2)
        points = _rotate(local_by_color[color_index], angle) + center[None, :]
        order = rng.permutation(DOT_COUNT)
        fields.append(
            {
                "color": color,
                "points": [
                    [round(float(x), 4), round(float(y), 4)]
                    for x, y in points[order]
                ],
            }
        )

    return {
        "canvas": [CANVAS[0], CANVAS[1]],
        "background": list(BACKGROUND),
        "dot_radius": round(float(rng.uniform(7.0, 8.5)), 4),
        "fields": fields,
    }


def render(scene: dict) -> Image.Image:
    """Render only the visible RGB evidence in a deterministic supersampled pass."""

    width, height = (int(value) for value in scene["canvas"])
    background = tuple(int(value) for value in scene["background"])
    image = Image.new(
        "RGB", (width * SUPERSAMPLE, height * SUPERSAMPLE), background
    )
    draw = ImageDraw.Draw(image)
    radius = float(scene["dot_radius"])
    for field in scene["fields"]:
        ink = PALETTE[str(field["color"])]
        for x, y in field["points"]:
            cx = float(x) * SUPERSAMPLE
            cy = float(y) * SUPERSAMPLE
            rr = radius * SUPERSAMPLE
            draw.ellipse(
                (
                    round(cx - rr),
                    round(cy - rr),
                    round(cx + rr),
                    round(cy + rr),
                ),
                fill=ink,
            )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    scores = [
        (half_turn_error(field["points"]), str(field["color"]))
        for field in scene["fields"]
    ]
    return min(scores, key=lambda item: (item[0], COLORS.index(item[1])))[1]


def margin(scene: dict) -> float:
    scores = sorted(half_turn_error(field["points"]) for field in scene["fields"])
    if len(scores) < 2:
        return 0.0
    return round(float(scores[1] - scores[0]), 8)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse same-color draw order, a nontrivial pixel-identical alias."""

    twin = copy.deepcopy(scene)
    for field in twin["fields"]:
        field["points"] = list(reversed(field["points"]))
    return [("reverse_same_colour_point_order", twin)]
