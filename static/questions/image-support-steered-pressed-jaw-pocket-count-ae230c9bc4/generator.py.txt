"""Deterministic renderer for the pressed-jaw sealed-chamber world.

Two disconnected dark jaws face each other across a white channel.  Each jaw is
an upright bar carrying a row of straight fingers, and the fingers meet in facing
pairs at the same heights.  Sliding the right jaw straight left until the two
jaws first touch turns every pair that ties for the smallest drawn fingertip gap
into a wall across the channel and leaves every other pair as a passage.  The
decision is the number of white chambers fenced between consecutive walls, which
is one fewer than the number of walls.

Neither jaw alone determines any gap: a pair's gap is the channel span minus the
sum of its two reaches, and the split of a pair's reach total between the left
and the right finger is sampled independently of the answer.  The outermost pair
at each end is always flush, so at least one chamber always forms.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS_W = 940
CANVAS_H = 640
MARGIN = 40

INK = (18, 18, 20)
BACKGROUND = (255, 255, 255)

DECOR_COLORS = (
    (214, 40, 40),
    (40, 90, 220),
    (30, 150, 60),
    (240, 150, 20),
    (150, 60, 200),
    (20, 170, 170),
    (150, 150, 155),
)
DECOR_KINDS = ("square", "ring", "dot", "bar")

MIN_REACH = 20
QUARANTINE_WIDTH = 22


# --------------------------------------------------------------------------
# latent scene sampling
# --------------------------------------------------------------------------
def _pockets_from_contacts(contacts: list[bool]) -> int:
    """Each flush finger pair walls the channel; chambers sit between walls."""
    walls = sum(1 for c in contacts if c)
    return max(0, walls - 1)


def _sample_contacts(rng: random.Random, n: int, target: int) -> list[bool] | None:
    """The outermost pair on each end is always flush, so chambers never vanish.

    `target` chambers therefore need `target - 1` further flush pairs among the
    interior bands, and at least one interior band must stay open.
    """
    interior = target - 1
    if interior < 0 or interior > n - 3:
        return None
    slots = list(range(1, n - 1))
    rng.shuffle(slots)
    chosen = set(slots[:interior])
    return [i in chosen or i in (0, n - 1) for i in range(n)]


def _place_decor(
    rng: random.Random, blocked: list[tuple[int, int, int, int]], count: int
) -> list[dict[str, Any]]:
    decor: list[dict[str, Any]] = []
    tries = 0
    while len(decor) < count and tries < 400:
        tries += 1
        size = rng.randint(16, 34)
        x = rng.randint(12, CANVAS_W - size - 12)
        y = rng.randint(12, CANVAS_H - size - 12)
        kind = rng.choice(DECOR_KINDS)
        w = size if kind != "bar" else size * 2
        h = size
        box = (x - 10, y - 10, x + w + 10, y + h + 10)
        if any(
            box[0] < bx1 and bx0 < box[2] and box[1] < by1 and by0 < box[3]
            for (bx0, by0, bx1, by1) in blocked
        ):
            continue
        if any(
            box[0] < d["x"] + d["w"] + 8
            and d["x"] - 8 < box[2]
            and box[1] < d["y"] + d["h"] + 8
            and d["y"] - 8 < box[3]
            for d in decor
        ):
            continue
        decor.append(
            {
                "kind": kind,
                "x": x,
                "y": y,
                "w": w,
                "h": h,
                "color": list(rng.choice(DECOR_COLORS)),
            }
        )
    return decor


def sample_scene(seed: int) -> dict:
    rng = random.Random(0xC0FFEE ^ (seed * 2654435761 % (1 << 32)))
    for _ in range(600):
        n = rng.choice([5, 6, 6, 7])
        band_h = rng.choice([54, 60, 66]) if n < 7 else rng.choice([54, 60])
        finger_h = band_h - 30
        target = rng.choice([1, 2, 3])
        contacts = _sample_contacts(rng, n, target)
        if contacts is None:
            continue
        reach = rng.randint(90, 130)
        # smallest open strip width; occasionally near the resolution boundary
        if rng.random() < 0.14:
            base = rng.randint(10, QUARANTINE_WIDTH - 2)
        else:
            base = rng.randint(24, 46)
        top = reach - 2 * MIN_REACH
        if base > top:
            continue
        recessed_idx = [i for i, c in enumerate(contacts) if not c]
        widths = {recessed_idx[0]: base}
        for i in recessed_idx[1:]:
            widths[i] = rng.randint(base, top)
        rng.shuffle(recessed_idx)
        # guarantee the declared minimum is actually attained
        widths[recessed_idx[0]] = base

        teeth = []
        ok = True
        for i in range(n):
            width = 0 if contacts[i] else widths[i]
            t = reach - width
            if t < 2 * MIN_REACH:
                ok = False
                break
            a = rng.randint(MIN_REACH, t - MIN_REACH)
            teeth.append([a, t - a])
        if not ok:
            continue

        clearance = reach + rng.randint(110, 170)
        span = reach + clearance
        bb_l = rng.choice([56, 64, 72])
        bb_r = rng.choice([56, 64, 72])
        total_w = bb_l + span + bb_r
        total_h = n * band_h
        if total_w > CANVAS_W - 2 * MARGIN or total_h > CANVAS_H - 2 * MARGIN:
            continue
        x0 = rng.randint(MARGIN, CANVAS_W - total_w - MARGIN)
        y0 = rng.randint(MARGIN, CANVAS_H - total_h - MARGIN)

        jaw = {
            "x0": x0,
            "y0": y0,
            "bb_l": bb_l,
            "bb_r": bb_r,
            "band_h": band_h,
            "finger_h": finger_h,
            "bands": n,
            "span": span,
        }
        blocked = [
            (x0 - 6, y0 - 6, x0 + total_w + 6, y0 + total_h + 6)
        ]
        decor = _place_decor(rng, blocked, rng.randint(3, 7))
        return {"jaw": jaw, "teeth": teeth, "decor": decor}
    raise RuntimeError("scene sampling failed")


# --------------------------------------------------------------------------
# geometry helpers shared by the renderer and the analytic gold
# --------------------------------------------------------------------------
def _jaw_rects(scene: dict) -> list[tuple[int, int, int, int]]:
    """Two upright bars plus one pair of facing fingers per band."""
    jaw = scene["jaw"]
    x0, y0 = jaw["x0"], jaw["y0"]
    bb_l, bb_r = jaw["bb_l"], jaw["bb_r"]
    band_h, n, span = jaw["band_h"], jaw["bands"], jaw["span"]
    finger_h = jaw["finger_h"]
    left_face = x0 + bb_l
    right_face = left_face + span
    rects = [
        (x0, y0, left_face, y0 + n * band_h),
        (right_face, y0, right_face + bb_r, y0 + n * band_h),
    ]
    pad = (band_h - finger_h) // 2
    for i, (a, b) in enumerate(scene["teeth"]):
        top = y0 + i * band_h + pad
        bottom = top + finger_h
        rects.append((left_face, top, left_face + a, bottom))
        rects.append((right_face - b, top, right_face, bottom))
    return rects


def _strip_widths(scene: dict) -> list[int]:
    """Gap each pair keeps after closing: its own reach total below the deepest."""
    totals = [a + b for (a, b) in scene["teeth"]]
    deepest = max(totals)
    return [deepest - t for t in totals]


def _contacts(scene: dict) -> list[bool]:
    return [w == 0 for w in _strip_widths(scene)]


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (CANVAS_W, CANVAS_H), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for item in scene["decor"]:
        color = tuple(item["color"])
        x, y, w, h = item["x"], item["y"], item["w"], item["h"]
        if item["kind"] == "square":
            draw.rectangle([x, y, x + w, y + h], fill=color)
        elif item["kind"] == "bar":
            draw.rectangle([x, y + h // 3, x + w, y + 2 * h // 3], fill=color)
        elif item["kind"] == "ring":
            draw.ellipse([x, y, x + w, y + h], outline=color, width=5)
        else:
            draw.ellipse([x + w // 4, y + h // 4, x + 3 * w // 4, y + 3 * h // 4], fill=color)
    for (rx0, ry0, rx1, ry1) in _jaw_rects(scene):
        draw.rectangle([rx0, ry0, rx1 - 1, ry1 - 1], fill=INK)
    return image


# --------------------------------------------------------------------------
# analytic gold, margin, quarantine, symmetries
# --------------------------------------------------------------------------
def analytic_gold(scene: dict) -> str:
    return str(_pockets_from_contacts(_contacts(scene)))


def margin(scene: dict) -> float:
    widths = [w for w in _strip_widths(scene) if w > 0]
    if not widths:
        return 0.0
    return float(min(widths))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_WIDTH


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases: every scene field is painted at the place it names."""
    return []
