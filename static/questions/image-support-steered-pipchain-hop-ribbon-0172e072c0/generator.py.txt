"""Renderer for the pipchain_hop_ribbon world.

One thin zig-zagging ribbon of pip-marked cells is drawn across a medium sized
band of an otherwise inert canvas.  A solid tab attached to the top edge of
one cell marks where a token starts.  The token then makes exactly three hops:
at each stop it reads the number of pips in the cell it currently occupies and
advances that many cells forward along the ribbon.  The question asks for the
number of pips in the cell where the token finally stops, so the decisive
evidence is the traversed stretch of the ribbon, not any single endpoint.

The renderer owns the latent scene and the analytic gold.  The pixel-only
inverse arm in ``oracle.py`` shares no code with it.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "pipchain_hop_ribbon-renderer-0.1.0"

WIDTH = 840
HEIGHT = 480

CELLS = 14
CELL = 32
HGAP = 12
NECK = 6
BORDER = 2
WOBBLE = 8
TAB_LEN = 12
TAB_HALF = 7
HOPS = 3

BAND_W = CELLS * CELL + (CELLS - 1) * HGAP  # 604
BAND_H = CELL + 2 * WOBBLE  # 48

BG = (247, 246, 242)
INK = (30, 32, 38)
DECO = (168, 168, 172)

FILL_BASE = {
    "red": (226, 96, 86),
    "green": (96, 178, 110),
    "blue": (92, 134, 222),
    "amber": (232, 180, 70),
}
FILL_NAMES = ("red", "green", "blue", "amber")

ANSWERS = ("1", "2", "3", "4")

# Dice-style pip offsets on a 3x3 lattice, in units of the pip offset radius.
PIP_LAYOUT = {
    1: ((0, 0),),
    2: ((-1, -1), (1, 1)),
    3: ((-1, -1), (0, 0), (1, 1)),
    4: ((-1, -1), (1, -1), (-1, 1), (1, 1)),
}

QUARANTINE_MARGIN = 2.6


# --------------------------------------------------------------- chain layout
def _wobble(rng: random.Random) -> list[int]:
    """A small vertical random walk, so the ribbon is a wandering thin route."""
    offsets = [0]
    for _ in range(CELLS - 1):
        step = rng.choice((-4, 0, 4))
        offsets.append(max(-WOBBLE, min(WOBBLE, offsets[-1] + step)))
    return offsets


def _replay(pips: list[int], start: int, length: int) -> int | None:
    """Three data-dependent hops along an ordered chain of ``length`` cells."""
    index = start
    for _ in range(HOPS):
        index += pips[index]
        if index >= length:
            return None
    return index


def _alt_answer(pips_by_chain: list[int], order: list[int], start_chain: int) -> str | None:
    """Replay the same rule under a different assumed ribbon ordering."""
    remap = [pips_by_chain[i] for i in order]
    try:
        start = order.index(start_chain)
    except ValueError:  # pragma: no cover - defensive
        return None
    landed = _replay(remap, start, len(remap))
    return None if landed is None else str(remap[landed])


# --------------------------------------------------------------- scene sample
def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    target = ANSWERS[seed % len(ANSWERS)]

    for _ in range(4000):
        pips = [rng.randint(1, 4) for _ in range(CELLS)]
        start = rng.randint(0, 3)
        landed = _replay(pips, start, CELLS)
        if landed is None or landed < 8:
            continue
        answer = str(pips[landed])
        if answer != target:
            continue
        # Anti-shortcut counterbalancing: the answer must not be readable from
        # the start cell, either ribbon end, the modal pip count, or a replay
        # that walks the ribbon from the wrong end.
        if answer in {str(pips[start]), str(pips[0]), str(pips[CELLS - 1])}:
            continue
        counts = [pips.count(v) for v in (1, 2, 3, 4)]
        top = max(counts)
        if counts.count(top) == 1 and pips.count(int(answer)) == top:
            continue
        reverse = _alt_answer(pips, list(range(CELLS - 1, -1, -1)), start)
        if reverse == answer:
            continue

        pip_offset = rng.choice((7, 8))
        pip_radius = round(rng.uniform(2.8, 3.6), 2)
        if CELL / 2 - BORDER - (pip_offset + pip_radius) < 2.3:
            continue

        fills = [rng.choice(FILL_NAMES) for _ in range(CELLS)]
        # Colour is a pure distractor: forbid the landing cell's colour from
        # being unique, so hue can never single the answer cell out.
        if fills.count(fills[landed]) < 3:
            continue

        x0 = rng.randint(40, WIDTH - BAND_W - 40)
        y0 = rng.randint(80, HEIGHT - BAND_H - 80)
        wobble = _wobble(rng)
        jitter = [
            tuple(rng.randint(-10, 10) for _ in range(3)) for _ in range(len(FILL_NAMES))
        ]
        decor = _sample_decor(rng, x0, y0)
        return {
            "wobble": wobble,
            "pips": pips,
            "start": start,
            "fills": fills,
            "fill_jitter": [list(j) for j in jitter],
            "pip_offset": pip_offset,
            "pip_radius": pip_radius,
            "band_x": x0,
            "band_y": y0,
            "decor": decor,
        }
    raise RuntimeError(f"could not sample a balanced scene for seed {seed}")


def _sample_decor(rng: random.Random, x0: int, y0: int) -> list[dict]:
    """Inert grey marks, always clear of the ribbon's neighbourhood."""
    keep = (x0 - 18, y0 - TAB_LEN - 26, x0 + BAND_W + 18, y0 + BAND_H + 22)
    out: list[dict] = []
    for _ in range(260):
        if len(out) >= 6:
            break
        size = rng.randint(6, 14)
        cx = rng.randint(14, WIDTH - 14)
        cy = rng.randint(14, HEIGHT - 14)
        if keep[0] - size <= cx <= keep[2] + size and keep[1] - size <= cy <= keep[3] + size:
            continue
        out.append({"kind": rng.choice(("square", "disk", "bar")), "cx": cx, "cy": cy, "size": size})
    return out


# --------------------------------------------------------------------- render
def _cell_rect(scene: dict, position: int) -> tuple[int, int, int, int]:
    x = scene["band_x"] + position * (CELL + HGAP)
    y = scene["band_y"] + WOBBLE + scene["wobble"][position]
    return x, y, x + CELL, y + CELL


def _fill_rgb(scene: dict, name: str) -> tuple[int, int, int]:
    base = FILL_BASE[name]
    jit = scene["fill_jitter"][FILL_NAMES.index(name)]
    return tuple(max(0, min(255, base[i] + jit[i])) for i in range(3))


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)

    for mark in scene["decor"]:
        cx, cy, size = mark["cx"], mark["cy"], mark["size"]
        if mark["kind"] == "square":
            draw.rectangle((cx - size, cy - size, cx + size, cy + size), fill=DECO)
        elif mark["kind"] == "disk":
            draw.ellipse((cx - size, cy - size, cx + size, cy + size), fill=DECO)
        else:
            draw.rectangle((cx - size, cy - 3, cx + size, cy + 3), fill=DECO)

    # Bridges first: each link is a horizontal bar spanning the gap between two
    # consecutive cells, so the whole ribbon is one connected strip.
    for position in range(CELLS - 1):
        left = _cell_rect(scene, position)
        right = _cell_rect(scene, position + 1)
        mid_y = (left[1] + left[3] + right[1] + right[3]) // 4
        draw.rectangle(
            (left[2] - 1, mid_y - NECK // 2, right[0] + 1, mid_y + NECK // 2), fill=INK
        )

    # Start tab: a solid triangle welded to the top edge of the start cell.
    rect = _cell_rect(scene, scene["start"])
    tip_x = (rect[0] + rect[2]) // 2
    draw.polygon(
        [
            (tip_x - TAB_HALF, rect[1] - TAB_LEN),
            (tip_x + TAB_HALF, rect[1] - TAB_LEN),
            (tip_x, rect[1] + 1),
        ],
        fill=INK,
    )

    # Cells: ink frame, coloured body, dark pips.
    offset = scene["pip_offset"]
    radius = scene["pip_radius"]
    for position in range(CELLS):
        x0, y0, x1, y1 = _cell_rect(scene, position)
        draw.rectangle((x0, y0, x1, y1), fill=INK)
        draw.rectangle(
            (x0 + BORDER, y0 + BORDER, x1 - BORDER, y1 - BORDER),
            fill=_fill_rgb(scene, scene["fills"][position]),
        )
        cx = (x0 + x1) / 2.0
        cy = (y0 + y1) / 2.0
        for dx, dy in PIP_LAYOUT[scene["pips"][position]]:
            px = cx + dx * offset
            py = cy + dy * offset
            draw.ellipse((px - radius, py - radius, px + radius, py + radius), fill=INK)
    return image


# ----------------------------------------------------------------------- gold
def analytic_gold(scene: dict) -> str:
    landed = _replay(scene["pips"], scene["start"], CELLS)
    if landed is None:
        raise ValueError("scene replays off the end of the ribbon")
    return str(scene["pips"][landed])


def _colour_separation(scene: dict) -> float:
    rgbs = [_fill_rgb(scene, name) for name in FILL_NAMES]
    best = 1e9
    for i in range(len(rgbs)):
        for j in range(i + 1, len(rgbs)):
            best = min(best, max(abs(rgbs[i][k] - rgbs[j][k]) for k in range(3)))
        best = min(best, max(abs(rgbs[i][k] - BG[k]) for k in range(3)))
    return float(best)


def margin(scene: dict) -> float:
    """Composite legibility margin in pixels for the decisive marks."""
    offset = scene["pip_offset"]
    radius = scene["pip_radius"]
    pip_border = CELL / 2.0 - BORDER - (offset + radius)
    pip_gap = 2.0 * offset - 2.0 * radius
    neck_clear = (CELL - NECK) / 2.0
    cell_gap = float(HGAP)  # background gap bridged by one link
    tab_clear = float(TAB_LEN - 2)
    colour = _colour_separation(scene) / 12.0
    return round(min(pip_border, pip_gap, neck_clear, cell_gap, tab_clear, colour), 4)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist: every sampled field is painted (see candidate.json)."""
    return []
