"""Independent image-only inverse for Reciprocal Selector Braid."""

from __future__ import annotations

from PIL import Image, ImageDraw

CANVAS = (1100, 860)
INK = (35, 40, 48)
MUTED = (218, 224, 228)
SHAPE = (18, 126, 111)
BADGE = (62, 48, 96)
PANEL = (255, 255, 255)
COLORS = {"amber": (219, 139, 22), "cyan": (18, 153, 177), "violet": (133, 76, 181)}
ROW_Y = (320, 410, 500, 590)


class OracleError(ValueError):
    pass


def _shape_template(name: str) -> tuple[int, ...]:
    image = Image.new("RGB", (80, 80), PANEL)
    draw, cx, cy = ImageDraw.Draw(image), 40, 40
    if name == "circle":
        draw.ellipse((cx - 26, cy - 26, cx + 26, cy + 26), fill=SHAPE)
    elif name == "square":
        draw.rectangle((cx - 24, cy - 24, cx + 24, cy + 24), fill=SHAPE)
    elif name == "triangle":
        draw.polygon(((cx, cy - 30), (cx - 30, cy + 24), (cx + 30, cy + 24)), fill=SHAPE)
    elif name == "diamond":
        draw.polygon(((cx, cy - 31), (cx - 31, cy), (cx, cy + 31), (cx + 31, cy)), fill=SHAPE)
    return tuple(1 if pixel == SHAPE else 0 for pixel in image.getdata())


def _read_shape(image: Image.Image, cx: int, cy: int) -> str:
    crop = image.crop((cx - 40, cy - 40, cx + 40, cy + 40))
    observed = tuple(1 if pixel == SHAPE else 0 for pixel in crop.getdata())
    scores = {
        name: sum(a != b for a, b in zip(observed, _shape_template(name)))
        for name in ("circle", "square", "triangle", "diamond")
    }
    best = min(scores, key=scores.get)
    if scores[best] != 0 or sum(observed) < 1500:
        raise OracleError("damaged or unknown shape")
    return best


def _read_badge(image: Image.Image, cx: int, cy: int) -> tuple[int, ...]:
    px = image.load()
    if px[cx - 51, cy] != INK or px[cx + 51, cy] != INK:
        raise OracleError("badge border missing")
    bits = []
    for i in range(6):
        col, row = i % 3, i // 3
        x0, y0 = cx - 37 + col * 26, cy - 25 + row * 31
        samples = [px[x, y] for x in range(x0 + 3, x0 + 18, 5)
                   for y in range(y0 + 3, y0 + 18, 5)]
        if all(pixel == BADGE for pixel in samples):
            bits.append(1)
        elif all(pixel == MUTED for pixel in samples):
            bits.append(0)
        else:
            raise OracleError("badge cell damaged")
    if sum(bits) != 3:
        raise OracleError("invalid badge")
    return tuple(bits)


def _selector_template(side: str) -> tuple[int, ...]:
    image = Image.new("RGB", (120, 60), PANEL)
    draw, cx, cy = ImageDraw.Draw(image), 60, 30
    if side == "left":
        draw.line((cx + 38, cy, cx - 30, cy), fill=INK, width=7)
        draw.polygon(((cx - 48, cy), (cx - 24, cy - 16), (cx - 24, cy + 16)), fill=INK)
    else:
        draw.line((cx - 38, cy, cx + 30, cy), fill=INK, width=7)
        draw.polygon(((cx + 48, cy), (cx + 24, cy - 16), (cx + 24, cy + 16)), fill=INK)
    return tuple(1 if pixel == INK else 0 for pixel in image.getdata())


def _read_selector(image: Image.Image, cx: int, cy: int) -> str:
    crop = image.crop((cx - 60, cy - 30, cx + 60, cy + 30))
    observed = tuple(1 if pixel == INK else 0 for pixel in crop.getdata())
    scores = {side: sum(a != b for a, b in zip(observed, _selector_template(side)))
              for side in ("left", "right")}
    best = min(scores, key=scores.get)
    if scores[best] != 0:
        raise OracleError("damaged selector")
    return best


def decision_from_image(image: Image.Image) -> str:
    image = image.convert("RGB")
    if image.size != CANVAS:
        raise OracleError("wrong canvas")
    px = image.load()
    for x, y in ((550, 18), (45, 400), (1055, 400)):
        if px[x, y] != INK:
            raise OracleError("panel boundary missing")

    first_key = _read_shape(image, 400, 128)
    second_key = _read_shape(image, 700, 128)
    if first_key == second_key:
        raise OracleError("query keys must differ")

    records, badges_seen = {}, set()
    for y in ROW_Y:
        key = _read_shape(image, 155, y)
        record = {
            "selector": _read_selector(image, 650, y),
            "left": _read_badge(image, 500, y),
            "right": _read_badge(image, 800, y),
        }
        if key in records:
            raise OracleError("duplicate key")
        records[key] = record
        badges_seen.update((record["left"], record["right"]))
    if len(records) != 4 or len(badges_seen) != 8:
        raise OracleError("duplicate table mark")

    first, second = records[first_key], records[second_key]
    target = (second[first["selector"]], first[second["selector"]])
    matches = []
    for cx in (190, 550, 910):
        border = px[cx, 680]
        names = [name for name, value in COLORS.items() if border == value]
        if len(names) != 1:
            raise OracleError("answer border damaged")
        pair = (_read_badge(image, cx - 65, 775), _read_badge(image, cx + 65, 775))
        if pair == target:
            matches.append(names[0])
    if len(matches) != 1:
        raise OracleError("non-unique reciprocal pair")
    return matches[0]
