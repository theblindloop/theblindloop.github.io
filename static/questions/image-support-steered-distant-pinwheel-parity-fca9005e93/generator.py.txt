"""Deterministic scene sampling and rendering for distant pinwheel parity."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw


PALETTES = (
    ((247, 247, 242), (232, 237, 231), (28, 91, 176), (202, 55, 50)),
    ((250, 246, 238), (235, 239, 246), (40, 103, 184), (211, 66, 48)),
    ((246, 248, 250), (239, 235, 225), (31, 112, 168), (196, 48, 62)),
)


def _majority_votes(rng: random.Random, sign: int) -> list[int]:
    votes = [sign] * 5 + [-sign] * 4
    rng.shuffle(votes)
    return votes


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    background, panel_color, ink_color, pin_color = PALETTES[seed % len(PALETTES)]
    relation_same = (seed % 2) == 0
    left_sign = 1 if rng.random() < 0.5 else -1
    right_sign = left_sign if relation_same else -left_sign
    centers = [[155 + rng.randint(-5, 5), 200 + rng.randint(-10, 10)],
               [645 + rng.randint(-5, 5), 200 + rng.randint(-10, 10)]]
    angles: list[list[float]] = []
    radii: list[list[float]] = []
    sizes: list[list[list[float]]] = []
    for _panel in range(2):
        rotation = rng.uniform(0.0, 2.0 * math.pi)
        angles.append([
            rotation + i * 2.0 * math.pi / 9.0 + rng.uniform(-0.045, 0.045)
            for i in range(9)
        ])
        radii.append([rng.uniform(80.0, 92.0) for _ in range(9)])
        sizes.append([[rng.uniform(30.0, 35.0), rng.uniform(17.0, 21.0)] for _ in range(9)])
    order = list(range(18))
    rng.shuffle(order)
    return {
        "width": 800,
        "height": 400,
        "background": list(background),
        "panel_color": list(panel_color),
        "ink_color": list(ink_color),
        "pin_color": list(pin_color),
        "panel_centers": centers,
        "panel_radius": 145,
        "petal_angles": angles,
        "vote_signs": [_majority_votes(rng, left_sign), _majority_votes(rng, right_sign)],
        "petal_radii": radii,
        "petal_sizes": sizes,
        "draw_order": order,
    }


def _triangle(scene: dict, panel: int, index: int) -> list[tuple[float, float]]:
    cx, cy = scene["panel_centers"][panel]
    theta = float(scene["petal_angles"][panel][index])
    radius = float(scene["petal_radii"][panel][index])
    length, breadth = map(float, scene["petal_sizes"][panel][index])
    vote = int(scene["vote_signs"][panel][index])
    radial = (math.cos(theta), math.sin(theta))
    tangent = (-math.sin(theta) * vote, math.cos(theta) * vote)
    px, py = cx + radius * radial[0], cy + radius * radial[1]
    tip = (px + 0.62 * length * tangent[0], py + 0.62 * length * tangent[1])
    base = (px - 0.48 * length * tangent[0], py - 0.48 * length * tangent[1])
    side = (-tangent[1], tangent[0])
    return [
        tip,
        (base[0] + 0.5 * breadth * side[0], base[1] + 0.5 * breadth * side[1]),
        (base[0] - 0.5 * breadth * side[0], base[1] - 0.5 * breadth * side[1]),
    ]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    radius = int(scene["panel_radius"])
    for cx, cy in scene["panel_centers"]:
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius),
                     fill=tuple(scene["panel_color"]), outline=(178, 182, 180), width=3)
    for code in scene["draw_order"]:
        panel, index = divmod(int(code), 9)
        draw.polygon(_triangle(scene, panel, index), fill=tuple(scene["ink_color"]))
    for cx, cy in scene["panel_centers"]:
        draw.ellipse((cx - 8, cy - 8, cx + 8, cy + 8), fill=tuple(scene["pin_color"]))
        draw.ellipse((cx - 2, cy - 2, cx + 2, cy + 2), fill=(255, 255, 255))
    return image


def _panel_signs(scene: dict) -> tuple[int, int]:
    signs = []
    for votes in scene["vote_signs"]:
        total = sum(int(value) for value in votes)
        signs.append(1 if total > 0 else -1 if total < 0 else 0)
    return signs[0], signs[1]


def analytic_gold(scene: dict) -> str:
    left = sum(int(value) for value in scene["vote_signs"][0])
    right = sum(int(value) for value in scene["vote_signs"][1])
    return "same" if (left > 0) == (right > 0) else "opposite"


def margin(scene: dict) -> float:
    totals = [abs(sum(int(value) for value in votes)) for votes in scene["vote_signs"]]
    return float(min(totals))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 1.0 or any(len(votes) != 9 for votes in scene["vote_signs"])


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["draw_order"] = list(reversed(twin["draw_order"]))
    return [("reverse_nonoverlapping_draw_order", twin)]
