"""Independent PNG-only inverse arm for distant pinwheel parity."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result: list[np.ndarray] = []
    for y0, x0 in np.argwhere(mask):
        if seen[y0, x0]:
            continue
        stack = [(int(y0), int(x0))]
        seen[y0, x0] = True
        points = []
        while stack:
            y, x = stack.pop()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= minimum:
            result.append(np.asarray(points, dtype=np.float64))
    return result


def _center(component: np.ndarray) -> np.ndarray:
    # Return Cartesian x,y even though component rows are y,x.
    return np.asarray([component[:, 1].mean(), component[:, 0].mean()])


def _tip_direction(component: np.ndarray) -> np.ndarray:
    xy = component[:, ::-1]
    center = xy.mean(axis=0)
    centered = xy - center
    covariance = centered.T @ centered / len(centered)
    values, vectors = np.linalg.eigh(covariance)
    axis = vectors[:, int(np.argmax(values))]
    projections = centered @ axis
    # A filled triangle extends farther from its centroid toward its tip.
    positive = float(projections.max())
    negative = float(-projections.min())
    if abs(positive - negative) < 1.2:
        raise ValueError("arrowhead direction is not resolved")
    return axis if positive > negative else -axis


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[:, :, 0], array[:, :, 1], array[:, :, 2]
    ink_mask = (blue > red + 35) & (blue > green + 18) & (blue > 125)
    pin_mask = (red > green + 60) & (red > blue + 55) & (red > 135)
    pins = _components(pin_mask, 70)
    arrows = _components(ink_mask, 120)
    if len(pins) != 2 or len(arrows) != 18:
        raise ValueError(f"expected two pins and eighteen arrowheads, found {len(pins)} and {len(arrows)}")
    pin_centers = sorted((_center(item) for item in pins), key=lambda point: point[0])
    groups: list[list[np.ndarray]] = [[], []]
    for arrow in arrows:
        center = _center(arrow)
        distances = [float(np.linalg.norm(center - pin)) for pin in pin_centers]
        panel = int(np.argmin(distances))
        if distances[panel] > 135.0:
            raise ValueError("arrowhead lies outside either pinwheel")
        groups[panel].append(arrow)
    if [len(group) for group in groups] != [9, 9]:
        raise ValueError("each pinwheel must contain exactly nine arrowheads")
    majorities = []
    for pin, group in zip(pin_centers, groups, strict=True):
        votes = []
        for arrow in group:
            center = _center(arrow)
            radial = center - pin
            direction = _tip_direction(arrow)
            cross = radial[0] * direction[1] - radial[1] * direction[0]
            if abs(cross) < 55.0:
                raise ValueError("arrowhead is insufficiently tangential")
            votes.append(1 if cross > 0 else -1)
        total = sum(votes)
        if total == 0:
            raise ValueError("pinwheel majority is tied")
        majorities.append(1 if total > 0 else -1)
    return "same" if majorities[0] == majorities[1] else "opposite"
