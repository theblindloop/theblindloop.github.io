"""Deterministic renderer for rearranged network correspondence."""

from __future__ import annotations

import copy
import itertools
import math
import random

from PIL import Image, ImageDraw


WIDTH = 900
HEIGHT = 500
SCALE = 2
EDGES = ((0, 1), (0, 2), (2, 3), (0, 4), (4, 5), (5, 6))
COLOR_RGB = {
    "red": (220, 68, 72),
    "green": (48, 166, 92),
    "blue": (62, 116, 218),
}


def _make_layout(rng: random.Random) -> list[list[int]]:
    """Lay out the three unequal arms independently around a center."""
    rotation = rng.uniform(-35.0, 35.0)
    slots = [rotation + value for value in (0.0, 120.0, 240.0)]
    rng.shuffle(slots)
    branches = ((1,), (2, 3), (4, 5, 6))
    points: list[list[int] | None] = [None] * 7
    points[0] = [0, 0]
    for branch, base_angle in zip(branches, slots):
        x = 0.0
        y = 0.0
        bend = rng.uniform(-8.0, 8.0)
        for depth, vertex in enumerate(branch):
            angle = math.radians(base_angle + bend * depth)
            step = rng.randint(52, 60)
            x += step * math.cos(angle)
            y += step * math.sin(angle)
            points[vertex] = [round(x), round(y)]
    return [point for point in points if point is not None]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = int(seed) % 7
    desired = ("red", "green", "blue")[int(seed) % 3]
    majority = ("red", "green", "blue")[(int(seed) // 3) % 3]
    pool = ["red", "red", "green", "green", "blue", "blue", majority]
    pool.remove(desired)
    rng.shuffle(pool)
    right_colors = pool[:target] + [desired] + pool[target:]
    edge_order = list(range(len(EDGES)))
    rng.shuffle(edge_order)
    return {
        "target": target,
        "right_colors": right_colors,
        "left_layout": _make_layout(rng),
        "right_layout": _make_layout(rng),
        "node_radius": rng.choice([17, 18, 19]),
        "edge_order": edge_order,
    }


def _scaled_box(box: tuple[int, int, int, int]) -> tuple[int, int, int, int]:
    return tuple(value * SCALE for value in box)


def _absolute(layout: list[list[int]], center: tuple[int, int]) -> list[tuple[int, int]]:
    return [(center[0] + int(point[0]), center[1] + int(point[1])) for point in layout]


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    positions: list[tuple[int, int]],
    radius: int,
    edge_order: list[int],
    fills: list[tuple[int, int, int]],
    target: int | None,
) -> None:
    scaled = [(x * SCALE, y * SCALE) for x, y in positions]
    for edge_index in edge_order:
        a, b = EDGES[int(edge_index)]
        draw.line(
            [scaled[a], scaled[b]],
            fill=(34, 38, 43),
            width=5 * SCALE,
        )
    outer = (radius + 3) * SCALE
    inner = radius * SCALE
    for index, ((x, y), fill) in enumerate(zip(scaled, fills)):
        draw.ellipse((x - outer, y - outer, x + outer, y + outer), fill=(28, 31, 35))
        draw.ellipse((x - inner, y - inner, x + inner, y + inner), fill=fill)
        if target == index:
            core = 7 * SCALE
            draw.ellipse((x - core, y - core, x + core, y + core), fill=(205, 48, 167))


def render(scene: dict) -> Image.Image:
    canvas = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), (244, 246, 249))
    draw = ImageDraw.Draw(canvas)
    draw.rounded_rectangle(
        _scaled_box((18, 24, 438, 476)),
        radius=18 * SCALE,
        fill=(251, 252, 253),
        outline=(147, 155, 166),
        width=2 * SCALE,
    )
    draw.rounded_rectangle(
        _scaled_box((462, 24, 882, 476)),
        radius=18 * SCALE,
        fill=(251, 252, 253),
        outline=(147, 155, 166),
        width=2 * SCALE,
    )
    # A neutral arrow makes the panel correspondence direction explicit.
    draw.line(_scaled_box((442, 250, 456, 250)), fill=(105, 112, 122), width=3 * SCALE)
    draw.polygon(
        [(456 * SCALE, 250 * SCALE), (449 * SCALE, 245 * SCALE), (449 * SCALE, 255 * SCALE)],
        fill=(105, 112, 122),
    )

    radius = int(scene["node_radius"])
    edge_order = [int(value) for value in scene["edge_order"]]
    left_positions = _absolute(scene["left_layout"], (228, 250))
    right_positions = _absolute(scene["right_layout"], (672, 250))
    _draw_panel(
        draw,
        left_positions,
        radius,
        edge_order,
        [(210, 218, 227)] * 7,
        int(scene["target"]),
    )
    _draw_panel(
        draw,
        right_positions,
        radius,
        edge_order,
        [COLOR_RGB[str(name)] for name in scene["right_colors"]],
        None,
    )
    return canvas.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["right_colors"][int(scene["target"])])


def _assignment_gap() -> float:
    edge_set = {tuple(sorted(edge)) for edge in EDGES}
    scores = []
    for permutation in itertools.permutations(range(7)):
        score = sum(
            tuple(sorted((permutation[a], permutation[b]))) in edge_set for a, b in EDGES
        )
        scores.append(score)
    scores.sort(reverse=True)
    return float(scores[0] - scores[1])


RUNNER_UP_GAP = _assignment_gap()


def margin(scene: dict) -> float:
    del scene
    return RUNNER_UP_GAP


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["edge_order"] = list(reversed(twin["edge_order"]))
    return [("reverse_identical_edge_draw_order", twin)]
