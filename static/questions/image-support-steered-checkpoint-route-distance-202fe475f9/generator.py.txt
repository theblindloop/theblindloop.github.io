"""Deterministic renderer and analytic arm for Checkpoint Route Distance."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 800
HEIGHT = 600
QUARANTINE_PX = 80.0
INK = (38, 43, 49)
MUTED = (102, 108, 112)
ROUTE_PALETTE = (
    (12, 129, 158),
    (25, 116, 180),
    (18, 145, 124),
)
CHECKPOINT_PALETTE = (
    (222, 151, 28),
    (214, 132, 35),
    (232, 166, 31),
)
BACKGROUND_PALETTE = (
    (248, 247, 243),
    (246, 248, 247),
    (249, 247, 245),
)


def _font(size: int, *, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _route_lengths(scene: dict[str, Any]) -> tuple[float, float]:
    vertices = [tuple(int(axis) for axis in point) for point in scene["vertices"]]
    checkpoint = tuple(int(axis) for axis in scene["checkpoint"])
    if len(vertices) < 9 or vertices.count(checkpoint) != 1:
        raise ValueError("route must contain the checkpoint exactly once")
    cumulative = [0.0]
    for first, second in zip(vertices, vertices[1:], strict=False):
        dx = abs(second[0] - first[0])
        dy = abs(second[1] - first[1])
        if (dx == 0) == (dy == 0):
            raise ValueError("route segments must be nonzero and orthogonal")
        cumulative.append(cumulative[-1] + dx + dy)
    checkpoint_index = vertices.index(checkpoint)
    endpoint_a, endpoint_b = vertices[0], vertices[-1]
    distance_a = cumulative[checkpoint_index]
    distance_b = cumulative[-1] - distance_a
    if endpoint_a[0] == endpoint_b[0]:
        raise ValueError("route endpoints must have distinct horizontal positions")
    if endpoint_a[0] < endpoint_b[0]:
        return distance_a, distance_b
    return distance_b, distance_a


def _base_scene(group: int) -> dict[str, Any]:
    rng = random.Random(0xC4EC_2026 + group * 1_000_003)
    near_boundary = group % 7 == 0
    target = (48.0, 78.0) if near_boundary else (110.0, 195.0)

    for _ in range(20_000):
        y0 = rng.randint(96, 112)
        left_steps = [rng.randint(48, 68), rng.randint(54, 78), rng.randint(48, 68)]
        y1 = y0 + left_steps[0]
        y2 = y1 + left_steps[1]
        y3 = y2 + left_steps[2]
        if near_boundary:
            down1 = rng.randint(88, 112)
            back = rng.randint(34, 54)
            down2 = rng.randint(88, 112)
        else:
            down1 = rng.randint(116, 146)
            back = rng.randint(64, 94)
            down2 = rng.randint(116, 146)
        y4 = y3 + down1
        y5 = y4 - back
        y6 = y5 + down2
        left_vertical = sum(left_steps)
        right_vertical = down1 + back + down2
        difference = right_vertical - left_vertical
        if target[0] <= difference <= target[1] and y6 <= 510 and y6 - y0 >= 330:
            break
    else:
        raise RuntimeError("could not sample a separated route pair")

    tx = rng.randint(-8, 8)
    x0, checkpoint_x, x7 = 65 + tx, 400 + tx, 735 + tx
    connectors = [
        154 + tx + rng.randint(-7, 7),
        250 + tx + rng.randint(-7, 7),
        338 + tx + rng.randint(-5, 5),
        462 + tx + rng.randint(-5, 5),
        550 + tx + rng.randint(-7, 7),
        646 + tx + rng.randint(-7, 7),
    ]
    xs = [x0, *connectors, x7]
    ys = [y0, y1, y2, y3, y4, y5, y6]
    vertices: list[list[int]] = [[xs[0], ys[0]]]
    for index in range(1, 7):
        vertices.append([xs[index], ys[index - 1]])
        vertices.append([xs[index], ys[index]])
        if index == 3:
            vertices.append([checkpoint_x, ys[index]])
    vertices.append([xs[7], ys[6]])
    checkpoint = [checkpoint_x, y3]
    scene = {
        "width": WIDTH,
        "height": HEIGHT,
        "vertices": vertices,
        "checkpoint": checkpoint,
        "route_rgb": list(ROUTE_PALETTE[group % len(ROUTE_PALETTE)]),
        "checkpoint_rgb": list(CHECKPOINT_PALETTE[(group // 3) % len(CHECKPOINT_PALETTE)]),
        "background_rgb": list(BACKGROUND_PALETTE[(group // 5) % len(BACKGROUND_PALETTE)]),
    }
    left, right = _route_lengths(scene)
    if not left < right or not target[0] <= right - left <= target[1]:
        raise AssertionError("base route separation changed")
    return scene


def sample_scene(seed: int) -> dict[str, Any]:
    """Return mirrored pairs with equal ink and opposite answer identities."""

    seed = int(seed)
    group = seed // 2
    scene = _base_scene(group)
    if seed % 2:
        mirrored = [[WIDTH - int(x), int(y)] for x, y in scene["vertices"]]
        scene = dict(scene)
        scene["vertices"] = list(reversed(mirrored))
        scene["checkpoint"] = [WIDTH - int(scene["checkpoint"][0]), int(scene["checkpoint"][1])]
    return scene


def analytic_gold(scene: dict[str, Any]) -> str:
    left, right = _route_lengths(scene)
    if left == right:
        raise ValueError("checkpoint route lengths are tied")
    return "left" if left < right else "right"


def margin(scene: dict[str, Any]) -> float:
    left, right = _route_lengths(scene)
    return float(abs(left - right))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) <= QUARANTINE_PX


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    twin = dict(scene)
    twin["vertices"] = list(reversed([list(point) for point in scene["vertices"]]))
    return [("reverse_route_storage", twin)]


def render(scene: dict[str, Any]) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    vertices = [tuple(int(axis) for axis in point) for point in scene["vertices"]]
    checkpoint = tuple(int(axis) for axis in scene["checkpoint"])
    route = tuple(int(value) for value in scene["route_rgb"])
    gold = tuple(int(value) for value in scene["checkpoint_rgb"])
    background = tuple(int(value) for value in scene["background_rgb"])
    if (width, height) != (WIDTH, HEIGHT):
        raise ValueError("unexpected canvas size")
    _route_lengths(scene)
    if not all(42 <= x <= width - 42 and 88 <= y <= height - 62 for x, y in vertices):
        raise ValueError("route leaves the safe canvas")
    if not all(0 <= channel <= 255 for color in (route, gold, background) for channel in color):
        raise ValueError("RGB channel outside range")

    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)
    draw.text((width // 2, 26), "FOLLOW THE ROUTE", fill=INK, font=_font(25, bold=True), anchor="mm")
    draw.text(
        (width // 2, 58),
        "Compare distance along the teal line — not straight-line distance",
        fill=MUTED,
        font=_font(15),
        anchor="mm",
    )
    cx, cy = checkpoint
    draw.ellipse((cx - 15, cy - 15, cx + 15, cy + 15), outline=gold, width=5)
    draw.line(vertices, fill=route, width=7, joint="curve")
    for x, y in (vertices[0], vertices[-1]):
        draw.ellipse((x - 7, y - 7, x + 7, y + 7), fill=route)
    draw.text((50, 558), "LEFT END", fill=INK, font=_font(16, bold=True), anchor="lm")
    draw.text((750, 558), "RIGHT END", fill=INK, font=_font(16, bold=True), anchor="rm")
    return image
