"""Independent pixel-only inverse for Checkpoint Route Distance.

The final RGB raster is the sole input. Cool ink yields one connected route;
warm ink yields the checkpoint ring. The route is measured column by column:
horizontal travel is the checkpoint-to-endpoint x displacement and each tall
column cluster contributes one recovered vertical run. No renderer values,
scene payload, filename, metadata, or analytic answer are available here.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


def _masks(image: Image.Image) -> tuple[np.ndarray, np.ndarray]:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[:, :, 0], array[:, :, 1], array[:, :, 2]
    chroma = array.max(axis=2) - array.min(axis=2)
    route = (
        (chroma >= 55)
        & ((green - red) >= 48)
        & ((blue - red) >= 44)
        & (green >= 100)
    )
    checkpoint = (
        (chroma >= 100)
        & ((red - green) >= 52)
        & ((green - blue) >= 80)
        & (red >= 190)
    )
    return route, checkpoint


def _component_size(mask: np.ndarray, start: tuple[int, int]) -> int:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    stack = [start]
    seen[start] = True
    count = 0
    while stack:
        y, x = stack.pop()
        count += 1
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                seen[ny, nx] = True
                stack.append((ny, nx))
    return count


def _groups(values: np.ndarray) -> list[list[int]]:
    groups: list[list[int]] = []
    for value in values.tolist():
        if not groups or value > groups[-1][-1] + 1:
            groups.append([int(value)])
        else:
            groups[-1].append(int(value))
    return groups


def _measure(image: Image.Image) -> tuple[float, float]:
    route, checkpoint_mask = _masks(image)
    ys, xs = np.nonzero(route)
    if len(xs) < 4700:
        raise ValueError("teal route is missing or materially erased")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    if not image.width * 0.80 <= x1 - x0 + 1 <= image.width * 0.90:
        raise ValueError("route horizontal extent is implausible")
    if not image.height * 0.54 <= y1 - y0 + 1 <= image.height * 0.74:
        raise ValueError("route vertical extent is implausible")
    first = tuple(int(value) for value in np.argwhere(route)[0])
    if _component_size(route, first) != int(route.sum()):
        raise ValueError("teal route is severed")

    checkpoint_points = np.argwhere(checkpoint_mask)
    if not 220 <= len(checkpoint_points) <= 520:
        raise ValueError("gold checkpoint ring is missing or malformed")
    checkpoint_y, checkpoint_x = checkpoint_points.mean(axis=0)
    if not image.width * 0.46 <= checkpoint_x <= image.width * 0.54:
        raise ValueError("checkpoint horizontal position is implausible")
    if not route[max(0, int(round(checkpoint_y)) - 5):int(round(checkpoint_y)) + 6,
                 max(0, int(round(checkpoint_x)) - 5):int(round(checkpoint_x)) + 6].any():
        raise ValueError("checkpoint is not visibly on the route")

    spans = np.full(image.width, -1, dtype=np.int32)
    for x in range(x0, x1 + 1):
        column_y = np.flatnonzero(route[:, x])
        if len(column_y):
            spans[x] = int(column_y[-1] - column_y[0])
    if int(np.sum(spans[x0:x1 + 1] < 0)) > 2:
        raise ValueError("route has missing interior columns")
    thin = spans[(spans >= 4) & (spans <= 15)]
    if len(thin) < image.width * 0.55:
        raise ValueError("horizontal route support is incomplete")
    thickness_span = float(np.median(thin))
    tall_columns = np.flatnonzero(spans > thickness_span + 25)
    connector_groups = _groups(tall_columns)
    if len(connector_groups) != 6 or any(not 5 <= len(group) <= 11 for group in connector_groups):
        raise ValueError("six vertical route runs could not be recovered")

    left_vertical = 0.0
    right_vertical = 0.0
    left_count = 0
    right_count = 0
    for group in connector_groups:
        center_x = float(np.median(group))
        travel = float(max(spans[x] for x in group)) - thickness_span
        if travel < 30:
            raise ValueError("vertical route run is too short")
        if center_x < checkpoint_x:
            left_vertical += travel
            left_count += 1
        else:
            right_vertical += travel
            right_count += 1
    if (left_count, right_count) != (3, 3):
        raise ValueError("checkpoint does not split the route into three runs per side")
    left = float(checkpoint_x - x0) + left_vertical
    right = float(x1 - checkpoint_x) + right_vertical
    return left, right


def decision_from_image(image: Image.Image) -> str:
    left, right = _measure(image)
    if abs(left - right) <= 12:
        raise ValueError("pixel-measured route distances are tied")
    return "left" if left < right else "right"
