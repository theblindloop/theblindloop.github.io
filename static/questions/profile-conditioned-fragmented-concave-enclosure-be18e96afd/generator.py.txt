"""Deterministic latent sampler and independent forward renderer."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

SCALE = 4
QUARANTINE_PX = 10.0


def _radius(scene: dict, theta: float) -> float:
    return float(scene["base_radius"]) * (
        1.0 + float(scene["amplitude"]) * math.cos(int(scene["lobes"]) * theta + float(scene["phase"]))
    )


def _point(scene: dict, theta: float) -> tuple[float, float]:
    r = _radius(scene, theta)
    cx, cy = scene["center"]
    return cx + r * math.cos(theta), cy + r * math.sin(theta)


def _tangent_angle(scene: dict, theta: float) -> float:
    eps = 0.002
    x0, y0 = _point(scene, theta - eps)
    x1, y1 = _point(scene, theta + eps)
    return math.atan2(y1 - y0, x1 - x0)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    width = height = 384
    cx = rng.uniform(184.0, 200.0)
    cy = rng.uniform(184.0, 200.0)
    base = rng.uniform(105.0, 116.0)
    lobes = rng.choice((3, 4, 5))
    amplitude = rng.uniform(0.25, 0.31)
    phase = rng.uniform(0.0, 2.0 * math.pi)
    fragment_count = rng.choice((24, 27, 30))
    stroke_length = rng.uniform(13.0, 17.0)
    stroke_width = rng.choice((4, 5))

    # The outside marker occupies a concave bay: outside the implied contour but
    # inside its convex hull. The inside marker occupies a counterbalanced random ray.
    bay_theta = (math.pi - phase) / lobes + 2.0 * math.pi * rng.randrange(lobes) / lobes
    contour_at_bay = base * (1.0 - amplitude)
    # Include deterministic, still-off-quarantine near-boundary cases so the
    # gallery exercises the finite-resolution decision boundary.
    outside_gap = rng.uniform(11.5, 14.5) if int(seed) % 6 == 0 else rng.uniform(20.0, 27.0)
    outside_r = contour_at_bay + outside_gap
    outside = (cx + outside_r * math.cos(bay_theta), cy + outside_r * math.sin(bay_theta))
    inside_theta = bay_theta + rng.uniform(1.0, 2.0)
    inside_r = rng.uniform(24.0, 42.0)
    inside = (cx + inside_r * math.cos(inside_theta), cy + inside_r * math.sin(inside_theta))

    inside_color = "red" if int(seed) % 2 == 0 else "blue"
    if inside_color == "red":
        red_position, blue_position = inside, outside
    else:
        red_position, blue_position = outside, inside
    red_choices = ((214, 55, 65), (225, 70, 45), (195, 45, 75))
    blue_choices = ((42, 105, 210), (35, 125, 200), (65, 85, 215))
    backgrounds = ((250, 248, 242), (246, 250, 249), (250, 247, 252))
    return {
        "width": width,
        "height": height,
        "center": [round(cx, 5), round(cy, 5)],
        "base_radius": round(base, 5),
        "lobes": lobes,
        "amplitude": round(amplitude, 6),
        "phase": round(phase, 7),
        "fragment_count": fragment_count,
        "stroke_length": round(stroke_length, 5),
        "stroke_width": stroke_width,
        "red_position": [round(red_position[0], 5), round(red_position[1], 5)],
        "blue_position": [round(blue_position[0], 5), round(blue_position[1], 5)],
        "inside_color": inside_color,
        "red_rgb": list(rng.choice(red_choices)),
        "blue_rgb": list(rng.choice(blue_choices)),
        "background_rgb": list(rng.choice(backgrounds)),
    }


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    background = tuple(scene["background_rgb"])
    image = Image.new("RGB", (width * SCALE, height * SCALE), background)
    draw = ImageDraw.Draw(image)
    count = int(scene["fragment_count"])
    length = float(scene["stroke_length"])
    line_width = int(scene["stroke_width"]) * SCALE
    # Accessed explicitly because marker assignment makes the label visible.
    inside_color = scene["inside_color"]
    if inside_color not in {"red", "blue"}:
        raise ValueError("invalid inside_color")
    for index in range(count):
        theta = 2.0 * math.pi * index / count
        x, y = _point(scene, theta)
        tangent = _tangent_angle(scene, theta)
        dx = 0.5 * length * math.cos(tangent)
        dy = 0.5 * length * math.sin(tangent)
        draw.line(
            ((x - dx) * SCALE, (y - dy) * SCALE, (x + dx) * SCALE, (y + dy) * SCALE),
            fill=(25, 25, 28),
            width=line_width,
        )
    marker_radius = 8.0
    for key, rgb in (("red_position", scene["red_rgb"]), ("blue_position", scene["blue_rgb"])):
        x, y = scene[key]
        box = (
            (x - marker_radius) * SCALE,
            (y - marker_radius) * SCALE,
            (x + marker_radius) * SCALE,
            (y + marker_radius) * SCALE,
        )
        draw.ellipse(box, fill=tuple(rgb))
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["inside_color"])


def margin(scene: dict) -> float:
    cx, cy = scene["center"]
    clearances = []
    for key in ("red_position", "blue_position"):
        x, y = scene[key]
        theta = math.atan2(y - cy, x - cx)
        radial = math.hypot(x - cx, y - cy)
        clearances.append(abs(_radius(scene, theta) - radial))
    return min(clearances)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
