"""Scene-free inverse arm operating only on raster pixels."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y0, x0 in np.argwhere(mask):
        if seen[y0, x0]:
            continue
        stack = [(int(y0), int(x0))]
        seen[y0, x0] = True
        xs, ys = [], []
        while stack:
            y, x = stack.pop()
            xs.append(x)
            ys.append(y)
            for yy, xx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= yy < height and 0 <= xx < width and mask[yy, xx] and not seen[yy, xx]:
                    seen[yy, xx] = True
                    stack.append((yy, xx))
        if len(xs) >= minimum:
            output.append((float(np.mean(xs)), float(np.mean(ys)), len(xs)))
    return output


def _marker_center(arr: np.ndarray, color: str) -> tuple[float, float]:
    r, g, b = arr[..., 0], arr[..., 1], arr[..., 2]
    if color == "red":
        mask = (r > 145) & (r > g * 1.45) & (r > b * 1.35)
    else:
        mask = (b > 135) & (b > r * 1.35) & (b > g * 1.18)
    ys, xs = np.nonzero(mask)
    if len(xs) < 80:
        raise ValueError(f"missing {color} marker")
    return float(xs.mean()), float(ys.mean())


def _inside(point: tuple[float, float], polygon: list[tuple[float, float]]) -> bool:
    x, y = point
    result = False
    previous = polygon[-1]
    for current in polygon:
        x1, y1 = previous
        x2, y2 = current
        if (y1 > y) != (y2 > y):
            crossing_x = (x2 - x1) * (y - y1) / (y2 - y1) + x1
            if x < crossing_x:
                result = not result
        previous = current
    return result


def reconstruct(image: Image.Image) -> tuple[list[tuple[float, float]], dict[str, tuple[float, float]]]:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    black = (arr[..., 0] < 90) & (arr[..., 1] < 90) & (arr[..., 2] < 95)
    pieces = _components(black, minimum=16)
    if len(pieces) < 18:
        raise ValueError("insufficient contour fragments")
    centers = [(x, y) for x, y, _ in pieces]
    cx = sum(x for x, _ in centers) / len(centers)
    cy = sum(y for _, y in centers) / len(centers)
    centers.sort(key=lambda p: math.atan2(p[1] - cy, p[0] - cx))
    markers = {"red": _marker_center(arr, "red"), "blue": _marker_center(arr, "blue")}
    return centers, markers


def decision_from_image(image: Image.Image) -> str:
    polygon, markers = reconstruct(image)
    states = {color: _inside(point, polygon) for color, point in markers.items()}
    winners = [color for color, state in states.items() if state]
    if len(winners) != 1:
        raise ValueError(f"expected exactly one contained marker, got {states}")
    return winners[0]
