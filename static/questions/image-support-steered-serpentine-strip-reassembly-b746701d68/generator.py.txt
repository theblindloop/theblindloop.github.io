"""Renderer and analytic arm for a connected-route reassembly question world."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw


WIDTH = 640
HEIGHT = 480
PANEL_COUNT = 4
STRIP_WIDTH = 320
STRIP_HEIGHT = 120
PANEL_WIDTH = STRIP_WIDTH // PANEL_COUNT
SYMBOLS = ("wave", "fork")
INK_WIDTH = 8

PALETTES = (
    {"route": (29, 61, 82), "ink": (247, 222, 166), "accent": (72, 174, 127)},
    {"route": (42, 72, 58), "ink": (244, 228, 190), "accent": (83, 181, 132)},
    {"route": (57, 48, 91), "ink": (248, 220, 176), "accent": (75, 168, 139)},
)
PAPERS = ((248, 246, 239), (242, 247, 242), (247, 243, 249))


def _line_points(start: tuple[float, float], end: tuple[float, float], count: int = 48) -> list[tuple[float, float]]:
    return [
        (
            start[0] + (end[0] - start[0]) * index / count,
            start[1] + (end[1] - start[1]) * index / count,
        )
        for index in range(count + 1)
    ]


def _strip_paths(symbol: str) -> list[list[tuple[float, float]]]:
    if symbol == "wave":
        return [
            [
                (float(x), 60.0 + 16.0 * math.sin((float(x) - 16.0) * 2.0 * math.pi / 142.0))
                for x in range(16, 305, 2)
            ]
        ]
    if symbol == "fork":
        return [
            _line_points((176.0, 101.0), (176.0, 60.0)),
            _line_points((18.0, 29.0), (176.0, 60.0)) + _line_points((176.0, 60.0), (302.0, 29.0)),
            _line_points((18.0, 91.0), (66.0, 68.0)),
            _line_points((254.0, 68.0), (302.0, 91.0)),
        ]
    raise ValueError(f"unknown symbol {symbol!r}")


def sample_scene(seed: int) -> dict:
    """Sample a balanced, nuisance-randomized scene."""

    rng = random.Random(int(seed) * 1000003 + 739391)
    symbol = SYMBOLS[int(seed) & 1]
    route_width = rng.uniform(330.0, 360.0)
    route_height = rng.uniform(60.0, 72.0)
    scene = {
        "width": WIDTH,
        "height": HEIGHT,
        "route_x": rng.uniform(310.0, 330.0),
        "route_y": rng.uniform(236.0, 244.0),
        "route_width": route_width,
        "route_height": route_height,
        "stroke": rng.uniform(38.0, 44.0),
        "palette": rng.randrange(len(PALETTES)),
        "accent_phase": rng.randrange(4),
        "paper_tone": rng.randrange(len(PAPERS)),
        "symbol": symbol,
    }
    return scene


def _route_points(scene: dict) -> list[tuple[float, float]]:
    cx = float(scene["route_x"])
    cy = float(scene["route_y"])
    route_width = float(scene["route_width"])
    route_height = float(scene["route_height"])
    left = cx - route_width / 2.0
    right = cx + route_width / 2.0
    top = cy - route_height / 2.0
    bottom = cy + route_height / 2.0
    step = route_width / PANEL_COUNT
    return [
        (left, top),
        (left + step, top),
        (left + step, bottom),
        (left + 2.0 * step, bottom),
        (left + 2.0 * step, top),
        (left + 3.0 * step, top),
        (left + 3.0 * step, bottom),
        (right, bottom),
    ]


def _map_strip_point(scene: dict, x: float, y: float) -> tuple[float, float]:
    points = _route_points(scene)
    left = points[0][0]
    step = float(scene["route_width"]) / PANEL_COUNT
    panel = min(PANEL_COUNT - 1, max(0, int(x // PANEL_WIDTH)))
    local_x = (x - panel * PANEL_WIDTH) / PANEL_WIDTH
    image_x = left + (panel + local_x) * step
    base_y = points[0][1] if panel % 2 == 0 else points[2][1]
    scale_y = float(scene["stroke"]) * 0.72 / 60.0
    image_y = base_y + (y - 60.0) * scale_y
    return image_x, image_y


def render(scene: dict) -> Image.Image:
    """Render one connected folded route and its embedded strip ink."""

    if (int(scene["width"]), int(scene["height"])) != (WIDTH, HEIGHT):
        raise ValueError("unsupported canvas size")
    palette = PALETTES[int(scene["palette"]) % len(PALETTES)]
    background = PAPERS[int(scene["paper_tone"]) % len(PAPERS)]
    image = Image.new("RGB", (WIDTH, HEIGHT), background)
    draw = ImageDraw.Draw(image)
    points = _route_points(scene)
    stroke = max(24, int(round(float(scene["stroke"]))))
    route_points = [(int(round(x)), int(round(y))) for x, y in points]
    draw.line(route_points, fill=palette["route"], width=stroke, joint="curve")
    radius = stroke // 2
    for x, y in route_points:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=palette["route"])

    # The bright mark is drawn panel by panel.  It is not a canonical symbol
    # until the alternating panel baselines are translated and concatenated.
    for path in _strip_paths(str(scene["symbol"])):
        for panel in range(PANEL_COUNT):
            lo = panel * PANEL_WIDTH
            hi = (panel + 1) * PANEL_WIDTH
            panel_points = [point for point in path if lo <= point[0] <= hi]
            if len(panel_points) >= 2:
                mapped = [_map_strip_point(scene, x, y) for x, y in panel_points]
                draw.line(
                    [(int(round(x)), int(round(y))) for x, y in mapped],
                    fill=palette["ink"],
                    width=INK_WIDTH,
                    joint="curve",
                )

    # Embedded green caps and join ticks expose route order without adding a
    # second carrier component.  The dark route remains the transform anchor.
    accent = palette["accent"]
    phase = int(scene["accent_phase"]) % 4
    for index, (x, y) in enumerate((route_points[0], route_points[-1])):
        if index == 0:
            draw.ellipse((x - 9, y - 9, x + 9, y + 9), fill=accent)
            draw.polygon([(x - 15, y), (x - 5, y - 10), (x - 5, y + 10)], fill=accent)
        else:
            draw.ellipse((x - 9, y - 9, x + 9, y + 9), outline=accent, width=4)
    for join_index in range(1, PANEL_COUNT):
        x = route_points[2 * join_index - 1][0]
        y = route_points[2 * join_index - 1][1]
        tick_color = accent if (join_index + phase) % 2 else palette["ink"]
        draw.line((x, y - 14, x, y + 14), fill=tick_color, width=4)
    return image


def analytic_gold(scene: dict) -> str:
    """Analytic label; the only label-bearing scene input is the symbol."""

    symbol = scene["symbol"]
    if symbol not in SYMBOLS:
        raise ValueError(f"unknown symbol {symbol!r}")
    return symbol


def margin(scene: dict) -> float:
    """One-sided margin for route thickness and panel separation."""

    stroke = float(scene["stroke"])
    height = float(scene["route_height"])
    return float(min(stroke / 32.0, height / 48.0, 1.0))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.08


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
