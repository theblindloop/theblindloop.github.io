"""Tracing world: how many black waypoints lie between the colored dots on-path.

A single winding blue path (one connected, non-self-crossing polyline) is drawn.
An orange port ring sits near the start of the path. Two distinctly coloured
terminal dots (red and green) sit on the path, together with several black
waypoint dots, some of which lie on the path arc strictly between the red and
green dots and some of which lie elsewhere (beyond either terminal) as
distractors. Walking from the orange port along the blue path, the solution is
the number of black waypoints encountered strictly between the red and green
dots. The count is a geodesic/on-path fact: because the path winds back on
itself in a serpentine, straight-line order need not equal on-path order, so the
task cannot be solved by Euclidean projection alone.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "count_between_on_path-0.1.0"

CANVAS = (480, 480)
BG = (255, 255, 255)
COLOR_BLUE = (40, 90, 210)
COLOR_RED = (225, 40, 40)
COLOR_GREEN = (40, 170, 70)
COLOR_BLACK = (30, 30, 30)
COLOR_ORANGE = (245, 150, 20)

STROKE = 20          # blue stroke thickness (px)
PORT_R = 5          # orange port radius
MARK_R = 4          # coloured/black marker radius
MARGIN_PAD = 46     # keep geometry away from image borders
# Minimum Euclidean gap between non-adjacent serpentine lanes so the nearest
# blue stroke to any marker is unambiguously its own lane.
LANE_GAP = 30.0
# Minimum arc distance between any two consecutive markers on the path.
ARC_SEP = 30.0
# Minimum distance from path endpoints for markers.
END_MARGIN = 40.0

ANSWER_VALUES = [0, 1, 2, 3]


def _pts(points: list) -> list[tuple[float, float]]:
    return [(float(p[0]), float(p[1])) for p in points]


def _path_length(points: list[tuple[float, float]]) -> float:
    return sum(math.dist(a, b) for a, b in zip(points, points[1:]))


def _point_at(points: list[tuple[float, float]], s: float) -> tuple[float, float]:
    """Point at arc-distance ``s`` along the polyline (s within total length)."""
    remaining = s
    for a, b in zip(points, points[1:]):
        seg = math.dist(a, b)
        if remaining <= seg or seg == 0:
            t = 0.0 if seg == 0 else remaining / seg
            return (a[0] + (b[0] - a[0]) * t, a[1] + (b[1] - a[1]) * t)
        remaining -= seg
    return points[-1]


def _sample(points: list[tuple[float, float]], step: float = 2.0) -> list[tuple[float, float]]:
    out: list[tuple[float, float]] = []
    for a, b in zip(points, points[1:]):
        d = math.dist(a, b)
        n = max(1, int(d / step))
        for i in range(n + 1):
            t = i / n
            out.append((a[0] + (b[0] - a[0]) * t, a[1] + (b[1] - a[1]) * t))
    return out


def _min_pairwise_distances(samples: list[tuple[float, float]]) -> float:
    best = float("inf")
    for i in range(len(samples)):
        for j in range(i + 1, len(samples)):
            d = math.dist(samples[i], samples[j])
            if d < best:
                best = d
    return best


def _serpentine_clearance(points: list[tuple[float, float]]) -> float:
    """Min distance between any two samples of the path that are far in arc."""
    n = len(points)
    samples_all = _sample(points)
    # We approximate self-clearance by the min distance among samples that are
    # not near-neighbours along the polyline (arc-neighbour samples are the
    # legitimately-adjacent stroke which we ignore via a per-vertex check below).
    best = float("inf")
    for i in range(len(samples_all)):
        for j in range(i + 1, len(samples_all)):
            d = math.dist(samples_all[i], samples_all[j])
            if d < best:
                best = d
    return best


def _rejection_loop(rng: random.Random) -> dict:
    w, h = CANVAS
    lo, hi = MARGIN_PAD, w - MARGIN_PAD
    for _ in range(20000):
        # Zigzag (serpentine) path: horizontal lanes at vertically separated
        # heights, alternating left/right. Lanes are placed lane_vert_gap apart,
        # so non-adjacent lanes are >= 2*lane_vert_gap apart and the stroke never
        # touches itself. The path is one connected, non-crossing polyline.
        lane_vert_gap = LANE_GAP  # >= LANE_GAP keeps non-adjacent lanes clear
        n_lanes = rng.randint(4, 7)
        y_lo = lo + 10
        y_hi = hi - 10
        usable_h = y_hi - y_lo
        span_h = (n_lanes - 1) * lane_vert_gap
        span_h = min(span_h, usable_h)
        step_h = span_h / (n_lanes - 1) if n_lanes > 1 else 0.0
        x_start = rng.uniform(lo + 15, lo + 50)
        x_end = rng.uniform(hi - 50, hi - 15)
        path = []
        for lane in range(n_lanes):
            y = y_lo + lane * step_h
            if lane % 2 == 0:
                path.append((x_start, y))
                path.append((x_end, y))
            else:
                path.append((x_end, y))
                path.append((x_start, y))
        path = _pts(path)
        total = _path_length(path)
        if total < END_MARGIN * 4:
            continue

        # Port near the start.
        s_port = END_MARGIN * rng.uniform(1.0, 1.5)
        port = _point_at(path, s_port)

        # Choose red and green arc positions.
        lo_s = END_MARGIN + 20
        hi_s = total - END_MARGIN - 20
        if hi_s - lo_s < 6 * ARC_SEP:
            continue
        s_red = rng.uniform(lo_s, lo_s + (hi_s - lo_s) * 0.35)
        span = hi_s - s_red
        s_green = s_red + rng.uniform(3 * ARC_SEP, span)
        # number of black waypoints between red and green.
        k_between = rng.choice(ANSWER_VALUES)
        between_gap = (s_green - s_red) / (k_between + 1)
        if between_gap < ARC_SEP:
            continue
        s_black_between = [s_red + between_gap * (i + 1) for i in range(k_between)]
        # distractor black waypoints outside the interval.
        k_outside = rng.choice([0, 1, 2])
        s_black_outside = []
        for _ in range(k_outside):
            if rng.random() < 0.5:
                s_black_outside.append(rng.uniform(s_port + ARC_SEP, s_red - ARC_SEP))
            else:
                s_black_outside.append(rng.uniform(s_green + ARC_SEP, hi_s))

        arc_positions = [s_port, s_red, s_green] + s_black_between + s_black_outside
        arc_positions.sort()
        ok = True
        for a, b in zip(arc_positions, arc_positions[1:]):
            if b - a < ARC_SEP:
                ok = False
                break
        if not ok:
            continue

        red_pos = _point_at(path, s_red)
        green_pos = _point_at(path, s_green)
        black_between_pos = [_point_at(path, s) for s in s_black_between]
        black_outside_pos = [_point_at(path, s) for s in s_black_outside]

        marks = [{"x": red_pos[0], "y": red_pos[1], "color": "red"},
                 {"x": green_pos[0], "y": green_pos[1], "color": "green"}]
        for (x, y) in black_between_pos:
            marks.append({"x": x, "y": y, "color": "black"})
        for (x, y) in black_outside_pos:
            marks.append({"x": x, "y": y, "color": "black"})
        rng.shuffle(marks)

        allgood = True
        for m in marks + [{"x": port[0], "y": port[1]}]:
            if not (20 < m["x"] < w - 20 and 20 < m["y"] < h - 20):
                allgood = False
                break
        if not allgood:
            continue

        scene = {
            "path": path,
            "port": [float(port[0]), float(port[1])],
            "marks": marks,
            "answer": str(k_between),
        }
        return scene
    raise RuntimeError("could not satisfy path constraints")


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    return _rejection_loop(rng)


def _draw_polyline(draw: ImageDraw.ImageDraw, pts, color, width):
    for a, b in zip(pts, pts[1:]):
        draw.line([a, b], fill=color, width=width, joint="curve")


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", CANVAS, BG)
    draw = ImageDraw.Draw(img)
    path = _pts(scene["path"])
    _draw_polyline(draw, path, COLOR_BLUE, STROKE)
    colors = {"red": COLOR_RED, "green": COLOR_GREEN, "black": COLOR_BLACK}
    for mark in scene["marks"]:
        r = MARK_R
        px, py = float(mark["x"]), float(mark["y"])
        draw.ellipse([px - r, py - r, px + r, py + r], fill=colors[mark["color"]])
    px, py = float(scene["port"][0]), float(scene["port"][1])
    draw.ellipse([px - PORT_R, py - PORT_R, px + PORT_R, py + PORT_R], fill=COLOR_ORANGE)
    return img


def _arc_position(scene: dict, pt: tuple[float, float]) -> float:
    """Arc distance from the path start to the projection of ``pt`` on the path."""
    best = None
    best_d = float("inf")
    cum = 0.0
    for a, b in zip(_pts(scene["path"]), _pts(scene["path"])[1:]):
        # project pt onto segment a-b
        abx, aby = b[0] - a[0], b[1] - a[1]
        seg2 = abx * abx + aby * aby
        t = 0.0 if seg2 == 0 else ((pt[0] - a[0]) * abx + (pt[1] - a[1]) * aby) / seg2
        t = max(0.0, min(1.0, t))
        proj = (a[0] + abx * t, a[1] + aby * t)
        d = math.dist(pt, proj)
        if d < best_d:
            best_d = d
            best = cum + math.dist(a, proj)
        cum += math.dist(a, b)
    return best


def analytic_gold(scene: dict) -> str:
    """Number of black waypoints between red and green along the path."""
    red = [m for m in scene["marks"] if m["color"] == "red"][0]
    green = [m for m in scene["marks"] if m["color"] == "green"][0]
    ar = _arc_position(scene, (red["x"], red["y"]))
    ag = _arc_position(scene, (green["x"], green["y"]))
    lo, hi = min(ar, ag), max(ar, ag)
    count = 0
    for m in scene["marks"]:
        if m["color"] == "black":
            am = _arc_position(scene, (m["x"], m["y"]))
            if lo < am < hi:
                count += 1
    return str(count)


def margin(scene: dict) -> float:
    """Minimum arc separation between consecutive on-path markers (px)."""
    pos = [scene["port"]] + [[m["x"], m["y"]] for m in scene["marks"]]
    arc = sorted(_arc_position(scene, (p[0], p[1])) for p in pos)
    return min(b - a for a, b in zip(arc, arc[1:]))


def is_quarantined(scene: dict) -> bool:
    w, h = CANVAS
    if margin(scene) < ARC_SEP * 0.8:
        return True
    for m in scene["marks"]:
        if not (20 <= m["x"] <= w - 20 and 20 <= m["y"] <= h - 20):
            return True
    return False


def latent_symmetries(scene: dict) -> list:
    """No nontrivial render-invariant transform: colours and positions are
    directly visible and any permutation of marker colours changes both the
    pixels and the gold. Declared empty (not_applicable)."""
    return []
