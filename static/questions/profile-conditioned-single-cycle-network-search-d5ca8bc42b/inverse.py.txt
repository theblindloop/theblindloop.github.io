"""Independent image-only inverse arm for the single-cycle network world.

The oracle receives only the final RGB image.  It finds the 24 bright chromatic
node disks, groups them into four visible regions, measures dark ink along every
possible node-pair segment, reconstructs each adjacency matrix, and requires one
connected 2-regular graph.  It deliberately shares no renderer or scene code.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

_LABELS = "ABCD"


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    locations = (np.flatnonzero(mask)).tolist()
    remaining = set(locations)
    result: list[np.ndarray] = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        pixels = [start]
        while stack:
            value = stack.pop()
            y, x = divmod(value, width)
            neighbors = []
            if x:
                neighbors.append(value - 1)
            if x + 1 < width:
                neighbors.append(value + 1)
            if y:
                neighbors.append(value - width)
            if y + 1 < height:
                neighbors.append(value + width)
            for neighbor in neighbors:
                if neighbor in remaining:
                    remaining.remove(neighbor)
                    stack.append(neighbor)
                    pixels.append(neighbor)
        if len(pixels) >= 220:
            result.append(np.asarray(pixels, dtype=np.int64))
    return result


def _node_centers(array: np.ndarray) -> list[tuple[float, float]]:
    high = array.max(axis=2).astype(np.int16)
    low = array.min(axis=2).astype(np.int16)
    node_mask = ((high - low) > 72) & (high > 148) & (low < 180)
    components = _components(node_mask)
    width = array.shape[1]
    centers = []
    for component in components:
        ys = component // width
        xs = component % width
        area = len(component)
        span_x = int(xs.max() - xs.min() + 1)
        span_y = int(ys.max() - ys.min() + 1)
        if 260 <= area <= 900 and 16 <= span_x <= 38 and 16 <= span_y <= 38:
            centers.append((float(xs.mean()), float(ys.mean())))
    if len(centers) != 24:
        raise ValueError(f"expected 24 visible node disks, recovered {len(centers)}")
    return centers


def _group_panels(centers: list[tuple[float, float]]) -> list[list[tuple[float, float]]]:
    ordered_x = sorted(point[0] for point in centers)
    ordered_y = sorted(point[1] for point in centers)
    split_x = (ordered_x[11] + ordered_x[12]) / 2.0
    split_y = (ordered_y[11] + ordered_y[12]) / 2.0
    panels: list[list[tuple[float, float]]] = [[], [], [], []]
    for point in centers:
        column = int(point[0] > split_x)
        row = int(point[1] > split_y)
        panels[row * 2 + column].append(point)
    if any(len(panel) != 6 for panel in panels):
        raise ValueError(f"visible regions do not contain six nodes each: {[len(p) for p in panels]}")
    return panels


def _dark_fraction(array: np.ndarray, first: tuple[float, float], second: tuple[float, float]) -> float:
    x0, y0 = first
    x1, y1 = second
    dx, dy = x1 - x0, y1 - y0
    length = math.hypot(dx, dy)
    if length < 35:
        return 0.0
    nx, ny = -dy / length, dx / length
    samples = 84
    hits = 0
    total = 0
    luminance = (
        0.2126 * array[:, :, 0].astype(np.float32)
        + 0.7152 * array[:, :, 1].astype(np.float32)
        + 0.0722 * array[:, :, 2].astype(np.float32)
    )
    for value in np.linspace(0.16, 0.84, samples):
        base_x = x0 + value * dx
        base_y = y0 + value * dy
        strip_hit = False
        for offset in (-2.0, 0.0, 2.0):
            x = int(round(base_x + offset * nx))
            y = int(round(base_y + offset * ny))
            if 0 <= y < array.shape[0] and 0 <= x < array.shape[1]:
                strip_hit = strip_hit or bool(luminance[y, x] < 145.0)
        hits += int(strip_hit)
        total += 1
    return hits / max(total, 1)


def _recover_edges(array: np.ndarray, nodes: list[tuple[float, float]]) -> list[tuple[int, int]]:
    # Give nodes stable cyclic indices solely to express an adjacency matrix;
    # the graph property itself is invariant to this indexing.
    cx = sum(point[0] for point in nodes) / len(nodes)
    cy = sum(point[1] for point in nodes) / len(nodes)
    ordered = sorted(nodes, key=lambda point: math.atan2(point[1] - cy, point[0] - cx))
    edges = []
    for first in range(6):
        for second in range(first + 1, 6):
            fraction = _dark_fraction(array, ordered[first], ordered[second])
            if fraction >= 0.72:
                edges.append((first, second))
    if len(edges) != 6:
        raise ValueError(f"expected six visible edges in a panel, recovered {len(edges)}")
    return edges


def _is_single_cycle(edges: list[tuple[int, int]]) -> bool:
    degrees = [0] * 6
    adjacency = [set() for _ in range(6)]
    for first, second in edges:
        degrees[first] += 1
        degrees[second] += 1
        adjacency[first].add(second)
        adjacency[second].add(first)
    reached = {0}
    pending = [0]
    while pending:
        node = pending.pop()
        for neighbor in adjacency[node]:
            if neighbor not in reached:
                reached.add(neighbor)
                pending.append(neighbor)
    return degrees == [2] * 6 and len(reached) == 6


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    panels = _group_panels(_node_centers(array))
    winners = []
    for index, nodes in enumerate(panels):
        if _is_single_cycle(_recover_edges(array, nodes)):
            winners.append(index)
    if len(winners) != 1:
        raise ValueError(f"pixel evidence does not identify one single-cycle panel: {winners}")
    return _LABELS[winners[0]]
