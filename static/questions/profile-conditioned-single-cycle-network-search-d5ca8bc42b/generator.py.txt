"""Deterministic latent sampler, analytic graph rule, and Pillow renderer."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw, ImageFont

_LABELS = "ABCD"
_PANEL_CENTERS = ((202, 240), (566, 240), (202, 574), (566, 574))
_GRAPH_RADIUS = 102.0

# All four templates contain six edges with the same regular-hexagon chord
# multiset: two side chords, two skip-one chords, and two diameters.  Thus their
# ideal total line length is identical.  Only the first template is a connected
# 2-regular graph.  The second is two disjoint triangles; the last two are
# connected but branched.
_TEMPLATES = (
    ((0, 1), (0, 2), (1, 4), (2, 5), (3, 4), (3, 5)),
    ((0, 1), (0, 3), (1, 3), (2, 4), (2, 5), (4, 5)),
    ((0, 1), (0, 2), (0, 3), (1, 3), (2, 5), (4, 5)),
    ((0, 1), (0, 2), (0, 3), (2, 5), (3, 4), (3, 5)),
)

_PALETTES = (
    {"node": "#E87326", "edge": "#243B59", "accent": "#EDF1F5"},
    {"node": "#C33F7A", "edge": "#31583D", "accent": "#F1EEF3"},
    {"node": "#209AA7", "edge": "#563465", "accent": "#EEF2F1"},
)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    path = f"/usr/share/fonts/truetype/dejavu/{name}"
    try:
        return ImageFont.truetype(path, size=size)
    except OSError:
        return ImageFont.load_default()


def _mapped_edges(edges: tuple[tuple[int, int], ...], shift: int, reflected: bool) -> list[list[int]]:
    def remap(value: int) -> int:
        signed = -value if reflected else value
        return (signed + shift) % 6

    return [[remap(a), remap(b)] for a, b in edges]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target_index = int(seed) % 4
    distractors = [1, 2, 3]
    rng.shuffle(distractors)
    kinds = list(distractors)
    kinds.insert(target_index, 0)
    panels = []
    for kind in kinds:
        panels.append(
            _mapped_edges(
                _TEMPLATES[kind],
                shift=rng.randrange(6),
                reflected=bool(rng.randrange(2)),
            )
        )
    palette = dict(_PALETTES[rng.randrange(len(_PALETTES))])
    return {
        "panel_edges": panels,
        "phase_degrees": rng.choice([0, 10, 20, 30, 40, 50]),
        "canvas_size": [768, 768],
        "palette": palette,
    }


def _node_xy(center: tuple[int, int], phase_degrees: float) -> list[tuple[float, float]]:
    phase = math.radians(float(phase_degrees) - 90.0)
    return [
        (
            center[0] + _GRAPH_RADIUS * math.cos(phase + index * math.pi / 3.0),
            center[1] + _GRAPH_RADIUS * math.sin(phase + index * math.pi / 3.0),
        )
        for index in range(6)
    ]


def render(scene: dict) -> Image.Image:
    width, height = [int(value) for value in scene["canvas_size"]]
    palette = scene["palette"]
    phase = float(scene["phase_degrees"])
    panel_edges = scene["panel_edges"]
    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), "#FAF9F6")
    draw = ImageDraw.Draw(image)

    def box(coords: tuple[float, float, float, float], **kwargs) -> None:
        draw.rounded_rectangle(tuple(int(v * scale) for v in coords), **kwargs)

    title_font = _font(24 * scale, bold=True)
    label_font = _font(27 * scale, bold=True)
    title = "FIND THE ONE CLOSED LOOP"
    title_box = draw.textbbox((0, 0), title, font=title_font)
    draw.text(
        ((width * scale - (title_box[2] - title_box[0])) / 2, 18 * scale),
        title,
        fill="#2D333B",
        font=title_font,
    )
    draw.line((92 * scale, 58 * scale, 676 * scale, 58 * scale), fill="#D5D9DE", width=2 * scale)

    for panel_index, center in enumerate(_PANEL_CENTERS):
        cx, cy = center
        box(
            (cx - 164, cy - 151, cx + 164, cy + 151),
            radius=18 * scale,
            fill="#FFFFFF",
            outline="#CBD1D8",
            width=2 * scale,
        )
        label = _LABELS[panel_index]
        label_xy = ((cx - 147) * scale, (cy - 139) * scale)
        draw.text(label_xy, label, fill="#2D333B", font=label_font)
        nodes = _node_xy(center, phase)
        for a, b in panel_edges[panel_index]:
            ax, ay = nodes[int(a)]
            bx, by = nodes[int(b)]
            draw.line(
                (int(ax * scale), int(ay * scale), int(bx * scale), int(by * scale)),
                fill=palette["edge"],
                width=7 * scale,
            )
        radius = 13 * scale
        for x, y in nodes:
            xx, yy = int(x * scale), int(y * scale)
            draw.ellipse(
                (xx - radius, yy - radius, xx + radius, yy + radius),
                fill=palette["node"],
                outline="#FFFFFF",
                width=2 * scale,
            )

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    winners: list[int] = []
    for panel_index, edges in enumerate(scene["panel_edges"]):
        degrees = [0] * 6
        adjacency = [set() for _ in range(6)]
        for first, second in edges:
            a, b = int(first), int(second)
            degrees[a] += 1
            degrees[b] += 1
            adjacency[a].add(b)
            adjacency[b].add(a)
        reached = {0}
        pending = [0]
        while pending:
            node = pending.pop()
            for neighbor in adjacency[node]:
                if neighbor not in reached:
                    reached.add(neighbor)
                    pending.append(neighbor)
        if degrees == [2] * 6 and len(reached) == 6:
            winners.append(panel_index)
    if len(winners) != 1:
        raise ValueError(f"expected one connected 2-regular panel, found {winners}")
    return _LABELS[winners[0]]


def _panel_violation(edges: list[list[int]]) -> int:
    degrees = [0] * 6
    adjacency = [set() for _ in range(6)]
    for first, second in edges:
        a, b = int(first), int(second)
        degrees[a] += 1
        degrees[b] += 1
        adjacency[a].add(b)
        adjacency[b].add(a)
    reached = set()
    components = 0
    for root in range(6):
        if root in reached:
            continue
        components += 1
        reached.add(root)
        pending = [root]
        while pending:
            node = pending.pop()
            for neighbor in adjacency[node]:
                if neighbor not in reached:
                    reached.add(neighbor)
                    pending.append(neighbor)
    return sum(abs(value - 2) for value in degrees) + 2 * (components - 1)


def margin(scene: dict) -> float:
    scores = sorted(_panel_violation(edges) for edges in scene["panel_edges"])
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    try:
        decision = analytic_gold(scene)
    except (ValueError, IndexError, TypeError):
        return True
    return decision not in _LABELS or margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["panel_edges"] = [
        [[second, first] for first, second in edges] for edges in scene["panel_edges"]
    ]
    return [("reverse_all_undirected_edge_endpoints", twin)]
