"""Forward arm for quota_drain_race.

Two framed panels stand apart inside one middle band of the canvas.

* The left panel is the quota card.  It has four rows; each row holds a
  coloured disc on the left and a glyph of one, two or three black dots on the
  right.  The dot count is that colour's quota: how many deliveries of that
  colour are still owed.
* The right panel is the delivery lane.  It is a tall strip cut into twelve
  cells stacked one above another, each holding one coloured disc.  Read top to
  bottom, those twelve discs are the deliveries in arrival order.

Walking the lane downwards, each disc pays off one unit of its own colour's
quota.  The decision is the colour whose quota reaches zero first -- a race to
zero, not an accumulation and not a lookup.  Every colour appears exactly three
times in the lane and every quota is at most three, so every colour does
eventually run out and exactly one runs out first; because the per-colour lane
counts are all equal, colour frequency, total ink and object counts carry no
signal at all, and the winner is fixed only by the interaction between the
quotas on one panel and the arrival order on the other.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "quota_drain_race-renderer-0.1.0"

CANVAS = 960
BACKGROUND = (247, 247, 244)
INK = (26, 26, 26)
DECOR = (222, 222, 218)

COLOUR_NAMES = ("teal", "violet", "orange", "moss")
COLOUR_RGB = {
    "teal": (18, 140, 148),
    "violet": (128, 60, 168),
    "orange": (226, 118, 30),
    "moss": (108, 150, 44),
}

# Panel geometry.  Both panels live in one middle band; an empty corridor of
# more than two hundred pixels separates them, so the decisive support is two
# disconnected components inside a single medium-sized region.
CARD_W = 180
CARD_H = 280
CARD_ROWS = 4
CARD_COLS = 2
CARD_CELL_W = CARD_W // CARD_COLS
CARD_CELL_H = CARD_H // CARD_ROWS
CARD_X_RANGE = (180, 192)
CARD_Y_RANGE = (400, 412)

LANE_CELLS = 12
LANE_W = 104
LANE_CELL_H = 28
LANE_H = LANE_CELLS * LANE_CELL_H
LANE_X_RANGE = (650, 662)
LANE_Y_RANGE = (372, 384)

CARD_CHIP_R = 18
LANE_CHIP_R = 10
PIP_R = 5
FRAME_W = 3
GRID_W = 2

PER_COLOUR_DELIVERIES = 3
MIN_WINNER_STEP = 4
QUARANTINE_CLEARANCE = 2.0


# --------------------------------------------------------------------------
# scene mechanics
# --------------------------------------------------------------------------
def _drain(quotas: dict, lane: list[str]) -> dict:
    """Walk the lane and record when each colour's quota reaches zero."""
    paid = {colour: 0 for colour in COLOUR_NAMES}
    emptied: dict[str, int] = {}
    for step, colour in enumerate(lane):
        if colour in emptied:
            continue
        paid[colour] += 1
        if paid[colour] >= quotas[colour]:
            emptied[colour] = step
    if not emptied:
        raise ValueError("no colour ever runs out")
    order = sorted(emptied.items(), key=lambda item: item[1])
    winner, winner_step = order[0]
    runner_step = order[1][1] if len(order) > 1 else len(lane) + 1
    return {
        "emptied": emptied,
        "winner": winner,
        "winner_step": winner_step,
        "runner_up": order[1][0] if len(order) > 1 else None,
        "margin": float(runner_step - winner_step),
    }


def _shortcut_colour(quotas: dict) -> str | None:
    """Colour a quota-card-only reader would name: the unique smallest quota."""
    smallest = min(quotas.values())
    lowest = [c for c in COLOUR_NAMES if quotas[c] == smallest]
    return lowest[0] if len(lowest) == 1 else None


def _decor(rng: random.Random, blocked: list[tuple[int, int, int, int]]) -> list[dict]:
    items: list[dict] = []
    attempts = 0
    while len(items) < 14 and attempts < 400:
        attempts += 1
        kind = rng.choice(("disc", "box", "bar"))
        size = rng.randint(16, 44)
        x = rng.randint(14, CANVAS - 14 - size)
        y = rng.randint(14, CANVAS - 14 - size)
        box = (x, y, x + size, y + size)
        if any(
            box[0] < bx1 + 26
            and box[2] > bx0 - 26
            and box[1] < by1 + 26
            and box[3] > by0 - 26
            for bx0, by0, bx1, by1 in blocked
        ):
            continue
        items.append({"kind": kind, "box": list(box)})
    return items


def sample_scene(seed: int) -> dict:
    """Deterministic scene sample with a balanced, counterbalanced answer."""
    rng = random.Random(seed)
    target = COLOUR_NAMES[seed % 4]
    # The colour of the very first delivery cycles independently of the answer,
    # so over any run of sixteen seeds every (answer, first-disc) pair occurs.
    first_colour = COLOUR_NAMES[((seed // 4) + seed) % 4]
    # One scene in eight is deliberately drawn so that the unique smallest
    # quota is also the winner; those rows are quarantined so a reader who only
    # looks at the quota card is never right on a headline scene.
    want_shortcut = seed % 8 == 5
    want_tight = seed % 6 == 2
    want_wide = seed % 6 == 5
    pip_gap = 13 + (seed % 5)

    pool = [c for c in COLOUR_NAMES for _ in range(PER_COLOUR_DELIVERIES)]
    for _ in range(200000):
        quotas = {colour: rng.randint(1, 3) for colour in COLOUR_NAMES}
        lane = list(pool)
        rng.shuffle(lane)
        if lane[0] != first_colour:
            continue
        drained = _drain(quotas, lane)
        if drained["winner"] != target:
            continue
        if drained["winner_step"] < MIN_WINNER_STEP:
            continue
        if (_shortcut_colour(quotas) == drained["winner"]) != want_shortcut:
            continue
        if want_tight and drained["margin"] != 1.0:
            continue
        if want_wide and drained["margin"] < 4.0:
            continue
        break
    else:  # pragma: no cover - the rejection loop always succeeds in practice
        raise RuntimeError("could not sample a balanced scene")

    card_order = list(COLOUR_NAMES)
    rng.shuffle(card_order)
    card_x = rng.randint(*CARD_X_RANGE)
    card_y = rng.randint(*CARD_Y_RANGE)
    lane_x = rng.randint(*LANE_X_RANGE)
    lane_y = rng.randint(*LANE_Y_RANGE)
    blocked = [
        (card_x, card_y, card_x + CARD_W, card_y + CARD_H),
        (lane_x, lane_y, lane_x + LANE_W, lane_y + LANE_H),
    ]
    return {
        "canvas": CANVAS,
        "card_x": card_x,
        "card_y": card_y,
        "lane_x": lane_x,
        "lane_y": lane_y,
        "card_order": card_order,
        "quotas": {colour: quotas[colour] for colour in COLOUR_NAMES},
        "lane": lane,
        "pip_r": PIP_R,
        "pip_gap": pip_gap,
        "decor": _decor(rng, blocked),
    }


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def _pips(draw: ImageDraw.ImageDraw, cx: float, cy: float, count: int, scene: dict) -> None:
    radius = scene["pip_r"]
    gap = scene["pip_gap"]
    offset = -(count - 1) * gap / 2.0
    for index in range(count):
        px = cx + offset + index * gap
        draw.ellipse((px - radius, cy - radius, px + radius, cy + radius), fill=INK)


def _disc(draw: ImageDraw.ImageDraw, cx: float, cy: float, radius: int, colour) -> None:
    draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=colour)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (scene["canvas"], scene["canvas"]), BACKGROUND)
    draw = ImageDraw.Draw(image)

    for item in scene["decor"]:
        x0, y0, x1, y1 = item["box"]
        if item["kind"] == "disc":
            draw.ellipse((x0, y0, x1, y1), fill=DECOR)
        elif item["kind"] == "box":
            draw.rectangle((x0, y0, x1, y1), outline=DECOR, width=3)
        else:
            draw.rectangle(
                (x0, y0 + (y1 - y0) // 2, x1, y0 + (y1 - y0) // 2 + 4), fill=DECOR
            )

    # --- quota card -------------------------------------------------------
    cx0, cy0 = scene["card_x"], scene["card_y"]
    draw.rectangle((cx0, cy0, cx0 + CARD_W, cy0 + CARD_H), outline=INK, width=FRAME_W)
    x = cx0 + CARD_CELL_W
    draw.rectangle((x - GRID_W // 2, cy0, x + GRID_W // 2, cy0 + CARD_H), fill=INK)
    for row in range(1, CARD_ROWS):
        y = cy0 + row * CARD_CELL_H
        draw.rectangle((cx0, y - GRID_W // 2, cx0 + CARD_W, y + GRID_W // 2), fill=INK)
    for row, colour in enumerate(scene["card_order"]):
        cy = cy0 + row * CARD_CELL_H + CARD_CELL_H // 2
        _disc(draw, cx0 + CARD_CELL_W // 2, cy, CARD_CHIP_R, COLOUR_RGB[colour])
        _pips(
            draw,
            cx0 + CARD_CELL_W + CARD_CELL_W // 2,
            cy,
            scene["quotas"][colour],
            scene,
        )

    # --- delivery lane ----------------------------------------------------
    lx0, ly0 = scene["lane_x"], scene["lane_y"]
    draw.rectangle((lx0, ly0, lx0 + LANE_W, ly0 + LANE_H), outline=INK, width=FRAME_W)
    for cell in range(1, LANE_CELLS):
        y = ly0 + cell * LANE_CELL_H
        draw.rectangle((lx0, y - GRID_W // 2, lx0 + LANE_W, y + GRID_W // 2), fill=INK)
    for cell, colour in enumerate(scene["lane"]):
        cy = ly0 + cell * LANE_CELL_H + LANE_CELL_H // 2
        _disc(draw, lx0 + LANE_W // 2, cy, LANE_CHIP_R, COLOUR_RGB[colour])

    return image


# --------------------------------------------------------------------------
# analytic side
# --------------------------------------------------------------------------
def _replay(scene: dict) -> dict:
    return _drain(dict(scene["quotas"]), list(scene["lane"]))


def analytic_gold(scene: dict) -> str:
    return _replay(scene)["winner"]


def margin(scene: dict) -> float:
    """Runner-up margin in deliveries.

    The winner's quota empties at some cell of the lane; the runner-up's quota
    empties later.  The margin is how many further deliveries the lane has to
    hand out before the second colour also runs out, which is exactly how far
    the answer sits from flipping when arrivals are reordered.
    """
    return _replay(scene)["margin"]


def clearance(scene: dict) -> float:
    """Pixel gap between two adjacent quota dots; below two they merge."""
    return float(scene["pip_gap"] - 2 * scene["pip_r"])


def is_quarantined(scene: dict) -> bool:
    if clearance(scene) < QUARANTINE_CLEARANCE:
        return True
    return _shortcut_colour(scene["quotas"]) == analytic_gold(scene)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases: every sampled field is painted into the raster."""
    return []
