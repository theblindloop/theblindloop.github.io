"""Independent image-only inverse for the corner-guided spiral badge."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

RED = (220, 58, 62)
BLUE = (42, 112, 190)
CELL_COLORS = {(30,53,72),(230,213,169),(64,47,82),(220,226,185),(35,67,58),(232,207,180)}
DARK = {(30,53,72),(64,47,82),(35,67,58)}
KEY_BG = (236,232,242)
LABELS = ("X", "plus", "T", "L")
TEMPLATES = {
    "X": [[int(r == c or r + c == 4) for c in range(5)] for r in range(5)],
    "plus": [[int(r == 2 or c == 2) for c in range(5)] for r in range(5)],
    "T": [[int(r == 0 or c == 2) for c in range(5)] for r in range(5)],
    "L": [[int(c == 0 or r == 4) for c in range(5)] for r in range(5)],
}
CORNERS = ((0,0),(0,4),(4,4),(4,0))


def _components(mask: np.ndarray) -> list[tuple[float,float,int]]:
    work = mask.copy(); out = []
    h, w = work.shape
    for y, x in zip(*np.nonzero(work)):
        if not work[y, x]: continue
        work[y, x] = False; q = deque([(int(y),int(x))]); pts=[]
        while q:
            yy, xx = q.popleft(); pts.append((yy,xx))
            for dy, dx in ((-1,0),(1,0),(0,-1),(0,1)):
                ny,nx=yy+dy,xx+dx
                if 0<=ny<h and 0<=nx<w and work[ny,nx]:
                    work[ny,nx]=False; q.append((ny,nx))
        ys=[p[0] for p in pts]; xs=[p[1] for p in pts]
        out.append((sum(xs)/len(xs),sum(ys)/len(ys),len(pts)))
    return out


def _spiral(start: int, nxt: int) -> list[tuple[int,int]]:
    if nxt not in ((start-1)%4,(start+1)%4): raise ValueError("ambiguous fiducials")
    r,c=CORNERS[start]; nr,nc=CORNERS[nxt]
    dr=0 if nr==r else (1 if nr>r else -1); dc=0 if nc==c else (1 if nc>c else -1)
    clockwise=nxt==(start+1)%4; path=[]; used=set()
    for _ in range(25):
        path.append((r,c)); used.add((r,c)); rr,cc=r+dr,c+dc
        if not (0<=rr<5 and 0<=cc<5 and (rr,cc) not in used):
            dr,dc=((dc,-dr) if clockwise else (-dc,dr)); rr,cc=r+dr,c+dc
        r,c=rr,cc
    return path


def _one_dot(arr: np.ndarray, color: tuple[int,int,int]) -> tuple[float,float]:
    comps=[p for p in _components(np.all(arr==color,axis=2)) if 25<=p[2]<=55]
    if len(comps)!=1: raise ValueError("missing or ambiguous corner fiducial")
    return comps[0][0],comps[0][1]


def decision_from_image(image: Image.Image) -> str:
    arr=np.asarray(image.convert("RGB"))
    cell_mask=np.zeros(arr.shape[:2],dtype=bool)
    for color in CELL_COLORS: cell_mask |= np.all(arr==color,axis=2)
    comps=[p for p in _components(cell_mask) if 20<=p[2]<=30]
    if len(comps)!=25: raise ValueError("incomplete 5 by 5 panel")
    xs=sorted({round(p[0],1) for p in comps}); ys=sorted({round(p[1],1) for p in comps})
    if len(xs)!=5 or len(ys)!=5: raise ValueError("malformed cell lattice")
    grid=[[0]*5 for _ in range(5)]
    for cx,cy,_ in comps:
        c=min(range(5),key=lambda i:abs(xs[i]-cx)); r=min(range(5),key=lambda i:abs(ys[i]-cy))
        rgb=tuple(int(v) for v in arr[round(cy),round(cx)])
        grid[r][c]=int(rgb in DARK)
    red=_one_dot(arr,RED); blue=_one_dot(arr,BLUE)
    keymask=np.all(arr==KEY_BG,axis=2); ky,kx=np.nonzero(keymask)
    if not len(kx): raise ValueError("missing detached key")
    keycx=(int(kx.min())+int(kx.max()))/2; keycy=(int(ky.min())+int(ky.max()))/2
    corner_xy=((keycx-7,keycy-7),(keycx+7,keycy-7),(keycx+7,keycy+7),(keycx-7,keycy+7))
    start=min(range(4),key=lambda i:(red[0]-corner_xy[i][0])**2+(red[1]-corner_xy[i][1])**2)
    nxt=min(range(4),key=lambda i:(blue[0]-corner_xy[i][0])**2+(blue[1]-corner_xy[i][1])**2)
    stream=[grid[r][c] for r,c in _spiral(start,nxt)]
    decoded=[stream[i*5:(i+1)*5] for i in range(5)]
    scores=[]
    for label in LABELS:
        dist=sum(decoded[r][c]!=TEMPLATES[label][r][c] for r in range(5) for c in range(5))
        scores.append((25-dist,label))
    scores.sort(reverse=True)
    if scores[0][0]-scores[1][0]<3: raise ValueError("ambiguous symbol")
    return scores[0][1]
