"""Deterministic scene sampler and renderer for corner-guided spiral reassembly."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

N = 5
CELL = 5
GAP = 1
STEP = CELL + GAP
PANEL = N * CELL + (N - 1) * GAP
BG = (247, 246, 240)
VIOLET = (104, 72, 150)
RED = (220, 58, 62)
BLUE = (42, 112, 190)
PALE_INK = (218, 216, 224)
KEY_BG = (236, 232, 242)
PALETTES = (
    ((30, 53, 72), (230, 213, 169)),
    ((64, 47, 82), (220, 226, 185)),
    ((35, 67, 58), (232, 207, 180)),
)
LABELS = ("X", "plus", "T", "L")
CORNERS = ((0, 0), (0, 4), (4, 4), (4, 0))
TEMPLATES = {
    "X": [[int(r == c or r + c == 4) for c in range(5)] for r in range(5)],
    "plus": [[int(r == 2 or c == 2) for c in range(5)] for r in range(5)],
    "T": [[int(r == 0 or c == 2) for c in range(5)] for r in range(5)],
    "L": [[int(c == 0 or r == 4) for c in range(5)] for r in range(5)],
}


def _spiral(start_corner: int, next_corner: int) -> list[tuple[int, int]]:
    if next_corner not in ((start_corner - 1) % 4, (start_corner + 1) % 4):
        raise ValueError("corner markers must be adjacent")
    r, c = CORNERS[start_corner]
    nr, nc = CORNERS[next_corner]
    dr = 0 if nr == r else (1 if nr > r else -1)
    dc = 0 if nc == c else (1 if nc > c else -1)
    clockwise = next_corner == (start_corner + 1) % 4
    path: list[tuple[int, int]] = []
    used: set[tuple[int, int]] = set()
    for _ in range(25):
        path.append((r, c)); used.add((r, c))
        rr, cc = r + dr, c + dc
        if not (0 <= rr < N and 0 <= cc < N and (rr, cc) not in used):
            dr, dc = ((dc, -dr) if clockwise else (-dc, dr))
            rr, cc = r + dr, c + dc
        r, c = rr, cc
    return path


def _classify(bits: list[list[int]]) -> tuple[str, float]:
    scores = []
    for label in LABELS:
        dist = sum(bits[r][c] != TEMPLATES[label][r][c] for r in range(5) for c in range(5))
        scores.append((25 - dist, label))
    scores.sort(reverse=True)
    return scores[0][1], float(scores[0][0] - scores[1][0])


def _decoded(scene: dict) -> list[list[int]]:
    path = _spiral(int(scene["start_corner"]), int(scene["next_corner"]))
    stream = [int(scene["panel_bits"][r][c]) for r, c in path]
    return [stream[i * 5:(i + 1) * 5] for i in range(5)]


def sample_scene(seed: int) -> dict:
    # Every four consecutive seeds share nuisances while cycling through labels.
    rng = random.Random(seed // 4)
    label = LABELS[seed % len(LABELS)]
    start = rng.randrange(4)
    clockwise = rng.randrange(2) == 0
    nxt = (start + (1 if clockwise else -1)) % 4
    stream = [v for row in TEMPLATES[label] for v in row]
    panel = [[0] * 5 for _ in range(5)]
    for value, (r, c) in zip(stream, _spiral(start, nxt), strict=True):
        panel[r][c] = value
    placements = (((2, 2), (4, 4)), ((2, 5), (4, 3)), ((5, 2), (3, 4)), ((5, 5), (3, 3)))
    (pr, pc), (kr, kc) = rng.choice(placements)
    panel_cx, panel_cy = 32 + 64 * pc, 32 + 64 * pr
    key_cx, key_cy = 32 + 64 * kc, 32 + 64 * kr
    return {
        "panel_bits": panel,
        "start_corner": start,
        "next_corner": nxt,
        "palette_index": rng.randrange(len(PALETTES)),
        "offset_x": panel_cx - PANEL // 2,
        "offset_y": panel_cy - PANEL // 2,
        "key_x": key_cx,
        "key_y": key_cy,
    }


def analytic_gold(scene: dict) -> str:
    return _classify(_decoded(scene))[0]


def margin(scene: dict) -> float:
    return _classify(_decoded(scene))[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 3.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (512, 512), BG)
    draw = ImageDraw.Draw(image)
    x0, y0 = int(scene["offset_x"]), int(scene["offset_y"])
    dark, light = PALETTES[int(scene["palette_index"])]
    # Four tiny detached brackets delimit the micro-raster.
    q = 5
    for sx, sy in ((-1, -1), (1, -1), (1, 1), (-1, 1)):
        cx = x0 - 3 if sx < 0 else x0 + PANEL + 2
        cy = y0 - 3 if sy < 0 else y0 + PANEL + 2
        draw.line((cx, cy, cx + sx * q, cy), fill=VIOLET, width=2)
        draw.line((cx, cy, cx, cy + sy * q), fill=VIOLET, width=2)
    bits = scene["panel_bits"]
    for r in range(5):
        for c in range(5):
            x = x0 + c * STEP; y = y0 + r * STEP
            draw.rectangle((x, y, x + CELL - 1, y + CELL - 1), fill=dark if bits[r][c] else light)
    # The detached miniature key repeats the panel's four corner locations.
    kx, ky = int(scene["key_x"]), int(scene["key_y"])
    draw.rectangle((kx - 12, ky - 12, kx + 12, ky + 12), fill=KEY_BG, outline=VIOLET, width=2)
    for corner, color in ((int(scene["start_corner"]), RED), (int(scene["next_corner"]), BLUE)):
        r, c = CORNERS[corner]
        cx = kx + (-7 if c == 0 else 7)
        cy = ky + (-7 if r == 0 else 7)
        draw.ellipse((cx - 3, cy - 3, cx + 3, cy + 3), fill=color)
    # Remote constant marks are nondiagnostic and prevent the badge position from being the only visual structure.
    for cx, cy in ((28, 28), (484, 28), (28, 484), (484, 484)):
        draw.ellipse((cx - 3, cy - 3, cx + 3, cy + 3), outline=PALE_INK, width=1)
    return image
