"""Deterministic scene sampler and raster renderer for ordered branch rails."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 800
HEIGHT = 800
BACKGROUND = [250, 250, 247]
STROKE_WIDTH = 7
COLORS = ("red", "blue", "green", "purple")
PALETTE = {
    "red": (211, 57, 63),
    "blue": (43, 99, 176),
    "green": (28, 137, 86),
    "purple": (126, 70, 168),
}
TARGET_WORD = ("up", "down", "up", "down")
WORDS = (
    TARGET_WORD,
    ("down", "up", "down", "up"),
    ("up", "up", "down", "down"),
    ("up", "down", "down", "up"),
)
SLOTS = ((250, 150), (550, 350), (250, 550), (550, 650))
BRANCH_X = (-60, -20, 20, 60)
HALF_SPINE = 96
BRANCH_LENGTH = 20


def sample_scene(seed: int) -> dict:
    """Return a balanced deterministic scene without storing its answer."""
    rng = random.Random(int(seed))
    target_color_index = int(seed) % len(COLORS)
    word_order = [word for word in WORDS[1:]]
    rng.shuffle(word_order)
    words_by_color: list[tuple[str, ...]] = []
    distractor_index = 0
    for color_index in range(len(COLORS)):
        if color_index == target_color_index:
            words_by_color.append(TARGET_WORD)
        else:
            words_by_color.append(word_order[distractor_index])
            distractor_index += 1

    shift = (int(seed) // len(COLORS)) % len(SLOTS)
    if (int(seed) // 16) % 2:
        slot_indices = [((shift - i) % len(SLOTS)) for i in range(len(SLOTS))]
    else:
        slot_indices = [((i + shift) % len(SLOTS)) for i in range(len(SLOTS))]

    rails = []
    for color_index, color in enumerate(COLORS):
        rails.append(
            {
                "color": color,
                "center": list(SLOTS[slot_indices[color_index]]),
                "word": list(words_by_color[color_index]),
            }
        )
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
        "stroke_width": STROKE_WIDTH,
        "rails": rails,
    }


def _draw_rail(draw: ImageDraw.ImageDraw, rail: dict, stroke_width: int) -> None:
    cx, cy = (int(value) for value in rail["center"])
    rgb = PALETTE[str(rail["color"])]
    # A quiet frame makes the four candidate regions explicit but carries no label.
    draw.rounded_rectangle(
        (cx - 112, cy - 38, cx + 112, cy + 38),
        radius=12,
        outline=(219, 220, 216),
        width=2,
    )
    draw.line((cx - HALF_SPINE, cy, cx + HALF_SPINE, cy), fill=rgb, width=stroke_width)
    for offset, side in zip(BRANCH_X, rail["word"], strict=True):
        end_y = cy - BRANCH_LENGTH if side == "up" else cy + BRANCH_LENGTH
        draw.line((cx + offset, cy, cx + offset, end_y), fill=rgb, width=stroke_width)


def render(scene: dict) -> Image.Image:
    image = Image.new(
        "RGB",
        (int(scene["width"]), int(scene["height"])),
        tuple(int(value) for value in scene["background"]),
    )
    draw = ImageDraw.Draw(image)
    for rail in scene["rails"]:
        _draw_rail(draw, rail, int(scene["stroke_width"]))
    return image


def _scores(scene: dict) -> list[tuple[int, str]]:
    scores = []
    for rail in scene["rails"]:
        score = sum(a == b for a, b in zip(rail["word"], TARGET_WORD, strict=True))
        scores.append((score, str(rail["color"])))
    return sorted(scores, reverse=True)


def analytic_gold(scene: dict) -> str:
    winners = [
        str(rail["color"])
        for rail in scene["rails"]
        if tuple(rail["word"]) == TARGET_WORD
    ]
    if len(winners) != 1:
        raise ValueError(f"expected one exact target rail, found {len(winners)}")
    return winners[0]


def margin(scene: dict) -> float:
    scores = _scores(scene)
    return float(scores[0][0] - scores[1][0])


def is_quarantined(scene: dict) -> bool:
    scores = _scores(scene)
    return scores[0][0] < len(TARGET_WORD) or scores[0][0] == scores[1][0] or margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["rails"] = list(reversed(twin["rails"]))
    return [("reverse_storage_order", twin)]
