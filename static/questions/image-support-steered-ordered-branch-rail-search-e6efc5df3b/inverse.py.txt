"""Independent pixel-only inverse for ordered branch rails.

The inverse uses only the public visible palette and geometry described by the
question. It enumerates colored components, validates rail morphology, locates
the horizontal spine, recovers branch sides in x order, and requires a unique
exact winner with a runner-up Hamming margin of at least two.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE = {
    "red": (211, 57, 63),
    "blue": (43, 99, 176),
    "green": (28, 137, 86),
    "purple": (126, 70, 168),
}
TARGET = ("up", "down", "up", "down")


def _groups(values: np.ndarray) -> list[np.ndarray]:
    indices = np.flatnonzero(values)
    if not len(indices):
        return []
    cuts = np.flatnonzero(np.diff(indices) > 1) + 1
    return [group for group in np.split(indices, cuts) if len(group)]


def _component_count(mask: np.ndarray) -> int:
    pending = mask.copy()
    count = 0
    height, width = pending.shape
    while pending.any():
        y, x = (int(value) for value in np.argwhere(pending)[0])
        count += 1
        queue = deque([(y, x)])
        pending[y, x] = False
        while queue:
            cy, cx = queue.popleft()
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and pending[ny, nx]:
                    pending[ny, nx] = False
                    queue.append((ny, nx))
    return count


def _word_from_mask(mask: np.ndarray) -> tuple[str, ...] | None:
    ys, xs = np.where(mask)
    if len(xs) < 1400 or _component_count(mask) != 1:
        return None
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    if not (188 <= x1 - x0 + 1 <= 198 and 39 <= y1 - y0 + 1 <= 45):
        return None

    row_counts = mask.sum(axis=1)
    spine_y = int(np.argmax(row_counts))
    if int(row_counts[spine_y]) < 185:
        return None
    upper = mask[: max(0, spine_y - 6), :].any(axis=0)
    lower = mask[min(mask.shape[0], spine_y + 7) :, :].any(axis=0)
    branch_groups = _groups(upper | lower)
    if len(branch_groups) != 4:
        return None

    word = []
    centers = []
    for group in branch_groups:
        up_ink = int(mask[: max(0, spine_y - 6), group].sum())
        down_ink = int(mask[min(mask.shape[0], spine_y + 7) :, group].sum())
        if min(up_ink, down_ink) != 0 or max(up_ink, down_ink) < 70:
            return None
        word.append("up" if up_ink > down_ink else "down")
        centers.append(float(group.mean()))
    gaps = np.diff(centers)
    if len(gaps) != 3 or np.any(np.abs(gaps - np.median(gaps)) > 2.0):
        return None
    return tuple(word)


def analyze_image(image: Image.Image) -> dict:
    array = np.asarray(image.convert("RGB"), dtype=np.int32)
    measured = {}
    for name, rgb in PALETTE.items():
        delta = array - np.asarray(rgb, dtype=np.int32)
        mask = np.sum(delta * delta, axis=2) <= 35 * 35
        word = _word_from_mask(mask)
        if word is not None:
            measured[name] = {
                "word": word,
                "score": sum(a == b for a, b in zip(word, TARGET, strict=True)),
            }
    if not measured:
        raise ValueError("no complete colored branch rails found")
    ranked = sorted(
        ((entry["score"], name) for name, entry in measured.items()), reverse=True
    )
    best_score = ranked[0][0]
    winners = [name for score, name in ranked if score == best_score]
    runner_score = ranked[len(winners)][0] if len(ranked) > len(winners) else -1
    score_margin = best_score - runner_score
    if best_score != len(TARGET) or len(winners) != 1 or score_margin < 2:
        raise ValueError(
            f"ambiguous rail search: best={best_score}, winners={winners}, margin={score_margin}"
        )
    return {
        "decision": winners[0],
        "margin": float(score_margin),
        "rail_count": len(measured),
        "scores": {name: int(entry["score"]) for name, entry in measured.items()},
    }


def decision_from_image(image: Image.Image) -> str:
    return str(analyze_image(image)["decision"])
