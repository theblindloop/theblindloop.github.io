"""Pixel-only inverse arm for the collinear dot-lane search world.

The oracle receives a rasterized image and nothing else.  It recovers the two
bands and three columns of framed panels from drawn ink, isolates the dark dots
inside each panel, groups the dot pixels into blobs, fits a total-least-squares
line to the blob centres of every lane, and selects the lane whose largest
perpendicular residual is smallest.  It abstains when the layout or the blob
count is not recovered or when the runner-up gap is too small.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

AMBIGUOUS = "ambiguous"

PANEL_COLUMNS = 3
PANEL_ROWS = 2
N_PANELS = PANEL_COLUMNS * PANEL_ROWS
DOTS_PER_LANE = 5

DRAWN_THRESHOLD = 40  # frames and dots
INK_THRESHOLD = 330  # dots only; frames are mid-tone
BLOB_RADIUS = 15.0
MIN_BLOB_PIXELS = 60
MAX_BLOB_PIXELS = 400
PANEL_INSET = 8
ABSTAIN_MARGIN_PX = 4.0


def _modal_colour(pixels: np.ndarray) -> np.ndarray:
    flat = pixels.reshape(-1, 3).astype(np.int64)
    keys = (flat[:, 0] << 16) | (flat[:, 1] << 8) | flat[:, 2]
    values, counts = np.unique(keys, return_counts=True)
    key = int(values[int(np.argmax(counts))])
    return np.array([(key >> 16) & 255, (key >> 8) & 255, key & 255], dtype=np.int64)


def _runs(flags: np.ndarray, min_length: int = 6) -> list:
    runs = []
    start = None
    for index, flag in enumerate(flags):
        if flag and start is None:
            start = index
        elif not flag and start is not None:
            if index - start >= min_length:
                runs.append((start, index))
            start = None
    if start is not None and len(flags) - start >= min_length:
        runs.append((start, len(flags)))
    return runs


def _blob_centres(mask: np.ndarray) -> list:
    """Group dot pixels into well-separated blobs and return their centroids."""
    coords = np.argwhere(mask)  # rows of (y, x)
    centres = []
    while coords.shape[0] > 0 and len(centres) <= DOTS_PER_LANE:
        seed = coords[0].astype(np.float64)
        close = np.hypot(coords[:, 0] - seed[0], coords[:, 1] - seed[1]) <= BLOB_RADIUS
        blob = coords[close]
        if not (MIN_BLOB_PIXELS <= blob.shape[0] <= MAX_BLOB_PIXELS):
            return []
        centres.append((float(blob[:, 1].mean()), float(blob[:, 0].mean())))
        coords = coords[~close]
    if coords.shape[0] > 0 or len(centres) != DOTS_PER_LANE:
        return []
    return centres


def _deviation(points) -> float:
    count = len(points)
    mx = sum(p[0] for p in points) / count
    my = sum(p[1] for p in points) / count
    sxx = sum((p[0] - mx) ** 2 for p in points)
    syy = sum((p[1] - my) ** 2 for p in points)
    sxy = sum((p[0] - mx) * (p[1] - my) for p in points)
    theta = 0.5 * math.atan2(2.0 * sxy, sxx - syy)
    nx, ny = -math.sin(theta), math.cos(theta)
    return max(abs((p[0] - mx) * nx + (p[1] - my) * ny) for p in points)


def analyse(image: Image.Image) -> dict:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int64)
    background = _modal_colour(pixels)
    distance = np.abs(pixels - background).sum(axis=2)
    drawn = distance > DRAWN_THRESHOLD
    ink = distance > INK_THRESHOLD

    empty = {"decision": AMBIGUOUS, "margin": 0.0, "deviations": []}
    bands = _runs(drawn.any(axis=1))
    if len(bands) != PANEL_ROWS:
        return empty

    boxes = []
    for band_top, band_bottom in bands:
        columns = _runs(drawn[band_top:band_bottom, :].any(axis=0))
        if len(columns) != PANEL_COLUMNS:
            return empty
        for left, right in columns:
            boxes.append((left, right, band_top, band_bottom))

    deviations = []
    for left, right, top, bottom in boxes:
        window = ink[
            top + PANEL_INSET : bottom - PANEL_INSET,
            left + PANEL_INSET : right - PANEL_INSET,
        ]
        if window.size == 0:
            return empty
        centres = _blob_centres(window)
        if len(centres) != DOTS_PER_LANE:
            return empty
        deviations.append(_deviation(centres))

    order = sorted(range(N_PANELS), key=lambda index: deviations[index])
    gap = deviations[order[1]] - deviations[order[0]]
    if gap < ABSTAIN_MARGIN_PX:
        return {
            "decision": AMBIGUOUS,
            "margin": round(float(gap), 9),
            "deviations": deviations,
        }
    return {
        "decision": f"panel-{order[0] + 1}",
        "margin": round(float(gap), 9),
        "deviations": deviations,
    }


def decision_from_image(image: Image.Image) -> str:
    return analyse(image)["decision"]


def margin_from_image(image: Image.Image) -> float:
    return round(float(analyse(image)["margin"]), 9)
