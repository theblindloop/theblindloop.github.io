"""Independent pixel-only inverse arm for the diag_distance_compare world.

Receives ONLY the rasterized PNG (plus the world's public color naming
convention) and re-measures, from pixels alone, the perpendicular distance of
each colored dot's center to the gray diagonal line, then answers which color
has the larger distance. It shares no geometry or code with renderer.render(),
and it never imports the renderer, prompts, generator, verifier, or any
analytic-gold code.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "diag_distance_compare-oracle-0.1.0"

PALETTE = ["red", "blue"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
}
LINE_RGB = (70, 70, 70)
BACKGROUND = (250, 250, 250)


def _distance_to_line(point, center, direction) -> float:
    """Perpendicular distance from point to the line through center with `direction`."""
    dx, dy = direction
    vx = point[0] - center[0]
    vy = point[1] - center[1]
    return abs(dx * vy - dy * vx)


def _fit_line(points: np.ndarray):
    """Return (center, unit direction) of the best-fit line through gray pixels."""
    center = points.mean(axis=0)
    centered = points - center
    cov = centered.T @ centered
    eigenvalues, eigenvectors = np.linalg.eigh(cov)
    principal = eigenvectors[:, int(np.argmax(eigenvalues))]
    norm = float(np.linalg.norm(principal))
    if norm == 0:
        return center, np.array([1.0, 0.0])
    return center, principal / norm


def _decision_from_segmentation(labels: np.ndarray) -> str:
    """Return the farther-dot color from a nearest-color label array."""
    # Gray diagonal reference line: fit from all gray-labelled pixels.
    ref_line_idx = 2
    ly = int(ref_line_idx)
    gray_ys, gray_xs = np.nonzero(labels == ly)
    if gray_ys.size < 20:
        return "none"
    gray_points = np.stack([gray_xs.astype(np.float64), gray_ys.astype(np.float64)], axis=1)
    center, direction = _fit_line(gray_points)

    distances: dict[str, float] = {}
    for idx, color in enumerate(PALETTE):
        ys, xs = np.nonzero(labels == idx)
        if ys.size == 0:
            return "none"
        dot_center = np.array([xs.mean(), ys.mean()])
        distances[color] = _distance_to_line(dot_center, center, direction)

    d_red = distances["red"]
    d_blue = distances["blue"]
    if abs(d_red - d_blue) < 0.5:
        return "none"
    return "red" if d_red > d_blue else "blue"


def decision_from_image(image: Image.Image) -> str:
    """Color of the dot whose perpendicular distance to the diagonal is larger."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    refs = np.stack(
        [
            np.array(COLOR_RGB[c], dtype=np.float32) for c in PALETTE
        ]
        + [np.array(LINE_RGB, dtype=np.float32)]
        + [np.array(BACKGROUND, dtype=np.float32)],
        axis=0,
    )
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    norms = np.linalg.norm(diffs, axis=-1)
    labels = np.argmin(norms, axis=-1)
    return _decision_from_segmentation(labels)
