"""Measurement world: which colored dot is farther from the gray diagonal.

A dark-gray diagonal reference segment is drawn across a light canvas. Two
solid colored disks (red and blue) sit on the canvas. The decision measures,
for each colored dot, the perpendicular distance from that dot's center to the
diagonal line, and answers which color has the larger perpendicular distance.

Latent scene z carries ``red`` and ``blue`` (each a dot center (x, y)) and
``line`` (the two endpoints of the diagonal segment). The reference line is
always rendered as visible gray ink and the two dot colors are visible marks,
so every latent field is directly readable from the raster. Dot positions are
sampled so that neither color, position, nor ink amount alone can reveal the
winner (which color is farther is randomized); the measured quantity is a
per-point perpendicular distance to a shared reference line.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "diag_distance_compare-0.1.0"

PALETTE = ["red", "blue"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
}
LINE_RGB = (70, 70, 70)
BACKGROUND = (250, 250, 250)

WIDTH = 520
HEIGHT = 400

DOT_RADIUS = 14

# Fixed diagonal reference segment (drawn as visible gray ink every scene).
LINE_A = (80, 60)
LINE_B = (WIDTH - 80, HEIGHT - 62)

# Minimum center distance from the line (keeps the disk clear of the line ink).
MIN_LINE_CLEARANCE = 22
# Minimum perpendicular-distance gap between winner and loser for emission.
MIN_MARGIN = 9.0
# Below this distance gap the scene is a finite-resolution boundary case.
QUARANTINE = 6.0


def _distance_to_line(point, a, b) -> float:
    """Perpendicular distance from point to the infinite line through a-b."""
    x, y = point
    ax, ay = a
    bx, by = b
    dx, dy = bx - ax, by - ay
    length = math.hypot(dx, dy)
    if length == 0:
        return 0.0
    return abs(dx * (y - ay) - dy * (x - ax)) / length


def _inside_region(rng) -> tuple[float, float]:
    """A random dot center comfortably inside the canvas."""
    margin = int(DOT_RADIUS) + 8
    x = rng.uniform(margin + 10, WIDTH - margin - 10)
    y = rng.uniform(margin + 10, HEIGHT - margin - 10)
    return (x, y)


def sample_scene(seed: int) -> dict:
    """Deterministic scene with a guaranteed, counterbalanced farther dot."""
    rng = random.Random(seed * 7919 + 3)
    line = (LINE_A, LINE_B)
    winner_color = PALETTE[seed % 2]
    loser_color = next(c for c in PALETTE if c != winner_color)

    while True:
        pw = _inside_region(rng)
        dw = _distance_to_line(pw, *line)
        if dw < MIN_LINE_CLEARANCE:
            continue
        pl = _inside_region(rng)
        dl = _distance_to_line(pl, *line)
        if dl < MIN_LINE_CLEARANCE:
            continue
        if math.hypot(pl[0] - pw[0], pl[1] - pw[1]) < 3 * DOT_RADIUS:
            continue
        if dw - dl < MIN_MARGIN:
            continue
        break

    red = pw if winner_color == "red" else pl
    blue = pw if winner_color == "blue" else pl
    return {
        "red": (round(red[0], 2), round(red[1], 2)),
        "blue": (round(blue[0], 2), round(blue[1], 2)),
        "line": ((LINE_A[0], LINE_A[1]), (LINE_B[0], LINE_B[1])),
        "width": WIDTH,
        "height": HEIGHT,
        "background": "light",
    }


def render(scene: dict) -> Image.Image:
    """Render I = R(z): gray diagonal segment plus two colored dots."""
    img = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BACKGROUND)
    draw = ImageDraw.Draw(img)
    (ax, ay), (bx, by) = scene["line"]
    draw.line([(ax, ay), (bx, by)], fill=LINE_RGB, width=4)
    r = DOT_RADIUS
    red_cx, red_cy = scene["red"]
    draw.ellipse([red_cx - r, red_cy - r, red_cx + r, red_cy + r], fill=COLOR_RGB["red"])
    blue_cx, blue_cy = scene["blue"]
    draw.ellipse([blue_cx - r, blue_cy - r, blue_cx + r, blue_cy + r], fill=COLOR_RGB["blue"])
    return img


def analytic_gold(scene: dict) -> str:
    """Color of the dot whose perpendicular distance to the diagonal is larger."""
    line = scene["line"]
    d_red = _distance_to_line(scene["red"], line[0], line[1])
    d_blue = _distance_to_line(scene["blue"], line[0], line[1])
    return "red" if d_red > d_blue else "blue"


def margin(scene: dict) -> float:
    """Verifiability margin: the perpendicular-distance gap (pixels)."""
    line = scene["line"]
    d_red = _distance_to_line(scene["red"], line[0], line[1])
    d_blue = _distance_to_line(scene["blue"], line[0], line[1])
    return float(abs(d_red - d_blue))


def is_quarantined(scene: dict) -> bool:
    """Near-boundary: the perpendicular-distance gap is finite-resolution small."""
    return margin(scene) < QUARANTINE


def latent_symmetries(scene: dict) -> list:
    """not_applicable: no nontrivial pixel-identical latent symmetry.

    Every latent field (the red dot center, the blue dot center, and the two
    endpoints of the diagonal) is directly rendered as visible ink: the dot
    centers are the drawn dot centers and the diagonal is the drawn gray
    segment. Swapping the two colors changes which color is correct, and any
    global reflection/rotation turns the raster into a distinct image while
    changing the relative distances, so no latent transform is both pixel-
    identical and gold-preserving.
    """
    return []
