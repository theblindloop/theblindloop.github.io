"""Independent image-only inverse for the second-difference triptych.

This module reimplements fiducial recovery, crop geometry, glyph templates, and
the three-panel arithmetic from public question text. It imports no renderer,
prompt, generator, verifier, or analytic-gold code and accepts only a final PNG.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image, ImageDraw

SYMBOLS = ("ring", "cross", "arrow", "wave")
RED = np.array((224, 58, 62), dtype=np.float32)
GREEN = np.array((47, 170, 92), dtype=np.float32)
BLUE = np.array((54, 92, 224), dtype=np.float32)
BASE_LEVEL = 128.0
SIGNAL_LEVEL = 46.0
MARGIN_QUARANTINE = 0.06


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[float, float, int]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in (
                (cy - 1, cx),
                (cy + 1, cx),
                (cy, cx - 1),
                (cy, cx + 1),
            ):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= 20:
            ys = np.array([point[0] for point in points], dtype=np.float64)
            xs = np.array([point[1] for point in points], dtype=np.float64)
            found.append((float(xs.mean()), float(ys.mean()), len(points)))
    return found


def _role_interior(arr: np.ndarray, color: np.ndarray) -> np.ndarray:
    distance = np.linalg.norm(arr - color[None, None, :], axis=2)
    marks = _components(distance < 34.0)
    if len(marks) != 4:
        raise ValueError(f"expected four role fiducials, found {len(marks)}")
    xs = sorted(mark[0] for mark in marks)
    ys = sorted(mark[1] for mark in marks)
    left = 0.5 * (xs[0] + xs[1])
    right = 0.5 * (xs[2] + xs[3])
    top = 0.5 * (ys[0] + ys[1])
    bottom = 0.5 * (ys[2] + ys[3])
    size = int(round((right - left) - 24.0))
    if size < 140 or size > 220 or abs((bottom - top) - (size + 24)) > 3.0:
        raise ValueError("role fiducials do not define a supported square")
    x0 = int(round(left + 12.0))
    y0 = int(round(top + 12.0))
    if x0 < 0 or y0 < 0 or x0 + size > arr.shape[1] or y0 + size > arr.shape[0]:
        raise ValueError("role crop is outside image")
    return arr[y0 : y0 + size, x0 : x0 + size, 0].astype(np.float32)


def _glyph_mask(symbol: str, size: int, stroke_variant: int) -> np.ndarray:
    canvas = Image.new("L", (int(size), int(size)), 0)
    draw = ImageDraw.Draw(canvas)
    width = max(8, int(round(size * (0.052 + 0.004 * int(stroke_variant)))))
    s = float(size - 1)
    if symbol == "ring":
        pad_x = int(round(size * 0.23))
        pad_y = int(round(size * 0.18))
        draw.ellipse((pad_x, pad_y, size - 1 - pad_x, size - 1 - pad_y), outline=255, width=width)
    elif symbol == "cross":
        cx = int(round(s * 0.50))
        draw.line((cx, int(round(s * 0.18)), cx, int(round(s * 0.82))), fill=255, width=width)
        draw.line((int(round(s * 0.18)), cx, int(round(s * 0.82)), cx), fill=255, width=width)
    elif symbol == "arrow":
        cx = int(round(s * 0.50))
        draw.line((cx, int(round(s * 0.80)), cx, int(round(s * 0.23))), fill=255, width=width)
        draw.line(
            (int(round(s * 0.29)), int(round(s * 0.42)), cx, int(round(s * 0.18)), int(round(s * 0.71)), int(round(s * 0.42))),
            fill=255,
            width=width,
            joint="curve",
        )
    else:
        points = []
        for index in range(49):
            x = 0.16 + 0.68 * index / 48.0
            y = 0.50 + 0.23 * np.sin(2.0 * np.pi * (x - 0.16) / 0.68)
            points.append((int(round(s * x)), int(round(s * y))))
        draw.line(points, fill=255, width=width, joint="curve")
    return np.asarray(canvas, dtype=np.float32) / 255.0


def _scores(transformed: np.ndarray) -> list[tuple[float, str]]:
    size = int(transformed.shape[0])
    lo = max(0, int(round(0.10 * size)))
    hi = min(size, int(round(0.90 * size)))
    scores: list[tuple[float, str]] = []
    for symbol in SYMBOLS:
        best = float("inf")
        for variant in (0, 1):
            target = BASE_LEVEL - SIGNAL_LEVEL * _glyph_mask(symbol, size, variant)
            mse = float(np.mean((transformed[lo:hi, lo:hi] - target[lo:hi, lo:hi]) ** 2))
            best = min(best, mse)
        scores.append((best, symbol))
    return sorted(scores, key=lambda item: (item[0], item[1]))


def confidence_margin(image: Image.Image) -> float:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    red = _role_interior(arr, RED)
    green = _role_interior(arr, GREEN)
    blue = _role_interior(arr, BLUE)
    if red.shape != green.shape or red.shape != blue.shape:
        raise ValueError("role panels have different sizes")
    # The declared second difference: the shared carrier cancels exactly.
    transformed = 0.5 * (red + blue) - green + BASE_LEVEL
    ranked = _scores(transformed)
    return float((ranked[1][0] - ranked[0][0]) / 1000.0)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    red = _role_interior(arr, RED)
    green = _role_interior(arr, GREEN)
    blue = _role_interior(arr, BLUE)
    if red.shape != green.shape or red.shape != blue.shape:
        raise ValueError("role panels have different sizes")
    transformed = 0.5 * (red + blue) - green + BASE_LEVEL
    ranked = _scores(transformed)
    gap = float((ranked[1][0] - ranked[0][0]) / 1000.0)
    if not np.isfinite(gap) or gap < MARGIN_QUARANTINE:
        raise ValueError(f"ambiguous second-difference reconstruction margin {gap:.4f}")
    return ranked[0][1]
