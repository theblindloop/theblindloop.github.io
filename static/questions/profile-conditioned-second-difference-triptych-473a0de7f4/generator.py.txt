"""Deterministic renderer and analytic gold for the triptych transform world."""

from __future__ import annotations

import itertools
import random

import numpy as np
from PIL import Image, ImageDraw

RENDERER_VERSION = "second_difference_triptych-0.1.0"
CANVAS_SIZE = (1200, 520)
SYMBOLS = ("ring", "cross", "arrow", "wave")
ROLES = ("red", "green", "blue")
ROLE_COLORS = {
    "red": (224, 58, 62),
    "green": (47, 170, 92),
    "blue": (54, 92, 224),
}
PANEL_SIZES = (168, 184, 200)
BACKGROUND = (249, 249, 249)
PANEL_BACKGROUND = (236, 236, 236)
FRAME = (72, 72, 72)
MARK_RADIUS = 7
BASE_LEVEL = 128.0
SIGNAL_LEVEL = 46.0
MARGIN_QUARANTINE = 0.06


def _glyph_mask(symbol: str, size: int, stroke_variant: int = 0) -> np.ndarray:
    """Draw one broad, low-frequency glyph; the transformed image uses this mask."""

    if symbol not in SYMBOLS:
        raise ValueError(f"unknown symbol {symbol!r}")
    canvas = Image.new("L", (int(size), int(size)), 0)
    draw = ImageDraw.Draw(canvas)
    width = max(8, int(round(size * (0.052 + 0.004 * int(stroke_variant)))))
    s = float(size - 1)
    if symbol == "ring":
        pad_x = int(round(size * 0.23))
        pad_y = int(round(size * 0.18))
        draw.ellipse((pad_x, pad_y, size - 1 - pad_x, size - 1 - pad_y), outline=255, width=width)
    elif symbol == "cross":
        cx = int(round(s * 0.50))
        draw.line((cx, int(round(s * 0.18)), cx, int(round(s * 0.82))), fill=255, width=width)
        draw.line((int(round(s * 0.18)), cx, int(round(s * 0.82)), cx), fill=255, width=width)
    elif symbol == "arrow":
        cx = int(round(s * 0.50))
        draw.line((cx, int(round(s * 0.80)), cx, int(round(s * 0.23))), fill=255, width=width)
        draw.line(
            (int(round(s * 0.29)), int(round(s * 0.42)), cx, int(round(s * 0.18)), int(round(s * 0.71)), int(round(s * 0.42))),
            fill=255,
            width=width,
            joint="curve",
        )
    else:  # wave
        points = []
        for index in range(49):
            x = 0.16 + 0.68 * index / 48.0
            y = 0.50 + 0.23 * np.sin(2.0 * np.pi * (x - 0.16) / 0.68)
            points.append((int(round(s * x)), int(round(s * y))))
        draw.line(points, fill=255, width=width, joint="curve")
    return np.asarray(canvas, dtype=np.float32) / 255.0


def _carrier(size: int, phase: int) -> np.ndarray:
    """High-frequency common carrier shared by all three role panels."""

    generator = np.random.default_rng(31000 + int(size) * 97 + int(phase) * 131)
    noise = generator.integers(-50, 51, size=(int(size), int(size))).astype(np.float32)
    yy, xx = np.indices((int(size), int(size)), dtype=np.float32)
    ripple = 9.0 * np.sin((xx + 2.0 * yy + 3.0 * int(phase)) * np.pi / 3.0)
    return noise + ripple


def _glyph_level(scene: dict) -> np.ndarray:
    size = int(scene["panel_size"])
    mask = _glyph_mask(str(scene["glyph"]), size, int(scene["stroke_variant"]))
    return BASE_LEVEL - SIGNAL_LEVEL * mask


def _panel_pixels(scene: dict, role: str) -> np.ndarray:
    size = int(scene["panel_size"])
    carrier = _carrier(size, int(scene["carrier_phase"]))
    values = BASE_LEVEL + carrier
    if role in ("red", "blue"):
        values = values - SIGNAL_LEVEL * _glyph_mask(
            str(scene["glyph"]), size, int(scene["stroke_variant"])
        )
    return np.clip(np.rint(values), 0, 255).astype(np.uint8)


def _origins(scene: dict) -> tuple[tuple[int, int], tuple[int, int], tuple[int, int]]:
    size = int(scene["panel_size"])
    outer = size + 48
    presets = (
        ((62, 124), (400, 78), (738, 136)),
        ((48, 76), (392, 148), (736, 96)),
        ((72, 154), (402, 102), (732, 70)),
        ((40, 120), (386, 62), (730, 156)),
    )
    chosen = presets[int(scene["layout"]) % len(presets)]
    if max(x + outer for x, _ in chosen) > CANVAS_SIZE[0] - 20:
        raise ValueError("panel layout exceeds canvas")
    if max(y + outer for _, y in chosen) > CANVAS_SIZE[1] - 12:
        raise ValueError("panel layout exceeds canvas")
    return chosen


def _draw_panel(image: Image.Image, origin: tuple[int, int], pixels: np.ndarray, role: str) -> None:
    x0, y0 = int(origin[0]), int(origin[1])
    size = int(pixels.shape[0])
    outer = size + 48
    tile = Image.fromarray(pixels, mode="L").convert("RGB")
    image.paste(tile, (x0 + 24, y0 + 24))
    draw = ImageDraw.Draw(image)
    draw.rectangle((x0, y0, x0 + outer - 1, y0 + outer - 1), outline=FRAME, width=2)
    centers = (
        (x0 + 12, y0 + 12),
        (x0 + size + 36, y0 + 12),
        (x0 + 12, y0 + size + 36),
        (x0 + size + 36, y0 + size + 36),
    )
    for cx, cy in centers:
        draw.ellipse(
            (cx - MARK_RADIUS, cy - MARK_RADIUS, cx + MARK_RADIUS, cy + MARK_RADIUS),
            fill=ROLE_COLORS[role],
        )


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS_SIZE, BACKGROUND)
    origins = _origins(scene)
    order = list(itertools.permutations(ROLES))[int(scene["role_order"]) % 6]
    for origin, role in zip(origins, order):
        _draw_panel(image, origin, _panel_pixels(scene, role), role)
    return image


def sample_scene(seed: int) -> dict:
    seed = int(seed)
    rng = random.Random(seed)
    return {
        "seed": seed,
        "glyph": SYMBOLS[(seed + 1) % len(SYMBOLS)],
        "role_order": rng.randrange(6),
        "layout": rng.randrange(4),
        "panel_size": PANEL_SIZES[rng.randrange(len(PANEL_SIZES))],
        "stroke_variant": rng.randrange(2),
        "carrier_phase": rng.randrange(12),
    }


def _scores(transformed: np.ndarray) -> list[tuple[float, str]]:
    size = int(transformed.shape[0])
    lo = max(0, int(round(0.10 * size)))
    hi = min(size, int(round(0.90 * size)))
    scores: list[tuple[float, str]] = []
    for symbol in SYMBOLS:
        best = float("inf")
        for variant in (0, 1):
            target = _glyph_level({"glyph": symbol, "panel_size": size, "stroke_variant": variant})
            mse = float(np.mean((transformed[lo:hi, lo:hi] - target[lo:hi, lo:hi]) ** 2))
            best = min(best, mse)
        scores.append((best, symbol))
    return sorted(scores, key=lambda item: (item[0], item[1]))


def analytic_gold(scene: dict) -> str:
    return str(scene["glyph"])


def margin(scene: dict) -> float:
    ranked = _scores(_glyph_level(scene))
    return max(0.0, float((ranked[1][0] - ranked[0][0]) / 1000.0))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < MARGIN_QUARANTINE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # No aliases are declared: role identity, layout, size, carrier phase, and
    # stroke are rendered nuisance fields; glyph is the sole analytic label input.
    return []
