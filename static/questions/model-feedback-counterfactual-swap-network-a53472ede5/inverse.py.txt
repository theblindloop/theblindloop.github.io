"""Independent image-only inverse for the visible swap-network grammar."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

WIDTH = 420
HEIGHT = 420
LANE_X = (54, 132, 210, 288, 366)
GATE_Y = (116, 151, 186, 221, 256, 291, 326)
TOKEN_RGB = {
    "A": (242, 169, 0),
    "B": (25, 118, 210),
    "C": (46, 157, 80),
    "D": (123, 75, 196),
    "E": (230, 75, 106),
}
INK_RGB = (23, 33, 43)
RED_RGB = (213, 38, 47)


class OracleError(ValueError):
    pass


def _near(arr: np.ndarray, rgb: tuple[int, int, int], radius: float) -> np.ndarray:
    delta = arr.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2, dtype=np.int32) <= radius * radius


def _nearest_index(value: float, choices: tuple[int, ...], tolerance: float) -> int:
    index = min(range(len(choices)), key=lambda i: abs(choices[i] - value))
    if abs(choices[index] - value) > tolerance:
        raise OracleError(f"mark at {value:.1f} is outside the visible grammar")
    return index


def _recover_top_order(arr: np.ndarray) -> list[str]:
    order: list[str | None] = [None] * len(LANE_X)
    for identity, rgb in TOKEN_RGB.items():
        mask = _near(arr[:92], rgb, 48.0)
        ys, xs = np.nonzero(mask)
        if len(xs) < 650:
            raise OracleError(f"token {identity} has too little recoverable color")
        cx = float(xs.mean())
        lane = _nearest_index(cx, LANE_X, 8.0)
        if order[lane] is not None:
            raise OracleError("two token colors recover to one lane")
        order[lane] = identity
    if any(value is None for value in order):
        raise OracleError("not all five top lanes contain one token")
    return [str(value) for value in order]


def _recover_gates(arr: np.ndarray) -> list[int]:
    ink = _near(arr, INK_RGB, 65.0)
    gates: list[int] = []
    for y in GATE_Y:
        scores = []
        for left in range(4):
            midpoint = (LANE_X[left] + LANE_X[left + 1]) // 2
            patch = ink[y - 7 : y + 8, midpoint - 7 : midpoint + 8]
            scores.append(int(patch.sum()))
        best = max(range(4), key=scores.__getitem__)
        ranked = sorted(scores, reverse=True)
        if ranked[0] < 35 or ranked[1] > 12:
            raise OracleError(f"gate row {y} is missing or has multiple crossings: {scores}")
        gates.append(best)
    return gates


def _recover_intervention(arr: np.ndarray, gates: list[int]) -> int:
    red = _near(arr, RED_RGB, 45.0)
    mask = red.copy()
    mask[:92] = False
    mask[350:] = False
    ys, xs = np.nonzero(mask)
    if len(xs) < 300:
        raise OracleError("red intervention box is missing")
    cy, cx = float(ys.mean()), float(xs.mean())
    row = _nearest_index(cy, GATE_Y, 7.0)
    midpoints = tuple((LANE_X[i] + LANE_X[i + 1]) // 2 for i in range(4))
    pair = _nearest_index(cx, midpoints, 8.0)
    if gates[row] != pair:
        raise OracleError("red intervention box does not surround the recovered crossing")
    return row


def _recover_target(arr: np.ndarray) -> int:
    red = _near(arr, RED_RGB, 45.0)
    mask = red.copy()
    mask[:350] = False
    ys, xs = np.nonzero(mask)
    if len(xs) < 180:
        raise OracleError("red target ring is missing")
    if not 372.0 <= float(ys.mean()) <= 392.0:
        raise OracleError("red destination mark is misplaced")
    return _nearest_index(float(xs.mean()), LANE_X, 7.0)


def decision_from_image(image: Image.Image) -> str:
    if image.size != (WIDTH, HEIGHT):
        raise OracleError(f"expected {(WIDTH, HEIGHT)} image, got {image.size}")
    arr = np.asarray(image.convert("RGB"))
    order = _recover_top_order(arr)
    gates = _recover_gates(arr)
    skipped = _recover_intervention(arr, gates)
    target = _recover_target(arr)
    state = list(order)
    for index, left in enumerate(gates):
        if index == skipped:
            continue
        state[left], state[left + 1] = state[left + 1], state[left]
    answer = state[target]
    if answer not in TOKEN_RGB:
        raise OracleError("recovered token identity is invalid")
    return answer
