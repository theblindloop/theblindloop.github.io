"""Deterministic Pillow renderer for a counterfactual swap network."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 420
HEIGHT = 420
SCALE = 3
LANE_X = (54, 132, 210, 288, 366)
GATE_Y = (116, 151, 186, 221, 256, 291, 326)
TOP_Y = 57
TRACK_TOP = 82
TRACK_BOTTOM = 365
TARGET_Y = 382
TOKEN_COLORS = {
    "A": "#F2A900",
    "B": "#1976D2",
    "C": "#2E9D50",
    "D": "#7B4BC4",
    "E": "#E64B6A",
}
INK = "#17212B"
RED = "#D5262F"
GRAY = "#86909A"


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size * SCALE
    )


def _simulate(token_order: list[str], gates: list[int], skipped: int | None) -> list[str]:
    state = list(token_order)
    for index, left in enumerate(gates):
        if skipped is not None and index == skipped:
            continue
        state[left], state[left + 1] = state[left + 1], state[left]
    return state


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    tokens = list(TOKEN_COLORS)
    rng.shuffle(tokens)

    # Adjacent repetitions are forbidden, which keeps every drawn X isolated
    # from an identical immediately following undo while retaining varied paths.
    gates: list[int] = []
    while len(gates) < len(GATE_Y):
        candidate = rng.randrange(4)
        if not gates or candidate != gates[-1]:
            gates.append(candidate)
    intervention_index = rng.randrange(1, len(gates) - 1)

    factual = _simulate(tokens, gates, skipped=None)
    counterfactual = _simulate(tokens, gates, skipped=intervention_index)
    changed_lanes = [i for i, pair in enumerate(zip(factual, counterfactual)) if pair[0] != pair[1]]
    # Removing a transposition always changes exactly two final identities. Query
    # one of those lanes so the intervention is causally relevant in every scene.
    target_lane = changed_lanes[rng.randrange(len(changed_lanes))]
    return {
        "token_order": tokens,
        "gates": gates,
        "intervention_index": intervention_index,
        "target_lane": target_lane,
    }


def analytic_gold(scene: dict[str, Any]) -> str:
    token_order = [str(value) for value in scene["token_order"]]
    gates = [int(value) for value in scene["gates"]]
    intervention_index = int(scene["intervention_index"])
    target_lane = int(scene["target_lane"])
    return _simulate(token_order, gates, skipped=intervention_index)[target_lane]


def margin(scene: dict[str, Any]) -> float:
    # Minimum center-to-mark clearance in the fixed visual grammar. It is
    # intentionally conservative and far above antialiasing uncertainty.
    return 12.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        tokens = list(scene["token_order"])
        gates = [int(value) for value in scene["gates"]]
        skipped = int(scene["intervention_index"])
        target = int(scene["target_lane"])
    except (KeyError, TypeError, ValueError):
        return True
    return not (
        sorted(tokens) == list("ABCDE")
        and len(gates) == len(GATE_Y)
        and all(0 <= gate < 4 for gate in gates)
        and all(a != b for a, b in zip(gates, gates[1:]))
        and 0 < skipped < len(gates) - 1
        and 0 <= target < 5
    )


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict]]:
    return []


def render(scene: dict[str, Any]) -> Image.Image:
    token_order = [str(value) for value in scene["token_order"]]
    gates = [int(value) for value in scene["gates"]]
    intervention_index = int(scene["intervention_index"])
    target_lane = int(scene["target_lane"])

    canvas = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), "white")
    draw = ImageDraw.Draw(canvas)
    sx = lambda value: int(round(value * SCALE))

    # Short instruction labels are redundant with the prompt but make the two
    # special marks visually self-identifying.
    draw.text((sx(14), sx(12)), "TOKENS", fill=GRAY, font=_font(12))

    track_width = sx(4)
    for x in LANE_X:
        draw.line((sx(x), sx(TRACK_TOP), sx(x), sx(TRACK_BOTTOM)), fill=INK, width=track_width)

    # Render each swap as two continuous diagonals after erasing the local
    # verticals. There is exactly one crossing at each distinct height.
    half_h = 12
    for index, (y, left) in enumerate(zip(GATE_Y, gates)):
        x0, x1 = LANE_X[left], LANE_X[left + 1]
        erase_width = sx(10)
        for x in (x0, x1):
            draw.line(
                (sx(x), sx(y - half_h - 1), sx(x), sx(y + half_h + 1)),
                fill="white",
                width=erase_width,
            )
        draw.line(
            (sx(x0), sx(y - half_h), sx(x1), sx(y + half_h)),
            fill=INK,
            width=track_width,
        )
        draw.line(
            (sx(x1), sx(y - half_h), sx(x0), sx(y + half_h)),
            fill=INK,
            width=track_width,
        )
        if index == intervention_index:
            pad_x, pad_y = 13, 18
            draw.rounded_rectangle(
                (sx(x0 - pad_x), sx(y - pad_y), sx(x1 + pad_x), sx(y + pad_y)),
                radius=sx(7),
                outline=RED,
                width=sx(4),
            )

    # Identity disks and large white letters provide redundant visible binding;
    # the independent oracle identifies disks by their fixed, separated colors.
    radius = 23
    font = _font(24)
    for lane, identity in enumerate(token_order):
        x = LANE_X[lane]
        draw.ellipse(
            (sx(x - radius), sx(TOP_Y - radius), sx(x + radius), sx(TOP_Y + radius)),
            fill=TOKEN_COLORS[identity],
            outline=INK,
            width=sx(2),
        )
        box = draw.textbbox((0, 0), identity, font=font)
        tw, th = box[2] - box[0], box[3] - box[1]
        draw.text(
            (sx(x) - tw / 2, sx(TOP_Y) - th / 2 - sx(2)),
            identity,
            fill="white",
            font=font,
        )

    # Every lane ends in a port. Only the queried port is a thick red ring.
    for lane, x in enumerate(LANE_X):
        if lane == target_lane:
            r = 14
            draw.ellipse(
                (sx(x - r), sx(TARGET_Y - r), sx(x + r), sx(TARGET_Y + r)),
                fill="white",
                outline=RED,
                width=sx(5),
            )
        else:
            r = 7
            draw.ellipse(
                (sx(x - r), sx(TARGET_Y - r), sx(x + r), sx(TARGET_Y + r)),
                fill="white",
                outline=GRAY,
                width=sx(2),
            )

    return canvas.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
