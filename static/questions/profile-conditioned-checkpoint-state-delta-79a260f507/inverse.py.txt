"""Independent pixel-only inverse for the checkpoint-state-delta world.

This module intentionally duplicates only public raster coordinates and palette
sampling.  It does not import renderer, prompts, generator, verifier, or gold
code, and it receives only a PIL image.
"""

from __future__ import annotations

from PIL import Image


LABELS = ("A", "B", "C", "D")
STATE_COUNT = 3
IDENTITY_PALETTE = (
    (207, 73, 88),
    (38, 126, 180),
    (111, 82, 169),
    (43, 148, 105),
)
STATE_PALETTE = (
    (226, 124, 54),
    (40, 145, 185),
    (157, 89, 163),
)
CANVAS = (1400, 820)
PANEL_LEFTS = (55, 745)
PANEL_TOP = 260
TILE_H = 165
SLOT_OFFSETS = ((35, 80), (315, 80), (35, 290), (315, 290))
IDENTITY_STRIP_W = 52
STATE_MARKER_OFFSET_X = 150
STATE_MARKER_OFFSET_Y = TILE_H // 2
IDENTITY_KEY_X = 55
IDENTITY_KEY_Y = 115
IDENTITY_KEY_STEP = 125
STATE_KEY_X = 590
STATE_KEY_Y = 115
STATE_KEY_STEP = 175


def _key_color(frame: Image.Image, xy: tuple[int, int]) -> tuple[int, int, int]:
    color = frame.getpixel(xy)
    if not isinstance(color, tuple) or len(color) != 3:
        raise ValueError("key sample is not RGB")
    return tuple(int(value) for value in color)


def _read_palettes(frame: Image.Image) -> tuple[dict[tuple[int, int, int], int], dict[tuple[int, int, int], int]]:
    identities: dict[tuple[int, int, int], int] = {}
    for label in range(len(LABELS)):
        left = IDENTITY_KEY_X + label * IDENTITY_KEY_STEP
        color = _key_color(frame, (left + 18, IDENTITY_KEY_Y + 16))
        if color in identities:
            raise ValueError("identity key reuses a color")
        if color not in IDENTITY_PALETTE:
            raise ValueError("identity key color is not in the public palette")
        identities[color] = label

    states: dict[tuple[int, int, int], int] = {}
    for state in range(STATE_COUNT):
        cx = STATE_KEY_X + state * STATE_KEY_STEP + 18
        color = _key_color(frame, (cx, STATE_KEY_Y + 16))
        if color in states:
            raise ValueError("state key reuses a color")
        if color not in STATE_PALETTE:
            raise ValueError("state key color is not in the public palette")
        states[color] = state
    return identities, states


def _snapshot(frame: Image.Image, identities: dict, states: dict, panel: int) -> list[int]:
    values: list[int | None] = [None] * len(LABELS)
    seen: set[int] = set()
    panel_left = PANEL_LEFTS[panel]
    for dx, dy in SLOT_OFFSETS:
        identity_color = _key_color(frame, (panel_left + dx + 14, PANEL_TOP + dy + TILE_H // 2))
        if identity_color not in identities:
            raise ValueError("panel tile has no readable identity stripe")
        label = identities[identity_color]
        if label in seen:
            raise ValueError("panel repeats an identity")
        seen.add(label)
        state_color = _key_color(
            frame,
            (panel_left + dx + STATE_MARKER_OFFSET_X, PANEL_TOP + dy + STATE_MARKER_OFFSET_Y),
        )
        if state_color not in states:
            raise ValueError("panel tile has no readable state marker")
        values[label] = states[state_color]
    if seen != set(range(len(LABELS))) or any(value is None for value in values):
        raise ValueError("panel does not expose exactly four identities")
    return [int(value) for value in values]


def decision_from_image(image: Image.Image) -> str:
    """Decode both visible snapshots and return the unique changed identity."""

    frame = image.convert("RGB")
    if frame.size != CANVAS:
        raise ValueError(f"unexpected raster size {frame.size}")
    identities, states = _read_palettes(frame)
    first = _snapshot(frame, identities, states, 0)
    second = _snapshot(frame, identities, states, 1)
    changed = [index for index, (left, right) in enumerate(zip(first, second)) if left != right]
    if len(changed) != 1:
        raise ValueError("pixel evidence does not contain exactly one state change")
    return LABELS[changed[0]]
