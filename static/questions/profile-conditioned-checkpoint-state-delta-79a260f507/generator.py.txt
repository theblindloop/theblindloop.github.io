"""Renderer and analytic side for the checkpoint-state-delta world.

The image contains two ordered snapshots of four identity-coded objects.  Each
object has one of three keyed categorical states, and exactly one identity has
a different state in the second snapshot.
"""

from __future__ import annotations

import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont


LABELS = ("A", "B", "C", "D")
STATE_NAMES = ("CALM", "ACTIVE", "PAUSED")
CANVAS = (1400, 820)

BACKGROUND = (247, 249, 252)
INK = (35, 45, 58)
MUTED = (91, 104, 119)
GRID = (185, 195, 207)
CARD = (255, 255, 255)
IDENTITY_PALETTE = (
    (207, 73, 88),
    (38, 126, 180),
    (111, 82, 169),
    (43, 148, 105),
)
STATE_PALETTE = (
    (226, 124, 54),
    (40, 145, 185),
    (157, 89, 163),
)
PANEL_TINTS = (
    (255, 255, 255),
    (247, 251, 254),
    (253, 249, 244),
)

PANEL_LEFTS = (55, 745)
PANEL_TOP = 260
PANEL_W = 600
PANEL_H = 500
TILE_W = 250
TILE_H = 165
SLOT_OFFSETS = ((35, 80), (315, 80), (35, 290), (315, 290))
IDENTITY_STRIP_W = 52
STATE_MARKER_OFFSET_X = 150
STATE_MARKER_OFFSET_Y = TILE_H // 2

IDENTITY_KEY_X = 55
IDENTITY_KEY_Y = 115
IDENTITY_KEY_STEP = 125
STATE_KEY_X = 590
STATE_KEY_Y = 115
STATE_KEY_STEP = 175
KEY_RADIUS = 15

_FONT_PATHS = (
    "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
    "/usr/share/fonts/dejavu/DejaVuSans.ttf",
)


def _font(size: int):
    for candidate in _FONT_PATHS:
        if Path(candidate).is_file():
            return ImageFont.truetype(candidate, size=size)
    return ImageFont.load_default()


def _centered_text(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font, fill):
    box = draw.textbbox((0, 0), text, font=font)
    width = box[2] - box[0]
    height = box[3] - box[1]
    draw.text((xy[0] - width / 2, xy[1] - height / 2 - box[1]), text, font=font, fill=fill)


def _identity_color(scene: dict, label: int) -> tuple[int, int, int]:
    return IDENTITY_PALETTE[int(scene["label_palette_order"][int(label)])]


def _state_color(scene: dict, state: int) -> tuple[int, int, int]:
    return STATE_PALETTE[int(scene["state_palette_order"][int(state)])]


def _draw_state_marker(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    state: int,
    color: tuple[int, int, int],
    radius: int,
) -> None:
    cx, cy = center
    radius = max(2, int(radius))
    if int(state) == 0:
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=color, outline=INK, width=2)
    elif int(state) == 1:
        draw.rounded_rectangle(
            (cx - radius, cy - radius, cx + radius, cy + radius),
            radius=max(2, radius // 4),
            fill=color,
            outline=INK,
            width=2,
        )
    else:
        draw.polygon(
            ((cx, cy - radius), (cx + radius, cy), (cx, cy + radius), (cx - radius, cy)),
            fill=color,
            outline=INK,
        )


def _draw_identity_key(draw: ImageDraw.ImageDraw, scene: dict, label_font, small_font) -> None:
    draw.text((IDENTITY_KEY_X, IDENTITY_KEY_Y - 34), "IDENTITY KEY", font=small_font, fill=MUTED)
    for label, name in enumerate(LABELS):
        left = IDENTITY_KEY_X + label * IDENTITY_KEY_STEP
        draw.rectangle((left, IDENTITY_KEY_Y, left + 36, IDENTITY_KEY_Y + 32), fill=_identity_color(scene, label), outline=INK, width=1)
        _centered_text(draw, (left + 78, IDENTITY_KEY_Y + 16), name, label_font, INK)


def _draw_state_key(draw: ImageDraw.ImageDraw, scene: dict, label_font, small_font) -> None:
    draw.text((STATE_KEY_X, STATE_KEY_Y - 34), "STATE KEY", font=small_font, fill=MUTED)
    for state, name in enumerate(STATE_NAMES):
        cx = STATE_KEY_X + state * STATE_KEY_STEP + 18
        cy = STATE_KEY_Y + 16
        _draw_state_marker(draw, (cx, cy), state, _state_color(scene, state), KEY_RADIUS)
        _centered_text(draw, (cx + 78, cy), name, label_font, INK)


def _draw_tile(
    draw: ImageDraw.ImageDraw,
    scene: dict,
    left: int,
    top: int,
    label: int,
    state: int,
    label_font,
    state_font,
) -> None:
    draw.rounded_rectangle(
        (left, top, left + TILE_W, top + TILE_H),
        radius=12,
        fill=CARD,
        outline=GRID,
        width=2,
    )
    identity_color = _identity_color(scene, label)
    draw.rectangle((left, top, left + IDENTITY_STRIP_W, top + TILE_H), fill=identity_color)
    _centered_text(draw, (left + IDENTITY_STRIP_W / 2, top + TILE_H / 2), LABELS[label], label_font, (255, 255, 255))
    _draw_state_marker(
        draw,
        (left + STATE_MARKER_OFFSET_X, top + STATE_MARKER_OFFSET_Y),
        state,
        _state_color(scene, state),
        int(scene["state_marker_radius"]),
    )
    _centered_text(draw, (left + 205, top + TILE_H / 2), STATE_NAMES[state], state_font, MUTED)


def _draw_arrow(draw: ImageDraw.ImageDraw) -> None:
    y = PANEL_TOP + PANEL_H // 2
    start = (PANEL_LEFTS[0] + PANEL_W + 12, y)
    end = (PANEL_LEFTS[1] - 12, y)
    draw.line((*start, *end), fill=MUTED, width=4)
    x, yy = end
    draw.polygon(((x, yy), (x - 15, yy - 9), (x - 15, yy + 9)), fill=MUTED)


def sample_scene(seed: int) -> dict:
    """Sample balanced two-panel trajectories and independent visual nuisances."""

    logic_rng = random.Random(int(seed) * 1000003 + 17)
    style_rng = random.Random(int(seed) * 1000033 + 31)
    changed_label = int(seed) % len(LABELS)
    first = [logic_rng.randrange(len(STATE_NAMES)) for _ in LABELS]
    second = list(first)
    second[changed_label] = (first[changed_label] + 1 + logic_rng.randrange(2)) % len(STATE_NAMES)

    label_palette_order = list(range(len(LABELS)))
    state_palette_order = list(range(len(STATE_NAMES)))
    style_rng.shuffle(label_palette_order)
    style_rng.shuffle(state_palette_order)
    slot_orders = []
    for _ in range(2):
        order = list(range(len(LABELS)))
        style_rng.shuffle(order)
        slot_orders.append(order)
    panel_tints = [style_rng.randrange(len(PANEL_TINTS)) for _ in range(2)]

    return {
        "canvas": list(CANVAS),
        "panel_states": [first, second],
        "label_palette_order": label_palette_order,
        "state_palette_order": state_palette_order,
        "slot_orders": slot_orders,
        "panel_tints": panel_tints,
        "identity_marker_radius": 7 if int(seed) % 13 == 0 else 20,
        "state_marker_radius": 7 if int(seed) % 11 == 0 else 20,
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", tuple(scene["canvas"]), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(27)
    section_font = _font(20)
    label_font = _font(20)
    state_font = _font(14)
    small_font = _font(15)

    draw.text((55, 26), "CHECKPOINT STATE DELTA", font=title_font, fill=INK)
    draw.text(
        (55, 70),
        "Bind each identity to its keyed state, then compare the ordered snapshots.",
        font=small_font,
        fill=MUTED,
    )
    _draw_identity_key(draw, scene, label_font, small_font)
    _draw_state_key(draw, scene, label_font, small_font)
    draw.text((1115, 108), "PANEL ORDER", font=small_font, fill=MUTED)
    draw.text((1115, 136), "P1  →  P2", font=section_font, fill=INK)
    draw.text((1115, 177), "Exactly one keyed state changes.", font=small_font, fill=MUTED)

    for panel in range(2):
        left = PANEL_LEFTS[panel]
        draw.rounded_rectangle(
            (left, PANEL_TOP, left + PANEL_W, PANEL_TOP + PANEL_H),
            radius=14,
            fill=PANEL_TINTS[int(scene["panel_tints"][panel])],
            outline=INK,
            width=2,
        )
        _centered_text(draw, (left + PANEL_W / 2, PANEL_TOP + 29), f"P{panel + 1}", section_font, INK)
        draw.line((left + 22, PANEL_TOP + 55, left + PANEL_W - 22, PANEL_TOP + 55), fill=GRID, width=1)
        for slot, (dx, dy) in enumerate(SLOT_OFFSETS):
            label = int(scene["slot_orders"][panel][slot])
            state = int(scene["panel_states"][panel][label])
            _draw_tile(
                draw,
                scene,
                left + dx,
                PANEL_TOP + dy,
                label,
                state,
                label_font,
                state_font,
            )
    _draw_arrow(draw)
    return image


def analytic_gold(scene: dict) -> str:
    """Return the sole identity whose visible categorical state differs."""

    snapshots = scene["panel_states"]
    if len(snapshots) != 2 or any(len(row) != len(LABELS) for row in snapshots):
        raise ValueError("expected two four-identity snapshots")
    changed = [index for index in range(len(LABELS)) if snapshots[0][index] != snapshots[1][index]]
    if len(changed) != 1:
        raise ValueError("checkpoint does not contain exactly one changed identity")
    return LABELS[changed[0]]


def margin(scene: dict) -> float:
    """Visible decoding clearance, with negative values reserved for quarantine."""

    return float(min(int(scene["identity_marker_radius"]), int(scene["state_marker_radius"])) - 12)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent alias is declared: every label-bearing field is rendered."""

    return []
