"""Independent pixel-only inverse arm for the nesting-depth-match world.

Receives ONLY the final PNG. It never imports or reads the renderer, the analytic
scene, prompts, or any renderer state. It recovers the decision by:

1. marking the three dot colors and taking each dot's centroid;
2. taking the centroid of all black (ring) ink as the shared ring center;
3. labeling each black connected component (one per ring loop) and estimating
   that ring's radius as the mean distance of its ink from the shared center;
4. computing each dot's nesting depth as the number of rings whose radius
   exceeds the dot's distance from the shared center;
5. returning which of red/blue shares green's nesting depth (blue on a
   blue-only exact match, else red).
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "nesting_depth_match-oracle-0.4.0"

_DOT_RGB = {
    "green": (0, 128, 0),
    "red": (255, 0, 0),
    "blue": (0, 0, 255),
}
_RING_MAX = 90


def _centroid(mask: np.ndarray) -> tuple[float, float] | None:
    ys, xs = np.nonzero(mask)
    if len(xs) == 0:
        return None
    return (float(xs.mean()), float(ys.mean()))


def _components(black: np.ndarray) -> list[np.ndarray]:
    """4-connected components of a boolean mask, returned as a list of masks."""
    h, w = black.shape
    labels = np.zeros_like(black, dtype=np.int32)
    current = 0
    for y in range(h):
        for x in range(w):
            if not black[y, x] or labels[y, x]:
                continue
            current += 1
            labels[y, x] = current
            stack = [(x, y)]
            while stack:
                cx, cy = stack.pop()
                for nx, ny in ((cx + 1, cy), (cx - 1, cy), (cx, cy + 1), (cx, cy - 1)):
                    if (
                        0 <= nx < w
                        and 0 <= ny < h
                        and black[ny, nx]
                        and labels[ny, nx] == 0
                    ):
                        labels[ny, nx] = current
                        stack.append((nx, ny))
    comps: list[np.ndarray] = []
    for label in range(1, current + 1):
        comps.append(labels == label)
    return comps


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.int16)

    dots: dict[str, tuple[float, float]] = {}
    for name, (r, g, b) in _DOT_RGB.items():
        mask = (arr[..., 0] == r) & (arr[..., 1] == g) & (arr[..., 2] == b)
        c = _centroid(mask)
        if c is not None:
            dots[name] = c

    black = (arr[..., 0] <= _RING_MAX) & (arr[..., 1] <= _RING_MAX) & (arr[..., 2] <= _RING_MAX)
    center = _centroid(black)
    if center is None:
        return "blue"

    radii: list[float] = []
    for comp in _components(black):
        ys, xs = np.nonzero(comp)
        dists = np.hypot(xs.astype(float) - center[0], ys.astype(float) - center[1])
        if len(dists):
            radii.append(float(dists.mean()))

    missing = any(color not in dots for color in ("green", "red", "blue"))

    def depth(px: float, py: float) -> int:
        d = math.hypot(px - center[0], py - center[1])
        return sum(1 for r in radii if d < r)

    if missing or not radii:
        return "blue"

    target = depth(dots["green"][0], dots["green"][1])
    red_match = depth(dots["red"][0], dots["red"][1]) == target
    blue_match = depth(dots["blue"][0], dots["blue"][1]) == target
    if blue_match and not red_match:
        return "blue"
    return "red"
