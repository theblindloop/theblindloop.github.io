"""Independent PNG-only inverse arm for the threaded remainder register."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

BACKGROUND = (248, 246, 240)
ROUTE_COLOR = (38, 103, 178)
START_COLOR = (31, 151, 91)
END_COLOR = (205, 62, 66)
BIT_COLOR = (38, 41, 48)


def _mask(array: np.ndarray, color: tuple[int, int, int]) -> np.ndarray:
    return np.all(array == np.asarray(color, dtype=np.uint8), axis=2)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        pixels = []
        while stack:
            cy, cx = stack.pop()
            pixels.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dy or dx):
                        continue
                    ny, nx = cy + dy, cx + dx
                    if (
                        0 <= ny < height
                        and 0 <= nx < width
                        and mask[ny, nx]
                        and not seen[ny, nx]
                    ):
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        output.append(np.asarray(pixels, dtype=np.int32))
    return output


def _centroid(component: np.ndarray) -> tuple[float, float]:
    return float(component[:, 1].mean()), float(component[:, 0].mean())


def _marker(mask: np.ndarray, label: str) -> np.ndarray:
    pixels = np.argwhere(mask).astype(np.int32)
    components = _components(mask)
    if not 250 <= len(pixels) <= 520 or not 1 <= len(components) <= 2:
        raise ValueError(f"{label}: damaged endpoint ring")
    return pixels


def _nearest_route(route: np.ndarray, center: tuple[float, float]) -> tuple[int, int, float]:
    cx, cy = center
    squared = (route[:, 1] - cx) ** 2 + (route[:, 0] - cy) ** 2
    index = int(np.argmin(squared))
    return int(route[index, 0]), int(route[index, 1]), float(np.sqrt(squared[index]))


def _geodesic(route_mask: np.ndarray, start: tuple[int, int]) -> np.ndarray:
    height, width = route_mask.shape
    distances = np.full((height, width), -1, dtype=np.int32)
    queue = deque([start])
    distances[start] = 0
    while queue:
        y, x = queue.popleft()
        for dy in (-1, 0, 1):
            for dx in (-1, 0, 1):
                if not (dy or dx):
                    continue
                ny, nx = y + dy, x + dx
                if (
                    0 <= ny < height
                    and 0 <= nx < width
                    and route_mask[ny, nx]
                    and distances[ny, nx] < 0
                ):
                    distances[ny, nx] = distances[y, x] + 1
                    queue.append((ny, nx))
    return distances


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.shape != (800, 800, 3):
        raise ValueError(f"expected 800x800 RGB raster, got {array.shape}")
    if int(_mask(array, BACKGROUND).sum()) < 590000:
        raise ValueError("background coverage is inconsistent with the public canvas")

    route_mask = _mask(array, ROUTE_COLOR)
    route_components = _components(route_mask)
    if len(route_components) != 1:
        raise ValueError(f"thread must be one connected component, found {len(route_components)}")
    route = route_components[0]
    if not 11000 <= len(route) <= 19000:
        raise ValueError(f"thread ink area {len(route)} is outside the reading floor")

    start_component = _marker(_mask(array, START_COLOR), "green START")
    end_component = _marker(_mask(array, END_COLOR), "red END")
    start_pixel = _nearest_route(route, _centroid(start_component))
    end_pixel = _nearest_route(route, _centroid(end_component))
    if start_pixel[2] > 5.0 or end_pixel[2] > 5.0:
        raise ValueError("thread does not meet both endpoint rings")
    distances = _geodesic(route_mask, (start_pixel[0], start_pixel[1]))
    if distances[end_pixel[0], end_pixel[1]] < 1900:
        raise ValueError("END is not reachable far enough along the thread")

    bead_components = _components(_mask(array, BIT_COLOR))
    if len(bead_components) != 11:
        raise ValueError(f"expected eleven beads, found {len(bead_components)}")
    ordered: list[tuple[int, int]] = []
    for component in bead_components:
        ys, xs = component[:, 0], component[:, 1]
        width = int(xs.max() - xs.min() + 1)
        height = int(ys.max() - ys.min() + 1)
        area = len(component)
        if not (16 <= width <= 18 and 16 <= height <= 18 and 85 <= area <= 260):
            raise ValueError("a bead has damaged geometry")
        bit = 1 if area >= 175 else 0
        ry, rx, clearance = _nearest_route(route, _centroid(component))
        if not 7.0 <= clearance <= 16.0:
            raise ValueError("a bead is not clearly bound to the thread")
        distance = int(distances[ry, rx])
        if distance < 0:
            raise ValueError("a bead cannot be reached from START")
        ordered.append((distance, bit))

    ordered.sort()
    if any(right[0] - left[0] < 80 for left, right in zip(ordered, ordered[1:])):
        raise ValueError("bead order is ambiguous on the thread")
    bits = [bit for _, bit in ordered]
    if bits.count(1) != 5 or bits.count(0) != 6:
        raise ValueError("fixed-weight bead balance is corrupted")

    remainder = 0
    for bit in bits:
        remainder = (2 * remainder + bit) % 3
    return str(remainder)
