"""Deterministic renderer for the threaded modulo-three register world."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 800
HEIGHT = 800
BACKGROUND = (248, 246, 240)
ROUTE_COLOR = (38, 103, 178)
START_COLOR = (31, 151, 91)
END_COLOR = (205, 62, 66)
BIT_COLOR = (38, 41, 48)
BIT_COUNT = 11
ONE_COUNT = 5

_BASE_ROUTE = ((30, 30), (770, 30), (770, 350), (30, 350), (30, 650), (770, 650))


def _remainder(bits: list[int]) -> str:
    value = 0
    for bit in bits:
        if bit not in (0, 1):
            raise ValueError(f"invalid bit {bit!r}")
        value = (2 * value + bit) % 3
    return str(value)


def _sample_bits(rng: random.Random, target: str) -> list[int]:
    base = [1] * ONE_COUNT + [0] * (BIT_COUNT - ONE_COUNT)
    for _ in range(10000):
        bits = list(base)
        rng.shuffle(bits)
        if _remainder(bits) == target:
            return bits
    raise RuntimeError(f"could not sample fixed-weight stream for remainder {target}")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x39A71D)
    target = str(int(seed) % 3)
    return {
        "bits": _sample_bits(rng, target),
        "route_variant": rng.randrange(4),
        "offset_x": rng.choice((-4, 0, 4)),
        "offset_y": rng.choice((-4, 0, 4)),
    }


def _route_points(scene: dict) -> list[tuple[float, float]]:
    variant = int(scene["route_variant"])
    dx = int(scene["offset_x"])
    dy = int(scene["offset_y"])
    points = []
    for x, y in _BASE_ROUTE:
        if variant & 1:
            x = WIDTH - x
        if variant & 2:
            y = HEIGHT - y
        points.append((float(x + dx), float(y + dy)))
    return points


def _point_at_fraction(
    points: list[tuple[float, float]], fraction: float
) -> tuple[float, float, float, float]:
    lengths = [math.dist(a, b) for a, b in zip(points, points[1:])]
    remaining = fraction * sum(lengths)
    for (x0, y0), (x1, y1), length in zip(points, points[1:], lengths):
        if remaining <= length:
            t = remaining / length
            return (
                x0 + t * (x1 - x0),
                y0 + t * (y1 - y0),
                (x1 - x0) / length,
                (y1 - y0) / length,
            )
        remaining -= length
    x0, y0 = points[-2]
    x1, y1 = points[-1]
    length = math.dist(points[-2], points[-1])
    return x1, y1, (x1 - x0) / length, (y1 - y0) / length


def _segment_distance(
    point: tuple[float, float], a: tuple[float, float], b: tuple[float, float]
) -> float:
    px, py = point
    ax, ay = a
    bx, by = b
    vx, vy = bx - ax, by - ay
    t = max(0.0, min(1.0, ((px - ax) * vx + (py - ay) * vy) / (vx * vx + vy * vy)))
    return math.hypot(px - (ax + t * vx), py - (ay + t * vy))


def _station_centers(scene: dict) -> list[tuple[float, float]]:
    points = _route_points(scene)
    centers: list[tuple[float, float]] = []
    for index in range(BIT_COUNT):
        fraction = 0.055 + index * (0.89 / (BIT_COUNT - 1))
        x, y, tx, ty = _point_at_fraction(points, fraction)
        choices = ((x - ty * 12.0, y + tx * 12.0), (x + ty * 12.0, y - tx * 12.0))
        scored = []
        for choice in choices:
            border = min(choice[0], WIDTH - choice[0], choice[1], HEIGHT - choice[1])
            distances = sorted(
                _segment_distance(choice, a, b) for a, b in zip(points, points[1:])
            )
            scored.append((border + 0.15 * sum(distances[:2]), choice))
        if abs(scored[0][0] - scored[1][0]) < 1e-6:
            centers.append(choices[index % 2])
        else:
            centers.append(max(scored, key=lambda item: item[0])[1])
    return centers


def render(scene: dict) -> Image.Image:
    bits = list(scene["bits"])
    if len(bits) != BIT_COUNT:
        raise ValueError(f"expected {BIT_COUNT} bits")
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    points = _route_points(scene)
    pixel_points = [(round(x), round(y)) for x, y in points]
    sx, sy = pixel_points[0]
    ex, ey = pixel_points[-1]

    # Endpoint rings are painted first so the blue centerline remains one component.
    draw.ellipse((sx - 15, sy - 15, sx + 15, sy + 15), outline=START_COLOR, width=5)
    draw.ellipse((ex - 15, ey - 15, ex + 15, ey + 15), outline=END_COLOR, width=5)
    draw.line(pixel_points, fill=ROUTE_COLOR, width=5, joint="curve")
    for x, y in pixel_points:
        draw.ellipse((x - 2, y - 2, x + 2, y + 2), fill=ROUTE_COLOR)

    for bit, (x, y) in zip(bits, _station_centers(scene), strict=True):
        box = (round(x - 8), round(y - 8), round(x + 8), round(y + 8))
        if bit == 1:
            draw.ellipse(box, fill=BIT_COLOR)
        elif bit == 0:
            draw.ellipse(box, outline=BIT_COLOR, width=3)
        else:
            raise ValueError(f"invalid bit {bit!r}")
    return image


def analytic_gold(scene: dict) -> str:
    return _remainder(list(scene["bits"]))


def margin(scene: dict) -> float:
    centers = _station_centers(scene)
    along_gap = min(math.dist(a, b) for a, b in zip(centers, centers[1:])) - 17.0
    return round(min(9.0, along_gap), 6)


def is_quarantined(scene: dict) -> bool:
    try:
        bits = list(scene["bits"])
        return (
            len(bits) != BIT_COUNT
            or sum(bit == 1 for bit in bits) != ONE_COUNT
            or any(bit not in (0, 1) for bit in bits)
            or margin(scene) < 6.0
            or analytic_gold(scene) not in {"0", "1", "2"}
        )
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
