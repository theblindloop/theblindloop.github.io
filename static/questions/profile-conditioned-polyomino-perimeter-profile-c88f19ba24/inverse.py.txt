"""Independent image-only inverse for the perimeter-profile world."""

from __future__ import annotations

from collections import deque
import numpy as np

IDENTITY_RGB = {
    "cyan": (35, 181, 190),
    "magenta": (210, 65, 154),
    "green": (66, 170, 91),
    "orange": (235, 145, 43),
}
BORDER_RGB = {
    "red": (202, 55, 58),
    "blue": (53, 101, 204),
    "gold": (220, 169, 34),
}
REFERENCE_RGB = (92, 96, 108)


def _components(mask):
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        stack = [(int(x), int(y))]
        seen[y, x] = True
        points = []
        while stack:
            px, py = stack.pop()
            points.append((px, py))
            for nx, ny in ((px + 1, py), (px - 1, py), (px, py + 1), (px, py - 1)):
                if 0 <= nx < w and 0 <= ny < h and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((nx, ny))
        result.append(points)
    return result


def _bbox_for_rgb(array, rgb):
    mask = np.all(array == rgb, axis=2)
    ys, xs = np.nonzero(mask)
    if len(xs) < 50:
        raise ValueError("missing card border")
    return int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())


def _inside(box, x, y):
    return box[0] < x < box[2] and box[1] < y < box[3]


def _card_profile(array, box):
    profile = []
    for rgb in IDENTITY_RGB.values():
        mask = np.all(array == rgb, axis=2)
        cells = []
        for component in _components(mask):
            if len(component) < 80:
                continue
            cx = sum(x for x, _ in component) / len(component)
            cy = sum(y for _, y in component) / len(component)
            if _inside(box, cx, cy):
                xs = [p[0] for p in component]
                ys = [p[1] for p in component]
                cells.append((cx, cy, max(xs) - min(xs) + 1, max(ys) - min(ys) + 1))
        if len(cells) != 6:
            raise ValueError(f"expected six visible cells, found {len(cells)}")
        side = float(np.median([min(c[2], c[3]) for c in cells]))
        adjacency = 0
        for i in range(6):
            for j in range(i + 1, 6):
                dx = abs(cells[i][0] - cells[j][0])
                dy = abs(cells[i][1] - cells[j][1])
                if (dx <= 1.5 and side + 0.5 <= dy <= side + 4.5) or (
                    dy <= 1.5 and side + 0.5 <= dx <= side + 4.5
                ):
                    adjacency += 1
        profile.append(24 - 2 * adjacency)
    return tuple(profile)


def decision_from_image(image) -> str:
    array = np.asarray(image.convert("RGB"))
    reference_box = _bbox_for_rgb(array, REFERENCE_RGB)
    option_boxes = {name: _bbox_for_rgb(array, rgb) for name, rgb in BORDER_RGB.items()}
    target = _card_profile(array, reference_box)
    matches = [name for name, box in option_boxes.items() if _card_profile(array, box) == target]
    if len(matches) != 1:
        raise ValueError(f"image has {len(matches)} matching options")
    return matches[0]
