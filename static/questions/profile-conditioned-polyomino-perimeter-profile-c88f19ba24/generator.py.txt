"""Renderer for color-bound polyomino perimeter correspondence."""

from __future__ import annotations

import random
from collections import defaultdict
from PIL import Image, ImageDraw, ImageFont

W, H = 1024, 700
BACKGROUND = (247, 245, 239)
CARD = (255, 255, 252)
INK = (35, 38, 45)
REFERENCE_BORDER = (92, 96, 108)
COLORS = {
    "cyan": (35, 181, 190),
    "magenta": (210, 65, 154),
    "green": (66, 170, 91),
    "orange": (235, 145, 43),
}
BORDERS = {
    "red": (202, 55, 58),
    "blue": (53, 101, 204),
    "gold": (220, 169, 34),
}
IDENTITIES = tuple(COLORS)


def _norm(cells):
    min_x = min(x for x, _ in cells)
    min_y = min(y for _, y in cells)
    return sorted([[x - min_x, y - min_y] for x, y in cells])


def _perimeter(cells) -> int:
    occupied = {tuple(p) for p in cells}
    return sum(
        (x + dx, y + dy) not in occupied
        for x, y in occupied
        for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1))
    )


def _shape_pool():
    """Deterministically enumerate many fixed hexominoes by bounded growth."""
    shapes = {((0, 0),)}
    for _ in range(5):
        grown = set()
        for shape in shapes:
            cells = set(shape)
            for x, y in tuple(cells):
                for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    q = (x + dx, y + dy)
                    if q not in cells:
                        grown.add(tuple(map(tuple, _norm(cells | {q}))))
        shapes = grown
    pools = defaultdict(list)
    for shape in sorted(shapes):
        p = _perimeter(shape)
        if p in (10, 12, 14):
            pools[p].append([list(q) for q in shape])
    return dict(pools)


SHAPES = _shape_pool()


def _pick_shapes(rng: random.Random, profile: list[int]) -> dict[str, list[list[int]]]:
    result = {}
    for color, perimeter in zip(IDENTITIES, profile):
        result[color] = rng.choice(SHAPES[perimeter])
    return result


def _transform(cells, turns: int):
    pts = [tuple(p) for p in cells]
    for _ in range(turns % 4):
        pts = [(-y, x) for x, y in pts]
    return _norm(pts)


def _profile(shapes: dict[str, list[list[int]]]) -> list[int]:
    return [_perimeter(shapes[color]) for color in IDENTITIES]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x50E17E2)
    base = [10, 12, 14, 14] if seed % 2 == 0 else [10, 10, 12, 14]
    rng.shuffle(base)
    reference_shapes = _pick_shapes(rng, base)

    permutations = sorted(set(__import__("itertools").permutations(base)))
    distractors = [list(p) for p in permutations if list(p) != base]
    rng.shuffle(distractors)
    option_profiles = [base, distractors[0], distractors[1]]
    correct_position = (seed // 3) % 3
    option_profiles[0], option_profiles[correct_position] = (
        option_profiles[correct_position], option_profiles[0]
    )
    option_shapes = [_pick_shapes(rng, p) for p in option_profiles]

    correct_color = ("red", "blue", "gold")[seed % 3]
    remaining_colors = [c for c in BORDERS if c != correct_color]
    if seed % 2:
        remaining_colors.reverse()
    border_colors = [None, None, None]
    border_colors[correct_position] = correct_color
    for i in range(3):
        if border_colors[i] is None:
            border_colors[i] = remaining_colors.pop(0)

    def placement():
        slots = list(range(4))
        rng.shuffle(slots)
        return {
            color: {"slot": slots[j], "turns": rng.randrange(4)}
            for j, color in enumerate(IDENTITIES)
        }

    return {
        "reference_shapes": reference_shapes,
        "option_shapes": option_shapes,
        "placements": {"reference": placement(), "options": [placement() for _ in range(3)]},
        "border_colors": border_colors,
    }


def _font(size: int):
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _draw_card(draw, box, shapes, placement, border_rgb, label):
    x0, y0, x1, y1 = box
    draw.rounded_rectangle(box, radius=15, fill=CARD, outline=border_rgb, width=7)
    draw.text((x0 + 14, y0 + 10), label, fill=INK, font=_font(17))
    top = y0 + 38
    slot_w = (x1 - x0 - 24) / 2
    slot_h = (y1 - top - 14) / 2
    step, cell = 16, 14
    for color in IDENTITIES:
        item = placement[color]
        slot = item["slot"]
        col, row = slot % 2, slot // 2
        sx = x0 + 12 + col * slot_w
        sy = top + row * slot_h
        pts = _transform(shapes[color], item["turns"])
        pw = (max(p[0] for p in pts) + 1) * step - (step - cell)
        ph = (max(p[1] for p in pts) + 1) * step - (step - cell)
        ox = int(sx + (slot_w - pw) / 2)
        oy = int(sy + (slot_h - ph) / 2)
        for gx, gy in pts:
            xx, yy = ox + gx * step, oy + gy * step
            draw.rectangle((xx, yy, xx + cell - 1, yy + cell - 1), fill=COLORS[color])


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (W, H), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((W // 2, 13), "MATCH THE EXPOSED-EDGE PROFILE", anchor="ma", fill=INK, font=_font(23))
    _draw_card(
        draw, (302, 52, 722, 336), scene["reference_shapes"],
        scene["placements"]["reference"], REFERENCE_BORDER, "REFERENCE"
    )
    boxes = [(22, 370, 326, 680), (360, 370, 664, 680), (698, 370, 1002, 680)]
    for i, box in enumerate(boxes):
        name = scene["border_colors"][i]
        _draw_card(
            draw, box, scene["option_shapes"][i], scene["placements"]["options"][i],
            BORDERS[name], f"OPTION {i + 1}"
        )
    return image


def analytic_gold(scene: dict) -> str:
    target = _profile(scene["reference_shapes"])
    matches = [i for i, shapes in enumerate(scene["option_shapes"]) if _profile(shapes) == target]
    if len(matches) != 1:
        raise ValueError(f"expected one match, found {matches}")
    return scene["border_colors"][matches[0]]


def margin(scene: dict) -> float:
    target = _profile(scene["reference_shapes"])
    distances = sorted(
        sum(abs(a - b) for a, b in zip(target, _profile(shapes)))
        for shapes in scene["option_shapes"]
    )
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    try:
        all_shapes = [scene["reference_shapes"], *scene["option_shapes"]]
        if any(set(shapes) != set(IDENTITIES) for shapes in all_shapes):
            return True
        if any(len(cells) != 6 for shapes in all_shapes for cells in shapes.values()):
            return True
        target = _profile(scene["reference_shapes"])
        return sum(_profile(shapes) == target for shapes in scene["option_shapes"]) != 1
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
