"""Deterministic sampler, simulator, and rasterizer for a self-editing cursor walk."""

from __future__ import annotations

import copy
import random
from functools import lru_cache

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1120
HEIGHT = 820
GRID_N = 7
GRID_X = 54
GRID_Y = 116
CELL = 70
GAP = 7
PITCH = CELL + GAP
STEPS = 18

BG = (246, 245, 240)
PAPER = (255, 255, 255)
INK = (34, 42, 54)
MUTED = (105, 114, 124)
DARK = (46, 58, 76)
LIGHT = (226, 232, 228)
START = (24, 151, 96)
GOALS = ((35, 112, 180), (224, 132, 31), (155, 80, 166), (207, 63, 67))
DIRECTIONS = ((-1, 0), (0, 1), (1, 0), (0, -1))


@lru_cache(maxsize=None)
def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{name}", size)
    except OSError:
        return ImageFont.load_default()


def _walk(
    tiles: list[list[int]],
    start: list[int] | tuple[int, int],
    heading: int,
    steps: int,
    *,
    mutate: bool = True,
    reverse_rule: bool = False,
) -> tuple[tuple[int, int], list[list[int]], list[tuple[int, int]]]:
    state = [list(row) for row in tiles]
    row, col = map(int, start)
    direction = int(heading)
    if len(state) != GRID_N or any(len(line) != GRID_N for line in state):
        raise ValueError("invalid grid shape")
    if not (0 <= row < GRID_N and 0 <= col < GRID_N and 0 <= direction < 4):
        raise ValueError("invalid start state")
    trace = [(row, col)]
    for _ in range(int(steps)):
        face = int(state[row][col])
        if face not in (0, 1):
            raise ValueError("invalid cell face")
        turn = -1 if face else 1
        if reverse_rule:
            turn = -turn
        direction = (direction + turn) % 4
        if mutate:
            state[row][col] = 1 - face
        dr, dc = DIRECTIONS[direction]
        row, col = row + dr, col + dc
        if not (0 <= row < GRID_N and 0 <= col < GRID_N):
            raise ValueError("cursor left the board")
        trace.append((row, col))
    return (row, col), state, trace


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    answer_index = int(seed) % 4
    for _ in range(50000):
        tiles = [[rng.randrange(2) for _ in range(GRID_N)] for _ in range(GRID_N)]
        dark_count = sum(map(sum, tiles))
        if not 18 <= dark_count <= 31:
            continue
        start = [rng.randrange(2, 5), rng.randrange(2, 5)]
        heading = rng.randrange(4)
        try:
            final, _, trace = _walk(tiles, start, heading, STEPS)
            no_flip, _, _ = _walk(tiles, start, heading, STEPS, mutate=False)
            reversed_turn, _, _ = _walk(tiles, start, heading, STEPS, reverse_rule=True)
        except ValueError:
            continue
        one_short = trace[-2]
        counterprograms = [no_flip, reversed_turn, one_short]
        if len({final, *counterprograms}) != 4:
            continue
        revisits = len(trace) - len(set(trace))
        if revisits < 3 or len(set(trace)) < 9 or final == tuple(start):
            continue
        rng.shuffle(counterprograms)
        goals: list[list[int] | None] = [None, None, None, None]
        goals[answer_index] = list(final)
        for index, position in zip((i for i in range(4) if i != answer_index), counterprograms):
            goals[index] = list(position)
        draw_order = list(range(GRID_N * GRID_N))
        rng.shuffle(draw_order)
        return {
            "tiles": tiles,
            "start": start,
            "heading": heading,
            "steps": STEPS,
            "goal_positions": goals,
            "cell_draw_order": draw_order,
        }
    raise RuntimeError(f"could not sample a valid scene for seed {seed}")


def analytic_gold(scene: dict) -> str:
    final, _, _ = _walk(scene["tiles"], scene["start"], scene["heading"], scene["steps"])
    goals = [tuple(map(int, position)) for position in scene["goal_positions"]]
    if len(goals) != 4 or len(set(goals)) != 4:
        raise ValueError("goal positions must be unique")
    matches = [index for index, position in enumerate(goals) if position == final]
    if len(matches) != 1:
        raise ValueError("final cursor must occupy exactly one goal")
    return f"goal-{'ABCD'[matches[0]]}"


def margin(scene: dict) -> float:
    analytic_gold(scene)
    return 1.0


def is_quarantined(scene: dict) -> bool:
    analytic_gold(scene)
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["cell_draw_order"] = list(reversed(scene["cell_draw_order"]))
    return [("reverse_disjoint_cell_draw_order", twin)]


def _centered(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font, fill) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2 - box[1]), text, font=font, fill=fill)


def _cell_box(row: int, col: int) -> tuple[int, int, int, int]:
    x = GRID_X + col * PITCH
    y = GRID_Y + row * PITCH
    return (x, y, x + CELL, y + CELL)


def _turn_icon(draw: ImageDraw.ImageDraw, center: tuple[int, int], right: bool, color=INK) -> None:
    x, y = center
    if right:
        draw.arc((x - 22, y - 20, x + 22, y + 24), 205, 30, fill=color, width=6)
        draw.polygon(((x + 21, y - 2), (x + 7, y - 5), (x + 16, y + 8)), fill=color)
    else:
        draw.arc((x - 22, y - 20, x + 22, y + 24), 150, 335, fill=color, width=6)
        draw.polygon(((x - 21, y - 2), (x - 7, y - 5), (x - 16, y + 8)), fill=color)


def _start_arrow(draw: ImageDraw.ImageDraw, row: int, col: int, heading: int) -> None:
    left, top, right, bottom = _cell_box(row, col)
    cx, cy = (left + right) // 2, (top + bottom) // 2
    dr, dc = DIRECTIONS[int(heading)]
    pr, pc = dc, -dr
    draw.ellipse((cx - 27, cy - 27, cx + 27, cy + 27), outline=START, width=6)
    draw.line((cx - 16 * dc, cy - 16 * dr, cx + 4 * dc, cy + 4 * dr), fill=START, width=9)
    tip = (cx + 21 * dc, cy + 21 * dr)
    base = (cx + 1 * dc, cy + 1 * dr)
    draw.polygon((tip, (base[0] + 11 * pc, base[1] + 11 * pr), (base[0] - 11 * pc, base[1] - 11 * pr)), fill=START)


def _goal_badge(draw: ImageDraw.ImageDraw, row: int, col: int, index: int) -> None:
    left, top, right, bottom = _cell_box(row, col)
    cx, cy = right - 17, bottom - 17
    draw.ellipse((cx - 16, cy - 16, cx + 16, cy + 16), fill=GOALS[index], outline=PAPER, width=3)
    _centered(draw, (cx, cy), "ABCD"[index], _font(17, True), PAPER)


def render(scene: dict) -> Image.Image:
    analytic_gold(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    draw.text((54, 24), "FLIP-TURN WALK", font=_font(32, True), fill=INK)
    draw.text((54, 64), f"{scene['steps']} MOVES  •  revisits use the square's new shade", font=_font(20, True), fill=MUTED)

    for flat in scene["cell_draw_order"]:
        row, col = divmod(int(flat), GRID_N)
        box = _cell_box(row, col)
        fill = DARK if int(scene["tiles"][row][col]) else LIGHT
        draw.rounded_rectangle(box, radius=10, fill=fill, outline=INK, width=3)

    for index, position in enumerate(scene["goal_positions"]):
        _goal_badge(draw, int(position[0]), int(position[1]), index)
    _start_arrow(draw, int(scene["start"][0]), int(scene["start"][1]), int(scene["heading"]))

    draw.rounded_rectangle((635, 96, 1072, 700), radius=20, fill=PAPER, outline=INK, width=4)
    _centered(draw, (853, 135), "EACH MOVE", _font(27, True), INK)
    draw.line((680, 162, 1027, 162), fill=LIGHT, width=4)

    draw.rounded_rectangle((680, 190, 1027, 292), radius=14, fill=LIGHT, outline=MUTED, width=3)
    draw.rounded_rectangle((700, 208, 766, 274), radius=9, fill=LIGHT, outline=INK, width=3)
    draw.text((792, 203), "LIGHT", font=_font(24, True), fill=INK)
    draw.text((792, 239), "TURN RIGHT", font=_font(22, True), fill=INK)
    _turn_icon(draw, (983, 241), True)

    draw.rounded_rectangle((680, 314, 1027, 416), radius=14, fill=DARK, outline=MUTED, width=3)
    draw.rounded_rectangle((700, 332, 766, 398), radius=9, fill=DARK, outline=PAPER, width=3)
    draw.text((792, 327), "DARK", font=_font(24, True), fill=PAPER)
    draw.text((792, 363), "TURN LEFT", font=_font(22, True), fill=PAPER)
    _turn_icon(draw, (983, 365), False, PAPER)

    _centered(draw, (853, 458), "TURN  →  FLIP  →  MOVE", _font(23, True), INK)
    draw.line((680, 488, 1027, 488), fill=LIGHT, width=4)
    _centered(draw, (853, 527), "GOAL BADGES", _font(22, True), MUTED)
    for index, color in enumerate(GOALS):
        cx = 720 + index * 89
        cy = 582
        draw.ellipse((cx - 25, cy - 25, cx + 25, cy + 25), fill=color, outline=INK, width=3)
        _centered(draw, (cx, cy), "ABCD"[index], _font(24, True), PAPER)
        _centered(draw, (cx, 627), f"goal-{'ABCD'[index]}", _font(15, True), INK)
    _centered(draw, (853, 670), "Where is the cursor after move 18?", _font(19, True), INK)
    return image
