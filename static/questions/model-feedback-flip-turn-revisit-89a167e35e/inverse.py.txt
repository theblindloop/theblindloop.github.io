"""Independent fixed-layout PNG decoder and self-editing walk executor."""

from __future__ import annotations

from PIL import Image

WIDTH = 1120
HEIGHT = 820
GRID_N = 7
GRID_X = 54
GRID_Y = 116
CELL = 70
GAP = 7
PITCH = CELL + GAP
STEPS = 18
DARK = (46, 58, 76)
LIGHT = (226, 232, 228)
START = (24, 151, 96)
GOALS = ((35, 112, 180), (224, 132, 31), (155, 80, 166), (207, 63, 67))
DIRECTIONS = ((-1, 0), (0, 1), (1, 0), (0, -1))


class OracleAbstention(ValueError):
    pass


def _near(pixel: tuple[int, ...], color: tuple[int, int, int], tolerance: int = 8) -> bool:
    return len(pixel) >= 3 and max(abs(int(pixel[i]) - color[i]) for i in range(3)) <= tolerance


def _cell_origin(row: int, col: int) -> tuple[int, int]:
    return GRID_X + col * PITCH, GRID_Y + row * PITCH


def _decode_face(pixel: tuple[int, ...]) -> int:
    if _near(pixel, LIGHT):
        return 0
    if _near(pixel, DARK):
        return 1
    raise OracleAbstention(f"unrecognized cell face {pixel[:3]}")


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    if rgb.size != (WIDTH, HEIGHT):
        raise OracleAbstention(f"unexpected image size {rgb.size}")
    px = rgb.load()

    tiles: list[list[int]] = []
    for row in range(GRID_N):
        line = []
        for col in range(GRID_N):
            x, y = _cell_origin(row, col)
            line.append(_decode_face(px[x + 14, y + 14]))
        tiles.append(line)

    start_cells: list[tuple[int, int]] = []
    for row in range(GRID_N):
        for col in range(GRID_N):
            x, y = _cell_origin(row, col)
            found = False
            for yy in range(y + 18, y + 53, 4):
                for xx in range(x + 18, x + 53, 4):
                    if _near(px[xx, yy], START, 3):
                        found = True
                        break
                if found:
                    break
            if found:
                start_cells.append((row, col))
    if len(start_cells) != 1:
        raise OracleAbstention(f"expected one start arrow, found {len(start_cells)}")
    row, col = start_cells[0]
    x, y = _cell_origin(row, col)
    cx, cy = x + CELL // 2, y + CELL // 2
    heading_hits = []
    for heading, (dr, dc) in enumerate(DIRECTIONS):
        if _near(px[cx + 19 * dc, cy + 19 * dr], START, 3):
            heading_hits.append(heading)
    if len(heading_hits) != 1:
        raise OracleAbstention(f"ambiguous start heading {heading_hits}")
    heading = heading_hits[0]

    goal_positions: list[tuple[int, int] | None] = [None, None, None, None]
    for grid_row in range(GRID_N):
        for grid_col in range(GRID_N):
            ox, oy = _cell_origin(grid_row, grid_col)
            probe = px[ox + CELL - 17, oy + CELL - 8]
            matches = [index for index, color in enumerate(GOALS) if _near(probe, color, 3)]
            if len(matches) > 1:
                raise OracleAbstention("ambiguous goal color")
            if matches:
                index = matches[0]
                if goal_positions[index] is not None:
                    raise OracleAbstention(f"duplicate goal-{'ABCD'[index]}")
                goal_positions[index] = (grid_row, grid_col)
    if any(position is None for position in goal_positions):
        raise OracleAbstention("missing goal badge")
    if len(set(goal_positions)) != 4:
        raise OracleAbstention("goal badges must occupy distinct cells")

    state = [line[:] for line in tiles]
    for _ in range(STEPS):
        face = state[row][col]
        heading = (heading + (-1 if face else 1)) % 4
        state[row][col] = 1 - face
        dr, dc = DIRECTIONS[heading]
        row, col = row + dr, col + dc
        if not (0 <= row < GRID_N and 0 <= col < GRID_N):
            raise OracleAbstention("visible program leaves the board")
    final = (row, col)
    matches = [index for index, position in enumerate(goal_positions) if position == final]
    if len(matches) != 1:
        raise OracleAbstention("final cursor is not on one unique goal")
    return f"goal-{'ABCD'[matches[0]]}"
