"""Deterministic latent sampler and renderer for triangle-feasible bar search."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 760, 560
SUPERSAMPLE = 3
LABELS = ("A", "B", "C", "D")
PANEL_CENTERS = ((205.0, 160.0), (555.0, 160.0), (205.0, 400.0), (555.0, 400.0))
PANEL_BOXES = ((35, 35, 375, 275), (385, 35, 725, 275), (35, 285, 375, 525), (385, 285, 725, 525))
SLOT_Y = (-52.0, 0.0, 52.0)
BAR_HEIGHT = 18.0
QUARANTINE_BOUNDARY_PX = 8.0
QUARANTINE_RUNNER_UP_PX = 20.0

PALETTES = (
    ("#D62828", "#2563EB", "#16A34A"),
    ("#7B2CBF", "#E76F51", "#00798C"),
    ("#C1121F", "#0A9396", "#6A4C93"),
)

# The target maximum occupies every within-scene rank across four consecutive
# seed blocks. The ratios below also make target total-ink rank match that rank,
# preventing either maximum length or total ink from serving as a stable cue.
MAXIMA_BY_TARGET_RANK = (
    (86, 124, 174, 230),
    (78, 116, 170, 228),
    (76, 106, 156, 222),
    (72, 100, 140, 202),
)
VALID_SHORT_RATIOS = ((0.60, 0.70), (0.64, 0.66), (0.57, 0.75))
INVALID_SHORT_RATIOS = ((0.36, 0.42), (0.34, 0.44), (0.38, 0.39))


def _score_lengths(lengths: list[int]) -> int:
    ordered = sorted(int(value) for value in lengths)
    return ordered[0] + ordered[1] - ordered[2]


def _panel_lengths(panel: list[dict]) -> list[int]:
    if len(panel) != 3:
        raise ValueError("each rack must contain exactly three bars")
    slots = sorted(int(bar["slot"]) for bar in panel)
    colors = sorted(int(bar["color_class"]) for bar in panel)
    if slots != [0, 1, 2] or colors != [0, 1, 2]:
        raise ValueError("each rack must use every slot and color exactly once")
    return [int(bar["length"]) for bar in panel]


def _panel_scores(panels: list[list[dict]]) -> list[int]:
    if len(panels) != 4:
        raise ValueError("the world requires exactly four racks")
    return [_score_lengths(_panel_lengths(panel)) for panel in panels]


def sample_scene(seed: int) -> dict:
    """Sample one feasible triple and three infeasible, cue-balanced triples."""

    seed = int(seed)
    rng = random.Random(seed)
    target = seed % 4
    target_rank = (seed // 4) % 4
    maxima = [value + rng.randint(-2, 2) for value in MAXIMA_BY_TARGET_RANK[target_rank]]
    target_maximum = maxima[target_rank]
    loser_maxima = [value for index, value in enumerate(maxima) if index != target_rank]
    rng.shuffle(loser_maxima)
    loser_ratios = list(INVALID_SHORT_RATIOS)
    rng.shuffle(loser_ratios)

    triples: list[list[int]] = []
    loser_index = 0
    for panel_index in range(4):
        if panel_index == target:
            ratios = rng.choice(VALID_SHORT_RATIOS)
            maximum = target_maximum
        else:
            ratios = loser_ratios[loser_index]
            maximum = loser_maxima[loser_index]
            loser_index += 1
        lengths = [round(maximum * ratios[0]), round(maximum * ratios[1]), maximum]
        if panel_index == target and _score_lengths(lengths) <= 0:
            raise AssertionError("target triple is not feasible")
        if panel_index != target and _score_lengths(lengths) >= 0:
            raise AssertionError("distractor triple is feasible")
        rng.shuffle(lengths)
        triples.append(lengths)

    panels: list[list[dict]] = []
    for lengths in triples:
        slots = [0, 1, 2]
        colors = [0, 1, 2]
        rng.shuffle(colors)
        panel = [
            {"slot": slot, "length": int(length), "color_class": int(color)}
            for slot, length, color in zip(slots, lengths, colors, strict=True)
        ]
        rng.shuffle(panel)
        panels.append(panel)
    return {
        "panels": panels,
        "palette": list(rng.choice(PALETTES)),
        "offsets": [[rng.randint(-4, 4), rng.randint(-3, 3)] for _ in range(4)],
    }


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    path = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
    try:
        return ImageFont.truetype(path, size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    """Render explicit bar slots; record order is deliberately unobservable."""

    panels = scene["panels"]
    palette = scene["palette"]
    offsets = scene["offsets"]
    if len(panels) != 4 or len(palette) != 3 or len(offsets) != 4:
        raise ValueError("malformed scene")
    scale = SUPERSAMPLE
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), "#FFFFFF")
    draw = ImageDraw.Draw(image)
    label_font = _font(24 * scale)
    for index, (box, center) in enumerate(zip(PANEL_BOXES, PANEL_CENTERS, strict=True)):
        draw.rounded_rectangle(
            tuple(int(value * scale) for value in box),
            radius=14 * scale,
            fill="#FAFBFC",
            outline="#AEB8C2",
            width=2 * scale,
        )
        draw.text(
            ((box[0] + 16) * scale, (box[1] + 12) * scale),
            LABELS[index],
            fill="#343A40",
            font=label_font,
        )
        cx = (center[0] + float(offsets[index][0])) * scale
        cy = (center[1] + float(offsets[index][1])) * scale
        for bar in panels[index]:
            length = float(bar["length"]) * scale
            y = cy + SLOT_Y[int(bar["slot"])] * scale
            half_height = BAR_HEIGHT * scale / 2.0
            draw.rounded_rectangle(
                (cx - length / 2.0, y - half_height, cx + length / 2.0, y + half_height),
                radius=int(6 * scale),
                fill=palette[int(bar["color_class"])],
            )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    scores = _panel_scores(scene["panels"])
    winners = [index for index, score in enumerate(scores) if score > 0]
    if len(winners) != 1:
        raise ValueError("scene does not have exactly one triangle-feasible rack")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    scores = sorted(_panel_scores(scene["panels"]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    scores = _panel_scores(scene["panels"])
    return min(abs(score) for score in scores) < QUARANTINE_BOUNDARY_PX or margin(scene) < QUARANTINE_RUNNER_UP_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Lists are unordered records because every bar carries an explicit slot."""

    twin = copy.deepcopy(scene)
    twin["panels"] = [list(reversed(panel)) for panel in scene["panels"]]
    return [("reverse_unordered_bar_records", twin)]
