"""Independent final-PNG-only inverse arm for triangle-feasible bar search.

Only the public raster convention is used: four reading-order racks, each with
three saturated horizontal bars. No renderer, scene, prompt, generator, verifier,
or analytic-gold module is imported.
"""

from __future__ import annotations

from collections import Counter, deque

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")
MIN_COMPONENT_AREA = 250
MIN_BAR_WIDTH = 20
MAX_BAR_HEIGHT = 24
ABSTAIN_BOUNDARY_PX = 8
ABSTAIN_RUNNER_UP_PX = 20


def _components(mask: np.ndarray, rgb: np.ndarray) -> list[dict]:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    components: list[dict] = []
    for start_y, start_x in np.argwhere(mask):
        sy, sx = int(start_y), int(start_x)
        if visited[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        visited[sy, sx] = True
        pixels: list[tuple[int, int]] = []
        tones: Counter[tuple[int, int, int]] = Counter()
        while queue:
            y, x = queue.popleft()
            pixels.append((y, x))
            tones[tuple(int(value) for value in rgb[y, x])] += 1
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    queue.append((ny, nx))
        if len(pixels) < MIN_COMPONENT_AREA:
            continue
        ys = [value[0] for value in pixels]
        xs = [value[1] for value in pixels]
        components.append(
            {
                "area": len(pixels),
                "x0": min(xs),
                "x1": max(xs),
                "y0": min(ys),
                "y1": max(ys),
                "cx": (min(xs) + max(xs) + 1) / 2.0,
                "cy": (min(ys) + max(ys) + 1) / 2.0,
                "tone": tones.most_common(1)[0][0],
            }
        )
    return components


def _panel_index(component: dict, width: int, height: int) -> int:
    return (2 if component["cy"] >= height / 2 else 0) + (1 if component["cx"] >= width / 2 else 0)


def _score(components: list[dict]) -> int:
    if len(components) != 3:
        raise ValueError("abstain: every rack must contain exactly three bars")
    lengths: list[int] = []
    for component in components:
        bar_width = int(component["x1"] - component["x0"] + 1)
        bar_height = int(component["y1"] - component["y0"] + 1)
        if bar_width < MIN_BAR_WIDTH or bar_height > MAX_BAR_HEIGHT or bar_width <= bar_height:
            raise ValueError("abstain: recovered colored component is not a visible horizontal bar")
        lengths.append(bar_width)
    lengths.sort()
    return lengths[0] + lengths[1] - lengths[2]


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    saturation = rgb.max(axis=2) - rgb.min(axis=2)
    colored = saturation >= 55
    components = _components(colored, rgb)
    if len(components) != 12:
        raise ValueError(f"abstain: expected 12 colored bars, recovered {len(components)}")
    tones = Counter(component["tone"] for component in components)
    if len(tones) != 3 or sorted(tones.values()) != [4, 4, 4]:
        raise ValueError("abstain: the three globally balanced bar colors were not recovered")
    height, width = colored.shape
    panels: list[list[dict]] = [[] for _ in LABELS]
    for component in components:
        panels[_panel_index(component, width, height)].append(component)
    scores = [_score(panel) for panel in panels]
    winners = [index for index, score in enumerate(scores) if score > 0]
    if len(winners) != 1:
        raise ValueError("abstain: raster does not contain one triangle-feasible rack")
    if min(abs(score) for score in scores) < ABSTAIN_BOUNDARY_PX:
        raise ValueError("abstain: a rack is too close to triangle equality")
    order = sorted(range(4), key=lambda index: scores[index], reverse=True)
    if scores[order[0]] - scores[order[1]] < ABSTAIN_RUNNER_UP_PX:
        raise ValueError("abstain: winner lacks a safe runner-up surplus gap")
    return LABELS[winners[0]]
