"""Independent image-only inverse for detached odd-overlap stamping."""

from __future__ import annotations

import numpy as np
from PIL import Image

FRAME = 3
PANEL_SIZE = 56
BOARD_N = 5
STAMP_N = 2
OUTPUT_N = 6
INK = np.array((30, 35, 48), dtype=np.uint8)
PALETTES = (
    (np.array((18, 151, 160), dtype=np.uint8), np.array((225, 91, 70), dtype=np.uint8)),
    (np.array((24, 137, 154), dtype=np.uint8), np.array((213, 101, 76), dtype=np.uint8)),
)
NAMES = ("X", "plus", "T", "L")


def _template(name: str) -> np.ndarray:
    n = OUTPUT_N
    mid = n // 2
    grid = np.zeros((n, n), dtype=np.uint8)
    if name == "X":
        for r in range(n):
            grid[r, r] = grid[r, n - 1 - r] = 1
    elif name == "plus":
        grid[mid - 1:mid + 1, :] = 1
        grid[:, mid - 1:mid + 1] = 1
    elif name == "T":
        grid[:2, :] = 1
        grid[:, mid - 1:mid + 1] = 1
    elif name == "L":
        grid[:, :2] = 1
        grid[-2:, :] = 1
    else:
        raise ValueError(name)
    return grid


TEMPLATES = {name: _template(name) for name in NAMES}


def _locate_frame(arr: np.ndarray, colors: tuple[np.ndarray, ...]) -> tuple[int, int]:
    masks = [np.all(arr == color, axis=2) for color in colors]
    matches = [(index, mask) for index, mask in enumerate(masks) if int(mask.sum()) > 0]
    if len(matches) != 1:
        raise ValueError("missing or ambiguous colored frame")
    _, mask = matches[0]
    ys, xs = np.nonzero(mask)
    x0, x1, y0, y1 = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    if x1 - x0 + 1 != PANEL_SIZE or y1 - y0 + 1 != PANEL_SIZE:
        raise ValueError("damaged frame bounds")
    expected_frame_pixels = PANEL_SIZE * PANEL_SIZE - (PANEL_SIZE - 2 * FRAME) ** 2
    if int(mask.sum()) != expected_frame_pixels:
        raise ValueError("damaged colored frame")
    return x0, y0


def _read_grid(arr: np.ndarray, x0: int, y0: int, n: int) -> np.ndarray:
    inner = PANEL_SIZE - 2 * FRAME
    if inner % n:
        raise ValueError("invalid framed grid")
    cell = inner // n
    grid = np.zeros((n, n), dtype=np.uint8)
    for r in range(n):
        for c in range(n):
            cx = x0 + FRAME + c * cell + cell // 2
            cy = y0 + FRAME + r * cell + cell // 2
            radius = max(2, cell // 4)
            patch = arr[cy - radius:cy + radius + 1, cx - radius:cx + radius + 1]
            votes = int(np.all(patch == INK, axis=2).sum())
            expected = (2 * radius + 1) ** 2
            if votes == expected:
                grid[r, c] = 1
            elif votes != 0:
                raise ValueError("ambiguous or damaged grid cell")
    return grid


def _convolve(board: np.ndarray, stamp: np.ndarray) -> np.ndarray:
    output = np.zeros((OUTPUT_N, OUTPUT_N), dtype=np.uint8)
    for r, c in np.argwhere(board == 1):
        output[r:r + STAMP_N, c:c + STAMP_N] ^= stamp
    return output


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    board_colors = tuple(pair[0] for pair in PALETTES)
    stamp_colors = tuple(pair[1] for pair in PALETTES)
    bx, by = _locate_frame(arr, board_colors)
    sx, sy = _locate_frame(arr, stamp_colors)
    board = _read_grid(arr, bx, by, BOARD_N)
    stamp = _read_grid(arr, sx, sy, STAMP_N)
    output = _convolve(board, stamp)
    ranked = sorted((int(np.count_nonzero(output != grid)), name)
                    for name, grid in TEMPLATES.items())
    if ranked[1][0] - ranked[0][0] < 4:
        raise ValueError("ambiguous odd-overlap symbol")
    return ranked[0][1]
