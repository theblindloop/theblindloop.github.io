"""Render a placement board and detached stamp whose odd overlaps reveal a glyph."""

from __future__ import annotations

import random
from functools import lru_cache
from typing import Iterable

from PIL import Image, ImageDraw

CANVAS = 512
BOARD_N = 5
STAMP_N = 2
OUTPUT_N = 6
FRAME = 3
PANEL_SIZE = 56
BOARD_CELL = 10
STAMP_CELL = 25
BACKGROUND = (250, 249, 246)
PALE = (239, 241, 235)
GRID = (183, 188, 184)
INK = (30, 35, 48)
PALETTES = (
    ((18, 151, 160), (225, 91, 70)),
    ((24, 137, 154), (213, 101, 76)),
)
NAMES = ("X", "plus", "T", "L")


def _template(name: str, n: int = OUTPUT_N) -> list[list[int]]:
    mid = n // 2
    grid = [[0 for _ in range(n)] for _ in range(n)]
    if name == "X":
        for r in range(n):
            grid[r][r] = 1
            grid[r][n - 1 - r] = 1
    elif name == "plus":
        for i in range(n):
            grid[mid - 1][i] = grid[mid][i] = 1
            grid[i][mid - 1] = grid[i][mid] = 1
    elif name == "T":
        for i in range(n):
            grid[0][i] = grid[1][i] = 1
            grid[i][mid - 1] = grid[i][mid] = 1
    elif name == "L":
        for i in range(n):
            grid[i][0] = grid[i][1] = 1
            grid[n - 2][i] = grid[n - 1][i] = 1
    else:
        raise ValueError(name)
    return grid


TEMPLATES = {name: _template(name) for name in NAMES}
BOARD_TEMPLATES = {name: _template(name, BOARD_N) for name in NAMES}


def _convolve(board: list[list[int]], stamp: list[list[int]]) -> list[list[int]]:
    out = [[0 for _ in range(OUTPUT_N)] for _ in range(OUTPUT_N)]
    for br in range(BOARD_N):
        for bc in range(BOARD_N):
            if not board[br][bc]:
                continue
            for sr in range(STAMP_N):
                for sc in range(STAMP_N):
                    if stamp[sr][sc]:
                        out[br + sr][bc + sc] ^= 1
    return out


def _distances(grid: list[list[int]], templates: dict[str, list[list[int]]] = TEMPLATES) -> dict[str, int]:
    n = len(grid)
    return {name: sum(grid[r][c] != target[r][c] for r in range(n) for c in range(n))
            for name, target in templates.items()}


@lru_cache(maxsize=512)
def _balanced_pair(seed: int) -> tuple[tuple[tuple[int, ...], ...], tuple[tuple[int, ...], ...]]:
    """Search deterministically for equal-ink sources with a clear composed label.

    The 5x5 board always has ten marks and the 2x2 stamp always has two. A board
    is rejected when its own unique nearest 5x5 glyph is the requested answer,
    preventing the placement panel from being a direct symbol shortcut.
    """
    rng = random.Random(int(seed))
    target_name = NAMES[int(seed) % len(NAMES)]
    output_masks = {
        name: sum(int(TEMPLATES[name][r][c]) << (r * OUTPUT_N + c)
                  for r in range(OUTPUT_N) for c in range(OUTPUT_N))
        for name in NAMES
    }
    board_masks = {
        name: sum(int(BOARD_TEMPLATES[name][r][c]) << (r * BOARD_N + c)
                  for r in range(BOARD_N) for c in range(BOARD_N))
        for name in NAMES
    }
    for _ in range(500_000):
        board_positions = rng.sample(range(BOARD_N * BOARD_N), 10)
        stamp_positions = rng.sample(range(STAMP_N * STAMP_N), 2)
        board_mask = sum(1 << position for position in board_positions)
        board_ranked = sorted(((board_mask ^ board_masks[name]).bit_count(), name)
                              for name in NAMES)
        if board_ranked[0][1] == target_name and board_ranked[0][0] < board_ranked[1][0]:
            continue
        output_mask = 0
        for position in board_positions:
            br, bc = divmod(position, BOARD_N)
            for stamp_position in stamp_positions:
                sr, sc = divmod(stamp_position, STAMP_N)
                output_mask ^= 1 << ((br + sr) * OUTPUT_N + bc + sc)
        ranked = sorted(((output_mask ^ output_masks[name]).bit_count(), name) for name in NAMES)
        if ranked[0][1] == target_name and ranked[0][0] <= 8 and ranked[1][0] - ranked[0][0] >= 4:
            board = [[int(r * BOARD_N + c in board_positions) for c in range(BOARD_N)]
                     for r in range(BOARD_N)]
            stamp = [[int(r * STAMP_N + c in stamp_positions) for c in range(STAMP_N)]
                     for r in range(STAMP_N)]
            return tuple(map(tuple, board)), tuple(map(tuple, stamp))
    raise RuntimeError(f"could not construct balanced stamp pair for seed {seed}")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x5A17)
    board, stamp = _balanced_pair(int(seed))
    return {
        "board": [list(row) for row in board],
        "stamp": [list(row) for row in stamp],
        "origin_x": 96 + rng.randint(-4, 4),
        "origin_y": 224 + rng.randint(-4, 4),
        "palette_variant": rng.randrange(len(PALETTES)),
        "board_left": bool(rng.randrange(2)),
    }


def _validate(value: Iterable[Iterable[int]], n: int) -> list[list[int]]:
    grid = [[int(v) for v in row] for row in value]
    if len(grid) != n or any(len(row) != n for row in grid):
        raise ValueError(f"grid must be {n} by {n}")
    if any(v not in (0, 1) for row in grid for v in row):
        raise ValueError("grid cells must be binary")
    return grid


def _draw_panel(draw: ImageDraw.ImageDraw, x: int, y: int, grid: list[list[int]],
                cell: int, frame_color: tuple[int, int, int]) -> None:
    n = len(grid)
    size = 2 * FRAME + n * cell
    if size != PANEL_SIZE:
        raise ValueError("unexpected panel size")
    draw.rectangle((x, y, x + size - 1, y + size - 1), fill=frame_color)
    draw.rectangle((x + FRAME, y + FRAME, x + size - FRAME - 1,
                    y + size - FRAME - 1), fill=PALE)
    for i in range(1, n):
        xx = x + FRAME + i * cell
        yy = y + FRAME + i * cell
        draw.line((xx, y + FRAME, xx, y + size - FRAME - 1), fill=GRID, width=1)
        draw.line((x + FRAME, yy, x + size - FRAME - 1, yy), fill=GRID, width=1)
    for r in range(n):
        for c in range(n):
            if grid[r][c]:
                cx = x + FRAME + c * cell + cell // 2
                cy = y + FRAME + r * cell + cell // 2
                radius = max(2, cell // 4)
                draw.rectangle((cx - radius, cy - radius, cx + radius, cy + radius), fill=INK)


def render(scene: dict) -> Image.Image:
    board = _validate(scene["board"], BOARD_N)
    stamp = _validate(scene["stamp"], STAMP_N)
    x = int(scene["origin_x"])
    y = int(scene["origin_y"])
    palette = PALETTES[int(scene["palette_variant"])]
    board_x, stamp_x = (x, 344) if bool(scene["board_left"]) else (344, x)
    image = Image.new("RGB", (CANVAS, CANVAS), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_panel(draw, board_x, y, board, BOARD_CELL, palette[0])
    _draw_panel(draw, stamp_x, y, stamp, STAMP_CELL, palette[1])
    return image


def analytic_gold(scene: dict) -> str:
    output = _convolve(scene["board"], scene["stamp"])
    ranked = sorted((_distances(output)[name], name) for name in NAMES)
    return ranked[0][1]


def margin(scene: dict) -> float:
    output = _convolve(scene["board"], scene["stamp"])
    ranked = sorted(_distances(output).values())
    return float(ranked[1] - ranked[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
