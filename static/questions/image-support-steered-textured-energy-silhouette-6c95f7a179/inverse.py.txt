"""Independent pixel-only inverse for texture-energy silhouettes."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


GRID = 24


def _background(array: np.ndarray) -> np.ndarray:
    border = np.concatenate(
        (
            array[:3].reshape(-1, 3),
            array[-3:].reshape(-1, 3),
            array[:, :3].reshape(-1, 3),
            array[:, -3:].reshape(-1, 3),
        ),
        axis=0,
    )
    return np.median(border, axis=0).astype(np.float32)


def _plaque_box(array: np.ndarray, background: np.ndarray) -> tuple[int, int, int, int]:
    distance = np.linalg.norm(array.astype(np.float32) - background, axis=2)
    mask = distance > 7.0
    ys, xs = np.nonzero(mask)
    if len(xs) < 9000:
        raise ValueError("texture plaque is missing")
    left, right = int(xs.min()), int(xs.max()) + 1
    top, bottom = int(ys.min()), int(ys.max()) + 1
    width, height = right - left, bottom - top
    if not 125 <= width <= 150 or not 125 <= height <= 150:
        raise ValueError("plaque extent is not a stable regional field")
    return left, top, right, bottom


def _block_energy(array: np.ndarray, box: tuple[int, int, int, int], background: np.ndarray) -> np.ndarray:
    left, top, right, bottom = box
    crop = array[top + 4 : bottom - 4, left + 4 : right - 4].astype(np.float32)
    base = np.percentile(crop.reshape(-1, 3), 90.0, axis=0)
    signal = np.linalg.norm(crop - base, axis=2)
    # A small fixed blur makes the statistic texture energy, not one accidental dash.
    smooth = signal
    for axis in (0, 1):
        smooth = (np.roll(smooth, -1, axis) + 2.0 * smooth + np.roll(smooth, 1, axis)) / 4.0
    height, width = smooth.shape
    result = np.zeros((GRID, GRID), dtype=np.float32)
    for row in range(GRID):
        y0, y1 = row * height // GRID, (row + 1) * height // GRID
        for col in range(GRID):
            x0, x1 = col * width // GRID, (col + 1) * width // GRID
            result[row, col] = float(np.mean(smooth[y0:y1, x0:x1]))
    return result


def _smooth_template(metric: np.ndarray, limit: float) -> np.ndarray:
    edge = 0.12
    value = np.clip((limit + edge - metric) / (2.0 * edge), 0.0, 1.0)
    return value * value * (3.0 - 2.0 * value)


def _templates() -> tuple[np.ndarray, np.ndarray]:
    coords = (np.arange(GRID, dtype=np.float32) + 0.5) / GRID * 2.0 - 1.0
    xx, yy = np.meshgrid(coords, coords)
    circle = _smooth_template(np.sqrt(xx * xx + yy * yy), 0.70)
    diamond = _smooth_template(np.abs(xx) + np.abs(yy), 0.90)
    return circle, diamond


def _correlation(values: np.ndarray, template: np.ndarray) -> float:
    left = values.astype(np.float64).ravel()
    right = template.astype(np.float64).ravel()
    left -= left.mean()
    right -= right.mean()
    denominator = float(np.linalg.norm(left) * np.linalg.norm(right))
    if denominator <= 1e-9:
        raise ValueError("texture field has no measurable energy contrast")
    return float(np.dot(left, right) / denominator)


def decision_from_image(image: Image.Image) -> str:
    """Return only a label inferred from the final RGB raster."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[2] != 3:
        raise ValueError("expected an RGB image")
    background = _background(array)
    box = _plaque_box(array, background)
    left, top, right, bottom = box
    interior = array[top + 3 : bottom - 3, left + 3 : right - 3].astype(np.float32)
    presence = np.linalg.norm(interior - background, axis=2) > 7.0
    if float(np.mean(presence)) < 0.91:
        raise ValueError("texture plaque has a large erased region")
    energy = _block_energy(array, box, background)
    circle, diamond = _templates()
    scores = (_correlation(energy, circle), _correlation(energy, diamond))
    order = np.argsort(scores)
    if float(scores[order[-1]] - scores[order[-2]]) < 0.035:
        raise ValueError("texture silhouette is below the decision margin")
    return "circle" if scores[0] > scores[1] else "diamond"
