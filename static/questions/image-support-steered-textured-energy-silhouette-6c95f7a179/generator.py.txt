"""Deterministic textured-plaque renderer and analytic gold."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw


WIDTH = 384
HEIGHT = 384
SUPER = 4
GRID = 24
BASE = (239, 242, 244)
PALETTE: tuple[tuple[int, int, int], ...] = (
    (46, 80, 111),
    (94, 58, 123),
    (37, 108, 101),
    (145, 73, 48),
)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample label-balanced symbols and label-independent raster nuisances."""

    seed = int(seed)
    label_rng = random.Random(seed ^ 0xA5C31D)
    nuisance_rng = random.Random(seed ^ 0x51E7B9)
    return {
        "symbol": "circle" if label_rng.randrange(2) == 0 else "diamond",
        "cx": round(192.0 + nuisance_rng.uniform(-1.0, 1.0), 4),
        "cy": round(192.0 + nuisance_rng.uniform(-1.0, 1.0), 4),
        "scale": round(nuisance_rng.uniform(0.96, 0.985), 5),
        "texture_rgb": list(nuisance_rng.choice(PALETTE)),
        "phase": nuisance_rng.randrange(1_000_000),
        "stroke_budget": 3 + nuisance_rng.randrange(2),
    }


def _smooth_inside(metric: float, limit: float) -> float:
    edge = 0.095
    value = max(0.0, min(1.0, (limit + edge - metric) / (2.0 * edge)))
    return value * value * (3.0 - 2.0 * value)


def _shape_energy(scene: dict[str, Any], x: float, y: float) -> float:
    symbol = str(scene["symbol"])
    if symbol == "circle":
        metric = (x * x + y * y) ** 0.5
        inside = _smooth_inside(metric, 0.71)
    elif symbol == "diamond":
        metric = abs(x) + abs(y)
        inside = _smooth_inside(metric, 0.895)
    else:
        raise ValueError(f"unknown symbol {symbol!r}")
    return 0.12 + 0.84 * inside


def _mix(base: tuple[int, int, int], ink: tuple[int, int, int], alpha: float) -> tuple[int, int, int]:
    return tuple(round((1.0 - alpha) * base[index] + alpha * ink[index]) for index in range(3))


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (WIDTH * SUPER, HEIGHT * SUPER), (255, 255, 255))
    draw = ImageDraw.Draw(image)
    side = 140.0 * float(scene["scale"])
    cx = float(scene["cx"])
    cy = float(scene["cy"])
    left = cx - side / 2.0
    top = cy - side / 2.0
    right = cx + side / 2.0
    bottom = cy + side / 2.0
    plaque = tuple(int(value) for value in BASE)
    draw.rectangle(
        (round(left * SUPER), round(top * SUPER), round(right * SUPER), round(bottom * SUPER)),
        fill=plaque,
    )

    ink = tuple(int(value) for value in scene["texture_rgb"])
    rng = random.Random(int(scene["phase"]))
    budget = int(scene["stroke_budget"])
    cell = side / GRID
    for row in range(GRID):
        for col in range(GRID):
            nx = (col + 0.5) / GRID * 2.0 - 1.0
            ny = (row + 0.5) / GRID * 2.0 - 1.0
            energy = _shape_energy(scene, nx, ny)
            color = _mix(plaque, ink, 0.16 + 0.76 * energy)
            width = max(1, round((0.70 + 1.20 * energy) * SUPER))
            length = (2.0 + 4.0 * energy) * SUPER
            for _ in range(budget):
                offset_x = rng.uniform(-0.25, 0.25) * cell * SUPER
                offset_y = rng.uniform(-0.25, 0.25) * cell * SUPER
                angle = rng.uniform(-0.82, 0.82)
                dx = 0.5 * length * __import__("math").cos(angle)
                dy = 0.5 * length * __import__("math").sin(angle)
                px = (left + (col + 0.5) * cell) * SUPER + offset_x
                py = (top + (row + 0.5) * cell) * SUPER + offset_y
                draw.line((round(px - dx), round(py - dy), round(px + dx), round(py + dy)), fill=color, width=width)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS).convert("RGB")


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(scene["symbol"])


def margin(scene: dict[str, Any]) -> float:
    # The sampled contrast is deliberately well clear of the finite-resolution boundary.
    return 0.24 * float(scene["scale"])


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 0.14


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No distinct latent aliases are exposed by this compact parameterization."""

    return []
