"""Renderer for the comb-merge (sweep-line interleave) correspondence world.

A dark board is littered with decorative coloured squares. Somewhere on the
board sits one small cream CARD. Everything that decides the answer is drawn
inside that card:

* six small square BEADS of six different colours, arranged in two staggered
  feed rows of three (an upper row and a lower row);
* a TRACK of six equally spaced square slots below the rows, five drawn as
  empty outlines and exactly one filled solid.

The beads are combed into the track by a left-to-right sweep line: the bead
whose centre the sweep reaches first takes the leftmost slot, and so on. The
sweep order is a *merge* of the two rows, so it cannot be read off either row
alone; the decision is the colour (and source row) of the bead that lands in
the marked slot.

The decisive quantity is the horizontal centre order of the six beads across
both rows. Its boundary event is a tie: two beads in different rows sharing a
centre x, where the sweep reaches both at once and the merge is undefined. The
margin is the smallest horizontal centre gap between beads that are consecutive
in the sweep.

All decisive marks live inside one small card whose board position is resampled
per scene; every mark off the card is decoration.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "riffle_track_source_bind-0.1.0"

CANVAS_W = 640
CANVAS_H = 480

BOARD = (28, 34, 46)
CREAM = (240, 232, 208)
SLOT = (70, 76, 96)
GREY_DECOR = (74, 80, 94)

PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (206, 58, 54),
    "orange": (226, 132, 32),
    "yellow": (222, 198, 44),
    "green": (58, 150, 80),
    "blue": (52, 96, 204),
    "purple": (140, 66, 178),
}
COLOR_NAMES = ("red", "orange", "yellow", "green", "blue", "purple")

BEAD = 9          # bead square side
SLOT_SIDE = 11    # track slot square side
SLOT_PITCH = 16
PAD = 8
INNER_W = 108
INNER_H = 56
CARD_W = INNER_W + 2 * PAD   # 124
CARD_H = INNER_H + 2 * PAD   # 72

UPPER_OY = 6
LOWER_OY = 26
TRACK_OY = 50
SLOT_OX = 10
BEAD_X_MIN = 6
BEAD_X_MAX = 102
SAME_ROW_MIN = 13

QUARANTINE_MARGIN = 4.0

# margin bands, indexed by (seed // 6) % 3
MARGIN_BANDS = ((8, 16), (4, 6), (2, 3))


def _row_assignments(xs: list[int]) -> list[tuple[int, ...]]:
    """Every 3-upper / 3-lower labelling that keeps same-row beads apart."""
    out: list[tuple[int, ...]] = []
    for mask in range(64):
        bits = [(mask >> i) & 1 for i in range(6)]
        if sum(bits) != 3:
            continue
        ok = True
        for row in (0, 1):
            members = [xs[i] for i in range(6) if bits[i] == row]
            for a, b in zip(members, members[1:]):
                if b - a < SAME_ROW_MIN:
                    ok = False
                    break
            if not ok:
                break
        if ok:
            out.append(tuple(bits))
    return out


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed ^ 0x5EED)
    answer_color = COLOR_NAMES[seed % 6]
    band = (seed // 6) % 3
    lo, hi = MARGIN_BANDS[band]
    want_upper = ((seed // 2) % 2) == 0

    xs: list[int] = []
    rows: tuple[int, ...] = ()
    for _ in range(400):
        min_gap = rng.randint(lo, hi)
        cap = min(22, (BEAD_X_MAX - BEAD_X_MIN) // 4)
        cap = max(cap, min_gap)
        gaps = [min_gap] + [rng.randint(min_gap, cap) for _ in range(4)]
        rng.shuffle(gaps)
        span = sum(gaps)
        if span > BEAD_X_MAX - BEAD_X_MIN:
            continue
        start = BEAD_X_MIN + rng.randint(0, BEAD_X_MAX - BEAD_X_MIN - span)
        cand = [start]
        for gap in gaps:
            cand.append(cand[-1] + gap)
        options = _row_assignments(cand)
        if not options:
            continue
        mark = rng.randrange(6)
        preferred = [opt for opt in options if (opt[mark] == 0) == want_upper]
        chosen = rng.choice(preferred or options)
        xs, rows = cand, chosen
        break
    else:  # pragma: no cover - the search above always succeeds in practice
        raise RuntimeError("failed to sample a comb layout")

    others = [name for name in COLOR_NAMES if name != answer_color]
    rng.shuffle(others)
    colors = list(others)
    colors.insert(mark, answer_color)

    px = rng.randint(10, CANVAS_W - CARD_W - 10)
    py = rng.randint(10, CANVAS_H - CARD_H - 10)

    clutter = _sample_clutter(rng, px, py)

    return {
        "panel": [px, py],
        "xs": xs,
        "rows": list(rows),
        "colors": colors,
        "mark": mark,
        "clutter": clutter,
    }


def _sample_clutter(rng: random.Random, px: int, py: int) -> list[dict]:
    """Decorative marks off the card: three per palette colour plus grey, so
    per-colour decorative ink is equal and carries no information."""
    keep_out = (px - 10, py - 10, px + CARD_W + 10, py + CARD_H + 10)
    items: list[dict] = []
    plan = [(name, 3) for name in COLOR_NAMES] + [("grey", 6)]
    for name, count in plan:
        placed = 0
        guard = 0
        while placed < count and guard < 4000:
            guard += 1
            size = rng.choice((5, 7, 9))
            x = rng.randint(4, CANVAS_W - size - 4)
            y = rng.randint(4, CANVAS_H - size - 4)
            if x + size >= keep_out[0] and x <= keep_out[2] and y + size >= keep_out[1] and y <= keep_out[3]:
                continue
            if any(
                abs(x - it["x"]) < 12 and abs(y - it["y"]) < 12 for it in items
            ):
                continue
            items.append({"color": name, "x": x, "y": y, "size": size})
            placed += 1
    return items


def _sweep_order(scene: dict) -> list[int]:
    """Bead indices in sweep order.  `xs` is sampled ascending, so this is the
    identity; it is recomputed rather than assumed."""
    return sorted(range(6), key=lambda i: scene["xs"][i])


def analytic_gold(scene: dict) -> str:
    order = _sweep_order(scene)
    index = order[scene["mark"]]
    row = "upper" if scene["rows"][index] == 0 else "lower"
    return f"{row}_{scene['colors'][index]}"


def margin(scene: dict) -> float:
    xs = sorted(scene["xs"])
    return float(min(b - a for a, b in zip(xs, xs[1:])))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The decorative clutter list is consumed as a set: the renderer sorts it
    before drawing, so reversing it re-draws a byte-identical raster."""
    return [("clutter_order", {**scene, "clutter": list(reversed(scene["clutter"]))})]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (CANVAS_W, CANVAS_H), BOARD)
    draw = ImageDraw.Draw(image)

    for item in sorted(
        scene["clutter"], key=lambda it: (it["y"], it["x"], it["size"], it["color"])
    ):
        color = GREY_DECOR if item["color"] == "grey" else PALETTE[item["color"]]
        x, y, size = item["x"], item["y"], item["size"]
        draw.rectangle([x, y, x + size - 1, y + size - 1], fill=color)

    px, py = scene["panel"]
    draw.rectangle([px, py, px + CARD_W - 1, py + CARD_H - 1], fill=CREAM)
    ix, iy = px + PAD, py + PAD

    for i in range(6):
        cx = ix + scene["xs"][i]
        cy = iy + (UPPER_OY if scene["rows"][i] == 0 else LOWER_OY)
        half = BEAD // 2
        draw.rectangle(
            [cx - half, cy - half, cx + half, cy + half], fill=PALETTE[scene["colors"][i]]
        )

    half = SLOT_SIDE // 2
    for slot in range(6):
        cx = ix + SLOT_OX + SLOT_PITCH * slot
        cy = iy + TRACK_OY
        box = [cx - half, cy - half, cx + half, cy + half]
        if slot == scene["mark"]:
            draw.rectangle(box, fill=SLOT)
        else:
            draw.rectangle(box, outline=SLOT, width=2)

    return image
