"""Deterministic latent sampler and renderer for one-gate connectivity gain."""

from __future__ import annotations

import copy
import itertools
import random
from collections import deque

from PIL import Image, ImageDraw

CANVAS = 512
GRID_N = 4
BASE_LEFT = 56
BASE_TOP = 56
CELL = 100
MARKER_RADIUS = 14
DOOR_INSET = 24
BACKGROUND = "#ffffff"
WALL = "#17191d"
ORANGE = "#ee8422"
MARKER_COLORS = ("#2868d8", "#1c9b58", "#9148b7")


def _all_edges() -> list[tuple[int, int, int, int]]:
    edges: list[tuple[int, int, int, int]] = []
    for row in range(GRID_N):
        for col in range(GRID_N):
            if col + 1 < GRID_N:
                edges.append((row, col, row, col + 1))
            if row + 1 < GRID_N:
                edges.append((row, col, row + 1, col))
    return edges


ALL_EDGES = tuple(_all_edges())


def _canonical_edge(edge: list[int] | tuple[int, ...]) -> tuple[int, int, int, int]:
    if len(edge) != 4:
        raise ValueError("an edge needs two grid endpoints")
    left = (int(edge[0]), int(edge[1]))
    right = (int(edge[2]), int(edge[3]))
    if left > right:
        left, right = right, left
    value = (left[0], left[1], right[0], right[1])
    if value not in ALL_EDGES:
        raise ValueError(f"not an internal grid edge: {edge!r}")
    return value


def _cell_id(row: int, col: int) -> int:
    return row * GRID_N + col


class _DisjointSet:
    def __init__(self) -> None:
        self.parent = list(range(GRID_N * GRID_N))
        self.count = GRID_N * GRID_N

    def find(self, value: int) -> int:
        while self.parent[value] != value:
            self.parent[value] = self.parent[self.parent[value]]
            value = self.parent[value]
        return value

    def union(self, left: int, right: int) -> bool:
        a, b = self.find(left), self.find(right)
        if a == b:
            return False
        self.parent[b] = a
        self.count -= 1
        return True


def _components(open_edges: list[list[int]] | tuple[tuple[int, ...], ...], *,
                extra_edge: list[int] | tuple[int, ...] | None = None) -> list[list[tuple[int, int]]]:
    opened = {_canonical_edge(edge) for edge in open_edges}
    if extra_edge is not None:
        opened.add(_canonical_edge(extra_edge))
    adjacency: dict[tuple[int, int], list[tuple[int, int]]] = {
        (row, col): [] for row in range(GRID_N) for col in range(GRID_N)
    }
    for edge in opened:
        first = (edge[0], edge[1])
        second = (edge[2], edge[3])
        adjacency[first].append(second)
        adjacency[second].append(first)
    unseen = set(adjacency)
    groups: list[list[tuple[int, int]]] = []
    while unseen:
        start = min(unseen)
        unseen.remove(start)
        queue = deque([start])
        group = [start]
        while queue:
            cell = queue.popleft()
            for neighbor in adjacency[cell]:
                if neighbor in unseen:
                    unseen.remove(neighbor)
                    queue.append(neighbor)
                    group.append(neighbor)
        groups.append(sorted(group))
    return groups


def _marker_labels(open_edges: list[list[int]], markers: list[dict], *,
                   extra_edge: list[int] | None = None) -> list[int]:
    groups = _components(open_edges, extra_edge=extra_edge)
    group_of = {cell: index for index, group in enumerate(groups) for cell in group}
    return [group_of[(int(item["cell"][0]), int(item["cell"][1]))] for item in markers]


def _new_pair_count(open_edges: list[list[int]], highlight_edge: list[int],
                    markers: list[dict]) -> int:
    before = _marker_labels(open_edges, markers)
    after = _marker_labels(open_edges, markers, extra_edge=highlight_edge)
    return sum(
        before[left] != before[right] and after[left] == after[right]
        for left, right in itertools.combinations(range(len(markers)), 2)
    )


def sample_scene(seed: int) -> dict:
    """Sample a fixed-ink partition with target pair gain cycling 0, 1, 2."""

    seed = int(seed)
    rng = random.Random(seed)
    target = seed % 3
    for _ in range(8000):
        shuffled = list(ALL_EDGES)
        rng.shuffle(shuffled)
        dsu = _DisjointSet()
        opened: set[tuple[int, int, int, int]] = set()
        for edge in shuffled:
            if dsu.count <= 3:
                break
            if dsu.union(_cell_id(edge[0], edge[1]), _cell_id(edge[2], edge[3])):
                opened.add(edge)
        if dsu.count != 3 or len(opened) != 13:
            continue

        groups = _components([list(edge) for edge in opened])
        if len(groups) != 3:
            continue
        group_of = {cell: index for index, group in enumerate(groups) for cell in group}
        crossing: list[tuple[tuple[int, int, int, int], int, int]] = []
        for edge in ALL_EDGES:
            if edge in opened:
                continue
            left_group = group_of[(edge[0], edge[1])]
            right_group = group_of[(edge[2], edge[3])]
            if left_group != right_group:
                crossing.append((edge, left_group, right_group))
        rng.shuffle(crossing)
        choice: tuple[tuple[int, int, int, int], list[tuple[int, int]]] | None = None
        for edge, first_index, second_index in crossing:
            first = groups[first_index]
            second = groups[second_index]
            third_index = next(index for index in range(3) if index not in {first_index, second_index})
            third = groups[third_index]
            if target == 2:
                double, single = (first, second) if len(first) >= 2 else (second, first)
                if len(double) < 2:
                    continue
                cells = rng.sample(double, 2) + [rng.choice(single)]
            elif target == 1:
                cells = [rng.choice(first), rng.choice(second), rng.choice(third)]
            else:
                occupied_gate_side = first if len(first) >= 2 else second
                if len(occupied_gate_side) < 2:
                    continue
                cells = rng.sample(occupied_gate_side, 2) + [rng.choice(third)]
            choice = (edge, cells)
            break
        if choice is None:
            continue

        highlight, cells = choice
        rng.shuffle(cells)
        colors = list(MARKER_COLORS)
        rng.shuffle(colors)
        markers = [
            {"cell": [cell[0], cell[1]], "color": color}
            for cell, color in zip(cells, colors, strict=True)
        ]
        pair_index = (seed // 3) % 48
        offsets = ((-10, -6), (0, 0), (10, 6), (-8, 8), (8, -8), (0, 10))
        scene = {
            "open_edges": [list(edge) for edge in sorted(opened)],
            "highlight_edge": list(highlight),
            "markers": markers,
            "wall_width": (10, 12, 14)[pair_index % 3],
            "transform": pair_index % 8,
            "offset": list(offsets[pair_index % len(offsets)]),
        }
        if _new_pair_count(scene["open_edges"], scene["highlight_edge"], scene["markers"]) == target:
            return scene
    raise RuntimeError(f"could not sample a valid scene for seed {seed}")


def _wall_box(edge: list[int] | tuple[int, ...], width: int, left: int, top: int,
              *, inset: int = 0) -> tuple[int, int, int, int]:
    row, col, next_row, next_col = _canonical_edge(edge)
    half = width // 2
    if row == next_row:
        x = left + max(col, next_col) * CELL
        y0 = top + row * CELL + inset
        y1 = top + (row + 1) * CELL - inset
        return (x - half, y0, x + half - 1, y1 - 1)
    y = top + max(row, next_row) * CELL
    x0 = left + col * CELL + inset
    x1 = left + (col + 1) * CELL - inset
    return (x0, y - half, x1 - 1, y + half - 1)


def _apply_transform(image: Image.Image, code: int) -> Image.Image:
    operations = (
        None,
        Image.Transpose.ROTATE_90,
        Image.Transpose.ROTATE_180,
        Image.Transpose.ROTATE_270,
        Image.Transpose.FLIP_LEFT_RIGHT,
        Image.Transpose.FLIP_TOP_BOTTOM,
        Image.Transpose.TRANSPOSE,
        Image.Transpose.TRANSVERSE,
    )
    operation = operations[int(code) % len(operations)]
    return image if operation is None else image.transpose(operation)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (CANVAS, CANVAS), BACKGROUND)
    draw = ImageDraw.Draw(image)
    width = int(scene["wall_width"])
    dx, dy = int(scene["offset"][0]), int(scene["offset"][1])
    left, top = BASE_LEFT + dx, BASE_TOP + dy
    right, bottom = left + GRID_N * CELL, top + GRID_N * CELL
    half = width // 2
    outer = (
        (left - half, top - half, right + half - 1, top + half - 1),
        (left - half, bottom - half, right + half - 1, bottom + half - 1),
        (left - half, top - half, left + half - 1, bottom + half - 1),
        (right - half, top - half, right + half - 1, bottom + half - 1),
    )
    for box in outer:
        draw.rectangle(box, fill=WALL)

    opened = {_canonical_edge(edge) for edge in scene["open_edges"]}
    highlight = _canonical_edge(scene["highlight_edge"])
    for edge in ALL_EDGES:
        if edge not in opened:
            draw.rectangle(_wall_box(edge, width, left, top), fill=WALL)
    draw.rectangle(_wall_box(highlight, width, left, top, inset=DOOR_INSET), fill=ORANGE)

    for marker in scene["markers"]:
        row, col = int(marker["cell"][0]), int(marker["cell"][1])
        x = left + col * CELL + CELL // 2
        y = top + row * CELL + CELL // 2
        draw.ellipse(
            (x - MARKER_RADIUS, y - MARKER_RADIUS, x + MARKER_RADIUS, y + MARKER_RADIUS),
            fill=str(marker["color"]),
        )
    return _apply_transform(image, int(scene["transform"]))


def analytic_gold(scene: dict) -> str:
    open_edges = scene["open_edges"]
    highlight_edge = scene["highlight_edge"]
    markers = scene["markers"]
    return str(_new_pair_count(open_edges, highlight_edge, markers))


def margin(scene: dict) -> float:
    """Runner-up margin in answer-count units (nearest alternative is one away)."""

    value = int(analytic_gold(scene))
    return float(min(abs(value - other) for other in (0, 1, 2) if other != value))


def is_quarantined(scene: dict) -> bool:
    try:
        if int(scene["wall_width"]) < 10:
            return True
        if len(scene["markers"]) != 3 or len(scene["open_edges"]) != 13:
            return True
        if _canonical_edge(scene["highlight_edge"]) in {
            _canonical_edge(edge) for edge in scene["open_edges"]
        }:
            return True
        return analytic_gold(scene) not in {"0", "1", "2"}
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["open_edges"] = [
        [edge[2], edge[3], edge[0], edge[1]] for edge in reversed(twin["open_edges"])
    ]
    edge = twin["highlight_edge"]
    twin["highlight_edge"] = [edge[2], edge[3], edge[0], edge[1]]
    twin["markers"] = list(reversed(twin["markers"]))
    return [("serialization_order", twin)]


def raster_clearance(scene: dict) -> float:
    return float(min(int(scene["wall_width"]) / 2.0, MARKER_RADIUS, (CELL - 2 * DOOR_INSET) / 2.0))
