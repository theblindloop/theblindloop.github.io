"""Independent pixel-only inverse for highlighted-gate connectivity gain."""

from __future__ import annotations

import itertools
from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, *, minimum: int) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result: list[list[tuple[int, int]]] = []
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = deque([(y, x)])
        pixels: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            pixels.append((cy, cx))
            for ny in range(max(0, cy - 1), min(height, cy + 2)):
                for nx in range(max(0, cx - 1), min(width, cx + 2)):
                    if mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        if len(pixels) >= minimum:
            result.append(pixels)
    return result


def _centroid(pixels: list[tuple[int, int]]) -> tuple[int, int]:
    return (
        round(sum(y for y, _ in pixels) / len(pixels)),
        round(sum(x for _, x in pixels) / len(pixels)),
    )


def _labels_at(free: np.ndarray, starts: list[tuple[int, int]]) -> list[int]:
    height, width = free.shape
    labels = np.full((height, width), -1, dtype=np.int16)
    next_label = 0
    for sy, sx in starts:
        if not free[sy, sx]:
            raise ValueError("marker center is not in traversable space")
        if labels[sy, sx] >= 0:
            continue
        labels[sy, sx] = next_label
        queue = [sy * width + sx]
        while queue:
            value = queue.pop()
            y, x = divmod(value, width)
            if y and free[y - 1, x] and labels[y - 1, x] < 0:
                labels[y - 1, x] = next_label
                queue.append(value - width)
            if y + 1 < height and free[y + 1, x] and labels[y + 1, x] < 0:
                labels[y + 1, x] = next_label
                queue.append(value + width)
            if x and free[y, x - 1] and labels[y, x - 1] < 0:
                labels[y, x - 1] = next_label
                queue.append(value - 1)
            if x + 1 < width and free[y, x + 1] and labels[y, x + 1] < 0:
                labels[y, x + 1] = next_label
                queue.append(value + 1)
        next_label += 1
    return [int(labels[y, x]) for y, x in starts]


def pair_gain_from_image(image: Image.Image) -> int:
    """Recover before/after partitions solely from the raster and count gained pairs."""

    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if rgb.ndim != 3 or rgb.shape[0] < 96 or rgb.shape[1] < 96:
        raise ValueError("image is too small")
    rgb = rgb[::2, ::2]
    red = rgb[:, :, 0].astype(np.int16)
    green = rgb[:, :, 1].astype(np.int16)
    blue = rgb[:, :, 2].astype(np.int16)
    high = np.maximum(np.maximum(red, green), blue)
    low = np.minimum(np.minimum(red, green), blue)
    dark = high < 90
    orange = (red > 170) & (green > 70) & (green < 175) & (blue < 90) & ((red - blue) > 100)
    orange_parts = _components(orange, minimum=80)
    if len(orange_parts) != 1:
        raise ValueError(f"expected one orange barrier segment, found {len(orange_parts)}")
    orange_pixels = orange_parts[0]
    ys = [item[0] for item in orange_pixels]
    xs = [item[1] for item in orange_pixels]
    if max(max(xs) - min(xs), max(ys) - min(ys)) < 20:
        raise ValueError("orange evidence is not an elongated barrier segment")

    saturated = (high - low > 55) & (high < 250)
    marker_mask = saturated & ~orange & ~dark
    marker_parts = _components(marker_mask, minimum=80)
    if len(marker_parts) != 3:
        raise ValueError(f"expected three colored markers, found {len(marker_parts)}")
    starts = [_centroid(part) for part in marker_parts]

    before = _labels_at(~(dark | orange), starts)
    after = _labels_at(~dark, starts)
    gain = sum(
        before[left] != before[right] and after[left] == after[right]
        for left, right in itertools.combinations(range(3), 2)
    )
    if gain not in {0, 1, 2}:
        raise ValueError(f"unexpected connectivity gain {gain}")
    return int(gain)


def decision_from_image(image: Image.Image) -> str:
    return str(pair_gain_from_image(image))
