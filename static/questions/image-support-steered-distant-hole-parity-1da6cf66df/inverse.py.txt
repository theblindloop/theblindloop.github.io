"""Independent inverse arm: infer hole parity from RGB pixels only."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

CYAN = np.array((22, 138, 173), dtype=np.int16)
ORANGE = np.array((224, 122, 40), dtype=np.int16)


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[list[tuple[int, int]]] = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        seen[sy, sx] = True
        queue = deque([(sy, sx)])
        group: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            group.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        groups.append(group)
    return groups


def _hole_count(array: np.ndarray, target: np.ndarray) -> int:
    distance = np.max(np.abs(array.astype(np.int16) - target), axis=2)
    mask = distance <= 4
    ys, xs = np.nonzero(mask)
    if not len(xs):
        raise ValueError("colored region missing")
    y0, y1, x0, x1 = int(ys.min()), int(ys.max()), int(xs.min()), int(xs.max())
    height, width = y1 - y0 + 1, x1 - x0 + 1
    if not (238 <= width <= 242 and 246 <= height <= 250):
        raise ValueError("colored outer boundary has invalid extent")
    crop = mask[y0:y1 + 1, x0:x1 + 1]
    if not (crop[0].all() and crop[-1].all() and crop[:, 0].all() and crop[:, -1].all()):
        raise ValueError("colored outer boundary is open")
    # A complete four-sided colored boundary is itself one connected component.
    # Every permitted white component below is strictly internal and small, so
    # it cannot disconnect that boundary-bearing component. This avoids a slow
    # flood fill over roughly sixty thousand colored pixels per panel.
    holes = []
    for group in _components(~crop):
        touches = any(y in (0, height - 1) or x in (0, width - 1) for y, x in group)
        if not touches:
            holes.append(group)
    if not 1 <= len(holes) <= 4:
        raise ValueError("unsupported number of enclosed pockets")
    for group in holes:
        points = np.asarray(group)
        area = len(group)
        hh = int(points[:, 0].max() - points[:, 0].min() + 1)
        ww = int(points[:, 1].max() - points[:, 1].min() + 1)
        if not (1050 <= area <= 1250 and 32 <= hh <= 36 and 32 <= ww <= 36):
            raise ValueError("enclosed pocket is not a robust square")
    return len(holes)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    cyan_holes = _hole_count(array, CYAN)
    orange_holes = _hole_count(array, ORANGE)
    return "same" if cyan_holes % 2 == orange_holes % 2 else "different"
