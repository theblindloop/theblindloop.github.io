"""Deterministic renderer and analytic arm for two distant perforated regions."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 800, 320
BACKGROUND = (247, 245, 239)
CYAN = (22, 138, 173)
ORANGE = (224, 122, 40)
LEFT_X, RIGHT_X = 30, 530

# The ordering alternates same/different parity and balances which color has the
# larger count. Every consecutive block of eight seeds contains four of each class.
COUNT_PAIRS = (
    (1, 3), (1, 2), (2, 4), (4, 3),
    (3, 1), (2, 1), (4, 2), (3, 4),
)


def _centers(seed: int, count: int, panel_x: int, panel_y: int) -> list[list[int]]:
    anchors = [
        (58, 62), (120, 62), (182, 62),
        (58, 124), (120, 124), (182, 124),
        (58, 186), (120, 186), (182, 186),
    ]
    rng = random.Random(seed)
    chosen = rng.sample(anchors, count)
    # Preserve random draw order to create a tested latent alias.
    return [[panel_x + x, panel_y + y] for x, y in chosen]


def sample_scene(seed: int) -> dict:
    cyan_holes, orange_holes = COUNT_PAIRS[seed % len(COUNT_PAIRS)]
    cyan_side = "left" if ((seed // len(COUNT_PAIRS)) + seed) % 2 == 0 else "right"
    panel_y = 36 + random.Random(seed ^ 0xA51C).choice((-4, 0, 4))
    cyan_x = LEFT_X if cyan_side == "left" else RIGHT_X
    orange_x = RIGHT_X if cyan_side == "left" else LEFT_X
    return {
        "seed": int(seed),
        "cyan_side": cyan_side,
        "cyan_holes": cyan_holes,
        "orange_holes": orange_holes,
        "cyan_centers": _centers(seed ^ 0xC1A0, cyan_holes, cyan_x, panel_y),
        "orange_centers": _centers(seed ^ 0x0A93, orange_holes, orange_x, panel_y),
        "panel_y": panel_y,
        "panel_width": 240,
        "panel_height": 248,
        "hole_size": 34,
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    panel_y = int(scene["panel_y"])
    panel_width = int(scene["panel_width"])
    panel_height = int(scene["panel_height"])
    hole_size = int(scene["hole_size"])
    cyan_x = LEFT_X if scene["cyan_side"] == "left" else RIGHT_X
    orange_x = RIGHT_X if scene["cyan_side"] == "left" else LEFT_X

    if len(scene["cyan_centers"]) != int(scene["cyan_holes"]):
        raise ValueError("cyan count/layout mismatch")
    if len(scene["orange_centers"]) != int(scene["orange_holes"]):
        raise ValueError("orange count/layout mismatch")

    for x, color, centers in (
        (cyan_x, CYAN, scene["cyan_centers"]),
        (orange_x, ORANGE, scene["orange_centers"]),
    ):
        draw.rectangle(
            (x, panel_y, x + panel_width - 1, panel_y + panel_height - 1),
            fill=color,
        )
        radius = hole_size // 2
        for cx, cy in centers:
            draw.rectangle((cx - radius, cy - radius, cx + radius - 1, cy + radius - 1), fill=BACKGROUND)
    return image


def analytic_gold(scene: dict) -> str:
    return "same" if int(scene["cyan_holes"]) % 2 == int(scene["orange_holes"]) % 2 else "different"


def margin(scene: dict) -> float:
    # With the fixed anchor lattice, the limiting clearance is the 62px anchor
    # spacing minus a 34px pocket, or the nearest 28px border clearance.
    return float(min(28, 62 - int(scene["hole_size"])))


def is_quarantined(scene: dict) -> bool:
    if int(scene["hole_size"]) < 24 or margin(scene) < 16:
        return True
    for key in ("cyan_centers", "orange_centers"):
        centers = scene[key]
        for i, (x1, y1) in enumerate(centers):
            for x2, y2 in centers[i + 1:]:
                if max(abs(x1 - x2), abs(y1 - y2)) < int(scene["hole_size"]) + 16:
                    return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    key = "cyan_centers" if len(twin["cyan_centers"]) > 1 else "orange_centers"
    twin[key] = list(reversed(twin[key]))
    return [(f"reverse_{key}", twin)]
