"""Renderer for the paired-register difference world.

The label is carried only by the relation between two separate 8-by-8 raster
registers.  Each register receives the same randomized carrier, with cells
flipped in one register according to the selected symbol template.  The two
registers have equal ink counts for both labels, so absolute darkness and
single-panel texture do not answer the question.
"""

from __future__ import annotations

import math
import random
from typing import Any

import numpy as np
from PIL import Image, ImageDraw


WIDTH = 512
HEIGHT = 512
SUPERSAMPLE = 4
TILE_SIDE = 148
GRID_SIZE = 8
CELL_SIZE = 12
INNER_OFFSET = 26
FRAME_WIDTH = 5

FRAME_PALETTE = ((47, 103, 142), (106, 71, 139), (44, 119, 91))
NOTCH_PALETTE = ((216, 165, 60), (190, 91, 54))
TONE_PAIRS = (
    ((225, 234, 240), (44, 61, 78)),
    ((239, 226, 211), (83, 49, 37)),
    ((220, 237, 218), (35, 73, 49)),
)
BACKGROUND_PALETTE = ((248, 247, 242), (245, 248, 250), (249, 244, 247))
REMOTE_COLORS = ((170, 177, 185), (157, 166, 175), (181, 169, 177))

TEMPLATES: dict[str, tuple[str, ...]] = {
    "ring": (
        "00111100",
        "01100110",
        "11000011",
        "11000011",
        "11000011",
        "11000011",
        "01100110",
        "00111100",
    ),
    "cross": (
        "10011001",
        "00011000",
        "00011000",
        "11111111",
        "11111111",
        "00011000",
        "00011000",
        "10011001",
    ),
}
TEMPLATE_GAP = sum(
    left != right
    for left_row, right_row in zip(TEMPLATES["ring"], TEMPLATES["cross"])
    for left, right in zip(left_row, right_row)
) / float(GRID_SIZE * GRID_SIZE)


def _template_bits(symbol: str) -> list[list[int]]:
    return [[int(value) for value in row] for row in TEMPLATES[symbol]]


def _carrier_for(symbol: str, rng: random.Random) -> list[str]:
    """Make a balanced carrier with exactly sixteen overlaps with the template."""

    template = _template_bits(symbol)
    template_ones = [(row, col) for row in range(GRID_SIZE) for col in range(GRID_SIZE) if template[row][col]]
    template_zeros = [(row, col) for row in range(GRID_SIZE) for col in range(GRID_SIZE) if not template[row][col]]
    rng.shuffle(template_ones)
    rng.shuffle(template_zeros)
    chosen = set(template_ones[:16] + template_zeros[:16])
    return [
        "".join("1" if (row, col) in chosen else "0" for col in range(GRID_SIZE))
        for row in range(GRID_SIZE)
    ]


def _outside_registers(
    x: float, y: float, marks: list[dict[str, Any]], other: list[tuple[float, float]]
) -> bool:
    if x < -20 or y < -20 or x > WIDTH + 20 or y > HEIGHT + 20:
        return False
    for ox, oy in other:
        if math.dist((x, y), (ox, oy)) < TILE_SIDE * 0.72:
            return False
    return True


def sample_scene(seed: int) -> dict:
    """Sample one deterministic, nuisance-randomized pair of registers."""

    seed = int(seed)
    rng = random.Random(seed)
    symbol = "ring" if seed % 2 == 0 else "cross"
    tone_a, tone_b = TONE_PAIRS[rng.randrange(len(TONE_PAIRS))]
    background = BACKGROUND_PALETTE[rng.randrange(len(BACKGROUND_PALETTE))]
    frame_color = FRAME_PALETTE[rng.randrange(len(FRAME_PALETTE))]
    notch_color = NOTCH_PALETTE[rng.randrange(len(NOTCH_PALETTE))]
    variant = "soft" if rng.randrange(5) == 0 else "standard"
    carrier = _carrier_for(symbol, rng)

    if rng.randrange(2) == 0:
        first_anchor = (rng.randint(30, 58), rng.randint(30, 58))
        second_anchor = (rng.randint(330, 354), rng.randint(330, 354))
    else:
        first_anchor = (rng.randint(330, 354), rng.randint(30, 58))
        second_anchor = (rng.randint(30, 58), rng.randint(330, 354))

    anchors = [first_anchor, second_anchor]
    registers = [
        {
            "x": int(x),
            "y": int(y),
            "rotation": rng.randrange(4) * 90,
        }
        for x, y in anchors
    ]
    if rng.randrange(2):
        registers.reverse()

    remote_marks: list[dict[str, Any]] = []
    for index in range(8):
        for _ in range(300):
            x = rng.uniform(16.0, WIDTH - 16.0)
            y = rng.uniform(16.0, HEIGHT - 16.0)
            if _outside_registers(x, y, remote_marks, anchors):
                break
        if index % 2 == 0:
            remote_marks.append(
                {
                    "kind": "dot",
                    "x": round(x, 3),
                    "y": round(y, 3),
                    "size": round(rng.uniform(3.0, 7.0), 3),
                }
            )
        else:
            remote_marks.append(
                {
                    "kind": "line",
                    "x": round(x, 3),
                    "y": round(y, 3),
                    "dx": round(rng.uniform(-20.0, 20.0), 3),
                    "dy": round(rng.uniform(-20.0, 20.0), 3),
                }
            )

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(background),
        "registers": registers,
        "tone_pair": [list(tone_a), list(tone_b)],
        "carrier": carrier,
        "symbol": symbol,
        "remote_marks": remote_marks,
        "frame_color": list(frame_color),
        "notch_color": list(notch_color),
        "variant": variant,
    }


def _blend(color: tuple[int, int, int], background: tuple[int, int, int], amount: float) -> tuple[int, int, int]:
    return tuple(
        int(round(amount * value + (1.0 - amount) * base))
        for value, base in zip(color, background)
    )


def _draw_remote_marks(
    draw: ImageDraw.ImageDraw, marks: list[dict[str, Any]], scale: int, color: tuple[int, int, int]
) -> None:
    for mark in marks:
        kind = mark["kind"]
        x = float(mark["x"]) * scale
        y = float(mark["y"]) * scale
        if kind == "dot":
            radius = float(mark["size"]) * scale / 2.0
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)
        else:
            draw.line(
                (
                    x,
                    y,
                    x + float(mark["dx"]) * scale,
                    y + float(mark["dy"]) * scale,
                ),
                fill=color,
                width=max(2, scale),
            )


def _register_tile(scene: dict, register: dict, index: int, scale: int) -> Image.Image:
    background = tuple(int(value) for value in scene["background"])
    tile = Image.new("RGB", (TILE_SIDE * scale, TILE_SIDE * scale), background)
    draw = ImageDraw.Draw(tile)
    frame = tuple(int(value) for value in scene["frame_color"])
    notch = tuple(int(value) for value in scene["notch_color"])
    tone_a = tuple(int(value) for value in scene["tone_pair"][0])
    tone_b = tuple(int(value) for value in scene["tone_pair"][1])
    amount = 0.90 if scene["variant"] == "soft" else 1.0
    tone_a = _blend(tone_a, background, amount)
    tone_b = _blend(tone_b, background, amount)
    carrier = scene["carrier"]
    template = _template_bits(scene["symbol"])
    for row in range(GRID_SIZE):
        for col in range(GRID_SIZE):
            carrier_bit = int(carrier[row][col])
            output_bit = template[row][col]
            bit = carrier_bit if index == 0 else carrier_bit ^ output_bit
            color = tone_b if bit else tone_a
            left = (INNER_OFFSET + col * CELL_SIZE) * scale
            top = (INNER_OFFSET + row * CELL_SIZE) * scale
            right = (INNER_OFFSET + (col + 1) * CELL_SIZE) * scale - 1
            bottom = (INNER_OFFSET + (row + 1) * CELL_SIZE) * scale - 1
            draw.rectangle((left, top, right, bottom), fill=color)

    border = [4 * scale, 4 * scale, (TILE_SIDE - 5) * scale, (TILE_SIDE - 5) * scale]
    draw.rectangle(border, outline=frame, width=FRAME_WIDTH * scale)
    # A filled triangular corner notch, isolated from the square border.
    draw.polygon(
        [
            (12 * scale, 12 * scale),
            (25 * scale, 12 * scale),
            (12 * scale, 25 * scale),
        ],
        fill=notch,
    )
    if scene["variant"] == "soft":
        draw.line(
            [(29 * scale, 12 * scale), (38 * scale, 12 * scale)],
            fill=notch,
            width=max(1, scale),
        )

    rotation = int(register["rotation"]) % 360
    if rotation == 90:
        tile = tile.transpose(Image.Transpose.ROTATE_90)
    elif rotation == 180:
        tile = tile.transpose(Image.Transpose.ROTATE_180)
    elif rotation == 270:
        tile = tile.transpose(Image.Transpose.ROTATE_270)
    return tile


def render(scene: dict) -> Image.Image:
    scale = SUPERSAMPLE
    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(int(value) for value in scene["background"])
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    remote_color = REMOTE_COLORS[int(scene["variant"] == "soft") % len(REMOTE_COLORS)]
    _draw_remote_marks(draw, scene["remote_marks"], scale, remote_color)
    for index, register in enumerate(scene["registers"]):
        tile = _register_tile(scene, register, index, scale)
        image.paste(tile, (int(register["x"]) * scale, int(register["y"]) * scale))
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """Closed-form label: the symbol visibly encoded by the pair relation."""

    return str(scene["symbol"])


def margin(scene: dict) -> float:
    """Normalized runner-up gap of the displayed two-template classifier."""

    return float(TEMPLATE_GAP)


def is_quarantined(scene: dict) -> bool:
    """All sampled scenes use stable cell contrast and are outside the boundary band."""

    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases are declared for this fully rendered parametrization."""

    return []
