"""Latent sampler, raster renderer, and analytic arm."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS = 512
BG = (250, 249, 246)
FRAME = (222, 226, 226)
INK = (17, 83, 133)


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    wanted = 1 if int(seed) % 2 == 0 else -1  # +1 means rising wins.
    total_width = 88
    desired = wanted * rng.choice((28, 32, 36, 40))
    rising_total = (total_width + desired) // 2
    falling_total = total_width - rising_total

    def partition(total: int) -> list[int]:
        for _ in range(1000):
            first = rng.randint(6, 26)
            second = rng.randint(6, 26)
            third = total - first - second
            if 6 <= third <= 26:
                values = [first, second, third]
                rng.shuffle(values)
                return values
        raise RuntimeError("unable to partition directional run")

    rising_widths = partition(rising_total)
    falling_widths = partition(falling_total)
    signs = [1, 1, 1, -1, -1, -1]
    rng.shuffle(signs)
    widths = [rising_widths.pop() if sign == 1 else falling_widths.pop() for sign in signs]
    heights = [rng.randint(6, 8) for _ in range(6)]
    local = [(0, 0)]
    for sign, width, height in zip(signs, widths, heights):
        x, y = local[-1]
        local.append((x + width, y - sign * height))
    min_y = min(y for _, y in local)
    max_y = max(y for _, y in local)
    span_x = local[-1][0]
    span_y = max_y - min_y
    left = rng.randint(70, CANVAS - span_x - 70)
    # Randomize among grid bands while keeping the thin trace within one band;
    # this makes local ablation support consistently pathlike rather than squat.
    band = rng.randint(1, 6)
    top = band * 64 + rng.randint(18, 64 - span_y - 18)
    vertices = [[left + x, top + (y - min_y)] for x, y in local]
    return {"vertices": vertices, "trace_width": 6}


def render(scene: dict[str, Any]) -> Image.Image:
    vertices = [(int(x), int(y)) for x, y in scene["vertices"]]
    trace_width = int(scene["trace_width"])
    scale = 3
    image = Image.new("RGB", (CANVAS * scale, CANVAS * scale), BG)
    draw = ImageDraw.Draw(image)
    xs = [p[0] for p in vertices]
    ys = [p[1] for p in vertices]
    pad_x, pad_y = 13, 16
    box = (
        (min(xs) - pad_x) * scale,
        (min(ys) - pad_y) * scale,
        (max(xs) + pad_x) * scale,
        (max(ys) + pad_y) * scale,
    )
    draw.rounded_rectangle(box, radius=8 * scale, outline=FRAME, width=1 * scale)
    points = [(x * scale, y * scale) for x, y in vertices]
    draw.line(points, fill=INK, width=trace_width * scale, joint="curve")
    radius = trace_width * scale / 2
    for x, y in points:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=INK)
    return image.resize((CANVAS, CANVAS), Image.Resampling.LANCZOS)


def _signed_run(scene: dict[str, Any]) -> int:
    vertices = scene["vertices"]
    total = 0
    for (x1, y1), (x2, y2) in zip(vertices, vertices[1:]):
        width = int(x2) - int(x1)
        total += width if int(y2) < int(y1) else -width
    return total


def analytic_gold(scene: dict[str, Any]) -> str:
    return "rising" if _signed_run(scene) > 0 else "falling"


def margin(scene: dict[str, Any]) -> float:
    return float(abs(_signed_run(scene)))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 16.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
