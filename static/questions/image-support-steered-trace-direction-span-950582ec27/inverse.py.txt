"""Independent inverse arm using only pixels from the final raster."""

from __future__ import annotations

from PIL import Image
import numpy as np


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = pixels[:, :, 0], pixels[:, :, 1], pixels[:, :, 2]
    mask = (red < 75) & (green < 130) & (blue > 105) & ((blue - red) > 60)
    yy, xx = np.nonzero(mask)
    if len(xx) < 250:
        return "abstain"
    x0, x1 = int(xx.min()), int(xx.max())
    if not 70 <= x1 - x0 <= 120:
        return "abstain"
    columns = []
    for x in range(x0, x1 + 1):
        values = np.nonzero(mask[:, x])[0]
        if len(values) < 2:
            return "abstain"
        columns.append(float(np.median(values)))
    center = np.asarray(columns, dtype=float)
    # A short symmetric baseline suppresses raster stair-steps but preserves turns.
    center = np.convolve(np.pad(center, (2, 2), mode="edge"), np.ones(5) / 5, mode="valid")
    lag = 4
    delta = center[2 * lag :] - center[: -2 * lag]
    if len(delta) < 40:
        return "abstain"
    # Positive image-y motion is falling; negative image-y motion is rising.
    rising = int(np.count_nonzero(delta < -0.45))
    falling = int(np.count_nonzero(delta > 0.45))
    if rising == falling or abs(rising - falling) < 8:
        return "abstain"
    return "rising" if rising > falling else "falling"
