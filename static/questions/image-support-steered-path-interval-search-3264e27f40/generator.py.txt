"""Deterministic latent world for a route-ordered visual search task."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 768
HEIGHT = 512
COLORS = ("red", "blue", "green", "purple")
PALETTE = {
    "red": (207, 66, 62),
    "blue": (45, 111, 181),
    "green": (49, 143, 91),
    "purple": (132, 76, 173),
}
DARK = (35, 40, 47)
GOLD = (221, 169, 43)
BACKGROUNDS = ((248, 247, 242), (244, 248, 247), (248, 245, 249))
ROUTE_WIDTH = 7
RING_OUTER = 17
RING_INNER = 10

_BASE_ROUTE = (
    (50, 112),
    (232, 78),
    (154, 232),
    (370, 156),
    (286, 370),
    (530, 268),
    (448, 446),
    (666, 368),
    (724, 132),
)


def _all_pairings(values: tuple[int, ...]) -> Iterable[tuple[tuple[int, int], ...]]:
    if not values:
        yield ()
        return
    first = values[0]
    for index in range(1, len(values)):
        second = values[index]
        rest = values[1:index] + values[index + 1 :]
        for tail in _all_pairings(rest):
            yield ((first, second),) + tail


_ELIGIBLE_PAIRINGS = tuple(
    pairing
    for pairing in _all_pairings(tuple(range(8)))
    if sum(abs(left - right) == 3 for left, right in pairing) == 1
)


def _make_sequence(seed: int, rng: random.Random) -> list[str]:
    """Pair all eight slots, with exactly one pair three positions apart."""

    target = COLORS[seed % len(COLORS)]
    pairing = list(rng.choice(_ELIGIBLE_PAIRINGS))
    target_pair = next(pair for pair in pairing if abs(pair[0] - pair[1]) == 3)
    other_pairs = [pair for pair in pairing if pair != target_pair]
    other_colors = [color for color in COLORS if color != target]
    rng.shuffle(other_pairs)
    rng.shuffle(other_colors)
    assignment = [(target_pair, target), *zip(other_pairs, other_colors)]
    sequence = [""] * 8
    for pair, color in assignment:
        for position in pair:
            sequence[position] = color
    return sequence


def _transform_point(point: tuple[int, int], transform: int) -> tuple[int, int]:
    x, y = point
    if transform & 1:
        x = WIDTH - x
    if transform & 2:
        y = HEIGHT - y
    return x, y


def sample_scene(seed: int) -> dict:
    """Sample balanced answer colors and nuisance geometry from an integer seed."""

    rng = random.Random(int(seed))
    sequence = _make_sequence(int(seed), rng)
    jittered = []
    for index, (x, y) in enumerate(_BASE_ROUTE):
        # Keep endpoints and bends visibly separated while varying all coordinates.
        scale = 4 if index in (0, len(_BASE_ROUTE) - 1) else 7
        jittered.append((x + rng.randint(-scale, scale), y + rng.randint(-scale, scale)))
    transform = (int(seed) // 4) % 4
    route = [_transform_point(point, transform) for point in jittered]
    return {
        "sequence": sequence,
        "route_points": [list(point) for point in route],
        "background": list(BACKGROUNDS[(int(seed) // 16) % len(BACKGROUNDS)]),
    }


def _ring_center(left: tuple[int, int], right: tuple[int, int]) -> tuple[int, int]:
    return ((left[0] + right[0]) // 2, (left[1] + right[1]) // 2)


def _disk(draw: ImageDraw.ImageDraw, center: tuple[int, int], radius: int, fill: tuple[int, int, int]) -> None:
    x, y = center
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=fill)


def render(scene: dict) -> Image.Image:
    """Render colored checkpoint rings around one continuous dark polyline."""

    points = [tuple(map(int, point)) for point in scene["route_points"]]
    sequence = list(scene["sequence"])
    background = tuple(map(int, scene["background"]))
    if len(points) != 9 or len(sequence) != 8:
        raise ValueError("expected nine route vertices and eight checkpoint colors")
    image = Image.new("RGB", (WIDTH, HEIGHT), background)
    draw = ImageDraw.Draw(image)

    # Halos are drawn first; the route is redrawn last so its support stays unbroken.
    for endpoint in (points[0], points[-1]):
        _disk(draw, endpoint, 20, GOLD)
        _disk(draw, endpoint, 12, background)
    for index, color in enumerate(sequence):
        center = _ring_center(points[index], points[index + 1])
        _disk(draw, center, RING_OUTER, PALETTE[color])
        _disk(draw, center, RING_INNER, background)

    for left, right in zip(points, points[1:]):
        draw.line((left, right), fill=DARK, width=ROUTE_WIDTH)
    joint_radius = ROUTE_WIDTH // 2
    for point in points:
        _disk(draw, point, joint_radius, DARK)
    return image


def _scores(sequence: list[str]) -> dict[str, float]:
    scores: dict[str, float] = {}
    for color in COLORS:
        positions = [index for index, value in enumerate(sequence) if value == color]
        if len(positions) != 2:
            scores[color] = -99.0
            continue
        intervening = positions[1] - positions[0] - 1
        scores[color] = -float(abs(intervening - 2))
    return scores


def analytic_gold(scene: dict) -> str:
    sequence = list(scene["sequence"])
    scores = _scores(sequence)
    winners = [color for color, score in scores.items() if score == 0.0]
    if len(winners) != 1:
        raise ValueError(f"expected one exact interval winner, found {winners}")
    return winners[0]


def margin(scene: dict) -> float:
    values = sorted(_scores(list(scene["sequence"])).values(), reverse=True)
    return float(values[0] - values[1])


def is_quarantined(scene: dict) -> bool:
    sequence = list(scene["sequence"])
    if len(sequence) != 8 or any(sequence.count(color) != 2 for color in COLORS):
        return True
    scores = _scores(sequence)
    return sum(value == 0.0 for value in scores.values()) != 1 or margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse an undirected route's latent parameterization without changing pixels."""

    twin = dict(scene)
    twin["route_points"] = list(reversed(scene["route_points"]))
    twin["sequence"] = list(reversed(scene["sequence"]))
    return [("reverse_route_parameterization", twin)]
