"""Deterministic renderer for a fiducial-defined polar band reveal."""

from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image, ImageDraw


RENDERER_VERSION = "polar-band-unwrap-renderer-0.1.0"
SYMBOLS = ("wave", "ladder", "bowtie", "key")
WIDTH, HEIGHT = 640, 512
CARRIER_PERIOD = 8
BACKGROUND_RGB = (246, 246, 246)
PALETTES = (
    ((38, 112, 230), (224, 66, 70)),
    ((35, 170, 106), (194, 72, 202)),
    ((225, 132, 38), (61, 101, 214)),
    ((166, 71, 194), (45, 161, 91)),
)


def _segment_band(x, y, x1, y1, x2, y2, width):
    dx, dy = x2 - x1, y2 - y1
    t = ((x - x1) * dx + (y - y1) * dy) / (dx * dx + dy * dy)
    t = np.clip(t, 0.0, 1.0)
    px, py = x1 + t * dx, y1 + t * dy
    return (x - px) ** 2 + (y - py) ** 2 <= width * width


def _disk(x, y, cx, cy, radius):
    return (x - cx) ** 2 + (y - cy) ** 2 <= radius * radius


def _symbol_mask(u, v, symbol, dx, dy, scale):
    """Return a broad normalized symbol mask; this is the renderer's only label use."""

    x = (u - 0.5 - dx) / scale + 0.5
    y = (v - 0.5 - dy) / scale + 0.5
    if symbol == "wave":
        curve = 0.5 + 0.21 * np.sin(2.0 * math.pi * 1.35 * (x - 0.12))
        return (x >= 0.12) & (x <= 0.88) & (np.abs(y - curve) <= 0.055)
    if symbol == "ladder":
        parts = [
            _segment_band(x, y, 0.27, 0.19, 0.27, 0.81, 0.045),
            _segment_band(x, y, 0.73, 0.19, 0.73, 0.81, 0.045),
            _segment_band(x, y, 0.27, 0.28, 0.73, 0.28, 0.045),
            _segment_band(x, y, 0.27, 0.50, 0.73, 0.50, 0.045),
            _segment_band(x, y, 0.27, 0.72, 0.73, 0.72, 0.045),
        ]
        return np.logical_or.reduce(parts)
    if symbol == "bowtie":
        parts = [
            _segment_band(x, y, 0.20, 0.22, 0.80, 0.50, 0.050),
            _segment_band(x, y, 0.80, 0.50, 0.20, 0.78, 0.050),
            _segment_band(x, y, 0.20, 0.78, 0.80, 0.22, 0.050),
            _segment_band(x, y, 0.80, 0.22, 0.20, 0.50, 0.050),
        ]
        return np.logical_or.reduce(parts)
    if symbol == "key":
        parts = [
            _disk(x, y, 0.50, 0.27, 0.135),
            _segment_band(x, y, 0.50, 0.34, 0.50, 0.78, 0.055),
            _segment_band(x, y, 0.50, 0.68, 0.68, 0.68, 0.055),
            _segment_band(x, y, 0.50, 0.78, 0.65, 0.78, 0.055),
        ]
        return np.logical_or.reduce(parts)
    raise ValueError(f"unknown symbol {symbol!r}")


def _polar_point(cx, cy, radius, angle_deg):
    angle = math.radians(angle_deg)
    return (cx + radius * math.cos(angle), cy + radius * math.sin(angle))


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; symbol identity is independent of nuisance RNG draws."""

    seed = int(seed)
    symbol = SYMBOLS[seed % len(SYMBOLS)]
    rng = random.Random((seed * 1000003 + 0x5EEDBEEF) & 0xFFFFFFFF)
    inner = rng.uniform(92.0, 116.0)
    outer = inner + rng.uniform(112.0, 132.0)
    scene = {
        "symbol": symbol,
        "cx": rng.uniform(304.0, 336.0),
        "cy": rng.uniform(216.0, 252.0),
        "inner_radius": inner,
        "outer_radius": outer,
        "start_deg": rng.uniform(26.0, 48.0),
        "span_deg": rng.uniform(86.0, 124.0),
        "carrier_phase": rng.randrange(CARRIER_PERIOD),
        "carrier_amplitude": 0.0,
        "signal_strength": 0.0,
        "base_level": rng.uniform(201.0, 218.0),
        "symbol_dx": rng.uniform(-0.032, 0.032),
        "symbol_dy": rng.uniform(-0.032, 0.032),
        "symbol_scale": rng.uniform(0.91, 1.04),
        "palette_id": rng.randrange(len(PALETTES)),
        "marker_swap": bool(rng.randrange(2)),
        "marker_radius": rng.uniform(9.0, 12.0),
        "width": WIDTH,
        "height": HEIGHT,
    }
    if seed % 5 == 0:
        scene["carrier_amplitude"] = rng.uniform(24.0, 30.0)
        scene["signal_strength"] = rng.uniform(51.0, 60.0)
    else:
        scene["carrier_amplitude"] = rng.uniform(15.0, 22.0)
        scene["signal_strength"] = rng.uniform(72.0, 88.0)
    return scene


def render(scene: dict) -> Image.Image:
    """Render the curved carrier and four visible polar-frame fiducials."""

    width, height = int(scene["width"]), int(scene["height"])
    yy, xx = np.mgrid[0:height, 0:width]
    cx, cy = float(scene["cx"]), float(scene["cy"])
    radius = np.hypot(xx - cx, yy - cy)
    angle = np.degrees(np.arctan2(yy - cy, xx - cx)) % 360.0
    start = float(scene["start_deg"])
    span = float(scene["span_deg"])
    inner, outer = float(scene["inner_radius"]), float(scene["outer_radius"])
    sector = (radius >= inner) & (radius <= outer) & (angle >= start) & (angle <= start + span)
    u = (angle - start) / span
    v = (radius - inner) / (outer - inner)
    phase = int(scene["carrier_phase"]) % CARRIER_PERIOD
    phase_unit = phase / float(CARRIER_PERIOD)
    carrier = float(scene["carrier_amplitude"]) * (
        0.58 * np.sin(2.0 * math.pi * (34.0 * u + phase_unit))
        + 0.42 * np.cos(2.0 * math.pi * (22.0 * v + 1.7 * phase_unit))
    )
    signal = _symbol_mask(
        u,
        v,
        str(scene["symbol"]),
        float(scene["symbol_dx"]),
        float(scene["symbol_dy"]),
        float(scene["symbol_scale"]),
    )
    gray = float(scene["base_level"]) + carrier - float(scene["signal_strength"]) * signal
    pixels = np.empty((height, width, 3), dtype=np.uint8)
    pixels[:, :, :] = np.array(BACKGROUND_RGB, dtype=np.uint8)
    clipped = np.clip(gray, 0.0, 255.0).astype(np.uint8)
    for channel in range(3):
        pixels[:, :, channel][sector] = clipped[sector]
    image = Image.fromarray(pixels, mode="RGB")
    draw = ImageDraw.Draw(image)
    palette = PALETTES[int(scene["palette_id"]) % len(PALETTES)]
    first, second = (palette[1], palette[0]) if scene["marker_swap"] else palette
    marker_radius = float(scene["marker_radius"])
    for color, radial in ((first, start), (second, start + span)):
        for ring_radius in (inner, outer):
            px, py = _polar_point(cx, cy, ring_radius, radial)
            box = (
                int(round(px - marker_radius)),
                int(round(py - marker_radius)),
                int(round(px + marker_radius)),
                int(round(py + marker_radius)),
            )
            draw.ellipse(box, fill=color)
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def margin(scene: dict) -> float:
    """Visible transform margin: low-frequency signal above carrier amplitude."""

    return float(scene["signal_strength"]) - float(scene["carrier_amplitude"])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 34.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["carrier_phase"] = int(scene["carrier_phase"]) + CARRIER_PERIOD
    return [("carrier_phase_period", twin)]
