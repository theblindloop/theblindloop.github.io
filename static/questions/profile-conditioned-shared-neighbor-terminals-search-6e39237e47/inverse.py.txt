"""Independent image-only inverse arm for shared-neighbor search."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

DARK = np.array((38, 50, 56), dtype=np.int16)
BORDER = np.array((165, 181, 194), dtype=np.int16)
GRAY = np.array((190, 200, 210), dtype=np.int16)
MAGENTA = np.array((205, 52, 128), dtype=np.int16)
AMBER = np.array((240, 164, 45), dtype=np.int16)


def _near(arr: np.ndarray, color: np.ndarray, tolerance: int = 8) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - color), axis=2) <= tolerance


def _components(mask: np.ndarray, minimum: int = 8) -> list[tuple[float, float, int, tuple[int, int, int, int]]]:
    h, w = mask.shape
    seen = np.zeros((h, w), dtype=bool)
    found = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        pts = []
        while queue:
            cy, cx = queue.popleft()
            pts.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(pts) >= minimum:
            ys = np.fromiter((p[0] for p in pts), dtype=float)
            xs = np.fromiter((p[1] for p in pts), dtype=float)
            found.append((float(xs.mean()), float(ys.mean()), len(pts),
                          (int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max()))))
    return found


def _one_center(arr: np.ndarray, color: np.ndarray, bbox: tuple[int, int, int, int]) -> tuple[float, float]:
    x0, y0, x1, y1 = bbox
    crop = _near(arr[y0:y1 + 1, x0:x1 + 1], color)
    comps = _components(crop, minimum=40)
    if len(comps) != 1:
        raise ValueError(f"expected one colored terminal, found {len(comps)}")
    return comps[0][0] + x0, comps[0][1] + y0


def _linked(dark: np.ndarray, start: tuple[float, float], end: tuple[float, float]) -> bool:
    h, w = dark.shape
    hits = []
    for t in np.linspace(0.24, 0.76, 45):
        x = int(round(start[0] + t * (end[0] - start[0])))
        y = int(round(start[1] + t * (end[1] - start[1])))
        x0, x1 = max(0, x - 2), min(w, x + 3)
        y0, y1 = max(0, y - 2), min(h, y + 3)
        hits.append(bool(dark[y0:y1, x0:x1].any()))
    return sum(hits) / len(hits) > 0.82


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    border_components = _components(_near(arr, BORDER), minimum=500)
    panels = sorted(border_components, key=lambda item: item[0])
    if len(panels) != 4:
        raise ValueError(f"expected four visible panel borders, found {len(panels)}")
    dark = _near(arr, DARK)
    shared_counts = []
    for _, _, _, bbox in panels:
        left = _one_center(arr, MAGENTA, bbox)
        right = _one_center(arr, AMBER, bbox)
        x0, y0, x1, y1 = bbox
        gray_crop = _near(arr[y0:y1 + 1, x0:x1 + 1], GRAY)
        gray = _components(gray_crop, minimum=60)
        nodes = sorted(((g[0] + x0, g[1] + y0) for g in gray), key=lambda p: p[1])
        if len(nodes) != 4:
            raise ValueError(f"expected four gray circles, found {len(nodes)}")
        shared_counts.append(sum(_linked(dark, left, node) and _linked(dark, right, node)
                                 for node in nodes))
    winners = [i for i, count in enumerate(shared_counts) if count == 1]
    if len(winners) != 1:
        raise ValueError(f"image has no unique one-shared-neighbor panel: {shared_counts}")
    return "ABCD"[winners[0]]
