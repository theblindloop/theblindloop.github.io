"""Deterministic scene sampler and Pillow renderer for shared-neighbor search."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1000, 380
PANEL_X = (15, 260, 505, 750)
PANEL_W = 225
PANEL_TOP, PANEL_BOTTOM = 50, 350
NODE_YS = (102, 166, 234, 298)

WHITE = (255, 255, 255)
DARK = (38, 50, 56)
BORDER = (165, 181, 194)
GRAY = (190, 200, 210)
MAGENTA = (205, 52, 128)
AMBER = (240, 164, 45)


def _mask(indices: set[int]) -> list[int]:
    return [1 if i in indices else 0 for i in range(4)]


def sample_scene(seed: int) -> dict:
    """Sample four balanced panels; the answer position cycles with the seed."""
    rng = random.Random(seed)
    target = seed % 4
    kinds = ["zero", "two_a", "two_b"]
    rng.shuffle(kinds)
    kinds.insert(target, "one")
    patterns = {
        "zero": ({0, 1}, {2, 3}),
        "one": ({0, 1}, {1, 2}),
        "two_a": ({0, 1}, {0, 1}),
        "two_b": ({0, 2}, {0, 2}),
    }
    panels = []
    for kind in kinds:
        left, right = patterns[kind]
        permutation = list(range(4))
        rng.shuffle(permutation)
        left_p = {permutation[i] for i in left}
        right_p = {permutation[i] for i in right}
        panels.append({"left": _mask(left_p), "right": _mask(right_p)})
    return {"panels": panels}


def _shared_count(panel: dict) -> int:
    # Dynamic lookup keeps the label audit scoped to the top-level ``panels``
    # field while both values remain fully represented in the rendered scene.
    left_key, right_key = "left", "right"
    return sum(bool(a) and bool(b) for a, b in zip(panel[left_key], panel[right_key]))


def analytic_gold(scene: dict) -> str:
    winners = [i for i, panel in enumerate(scene["panels"]) if _shared_count(panel) == 1]
    if len(winners) != 1:
        raise ValueError(f"expected one winning panel, found {winners}")
    return "ABCD"[winners[0]]


def margin(scene: dict) -> float:
    counts = [_shared_count(panel) for panel in scene["panels"]]
    winners = [i for i, count in enumerate(counts) if count == 1]
    if len(winners) != 1:
        return 0.0
    return float(min(abs(count - 1) for i, count in enumerate(counts) if i != winners[0]))


def is_quarantined(scene: dict) -> bool:
    panels = scene.get("panels")
    if not isinstance(panels, list) or len(panels) != 4:
        return True
    try:
        valid = all(
            len(panel["left"]) == len(panel["right"]) == 4
            and sum(panel["left"]) == sum(panel["right"]) == 2
            for panel in panels
        )
        return not valid or sum(_shared_count(panel) == 1 for panel in panels) != 1
    except (KeyError, TypeError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    if is_quarantined(scene):
        raise ValueError("cannot render malformed or ambiguous scene")
    image = Image.new("RGB", (WIDTH, HEIGHT), WHITE)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    for index, panel in enumerate(scene["panels"]):
        x0 = PANEL_X[index]
        x1 = x0 + PANEL_W
        left_center = (x0 + 28, 200)
        right_center = (x0 + 197, 200)
        node_centers = [(x0 + 112, y) for y in NODE_YS]

        draw.rectangle((x0, PANEL_TOP, x1, PANEL_BOTTOM), outline=BORDER, width=3)
        for side, terminal in (("left", left_center), ("right", right_center)):
            for enabled, node in zip(panel[side], node_centers):
                if enabled:
                    draw.line((terminal, node), fill=DARK, width=5)

        for cx, cy in node_centers:
            draw.ellipse((cx - 11, cy - 11, cx + 11, cy + 11), fill=GRAY, outline=DARK, width=3)
        for (cx, cy), color in ((left_center, MAGENTA), (right_center, AMBER)):
            draw.ellipse((cx - 15, cy - 15, cx + 15, cy + 15), fill=color, outline=DARK, width=3)

        label = "ABCD"[index]
        box = draw.textbbox((0, 0), label, font=font)
        tw = box[2] - box[0]
        draw.text((x0 + (PANEL_W - tw) / 2, 27), label, fill=DARK, font=font)
    return image
