"""Independent pixel-only inverse arm for the coupled-pair world.

Recovers the decision (the LEADER colour of the pair reached after composing
the coupled pair-transition ``k`` times from the START pair, where ``k`` is the
number of STEPS dots) from the final rasterized PNG alone. It reads the nine
KEY rows by sampling the fixed source/target squares to build the ordered-pair
-> next-pair map, reads the two START discs to fix (L0, F0), counts the STEPS
dots by scanning the grey row band, composes the discovered pair transition
``k`` times, and returns the LEADER colour name. No renderer / prompts /
generate / verify / analytic-gold code is imported.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

COLORS = [
    (222, 55, 60),    # red
    (45, 95, 215),    # blue
    (50, 160, 72),    # green
]
COLOR_NAMES = ["red", "blue", "green"]
GREY = (85, 85, 85)

N = 3
SQUARE = 22
SRC_LX, SRC_FX = 96, 128
TGT_LX, TGT_FX = 292, 324
KEYY0, ROW_STEP = 80, 38
START_L_CX, START_F_CX, START_CY = 150, 220, 470
STEPS_Y = 555
DOT_MIN_GAP = 6


def _nearest_colour(rgb: tuple[int, int, int]) -> int:
    best = 0
    best_d = float("inf")
    for index, colour in enumerate(COLORS):
        d = sum((rgb[i] - colour[i]) ** 2 for i in range(3))
        if d < best_d:
            best_d = d
            best = index
    return best


def _pixel(arr: np.ndarray, x: int, y: int) -> tuple[int, int, int]:
    return (int(arr[y, x, 0]), int(arr[y, x, 1]), int(arr[y, x, 2]))


def _count_dots(arr: np.ndarray) -> int:
    mask = np.all(arr == GREY, axis=-1)
    band = mask[STEPS_Y - 1 : STEPS_Y + 2]
    cols = np.any(band, axis=0)
    runs: list[int] = []
    in_run = False
    last = None
    for x in range(cols.shape[0]):
        if cols[x]:
            if not in_run or (last is not None and x - last > DOT_MIN_GAP):
                runs.append(x)
                in_run = True
            last = x
        else:
            in_run = False
    return len(runs)


def decision_from_image(image: Image.Image) -> str:
    """Return the final LEADER colour name after ``k`` composed transitions."""
    arr = np.asarray(image.convert("RGB"))

    table: list[tuple[int, int]] = []
    for row in range(N * N):
        y = KEYY0 + row * ROW_STEP
        src_leader = _nearest_colour(_pixel(arr, SRC_LX + SQUARE // 2, y + SQUARE // 2))
        src_follower = _nearest_colour(_pixel(arr, SRC_FX + SQUARE // 2, y + SQUARE // 2))
        next_leader = _nearest_colour(_pixel(arr, TGT_LX + SQUARE // 2, y + SQUARE // 2))
        next_follower = _nearest_colour(_pixel(arr, TGT_FX + SQUARE // 2, y + SQUARE // 2))
        pair_index = src_leader * N + src_follower
        table.append((pair_index, (next_leader, next_follower)))
    pair_map = {pair_index: nxt for pair_index, nxt in table}

    leader = _nearest_colour(_pixel(arr, START_L_CX, START_CY))
    follower = _nearest_colour(_pixel(arr, START_F_CX, START_CY))
    k = _count_dots(arr)
    if k <= 0:
        k = 1

    for _ in range(k):
        next_leader, next_follower = pair_map[leader * N + follower]
        leader, follower = next_leader, next_follower
    return COLOR_NAMES[leader]
