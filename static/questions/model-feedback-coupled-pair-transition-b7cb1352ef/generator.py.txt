"""Coupled-pair state-transition world (state-transition reasoning).

A single running state is an ordered pair of two colours (LEADER, FOLLOWER)
drawn from a three-colour alphabet. A KEY table holds nine entries: the row for
each ordered source pair (L, F) shows the next pair (L', F') that the state
turns into under exactly one application of the transition. Because the new
LEADER and the new FOLLOWER both depend on the current pair jointly, the
transition is a coupled binary function over a nine-state space, not a
repeated unary (colour -> colour) map. A row of STEPS dots fixes K, the number
of times the transition must be composed in sequence. Starting from the START
pair (L0, F0), the solver evolves the ordered-pair state for exactly K discrete
transitions, each step reading the current pair's KEY row to move to the next
pair, and answers with the final LEADER colour.

The renderer owns scene sampling, rasterization and the structural analytic
gold (the LEADER colour of the pair reached after K composed transitions). The
independent pixel-only inverse arm (``oracle.py``) recovers the same decision
from the final PNG alone by reading the nine KEY rows into a pair map, reading
the START pair and counting the STEPS dots.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw, ImageFont

WIDTH = 560
HEIGHT = 640
BACKGROUND = (255, 255, 255)
GREY = (85, 85, 85)
LABEL = (40, 40, 40)

COLORS = [
    (222, 55, 60),    # red
    (45, 95, 215),    # blue
    (50, 160, 72),    # green
]
COLOR_NAMES = ["red", "blue", "green"]

N = 3
SQUARE = 22

# KEY table geometry: nine rows (one per ordered source pair), top to bottom.
SRC_LX = 96
SRC_FX = 128
TGT_LX = 292
TGT_FX = 324
KEYY0 = 80
ROW_STEP = 38

# START pair: two discs.
START_L_CX, START_F_CX = 150, 220
START_CY = 470
START_R = 22

# STEPS dots.
STEPS_Y = 555
STEPS_X0 = 132
STEPS_SPACING = 34
DOT_R = 8

K_LO, K_HI = 2, 4

_FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"


def sample_scene(seed: int) -> dict:
    """Sample one latent scene deterministically from ``seed``."""
    rng = np.random.default_rng(seed)
    # table[r] = (next leader, next follower) for source pair r = leader*N+follower.
    table = [(int(rng.integers(0, N)), int(rng.integers(0, N))) for _ in range(N * N)]
    start = (int(rng.integers(0, N)), int(rng.integers(0, N)))
    k = int(rng.integers(K_LO, K_HI + 1))
    return {
        "table": table,
        "start": start,
        "k": k,
    }


def analytic_gold(scene: dict) -> str:
    """Structural gold: the LEADER colour of the pair after ``k`` transitions."""
    table = scene["table"]
    state = scene["start"]
    k = scene["k"]
    leader, follower = state
    for _ in range(k):
        next_leader, next_follower = table[leader * N + follower]
        leader, follower = next_leader, next_follower
    return COLOR_NAMES[leader]


def margin(scene: dict) -> float:
    """The transition is fully deterministic; the decision is unambiguous."""
    return 1.0


def is_quarantined(scene: dict) -> bool:
    """No illegibility: colours are fixed, K is 2..4, geometry is fixed."""
    return False


def _fill_arrow(draw, x0, x1, mid_y):
    draw.line([x0, mid_y, x1, mid_y], fill=GREY, width=3)
    draw.polygon(
        [(x1, mid_y), (x1 - 12, mid_y - 8), (x1 - 12, mid_y + 8)],
        fill=GREY,
    )


def render(scene: dict) -> Image.Image:
    """Rasterize the latent scene (a deterministic PNG)."""
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(img)
    try:
        font_big = ImageFont.truetype(_FONT_PATH, 24)
        font_small = ImageFont.truetype(_FONT_PATH, 18)
    except OSError:
        font_big = ImageFont.load_default()
        font_small = ImageFont.load_default()

    draw.text((30, 36), "Coupled transition KEY", font=font_big, fill=LABEL)

    table = scene["table"]
    for row in range(N * N):
        y = KEYY0 + row * ROW_STEP
        src_leader, src_follower = divmod(row, N)
        next_leader, next_follower = table[row]
        draw.rectangle([SRC_LX, y, SRC_LX + SQUARE, y + SQUARE], fill=COLORS[src_leader])
        draw.rectangle([SRC_FX, y, SRC_FX + SQUARE, y + SQUARE], fill=COLORS[src_follower])
        mid_y = y + SQUARE // 2
        _fill_arrow(draw, SRC_FX + SQUARE + 6, TGT_LX - 8, mid_y)
        draw.rectangle([TGT_LX, y, TGT_LX + SQUARE, y + SQUARE], fill=COLORS[next_leader])
        draw.rectangle([TGT_FX, y, TGT_FX + SQUARE, y + SQUARE], fill=COLORS[next_follower])

    start_leader, start_follower = scene["start"]
    draw.text((60, START_CY - START_R - 34), "START pair (L F):", font=font_small, fill=LABEL)
    draw.ellipse(
        [
            START_L_CX - START_R,
            START_CY - START_R,
            START_L_CX + START_R,
            START_CY + START_R,
        ],
        fill=COLORS[start_leader],
    )
    draw.ellipse(
        [
            START_F_CX - START_R,
            START_CY - START_R,
            START_F_CX + START_R,
            START_CY + START_R,
        ],
        fill=COLORS[start_follower],
    )
    draw.text((START_L_CX, START_CY + START_R + 6), "L", font=font_small, fill=LABEL)
    draw.text((START_F_CX, START_CY + START_R + 6), "F", font=font_small, fill=LABEL)

    draw.text((30, STEPS_Y - 24), "STEPS:", font=font_small, fill=LABEL)
    for i in range(scene["k"]):
        cx = STEPS_X0 + i * STEPS_SPACING
        draw.ellipse([cx - DOT_R, STEPS_Y - DOT_R, cx + DOT_R, STEPS_Y + DOT_R], fill=GREY)

    return img


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The scene is injectively rendered; no latent aliases are declared."""
    return []
