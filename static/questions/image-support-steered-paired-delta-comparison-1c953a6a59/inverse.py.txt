"""Independent image-only inverse for the paired-change dot plot."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

# Public ink specification. No renderer or latent module is imported.
_CYAN = np.array((0, 145, 190), dtype=np.int16)
_ORANGE = np.array((230, 105, 25), dtype=np.int16)
_COLOR_TOLERANCE = 8
_MIN_AREA = 590
_MAX_AREA = 700


class OracleError(ValueError):
    """Raised when the raster does not expose four intact measurable markers."""


def _components(mask: np.ndarray) -> list[np.ndarray]:
    remaining = {tuple(point) for point in np.argwhere(mask)}
    groups: list[np.ndarray] = []
    while remaining:
        start = remaining.pop()
        queue = deque([start])
        points = [start]
        while queue:
            y, x = queue.popleft()
            for neighbor in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if neighbor in remaining:
                    remaining.remove(neighbor)
                    queue.append(neighbor)
                    points.append(neighbor)
        groups.append(np.asarray(points, dtype=np.float64))
    return groups


def _marker_centers(array: np.ndarray, color: np.ndarray, name: str) -> list[tuple[float, float]]:
    delta = np.abs(array.astype(np.int16) - color)
    mask = np.max(delta, axis=2) <= _COLOR_TOLERANCE
    groups = _components(mask)
    if len(groups) != 2:
        raise OracleError(f"{name} must have exactly two disconnected markers")

    centers: list[tuple[float, float]] = []
    for points in groups:
        area = len(points)
        if not _MIN_AREA <= area <= _MAX_AREA:
            raise OracleError(f"{name} marker area is inconsistent with an intact disk")
        ys = points[:, 0]
        xs = points[:, 1]
        width = int(xs.max() - xs.min() + 1)
        height = int(ys.max() - ys.min() + 1)
        fill = area / float(width * height)
        if not (27 <= width <= 30 and 27 <= height <= 30 and 0.72 <= fill <= 0.84):
            raise OracleError(f"{name} marker is clipped or not disk-like")
        centers.append((float(xs.mean()), float(ys.mean())))

    centers.sort(key=lambda point: point[0])
    if centers[1][0] - centers[0][0] < 120:
        raise OracleError(f"{name} PRE and POST columns are not visibly separated")
    return centers


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[2] != 3:
        raise OracleError("expected an RGB raster")
    cyan = _marker_centers(array, _CYAN, "cyan")
    orange = _marker_centers(array, _ORANGE, "orange")
    cyan_delta = cyan[0][1] - cyan[1][1]
    orange_delta = orange[0][1] - orange[1][1]
    difference = cyan_delta - orange_delta
    if abs(difference) < 0.5:
        raise OracleError("paired changes are tied at raster resolution")
    return "CYAN" if difference > 0 else "ORANGE"
