"""Deterministic renderer and analytic arm for a paired-change dot plot."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 512
HEIGHT = 512
LEFT_X = 160
RIGHT_X = 352
MARKER_RADIUS = 14
CYAN = (0, 145, 190)
ORANGE = (230, 105, 25)
BACKGROUND = (255, 255, 255)
QUARANTINE_MARGIN = 18.0


def _validated_pairs(scene: dict) -> tuple[tuple[int, int], tuple[int, int]]:
    cyan = scene["cyan_y"]
    orange = scene["orange_y"]
    if not isinstance(cyan, list) or not isinstance(orange, list):
        raise ValueError("marker coordinates must be lists")
    if len(cyan) != 2 or len(orange) != 2:
        raise ValueError("each sample must have PRE and POST coordinates")
    values = [*cyan, *orange]
    if any(isinstance(value, bool) or not isinstance(value, int) for value in values):
        raise ValueError("marker coordinates must be integers")
    if any(value < 135 or value > 377 for value in values):
        raise ValueError("marker lies outside the visible measurement range")
    return (cyan[0], cyan[1]), (orange[0], orange[1])


def sample_scene(seed: int) -> dict:
    """Sample balanced winners with varied signs, baselines, and margins."""

    rng = random.Random(int(seed))
    winner = "CYAN" if int(seed) % 2 == 0 else "ORANGE"
    if int(seed) % 7 == 0:
        gap = 10  # explicit finite-resolution quarantine coverage
    else:
        gap = (22, 30, 38, 48, 58, 68)[int(seed) % 6]

    for _ in range(2000):
        smaller_delta = rng.randint(-62, 34)
        larger_delta = smaller_delta + gap
        if winner == "CYAN":
            cyan_delta, orange_delta = larger_delta, smaller_delta
        else:
            cyan_delta, orange_delta = smaller_delta, larger_delta

        cyan_pre = rng.randint(175, 337)
        orange_pre = rng.randint(175, 337)
        cyan_post = cyan_pre - cyan_delta
        orange_post = orange_pre - orange_delta
        values = (cyan_pre, cyan_post, orange_pre, orange_post)
        if not all(145 <= value <= 367 for value in values):
            continue
        if abs(cyan_pre - orange_pre) < 52:
            continue
        if abs(cyan_post - orange_post) < 52:
            continue
        return {
            "cyan_y": [cyan_pre, cyan_post],
            "orange_y": [orange_pre, orange_post],
        }
    raise RuntimeError("could not sample a separated paired-change scene")


def render(scene: dict) -> Image.Image:
    cyan, orange = _validated_pairs(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    # A common measurement frame is visible, but the four endpoints remain
    # disconnected. The inverse arm deliberately recovers only those endpoints.
    draw.rounded_rectangle((92, 82, 420, 414), radius=10, fill=(249, 251, 253), outline=(95, 105, 115), width=2)
    for y in (128, 192, 256, 320, 384):
        draw.line((108, y, 404, y), fill=(221, 227, 232), width=1)
    for x in (LEFT_X, RIGHT_X):
        draw.line((x, 104, x, 392), fill=(210, 217, 223), width=1)
    draw.text((147, 424), "PRE", fill=(35, 42, 48), font=font)
    draw.text((338, 424), "POST", fill=(35, 42, 48), font=font)
    draw.text((195, 55), "PAIRED CHANGE", fill=(35, 42, 48), font=font)
    draw.text((50, 103), "HIGH", fill=(65, 72, 78), font=font)
    draw.text((55, 378), "LOW", fill=(65, 72, 78), font=font)

    for x, y in ((LEFT_X, cyan[0]), (RIGHT_X, cyan[1])):
        draw.ellipse((x - MARKER_RADIUS, y - MARKER_RADIUS, x + MARKER_RADIUS, y + MARKER_RADIUS), fill=CYAN)
    for x, y in ((LEFT_X, orange[0]), (RIGHT_X, orange[1])):
        draw.ellipse((x - MARKER_RADIUS, y - MARKER_RADIUS, x + MARKER_RADIUS, y + MARKER_RADIUS), fill=ORANGE)
    return image


def analytic_gold(scene: dict) -> str:
    cyan, orange = _validated_pairs(scene)
    cyan_delta = cyan[0] - cyan[1]
    orange_delta = orange[0] - orange[1]
    if cyan_delta == orange_delta:
        raise ValueError("tied paired changes are undefined")
    return "CYAN" if cyan_delta > orange_delta else "ORANGE"


def margin(scene: dict) -> float:
    cyan, orange = _validated_pairs(scene)
    cyan_delta = cyan[0] - cyan[1]
    orange_delta = orange[0] - orange[1]
    return float(abs(cyan_delta - orange_delta))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    _validated_pairs(scene)
    return []
