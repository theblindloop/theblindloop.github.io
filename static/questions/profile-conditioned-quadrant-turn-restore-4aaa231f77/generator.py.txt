"""Renderer and analytic label for a piecewise quarter-turn raster world."""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw

RENDERER_VERSION = "quadrant_turn_restore-0.1.0"
SYMBOLS = ("crown", "spear", "ribbon", "clover")
WIDTH, HEIGHT = 640, 460
TILE_SIZE = 324
PAD = 22
SIDE = 280
HALF = SIDE // 2
MARK = 8
PALETTES = (
    (18, 158, 181),
    (222, 64, 119),
    (75, 95, 211),
    (46, 168, 93),
    (226, 132, 25),
)


def _symbol_mask(
    symbol: str,
    dx: int = 0,
    dy: int = 0,
    scale: float = 1.0,
) -> np.ndarray:
    """Make a broad, low-frequency symbol on a square raster."""

    image = Image.new("L", (SIDE, SIDE), 0)
    draw = ImageDraw.Draw(image)
    cx = SIDE / 2 + dx
    cy = SIDE / 2 + dy
    sx = 1.0 * scale

    def p(x: float, y: float) -> tuple[int, int]:
        return (round(cx + x * sx), round(cy + y * sx))

    if symbol == "crown":
        points = [p(-72, 48), p(-64, -48), p(-24, -8), p(0, -62), p(25, -8), p(65, -48), p(72, 48)]
        draw.line(points, fill=255, width=max(10, round(15 * scale)), joint="curve")
        draw.line([p(-70, 48), p(70, 48)], fill=255, width=max(10, round(15 * scale)))
    elif symbol == "spear":
        draw.line([p(0, -64), p(0, 70)], fill=255, width=max(10, round(15 * scale)))
        draw.polygon([p(0, -86), p(-31, -35), p(0, -45), p(31, -35)], fill=255)
        draw.line([p(-30, 68), p(30, 68)], fill=255, width=max(10, round(14 * scale)))
    elif symbol == "ribbon":
        first = [p(x, 36 * np.sin((x + 58) / 30.0)) for x in range(-72, 73, 3)]
        second = [p(x, 36 * np.sin((x + 58) / 30.0) + 28) for x in range(-72, 73, 3)]
        draw.line(first, fill=255, width=max(9, round(13 * scale)), joint="curve")
        draw.line(second, fill=255, width=max(9, round(13 * scale)), joint="curve")
        draw.line([p(-70, -52), p(-45, -25)], fill=255, width=max(9, round(13 * scale)))
        draw.line([p(45, 25), p(70, 52)], fill=255, width=max(9, round(13 * scale)))
    elif symbol == "clover":
        radius = 38 * scale
        for ox, oy in ((-35, -30), (35, -30), (-35, 30), (35, 30)):
            draw.ellipse(
                [round(cx + ox * scale - radius), round(cy + oy * scale - radius),
                 round(cx + ox * scale + radius), round(cy + oy * scale + radius)],
                outline=255,
                width=max(9, round(13 * scale)),
            )
        draw.line([p(0, 30), p(0, 78)], fill=255, width=max(9, round(13 * scale)))
        draw.line([p(-24, 77), p(24, 77)], fill=255, width=max(9, round(13 * scale)))
    else:
        raise ValueError(f"unknown symbol {symbol!r}")
    return np.asarray(image, dtype=np.float32) / 255.0


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    near_boundary = int(seed) % 7 == 0
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "tile_x": rng.randint(42, 270),
        "tile_y": rng.randint(34, 96),
        "carrier_phase": rng.randrange(8),
        "carrier_amplitude": round(rng.uniform(72.0, 84.0) if near_boundary else rng.uniform(54.0, 68.0), 3),
        "signal_strength": round(rng.uniform(54.0, 64.0) if near_boundary else rng.uniform(78.0, 94.0), 3),
        "base_level": rng.randint(154, 198),
        "symbol_dx": rng.choice((-10, 0, 10)),
        "symbol_dy": rng.choice((-10, 0, 10)),
        "symbol_scale": rng.choice((0.94, 1.0, 1.06)),
        "marker_palette_id": rng.randrange(len(PALETTES)),
        "symbol": SYMBOLS[int(seed) % len(SYMBOLS)],
        "seed": int(seed),
    }


def _carrier(phase: int, amplitude: float) -> np.ndarray:
    yy, xx = np.indices((SIDE, SIDE))
    blocks = ((yy // 4) + ((xx + int(phase)) // 4)) % 2
    return np.where(blocks == 0, amplitude, -amplitude).astype(np.float32)


def _quarter_turn_encode(values: np.ndarray) -> np.ndarray:
    """Rotate each displayed quadrant away from the common orientation."""

    result = np.empty_like(values)
    h = HALF
    result[:h, :h] = np.rot90(values[:h, :h], -1)
    result[:h, h:] = np.rot90(values[:h, h:], 1)
    result[h:, :h] = np.rot90(values[h:, :h], 1)
    result[h:, h:] = np.rot90(values[h:, h:], -1)
    return result


def render(scene: dict) -> Image.Image:
    base = float(scene["base_level"])
    mask = _symbol_mask(
        str(scene["symbol"]),
        dx=int(scene["symbol_dx"]),
        dy=int(scene["symbol_dy"]),
        scale=float(scene["symbol_scale"]),
    )
    raw = np.clip(
        base
        - float(scene["signal_strength"]) * mask
        + _carrier(int(scene["carrier_phase"]), float(scene["carrier_amplitude"])),
        0.0,
        255.0,
    )
    scrambled = _quarter_turn_encode(raw)
    tile = np.full((TILE_SIZE, TILE_SIZE), base, dtype=np.float32)
    tile[PAD : PAD + SIDE, PAD : PAD + SIDE] = scrambled
    tile_rgb = Image.fromarray(np.rint(tile).astype(np.uint8), mode="L").convert("RGB")
    draw = ImageDraw.Draw(tile_rgb)
    draw.rectangle([0, 0, TILE_SIZE - 1, TILE_SIZE - 1], outline=(36, 36, 36), width=3)
    color = PALETTES[int(scene["marker_palette_id"]) % len(PALETTES)]
    for cx, cy in (
        (MARK, MARK),
        (TILE_SIZE - 1 - MARK, MARK),
        (MARK, TILE_SIZE - 1 - MARK),
        (TILE_SIZE - 1 - MARK, TILE_SIZE - 1 - MARK),
    ):
        draw.rectangle([cx - 7, cy - 7, cx + 7, cy + 7], fill=color)
    canvas = Image.new("RGB", (int(scene["width"]), int(scene["height"])), (255, 255, 255))
    canvas.paste(tile_rgb, (int(scene["tile_x"]), int(scene["tile_y"])))
    return canvas


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def margin(scene: dict) -> float:
    return round(float(scene["signal_strength"]) - 0.65 * float(scene["carrier_amplitude"]), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 14.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["carrier_phase"] = int(scene["carrier_phase"]) + 8
    return [("carrier_period_alias", twin)]
