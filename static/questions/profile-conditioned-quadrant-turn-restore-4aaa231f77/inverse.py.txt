"""Independent image-only inverse for quarter-turned quadrants."""

from __future__ import annotations

from collections import deque
from functools import lru_cache

import numpy as np
from PIL import Image, ImageDraw

SYMBOLS = ("crown", "spear", "ribbon", "clover")
TILE_SIZE = 324
PAD = 22
SIDE = 280
HALF = 140
MARK = 8


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    found: list[tuple[float, float, int]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= 50:
            found.append((float(np.mean([p[1] for p in points])), float(np.mean([p[0] for p in points])), len(points)))
    return found


def _find_square(image: Image.Image) -> np.ndarray:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    spread = rgb.max(axis=2).astype(int) - rgb.min(axis=2).astype(int)
    colored = (spread > 45) & (rgb.max(axis=2) > 100) & (rgb.min(axis=2) < 235)
    markers = _components(colored)
    if len(markers) < 4:
        raise ValueError("four colored corner fiducials were not found")
    markers = sorted(markers, key=lambda item: item[2], reverse=True)[:4]
    min_x = min(item[0] for item in markers)
    max_x = max(item[0] for item in markers)
    min_y = min(item[1] for item in markers)
    max_y = max(item[1] for item in markers)
    if max_x - min_x < SIDE or max_y - min_y < SIDE:
        raise ValueError("fiducials do not enclose the expected raster")
    tile_x = int(round(min_x - MARK))
    tile_y = int(round(min_y - MARK))
    gray = np.asarray(image.convert("L"), dtype=np.float32)
    square = gray[tile_y + PAD : tile_y + PAD + SIDE, tile_x + PAD : tile_x + PAD + SIDE]
    if square.shape != (SIDE, SIDE):
        raise ValueError("marked raster is incomplete")
    return square


def _quarter_turn_decode(values: np.ndarray) -> np.ndarray:
    result = np.empty_like(values)
    h = HALF
    result[:h, :h] = np.rot90(values[:h, :h], 1)
    result[:h, h:] = np.rot90(values[:h, h:], -1)
    result[h:, :h] = np.rot90(values[h:, :h], -1)
    result[h:, h:] = np.rot90(values[h:, h:], 1)
    return result


def _box_blur(values: np.ndarray) -> np.ndarray:
    size = 9
    radius = size // 2
    padded = np.pad(values, ((radius, radius), (radius, radius)), mode="edge")
    integral = np.pad(padded, ((1, 0), (1, 0)), mode="constant").cumsum(0).cumsum(1)
    return (
        integral[size:, size:]
        - integral[:-size, size:]
        - integral[size:, :-size]
        + integral[:-size, :-size]
    ) / float(size * size)


def _symbol_mask(symbol: str, dx: int, dy: int, scale: float) -> np.ndarray:
    image = Image.new("L", (SIDE, SIDE), 0)
    draw = ImageDraw.Draw(image)
    cx = SIDE / 2 + dx
    cy = SIDE / 2 + dy
    s = float(scale)

    def p(x: float, y: float) -> tuple[int, int]:
        return (round(cx + x * s), round(cy + y * s))

    if symbol == "crown":
        points = [p(-72, 48), p(-64, -48), p(-24, -8), p(0, -62), p(25, -8), p(65, -48), p(72, 48)]
        draw.line(points, fill=255, width=max(10, round(15 * s)), joint="curve")
        draw.line([p(-70, 48), p(70, 48)], fill=255, width=max(10, round(15 * s)))
    elif symbol == "spear":
        draw.line([p(0, -64), p(0, 70)], fill=255, width=max(10, round(15 * s)))
        draw.polygon([p(0, -86), p(-31, -35), p(0, -45), p(31, -35)], fill=255)
        draw.line([p(-30, 68), p(30, 68)], fill=255, width=max(10, round(14 * s)))
    elif symbol == "ribbon":
        first = [p(x, 36 * np.sin((x + 58) / 30.0)) for x in range(-72, 73, 3)]
        second = [p(x, 36 * np.sin((x + 58) / 30.0) + 28) for x in range(-72, 73, 3)]
        draw.line(first, fill=255, width=max(9, round(13 * s)), joint="curve")
        draw.line(second, fill=255, width=max(9, round(13 * s)), joint="curve")
        draw.line([p(-70, -52), p(-45, -25)], fill=255, width=max(9, round(13 * s)))
        draw.line([p(45, 25), p(70, 52)], fill=255, width=max(9, round(13 * s)))
    elif symbol == "clover":
        radius = 38 * s
        for ox, oy in ((-35, -30), (35, -30), (-35, 30), (35, 30)):
            draw.ellipse(
                [round(cx + ox * s - radius), round(cy + oy * s - radius),
                 round(cx + ox * s + radius), round(cy + oy * s + radius)],
                outline=255,
                width=max(9, round(13 * s)),
            )
        draw.line([p(0, 30), p(0, 78)], fill=255, width=max(9, round(13 * s)))
        draw.line([p(-24, 77), p(24, 77)], fill=255, width=max(9, round(13 * s)))
    else:
        raise ValueError(symbol)
    return np.asarray(image, dtype=np.float32) / 255.0


@lru_cache(maxsize=1)
def _template_bank() -> dict[str, tuple[np.ndarray, np.ndarray]]:
    bank: dict[str, tuple[np.ndarray, np.ndarray]] = {}
    values_x = []
    for scale in (0.94, 1.0, 1.06):
        for dx, dy in ((-10, -10), (-10, 0), (-10, 10), (0, -10), (0, 0), (0, 10), (10, -10), (10, 0), (10, 10)):
            values_x.append((scale, dx, dy))
    roi = np.ones((SIDE, SIDE), dtype=bool)
    roi[:18, :] = False
    roi[-18:, :] = False
    roi[:, :18] = False
    roi[:, -18:] = False
    for symbol in SYMBOLS:
        templates = np.stack([_box_blur(_symbol_mask(symbol, dx, dy, scale))[roi] for scale, dx, dy in values_x]).astype(np.float32)
        centered = templates - templates.mean(axis=1, keepdims=True)
        norms = np.linalg.norm(centered, axis=1) + 1e-6
        bank[symbol] = centered, norms
    return bank


def _decode(image: Image.Image) -> tuple[str, float]:
    restored = _quarter_turn_decode(_find_square(image))
    blurred = _box_blur(restored)
    baseline = float(np.median(blurred))
    observed = (baseline - blurred).astype(np.float32)
    roi = np.ones((SIDE, SIDE), dtype=bool)
    roi[:18, :] = False
    roi[-18:, :] = False
    roi[:, :18] = False
    roi[:, -18:] = False
    vector = observed[roi]
    vector -= vector.mean()
    vector_norm = float(np.linalg.norm(vector)) + 1e-6
    scores: list[tuple[float, str]] = []
    for symbol in SYMBOLS:
        templates, norms = _template_bank()[symbol]
        correlations = (templates @ vector) / (norms * vector_norm)
        scores.append((float(np.max(correlations)), symbol))
    scores.sort(reverse=True)
    top, runner = scores[0], scores[1]
    margin = top[0] - runner[0]
    if top[0] < 0.10 or margin < 0.012:
        raise ValueError("restored symbol is ambiguous")
    return top[1], float(margin)


@lru_cache(maxsize=4096)
def _decode_cached(data: bytes, width: int, height: int) -> tuple[str, float]:
    rgb = np.frombuffer(data, dtype=np.uint8).reshape((height, width, 3))
    return _decode(Image.fromarray(rgb, mode="RGB"))


def decision_from_image(image: Image.Image) -> str:
    """Classify the symbol from the final raster and nothing else."""

    rgb = image.convert("RGB")
    return _decode_cached(rgb.tobytes(), rgb.width, rgb.height)[0]


def confidence_margin(image: Image.Image) -> float:
    rgb = image.convert("RGB")
    return _decode_cached(rgb.tobytes(), rgb.width, rgb.height)[1]
