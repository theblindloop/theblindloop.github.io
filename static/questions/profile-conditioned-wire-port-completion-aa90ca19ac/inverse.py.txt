"""Independent pixel-only inverse arm using only raster color components."""
from __future__ import annotations

from collections import deque
import numpy as np

PALETTE = {"red": (220, 62, 62), "blue": (55, 110, 220), "gold": (230, 168, 35)}


def _components(mask: np.ndarray, minimum: int = 1):
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        todo = [(int(y), int(x))]
        seen[y, x] = True
        pts = []
        while todo:
            yy, xx = todo.pop()
            pts.append((yy, xx))
            for dy, dx in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                ny, nx = yy + dy, xx + dx
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    todo.append((ny, nx))
        if len(pts) >= minimum:
            output.append(np.asarray(pts, dtype=np.float64))
    return output


def _centers_for_color(arr: np.ndarray, rgb):
    delta = np.max(np.abs(arr.astype(np.int16) - np.asarray(rgb, dtype=np.int16)), axis=2)
    return [component.mean(axis=0) for component in _components(delta <= 4, minimum=80)]


def decision_from_image(image) -> str:
    arr = np.asarray(image.convert("RGB"))
    terminals = []
    for name, rgb in PALETTE.items():
        terminals.extend((center, name) for center in _centers_for_color(arr, rgb))
    if len(terminals) != 18:
        raise ValueError(f"expected 18 colored terminals, found {len(terminals)}")

    # Group terminals into the three visually separated option bands by y gaps.
    terminals.sort(key=lambda item: item[0][0])
    bands = [terminals[i:i + 6] for i in range(0, 18, 6)]
    scores = []
    for band in bands:
        ys = [center[0] for center, _ in band]
        xs = [center[1] for center, _ in band]
        y0, y1 = int(min(ys) - 18), int(max(ys) + 19)
        x0, x1 = int(min(xs) - 5), int(max(xs) + 6)
        crop = arr[y0:y1, x0:x1]
        # Wire ink is near (32,34,42); frame, text, and socket colors are excluded.
        target = np.asarray((32, 34, 42), dtype=np.int16)
        dark = np.max(np.abs(crop.astype(np.int16) - target), axis=2) <= 4
        wires = _components(dark, minimum=100)
        if len(wires) != 3:
            raise ValueError(f"expected three wire components in a panel, found {len(wires)}")
        matched = 0
        for wire in wires:
            global_pts = wire + np.asarray((y0, x0), dtype=np.float64)
            distances = []
            for center, color in band:
                diff = global_pts - center
                distances.append((float(np.min(np.sum(diff * diff, axis=1))), color))
            distances.sort(key=lambda item: item[0])
            if distances[0][1] == distances[1][1]:
                matched += 1
        scores.append(matched)
    winners = [i for i, score in enumerate(scores) if score == 3]
    if len(winners) != 1:
        raise ValueError(f"no unique complete option; scores={scores}")
    return ("A", "B", "C")[winners[0]]
