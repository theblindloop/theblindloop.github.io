"""Deterministic renderer for the wire-terminal correspondence world."""
from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 720, 600
COLORS = ("red", "blue", "gold")
RGB = {"red": (220, 62, 62), "blue": (55, 110, 220), "gold": (230, 168, 35)}
LABELS = ("A", "B", "C")

# Five Catalan non-crossing perfect matchings around six clockwise positions.
MATCHINGS = (
    ((0, 1), (2, 3), (4, 5)),
    ((0, 1), (2, 5), (3, 4)),
    ((0, 3), (1, 2), (4, 5)),
    ((0, 5), (1, 2), (3, 4)),
    ((0, 5), (1, 4), (2, 3)),
)


def _score(option: dict) -> int:
    colors = option["terminal_colors"]
    return sum(colors[a] == colors[b] for a, b in option["pairs"])


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    winner = seed % 3
    options = []
    for option_index in range(3):
        pairs = [list(pair) for pair in rng.choice(MATCHINGS)]
        pair_colors = list(COLORS)
        rng.shuffle(pair_colors)
        terminals = [None] * 6
        for color, (a, b) in zip(pair_colors, pairs):
            terminals[a] = color
            terminals[b] = color
        if option_index != winner:
            # Swap endpoints belonging to two distinct pairs. Counts remain two of
            # each color, but exactly one wire stays matched (runner-up score 1).
            p, q = rng.sample(range(3), 2)
            a = pairs[p][rng.randrange(2)]
            b = pairs[q][rng.randrange(2)]
            terminals[a], terminals[b] = terminals[b], terminals[a]
        options.append({"terminal_colors": terminals, "pairs": pairs})
    return {"options": options}


def analytic_gold(scene: dict) -> str:
    options = scene["options"]
    scores = []
    for option in options:
        colors = option["terminal_colors"]
        scores.append(sum(colors[a] == colors[b] for a, b in option["pairs"]))
    winners = [i for i, score in enumerate(scores) if score == 3]
    return LABELS[winners[0]] if len(winners) == 1 else "AMBIGUOUS"


def margin(scene: dict) -> float:
    scores = sorted((_score(option) for option in scene["options"]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    return analytic_gold(scene) == "AMBIGUOUS" or margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int):
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (250, 250, 253))
    draw = ImageDraw.Draw(image)
    draw.text((360, 18), "TRACE THE THREE WIRES", fill=(38, 40, 50), font=_font(24), anchor="ma")
    draw.text((360, 48), "Find the option whose wire ends always match in color", fill=(70, 73, 85), font=_font(15), anchor="ma")
    # Clockwise perimeter locations: top-left, top-right, right, bottom-right,
    # bottom-left, left. Straight chords for MATCHINGS never cross.
    offsets = ((-125, -44), (125, -44), (190, 0), (125, 44), (-125, 44), (-190, 0))
    for index, option in enumerate(scene["options"]):
        top = 82 + index * 168
        left, right, bottom = 72, 670, top + 142
        draw.rounded_rectangle((left, top, right, bottom), radius=15, fill=(255, 255, 255), outline=(202, 205, 216), width=4)
        draw.text((39, top + 71), LABELS[index], fill=(38, 40, 50), font=_font(30), anchor="mm")
        cx, cy = 371, top + 71
        pts = [(cx + dx, cy + dy) for dx, dy in offsets]
        for a, b in option["pairs"]:
            draw.line((pts[a], pts[b]), fill=(32, 34, 42), width=8)
        for point, color in zip(pts, option["terminal_colors"]):
            x, y = point
            draw.ellipse((x - 13, y - 13, x + 13, y + 13), fill=RGB[color])
            draw.ellipse((x - 4, y - 4, x + 4, y + 4), fill=(255, 255, 255))
    return image
