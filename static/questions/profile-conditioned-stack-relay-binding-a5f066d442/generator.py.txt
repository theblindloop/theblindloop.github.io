"""Deterministic scene sampling and integer raster rendering for stack_relay_binding.

Six evenly spaced columns of grey square tokens stand on one black baseline.
A black triangular pointer above the row selects exactly one column.  The
number of tokens in that column is read as a count, and that count is then
spent as a *position*: the answer is the tag colour of the column at that
place counting from the left edge.  The scene therefore carries two readings
of two different kinds, a cardinal one and an ordinal one, and the second is
addressed by the first.  Nothing about the pointed column's own tag, its own
height rank, or its own place in the row determines the answer.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "stack_relay_binding-renderer-0.1.0"

WIDTH = 940
HEIGHT = 460
BACKGROUND = (255, 255, 255)

COLUMNS = 6
COLOR_NAMES = ("red", "orange", "green", "blue", "purple", "brown")
COLOR_RGB = {
    "red": (208, 30, 38),
    "orange": (232, 120, 20),
    "green": (16, 138, 60),
    "blue": (30, 66, 208),
    "purple": (128, 36, 160),
    "brown": (124, 74, 32),
}
TOKEN_RGB = (150, 150, 150)
INK_RGB = (0, 0, 0)

BASELINE_Y = 372
BASELINE_H = 4
BASELINE_X0 = 40
BASELINE_X1 = 900
TOKEN = 24
TOKEN_HALF = TOKEN // 2
STACK_LIFT = 18
DISK_CY = 416
DISK_R = 15
ARROW_TIP_Y = 88
ARROW_TOP_Y = 44
ARROW_HALF = 16

PITCH_MIN = 124
PITCH_MAX = 142
X0_MIN = 92
X0_MAX = 112
GAP_MIN = 17
GAP_MAX = 23

QUARANTINE_MARGIN = 16.0
STRUCTURAL_MARGIN_CAP = 18.0


def _column_x(scene: dict, index: int) -> int:
    return int(scene["x0"]) + index * int(scene["pitch"])


def _arrow_x(scene: dict) -> int:
    return _column_x(scene, int(scene["start"])) + int(scene["arrow_dx"])


def _pointer_margin(scene: dict) -> float:
    """Horizontal pixels the pointer would have to slide to select elsewhere."""

    tip = _arrow_x(scene)
    distances = sorted(abs(tip - _column_x(scene, i)) for i in range(COLUMNS))
    return float((distances[1] - distances[0]) / 2.0)


def _try_scene(
    rng: random.Random,
    colour_phase: int,
    target: int,
    self_flag: bool,
    tall_flag: bool,
    tight: bool,
) -> dict | None:
    counts = [rng.randint(1, COLUMNS) for _ in range(COLUMNS)]
    if self_flag:
        start = target
    else:
        start = rng.choice([i for i in range(COLUMNS) if i != target])
    counts[start] = target + 1

    top = max(counts)
    if counts.count(top) != 1:
        return None
    if not self_flag and (counts.index(top) == target) != tall_flag:
        return None

    # The answer colour is pinned to a seed cycle, the remaining tags are a
    # free permutation, so tag colour never predicts position or height.
    answer_colour = COLOR_NAMES[colour_phase]
    rest = [name for name in COLOR_NAMES if name != answer_colour]
    rng.shuffle(rest)
    tags = list(rest)
    tags.insert(target, answer_colour)

    pitch = rng.randrange(PITCH_MIN, PITCH_MAX + 1)
    gap = rng.randrange(GAP_MIN, GAP_MAX + 1)
    x0 = rng.randrange(X0_MIN, X0_MAX + 1)

    # Sample the runner-up pointer margin first, then realise it as an offset
    # aimed at the interior so the second-nearest column is a real rival.
    if tight:
        wanted = rng.randint(9, 15)
    else:
        wanted = rng.randint(22, pitch // 2 - 4)
    offset = pitch // 2 - wanted
    if start == 0:
        sign = 1
    elif start == COLUMNS - 1:
        sign = -1
    else:
        sign = rng.choice((-1, 1))

    scene = {
        "counts": counts,
        "tags": tags,
        "start": start,
        "pitch": pitch,
        "x0": x0,
        "gap": gap,
        "arrow_dx": sign * offset,
    }
    if _column_x(scene, COLUMNS - 1) + TOKEN_HALF > BASELINE_X1 - 10:
        return None
    stack = COLUMNS * TOKEN + (COLUMNS - 1) * gap
    if BASELINE_Y - STACK_LIFT - stack < ARROW_TIP_Y + 6:
        return None
    if abs(_pointer_margin(scene) - wanted) > 1e-9:
        return None
    return scene


def sample_scene(seed: int) -> dict:
    """Deterministic latent scene for one integer seed."""

    value = int(seed)
    rng = random.Random(value * 7919 + 13)
    # The answer colour rides an exact seed cycle; the answer position, the
    # self-pointing case and the tallest-stack case ride independent draws so
    # that no cheap positional reading beats chance.
    phases = random.Random(value * 104729 + 7)
    colour_phase = value % COLUMNS
    target = phases.randrange(COLUMNS)
    self_flag = phases.randrange(COLUMNS) == 0
    # A target at the last place already forces the pointed stack to the
    # maximum height, so the tallest stack cannot additionally be made to sit
    # over the target there; that combination is simply not requested.
    tall_flag = (
        phases.randrange(COLUMNS) == 0
        and not self_flag
        and target != COLUMNS - 1
    )
    tight = value % 5 == 0
    for _ in range(60000):
        scene = _try_scene(rng, colour_phase, target, self_flag, tall_flag, tight)
        if scene is not None:
            return scene
    raise RuntimeError(f"no admissible scene for seed {seed}")


def analytic_gold(scene: dict) -> str:
    """Tag colour of the column whose place equals the pointed column's count."""

    place = int(scene["counts"][int(scene["start"])])
    return str(scene["tags"][place - 1])


def margin(scene: dict) -> float:
    """Pixels of perturbation the picture tolerates before the answer moves."""

    return float(
        min(_pointer_margin(scene), float(int(scene["gap"])), STRUCTURAL_MARGIN_CAP)
    )


def is_quarantined(scene: dict) -> bool:
    """Pointers close to the midpoint between two columns leave headline claims."""

    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No pixel-identical latent alias exists; see candidate.json."""

    return []


def render(scene: dict) -> Image.Image:
    """Rasterize one scene with hard integer edges and no antialiasing."""

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    gap = int(scene["gap"])

    draw.rectangle(
        [BASELINE_X0, BASELINE_Y, BASELINE_X1, BASELINE_Y + BASELINE_H - 1],
        fill=INK_RGB,
    )

    for index in range(COLUMNS):
        x = _column_x(scene, index)
        for level in range(int(scene["counts"][index])):
            bottom = BASELINE_Y - STACK_LIFT - level * (TOKEN + gap)
            draw.rectangle(
                [x - TOKEN_HALF, bottom - TOKEN + 1, x + TOKEN_HALF - 1, bottom],
                fill=TOKEN_RGB,
            )
        draw.ellipse(
            [x - DISK_R, DISK_CY - DISK_R, x + DISK_R, DISK_CY + DISK_R],
            fill=COLOR_RGB[scene["tags"][index]],
        )

    tip = _arrow_x(scene)
    draw.polygon(
        [
            (tip, ARROW_TIP_Y),
            (tip - ARROW_HALF, ARROW_TOP_Y),
            (tip + ARROW_HALF, ARROW_TOP_Y),
        ],
        fill=INK_RGB,
    )
    return image
