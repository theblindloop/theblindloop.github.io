"""Independent image-only inverse for the area-ledger atlas."""

from __future__ import annotations

from PIL import Image

FEATURES = [(68, 188, 207), (236, 113, 87), (151, 111, 204), (151, 197, 91)]
TAGS = {
    "gold": (239, 184, 47),
    "red": (218, 57, 68),
    "green": (47, 164, 96),
    "blue": (48, 105, 207),
}


def _centroid_for_color(image: Image.Image, rgb: tuple[int, int, int]) -> tuple[float, float]:
    pixels = image.load()
    width, height = image.size
    xs = []
    ys = []
    for y in range(height):
        for x in range(width):
            if pixels[x, y] == rgb:
                xs.append(x)
                ys.append(y)
    if len(xs) < 100:
        raise ValueError("missing or damaged tag")
    return sum(xs) / len(xs), sum(ys) / len(ys)


def decision_from_image(image: Image.Image) -> str:
    picture = image.convert("RGB")
    tag_centers = {name: _centroid_for_color(picture, rgb) for name, rgb in TAGS.items()}
    background = picture.getpixel((0, 0))
    core_left = round(min(center[0] for center in tag_centers.values()))
    core_right = round(max(center[0] for center in tag_centers.values()))
    core_top = round(min(center[1] for center in tag_centers.values()))
    core_bottom = round(max(center[1] for center in tag_centers.values()))
    core_background = sum(
        picture.getpixel((x, y)) == background
        for y in range(core_top, core_bottom + 1)
        for x in range(core_left, core_right + 1)
    )
    if core_background:
        raise ValueError("broken central atlas bridge")
    totals = {name: [0, 0, 0, 0] for name in TAGS}
    pixels = picture.load()
    width, height = picture.size
    feature_index = {rgb: index for index, rgb in enumerate(FEATURES)}
    for y in range(height):
        for x in range(width):
            index = feature_index.get(pixels[x, y])
            if index is None:
                continue
            owner = min(
                tag_centers,
                key=lambda name: (x - tag_centers[name][0]) ** 2
                + (y - tag_centers[name][1]) ** 2,
            )
            totals[owner][index] += 1
    if any(sum(values) < 9000 for values in totals.values()):
        raise ValueError("incomplete mosaic")
    reference = totals["gold"]
    distances = {
        name: sum(abs(a - b) for a, b in zip(reference, totals[name]))
        for name in ("red", "green", "blue")
    }
    ordered = sorted(distances.items(), key=lambda item: (item[1], item[0]))
    if ordered[0][1] == ordered[1][1]:
        raise ValueError("ambiguous ledger match")
    return ordered[0][0]
