"""Deterministic latent sampler and Pillow renderer."""

from __future__ import annotations

import random
from collections import Counter

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 768, 640
BACKGROUND = (248, 246, 239)
NAVY = (25, 38, 62)
PANEL_INK = [(68, 188, 207), (236, 113, 87), (151, 111, 204), (151, 197, 91)]
TAG_RGB = {
    "gold": (239, 184, 47),
    "red": (218, 57, 68),
    "green": (47, 164, 96),
    "blue": (48, 105, 207),
}
TAG_NAMES = ("red", "green", "blue")
BASE_COUNTS = (4, 5, 7, 9)
POSITIONS = ((0, 0), (1, 0), (0, 1), (1, 1))


def _grid_from_counts(counts: list[int], rng: random.Random) -> list[list[int]]:
    values = [color for color, count in enumerate(counts) for _ in range(count)]
    rng.shuffle(values)
    return [values[row * 5 : (row + 1) * 5] for row in range(5)]


def _ledger(grid: list[list[int]]) -> tuple[int, ...]:
    counts = Counter(value for row in grid for value in row)
    return tuple(counts[index] for index in range(4))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    reference_counts = list(BASE_COUNTS)
    rng.shuffle(reference_counts)
    correct_tag = TAG_NAMES[int(seed) % len(TAG_NAMES)]
    ledgers: dict[str, list[int]] = {correct_tag: list(reference_counts)}
    other_tags = [tag for tag in TAG_NAMES if tag != correct_tag]
    rng.shuffle(other_tags)
    for tag, amount in zip(other_tags, (1, 2)):
        counts = list(reference_counts)
        donors = [index for index, value in enumerate(counts) if value > amount + 1]
        donor = rng.choice(donors)
        receiver = rng.choice([index for index in range(4) if index != donor])
        counts[donor] -= amount
        counts[receiver] += amount
        ledgers[tag] = counts
    position_tags = ["gold", *TAG_NAMES]
    rng.shuffle(position_tags)
    grids = []
    for tag in position_tags:
        counts = reference_counts if tag == "gold" else ledgers[tag]
        grids.append(_grid_from_counts(list(counts), rng))
    return {
        "grids": grids,
        "tags": position_tags,
        "cell_size": 28 + (int(seed) % 2),
        "atlas_origin": [96 + (int(seed) % 3), 82 + ((int(seed) // 3) % 3)],
    }


def _panel_geometry(origin: list[int], cell_size: int, position: tuple[int, int]):
    ox, oy = origin
    column, row = position
    left = ox + 44 + column * 270
    top = oy + 54 + row * 228
    grid_w = 5 * cell_size
    grid_x = left + (174 - grid_w) // 2
    grid_y = top + 30
    return grid_x, grid_y, (grid_x + grid_w // 2, grid_y + grid_w // 2)


def render(scene: dict) -> Image.Image:
    grids = scene["grids"]
    tags = scene["tags"]
    cell_size = int(scene["cell_size"])
    origin = list(scene["atlas_origin"])
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    ox, oy = origin
    draw.rounded_rectangle((ox, oy, ox + 576, oy + 492), radius=30, fill=NAVY)
    draw.rounded_rectangle((ox + 18, oy + 18, ox + 558, oy + 474), radius=22, outline=(61, 77, 104), width=3)
    for grid, tag, position in zip(grids, tags, POSITIONS):
        grid_x, grid_y, tag_center = _panel_geometry(origin, cell_size, position)
        del tag_center
        grid_w = 5 * cell_size
        draw.rectangle(
            (grid_x - 7, grid_y - 7, grid_x + grid_w + 6, grid_y + grid_w + 6),
            outline=TAG_RGB[tag], width=5,
        )
        for row in range(5):
            for column in range(5):
                x0 = grid_x + column * cell_size + 2
                y0 = grid_y + row * cell_size + 2
                x1 = grid_x + (column + 1) * cell_size - 3
                y1 = grid_y + (row + 1) * cell_size - 3
                draw.rectangle((x0, y0, x1, y1), fill=PANEL_INK[grid[row][column]])
    return image


def analytic_gold(scene: dict) -> str:
    grids = scene["grids"]
    tags = scene["tags"]
    by_tag = {tag: _ledger(grid) for tag, grid in zip(tags, grids)}
    reference = by_tag["gold"]
    distances = {
        tag: sum(abs(a - b) for a, b in zip(reference, by_tag[tag]))
        for tag in TAG_NAMES
    }
    return min(distances, key=distances.get)


def margin(scene: dict) -> float:
    grids = scene["grids"]
    tags = scene["tags"]
    by_tag = {tag: _ledger(grid) for tag, grid in zip(tags, grids)}
    reference = by_tag["gold"]
    distances = sorted(
        sum(abs(a - b) for a, b in zip(reference, by_tag[tag])) for tag in TAG_NAMES
    )
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    grids = scene["grids"]
    tags = scene["tags"]
    if sorted(tags) != ["blue", "gold", "green", "red"]:
        return True
    if any(len(grid) != 5 or any(len(row) != 5 for row in grid) for grid in grids):
        return True
    by_tag = {tag: _ledger(grid) for tag, grid in zip(tags, grids)}
    if any(sum(ledger) != 25 or min(ledger) < 1 for ledger in by_tag.values()):
        return True
    reference = by_tag["gold"]
    distances = sorted(
        sum(abs(a - b) for a, b in zip(reference, by_tag[tag])) for tag in TAG_NAMES
    )
    return distances[0] != 0 or distances[1] <= distances[0]


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
