"""Independent pixel-only inverse arm for the lowercase-m hump question.

This module intentionally duplicates only the public visual rule: the target is
the sole chromatic connected trace, while proof-sheet furniture is gray.  It
does not import or inspect renderer state, latent fields, manifests, filenames,
or analytic gold.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "lowercase-m-humps-oracle-0.1.0"


class OracleAbstention(ValueError):
    """The raster does not expose a single resolvable practice stroke."""


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Return 8-connected foreground components as arrays of (y, x) pixels."""

    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for sy, sx in np.argwhere(mask):
        y0, x0 = int(sy), int(sx)
        if seen[y0, x0]:
            continue
        stack = [(y0, x0)]
        seen[y0, x0] = True
        points: list[tuple[int, int]] = []
        while stack:
            y, x = stack.pop()
            points.append((y, x))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dy == 0 and dx == 0:
                        continue
                    ny, nx = y + dy, x + dx
                    if (
                        0 <= ny < height
                        and 0 <= nx < width
                        and mask[ny, nx]
                        and not seen[ny, nx]
                    ):
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        output.append(np.asarray(points, dtype=np.int32))
    return output


def _bridge_short_false_gaps(values: np.ndarray, maximum: int = 2) -> np.ndarray:
    result = values.copy()
    index = 0
    while index < len(result):
        if result[index]:
            index += 1
            continue
        end = index
        while end < len(result) and not result[end]:
            end += 1
        if index > 0 and end < len(result) and end - index <= maximum:
            result[index:end] = True
        index = end
    return result


def _true_run_lengths(values: np.ndarray) -> list[int]:
    lengths: list[int] = []
    index = 0
    while index < len(values):
        if not values[index]:
            index += 1
            continue
        end = index
        while end < len(values) and values[end]:
            end += 1
        lengths.append(end - index)
        index = end
    return lengths


def _count_arches(component: np.ndarray, shape: tuple[int, int]) -> int:
    mask = np.zeros(shape, dtype=bool)
    mask[component[:, 0], component[:, 1]] = True
    xs_all = component[:, 1]
    ys_all = component[:, 0]
    x_min, x_max = int(xs_all.min()), int(xs_all.max())
    y_min, y_max = int(ys_all.min()), int(ys_all.max())
    box_width = x_max - x_min + 1
    box_height = y_max - y_min + 1
    if not 64 <= box_width <= 112 or not 15 <= box_height <= 48:
        raise OracleAbstention(
            f"colored trace has unsupported visible extent {box_width}x{box_height}"
        )

    # The first few columns contain the fixed left stem.  Thereafter, the median
    # colored y in each column is a direct centerline measurement of the humps.
    trim = max(6, int(round(box_width * 0.08)))
    start = x_min + trim
    xs = np.arange(start, x_max + 1, dtype=np.int32)
    centers = np.full(len(xs), np.nan, dtype=np.float64)
    for index, x in enumerate(xs):
        y_values = np.flatnonzero(mask[:, x])
        if len(y_values):
            centers[index] = float(np.median(y_values))
    present = np.isfinite(centers)
    if present.sum() < 54 or present.sum() / len(centers) < 0.94:
        raise OracleAbstention("colored trace is horizontally incomplete")
    centers = np.interp(np.arange(len(centers)), np.flatnonzero(present), centers[present])
    if len(centers) >= 5:
        centers = np.convolve(centers, np.ones(5, dtype=np.float64) / 5.0, mode="same")
        centers[:2] = centers[2]
        centers[-2:] = centers[-3]

    # Baseline touches are the bottommost centerline samples.  A high quantile
    # is robust to the stem remnant and to antialiasing without using coordinates.
    baseline = float(np.quantile(centers, 0.90))
    relief = baseline - centers
    peak = float(np.quantile(relief, 0.98))
    if peak < 7.0:
        raise OracleAbstention(f"humps are too shallow ({peak:.2f}px)")
    active = relief > max(3.1, 0.40 * peak)
    active = _bridge_short_false_gaps(active, maximum=2)
    runs = [length for length in _true_run_lengths(active) if length >= 6]
    count = len(runs)
    if count not in (2, 3):
        raise OracleAbstention(f"resolved {count} complete arch runs, expected two or three")
    if min(runs) < 7:
        raise OracleAbstention("one arch is too narrow for a stable decision")
    return count


def decision_from_image(image: Image.Image) -> str:
    """Recover ``two`` or ``three`` from the final raster alone."""

    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    channel_max = array.max(axis=2)
    channel_min = array.min(axis=2)
    # The question declares the target as the small colored stroke; all other
    # page furniture is achromatic.  This threshold accepts anti-aliased rims.
    chromatic = (channel_max - channel_min >= 43) & (channel_max >= 78)
    components = _components(chromatic)
    significant = [component for component in components if len(component) >= 10]
    if len(significant) != 1:
        raise OracleAbstention(
            f"expected one connected colored trace, found {len(significant)}"
        )
    component = significant[0]
    if len(component) < 330:
        raise OracleAbstention("colored trace has insufficient ink")
    count = _count_arches(component, chromatic.shape)
    return "two" if count == 2 else "three"
