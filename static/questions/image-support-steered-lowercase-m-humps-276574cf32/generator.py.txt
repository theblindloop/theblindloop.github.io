"""Deterministic latent sampler and renderer for a prior-conflict letter proof.

The familiar control has two humps.  Its matched counterfactual has three.
Both classes use the same canvas, stem, endpoints, bounding box, palette
distribution, position distribution, and wording.  The wave amplitude is
inversely proportional to the hump count, so the colored wave's arclength (and
therefore total ink to first order) is matched instead of growing with count.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "lowercase-m-humps-renderer-0.1.0"

_COLORS: tuple[tuple[int, int, int], ...] = (
    (17, 119, 128),   # teal
    (194, 72, 42),    # terracotta
    (91, 72, 158),    # violet
    (27, 132, 79),    # green
)
_BACKGROUND = (246, 244, 238)
_SCALE = 4


def sample_scene(seed: int) -> dict:
    """Sample one balanced deterministic scene from an integer seed."""

    rng = random.Random((int(seed) ^ 0x6D5A56A9) & 0xFFFFFFFFFFFFFFFF)
    hump_count = 2 + (int(seed) & 1)
    column = rng.randrange(1, 6)
    row = rng.randrange(1, 7)
    return {
        "width": 512,
        "height": 512,
        "background": list(_BACKGROUND),
        "hump_count": hump_count,
        # Keep the trace inside one high-resolution ablation row while moving
        # it among many absolute page locations.
        "x": 64 * column + 22 + rng.randrange(-3, 4),
        "baseline": 64 * row + 41 + rng.randrange(-2, 3),
        "span": rng.choice((78, 80, 82, 84)),
        "curve_strength": rng.choice((29.0, 30.0, 31.0, 32.0)),
        "stem_height": rng.choice((21.0, 22.0, 23.0)),
        "stroke_width": rng.choice((5, 6, 7)),
        "color_index": rng.randrange(len(_COLORS)),
        "guide_variant": rng.randrange(3),
        "guide_offset": rng.randrange(-5, 6),
    }


def _scaled_box(box: tuple[float, float, float, float], scale: int) -> tuple[int, int, int, int]:
    return tuple(int(round(value * scale)) for value in box)  # type: ignore[return-value]


def _draw_neutral_world(scene: dict, draw: ImageDraw.ImageDraw, scale: int) -> None:
    width = int(scene["width"])
    height = int(scene["height"])
    variant = int(scene["guide_variant"])
    offset = int(scene["guide_offset"])
    x = float(scene["x"])
    baseline = float(scene["baseline"])
    span = float(scene["span"])
    stem_height = float(scene["stem_height"])

    # A restrained typographer's proof sheet.  Every mark is achromatic and
    # therefore nondiagnostic to the independent chroma-only inverse arm.
    draw.rounded_rectangle(
        _scaled_box((24, 24, width - 24, height - 24), scale),
        radius=12 * scale,
        fill=(251, 250, 246),
        outline=(210, 208, 201),
        width=2 * scale,
    )
    header_y = 48 + variant * 5
    draw.rounded_rectangle(
        _scaled_box((48, header_y, 132 + 12 * variant, header_y + 8), scale),
        radius=4 * scale,
        fill=(166, 165, 160),
    )
    draw.rounded_rectangle(
        _scaled_box((48, header_y + 17, 96 + 7 * variant, header_y + 22), scale),
        radius=2 * scale,
        fill=(207, 205, 199),
    )
    for index in range(5):
        yy = 120 + index * 72 + offset
        draw.line(
            [(48 * scale, yy * scale), ((width - 48) * scale, yy * scale)],
            fill=(229, 227, 220),
            width=1 * scale,
        )
    # The local proof inset moves with the decisive stroke but contains no
    # answer-varying geometry of its own.
    draw.rounded_rectangle(
        _scaled_box((x - 17, baseline - stem_height - 16, x + span + 17, baseline + 16), scale),
        radius=8 * scale,
        fill=(255, 254, 250),
        outline=(199, 198, 192),
        width=1 * scale,
    )
    draw.line(
        [(int((x - 9) * scale), int(baseline * scale)),
         (int((x + span + 9) * scale), int(baseline * scale))],
        fill=(187, 186, 181),
        width=1 * scale,
    )
    tick = 6 + variant
    for tx in (x - 9, x + span + 9):
        draw.line(
            [(int(tx * scale), int((baseline - tick) * scale)),
             (int(tx * scale), int((baseline + tick) * scale))],
            fill=(187, 186, 181),
            width=1 * scale,
        )
    # Remote calibration chips balance page complexity without chromatic cues.
    chip_y = height - 66 - 3 * variant
    for index, shade in enumerate((181, 202, 220, 194)):
        cx = 64 + index * 27
        draw.rectangle(
            _scaled_box((cx, chip_y, cx + 17, chip_y + 9), scale),
            fill=(shade, shade, shade),
        )


def _trace_points(scene: dict, scale: int) -> list[tuple[int, int]]:
    x = float(scene["x"])
    baseline = float(scene["baseline"])
    span = float(scene["span"])
    hump_count = int(scene["hump_count"])
    curve_strength = float(scene["curve_strength"])
    amplitude = curve_strength / hump_count
    samples = max(240, int(span * 5))
    points: list[tuple[int, int]] = []
    for index in range(samples + 1):
        t = index / samples
        arch = math.sin(math.pi * hump_count * t) ** 2
        px = x + span * t
        py = baseline - amplitude * arch
        points.append((int(round(px * scale)), int(round(py * scale))))
    return points


def render(scene: dict) -> Image.Image:
    """Render the final RGB raster; no text or metadata carries the answer."""

    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(int(value) for value in scene["background"])
    x = float(scene["x"])
    baseline = float(scene["baseline"])
    stem_height = float(scene["stem_height"])
    stroke_width = int(scene["stroke_width"])
    color_index = int(scene["color_index"])
    image = Image.new("RGB", (width * _SCALE, height * _SCALE), background)
    draw = ImageDraw.Draw(image)
    _draw_neutral_world(scene, draw, _SCALE)
    color = _COLORS[color_index]
    line_width = stroke_width * _SCALE
    # The left stem fixes the vertical extent across both answer classes.  Its
    # lower endpoint is exactly the first baseline touch of the arch sequence.
    draw.line(
        [(int(round(x * _SCALE)), int(round((baseline - stem_height) * _SCALE))),
         (int(round(x * _SCALE)), int(round(baseline * _SCALE)))],
        fill=color,
        width=line_width,
    )
    draw.line(_trace_points(scene, _SCALE), fill=color, width=line_width, joint="curve")
    radius = line_width // 2
    # Round the two true endpoints without adding a marker lookup cue.
    for px, py in (
        (x * _SCALE, (baseline - stem_height) * _SCALE),
        ((x + float(scene["span"])) * _SCALE, baseline * _SCALE),
    ):
        draw.ellipse(
            (int(px - radius), int(py - radius), int(px + radius), int(py + radius)),
            fill=color,
        )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    count = int(scene["hump_count"])
    if count == 2:
        return "two"
    if count == 3:
        return "three"
    raise ValueError(f"unsupported hump count {count}")


def _resolution_terms(scene: dict) -> tuple[float, float]:
    count = int(scene["hump_count"])
    amplitude = float(scene["curve_strength"]) / count
    wavelength = float(scene["span"]) / count
    return amplitude, wavelength


def margin(scene: dict) -> float:
    """Pixel headroom before a hump becomes too shallow or narrow to resolve."""

    amplitude, wavelength = _resolution_terms(scene)
    return float(min(amplitude - 7.0, wavelength - 20.0))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.5


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # All sampled fields intentionally alter pixels; v0.4 accepts this explicit
    # injective/not-applicable declaration after its rendered-field audit.
    return []
