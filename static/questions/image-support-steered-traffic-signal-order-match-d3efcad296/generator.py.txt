"""Deterministic two-signal renderer and analytic arm."""

from __future__ import annotations

import itertools
import random
from copy import deepcopy

from PIL import Image, ImageDraw

COLORS = {
    "red": (222, 52, 58),
    "amber": (245, 174, 35),
    "green": (43, 166, 88),
}
ORDERS = tuple(itertools.permutations(("red", "amber", "green")))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x6A09E667)
    left_order = list(ORDERS[rng.randrange(len(ORDERS))])
    same = int(seed) % 2 == 0
    if same:
        right_order = list(left_order)
    else:
        alternatives = [order for order in ORDERS if list(order) != left_order]
        right_order = list(alternatives[rng.randrange(len(alternatives))])
    return {
        "width": 768,
        "height": 432,
        "left_order": left_order,
        "right_order": right_order,
        "left_x": rng.randint(103, 143),
        "right_x": rng.randint(625, 665),
        "top_y": rng.randint(47, 67),
        "lamp_radius": rng.randint(27, 35),
        "slot_gap": rng.randint(20, 32),
        "housing_width": rng.randint(112, 132),
        "post_height": rng.randint(48, 70),
        "decoration_phase": rng.randrange(8),
    }


def _scaled_box(box: tuple[float, float, float, float], scale: int) -> tuple[int, int, int, int]:
    return tuple(int(round(value * scale)) for value in box)


def _draw_signal(draw: ImageDraw.ImageDraw, scene: dict, center_x: int, order: list[str], scale: int) -> None:
    radius = int(scene["lamp_radius"])
    slot_step = 2 * radius + int(scene["slot_gap"])
    top_y = int(scene["top_y"])
    housing_width = int(scene["housing_width"])
    padding_y = 16
    housing_top = top_y - padding_y
    housing_bottom = top_y + 2 * slot_step + 2 * radius + padding_y
    left = center_x - housing_width // 2
    right = center_x + housing_width // 2

    shadow = _scaled_box((left + 7, housing_top + 8, right + 7, housing_bottom + 8), scale)
    draw.rounded_rectangle(shadow, radius=15 * scale, fill=(149, 151, 153))
    housing = _scaled_box((left, housing_top, right, housing_bottom), scale)
    draw.rounded_rectangle(
        housing,
        radius=15 * scale,
        fill=(48, 53, 58),
        outline=(20, 24, 28),
        width=4 * scale,
    )

    for slot, color_name in enumerate(order):
        cy = top_y + radius + slot * slot_step
        hood = _scaled_box(
            (center_x - radius - 10, cy - radius - 9, center_x + radius + 10, cy + 7),
            scale,
        )
        draw.ellipse(hood, fill=(24, 28, 32))
        circle = _scaled_box(
            (center_x - radius, cy - radius, center_x + radius, cy + radius), scale
        )
        draw.ellipse(circle, fill=COLORS[color_name], outline=(11, 14, 17), width=3 * scale)

    post_top = housing_bottom
    post_bottom = min(int(scene["height"]) - 25, post_top + int(scene["post_height"]))
    draw.rectangle(
        _scaled_box((center_x - 10, post_top, center_x + 10, post_bottom), scale),
        fill=(65, 69, 73),
    )
    draw.rounded_rectangle(
        _scaled_box((center_x - 38, post_bottom - 7, center_x + 38, post_bottom + 8), scale),
        radius=6 * scale,
        fill=(65, 69, 73),
    )


def render(scene: dict) -> Image.Image:
    scale = 3
    width = int(scene["width"])
    height = int(scene["height"])
    canvas = Image.new("RGB", (width * scale, height * scale), (239, 242, 244))
    draw = ImageDraw.Draw(canvas)

    # Pale road-line dashes provide answer-neutral parity-controlled decoration.
    phase = int(scene["decoration_phase"]) % 2
    for index, x in enumerate(range(28, width - 28, 70)):
        if index % 2 == phase:
            draw.rounded_rectangle(
                _scaled_box((x, height - 17, x + 38, height - 11), scale),
                radius=3 * scale,
                fill=(205, 211, 215),
            )

    _draw_signal(draw, scene, int(scene["left_x"]), list(scene["left_order"]), scale)
    _draw_signal(draw, scene, int(scene["right_x"]), list(scene["right_order"]), scale)
    return canvas.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "yes" if tuple(scene["left_order"]) == tuple(scene["right_order"]) else "no"


def margin(scene: dict) -> float:
    radius = float(scene["lamp_radius"])
    gap = float(scene["slot_gap"])
    horizontal_separation = float(scene["right_x"] - scene["left_x"] - scene["housing_width"])
    return min(2.0 * radius, gap, horizontal_separation)


def is_quarantined(scene: dict) -> bool:
    return float(scene["lamp_radius"]) < 22.0 or float(scene["slot_gap"]) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["decoration_phase"] = int(scene["decoration_phase"]) + 2
    return [("decoration_phase_plus_two", twin)]
