"""Independent pixel-only inverse arm for the two traffic-signal world."""

from __future__ import annotations

import numpy as np
from PIL import Image

PALETTE = {
    "red": np.asarray((222, 52, 58), dtype=np.int16),
    "amber": np.asarray((245, 174, 35), dtype=np.int16),
    "green": np.asarray((43, 166, 88), dtype=np.int16),
}
MAX_COLOR_DISTANCE = 38.0
# The smallest sampled full lamp contributes at least 1,786 in-palette pixels
# across a 300-scene calibration sweep. Requiring 1,500 preserves >250 px of
# headroom while making partial evidence erasure explicitly abstaining.
MIN_HALF_AREA = 1500


def _lamp_centers(image: Image.Image) -> dict[str, tuple[tuple[float, float], tuple[float, float]]]:
    array = np.asarray(image.convert("RGB"), dtype=np.int32)
    height, width = array.shape[:2]
    recovered = {}
    for name, color in PALETTE.items():
        delta = array - color.astype(np.int32)
        distance = np.sqrt(np.sum(delta * delta, axis=2))
        ys, xs = np.nonzero(distance <= MAX_COLOR_DISTANCE)
        if len(xs) < 2 * MIN_HALF_AREA:
            raise ValueError(f"insufficient {name} lamp pixels")
        left = xs < width / 2.0
        right = ~left
        if int(left.sum()) < MIN_HALF_AREA or int(right.sum()) < MIN_HALF_AREA:
            raise ValueError(f"missing {name} lamp on one side")
        left_center = (float(xs[left].mean()), float(ys[left].mean()))
        right_center = (float(xs[right].mean()), float(ys[right].mean()))
        if left_center[0] >= width * 0.42 or right_center[0] <= width * 0.58:
            raise ValueError("signal heads are not visibly separated")
        if not (height * 0.05 < left_center[1] < height * 0.82):
            raise ValueError("left lamp lies outside the signal region")
        if not (height * 0.05 < right_center[1] < height * 0.82):
            raise ValueError("right lamp lies outside the signal region")
        recovered[name] = (left_center, right_center)
    return recovered


def decision_from_image(image: Image.Image) -> str:
    centers = _lamp_centers(image)
    left = tuple(sorted(PALETTE, key=lambda name: centers[name][0][1]))
    right = tuple(sorted(PALETTE, key=lambda name: centers[name][1][1]))
    left_y = sorted(centers[name][0][1] for name in PALETTE)
    right_y = sorted(centers[name][1][1] for name in PALETTE)
    if min(np.diff(left_y)) < 35.0 or min(np.diff(right_y)) < 35.0:
        raise ValueError("vertical lamp ordering is ambiguous")
    return "yes" if left == right else "no"
