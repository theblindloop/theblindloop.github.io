"""Deterministic raster world for counting the visible sides of a sign."""

from __future__ import annotations

import math
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

RENDERER_VERSION = "stop-sign-side-count-renderer-0.1.0"
WIDTH = 640
HEIGHT = 640
BACKGROUND = "white"
SIDE_COUNTS = tuple(range(5, 11))
BODY_COLORS = ("#d94747", "#3277b6", "#36935f", "#c58a22", "#8153a5")
BORDER_COLOR = "#20242b"
TEXT_COLOR = "white"
FONT_PATHS = (
    "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf",
    "/usr/share/fonts/dejavu/DejaVuSans-Bold.ttf",
)


def _polygon_vertices(
    side_count: int, center_x: float, center_y: float, radius: float, rotation: float
) -> list[list[float]]:
    return [
        [
            center_x + radius * math.cos(rotation + 2.0 * math.pi * index / side_count),
            center_y + radius * math.sin(rotation + 2.0 * math.pi * index / side_count),
        ]
        for index in range(side_count)
    ]


def sample_scene(seed: int) -> dict:
    """Sample one balanced, fully visible sign scene from an integer seed."""
    rng = random.Random(int(seed))
    side_count = SIDE_COUNTS[int(seed) % len(SIDE_COUNTS)]
    # Compensate the regular-polygon area coefficient before an independent scale
    # draw, so side count does not trivially predict total ink.
    target_area = 160_000.0
    coefficient = side_count * math.sin(2.0 * math.pi / side_count) / 2.0
    radius = math.sqrt(target_area / coefficient) * rng.uniform(0.92, 1.08)
    if int(seed) % 31 == 0:
        # A small but still raster-answerable boundary case for the gallery.
        radius *= 0.42
    center_x = WIDTH / 2.0 + rng.uniform(-42.0, 42.0)
    center_y = HEIGHT / 2.0 + rng.uniform(-42.0, 42.0)
    rotation = rng.uniform(0.0, 2.0 * math.pi)
    body_color = BODY_COLORS[(int(seed) * 17 + 11) % len(BODY_COLORS)]
    label_size = max(18, int(radius * rng.uniform(0.27, 0.33)))
    border_width = 7 + (int(seed) % 3)
    vertices = _polygon_vertices(side_count, center_x, center_y, radius, rotation)
    return {
        "side_count": side_count,
        "vertices": vertices,
        "center_x": round(center_x, 6),
        "center_y": round(center_y, 6),
        "radius": round(radius, 6),
        "rotation": round(rotation, 12),
        "body_color": body_color,
        "border_color": BORDER_COLOR,
        "border_width": border_width,
        "label_size": label_size,
        "label_text": "STOP",
        "text_color": TEXT_COLOR,
        "width": WIDTH,
        "height": HEIGHT,
        "background": BACKGROUND,
        "antialias": True,
    }


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    for path in FONT_PATHS:
        if Path(path).is_file():
            return ImageFont.truetype(path, size=size)
    return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    """Render only from the serialized scene fields; no hidden answer styling."""
    side_count = int(scene["side_count"])
    vertices = scene["vertices"]
    if len(vertices) != side_count:
        raise ValueError("vertex count and side_count disagree")
    width = int(scene["width"])
    height = int(scene["height"])
    supersample = 4 if bool(scene["antialias"]) else 1
    image = Image.new(
        "RGB", (width * supersample, height * supersample), scene["background"]
    )
    draw = ImageDraw.Draw(image)
    center = (float(scene["center_x"]), float(scene["center_y"]))
    radius = float(scene["radius"])
    border = max(1.0, float(scene["border_width"]))
    outer_scale = (radius + border / 2.0) / radius
    inner_scale = max(1.0, radius - border / 2.0) / radius
    outer = [
        ((center[0] + (float(x) - center[0]) * outer_scale) * supersample,
         (center[1] + (float(y) - center[1]) * outer_scale) * supersample)
        for x, y in vertices
    ]
    inner = [
        ((center[0] + (float(x) - center[0]) * inner_scale) * supersample,
         (center[1] + (float(y) - center[1]) * inner_scale) * supersample)
        for x, y in vertices
    ]
    draw.polygon(outer, fill=scene["border_color"])
    draw.polygon(inner, fill=scene["body_color"])
    font = _font(max(8, int(scene["label_size"]) * supersample))
    label = str(scene["label_text"])
    cx = float(scene["center_x"]) * supersample
    cy = float(scene["center_y"]) * supersample
    box = draw.textbbox((0, 0), label, font=font)
    tx = cx - (box[2] - box[0]) / 2.0 - box[0]
    ty = cy - (box[3] - box[1]) / 2.0 - box[1]
    draw.text((tx, ty), label, fill=scene["text_color"], font=font)
    if supersample > 1:
        image = image.resize((width, height), Image.Resampling.LANCZOS)
    return image


def analytic_gold(scene: dict) -> str:
    return str(int(scene["side_count"]))


def _edge_lengths(scene: dict) -> list[float]:
    points = [(float(x), float(y)) for x, y in scene["vertices"]]
    return [
        math.dist(points[index], points[(index + 1) % len(points)])
        for index in range(len(points))
    ]


def margin(scene: dict) -> float:
    """Positive clearance of the shortest visible edge and canvas border."""
    vertices = [(float(x), float(y)) for x, y in scene["vertices"]]
    shortest_edge = min(_edge_lengths(scene))
    border_clearance = min(
        min(x, float(scene["width"]) - x, y, float(scene["height"]) - y)
        for x, y in vertices
    )
    return float(min(shortest_edge - 18.0, border_clearance - 12.0))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Cyclic list order is a nontrivial pixel-identical latent representation alias."""
    vertices = list(scene["vertices"])
    twin = dict(scene)
    twin["vertices"] = vertices[1:] + vertices[:1]
    return [("cyclic_vertex_order", twin)]
