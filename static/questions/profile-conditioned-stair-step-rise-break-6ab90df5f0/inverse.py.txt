"""Pixel-only inverse arm for `stair_step_rise_break`.

Receives a PNG and nothing else: no latent scene, no renderer, no prompts, no
analytic gold.  It segments the one large saturated body on the page, reads the
page height of its top edge column by column, groups those columns into the
level treads of the flight, and reports whether every tread is drawn higher than
the tread before it.  It abstains whenever the flight is not intact, the treads
are not well formed, or two neighbouring treads sit within a few pixels of the
painted boundary of the rule.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "stair_step_rise_break-oracle-0.1.0"

ABSTAIN = "abstain"

CHROMA_MIN = 60          # saturated ink versus the colourless page furniture
MIN_COLUMN_INK = 20      # a tread column is far taller than any loose mark
MIN_TREAD_PX = 12        # a tread narrower than this is not a tread
TREAD_LO, TREAD_HI = 0.62, 1.45   # admissible spread of tread widths
MIN_STEP_PX = 8.0        # decision floor: closer than this and we abstain
MIN_TREADS, MAX_TREADS = 4, 9


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    chroma = arr.max(axis=-1) - arr.min(axis=-1)
    mask = chroma > CHROMA_MIN
    if not mask.any():
        return ABSTAIN

    counts = mask.sum(axis=0)
    cols = np.flatnonzero(counts >= MIN_COLUMN_INK)
    if cols.size < MIN_TREADS * MIN_TREAD_PX:
        return ABSTAIN
    # The flight must be one unbroken run of columns.
    if cols[-1] - cols[0] + 1 != cols.size:
        return ABSTAIN

    tops = np.zeros(cols.size, dtype=np.int64)
    bots = np.zeros(cols.size, dtype=np.int64)
    for i, x in enumerate(cols):
        ys = np.flatnonzero(mask[:, x])
        # each tread column is one solid vertical run down to the ground
        if ys[-1] - ys[0] + 1 != ys.size:
            return ABSTAIN
        tops[i] = ys[0]
        bots[i] = ys[-1]
    if int(bots.max() - bots.min()) > 3:
        return ABSTAIN

    # Group the columns into level treads.
    treads: list[list[int]] = []
    start = 0
    for i in range(1, cols.size + 1):
        if i == cols.size or abs(int(tops[i]) - int(tops[start])) > 1:
            treads.append([start, i - 1])
            start = i
    widths = [b - a + 1 for a, b in treads]
    if not (MIN_TREADS <= len(treads) <= MAX_TREADS):
        return ABSTAIN
    if min(widths) < MIN_TREAD_PX:
        return ABSTAIN
    med = float(np.median(widths))
    if med <= 0 or min(widths) < TREAD_LO * med or max(widths) > TREAD_HI * med:
        return ABSTAIN

    levels = [float(np.median(tops[a:b + 1])) for a, b in treads]
    rises = [levels[i] - levels[i + 1] for i in range(len(levels) - 1)]  # up is positive
    if min(abs(r) for r in rises) < MIN_STEP_PX:
        return ABSTAIN
    return "yes" if all(r > 0 for r in rises) else "no"
