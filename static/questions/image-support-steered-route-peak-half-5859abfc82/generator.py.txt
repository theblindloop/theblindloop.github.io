"""Renderer and analytic arm for peak position along a measured route."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

CANVAS = (640, 420)
BACKGROUND = (246, 247, 244)
CARD = (255, 255, 255)
BORDER = (216, 220, 217)
CYAN = (16, 166, 181)
CORAL = (226, 82, 68)
BLUE = (49, 103, 198)
INK = (48, 57, 63)


def _lengths(points: list[list[int]]) -> tuple[list[float], float]:
    cumulative = [0.0]
    for left, right in zip(points, points[1:]):
        cumulative.append(cumulative[-1] + math.dist(left, right))
    return cumulative, cumulative[-1]


def _raw_decision(points: list[list[int]]) -> tuple[str, float]:
    cumulative, total = _lengths(points)
    peak_index = min(range(len(points)), key=lambda index: points[index][1])
    signed = total - 2.0 * cumulative[peak_index]
    return ("before" if signed > 0 else "after"), abs(signed)


def sample_scene(seed: int) -> dict:
    """Sample a balanced full-width x-monotone route with one central peak."""
    rng = random.Random(int(seed))
    wanted = "before" if int(seed) % 2 == 0 else "after"
    for _ in range(5000):
        # Nine ordered vertices span the card. The peak stays at the middle
        # vertex, so horizontal peak position cannot reveal path-length half.
        raw_gaps = [rng.randint(52, 90) for _ in range(8)]
        scale = 542.0 / sum(raw_gaps)
        xs = [49]
        for gap in raw_gaps:
            xs.append(xs[-1] + max(43, round(gap * scale)))
        # Correct rounding drift without moving the central peak to a fixed x.
        drift = 591 - xs[-1]
        for index in range(1, len(xs)):
            xs[index] += round(drift * index / 8)

        ys = [rng.randint(178, 242) for _ in range(9)]
        ys[4] = rng.randint(67, 84)
        # One pronounced detour lengthens the required side, while a smaller
        # opposite-side excursion prevents a flat-tail cue. This keeps the
        # route's grid support thin instead of filling its bounding box.
        long_index = 6 if wanted == "before" else 2
        short_index = 2 if wanted == "before" else 6
        ys[long_index] = rng.randint(326, 348)
        ys[short_index] = rng.randint(242, 270)
        if min(ys[:4] + ys[5:]) - ys[4] < 58:
            continue
        points = [[int(x), int(y)] for x, y in zip(xs, ys)]
        decision, decision_margin = _raw_decision(points)
        if decision != wanted or decision_margin < 86:
            continue

        # Reflection and translation change start side and absolute placement
        # without changing the directed path-length decision.
        if rng.random() < 0.5:
            points = [[640 - x, y] for x, y in points]
        dy = rng.randint(-10, 10)
        points = [[x, y + dy] for x, y in points]
        return {"points": points}
    raise RuntimeError(f"could not sample a robust route for seed {seed}")


def analytic_gold(scene: dict) -> str:
    return _raw_decision(scene["points"])[0]


def margin(scene: dict) -> float:
    # Runner-up margin in pixels: twice the distance between the peak's
    # cumulative position and exact route halfway.
    return round(_raw_decision(scene["points"])[1], 6)


def is_quarantined(scene: dict) -> bool:
    points = scene["points"]
    ys = sorted(point[1] for point in points)
    return ys[1] - ys[0] < 48 or margin(scene) < 76


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    points = [tuple(point) for point in scene["points"]]
    image = Image.new("RGB", CANVAS, BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((20, 20, 620, 400), radius=22, fill=CARD, outline=BORDER, width=2)
    draw.text((40, 35), "CUMULATIVE ROUTE MEASUREMENT", fill=INK)
    draw.line(points, fill=CYAN, width=9, joint="curve")
    for x, y in points[1:-1]:
        draw.ellipse((x - 4, y - 4, x + 4, y + 4), fill=CYAN)

    for (x, y), color in ((points[0], CORAL), (points[-1], BLUE)):
        draw.ellipse((x - 12, y - 12, x + 12, y + 12), fill=CARD)
        draw.ellipse((x - 9, y - 9, x + 9, y + 9), fill=color)
    return image
