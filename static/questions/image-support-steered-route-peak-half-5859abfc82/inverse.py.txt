"""Independent pixel-only inverse for peak position along a route."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _mask_near(rgb: np.ndarray, target: tuple[int, int, int], tolerance: int = 14) -> np.ndarray:
    delta = np.abs(rgb.astype(np.int16) - np.asarray(target, dtype=np.int16))
    return np.max(delta, axis=2) <= tolerance


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[list[tuple[int, int]]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        group: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            group.append((cy, cx))
            for ny in range(max(0, cy - 1), min(height, cy + 2)):
                for nx in range(max(0, cx - 1), min(width, cx + 2)):
                    if mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        groups.append(group)
    return groups


def _centroid(mask: np.ndarray) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 100:
        raise ValueError("missing endpoint")
    return float(xs.mean()), float(ys.mean())


def _centerline(route: np.ndarray, forward: bool) -> list[tuple[float, float]]:
    ys, xs = np.nonzero(route)
    if not len(xs):
        return []
    samples: list[tuple[float, float]] = []
    for x in range(int(xs.min()), int(xs.max()) + 1):
        column_y = np.flatnonzero(route[:, x])
        if len(column_y):
            samples.append((float(x), float(np.median(column_y))))
    if not forward:
        samples.reverse()
    return samples


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"))
    route = _mask_near(rgb, (16, 166, 181))
    coral = _mask_near(rgb, (226, 82, 68))
    blue = _mask_near(rgb, (49, 103, 198))
    if int(route.sum()) < 3200:
        return "abstain"
    substantial = [group for group in _components(route) if len(group) >= 40]
    if len(substantial) != 1 or len(substantial[0]) < int(route.sum()) * 0.985:
        return "abstain"
    try:
        start_x, _ = _centroid(coral)
        finish_x, _ = _centroid(blue)
    except ValueError:
        return "abstain"
    if abs(start_x - finish_x) < 420:
        return "abstain"

    samples = _centerline(route, forward=start_x < finish_x)
    if len(samples) < 500:
        return "abstain"
    cumulative = [0.0]
    for left, right in zip(samples, samples[1:]):
        cumulative.append(cumulative[-1] + float(np.hypot(right[0] - left[0], right[1] - left[1])))
    peak_y = min(point[1] for point in samples)
    peak_indices = [index for index, point in enumerate(samples) if point[1] <= peak_y + 1.0]
    peak_index = peak_indices[len(peak_indices) // 2]
    signed = cumulative[-1] - 2.0 * cumulative[peak_index]
    if abs(signed) < 35:
        return "abstain"
    return "before" if signed > 0 else "after"
