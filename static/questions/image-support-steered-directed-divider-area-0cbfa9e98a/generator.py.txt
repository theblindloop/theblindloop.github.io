"""Deterministic raster world for directed chamber partitions."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 512
CHAMBER = (78, 72, 434, 440)
YS = (79, 137, 198, 259, 320, 381, 433)
NAVY = (36, 55, 82)
CREAM = (247, 236, 201)
GREEN = (39, 174, 96)
RED = (222, 74, 74)
BACKGROUND = (244, 247, 250)


def _signed_area_difference(scene: dict) -> float:
    xs = scene["path_xs"]
    # Screen-left floor minus screen-right floor for the monotone divider.
    total = 0.0
    mid_x = (CHAMBER[0] + CHAMBER[2]) / 2.0
    for index in range(len(YS) - 1):
        mean_x = (xs[index] + xs[index + 1]) / 2.0
        total += 2.0 * (mean_x - mid_x) * (YS[index + 1] - YS[index])
    return total


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    desired = 1 if int(seed) % 2 == 0 else -1
    for _ in range(1000):
        endpoints = [256 + rng.randint(-24, 24), 256 + rng.randint(-24, 24)]
        interior = [
            256 + desired * 52 + rng.randint(-68, 68)
            for _ in range(len(YS) - 2)
        ]
        xs = [endpoints[0], *interior, endpoints[1]]
        if min(xs) < 154 or max(xs) > 358:
            continue
        scene = {"path_xs": xs}
        signed = _signed_area_difference(scene)
        if signed * desired > 12000:
            return scene
    raise RuntimeError("could not sample a separated chamber partition")


def render(scene: dict) -> Image.Image:
    xs = scene["path_xs"]
    if len(xs) != len(YS):
        raise ValueError("path_xs has the wrong length")
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle(CHAMBER, radius=24, fill=CREAM, outline=NAVY, width=12)
    points = list(zip(xs, YS))
    draw.line(points, fill=NAVY, width=17, joint="curve")
    # Endpoint rings cover the divider but do not encode the answer.
    for (x, y), color in ((points[0], GREEN), (points[-1], RED)):
        draw.ellipse((x - 13, y - 13, x + 13, y + 13), fill=color, outline=(255, 255, 255), width=4)
    # A small arrow inside the divider reinforces its green-to-red direction.
    k = 3
    x0, y0 = points[k]
    x1, y1 = points[k + 1]
    dx, dy = x1 - x0, y1 - y0
    length = max(1.0, (dx * dx + dy * dy) ** 0.5)
    ux, uy = dx / length, dy / length
    px, py = -uy, ux
    cx, cy = x0 + 0.35 * dx, y0 + 0.35 * dy
    tip = (cx + 9 * ux, cy + 9 * uy)
    left = (cx - 7 * ux + 6 * px, cy - 7 * uy + 6 * py)
    right = (cx - 7 * ux - 6 * px, cy - 7 * uy - 6 * py)
    draw.polygon((tip, left, right), fill=(255, 255, 255))
    return image


def analytic_gold(scene: dict) -> str:
    # Green is always the upper endpoint, so travel-right is screen-left.
    return "right" if _signed_area_difference(scene) > 0 else "left"


def margin(scene: dict) -> float:
    return abs(_signed_area_difference(scene))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8000.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
