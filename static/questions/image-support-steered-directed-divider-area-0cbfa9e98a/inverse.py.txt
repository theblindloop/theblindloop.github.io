"""Independent pixel-only inverse for the directed divider chamber."""

from __future__ import annotations

import numpy as np
from PIL import Image

CREAM = np.array((247, 236, 201), dtype=np.int16)
GREEN = np.array((39, 174, 96), dtype=np.int16)
RED = np.array((222, 74, 74), dtype=np.int16)


def _mask_near(pixels: np.ndarray, color: np.ndarray, tolerance: int = 8) -> np.ndarray:
    delta = pixels.astype(np.int16) - color
    return np.max(np.abs(delta), axis=2) <= tolerance


def _centroid(mask: np.ndarray) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 80:
        raise ValueError("endpoint marker is missing")
    return float(xs.mean()), float(ys.mean())


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"))
    if pixels.ndim != 3 or min(pixels.shape[:2]) < 64:
        return "abstain"

    cream = _mask_near(pixels, CREAM)
    green = _mask_near(pixels, GREEN)
    red = _mask_near(pixels, RED)
    try:
        gx, gy = _centroid(green)
        rx, ry = _centroid(red)
    except ValueError:
        return "abstain"
    if abs(ry - gy) < pixels.shape[0] * 0.35:
        return "abstain"

    ys_all, xs_all = np.nonzero(cream)
    if len(xs_all) < 50000:
        return "abstain"
    x_min, x_max = int(xs_all.min()), int(xs_all.max())
    y_min, y_max = int(ys_all.min()), int(ys_all.max())
    box_area = (x_max - x_min + 1) * (y_max - y_min + 1)
    # A missing 1/8-scale chamber tile drops this well below the clean range;
    # the lower clean bound accommodates longer zig-zag dividers.
    if box_area <= 0 or len(xs_all) / box_area < 0.900:
        return "abstain"

    screen_left = 0
    screen_right = 0
    valid_rows = 0
    lo_y = max(y_min, int(min(gy, ry)) + 15)
    hi_y = min(y_max, int(max(gy, ry)) - 15)
    for y in range(lo_y, hi_y + 1):
        xs = np.flatnonzero(cream[y, x_min : x_max + 1]) + x_min
        if len(xs) < 120:
            continue
        gaps = np.diff(xs)
        if len(gaps) == 0:
            continue
        split = int(np.argmax(gaps))
        if gaps[split] < 7:
            continue
        left_count = split + 1
        right_count = len(xs) - left_count
        if min(left_count, right_count) < 35:
            continue
        screen_left += left_count
        screen_right += right_count
        valid_rows += 1
    if valid_rows < max(180, int(0.65 * max(1, hi_y - lo_y + 1))):
        return "abstain"

    difference = screen_left - screen_right
    if abs(difference) < 900:
        return "abstain"
    # For downward travel, screen-left is the travel-right side; invert if upward.
    if gy < ry:
        return "right" if difference > 0 else "left"
    return "left" if difference > 0 else "right"
