"""Analytic sampler and Pillow renderer for a compact erosion reveal."""

from __future__ import annotations

import random
from typing import Iterable

import numpy as np
from PIL import Image, ImageDraw

CANVAS = 512
N = 11
SYMBOLS = ("X", "plus", "ring", "U")
TOTAL_DARK = 58

_BASE = {
    "X": ("10001", "01010", "00100", "01010", "10001"),
    "plus": ("00100", "00100", "11111", "00100", "00100"),
    "ring": ("01110", "10001", "10001", "10001", "01110"),
    "U": ("10001", "10001", "10001", "10001", "01110"),
}

_PALETTES = (
    ((35, 48, 76), (238, 229, 202), (118, 73, 160)),
    ((42, 60, 48), (236, 225, 192), (126, 76, 158)),
    ((55, 42, 72), (231, 235, 208), (111, 67, 151)),
)


def _erode(grid: np.ndarray) -> np.ndarray:
    out = np.zeros_like(grid, dtype=bool)
    for y in range(1, N - 1):
        for x in range(1, N - 1):
            out[y, x] = bool(np.all(grid[y - 1 : y + 2, x - 1 : x + 2]))
    return out


def _target(symbol: str) -> np.ndarray:
    seed = np.zeros((N, N), dtype=bool)
    for y, row in enumerate(_BASE[symbol], start=3):
        for x, value in enumerate(row, start=3):
            seed[y, x] = value == "1"
    carrier = np.zeros_like(seed)
    for y, x in np.argwhere(seed):
        carrier[y - 1 : y + 2, x - 1 : x + 2] = True
    return _erode(carrier)


_TARGETS = {symbol: _target(symbol) for symbol in SYMBOLS}


def _make_grid(symbol: str, rng: random.Random) -> np.ndarray:
    seed = np.zeros((N, N), dtype=bool)
    for y, row in enumerate(_BASE[symbol], start=3):
        for x, value in enumerate(row, start=3):
            seed[y, x] = value == "1"
    grid = np.zeros_like(seed)
    for y, x in np.argwhere(seed):
        grid[y - 1 : y + 2, x - 1 : x + 2] = True

    # Equalize total visible ink with randomly spread, strictly removable decoys.
    candidates = [(y, x) for y in range(N) for x in range(N) if not grid[y, x]]
    rng.shuffle(candidates)
    expected = _TARGETS[symbol]
    while int(grid.sum()) < TOTAL_DARK and candidates:
        y, x = candidates.pop()
        trial = grid.copy()
        trial[y, x] = True
        if np.array_equal(_erode(trial), expected):
            grid = trial
    if int(grid.sum()) != TOTAL_DARK or not np.array_equal(_erode(grid), expected):
        raise RuntimeError("could not construct balanced removable decoys")
    return grid


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    symbol = SYMBOLS[int(seed) % len(SYMBOLS)]
    cell_size = rng.choice((5, 6))
    side = N * cell_size
    origin_x = rng.randint(72, CANVAS - 72 - side)
    origin_y = rng.randint(72, CANVAS - 72 - side)
    dark, pale, frame = _PALETTES[rng.randrange(len(_PALETTES))]
    grid = _make_grid(symbol, rng)
    return {
        "symbol": symbol,
        "grid": ["".join("1" if value else "0" for value in row) for row in grid],
        "origin_x": origin_x,
        "origin_y": origin_y,
        "cell_size": cell_size,
        "dark": list(dark),
        "pale": list(pale),
        "frame": list(frame),
    }


def _grid_from_scene(scene: dict) -> np.ndarray:
    rows = scene["grid"]
    if len(rows) != N or any(len(row) != N for row in rows):
        raise ValueError("grid must be 11 by 11")
    return np.asarray([[value == "1" for value in row] for row in rows], dtype=bool)


def render(scene: dict) -> Image.Image:
    # Access every declared latent field: the rendered-field audit can see them.
    symbol = scene["symbol"]
    if symbol not in SYMBOLS:
        raise ValueError("unknown symbol")
    grid = _grid_from_scene(scene)
    ox, oy = int(scene["origin_x"]), int(scene["origin_y"])
    cell = int(scene["cell_size"])
    dark, pale, frame = tuple(scene["dark"]), tuple(scene["pale"]), tuple(scene["frame"])
    if cell not in (5, 6):
        raise ValueError("unsupported cell size")
    side = N * cell
    image = Image.new("RGB", (CANVAS, CANVAS), (250, 250, 248))
    draw = ImageDraw.Draw(image)
    fw = 4
    draw.rectangle((ox - fw, oy - fw, ox + side + fw - 1, oy + side + fw - 1), fill=frame)
    draw.rectangle((ox, oy, ox + side - 1, oy + side - 1), fill=pale)
    for y, x in np.argwhere(grid):
        x0, y0 = ox + int(x) * cell, oy + int(y) * cell
        draw.rectangle((x0, y0, x0 + cell - 1, y0 + cell - 1), fill=dark)
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def _scores(scene: dict) -> list[int]:
    observed = _erode(_grid_from_scene(scene))
    return [int(np.count_nonzero(observed != _TARGETS[name])) for name in SYMBOLS]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene))
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    scores = _scores(scene)
    ordered = sorted(scores)
    return scores[SYMBOLS.index(analytic_gold(scene))] != ordered[0] or ordered[1] - ordered[0] < 3


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
