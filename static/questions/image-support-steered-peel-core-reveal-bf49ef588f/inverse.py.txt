"""Independent pixel-only inverse: locate, erode, and template-score the tile."""

from __future__ import annotations

import numpy as np
from PIL import Image

N = 11
LABELS = ("X", "plus", "ring", "U")
_GLYPHS = {
    "X": ("10001", "01010", "00100", "01010", "10001"),
    "plus": ("00100", "00100", "11111", "00100", "00100"),
    "ring": ("01110", "10001", "10001", "10001", "01110"),
    "U": ("10001", "10001", "10001", "10001", "01110"),
}


def _peel(values: np.ndarray) -> np.ndarray:
    result = np.zeros((N, N), dtype=bool)
    for row in range(1, N - 1):
        for col in range(1, N - 1):
            result[row, col] = bool(values[row - 1 : row + 2, col - 1 : col + 2].all())
    return result


def _reference(label: str) -> np.ndarray:
    points = np.zeros((N, N), dtype=bool)
    for row, text_row in enumerate(_GLYPHS[label], start=3):
        for col, value in enumerate(text_row, start=3):
            points[row, col] = value == "1"
    thick = np.zeros_like(points)
    for row, col in np.argwhere(points):
        thick[row - 1 : row + 2, col - 1 : col + 2] = True
    return _peel(thick)


_REFERENCES = {label: _reference(label) for label in LABELS}


def _extract(image: Image.Image) -> np.ndarray:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    border = np.concatenate(
        (pixels[:3].reshape(-1, 3), pixels[-3:].reshape(-1, 3), pixels[:, :3].reshape(-1, 3), pixels[:, -3:].reshape(-1, 3)),
        axis=0,
    )
    background = np.median(border, axis=0)
    foreground = np.linalg.norm(pixels - background, axis=2) > 15.0
    ys, xs = np.where(foreground)
    if len(xs) < 100:
        raise ValueError("framed tile is missing")
    left, right, top, bottom = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    width, height = right - left + 1, bottom - top + 1
    if width != height or (width - 8) % N or (width - 8) // N not in (5, 6):
        raise ValueError("framed tile geometry is incomplete")
    cell = (width - 8) // N
    ox, oy = left + 4, top + 4
    samples = np.asarray(
        [[pixels[oy + row * cell + cell // 2, ox + col * cell + cell // 2] for col in range(N)] for row in range(N)],
        dtype=np.int16,
    )
    colors = np.unique(samples.reshape(-1, 3), axis=0)
    if len(colors) != 2:
        raise ValueError("tile contains missing or ambiguous cell colors")
    luminance = colors @ np.asarray([0.2126, 0.7152, 0.0722])
    if float(abs(luminance[1] - luminance[0])) < 80.0:
        raise ValueError("cell tones are not separable")
    dark_color = colors[int(np.argmin(luminance))]
    grid = np.all(samples == dark_color, axis=2)
    if int(grid.sum()) != 58:
        raise ValueError("balanced tile evidence is incomplete")
    return grid


def decision_details(image: Image.Image) -> tuple[str, float]:
    peeled = _peel(_extract(image))
    scores = {label: int(np.count_nonzero(peeled != target)) for label, target in _REFERENCES.items()}
    ordered = sorted(scores.items(), key=lambda item: (item[1], LABELS.index(item[0])))
    margin = float(ordered[1][1] - ordered[0][1])
    if ordered[0][1] > 1 or margin < 3:
        raise ValueError("peeled symbol is ambiguous")
    return ordered[0][0], margin


def decision_from_image(image: Image.Image) -> str:
    return decision_details(image)[0]
