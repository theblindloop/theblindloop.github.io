"""Independent raster-only inverse for the marked-edge world."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _centroid(mask: np.ndarray) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if len(xs) == 0:
        raise ValueError("missing color marker")
    return float(xs.mean()), float(ys.mean())


def _nearby(mask: np.ndarray, center: tuple[float, float], radius: float) -> list[tuple[int, int]]:
    cx, cy = center
    y0 = max(0, int(cy - radius))
    y1 = min(mask.shape[0], int(cy + radius + 1))
    x0 = max(0, int(cx - radius))
    x1 = min(mask.shape[1], int(cx + radius + 1))
    ys, xs = np.nonzero(mask[y0:y1, x0:x1])
    result = []
    limit = radius * radius
    for yy, xx in zip(ys + y0, xs + x0):
        if (float(xx) - cx) ** 2 + (float(yy) - cy) ** 2 <= limit:
            result.append((int(yy), int(xx)))
    return result


def _reachable(mask: np.ndarray, seeds: list[tuple[int, int]]) -> np.ndarray:
    visited = np.zeros(mask.shape, dtype=bool)
    queue: deque[tuple[int, int]] = deque()
    for y, x in seeds:
        if 0 <= y < mask.shape[0] and 0 <= x < mask.shape[1] and mask[y, x] and not visited[y, x]:
            visited[y, x] = True
            queue.append((y, x))
    height, width = mask.shape
    while queue:
        y, x = queue.popleft()
        for dy in (-1, 0, 1):
            for dx in (-1, 0, 1):
                if not dx and not dy:
                    continue
                ny, nx = y + dy, x + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    queue.append((ny, nx))
    return visited


def decision_from_image(image: Image.Image) -> str:
    """Return the surviving terminal color using only the final RGB pixels."""

    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = rgb[:, :, 0], rgb[:, :, 1], rgb[:, :, 2]
    red_mask = (red > 145) & (red > green * 1.45) & (red > blue * 1.25)
    blue_mask = (blue > 135) & (blue > red * 1.25) & (blue > green * 1.12)
    green_mask = (green > 105) & (green > red * 1.18) & (green > blue * 1.02)
    purple_mask = (red > 88) & (blue > 88) & (green < 150) & (red > green * 1.15) & (blue > green * 1.20)
    dark_mask = (rgb.max(axis=2) < 132) & ((rgb.max(axis=2) - rgb.min(axis=2)) < 72)

    source_center = _centroid(green_mask)
    red_center = _centroid(red_mask)
    blue_center = _centroid(blue_mask)
    marker_center = _centroid(purple_mask)

    # The ring is rendered behind the dark centerline. Removing a generous
    # disk around its pixel centroid therefore simulates closing that edge.
    cut = dark_mask.copy()
    yy, xx = np.ogrid[:cut.shape[0], :cut.shape[1]]
    cut_radius = 27.0
    cut[(xx - marker_center[0]) ** 2 + (yy - marker_center[1]) ** 2 <= cut_radius * cut_radius] = False

    source_seeds = _nearby(cut, source_center, 55.0)
    reachable = _reachable(cut, source_seeds)
    red_reachable = any(reachable[y, x] for y, x in _nearby(cut, red_center, 55.0))
    blue_reachable = any(reachable[y, x] for y, x in _nearby(cut, blue_center, 55.0))
    if red_reachable and not blue_reachable:
        return "red"
    if blue_reachable and not red_reachable:
        return "blue"
    raise ValueError("edge-cut reachability is ambiguous")
