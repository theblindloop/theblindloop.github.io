"""Deterministic renderer and analytic label for the marked-edge world."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw


WIDTH = 640
HEIGHT = 480
SCALE = 3


def _palette(palette_id: int) -> dict[str, tuple[int, int, int]]:
    palettes = (
        {
            "background": (248, 249, 246),
            "route": (34, 48, 62),
            "green": (38, 174, 104),
            "red": (220, 70, 73),
            "blue": (61, 111, 216),
            "purple": (161, 73, 181),
            "distractor": (205, 213, 217),
        },
        {
            "background": (246, 249, 251),
            "route": (41, 53, 69),
            "green": (29, 165, 113),
            "red": (222, 76, 80),
            "blue": (65, 105, 222),
            "purple": (174, 79, 164),
            "distractor": (201, 211, 219),
        },
    )
    return palettes[int(palette_id) % len(palettes)]


def _transform(point: Iterable[float], scene: dict) -> tuple[float, float]:
    x, y = float(point[0]), float(point[1])
    cx, cy = float(scene["width"]) / 2.0, float(scene["height"]) / 2.0
    dx, dy = x - cx, y - cy
    rotation = int(scene["rotation"]) % 360
    if rotation == 90:
        dx, dy = -dy, dx
    elif rotation == 180:
        dx, dy = -dx, -dy
    elif rotation == 270:
        dx, dy = dy, -dx
    tx, ty = scene["translation"]
    return (cx + dx + float(tx), cy + dy + float(ty))


def _points(scene: dict, points: Iterable[Iterable[float]]) -> list[tuple[float, float]]:
    return [_transform(point, scene) for point in points]


def sample_scene(seed: int) -> dict:
    """Sample one balanced scene from a deterministic integer seed."""

    rng = random.Random(int(seed))
    ordinal = int(seed) % 1000
    palette_id = rng.randrange(2)
    branch_flip = rng.randrange(2)
    rotation = rng.choice((0, 90, 180, 270))
    translation = [rng.randint(-18, 18), rng.randint(-14, 14)]
    source = [110.0, 240.0]
    junction = [310.0 + rng.randint(-12, 12), 240.0 + rng.randint(-10, 10)]
    upper = 1 if branch_flip == 0 else -1
    lift = rng.choice((82, 96, 110))
    bend_x = 382.0 + rng.randint(-14, 14)
    end_x = 532.0 + rng.randint(-10, 10)
    end_y = 240.0 - upper * (lift + rng.randint(18, 34))
    branch0 = [
        list(junction),
        [bend_x - 26.0, 240.0 - upper * lift],
        [bend_x + 54.0, 240.0 - upper * (lift + 12.0)],
        [end_x, end_y],
    ]
    branch1 = [
        list(junction),
        [bend_x - 26.0, 240.0 + upper * lift],
        [bend_x + 54.0, 240.0 + upper * (lift + 12.0)],
        [end_x, 480.0 - end_y],
    ]
    branch_paths = [branch0, branch1]
    marked_branch = rng.randrange(2)
    answer_color = "red" if ordinal % 2 == 0 else "blue"
    other_color = "blue" if answer_color == "red" else "red"
    terminal_colors = [None, None]
    terminal_colors[1 - marked_branch] = answer_color
    terminal_colors[marked_branch] = other_color
    marker_centers = [
        [
            (path[1][0] + path[2][0]) / 2.0,
            (path[1][1] + path[2][1]) / 2.0,
        ]
        for path in branch_paths
    ]
    distractors = [
        [[82.0, 82.0], [190.0, 62.0], [252.0, 88.0]],
        [[410.0, 420.0], [470.0, 398.0], [580.0, 418.0]],
    ]
    marker_clearance = float(min(lift - 24, 72))
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(_palette(palette_id)["background"]),
        "route_color": list(_palette(palette_id)["route"]),
        "source_color": "green",
        "terminal_colors": terminal_colors,
        "marker_color": "purple",
        "source": source,
        "junction": junction,
        "branch_paths": branch_paths,
        "marked_branch": marked_branch,
        "marker_centers": marker_centers,
        "marker_clearance": marker_clearance,
        "translation": translation,
        "rotation": rotation,
        "stroke_width": 9,
        "terminal_radius": 23,
        "marker_radius": 23,
        "palette_id": palette_id,
        "distractors": distractors,
    }


def render(scene: dict) -> Image.Image:
    palette = _palette(scene["palette_id"])
    scale = SCALE
    image = Image.new(
        "RGB",
        (int(scene["width"]) * scale, int(scene["height"]) * scale),
        palette["background"],
    )
    draw = ImageDraw.Draw(image)

    def scaled(point: tuple[float, float]) -> tuple[int, int]:
        return (round(point[0] * scale), round(point[1] * scale))

    def line(points: list[tuple[float, float]], fill, width: int) -> None:
        draw.line([scaled(point) for point in points], fill=fill, width=width * scale, joint="curve")

    distractor_color = palette["distractor"]
    for path in scene["distractors"]:
        line(_points(scene, path), distractor_color, 4)

    marker_center = scene["marker_centers"][int(scene["marked_branch"])]
    marker = scaled(_transform(marker_center, scene))
    marker_radius = int(scene["marker_radius"]) * scale
    marker_width = 5 * scale
    draw.ellipse(
        (
            marker[0] - marker_radius,
            marker[1] - marker_radius,
            marker[0] + marker_radius,
            marker[1] + marker_radius,
        ),
        outline=palette["purple"],
        width=marker_width,
    )

    route_color = tuple(scene["route_color"])
    line(_points(scene, [scene["source"], scene["junction"]]), route_color, int(scene["stroke_width"]))
    for path in scene["branch_paths"]:
        line(_points(scene, path), route_color, int(scene["stroke_width"]))

    source = scaled(_transform(scene["source"], scene))
    source_radius = int(scene["terminal_radius"]) * scale
    draw.ellipse(
        (source[0] - source_radius, source[1] - source_radius, source[0] + source_radius, source[1] + source_radius),
        fill=palette["green"],
        outline=(255, 255, 255),
        width=2 * scale,
    )
    for index, path in enumerate(scene["branch_paths"]):
        terminal = scaled(_transform(path[-1], scene))
        color = palette[str(scene["terminal_colors"][index])]
        draw.ellipse(
            (
                terminal[0] - source_radius,
                terminal[1] - source_radius,
                terminal[0] + source_radius,
                terminal[1] + source_radius,
            ),
            fill=color,
            outline=(255, 255, 255),
            width=2 * scale,
        )

    return image.resize((int(scene["width"]), int(scene["height"])), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    marked = int(scene["marked_branch"])
    colors = scene["terminal_colors"]
    return str(colors[1 - marked])


def margin(scene: dict) -> float:
    return float(scene["marker_clearance"])


def is_quarantined(scene: dict) -> bool:
    return float(scene["marker_clearance"]) < 38.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
