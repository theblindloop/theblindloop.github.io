"""Deterministic renderer for the radial-cadence medallion world."""

from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image, ImageDraw

CANVAS_SIZE = 512
BACKGROUND = (246, 244, 238)
PALETTE = (
    (38, 104, 154),
    (113, 63, 158),
    (177, 77, 47),
    (37, 132, 91),
    (164, 57, 113),
)
DISTRACTOR_RGB = (154, 152, 146)


def sample_scene(seed: int) -> dict:
    """Sample balanced classes and four visible difficulty levels."""
    value = int(seed)
    rng = random.Random(value ^ 0x5A17_CADE)
    magnitude = (0.75, 1.05, 1.35, 1.65)[(value // 2) % 4]
    gap_delta = magnitude if value % 2 == 0 else -magnitude
    center_x = rng.choice((160, 352))
    center_y = rng.choice((160, 352))

    distractors: list[dict] = []
    while len(distractors) < 8:
        x = rng.randrange(28, CANVAS_SIZE - 28)
        y = rng.randrange(28, CANVAS_SIZE - 28)
        if math.hypot(x - center_x, y - center_y) < 82:
            continue
        distractors.append(
            {
                "x": x,
                "y": y,
                "kind": rng.choice(("dot", "square", "ring")),
                "size": rng.choice((3, 4, 5)),
            }
        )

    return {
        "canvas_size": CANVAS_SIZE,
        "center_x": center_x,
        "center_y": center_y,
        "medallion_radius": rng.choice((47.0, 48.0)),
        "inner_radius": rng.choice((8.5, 9.0, 9.5)),
        "mean_gap": rng.choice((7.45, 7.65, 7.85)),
        "gap_delta": gap_delta,
        "ridge_width": rng.choice((1.25, 1.4, 1.55)),
        "ridge_amplitude": rng.choice((43.0, 47.0, 51.0)),
        "base_rgb": list(rng.choice(PALETTE)),
        "background_rgb": list(BACKGROUND),
        "distractor_marks": distractors,
    }


def _ridge_radii(scene: dict) -> list[float]:
    """Five radii whose four gaps have a signed linear outward trend."""
    inner = float(scene["inner_radius"])
    mean_gap = float(scene["mean_gap"])
    delta = float(scene["gap_delta"])
    gaps = [mean_gap + coefficient * delta for coefficient in (-1.5, -0.5, 0.5, 1.5)]
    radii = [inner]
    for gap in gaps:
        radii.append(radii[-1] + gap)
    return radii


def render(scene: dict) -> Image.Image:
    canvas = int(scene["canvas_size"])
    if canvas != CANVAS_SIZE:
        raise ValueError("this world uses a fixed 512 by 512 canvas")
    background = np.asarray(scene["background_rgb"], dtype=np.float64)
    array = np.empty((canvas, canvas, 3), dtype=np.float64)
    array[:] = background

    # Remote achromatic marks are visually present but invisible to the chroma
    # segmentation used by the inverse arm.
    base_image = Image.fromarray(np.uint8(np.rint(array)), mode="RGB")
    draw = ImageDraw.Draw(base_image)
    for mark in scene["distractor_marks"]:
        x, y, size = int(mark["x"]), int(mark["y"]), int(mark["size"])
        bounds = (x - size, y - size, x + size, y + size)
        kind = str(mark["kind"])
        if kind == "dot":
            draw.ellipse(bounds, fill=DISTRACTOR_RGB)
        elif kind == "square":
            draw.rectangle(bounds, fill=DISTRACTOR_RGB)
        else:
            draw.ellipse(bounds, outline=DISTRACTOR_RGB, width=2)
    array = np.asarray(base_image, dtype=np.float64).copy()

    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    medallion_radius = float(scene["medallion_radius"])
    if min(cx, cy) - medallion_radius < 24 or max(cx, cy) + medallion_radius > canvas - 24:
        raise ValueError("medallion is too close to the canvas edge")
    yy, xx = np.mgrid[0:canvas, 0:canvas]
    distance = np.hypot(xx - cx, yy - cy)
    coverage = np.clip(medallion_radius + 0.5 - distance, 0.0, 1.0)
    inside = distance <= medallion_radius - 0.5

    width = float(scene["ridge_width"])
    ridge_field = np.zeros_like(distance)
    for radius in _ridge_radii(scene):
        ridge_field += np.exp(-0.5 * ((distance - radius) / width) ** 2)
    # The class changes ridge positions, so subtract the actual disk mean.  This
    # holds mean luminance label-neutral rather than exposing an ink proxy.
    ridge_field -= float(ridge_field[inside].mean())
    modulation = float(scene["ridge_amplitude"]) * ridge_field
    interior_rgb = np.asarray(scene["base_rgb"], dtype=np.float64)[None, None, :] + modulation[:, :, None]
    if interior_rgb[inside].min() < 0.0 or interior_rgb[inside].max() > 255.0:
        raise ValueError("sampled ridge field clips the palette")
    array = array * (1.0 - coverage[:, :, None]) + interior_rgb * coverage[:, :, None]
    return Image.fromarray(np.uint8(np.clip(np.rint(array), 0, 255)), mode="RGB")


def analytic_gold(scene: dict) -> str:
    return "grow" if float(scene["gap_delta"]) > 0.0 else "shrink"


def margin(scene: dict) -> float:
    """Runner-up score gap: outermost gap minus innermost gap, in pixels."""
    return 3.0 * abs(float(scene["gap_delta"]))


def is_quarantined(scene: dict) -> bool:
    return abs(float(scene["gap_delta"])) <= 0.65


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["distractor_marks"] = list(reversed(scene["distractor_marks"]))
    return [("reverse_same_ink_distractor_order", twin)]
