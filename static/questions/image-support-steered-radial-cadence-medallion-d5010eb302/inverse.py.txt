"""Independent PNG-only inverse for radial cadence.

The inverse locates the sole connected chromatic medallion, validates its disk
geometry, constructs an angularly pooled luminance profile, discovers five
ridge maxima, and compares the ordered radial gaps.  It imports no forward-arm
module and receives no latent scene values.
"""

from __future__ import annotations

from collections import deque
from dataclasses import dataclass

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    """The raster does not expose an intact, unambiguous public instance."""


@dataclass(frozen=True)
class Measurement:
    decision: str
    ridge_radii: tuple[float, ...]
    radial_gaps: tuple[float, ...]
    runner_up_gap_px: float
    medallion_center: tuple[float, float]
    medallion_radius: float
    disk_fill: float


def _components(mask: np.ndarray) -> list[np.ndarray]:
    work = np.asarray(mask, dtype=bool).copy()
    height, width = work.shape
    output: list[np.ndarray] = []
    for start_y, start_x in np.argwhere(work):
        sy, sx = int(start_y), int(start_x)
        if not work[sy, sx]:
            continue
        work[sy, sx] = False
        queue: deque[tuple[int, int]] = deque([(sy, sx)])
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        output.append(np.asarray(points, dtype=np.float64))
    return sorted(output, key=len, reverse=True)


def _chromatic_mask(array: np.ndarray) -> np.ndarray:
    border = np.concatenate(
        (array[:3].reshape(-1, 3), array[-3:].reshape(-1, 3), array[:, :3].reshape(-1, 3), array[:, -3:].reshape(-1, 3)),
        axis=0,
    )
    background = np.median(border.astype(np.float32), axis=0)
    delta = np.linalg.norm(array.astype(np.float32) - background, axis=2)
    chroma = array.max(axis=2).astype(np.int16) - array.min(axis=2).astype(np.int16)
    return (delta >= 44.0) & (chroma >= 42)


def _recover_disk(array: np.ndarray) -> tuple[np.ndarray, float, float, float, float]:
    mask = _chromatic_mask(array)
    components = _components(mask)
    if len(components) != 1:
        raise OracleAbstention("expected one connected chromatic medallion")
    points = components[0]
    if not 6750 <= len(points) <= 7500:
        raise OracleAbstention("chromatic area is inconsistent with an intact medallion")
    ys, xs = points[:, 0], points[:, 1]
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    width, height = x1 - x0 + 1, y1 - y0 + 1
    if not (94 <= width <= 99 and 94 <= height <= 99 and abs(width - height) <= 2):
        raise OracleAbstention("chromatic component lacks the declared compact circular extent")
    cx = (x0 + x1) / 2.0
    cy = (y0 + y1) / 2.0
    radius = (width + height) / 4.0
    expected_area = np.pi * radius * radius
    disk_fill = float(len(points) / expected_area)
    if not 0.955 <= disk_fill <= 1.035:
        raise OracleAbstention("medallion interior is damaged")
    radial = np.hypot(xs - cx, ys - cy)
    if np.percentile(radial, 99.5) - np.percentile(radial, 0.5) < 43.0:
        raise OracleAbstention("medallion does not expose a filled interior")
    return mask, cx, cy, radius, disk_fill


def _radial_profile(array: np.ndarray, mask: np.ndarray, cx: float, cy: float, radius: float) -> tuple[np.ndarray, np.ndarray]:
    yy, xx = np.mgrid[0:array.shape[0], 0:array.shape[1]]
    radial = np.hypot(xx - cx, yy - cy)
    gray = array.astype(np.float64).mean(axis=2)
    scale = 1
    bins = np.floor(radial * scale).astype(np.int32)
    valid = mask & (radial <= radius - 3.5)
    count = np.bincount(bins[valid], minlength=int(radius * scale) + 2).astype(np.float64)
    total = np.bincount(bins[valid], weights=gray[valid], minlength=len(count))
    if np.any(count[5:int((radius - 4) * scale)] < 8):
        raise OracleAbstention("radial profile has missing angular support")
    profile = np.divide(total, count, out=np.zeros_like(total), where=count > 0)
    kernel = np.asarray([1.0, 2.0, 1.0]) / 4.0
    smooth = np.convolve(profile, kernel, mode="same")
    radii = (np.arange(len(smooth), dtype=np.float64) + 0.5) / scale
    return radii, smooth


def _ridge_peaks(radii: np.ndarray, profile: np.ndarray, disk_radius: float) -> tuple[float, ...]:
    valid = (radii >= 5.5) & (radii <= disk_radius - 5.0)
    baseline = float(np.median(profile[valid]))
    candidates = [
        index
        for index in range(2, len(profile) - 2)
        if valid[index]
        and profile[index] >= profile[index - 1]
        and profile[index] > profile[index + 1]
        and profile[index] - baseline >= 12.0
    ]
    selected: list[int] = []
    for index in sorted(candidates, key=lambda item: float(profile[item]), reverse=True):
        if all(abs(index - other) >= 4 for other in selected):
            selected.append(index)
        if len(selected) == 5:
            break
    if len(selected) != 5:
        raise OracleAbstention(f"recovered {len(selected)} resolvable ridges instead of five")
    refined: list[float] = []
    for index in sorted(selected):
        local = np.arange(max(0, index - 1), min(len(profile), index + 2))
        weights = np.maximum(profile[local] - baseline, 0.0)
        if float(weights.sum()) <= 0.0:
            raise OracleAbstention("ridge peak has no positive prominence")
        refined.append(float(np.sum(radii[local] * weights) / weights.sum()))
    return tuple(refined)


def measure_image(image: Image.Image) -> Measurement:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.shape != (512, 512, 3):
        raise OracleAbstention("unsupported raster dimensions")
    mask, cx, cy, radius, disk_fill = _recover_disk(array)
    radial_axis, profile = _radial_profile(array, mask, cx, cy, radius)
    ridge_radii = _ridge_peaks(radial_axis, profile, radius)
    gaps = tuple(float(value) for value in np.diff(np.asarray(ridge_radii)))
    step_changes = np.diff(np.asarray(gaps))
    score = float(gaps[-1] - gaps[0])
    if abs(score) <= 1.6:
        raise OracleAbstention("ridge cadence is too close to the visible tie")
    if score > 0.0 and not np.all(step_changes > -0.20):
        raise OracleAbstention("ridge gaps do not form one growing cadence")
    if score < 0.0 and not np.all(step_changes < 0.20):
        raise OracleAbstention("ridge gaps do not form one shrinking cadence")
    return Measurement(
        decision="grow" if score > 0.0 else "shrink",
        ridge_radii=ridge_radii,
        radial_gaps=gaps,
        runner_up_gap_px=abs(score),
        medallion_center=(cx, cy),
        medallion_radius=radius,
        disk_fill=disk_fill,
    )


def decision_from_image(image: Image.Image) -> str:
    return measure_image(image).decision
