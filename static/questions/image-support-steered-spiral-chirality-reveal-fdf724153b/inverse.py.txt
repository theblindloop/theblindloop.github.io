"""Pixel-only inverse arm for the spiral-chirality reveal world.

Receives one rasterized PNG and nothing else: no latent scene, no renderer, no
prompt or gold code. It segments the ink, keeps the largest connected stroke,
finds the beaded tip, parameterizes the stroke by geodesic distance from that
tip, recovers the quantized heading sequence, groups the four-turn detour
signatures, reads each detour's handedness relative to travel, and classifies
the twenty-five-bit pattern.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "spiral_chirality_reveal-oracle-0.1.0"
ABSTAIN = "undecided"
STATIONS = 25
SAMPLE_STEP = 2
CLUSTER_GAP = 24
TIP_GUARD = 26

GLYPH_CELLS: dict[str, tuple[int, ...]] = {
    "tee": (0, 1, 2, 3, 4, 7, 12, 17, 22),
    "ell": (0, 5, 10, 15, 20, 21, 22, 23, 24),
    "plus": (2, 7, 10, 11, 12, 13, 14, 17, 22),
    "ex": (0, 4, 6, 8, 12, 16, 18, 20, 24),
}


def _largest_component(ink: np.ndarray) -> np.ndarray:
    height, width = ink.shape
    seen = np.zeros_like(ink)
    best = np.zeros_like(ink)
    best_size = 0
    for sy in range(height):
        row = np.flatnonzero(ink[sy] & ~seen[sy])
        for sx in row:
            if seen[sy, sx]:
                continue
            stack = [(sy, int(sx))]
            seen[sy, sx] = True
            pixels = []
            while stack:
                y, x = stack.pop()
                pixels.append((y, x))
                for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                    if 0 <= ny < height and 0 <= nx < width and ink[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
            if len(pixels) > best_size:
                best_size = len(pixels)
                best = np.zeros_like(ink)
                for y, x in pixels:
                    best[y, x] = True
    return best


def _bfs(mask: np.ndarray, start: tuple[int, int]) -> np.ndarray:
    height, width = mask.shape
    dist = np.full(mask.shape, -1, dtype=np.int32)
    dist[start] = 0
    queue = deque([start])
    while queue:
        y, x = queue.popleft()
        step = dist[y, x] + 1
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and dist[ny, nx] < 0:
                dist[ny, nx] = step
                queue.append((ny, nx))
    return dist


def _farthest(dist: np.ndarray) -> tuple[int, int]:
    flat = int(np.argmax(dist))
    return int(flat // dist.shape[1]), int(flat % dist.shape[1])


def _heading(vector: tuple[float, float]) -> tuple[int, int] | None:
    dx, dy = vector
    if abs(dx) >= abs(dy):
        if abs(dx) < 1.0:
            return None
        return (1, 0) if dx > 0 else (-1, 0)
    if abs(dy) < 1.0:
        return None
    return (0, 1) if dy > 0 else (0, -1)


def _bits_from_image(image: Image.Image) -> list[int] | None:
    arr = np.asarray(image.convert("RGB"))
    flat = arr.reshape(-1, 3)
    colors, counts = np.unique(flat, axis=0, return_counts=True)
    background = colors[int(np.argmax(counts))]
    ink = ~np.all(arr == background, axis=-1)
    if ink.sum() < 200:
        return None
    stroke = _largest_component(ink)
    seed = _farthest(_bfs(stroke, tuple(int(v) for v in np.argwhere(stroke)[0])))
    first = _bfs(stroke, seed)
    tip_a = seed
    tip_b = _farthest(first)

    def bulk(tip: tuple[int, int]) -> int:
        y, x = tip
        window = stroke[max(0, y - 8) : y + 9, max(0, x - 8) : x + 9]
        return int(window.sum())

    start = tip_a if bulk(tip_a) >= bulk(tip_b) else tip_b
    dist = _bfs(stroke, start)
    if int((dist < 0)[stroke].sum()) > 0:
        return None
    values = dist[stroke]
    span = int(values.max())
    ys, xs = np.nonzero(stroke)
    order = dist[ys, xs] // SAMPLE_STEP
    buckets = int(span // SAMPLE_STEP) + 1
    sums_x = np.bincount(order, weights=xs, minlength=buckets)
    sums_y = np.bincount(order, weights=ys, minlength=buckets)
    hits = np.bincount(order, minlength=buckets)
    if (hits[:buckets] == 0).any():
        return None
    px = sums_x / hits
    py = sums_y / hits

    headings: list[tuple[int, int] | None] = [None] * buckets
    for index in range(2, buckets - 2):
        headings[index] = _heading((px[index + 2] - px[index - 2], py[index + 2] - py[index - 2]))

    runs: list[tuple[tuple[int, int], int, int]] = []
    index = 2
    while index < buckets - 2:
        current = headings[index]
        stop = index
        while stop < buckets - 2 and headings[stop] == current:
            stop += 1
        if current is not None and (stop - index) >= 2:
            runs.append((current, index, stop - 1))
        index = stop

    merged: list[tuple[tuple[int, int], int, int]] = []
    for run in runs:
        if merged and merged[-1][0] == run[0]:
            merged[-1] = (run[0], merged[-1][1], run[2])
        else:
            merged.append(run)

    turns: list[tuple[float, int]] = []
    for before, after in zip(merged, merged[1:]):
        (dx1, dy1), (dx2, dy2) = before[0], after[0]
        cross = dx1 * dy2 - dy1 * dx2
        if cross == 0:
            return None
        position = (before[2] + after[1]) / 2.0 * SAMPLE_STEP
        if position < TIP_GUARD or position > span - TIP_GUARD:
            continue
        turns.append((position, 1 if cross > 0 else -1))

    clusters: list[list[tuple[float, int]]] = []
    for turn in turns:
        if clusters and turn[0] - clusters[-1][-1][0] <= CLUSTER_GAP:
            clusters[-1].append(turn)
        else:
            clusters.append([turn])
    detours = [cluster for cluster in clusters if len(cluster) == 4]
    if len(detours) != STATIONS or any(len(c) not in (1, 4) for c in clusters):
        return None
    # a left turn is a negative cross product in image coordinates (y downward)
    return [1 if cluster[0][1] < 0 else 0 for cluster in detours]


def decision_from_image(image: Image.Image) -> str:
    """Glyph recovered from the final pixels alone, or an abstention."""
    bits = _bits_from_image(image)
    if bits is None:
        return ABSTAIN
    scores = []
    for glyph, cells in sorted(GLYPH_CELLS.items()):
        target = [1 if i in set(cells) else 0 for i in range(STATIONS)]
        scores.append((sum(1 for a, b in zip(bits, target) if a != b), glyph))
    scores.sort()
    if scores[0][0] > 2 or scores[0][0] == scores[1][0]:
        return ABSTAIN
    return scores[0][1]


def decision_margin_from_image(image: Image.Image) -> float:
    """Runner-up separation of the pixel-only decision, in grid cells."""
    bits = _bits_from_image(image)
    if bits is None:
        return 0.0
    scores = sorted(
        sum(1 for a, b in zip(bits, [1 if i in set(cells) else 0 for i in range(STATIONS)]) if a != b)
        for cells in GLYPH_CELLS.values()
    )
    return float(scores[1] - scores[0])
