"""Enclosed-region marker selection world.

Global / figure-ground topology reveal: thick black line-work partitions the
white canvas into white regions. Exactly one of several coloured marker dots
sits inside a white region that is completely sealed off by black line on every
side (so that region never touches the picture edge); the other coloured marker
dots sit in white regions that all open onto the picture boundary. The answer is
the colour of the marker that lies in the fully enclosed white pocket, which is a
purely global, price-of-topology judgement over the whole raster.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "enclosed_region_marker-0.1.0"

CANVAS = 512
BACKGROUND = "white"
INK_COLOR = (20, 20, 20)

# Distinct saturated marker colours. The decision labels are these colour names.
MARKER_COLORS = {
    "red": (220, 40, 40),
    "green": (40, 175, 70),
    "blue": (45, 85, 225),
    "orange": (240, 145, 35),
}
MARKER_RADIUS = 8.0

WALL_THICKNESS = 18.0
BAR_THICKNESS = 16.0
RING_A_RANGE = (70.0, 95.0)
RING_B_RANGE = (70.0, 95.0)
RING_CENTER_RANGE = (165.0, 347.0)

# A marker that is this close to the seal wall is ambiguous at pixel scale.
QUARANTINE_MARGIN = 12.0


def sample_scene(seed: int) -> dict:
    """Deterministically sample a full latent scene from an integer seed."""
    rng = random.Random(seed)

    answer_color = rng.choice(sorted(MARKER_COLORS))

    cx = rng.uniform(*RING_CENTER_RANGE)
    cy = rng.uniform(*RING_CENTER_RANGE)
    a = rng.uniform(*RING_A_RANGE)
    b = rng.uniform(*RING_B_RANGE)
    wall = WALL_THICKNESS
    ring = {"cx": cx, "cy": cy, "a": a, "b": b, "wall": wall}

    bars = []
    # Two horizontal bars hugging the top and bottom margins.
    top_y = rng.uniform(16.0, 40.0)
    bars.append({"x1": rng.uniform(60.0, 90.0), "y1": top_y,
                 "x2": rng.uniform(422.0, 452.0), "y2": top_y,
                 "thickness": BAR_THICKNESS})
    bot_y = rng.uniform(472.0, 496.0)
    bars.append({"x1": rng.uniform(60.0, 90.0), "y1": bot_y,
                 "x2": rng.uniform(422.0, 452.0), "y2": bot_y,
                 "thickness": BAR_THICKNESS})
    # Two vertical bars hugging the left and right margins.
    left_x = rng.uniform(16.0, 40.0)
    bars.append({"x1": left_x, "y1": rng.uniform(90.0, 140.0),
                 "x2": left_x, "y2": rng.uniform(300.0, 360.0),
                 "thickness": BAR_THICKNESS})
    right_x = rng.uniform(472.0, 496.0)
    bars.append({"x1": right_x, "y1": rng.uniform(160.0, 220.0),
                 "x2": right_x, "y2": rng.uniform(392.0, 452.0),
                 "thickness": BAR_THICKNESS})

    def inside_ring(x: float, y: float, pad: float = 0.0) -> bool:
        dx = (x - cx) / (a - pad)
        dy = (y - cy) / (b - pad)
        return (dx * dx + dy * dy) < 1.0

    # Target marker deep inside the ring.
    angle = rng.uniform(0.0, 2.0 * math.pi)
    rad = rng.uniform(0.0, 0.4 * min(a, b))
    tx = cx + rad * math.cos(angle)
    ty = cy + rad * math.sin(angle)
    target = {"color": answer_color, "x": tx, "y": ty, "radius": MARKER_RADIUS}

    def dist_to_segment(px, py, x1, y1, x2, y2):
        vx, vy = x2 - x1, y2 - y1
        wx, wy = px - x1, py - y1
        denom = vx * vx + vy * vy
        if denom == 0:
            return math.hypot(px - x1, py - y1)
        t = max(0.0, min(1.0, (wx * vx + wy * vy) / denom))
        return math.hypot(px - (x1 + t * vx), py - (y1 + t * vy))

    # A distractor must sit well away from the seal wall and from every bar so
    # that its white region is unambiguously part of the open exterior.
    def distractor_too_close(px, py):
        if inside_ring(px, py, pad=-(wall + MARKER_RADIUS + 6.0)):
            return True
        for seg in bars:
            if dist_to_segment(px, py, seg["x1"], seg["y1"], seg["x2"], seg["y2"]) < (
                seg["thickness"] / 2.0 + MARKER_RADIUS + 3.0
            ):
                return True
        return False

    # Three distractor markers placed in the exterior white corners, always
    # well outside the ring's seal wall and off the black line-work.
    corner_boxes = [
        (30.0, 190.0, 30.0, 190.0),   # top-left
        (322.0, 482.0, 30.0, 190.0),  # top-right
        (30.0, 190.0, 322.0, 482.0),  # bottom-left
        (322.0, 482.0, 322.0, 482.0),  # bottom-right
    ]
    distractors = []
    distractor_colors = [c for c in sorted(MARKER_COLORS) if c != answer_color]
    box_indices = rng.sample(range(4), 3)
    for k, bi in enumerate(box_indices):
        xlo, xhi, ylo, yhi = corner_boxes[bi]
        found = False
        for _attempt in range(120):
            px = rng.uniform(xlo, xhi)
            py = rng.uniform(ylo, yhi)
            if not distractor_too_close(px, py):
                distractors.append(
                    {"color": distractor_colors[k], "x": px, "y": py,
                     "radius": MARKER_RADIUS}
                )
                found = True
                break
        if not found:  # pragmatic fallback: the box's absolute image corner is
            # always clear white exterior, far from the ring wall and the bars.
            corner = ((12.0, 12.0), (500.0, 12.0), (12.0, 500.0), (500.0, 500.0))[bi]
            distractors.append(
                {"color": distractor_colors[k], "x": corner[0],
                 "y": corner[1], "radius": MARKER_RADIUS}
            )

    markers = [target] + distractors

    return {
        "canvas": CANVAS,
        "background": BACKGROUND,
        "ink_color": list(INK_COLOR),
        "ring": ring,
        "bars": bars,
        "markers": markers,
        "answer_color": answer_color,
        "ring_a": round(a, 3),
        "ring_b": round(b, 3),
        "wall": round(wall, 3),
    }


def _draw_line(draw: ImageDraw.ImageDraw, seg: dict, scale: float) -> None:
    t = float(seg["thickness"]) * scale
    p1 = (float(seg["x1"]) * scale, float(seg["y1"]) * scale)
    p2 = (float(seg["x2"]) * scale, float(seg["y2"]) * scale)
    draw.line([p1, p2], fill=INK_COLOR, width=int(round(t)))


def render(scene: dict, supersample: int = 4) -> Image.Image:
    """Render the line-work scene using supersampled antialiased drawing."""
    ans = scene["answer_color"]
    if ans not in MARKER_COLORS:
        raise ValueError(f"unknown answer colour {ans!r}")
    width = int(scene["canvas"])
    ss = supersample
    img = Image.new("RGB", (width * ss, width * ss), scene["background"])
    draw = ImageDraw.Draw(img)

    ring = scene["ring"]
    cx = float(ring["cx"]) * ss
    cy = float(ring["cy"]) * ss
    a = float(ring["a"]) * ss
    b = float(ring["b"]) * ss
    wall = float(ring["wall"]) * ss
    # Draw the sealed ring as a thick closed elliptical wall.
    draw.ellipse([cx - a - wall, cy - b - wall, cx + a + wall, cy + b + wall],
                 fill=INK_COLOR)
    draw.ellipse([cx - a, cy - b, cx + a, cy + b], fill=scene["background"])

    for seg in scene["bars"]:
        _draw_line(draw, seg, ss)

    for marker in scene["markers"]:
        color = MARKER_COLORS[marker["color"]]
        r = float(marker["radius"]) * ss
        mx = float(marker["x"]) * ss
        my = float(marker["y"]) * ss
        draw.ellipse([mx - r, my - r, mx + r, my + r], fill=color)

    return img.resize((width, width), Image.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """The answer is the colour of the marker inside the sealed region."""
    return scene["answer_color"]


def _marker_clearance(scene: dict) -> float:
    ring = scene["ring"]
    cx, cy = float(ring["cx"]), float(ring["cy"])
    a, b = float(ring["a"]), float(ring["b"])
    good = 0.0
    for marker in scene["markers"]:
        if marker["color"] == scene["answer_color"]:
            mx, my = float(marker["x"]), float(marker["y"])
            t = math.sqrt(((mx - cx) / a) ** 2 + ((my - cy) / b) ** 2)
            good = (1.0 - t) * min(a, b)
            break
    return round(good, 3)


def margin(scene: dict) -> float:
    """One-sided verifiability margin: distance from the target marker to the
    inner seal wall (in pixels). The further from the wall, the clearer that the
    pocket is fully enclosed; tiny clearances are ambiguous and quarantined."""
    return _marker_clearance(scene)


def is_quarantined(scene: dict) -> bool:
    return bool(_marker_clearance(scene) < QUARANTINE_MARGIN)


def latent_symmetries(scene: dict) -> list:
    """No declared latent symmetry.

    The sealed pocket geometry and marker colouring are all directly visible;
    the enclosed-marker colour is an injective function of the rendered scene,
    and no render-invariant relabelling leaves both pixels and gold fixed.
    """
    return []
