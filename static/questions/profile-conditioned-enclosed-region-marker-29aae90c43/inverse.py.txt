"""Independent pixel-only inverse arm for the enclosed-region marker world.

Receives only the rendered PNG. It segments the dark line-work (ink) from the
light background, locates each coloured marker dot, floods through the
non-ink region from each marker to test whether that white region reaches the
picture border, and reads off the colour of the single marker whose white region
is fully sealed (never touches the border). That colour is the answer.

This module imports neither renderer, prompts, generate, verify, nor any
analytic-gold or scene code, and never receives latent scene data.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

# Pixels at least this bright in every channel are background white.
_WHITE_MIN = 200
# Pixels darker than this (max channel) are the black line-work / ink.
_INK_MAX = 70
# A marker component must occupy at least this many pixels.
_MIN_MARKER_PIXELS = 4

# Palette used to name each marker by its nearest fixed colour.
_PALETTE = {
    "red": np.array([220, 40, 40]),
    "green": np.array([40, 175, 70]),
    "blue": np.array([45, 85, 225]),
    "orange": np.array([240, 145, 35]),
}


def _marker_components(arr: np.ndarray):  # -> list[(x, y, colour)]
    """Find 4-connected coloured marker blobs and name their colour."""
    h, w, _ = arr.shape
    r = arr[:, :, 0].astype(int)
    g = arr[:, :, 1].astype(int)
    b = arr[:, :, 2].astype(int)
    # Markers are the only strongly saturated objects (black ink and white
    # background are both low-chroma). Antialiased edge greys are excluded by
    # the saturation floor, so only the coloured discs survive as blobs.
    mx = np.maximum(np.maximum(r, g), b)
    mn = np.minimum(np.minimum(r, g), b)
    marker_mask = (mx - mn) >= 60

    visited = np.zeros((h, w), dtype=bool)
    coords = set(map(tuple, np.argwhere(marker_mask)))
    components = []
    for start in coords:
        if visited[start]:
            continue
        stack = {start}
        blob = []
        while stack:
            cell = stack.pop()
            if visited[cell]:
                continue
            visited[cell] = True
            blob.append(cell)
            rr, cc = cell
            for nr, nc in ((rr + 1, cc), (rr - 1, cc), (rr, cc + 1), (rr, cc - 1)):
                if 0 <= nr < h and 0 <= nc < w and marker_mask[nr, nc] and not visited[nr, nc]:
                    stack.add((nr, nc))
        if len(blob) >= _MIN_MARKER_PIXELS:
            xs = sum(p[1] for p in blob) / len(blob)
            ys = sum(p[0] for p in blob) / len(blob)
            colour = _classify(arr[int(round(ys)), int(round(xs))])
            components.append((xs, ys, colour))
    return components


def _classify(rgb) -> str:
    rgb = np.asarray(rgb, dtype=float)
    best, best_name = None, None
    for name, pcolour in _PALETTE.items():
        dist = float(np.linalg.norm(rgb - pcolour))
        if best is None or dist < best:
            best, best_name = dist, name
    return best_name


def _reaches_border(arr: np.ndarray, start: tuple[int, int]) -> bool:
    """Flood through non-ink pixels from start; report whether the border is met."""
    h, w, _ = arr.shape
    is_ink = (np.max(arr, axis=2) < _INK_MAX)
    frontier = {start}
    seen = np.zeros((h, w), dtype=bool)
    while frontier:
        cell = frontier.pop()
        rr, cc = cell
        if rr == 0 or rr == h - 1 or cc == 0 or cc == w - 1:
            return True
        for nr, nc in ((rr + 1, cc), (rr - 1, cc), (rr, cc + 1), (rr, cc - 1)):
            if 0 <= nr < h and 0 <= nc < w and not is_ink[nr, nc] and not seen[nr, nc]:
                seen[nr, nc] = True
                frontier.add((nr, nc))
    return False


def decision_from_image(image: Image.Image) -> str:
    """Recover the enclosed-marker colour from the rasterized PNG alone."""
    arr = np.asarray(image.convert("RGB")).astype(np.int32)
    markers = _marker_components(arr)
    for (x, y, colour) in markers:
        start = (int(round(y)), int(round(x)))
        if not _reaches_border(arr, start):
            return colour
    return None if not markers else markers[0][2]
