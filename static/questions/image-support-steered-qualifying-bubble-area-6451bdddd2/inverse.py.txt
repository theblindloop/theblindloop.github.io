"""Independent image-only inverse for threshold-qualified bubble area."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

CORAL = np.asarray((223, 102, 95), dtype=np.float32)
TEAL = np.asarray((41, 150, 147), dtype=np.float32)
GOLD = np.asarray((194, 142, 50), dtype=np.float32)
THRESHOLD = np.asarray((79, 104, 117), dtype=np.float32)


def _mask(array: np.ndarray, color: np.ndarray, tolerance: float) -> np.ndarray:
    return np.linalg.norm(array - color, axis=2) <= tolerance


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y = int(y)
        x = int(x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for ny, nx in (
                (cy - 1, cx),
                (cy + 1, cx),
                (cy, cx - 1),
                (cy, cx + 1),
                (cy - 1, cx - 1),
                (cy - 1, cx + 1),
                (cy + 1, cx - 1),
                (cy + 1, cx + 1),
            ):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= minimum:
            found.append(np.asarray(points, dtype=np.int32))
    return found


def _largest(mask: np.ndarray) -> np.ndarray:
    parts = _components(mask, 80)
    if not parts:
        raise ValueError("colored bubble is absent")
    return max(parts, key=len)


def _thresholds(array: np.ndarray) -> tuple[float, float]:
    mask = _mask(array, THRESHOLD, 20.0)
    if int(mask.sum()) < 160:
        raise ValueError("both cutoff lines are absent")
    column_counts = mask.sum(axis=0)
    row_counts = mask.sum(axis=1)
    x_cut = float(int(np.argmax(column_counts)))
    y_cut = float(int(np.argmax(row_counts)))
    if float(column_counts.max()) < 25.0 or float(row_counts.max()) < 25.0:
        raise ValueError("cutoff lines are too short or unstable")
    return x_cut, y_cut


def _bubble(mask: np.ndarray) -> tuple[float, float, float]:
    points = _largest(mask)
    y = float(points[:, 0].mean())
    x = float(points[:, 1].mean())
    area = float(len(points))
    equivalent_radius = math.sqrt(area / math.pi)
    if equivalent_radius < 7.0:
        raise ValueError("bubble is too small")
    return x, y, area


def decision_from_image(image: Image.Image) -> str:
    """Recover eligibility and compare colored filled areas from the PNG only."""

    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    if array.shape[:2] != (512, 512):
        raise ValueError("unexpected canvas size")
    x_cut, y_cut = _thresholds(array)
    coral = _bubble(_mask(array, CORAL, 42.0))
    teal = _bubble(_mask(array, TEAL, 42.0))
    gold = _bubble(_mask(array, GOLD, 42.0))
    bubbles = {"coral": coral, "teal": teal, "gold": gold}
    eligible = [
        name
        for name, (x, y, _area) in bubbles.items()
        if x > x_cut + 8.0 and y < y_cut - 8.0
    ]
    if set(eligible) != {"coral", "teal"}:
        raise ValueError("threshold qualification is unresolved")
    difference = coral[2] - teal[2]
    if abs(difference) < 30.0:
        raise ValueError("qualified filled areas are unresolved")
    return "coral" if difference > 0.0 else "teal"
