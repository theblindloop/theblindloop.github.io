"""Renderer and analytic arm for a compact threshold-qualified bubble chart."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
CARD_W = 134
CARD_H = 116

CORAL = "#df665f"
TEAL = "#299693"
GOLD = "#c28e32"
THRESHOLD = "#4f6875"
FRAME = "#b9c7ca"
CARD_PALE = ("#fffdf7", "#fbfffd", "#fffafd")
BACKGROUNDS = ("#f5f8f6", "#f4f7fb", "#faf6f1")
REMOTE = ("#c4d1d0", "#c8d1de", "#d8c9c0")
MIN_MARGIN = 100.0


def _qualified(scene: dict[str, Any]) -> list[dict[str, Any]]:
    x_cut = float(scene["x_threshold"])
    y_cut = float(scene["y_threshold"])
    return [
        bubble
        for bubble in scene["bubbles"]
        if float(bubble["x"]) > x_cut and float(bubble["y"]) < y_cut
    ]


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced chart with a compact, local decision card."""

    rng = random.Random(int(seed))
    card_x = rng.choice((60, 92, 124, 156, 188, 220, 252, 284, 316))
    card_y = rng.choice((54, 86, 118, 150, 182, 214, 246, 278, 310))
    x_threshold = card_x + 46
    y_threshold = card_y + 82
    target_positions = [
        (card_x + 67, card_y + 38),
        (card_x + 103, card_y + 55),
    ]
    if rng.randrange(2):
        target_positions.reverse()

    near_boundary = int(seed) % 13 == 0
    if near_boundary:
        radii = [20, 21]
    else:
        radii = [14, 22]
    coral_is_larger = int(seed) % 2 == 0
    if not coral_is_larger:
        radii.reverse()
    if rng.randrange(2):
        coral_position, teal_position = target_positions
    else:
        teal_position, coral_position = target_positions

    bubbles = [
        {
            "name": "coral",
            "x": int(coral_position[0]),
            "y": int(coral_position[1]),
            "radius": int(radii[0]),
            "color": CORAL,
        },
        {
            "name": "teal",
            "x": int(teal_position[0]),
            "y": int(teal_position[1]),
            "radius": int(radii[1]),
            "color": TEAL,
        },
        {
            "name": "gold",
            "x": int(card_x + 24),
            "y": int(card_y + 43),
            "radius": 24,
            "color": GOLD,
        },
    ]
    return {
        "bubbles": bubbles,
        "x_threshold": int(x_threshold),
        "y_threshold": int(y_threshold),
        "card_x": int(card_x),
        "card_y": int(card_y),
        "background": BACKGROUNDS[rng.randrange(len(BACKGROUNDS))],
        "card_background": CARD_PALE[rng.randrange(len(CARD_PALE))],
        "remote_variant": int(rng.randrange(20)),
        "style": int(rng.randrange(3)),
    }


def _draw_remote_context(draw: ImageDraw.ImageDraw, variant: int, style: int) -> None:
    """Draw pale, non-oracle context outside the compact card."""

    color = REMOTE[(variant + style) % len(REMOTE)]
    for index in range(5):
        x = 18 + ((variant * 31 + index * 73) % 108)
        y = 24 + ((variant * 19 + index * 37) % 84)
        draw.line((x, y, x + 28 + index * 4, y), fill=color, width=2)
        draw.line((x, y + 8, x + 18 + index * 2, y + 8), fill=REMOTE[1], width=2)
    for index in range(4):
        x = 360 + ((variant * 23 + index * 29) % 112)
        y = 354 + ((variant * 17 + index * 21) % 92)
        draw.line((x, y, min(WIDTH - 12, x + 42), y), fill=REMOTE[2], width=2)
        draw.line((x, y + 8, min(WIDTH - 12, x + 24), y + 8), fill=color, width=2)


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the final RGB image."""

    image = Image.new("RGB", (WIDTH, HEIGHT), scene["background"])
    draw = ImageDraw.Draw(image)
    _draw_remote_context(draw, int(scene["remote_variant"]), int(scene["style"]))
    left = int(scene["card_x"])
    top = int(scene["card_y"])
    draw.rounded_rectangle(
        (left, top, left + CARD_W, top + CARD_H),
        radius=10,
        fill=scene["card_background"],
        outline=FRAME,
        width=3,
    )
    x_cut = int(scene["x_threshold"])
    y_cut = int(scene["y_threshold"])
    draw.line((x_cut, top + 19, x_cut, top + 70), fill=THRESHOLD, width=5)
    draw.line((left + 39, y_cut, left + 119, y_cut), fill=THRESHOLD, width=5)
    draw.line((x_cut - 7, top + 19, x_cut + 7, top + 19), fill=THRESHOLD, width=3)
    draw.line((left + 119, y_cut - 7, left + 119, y_cut + 7), fill=THRESHOLD, width=3)
    for bubble in scene["bubbles"]:
        x = int(bubble["x"])
        y = int(bubble["y"])
        radius = int(bubble["radius"])
        draw.ellipse(
            (x - radius, y - radius, x + radius, y + radius),
            fill=bubble["color"],
            outline="#ffffff",
            width=2,
        )
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Select the larger colored bubble after both visible cutoff tests."""

    qualified = {bubble["name"]: bubble for bubble in _qualified(scene)}
    if set(qualified) != {"coral", "teal"}:
        raise ValueError("scene does not contain exactly two qualified answer bubbles")
    coral_area = float(qualified["coral"]["radius"]) ** 2
    teal_area = float(qualified["teal"]["radius"]) ** 2
    if coral_area == teal_area:
        raise ValueError("qualified bubble areas tie")
    return "coral" if coral_area > teal_area else "teal"


def margin(scene: dict[str, Any]) -> float:
    """Return the signed radius-squared difference for the two survivors."""

    qualified = {bubble["name"]: bubble for bubble in _qualified(scene)}
    return float(qualified["coral"]["radius"] ** 2 - qualified["teal"]["radius"] ** 2)


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Quarantine near-equal filled areas."""

    return abs(margin(scene)) < MIN_MARGIN


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No nontrivial pixel-identical latent aliases are declared."""

    return []
