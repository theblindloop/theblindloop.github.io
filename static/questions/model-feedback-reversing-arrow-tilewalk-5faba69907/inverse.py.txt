"""Independent image-only decoder for the reversing-arrow tile walk."""

from __future__ import annotations

import numpy as np
from PIL import Image

CANVAS = (660, 660)
N, CELL, X0, Y0, STEPS = 5, 100, 80, 80, 7
DIRS = ((0, -1), (1, 0), (0, 1), (-1, 0))
TOKEN_COLORS = (np.array((220, 45, 55)), np.array((25, 105, 220)))
GOAL_COLORS = {"amber": np.array((235, 150, 20)), "cyan": np.array((0, 165, 185)),
               "violet": np.array((125, 70, 195))}


class OracleError(ValueError):
    pass


def _near(crop, rgb, distance=25):
    delta = crop.astype(np.int32) - rgb.astype(np.int32)
    return np.sum(delta * delta, axis=2) < distance * distance


def _arrow_direction(arr, cx, cy, rgb):
    crop = arr[cy - 20:cy + 21, cx - 20:cx + 21]
    yy, xx = np.nonzero(_near(crop, rgb))
    if not 210 <= len(xx) <= 430:
        raise OracleError(f"unreadable arrow ink count {len(xx)}")
    width, height = int(xx.max() - xx.min()), int(yy.max() - yy.min())
    mean_x, mean_y = float(xx.mean() - 20), float(yy.mean() - 20)
    if height > width + 5:
        return 0 if mean_y < -0.8 else 2
    if width > height + 5:
        return 3 if mean_x < -0.8 else 1
    raise OracleError("arrow orientation ambiguous")


def _decode_grid(arr):
    result = []
    for y in range(N):
        row = []
        for x in range(N):
            cx, cy = X0 + x * CELL + CELL // 2, Y0 + y * CELL + CELL // 2
            row.append([_arrow_direction(arr, cx, cy - 22, TOKEN_COLORS[0]),
                        _arrow_direction(arr, cx, cy + 22, TOKEN_COLORS[1])])
        result.append(row)
    return result


def _decode_starts(arr):
    found = [[], []]
    for y in range(N):
        for x in range(N):
            red_crop = arr[Y0 + y * CELL + 5:Y0 + y * CELL + 31,
                           X0 + x * CELL + 5:X0 + x * CELL + 31]
            blue_crop = arr[Y0 + y * CELL + 5:Y0 + y * CELL + 31,
                            X0 + x * CELL + 69:X0 + x * CELL + 95]
            if int(_near(red_crop, TOKEN_COLORS[0]).sum()) > 180:
                found[0].append((x, y))
            if int(_near(blue_crop, TOKEN_COLORS[1]).sum()) > 180:
                found[1].append((x, y))
    if any(len(values) != 1 for values in found):
        raise OracleError(f"expected one start per token: {found}")
    if found[0][0] == found[1][0]:
        raise OracleError("tokens share a start")
    return found[0][0], found[1][0]


def _decode_goals(arr):
    found = {}
    for y in range(N):
        for x in range(N):
            crop = arr[Y0 + y * CELL + 3:Y0 + (y + 1) * CELL - 3,
                       X0 + x * CELL + 3:X0 + (x + 1) * CELL - 3]
            hits = [name for name, rgb in GOAL_COLORS.items() if int(_near(crop, rgb).sum()) > 1800]
            if len(hits) > 1:
                raise OracleError("multiple goal colors on tile")
            if hits:
                found[(x, y)] = hits[0]
    if len(found) != 3 or set(found.values()) != set(GOAL_COLORS):
        raise OracleError("expected one goal of each color")
    return found


def _step(cell, direction):
    dx, dy = DIRS[direction]
    return cell[0] + dx, cell[1] + dy


def decision_from_image(image: Image.Image) -> str:
    if image.size != CANVAS:
        raise OracleError(f"wrong image size {image.size}")
    arr = np.asarray(image.convert("RGB"))
    board = _decode_grid(arr)
    red, blue = _decode_starts(arr)
    goals = _decode_goals(arr)
    states, reds, used = set(), [red], set()
    reread = False
    for _ in range(STEPS):
        old_red, old_blue = red, blue
        red_key, blue_key = (old_blue[0], old_blue[1], 0), (old_red[0], old_red[1], 1)
        reread = reread or red_key in used or blue_key in used
        used.update((red_key, blue_key))
        red_dir = board[old_blue[1]][old_blue[0]][0]
        blue_dir = board[old_red[1]][old_red[0]][1]
        red, blue = _step(old_red, red_dir), _step(old_blue, blue_dir)
        if not (0 <= red[0] < N and 0 <= red[1] < N and 0 <= blue[0] < N and 0 <= blue[1] < N):
            raise OracleError("walk leaves grid")
        board[old_blue[1]][old_blue[0]][0] = (red_dir + 2) % 4
        board[old_red[1]][old_red[0]][1] = (blue_dir + 2) % 4
        signature = (red, blue, tuple(tuple(tuple(pair) for pair in row) for row in board))
        if signature in states:
            raise OracleError("complete state repeats")
        states.add(signature)
        reds.append(red)
    if not reread:
        raise OracleError("no reversed arrow is read again")
    if red not in goals:
        raise OracleError("final red tile is not a goal")
    if any(tile in goals for tile in reds[:-1]):
        raise OracleError("red visits a goal early")
    return goals[red]
