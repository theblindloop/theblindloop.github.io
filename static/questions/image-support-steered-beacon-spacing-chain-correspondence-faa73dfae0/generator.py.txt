"""Deterministic renderer and analytic arm for the checkpoint-spacing world."""

from __future__ import annotations

import itertools
import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "beacon-spacing-chain-renderer-0.1.0"
WIDTH = 1200
HEIGHT = 420
BACKGROUND = (248, 250, 252)
PANEL_FILL = (240, 245, 249)
PANEL_EDGE = (205, 216, 226)
TEXT = (40, 52, 64)
NAVY = (24, 52, 84)
GOLD = (214, 156, 39)
WHITE = (255, 255, 255)

BASE_GAPS = (36, 48, 60, 72, 84)
FIELD_BOUNDS = ((60, 420), (420, 780), (780, 1140))
ROLES = ("R", "A", "B")
QUARANTINE_MARGIN = 24.0


def _score(reference: list[int], candidate_left_to_right: list[int]) -> int:
    transformed = list(reversed(candidate_left_to_right))
    return int(sum(abs(int(a) - int(b)) for a, b in zip(reference, transformed, strict=True)))


def _candidate_map(records: list[dict[str, Any]]) -> dict[str, list[int]]:
    result: dict[str, list[int]] = {}
    for record in records:
        slot = str(record["slot"])
        result[slot] = [int(value) for value in record["gaps"]]
    return result


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced scene with equal gap multisets and a reversed winner."""

    rng = random.Random(int(seed))
    reference = list(BASE_GAPS)
    rng.shuffle(reference)
    target = list(reversed(reference))
    wrong_pool = []
    for permutation in itertools.permutations(BASE_GAPS):
        wrong = list(permutation)
        if wrong == target:
            continue
        if _score(reference, wrong) >= 48:
            wrong_pool.append(wrong)
    wrong = list(rng.choice(wrong_pool))

    winner = "A" if int(seed) % 2 == 0 else "B"
    records = [
        {"slot": "A", "gaps": target if winner == "A" else wrong},
        {"slot": "B", "gaps": wrong if winner == "A" else target},
    ]
    if rng.random() < 0.5:
        records.reverse()

    phase = rng.choice((0.0, 0.24, -0.28, 0.41))
    return {
        "reference_gaps": reference,
        "candidate_gaps": records,
        "layout": {
            "width": WIDTH,
            "height": HEIGHT,
            "field_bounds": [list(bounds) for bounds in FIELD_BOUNDS],
            "trace_start": 90,
            "trace_end": 1110,
            "route_phase": phase,
        },
    }


def analytic_gold(scene: dict[str, Any]) -> str:
    reference = [int(value) for value in scene["reference_gaps"]]
    candidates = _candidate_map(scene["candidate_gaps"])
    scores = {slot: _score(reference, candidates[slot]) for slot in ("A", "B")}
    if scores["A"] == scores["B"]:
        raise ValueError("candidate gap-rhythm scores tie")
    return "A" if scores["A"] < scores["B"] else "B"


def margin(scene: dict[str, Any]) -> float:
    reference = [int(value) for value in scene["reference_gaps"]]
    candidates = _candidate_map(scene["candidate_gaps"])
    score_a = _score(reference, candidates["A"])
    score_b = _score(reference, candidates["B"])
    return float(abs(score_a - score_b))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def _route_y(x: int, phase: float) -> int:
    t = float(x - 90)
    y = 218.0 + 17.0 * math.sin(t / 72.0 + phase) + 7.0 * math.sin(t / 31.0 - 0.4 * phase)
    return int(round(y))


def _flag_xs(gaps: list[int], bounds: tuple[int, int]) -> list[int]:
    start, end = bounds
    x = start + 30
    positions = [x]
    for gap in gaps:
        x += int(gap)
        positions.append(x)
    if positions[-1] != end - 30:
        raise ValueError("gap sequence does not fill the declared field")
    return positions


def _draw_slot_badge(draw: ImageDraw.ImageDraw, x: int, slot: str) -> None:
    draw.rectangle((x, 128, x + 30, 158), fill=WHITE, outline=PANEL_EDGE, width=1)
    font = ImageFont.load_default()
    draw.text((x + 11, 137), slot, fill=TEXT, font=font)


def render(scene: dict[str, Any]) -> Image.Image:
    layout = scene["layout"]
    width = int(layout["width"])
    height = int(layout["height"])
    bounds = [tuple(int(value) for value in item) for item in layout["field_bounds"]]
    phase = float(layout["route_phase"])
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    draw.text((60, 30), "REVERSE CHECKPOINT SPACING", fill=TEXT, font=font)
    draw.text((60, 48), "follow the single navy route; compare the five gaps", fill=TEXT, font=font)

    for index, (slot, bound) in enumerate(zip(ROLES, bounds, strict=True)):
        x0, x1 = bound
        draw.rectangle((x0, 112, x1, 320), fill=PANEL_FILL, outline=PANEL_EDGE, width=2)
        _draw_slot_badge(draw, x0 + 16, slot)
        caption = "REFERENCE" if slot == "R" else f"CANDIDATE {slot}"
        draw.text((x0 + 56, 136), caption, fill=TEXT, font=font)
        draw.text((x0 + 56, 152), "six attached flags", fill=TEXT, font=font)
        if index < 2:
            draw.line((x1, 112, x1, 320), fill=PANEL_EDGE, width=1)

    trace_start = int(layout["trace_start"])
    trace_end = int(layout["trace_end"])
    route_points = [(x, _route_y(x, phase)) for x in range(trace_start, trace_end + 1, 4)]
    if route_points[-1][0] != trace_end:
        route_points.append((trace_end, _route_y(trace_end, phase)))
    draw.line(route_points, fill=NAVY, width=9, joint="curve")

    candidates = _candidate_map(scene["candidate_gaps"])
    all_gaps = {"R": [int(value) for value in scene["reference_gaps"]], **candidates}
    for slot, bound in zip(ROLES, bounds, strict=True):
        for x in _flag_xs(all_gaps[slot], bound):
            y = _route_y(x, phase)
            # The flag stops in the upper edge of the stroke, leaving the
            # central navy route continuous beneath every checkpoint.
            draw.rectangle((x - 6, y - 34, x + 6, y - 3), fill=GOLD)

    return image


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    twin = dict(scene)
    twin["candidate_gaps"] = [
        {"slot": record["slot"], "gaps": list(record["gaps"])}
        for record in reversed(scene["candidate_gaps"])
    ]
    return [("candidate_record_order", twin)]
