"""Independent pixel-only inverse for the checkpoint-spacing world."""

from __future__ import annotations

import numpy as np
from PIL import Image


_NAVY = np.asarray((24, 52, 84), dtype=np.int16)
_GOLD = np.asarray((214, 156, 39), dtype=np.int16)
_NAVY_TOL = 8
_GOLD_TOL = 8


def _colour_mask(array: np.ndarray, colour: np.ndarray, tolerance: int) -> np.ndarray:
    delta = np.abs(array.astype(np.int16) - colour.reshape(1, 1, 3))
    return np.all(delta <= tolerance, axis=2)


def _horizontal_runs(mask: np.ndarray) -> list[tuple[int, int]]:
    active = np.any(mask, axis=0)
    padded = np.concatenate(([False], active, [False]))
    changes = np.flatnonzero(padded[1:] != padded[:-1])
    return [(int(changes[i]), int(changes[i + 1] - 1)) for i in range(0, len(changes), 2)]


def _route_is_complete(navy: np.ndarray) -> None:
    active = np.any(navy, axis=0)
    runs = _horizontal_runs(navy)
    if not runs:
        raise ValueError("navy route is absent")
    longest = max(runs, key=lambda pair: pair[1] - pair[0])
    width = navy.shape[1]
    if longest[0] > int(0.12 * width) or longest[1] < int(0.88 * width):
        raise ValueError("navy route does not span all three fields")
    if not np.all(active[longest[0] : longest[1] + 1]):
        raise ValueError("navy route has a visible break")
    if len(runs) != 1:
        raise ValueError("more than one disconnected horizontal route remains")


def _flag_centers(gold: np.ndarray, left: int, right: int) -> list[float]:
    runs = _horizontal_runs(gold[:, left:right])
    centers = [0.5 * (start + end) + left for start, end in runs if end - start + 1 >= 5]
    if len(centers) != 6:
        raise ValueError("each field must expose exactly six gold checkpoint flags")
    centers.sort()
    if any((b - a) < 24.0 for a, b in zip(centers, centers[1:])):
        raise ValueError("checkpoint flags are not separately recoverable")
    return centers


def _gap_code(centers: list[float]) -> list[float]:
    return [b - a for a, b in zip(centers, centers[1:])]


def _score(reference: list[float], candidate: list[float]) -> float:
    transformed = list(reversed(candidate))
    return float(sum(abs(a - b) for a, b in zip(reference, transformed, strict=True)))


def decision_from_image(image: Image.Image) -> str:
    """Recover the candidate winner from only the final RGB pixels."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    navy = _colour_mask(array, _NAVY, _NAVY_TOL)
    gold = _colour_mask(array, _GOLD, _GOLD_TOL)
    _route_is_complete(navy)
    width = array.shape[1]
    third = width / 3.0
    reference = _gap_code(_flag_centers(gold, 0, int(round(third))))
    candidate_a = _gap_code(_flag_centers(gold, int(round(third)), int(round(2.0 * third))))
    candidate_b = _gap_code(_flag_centers(gold, int(round(2.0 * third)), width))
    score_a = _score(reference, candidate_a)
    score_b = _score(reference, candidate_b)
    if score_a == score_b:
        raise ValueError("pixel gap-rhythm scores are tied")
    return "A" if score_a < score_b else "B"
