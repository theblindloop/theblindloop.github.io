"""Analytic sampler and raster renderer for the paired-link-cut world."""

from __future__ import annotations

import copy
import random
from collections import deque

from PIL import Image, ImageDraw

CANVAS = [480, 360]
PALETTE = {
    "background": [250, 250, 247],
    "gray": [116, 123, 132],
    "black": [25, 28, 34],
    "red": [218, 55, 72],
    "green": [38, 160, 91],
    "blue": [54, 104, 218],
}
COLORS = ("red", "green", "blue")

# An irregular, well-separated cycle. Edge i joins node i to node (i+1) mod 8.
_BASE_NODES = [
    [64, 180], [116, 82], [237, 48], [363, 88],
    [420, 181], [357, 278], [235, 314], [112, 274],
]


def _transform_nodes(nodes: list[list[int]], code: int, dx: int, dy: int) -> list[list[int]]:
    out = []
    for x, y in nodes:
        if code & 1:
            x = CANVAS[0] - x
        if code & 2:
            y = CANVAS[1] - y
        out.append([x + dx, y + dy])
    return out


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    nodes = [[x + rng.randint(-4, 4), y + rng.randint(-4, 4)] for x, y in _BASE_NODES]
    nodes = _transform_nodes(nodes, (seed // 5) % 4, rng.randint(-4, 4), rng.randint(-3, 3))
    edges = [[i, (i + 1) % 8] for i in range(8)]

    # Rotate the two-terminal problem around the ring to break absolute-position cues.
    start = (seed // 3) % 8
    terminals = [start, (start + 4) % 8]
    winner_role = [(start + 1) % 8, (start - 3) % 8]  # one edge on each terminal route
    upper_loser = [start % 8, (start + 2) % 8]
    lower_loser = [(start - 1) % 8, (start - 2) % 8]
    roles = [winner_role, upper_loser, lower_loser]
    shift = seed % 3
    assigned = [COLORS[(shift + i) % 3] for i in range(3)]
    colored_pairs = {assigned[i]: list(roles[i]) for i in range(3)}
    return {
        "canvas": list(CANVAS),
        "nodes": nodes,
        "edges": edges,
        "terminals": terminals,
        "colored_pairs": colored_pairs,
        "palette": copy.deepcopy(PALETTE),
        "node_radius": 12,
        "terminal_half_size": 16,
        "edge_width": 7,
    }


def render(scene: dict) -> Image.Image:
    width, height = scene["canvas"]
    palette = scene["palette"]
    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), tuple(palette["background"]))
    draw = ImageDraw.Draw(image)
    edge_colors = {index: "gray" for index in range(len(scene["edges"]))}
    for color, indices in scene["colored_pairs"].items():
        for index in indices:
            edge_colors[index] = color
    for index, (a, b) in enumerate(scene["edges"]):
        draw.line(
            [(scene["nodes"][a][0] * scale, scene["nodes"][a][1] * scale),
             (scene["nodes"][b][0] * scale, scene["nodes"][b][1] * scale)],
            fill=tuple(palette[edge_colors[index]]), width=scene["edge_width"] * scale,
        )
    radius = scene["node_radius"] * scale
    half = scene["terminal_half_size"] * scale
    for index, (x, y) in enumerate(scene["nodes"]):
        cx, cy = x * scale, y * scale
        if index in scene["terminals"]:
            draw.rectangle(
                [cx - half, cy - half, cx + half, cy + half],
                fill=tuple(palette["black"]), outline=tuple(palette["background"]), width=2 * scale,
            )
        else:
            draw.ellipse(
                [cx - radius, cy - radius, cx + radius, cy + radius],
                fill=tuple(palette["black"]), outline=tuple(palette["background"]), width=2 * scale,
            )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def _connected(edges: list[tuple[int, int]], source: int, target: int, removed: set[int]) -> bool:
    adjacency: dict[int, list[int]] = {}
    for index, (a, b) in enumerate(edges):
        if index in removed:
            continue
        adjacency.setdefault(a, []).append(b)
        adjacency.setdefault(b, []).append(a)
    seen = {source}
    queue = deque([source])
    while queue:
        node = queue.popleft()
        if node == target:
            return True
        for nxt in adjacency.get(node, []):
            if nxt not in seen:
                seen.add(nxt)
                queue.append(nxt)
    return False


def _qualifies(scene: dict, indices: list[int]) -> bool:
    edges = [tuple(edge) for edge in scene["edges"]]
    source, target = scene["terminals"]
    a, b = indices
    return (
        _connected(edges, source, target, {a})
        and _connected(edges, source, target, {b})
        and not _connected(edges, source, target, {a, b})
    )


def analytic_gold(scene: dict) -> str:
    winners = [color for color in COLORS if _qualifies(scene, scene["colored_pairs"][color])]
    if len(winners) != 1:
        raise ValueError(f"expected one minimal two-link cut, found {winners}")
    return winners[0]


def margin(scene: dict) -> float:
    # Winner disconnects (score 1); runners-up leave a route (score 0).
    scores = sorted((1.0 if _qualifies(scene, pair) else 0.0
                     for pair in scene["colored_pairs"].values()), reverse=True)
    return scores[0] - scores[1]


def is_quarantined(scene: dict) -> bool:
    try:
        if sorted(len(pair) for pair in scene["colored_pairs"].values()) != [2, 2, 2]:
            return True
        edges = [tuple(edge) for edge in scene["edges"]]
        source, target = scene["terminals"]
        all_singles_safe = all(
            _connected(edges, source, target, {edge_index})
            for pair in scene["colored_pairs"].values() for edge_index in pair
        )
        return not all_singles_safe or margin(scene) < 1.0 or analytic_gold(scene) not in COLORS
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    count = len(scene["nodes"])
    remap = {old: count - 1 - old for old in range(count)}
    twin["nodes"] = list(reversed(scene["nodes"]))
    twin["edges"] = [[remap[a], remap[b]] for a, b in scene["edges"]]
    twin["terminals"] = [remap[i] for i in scene["terminals"]]
    # Colored pairs index edges, and edge order is deliberately unchanged.
    return [("reverse_node_identifier_order", twin)]
