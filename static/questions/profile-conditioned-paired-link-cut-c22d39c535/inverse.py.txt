"""Independent pixel-only graph recovery for paired-link-cut."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_NAMES = ("background", "gray", "black", "red", "green", "blue")
_RGB = np.array([
    [250, 250, 247], [116, 123, 132], [25, 28, 34],
    [218, 55, 72], [38, 160, 91], [54, 104, 218],
], dtype=np.int32)


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        pixels = []
        while stack:
            cy, cx = stack.pop()
            pixels.append((cy, cx))
            for ny in range(max(0, cy - 1), min(height, cy + 2)):
                for nx in range(max(0, cx - 1), min(width, cx + 2)):
                    if mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        if len(pixels) >= minimum:
            found.append(np.asarray(pixels, dtype=np.int32))
    return found


def _connected(edges: list[tuple[int, int]], source: int, target: int, removed: set[int]) -> bool:
    adjacency: dict[int, list[int]] = {}
    for index, (a, b) in enumerate(edges):
        if index in removed:
            continue
        adjacency.setdefault(a, []).append(b)
        adjacency.setdefault(b, []).append(a)
    seen = {source}
    queue = deque([source])
    while queue:
        u = queue.popleft()
        if u == target:
            return True
        for v in adjacency.get(u, []):
            if v not in seen:
                seen.add(v)
                queue.append(v)
    return False


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int32)
    distances = ((array[:, :, None, :] - _RGB[None, None, :, :]) ** 2).sum(axis=3)
    labels = distances.argmin(axis=2)

    node_components = _components(labels == 2, 180)
    if len(node_components) != 8:
        raise ValueError(f"expected 8 black stations, found {len(node_components)}")
    centers = [(float(p[:, 1].mean()), float(p[:, 0].mean())) for p in node_components]
    areas = [len(p) for p in node_components]
    terminals = sorted(range(8), key=lambda i: areas[i], reverse=True)[:2]
    if min(areas[i] for i in terminals) < 680 or max(areas[i] for i in range(8) if i not in terminals) > 520:
        raise ValueError("square terminals are not distinguishable from round stations")

    edges: list[tuple[int, int]] = []
    colors: list[str] = []
    for label, name in ((1, "gray"), (3, "red"), (4, "green"), (5, "blue")):
        expected = 2
        components = _components(labels == label, 80)
        if len(components) != expected:
            raise ValueError(f"expected {expected} separate {name} links, found {len(components)}")
        for pixels in components:
            xy = pixels[:, ::-1].astype(np.float64)
            proximity = []
            for index, (cx, cy) in enumerate(centers):
                dist2 = ((xy[:, 0] - cx) ** 2 + (xy[:, 1] - cy) ** 2).min()
                proximity.append((float(dist2), index))
            endpoints = sorted(proximity)[:2]
            if endpoints[1][0] > 420.0:
                raise ValueError("link has no unambiguous station endpoints")
            edge = tuple(sorted((endpoints[0][1], endpoints[1][1])))
            if edge[0] == edge[1] or edge in edges:
                raise ValueError("ambiguous or duplicate recovered link")
            edges.append(edge)
            colors.append(name)

    if len(edges) != 8 or any(colors.count(color) != 2 for color in ("red", "green", "blue", "gray")):
        raise ValueError("wrong link inventory")
    winners = []
    for color in ("red", "green", "blue"):
        pair = [i for i, name in enumerate(colors) if name == color]
        a, b = pair
        if (_connected(edges, terminals[0], terminals[1], {a})
                and _connected(edges, terminals[0], terminals[1], {b})
                and not _connected(edges, terminals[0], terminals[1], {a, b})):
            winners.append(color)
    if len(winners) != 1:
        raise ValueError(f"expected one minimal two-link cut, found {winners}")
    return winners[0]
