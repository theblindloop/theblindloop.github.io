"""Deterministic latent sampler and raster renderer for a spiral-growth world."""

from __future__ import annotations

import colorsys
import math
import random
from PIL import Image, ImageDraw


def _hex(rgb: tuple[int, int, int]) -> str:
    return "#" + "".join(f"{v:02x}" for v in rgb)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    hue = rng.random()
    dark = tuple(round(v * 255) for v in colorsys.hsv_to_rgb(hue, 0.84, 0.46))
    light = tuple(round(v * 255) for v in colorsys.hsv_to_rgb((hue + 0.035) % 1, 0.60, 0.92))
    bgv = rng.randint(244, 252)
    return {
        "width": 384,
        "height": 384,
        "center_x": rng.uniform(183.0, 201.0),
        "center_y": rng.uniform(183.0, 201.0),
        "phase": rng.uniform(0.0, 2.0 * math.pi),
        "handedness": "clockwise" if int(seed) % 2 == 0 else "counterclockwise",
        "inner_radius": rng.uniform(12.0, 18.0),
        "radial_growth": rng.uniform(20.0, 23.0),
        "turns": rng.uniform(0.74, 0.84),
        "stroke": rng.uniform(7.0, 9.0),
        "background": _hex((bgv, bgv, min(255, bgv + 2))),
        "ink": [_hex(dark), _hex(light)],
    }


def _rgb(value: str) -> tuple[int, int, int]:
    value = value.lstrip("#")
    return tuple(int(value[i : i + 2], 16) for i in (0, 2, 4))


def render(scene: dict) -> Image.Image:
    # All declared latent fields are deliberately read here for the rendered-field audit.
    width, height = int(scene["width"]), int(scene["height"])
    cx, cy = float(scene["center_x"]), float(scene["center_y"])
    phase = round(float(scene["phase"]) % (2.0 * math.pi), 8)
    handedness = scene["handedness"]
    inner = float(scene["inner_radius"])
    growth = float(scene["radial_growth"])
    turns = float(scene["turns"])
    stroke = float(scene["stroke"])
    background = scene["background"]
    ink0, ink1 = map(_rgb, scene["ink"])
    sign = 1.0 if handedness == "clockwise" else -1.0
    ss = 3
    image = Image.new("RGB", (width * ss, height * ss), background)
    draw = ImageDraw.Draw(image)
    steps = 420
    points: list[tuple[float, float]] = []
    colors: list[tuple[int, int, int]] = []
    for i in range(steps + 1):
        u = i / steps
        t = u * turns * 2.0 * math.pi
        radius = inner + growth * t
        angle = phase + sign * t
        points.append(((cx + radius * math.cos(angle)) * ss, (cy + radius * math.sin(angle)) * ss))
        colors.append(tuple(round((1.0 - u) * ink0[k] + u * ink1[k]) for k in range(3)))
    # Overlapping short strokes make one connected, smoothly graded trail.
    for i in range(steps):
        draw.line([points[i], points[i + 1]], fill=colors[i], width=round(stroke * ss + 2), joint="curve")
    # Round caps are part of the same component and expose the two true ends.
    cap = stroke * ss / 2.0
    for point, color in ((points[0], colors[0]), (points[-1], colors[-1])):
        draw.ellipse([point[0] - cap, point[1] - cap, point[0] + cap, point[1] + cap], fill=color)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["handedness"])


def margin(scene: dict) -> float:
    return round(float(scene["turns"]) - 0.50, 9)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.18


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["phase"] = float(scene["phase"]) + 2.0 * math.pi
    return [("phase_plus_full_turn", twin)]
