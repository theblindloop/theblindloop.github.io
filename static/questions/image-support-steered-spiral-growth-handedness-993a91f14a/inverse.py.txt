"""Independent pixel-only inverse arm for spiral growth handedness."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    pass


def _foreground(array: np.ndarray) -> np.ndarray:
    border = np.concatenate((array[:3].reshape(-1, 3), array[-3:].reshape(-1, 3), array[:, :3].reshape(-1, 3), array[:, -3:].reshape(-1, 3)))
    background = np.median(border, axis=0)
    distance = np.linalg.norm(array.astype(np.float64) - background, axis=2)
    return distance > 32.0


def _components(mask: np.ndarray) -> list[int]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    sizes: list[int] = []
    for yy, xx in np.argwhere(mask):
        y, x = int(yy), int(xx)
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = deque([(y, x)])
        size = 0
        while queue:
            py, px = queue.popleft()
            size += 1
            for ny, nx in ((py - 1, px), (py + 1, px), (py, px - 1), (py, px + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        sizes.append(size)
    return sorted(sizes, reverse=True)


def _ordered_centers(array: np.ndarray, mask: np.ndarray) -> np.ndarray:
    ys, xs = np.nonzero(mask)
    if len(xs) < 700:
        raise OracleAbstention("too little trail ink")
    rgb = array[ys, xs].astype(np.float64)
    # Ignore pale antialias fringe, then order trail pieces along its dominant color ramp.
    centered_rgb = rgb - rgb.mean(axis=0)
    _, singular, vt = np.linalg.svd(centered_rgb, full_matrices=False)
    if singular[0] < 300.0:
        raise OracleAbstention("trail has no recoverable longitudinal color progression")
    projection = centered_rgb @ vt[0]
    lo, hi = np.percentile(projection, [2.0, 98.0])
    if hi - lo < 20.0:
        raise OracleAbstention("color progression too weak")
    edges = np.linspace(lo, hi, 33)
    centers = []
    brightness = []
    for left, right in zip(edges[:-1], edges[1:]):
        take = (projection >= left) & (projection < right)
        if take.sum() < 12:
            raise OracleAbstention("missing interior trail band")
        centers.append((float(xs[take].mean()), float(ys[take].mean())))
        mean_rgb = rgb[take].mean(axis=0)
        brightness.append(float(mean_rgb @ np.asarray([0.2126, 0.7152, 0.0722])))
    path = np.asarray(centers)
    # PCA sign is arbitrary. The trail is deliberately shaded darker-to-lighter
    # from tight end to wide end, so use only that ordinal cue to orient the trace.
    # The class still comes from the geometry of all ordered interior bands.
    if brightness[0] > brightness[-1]:
        path = path[::-1]
    return path


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    mask = _foreground(array)
    sizes = _components(mask)
    if not sizes or sizes[0] < 700:
        raise OracleAbstention("no usable trail")
    # A valid image has one continuous trail. Destructive removal of an interior
    # portion must therefore abstain instead of classifying disconnected remnants.
    if len(sizes) > 1 and sizes[1] > max(20, int(0.015 * sizes[0])):
        raise OracleAbstention("trail is disconnected")
    path = _ordered_centers(array, mask)
    steps = np.diff(path, axis=0)
    if np.max(np.linalg.norm(steps, axis=1)) > 90.0:
        raise OracleAbstention("ordered trace has a missing band")
    # Sum local oriented area along the ordered global trace. In raster coordinates
    # (y increases downward), positive angular progression is visually clockwise.
    anchor = np.median(path[5:16], axis=0)
    p = path - anchor
    signed = float(np.sum(p[:-1, 0] * p[1:, 1] - p[:-1, 1] * p[1:, 0]))
    scale = float(np.sum(np.linalg.norm(p[:-1], axis=1) * np.linalg.norm(np.diff(p, axis=0), axis=1)))
    if scale <= 0.0 or abs(signed) / scale < 0.12:
        raise OracleAbstention("handedness fit is ambiguous")
    return "clockwise" if signed > 0.0 else "counterclockwise"
