"""Independent image-only oracle for the S-curve port-link world.

The oracle works only from the raster.  It recovers the free (pale) background
and the two coloured ports by simple colour masks, then flood-fills the pale
field starting from the blue port.  If the flood reaches the red port the two
ports lie in one connected pale region ("yes"); otherwise they are separated by
the dark walls ("no").  No latent scene state is used.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


def _mask(arr, centre, tol):
    return (
        (np.abs(arr[:, :, 0].astype(np.int16) - centre[0]) < tol)
        & (np.abs(arr[:, :, 1].astype(np.int16) - centre[1]) < tol)
        & (np.abs(arr[:, :, 2].astype(np.int16) - centre[2]) < tol)
    )


def _flood(shape, seeds, passable):
    h, w = shape
    seen = np.zeros(shape, dtype=bool)
    stack = list(seeds)
    for (y, x) in stack:
        seen[y, x] = True
    while stack:
        y, x = stack.pop()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < h and 0 <= nx < w and passable[ny, nx] and not seen[ny, nx]:
                seen[ny, nx] = True
                stack.append((ny, nx))
    return seen


def decision_from_image(image: Image.Image) -> str:
    if image.mode != "RGB":
        image = image.convert("RGB")
    arr = np.asarray(image, dtype=np.int16)
    bg = _mask(arr, (250, 250, 246), 40)
    blue = _mask(arr, (34, 66, 214), 70)
    red = _mask(arr, (216, 58, 44), 70)
    if not blue.any() or not red.any():
        raise ValueError("port colour not found")
    passable = bg | blue | red
    blue_seeds = [(int(y), int(x)) for y, x in zip(*np.where(blue))]
    reached = _flood(passable.shape, blue_seeds, passable)
    return "yes" if reached[red].any() else "no"
