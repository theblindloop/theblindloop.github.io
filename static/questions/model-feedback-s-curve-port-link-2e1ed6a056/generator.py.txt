"""Deterministic renderer and analytic gold for the S-curve port-link world.

A pale field is split by three thick vertical walls into four vertical bands.
The left-most and right-most bands each hold a coloured port (blue on the left,
red on the right).  The outer two walls are always open at the bottom and the
middle wall is open at the top unless it is sealed.  When the middle wall has
its top gap, the pale background forms one continuous S-corridor that winds
from the blue port, under the outer walls, over the middle wall, and reaches
the red port: the two ports are connected.  When the middle wall is sealed to
full height it splits the pale field into two components and the ports are
disconnected.  A decorative key square in one corner is not part of the field.
The decision is whether the two ports lie in the same connected pale region.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

W = 800
H = 560
BG = (250, 250, 246)
INK = (42, 46, 54)
BLUE = (34, 66, 214)
RED = (216, 58, 44)
WALL_W = 26

PORT_L = (70, 280)
PORT_R = (724, 280)
PORT_RAD = 20

WALL_A_X = 256
WALL_B_X = 400
WALL_C_X = 544
TOP = 0
BOT = H
OPEN_A_FLOOR = 330
OPEN_C_FLOOR = 330
OPEN_B_CEIL = 240

PROMPT_FAMILIES = ("pf1_s_corridor_link", "pf2_winding_channel", "pf3_port_connectivity")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    sealed = bool(rng.randrange(2))
    variant = rng.randrange(4)
    return {"sealed": sealed, "variant": variant}


def _vwall(draw, x: float, y0: float, y1: float) -> None:
    half = WALL_W / 2
    draw.rectangle([x - half, y0, x + half, y1], fill=INK)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (W, H), BG)
    draw = ImageDraw.Draw(image)
    sealed = scene["sealed"]
    variant = scene["variant"]
    _vwall(draw, WALL_A_X, TOP, OPEN_A_FLOOR)
    if sealed:
        _vwall(draw, WALL_B_X, TOP, BOT)
    else:
        _vwall(draw, WALL_B_X, OPEN_B_CEIL, BOT)
    _vwall(draw, WALL_C_X, TOP, OPEN_C_FLOOR)
    for (px, py), colour in ((PORT_L, BLUE), (PORT_R, RED)):
        draw.ellipse([px - PORT_RAD, py - PORT_RAD, px + PORT_RAD, py + PORT_RAD], fill=colour)
    kx = 30 + (variant % 3) * 16
    draw.rectangle([kx, 12, kx + 22, 34], outline=INK, width=4)
    return image


def analytic_gold(scene: dict) -> str:
    return "no" if scene["sealed"] else "yes"


def margin(scene: dict) -> float:
    return 40.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list:
    return []
