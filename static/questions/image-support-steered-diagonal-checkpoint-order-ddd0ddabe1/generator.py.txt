from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

BACKGROUND = "#fcfcf8"
HALO = "#a1beda"
TRACE = "#17345d"
START = "#18a558"
FINISH = "#dc3f45"
CHECKPOINT = "#f2b72f"
DECORATION = "#d8d8d2"
WHITE = "#ffffff"


def _transform(points: list[tuple[int, int]], rotation: int, reflect: bool,
               center: tuple[int, int]) -> list[list[int]]:
    xs = [point[0] for point in points]
    ys = [point[1] for point in points]
    ox = (min(xs) + max(xs)) / 2.0
    oy = (min(ys) + max(ys)) / 2.0
    transformed: list[list[int]] = []
    for px, py in points:
        x, y = px - ox, py - oy
        if reflect:
            y = -y
        for _ in range(int(rotation) % 4):
            x, y = -y, x
        transformed.append([int(round(center[0] + x)), int(round(center[1] + y))])
    return transformed


def _template(variant: int) -> list[tuple[int, int]]:
    # One diagonal is surrounded by axis-aligned runs in a very thin strip.
    # Alternating vertical runs prevent a single global heading from revealing
    # the interior event while leaving the raster centerline unambiguous.
    if int(variant) % 2 == 0:
        return [(0, 16), (28, 16), (28, 0), (76, 16), (76, 32), (104, 32), (104, 16), (136, 16)]
    return [(0, 16), (30, 16), (30, 0), (76, 16), (76, 32), (104, 32), (104, 16), (136, 16)]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    event_order = "checkpoint" if int(seed) % 2 == 0 else "diagonal"
    centers = (96, 160, 224, 288, 352, 416)
    points = _transform(
        _template(rng.randrange(2)), rng.randrange(4), bool(rng.randrange(2)),
        (rng.choice(centers), rng.choice(centers)),
    )
    # Pale flecks are deliberately far from the compact route and never enter
    # either analytic gold or the inverse arm.
    decorations = []
    for _ in range(3):
        decorations.append([rng.randrange(18, 494), rng.randrange(18, 494), rng.choice((2, 3))])
    return {
        "width": 512,
        "height": 512,
        "points": points,
        "stroke": rng.choice((3, 4)),
        "event_order": event_order,
        "route_color": TRACE,
        "start_color": START,
        "finish_color": FINISH,
        "checkpoint_color": CHECKPOINT,
        "background": BACKGROUND,
        "decorations": decorations,
    }


def _midpoint(a: tuple[int, int], b: tuple[int, int]) -> tuple[int, int]:
    return int(round((a[0] + b[0]) / 2)), int(round((a[1] + b[1]) / 2))


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), scene["background"])
    draw = ImageDraw.Draw(image)
    for x, y, radius in scene["decorations"]:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=DECORATION)
    points = [tuple(map(int, point)) for point in scene["points"]]
    stroke = int(scene["stroke"])
    draw.line(points, fill=HALO, width=stroke + 6, joint="curve")
    # A long axis-aligned run before (segment 0) or after (segment 6) the sole
    # diagonal carries the ring, separating both events by full bends.
    checkpoint_segment = 0 if scene["event_order"] == "checkpoint" else 6
    cx, cy = _midpoint(points[checkpoint_segment], points[checkpoint_segment + 1])
    draw.ellipse((cx - 8, cy - 8, cx + 8, cy + 8), outline=WHITE, width=5)
    draw.ellipse((cx - 8, cy - 8, cx + 8, cy + 8), outline=scene["checkpoint_color"], width=3)
    # Paint the thin route last through the checkpoint ring so color
    # segmentation retains one unbroken centerline.
    draw.line(points, fill=scene["route_color"], width=stroke, joint="curve")
    for point, color in ((points[0], scene["start_color"]), (points[-1], scene["finish_color"])):
        x, y = point
        draw.ellipse((x - 7, y - 7, x + 7, y + 7), fill=WHITE)
        draw.ellipse((x - 5, y - 5, x + 5, y + 5), fill=color)
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["event_order"])


def margin(scene: dict) -> float:
    # The runner-up feature is an axis-aligned run. Report the sole diagonal's
    # angular separation from the nearest axis in degrees.
    points = [tuple(map(float, point)) for point in scene["points"]]
    dx = abs(points[3][0] - points[2][0])
    dy = abs(points[3][1] - points[2][1])
    return round(math.degrees(math.atan2(min(dx, dy), max(dx, dy))), 6)


def is_quarantined(scene: dict) -> bool:
    points = [tuple(map(int, point)) for point in scene["points"]]
    lengths = [math.dist(a, b) for a, b in zip(points, points[1:])]
    edge = min(
        min(x, y, int(scene["width"]) - 1 - x, int(scene["height"]) - 1 - y)
        for x, y in points
    )
    return margin(scene) < 12.0 or min(lengths) < 16.0 or edge < 24


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    transformed = dict(scene)
    transformed["route_color"] = str(scene["route_color"]).upper()
    return [("hex_case_alias", transformed)]
