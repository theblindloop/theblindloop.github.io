from __future__ import annotations

from collections import deque
import math

import numpy as np
from PIL import Image

_TRACE = np.array((23, 52, 93), dtype=np.int16)
_START = np.array((24, 165, 88), dtype=np.int16)
_FINISH = np.array((220, 63, 69), dtype=np.int16)
_CHECKPOINT = np.array((242, 183, 47), dtype=np.int16)


def _mask(array: np.ndarray, color: np.ndarray, tolerance: int = 8) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _centroid(mask: np.ndarray, name: str, low: int, high: int) -> tuple[float, float]:
    yy, xx = np.nonzero(mask)
    if not low <= len(xx) <= high:
        raise ValueError(f"missing or malformed {name} marker")
    return float(xx.mean()), float(yy.mean())


def _component_count(mask: np.ndarray) -> int:
    unseen = {tuple(point) for point in np.argwhere(mask).tolist()}
    count = 0
    while unseen:
        count += 1
        stack = [unseen.pop()]
        while stack:
            y, x = stack.pop()
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    neighbor = (y + dy, x + dx)
                    if neighbor in unseen:
                        unseen.remove(neighbor)
                        stack.append(neighbor)
    return count


def _nearest(points: np.ndarray, center: tuple[float, float]) -> tuple[int, int]:
    delta = points.astype(float) - np.array((center[1], center[0]))
    y, x = points[int(np.argmin(np.sum(delta * delta, axis=1)))]
    return int(y), int(x)


def _shortest_path(mask: np.ndarray, start: tuple[int, int], finish: tuple[int, int]):
    queue = deque([start])
    parent = {start: None}
    while queue:
        point = queue.popleft()
        if point == finish:
            break
        y, x = point
        for dy in (-1, 0, 1):
            for dx in (-1, 0, 1):
                if not (dx or dy):
                    continue
                neighbor = (y + dy, x + dx)
                if (0 <= neighbor[0] < mask.shape[0] and 0 <= neighbor[1] < mask.shape[1]
                        and mask[neighbor] and neighbor not in parent):
                    parent[neighbor] = point
                    queue.append(neighbor)
    if finish not in parent:
        raise ValueError("navy centerline is broken")
    ordered = []
    point = finish
    while point is not None:
        ordered.append((float(point[1]), float(point[0])))
        point = parent[point]
    return list(reversed(ordered))


def _runs(indices: list[int]) -> list[list[int]]:
    groups: list[list[int]] = []
    for index in indices:
        if not groups or index > groups[-1][-1] + 2:
            groups.append([index])
        else:
            groups[-1].append(index)
    return groups


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 64 or array.shape[1] < 64:
        raise ValueError("unsupported raster")
    start_center = _centroid(_mask(array, _START), "green", 60, 140)
    finish_center = _centroid(_mask(array, _FINISH), "red", 60, 140)
    checkpoint_center = _centroid(_mask(array, _CHECKPOINT), "yellow", 35, 150)
    trace = _mask(array, _TRACE)
    trace_points = np.argwhere(trace)
    if not 350 <= len(trace_points) <= 1700 or _component_count(trace) != 1:
        raise ValueError("missing, broken, or extraneous navy centerline")
    ordered = _shortest_path(
        trace, _nearest(trace_points, start_center), _nearest(trace_points, finish_center)
    )
    if len(ordered) < 100:
        raise ValueError("route is too short to trace")

    checkpoint_index = min(
        range(len(ordered)),
        key=lambda index: math.dist(ordered[index], checkpoint_center),
    )
    if math.dist(ordered[checkpoint_index], checkpoint_center) > 4.0:
        raise ValueError("checkpoint is not attached to the route")

    # With a 12-pixel chord, a true diagonal supplies a long contiguous band in
    # which both coordinate changes are substantial. A right-angle corner can
    # create only a short band. This derives the feature from pixels and uses no
    # renderer coordinates or latent segment index.
    lag = 12
    diagonal_windows = []
    for index in range(0, len(ordered) - lag):
        dx = abs(ordered[index + lag][0] - ordered[index][0])
        dy = abs(ordered[index + lag][1] - ordered[index][1])
        major = max(dx, dy)
        ratio = min(dx, dy) / major if major else 0.0
        if major >= 8.0 and ratio >= 0.16:
            diagonal_windows.append(index)
    groups = sorted(_runs(diagonal_windows), key=len, reverse=True)
    if not groups or len(groups[0]) < 24:
        raise ValueError("no stable diagonal run")
    if len(groups) > 1 and len(groups[1]) >= 0.7 * len(groups[0]):
        raise ValueError("more than one diagonal candidate")
    diagonal_index = int(round(sum(groups[0]) / len(groups[0]) + lag / 2.0))
    separation = checkpoint_index - diagonal_index
    if abs(separation) < 20:
        raise ValueError("checkpoint/diagonal order is ambiguous")
    return "checkpoint" if separation < 0 else "diagonal"
