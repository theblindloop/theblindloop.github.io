"""Deterministic renderer for aligned XOR cards."""
from __future__ import annotations

import random
import numpy as np
from PIL import Image, ImageDraw

SYMBOLS = ("circle", "triangle", "plus", "X")
W, H = 640, 360
CARD = 250
PATCH = 105
SCALE = 5


def _templates() -> dict[str, np.ndarray]:
    yy, xx = np.mgrid[-10:11, -10:11]
    r = np.sqrt(xx * xx + yy * yy)
    circle = (r >= 7.5) & (r <= 10.2)
    triangle = np.zeros((21, 21), dtype=bool)
    for y in range(21):
        half = y / 2.0
        triangle[y] = (np.abs(np.arange(21) - 10) >= half - 1.35) & (np.abs(np.arange(21) - 10) <= half + 1.35)
    triangle[18:21, 1:20] = True
    plus = (((np.abs(xx) <= 2) & (np.abs(yy) <= 9)) | ((np.abs(yy) <= 2) & (np.abs(xx) <= 9)))
    ex = (((np.abs(xx - yy) <= 1) | (np.abs(xx + yy) <= 1)) & (np.abs(xx) <= 9) & (np.abs(yy) <= 9))
    return {"circle": circle, "triangle": triangle, "plus": plus, "X": ex}


def _expanded(symbol: str) -> np.ndarray:
    return np.kron(_templates()[symbol], np.ones((SCALE, SCALE), dtype=bool))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    symbol = SYMBOLS[int(seed) % len(SYMBOLS)]
    other = [s for s in SYMBOLS if s != symbol]
    return {
        "symbol": symbol,
        "decoy": rng.choice(other),
        "split_seed": rng.randrange(1, 2**31),
        "left_cross_x": rng.randrange(112, 139),
        "left_cross_y": rng.randrange(112, 139),
        "right_cross_x": rng.randrange(112, 139),
        "right_cross_y": rng.randrange(112, 139),
        "swap_cards": bool(rng.randrange(2)),
        "accent": rng.choice(["cyan_a", "cyan_b", "turquoise"]),
    }


def _patterns(scene: dict) -> tuple[np.ndarray, np.ndarray]:
    target = _expanded(scene["symbol"])
    decoy_shape = _expanded(scene["decoy"])
    rng = np.random.default_rng(int(scene["split_seed"]))
    split = rng.random((PATCH, PATCH)) < 0.5
    # Shared structured decoy plus fixed-cardinality texture.  Shared pixels
    # disappear under XOR, while neither individual card cleanly shows target.
    decoy = decoy_shape.copy()
    noise_order = rng.permutation(PATCH * PATCH)
    desired_total_pair_ink = 6200
    target_n = int(target.sum())
    desired_decoy_outside = max(0, (desired_total_pair_ink - target_n) // 2)
    outside = ~target
    decoy &= outside
    current = int(decoy.sum())
    if current < desired_decoy_outside:
        for flat in noise_order:
            y, x = divmod(int(flat), PATCH)
            if outside[y, x] and not decoy[y, x]:
                decoy[y, x] = True
                current += 1
                if current >= desired_decoy_outside:
                    break
    elif current > desired_decoy_outside:
        for flat in noise_order:
            y, x = divmod(int(flat), PATCH)
            if decoy[y, x]:
                decoy[y, x] = False
                current -= 1
                if current <= desired_decoy_outside:
                    break
    first = np.logical_xor(decoy, target & split)
    second = np.logical_xor(decoy, target & ~split)
    # split_phase is encoded by swap_cards: toggling it swaps the complementary
    # target shares; page ordering swaps too, yielding a pixel-identical alias.
    if scene["swap_cards"]:
        first, second = second, first
    return first, second


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (W, H), (247, 245, 239))
    draw = ImageDraw.Draw(image)
    patterns = _patterns(scene)
    slots = [(45, 55), (345, 55)]
    crosses = [
        (int(scene["left_cross_x"]), int(scene["left_cross_y"])),
        (int(scene["right_cross_x"]), int(scene["right_cross_y"])),
    ]
    # The cross coordinates travel with card content when swap_cards is toggled.
    if scene["swap_cards"]:
        crosses = crosses[::-1]
    accent = {"cyan_a": (0, 205, 220), "cyan_b": (0, 205, 220), "turquoise": (0, 190, 190)}[scene["accent"]]
    for (sx, sy), (cx, cy), mask in zip(slots, crosses, patterns):
        draw.rectangle((sx, sy, sx + CARD - 1, sy + CARD - 1), fill="white", outline=(80, 80, 80), width=3)
        top = cy - PATCH // 2
        left = cx - PATCH // 2
        ys, xs = np.nonzero(mask)
        for x, y in zip(xs.tolist(), ys.tolist()):
            px, py = sx + left + x, sy + top + y
            draw.point((px, py), fill=(18, 18, 18))
        # Registration cross is overprinted and therefore independently visible.
        ax, ay = sx + cx, sy + cy
        draw.line((ax - 11, ay, ax + 11, ay), fill=accent, width=5)
        draw.line((ax, ay - 11, ax, ay + 11), fill=accent, width=5)
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def _mask_margin(symbol: str) -> float:
    masks = _templates()
    target = masks[symbol].copy()
    target[8:13, :] = False
    target[:, 8:13] = False
    distances = []
    for name, mask in masks.items():
        if name == symbol:
            continue
        comp = mask.copy()
        comp[8:13, :] = False
        comp[:, 8:13] = False
        distances.append(float(np.mean(target != comp)))
    return min(distances)


def margin(scene: dict) -> float:
    return round(_mask_margin(scene["symbol"]), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.08


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    if scene["accent"] not in ("cyan_a", "cyan_b"):
        return []
    twin = dict(scene)
    twin["accent"] = "cyan_b" if scene["accent"] == "cyan_a" else "cyan_a"
    return [("registration_palette_alias", twin)]
