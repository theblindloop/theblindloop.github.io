"""Pixel-only inverse arm.  This module deliberately imports no world code."""
from __future__ import annotations

import numpy as np
from PIL import Image

N = 105


def _templates() -> dict[str, np.ndarray]:
    yy, xx = np.mgrid[-10:11, -10:11]
    radius = np.hypot(xx, yy)
    ring = (radius >= 7.5) & (radius <= 10.2)
    tri = np.zeros((21, 21), bool)
    columns = np.arange(21)
    for row in range(21):
        edge = row / 2.0
        tri[row] = (np.abs(columns - 10) >= edge - 1.35) & (np.abs(columns - 10) <= edge + 1.35)
    tri[18:21, 1:20] = True
    plus_mask = (((abs(xx) <= 2) & (abs(yy) <= 9)) | ((abs(yy) <= 2) & (abs(xx) <= 9)))
    x_mask = (((abs(xx - yy) <= 1) | (abs(xx + yy) <= 1)) & (abs(xx) <= 9) & (abs(yy) <= 9))
    block = np.ones((5, 5), bool)
    return {"circle": np.kron(ring, block), "triangle": np.kron(tri, block), "plus": np.kron(plus_mask, block), "X": np.kron(x_mask, block)}


def _cross_centers(rgb: np.ndarray) -> list[tuple[int, int]]:
    # All allowed registration colors have low red and strong green/blue.
    cyan = (rgb[:, :, 0] < 40) & (rgb[:, :, 1] > 160) & (rgb[:, :, 2] > 160)
    centers = []
    for lo, hi in ((0, rgb.shape[1] // 2), (rgb.shape[1] // 2, rgb.shape[1])):
        ys, xs = np.nonzero(cyan[:, lo:hi])
        if len(xs) < 40:
            raise ValueError("missing registration cross")
        centers.append((int(round(float(xs.mean() + lo))), int(round(float(ys.mean())))))
    return centers


def _extract(black: np.ndarray, center: tuple[int, int]) -> np.ndarray:
    cx, cy = center
    half = N // 2
    patch = black[cy-half:cy+half+1, cx-half:cx+half+1]
    if patch.shape != (N, N):
        raise ValueError("registration crop leaves image")
    return patch


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"))
    centers = _cross_centers(rgb)
    black = np.max(rgb, axis=2) < 80
    revealed = np.logical_xor(_extract(black, centers[0]), _extract(black, centers[1]))
    # Ignore the visibly overprinted cross footprint in both observation/template.
    valid = np.ones((N, N), bool)
    c = N // 2
    valid[c-3:c+4, :] = False
    valid[:, c-3:c+4] = False
    scores = {name: float(np.mean(revealed[valid] == mask[valid])) for name, mask in _templates().items()}
    ordered = sorted(scores, key=lambda name: (-scores[name], name))
    if scores[ordered[0]] - scores[ordered[1]] < 0.08:
        raise ValueError("ambiguous XOR symbol")
    return ordered[0]
