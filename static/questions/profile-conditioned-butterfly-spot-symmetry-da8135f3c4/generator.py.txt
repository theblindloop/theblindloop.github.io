"""Scene sampler, renderer and analytic gold for `butterfly_spot_symmetry`.

The page shows one butterfly.  Its two wings are drawn as an exact mirror pair:
the left half of the wing ink is painted once and its mirror image is pasted on
the other side of the body, so wing shape carries no information at all.  Each
wing then receives one round spot of the same colour and the same radius.  The
only asymmetry a page can hold is how far each spot sits from the centre line
of the body, and the decision is which spot sits farther out.

The canonical prior -- a butterfly's wing markings are mirror images of each
other -- is painted by the `even` class and contradicted by the `left` and
`right` classes with identical wording, palette, wing shape, spot size, spot
height, ink total and page furniture.  The prior is recorded as a diagnostic and
is never gold.  Both spot distances are drawn from one sampled pair
`(base + gap, base - gap)`; the class only decides which side receives which, so
the sum of the two distances, the total coloured ink and every other cheap cue
are independent of the answer by construction.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageOps

RENDERER_VERSION = "butterfly_spot_symmetry-renderer-0.1.0"

DECISIONS = ("left", "right", "even")

EVEN_MAX = 6.0          # a difference this small or smaller is "same distance"
QUARANTINE_PX = 4.0     # nearer than this to the boundary: quarantined

HALF_W, HALF_H = 210, 300

INKS = (
    (214, 74, 46),
    (46, 116, 208),
    (126, 74, 190),
    (28, 150, 110),
    (206, 150, 20),
    (190, 60, 140),
)
BODY_INKS = ((38, 38, 44), (52, 44, 40), (40, 48, 44), (46, 40, 50))
GREY = (152, 152, 152)

SCENE_FIELDS = (
    "width", "height", "cx", "cy",
    "upper_dx", "upper_dy", "upper_rx", "upper_ry",
    "lower_dx", "lower_dy", "lower_rx", "lower_ry",
    "body_w", "body_h", "ant_len",
    "wing_ink", "spot_ink", "body_ink",
    "spot_r", "spot_dy", "k_left", "k_right",
    "frame_inset", "tick_len", "deco",
)


# --------------------------------------------------------------------------- #
# sampling
# --------------------------------------------------------------------------- #
def sample_scene(seed: int) -> dict:
    for attempt in range(64):
        try:
            return _sample_once(seed * 1009 + attempt)
        except ValueError:
            continue
    raise RuntimeError(f"could not sample a scene for seed {seed}")


def _sample_once(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    width, height = 640, 480
    cx = rng.randint(296, 344)
    cy = rng.randint(232, 250)

    upper_rx = rng.randint(74, 86)
    upper_ry = rng.randint(58, 68)
    upper_dx = rng.randint(58, 70)
    upper_dy = rng.randint(60, 72)
    lower_rx = rng.randint(50, 60)
    lower_ry = rng.randint(44, 52)
    lower_dx = rng.randint(40, 50)
    lower_dy = rng.randint(50, 62)

    body_w = rng.randint(9, 12)
    body_h = rng.randint(98, 114)
    ant_len = rng.randint(26, 34)

    wi = rng.randrange(len(INKS))
    si = (wi + rng.randrange(1, len(INKS))) % len(INKS)
    wing_ink = INKS[wi]
    spot_ink = INKS[si]
    body_ink = BODY_INKS[rng.randrange(len(BODY_INKS))]

    spot_r = rng.randint(11, 14)
    spot_dy = upper_dy + rng.randint(-5, 5)      # height above cy
    base = upper_dx + rng.randint(-6, 6)         # mean distance from centre line
    gap = rng.randint(9, 17)                     # half of the drawn difference

    decision = DECISIONS[rng.randrange(3)]
    if decision == "even":
        k_left = k_right = base
    elif decision == "left":
        k_left, k_right = base + gap, base - gap
    else:
        k_left, k_right = base - gap, base + gap

    scene = {
        "width": width, "height": height, "cx": cx, "cy": cy,
        "upper_dx": upper_dx, "upper_dy": upper_dy,
        "upper_rx": upper_rx, "upper_ry": upper_ry,
        "lower_dx": lower_dx, "lower_dy": lower_dy,
        "lower_rx": lower_rx, "lower_ry": lower_ry,
        "body_w": body_w, "body_h": body_h, "ant_len": ant_len,
        "wing_ink": list(wing_ink), "spot_ink": list(spot_ink),
        "body_ink": list(body_ink),
        "spot_r": spot_r, "spot_dy": spot_dy,
        "k_left": k_left, "k_right": k_right,
        "frame_inset": rng.randint(12, 20),
        "tick_len": rng.randint(14, 22),
        "deco": [
            [rng.randint(60, 580), rng.randint(36, 60), rng.randint(5, 8)]
            for _ in range(rng.randint(2, 4))
        ],
    }
    _assert_spots_on_wing(scene)
    return scene


def _assert_spots_on_wing(scene: dict) -> None:
    """Both spots sit well inside the upper wing and clear of the body."""
    rx = scene["upper_rx"] - scene["spot_r"] - 5.0
    ry = scene["upper_ry"] - scene["spot_r"] - 5.0
    for k in (scene["k_left"], scene["k_right"]):
        d = k + 0.5
        u = ((d - scene["upper_dx"]) / rx) ** 2 + (
            (scene["spot_dy"] - scene["upper_dy"]) / ry
        ) ** 2
        if u >= 1.0:
            raise ValueError("spot would fall off the wing")
        if d - scene["spot_r"] <= scene["body_w"] + 4:
            raise ValueError("spot would touch the body")


# --------------------------------------------------------------------------- #
# rendering
# --------------------------------------------------------------------------- #
def render(scene: dict) -> Image.Image:
    w, h = int(scene["width"]), int(scene["height"])
    cx, cy = int(scene["cx"]), int(scene["cy"])
    img = Image.new("RGB", (w, h), (255, 255, 255))
    d = ImageDraw.Draw(img)

    # colourless page furniture
    inset = int(scene["frame_inset"])
    d.rectangle([inset, inset, w - 1 - inset, h - 1 - inset], outline=GREY, width=2)
    t = int(scene["tick_len"])
    for (x0, y0, dx, dy) in (
        (inset + 6, inset + 6, 1, 0), (inset + 6, inset + 6, 0, 1),
        (w - 1 - inset - 6, h - 1 - inset - 6, -1, 0),
        (w - 1 - inset - 6, h - 1 - inset - 6, 0, -1),
    ):
        d.line([x0, y0, x0 + dx * t, y0 + dy * t], fill=GREY, width=2)
    for (dx, dy, r) in scene["deco"]:
        d.polygon(
            [(dx, dy - r), (dx + r, dy), (dx, dy + r), (dx - r, dy)], fill=GREY
        )

    # wings: paint one half, then paste it and its mirror around x = cx - 0.5
    half = Image.new("RGBA", (HALF_W, HALF_H), (0, 0, 0, 0))
    dh = ImageDraw.Draw(half)
    mid = HALF_H // 2
    wing = tuple(scene["wing_ink"])
    for (dx, dy, rx, ry) in (
        (scene["upper_dx"], scene["upper_dy"], scene["upper_rx"], scene["upper_ry"]),
        (scene["lower_dx"], -scene["lower_dy"], scene["lower_rx"], scene["lower_ry"]),
    ):
        ux = HALF_W - int(dx)
        uy = mid - int(dy)
        dh.ellipse([ux - int(rx), uy - int(ry), ux + int(rx), uy + int(ry)], fill=wing)
    img.paste(half, (cx - HALF_W, cy - mid), half)
    flipped = ImageOps.mirror(half)
    img.paste(flipped, (cx, cy - mid), flipped)

    # body and antennae, drawn symmetrically about x = cx - 0.5
    bw, bh = int(scene["body_w"]), int(scene["body_h"])
    d.ellipse([cx - 1 - bw, cy - bh, cx + bw, cy + bh], fill=tuple(scene["body_ink"]))
    a = int(scene["ant_len"])
    d.line([cx - 1, cy - bh, cx - 1 - a, cy - bh - a], fill=GREY, width=3)
    d.line([cx, cy - bh, cx + a, cy - bh - a], fill=GREY, width=3)

    # the two wing spots
    r = int(scene["spot_r"])
    sy = cy - int(scene["spot_dy"])
    spot = tuple(scene["spot_ink"])
    xl = cx - 1 - int(scene["k_left"])
    xr = cx + int(scene["k_right"])
    d.ellipse([xl - r, sy - r, xl + r, sy + r], fill=spot)
    d.ellipse([xr - r, sy - r, xr + r, sy + r], fill=spot)
    return img


# --------------------------------------------------------------------------- #
# analytic gold
# --------------------------------------------------------------------------- #
def delta(scene: dict) -> float:
    """Left spot distance minus right spot distance, in canvas pixels."""
    return float(int(scene["k_left"]) - int(scene["k_right"]))


def analytic_gold(scene: dict) -> str:
    dl = delta(scene)
    if abs(dl) <= EVEN_MAX:
        return "even"
    return "left" if dl > 0 else "right"


def margin(scene: dict) -> float:
    """Pixels the drawn difference must move before the answer flips."""
    return float(abs(abs(delta(scene)) - EVEN_MAX))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def runner_up(scene: dict) -> str:
    dl = delta(scene)
    if abs(dl) <= EVEN_MAX:
        return "left" if dl > 0 else "right" if dl < 0 else "left"
    return "even"


def variant(scene: dict) -> str:
    return "canonical" if analytic_gold(scene) == "even" else "counterfactual"


def prior_answer(scene: dict) -> str:
    """The canonical prior always says the markings mirror each other."""
    return "even"


def latent_symmetries(scene: dict) -> list:
    """No latent aliases are declared: every field is painted into the raster."""
    return []


def ink_pixels(image: Image.Image) -> int:
    arr = image.convert("RGB")
    return sum(1 for px in arr.getdata() if px != (255, 255, 255))


def spot_sum(scene: dict) -> int:
    return int(scene["k_left"]) + int(scene["k_right"])


# --------------------------------------------------------------------------- #
# declared edits used by the verifier and the tests
# --------------------------------------------------------------------------- #
def translate(scene: dict, dx: float, dy: float) -> dict:
    alt = dict(scene)
    alt["cx"] = int(scene["cx"] + dx)
    alt["cy"] = int(scene["cy"] + dy)
    return alt


def recolor(scene: dict) -> dict:
    alt = dict(scene)
    wi = INKS.index(tuple(scene["wing_ink"]))
    si = INKS.index(tuple(scene["spot_ink"]))
    alt["wing_ink"] = list(INKS[(wi + 2) % len(INKS)])
    alt["spot_ink"] = list(INKS[(si + 3) % len(INKS)])
    if alt["wing_ink"] == alt["spot_ink"]:
        alt["spot_ink"] = list(INKS[(si + 4) % len(INKS)])
    bi = BODY_INKS.index(tuple(scene["body_ink"]))
    alt["body_ink"] = list(BODY_INKS[(bi + 1) % len(BODY_INKS)])
    return alt


def restyle(scene: dict, seed: int) -> dict:
    """Permute the colourless decoration only."""
    rng = random.Random(seed * 31 + 7)
    alt = dict(scene)
    alt["frame_inset"] = rng.randint(12, 20)
    alt["tick_len"] = rng.randint(14, 22)
    alt["deco"] = [
        [rng.randint(60, 580), rng.randint(36, 60), rng.randint(5, 8)]
        for _ in range(rng.randint(2, 4))
    ]
    alt["ant_len"] = rng.randint(26, 34)
    return alt


def rescale(scene: dict) -> dict:
    """Grow the wings and both spots together: the answer is untouched."""
    alt = dict(scene)
    alt["upper_rx"] = int(scene["upper_rx"]) + 5
    alt["upper_ry"] = int(scene["upper_ry"]) + 4
    alt["lower_rx"] = int(scene["lower_rx"]) + 4
    alt["lower_ry"] = int(scene["lower_ry"]) + 3
    alt["spot_r"] = int(scene["spot_r"]) + 2
    _assert_spots_on_wing(alt)
    return alt


def _base_gap(scene: dict) -> tuple[int, int]:
    kl, kr = int(scene["k_left"]), int(scene["k_right"])
    base = (kl + kr) // 2
    gap = abs(kl - kr) // 2
    if gap < 9:
        gap = 9
    return base, gap


def set_class(scene: dict, target: str) -> dict:
    """Repaint the same pair of distances as the requested class."""
    if target not in DECISIONS:
        raise ValueError(f"unknown class: {target}")
    base, gap = _base_gap(scene)
    alt = dict(scene)
    if target == "even":
        alt["k_left"] = alt["k_right"] = base
    elif target == "left":
        alt["k_left"], alt["k_right"] = base + gap, base - gap
    else:
        alt["k_left"], alt["k_right"] = base - gap, base + gap
    _assert_spots_on_wing(alt)
    return alt


def mirror_class(decision: str) -> str:
    """Mirroring the page left to right swaps the two sides."""
    if decision == "left":
        return "right"
    if decision == "right":
        return "left"
    return decision


def collapse_margin(scene: dict) -> dict:
    """Draw the two distances exactly on the painted boundary."""
    base, _ = _base_gap(scene)
    alt = dict(scene)
    alt["k_left"] = base + 3
    alt["k_right"] = base - 3
    _assert_spots_on_wing(alt)
    return alt
