"""Pixel-only inverse arm for `butterfly_spot_symmetry`.

Receives a PNG and nothing else: no latent scene, no renderer, no prompts, no
analytic gold.  It groups the page into flat colour classes, takes the dark
tall narrow class as the body and its horizontal centre as the centre line,
takes the smaller of the two saturated classes as the pair of wing spots,
splits those spot pixels at the centre line, and reports which side's spot
centre lies farther from that line.  It abstains whenever the body is not a
well formed bar, the spot class is not two round marks of equal size one per
side, or the two distances land in the painted borderline band.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "butterfly_spot_symmetry-oracle-0.1.0"

ABSTAIN = "abstain"

MIN_CLASS_PX = 120        # a colour class smaller than this is stray ink
CHROMA_MIN = 40           # saturated wing / spot ink versus grey furniture
BODY_ASPECT_MIN = 3.0     # the body bar is far taller than it is wide
BODY_MIN_PX = 400
SPOT_MIN_PX = 200
SPOT_BALANCE = 0.90       # the two spots are the same size
SPOT_ASPECT_LO, SPOT_ASPECT_HI = 0.84, 1.20
SPOT_FILL_LO, SPOT_FILL_HI = 0.66, 0.90   # a disc fills about pi/4 of its box
EVEN_MAX = 6.0            # declared: equal, or apart by at least 16 px
ABSTAIN_BAND = 4.0        # closer than this to the painted boundary: abstain


def _classes(arr: np.ndarray) -> list[dict]:
    codes = (
        arr[:, :, 0].astype(np.int64) * 65536
        + arr[:, :, 1].astype(np.int64) * 256
        + arr[:, :, 2].astype(np.int64)
    )
    values, counts = np.unique(codes, return_counts=True)
    out: list[dict] = []
    for value, count in zip(values.tolist(), counts.tolist()):
        if count < MIN_CLASS_PX:
            continue
        r, g, b = (value >> 16) & 255, (value >> 8) & 255, value & 255
        if r > 245 and g > 245 and b > 245:
            continue  # the page
        ys, xs = np.nonzero(codes == value)
        out.append(
            {
                "count": int(count),
                "chroma": int(max(r, g, b) - min(r, g, b)),
                "luma": int(r) * 299 + int(g) * 587 + int(b) * 114,
                "xs": xs,
                "ys": ys,
            }
        )
    return out


def _round_mark(xs: np.ndarray, ys: np.ndarray) -> bool:
    w = float(xs.max() - xs.min() + 1)
    h = float(ys.max() - ys.min() + 1)
    if w < 8.0 or h < 8.0:
        return False
    if not (SPOT_ASPECT_LO <= w / h <= SPOT_ASPECT_HI):
        return False
    fill = float(xs.size) / (w * h)
    return SPOT_FILL_LO <= fill <= SPOT_FILL_HI


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    classes = _classes(arr)
    if len(classes) < 3:
        return ABSTAIN

    grey = [c for c in classes if c["chroma"] < CHROMA_MIN]
    saturated = [c for c in classes if c["chroma"] >= CHROMA_MIN]
    if not grey or len(saturated) != 2:
        return ABSTAIN

    body = min(grey, key=lambda c: c["luma"])
    if body["count"] < BODY_MIN_PX:
        return ABSTAIN
    bw = float(body["xs"].max() - body["xs"].min() + 1)
    bh = float(body["ys"].max() - body["ys"].min() + 1)
    if bw <= 0 or bh / bw < BODY_ASPECT_MIN:
        return ABSTAIN
    axis = float(body["xs"].mean())

    wing = max(saturated, key=lambda c: c["count"])
    spot = min(saturated, key=lambda c: c["count"])
    if spot["count"] >= wing["count"]:
        return ABSTAIN

    xs, ys = spot["xs"].astype(np.float64), spot["ys"]
    left = xs < axis
    right = xs > axis
    nl, nr = int(left.sum()), int(right.sum())
    if nl < SPOT_MIN_PX or nr < SPOT_MIN_PX:
        return ABSTAIN
    if min(nl, nr) / float(max(nl, nr)) < SPOT_BALANCE:
        return ABSTAIN
    if not _round_mark(xs[left], ys[left]) or not _round_mark(xs[right], ys[right]):
        return ABSTAIN

    d_left = axis - float(xs[left].mean())
    d_right = float(xs[right].mean()) - axis
    delta = d_left - d_right
    if abs(abs(delta) - EVEN_MAX) < ABSTAIN_BAND:
        return ABSTAIN
    if abs(delta) <= EVEN_MAX:
        return "even"
    return "left" if delta > 0 else "right"
