"""Fork-branch tracing world: which coloured terminal ends the LONGER branch.

A single connected blue network is drawn. It begins at an orange ring on the
left, runs along a shared trunk to a junction (fork) in the middle, and then
splits into two mutually exclusive branches. The upper branch ends in one
coloured terminal dot and the lower branch ends in the other. The two coloured
terminals are drawn from {red, green}; which colour sits on which branch, and
which branch is the longer one along the stroke, are both randomised.

The question identifies the fork by construction and asks which coloured
terminal sits at the far end of the LONGER branch, i.e. the branch whose on-in
stroke length (from the junction to its terminal end) is the greater. Because
both branches share the same trunk up to the junction, comparing the on-stroke
(junction-to-terminal) branch lengths is equivalent to comparing the geodesic
distance from the ring to each terminal. The branches are serpentines that fold
back and forth, so the longer branch need not be the one with the greater
straight-line (Euclidean) reach; the answer is a branch-geodesic fact that must
be recovered by tracing the shared trunk to the fork and then ranking the two
branch strokes, not by local proximity or extent.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "fork_branch_length-0.1.0"

CANVAS = (480, 480)
BG = (255, 255, 255)
COLOR_NET = (35, 95, 210)
COLOR_RING = (250, 150, 40)
COLOR_RED = (225, 40, 40)
COLOR_GREEN = (35, 165, 75)

STROKE = 18
TERM_R = 6
PAD = 40
MIN_MARGIN = 60.0
BRANCH_CLEAR = 28.0

TERMINAL_COLORS = ["red", "green"]


def _pts(points: Any) -> list[tuple[float, float]]:
    return [(float(p[0]), float(p[1])) for p in points]


def _path_length(points: list[tuple[float, float]]) -> float:
    return sum(math.dist(a, b) for a, b in zip(points, points[1:]))


def _seg_dist(p, a, b):
    ax, ay = a
    bx, by = b
    px, py = p
    dx, dy = bx - ax, by - ay
    if dx == 0 and dy == 0:
        return math.dist(p, a)
    t = ((px - ax) * dx + (py - ay) * dy) / (dx * dx + dy * dy)
    t = max(0.0, min(1.0, t))
    return math.hypot(px - (ax + t * dx), py - (ay + t * dy))


def _dist_to_polyline(p, pts):
    best = float("inf")
    for a, b in zip(pts, pts[1:]):
        best = min(best, _seg_dist(p, a, b))
    return best


def _branch_upper(rng: random.Random, fx: float) -> list:
    w, h = CANVAS
    x_lo = fx + 8.0
    x_hi = float(w - PAD - 10)
    n_lanes = rng.randint(3, 5)
    y_top = float(PAD + 18)
    y_bot = float(218.0)
    step = (y_bot - y_top) / (n_lanes - 1) if n_lanes > 1 else 0.0
    pts = [(float(fx), 240.0)]
    for lane in range(n_lanes):
        y = y_bot - lane * step
        if lane % 2 == 0:
            pts.append((x_lo, y))
            pts.append((x_hi, y))
        else:
            pts.append((x_hi, y))
            pts.append((x_lo, y))
    return pts


def _branch_lower(rng: random.Random, fx: float) -> list:
    w, h = CANVAS
    x_lo = fx + 8.0
    x_hi = float(w - PAD - 10)
    n_lanes = rng.randint(3, 5)
    y_top = float(262.0)
    y_bot = float(h - PAD - 18)
    step = (y_bot - y_top) / (n_lanes - 1) if n_lanes > 1 else 0.0
    pts = [(float(fx), 240.0)]
    for lane in range(n_lanes):
        y = y_top + lane * step
        if lane % 2 == 0:
            pts.append((x_lo, y))
            pts.append((x_hi, y))
        else:
            pts.append((x_hi, y))
            pts.append((x_lo, y))
    return pts


def _sample_scene(rng: random.Random) -> dict:
    w, h = CANVAS
    for _ in range(40000):
        fx = rng.uniform(150.0, 230.0)
        rx = rng.uniform(55.0, 105.0)
        ring = [float(rx), 240.0]
        fork = [float(fx), 240.0]
        trunk = [ring, fork]
        try:
            up = _branch_upper(rng, fx)
            lo = _branch_lower(rng, fx)
        except Exception:
            continue
        dist_a = _path_length(_pts(trunk)) + _path_length(_pts(up))
        dist_b = _path_length(_pts(trunk)) + _path_length(_pts(lo))
        if abs(dist_a - dist_b) < MIN_MARGIN:
            continue
        longer = "a" if dist_a > dist_b else "b"
        term_a = up[-1]
        term_b = lo[-1]
        # shuffle which colour is on the upper / lower branch
        colors = list(TERMINAL_COLORS)
        rng.shuffle(colors)
        color_a, color_b = colors[0], colors[1]
        terminals = [
            {"color": color_a, "pos": [float(term_a[0]), float(term_a[1])], "branch": "a"},
            {"color": color_b, "pos": [float(term_b[0]), float(term_b[1])], "branch": "b"},
        ]
        answer = color_a if longer == "a" else color_b

        # validity checks
        ok = True
        positions = {"a": [float(term_a[0]), float(term_a[1])],
                     "b": [float(term_b[0]), float(term_b[1])]}
        for key, pos in positions.items():
            if not (20 <= pos[0] <= w - 20 and 20 <= pos[1] <= h - 20):
                ok = False
                break
        if not ok:
            continue
        # terminal clear of the other branch's ink and of the fork/trunk
        pa, pb = positions["a"], positions["b"]
        clear_a = min(
            _dist_to_polyline(pa, _pts(lo)),
            _dist_to_polyline(pa, _pts(trunk)),
            math.dist(pa, fork),
        )
        clear_b = min(
            _dist_to_polyline(pb, _pts(up)),
            _dist_to_polyline(pb, _pts(trunk)),
            math.dist(pb, fork),
        )
        if clear_a < BRANCH_CLEAR or clear_b < BRANCH_CLEAR:
            continue
        network = [
            {"seg": "trunk", "pts": trunk},
            {"seg": "branch_a", "pts": up},
            {"seg": "branch_b", "pts": lo},
        ]
        scene = {
            "network": network,
            "ring": ring,
            "fork": fork,
            "terminals": terminals,
            "answer": answer,
        }
        return scene
    raise RuntimeError("could not satisfy fork-branch constraints")


def sample_scene(seed: int) -> dict:
    return _sample_scene(random.Random(seed))


def _draw_polyline(draw: ImageDraw.ImageDraw, pts, width):
    for a, b in zip(pts, pts[1:]):
        draw.line([a, b], fill=COLOR_NET, width=width, joint="curve")


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", CANVAS, BG)
    draw = ImageDraw.Draw(img)
    for route in scene["network"]:
        _draw_polyline(draw, _pts(route["pts"]), STROKE)
    term_rgb = {"red": COLOR_RED, "green": COLOR_GREEN}
    rx, ry = float(scene["ring"][0]), float(scene["ring"][1])
    rr = 9
    draw.ellipse([rx - rr, ry - rr, rx + rr, ry + rr], fill=COLOR_RING)
    for term in scene["terminals"]:
        tx, ty = float(term["pos"][0]), float(term["pos"][1])
        draw.ellipse([tx - TERM_R, ty - TERM_R, tx + TERM_R, ty + TERM_R],
                     fill=term_rgb[term["color"]])
    return img


def analytic_gold(scene: dict) -> str:
    """Colour of the terminal at the end of the longer branch."""
    branches = {r["seg"]: r for r in scene["network"] if r["seg"] != "trunk"}
    trunk = _pts(scene["network"][0]["pts"])
    dist_a = _path_length(trunk) + _path_length(_pts(branches["branch_a"]["pts"]))
    dist_b = _path_length(trunk) + _path_length(_pts(branches["branch_b"]["pts"]))
    longer = "a" if dist_a > dist_b else "b"
    term = scene["terminals"]
    for t in term:
        if t["branch"] == longer:
            return t["color"]
    raise RuntimeError("longer branch has no terminal")


def margin(scene: dict) -> float:
    """Geodesic (on-stroke) difference between ring-to-terminal distances in px."""
    branches = {r["seg"]: r for r in scene["network"] if r["seg"] != "trunk"}
    trunk = _pts(scene["network"][0]["pts"])
    dist_a = _path_length(trunk) + _path_length(_pts(branches["branch_a"]["pts"]))
    dist_b = _path_length(trunk) + _path_length(_pts(branches["branch_b"]["pts"]))
    return float(abs(dist_a - dist_b))


def is_quarantined(scene: dict) -> bool:
    w, h = CANVAS
    if margin(scene) < MIN_MARGIN:
        return True
    branches = {r["seg"]: r for r in scene["network"] if r["seg"] != "trunk"}
    up = _pts(branches["branch_a"]["pts"])
    lo = _pts(branches["branch_b"]["pts"])
    trunk = _pts(scene["network"][0]["pts"])
    fork = (float(scene["fork"][0]), float(scene["fork"][1]))
    for term in scene["terminals"]:
        pos = (float(term["pos"][0]), float(term["pos"][1]))
        if not (20 <= pos[0] <= w - 20 and 20 <= pos[1] <= h - 20):
            return True
        other = up if term["branch"] == "b" else lo
        if min(_dist_to_polyline(pos, other),
               _dist_to_polyline(pos, trunk), math.dist(pos, fork)) < BRANCH_CLEAR:
            return True
    return False


def latent_symmetries(scene: dict) -> list:
    return []
