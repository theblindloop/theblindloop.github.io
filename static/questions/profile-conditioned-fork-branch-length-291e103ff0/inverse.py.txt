"""Pixel-only inverse arm for the fork-branch-length world.

Reads only the final RGB PNG. It segments the rendered palette (blue network,
orange ring, red/green terminal dots, white background). It measures, along the
blue ink alone, the geodesic (on-stroke) distance from the orange ring to each
coloured terminal using a breadth-first wave over the blue pixels, and returns
the colour of the terminal whose on-stroke distance from the ring is the
greater. Because both branches share the same trunk up to the junction, the
greater ring-to-terminal geodesic distance identifies the longer branch.

Imports only numpy/PIL; it never reads renderer, prompt, generator, verifier or
gold code and it never receives latent scene data.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_PALETTE = {
    "net": (35, 95, 210),
    "ring": (250, 150, 40),
    "red": (225, 40, 40),
    "green": (35, 165, 75),
    "white": (255, 255, 255),
}
_NAMES = ["net", "ring", "red", "green", "white"]


def _label_pixels(arr: np.ndarray) -> np.ndarray:
    palette = np.array([*_PALETTE.values()], dtype=np.float32)
    dists = np.stack([np.linalg.norm(arr - c, axis=-1) for c in palette], axis=-1)
    return dists.argmin(axis=-1)


def _geodesic_distances(blue: np.ndarray, ring: np.ndarray) -> np.ndarray:
    """8-connected BFS distance over blue pixels seeded from blue near the ring."""
    h, w = blue.shape
    dist = np.full((h, w), -1, dtype=np.int32)
    if not blue.any() or not ring.any():
        return dist
    seed = np.zeros_like(blue)
    for dy in (-1, 0, 1):
        for dx in (-1, 0, 1):
            seed |= np.roll(ring, (dy, dx), axis=(0, 1))
    seed = seed & blue
    frontier = list(zip(*np.nonzero(seed)))
    d = 0
    for (y, x) in frontier:
        dist[y, x] = d
    while frontier:
        d += 1
        nxt = set()
        for (y, x) in frontier:
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dy == 0 and dx == 0:
                        continue
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < h and 0 <= nx < w and blue[ny, nx] and dist[ny, nx] == -1:
                        dist[ny, nx] = d
                        nxt.add((ny, nx))
        frontier = list(nxt)
    return dist


def _terminal_distance(term_mask: np.ndarray, dist: np.ndarray, radius: int = 12) -> int:
    """Max geodesic distance among blue pixels near the terminal centroid."""
    ys, xs = np.nonzero(term_mask)
    if len(ys) == 0:
        raise ValueError("terminal dot not found")
    cy, cx = int(np.mean(ys)), int(np.mean(xs))
    h, w = dist.shape
    y0, y1 = max(0, cy - radius), min(h, cy + radius + 1)
    x0, x1 = max(0, cx - radius), min(w, cx + radius + 1)
    window = dist[y0:y1, x0:x1]
    valid = window[window >= 0]
    if len(valid) == 0:
        raise ValueError("terminal has no neighbouring blue ink")
    return int(valid.max())


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label_pixels(arr)
    idx = {name: i for i, name in enumerate(_NAMES)}
    blue = labels == idx["net"]
    ring = labels == idx["ring"]
    if not blue.any() or not ring.any():
        raise ValueError("network or ring not found")
    dist = _geodesic_distances(blue, ring)
    da = _terminal_distance(labels == idx["red"], dist)
    db = _terminal_distance(labels == idx["green"], dist)
    if da == db:
        raise ValueError("inconclusive on-stroke tie between terminals")
    return "red" if da > db else "green"
