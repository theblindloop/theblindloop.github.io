"""Independent pixel-only inverse for directed checkpoint order."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

FRAME = (190, 198, 210)
INK = (35, 39, 48)
START = (36, 166, 99)
FINISH = (222, 72, 72)
CHECKPOINTS = {
    "C": (35, 177, 205),
    "M": (211, 70, 165),
    "O": (238, 138, 42),
    "P": (126, 87, 194),
}


def _components(mask: np.ndarray) -> list[tuple[np.ndarray, np.ndarray]]:
    seen = np.zeros(mask.shape, dtype=bool)
    results = []
    height, width = mask.shape
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        ys, xs = [], []
        while queue:
            cy, cx = queue.popleft()
            ys.append(cy)
            xs.append(cx)
            for ny, nx in ((cy-1, cx), (cy+1, cx), (cy, cx-1), (cy, cx+1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        results.append((np.asarray(ys), np.asarray(xs)))
    return results


def _color_components(array: np.ndarray, color: tuple[int, int, int], minimum: int) -> list[dict]:
    mask = np.all(array == np.asarray(color, dtype=np.uint8), axis=2)
    output = []
    for ys, xs in _components(mask):
        if len(xs) >= minimum:
            output.append({"cx": float(xs.mean()), "cy": float(ys.mean()), "area": len(xs), "bbox": (int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max()))})
    return output


def _inside(point: dict, box: tuple[int, int, int, int]) -> bool:
    x0, y0, x1, y1 = box
    return x0 < point["cx"] < x1 and y0 < point["cy"] < y1


def _joined(a: dict, b: dict, ink: np.ndarray) -> bool:
    dx, dy = b["cx"] - a["cx"], b["cy"] - a["cy"]
    if min(abs(dx), abs(dy)) > 2.0:
        return False
    hits = 0
    for t in np.linspace(0.42, 0.58, 5):
        x = int(round(a["cx"] + t * dx))
        y = int(round(a["cy"] + t * dy))
        y0, y1 = max(0, y-2), min(ink.shape[0], y+3)
        x0, x1 = max(0, x-2), min(ink.shape[1], x+3)
        hits += bool(ink[y0:y1, x0:x1].mean() > 0.45)
    return hits >= 4


def _sequence(vertices: list[dict], ink: np.ndarray) -> str:
    count = len(vertices)
    adjacency = [[] for _ in vertices]
    for i in range(count):
        for j in range(i + 1, count):
            if _joined(vertices[i], vertices[j], ink):
                adjacency[i].append(j)
                adjacency[j].append(i)
    if sorted(len(items) for items in adjacency) != [1, 1, 2, 2, 2, 2]:
        raise ValueError("route pixels do not form a six-node chain")
    order = [0]
    prior = -1
    while len(order) < count:
        choices = [item for item in adjacency[order[-1]] if item != prior]
        if len(choices) != 1:
            raise ValueError("route traversal is ambiguous")
        prior, current = order[-1], choices[0]
        if current in order:
            raise ValueError("route contains a cycle")
        order.append(current)
    if order[-1] != count - 1:
        raise ValueError("route does not end at red marker")
    return "".join(vertices[index]["symbol"] for index in order[1:-1])


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    frames = _color_components(array, FRAME, 500)
    if len(frames) != 4:
        raise ValueError("expected four visible panel frames")
    boxes = [item["bbox"] for item in sorted(frames, key=lambda component: (component["cy"], component["cx"]))]
    starts = _color_components(array, START, 100)
    finishes = _color_components(array, FINISH, 100)
    checkpoints = []
    for symbol, color in CHECKPOINTS.items():
        for component in _color_components(array, color, 100):
            component["symbol"] = symbol
            checkpoints.append(component)
    if len(starts) != 4 or len(finishes) != 4 or len(checkpoints) != 16:
        raise ValueError("missing colored route evidence")
    ink = np.all(array == np.asarray(INK, dtype=np.uint8), axis=2)
    sequences = []
    for box in boxes:
        panel_starts = [item for item in starts if _inside(item, box)]
        panel_finishes = [item for item in finishes if _inside(item, box)]
        panel_checkpoints = [item for item in checkpoints if _inside(item, box)]
        if len(panel_starts) != 1 or len(panel_finishes) != 1 or len(panel_checkpoints) != 4:
            raise ValueError("panel grouping failed")
        start = dict(panel_starts[0], symbol="S")
        finish = dict(panel_finishes[0], symbol="F")
        sequences.append(_sequence([start, *panel_checkpoints, finish], ink))
    winners = [index for index, value in enumerate(sequences[1:]) if value == sequences[0]]
    if len(winners) != 1:
        raise ValueError("pixel evidence has no unique match")
    return ("A", "B", "C")[winners[0]]
