"""Deterministic directed checkpoint-order correspondence world."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 900, 620
BACKGROUND = (248, 249, 252)
FRAME = (190, 198, 210)
INK = (35, 39, 48)
START = (36, 166, 99)
FINISH = (222, 72, 72)
CHECKPOINTS = {
    "C": (35, 177, 205),
    "M": (211, 70, 165),
    "O": (238, 138, 42),
    "P": (126, 87, 194),
}
SAFE_TURNS = ("LLRL", "LLRR", "LRLL", "LRLR", "LRRL", "RLLR", "RLRL", "RLRR", "RRLL", "RRLR")
PANELS = ((325, 34, 575, 274), (35, 344, 285, 584), (325, 344, 575, 584), (615, 344, 865, 584))


def _hamming(a: str, b: str) -> int:
    return sum(x != y for x, y in zip(a, b))


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    symbols = list(CHECKPOINTS)
    rng.shuffle(symbols)
    reference = "".join(symbols)
    permutations = []
    import itertools
    for value in itertools.permutations(reference):
        text = "".join(value)
        if text != reference and _hamming(text, reference) >= 2:
            permutations.append(text)
    rng.shuffle(permutations)
    answer_index = seed % 3
    options = permutations[:2]
    options.insert(answer_index, reference)

    geometries = []
    rotations = []
    lengths = []
    for _ in range(4):
        geometries.append(SAFE_TURNS[rng.randrange(len(SAFE_TURNS))])
        rotations.append(rng.randrange(4))
        values = [32, 36, 40, 44, 48]
        rng.shuffle(values)
        lengths.append(values)
    return {
        "sequences": [reference, *options],
        "geometries": geometries,
        "rotations": rotations,
        "lengths": lengths,
    }


def _points(pattern: str, lengths: Iterable[int], rotation: int) -> list[tuple[float, float]]:
    direction = rotation % 4
    vectors = ((1, 0), (0, 1), (-1, 0), (0, -1))
    x = y = 0.0
    result = [(x, y)]
    values = list(lengths)
    for index, distance in enumerate(values):
        dx, dy = vectors[direction]
        x += dx * distance
        y += dy * distance
        result.append((x, y))
        if index < len(pattern):
            direction = (direction + (1 if pattern[index] == "R" else -1)) % 4
    return result


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _center(points: list[tuple[float, float]], box: tuple[int, int, int, int]) -> list[tuple[int, int]]:
    x0, y0, x1, y1 = box
    min_x, max_x = min(x for x, _ in points), max(x for x, _ in points)
    min_y, max_y = min(y for _, y in points), max(y for _, y in points)
    cx, cy = (min_x + max_x) / 2, (min_y + max_y) / 2
    target_x, target_y = (x0 + x1) / 2, (y0 + y1) / 2 + 12
    return [(round(x - cx + target_x), round(y - cy + target_y)) for x, y in points]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    labels = ("REFERENCE", "A", "B", "C")
    for index, box in enumerate(PANELS):
        draw.rectangle(box, outline=FRAME, width=3)
        label = labels[index]
        font = _font(17 if index == 0 else 21, bold=True)
        bbox = draw.textbbox((0, 0), label, font=font)
        draw.text(((box[0] + box[2] - (bbox[2] - bbox[0])) / 2, box[1] + 10), label, fill=INK, font=font)
        raw = _points(scene["geometries"][index], scene["lengths"][index], scene["rotations"][index])
        points = _center(raw, box)
        draw.line(points, fill=INK, width=9, joint="curve")
        radius = 10
        for (x, y), symbol in zip(points[1:-1], scene["sequences"][index]):
            color = CHECKPOINTS[symbol]
            draw.rounded_rectangle((x-radius, y-radius, x+radius, y+radius), radius=3, fill=color, outline=(255, 255, 255), width=2)
        for (x, y), color in ((points[0], START), (points[-1], FINISH)):
            draw.ellipse((x-radius-1, y-radius-1, x+radius+1, y+radius+1), fill=color, outline=(255, 255, 255), width=2)
    title = "TRACE GREEN → RED  •  MATCH CHECKPOINT COLOR ORDER"
    font = _font(18, bold=True)
    bbox = draw.textbbox((0, 0), title, font=font)
    draw.text(((WIDTH - (bbox[2] - bbox[0])) / 2, 300), title, fill=INK, font=font)
    return image


def analytic_gold(scene: dict) -> str:
    reference = scene["sequences"][0]
    winners = [i for i, value in enumerate(scene["sequences"][1:]) if value == reference]
    if len(winners) != 1:
        raise ValueError("scene does not have exactly one matching sequence")
    return ("A", "B", "C")[winners[0]]


def margin(scene: dict) -> float:
    reference = scene["sequences"][0]
    scores = sorted((4 - _hamming(reference, value) for value in scene["sequences"][1:]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 2.0
    except (KeyError, ValueError, TypeError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = {
        "sequences": [*scene["sequences"]],
        "geometries": [*scene["geometries"]],
        "rotations": [value + 4 for value in scene["rotations"]],
        "lengths": [list(values) for values in scene["lengths"]],
    }
    return [("full_turn_rotation_alias", twin)]
