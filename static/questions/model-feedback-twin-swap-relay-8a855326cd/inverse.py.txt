"""Independent fixed-layout PNG decoder and twin-swap walk executor."""

from __future__ import annotations

from PIL import Image

WIDTH = 1120
HEIGHT = 820
GRID_N = 7
GRID_X = 48
GRID_Y = 112
CELL = 70
GAP = 7
PITCH = CELL + GAP
STEPS = 20
PAIR_COUNT = 10
DARK = (45, 57, 75)
LIGHT = (226, 232, 228)
START = (24, 151, 96)
GOALS = ((35, 112, 180), (224, 132, 31), (155, 80, 166), (207, 63, 67))
PAIR_COLORS = (
    (0, 126, 167), (230, 134, 28), (126, 87, 194), (206, 70, 65), (22, 143, 82),
    (53, 95, 184), (197, 75, 140), (116, 108, 56), (0, 151, 144), (177, 75, 0),
)
DIRECTIONS = ((-1, 0), (0, 1), (1, 0), (0, -1))


class OracleAbstention(ValueError):
    pass


def _near(pixel: tuple[int, ...], color: tuple[int, int, int], tolerance: int = 7) -> bool:
    return len(pixel) >= 3 and max(abs(int(pixel[i]) - color[i]) for i in range(3)) <= tolerance


def _origin(row: int, col: int) -> tuple[int, int]:
    return GRID_X + col * PITCH, GRID_Y + row * PITCH


def _face(pixel: tuple[int, ...]) -> int:
    if _near(pixel, LIGHT):
        return 0
    if _near(pixel, DARK):
        return 1
    raise OracleAbstention(f"unrecognized square shade {pixel[:3]}")


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    if rgb.size != (WIDTH, HEIGHT):
        raise OracleAbstention(f"unexpected image size {rgb.size}")
    px = rgb.load()

    tiles: list[list[int]] = []
    pair_cells: list[list[tuple[int, int]]] = [[] for _ in range(PAIR_COUNT)]
    pair_at: dict[tuple[int, int], int] = {}
    for row in range(GRID_N):
        line: list[int] = []
        for col in range(GRID_N):
            x, y = _origin(row, col)
            line.append(_face(px[x + CELL - 12, y + 12]))
            probe = px[x + 10, y + 20]
            matches = [index for index, color in enumerate(PAIR_COLORS) if _near(probe, color, 4)]
            if len(matches) > 1:
                raise OracleAbstention("ambiguous pair marker")
            if matches:
                index = matches[0]
                pair_cells[index].append((row, col))
                pair_at[(row, col)] = index
        tiles.append(line)
    if any(len(cells) != 2 for cells in pair_cells):
        raise OracleAbstention(f"every letter must occur twice: {[len(cells) for cells in pair_cells]}")

    start_cells: list[tuple[int, int]] = []
    for row in range(GRID_N):
        for col in range(GRID_N):
            x, y = _origin(row, col)
            found = False
            for yy in range(y + 17, y + 54, 4):
                for xx in range(x + 17, x + 54, 4):
                    if _near(px[xx, yy], START, 3):
                        found = True
                        break
                if found:
                    break
            if found:
                start_cells.append((row, col))
    if len(start_cells) != 1:
        raise OracleAbstention(f"expected one start arrow, found {len(start_cells)}")
    row, col = start_cells[0]
    x, y = _origin(row, col)
    cx, cy = x + CELL // 2, y + CELL // 2
    heading_hits = []
    for heading, (dr, dc) in enumerate(DIRECTIONS):
        if _near(px[cx + 19 * dc, cy + 19 * dr], START, 3):
            heading_hits.append(heading)
    if len(heading_hits) != 1:
        raise OracleAbstention(f"ambiguous start heading {heading_hits}")
    heading = heading_hits[0]

    goal_positions: list[tuple[int, int] | None] = [None] * 4
    for grid_row in range(GRID_N):
        for grid_col in range(GRID_N):
            ox, oy = _origin(grid_row, grid_col)
            probe = px[ox + CELL - 17, oy + CELL - 7]
            matches = [index for index, color in enumerate(GOALS) if _near(probe, color, 3)]
            if len(matches) > 1:
                raise OracleAbstention("ambiguous goal badge")
            if matches:
                index = matches[0]
                if goal_positions[index] is not None:
                    raise OracleAbstention(f"duplicate goal-{'ABCD'[index]}")
                goal_positions[index] = (grid_row, grid_col)
    if any(position is None for position in goal_positions) or len(set(goal_positions)) != 4:
        raise OracleAbstention("four distinct goal badges required")

    state = [line[:] for line in tiles]
    for _ in range(STEPS):
        face = state[row][col]
        heading = (heading + (-1 if face else 1)) % 4
        here = (row, col)
        if here in pair_at:
            peers = pair_cells[pair_at[here]]
            target = peers[1] if peers[0] == here else peers[0]
            state[row][col], state[target[0]][target[1]] = (
                state[target[0]][target[1]], state[row][col]
            )
        else:
            state[row][col] ^= 1
        dr, dc = DIRECTIONS[heading]
        row, col = row + dr, col + dc
        if not (0 <= row < GRID_N and 0 <= col < GRID_N):
            raise OracleAbstention("visible program leaves board")
    final = (row, col)
    matches = [index for index, position in enumerate(goal_positions) if position == final]
    if len(matches) != 1:
        raise OracleAbstention("final cursor is not on one goal")
    return f"goal-{'ABCD'[matches[0]]}"
