"""Render three competing directed multiplicative rate paths."""

from __future__ import annotations

from fractions import Fraction
import random

from PIL import Image, ImageDraw

CANVAS = (1200, 760)
ANSWERS = ("amber", "cyan", "violet")
RGB = {
    "background": (255, 255, 255), "panel": (247, 249, 252), "ink": (31, 38, 46),
    "green": (45, 151, 91), "red": (210, 61, 67), "blue": (38, 111, 187),
    "purple": (126, 73, 181), "orange": (218, 103, 44), "teal": (24, 145, 139),
    "amber": (226, 151, 28), "cyan": (18, 157, 181), "violet": (141, 77, 190),
}
ROW_Y = (35, 275, 515)
ROW_H = 210
NODE_X = (85, 425, 765, 1105)
FACTOR_X = (255, 595, 935)
FACTOR_OPTIONS = (
    (2, 3), (3, 2), (3, 4), (4, 3), (4, 5), (5, 4),
    (5, 6), (6, 5), (5, 7), (7, 5), (7, 8), (8, 7),
    (8, 9), (9, 8), (3, 5), (5, 3), (4, 7), (7, 4),
)
DIGIT_SEGMENTS = {
    0: "abcdef", 1: "bc", 2: "abdeg", 3: "abcdg", 4: "bcfg",
    5: "acdfg", 6: "acdefg", 7: "abc", 8: "abcdefg", 9: "abcdfg",
}
BLOCK_FONT = {
    "S": ("1111", "1000", "1000", "1111", "0001", "0001", "1111"),
    "T": ("11111", "00100", "00100", "00100", "00100", "00100", "00100"),
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "R": ("11110", "10001", "10001", "11110", "10100", "10010", "10001"),
    "F": ("11111", "10000", "10000", "11110", "10000", "10000", "10000"),
    "I": ("111", "010", "010", "010", "010", "010", "111"),
    "N": ("10001", "11001", "10101", "10101", "10101", "10011", "10001"),
    "H": ("10001", "10001", "10001", "11111", "10001", "10001", "10001"),
}


def _product(route: list[list[int]] | tuple[tuple[int, int], ...]) -> Fraction:
    value = Fraction(1, 1)
    for numerator, denominator in route:
        if not 2 <= int(numerator) <= 9 or not 2 <= int(denominator) <= 9:
            raise ValueError("factor digits must be in [2, 9]")
        value *= Fraction(int(numerator), int(denominator))
    return value


def _route_pool() -> list[tuple[Fraction, tuple[tuple[int, int], ...]]]:
    by_value: dict[Fraction, tuple[tuple[int, int], ...]] = {}
    for a in FACTOR_OPTIONS:
        for b in FACTOR_OPTIONS:
            for c in FACTOR_OPTIONS:
                if len({a, b, c}) < 3:
                    continue
                route = (a, b, c); value = _product(route)
                if Fraction(3, 4) <= value <= Fraction(13, 4):
                    by_value.setdefault(value, route)
    return sorted((value, route) for value, route in by_value.items())


ROUTE_POOL = _route_pool()
WINDOWS = tuple(
    i for i in range(len(ROUTE_POOL) - 2)
    if ROUTE_POOL[i + 2][0] > ROUTE_POOL[i + 1][0] > ROUTE_POOL[i][0]
    and Fraction(1, 120) <= (ROUTE_POOL[i + 2][0] - ROUTE_POOL[i + 1][0]) / ROUTE_POOL[i + 1][0]
    and (ROUTE_POOL[i + 2][0] - ROUTE_POOL[i][0]) / ROUTE_POOL[i][0] <= Fraction(1, 12)
)


def sample_scene(seed: int) -> dict:
    rng = random.Random((int(seed) ^ 0xC0A7E5) & 0xFFFFFFFFFFFFFFFF)
    base = WINDOWS[int(seed) % len(WINDOWS)]
    routes = [list(map(list, ROUTE_POOL[base + offset][1])) for offset in range(3)]
    rng.shuffle(routes)
    products = [_product(route) for route in routes]
    target_slot = products.index(max(products))
    answer_color = ANSWERS[(int(seed) // 3) % 3]
    remaining = [color for color in ANSWERS if color != answer_color]; rng.shuffle(remaining)
    tag_colors: list[str | None] = [None, None, None]; tag_colors[target_slot] = answer_color
    for slot in range(3):
        if tag_colors[slot] is None:
            tag_colors[slot] = remaining.pop()
    scene = {"route_factors": routes, "tag_colors": tag_colors}
    if is_quarantined(scene):
        raise RuntimeError("constructed scene is quarantined")
    return scene


def _draw_digit(draw: ImageDraw.ImageDraw, digit: int, x: int, y: int) -> None:
    boxes = {
        "a": (x + 4, y, x + 19, y + 4), "b": (x + 19, y + 4, x + 23, y + 21),
        "c": (x + 19, y + 23, x + 23, y + 40), "d": (x + 4, y + 40, x + 19, y + 44),
        "e": (x, y + 23, x + 4, y + 40), "f": (x, y + 4, x + 4, y + 21),
        "g": (x + 4, y + 20, x + 19, y + 24),
    }
    for segment in DIGIT_SEGMENTS[int(digit)]:
        draw.rectangle(boxes[segment], fill=RGB["ink"])


def _draw_factor(draw: ImageDraw.ImageDraw, cx: int, cy: int, pair: list[int]) -> None:
    draw.rounded_rectangle((cx - 61, cy - 36, cx + 61, cy + 36), radius=11,
                           fill=RGB["background"], outline=RGB["ink"], width=4)
    y = cy - 22; _draw_digit(draw, int(pair[0]), cx - 37, y)
    draw.ellipse((cx - 5, cy - 13, cx + 3, cy - 5), fill=RGB["ink"])
    draw.ellipse((cx - 5, cy + 5, cx + 3, cy + 13), fill=RGB["ink"])
    _draw_digit(draw, int(pair[1]), cx + 12, y)


def _block_text(draw: ImageDraw.ImageDraw, text: str, center_x: int, top: int, scale: int = 3) -> None:
    widths = [len(BLOCK_FONT[ch][0]) * scale for ch in text]
    total = sum(widths) + (len(text) - 1) * 2 * scale; x = center_x - total // 2
    for ch, width in zip(text, widths):
        for row, bits in enumerate(BLOCK_FONT[ch]):
            for col, bit in enumerate(bits):
                if bit == "1":
                    draw.rectangle((x + col * scale, top + row * scale,
                                    x + (col + 1) * scale - 1, top + (row + 1) * scale - 1), fill=RGB["ink"])
        x += width + 2 * scale


def _node(draw: ImageDraw.ImageDraw, x: int, y: int, kind: int, fill: tuple[int, int, int]) -> None:
    if kind == 0:
        draw.ellipse((x - 37, y - 37, x + 37, y + 37), fill=fill, outline=RGB["ink"], width=4)
    elif kind == 1:
        pts = ((x, y - 42), (x - 42, y + 32), (x + 42, y + 32)); draw.polygon(pts, fill=fill); draw.line(pts + (pts[0],), fill=RGB["ink"], width=4, joint="curve")
    elif kind == 2:
        draw.rectangle((x - 36, y - 36, x + 36, y + 36), fill=fill, outline=RGB["ink"], width=4)
    else:
        pts = ((x, y - 42), (x - 38, y - 20), (x - 38, y + 20), (x, y + 42), (x + 38, y + 20), (x + 38, y - 20)); draw.polygon(pts, fill=fill); draw.line(pts + (pts[0],), fill=RGB["ink"], width=4, joint="curve")


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS, RGB["background"]); draw = ImageDraw.Draw(image)
    palettes = ((RGB["blue"], RGB["purple"]), (RGB["orange"], RGB["teal"]), (RGB["purple"], RGB["blue"]))
    for row, (y, factors, color) in enumerate(zip(ROW_Y, scene["route_factors"], scene["tag_colors"])):
        draw.rounded_rectangle((18, y, 1182, y + ROW_H), radius=22, fill=RGB["panel"], outline=RGB[str(color)], width=10)
        cy = y + 117
        for edge in range(3):
            x0, x1 = NODE_X[edge] + 44, NODE_X[edge + 1] - 46
            draw.line((x0, cy, x1, cy), fill=RGB["ink"], width=8)
            draw.polygon(((x1, cy), (x1 - 22, cy - 16), (x1 - 22, cy + 16)), fill=RGB["ink"])
        _node(draw, NODE_X[0], cy, 0, RGB["green"])
        _node(draw, NODE_X[1], cy, 1 if row != 1 else 2, palettes[row][0])
        _node(draw, NODE_X[2], cy, 2 if row != 1 else 3, palettes[row][1])
        _node(draw, NODE_X[3], cy, 3, RGB["red"])
        for cx, pair in zip(FACTOR_X, factors): _draw_factor(draw, cx, cy, pair)
        _block_text(draw, "START", NODE_X[0], y + 27, 3); _block_text(draw, "FINISH", NODE_X[3], y + 27, 3)
    return image


def analytic_gold(scene: dict) -> str:
    if len(scene["route_factors"]) != 3 or set(scene["tag_colors"]) != set(ANSWERS):
        raise ValueError("malformed route binding")
    values = [_product(route) for route in scene["route_factors"]]; maximum = max(values)
    matches = [i for i, value in enumerate(values) if value == maximum]
    if len(matches) != 1: raise ValueError("path products do not have a unique maximum")
    return str(scene["tag_colors"][matches[0]])


def margin(scene: dict) -> float:
    values = sorted((_product(route) for route in scene["route_factors"]), reverse=True)
    return float((values[0] - values[1]) / values[1]) if len(values) == 3 and values[0] > values[1] else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < (1 / 120) or analytic_gold(scene) not in ANSWERS
    except (KeyError, TypeError, ValueError, ZeroDivisionError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
