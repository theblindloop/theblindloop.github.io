"""Independent image-only inverse for competing multiplicative rate paths."""

from __future__ import annotations
from fractions import Fraction
import numpy as np
from PIL import Image

CANVAS = (1200, 760)
INK = (31, 38, 46); BACKGROUND = (255, 255, 255); PANEL = (247, 249, 252)
TAG_RGB = {"amber": (226, 151, 28), "cyan": (18, 157, 181), "violet": (141, 77, 190)}
NODE_RGB = ((45, 151, 91), (210, 61, 67), (38, 111, 187), (126, 73, 181), (218, 103, 44), (24, 145, 139))
ALLOWED = {INK, BACKGROUND, PANEL, *TAG_RGB.values(), *NODE_RGB}
ROW_Y = (35, 275, 515); FACTOR_X = (255, 595, 935)
SEGMENT_POINTS = {"a": (11, 2), "b": (21, 12), "c": (21, 32), "d": (11, 42), "e": (2, 32), "f": (2, 12), "g": (11, 22)}
SEGMENT_DIGIT = {
    frozenset("abcdef"): 0, frozenset("bc"): 1, frozenset("abdeg"): 2,
    frozenset("abcdg"): 3, frozenset("bcfg"): 4, frozenset("acdfg"): 5,
    frozenset("acdefg"): 6, frozenset("abc"): 7, frozenset("abcdefg"): 8,
    frozenset("abcdfg"): 9,
}


class OracleError(ValueError):
    pass


def _digit(arr: np.ndarray, x: int, y: int) -> int:
    active = set()
    for name, (dx, dy) in SEGMENT_POINTS.items():
        patch = arr[y + dy - 1:y + dy + 2, x + dx - 1:x + dx + 2]
        if patch.shape != (3, 3, 3): raise OracleError("digit patch outside canvas")
        ink = int(np.all(patch == INK, axis=2).sum())
        if ink >= 7: active.add(name)
        elif ink != 0: raise OracleError("damaged seven-segment stroke")
    try: digit = SEGMENT_DIGIT[frozenset(active)]
    except KeyError as exc: raise OracleError(f"invalid seven-segment digit {sorted(active)}") from exc
    if not 2 <= digit <= 9: raise OracleError("rate digit is outside [2, 9]")
    return digit


def _factor(arr: np.ndarray, cx: int, cy: int) -> tuple[int, int]:
    y = cy - 22; numerator = _digit(arr, cx - 37, y); denominator = _digit(arr, cx + 12, y)
    for yy in (cy - 9, cy + 9):
        if not np.all(arr[yy - 2:yy + 3, cx - 3:cx + 2] == INK): raise OracleError("ratio colon is missing")
    card = arr[cy - 31:cy + 32, cx - 56:cx + 57]
    if int(np.all(card == PANEL, axis=2).sum()) or any(np.all(card == color, axis=2).any() for color in TAG_RGB.values()):
        raise OracleError("factor card is contaminated")
    return numerator, denominator


def _route_color(arr: np.ndarray, y: int) -> str:
    counts = {name: int(np.all(arr[y:y + 12, 200:1000] == rgb, axis=2).sum()) for name, rgb in TAG_RGB.items()}
    names = [name for name, count in counts.items() if count > 6000]
    if len(names) != 1: raise OracleError("route border color is missing or ambiguous")
    return names[0]


def _validate_route_marks(arr: np.ndarray, y: int) -> None:
    cy = y + 117
    for cx in FACTOR_X:
        for x0, x1 in ((cx - 112, cx - 66), (cx + 66, cx + 112)):
            patch = arr[cy - 2:cy + 3, x0:x1]
            if patch.size == 0 or int(np.all(patch == INK, axis=2).sum()) < 210: raise OracleError("directed route is broken")
    if not np.all(arr[cy - 10:cy + 11, 75:96] == NODE_RGB[0]): raise OracleError("START node is damaged")
    if not np.all(arr[cy - 10:cy + 11, 1095:1116] == NODE_RGB[1]): raise OracleError("FINISH node is damaged")


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    if rgb.size != CANVAS: raise OracleError("wrong canvas")
    colors = rgb.getcolors(maxcolors=64)
    if colors is None or not {tuple(color) for _, color in colors} <= ALLOWED: raise OracleError("unknown raster color")
    arr = np.asarray(rgb); route_colors = []; products = []
    for y in ROW_Y:
        route_colors.append(_route_color(arr, y)); _validate_route_marks(arr, y); value = Fraction(1, 1)
        for cx in FACTOR_X:
            numerator, denominator = _factor(arr, cx, y + 117); value *= Fraction(numerator, denominator)
        products.append(value)
    if set(route_colors) != set(TAG_RGB): raise OracleError("route colors are not a permutation")
    maximum = max(products); matches = [i for i, value in enumerate(products) if value == maximum]
    if len(matches) != 1: raise OracleError("path products do not have a unique maximum")
    return route_colors[matches[0]]
