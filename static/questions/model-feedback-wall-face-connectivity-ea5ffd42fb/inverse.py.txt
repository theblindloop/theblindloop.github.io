"""Independent pixel-only inverse for the white-face connectivity question.

The oracle does not know the cell grid or the latent door list.  It segments
the dark walls and the three colored disks directly from the PNG, coarsens the
wall mask only after segmentation (the generous doors are much wider than one
coarse sample), flood-fills the non-wall pixels from the image border, and
returns the unique marker whose component is not exterior.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "wall_face_connectivity-oracle-1.0.0"
WIDTH = 720
HEIGHT = 560
SCALE = 4
WALL = np.array((37, 45, 57), dtype=np.int16)
MARKERS = {
    "red": np.array((218, 61, 70), dtype=np.int16),
    "blue": np.array((54, 106, 207), dtype=np.int16),
    "orange": np.array((235, 139, 38), dtype=np.int16),
}


class OracleError(ValueError):
    """Raised when the raster does not expose one clear answer."""


def _wall_mask(array: np.ndarray) -> np.ndarray:
    # The wall is the only dark ink.  The per-channel test retains its solid
    # interior through LANCZOS edges while excluding all three bright markers.
    return (
        (array[:, :, 0] < 120)
        & (array[:, :, 1] < 130)
        & (array[:, :, 2] < 145)
    )


def _marker_center(array: np.ndarray, name: str) -> tuple[float, float]:
    target = MARKERS[name]
    delta = array.astype(np.int32) - target.astype(np.int32)
    mask = np.sum(delta * delta, axis=2) <= 58 * 58
    ys, xs = np.where(mask)
    if len(xs) < 300:
        raise OracleError(f"marker {name} is missing or too small")
    return float(xs.mean()), float(ys.mean())


def _outside_open_pixels(wall: np.ndarray) -> np.ndarray:
    """Flood-fill a 4-connected coarse non-wall mask from its border."""

    height, width = wall.shape
    if height % SCALE or width % SCALE:
        raise OracleError("image dimensions are not compatible with raster probe")
    # Any wall ink in a coarse block marks that block as blocked.  A real door
    # is 44 px wide and remains an eleven-block opening, while a 16 px wall is
    # four blocked blocks; this is a conservative topology-preserving probe.
    coarse = wall.reshape(height // SCALE, SCALE, width // SCALE, SCALE).any(axis=(1, 3))
    ch, cw = coarse.shape
    seen = np.zeros((ch, cw), dtype=bool)
    queue: deque[tuple[int, int]] = deque()
    for x in range(cw):
        for y in (0, ch - 1):
            if not coarse[y, x] and not seen[y, x]:
                seen[y, x] = True
                queue.append((y, x))
    for y in range(ch):
        for x in (0, cw - 1):
            if not coarse[y, x] and not seen[y, x]:
                seen[y, x] = True
                queue.append((y, x))
    while queue:
        y, x = queue.popleft()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < ch and 0 <= nx < cw and not coarse[ny, nx] and not seen[ny, nx]:
                seen[ny, nx] = True
                queue.append((ny, nx))
    return seen


def decision_from_image(image: Image.Image) -> str:
    if image.size != (WIDTH, HEIGHT):
        raise OracleError(f"unexpected image size {image.size}")
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    wall = _wall_mask(array)
    outside = _outside_open_pixels(wall)
    answers: list[str] = []
    for name in ("red", "blue", "orange"):
        x, y = _marker_center(array, name)
        coarse_x = min(outside.shape[1] - 1, max(0, int(x) // SCALE))
        coarse_y = min(outside.shape[0] - 1, max(0, int(y) // SCALE))
        if not bool(outside[coarse_y, coarse_x]):
            answers.append(name)
    if len(answers) != 1:
        raise OracleError(f"expected one sealed marker, found {answers}")
    return answers[0]
