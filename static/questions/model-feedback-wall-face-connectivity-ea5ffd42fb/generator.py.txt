"""Fresh maze renderer for the white-face connectivity question.

The label is a property of the white complement of a dark wall arrangement:
starting at the image border, which coloured marker lies in a white component
that cannot be reached through the visible door gaps?  The scene stores only
rendered operands; there is no hidden answer field.
"""

from __future__ import annotations

import random
from collections import deque

from PIL import Image, ImageDraw

WIDTH = 720
HEIGHT = 560
COLS = 4
ROWS = 3
GRID_X = 100
GRID_Y = 70
CELL_W = 130
CELL_H = 130
GRID_RIGHT = GRID_X + COLS * CELL_W
GRID_BOTTOM = GRID_Y + ROWS * CELL_H

BACKGROUND = (249, 248, 244)
WALL = (37, 45, 57)
COLORS = {
    "red": (218, 61, 70),
    "blue": (54, 106, 207),
    "orange": (235, 139, 38),
}

RENDERER_VERSION = "wall_face_connectivity-renderer-1.0.0"
_OUTSIDE = COLS * ROWS
_ALL_V_DOORS = [(boundary, row) for boundary in range(1, COLS) for row in range(ROWS)]
_ALL_H_DOORS = [(col, boundary) for boundary in range(1, ROWS) for col in range(COLS)]
_ALL_OUTER_DOORS = [(side, index) for side in ("left", "right") for index in range(ROWS)] + [
    (side, index) for side in ("top", "bottom") for index in range(COLS)
]


def _cell_id(col: int, row: int) -> int:
    return row * COLS + col


def _cell_from_id(index: int) -> tuple[int, int]:
    return index % COLS, index // COLS


def _cell_center(col: int, row: int) -> tuple[float, float]:
    return (
        GRID_X + (col + 0.5) * CELL_W,
        GRID_Y + (row + 0.5) * CELL_H,
    )


def _normalise_door_lists(scene: dict) -> tuple[list[tuple[int, int]], list[tuple[int, int]], dict[str, list[int]]]:
    vertical = sorted((int(item[0]), int(item[1])) for item in scene["v_doors"])
    horizontal = sorted((int(item[0]), int(item[1])) for item in scene["h_doors"])
    outer = {
        side: sorted(int(value) for value in scene["outer_doors"].get(side, []))
        for side in ("left", "right", "top", "bottom")
    }
    return vertical, horizontal, outer


def _adjacency(scene: dict) -> dict[int, set[int]]:
    """Build the abstract cell adjacency induced by visible door gaps."""

    vertical, horizontal, outer = _normalise_door_lists(scene)
    graph: dict[int, set[int]] = {index: set() for index in range(_OUTSIDE + 1)}
    for boundary, row in vertical:
        left = _cell_id(boundary - 1, row)
        right = _cell_id(boundary, row)
        graph[left].add(right)
        graph[right].add(left)
    for col, boundary in horizontal:
        above = _cell_id(col, boundary - 1)
        below = _cell_id(col, boundary)
        graph[above].add(below)
        graph[below].add(above)
    for row in outer["left"]:
        cell = _cell_id(0, row)
        graph[_OUTSIDE].add(cell)
        graph[cell].add(_OUTSIDE)
    for row in outer["right"]:
        cell = _cell_id(COLS - 1, row)
        graph[_OUTSIDE].add(cell)
        graph[cell].add(_OUTSIDE)
    for col in outer["top"]:
        cell = _cell_id(col, 0)
        graph[_OUTSIDE].add(cell)
        graph[cell].add(_OUTSIDE)
    for col in outer["bottom"]:
        cell = _cell_id(col, ROWS - 1)
        graph[_OUTSIDE].add(cell)
        graph[cell].add(_OUTSIDE)
    return graph


def _outside_distances(scene: dict) -> dict[int, int]:
    graph = _adjacency(scene)
    distances = {_OUTSIDE: 0}
    queue: deque[int] = deque([_OUTSIDE])
    while queue:
        node = queue.popleft()
        for neighbour in sorted(graph[node]):
            if neighbour not in distances:
                distances[neighbour] = distances[node] + 1
                queue.append(neighbour)
    return distances


def _component_from(scene: dict, start: int) -> set[int]:
    graph = _adjacency(scene)
    found = {start}
    queue: deque[int] = deque([start])
    while queue:
        node = queue.popleft()
        for neighbour in graph[node]:
            if neighbour not in found:
                found.add(neighbour)
                queue.append(neighbour)
    return found


def analytic_gold(scene: dict) -> str:
    """Return the colour whose marked cell is not in the outside component."""

    distances = _outside_distances(scene)
    cells = [tuple(int(value) for value in position) for position in scene["marker_cells"]]
    colours = [str(value) for value in scene["marker_colors"]]
    if len(cells) != 3 or len(colours) != 3 or len(set(cells)) != 3:
        raise ValueError("expected three distinct marker cells and three colours")
    unreachable = [index for index, (col, row) in enumerate(cells) if _cell_id(col, row) not in distances]
    if len(unreachable) != 1:
        raise ValueError(f"expected one sealed marker, found {len(unreachable)}")
    return colours[unreachable[0]]


def margin(scene: dict) -> float:
    """Visible clearance margin for the discrete wall/door boundary."""

    wall_width = float(scene["wall_width"])
    door_gap = float(scene["door_gap"])
    marker_radius = float(scene["marker_radius"])
    marker_clearance = min(CELL_W / 2.0 - marker_radius - wall_width / 2.0, CELL_H / 2.0 - marker_radius - wall_width / 2.0)
    return float(min(door_gap - wall_width, marker_clearance))


def is_quarantined(scene: dict) -> bool:
    try:
        if margin(scene) < 14.0:
            return True
        vertical, horizontal, outer = _normalise_door_lists(scene)
        if any(item not in _ALL_V_DOORS for item in vertical) or len(set(vertical)) != len(vertical):
            return True
        if any(item not in _ALL_H_DOORS for item in horizontal) or len(set(horizontal)) != len(horizontal):
            return True
        for side, values in outer.items():
            limit = ROWS if side in {"left", "right"} else COLS
            if any(value < 0 or value >= limit for value in values) or len(set(values)) != len(values):
                return True
        cells = [tuple(int(value) for value in position) for position in scene["marker_cells"]]
        if len(cells) != 3 or len(set(cells)) != 3 or any(
            col < 0 or col >= COLS or row < 0 or row >= ROWS for col, row in cells
        ):
            return True
        if len({str(value) for value in scene["marker_colors"]}) != 3:
            return True
        analytic_gold(scene)
        return False
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """There is no unused record-order or label alias in this parametrization."""

    return []


def _sample_topology(rng: random.Random) -> tuple[list[tuple[int, int]], list[tuple[int, int]], dict[str, list[int]], list[int]]:
    """Sample a maze whose outside component and sealed component are nontrivial."""

    for _ in range(4000):
        v_doors = sorted(rng.sample(_ALL_V_DOORS, rng.randint(6, len(_ALL_V_DOORS))))
        h_doors = sorted(rng.sample(_ALL_H_DOORS, rng.randint(4, 7)))
        chosen_outer = rng.sample(_ALL_OUTER_DOORS, rng.randint(2, 4))
        outer = {side: [] for side in ("left", "right", "top", "bottom")}
        for side, index in chosen_outer:
            outer[side].append(index)
        graph_scene = {"v_doors": v_doors, "h_doors": h_doors, "outer_doors": outer}
        distances = _outside_distances(graph_scene)
        reachable = [index for index in range(_OUTSIDE) if index in distances]
        unreachable = [index for index in range(_OUTSIDE) if index not in distances]
        if len(reachable) < 6 or len(unreachable) < 2:
            continue
        sealed_start = rng.choice(unreachable)
        sealed_component = _component_from(graph_scene, sealed_start)
        if len(sealed_component & set(range(_OUTSIDE))) < 2:
            continue
        far_reachable = [index for index in reachable if distances[index] >= 2]
        if len(far_reachable) < 2:
            continue
        cells = [sealed_start, *rng.sample(far_reachable, 2)]
        rng.shuffle(cells)
        return v_doors, h_doors, outer, cells
    raise RuntimeError("could not sample a nontrivial sealed-face maze")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) * 1009 + 9176)
    v_doors, h_doors, outer, marker_ids = _sample_topology(rng)
    marker_cells = [list(_cell_from_id(index)) for index in marker_ids]
    marker_colors = ["red", "blue", "orange"]
    rng.shuffle(marker_colors)
    return {
        "v_doors": [list(item) for item in v_doors],
        "h_doors": [list(item) for item in h_doors],
        "outer_doors": {side: list(values) for side, values in sorted(outer.items())},
        "marker_cells": marker_cells,
        "marker_colors": marker_colors,
        "wall_width": 16,
        "door_gap": 44,
        "marker_radius": 22,
    }


def _draw_wall(draw: ImageDraw.ImageDraw, p1: tuple[float, float], p2: tuple[float, float], scale: int, width: int) -> None:
    draw.line(
        [(round(p1[0] * scale), round(p1[1] * scale)), (round(p2[0] * scale), round(p2[1] * scale))],
        fill=WALL,
        width=round(width * scale),
    )


def _draw_interval_with_door(
    draw: ImageDraw.ImageDraw,
    axis: str,
    fixed: float,
    start: float,
    end: float,
    door: bool,
    scale: int,
    width: int,
    gap: int,
) -> None:
    if not door:
        p1 = (fixed, start) if axis == "vertical" else (start, fixed)
        p2 = (fixed, end) if axis == "vertical" else (end, fixed)
        _draw_wall(draw, p1, p2, scale, width)
        return
    centre = (start + end) / 2.0
    half = gap / 2.0
    first_end = centre - half
    second_start = centre + half
    if first_end > start:
        p1 = (fixed, start) if axis == "vertical" else (start, fixed)
        p2 = (fixed, first_end) if axis == "vertical" else (first_end, fixed)
        _draw_wall(draw, p1, p2, scale, width)
    if second_start < end:
        p1 = (fixed, second_start) if axis == "vertical" else (second_start, fixed)
        p2 = (fixed, end) if axis == "vertical" else (end, fixed)
        _draw_wall(draw, p1, p2, scale, width)


def render(scene: dict) -> Image.Image:
    scale = 4
    wall_width = int(scene["wall_width"])
    gap = int(scene["door_gap"])
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    v_doors, h_doors, outer = _normalise_door_lists(scene)
    v_set = set(v_doors)
    h_set = set(h_doors)

    for row in range(ROWS):
        y0 = GRID_Y + row * CELL_H
        y1 = y0 + CELL_H
        _draw_interval_with_door(draw, "vertical", GRID_X, y0, y1, row in outer["left"], scale, wall_width, gap)
        _draw_interval_with_door(draw, "vertical", GRID_RIGHT, y0, y1, row in outer["right"], scale, wall_width, gap)
    for col in range(COLS):
        x0 = GRID_X + col * CELL_W
        x1 = x0 + CELL_W
        _draw_interval_with_door(draw, "horizontal", GRID_Y, x0, x1, col in outer["top"], scale, wall_width, gap)
        _draw_interval_with_door(draw, "horizontal", GRID_BOTTOM, x0, x1, col in outer["bottom"], scale, wall_width, gap)

    for boundary in range(1, COLS):
        x = GRID_X + boundary * CELL_W
        for row in range(ROWS):
            y0 = GRID_Y + row * CELL_H
            _draw_interval_with_door(draw, "vertical", x, y0, y0 + CELL_H, (boundary, row) in v_set, scale, wall_width, gap)
    for boundary in range(1, ROWS):
        y = GRID_Y + boundary * CELL_H
        for col in range(COLS):
            x0 = GRID_X + col * CELL_W
            _draw_interval_with_door(draw, "horizontal", y, x0, x0 + CELL_W, (col, boundary) in h_set, scale, wall_width, gap)

    for position, colour in zip(scene["marker_cells"], scene["marker_colors"]):
        col, row = int(position[0]), int(position[1])
        cx, cy = _cell_center(col, row)
        radius = float(scene["marker_radius"])
        box = [round((cx - radius) * scale), round((cy - radius) * scale), round((cx + radius) * scale), round((cy + radius) * scale)]
        draw.ellipse(box, fill=COLORS[str(colour)])

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
