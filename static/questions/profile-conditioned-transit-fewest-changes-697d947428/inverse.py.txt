"""Pixel-only oracle for transit_fewest_changes.

Receives the final PNG and nothing else: no scene, no renderer, no gold. It
segments the station discs and the coloured strokes, attributes every stroke
fragment to the two discs it runs into, rebuilds the line/station incidence
graph, and returns the fewest number of line switches between the green and the
red disc. It abstains with "unknown" whenever the reconstruction is ambiguous.
"""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "transit_fewest_changes-oracle-0.1.0"

ABSTAIN = "unknown"

_STATION_COLORS = {
    "plain": (35, 35, 35),
    "green": (35, 150, 60),
    "red": (215, 40, 40),
}
_LINE_COLORS = {
    "blue": (25, 85, 215),
    "orange": (235, 145, 20),
    "purple": (150, 55, 195),
    "teal": (20, 155, 150),
}
_TOL = 40.0
_MIN_DISC_PIXELS = 250
_MIN_STROKE_PIXELS = 60
_MIN_DISC_GAP = 14.0


def _mask(arr: np.ndarray, rgb) -> np.ndarray:
    d = arr.astype(np.int32) - np.array(rgb, dtype=np.int32)
    return (d * d).sum(axis=2) <= _TOL * _TOL


def _components(mask: np.ndarray) -> list[np.ndarray]:
    seen = np.zeros(mask.shape, dtype=bool)
    out: list[np.ndarray] = []
    h, w = mask.shape
    for y0, x0 in np.argwhere(mask):
        if seen[y0, x0]:
            continue
        stack = deque([(int(y0), int(x0))])
        seen[y0, x0] = True
        pixels = []
        while stack:
            y, x = stack.pop()
            pixels.append((y, x))
            for dy, dx in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                ny, nx = y + dy, x + dx
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        out.append(np.array(pixels, dtype=np.int32))
    return out


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))

    discs: list[dict] = []
    for kind, rgb in _STATION_COLORS.items():
        for comp in _components(_mask(arr, rgb)):
            if comp.shape[0] < _MIN_DISC_PIXELS:
                continue
            cy, cx = comp[:, 0].mean(), comp[:, 1].mean()
            discs.append({
                "kind": kind,
                "cy": float(cy),
                "cx": float(cx),
                "r": math.sqrt(comp.shape[0] / math.pi),
            })

    if len(discs) < 6:
        return ABSTAIN
    if sum(d["kind"] == "green" for d in discs) != 1:
        return ABSTAIN
    if sum(d["kind"] == "red" for d in discs) != 1:
        return ABSTAIN

    radius = float(np.median([d["r"] for d in discs]))
    for i, a in enumerate(discs):
        for b in discs[i + 1:]:
            gap = math.hypot(a["cy"] - b["cy"], a["cx"] - b["cx"]) - 2 * radius
            if gap < _MIN_DISC_GAP:
                return ABSTAIN

    reach = radius + 9.0
    centres = np.array([[d["cy"], d["cx"]] for d in discs], dtype=np.float64)

    line_stations: dict[str, set[int]] = {}
    for name, rgb in _LINE_COLORS.items():
        comps = [c for c in _components(_mask(arr, rgb)) if c.shape[0] >= _MIN_STROKE_PIXELS]
        if not comps:
            return ABSTAIN
        served: set[int] = set()
        for comp in comps:
            pts = comp.astype(np.float64)
            touched = []
            for idx, centre in enumerate(centres):
                dy = pts[:, 0] - centre[0]
                dx = pts[:, 1] - centre[1]
                if float(np.sqrt(dy * dy + dx * dx).min()) <= reach:
                    touched.append(idx)
            if len(touched) != 2:
                return ABSTAIN
            served.update(touched)
        line_stations[name] = served

    green = next(i for i, d in enumerate(discs) if d["kind"] == "green")
    red = next(i for i, d in enumerate(discs) if d["kind"] == "red")

    names = list(line_stations)
    start = [n for n in names if green in line_stations[n]]
    goal = {n for n in names if red in line_stations[n]}
    if not start or not goal:
        return ABSTAIN

    dist = {n: 0 for n in start}
    queue = deque(start)
    while queue:
        n = queue.popleft()
        if n in goal:
            return str(dist[n])
        for m in names:
            if m in dist:
                continue
            if line_stations[n] & line_stations[m]:
                dist[m] = dist[n] + 1
                queue.append(m)
    return ABSTAIN
