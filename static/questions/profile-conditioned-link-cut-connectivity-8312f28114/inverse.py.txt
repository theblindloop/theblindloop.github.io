"""Independent pixel-only inverse arm for the link-cut connectivity world.

Receives ONLY the final PNG. It never imports or reads the renderer, prompts,
generator, verifier, or any latent scene data. It recovers the decision by:

1. segmenting the picture by color:
     - dark : the ordinary network links and neutral nodes;
     - green: the start node;
     - red  : the goal node;
     - orange: the candidate bridge (the link that would be removed);
2. building a "travelable" mask of everything EXCEPT the orange bridge
   (i.e. the network that would remain after the orange link is removed);
3. flood-filling from the green start node through that mask and checking
   whether any red goal pixel is reached.

If green can reach a red pixel without passing through orange ink, the two
marked nodes remain connected after the bridge is removed (answer 1); otherwise
they become disconnected (answer 0).
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "link_cut_connectivity-oracle-0.4.0"


def _masks(arr: np.ndarray) -> tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
    r = arr[..., 0].astype(np.int16)
    g = arr[..., 1].astype(np.int16)
    b = arr[..., 2].astype(np.int16)
    is_green = (g > 95) & (g >= r + 40) & (g >= b + 40) & (r <= 110)
    is_red = (r > 150) & (r >= g + 80) & (r >= b + 80) & (g <= 100) & (b <= 100)
    is_orange = (r > 170) & (g >= 60) & (g <= 235) & (b <= 120)
    is_dark = (r <= 130) & (g <= 130) & (b <= 130)
    return is_green, is_red, is_orange, is_dark


def _flood_reaches_red(travel: np.ndarray, red: np.ndarray, seed) -> bool:
    h, w = travel.shape
    reached = np.zeros_like(travel, dtype=bool)
    stack = [seed]
    reached[seed[0], seed[1]] = True
    if red[seed[0], seed[1]]:
        return True
    while stack:
        y, x = stack.pop()
        for ny, nx in ((y + 1, x), (y - 1, x), (y, x + 1), (y, x - 1)):
            if 0 <= ny < h and 0 <= nx < w and travel[ny, nx] and not reached[ny, nx]:
                reached[ny, nx] = True
                if red[ny, nx]:
                    return True
                stack.append((ny, nx))
    return False


def decision_from_image(image: Image.Image) -> int:
    """1 if the marked nodes remain connected after removing the orange bridge.

    Derived from the final PNG alone.
    """
    arr = np.asarray(image.convert("RGB"))
    is_green, is_red, _is_orange, is_dark = _masks(arr)
    travel = (is_dark | is_green | is_red) & ~_is_orange
    greens = np.argwhere(is_green)
    if greens.size == 0:
        return 0
    seed = (int(greens[0, 0]), int(greens[0, 1]))
    return 1 if _flood_reaches_red(travel, is_red, seed) else 0
