"""Link-cut connectivity world renderer.

A small abstract network is drawn as thick dark links connecting circular nodes
on a white background. Two nodes are marked -- a green "start" node and a red
"goal" node. Exactly one link is highlighted in orange and is the "candidate
bridge". The decision is whether the green and red nodes `remain connected` to
each other through the remaining black links after that orange link is removed.

The scene is a plain dict so the verifier's AST reachability audit is side-effect
free:

    z = {
        "width": 520,
        "height": 380,
        "background": "white",
        "nodes": [[x, y, kind], ...],   # kind in {"start","goal","neutral"}
        "edges": [[i, j], ...],         # node indices
        "bridge_edge_index": k,         # edge drawn in orange (the candidate bridge)
        "link_width": 14.0,
        "node_radius": 14.0,
    }

Two canonical layouts are sampled:
  * "chain"  : a single simple path  start - n1 - n2 - goal ; the bridge is the
               middle edge. Removing it splits the path, so start and goal become
               DISCONNECTED (answer "no").
  * "cycle"  : a diamond  start - a - goal - b - start (a single cycle). Removing
               any single edge of a cycle leaves a connected path between start
               and goal, so they REMAIN connected (answer "yes").
"""

from __future__ import annotations

import math
import random
from collections.abc import Iterator

from PIL import Image, ImageDraw

RENDERER_VERSION = "link_cut_connectivity-0.4.0"

BACKGROUND = (255, 255, 255)
DARK = (45, 45, 45)
GREEN = (30, 150, 55)
RED = (210, 30, 45)
ORANGE = (255, 150, 0)

WIDTH_DEFAULT = 520
HEIGHT_DEFAULT = 380
LINK_WIDTH = 14.0
NODE_RADIUS = 14.0
QUARANTINE_MIN = 6.0

MIN_X = 80
MAX_X = 440


def _build_chain(rng: random.Random) -> dict:
    cx = 300.0
    base_y = [100.0, 190.0, 190.0, 280.0]
    xs = [MIN_X, 204.0, 316.0, MAX_X]
    nodes = []
    for i, (x, by) in enumerate(zip(xs, base_y)):
        jx = rng.uniform(-12.0, 12.0)
        jy = rng.uniform(-14.0, 14.0)
        kind = "start" if i == 0 else "goal" if i == 3 else "neutral"
        nodes.append([round(x + jx, 1), round(by + jy, 1), kind])
    edges = [[0, 1], [1, 2], [2, 3]]
    return {
        "width": WIDTH_DEFAULT,
        "height": HEIGHT_DEFAULT,
        "background": "white",
        "nodes": nodes,
        "edges": edges,
        "bridge_edge_index": 1,
        "link_width": LINK_WIDTH,
        "node_radius": NODE_RADIUS,
    }


def _build_cycle(rng: random.Random) -> dict:
    nodes = [
        [MIN_X, 180.0, "start"],
        [250.0, 90.0, "neutral"],
        [MAX_X, 200.0, "goal"],
        [250.0, 300.0, "neutral"],
    ]
    for node in nodes:
        node[0] = round(node[0] + rng.uniform(-14.0, 14.0), 1)
        node[1] = round(node[1] + rng.uniform(-14.0, 14.0), 1)
    edges = [[0, 1], [1, 2], [2, 3], [3, 0]]
    return {
        "width": WIDTH_DEFAULT,
        "height": HEIGHT_DEFAULT,
        "background": "white",
        "nodes": nodes,
        "edges": edges,
        "bridge_edge_index": rng.randint(0, 3),
        "link_width": LINK_WIDTH,
        "node_radius": NODE_RADIUS,
    }


def sample_scene(rng: random.Random, layout: str | None = None) -> dict:
    """Sample a robust chain or cycle network scene."""
    if layout is None:
        layout = "chain" if rng.random() < 0.5 else "cycle"
    if layout == "cycle":
        return _build_cycle(rng)
    return _build_chain(rng)


_NODE_COLOR = {"start": GREEN, "goal": RED, "neutral": DARK}


def render(scene: dict, supersample: int = 3) -> Image.Image:
    """Render I = R(z). Deterministic pure function of the scene dict."""
    width = int(scene["width"])
    height = int(scene["height"])
    background = scene["background"]
    nodes = scene["nodes"]
    edges = scene["edges"]
    bridge = int(scene["bridge_edge_index"])
    link_width = float(scene["link_width"])
    node_radius = float(scene["node_radius"])

    ss = supersample
    img = Image.new("RGB", (width * ss, height * ss), background)
    draw = ImageDraw.Draw(img)

    positions = [(float(n[0]) * ss, float(n[1]) * ss) for n in nodes]
    for index, (a, b) in enumerate(edges):
        color = ORANGE if index == bridge else DARK
        draw.line(
            [positions[a], positions[b]],
            fill=color,
            width=max(1, int(round(link_width * ss))),
        )
    for (x, y, kind) in nodes:
        color = _NODE_COLOR[kind]
        r = node_radius * ss
        draw.ellipse(
            [x * ss - r, y * ss - r, x * ss + r, y * ss + r],
            fill=color,
        )
    if ss > 1:
        img = img.resize((width, height), Image.LANCZOS)
    return img


def _graph_from_scene(scene: dict) -> tuple[list[dict], list[tuple[int, int]], int]:
    nodes = scene["nodes"]
    edges = scene["edges"]
    bridge = int(scene["bridge_edge_index"])
    return nodes, edges, bridge


def analytic_gold(scene: dict) -> int:
    """1 if start and goal remain connected after removing the bridge, else 0.

    Computed purely from the scene graph, never from pixels.
    """
    nodes, edges, bridge = _graph_from_scene(scene)
    n = len(nodes)
    start = next(i for i, node in enumerate(nodes) if node[2] == "start")
    goal = next(i for i, node in enumerate(nodes) if node[2] == "goal")
    adj: list[list[int]] = [[] for _ in range(n)]
    for index, (a, b) in enumerate(edges):
        if index == bridge:
            continue
        adj[a].append(b)
        adj[b].append(a)
    seen = {start}
    stack = [start]
    while stack:
        cur = stack.pop()
        for nxt in adj[cur]:
            if nxt not in seen:
                seen.add(nxt)
                stack.append(nxt)
    return 1 if goal in seen else 0


def margin(scene: dict) -> float:
    """Continuous robustness margin = smallest link/node half-extent clearance.

    All scenes are constructed with wide clearances so the value is comfortably
    above the quarantine floor. Small values indicate fragile, near-tangent
    connectivity that should be quarantined.
    """
    link_width = float(scene["link_width"])
    node_radius = float(scene["node_radius"])
    nodes = scene["nodes"]
    best = min(link_width / 2.0, node_radius)
    for i in range(len(nodes)):
        for j in range(i + 1, len(nodes)):
            dx = nodes[i][0] - nodes[j][0]
            dy = nodes[i][1] - nodes[j][1]
            clearance = math.hypot(dx, dy) - node_radius - node_radius
            best = min(best, clearance / 2.0)
    return float(best)


def is_quarantined(scene: dict) -> bool:
    """Quarantine scenes whose distinguishing features are too thin to trust."""
    return margin(scene) < QUARANTINE_MIN


def latent_symmetries(scene: dict) -> Iterator[dict]:
    """No nontrivial render-invariant symmetry: every field changes the picture.

    The marker colors (green/red), the orange bridge, and the link geometry are
    all semantically tied to the decision and are rendered directly. The scene
    is a canonical, minimal parametrization, so the world declares
    not_applicable for the latent-alias gate.
    """
    return iter(())
