"""Independent pixel-only inverse for steering-wheel side placement."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_MIN_CHROMA = 58
_MIN_AREA = 650
_MAX_AREA = 1700
_MIN_CENTER_DISTANCE = 72.0


class OracleAbstention(ValueError):
    pass


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dy == 0 and dx == 0:
                        continue
                    ny, nx = cy + dy, cx + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        output.append(np.asarray(points, dtype=np.int32))
    return output


def _wheel(array: np.ndarray) -> np.ndarray:
    high = array.max(axis=2).astype(np.int16)
    low = array.min(axis=2).astype(np.int16)
    mask = (high - low >= _MIN_CHROMA) & (high >= 115)
    candidates = []
    for points in _components(mask):
        area = len(points)
        if area < 20:
            continue
        ys, xs = points[:, 0], points[:, 1]
        height = int(ys.max() - ys.min() + 1)
        width = int(xs.max() - xs.min() + 1)
        fill = area / float(width * height)
        if not _MIN_AREA <= area <= _MAX_AREA:
            raise OracleAbstention(f"chromatic component has unsupported area {area}")
        if not (41 <= width <= 53 and 41 <= height <= 53):
            raise OracleAbstention("chromatic component has unsupported wheel diameter")
        if not 0.90 <= width / height <= 1.11 or not 0.36 <= fill <= 0.70:
            raise OracleAbstention("chromatic component is not a compact steering wheel")
        candidates.append(points)
    if len(candidates) != 1:
        raise OracleAbstention(f"expected one connected chromatic steering wheel, found {len(candidates)}")
    return candidates[0]


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape != (512, 512, 3):
        raise OracleAbstention("expected a 512 by 512 RGB raster")
    points = _wheel(array)
    cy = float(points[:, 0].mean())
    cx = float(points[:, 1].mean())
    if not 180.0 <= cy <= 350.0:
        raise OracleAbstention("wheel lies outside the visible dashboard band")
    signed = cx - array.shape[1] / 2.0
    if abs(signed) <= _MIN_CENTER_DISTANCE:
        raise OracleAbstention("wheel is too close to the bilateral midpoint")
    return "left" if signed < 0.0 else "right"
