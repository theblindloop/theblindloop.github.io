"""Deterministic matched renderer for steering-wheel side placement."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

CANVAS_SIZE = 512
SUPERSAMPLE = 4
MID_X = CANVAS_SIZE // 2
QUARANTINE_DISTANCE = 72.0

WHEEL_PALETTE = (
    (216, 49, 72),
    (35, 112, 220),
    (28, 157, 93),
    (226, 139, 24),
    (145, 67, 196),
)
BACKGROUND_TONES = ((246, 244, 238), (239, 244, 247), (246, 241, 246))
DASHBOARD_TONES = ((59, 65, 70), (66, 68, 73), (48, 62, 68))


def sample_scene(seed: int) -> dict:
    """Adjacent seeds are matched left/right counterfactuals."""

    seed = int(seed)
    pair = seed // 2
    rng = random.Random(pair ^ 0x57EE71)
    return {
        "background_tone": pair % len(BACKGROUND_TONES),
        "canvas_size": CANVAS_SIZE,
        "dashboard_tone": (pair // 3) % len(DASHBOARD_TONES),
        "distance_code": rng.randrange(3),
        "trim_variant": rng.randrange(12),
        "wheel_color_code": (pair // 2) % len(WHEEL_PALETTE),
        "wheel_radius": (21, 23, 25)[rng.randrange(3)],
        "wheel_side": "left" if seed % 2 == 0 else "right",
        "wheel_y": (218, 250, 282, 314)[rng.randrange(4)],
    }


def _box(box: tuple[float, float, float, float], scale: int) -> tuple[int, int, int, int]:
    return tuple(round(value * scale) for value in box)


def _point(point: tuple[float, float], scale: int) -> tuple[int, int]:
    return tuple(round(value * scale) for value in point)


def _wheel_center(scene: dict) -> tuple[int, int]:
    distance = (96, 112, 128)[int(scene["distance_code"]) % 3]
    side = str(scene["wheel_side"])
    if side not in {"left", "right"}:
        raise ValueError("wheel_side must be left or right")
    x = MID_X - distance if side == "left" else MID_X + distance
    return x, int(scene["wheel_y"])


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas_size"])
    if size != CANVAS_SIZE:
        raise ValueError("this world uses a fixed 512-pixel canvas")
    background = BACKGROUND_TONES[int(scene["background_tone"]) % len(BACKGROUND_TONES)]
    panel = DASHBOARD_TONES[int(scene["dashboard_tone"]) % len(DASHBOARD_TONES)]
    variant = int(scene["trim_variant"])
    radius = int(scene["wheel_radius"])
    wheel = WHEEL_PALETTE[int(scene["wheel_color_code"]) % len(WHEEL_PALETTE)]
    cx, cy = _wheel_center(scene)

    ss = SUPERSAMPLE
    image = Image.new("RGB", (size * ss, size * ss), background)
    draw = ImageDraw.Draw(image)
    edge = tuple(max(0, value - 24) for value in panel)
    trim = tuple(min(255, value + 72) for value in panel)
    glass = tuple(min(255, value + 112) for value in panel)
    faint = tuple(max(0, value - 10) for value in background)

    # Remote marks vary with nuisance style and never carry the side label.
    for index in range(3):
        x = 34 + ((variant * 47 + index * 181) % 444)
        y = 30 if (variant + index) % 2 == 0 else 482
        draw.rounded_rectangle(_box((x - 18, y - 3, x + 18, y + 3), ss), radius=3 * ss, fill=faint)

    # A bilaterally symmetric, centered car interior.
    draw.rounded_rectangle(_box((52, 86, 460, 426), ss), radius=36 * ss, fill=panel, outline=edge, width=4 * ss)
    draw.rounded_rectangle(_box((78, 108, 434, 198), ss), radius=22 * ss, fill=glass, outline=trim, width=3 * ss)
    draw.polygon([_point((256, 108), ss), _point((246, 198), ss), _point((266, 198), ss)], fill=trim)
    draw.rounded_rectangle(_box((210, 214, 302, 393), ss), radius=16 * ss, fill=edge, outline=trim, width=3 * ss)
    draw.rounded_rectangle(_box((228, 236, 284, 286), ss), radius=7 * ss, fill=glass)
    for y in (309, 339, 369):
        draw.rounded_rectangle(_box((237, y - 4, 275, y + 4), ss), radius=4 * ss, fill=trim)
    for vent_x in (105, 407):
        draw.ellipse(_box((vent_x - 20, 224, vent_x + 20, 264), ss), fill=edge, outline=trim, width=3 * ss)
        draw.line([_point((vent_x - 11, 244), ss), _point((vent_x + 11, 244), ss)], fill=trim, width=3 * ss)
    draw.line([_point((74, 405), ss), _point((438, 405), ss)], fill=trim, width=3 * ss)

    # One connected chromatic wheel: thick annulus, three spokes, and hub.
    draw.ellipse(_box((cx - radius, cy - radius, cx + radius, cy + radius), ss), fill=wheel)
    inner = max(8, radius - 8)
    draw.ellipse(_box((cx - inner, cy - inner, cx + inner, cy + inner), ss), fill=panel)
    spoke_width = 6 * ss
    draw.line([_point((cx, cy), ss), _point((cx, cy - inner - 2), ss)], fill=wheel, width=spoke_width)
    draw.line([_point((cx, cy), ss), _point((cx - inner + 1, cy + inner - 2), ss)], fill=wheel, width=spoke_width)
    draw.line([_point((cx, cy), ss), _point((cx + inner - 1, cy + inner - 2), ss)], fill=wheel, width=spoke_width)
    draw.ellipse(_box((cx - 7, cy - 7, cx + 7, cy + 7), ss), fill=wheel)

    return image.resize((size, size), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    side = str(scene["wheel_side"])
    if side not in {"left", "right"}:
        raise ValueError("wheel_side must be left or right")
    return side


def margin(scene: dict) -> float:
    cx, _ = _wheel_center(scene)
    return float(abs(cx - MID_X))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= QUARANTINE_DISTANCE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["wheel_color_code"] = int(twin["wheel_color_code"]) + len(WHEEL_PALETTE)
    return [("one_full_palette_cycle", twin)]
