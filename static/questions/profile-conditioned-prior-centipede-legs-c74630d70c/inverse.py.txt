"""Independent pixel-only inverse arm for the centipede-leg world.

Receives ONLY the final PNG and returns the number of legs drawn as a string.
It shares no geometry constants with renderer.render except the dark leg/body
colour, which is part of the visible image. It never imports renderer, prompts,
generate, verify, or analytic-gold code, and accepts no latent scene data.

Method: the centipede body and legs are the only dark pixels. The oracle
colour-segments them and finds the body band as the rows whose dark extent spans
almost the full body width (the wide horizontal bar). Upward legs occupy only
the rows strictly above that band and downward legs only the rows strictly
below it; because legs do not touch each other, each leg appears in its band as
a separated vertical stroke. Counting the contiguous column-runs of dark pixels
in the upper band and in the lower band and summing gives the total number of
legs.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "prior_centipede_legs-oracle-0.1.0"

_DARK = np.array((40, 26, 14), dtype=np.float32)
_COLOR_DIST = 60.0
_BODY_THRESH = 220.0


def _run_count(present: np.ndarray) -> int:
    """Count contiguous runs of True in a per-column boolean signal."""
    wrapped = np.concatenate([present.astype(np.int8), present[:1].astype(np.int8)])
    transitions = int(np.sum(np.abs(np.diff(wrapped))))
    return transitions // 2


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    mask = np.linalg.norm(arr - _DARK, axis=-1) < _COLOR_DIST
    h = mask.shape[0]
    row_count = mask.sum(axis=1).astype(np.float64)
    body_rows = np.nonzero(row_count > _BODY_THRESH)[0]
    if len(body_rows) == 0:
        return "0"
    body_top = int(body_rows.min())
    body_bot = int(body_rows.max())
    upper = mask[0:max(body_top - 3, 0), :]
    lower = mask[min(body_bot + 4, h):, :]
    up = _run_count(upper.any(axis=0)) if upper.shape[0] > 0 else 0
    low = _run_count(lower.any(axis=0)) if lower.shape[0] > 0 else 0
    return str(up + low)


def measure(image: Image.Image) -> dict:
    """Diagnostic measurement (not used for the decision)."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    mask = np.linalg.norm(arr - _DARK, axis=-1) < _COLOR_DIST
    row_count = mask.sum(axis=1)
    body_rows = np.nonzero(row_count > _BODY_THRESH)[0]
    return {
        "dark_px": int(mask.sum()),
        "body_rows": (int(body_rows.min()), int(body_rows.max())) if len(body_rows) else None,
        "size": image.size,
    }
