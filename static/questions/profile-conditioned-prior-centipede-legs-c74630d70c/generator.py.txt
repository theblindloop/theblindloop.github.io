"""Centipede-leg count world under a contradictory canonical prior.

Profile: paper-semantic@0.3.0:prior_conflict. Report visible truth (the actual
number of legs) when a familiar canonical prior ("a centipede has a hundred
legs -- its name literally means hundred-legged") is contradicted by the actual
image, which in the counterfactual scenes shows a centipede with only a handful
of leg pairs.

Latent z = {p, x_off, reflected, width, height}. ``p`` is the number of leg
pairs drawn along the body, giving 2*p legs total. ``x_off`` translates the
whole animal horizontally. ``reflected`` selects the (equivariant) vertically
reflected drawing: because the centipede's paired-leg body is exactly symmetric
about its horizontal midline, both polarities render pixel-identically, giving a
declared, exercised latent symmetry. The closed-form gold is

    decision = str(2 * p)

a scene parameter, not a pixel measurement.

The centipede is drawn as a dark horizontal body bar with ``p`` equally spaced
leg pairs; each pair has one upward leg and one downward leg stemming from the
body. All body and leg pixels share one dark colour, and the legs do not touch
one another. The pixel-only inverse arm colour-segments the body, locates the
body band by its row width, and counts the number of separated upward and
downward leg protrusions, whose sum is exactly the number of legs.

The continuous verifiability margin is the horizontal gap between adjacent leg
columns in pixels (body span / (p-1) minus the leg width). When that gap is too
small the legs crowd into each other and the count becomes pixel-ambiguous, so
such scenes are quarantined.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "prior_centipede_legs-0.1.0"

CANVAS = (512, 512)
BG = (255, 255, 255)
DARK = (40, 26, 14)   # the only dark, counted pixels (body + legs)

LEG_WIDTH = 16
BODY_X0 = 120
BODY_X1 = 392          # body horizontal span 272 px
BODY_TOP = 241
BODY_BOT = 270         # body thickness 30 px, centred on y = 255.5
LEG_TOP = 171
LEG_BOT = 340          # legs extend 70 px above/below the body

SUPPORTED_P = tuple(range(3, 13))   # 3..12 pairs -> 6..24 legs
QUAR_MARGIN_PX = 10.0


def margin(scene: dict) -> float:
    """Horizontal gap (px) between adjacent leg columns; the count robustness."""
    p = int(scene["p"])
    span = float(BODY_X1 - BODY_X0)
    return span / (p - 1) - LEG_WIDTH


def is_quarantined(scene: dict) -> bool:
    p = int(scene["p"])
    if p not in SUPPORTED_P:
        return True
    return margin(scene) < QUAR_MARGIN_PX


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    while True:
        p = rng.choice(SUPPORTED_P)
        scene = {
            "p": p,
            "x_off": rng.randint(-8, 8),
            "reflected": False,
            "width": CANVAS[0],
            "height": CANVAS[1],
        }
        if not is_quarantined(scene):
            return scene


def render(scene: dict) -> Image.Image:
    w, h = scene["width"], scene["height"]
    p = int(scene["p"])
    off = int(scene.get("x_off", 0))

    img = Image.new("RGB", (w, h), BG)
    draw = ImageDraw.Draw(img)
    draw.rectangle(
        [BODY_X0 + off, BODY_TOP, BODY_X1 + off, BODY_BOT], fill=DARK
    )
    half = LEG_WIDTH // 2
    for k in range(p):
        cx = BODY_X0 + off + int(round((BODY_X1 - BODY_X0) * k / (p - 1)))
        # upper leg (above the body) and lower leg (below the body)
        draw.rectangle([cx - half, LEG_TOP, cx + half, BODY_TOP - 1], fill=DARK)
        draw.rectangle([cx - half, BODY_BOT + 1, cx + half, LEG_BOT], fill=DARK)
    if scene.get("reflected", False):
        img = img.transpose(Image.FLIP_TOP_BOTTOM)
    return img


def analytic_gold(scene: dict) -> str:
    return str(2 * int(scene["p"]))


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """One declared, exercised reflection: flip the drawing about its midline.

    The centipede's paired-leg body is exactly symmetric about its horizontal
    midline (the body is a centred bar and each leg pair has a same-length
    upward and downward leg at the same column), so reflecting the whole scene
    vertically maps the figure to itself pixel-identically while preserving the
    leg count (gold). The ``reflected`` flag selects the equivariant polarity.
    """
    twin = dict(scene, reflected=not bool(scene.get("reflected", False)))
    return [("reflect_about_midline", twin)]
