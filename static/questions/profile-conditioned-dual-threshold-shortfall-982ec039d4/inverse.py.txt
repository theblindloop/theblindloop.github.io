"""Independent image-only inverse for the dual-threshold shortfall plot."""

from __future__ import annotations

import numpy as np
from PIL import Image

_PALETTE = {
    "red": (211, 54, 58),
    "blue": (39, 103, 189),
    "green": (34, 145, 79),
    "purple": (139, 71, 171),
}
_TARGET = np.asarray((184, 62, 139), dtype=np.int32)


def _point_centers(arr: np.ndarray) -> dict[str, tuple[float, float]]:
    centers: dict[str, tuple[float, float]] = {}
    work = arr.astype(np.int32)
    for name, rgb in _PALETTE.items():
        delta = work - np.asarray(rgb, dtype=np.int32)
        mask = np.sum(delta * delta, axis=2, dtype=np.int32) < 1500
        ys, xs = np.nonzero(mask)
        if len(xs) < 140:
            raise ValueError(f"missing or damaged {name} point")
        if int(xs.max() - xs.min()) > 32 or int(ys.max() - ys.min()) > 32:
            raise ValueError(f"invalid {name} point geometry")
        centers[name] = (float(xs.mean()), float(ys.mean()))
    return centers


def _targets(arr: np.ndarray) -> tuple[float, float]:
    work = arr.astype(np.int32)
    delta = work - _TARGET
    mask = np.sum(delta * delta, axis=2, dtype=np.int32) < 650
    column_scores = mask.sum(axis=0)
    row_scores = mask.sum(axis=1)
    columns = np.flatnonzero(column_scores > 105)
    rows = np.flatnonzero(row_scores > 175)
    if len(columns) < 2 or len(rows) < 2:
        raise ValueError("missing or damaged dashed targets")
    if int(columns.max() - columns.min()) > 8 or int(rows.max() - rows.min()) > 8:
        raise ValueError("ambiguous dashed target geometry")
    return float(columns.mean()), float(rows.mean())


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.ndim != 3 or arr.shape[0] < 450 or arr.shape[1] < 650:
        raise ValueError("image is not the declared dual-target plot raster")
    centers = _point_centers(arr)
    tx, ty = _targets(arr)
    scores = {
        color: max(0.0, tx - center[0]) + max(0.0, center[1] - ty)
        for color, center in centers.items()
    }
    ordered = sorted(scores, key=scores.get)
    if scores[ordered[1]] - scores[ordered[0]] < 6.0:
        return "abstain"
    return ordered[0]
