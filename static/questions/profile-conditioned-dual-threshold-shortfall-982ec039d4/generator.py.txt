"""Deterministic dual-threshold shortfall scenes and Pillow renderer."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 720, 500
COLORS = ("red", "blue", "green", "purple")
RGB = {
    "red": (211, 54, 58),
    "blue": (39, 103, 189),
    "green": (34, 145, 79),
    "purple": (139, 71, 171),
}
TARGET_RGB = (184, 62, 139)
BACKGROUND = (250, 250, 247)


def _shortfalls(scene: dict) -> dict[str, float]:
    tx, ty = float(scene["target_x"]), float(scene["target_y"])
    return {
        color: max(0.0, tx - float(point[0])) + max(0.0, float(point[1]) - ty)
        for color, point in zip(scene["colors"], scene["points"], strict=True)
    }


def sample_scene(seed: int) -> dict:
    """Return a balanced scene with a safely separated minimum shortfall."""
    rng = random.Random(int(seed))
    winner_index = int(seed) % len(COLORS)
    winner_style = (int(seed) // len(COLORS)) % 3
    target_x = rng.randint(344, 376)
    target_y = rng.randint(246, 274)

    # Each pair is (x shortfall, y shortfall). Zero means that target is met;
    # the satisfying coordinate is placed a random safe distance past its line.
    if winner_style == 0:  # winner misses both
        roles = [(rng.randint(34, 40), rng.randint(32, 38)),
                 (rng.randint(98, 106), 0),
                 (0, rng.randint(112, 122)),
                 (rng.randint(73, 80), rng.randint(72, 79))]
    elif winner_style == 1:  # winner misses x only
        roles = [(rng.randint(65, 72), 0),
                 (rng.randint(43, 48), rng.randint(44, 49)),
                 (0, rng.randint(108, 118)),
                 (rng.randint(127, 138), 0)]
    else:  # winner misses y only
        roles = [(0, rng.randint(65, 72)),
                 (rng.randint(44, 49), rng.randint(43, 48)),
                 (rng.randint(106, 116), 0),
                 (0, rng.randint(128, 139))]

    remaining = [i for i in range(4) if i != winner_index]
    rng.shuffle(remaining)
    roles_by_color: list[tuple[int, int] | None] = [None] * 4
    roles_by_color[winner_index] = roles[0]
    for color_index, role in zip(remaining, roles[1:], strict=True):
        roles_by_color[color_index] = role

    points: list[list[int]] = []
    for dx, dy in roles_by_color:  # type: ignore[misc]
        x = target_x - dx if dx else target_x + rng.randint(52, 132)
        y = target_y + dy if dy else target_y - rng.randint(48, 126)
        points.append([x, y])

    scene = {
        "colors": list(COLORS),
        "points": points,
        "target_x": target_x,
        "target_y": target_y,
    }
    if is_quarantined(scene):
        raise RuntimeError("constructed scene unexpectedly quarantined")
    return scene


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    path = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf" if bold else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
    try:
        return ImageFont.truetype(path, size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    tx, ty = int(scene["target_x"]), int(scene["target_y"])

    draw.text((WIDTH * scale // 2, 27 * scale), "Combined shortfall to two targets",
              fill=(30, 34, 39), font=_font(22 * scale, True), anchor="mm")
    draw.text((WIDTH * scale // 2, 58 * scale),
              "shortfall = left gap (if any) + below gap (if any)",
              fill=(67, 71, 77), font=_font(15 * scale), anchor="mm")

    left, top, right, bottom = 72, 92, 664, 448
    for x in range(112, 665, 46):
        draw.line((x * scale, top * scale, x * scale, bottom * scale), fill=(226, 228, 226), width=scale)
    for y in range(116, 449, 44):
        draw.line((left * scale, y * scale, right * scale, y * scale), fill=(226, 228, 226), width=scale)
    draw.line((left * scale, bottom * scale, right * scale, bottom * scale), fill=(83, 88, 94), width=2 * scale)
    draw.line((left * scale, top * scale, left * scale, bottom * scale), fill=(83, 88, 94), width=2 * scale)

    def dashed_line(x1: int, y1: int, x2: int, y2: int) -> None:
        length = int(round(math.hypot(x2 - x1, y2 - y1)))
        for pos in range(0, length + 1, 20):
            stop = min(pos + 12, length)
            ax = x1 + (x2 - x1) * pos / length
            ay = y1 + (y2 - y1) * pos / length
            bx = x1 + (x2 - x1) * stop / length
            by = y1 + (y2 - y1) * stop / length
            draw.line((ax * scale, ay * scale, bx * scale, by * scale), fill=TARGET_RGB, width=3 * scale)

    dashed_line(tx, top, tx, bottom)
    dashed_line(left, ty, right, ty)
    draw.text((tx * scale, 467 * scale), "x target", fill=(71, 73, 77), font=_font(12 * scale), anchor="mm")
    draw.text((48 * scale, ty * scale), "y target", fill=(71, 73, 77), font=_font(12 * scale), anchor="mm")

    for color, point in zip(scene["colors"], scene["points"], strict=True):
        cx, cy = map(int, point)
        radius = 13
        diamond = [(cx * scale, (cy - radius) * scale), ((cx + radius) * scale, cy * scale),
                   (cx * scale, (cy + radius) * scale), ((cx - radius) * scale, cy * scale)]
        draw.polygon(diamond, fill=RGB[color], outline=(255, 255, 255), width=2 * scale)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    scores = _shortfalls(scene)
    ordered = sorted(scores, key=scores.get)
    if scores[ordered[1]] - scores[ordered[0]] <= 0:
        return "abstain"
    return ordered[0]


def margin(scene: dict) -> float:
    values = sorted(_shortfalls(scene).values())
    return float(values[1] - values[0])


def is_quarantined(scene: dict) -> bool:
    tx, ty = float(scene["target_x"]), float(scene["target_y"])
    points = [(float(x), float(y)) for x, y in scene["points"]]
    target_clearance = min(min(abs(x - tx), abs(y - ty)) for x, y in points)
    point_clearance = min(
        math.hypot(a[0] - b[0], a[1] - b[1])
        for index, a in enumerate(points) for b in points[index + 1:]
    )
    return margin(scene) < 12.0 or target_clearance < 28.0 or point_clearance < 32.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
