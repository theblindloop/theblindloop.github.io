"""Independent pixel-only inverse for switchyard reachability."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

GRAY = np.array((74, 85, 99), dtype=np.int16)
MARKERS = {
    "red": np.array((211, 52, 61), dtype=np.int16),
    "green": np.array((31, 157, 98), dtype=np.int16),
    "blue": np.array((42, 101, 210), dtype=np.int16),
}


def _near(array: np.ndarray, color: np.ndarray, tolerance: int = 8) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _centroid(mask: np.ndarray, name: str) -> tuple[int, int]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 20:
        raise ValueError(f"missing {name} station marker")
    return int(round(float(xs.mean()))), int(round(float(ys.mean())))


def _labels(mask: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    labels = np.zeros((height, width), dtype=np.int32)
    current = 0
    for y, x in zip(*np.nonzero(mask)):
        if labels[y, x]:
            continue
        current += 1
        labels[y, x] = current
        queue = deque([(int(y), int(x))])
        while queue:
            cy, cx = queue.popleft()
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and labels[ny, nx] == 0:
                    labels[ny, nx] = current
                    queue.append((ny, nx))
    return labels


def _component_at(labels: np.ndarray, point: tuple[int, int]) -> int:
    x, y = point
    # A filled marker can create a tiny gray collar island at a junction under
    # some combinations of incident edge colors. Inspect the full local
    # neighborhood and bind to the largest global component touching it, which
    # is the path network rather than that decorative collar.
    radius = 19
    y0, y1 = max(0, y - radius), min(labels.shape[0], y + radius + 1)
    x0, x1 = max(0, x - radius), min(labels.shape[1], x + radius + 1)
    nearby = set(int(value) for value in np.unique(labels[y0:y1, x0:x1]) if value > 0)
    if not nearby:
        raise ValueError("station marker is not attached to a gray junction")
    return max(nearby, key=lambda value: int((labels == value).sum()))


def decision_from_image(image: Image.Image) -> str:
    """Return the station in the black marker's gray connected component."""
    array = np.asarray(image.convert("RGB"))
    gray_mask = _near(array, GRAY)
    if int(gray_mask.sum()) < 500:
        raise ValueError("no usable gray switchyard network")
    labels = _labels(gray_mask)

    black_mask = np.max(array, axis=2) < 45
    black = _centroid(black_mask, "black")
    black_component = _component_at(labels, black)
    winners: list[str] = []
    for name, rgb in MARKERS.items():
        marker = _centroid(_near(array, rgb), name)
        if _component_at(labels, marker) == black_component:
            winners.append(name)
    if len(winners) != 1:
        raise ValueError(f"pixel topology has {len(winners)} reachable colored stations")
    return winners[0]
