"""Deterministic renderer for the median rank-filter reveal world.

The visible tile is a broad binary glyph corrupted by a per-scene
salt-and-pepper carrier.  The question's declared 7x7 median is the
answer-bearing operation.  This module deliberately contains the analytic
scene construction; the pixel-only inverse is implemented independently in
``oracle.py``.
"""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageColor, ImageDraw


CANVAS_SIZE = 512
MARK_SIZE = 16
MARK_GAP = 8
FILTER_SIZE = 7
TEMPLATE_SIZE = 128
LOW_TONE = 42
HIGH_TONE = 214
SYMBOLS = ("fan", "gate", "orb", "sail")
MARKER_PALETTE = (
    "#d1495b",
    "#247ba0",
    "#2a9d8f",
    "#8e5bb7",
    "#e08e0b",
)


def _line(draw: ImageDraw.ImageDraw, points: list[tuple[int, int]], width: int) -> None:
    draw.line(points, fill=255, width=width, joint="curve")


def _symbol_mask(symbol: str, size: int) -> np.ndarray:
    """Return a dark-stroke mask in a normalized square raster."""

    if symbol not in SYMBOLS:
        raise ValueError(f"unknown symbol {symbol!r}")
    image = Image.new("L", (size, size), 0)
    draw = ImageDraw.Draw(image)
    w = max(7, int(round(size * 0.075)))
    q = lambda value: int(round(value * (size - 1)))

    if symbol == "fan":
        hub = (q(0.50), q(0.78))
        for endpoint in ((0.22, 0.22), (0.50, 0.14), (0.78, 0.22)):
            _line(draw, [hub, (q(endpoint[0]), q(endpoint[1]))], w)
        _line(draw, [(q(0.34), q(0.80)), (q(0.66), q(0.80))], w)
    elif symbol == "gate":
        _line(draw, [(q(0.28), q(0.82)), (q(0.28), q(0.40)), (q(0.50), q(0.18)),
                     (q(0.72), q(0.40)), (q(0.72), q(0.82))], w)
        _line(draw, [(q(0.22), q(0.82)), (q(0.78), q(0.82))], w)
        _line(draw, [(q(0.33), q(0.55)), (q(0.67), q(0.55))], w)
    elif symbol == "orb":
        box = [q(0.23), q(0.18), q(0.77), q(0.72)]
        draw.ellipse(box, outline=255, width=w)
        _line(draw, [(q(0.50), q(0.72)), (q(0.50), q(0.88))], w)
        _line(draw, [(q(0.38), q(0.88)), (q(0.62), q(0.88))], w)
    elif symbol == "sail":
        _line(draw, [(q(0.50), q(0.14)), (q(0.50), q(0.86))], w)
        _line(draw, [(q(0.47), q(0.20)), (q(0.22), q(0.72)), (q(0.47), q(0.72)),
                     (q(0.47), q(0.20))], w)
        _line(draw, [(q(0.53), q(0.36)), (q(0.78), q(0.72)), (q(0.53), q(0.72))], w)
        _line(draw, [(q(0.30), q(0.86)), (q(0.70), q(0.86))], w)

    return np.asarray(image, dtype=np.uint8) > 127


def _median_filter(values: np.ndarray) -> np.ndarray:
    radius = FILTER_SIZE // 2
    padded = np.pad(values, ((radius, radius), (radius, radius)), mode="edge")
    windows = np.lib.stride_tricks.sliding_window_view(
        padded, (FILTER_SIZE, FILTER_SIZE)
    )
    return np.median(windows, axis=(-1, -2)).astype(np.uint8)


def _encoded_tile(scene: dict) -> np.ndarray:
    side = int(scene["side"])
    mask = _symbol_mask(str(scene["symbol"]), side)
    signal = np.where(mask, LOW_TONE, HIGH_TONE).astype(np.uint8)
    rng = np.random.default_rng(int(scene["carrier_seed"]))
    flips = rng.random((side, side)) < float(scene["noise_rate"])
    encoded = signal.copy()
    encoded[flips] = np.where(encoded[flips] == LOW_TONE, HIGH_TONE, LOW_TONE)
    return encoded


def sample_scene(seed: int) -> dict:
    """Sample one balanced, translated marked raster scene."""

    rng = random.Random(int(seed))
    side = (208, 224, 240)[int(seed) % 3]
    symbol = SYMBOLS[(int(seed) * 5 + 1) % len(SYMBOLS)]
    marker_color = MARKER_PALETTE[(int(seed) * 3 + 2) % len(MARKER_PALETTE)]
    noise_levels = (0.30, 0.34, 0.38, 0.42)
    noise_rate = noise_levels[(int(seed) // 3) % len(noise_levels)]
    margin = MARK_SIZE + MARK_GAP + 14
    x = rng.randint(margin, CANVAS_SIZE - side - margin - 1)
    y = rng.randint(margin, CANVAS_SIZE - side - margin - 1)
    return {
        "canvas_size": CANVAS_SIZE,
        "side": side,
        "x": x,
        "y": y,
        "symbol": symbol,
        "marker_color": marker_color,
        "noise_rate": noise_rate,
        "carrier_seed": rng.randrange(1, 2**31 - 1),
    }


def render(scene: dict) -> Image.Image:
    """Render only from the public scene dictionary."""

    canvas_size = int(scene["canvas_size"])
    side = int(scene["side"])
    x = int(scene["x"])
    y = int(scene["y"])
    marker_color = ImageColor.getrgb(str(scene["marker_color"]))
    array = np.full((canvas_size, canvas_size, 3), 255, dtype=np.uint8)
    tile = _encoded_tile(scene)
    array[y : y + side, x : x + side, :] = tile[:, :, None]

    positions = (
        (x - MARK_GAP - MARK_SIZE, y - MARK_GAP - MARK_SIZE),
        (x + side + MARK_GAP, y - MARK_GAP - MARK_SIZE),
        (x - MARK_GAP - MARK_SIZE, y + side + MARK_GAP),
        (x + side + MARK_GAP, y + side + MARK_GAP),
    )
    rgb = np.asarray(marker_color, dtype=np.uint8)
    for px, py in positions:
        array[py : py + MARK_SIZE, px : px + MARK_SIZE, :] = rgb
    return Image.fromarray(array, mode="RGB")


def _classification_scores(tile: np.ndarray) -> dict[str, float]:
    restored = _median_filter(tile)
    normalized = Image.fromarray(restored, mode="L").resize(
        (TEMPLATE_SIZE, TEMPLATE_SIZE), Image.Resampling.BILINEAR
    )
    values = np.asarray(normalized, dtype=np.float32)
    darkness = np.clip((HIGH_TONE - values) / (HIGH_TONE - LOW_TONE), 0.0, 1.0)
    scores = {}
    for symbol in SYMBOLS:
        template = _symbol_mask(symbol, TEMPLATE_SIZE).astype(np.float32)
        scores[symbol] = float(1.0 - np.mean(np.abs(darkness - template)))
    return scores


def _top_margin(scores: dict[str, float]) -> float:
    values = sorted(scores.values(), reverse=True)
    return float(values[0] - values[1])


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def margin(scene: dict) -> float:
    return _top_margin(_classification_scores(_encoded_tile(scene)))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.035 or float(scene["noise_rate"]) > 0.43


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases are claimed; every sampled field is rendered."""

    return []
