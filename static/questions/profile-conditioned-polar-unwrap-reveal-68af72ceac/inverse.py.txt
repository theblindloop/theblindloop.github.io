"""Independent pixel-only inverse arm for polar_unwrap_reveal.

Receives ONLY the final PNG. It locates the ring (fiducial), executes the
declared raster transform (unwrap: radius->row, angle->column), extracts the
reconstructed glyph grid, and classifies it against the two block candidates A
and B declared by the public question contract. No scene data, no analytic gold.

Polarity is recovered from the pixels: the dominant canvas color is the
background; ink is anything that differs strongly from it. This makes the arm
invariant to the global colour inversion used as a label-preserving nuisance.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "polar_unwrap_reveal-oracle-0.1.0"

CX = 256.0
CY = 256.0
R_IN = 130.0
R_OUT = 196.0
G_ROWS = 15
G_COLS = 12
CELL = 3

TEMPLATE_A = [
    list("............"),
    list("....##......"),
    list("....##......"),
    list("...#..#....."),
    list("...#..#....."),
    list("..#....#...."),
    list("..#....#...."),
    list(".#......#..."),
    list(".#......#..."),
    list(".########..."),
    list("#........#.."),
    list("#........#.."),
    list("#........#.."),
    list("............"),
    list("............"),
]
TEMPLATE_B = [
    list("............"),
    list("##########.."),
    list("#........#.."),
    list("#........#.."),
    list("#........#.."),
    list("##########.."),
    list("#........#.."),
    list("#........#.."),
    list("#........#.."),
    list("##########.."),
    list("............"),
    list("............"),
    list("............"),
    list("............"),
    list("............"),
]
TEMPLATES = {"A": TEMPLATE_A, "B": TEMPLATE_B}


def _template_mask(glyph: str) -> np.ndarray:
    return np.array([[c == "#" for c in row] for row in TEMPLATES[glyph]], dtype=bool)


def _unwrap(mask: np.ndarray) -> np.ndarray:
    grid = np.zeros((G_ROWS, G_COLS), dtype=bool)
    for row in range(G_ROWS):
        for col in range(G_COLS):
            frac = col / G_COLS
            theta = 2.0 * np.pi * frac
            r = R_IN + (R_OUT - R_IN) * (row + 0.5) / G_ROWS
            x = int(round(CX + r * np.cos(theta)))
            y = int(round(CY + r * np.sin(theta)))
            votes = 0
            for dy in range(-CELL // 2, CELL // 2 + 1):
                for dx in range(-CELL // 2, CELL // 2 + 1):
                    yy = y + dy
                    xx = x + dx
                    if 0 <= yy < mask.shape[0] and 0 <= xx < mask.shape[1] and mask[yy, xx]:
                        votes += 1
            grid[row, col] = votes > 0
    return grid


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("L"), dtype=np.uint8)
    bg_value = float(np.bincount(arr.ravel(), minlength=256).argmax())
    ink = np.abs(arr.astype(np.float32) - bg_value) > 40.0
    grid = _unwrap(ink)
    scores = {}
    for glyph, tmpl in TEMPLATES.items():
        t = _template_mask(glyph)
        scores[glyph] = float((grid & t).sum()) / float(t.sum())
    return "A" if scores["A"] >= scores["B"] else "B"


def reveal_margin(image: Image.Image) -> float:
    arr = np.asarray(image.convert("L"), dtype=np.uint8)
    bg_value = float(np.bincount(arr.ravel(), minlength=256).argmax())
    ink = np.abs(arr.astype(np.float32) - bg_value) > 40.0
    grid = _unwrap(ink)
    scores = {}
    for glyph, tmpl in TEMPLATES.items():
        t = _template_mask(glyph)
        scores[glyph] = float((grid & t).sum()) / float(t.sum())
    return abs(scores["A"] - scores["B"])
