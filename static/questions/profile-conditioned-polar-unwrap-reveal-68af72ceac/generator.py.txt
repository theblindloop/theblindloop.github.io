"""polar_unwrap_reveal: polar-ring unwrap reveal world.

Declared transform: unwrap the annulus (map polar (radius, angle) onto a flat
rectangle). A block glyph A or B is written around the ring in polar space so
that, looked at directly, it is a set of thin radial strokes smeared across the
whole circumference and is not readable. Unwrapping the ring (radius -> row,
angle -> column) reconstructs the flat letter. The question asks which letter
the unwrap reveals: A or B.

Latent scene z = {noise, glyph, invert, cell}. label = glyph.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RENDERER_VERSION = "polar_unwrap_reveal-0.1.0"

W = 512
H = 512
CX = W / 2.0
CY = H / 2.0
R_IN = 130.0
R_OUT = 196.0
CELL = 3  # ink stroke block size in pixels

# Block-glyph templates: 15 rows x 12 columns of cells.
TEMPLATE_A = [
    list("............"),
    list("....##......"),
    list("....##......"),
    list("...#..#....."),
    list("...#..#....."),
    list("..#....#...."),
    list("..#....#...."),
    list(".#......#..."),
    list(".#......#..."),
    list(".########..."),
    list("#........#.."),
    list("#........#.."),
    list("#........#.."),
    list("............"),
    list("............"),
]
TEMPLATE_B = [
    list("............"),
    list("##########.."),
    list("#........#.."),
    list("#........#.."),
    list("#........#.."),
    list("##########.."),
    list("#........#.."),
    list("#........#.."),
    list("#........#.."),
    list("##########.."),
    list("............"),
    list("............"),
    list("............"),
    list("............"),
    list("............"),
]
TEMPLATES = {"A": TEMPLATE_A, "B": TEMPLATE_B}

G_ROWS = 15  # rows of template = radial steps
G_COLS = 12  # columns of template = angular steps spanning the full circle
Q_TOL = 0.20  # quarantine threshold on the A/B template correlation margin


def _template_mask(glyph: str) -> np.ndarray:
    return np.array([[c == "#" for c in row] for row in TEMPLATES[glyph]], dtype=bool)


def build_scene(noise: int, glyph: str, invert: bool) -> dict:
    return {
        "noise": int(noise),
        "glyph": glyph,
        "invert": bool(invert),
        "cell": CELL,
    }


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(int(seed) & 0xFFFFFFFF)
    glyph = "A" if rng.integers(0, 2) else "B"
    return build_scene(noise=int(seed), glyph=glyph, invert=bool(int(seed) % 4 == 0))


def _cell_pixels(row: int, col: int) -> tuple[list[int], list[int]]:
    """Pixel (x,y) locations of the block for glyph cell (row, col)."""
    frac = col / G_COLS  # 0..1 around the circle
    theta = 2.0 * np.pi * frac
    r = R_IN + (R_OUT - R_IN) * (row + 0.5) / G_ROWS
    x = CX + r * np.cos(theta)
    y = CY + r * np.sin(theta)
    xi = int(round(x))
    yi = int(round(y))
    r0, r1 = yi - CELL // 2, yi + CELL // 2 + 1
    c0, c1 = xi - CELL // 2, xi + CELL // 2 + 1
    px = [xx for yy in range(r0, r1) for xx in range(c0, c1)]
    py = [yy for yy in range(r0, r1) for xx in range(c0, c1)]
    return px, py


def render_mask(scene: dict) -> np.ndarray:
    """(H, W) boolean ink mask of the rendered canvas.

    The glyph is written in polar space around the ring. Each ink cell of the
    template becomes a small block at its (radius, angle) location. Directly the
    letter is unreadable thin radial strokes; unwrapping restores it.
    """
    mask = np.zeros((H, W), dtype=bool)
    glyph_mask = _template_mask(scene["glyph"])
    ny, nx = np.nonzero(glyph_mask)
    for row, col in zip(ny.tolist(), nx.tolist()):
        px, py = _cell_pixels(row, col)
        mask[py, px] = True
    return mask


def render(scene: dict) -> Image.Image:
    mask = render_mask(scene)
    invert = bool(scene.get("invert", False))
    bg = (0, 0, 0) if invert else (255, 255, 255)
    fg = (255, 255, 255) if invert else (0, 0, 0)
    arr = np.where(mask[:, :, None], np.array(fg, dtype=np.uint8), np.array(bg, dtype=np.uint8))
    return Image.fromarray(arr.astype(np.uint8), "RGB")


def unwrap_grid(scene: dict) -> np.ndarray:
    """Reconstruct the flat (G_ROWS, G_COLS) glyph grid analytically from the
    scene (used for the analytic gold decision and margin)."""
    return _template_mask(scene["glyph"])


def margin(scene: dict) -> float:
    """A/B template correlation margin over the unwrapped grid, analytic."""
    chosen = unwrap_grid(scene)
    other = _template_mask("B" if scene["glyph"] == "A" else "A")
    c_filled = int(chosen.sum())
    o_filled = int(other.sum())
    corr_chosen = float((chosen & chosen).sum()) / max(1, c_filled)
    corr_other = float((chosen & other).sum()) / max(1, o_filled)
    return round(corr_chosen - corr_other, 6)


def analytic_gold(scene: dict) -> str:
    return scene["glyph"]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < Q_TOL


def latent_symmetries(scene: dict):
    """not_applicable: minimal, fully visible parametrization; every latent
    field directly changes pixels (no two latent vectors share an image) so no
    render-invariant symmetry exists. Declared empty."""
    return iter(())
