"""Independent image-only inverse for converted bottleneck capacity."""

from __future__ import annotations
from fractions import Fraction
import numpy as np
from PIL import Image

CANVAS = (1200, 760)
INK = (31, 38, 46)
BACKGROUND = (255, 255, 255)
PANEL = (247, 249, 252)
CARD = (232, 237, 242)
TAG_RGB = {"amber": (226, 151, 28), "cyan": (18, 157, 181), "violet": (141, 77, 190)}
ALLOWED = {INK, BACKGROUND, PANEL, CARD, *TAG_RGB.values()}
ROW_Y = (35, 275, 515)
CAP_X = (100, 600, 1100)
RATE_X = (350, 850)
SEGMENT_POINTS = {"a": (11, 2), "b": (21, 12), "c": (21, 32), "d": (11, 42), "e": (2, 32), "f": (2, 12), "g": (11, 22)}
SEGMENT_DIGIT = {
    frozenset("abcdef"): 0, frozenset("bc"): 1, frozenset("abdeg"): 2,
    frozenset("abcdg"): 3, frozenset("bcfg"): 4, frozenset("acdfg"): 5,
    frozenset("acdefg"): 6, frozenset("abc"): 7, frozenset("abcdefg"): 8,
    frozenset("abcdfg"): 9,
}


class OracleError(ValueError):
    pass


def _digit(arr: np.ndarray, x: int, y: int) -> int:
    active = set()
    for name, (dx, dy) in SEGMENT_POINTS.items():
        patch = arr[y + dy - 1:y + dy + 2, x + dx - 1:x + dx + 2]
        if patch.shape != (3, 3, 3):
            raise OracleError("digit patch outside canvas")
        ink = int(np.all(patch == INK, axis=2).sum())
        if ink >= 7:
            active.add(name)
        elif ink != 0:
            raise OracleError("damaged seven-segment stroke")
    try:
        return SEGMENT_DIGIT[frozenset(active)]
    except KeyError as exc:
        raise OracleError(f"invalid seven-segment digit {sorted(active)}") from exc


def _number(arr: np.ndarray, cx: int, cy: int) -> int:
    y = cy - 14
    value = 10 * _digit(arr, cx - 27, y) + _digit(arr, cx + 4, y)
    if not 10 <= value <= 99:
        raise OracleError("capacity outside range")
    patch = arr[cy - 65:cy + 64, cx - 64:cx + 65]
    if int(np.all(patch == CARD, axis=2).sum()) < 7000:
        raise OracleError("capacity card is damaged")
    return value


def _rate(arr: np.ndarray, cx: int, cy: int) -> tuple[int, int]:
    y = cy - 22
    p, q = _digit(arr, cx - 37, y), _digit(arr, cx + 12, y)
    if p < 2 or q < 1 or p <= q:
        raise OracleError("invalid conversion")
    for yy in (cy - 9, cy + 9):
        if not np.all(arr[yy - 2:yy + 3, cx - 3:cx + 2] == INK):
            raise OracleError("conversion colon is missing")
    return p, q


def _route_color(arr: np.ndarray, y: int) -> str:
    counts = {name: int(np.all(arr[y:y + 12, 200:1000] == rgb, axis=2).sum()) for name, rgb in TAG_RGB.items()}
    names = [name for name, count in counts.items() if count > 6000]
    if len(names) != 1:
        raise OracleError("line border color is missing or ambiguous")
    return names[0]


def _validate_arrows(arr: np.ndarray, cy: int) -> None:
    for x0, x1 in ((175, 282), (418, 520), (675, 782), (918, 1020)):
        patch = arr[cy - 2:cy + 3, x0:x1]
        if int(np.all(patch == INK, axis=2).sum()) < 450:
            raise OracleError("flow arrow is broken")


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    if rgb.size != CANVAS:
        raise OracleError("wrong canvas")
    colors = rgb.getcolors(maxcolors=64)
    if colors is None or not {tuple(color) for _, color in colors} <= ALLOWED:
        raise OracleError("unknown raster color")
    arr = np.asarray(rgb)
    colors_seen = []
    line_values = []
    for y in ROW_Y:
        cy = y + 124
        colors_seen.append(_route_color(arr, y))
        _validate_arrows(arr, cy)
        caps = [_number(arr, cx, cy) for cx in CAP_X]
        rates = [_rate(arr, cx, cy) for cx in RATE_X]
        converted = (
            Fraction(caps[0]) * Fraction(rates[0][1], rates[0][0]) * Fraction(rates[1][1], rates[1][0]),
            Fraction(caps[1]) * Fraction(rates[1][1], rates[1][0]),
            Fraction(caps[2]),
        )
        line_values.append(min(converted))
    if set(colors_seen) != set(TAG_RGB):
        raise OracleError("line colors are not a permutation")
    maximum = max(line_values)
    matches = [i for i, value in enumerate(line_values) if value == maximum]
    if len(matches) != 1:
        raise OracleError("line capacities do not have a unique maximum")
    return colors_seen[matches[0]]
