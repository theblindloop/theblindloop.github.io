"""Render parallel production lines with converted stage bottlenecks."""

from __future__ import annotations

from fractions import Fraction
import random

from PIL import Image, ImageDraw

CANVAS = (1200, 760)
ANSWERS = ("amber", "cyan", "violet")
RGB = {
    "background": (255, 255, 255), "panel": (247, 249, 252), "ink": (31, 38, 46),
    "card": (232, 237, 242), "amber": (226, 151, 28), "cyan": (18, 157, 181),
    "violet": (141, 77, 190),
}
ROW_Y = (35, 275, 515)
ROW_H = 210
CAP_X = (100, 600, 1100)
RATE_X = (350, 850)
DIGIT_SEGMENTS = {
    0: "abcdef", 1: "bc", 2: "abdeg", 3: "abcdg", 4: "bcfg",
    5: "acdfg", 6: "acdefg", 7: "abc", 8: "abcdefg", 9: "abcdfg",
}
BLOCK_FONT = {
    "M": ("10001", "11011", "10101", "10101", "10001", "10001", "10001"),
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "X": ("10001", "10001", "01010", "00100", "01010", "10001", "10001"),
    "I": ("111", "010", "010", "010", "010", "010", "111"),
    "N": ("10001", "11001", "10101", "10101", "10101", "10011", "10001"),
    "P": ("11110", "10001", "10001", "11110", "10000", "10000", "10000"),
    "U": ("10001", "10001", "10001", "10001", "10001", "10001", "01110"),
    "T": ("11111", "00100", "00100", "00100", "00100", "00100", "00100"),
    "D": ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
    "F": ("11111", "10000", "10000", "11110", "10000", "10000", "10000"),
    "S": ("11111", "10000", "10000", "11111", "00001", "00001", "11111"),
    "H": ("10001", "10001", "10001", "11111", "10001", "10001", "10001"),
}
CONVERSIONS = ((2, 1), (3, 2), (4, 3), (5, 4), (3, 1), (5, 3), (4, 1), (5, 2))
CAPACITIES = tuple(range(18, 97, 3))


def _finish_limits(caps: list[int], rates: list[list[int]]) -> tuple[Fraction, Fraction, Fraction]:
    if len(caps) != 3 or len(rates) != 2:
        raise ValueError("each line needs three capacities and two conversions")
    r0 = Fraction(int(rates[0][1]), int(rates[0][0]))
    r1 = Fraction(int(rates[1][1]), int(rates[1][0]))
    return Fraction(int(caps[0])) * r0 * r1, Fraction(int(caps[1])) * r1, Fraction(int(caps[2]))


def _line_capacity(caps: list[int], rates: list[list[int]]) -> Fraction:
    return min(_finish_limits(caps, rates))


def sample_scene(seed: int) -> dict:
    rng = random.Random((int(seed) ^ 0xB0771E) & 0xFFFFFFFFFFFFFFFF)
    for _ in range(10000):
        lines = []
        for bottleneck_slot in rng.sample((0, 1, 2), 3):
            conversions = [list(rng.choice(CONVERSIONS)), list(rng.choice(CONVERSIONS))]
            candidates = []
            for _attempt in range(400):
                capacities = [rng.choice(CAPACITIES) for _ in range(3)]
                line = {"capacities": capacities, "conversions": conversions}
                limits = _finish_limits(capacities, conversions)
                if limits.index(min(limits)) == bottleneck_slot and len(set(limits)) == 3:
                    candidates.append((min(limits), line))
            if not candidates:
                break
            _, line = rng.choice(candidates)
            lines.append(line)
        if len(lines) != 3:
            continue
        values = [_line_capacity(line["capacities"], line["conversions"]) for line in lines]
        order = sorted(range(3), key=lambda i: values[i], reverse=True)
        if len(set(values)) != 3:
            continue
        relative_gap = (values[order[0]] - values[order[1]]) / values[order[1]]
        spread = (values[order[0]] - values[order[2]]) / values[order[2]]
        if not (Fraction(1, 30) <= relative_gap <= Fraction(1, 6) and spread <= Fraction(1, 2)):
            continue
        target_slot = order[0]
        answer_color = ANSWERS[(int(seed) // 3) % 3]
        remaining = [c for c in ANSWERS if c != answer_color]
        rng.shuffle(remaining)
        colors: list[str | None] = [None, None, None]
        colors[target_slot] = answer_color
        for slot in range(3):
            if colors[slot] is None:
                colors[slot] = remaining.pop()
        scene = {
            "capacities": [line["capacities"] for line in lines],
            "conversions": [line["conversions"] for line in lines],
            "tag_colors": colors,
        }
        if not is_quarantined(scene):
            return scene
    raise RuntimeError("could not construct a valid capacity scene")


def _draw_digit(draw: ImageDraw.ImageDraw, digit: int, x: int, y: int) -> None:
    boxes = {
        "a": (x + 4, y, x + 19, y + 4), "b": (x + 19, y + 4, x + 23, y + 21),
        "c": (x + 19, y + 23, x + 23, y + 40), "d": (x + 4, y + 40, x + 19, y + 44),
        "e": (x, y + 23, x + 4, y + 40), "f": (x, y + 4, x + 4, y + 21),
        "g": (x + 4, y + 20, x + 19, y + 24),
    }
    for segment in DIGIT_SEGMENTS[int(digit)]:
        draw.rectangle(boxes[segment], fill=RGB["ink"])


def _draw_number(draw: ImageDraw.ImageDraw, value: int, cx: int, y: int) -> None:
    if not 10 <= int(value) <= 99:
        raise ValueError("capacity must have two digits")
    tens, ones = divmod(int(value), 10)
    _draw_digit(draw, tens, cx - 27, y)
    _draw_digit(draw, ones, cx + 4, y)


def _draw_rate(draw: ImageDraw.ImageDraw, cx: int, cy: int, pair: list[int]) -> None:
    draw.rounded_rectangle((cx - 61, cy - 36, cx + 61, cy + 36), radius=11,
                           fill=RGB["background"], outline=RGB["ink"], width=4)
    y = cy - 22
    _draw_digit(draw, int(pair[0]), cx - 37, y)
    draw.ellipse((cx - 5, cy - 13, cx + 3, cy - 5), fill=RGB["ink"])
    draw.ellipse((cx - 5, cy + 5, cx + 3, cy + 13), fill=RGB["ink"])
    _draw_digit(draw, int(pair[1]), cx + 12, y)


def _block_text(draw: ImageDraw.ImageDraw, text: str, center_x: int, top: int, scale: int = 3) -> None:
    widths = [len(BLOCK_FONT[ch][0]) * scale for ch in text]
    total = sum(widths) + (len(text) - 1) * 2 * scale
    x = center_x - total // 2
    for ch, width in zip(text, widths):
        for row, bits in enumerate(BLOCK_FONT[ch]):
            for col, bit in enumerate(bits):
                if bit == "1":
                    draw.rectangle((x + col * scale, top + row * scale,
                                    x + (col + 1) * scale - 1, top + (row + 1) * scale - 1), fill=RGB["ink"])
        x += width + 2 * scale


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS, RGB["background"])
    draw = ImageDraw.Draw(image)
    for y, capacities, conversions, color in zip(ROW_Y, scene["capacities"], scene["conversions"], scene["tag_colors"]):
        draw.rounded_rectangle((18, y, 1182, y + ROW_H), radius=22, fill=RGB["panel"], outline=RGB[str(color)], width=10)
        cy = y + 124
        for cx, label in zip(CAP_X, ("INPUT", "MID", "FINISH")):
            _block_text(draw, label, cx, y + 20, 2)
        for edge in range(2):
            left = CAP_X[edge] + 72
            right = CAP_X[edge + 1] - 76
            draw.line((left, cy, right, cy), fill=RGB["ink"], width=8)
            draw.polygon(((right, cy), (right - 22, cy - 16), (right - 22, cy + 16)), fill=RGB["ink"])
        for cx, capacity in zip(CAP_X, capacities):
            draw.rounded_rectangle((cx - 69, cy - 70, cx + 69, cy + 68), radius=14,
                                   fill=RGB["card"], outline=RGB["ink"], width=5)
            _block_text(draw, "MAX", cx, cy - 57, 3)
            _draw_number(draw, int(capacity), cx, cy - 14)
        for cx, pair in zip(RATE_X, conversions):
            _draw_rate(draw, cx, cy, pair)
    return image


def analytic_gold(scene: dict) -> str:
    if len(scene["capacities"]) != 3 or len(scene["conversions"]) != 3 or set(scene["tag_colors"]) != set(ANSWERS):
        raise ValueError("malformed line binding")
    values = [_line_capacity(caps, rates) for caps, rates in zip(scene["capacities"], scene["conversions"])]
    maximum = max(values)
    matches = [i for i, value in enumerate(values) if value == maximum]
    if len(matches) != 1:
        raise ValueError("line capacities do not have a unique maximum")
    return str(scene["tag_colors"][matches[0]])


def margin(scene: dict) -> float:
    values = sorted((_line_capacity(caps, rates) for caps, rates in zip(scene["capacities"], scene["conversions"])), reverse=True)
    return float((values[0] - values[1]) / values[1]) if len(values) == 3 and values[0] > values[1] else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        if any(len(set(_finish_limits(caps, rates))) != 3 for caps, rates in zip(scene["capacities"], scene["conversions"])):
            return True
        return not (1 / 30 <= margin(scene) <= 1 / 6) or analytic_gold(scene) not in ANSWERS
    except (KeyError, TypeError, ValueError, ZeroDivisionError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
