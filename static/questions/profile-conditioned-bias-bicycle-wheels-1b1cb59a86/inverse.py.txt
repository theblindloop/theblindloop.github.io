"""Independent PNG-only inverse arm for the bicycle-wheel question."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

MIN_COMPONENT_AREA = 850
MAX_COMPONENT_AREA = 7000


def _colored_mask(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    hi = arr.max(axis=2)
    lo = arr.min(axis=2)
    # Keeps all three supported saturated wheel palettes, while excluding gray
    # road/frame/background. Orange reflectors survive but are tiny components.
    return ((hi - lo) >= 58) & (hi >= 105)


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[int, int, int, int, int]] = []
    for y0, x0 in zip(*np.nonzero(mask & ~seen)):
        if seen[y0, x0]:
            continue
        queue = deque([(int(y0), int(x0))])
        seen[y0, x0] = True
        area = 0
        xmin = xmax = int(x0)
        ymin = ymax = int(y0)
        while queue:
            y, x = queue.popleft()
            area += 1
            xmin, xmax = min(xmin, x), max(xmax, x)
            ymin, ymax = min(ymin, y), max(ymax, y)
            for yy, xx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= yy < h and 0 <= xx < w and mask[yy, xx] and not seen[yy, xx]:
                    seen[yy, xx] = True
                    queue.append((yy, xx))
        found.append((area, xmin, ymin, xmax, ymax))
    return found


def decision_from_image(image: Image.Image) -> str:
    """Count large, approximately square saturated ring components from pixels."""
    rings = 0
    for area, xmin, ymin, xmax, ymax in _components(_colored_mask(image)):
        width = xmax - xmin + 1
        height = ymax - ymin + 1
        ratio = width / max(1.0, float(height))
        if MIN_COMPONENT_AREA <= area <= MAX_COMPONENT_AREA and 0.86 <= ratio <= 1.14 and width >= 90:
            rings += 1
    if rings not in (1, 2, 3):
        raise ValueError(f"raster does not contain 1, 2, or 3 measurable wheel rings (found {rings})")
    return str(rings)
