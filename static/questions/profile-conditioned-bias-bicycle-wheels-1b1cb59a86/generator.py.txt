"""Deterministic latent sampler and Pillow renderer for bicycle-wheel counts."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 384
SCALE = 3
BACKGROUND = (248, 247, 242)
INK = (37, 43, 49)
ROAD = (155, 158, 160)
PALETTES = ((28, 122, 205), (26, 151, 91), (184, 55, 151))
SLOT_X = (124, 256, 388)
BASE_Y = 268


def sample_scene(seed: int) -> dict:
    """Sample matched scenes; consecutive seeds cycle through all answer classes."""
    rng = random.Random(int(seed))
    wheel_count = int(seed) % 3 + 1
    selected = sorted(rng.sample(range(3), wheel_count))
    dx = rng.randint(-9, 9)
    dy = rng.randint(-4, 4)
    radius = rng.randint(57, 60)
    ring_width = rng.randint(8, 10)
    palette = list(PALETTES[rng.randrange(len(PALETTES))])
    phase = rng.randrange(0, 30)

    # The frame is stable across count classes. Segment order is a true latent
    # alias because all segments are rendered with the same opaque ink.
    segments = [
        [124, 268, 202, 178],
        [202, 178, 275, 268],
        [275, 268, 124, 268],
        [202, 178, 330, 178],
        [330, 178, 388, 268],
        [275, 268, 330, 178],
        [194, 168, 218, 168],
        [330, 178, 350, 135],
        [350, 135, 374, 135]
    ]
    return {
        "wheel_count": wheel_count,
        "selected_slots": selected,
        "offset_x": dx,
        "offset_y": dy,
        "wheel_radius": radius,
        "ring_width": ring_width,
        "palette_rgb": palette,
        "spoke_phase": phase,
        "frame_segments": segments,
    }


def _validate_scene(scene: dict) -> None:
    if int(scene["wheel_count"]) != len(scene["selected_slots"]):
        raise ValueError("wheel_count and selected_slots disagree")
    if not 1 <= int(scene["wheel_count"]) <= 3:
        raise ValueError("wheel_count must be 1, 2, or 3")


def _line(draw: ImageDraw.ImageDraw, points: list[tuple[int, int]], fill: tuple[int, int, int], width: int) -> None:
    draw.line([(x * SCALE, y * SCALE) for x, y in points], fill=fill, width=width * SCALE, joint="curve")


def render(scene: dict) -> Image.Image:
    """Render only public visual fields into one antialiased RGB image."""
    _validate_scene(scene)
    dx = int(scene["offset_x"])
    dy = int(scene["offset_y"])
    radius = int(scene["wheel_radius"])
    ring_width = int(scene["ring_width"])
    color = tuple(int(v) for v in scene["palette_rgb"])
    phase = math.radians(int(scene["spoke_phase"]))

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _line(draw, [(42, BASE_Y + radius + 9 + dy), (470, BASE_Y + radius + 9 + dy)], ROAD, 3)

    # Thin spokes are rendered first; colored rings are rendered last so every
    # counted component is visibly complete rather than hidden by a spoke.
    for slot in scene["selected_slots"]:
        cx = SLOT_X[int(slot)] + dx
        cy = BASE_Y + dy
        for k in range(8):
            angle = phase + k * math.pi / 4.0
            ex = int(round(cx + (radius - 3) * math.cos(angle)))
            ey = int(round(cy + (radius - 3) * math.sin(angle)))
            _line(draw, [(cx, cy), (ex, ey)], INK, 2)
        rr = 4 * SCALE
        draw.ellipse((cx * SCALE - rr, cy * SCALE - rr, cx * SCALE + rr, cy * SCALE + rr), fill=INK)

    for x1, y1, x2, y2 in scene["frame_segments"]:
        _line(draw, [(x1 + dx, y1 + dy), (x2 + dx, y2 + dy)], INK, 7)

    # Small fixed orange reflectors prevent a single-color icon while remaining
    # far below the oracle's component-size threshold.
    for x, y in ((202, 178), (330, 178)):
        rr = 5 * SCALE
        draw.ellipse(((x + dx) * SCALE - rr, (y + dy) * SCALE - rr,
                      (x + dx) * SCALE + rr, (y + dy) * SCALE + rr), fill=(226, 126, 34))

    for slot in scene["selected_slots"]:
        cx = (SLOT_X[int(slot)] + dx) * SCALE
        cy = (BASE_Y + dy) * SCALE
        r = radius * SCALE
        draw.ellipse((cx - r, cy - r, cx + r, cy + r), outline=color, width=ring_width * SCALE)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(int(scene["wheel_count"]))


def margin(scene: dict) -> float:
    selected = sorted(int(v) for v in scene["selected_slots"])
    if len(selected) < 2:
        return 32.0
    radius = float(scene["wheel_radius"])
    return min(float(SLOT_X[b] - SLOT_X[a]) - 2.0 * radius for a, b in zip(selected, selected[1:]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["frame_segments"] = list(reversed(twin["frame_segments"]))
    return [("reverse_frame_segment_construction_order", twin)]
