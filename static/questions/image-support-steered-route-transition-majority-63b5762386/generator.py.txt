"""Deterministic latent sampler and raster renderer."""

from __future__ import annotations

import itertools
import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 768
HEIGHT = 512
SCALE = 2
BACKGROUND = "#f7f3e8"
NAVY = "#1e3049"
CYAN = "#00aeb9"
ORANGE = "#ee7b24"
GREEN = "#25ab5b"
RED = "#d63e41"
TETHER = "#92979d"


def _transition_count(colors: list[str]) -> int:
    return sum(a != b for a, b in zip(colors, colors[1:]))


def _sequence_pool(decision: str, preferred: int) -> list[list[str]]:
    pool: list[list[str]] = []
    for cyan_positions in itertools.combinations(range(8), 4):
        cyan = set(cyan_positions)
        seq = ["cyan" if i in cyan else "orange" for i in range(8)]
        transitions = _transition_count(seq)
        winner = "switches" if transitions > 7 - transitions else "repeats"
        if winner == decision and transitions == preferred:
            pool.append(seq)
    return pool


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    decision = "switches" if seed % 2 == 0 else "repeats"
    preferred = (4, 5, 6)[(seed // 2) % 3] if decision == "switches" else (1, 2, 3)[(seed // 2) % 3]
    pool = _sequence_pool(decision, preferred)
    colors = list(pool[rng.randrange(len(pool))])

    xs = [56 + i * 73 for i in range(10)]
    ys: list[int] = []
    last = rng.randint(215, 297)
    for i in range(10):
        target = rng.randint(205, 307)
        if i and abs(target - last) < 38:
            target = 205 if last > 256 else 307
        ys.append(target)
        last = target
    route_points = [[x, y] for x, y in zip(xs, ys)]
    checkpoint_xs = [116 + 75 * i + rng.randint(-9, 9) for i in range(8)]
    checkpoint_sides = [rng.choice([-1, 1]) for _ in range(8)]
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": BACKGROUND,
        "route_points": route_points,
        "checkpoint_xs": checkpoint_xs,
        "checkpoint_colors": colors,
        "checkpoint_sides": checkpoint_sides,
    }


def _route_y(points: list[list[int]], x: float) -> float:
    for (x0, y0), (x1, y1) in zip(points, points[1:]):
        if x0 <= x <= x1:
            t = (x - x0) / (x1 - x0)
            return y0 + t * (y1 - y0)
    raise ValueError("checkpoint lies outside route")


def render(scene: dict[str, Any]) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    background = str(scene["background"])
    route_points = [[int(v) for v in point] for point in scene["route_points"]]
    checkpoint_xs = [int(v) for v in scene["checkpoint_xs"]]
    checkpoint_colors = [str(v) for v in scene["checkpoint_colors"]]
    checkpoint_sides = [int(v) for v in scene["checkpoint_sides"]]
    if not (len(checkpoint_xs) == len(checkpoint_colors) == len(checkpoint_sides) == 8):
        raise ValueError("exactly eight checkpoints are required")

    image = Image.new("RGB", (width * SCALE, height * SCALE), background)
    draw = ImageDraw.Draw(image)
    pts = [(x * SCALE, y * SCALE) for x, y in route_points]

    flag_centers: list[tuple[float, float]] = []
    for x, side in zip(checkpoint_xs, checkpoint_sides):
        y = _route_y(route_points, x)
        fy = y + side * 27
        flag_centers.append((x, fy))
        draw.line([(x * SCALE, y * SCALE), (x * SCALE, fy * SCALE)], fill=TETHER, width=3 * SCALE)

    draw.line(pts, fill=NAVY, width=11 * SCALE, joint="curve")
    for x, y in route_points[1:-1]:
        r = 5.5 * SCALE
        draw.ellipse((x * SCALE - r, y * SCALE - r, x * SCALE + r, y * SCALE + r), fill=NAVY)

    endpoint_radius = 13 * SCALE
    for (x, y), color in ((route_points[0], GREEN), (route_points[-1], RED)):
        cx, cy = x * SCALE, y * SCALE
        draw.ellipse(
            (cx - endpoint_radius, cy - endpoint_radius, cx + endpoint_radius, cy + endpoint_radius),
            fill=color,
            outline=BACKGROUND,
            width=3 * SCALE,
        )

    palette = {"cyan": CYAN, "orange": ORANGE}
    flag_radius = 10 * SCALE
    for (cx0, cy0), color_name in zip(flag_centers, checkpoint_colors):
        cx, cy = cx0 * SCALE, cy0 * SCALE
        draw.ellipse(
            (cx - flag_radius, cy - flag_radius, cx + flag_radius, cy + flag_radius),
            fill=palette[color_name],
            outline=background,
            width=2 * SCALE,
        )

    font = ImageFont.load_default()
    draw.text((34 * SCALE, 20 * SCALE), "START", fill=TETHER, font=font)
    draw.text(((width - 92) * SCALE, 20 * SCALE), "FINISH", fill=TETHER, font=font)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    colors = [str(value) for value in scene["checkpoint_colors"]]
    switches = _transition_count(colors)
    return "switches" if switches > len(colors) - 1 - switches else "repeats"


def margin(scene: dict[str, Any]) -> float:
    colors = [str(value) for value in scene["checkpoint_colors"]]
    switches = _transition_count(colors)
    return float(abs(switches - (len(colors) - 1 - switches)))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 1.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
