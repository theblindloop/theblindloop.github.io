"""Independent pixel-only inverse for the checkpoint-transition question."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_COLORS = {
    "navy": np.array((30, 48, 73), dtype=np.int16),
    "cyan": np.array((0, 174, 185), dtype=np.int16),
    "orange": np.array((238, 123, 36), dtype=np.int16),
    "green": np.array((37, 171, 91), dtype=np.int16),
    "red": np.array((214, 62, 65), dtype=np.int16),
}


def _mask(arr: np.ndarray, name: str, radius: int = 48) -> np.ndarray:
    delta = arr.astype(np.int32) - _COLORS[name].astype(np.int32)
    return np.sum(delta * delta, axis=2) <= radius * radius


def _components(mask: np.ndarray, minimum: int) -> list[tuple[float, float, int]]:
    work = mask.copy()
    h, w = work.shape
    found: list[tuple[float, float, int]] = []
    for y0, x0 in zip(*np.nonzero(work)):
        if not work[y0, x0]:
            continue
        work[y0, x0] = False
        queue = deque([(int(y0), int(x0))])
        ys: list[int] = []
        xs: list[int] = []
        while queue:
            y, x = queue.popleft()
            ys.append(y)
            xs.append(x)
            for yy, xx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= yy < h and 0 <= xx < w and work[yy, xx]:
                    work[yy, xx] = False
                    queue.append((yy, xx))
        if len(xs) >= minimum:
            found.append((float(np.mean(xs)), float(np.mean(ys)), len(xs)))
    return found


def _single_marker(arr: np.ndarray, name: str) -> tuple[float, float]:
    comps = _components(_mask(arr, name), 180)
    if len(comps) != 1:
        raise ValueError(f"expected one {name} endpoint marker")
    return comps[0][0], comps[0][1]


def _ordered_flags(arr: np.ndarray) -> list[str]:
    flags: list[tuple[float, float, str]] = []
    for name in ("cyan", "orange"):
        for x, y, area in _components(_mask(arr, name), 150):
            if not 180 <= area <= 430:
                raise ValueError("checkpoint marker has invalid area")
            flags.append((x, y, name))
    if len(flags) != 8 or sum(name == "cyan" for _, _, name in flags) != 4:
        raise ValueError("expected eight balanced checkpoint dots")
    flags.sort(key=lambda item: item[0])
    if min(b[0] - a[0] for a, b in zip(flags, flags[1:])) < 42:
        raise ValueError("checkpoint ordering is not pixel-resolved")
    return [name for _, _, name in flags]


def _validate_ordered_centerline(arr: np.ndarray, flags: list[tuple[float, float, str]]) -> None:
    navy = _mask(arr, "navy", 42)
    green_x, _ = _single_marker(arr, "green")
    red_x, _ = _single_marker(arr, "red")
    if red_x - green_x < arr.shape[1] * 0.72:
        raise ValueError("route does not span the image")
    left = int(round(green_x)) + 11
    right = int(round(red_x)) - 11
    centerline: dict[int, float] = {}
    missing_run = 0
    max_missing = 0
    for x in range(left, right + 1):
        ys = np.nonzero(navy[:, x])[0]
        if len(ys) >= 3:
            centerline[x] = float(np.median(ys))
            missing_run = 0
        else:
            missing_run += 1
            max_missing = max(max_missing, missing_run)
    if max_missing > 2 or len(centerline) < 0.96 * (right - left + 1):
        raise ValueError("navy centerline is broken")
    for x, y, _ in flags:
        xi = int(round(x))
        nearby = [centerline[col] for col in range(xi - 2, xi + 3) if col in centerline]
        if not nearby:
            raise ValueError("checkpoint cannot be bound to route")
        distance = abs(y - float(np.mean(nearby)))
        if not 13 <= distance <= 42:
            raise ValueError("checkpoint tether distance is invalid")


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    flags: list[tuple[float, float, str]] = []
    for name in ("cyan", "orange"):
        for x, y, area in _components(_mask(arr, name), 150):
            if not 180 <= area <= 430:
                raise ValueError("checkpoint marker has invalid area")
            flags.append((x, y, name))
    if len(flags) != 8 or sum(name == "cyan" for _, _, name in flags) != 4:
        raise ValueError("expected eight balanced checkpoint dots")
    flags.sort(key=lambda item: item[0])
    if min(b[0] - a[0] for a, b in zip(flags, flags[1:])) < 42:
        raise ValueError("checkpoint ordering is not pixel-resolved")
    _validate_ordered_centerline(arr, flags)
    colors = [name for _, _, name in flags]
    switches = sum(a != b for a, b in zip(colors, colors[1:]))
    repeats = len(colors) - 1 - switches
    if switches == repeats:
        raise ValueError("transition majority is tied")
    return "switches" if switches > repeats else "repeats"
