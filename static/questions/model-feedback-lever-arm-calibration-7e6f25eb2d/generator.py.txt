"""Deterministic sampler and renderer for coupled lever-arm calibration."""

from __future__ import annotations

import collections
import itertools
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1200, 960
SIZE = (WIDTH, HEIGHT)
BACKGROUND = (247, 245, 238)
CARD = (255, 253, 248)
CLUE_CARD = (239, 246, 250)
INK = (35, 42, 54)
MUTED = (104, 111, 120)
FULCRUM = (220, 187, 79)
COLORS = ((201, 67, 64), (48, 111, 176), (43, 147, 103), (222, 153, 35))
MASSES = (3, 4, 5, 7)
DECISIONS = ("beam-1", "beam-2", "beam-3", "beam-4")
STEP = 48
BEAM_Y_OFFSET = 122

# Each pair is (mass role, signed arm distance). Negative is left of the
# fulcrum. Every clue has zero torque for the unique positive scale 3:4:5:7.
CLUE_SPECS = (
    ((0, -4), (1, 3)),
    ((0, -1), (1, -1), (3, 1)),
    ((1, -1), (2, -1), (0, 3)),
    ((0, -1), (2, -1), (1, 2)),
)
CLUE_RECTS = ((30, 56, 585, 246), (615, 56, 1170, 246), (30, 266, 585, 456), (615, 266, 1170, 456))
OPTION_RECTS = ((30, 536, 585, 726), (615, 536, 1170, 726), (30, 746, 585, 936), (615, 746, 1170, 936))

_BOLD_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
FONT_TINY = ImageFont.truetype(_BOLD_PATH, 16)
FONT_MEDIUM = ImageFont.truetype(_BOLD_PATH, 28)


def _option_pools() -> dict[int, tuple[tuple[tuple[int, ...], tuple[int, ...]], ...]]:
    pools: dict[int, list[tuple[tuple[int, ...], tuple[int, ...]]]] = {0: [], 1: [], 2: [], 3: []}
    for sides in itertools.product((-1, 1), repeat=4):
        if len(set(sides)) != 2:
            continue
        for distances in itertools.product(range(1, 5), repeat=4):
            if max(collections.Counter(zip(sides, distances)).values()) > 2:
                continue
            torque = sum(side * distance * mass for side, distance, mass in zip(sides, distances, MASSES))
            delta = abs(torque)
            if delta in pools:
                pools[delta].append((tuple(sides), tuple(distances)))
    return {delta: tuple(items) for delta, items in pools.items()}


OPTION_POOLS = _option_pools()


def sample_scene(seed: int) -> dict:
    seed = int(seed)
    rng = random.Random(seed)
    correct = OPTION_POOLS[0][rng.randrange(len(OPTION_POOLS[0]))]
    distractors = [OPTION_POOLS[delta][rng.randrange(len(OPTION_POOLS[delta]))] for delta in (1, 2, 3)]
    answer_slot = seed % 4
    options = distractors[:]
    options.insert(answer_slot, correct)
    role_colors = list(range(4))
    rng.shuffle(role_colors)
    return {
        "role_colors": role_colors,
        "option_sides": [list(item[0]) for item in options],
        "option_distances": [list(item[1]) for item in options],
    }


def _validated(scene: dict) -> tuple[list[int], tuple[tuple[int, ...], ...], tuple[tuple[int, ...], ...]]:
    role_colors = [int(value) for value in scene["role_colors"]]
    sides = tuple(tuple(int(value) for value in row) for row in scene["option_sides"])
    distances = tuple(tuple(int(value) for value in row) for row in scene["option_distances"])
    if sorted(role_colors) != [0, 1, 2, 3]:
        raise ValueError("role_colors must be a permutation")
    if len(sides) != 4 or len(distances) != 4:
        raise ValueError("four test beams are required")
    for side_row, distance_row in zip(sides, distances):
        if len(side_row) != 4 or set(side_row) - {-1, 1} or len(set(side_row)) != 2:
            raise ValueError("each beam must place four loads on both sides")
        if len(distance_row) != 4 or any(value not in (1, 2, 3, 4) for value in distance_row):
            raise ValueError("arm distances must be in 1..4")
        if max(collections.Counter(zip(side_row, distance_row)).values()) > 2:
            raise ValueError("at most two loads may share one hook")
    return role_colors, sides, distances


def _torque(side_row: tuple[int, ...], distance_row: tuple[int, ...]) -> int:
    return sum(side * distance * mass for side, distance, mass in zip(side_row, distance_row, MASSES))


def analytic_gold(scene: dict) -> str:
    try:
        _colors, sides, distances = _validated(scene)
    except (KeyError, TypeError, ValueError):
        return "abstain"
    matches = [index for index, rows in enumerate(zip(sides, distances)) if _torque(*rows) == 0]
    return DECISIONS[matches[0]] if len(matches) == 1 else "abstain"


def margin(scene: dict) -> float:
    try:
        _colors, sides, distances = _validated(scene)
    except (KeyError, TypeError, ValueError):
        return 0.0
    wrong = [abs(_torque(*rows)) for rows in zip(sides, distances) if _torque(*rows) != 0]
    return float(min(wrong)) if wrong else 0.0


def is_quarantined(scene: dict) -> bool:
    return analytic_gold(scene) == "abstain" or margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _centered_text(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str, font: ImageFont.FreeTypeFont, fill: tuple[int, int, int]) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2 - box[1]), text, font=font, fill=fill)


def _draw_beam(draw: ImageDraw.ImageDraw, rect: tuple[int, int, int, int], loads: tuple[tuple[int, int], ...], role_colors: list[int], label: str, clue: bool) -> None:
    x0, y0, x1, y1 = rect
    draw.rounded_rectangle(rect, radius=16, fill=CLUE_CARD if clue else CARD, outline=INK, width=4)
    draw.rounded_rectangle((x0 + 12, y0 + 11, x0 + 112, y0 + 43), radius=8, fill=INK)
    _centered_text(draw, (x0 + 62, y0 + 27), label, FONT_TINY, (255, 255, 255))
    cx = (x0 + x1) // 2
    beam_y = y0 + BEAM_Y_OFFSET
    draw.line((cx - 4 * STEP - 11, beam_y, cx + 4 * STEP + 11, beam_y), fill=INK, width=8)
    for side in (-1, 1):
        for distance in range(1, 5):
            x = cx + side * distance * STEP
            draw.line((x, beam_y - 8, x, beam_y + 8), fill=INK, width=4)
            _centered_text(draw, (x, beam_y + 20), str(distance), FONT_TINY, MUTED)
    draw.polygon(((cx, beam_y + 6), (cx - 28, beam_y + 57), (cx + 28, beam_y + 57)), fill=FULCRUM, outline=INK)
    grouped: dict[int, list[int]] = {}
    for role, signed_distance in loads:
        grouped.setdefault(signed_distance, []).append(role)
    for signed_distance, roles in sorted(grouped.items()):
        x = cx + signed_distance * STEP
        draw.line((x, beam_y - 4, x, beam_y - 26 - 39 * (len(roles) - 1)), fill=INK, width=4)
        for level, role in enumerate(roles):
            y = beam_y - 31 - level * 39
            color = COLORS[role_colors[role]]
            draw.ellipse((x - 17, y - 17, x + 17, y + 17), fill=color, outline=INK, width=3)
            draw.ellipse((x - 4, y - 4, x + 4, y + 4), fill=(255, 255, 255))


def render(scene: dict) -> Image.Image:
    role_colors, sides, distances = _validated(scene)
    image = Image.new("RGB", SIZE, BACKGROUND)
    draw = ImageDraw.Draw(image)
    _centered_text(draw, (WIDTH // 2, 27), "LEVEL CLUES  ·  TICKS SHOW ARM DISTANCE", FONT_MEDIUM, INK)
    for index, (rect, spec) in enumerate(zip(CLUE_RECTS, CLUE_SPECS), 1):
        _draw_beam(draw, rect, spec, role_colors, f"CLUE {index}", True)
    _centered_text(draw, (WIDTH // 2, 498), "WHICH TEST BEAM STAYS LEVEL?", FONT_MEDIUM, INK)
    for index, (rect, side_row, distance_row) in enumerate(zip(OPTION_RECTS, sides, distances), 1):
        loads = tuple((role, side_row[role] * distance_row[role]) for role in range(4))
        _draw_beam(draw, rect, loads, role_colors, f"BEAM {index}", False)
    return image
