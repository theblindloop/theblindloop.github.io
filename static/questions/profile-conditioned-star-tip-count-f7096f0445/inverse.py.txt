"""Independent pixel-only inverse for the visible number of star tips.

The oracle does not know renderer state or scene data.  It segments the only
large non-background silhouette, measures its radial boundary around the pixel
centroid, and selects the dominant 4--7 cycle in that boundary.  The returned
decision is therefore a property of the raster alone.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    pass


def _foreground_mask(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    h, w, _ = arr.shape
    edge = max(4, min(h, w) // 32)
    border = np.concatenate(
        [
            arr[:edge].reshape(-1, 3),
            arr[-edge:].reshape(-1, 3),
            arr[:, :edge].reshape(-1, 3),
            arr[:, -edge:].reshape(-1, 3),
        ],
        axis=0,
    )
    background = np.median(border, axis=0)
    distance = np.linalg.norm(arr - background, axis=2)
    mask = distance > 45.0
    if int(mask.sum()) < 2500:
        raise OracleAbstention("no large foreground silhouette")
    ys, xs = np.nonzero(mask)
    if xs.min() <= 1 or ys.min() <= 1 or xs.max() >= w - 2 or ys.max() >= h - 2:
        raise OracleAbstention("foreground touches the image boundary")
    return mask


def _radial_profile(mask: np.ndarray, bins: int = 720) -> np.ndarray:
    ys, xs = np.nonzero(mask)
    # Area centroid is stable for every regular filled star and requires no
    # shared center coordinate.
    cx = float(xs.mean())
    cy = float(ys.mean())
    dx = xs.astype(np.float64) - cx
    dy = ys.astype(np.float64) - cy
    radii = np.hypot(dx, dy)
    angles = (np.arctan2(dy, dx) + 2.0 * math.pi) % (2.0 * math.pi)
    indices = np.floor(angles * bins / (2.0 * math.pi)).astype(np.int64) % bins
    profile = np.zeros(bins, dtype=np.float64)
    np.maximum.at(profile, indices, radii)
    known = np.flatnonzero(profile > 0)
    if len(known) < bins * 0.75:
        raise OracleAbstention("radial contour is too sparse")
    # Circular interpolation fills the few angular bins missed by pixel centers.
    extended_x = np.concatenate((known - bins, known, known + bins))
    extended_y = np.concatenate((profile[known], profile[known], profile[known]))
    profile = np.interp(np.arange(bins), extended_x, extended_y)
    kernel = np.ones(9, dtype=np.float64) / 9.0
    padded = np.concatenate((profile[-4:], profile, profile[:4]))
    return np.convolve(padded, kernel, mode="valid")


def _measure(image: Image.Image) -> tuple[int, float]:
    profile = _radial_profile(_foreground_mask(image))
    centered = profile - profile.mean()
    bins = len(profile)
    angles = 2.0 * math.pi * np.arange(bins) / bins
    amplitudes = {}
    for cycles in range(4, 8):
        coefficient = np.sum(centered * np.exp(-1j * cycles * angles))
        amplitudes[cycles] = float(abs(coefficient))
    ranked = sorted(amplitudes.items(), key=lambda item: item[1], reverse=True)
    winner, best = ranked[0]
    runner_up = ranked[1][1]
    decision_margin = best - runner_up
    if best < 2500.0 or decision_margin < 900.0 or best < runner_up * 1.45:
        raise OracleAbstention(
            f"ambiguous radial spectrum: best={best:.1f}, runner_up={runner_up:.1f}"
        )
    return winner, decision_margin


def decision_from_image(image: Image.Image) -> str:
    decision, _ = _measure(image)
    return str(decision)
