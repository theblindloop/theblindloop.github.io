"""Deterministic renderer for a single filled radial star silhouette."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 384
HEIGHT = 384
SCALE = 4
POINT_COUNTS = (4, 5, 6, 7)
FILLS = (
    (34, 75, 138),
    (145, 47, 74),
    (35, 112, 91),
    (109, 62, 151),
    (161, 83, 24),
    (35, 91, 112),
)
BACKGROUNDS = ((250, 248, 241), (245, 249, 252), (250, 246, 250), (247, 249, 244))


def sample_scene(seed: int) -> dict:
    """Sample balanced count identities with style independent of the count."""
    rng = random.Random(int(seed))
    points = POINT_COUNTS[int(seed) % len(POINT_COUNTS)]
    outer_radius = rng.uniform(112.0, 136.0)
    # Polygon area = n * R^2 * inner_ratio * sin(pi/n).  Sampling a
    # count-independent area factor removes count as a shortcut through ink area.
    area_factor = rng.uniform(1.28, 1.38)
    inner_ratio = area_factor / (points * math.sin(math.pi / points))
    return {
        "points": points,
        "rotation_deg": rng.uniform(0.0, 360.0),
        "center_x": rng.uniform(181.0, 203.0),
        "center_y": rng.uniform(181.0, 203.0),
        "outer_radius": outer_radius,
        "inner_ratio": inner_ratio,
        "fill_rgb": list(rng.choice(FILLS)),
        "background_rgb": list(rng.choice(BACKGROUNDS)),
    }


def _vertices(scene: dict) -> list[tuple[float, float]]:
    n = int(scene["points"])
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    outer = float(scene["outer_radius"])
    inner = outer * float(scene["inner_ratio"])
    base = math.radians(float(scene["rotation_deg"]) % 360.0)
    vertices = []
    for index in range(2 * n):
        radius = outer if index % 2 == 0 else inner
        angle = base + index * math.pi / n
        vertices.append((cx + radius * math.cos(angle), cy + radius * math.sin(angle)))
    return vertices


def render(scene: dict) -> Image.Image:
    background = tuple(int(v) for v in scene["background_rgb"])
    fill = tuple(int(v) for v in scene["fill_rgb"])
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), background)
    draw = ImageDraw.Draw(image)
    vertices = [(round(x * SCALE), round(y * SCALE)) for x, y in _vertices(scene)]
    draw.polygon(vertices, fill=fill)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(int(scene["points"]))


def margin(scene: dict) -> float:
    """Visible radial peak-to-notch depth, in final-image pixels."""
    return float(scene["outer_radius"]) * (1.0 - float(scene["inner_ratio"]))


def is_quarantined(scene: dict) -> bool:
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    radius = float(scene["outer_radius"])
    canvas_clearance = min(cx - radius, cy - radius, WIDTH - cx - radius, HEIGHT - cy - radius)
    return margin(scene) < 36.0 or canvas_clearance < 35.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["rotation_deg"] = float(twin["rotation_deg"]) + 360.0
    return [("rotation_plus_full_turn", twin)]
