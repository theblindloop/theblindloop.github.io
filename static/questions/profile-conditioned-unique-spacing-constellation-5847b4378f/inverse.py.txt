"""Independent image-only inverse for the unique-spacing constellation."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


_COLORS = ("red", "blue", "green")
_PALETTE = np.array(
    [[220, 50, 50], [45, 90, 210], [40, 170, 90]], dtype=np.float32
)
_DISTANCE_TOLERANCE = 3.0
_MIN_COMPONENT_AREA = 24


def _components(mask: np.ndarray) -> list[tuple[float, float]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    centers: list[tuple[float, float]] = []
    for y in range(height):
        for x in range(width):
            if not mask[y, x] or seen[y, x]:
                continue
            stack = [(y, x)]
            seen[y, x] = True
            pixels: list[tuple[int, int]] = []
            while stack:
                py, px = stack.pop()
                pixels.append((py, px))
                for dy in (-1, 0, 1):
                    for dx in (-1, 0, 1):
                        if not (dx or dy):
                            continue
                        qy, qx = py + dy, px + dx
                        if (
                            0 <= qy < height
                            and 0 <= qx < width
                            and mask[qy, qx]
                            and not seen[qy, qx]
                        ):
                            seen[qy, qx] = True
                            stack.append((qy, qx))
            if len(pixels) >= _MIN_COMPONENT_AREA:
                centers.append(
                    (
                        float(sum(px + 0.5 for py, px in pixels) / len(pixels)),
                        float(sum(py + 0.5 for py, px in pixels) / len(pixels)),
                    )
                )
    return centers


def _repeat_burden(points: list[tuple[float, float]]) -> int:
    distances = sorted(
        math.hypot(x1 - x2, y1 - y2)
        for i, (x1, y1) in enumerate(points)
        for x2, y2 in points[i + 1 :]
    )
    burden = 0
    i = 0
    while i < len(distances):
        j = i + 1
        while j < len(distances) and distances[j] - distances[i] <= _DISTANCE_TOLERANCE:
            j += 1
        if j - i > 1:
            burden += j - i - 1
        i = j
    return burden


def _centers_for_color(arr: np.ndarray, index: int) -> list[tuple[float, float]]:
    ink = _PALETTE[index]
    distance = np.linalg.norm(arr - ink, axis=2)
    other = np.delete(_PALETTE, index, axis=0)
    nearest_other = np.min(
        np.linalg.norm(arr[:, :, None, :] - other[None, None, :, :], axis=3), axis=2
    )
    mask = (distance < 170.0) & (distance <= nearest_other + 12.0)
    centers = _components(mask)
    if len(centers) != 8:
        raise ValueError(f"expected eight {_COLORS[index]} dot components, found {len(centers)}")
    return centers


def decision_from_image(image: Image.Image) -> str:
    """Recover the color with the smallest complete pair-distance repeat burden."""

    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    scores = {
        color: _repeat_burden(_centers_for_color(arr, index))
        for index, color in enumerate(_COLORS)
    }
    return min(_COLORS, key=lambda color: (scores[color], _COLORS.index(color)))
