"""Renderer and analytic label for the unique-spacing constellation world."""

from __future__ import annotations

import copy
import math
import random
from typing import Iterable

from PIL import Image, ImageDraw


RENDERER_VERSION = "unique_spacing_constellation-0.1.0"
COLORS = ("red", "blue", "green")
PALETTE = {
    "red": (220, 50, 50),
    "blue": (45, 90, 210),
    "green": (40, 170, 90),
}
CANVAS_SIZE = (768, 420)
LANE_X = (128.0, 384.0, 640.0)
LANE_Y = 210.0
DOT_COUNT = 8
DOT_RADIUS = 8.0
DISTANCE_TOLERANCE = 3.0
QUARANTINE_GAP = 2.0
SUPERSAMPLE = 4


def _pair_distances(points: Iterable[Iterable[float]]) -> list[float]:
    values = [(float(p[0]), float(p[1])) for p in points]
    return sorted(
        math.hypot(x1 - x2, y1 - y2)
        for i, (x1, y1) in enumerate(values)
        for x2, y2 in values[i + 1 :]
    )


def _repeat_burden(points: Iterable[Iterable[float]], tolerance: float) -> int:
    """Count extra members in tolerance-clusters of the pair-distance multiset."""

    distances = _pair_distances(points)
    burden = 0
    i = 0
    while i < len(distances):
        j = i + 1
        while j < len(distances) and distances[j] - distances[i] <= tolerance:
            j += 1
        if j - i > 1:
            burden += j - i - 1
        i = j
    return burden


def _min_separation(points: Iterable[Iterable[float]]) -> float:
    values = [(float(p[0]), float(p[1])) for p in points]
    if len(values) < 2:
        return float("inf")
    return min(
        math.hypot(x1 - x2, y1 - y2)
        for i, (x1, y1) in enumerate(values)
        for x2, y2 in values[i + 1 :]
    )


def _rotate(points: list[tuple[float, float]], angle: float) -> list[tuple[float, float]]:
    ca, sa = math.cos(angle), math.sin(angle)
    return [(ca * x - sa * y, sa * x + ca * y) for x, y in points]


def _normalize(points: list[tuple[float, float]], angle: float) -> list[list[float]]:
    """Apply a shared pose, then give every pattern the same visible footprint."""

    posed = _rotate(points, angle)
    mean_x = sum(p[0] for p in posed) / len(posed)
    mean_y = sum(p[1] for p in posed) / len(posed)
    centered = [(x - mean_x, y - mean_y) for x, y in posed]
    span_x = max(p[0] for p in centered) - min(p[0] for p in centered)
    span_y = max(p[1] for p in centered) - min(p[1] for p in centered)
    if span_x <= 1e-9 or span_y <= 1e-9:
        raise ValueError("degenerate constellation footprint")
    sx, sy = 132.0 / span_x, 116.0 / span_y
    return [[round(x * sx, 5), round(y * sy, 5)] for x, y in centered]


def _target_pattern(rng: random.Random, angle: float) -> list[list[float]]:
    """Find an irregular eight-point set with a low spacing-collision burden."""

    best: tuple[tuple[int, int, float], list[list[float]]] | None = None
    for _ in range(1500):
        raw = [(rng.uniform(-1.0, 1.0), rng.uniform(-1.0, 1.0)) for _ in range(DOT_COUNT)]
        points = _normalize(raw, angle)
        separation = _min_separation(points)
        if separation < 20.0:
            continue
        burden = _repeat_burden(points, DISTANCE_TOLERANCE)
        key = (burden, int(round(100.0 / separation)), -sum(_pair_distances(points)))
        if best is None or key < best[0]:
            best = (key, points)
        if burden <= 1 and separation >= 24.0:
            return points
    if best is not None:
        return best[1]
    fallback = [
        (0.0, 0.0), (1.0, 0.2), (3.8, -0.6), (8.9, 0.8),
        (15.1, -0.4), (22.5, 0.7), (31.8, -0.5), (34.0, 0.4),
    ]
    return _normalize(fallback, angle)


def _central_symmetric_decoy(rng: random.Random, angle: float) -> list[list[float]]:
    for _ in range(200):
        base: list[tuple[float, float]] = []
        while len(base) < 4:
            point = (rng.uniform(-4.0, 4.0), rng.uniform(-3.0, 3.0))
            trial = base + [point]
            if _min_separation(trial + [(-x, -y) for x, y in trial]) > 1.0:
                base.append(point)
        raw = base + [(-x, -y) for x, y in base]
        points = _normalize(raw, angle)
        if _min_separation(points) >= 20.0:
            return points
    raise RuntimeError("could not sample central-symmetry decoy")


def _two_row_decoy(rng: random.Random, angle: float) -> list[list[float]]:
    for _ in range(200):
        xs = sorted(rng.uniform(-4.0, 4.0) for _ in range(4))
        if min(b - a for a, b in zip(xs, xs[1:])) < 0.8:
            continue
        raw = [(x, -1.5) for x in xs] + [(x, 1.5) for x in xs]
        points = _normalize(raw, angle)
        if _min_separation(points) >= 20.0:
            return points
    raise RuntimeError("could not sample two-row decoy")


def _score_by_color(scene: dict) -> dict[str, int]:
    tolerance = float(scene["distance_tolerance"])
    return {
        color: _repeat_burden(scene["point_sets"][color], tolerance)
        for color in COLORS
    }


def sample_scene(seed: int) -> dict:
    """Sample one balanced, comfortably off-boundary scene from an integer seed."""

    rng = random.Random(int(seed))
    target_color = COLORS[rng.randrange(len(COLORS))]
    lane_order = list(COLORS)
    rng.shuffle(lane_order)
    lane_centers = {
        color: [LANE_X[index], LANE_Y] for index, color in enumerate(lane_order)
    }
    pose_angle = rng.uniform(-math.pi, math.pi)
    unique = _target_pattern(rng, pose_angle)
    symmetric = _central_symmetric_decoy(rng, pose_angle)
    rows = _two_row_decoy(rng, pose_angle)
    decoy_colors = [color for color in COLORS if color != target_color]
    rng.shuffle(decoy_colors)
    point_sets: dict[str, list[list[float]]] = {target_color: unique}
    point_sets[decoy_colors[0]] = symmetric
    point_sets[decoy_colors[1]] = rows
    for color in COLORS:
        cx, cy = lane_centers[color]
        point_sets[color] = [
            [round(cx + x, 5), round(cy + y, 5)] for x, y in point_sets[color]
        ]
    scene = {
        "target_color": target_color,
        "lane_order": lane_order,
        "palette": {color: list(PALETTE[color]) for color in COLORS},
        "canvas_size": list(CANVAS_SIZE),
        "lane_centers": lane_centers,
        "point_sets": point_sets,
        "dot_radius": DOT_RADIUS,
        "distance_tolerance": DISTANCE_TOLERANCE,
        "background": [255, 255, 255],
    }
    if margin(scene) < QUARANTINE_GAP:
        return sample_scene(int(seed) + 7919)
    return scene


def render(scene: dict) -> Image.Image:
    width, height = (int(value) for value in scene["canvas_size"])
    image = Image.new(
        "RGB",
        (width * SUPERSAMPLE, height * SUPERSAMPLE),
        tuple(int(value) for value in scene["background"]),
    )
    draw = ImageDraw.Draw(image)
    target = scene["target_color"]
    lane_order = list(scene["lane_order"])
    draw_order = [target] + [color for color in lane_order if color != target]
    radius = float(scene["dot_radius"])
    for color in draw_order:
        ink = tuple(int(value) for value in scene["palette"][color])
        for x, y in scene["point_sets"][color]:
            box = [
                int(round((float(x) - radius) * SUPERSAMPLE)),
                int(round((float(y) - radius) * SUPERSAMPLE)),
                int(round((float(x) + radius) * SUPERSAMPLE)),
                int(round((float(y) + radius) * SUPERSAMPLE)),
            ]
            draw.ellipse(box, fill=ink)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """The declared answer is the color assigned the unique-spacing pattern."""

    return scene["target_color"]


def margin(scene: dict) -> float:
    scores = sorted(_score_by_color(scene).values())
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["point_sets"] = {
        color: list(reversed(points)) for color, points in scene["point_sets"].items()
    }
    return [("reverse_point_record_order", twin)]
