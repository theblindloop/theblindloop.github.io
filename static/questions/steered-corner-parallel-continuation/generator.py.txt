"""Corner parallel-continuation world.

Two short dark strokes are placed far apart, one in the upper-left region and
one in the lower-right region of the canvas. Each stroke points in one of four
cardinal-cum-diagonal directions (0, 45, 90, 135 degrees). The decision asks
whether the two distant strokes share the SAME direction (reading as two
continuing parallel rails, a global continuation Gestalt) or diverge at a RIGHT
ANGLE (a corner break). The answer requires binding the two disconnected,
widely separated components: neither stroke alone fixes it, and erasing either
distant stroke destroys the comparison.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

BG = (255, 255, 255)
INK = (28, 34, 66)

CANVAS_W = 512
CANVAS_H = 512

ANGLES = [0, 45, 90, 135]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    bar1_angle = ANGLES[seed % len(ANGLES)]
    if (seed // len(ANGLES)) % 2 == 0:
        bar2_angle = bar1_angle
    else:
        bar2_angle = (bar1_angle + 90) % 180
    bar1_cx = rng.randint(80, 92)
    bar1_cy = rng.randint(80, 92)
    bar2_cx = rng.randint(420, 432)
    bar2_cy = rng.randint(420, 432)
    return {
        "bar1_angle": int(bar1_angle),
        "bar2_angle": int(bar2_angle),
        "bar1_cx": bar1_cx,
        "bar1_cy": bar1_cy,
        "bar2_cx": bar2_cx,
        "bar2_cy": bar2_cy,
        "length": 46,
        "width": 14,
        "canvas_w": CANVAS_W,
        "canvas_h": CANVAS_H,
    }


def _endpoints(cx: float, cy: float, angle: float, length: float):
    rad = math.radians(angle)
    hx, hy = (length / 2.0) * math.cos(rad), (length / 2.0) * math.sin(rad)
    return (cx - hx, cy - hy), (cx + hx, cy + hy)


def _draw_bar(draw: ImageDraw.ImageDraw, scene: dict, cx: float, cy: float, angle: float) -> None:
    (x0, y0), (x1, y1) = _endpoints(cx, cy, angle, scene["length"])
    pad = 4
    x0p, y0p = max(pad, min(scene["canvas_w"] - 1 - pad, x0)), max(
        pad, min(scene["canvas_h"] - 1 - pad, y0)
    )
    x1p, y1p = max(pad, min(scene["canvas_w"] - 1 - pad, x1)), max(
        pad, min(scene["canvas_h"] - 1 - pad, y1)
    )
    draw.line([x0p, y0p, x1p, y1p], fill=INK, width=int(scene["width"]))


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (scene["canvas_w"], scene["canvas_h"]), BG)
    draw = ImageDraw.Draw(img)
    _draw_bar(draw, scene, scene["bar1_cx"], scene["bar1_cy"], scene["bar1_angle"])
    _draw_bar(draw, scene, scene["bar2_cx"], scene["bar2_cy"], scene["bar2_angle"])
    return img


def _circular_diff(a: float, b: float) -> float:
    d = abs(a - b) % 180.0
    return min(d, 180.0 - d)


def analytic_gold(scene: dict) -> str:
    if scene["bar1_angle"] == scene["bar2_angle"]:
        return "yes"
    return "no"


def margin(scene: dict) -> float:
    diff = _circular_diff(scene["bar1_angle"], scene["bar2_angle"])
    return float(abs(diff - 45.0))


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> iter:
    return iter(())
