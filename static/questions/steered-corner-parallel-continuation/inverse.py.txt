"""Pixel-only oracle for the corner parallel-continuation world.

The oracle never imports the renderer, prompts, generator, verifier, or any gold
code. It receives only the rendered image. It colour-segments the two dark
strokes, splits them into their two disconnected components, recovers each
stroke's principal direction, and reports whether the two distant orientations
agree (parallel) or are at a right angle.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

INK = (28, 34, 66)


def _principal_angle(mask: np.ndarray) -> float:
    ys, xs = np.nonzero(mask)
    if len(ys) < 3:
        raise ValueError("stroke component too small to fit an orientation")
    centroid = np.asarray([float(ys.mean()), float(xs.mean())])
    centered = np.stack((ys, xs), axis=1).astype(np.float64) - centroid
    cov = centered.T @ centered / len(centered)
    values, vectors = np.linalg.eigh(cov)
    axis = vectors[:, int(np.argmax(values))]
    vy, vx = float(axis[0]), float(axis[1])
    angle = math.degrees(math.atan2(vy, vx))
    if angle < 0.0:
        angle += 180.0
    if angle >= 180.0:
        angle -= 180.0
    return angle


def _components(mask: np.ndarray) -> list[np.ndarray]:
    ys, xs = np.nonzero(mask)
    if not len(ys):
        raise ValueError("no stroke pixels present")
    seen = set()
    comps = []
    for y, x in zip(ys.tolist(), xs.tolist(), strict=True):
        if (y, x) in seen:
            continue
        stack = [(y, x)]
        seen.add((y, x))
        pixels = []
        while stack:
            cy, cx = stack.pop()
            pixels.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if (
                    0 <= ny < mask.shape[0]
                    and 0 <= nx < mask.shape[1]
                    and mask[ny, nx]
                    and (ny, nx) not in seen
                ):
                    seen.add((ny, nx))
                    stack.append((ny, nx))
        comps.append(pixels)
    return comps


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    mask = np.all(arr == INK, axis=2)
    comps = _components(mask)
    if len(comps) != 2:
        raise ValueError("expected exactly two disconnected stroke components")
    masks = [np.zeros(mask.shape, dtype=bool) for _ in comps]
    for comp, out in zip(comps, masks, strict=True):
        for y, x in comp:
            out[y, x] = True
    if len(masks) != 2:
        raise ValueError("unexpected component count")
    a0 = _principal_angle(masks[0])
    a1 = _principal_angle(masks[1])
    d = abs(a0 - a1) % 180.0
    diff = min(d, 180.0 - d)
    return "yes" if diff < 45.0 else "no"
