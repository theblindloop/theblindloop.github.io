"""Marked-seam twin-cluster world.

A row of domino-like tiles is drawn on a plain background. Every half tile
carries a compact cluster of dark dots. Exactly one small red arrow sits under
the row, pointing up at the butt seam between two neighbouring tiles. The
decision is whether the two dot clusters flanking that arrow hold the same
number of dots.

The familiar canonical prior (a laid domino chain joins halves that match) says
"same"; the sampler produces matched control (same) and counterfactual (differ)
scenes with identical wording, identical styling and balanced count sums, so the
prior is recorded for analysis and never used as gold.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "domino_seam_twin-renderer-0.1.0"

# Fixed ink colours. Nothing is antialiased, so every mark keeps its exact RGB
# value and the pixel-only oracle can segment by exact colour equality.
BG = (255, 255, 255)
DOT = (30, 30, 30)
MARK = (198, 32, 44)
EDGE = (96, 88, 72)

TILE_FILLS = [(246, 240, 214), (232, 238, 246), (241, 232, 241), (236, 244, 232)]
DECOR_COLORS = [(0, 140, 140), (150, 150, 30), (170, 170, 170), (122, 60, 160)]

# Grouping radius shared with (but re-implemented in) the pixel oracle: dots
# closer than this belong to the same cluster.
R_REF = 34.0
QUARANTINE_MARGIN = 4.0

WIDTH = 940
HEIGHT = 520
HALF_W = 84
TILE_H = 96
TILE_GAP = 8

# Compact, single-link connected dot patterns on a 3x3 grid of (row, col) slots.
PATTERNS: dict[int, list[list[tuple[int, int]]]] = {
    1: [[(1, 1)], [(0, 1)], [(2, 1)], [(1, 0)], [(1, 2)]],
    2: [[(1, 0), (1, 1)], [(1, 1), (1, 2)], [(0, 1), (1, 1)], [(1, 1), (2, 1)]],
    3: [
        [(1, 0), (1, 1), (1, 2)],
        [(0, 1), (1, 1), (2, 1)],
        [(0, 0), (1, 1), (2, 2)],
        [(0, 1), (1, 0), (1, 1)],
    ],
    4: [
        [(0, 0), (0, 1), (1, 0), (1, 1)],
        [(0, 1), (1, 0), (1, 1), (1, 2)],
        [(0, 0), (0, 2), (1, 1), (2, 0)],
        [(1, 1), (1, 2), (2, 1), (2, 2)],
    ],
    5: [
        [(0, 0), (0, 2), (1, 1), (2, 0), (2, 2)],
        [(0, 0), (0, 1), (1, 0), (1, 1), (2, 2)],
        [(0, 1), (1, 0), (1, 1), (1, 2), (2, 1)],
    ],
    6: [
        [(0, 0), (0, 1), (0, 2), (1, 0), (1, 1), (1, 2)],
        [(1, 0), (1, 1), (1, 2), (2, 0), (2, 1), (2, 2)],
    ],
}


def _pitch(half_w: int, tile_gap: int) -> int:
    return 2 * half_w + tile_gap


def _seam_x(scene: dict) -> float:
    return (
        scene["row_x"]
        + scene["marked_joint"] * _pitch(scene["half_w"], scene["tile_gap"])
        + 2 * scene["half_w"]
        + scene["tile_gap"] / 2.0
    )


def _joint_clusters(scene: dict) -> tuple[list, list]:
    """The two dot clusters flanking the marked seam."""
    joint = scene["marked_joint"]
    halves = scene["halves"]
    return halves[2 * joint + 1], halves[2 * joint + 2]


def analytic_gold(scene: dict) -> str:
    """`same` when the two flanking clusters hold equally many dots."""
    left, right = _joint_clusters(scene)
    return "same" if len(left) == len(right) else "differ"


def _dot_positions(half_x: int, row_y: int, half_w: int, tile_h: int, spacing: int, pattern):
    cx = half_x + half_w // 2
    cy = row_y + tile_h // 2
    return [[cx + (col - 1) * spacing, cy + (row - 1) * spacing] for row, col in pattern]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    n_tiles = rng.choice([4, 5])
    spacing = rng.choice([16, 17, 18])
    dot_r = rng.choice([5, 6, 7])
    pitch = _pitch(HALF_W, TILE_GAP)
    row_w = n_tiles * pitch - TILE_GAP
    row_x = rng.randint(20, max(20, WIDTH - row_w - 20))
    row_y = rng.randint(150, 250)
    joint = rng.randrange(n_tiles - 1)

    same = rng.random() < 0.5
    left_n = rng.randint(1, 6)
    if same:
        right_n = left_n
    else:
        deltas = [d for d in (1, 2, 3) if 1 <= left_n + d <= 6 or 1 <= left_n - d <= 6]
        delta = rng.choice(deltas)
        signs = [s for s in (1, -1) if 1 <= left_n + s * delta <= 6]
        right_n = left_n + rng.choice(signs) * delta

    counts = [rng.randint(1, 6) for _ in range(2 * n_tiles)]
    counts[2 * joint + 1] = left_n
    counts[2 * joint + 2] = right_n

    halves = []
    for index, count in enumerate(counts):
        tile = index // 2
        half_x = row_x + tile * pitch + (index % 2) * HALF_W
        pattern = rng.choice(PATTERNS[count])
        halves.append(_dot_positions(half_x, row_y, HALF_W, TILE_H, spacing, pattern))

    decor = []
    for _ in range(rng.randint(6, 11)):
        kind = rng.choice(["ring", "square", "tri", "bar"])
        color = rng.randrange(len(DECOR_COLORS))
        x = rng.randint(30, WIDTH - 60)
        y = rng.choice([rng.randint(24, 110), rng.randint(row_y + TILE_H + 60, HEIGHT - 40)])
        decor.append([x, y, rng.randint(9, 20), kind, color])

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "half_w": HALF_W,
        "tile_h": TILE_H,
        "tile_gap": TILE_GAP,
        "n_tiles": n_tiles,
        "row_x": row_x,
        "row_y": row_y,
        "dot_r": dot_r,
        "spacing": spacing,
        "halves": halves,
        "marked_joint": joint,
        "decor": decor,
        "palette": rng.randrange(len(TILE_FILLS)),
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (scene["width"], scene["height"]), BG)
    draw = ImageDraw.Draw(image)
    fill = TILE_FILLS[scene["palette"] % len(TILE_FILLS)]
    pitch = _pitch(scene["half_w"], scene["tile_gap"])

    for x, y, size, kind, color in scene["decor"]:
        ink = DECOR_COLORS[color % len(DECOR_COLORS)]
        if kind == "ring":
            draw.ellipse([x, y, x + size, y + size], outline=ink, width=3)
        elif kind == "square":
            draw.rectangle([x, y, x + size, y + size], outline=ink, width=3)
        elif kind == "tri":
            draw.polygon([(x, y + size), (x + size, y + size), (x + size / 2, y)], fill=ink)
        else:
            draw.rectangle([x, y, x + size, y + max(3, size // 4)], fill=ink)

    for tile in range(scene["n_tiles"]):
        x0 = scene["row_x"] + tile * pitch
        y0 = scene["row_y"]
        draw.rectangle(
            [x0, y0, x0 + 2 * scene["half_w"], y0 + scene["tile_h"]], fill=fill, outline=EDGE, width=3
        )
        draw.line(
            [x0 + scene["half_w"], y0 + 6, x0 + scene["half_w"], y0 + scene["tile_h"] - 6],
            fill=EDGE,
            width=2,
        )

    for cluster in scene["halves"]:
        for cx, cy in cluster:
            r = scene["dot_r"]
            draw.ellipse([cx - r, cy - r, cx + r, cy + r], fill=DOT)

    seam = _seam_x(scene)
    top = scene["row_y"] + scene["tile_h"] + 100
    draw.polygon(
        [(seam, top), (seam - 10, top + 17), (seam + 10, top + 17)],
        fill=MARK,
    )
    return image


def _mst_max_edge(points: list) -> float:
    if len(points) < 2:
        return 0.0
    inside = [points[0]]
    rest = list(points[1:])
    longest = 0.0
    while rest:
        best = None
        for index, candidate in enumerate(rest):
            for anchor in inside:
                distance = ((candidate[0] - anchor[0]) ** 2 + (candidate[1] - anchor[1]) ** 2) ** 0.5
                if best is None or distance < best[0]:
                    best = (distance, index)
        distance, index = best
        longest = max(longest, distance)
        inside.append(rest.pop(index))
    return longest


def _median(values: list[float]) -> float:
    ordered = sorted(values)
    middle = len(ordered) // 2
    if len(ordered) % 2:
        return float(ordered[middle])
    return (ordered[middle - 1] + ordered[middle]) / 2.0


def margin(scene: dict) -> float:
    """Smallest pixel slack protecting the whole image-only decision chain."""
    halves = scene["halves"]
    dot_r = scene["dot_r"]
    terms = []

    flat = [(index, point) for index, cluster in enumerate(halves) for point in cluster]
    rim_gap = 1e9
    separation = 1e9
    for i in range(len(flat)):
        for j in range(i + 1, len(flat)):
            (ci, pi), (cj, pj) = flat[i], flat[j]
            distance = ((pi[0] - pj[0]) ** 2 + (pi[1] - pj[1]) ** 2) ** 0.5
            rim_gap = min(rim_gap, distance - 2 * dot_r)
            if ci != cj:
                separation = min(separation, distance)
    terms.append(rim_gap)
    terms.append(separation - R_REF)
    terms.append(R_REF - max(_mst_max_edge(cluster) for cluster in halves))

    means = [sum(point[0] for point in cluster) / len(cluster) for cluster in halves]
    ordered = sorted(means)
    threshold = _median([ordered[k + 1] - ordered[k] for k in range(len(ordered) - 1)])
    seam = _seam_x(scene)
    joint = scene["marked_joint"]
    near = [2 * joint + 1, 2 * joint + 2]
    terms.append(threshold - max(abs(means[k] - seam) for k in near))
    terms.append(min(abs(means[k] - seam) for k in range(len(means)) if k not in near) - threshold)
    return round(float(min(terms)), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist: every sampled field is drawn, and the gold reads
    only dot-cluster membership counts that the raster shows directly."""
    return []


def canonical_prior(scene: dict) -> str:
    """Familiar prior for analysis only: laid domino chains join equal halves."""
    return "same"


def scene_targets(scene: dict) -> dict[str, Any]:
    left, right = _joint_clusters(scene)
    return {
        "left_count": len(left),
        "right_count": len(right),
        "decision": analytic_gold(scene),
        "count_sum": len(left) + len(right),
        "canonical_prior": canonical_prior(scene),
        "variant": "control" if analytic_gold(scene) == canonical_prior(scene) else "counterfactual",
    }
