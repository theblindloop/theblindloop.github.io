"""Independent pixel-only inverse arm for the marked-seam twin-cluster world.

It receives a PNG and nothing else: no scene, no renderer import, no analytic
gold. It re-derives the decision by segmenting the two fixed ink colours,
labelling dot blobs, grouping blobs into clusters, locating the red arrow, and
comparing the sizes of the two clusters that flank the arrow.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "domino_seam_twin-oracle-0.1.0"

DOT_RGB = (30, 30, 30)
MARK_RGB = (198, 32, 44)

GROUP_RADIUS = 34.0
MIN_BLOB_AREA = 40
MAX_BLOB_AREA = 400
UNRESOLVED = "unresolved"


def _exact_mask(arr: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    return (arr[:, :, 0] == rgb[0]) & (arr[:, :, 1] == rgb[1]) & (arr[:, :, 2] == rgb[2])


def _blobs(mask: np.ndarray) -> list[tuple[float, float, int]]:
    """4-connected components of a sparse boolean mask, as (x, y, area)."""
    pixels = {(int(y), int(x)) for y, x in np.argwhere(mask)}
    found = []
    while pixels:
        start = pixels.pop()
        stack = [start]
        component = [start]
        while stack:
            y, x = stack.pop()
            for neighbour in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if neighbour in pixels:
                    pixels.discard(neighbour)
                    stack.append(neighbour)
                    component.append(neighbour)
        area = len(component)
        ys = [p[0] for p in component]
        xs = [p[1] for p in component]
        height = max(ys) - min(ys) + 1
        width = max(xs) - min(xs) + 1
        if not MIN_BLOB_AREA <= area <= MAX_BLOB_AREA:
            continue
        if max(width, height) > 2.0 * min(width, height):
            continue  # not a round dot
        found.append((sum(xs) / area, sum(ys) / area, area))
    return found


def _group(points: list[tuple[float, float, int]]) -> list[list[tuple[float, float, int]]]:
    """Single-link grouping at GROUP_RADIUS."""
    parent = list(range(len(points)))

    def find(a: int) -> int:
        while parent[a] != a:
            parent[a] = parent[parent[a]]
            a = parent[a]
        return a

    for i in range(len(points)):
        for j in range(i + 1, len(points)):
            dx = points[i][0] - points[j][0]
            dy = points[i][1] - points[j][1]
            if (dx * dx + dy * dy) ** 0.5 <= GROUP_RADIUS:
                parent[find(i)] = find(j)
    clusters: dict[int, list[tuple[float, float, int]]] = {}
    for index, point in enumerate(points):
        clusters.setdefault(find(index), []).append(point)
    return list(clusters.values())


def _median(values: list[float]) -> float:
    ordered = sorted(values)
    middle = len(ordered) // 2
    if len(ordered) % 2:
        return float(ordered[middle])
    return (ordered[middle - 1] + ordered[middle]) / 2.0


def measure(image: Image.Image) -> dict:
    arr = np.asarray(image.convert("RGB"))
    marker = np.argwhere(_exact_mask(arr, MARK_RGB))
    dots = _blobs(_exact_mask(arr, DOT_RGB))
    report = {"marker_pixels": int(len(marker)), "dot_blobs": len(dots)}
    if len(marker) < 40 or len(dots) < 2:
        report["decision"] = UNRESOLVED
        return report

    arrow_x = float(marker[:, 1].mean())
    clusters = _group(dots)
    if len(clusters) < 3:
        report["decision"] = UNRESOLVED
        return report

    means = sorted(sum(p[0] for p in cluster) / len(cluster) for cluster in clusters)
    threshold = _median([means[k + 1] - means[k] for k in range(len(means) - 1)])

    left = right = None
    for cluster in clusters:
        mean_x = sum(p[0] for p in cluster) / len(cluster)
        if abs(mean_x - arrow_x) > threshold:
            continue
        if mean_x < arrow_x and (left is None or mean_x > left[0]):
            left = (mean_x, len(cluster))
        elif mean_x > arrow_x and (right is None or mean_x < right[0]):
            right = (mean_x, len(cluster))
    if left is None or right is None:
        report["decision"] = UNRESOLVED
        return report

    report.update(
        {
            "arrow_x": arrow_x,
            "left_count": left[1],
            "right_count": right[1],
            "decision": "same" if left[1] == right[1] else "differ",
        }
    )
    return report


def decision_from_image(image: Image.Image) -> str:
    return str(measure(image)["decision"])
