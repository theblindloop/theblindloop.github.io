"""Pixel-only inverse arm for the bead-trail world.

Receives one rasterised PNG and nothing else.  It rebuilds the global contour
graph from the dark ink, checks that the ink is exactly one closed thin cycle,
walks the cycle to obtain an ordered traversal, projects each coloured bead onto
that traversal, and finally reads off the bead two steps away from red in the
recovered 4-cycle.  Any break in the trace, any missing bead, or any ambiguous
projection yields ``unknown`` instead of a guess.
"""

from __future__ import annotations

import numpy as np

REFERENCE_BEADS = {
    "red": (208, 44, 44),
    "green": (32, 142, 66),
    "blue": (48, 86, 206),
    "amber": (232, 166, 26),
}
INK_MAX_CHANNEL = 110
BEAD_TOLERANCE = 120
MIN_BEAD_PIXELS = 90
UNKNOWN = "unknown"


def _ink_mask(pixels):
    return pixels.max(axis=2) <= INK_MAX_CHANNEL


def _trace_cycle(mask):
    """Return the ink pixels in traversal order, or None if not a single cycle."""
    ys, xs = np.nonzero(mask)
    if ys.size < 200:
        return None
    points = list(zip(xs.tolist(), ys.tolist()))
    present = set(points)
    links = {}
    for (x, y) in points:
        found = [
            (x + dx, y + dy)
            for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1))
            if (x + dx, y + dy) in present
        ]
        if len(found) != 2:
            return None
        links[(x, y)] = found
    start = points[0]
    order = [start]
    previous = None
    current = start
    while True:
        left, right = links[current]
        nxt = left if left != previous else right
        previous, current = current, nxt
        if current == start:
            break
        order.append(current)
        if len(order) > len(points):
            return None
    if len(order) != len(points):
        return None
    return np.asarray(order, dtype=np.float64)


def _bead_centroids(pixels, ink):
    centroids = {}
    for name, rgb in REFERENCE_BEADS.items():
        distance = np.abs(pixels - np.array(rgb, dtype=np.int64)).sum(axis=2)
        mask = (distance <= BEAD_TOLERANCE) & ~ink
        count = int(mask.sum())
        if count < MIN_BEAD_PIXELS:
            return None
        ys, xs = np.nonzero(mask)
        spread = max(ys.max() - ys.min(), xs.max() - xs.min())
        if spread > 80:
            return None
        centroids[name] = (float(xs.mean()), float(ys.mean()))
    return centroids


def decision_from_image(image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int64)
    ink = _ink_mask(pixels)
    trace = _trace_cycle(ink)
    if trace is None:
        return UNKNOWN
    centroids = _bead_centroids(pixels, ink)
    if centroids is None:
        return UNKNOWN
    total = trace.shape[0]
    placements = {}
    for name, (cx, cy) in centroids.items():
        offsets = np.hypot(trace[:, 0] - cx, trace[:, 1] - cy)
        best = int(np.argmin(offsets))
        if offsets[best] > 6.0:
            return UNKNOWN
        placements[name] = best
    ordered = sorted(placements, key=lambda name: placements[name])
    steps = [
        (placements[ordered[(k + 1) % 4]] - placements[ordered[k]]) % total
        for k in range(4)
    ]
    if min(steps) < 8:
        return UNKNOWN
    anchor = ordered.index("red")
    return ordered[(anchor + 2) % 4]
