"""Bead trail: cyclic order of coloured beads along one closed corridor outline.

The scene is a single closed, thin, rectilinear outline (the boundary of a
self-avoiding lattice corridor) that meanders across the whole page.  Four small
coloured beads sit exactly on that outline.  The decision is which bead is the
*opposite* one of the red bead when the outline is walked as a cycle: the second
bead met when travelling along the trail away from red, which is the same bead in
either direction.

Spatial proximity is deliberately uninformative: the two walls of the corridor
run close together in the picture but are separated by half the trail length, so
the answer can only be obtained by following the whole connected trace.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CELL = 68
GRID_W = 11
GRID_H = 7
MARGIN = 40
CANVAS_W = 2 * MARGIN + GRID_W * CELL
CANVAS_H = 2 * MARGIN + GRID_H * CELL
BEAD_RADIUS = 11

BEAD_ORDER = ("red", "green", "blue", "amber")
BEAD_RGB = {
    "red": (208, 44, 44),
    "green": (32, 142, 66),
    "blue": (48, 86, 206),
    "amber": (232, 166, 26),
}
INK_CHOICES = ((28, 32, 40), (38, 30, 34), (28, 40, 36), (34, 34, 30))
BG_CHOICES = ((250, 250, 247), (247, 249, 252), (251, 248, 244))

MIN_TRAIL_CELLS = 15
MAX_TRAIL_CELLS = 24
MIN_BEAD_GAP = 34.0
QUARANTINE_GAP = 66.0
MIN_BEAD_EUCLID = 30.0
MIN_CORNER_CLEARANCE = 24.0


def _neighbours(cell):
    i, j = cell
    return ((i + 1, j), (i - 1, j), (i, j + 1), (i, j - 1))


def _ring8(cell):
    i, j = cell
    return tuple(
        (i + di, j + dj)
        for di in (-1, 0, 1)
        for dj in (-1, 0, 1)
        if not (di == 0 and dj == 0)
    )


def _admissible(cand, previous, occupied):
    """A corridor stays one cell wide and never pinches its own outline."""
    ci, cj = cand
    if not (0 <= ci < GRID_W and 0 <= cj < GRID_H):
        return False
    if cand in occupied:
        return False
    for other in _neighbours(cand):
        if other in occupied and other != previous:
            return False
    for di, dj in ((1, 1), (1, -1), (-1, 1), (-1, -1)):
        diagonal = (ci + di, cj + dj)
        if diagonal not in occupied:
            continue
        if (ci + di, cj) not in occupied and (ci, cj + dj) not in occupied:
            return False
    return True


def _grow_trail(rng, target):
    """Depth-first self-avoiding lattice corridor of the requested length."""
    start = (rng.randrange(GRID_W), rng.randrange(GRID_H))
    trail = [start]
    occupied = {start}
    frontier = [_shuffled_steps(rng, start)]
    expansions = 0
    while trail and len(trail) < target and expansions < 6000:
        expansions += 1
        options = frontier[-1]
        placed = False
        while options:
            cand = options.pop()
            previous = trail[-1]
            if _admissible(cand, previous, occupied):
                trail.append(cand)
                occupied.add(cand)
                frontier.append(_shuffled_steps(rng, cand))
                placed = True
                break
        if not placed:
            occupied.discard(trail.pop())
            frontier.pop()
    return trail


def _shuffled_steps(rng, cell):
    steps = list(_neighbours(cell))
    rng.shuffle(steps)
    return steps


def _simply_connected(cells):
    """The padded complement of the corridor must be a single region."""
    region = set(cells)
    seen = {(-1, -1)}
    stack = [(-1, -1)]
    total = 0
    while stack:
        i, j = stack.pop()
        total += 1
        for ni, nj in ((i + 1, j), (i - 1, j), (i, j + 1), (i, j - 1)):
            if not (-1 <= ni <= GRID_W and -1 <= nj <= GRID_H):
                continue
            if (ni, nj) in region or (ni, nj) in seen:
                continue
            seen.add((ni, nj))
            stack.append((ni, nj))
    return total == (GRID_W + 2) * (GRID_H + 2) - len(region)


def _boundary_cycle(cells):
    """Ordered lattice-point cycle of the corridor boundary, collinear-merged."""
    region = set(cells)
    edges = []
    for (i, j) in cells:
        if (i, j - 1) not in region:
            edges.append(((i, j), (i + 1, j)))
        if (i, j + 1) not in region:
            edges.append(((i, j + 1), (i + 1, j + 1)))
        if (i - 1, j) not in region:
            edges.append(((i, j), (i, j + 1)))
        if (i + 1, j) not in region:
            edges.append(((i + 1, j), (i + 1, j + 1)))
    adjacency = {}
    for a, b in edges:
        adjacency.setdefault(a, []).append(b)
        adjacency.setdefault(b, []).append(a)
    if any(len(v) != 2 for v in adjacency.values()):
        return None
    start = min(adjacency)
    order = [start]
    previous = None
    current = start
    while True:
        step = [p for p in adjacency[current] if p != previous]
        if not step:
            return None
        nxt = step[0]
        previous, current = current, nxt
        if current == start:
            break
        order.append(current)
        if len(order) > len(adjacency):
            return None
    if len(order) != len(adjacency):
        return None
    merged = []
    n = len(order)
    for k in range(n):
        prv = order[(k - 1) % n]
        cur = order[k]
        nxt = order[(k + 1) % n]
        if (cur[0] - prv[0]) * (nxt[1] - cur[1]) != (cur[1] - prv[1]) * (nxt[0] - cur[0]):
            merged.append(cur)
    return merged if len(merged) >= 8 else None


def _to_pixels(lattice_points):
    return [[MARGIN + i * CELL, MARGIN + j * CELL] for i, j in lattice_points]


def _segment_lengths(polyline):
    lengths = []
    n = len(polyline)
    for k in range(n):
        ax, ay = polyline[k]
        bx, by = polyline[(k + 1) % n]
        lengths.append(abs(bx - ax) + abs(by - ay))
    return lengths


def _point_at(polyline, lengths, distance):
    total = sum(lengths)
    remaining = distance % total
    for k, length in enumerate(lengths):
        if remaining <= length:
            ax, ay = polyline[k]
            bx, by = polyline[(k + 1) % len(polyline)]
            frac = remaining / length if length else 0.0
            return [ax + (bx - ax) * frac, ay + (by - ay) * frac]
        remaining -= length
    return list(polyline[0])


def _corner_clearance(lengths, distance):
    total = sum(lengths)
    remaining = distance % total
    for length in lengths:
        if remaining <= length:
            return min(remaining, length - remaining)
        remaining -= length
    return 0.0


def sample_scene(seed: int) -> dict:
    rng = random.Random((int(seed) * 1000003 + 17) & 0xFFFFFFFF)
    for _ in range(4000):
        trail = _grow_trail(rng, rng.randint(MIN_TRAIL_CELLS, MAX_TRAIL_CELLS))
        if len(trail) < MIN_TRAIL_CELLS or not _simply_connected(trail):
            continue
        xs = [c[0] for c in trail]
        ys = [c[1] for c in trail]
        if (max(xs) - min(xs) + 1) < GRID_W - 3 or (max(ys) - min(ys) + 1) < GRID_H - 2:
            continue
        cycle = _boundary_cycle(trail)
        if cycle is None:
            continue
        polyline = _to_pixels(cycle)
        lengths = _segment_lengths(polyline)
        total = sum(lengths)
        if total < 1200:
            continue

        tight = rng.random() < 0.34
        placement = None
        for _ in range(400):
            offsets = sorted(rng.uniform(0.0, total) for _ in range(4))
            gaps = [
                offsets[(k + 1) % 4] - offsets[k] + (total if k == 3 else 0.0)
                for k in range(4)
            ]
            low = min(gaps)
            if low < MIN_BEAD_GAP:
                continue
            if tight and low >= QUARANTINE_GAP:
                continue
            if not tight and low < QUARANTINE_GAP + 6.0:
                continue
            if any(_corner_clearance(lengths, d) < MIN_CORNER_CLEARANCE for d in offsets):
                continue
            points = [_point_at(polyline, lengths, d) for d in offsets]
            close = False
            for a in range(4):
                for b in range(a + 1, 4):
                    dx = points[a][0] - points[b][0]
                    dy = points[a][1] - points[b][1]
                    if (dx * dx + dy * dy) ** 0.5 < MIN_BEAD_EUCLID:
                        close = True
            if close:
                continue
            placement = (offsets, gaps, points, low)
            break
        if placement is None:
            continue
        offsets, gaps, points, low = placement

        red_slot = rng.randrange(4)
        rest = [c for c in BEAD_ORDER if c != "red"]
        rng.shuffle(rest)
        colours = [None, None, None, None]
        colours[red_slot] = "red"
        fill = 0
        for slot in range(4):
            if colours[slot] is None:
                colours[slot] = rest[fill]
                fill += 1

        ink = list(INK_CHOICES[rng.randrange(len(INK_CHOICES))])
        bg = list(BG_CHOICES[rng.randrange(len(BG_CHOICES))])
        return {
            "canvas": [CANVAS_W, CANVAS_H],
            "loop_xy": [[round(x, 3), round(y, 3)] for x, y in polyline],
            "bead_xy": [[round(x, 3), round(y, 3)] for x, y in points],
            "bead_colors": colours,
            "bead_offsets": [round(d, 3) for d in offsets],
            "bead_gaps": [round(g, 3) for g in gaps],
            "trail_length": round(total, 3),
            "min_gap": round(low, 3),
            "bead_radius": BEAD_RADIUS,
            "ink_rgb": ink,
            "bg_rgb": bg,
            "orientation": 1,
            "start_index": 0,
        }
    raise RuntimeError("sample_scene: no admissible scene found")


def render(scene: dict) -> Image.Image:
    width, height = scene["canvas"]
    image = Image.new("RGB", (int(width), int(height)), tuple(scene["bg_rgb"]))
    draw = ImageDraw.Draw(image)
    radius = int(scene["bead_radius"])
    for (cx, cy), name in zip(scene["bead_xy"], scene["bead_colors"]):
        draw.ellipse(
            [round(cx) - radius, round(cy) - radius, round(cx) + radius, round(cy) + radius],
            fill=BEAD_RGB[name],
        )
    loop = [(round(x), round(y)) for x, y in scene["loop_xy"]]
    shift = int(scene["start_index"]) % len(loop)
    loop = loop[shift:] + loop[:shift]
    if int(scene["orientation"]) < 0:
        loop = list(reversed(loop))
    ink = tuple(scene["ink_rgb"])
    for k in range(len(loop)):
        draw.line([loop[k], loop[(k + 1) % len(loop)]], fill=ink, width=1)
    return image


def analytic_gold(scene: dict) -> str:
    order = list(scene["bead_colors"])
    anchor = order.index("red")
    return order[(anchor + 2) % 4]


def margin(scene: dict) -> float:
    return float(scene["min_gap"])


def is_quarantined(scene: dict) -> bool:
    return float(scene["min_gap"]) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list:
    """Pixel-identical relabellings of the latent trail and bead cycle."""
    twins = []

    reversed_trail = dict(scene)
    reversed_trail["orientation"] = -int(scene["orientation"])
    twins.append(("reverse_trail_traversal", reversed_trail))

    rotated_trail = dict(scene)
    rotated_trail["start_index"] = (int(scene["start_index"]) + 3) % len(scene["loop_xy"])
    twins.append(("rotate_trail_start", rotated_trail))

    rotated_beads = dict(scene)
    rotated_beads["bead_xy"] = scene["bead_xy"][1:] + scene["bead_xy"][:1]
    rotated_beads["bead_colors"] = scene["bead_colors"][1:] + scene["bead_colors"][:1]
    twins.append(("rotate_bead_cycle", rotated_beads))

    mirrored_beads = dict(scene)
    mirrored_beads["bead_xy"] = list(reversed(scene["bead_xy"]))
    mirrored_beads["bead_colors"] = list(reversed(scene["bead_colors"]))
    twins.append(("reverse_bead_cycle", mirrored_beads))

    return twins
