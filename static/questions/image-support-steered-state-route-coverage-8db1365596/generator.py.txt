"""Renderer and analytic arm for the connected-route state-coverage world."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "state-route-coverage-0.1.0"
CANVAS = (640, 480)
STATION_COUNT = 9
STATE_LABELS = ("A", "B", "C", "D")
PALETTE = {
    "background": (249, 251, 253),
    "route": (36, 61, 84),
    "operation": (218, 96, 57),
    "state_a": (191, 69, 68),
    "state_b": (41, 149, 94),
    "state_c": (65, 105, 190),
    "state_d": (132, 83, 169),
    "neutral": (169, 177, 187),
    "text": (29, 40, 52),
}


def _colors(scene: dict[str, Any]) -> dict[str, tuple[int, int, int]]:
    return {
        name: tuple(int(channel) for channel in value)
        for name, value in scene["palette"].items()
    }


def _trace(initial_state: int, operations: list[int]) -> list[int]:
    state = int(initial_state) % 4
    trace = [state]
    for operation in operations:
        state = (state + int(operation)) % 4
        trace.append(state)
    return trace


def _missing(trace: list[int]) -> list[int]:
    present = set(int(value) % 4 for value in trace)
    return [state for state in range(4) if state not in present]


def _coverage_margin(trace: list[int]) -> float:
    counts = [trace.count(state) for state in range(4)]
    missing = _missing(trace)
    if len(missing) != 1:
        return 0.0
    return float(min(counts[state] for state in range(4) if state not in missing))


def _route_points(rng: random.Random) -> list[list[int]]:
    """Return a monotone-x, thin trace contained in a central region."""

    xs = [112, 155, 198, 241, 284, 327, 370, 413, 456, 499, 542]
    # Keep the route visibly serpentine but within two central ablation rows.
    # Its horizontal span supplies the regional scale; the narrow vertical
    # envelope keeps the decisive support a single elongated component.
    base_y = [260, 278, 238, 284, 242, 286, 250, 276, 252, 272, 260]
    phase = rng.randrange(4)
    offsets = (-10, 0, 10, 5)
    points: list[list[int]] = []
    for index, (x, y) in enumerate(zip(xs, base_y, strict=True)):
        jitter = 0 if index in (0, len(xs) - 1) else rng.randint(-3, 3)
        points.append([x + jitter, y + offsets[phase]])
    return points


def _distractors(rng: random.Random) -> list[dict[str, int]]:
    anchors = [(46, 62), (594, 62), (48, 418), (592, 416), (82, 112), (564, 112)]
    rng.shuffle(anchors)
    return [
        {
            "kind": index % 2,
            "x": x + rng.randint(-7, 7),
            "y": y + rng.randint(-7, 7),
            "size": rng.randint(8, 12),
        }
        for index, (x, y) in enumerate(anchors[:4])
    ]


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced scene with exactly one absent state."""

    rng = random.Random(int(seed))
    route_points = _route_points(rng)
    for _ in range(400):
        operations = [rng.choice((-1, 0, 1)) for _ in range(STATION_COUNT)]
        base_initial = rng.randrange(4)
        trace = _trace(base_initial, operations)
        missing = _missing(trace)
        if len(missing) == 1 and _coverage_margin(trace) > 0.0:
            break
    else:
        raise RuntimeError("could not sample a unique missing state")

    # A cyclic relabeling of the visible start badge shifts every visited state
    # while leaving the operation silhouettes and route geometry unchanged.
    desired_missing = int(seed) % 4
    shift = (desired_missing - missing[0]) % 4
    initial_state = (base_initial + shift) % 4
    return {
        "initial_state": initial_state,
        "operations": operations,
        "route_points": route_points,
        "distractors": _distractors(rng),
        "palette": {name: list(value) for name, value in PALETTE.items()},
        "route_width": 6,
        "marker_size": 14,
        "canvas": list(CANVAS),
    }


def _draw_operation(
    draw: ImageDraw.ImageDraw,
    center: list[int],
    operation: int,
    radius: int,
    color: tuple[int, int, int],
) -> None:
    x, y = center
    if operation == 1:
        draw.polygon(
            [(x - radius, y - radius), (x - radius, y + radius), (x + radius, y)],
            fill=color,
        )
    elif operation == -1:
        draw.polygon(
            [(x + radius, y - radius), (x + radius, y + radius), (x - radius, y)],
            fill=color,
        )
    else:
        draw.rectangle(
            (x - radius, y - radius, x + radius, y + radius),
            outline=color,
            width=4,
        )


def _draw_state_badge(
    draw: ImageDraw.ImageDraw,
    center: list[int],
    state: int,
    radius: int,
    colors: dict[str, tuple[int, int, int]],
    font: ImageFont.ImageFont,
) -> None:
    x, y = center
    state_color = colors["state_a" if state == 0 else "state_b" if state == 1 else "state_c" if state == 2 else "state_d"]
    draw.ellipse(
        (x - radius, y - radius, x + radius, y + radius),
        fill=state_color,
        outline=colors["route"],
        width=3,
    )
    label = STATE_LABELS[state]
    bbox = draw.textbbox((0, 0), label, font=font)
    tw, th = bbox[2] - bbox[0], bbox[3] - bbox[1]
    draw.text((x - tw / 2, y - th / 2 - 1), label, fill=colors["background"], font=font)


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the route, visible stations, start badge, and nuisance marks."""

    width, height = (int(value) for value in scene["canvas"])
    colors = _colors(scene)
    image = Image.new("RGB", (width, height), colors["background"])
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    for distractor in scene["distractors"]:
        x, y, size = (int(distractor[key]) for key in ("x", "y", "size"))
        if int(distractor["kind"]) == 0:
            draw.ellipse((x - size, y - size, x + size, y + size), outline=colors["neutral"], width=3)
        else:
            draw.rectangle((x - size, y - size, x + size, y + size), outline=colors["neutral"], width=3)

    points = [tuple(int(value) for value in point) for point in scene["route_points"]]
    draw.line(points, fill=colors["route"], width=int(scene["route_width"]), joint="curve")

    marker_radius = int(scene["marker_size"])
    for index, (point, operation) in enumerate(zip(points[1:-1], scene["operations"], strict=True), start=1):
        center = [int(point[0]), int(point[1])]
        _draw_operation(draw, center, int(operation), marker_radius, colors["operation"])
        label = str(index)
        bbox = draw.textbbox((0, 0), label, font=font)
        tw = bbox[2] - bbox[0]
        draw.text((center[0] - tw / 2, center[1] - marker_radius - 18), label, fill=colors["text"], font=font)

    _draw_state_badge(
        draw,
        [int(value) for value in points[0]],
        int(scene["initial_state"]),
        marker_radius + 7,
        colors,
        font,
    )
    end_x, end_y = points[-1]
    draw.ellipse((end_x - 10, end_y - 10, end_x + 10, end_y + 10), outline=colors["neutral"], width=3)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Compute the missing label from the visible start and station fields."""

    trace = _trace(int(scene["initial_state"]), [int(value) for value in scene["operations"]])
    missing = _missing(trace)
    if len(missing) != 1:
        raise ValueError("scene does not have a unique missing state")
    return STATE_LABELS[missing[0]]


def margin(scene: dict[str, Any]) -> float:
    trace = _trace(int(scene["initial_state"]), [int(value) for value in scene["operations"]])
    return round(_coverage_margin(trace), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    if len(scene.get("operations", [])) != STATION_COUNT:
        return True
    trace = _trace(int(scene.get("initial_state", -1)), [int(value) for value in scene["operations"]])
    return len(_missing(trace)) != 1 or margin(scene) <= 0.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict]]:
    # Every answer-bearing field is visible in the raster, so this candidate
    # declares no latent pixel-identical aliases.
    return []
