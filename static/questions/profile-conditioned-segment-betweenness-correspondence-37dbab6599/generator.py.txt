"""Segment-betweenness correspondence world renderer (paper-semantic family).

A cross-panel *segment-extent (betweenness)* correspondence task. Each panel
shows two gray ``anchors`` and one colored ``target`` dot. The two anchors
define a segment. The reference panel's colored target is placed either clearly
``between`` the two anchors (inside the span, on the line) or clearly ``beyond``
them (past one anchor, on the continuation of the line) - randomized. Three
candidate panels A, B, C each show their own two anchors (re-arranged elsewhere
in the raster) and one colored target. Exactly one candidate reproduces the
reference's between/outside *relationship* (its target inside-vs-beyond its own
two anchors matches the reference's state); the other two have the opposite
state. The question asks which candidate A, B, or C reproduces the same
betweenness relationship as the reference.

This is *not* a collinearity world: here the target is always collinear with the
two anchors, and the decision is the target's *along-segment extent* (inside the
anchor span vs. beyond it), not its perpendicular offset from the anchor line.

The pixel-only inverse arm must, in every panel, recover the two anchors and the
colored target, reconstruct the anchor-to-anchor line and its midpoint, project
the target onto the line, and read whether the projection lies inside the anchor
span or beyond one end, then match the candidate whose relation equals the
reference's.

Latent state z (every scene field is rendered - none hidden):

    panels : ordered list of four panels [reference, A, B, C]; each panel is
             {an1: [x, y], an2: [x, y], target: [x, y]}. The target position of
             each panel encodes that panel's between/outside extent state.
    width  : raster width.
    height : raster height.

The analytic gold is the label (A/B/C) of the candidate panel whose target's
between/outside relationship to its own anchors equals the reference's. Because
panels (all anchor and target positions) are drawn verbatim, the gold is
recoverable from pixels alone; no gold-only field exists.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "segment_betweenness_correspondence-0.1.0"

BACKGROUND = (255, 255, 255)
ANCHOR_COLOR = (95, 95, 95)

TARGET_PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (235, 40, 40),
    "green": (0, 170, 70),
    "blue": (40, 60, 235),
    "orange": (245, 140, 20),
}

# Public visual geometry (constant across scenes for ink/position counterbalance).
WIDTH = 900
HEIGHT = 940
BAND_CY = [160, 390, 620, 850]   # reference, A, B, C top to bottom
RADIUS = 15
ANCHOR_T = 70                    # anchors sit +/-ANCHOR_T along the panel line
BETWEEN_MAX = 25                 # max |offset| for a between-state target
EXT = 50                         # additional offset beyond one anchor for outside
THRESH = ANCHOR_T                # between/outside classifier threshold (pixels)
MAX_ANGLE = 0.45                 # radians; keeps panels compact / bands separated

_MARGIN_FLOOR = 2.0


def _along_offset(panel: dict) -> float:
    """Signed projection of the target onto the anchor line, in pixels, measured
    from the midpoint between the two anchors (positive toward anchor 2)."""
    (ax1, ay1) = panel["an1"]
    (ax2, ay2) = panel["an2"]
    tx, ty = panel["target"]
    dx, dy = ax2 - ax1, ay2 - ay1
    length = math.hypot(dx, dy)
    if length == 0:
        raise ValueError("degenerate anchor pair")
    ux, uy = dx / length, dy / length
    mx, my = (ax1 + ax2) / 2.0, (ay1 + ay2) / 2.0
    return (tx - mx) * ux + (ty - my) * uy


def _state(panel: dict) -> str:
    return "between" if abs(_along_offset(panel)) < THRESH else "outside"


def _build_panel(rng: random.Random, yc: float, between: bool) -> dict:
    angle = rng.uniform(-MAX_ANGLE, MAX_ANGLE)
    c, s = math.cos(angle), math.sin(angle)
    ux, uy = c, s          # unit along-line direction
    cx = WIDTH // 2
    if between:
        off = rng.uniform(-BETWEEN_MAX, BETWEEN_MAX)
    else:
        off = ANCHOR_T + EXT
    return {
        "an1": [round(cx - ANCHOR_T * ux), round(yc - ANCHOR_T * uy)],
        "an2": [round(cx + ANCHOR_T * ux), round(yc + ANCHOR_T * uy)],
        "target": [round(cx + off * ux), round(yc + off * uy)],
        "target_color": rng.choice(list(TARGET_PALETTE)),
    }


def sample_scene(seed: int) -> dict:
    """Sample a balanced, deterministic latent scene (dict)."""
    rng = random.Random(seed)
    ref_between = rng.random() < 0.5
    match_slot = rng.randrange(3)
    cand_states = [not ref_between, not ref_between, not ref_between]
    cand_states[match_slot] = ref_between

    panels = [_build_panel(rng, BAND_CY[0], ref_between)]
    for i in range(3):
        panels.append(_build_panel(rng, BAND_CY[i + 1], cand_states[i]))

    return {
        "panels": panels,
        "width": WIDTH,
        "height": HEIGHT,
    }


def _draw_dot(draw, pos, color, radius: int = RADIUS) -> None:
    x, y = int(pos[0]), int(pos[1])
    draw.ellipse([x - radius, y - radius, x + radius, y + radius], fill=color)


def render(scene: dict) -> Image.Image:
    w = int(scene["width"])
    h = int(scene["height"])
    img = Image.new("RGB", (w, h), BACKGROUND)
    draw = ImageDraw.Draw(img)

    for panel in scene["panels"]:
        ax1, ay1 = panel["an1"]
        ax2, ay2 = panel["an2"]
        draw.ellipse(
            [ax1 - RADIUS, ay1 - RADIUS, ax1 + RADIUS, ay1 + RADIUS], fill=ANCHOR_COLOR
        )
        draw.ellipse(
            [ax2 - RADIUS, ay2 - RADIUS, ax2 + RADIUS, ay2 + RADIUS], fill=ANCHOR_COLOR
        )
        _draw_dot(draw, panel["target"], TARGET_PALETTE[panel["target_color"]])
    return img


def analytic_gold(scene: dict) -> str:
    """Decision: label (A/B/C) of the candidate panel whose target between/outside
    relationship to its own anchors matches the reference panel's state."""
    panels = scene["panels"]
    states = []
    for p in panels:
        a1x, a1y = p["an1"]
        a2x, a2y = p["an2"]
        tx, ty = p["target"]
        dx = a2x - a1x
        dy = a2y - a1y
        length = math.hypot(dx, dy)
        ux = dx / length
        uy = dy / length
        mx = (a1x + a2x) / 2.0
        my = (a1y + a2y) / 2.0
        offset = (tx - mx) * ux + (ty - my) * uy
        states.append("between" if abs(offset) < THRESH else "outside")
    ref_state = states[0]
    for i in range(1, 4):
        if states[i] == ref_state:
            return "ABC"[i - 1]
    raise ValueError("no candidate reproduced the reference betweenness relationship")


def margin(scene: dict) -> float:
    """One-sided verifiability margin: minimum, over every panel, of the distance
    from that panel's target to the between/outside classification boundary
    (|offset| = THRESH)."""
    return float(min(abs(abs(_along_offset(p)) - THRESH) for p in scene["panels"]))


def is_quarantined(scene: dict) -> bool:
    """Quarantine only a genuinely ambiguous scene. Between targets sit at most
    BETWEEN_MAX (25) px from the midpoint while outside targets sit 120 px beyond
    one anchor; the midpoint-to-threshold distance is therefore always >= 45 px,
    so ambiguous scenes are never produced by the sampler."""
    return margin(scene) < _MARGIN_FLOOR


def latent_symmetries(scene: dict) -> list:
    """not_applicable: every label-influencing field (all anchor and target
    positions) is rendered verbatim, so the raster uniquely determines each
    panel's between/outside state and the matching candidate; no pixel-identical
    transform is declared."""
    return []
