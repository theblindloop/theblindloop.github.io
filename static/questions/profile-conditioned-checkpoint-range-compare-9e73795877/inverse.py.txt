"""Independent image-only inverse for paired checkpoint trace range."""

from __future__ import annotations

import numpy as np
from PIL import Image


CHECKPOINTS = 9


def _colored_mask(rgb: np.ndarray) -> np.ndarray:
    channels = rgb.astype(np.int16)
    spread = channels.max(axis=2) - channels.min(axis=2)
    return (spread >= 38) & (channels.min(axis=2) <= 225)


def _range_from_panel(mask: np.ndarray, x0: int, x1: int) -> float:
    crop = mask[:, x0:x1]
    ys, xs = np.nonzero(crop)
    if len(xs) < 180:
        raise ValueError("colored trace is missing or too small")
    x_min = int(xs.min())
    x_max = int(xs.max())
    span = x_max - x_min
    if span < 250:
        raise ValueError("trace endpoint span is too short")
    # Filled markers extend the colored bounding box by roughly one radius.
    # Estimate that visible offset from the total span, then sample a small
    # window around nine equally spaced marker centers.
    edge_offset = max(5, min(14, int(round(span * 0.025))))
    first = x_min + edge_offset
    last = x_max - edge_offset
    marker_x = np.linspace(first, last, CHECKPOINTS)
    centers = []
    for position in marker_x:
        window = np.abs(xs - position) <= 6.0
        local_y = ys[window]
        if len(local_y) < 8:
            raise ValueError("checkpoint marker cannot be recovered")
        centers.append(float(np.median(local_y)))
    measured = max(centers) - min(centers)
    if measured < 14.0:
        raise ValueError("vertical range is too small to resolve")
    return measured


def decision_from_image(image: Image.Image) -> str:
    """Return A or B using only the final RGB raster."""

    rgb = np.asarray(image.convert("RGB"))
    if rgb.ndim != 3 or rgb.shape[2] != 3 or rgb.shape[1] < 800 or rgb.shape[0] < 400:
        raise ValueError("image is not a usable paired trace panel")
    mask = _colored_mask(rgb)
    midpoint = rgb.shape[1] // 2
    range_a = _range_from_panel(mask, 0, midpoint)
    range_b = _range_from_panel(mask, midpoint, rgb.shape[1])
    if abs(range_a - range_b) < 14.0:
        raise ValueError("vertical-range comparison is ambiguous")
    return "A" if range_a > range_b else "B"
