"""Deterministic paired checkpoint-trace renderer and analytic world."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw


RENDERER_VERSION = "checkpoint_range_compare-0.1.0"
CANVAS = (960, 480)
PANEL_BOXES = {
    "A": (28, 44, 452, 436),
    "B": (508, 44, 932, 436),
}
PLOT_BOXES = {
    "A": (78, 92, 402, 390),
    "B": (558, 92, 882, 390),
}
CHECKPOINTS = 9
QUARANTINE_GAP = 32.0
PALETTE = (
    (218, 55, 70),
    (44, 105, 213),
    (39, 157, 100),
    (157, 74, 181),
    (226, 132, 36),
    (25, 157, 170),
)


def _vertical_range(points: list[list[float]]) -> float:
    ys = [float(point[1]) for point in points]
    return max(ys) - min(ys)


def _profile(rng: random.Random, amplitude: float, center: float) -> list[float]:
    raw = [rng.uniform(-1.0, 1.0) for _ in range(CHECKPOINTS)]
    low = min(raw)
    high = max(raw)
    normalized = [(value - low) / (high - low) for value in raw]
    return [center - amplitude / 2.0 + amplitude * value for value in normalized]


def _points(
    plot_box: tuple[int, int, int, int], values: list[float]
) -> list[list[float]]:
    left, top, right, bottom = plot_box
    x_step = (right - left) / float(CHECKPOINTS - 1)
    height = float(bottom - top)
    return [
        [round(left + index * x_step, 3), round(top + (1.0 - value) * height, 3)]
        for index, value in enumerate(values)
    ]


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one balanced paired-trace scene from an integer seed."""

    geometry_rng = random.Random(int(seed) * 1000003 + 17)
    style_rng = random.Random(int(seed) * 9176 + 401)
    winner = "A" if int(seed) % 2 else "B"
    near_boundary = int(seed) % 11 == 0

    if near_boundary:
        winner_amplitude = geometry_rng.uniform(0.31, 0.34)
        loser_amplitude = winner_amplitude - geometry_rng.uniform(0.075, 0.085)
    else:
        winner_amplitude = geometry_rng.uniform(0.55, 0.66)
        loser_amplitude = geometry_rng.uniform(0.22, 0.34)

    amplitude_a = winner_amplitude if winner == "A" else loser_amplitude
    amplitude_b = winner_amplitude if winner == "B" else loser_amplitude
    center_a = geometry_rng.uniform(0.47, 0.53)
    center_b = geometry_rng.uniform(0.47, 0.53)
    values_a = _profile(geometry_rng, amplitude_a, center_a)
    values_b = _profile(geometry_rng, amplitude_b, center_b)
    points_a = _points(PLOT_BOXES["A"], values_a)
    points_b = _points(PLOT_BOXES["B"], values_b)

    color_a = style_rng.choice(PALETTE)
    color_b = style_rng.choice(PALETTE)
    return {
        "points_a": points_a,
        "points_b": points_b,
        "color_a": list(color_a),
        "color_b": list(color_b),
        "stroke_a": round(style_rng.uniform(4.0, 7.0), 2),
        "stroke_b": round(style_rng.uniform(4.0, 7.0), 2),
        "marker_a": round(style_rng.uniform(7.0, 10.0), 2),
        "marker_b": round(style_rng.uniform(7.0, 10.0), 2),
        "canvas": [CANVAS[0], CANVAS[1]],
        "background": "#ffffff",
    }


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    label: str,
    points: list[list[float]],
    color: list[int],
    stroke: float,
    marker: float,
    scale: int,
) -> None:
    frame = tuple(int(value * scale) for value in PANEL_BOXES[label])
    plot = tuple(int(value * scale) for value in PLOT_BOXES[label])
    draw.rectangle(frame, outline=(183, 188, 198), width=max(1, scale))
    left, top, right, bottom = plot
    draw.rectangle(plot, outline=(216, 220, 228), width=max(1, scale))
    for fraction in (0.25, 0.5, 0.75):
        y = int(round(top + fraction * (bottom - top)))
        draw.line((left, y, right, y), fill=(239, 241, 245), width=max(1, scale))
    draw.text((frame[0] + 12 * scale, frame[1] + 10 * scale), label, fill=(30, 35, 45))

    scaled_points = [(int(round(point[0] * scale)), int(round(point[1] * scale))) for point in points]
    draw.line(
        scaled_points,
        fill=tuple(color),
        width=max(1, int(round(stroke * scale))),
        joint="curve",
    )
    radius = marker * scale
    for x, y in scaled_points:
        draw.ellipse(
            (int(round(x - radius)), int(round(y - radius)), int(round(x + radius)), int(round(y + radius))),
            fill=tuple(color),
        )


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only from the supplied scene dictionary."""

    width, height = (int(scene["canvas"][0]), int(scene["canvas"][1]))
    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), scene["background"])
    draw = ImageDraw.Draw(image)
    _draw_panel(draw, "A", scene["points_a"], scene["color_a"], scene["stroke_a"], scene["marker_a"], scale)
    _draw_panel(draw, "B", scene["points_b"], scene["color_b"], scene["stroke_b"], scene["marker_b"], scale)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    range_a = _vertical_range(scene["points_a"])
    range_b = _vertical_range(scene["points_b"])
    return "A" if range_a > range_b else "B"


def margin(scene: dict[str, Any]) -> float:
    return abs(_vertical_range(scene["points_a"]) - _vertical_range(scene["points_b"]))


def is_quarantined(scene: dict[str, Any]) -> bool:
    if margin(scene) < QUARANTINE_GAP:
        return True
    for points in (scene["points_a"], scene["points_b"]):
        if len(points) != CHECKPOINTS:
            return True
        if any(float(right[0]) <= float(left[0]) + 12.0 for left, right in zip(points, points[1:])):
            return True
        if min(float(point[1]) for point in points) < 82.0 or max(float(point[1]) for point in points) > 400.0:
            return True
    return False


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No aliases are declared; the answer-bearing coordinates are visible."""

    return []
