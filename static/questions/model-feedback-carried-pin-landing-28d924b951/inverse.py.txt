"""Independent PNG-only inverse arm for carried-pin landing."""

from __future__ import annotations

import itertools
import math

import numpy as np
from PIL import Image

WIDTH = 1150
HEIGHT = 790
GRID_N = 9
MAIN_CELL = 50
OPTION_CELL = 26
MAIN_ORIGIN = (45, 180)
OPTION_ORIGINS = ((610, 175), (890, 175), (610, 475), (890, 475))

BLUE = np.array((37, 99, 235), dtype=np.uint8)
AMBER = np.array((245, 158, 11), dtype=np.uint8)
RED = np.array((220, 38, 38), dtype=np.uint8)
GREEN = np.array((22, 163, 74), dtype=np.uint8)


class OracleAbstention(ValueError):
    """The required visible geometric marks cannot be decoded confidently."""


def _mask(arr: np.ndarray, color: np.ndarray) -> np.ndarray:
    return np.all(arr == color, axis=2)


def _cell_set(
    arr: np.ndarray, origin: tuple[int, int], cell: int, color: np.ndarray, label: str
) -> list[np.ndarray]:
    found: list[np.ndarray] = []
    for y in range(GRID_N):
        for x in range(GRID_N):
            cx = int(round(origin[0] + (x + 0.5) * cell))
            cy = int(round(origin[1] + (y + 0.5) * cell))
            radius = 7 if cell >= 40 else 4
            patch = arr[cy - radius : cy + radius + 1, cx - radius : cx + radius + 1]
            threshold = 180 if cell >= 40 else 60
            if int(_mask(patch, color).sum()) >= threshold:
                found.append(np.array((float(x), float(y))))
    if len(found) != 5:
        raise OracleAbstention(f"{label}: expected five colored cells, found {len(found)}")
    return found


def _fit_axis(arr: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    ox, oy = MAIN_ORIGIN
    size = GRID_N * MAIN_CELL
    crop = arr[oy - 6 : oy + size + 7, ox - 6 : ox + size + 7]
    ys, xs = np.nonzero(_mask(crop, RED))
    if not 500 <= len(xs) <= 4500:
        raise OracleAbstention(f"reflection line has {len(xs)} red pixels")
    points_px = np.column_stack((xs + ox - 6, ys + oy - 6)).astype(np.float64)
    points = np.empty_like(points_px)
    points[:, 0] = (points_px[:, 0] - ox) / MAIN_CELL - 0.5
    points[:, 1] = (points_px[:, 1] - oy) / MAIN_CELL - 0.5
    center = points.mean(axis=0)
    _, _, vt = np.linalg.svd(points - center, full_matrices=False)
    direction = vt[0]
    normal = np.array((-direction[1], direction[0]))
    residual = np.abs((points - center) @ normal)
    extent = float(np.ptp((points - center) @ direction))
    if float(np.quantile(residual, 0.95)) > 0.13 or extent < 4.0:
        raise OracleAbstention("malformed reflection line")
    return center, direction


def _fit_pin(arr: np.ndarray) -> np.ndarray:
    ox, oy = MAIN_ORIGIN
    size = GRID_N * MAIN_CELL
    crop = arr[oy - 20 : oy + size + 21, ox - 20 : ox + size + 21]
    ys, xs = np.nonzero(_mask(crop, GREEN))
    if not 480 <= len(xs) <= 620:
        raise OracleAbstention(f"pin has {len(xs)} green pixels")
    points = np.column_stack((xs + ox - 20, ys + oy - 20)).astype(np.float64)
    center_px = points.mean(axis=0)
    radii = np.linalg.norm(points - center_px, axis=1)
    if float(radii.max()) > 14.5 or float(np.quantile(radii, 0.85)) < 9.0:
        raise OracleAbstention("malformed pin")
    return np.array(((center_px[0] - ox) / MAIN_CELL - 0.5,
                     (center_px[1] - oy) / MAIN_CELL - 0.5))


def _set_error(expected: list[np.ndarray], observed: list[np.ndarray]) -> float:
    best = math.inf
    for permutation in itertools.permutations(observed):
        best = min(best, max(float(np.linalg.norm(a - b)) for a, b in zip(expected, permutation)))
    return best


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    if rgb.size != (WIDTH, HEIGHT):
        raise OracleAbstention(f"unexpected raster size {rgb.size}")
    arr = np.asarray(rgb, dtype=np.uint8)
    source = _cell_set(arr, MAIN_ORIGIN, MAIN_CELL, BLUE, "source")
    options = [
        _cell_set(arr, origin, OPTION_CELL, AMBER, f"option {index + 1}")
        for index, origin in enumerate(OPTION_ORIGINS)
    ]
    axis_center, axis_direction = _fit_axis(arr)
    pin = _fit_pin(arr)
    pin_delta = pin - axis_center
    moved_pin = axis_center + 2.0 * axis_direction * float(pin_delta @ axis_direction) - pin_delta
    expected: list[np.ndarray] = []
    for point in source:
        delta = point - axis_center
        reflected = axis_center + 2.0 * axis_direction * float(delta @ axis_direction) - delta
        dx, dy = reflected - moved_pin
        expected.append(moved_pin + np.array((-dy, dx)))
    errors = [_set_error(expected, option) for option in options]
    order = sorted(range(4), key=lambda index: errors[index])
    if errors[order[0]] > 0.18:
        raise OracleAbstention(f"no exact carried-pin landing: {errors}")
    if errors[order[1]] - errors[order[0]] < 0.45:
        raise OracleAbstention(f"landing is not unique: {errors}")
    return f"option-{order[0] + 1}"
