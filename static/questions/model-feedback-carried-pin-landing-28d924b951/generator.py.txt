"""Deterministic renderer for a reflection with a transported pivot."""

from __future__ import annotations

import copy
import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1150
HEIGHT = 790
GRID_N = 9
MAIN_CELL = 50
OPTION_CELL = 26
MAIN_ORIGIN = (45, 180)
OPTION_ORIGINS = ((610, 175), (890, 175), (610, 475), (890, 475))

WHITE = (255, 255, 255)
INK = (31, 41, 55)
GRID = (207, 215, 225)
BLUE = (37, 99, 235)
AMBER = (245, 158, 11)
RED = (220, 38, 38)
GREEN = (22, 163, 74)
PALE = (248, 250, 252)

FONT_BOLD = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
FONT_REGULAR = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"

_BASE_SHAPES = (
    ((0, 0), (0, 1), (0, 2), (1, 2), (2, 2)),
    ((0, 0), (1, 0), (2, 0), (0, 1), (1, 2)),
    ((0, 0), (1, 0), (1, 1), (2, 1), (2, 2)),
    ((0, 0), (0, 1), (1, 1), (1, 2), (2, 1)),
    ((0, 0), (1, 0), (2, 0), (3, 0), (1, 1)),
    ((0, 0), (0, 1), (0, 2), (1, 0), (2, 1)),
)


def _shape_variant(shape: Iterable[tuple[int, int]], code: int) -> set[tuple[int, int]]:
    points = []
    for x, y in shape:
        if code >= 4:
            x = -x
        for _ in range(code % 4):
            x, y = -y, x
        points.append((x, y))
    min_x = min(x for x, _ in points)
    min_y = min(y for _, y in points)
    return {(x - min_x, y - min_y) for x, y in points}


def _reflect_point(point: tuple[int, int], axis: dict) -> tuple[int, int]:
    x, y = point
    k = int(axis["k"])
    kind = axis["kind"]
    if kind == "vertical":
        return 2 * k - x, y
    if kind == "horizontal":
        return x, 2 * k - y
    if kind == "diag_down":
        return y - k, x + k
    if kind == "diag_up":
        return k - y, k - x
    raise ValueError(f"unknown axis kind {kind!r}")


def _rotate_cw(point: tuple[int, int], pivot: tuple[int, int]) -> tuple[int, int]:
    x, y = point
    px, py = pivot
    return px - (y - py), py + (x - px)


def _rotate_ccw(point: tuple[int, int], pivot: tuple[int, int]) -> tuple[int, int]:
    x, y = point
    px, py = pivot
    return px + (y - py), py - (x - px)


def _axis_side(point: tuple[int, int], axis: dict) -> int:
    x, y = point
    k = int(axis["k"])
    if axis["kind"] == "vertical":
        return x - k
    if axis["kind"] == "horizontal":
        return y - k
    if axis["kind"] == "diag_down":
        return y - x - k
    return x + y - k


def _inside(points: Iterable[tuple[int, int]]) -> bool:
    return all(0 <= x < GRID_N and 0 <= y < GRID_N for x, y in points)


def _serial(points: Iterable[tuple[int, int]]) -> list[list[int]]:
    return [[x, y] for x, y in sorted(points, key=lambda p: (p[1], p[0]))]


def _correct_set(scene: dict) -> set[tuple[int, int]]:
    source = {tuple(point) for point in scene["source"]}
    pin = tuple(scene["pin"])
    reflected = {_reflect_point(point, scene["axis"]) for point in source}
    moved_pin = _reflect_point(pin, scene["axis"])
    return {_rotate_cw(point, moved_pin) for point in reflected}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0xC471E9A3)
    axis_choices = (
        ("vertical", tuple(range(2, 7))),
        ("horizontal", tuple(range(2, 7))),
        ("diag_down", tuple(range(-2, 3))),
        ("diag_up", tuple(range(6, 11))),
    )
    for _ in range(30000):
        shape = _shape_variant(rng.choice(_BASE_SHAPES), rng.randrange(8))
        shape_w = max(x for x, _ in shape) + 1
        shape_h = max(y for _, y in shape) + 1
        tx = rng.randrange(GRID_N - shape_w + 1)
        ty = rng.randrange(GRID_N - shape_h + 1)
        source = {(x + tx, y + ty) for x, y in shape}

        kind, values = rng.choice(axis_choices)
        axis = {"kind": kind, "k": int(rng.choice(values))}
        source_sides = [_axis_side(point, axis) for point in source]
        if not (all(v >= 1 for v in source_sides) or all(v <= -1 for v in source_sides)):
            continue
        reflected = {_reflect_point(point, axis) for point in source}
        if not _inside(reflected):
            continue

        pin = (rng.randrange(1, GRID_N - 1), rng.randrange(1, GRID_N - 1))
        if pin in source or abs(_axis_side(pin, axis)) < 1:
            continue
        if (_axis_side(pin, axis) > 0) != (source_sides[0] > 0):
            continue
        moved_pin = _reflect_point(pin, axis)
        if not _inside((moved_pin,)) or moved_pin in reflected:
            continue

        correct = {_rotate_cw(point, moved_pin) for point in reflected}
        if not _inside(correct) or moved_pin in correct:
            continue
        wrong_pool = (
            {_rotate_cw(point, pin) for point in reflected},
            {_rotate_ccw(point, moved_pin) for point in reflected},
            reflected,
            {_rotate_cw(point, pin) for point in source},
            {_rotate_cw(_rotate_cw(point, moved_pin), moved_pin) for point in reflected},
        )
        usable: list[set[tuple[int, int]]] = []
        for wrong in wrong_pool:
            if (
                _inside(wrong)
                and wrong != correct
                and wrong not in usable
                and len(correct & wrong) <= 3
            ):
                usable.append(wrong)
        if len(usable) < 3:
            continue
        rng.shuffle(usable)
        chosen = usable[:3]
        correct_index = int(seed) % 4
        option_sets = []
        wrong_index = 0
        for index in range(4):
            if index == correct_index:
                option_sets.append(correct)
            else:
                option_sets.append(chosen[wrong_index])
                wrong_index += 1
        draw_order = list(range(4))
        rng.shuffle(draw_order)
        scene = {
            "source": _serial(source),
            "axis": axis,
            "pin": list(pin),
            "options": [_serial(option) for option in option_sets],
            "option_draw_order": draw_order,
        }
        analytic_gold(scene)
        return scene
    raise RuntimeError("could not sample a clear carried-pin scene")


def analytic_gold(scene: dict) -> str:
    correct = _correct_set(scene)
    options = [{tuple(point) for point in option} for option in scene["options"]]
    winners = [index for index, option in enumerate(options) if option == correct]
    if len(winners) != 1:
        raise ValueError(f"scene has {len(winners)} exact landing options")
    return f"option-{winners[0] + 1}"


def margin(scene: dict) -> float:
    correct = _correct_set(scene)
    overlaps = sorted(
        (len(correct & {tuple(point) for point in option}) for option in scene["options"]),
        reverse=True,
    )
    return float(overlaps[0] - overlaps[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["option_draw_order"] = list(reversed(scene["option_draw_order"]))
    return [("reverse_disjoint_option_painter_order", twin)]


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(FONT_BOLD if bold else FONT_REGULAR, size)


def _center(origin: tuple[int, int], cell: int, point: tuple[float, float]) -> tuple[float, float]:
    x, y = point
    return origin[0] + (x + 0.5) * cell, origin[1] + (y + 0.5) * cell


def _axis_endpoints(axis: dict) -> tuple[tuple[float, float], tuple[float, float]]:
    low, high = -0.42, GRID_N - 0.58
    kind, k = axis["kind"], float(axis["k"])
    if kind == "vertical":
        a, b = (k, low), (k, high)
    elif kind == "horizontal":
        a, b = (low, k), (high, k)
    elif kind == "diag_down":
        x0, x1 = max(low, low - k), min(high, high - k)
        a, b = (x0, x0 + k), (x1, x1 + k)
    else:
        x0, x1 = max(low, k - high), min(high, k - low)
        a, b = (x0, k - x0), (x1, k - x1)
    return _center(MAIN_ORIGIN, MAIN_CELL, a), _center(MAIN_ORIGIN, MAIN_CELL, b)


def _dashed_line(draw: ImageDraw.ImageDraw, start: tuple[float, float], end: tuple[float, float]) -> None:
    dx, dy = end[0] - start[0], end[1] - start[1]
    length = math.hypot(dx, dy)
    ux, uy = dx / length, dy / length
    distance = 0.0
    while distance < length:
        stop = min(distance + 20.0, length)
        draw.line(
            (start[0] + ux * distance, start[1] + uy * distance,
             start[0] + ux * stop, start[1] + uy * stop),
            fill=RED, width=8,
        )
        distance += 32.0


def _draw_grid(draw: ImageDraw.ImageDraw, origin: tuple[int, int], cell: int) -> None:
    size = GRID_N * cell
    x, y = origin
    draw.rectangle((x, y, x + size, y + size), fill=PALE, outline=INK, width=2)
    for step in range(1, GRID_N):
        draw.line((x + step * cell, y, x + step * cell, y + size), fill=GRID, width=1)
        draw.line((x, y + step * cell, x + size, y + step * cell), fill=GRID, width=1)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), WHITE)
    draw = ImageDraw.Draw(image)
    draw.text((34, 18), "CARRY THE PIN THROUGH THE REFLECTION", fill=INK, font=_font(30, True))
    draw.text((36, 65), "1  REFLECT blue + green pin", fill=INK, font=_font(22, True))
    draw.text((450, 65), "2  TURN blue 90° CLOCKWISE around moved pin", fill=INK, font=_font(22, True))
    draw.text((45, 137), "START", fill=INK, font=_font(24, True))
    draw.text((610, 137), "AMBER OPTIONS — SAME GRID", fill=INK, font=_font(24, True))

    _draw_grid(draw, MAIN_ORIGIN, MAIN_CELL)
    axis_a, axis_b = _axis_endpoints(scene["axis"])
    _dashed_line(draw, axis_a, axis_b)
    for x, y in scene["source"]:
        left = MAIN_ORIGIN[0] + x * MAIN_CELL + 4
        top = MAIN_ORIGIN[1] + y * MAIN_CELL + 4
        draw.rectangle((left, top, left + MAIN_CELL - 8, top + MAIN_CELL - 8), fill=BLUE)
    px, py = _center(MAIN_ORIGIN, MAIN_CELL, tuple(scene["pin"]))
    draw.ellipse((px - 17, py - 17, px + 17, py + 17), fill=WHITE, outline=INK, width=3)
    draw.ellipse((px - 13, py - 13, px + 13, py + 13), fill=GREEN)

    for index in scene.get("option_draw_order", list(range(4))):
        origin = OPTION_ORIGINS[int(index)]
        _draw_grid(draw, origin, OPTION_CELL)
        for x, y in scene["options"][int(index)]:
            left = origin[0] + x * OPTION_CELL + 3
            top = origin[1] + y * OPTION_CELL + 3
            draw.rectangle((left, top, left + OPTION_CELL - 6, top + OPTION_CELL - 6), fill=AMBER)
        badge_x, badge_y = origin[0] - 18, origin[1] + 18
        draw.ellipse((badge_x - 16, badge_y - 16, badge_x + 16, badge_y + 16), fill=INK)
        draw.text((badge_x, badge_y - 1), str(int(index) + 1), fill=WHITE, font=_font(20, True), anchor="mm")
    return image
