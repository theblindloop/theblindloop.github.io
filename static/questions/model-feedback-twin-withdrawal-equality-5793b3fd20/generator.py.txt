"""Deterministic renderer and analytic arm for the twin-withdrawal world.

Two rows of nine square cells are drawn, one above the other; each cell is
filled either dark or pale.  Beneath one cell of EACH row sits a small solid
arrowhead pointing straight up at that cell, marking it.

The reasoning program is a counterfactual intervention that is relational and
paired.  The two drawn nine-cell rows are the actually rendered world.  The
question asks about a state that is NEVER drawn: the pair of shorter eight-cell
rows that WOULD result if the arrow-marked cell were withdrawn from EACH row
and the remaining cells slid together to close the gap, preserving order.  The
verdict is whether the two surviving rows then read identically in dark and
pale.  The answer is carried only by performing BOTH withdrawals and comparing
the two never-rendered survivors; no crop, no single cell, no count, and no
comparison of the drawn rows decides it, because the drawn rows include the to-be-
removed cells and so generally differ from the survivors.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 760
HEIGHT = 620
N_CELLS = 9
CELL = 52
GAP = 12
ROW_A_Y = 170
ROW_B_Y = 396
ARROW_TOP_GAP = 26
ARROW_H = 42
ARROW_HALFW = 17
BACKGROUND = (252, 252, 252)
OUTLINE = (120, 120, 120)
ON_FILL = (40, 40, 40)
OFF_FILL = (236, 236, 236)
ARROW_FILL = (40, 40, 40)


def _survivor_pair(rng: random.Random, target: str) -> tuple[list[int], list[int]]:
    """A pair of 8-cell post-withdrawal rows with the requested equality class."""
    length = N_CELLS - 1
    left = [rng.randint(0, 1) for _ in range(length)]
    if target == "yes":
        return left, list(left)
    right = [rng.randint(0, 1) for _ in range(length)]
    if right == left:
        right[0] = 1 - right[0]
    return left, right


def sample_scene(seed: int, target: str | None = None) -> dict:
    rng = random.Random(int(seed))
    if target is None:
        target = "yes" if rng.randint(0, 1) == 0 else "no"
    survivor_a, survivor_b = _survivor_pair(rng, target)
    mark_a = rng.randint(0, N_CELLS - 1)
    mark_b = rng.randint(0, N_CELLS - 1)
    insert_a = rng.randint(0, 1)
    insert_b = rng.randint(0, 1)
    row_a = survivor_a[:mark_a] + [insert_a] + survivor_a[mark_a:]
    row_b = survivor_b[:mark_b] + [insert_b] + survivor_b[mark_b:]
    return {
        "canvas_size": [WIDTH, HEIGHT],
        "row_a": row_a,
        "row_b": row_b,
        "mark_a": mark_a,
        "mark_b": mark_b,
        "cell_px": CELL,
        "gap_px": GAP,
        "row_a_y": ROW_A_Y,
        "row_b_y": ROW_B_Y,
        "arrow_top_gap": ARROW_TOP_GAP,
        "arrow_h": ARROW_H,
        "arrow_halfw": ARROW_HALFW,
    }


def _row_left() -> int:
    total = N_CELLS * CELL + (N_CELLS - 1) * GAP
    return (WIDTH - total) // 2


def _draw_row(draw: ImageDraw.ImageDraw, row: list[int], row_y: int) -> list[int]:
    left = _row_left()
    centers: list[int] = []
    for index, value in enumerate(row):
        x = left + index * (CELL + GAP)
        cx = x + CELL // 2
        centers.append(cx)
        draw.rectangle(
            (x, row_y, x + CELL, row_y + CELL),
            fill=ON_FILL if value == 1 else OFF_FILL,
            outline=OUTLINE,
            width=3,
        )
    return centers


def _draw_arrow(
    draw: ImageDraw.ImageDraw, pivot_cx: int, row_y: int, top_gap: int, h: int, halfw: int
) -> None:
    top = row_y + CELL + top_gap
    base = top + h
    draw.polygon(
        [(pivot_cx, top), (pivot_cx - halfw, base), (pivot_cx + halfw, base)],
        fill=ARROW_FILL,
    )


def render(scene: dict) -> Image.Image:
    width, height = (int(value) for value in scene["canvas_size"])
    row_a = [int(value) for value in scene["row_a"]]
    row_b = [int(value) for value in scene["row_b"]]
    mark_a = int(scene["mark_a"])
    mark_b = int(scene["mark_b"])
    row_a_y = int(scene["row_a_y"])
    row_b_y = int(scene["row_b_y"])
    top_gap = int(scene["arrow_top_gap"])
    h = int(scene["arrow_h"])
    halfw = int(scene["arrow_halfw"])
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    centers_a = _draw_row(draw, row_a, row_a_y)
    centers_b = _draw_row(draw, row_b, row_b_y)
    _draw_arrow(draw, centers_a[mark_a], row_a_y, top_gap, h, halfw)
    _draw_arrow(draw, centers_b[mark_b], row_b_y, top_gap, h, halfw)
    return image


def _delete_at(seq: list[int], index: int) -> list[int]:
    return seq[:index] + seq[index + 1 :]


def analytic_gold(scene: dict) -> str:
    row_a = [int(value) for value in scene["row_a"]]
    row_b = [int(value) for value in scene["row_b"]]
    mark_a = int(scene["mark_a"])
    mark_b = int(scene["mark_b"])
    survivor_a = _delete_at(row_a, mark_a)
    survivor_b = _delete_at(row_b, mark_b)
    return "yes" if survivor_a == survivor_b else "no"


def margin(scene: dict) -> float:
    return 1.0


def is_quarantined(scene: dict) -> bool:
    row_a = scene["row_a"]
    row_b = scene["row_b"]
    mark_a = int(scene["mark_a"])
    mark_b = int(scene["mark_b"])
    if len(row_a) != N_CELLS or len(row_b) != N_CELLS:
        return True
    if any(value not in (0, 1) for value in row_a + row_b):
        return True
    if mark_a < 0 or mark_a >= N_CELLS or mark_b < 0 or mark_b >= N_CELLS:
        return True
    return False


def latent_symmetries(scene: dict) -> list:
    return []
