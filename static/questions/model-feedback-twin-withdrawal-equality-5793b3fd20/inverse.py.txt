"""Independent PNG-only inverse arm for the twin-withdrawal world.

Organized as two horizontal cell rows.  In each row we segment the nine
dark/pale square cells by structural masking and locate the column the
up-pointing dark arrowhead beneath that row marks, then withdraw exactly that
cell and report whether the two resulting eight-cell survivors (both never
drawn) are equal.  Operates only on pixels; uses no renderer geometry, scene
parameters, prompt, or gold code.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

N_CELLS = 9
ARROW_COUNT = 2


def _structure_mask(arr: np.ndarray) -> np.ndarray:
    chroma = arr.max(axis=2).astype(np.int16) - arr.min(axis=2).astype(np.int16)
    grays = chroma <= 16
    notwhite = (255 - arr.mean(axis=2)) >= 10
    return grays & notwhite


def _columns_at(band: np.ndarray) -> list[tuple[int, int]]:
    cols = band.any(axis=0)
    trans = np.diff(cols.astype(np.int8))
    starts = np.flatnonzero(trans == 1)
    ends = np.flatnonzero(trans == -1) + 1
    if cols[0]:
        starts = np.concatenate(([0], starts))
    if cols[-1]:
        ends = np.concatenate((ends, [len(cols)]))
    pairs = [(int(s), int(e)) for s, e in zip(starts, ends) if int(e) - int(s) >= 20]
    pairs.sort()
    return pairs


def _dark_mask(arr: np.ndarray) -> np.ndarray:
    return arr.mean(axis=2) < 120


def _cell_band_groups(struct: np.ndarray, y_lo: int, y_hi: int) -> list[tuple[int, int]]:
    rows = np.where(struct[y_lo:y_hi].any(axis=1))[0] + y_lo
    if len(rows) == 0:
        return []
    groups: list[tuple[int, int]] = []
    start = rows[0]
    prev = rows[0]
    for r in rows[1:]:
        if r > prev + 1:
            groups.append((start, prev))
            start = r
        prev = r
    groups.append((start, prev))
    return groups


def _band_span(struct: np.ndarray, top: int, bot: int) -> int:
    band = struct[top : bot + 1, :]
    cols = np.flatnonzero(band.any(axis=0))
    if len(cols) == 0:
        return 0
    return int(cols.max() - cols.min())


def _extract_row(arr: np.ndarray, struct: np.ndarray, top: int, bot: int) -> tuple[list[int], list[int]]:
    y = (top + bot) // 2
    band = struct[y - 2 : y + 3, :]
    pairs = _columns_at(band)
    if len(pairs) != N_CELLS:
        raise ValueError(f"expected {N_CELLS} cells in a row band, found {len(pairs)}")
    bits: list[int] = []
    centers: list[int] = []
    for start, end in pairs:
        x = (start + end) // 2
        centers.append(x)
        px = arr[y, x].astype(np.int16)
        bits.append(1 if float(px.mean()) < 120 else 0)
    return bits, centers


def _find_mark(arr: np.ndarray, centers: list[int], after_y: int, before_y: int) -> int:
    dark = _dark_mask(arr)
    region = np.zeros_like(dark)
    lo = max(0, after_y)
    hi = min(arr.shape[0], before_y)
    if hi <= lo:
        raise ValueError("empty arrow search region")
    region[lo:hi, :] = dark[lo:hi, :]
    ys, xs = np.nonzero(region)
    if len(xs) == 0:
        raise ValueError("no arrowhead found below the row")
    ax = float(xs.mean())
    return min(range(len(centers)), key=lambda i: abs(centers[i] - ax))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    struct = _structure_mask(arr)
    height = arr.shape[0]
    half = height // 2
    top_groups = _cell_band_groups(struct, 0, half)
    bot_groups = _cell_band_groups(struct, half, height)
    if not top_groups or not bot_groups:
        raise ValueError("could not find two row bands")
    tgroup = max(top_groups, key=lambda g: _band_span(struct, g[0], g[1]))
    bgroup = max(bot_groups, key=lambda g: _band_span(struct, g[0], g[1]))
    bits_a, centers_a = _extract_row(arr, struct, tgroup[0], tgroup[1])
    bits_b, centers_b = _extract_row(arr, struct, bgroup[0], bgroup[1])
    mark_a = _find_mark(arr, centers_a, tgroup[1] + 8, half)
    mark_b = _find_mark(arr, centers_b, bgroup[1] + 8, height)
    survivor_a = bits_a[:mark_a] + bits_a[mark_a + 1 :]
    survivor_b = bits_b[:mark_b] + bits_b[mark_b + 1 :]
    return "yes" if survivor_a == survivor_b else "no"
