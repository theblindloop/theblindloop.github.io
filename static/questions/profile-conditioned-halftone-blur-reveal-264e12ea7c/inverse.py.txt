"""Independent pixel-only inverse that explicitly low-pass filters the raster."""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageFilter

GRID = 17
MIN_MARGIN = 0.12
FRAME_RGB = (20, 174, 196)
EXPECTED_INK = 68


def _segment_distance(xx: np.ndarray, yy: np.ndarray, a: tuple[float, float], b: tuple[float, float]) -> np.ndarray:
    ax, ay = a
    bx, by = b
    dx, dy = bx - ax, by - ay
    t = np.clip(((xx - ax) * dx + (yy - ay) * dy) / (dx * dx + dy * dy), 0.0, 1.0)
    return np.hypot(xx - (ax + t * dx), yy - (ay + t * dy))


def _reference_masks() -> dict[str, np.ndarray]:
    yy, xx = np.mgrid[:GRID, :GRID]
    c = 8
    rr = np.hypot(xx-c, yy-c)
    circle = (rr >= 5.0) & (rr <= 6.5)
    triangle = np.minimum.reduce([
        _segment_distance(xx, yy, (8, 2), (2, 14)),
        _segment_distance(xx, yy, (2, 14), (14, 14)),
        _segment_distance(xx, yy, (14, 14), (8, 2)),
    ]) <= 0.9
    plus = (((np.abs(xx-c) <= 1) & (np.abs(yy-c) <= 6)) |
            ((np.abs(yy-c) <= 1) & (np.abs(xx-c) <= 6)))
    cross = (((np.abs((xx-c)-(yy-c)) <= 1) | (np.abs((xx-c)+(yy-c)) <= 1)) &
             (np.abs(xx-c) <= 6) & (np.abs(yy-c) <= 6))
    plus[c, c] = False
    cross[c, c] = False
    return {"circle": circle, "triangle": triangle, "plus": plus, "x": cross}


REFERENCES = _reference_masks()


def _frame_bounds(arr: np.ndarray) -> tuple[int, int, int, int]:
    delta = arr.astype(np.int32) - np.asarray(FRAME_RGB, dtype=np.int32)
    marker = np.sum(delta * delta, axis=2, dtype=np.int64) < 35 * 35
    yy, xx = np.nonzero(marker)
    if len(xx) < 2500:
        raise ValueError("cyan frame missing or incomplete")
    left, right, top, bottom = int(xx.min()), int(xx.max()), int(yy.min()), int(yy.max())
    if not (270 <= right-left <= 290 and 270 <= bottom-top <= 290):
        raise ValueError("cyan frame has invalid geometry")
    return left+5, top+5, right-4, bottom-4


def _lowpass_cells(image: Image.Image, bounds: tuple[int, int, int, int]) -> np.ndarray:
    crop = image.crop(bounds).convert("L").filter(ImageFilter.GaussianBlur(radius=4.5))
    arr = np.asarray(crop, dtype=np.float32)
    h, w = arr.shape
    means = np.zeros((GRID, GRID), dtype=float)
    for row in range(GRID):
        for col in range(GRID):
            cy = int(round((row + 0.5) * h / GRID))
            cx = int(round((col + 0.5) * w / GRID))
            patch = arr[max(0, cy-3):min(h, cy+4), max(0, cx-3):min(w, cx+4)]
            means[row, col] = float(patch.mean())
    order = np.argsort(means, axis=None)
    observed = np.zeros(GRID * GRID, dtype=bool)
    observed[order[:EXPECTED_INK]] = True
    return observed.reshape(GRID, GRID)


def _classify(image: Image.Image) -> tuple[str, float]:
    rgb = image.convert("RGB")
    cells = _lowpass_cells(rgb, _frame_bounds(np.asarray(rgb)))
    ranking = sorted(
        ((1.0 - float(np.mean(cells != mask)), label) for label, mask in REFERENCES.items()),
        reverse=True,
    )
    gap = ranking[0][0] - ranking[1][0]
    if gap < MIN_MARGIN:
        raise ValueError(f"ambiguous low-pass glyph (runner-up gap {gap:.3f})")
    return ranking[0][1], gap


def decision_from_image(image: Image.Image) -> str:
    return _classify(image)[0]
