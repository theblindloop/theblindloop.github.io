"""Twin-ball gauge consensus world.

Two disconnected colored ball gauges each sit in one of two clearly separated
vertical zones (top / bottom), delimited by a visible green midline tick inside
each channel. The decision asks whether the two ball readings AGREE on the same
zone (revealing a consistent global state) or conflict. The decisive pixel
support is the pair of coloured balls plus the two reference ticks: all local
and disconnected, i.e. focal + multipart.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

BG = (255, 255, 255)
CHANNEL = (214, 214, 214)
TICK = (60, 160, 40)
BLUE = (30, 90, 220)
RED = (220, 60, 40)

CANVAS_W = 512
CANVAS_H = 512

COMBOS = [(False, False), (False, True), (True, False), (True, True)]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    channel_w = 36
    channel_h = 220
    ball_r = 16
    gap = 64
    left_high, right_high = COMBOS[seed % len(COMBOS)]
    ox = rng.randint(90, 320)
    oy = rng.randint(120, 280)
    return {
        "left_high": left_high,
        "right_high": right_high,
        "ox": ox,
        "oy": oy,
        "channel_w": channel_w,
        "channel_h": channel_h,
        "ball_r": ball_r,
        "gap": gap,
        "canvas_w": CANVAS_W,
        "canvas_h": CANVAS_H,
    }


def _geometry(scene: dict) -> tuple[float, float, float, float, float]:
    """Return (left_x0, right_x0, left_ball_cx, right_ball_cx, midline_y)."""
    ox, oy = scene["ox"], scene["oy"]
    cw, ch = scene["channel_w"], scene["channel_h"]
    gap = scene["gap"]
    left_x0 = float(ox)
    right_x0 = float(ox + cw + gap)
    left_cx = ox + cw / 2.0
    right_cx = right_x0 + cw / 2.0
    midline = oy + ch / 2.0
    return left_x0, right_x0, left_cx, right_cx, midline


def _ball_cy(scene: dict, high: bool) -> float:
    oy = scene["oy"]
    ch = scene["channel_h"]
    br = scene["ball_r"]
    if high:
        return oy + br + 12
    return oy + ch - (br + 12)


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (scene["canvas_w"], scene["canvas_h"]), BG)
    d = ImageDraw.Draw(img)
    cw, ch = scene["channel_w"], scene["channel_h"]
    br = scene["ball_r"]
    oy = scene["oy"]
    left_x0, right_x0, left_cx, right_cx, midline = _geometry(scene)
    for x0 in (left_x0, right_x0):
        d.rectangle([x0, oy, x0 + cw - 1, oy + ch - 1], fill=CHANNEL)
    for x0 in (left_x0, right_x0):
        d.line([x0, midline, x0 + cw - 1, midline], fill=TICK, width=3)
    for cx, high, color in (
        (left_cx, scene["left_high"], BLUE),
        (right_cx, scene["right_high"], RED),
    ):
        cy = _ball_cy(scene, high)
        d.ellipse([cx - br, cy - br, cx + br, cy + br], fill=color)
    return img


def analytic_gold(scene: dict) -> str:
    return "yes" if scene["left_high"] == scene["right_high"] else "no"


def margin(scene: dict) -> float:
    _, _, _, _, midline = _geometry(scene)
    distances = [abs(_ball_cy(scene, high) - midline) for high in (scene["left_high"], scene["right_high"])]
    return float(min(distances))


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> iter:
    return iter(())
