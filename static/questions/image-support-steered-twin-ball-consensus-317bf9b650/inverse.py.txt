"""Pixel-only oracle for twin-ball gauge consensus.

The oracle never imports the renderer, prompts, generator, verifier, or any gold
code. It receives only the rendered image and independently recovers the two
ball-zone readings from pixel colours, then reports whether they agree.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

BLUE = (30, 90, 220)
RED = (220, 60, 40)
TICK = (60, 160, 40)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    blue = np.all(arr == BLUE, axis=2)
    red = np.all(arr == RED, axis=2)
    green = np.all(arr == TICK, axis=2)

    gy, gx = np.nonzero(green)
    if len(gy) == 0:
        raise ValueError("missing reference midline ticks")
    midline = float(gy.mean())

    def classify(mask: np.ndarray) -> bool:
        ys, xs = np.nonzero(mask)
        if len(ys) == 0:
            raise ValueError("missing coloured marker")
        return float(ys.mean()) < midline

    left_high = classify(blue)
    right_high = classify(red)
    return "yes" if left_high == right_high else "no"
