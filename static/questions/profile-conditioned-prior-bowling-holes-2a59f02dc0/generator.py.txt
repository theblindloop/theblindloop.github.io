"""Prior-conflict bowling-ball world: count visible finger holes.

The latent scene holds a bowling ball seen from directly above with ``n``
finger holes. A familiar canonical prior asserts that a bowling ball has THREE
finger holes; the counterfactual renders a non-canonical count and the decision
must be read from the visible holes alone, never from the prior. The pixel
oracle counts the actual dark hole blobs inside the ball.

Latent alias mode is ``not_applicable``: every latent field is directly
reflected in visible geometry (ball size, rotation, hole count, hole size and
orbit radius), so the rendering is injective in the latent scene and there is
no render-invariant alias. The canonical-prior name "bowling ball" is a
matched-control/counterfactual manipulation of the *question*, not of any
latent field.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

CANVAS = 512
CX = 256.0
CY = 256.0
BALL_RADIUS = 180.0
# Holes are punched on a ring (orbit) comfortably inside the ball so they never
# touch the ball rim and are always isolated, countable blobs.
HOLE_ORBIT = 66.0
HOLE_RADIUS = 20.0

# Supported off-quarantine counts. The canonical prior is 3; 2 and 4 are the
# counterfactuals that contradict it. All remain far enough apart (>= 88px
# centre separation at an 66px orbit) that each hole is an isolated blob.
SUPPORTED_N = (3, 2, 4)

# A count is quarantined when holes crowd so close that blobs can touch and be
# miscounted at pixel scale. At the smallest supported gap (2*66*sin(45deg) ~=
# 93px for n=4) with a ~20px hole radius, blobs are separated by ~53px, far
# above this threshold, so supported counts are never ambiguous.
QUAR_MIN_GAP_PX = 46.0

BACKGROUND = (255, 255, 255)
BALL_COLOR = (66, 74, 138)    # indigo ball body
HOLE_COLOR = (10, 12, 30)     # near-black finger holes


def sample_scene(seed: int) -> dict:
    """Deterministic scene dict from an integer seed."""
    rng = random.Random(seed)
    n = SUPPORTED_N[rng.randrange(len(SUPPORTED_N))]
    # start_angle, ball_radius, hole_radius and orbit vary independently of n so
    # that neither position nor object size predicts the count (cheap-cue
    # counterbalance).
    start_angle = rng.uniform(0.0, 360.0)
    ball_radius = BALL_RADIUS + rng.uniform(-6.0, 8.0)
    hole_radius = HOLE_RADIUS + rng.uniform(-3.0, 3.0)
    orbit = HOLE_ORBIT + rng.uniform(-5.0, 6.0)
    return {
        "n": n,
        "start_angle": round(start_angle, 6),
        "ball_radius": round(ball_radius, 3),
        "hole_radius": round(hole_radius, 3),
        "orbit": round(orbit, 3),
        "canvas": CANVAS,
    }


def _pol(angle_deg: float, radius: float) -> tuple[float, float]:
    rad = math.radians(angle_deg)
    return (CX + radius * math.cos(rad), CY + radius * math.sin(rad))


def _hole_pos(scene: dict, k: int) -> tuple[float, float]:
    n = int(scene["n"])
    step = 360.0 / n
    angle = float(scene["start_angle"]) + k * step
    return _pol(angle, float(scene["orbit"]))


def render(scene: dict) -> Image.Image:
    """Render the bowling ball to an RGB image."""
    img = Image.new("RGB", (int(scene["canvas"]), int(scene["canvas"])), BACKGROUND)
    draw = ImageDraw.Draw(img)
    ball_r = float(scene["ball_radius"])
    hole_r = float(scene["hole_radius"])
    draw.ellipse([CX - ball_r, CY - ball_r, CX + ball_r, CY + ball_r], fill=BALL_COLOR)
    n = int(scene["n"])
    for k in range(n):
        px, py = _hole_pos(scene, k)
        draw.ellipse([px - hole_r, py - hole_r, px + hole_r, py + hole_r], fill=HOLE_COLOR)
    return img


def analytic_gold(scene: dict) -> str:
    """Decision = the visible count of finger holes, as a string."""
    return str(scene["n"])


def margin(scene: dict) -> float:
    """Verifiability margin = minimum pairwise hole-centre distance (px)."""
    n = int(scene["n"])
    if n < 2:
        return float("inf")
    orbit = float(scene["orbit"])
    return 2.0 * orbit * math.sin(math.pi / n)


def is_quarantined(scene: dict) -> bool:
    """Quarantine crowded counts that are ambiguous at pixel scale."""
    if int(scene["n"]) not in SUPPORTED_N:
        return True
    return margin(scene) < QUAR_MIN_GAP_PX


def latent_symmetries(scene: dict):
    """not_applicable: empty, the latent rendering is injective."""
    return []
