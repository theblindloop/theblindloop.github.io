"""Deterministic latent sampler and raster renderer for Checkpoint Half Tally."""

from __future__ import annotations

import copy
import random
from typing import Iterable

from PIL import Image, ImageDraw

CANVAS_SIZE = 600
BACKGROUND = (248, 247, 242)
NAVY = (28, 43, 61)
ROUTE_HALO = (205, 216, 224)
BLUE = (38, 105, 201)
ORANGE = (232, 126, 38)
RED = (205, 55, 62)
GREEN = (34, 148, 92)


def _d4(point: tuple[int, int], code: int) -> tuple[int, int]:
    """Apply one of the eight square symmetries to an integer point."""

    x, y = point
    m = CANVAS_SIZE - 1
    transforms = (
        (x, y),
        (m - x, y),
        (x, m - y),
        (m - x, m - y),
        (y, x),
        (m - y, x),
        (y, m - x),
        (m - y, m - x),
    )
    return transforms[code]


def _station(points: list[tuple[int, int]], fraction: float) -> tuple[int, int]:
    lengths = [abs(b[0] - a[0]) + abs(b[1] - a[1]) for a, b in zip(points, points[1:])]
    target = fraction * sum(lengths)
    walked = 0.0
    for (x0, y0), (x1, y1), length in zip(points, points[1:], lengths):
        if target <= walked + length:
            offset = target - walked
            if x0 == x1:
                sign = 1 if y1 >= y0 else -1
                return x0, int(round(y0 + sign * offset))
            sign = 1 if x1 >= x0 else -1
            return int(round(x0 + sign * offset)), y0
        walked += length
    return points[-1]


def _offset_station(
    points: list[tuple[int, int]], fraction: float, side: int, offset: int = 22
) -> tuple[int, int]:
    """Place a ring beside, rather than overpainting, its route station."""

    station = _station(points, fraction)
    lengths = [abs(b[0] - a[0]) + abs(b[1] - a[1]) for a, b in zip(points, points[1:])]
    target = fraction * sum(lengths)
    walked = 0.0
    for (x0, y0), (x1, y1), length in zip(points, points[1:], lengths):
        if target <= walked + length:
            if x0 == x1:
                return station[0] + side * offset, station[1]
            return station[0], station[1] + side * offset
        walked += length
    return station


def _segment_station(
    start: tuple[int, int], end: tuple[int, int], fraction: float, side: int, offset: int = 22
) -> tuple[int, int]:
    import math

    x0, y0 = start
    x1, y1 = end
    x = int(round(x0 + fraction * (x1 - x0)))
    y = int(round(y0 + fraction * (y1 - y0)))
    length = math.hypot(x1 - x0, y1 - y0)
    if length <= 0:
        raise ValueError("route segment has zero length")
    nx = -(y1 - y0) / length
    ny = (x1 - x0) / length
    return int(round(x + side * offset * nx)), int(round(y + side * offset * ny))


def _color_word(seed: int, rng: random.Random) -> list[str]:
    """Return four red/four green labels with a paired, balanced winner."""

    # Adjacent even/odd seeds share geometry and unordered color totals while
    # taking opposite answers.  Magnitude 2 is the nearest non-tied discrete
    # case; magnitude 4 supplies an easier comparison without changing ink.
    red_in_winner = 4 if rng.randrange(3) == 0 else 3
    red_in_loser = 4 - red_in_winner
    winner_offsets = set(rng.sample(range(4), red_in_winner))
    loser_offsets = set(rng.sample(range(4), red_in_loser))
    winner = ["red" if i in winner_offsets else "green" for i in range(4)]
    loser = ["red" if i in loser_offsets else "green" for i in range(4)]
    return winner + loser if seed % 2 == 0 else loser + winner


def sample_scene(seed: int) -> dict:
    """Sample a counterbalanced, non-self-intersecting folded route."""

    pair_seed = seed - (seed % 2)
    rng = random.Random((pair_seed * 1_000_003) ^ 0x4A6F7921)

    # A sparse diagonal zigzag spans distant corners but stays inside a thin
    # corridor.  Local reversals make encounter order a route property while
    # avoiding crossings and the diffuse support of a dense serpentine.
    j = lambda radius: rng.randrange(-radius, radius + 1)
    base = [
        (64 + j(7), 529 + j(7)),
        (218 + j(8), 430 + j(8)),
        (143 + j(8), 326 + j(8)),
        (357 + j(8), 352 + j(8)),
        (278 + j(8), 198 + j(8)),
        (468 + j(8), 145 + j(8)),
        (536 + j(7), 62 + j(7)),
    ]
    # Segment-local stations stay well clear of every bend.  The list order is
    # the encounter order and intentionally places the 4/4 split inside the
    # long middle arm rather than at a geometry boundary.
    station_specs = (
        (0, 0.25, 1),
        (0, 0.72, -1),
        (1, 0.48, 1),
        (2, 0.48, -1),
        (3, 0.48, 1),
        (4, 0.50, -1),
        (5, 0.28, 1),
        (5, 0.74, -1),
    )
    stations = [
        _segment_station(base[segment], base[segment + 1], fraction, side)
        for segment, fraction, side in station_specs
    ]
    transform_code = rng.randrange(8)
    route = [_d4(point, transform_code) for point in base]
    checkpoints = [_d4(point, transform_code) for point in stations]
    colors = _color_word(seed, rng)

    return {
        "route_points": [list(point) for point in route],
        "start_center": list(route[0]),
        "finish_center": list(route[-1]),
        "checkpoint_centers": [list(point) for point in checkpoints],
        "checkpoint_colors": colors,
    }


def _ring(draw: ImageDraw.ImageDraw, center: Iterable[int], radius: int, color: tuple[int, int, int], width: int) -> None:
    x, y = (int(value) for value in center)
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), outline=color, width=width)


def render(scene: dict) -> Image.Image:
    """Render only public scene fields; no answer-specific annotation is drawn."""

    route_points = [tuple(map(int, point)) for point in scene["route_points"]]
    start_center = tuple(map(int, scene["start_center"]))
    finish_center = tuple(map(int, scene["finish_center"]))
    checkpoint_centers = [tuple(map(int, point)) for point in scene["checkpoint_centers"]]
    checkpoint_colors = list(scene["checkpoint_colors"])
    if len(checkpoint_centers) != 8 or len(checkpoint_colors) != 8:
        raise ValueError("exactly eight checkpoints are required")

    image = Image.new("RGB", (CANVAS_SIZE, CANVAS_SIZE), BACKGROUND)
    draw = ImageDraw.Draw(image)

    # Quiet drafting-board frame and corner registration marks are visual
    # context only; the inverse arm ignores their pale palette.
    draw.rounded_rectangle((22, 22, 577, 577), radius=18, outline=(226, 224, 216), width=2)
    for x, y in ((38, 38), (562, 38), (38, 562), (562, 562)):
        draw.line((x - 5, y, x + 5, y), fill=(218, 216, 208), width=1)
        draw.line((x, y - 5, x, y + 5), fill=(218, 216, 208), width=1)

    # Endpoint rings are laid down first.  The route then cuts a single radial
    # opening in each ring while remaining visibly and pixel-wise continuous.
    _ring(draw, start_center, 16, BLUE, 5)
    _ring(draw, finish_center, 16, ORANGE, 5)

    # Drawing vertex caps explicitly makes the raster independent of polyline
    # enumeration direction, which is exercised by latent_symmetries().
    draw.line(route_points, fill=ROUTE_HALO, width=15)
    for x, y in route_points:
        draw.ellipse((x - 7, y - 7, x + 7, y + 7), fill=ROUTE_HALO)
    draw.line(route_points, fill=NAVY, width=5)
    for x, y in route_points:
        draw.ellipse((x - 2, y - 2, x + 2, y + 2), fill=NAVY)

    palette = {"red": RED, "green": GREEN}
    for center, color_name in zip(checkpoint_centers, checkpoint_colors, strict=True):
        if color_name not in palette:
            raise ValueError(f"unsupported checkpoint color {color_name!r}")
        _ring(draw, center, 11, palette[color_name], 4)

    return image


def _red_counts(scene: dict) -> tuple[int, int]:
    colors = list(scene["checkpoint_colors"])
    if len(colors) != 8 or any(color not in {"red", "green"} for color in colors):
        raise ValueError("checkpoint_colors must be eight red/green labels")
    return colors[:4].count("red"), colors[4:].count("red")


def analytic_gold(scene: dict) -> str:
    first, last = _red_counts(scene)
    if first == last:
        raise ValueError("tied scene has no forced comparison answer")
    return "first" if first > last else "last"


def margin(scene: dict) -> float:
    first, last = _red_counts(scene)
    return float(abs(first - last))


def is_quarantined(scene: dict) -> bool:
    first, last = _red_counts(scene)
    return first == last


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["route_points"] = list(reversed(twin["route_points"]))
    return [("reverse_polyline_storage", twin)]
