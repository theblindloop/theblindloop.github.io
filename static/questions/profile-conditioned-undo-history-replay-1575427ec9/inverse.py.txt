"""Scene-free pixel inverse arm for undo_history_replay.

Receives only a rendered RGB image. It re-finds the drawn row of panels, reads
the leftmost panel as the pot's starting colour, classifies every card to its
right as a paint card or an undo card from the drawn glyph alone, replays the
declared undo semantics in drawn left-to-right order, and reports the final
colour and the number of surviving paint cards.

It imports no renderer, prompt, generator, verifier or analytic-gold code and
never receives latent scene data.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "undo_history_replay-oracle-0.1.0"

_PANEL_BG = (252, 246, 228)
_PALETTE = {
    "red": (214, 38, 38),
    "blue": (40, 84, 204),
    "green": (26, 150, 72),
    "amber": (232, 150, 20),
    "violet": (140, 58, 180),
}
_COUNT_WORDS = ("zero", "one", "two", "three", "four")
_SAT_BOUNDARY = 45.0


def _runs(flags: np.ndarray, min_len: int) -> list[tuple[int, int]]:
    out: list[tuple[int, int]] = []
    start = None
    for i, f in enumerate(flags):
        if f and start is None:
            start = i
        elif not f and start is not None:
            if i - start >= min_len:
                out.append((start, i - 1))
            start = None
    if start is not None and len(flags) - start >= min_len:
        out.append((start, len(flags) - 1))
    return out


def _read_glyph(arr: np.ndarray, r0: int, r1: int, c0: int, c1: int):
    """Return the mean RGB of the ink drawn inside one panel, or None."""
    sub = arr[r0:r1 + 1, c0:c1 + 1].astype(np.int64)
    bg = np.array(_PANEL_BG, dtype=np.int64)
    ink = np.abs(sub - bg).sum(axis=2) > 90
    if int(ink.sum()) < 60:
        return None
    return sub[ink].mean(axis=0)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    bg = np.array(_PANEL_BG, dtype=np.int64)
    cream = np.abs(arr.astype(np.int64) - bg).sum(axis=2) <= 24

    row_runs = _runs(cream.any(axis=1), 40)
    if not row_runs:
        raise ValueError("no drawn panel row found")
    r0, r1 = max(row_runs, key=lambda r: r[1] - r[0])
    band = cream[r0:r1 + 1]
    col_runs = _runs(band.any(axis=0), 40)
    if len(col_runs) < 2:
        raise ValueError("fewer than two drawn panels found")

    inset = 10
    glyphs = []
    for c0, c1 in col_runs:
        glyphs.append(_read_glyph(arr, r0 + inset, r1 - inset, c0 + inset, c1 - inset))

    if glyphs[0] is None:
        raise ValueError("the leftmost panel carries no drawn starting colour")

    def name_of(rgb) -> str:
        return min(_PALETTE, key=lambda n: float(np.abs(rgb - np.array(_PALETTE[n])).sum()))

    stack = [name_of(glyphs[0])]
    for g in glyphs[1:]:
        if g is None:
            raise ValueError("an empty card panel was drawn")
        if float(g.max() - g.min()) < _SAT_BOUNDARY:  # grey glyph: an undo card
            if len(stack) > 1:
                stack.pop()
        else:
            stack.append(name_of(g))

    survivors = min(len(stack) - 1, len(_COUNT_WORDS) - 1)
    return f"{stack[-1]}|{_COUNT_WORDS[survivors]}"
