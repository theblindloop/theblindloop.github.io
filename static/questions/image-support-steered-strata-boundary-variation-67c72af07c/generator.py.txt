"""Deterministic renderer and analytic arm for stratum boundary variation."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 640
HEIGHT = 420
STEP_COUNT = 16
STEP_WIDTH = 34
QUARANTINE_PX = 12.0
BACKGROUND = (247, 245, 239)
INK = (43, 48, 53)
GRID = (211, 211, 204)
PALETTE = (
    (17, 137, 128),
    (32, 119, 158),
    (49, 137, 91),
    (111, 83, 165),
    (181, 86, 53),
    (173, 113, 25),
)


def _font(size: int, *, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _profile(rng: random.Random) -> list[int]:
    """Draw a zero-mean, everywhere-changing 16-step offset profile."""

    for _ in range(20_000):
        values = [rng.randint(-14, 14) for _ in range(STEP_COUNT - 1)]
        final = -sum(values)
        if not -14 <= final <= 14:
            continue
        values.append(final)
        if min(abs(right - left) for left, right in zip(values, values[1:])) < 3:
            continue
        variation = sum(abs(right - left) for left, right in zip(values, values[1:]))
        if 120 <= variation <= 230:
            return values
    raise RuntimeError("could not sample a valid stepped profile")


def _profile_variation(values: list[int]) -> int:
    if len(values) != STEP_COUNT:
        raise ValueError("boundary profile must contain sixteen levels")
    return sum(abs(int(right) - int(left)) for left, right in zip(values, values[1:]))


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample cue-balanced pairs with opposite answers and identical total ink."""

    pair_seed = int(seed) // 2
    rng = random.Random(0x57A7_A021 + pair_seed * 1_000_003)
    near_boundary = pair_seed % 7 == 0
    low_gap, high_gap = ((8, 12) if near_boundary else (26, 44))
    for _ in range(10_000):
        rough = _profile(rng)
        smooth = _profile(rng)
        difference = _profile_variation(rough) - _profile_variation(smooth)
        if low_gap <= difference <= high_gap:
            break
        if -high_gap <= difference <= -low_gap:
            rough, smooth = smooth, rough
            break
    else:
        raise RuntimeError("could not sample separated boundary variations")

    top_center = 90 + 2 * (pair_seed % 3)
    bottom_center = 330 - 2 * (pair_seed % 3)
    if int(seed) % 2 == 0:
        top_offsets, bottom_offsets = rough, smooth
    else:
        top_offsets, bottom_offsets = smooth, rough
    return {
        "top_levels": [top_center + value for value in top_offsets],
        "bottom_levels": [bottom_center + value for value in bottom_offsets],
        "fill_rgb": list(PALETTE[pair_seed % len(PALETTE)]),
        "x0": 44 + 3 * (pair_seed % 5),
        "step_width": STEP_WIDTH,
        "canvas_width": WIDTH,
        "canvas_height": HEIGHT,
    }


def _variations(scene: dict[str, Any]) -> tuple[int, int]:
    top = [int(value) for value in scene["top_levels"]]
    bottom = [int(value) for value in scene["bottom_levels"]]
    return _profile_variation(top), _profile_variation(bottom)


def analytic_gold(scene: dict[str, Any]) -> str:
    top, bottom = _variations(scene)
    if top == bottom:
        raise ValueError("boundary variations tie")
    return "top" if top > bottom else "bottom"


def margin(scene: dict[str, Any]) -> float:
    top, bottom = _variations(scene)
    return float(abs(top - bottom))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) <= QUARANTINE_PX


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []


def render(scene: dict[str, Any]) -> Image.Image:
    top = [int(value) for value in scene["top_levels"]]
    bottom = [int(value) for value in scene["bottom_levels"]]
    fill = tuple(int(value) for value in scene["fill_rgb"])
    x0 = int(scene["x0"])
    step_width = int(scene["step_width"])
    width = int(scene["canvas_width"])
    height = int(scene["canvas_height"])
    if len(top) != STEP_COUNT or len(bottom) != STEP_COUNT:
        raise ValueError("renderer requires sixteen top and bottom levels")
    if step_width < 20 or x0 < 30 or x0 + STEP_COUNT * step_width > width - 30:
        raise ValueError("stepped stratum does not fit horizontally")
    if min(bottom[index] - top[index] for index in range(STEP_COUNT)) < height * 0.40:
        raise ValueError("stratum is not a thick connected area")
    if min(top) < 60 or max(bottom) > height - 60:
        raise ValueError("stratum does not fit vertically")
    if not all(0 <= value <= 255 for value in fill):
        raise ValueError("fill_rgb is outside RGB range")

    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    x1 = x0 + STEP_COUNT * step_width

    for guide_y in range(70, height - 49, 40):
        draw.line((x0 - 12, guide_y, x1 + 12, guide_y), fill=GRID, width=1)
    for index in range(STEP_COUNT + 1):
        tick_x = x0 + index * step_width
        draw.line((tick_x, 55, tick_x, 64), fill=GRID, width=1)
        draw.line((tick_x, height - 64, tick_x, height - 55), fill=GRID, width=1)

    for index, (top_y, bottom_y) in enumerate(zip(top, bottom, strict=True)):
        left = x0 + index * step_width
        right = left + step_width - 1
        draw.rectangle((left, top_y, right, bottom_y), fill=fill)

    title_font = _font(19, bold=True)
    label_font = _font(13, bold=True)
    small_font = _font(12)
    draw.text((x0, 15), "STRATUM EDGE VARIATION", font=title_font, fill=INK)
    draw.text((x0 + 315, 20), "equal-width steps", font=small_font, fill=INK)
    draw.text((9, 72), "TOP", font=label_font, fill=INK)
    draw.text((x1 - 63, height - 73), "BOTTOM", font=label_font, fill=INK)
    draw.text((x0, height - 31), "sum vertical jumps  •  horizontal runs = 0", font=small_font, fill=INK)
    return image
