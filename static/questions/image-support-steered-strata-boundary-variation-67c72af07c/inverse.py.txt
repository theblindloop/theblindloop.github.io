"""Independent final-raster inverse for stratum boundary variation.

The inverse segments the sole saturated filled area, validates its connected
column-convex span and equal-width step structure, recovers both contour level
sequences, and independently composes their total variations.  It uses no
renderer constants, latent data, source files, metadata, or fixed coordinates.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


def _filled_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    high = array.max(axis=2)
    low = array.min(axis=2)
    chroma = high - low
    return (chroma >= 62) & (high >= 105) & (low <= 185)


def _runs(top: np.ndarray, bottom: np.ndarray) -> tuple[list[int], list[int], list[int]]:
    top_levels: list[int] = []
    bottom_levels: list[int] = []
    widths: list[int] = []
    start = 0
    for index in range(1, len(top) + 1):
        changed = index == len(top) or top[index] != top[start] or bottom[index] != bottom[start]
        if changed:
            top_levels.append(int(top[start]))
            bottom_levels.append(int(bottom[start]))
            widths.append(index - start)
            start = index
    return top_levels, bottom_levels, widths


def _measure(image: Image.Image) -> tuple[int, int]:
    mask = _filled_mask(image)
    ys, xs = np.nonzero(mask)
    if len(xs) < image.width * image.height * 0.22:
        raise ValueError("filled stratum is missing or materially erased")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    width = x1 - x0 + 1
    height = y1 - y0 + 1
    if width < image.width * 0.76 or height < image.height * 0.50:
        raise ValueError("filled stratum has implausible extent")

    sub = mask[:, x0 : x1 + 1]
    if not bool(sub.any(axis=0).all()):
        raise ValueError("filled stratum has a severed horizontal span")
    top = np.argmax(sub, axis=0)
    bottom = sub.shape[0] - 1 - np.argmax(sub[::-1], axis=0)
    counts = sub.sum(axis=0)
    if int(np.max((bottom - top + 1) - counts)) != 0:
        raise ValueError("filled stratum contains an internal erasure")

    top_levels, bottom_levels, widths = _runs(top, bottom)
    if not 12 <= len(widths) <= 20:
        raise ValueError("boundary does not expose the expected repeated step structure")
    median_width = int(np.median(np.asarray(widths)))
    if median_width < image.width * 0.035 or max(abs(value - median_width) for value in widths) > 2:
        raise ValueError("step intervals are not equal-width after raster recovery")
    if min(abs(right - left) for left, right in zip(top_levels, top_levels[1:])) < 2:
        raise ValueError("top boundary has an unresolved jump")
    if min(abs(right - left) for left, right in zip(bottom_levels, bottom_levels[1:])) < 2:
        raise ValueError("bottom boundary has an unresolved jump")

    top_total = sum(abs(right - left) for left, right in zip(top_levels, top_levels[1:]))
    bottom_total = sum(
        abs(right - left) for left, right in zip(bottom_levels, bottom_levels[1:])
    )
    if top_total == bottom_total:
        raise ValueError("boundary totals tie at finite resolution")
    return top_total, bottom_total


def decision_from_image(image: Image.Image) -> str:
    top, bottom = _measure(image)
    return "top" if top > bottom else "bottom"
