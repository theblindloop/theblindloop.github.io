"""Deterministic forest-plot renderer and analytic arm.

The analytic arm operates on scene geometry.  It does not call the independent
image oracle, which separately reconstructs the same decision from RGB pixels.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

COLORS: dict[str, tuple[int, int, int]] = {
    "red": (210, 52, 58),
    "blue": (42, 101, 176),
    "green": (0, 139, 105),
    "orange": (230, 126, 34),
}
COLOR_NAMES: tuple[str, ...] = tuple(COLORS)
REFERENCE = (92, 96, 103)
INK = (35, 43, 55)
LIGHT = (219, 224, 230)


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size=size)
    except OSError:
        return ImageFont.load_default()


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one balanced, well-separated four-interval scene."""

    rng = random.Random(int(seed))
    width, height = 640, 480
    reference_x = 300 + rng.randint(-18, 18)
    color_order = list(COLOR_NAMES)
    winner = color_order[int(seed) % len(color_order)]

    widths = [96, 120, 146, 172]
    rng.shuffle(widths)  # interval width does not reveal the answer role
    row_order = list(color_order)
    rng.shuffle(row_order)  # row position does not reveal the answer color

    # Four deterministic boundary bands make the submitted gallery include
    # easy and near-boundary examples while remaining outside quarantine.
    winner_gap = (12, 19, 31, 45)[(int(seed) // 4) % 4]
    other_colors = [name for name in color_order if name != winner]
    rng.shuffle(other_colors)
    intervals: dict[str, list[int]] = {}

    role_by_color = {
        winner: "excluded_right",
        other_colors[0]: "crossing_right_center",
        other_colors[1]: "crossing",
        other_colors[2]: "left_only",
    }
    width_by_color = dict(zip(color_order, widths))
    for color in color_order:
        span = width_by_color[color]
        role = role_by_color[color]
        if role == "excluded_right":
            left = reference_x + winner_gap
            right = left + span
        elif role == "crossing_right_center":
            # Its midpoint can lie farther right than the winner's midpoint,
            # defeating a center-only shortcut while its left cap still crosses.
            left = reference_x - (12 + rng.randint(0, 9))
            right = left + span
        elif role == "crossing":
            left = reference_x - (30 + rng.randint(0, 15))
            right = left + span
        else:
            right = reference_x - (14 + rng.randint(0, 18))
            left = right - span
        intervals[color] = [int(left), int(right)]

    return {
        "canvas_width": width,
        "canvas_height": height,
        "background": [248, 250, 252],
        "reference_x": int(reference_x),
        "color_order": color_order,
        "row_order": row_order,
        "intervals": intervals,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render a compact scientific forest plot from the complete scene."""

    width = int(scene["canvas_width"])
    height = int(scene["canvas_height"])
    image = Image.new("RGB", (width, height), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    reference_x = int(scene["reference_x"])
    intervals = scene["intervals"]
    color_order = list(scene["color_order"])
    row_order = list(scene["row_order"])

    draw.text((38, 22), "UNCERTAINTY INTERVALS", fill=INK, font=_font(22, True))
    draw.text(
        (39, 49),
        "Complete exclusion requires the full interval to clear the reference",
        fill=(85, 94, 106),
        font=_font(13),
    )
    plot_left, plot_right, plot_top, plot_bottom = 76, width - 38, 92, height - 58
    draw.rounded_rectangle(
        (plot_left, plot_top, plot_right, plot_bottom),
        radius=8,
        fill=(255, 255, 255),
        outline=LIGHT,
        width=2,
    )

    row_y = [136, 211, 286, 361]
    for idx, y in enumerate(row_y):
        draw.line((plot_left + 14, y, plot_right - 14, y), fill=(236, 239, 243), width=1)
        draw.text((plot_left + 12, y - 28), f"S{idx + 1}", fill=(116, 125, 137), font=_font(12, True))

    for y0 in range(plot_top + 12, plot_bottom - 5, 16):
        draw.rectangle((reference_x - 2, y0, reference_x + 2, min(y0 + 8, plot_bottom - 7)), fill=REFERENCE)
    draw.text((reference_x - 38, plot_bottom + 12), "REFERENCE", fill=REFERENCE, font=_font(11, True))

    # Access color_order explicitly as part of the rendered label binding.
    if set(color_order) != set(COLORS):
        raise ValueError("scene color_order is invalid")
    for y, color in zip(row_y, row_order):
        left, right = (int(value) for value in intervals[color])
        rgb = COLORS[color]
        draw.line((left, y, right, y), fill=rgb, width=7)
        draw.line((left, y - 12, left, y + 12), fill=rgb, width=5)
        draw.line((right, y - 12, right, y + 12), fill=rgb, width=5)
        center = (left + right) // 2
        draw.ellipse((center - 8, y - 8, center + 8, y + 8), fill=(255, 255, 255), outline=rgb, width=4)

    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    intervals = scene["intervals"]
    color_order = scene["color_order"]
    reference_x = float(scene["reference_x"])
    winners = [color for color in color_order if float(intervals[color][0]) > reference_x]
    if len(winners) != 1:
        raise ValueError(f"expected exactly one fully-right interval, found {winners}")
    return str(winners[0])


def _scores(scene: dict[str, Any]) -> dict[str, float]:
    intervals = scene["intervals"]
    reference_x = float(scene["reference_x"])
    return {color: float(values[0]) - reference_x for color, values in intervals.items()}


def margin(scene: dict[str, Any]) -> float:
    """Minimum endpoint clearance from the visible decision boundary, in px."""

    return min(abs(value) for value in _scores(scene).values())


def runner_up_margin(scene: dict[str, Any]) -> float:
    ranked = sorted(_scores(scene).values(), reverse=True)
    return ranked[0] - ranked[1]


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 10.0 or runner_up_margin(scene) < 20.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
