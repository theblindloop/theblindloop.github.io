"""Independent PNG-only inverse arm for forest interval exclusion.

This module deliberately duplicates only the public color vocabulary.  It has
no access to sampling, renderer geometry, analytic gold, manifests, or paths.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (210, 52, 58),
    "blue": (42, 101, 176),
    "green": (0, 139, 105),
    "orange": (230, 126, 34),
}


def decision_from_image(image: Image.Image) -> str:
    """Recover the unique interval with a left cap right of the dashed line."""

    rgb = image.convert("RGB")
    width, height = rgb.size
    if width < 200 or height < 160:
        raise ValueError("image is too small for interval measurement")
    pixels = np.asarray(rgb, dtype=np.int16)

    # A neutral-gray vertical dashed reference produces the strongest column
    # support. Chromatic intervals and near-white grid lines are excluded.
    channel_spread = np.max(pixels, axis=2) - np.min(pixels, axis=2)
    neutral = (
        (pixels[:, :, 0] >= 65)
        & (pixels[:, :, 0] <= 125)
        & (channel_spread <= 14)
    )
    gray_counts = np.sum(neutral, axis=0)
    peak = int(np.max(gray_counts, initial=0))
    if peak < max(35, height // 10):
        raise ValueError("visible dashed reference line was not recovered")
    peak_columns = np.flatnonzero(gray_counts >= peak - 3)
    if peak_columns.size == 0 or int(peak_columns[-1] - peak_columns[0]) > 10:
        raise ValueError("reference-line fit is ambiguous")
    reference_x = float(np.mean(peak_columns))

    left_edges: dict[str, int] = {}
    for name, target in PALETTE.items():
        target_array = np.asarray(target, dtype=np.int16)
        mask = np.max(np.abs(pixels - target_array), axis=2) <= 8
        ys, xs = np.nonzero(mask)
        if xs.size < 250:
            raise ValueError(f"{name} interval is missing or corrupted")
        if int(np.max(xs) - np.min(xs)) < 55 or int(np.max(ys) - np.min(ys)) < 18:
            raise ValueError(f"{name} mark is not an interval with endpoint caps")
        left_edges[name] = int(np.min(xs))

    scores = {name: left - reference_x for name, left in left_edges.items()}
    winners = [name for name, score in scores.items() if score > 5.0]
    if len(winners) != 1:
        raise ValueError(f"expected one fully-right interval, recovered {winners}")
    ranked = sorted(scores.values(), reverse=True)
    if min(abs(value) for value in scores.values()) < 5.0 or ranked[0] - ranked[1] < 12.0:
        raise ValueError("endpoint decision is raster-ambiguous")
    return winners[0]
