"""Renderer and analytic arm for balanced-symbol substitution correspondence."""

from __future__ import annotations

import copy
import itertools
import json
import math
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 700
BACKGROUND = (250, 249, 245)
INK = (25, 29, 35)
NAVY = (42, 68, 102)
MAGENTA = (194, 45, 133)
COLOR_RGB = {
    "red": (210, 65, 62),
    "green": (45, 151, 91),
    "blue": (55, 106, 190),
}
ANSWER_COLORS = ("red", "green", "blue")
FONT_PATH = Path("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf")


def _font(size: int) -> ImageFont.ImageFont:
    try:
        return ImageFont.truetype(str(FONT_PATH), size)
    except OSError:
        return ImageFont.load_default()


def _primitive_weights(equations: list[dict]) -> tuple[int, int, int]:
    """Recover the unique positive primitive integer ratio implied by balances."""
    solutions: set[tuple[int, int, int]] = set()
    for weights in itertools.product(range(1, 7), repeat=3):
        if all(
            sum(a * w for a, w in zip(row["left"], weights))
            == sum(a * w for a, w in zip(row["right"], weights))
            for row in equations
        ):
            divisor = math.gcd(math.gcd(weights[0], weights[1]), weights[2])
            solutions.add(tuple(value // divisor for value in weights))
    if len(solutions) != 1:
        raise ValueError(f"balance system does not determine one positive ratio: {solutions}")
    return next(iter(solutions))


def _analysis(scene: dict) -> tuple[str, float, bool]:
    equations = scene["equations"]
    reference = scene["reference"]
    candidates = scene["candidates"]
    candidate_colors = scene["candidate_colors"]
    weights = _primitive_weights(equations)
    target = sum(count * weight for count, weight in zip(reference, weights))
    totals = [sum(count * weight for count, weight in zip(group, weights)) for group in candidates]
    matches = [index for index, total in enumerate(totals) if total == target]
    gaps = [abs(total - target) for total in totals if total != target]
    quarantined = len(matches) != 1 or not gaps or min(gaps) < 1
    if len(matches) != 1:
        raise ValueError(f"expected one equivalent tray, found {matches}")
    return candidate_colors[matches[0]], float(min(gaps)), quarantined


def sample_scene(seed: int) -> dict:
    """Create a deterministic, balanced scene with no answer-position shortcut."""
    rng = random.Random(int(seed))
    glyphs = [0, 1, 2]
    rng.shuffle(glyphs)
    light, medium, heavy = glyphs

    def counts(*items: int) -> list[int]:
        result = [0, 0, 0]
        for item in items:
            result[item] += 1
        return result

    equations = [
        {"left": counts(light, light), "right": counts(medium)},
        {"left": counts(light, medium), "right": counts(heavy)},
    ]
    for row in equations:
        if rng.randrange(2):
            row["left"], row["right"] = row["right"], row["left"]
    rng.shuffle(equations)

    triple_medium = counts(medium, medium, medium)
    mixed = counts(light, medium, heavy)
    if int(seed) % 2:
        reference, equivalent = mixed, triple_medium
    else:
        reference, equivalent = triple_medium, mixed

    weights = [0, 0, 0]
    weights[light], weights[medium], weights[heavy] = 1, 2, 3
    all_groups = [list(group) for group in itertools.product(range(4), repeat=3) if sum(group) == 3]
    distractors = [
        group
        for group in all_groups
        if sum(a * b for a, b in zip(group, weights)) != 6
    ]
    near = [group for group in distractors if abs(sum(a * b for a, b in zip(group, weights)) - 6) == 1]
    far = [group for group in distractors if abs(sum(a * b for a, b in zip(group, weights)) - 6) >= 2]
    rng.shuffle(near)
    rng.shuffle(far)
    chosen_distractors = [near[0], far[0]]

    correct_position = (int(seed) // 3) % 3
    candidates: list[list[int]] = []
    distractor_iter = iter(chosen_distractors)
    for position in range(3):
        candidates.append(list(equivalent) if position == correct_position else list(next(distractor_iter)))

    answer_color = ANSWER_COLORS[int(seed) % 3]
    other_colors = [color for color in ANSWER_COLORS if color != answer_color]
    if (int(seed) // 9) % 2:
        other_colors.reverse()
    candidate_colors: list[str] = []
    other_iter = iter(other_colors)
    for position in range(3):
        candidate_colors.append(answer_color if position == correct_position else next(other_iter))

    return {
        "equations": equations,
        "reference": reference,
        "candidates": candidates,
        "candidate_colors": candidate_colors,
    }


def _draw_glyph(draw: ImageDraw.ImageDraw, glyph: int, cx: int, cy: int) -> None:
    if glyph == 0:  # circle
        draw.ellipse((cx - 22, cy - 22, cx + 22, cy + 22), fill=INK)
    elif glyph == 1:  # square
        draw.rectangle((cx - 20, cy - 20, cx + 20, cy + 20), fill=INK)
    elif glyph == 2:  # triangle
        draw.polygon(((cx, cy - 24), (cx - 24, cy + 21), (cx + 24, cy + 21)), fill=INK)
    else:
        raise ValueError(f"unknown glyph {glyph}")


def _draw_group(
    draw: ImageDraw.ImageDraw,
    box: tuple[int, int, int, int],
    group: list[int],
    border: tuple[int, int, int],
    width: int = 5,
) -> None:
    x0, y0, x1, y1 = box
    draw.rounded_rectangle(box, radius=12, fill=(255, 255, 255), outline=border, width=width)
    glyphs = [glyph for glyph, count in enumerate(group) for _ in range(count)]
    if not glyphs:
        return
    spacing = 62
    start = (x0 + x1) // 2 - spacing * (len(glyphs) - 1) // 2
    cy = (y0 + y1) // 2
    for index, glyph in enumerate(glyphs):
        _draw_glyph(draw, glyph, start + spacing * index, cy)


def _equation_sort_key(row: dict) -> str:
    return json.dumps(row, sort_keys=True, separators=(",", ":"))


def render(scene: dict) -> Image.Image:
    equations = sorted(scene["equations"], key=_equation_sort_key)
    reference = scene["reference"]
    candidates = scene["candidates"]
    candidate_colors = scene["candidate_colors"]

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(25)
    label_font = _font(18)
    draw.text((WIDTH // 2, 25), "BALANCED SYMBOL RULES", anchor="mm", fill=NAVY, font=title_font)

    row_y = (64, 194)
    for row, y0 in zip(equations, row_y):
        left_box = (90, y0, 390, y0 + 104)
        right_box = (570, y0, 870, y0 + 104)
        _draw_group(draw, left_box, row["left"], NAVY)
        _draw_group(draw, right_box, row["right"], NAVY)
        mid_y = y0 + 52
        draw.rounded_rectangle((449, mid_y - 13, 511, mid_y - 5), radius=4, fill=NAVY)
        draw.rounded_rectangle((449, mid_y + 5, 511, mid_y + 13), radius=4, fill=NAVY)

    draw.line((38, 330, 922, 330), fill=(188, 192, 197), width=2)
    draw.text((130, 363), "REFERENCE", anchor="mm", fill=MAGENTA, font=label_font)
    draw.text((610, 363), "CANDIDATES", anchor="mm", fill=NAVY, font=label_font)

    ref_box = (22, 390, 238, 555)
    _draw_group(draw, ref_box, reference, MAGENTA, width=7)
    candidate_boxes = [(260, 390, 476, 555), (499, 390, 715, 555), (738, 390, 954, 555)]
    for group, color, box in zip(candidates, candidate_colors, candidate_boxes):
        _draw_group(draw, box, group, COLOR_RGB[color], width=7)

    draw.text(
        (WIDTH // 2, 603),
        "Use both balance rows.  Equal groups may look different.",
        anchor="mm",
        fill=(78, 83, 91),
        font=_font(19),
    )
    draw.text(
        (WIDTH // 2, 647),
        "Which colored tray equals the magenta tray?",
        anchor="mm",
        fill=INK,
        font=_font(23),
    )
    return image


def analytic_gold(scene: dict) -> str:
    return _analysis(scene)[0]


def margin(scene: dict) -> float:
    return _analysis(scene)[1]


def is_quarantined(scene: dict) -> bool:
    return _analysis(scene)[2]


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["equations"] = list(reversed(twin["equations"]))
    return [("reverse_equation_storage", twin)]
