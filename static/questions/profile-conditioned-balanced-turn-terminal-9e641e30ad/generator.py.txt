"""Deterministic visual world and independent analytic arm."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 720, 300
BG = "#f4f5f2"
INK = "#30343a"
CYAN = "#00a0b4"
GREEN = "#20a464"
COLORS = {"orange": "#ee8b2c", "purple": "#9467d3", "blue": "#3979cf"}

# Each polyline has segment lengths 60, 50, 40, 60, 30 and four bends.
# Only `balanced` has two turns of each handedness.
_BASE = {
    "balanced": [[-80, 40], [-20, 40], [-20, -10], [20, -10], [20, -70], [50, -70]],
    "left_heavy": [[-80, 40], [-20, 40], [-20, -10], [-60, -10], [-60, -70], [-90, -70]],
    "right_heavy": [[-80, -40], [-20, -40], [-20, 10], [-60, 10], [-60, 70], [-90, 70]],
}


def _transform(point: list[int], code: int) -> tuple[int, int]:
    x, y = point
    if code >= 4:
        x = -x
    for _ in range(code % 4):
        x, y = -y, x
    return x, y


def _scene_routes(scene: dict) -> list[dict]:
    routes = []
    for slot, (kind, terminal, transform, shift) in enumerate(
        zip(scene["route_kinds"], scene["terminal_colors"], scene["transforms"], scene["shifts"])
    ):
        cx, cy = 120 + 240 * slot + int(shift[0]), 150 + int(shift[1])
        points = []
        for point in _BASE[kind]:
            x, y = _transform(point, int(transform))
            points.append([cx + x, cy + y])
        routes.append({"kind": kind, "terminal": terminal, "points": points})
    return routes


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    answer_color = ("orange", "purple", "blue")[int(seed) % 3]
    other_colors = [color for color in ("orange", "purple", "blue") if color != answer_color]
    rng.shuffle(other_colors)
    color_for_kind = {
        "balanced": answer_color,
        "left_heavy": other_colors[0],
        "right_heavy": other_colors[1],
    }
    route_kinds = ["balanced", "left_heavy", "right_heavy"]
    rng.shuffle(route_kinds)
    return {
        "route_kinds": route_kinds,
        "terminal_colors": [color_for_kind[kind] for kind in route_kinds],
        "transforms": [rng.randrange(8) for _ in route_kinds],
        "shifts": [[rng.randint(-6, 6), rng.randint(-6, 6)] for _ in route_kinds],
    }


def _turn_counts(points: list[list[int]]) -> tuple[int, int]:
    directions = []
    for a, b in zip(points, points[1:]):
        dx, dy = b[0] - a[0], b[1] - a[1]
        if (dx == 0) == (dy == 0):
            raise ValueError("every segment must be horizontal or vertical")
        directions.append((0 if dx == 0 else (1 if dx > 0 else -1), 0 if dy == 0 else (1 if dy > 0 else -1)))
    signs = []
    for first, second in zip(directions, directions[1:]):
        cross = first[0] * second[1] - first[1] * second[0]
        if cross == 0:
            raise ValueError("every internal vertex must be a 90-degree bend")
        signs.append(1 if cross > 0 else -1)
    return signs.count(1), signs.count(-1)


def analytic_gold(scene: dict) -> str:
    scores = []
    for route in _scene_routes(scene):
        positive, negative = _turn_counts(route["points"])
        scores.append((abs(positive - negative), route["terminal"]))
    winners = [terminal for score, terminal in scores if score == min(value for value, _ in scores)]
    if len(winners) != 1:
        raise ValueError("scene has no unique turn-balanced cable")
    return winners[0]


def margin(scene: dict) -> float:
    scores = []
    for route in _scene_routes(scene):
        positive, negative = _turn_counts(route["points"])
        scores.append(abs(positive - negative))
    scores.sort()
    return float(scores[1] - scores[0])


def _segments(points: list[list[int]]) -> list[tuple[tuple[int, int], tuple[int, int]]]:
    return [(tuple(a), tuple(b)) for a, b in zip(points, points[1:])]


def _self_intersects(points: list[list[int]]) -> bool:
    segments = _segments(points)
    for i, (a, b) in enumerate(segments):
        for j, (c, d) in enumerate(segments):
            if j <= i + 1:
                continue
            if a[1] == b[1] and c[0] == d[0]:
                if min(a[0], b[0]) <= c[0] <= max(a[0], b[0]) and min(c[1], d[1]) <= a[1] <= max(c[1], d[1]):
                    return True
            if a[0] == b[0] and c[1] == d[1]:
                if min(c[0], d[0]) <= a[0] <= max(c[0], d[0]) and min(a[1], b[1]) <= c[1] <= max(a[1], b[1]):
                    return True
    return False


def is_quarantined(scene: dict) -> bool:
    try:
        routes = _scene_routes(scene)
        scores = []
        for route in routes:
            points = route["points"]
            if len(points) != 6 or _self_intersects(points):
                return True
            counts = _turn_counts(points)
            scores.append(abs(counts[0] - counts[1]))
        return sorted(scores) != [0, 2, 2] or len({route["terminal"] for route in routes}) != 3
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    for slot in range(3):
        x0 = 12 + 240 * slot
        draw.rounded_rectangle((x0, 18, x0 + 216, 282), radius=18, fill="#ffffff", outline="#d6d9dc", width=2)
    for route in _scene_routes(scene):
        points = [tuple(point) for point in route["points"]]
        draw.line(points, fill=INK, width=9, joint="curve")
        sx, sy = points[0]
        ex, ey = points[-1]
        draw.ellipse((sx - 13, sy - 13, sx + 13, sy + 13), fill="#ffffff", outline=GREEN, width=6)
        draw.rounded_rectangle((ex - 13, ey - 13, ex + 13, ey + 13), radius=4, fill=COLORS[route["terminal"]], outline="#ffffff", width=3)
        # Public, pixel-exact centerline makes ordered tracing independently recoverable.
        draw.line(points, fill=CYAN, width=1)
    return image
