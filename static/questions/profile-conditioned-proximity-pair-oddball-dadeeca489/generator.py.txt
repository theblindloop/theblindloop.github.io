"""Pair-proximity oddball quadrant search world.

The image is a 2x2 grid of quadrants split by a light divider. Each quadrant
holds exactly two dark filled dots. Exactly one of the four quadrants has its
two dots very close together (a small intra-pair centre-to-centre distance);
the other three quadrants each have their two dots clearly far apart.

The decision is the quadrant whose two dots are closest together. Every
attribute that decides the answer (the four intra-pair distances) is drawn in
the raster; nothing else in the scene is the answer.
"""

from __future__ import annotations

import math

from PIL import Image, ImageDraw

SIZE = 480
BG = (255, 255, 255)
DOT = (25, 25, 25)
DIV = (205, 205, 205)
R = 12
MG = SIZE // 2  # 240, split line

SECTORS = ("TL", "TR", "BL", "BR")
SECTOR_CENTERS = {
    "TL": (SIZE // 4, SIZE // 4),        # (120, 120)
    "TR": (3 * SIZE // 4, SIZE // 4),    # (360, 120)
    "BL": (SIZE // 4, 3 * SIZE // 4),    # (120, 360)
    "BR": (3 * SIZE // 4, 3 * SIZE // 4),# (360, 360)
}

MIN_CLOSE = 48.0   # winner intra-pair distance band
MAX_CLOSE = 88.0
MIN_FAR = 120.0    # distractor intra-pair distance band
MAX_FAR = 176.0
_QUARANTINE_THRESH = 40.0


def sample_scene(seed: int) -> dict:
    import numpy as np

    rng = np.random.default_rng(seed)
    winner = SECTORS[int(rng.integers(0, len(SECTORS)))]
    dists = {}
    for sector in SECTORS:
        if sector == winner:
            dists[sector] = float(rng.uniform(MIN_CLOSE, MAX_CLOSE))
        else:
            dists[sector] = float(rng.uniform(MIN_FAR, MAX_FAR))
    return {"dists": dists}


def _draw_pair(
    draw: ImageDraw.ImageDraw,
    cx: float,
    cy: float,
    d: float,
    angle: float,
) -> None:
    ux = math.cos(angle)
    uy = math.sin(angle)
    for sign in (-1, 1):
        x = cx + sign * (d / 2.0) * ux
        y = cy + sign * (d / 2.0) * uy
        draw.ellipse(
            [x - R, y - R, x + R, y + R],
            fill=DOT,
        )


def render(scene: dict) -> Image.Image:
    import numpy as np

    img = Image.new("RGB", (SIZE, SIZE), BG)
    draw = ImageDraw.Draw(img)
    draw.line([MG, 0, MG, SIZE], fill=DIV, width=2)
    draw.line([0, MG, SIZE, MG], fill=DIV, width=2)
    rng = np.random.default_rng(0)
    for sector in SECTORS:
        cx, cy = SECTOR_CENTERS[sector]
        angle = float(rng.uniform(0.0, math.pi))
        _draw_pair(draw, cx, cy, scene["dists"][sector], angle)
    return img


def _winner(scene: dict) -> str:
    return min(scene["dists"], key=lambda s: scene["dists"][s])


def analytic_gold(scene: dict) -> str:
    return _winner(scene)


def margin(scene: dict) -> float:
    ordered = sorted(scene["dists"].values())
    if len(ordered) < 2:
        return float(MAX_FAR)
    return float(ordered[1] - ordered[0])


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < _QUARANTINE_THRESH)


def latent_symmetries(scene: dict) -> list:
    return []
