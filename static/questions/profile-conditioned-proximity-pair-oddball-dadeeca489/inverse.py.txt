"""Independent pixel-only inverse arm for proximity_pair_oddball.

Receives only the final RGB image. It separates each of the four quadrant
cells, thresholds the dark dot ink, and labels the connected dark components in
every cell. Each quadrant holds exactly two dark dots, whose centers are the
component centroids. It measures each quadrant's intra-pair centre-to-centre
distance and selects the quadrant with the smallest such distance (the closest
pair). Imports nothing from renderer/prompts/generate/verify.
"""

from __future__ import annotations

from collections import deque

import numpy as np

SIZE = 480
MG = SIZE // 2  # 240
DARK_SUM = 300  # dot ink (25,25,25) sums to 75; divider (205,205,205) to 615

SECTORS = ("TL", "TR", "BL", "BR")
_CELL_X = {"TL": (0, MG), "TR": (MG, SIZE), "BL": (0, MG), "BR": (MG, SIZE)}
_CELL_Y = {"TL": (0, MG), "TR": (0, MG), "BL": (MG, SIZE), "BR": (MG, SIZE)}


def _cell_centroids(dark: np.ndarray, y0: int, y1: int, x0: int, x1: int):
    cell = dark[y0:y1, x0:x1]
    h, w = cell.shape
    label = np.zeros((h, w), dtype=np.int32)
    current = 0
    centroids = {}
    for i in range(h):
        for j in range(w):
            if not cell[i, j] or label[i, j]:
                continue
            current += 1
            q = deque([(i, j)])
            label[i, j] = current
            sx = sy = 0
            count = 0
            while q:
                ci, cj = q.popleft()
                sx += cj
                sy += ci
                count += 1
                for ni, nj in ((ci - 1, cj), (ci + 1, cj), (ci, cj - 1), (ci, cj + 1)):
                    if 0 <= ni < h and 0 <= nj < w and cell[ni, nj] and not label[ni, nj]:
                        label[ni, nj] = current
                        q.append((ni, nj))
            centroids[current] = (sx / count, sy / count)
    values = list(centroids.values())
    if len(values) >= 2:
        return values
    return None


def _pair_distance(dark: np.ndarray, y0: int, y1: int, x0: int, x1: int) -> float:
    centers = _cell_centroids(dark, y0, y1, x0, x1)
    if centers is None:
        return float("inf")
    (x1c, y1c), (x2c, y2c) = centers[0], centers[1]
    return float(((x1c - x2c) ** 2 + (y1c - y2c) ** 2) ** 0.5)


def decision_from_image(image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    dark = arr.sum(axis=2) < DARK_SUM
    best_sector = SECTORS[0]
    best_dist = float("inf")
    for sector in SECTORS:
        x0, x1 = _CELL_X[sector]
        y0, y1 = _CELL_Y[sector]
        dist = _pair_distance(dark, y0, y1, x0, x1)
        if dist < best_dist:
            best_dist = dist
            best_sector = sector
    return best_sector
