"""Renderer and analytic specification for a color-keyed residual board.

Each of three panels contains one bar for each of four repeated colors.  All
panels use the same four distinct integer heights, but the candidate panels
permute which color receives each height.  The decision compares the sum of
absolute height residuals from REFERENCE after matching by color.
"""

from __future__ import annotations

import itertools
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

RENDERER_VERSION = "color_residual_compare-renderer-0.1.0"

WIDTH = 1260
HEIGHT = 820
PANEL_X = (35, 435, 835)
PANEL_WIDTH = 390
PANEL_TOP = 145
PANEL_BOTTOM = 700
BASELINE = 620
LEVEL_PITCH = 48
BAR_WIDTH = 42
BAR_OFFSETS = (91, 161, 231, 301)
LEVEL_CHOICES = tuple(range(1, 7))
COLOR_NAMES = ("red", "blue", "green", "violet")
COLOR_SHORT = {"red": "R", "blue": "B", "green": "G", "violet": "V"}
BAR_COLORS = {
    "red": (210, 70, 67),
    "blue": (52, 101, 184),
    "green": (47, 145, 91),
    "violet": (145, 78, 174),
}

BACKGROUND = (249, 249, 246)
PANEL_FILL = (255, 255, 255)
INK = (38, 49, 58)
SUBTLE = (139, 151, 157)
GUIDE = (220, 226, 226)


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _centered_text(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font, fill) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2), text, font=font, fill=fill)


def _residual(reference: Iterable[int], candidate: Iterable[int]) -> int:
    return sum(abs(int(value) - int(target)) for value, target in zip(candidate, reference, strict=True))


def _validate_scene(scene: dict) -> tuple[list[int], list[int], list[int]]:
    reference = [int(value) for value in scene["reference_values"]]
    candidate_a = [int(value) for value in scene["candidate_a_values"]]
    candidate_b = [int(value) for value in scene["candidate_b_values"]]
    if any(len(values) != len(COLOR_NAMES) for values in (reference, candidate_a, candidate_b)):
        raise ValueError("the board needs four values in each panel")
    if any(value not in LEVEL_CHOICES for value in reference + candidate_a + candidate_b):
        raise ValueError("bar heights must be integer ruler levels from 1 through 6")
    if any(len(set(values)) != len(COLOR_NAMES) for values in (reference, candidate_a, candidate_b)):
        raise ValueError("each panel needs four distinct visible levels")
    if sorted(reference) != sorted(candidate_a) or sorted(reference) != sorted(candidate_b):
        raise ValueError("all panels must use one shared height multiset")
    return reference, candidate_a, candidate_b


def _pairs_for_target(reference: tuple[int, ...], target: str, max_gap: int = 8) -> list[tuple[tuple[int, ...], tuple[int, ...]]]:
    permutations = sorted(set(itertools.permutations(reference)))
    pairs: list[tuple[tuple[int, ...], tuple[int, ...]]] = []
    for candidate_a in permutations:
        error_a = _residual(reference, candidate_a)
        for candidate_b in permutations:
            delta = error_a - _residual(reference, candidate_b)
            if target == "A" and 2 <= delta <= max_gap:
                pairs.append((candidate_a, candidate_b))
            elif target == "B" and -max_gap <= delta <= -2:
                pairs.append((candidate_a, candidate_b))
            elif target == "tie" and delta == 0 and candidate_a != candidate_b:
                pairs.append((candidate_a, candidate_b))
    return pairs


def sample_scene(seed: int) -> dict:
    """Sample one deterministic matched-multiset panel scene.

    Even non-boundary seeds target A and odd non-boundary seeds target B.  A
    sparse residue class supplies visibly declared equality boundaries for the
    quarantined tie class.
    """

    rng = random.Random(int(seed))
    levels = tuple(sorted(rng.sample(list(LEVEL_CHOICES), len(COLOR_NAMES))))
    reference = tuple(rng.choice(sorted(set(itertools.permutations(levels)))))
    if int(seed) % 19 == 0:
        target = "tie"
    else:
        target = "A" if int(seed) % 2 == 0 else "B"
    pairs = _pairs_for_target(reference, target)
    if not pairs:
        raise RuntimeError(f"no {target} residual pair exists for levels {levels}")
    candidate_a, candidate_b = rng.choice(pairs)
    return {
        "reference_values": list(reference),
        "candidate_a_values": list(candidate_a),
        "candidate_b_values": list(candidate_b),
    }


def render(scene: dict) -> Image.Image:
    reference, candidate_a, candidate_b = _validate_scene(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(28)
    body_font = _font(18)
    panel_font = _font(21)
    small_font = _font(15)
    number_font = _font(17)

    draw.text((42, 23), "COLOR RESIDUAL BOARD", font=title_font, fill=INK)
    draw.text((42, 61), "The same four colors repeat; use the shared ruler to compare heights.", font=body_font, fill=INK)
    draw.text((42, 91), "REFERENCE is the baseline. Bars are matched by color, not by position.", font=small_font, fill=SUBTLE)

    panel_values = (reference, candidate_a, candidate_b)
    panel_titles = ("REFERENCE", "CANDIDATE A", "CANDIDATE B")
    for panel_index, (x0, values, title) in enumerate(zip(PANEL_X, panel_values, panel_titles, strict=True)):
        _centered_text(draw, (x0 + PANEL_WIDTH / 2, 125), title, panel_font, INK)
        draw.rectangle((x0, PANEL_TOP, x0 + PANEL_WIDTH, PANEL_BOTTOM), fill=PANEL_FILL, outline=INK, width=2)
        ruler_x = x0 + 49
        for level in range(0, 7):
            y = BASELINE - level * LEVEL_PITCH
            draw.line((ruler_x, y, x0 + PANEL_WIDTH - 16, y), fill=GUIDE, width=1)
            draw.line((ruler_x - 7, y, ruler_x + 7, y), fill=SUBTLE, width=2)
            _centered_text(draw, (x0 + 24, y), str(level), number_font, INK)
        draw.line((ruler_x, BASELINE - 6 * LEVEL_PITCH, ruler_x, BASELINE), fill=SUBTLE, width=2)
        for color_name, value, offset in zip(COLOR_NAMES, values, BAR_OFFSETS, strict=True):
            left = x0 + offset
            top = BASELINE - int(value) * LEVEL_PITCH
            draw.rectangle((left, top, left + BAR_WIDTH, BASELINE), fill=BAR_COLORS[color_name])
            _centered_text(draw, (left + BAR_WIDTH / 2, BASELINE + 27), COLOR_SHORT[color_name], small_font, INK)
        if panel_index == 0:
            draw.text((x0 + 15, PANEL_BOTTOM - 28), "height levels", font=small_font, fill=SUBTLE)

    draw.text((42, 752), "For each color: take |candidate height − reference height|, then add the four gaps.", font=body_font, fill=INK)
    draw.text((42, 783), "The answer is the candidate with the larger total; equal totals are tie.", font=small_font, fill=SUBTLE)
    return image


def analytic_gold(scene: dict) -> str:
    reference, candidate_a, candidate_b = _validate_scene(scene)
    error_a = _residual(reference, candidate_a)
    error_b = _residual(reference, candidate_b)
    if error_a > error_b:
        return "A"
    if error_b > error_a:
        return "B"
    return "tie"


def margin(scene: dict) -> float:
    reference, candidate_a, candidate_b = _validate_scene(scene)
    return float(_residual(reference, candidate_a) - _residual(reference, candidate_b))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) == 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # Every array entry changes one rendered color's bar top.  No latent alias
    # is declared, so the v0.4 tri-state gate should report not_applicable.
    return []
