"""Independent image-only inverse for the color-residual board.

The inverse intentionally duplicates only observable geometry.  It segments
each exact bar color inside each panel, measures the top of the component
against the common ruler, matches values by color, and computes two L1 sums.
It does not import any candidate module or accept latent scene data.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "color_residual_compare-oracle-0.1.0"

_WIDTH = 1260
_HEIGHT = 820
_PANEL_X = (35, 435, 835)
_PANEL_WIDTH = 390
_BASELINE = 620
_LEVEL_PITCH = 48.0
_BAR_COLORS = {
    "red": np.array((210, 70, 67), dtype=np.uint8),
    "blue": np.array((52, 101, 184), dtype=np.uint8),
    "green": np.array((47, 145, 91), dtype=np.uint8),
    "violet": np.array((145, 78, 174), dtype=np.uint8),
}
_COLOR_NAMES = ("red", "blue", "green", "violet")
_MIN_BAR_PIXELS = 500


def _components(mask: np.ndarray) -> list[np.ndarray]:
    ys, xs = np.nonzero(mask)
    seen = np.zeros(mask.shape, dtype=bool)
    components: list[np.ndarray] = []
    h, w = mask.shape
    for y0, x0 in zip(ys.tolist(), xs.tolist()):
        if seen[y0, x0]:
            continue
        queue = deque([(y0, x0)])
        seen[y0, x0] = True
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dy == 0 and dx == 0:
                        continue
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        components.append(np.asarray(points, dtype=np.int32))
    return components


def _panel_values(arr: np.ndarray, panel_x: int) -> dict[str, int]:
    values: dict[str, int] = {}
    for color_name in _COLOR_NAMES:
        mask = np.all(arr == _BAR_COLORS[color_name], axis=2)
        mask[:250, :] = False
        mask[_BASELINE + 2 :, :] = False
        mask[:, :panel_x] = False
        mask[:, panel_x + _PANEL_WIDTH + 1 :] = False
        components = [component for component in _components(mask) if len(component) >= _MIN_BAR_PIXELS]
        if len(components) != 1:
            raise ValueError(f"expected one {color_name} bar in panel, found {len(components)}")
        component = components[0]
        top = int(component[:, 0].min())
        bottom = int(component[:, 0].max())
        if bottom != _BASELINE:
            raise ValueError(f"{color_name} bar does not reach the common baseline")
        value = int(round((_BASELINE - top) / _LEVEL_PITCH))
        if value < 1 or value > 6 or abs((_BASELINE - top) - value * _LEVEL_PITCH) > 1.0:
            raise ValueError(f"{color_name} bar is not aligned to the ruler")
        values[color_name] = value
    if len(set(values.values())) != 4:
        raise ValueError("panel does not contain four distinct levels")
    return values


def _residual(reference: dict[str, int], candidate: dict[str, int]) -> int:
    return sum(abs(candidate[name] - reference[name]) for name in _COLOR_NAMES)


def decision_from_image(image: Image.Image) -> str:
    """Return ``A``, ``B``, or ``tie`` from the raster alone."""

    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.shape != (_HEIGHT, _WIDTH, 3):
        raise ValueError(f"unexpected image shape {arr.shape}")
    reference = _panel_values(arr, _PANEL_X[0])
    candidate_a = _panel_values(arr, _PANEL_X[1])
    candidate_b = _panel_values(arr, _PANEL_X[2])
    reference_levels = sorted(reference.values())
    if sorted(candidate_a.values()) != reference_levels or sorted(candidate_b.values()) != reference_levels:
        raise ValueError("panels do not share one height multiset")
    error_a = _residual(reference, candidate_a)
    error_b = _residual(reference, candidate_b)
    if error_a > error_b:
        return "A"
    if error_b > error_a:
        return "B"
    return "tie"
