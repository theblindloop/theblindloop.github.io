"""Renderer and analytic arm for the compact irreversible seal gate."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
CELL = 38
COLS = 3
ROWS = 2
PATCH_W = CELL * COLS
PATCH_H = CELL * ROWS
BACKGROUND = (244, 241, 232)
INK = (248, 247, 240)
GRID = (184, 190, 187)
GOLD = (224, 174, 54)
PALETTES = ((38, 83, 101), (91, 59, 105), (42, 91, 72))
COMMAND_MULTISET = ("toggle", "toggle", "seal", "wait", "wait")


def _replay(initial: int, commands: list[str]) -> int:
    lamp = int(initial)
    gate_open = True
    for command in commands:
        if command == "seal":
            gate_open = False
        elif command == "toggle" and gate_open:
            lamp ^= 1
    return lamp


def sample_scene(seed: int) -> dict:
    """Sample balanced decisions while varying placement, palette, and order."""
    rng = random.Random(int(seed) ^ 0x5EA1A7E)
    initial = rng.randrange(2)
    target = int(seed) & 1
    permutations = sorted(set(itertools.permutations(COMMAND_MULTISET)))
    eligible = [list(item) for item in permutations if _replay(initial, list(item)) == target]
    commands = eligible[rng.randrange(len(eligible))]
    return {
        "init_state": initial,
        "commands": commands,
        "origin_x": rng.randrange(42, WIDTH - PATCH_W - 42),
        "origin_y": rng.randrange(42, HEIGHT - PATCH_H - 42),
        "palette": rng.randrange(len(PALETTES)),
    }


def _cell_box(x0: int, y0: int, index: int) -> tuple[int, int, int, int]:
    row, col = divmod(index, COLS)
    left = x0 + col * CELL
    top = y0 + row * CELL
    return left, top, left + CELL - 1, top + CELL - 1


def _dot(draw: ImageDraw.ImageDraw, cx: int, cy: int, radius: int, *, fill=None, outline=None, width=1):
    draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=fill, outline=outline, width=width)


def render(scene: dict) -> Image.Image:
    initial = int(scene["init_state"])
    commands = list(scene["commands"])
    x0 = int(scene["origin_x"])
    y0 = int(scene["origin_y"])
    tile = PALETTES[int(scene["palette"])]

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rectangle((x0, y0, x0 + PATCH_W - 1, y0 + PATCH_H - 1), fill=tile)
    for col in (1, 2):
        x = x0 + col * CELL
        draw.rectangle((x - 1, y0, x, y0 + PATCH_H - 1), fill=GRID)
    y = y0 + CELL
    draw.rectangle((x0, y - 1, x0 + PATCH_W - 1, y), fill=GRID)

    # The gold inset declares the first tile as START.
    sx0, sy0, sx1, sy1 = _cell_box(x0, y0, 0)
    draw.rectangle((sx0 + 3, sy0 + 3, sx1 - 3, sy1 - 3), outline=GOLD, width=3)
    scx, scy = (sx0 + sx1) // 2, (sy0 + sy1) // 2
    if initial:
        _dot(draw, scx, scy, 7, fill=INK)
    else:
        _dot(draw, scx, scy, 7, outline=INK, width=3)

    for index, command in enumerate(commands, start=1):
        left, top, right, bottom = _cell_box(x0, y0, index)
        cx, cy = (left + right) // 2, (top + bottom) // 2
        if command == "toggle":
            draw.rectangle((cx - 7, cy - 9, cx - 3, cy + 9), fill=INK)
            draw.rectangle((cx + 3, cy - 9, cx + 7, cy + 9), fill=INK)
        elif command == "seal":
            draw.rectangle((cx - 9, cy - 9, cx + 9, cy + 9), fill=INK)
        elif command == "wait":
            _dot(draw, cx, cy, 4, fill=INK)
        else:
            raise ValueError(f"unknown command {command!r}")
    return image


def analytic_gold(scene: dict) -> str:
    state = int(scene["init_state"])
    open_gate = True
    for command in scene["commands"]:
        if command == "seal":
            open_gate = False
        elif command == "toggle" and open_gate:
            state ^= 1
    return "yes" if state else "no"


def margin(scene: dict) -> float:
    # Nearest command-glyph pixel-count bands are separated by at least 70 px.
    return 70.0


def is_quarantined(scene: dict) -> bool:
    commands = list(scene.get("commands", []))
    return not (
        scene.get("init_state") in (0, 1)
        and len(commands) == 5
        and commands.count("toggle") == 2
        and commands.count("seal") == 1
        and commands.count("wait") == 2
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
