"""Independent pixel-only parser for Seal Gate Patch.

This module intentionally contains its own palette, geometry checks, glyph
classification, and replay.  It imports no world-generation or gold code.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_SIZE = (512, 512)
_CELL = 38
_PATCH_W = 114
_PATCH_H = 76
_TILES = np.asarray(((38, 83, 101), (91, 59, 105), (42, 91, 72)), dtype=np.int32)


def _recover_patch(array: np.ndarray) -> tuple[int, int, np.ndarray]:
    rgb = array.astype(np.int32)
    distances = np.sqrt(((rgb[:, :, None, :] - _TILES[None, None, :, :]) ** 2).sum(axis=3))
    nearest = distances.min(axis=2)
    mask = nearest < 8.0
    ys, xs = np.nonzero(mask)
    if len(xs) < 6500:
        raise ValueError("tile field is missing or materially erased")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    if (x1 - x0 + 1, y1 - y0 + 1) != (_PATCH_W, _PATCH_H):
        raise ValueError("tile field has invalid bounds")
    # All cells must retain most of their solid interior. This makes partial
    # erasure an explicit abstention rather than a guessed replay.
    for index in range(6):
        row, col = divmod(index, 3)
        left, top = x0 + col * _CELL, y0 + row * _CELL
        interior = mask[top + 2 : top + _CELL - 2, left + 2 : left + _CELL - 2]
        if interior.shape != (34, 34) or float(interior.mean()) < 0.52:
            raise ValueError("one or more tiles are incomplete")
    return x0, y0, mask


def _white_count(array: np.ndarray, x0: int, y0: int, index: int) -> int:
    row, col = divmod(index, 3)
    left, top = x0 + col * _CELL, y0 + row * _CELL
    crop = array[top + 7 : top + 31, left + 7 : left + 31]
    white = (crop[:, :, 0] > 235) & (crop[:, :, 1] > 235) & (crop[:, :, 2] > 225)
    return int(white.sum())


def _command_from_count(count: int) -> str:
    if 35 <= count <= 90:
        return "wait"
    if 145 <= count <= 230:
        return "toggle"
    if 300 <= count <= 400:
        return "seal"
    raise ValueError(f"ambiguous command glyph ({count} bright pixels)")


def decision_from_image(image: Image.Image) -> str:
    if image.size != _SIZE:
        raise ValueError("unexpected canvas size")
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    x0, y0, _ = _recover_patch(array)
    start_count = _white_count(array, x0, y0, 0)
    if 90 <= start_count <= 125:
        state = 0
    elif 145 <= start_count <= 210:
        state = 1
    else:
        raise ValueError(f"ambiguous START lamp ({start_count} bright pixels)")

    commands = [_command_from_count(_white_count(array, x0, y0, index)) for index in range(1, 6)]
    if sorted(commands) != ["seal", "toggle", "toggle", "wait", "wait"]:
        raise ValueError("command multiset is invalid")
    gate_open = True
    for command in commands:
        if command == "seal":
            gate_open = False
        elif command == "toggle" and gate_open:
            state ^= 1
    return "yes" if state else "no"
