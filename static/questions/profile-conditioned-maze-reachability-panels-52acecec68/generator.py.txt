"""Deterministic three-panel maze renderer and analytic topology oracle."""

from __future__ import annotations

import random
from collections import deque
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "maze_reachability_panels-0.1.0"

WIDTH = 780
HEIGHT = 300
PANEL_ORIGINS = (10, 270, 530)
PANEL_TOP = 30
PANEL_SIZE = 240
GRID_SIZE = 11
CELL_SIZE = 20
GRID_OFFSET = 10
OPEN_EDGE_COUNT = 64
MARKER_RADIUS = 5
WALL_WIDTHS = (6, 7, 8)
QUARANTINE_CLEARANCE = 4.0
POSITIONS = ("left", "center", "right")

BACKGROUND = (255, 255, 255)
BLACK = (28, 31, 38)
RED = (218, 54, 61)
GREEN = (48, 158, 91)


def _edge_key(a: tuple[int, int], b: tuple[int, int]) -> tuple[int, int, int, int]:
    """Canonical undirected grid edge."""
    return (*a, *b) if a <= b else (*b, *a)


def _all_edges() -> list[tuple[int, int, int, int]]:
    edges: list[tuple[int, int, int, int]] = []
    for row in range(GRID_SIZE):
        for col in range(GRID_SIZE):
            if col + 1 < GRID_SIZE:
                edges.append(_edge_key((row, col), (row, col + 1)))
            if row + 1 < GRID_SIZE:
                edges.append(_edge_key((row, col), (row + 1, col)))
    return edges


def _adjacency(open_edges: list[list[int]] | list[tuple[int, int, int, int]]) -> dict[tuple[int, int], set[tuple[int, int]]]:
    graph = {
        (row, col): set()
        for row in range(GRID_SIZE)
        for col in range(GRID_SIZE)
    }
    for edge in open_edges:
        row_a, col_a, row_b, col_b = (int(value) for value in edge)
        first = (row_a, col_a)
        second = (row_b, col_b)
        graph[first].add(second)
        graph[second].add(first)
    return graph


def _reachable_graph(
    start: tuple[int, int],
    goal: tuple[int, int],
    open_edges: set[tuple[int, int, int, int]] | list[list[int]] | list[tuple[int, int, int, int]],
) -> bool:
    if isinstance(open_edges, set):
        graph_edges = open_edges
    else:
        graph_edges = {
            _edge_key((int(edge[0]), int(edge[1])), (int(edge[2]), int(edge[3])))
            for edge in open_edges
        }
    graph = _adjacency([list(edge) for edge in graph_edges])
    queue: deque[tuple[int, int]] = deque([start])
    seen = {start}
    while queue:
        current = queue.popleft()
        if current == goal:
            return True
        for neighbor in graph[current]:
            if neighbor not in seen:
                seen.add(neighbor)
                queue.append(neighbor)
    return False


def _sample_marker_pair(rng: random.Random) -> tuple[tuple[int, int], tuple[int, int]]:
    cells = [(row, col) for row in range(GRID_SIZE) for col in range(GRID_SIZE)]
    start = rng.choice(cells)
    choices = [
        cell
        for cell in cells
        if abs(cell[0] - start[0]) + abs(cell[1] - start[1]) >= 8
    ]
    return start, rng.choice(choices)


def _short_path(
    start: tuple[int, int], goal: tuple[int, int], rng: random.Random
) -> set[tuple[int, int, int, int]]:
    current = start
    path: set[tuple[int, int, int, int]] = set()
    while current != goal:
        row, col = current
        candidates: list[tuple[int, int]] = []
        if row < goal[0]:
            candidates.append((row + 1, col))
        if row > goal[0]:
            candidates.append((row - 1, col))
        if col < goal[1]:
            candidates.append((row, col + 1))
        if col > goal[1]:
            candidates.append((row, col - 1))
        rng.shuffle(candidates)
        next_cell = candidates[0]
        path.add(_edge_key(current, next_cell))
        current = next_cell
    return path


def _connected_edges(
    start: tuple[int, int], goal: tuple[int, int], rng: random.Random
) -> set[tuple[int, int, int, int]]:
    all_edges = _all_edges()
    chosen = _short_path(start, goal, rng)
    remaining = [edge for edge in all_edges if edge not in chosen]
    rng.shuffle(remaining)
    chosen.update(remaining[: OPEN_EDGE_COUNT - len(chosen)])
    return chosen


def _disconnected_edges(
    start: tuple[int, int], goal: tuple[int, int], rng: random.Random
) -> set[tuple[int, int, int, int]]:
    all_edges = _all_edges()
    for _ in range(500):
        candidate = set(rng.sample(all_edges, OPEN_EDGE_COUNT))
        if not _reachable_graph(start, goal, candidate):
            return candidate
    # The sparse graph above normally succeeds immediately. The fallback keeps
    # the construction total without ever relaxing the disconnected invariant.
    candidate = set()
    for edge in all_edges:
        if edge[0] < GRID_SIZE // 2 and edge[2] < GRID_SIZE // 2:
            candidate.add(edge)
    for edge in all_edges:
        if len(candidate) >= OPEN_EDGE_COUNT:
            break
        if edge not in candidate and not (
            edge[0] == GRID_SIZE // 2 - 1 and edge[2] == GRID_SIZE // 2
        ):
            candidate.add(edge)
    if _reachable_graph(start, goal, candidate):
        raise RuntimeError("could not construct a disconnected maze")
    return candidate


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one scene with exactly one reachable marker pair."""
    rng = random.Random(int(seed))
    start, goal = _sample_marker_pair(rng)
    true_panel = rng.randrange(3)
    wall_width = rng.choice(WALL_WIDTHS)
    panels: list[dict[str, Any]] = []
    open_edges_by_panel: list[list[list[int]]] = []
    for index in range(3):
        if index == true_panel:
            edges = _connected_edges(start, goal, rng)
        else:
            edges = _disconnected_edges(start, goal, rng)
        panels.append({"wall_width": wall_width})
        open_edges_by_panel.append([list(edge) for edge in sorted(edges)])
    scene = {
        "panels": panels,
        "start": [start[0], start[1]],
        "goal": [goal[0], goal[1]],
        "open_edges": open_edges_by_panel,
    }
    if sum(_panel_reachable(scene, index) for index in range(3)) != 1:
        raise RuntimeError("sampled maze scene does not have a unique answer")
    return scene


def _panel_reachable(scene: dict[str, Any], index: int) -> bool:
    start = (int(scene["start"][0]), int(scene["start"][1]))
    goal = (int(scene["goal"][0]), int(scene["goal"][1]))
    return _reachable_graph(start, goal, scene["open_edges"][index])


def _wall_vertical(
    draw: ImageDraw.ImageDraw, x: int, y0: int, y1: int, width: int
) -> None:
    half = width // 2
    draw.rectangle((x - half, y0, x + half - 1, y1 - 1), fill=BLACK)


def _wall_horizontal(
    draw: ImageDraw.ImageDraw, x0: int, x1: int, y: int, width: int
) -> None:
    half = width // 2
    draw.rectangle((x0, y - half, x1 - 1, y + half - 1), fill=BLACK)


def _cell_center(origin_x: int, origin_y: int, cell: list[int] | tuple[int, int]) -> tuple[int, int]:
    row, col = int(cell[0]), int(cell[1])
    return (
        origin_x + col * CELL_SIZE + CELL_SIZE // 2,
        origin_y + row * CELL_SIZE + CELL_SIZE // 2,
    )


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only public scene fields into a fixed RGB raster."""
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    all_edges = _all_edges()
    for index, panel in enumerate(scene["panels"]):
        x0 = PANEL_ORIGINS[index]
        y0 = PANEL_TOP
        draw.rectangle(
            (x0, y0, x0 + PANEL_SIZE - 1, y0 + PANEL_SIZE - 1),
            fill=BACKGROUND,
            outline=BLACK,
            width=6,
        )
        origin_x = x0 + GRID_OFFSET
        origin_y = y0 + GRID_OFFSET
        wall_width = int(panel["wall_width"])
        open_edges = {
            _edge_key((int(edge[0]), int(edge[1])), (int(edge[2]), int(edge[3])))
            for edge in scene["open_edges"][index]
        }
        _wall_horizontal(
            draw, origin_x, origin_x + GRID_SIZE * CELL_SIZE + 1, origin_y, wall_width
        )
        _wall_horizontal(
            draw,
            origin_x,
            origin_x + GRID_SIZE * CELL_SIZE + 1,
            origin_y + GRID_SIZE * CELL_SIZE,
            wall_width,
        )
        _wall_vertical(
            draw, origin_x, origin_y, origin_y + GRID_SIZE * CELL_SIZE + 1, wall_width
        )
        _wall_vertical(
            draw,
            origin_x + GRID_SIZE * CELL_SIZE,
            origin_y,
            origin_y + GRID_SIZE * CELL_SIZE + 1,
            wall_width,
        )
        for row_a, col_a, row_b, col_b in all_edges:
            edge = (row_a, col_a, row_b, col_b)
            if edge in open_edges:
                continue
            if row_a == row_b:
                boundary_x = origin_x + max(col_a, col_b) * CELL_SIZE
                _wall_vertical(
                    draw,
                    boundary_x,
                    origin_y + row_a * CELL_SIZE,
                    origin_y + (row_a + 1) * CELL_SIZE + 1,
                    wall_width,
                )
            else:
                boundary_y = origin_y + max(row_a, row_b) * CELL_SIZE
                _wall_horizontal(
                    draw,
                    origin_x + col_a * CELL_SIZE,
                    origin_x + (col_a + 1) * CELL_SIZE + 1,
                    boundary_y,
                    wall_width,
                )
        red_x, red_y = _cell_center(origin_x, origin_y, scene["start"])
        green_x, green_y = _cell_center(origin_x, origin_y, scene["goal"])
        draw.ellipse(
            (red_x - MARKER_RADIUS, red_y - MARKER_RADIUS,
             red_x + MARKER_RADIUS, red_y + MARKER_RADIUS),
            fill=RED,
        )
        draw.ellipse(
            (green_x - MARKER_RADIUS, green_y - MARKER_RADIUS,
             green_x + MARKER_RADIUS, green_y + MARKER_RADIUS),
            fill=GREEN,
        )
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the unique panel whose visible grid connects both markers."""
    reachable = [_panel_reachable(scene, index) for index in range(3)]
    if sum(reachable) != 1:
        raise ValueError(f"scene must have exactly one reachable panel: {reachable}")
    return POSITIONS[reachable.index(True)]


def margin(scene: dict[str, Any]) -> float:
    """Conservative raster margin: the thinnest visible wall/marker encoding."""
    widths = [float(panel["wall_width"]) for panel in scene["panels"]]
    return float(min(min(widths), CELL_SIZE - 2 * MARKER_RADIUS))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < QUARANTINE_CLEARANCE


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """The ordered panel list is public; no latent alias is declared."""
    return []
