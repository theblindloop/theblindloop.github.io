"""Independent image-only inverse for maze reachability."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "maze_reachability_panels-oracle-0.1.0"

BLACK = np.array((28, 31, 38), dtype=np.int16)
RED = np.array((218, 54, 61), dtype=np.int16)
GREEN = np.array((48, 158, 91), dtype=np.int16)
MARKER_TOLERANCE = 30.0
BLACK_TOLERANCE = 70.0


def _distance_to(rgb: np.ndarray, color: np.ndarray) -> np.ndarray:
    delta = rgb.astype(np.float32) - color.astype(np.float32)
    return np.sqrt(np.sum(delta * delta, axis=2))


def _marker_center(mask: np.ndarray) -> tuple[int, int]:
    points = np.argwhere(mask)
    if len(points) < 20:
        raise ValueError("marker color is absent or too small")
    y, x = np.rint(points.mean(axis=0)).astype(int)
    return int(y), int(x)


def _frame_bounds(dark: np.ndarray, x_start: int, x_end: int) -> tuple[int, int, int, int]:
    """Find a framed panel from long dark runs, without scene coordinates."""
    local = dark[:, x_start:x_end]
    row_counts = local.sum(axis=1)
    candidate_rows = np.flatnonzero(row_counts >= 0.70 * local.shape[1])
    if len(candidate_rows) < 2:
        raise ValueError("could not identify the horizontal panel frame")
    top = int(candidate_rows[0])
    bottom = int(candidate_rows[-1])
    panel = dark[top : bottom + 1, x_start:x_end]
    col_counts = panel.sum(axis=0)
    candidate_cols = np.flatnonzero(col_counts >= 0.70 * panel.shape[0])
    if len(candidate_cols) < 2:
        raise ValueError("could not identify the vertical panel frame")
    left = x_start + int(candidate_cols[0])
    right = x_start + int(candidate_cols[-1])
    return top, bottom, left, right


def _reachable_pixels(
    passable: np.ndarray,
    start: tuple[int, int],
    goal: tuple[int, int],
    bounds: tuple[int, int, int, int],
) -> bool:
    top, bottom, left, right = bounds
    inner_top, inner_bottom = top + 6, bottom - 5
    inner_left, inner_right = left + 6, right - 5
    if not (inner_top <= start[0] <= inner_bottom and inner_left <= start[1] <= inner_right):
        raise ValueError("red marker is outside its framed panel")
    if not (inner_top <= goal[0] <= inner_bottom and inner_left <= goal[1] <= inner_right):
        raise ValueError("green marker is outside its framed panel")
    view = passable[inner_top : inner_bottom + 1, inner_left : inner_right + 1]
    origin = (inner_top, inner_left)
    start_local = (start[0] - origin[0], start[1] - origin[1])
    goal_local = (goal[0] - origin[0], goal[1] - origin[1])
    if not view[start_local] or not view[goal_local]:
        raise ValueError("marker is hidden by a dark wall")
    queue: deque[tuple[int, int]] = deque([start_local])
    seen = np.zeros(view.shape, dtype=bool)
    seen[start_local] = True
    while queue:
        row, col = queue.popleft()
        if (row, col) == goal_local:
            return True
        for next_row, next_col in (
            (row - 1, col),
            (row + 1, col),
            (row, col - 1),
            (row, col + 1),
        ):
            if (
                0 <= next_row < view.shape[0]
                and 0 <= next_col < view.shape[1]
                and view[next_row, next_col]
                and not seen[next_row, next_col]
            ):
                seen[next_row, next_col] = True
                queue.append((next_row, next_col))
    return False


def decision_from_image(image: Image.Image) -> str:
    """Return ``left``, ``center``, or ``right`` from the PNG alone."""
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    if rgb.ndim != 3 or rgb.shape[2] != 3 or rgb.shape[1] < 180:
        raise ValueError("image is not a valid three-panel raster")
    red = _distance_to(rgb, RED) <= MARKER_TOLERANCE
    green = _distance_to(rgb, GREEN) <= MARKER_TOLERANCE
    dark = _distance_to(rgb, BLACK) <= BLACK_TOLERANCE
    passable = ~dark
    reachable: list[bool] = []
    width = rgb.shape[1]
    for index in range(3):
        x_start = index * width // 3
        x_end = (index + 1) * width // 3
        bounds = _frame_bounds(dark, x_start, x_end)
        red_yx = _marker_center(red[:, x_start:x_end])
        green_yx = _marker_center(green[:, x_start:x_end])
        red_global = (red_yx[0], red_yx[1] + x_start)
        green_global = (green_yx[0], green_yx[1] + x_start)
        reachable.append(_reachable_pixels(passable, red_global, green_global, bounds))
    if sum(reachable) != 1:
        raise ValueError(f"expected one reachable panel, measured {reachable}")
    return ("left", "center", "right")[reachable.index(True)]
