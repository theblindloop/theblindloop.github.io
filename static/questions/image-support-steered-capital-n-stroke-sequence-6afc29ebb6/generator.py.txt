"""Renderer and analytic arm for the capital-N stroke-sequence world."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

CANVAS = 384
TARGET_DIRECTIONS = ((0.0, -1.0), (1.0, 1.0), (0.0, -1.0))
BLUE_PALETTE = ((20, 104, 205), (12, 122, 198), (42, 91, 201))
RED_PALETTE = ((214, 48, 55), (196, 38, 62), (225, 65, 42))


def sample_scene(seed: int) -> dict:
    """Return balanced canonical/counterfactual routes with matched nuisance style."""
    rng = random.Random(int(seed))
    side = rng.randrange(190, 231)
    left = rng.randrange(68, CANVAS - side - 67)
    top = rng.randrange(78, CANVAS - side - 54)
    right, bottom = left + side, top + side
    j = lambda: rng.randrange(-5, 6)
    p0 = [left + j(), bottom + j()]
    p3 = [right + j(), top + j()]
    if int(seed) % 2 == 0:
        # Canonical single-stroke N: up, down-right, up.
        points = [p0, [left + j(), top + j()], [right + j(), bottom + j()], p3]
    else:
        # Matched counterfactual: right, up-left, right. Same endpoints and lengths.
        points = [p0, [right + j(), bottom + j()], [left + j(), top + j()], p3]
    return {
        "points": points,
        "start_index": 0,
        "stroke_color": list(rng.choice(BLUE_PALETTE)),
        "start_color": list(rng.choice(RED_PALETTE)),
        "background": [249, 248, 244],
        "stroke_width": rng.randrange(10, 15),
        "width": CANVAS,
        "height": CANVAS,
    }


def _ordered_points(scene: dict) -> list[list[int]]:
    points = scene["points"]
    start_index = int(scene["start_index"])
    if start_index == 0:
        return [list(p) for p in points]
    if start_index == len(points) - 1:
        return [list(p) for p in reversed(points)]
    raise ValueError("start_index must name one endpoint")


def render(scene: dict) -> Image.Image:
    """Render a neutral practice card; only the blue route determines the answer."""
    w, h = int(scene["width"]), int(scene["height"])
    ss = 4
    bg = tuple(scene["background"])
    image = Image.new("RGB", (w * ss, h * ss), bg)
    draw = ImageDraw.Draw(image)
    # Neutral card context is identical across classes and deliberately not read by the oracle.
    draw.rounded_rectangle(
        (18 * ss, 18 * ss, (w - 18) * ss, (h - 18) * ss),
        radius=15 * ss,
        outline=(181, 183, 187),
        width=2 * ss,
    )
    font = ImageFont.load_default(size=15 * ss)
    draw.text((w * ss // 2, 37 * ss), "CAPITAL N PRACTICE", fill=(92, 94, 99), anchor="mm", font=font)

    points = [(int(p[0]) * ss, int(p[1]) * ss) for p in _ordered_points(scene)]
    sw = int(scene["stroke_width"]) * ss
    blue = tuple(scene["stroke_color"])
    draw.line(points, fill=blue, width=sw, joint="curve")
    radius = sw // 2
    for x, y in points[1:-1]:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=blue)

    # A visible start marker makes traversal direction pixel-answerable.
    sx, sy = points[0]
    rr = int(scene["stroke_width"] * 0.82 * ss)
    red = tuple(scene["start_color"])
    draw.ellipse((sx - rr, sy - rr, sx + rr, sy + rr), fill=red)
    draw.text((sx, sy + 19 * ss), "START", fill=(92, 94, 99), anchor="ma", font=font)
    return image.resize((w, h), Image.Resampling.LANCZOS)


def _angle_error(a: tuple[float, float], b: tuple[float, float]) -> float:
    dot = a[0] * b[0] + a[1] * b[1]
    na = math.hypot(*a)
    nb = math.hypot(*b)
    cosine = max(-1.0, min(1.0, dot / (na * nb)))
    return math.degrees(math.acos(cosine))


def _errors(scene: dict) -> list[float]:
    points = _ordered_points(scene)
    runs = [(b[0] - a[0], b[1] - a[1]) for a, b in zip(points, points[1:])]
    return [_angle_error(run, target) for run, target in zip(runs, TARGET_DIRECTIONS)]


def analytic_gold(scene: dict) -> str:
    """Gold is computed only from the visibly started point sequence."""
    return "yes" if max(_errors(scene)) < 35.0 else "no"


def margin(scene: dict) -> float:
    """Angular distance in degrees from the 35-degree composition boundary."""
    return min(abs(error - 35.0) for error in _errors(scene))


def is_quarantined(scene: dict) -> bool:
    points = _ordered_points(scene)
    shortest = min(math.dist(a, b) for a, b in zip(points, points[1:]))
    return margin(scene) < 12.0 or shortest < 120.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse the latent enumeration while preserving the visible START endpoint."""
    twin = deepcopy(scene)
    twin["points"] = list(reversed(twin["points"]))
    twin["start_index"] = len(twin["points"]) - 1
    return [("reverse_enumeration", twin)]
