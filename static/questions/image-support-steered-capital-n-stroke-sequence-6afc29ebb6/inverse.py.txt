"""Independent image-only inverse arm for the connected stroke."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _blue_mask(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    r, g, b = arr[..., 0], arr[..., 1], arr[..., 2]
    return (b > 135) & (b - r > 65) & (b - g > 35)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    seen = np.zeros(mask.shape, dtype=bool)
    components: list[np.ndarray] = []
    height, width = mask.shape
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        pixels: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            pixels.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(pixels) >= 40:
            components.append(np.asarray(pixels, dtype=np.int32))
    return components


def decision_from_image(image: Image.Image) -> str:
    """Classify side-spanning versus top/bottom-spanning runs in the blue route.

    The oracle locates the route from color pixels, checks it remains one connected
    component, normalizes by its measured bounding box, and uses coverage along all
    four sides. It receives no coordinates, scene state, or analytic answer.
    """
    mask = _blue_mask(image)
    components = _components(mask)
    if len(components) != 1 or len(components[0]) < 900:
        raise ValueError("blue route is missing, fragmented, or too small")
    pixels = components[0]
    y0, x0 = pixels.min(axis=0)
    y1, x1 = pixels.max(axis=0)
    bw, bh = int(x1 - x0 + 1), int(y1 - y0 + 1)
    if min(bw, bh) < 150 or not 0.78 <= bw / bh <= 1.28:
        raise ValueError("route bounding box is not a clear regional square")
    crop = mask[y0 : y1 + 1, x0 : x1 + 1]
    xb = max(8, round(0.15 * bw))
    yb = max(8, round(0.15 * bh))
    left_rows = np.count_nonzero(crop[:, :xb].any(axis=1)) / bh
    right_rows = np.count_nonzero(crop[:, -xb:].any(axis=1)) / bh
    top_cols = np.count_nonzero(crop[:yb, :].any(axis=0)) / bw
    bottom_cols = np.count_nonzero(crop[-yb:, :].any(axis=0)) / bw
    evidence_margin = (left_rows + right_rows) - (top_cols + bottom_cols)
    if abs(evidence_margin) < 0.45:
        raise ValueError("ordered-run evidence has insufficient runner-up margin")
    return "yes" if evidence_margin > 0 else "no"
