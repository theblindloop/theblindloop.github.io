"""Deterministic renderer and analytic state replay for Prefix Flip Stack."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 560
SHAPES = ("circle", "triangle", "square", "diamond")
PASTELS = ("peach", "mint", "lavender", "sand")
RGB = {
    "white": (250, 251, 253),
    "dark": (30, 35, 44),
    "muted": (105, 113, 126),
    "blue": (38, 108, 218),
    "gray": (205, 211, 220),
    "cyan": (0, 156, 184),
    "magenta": (207, 45, 132),
    "peach": (255, 211, 188),
    "mint": (190, 234, 214),
    "lavender": (218, 204, 244),
    "sand": (241, 224, 169),
}


def _apply_flips(
    order: list[str], faces: list[bool], flip_lengths: list[int]
) -> tuple[list[str], list[bool]]:
    order = list(order)
    faces = list(faces)
    for k in flip_lengths:
        order[:k] = reversed(order[:k])
        faces[:k] = [not value for value in reversed(faces[:k])]
    return order, faces


def _preimage_index(flip_lengths: list[int], target_slot: int) -> tuple[int, int]:
    tagged = list(range(4))
    toggles = [0, 0, 0, 0]
    for k in flip_lengths:
        tagged[:k] = reversed(tagged[:k])
        toggles[:k] = [value ^ 1 for value in reversed(toggles[:k])]
    return tagged[target_slot], toggles[target_slot]


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed) ^ 0xA51C9E)
    desired_shape = SHAPES[int(seed) % len(SHAPES)]
    desired_face = bool((int(seed) // len(SHAPES)) % 2)
    target_slot = rng.randrange(4)

    flip_lengths = [rng.choice((1, 2, 3, 4)) for _ in range(4)]
    if len(set(flip_lengths)) == 1:
        flip_lengths[1] = 2 if flip_lengths[0] != 2 else 3

    source_slot, toggle = _preimage_index(flip_lengths, target_slot)
    initial_order = list(SHAPES)
    rng.shuffle(initial_order)
    current = initial_order.index(desired_shape)
    initial_order[current], initial_order[source_slot] = (
        initial_order[source_slot],
        initial_order[current],
    )

    required_face = desired_face ^ bool(toggle)
    initial_faces = [False] * 4
    initial_faces[source_slot] = required_face
    remaining = [i for i in range(4) if i != source_slot]
    rng.shuffle(remaining)
    for index in remaining[: 2 - int(required_face)]:
        initial_faces[index] = True

    token_colors = list(PASTELS)
    rng.shuffle(token_colors)
    layout_shift = [rng.randint(-8, 8), rng.randint(-6, 6)]
    scene = {
        "initial_order": initial_order,
        "initial_faces": initial_faces,
        "flip_lengths": flip_lengths,
        "target_slot": target_slot,
        "token_colors": token_colors,
        "layout_shift": layout_shift,
    }
    assert analytic_gold(scene) == f"{desired_shape} {'up' if desired_face else 'down'}"
    return scene


def _draw_shape(draw: ImageDraw.ImageDraw, shape: str, cx: int, cy: int) -> None:
    ink = RGB["dark"]
    if shape == "circle":
        draw.ellipse((cx - 20, cy - 20, cx + 20, cy + 20), fill=ink)
    elif shape == "triangle":
        draw.polygon(((cx, cy - 20), (cx - 21, cy + 20), (cx + 21, cy + 20)), fill=ink)
    elif shape == "square":
        draw.rectangle((cx - 20, cy - 20, cx + 20, cy + 20), fill=ink)
    elif shape == "diamond":
        draw.polygon(((cx, cy - 21), (cx - 21, cy), (cx, cy + 21), (cx + 21, cy)), fill=ink)
    else:
        raise ValueError(f"unknown shape {shape!r}")


def render(scene: dict[str, Any]) -> Image.Image:
    order = list(scene["initial_order"])
    faces = [bool(value) for value in scene["initial_faces"]]
    flips = [int(value) for value in scene["flip_lengths"]]
    target = int(scene["target_slot"])
    colors = list(scene["token_colors"])
    shift_x, shift_y = [int(value) for value in scene["layout_shift"]]
    if sorted(order) != sorted(SHAPES) or len(faces) != 4 or len(colors) != 4:
        raise ValueError("scene must contain four unique shapes, faces, and colors")
    if len(flips) != 4 or any(k not in (1, 2, 3, 4) for k in flips):
        raise ValueError("scene must contain four prefix lengths in 1..4")

    image = Image.new("RGB", (WIDTH, HEIGHT), RGB["white"])
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    x0, y0 = 72 + shift_x, 108 + shift_y
    tile_w, tile_h, gap = 174, 72, 14

    draw.text((x0, 56 + shift_y), "START STACK", fill=RGB["dark"], font=font)
    draw.text((326 + shift_x, 132 + shift_y), "FLIP CARDS  (read left to right)", fill=RGB["dark"], font=font)
    draw.text((54 + shift_x, 474 + shift_y), "solid = face-up     hollow = face-down", fill=RGB["muted"], font=font)

    for slot, (shape, face, color) in enumerate(zip(order, faces, colors, strict=True)):
        top = y0 + slot * (tile_h + gap)
        draw.rectangle((x0, top, x0 + tile_w, top + tile_h), fill=RGB[color])
        _draw_shape(draw, shape, x0 + 49, top + tile_h // 2)
        dot_x, dot_y, radius = x0 + 136, top + tile_h // 2, 12
        if face:
            draw.ellipse((dot_x - radius, dot_y - radius, dot_x + radius, dot_y + radius), fill=RGB["cyan"])
        else:
            draw.ellipse(
                (dot_x - radius, dot_y - radius, dot_x + radius, dot_y + radius),
                outline=RGB["cyan"],
                width=4,
            )

    target_y = y0 + target * (tile_h + gap) + tile_h // 2
    draw.polygon(
        ((x0 - 12, target_y), (x0 - 40, target_y - 17), (x0 - 40, target_y + 17)),
        fill=RGB["magenta"],
    )
    draw.text((x0 - 68, target_y - 31), "TARGET", fill=RGB["magenta"], font=font)

    card_x0, card_y0 = 326 + shift_x, 184 + shift_y
    bar_w, bar_h, pitch = 76, 18, 29
    for card, k in enumerate(flips):
        left = card_x0 + card * 142
        for row in range(4):
            top = card_y0 + row * pitch
            color = RGB["blue"] if row < k else RGB["gray"]
            draw.rectangle((left, top, left + bar_w, top + bar_h), fill=color)
        draw.text((left + 29, card_y0 + 126), f"{card + 1}", fill=RGB["muted"], font=font)

    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    order, faces = _apply_flips(
        list(scene["initial_order"]),
        [bool(value) for value in scene["initial_faces"]],
        [int(value) for value in scene["flip_lengths"]],
    )
    target = int(scene["target_slot"])
    return f"{order[target]} {'up' if faces[target] else 'down'}"


def margin(scene: dict[str, Any]) -> float:
    # Minimum designed gap (px) among card bars, stack slots, and target choices.
    _ = scene["flip_lengths"]
    return 11.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    return not (
        sorted(scene["initial_order"]) == sorted(SHAPES)
        and len(scene["initial_faces"]) == 4
        and len(scene["flip_lengths"]) == 4
        and all(int(k) in (1, 2, 3, 4) for k in scene["flip_lengths"])
        and int(scene["target_slot"]) in range(4)
    )


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
