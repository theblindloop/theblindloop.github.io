"""Independent pixel-only inverse arm for the rubber-band constellation.

This module deliberately has no imports from any candidate-world module. It
segments the final raster, recovers component centroids, constructs the dark
constellation's convex hull, and returns the color with the largest signed
inside score.
"""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image


_PROTOTYPES = np.array(
    [
        (248, 247, 242),
        (39, 45, 51),
        (216, 67, 58),
        (45, 111, 195),
        (42, 148, 91),
        (232, 137, 38),
    ],
    dtype=np.int32,
)
_NAMES = ("red", "blue", "green", "orange")


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[float, float, int]] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        sx = sy = count = 0
        while queue:
            cy, cx = queue.popleft()
            sx += cx
            sy += cy
            count += 1
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if count >= 24:
            found.append((sx / count, sy / count, count))
    return found


def _turn(a: tuple[float, float], b: tuple[float, float], p: tuple[float, float]) -> float:
    return (b[0] - a[0]) * (p[1] - a[1]) - (b[1] - a[1]) * (p[0] - a[0])


def _hull(points: list[tuple[float, float]]) -> list[tuple[float, float]]:
    pts = sorted(set(points))
    if len(pts) < 3:
        raise ValueError("dark-dot evidence is missing")
    low: list[tuple[float, float]] = []
    for point in pts:
        while len(low) >= 2 and _turn(low[-2], low[-1], point) <= 0:
            low.pop()
        low.append(point)
    high: list[tuple[float, float]] = []
    for point in reversed(pts):
        while len(high) >= 2 and _turn(high[-2], high[-1], point) <= 0:
            high.pop()
        high.append(point)
    return low[:-1] + high[:-1]


def _inside_score(hull: list[tuple[float, float]], point: tuple[float, float]) -> float:
    distances = []
    for index, a in enumerate(hull):
        b = hull[(index + 1) % len(hull)]
        distances.append(_turn(a, b, point) / math.hypot(b[0] - a[0], b[1] - a[1]))
    return min(distances)


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int32)
    squared = ((pixels[:, :, None, :] - _PROTOTYPES[None, None, :, :]) ** 2).sum(axis=3)
    labels = squared.argmin(axis=2)

    dark = _components(labels == 1)
    if len(dark) < 6:
        raise ValueError(f"expected a dark constellation, found {len(dark)} components")
    hull = _hull([(item[0], item[1]) for item in dark])

    candidates: dict[str, tuple[float, float]] = {}
    for palette_index, name in enumerate(_NAMES, start=2):
        components = _components(labels == palette_index)
        if len(components) != 1:
            raise ValueError(f"expected exactly one {name} candidate, found {len(components)}")
        candidates[name] = (components[0][0], components[0][1])

    scores = {name: _inside_score(hull, point) for name, point in candidates.items()}
    winner = max(scores, key=scores.get)
    if scores[winner] <= 1.0:
        raise ValueError(f"no colored center is securely inside the implied hull: {scores}")
    if sum(value > 0.0 for value in scores.values()) != 1:
        raise ValueError(f"membership is not unique: {scores}")
    return winner
