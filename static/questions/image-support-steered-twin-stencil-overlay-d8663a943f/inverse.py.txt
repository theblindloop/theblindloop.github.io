"""Independent image-only inverse for the declared two-stencil intersection."""

from __future__ import annotations

import numpy as np
from PIL import Image

COLORS = (
    (15, 145, 185), (232, 112, 24),
    (122, 73, 178), (38, 153, 88),
    (39, 102, 190), (204, 67, 105),
)
DARK = np.asarray((28, 31, 38), dtype=np.int16)


def _template(coords: set[tuple[int, int]]) -> np.ndarray:
    out = np.zeros((7, 7), dtype=bool)
    for y, x in coords:
        out[y, x] = True
    return out


TEMPLATES = {
    "X": _template({(i, i) for i in range(7)} | {(i, 6 - i) for i in range(7)}),
    "plus": _template({(3, i) for i in range(7)} | {(i, 3) for i in range(7)}),
    "T": _template({(0, i) for i in range(7)} | {(i, 3) for i in range(1, 7)}),
    "L": _template({(i, 0) for i in range(7)} | {(6, i) for i in range(1, 7)}),
}


def _find_tiles(array: np.ndarray) -> list[tuple[int, int, int, int]]:
    found = []
    work = array.astype(np.int16)
    for color in COLORS:
        distance = np.linalg.norm(work - np.asarray(color, dtype=np.int16), axis=2)
        ys, xs = np.nonzero(distance < 10.0)
        if len(xs) < 200:
            continue
        box = (int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max()))
        width, height = box[2] - box[0] + 1, box[3] - box[1] + 1
        if not (29 <= width <= 32 and 29 <= height <= 32):
            raise ValueError("a colored tile fiducial is incomplete")
        found.append(box)
    if len(found) != 2:
        raise ValueError("exactly two visible colored tile fiducials are required")
    return found


def _read_tile(array: np.ndarray, box: tuple[int, int, int, int]) -> np.ndarray:
    x0, y0, _, _ = box
    mask = np.zeros((7, 7), dtype=bool)
    for y in range(7):
        for x in range(7):
            patch = array[y0 + 6 + 3 * y:y0 + 8 + 3 * y, x0 + 6 + 3 * x:x0 + 8 + 3 * x].astype(np.int16)
            if patch.shape != (2, 2, 3):
                raise ValueError("tile sample window is clipped")
            mask[y, x] = float(np.linalg.norm(patch - DARK, axis=2).mean()) < 35.0
    count = int(mask.sum())
    if not 29 <= count <= 33:
        raise ValueError("tile ink count is outside the supported range")
    return mask


def measure_from_image(image: Image.Image) -> tuple[str, float, int]:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    masks = [_read_tile(array, box) for box in _find_tiles(array)]
    revealed = masks[0] & masks[1]
    ranked = sorted((int(np.count_nonzero(revealed != template)), label) for label, template in TEMPLATES.items())
    best_distance, label = ranked[0]
    runner_gap = float(ranked[1][0] - best_distance)
    if best_distance > 1 or runner_gap < 8:
        raise ValueError("intersection symbol is ambiguous")
    return label, runner_gap, best_distance


def decision_from_image(image: Image.Image) -> str:
    return measure_from_image(image)[0]
