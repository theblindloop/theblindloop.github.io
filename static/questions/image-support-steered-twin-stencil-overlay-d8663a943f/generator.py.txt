"""Deterministic scene sampler and Pillow renderer for twin stencil intersection."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 512
BACKGROUND = (248, 246, 240)
DARK = (28, 31, 38)
GRID = (220, 218, 211)
DECOY = (215, 212, 203)
PAIRS = (
    ((15, 145, 185), (232, 112, 24)),
    ((122, 73, 178), (38, 153, 88)),
    ((39, 102, 190), (204, 67, 105)),
)
LABELS = ("X", "plus", "T", "L")


def _coords_to_rows(coords: Iterable[tuple[int, int]]) -> list[str]:
    points = set(coords)
    return ["".join("1" if (y, x) in points else "0" for x in range(7)) for y in range(7)]


TEMPLATES = {
    "X": _coords_to_rows({(i, i) for i in range(7)} | {(i, 6 - i) for i in range(7)}),
    "plus": _coords_to_rows({(3, i) for i in range(7)} | {(i, 3) for i in range(7)}),
    "T": _coords_to_rows({(0, i) for i in range(7)} | {(i, 3) for i in range(1, 7)}),
    "L": _coords_to_rows({(i, 0) for i in range(7)} | {(6, i) for i in range(1, 7)}),
}


def _rows_to_set(rows: list[str]) -> set[tuple[int, int]]:
    return {(y, x) for y, row in enumerate(rows) for x, value in enumerate(row) if value == "1"}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    label = LABELS[int(seed) % len(LABELS)]
    target = _rows_to_set(TEMPLATES[label])
    complement = [(y, x) for y in range(7) for x in range(7) if (y, x) not in target]
    rng.shuffle(complement)
    # The 36 non-symbol cells are split evenly: each dense tile has 31 dark
    # cells, but their intersection contains exactly the 13-cell symbol.
    tile_a = target | set(complement[:18])
    tile_b = target | set(complement[18:])
    pair = list(PAIRS[rng.randrange(len(PAIRS))])
    left_grid = rng.randrange(1, 5)  # second center is exactly two 8x8 cells away
    row_grid = rng.randrange(1, 7)
    centers = [(64 * left_grid + 32, 64 * row_grid + 32), (64 * (left_grid + 2) + 32, 64 * row_grid + 32)]
    if rng.random() < 0.5:
        tile_a, tile_b = tile_b, tile_a
    if rng.random() < 0.5:
        pair.reverse()
    decoys: list[list[int]] = []
    for _ in range(4):
        while True:
            point = [rng.randrange(30, 482), rng.randrange(30, 482)]
            if all((point[0] - cx) ** 2 + (point[1] - cy) ** 2 > 95**2 for cx, cy in centers):
                decoys.append(point)
                break
    return {
        "tile_a_rows": _coords_to_rows(tile_a),
        "tile_b_rows": _coords_to_rows(tile_b),
        "tile_a_center": list(centers[0]),
        "tile_b_center": list(centers[1]),
        "tile_a_color": list(pair[0]),
        "tile_b_color": list(pair[1]),
        "decoy_centers": decoys,
    }


def _draw_tile(draw: ImageDraw.ImageDraw, rows: list[str], center: list[int], color: list[int]) -> None:
    cx, cy = center
    x0, y0 = cx - 15, cy - 15
    draw.rectangle((x0, y0, x0 + 30, y0 + 30), fill=BACKGROUND, outline=tuple(color), width=3)
    for i in range(8):
        q = x0 + 5 + 3 * i
        draw.line((q, y0 + 5, q, y0 + 26), fill=GRID, width=1)
        q = y0 + 5 + 3 * i
        draw.line((x0 + 5, q, x0 + 26, q), fill=GRID, width=1)
    for y, row in enumerate(rows):
        for x, value in enumerate(row):
            if value == "1":
                px, py = x0 + 6 + 3 * x, y0 + 6 + 3 * y
                draw.rectangle((px, py, px + 1, py + 1), fill=DARK)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for cx, cy in scene["decoy_centers"]:
        draw.rectangle((cx - 5, cy - 5, cx + 5, cy + 5), outline=DECOY, width=2)
    _draw_tile(draw, scene["tile_a_rows"], scene["tile_a_center"], scene["tile_a_color"])
    _draw_tile(draw, scene["tile_b_rows"], scene["tile_b_center"], scene["tile_b_color"])
    return image


def _intersection_rows(a: list[str], b: list[str]) -> list[str]:
    return ["".join("1" if x == y == "1" else "0" for x, y in zip(ra, rb, strict=True)) for ra, rb in zip(a, b, strict=True)]


def _distance(a: list[str], b: list[str]) -> int:
    return sum(x != y for ra, rb in zip(a, b, strict=True) for x, y in zip(ra, rb, strict=True))


def analytic_gold(scene: dict) -> str:
    observed = _intersection_rows(scene["tile_a_rows"], scene["tile_b_rows"])
    ranked = sorted((_distance(observed, rows), label) for label, rows in TEMPLATES.items())
    return ranked[0][1]


def margin(scene: dict) -> float:
    observed = _intersection_rows(scene["tile_a_rows"], scene["tile_b_rows"])
    ranked = sorted(_distance(observed, rows) for rows in TEMPLATES.values())
    return float(ranked[1] - ranked[0])


def is_quarantined(scene: dict) -> bool:
    observed = _intersection_rows(scene["tile_a_rows"], scene["tile_b_rows"])
    ranked = sorted(_distance(observed, rows) for rows in TEMPLATES.values())
    return ranked[0] > 1 or ranked[1] - ranked[0] < 8


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = {key: (list(value) if isinstance(value, list) else value) for key, value in scene.items()}
    for suffix in ("rows", "center", "color"):
        left, right = f"tile_a_{suffix}", f"tile_b_{suffix}"
        twin[left], twin[right] = twin[right], twin[left]
    return [("swap_commutative_tiles", twin)]
