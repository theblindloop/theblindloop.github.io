"""Independent pixel-only inverse for the midpoint construction."""
from __future__ import annotations

from collections import Counter, deque
import numpy as np
from PIL import Image

FRAME_RGB = (8, 145, 178)


def _templates() -> dict[str, set[tuple[int, int]]]:
    plus = [(4, y) for y in range(1, 8)] + [(x, 4) for x in range(1, 8) if x != 4]
    ex = [(i, i) for i in range(1, 8)] + [(i, 8 - i) for i in range(1, 8) if i != 4]
    diamond = [(x, y) for y in range(1, 8) for x in range(1, 8)
               if abs(x - 4) + abs(y - 4) == 3] + [(4, 4)]
    square = ([(x, y) for y in (1, 7) for x in (1, 3, 5, 7)]
              + [(x, y) for x in (1, 7) for y in (3, 5)] + [(4, 4)])
    return {"plus": set(plus), "X": set(ex), "diamond": set(diamond), "square": set(square)}


TEMPLATES = _templates()


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))]); seen[y, x] = True
        xs, ys = [], []
        while queue:
            yy, xx = queue.popleft(); xs.append(xx); ys.append(yy)
            for dy, dx in ((-1,0),(1,0),(0,-1),(0,1)):
                ny, nx = yy + dy, xx + dx
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True; queue.append((ny, nx))
        if 80 <= len(xs) <= 500:
            found.append((float(sum(xs))/len(xs), float(sum(ys))/len(ys), len(xs)))
    return found


def _recover_points(image: Image.Image) -> set[tuple[int, int]]:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    frame_mask = np.all(arr == np.array(FRAME_RGB, dtype=np.uint8), axis=2)
    ys, xs = np.nonzero(frame_mask)
    if len(xs) < 500:
        raise ValueError("visible frame is missing")
    fx, fy, rx, by = int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())
    if abs((rx - fx) - (by - fy)) > 3:
        raise ValueError("frame is not square")
    cell = (rx - fx) / 9.0
    counts = Counter(map(tuple, arr.reshape(-1, 3).tolist()))
    groups = []
    for rgb, count in counts.items():
        if count < 160 or max(rgb) - min(rgb) < 75 or max(rgb) < 120 or rgb == FRAME_RGB:
            continue
        comps = _components(np.all(arr == np.array(rgb, dtype=np.uint8), axis=2))
        if len(comps) == 2:
            groups.append(comps)
    if len(groups) != 13:
        raise ValueError(f"expected 13 recoverable equal-color pairs, found {len(groups)}")
    points = set()
    for a, b in groups:
        mx, my = (a[0] + b[0]) / 2.0, (a[1] + b[1]) / 2.0
        gx = int(round((mx - fx) / cell - 0.5))
        gy = int(round((my - fy) / cell - 0.5))
        expected_x, expected_y = fx + (gx + 0.5) * cell, fy + (gy + 0.5) * cell
        if abs(mx - expected_x) > 2.0 or abs(my - expected_y) > 2.0:
            raise ValueError("constructed midpoint is off the visible coordinate lattice")
        points.add((gx, gy))
    if len(points) != 13:
        raise ValueError("constructed midpoints collide")
    return points


def decision_details(image: Image.Image) -> tuple[str, float]:
    points = _recover_points(image)
    scores = sorted([(len(points & template), name) for name, template in TEMPLATES.items()], reverse=True)
    gap = scores[0][0] - scores[1][0]
    if scores[0][0] != 13 or gap < 4:
        raise ValueError("ambiguous midpoint constellation")
    return scores[0][1], float(gap)


def decision_from_image(image: Image.Image) -> str:
    return decision_details(image)[0]
