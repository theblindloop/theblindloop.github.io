"""Deterministic renderer for the paired-midpoint reveal world."""
from __future__ import annotations

import copy
import random
from PIL import Image, ImageDraw


def _templates() -> dict[str, tuple[tuple[int, int], ...]]:
    plus = [(4, y) for y in range(1, 8)] + [(x, 4) for x in range(1, 8) if x != 4]
    ex = [(i, i) for i in range(1, 8)] + [(i, 8 - i) for i in range(1, 8) if i != 4]
    diamond = [(x, y) for y in range(1, 8) for x in range(1, 8)
               if abs(x - 4) + abs(y - 4) == 3] + [(4, 4)]
    square = ([(x, y) for y in (1, 7) for x in (1, 3, 5, 7)]
              + [(x, y) for x in (1, 7) for y in (3, 5)] + [(4, 4)])
    return {name: tuple(sorted(points)) for name, points in
            {"plus": plus, "X": ex, "diamond": diamond, "square": square}.items()}


TEMPLATES = _templates()
PALETTE = (
    "#d62828", "#f07100", "#e3b505", "#4d9f35", "#008f5a",
    "#0096a6", "#1475d1", "#3446c7", "#7137b8", "#a72db5",
    "#d12c83", "#c33d55", "#8a551c", "#587d16", "#176f8f",
)
FRAME = "#0891b2"


def _grid_points(scene: dict) -> set[tuple[int, int]]:
    fx, fy, cell = int(scene["frame_x"]), int(scene["frame_y"]), int(scene["cell_px"])
    points = set()
    for item in scene["pairs"]:
        ax, ay = item["a"]
        bx, by = item["b"]
        mx, my = (float(ax) + float(bx)) / 2, (float(ay) + float(by)) / 2
        gx = round((mx - fx) / cell - 0.5)
        gy = round((my - fy) / cell - 0.5)
        points.add((int(gx), int(gy)))
    return points


def _scores(scene: dict) -> list[tuple[int, str]]:
    points = _grid_points(scene)
    return sorted([(len(points & set(template)), name) for name, template in TEMPLATES.items()], reverse=True)


def _clear(center: tuple[int, int], occupied: list[tuple[int, int]], distance: int) -> bool:
    return all((center[0] - x) ** 2 + (center[1] - y) ** 2 >= distance ** 2 for x, y in occupied)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    symbol = tuple(TEMPLATES)[int(seed) % len(TEMPLATES)]
    cell = rng.choice((52, 54, 56))
    fx, fy = rng.randint(54, 76), rng.randint(54, 76)
    radius = rng.choice((8, 9))
    colors = list(PALETTE)
    rng.shuffle(colors)
    occupied: list[tuple[int, int]] = []
    pairs = []
    # Every endpoint stays within 15 px of its midpoint. Since logical midpoint
    # cells are at least 52 px apart, different pair neighborhoods cannot touch.
    vector_pool = [(14, 0), (0, 14), (10, 10), (10, -10)]
    rng.shuffle(vector_pool)
    for index, (gx, gy) in enumerate(TEMPLATES[symbol]):
        mx = fx + int((gx + 0.5) * cell)
        my = fy + int((gy + 0.5) * cell)
        choices = list(vector_pool)
        rng.shuffle(choices)
        chosen = None
        for dx, dy in choices:
            a, b = (mx - dx, my - dy), (mx + dx, my + dy)
            if (_clear(a, occupied, 2 * radius + 3) and _clear(b, occupied, 2 * radius + 3)
                    and fx + 12 < a[0] < fx + 9 * cell - 12
                    and fy + 12 < a[1] < fy + 9 * cell - 12
                    and fx + 12 < b[0] < fx + 9 * cell - 12
                    and fy + 12 < b[1] < fy + 9 * cell - 12):
                chosen = (a, b)
                break
        if chosen is None:
            raise RuntimeError(f"could not place endpoint pair for seed {seed}")
        occupied.extend(chosen)
        pairs.append({"a": list(chosen[0]), "b": list(chosen[1]), "color": colors[index]})
    rng.shuffle(pairs)
    distractors = []
    for _ in range(9):
        for _attempt in range(500):
            point = (rng.randint(fx + 18, fx + 9 * cell - 18),
                     rng.randint(fy + 18, fy + 9 * cell - 18))
            if _clear(point, occupied + distractors, 2 * radius + 3):
                distractors.append(point)
                break
    return {
        "pairs": pairs, "distractors": [list(p) for p in distractors],
        "frame_x": fx, "frame_y": fy, "cell_px": cell, "radius": radius,
        "background": rng.choice(("#ffffff", "#fffdf8", "#f8fafc")),
    }


def render(scene: dict) -> Image.Image:
    pairs = scene["pairs"]
    distractors = scene["distractors"]
    fx, fy = int(scene["frame_x"]), int(scene["frame_y"])
    cell, radius = int(scene["cell_px"]), int(scene["radius"])
    background = scene["background"]
    image = Image.new("RGB", (620, 620), background)
    draw = ImageDraw.Draw(image)
    draw.rectangle((fx, fy, fx + 9 * cell, fy + 9 * cell), outline=FRAME, width=5)
    for x, y in distractors:
        draw.ellipse((x-radius-2, y-radius-2, x+radius+2, y+radius+2), fill="#263142")
        draw.ellipse((x-radius, y-radius, x+radius, y+radius), fill="#8b949e")
    for item in pairs:
        for x, y in (item["a"], item["b"]):
            draw.ellipse((x-radius-2, y-radius-2, x+radius+2, y+radius+2), fill="#263142")
            draw.ellipse((x-radius, y-radius, x+radius, y+radius), fill=item["color"])
    return image


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    if scores[0][0] != 13:
        raise ValueError("midpoints do not form a declared constellation")
    return scores[0][1]


def margin(scene: dict) -> float:
    scores = _scores(scene)
    return float(scores[0][0] - scores[1][0])


def is_quarantined(scene: dict) -> bool:
    pairs = scene.get("pairs", [])
    colors = [item.get("color") for item in pairs]
    valid = len(pairs) == 13 and len(set(colors)) == 13
    valid &= len(_grid_points(scene)) == 13
    return not (valid and margin(scene) >= 4)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["pairs"] = list(reversed(twin["pairs"]))
    for item in twin["pairs"]:
        item["a"], item["b"] = item["b"], item["a"]
    return [("reverse_pairs_and_swap_endpoints", twin)]
