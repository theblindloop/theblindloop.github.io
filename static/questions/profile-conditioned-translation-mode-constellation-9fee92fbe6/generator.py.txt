"""Deterministic sampler, analytic displacement-mode arm, and RGB renderer."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

COLOR_NAMES = ("red", "blue", "green", "orange")
GUIDE_RGB = {
    "red": (211, 55, 48),
    "blue": (42, 104, 177),
    "green": (34, 139, 78),
    "orange": (225, 126, 35),
}
ANGLE_TOL_DEG = 2.5
LENGTH_TOL_PX = 3.0
MIN_PAIR_LENGTH_PX = 35.0


def _angle_distance(a: float, b: float) -> float:
    delta = abs(float(a) - float(b)) % 180.0
    return min(delta, 180.0 - delta)


def _support(points: list[list[int]], guide_angle: float) -> int:
    """Largest equal-length cluster among point pairs parallel to one guide."""
    lengths: list[float] = []
    for index, first in enumerate(points):
        for second in points[index + 1 :]:
            dx = float(second[0] - first[0])
            dy = float(second[1] - first[1])
            distance = math.hypot(dx, dy)
            if distance < MIN_PAIR_LENGTH_PX:
                continue
            angle = math.degrees(math.atan2(dy, dx)) % 180.0
            if _angle_distance(angle, guide_angle) <= ANGLE_TOL_DEG:
                lengths.append(distance)
    if not lengths:
        return 0
    return max(sum(abs(other - center) <= LENGTH_TOL_PX for other in lengths) for center in lengths)


def _score_by_color(scene: dict) -> dict[str, int]:
    points = scene["points"]
    guides = scene["guides"]
    return {guide["color"]: _support(points, float(guide["angle"])) for guide in guides}


def _well_separated(candidate: tuple[int, int], points: list[list[int]], floor: float) -> bool:
    return all(math.hypot(candidate[0] - x, candidate[1] - y) >= floor for x, y in points)


def sample_scene(seed: int) -> dict:
    """Sample interleaved translated copies, distractors, and balanced guide identities."""
    rng = random.Random(int(seed))
    answer_color = COLOR_NAMES[int(seed) % len(COLOR_NAMES)]
    width, height, panel_bottom = 640, 560, 405

    for _attempt in range(600):
        raw_angle = rng.uniform(6.0, 174.0)
        raw_length = rng.randint(106, 128)
        dx = int(round(raw_length * math.cos(math.radians(raw_angle))))
        dy = int(round(raw_length * math.sin(math.radians(raw_angle))))
        if math.hypot(dx, dy) < 100:
            continue
        true_angle = math.degrees(math.atan2(dy, dx)) % 180.0
        motif_count = rng.choice((8, 9, 10))
        distractor_count = rng.choice((4, 5, 6))

        x_low = max(48, 48 - dx)
        x_high = min(width - 49, width - 49 - dx)
        y_low = max(42, 42 - dy)
        y_high = min(panel_bottom - 35, panel_bottom - 35 - dy)
        if x_low >= x_high or y_low >= y_high:
            continue

        base: list[list[int]] = []
        all_points: list[list[int]] = []
        for _ in range(6000):
            if len(base) >= motif_count:
                break
            point = (rng.randint(x_low, x_high), rng.randint(y_low, y_high))
            twin = (point[0] + dx, point[1] + dy)
            if not _well_separated(point, all_points, 25.0):
                continue
            if not _well_separated(twin, all_points, 25.0):
                continue
            if math.hypot(point[0] - twin[0], point[1] - twin[1]) < 25.0:
                continue
            base.append([point[0], point[1]])
            all_points.extend(([point[0], point[1]], [twin[0], twin[1]]))
        if len(base) != motif_count:
            continue

        for _ in range(6000):
            if len(all_points) >= motif_count * 2 + distractor_count:
                break
            candidate = (rng.randint(48, width - 49), rng.randint(42, panel_bottom - 35))
            if _well_separated(candidate, all_points, 25.0):
                all_points.append([candidate[0], candidate[1]])
        if len(all_points) != motif_count * 2 + distractor_count:
            continue

        remaining = [color for color in COLOR_NAMES if color != answer_color]
        rng.shuffle(remaining)
        offsets = [rng.uniform(29, 41), rng.uniform(65, 82), rng.uniform(108, 132)]
        rng.shuffle(offsets)
        guides = [{"color": answer_color, "angle": true_angle}]
        guides.extend(
            {"color": color, "angle": (true_angle + offset) % 180.0}
            for color, offset in zip(remaining, offsets, strict=True)
        )
        rng.shuffle(guides)
        rng.shuffle(all_points)
        scene = {
            "width": width,
            "height": height,
            "panel_bottom": panel_bottom,
            "points": all_points,
            "radius": rng.choice((7, 8)),
            "guides": guides,
            "guide_length": 86,
            "guide_width": 6,
            "background": [248, 247, 242],
            "dot_color": [27, 32, 43],
            "frame_color": [178, 180, 178],
        }
        scores = _score_by_color(scene)
        ordered = sorted(scores, key=lambda color: (-scores[color], color))
        if ordered[0] == answer_color and scores[ordered[0]] >= motif_count and (
            scores[ordered[0]] - scores[ordered[1]] >= 4
        ):
            return scene
    raise RuntimeError("could not sample a uniquely supported translation mode")


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    panel_bottom = int(scene["panel_bottom"])
    points = scene["points"]
    radius = int(scene["radius"])
    guides = scene["guides"]
    guide_length = int(scene["guide_length"])
    guide_width = int(scene["guide_width"])
    background = tuple(scene["background"])
    dot_color = tuple(scene["dot_color"])
    frame_color = tuple(scene["frame_color"])

    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)
    draw.rectangle((20, 20, width - 21, panel_bottom - 10), outline=frame_color, width=2)
    draw.line((20, panel_bottom + 2, width - 21, panel_bottom + 2), fill=frame_color, width=2)
    for x, y in points:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=dot_color)

    cell_width = width / 4.0
    guide_y = panel_bottom + (height - panel_bottom) / 2.0 + 4.0
    for index, guide in enumerate(guides):
        cx = (index + 0.5) * cell_width
        angle = math.radians(float(guide["angle"]))
        ux = math.cos(angle) * guide_length / 2.0
        uy = math.sin(angle) * guide_length / 2.0
        endpoints = [(cx - ux, guide_y - uy), (cx + ux, guide_y + uy)]
        color = GUIDE_RGB[guide["color"]]
        draw.line(endpoints, fill=color, width=guide_width)
        endpoint_radius = guide_width
        for ex, ey in endpoints:
            draw.ellipse(
                (ex - endpoint_radius, ey - endpoint_radius, ex + endpoint_radius, ey + endpoint_radius),
                outline=color,
                width=2,
            )
    return image


def analytic_gold(scene: dict) -> str:
    scores = _score_by_color(scene)
    return min(scores, key=lambda color: (-scores[color], color))


def margin(scene: dict) -> float:
    ordered = sorted(_score_by_color(scene).values(), reverse=True)
    return float(ordered[0] - ordered[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 3.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["points"] = list(reversed(twin["points"]))
    return [("reverse_point_order", twin)]
