from __future__ import annotations

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        points = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        output.append(np.asarray(points, dtype=np.float64))
    return output


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    r, g, b = array[..., 0], array[..., 1], array[..., 2]
    gold = (r > 175) & (g > 105) & (g < 220) & (b < 100) & ((r - b) > 80)
    gold_parts = [part for part in _components(gold) if len(part) >= 12]
    if not gold_parts:
        raise ValueError("center marker missing")
    marker = max(gold_parts, key=len)
    cy, cx = marker.mean(axis=0)
    chroma = np.max(array, axis=2) - np.min(array, axis=2)
    specimen = (chroma > 35) & ~gold
    parts = [part for part in _components(specimen) if len(part) >= 300]
    if len(parts) != 1:
        raise ValueError("connected specimen missing or corrupted")
    points = parts[0]
    if len(points) < 5000:
        raise ValueError("insufficient specimen area")
    # The rendered specimen is centrally symmetric about the visible marker.
    # Verify that observable invariant before trusting a moment measured from a
    # potentially erased or cropped region. This uses only the raster mask.
    iy, ix = np.rint([cy, cx]).astype(int)
    ys = points[:, 0].astype(int)
    xs = points[:, 1].astype(int)
    rys = 2 * iy - ys
    rxs = 2 * ix - xs
    valid = (rys >= 0) & (rys < specimen.shape[0]) & (rxs >= 0) & (rxs < specimen.shape[1])
    matched = np.zeros(len(points), dtype=bool)
    matched[valid] = specimen[rys[valid], rxs[valid]]
    if float(np.mean(~matched)) > 0.012:
        raise ValueError("specimen symmetry corrupted")
    mx = float(np.square(points[:, 1] - cx).sum())
    my = float(np.square(points[:, 0] - cy).sum())
    if abs(mx - my) < 500_000:
        raise ValueError("raster moment ambiguity")
    return "horizontal" if mx > my else "vertical"
