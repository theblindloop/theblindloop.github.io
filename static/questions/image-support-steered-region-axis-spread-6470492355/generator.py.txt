from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = 384
BACKGROUND = (248, 247, 242)
GOLD = (242, 181, 48)
PALETTE = [(24, 151, 156), (102, 92, 190), (219, 91, 106), (47, 122, 184)]
QUARANTINE_MOMENT = 2_000_000.0


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    horizontal_wins = bool(seed % 2)
    major = rng.randint(76, 92)
    minor = rng.randint(45, 60)
    h_reach, v_reach = (major, minor) if horizontal_wins else (minor, major)
    h_thick = rng.randint(31, 42)
    v_thick = rng.randint(31, 42)
    return {
        "center_x": rng.randint(176, 208),
        "center_y": rng.randint(176, 208),
        "horizontal_reach": h_reach,
        "vertical_reach": v_reach,
        "horizontal_half_thickness": h_thick,
        "vertical_half_thickness": v_thick,
        "color": list(PALETTE[rng.randrange(len(PALETTE))]),
        "canvas_width": CANVAS,
        "canvas_height": CANVAS,
    }


def _rectangles(scene: dict) -> tuple[tuple[int, int, int, int], tuple[int, int, int, int]]:
    cx, cy = scene["center_x"], scene["center_y"]
    hr, vr = scene["horizontal_reach"], scene["vertical_reach"]
    ht, vt = scene["horizontal_half_thickness"], scene["vertical_half_thickness"]
    return (cx - hr, cy - ht, cx + hr, cy + ht), (cx - vt, cy - vr, cx + vt, cy + vr)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (scene["canvas_width"], scene["canvas_height"]), BACKGROUND)
    draw = ImageDraw.Draw(image)
    color = tuple(scene["color"])
    horizontal, vertical = _rectangles(scene)
    draw.rounded_rectangle(horizontal, radius=18, fill=color)
    draw.rounded_rectangle(vertical, radius=18, fill=color)
    cx, cy = scene["center_x"], scene["center_y"]
    draw.ellipse((cx - 7, cy - 7, cx + 7, cy + 7), fill=GOLD, outline=(91, 66, 22), width=2)
    return image


def _axis_score(scene: dict) -> float:
    """Signed rectangular-union moment model; rounded caps preserve its sign."""
    hr = scene["horizontal_reach"]
    vr = scene["vertical_reach"]
    ht = scene["horizontal_half_thickness"]
    vt = scene["vertical_half_thickness"]
    sum_hr2 = hr * (hr + 1) * (2 * hr + 1) / 3.0
    sum_vr2 = vr * (vr + 1) * (2 * vr + 1) / 3.0
    sum_ht2 = ht * (ht + 1) * (2 * ht + 1) / 3.0
    sum_vt2 = vt * (vt + 1) * (2 * vt + 1) / 3.0
    mx = (2 * ht + 1) * sum_hr2 + (2 * vr + 1) * sum_vt2 - (2 * ht + 1) * sum_vt2
    my = (2 * hr + 1) * sum_ht2 + (2 * vt + 1) * sum_vr2 - (2 * vt + 1) * sum_ht2
    return mx - my


def analytic_gold(scene: dict) -> str:
    return "horizontal" if _axis_score(scene) > 0 else "vertical"


def margin(scene: dict) -> float:
    return abs(_axis_score(scene))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MOMENT


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
