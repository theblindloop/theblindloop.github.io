"""Renderer and analytic arm for paired cross-panel dot-range contraction."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 760, 410
PANEL_X = (40, 410)
PANEL_W = 310
LOCAL_MIN, LOCAL_MAX = 25.0, 285.0
DOT_RADIUS = 7
PALETTE = {
    "blue": (43, 119, 188),
    "orange": (230, 116, 34),
    "green": (42, 148, 76),
}


def _five_positions(rng: random.Random, extent: float) -> list[float]:
    left = rng.uniform(LOCAL_MIN, LOCAL_MAX - extent)
    fractions = [0.0, rng.uniform(0.20, 0.31), rng.uniform(0.43, 0.57), rng.uniform(0.69, 0.80), 1.0]
    return [round(left + extent * fraction, 3) for fraction in fractions]


def sample_scene(seed: int) -> dict:
    """Create a balanced, well-separated scene from one deterministic seed."""
    rng = random.Random(int(seed))
    colors = ["blue", "orange", "green"]
    winner = int(seed) % len(colors)
    others = [index for index in range(3) if index != winner]
    rng.shuffle(others)
    contractions = [0.0, 0.0, 0.0]
    contractions[winner] = rng.uniform(72.0, 92.0)
    contractions[others[0]] = rng.uniform(20.0, 42.0)
    contractions[others[1]] = rng.uniform(-28.0, 8.0)

    x_a: list[list[float]] = []
    x_b: list[list[float]] = []
    for contraction in contractions:
        range_a = rng.uniform(175.0, 225.0)
        range_b = range_a - contraction
        x_a.append(_five_positions(rng, range_a))
        x_b.append(_five_positions(rng, range_b))

    y_rows = [125.0, 220.0, 315.0]
    rng.shuffle(y_rows)
    return {"colors": colors, "x_a": x_a, "x_b": x_b, "y_rows": y_rows}


def _scores(scene: dict) -> dict[str, float]:
    colors = scene["colors"]
    x_a = scene["x_a"]
    x_b = scene["x_b"]
    return {
        color: (max(a) - min(a)) - (max(b) - min(b))
        for color, a, b in zip(colors, x_a, x_b)
    }


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    return max(scores, key=scores.get)


def margin(scene: dict) -> float:
    values = sorted(_scores(scene).values(), reverse=True)
    return float(values[0] - values[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = {
        "colors": list(scene["colors"]),
        "x_a": [list(reversed(row)) for row in scene["x_a"]],
        "x_b": [list(reversed(row)) for row in scene["x_b"]],
        "y_rows": list(scene["y_rows"]),
    }
    return [("reverse_coordinate_storage_order", twin)]


def _font(size: int) -> ImageFont.ImageFont:
    try:
        return ImageFont.truetype("DejaVuSans.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _draw_dots(draw: ImageDraw.ImageDraw, panel_left: int, xs: Iterable[float], y: float, color: tuple[int, int, int], scale: int) -> None:
    radius = DOT_RADIUS * scale
    for local_x in xs:
        cx = (panel_left + local_x) * scale
        cy = y * scale
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=color, outline=(25, 31, 38), width=scale)


def render(scene: dict) -> Image.Image:
    colors = scene["colors"]
    x_a = scene["x_a"]
    x_b = scene["x_b"]
    y_rows = scene["y_rows"]
    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), (250, 250, 247))
    draw = ImageDraw.Draw(image)
    title_font = _font(20 * scale)
    panel_font = _font(18 * scale)
    small_font = _font(13 * scale)
    draw.text((WIDTH * scale / 2, 24 * scale), "Horizontal ranges in two panels", fill=(30, 34, 40), anchor="mm", font=title_font)
    for label, left in zip(("A", "B"), PANEL_X):
        box = (left * scale, 72 * scale, (left + PANEL_W) * scale, 365 * scale)
        draw.rounded_rectangle(box, radius=8 * scale, fill=(255, 255, 255), outline=(116, 122, 130), width=scale)
        draw.text(((left + PANEL_W / 2) * scale, 54 * scale), f"Panel {label}", fill=(34, 38, 44), anchor="mm", font=panel_font)
        for y in y_rows:
            draw.line(((left + 15) * scale, y * scale, (left + PANEL_W - 15) * scale, y * scale), fill=(224, 226, 228), width=scale)
    for index, color_name in enumerate(colors):
        color = PALETTE[color_name]
        y = y_rows[index]
        _draw_dots(draw, PANEL_X[0], x_a[index], y, color, scale)
        _draw_dots(draw, PANEL_X[1], x_b[index], y, color, scale)
    draw.text((WIDTH * scale / 2, 391 * scale), "Range = rightmost dot − leftmost dot", fill=(78, 82, 88), anchor="mm", font=small_font)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
