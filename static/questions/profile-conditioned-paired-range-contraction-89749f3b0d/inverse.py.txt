"""Independent pixel-only inverse arm."""

from __future__ import annotations

import numpy as np
from PIL import Image

COLOR_RGB = {
    "blue": np.array((43, 119, 188), dtype=np.int16),
    "orange": np.array((230, 116, 34), dtype=np.int16),
    "green": np.array((42, 148, 76), dtype=np.int16),
}


def _range_for(mask: np.ndarray, left: int, right: int) -> float:
    ys, xs = np.nonzero(mask[:, left:right])
    if xs.size < 100:
        raise ValueError("insufficient colored evidence")
    occupied = np.flatnonzero(np.bincount(xs, minlength=right - left) >= 3)
    if occupied.size < 10:
        raise ValueError("colored dot extrema are not recoverable")
    return float(occupied[-1] - occupied[0])


def decision_from_image(image: Image.Image) -> str:
    # int32 prevents overflow when three squared channel deltas are summed.
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    if arr.shape[1] < 600 or arr.shape[0] < 300:
        raise ValueError("unexpected raster dimensions")
    scores: dict[str, float] = {}
    split = arr.shape[1] // 2
    for name, rgb in COLOR_RGB.items():
        distance = np.sqrt(np.sum((arr - rgb.astype(np.int32)) ** 2, axis=2))
        mask = distance < 45.0
        range_a = _range_for(mask, 0, split)
        range_b = _range_for(mask, split, arr.shape[1])
        scores[name] = range_a - range_b
    ordered = sorted(scores, key=scores.get, reverse=True)
    if scores[ordered[0]] - scores[ordered[1]] < 8.0:
        raise ValueError("ambiguous winner margin")
    return ordered[0]
