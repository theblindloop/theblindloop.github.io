"""Deterministic renderer and analytic arm for nested contour panels."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw


SUPER = 4
BACKGROUND = (248, 247, 243)
COLORS = {
    "red": (217, 75, 72),
    "blue": (61, 115, 199),
}
DECOR_INK = (186, 184, 178)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample two equal-scale, widely separated contour stacks.

    Consecutive scene seeds alternate the winning color so the public evidence
    remains balanced for every generation seed while all geometric nuisance
    fields continue to vary independently.
    """

    rng = random.Random(int(seed))
    low = rng.choice((2, 3))
    gap_in_layers = rng.choice((1, 2))
    high = min(5, low + gap_in_layers)
    red_wins = int(seed) % 2 == 0
    if red_wins:
        red_depth, blue_depth = high, low
    else:
        red_depth, blue_depth = low, high

    return {
        "width": 512,
        "height": 512,
        "red_depth": int(red_depth),
        "blue_depth": int(blue_depth),
        "red_center": [86 + rng.randint(-6, 6), 208 + rng.randint(-16, 16)],
        "blue_center": [426 + rng.randint(-6, 6), 304 + rng.randint(-16, 16)],
        "panel_radius": rng.choice((68, 72, 76)),
        "ring_gap": rng.choice((8, 9)),
        "stroke_width": rng.choice((4, 5)),
        "color_order": list(rng.choice((("red", "blue"), ("blue", "red")))),
        "background": list(BACKGROUND),
        "decor_seed": rng.randrange(1, 2**31 - 1),
        "translation": [rng.randint(-7, 7), rng.randint(-12, 12)],
        "scale": rng.choice((0.96, 1.0, 1.04)),
    }


def _scaled_point(scene: dict[str, Any], center: list[int], x: float, y: float) -> tuple[int, int]:
    scale = float(scene["scale"])
    translation = scene["translation"]
    px = (float(center[0]) + float(translation[0]) + x * scale) * SUPER
    py = (float(center[1]) + float(translation[1]) + y * scale) * SUPER
    return int(round(px)), int(round(py))


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    scene: dict[str, Any],
    center: list[int],
    depth: int,
    color_name: str,
) -> None:
    radius = float(scene["panel_radius"]) * float(scene["scale"])
    layer_gap = float(scene["ring_gap"]) * float(scene["scale"])
    width = max(1, int(round(float(scene["stroke_width"]) * SUPER)))
    ink = COLORS[color_name]
    for layer in range(int(depth)):
        ring_radius = radius - layer * (layer_gap + float(scene["stroke_width"]) * float(scene["scale"]))
        if ring_radius <= 7.0:
            raise ValueError("sampled contour stack has insufficient inner clearance")
        cx, cy = _scaled_point(scene, center, 0.0, 0.0)
        delta = int(round(ring_radius * SUPER))
        draw.ellipse((cx - delta, cy - delta, cx + delta, cy + delta), outline=ink, width=width)


def _draw_decorations(draw: ImageDraw.ImageDraw, scene: dict[str, Any]) -> None:
    """Add identical-kind, neutral remote marks that do not encode the answer."""

    rng = random.Random(int(scene["decor_seed"]))
    width = int(scene["width"]) * SUPER
    height = int(scene["height"]) * SUPER
    for base_x, base_y in ((42, 44), (470, 48), (48, 468), (464, 462)):
        x = (base_x + rng.randint(-9, 9)) * SUPER
        y = (base_y + rng.randint(-9, 9)) * SUPER
        radius = rng.choice((2, 3)) * SUPER
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=DECOR_INK)
    for start_x, start_y, length in ((18, 150, 34), (460, 344, 34)):
        offset = rng.randint(-5, 5)
        y = (start_y + offset) * SUPER
        x0 = (start_x + offset) * SUPER
        x1 = min(width - SUPER, x0 + length * SUPER)
        draw.line((x0, y, x1, y), fill=DECOR_INK, width=SUPER)
    # Use the height field in the drawing path so every declared nuisance field
    # is part of the rendered contract.
    if height < 32:
        raise ValueError("canvas is too small")


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the final RGB PNG image at high resolution, then downsample."""

    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(int(value) for value in scene["background"])
    image = Image.new("RGB", (width * SUPER, height * SUPER), background)
    draw = ImageDraw.Draw(image)
    _draw_decorations(draw, scene)

    # Keep label fields as explicit accesses: the rendered-field audit can see
    # that both analytic inputs are visibly represented.
    red_depth = int(scene["red_depth"])
    blue_depth = int(scene["blue_depth"])
    red_center = list(scene["red_center"])
    blue_center = list(scene["blue_center"])
    color_order = list(scene["color_order"])
    centers = {"red": red_center, "blue": blue_center}
    depths = {"red": red_depth, "blue": blue_depth}
    for color_name in color_order:
        if color_name not in COLORS:
            raise ValueError(f"unknown panel color {color_name!r}")
        _draw_panel(draw, scene, centers[color_name], depths[color_name], color_name)

    return image.resize((width, height), Image.Resampling.LANCZOS).convert("RGB")


def analytic_gold(scene: dict[str, Any]) -> str:
    """Select the color whose visible contour stack is deeper."""

    red_depth = int(scene["red_depth"])
    blue_depth = int(scene["blue_depth"])
    if red_depth == blue_depth:
        raise ValueError("nested contour duel is tied")
    return "red" if red_depth > blue_depth else "blue"


def margin(scene: dict[str, Any]) -> float:
    """Report the physical ring-count gap in rendered pixels."""

    red_depth = int(scene["red_depth"])
    blue_depth = int(scene["blue_depth"])
    return round(abs(red_depth - blue_depth) * float(scene["ring_gap"]) * float(scene["scale"]), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Exclude ties and contour stacks that approach the raster boundary."""

    red_depth = int(scene["red_depth"])
    blue_depth = int(scene["blue_depth"])
    smallest_inner = min(red_depth, blue_depth)
    inner_radius = float(scene["panel_radius"]) - (smallest_inner - 1) * (
        float(scene["ring_gap"]) + float(scene["stroke_width"])
    )
    return (
        red_depth == blue_depth
        or float(scene["ring_gap"]) * float(scene["scale"]) < 8.0
        or inner_radius < 7.0
    )


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No intentionally pixel-identical latent aliases are declared."""

    del scene
    return []
