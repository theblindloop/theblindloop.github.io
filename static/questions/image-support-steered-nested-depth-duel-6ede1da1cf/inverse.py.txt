"""Independent pixel-only oracle for the nested contour duel."""

from __future__ import annotations

import numpy as np
from PIL import Image


PALETTE = {
    "red": np.asarray([217, 75, 72], dtype=np.float32),
    "blue": np.asarray([61, 115, 199], dtype=np.float32),
}


def _components(mask: np.ndarray, minimum: int = 60) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    found: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y = int(y)
        x = int(x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    ny, nx = cy + dy, cx + dx
                    if (
                        0 <= ny < height
                        and 0 <= nx < width
                        and mask[ny, nx]
                        and not seen[ny, nx]
                    ):
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        if len(points) >= minimum:
            found.append(np.asarray(points, dtype=np.int32))
    return found


def _color_mask(array: np.ndarray, color: str) -> np.ndarray:
    target = PALETTE[color]
    distances = np.linalg.norm(array.astype(np.float32) - target, axis=2)
    # The saturated core of each supersampled stroke is comfortably inside this
    # radius; pale background and gray decoration are far outside it.
    return distances <= 82.0


def _ring_count(array: np.ndarray, color: str) -> int:
    components = _components(_color_mask(array, color))
    if not 2 <= len(components) <= 5:
        raise ValueError(f"expected 2-5 {color} contour components, found {len(components)}")

    centers = []
    radii = []
    for component in components:
        y0, x0 = component.min(axis=0)
        y1, x1 = component.max(axis=0)
        component_center = np.asarray([(y0 + y1) / 2.0, (x0 + x1) / 2.0], dtype=np.float32)
        centers.append(component_center)
        radii.append(float((y1 - y0 + x1 - x0) / 4.0))
        # A destructive raster ablation may leave a long open arc that still
        # looks like a plausible component.  Count only components with ink in
        # every angular sector around their fitted center, which is an image-
        # only closure test for a complete visible ring.
        offsets = component.astype(np.float32) - component_center
        angles = (np.arctan2(offsets[:, 0], offsets[:, 1]) + np.pi) / (2.0 * np.pi)
        bins = np.floor(angles * 16.0).astype(np.int32)
        if len(np.unique(bins)) < 16:
            raise ValueError(f"{color} component is an open contour fragment")

    center_array = np.asarray(centers, dtype=np.float32)
    center_reference = np.median(center_array, axis=0)
    if float(np.linalg.norm(center_array - center_reference, axis=1).max()) > 14.0:
        raise ValueError(f"{color} contours do not share a stable center")

    ordered_radii = np.sort(np.asarray(radii, dtype=np.float32))
    if float(ordered_radii[0]) < 8.0 or float(ordered_radii[-1]) > 110.0:
        raise ValueError(f"{color} contour radius is outside the stable range")
    if float(np.diff(ordered_radii).min()) < 7.0:
        raise ValueError(f"{color} contours have an ambiguous or split layer")
    return len(components)


def decision_from_image(image: Image.Image) -> str:
    """Return the color with more visible nested contour layers."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    red_count = _ring_count(array, "red")
    blue_count = _ring_count(array, "blue")
    if red_count == blue_count:
        raise ValueError("contour stacks are tied or indistinguishable")
    return "red" if red_count > blue_count else "blue"
