"""Deterministic latent scene sampler and PIL renderer."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw


def _signed_excursions(rng: random.Random, large: bool) -> list[int]:
    if large:
        magnitudes = [rng.randint(29, 38) for _ in range(3)]
    else:
        magnitudes = [rng.randint(12, 20) for _ in range(3)]
    steps = [value for magnitude in magnitudes for value in (magnitude, -magnitude)]
    rng.shuffle(steps)
    # The half always returns to its starting height, removing endpoint shortcuts.
    return steps


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    width, height = 768, 480
    divider_x = rng.randint(370, 398)
    x0, x12 = rng.randint(46, 64), rng.randint(704, 722)
    left_x = [round(x0 + (divider_x - x0) * i / 6) for i in range(7)]
    right_x = [round(divider_x + (x12 - divider_x) * i / 6) for i in range(1, 7)]
    target_left = int(seed) % 2 == 0
    left_steps = _signed_excursions(rng, target_left)
    right_steps = _signed_excursions(rng, not target_left)
    y = rng.randint(218, 262)
    points: list[list[int]] = [[left_x[0], y]]
    for x, dy in zip(left_x[1:], left_steps):
        y += dy
        points.append([x, y])
    for x, dy in zip(right_x, right_steps):
        y += dy
        points.append([x, y])
    return {
        "points": points,
        "divider_x": divider_x,
        "width": width,
        "height": height,
        "stroke_width": rng.choice([6, 7, 8]),
        "route_color": [18, 157, 187],
        "divider_color": [224, 128, 38],
        "background": [250, 250, 247],
    }


def _totals(scene: dict) -> tuple[float, float]:
    points = scene["points"]
    divider_x = float(scene["divider_x"])
    left = right = 0.0
    for (x0, y0), (x1, y1) in zip(points, points[1:]):
        vertical = abs(float(y1) - float(y0))
        midpoint = (float(x0) + float(x1)) / 2.0
        if midpoint < divider_x:
            left += vertical
        else:
            right += vertical
    return left, right


def analytic_gold(scene: dict) -> str:
    left, right = _totals(scene)
    return "left" if left > right else "right"


def margin(scene: dict) -> float:
    left, right = _totals(scene)
    return abs(left - right)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 36.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(scene["background"])
    route_color = tuple(scene["route_color"])
    divider_color = tuple(scene["divider_color"])
    points = [(int(x), int(y)) for x, y in scene["points"]]
    divider_x = int(scene["divider_x"])
    stroke_width = int(scene["stroke_width"])
    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    # A visible measurement frame gives context but carries no decision value.
    frame = (118, 125, 128)
    draw.line([(28 * scale, 52 * scale), (28 * scale, 428 * scale)], fill=frame, width=2 * scale)
    draw.line([(28 * scale, 428 * scale), ((width - 28) * scale, 428 * scale)], fill=frame, width=2 * scale)
    for y in range(72, 409, 24):
        draw.line(
            [(divider_x * scale, y * scale), (divider_x * scale, min(y + 12, 420) * scale)],
            fill=divider_color,
            width=3 * scale,
        )
    draw.line(
        [(x * scale, y * scale) for x, y in points],
        fill=route_color,
        width=stroke_width * scale,
        joint="curve",
    )
    radius = max(3, stroke_width // 2) * scale
    for x, y in points:
        draw.ellipse(
            (x * scale - radius, y * scale - radius, x * scale + radius, y * scale + radius),
            fill=route_color,
        )
    return image.resize((width, height), Image.Resampling.LANCZOS)
