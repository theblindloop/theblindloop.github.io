"""Independent pixel-only inverse arm for route variation."""

from __future__ import annotations

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    pass


def _masks(image: Image.Image) -> tuple[np.ndarray, np.ndarray]:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[:, :, 0], array[:, :, 1], array[:, :, 2]
    cyan = (red < 105) & (green > 105) & (blue > 130) & ((blue - red) > 70)
    orange = (red > 160) & (green > 65) & (green < 185) & (blue < 105) & ((red - blue) > 100)
    return cyan, orange


def _measure(image: Image.Image) -> tuple[float, float]:
    cyan, orange = _masks(image)
    height, width = cyan.shape
    if int(cyan.sum()) < 1800:
        raise OracleAbstention("too little cyan route ink")
    orange_y, orange_x = np.nonzero(orange)
    if len(orange_x) < 120 or np.ptp(orange_y) < height * 0.45:
        raise OracleAbstention("visible dashed divider is missing")
    divider_x = float(np.median(orange_x))
    xs = np.flatnonzero(cyan.any(axis=0))
    if len(xs) < width * 0.76 or (xs[-1] - xs[0]) < width * 0.80:
        raise OracleAbstention("route does not span the image")
    if xs[0] > width * 0.11 or xs[-1] < width * 0.89:
        raise OracleAbstention("route endpoint evidence is missing")
    gaps = np.diff(xs)
    if len(gaps) and int(gaps.max()) > max(9, width // 70):
        raise OracleAbstention("route is visibly broken")

    full_x = np.arange(int(xs[0]), int(xs[-1]) + 1)
    samples_x: list[int] = []
    samples_y: list[float] = []
    for x in xs:
        ys = np.flatnonzero(cyan[:, x])
        if len(ys):
            samples_x.append(int(x))
            samples_y.append(float(np.median(ys)))
    center_y = np.interp(full_x, np.asarray(samples_x), np.asarray(samples_y))
    # Light smoothing suppresses antialias stair-steps without erasing real excursions.
    kernel = np.ones(7, dtype=np.float64) / 7.0
    smooth_y = np.convolve(np.pad(center_y, (3, 3), mode="edge"), kernel, mode="valid")
    dy = np.abs(np.diff(smooth_y))
    seg_mid_x = (full_x[:-1] + full_x[1:]) / 2.0
    left = float(dy[seg_mid_x < divider_x].sum())
    right = float(dy[seg_mid_x >= divider_x].sum())
    if abs(left - right) < 18.0:
        raise OracleAbstention("pixel decision is too close")
    return left, right


def decision_from_image(image: Image.Image) -> str:
    left, right = _measure(image)
    return "left" if left > right else "right"
