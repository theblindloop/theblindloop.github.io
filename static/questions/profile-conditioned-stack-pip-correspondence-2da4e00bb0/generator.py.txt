"""Deterministic renderer and analytic arm for stack-to-pip correspondence."""

from __future__ import annotations

import copy
import itertools
import math
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 960, 720
SCALE = 2
BACKGROUND = (247, 248, 250)
CARD_FILL = (255, 255, 252)
INK = (25, 29, 36)
DIVIDER = (207, 212, 220)
FRAME_COLORS = {
    "reference": (190, 50, 150),
    "red": (218, 68, 73),
    "green": (43, 157, 104),
    "blue": (54, 105, 204),
}
COLORS = ("red", "green", "blue")
CARD_IDS = ("reference",) + COLORS
CARD_W, CARD_H = 400, 270
CARD_SLOTS = ((45, 35), (515, 35), (45, 415), (515, 415))
SQUARE_HALF = 12
PIP_RADIUS = 11
PIP_SITES = (
    (-32, -58), (0, -58), (32, -58),
    (-32, 0), (0, 0), (32, 0),
    (-32, 58), (0, 58), (32, 58),
)


def _candidate_scores(scene: dict) -> dict[str, int]:
    reference = list(scene["reference_counts"])
    candidates = scene["candidate_counts"]
    return {
        color: sum(left == right for left, right in zip(reference, candidates[color]))
        for color in COLORS
    }


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene with equal-ink candidate signatures."""
    rng = random.Random(int(seed))
    reference = rng.sample(range(1, 6), 3)
    permutations = [
        list(values)
        for values in itertools.permutations(reference)
        if list(values) != reference
    ]
    rng.shuffle(permutations)
    correct_color = COLORS[int(seed) % len(COLORS)]
    wrong = iter(permutations[:2])
    candidate_counts = {
        color: list(reference) if color == correct_color else list(next(wrong))
        for color in COLORS
    }

    card_slots = list(CARD_IDS)
    rng.shuffle(card_slots)
    mark_layouts: dict[str, list[list[list[int]]]] = {"reference": []}
    for count in reference:
        x_jitter = rng.randint(-4, 4)
        points = [[x_jitter, 88 - 38 * index] for index in range(count)]
        rng.shuffle(points)
        mark_layouts["reference"].append(points)
    for color in COLORS:
        groups = []
        for count in candidate_counts[color]:
            sites = rng.sample(list(PIP_SITES), count)
            points = [
                [x + rng.randint(-2, 2), y + rng.randint(-2, 2)] for x, y in sites
            ]
            rng.shuffle(points)
            groups.append(points)
        mark_layouts[color] = groups

    scene = {
        "reference_counts": reference,
        "candidate_counts": candidate_counts,
        "card_slots": card_slots,
        "mark_layouts": mark_layouts,
    }
    if is_quarantined(scene):
        raise RuntimeError("sampler produced a quarantined scene")
    return scene


def analytic_gold(scene: dict) -> str:
    """Return the border color whose ordered pip counts equal the stacks."""
    reference = list(scene["reference_counts"])
    candidates = scene["candidate_counts"]
    matches = [color for color in COLORS if list(candidates[color]) == reference]
    if len(matches) != 1:
        raise ValueError("there must be exactly one ordered cardinality match")
    return matches[0]


def margin(scene: dict) -> float:
    """Best-minus-runner-up count of matching ordered compartments."""
    ranked = sorted(_candidate_scores(scene).values(), reverse=True)
    return float(ranked[0] - ranked[1]) if len(ranked) >= 2 else 0.0


def _layout_ok(points: object, expected_count: int) -> bool:
    if not isinstance(points, list) or len(points) != expected_count:
        return False
    parsed: list[tuple[float, float]] = []
    for point in points:
        if (
            not isinstance(point, list)
            or len(point) != 2
            or not all(isinstance(value, (int, float)) for value in point)
        ):
            return False
        x, y = float(point[0]), float(point[1])
        if not (-42 <= x <= 42 and -78 <= y <= 100):
            return False
        parsed.append((x, y))
    return all(
        math.hypot(ax - bx, ay - by) >= 26
        for index, (ax, ay) in enumerate(parsed)
        for bx, by in parsed[index + 1 :]
    )


def is_quarantined(scene: dict) -> bool:
    """Reject malformed, ambiguous, or finite-resolution unsafe scenes."""
    try:
        reference = list(scene["reference_counts"])
        candidates = scene["candidate_counts"]
        slots = list(scene["card_slots"])
        layouts = scene["mark_layouts"]
        if len(reference) != 3 or len(set(reference)) != 3:
            return True
        if any(not isinstance(value, int) or not 1 <= value <= 5 for value in reference):
            return True
        if set(candidates) != set(COLORS) or set(layouts) != set(CARD_IDS):
            return True
        if sorted(slots) != sorted(CARD_IDS):
            return True
        signatures = [tuple(candidates[color]) for color in COLORS]
        if len(set(signatures)) != 3:
            return True
        if any(sorted(signature) != sorted(reference) for signature in signatures):
            return True
        expected = {"reference": reference, **candidates}
        for card_id in CARD_IDS:
            groups = layouts[card_id]
            if not isinstance(groups, list) or len(groups) != 3:
                return True
            if any(
                not _layout_ok(points, count)
                for points, count in zip(groups, expected[card_id])
            ):
                return True
        _ = analytic_gold(scene)
        return margin(scene) < 2.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse non-overlapping mark draw lists: pixels and gold stay identical."""
    twin = copy.deepcopy(scene)
    for groups in twin["mark_layouts"].values():
        for points in groups:
            points.reverse()
    return [("reverse_mark_draw_order", twin)]


def _scaled_box(box: tuple[int, int, int, int]) -> tuple[int, int, int, int]:
    return tuple(value * SCALE for value in box)


def _draw_card(
    draw: ImageDraw.ImageDraw,
    card_id: str,
    box: tuple[int, int],
    counts: list[int],
    groups: list[list[list[int]]],
) -> None:
    x, y = box
    draw.rounded_rectangle(
        _scaled_box((x, y, x + CARD_W, y + CARD_H)),
        radius=14 * SCALE,
        fill=CARD_FILL,
        outline=FRAME_COLORS[card_id],
        width=8 * SCALE,
    )
    for boundary in (1, 2):
        sx = round(x + boundary * CARD_W / 3)
        draw.line(
            [(sx * SCALE, (y + 13) * SCALE), (sx * SCALE, (y + CARD_H - 13) * SCALE)],
            fill=DIVIDER,
            width=3 * SCALE,
        )
    for group_index, (count, points) in enumerate(zip(counts, groups)):
        if len(points) != count:
            raise ValueError("layout count differs from displayed cardinality")
        cx = round(x + (group_index + 0.5) * CARD_W / 3)
        cy = y + CARD_H // 2
        for dx, dy in points:
            px, py = cx + int(dx), cy + int(dy)
            if card_id == "reference":
                draw.rectangle(
                    _scaled_box(
                        (px - SQUARE_HALF, py - SQUARE_HALF, px + SQUARE_HALF, py + SQUARE_HALF)
                    ),
                    fill=INK,
                )
            else:
                draw.ellipse(
                    _scaled_box((px - PIP_RADIUS, py - PIP_RADIUS, px + PIP_RADIUS, py + PIP_RADIUS)),
                    fill=INK,
                )


def render(scene: dict) -> Image.Image:
    """Render all answer-bearing state into one antialiased PNG-ready image."""
    reference = scene["reference_counts"]
    candidates = scene["candidate_counts"]
    slots = scene["card_slots"]
    layouts = scene["mark_layouts"]
    if len(slots) != 4:
        raise ValueError("four card slots are required")
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for position, card_id in zip(CARD_SLOTS, slots):
        counts = reference if card_id == "reference" else candidates[card_id]
        _draw_card(draw, card_id, position, counts, layouts[card_id])
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
