"""Renderer and analytic arm for the woven-swatch organization world."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
ROWS = 6
COLS = 6
PALETTES = (
    ((37, 99, 164), (232, 166, 55)),
    ((34, 139, 115), (193, 74, 91)),
    ((112, 78, 170), (221, 142, 62)),
    ((34, 121, 180), (205, 82, 138)),
)
BACKGROUNDS = ((247, 244, 236), (244, 247, 243), (247, 245, 248))


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    order = "stripes" if seed % 2 == 0 else "checkerboard"
    cell = rng.choice((14, 15, 16, 17))
    span = ROWS * cell
    # The decisive inset moves widely while always staying away from the border.
    left = rng.randint(42, WIDTH - span - 42)
    top = rng.randint(42, HEIGHT - span - 42)
    return {
        "order": order,
        "phase": rng.randrange(2),
        "palette_swap": rng.randrange(2),
        "palette": rng.randrange(len(PALETTES)),
        "left": left,
        "top": top,
        "cell": cell,
        "rows": ROWS,
        "cols": COLS,
        "background": rng.randrange(len(BACKGROUNDS)),
    }


def _bit(scene: dict, row: int, col: int) -> int:
    if scene["order"] == "stripes":
        base = row & 1
    else:
        base = (row + col) & 1
    return base ^ int(scene["phase"])


def render(scene: dict) -> Image.Image:
    background = BACKGROUNDS[int(scene["background"]) % len(BACKGROUNDS)]
    image = Image.new("RGB", (WIDTH, HEIGHT), background)
    draw = ImageDraw.Draw(image)
    left = int(scene["left"])
    top = int(scene["top"])
    cell = int(scene["cell"])
    rows = int(scene["rows"])
    cols = int(scene["cols"])
    colors = PALETTES[int(scene["palette"]) % len(PALETTES)]
    if int(scene["palette_swap"]):
        colors = colors[::-1]

    # Butt-jointed cells make a single thick, connected, area-like swatch.
    for row in range(rows):
        for col in range(cols):
            x0 = left + col * cell
            y0 = top + row * cell
            draw.rectangle(
                (x0, y0, x0 + cell - 1, y0 + cell - 1),
                fill=colors[_bit(scene, row, col)],
            )

    # A neutral frame exposes the exact field boundary without carrying the label.
    right = left + cols * cell
    bottom = top + rows * cell
    draw.rounded_rectangle(
        (left - 5, top - 5, right + 4, bottom + 4),
        radius=7,
        outline=(72, 72, 72),
        width=3,
    )
    # Neutral corner studs make scale and orientation explicit, but are nondiagnostic.
    for x, y in ((left - 9, top - 9), (right + 8, top - 9), (left - 9, bottom + 8), (right + 8, bottom + 8)):
        draw.ellipse((x - 2, y - 2, x + 2, y + 2), fill=(100, 100, 100))
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["order"])


def margin(scene: dict) -> float:
    # Every ideal 6x6 scene has all 30 horizontal adjacencies voting one way.
    return 1.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # Complementing the abstract phase while swapping the palette assignment is
    # pixel-identical, yet is a nontrivial change to the latent representation.
    twin = deepcopy(scene)
    twin["phase"] = 1 - int(scene["phase"])
    twin["palette_swap"] = 1 - int(scene["palette_swap"])
    return [("phase_palette_complement", twin)]
