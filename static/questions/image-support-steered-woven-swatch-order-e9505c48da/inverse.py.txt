"""Independent pixel-only inverse arm for compact woven swatches.

The implementation knows only the public 6 by 6 question contract. It locates
the saturated swatch without fixed coordinates or renderer state, reconstructs
two palette-neutral color classes, and scores whole-field adjacency grammar.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ROWS = 6
COLS = 6
MIN_GAP = 0.35


class OracleAbstention(ValueError):
    pass


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen: set[tuple[int, int]] = set()
    groups: list[list[tuple[int, int]]] = []
    for yy, xx in np.argwhere(mask):
        start = (int(yy), int(xx))
        if start in seen:
            continue
        seen.add(start)
        stack = [start]
        group: list[tuple[int, int]] = []
        while stack:
            y, x = stack.pop()
            group.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                point = (ny, nx)
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and point not in seen:
                    seen.add(point)
                    stack.append(point)
        groups.append(group)
    return groups


def _grid_from_image(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    chroma = array.max(axis=2) - array.min(axis=2)
    saturated = chroma > 48.0
    groups = _components(saturated)
    if not groups:
        raise OracleAbstention("no colored swatch")
    group = max(groups, key=len)
    ys = np.fromiter((point[0] for point in group), dtype=np.int32)
    xs = np.fromiter((point[1] for point in group), dtype=np.int32)
    y0, y1 = int(ys.min()), int(ys.max()) + 1
    x0, x1 = int(xs.min()), int(xs.max()) + 1
    height, width = y1 - y0, x1 - x0
    if not (82 <= height <= 104 and 82 <= width <= 104 and abs(height - width) <= 2):
        raise OracleAbstention("colored field has an invalid extent")
    local = saturated[y0:y1, x0:x1]
    if float(local.mean()) < 0.985:
        raise OracleAbstention("colored field is incomplete")

    means = np.zeros((ROWS, COLS, 3), dtype=np.float64)
    for row in range(ROWS):
        ay = y0 + (2 * row + 1) * height // (2 * ROWS)
        for col in range(COLS):
            ax = x0 + (2 * col + 1) * width // (2 * COLS)
            chip = array[max(y0, ay - 2):min(y1, ay + 3), max(x0, ax - 2):min(x1, ax + 3)]
            means[row, col] = np.median(chip.reshape(-1, 3), axis=0)

    # Find the most separated observed pair, then label every cell by proximity.
    flat = means.reshape(-1, 3)
    distances = np.linalg.norm(flat[:, None, :] - flat[None, :, :], axis=2)
    first, second = np.unravel_index(int(np.argmax(distances)), distances.shape)
    anchors = flat[[first, second]]
    if float(np.linalg.norm(anchors[0] - anchors[1])) < 70.0:
        raise OracleAbstention("swatch colors are not separable")
    grid = np.argmin(np.linalg.norm(means[:, :, None, :] - anchors[None, None, :, :], axis=3), axis=2)
    fit = np.min(np.linalg.norm(means[:, :, None, :] - anchors[None, None, :, :], axis=3), axis=2)
    if float(np.max(fit)) > 18.0:
        raise OracleAbstention("cells do not form a clean two-color field")
    if not 14 <= int(grid.sum()) <= 22:
        raise OracleAbstention("color totals are implausibly imbalanced")
    return grid


def _scores(grid: np.ndarray) -> tuple[float, float]:
    horizontal_equal = np.mean(grid[:, 1:] == grid[:, :-1])
    horizontal_alternate = np.mean(grid[:, 1:] != grid[:, :-1])
    vertical_alternate = np.mean(grid[1:, :] != grid[:-1, :])
    # Stripes: same within every row and alternate between rows.
    stripe_score = float((horizontal_equal + vertical_alternate) / 2.0)
    # Checkerboard: alternate along both image axes.
    checker_score = float((horizontal_alternate + vertical_alternate) / 2.0)
    return stripe_score, checker_score


def decision_from_image(image: Image.Image) -> str:
    grid = _grid_from_image(image)
    stripe_score, checker_score = _scores(grid)
    gap = abs(stripe_score - checker_score)
    if max(stripe_score, checker_score) < 0.88 or gap < MIN_GAP:
        raise OracleAbstention("ambiguous organization")
    return "stripes" if stripe_score > checker_score else "checkerboard"
