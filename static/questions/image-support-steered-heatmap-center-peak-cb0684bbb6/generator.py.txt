"""Deterministic analytic scene sampler and Pillow renderer."""

from __future__ import annotations

import colorsys
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
CELL = 18
TILE = 3 * CELL
BACKGROUND = (248, 248, 246)
GRID_INK = (55, 58, 62)
NEUTRAL_INK = (184, 186, 184)
HUES = (0.58, 0.82, 0.45, 0.035)
EDGE_INDICES = (1, 3, 5, 7)
CORNER_INDICES = (0, 2, 6, 8)


def _rank_colors(palette: int) -> list[tuple[int, int, int]]:
    hue = HUES[int(palette) % len(HUES)]
    colors = []
    for rank in range(9):
        value = 0.97 - 0.068 * rank
        rgb = colorsys.hsv_to_rgb(hue, 0.72, value)
        colors.append(tuple(round(channel * 255) for channel in rgb))
    return colors


def _nuisance_state(seed: int) -> tuple[int, int, int, list[dict[str, int]]]:
    # Clearing the low bit makes adjacent YES/NO seeds share all nuisances.
    rng = random.Random((int(seed) & ~1) ^ 0x4A17C3)
    grid_x = rng.randrange(1, 7)
    grid_y = rng.randrange(1, 7)
    tile_x = grid_x * 64 + 5
    tile_y = grid_y * 64 + 5
    palette = rng.randrange(len(HUES))

    distractors: list[dict[str, int]] = []
    attempts = 0
    while len(distractors) < 9 and attempts < 300:
        attempts += 1
        x = rng.randrange(22, WIDTH - 22)
        y = rng.randrange(22, HEIGHT - 22)
        if abs(x - (tile_x + TILE // 2)) < 72 and abs(y - (tile_y + TILE // 2)) < 72:
            continue
        if any((x - item["x"]) ** 2 + (y - item["y"]) ** 2 < 34**2 for item in distractors):
            continue
        distractors.append(
            {"x": x, "y": y, "kind": rng.randrange(3), "size": rng.randrange(5, 10)}
        )
    if len(distractors) != 9:
        raise RuntimeError("could not place neutral registration marks")
    return tile_x, tile_y, palette, distractors


def sample_scene(seed: int) -> dict[str, Any]:
    tile_x, tile_y, palette, distractors = _nuisance_state(seed)
    layout_rng = random.Random((int(seed) & ~1) ^ 0xB82F91)
    low = [0, 1, 2, 3]
    high = [5, 6, 7, 8]
    layout_rng.shuffle(low)
    layout_rng.shuffle(high)
    values = [0] * 9
    values[4] = 4
    edge_values, corner_values = (low, high) if int(seed) & 1 else (high, low)
    for index, value in zip(EDGE_INDICES, edge_values, strict=True):
        values[index] = value
    for index, value in zip(CORNER_INDICES, corner_values, strict=True):
        values[index] = value
    return {
        "values": values,
        "tile_x": tile_x,
        "tile_y": tile_y,
        "palette": palette,
        "distractors": distractors,
    }


def _draw_distractor(draw: ImageDraw.ImageDraw, item: dict[str, int]) -> None:
    x, y, size, kind = item["x"], item["y"], item["size"], item["kind"]
    if kind == 0:
        draw.ellipse((x - size, y - size, x + size, y + size), outline=NEUTRAL_INK, width=2)
    elif kind == 1:
        draw.line((x - size, y, x + size, y), fill=NEUTRAL_INK, width=2)
        draw.line((x, y - size, x, y + size), fill=NEUTRAL_INK, width=2)
    else:
        draw.polygon(
            ((x, y - size), (x - size, y + size), (x + size, y + size)),
            outline=NEUTRAL_INK,
        )


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for item in scene["distractors"]:
        _draw_distractor(draw, item)

    x = int(scene["tile_x"])
    y = int(scene["tile_y"])
    colors = _rank_colors(int(scene["palette"]))
    draw.rectangle((x - 3, y - 3, x + TILE + 2, y + TILE + 2), fill=GRID_INK)
    for index, rank in enumerate(scene["values"]):
        row, column = divmod(index, 3)
        x0 = x + column * CELL
        y0 = y + row * CELL
        draw.rectangle((x0, y0, x0 + CELL - 1, y0 + CELL - 1), fill=colors[int(rank)])
    for offset in (CELL, 2 * CELL):
        draw.line((x + offset, y, x + offset, y + TILE - 1), fill=GRID_INK, width=1)
        draw.line((x, y + offset, x + TILE - 1, y + offset), fill=GRID_INK, width=1)
    return image


def _signed_rank_gap(scene: dict[str, Any]) -> int:
    values = scene["values"]
    return int(values[4]) - max(int(values[index]) for index in EDGE_INDICES)


def analytic_gold(scene: dict[str, Any]) -> str:
    return "yes" if _signed_rank_gap(scene) > 0 else "no"


def margin(scene: dict[str, Any]) -> float:
    return float(abs(_signed_rank_gap(scene)))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 1.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
