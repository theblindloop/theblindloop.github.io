"""Independent pixel-only inverse for the compact heatmap tile."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _luminance(rgb: np.ndarray) -> float:
    vector = np.asarray(rgb, dtype=np.float64)
    return float(0.2126 * vector[0] + 0.7152 * vector[1] + 0.0722 * vector[2])


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[2] != 3:
        raise ValueError("expected an RGB raster")

    high = array.max(axis=2).astype(np.int16)
    low = array.min(axis=2).astype(np.int16)
    saturated = (high - low) >= 45
    points = np.argwhere(saturated)
    if len(points) < 2400:
        raise ValueError("complete heatmap tile not found")

    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    height = int(y1 - y0 + 1)
    width = int(x1 - x0 + 1)
    if not (51 <= width <= 56 and 51 <= height <= 56):
        raise ValueError("heatmap extent is unresolved")
    local_mask = saturated[y0 : y1 + 1, x0 : x1 + 1]
    if float(local_mask.mean()) < 0.91:
        raise ValueError("heatmap tile is incomplete")
    if int(saturated.sum()) != int(local_mask.sum()):
        raise ValueError("multiple chromatic regions found")

    cell_w = width / 3.0
    cell_h = height / 3.0
    lightness: list[float] = []
    for row in range(3):
        for column in range(3):
            # Stay two pixels inside each recovered cell to avoid grid strokes.
            xa = int(round(x0 + column * cell_w)) + 2
            xb = int(round(x0 + (column + 1) * cell_w)) - 2
            ya = int(round(y0 + row * cell_h)) + 2
            yb = int(round(y0 + (row + 1) * cell_h)) - 2
            patch = array[ya:yb, xa:xb].reshape(-1, 3)
            if len(patch) < 80:
                raise ValueError("cell interior is too small")
            color = np.median(patch, axis=0)
            lightness.append(_luminance(color))

    center = lightness[4]
    neighbors = [lightness[index] for index in (1, 3, 5, 7)]
    signed_gap = min(value - center for value in neighbors)
    if abs(signed_gap) < 4.0:
        raise ValueError("center-neighbor lightness boundary is ambiguous")
    return "yes" if signed_gap > 0.0 else "no"
