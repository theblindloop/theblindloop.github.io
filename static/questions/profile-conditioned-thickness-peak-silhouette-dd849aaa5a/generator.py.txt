"""Pillow renderer and analytic scene contract for one thickness-field question."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "thickness_peak_silhouette-0.1.0"
WIDTH = 900
HEIGHT = 360
PANEL_TOP = 50
PANEL_WIDTH = 260
PANEL_HEIGHT = 260
PANEL_LEFTS = (28, 320, 612)
SUPERSAMPLE = 3
COLORS = ("red", "blue", "green")
PALETTE = {
    "red": (218, 72, 78),
    "blue": (58, 112, 214),
    "green": (48, 158, 103),
}
FRAME = (232, 236, 241)
BACKGROUND = (255, 255, 255)
CANONICAL_SIZE = 220
PEAK_SEPARATION = 22
PEAK_DEPTH_FRACTION = 0.42


def _base_shape(kind: str, size: int = CANONICAL_SIZE) -> Image.Image:
    """Create a canonical filled silhouette; geometry is not used by the oracle."""

    mask = Image.new("L", (size, size), 0)
    draw = ImageDraw.Draw(mask)
    q = size / CANONICAL_SIZE

    def ellipse(cx: float, cy: float, radius: float) -> None:
        draw.ellipse(
            ((cx - radius) * q, (cy - radius) * q,
             (cx + radius) * q, (cy + radius) * q),
            fill=255,
        )

    def line(a: tuple[float, float], b: tuple[float, float], width: float) -> None:
        draw.line((a[0] * q, a[1] * q, b[0] * q, b[1] * q), fill=255, width=max(1, round(width * q)))

    if kind == "single":
        draw.ellipse((49 * q, 57 * q, 171 * q, 163 * q), fill=255)
    elif kind == "double":
        ellipse(73, 110, 39)
        ellipse(147, 110, 39)
        line((73, 110), (147, 110), 24)
    elif kind == "triple":
        centers = ((110, 66), (70, 137), (150, 137))
        for center in centers:
            ellipse(center[0], center[1], 28)
        for first, second in ((centers[0], centers[1]), (centers[0], centers[2]), (centers[1], centers[2])):
            line(first, second, 18)
    else:
        raise ValueError(f"unknown shape kind {kind!r}")
    return mask


_BASE_AREAS = {kind: _base_shape(kind).histogram()[255] for kind in ("single", "double", "triple")}
_REFERENCE_AREA = sum(_BASE_AREAS.values()) / len(_BASE_AREAS)


def _shape_mask(kind: str, jitter: float, rotation: float, supersample: int) -> Image.Image:
    base = _base_shape(kind, CANONICAL_SIZE * supersample)
    area_scale = math.sqrt(_REFERENCE_AREA / _BASE_AREAS[kind])
    scale = area_scale * jitter
    new_size = max(8, round(base.width * scale))
    base = base.resize((new_size, new_size), Image.Resampling.LANCZOS)
    return base.rotate(rotation, resample=Image.Resampling.BICUBIC, expand=False)


def sample_scene(seed: int) -> dict[str, Any]:
    """Return a deterministic scene; answer identity is independent of panel slot."""

    rng = random.Random(int(seed))
    target_color = rng.choice(list(COLORS))
    color_order = list(COLORS)
    rng.shuffle(color_order)
    decoys = ["double", "triple"]
    rng.shuffle(decoys)
    panels: list[dict[str, Any]] = []
    for slot, color in enumerate(color_order):
        kind = "single" if color == target_color else decoys.pop()
        panels.append(
            {
                "slot": slot,
                "color": color,
                "kind": kind,
                "jitter": round(rng.uniform(0.96, 1.04), 6),
                "rotation": round(rng.uniform(-18.0, 18.0), 6),
                "offset": [round(rng.uniform(-13.0, 13.0), 6), round(rng.uniform(-10.0, 10.0), 6)],
            }
        )
    return {"panels": panels}


def render(scene: dict[str, Any]) -> Image.Image:
    """Render three same-frame filled silhouettes to an RGB PNG-compatible image."""

    ss = SUPERSAMPLE
    image = Image.new("RGB", (WIDTH * ss, HEIGHT * ss), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for left in PANEL_LEFTS:
        rect = (
            left * ss,
            PANEL_TOP * ss,
            (left + PANEL_WIDTH) * ss,
            (PANEL_TOP + PANEL_HEIGHT) * ss,
        )
        draw.rounded_rectangle(rect, radius=12 * ss, fill=(249, 250, 252), outline=FRAME, width=2 * ss)

    for panel in scene["panels"]:
        slot = int(panel["slot"])
        if slot not in (0, 1, 2):
            raise ValueError("panel slot outside the three visible slots")
        left = PANEL_LEFTS[slot]
        mask = _shape_mask(
            str(panel["kind"]),
            float(panel["jitter"]),
            float(panel["rotation"]),
            ss,
        )
        cx = (left + PANEL_WIDTH / 2.0 + float(panel["offset"][0])) * ss
        cy = (PANEL_TOP + PANEL_HEIGHT / 2.0 + float(panel["offset"][1])) * ss
        xy = (round(cx - mask.width / 2), round(cy - mask.height / 2))
        image.paste(PALETTE[str(panel["color"])], xy, mask)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the visible color attached to the single-lobe silhouette family."""

    for panel in scene["panels"]:
        if panel["kind"] == "single":
            return str(panel["color"])
    raise ValueError("scene has no single-peak silhouette")


def margin(scene: dict[str, Any]) -> float:
    """Return the ideal one-peak score gap against the runner-up family."""

    scores = []
    for panel in scene["panels"]:
        kind = panel["kind"]
        scores.append(1.0 if kind == "single" else 0.0)
    ordered = sorted(scores, reverse=True)
    return float(ordered[0] - ordered[1])


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Exclude only a small top-gap or malformed-family case from headline classes."""

    kinds = [panel["kind"] for panel in scene["panels"]]
    return margin(scene) < 0.18 or kinds.count("single") != 1


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No latent aliases are claimed; all label-bearing panel records are rendered."""

    return []
