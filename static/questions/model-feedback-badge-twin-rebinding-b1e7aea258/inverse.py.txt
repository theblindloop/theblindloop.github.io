"""Independent image-only inverse for Badge Twin Rebinding."""

from __future__ import annotations

import collections
from PIL import Image, ImageDraw

CANVAS = (1000, 800)
INK = (35, 40, 48)
MUTED = (218, 224, 228)
SHAPE = (25, 126, 111)
BADGE = (58, 48, 92)
PANEL = (255, 255, 255)
ACCENT = (194, 47, 62)
COLORS = {"amber": (219, 139, 22), "cyan": (18, 153, 177), "violet": (133, 76, 181)}
ROW_Y = (315, 435, 555, 675)
FIELD_X = {"left": 450, "right": 750}


class OracleError(ValueError):
    pass


def _shape_template(name: str) -> tuple[int, ...]:
    image = Image.new("RGB", (80, 80), PANEL)
    draw, cx, cy = ImageDraw.Draw(image), 40, 40
    if name == "circle": draw.ellipse((cx - 25, cy - 25, cx + 25, cy + 25), fill=SHAPE)
    elif name == "square": draw.rectangle((cx - 23, cy - 23, cx + 23, cy + 23), fill=SHAPE)
    elif name == "triangle": draw.polygon(((cx, cy - 29), (cx - 29, cy + 23), (cx + 29, cy + 23)), fill=SHAPE)
    elif name == "diamond": draw.polygon(((cx, cy - 30), (cx - 30, cy), (cx, cy + 30), (cx + 30, cy)), fill=SHAPE)
    return tuple(pixel == SHAPE for pixel in image.getdata())


def _read_shape(image: Image.Image, cx: int, cy: int) -> str:
    crop = image.crop((cx - 40, cy - 40, cx + 40, cy + 40))
    observed = tuple(pixel == SHAPE for pixel in crop.getdata())
    scores = {name: sum(a != b for a, b in zip(observed, _shape_template(name)))
              for name in ("circle", "square", "triangle", "diamond")}
    best = min(scores, key=scores.get)
    if scores[best] != 0 or sum(observed) < 1500:
        raise OracleError("damaged or unknown row shape")
    return best


def _read_badge(image: Image.Image, cx: int, cy: int) -> tuple[int, ...]:
    px = image.load()
    if px[cx - 57, cy] != INK or px[cx + 57, cy] != INK:
        raise OracleError("badge border missing")
    bits = []
    for index in range(6):
        col, row = index % 3, index // 3
        x0, y0 = cx - 42 + col * 29, cy - 27 + row * 33
        samples = [px[x, y] for x in range(x0 + 4, x0 + 20, 5)
                   for y in range(y0 + 4, y0 + 20, 5)]
        if all(pixel == BADGE for pixel in samples): bits.append(1)
        elif all(pixel == MUTED for pixel in samples): bits.append(0)
        else: raise OracleError("badge cell damaged")
    if sum(bits) != 3:
        raise OracleError("invalid badge")
    return tuple(bits)


def _read_field(image: Image.Image, cx: int, cy: int) -> tuple[tuple[int, ...], str]:
    px = image.load()
    badge = _read_badge(image, cx - 25, cy)
    if px[cx + 52, cy] != INK:
        raise OracleError("badge-to-dot link missing")
    dot = px[cx + 94, cy]
    names = [name for name, value in COLORS.items() if dot == value]
    if len(names) != 1:
        raise OracleError("unknown dot color")
    return badge, names[0]


def decision_from_image(image: Image.Image) -> str:
    image = image.convert("RGB")
    if image.size != CANVAS:
        raise OracleError("wrong canvas")
    px = image.load()
    for x, y in ((500, 25), (45, 400), (955, 400)):
        if px[x, y] != INK:
            raise OracleError("panel boundary missing")
    start_key = _read_shape(image, 210, 125)
    left_marked = px[477, 122] == ACCENT
    right_marked = px[657, 122] == ACCENT
    if left_marked == right_marked:
        raise OracleError("START field marker is ambiguous")
    take_side = "left" if left_marked else "right"
    records = {}
    occurrences = collections.Counter()
    for y in ROW_Y:
        if px[235, y] != INK:
            raise OracleError("row arrow missing")
        key = _read_shape(image, 165, y)
        fields = {side: _read_field(image, cx, y) for side, cx in FIELD_X.items()}
        if key in records:
            raise OracleError("duplicate row key")
        records[key] = fields
        occurrences.update(field[0] for field in fields.values())
    if set(records) != {"circle", "square", "triangle", "diamond"}:
        raise OracleError("missing row key")
    if len(occurrences) != 4 or set(occurrences.values()) != {2}:
        raise OracleError("badges must occur in pairs")
    for badge in occurrences:
        containing = {key for key, fields in records.items()
                      if any(field[0] == badge for field in fields.values())}
        if len(containing) != 2:
            raise OracleError("badge twin occurs within one row")
    selected = records[start_key][take_side][0]
    twins = [field[1] for key, fields in records.items() if key != start_key
             for field in fields.values() if field[0] == selected]
    if len(twins) != 1:
        raise OracleError("selected badge lacks unique external twin")
    return twins[0]
