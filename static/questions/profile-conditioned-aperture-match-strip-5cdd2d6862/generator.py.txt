"""Aperture-match strip world.

Latent scene z = {width, height, palette, thickness, key, plates, panels}.

A key box at the top holds two horizontal plates separated by a vertical
reference aperture. Four panels below the key, arranged left to right, each hold
two vertical posts separated by a horizontal aperture. Every bar in the image
(plate or post) has the same visible thickness. Exactly one panel's aperture is
exactly as wide as the reference aperture is tall; every other panel's aperture
differs from the reference by more than one bar thickness.

The decision is the panel carrying the matching aperture. The margin is the
smallest absolute aperture mismatch among the non-matching panels, in pixels.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "aperture_match_strip-0.1.0"

WIDTH = 640
HEIGHT = 440
STRIP_Y = 168
THICKNESS = 12
PANEL_COUNT = 4
PANEL_WIDTH = WIDTH // PANEL_COUNT
SUPERSAMPLE = 3

KEY_LEFT = 226
KEY_RIGHT = 414
KEY_TOP = 18
KEY_BOTTOM = 152

MIN_MISMATCH = 18.0
QUARANTINE_MISMATCH = 21.0
REF_MIN = 56
REF_MAX = 84
GAP_FLOOR = 30
GAP_CEIL = 112

PANELS = ("panel_1", "panel_2", "panel_3", "panel_4")

PALETTES = (
    {"background": (255, 255, 255), "bar": (26, 26, 30), "frame": (176, 176, 176)},
    {"background": (247, 245, 239), "bar": (18, 32, 46), "frame": (170, 172, 174)},
    {"background": (242, 246, 250), "bar": (42, 22, 24), "frame": (168, 176, 184)},
    {"background": (249, 250, 243), "bar": (22, 42, 28), "frame": (180, 176, 166)},
)


def _panel_index(cx: float) -> int:
    return min(PANEL_COUNT - 1, max(0, int(cx // PANEL_WIDTH)))


def _reference_aperture(scene: dict) -> float:
    plates = sorted(scene["plates"], key=lambda plate: plate["y"])
    return float(plates[1]["y"] - (plates[0]["y"] + scene["thickness"]))


def _panel_apertures(scene: dict) -> dict:
    apertures: dict = {}
    for panel in scene["panels"]:
        apertures[PANELS[_panel_index(panel["cx"])]] = float(panel["gap"])
    return apertures


def analytic_gold(scene: dict) -> str:
    """Panel whose horizontal aperture equals the vertical reference aperture."""
    reference = _reference_aperture(scene)
    apertures = _panel_apertures(scene)
    if len(apertures) != PANEL_COUNT:
        raise ValueError("scene does not populate every panel exactly once")
    winners = [name for name, gap in apertures.items() if gap == reference]
    if len(winners) != 1:
        raise ValueError(f"scene has no unique aperture match: {sorted(winners)}")
    return winners[0]


def margin(scene: dict) -> float:
    """Smallest aperture mismatch among the non-matching panels, in pixels."""
    reference = _reference_aperture(scene)
    mismatches = [
        abs(gap - reference) for gap in _panel_apertures(scene).values() if gap != reference
    ]
    if not mismatches:
        raise ValueError("scene has no competing panel")
    return round(min(mismatches), 6)


def is_quarantined(scene: dict) -> bool:
    """Near-boundary scenes whose closest competitor is hard to separate."""
    return bool(margin(scene) < QUARANTINE_MISMATCH)


def latent_symmetries(scene: dict):
    """Storage-order aliases: the plate list and the panel list are unordered."""
    plates_swapped = dict(scene, plates=list(reversed(scene["plates"])))
    panels_shuffled = dict(scene, panels=[scene["panels"][i] for i in (2, 0, 3, 1)])
    return [
        ("plate_record_order", plates_swapped),
        ("panel_record_order", panels_shuffled),
    ]


def _ink(panel: dict, thickness: int) -> int:
    return 2 * thickness * panel["height"]


def sample_scene(seed: int) -> dict:
    rng = random.Random(repr(("aperture_match_strip", seed)))
    target = seed % PANEL_COUNT
    below_count = (seed // PANEL_COUNT) % PANEL_COUNT
    palette_index = rng.randrange(len(PALETTES))
    thickness = THICKNESS
    for _ in range(4000):
        reference = rng.randint(REF_MIN, REF_MAX)
        low_cap = min(52, reference - GAP_FLOOR - 2)
        high_cap = min(52, GAP_CEIL - reference)
        if low_cap < MIN_MISMATCH or high_cap < MIN_MISMATCH:
            continue
        others = [i for i in range(PANEL_COUNT) if i != target]
        rng.shuffle(others)
        below = set(others[: min(below_count, len(others))])
        gaps = {}
        for index in range(PANEL_COUNT):
            if index == target:
                gaps[index] = reference
            elif index in below:
                gaps[index] = reference - rng.randint(int(MIN_MISMATCH), int(low_cap))
            else:
                gaps[index] = reference + rng.randint(int(MIN_MISMATCH), int(high_cap))
        heights = [rng.randint(96, 168) for _ in range(PANEL_COUNT)]
        if heights[target] in (max(heights), min(heights)):
            continue
        panels = []
        for index in range(PANEL_COUNT):
            cx = index * PANEL_WIDTH + PANEL_WIDTH / 2 + rng.randint(-8, 8)
            top = rng.randint(STRIP_Y + 14, HEIGHT - 14 - heights[index])
            panels.append(
                {
                    "cx": float(cx),
                    "gap": int(gaps[index]),
                    "top": int(top),
                    "height": int(heights[index]),
                }
            )
        plate_length = rng.randint(96, 148)
        plate_x = (KEY_LEFT + KEY_RIGHT) / 2 - plate_length / 2 + rng.randint(-10, 10)
        top_room = KEY_BOTTOM - 10 - (2 * thickness + reference)
        if top_room <= KEY_TOP + 10:
            continue
        upper_y = rng.randint(KEY_TOP + 10, int(top_room))
        plates = [
            {"x": float(plate_x), "y": int(upper_y), "length": int(plate_length)},
            {
                "x": float(plate_x),
                "y": int(upper_y + thickness + reference),
                "length": int(plate_length),
            },
        ]
        rng.shuffle(plates)
        scene = {
            "width": WIDTH,
            "height": HEIGHT,
            "palette": palette_index,
            "thickness": thickness,
            "plates": plates,
            "panels": panels,
        }
        if analytic_gold(scene) != PANELS[target]:
            continue
        if margin(scene) < MIN_MISMATCH:
            continue
        return scene
    raise RuntimeError(f"could not sample a valid scene for seed {seed}")


def render(scene: dict) -> Image.Image:
    palette = PALETTES[scene["palette"] % len(PALETTES)]
    width = scene["width"]
    height = scene["height"]
    thickness = scene["thickness"]
    ss = SUPERSAMPLE
    image = Image.new("RGB", (width * ss, height * ss), palette["background"])
    draw = ImageDraw.Draw(image)
    line = max(1, ss // 2)

    def box(x0, y0, x1, y1, fill):
        draw.rectangle(
            [(round(x0 * ss), round(y0 * ss)), (round(x1 * ss) - 1, round(y1 * ss) - 1)],
            fill=fill,
        )

    draw.rectangle(
        [(0, 0), (width * ss - 1, height * ss - 1)], outline=palette["frame"], width=line
    )
    draw.line(
        [(0, STRIP_Y * ss), (width * ss, STRIP_Y * ss)], fill=palette["frame"], width=line
    )
    for index in range(1, PANEL_COUNT):
        x = index * PANEL_WIDTH * ss
        draw.line([(x, STRIP_Y * ss), (x, height * ss)], fill=palette["frame"], width=line)
    draw.rectangle(
        [
            (KEY_LEFT * ss, KEY_TOP * ss),
            (KEY_RIGHT * ss - 1, KEY_BOTTOM * ss - 1),
        ],
        outline=palette["frame"],
        width=line,
    )
    for plate in scene["plates"]:
        box(plate["x"], plate["y"], plate["x"] + plate["length"], plate["y"] + thickness, palette["bar"])
    for panel in scene["panels"]:
        left_inner = panel["cx"] - panel["gap"] / 2.0
        right_inner = panel["cx"] + panel["gap"] / 2.0
        top = panel["top"]
        bottom = panel["top"] + panel["height"]
        box(left_inner - thickness, top, left_inner, bottom, palette["bar"])
        box(right_inner, top, right_inner + thickness, bottom, palette["bar"])
    return image.resize((width, height), Image.LANCZOS)
