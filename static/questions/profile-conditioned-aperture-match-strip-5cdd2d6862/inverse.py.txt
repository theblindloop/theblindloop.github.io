"""Pixel-only decision arm for the aperture-match strip world.

Receives a rendered RGB image and nothing else. It thresholds the dark bar ink,
labels connected components, splits them into wide plates and tall posts by
bounding-box aspect, measures the vertical clearance between the two plates and
the horizontal clearance between each panel's two posts, and returns the panel
whose clearance best matches the plate clearance.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

DARK_LUMINANCE = 110.0
MIN_COMPONENT_AREA = 300
PANEL_COUNT = 4
MATCH_TOLERANCE = 9.0
DECISION_MARGIN = 6.0
AMBIGUOUS = "ambiguous"
PANELS = ("panel_1", "panel_2", "panel_3", "panel_4")


def _dark_mask(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.float64)
    luminance = 0.299 * arr[:, :, 0] + 0.587 * arr[:, :, 1] + 0.114 * arr[:, :, 2]
    return luminance < DARK_LUMINANCE


def _boxes(mask: np.ndarray) -> list:
    height, width = mask.shape
    seen = np.zeros((height, width), dtype=bool)
    found = []
    for start_y in range(height):
        for start_x in range(width):
            if not mask[start_y, start_x] or seen[start_y, start_x]:
                continue
            stack = [(start_y, start_x)]
            seen[start_y, start_x] = True
            area = 0
            y0 = y1 = start_y
            x0 = x1 = start_x
            while stack:
                y, x = stack.pop()
                area += 1
                y0 = min(y0, y)
                y1 = max(y1, y)
                x0 = min(x0, x)
                x1 = max(x1, x)
                for ny in (y - 1, y, y + 1):
                    if ny < 0 or ny >= height:
                        continue
                    for nx in (x - 1, x, x + 1):
                        if nx < 0 or nx >= width:
                            continue
                        if mask[ny, nx] and not seen[ny, nx]:
                            seen[ny, nx] = True
                            stack.append((ny, nx))
            if area >= MIN_COMPONENT_AREA:
                found.append({"y0": y0, "y1": y1, "x0": x0, "x1": x1, "area": area})
    return found


def decision_from_image(image: Image.Image) -> str:
    mask = _dark_mask(image)
    height, width = mask.shape
    plates = []
    posts = []
    for box in _boxes(mask):
        box_w = box["x1"] - box["x0"] + 1
        box_h = box["y1"] - box["y0"] + 1
        if box_w > box_h:
            plates.append(box)
        elif box_h > box_w:
            posts.append(box)
        else:
            return AMBIGUOUS
    if len(plates) != 2 or len(posts) != 2 * PANEL_COUNT:
        return AMBIGUOUS
    plates.sort(key=lambda box: box["y0"])
    reference = float(plates[1]["y0"] - plates[0]["y1"] - 1)
    columns: dict = {index: [] for index in range(PANEL_COUNT)}
    lane = width / PANEL_COUNT
    for post in posts:
        center_x = (post["x0"] + post["x1"]) / 2.0
        columns[min(PANEL_COUNT - 1, int(center_x // lane))].append(post)
    scored = []
    for index in range(PANEL_COUNT):
        pair = columns[index]
        if len(pair) != 2:
            return AMBIGUOUS
        pair.sort(key=lambda box: box["x0"])
        clearance = float(pair[1]["x0"] - pair[0]["x1"] - 1)
        scored.append((abs(clearance - reference), PANELS[index]))
    scored.sort()
    if scored[0][0] > MATCH_TOLERANCE:
        return AMBIGUOUS
    if scored[1][0] - scored[0][0] < DECISION_MARGIN:
        return AMBIGUOUS
    return scored[0][1]
