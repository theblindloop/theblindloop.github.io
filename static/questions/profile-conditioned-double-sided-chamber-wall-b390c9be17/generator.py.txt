from __future__ import annotations

import random
from typing import Dict, List, Tuple

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 480
WALL = 14
COLORS = {
    "red": (220, 52, 52),
    "green": (38, 158, 82),
    "blue": (48, 96, 205),
}


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    order = ["red", "green", "blue"]
    rng.shuffle(order)
    return {"transform": rng.randrange(4), "color_order": order}


def _rot(point: Tuple[int, int], turns: int) -> Tuple[int, int]:
    x, y = point
    for _ in range(turns % 4):
        x, y = WIDTH - y, x
    return x, y


def _line(draw: ImageDraw.ImageDraw, a: Tuple[int, int], b: Tuple[int, int], fill) -> None:
    draw.line([a, b], fill=fill, width=WALL)


def render(scene: dict) -> Image.Image:
    turns = int(scene["transform"]) % 4
    color_order = list(scene["color_order"])
    image = Image.new("RGB", (WIDTH, HEIGHT), "white")
    draw = ImageDraw.Draw(image)
    black = (22, 25, 29)

    walls = [
        ((90, 90), (390, 90)), ((390, 90), (390, 390)),
        ((390, 390), (90, 390)), ((90, 390), (90, 90)),
        ((240, 90), (240, 390)), ((90, 240), (390, 240)),
    ]
    for a, b in walls:
        _line(draw, _rot(a, turns), _rot(b, turns), black)

    # Roles: bounded/bounded, bounded/exterior, exterior/exterior.
    candidates = [
        ((240, 115), (240, 215)),
        ((90, 265), (90, 365)),
        ((440, 170), (440, 270)),
    ]
    for color, (a, b) in zip(color_order, candidates):
        _line(draw, _rot(a, turns), _rot(b, turns), COLORS[color])
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["color_order"][0])


def margin(scene: dict) -> float:
    # Enclosed-side counts are 2, 1, and 0, hence winner minus runner-up = 1.
    return 1.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = {"transform": int(scene["transform"]) + 4, "color_order": list(scene["color_order"])}
    return [("full_rotation_alias", twin)]
