from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE = {
    "red": np.array((220, 52, 52), dtype=np.int16),
    "green": np.array((38, 158, 82), dtype=np.int16),
    "blue": np.array((48, 96, 205), dtype=np.int16),
}


def _exterior(white: np.ndarray) -> np.ndarray:
    h, w = white.shape
    seen = np.zeros((h, w), dtype=bool)
    queue = deque()
    for x in range(w):
        if white[0, x]: seen[0, x] = True; queue.append((0, x))
        if white[h - 1, x]: seen[h - 1, x] = True; queue.append((h - 1, x))
    for y in range(1, h - 1):
        if white[y, 0]: seen[y, 0] = True; queue.append((y, 0))
        if white[y, w - 1]: seen[y, w - 1] = True; queue.append((y, w - 1))
    while queue:
        y, x = queue.popleft()
        for yy, xx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= yy < h and 0 <= xx < w and white[yy, xx] and not seen[yy, xx]:
                seen[yy, xx] = True
                queue.append((yy, xx))
    return seen


def _side_samples(mask: np.ndarray) -> tuple[tuple[int, int], tuple[int, int]]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 50:
        raise ValueError("colored wall segment is missing")
    xmin, xmax, ymin, ymax = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    cx, cy = (xmin + xmax) // 2, (ymin + ymax) // 2
    if (ymax - ymin) > (xmax - xmin):
        return (cy, xmin - 4), (cy, xmax + 4)
    return (ymin - 4, cx), (ymax + 4, cx)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    white = np.all(arr > 245, axis=2)
    outside = _exterior(white)
    scores = {}
    for name, rgb in PALETTE.items():
        mask = np.max(np.abs(arr - rgb), axis=2) <= 3
        p, q = _side_samples(mask)
        h, w = white.shape
        if not all(0 <= y < h and 0 <= x < w and white[y, x] for y, x in (p, q)):
            raise ValueError("candidate sides are not visibly clear")
        scores[name] = int(not outside[p]) + int(not outside[q])
    best = max(scores.values())
    winners = [name for name, score in scores.items() if score == best]
    if len(winners) != 1 or best != 2:
        raise ValueError("no unique two-enclosed-side wall")
    return winners[0]
