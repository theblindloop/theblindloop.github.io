"""Independent pixel-only inverse for the edge-alternating badge world."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _runs(values: np.ndarray, threshold: float) -> list[tuple[int, int]]:
    mask = values > threshold
    result: list[tuple[int, int]] = []
    start = None
    for index, active in enumerate(mask.tolist() + [False]):
        if active and start is None:
            start = index
        elif not active and start is not None:
            if index - start >= 8:
                result.append((start, index - 1))
            start = None
    return result


def _slot_centers(array: np.ndarray, saturated: np.ndarray) -> list[float]:
    height, _ = saturated.shape
    # Prefer the four dark visible marker bodies, which survive badge ablations.
    dark = np.all(array < 85, axis=2)
    bottom = dark.copy()
    bottom[: int(height * 0.62)] = False
    marker_runs = _runs(bottom.sum(axis=0), 10)
    markers = [(left + right) / 2.0 for left, right in marker_runs if right - left >= 20]
    if len(markers) == 4:
        return markers

    # If a marker is damaged, the four full colored badge projections are an
    # independent slot source. These two sources are vertically separated.
    badge_runs = _runs(saturated.sum(axis=0), 12)
    badges = [(left + right) / 2.0 for left, right in badge_runs if right - left >= 24]
    if len(badges) == 4:
        return badges
    raise ValueError("could not enumerate four visible candidate slots")


def _cell_color(array: np.ndarray, x0: int, y0: int, x1: int, y1: int) -> np.ndarray:
    height, width = array.shape[:2]
    x0, x1 = max(0, x0), min(width, x1)
    y0, y1 = max(0, y0), min(height, y1)
    if x1 - x0 < 5 or y1 - y0 < 5:
        raise ValueError("candidate cell is out of bounds")
    pixels = array[y0:y1, x0:x1].reshape(-1, 3)
    color = np.median(pixels, axis=0)
    if float(color.max() - color.min()) < 45.0 or float(color.max()) > 250.0:
        raise ValueError("candidate cell lacks a complete colored area")
    return color.astype(np.float64)


def _score_badge(array: np.ndarray, center_x: float, center_y: float, side: int) -> int:
    half = side / 2.0
    quarter = side / 4.0
    inset = max(7, side // 14)
    colors = []
    for row in range(2):
        for column in range(2):
            cx = center_x + (-quarter if column == 0 else quarter)
            cy = center_y + (-quarter if row == 0 else quarter)
            colors.append(
                _cell_color(
                    array,
                    int(round(cx - quarter + inset)),
                    int(round(cy - quarter + inset)),
                    int(round(cx + quarter - inset)),
                    int(round(cy + quarter - inset)),
                )
            )

    # Build the two visible color classes without assuming palette names.
    representatives: list[np.ndarray] = []
    classes: list[int] = []
    for color in colors:
        match = next(
            (index for index, ref in enumerate(representatives)
             if float(np.linalg.norm(color - ref)) < 38.0),
            None,
        )
        if match is None:
            representatives.append(color)
            match = len(representatives) - 1
        classes.append(match)
    if len(representatives) != 2 or sorted(classes.count(i) for i in range(2)) != [2, 2]:
        raise ValueError("badge does not contain two balanced visible colors")
    edges = ((0, 1), (2, 3), (0, 2), (1, 3))
    return sum(classes[a] == classes[b] for a, b in edges)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    chroma = array.max(axis=2).astype(np.int16) - array.min(axis=2).astype(np.int16)
    saturated = (chroma > 45) & (array.max(axis=2) < 252)
    if int(saturated.sum()) < 1000:
        raise ValueError("insufficient colored evidence")
    centers = sorted(_slot_centers(array, saturated))
    ys, xs = np.nonzero(saturated)
    # Robust to destruction of any one badge because the other three share y extent.
    center_y = float(np.median(ys))
    badge_runs = _runs(saturated.sum(axis=0), 12)
    widths = [right - left + 1 for left, right in badge_runs if right - left >= 60]
    side = int(round(float(np.median(widths)))) if widths else 112
    if not 88 <= side <= 132:
        raise ValueError("colored candidate scale is ambiguous")

    scores = []
    for center in centers:
        try:
            score = _score_badge(array, center, center_y, side)
        except ValueError:
            score = 99
        scores.append(score)
    winners = [index for index, value in enumerate(scores) if value == 0]
    finite_runners = sorted(value for value in scores if value < 99)
    if len(winners) != 1 or len(finite_runners) < 2 or finite_runners[1] < 1:
        raise ValueError("no unique alternating winner with runner-up margin")
    return ("A", "B", "C", "D")[winners[0]]
