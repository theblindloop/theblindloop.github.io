"""Independent image-only inverse for the connected trace measurement.

Only the final RGB image is accepted.  The inverse segments a small public
palette, requires one connected route from the colored start to the terminal,
localizes the amber split marker, and follows a shortest raster path before
comparing upward pixel travel on the two ordered halves.
"""

from __future__ import annotations

from collections import deque
import math

import numpy as np
from PIL import Image

ROUTE_PALETTE = np.asarray(
    [[26, 110, 118], [32, 123, 104], [36, 95, 124], [23, 107, 100]], dtype=np.float32
)
START = np.asarray([112, 71, 173], dtype=np.float32)
TERMINAL = np.asarray([102, 114, 125], dtype=np.float32)
SPLIT = np.asarray([230, 163, 41], dtype=np.float32)
COLOR_TOLERANCE = 48.0
MARKER_TOLERANCE = 58.0
MIN_MARKER_PIXELS = 45
GRID_FACTOR = 2
MIN_PATH_CELLS = 120
MIN_DECISION_MARGIN = 6


def _mask(array: np.ndarray, palette: np.ndarray, tolerance: float) -> np.ndarray:
    if palette.ndim == 1:
        distance = np.linalg.norm(array - palette.reshape(1, 1, 3), axis=2)
        return distance <= tolerance
    distances = np.stack(
        [np.linalg.norm(array - color.reshape(1, 1, 3), axis=2) for color in palette], axis=0
    )
    return distances.min(axis=0) <= tolerance


def _downsample(mask: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    if height % GRID_FACTOR or width % GRID_FACTOR:
        raise ValueError("unexpected image dimensions")
    return mask.reshape(height // GRID_FACTOR, GRID_FACTOR, width // GRID_FACTOR, GRID_FACTOR).any(axis=(1, 3))


def _marker_centroid(mask: np.ndarray, label: str) -> tuple[float, float]:
    points = np.argwhere(mask)
    if len(points) < MIN_MARKER_PIXELS:
        raise ValueError(f"{label} marker is absent")
    center = points.mean(axis=0)
    return float(center[0]), float(center[1])


def _nearest_route_cell(route: np.ndarray, center: tuple[float, float]) -> tuple[int, int]:
    points = np.argwhere(route)
    if not len(points):
        raise ValueError("trace is absent")
    target = np.asarray(center, dtype=np.float32)
    distances = ((points.astype(np.float32) - target) ** 2).sum(axis=1)
    y, x = points[int(np.argmin(distances))]
    return int(y), int(x)


def _path(route: np.ndarray, start: tuple[int, int], terminal: tuple[int, int]) -> list[tuple[int, int]]:
    height, width = route.shape
    queue: deque[tuple[int, int]] = deque([start])
    parent: dict[tuple[int, int], tuple[int, int] | None] = {start: None}
    while queue:
        current = queue.popleft()
        if current == terminal:
            break
        cy, cx = current
        for dy in (-1, 0, 1):
            for dx in (-1, 0, 1):
                if not (dy or dx):
                    continue
                neighbor = (cy + dy, cx + dx)
                ny, nx = neighbor
                if 0 <= ny < height and 0 <= nx < width and route[ny, nx] and neighbor not in parent:
                    parent[neighbor] = current
                    queue.append(neighbor)
    if terminal not in parent:
        raise ValueError("trace does not form one complete start-to-terminal component")
    ordered: list[tuple[int, int]] = []
    cursor: tuple[int, int] | None = terminal
    while cursor is not None:
        ordered.append(cursor)
        cursor = parent[cursor]
    ordered.reverse()
    if len(ordered) < MIN_PATH_CELLS:
        raise ValueError("trace path is too short")
    return ordered


def _closest_index(path: list[tuple[int, int]], center: tuple[float, float]) -> int:
    target = np.asarray(center, dtype=np.float32)
    points = np.asarray(path, dtype=np.float32)
    distances = ((points - target) ** 2).sum(axis=1)
    index = int(np.argmin(distances))
    if index < len(path) * 0.25 or index > len(path) * 0.75:
        raise ValueError("amber marker is not an interior split")
    return index


def _upward_pixels(path: list[tuple[int, int]]) -> int:
    return sum(max(0, int(left[0]) - int(right[0])) for left, right in zip(path, path[1:]))


def decision_from_image(image: Image.Image) -> str:
    """Return ``before`` or ``after`` from the PNG alone, or abstain."""

    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    if array.shape[:2] != (512, 512):
        raise ValueError("unexpected canvas size")
    route = _downsample(_mask(array, ROUTE_PALETTE, COLOR_TOLERANCE))
    start = _downsample(_mask(array, START, MARKER_TOLERANCE))
    terminal = _downsample(_mask(array, TERMINAL, MARKER_TOLERANCE))
    split = _downsample(_mask(array, SPLIT, MARKER_TOLERANCE))
    start_center = _marker_centroid(start, "start")
    terminal_center = _marker_centroid(terminal, "terminal")
    split_center = _marker_centroid(split, "amber")
    start_cell = _nearest_route_cell(route, start_center)
    terminal_cell = _nearest_route_cell(route, terminal_center)
    ordered = _path(route, start_cell, terminal_cell)
    split_index = _closest_index(ordered, split_center)
    before = _upward_pixels(ordered[: split_index + 1])
    after = _upward_pixels(ordered[split_index:])
    if abs(before - after) < MIN_DECISION_MARGIN:
        raise ValueError("two integrated travel measurements are unresolved")
    return "before" if before > after else "after"
