"""Renderer and analytic arm for directed trace upward-variation comparison.

The trace has strictly monotone x coordinates, which makes it visibly a single
non-self-intersecting path while its y profile contains separated meanders.
The answer is the half with the larger sum of positive upward y displacements
when the path is followed from the colored start marker to its arrowhead.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
ROUTE_COLORS = ("#1a6e76", "#207b68", "#245f7c", "#176b64")
BACKGROUNDS = ("#ffffff", "#fcfdf9", "#fbfcff")
START_COLOR = "#7047ad"
TERMINAL_COLOR = "#66727d"
SPLIT_COLOR = "#e6a329"
STROKE = 7
MARKER_RADIUS = 13
ARROW_LENGTH = 25.0
ARROW_WIDTH = 12.0
MIN_MARGIN = 28.0


def _upward_travel(points: list[list[int]] | list[tuple[int, int]]) -> float:
    """Sum only positive vertical movement in image coordinates (y decreases up)."""

    return sum(max(0.0, float(left[1]) - float(right[1])) for left, right in zip(points, points[1:]))


def _profile_pair(rng: random.Random, first_is_rough: bool) -> list[int]:
    """Make two joined halves: one jagged and one gently varying."""

    rough = [0, -76, 10, -84, 8, -72, 12, -79, 0]
    calm = [0, 14, -8, 11, -5, 13, -6, 10, 0]
    first, second = (rough, calm) if first_is_rough else (calm, rough)
    profile = first + second[1:]
    # Small visible nuisance variation leaves the large rough-vs-calm margin
    # intact and prevents an exact template from being a useful shortcut.
    jitter = [0] + [rng.randint(-4, 4) for _ in range(len(profile) - 2)] + [0]
    return [int(a + b) for a, b in zip(profile, jitter)]


def _make_route(rng: random.Random) -> tuple[list[list[int]], int]:
    winner_before = bool(rng.randrange(2))
    base_y = rng.randint(222, 286)
    profile = _profile_pair(rng, winner_before)
    xs: list[int] = []
    for index in range(17):
        # The step is much larger than the jitter, so x remains strictly
        # monotone and non-adjacent segments cannot cross.
        xs.append(round(42.0 + index * 26.75 + rng.uniform(-3.0, 3.0)))
    ys = [base_y + value for value in profile]
    route = [[x, y] for x, y in zip(xs, ys)]
    if rng.randrange(2):
        route = [[WIDTH - x, y] for x, y in reversed(route)]
    if rng.randrange(2):
        route = [[x, HEIGHT - y] for x, y in route]
    return route, 8


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one balanced, deterministic wide-span trace scene."""

    rng = random.Random(int(seed))
    route, split_index = _make_route(rng)
    return {
        "route": route,
        "split_index": split_index,
        "route_color": ROUTE_COLORS[rng.randrange(len(ROUTE_COLORS))],
        "background": BACKGROUNDS[rng.randrange(len(BACKGROUNDS))],
        "stroke": STROKE,
        "start_color": START_COLOR,
        "terminal_color": TERMINAL_COLOR,
        "split_color": SPLIT_COLOR,
        "marker_radius": MARKER_RADIUS,
    }


def _draw_marker(draw: ImageDraw.ImageDraw, point: list[int], color: str, radius: int) -> None:
    x, y = point
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)


def _arrow_polygon(route: list[list[int]], length: float, width: float) -> list[tuple[float, float]]:
    left, right = route[-2], route[-1]
    dx, dy = float(right[0] - left[0]), float(right[1] - left[1])
    norm = math.hypot(dx, dy)
    ux, uy = dx / norm, dy / norm
    nx, ny = -uy, ux
    tip = (right[0] + ux * length, right[1] + uy * length)
    base = (right[0] - ux * 3.0, right[1] - uy * 3.0)
    return [tip, (base[0] + nx * width, base[1] + ny * width), (base[0] - nx * width, base[1] - ny * width)]


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the final RGB raster; markers precede the continuous trace."""

    image = Image.new("RGB", (WIDTH, HEIGHT), scene["background"])
    draw = ImageDraw.Draw(image)
    route = scene["route"]
    radius = int(scene["marker_radius"])
    _draw_marker(draw, route[0], scene["start_color"], radius)
    _draw_marker(draw, route[-1], scene["terminal_color"], radius)
    _draw_marker(draw, route[int(scene["split_index"])], scene["split_color"], radius)
    draw.line([tuple(point) for point in route], fill=scene["route_color"], width=int(scene["stroke"]), joint="curve")
    draw.polygon(_arrow_polygon(route, ARROW_LENGTH, ARROW_WIDTH), fill=scene["route_color"])
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Compute the visible measurement from the route and the visible split."""

    route = scene["route"]
    split_index = int(scene["split_index"])
    before = _upward_travel(route[: split_index + 1])
    after = _upward_travel(route[split_index:])
    return "before" if before > after else "after"


def margin(scene: dict[str, Any]) -> float:
    route = scene["route"]
    split_index = int(scene["split_index"])
    before = _upward_travel(route[: split_index + 1])
    after = _upward_travel(route[split_index:])
    return abs(before - after)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < MIN_MARGIN


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No nontrivial latent aliases: every label input is visibly rendered."""

    return []
