"""Scene-free inverse arm for the final PNG.

The oracle discovers the repeated body-fill color from raster statistics, splits
that color into vertically separated components, measures their pixel bounding
diameters, and reports the tier of the unique maximum. It receives no palette,
coordinates, scene dictionary, filename, or analytic answer.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

TIERS = ("top", "middle", "bottom")


def _runs(values: np.ndarray) -> list[tuple[int, int]]:
    if len(values) == 0:
        return []
    cuts = np.flatnonzero(np.diff(values) > 1)
    starts = np.r_[0, cuts + 1]
    ends = np.r_[cuts, len(values) - 1]
    return [(int(values[a]), int(values[b])) for a, b in zip(starts, ends)]


def measure_from_image(image: Image.Image) -> dict:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.ndim != 3 or arr.shape[0] < 100 or arr.shape[1] < 100:
        raise ValueError("image is too small for three body components")
    colors, counts = np.unique(arr.reshape(-1, 3), axis=0, return_counts=True)
    order = np.argsort(counts)[::-1]
    viable = []
    for color_index in order[:80]:
        color = colors[color_index]
        count = int(counts[color_index])
        mean = float(color.mean())
        spread = int(color.max()) - int(color.min())
        if count < 1500 or not (95.0 < mean < 235.0) or spread < 15:
            continue
        mask = np.all(arr == color, axis=2)
        row_ids = np.flatnonzero(mask.any(axis=1))
        components = []
        for y0, y1 in _runs(row_ids):
            slab = mask[y0:y1 + 1]
            area = int(slab.sum())
            xs = np.flatnonzero(slab.any(axis=0))
            if len(xs) == 0:
                continue
            x0, x1 = int(xs[0]), int(xs[-1])
            height = y1 - y0 + 1
            width = x1 - x0 + 1
            if area >= 500 and height >= 30 and width >= 30:
                components.append({
                    "x0": x0, "x1": x1, "y0": y0, "y1": y1,
                    "area": area, "width": width, "height": height,
                })
        if len(components) == 3:
            components.sort(key=lambda item: (item["y0"] + item["y1"]) / 2.0)
            score = sum(item["area"] for item in components)
            viable.append((score, tuple(int(v) for v in color), components))
    if not viable:
        raise ValueError("could not discover three repeated body-fill components")
    _, color, components = max(viable, key=lambda item: item[0])
    diameters = [max(item["width"], item["height"]) for item in components]
    ranked = sorted(range(3), key=lambda index: diameters[index], reverse=True)
    runner_up_gap = float(diameters[ranked[0]] - diameters[ranked[1]])
    if runner_up_gap < 8.0:
        raise ValueError(f"ambiguous top-two diameter gap: {runner_up_gap:.1f}px")
    return {
        "decision": TIERS[ranked[0]],
        "runner_up_margin": runner_up_gap,
        "body_color": list(color),
        "diameters": diameters,
        "components": components,
    }


def decision_from_image(image: Image.Image) -> str:
    return str(measure_from_image(image)["decision"])
