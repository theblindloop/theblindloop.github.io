"""Deterministic scene sampler and renderer for the snowman tier world."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw

WIDTH = 448
HEIGHT = 512
SCALE = 3
GAP = 9
RADII = (46, 60, 74)
TIERS = ("top", "middle", "bottom")
BODY_PALETTES = (
    (126, 210, 232),
    (171, 178, 232),
    (139, 211, 184),
    (224, 164, 190),
)
BACKGROUNDS = ((248, 251, 253), (252, 248, 241))


def sample_scene(seed: int) -> dict:
    """Return one of all six size orders plus independent nuisance factors."""
    rng = random.Random(int(seed) ^ 0x5A17B04D)
    permutations = tuple(itertools.permutations(RADII))
    return {
        "tier_radii": list(permutations[int(seed) % len(permutations)]),
        "palette_index": rng.randrange(len(BODY_PALETTES)),
        "center_x": rng.randint(214, 234),
        "mirror": bool(rng.randrange(2)),
        "decoration_variant": rng.randrange(3),
        "background_variant": rng.randrange(2),
    }


def _layout(scene: dict) -> list[tuple[int, int, int]]:
    radii = [int(value) for value in scene["tier_radii"]]
    x = int(scene["center_x"])
    y0 = 75 + radii[0]
    y1 = y0 + radii[0] + GAP + radii[1]
    y2 = y1 + radii[1] + GAP + radii[2]
    return [(x, y0, radii[0]), (x, y1, radii[1]), (x, y2, radii[2])]


def _box(cx: float, cy: float, radius: float) -> tuple[int, int, int, int]:
    return tuple(int(round(v * SCALE)) for v in (cx - radius, cy - radius, cx + radius, cy + radius))


def _points(points: list[tuple[float, float]]) -> list[tuple[int, int]]:
    return [(int(round(x * SCALE)), int(round(y * SCALE))) for x, y in points]


def render(scene: dict) -> Image.Image:
    """Render an antialiased, matched-style three-tier snowman."""
    background = BACKGROUNDS[int(scene["background_variant"]) % len(BACKGROUNDS)]
    body = BODY_PALETTES[int(scene["palette_index"]) % len(BODY_PALETTES)]
    canvas = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), background)
    draw = ImageDraw.Draw(canvas)
    layout = _layout(scene)
    x = int(scene["center_x"])
    mirror = -1 if bool(scene["mirror"]) else 1
    decor = int(scene["decoration_variant"])
    dark = (43, 53, 62)
    brown = (105, 72, 48)
    red = (202, 66, 74)
    orange = (238, 132, 45)
    snow_shadow = (218, 226, 232)

    # Ground and stick arms sit behind the body, so all body rims remain legible.
    draw.ellipse(_box(x, 469, 112), fill=snow_shadow)
    _, middle_y, middle_r = layout[1]
    arm_lift = (-18, 4, 20)[decor]
    left = [(x - middle_r + 2, middle_y - 8), (x - middle_r - 45, middle_y - 37 - arm_lift)]
    right = [(x + middle_r - 2, middle_y - 8), (x + middle_r + 45, middle_y - 37 + arm_lift)]
    if mirror < 0:
        left, right = ([(2 * x - px, py) for px, py in right], [(2 * x - px, py) for px, py in left])
    draw.line(_points(left), fill=brown, width=6 * SCALE)
    draw.line(_points(right), fill=brown, width=6 * SCALE)
    for branch, direction in ((left, -1), (right, 1)):
        ex, ey = branch[-1]
        draw.line(_points([(ex, ey), (ex + 16 * direction, ey - 13)]), fill=brown, width=4 * SCALE)

    # Fixed multiset of radii: answer identity changes without changing total body ink.
    for cx, cy, radius in reversed(layout):
        draw.ellipse(_box(cx, cy, radius), fill=body, outline=dark, width=5 * SCALE)

    top_x, top_y, top_r = layout[0]
    # Hat, face, scarf, and buttons make the iconic prior salient but never encode the answer.
    brim_y = top_y - top_r - 5
    draw.rounded_rectangle(
        (int((top_x - 39) * SCALE), int((brim_y - 4) * SCALE),
         int((top_x + 39) * SCALE), int((brim_y + 5) * SCALE)),
        radius=3 * SCALE,
        fill=dark,
    )
    draw.rectangle(
        (int((top_x - 27) * SCALE), int((brim_y - 38) * SCALE),
         int((top_x + 27) * SCALE), int((brim_y - 3) * SCALE)),
        fill=dark,
    )
    eye_y = top_y - max(10, top_r * 0.20)
    eye_dx = max(12, top_r * 0.28)
    for ex in (top_x - eye_dx, top_x + eye_dx):
        draw.ellipse(_box(ex, eye_y, 4), fill=dark)
    nose_y = top_y + 1
    nose_tip = top_x + mirror * min(30, top_r * 0.54)
    draw.polygon(_points([(top_x - 2, nose_y - 5), (top_x - 2, nose_y + 5), (nose_tip, nose_y)]), fill=orange)
    mouth_y = top_y + max(14, top_r * 0.31)
    for dx in (-13, 0, 13):
        draw.ellipse(_box(top_x + dx, mouth_y + abs(dx) * 0.11, 2.5), fill=dark)

    # The scarf occupies the inter-tier gap, not a body diameter extremum.
    scarf_y = top_y + top_r + GAP / 2
    draw.rounded_rectangle(
        (int((x - 34) * SCALE), int((scarf_y - 4) * SCALE),
         int((x + 34) * SCALE), int((scarf_y + 4) * SCALE)),
        radius=3 * SCALE,
        fill=red,
    )
    tail_side = mirror
    draw.polygon(_points([
        (x + tail_side * 20, scarf_y + 2),
        (x + tail_side * 38, scarf_y + 34),
        (x + tail_side * 23, scarf_y + 39),
        (x + tail_side * 10, scarf_y + 5),
    ]), fill=red)

    # Same decoration count for all answer classes; only its harmless layout changes.
    for tier_index in (1, 2):
        cx, cy, radius = layout[tier_index]
        offsets = (-0.24, 0.04, 0.29) if decor != 1 else (-0.29, -0.02, 0.25)
        for frac in offsets:
            draw.ellipse(_box(cx, cy + radius * frac, 4), fill=dark)

    return canvas.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    radii = [int(value) for value in scene["tier_radii"]]
    return TIERS[max(range(3), key=lambda index: radii[index])]


def margin(scene: dict) -> float:
    radii = sorted((int(value) for value in scene["tier_radii"]), reverse=True)
    return float(2 * (radii[0] - radii[1]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Exercise two real representation aliases that leave the raster unchanged."""
    palette_twin = dict(scene)
    palette_twin["palette_index"] = int(scene["palette_index"]) + len(BODY_PALETTES)
    background_twin = dict(scene)
    background_twin["background_variant"] = int(scene["background_variant"]) + len(BACKGROUNDS)
    return [("cyclic_palette_index_alias", palette_twin), ("cyclic_background_index_alias", background_twin)]
