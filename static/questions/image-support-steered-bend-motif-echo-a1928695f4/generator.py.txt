"""Deterministic renderer for a paired bend-motif trail."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 320
BACKGROUND = (247, 246, 242)
PALETTES = (
    ((35, 43, 54), (236, 142, 47), (112, 78, 190), (33, 171, 154)),
    ((39, 45, 48), (112, 78, 190), (33, 171, 154), (236, 142, 47)),
    ((43, 39, 52), (224, 72, 91), (42, 157, 189), (236, 165, 48)),
    ((32, 47, 48), (42, 157, 189), (236, 165, 48), (224, 72, 91)),
)


def _motifs() -> tuple[tuple[int, ...], ...]:
    """Balanced non-template motifs whose cumulative excursions are symmetric."""
    output = []
    for values in itertools.product((-1, 1), repeat=8):
        if sum(values) != 0:
            continue
        cumulative = [0]
        for value in values:
            cumulative.append(cumulative[-1] + value)
        if max(cumulative) == -min(cumulative) and len(set(values)) == 2:
            output.append(values)
    return tuple(output)


MOTIFS = _motifs()


def sample_scene(seed: int) -> dict:
    """Adjacent seeds are a nuisance-matched same/opposite pair."""
    rng = random.Random(seed // 2)
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "x_start": rng.uniform(30.0, 42.0),
        "x_end": rng.uniform(470.0, 482.0),
        "base_y": rng.uniform(139.0, 181.0),
        "amplitude": rng.uniform(19.0, 24.0),
        "stroke": rng.choice((8.0, 9.0, 10.0)),
        "motif": list(rng.choice(MOTIFS)),
        "relation": "same" if seed % 2 == 0 else "opposite",
        "palette": (seed // 2) % len(PALETTES),
    }


def _vertices(scene: dict) -> list[tuple[float, float]]:
    motif = [int(value) for value in scene["motif"]]
    relation = str(scene["relation"])
    second = motif if relation == "same" else [-value for value in motif]
    signs = motif + second
    x0 = float(scene["x_start"])
    x1 = float(scene["x_end"])
    step = (x1 - x0) / len(signs)
    amplitude = float(scene["amplitude"])
    y = float(scene["base_y"])
    points = [(x0, y)]
    for index, sign in enumerate(signs, start=1):
        y += amplitude * sign
        points.append((x0 + step * index, y))
    return points


def render(scene: dict) -> Image.Image:
    scale = 4
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    trail, round_color, diamond_color, square_color = PALETTES[int(scene["palette"])]
    points = _vertices(scene)
    scaled = [(round(x * scale), round(y * scale)) for x, y in points]
    draw.line(
        scaled,
        fill=trail,
        width=round(float(scene["stroke"]) * scale),
        joint="curve",
    )

    # These small visible beads expose all sixteen segment boundaries even when
    # adjacent slope signs match and would otherwise form one collinear run.
    bead_radius = 4.0 * scale
    for index, (bx, by) in enumerate(scaled[1:-1], start=1):
        if index == 8:
            continue
        draw.ellipse(
            (bx - bead_radius, by - bead_radius, bx + bead_radius, by + bead_radius),
            fill=diamond_color,
        )

    sx, sy = scaled[0]
    radius = 13.0 * scale
    draw.ellipse((sx - radius, sy - radius, sx + radius, sy + radius), fill=round_color)

    mx, my = scaled[8]
    radius = 15.0 * scale
    draw.polygon(
        ((mx, my - radius), (mx + radius, my), (mx, my + radius), (mx - radius, my)),
        fill=diamond_color,
    )

    ex, ey = scaled[-1]
    half = 12.0 * scale
    draw.rectangle((ex - half, ey - half, ex + half, ey + half), fill=square_color)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["relation"])


def margin(scene: dict) -> float:
    run = (float(scene["x_end"]) - float(scene["x_start"])) / 16.0
    return float(scene["amplitude"]) / run


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.45


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
