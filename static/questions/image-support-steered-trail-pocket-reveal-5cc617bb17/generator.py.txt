"""Analytic scene sampler and independent forward rasterizer."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
SUPERSAMPLE = 4
PALETTE = ("#17324d", "#4b2142", "#16453a", "#3d315b")


def sample_scene(seed: int) -> dict:
    """Sample balanced topology with independently randomized visible nuisances."""
    rng = random.Random(int(seed))
    pocket_count = 1 + (int(seed) & 1)
    orientation = "horizontal" if rng.random() < 0.5 else "vertical"
    # In the long direction, a grid boundary centers the 84--91 px trace inside
    # two overlapping 8x8 ablation columns.  The short direction uses a cell
    # center, keeping decisive support to one row.  Choices randomize location.
    long_center = rng.choice((128, 192, 256, 320, 384))
    short_center = rng.choice((96, 160, 224, 288, 352, 416))
    if orientation == "horizontal":
        cx, cy = long_center, short_center
    else:
        cx, cy = short_center, long_center
    radius_long = rng.choice((20, 21, 22))
    radius_short = rng.choice((13, 14, 15))
    stroke = rng.choice((5, 6))
    # Every sixth seed is a visible, still-answerable near-boundary case.
    gap_degrees = 22 if int(seed) % 6 == 0 else rng.choice((30, 34, 38))
    return {
        "pocket_count": pocket_count,
        "cx": cx,
        "cy": cy,
        "orientation": orientation,
        "radius_long": radius_long,
        "radius_short": radius_short,
        "stroke": stroke,
        "gap_degrees": gap_degrees,
        "open_lobe": rng.choice(("first", "second")),
        "palette_index": rng.randrange(len(PALETTE)),
    }


def _point(scene: dict, u: float, v: float) -> tuple[float, float]:
    if scene["orientation"] == "horizontal":
        return scene["cx"] + u, scene["cy"] + v
    return scene["cx"] + v, scene["cy"] + u


def _ellipse_points(
    scene: dict, center_u: float, start_deg: float = 0.0, end_deg: float = 360.0
) -> list[tuple[float, float]]:
    steps = max(72, int(abs(end_deg - start_deg) * 0.45))
    points = []
    for index in range(steps + 1):
        angle = math.radians(start_deg + (end_deg - start_deg) * index / steps)
        u = center_u + scene["radius_long"] * math.cos(angle)
        v = scene["radius_short"] * math.sin(angle)
        points.append(_point(scene, u, v))
    return points


def render(scene: dict) -> Image.Image:
    """Draw two touching loop traces; exactly one is opened for the one-pocket class."""
    scale = SUPERSAMPLE
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), "white")
    draw = ImageDraw.Draw(image)
    color = PALETTE[int(scene["palette_index"])]
    stroke = int(scene["stroke"] * scale)
    separation = 2.0 * scene["radius_long"] - 0.75 * scene["stroke"]
    centers = (-separation / 2.0, separation / 2.0)

    for center_u in centers:
        points = _ellipse_points(scene, center_u)
        draw.line(
            [(round(x * scale), round(y * scale)) for x, y in points],
            fill=color,
            width=stroke,
            joint="curve",
        )

    if scene["pocket_count"] == 1:
        lobe_index = 0 if scene["open_lobe"] == "first" else 1
        outer_angle = 180.0 if lobe_index == 0 else 0.0
        half = float(scene["gap_degrees"]) / 2.0
        gap_points = _ellipse_points(
            scene, centers[lobe_index], outer_angle - half, outer_angle + half
        )
        # A white over-stroke opens the contour by more than the antialias fringe.
        draw.line(
            [(round(x * scale), round(y * scale)) for x, y in gap_points],
            fill="white",
            width=int((scene["stroke"] + 3) * scale),
            joint="curve",
        )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "one pocket" if scene["pocket_count"] == 1 else "two pockets"


def margin(scene: dict) -> float:
    """Conservative pixel clearance to the opposite complement topology."""
    if scene["pocket_count"] == 2:
        return float(min(scene["stroke"], 2 * scene["radius_short"] - scene["stroke"]))
    aperture = 2.0 * scene["radius_short"] * math.sin(
        math.radians(float(scene["gap_degrees"])) / 2.0
    )
    return float(aperture)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
