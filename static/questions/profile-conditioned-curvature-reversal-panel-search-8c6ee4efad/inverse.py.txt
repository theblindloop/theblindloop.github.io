"""Independent pixel-only inverse for the curvature-reversal search."""

from __future__ import annotations

import numpy as np
from PIL import Image

PANEL_RECTS = (
    (70, 80, 470, 340),
    (530, 80, 930, 340),
    (70, 420, 470, 680),
    (530, 420, 930, 680),
)
LABELS = ("A", "B", "C", "D")
QUARANTINE_GAP = 0.5


def _is_stroke_pixel(pixel: tuple[int, int, int]) -> bool:
    red, green, blue = pixel
    return red < 60 and green < 80 and blue < 100


def _largest_component(image: Image.Image, rect: tuple[int, int, int, int]):
    x0, y0, x1, y1 = rect
    # The interior crop excludes the panel border and the label, leaving the
    # single open stroke as the only eligible dark connected component.
    left, top, right, bottom = x0 + 28, y0 + 48, x1 - 28, y1 - 28
    pixels = image.load()
    visited: set[tuple[int, int]] = set()
    largest: list[tuple[int, int]] = []
    for y in range(top, bottom + 1):
        for x in range(left, right + 1):
            if (x, y) in visited or not _is_stroke_pixel(pixels[x, y]):
                continue
            stack = [(x, y)]
            visited.add((x, y))
            component: list[tuple[int, int]] = []
            while stack:
                px, py = stack.pop()
                component.append((px, py))
                for nx, ny in (
                    (px - 1, py),
                    (px + 1, py),
                    (px, py - 1),
                    (px, py + 1),
                ):
                    if (
                        nx < left
                        or nx > right
                        or ny < top
                        or ny > bottom
                        or (nx, ny) in visited
                        or not _is_stroke_pixel(pixels[nx, ny])
                    ):
                        continue
                    visited.add((nx, ny))
                    stack.append((nx, ny))
            if len(component) > len(largest):
                largest = component
    return largest


def _centerline(component: list[tuple[int, int]]) -> tuple[np.ndarray, np.ndarray] | None:
    if len(component) < 800:
        return None
    by_x: dict[int, list[int]] = {}
    for x, y in component:
        by_x.setdefault(x, []).append(y)
    columns = sorted(by_x)
    if len(columns) < 80 or columns[-1] - columns[0] < 160:
        return None
    xs = np.asarray(columns, dtype=float)
    ys = np.asarray(
        [float(np.median(np.asarray(by_x[column], dtype=float))) for column in columns],
        dtype=float,
    )
    return xs, ys


def _reversal_score(component: list[tuple[int, int]]) -> float | None:
    centerline = _centerline(component)
    if centerline is None:
        return None
    xs, ys = centerline
    normalized_x = (xs - float(xs.mean())) / max(float(np.ptp(xs)) / 2.0, 1.0)
    try:
        coefficients = np.polyfit(normalized_x, ys, 3)
    except (FloatingPointError, np.linalg.LinAlgError, ValueError):
        return None
    # A cubic centerline has an affine second derivative. Sampling away from
    # its possible zero crossing makes the discrete sign change observable.
    sample_x = np.asarray((-0.8, -0.6, -0.4, -0.2, 0.2, 0.4, 0.6, 0.8))
    second = 6.0 * coefficients[0] * sample_x + 2.0 * coefficients[1]
    signs = [1 if value > 1.0 else -1 if value < -1.0 else 0 for value in second]
    signs = [sign for sign in signs if sign]
    return float(sum(left != right for left, right in zip(signs, signs[1:])))


def decision_from_image(image: Image.Image) -> str:
    """Return a panel label from the supplied raster only, or ABSTAIN."""

    try:
        rgb = image.convert("RGB")
        scores = []
        for rect in PANEL_RECTS:
            score = _reversal_score(_largest_component(rgb, rect))
            if score is None:
                return "ABSTAIN"
            scores.append(score)
        order = sorted(range(4), key=lambda index: scores[index], reverse=True)
        if scores[order[0]] != 1.0:
            return "ABSTAIN"
        if scores[order[0]] - scores[order[1]] < QUARANTINE_GAP:
            return "ABSTAIN"
        return LABELS[order[0]]
    except (ArithmeticError, IndexError, TypeError, ValueError):
        return "ABSTAIN"
