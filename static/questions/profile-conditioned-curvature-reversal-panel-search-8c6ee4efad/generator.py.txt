"""Deterministic renderer and analytic labeler for the curvature search world."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1000
HEIGHT = 760
PANEL_RECTS = (
    (70, 80, 470, 340),
    (530, 80, 930, 340),
    (70, 420, 470, 680),
    (530, 420, 930, 680),
)
LABELS = ("A", "B", "C", "D")
PANEL_FILL = (250, 251, 253)
PANEL_OUTLINE = (155, 166, 178)
TEXT_COLOR = (78, 92, 108)
STROKE_COLOR = (25, 43, 64)
STROKE_WIDTH = 10
LOCAL_XS = tuple(range(68, 325, 16))
QUARANTINE_GAP = 0.5


def _turn_sign(a: list[int], b: list[int], c: list[int]) -> int:
    """Return the signed local bend of three ordered centerline points."""

    first_x = b[0] - a[0]
    first_y = b[1] - a[1]
    second_x = c[0] - b[0]
    second_y = c[1] - b[1]
    cross = first_x * second_y - first_y * second_x
    if abs(cross) < 1.0:
        return 0
    return 1 if cross > 0 else -1


def _reversal_count(points: list[list[int]]) -> int:
    signs = [
        sign
        for index in range(1, len(points) - 1)
        if (sign := _turn_sign(points[index - 1], points[index], points[index + 1]))
    ]
    return sum(left != right for left, right in zip(signs, signs[1:]))


def _panel_scores(scene: dict) -> list[float]:
    return [float(_reversal_count(panel["points"])) for panel in scene["panels"]]


def analytic_gold(scene: dict) -> str:
    scores = _panel_scores(scene)
    best = max(range(len(scores)), key=lambda index: scores[index])
    return LABELS[best]


def margin(scene: dict) -> float:
    ordered = sorted(_panel_scores(scene), reverse=True)
    if len(ordered) < 2:
        return 0.0
    return float(ordered[0] - ordered[1])


def is_quarantined(scene: dict) -> bool:
    scores = _panel_scores(scene)
    return not scores or max(scores) != 1.0 or margin(scene) < QUARANTINE_GAP


def _curve_points(
    origin: tuple[int, int],
    x_shift: int,
    y_shift: int,
    *,
    target: bool,
    vertical_sign: int,
) -> list[list[int]]:
    """Construct either the one-reversal cubic or a constant-sign parabola."""

    origin_x, origin_y = origin
    center_x = 196.0
    if target:
        center_y = 130.0
    else:
        # The two signs use opposite vertical offsets so all curve masks have
        # the same local height range despite the slanted parabola.
        center_y = 135.0 if vertical_sign > 0 else 125.0
    points = []
    for local_x in LOCAL_XS:
        u = (local_x - center_x) / 128.0
        if target:
            value = vertical_sign * 115.0 * (u**3 - 0.78 * u)
        else:
            value = vertical_sign * (30.0 * (u * u - 1.0) + 25.0 * u)
        points.append(
            [
                origin_x + local_x + x_shift,
                origin_y + int(round(center_y + value + y_shift)),
            ]
        )
    return points


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target_index = int(seed) % 4
    panels = []
    for index, rect in enumerate(PANEL_RECTS):
        origin = (rect[0], rect[1])
        x_shift = rng.randint(-8, 8)
        y_shift = rng.randint(-5, 5)
        vertical_sign = 1 if rng.randrange(2) == 0 else -1
        panels.append(
            {
                "points": _curve_points(
                    origin,
                    x_shift,
                    y_shift,
                    target=index == target_index,
                    vertical_sign=vertical_sign,
                )
            }
        )
    scene = {"panels": panels}
    if is_quarantined(scene):
        raise RuntimeError(f"seed {seed} produced a quarantined scene")
    return scene


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), "white")
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    for index, panel in enumerate(scene["panels"]):
        x0, y0, x1, y1 = PANEL_RECTS[index]
        draw.rounded_rectangle(
            (x0, y0, x1, y1),
            radius=12,
            fill=PANEL_FILL,
            outline=PANEL_OUTLINE,
            width=2,
        )
        draw.text((x0 + 14, y0 + 10), LABELS[index], fill=TEXT_COLOR, font=font)
        points = [tuple(point) for point in panel["points"]]
        # Draw each undirected segment in canonical endpoint order. This keeps
        # the raster exactly invariant when the latent point list is reversed.
        for first, second in zip(points, points[1:]):
            if first > second:
                first, second = second, first
            draw.line(
                [first, second],
                fill=STROKE_COLOR,
                width=STROKE_WIDTH,
            )
    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    for panel in twin["panels"]:
        panel["points"] = list(reversed(panel["points"]))
    return [("reverse_point_order", twin)]
