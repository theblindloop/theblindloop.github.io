"""Independent PNG-only inverse for the coupled transition table."""

from __future__ import annotations

import numpy as np
from PIL import Image

EXPECTED_SIZE = (900, 650)
STATE_COUNT, ROUNDS = 4, 5
TABLE_X, TABLE_Y, TABLE_CELL, TABLE_STEP = 110, 130, 56, 64
HEADER_SIZE, HEADER_TOP, ROW_HEADER_LEFT = 44, 72, 48
START_TOP, START_SIZE = 112, 56
START_LEFTS = (565, 685)
OPTION_TOPS = (365, 417, 469, 521, 573)
OPTION_LEFTS, OPTION_SIZE = (585, 685), 42
PALETTE = np.array(
    ((201, 79, 85), (223, 139, 36), (38, 146, 127), (69, 111, 179)),
    dtype=np.float32,
)
NAVY = np.array((37, 59, 86), dtype=np.float32)


class OracleError(ValueError):
    pass


def _read_chip(array: np.ndarray, left: int, top: int, size: int) -> int:
    half = size // 2
    radius = max(2, size // 14)
    offsets = ((half, 9), (half, size - 9), (9, half), (size - 9, half))
    patches = []
    for dx, dy in offsets:
        patches.append(array[top + dy - radius : top + dy + radius + 1, left + dx - radius : left + dx + radius + 1])
    pixels = np.concatenate([patch.reshape(-1, 3) for patch in patches]).astype(np.float32)
    mean = pixels.mean(axis=0)
    distances = np.linalg.norm(PALETTE - mean, axis=1)
    order = np.argsort(distances)
    if float(distances[order[0]]) > 13.0 or float(distances[order[1]] - distances[order[0]]) < 35.0:
        raise OracleError("missing or ambiguous state chip")
    if float(np.linalg.norm(pixels - PALETTE[order[0]], axis=1).mean()) > 14.0:
        raise OracleError("state chip is materially corrupted")
    return int(order[0])


def _read_table(array: np.ndarray) -> list[list[int]]:
    table = []
    for row in range(STATE_COUNT):
        values = []
        for column in range(STATE_COUNT):
            values.append(
                _read_chip(
                    array,
                    TABLE_X + column * TABLE_STEP,
                    TABLE_Y + row * TABLE_STEP,
                    TABLE_CELL,
                )
            )
        table.append(values)
    return table


def _validate_axis_keys(array: np.ndarray) -> None:
    for state in range(STATE_COUNT):
        centre_x = TABLE_X + state * TABLE_STEP + TABLE_CELL // 2
        top_state = _read_chip(array, centre_x - HEADER_SIZE // 2, HEADER_TOP, HEADER_SIZE)
        centre_y = TABLE_Y + state * TABLE_STEP + TABLE_CELL // 2
        left_state = _read_chip(array, ROW_HEADER_LEFT, centre_y - HEADER_SIZE // 2, HEADER_SIZE)
        if top_state != state or left_state != state:
            raise OracleError("row or column key is missing or permuted")


def _read_pair(array: np.ndarray, lefts: tuple[int, int], top: int, size: int) -> tuple[int, int]:
    return (_read_chip(array, lefts[0], top, size), _read_chip(array, lefts[1], top, size))


def _validate_option_labels(array: np.ndarray) -> None:
    for top in OPTION_TOPS:
        cy = top + OPTION_SIZE // 2
        patch = array[cy - 17 : cy + 18, 478:513].astype(np.float32)
        navy_count = int((np.linalg.norm(patch - NAVY, axis=2) < 18.0).sum())
        white_count = int((patch.mean(axis=2) > 235.0).sum())
        if navy_count < 650 or white_count < 22:
            raise OracleError("an option label disk or glyph is missing")


def decision_from_image(image: Image.Image) -> str:
    if image.size != EXPECTED_SIZE:
        raise OracleError(f"unexpected raster size {image.size}")
    array = np.asarray(image.convert("RGB"))
    _validate_axis_keys(array)
    _validate_option_labels(array)
    table = _read_table(array)
    left, right = _read_pair(array, START_LEFTS, START_TOP, START_SIZE)
    options = [_read_pair(array, OPTION_LEFTS, top, OPTION_SIZE) for top in OPTION_TOPS]
    if len(set(options)) != 5:
        raise OracleError("option pairs are not distinct")
    for _ in range(ROUNDS):
        old_left, old_right = left, right
        left = table[old_left][old_right]
        right = table[old_right][old_left]
    matches = [index for index, option in enumerate(options) if option == (left, right)]
    if len(matches) != 1:
        raise OracleError("recovered final pair does not match exactly one option")
    return "ABCDE"[matches[0]]
