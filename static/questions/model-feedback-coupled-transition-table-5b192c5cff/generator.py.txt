"""Deterministic sampler, renderer, and analytic coupled-state recurrence."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 900, 650
STATE_COUNT, ROUNDS = 4, 5
TABLE_X, TABLE_Y, TABLE_CELL, TABLE_STEP = 110, 130, 56, 64
HEADER_SIZE, HEADER_TOP, ROW_HEADER_LEFT = 44, 72, 48
START_TOP, START_SIZE = 112, 56
START_LEFTS = (565, 685)
OPTION_TOPS = (365, 417, 469, 521, 573)
OPTION_LEFTS, OPTION_SIZE = (585, 685), 42
NAVY = "#253b56"
GOLD = "#d18a17"
BACKGROUND = "#ffffff"
PANEL = "#f2f5f7"
GRID = "#b7c2ca"
STATE_COLORS = ("#c94f55", "#df8b24", "#26927f", "#456fb3")
FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"

Pair = tuple[int, int]
Table = tuple[tuple[int, ...], ...]


def _state(value: object) -> int:
    result = int(value)
    if result not in range(STATE_COUNT):
        raise ValueError("state must be in 0..3")
    return result


def _pair(value: object) -> Pair:
    if not isinstance(value, (list, tuple)) or len(value) != 2:
        raise ValueError("ordered state pair must have length two")
    return (_state(value[0]), _state(value[1]))


def _table(value: object) -> Table:
    if not isinstance(value, (list, tuple)) or len(value) != STATE_COUNT:
        raise ValueError("table must have four rows")
    rows: list[tuple[int, ...]] = []
    for row in value:
        if not isinstance(row, (list, tuple)) or len(row) != STATE_COUNT:
            raise ValueError("every table row must have four entries")
        rows.append(tuple(_state(item) for item in row))
    return tuple(rows)


def _valid_scene(scene: dict) -> tuple[Table, Pair, list[Pair]]:
    if set(scene) != {"table", "start_states", "options"}:
        raise ValueError("scene fields do not match declared label inputs")
    table = _table(scene["table"])
    start = _pair(scene["start_states"])
    if not isinstance(scene["options"], list):
        raise ValueError("options must be a list")
    options = [_pair(value) for value in scene["options"]]
    if len(options) != 5 or len(set(options)) != 5:
        raise ValueError("five distinct ordered-pair options are required")
    return table, start, options


def _step(pair: Pair, table: Table) -> Pair:
    left, right = pair
    return (table[left][right], table[right][left])


def _evolve(pair: Pair, table: Table, rounds: int = ROUNDS) -> Pair:
    result = pair
    for _ in range(int(rounds)):
        result = _step(result, table)
    return result


def _trace(pair: Pair, table: Table) -> list[Pair]:
    history = [pair]
    for _ in range(ROUNDS):
        history.append(_step(history[-1], table))
    return history


def _wrong_evolve(pair: Pair, table: Table, mode: str) -> Pair:
    left, right = pair
    for _ in range(ROUNDS):
        if mode == "in_place":
            left = table[left][right]
            right = table[right][left]
        elif mode == "same_orientation":
            left, right = table[left][right], table[left][right]
        elif mode == "fixed_axes":
            left, right = table[left][right], table[left][right]
        else:
            raise ValueError("unknown wrong transition")
    return (left, right)


def _distance(first: Pair, second: Pair) -> int:
    return int(first[0] != second[0]) + int(first[1] != second[1])


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    for _ in range(6000):
        rows: list[list[int]] = []
        for _row in range(STATE_COUNT):
            values = list(range(STATE_COUNT))
            rng.shuffle(values)
            rows.append(values)
        table = _table(rows)
        start = (rng.randrange(STATE_COUNT), rng.randrange(STATE_COUNT))
        if start[0] == start[1]:
            continue
        history = _trace(start, table)
        target = history[-1]
        ordered_rounds = [pair for pair in history[:-1] if table[pair[0]][pair[1]] != table[pair[1]][pair[0]]]
        left_changes = sum(a[0] != b[0] for a, b in zip(history, history[1:]))
        right_changes = sum(a[1] != b[1] for a, b in zip(history, history[1:]))
        in_place = _wrong_evolve(start, table, "in_place")
        if not (
            len(set(history)) >= 4
            and len(ordered_rounds) >= 3
            and left_changes >= 3
            and right_changes >= 3
            and target != history[-2]
            and in_place != target
        ):
            continue

        preferred = [
            history[-2],
            _evolve(start, table, ROUNDS + 1),
            in_place,
            (target[1], target[0]),
            (table[start[0]][start[1]], table[start[1]][start[0]]),
            _wrong_evolve(start, table, "same_orientation"),
        ]
        distractors: list[Pair] = []
        for option in preferred:
            if option != target and option not in distractors:
                distractors.append(option)
        all_pairs = [(left, right) for left in range(STATE_COUNT) for right in range(STATE_COUNT)]
        rng.shuffle(all_pairs)
        for option in all_pairs:
            if option != target and option not in distractors:
                distractors.append(option)
        if len(distractors) < 4:
            continue
        options = distractors[:4]
        options.insert(int(seed) % 5, target)
        return {
            "table": [list(row) for row in table],
            "start_states": list(start),
            "options": [list(option) for option in options],
        }
    raise RuntimeError("could not sample a feedback-rich coupled transition table")


def analytic_gold(scene: dict) -> str:
    table, start, options = _valid_scene(scene)
    target = _evolve(start, table)
    matches = [index for index, option in enumerate(options) if option == target]
    if len(matches) != 1:
        raise ValueError("the final ordered pair must occur exactly once")
    return "ABCDE"[matches[0]]


def margin(scene: dict) -> float:
    table, start, options = _valid_scene(scene)
    target = _evolve(start, table)
    distances = [_distance(target, option) for option in options if option != target]
    return float(min(distances)) if distances else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        table, start, options = _valid_scene(scene)
        target = _evolve(start, table)
        return sum(option == target for option in options) != 1 or margin(scene) < 1.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    _valid_scene(scene)
    return []


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(FONT_PATH, size=size)


def _draw_chip(draw: ImageDraw.ImageDraw, state: int, left: int, top: int, size: int, *, outline: str | None = None) -> None:
    checked = _state(state)
    draw.rounded_rectangle(
        (left, top, left + size, top + size),
        radius=max(7, size // 6),
        fill=STATE_COLORS[checked],
        outline=outline or NAVY,
        width=2,
    )
    draw.text(
        (left + size // 2, top + size // 2),
        str(checked + 1),
        fill="white",
        font=_font(max(18, size // 2)),
        anchor="mm",
    )


def render(scene: dict) -> Image.Image:
    table, start, options = _valid_scene(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((WIDTH // 2, 22), "COUPLED TRANSITION TABLE", fill=NAVY, font=_font(24), anchor="mm")

    draw.rounded_rectangle((28, 50, 386, 420), radius=14, fill=PANEL, outline=GRID, width=2)
    draw.text((238, 57), "COLUMN = old RIGHT", fill=NAVY, font=_font(16), anchor="mm")
    draw.text((68, 107), "ROW =\nold LEFT", fill=NAVY, font=_font(14), anchor="mm", align="center")
    for index in range(STATE_COUNT):
        centre_x = TABLE_X + index * TABLE_STEP + TABLE_CELL // 2
        _draw_chip(draw, index, centre_x - HEADER_SIZE // 2, HEADER_TOP, HEADER_SIZE, outline=GOLD)
        centre_y = TABLE_Y + index * TABLE_STEP + TABLE_CELL // 2
        _draw_chip(draw, index, ROW_HEADER_LEFT, centre_y - HEADER_SIZE // 2, HEADER_SIZE, outline=GOLD)
    for row in range(STATE_COUNT):
        for column in range(STATE_COUNT):
            left = TABLE_X + column * TABLE_STEP
            top = TABLE_Y + row * TABLE_STEP
            _draw_chip(draw, table[row][column], left, top, TABLE_CELL)
    draw.text((207, 402), "CELL = next state", fill=NAVY, font=_font(15), anchor="mm")

    draw.line((418, 58, 418, 626), fill=GRID, width=2)
    draw.text((657, 70), "START", fill=GOLD, font=_font(22), anchor="mm")
    draw.text((START_LEFTS[0] + START_SIZE // 2, 98), "LEFT", fill=NAVY, font=_font(16), anchor="mm")
    draw.text((START_LEFTS[1] + START_SIZE // 2, 98), "RIGHT", fill=NAVY, font=_font(16), anchor="mm")
    _draw_chip(draw, start[0], START_LEFTS[0], START_TOP, START_SIZE, outline=GOLD)
    _draw_chip(draw, start[1], START_LEFTS[1], START_TOP, START_SIZE, outline=GOLD)

    draw.rounded_rectangle((456, 190, 861, 312), radius=12, fill=PANEL, outline=GRID, width=2)
    draw.text((475, 211), "EACH ROUND — USE THE OLD PAIR", fill=NAVY, font=_font(17), anchor="lm")
    draw.text((475, 244), "new LEFT  = table[ LEFT, RIGHT ]", fill=NAVY, font=_font(17), anchor="lm")
    draw.text((475, 274), "new RIGHT = table[ RIGHT, LEFT ]", fill=NAVY, font=_font(17), anchor="lm")
    draw.text((475, 298), "CHANGE BOTH TOGETHER  •  5 ROUNDS", fill=GOLD, font=_font(15), anchor="lm")

    draw.text((657, 338), "OPTIONS — final ordered pair", fill=NAVY, font=_font(17), anchor="mm")
    draw.text((OPTION_LEFTS[0] + OPTION_SIZE // 2, 354), "LEFT", fill=NAVY, font=_font(13), anchor="mm")
    draw.text((OPTION_LEFTS[1] + OPTION_SIZE // 2, 354), "RIGHT", fill=NAVY, font=_font(13), anchor="mm")
    for label, option, top in zip("ABCDE", options, OPTION_TOPS):
        cy = top + OPTION_SIZE // 2
        draw.ellipse((475, cy - 20, 515, cy + 20), fill=NAVY)
        draw.text((495, cy), label, fill="white", font=_font(22), anchor="mm")
        _draw_chip(draw, option[0], OPTION_LEFTS[0], top, OPTION_SIZE)
        _draw_chip(draw, option[1], OPTION_LEFTS[1], top, OPTION_SIZE)
    return image
