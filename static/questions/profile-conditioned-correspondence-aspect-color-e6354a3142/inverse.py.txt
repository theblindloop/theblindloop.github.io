"""Independent pixel-only inverse arm for the aspect-ratio correspondence world.

Receives ONLY the final rasterized PNG and the public question contract (this
file's own visual spec). It never imports the renderer, prompts, generator,
verifier, or analytic-gold code, and never reads scene data.

Public visual spec:

  * the top panel holds one neutral gray filled axis-aligned rectangle (the
    reference) whose width/height aspect ratio comes from the canonical set
    {0.5, 0.667, 1.0, 1.5, 2.0};
  * a horizontal dark divider separates it from a column of three colored
    filled rectangles, one red, one green, one blue, each with a different
    aspect ratio;
  * the task: decide which rectangle color has exactly the same width/height
    aspect ratio as the gray reference.

The oracle, from pixels alone: finds the divider, locates the gray reference
rectangle above it and the three colored rectangles below it, measures each
rectangle's width and height from its axis-aligned pixel bounding box, computes
the width/height ratio, bins it to the nearest canonical ratio, and returns the
color of the candidate rectangle whose binned ratio equals the reference's.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "correspondence_aspect_color-oracle-0.1.0"

PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (235, 40, 40),
    "green": (0, 170, 70),
    "blue": (40, 60, 235),
}
GRAY = (150, 150, 150)
DIVIDER_DARK = (70, 70, 70)
CANONICAL_RATIOS = [1.0 / 2.0, 2.0 / 3.0, 1.0, 3.0 / 2.0, 2.0]
_COLOR_DIST = 70.0
_DARK = 140.0


def _classify(arr: np.ndarray, color: tuple[int, int, int]) -> np.ndarray:
    c = np.asarray(color, dtype=np.float32)
    d = np.linalg.norm(arr - c[None, None, :], axis=-1)
    return d < _COLOR_DIST


def _bbox_rows_cols(mask: np.ndarray) -> tuple[list, list] | None:
    ys, xs = np.nonzero(mask)
    if ys.size == 0:
        return None
    return (ys, xs)


def _rect_ratio_from_mask(mask: np.ndarray) -> float | None:
    pair = _bbox_rows_cols(mask)
    if pair is None:
        return None
    ys, xs = pair
    height = int(ys.max() - ys.min() + 1)
    width = int(xs.max() - xs.min() + 1)
    if height <= 0:
        raise ValueError("degenerate rectangle height")
    return width / height


def _bin_ratio(ratio: float) -> float:
    return min(CANONICAL_RATIOS, key=lambda value: abs(value - ratio))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.float32)

    dark = _classify(arr, DIVIDER_DARK) | (arr.max(axis=2) < _DARK)
    line_counts = dark.sum(axis=1)
    grid = line_counts > 0.45 * dark.shape[1]
    rows = np.where(grid)[0]
    if rows.size == 0:
        raise ValueError("no divider line found")
    divider_mid = float(rows.mean())

    gray_mask = _classify(arr, GRAY)
    ref_mask = gray_mask.copy()
    if divider_mid < ref_mask.shape[0]:
        ref_mask[int(divider_mid):, :] = False
    ref_ratio = _rect_ratio_from_mask(ref_mask)
    if ref_ratio is None:
        raise ValueError("no gray reference rectangle above the divider")
    ref_bin = _bin_ratio(ref_ratio)

    color_masks = {
        name: _classify(arr, rgb) & _below(divider_mid, arr.shape[0])
        for name, rgb in PALETTE.items()
    }

    match_color = None
    matched = 0
    for name, mask in color_masks.items():
        if not mask.any():
            raise ValueError(f"candidate {name} rectangle is missing")
        ratio = _rect_ratio_from_mask(mask)
        bin_value = _bin_ratio(ratio)
        if abs(bin_value - ref_bin) < 1e-9:
            match_color = name
            matched += 1
    if matched != 1 or match_color is None:
        raise ValueError(f"expected exactly one candidate to match, matched={matched}")
    return match_color


def _below(divider_mid: float, height: int) -> np.ndarray:
    rows = np.arange(height, dtype=np.float32) > divider_mid + 1.0
    return rows[:, None]


if __name__ == "__main__":
    import sys

    img = Image.open(sys.argv[1]).convert("RGB")
    print(decision_from_image(img))
