"""Renderer and analytic arm for the perspective checksum sheet world.

The visible object is one filled, convex quadrilateral rather than a collection
of panels.  Its canonical texture contains a large four-quadrant glyph.  The
question explicitly asks for a perspective rectification before reading that
glyph.  ``oracle.py`` repeats the inverse from pixels with separate code and
does not import this module.
"""

from __future__ import annotations

import math
import random
from typing import Iterable

import numpy as np
from PIL import Image, ImageDraw


WIDTH = 640
HEIGHT = 640
CANONICAL = 440
MARKER_POINTS = ((30.0, 30.0), (410.0, 30.0), (410.0, 410.0), (30.0, 410.0))
BACKGROUND = (248, 247, 242)

PALETTES = (
    {
        "sheet": (34, 111, 123),
        "shade": (24, 83, 98),
        "glyph": (244, 228, 177),
        "marker": (226, 165, 54),
        "border": (18, 48, 63),
    },
    {
        "sheet": (39, 101, 133),
        "shade": (28, 72, 108),
        "glyph": (241, 224, 170),
        "marker": (231, 174, 63),
        "border": (22, 43, 70),
    },
    {
        "sheet": (44, 121, 107),
        "shade": (31, 91, 83),
        "glyph": (244, 231, 183),
        "marker": (218, 157, 48),
        "border": (22, 57, 53),
    },
)

SYMBOLS = ("lozenge", "chevron")


def _convex(corners: list[list[float]]) -> bool:
    signs = []
    for index in range(4):
        ax, ay = corners[index]
        bx, by = corners[(index + 1) % 4]
        cx, cy = corners[(index + 2) % 4]
        cross = (bx - ax) * (cy - by) - (by - ay) * (cx - bx)
        signs.append(cross)
    return all(value > 0 for value in signs) or all(value < 0 for value in signs)


def _fit_homography(source: np.ndarray, target: np.ndarray) -> np.ndarray:
    """Solve the eight inhomogeneous equations for source -> target."""

    rows: list[list[float]] = []
    values: list[float] = []
    for (x, y), (u, v) in zip(source, target, strict=True):
        rows.append([x, y, 1.0, 0.0, 0.0, 0.0, -u * x, -u * y])
        values.append(float(u))
        rows.append([0.0, 0.0, 0.0, x, y, 1.0, -v * x, -v * y])
        values.append(float(v))
    coefficients = np.linalg.solve(np.asarray(rows, dtype=float), np.asarray(values, dtype=float))
    return np.array(
        [
            [coefficients[0], coefficients[1], coefficients[2]],
            [coefficients[3], coefficients[4], coefficients[5]],
            [coefficients[6], coefficients[7], 1.0],
        ],
        dtype=float,
    )


def _sample_quad(rng: random.Random) -> list[list[float]]:
    cx = rng.uniform(308.0, 332.0)
    cy = rng.uniform(308.0, 332.0)
    half_x = rng.uniform(218.0, 232.0)
    half_y = rng.uniform(208.0, 228.0)
    for _ in range(100):
        jitter = [[rng.uniform(-42.0, 42.0), rng.uniform(-38.0, 38.0)] for _ in range(4)]
        corners = [
            [cx - half_x + jitter[0][0], cy - half_y + jitter[0][1]],
            [cx + half_x + jitter[1][0], cy - half_y + jitter[1][1]],
            [cx + half_x + jitter[2][0], cy + half_y + jitter[2][1]],
            [cx - half_x + jitter[3][0], cy + half_y + jitter[3][1]],
        ]
        if _convex(corners) and all(28.0 < x < WIDTH - 28.0 and 28.0 < y < HEIGHT - 28.0 for x, y in corners):
            return corners
    return [
        [84.0, 88.0],
        [556.0, 79.0],
        [568.0, 560.0],
        [72.0, 552.0],
    ]


def sample_scene(seed: int) -> dict:
    """Return a deterministic JSON-friendly scene for one seed."""

    rng = random.Random(int(seed) * 1000003 + 9176)
    # Alternating seed parity gives deterministic class balance while every
    # visual nuisance comes from an independent stream of random draws.
    symbol = SYMBOLS[int(seed) & 1]
    scene = {
        "width": WIDTH,
        "height": HEIGHT,
        "corners": _sample_quad(rng),
        "symbol": symbol,
        "palette": rng.randrange(len(PALETTES)),
        "border_width": rng.uniform(9.0, 14.0),
        "texture_phase": rng.randrange(4),
        "alias_cycle": rng.randrange(4),
    }
    return scene


def _glyph_points(symbol: str) -> list[tuple[float, float]]:
    if symbol == "lozenge":
        return [(40.0, 220.0), (220.0, 40.0), (400.0, 220.0), (220.0, 400.0), (40.0, 220.0)]
    if symbol == "chevron":
        return [(38.0, 72.0), (128.0, 368.0), (220.0, 72.0), (312.0, 368.0), (402.0, 72.0)]
    raise ValueError(f"unknown symbol {symbol!r}")


def _canonical_sheet(scene: dict) -> Image.Image:
    palette = PALETTES[int(scene["palette"]) % len(PALETTES)]
    image = Image.new("RGB", (CANONICAL, CANONICAL), palette["sheet"])
    draw = ImageDraw.Draw(image)
    phase = (int(scene["alias_cycle"]) + int(scene["texture_phase"])) % 4

    # A full-field woven texture keeps the decisive support area-like instead
    # of making the answer a sparse collection of marks.
    stripe = 36
    for row in range(-1, CANONICAL // stripe + 2):
        y0 = row * stripe
        if (row + phase) % 2 == 0:
            draw.rectangle((0, y0, CANONICAL, y0 + 15), fill=palette["shade"])
    for column in range(-1, CANONICAL // stripe + 2):
        x0 = column * stripe
        if (column + phase + 1) % 3 == 0:
            draw.rectangle((x0, 0, x0 + 11, CANONICAL), fill=palette["shade"])

    # The glyph crosses all four quadrants.  Its raw perspective image is not
    # a canonical symbol until the declared transform is carried out.
    line_width = 23
    draw.line(_glyph_points(scene["symbol"]), fill=palette["glyph"], width=line_width, joint="curve")

    # Four equal corner fiducials expose the projective frame.  They sit well
    # inside the border so their centroid is a stable proxy for the true
    # corresponding canonical point.
    marker_radius = 13
    for x, y in MARKER_POINTS:
        draw.ellipse((x - marker_radius, y - marker_radius, x + marker_radius, y + marker_radius), fill=palette["marker"])
        draw.ellipse((x - 4, y - 4, x + 4, y + 4), fill=palette["border"])
    border = max(8, int(round(float(scene["border_width"]))))
    draw.rectangle((8, 8, CANONICAL - 8, CANONICAL - 8), outline=palette["border"], width=border)
    return image


def render(scene: dict) -> Image.Image:
    """Render the continuous sheet through a projective warp."""

    width = int(scene["width"])
    height = int(scene["height"])
    if (width, height) != (WIDTH, HEIGHT):
        raise ValueError("unsupported canvas size")
    source = np.asarray(
        [[0.0, 0.0], [CANONICAL, 0.0], [CANONICAL, CANONICAL], [0.0, CANONICAL]], dtype=float
    )
    target = np.asarray(scene["corners"], dtype=float)
    if target.shape != (4, 2) or not _convex(target.tolist()):
        raise ValueError("scene quadrilateral is not convex")
    homography = _fit_homography(source, target)
    inverse = np.linalg.inv(homography)
    inverse = inverse / inverse[2, 2]
    coefficients = (
        float(inverse[0, 0]),
        float(inverse[0, 1]),
        float(inverse[0, 2]),
        float(inverse[1, 0]),
        float(inverse[1, 1]),
        float(inverse[1, 2]),
        float(inverse[2, 0]),
        float(inverse[2, 1]),
    )
    transformed = _canonical_sheet(scene).transform(
        (width, height), Image.Transform.PERSPECTIVE, coefficients, resample=Image.Resampling.BICUBIC, fillcolor=BACKGROUND
    )
    mask = Image.new("L", (width, height), 0)
    mask_draw = ImageDraw.Draw(mask)
    mask_draw.polygon([tuple(point) for point in target.tolist()], fill=255)
    canvas = Image.new("RGB", (width, height), BACKGROUND)
    canvas.paste(transformed, (0, 0), mask)
    draw = ImageDraw.Draw(canvas)
    palette = PALETTES[int(scene["palette"]) % len(PALETTES)]
    border = max(5, int(round(float(scene["border_width"]))))
    outline = [tuple(point) for point in target.tolist()]
    draw.line(outline + [outline[0]], fill=palette["border"], width=border, joint="curve")
    return canvas


def analytic_gold(scene: dict) -> str:
    """Return only the declared symbol label; the cycle is a render alias."""

    symbol = scene["symbol"]
    normalized_cycle = int(scene["alias_cycle"]) % 4
    if normalized_cycle not in range(4):  # keeps the label input explicit to the audit
        raise ValueError("invalid cycle")
    if symbol not in SYMBOLS:
        raise ValueError(f"unknown symbol {symbol!r}")
    return symbol


def margin(scene: dict) -> float:
    """One-sided render-verifiability margin for the sheet frame."""

    corners = np.asarray(scene["corners"], dtype=float)
    edges = np.linalg.norm(np.roll(corners, -1, axis=0) - corners, axis=1)
    signed_cross = []
    for index in range(4):
        a = corners[index]
        b = corners[(index + 1) % 4]
        c = corners[(index + 2) % 4]
        first = b - a
        second = c - b
        signed_cross.append(float(first[0] * second[1] - first[1] * second[0]))
    area_scale = abs(sum(signed_cross)) / (2.0 * WIDTH * HEIGHT)
    edge_scale = float(edges.min() / WIDTH)
    border_clearance = float(
        min(corners[:, 0].min(), corners[:, 1].min(), WIDTH - corners[:, 0].max(), HEIGHT - corners[:, 1].max()) / WIDTH
    )
    return float(min(area_scale, edge_scale, border_clearance))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.08


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["corners"] = [list(point) for point in scene["corners"]]
    twin["alias_cycle"] = int(scene["alias_cycle"]) + 4
    return [("one_full_texture_cycle", twin)]
