"""State-visible swaps: track a ringed token's slot across ordered swaps.

The image is a vertical timeline of PANELS=4 panels. Each panel shows a row of
4 slots (boxes); every slot holds one token disk. Exactly one token is uniquely
colored across the whole image (colors are a permutation of the 4-color
palette), so identity is carried by color. A magenta ring marks the tracked
token in the top (first) panel.

Between each pair of consecutive panels sits a thin "swap strip" showing one
cyan exchange mark at a boundary between two adjacent slots; that mark means the
two tokens at those slots exchange positions as you move to the next panel.

The decision is the slot (1-4) where the ringed token lives in the FINAL (bottom)
panel. It is obtained by replaying the visible swap boundaries on the marked
start slot; the scene below is guaranteed to contain exactly one consistent
replay because each swap is an unambiguous adjacent transposition.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

N = 4
PANELS = 4

LEFT_M = 60
TOP_M = 70
SLOT_W = 110
SLOT_H = 90
GAP = 30
STRIP_H = 40
TOKEN_R = 20
RING_R = TOKEN_R + 8

WIDTH = LEFT_M + N * SLOT_W + (N - 1) * GAP + LEFT_M
HEIGHT = TOP_M + PANELS * SLOT_H + (PANELS - 1) * STRIP_H + 50

BG = (255, 255, 255)
BORDER = (95, 95, 95)
TOKEN_COLORS = [
    (210, 40, 40),
    (40, 170, 40),
    (50, 80, 220),
    (240, 160, 30),
]
RING = (210, 0, 210)
SWAP = (0, 160, 190)

SLOT_LABELS = {0: "slot 1", 1: "slot 2", 2: "slot 3", 3: "slot 4"}


def _col_center(i: int) -> int:
    return LEFT_M + SLOT_W // 2 + i * (SLOT_W + GAP)


def _col_x0(i: int) -> int:
    return _col_center(i) - SLOT_W // 2


def _boundary_x(b: int) -> int:
    return (_col_center(b) + _col_center(b + 1)) // 2


def _panel_top(k: int) -> int:
    return TOP_M + k * (SLOT_H + STRIP_H)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    colors = [0, 1, 2, 3]
    rng.shuffle(colors)
    arrangement = list(colors)
    start = rng.randint(0, N - 1)
    swaps = [rng.randint(0, N - 2) for _ in range(PANELS - 1)]
    arrangements = [list(arrangement)]
    for boundary in swaps:
        arrangement = list(arrangement)
        arrangement[boundary], arrangement[boundary + 1] = (
            arrangement[boundary + 1],
            arrangement[boundary],
        )
        arrangements.append(arrangement)
    return {"tokens": arrangements, "swaps": swaps, "start": start}


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)
    arrangements = scene["tokens"]
    swaps = scene["swaps"]
    start = scene["start"]

    for k in range(PANELS):
        top = _panel_top(k)
        for i in range(N):
            x0 = _col_x0(i)
            draw.rectangle(
                [x0, top, x0 + SLOT_W, top + SLOT_H], outline=BORDER, width=3
            )
            cx = _col_center(i)
            cy = top + SLOT_H // 2
            color = TOKEN_COLORS[arrangements[k][i]]
            draw.ellipse(
                [cx - TOKEN_R, cy - TOKEN_R, cx + TOKEN_R, cy + TOKEN_R], fill=color
            )
        if k == 0:
            cx = _col_center(start)
            cy = top + SLOT_H // 2
            draw.ellipse(
                [cx - RING_R, cy - RING_R, cx + RING_R, cy + RING_R],
                outline=RING,
                width=5,
            )

    for k, boundary in enumerate(swaps):
        strip_center_y = _panel_top(k) + SLOT_H + STRIP_H // 2
        bx = _boundary_x(boundary)
        r = 13
        draw.line([bx - r, strip_center_y - r, bx + r, strip_center_y + r], fill=SWAP, width=5)
        draw.line([bx - r, strip_center_y + r, bx + r, strip_center_y - r], fill=SWAP, width=5)
    return img


def analytic_gold(scene: dict) -> str:
    position = scene["start"]
    for boundary in scene["swaps"]:
        if position == boundary:
            position = boundary + 1
        elif position == boundary + 1:
            position = boundary
    return SLOT_LABELS[position]


def margin(scene: dict) -> float:
    return 10.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    """No nontrivial render-invariant latent symmetry: every latent field
    (``tokens``, ``swaps``, ``start``) directly changes the pixels -- ``tokens``
    fix the drawn disk colors in each slot, ``swaps`` fix the cyan exchange
    marks, and ``start`` fixes which panel-1 token is ringed. Any permutation of
    slots or colors changes the raster, so no alias exists."""
    return []
