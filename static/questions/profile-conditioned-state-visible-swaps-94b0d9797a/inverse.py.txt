"""Independent pixel-only inverse arm for the state-visible-swaps world.

Receives only the final PNG. From the raster alone it:
  1. labels pixels against the fixed palette and recovers the 4 slot columns
     from the panel-box border edges;
  2. finds the magenta ring in the first panel to get the tracked token's start
     slot;
  3. reads each cyan exchange mark's boundary to recover the per-step swaps in
     panel order;
  4. replays the swaps from the marked start slot and returns the final slot.

No latent scene, renderer, prompt, or gold code is accessed.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

BG = (255, 255, 255)
BORDER = (95, 95, 95)
TOKEN_COLORS = [
    (210, 40, 40),
    (40, 170, 40),
    (50, 80, 220),
    (240, 160, 30),
]
RING = (210, 0, 210)
SWAP = (0, 160, 190)

PALETTE = [BG, BORDER, *TOKEN_COLORS, RING, SWAP]
_BG, _BRD, _TC0, _TC1, _TC2, _TC3, _RING, _SWAP = range(8)

EXPECTED_COLUMNS = 4
SLOT_W = 110
SLOT_H = 90
STRIP_H = 40
MARK_HALF = 13
SLOT_LABELS = {0: "slot 1", 1: "slot 2", 2: "slot 3", 3: "slot 4"}


class UnreadableImageError(ValueError):
    """The raster does not expose the expected panel structure."""


def _label(arr: np.ndarray) -> np.ndarray:
    dists = np.stack(
        [np.linalg.norm(arr - np.array(c, dtype=np.float32), axis=-1) for c in PALETTE],
        axis=-1,
    )
    return dists.argmin(axis=-1)


def _cluster(sorted_vals: list[int], tol: int) -> list[int]:
    if not sorted_vals:
        return []
    out = [[sorted_vals[0]]]
    for v in sorted_vals[1:]:
        if v - out[-1][-1] > tol:
            out.append([v])
        else:
            out[-1].append(v)
    return [r[len(r) // 2] for r in out]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label(arr)
    h, w = labels.shape

    border_ys, border_xs = np.nonzero(labels == _BRD)
    if len(border_xs) == 0:
        raise UnreadableImageError("no panel borders found")

    counts = np.bincount(border_xs, minlength=w)
    edge_cols = [int(x) for x in range(w) if counts[x] > 40]
    edges = _cluster(sorted(set(edge_cols)), 6)
    if len(edges) < 2 * EXPECTED_COLUMNS:
        raise UnreadableImageError(
            f"expected {2 * EXPECTED_COLUMNS} box side edges, found {len(edges)}"
        )
    edge = edges[: 2 * EXPECTED_COLUMNS]
    col_centers = [(edge[2 * i] + edge[2 * i + 1]) // 2 for i in range(EXPECTED_COLUMNS)]

    ring_pts = list(zip(*np.nonzero(labels == _RING)))
    if not ring_pts:
        raise UnreadableImageError("no magenta ring found")
    ring_x = sum(p[1] for p in ring_pts) // len(ring_pts)
    start = min(range(EXPECTED_COLUMNS), key=lambda i: abs(col_centers[i] - ring_x))

    top_edge = int(border_ys.min())

    swap_ys, swap_xs = np.nonzero(labels == _SWAP)
    if len(swap_xs) == 0:
        raise UnreadableImageError("no swap marks found")

    strip_ops = []
    for b in range(EXPECTED_COLUMNS - 1):
        strip_ops.append((edge[2 * b + 1] + edge[2 * b + 2]) // 2)

    strips = []
    prev_top = top_edge
    for _step in range(3):
        lo = prev_top + SLOT_H - MARK_HALF
        hi = prev_top + SLOT_H + STRIP_H + MARK_HALF
        xs_here = [int(x) for (y, x) in zip(swap_ys.tolist(), swap_xs.tolist()) if lo <= y <= hi]
        if not xs_here:
            raise UnreadableImageError(f"swap strip {_step} has no mark")
        mean_bx = sum(xs_here) // len(xs_here)
        distances = [abs(strip_ops[b] - mean_bx) for b in range(EXPECTED_COLUMNS - 1)]
        strips.append(min(range(EXPECTED_COLUMNS - 1), key=lambda b: distances[b]))
        prev_top = prev_top + SLOT_H + STRIP_H

    position = start
    for boundary in strips:
        if position == boundary:
            position = boundary + 1
        elif position == boundary + 1:
            position = boundary
    return SLOT_LABELS[position]
