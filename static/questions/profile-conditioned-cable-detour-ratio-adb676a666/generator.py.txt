"""Deterministic cable-detour visual world and analytic arm."""

from __future__ import annotations

import copy
import math
import random
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 720, 480
BG = "#f3f5f4"
INK = "#30343a"
CYAN = "#00a0b4"
GREEN = "#20a464"
COLORS = {"orange": "#ee8b2c", "purple": "#9467d3", "blue": "#3979cf"}

# Each set deliberately breaks rankings by raw length and endpoint chord while
# preserving a large ratio gap.  All routes have exactly six 90-degree bends.
_SETS = (
    ((520, "low"), (600, "mid"), (360, "high")),
    ((580, "low"), (560, "mid"), (300, "high")),
    ((450, "low"), (490, "mid"), (500, "high")),
)
_Y_LEVELS = {
    "low": (65, 45, 75, 65),       # total vertical travel 60
    "mid": (100, 25, 100, 70),    # total vertical travel 180
    "high": (115, 15, 115, 15),   # total vertical travel 300
}


def _route_points(dx: int, tier: str, row: int, mirror_x: bool, mirror_y: bool) -> list[list[int]]:
    x0 = (WIDTH - dx) // 2
    shares = (0.18, 0.26, 0.22)
    x1 = x0 + round(dx * shares[0])
    x2 = x1 + round(dx * shares[1])
    x3 = x2 + round(dx * shares[2])
    xs = (x0, x1, x2, x3, x0 + dx)
    ys = list(_Y_LEVELS[tier])
    if mirror_x:
        xs = tuple(WIDTH - x for x in xs)
    if mirror_y:
        ys = [140 - y for y in ys]
    yoff = 10 + 155 * row
    ys = [yoff + y for y in ys]
    return [
        [xs[0], ys[0]], [xs[1], ys[0]], [xs[1], ys[1]], [xs[2], ys[1]],
        [xs[2], ys[2]], [xs[3], ys[2]], [xs[3], ys[3]], [xs[4], ys[3]],
    ]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    answer_color = ("orange", "purple", "blue")[int(seed) % 3]
    other_colors = [c for c in ("orange", "purple", "blue") if c != answer_color]
    rng.shuffle(other_colors)
    tier_colors = {"high": answer_color, "mid": other_colors[0], "low": other_colors[1]}
    specs = list(_SETS[(int(seed) // 3) % len(_SETS)])
    rng.shuffle(specs)
    routes = []
    for row, (dx, tier) in enumerate(specs):
        routes.append({
            "tier": tier,
            "terminal": tier_colors[tier],
            "points": _route_points(dx, tier, row, bool(rng.randrange(2)), bool(rng.randrange(2))),
        })
    return {"routes": routes, "width": WIDTH, "height": HEIGHT, "background": BG, "stroke": 11}


def _route_ratio(points: list[list[int]]) -> float:
    length = sum(math.dist(a, b) for a, b in zip(points, points[1:]))
    chord = math.dist(points[0], points[-1])
    if chord <= 0:
        raise ValueError("coincident endpoint centers")
    return length / chord


def analytic_gold(scene: dict) -> str:
    scored = [(_route_ratio(route["points"]), route["terminal"]) for route in scene["routes"]]
    best = max(score for score, _ in scored)
    winners = [terminal for score, terminal in scored if abs(score - best) < 1e-12]
    if len(winners) != 1:
        raise ValueError("scene has no unique maximum detour ratio")
    return winners[0]


def margin(scene: dict) -> float:
    scores = sorted((_route_ratio(route["points"]) for route in scene["routes"]), reverse=True)
    return float(scores[0] - scores[1])


def _valid_polyline(points: list[list[int]]) -> bool:
    if len(points) != 8:
        return False
    directions = []
    for a, b in zip(points, points[1:]):
        dx, dy = b[0] - a[0], b[1] - a[1]
        if (dx == 0) == (dy == 0) or (dx == 0 and dy == 0):
            return False
        directions.append(dx == 0)
    return all(a != b for a, b in zip(directions, directions[1:]))


def is_quarantined(scene: dict) -> bool:
    try:
        routes = scene["routes"]
        if len(routes) != 3 or {r["terminal"] for r in routes} != set(COLORS):
            return True
        if any(not _valid_polyline(r["points"]) for r in routes):
            return True
        scores = sorted((_route_ratio(r["points"]) for r in routes), reverse=True)
        return scores[0] - scores[1] < 0.18
    except (KeyError, TypeError, ValueError, ZeroDivisionError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    transformed = copy.deepcopy(scene)
    transformed["routes"] = list(reversed(transformed["routes"]))
    return [("reverse_route_storage_order", transformed)]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), scene["background"])
    draw = ImageDraw.Draw(image)
    for row in range(3):
        top = 12 + 155 * row
        draw.rounded_rectangle((16, top, WIDTH - 16, top + 140), radius=16, fill="#ffffff", outline="#d6d9dc", width=2)
    for route in scene["routes"]:
        points = [tuple(p) for p in route["points"]]
        draw.line(points, fill=INK, width=int(scene["stroke"]), joint="curve")
        sx, sy = points[0]
        ex, ey = points[-1]
        draw.ellipse((sx - 13, sy - 13, sx + 13, sy + 13), fill="#ffffff", outline=GREEN, width=6)
        draw.rounded_rectangle((ex - 13, ey - 13, ex + 13, ey + 13), radius=4, fill=COLORS[route["terminal"]], outline="#ffffff", width=3)
        # Exact public centerline supports a renderer-independent pixel trace.
        draw.line(points, fill=CYAN, width=1)
    return image
