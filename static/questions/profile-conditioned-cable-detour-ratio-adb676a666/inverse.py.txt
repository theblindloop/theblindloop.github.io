"""Independent pixel-only inverse arm for cable detour ratio."""

from __future__ import annotations

from collections import deque
import math
import numpy as np
from PIL import Image

_CYAN = np.array([0, 160, 180], dtype=np.uint8)
_GREEN = np.array([32, 164, 100], dtype=np.uint8)
_TERMINALS = {
    "orange": np.array([238, 139, 44], dtype=np.uint8),
    "purple": np.array([148, 103, 211], dtype=np.uint8),
    "blue": np.array([57, 121, 207], dtype=np.uint8),
}


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        group = []
        while queue:
            cy, cx = queue.popleft()
            group.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(group) > 250:
            groups.append(group)
    return groups


def _ordered_path(group: list[tuple[int, int]], green_points: np.ndarray) -> list[tuple[int, int]]:
    pixels = set(group)
    endpoints = []
    for point in group:
        y, x = point
        degree = sum((y + dy, x + dx) in pixels for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)))
        if degree == 1:
            endpoints.append(point)
        elif degree != 2:
            raise ValueError("centerline is branched, thickened, or broken")
    if len(endpoints) != 2:
        raise ValueError("centerline does not have exactly two endpoints")

    def green_distance(point: tuple[int, int]) -> int:
        delta = green_points - np.asarray(point)
        return int(np.min((delta * delta).sum(axis=1)))

    start = min(endpoints, key=green_distance)
    if green_distance(start) > 20 * 20:
        raise ValueError("route lacks a green start marker")
    ordered, previous, current = [start], None, start
    while len(ordered) < len(group):
        y, x = current
        choices = [(y + dy, x + dx) for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1))
                   if (y + dy, x + dx) in pixels and (y + dy, x + dx) != previous]
        if not choices:
            break
        if len(choices) != 1:
            raise ValueError("ambiguous centerline ordering")
        previous, current = current, choices[0]
        ordered.append(current)
    if len(ordered) != len(group):
        raise ValueError("failed to trace the complete centerline")
    return ordered


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    groups = _components(np.all(array == _CYAN, axis=2))
    if len(groups) != 3:
        raise ValueError(f"expected three cable centerlines, found {len(groups)}")
    green_points = np.argwhere(np.all(array == _GREEN, axis=2))
    if len(green_points) < 150:
        raise ValueError("green endpoint evidence is missing")
    terminal_points = {label: np.argwhere(np.all(array == color, axis=2)) for label, color in _TERMINALS.items()}
    if any(len(points) < 100 for points in terminal_points.values()):
        raise ValueError("colored endpoint evidence is missing")

    scored = []
    for group in groups:
        path = _ordered_path(group, green_points)
        endpoint = np.asarray(path[-1])
        distances = {label: int(np.min(((points - endpoint) ** 2).sum(axis=1))) for label, points in terminal_points.items()}
        label = min(distances, key=distances.get)
        if distances[label] > 20 * 20:
            raise ValueError("centerline lacks a colored terminal")
        # Four-connected exact centerline: each graph step is one pixel.
        length = float(len(path) - 1)
        chord = math.dist(path[0], path[-1])
        if chord <= 0:
            raise ValueError("coincident endpoint centers")
        scored.append((length / chord, label))
    scored.sort(reverse=True)
    if scored[0][0] - scored[1][0] < 0.18:
        raise ValueError("pixel ratio margin is ambiguous")
    return scored[0][1]
