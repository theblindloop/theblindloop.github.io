"""Deterministic renderer for three-way occlusion-cycle visual search."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

COLORS = {
    "red": (203, 54, 68),
    "blue": (35, 108, 181),
    "green": (35, 143, 92),
}
INK = (31, 38, 48)
FRAME = (151, 159, 169)
PANEL_BG = (249, 250, 252)
LABELS = ("A", "B", "C", "D")
QUESTION = (
    "Which labeled panel shows RED over BLUE, BLUE over GREEN, and GREEN over RED? "
    "At a crossing, the uninterrupted strip is over the interrupted strip. A-D run left to right."
)

# Local endpoints form three well-separated pairwise intersections: the red-blue,
# red-green, and blue-green crossings sit at the corners of a visible triangle.
BASE_LINES = {
    "red": ((28, 58), (192, 58)),
    "blue": ((68, 24), (68, 166)),
    "green": ((28, 148), (192, 20)),
}
INTERSECTIONS = {
    frozenset(("red", "blue")): (68, 58),
    frozenset(("red", "green")): (143, 58),
    frozenset(("blue", "green")): (68, 117),
}
TARGET_TOP = {
    frozenset(("red", "blue")): "red",
    frozenset(("blue", "green")): "blue",
    frozenset(("green", "red")): "green",
}
REVERSE_TOP = {
    pair: next(color for color in pair if color != top)
    for pair, top in TARGET_TOP.items()
}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    # Source panel 0 is the requested cycle; all three distractors are its exact
    # reversal. Every panel therefore has identical crossing count and has every
    # color over once and under once.
    panel_cycles = ["target", "reverse", "reverse", "reverse"]
    panel_order = list(range(4))
    rng.shuffle(panel_order)
    layouts = [
        {"dx": rng.randint(-3, 3), "dy": rng.randint(-3, 3)}
        for _ in range(4)
    ]
    return {
        "panel_cycles": panel_cycles,
        "panel_order": panel_order,
        "layouts": layouts,
        "style_variant": rng.randrange(2),
    }


def _shift_line(line: tuple[tuple[int, int], tuple[int, int]], dx: int, dy: int):
    return tuple((x + dx, y + dy) for x, y in line)


def _local_segment(
    line: tuple[tuple[int, int], tuple[int, int]],
    point: tuple[int, int],
    half_length: float = 17.0,
):
    (x0, y0), (x1, y1) = line
    length = ((x1 - x0) ** 2 + (y1 - y0) ** 2) ** 0.5
    ux, uy = (x1 - x0) / length, (y1 - y0) / length
    px, py = point
    return (
        (round(px - ux * half_length), round(py - uy * half_length)),
        (round(px + ux * half_length), round(py + uy * half_length)),
    )


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    box: tuple[int, int, int, int],
    cycle: str,
    layout: dict,
) -> None:
    x0, y0, x1, y1 = box
    draw.rounded_rectangle(box, radius=9, fill=PANEL_BG, outline=FRAME, width=2)
    dx, dy = int(layout["dx"]), int(layout["dy"])
    lines = {
        color: tuple((x0 + x, y0 + y) for x, y in _shift_line(line, dx, dy))
        for color, line in BASE_LINES.items()
    }
    for color in ("red", "blue", "green"):
        draw.line(lines[color], fill=COLORS[color], width=11)

    top_map = TARGET_TOP if cycle == "target" else REVERSE_TOP
    for pair, local_point in INTERSECTIONS.items():
        px, py = x0 + local_point[0] + dx, y0 + local_point[1] + dy
        top = top_map[pair]
        # A white underpass gap makes the lower strip visibly interrupted. The
        # top strip is then redrawn through the center with its full width.
        draw.ellipse((px - 11, py - 11, px + 11, py + 11), fill=PANEL_BG)
        draw.line(_local_segment(lines[top], (px, py)), fill=COLORS[top], width=11)


def render(scene: dict) -> Image.Image:
    _ = scene["style_variant"]  # deliberate serialized render-invariant alias
    image = Image.new("RGB", (1000, 430), (255, 255, 255))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    draw.text((25, 22), QUESTION, fill=INK, font=font)
    boxes = (
        (20, 106, 240, 286),
        (267, 106, 487, 286),
        (514, 106, 734, 286),
        (761, 106, 981, 286),
    )
    for position, box in enumerate(boxes):
        source = scene["panel_order"][position]
        _draw_panel(
            draw,
            box,
            scene["panel_cycles"][source],
            scene["layouts"][source],
        )
        draw.text(((box[0] + box[2]) // 2 - 3, 310), LABELS[position], fill=INK, font=font)
    return image


def analytic_gold(scene: dict) -> str:
    matches = [
        position
        for position, source in enumerate(scene["panel_order"])
        if scene["panel_cycles"][source] == "target"
    ]
    if len(matches) != 1:
        raise ValueError("scene does not contain exactly one target cycle")
    return LABELS[matches[0]]


def margin(scene: dict) -> float:
    matches = [
        3.0 if scene["panel_cycles"][source] == "target" else 0.0
        for source in scene["panel_order"]
    ]
    ordered = sorted(matches, reverse=True)
    return ordered[0] - ordered[1] if len(ordered) == 4 else 0.0


def is_quarantined(scene: dict) -> bool:
    valid = {"target", "reverse"}
    cycles = scene.get("panel_cycles", [])
    return (
        len(cycles) != 4
        or any(cycle not in valid for cycle in cycles)
        or sum(cycle == "target" for cycle in cycles) != 1
        or margin(scene) < 1.0
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["style_variant"] = 1 - int(twin["style_variant"])
    return [("serialization_style_alias", twin)]
