"""Independent pixel-only inverse arm for the occlusion-cycle panels."""

from __future__ import annotations

import numpy as np
from PIL import Image

PALETTE = {
    "red": (203, 54, 68),
    "blue": (35, 108, 181),
    "green": (35, 143, 92),
}
LABELS = ("A", "B", "C", "D")
PAIRS = (("red", "blue"), ("blue", "green"), ("green", "red"))
WANTED = {("red", "blue"): "red", ("blue", "green"): "blue", ("green", "red"): "green"}


def _masks(image: Image.Image) -> dict[str, np.ndarray]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    masks = {}
    for name, rgb in PALETTE.items():
        delta = arr - np.asarray(rgb, dtype=np.int32)
        mask = np.sum(delta * delta, axis=2, dtype=np.int32) <= 34 * 34
        if int(mask.sum()) < 500:
            raise ValueError(f"missing {name} strip evidence")
        masks[name] = mask
    return masks


def _runs(values: np.ndarray) -> list[tuple[int, int]]:
    ordered = np.flatnonzero(values)
    if len(ordered) == 0:
        return []
    cuts = np.flatnonzero(np.diff(ordered) > 1) + 1
    groups = np.split(ordered, cuts)
    return [(int(group[0]), int(group[-1])) for group in groups]


def _panel_regions(masks: dict[str, np.ndarray]) -> list[tuple[int, int]]:
    occupied = np.any(np.stack(list(masks.values()), axis=0), axis=0)
    runs = _runs(occupied.any(axis=0))
    # Within-panel interruption gaps never empty an entire x column; only the
    # broad white gutters split these four projection runs.
    if len(runs) != 4 or any(hi - lo < 140 for lo, hi in runs):
        raise ValueError("expected four separated colored panels")
    return runs


def _fit_line(mask: np.ndarray, xlo: int, xhi: int) -> tuple[np.ndarray, np.ndarray]:
    ys, xs = np.nonzero(mask[:, xlo : xhi + 1])
    xs = xs.astype(np.float64) + xlo
    ys = ys.astype(np.float64)
    if len(xs) < 120:
        raise ValueError("strip is too incomplete to fit")
    points = np.column_stack((xs, ys))
    center = points.mean(axis=0)
    centered = points - center
    _, singular, vh = np.linalg.svd(centered, full_matrices=False)
    if singular[0] < 80 or singular[0] / max(singular[1], 1e-6) < 4.0:
        raise ValueError("colored component is not an elongated strip")
    direction = vh[0]
    return center, direction


def _intersection(
    first: tuple[np.ndarray, np.ndarray], second: tuple[np.ndarray, np.ndarray]
) -> np.ndarray:
    p, u = first
    q, v = second
    matrix = np.column_stack((u, -v))
    if abs(float(np.linalg.det(matrix))) < 0.18:
        raise ValueError("strip pair is nearly parallel")
    t = np.linalg.solve(matrix, q - p)[0]
    return p + t * u


def _nearest_distance(mask: np.ndarray, point: np.ndarray, radius: int = 18) -> float:
    x, y = float(point[0]), float(point[1])
    x0, x1 = max(0, int(x) - radius), min(mask.shape[1], int(x) + radius + 1)
    y0, y1 = max(0, int(y) - radius), min(mask.shape[0], int(y) + radius + 1)
    ys, xs = np.nonzero(mask[y0:y1, x0:x1])
    if len(xs) == 0:
        return float("inf")
    xs = xs.astype(np.float64) + x0
    ys = ys.astype(np.float64) + y0
    return float(np.sqrt(np.min((xs - x) ** 2 + (ys - y) ** 2)))


def _panel_score(masks: dict[str, np.ndarray], region: tuple[int, int]) -> int:
    xlo, xhi = region
    lines = {name: _fit_line(mask, xlo, xhi) for name, mask in masks.items()}
    score = 0
    for first, second in PAIRS:
        point = _intersection(lines[first], lines[second])
        if not (xlo + 10 <= point[0] <= xhi - 10 and 10 <= point[1] <= masks[first].shape[0] - 10):
            raise ValueError("fitted crossing lies outside its panel")
        d_first = _nearest_distance(masks[first], point)
        d_second = _nearest_distance(masks[second], point)
        if not np.isfinite(d_first + d_second) or abs(d_first - d_second) < 4.0:
            raise ValueError("over-under relation is ambiguous")
        top = first if d_first < d_second else second
        if top == WANTED[(first, second)]:
            score += 1
    return score


def decision_from_image(image: Image.Image) -> str:
    masks = _masks(image)
    regions = _panel_regions(masks)
    scores = [_panel_score(masks, region) for region in regions]
    order = sorted(range(4), key=lambda index: scores[index], reverse=True)
    if scores[order[0]] != 3 or scores[order[0]] - scores[order[1]] < 1:
        raise ValueError(f"pixel evidence has no unique requested cycle: {scores}")
    return LABELS[order[0]]
