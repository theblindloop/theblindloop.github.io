"""Deterministic raster world for first identity-specific state recurrence."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1280, 720
BACKGROUND = (245, 247, 251)
PAPER = (255, 255, 255)
CARD = (250, 251, 253)
ARROW_BG = (232, 235, 241)
INK = (28, 37, 51)
MUTED = (94, 107, 128)
BORDER = (207, 214, 225)
LABELS = tuple("ABCD")
DIRECTION_NAMES = ("NORTH", "EAST", "SOUTH", "WEST")

PALETTES = (
    ((220, 52, 77), (39, 112, 220), (27, 157, 107), (232, 132, 31)),
    ((140, 70, 211), (13, 154, 173), (220, 61, 145), (205, 154, 25)),
    ((205, 65, 54), (52, 133, 197), (78, 158, 57), (221, 116, 54)),
)

_PIP_OFFSETS = (
    ((0, 0),),
    ((-7, 0), (7, 0)),
    ((0, -7), (-7, 6), (7, 6)),
    ((-7, -7), (7, -7), (-7, 7), (7, 7)),
)
_DIRECTION_VECTORS = ((0, -1), (1, 0), (0, 1), (-1, 0))


def _font(size: int, bold: bool = True) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _first_return(states: list[list[int]], identity: int) -> int | None:
    initial = int(states[0][identity])
    left = False
    for panel in range(1, len(states)):
        direction = int(states[panel][identity])
        if direction != initial:
            left = True
        elif left:
            return panel
    return None


def _winner(states: list[list[int]]) -> int:
    returns = [(step, identity) for identity in range(4) if (step := _first_return(states, identity)) is not None]
    if not returns:
        raise ValueError("timeline has no return event")
    earliest = min(step for step, _ in returns)
    winners = [identity for step, identity in returns if step == earliest]
    if len(winners) != 1:
        raise ValueError("timeline lacks a unique earliest return")
    return winners[0]


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced five-panel recurrence timeline."""
    seed = int(seed)
    rng = random.Random(seed ^ 0x51A7E)
    target = seed % 4
    return_panel = 2 + ((seed // 4) % 3)

    initials = [rng.randrange(4) for _ in range(4)]
    states = [[initials[identity] for identity in range(4)] for _ in range(5)]

    def other(initial: int) -> int:
        return (initial + 1 + rng.randrange(3)) % 4

    # Winner leaves immediately, avoids its home direction, then uniquely returns.
    for panel in range(1, return_panel):
        states[panel][target] = other(initials[target])
    states[return_panel][target] = initials[target]
    for panel in range(return_panel + 1, 5):
        states[panel][target] = rng.randrange(4)

    distractors = [identity for identity in range(4) if identity != target]
    rng.shuffle(distractors)
    persistent, later_return, delayed_departure = distractors

    # Persistent is a hard negative: it equals panel 1 throughout but never leaves.
    for panel in range(5):
        states[panel][persistent] = initials[persistent]

    # This token leaves immediately and may return, but strictly after the winner.
    for panel in range(1, 5):
        states[panel][later_return] = other(initials[later_return])
    if return_panel < 4:
        states[return_panel + 1][later_return] = initials[later_return]

    # This token repeats home first, departs later, and never returns.
    leave_panel = rng.choice((2, 3))
    for panel in range(leave_panel, 5):
        states[panel][delayed_departure] = other(initials[delayed_departure])

    if _winner(states) != target:
        raise AssertionError("sampler failed to construct its unique recurrence")

    target_color = (seed // 4) % 4
    color_order = [-1] * 4
    color_order[target] = target_color
    remaining_colors = [value for value in range(4) if value != target_color]
    rng.shuffle(remaining_colors)
    for identity, color in zip((i for i in range(4) if i != target), remaining_colors, strict=True):
        color_order[identity] = color

    lane_orders = []
    for _ in range(5):
        order = list(range(4))
        rng.shuffle(order)
        lane_orders.append(order)

    return {
        "states": states,
        "lane_orders": lane_orders,
        "color_order": color_order,
        "palette_variant": (seed // 16) % len(PALETTES),
        "layout_dx": rng.randint(-6, 6),
        "layout_dy": rng.randint(-5, 5),
    }


def analytic_gold(scene: dict[str, Any]) -> str:
    states = scene["states"]
    return LABELS[_winner(states)]


def margin(scene: dict[str, Any]) -> float:
    # Arrowhead mass shifts the pixel centroid by more than five pixels along
    # the winning axis; three is the conservative nearest-direction threshold.
    return 5.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 3.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []


def _centered_text(draw: ImageDraw.ImageDraw, center: tuple[float, float], text: str, font: Any, fill: tuple[int, int, int]) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((center[0] - (box[2] - box[0]) / 2, center[1] - (box[3] - box[1]) / 2 - box[1]), text, font=font, fill=fill)


def _draw_token(draw: ImageDraw.ImageDraw, center: tuple[int, int], identity: int, color: tuple[int, int, int]) -> None:
    cx, cy = center
    radius = 23
    draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=color)
    for ox, oy in _PIP_OFFSETS[identity]:
        draw.ellipse((cx + ox - 3, cy + oy - 3, cx + ox + 3, cy + oy + 3), fill=INK)


def _draw_arrow(draw: ImageDraw.ImageDraw, center: tuple[int, int], direction: int) -> None:
    cx, cy = center
    dx, dy = _DIRECTION_VECTORS[int(direction)]
    px, py = -dy, dx
    tail = (cx - 8 * dx, cy - 8 * dy)
    neck = (cx + 9 * dx, cy + 9 * dy)
    tip = (cx + 20 * dx, cy + 20 * dy)
    draw.line((*tail, *neck), fill=INK, width=6)
    draw.polygon(
        (
            tip,
            (cx + 7 * dx + 8 * px, cy + 7 * dy + 8 * py),
            (cx + 7 * dx - 8 * px, cy + 7 * dy - 8 * py),
        ),
        fill=INK,
    )


def render(scene: dict[str, Any]) -> Image.Image:
    states = [[int(value) for value in panel] for panel in scene["states"]]
    lane_orders = [[int(value) for value in panel] for panel in scene["lane_orders"]]
    color_order = [int(value) for value in scene["color_order"]]
    palette_variant = int(scene["palette_variant"])
    dx = int(scene["layout_dx"])
    dy = int(scene["layout_dy"])
    if len(states) != 5 or any(len(panel) != 4 for panel in states):
        raise ValueError("states must be a 5 by 4 ledger")
    if len(lane_orders) != 5 or any(sorted(panel) != list(range(4)) for panel in lane_orders):
        raise ValueError("each panel must contain every identity once")
    if sorted(color_order) != list(range(4)):
        raise ValueError("color_order must be a permutation")
    palette = PALETTES[palette_variant]

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((17, 14, WIDTH - 17, HEIGHT - 14), radius=24, fill=PAPER, outline=(220, 226, 235), width=2)
    draw.text((42, 31), "FIRST HOME-DIRECTION RETURN", font=_font(24), fill=INK)
    draw.text((42, 67), "Read left to right. A return counts only after that token has pointed away from its own first direction.", font=_font(16, False), fill=MUTED)
    draw.text((42, 104), "PIP KEY", font=_font(14), fill=INK)
    draw.text((119, 104), "A = 1     B = 2     C = 3     D = 4", font=_font(15, False), fill=MUTED)
    draw.line((42, 142, WIDTH - 42, 142), fill=(221, 226, 235), width=2)

    panel_w, panel_h, gap = 224, 505, 24
    start_x = (WIDTH - (5 * panel_w + 4 * gap)) // 2 + dx
    top = 159 + dy
    for panel in range(5):
        left = start_x + panel * (panel_w + gap)
        right = left + panel_w
        draw.rounded_rectangle((left, top, right, top + panel_h), radius=15, fill=CARD, outline=BORDER, width=3)
        _centered_text(draw, ((left + right) / 2, top + 25), f"PANEL {panel + 1}", _font(16), INK)
        draw.text((left + 23, top + 56), "TOKEN", font=_font(10), fill=MUTED)
        draw.text((left + 105, top + 56), "DIRECTION", font=_font(10), fill=MUTED)
        draw.line((left + 17, top + 75, right - 17, top + 75), fill=BORDER, width=2)
        for lane, identity in enumerate(lane_orders[panel]):
            cy = top + 116 + lane * 91
            token_center = (left + 55, cy)
            box_left, box_top = left + 116, cy - 27
            draw.rectangle((box_left, box_top, box_left + 54, box_top + 54), fill=ARROW_BG)
            _draw_token(draw, token_center, identity, palette[color_order[identity]])
            _draw_arrow(draw, (box_left + 27, box_top + 27), states[panel][identity])
        if panel < 4:
            arrow_x = right + gap // 2
            arrow_y = top + panel_h // 2
            draw.polygon(((arrow_x - 5, arrow_y - 8), (arrow_x + 8, arrow_y), (arrow_x - 5, arrow_y + 8)), fill=(137, 149, 168))
    return image
