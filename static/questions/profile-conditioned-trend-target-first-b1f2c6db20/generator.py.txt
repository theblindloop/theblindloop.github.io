"""Deterministic latent sampler and Pillow renderer for target-crossing plots."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 420
SCALE = 3
COLOR_RGB = {
    "red": (214, 55, 62),
    "blue": (40, 105, 210),
    "green": (36, 151, 92),
    "purple": (142, 68, 173),
}
COLORS = tuple(COLOR_RGB)
TARGET_RGB = (235, 139, 35)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    answer_color = COLORS[int(seed) % len(COLORS)]
    dx = rng.randint(-10, 10)
    dy = rng.randint(-8, 8)
    left, top, right, bottom = 72 + dx, 44 + dy, 604 + dx, 362 + dy
    target_y = rng.randint(190, 228) + dy

    first = rng.randint(left + 82, left + 128)
    gaps = [rng.randint(48, 72), rng.randint(54, 82), rng.randint(48, 76)]
    ordered = [first]
    for gap in gaps:
        ordered.append(ordered[-1] + gap)
    if ordered[-1] > right - 64:
        shift = ordered[-1] - (right - 64)
        ordered = [value - shift for value in ordered]

    remaining_colors = [color for color in COLORS if color != answer_color]
    rng.shuffle(remaining_colors)
    crossings = {answer_color: float(ordered[0])}
    crossings.update({color: float(x) for color, x in zip(remaining_colors, ordered[1:])})

    slopes: dict[str, float] = {}
    for color in COLORS:
        crossing = crossings[color]
        # In image coordinates negative slope means the trend rises left-to-right.
        max_left = (bottom - 18 - target_y) / max(crossing - left, 1.0)
        max_right = (target_y - (top + 18)) / max(right - crossing, 1.0)
        upper = min(0.34, max_left, max_right)
        lower = min(0.15, upper * 0.72)
        slopes[color] = -rng.uniform(max(0.10, lower), max(0.105, upper))

    return {
        "colors": list(COLORS),
        "crossings": crossings,
        "slopes": slopes,
        "target_y": float(target_y),
        "plot_offset": [dx, dy],
    }


def _scaled(point: tuple[float, float]) -> tuple[int, int]:
    return (round(point[0] * SCALE), round(point[1] * SCALE))


def render(scene: dict) -> Image.Image:
    colors = scene["colors"]
    crossings = scene["crossings"]
    slopes = scene["slopes"]
    target_y = float(scene["target_y"])
    dx, dy = scene["plot_offset"]
    left, top, right, bottom = 72 + dx, 44 + dy, 604 + dx, 362 + dy

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), (250, 250, 248))
    draw = ImageDraw.Draw(image)
    # Pale chart area and a visible measurement grid.
    draw.rectangle([*_scaled((left, top)), *_scaled((right, bottom))], fill=(255, 255, 255))
    for x in range(left + 52, right, 52):
        draw.line([_scaled((x, top)), _scaled((x, bottom))], fill=(226, 229, 232), width=2 * SCALE)
    for y in range(top + 53, bottom, 53):
        draw.line([_scaled((left, y)), _scaled((right, y))], fill=(226, 229, 232), width=2 * SCALE)

    axis = (75, 79, 85)
    draw.line([_scaled((left, bottom)), _scaled((right + 9, bottom))], fill=axis, width=3 * SCALE)
    draw.line([_scaled((left, bottom)), _scaled((left, top - 8))], fill=axis, width=3 * SCALE)
    draw.polygon([_scaled((right + 9, bottom)), _scaled((right, bottom - 5)), _scaled((right, bottom + 5))], fill=axis)
    draw.polygon([_scaled((left, top - 8)), _scaled((left - 5, top + 1)), _scaled((left + 5, top + 1))], fill=axis)

    # The target is drawn first; colored traces remain continuous at crossings.
    dash = 14
    x = left + 2
    while x < right:
        draw.line(
            [_scaled((x, target_y)), _scaled((min(x + dash, right), target_y))],
            fill=TARGET_RGB,
            width=4 * SCALE,
        )
        x += dash + 9

    for color in colors:
        crossing = float(crossings[color])
        slope = float(slopes[color])
        y_left = target_y + slope * (left - crossing)
        y_right = target_y + slope * (right - crossing)
        draw.line(
            [_scaled((left, y_left)), _scaled((right, y_right))],
            fill=COLOR_RGB[color],
            width=7 * SCALE,
        )
        # Round endpoint caps keep all traces equally legible.
        radius = 3.5 * SCALE
        for px, py in (_scaled((left, y_left)), _scaled((right, y_right))):
            draw.ellipse([px - radius, py - radius, px + radius, py + radius], fill=COLOR_RGB[color])

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    crossings = scene["crossings"]
    colors = scene["colors"]
    return min(colors, key=lambda color: float(crossings[color]))


def margin(scene: dict) -> float:
    ordered = sorted(float(value) for value in scene["crossings"].values())
    return ordered[1] - ordered[0]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 18.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
