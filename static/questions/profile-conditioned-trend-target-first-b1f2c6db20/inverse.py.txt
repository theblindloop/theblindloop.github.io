"""Independent inverse arm: recover the answer from RGB pixels only."""

from __future__ import annotations

import numpy as np
from PIL import Image

COLOR_RGB = {
    "red": (214, 55, 62),
    "blue": (40, 105, 210),
    "green": (36, 151, 92),
    "purple": (142, 68, 173),
}
TARGET_RGB = (235, 139, 35)


def _mask_near(array: np.ndarray, rgb: tuple[int, int, int], radius: float) -> np.ndarray:
    delta = array.astype(np.float32) - np.asarray(rgb, dtype=np.float32)
    return np.sum(delta * delta, axis=2) <= radius * radius


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    target_mask = _mask_near(array, TARGET_RGB, 42.0)
    ty, tx = np.nonzero(target_mask)
    if len(tx) < 180:
        raise ValueError("orange dashed target is absent or unreadable")
    target_y = float(np.median(ty))

    crossings: dict[str, float] = {}
    for name, rgb in COLOR_RGB.items():
        mask = _mask_near(array, rgb, 44.0)
        ys, xs = np.nonzero(mask)
        if len(xs) < 900 or float(xs.max() - xs.min()) < 350:
            raise ValueError(f"{name} trace is absent or too short")
        design = np.column_stack((xs.astype(np.float64), np.ones(len(xs))))
        slope, intercept = np.linalg.lstsq(design, ys.astype(np.float64), rcond=None)[0]
        if not -0.6 < slope < -0.06:
            raise ValueError(f"{name} trace fit has implausible slope")
        crossing = (target_y - intercept) / slope
        if not 60.0 < crossing < array.shape[1] - 20.0:
            raise ValueError(f"{name} crossing lies outside the plot")
        crossings[name] = float(crossing)

    ordered = sorted(crossings, key=crossings.get)
    if crossings[ordered[1]] - crossings[ordered[0]] < 10.0:
        raise ValueError("first crossing is pixel-ambiguous")
    return ordered[0]
