"""Pixel-only inverse arm for the dwell-band world.

Receives one rendered RGB image and nothing else: no scene, no renderer code,
no gold. It segments the shaded chart panel, recovers the four band boundaries
from row-wise shade changes, traces the thin dark stroke column by column, and
returns the 1-based index from the top of the band with the largest column
dwell.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

CHROMA_TOL = 10
PANEL_LO = 150
PANEL_HI = 250
INK_HI = 90
MIN_BAND_ROWS = 30
EXPECTED_BANDS = 4
ABSTAIN = "abstain"


def _fields(image: Image.Image):
    arr = np.asarray(image.convert("RGB")).astype(np.int16)
    chroma = arr.max(axis=2) - arr.min(axis=2)
    value = arr.mean(axis=2)
    achromatic = chroma <= CHROMA_TOL
    panel_like = achromatic & (value >= PANEL_LO) & (value <= PANEL_HI)
    ink = achromatic & (value <= INK_HI)
    return value, panel_like, ink


def _panel_box(panel_like: np.ndarray):
    rows = np.flatnonzero(panel_like.any(axis=1))
    cols = np.flatnonzero(panel_like.any(axis=0))
    if rows.size < 120 or cols.size < 120:
        return None
    return int(rows[0]), int(rows[-1]), int(cols[0]), int(cols[-1])


def _band_starts(value: np.ndarray, panel_like: np.ndarray, box) -> list[int] | None:
    r0, r1, c0, c1 = box
    shades: list[int] = []
    for row in range(r0, r1 + 1):
        mask = panel_like[row, c0 : c1 + 1]
        if not mask.any():
            shades.append(-1)
            continue
        values = np.rint(value[row, c0 : c1 + 1][mask]).astype(np.int64)
        shades.append(int(np.bincount(values, minlength=256).argmax()))
    # rows with no plain shade left visible inherit the shade above them
    for index in range(len(shades)):
        if shades[index] < 0:
            shades[index] = shades[index - 1] if index else -1
    if shades[0] < 0:
        return None
    starts = [r0]
    for index in range(1, len(shades)):
        if shades[index] != shades[index - 1]:
            starts.append(r0 + index)
    if len(starts) != EXPECTED_BANDS:
        return None
    spans = [starts[k + 1] - starts[k] for k in range(len(starts) - 1)] + [r1 + 1 - starts[-1]]
    if min(spans) < MIN_BAND_ROWS:
        return None
    return starts


def decision_from_image(image: Image.Image) -> str:
    value, panel_like, ink = _fields(image)
    box = _panel_box(panel_like)
    if box is None:
        return ABSTAIN
    starts = _band_starts(value, panel_like, box)
    if starts is None:
        return ABSTAIN
    r0, r1, c0, c1 = box
    counts = [0] * len(starts)
    covered = 0
    for column in range(c0, c1 + 1):
        rows = np.flatnonzero(ink[r0 : r1 + 1, column])
        if rows.size == 0:
            continue
        covered += 1
        middle = r0 + (int(rows[0]) + int(rows[-1])) / 2.0
        index = 0
        for k in range(1, len(starts)):
            if starts[k] <= middle:
                index = k
        counts[index] += 1
    if covered < (c1 - c0 + 1) * 0.9:
        return ABSTAIN
    order = sorted(range(len(counts)), key=lambda k: (-counts[k], k))
    return str(order[0] + 1)
