"""Independent pixel-only inverse for the reference-contour world.

This module intentionally knows only the public raster layout and palette.  It
extracts the reference and each of twelve candidate regions from the PNG,
normalizes every connected ink component for translation, and scores all masks
against the reference mask.  It never reads a scene, renderer, prompt, or
analytic gold value.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


_SIZE = (960, 760)
_INK = np.array((25, 47, 75), dtype=np.int16)
_CROP_HALF = 58
_REFERENCE_CENTER = (166, 105)
_PANEL_BOXES = (
    (32, 220, 464, 470),
    (496, 220, 928, 470),
    (32, 490, 464, 740),
    (496, 490, 928, 740),
)
_SLOT_OFFSETS = ((96, 145), (216, 145), (336, 145))


def _crop(image: Image.Image, center: tuple[int, int]) -> np.ndarray:
    cx, cy = center
    left, top = cx - _CROP_HALF, cy - _CROP_HALF
    right, bottom = cx + _CROP_HALF, cy + _CROP_HALF
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    if left < 0 or top < 0 or right > arr.shape[1] or bottom > arr.shape[0]:
        raise ValueError("contour crop falls outside the public canvas")
    patch = arr[top:bottom, left:right]
    distance = np.linalg.norm(patch - _INK, axis=2)
    # Labels and pale panel borders are deliberately a different palette; this
    # threshold keeps only the contour stroke and its immediate antialiasing.
    return distance <= 72.0


def _signature(mask: np.ndarray) -> np.ndarray:
    ys, xs = np.where(mask)
    if len(xs) < 40:
        raise ValueError("candidate crop has no measurable contour")
    y0, y1 = int(ys.min()), int(ys.max())
    x0, x1 = int(xs.min()), int(xs.max())
    patch = mask[y0 : y1 + 1, x0 : x1 + 1]
    canvas = np.zeros((120, 120), dtype=bool)
    if patch.shape[0] > 120 or patch.shape[1] > 120:
        raise ValueError("contour is too large for the public crop")
    top = (120 - patch.shape[0]) // 2
    left = (120 - patch.shape[1]) // 2
    canvas[top : top + patch.shape[0], left : left + patch.shape[1]] = patch
    return canvas


def _score(reference: np.ndarray, candidate: np.ndarray) -> int:
    return int(np.count_nonzero(np.logical_xor(reference, candidate)))


def decision_from_image(image: Image.Image) -> str:
    """Return A/B/C/D from the PNG alone."""
    if image.size != _SIZE:
        raise ValueError(f"expected {_SIZE} image, got {image.size}")
    reference = _signature(_crop(image, _REFERENCE_CENTER))
    panel_scores: list[int] = []
    for x0, y0, _x1, _y1 in _PANEL_BOXES:
        scores = []
        for offset_x, offset_y in _SLOT_OFFSETS:
            candidate = _signature(_crop(image, (x0 + offset_x, y0 + offset_y)))
            scores.append(_score(reference, candidate))
        panel_scores.append(min(scores))
    best = min(panel_scores)
    winners = [index for index, score in enumerate(panel_scores) if score == best]
    if len(winners) != 1:
        raise ValueError(f"ambiguous contour match: panel scores {panel_scores}")
    return "ABCD"[winners[0]]
