"""Renderer and analytic scorer for a reference-contour visual search world.

The scene contains no answer-only field.  ``reference`` and ``panels`` are the
only scene fields used by both rendering and analytic scoring; panel identity
is recovered by comparing visible point sequences, while the PNG inverse uses
only the rasterized contour masks.
"""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont


WIDTH, HEIGHT = 960, 760
BACKGROUND = (250, 249, 246)
INK = (25, 47, 75)
LABEL = (83, 101, 123)
BORDER = (198, 208, 220)
PANEL_FILL = (247, 249, 251)
REFERENCE_FRAME = (32, 24, 300, 184)
REFERENCE_CENTER = (166, 105)
PANEL_BOXES = (
    (32, 220, 464, 470),
    (496, 220, 928, 470),
    (32, 490, 464, 740),
    (496, 490, 928, 740),
)
SLOT_OFFSETS = ((96, 145), (216, 145), (336, 145))

# A deliberately asymmetric, closed nine-corner contour.  The reference and
# its exact copy use the same integer geometry; distractors move one corner by
# a visible amount while retaining the same count, stroke, and rough perimeter.
BASE_POINTS = (
    (-40, -16),
    (-28, -37),
    (4, -42),
    (37, -29),
    (30, -6),
    (42, 14),
    (23, 39),
    (-11, 34),
    (-40, 18),
)
TEMPLATE_ADJUSTMENTS = (
    (),
    ((1, 4, 3), (5, -3, 2)),
    ((3, -4, 2), (7, -4, -3)),
    ((0, 0, 5), (6, 4, -4)),
)
MUTATIONS = (
    (2, (8, 2)),
    (3, (-8, 2)),
    (4, (7, -4)),
    (5, (-7, 2)),
    (6, (6, -6)),
    (7, (-7, 3)),
    (8, (6, -4)),
    (1, (7, 4)),
    (2, (-7, 5)),
    (3, (5, -7)),
    (6, (-6, 5)),
    (0, (8, 3)),
)


def _reference_points(template: int) -> list[list[int]]:
    points = [list(point) for point in BASE_POINTS]
    for index, dx, dy in TEMPLATE_ADJUSTMENTS[template % len(TEMPLATE_ADJUSTMENTS)]:
        points[index][0] += dx
        points[index][1] += dy
    return points


def _mutated(points: list[list[int]], mutation_index: int, near: bool) -> list[list[int]]:
    out = [list(point) for point in points]
    point_index, (dx, dy) = MUTATIONS[mutation_index % len(MUTATIONS)]
    if near:
        # The first member of each panel's distractor trio becomes a visible
        # near-boundary perturbation.  It is still not pixel-identical.
        dx = 2 if dx >= 0 else -2
        dy = 1 if dy >= 0 else -1
    out[point_index][0] += dx
    out[point_index][1] += dy
    return out


def sample_scene(seed: int) -> dict:
    """Sample one deterministic, JSON-serializable scene."""
    rng = random.Random(int(seed))
    template = rng.randrange(len(TEMPLATE_ADJUSTMENTS))
    reference = _reference_points(template)
    target_panel = (int(seed) * 7 + 1) % 4
    near = int(seed) % 13 == 0
    variants = [_mutated(reference, index, near) for index in range(len(MUTATIONS))]

    panels: list[list[list[list[int]]]] = []
    for panel_index in range(4):
        shapes = [variants[panel_index * 3 + offset] for offset in range(3)]
        if panel_index == target_panel:
            shapes[0] = [list(point) for point in reference]
        rng.shuffle(shapes)
        panels.append(shapes)
    return {"reference": reference, "panels": panels}


def _shape_distance(reference: list[list[int]], shape: list[list[int]]) -> float:
    if len(reference) != len(shape):
        return float("inf")
    return float(
        sum(abs(int(a[0]) - int(b[0])) + abs(int(a[1]) - int(b[1])) for a, b in zip(reference, shape))
    )


def _panel_scores(scene: dict) -> list[float]:
    reference = scene["reference"]
    panels = scene["panels"]
    return [min(_shape_distance(reference, shape) for shape in panel) for panel in panels]


def analytic_gold(scene: dict) -> str:
    scores = _panel_scores(scene)
    best = min(scores)
    winners = [index for index, score in enumerate(scores) if score == best]
    if len(winners) != 1:
        raise ValueError("scene has no unique contour-match panel")
    return "ABCD"[winners[0]]


def margin(scene: dict) -> float:
    scores = sorted(_panel_scores(scene))
    if len(scores) < 2:
        return 0.0
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    scores = _panel_scores(scene)
    if len(scores) != 4:
        return True
    best = min(scores)
    # Only a tied best panel is ambiguous; tied runner-up panels do not affect
    # the selected answer and should not erase otherwise balanced classes.
    return scores.count(best) != 1 or margin(scene) < 6.0


def _translated(points: Iterable[Iterable[int]], center: tuple[int, int]) -> list[tuple[int, int]]:
    cx, cy = center
    return [(cx + int(point[0]), cy + int(point[1])) for point in points]


def _draw_contour(draw: ImageDraw.ImageDraw, points: list[list[int]], center: tuple[int, int]) -> None:
    contour = _translated(points, center)
    draw.line(contour + [contour[0]], fill=INK, width=6, joint="curve")


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    draw.text((40, 8), "REFERENCE OUTLINE", fill=LABEL, font=font)
    draw.rectangle(REFERENCE_FRAME, outline=BORDER, width=2)
    _draw_contour(draw, scene["reference"], REFERENCE_CENTER)

    labels = "ABCD"
    for panel_index, (x0, y0, x1, y1) in enumerate(PANEL_BOXES):
        draw.rectangle((x0, y0, x1, y1), fill=PANEL_FILL, outline=BORDER, width=2)
        draw.text((x0 + 14, y0 + 12), labels[panel_index], fill=LABEL, font=font)
        for shape, (offset_x, offset_y) in zip(scene["panels"][panel_index], SLOT_OFFSETS):
            _draw_contour(draw, shape, (x0 + offset_x, y0 + offset_y))
    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """There are no unused latent fields or pixel-identical label aliases."""
    return []
