"""Deterministic renderer and analytic label for the pattern-count world."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "color-bound-pattern-count-correspondence-0.1.0"
WIDTH, HEIGHT = 1000, 430
BACKGROUND = (250, 249, 244)
LIGHT = (255, 255, 252)
INK = (35, 39, 44)
COLORS = {
    "red": (214, 61, 57),
    "blue": (43, 103, 171),
    "green": (42, 145, 84),
    "orange": (226, 132, 42),
    "purple": (126, 74, 164),
}
COLOR_ORDER = tuple(COLORS)
PANEL_CENTERS = ((170, 220), (500, 220), (830, 220))
PANEL_BOXES = ((20, 78, 320, 390), (350, 78, 650, 390), (680, 78, 980, 390))
SLOTS = ((-100, -72), (-34, -72), (32, -72), (-67, 35), (-1, 35))
TILE_W, TILE_H = 52, 78


def _color_map(panel: dict) -> dict[str, int]:
    return {str(color): int(count) for color, count in panel.items()}


def _valid_panel(panel: dict) -> bool:
    try:
        colors = [str(color) for color in panel]
        counts = [int(count) for count in panel.values()]
        return (
            len(panel) == 5
            and sorted(colors) == sorted(COLOR_ORDER)
            and sorted(counts) == [0, 1, 2, 3, 4]
        )
    except (KeyError, TypeError, ValueError):
        return False


def _score(reference: dict, candidate: dict) -> int:
    reference_map = _color_map(reference)
    candidate_map = _color_map(candidate)
    return sum(reference_map[color] == candidate_map[color] for color in COLOR_ORDER)


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene whose answer-bearing fields are all rendered."""
    rng = random.Random(int(seed))
    colors = list(COLOR_ORDER)
    rng.shuffle(colors)
    counts = list(range(5))
    rng.shuffle(counts)
    def panel_from_mapping(mapping: dict[str, int]) -> dict:
        order = list(COLOR_ORDER)
        rng.shuffle(order)
        return {color: mapping[color] for color in order}

    reference_map = {color: count for color, count in zip(colors, counts)}
    reference = panel_from_mapping(reference_map)
    matching = panel_from_mapping(reference_map)
    wrong_map = dict(reference_map)
    first, second = rng.sample(list(COLOR_ORDER), 2)
    wrong_map[first], wrong_map[second] = wrong_map[second], wrong_map[first]
    wrong = panel_from_mapping(wrong_map)
    if int(seed) % 2 == 0:
        candidate_a, candidate_b = matching, wrong
    else:
        candidate_a, candidate_b = wrong, matching
    return {
        "reference": reference,
        "candidate_a": candidate_a,
        "candidate_b": candidate_b,
    }


def _draw_tile(draw: ImageDraw.ImageDraw, cx: int, cy: int, slot: int, color_name: str, count: int) -> None:
    left = int(round(cx + SLOTS[slot][0] - TILE_W / 2))
    top = int(round(cy + SLOTS[slot][1] - TILE_H / 2))
    right, bottom = left + TILE_W, top + TILE_H
    color = COLORS[str(color_name)]
    draw.rectangle((left, top, right, bottom), fill=color, outline=INK, width=2)
    for index in range(count):
        y = top + 13 + index * 12
        draw.rectangle((left + 10, y, right - 10, y + 4), fill=LIGHT)


def _draw_panel(draw: ImageDraw.ImageDraw, panel: dict, center: tuple[int, int], label: str) -> None:
    cx, cy = center
    for slot, color in enumerate(COLOR_ORDER):
        _draw_tile(draw, cx, cy, slot, color, int(panel[color]))
    draw.text((cx, 96), "REFERENCE" if label == "REFERENCE" else "CANDIDATE", fill=INK, anchor="ma")
    draw.text((cx, 408), label, fill=INK, anchor="ma")


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((WIDTH // 2, 26), "COLOR-BOUND INTERIOR PATTERN", fill=INK, anchor="ma")
    for box in PANEL_BOXES:
        draw.rectangle(box, outline=INK, width=2)
    _draw_panel(draw, scene["reference"], PANEL_CENTERS[0], "REFERENCE")
    _draw_panel(draw, scene["candidate_a"], PANEL_CENTERS[1], "A")
    _draw_panel(draw, scene["candidate_b"], PANEL_CENTERS[2], "B")
    return image


def analytic_gold(scene: dict) -> str:
    reference = scene["reference"]
    candidate_a = scene["candidate_a"]
    candidate_b = scene["candidate_b"]
    score_a = _score(reference, candidate_a)
    score_b = _score(reference, candidate_b)
    if score_a == score_b:
        raise ValueError("candidate pattern profiles are tied")
    return "A" if score_a > score_b else "B"


def margin(scene: dict) -> float:
    reference = scene["reference"]
    return float(abs(_score(reference, scene["candidate_a"]) - _score(reference, scene["candidate_b"])))


def is_quarantined(scene: dict) -> bool:
    try:
        panels = [scene["reference"], scene["candidate_a"], scene["candidate_b"]]
        if not all(_valid_panel(panel) for panel in panels):
            return True
        return margin(scene) < 1.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
