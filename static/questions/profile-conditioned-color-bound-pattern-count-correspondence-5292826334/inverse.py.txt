"""Independent image-only inverse for the pattern-count correspondence."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

WIDTH, HEIGHT = 1000, 430
PANEL_X_RANGES = ((20, 320), (350, 650), (680, 980))
PANEL_Y_RANGE = (78, 390)
COLORS = (
    (214, 61, 57),
    (43, 103, 171),
    (42, 145, 84),
    (226, 132, 42),
    (126, 74, 164),
)
LIGHT = (255, 255, 252)


def _largest_component(mask: np.ndarray) -> tuple[int, int, int, int, int]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 300:
        raise ValueError("colored tile is missing or too small")
    remaining = set(zip(xs.tolist(), ys.tolist()))
    best: list[tuple[int, int]] = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        component = [start]
        while stack:
            x, y = stack.pop()
            for neighbor in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
                if neighbor in remaining:
                    remaining.remove(neighbor)
                    stack.append(neighbor)
                    component.append(neighbor)
        if len(component) > len(best):
            best = component
    if len(best) < 300:
        raise ValueError("colored tile component is ambiguous")
    bx = [point[0] for point in best]
    by = [point[1] for point in best]
    return min(bx), min(by), max(bx), max(by), len(best)


def _count_bands(array: np.ndarray, x0: int, x1: int, color: tuple[int, int, int]) -> int:
    y0, y1 = PANEL_Y_RANGE
    panel = array[y0:y1, x0:x1]
    mask = np.all(panel == np.asarray(color, dtype=np.uint8), axis=2)
    left, top, right, bottom, _ = _largest_component(mask)
    if right - left < 30 or bottom - top < 55:
        raise ValueError("tile bounds are too small")
    light = np.all(panel == np.asarray(LIGHT, dtype=np.uint8), axis=2)
    x_left = left + 6
    x_right = right - 5
    active = []
    for y in range(top, bottom + 1):
        active.append(int(light[y, x_left:x_right + 1].sum()) >= 20)
    groups = 0
    in_group = False
    for value in active:
        if value and not in_group:
            groups += 1
            in_group = True
        elif not value:
            in_group = False
    if groups > 4:
        raise ValueError("too many interior band groups")
    return groups


def _panel_profile(array: np.ndarray, x_range: tuple[int, int]) -> dict[int, int]:
    return {index: _count_bands(array, x_range[0], x_range[1], color) for index, color in enumerate(COLORS)}


def _score(reference: dict[int, int], candidate: dict[int, int]) -> int:
    return sum(reference[key] == candidate[key] for key in reference)


def decision_from_image(image: Image.Image) -> str:
    """Recover A/B from RGB pixels using color masks, bounds, and row groups."""
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.shape[:2] != (HEIGHT, WIDTH):
        raise ValueError(f"unexpected raster size {array.shape[:2]}")
    reference = _panel_profile(array, PANEL_X_RANGES[0])
    score_a = _score(reference, _panel_profile(array, PANEL_X_RANGES[1]))
    score_b = _score(reference, _panel_profile(array, PANEL_X_RANGES[2]))
    if score_a == score_b:
        raise ValueError("pixel profile scores are tied")
    return "A" if score_a > score_b else "B"
