"""Deterministic renderer and analytic arm for Doublet Run Ribbon."""

from __future__ import annotations

import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 900, 600
BACKGROUND = (247, 245, 239)
INK = (42, 47, 57)
CYAN = (36, 184, 201)
CORAL = (235, 105, 86)
GOLD = (238, 184, 48)
PALE = (231, 228, 219)
SPAN_WIDTH = 64
SPAN_COUNT = 8
ROUTE_LEFT = 194
ROUTE_CENTER_Y = 337


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    path = Path("/usr/share/fonts/truetype/dejavu") / name
    return ImageFont.truetype(str(path), size=size)


def _replay_spans(spans: list[str]) -> str:
    """Toggle once per maximal run of exactly two equal span colors."""
    if not spans or any(span not in ("C", "V") for span in spans):
        raise ValueError("spans must be a nonempty C/V sequence")
    qualifying_runs = 0
    run_length = 1
    for previous, current in zip(spans, spans[1:]):
        if current == previous:
            run_length += 1
        else:
            qualifying_runs += int(run_length == 2)
            run_length = 1
    qualifying_runs += int(run_length == 2)
    return "yes" if qualifying_runs % 2 else "no"


def sample_scene(seed: int) -> dict:
    """Balance answers while fixing total ink at four spans of each color."""
    rng = random.Random(int(seed))
    target = ("no", "yes")[int(seed) % 2]
    for _ in range(5000):
        spans = ["C"] * 4 + ["V"] * 4
        rng.shuffle(spans)
        if _replay_spans(spans) == target:
            return {
                "spans": spans,
                "thickness": 22 + 3 * (int(seed) % 5),
            }
    raise RuntimeError("could not sample a balanced operation sequence")


def analytic_gold(scene: dict) -> str:
    return _replay_spans(list(scene["spans"]))


def margin(scene: dict) -> float:
    return float(scene["thickness"])


def is_quarantined(scene: dict) -> bool:
    spans = list(scene.get("spans", []))
    return (
        len(spans) != SPAN_COUNT
        or spans.count("C") != 4
        or spans.count("V") != 4
        or int(scene.get("thickness", 0)) < 20
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    spans = list(scene["spans"])
    thickness = int(scene["thickness"])
    if len(spans) != SPAN_COUNT:
        raise ValueError("the public world requires exactly eight spans")

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(31, True)
    body_font = _font(19)
    small_bold = _font(17, True)
    label_font = _font(18, True)

    draw.text((70, 54), "WATCH THE COLOR RUNS", fill=INK, font=title_font)
    draw.text(
        (70, 101),
        "The lamp begins OFF",
        fill=INK,
        font=body_font,
    )

    # Compact visible initial state. The public question also declares it; the
    # image-only oracle intentionally derives the answer from the ribbon.
    box = (344, 154, 556, 246)
    draw.rounded_rectangle(box, radius=14, fill=(252, 251, 247), outline=INK, width=3)
    draw.text((371, 171), "START", fill=INK, font=small_bold)
    draw.ellipse((472, 169, 506, 203), fill=PALE, outline=INK, width=3)
    draw.text((459, 211), "OFF", fill=INK, font=label_font)
    draw.line((466, 235, 512, 235), fill=GOLD, width=6)

    y0 = ROUTE_CENTER_Y - thickness // 2
    y1 = y0 + thickness - 1
    route_right = ROUTE_LEFT + SPAN_COUNT * SPAN_WIDTH

    # The black backing and touching endpoint symbols make one connected trace.
    draw.rounded_rectangle(
        (ROUTE_LEFT - 5, y0 - 5, route_right + 5, y1 + 5),
        radius=(thickness + 10) // 2,
        fill=INK,
    )
    draw.ellipse(
        (ROUTE_LEFT - 45, ROUTE_CENTER_Y - 25, ROUTE_LEFT + 5, ROUTE_CENTER_Y + 25),
        fill=GOLD,
        outline=INK,
        width=4,
    )
    for index, span in enumerate(spans):
        x0 = ROUTE_LEFT + index * SPAN_WIDTH + 2
        x1 = ROUTE_LEFT + (index + 1) * SPAN_WIDTH - 3
        color = CYAN if span == "C" else CORAL
        draw.rectangle((x0, y0, x1, y1), fill=color)

    tip_x = route_right + 42
    draw.polygon(
        [
            (route_right + 1, ROUTE_CENTER_Y - 26),
            (tip_x, ROUTE_CENTER_Y),
            (route_right + 1, ROUTE_CENTER_Y + 26),
        ],
        fill=INK,
    )
    draw.text((ROUTE_LEFT - 55, ROUTE_CENTER_Y + 48), "START", fill=INK, font=small_bold)
    draw.text((236, 431), "GROUP ADJACENT SPANS OF THE SAME COLOR", fill=INK, font=small_bold)
    draw.text((245, 463), "A RUN OF EXACTLY 2 FLIPS THE LAMP ONCE", fill=INK, font=small_bold)
    draw.text((258, 527), "Include the final run when you reach the arrow.", fill=INK, font=body_font)
    return image
