"""Scene sampling, rendering, and analytic gold for wire_bend_bead_spacing_binding.

Two panels each show a thin rectilinear "wire" of the same total length. The wire
carries four beads at fixed distances measured along the wire. The lower wire is
bent differently and may be traversed from either end, so only the spacing
pattern of the beads identifies which bead is which.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "wire_bend_bead_spacing_binding-renderer-0.1.0"

WIDTH = 560
HEIGHT = 660
DIVIDER_TOP = 322
DIVIDER_BOTTOM = 326
PANEL_TOP = (30, 30, 530, 300)  # x0, y0, x1, y1 usable region of the upper panel
PANEL_BOTTOM = (30, 356, 530, 626)

TOTAL_LENGTH = 680
WIRE_WIDTH = 3
BEAD_RADIUS = 9
RING_RADIUS = 16
RING_WIDTH = 3

BACKGROUND = (255, 255, 255)
WIRE_COLOR = (70, 70, 70)
GRAY_BEAD = (150, 150, 150)
RING_COLOR = (0, 0, 0)
DIVIDER_COLOR = (0, 0, 0)

COLOR_NAMES = ("red", "blue", "green", "orange")
COLOR_RGB = {
    "red": (220, 40, 40),
    "blue": (40, 80, 220),
    "green": (30, 150, 60),
    "orange": (245, 150, 20),
}

MIN_BEAD_GAP = 85
END_CLEARANCE = 70
CORNER_CLEARANCE = 28
QUARANTINE_MARGIN = 90.0
MIN_SAMPLED_MARGIN = 40.0


# --------------------------------------------------------------------------
# geometry helpers
# --------------------------------------------------------------------------
def _serpentine(geom: dict) -> list[tuple[int, int]]:
    """Vertices of a five-segment serpentine: H, V, H, V, H."""
    x, y = geom["x0"], geom["y0"]
    sign = geom["sign"]
    pts = [(x, y)]
    x += sign * geom["a"]
    pts.append((x, y))
    y += geom["p"]
    pts.append((x, y))
    x -= sign * geom["b"]
    pts.append((x, y))
    y += geom["q"]
    pts.append((x, y))
    x += sign * geom["c"]
    pts.append((x, y))
    return pts


def _segment_lengths(geom: dict) -> list[int]:
    return [geom["a"], geom["p"], geom["b"], geom["q"], geom["c"]]


def _corner_offsets(geom: dict) -> list[int]:
    lengths = _segment_lengths(geom)
    out, run = [], 0
    for length in lengths[:-1]:
        run += length
        out.append(run)
    return out


def _point_at(vertices: list[tuple[int, int]], distance: float) -> tuple[float, float]:
    remaining = distance
    for start, end in zip(vertices, vertices[1:]):
        seg = abs(end[0] - start[0]) + abs(end[1] - start[1])
        if remaining <= seg:
            frac = remaining / seg if seg else 0.0
            return (start[0] + (end[0] - start[0]) * frac, start[1] + (end[1] - start[1]) * frac)
        remaining -= seg
    return (float(vertices[-1][0]), float(vertices[-1][1]))


def _sample_geometry(rng: random.Random, panel: tuple[int, int, int, int], offsets: list[int]) -> dict:
    """Sample a serpentine of exactly TOTAL_LENGTH that fits and clears every bead."""
    x0p, y0p, x1p, y1p = panel
    span_x = x1p - x0p
    span_y = y1p - y0p
    for _ in range(4000):
        p = rng.randrange(55, 101)
        q = rng.randrange(55, 101)
        if p + q > span_y - 40:
            continue
        rest = TOTAL_LENGTH - p - q
        a = rng.randrange(90, 251)
        b = rng.randrange(90, 251)
        c = rest - a - b
        if not 90 <= c <= 250:
            continue
        sign = rng.choice((1, -1))
        xs = [0, a, a - b, a - b + c]
        extent = max(xs) - min(xs)
        if extent > span_x:
            continue
        left = min(xs)
        slack_x = span_x - extent
        slack_y = span_y - (p + q)
        if sign == 1:
            x0 = x0p - left + rng.randrange(0, slack_x + 1)
        else:
            x0 = x1p + left - rng.randrange(0, slack_x + 1)
        y0 = y0p + rng.randrange(0, slack_y + 1)
        geom = {"x0": int(x0), "y0": int(y0), "a": a, "p": p, "b": b, "q": q, "c": c, "sign": sign}
        corners = _corner_offsets(geom)
        if any(abs(off - corner) < CORNER_CLEARANCE for off in offsets for corner in corners):
            continue
        return geom
    raise RuntimeError("failed to sample a wire geometry")


def _margin_px(offsets: list[int]) -> float:
    """L1 mismatch (in pixels) between the true spacing signature and its reversal."""
    total = 0.0
    n = len(offsets)
    for i in range(n):
        total += abs(TOTAL_LENGTH - offsets[n - 1 - i] - offsets[i])
    return float(total)


def _sample_offsets(rng: random.Random, near_boundary: bool) -> list[int]:
    if near_boundary:
        for _ in range(4000):
            r0 = rng.randrange(END_CLEARANCE, 220)
            r1 = rng.randrange(r0 + MIN_BEAD_GAP, 330)
            d0 = rng.choice((-1, 1)) * rng.randrange(8, 23)
            d1 = rng.choice((-1, 1)) * rng.randrange(8, 23)
            if not 20 <= abs(d0) + abs(d1) < 45:
                continue
            r2 = TOTAL_LENGTH - r1 + d1
            r3 = TOTAL_LENGTH - r0 + d0
            offsets = [r0, r1, r2, r3]
            if offsets != sorted(offsets):
                continue
            if r3 > TOTAL_LENGTH - END_CLEARANCE:
                continue
            if min(b - a for a, b in zip(offsets, offsets[1:])) < MIN_BEAD_GAP:
                continue
            return offsets
        raise RuntimeError("failed to sample near-boundary offsets")
    for _ in range(4000):
        picks = sorted(rng.randrange(END_CLEARANCE, TOTAL_LENGTH - END_CLEARANCE + 1, 5) for _ in range(4))
        if min(b - a for a, b in zip(picks, picks[1:])) < MIN_BEAD_GAP:
            continue
        if _margin_px(picks) < 130.0:
            continue
        return list(picks)
    raise RuntimeError("failed to sample offsets")


# --------------------------------------------------------------------------
# public interface
# --------------------------------------------------------------------------
def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    near_boundary = rng.random() < 0.22
    offsets = _sample_offsets(rng, near_boundary)
    reversed_flag = rng.choice((False, True))
    if reversed_flag:
        lower_offsets = sorted(TOTAL_LENGTH - off for off in offsets)
    else:
        lower_offsets = list(offsets)
    top_geom = _sample_geometry(rng, PANEL_TOP, offsets)
    bottom_geom = _sample_geometry(rng, PANEL_BOTTOM, lower_offsets)
    colors = list(COLOR_NAMES)
    rng.shuffle(colors)
    query_index = rng.randrange(4)
    return {
        "offsets": [int(v) for v in offsets],
        "lower_offsets": [int(v) for v in lower_offsets],
        "reversed": bool(reversed_flag),
        "colors": colors,
        "query_index": int(query_index),
        "top_geom": top_geom,
        "bottom_geom": bottom_geom,
        "total_length": TOTAL_LENGTH,
        "near_boundary": bool(near_boundary),
    }


def scene_panels(scene: dict) -> tuple[list[tuple[float, float]], list[tuple[float, float]]]:
    """Bead centers in the upper and lower panels, ordered by distance along each wire."""
    top = _serpentine(scene["top_geom"])
    bottom = _serpentine(scene["bottom_geom"])
    upper = [_point_at(top, off) for off in scene["offsets"]]
    lower = [_point_at(bottom, off) for off in scene["lower_offsets"]]
    return upper, lower


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rectangle([0, DIVIDER_TOP, WIDTH - 1, DIVIDER_BOTTOM], fill=DIVIDER_COLOR)

    top_pts = _serpentine(scene["top_geom"])
    bottom_pts = _serpentine(scene["bottom_geom"])
    for pts in (top_pts, bottom_pts):
        for start, end in zip(pts, pts[1:]):
            draw.line([start, end], fill=WIRE_COLOR, width=WIRE_WIDTH)

    upper, lower = scene_panels(scene)
    for center, name in zip(upper, scene["colors"]):
        _disk(draw, center, BEAD_RADIUS, COLOR_RGB[name])
    for center in lower:
        _disk(draw, center, BEAD_RADIUS, GRAY_BEAD)

    query = lower[int(scene["query_index"])]
    draw.ellipse(
        [query[0] - RING_RADIUS, query[1] - RING_RADIUS, query[0] + RING_RADIUS, query[1] + RING_RADIUS],
        outline=RING_COLOR,
        width=RING_WIDTH,
    )
    return image


def _disk(draw: ImageDraw.ImageDraw, center: tuple[float, float], radius: int, fill) -> None:
    draw.ellipse(
        [center[0] - radius, center[1] - radius, center[0] + radius, center[1] + radius], fill=fill
    )


def analytic_gold(scene: dict) -> str:
    """Colour of the upper bead that the ringed lower bead corresponds to.

    The traversal direction is read back from the two drawn offset lists rather
    than from any unrendered flag, so every input to the label is on the raster.
    """
    upper = [int(v) for v in scene["offsets"]]
    lower = [int(v) for v in scene["lower_offsets"]]
    index = int(scene["query_index"])
    if lower != upper:
        index = len(upper) - 1 - index
    return scene["colors"][index]


def margin(scene: dict) -> float:
    return _margin_px([int(v) for v in scene["offsets"]])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases: every label input is drawn into the raster."""
    return []
