"""Parallel-pair panel search world.

Latent scene z = {width, height, palette, bars}. Each of four panels holds four
identical dark bars whose only free attributes are position and axis direction.
Exactly one panel contains two bars with an identical axis direction (an exactly
parallel pair); every other within-panel pair differs by at least MIN_SEPARATION
degrees. The decision is the panel holding the parallel pair, and the margin is
the smallest direction difference of any non-parallel within-panel pair.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "parallel_pair_panel-0.1.0"

CANVAS = 560
PANEL = CANVAS // 2
INSET = 14
CELL = (PANEL - 2 * INSET) // 2
BAR_LENGTH = 48.0
BAR_WIDTH = 9.0
CELL_JITTER = 14.0
SUPERSAMPLE = 3
DIRECTION_COUNT = 8
DIRECTION_STEP = 180.0 / DIRECTION_COUNT
BARS_PER_PANEL = 4
MIN_SEPARATION = 3.0
QUARANTINE_MARGIN = 8.0
JITTER_BUDGETS = (2.0, 2.0, 4.0, 7.0, 9.5)

PANELS = ("top_left", "top_right", "bottom_left", "bottom_right")

PALETTES = (
    {"background": (255, 255, 255), "bar": (26, 26, 30), "divider": (176, 176, 176)},
    {"background": (248, 246, 240), "bar": (18, 32, 46), "divider": (168, 170, 172)},
    {"background": (243, 246, 250), "bar": (40, 22, 24), "divider": (170, 178, 186)},
    {"background": (250, 250, 244), "bar": (22, 40, 26), "divider": (180, 176, 168)},
)


def _fold(delta: float) -> float:
    """Smallest unsigned difference between two axis directions (mod 180)."""
    delta = abs(delta) % 180.0
    return min(delta, 180.0 - delta)


def _panel_of(x: float, y: float) -> str:
    row = "top" if y < CANVAS / 2 else "bottom"
    col = "left" if x < CANVAS / 2 else "right"
    return f"{row}_{col}"


def _grouped(bars: list) -> dict:
    groups: dict = {panel: [] for panel in PANELS}
    for bar in bars:
        groups[_panel_of(bar["x"], bar["y"])].append(bar)
    return groups


def _pair_gaps(entries: list) -> list:
    gaps = []
    for i in range(len(entries)):
        for j in range(i + 1, len(entries)):
            gaps.append(_fold(entries[i] - entries[j]))
    return gaps


def analytic_gold(scene: dict) -> str:
    """Panel whose four bar directions contain a repeated (exactly parallel) pair."""
    winners = []
    for panel, bars in _grouped(scene["bars"]).items():
        angles = [bar["angle_deg"] % 180.0 for bar in bars]
        if any(gap == 0.0 for gap in _pair_gaps(angles)):
            winners.append(panel)
    if len(winners) != 1:
        raise ValueError(f"scene does not have a unique parallel-pair panel: {winners}")
    return winners[0]


def margin(scene: dict) -> float:
    """Smallest nonzero within-panel direction difference, in degrees."""
    gaps = []
    for bars in _grouped(scene["bars"]).values():
        angles = [bar["angle_deg"] % 180.0 for bar in bars]
        gaps.extend(gap for gap in _pair_gaps(angles) if gap > 0.0)
    return round(min(gaps), 6)


def is_quarantined(scene: dict) -> bool:
    """Near-boundary scenes whose closest non-parallel pair is hard to separate."""
    return bool(margin(scene) < QUARANTINE_MARGIN)


def latent_symmetries(scene: dict):
    """Exact latent aliases: axis direction is pi-periodic and bar order is free."""
    flipped = {
        key: ([dict(bar, angle_deg=bar["angle_deg"] + 180.0) for bar in value] if key == "bars" else value)
        for key, value in scene.items()
    }
    reordered = {
        key: (list(reversed(value)) if key == "bars" else value) for key, value in scene.items()
    }
    return [("axis_period_180", flipped), ("bar_order_permutation", reordered)]


def sample_scene(seed: int) -> dict:
    rng = random.Random(("parallel_pair_panel", seed).__repr__())
    target = PANELS[seed % len(PANELS)]
    palette_index = rng.randrange(len(PALETTES))
    for _ in range(4000):
        directions = {}
        duplicate_dir = None
        for panel in PANELS:
            if panel == target:
                chosen = rng.sample(range(DIRECTION_COUNT), BARS_PER_PANEL - 1)
                duplicate_dir = rng.choice(chosen)
                slots = chosen + [duplicate_dir]
            else:
                slots = rng.sample(range(DIRECTION_COUNT), BARS_PER_PANEL)
            rng.shuffle(slots)
            directions[panel] = slots
        elsewhere = any(
            duplicate_dir in slots for panel, slots in directions.items() if panel != target
        )
        if not elsewhere:
            continue
        budget = rng.choice(JITTER_BUDGETS)
        bars = []
        for panel, slots in directions.items():
            row = 0 if panel.startswith("top") else 1
            col = 0 if panel.endswith("left") else 1
            offsets = {}
            cells = [(0, 0), (0, 1), (1, 0), (1, 1)]
            rng.shuffle(cells)
            for slot_index, direction in enumerate(slots):
                if direction not in offsets:
                    offsets[direction] = rng.uniform(-budget, budget)
                angle = (direction * DIRECTION_STEP + offsets[direction]) % 180.0
                cell_row, cell_col = cells[slot_index]
                x = col * PANEL + INSET + CELL * (cell_col + 0.5) + rng.uniform(-CELL_JITTER, CELL_JITTER)
                y = row * PANEL + INSET + CELL * (cell_row + 0.5) + rng.uniform(-CELL_JITTER, CELL_JITTER)
                bars.append({"x": round(x, 4), "y": round(y, 4), "angle_deg": round(angle, 6)})
        scene = {
            "width": CANVAS,
            "height": CANVAS,
            "palette": palette_index,
            "bars": bars,
        }
        if analytic_gold(scene) != target:
            continue
        if margin(scene) < MIN_SEPARATION:
            continue
        return scene
    raise RuntimeError(f"could not sample a valid scene for seed {seed}")


def render(scene: dict) -> Image.Image:
    palette = PALETTES[scene["palette"] % len(PALETTES)]
    width = scene["width"]
    height = scene["height"]
    ss = SUPERSAMPLE
    image = Image.new("RGB", (width * ss, height * ss), palette["background"])
    draw = ImageDraw.Draw(image)
    mid_x = width * ss / 2
    mid_y = height * ss / 2
    line = max(1, ss)
    draw.line([(mid_x, 0), (mid_x, height * ss)], fill=palette["divider"], width=line)
    draw.line([(0, mid_y), (width * ss, mid_y)], fill=palette["divider"], width=line)
    draw.rectangle(
        [(0, 0), (width * ss - 1, height * ss - 1)], outline=palette["divider"], width=line
    )
    for bar in scene["bars"]:
        theta = math.radians(bar["angle_deg"] % 180.0)
        ux, uy = math.cos(theta), math.sin(theta)
        half_l, half_w = BAR_LENGTH / 2.0, BAR_WIDTH / 2.0
        polygon = []
        for sl, sw in ((1, 1), (1, -1), (-1, -1), (-1, 1)):
            px = bar["x"] + sl * half_l * ux - sw * half_w * uy
            py = bar["y"] + sl * half_l * uy + sw * half_w * ux
            polygon.append((round(px * ss), round(py * ss)))
        draw.polygon(polygon, fill=palette["bar"])
    return image.resize((width, height), Image.LANCZOS)
