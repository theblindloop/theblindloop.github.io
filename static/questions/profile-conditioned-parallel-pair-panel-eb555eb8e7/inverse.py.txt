"""Pixel-only decision arm for the parallel-pair panel world.

Receives a rendered RGB image and nothing else. It segments the dark bar ink,
labels connected components, estimates every bar's axis direction from its
second-order image moments, groups bars by panel using the image midlines, and
returns the panel whose closest pair of bar directions is the parallel one.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

DARK_LUMINANCE = 110.0
MIN_COMPONENT_AREA = 60
PANELS = ("top_left", "top_right", "bottom_left", "bottom_right")
AMBIGUOUS = "ambiguous"


def _dark_mask(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.float64)
    luminance = 0.299 * arr[:, :, 0] + 0.587 * arr[:, :, 1] + 0.114 * arr[:, :, 2]
    return luminance < DARK_LUMINANCE


def _components(mask: np.ndarray) -> list:
    height, width = mask.shape
    labels = np.zeros((height, width), dtype=np.int32)
    blobs = []
    current = 0
    for start_y in range(height):
        row = mask[start_y]
        for start_x in range(width):
            if not row[start_x] or labels[start_y, start_x]:
                continue
            current += 1
            stack = [(start_y, start_x)]
            labels[start_y, start_x] = current
            pixels = []
            while stack:
                y, x = stack.pop()
                pixels.append((y, x))
                for ny in (y - 1, y, y + 1):
                    if ny < 0 or ny >= height:
                        continue
                    for nx in (x - 1, x, x + 1):
                        if nx < 0 or nx >= width:
                            continue
                        if mask[ny, nx] and not labels[ny, nx]:
                            labels[ny, nx] = current
                            stack.append((ny, nx))
            if len(pixels) >= MIN_COMPONENT_AREA:
                blobs.append(np.array(pixels, dtype=np.float64))
    return blobs


def _axis_direction(pixels: np.ndarray) -> float:
    ys = pixels[:, 0]
    xs = pixels[:, 1]
    cx = xs.mean()
    cy = ys.mean()
    dx = xs - cx
    dy = ys - cy
    mu20 = float((dx * dx).mean())
    mu02 = float((dy * dy).mean())
    mu11 = float((dx * dy).mean())
    angle = 0.5 * math.atan2(2.0 * mu11, mu20 - mu02)
    return math.degrees(angle) % 180.0


def _fold(delta: float) -> float:
    delta = abs(delta) % 180.0
    return min(delta, 180.0 - delta)


def decision_from_image(image: Image.Image) -> str:
    mask = _dark_mask(image)
    height, width = mask.shape
    groups: dict = {panel: [] for panel in PANELS}
    for pixels in _components(mask):
        cy = float(pixels[:, 0].mean())
        cx = float(pixels[:, 1].mean())
        row = "top" if cy < height / 2 else "bottom"
        col = "left" if cx < width / 2 else "right"
        groups[f"{row}_{col}"].append(_axis_direction(pixels))
    scored = []
    for panel in PANELS:
        angles = groups[panel]
        if len(angles) < 2:
            return AMBIGUOUS
        gaps = [
            _fold(angles[i] - angles[j])
            for i in range(len(angles))
            for j in range(i + 1, len(angles))
        ]
        scored.append((min(gaps), panel))
    scored.sort()
    if len(scored) < 2 or scored[1][0] - scored[0][0] < 1.0:
        return AMBIGUOUS
    return scored[0][1]
