"""Renderer and analytic arm for the phase-marked lattice-decimation world."""

from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image, ImageDraw

CANVAS = 480
GRID_N = 30
OUT_N = 15
CELL = 12
CYAN = (0, 174, 190)
MAGENTA = (220, 25, 150)
GRID = (218, 222, 226)
INKS = ((24, 27, 34), (31, 55, 105), (73, 35, 91), (35, 79, 71))
SYMBOLS = ("circle", "diamond", "plus", "X")
TOTAL_DARK_CELLS = 450
QUARANTINE_GAP = 4.0


def _templates() -> dict[str, np.ndarray]:
    yy, xx = np.mgrid[-7:8, -7:8]
    return {
        "circle": np.abs(np.sqrt(xx * xx + yy * yy) - 5.0) <= 1.0,
        "diamond": np.abs(np.abs(xx) + np.abs(yy) - 6) <= 1,
        "plus": ((np.abs(xx) <= 1) & (np.abs(yy) <= 6))
        | ((np.abs(yy) <= 1) & (np.abs(xx) <= 6)),
        "X": (np.abs(np.abs(xx) - np.abs(yy)) <= 1)
        & (np.abs(xx) <= 6)
        & (np.abs(yy) <= 6),
    }


def _reconstructed(scene: dict) -> np.ndarray:
    mask = _templates()[scene["symbol"]].copy()
    for row, col in scene["flip_cells"]:
        mask[int(row), int(col)] = ~mask[int(row), int(col)]
    return mask


def _scores(mask: np.ndarray) -> dict[str, int]:
    return {name: int(np.count_nonzero(mask != ref)) for name, ref in _templates().items()}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x5A17C3)
    symbol = SYMBOLS[int(seed) % len(SYMBOLS)]
    row_phase = rng.randrange(2)
    col_phase = rng.randrange(2)
    offset_x = rng.randrange(48, 69)
    offset_y = rng.randrange(48, 69)
    ink_index = rng.randrange(len(INKS))
    cells = [(r, c) for r in range(OUT_N) for c in range(OUT_N)]
    rng.shuffle(cells)
    flip_cells = [list(value) for value in cells[: (int(seed) // 4) % 7]]
    provisional = {
        "symbol": symbol,
        "row_phase": row_phase,
        "col_phase": col_phase,
        "offset_x": offset_x,
        "offset_y": offset_y,
        "ink_index": ink_index,
        "noise_cells": [],
        "flip_cells": flip_cells,
    }
    retained = _reconstructed(provisional)
    retained_positions = {
        (row_phase + 2 * r, col_phase + 2 * c)
        for r in range(OUT_N)
        for c in range(OUT_N)
    }
    discarded = [
        (r, c)
        for r in range(GRID_N)
        for c in range(GRID_N)
        if (r, c) not in retained_positions
    ]
    rng.shuffle(discarded)
    noise_count = TOTAL_DARK_CELLS - int(retained.sum())
    provisional["noise_cells"] = [list(value) for value in discarded[:noise_count]]
    assert analytic_gold(provisional) == symbol
    assert margin(provisional) > QUARANTINE_GAP
    return provisional


def _full_grid(scene: dict) -> np.ndarray:
    grid = np.zeros((GRID_N, GRID_N), dtype=bool)
    for row, col in scene["noise_cells"]:
        grid[int(row), int(col)] = True
    target = _reconstructed(scene)
    row_phase = int(scene["row_phase"])
    col_phase = int(scene["col_phase"])
    for row in range(OUT_N):
        for col in range(OUT_N):
            grid[row_phase + 2 * row, col_phase + 2 * col] = bool(target[row, col])
    return grid


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (CANVAS, CANVAS), (250, 249, 246))
    draw = ImageDraw.Draw(image)
    left = int(scene["offset_x"])
    top = int(scene["offset_y"])
    ink = INKS[int(scene["ink_index"])]
    grid = _full_grid(scene)
    for row in range(GRID_N):
        for col in range(GRID_N):
            if grid[row, col]:
                x0 = left + col * CELL + 1
                y0 = top + row * CELL + 1
                draw.rectangle((x0, y0, x0 + CELL - 2, y0 + CELL - 2), fill=ink)
    for index in range(GRID_N + 1):
        x = left + index * CELL
        y = top + index * CELL
        draw.line((x, top, x, top + GRID_N * CELL), fill=GRID, width=1)
        draw.line((left, y, left + GRID_N * CELL, y), fill=GRID, width=1)
    frame = (left - 4, top - 4, left + GRID_N * CELL + 4, top + GRID_N * CELL + 4)
    draw.rectangle(frame, outline=CYAN, width=3)
    col_center = left + int(scene["col_phase"]) * CELL + CELL // 2
    row_center = top + int(scene["row_phase"]) * CELL + CELL // 2
    draw.polygon(
        [(col_center - 5, top - 17), (col_center + 5, top - 17), (col_center, top - 7)],
        fill=MAGENTA,
    )
    draw.polygon(
        [(left - 17, row_center - 5), (left - 17, row_center + 5), (left - 7, row_center)],
        fill=MAGENTA,
    )
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def margin(scene: dict) -> float:
    scores = sorted(_scores(_reconstructed(scene)).values())
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
