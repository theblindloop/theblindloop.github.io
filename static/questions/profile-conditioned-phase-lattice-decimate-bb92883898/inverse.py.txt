"""Independent inverse arm. It receives only final raster pixels."""

from __future__ import annotations

import numpy as np
from PIL import Image

GRID_N = 30
OUT_N = 15


def _glyph_templates() -> dict[str, np.ndarray]:
    # Re-expressed independently as explicit point predicates on the public 15x15 result.
    result: dict[str, np.ndarray] = {}
    for name in ("circle", "diamond", "plus", "X"):
        mask = np.zeros((OUT_N, OUT_N), dtype=bool)
        for row in range(OUT_N):
            for col in range(OUT_N):
                x, y = col - 7, row - 7
                if name == "circle":
                    keep = 4.0 <= (x * x + y * y) ** 0.5 <= 6.0
                elif name == "diamond":
                    keep = 5 <= abs(x) + abs(y) <= 7
                elif name == "plus":
                    keep = (abs(x) <= 1 and abs(y) <= 6) or (
                        abs(y) <= 1 and abs(x) <= 6
                    )
                else:
                    keep = abs(abs(x) - abs(y)) <= 1 and abs(x) <= 6 and abs(y) <= 6
                mask[row, col] = keep
        result[name] = mask
    return result


def _recover(image: Image.Image) -> tuple[np.ndarray, float]:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    cyan = (rgb[:, :, 0] < 20) & (rgb[:, :, 1] > 145) & (rgb[:, :, 2] > 160)
    ys, xs = np.nonzero(cyan)
    if len(xs) < 100:
        raise ValueError("cyan frame is missing")
    min_x, max_x = int(xs.min()), int(xs.max())
    min_y, max_y = int(ys.min()), int(ys.max())
    if not (360 <= max_x - min_x <= 370 and 360 <= max_y - min_y <= 370):
        raise ValueError("cyan frame has unexpected geometry")
    left, top = min_x + 4, min_y + 4
    span_x, span_y = max_x - min_x - 8, max_y - min_y - 8
    cell_x, cell_y = span_x / GRID_N, span_y / GRID_N
    if not (11.8 <= cell_x <= 12.2 and 11.8 <= cell_y <= 12.2):
        raise ValueError("cell pitch cannot be inferred")

    magenta = (rgb[:, :, 0] > 180) & (rgb[:, :, 1] < 70) & (rgb[:, :, 2] > 110)
    top_band = magenta[max(0, top - 22) : top - 4, left : left + 30]
    side_band = magenta[top : top + 30, max(0, left - 22) : left - 4]
    ty, tx = np.nonzero(top_band)
    sy, sx = np.nonzero(side_band)
    if len(tx) < 15 or len(sy) < 15:
        raise ValueError("magenta sampling tabs are missing")
    tab_col = left + float(tx.mean())
    tab_row = top + float(sy.mean())
    col_phase = int(round((tab_col - left - cell_x / 2) / cell_x))
    row_phase = int(round((tab_row - top - cell_y / 2) / cell_y))
    if row_phase not in (0, 1) or col_phase not in (0, 1):
        raise ValueError("sampling phase is ambiguous")

    sampled = np.zeros((OUT_N, OUT_N), dtype=bool)
    for row in range(OUT_N):
        for col in range(OUT_N):
            source_row = row_phase + 2 * row
            source_col = col_phase + 2 * col
            px = int(round(left + (source_col + 0.5) * cell_x))
            py = int(round(top + (source_row + 0.5) * cell_y))
            patch = rgb[py - 2 : py + 3, px - 2 : px + 3]
            luminance = 0.299 * patch[:, :, 0] + 0.587 * patch[:, :, 1] + 0.114 * patch[:, :, 2]
            sampled[row, col] = float(np.median(luminance)) < 135.0
    distances = {
        name: int(np.count_nonzero(sampled != template))
        for name, template in _glyph_templates().items()
    }
    ranked = sorted(distances.items(), key=lambda item: (item[1], item[0]))
    gap = float(ranked[1][1] - ranked[0][1])
    if gap <= 4:
        raise ValueError(f"ambiguous reconstruction: runner-up gap {gap:g}")
    return sampled, gap


def decision_with_margin(image: Image.Image) -> tuple[str, float]:
    sampled, gap = _recover(image)
    distances = {
        name: int(np.count_nonzero(sampled != template))
        for name, template in _glyph_templates().items()
    }
    winner = min(distances, key=lambda name: (distances[name], name))
    return winner, gap


def decision_from_image(image: Image.Image) -> str:
    return decision_with_margin(image)[0]
