"""Renderer, sampler and analytic gold for carry_counter_replay.

Four bounded counters stand side by side.  Each counter is a column of four
square cells that fill from the bottom.  A left-to-right row of step cards each
names one counter (by that counter's own header colour) and a number of steps
(by its black dots).  One step shades one more cell of that counter; a step
applied to a counter whose four cells are already shaded instead clears that
counter to zero shaded cells and sends one step to the counter immediately on
its left, which obeys the same rule, so an overflow can cascade.  A step carried
out of the leftmost counter is discarded.

The answer is how many cells the magenta-ringed counter shows once every card
has been applied in drawn left-to-right order.  No drawn configuration shows the
final state: only the starting state and the cards are drawn.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "carry_counter_replay-renderer@1.0.0"

N_COUNTERS = 4
N_CELLS = 4  # a counter shows 0, 1, 2, 3 or 4 shaded cells

W, H = 620, 360

WHITE = (255, 255, 255)
NAVY = (38, 58, 120)  # a shaded counter cell; used nowhere else
CELL_EDGE = (170, 170, 178)
BLACK = (0, 0, 0)
MAGENTA = (200, 0, 190)  # the query ring; used nowhere else
GREY = (140, 140, 140)
CARD_EDGE = (60, 60, 60)

COLOR_NAMES = ("red", "blue", "green", "orange")
COLOR_RGB = {
    "red": (214, 44, 44),
    "blue": (36, 110, 214),
    "green": (30, 150, 70),
    "orange": (230, 140, 20),
}

CW, CH = 46, 24  # counter width, cell height
COL_GAP = 34
HEADER_H = 16
CARD_W, CARD_H = 64, 52
DOT_D = 8

_GAP_CHOICES = (4, 5, 5, 6, 6, 7, 7, 8, 8, 9, 9, 10, 10)
QUARANTINE_MARGIN = 5.0


# ---------------------------------------------------------------- replay ----
def _step(fills: list[int], index: int, carries: list[int]) -> None:
    """One step on `index`, with cascading carry to the left on overflow."""
    if fills[index] < N_CELLS:
        fills[index] += 1
        return
    fills[index] = 0
    if index > 0:
        carries.append(index - 1)
        _step(fills, index - 1, carries)


def _replay(scene: dict[str, Any]) -> tuple[list[int], list[int]]:
    fills = _starts(scene)
    carries: list[int] = []
    for card in _ordered_cards(scene):
        for _ in range(int(card["steps"])):
            _step(fills, int(card["target"]), carries)
    return fills, carries


def _capped_replay(scene: dict[str, Any]) -> list[int]:
    """The same cards replayed with overflow ignored instead of carried."""
    fills = _starts(scene)
    for card in _ordered_cards(scene):
        for _ in range(int(card["steps"])):
            i = int(card["target"])
            if fills[i] < N_CELLS:
                fills[i] += 1
    return fills


def _starts(scene: dict[str, Any]) -> list[int]:
    fills = [0] * N_COUNTERS
    for counter in scene["counters"]:
        fills[int(counter["index"])] = int(counter["start"])
    return fills


def _ordered_cards(scene: dict[str, Any]) -> list[dict[str, Any]]:
    return sorted(scene["cards"], key=lambda c: int(c["slot"]))


# --------------------------------------------------------------- sampling ---
def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(seed * 7919 + 13)

    target_answer = seed % (N_CELLS + 1)
    query = (seed // 5) % N_COUNTERS
    block_capped = ((seed // 20) % 2) == 0
    require_change = ((seed // 40) % 5) != 0

    # Constraint levels, strongest first: the answer class and the presence of at
    # least one carry always hold; the two shortcut blocks are relaxed in a fixed
    # order if the requested answer class cannot satisfy them for this query column.
    counters = None
    cards = None
    for level in (0, 1, 2):
        want_change = require_change and level == 0
        want_capped = block_capped and level <= 1
        for _ in range(4000):
            starts = [rng.randrange(0, N_CELLS + 1) for _ in range(N_COUNTERS)]
            n_cards = rng.randint(4, 6)
            cand = [
                {"slot": s, "target": rng.randrange(N_COUNTERS), "steps": rng.randint(1, 3)}
                for s in range(n_cards)
            ]
            probe = {
                "counters": [{"index": i, "start": v} for i, v in enumerate(starts)],
                "cards": cand,
            }
            fills, carries = _replay(probe)
            if fills[query] != target_answer or not carries:
                continue
            if want_change and fills[query] == starts[query]:
                continue
            if want_capped and _capped_replay(probe)[query] == fills[query]:
                continue
            counters = [{"index": i, "start": v} for i, v in enumerate(starts)]
            cards = cand
            break
        if counters is not None:
            break
    if counters is None or cards is None:  # pragma: no cover - constraints are feasible
        raise RuntimeError(f"could not sample a scene for seed {seed}")

    names = list(COLOR_NAMES)
    rng.shuffle(names)
    palette = {str(i): names[i] for i in range(N_COUNTERS)}

    gaps = {
        "cell": rng.choice(_GAP_CHOICES),
        "dot": rng.choice(_GAP_CHOICES),
        "card": rng.randint(12, 18),
    }
    canvas = {"w": W, "h": H, "ox": 34 + rng.randint(0, 12), "oy": 20 + rng.randint(0, 10)}

    # store the two structures in an order that carries no information
    rng.shuffle(counters)
    shuffled_cards = list(cards)
    rng.shuffle(shuffled_cards)

    return {
        "counters": counters,
        "cards": shuffled_cards,
        "query": query,
        "palette": palette,
        "gaps": gaps,
        "canvas": canvas,
    }


# ------------------------------------------------------------------ gold ----
def analytic_gold(scene: dict[str, Any]) -> str:
    fills, _ = _replay(scene)
    return str(fills[int(scene["query"])])


def margin(scene: dict[str, Any]) -> float:
    g = scene["gaps"]
    return float(min(int(g["cell"]), int(g["dot"]), int(g["card"])))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """Storage-order permutations that must be pixel-identical and gold-preserving."""
    out: list[tuple[str, dict[str, Any]]] = []

    rev_counters = dict(scene)
    rev_counters["counters"] = list(reversed(scene["counters"]))
    out.append(("reverse_counter_bag", rev_counters))

    rev_cards = dict(scene)
    rev_cards["cards"] = list(reversed(scene["cards"]))
    out.append(("reverse_card_list", rev_cards))

    rot_cards = dict(scene)
    cards = list(scene["cards"])
    rot_cards["cards"] = cards[1:] + cards[:1]
    out.append(("rotate_card_list", rot_cards))

    return out


# ---------------------------------------------------------------- render ----
def _layout(scene: dict[str, Any]) -> dict[str, Any]:
    ox = int(scene["canvas"]["ox"])
    oy = int(scene["canvas"]["oy"])
    g_cell = int(scene["gaps"]["cell"])
    cells_top = oy + HEADER_H + 10
    block_h = N_CELLS * CH + (N_CELLS - 1) * g_cell
    base_y = cells_top + block_h
    ring_cy = base_y + 22
    cards_top = ring_cy + 20
    return {
        "ox": ox,
        "oy": oy,
        "g_cell": g_cell,
        "cells_top": cells_top,
        "base_y": base_y,
        "ring_cy": ring_cy,
        "cards_top": cards_top,
    }


def _col_x(ox: int, i: int) -> int:
    return ox + i * (CW + COL_GAP)


def render(scene: dict[str, Any]) -> Image.Image:
    lay = _layout(scene)
    ox, oy = lay["ox"], lay["oy"]
    g_cell = lay["g_cell"]
    g_dot = int(scene["gaps"]["dot"])
    g_card = int(scene["gaps"]["card"])
    starts = _starts(scene)
    palette = scene["palette"]

    img = Image.new("RGB", (W, H), WHITE)
    d = ImageDraw.Draw(img)

    for i in range(N_COUNTERS):
        x0 = _col_x(ox, i)
        x1 = x0 + CW
        d.rectangle([x0, oy, x1, oy + HEADER_H], fill=COLOR_RGB[palette[str(i)]])
        for j in range(N_CELLS):  # j = 0 is the bottom cell
            top = lay["cells_top"] + (N_CELLS - 1 - j) * (CH + g_cell)
            box = [x0, top, x1, top + CH]
            if j < starts[i]:
                d.rectangle(box, fill=NAVY, outline=NAVY)
            else:
                d.rectangle(box, fill=WHITE, outline=CELL_EDGE)

    q = int(scene["query"])
    cx = _col_x(ox, q) + CW // 2
    cy = lay["ring_cy"]
    d.ellipse([cx - 11, cy - 11, cx + 11, cy + 11], outline=MAGENTA, width=3)

    cards = _ordered_cards(scene)
    cy_top = lay["cards_top"]
    d.line([ox, cy_top - 12, ox + len(cards) * (CARD_W + g_card) - g_card, cy_top - 12],
           fill=GREY, width=2)
    tip = ox + len(cards) * (CARD_W + g_card) - g_card
    d.polygon([(tip, cy_top - 18), (tip, cy_top - 6), (tip + 12, cy_top - 12)], fill=GREY)

    for k, card in enumerate(cards):
        x0 = ox + k * (CARD_W + g_card)
        d.rectangle([x0, cy_top, x0 + CARD_W, cy_top + CARD_H], fill=WHITE, outline=CARD_EDGE)
        sw = 18
        sx = x0 + (CARD_W - sw) // 2
        d.rectangle([sx, cy_top + 6, sx + sw, cy_top + 6 + sw],
                    fill=COLOR_RGB[palette[str(int(card["target"]))]])
        n = int(card["steps"])
        span = n * DOT_D + (n - 1) * g_dot
        dx = x0 + (CARD_W - span) // 2
        dy = cy_top + CARD_H - 8 - DOT_D
        for t in range(n):
            left = dx + t * (DOT_D + g_dot)
            d.ellipse([left, dy, left + DOT_D - 1, dy + DOT_D - 1], fill=BLACK)

    return img
