"""Pixel-only inverse arm for disconnected shared-boundary multiplicity."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

COLORS = {
    "red": np.array((217, 52, 58)),
    "green": np.array((22, 139, 82)),
    "blue": np.array((39, 103, 199)),
}


def _components(mask: np.ndarray, diagonal: bool = False) -> tuple[np.ndarray, list[int]]:
    height, width = mask.shape
    labels = np.full((height, width), -1, dtype=np.int32)
    sizes: list[int] = []
    neighbors = ((-1, 0), (1, 0), (0, -1), (0, 1))
    if diagonal:
        neighbors += ((-1, -1), (-1, 1), (1, -1), (1, 1))
    for y in range(height):
        for x in range(width):
            if not mask[y, x] or labels[y, x] >= 0:
                continue
            label = len(sizes)
            labels[y, x] = label
            queue = deque([(y, x)])
            size = 0
            while queue:
                yy, xx = queue.popleft()
                size += 1
                for dy, dx in neighbors:
                    y2, x2 = yy + dy, xx + dx
                    if 0 <= y2 < height and 0 <= x2 < width and mask[y2, x2] and labels[y2, x2] < 0:
                        labels[y2, x2] = label
                        queue.append((y2, x2))
            sizes.append(size)
    return labels, sizes


def _dilate(mask: np.ndarray, radius: int) -> np.ndarray:
    padded = np.pad(mask, radius)
    out = np.zeros_like(mask)
    height, width = mask.shape
    for dy in range(2 * radius + 1):
        for dx in range(2 * radius + 1):
            out |= padded[dy:dy + height, dx:dx + width]
    return out


def _marker_floor_labels(arr: np.ndarray, floor_labels: np.ndarray, rgb: np.ndarray) -> list[int]:
    distance = np.sqrt(np.sum((arr - rgb) ** 2, axis=2))
    color_mask = distance < 38
    dot_labels, sizes = _components(color_mask)
    found = []
    for label, size in enumerate(sizes):
        if size < 12:
            continue
        ys, xs = np.where(dot_labels == label)
        cy, cx = int(round(float(ys.mean()))), int(round(float(xs.mean())))
        floor = int(floor_labels[cy, cx])
        if floor >= 0:
            found.append(floor)
    return found


def _shared_sections(black: np.ndarray, floor_labels: np.ndarray, a: int, b: int) -> int:
    near_a = _dilate(floor_labels == a, 3)
    near_b = _dilate(floor_labels == b, 3)
    contact = black & near_a & near_b
    _, sizes = _components(contact, diagonal=True)
    return sum(size >= 6 for size in sizes)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)[::4, ::4]
    black = np.max(arr, axis=2) < 70
    floor_labels, floor_sizes = _components(~black)
    if len(floor_sizes) < 5:
        return "abstain"
    outside = int(floor_labels[0, 0])

    counts = {}
    for color, rgb in COLORS.items():
        labels = _marker_floor_labels(arr, floor_labels, rgb)
        if len(labels) != 2 or labels[0] == labels[1] or outside in labels:
            return "abstain"
        counts[color] = _shared_sections(black, floor_labels, labels[0], labels[1])
    if sorted(counts.values()) != [1, 2, 3]:
        return "abstain"
    matches = [color for color, value in counts.items() if value == 2]
    return matches[0] if len(matches) == 1 else "abstain"
