"""Deterministic renderer and independent analytic arm for boundary multiplicity."""

from __future__ import annotations

import random
from collections import defaultdict, deque
from copy import deepcopy

from PIL import Image, ImageDraw

SIZE = 512
GRID_N = 8
GRID_LEFT = 48
CELL = 52
GRID_RIGHT = GRID_LEFT + GRID_N * CELL
WALL = 9
MARKER_RADIUS = 12
PALETTE = {"red": "#d9343a", "green": "#168b52", "blue": "#2767c7"}


def _base_layout() -> list[list[int]]:
    """Connected chambers with A-B contact multiplicities 2, 3, and 1."""
    layout = [[-1 for _ in range(GRID_N)] for _ in range(GRID_N)]
    # Common chamber A: a vertical spine.
    for r in range(1, 7):
        layout[r][3] = 0
    # B touches A in two separated rows and connects behind those contacts.
    for r in range(1, 6):
        layout[r][1] = 1
    layout[0][1] = 1
    layout[2][2] = 1
    layout[4][2] = 1
    # C touches A in three separated rows and connects by a parallel spine.
    for r in range(1, 6):
        layout[r][5] = 2
    for r in (1, 3, 5):
        layout[r][4] = 2
    # D has two wall edges meeting at one corner, hence one connected section.
    for c in range(4, 8):
        layout[6][c] = 3
    for c in range(3, 7):
        layout[7][c] = 3

    # Give each remaining connected patch its own chamber ID.
    next_id = 4
    for r in range(GRID_N):
        for c in range(GRID_N):
            if layout[r][c] != -1:
                continue
            layout[r][c] = next_id
            queue = deque([(r, c)])
            while queue:
                rr, cc = queue.popleft()
                for r2, c2 in ((rr - 1, cc), (rr + 1, cc), (rr, cc - 1), (rr, cc + 1)):
                    if 0 <= r2 < GRID_N and 0 <= c2 < GRID_N and layout[r2][c2] == -1:
                        layout[r2][c2] = next_id
                        queue.append((r2, c2))
            next_id += 1
    return layout


def _boundary_edges(layout: list[list[int]]) -> dict[tuple[int, int], set[tuple[int, int, str]]]:
    edges: dict[tuple[int, int], set[tuple[int, int, str]]] = defaultdict(set)
    for r in range(GRID_N):
        for c in range(GRID_N):
            here = layout[r][c]
            if c + 1 < GRID_N and layout[r][c + 1] != here:
                pair = tuple(sorted((here, layout[r][c + 1])))
                edges[pair].add((2 * r + 1, 2 * c + 2, "v"))
            if r + 1 < GRID_N and layout[r + 1][c] != here:
                pair = tuple(sorted((here, layout[r + 1][c])))
                edges[pair].add((2 * r + 2, 2 * c + 1, "h"))
    return edges


def _contact_components(layout: list[list[int]], a: int, b: int) -> int:
    edges = _boundary_edges(layout).get(tuple(sorted((a, b))), set())
    by_endpoint: dict[tuple[int, int], list[tuple[int, int, str]]] = defaultdict(list)
    for edge in edges:
        y, x, orientation = edge
        endpoints = ((y - 1, x), (y + 1, x)) if orientation == "v" else ((y, x - 1), (y, x + 1))
        for endpoint in endpoints:
            by_endpoint[endpoint].append(edge)
    seen: set[tuple[int, int, str]] = set()
    count = 0
    for start in edges:
        if start in seen:
            continue
        count += 1
        seen.add(start)
        queue = [start]
        while queue:
            y, x, orientation = queue.pop()
            endpoints = ((y - 1, x), (y + 1, x)) if orientation == "v" else ((y, x - 1), (y, x + 1))
            for endpoint in endpoints:
                for neighbor in by_endpoint[endpoint]:
                    if neighbor not in seen:
                        seen.add(neighbor)
                        queue.append(neighbor)
    return count


def _transform(layout: list[list[int]], turns: int, mirror: bool) -> list[list[int]]:
    result = deepcopy(layout)
    if mirror:
        result = [row[::-1] for row in result]
    for _ in range(turns % 4):
        result = [list(row) for row in zip(*result[::-1])]
    return result


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    layout = _transform(_base_layout(), rng.randrange(4), bool(rng.randrange(2)))
    # Relabel region IDs to prevent their numeric values becoming meaningful.
    ids = sorted({value for row in layout for value in row})
    shuffled = ids[:]
    rng.shuffle(shuffled)
    relabel = dict(zip(ids, shuffled))
    layout = [[relabel[value] for value in row] for row in layout]
    common, two_region, three_region, one_region = (relabel[i] for i in range(4))

    colors = list(PALETTE)
    target_color = colors[int(seed) % len(colors)]
    remaining = [color for color in colors if color != target_color]
    rng.shuffle(remaining)
    roles = {target_color: two_region, remaining[0]: one_region, remaining[1]: three_region}

    common_cells = [(r, c) for r in range(GRID_N) for c in range(GRID_N) if layout[r][c] == common]
    rng.shuffle(common_cells)
    markers: dict[str, list[list[int]]] = {}
    for index, color in enumerate(colors):
        partner = roles[color]
        partner_cells = [(r, c) for r in range(GRID_N) for c in range(GRID_N) if layout[r][c] == partner]
        markers[color] = [list(common_cells[index]), list(rng.choice(partner_cells))]
    return {"layout": layout, "markers": markers}


def render(scene: dict) -> Image.Image:
    layout = scene["layout"]
    image = Image.new("RGB", (SIZE, SIZE), "#e8edf2")
    draw = ImageDraw.Draw(image)
    draw.rectangle((GRID_LEFT, GRID_LEFT, GRID_RIGHT, GRID_RIGHT), fill="white")
    draw.rectangle((GRID_LEFT, GRID_LEFT, GRID_RIGHT, GRID_RIGHT), outline="#111111", width=WALL)
    half = WALL // 2
    for r in range(GRID_N):
        for c in range(1, GRID_N):
            if layout[r][c - 1] != layout[r][c]:
                x = GRID_LEFT + c * CELL
                y0 = GRID_LEFT + r * CELL
                draw.rectangle((x - half, y0 - half, x + half, y0 + CELL + half), fill="#111111")
    for r in range(1, GRID_N):
        for c in range(GRID_N):
            if layout[r - 1][c] != layout[r][c]:
                y = GRID_LEFT + r * CELL
                x0 = GRID_LEFT + c * CELL
                draw.rectangle((x0 - half, y - half, x0 + CELL + half, y + half), fill="#111111")
    for color, cells in scene["markers"].items():
        for r, c in cells:
            x = GRID_LEFT + c * CELL + CELL // 2
            y = GRID_LEFT + r * CELL + CELL // 2
            draw.ellipse(
                (x - MARKER_RADIUS, y - MARKER_RADIUS, x + MARKER_RADIUS, y + MARKER_RADIUS),
                fill=PALETTE[color], outline="white", width=2,
            )
    return image


def _marker_regions(scene: dict, color: str) -> tuple[int, int]:
    layout = scene["layout"]
    return tuple(layout[r][c] for r, c in scene["markers"][color])


def analytic_gold(scene: dict) -> str:
    counts = {}
    for color in PALETTE:
        a, b = _marker_regions(scene, color)
        if a == b:
            return "abstain"
        counts[color] = _contact_components(scene["layout"], a, b)
    matches = [color for color, value in counts.items() if value == 2]
    return matches[0] if len(matches) == 1 else "abstain"


def margin(scene: dict) -> float:
    gaps = []
    for color in PALETTE:
        a, b = _marker_regions(scene, color)
        value = _contact_components(scene["layout"], a, b)
        if value != 2:
            gaps.append(abs(value - 2))
    return float(min(gaps)) if gaps else 0.0


def is_quarantined(scene: dict) -> bool:
    return analytic_gold(scene) == "abstain" or margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    transformed = deepcopy(scene)
    ids = sorted({value for row in scene["layout"] for value in row})
    mapping = {value: ids[(index + 1) % len(ids)] for index, value in enumerate(ids)}
    transformed["layout"] = [[mapping[value] for value in row] for row in scene["layout"]]
    return [("bijective_region_relabel", transformed)]
