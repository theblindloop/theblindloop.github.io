"""Deterministic renderer for the connected snowflake-arm world."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

ARM_COUNTS = (5, 6, 7, 8)
CANVAS_SIZE = 512
TURN_STEPS = 840
SUPERSAMPLE = 3
PROFILE_POINTS = 2520

PALETTES = (
    ("#246BCE", "#F7F3E8"),
    ("#137D76", "#F6F1E7"),
    ("#6B57B8", "#F8F3E9"),
    ("#35556F", "#FAF4E8"),
)


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene from an integer seed.

    Consecutive seeds cycle through all four answers.  Palette and scale cycle
    on slower, orthogonal schedules, while placement and rotation use a local
    PRNG.  Thus canonical six-arm and counterfactual variants share styling.
    """

    seed = int(seed)
    rng = random.Random(seed ^ 0x5F10A6E)
    foreground, background = PALETTES[(seed // 4) % len(PALETTES)]
    return {
        "arm_count": ARM_COUNTS[seed % len(ARM_COUNTS)],
        "background": background,
        "canvas_size": CANVAS_SIZE,
        "center_x": 256 + rng.randint(-10, 10),
        "center_y": 256 + rng.randint(-10, 10),
        "foreground": foreground,
        "rotation_step": rng.randrange(TURN_STEPS),
        "scale_percent": (92, 96, 100)[(seed // 16) % 3],
    }


def _radius_at(scene: dict, theta: float) -> float:
    """Radial silhouette with one long tip and two short tips per sector."""

    arms = int(scene["arm_count"])
    rotation = 2.0 * math.pi * (int(scene["rotation_step"]) % TURN_STEPS) / TURN_STEPS
    phase = (((theta - rotation) * arms / (2.0 * math.pi) + 0.5) % 1.0) - 0.5
    primary = max(0.0, 1.0 - abs(phase) / 0.16)
    side = max(0.0, 1.0 - abs(abs(phase) - 0.29) / 0.085)
    scale = int(scene["scale_percent"]) / 100.0
    # The base disk makes the evidence one thick connected, area-like region.
    # Because every arm count repeats the same unit-sector profile, continuous
    # silhouette area is independent of arm count at a fixed scale.
    return scale * (168.0 + 52.0 * primary + 26.0 * side)


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas_size"])
    ss = SUPERSAMPLE
    image = Image.new("RGB", (size * ss, size * ss), str(scene["background"]))
    draw = ImageDraw.Draw(image)
    cx = float(scene["center_x"]) * ss
    cy = float(scene["center_y"]) * ss
    points = []
    for index in range(PROFILE_POINTS):
        theta = 2.0 * math.pi * index / PROFILE_POINTS
        radius = _radius_at(scene, theta) * ss
        points.append(
            (
                round(cx + radius * math.cos(theta)),
                round(cy + radius * math.sin(theta)),
            )
        )
    draw.polygon(points, fill=str(scene["foreground"]))
    return image.resize((size, size), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(int(scene["arm_count"]))


def margin(scene: dict) -> float:
    """Primary-tip radial prominence over the shorter side-tip prominence."""

    return 26.0 * int(scene["scale_percent"]) / 100.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["rotation_step"] = int(twin["rotation_step"]) + TURN_STEPS
    return [("one_full_turn", twin)]
