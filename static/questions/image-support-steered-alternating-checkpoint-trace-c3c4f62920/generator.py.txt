"""Deterministic latent sampler and raster renderer for a global trace task."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 640
HEIGHT = 480
CHECKPOINT_RED = (210, 55, 65)
CHECKPOINT_GREEN = (35, 158, 92)

_ROUTE_PALETTES = (
    (23, 43, 70),
    (31, 48, 76),
    (39, 42, 68),
)
_BACKGROUNDS = ((248, 247, 242), (247, 249, 246), (250, 247, 245))
_STARTS = ((35, 105, 210), (43, 113, 218), (31, 98, 201))
_FINISHES = ((238, 132, 42), (232, 122, 35), (242, 143, 50))


def _transform_points(points: list[tuple[int, int]], mode: int) -> list[list[int]]:
    transformed = []
    for x, y in points:
        if mode & 1:
            x = WIDTH - x
        if mode & 2:
            y = HEIGHT - y
        transformed.append([int(x), int(y)])
    return transformed


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; consecutive integer seeds alternate answers."""

    rng = random.Random(int(seed))
    top = 56 + rng.randint(-7, 7)
    lower = 424 + rng.randint(-7, 7)
    base = [
        (46 + rng.randint(-4, 5), top),
        (300 + rng.randint(-12, 12), 145 + rng.randint(-10, 10)),
        (190 + rng.randint(-10, 10), 260 + rng.randint(-10, 10)),
        (470 + rng.randint(-12, 12), 200 + rng.randint(-10, 10)),
        (350 + rng.randint(-10, 10), 360 + rng.randint(-10, 10)),
        (594 + rng.randint(-5, 4), lower),
    ]
    points = _transform_points(base, rng.randrange(4))
    if rng.randrange(2):
        points.reverse()

    # Place checkpoints in segment interiors.  The long central segment carries
    # two, while every other segment carries one, so no marker can occlude an
    # acute corner and all six remain spread along the complete trace.
    metric_points = [(float(x), float(y)) for x, y in points]
    segment_lengths = [
        math.hypot(b[0] - a[0], b[1] - a[1])
        for a, b in zip(metric_points, metric_points[1:])
    ]
    total_length = sum(segment_lengths)
    schedule = [(0, 0.45), (1, 0.50), (2, 0.28), (2, 0.74), (3, 0.52), (4, 0.55)]
    consumed = [0.0]
    for length in segment_lengths:
        consumed.append(consumed[-1] + length)
    fractions = []
    for segment, local in schedule:
        local += rng.uniform(-0.025, 0.025)
        fractions.append(round((consumed[segment] + local * segment_lengths[segment]) / total_length, 4))

    phase = rng.randrange(2)
    colors = ["red" if (index + phase) % 2 == 0 else "green" for index in range(6)]
    if int(seed) % 2:
        swap_at = rng.randrange(5)
        colors[swap_at], colors[swap_at + 1] = colors[swap_at + 1], colors[swap_at]

    palette_index = rng.randrange(len(_ROUTE_PALETTES))
    return {
        "route_points": points,
        "checkpoint_indices": fractions,
        "checkpoint_colors": colors,
        "route_color": list(_ROUTE_PALETTES[palette_index]),
        "background_color": list(_BACKGROUNDS[palette_index]),
        "start_color": list(_STARTS[palette_index]),
        "finish_color": list(_FINISHES[palette_index]),
    }


def _polyline_lengths(points: Iterable[Iterable[float]]) -> tuple[list[tuple[float, float]], list[float], float]:
    pts = [(float(point[0]), float(point[1])) for point in points]
    lengths = [math.hypot(b[0] - a[0], b[1] - a[1]) for a, b in zip(pts, pts[1:])]
    return pts, lengths, sum(lengths)


def _station(points: list[tuple[float, float]], lengths: list[float], total: float, fraction: float) -> tuple[float, float, float, float]:
    target = max(0.0, min(1.0, float(fraction))) * total
    consumed = 0.0
    for (x0, y0), (x1, y1), length in zip(points, points[1:], lengths):
        if target <= consumed + length:
            local = (target - consumed) / length
            x = x0 + local * (x1 - x0)
            y = y0 + local * (y1 - y0)
            return x, y, (x1 - x0) / length, (y1 - y0) / length
        consumed += length
    x0, y0 = points[-2]
    x1, y1 = points[-1]
    length = lengths[-1]
    return x1, y1, (x1 - x0) / length, (y1 - y0) / length


def render(scene: dict) -> Image.Image:
    """Draw a thin, continuous route and six route-adjacent colored checkpoints."""

    route_points = scene["route_points"]
    checkpoint_indices = scene["checkpoint_indices"]
    checkpoint_colors = scene["checkpoint_colors"]
    route_color = tuple(scene["route_color"])
    background = tuple(scene["background_color"])
    start_color = tuple(scene["start_color"])
    finish_color = tuple(scene["finish_color"])

    image = Image.new("RGB", (WIDTH, HEIGHT), background)
    draw = ImageDraw.Draw(image)
    points = [(int(point[0]), int(point[1])) for point in route_points]
    draw.line(points, fill=route_color, width=9, joint="curve")

    metric_points, lengths, total = _polyline_lengths(route_points)
    for index, (fraction, name) in enumerate(zip(checkpoint_indices, checkpoint_colors)):
        x, y, tx, ty = _station(metric_points, lengths, total, fraction)
        side = 1.0 if index % 2 == 0 else -1.0
        # Keep the colored disk close enough to be an unambiguous route marker
        # while preserving the dark centreline underneath every checkpoint.
        cx = x + side * (-ty) * 14.0
        cy = y + side * tx * 14.0
        radius = 8
        color = CHECKPOINT_RED if name == "red" else CHECKPOINT_GREEN
        draw.ellipse(
            (round(cx - radius), round(cy - radius), round(cx + radius), round(cy + radius)),
            fill=color,
        )

    for center, color in ((points[0], start_color), (points[-1], finish_color)):
        x, y = center
        draw.ellipse((x - 16, y - 16, x + 16, y + 16), outline=color, width=5)
    return image


def analytic_gold(scene: dict) -> str:
    colors = scene["checkpoint_colors"]
    return "yes" if all(left != right for left, right in zip(colors, colors[1:])) else "no"


def margin(scene: dict) -> float:
    """Runner-up margin in adjacency-vote units around the 4.5/5 boundary."""

    colors = scene["checkpoint_colors"]
    alternating_edges = sum(left != right for left, right in zip(colors, colors[1:]))
    return abs(float(alternating_edges) - 4.5)


def is_quarantined(scene: dict) -> bool:
    fractions = [float(value) for value in scene["checkpoint_indices"]]
    valid_colors = all(value in {"red", "green"} for value in scene["checkpoint_colors"])
    return not (
        len(scene["route_points"]) == 6
        and len(fractions) == 6
        and len(scene["checkpoint_colors"]) == 6
        and valid_colors
        and all(b - a >= 0.08 for a, b in zip(fractions, fractions[1:]))
        and 0.06 <= fractions[0]
        and fractions[-1] <= 0.94
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
