"""Independent pixel-only inverse for crossbuck board alignment."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


def _dominant_fill_masks(image: Image.Image) -> list[np.ndarray]:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int32)
    spread = rgb.max(axis=2) - rgb.min(axis=2)
    eligible = (spread > 65) & (rgb.max(axis=2) > 105)
    pixels = rgb[eligible]
    if len(pixels) < 1500:
        raise ValueError("insufficient colored-board evidence")
    quantized = ((pixels + 4) // 8) * 8
    colors, counts = np.unique(quantized, axis=0, return_counts=True)
    order = np.argsort(counts)[::-1]
    centers: list[np.ndarray] = []
    for index in order:
        center = colors[index]
        if counts[index] < 350:
            break
        if all(float(np.linalg.norm(center - prior)) > 70.0 for prior in centers):
            centers.append(center)
        if len(centers) == 2:
            break
    if len(centers) != 2:
        raise ValueError("could not isolate two board colors")
    masks = []
    for center in centers:
        distance = np.sqrt(((rgb - center) ** 2).sum(axis=2))
        mask = eligible & (distance < 42.0)
        if int(mask.sum()) < 900:
            raise ValueError("colored board is too small or occluded")
        masks.append(mask)
    return masks


def _axis_angle(mask: np.ndarray) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    points = np.column_stack((xs, ys)).astype(float)
    points -= points.mean(axis=0)
    covariance = points.T @ points / len(points)
    values, vectors = np.linalg.eigh(covariance)
    major = vectors[:, int(np.argmax(values))]
    angle = math.degrees(math.atan2(major[1], major[0])) % 180.0
    elongation = float(values.max() / max(values.min(), 1e-6))
    if elongation < 8.0:
        raise ValueError("board axis is not sufficiently elongated")
    return angle, elongation


def _distance_to_family(angle: float, family: tuple[float, float]) -> float:
    return min(abs(((angle - target + 90.0) % 180.0) - 90.0) for target in family)


def decision_from_image(image: Image.Image) -> str:
    angles = [_axis_angle(mask)[0] for mask in _dominant_fill_masks(image)]
    axis_score = sum(_distance_to_family(angle, (0.0, 90.0)) for angle in angles)
    diagonal_score = sum(_distance_to_family(angle, (45.0, 135.0)) for angle in angles)
    runner_up_margin = abs(axis_score - diagonal_score)
    separation = abs(((angles[0] - angles[1] + 90.0) % 180.0) - 90.0)
    separation = min(separation, 180.0 - separation)
    if runner_up_margin < 30.0 or abs(separation - 90.0) > 14.0:
        raise ValueError("ambiguous board-axis evidence")
    return "plus" if axis_score < diagonal_score else "X"
