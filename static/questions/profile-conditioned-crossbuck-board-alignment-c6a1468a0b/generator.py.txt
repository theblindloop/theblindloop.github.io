"""Deterministic crossbuck renderer and analytic arm."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 480
HEIGHT = 480
SCALE = 3
PALETTES = (
    ((34, 132, 205), (238, 126, 47)),
    ((26, 156, 143), (221, 72, 126)),
    ((94, 92, 196), (231, 155, 36)),
    ((28, 146, 190), (215, 75, 75)),
)
BACKGROUNDS = ((238, 235, 225), (230, 236, 239), (241, 232, 236), (232, 239, 228))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    alignment = "X" if int(seed) % 2 == 0 else "plus"
    length = rng.randrange(268, 321, 4)
    width = rng.randrange(42, 55, 2)
    palette = rng.randrange(len(PALETTES))
    colors = PALETTES[palette]
    if rng.randrange(2):
        colors = (colors[1], colors[0])
    return {
        "alignment": alignment,
        "center_x": rng.randint(220, 260),
        "center_y": rng.randint(185, 220),
        "board_length": length,
        "board_width": width,
        "board_color": [list(colors[0]), list(colors[1])],
        "post_height": rng.randint(178, 215),
        "background_color": list(BACKGROUNDS[rng.randrange(len(BACKGROUNDS))]),
        "board_order": rng.randrange(2),
        "label_order": rng.randrange(2),
        "bolt_offset": rng.randint(14, 24),
    }


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    path = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
    try:
        return ImageFont.truetype(path, size)
    except OSError:
        return ImageFont.load_default()


def _board_tile(length: int, width: int, color: tuple[int, int, int], label: str, bolt_offset: int) -> Image.Image:
    pad = 8 * SCALE
    tile = Image.new("RGBA", (length * SCALE + 2 * pad, width * SCALE + 2 * pad), (0, 0, 0, 0))
    draw = ImageDraw.Draw(tile)
    box = (pad, pad, pad + length * SCALE - 1, pad + width * SCALE - 1)
    draw.rounded_rectangle(box, radius=5 * SCALE, fill=color + (255,), outline=(45, 48, 50, 255), width=3 * SCALE)
    font = _font(20 * SCALE)
    bounds = draw.textbbox((0, 0), label, font=font)
    tw, th = bounds[2] - bounds[0], bounds[3] - bounds[1]
    tx = tile.width // 2 - tw // 2
    ty = tile.height // 2 - th // 2 - bounds[1]
    draw.text((tx, ty), label, font=font, fill=(35, 37, 40, 255))
    cy = tile.height // 2
    for x in (pad + bolt_offset * SCALE, pad + (length - bolt_offset) * SCALE):
        r = 4 * SCALE
        draw.ellipse((x-r, cy-r, x+r, cy+r), fill=(242, 235, 196, 255), outline=(48, 50, 52, 255), width=SCALE)
    return tile


def _paste_board(image: Image.Image, scene: dict, board_index: int, angle: float, label: str) -> None:
    colors = scene["board_color"]
    color = tuple(int(v) for v in colors[board_index])
    tile = _board_tile(int(scene["board_length"]), int(scene["board_width"]), color, label, int(scene["bolt_offset"]))
    rotated = tile.rotate(angle, resample=Image.Resampling.BICUBIC, expand=True)
    cx = int(scene["center_x"]) * SCALE
    cy = int(scene["center_y"]) * SCALE
    image.alpha_composite(rotated, (cx - rotated.width // 2, cy - rotated.height // 2))


def render(scene: dict) -> Image.Image:
    bg = tuple(int(v) for v in scene["background_color"])
    image = Image.new("RGBA", (WIDTH * SCALE, HEIGHT * SCALE), bg + (255,))
    draw = ImageDraw.Draw(image)
    cx = int(scene["center_x"]) * SCALE
    cy = int(scene["center_y"]) * SCALE
    ground = (int(scene["center_y"]) + int(scene["post_height"])) * SCALE
    # Neutral support and track marks make the object recognizable but are excluded by saturation.
    post_w = 24 * SCALE
    draw.rounded_rectangle((cx-post_w//2, cy, cx+post_w//2, ground), radius=5*SCALE,
                           fill=(104, 108, 111, 255), outline=(49, 52, 54, 255), width=3*SCALE)
    draw.line((45*SCALE, ground, 435*SCALE, ground), fill=(72, 75, 77, 255), width=5*SCALE)
    draw.line((65*SCALE, ground+28*SCALE, 415*SCALE, ground+28*SCALE), fill=(72, 75, 77, 255), width=5*SCALE)
    for x in range(85, 416, 55):
        draw.line((x*SCALE, ground-7*SCALE, (x-8)*SCALE, ground+36*SCALE), fill=(118, 106, 88, 255), width=7*SCALE)

    angles = (45.0, -45.0) if scene["alignment"] == "X" else (0.0, 90.0)
    labels = ("RAIL", "ROAD") if int(scene["label_order"]) == 0 else ("ROAD", "RAIL")
    order = (0, 1) if int(scene["board_order"]) == 0 else (1, 0)
    for index in order:
        _paste_board(image, scene, index, angles[index], labels[index])

    final = image.convert("RGB").resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
    return final


def analytic_gold(scene: dict) -> str:
    alignment = scene["alignment"]
    if alignment not in {"X", "plus"}:
        raise ValueError("invalid alignment")
    return alignment


def margin(scene: dict) -> float:
    # The closest runner-up family is 45 degrees away for either exact configuration.
    return 45.0


def is_quarantined(scene: dict) -> bool:
    return int(scene["board_length"]) < 250 or int(scene["board_width"]) < 36


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
