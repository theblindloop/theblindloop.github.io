"""Independent image-only inverse arm for column_bend_rectify_reveal.

Receives only the final PNG. It locates the magenta fiducial frame, levels the
periodic column bend with the declared (public-contract) constants, and reads
whether the surviving solid glyph has its mass above or below the panel centre,
yielding the answer A (upward triangle, mass low) or B (downward triangle,
mass high).
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

_BG = (240, 240, 240)
_FRAME = (255, 0, 255)
_AMP = 6
_PERIOD = 32
_PANEL = 112


def _find_panel(mask: np.ndarray) -> tuple[int, int, int, int]:
    """Return x0, y0, w, h bounding box of frame pixels, or None-ish (-1,...)."""
    ys, xs = np.where(mask)
    if ys.size == 0:
        return (-1, -1, 0, 0)
    return int(xs.min()), int(ys.min()), int(xs.max() - xs.min() + 1), int(ys.max() - ys.min() + 1)


def _straighten(panel: np.ndarray, bg: tuple) -> np.ndarray:
    out = panel.copy().astype(np.float64)
    h, w = panel.shape[:2]
    shifts = np.round(_AMP * np.sin(2.0 * math.pi * np.arange(w) / _PERIOD)).astype(np.int32)
    for j in range(w):
        s = int(shifts[j])
        if s == 0:
            continue
        col = out[:, j].copy()
        if s > 0:
            col[0:h - s] = col[s:h]
            col[h - s:h] = bg
        else:
            col[-s:h] = col[0:h + s]
            col[0:-s] = bg
        out[:, j] = col
    return out


def _content_mask(panel: np.ndarray, bg: tuple) -> np.ndarray:
    return ~np.all(panel.astype(np.int64) == np.array(bg, dtype=np.int64), axis=-1)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    frame = np.all(arr == np.array(_FRAME, dtype=np.uint8), axis=-1)
    x0, y0, w, h = _find_panel(frame)
    if x0 < 0 or w < _PANEL or h < _PANEL:
        return ""
    x1 = x0 + min(w, _PANEL)
    y1 = y0 + min(h, _PANEL)
    panel = arr[y0:y1, x0:x1]
    straightened = _straighten(panel, _BG)
    mask = _content_mask(straightened, _BG)
    idx = np.argwhere(mask)
    if idx.size == 0:
        return ""
    centre = (panel.shape[0] - 1) / 2.0
    centroid = float(idx[:, 0].mean())
    return "A" if centroid > centre else "B"
