"""Column-bend rectification reveal world.

A coarse abstract glyph (an upward- or downward-pointing solid triangle) is
drawn in a flat, rectangular panel that is framed by a magenta fiducial border.
Every vertical strip (column) of the panel is then independently shifted
vertically by a periodic amount

    shift(x) = round(AMP * sin(2*pi * x / PERIOD))

so straight vertical strokes become a smooth vertical wave and the glyph is
tilted / bent out of reading. The declared transform is to undo that known
periodic column bend (rectify) -- shifting each strip back by -shift(x) -- and
read the level symbol that appears: an upward triangle (A) or a downward
triangle (B).
"""

from __future__ import annotations

import math
import random
from pathlib import Path

import numpy as np
from PIL import Image

RENDERER_VERSION = "column_bend_rectify_reveal-0.1.0"

CELL = 16
GRID = 5
BLOCK = CELL * GRID
PANEL = 112
AMP = 6
PERIOD = 32
MIN_MARGIN = 3.0

CANVAS = 512
BG_RGB = (240, 240, 240)
GLYPH_RGB = (20, 60, 200)
FRAME_RGB = (255, 0, 255)

# 5x5 glyph masks. "A" = upward-pointing triangle (apex up, mass low).
_M = 5
_UP = np.zeros((_M, _M), dtype=np.int32)
_DOWN = np.zeros((_M, _M), dtype=np.int32)
for _r in range(_M):
    _half = _r
    for _c in range(_M):
        if abs(_c - (_M // 2)) <= _half:
            _UP[_r, _c] = 1
    for _c in range(_M):
        if abs(_c - (_M // 2)) <= (_M - 1 - _r):
            _DOWN[_r, _c] = 1

GLYPHS = {"A": _UP, "B": _DOWN}


def _shift_values(count: int) -> np.ndarray:
    xs = np.arange(count, dtype=np.float64)
    return np.round(AMP * np.sin(2.0 * math.pi * xs / PERIOD)).astype(np.int32)


def _glyph_block(glyph: str) -> np.ndarray:
    mask = GLYPHS[glyph]
    block = np.kron(mask, np.ones((CELL, CELL), dtype=np.int32))
    return np.where(block == 1, 1, 0)


def _centroid_y_of(arr: np.ndarray, bg: tuple) -> float:
    ys, xs = np.where(np.any(arr != np.array(bg, dtype=arr.dtype), axis=-1) | (arr[..., 0] != bg[0]))
    with np.errstate(all="ignore"):
        mask = ~np.all(arr == np.array(bg, dtype=arr.dtype), axis=-1)
    idx = np.argwhere(mask)
    if idx.size == 0:
        return float("nan")
    return float(idx[:, 0].mean())


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    glyph = "A" if rng.randrange(2) == 0 else "B"
    px = rng.randint(40, CANVAS - PANEL - 40)
    py = rng.randint(40, CANVAS - PANEL - 40)
    scene = {
        "glyph": glyph,
        "panel_x": px,
        "panel_y": py,
        "cell": CELL,
        "warp_amp": AMP,
        "warp_period": PERIOD,
        "glyph_rgb": list(GLYPH_RGB),
        "bg_rgb": list(BG_RGB),
        "canvas": CANVAS,
    }
    return scene


def _panel(content: np.ndarray, bg: tuple) -> np.ndarray:
    block = _glyph_block(content)
    panel = np.full((PANEL, PANEL, 3), bg, dtype=np.uint8)
    off = (PANEL - BLOCK) // 2
    panel[off:off + BLOCK, off:off + BLOCK] = np.where(
        block[..., None] == 1, np.array(GLYPH_RGB, dtype=np.uint8), bg
    )
    return panel


def render(scene: dict) -> Image.Image:
    canvas = int(scene["canvas"])
    bg = tuple(int(v) for v in scene["bg_rgb"])
    glyph = str(scene["glyph"])
    px = int(scene["panel_x"])
    py = int(scene["panel_y"])
    amp = int(scene["warp_amp"])
    period = int(scene["warp_period"])

    img = np.full((canvas, canvas, 3), bg, dtype=np.uint8)

    source = _panel(glyph, bg).astype(np.float64)
    shifts = np.round(amp * np.sin(2.0 * math.pi * np.arange(PANEL) / period)).astype(np.int32)
    warped = np.full_like(source, bg, dtype=np.float64)
    for j in range(PANEL):
        s = int(shifts[j])
        if s == 0:
            warped[:, j] = source[:, j]
        elif s > 0:
            warped[s:, j] = source[:PANEL - s, j]
        else:
            warped[:PANEL + s, j] = source[-s:, j]

    panel_uint = np.clip(warped, 0, 255).astype(np.uint8)
    img[py:py + PANEL, px:px + PANEL] = panel_uint

    img[py, px:px + PANEL] = np.array(FRAME_RGB, dtype=np.uint8)
    img[py + PANEL - 1, px:px + PANEL] = np.array(FRAME_RGB, dtype=np.uint8)
    img[py:py + PANEL, px] = np.array(FRAME_RGB, dtype=np.uint8)
    img[py:py + PANEL, px + PANEL - 1] = np.array(FRAME_RGB, dtype=np.uint8)

    return Image.fromarray(img, "RGB")


def analytic_gold(scene: dict) -> str:
    return str(scene["glyph"])


def _signed_centroid_offset(scene: dict) -> float:
    glyph = str(scene["glyph"])
    bg = tuple(int(v) for v in scene["bg_rgb"])
    amp = int(scene["warp_amp"])
    period = int(scene["warp_period"])
    source = _panel(glyph, bg).astype(np.float64)
    shifts = np.round(amp * np.sin(2.0 * math.pi * np.arange(PANEL) / period)).astype(np.int32)
    warped = np.full_like(source, bg, dtype=np.float64)
    for j in range(PANEL):
        s = int(shifts[j])
        if s == 0:
            warped[:, j] = source[:, j]
        elif s > 0:
            warped[s:, j] = source[:PANEL - s, j]
        else:
            warped[:PANEL + s, j] = source[-s:, j]
    bg_arr = np.array(bg, dtype=np.float64)
    mask = np.any(warped != bg_arr, axis=-1)
    idx = np.argwhere(mask)
    if idx.size == 0:
        return 0.0
    center = (PANEL - 1) / 2.0
    return float(idx[:, 0].mean() - center)


def margin(scene: dict) -> float:
    return _signed_centroid_offset(scene)


def is_quarantined(scene: dict) -> bool:
    return abs(margin(scene)) < MIN_MARGIN


def latent_symmetries(scene: dict) -> list:
    return []
