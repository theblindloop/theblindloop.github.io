"""Deterministic local route-measurement world."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw


def _arc_midpoint(scene: dict[str, Any]) -> tuple[float, float]:
    points = [(float(x), float(y)) for x, y in scene["route_points"]]
    lengths = [math.hypot(b[0] - a[0], b[1] - a[1]) for a, b in zip(points, points[1:])]
    target = sum(lengths) / 2.0
    walked = 0.0
    for a, b, length in zip(points, points[1:], lengths):
        if walked + length >= target:
            t = (target - walked) / length
            return a[0] + t * (b[0] - a[0]), a[1] + t * (b[1] - a[1])
        walked += length
    return points[-1]


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(seed)
    width = height = 512
    # Centers of the non-overlap strips induced by the public 8x8 ablation grid.
    # Choosing among strips translates the cue while keeping its vertical support thin.
    band_x = rng.choice([96, 160, 224, 288, 352])
    band_y = rng.choice([96, 160, 224, 288, 352, 416])
    origin_x = band_x - rng.randint(42, 53)
    origin_y = band_y
    route_width = rng.choice([82, 88, 94])
    stroke = rng.choice([5, 6])
    xs = [origin_x + round(i * route_width / 8) for i in range(9)]
    phase = rng.uniform(0.0, math.tau)
    ys = []
    for i in range(9):
        wave = 5.0 * math.sin(phase + i * 0.83) + 2.0 * math.sin(phase * 0.7 + i * 1.61)
        ys.append(origin_y + round(wave))
    route_points = [[int(x), int(y)] for x, y in zip(xs, ys)]
    provisional = {
        "route_points": route_points,
        "reference_y": 0,
    }
    _, midpoint_y = _arc_midpoint(provisional)
    gap = rng.choice([7, 9, 12, 16])
    # Even/odd seed gives exact balance over each consecutive stress range.
    reference_y = round(midpoint_y + gap if seed % 2 == 0 else midpoint_y - gap)
    distractors = []
    for _ in range(5):
        while True:
            x, y = rng.randint(28, 484), rng.randint(28, 484)
            if abs(x - (origin_x + route_width / 2)) > 100 or abs(y - origin_y) > 75:
                distractors.append([x, y, rng.choice([2, 3])])
                break
    return {
        "width": width,
        "height": height,
        "background": [248, 250, 252],
        "route_color": [8, 157, 185],
        "reference_color": [83, 91, 103],
        "distractor_color": [218, 223, 229],
        "origin_x": origin_x,
        "origin_y": origin_y,
        "route_points": route_points,
        "reference_y": int(reference_y),
        "stroke": stroke,
        "distractors": distractors,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    scale = 4
    width, height = int(scene["width"]), int(scene["height"])
    background = tuple(scene["background"])
    route_color = tuple(scene["route_color"])
    reference_color = tuple(scene["reference_color"])
    distractor_color = tuple(scene["distractor_color"])
    origin_x, origin_y = int(scene["origin_x"]), int(scene["origin_y"])
    points = [(int(x) * scale, int(y) * scale) for x, y in scene["route_points"]]
    reference_y = int(scene["reference_y"])
    stroke = int(scene["stroke"])
    distractors = scene["distractors"]

    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    for x, y, radius in distractors:
        x, y, radius = int(x) * scale, int(y) * scale, int(radius) * scale
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=distractor_color)

    # The dashed reference shares the route's local horizontal extent.
    left = int(scene["route_points"][0][0])
    right = int(scene["route_points"][-1][0])
    dash, gap = 8, 5
    x = left
    while x < right:
        draw.line(
            ((x * scale, reference_y * scale), (min(x + dash, right) * scale, reference_y * scale)),
            fill=reference_color,
            width=2 * scale,
        )
        x += dash + gap
    draw.line(points, fill=route_color, width=stroke * scale, joint="curve")
    # Accessed to make the rendered placement parameters explicit, not hidden labels.
    if not (0 <= origin_x < width and 0 <= origin_y < height):
        raise ValueError("route origin outside canvas")
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    _, y = _arc_midpoint(scene)
    return "above" if y < float(scene["reference_y"]) else "below"


def margin(scene: dict[str, Any]) -> float:
    _, y = _arc_midpoint(scene)
    return abs(y - float(scene["reference_y"]))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) <= 6.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
