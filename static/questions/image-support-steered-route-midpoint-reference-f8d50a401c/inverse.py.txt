"""Independent image-only inverse for the local route measurement."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


class OracleError(ValueError):
    pass


def _route_centerline(array: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    r = array[:, :, 0].astype(np.int16)
    g = array[:, :, 1].astype(np.int16)
    b = array[:, :, 2].astype(np.int16)
    mask = (r < 70) & (g > 105) & (b > 125) & ((g - r) > 65) & ((b - r) > 80)
    ys, xs = np.nonzero(mask)
    if len(xs) < 280:
        raise OracleError("cyan route is missing or materially erased")
    columns = np.unique(xs)
    if int(columns[-1] - columns[0]) < 72 or len(columns) < 72:
        raise OracleError("cyan route does not retain a complete horizontal span")
    if len(columns) < int(columns[-1] - columns[0] + 1) * 0.90:
        raise OracleError("cyan route has a destructive interior gap")
    centers = np.asarray([float(np.median(np.nonzero(mask[:, x])[0])) for x in columns])
    return columns.astype(np.float64), centers


def _halfway_y(xs: np.ndarray, ys: np.ndarray) -> float:
    steps = np.hypot(np.diff(xs), np.diff(ys))
    target = float(steps.sum()) / 2.0
    cumulative = np.concatenate(([0.0], np.cumsum(steps)))
    index = int(np.searchsorted(cumulative, target, side="right") - 1)
    index = min(index, len(steps) - 1)
    fraction = (target - cumulative[index]) / max(float(steps[index]), 1e-9)
    return float(ys[index] + fraction * (ys[index + 1] - ys[index]))


def _reference_y(array: np.ndarray, x0: int, x1: int) -> float:
    r = array[:, :, 0].astype(np.int16)
    g = array[:, :, 1].astype(np.int16)
    b = array[:, :, 2].astype(np.int16)
    neutral = (
        (r >= 45) & (r <= 145) & (np.abs(r - g) < 24) & (np.abs(g - b) < 24)
    )
    neutral[:, : max(0, x0 - 3)] = False
    neutral[:, min(array.shape[1], x1 + 4) :] = False
    ys, _ = np.nonzero(neutral)
    if len(ys) < 35:
        raise OracleError("dashed reference is missing or materially erased")
    values, counts = np.unique(ys, return_counts=True)
    peak = int(values[int(np.argmax(counts))])
    selected = ys[np.abs(ys - peak) <= 2]
    if len(selected) < 30:
        raise OracleError("dashed reference cannot be localized")
    return float(np.median(selected))


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    xs, ys = _route_centerline(array)
    midpoint_y = _halfway_y(xs, ys)
    reference_y = _reference_y(array, int(xs[0]), int(xs[-1]))
    gap = midpoint_y - reference_y
    if not math.isfinite(gap) or abs(gap) <= 3.0:
        raise OracleError("midpoint is too close to the visible reference")
    return "above" if gap < 0 else "below"
