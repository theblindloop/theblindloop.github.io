"""Renderer, sampler and analytic gold for push_cart_replay.

A nine-cell track carries two discs, one red and one blue. A row of cream cards
is applied in drawn left-to-right order. Each card names one disc by a coloured
selector spot, a direction by a green arrow, and a step count by purple pips. A
step moves the named disc one cell; if the neighbouring cell already holds the
other disc, that other disc is pushed one further cell along. A step that would
carry either disc off the track is refused and the rest of that card is lost.

Only the starting layout is ever drawn; no later disc position appears in the
image. The answer is the pair of final cells.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "push_cart_replay-renderer-0.1.0"

# ---------------------------------------------------------------- geometry ---
CANVAS = (900, 430)
N_CELLS = 9
CELL_W = 66
CELL_H = 66
CELL_GAP = 8
CELL_PITCH = CELL_W + CELL_GAP          # 74
TRACK_X0 = 60
TRACK_Y0 = 60
DISC_R = 11

N_CARDS = 5
CARD_W = 130
CARD_H = 112
CARD_PITCH = 160
CARD_X0 = 55
CARD_Y0 = 208
SPOT_R = 13
ARROW_HW = 21           # half width of the green arrow
ARROW_HH = 15           # half height of the green arrow
PIP_R = 7
PIP_DX = 34

# ------------------------------------------------------------------ colours --
BG = (255, 255, 255)
CELL_FILL = (226, 226, 226)
CELL_EDGE = (95, 95, 95)
RED = (200, 45, 45)
BLUE = (45, 80, 200)
CARD_FILL = (250, 240, 205)
CARD_EDGE = (150, 140, 100)
ARROW_GREEN = (30, 140, 60)
PIP_PURPLE = (130, 40, 160)
ORDER_GREY = (120, 120, 120)

DISC_COLOUR = {"red": RED, "blue": BLUE}

_MODE_DX = {
    "wide": (0, 12),
    "near": (27, 30),
    "tight": (32, 35),
}


# --------------------------------------------------------------- primitives --
def cell_cx(index: int) -> float:
    """Centre x of cell ``index`` (0-based), in pixels."""
    return TRACK_X0 + index * CELL_PITCH + CELL_W / 2.0


def cell_cy() -> float:
    return TRACK_Y0 + CELL_H / 2.0


def card_x0(slot: int, jx: int) -> int:
    return CARD_X0 + slot * CARD_PITCH + jx


# ------------------------------------------------------------------ sampling --
def _sample_once(rng: random.Random) -> dict:
    mode = rng.choice(["wide", "wide", "wide", "near", "tight"])
    lo, hi = _MODE_DX[mode]

    cells = rng.sample(range(N_CELLS), 2)
    colours = ["red", "blue"]
    rng.shuffle(colours)
    discs = []
    for colour, cell in zip(colours, cells):
        dx = rng.randint(lo, hi) * rng.choice((-1, 1))
        discs.append({"colour": colour, "cell": cell, "dx": dx})
    rng.shuffle(discs)

    cards = []
    for slot in range(N_CARDS):
        cards.append(
            {
                "slot": slot,
                "target": rng.choice(("red", "blue")),
                "dir": rng.choice((-1, 1)),
                "steps": rng.randint(1, 2),
                "jx": rng.randint(-6, 6),
            }
        )
    rng.shuffle(cards)

    return {
        "canvas": list(CANVAS),
        "n_cells": N_CELLS,
        "mode": mode,
        "discs": discs,
        "cards": cards,
    }


def _start_cells(scene: dict) -> dict:
    """Start cell of each disc, read from drawn geometry, not from list order."""
    out = {}
    for disc in scene["discs"]:
        cx = cell_cx(disc["cell"]) + disc["dx"]
        best = min(range(scene["n_cells"]), key=lambda i: abs(cell_cx(i) - cx))
        out[disc["colour"]] = best
    return out


def _program(scene: dict) -> list[dict]:
    """Cards in drawn left-to-right order."""
    return sorted(scene["cards"], key=lambda c: card_x0(c["slot"], c["jx"]))


def _replay(start: dict, program: list[dict], interact: bool = True) -> dict:
    pos = dict(start)
    pushes = 0
    refused = 0
    for card in program:
        other = "blue" if card["target"] == "red" else "red"
        for _ in range(card["steps"]):
            nxt = pos[card["target"]] + card["dir"]
            if not 0 <= nxt < N_CELLS:
                refused += 1
                break
            if interact and nxt == pos[other]:
                shoved = pos[other] + card["dir"]
                if not 0 <= shoved < N_CELLS:
                    refused += 1
                    break
                pos[other] = shoved
                pushes += 1
            pos[card["target"]] = nxt
    return {"pos": pos, "pushes": pushes, "refused": refused}


def sample_scene(seed: int) -> dict:
    """Deterministically sample one scene that satisfies the shortcut blocks."""
    rng = random.Random(seed * 7919 + 13)
    for _ in range(4000):
        scene = _sample_once(rng)
        start = _start_cells(scene)
        if start["red"] == start["blue"]:
            continue
        # the drawn discs must not touch each other
        xs = {d["colour"]: cell_cx(d["cell"]) + d["dx"] for d in scene["discs"]}
        if abs(xs["red"] - xs["blue"]) < 3 * DISC_R:
            continue
        run = _replay(start, _program(scene))
        end = run["pos"]
        # interaction must be load bearing and visible in the answer
        if run["pushes"] < 1:
            continue
        if end["red"] == start["red"] or end["blue"] == start["blue"]:
            continue
        # a replay that ignores the other disc must give a different answer
        naive = _replay(start, _program(scene), interact=False)["pos"]
        if naive["red"] == end["red"] and naive["blue"] == end["blue"]:
            continue
        # the answer must not be readable off the last card's direction alone
        last = _program(scene)[-1]
        edge = N_CELLS - 1 if last["dir"] > 0 else 0
        if end[last["target"]] == edge:
            continue
        scene["_seed"] = seed
        return scene
    raise RuntimeError(f"no admissible scene for seed {seed}")


# ------------------------------------------------------------- analytic gold --
def analytic_gold(scene: dict) -> str:
    """``"<red final cell>|<blue final cell>"``, cells numbered from 1."""
    end = _replay(_start_cells(scene), _program(scene))["pos"]
    return f"{end['red'] + 1}|{end['blue'] + 1}"


def margin(scene: dict) -> float:
    """Least drawn clearance that any reading of the image depends on."""
    clearances = []
    for disc in scene["discs"]:
        # distance from the perpendicular bisector between neighbouring cells
        clearances.append(CELL_PITCH / 2.0 - abs(disc["dx"]))
    xs = sorted(cell_cx(d["cell"]) + d["dx"] for d in scene["discs"])
    clearances.append((xs[1] - xs[0]) - 2 * DISC_R)          # disc separation
    order = sorted(scene["cards"], key=lambda c: card_x0(c["slot"], c["jx"]))
    for a, b in zip(order, order[1:]):
        clearances.append(card_x0(b["slot"], b["jx"]) - (card_x0(a["slot"], a["jx"]) + CARD_W))
    clearances.append(float(PIP_DX - 2 * PIP_R))             # pip separation
    clearances.append(float(ARROW_HW - ARROW_HW / 3.0))      # arrow asymmetry
    return round(float(min(clearances)), 6)


QUARANTINE_THRESHOLD = 6.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_THRESHOLD


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Re-orderings of the unordered bags; each must be pixel identical."""
    out: list[tuple[str, dict]] = []
    flipped = dict(scene)
    flipped["discs"] = list(reversed(scene["discs"]))
    out.append(("discs_reversed", flipped))
    rev = dict(scene)
    rev["cards"] = list(reversed(scene["cards"]))
    out.append(("cards_reversed", rev))
    rot = dict(scene)
    rot["cards"] = scene["cards"][2:] + scene["cards"][:2]
    out.append(("cards_rotated", rot))
    both = dict(scene)
    both["discs"] = list(reversed(scene["discs"]))
    both["cards"] = list(reversed(scene["cards"]))
    out.append(("discs_and_cards_reversed", both))
    return out


# --------------------------------------------------------------- rendering ---
def _triangle(draw: ImageDraw.ImageDraw, cx: float, cy: float, direction: int) -> None:
    """Isoceles arrow head; apex on the side the arrow points to."""
    apex = (cx + direction * ARROW_HW, cy)
    back_x = cx - direction * ARROW_HW
    draw.polygon([apex, (back_x, cy - ARROW_HH), (back_x, cy + ARROW_HH)], fill=ARROW_GREEN)


def render(scene: dict) -> Image.Image:
    w, h = scene["canvas"]
    img = Image.new("RGB", (w, h), BG)
    d = ImageDraw.Draw(img)

    for i in range(scene["n_cells"]):
        x0 = TRACK_X0 + i * CELL_PITCH
        d.rectangle([x0, TRACK_Y0, x0 + CELL_W, TRACK_Y0 + CELL_H], fill=CELL_FILL, outline=CELL_EDGE, width=2)

    for disc in sorted(scene["discs"], key=lambda x: x["colour"]):
        cx = cell_cx(disc["cell"]) + disc["dx"]
        cy = cell_cy()
        d.ellipse([cx - DISC_R, cy - DISC_R, cx + DISC_R, cy + DISC_R], fill=DISC_COLOUR[disc["colour"]])

    for card in sorted(scene["cards"], key=lambda c: card_x0(c["slot"], c["jx"])):
        x0 = card_x0(card["slot"], card["jx"])
        y0 = CARD_Y0
        d.rectangle([x0, y0, x0 + CARD_W, y0 + CARD_H], fill=CARD_FILL, outline=CARD_EDGE, width=2)
        cx = x0 + CARD_W / 2.0
        d.ellipse(
            [cx - SPOT_R, y0 + 12, cx + SPOT_R, y0 + 12 + 2 * SPOT_R],
            fill=DISC_COLOUR[card["target"]],
        )
        _triangle(d, cx, y0 + 62, card["dir"])
        pips = card["steps"]
        first = cx - (pips - 1) * PIP_DX / 2.0
        for k in range(pips):
            px = first + k * PIP_DX
            py = y0 + 94
            d.ellipse([px - PIP_R, py - PIP_R, px + PIP_R, py + PIP_R], fill=PIP_PURPLE)

    # drawn statement of the application order
    ay = CARD_Y0 + CARD_H + 34
    d.line([110, ay, 800, ay], fill=ORDER_GREY, width=5)
    d.polygon([(820, ay), (800, ay - 12), (800, ay + 12)], fill=ORDER_GREY)
    return img
