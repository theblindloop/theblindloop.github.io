"""Independent pixel-only inverse for five-key group order."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_MIN_AREA = 2800
_MAX_AREA = 4400
_MIN_GAP_ADVANTAGE = 18.0


class OracleAbstention(ValueError):
    pass


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dy == 0 and dx == 0:
                        continue
                    ny, nx = cy + dy, cx + dx
                    if (
                        0 <= ny < height
                        and 0 <= nx < width
                        and mask[ny, nx]
                        and not seen[ny, nx]
                    ):
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        output.append(np.asarray(points, dtype=np.int32))
    return output


def _key_boxes(array: np.ndarray) -> list[tuple[int, int, int, int]]:
    high = array.max(axis=2).astype(np.int16)
    low = array.min(axis=2).astype(np.int16)
    chroma = high - low
    mask = (high <= 62) & (chroma <= 24)
    boxes = []
    for points in _components(mask):
        area = len(points)
        if area < 24:
            continue
        if not _MIN_AREA <= area <= _MAX_AREA:
            raise OracleAbstention(f"dark component has unsupported area {area}")
        ys, xs = points[:, 0], points[:, 1]
        x0, x1 = int(xs.min()), int(xs.max())
        y0, y1 = int(ys.min()), int(ys.max())
        width, height = x1 - x0 + 1, y1 - y0 + 1
        fill = area / float(width * height)
        if not (25 <= width <= 31 and 131 <= height <= 142):
            raise OracleAbstention("dark component is not a supported piano key")
        if not 0.88 <= fill <= 1.01:
            raise OracleAbstention("dark component is not a solid key rectangle")
        boxes.append((x0, y0, x1, y1))
    if len(boxes) != 5:
        raise OracleAbstention(f"expected five black keys, found {len(boxes)}")
    boxes.sort(key=lambda box: box[0])
    return boxes


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 320 or array.shape[1] < 320:
        raise OracleAbstention("raster is too small")
    boxes = _key_boxes(array)
    heights = [box[3] - box[1] + 1 for box in boxes]
    tops = [box[1] for box in boxes]
    if max(heights) - min(heights) > 2 or max(tops) - min(tops) > 2:
        raise OracleAbstention("keys are not a common aligned row")
    gaps = [float(boxes[index + 1][0] - boxes[index][2] - 1) for index in range(4)]
    order = sorted(range(4), key=lambda index: gaps[index], reverse=True)
    advantage = gaps[order[0]] - gaps[order[1]]
    if advantage <= _MIN_GAP_ADVANTAGE:
        raise OracleAbstention("no uniquely separated pair of key groups")
    if order[0] == 1:
        return "yes"
    if order[0] == 2:
        return "no"
    raise OracleAbstention("wide separator does not define groups of two and three")
