"""Deterministic renderer for visible piano-key group order."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

CANVAS_SIZE = 512
SUPERSAMPLE = 4
QUARANTINE_GAP_ADVANTAGE = 18.0

BACKGROUNDS = (
    (244, 241, 234),
    (237, 243, 246),
    (245, 238, 244),
    (239, 244, 235),
)
CASES = (
    (84, 112, 132),
    (126, 91, 118),
    (79, 119, 105),
    (131, 99, 70),
)
ACCENTS = (
    (218, 102, 85),
    (71, 145, 184),
    (224, 166, 64),
    (118, 165, 99),
)
BLACK_KEY = (22, 24, 30)


def sample_scene(seed: int) -> dict:
    """Return adjacent matched control/counterfactual scenes.

    Seeds 2k and 2k+1 share every visual nuisance. The even control places the
    wide separator after key two; the odd counterfactual places it after key
    three. Both have five equal keys and therefore identical key ink.
    """

    seed = int(seed)
    pair = seed // 2
    rng = random.Random(pair ^ 0x51A0C0DE)
    return {
        "background_code": pair % len(BACKGROUNDS),
        "case_code": (pair // len(BACKGROUNDS)) % len(CASES),
        "canvas_size": CANVAS_SIZE,
        "decor_code": rng.randrange(16),
        "group_order_code": seed % 2,
        "key_height": 132 + 4 * rng.randrange(3),
        "key_width": 26 + 2 * rng.randrange(3),
        "keyboard_x": 84 + 4 * rng.randrange(3),
        "keyboard_y": 148 + 4 * rng.randrange(3),
        "large_gap": 154 + 4 * rng.randrange(3),
        "small_gap": 12 + 2 * rng.randrange(3),
    }


def _scaled_box(box: tuple[float, float, float, float], scale: int) -> tuple[int, int, int, int]:
    return tuple(round(value * scale) for value in box)


def _group_sizes(scene: dict) -> tuple[int, int]:
    return (2, 3) if int(scene["group_order_code"]) % 2 == 0 else (3, 2)


def _key_boxes(scene: dict) -> list[tuple[float, float, float, float]]:
    left_count, right_count = _group_sizes(scene)
    width = float(scene["key_width"])
    height = float(scene["key_height"])
    small = float(scene["small_gap"])
    large = float(scene["large_gap"])
    x = float(scene["keyboard_x"])
    y = float(scene["keyboard_y"])
    boxes = []
    for group_index, count in enumerate((left_count, right_count)):
        for _ in range(count):
            boxes.append((x, y, x + width, y + height))
            x += width + small
        x -= small
        if group_index == 0:
            x += large
    return boxes


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas_size"])
    background = BACKGROUNDS[int(scene["background_code"]) % len(BACKGROUNDS)]
    case = CASES[int(scene["case_code"]) % len(CASES)]
    decor = int(scene["decor_code"])
    boxes = _key_boxes(scene)
    ss = SUPERSAMPLE

    image = Image.new("RGB", (size * ss, size * ss), background)
    draw = ImageDraw.Draw(image)

    # Nondiagnostic colored ornaments are paired and remote from the keyboard.
    accent = ACCENTS[decor % len(ACCENTS)]
    accent_2 = ACCENTS[(decor // 3 + 1) % len(ACCENTS)]
    for index, (cx, cy) in enumerate(((54, 66), (458, 66), (54, 446), (458, 446))):
        radius = 7 + ((decor + index * 3) % 5)
        color = accent if index % 2 == 0 else accent_2
        draw.ellipse(_scaled_box((cx - radius, cy - radius, cx + radius, cy + radius), ss), fill=color)

    left = min(box[0] for box in boxes) - 24
    right = max(box[2] for box in boxes) + 24
    top = min(box[1] for box in boxes) - 30
    bottom = max(box[3] for box in boxes) + 48
    case_edge = tuple(max(58, channel - 24) for channel in case)
    white_key = (248, 247, 241)
    key_line = (174, 176, 172)

    # A cohesive keyboard case makes the familiar object explicit while the
    # decisive black keys remain disconnected components.
    draw.rounded_rectangle(
        _scaled_box((left, top, right, bottom), ss),
        radius=18 * ss,
        fill=case,
        outline=case_edge,
        width=3 * ss,
    )
    bed_top = top + 22
    bed_bottom = bottom - 20
    draw.rounded_rectangle(
        _scaled_box((left + 14, bed_top, right - 14, bed_bottom), ss),
        radius=8 * ss,
        fill=white_key,
    )

    # White-key divisions are light enough never to enter the oracle's dark mask.
    division_count = 10
    usable_left, usable_right = left + 14, right - 14
    for index in range(1, division_count):
        x = usable_left + (usable_right - usable_left) * index / division_count
        draw.line(
            [(round(x * ss), round(bed_top * ss)), (round(x * ss), round(bed_bottom * ss))],
            fill=key_line,
            width=2 * ss,
        )

    for box in boxes:
        x0, y0, x1, y1 = box
        draw.rounded_rectangle(
            _scaled_box((x0, y0, x1, y1), ss),
            radius=4 * ss,
            fill=BLACK_KEY,
        )
        # A subtle highlight stays within the same connected dark component.
        draw.rounded_rectangle(
            _scaled_box((x0 + 4, y0 + 5, x1 - 4, y0 + 9), ss),
            radius=2 * ss,
            fill=(45, 47, 54),
        )

    return image.resize((size, size), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "yes" if int(scene["group_order_code"]) % 2 == 0 else "no"


def margin(scene: dict) -> float:
    return float(scene["large_gap"]) - float(scene["small_gap"])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= QUARANTINE_GAP_ADVANTAGE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["background_code"] = int(twin["background_code"]) + len(BACKGROUNDS)
    return [("one_full_background_palette_cycle", twin)]
