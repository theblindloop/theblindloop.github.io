"""Analytic sampler and renderer for center-tab tile assembly."""

from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image, ImageDraw

LABELS = ("circle", "diamond", "plus", "X")
N = 24
HALF = 12
K_PER_QUADRANT = 24
CELL = 10
CONTENT = HALF * CELL
CANVAS = 640
BASES = ((90, 90), (430, 90), (90, 430), (430, 430))
DESIRED_CORNERS = ((1, 1), (1, 0), (0, 1), (0, 0))
PALETTES = (
    ((20, 178, 205), (222, 38, 151)),
    ((35, 158, 222), (202, 46, 184)),
)


def _score(label: str, x: int, y: int) -> float:
    cx = cy = 11.5
    if label == "circle":
        return abs(math.hypot(x - cx, y - cy) - 8.2)
    if label == "diamond":
        return abs(abs(x - cx) + abs(y - cy) - 11.0)
    if label == "plus":
        return min(abs(x - cx), abs(y - cy))
    return min(abs((x - cx) - (y - cy)), abs((x - cx) + (y - cy)))


def _template(label: str) -> np.ndarray:
    ranked = sorted(
        ((_score(label, x, y), y, x) for y in range(HALF) for x in range(HALF))
    )[:K_PER_QUADRANT]
    mask = np.zeros((N, N), dtype=np.uint8)
    for _, y, x in ranked:
        for yy, xx in ((y, x), (y, N - 1 - x), (N - 1 - y, x), (N - 1 - y, N - 1 - x)):
            mask[yy, xx] = 1
    return mask


def _templates() -> dict[str, np.ndarray]:
    return {label: _template(label) for label in LABELS}


def _rotate_corner_clockwise(corner: tuple[int, int], turns: int) -> tuple[int, int]:
    row, col = corner
    for _ in range(turns % 4):
        row, col = col, 1 - row
    return row, col


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    return {
        "answer": LABELS[int(seed) % len(LABELS)],
        "rotations": [rng.randrange(4) for _ in range(4)],
        "offsets": [[rng.randint(-12, 12), rng.randint(-12, 12)] for _ in range(4)],
        "palette_variant": rng.randrange(len(PALETTES)),
    }


def render(scene: dict) -> Image.Image:
    answer = scene["answer"]
    rotations = scene["rotations"]
    offsets = scene["offsets"]
    frame_color, tab_color = PALETTES[int(scene["palette_variant"])]
    whole = _templates()[answer]
    quadrants = (
        whole[:HALF, :HALF], whole[:HALF, HALF:],
        whole[HALF:, :HALF], whole[HALF:, HALF:],
    )
    image = Image.new("RGB", (CANVAS, CANVAS), (248, 247, 242))
    draw = ImageDraw.Draw(image)
    for index, (base, offset, turns, tile) in enumerate(zip(BASES, offsets, rotations, quadrants)):
        x = int(base[0] + offset[0])
        y = int(base[1] + offset[1])
        outer = (x - 7, y - 7, x + CONTENT + 6, y + CONTENT + 6)
        draw.rectangle(outer, fill=frame_color)
        draw.rectangle((x, y, x + CONTENT - 1, y + CONTENT - 1), fill=(255, 255, 255))
        current = np.rot90(tile, -int(turns))
        for row, col in np.argwhere(current > 0):
            left = x + int(col) * CELL + 1
            top = y + int(row) * CELL + 1
            draw.rectangle((left, top, left + CELL - 3, top + CELL - 3), fill=(25, 25, 25))
        marker = _rotate_corner_clockwise(DESIRED_CORNERS[index], int(turns))
        mx = x - 7 if marker[1] == 0 else x + CONTENT - 3
        my = y - 7 if marker[0] == 0 else y + CONTENT - 3
        draw.rectangle((mx, my, mx + 10, my + 10), fill=tab_color)
    return image


def analytic_gold(scene: dict) -> str:
    return scene["answer"]


def margin(scene: dict) -> float:
    target = _templates()[scene["answer"]]
    distances = sorted(
        int(np.count_nonzero(target != other))
        for label, other in _templates().items()
        if label != scene["answer"]
    )
    return float(distances[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
