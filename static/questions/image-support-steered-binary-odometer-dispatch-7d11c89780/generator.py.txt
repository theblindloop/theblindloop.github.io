from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

W, H = 960, 480
BG = (248, 247, 242)
INK = (38, 43, 54)
NAVY = (26, 51, 79)
GOLD = (218, 164, 45)
CYAN = (31, 178, 190)
CORAL = (226, 91, 82)

def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    commands = [rng.choice((1, -1)) for _ in range(4)]
    target = seed % 4
    start = (target - sum(commands)) % 4
    return {
        "start_bits": [start // 2, start % 2],
        "commands": commands,
        "palette_code": rng.randrange(2),
    }

def _font(size: int, bold: bool = False):
    names = [
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf" if bold else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
        "/usr/share/fonts/truetype/liberation2/LiberationSans-Bold.ttf" if bold else "/usr/share/fonts/truetype/liberation2/LiberationSans-Regular.ttf",
    ]
    for name in names:
        try:
            return ImageFont.truetype(name, size)
        except OSError:
            pass
    return ImageFont.load_default()

def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (W, H), BG)
    d = ImageDraw.Draw(image)
    title = _font(25, True)
    small = _font(18, True)
    huge = _font(44, True)

    d.text((40, 24), "START COUNTER", fill=INK, font=title)
    d.text((675, 24), "COMMANDS", fill=INK, font=title)
    # A large disconnected left evidence component.
    d.rounded_rectangle((38, 78, 292, 404), radius=22, outline=GOLD, width=12)
    d.text((82, 105), "2s", fill=INK, font=small)
    d.text((211, 105), "1s", fill=INK, font=small)
    for i, bit in enumerate(scene["start_bits"]):
        cx = 104 + 126 * i
        box = (cx - 38, 178, cx + 38, 254)
        if int(bit):
            d.ellipse(box, fill=NAVY, outline=NAVY, width=1)
        else:
            d.ellipse(box, outline=NAVY, width=10)
    d.text((72, 292), "dark = 1", fill=INK, font=small)
    d.text((72, 329), "hollow = 0", fill=INK, font=small)

    # The right bank is far away, leaving a blank vertical gulf.
    positions = [(680, 92), (806, 92), (680, 256), (806, 256)]
    for index, (x, y) in enumerate(positions):
        command = int(scene["commands"][index])
        color = CYAN if command == 1 else CORAL
        d.rounded_rectangle((x, y, x + 112, y + 126), radius=16, fill=color)
        d.ellipse((x + 10, y + 9, x + 28, y + 27), fill=BG)
        d.text((x + 40, y + 33), "+" if command == 1 else "−", fill=INK, font=huge)
        d.text((x + 49, y + 94), str(index + 1), fill=INK, font=small)
    return image

def analytic_gold(scene: dict) -> str:
    value = 2 * int(scene["start_bits"][0]) + int(scene["start_bits"][1])
    for command in scene["commands"]:
        value = (value + int(command)) % 4
    return format(value, "02b")

def margin(scene: dict) -> float:
    return 32.0

def is_quarantined(scene: dict) -> bool:
    try:
        return not (
            len(scene["start_bits"]) == 2
            and all(int(v) in (0, 1) for v in scene["start_bits"])
            and len(scene["commands"]) == 4
            and all(int(v) in (-1, 1) for v in scene["commands"])
        )
    except Exception:
        return True

def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["palette_code"] = int(scene["palette_code"]) + 2
    return [("palette_period_two", twin)]
