from __future__ import annotations

import numpy as np
from PIL import Image

BG = np.array((248, 247, 242), dtype=np.uint8)
NAVY = np.array((26, 51, 79), dtype=np.uint8)
GOLD = np.array((218, 164, 45), dtype=np.uint8)
CYAN = np.array((31, 178, 190), dtype=np.uint8)
CORAL = np.array((226, 91, 82), dtype=np.uint8)

def _components(mask: np.ndarray):
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out = []
    for yy, xx in np.argwhere(mask):
        y, x = int(yy), int(xx)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        pts = []
        while stack:
            cy, cx = stack.pop()
            pts.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(pts) >= 20:
            ys = [p[0] for p in pts]
            xs = [p[1] for p in pts]
            out.append((len(pts), min(xs), min(ys), max(xs), max(ys)))
    return out

def decision_from_image(image: Image.Image) -> str:
    a = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if a.ndim != 3 or a.shape[0] < 300 or a.shape[1] < 600:
        raise ValueError("canvas is missing")

    gold = np.all(a == GOLD, axis=2)
    gold_parts = _components(gold)
    if len(gold_parts) != 1:
        raise ValueError("START frame is missing or damaged")
    ga, gx0, gy0, gx1, gy1 = gold_parts[0]
    if ga < 11800 or gx1 - gx0 < 235 or gy1 - gy0 < 305:
        raise ValueError("START frame is incomplete")

    navy = np.all(a == NAVY, axis=2)
    mid = (gx0 + gx1) // 2
    lamp_band = navy[max(0, gy0 + 70):min(a.shape[0], gy0 + 190)]
    left_ink = int(lamp_band[:, gx0:mid].sum())
    right_ink = int(lamp_band[:, mid:gx1 + 1].sum())
    def bit(count: int) -> int:
        if 4300 <= count <= 4900:
            return 1
        if 1700 <= count <= 2400:
            return 0
        raise ValueError("bit lamp is missing or ambiguous")
    start = 2 * bit(left_ink) + bit(right_ink)

    cards = []
    for color, delta in ((CYAN, 1), (CORAL, -1)):
        for area, x0, y0, x1, y1 in _components(np.all(a == color, axis=2)):
            if area < 11800 or x1 - x0 < 100 or y1 - y0 < 115:
                raise ValueError("command card is incomplete")
            cards.append((y0, x0, delta))
    if len(cards) != 4:
        raise ValueError("four command cards were not recovered")
    cards.sort(key=lambda item: (item[0] // 100, item[1]))
    value = start
    for _, __, delta in cards:
        value = (value + delta) % 4
    return format(value, "02b")
