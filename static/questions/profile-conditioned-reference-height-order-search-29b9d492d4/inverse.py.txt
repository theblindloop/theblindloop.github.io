"""Independent pixel-only inverse arm for reference_height_order_search.

Receives only a PIL image and the public question contract. It finds the three
bar heights of the reference group (top panel) and of each of the four quadrant
groups using the documented, fixed scene geometry, derives each group's
left-to-right rank permutation from its measured heights, and selects the
quadrant whose permutation exactly equals the reference group's permutation.

Imports nothing from the renderer, prompts, generator, or verifier.
"""

from __future__ import annotations

import numpy as np

# Documented fixed geometry of the rendered world (part of the public contract).
BAR_W = 24
REF_XS = (210, 240, 270)
REF_BASE = 175
GROUP_XS = {
    "TL": (92, 120, 148),
    "TR": (332, 360, 388),
    "BL": (92, 120, 148),
    "BR": (332, 360, 388),
}
GROUP_BASE = {"TL": 285, "TR": 285, "BL": 425, "BR": 425}
_H_DIV = 345  # horizontal quadrant divider


def _band(sector: str) -> tuple[int, int]:
    if sector == "REF":
        return (0, 210)
    if sector in ("TL", "TR"):
        return (0, 300)
    return (300, 480)


def _bar_height(arr, x, base, ymin, ymax) -> float:
    """Height of the filled bar whose top edge sits at centre column x+12, measured
    only within the horizontal band [ymin, ymax) of the group's cell."""
    col = int(x) + BAR_W // 2
    band = arr[ymin:ymax, col]
    dark = np.nonzero(band < 150)[0]
    if dark.size == 0:
        return 0.0
    top = int(ymin + dark.min())
    if top >= base:
        return 0.0
    return float(base - top)


def _permutation(heights) -> tuple:
    assert len(heights) == 3 and min(heights) > 0 and len(set(heights)) == 3, heights
    order = sorted(range(3), key=lambda i: heights[i])
    perm = [0, 0, 0]
    for rank, pos in enumerate(order):
        perm[pos] = rank
    return tuple(perm)


def decision_from_image(image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16).sum(axis=2)

    ref_heights = [_bar_height(arr, x, REF_BASE, *_band("REF")) for x in REF_XS]
    ref_perm = _permutation(ref_heights)

    sectors = ("TL", "TR", "BL", "BR")
    for sector in sectors:
        heights = [_bar_height(arr, x, GROUP_BASE[sector], *_band(sector))
                   for x in GROUP_XS[sector]]
        if _permutation(heights) == ref_perm:
            return sector
    return "TL"
