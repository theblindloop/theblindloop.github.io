"""Reference height-order visual search world.

A reference group of three dark bars is shown in the top panel; the three bars
sit on a shared baseline and each has a different height, so they define a
left-to-right height ORDER (a permutation). Four quadrant groups below each also
hold three bars on a shared baseline. Exactly one quadrant group has the SAME
left-to-right height order (the same arrangement of shortest/middle/tallest) as
the reference group. The decision is the quadrant of that matching group.

The matching criterion is the relative ORDER (rank permutation) of the three
heights, not their absolute values: every group uses different absolute heights,
so the answer cannot be read from absolute magnitude, ink, or position.
"""

from __future__ import annotations

import itertools

import numpy as np
from PIL import Image, ImageDraw

SIZE = 480
BG = (255, 255, 255)
BAR = (20, 20, 20)
BOX = (160, 160, 160)
DIV = (235, 235, 235)

BAR_W = 24
MIN_SEP = 14  # minimum pixel separation between any two heights within a group

SECTORS = ("TL", "TR", "BL", "BR")
GROUP_XS = {
    "TL": (92, 120, 148),
    "TR": (332, 360, 388),
    "BL": (92, 120, 148),
    "BR": (332, 360, 388),
}
GROUP_BASE = {
    "TL": 285,
    "TR": 285,
    "BL": 425,
    "BR": 425,
}
REF_XS = (210, 240, 270)
REF_BASE = 175

# All 3! rank permutations of (short, middle, tall) across left/mid/right bars.
_PERMS = list(itertools.permutations((0, 1, 2)))


def _sample_height_triple(rng: np.random.Generator) -> tuple[int, int, int]:
    """Return three distinct heights consistent with ascending ranks 0,1,2."""
    ranks = [0, 1, 2]
    heights = []
    lo, hi = 40, 120
    for _ in range(3):
        for _try in range(200):
            v = int(rng.integers(lo, hi + 1))
            if all(abs(v - h) >= MIN_SEP for h in heights):
                heights.append(v)
                break
        else:
            heights.append(lo)
    heights.sort()  # heights[0]=shortest, heights[2]=tallest
    return tuple(heights)


def _heights_for_perm(perm, rng) -> tuple[int, int, int]:
    """Assign three heights to left/mid/right positions realising rank-permutation.

    perm is a tuple giving, for positions left,mid,right, the rank (0=shortest,
    2=tallest) of the bar placed there.
    """
    ordered = list(_sample_height_triple(rng))  # index = rank
    return tuple(ordered[rank] for rank in perm)


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(seed)
    winner_sector = SECTORS[int(rng.integers(0, len(SECTORS)))]

    ref_perm = _PERMS[int(rng.integers(0, len(_PERMS)))]
    ref_heights = _heights_for_perm(ref_perm, rng)

    groups = {}
    for sector in SECTORS:
        if sector == winner_sector:
            groups[sector] = {"perm": ref_perm, "heights": _heights_for_perm(ref_perm, rng)}
            continue
        others = [p for p in _PERMS if p != ref_perm]
        perm = others[int(rng.integers(0, len(others)))]
        groups[sector] = {"perm": perm, "heights": _heights_for_perm(perm, rng)}

    return {
        "reference": {"perm": ref_perm, "heights": ref_heights},
        "groups": groups,
        "winner_sector": winner_sector,
    }


def _draw_bar(draw: ImageDraw.ImageDraw, x: int, base: int, height: int) -> None:
    draw.rectangle([x, base - height, x + BAR_W, base], fill=BAR)


def _draw_group(draw: ImageDraw.ImageDraw, xs, base, heights) -> None:
    for x, h in zip(xs, heights):
        _draw_bar(draw, x, base, h)


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (SIZE, SIZE), BG)
    draw = ImageDraw.Draw(img)

    # Reference panel outline.
    draw.rectangle([180, 40, 300, 195], outline=BOX, width=3)

    # Quadrant dividers (very light grey, distinct from dark bars).
    draw.line([SIZE // 2, 210, SIZE // 2, SIZE], fill=DIV, width=2)
    draw.line([0, 345, SIZE, 345], fill=DIV, width=2)

    ref = scene["reference"]
    _draw_group(draw, REF_XS, REF_BASE, ref["heights"])

    for sector in SECTORS:
        _draw_group(draw, GROUP_XS[sector], GROUP_BASE[sector],
                    scene["groups"][sector]["heights"])

    return img


def _perm_of_scene_group(heights) -> tuple:
    order = sorted(range(3), key=lambda i: heights[i])
    perm = [0, 0, 0]
    for rank, pos in enumerate(order):
        perm[pos] = rank
    return tuple(perm)


def _mismatch(gperm, ref_perm) -> int:
    return sum(1 for a, b in zip(gperm, ref_perm) if a != b)


def _decision(scene: dict) -> tuple[str, int]:
    ref_perm = tuple(scene["reference"]["perm"])
    mismatches = {}
    for sector in SECTORS:
        mismatches[sector] = _mismatch(tuple(scene["groups"][sector]["perm"]), ref_perm)
    winner = min(mismatches, key=lambda s: (mismatches[s], s))
    # Margin: difference between the closest distractor's mismatch and the
    # winner's (0). Winner is unique (mismatch 0); every distractor has >= 2.
    runner_up = min(m for s, m in mismatches.items() if s != winner)
    return winner, int(runner_up - mismatches[winner])


def analytic_gold(scene: dict) -> str:
    sector, _ = _decision(scene)
    return sector


def margin(scene: dict) -> float:
    _, m = _decision(scene)
    return float(m)


def is_quarantined(scene: dict) -> bool:
    # All groups are guaranteed distinct-height triples with a unique reference
    # match and a runner-up differing in at least two bar positions, so no
    # finite-resolution boundary case arises. Returns False always.
    return False


def latent_symmetries(scene: dict) -> list:
    return []
