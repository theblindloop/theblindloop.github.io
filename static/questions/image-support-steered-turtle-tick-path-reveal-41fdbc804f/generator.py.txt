"""Analytic world and rasterizer for Trail-Tick Turtle Reveal.

The answer program is carried by colored cross-ticks on one continuous trail.
Carrier geometry and every global nuisance are sampled independently of the
program.  The inverse implementation lives in oracle.py and sees only pixels.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (247, 244, 237)
TRAIL = (35, 45, 60)
START_RED = (205, 54, 63)
END_GOLD = (235, 188, 43)

PALETTES = (
    ((25, 181, 205), (239, 126, 48)),
    ((36, 193, 189), (229, 139, 42)),
    ((47, 165, 211), (242, 111, 57)),
)

# Each program has sixteen commands and exactly eight of each color.  The
# normalized drawings are, respectively, a diagonal stair, a vertical wave,
# and a square-wave crown.  Equal command/color/ink counts block cheap cues.
PROGRAMS = {
    "stair": "LR" * 8,
    "wave": "RLLR" * 4,
    "crown": "RRLL" * 4,
}
SYMBOLS = tuple(PROGRAMS)


def _mix(seed: int, salt: int) -> int:
    value = (seed ^ salt) & 0xFFFFFFFF
    value ^= value >> 16
    value = (value * 0x7FEB352D) & 0xFFFFFFFF
    value ^= value >> 15
    value = (value * 0x846CA68B) & 0xFFFFFFFF
    return value ^ (value >> 16)


def sample_scene(seed: int) -> dict:
    """Return one JSON-serializable, class-balanced deterministic scene."""
    rng = random.Random(_mix(int(seed), 0xA53C9E21))
    return {
        "symbol": SYMBOLS[int(seed) % len(SYMBOLS)],
        "route_variant": _mix(int(seed), 0x13579BDF) % 4,
        "palette": _mix(int(seed), 0x2468ACE0) % len(PALETTES),
        "rotation": _mix(int(seed), 0x31415926) % 4,
        "scale": round(rng.uniform(0.88, 0.94), 6),
        "offset_x": round(rng.uniform(-5.0, 5.0), 6),
        "offset_y": round(rng.uniform(-5.0, 5.0), 6),
    }


def _route_local(t: float, variant: int) -> tuple[float, float]:
    """A chord-spanning curve; the sinusoid is perpendicular to the chord."""
    x0, y0 = -210.0, 200.0
    x1, y1 = 210.0, -200.0
    dx, dy = x1 - x0, y1 - y0
    length = math.hypot(dx, dy)
    nx, ny = -dy / length, dx / length
    cycles = (1, 2, 1, 2)[variant]
    amplitude = (22.0, 18.0, -25.0, -20.0)[variant]
    wiggle = amplitude * math.sin(cycles * 2.0 * math.pi * t)
    return x0 + dx * t + nx * wiggle, y0 + dy * t + ny * wiggle


def _route_tangent(t: float, variant: int) -> tuple[float, float]:
    eps = 0.0005
    ax, ay = _route_local(max(0.0, t - eps), variant)
    bx, by = _route_local(min(1.0, t + eps), variant)
    dx, dy = bx - ax, by - ay
    norm = math.hypot(dx, dy)
    return dx / norm, dy / norm


def _place(point: tuple[float, float], scene: dict) -> tuple[float, float]:
    x, y = point
    quarter = int(scene["rotation"]) % 4
    for _ in range(quarter):
        x, y = -y, x
    scale = float(scene["scale"])
    return (
        WIDTH / 2.0 + float(scene["offset_x"]) + scale * x,
        HEIGHT / 2.0 + float(scene["offset_y"]) + scale * y,
    )


def _rotated_vector(vector: tuple[float, float], scene: dict) -> tuple[float, float]:
    x, y = vector
    for _ in range(int(scene["rotation"]) % 4):
        x, y = -y, x
    return x, y


def _disk(draw: ImageDraw.ImageDraw, center: tuple[float, float], radius: float, fill) -> None:
    x, y = center
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=fill)


def render(scene: dict) -> Image.Image:
    """Render a 512px flat-color PNG with no metadata or hidden channel."""
    symbol = scene["symbol"]
    variant = int(scene["route_variant"])
    palette = PALETTES[int(scene["palette"])]
    # Read all declared geometric nuisances through _place.
    _ = (scene["rotation"], scene["scale"], scene["offset_x"], scene["offset_y"])
    if symbol not in PROGRAMS:
        raise ValueError(f"unknown symbol {symbol!r}")

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    route = [_place(_route_local(i / 300.0, variant), scene) for i in range(301)]
    draw.line(route, fill=TRAIL, width=9, joint="curve")

    # A colored tick crosses (and therefore physically joins) the carrier at
    # each command station. Programs always contain eight L and eight R ticks.
    for index, command in enumerate(PROGRAMS[symbol]):
        t = (index + 1) / 17.0
        center = _place(_route_local(t, variant), scene)
        tx, ty = _rotated_vector(_route_tangent(t, variant), scene)
        nx, ny = -ty, tx
        half = 14.0 * float(scene["scale"])
        ends = [
            (center[0] - nx * half, center[1] - ny * half),
            (center[0] + nx * half, center[1] + ny * half),
        ]
        draw.line(ends, fill=palette[0 if command == "L" else 1], width=7)
        _disk(draw, center, 3.5, palette[0 if command == "L" else 1])

    start = _place(_route_local(0.0, variant), scene)
    end = _place(_route_local(1.0, variant), scene)
    # Hollow red ring: the dark trail remains visible inside and touches it.
    sx, sy = start
    draw.ellipse((sx - 14, sy - 14, sx + 14, sy + 14), outline=START_RED, width=8)

    # Filled gold triangle, pointed along the final direction of travel.
    tx, ty = _rotated_vector(_route_tangent(1.0, variant), scene)
    nx, ny = -ty, tx
    tip = (end[0] + tx * 15, end[1] + ty * 15)
    back = (end[0] - tx * 12, end[1] - ty * 12)
    triangle = [tip, (back[0] + nx * 13, back[1] + ny * 13), (back[0] - nx * 13, back[1] - ny * 13)]
    draw.polygon(triangle, fill=END_GOLD)
    return image


def analytic_gold(scene: dict) -> str:
    return scene["symbol"]


def _hamming(a: str, b: str) -> int:
    return sum(x != y for x, y in zip(a, b)) + abs(len(a) - len(b))


def margin(scene: dict) -> float:
    program = PROGRAMS[scene["symbol"]]
    distances = sorted(_hamming(program, candidate) for candidate in PROGRAMS.values())
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # The latent parametrization is injective over the rendered scene fields.
    return []
