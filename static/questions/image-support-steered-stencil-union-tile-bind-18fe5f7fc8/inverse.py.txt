"""Pixel-only inverse arm for stencil_union_tile_bind.

Receives the final PNG and the public question contract only.  It segments the
flat palette, recovers every framed 4x4 grid from its frame colour, reads the
16 cell states by geometry, forms the union of the two slate stencils and
returns the frame colour of the unique tile matching that union.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "stencil_union_tile_bind-oracle-0.1.0"
ABSTAIN = "abstain"

_BG = (247, 243, 232)
_SLATE = (54, 66, 84)
_INK = (18, 18, 18)
_HUES = {
    "red": (196, 52, 44),
    "blue": (38, 92, 184),
    "green": (36, 140, 74),
    "purple": (128, 58, 168),
}

_GRID = 4
_MIN_SIDE = 120
_MAX_SIDE = 170


def _components(mask: np.ndarray):
    """Label 4-connected components with an iterative flood fill."""
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out = []
    ys, xs = np.nonzero(mask)
    for sy, sx in zip(ys.tolist(), xs.tolist()):
        if seen[sy, sx]:
            continue
        stack = [(sy, sx)]
        seen[sy, sx] = True
        y0 = y1 = sy
        x0 = x1 = sx
        size = 0
        while stack:
            cy, cx = stack.pop()
            size += 1
            if cy < y0:
                y0 = cy
            if cy > y1:
                y1 = cy
            if cx < x0:
                x0 = cx
            if cx > x1:
                x1 = cx
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        out.append((y0, x0, y1, x1, size))
    return out


def _read_grid(arr: np.ndarray, box) -> int | None:
    y0, x0, y1, x1 = box
    side = y1 - y0 + 1
    if abs((x1 - x0 + 1) - side) > 3:
        return None
    inner = side - 8
    if inner <= 0 or inner % _GRID:
        # tolerate a small frame-width mismatch by re-deriving the cell pitch
        inner = side - 8
    step = inner / _GRID
    if step < 20:
        return None
    pattern = 0
    for index in range(16):
        col, row = index % _GRID, index // _GRID
        cx = int(round(x0 + 4 + (col + 0.5) * step))
        cy = int(round(y0 + 4 + (row + 0.5) * step))
        if not (0 <= cy < arr.shape[0] and 0 <= cx < arr.shape[1]):
            return None
        pixel = tuple(int(v) for v in arr[cy, cx])
        if pixel == _INK:
            pattern |= 1 << index
        elif pixel != _BG:
            return None
    return pattern


def _grids_for(arr: np.ndarray, rgb):
    mask = np.all(arr == np.array(rgb, dtype=arr.dtype), axis=-1)
    found = []
    for y0, x0, y1, x1, size in _components(mask):
        height = y1 - y0 + 1
        width = x1 - x0 + 1
        if not (_MIN_SIDE <= height <= _MAX_SIDE and _MIN_SIDE <= width <= _MAX_SIDE):
            continue
        # a frame is hollow: its ink is far below its bounding-box area
        if size > 0.6 * height * width:
            continue
        found.append((y0, x0, y1, x1))
    return found


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    if arr.ndim != 3 or arr.shape[0] < 200 or arr.shape[1] < 200:
        return ABSTAIN

    allowed = [_BG, _SLATE, _INK] + list(_HUES.values()) + [(222, 150, 40), (30, 150, 150)]
    known = np.zeros(arr.shape[:2], dtype=bool)
    for rgb in allowed:
        known |= np.all(arr == np.array(rgb, dtype=arr.dtype), axis=-1)
    if known.mean() < 0.999:
        return ABSTAIN

    stencil_boxes = _grids_for(arr, _SLATE)
    if len(stencil_boxes) != 2:
        return ABSTAIN
    union = 0
    for box in stencil_boxes:
        pattern = _read_grid(arr, box)
        if pattern is None or pattern == 0:
            return ABSTAIN
        union |= pattern

    tiles = {}
    for name, rgb in _HUES.items():
        boxes = _grids_for(arr, rgb)
        if len(boxes) != 1:
            return ABSTAIN
        pattern = _read_grid(arr, boxes[0])
        if pattern is None:
            return ABSTAIN
        tiles[name] = pattern
    if len(set(tiles.values())) != 4:
        return ABSTAIN

    hits = [name for name, pattern in tiles.items() if pattern == union]
    if len(hits) != 1:
        return ABSTAIN
    return hits[0]
