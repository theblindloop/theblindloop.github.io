"""Deterministic familiar-rainbow renderer with balanced radial color order."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 512
COLOR_NAMES = ("red", "orange", "yellow", "green", "blue")
COLORS = {
    "red": "#d62828",
    "orange": "#f47c20",
    "yellow": "#f6c744",
    "green": "#2a9d55",
    "blue": "#2468c9",
}
BACKGROUNDS = ("#eaf7ff", "#f1f6fb", "#e8f4f5")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x0A71C)
    outer_color = COLOR_NAMES[int(seed) % len(COLOR_NAMES)]
    inner = [name for name in COLOR_NAMES if name != outer_color]
    rng.shuffle(inner)
    return {
        "band_colors": [outer_color, *inner],
        "center_x": rng.randint(248, 264),
        "baseline_y": rng.randint(366, 382),
        "outer_radius": rng.randint(188, 198),
        "inner_radius": rng.randint(48, 58),
        "background_id": rng.randrange(len(BACKGROUNDS)),
        "cloud_order": ["left", "right"],
    }


def _radii(scene: dict) -> list[float]:
    """Six boundaries giving five equal-area semicircular annuli."""
    outer = float(scene["outer_radius"])
    inner = float(scene["inner_radius"])
    step = (outer * outer - inner * inner) / 5.0
    return [math.sqrt(outer * outer - i * step) for i in range(6)]


def _cloud(draw: ImageDraw.ImageDraw, x: float, y: float, scale: int) -> None:
    fill, outline = "#ffffff", "#a9c6d5"
    for dx, dy, radius in ((-35, -2, 30), (-8, -18, 37), (27, -3, 29)):
        box = tuple(round(v * scale) for v in (x + dx - radius, y + dy - radius, x + dx + radius, y + dy + radius))
        draw.ellipse(box, fill=fill, outline=outline, width=2 * scale)
    box = tuple(round(v * scale) for v in (x - 61, y - 8, x + 58, y + 28))
    draw.rounded_rectangle(box, radius=16 * scale, fill=fill, outline=outline, width=2 * scale)


def render(scene: dict) -> Image.Image:
    scale = 3
    bg = BACKGROUNDS[int(scene["background_id"]) % len(BACKGROUNDS)]
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), bg)
    draw = ImageDraw.Draw(image)
    cx = float(scene["center_x"])
    cy = float(scene["baseline_y"])
    radii = _radii(scene)
    colors = list(scene["band_colors"])
    if sorted(colors) != sorted(COLOR_NAMES):
        raise ValueError("band_colors must be a permutation of the five public colors")

    # Nested half-disks expose five equal-area bands despite different radii.
    for radius, name in zip(radii[:-1], colors):
        box = tuple(round(v * scale) for v in (cx - radius, cy - radius, cx + radius, cy + radius))
        draw.pieslice(box, start=180, end=360, fill=COLORS[name])
    radius = radii[-1]
    box = tuple(round(v * scale) for v in (cx - radius, cy - radius, cx + radius, cy + radius))
    draw.pieslice(box, start=180, end=360, fill=bg)

    # Familiar endpoint clouds leave the top radial evidence unobscured.
    positions = {
        "left": (cx - radii[0] + 19, cy + 1),
        "right": (cx + radii[0] - 19, cy + 1),
    }
    for key in scene["cloud_order"]:
        _cloud(draw, *positions[key], scale)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["band_colors"][0])


def margin(scene: dict) -> float:
    radii = _radii(scene)
    return float(min(radii[i] - radii[i + 1] for i in range(5)))


def is_quarantined(scene: dict) -> bool:
    outer = float(scene["outer_radius"])
    cx, cy = float(scene["center_x"]), float(scene["baseline_y"])
    return margin(scene) < 16.0 or cx - outer < 8 or cx + outer > WIDTH - 8 or cy - outer < 8 or cy > HEIGHT - 8


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["cloud_order"] = list(reversed(scene["cloud_order"]))
    return [("reverse_nonoverlapping_cloud_draw_order", twin)]
