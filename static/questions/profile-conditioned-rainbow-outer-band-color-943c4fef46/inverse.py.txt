"""Independent pixel-only inverse for the outermost rainbow-band color."""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageColor

_NAMES = ("red", "orange", "yellow", "green", "blue")
_RGB = np.array([ImageColor.getrgb(v) for v in ("#d62828", "#f47c20", "#f6c744", "#2a9d55", "#2468c9")], dtype=np.float32)


def measure(image: Image.Image) -> tuple[str, float]:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    distances = np.linalg.norm(arr[:, :, None, :] - _RGB[None, None, :, :], axis=3)
    nearest = distances.argmin(axis=2)
    best = distances.min(axis=2)
    top_y: list[int] = []
    for index, name in enumerate(_NAMES):
        ys, xs = np.nonzero((nearest == index) & (best < 58.0))
        if len(xs) < 900:
            raise ValueError(f"missing or corrupted {name} band")
        # Ignore small nearest-palette antialias fringes by using a low but
        # substantial support quantile rather than the first isolated pixel.
        top_y.append(int(np.quantile(ys, 0.01)))
    order = np.argsort(np.asarray(top_y))
    winner, runner = int(order[0]), int(order[1])
    gap = float(top_y[runner] - top_y[winner])
    if gap < 10.0:
        raise ValueError(f"ambiguous outer band; raster gap={gap}")
    return _NAMES[winner], gap


def decision_from_image(image: Image.Image) -> str:
    return measure(image)[0]
