"""Renderer and analytic world for global circular-flow coherence."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 960
HEIGHT = 340
PANEL_X = (160.0, 480.0, 800.0)
PANEL_Y = 170.0
WHITE = (255, 255, 255)
PANEL_EDGE = (211, 216, 220)
COLOR_RGB = {
    "red": (214, 61, 64),
    "blue": (48, 105, 192),
    "green": (49, 157, 95),
}
COLORS = tuple(COLOR_RGB)
MIN_MARGIN = 8.0


def _flow_center(points: list[list[float]], angles: list[float]) -> tuple[float, float]:
    """Fit the point where all line axes are normal to their radius vectors."""
    a00 = a01 = a11 = b0 = b1 = 0.0
    for point, angle in zip(points, angles):
        ux = math.cos(angle)
        uy = math.sin(angle)
        px, py = float(point[0]), float(point[1])
        a00 += ux * ux
        a01 += ux * uy
        a11 += uy * uy
        projection = ux * px + uy * py
        b0 += ux * projection
        b1 += uy * projection
    determinant = a00 * a11 - a01 * a01
    if determinant < 1e-8:
        raise ValueError("dash axes do not constrain a two-dimensional center")
    return (
        (b0 * a11 - b1 * a01) / determinant,
        (a00 * b1 - a01 * b0) / determinant,
    )


def _field_score(_unused: object, field: dict[str, Any]) -> float:
    points = field["dash_points"]
    angles = field["dash_angles"]
    center_x, center_y = _flow_center(points, angles)
    residuals = []
    for point, angle in zip(points, angles):
        dx = float(point[0]) - center_x
        dy = float(point[1]) - center_y
        radius = math.hypot(dx, dy)
        if radius < 1e-6:
            return -1.0
        residuals.append(abs(math.cos(angle) * dx + math.sin(angle) * dy) / radius)
    return 1.0 - sum(residuals) / len(residuals)


def _scores(scene: dict[str, Any]) -> list[tuple[str, float]]:
    return [(field["color"], _field_score(None, field)) for field in scene["fields"]]


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one balanced scene; the coherent color cycles independently of slot."""
    rng = random.Random(int(seed))
    count = rng.choice((16, 18))
    phase = rng.uniform(0.0, 2.0 * math.pi)
    nominal_radius = rng.uniform(78.0, 84.0)
    radii = [nominal_radius + rng.uniform(-3.0, 3.0) for _ in range(count)]
    point_angles = [
        phase + 2.0 * math.pi * index / count + rng.uniform(-0.025, 0.025)
        for index in range(count)
    ]
    length = round(rng.uniform(20.0, 23.0), 4)
    stroke = round(rng.uniform(5.2, 6.4), 4)

    coherent = [angle + math.pi / 2.0 for angle in point_angles]
    shift = max(3, count // 3)
    shifted = [coherent[(index + shift) % count] for index in range(count)]
    reversed_flow = [coherent[(shift - index) % count] for index in range(count)]
    role_angles = {
        "coherent": coherent,
        "shifted": shifted,
        "reversed": reversed_flow,
    }
    roles = list(role_angles)
    rng.shuffle(roles)

    coherent_color = COLORS[int(seed) % len(COLORS)]
    other_colors = [color for color in COLORS if color != coherent_color]
    rng.shuffle(other_colors)
    fields: list[dict[str, Any]] = []
    for slot, role in enumerate(roles):
        if role == "coherent":
            color = coherent_color
        else:
            color = other_colors.pop()
        center_x = PANEL_X[slot] + rng.uniform(-10.0, 10.0)
        center_y = PANEL_Y + rng.uniform(-9.0, 9.0)
        points = [
            [
                round(center_x + radius * math.cos(angle), 4),
                round(center_y + radius * math.sin(angle), 4),
            ]
            for radius, angle in zip(radii, point_angles)
        ]
        fields.append(
            {
                "slot": slot,
                "color": color,
                "dash_points": points,
                "dash_angles": [round(angle, 7) for angle in role_angles[role]],
                "length": length,
                "stroke": stroke,
            }
        )
    return {"fields": fields}


def render(scene: dict[str, Any], supersample: int = 4) -> Image.Image:
    scale = int(supersample)
    if scale < 1:
        raise ValueError("supersample must be positive")
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), WHITE)
    draw = ImageDraw.Draw(image)
    for center_x in PANEL_X:
        left = int(round((center_x - 132.0) * scale))
        top = int(round(38.0 * scale))
        right = int(round((center_x + 132.0) * scale))
        bottom = int(round(302.0 * scale))
        draw.rounded_rectangle(
            (left, top, right, bottom),
            radius=int(round(15.0 * scale)),
            outline=PANEL_EDGE,
            width=max(1, int(round(2.0 * scale))),
        )
    for field in sorted(scene["fields"], key=lambda item: int(item["slot"])):
        color = COLOR_RGB[field["color"]]
        half = float(field["length"]) / 2.0
        width = max(1, int(round(float(field["stroke"]) * scale)))
        for point, angle in zip(field["dash_points"], field["dash_angles"]):
            dx = half * math.cos(float(angle))
            dy = half * math.sin(float(angle))
            draw.line(
                (
                    (float(point[0]) - dx) * scale,
                    (float(point[1]) - dy) * scale,
                    (float(point[0]) + dx) * scale,
                    (float(point[1]) + dy) * scale,
                ),
                fill=color,
                width=width,
            )
    if scale != 1:
        image = image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    scores = sorted(_scores(scene), key=lambda item: (-item[1], item[0]))
    return scores[0][0]


def margin(scene: dict[str, Any]) -> float:
    scores = sorted((score for _, score in _scores(scene)), reverse=True)
    return round((scores[0] - scores[1]) * 100.0, 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        return margin(scene) < MIN_MARGIN
    except (ValueError, ZeroDivisionError):
        return True


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """The latent record order is irrelevant because slots are visible anchors."""
    return [("field_record_order", {"fields": list(reversed(scene["fields"]))})]
