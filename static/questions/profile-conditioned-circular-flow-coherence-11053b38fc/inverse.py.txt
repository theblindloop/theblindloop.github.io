"""Independent pixel-only inverse for circular-flow coherence."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


# These are public visual colors, repeated here rather than imported from the renderer.
PALETTE = {
    "red": (214, 61, 64),
    "blue": (48, 105, 192),
    "green": (49, 157, 95),
}
COLOR_TOLERANCE = 88.0
MIN_COMPONENT_PIXELS = 18


def _connected_components(mask: np.ndarray) -> list[np.ndarray]:
    ys, xs = np.nonzero(mask)
    if len(xs) == 0:
        return []
    y0, y1 = int(ys.min()), int(ys.max()) + 1
    x0, x1 = int(xs.min()), int(xs.max()) + 1
    local = mask[y0:y1, x0:x1]
    visited = np.zeros(local.shape, dtype=bool)
    components: list[np.ndarray] = []
    height, width = local.shape
    for yy, xx in zip(*np.nonzero(local)):
        if visited[yy, xx]:
            continue
        visited[yy, xx] = True
        stack = [(int(yy), int(xx))]
        points: list[tuple[float, float]] = []
        while stack:
            cy, cx = stack.pop()
            points.append((float(cx + x0), float(cy + y0)))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and local[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= MIN_COMPONENT_PIXELS:
            components.append(np.asarray(points, dtype=np.float64))
    return components


def _dash_features(image: Image.Image, color: tuple[int, int, int]) -> tuple[np.ndarray, np.ndarray]:
    pixels = np.asarray(image.convert("RGB"), dtype=np.float64)
    distance = np.sqrt(np.sum((pixels - np.asarray(color, dtype=np.float64)) ** 2, axis=2))
    components = _connected_components(distance <= COLOR_TOLERANCE)
    centers: list[np.ndarray] = []
    directions: list[np.ndarray] = []
    for component in components:
        center = component.mean(axis=0)
        centered = component - center
        covariance = centered.T @ centered / max(1, len(component) - 1)
        values, vectors = np.linalg.eigh(covariance)
        direction = vectors[:, int(np.argmax(values))]
        if values.max() < 2.0:
            continue
        centers.append(center)
        directions.append(direction / max(np.linalg.norm(direction), 1e-12))
    if len(centers) < 12:
        raise ValueError("not enough separated dash components for a field")
    # A clean scene has 16 or 18 dashes. Keep the largest components if a
    # threshold creates a tiny antialias fragment in a pathological raster.
    if len(centers) > 20:
        ranked = sorted(zip(components, centers, directions), key=lambda item: len(item[0]), reverse=True)[:20]
        centers = [item[1] for item in ranked]
        directions = [item[2] for item in ranked]
    return np.asarray(centers), np.asarray(directions)


def _coherence_score(points: np.ndarray, directions: np.ndarray) -> float:
    a = directions
    b = np.sum(directions * points, axis=1)
    center, _, _, _ = np.linalg.lstsq(a, b, rcond=None)
    radial = points - center
    radii = np.linalg.norm(radial, axis=1)
    if np.any(radii < 1.0):
        raise ValueError("dash field center is degenerate")
    residual = np.abs(np.sum(directions * radial, axis=1)) / radii
    return float(1.0 - np.mean(residual))


def decision_from_image(image: Image.Image) -> str:
    """Return the color with the highest fitted whole-field flow coherence."""
    scores = {}
    for name, color in PALETTE.items():
        points, directions = _dash_features(image, color)
        scores[name] = _coherence_score(points, directions)
    return max(scores, key=lambda name: (scores[name], name))
