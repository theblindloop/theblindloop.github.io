"""Deterministic renderer and analytic arm for the magnifier-prior world."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
SUPERSAMPLE = 4
DIRECTIONS = ("down-right", "down-left")
MIN_SIDE_HEADROOM_PX = 48.0

_INK_PALETTE = (
    (31, 55, 83),
    (70, 43, 78),
    (39, 72, 62),
    (67, 57, 47),
)
_BACKGROUNDS = (
    (244, 246, 241),
    (246, 242, 234),
    (238, 244, 247),
    (243, 239, 247),
)
_ACCENTS = (
    (232, 158, 55),
    (55, 177, 196),
    (224, 113, 126),
    (126, 177, 78),
)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one deterministic member of a style-matched mirrored pair."""

    seed = int(seed)
    rng = random.Random(((seed // 2) * 1_000_003) ^ 0x6D41474E)
    return {
        "direction": DIRECTIONS[seed % 2],
        "center_x": round(rng.uniform(232.0, 280.0), 3),
        "center_y": round(rng.uniform(174.0, 212.0), 3),
        "lens_radius": round(rng.uniform(82.0, 102.0), 3),
        "ring_width": rng.choice((15, 17, 19)),
        "handle_length": round(rng.uniform(118.0, 148.0), 3),
        "handle_width": rng.choice((27, 30, 33)),
        "ink_rgb": list(rng.choice(_INK_PALETTE)),
        "background_rgb": list(rng.choice(_BACKGROUNDS)),
        "accent_rgb": list(rng.choice(_ACCENTS)),
        "marker_phase": rng.randrange(4),
    }


def _handle_geometry(scene: dict[str, Any]) -> tuple[tuple[float, float], tuple[float, float]]:
    direction = str(scene["direction"])
    if direction not in DIRECTIONS:
        raise ValueError(f"invalid direction {direction!r}")
    sign = 1.0 if direction == "down-right" else -1.0
    unit_x = sign / math.sqrt(2.0)
    unit_y = 1.0 / math.sqrt(2.0)
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    radius = float(scene["lens_radius"])
    length = float(scene["handle_length"])
    start_d = radius * 0.72
    end_d = radius + length
    return (
        (cx + unit_x * start_d, cy + unit_y * start_d),
        (cx + unit_x * end_d, cy + unit_y * end_d),
    )


def _draw_marker(draw: ImageDraw.ImageDraw, scene: dict[str, Any]) -> None:
    """Draw one answer-independent bright marker with a periodic latent phase."""

    phase = int(scene["marker_phase"]) % 4
    positions = ((456, 256), (256, 456), (56, 256), (256, 56))
    x, y = positions[phase]
    scale = SUPERSAMPLE
    radius = 8 * scale
    fill = tuple(int(value) for value in scene["accent_rgb"])
    draw.ellipse(
        (x * scale - radius, y * scale - radius, x * scale + radius, y * scale + radius),
        fill=fill,
    )


def render(scene: dict[str, Any]) -> Image.Image:
    """Render a hollow lens ring with one attached diagonal handle."""

    scale = SUPERSAMPLE
    background = tuple(int(value) for value in scene["background_rgb"])
    ink = tuple(int(value) for value in scene["ink_rgb"])
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), background)
    draw = ImageDraw.Draw(image)
    _draw_marker(draw, scene)

    (sx, sy), (ex, ey) = _handle_geometry(scene)
    handle_width = int(scene["handle_width"]) * scale
    draw.line((sx * scale, sy * scale, ex * scale, ey * scale), fill=ink, width=handle_width)
    cap_r = handle_width / 2.0
    for x, y in ((sx * scale, sy * scale), (ex * scale, ey * scale)):
        draw.ellipse((x - cap_r, y - cap_r, x + cap_r, y + cap_r), fill=ink)

    cx = float(scene["center_x"]) * scale
    cy = float(scene["center_y"]) * scale
    radius = float(scene["lens_radius"]) * scale
    ring_width = int(scene["ring_width"]) * scale
    draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=ink)
    inner = radius - ring_width
    if inner <= 8 * scale:
        raise ValueError("ring leaves no measurable lens opening")
    draw.ellipse((cx - inner, cy - inner, cx + inner, cy + inner), fill=background)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    """Read only the visible handle-direction field permitted as label input."""

    value = str(scene["direction"])
    if value not in DIRECTIONS:
        raise ValueError(f"invalid direction {value!r}")
    return value


def margin(scene: dict[str, Any]) -> float:
    """Analytic lateral headroom of the handle midpoint over the ambiguity floor."""

    radius = float(scene["lens_radius"])
    length = float(scene["handle_length"])
    midpoint_distance = radius + length * 0.5
    return midpoint_distance / math.sqrt(2.0) - MIN_SIDE_HEADROOM_PX


def is_quarantined(scene: dict[str, Any]) -> bool:
    if margin(scene) <= 0.0:
        return True
    (_, _), (ex, ey) = _handle_geometry(scene)
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    radius = float(scene["lens_radius"])
    half_handle = float(scene["handle_width"]) / 2.0
    pad = 8.0
    return bool(
        cx - radius < pad
        or cx + radius > WIDTH - pad
        or cy - radius < pad
        or ex - half_handle < pad
        or ex + half_handle > WIDTH - pad
        or ey + half_handle > HEIGHT - pad
    )


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """Exercise the natural four-turn periodic alias of the marker phase."""

    twin = dict(scene)
    twin["marker_phase"] = int(scene["marker_phase"]) + 4
    return [("marker_phase_plus_full_period", twin)]
