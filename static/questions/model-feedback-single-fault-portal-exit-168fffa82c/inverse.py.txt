"""Independent image-only inverse for the hatch-robustness map."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

WIDTH, HEIGHT = 900, 720
ROWS, COLS, CELL = 13, 17, 38
ORIGIN_X, ORIGIN_Y = 127, 176
OPEN_RGB = (248, 250, 252)
WALL_RGB = (35, 50, 69)
START_RGB = (18, 150, 94)
HATCH_RGB = (128, 137, 151)
EXIT_RGB = {
    "amber": (230, 146, 27), "cyan": (8, 145, 178),
    "violet": (132, 70, 190), "lime": (87, 164, 57),
}
PORTAL_RGB = {
    "A": (45, 100, 210), "B": (205, 60, 135), "C": (220, 80, 40),
    "D": (0, 120, 105), "E": (125, 95, 25),
}


class OracleError(ValueError):
    pass


def _near(patch: np.ndarray, rgb: tuple[int, int, int], radius: float = 28.0) -> int:
    delta = patch.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return int((np.sum(delta * delta, axis=2, dtype=np.int32) <= radius * radius).sum())


def _recover(image: Image.Image) -> tuple[
    set[tuple[int, int]], tuple[int, int], dict[str, tuple[int, int]],
    dict[str, tuple[tuple[int, int], tuple[int, int]]], tuple[tuple[int, int], ...],
]:
    if image.size != (WIDTH, HEIGHT):
        raise OracleError(f"expected {(WIDTH, HEIGHT)} image, got {image.size}")
    arr = np.asarray(image.convert("RGB"))
    palette = {
        "start": START_RGB, "hatch": HATCH_RGB,
        **{f"exit:{name}": rgb for name, rgb in EXIT_RGB.items()},
        **{f"portal:{name}": rgb for name, rgb in PORTAL_RGB.items()},
    }
    found: dict[str, list[tuple[int, int]]] = {name: [] for name in palette}
    open_cells: set[tuple[int, int]] = set()
    for row in range(ROWS):
        for col in range(COLS):
            cx = ORIGIN_X + col * CELL + CELL // 2
            cy = ORIGIN_Y + row * CELL + CELL // 2
            patch = arr[cy - 11:cy + 12, cx - 11:cx + 12]
            scores = {name: _near(patch, rgb) for name, rgb in palette.items()}
            best = max(scores, key=scores.get)
            ranked = sorted(scores.values(), reverse=True)
            if scores[best] >= 105 and scores[best] >= 2 * max(1, ranked[1]):
                found[best].append((row, col))
                open_cells.add((row, col))
                continue
            open_score, wall_score = _near(patch, OPEN_RGB), _near(patch, WALL_RGB)
            if open_score >= 430 and open_score >= 5 * max(1, wall_score):
                open_cells.add((row, col))
            elif wall_score >= 430 and wall_score >= 5 * max(1, open_score):
                continue
            else:
                raise OracleError(f"cell {(row, col)} has no stable interior class")
    if len(found["start"]) != 1 or len(found["hatch"]) != 4:
        raise OracleError("expected one start and four hatches")
    exits: dict[str, tuple[int, int]] = {}
    for color in EXIT_RGB:
        values = found[f"exit:{color}"]
        if len(values) != 1:
            raise OracleError(f"expected one {color} exit, found {len(values)}")
        exits[color] = values[0]
    portals: dict[str, tuple[tuple[int, int], tuple[int, int]]] = {}
    for letter in PORTAL_RGB:
        values = found[f"portal:{letter}"]
        if len(values) != 2:
            raise OracleError(f"expected two {letter} portals, found {len(values)}")
        portals[letter] = (values[0], values[1])
    return open_cells, found["start"][0], exits, portals, tuple(found["hatch"])


def _reachable(open_cells: set[tuple[int, int]], start: tuple[int, int],
               target: tuple[int, int],
               portals: dict[str, tuple[tuple[int, int], tuple[int, int]]],
               blocked: tuple[int, int]) -> bool:
    jumps: dict[tuple[int, int], tuple[int, int]] = {}
    for first, second in portals.values():
        jumps[first], jumps[second] = second, first
    seen, queue = {start}, deque([start])
    while queue:
        row, col = queue.popleft()
        if (row, col) == target:
            return True
        neighbors = [(row - 1, col), (row + 1, col),
                     (row, col - 1), (row, col + 1)]
        if (row, col) in jumps:
            neighbors.append(jumps[(row, col)])
        for neighbor in neighbors:
            if neighbor in open_cells and neighbor != blocked and neighbor not in seen:
                seen.add(neighbor)
                queue.append(neighbor)
    return False


def decision_from_image(image: Image.Image) -> str:
    open_cells, start, exits, portals, hatches = _recover(image)
    answers = [
        color for color, target in exits.items()
        if all(_reachable(open_cells, start, target, portals, hatch) for hatch in hatches)
    ]
    if len(answers) != 1:
        raise OracleError(f"expected one single-fault-safe exit, found {answers}")
    return answers[0]
