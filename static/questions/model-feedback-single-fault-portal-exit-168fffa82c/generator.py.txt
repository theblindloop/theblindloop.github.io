"""Render a portal corridor whose exits have different single-hatch robustness."""

from __future__ import annotations

import random
from collections import deque
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 900
HEIGHT = 720
SCALE = 3
ROWS = 13
COLS = 17
CELL = 38
ORIGIN_X = 127
ORIGIN_Y = 176

OPEN_RGB = (248, 250, 252)
WALL_RGB = (35, 50, 69)
GRID_RGB = (94, 108, 125)
INK_RGB = (24, 34, 47)
MUTED_RGB = (88, 102, 120)
START_RGB = (18, 150, 94)
HATCH_RGB = (128, 137, 151)
EXIT_RGB = {
    "amber": (230, 146, 27),
    "cyan": (8, 145, 178),
    "violet": (132, 70, 190),
    "lime": (87, 164, 57),
}
PORTAL_RGB = {
    "A": (45, 100, 210),
    "B": (205, 60, 135),
    "C": (220, 80, 40),
    "D": (0, 120, 105),
    "E": (125, 95, 25),
}
COLORS = tuple(EXIT_RGB)


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size * SCALE
    )


def _transform(cell: tuple[int, int], variant: int) -> tuple[int, int]:
    row, col = cell
    if variant & 1:
        col = COLS - 1 - col
    if variant & 2:
        row = ROWS - 1 - row
    return row, col


def _base_geometry() -> tuple[
    set[tuple[int, int]],
    tuple[int, int],
    list[tuple[int, int]],
    dict[str, tuple[tuple[int, int], tuple[int, int]]],
    list[tuple[int, int]],
]:
    paths = [
        [(6, 1), (6, 2), (6, 3)],
        # First redundant stage: upper route uses A, lower route is spatial.
        [(6, 3), (5, 3), (4, 3), (3, 3), (3, 4), (3, 5)],
        [(5, 7), (5, 8), (6, 8)],
        [(6, 3), (7, 3), (8, 3), (9, 3), (9, 4), (9, 5),
         (9, 6), (8, 6), (7, 6), (6, 6), (6, 7), (6, 8)],
        # Second redundant stage: upper route is spatial, lower route uses B.
        [(6, 8), (5, 8), (4, 8), (3, 8), (3, 9), (3, 10),
         (3, 11), (3, 12), (4, 12), (5, 12), (6, 12)],
        [(6, 8), (7, 8)],
        [(9, 10), (9, 11), (9, 12), (8, 12), (7, 12), (6, 12)],
        [(6, 12), (6, 13), (6, 14), (6, 15)],
        # Three remote exit pockets hang from different hatch junctions.
        [(3, 3), (2, 3), (1, 3)],
        [(1, 14), (1, 15), (1, 16)],
        [(9, 3), (10, 3), (11, 3)],
        [(11, 14), (11, 15), (11, 16)],
        [(3, 11), (2, 11), (1, 11)],
        [(11, 8), (11, 7), (11, 6)],
    ]
    open_cells = {cell for path in paths for cell in path}
    start = (6, 1)
    # Roles 0..2 each fail under a different hatch; role 3 survives all four.
    exit_roles = [(1, 16), (11, 16), (11, 6), (6, 15)]
    portals = {
        "A": ((3, 5), (5, 7)),
        "B": ((7, 8), (9, 10)),
        "C": ((1, 3), (1, 14)),
        "D": ((11, 3), (11, 14)),
        "E": ((1, 11), (11, 8)),
    }
    hatches = [(3, 3), (9, 3), (3, 11), (9, 11)]
    return open_cells, start, exit_roles, portals, hatches


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    open_cells, start, exit_roles, portal_roles, hatches = _base_geometry()
    variant = (int(seed) // 4) % 4
    transform = lambda cell: _transform(cell, variant)

    safe_color = COLORS[int(seed) % len(COLORS)]
    other_colors = [color for color in COLORS if color != safe_color]
    rng.shuffle(other_colors)
    role_colors = [*other_colors, safe_color]

    letters = list(PORTAL_RGB)
    shuffled_letters = letters.copy()
    rng.shuffle(shuffled_letters)
    portals = {
        shown: [list(transform(cell)) for cell in portal_roles[role]]
        for shown, role in zip(shuffled_letters, letters)
    }
    return {
        "open_cells": [list(transform(cell)) for cell in sorted(open_cells)],
        "start": list(transform(start)),
        "exits": {
            color: list(transform(cell))
            for color, cell in zip(role_colors, exit_roles)
        },
        "portals": portals,
        "hatches": [list(transform(cell)) for cell in hatches],
    }


def _cell(value: object) -> tuple[int, int] | None:
    if not isinstance(value, list) or len(value) != 2:
        return None
    if any(isinstance(item, bool) or not isinstance(item, int) for item in value):
        return None
    row, col = int(value[0]), int(value[1])
    return (row, col) if 0 <= row < ROWS and 0 <= col < COLS else None


def _decode(scene: dict[str, Any]) -> tuple[
    set[tuple[int, int]],
    tuple[int, int],
    dict[str, tuple[int, int]],
    dict[str, tuple[tuple[int, int], tuple[int, int]]],
    tuple[tuple[int, int], ...],
] | None:
    try:
        raw_open = scene["open_cells"]
        start = _cell(scene["start"])
        raw_exits = scene["exits"]
        raw_portals = scene["portals"]
        raw_hatches = scene["hatches"]
    except (KeyError, TypeError):
        return None
    if not isinstance(raw_open, list) or not isinstance(raw_exits, dict):
        return None
    if not isinstance(raw_portals, dict) or not isinstance(raw_hatches, list):
        return None
    converted = [_cell(value) for value in raw_open]
    if any(value is None for value in converted) or start is None:
        return None
    open_cells = {value for value in converted if value is not None}
    if len(open_cells) != len(converted) or set(raw_exits) != set(EXIT_RGB):
        return None
    exits = {color: _cell(raw_exits[color]) for color in EXIT_RGB}
    if any(cell is None for cell in exits.values()):
        return None
    clean_exits = {color: cell for color, cell in exits.items() if cell is not None}
    if set(raw_portals) != set(PORTAL_RGB):
        return None
    portals: dict[str, tuple[tuple[int, int], tuple[int, int]]] = {}
    for letter in PORTAL_RGB:
        pair = raw_portals[letter]
        if not isinstance(pair, list) or len(pair) != 2:
            return None
        cells = tuple(_cell(value) for value in pair)
        if any(cell is None for cell in cells) or cells[0] == cells[1]:
            return None
        portals[letter] = (cells[0], cells[1])  # type: ignore[arg-type]
    hatch_cells = tuple(_cell(value) for value in raw_hatches)
    if len(hatch_cells) != 4 or any(cell is None for cell in hatch_cells):
        return None
    clean_hatches = tuple(cell for cell in hatch_cells if cell is not None)
    specials = [start, *clean_exits.values(), *clean_hatches,
                *(cell for pair in portals.values() for cell in pair)]
    if len(specials) != len(set(specials)) or any(cell not in open_cells for cell in specials):
        return None
    return open_cells, start, clean_exits, portals, clean_hatches


def _reachable(
    open_cells: set[tuple[int, int]],
    start: tuple[int, int],
    target: tuple[int, int],
    portals: dict[str, tuple[tuple[int, int], tuple[int, int]]],
    blocked: tuple[int, int] | None = None,
) -> bool:
    jumps: dict[tuple[int, int], tuple[int, int]] = {}
    for first, second in portals.values():
        jumps[first] = second
        jumps[second] = first
    if start == blocked or target == blocked:
        return False
    seen = {start}
    queue = deque([start])
    while queue:
        row, col = queue.popleft()
        if (row, col) == target:
            return True
        neighbors = [(row - 1, col), (row + 1, col),
                     (row, col - 1), (row, col + 1)]
        if (row, col) in jumps:
            neighbors.append(jumps[(row, col)])
        for neighbor in neighbors:
            if neighbor in open_cells and neighbor != blocked and neighbor not in seen:
                seen.add(neighbor)
                queue.append(neighbor)
    return False


def _robust_colors(scene: dict[str, Any]) -> list[str]:
    decoded = _decode(scene)
    if decoded is None:
        return []
    open_cells, start, exits, portals, hatches = decoded
    return [
        color for color, target in exits.items()
        if all(_reachable(open_cells, start, target, portals, hatch) for hatch in hatches)
    ]


def analytic_gold(scene: dict[str, Any]) -> str:
    answers = _robust_colors(scene)
    if len(answers) != 1:
        raise ValueError(f"scene has {len(answers)} single-fault-safe exits")
    return answers[0]


def margin(scene: dict[str, Any]) -> float:
    return 14.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    decoded = _decode(scene)
    if decoded is None or len(_robust_colors(scene)) != 1:
        return True
    open_cells, start, exits, portals, hatches = decoded
    if any(not _reachable(open_cells, start, target, portals) for target in exits.values()):
        return True
    safe = _robust_colors(scene)[0]
    return any(
        all(_reachable(open_cells, start, target, portals, hatch) for hatch in hatches)
        for color, target in exits.items() if color != safe
    )


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict]]:
    return []


def _centered(draw: ImageDraw.ImageDraw, center: tuple[int, int], text: str,
              font: ImageFont.FreeTypeFont, fill: tuple[int, int, int] | str) -> None:
    cx, cy = center
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((cx * SCALE - (box[2] - box[0]) / 2,
               cy * SCALE - (box[3] - box[1]) / 2 - 2 * SCALE),
              text, font=font, fill=fill)


def render(scene: dict[str, Any]) -> Image.Image:
    decoded = _decode(scene)
    if decoded is None:
        raise ValueError("cannot render malformed scene")
    open_cells, start, exits, portals, hatches = decoded
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), (238, 243, 248))
    draw = ImageDraw.Draw(image)
    draw.text((32 * SCALE, 18 * SCALE), "PORTAL MAZE  •  SURVIVE ANY ONE HATCH CLOSURE",
              font=_font(21), fill=INK_RGB)
    draw.text((33 * SCALE, 54 * SCALE),
              "Test each gray X separately  |  touching cells connect  |  equal letters jump",
              font=_font(13), fill=MUTED_RGB)

    legend_y = 113
    legend = [
        (START_RGB, "S = start", "circle"),
        (HATCH_RGB, "gray X = hatch", "square"),
        (PORTAL_RGB["A"], "A=A = portal", "circle"),
        (EXIT_RGB["amber"], "colors = exits", "diamond"),
    ]
    for x, (color, label, shape) in zip((85, 285, 500, 710), legend):
        if shape == "square":
            draw.rounded_rectangle(((x - 13) * SCALE, (legend_y - 13) * SCALE,
                                    (x + 13) * SCALE, (legend_y + 13) * SCALE),
                                   radius=4 * SCALE, fill=color, outline=INK_RGB, width=2 * SCALE)
        elif shape == "diamond":
            draw.polygon([(x * SCALE, (legend_y - 15) * SCALE),
                          ((x + 15) * SCALE, legend_y * SCALE),
                          (x * SCALE, (legend_y + 15) * SCALE),
                          ((x - 15) * SCALE, legend_y * SCALE)],
                         fill=color, outline=INK_RGB)
        else:
            draw.ellipse(((x - 13) * SCALE, (legend_y - 13) * SCALE,
                          (x + 13) * SCALE, (legend_y + 13) * SCALE),
                         fill=color, outline=INK_RGB, width=2 * SCALE)
        draw.text(((x + 23) * SCALE, (legend_y - 10) * SCALE), label,
                  font=_font(11), fill=INK_RGB)

    left, top = ORIGIN_X * SCALE, ORIGIN_Y * SCALE
    right = (ORIGIN_X + COLS * CELL) * SCALE
    bottom = (ORIGIN_Y + ROWS * CELL) * SCALE
    draw.rectangle((left - 5 * SCALE, top - 5 * SCALE,
                    right + 5 * SCALE, bottom + 5 * SCALE),
                   fill=WALL_RGB, outline=INK_RGB, width=2 * SCALE)
    for row in range(ROWS):
        for col in range(COLS):
            x0 = (ORIGIN_X + col * CELL) * SCALE
            y0 = (ORIGIN_Y + row * CELL) * SCALE
            draw.rectangle((x0, y0, x0 + CELL * SCALE, y0 + CELL * SCALE),
                           fill=OPEN_RGB if (row, col) in open_cells else WALL_RGB)

    def center(cell: tuple[int, int]) -> tuple[int, int]:
        row, col = cell
        return ORIGIN_X + col * CELL + CELL // 2, ORIGIN_Y + row * CELL + CELL // 2

    for cell in hatches:
        cx, cy = center(cell)
        draw.rounded_rectangle(((cx - 15) * SCALE, (cy - 15) * SCALE,
                                (cx + 15) * SCALE, (cy + 15) * SCALE),
                               radius=5 * SCALE, fill=HATCH_RGB,
                               outline=INK_RGB, width=2 * SCALE)
        _centered(draw, (cx, cy), "X", _font(16), "white")
    for letter, pair in portals.items():
        for cell in pair:
            cx, cy = center(cell)
            draw.ellipse(((cx - 15) * SCALE, (cy - 15) * SCALE,
                          (cx + 15) * SCALE, (cy + 15) * SCALE),
                         fill=PORTAL_RGB[letter], outline=INK_RGB, width=2 * SCALE)
            _centered(draw, (cx, cy), letter, _font(15), "white")
    sx, sy = center(start)
    draw.ellipse(((sx - 16) * SCALE, (sy - 16) * SCALE,
                  (sx + 16) * SCALE, (sy + 16) * SCALE),
                 fill=START_RGB, outline=INK_RGB, width=2 * SCALE)
    _centered(draw, (sx, sy), "S", _font(17), "white")
    for color, cell in exits.items():
        cx, cy = center(cell)
        draw.polygon([(cx * SCALE, (cy - 17) * SCALE),
                      ((cx + 17) * SCALE, cy * SCALE),
                      (cx * SCALE, (cy + 17) * SCALE),
                      ((cx - 17) * SCALE, cy * SCALE)],
                     fill=EXIT_RGB[color], outline=INK_RGB)
        _centered(draw, (cx, cy), "E", _font(14), "white")

    for col in range(COLS + 1):
        x = (ORIGIN_X + col * CELL) * SCALE
        draw.line((x, top, x, bottom), fill=GRID_RGB, width=SCALE)
    for row in range(ROWS + 1):
        y = (ORIGIN_Y + row * CELL) * SCALE
        draw.line((left, y, right, y), fill=GRID_RGB, width=SCALE)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
