"""Pixel-only inverse arm for the proper graph-coloring search.

The implementation discovers colored node components, groups them into four
panels from relative gaps, confirms every visible edge using dark pixels, then
counts same-color endpoint violations. It accepts no latent data and shares no
renderer or prompt code.
"""

from __future__ import annotations

from collections import deque
from dataclasses import dataclass
from math import hypot

import numpy as np
from PIL import Image


@dataclass(frozen=True)
class Node:
    x: float
    y: float
    color: int
    area: int


def _components(mask: np.ndarray, color: int) -> list[Node]:
    height, width = mask.shape
    visited = np.zeros(mask.shape, dtype=np.bool_)
    nodes: list[Node] = []
    for y_value, x_value in np.argwhere(mask):
        y = int(y_value)
        x = int(x_value)
        if visited[y, x]:
            continue
        queue = deque([(x, y)])
        visited[y, x] = True
        area = 0
        x_sum = 0
        y_sum = 0
        while queue:
            px, py = queue.popleft()
            area += 1
            x_sum += px
            y_sum += py
            for nx, ny in ((px - 1, py), (px + 1, py), (px, py - 1), (px, py + 1)):
                if 0 <= nx < width and 0 <= ny < height and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    queue.append((nx, ny))
        if area >= 400:
            nodes.append(Node(x_sum / area, y_sum / area, color, area))
    return nodes


def _largest_gap_split(values: list[float]) -> float:
    ordered = sorted(values)
    if len(ordered) < 2:
        raise ValueError("not enough visible components")
    gaps = [(ordered[index + 1] - ordered[index], index) for index in range(len(ordered) - 1)]
    gap, index = max(gaps)
    if gap < 70:
        raise ValueError("candidate panels are not visibly separated")
    return (ordered[index] + ordered[index + 1]) / 2.0


def _dark_fraction(
    dark: np.ndarray,
    first: Node,
    second: Node,
    *,
    samples: int = 25,
) -> float:
    height, width = dark.shape
    hits = 0
    for step in range(samples):
        t = 0.24 + 0.52 * step / (samples - 1)
        x = int(round(first.x + t * (second.x - first.x)))
        y = int(round(first.y + t * (second.y - first.y)))
        left = max(0, x - 2)
        right = min(width, x + 3)
        top = max(0, y - 2)
        bottom = min(height, y + 3)
        if bool(dark[top:bottom, left:right].any()):
            hits += 1
    return hits / samples


def _panel_violations(nodes: list[Node], dark: np.ndarray) -> int:
    if len(nodes) != 8:
        raise ValueError("each panel must contain exactly eight colored nodes")
    if sum(node.color == 0 for node in nodes) != 4:
        raise ValueError("each panel must contain four red and four blue nodes")
    if any(not 650 <= node.area <= 1450 for node in nodes):
        raise ValueError("a colored mark is not a full visible node")

    edges: list[tuple[int, int]] = []
    for first_index in range(len(nodes)):
        for second_index in range(first_index + 1, len(nodes)):
            first = nodes[first_index]
            second = nodes[second_index]
            distance = hypot(first.x - second.x, first.y - second.y)
            if 70 <= distance <= 122 and _dark_fraction(dark, first, second) >= 0.82:
                edges.append((first_index, second_index))
    if len(edges) != 10:
        raise ValueError("each panel must expose exactly ten unbroken dark edges")
    degrees = [0] * 8
    for first, second in edges:
        degrees[first] += 1
        degrees[second] += 1
    if sorted(degrees) != [2, 2, 2, 2, 3, 3, 3, 3]:
        raise ValueError("recovered edge incidence is not the visible ladder graph")
    return sum(nodes[first].color == nodes[second].color for first, second in edges)


def decision_from_image(image: Image.Image) -> str:
    """Return A-D only for a unique, safely separated pixel-derived winner."""

    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    if pixels.ndim != 3 or pixels.shape[0] < 300 or pixels.shape[1] < 500:
        raise ValueError("raster is too small")
    red = (
        (pixels[:, :, 0] > 145)
        & (pixels[:, :, 0] - pixels[:, :, 1] > 50)
        & (pixels[:, :, 0] - pixels[:, :, 2] > 35)
    )
    blue = (
        (pixels[:, :, 2] > 135)
        & (pixels[:, :, 2] - pixels[:, :, 0] > 35)
        & (pixels[:, :, 2] - pixels[:, :, 1] > 20)
    )
    nodes = _components(red, 0) + _components(blue, 1)
    if len(nodes) != 32:
        raise ValueError("expected exactly thirty-two visible colored nodes")

    x_split = _largest_gap_split([node.x for node in nodes])
    y_split = _largest_gap_split([node.y for node in nodes])
    grouped = [[], [], [], []]
    for node in nodes:
        column = int(node.x > x_split)
        row = int(node.y > y_split)
        grouped[row * 2 + column].append(node)

    dark = (
        (pixels[:, :, 0] < 100)
        & (pixels[:, :, 1] < 100)
        & (pixels[:, :, 2] < 105)
    )
    violations = [_panel_violations(panel, dark) for panel in grouped]
    winners = [index for index, count in enumerate(violations) if count == 0]
    if len(winners) != 1:
        raise ValueError("pixel evidence has no unique proper coloring")
    ordered = sorted(violations)
    if ordered[1] - ordered[0] < 2:
        raise ValueError("pixel-derived runner-up margin is unsafe")
    return ("A", "B", "C", "D")[winners[0]]
