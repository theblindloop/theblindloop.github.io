"""Deterministic balanced graph-coloring scenes rendered with Pillow."""

from __future__ import annotations

import itertools
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1000
HEIGHT = 620
LABELS = ("A", "B", "C", "D")
EDGES = (
    (0, 1), (1, 2), (2, 3),
    (4, 5), (5, 6), (6, 7),
    (0, 4), (1, 5), (2, 6), (3, 7),
)
PROPER = (0, 1, 0, 1, 1, 0, 1, 0)
PALETTES = (
    ((218, 52, 62), (42, 105, 207)),
    ((197, 42, 76), (34, 121, 196)),
    ((226, 73, 45), (52, 89, 190)),
)


def _violations(pattern: Iterable[int]) -> int:
    values = tuple(int(value) for value in pattern)
    return sum(values[left] == values[right] for left, right in EDGES)


def _balanced_patterns() -> list[tuple[int, ...]]:
    patterns: list[tuple[int, ...]] = []
    for ones in itertools.combinations(range(8), 4):
        values = [0] * 8
        for index in ones:
            values[index] = 1
        patterns.append(tuple(values))
    return patterns


def sample_scene(seed: int) -> dict:
    """Sample one four-panel scene with a counterbalanced answer position."""

    seed = int(seed)
    rng = random.Random(seed ^ 0x5A17C0DE)
    target = PROPER if ((seed // 4) % 2 == 0) else tuple(1 - bit for bit in PROPER)
    target_index = seed % 4
    requested_margin = 2 + ((seed // 4) % 4)

    invalid = [
        pattern
        for pattern in _balanced_patterns()
        if _violations(pattern) >= requested_margin and _violations(pattern) != 0
    ]
    anchors = [pattern for pattern in invalid if _violations(pattern) == requested_margin]
    rng.shuffle(invalid)
    rng.shuffle(anchors)
    distractors = [anchors[0]]
    for pattern in invalid:
        if pattern not in distractors:
            distractors.append(pattern)
        if len(distractors) == 3:
            break
    rng.shuffle(distractors)

    patterns: list[list[int]] = []
    distractor_index = 0
    for panel_index in range(4):
        if panel_index == target_index:
            patterns.append(list(target))
        else:
            patterns.append(list(distractors[distractor_index]))
            distractor_index += 1

    return {
        "patterns": patterns,
        "palette": seed % len(PALETTES),
        "offset_x": ((seed // 3) % 3 - 1) * 4,
        "offset_y": ((seed // 9) % 3 - 1) * 3,
    }


def _panel_boxes(offset_x: int, offset_y: int) -> list[tuple[int, int, int, int]]:
    lefts = (34 + offset_x, 518 + offset_x)
    tops = (92 + offset_y, 354 + offset_y)
    return [(x, y, x + 448, y + 235) for y in tops for x in lefts]


def _node_centers(box: tuple[int, int, int, int]) -> list[tuple[int, int]]:
    left, top, _, _ = box
    xs = [left + 74 + 100 * column for column in range(4)]
    return [(x, top + y) for y in (78, 170) for x in xs]


def render(scene: dict) -> Image.Image:
    """Render a complete RGB raster; all label inputs are visibly expressed."""

    patterns = scene["patterns"]
    palette_index = int(scene["palette"])
    offset_x = int(scene["offset_x"])
    offset_y = int(scene["offset_y"])
    red, blue = PALETTES[palette_index]

    scale = 2
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), (247, 247, 244))
    draw = ImageDraw.Draw(image)

    def box(values: tuple[int, int, int, int]) -> tuple[int, int, int, int]:
        return tuple(int(value * scale) for value in values)

    def point(values: tuple[int, int]) -> tuple[int, int]:
        return tuple(int(value * scale) for value in values)

    title_font = ImageFont.load_default(size=27 * scale)
    label_font = ImageFont.load_default(size=28 * scale)
    subtitle_font = ImageFont.load_default(size=17 * scale)
    draw.text(
        point((WIDTH // 2, 22)),
        "EDGE-COLOR CHECK",
        fill=(31, 34, 39),
        font=title_font,
        anchor="ma",
    )
    draw.text(
        point((WIDTH // 2, 59)),
        "Every dark line must join different colors",
        fill=(75, 78, 84),
        font=subtitle_font,
        anchor="ma",
    )

    for panel_index, (panel_box, pattern) in enumerate(
        zip(_panel_boxes(offset_x, offset_y), patterns, strict=True)
    ):
        draw.rounded_rectangle(
            box(panel_box),
            radius=16 * scale,
            fill=(255, 255, 253),
            outline=(151, 156, 164),
            width=3 * scale,
        )
        left, top, _, _ = panel_box
        draw.text(
            point((left + 18, top + 11)),
            LABELS[panel_index],
            fill=(27, 30, 35),
            font=label_font,
            anchor="la",
        )
        centers = _node_centers(panel_box)
        for first, second in EDGES:
            draw.line(
                (point(centers[first]), point(centers[second])),
                fill=(45, 48, 53),
                width=8 * scale,
            )
        for node_index, center in enumerate(centers):
            radius = 19
            cx, cy = center
            draw.ellipse(
                box((cx - radius, cy - radius, cx + radius, cy + radius)),
                fill=red if int(pattern[node_index]) == 0 else blue,
                outline=(255, 255, 253),
                width=3 * scale,
            )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS).convert("RGB")


def _violation_counts(patterns: Iterable[Iterable[int]]) -> list[int]:
    return [_violations(pattern) for pattern in patterns]


def analytic_gold(scene: dict) -> str:
    counts = _violation_counts(scene["patterns"])
    winners = [index for index, count in enumerate(counts) if count == 0]
    if len(winners) != 1:
        raise ValueError("scene does not have a unique proper coloring")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    counts = sorted(_violation_counts(scene["patterns"]))
    if len(counts) < 2:
        return 0.0
    return float(counts[1] - counts[0])


def is_quarantined(scene: dict) -> bool:
    patterns = scene["patterns"]
    if len(patterns) != 4 or any(len(pattern) != 8 for pattern in patterns):
        return True
    if any(sum(int(value) for value in pattern) != 4 for pattern in patterns):
        return True
    counts = _violation_counts(patterns)
    return counts.count(0) != 1 or margin(scene) < 2.0
