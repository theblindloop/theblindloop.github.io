"""Deterministic latent renderer for an occluded checker-phase field."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw


def _cells(scene: dict) -> list[tuple[int, int]]:
    rows = int(scene["rows"])
    side = int(scene["side_cols"])
    hidden = int(scene["hidden_cols"])
    left_phase = int(scene["left_phase"])
    flip = 1 if scene["relation"] == "opposite" else 0
    cells: list[tuple[int, int]] = []
    for row in range(rows):
        for col in range(side):
            if (row + col + left_phase) % 2 == 0:
                cells.append((row, col))
        for local_col in range(side):
            col = side + hidden + local_col
            if (row + col + left_phase + flip) % 2 == 0:
                cells.append((row, col))
    return cells


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    rows = rng.choice([8, 10, 12])
    side_cols = rng.choice([4, 6])
    hidden_cols = rng.choice([3, 4])
    pitch = rng.choice([25, 26, 27])
    radius = rng.choice([7, 8, 9])
    total_cols = 2 * side_cols + hidden_cols
    span_x = (total_cols - 1) * pitch
    span_y = (rows - 1) * pitch
    ox = (512 - span_x) // 2 + rng.randint(-6, 6)
    oy = (512 - span_y) // 2 + rng.randint(-6, 6)
    relation = "same" if int(seed) % 2 else "opposite"
    scene = {
        "rows": rows,
        "side_cols": side_cols,
        "hidden_cols": hidden_cols,
        "pitch": pitch,
        "origin": [ox, oy],
        "radius": radius,
        "left_phase": rng.randrange(2),
        "relation": relation,
        "turn": rng.randrange(4),
        "mirror": bool(rng.randrange(2)),
        "canvas": [512, 512],
        "palette": {"background": "#f8f5ee", "ink": "#17191d", "screen": "#b8bdc4", "screen_edge": "#737a84"},
    }
    order = list(range(len(_cells(scene))))
    rng.shuffle(order)
    scene["mark_order"] = order
    return scene


def render(scene: dict) -> Image.Image:
    width, height = map(int, scene["canvas"])
    palette = scene["palette"]
    image = Image.new("RGB", (width, height), palette["background"])
    draw = ImageDraw.Draw(image)
    ox, oy = map(int, scene["origin"])
    pitch = int(scene["pitch"])
    radius = int(scene["radius"])
    cells = _cells(scene)
    order = list(map(int, scene["mark_order"]))
    if sorted(order) != list(range(len(cells))):
        raise ValueError("mark_order is not a permutation")
    for index in order:
        row, col = cells[index]
        x, y = ox + col * pitch, oy + row * pitch
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=palette["ink"])

    side = int(scene["side_cols"])
    hidden = int(scene["hidden_cols"])
    x0 = ox + (side - 0.5) * pitch
    x1 = ox + (side + hidden - 0.5) * pitch
    y0 = oy - pitch * 0.62
    y1 = oy + (int(scene["rows"]) - 1) * pitch + pitch * 0.62
    draw.rounded_rectangle((round(x0), round(y0), round(x1), round(y1)), radius=7, fill=palette["screen"], outline=palette["screen_edge"], width=3)
    # Three neutral seams make the screen visibly opaque, not an empty gap.
    for fraction in (0.28, 0.5, 0.72):
        x = round(x0 + fraction * (x1 - x0))
        draw.line((x, round(y0 + 10), x, round(y1 - 10)), fill=palette["screen_edge"], width=1)

    if bool(scene["mirror"]):
        image = image.transpose(Image.Transpose.FLIP_LEFT_RIGHT)
    turns = int(scene["turn"]) % 4
    if turns:
        image = image.rotate(90 * turns, expand=False)
    return image


def analytic_gold(scene: dict) -> str:
    side = int(scene["side_cols"])
    hidden = int(scene["hidden_cols"])
    left = []
    right = []
    for row, col in _cells(scene):
        (left if col < side else right).append((row + col) % 2)
    if not left or not right or len(set(left)) != 1 or len(set(right)) != 1:
        raise ValueError("invalid checker field")
    return "same" if left[0] == right[0] else "opposite"


def margin(scene: dict) -> float:
    return float(min(len(_cells(scene)) // 2, int(scene["rows"]) * int(scene["side_cols"]) // 2))


def is_quarantined(scene: dict) -> bool:
    return int(scene["pitch"]) <= 2 * int(scene["radius"]) + 5 or int(scene["hidden_cols"]) < 2


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["mark_order"] = list(reversed(twin["mark_order"]))
    return [("reverse_disjoint_dot_painter_order", twin)]
