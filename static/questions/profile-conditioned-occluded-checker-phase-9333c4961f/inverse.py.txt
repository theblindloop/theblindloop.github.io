"""Independent PNG-only inverse for occluded checker phase."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[list[tuple[int, int]]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))]); seen[y, x] = True; group = []
        while queue:
            yy, xx = queue.popleft(); group.append((yy, xx))
            for dy, dx in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                ny, nx = yy + dy, xx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True; queue.append((ny, nx))
        groups.append(group)
    return groups


def _cluster(values: list[float], tolerance: float = 4.0) -> list[float]:
    centers: list[list[float]] = []
    for value in sorted(values):
        if not centers or value - sum(centers[-1]) / len(centers[-1]) > tolerance:
            centers.append([value])
        else:
            centers[-1].append(value)
    return [sum(group) / len(group) for group in centers]


def _nearest(value: float, centers: list[float]) -> int:
    return min(range(len(centers)), key=lambda index: abs(value - centers[index]))


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"))
    # Normalize the long screen axis to vertical using its neutral-gray mask.
    gray = (np.max(rgb, axis=2) - np.min(rgb, axis=2) < 20) & (rgb[:, :, 0] > 90) & (rgb[:, :, 0] < 210)
    ys, xs = np.nonzero(gray)
    if len(xs) < 1000:
        raise ValueError("screen not recovered")
    if (xs.max() - xs.min()) > (ys.max() - ys.min()):
        image = image.transpose(Image.Transpose.ROTATE_90)
        rgb = np.asarray(image.convert("RGB"))
        gray = (np.max(rgb, axis=2) - np.min(rgb, axis=2) < 20) & (rgb[:, :, 0] > 90) & (rgb[:, :, 0] < 210)
        ys, xs = np.nonzero(gray)
    screen_left, screen_right = float(xs.min()), float(xs.max())
    if screen_right - screen_left < 45 or ys.max() - ys.min() < 150:
        raise ValueError("invalid screen geometry")

    black = np.max(rgb, axis=2) < 75
    groups = [group for group in _components(black) if 80 <= len(group) <= 400]
    points = [(sum(x for y, x in group) / len(group), sum(y for y, x in group) / len(group)) for group in groups]
    if len(points) < 24:
        raise ValueError("too few dot components")
    left = [point for point in points if point[0] < screen_left - 5]
    right = [point for point in points if point[0] > screen_right + 5]
    if len(left) != len(right) or len(left) < 12:
        raise ValueError("both checker fields are required")

    x_centers = _cluster([x for x, _ in points])
    y_centers = _cluster([y for _, y in points])
    if len(x_centers) < 8 or len(y_centers) < 8:
        raise ValueError("lattice has insufficient extent")
    # Dot columns alternate rows, so every true column appears among the union.
    x_steps = [b - a for a, b in zip(x_centers, x_centers[1:])]
    y_steps = [b - a for a, b in zip(y_centers, y_centers[1:])]
    pitch = float(np.median(y_steps))
    if not 23 <= pitch <= 29 or max(abs(step - pitch) for step in y_steps) > 2.0:
        raise ValueError("uncertain row pitch")

    x0 = x_centers[0]
    y0 = y_centers[0]
    parity_by_side = [[], []]
    residuals = []
    for x, y in points:
        col = int(round((x - x0) / pitch))
        row = int(round((y - y0) / pitch))
        residuals.append(abs(x - (x0 + col * pitch)) + abs(y - (y0 + row * pitch)))
        side = 0 if x < screen_left else 1
        parity_by_side[side].append((row + col) % 2)
    if max(residuals) > 2.5:
        raise ValueError("dots do not register to one square lattice")
    if any(len(set(values)) != 1 for values in parity_by_side):
        raise ValueError("checker phase is internally inconsistent")
    return "same" if parity_by_side[0][0] == parity_by_side[1][0] else "opposite"
