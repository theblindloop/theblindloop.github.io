"""Deterministic connected-region scenes for bilateral area integration."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH = 768
HEIGHT = 768
BACKGROUND = [248, 247, 242]
ISLAND_RGB = [37, 104, 151]
RULE_RGB = [238, 174, 45]
FRAME_RGB = [173, 169, 157]
LABEL_RGB = [42, 45, 48]
CELL_SIZE = 22
GRID_SIZE = 10
CELL_COUNT = 60
HALF_COLUMNS = GRID_SIZE // 2
PANEL_CENTERS = ((192, 192), (576, 192), (192, 576), (576, 576))
LABELS = ("A", "B", "C", "D")
MIN_MARGIN_CELLS = 4.0

_ALL_CELLS = {(x, y) for y in range(GRID_SIZE) for x in range(GRID_SIZE)}
_BOUNDARY = {
    (x, y)
    for x, y in _ALL_CELLS
    if x in (0, GRID_SIZE - 1) or y in (0, GRID_SIZE - 1)
}


def _neighbors(point: tuple[int, int]) -> tuple[tuple[int, int], ...]:
    x, y = point
    return ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1))


def _is_connected(cells: set[tuple[int, int]]) -> bool:
    if not cells:
        return False
    pending = [next(iter(cells))]
    seen = {pending[0]}
    while pending:
        point = pending.pop()
        for neighbor in _neighbors(point):
            if neighbor in cells and neighbor not in seen:
                seen.add(neighbor)
                pending.append(neighbor)
    return len(seen) == len(cells)


def _side_counts(cells: list[list[int]] | set[tuple[int, int]]) -> tuple[int, int]:
    points = [(int(point[0]), int(point[1])) for point in cells]
    left = sum(x < HALF_COLUMNS for x, _ in points)
    return left, len(points) - left


def _grow_side(
    rng: random.Random,
    cells: set[tuple[int, int]],
    *,
    left_side: bool,
    target_count: int,
) -> None:
    def on_side(point: tuple[int, int]) -> bool:
        return (point[0] < HALF_COLUMNS) == left_side

    while sum(on_side(point) for point in cells) < target_count:
        frontier = sorted(
            point
            for point in (_ALL_CELLS - cells)
            if on_side(point) and any(neighbor in cells for neighbor in _neighbors(point))
        )
        if not frontier:
            raise RuntimeError("connected growth exhausted before reaching side-area target")
        # A small centerward preference keeps the masks area-like while the random
        # term produces many asymmetric silhouettes with identical totals/extents.
        weighted = []
        for point in frontier:
            x, y = point
            center_distance = abs(x - 4.5) + 0.35 * abs(y - 4.5)
            weight = 1.0 / (1.0 + center_distance)
            weighted.append((rng.random() ** (1.0 / weight), point))
        weighted.sort(reverse=True)
        cells.add(weighted[0][1])


def _sample_cells(rng: random.Random, left_count: int) -> list[list[int]]:
    """Grow one 60-cell connected mask with a prescribed half-area split.

    The complete outer cell ring is common to every candidate. Besides making
    total extent identical, it guarantees that every coarse ablation overlapping
    the panel removes genuine evidence from the single connected region.
    """
    right_count = CELL_COUNT - int(left_count)
    if not 18 <= left_count <= 42 or not 18 <= right_count <= 42:
        raise ValueError("requested side count cannot contain the common boundary")
    for _ in range(200):
        cells = set(_BOUNDARY)
        # Randomize which side grows first without changing the requested totals.
        if rng.randrange(2):
            _grow_side(rng, cells, left_side=True, target_count=left_count)
            _grow_side(rng, cells, left_side=False, target_count=right_count)
        else:
            _grow_side(rng, cells, left_side=False, target_count=right_count)
            _grow_side(rng, cells, left_side=True, target_count=left_count)
        reflected = {(GRID_SIZE - 1 - x, y) for x, y in cells}
        if len(cells) == CELL_COUNT and _is_connected(cells) and cells != reflected:
            return [list(point) for point in sorted(cells, key=lambda p: (p[1], p[0]))]
    raise RuntimeError("could not sample an asymmetric connected island")


def sample_scene(seed: int) -> dict:
    """Sample a class-balanced scene with no stored answer field."""
    seed = int(seed)
    rng = random.Random(seed)
    target_index = seed % len(LABELS)
    loser_differences = [4, 8, 12]
    rng.shuffle(loser_differences)
    signs = [-1, 1, -1, 1]
    rng.shuffle(signs)
    panels = []
    loser_rank = 0
    for index, label in enumerate(LABELS):
        if index == target_index:
            left_count = CELL_COUNT // 2
        else:
            difference = loser_differences[loser_rank]
            sign = signs[loser_rank]
            left_count = CELL_COUNT // 2 + sign * (difference // 2)
            loser_rank += 1
        cells = _sample_cells(rng, left_count)
        panels.append({"label": label, "cells": cells})
    draw_order = list(range(len(panels)))
    rng.shuffle(draw_order)
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
        "island_rgb": list(ISLAND_RGB),
        "rule_rgb": list(RULE_RGB),
        "frame_rgb": list(FRAME_RGB),
        "label_rgb": list(LABEL_RGB),
        "cell_size": CELL_SIZE,
        "panel_centers": [list(center) for center in PANEL_CENTERS],
        "panels": panels,
        "draw_order": draw_order,
    }


def _draw_panel(
    scene: dict,
    draw: ImageDraw.ImageDraw,
    index: int,
    font: ImageFont.ImageFont,
) -> None:
    cx, cy = (int(value) for value in scene["panel_centers"][index])
    panel = scene["panels"][index]
    frame_half = 150
    draw.rounded_rectangle(
        (cx - frame_half, cy - frame_half, cx + frame_half, cy + frame_half),
        radius=16,
        fill=(253, 252, 248),
        outline=tuple(scene["frame_rgb"]),
        width=3,
    )
    # The rule is drawn first, so the blue island remains one unbroken component.
    rule = tuple(scene["rule_rgb"])
    draw.line((cx, cy - 135, cx, cy + 135), fill=rule, width=5)
    draw.polygon(((cx - 10, cy - 126), (cx + 10, cy - 126), (cx, cy - 113)), fill=rule)
    draw.polygon(((cx - 10, cy + 126), (cx + 10, cy + 126), (cx, cy + 113)), fill=rule)

    cell_size = int(scene["cell_size"])
    origin_x = cx - GRID_SIZE * cell_size // 2
    origin_y = cy - GRID_SIZE * cell_size // 2
    for x, y in panel["cells"]:
        x0 = origin_x + int(x) * cell_size
        y0 = origin_y + int(y) * cell_size
        draw.rectangle(
            (x0, y0, x0 + cell_size - 1, y0 + cell_size - 1),
            fill=tuple(scene["island_rgb"]),
        )
    draw.text(
        (cx - frame_half + 14, cy - frame_half + 10),
        str(panel["label"]),
        fill=tuple(scene["label_rgb"]),
        font=font,
        anchor="la",
    )


def render(scene: dict) -> Image.Image:
    image = Image.new(
        "RGB",
        (int(scene["width"]), int(scene["height"])),
        tuple(int(value) for value in scene["background"]),
    )
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default(size=30)
    for index in scene["draw_order"]:
        _draw_panel(scene, draw, int(index), font)
    return image


def _rank(scene: dict) -> list[tuple[int, str]]:
    ranked = []
    for panel in scene["panels"]:
        left, right = _side_counts(panel["cells"])
        ranked.append((abs(left - right), str(panel["label"])))
    return sorted(ranked)


def analytic_gold(scene: dict) -> str:
    ranked = _rank(scene)
    if ranked[0][0] != 0 or sum(difference == 0 for difference, _ in ranked) != 1:
        raise ValueError("expected exactly one equal-half-area island")
    return ranked[0][1]


def margin(scene: dict) -> float:
    ranked = _rank(scene)
    return float(ranked[1][0] - ranked[0][0])


def is_quarantined(scene: dict) -> bool:
    ranked = _rank(scene)
    return not (
        ranked[0][0] == 0
        and sum(difference == 0 for difference, _ in ranked) == 1
        and margin(scene) >= MIN_MARGIN_CELLS
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["draw_order"] = list(reversed(twin["draw_order"]))
    return [("reverse_disjoint_panel_draw_order", twin)]
