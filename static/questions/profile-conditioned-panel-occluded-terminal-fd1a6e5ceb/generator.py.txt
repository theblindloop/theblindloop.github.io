"""panel_occluded_terminal renderer.

Latent scene: three grey pipes drawn as polylines across a white page, and three
opaque vertical panels painted on top of them. A pipe bends only in the open
gaps between panels; behind a panel it is one straight run that continues the
direction it entered with. Pipes cross each other only behind panels, so the
binding between a left-hand starting dot and a right-hand coloured square is
carried entirely by amodal continuation across the occluding panels.

The analytic label is the colour of the square terminating the pipe whose left
end is circled by the black ring.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "panel_occluded_terminal-renderer-0.1.0"

CANVAS = (860, 560)
Y_LO, Y_HI = 74.0, 486.0

# Panel geometry: [left face x, right face x). Pipe ink is visible up to
# left_face - 1 and again from right_face onwards.
PANELS = ((200, 284), (404, 488), (608, 692))
APPROACH = 46.0  # visible diagonal run before a panel
EXIT_RUN = 46.0  # visible diagonal run after a panel

X_START = 44.0
X_END = 816.0

MIN_GAP_OPEN = 34.0      # min vertical clearance between pipes on horizontal runs
MIN_GAP_FACE = 26.0      # min vertical clearance between stubs at a panel face
MAX_SLOPE = 1.0
MIN_SLOPE_DIFF = 0.28    # transversality of a hidden crossing
CROSS_INSET = 12.0       # hidden crossings must sit this far inside a panel

QUARANTINE_MARGIN_PX = 26.0

PIPE_COLOR = (88, 92, 104)
PANEL_FILL = (198, 202, 210)
PANEL_EDGE = (150, 155, 165)
MARK_COLOR = (0, 0, 0)
BACKGROUND = (255, 255, 255)

TERMINAL_COLORS = {
    "blue": (33, 98, 220),
    "orange": (232, 120, 20),
    "purple": (140, 60, 200),
}

PIPE_WIDTH = 7
PORT_RADIUS = 6
RING_RADIUS = 18
RING_WIDTH = 4
SQUARE_HALF = 14

_N_PIPES = 3


def _ordered_levels(rng: random.Random, order, gap: float):
    """Sample one y level per pipe so that ``order`` (top-to-bottom pipe ids)
    holds with at least ``gap`` clearance between neighbours."""
    span = Y_HI - Y_LO
    slack = span - gap * (len(order) - 1)
    if slack <= 0:
        return None
    cuts = sorted(rng.uniform(0.0, slack) for _ in range(len(order) - 1))
    levels = {}
    y = Y_LO + rng.uniform(0.0, slack - (cuts[-1] if cuts else 0.0))
    prev = 0.0
    for i, pipe in enumerate(order):
        if i > 0:
            y += gap + (cuts[i - 1] - prev)
            prev = cuts[i - 1]
        if y > Y_HI:
            return None
        levels[pipe] = y
    return levels


def _order_of(levels) -> tuple:
    return tuple(sorted(levels, key=lambda p: levels[p]))


def _min_gap(levels) -> float:
    ys = sorted(levels.values())
    return min(b - a for a, b in zip(ys, ys[1:]))


def _try_panel(rng: random.Random, entry, panel):
    """Given entry horizontal levels, sample the hidden straight runs of one panel.

    Returns (a, b, c, m) with a = levels at the left face, b = levels at the
    right face, c = levels where the pipes flatten out again, m = slopes.
    """
    xl, xr = panel
    ww = float(xr - xl)
    old_order = _order_of(entry)
    for _ in range(160):
        new_order = list(old_order)
        rng.shuffle(new_order)
        new_order = tuple(new_order)
        if new_order == old_order:
            continue
        b = _ordered_levels(rng, new_order, MIN_GAP_OPEN)
        if b is None:
            continue
        m = {p: (b[p] - entry[p]) / (APPROACH + ww) for p in entry}
        if max(abs(v) for v in m.values()) > MAX_SLOPE:
            continue
        a = {p: entry[p] + m[p] * APPROACH for p in entry}
        c = {p: b[p] + m[p] * EXIT_RUN for p in entry}
        if _order_of(a) != old_order or _min_gap(a) < MIN_GAP_FACE:
            continue
        if _order_of(c) != new_order or _min_gap(c) < MIN_GAP_OPEN:
            continue
        if min(c.values()) < Y_LO or max(c.values()) > Y_HI:
            continue
        ok = True
        for i, p in enumerate(entry):
            for q in list(entry)[i + 1:]:
                da = a[p] - a[q]
                db = b[p] - b[q]
                if da * db < 0:  # they swap: the crossing must be hidden and transversal
                    if abs(m[p] - m[q]) < MIN_SLOPE_DIFF:
                        ok = False
                        break
                    t = da / (da - db)
                    if not (CROSS_INSET <= t * ww <= ww - CROSS_INSET):
                        ok = False
                        break
        if ok:
            return a, b, c, m
    return None


def sample_scene(seed: int) -> dict:
    """Sample one latent scene deterministically from ``seed``."""
    rng = random.Random(0x9E37 ^ (seed * 2654435761 + 12345))
    pipes = tuple(range(_N_PIPES))
    for _ in range(400):
        start_order = list(pipes)
        rng.shuffle(start_order)
        entry = _ordered_levels(rng, tuple(start_order), MIN_GAP_OPEN)
        if entry is None:
            continue
        polylines = {p: [(X_START, entry[p])] for p in pipes}
        levels = entry
        good = True
        for panel in PANELS:
            xl, xr = panel
            got = _try_panel(rng, levels, panel)
            if got is None:
                good = False
                break
            a, b, c, _m = got
            for p in pipes:
                polylines[p].append((xl - APPROACH, levels[p]))
                polylines[p].append((float(xl), a[p]))
                polylines[p].append((float(xr), b[p]))
                polylines[p].append((xr + EXIT_RUN, c[p]))
            levels = c
        if not good:
            continue
        for p in pipes:
            polylines[p].append((X_END, levels[p]))

        end_order = _order_of(levels)
        start_rank = {p: i for i, p in enumerate(_order_of(entry))}
        end_rank = {p: i for i, p in enumerate(end_order)}

        ring_pipe = rng.randrange(_N_PIPES)
        # Anti-shortcut: reading the square at the ring's own height must fail.
        if start_rank[ring_pipe] == end_rank[ring_pipe]:
            continue

        colors = list(TERMINAL_COLORS)
        rng.shuffle(colors)
        terminal_color = {p: colors[i] for i, p in enumerate(pipes)}

        scene = {
            "canvas": list(CANVAS),
            "panels": [list(p) for p in PANELS],
            "pipes": [
                {
                    "points": [[round(x, 4), round(y, 4)] for x, y in polylines[p]],
                    "terminal_color": terminal_color[p],
                }
                for p in pipes
            ],
            "ring_pipe": ring_pipe,
            "style": {
                "pipe_width": PIPE_WIDTH,
                "port_radius": PORT_RADIUS,
                "ring_radius": RING_RADIUS,
                "square_half": SQUARE_HALF,
            },
        }
        return scene
    raise RuntimeError(f"could not sample a valid scene for seed {seed}")


def _face_levels(scene: dict):
    """Recover, per panel, the (entry-face, exit-face) levels of every pipe."""
    out = []
    for j, (xl, xr) in enumerate(scene["panels"]):
        a, b = {}, {}
        for i, pipe in enumerate(scene["pipes"]):
            pts = pipe["points"]
            a[i] = [y for x, y in pts if abs(x - xl) < 1e-6][0]
            b[i] = [y for x, y in pts if abs(x - xr) < 1e-6][0]
        out.append((a, b))
    return out


def analytic_gold(scene: dict) -> str:
    """Colour of the square terminating the pipe whose left end carries the ring."""
    return scene["pipes"][scene["ring_pipe"]]["terminal_color"]


def margin(scene: dict) -> float:
    """Smallest vertical separation, in pixels, between two pipes at any panel
    face. It is the distance by which a straight continuation across a panel
    must be resolved before the wrong stub would be picked up instead."""
    worst = float("inf")
    for a, b in _face_levels(scene):
        for levels in (a, b):
            ys = sorted(levels.values())
            worst = min(worst, min(hi - lo for lo, hi in zip(ys, ys[1:])))
    return round(float(worst), 4)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN_PX


def latent_symmetries(scene: dict) -> list:
    """No latent alias: every field that can move the label is painted.

    The label reads only the pipe polylines, the ring index, and the terminal
    colours. Each polyline is the drawn centreline (its hidden spans are the
    straight continuations of the visible stubs), the ring index is the drawn
    annulus, and a terminal colour is literally the fill of its square. No
    two scenes with different labels rasterise identically, so the alias gate
    is ``not_applicable`` rather than ``pass``.
    """
    return []


def render(scene: dict) -> Image.Image:
    w, h = scene["canvas"]
    img = Image.new("RGB", (w, h), BACKGROUND)
    draw = ImageDraw.Draw(img)
    half = scene["style"]["pipe_width"] / 2.0

    for pipe in scene["pipes"]:
        pts = [(float(x), float(y)) for x, y in pipe["points"]]
        draw.line(pts, fill=PIPE_COLOR, width=scene["style"]["pipe_width"])
        for x, y in pts[1:-1]:
            draw.ellipse([x - half, y - half, x + half, y + half], fill=PIPE_COLOR)

    for xl, xr in scene["panels"]:
        draw.rectangle([xl, 0, xr - 1, h - 1], fill=PANEL_FILL)
        draw.rectangle([xl, 0, xr - 1, h - 1], outline=PANEL_EDGE, width=3)

    for i, pipe in enumerate(scene["pipes"]):
        sx, sy = pipe["points"][0]
        r = scene["style"]["port_radius"]
        draw.ellipse([sx - r, sy - r, sx + r, sy + r], fill=MARK_COLOR)
        if i == scene["ring_pipe"]:
            rr = scene["style"]["ring_radius"]
            draw.ellipse(
                [sx - rr, sy - rr, sx + rr, sy + rr],
                outline=MARK_COLOR,
                width=RING_WIDTH,
            )
        ex, ey = pipe["points"][-1]
        s = scene["style"]["square_half"]
        draw.rectangle(
            [ex - s, ey - s, ex + s, ey + s], fill=TERMINAL_COLORS[pipe["terminal_color"]]
        )
    return img
