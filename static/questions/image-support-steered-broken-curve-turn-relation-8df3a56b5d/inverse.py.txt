from __future__ import annotations

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    parts: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        seen[y, x] = True
        stack = [(int(y), int(x))]
        pixels = []
        while stack:
            py, px = stack.pop()
            pixels.append((py, px))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    ny, nx = py + dy, px + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        if len(pixels) >= 25:
            parts.append(np.asarray(pixels, dtype=float)[:, ::-1])
    return parts


def _quadratic_bend(part: np.ndarray, origin: np.ndarray, u_axis: np.ndarray, v_axis: np.ndarray) -> float:
    centered = part - origin
    u = centered @ u_axis
    v = centered @ v_axis
    # Bin across the component so stroke thickness cannot dominate the fit.
    edges = np.linspace(float(u.min()), float(u.max()), 14)
    ub, vb = [], []
    for lo, hi in zip(edges[:-1], edges[1:]):
        pick = (u >= lo) & (u <= hi)
        if np.any(pick):
            ub.append(float(np.mean(u[pick])))
            vb.append(float(np.mean(v[pick])))
    if len(ub) < 7:
        raise ValueError("too little curve remains")
    return float(np.polyfit(np.asarray(ub), np.asarray(vb), 2)[0])


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    background = np.array([250, 249, 246], dtype=np.int16)
    mask = np.max(np.abs(rgb - background), axis=2) > 35
    parts = _components(mask)
    if len(parts) != 2:
        return "abstain"
    centers = np.asarray([part.mean(axis=0) for part in parts])
    delta = centers[1] - centers[0]
    norm = float(np.linalg.norm(delta))
    if norm < 45.0:
        return "abstain"
    u_axis = delta / norm
    v_axis = np.array([-u_axis[1], u_axis[0]])
    origin = centers.mean(axis=0)
    try:
        bends = [_quadratic_bend(part, origin, u_axis, v_axis) for part in parts]
    except (ValueError, np.linalg.LinAlgError):
        return "abstain"
    if min(abs(bend) for bend in bends) < 0.003:
        return "abstain"
    return "C" if bends[0] * bends[1] > 0 else "S"
