from __future__ import annotations

import math
import random
from PIL import Image, ImageDraw

SIZE = 384


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    same = (seed % 2) == 0
    sign_a = rng.choice([-1, 1])
    sign_b = sign_a if same else -sign_a
    # Regularly include visibly near-boundary and quarantined bows.
    if seed % 12 == 0:
        bows = [6, rng.randint(13, 18)]
    elif seed % 6 == 0:
        bows = [9, rng.randint(13, 18)]
    else:
        bows = [rng.randint(13, 18), rng.randint(13, 18)]
    rng.shuffle(bows)
    return {
        "bend_a": sign_a * bows[0],
        "bend_b": sign_b * bows[1],
        "cx": rng.choice([120, 168, 216, 264]),
        "cy": rng.choice([120, 168, 216, 264]),
        "rotation": rng.choice([0.0, 90.0]),
        "half_length": rng.randint(16, 18),
        "separation": 48,
        "width": 6,
        "color": rng.choice([[29, 91, 172], [24, 116, 147], [76, 67, 165]]),
    }


def _transform(x: float, y: float, scene: dict) -> tuple[float, float]:
    angle = math.radians(float(scene["rotation"]))
    ca, sa = math.cos(angle), math.sin(angle)
    return (
        float(scene["cx"]) + ca * x - sa * y,
        float(scene["cy"]) + sa * x + ca * y,
    )


def _fragment(center_x: float, bow: float, half_length: float, scene: dict) -> list[tuple[float, float]]:
    points = []
    for index in range(49):
        t = -1.0 + 2.0 * index / 48.0
        x = center_x + half_length * t
        y = bow * (1.0 - t * t)
        points.append(_transform(x, y, scene))
    return points


def render(scene: dict) -> Image.Image:
    scale = 4
    image = Image.new("RGB", (SIZE * scale, SIZE * scale), (250, 249, 246))
    draw = ImageDraw.Draw(image)
    half_length = float(scene["half_length"])
    separation = float(scene["separation"])
    color = tuple(scene["color"])
    fragments = (
        (-separation, float(scene["bend_a"])),
        (separation, float(scene["bend_b"])),
    )
    for center_x, bend in fragments:
        points = _fragment(center_x, bend, half_length, scene)
        draw.line([(x * scale, y * scale) for x, y in points], fill=color,
                  width=int(scene["width"]) * scale, joint="curve")
    return image.resize((SIZE, SIZE), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "C" if float(scene["bend_a"]) * float(scene["bend_b"]) > 0 else "S"


def margin(scene: dict) -> float:
    return min(abs(float(scene["bend_a"])), abs(float(scene["bend_b"])))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
