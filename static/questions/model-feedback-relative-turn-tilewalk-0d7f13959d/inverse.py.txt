"""Independent image-only decoder for the relative-turn tile walk."""

from __future__ import annotations

import numpy as np
from PIL import Image

CANVAS = (610, 590)
N, CELL, X0, Y0, STEPS = 5, 88, 85, 75, 8
DIRS = ((0, -1), (1, 0), (0, 1), (-1, 0))
TURN = {"L": -1, "D": 0, "R": 1}
COLORS = {
    "amber": np.array((235, 150, 20), dtype=np.int32),
    "cyan": np.array((0, 160, 190), dtype=np.int32),
    "violet": np.array((145, 75, 190), dtype=np.int32),
}


class OracleError(ValueError):
    pass


def _dark(arr: np.ndarray) -> np.ndarray:
    return np.all(arr < 80, axis=2)


def _decode_start(arr: np.ndarray) -> tuple[tuple[int, int], int]:
    regions = {
        0: (slice(5, 70), slice(X0, X0 + N * CELL)),
        1: (slice(Y0, Y0 + N * CELL), slice(540, 600)),
        2: (slice(520, 585), slice(X0, X0 + N * CELL)),
        3: (slice(Y0, Y0 + N * CELL), slice(5, 70)),
    }
    hits = []
    for side, (ys, xs) in regions.items():
        yy, xx = np.nonzero(_dark(arr[ys, xs]))
        if len(xx) > 450:
            xoff = xs.start or 0
            yoff = ys.start or 0
            hits.append((side, float(np.median(xx + xoff)), float(np.median(yy + yoff))))
    if len(hits) != 1:
        raise OracleError("start arrow missing or duplicated")
    side, cx, cy = hits[0]
    if side in (0, 2):
        col = int((cx - X0) // CELL)
        start = (col, 0 if side == 0 else N - 1)
        heading = 2 if side == 0 else 0
    else:
        row = int((cy - Y0) // CELL)
        start = (N - 1 if side == 1 else 0, row)
        heading = 3 if side == 1 else 1
    if not (0 <= start[0] < N and 0 <= start[1] < N):
        raise OracleError("start arrow is not aligned to a tile")
    return start, heading


def _decode_grid(arr: np.ndarray) -> list[list[str]]:
    grid: list[list[str]] = []
    for y in range(N):
        row = []
        for x in range(N):
            crop = arr[
                Y0 + y * CELL + 20:Y0 + (y + 1) * CELL - 20,
                X0 + x * CELL + 20:X0 + (x + 1) * CELL - 20,
            ]
            count = int(_dark(crop).sum())
            if 185 <= count <= 245:
                glyph = "D"
            elif 345 <= count <= 410:
                glyph = "L"
            elif 620 <= count <= 700:
                glyph = "R"
            else:
                raise OracleError(f"unreadable tile glyph at {(x, y)}: {count}")
            row.append(glyph)
        grid.append(row)
    return grid


def _decode_goals(arr: np.ndarray) -> dict[tuple[int, int], str]:
    found: dict[tuple[int, int], str] = {}
    for y in range(N):
        for x in range(N):
            crop = arr[
                Y0 + y * CELL + 4:Y0 + (y + 1) * CELL - 4,
                X0 + x * CELL + 4:X0 + (x + 1) * CELL - 4,
            ].astype(np.int32)
            matches = []
            for name, rgb in COLORS.items():
                distance = np.sqrt(np.sum((crop - rgb) ** 2, axis=2))
                if int((distance < 35).sum()) > 700:
                    matches.append(name)
            if len(matches) > 1:
                raise OracleError("multiple goal colors on one tile")
            if matches:
                found[(x, y)] = matches[0]
    if len(found) != 3 or set(found.values()) != set(COLORS):
        raise OracleError("expected exactly one goal of each color")
    return found


def decision_from_image(image: Image.Image) -> str:
    if image.size != CANVAS:
        raise OracleError(f"wrong image size {image.size}")
    arr = np.asarray(image.convert("RGB"))
    start, heading = _decode_start(arr)
    grid = _decode_grid(arr)
    goals = _decode_goals(arr)
    cell = start
    visited = {cell}
    for _ in range(STEPS):
        x, y = cell
        heading = (heading + TURN[grid[y][x]]) % 4
        dx, dy = DIRS[heading]
        cell = (x + dx, y + dy)
        if not (0 <= cell[0] < N and 0 <= cell[1] < N):
            raise OracleError("walk leaves the grid")
        if cell in visited:
            raise OracleError("walk repeats a tile")
        visited.add(cell)
    if cell not in goals:
        raise OracleError("final tile is not a colored goal")
    if any(tile in goals for tile in visited - {cell}):
        raise OracleError("goal is visited before the final move")
    return goals[cell]
