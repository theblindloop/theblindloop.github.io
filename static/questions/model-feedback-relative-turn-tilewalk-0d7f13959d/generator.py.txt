"""Deterministic latent sampler and renderer for a relative-turn grid automaton."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

CANVAS = (610, 590)
N = 5
CELL = 88
X0, Y0 = 85, 75
STEPS = 8
GOAL_COLORS = {
    "amber": (235, 150, 20),
    "cyan": (0, 160, 190),
    "violet": (145, 75, 190),
}
DIRS = ((0, -1), (1, 0), (0, 1), (-1, 0))  # up, right, down, left
TURN = {"L": -1, "D": 0, "R": 1}


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype("DejaVuSans-Bold.ttf", size)


def _advance(cell: tuple[int, int], heading: int, glyph: str) -> tuple[tuple[int, int], int]:
    heading = (heading + TURN[glyph]) % 4
    dx, dy = DIRS[heading]
    return (cell[0] + dx, cell[1] + dy), heading


def _path_for(rng: random.Random, side: int, offset: int) -> tuple[list[tuple[int, int]], list[int]]:
    if side == 0:  # top, facing down
        start, heading = (offset, 0), 2
    elif side == 1:  # right, facing left
        start, heading = (N - 1, offset), 3
    elif side == 2:  # bottom, facing up
        start, heading = (offset, N - 1), 0
    else:  # left, facing right
        start, heading = (0, offset), 1

    cells = [start]
    headings = [heading]

    def visit(cell: tuple[int, int], head: int) -> bool:
        if len(cells) == STEPS + 1:
            return True
        choices = [-1, 0, 1]
        rng.shuffle(choices)
        for delta in choices:
            nh = (head + delta) % 4
            dx, dy = DIRS[nh]
            nxt = (cell[0] + dx, cell[1] + dy)
            if not (0 <= nxt[0] < N and 0 <= nxt[1] < N) or nxt in cells:
                continue
            cells.append(nxt)
            headings.append(nh)
            if visit(nxt, nh):
                return True
            cells.pop()
            headings.pop()
        return False

    if not visit(start, heading):
        raise RuntimeError("could not construct tile walk")
    return cells, headings


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x5A17C0DE)
    side = int(seed) % 4
    offset = rng.randrange(1, N - 1)
    path, heads = _path_for(rng, side, offset)
    grid = [[rng.choice(("L", "D", "R")) for _ in range(N)] for _ in range(N)]
    for i in range(STEPS):
        delta = (heads[i + 1] - heads[i]) % 4
        glyph = {3: "L", 0: "D", 1: "R"}[delta]
        x, y = path[i]
        grid[y][x] = glyph
    endpoint = path[-1]
    answer = ("amber", "cyan", "violet")[int(seed) % 3]
    available = [(x, y) for y in range(N) for x in range(N) if (x, y) not in path]
    rng.shuffle(available)
    decoys = available[:2]
    other_colors = [c for c in GOAL_COLORS if c != answer]
    rng.shuffle(other_colors)
    goals = {answer: list(endpoint), other_colors[0]: list(decoys[0]), other_colors[1]: list(decoys[1])}
    return {
        "instructions": grid,
        "start": list(path[0]),
        "heading": heads[0],
        "goals": goals,
    }


def _simulate(scene: dict) -> tuple[tuple[int, int], list[tuple[int, int]]]:
    cell = tuple(scene["start"])
    heading = int(scene["heading"])
    trail = [cell]
    for _ in range(STEPS):
        x, y = cell
        cell, heading = _advance(cell, heading, scene["instructions"][y][x])
        trail.append(cell)
    return cell, trail


def analytic_gold(scene: dict) -> str:
    endpoint, _ = _simulate(scene)
    matches = [color for color, cell in scene["goals"].items() if tuple(cell) == endpoint]
    if len(matches) != 1:
        raise ValueError("endpoint does not have exactly one goal")
    return matches[0]


def margin(scene: dict) -> float:
    return 1.0 if not is_quarantined(scene) else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        grid = scene["instructions"]
        if len(grid) != N or any(len(row) != N for row in grid):
            return True
        if any(g not in TURN for row in grid for g in row):
            return True
        start = tuple(scene["start"])
        heading = int(scene["heading"])
        if heading not in range(4):
            return True
        endpoint, trail = _simulate(scene)
        if any(not (0 <= x < N and 0 <= y < N) for x, y in trail):
            return True
        if len(set(trail)) != len(trail):
            return True
        goal_cells = [tuple(v) for v in scene["goals"].values()]
        if set(scene["goals"]) != set(GOAL_COLORS) or len(set(goal_cells)) != 3:
            return True
        if endpoint not in goal_cells or any(cell in goal_cells for cell in trail[:-1]):
            return True
        side_ok = (
            (start[1] == 0 and heading == 2) or (start[0] == N - 1 and heading == 3)
            or (start[1] == N - 1 and heading == 0) or (start[0] == 0 and heading == 1)
        )
        return not side_ok
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _draw_start_arrow(draw: ImageDraw.ImageDraw, start: tuple[int, int], heading: int) -> None:
    x, y = start
    cx = X0 + x * CELL + CELL // 2
    cy = Y0 + y * CELL + CELL // 2
    if heading == 2:  # down from top
        draw.rectangle((cx - 7, 16, cx + 7, 48), fill="black")
        draw.polygon(((cx - 20, 45), (cx + 20, 45), (cx, 67)), fill="black")
    elif heading == 0:  # up from bottom
        draw.rectangle((cx - 7, 542, cx + 7, 574), fill="black")
        draw.polygon(((cx - 20, 545), (cx + 20, 545), (cx, 523)), fill="black")
    elif heading == 1:  # right from left
        draw.rectangle((16, cy - 7, 48, cy + 7), fill="black")
        draw.polygon(((45, cy - 20), (45, cy + 20), (67, cy)), fill="black")
    else:  # left from right
        draw.rectangle((542, cy - 7, 574, cy + 7), fill="black")
        draw.polygon(((545, cy - 20), (545, cy + 20), (523, cy)), fill="black")


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS, (248, 248, 246))
    draw = ImageDraw.Draw(image)
    # Checkerboard is deliberately very pale; it separates cells without encoding state.
    for y in range(N):
        for x in range(N):
            fill = (255, 255, 255) if (x + y) % 2 == 0 else (242, 245, 247)
            box = (X0 + x * CELL, Y0 + y * CELL, X0 + (x + 1) * CELL, Y0 + (y + 1) * CELL)
            draw.rectangle(box, fill=fill)
    for color, cell in scene["goals"].items():
        x, y = cell
        x1, y1 = X0 + x * CELL + 7, Y0 + y * CELL + 7
        x2, y2 = X0 + (x + 1) * CELL - 7, Y0 + (y + 1) * CELL - 7
        draw.rectangle((x1, y1, x2, y2), outline=GOAL_COLORS[color], width=8)
    for i in range(N + 1):
        draw.line((X0 + i * CELL, Y0, X0 + i * CELL, Y0 + N * CELL), fill=(45, 50, 55), width=3)
        draw.line((X0, Y0 + i * CELL, X0 + N * CELL, Y0 + i * CELL), fill=(45, 50, 55), width=3)
    font = _font(48)
    for y, row in enumerate(scene["instructions"]):
        for x, glyph in enumerate(row):
            cx, cy = X0 + x * CELL + CELL // 2, Y0 + y * CELL + CELL // 2
            shown = "•" if glyph == "D" else glyph
            box = draw.textbbox((0, 0), shown, font=font)
            draw.text((cx - (box[2] - box[0]) / 2, cy - (box[3] - box[1]) / 2 - box[1]), shown,
                      font=font, fill=(12, 15, 18))
    _draw_start_arrow(draw, tuple(scene["start"]), int(scene["heading"]))
    return image
