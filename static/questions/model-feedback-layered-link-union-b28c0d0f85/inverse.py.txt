"""Independent image-only inverse arm for Layered Link Union."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

WIDTH, HEIGHT = 1100, 800
DECISIONS = ("A", "B", "C", "D")
INK = np.array((43, 48, 58), dtype=np.int16)
NODE_COLORS = {
    "S": (218, 70, 70), "A": (245, 175, 55), "B": (70, 130, 220),
    "C": (105, 180, 80), "D": (150, 90, 200), "E": (50, 170, 170),
    "F": (230, 100, 150), "G": (160, 105, 65), "H": (60, 80, 150),
    "J": (80, 190, 220), "K": (200, 150, 40), "L": (40, 145, 90),
    "M": (120, 70, 150), "N": (235, 115, 45), "P": (125, 135, 145),
}
CARD_ORIGINS = ((40, 90), (560, 90), (40, 435), (560, 435))
ROW_OFFSETS = (92, 172, 252)


def _badge_id(arr: np.ndarray, cx: int, cy: int) -> str | None:
    patch = arr[cy - 23:cy + 24, cx - 23:cx + 24]
    counts = {}
    for node, rgb in NODE_COLORS.items():
        ref = np.array(rgb, dtype=np.int16)
        counts[node] = int((np.abs(patch.astype(np.int16) - ref).max(axis=2) <= 2).sum())
    node, count = max(counts.items(), key=lambda item: item[1])
    ordered = sorted(counts.values(), reverse=True)
    if count < 850 or (len(ordered) > 1 and ordered[1] > 80):
        return None
    return node


def _cable_present(arr: np.ndarray, x0: int, x1: int, cy: int) -> bool:
    strip = arr[cy - 5:cy + 6, x0 + 38:x1 - 38].astype(np.int16)
    exactish = np.abs(strip - INK).max(axis=2) <= 3
    return float(exactish.mean()) >= 0.88


def _read_edges(image: Image.Image) -> list[tuple[str, str]] | None:
    if image.size != (WIDTH, HEIGHT):
        return None
    arr = np.asarray(image.convert("RGB"))
    edges = []
    for x0, y0 in CARD_ORIGINS:
        for dy in ROW_OFFSETS:
            cy, lx, rx = y0 + dy, x0 + 108, x0 + 392
            left, right = _badge_id(arr, lx, cy), _badge_id(arr, rx, cy)
            if left is None or right is None or left == right:
                return None
            if not _cable_present(arr, lx, rx, cy):
                return None
            edges.append((left, right))
    return edges


def decision_from_image(image: Image.Image) -> str:
    edges = _read_edges(image)
    if edges is None:
        return "abstain"
    graph = {node: set() for node in NODE_COLORS}
    for left, right in edges:
        graph[left].add(right)
        graph[right].add(left)
    seen = {"S"}
    queue = deque(["S"])
    while queue:
        cur = queue.popleft()
        for nxt in graph[cur]:
            if nxt not in seen:
                seen.add(nxt)
                queue.append(nxt)
    matches = [node for node in DECISIONS if node in seen]
    return matches[0] if len(matches) == 1 else "abstain"
