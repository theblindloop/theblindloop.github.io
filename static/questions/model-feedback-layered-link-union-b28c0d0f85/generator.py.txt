"""Deterministic renderer and analytic graph arm for Layered Link Union."""

from __future__ import annotations

from collections import deque
import copy
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1100, 800
BACKGROUND = (244, 241, 234)
INK = (43, 48, 58)
CARD = (255, 255, 252)
CARD_EDGE = (91, 99, 112)
DECISIONS = ("A", "B", "C", "D")
NODE_COLORS = {
    "S": (218, 70, 70), "A": (245, 175, 55), "B": (70, 130, 220),
    "C": (105, 180, 80), "D": (150, 90, 200), "E": (50, 170, 170),
    "F": (230, 100, 150), "G": (160, 105, 65), "H": (60, 80, 150),
    "J": (80, 190, 220), "K": (200, 150, 40), "L": (40, 145, 90),
    "M": (120, 70, 150), "N": (235, 115, 45), "P": (125, 135, 145),
}
CARD_BOXES = ((40, 90, 540, 405), (560, 90, 1060, 405),
              (40, 435, 540, 750), (560, 435, 1060, 750))
ROW_OFFSETS = (92, 172, 252)
FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(FONT_PATH, size)


def _components(edges: list[list[list[str]]]) -> dict[str, set[str]]:
    graph: dict[str, set[str]] = {node: set() for node in NODE_COLORS}
    for card in edges:
        for left, right in card:
            graph[left].add(right)
            graph[right].add(left)
    out: dict[str, set[str]] = {}
    for start in graph:
        seen = {start}
        queue = deque([start])
        while queue:
            cur = queue.popleft()
            for nxt in graph[cur]:
                if nxt not in seen:
                    seen.add(nxt)
                    queue.append(nxt)
        out[start] = seen
    return out


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = DECISIONS[int(seed) % len(DECISIONS)]
    excluded = [node for node in DECISIONS if node != target]
    rng.shuffle(excluded)
    x0, x1, x2 = excluded
    # The S-to-target path crosses all four logical cards. Other answer sockets
    # live in three separate components. One harmless repeated cable keeps every
    # card equally dense without changing the graph partition.
    edges = [
        [["S", "E"], ["H", target], [x0, "J"]],
        [["E", "F"], ["J", "K"], [x1, "L"]],
        [["F", "G"], ["L", "M"], [x2, "N"]],
        [["G", "H"], ["N", "P"], ["J", "K"]],
    ]
    row_order = []
    endpoint_flips = []
    for _ in range(4):
        order = [0, 1, 2]
        rng.shuffle(order)
        row_order.append(order)
        endpoint_flips.append([bool(rng.getrandbits(1)) for _ in range(3)])
    card_order = [0, 1, 2, 3]
    rng.shuffle(card_order)
    scene = {
        "edges": edges,
        "card_order": card_order,
        "row_order": row_order,
        "endpoint_flips": endpoint_flips,
    }
    assert not is_quarantined(scene)
    return scene


def analytic_gold(scene: dict) -> str:
    component = _components(scene["edges"])["S"]
    matches = [node for node in DECISIONS if node in component]
    return matches[0] if len(matches) == 1 else "abstain"


def _min_card_count(scene: dict, goal: str) -> int:
    adjacency: dict[str, list[tuple[str, int]]] = {node: [] for node in NODE_COLORS}
    for card_id, card in enumerate(scene["edges"]):
        for left, right in card:
            adjacency[left].append((right, card_id))
            adjacency[right].append((left, card_id))
    queue = deque([("S", 0)])
    seen = {("S", 0)}
    best = 99
    while queue:
        node, mask = queue.popleft()
        if node == goal:
            best = min(best, mask.bit_count())
        for nxt, card_id in adjacency[node]:
            state = (nxt, mask | (1 << card_id))
            if state not in seen:
                seen.add(state)
                queue.append(state)
    return best


def margin(scene: dict) -> float:
    answer = analytic_gold(scene)
    return 1.0 if answer in DECISIONS else 0.0


def is_quarantined(scene: dict) -> bool:
    answer = analytic_gold(scene)
    return answer not in DECISIONS or _min_card_count(scene, answer) < 3


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    transformed = copy.deepcopy(scene)
    transformed["endpoint_flips"] = [
        [not flag for flag in card] for card in transformed["endpoint_flips"]
    ]
    return [("reverse_all_undirected_cable_records", transformed)]


def _canonical_endpoints(pair: list[str], flip: bool) -> tuple[str, str]:
    endpoints = list(reversed(pair)) if flip else list(pair)
    # Cable direction is latent bookkeeping only; the visible row is canonical.
    return tuple(sorted(endpoints))


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((WIDTH // 2, 29), "COMBINE ALL FOUR LINK CARDS", fill=INK,
              font=_font(30), anchor="mm")
    draw.text((WIDTH // 2, 61), "same letter = same socket", fill=(83, 89, 101),
              font=_font(18), anchor="mm")

    for visible_card, box in enumerate(CARD_BOXES):
        x0, y0, x1, y1 = box
        draw.rounded_rectangle(box, radius=18, fill=CARD, outline=CARD_EDGE, width=4)
        draw.text(((x0 + x1) // 2, y0 + 30), f"LINK CARD {visible_card + 1}",
                  fill=INK, font=_font(22), anchor="mm")
        logical_card = scene["card_order"][visible_card]
        for visible_row, dy in enumerate(ROW_OFFSETS):
            edge_index = scene["row_order"][logical_card][visible_row]
            pair = scene["edges"][logical_card][edge_index]
            flip = scene["endpoint_flips"][logical_card][edge_index]
            left, right = _canonical_endpoints(pair, flip)
            cy = y0 + dy
            lx, rx = x0 + 108, x0 + 392
            draw.line((lx, cy, rx, cy), fill=INK, width=14)
            draw.ellipse((lx - 31, cy - 31, lx + 31, cy + 31),
                         fill=NODE_COLORS[left], outline=INK, width=4)
            draw.ellipse((rx - 31, cy - 31, rx + 31, cy + 31),
                         fill=NODE_COLORS[right], outline=INK, width=4)
            for node, cx in ((left, lx), (right, rx)):
                text_fill = (255, 255, 255) if node in {"S", "H", "M"} else (25, 28, 34)
                draw.text((cx, cy + 1), node, fill=text_fill, font=_font(27), anchor="mm")
    return image
