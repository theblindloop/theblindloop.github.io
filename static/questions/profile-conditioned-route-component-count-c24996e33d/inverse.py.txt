"""Pixel-only inverse arm for the route-component-count world.

Reads only the final RGB PNG. Segments the blue stroke by nearest-colour
labelling, then counts the number of 4-connected components of the blue mask.
Because distinct routes never touch, each separate connected component is exactly
one route, so the component count is the answer.

No renderer, prompt, generator, verifier or scene data is imported or read.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_PALETTE = {
    "blue": (40, 90, 210),
    "white": (255, 255, 255),
}
_NAMES = ["blue", "white"]


def _label_pixels(arr: np.ndarray) -> np.ndarray:
    palette = np.array([*_PALETTE.values()], dtype=np.float32)
    dists = np.stack([np.linalg.norm(arr - c, axis=-1) for c in palette], axis=-1)
    return dists.argmin(axis=-1)


def _count_components(mask: np.ndarray) -> int:
    h, w = mask.shape
    seen = np.zeros((h, w), dtype=bool)
    count = 0
    for y in range(h):
        for x in range(w):
            if mask[y, x] and not seen[y, x]:
                count += 1
                stack = [(y, x)]
                seen[y, x] = True
                while stack:
                    yy, xx = stack.pop()
                    for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                        if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                            seen[ny, nx] = True
                            stack.append((ny, nx))
    return count


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label_pixels(arr)
    blue = labels == _NAMES.index("blue")
    if not blue.any():
        raise ValueError("no blue strokes found")
    return str(_count_components(blue))
