"""Route-component-count world: how many separate connected blue routes are drawn.

A scene contains between one and three distinct, non-touching blue routes. Each
route is a winding, self-avoiding polyline (a single connected component of blue
stroke). The routes never touch or cross and are separated by a clear gap. The
question asks how many separate (unconnected) blue routes are present; the answer
is the number of connected components of the blue stroke network. Because the
routes are serpentine polyline strokes rather than solid blobs, the decision is a
connectivity/tracing fact (partitioning the stroke network into connected
components), not a count of target-coloured disks amid distractors.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "route_component_count-0.1.0"

CANVAS = (480, 480)
BG = (255, 255, 255)
COLOR_BLUE = (40, 90, 210)

STROKE = 14                 # blue stroke thickness (px)
EDGE_PAD = 40               # keep routes away from the image border
BAND_GAP = 52               # nominal vertical gap between route bands
MIN_CLEAR = 26.0            # minimum clear gap between distinct routes (px)
ARC_SEP = 24.0              # min turn-length inside a zigzag route

ANSWER_VALUES = [1, 2, 3]


def _pts(points: list) -> list[tuple[float, float]]:
    return [(float(p[0]), float(p[1])) for p in points]


def _polyline_samples(points: list[tuple[float, float]], step: float = 2.0) -> list:
    out: list[tuple[float, float]] = []
    for a, b in zip(points, points[1:]):
        d = math.dist(a, b)
        n = max(1, int(d / step))
        for i in range(n + 1):
            t = i / n
            out.append((a[0] + (b[0] - a[0]) * t, a[1] + (b[1] - a[1]) * t))
    return out


def _clearance(routes: list, step: float = 1.5) -> float:
    """Minimum Euclidean distance between any two points of distinct routes."""
    all_pts = [_polyline_samples(r, step) for r in routes]
    best = float("inf")
    for i in range(len(all_pts)):
        for j in range(i + 1, len(all_pts)):
            for p in all_pts[i]:
                for q in all_pts[j]:
                    d = math.dist(p, q)
                    if d < best:
                        best = d
    return best


def _sample_one_route(rng: random.Random, x_lo: float, x_hi: float, y_lo: float, y_hi: float):
    """A single winding, self-avoiding polyline confined to the given box."""
    for _ in range(400):
        n_vert = rng.randint(4, 7)
        xs = sorted(rng.uniform(x_lo, x_hi) for _ in range(n_vert))
        pts = []
        y = float(rng.uniform(y_lo, y_hi))
        for x in xs:
            pts.append((x, y))
            y = float(rng.uniform(y_lo + min(8, (y_hi - y_lo) / 4),
                                  y_hi - min(8, (y_hi - y_lo) / 4)))
        return pts
    return [(x_lo, (y_lo + y_hi) / 2), (x_hi, (y_lo + y_hi) / 2)]


def _rejection_loop(rng: random.Random) -> dict:
    w, h = CANVAS
    for _ in range(20000):
        k = rng.choice(ANSWER_VALUES)
        lo, hi = EDGE_PAD, h - EDGE_PAD
        usable = hi - lo
        span = (k - 1) * BAND_GAP
        usable = min(span, usable)
        step_h = span / (k - 1) if k > 1 else 0.0
        x_lo, x_hi = EDGE_PAD, w - EDGE_PAD
        routes = []
        for i in range(k):
            y_lo = lo + i * step_h
            y_hi = lo + (i + 1) * step_h
            if (y_hi - y_lo) < 40:
                y_lo = lo - 20
                y_hi = lo + 20
            route = _sample_one_route(rng, x_lo, x_hi, y_lo, y_hi)
            route = _pts(route)
            routes.append(route)

        clear = _clearance(routes)
        if clear < MIN_CLEAR:
            continue
        # every vertex must stay comfortably inside the canvas after stroke
        inside = True
        for r in routes:
            for (x, y) in r:
                if not (EDGE_PAD - 10 <= x <= w - EDGE_PAD + 10
                        and EDGE_PAD - 10 <= y <= h - EDGE_PAD + 10):
                    inside = False
        if not inside:
            continue

        scene = {"routes": routes}
        return scene
    raise RuntimeError("could not satisfy route constraints")


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    return _rejection_loop(rng)


def _draw_polyline(draw: ImageDraw.ImageDraw, pts, width):
    for a, b in zip(pts, pts[1:]):
        draw.line([a, b], fill=COLOR_BLUE, width=width, joint="curve")


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", CANVAS, BG)
    draw = ImageDraw.Draw(img)
    for route in scene["routes"]:
        _draw_polyline(draw, _pts(route), STROKE)
    return img


def analytic_gold(scene: dict) -> str:
    """Number of separate (connected-component) blue routes."""
    return str(len(scene["routes"]))


def margin(scene: dict) -> float:
    """Minimum clear gap between distinct routes (px)."""
    return _clearance(_pts(r) for r in scene["routes"])


def is_quarantined(scene: dict) -> bool:
    w, h = CANVAS
    if margin(scene) < MIN_CLEAR:
        return True
    for r in scene["routes"]:
        for (x, y) in r:
            if not (EDGE_PAD - 10 <= x <= w - EDGE_PAD + 10
                    and EDGE_PAD - 10 <= y <= h - EDGE_PAD + 10):
                return True
    return False


def latent_symmetries(scene: dict) -> list:
    """No render-invariant transform: route shapes/positions are visible and any
    change that reconnects or separates strokes changes both pixels and the gold.
    Declared empty (not_applicable)."""
    return []
