"""Independent raster-only inverse arm for exclusive two-circle membership."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

RING_PALETTE = (
    np.array((225, 145, 24), dtype=float),
    np.array((202, 52, 112), dtype=float),
)
DOT_COLOR = np.array((20, 151, 151), dtype=float)
LABELS = ("A", "B", "C", "D")


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        pts = []
        while stack:
            yy, xx = stack.pop()
            pts.append((yy, xx))
            for ny, nx in ((yy-1, xx), (yy+1, xx), (yy, xx-1), (yy, xx+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(pts) >= minimum:
            groups.append(np.asarray(pts, dtype=float))
    return groups


def _fit_circle(points: np.ndarray) -> tuple[float, float, float]:
    if len(points) < 80:
        raise ValueError("too few circle pixels")
    y, x = points[:, 0], points[:, 1]
    a = np.column_stack((2.0 * x, 2.0 * y, np.ones_like(x)))
    cx, cy, q = np.linalg.lstsq(a, x * x + y * y, rcond=None)[0]
    radius = math.sqrt(max(0.0, q + cx * cx + cy * cy))
    return float(cx), float(cy), radius


def _panel_boxes(arr: np.ndarray) -> list[tuple[int, int, int, int]]:
    panel_color = np.array((247, 249, 252), dtype=float)
    groups = _components(np.linalg.norm(arr - panel_color, axis=2) < 8.0, 5000)
    boxes = []
    for pts in groups:
        ys, xs = pts[:, 0], pts[:, 1]
        boxes.append((int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())))
    return sorted(boxes, key=lambda b: b[0])


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=float)
    boxes = _panel_boxes(arr)
    if len(boxes) != 4:
        raise ValueError(f"expected four visible panels, found {len(boxes)}")
    scores = []
    clearances = []
    for x0, y0, x1, y1 in boxes:
        crop = arr[y0:y1 + 1, x0:x1 + 1]
        circles = []
        for color in RING_PALETTE:
            pts = np.column_stack(np.nonzero(np.linalg.norm(crop - color, axis=2) < 55.0)).astype(float)
            circles.append(_fit_circle(pts))
        dot_groups = _components(np.linalg.norm(crop - DOT_COLOR, axis=2) < 50.0, 35)
        if len(dot_groups) != 4:
            raise ValueError(f"expected four teal dots, found {len(dot_groups)}")
        dots = [(float(group[:, 1].mean()), float(group[:, 0].mean())) for group in dot_groups]
        score = 0
        local_clear = []
        for x, y in dots:
            bits = []
            for cx, cy, radius in circles:
                signed = radius - math.hypot(x - cx, y - cy)
                bits.append(signed > 0.0)
                local_clear.append(abs(signed))
            score += sum(bits) == 1
        scores.append(score)
        clearances.append(min(local_clear))
    order = sorted(range(4), key=lambda i: scores[i], reverse=True)
    if scores[order[0]] != 4 or scores[order[0]] - scores[order[1]] < 1:
        raise ValueError(f"ambiguous exclusive-membership scores {scores}")
    if min(clearances) < 4.0:
        raise ValueError(f"finite-resolution boundary clearance {clearances}")
    return LABELS[order[0]]
