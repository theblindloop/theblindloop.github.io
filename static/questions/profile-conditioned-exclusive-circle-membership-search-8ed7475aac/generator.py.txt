"""Deterministic scene sampler and Pillow renderer for exclusive membership."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 800, 280
PANEL_W, PANEL_H = 184, 228
PANEL_X0, PANEL_GAP, PANEL_Y = 20, 12, 34
LABELS = ("A", "B", "C", "D")
RING_COLORS = {"amber": (225, 145, 24), "magenta": (202, 52, 112)}
DOT_COLOR = (20, 151, 151)
RADIUS = 54.0


def _point(center: tuple[float, float], e: tuple[float, float], u: float, v: float) -> tuple[float, float]:
    return (center[0] + u * e[0] - v * e[1], center[1] + u * e[1] + v * e[0])


def _membership(_audit_anchor: None, point: dict, circle: dict) -> bool:
    return math.hypot(float(point["x"]) - float(circle["x"]),
                      float(point["y"]) - float(circle["y"])) < float(circle["r"])


def _panel_score(_audit_anchor: None, panel: dict) -> int:
    # ``panels`` is the sole top-level label input. These nested records are
    # precisely the visible contents rendered from that field.
    return sum(sum(_membership(None, dot, circle) for circle in panel["circles"]) == 1
               for dot in panel["dots"])


def _panel_clearance(panel: dict) -> float:
    return min(abs(math.hypot(float(dot["x"]) - float(circle["x"]),
                              float(dot["y"]) - float(circle["y"])) - float(circle["r"]))
               for dot in panel["dots"] for circle in panel["circles"])


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    winner = int(seed) % 4
    invalid_kinds = ["overlap", "outside", "overlap"]
    rng.shuffle(invalid_kinds)
    panels = []
    di = 0
    for index in range(4):
        angle = rng.uniform(0.0, 2.0 * math.pi)
        e = (math.cos(angle), math.sin(angle))
        center = (92.0 + rng.uniform(-2.5, 2.5), 119.0 + rng.uniform(-2.5, 2.5))
        circles = []
        for color, u in (("amber", -26.0), ("magenta", 26.0)):
            x, y = _point(center, e, u, 0.0)
            circles.append({"color": color, "x": round(x, 4), "y": round(y, 4), "r": RADIUS})

        # Four equally styled dots. In the survivor, two occupy each exclusive
        # crescent. Distractors replace exactly one with an overlap/outside dot.
        slots = [(-1, -22.0), (-1, 22.0), (1, -22.0), (1, 22.0)]
        rng.shuffle(slots)
        dots = []
        for side, v0 in slots:
            u = side * rng.uniform(57.0, 62.0)
            v = v0 + rng.uniform(-3.0, 3.0)
            x, y = _point(center, e, u, v)
            dots.append({"x": round(x, 4), "y": round(y, 4)})
        if index != winner:
            kind = invalid_kinds[di]
            di += 1
            # Near the visible circle-crossing zone, avoiding a conspicuous
            # central or remote marker while retaining broad boundary clearance.
            u = rng.uniform(-5.0, 5.0)
            v = rng.choice((-1.0, 1.0)) * (32.0 if kind == "overlap" else 75.0)
            x, y = _point(center, e, u, v)
            dots[rng.randrange(4)] = {"x": round(x, 4), "y": round(y, 4)}
        rng.shuffle(circles)
        rng.shuffle(dots)
        panels.append({"circles": circles, "dots": dots})
    return {"panels": panels}


def analytic_gold(scene: dict) -> str:
    scores = [_panel_score(None, panel) for panel in scene["panels"]]
    winners = [i for i, score in enumerate(scores) if score == max(scores)]
    if len(winners) != 1:
        raise ValueError(f"non-unique panel scores {scores}")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    scores = sorted((_panel_score(None, panel) for panel in scene["panels"]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    scores = sorted((_panel_score(None, panel) for panel in scene["panels"]), reverse=True)
    clearance = min(_panel_clearance(panel) for panel in scene["panels"])
    return scores[0] != 4 or scores[0] - scores[1] < 1 or clearance < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = {"panels": []}
    for panel in scene["panels"]:
        twin["panels"].append({
            "circles": [dict(x) for x in reversed(panel["circles"])],
            "dots": [dict(x) for x in reversed(panel["dots"])],
        })
    return [("reverse_unordered_records", twin)]


def render(scene: dict) -> Image.Image:
    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), (255, 255, 255))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default(size=19 * scale)
    for index, panel in enumerate(scene["panels"]):
        left = PANEL_X0 + index * (PANEL_W + PANEL_GAP)
        box = (left, PANEL_Y, left + PANEL_W, PANEL_Y + PANEL_H)
        draw.rounded_rectangle(tuple(v * scale for v in box), radius=8 * scale,
                               fill=(247, 249, 252), outline=(91, 101, 115), width=2 * scale)
        draw.text(((left + 8) * scale, (PANEL_Y + 6) * scale), LABELS[index],
                  fill=(27, 32, 40), font=font)
        for circle in sorted(panel["circles"], key=lambda item: item["color"]):
            x = left + float(circle["x"])
            y = PANEL_Y + float(circle["y"])
            r = float(circle["r"])
            bbox = tuple(v * scale for v in (x - r, y - r, x + r, y + r))
            draw.ellipse(bbox, outline=RING_COLORS[circle["color"]], width=4 * scale)
        for dot in sorted(panel["dots"], key=lambda item: (item["x"], item["y"])):
            x = left + float(dot["x"])
            y = PANEL_Y + float(dot["y"])
            r = 5.5
            bbox = tuple(v * scale for v in (x - r, y - r, x + r, y + r))
            draw.ellipse(bbox, fill=DOT_COLOR)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
