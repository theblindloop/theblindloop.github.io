"""Deterministic raster renderer for compact woven-crossing parity scenes."""

from __future__ import annotations

import copy
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 1000
HEIGHT = 1000
SUPERSAMPLE = 4
PALETTE = (
    (39, 94, 171),
    (196, 67, 74),
    (42, 137, 103),
    (173, 92, 174),
    (213, 119, 43),
    (34, 139, 151),
)
BACKGROUND = (247, 245, 239)
STATES = ("slash", "backslash")


def _sorted_badges(scene: dict[str, Any]) -> list[dict[str, Any]]:
    return sorted(scene["badges"], key=lambda badge: int(badge["cx"]))


def _state_values(scene: dict[str, Any]) -> list[str]:
    return [str(badge["state"]) for badge in _sorted_badges(scene)]


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the equality of the two visible crossing states.

    This intentionally reads only ``badges``: the crossing state is the sole
    analytic label input and is rendered as the continuous/slit diagonal.
    """

    states = _state_values(scene)
    if len(states) != 2 or any(state not in STATES for state in states):
        raise ValueError("scene must contain exactly two valid badges")
    return "yes" if states[0] == states[1] else "no"


def margin(scene: dict[str, Any]) -> float:
    badges = _sorted_badges(scene)
    if len(badges) != 2:
        return -1.0
    left, right = badges
    left_size = float(left["size"])
    right_size = float(right["size"])
    min_size = min(left_size, right_size)
    # The first term is the visible length of the background slit after the
    # stroke width is removed. The other terms keep components and canvas
    # borders comfortably separated.
    slit_clearance = 0.58 * 0.39 * min_size - 0.17 * min_size * 0.25
    component_gap = float(right["cx"]) - float(left["cx"]) - 0.39 * (left_size + right_size) - 12.0
    edge_clearance = min(
        float(left["cx"]) - 0.39 * left_size,
        float(right["cx"]) + 0.39 * right_size,
        float(left["cy"]) - 0.39 * left_size,
        float(right["cy"]) - 0.39 * right_size,
        HEIGHT - (float(left["cy"]) + 0.39 * left_size),
        HEIGHT - (float(right["cy"]) + 0.39 * right_size),
    )
    return float(min(slit_clearance, component_gap, edge_clearance))


def is_quarantined(scene: dict[str, Any]) -> bool:
    badges = _sorted_badges(scene)
    if len(badges) != 2:
        return True
    if any(str(badge.get("state")) not in STATES for badge in badges):
        return True
    if any(float(badge.get("size", 0)) < 48.0 for badge in badges):
        return True
    if float(badges[1]["cx"]) - float(badges[0]["cx"]) < 155.0:
        return True
    return margin(scene) < 6.0


def _point(cx: float, cy: float, half: float, slant: int, t: float) -> tuple[float, float]:
    return (cx + t * half, cy + slant * t * half)


def _line(
    draw: ImageDraw.ImageDraw,
    first: tuple[float, float],
    second: tuple[float, float],
    width: float,
    color: tuple[int, int, int],
) -> None:
    draw.line(
        [(round(first[0]), round(first[1])), (round(second[0]), round(second[1]))],
        fill=color,
        width=max(1, round(width)),
    )


def _draw_badge(
    draw: ImageDraw.ImageDraw,
    badge: dict[str, Any],
    scale: int,
) -> None:
    cx = float(badge["cx"]) * scale
    cy = float(badge["cy"]) * scale
    size = float(badge["size"]) * scale
    half = 0.39 * size
    stroke = 0.17 * size
    color = tuple(int(value) for value in badge["color"])
    state = str(badge["state"])
    if state not in STATES:
        raise ValueError(f"unknown crossing state {state!r}")

    continuous_slant = -1 if state == "slash" else 1
    under_slant = -continuous_slant
    slit_half_parameter = 0.58

    # Paint the interrupted band first. The continuous band then visibly sits
    # over it at the centre, while both bands remain one connected badge.
    _line(
        draw,
        _point(cx, cy, half, under_slant, -1.0),
        _point(cx, cy, half, under_slant, -slit_half_parameter),
        stroke,
        color,
    )
    _line(
        draw,
        _point(cx, cy, half, under_slant, slit_half_parameter),
        _point(cx, cy, half, under_slant, 1.0),
        stroke,
        color,
    )
    _line(
        draw,
        _point(cx, cy, half, continuous_slant, -1.0),
        _point(cx, cy, half, continuous_slant, 1.0),
        stroke,
        color,
    )


def render(scene: dict[str, Any]) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(int(value) for value in scene["background"])
    image = Image.new("RGB", (width * SUPERSAMPLE, height * SUPERSAMPLE), background)
    draw = ImageDraw.Draw(image)
    for badge in _sorted_badges(scene):
        _draw_badge(draw, badge, SUPERSAMPLE)
    return image.resize((width, height), Image.Resampling.LANCZOS).convert("RGB")


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    # Each template keeps the two crossing cues in one small area while moving
    # that area among several raster cells. The crossing cells stay separated.
    templates = (
        (312, 437, 562, 437),
        (437, 437, 687, 437),
        (312, 562, 562, 562),
        (437, 562, 687, 562),
    )
    left_x, base_y, right_x, _ = templates[rng.randrange(len(templates))]
    y = base_y
    left = left_x
    right = right_x
    left_size = rng.randint(58, 62)
    right_size = rng.randint(58, 62)
    fit = int(seed) % 2 == 0
    left_state = STATES[rng.randrange(len(STATES))]
    right_state = left_state if fit else ("backslash" if left_state == "slash" else "slash")
    colors = rng.sample(PALETTE, 2)
    badges = [
        {
            "cx": left,
            "cy": y,
            "size": left_size,
            "state": left_state,
            "color": list(colors[0]),
        },
        {
            "cx": right,
            "cy": y,
            "size": right_size,
            "state": right_state,
            "color": list(colors[1]),
        },
    ]
    return {
        "badges": badges,
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
    }


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    twin = copy.deepcopy(scene)
    twin["badges"] = list(reversed(twin["badges"]))
    return [("unordered_badge_record_permutation", twin)]
