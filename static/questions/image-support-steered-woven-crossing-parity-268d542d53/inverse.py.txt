"""Independent image-only inverse for woven crossing parity."""

from __future__ import annotations

import numpy as np
from PIL import Image

SAMPLE_T = 0.25


def _background(array: np.ndarray) -> np.ndarray:
    border = np.concatenate(
        (
            array[:3].reshape(-1, 3),
            array[-3:].reshape(-1, 3),
            array[:, :3].reshape(-1, 3),
            array[:, -3:].reshape(-1, 3),
        ),
        axis=0,
    )
    return np.median(border, axis=0)


def _foreground_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    background = _background(array)
    distance = np.sqrt(np.sum((array - background) ** 2, axis=2))
    return distance > 28.0


def _x_runs(mask: np.ndarray) -> list[tuple[int, int]]:
    occupied = np.any(mask, axis=0)
    starts = np.flatnonzero(occupied & ~np.r_[False, occupied[:-1]])
    ends = np.flatnonzero(occupied & ~np.r_[occupied[1:], False])
    return [(int(start), int(end) + 1) for start, end in zip(starts, ends, strict=True)]


def _component_boxes(mask: np.ndarray) -> list[tuple[np.ndarray, tuple[int, int, int, int]]]:
    runs = _x_runs(mask)
    if len(runs) != 2:
        raise ValueError(f"expected two separated woven badges, found {len(runs)} x-runs")
    components: list[tuple[np.ndarray, tuple[int, int, int, int]]] = []
    for start, end in runs:
        component = mask[:, start:end]
        ys, xs = np.nonzero(component)
        if len(xs) < 280:
            raise ValueError("woven badge component is too small")
        x0 = int(xs.min()) + start
        x1 = int(xs.max()) + start + 1
        y0 = int(ys.min())
        y1 = int(ys.max()) + 1
        full = np.zeros_like(mask)
        full[:, start:end] = component
        components.append((full, (x0, y0, x1, y1)))
    components.sort(key=lambda item: item[1][0])
    return components


def _patch_fraction(mask: np.ndarray, x: float, y: float, radius: int = 3) -> float:
    height, width = mask.shape
    cx = int(round(x))
    cy = int(round(y))
    x0, x1 = max(0, cx - radius), min(width, cx + radius + 1)
    y0, y1 = max(0, cy - radius), min(height, cy + radius + 1)
    patch = mask[y0:y1, x0:x1]
    return float(np.mean(patch)) if patch.size else 0.0


def _diagonal_score(component: np.ndarray, box: tuple[int, int, int, int], slant: int) -> float:
    x0, y0, x1, y1 = box
    cx = (x0 + x1 - 1) / 2.0
    cy = (y0 + y1 - 1) / 2.0
    half = 0.40 * min(float(x1 - x0), float(y1 - y0))
    first = _patch_fraction(component, cx - SAMPLE_T * half, cy - slant * SAMPLE_T * half)
    second = _patch_fraction(component, cx + SAMPLE_T * half, cy + slant * SAMPLE_T * half)
    return min(first, second)


def _crossing_state(component: np.ndarray, box: tuple[int, int, int, int]) -> str:
    slash = _diagonal_score(component, box, -1)
    backslash = _diagonal_score(component, box, 1)
    if max(slash, backslash) < 0.28 or abs(slash - backslash) < 0.22:
        raise ValueError("central crossing is missing or ambiguous")
    return "slash" if slash > backslash else "backslash"


def decision_from_image(image: Image.Image) -> str:
    """Recover the equality decision from RGB pixels only."""

    if image.width < 200 or image.height < 160:
        raise ValueError("image is too small for the woven-badge contract")
    mask = _foreground_mask(image)
    components = _component_boxes(mask)
    states = [_crossing_state(component, box) for component, box in components]
    return "yes" if states[0] == states[1] else "no"
