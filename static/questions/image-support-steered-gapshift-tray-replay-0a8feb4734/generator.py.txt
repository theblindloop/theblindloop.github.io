"""Renderer and analytic gold for the `gapshift_tray_replay` world.

One small framed cassette holds every decisive mark: a three-by-three tray of
eight coloured tiles and one empty square, and immediately below it a short
strip of four arrow glyphs.  Each arrow says which way the empty square moves;
the tile it moves onto slides back into the square the gap just left.  One tray
cell carries a thick black ring, and the question asks which colour stands in
the ringed cell once all four moves have been carried out.

`render` is a pure function of the scene dictionary.  Every scene field is a
plain JSON type, so a JSON round trip reproduces the identical raster.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "gapshift_tray_replay-renderer-0.1.0"

CANVAS_W = 900
CANVAS_H = 640
BG = (246, 246, 246)
PANEL = (252, 252, 252)
BED = (228, 228, 228)
INK = (0, 0, 0)

# cassette layout (all constants are fixed; only the position is sampled)
PAD = 10
CELL = 38
GAP = 5
PITCH = CELL + GAP            # 43
TRAY = 3 * CELL + 2 * GAP     # 124
DIVIDE = 6
MCELL = 28
MGAP = 4
MPITCH = MCELL + MGAP         # 32
STRIP_W = 4 * MCELL + 3 * MGAP  # 124
FRAME = 4
INNER_W = TRAY + 2 * PAD                       # 144
INNER_H = PAD + TRAY + DIVIDE + MCELL + PAD    # 178
OUTER_W = INNER_W + 2 * FRAME                  # 152
OUTER_H = INNER_H + 2 * FRAME                  # 186

COLOUR_NAMES = ("red", "green", "blue", "yellow")
COLOURS = {
    "red": (208, 32, 40),
    "green": (26, 140, 58),
    "blue": (36, 72, 196),
    "yellow": (226, 168, 20),
}
DIRECTIONS = ("U", "D", "L", "R")
_DELTA = {"U": (-1, 0), "D": (1, 0), "L": (0, -1), "R": (0, 1)}
_OPPOSITE = {"U": "D", "D": "U", "L": "R", "R": "L"}

N_MOVES = 4
APEX_MIN = 9
APEX_MAX = 12
RING_MIN = 3
RING_MAX = 5
MARGIN_FLOOR = 2.5


# ----------------------------------------------------------------- sampling


def _legal(hole: int, direction: str) -> bool:
    dr, dc = _DELTA[direction]
    r, c = divmod(hole, 3)
    return 0 <= r + dr < 3 and 0 <= c + dc < 3


def _apply(cells: list, hole: int, direction: str) -> tuple:
    dr, dc = _DELTA[direction]
    r, c = divmod(hole, 3)
    target = (r + dr) * 3 + (c + dc)
    out = list(cells)
    out[hole], out[target] = out[target], out[hole]
    return out, target


def _replay(cells: list, moves: list) -> list:
    grid = list(cells)
    hole = grid.index(-1)
    for direction in moves:
        grid, hole = _apply(grid, hole, direction)
    return grid


def _sample_decor(rng: random.Random, box: list) -> list:
    """Pale inert marks placed strictly outside the cassette bounding box."""
    x0, y0, x1, y1 = box
    decor = []
    attempts = 0
    while len(decor) < 13 and attempts < 400:
        attempts += 1
        w = rng.randint(16, 62)
        h = rng.randint(16, 62)
        px = rng.randint(6, CANVAS_W - w - 6)
        py = rng.randint(6, CANVAS_H - h - 6)
        if px < x1 + 10 and px + w > x0 - 10 and py < y1 + 10 and py + h > y0 - 10:
            continue
        kind = rng.choice(["rect", "disk", "wedge", "bar"])
        tone = rng.randrange(198, 221, 2)
        decor.append({"kind": kind, "box": [px, py, px + w, py + h], "tone": tone})
    return decor


def sample_scene(seed: int) -> dict:
    """Deterministic scene sample; the answer colour is balanced by seed."""
    rng = random.Random(seed * 6151 + 907)
    target = COLOUR_NAMES[seed % len(COLOUR_NAMES)]

    scene = None
    for _ in range(600):
        cells = [0, 0, 1, 1, 2, 2, 3, 3, -1]
        rng.shuffle(cells)
        hole = cells.index(-1)

        moves = []
        cursor = hole
        for _ in range(N_MOVES):
            options = [d for d in DIRECTIONS if _legal(cursor, d)]
            if moves:
                options = [d for d in options if d != _OPPOSITE[moves[-1]]] or options
            direction = rng.choice(sorted(options))
            moves.append(direction)
            _, cursor = _apply(cells, cursor, direction)

        final = _replay(cells, moves)
        movers = [
            i
            for i in range(9)
            if final[i] != -1 and cells[i] != -1 and final[i] != cells[i]
        ]
        if not movers:
            continue
        ring = rng.choice(movers)
        if COLOUR_NAMES[final[ring]] != target:
            continue

        ox = rng.randint(24, CANVAS_W - OUTER_W - 24)
        oy = rng.randint(24, CANVAS_H - OUTER_H - 24)
        scene = {
            "canvas": [CANVAS_W, CANVAS_H],
            "origin": [ox, oy],
            "cells": [int(v) for v in cells],
            "moves": list(moves),
            "apex": [rng.randint(APEX_MIN, APEX_MAX) for _ in range(N_MOVES)],
            "ring_cell": int(ring),
            "ring_thickness": rng.randint(RING_MIN, RING_MAX),
            "decor": _sample_decor(rng, [ox, oy, ox + OUTER_W, oy + OUTER_H]),
        }
        break

    if scene is None:  # pragma: no cover - the balanced search always succeeds
        raise RuntimeError(f"scene sampling failed for seed {seed}")
    return scene


# ---------------------------------------------------------------- rendering


def _tray_origin(scene: dict) -> tuple:
    ox, oy = scene["origin"]
    return ox + FRAME + PAD, oy + FRAME + PAD


def _strip_origin(scene: dict) -> tuple:
    tx, ty = _tray_origin(scene)
    return tx, ty + TRAY + DIVIDE


def render(scene: dict) -> Image.Image:
    """Pure function of `scene`: no randomness, no mutation, flat colours only."""
    width, height = int(scene["canvas"][0]), int(scene["canvas"][1])
    image = Image.new("RGB", (width, height), BG)
    draw = ImageDraw.Draw(image)

    for item in scene["decor"]:
        x0, y0, x1, y1 = (int(v) for v in item["box"])
        tone = int(item["tone"])
        fill = (tone, tone, tone)
        kind = item["kind"]
        if kind == "rect":
            draw.rectangle([x0, y0, x1, y1], fill=fill)
        elif kind == "disk":
            draw.ellipse([x0, y0, x1, y1], fill=fill)
        elif kind == "wedge":
            draw.polygon([(x0, y1), (x1, y1), ((x0 + x1) // 2, y0)], fill=fill)
        else:
            mid = (y0 + y1) // 2
            draw.rectangle([x0, mid - 3, x1, mid + 3], fill=fill)

    ox, oy = int(scene["origin"][0]), int(scene["origin"][1])
    draw.rectangle([ox, oy, ox + OUTER_W - 1, oy + OUTER_H - 1], fill=PANEL)
    for k in range(FRAME):
        draw.rectangle(
            [ox + k, oy + k, ox + OUTER_W - 1 - k, oy + OUTER_H - 1 - k], outline=INK
        )

    tx, ty = _tray_origin(scene)
    draw.rectangle([tx, ty, tx + TRAY - 1, ty + TRAY - 1], fill=BED)
    cells = scene["cells"]
    for index in range(9):
        row, col = divmod(index, 3)
        cx0 = tx + col * PITCH
        cy0 = ty + row * PITCH
        value = int(cells[index])
        if value >= 0:
            draw.rectangle(
                [cx0, cy0, cx0 + CELL - 1, cy0 + CELL - 1],
                fill=COLOURS[COLOUR_NAMES[value]],
            )

    ring = int(scene["ring_cell"])
    thickness = int(scene["ring_thickness"])
    row, col = divmod(ring, 3)
    rx0 = tx + col * PITCH
    ry0 = ty + row * PITCH
    for k in range(thickness):
        draw.rectangle(
            [rx0 + k, ry0 + k, rx0 + CELL - 1 - k, ry0 + CELL - 1 - k], outline=INK
        )

    sx, sy = _strip_origin(scene)
    draw.rectangle([sx, sy, sx + STRIP_W - 1, sy + MCELL - 1], fill=BED)
    for slot, direction in enumerate(scene["moves"]):
        cx = sx + slot * MPITCH + MCELL // 2
        cy = sy + MCELL // 2
        apex = int(scene["apex"][slot])
        half = 8
        if direction in ("U", "D"):
            sign = -1 if direction == "U" else 1
            points = [
                (cx, cy + sign * apex),
                (cx - half, cy - sign * apex),
                (cx + half, cy - sign * apex),
            ]
        else:
            sign = -1 if direction == "L" else 1
            points = [
                (cx + sign * apex, cy),
                (cx - sign * apex, cy - half),
                (cx - sign * apex, cy + half),
            ]
        draw.polygon(points, fill=INK)

    return image


# ------------------------------------------------------------------- labels


def analytic_gold(scene: dict) -> str:
    """Colour standing in the ringed tray cell after the four gap moves."""
    final = _replay([int(v) for v in scene["cells"]], list(scene["moves"]))
    value = final[int(scene["ring_cell"])]
    if value < 0:
        raise ValueError("ringed cell finishes empty; scene should be quarantined")
    return COLOUR_NAMES[value]


def margin(scene: dict) -> float:
    """Smallest pixel clearance of any decisive glyph in the cassette.

    The arrow term is the distance between an arrow's filled centroid and its
    slot centre, which is what separates the four possible readings of a move;
    the ring term is the ring's stroke width.
    """
    arrows = min(int(a) / 3.0 for a in scene["apex"])
    return float(round(min(arrows, float(int(scene["ring_thickness"]))), 6))


def is_quarantined(scene: dict) -> bool:
    """Reject replays that are not uniquely readable from the pixels."""
    cells = [int(v) for v in scene["cells"]]
    if sorted(cells) != [-1, 0, 0, 1, 1, 2, 2, 3, 3]:
        return True
    moves = list(scene["moves"])
    if len(moves) != N_MOVES or any(d not in DIRECTIONS for d in moves):
        return True
    hole = cells.index(-1)
    for direction in moves:
        if not _legal(hole, direction):
            return True
        _, hole = _apply(cells, hole, direction)
    final = _replay(cells, moves)
    ring = int(scene["ring_cell"])
    if final[ring] < 0:
        return True
    if cells[ring] < 0 or final[ring] == cells[ring]:
        return True
    return margin(scene) < MARGIN_FLOOR


def latent_symmetries(scene: dict) -> list:
    """Declared `not_applicable`: no latent transform participates in the gold."""
    return []
