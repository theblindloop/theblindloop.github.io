"""Image-only inverse arm: recover the proximity graph and count its groups.

This module never sees the latent scene. From the PNG alone it

1. separates three pixel classes by luminance/chroma alone: the near-black
   reference bar, the coloured dot ink, and the pale background;
2. measures the linking threshold as the horizontal pixel extent of the bar;
3. segments the dot ink into connected components (row-run union-find) and
   takes each component's centroid;
4. rebuilds the link graph {i~j : ||c_i - c_j|| < threshold} and returns the
   number of connected components.

Nothing about the answer is read locally: the returned symbol is a property of
the reconstructed global proximity graph. Colour, dot size and dot count are
never used as evidence.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "proximity_group_count-oracle-0.1.0"

DARK_MAX = 80  # every channel below this => reference-bar ink
INK_MIN_DELTA = 55  # distance from the background colour => dot ink
MIN_DOT_PIXELS = 40
MIN_BAR_PIXELS = 200


class OracleError(ValueError):
    """The image does not contain the structure the question declares."""


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Connected components (4-connectivity) via row runs and union-find."""
    height, width = mask.shape
    parent: list[int] = []

    def find(i: int) -> int:
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i: int, j: int) -> None:
        ri, rj = find(i), find(j)
        if ri != rj:
            parent[ri] = rj

    runs: list[tuple[int, int, int, int]] = []  # (row, start, end_exclusive, id)
    previous: list[tuple[int, int, int, int]] = []
    for row in range(height):
        line = mask[row]
        if not line.any():
            previous = []
            continue
        padded = np.concatenate(([False], line, [False]))
        edges = np.flatnonzero(padded[1:] != padded[:-1])
        current: list[tuple[int, int, int, int]] = []
        for start, end in zip(edges[0::2], edges[1::2]):
            index = len(parent)
            parent.append(index)
            entry = (row, int(start), int(end), index)
            current.append(entry)
            runs.append(entry)
            for _, p_start, p_end, p_index in previous:
                if p_start < end and start < p_end:
                    union(index, p_index)
        previous = current

    groups: dict[int, list[tuple[int, int, int]]] = {}
    for row, start, end, index in runs:
        groups.setdefault(find(index), []).append((row, start, end))

    out: list[np.ndarray] = []
    for pixel_runs in groups.values():
        coordinates = []
        for row, start, end in pixel_runs:
            for column in range(start, end):
                coordinates.append((column, row))
        out.append(np.asarray(coordinates, dtype=np.float64))
    return out


def measure(image: Image.Image) -> tuple[list[tuple[float, float]], float]:
    """Return (dot centres, linking threshold in pixels) from pixels alone."""
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    corners = np.array(
        [arr[0, 0], arr[0, -1], arr[-1, 0], arr[-1, -1]], dtype=np.int16
    )
    background = np.median(corners, axis=0)

    dark = (arr < DARK_MAX).all(axis=-1)
    distance = np.abs(arr - background).max(axis=-1)
    ink = (distance >= INK_MIN_DELTA) & ~dark

    bar_pixels = np.flatnonzero(dark.any(axis=0))
    if dark.sum() < MIN_BAR_PIXELS or bar_pixels.size == 0:
        raise OracleError("no reference bar found in the image")
    threshold = float(bar_pixels.max() - bar_pixels.min() + 1)

    centres: list[tuple[float, float]] = []
    for component in _components(ink):
        if len(component) < MIN_DOT_PIXELS:
            continue
        centres.append((float(component[:, 0].mean()) + 0.5, float(component[:, 1].mean()) + 0.5))
    if len(centres) < 2:
        raise OracleError(f"only {len(centres)} dot components found")
    return centres, threshold


def decision_from_image(image: Image.Image) -> str:
    """Number of proximity groups recovered from the image alone."""
    centres, threshold = measure(image)
    parent = list(range(len(centres)))

    def find(i: int) -> int:
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    for i in range(len(centres)):
        for j in range(i + 1, len(centres)):
            if math.dist(centres[i], centres[j]) < threshold:
                ri, rj = find(i), find(j)
                if ri != rj:
                    parent[ri] = rj
    return str(len({find(i) for i in range(len(centres))}))
