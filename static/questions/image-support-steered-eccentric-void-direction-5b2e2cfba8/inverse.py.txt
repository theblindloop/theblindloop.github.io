"""Independent pixel-only inverse arm for the eccentric circular opening."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class OracleError(ValueError):
    pass


def _components(mask: np.ndarray) -> list[tuple[np.ndarray, bool]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[np.ndarray, bool]] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points = []
        touches = False
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            touches |= cy == 0 or cx == 0 or cy == height - 1 or cx == width - 1
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        found.append((np.asarray(points, dtype=np.int32), touches))
    return found


def _measure(image: Image.Image) -> tuple[float, float]:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    if array.ndim != 3 or array.shape[2] != 3 or min(array.shape[:2]) < 64:
        raise OracleError("unsupported image")
    border = np.concatenate(
        [array[:3].reshape(-1, 3), array[-3:].reshape(-1, 3),
         array[:, :3].reshape(-1, 3), array[:, -3:].reshape(-1, 3)]
    )
    background = np.median(border, axis=0)
    delta = np.linalg.norm(array.astype(np.float32) - background.astype(np.float32), axis=2)
    foreground = delta > 55.0
    points = np.argwhere(foreground)
    if len(points) < 8000:
        raise OracleError("plaque is absent or too small")
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    width, height = int(x1 - x0 + 1), int(y1 - y0 + 1)
    if not 145 <= width <= 180 or not 145 <= height <= 180 or abs(width - height) > 3:
        raise OracleError("outer boundary is not a supported circle")
    outer_x = (float(x0) + float(x1)) / 2.0
    outer_y = (float(y0) + float(y1)) / 2.0
    radius = (width + height - 2.0) / 4.0

    # The opening is the sole background component enclosed by foreground.
    crop_bg = ~foreground[y0 : y1 + 1, x0 : x1 + 1]
    enclosed = [pts for pts, touches in _components(crop_bg) if not touches and len(pts) > 1200]
    if len(enclosed) != 1:
        raise OracleError("expected exactly one enclosed opening")
    hole = enclosed[0]
    hy0, hx0 = hole.min(axis=0)
    hy1, hx1 = hole.max(axis=0)
    hole_width, hole_height = int(hx1 - hx0 + 1), int(hy1 - hy0 + 1)
    if not 53 <= hole_width <= 72 or abs(hole_width - hole_height) > 3:
        raise OracleError("opening is not a supported circle")
    hole_x = float(x0) + (float(hx0) + float(hx1)) / 2.0
    hole_y = float(y0) + (float(hy0) + float(hy1)) / 2.0
    hole_radius = (hole_width + hole_height - 2.0) / 4.0
    if abs(hole_y - outer_y) > 2.0:
        raise OracleError("opening is not horizontally displaced")

    yy, xx = np.indices(foreground.shape, dtype=np.float32)
    predicted = ((xx - outer_x) ** 2 + (yy - outer_y) ** 2 <= (radius + 0.75) ** 2) & (
        (xx - hole_x) ** 2 + (yy - hole_y) ** 2 > (hole_radius + 0.25) ** 2
    )
    union = np.count_nonzero(predicted | foreground)
    overlap = np.count_nonzero(predicted & foreground)
    if union == 0 or overlap / union < 0.972:
        raise OracleError("plaque pixels do not support two complete circular fits")
    return outer_x, hole_x


def decision_from_image(image: Image.Image) -> str:
    outer_x, hole_x = _measure(image)
    displacement = hole_x - outer_x
    if abs(displacement) < 3.0:
        raise OracleError("opening direction is ambiguous")
    return "right" if displacement > 0 else "left"
