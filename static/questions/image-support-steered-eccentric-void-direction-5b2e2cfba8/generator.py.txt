"""Deterministic scene sampler and renderer for eccentric_void_direction."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS = 384
PALETTES = (
    ((246, 242, 232), (31, 111, 139)),
    ((244, 246, 239), (167, 67, 79)),
    ((242, 240, 248), (79, 91, 166)),
    ((248, 243, 235), (44, 126, 91)),
)


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    outer_radius = rng.randint(76, 88)
    hole_radius = rng.randint(27, 35)
    # Deterministically expose a few observable near-boundary cases while
    # guaranteeing alternating off-boundary answer identities.
    magnitude = rng.randint(6, 10) if seed % 11 == 0 else rng.randint(18, 28)
    sign = 1 if seed % 2 == 0 else -1
    margin = outer_radius + 14
    cx = rng.randint(margin, CANVAS - margin)
    cy = rng.randint(margin, CANVAS - margin)
    return {
        "canvas_size": CANVAS,
        "outer_center": [cx, cy],
        "outer_radius": outer_radius,
        "hole_radius": hole_radius,
        "hole_dx": sign * magnitude,
        "palette_index": rng.randrange(len(PALETTES)),
    }


def render(scene: dict[str, Any]) -> Image.Image:
    size = int(scene["canvas_size"])
    cx, cy = (int(value) for value in scene["outer_center"])
    outer_radius = int(scene["outer_radius"])
    hole_radius = int(scene["hole_radius"])
    hole_dx = int(scene["hole_dx"])
    background, ink = PALETTES[int(scene["palette_index"]) % len(PALETTES)]
    image = Image.new("RGB", (size, size), background)
    draw = ImageDraw.Draw(image)
    draw.ellipse(
        (cx - outer_radius, cy - outer_radius, cx + outer_radius, cy + outer_radius),
        fill=ink,
    )
    hx = cx + hole_dx
    draw.ellipse(
        (hx - hole_radius, cy - hole_radius, hx + hole_radius, cy + hole_radius),
        fill=background,
    )
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    return "right" if int(scene["hole_dx"]) > 0 else "left"


def margin(scene: dict[str, Any]) -> float:
    return float(abs(int(scene["hole_dx"])))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return abs(int(scene["hole_dx"])) < 12


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict]]:
    return []
