"""Independent image-only inverse for Opposed Row-Shear Residual."""

from __future__ import annotations

import numpy as np
from PIL import Image

N = 7
TEAL = np.array((22, 139, 139), dtype=np.int16)
CORAL = np.array((231, 111, 81), dtype=np.int16)
DARK = np.array((39, 50, 68), dtype=np.float64)
LIGHT = np.array((238, 232, 216), dtype=np.float64)


def _templates() -> dict[str, np.ndarray]:
    x = np.zeros((N, N), dtype=np.uint8)
    for i in range(N):
        x[i, i] = x[i, N - 1 - i] = 1
    plus = np.zeros((N, N), dtype=np.uint8)
    plus[3, :] = plus[:, 3] = 1
    tee = np.zeros((N, N), dtype=np.uint8)
    tee[0, :] = tee[:, 3] = 1
    ell = np.zeros((N, N), dtype=np.uint8)
    ell[:, 0] = ell[6, :] = 1
    return {"X": x, "plus": plus, "T": tee, "L": ell}


TEMPLATES = _templates()


def _bbox_for_color(arr: np.ndarray, color: np.ndarray) -> tuple[int, int, int, int]:
    distance = np.max(np.abs(arr.astype(np.int16) - color), axis=2)
    ys, xs = np.nonzero(distance <= 8)
    if len(xs) < 900:
        raise ValueError("colored panel fiducial is missing or damaged")
    x0, x1, y0, y1 = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    if not (134 <= x1 - x0 + 1 <= 144 and 134 <= y1 - y0 + 1 <= 144):
        raise ValueError("colored panel fiducial has invalid extent")
    # A damaged border must abstain even when its extreme pixels survive.
    border = np.zeros(distance.shape, dtype=bool)
    border[y0:y0 + 8, x0:x1 + 1] = True
    border[y1 - 7:y1 + 1, x0:x1 + 1] = True
    border[y0:y1 + 1, x0:x0 + 8] = True
    border[y0:y1 + 1, x1 - 7:x1 + 1] = True
    if float(np.mean(distance[border] <= 8)) < 0.86:
        raise ValueError("colored panel frame is materially corrupted")
    return x0, y0, x1, y1


def _read_grid(arr: np.ndarray, box: tuple[int, int, int, int]) -> np.ndarray:
    x0, y0, _, _ = box
    grid = np.zeros((N, N), dtype=np.uint8)
    for row in range(N):
        for col in range(N):
            x = x0 + 11 + col * 17
            y = y0 + 11 + row * 17
            patch = arr[y:y + 15, x:x + 15].astype(np.float64)
            dark_distance = np.linalg.norm(patch - DARK, axis=2)
            light_distance = np.linalg.norm(patch - LIGHT, axis=2)
            recognized = np.minimum(dark_distance, light_distance) < 18.0
            if patch.shape[:2] != (15, 15) or float(recognized.mean()) < 0.93:
                raise ValueError("grid cell is missing or corrupted")
            grid[row, col] = int(float(dark_distance.mean()) < float(light_distance.mean()))
    return grid


def _shear(grid: np.ndarray, direction: int) -> np.ndarray:
    return np.stack([np.roll(grid[row], direction * row) for row in range(N)])


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    if arr.ndim != 3 or arr.shape[0] < 250 or arr.shape[1] < 600:
        raise ValueError("raster is too small")
    teal = _read_grid(arr, _bbox_for_color(arr, TEAL))
    coral = _read_grid(arr, _bbox_for_color(arr, CORAL))
    residual = _shear(teal, 1) ^ _shear(coral, -1)
    ranked = sorted((int(np.count_nonzero(residual != glyph)), label) for label, glyph in TEMPLATES.items())
    if len(ranked) < 2 or ranked[1][0] - ranked[0][0] < 3:
        raise ValueError("template decision is ambiguous")
    return ranked[0][1]
