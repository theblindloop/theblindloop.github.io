"""Deterministic renderer and analytic arm for Opposed Row-Shear Residual."""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 800, 420
N = 7
PANEL = 140
PANEL_Y = 120
LEFT_X, RIGHT_X = 130, 530
CELL = 15
GAP = 2
GRID_OFFSET = 11

BG = (248, 246, 240)
CARD = (255, 255, 255)
TEAL = (22, 139, 139)
CORAL = (231, 111, 81)
DARK = (39, 50, 68)
LIGHT = (238, 232, 216)

LABELS = ("X", "plus", "T", "L")


def _templates() -> dict[str, np.ndarray]:
    out: dict[str, np.ndarray] = {}
    x = np.zeros((N, N), dtype=np.uint8)
    for i in range(N):
        x[i, i] = 1
        x[i, N - 1 - i] = 1
    out["X"] = x
    plus = np.zeros((N, N), dtype=np.uint8)
    plus[3, :] = 1
    plus[:, 3] = 1
    out["plus"] = plus
    tee = np.zeros((N, N), dtype=np.uint8)
    tee[0, :] = 1
    tee[:, 3] = 1
    out["T"] = tee
    ell = np.zeros((N, N), dtype=np.uint8)
    ell[:, 0] = 1
    ell[6, :] = 1
    out["L"] = ell
    return out


TEMPLATES = _templates()


def _shear(grid: np.ndarray, direction: int) -> np.ndarray:
    return np.stack([np.roll(grid[row], direction * row) for row in range(N)])


def _residual(scene: dict) -> np.ndarray:
    teal = np.asarray(scene["teal_grid"], dtype=np.uint8)
    coral = np.asarray(scene["coral_grid"], dtype=np.uint8)
    return _shear(teal, 1) ^ _shear(coral, -1)


def _rank(residual: np.ndarray) -> list[tuple[int, str]]:
    return sorted((int(np.count_nonzero(residual != glyph)), label) for label, glyph in TEMPLATES.items())


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(int(seed))
    label = LABELS[int(seed) % len(LABELS)]
    target = TEMPLATES[label].copy()

    # Every third scene is clean. Others are brought near (but not onto) the
    # nearest competing template while preserving the target's 13-cell ink.
    if int(seed) % 3:
        competitors = sorted(
            (int(np.count_nonzero(target != glyph)), other, glyph)
            for other, glyph in TEMPLATES.items()
            if other != label
        )
        distance, _, competitor = competitors[0]
        pair_count = max(1, (distance - 4) // 4)
        turn_off = np.argwhere((target == 1) & (competitor == 0))
        turn_on = np.argwhere((target == 0) & (competitor == 1))
        rng.shuffle(turn_off)
        rng.shuffle(turn_on)
        for y, x in turn_off[:pair_count]:
            target[y, x] = 0
        for y, x in turn_on[:pair_count]:
            target[y, x] = 1

    # Search for two source panels whose ink counts are answer-invariant. The
    # XOR target has odd weight, so 24 versus 25 is parity-consistent.
    for _ in range(20000):
        flat = np.zeros(N * N, dtype=np.uint8)
        flat[rng.choice(N * N, size=24, replace=False)] = 1
        teal = flat.reshape(N, N)
        sheared_coral = _shear(teal, 1) ^ target
        coral = _shear(sheared_coral, 1)
        if int(coral.sum()) == 25:
            return {
                "teal_grid": teal.tolist(),
                "coral_grid": coral.tolist(),
                "teal_left": bool(rng.integers(0, 2)),
            }
    raise RuntimeError("could not counterbalance panel ink")


def analytic_gold(scene: dict) -> str:
    return _rank(_residual(scene))[0][1]


def margin(scene: dict) -> float:
    ranked = _rank(_residual(scene))
    return float(ranked[1][0] - ranked[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 3.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _draw_panel(draw: ImageDraw.ImageDraw, x0: int, grid: list[list[int]], frame: tuple[int, int, int]) -> None:
    draw.rounded_rectangle((x0, PANEL_Y, x0 + PANEL - 1, PANEL_Y + PANEL - 1), radius=8, fill=frame)
    draw.rectangle((x0 + 8, PANEL_Y + 8, x0 + PANEL - 9, PANEL_Y + PANEL - 9), fill=CARD)
    for row in range(N):
        for col in range(N):
            x = x0 + GRID_OFFSET + col * (CELL + GAP)
            y = PANEL_Y + GRID_OFFSET + row * (CELL + GAP)
            fill = DARK if int(grid[row][col]) else LIGHT
            draw.rectangle((x, y, x + CELL - 1, y + CELL - 1), fill=fill)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    teal_x = LEFT_X if scene["teal_left"] else RIGHT_X
    coral_x = RIGHT_X if scene["teal_left"] else LEFT_X
    _draw_panel(draw, teal_x, scene["teal_grid"], TEAL)
    _draw_panel(draw, coral_x, scene["coral_grid"], CORAL)
    return image
