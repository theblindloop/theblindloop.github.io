"""Deterministic latent sampler and renderer for the radiation-blade world."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

CANVAS = 384
COUNTS = (2, 3, 4, 5)
INKS = ("#20242B", "#25213A", "#17313D")
PLAQUES = ("#F6C945", "#F4D35E", "#FFD166")
TOTAL_BLADE_SPAN_DEG = 144.0


def sample_scene(seed: int) -> dict:
    """Sample balanced count identities and nuisance styling from one integer seed."""
    rng = random.Random((int(seed) ^ 0x5A17C9E3) & 0xFFFFFFFF)
    blade_count = COUNTS[int(seed) % len(COUNTS)]
    scale = round(rng.uniform(0.91, 1.06), 4)
    markers = [
        [-116.0, -108.0],
        [116.0, -108.0],
        [-116.0, 108.0],
        [116.0, 108.0],
    ]
    rng.shuffle(markers)
    return {
        "blade_count": blade_count,
        "center_x": round(CANVAS / 2 + rng.uniform(-15.0, 15.0), 3),
        "center_y": round(CANVAS / 2 + rng.uniform(-15.0, 15.0), 3),
        "scale": scale,
        "rotation_deg": round(rng.uniform(0.0, 360.0), 3),
        "ink": INKS[(int(seed) // 4) % len(INKS)],
        "plaque": PLAQUES[(int(seed) // 12) % len(PLAQUES)],
        "markers": markers,
    }


def _sector_points(cx: float, cy: float, inner: float, outer: float, a0: float, a1: float):
    steps = 28
    outer_arc = []
    inner_arc = []
    for i in range(steps + 1):
        angle = math.radians(a0 + (a1 - a0) * i / steps)
        outer_arc.append((cx + outer * math.cos(angle), cy + outer * math.sin(angle)))
    for i in range(steps, -1, -1):
        angle = math.radians(a0 + (a1 - a0) * i / steps)
        inner_arc.append((cx + inner * math.cos(angle), cy + inner * math.sin(angle)))
    return outer_arc + inner_arc


def render(scene: dict) -> Image.Image:
    """Render the final RGB raster with supersampled, visibly separated components."""
    blade_count = int(scene["blade_count"])
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    scale = float(scene["scale"])
    rotation = float(scene["rotation_deg"])
    ink = str(scene["ink"])
    plaque = str(scene["plaque"])
    markers = list(scene["markers"])

    ss = 4
    image = Image.new("RGB", (CANVAS * ss, CANVAS * ss), "white")
    draw = ImageDraw.Draw(image)
    scx, scy = cx * ss, cy * ss
    plaque_half = 145.0 * scale * ss
    border_width = max(4, round(6.0 * scale * ss))
    draw.rounded_rectangle(
        [scx - plaque_half, scy - plaque_half, scx + plaque_half, scy + plaque_half],
        radius=24.0 * scale * ss,
        fill=plaque,
        outline=ink,
        width=border_width,
    )

    inner = 43.0 * scale * ss
    outer = 108.0 * scale * ss
    span = TOTAL_BLADE_SPAN_DEG / blade_count
    for index in range(blade_count):
        mid = rotation + index * 360.0 / blade_count
        points = _sector_points(scx, scy, inner, outer, mid - span / 2, mid + span / 2)
        draw.polygon(points, fill=ink)

    hub = 22.0 * scale * ss
    draw.ellipse([scx - hub, scy - hub, scx + hub, scy + hub], fill=ink)
    marker_half = 5.0 * scale * ss
    for dx, dy in markers:
        mx, my = scx + float(dx) * scale * ss, scy + float(dy) * scale * ss
        draw.rectangle(
            [mx - marker_half, my - marker_half, mx + marker_half, my + marker_half],
            fill=ink,
        )

    return image.resize((CANVAS, CANVAS), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(int(scene["blade_count"]))


def margin(scene: dict) -> float:
    """One-sided chord clearance between neighboring blades at their inner radius."""
    count = int(scene["blade_count"])
    scale = float(scene["scale"])
    open_angle = (360.0 - TOTAL_BLADE_SPAN_DEG) / count
    return 2.0 * (43.0 * scale) * math.sin(math.radians(open_angle) / 2.0)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Permuting non-overlapping equal-color marker draw order is pixel-identical."""
    twin = copy.deepcopy(scene)
    twin["markers"] = list(reversed(twin["markers"]))
    return [("reverse_equal_marker_order", twin)]
