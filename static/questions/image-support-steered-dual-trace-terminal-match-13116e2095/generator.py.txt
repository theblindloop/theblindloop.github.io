"""Deterministic renderer and analytic arm for dual trace terminal matching."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 800, 480
BACKGROUND = (250, 248, 242)
NAVY = (28, 48, 78)
GREEN = (34, 168, 92)
CYAN = (27, 170, 210)
ORANGE = (239, 126, 46)
GRAY = (145, 150, 157)
OUTLINE = (20, 30, 45)
STROKE = 10
MARKER_R = 14
QUARANTINE_MARGIN = 30.0


def _trail_points(x0: int, x1: int, y: int, variant: int, flip: bool) -> list[list[int]]:
    """Return a visibly bent but unambiguous centerline."""
    dx = x1 - x0
    sign = -1 if variant % 2 else 1
    amp = 16 + 5 * (variant // 2)
    points = [
        [x0, y],
        [x0 + dx // 3, y + sign * amp],
        [x0 + 2 * dx // 3, y - sign * (amp // 2)],
        [x1, y],
    ]
    return list(reversed(points)) if flip else points


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    same = (int(seed) & 1) == 0
    first_target = "cyan" if rng.randrange(2) == 0 else "orange"
    second_target = first_target if same else ("orange" if first_target == "cyan" else "cyan")
    islands = []
    for index, (x0, x1, target_color) in enumerate(
        ((157, 258, first_target), (542, 643, second_target))
    ):
        target_top = bool(rng.randrange(2))
        reverse = bool(rng.randrange(2))
        separation = rng.choice((76, 88, 100))
        center_y = 240 + rng.choice((-5, 0, 5))
        top_y = center_y - separation // 2
        bottom_y = center_y + separation // 2
        target_y = top_y if target_top else bottom_y
        decoy_y = bottom_y if target_top else top_y
        decoy_color = "orange" if target_color == "cyan" else "cyan"
        islands.append(
            {
                "target_path": _trail_points(x0, x1, target_y, rng.randrange(4), reverse),
                "decoy_path": _trail_points(x0, x1, decoy_y, rng.randrange(4), reverse),
                "target_start": [x1, target_y] if reverse else [x0, target_y],
                "decoy_start": [x1, decoy_y] if reverse else [x0, decoy_y],
                "target_terminal": [x0, target_y] if reverse else [x1, target_y],
                "decoy_terminal": [x0, decoy_y] if reverse else [x1, decoy_y],
                "target_color": target_color,
                "decoy_color": decoy_color,
                "lane_separation": separation,
                "island_index": index,
            }
        )
    return {"islands": islands}


def _ellipse(draw: ImageDraw.ImageDraw, center: list[int], radius: int, fill: tuple[int, int, int]) -> None:
    x, y = center
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=fill, outline=OUTLINE, width=2)


def _terminal_color(name: str) -> tuple[int, int, int]:
    return CYAN if name == "cyan" else ORANGE


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for island in scene["islands"]:
        draw.line([tuple(p) for p in island["target_path"]], fill=NAVY, width=STROKE, joint="curve")
        draw.line([tuple(p) for p in island["decoy_path"]], fill=NAVY, width=STROKE, joint="curve")
        _ellipse(draw, island["target_start"], MARKER_R, GREEN)
        _ellipse(draw, island["decoy_start"], MARKER_R, GRAY)
        _ellipse(draw, island["target_terminal"], MARKER_R, _terminal_color(island["target_color"]))
        _ellipse(draw, island["decoy_terminal"], MARKER_R, _terminal_color(island["decoy_color"]))
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    colors = [island["target_color"] for island in scene["islands"]]
    return "same" if colors[0] == colors[1] else "different"


def margin(scene: dict[str, Any]) -> float:
    return float(min(island["lane_separation"] for island in scene["islands"]) - 2 * MARKER_R - STROKE)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < QUARANTINE_MARGIN or len(scene["islands"]) != 2


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
