"""Independent pixel-only inverse arm based on color masks and connectivity."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

NAVY = np.array((28, 48, 78), dtype=np.int16)
GREEN = np.array((34, 168, 92), dtype=np.int16)
CYAN = np.array((27, 170, 210), dtype=np.int16)
ORANGE = np.array((239, 126, 46), dtype=np.int16)
TOLERANCE = 8


def _mask(array: np.ndarray, color: np.ndarray) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= TOLERANCE


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            output.append(np.asarray(points, dtype=np.int32))
    return output


def _centroid(component: np.ndarray) -> np.ndarray:
    return component.astype(np.float64).mean(axis=0)


def _near(component: np.ndarray, marker: np.ndarray, radius: float = 20.0) -> bool:
    c = _centroid(marker)
    delta = component.astype(np.float64) - c
    return float(np.min(np.sum(delta * delta, axis=1))) <= radius * radius


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 200 or array.shape[1] < 400:
        raise ValueError("unexpected raster")
    navy_components = _components(_mask(array, NAVY), 80)
    green_components = _components(_mask(array, GREEN), 100)
    cyan_components = _components(_mask(array, CYAN), 100)
    orange_components = _components(_mask(array, ORANGE), 100)
    if len(green_components) != 2 or len(cyan_components) != 2 or len(orange_components) != 2:
        raise ValueError("required markers are missing or corrupted")
    recovered: list[tuple[float, str]] = []
    for green in green_components:
        touching = [component for component in navy_components if _near(component, green)]
        if len(touching) != 1:
            raise ValueError("green start does not identify exactly one trail")
        trail = touching[0]
        terminals: list[str] = []
        terminals.extend("cyan" for marker in cyan_components if _near(trail, marker))
        terminals.extend("orange" for marker in orange_components if _near(trail, marker))
        if len(terminals) != 1:
            raise ValueError("traced trail does not reach exactly one terminal")
        recovered.append((float(_centroid(green)[1]), terminals[0]))
    recovered.sort()
    return "same" if recovered[0][1] == recovered[1][1] else "different"
