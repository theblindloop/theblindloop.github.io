"""Independent pixel-only inverse for metric constellation correspondence."""

from __future__ import annotations

import itertools
import math

import numpy as np
from PIL import Image

WIDTH, HEIGHT = 960, 360
PANEL_X_RANGES = ((20, 300), (340, 620), (660, 940))
PANEL_Y_RANGE = (82, 334)
COLORS = (
    ("red", (220, 50, 47)),
    ("blue", (43, 109, 176)),
    ("green", (46, 150, 85)),
    ("orange", (235, 137, 48)),
    ("purple", (132, 76, 170)),
)


def _largest_component_centroid(mask: np.ndarray) -> tuple[float, float]:
    """Find the centroid of the largest 4-connected exact-color component."""
    ys, xs = np.nonzero(mask)
    if len(xs) == 0:
        raise ValueError("missing colored component")
    remaining = set(zip(xs.tolist(), ys.tolist()))
    best: list[tuple[int, int]] = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        component = [start]
        while stack:
            x, y = stack.pop()
            for neighbor in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
                if neighbor in remaining:
                    remaining.remove(neighbor)
                    stack.append(neighbor)
                    component.append(neighbor)
        if len(component) > len(best):
            best = component
    if len(best) < 20:
        raise ValueError("colored component is too small")
    return (sum(x for x, _ in best) / len(best), sum(y for _, y in best) / len(best))


def _panel_points(array: np.ndarray, x_range: tuple[int, int]) -> dict[str, tuple[float, float]]:
    x0, x1 = x_range
    y0, y1 = PANEL_Y_RANGE
    panel = array[y0:y1, x0:x1]
    points = {}
    for name, rgb in COLORS:
        mask = np.all(panel == np.asarray(rgb, dtype=np.uint8), axis=2)
        x, y = _largest_component_centroid(mask)
        points[name] = (x + x0, y + y0)
    return points


def _signature(points: dict[str, tuple[float, float]]) -> tuple[tuple[str, str], ...]:
    values = []
    names = tuple(name for name, _ in COLORS)
    for left, right in itertools.combinations(names, 2):
        a, b = points[left], points[right]
        values.append((math.dist(a, b), (left, right)))
    if len({round(value, 6) for value, _ in values}) != len(values):
        raise ValueError("pixel pair distances are tied")
    return tuple(pair for _, pair in sorted(values, key=lambda item: (item[0], item[1])))


def _score(reference: tuple, candidate: tuple) -> int:
    return sum(left == right for left, right in zip(reference, candidate))


def decision_from_image(image: Image.Image) -> str:
    """Recover A/B using only exact RGB segmentation, centroids, and distances."""
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.shape[:2] != (HEIGHT, WIDTH):
        raise ValueError(f"unexpected raster size {array.shape[:2]}")
    reference = _signature(_panel_points(array, PANEL_X_RANGES[0]))
    score_a = _score(reference, _signature(_panel_points(array, PANEL_X_RANGES[1])))
    score_b = _score(reference, _signature(_panel_points(array, PANEL_X_RANGES[2])))
    if score_a == score_b:
        raise ValueError("pixel candidates have tied distance-rank scores")
    return "A" if score_a > score_b else "B"
