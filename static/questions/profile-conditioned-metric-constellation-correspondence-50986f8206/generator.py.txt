"""Renderer and analytic label for metric constellation correspondence.

The three panels contain five filled, uniquely colored point markers.  The
question is about the ordinal order of all ten labeled center-to-center
distances, so the renderer deliberately gives both candidate panels identical
geometry and marker areas; only their color-to-location assignment differs.
"""

from __future__ import annotations

import itertools
import math
import random
from pathlib import Path

from PIL import Image, ImageDraw

RENDERER_VERSION = "metric-constellation-correspondence-0.1.0"
WIDTH, HEIGHT = 960, 360
PANEL_CENTERS = ((160, 205), (480, 205), (800, 205))
PANEL_BOXES = ((20, 82, 300, 334), (340, 82, 620, 334), (660, 82, 940, 334))
COLORS = {
    "red": (220, 50, 47),
    "blue": (43, 109, 176),
    "green": (46, 150, 85),
    "orange": (235, 137, 48),
    "purple": (132, 76, 170),
}
COLOR_ORDER = tuple(COLORS)
BACKGROUND = (250, 250, 247)
INK = (34, 38, 42)
_TEMPLATES = (
    ((68.645, -46.918), (42.414, 3.027), (-74.222, 6.580), (-38.784, -54.727), (-6.066, 67.023)),
    ((54.704, 66.446), (-59.916, -51.957), (13.028, 10.649), (58.249, 10.276), (-59.020, 54.247)),
    ((-43.822, 15.454), (-49.681, -65.405), (22.653, -56.268), (-63.257, 57.488), (68.869, 40.100)),
    ((64.712, 54.817), (24.861, -49.366), (-9.441, -15.731), (-64.196, 34.060), (-59.078, -58.839)),
    ((-35.675, 33.925), (-68.803, -43.102), (57.790, -24.672), (65.264, 32.177), (28.109, 62.605)),
)


def _pair_distances(points: list[dict]) -> list[tuple[float, tuple[str, str]]]:
    positions = {point["color"]: (float(point["x"]), float(point["y"])) for point in points}
    values = []
    for left, right in itertools.combinations(COLOR_ORDER, 2):
        a, b = positions[left], positions[right]
        values.append((math.hypot(a[0] - b[0], a[1] - b[1]), (left, right)))
    return values


def _signature(points: list[dict]) -> tuple[tuple[str, str], ...]:
    """Return the strict short-to-long order of the ten labeled pairs."""
    values = _pair_distances(points)
    if len({round(value, 8) for value, _ in values}) != len(values):
        raise ValueError("pairwise distances are tied")
    return tuple(pair for _, pair in sorted(values, key=lambda item: (item[0], item[1])))


def _rank_score(reference: tuple, candidate: tuple) -> int:
    return sum(left == right for left, right in zip(reference, candidate))


def _minimum_separation(points: list[dict]) -> float:
    positions = [(float(point["x"]), float(point["y"])) for point in points]
    return min(math.dist(a, b) for a, b in itertools.combinations(positions, 2))


def _minimum_rank_gap(points: list[dict]) -> float:
    distances = sorted(value for value, _ in _pair_distances(points))
    return min(right - left for left, right in zip(distances, distances[1:]))


def _panel_is_safe(panel: dict) -> bool:
    try:
        _signature(panel["points"])
        return _minimum_separation(panel["points"]) >= 30.0 and _minimum_rank_gap(panel["points"]) >= 3.5
    except (KeyError, TypeError, ValueError):
        return False


def sample_scene(seed: int) -> dict:
    """Sample one balanced scene; all label-bearing fields are rendered."""
    rng = random.Random(int(seed))
    for _ in range(20_000):
        template = rng.choice(_TEMPLATES)
        reference_angle = rng.uniform(0.0, 2.0 * math.pi)
        reference_scale = rng.uniform(0.95, 1.02)
        ca, sa = math.cos(reference_angle), math.sin(reference_angle)
        reference_points = []
        for index, (x, y) in enumerate(template):
            reference_points.append(
                {
                    "x": reference_scale * (ca * x - sa * y),
                    "y": reference_scale * (sa * x + ca * y),
                    "color": COLOR_ORDER[index],
                }
            )

        permutation = list(COLOR_ORDER)
        rng.shuffle(permutation)
        for point, color in zip(reference_points, permutation):
            point["color"] = color

        angle = rng.uniform(0.0, 2.0 * math.pi)
        scale = rng.uniform(0.90, 1.04)
        reflection = -1.0 if rng.randrange(2) else 1.0
        ca, sa = math.cos(angle), math.sin(angle)
        transformed = []
        for point in reference_points:
            px = float(point["x"]) * reflection
            py = float(point["y"])
            transformed.append(
                {
                    "x": scale * (ca * px - sa * py),
                    "y": scale * (sa * px + ca * py),
                    "color": point["color"],
                }
            )
        swap_left, swap_right = sorted(rng.sample(range(5), 2))
        decoy = [dict(point) for point in transformed]
        decoy[swap_left]["color"], decoy[swap_right]["color"] = (
            decoy[swap_right]["color"],
            decoy[swap_left]["color"],
        )
        candidate_radius = rng.randint(10, 14)
        reference_radius = rng.randint(10, 14)
        matching = {"points": transformed, "radius": candidate_radius}
        wrong = {"points": decoy, "radius": candidate_radius}
        candidate_a, candidate_b = (matching, wrong) if rng.randrange(2) == 0 else (wrong, matching)
        scene = {
            "reference": {"points": reference_points, "radius": reference_radius},
            "candidate_a": candidate_a,
            "candidate_b": candidate_b,
        }
        if not all(_panel_is_safe(scene[key]) for key in ("reference", "candidate_a", "candidate_b")):
            continue
        if _rank_score(_signature(reference_points), _signature(candidate_a["points"])) == _rank_score(
            _signature(reference_points), _signature(candidate_b["points"])
        ):
            continue
        return scene
    raise RuntimeError("could not sample a finite-resolution-safe constellation")


def _draw_panel(draw: ImageDraw.ImageDraw, panel: dict, center: tuple[int, int], label: str, title: str) -> None:
    cx, cy = center
    for point in panel["points"]:
        x = int(round(cx + float(point["x"])))
        y = int(round(cy + float(point["y"])))
        radius = int(panel["radius"])
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=COLORS[point["color"]])
    draw.text((cx, 94), title, fill=INK, anchor="ma")
    draw.text((cx, 322), label, fill=INK, anchor="ma")


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((WIDTH // 2, 24), "COLOR-LABELED DISTANCE RANK", fill=INK, anchor="ma")
    for box in PANEL_BOXES:
        draw.rectangle(box, outline=INK, width=2)
    _draw_panel(draw, scene["reference"], PANEL_CENTERS[0], "REFERENCE", "REFERENCE")
    _draw_panel(draw, scene["candidate_a"], PANEL_CENTERS[1], "A", "CANDIDATE")
    _draw_panel(draw, scene["candidate_b"], PANEL_CENTERS[2], "B", "CANDIDATE")
    return image


def analytic_gold(scene: dict) -> str:
    reference = scene["reference"]
    candidate_a = scene["candidate_a"]
    candidate_b = scene["candidate_b"]
    reference_signature = _signature(reference["points"])
    score_a = _rank_score(reference_signature, _signature(candidate_a["points"]))
    score_b = _rank_score(reference_signature, _signature(candidate_b["points"]))
    if score_a == score_b:
        raise ValueError("candidate distance signatures are tied")
    return "A" if score_a > score_b else "B"


def margin(scene: dict) -> float:
    reference_signature = _signature(scene["reference"]["points"])
    score_a = _rank_score(reference_signature, _signature(scene["candidate_a"]["points"]))
    score_b = _rank_score(reference_signature, _signature(scene["candidate_b"]["points"]))
    return float(abs(score_a - score_b))


def is_quarantined(scene: dict) -> bool:
    try:
        panels = [scene["reference"], scene["candidate_a"], scene["candidate_b"]]
        if not all(_panel_is_safe(panel) for panel in panels):
            return True
        reference_signature = _signature(scene["reference"]["points"])
        scores = [
            _rank_score(reference_signature, _signature(scene[key]["points"]))
            for key in ("candidate_a", "candidate_b")
        ]
        return scores[0] == scores[1]
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
