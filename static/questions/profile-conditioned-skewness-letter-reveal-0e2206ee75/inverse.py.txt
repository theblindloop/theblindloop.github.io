"""Independent pixel-only inverse arm for skewness_letter_reveal.

Receives ONLY the final PNG. It:
  1. reads the luminance of every pixel;
  2. computes the local luminance SKEWNESS (third central moment normalized by
     sigma^3) of every block - the declared second-order-invariant, third-moment
     operator;
  3. correlates that skewness map against the two declared letter templates
     T and L;
  4. reports the template whose stroke region carries the higher positive-skew
     contrast.

No scene data, no analytic gold, no renderer imports.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "skewness_letter_reveal-oracle-0.1.0"

W = 512
NGRID = 16
CELL = W // NGRID


def _letter_masks() -> dict[str, np.ndarray]:
    r, c = np.indices((NGRID, NGRID))
    t = (r <= 0) | (np.abs(c - (NGRID // 2)) <= 1)
    l = (c <= 1) | (r >= NGRID - 2)
    return {"T": t, "L": l}


def _block_skewness(image: Image.Image) -> np.ndarray:
    """(NGRID, NGRID) map of the local luminance skewness of every block."""
    arr = np.asarray(image.convert("L"), dtype=np.float64)
    out = np.zeros((NGRID, NGRID), dtype=np.float64)
    for i in range(NGRID):
        for j in range(NGRID):
            block = arr[i * CELL:(i + 1) * CELL, j * CELL:(j + 1) * CELL].ravel()
            s = block.std()
            if s < 1e-9:
                out[i, j] = 0.0
            else:
                out[i, j] = float(np.mean(((block - block.mean()) / s) ** 3))
    return out


def _contrast(skew_map: np.ndarray, mask: np.ndarray) -> float:
    inside = skew_map[mask.astype(bool)].mean()
    outside = skew_map[~mask.astype(bool)].mean()
    return float(inside - outside)


def decision_from_image(image: Image.Image) -> str:
    skew_map = _block_skewness(image)
    masks = _letter_masks()
    st = _contrast(skew_map, masks["T"])
    sl = _contrast(skew_map, masks["L"])
    return "T" if st >= sl else "L"


def reveal_margin(image: Image.Image) -> float:
    skew_map = _block_skewness(image)
    masks = _letter_masks()
    st = _contrast(skew_map, masks["T"])
    sl = _contrast(skew_map, masks["L"])
    return abs(st - sl)
