"""skewness_letter_reveal: third-central-moment (skewness) letter reveal.

Declared transform: every 16x16 lattice block has the SAME mean brightness and
the SAME variance, but blocks that belong to a coarse letter glyph carry an
ASYMETRIC luminance distribution (a positive third central moment) while the
background blocks carry a symmetric Gaussian distribution (zero skewness). The
declared operation computes the local luminance SKEWNESS (third central moment
normalized by sigma^3) of every block; that map is flat in its mean and
variance but sharply positive on the glyph strokes, which reconstructs a clean
letter whose identity (T or L) is the answer.

Because the first and second moments are held constant across the whole field,
no mean-, variance-, range-, median-, parity-, or position-based operator can
read the letter (verified: such operators are at chance). Only the declared
per-block third-central-moment operator exposes it.

Latent scene z = {seedc (glyph/class seed), glyph, seedn (noise seed)}.
label = glyph.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RENDERER_VERSION = "skewness_letter_reveal-0.1.0"

W = 512
H = 512
NGRID = 16          # blocks per side
CELL = W // NGRID   # 32 px per block
BASE = 128.0
SIGMA_OUT = 45.0    # symmetric Gaussian noise std (background blocks)
P_HIGH = 0.25       # fraction of high pixels inside glyph strokes
# Two-point symmetric-mean, variance-matched modulation inside the glyph:
# 25% of pixels at BASE+A, 75% at BASE-B with B = A/3 keeps mean = BASE and
# variance = SIGMA_OUT^2 (equal to the background), while giving positive skew.
A = float(np.sqrt(3.0) * SIGMA_OUT)
B = A / 3.0
Q_TOL = 0.30


def _glyph_mask(glyph: str) -> np.ndarray:
    """(NGRID, NGRID) bool masks: T = top bar + central stem; L = left + bottom."""
    r, c = np.indices((NGRID, NGRID))
    if glyph == "T":
        return (r <= 0) | (np.abs(c - (NGRID // 2)) <= 1)
    # L
    return (c <= 1) | (r >= NGRID - 2)


def _letter_masks() -> dict[str, np.ndarray]:
    return {"T": _glyph_mask("T"), "L": _glyph_mask("L")}


def build_scene(seedc: int, glyph: str, seedn: int) -> dict:
    return {"seedc": int(seedc), "glyph": glyph, "seedn": int(seedn)}


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(int(seed) & 0xFFFFFFFF)
    glyph = "T" if rng.integers(0, 2) else "L"
    return build_scene(seedc=int(seed), glyph=glyph, seedn=int(seed) + 57)


def render_matrix(scene: dict) -> np.ndarray:
    """(H, W) float luminance raster.

    Background pixels are symmetric Gaussian noise centered on BASE with std
    SIGMA_OUT. Glyph-stroke pixels are drawn from an asymmetric two-point
    distribution (25% at BASE+A, 75% at BASE-B) whose mean equals BASE and whose
    variance equals SIGMA_OUT^2, so the whole field is flat in mean and variance
    and the letter is exposed only by its third central moment (skewness)."""
    mask = _glyph_mask(scene["glyph"]).astype(np.float64)
    rng = np.random.default_rng((int(scene["seedn"]) & 0xFFFFFFFF) ^ 0x9E3779B9)
    pix = rng.normal(BASE, SIGMA_OUT, size=(W, H))
    inside = np.repeat(np.repeat(mask, CELL, axis=0), CELL, axis=1).astype(bool)
    n = int(inside.sum())
    draws = rng.random(n)
    vals = np.where(draws < P_HIGH, BASE + A, BASE - B)
    pix[inside] = vals
    return pix[:W, :H]


def render(scene: dict) -> Image.Image:
    arr = render_matrix(scene)
    arr = np.clip(arr, 0, 255)
    canvas = np.repeat(arr[:, :, None], 3, axis=2).astype(np.uint8)
    canvas = canvas[:W, :H]
    return Image.fromarray(canvas, "RGB")


def margin(scene: dict) -> float:
    chosen = _glyph_mask(scene["glyph"])
    other = _glyph_mask("L" if scene["glyph"] == "T" else "T")
    o_filled = int(other.sum())
    overlap = float((chosen & other).sum()) / max(1, o_filled)
    return round(1.0 - overlap, 6)


def analytic_gold(scene: dict) -> str:
    return scene["glyph"]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < Q_TOL


def latent_symmetries(scene: dict):
    """not_applicable: every latent field (the class seed, the answer glyph, and
    the noise seed) directly changes the rendered pixels, so no distinct latent
    vector reproduces the same image. Declared empty."""
    return iter(())
