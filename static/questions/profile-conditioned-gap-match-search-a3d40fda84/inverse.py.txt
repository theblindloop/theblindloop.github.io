"""Independent pixel-only inverse arm for gap_match_search.

Receives only a PIL image and the public question contract. It locates the dark
stacked bar pair in the top reference panel and in each of the four quadrants,
measures each pair's vertical gap (the empty space between the facing edges of
the two bars), and selects the quadrant whose measured gap is closest to the
reference pair's gap.

Imports nothing from the renderer, prompts, generator, or verifier.
"""

from __future__ import annotations

import numpy as np

HB = 12.0  # bar height in pixels, public layout
COL_RANGE = {
    "REF": (210, 270),
    "TL": (90, 150),
    "TR": (330, 390),
    "BL": (90, 150),
    "BR": (330, 390),
}
Y_WINDOW = {
    "REF": (56, 168),
    "TL": (248, 360),
    "TR": (248, 360),
    "BL": (368, 480),
    "BR": (368, 480),
}


def _gap(arr, name: str) -> float:
    x0, x1 = COL_RANGE[name]
    y0, y1 = Y_WINDOW[name]
    region = arr[y0:y1, x0:x1]
    dark = np.min(region, axis=2) < 150
    rows = dark.any(axis=1)
    row_idx = np.nonzero(rows)[0]
    if row_idx.size == 0:
        return 0.0
    # Collapse consecutive rows into blocks (bars are solid, gap has no dark rows).
    blocks = []
    start = row_idx[0]
    prev = row_idx[0]
    for r in row_idx[1:]:
        if r - prev > 1:
            blocks.append((int(start), int(prev)))
            start = r
        prev = r
    blocks.append((int(start), int(prev)))
    blocks.sort(key=lambda b: b[0])
    if len(blocks) < 2:
        return 0.0
    top_bar_bottom = blocks[0][1]
    bottom_bar_top = blocks[1][0]
    return float(bottom_bar_top - top_bar_bottom - 1)


def decision_from_image(image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    ref_gap = _gap(arr, "REF")
    best_sector = None
    best_diff = float("inf")
    for sector in ("TL", "TR", "BL", "BR"):
        gap = _gap(arr, sector)
        diff = abs(gap - ref_gap)
        if diff < best_diff:
            best_diff = diff
            best_sector = sector
    return best_sector if best_sector is not None else "TL"
