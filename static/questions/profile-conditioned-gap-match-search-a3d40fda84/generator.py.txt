"""Reference-gap spacing search world.

A reference pair of stacked horizontal bars is shown in the top panel with a
specific vertical gap (empty space between the facing edges of the two bars).
Below, the image is split into four quadrants, each containing its own pair of
stacked horizontal bars with a different vertical gap. Exactly one quadrant's
bar pair has the same gap as the reference pair. The decision is the quadrant
containing that matching bar pair. All geometry and the decision are visible in
the raster; rendering is deterministic.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw

SIZE = 480
BG = (255, 255, 255)
BAR = (25, 25, 25)

HB = 12  # bar height in pixels
WB = 60  # bar width in pixels
HBAR_HALF = WB // 2

REF_CX = 240
REF_TOP = 60

SECTORS = ("TL", "TR", "BL", "BR")
SECTOR_CX = {"TL": 120, "TR": 360, "BL": 120, "BR": 360}
SECTOR_TOP = {"TL": 252, "TR": 252, "BL": 372, "BR": 372}

# Column ranges exactly enclose each bar so every bar row is fully occupied.
COL_RANGE = {
    "REF": (REF_CX - HBAR_HALF, REF_CX + HBAR_HALF),
    "TL": (SECTOR_CX["TL"] - HBAR_HALF, SECTOR_CX["TL"] + HBAR_HALF),
    "TR": (SECTOR_CX["TR"] - HBAR_HALF, SECTOR_CX["TR"] + HBAR_HALF),
    "BL": (SECTOR_CX["BL"] - HBAR_HALF, SECTOR_CX["BL"] + HBAR_HALF),
    "BR": (SECTOR_CX["BR"] - HBAR_HALF, SECTOR_CX["BR"] + HBAR_HALF),
}
# Vertical windows around each pair's bounding region.
Y_WINDOW = {
    "REF": (REF_TOP - 4, REF_TOP + 2 * HB + 80 + 4),
    "TL": (SECTOR_TOP["TL"] - 4, SECTOR_TOP["TL"] + 2 * HB + 80 + 4),
    "TR": (SECTOR_TOP["TR"] - 4, SECTOR_TOP["TR"] + 2 * HB + 80 + 4),
    "BL": (SECTOR_TOP["BL"] - 4, SECTOR_TOP["BL"] + 2 * HB + 80 + 4),
    "BR": (SECTOR_TOP["BR"] - 4, SECTOR_TOP["BR"] + 2 * HB + 80 + 4),
}

GAP_LO = 20
GAP_HI = 76
MIN_DIFF = 6  # minimum |gap - reference gap| for every distractor
QUARANTINE_THRESH = 4.0


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(seed)
    ref_gap = int(rng.integers(GAP_LO, GAP_HI + 1))

    winner_sector = SECTORS[int(rng.integers(0, len(SECTORS)))]
    gaps: dict[str, int] = {winner_sector: ref_gap}

    for sector in SECTORS:
        if sector in gaps:
            continue
        for _ in range(1000):
            candidate = int(rng.integers(GAP_LO, GAP_HI + 1))
            if abs(candidate - ref_gap) < MIN_DIFF:
                continue
            gaps[sector] = candidate
            break
        else:  # pragma: no cover - extremely unlikely path
            gaps[sector] = ref_gap + (MIN_DIFF if ref_gap < GAP_HI - MIN_DIFF else -MIN_DIFF)

    reference = {"gap": float(ref_gap), "cx": REF_CX, "top": REF_TOP}
    candidates = [
        {
            "gap": float(gaps[sector]),
            "sector": sector,
            "cx": SECTOR_CX[sector],
            "top": SECTOR_TOP[sector],
        }
        for sector in SECTORS
    ]
    return {"reference": reference, "candidates": candidates}


def _draw_pair(draw: ImageDraw.ImageDraw, cx: int, top: int, gap: int) -> None:
    x0 = cx - HBAR_HALF
    x1 = cx + HBAR_HALF
    draw.rectangle([x0, top, x1, top + HB], fill=BAR)
    draw.rectangle([x0, top + HB + gap, x1, top + HB + gap + HB], fill=BAR)


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (SIZE, SIZE), BG)
    draw = ImageDraw.Draw(img)
    ref = scene["reference"]
    _draw_pair(draw, int(ref["cx"]), int(ref["top"]), int(round(ref["gap"])))
    for cand in scene["candidates"]:
        _draw_pair(draw, int(cand["cx"]), int(cand["top"]), int(round(cand["gap"])))
    return img


def _decision(scene: dict) -> tuple[str, float]:
    ref_gap = scene["reference"]["gap"]
    best_sector = None
    best_diff = float("inf")
    for cand in scene["candidates"]:
        diff = abs(cand["gap"] - ref_gap)
        if diff < best_diff:
            best_diff = diff
            best_sector = cand["sector"]
    runner_up = min(
        abs(c["gap"] - ref_gap)
        for c in scene["candidates"]
        if c["sector"] != best_sector
    )
    return best_sector, float(runner_up)


def analytic_gold(scene: dict) -> str:
    sector, _ = _decision(scene)
    return sector


def margin(scene: dict) -> float:
    _, m = _decision(scene)
    return m


def is_quarantined(scene: dict) -> bool:
    _, m = _decision(scene)
    return bool(m < QUARANTINE_THRESH)


def latent_symmetries(scene: dict) -> list:
    return []
