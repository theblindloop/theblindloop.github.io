"""Independent RGB-only inverse renderer for the bilateral-aperture world."""

from __future__ import annotations

from typing import Any

import numpy as np
from PIL import Image

SECTORS = ("NW", "NE", "SW", "SE")
MIN_GLOBAL_DARK = 315_000
MIN_SECTOR_DARK = 68_000


def _components(mask: np.ndarray, minimum: int = 1) -> list[dict[str, Any]]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    result: list[dict[str, Any]] = []
    for y, x in np.argwhere(mask):
        y = int(y)
        x = int(x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for ny, nx in (
                (cy - 1, cx),
                (cy + 1, cx),
                (cy, cx - 1),
                (cy, cx + 1),
            ):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) < minimum:
            continue
        ys = [point[0] for point in points]
        xs = [point[1] for point in points]
        result.append(
            {
                "area": len(points),
                "bbox": (min(xs), min(ys), max(xs), max(ys)),
                "centroid": (sum(xs) / len(xs), sum(ys) / len(ys)),
            }
        )
    return result


def _sector_bounds(height: int, width: int) -> dict[str, tuple[int, int, int, int]]:
    half_width = width // 2
    half_height = height // 2
    return {
        "NW": (0, 0, half_width, half_height),
        "NE": (half_width, 0, width, half_height),
        "SW": (0, half_height, half_width, height),
        "SE": (half_width, half_height, width, height),
    }


def _window_widths(array: np.ndarray, bounds: tuple[int, int, int, int]) -> list[int]:
    x0, y0, x1, y1 = bounds
    local = np.all(array[y0:y1, x0:x1] >= 250, axis=2)
    holes = [item for item in _components(local, minimum=400) if item["area"] < 6000]
    if len(holes) != 2:
        raise ValueError("each sector must contain exactly two enclosed windows")
    widths = []
    for hole in holes:
        bx0, by0, bx1, by1 = hole["bbox"]
        if bx1 - bx0 < 20 or by1 - by0 < 20:
            raise ValueError("window is too small")
        if abs((bx1 - bx0) - (by1 - by0)) > 4:
            raise ValueError("window is not approximately round")
        widths.append(int(bx1 - bx0 + 1))
    return sorted(widths)


def decision_from_image(image: Image.Image) -> str:
    """Recover the unique equal-width window pair using only the final RGB image."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[2] != 3:
        raise ValueError("expected an RGB image")
    dark_color = np.asarray([28, 55, 92], dtype=np.int16)
    dark = np.linalg.norm(array.astype(np.int16) - dark_color, axis=2) <= 0
    # The plate is deliberately thick.  A four-pixel occupancy reduction is
    # enough to test its connected scaffold while avoiding a Python flood fill
    # over every full-resolution ink pixel.
    dark_components = _components(dark[::4, ::4], minimum=20)
    if len(dark_components) != 1:
        raise ValueError("the plate must be one connected dark component")
    if int(dark.sum()) < MIN_GLOBAL_DARK:
        raise ValueError("the connected plate is incomplete")

    gaps: list[tuple[int, str]] = []
    for sector, bounds in _sector_bounds(array.shape[0], array.shape[1]).items():
        x0, y0, x1, y1 = bounds
        if int(dark[y0:y1, x0:x1].sum()) < MIN_SECTOR_DARK:
            raise ValueError("a lobe is incomplete")
        widths = _window_widths(array, bounds)
        gaps.append((abs(widths[0] - widths[1]), sector))
    best = min(gap for gap, _ in gaps)
    winners = [sector for gap, sector in gaps if gap <= best + 1]
    if len(winners) != 1 or best > 1:
        raise ValueError("no unique equal-width pair")
    return winners[0]
