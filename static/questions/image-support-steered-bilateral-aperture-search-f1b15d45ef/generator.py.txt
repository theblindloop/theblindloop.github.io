"""Deterministic renderer and analytic arm for the bilateral-aperture search world."""

from __future__ import annotations

import copy
import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "bilateral_aperture_search-0.1.0"
SECTORS = ("NW", "NE", "SW", "SE")
WIDTH = 768
HEIGHT = 768
BACKGROUND = [235, 238, 243]
INK = [28, 55, 92]
WINDOW = [255, 255, 255]
CORNER_RADIUS = 56
HUB_BOX = [282, 282, 486, 486]
MARGIN_FLOOR = 8.0

LOBE_BOXES = {
    "NW": [58, 58, 350, 350],
    "NE": [418, 58, 710, 350],
    "SW": [58, 418, 350, 710],
    "SE": [418, 418, 710, 710],
}
WINDOW_CENTERS = {
    "NW": [[130, 204], [250, 204]],
    "NE": [[490, 204], [610, 204]],
    "SW": [[130, 564], [250, 564]],
    "SE": [[490, 564], [610, 564]],
}


def _pattern_deltas(seed: int) -> tuple[float, float, float]:
    """Return distractor separations with a visible near-boundary regime."""

    regimes = ((2.0, 3.0, 4.0), (3.0, 4.0, 6.0), (4.0, 5.0, 6.0))
    return regimes[int(seed) % len(regimes)]


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one balanced scene; the winning sector cycles with the seed."""

    seed = int(seed)
    rng = random.Random(seed)
    radius = float((24, 25, 26)[rng.randrange(3)])
    target_sector = SECTORS[seed % len(SECTORS)]
    deltas = list(_pattern_deltas(seed))
    rng.shuffle(deltas)

    lobes: list[dict[str, Any]] = []
    distractor_index = 0
    for sector in SECTORS:
        if sector == target_sector:
            radii = [radius, radius]
        else:
            delta = float(deltas[distractor_index])
            distractor_index += 1
            large = radius + delta
            small = math.sqrt(2.0 * radius * radius - large * large)
            radii = [large, small]
            if rng.random() < 0.5:
                radii.reverse()
        lobes.append(
            {
                "sector": sector,
                "box": list(LOBE_BOXES[sector]),
                "window_centers": [list(point) for point in WINDOW_CENTERS[sector]],
                "window_radii": [float(radii[0]), float(radii[1])],
            }
        )
    rng.shuffle(lobes)
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
        "ink": list(INK),
        "window": list(WINDOW),
        "lobes": lobes,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render all dark geometry first, then cut the eight white windows."""

    image = Image.new(
        "RGB",
        (int(scene["width"]), int(scene["height"])),
        tuple(int(value) for value in scene["background"]),
    )
    draw = ImageDraw.Draw(image)
    ink = tuple(int(value) for value in scene["ink"])
    window = tuple(int(value) for value in scene["window"])
    for lobe in scene["lobes"]:
        draw.rounded_rectangle(
            tuple(int(value) for value in lobe["box"]),
            radius=CORNER_RADIUS,
            fill=ink,
        )
    draw.rounded_rectangle(tuple(HUB_BOX), radius=48, fill=ink)
    for lobe in scene["lobes"]:
        for center, radius in zip(
            lobe["window_centers"], lobe["window_radii"], strict=True
        ):
            cx, cy = (float(center[0]), float(center[1]))
            r = float(radius)
            draw.ellipse((cx - r, cy - r, cx + r, cy + r), fill=window)
    return image


def _width_gaps(scene: dict[str, Any]) -> list[tuple[float, str]]:
    gaps = []
    for lobe in scene["lobes"]:
        radii = [float(value) for value in lobe["window_radii"]]
        gaps.append((abs(2.0 * radii[0] - 2.0 * radii[1]), str(lobe["sector"])))
    return gaps


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the unique sector whose two latent window widths are equal."""

    gaps = _width_gaps(scene)
    best_gap = min(gap for gap, _ in gaps)
    winners = [sector for gap, sector in gaps if abs(gap - best_gap) <= 1e-9]
    if len(winners) != 1 or best_gap > 1e-9:
        raise ValueError("scene has no unique equal-window lobe")
    return winners[0]


def margin(scene: dict[str, Any]) -> float:
    """Runner-up decision margin: the smallest non-winning width gap."""

    gaps = sorted(gap for gap, _ in _width_gaps(scene))
    nonzero = [gap for gap in gaps if gap > 1e-9]
    if not nonzero:
        raise ValueError("scene has no runner-up")
    return float(nonzero[0])


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Exclude width gaps too close to the pixel equality boundary."""

    return margin(scene) < MARGIN_FLOOR


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """Reordering lobe records is pixel-identical because sectors are explicit."""

    twin = copy.deepcopy(scene)
    twin["lobes"] = list(reversed(twin["lobes"]))
    return [("reorder_lobe_records", twin)]
