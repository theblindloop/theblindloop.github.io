"""Independent pixel-only arm for ``mitten_rack_overlap``.

Receives a rasterised PNG and nothing else: no latent scene, no analytic gold,
no renderer, prompt, generator or verifier code.  It segments the two dark grey
rack frames, validates that each is an undamaged rectangular frame, groups the
remaining chromatic blobs into flat-colour connected components, assigns each
blob to a rack interior or to the floor purely from geometry, and returns the
number of flat colours that occur on both racks.  It abstains whenever the
picture stops being legible under its own stated conditions.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "mitten_rack_overlap-oracle-0.1.0"
ABSTAIN = "abstain"

_BG = (255, 255, 255)
_RACK = (64, 64, 64)

_MIN_BLOB = 300
_FRAGMENT = 60
_MIN_RACK_AREA = 1200


def _packed(rgb: tuple[int, int, int]) -> int:
    return (int(rgb[0]) << 16) | (int(rgb[1]) << 8) | int(rgb[2])


def _components(flat: list[int], w: int, h: int, bg: int):
    """4-connected components of equal packed colour over non-background pixels."""
    seen = bytearray(w * h)
    comps = []
    for start in range(w * h):
        if seen[start] or flat[start] == bg:
            continue
        val = flat[start]
        seen[start] = 1
        stack = [start]
        area = 0
        x0 = x1 = start % w
        y0 = y1 = start // w
        while stack:
            p = stack.pop()
            area += 1
            x, y = p % w, p // w
            if x < x0:
                x0 = x
            if x > x1:
                x1 = x
            if y < y0:
                y0 = y
            if y > y1:
                y1 = y
            if x > 0 and not seen[p - 1] and flat[p - 1] == val:
                seen[p - 1] = 1
                stack.append(p - 1)
            if x + 1 < w and not seen[p + 1] and flat[p + 1] == val:
                seen[p + 1] = 1
                stack.append(p + 1)
            if y > 0 and not seen[p - w] and flat[p - w] == val:
                seen[p - w] = 1
                stack.append(p - w)
            if y + 1 < h and not seen[p + w] and flat[p + w] == val:
                seen[p + w] = 1
                stack.append(p + w)
        comps.append({"colour": val, "area": area, "box": (x0, y0, x1, y1)})
    return comps


def _sep(a, b) -> float:
    return max(max(a[0] - b[2], b[0] - a[2]), max(a[1] - b[3], b[1] - a[3]))


def _inside(inner, box, slack: float = 2.0) -> bool:
    return (
        box[0] >= inner[0] - slack
        and box[1] >= inner[1] - slack
        and box[2] <= inner[2] + slack
        and box[3] <= inner[3] + slack
    )


def _rack_stroke(mask, box) -> int:
    x0, y0, x1, y1 = box
    x = (x0 + x1) // 2
    t = 0
    while y0 + t <= y1 and mask[y0 + t, x]:
        t += 1
    return t


def _frame_ok(mask, comp) -> tuple[bool, int]:
    x0, y0, x1, y1 = comp["box"]
    w, h = x1 - x0 + 1, y1 - y0 + 1
    if not (120 <= w <= 320 and 120 <= h <= 360):
        return False, 0
    t = _rack_stroke(mask, comp["box"])
    if not 3 <= t <= 9 or w <= 2 * t + 20 or h <= 2 * t + 20:
        return False, 0
    ring = w * h - (w - 2 * t) * (h - 2 * t)
    if abs(comp["area"] - ring) > max(12, 0.02 * ring):
        return False, 0
    band = mask[y0 : y1 + 1, x0 : x1 + 1]
    interior = band[t : h - t, t : w - t]
    if interior.any():
        return False, 0
    if int(band.sum()) - int(interior.sum()) < 0.995 * ring:
        return False, 0
    for strip in (band[:t, :], band[h - t :, :], band[:, :t], band[:, w - t :]):
        if strip.mean() < 0.995:
            return False, 0
    return True, t


def _colour_gap(values) -> float:
    vals = sorted(set(values))
    if len(vals) < 2:
        return 442.0
    rgb = [((v >> 16) & 255, (v >> 8) & 255, v & 255) for v in vals]
    best = 442.0
    for i in range(len(rgb)):
        for j in range(i + 1, len(rgb)):
            d = sum((a - b) ** 2 for a, b in zip(rgb[i], rgb[j])) ** 0.5
            best = min(best, d)
    return best


def decision_from_image(image: "Image.Image") -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    h, w = arr.shape[0], arr.shape[1]
    packed = (arr[:, :, 0] << 16) | (arr[:, :, 1] << 8) | arr[:, :, 2]
    bg = _packed(_BG)
    rack_val = _packed(_RACK)
    comps = _components(packed.reshape(-1).tolist(), w, h, bg)

    for comp in comps:
        if _FRAGMENT <= comp["area"] < _MIN_BLOB:
            return ABSTAIN
    comps = [c for c in comps if c["area"] >= _MIN_BLOB]

    rack_mask = packed == rack_val
    racks = [c for c in comps if c["colour"] == rack_val and c["area"] >= _MIN_RACK_AREA]
    if len(racks) != 2 or any(c["colour"] == rack_val for c in comps if c not in racks):
        return ABSTAIN

    strokes = []
    for comp in racks:
        ok, t = _frame_ok(rack_mask, comp)
        if not ok:
            return ABSTAIN
        strokes.append(t)
    racks_sorted = sorted(zip(racks, strokes), key=lambda rs: rs[0]["box"][0] + rs[0]["box"][2])
    (ra, ta), (rb, tb) = racks_sorted
    if rb["box"][0] - ra["box"][2] < 80:
        return ABSTAIN

    blobs = [c for c in comps if c["colour"] != rack_val]
    if not 4 <= len(blobs) <= 16:
        return ABSTAIN

    sides: list[list[int]] = [[], []]
    for comp in blobs:
        box = comp["box"]
        placed = None
        for idx, (rack, t) in enumerate(((ra, ta), (rb, tb))):
            rx0, ry0, rx1, ry1 = rack["box"]
            if _inside((rx0 + t, ry0 + t, rx1 - t, ry1 - t), box):
                placed = idx
                break
        if placed is None:
            if all(_sep(box, r["box"]) >= 6 for r, _t in ((ra, ta), (rb, tb))):
                continue  # loose mitten on the floor: ignored by the stated rule
            return ABSTAIN
        sides[placed].append(comp["colour"])

    for side in sides:
        if not 2 <= len(side) <= 5 or len(set(side)) != len(side):
            return ABSTAIN
    if _colour_gap(sides[0] + sides[1]) < 60.0:
        return ABSTAIN

    shared = len(set(sides[0]) & set(sides[1]))
    if shared > 3:
        return ABSTAIN
    return str(shared)
