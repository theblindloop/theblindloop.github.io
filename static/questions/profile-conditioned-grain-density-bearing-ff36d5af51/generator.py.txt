"""Grain-density bearing world: a speck field whose crowding drifts one way.

The latent scene stores only rendered quantities: canvas, palette, speck radius,
the explicit list of speck centres and the four coloured marker discs. The ramp
direction is never stored. Both arms recover the direction of increasing grain
density from the realised speck field itself.
"""

from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 640
HEIGHT = 640
SPECK_R = 3.0
MARKER_R = 11.0
SPECK_COUNT = 800
MARKER_RING = 0.40 * WIDTH

ABSTAIN = "abstain"
CLASS_NAMES = ("red", "green", "blue", "orange")

QUARANTINE_DEG = 10.0
SAMPLING_FLOOR_DEG = 18.0
MIN_MARKER_SEPARATION_DEG = 55.0

BG_COLOR = (246, 245, 241)
SPECK_COLOR = (62, 62, 70)
MARKER_COLORS = {
    "red": (212, 40, 40),
    "green": (24, 140, 60),
    "blue": (38, 66, 204),
    "orange": (230, 132, 16),
}


def _angle_gap_deg(a: float, b: float) -> float:
    """Unsigned separation between two bearings, in degrees."""
    delta = math.degrees(abs(math.atan2(math.sin(a - b), math.cos(a - b))))
    return delta


def _uphill_bearing(scene: dict) -> float:
    """Bearing of the first moment of the realised grain field about the centre."""
    specks = scene["specks"]
    cx = float(scene["width"]) / 2.0
    cy = float(scene["height"]) / 2.0
    sx = sum(float(p[0]) for p in specks) / len(specks) - cx
    sy = sum(float(p[1]) for p in specks) / len(specks) - cy
    return math.atan2(sy, sx)


def _ranked_markers(scene: dict) -> list[tuple[float, str]]:
    """Markers ordered by how closely their bearing matches the uphill bearing."""
    bearing = _uphill_bearing(scene)
    cx = float(scene["width"]) / 2.0
    cy = float(scene["height"]) / 2.0
    ranked = [
        (
            _angle_gap_deg(math.atan2(float(m["y"]) - cy, float(m["x"]) - cx), bearing),
            str(m["name"]),
        )
        for m in scene["markers"]
    ]
    ranked.sort(key=lambda item: (item[0], item[1]))
    return ranked


def analytic_gold(scene: dict) -> str:
    ranked = _ranked_markers(scene)
    if len(ranked) < 2 or ranked[0][0] >= ranked[1][0]:
        return ABSTAIN
    return ranked[0][1]


def margin(scene: dict) -> float:
    """Degrees by which the winning marker beats the runner-up."""
    ranked = _ranked_markers(scene)
    if len(ranked) < 2:
        return 0.0
    return float(ranked[1][0] - ranked[0][0])


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_DEG)


def _sample_markers(rng: random.Random) -> list[dict]:
    cx, cy = WIDTH / 2.0, HEIGHT / 2.0
    while True:
        angles = sorted(rng.uniform(0.0, 2.0 * math.pi) for _ in range(4))
        gaps = [
            math.degrees(
                (angles[(i + 1) % 4] - angles[i]) % (2.0 * math.pi)
            )
            for i in range(4)
        ]
        if min(gaps) < MIN_MARKER_SEPARATION_DEG:
            continue
        names = list(CLASS_NAMES)
        rng.shuffle(names)
        return [
            {
                "name": names[i],
                "x": round(cx + MARKER_RING * math.cos(angles[i]), 3),
                "y": round(cy + MARKER_RING * math.sin(angles[i]), 3),
            }
            for i in range(4)
        ]


def _sample_specks(rng: random.Random, theta: float, markers: list[dict]) -> list[list[float]]:
    """Rejection sampling: a linear density ramp along theta, no speck overlap."""
    ux, uy = math.cos(theta), math.sin(theta)
    cx, cy = WIDTH / 2.0, HEIGHT / 2.0
    keep_out = (MARKER_R + SPECK_R + 3.0) ** 2
    min_sep_sq = 10.0 ** 2
    pts: list[list[float]] = []
    arr_x: list[float] = []
    arr_y: list[float] = []
    margin_px = SPECK_R + 4.0
    attempts = 0
    while len(pts) < SPECK_COUNT and attempts < 400000:
        attempts += 1
        x = rng.uniform(margin_px, WIDTH - margin_px)
        y = rng.uniform(margin_px, HEIGHT - margin_px)
        proj = ((x - cx) * ux + (y - cy) * uy) / (WIDTH / 2.0)
        weight = 0.625 + 0.375 * max(-1.0, min(1.0, proj))
        if rng.random() > weight:
            continue
        if any((x - float(m["x"])) ** 2 + (y - float(m["y"])) ** 2 < keep_out for m in markers):
            continue
        ok = True
        for px, py in zip(arr_x, arr_y):
            if (x - px) ** 2 + (y - py) ** 2 < min_sep_sq:
                ok = False
                break
        if not ok:
            continue
        pts.append([round(x, 3), round(y, 3)])
        arr_x.append(x)
        arr_y.append(y)
    return pts


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    for _ in range(200):
        markers = _sample_markers(rng)
        theta = rng.uniform(0.0, 2.0 * math.pi)
        specks = _sample_specks(rng, theta, markers)
        if len(specks) < SPECK_COUNT:
            continue
        scene = {
            "width": WIDTH,
            "height": HEIGHT,
            "bg_color": list(BG_COLOR),
            "speck_color": list(SPECK_COLOR),
            "speck_radius": SPECK_R,
            "marker_radius": MARKER_R,
            "marker_colors": {k: list(v) for k, v in MARKER_COLORS.items()},
            "markers": markers,
            "specks": specks,
        }
        if analytic_gold(scene) == ABSTAIN:
            continue
        if margin(scene) < SAMPLING_FLOOR_DEG:
            continue
        return scene
    raise RuntimeError(f"sampling failed for seed {seed}")


def _disc(draw: ImageDraw.ImageDraw, x: float, y: float, r: float, color: tuple) -> None:
    draw.ellipse([x - r, y - r, x + r, y + r], fill=color)


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width, height), tuple(scene["bg_color"]))
    draw = ImageDraw.Draw(image)
    speck_color = tuple(scene["speck_color"])
    radius = float(scene["speck_radius"])
    for x, y in scene["specks"]:
        _disc(draw, float(x), float(y), radius, speck_color)
    marker_radius = float(scene["marker_radius"])
    for m in scene["markers"]:
        _disc(
            draw,
            float(m["x"]),
            float(m["y"]),
            marker_radius,
            tuple(scene["marker_colors"][m["name"]]),
        )
    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical relabellings: the speck list and marker list are unordered."""
    specks = [list(p) for p in scene["specks"]]
    markers = [dict(m) for m in scene["markers"]]
    alias_a = dict(scene)
    alias_a["specks"] = specks[1:] + specks[:1]
    alias_b = dict(scene)
    alias_b["specks"] = list(reversed(specks))
    alias_b["markers"] = markers[1:] + markers[:1]
    alias_c = dict(scene)
    alias_c["markers"] = list(reversed(markers))
    return [
        ("speck_list_cycle", alias_a),
        ("speck_reverse_and_marker_cycle", alias_b),
        ("marker_list_reverse", alias_c),
    ]


def ink_field(image: Image.Image, cells: int = 16) -> np.ndarray:
    """Diagnostic helper (not used by the oracle): coarse grain-energy field."""
    arr = np.asarray(image.convert("RGB")).astype(np.int16)
    sat = arr.max(axis=2) - arr.min(axis=2)
    grey = (sat < 30) & (arr.mean(axis=2) < 160)
    h, w = grey.shape
    field = np.zeros((cells, cells), dtype=np.float64)
    for i in range(cells):
        for j in range(cells):
            block = grey[
                i * h // cells : (i + 1) * h // cells,
                j * w // cells : (j + 1) * w // cells,
            ]
            field[i, j] = float(block.sum())
    return field
