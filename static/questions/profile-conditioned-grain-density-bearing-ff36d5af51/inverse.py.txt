"""Pixel-only decision arm for the grain-density bearing world.

Receives one RGB image and nothing else. It rebuilds the grain-energy field from
the grey ink, takes the field's first moment about the canvas centre to obtain
the bearing in which the grain crowds, and names the coloured marker whose own
bearing from the centre is closest to it. No renderer, prompt, generator,
verifier or analytic-gold code is imported, and no latent scene is accepted.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ABSTAIN = "abstain"

# Colour vocabulary of the public question contract, not renderer state.
REFERENCE_COLORS = {
    "red": (220, 40, 40),
    "green": (30, 140, 60),
    "blue": (40, 70, 200),
    "orange": (230, 130, 20),
}

SATURATION_CUT = 55
GREY_SATURATION_CUT = 30
GREY_VALUE_CUT = 170
MIN_MARKER_PIXELS = 40
MIN_DECISION_GAP_DEG = 2.0


def _angle_gap_deg(a: float, b: float) -> float:
    return math.degrees(abs(math.atan2(math.sin(a - b), math.cos(a - b))))


def _masks(arr: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    high = arr.max(axis=2)
    low = arr.min(axis=2)
    sat = high - low
    value = arr.mean(axis=2)
    grain = (sat < GREY_SATURATION_CUT) & (value < GREY_VALUE_CUT)
    colored = sat >= SATURATION_CUT
    return grain, colored


def _grain_bearing(grain: np.ndarray) -> float:
    ys, xs = np.nonzero(grain)
    if xs.size == 0:
        raise ValueError("no grain ink found")
    height, width = grain.shape
    cx = width / 2.0
    cy = height / 2.0
    return math.atan2(float(ys.mean()) - cy, float(xs.mean()) - cx)


def _marker_bearings(arr: np.ndarray, colored: np.ndarray) -> dict[str, float]:
    ys, xs = np.nonzero(colored)
    if xs.size == 0:
        raise ValueError("no coloured markers found")
    height, width = colored.shape
    pixels = arr[ys, xs].astype(np.float64)
    names = list(REFERENCE_COLORS)
    refs = np.array([REFERENCE_COLORS[n] for n in names], dtype=np.float64)
    dist = ((pixels[:, None, :] - refs[None, :, :]) ** 2).sum(axis=2)
    assign = dist.argmin(axis=1)
    bearings: dict[str, float] = {}
    for index, name in enumerate(names):
        sel = assign == index
        if int(sel.sum()) < MIN_MARKER_PIXELS:
            continue
        mx = float(xs[sel].mean()) - width / 2.0
        my = float(ys[sel].mean()) - height / 2.0
        bearings[name] = math.atan2(my, mx)
    return bearings


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.int16)
    grain, colored = _masks(arr)
    try:
        bearing = _grain_bearing(grain)
        markers = _marker_bearings(arr, colored)
    except ValueError:
        return ABSTAIN
    if len(markers) < 2:
        return ABSTAIN
    ranked = sorted(
        ((_angle_gap_deg(b, bearing), name) for name, b in markers.items()),
        key=lambda item: (item[0], item[1]),
    )
    if ranked[1][0] - ranked[0][0] < MIN_DECISION_GAP_DEG:
        return ABSTAIN
    return ranked[0][1]
