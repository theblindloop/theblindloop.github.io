"""Independent PNG-only inverse arm for trace excursion spans."""

from __future__ import annotations

import numpy as np
from PIL import Image

COLORS = {
    "red": (205, 52, 52),
    "blue": (40, 105, 190),
    "green": (45, 150, 75),
}
ORANGE = (230, 126, 34)


def _mask_near(arr: np.ndarray, rgb: tuple[int, int, int], radius: float = 82.0) -> np.ndarray:
    target = np.asarray(rgb, dtype=np.float32)
    return np.linalg.norm(arr.astype(np.float32) - target, axis=2) < radius


def _row_clusters(mask: np.ndarray) -> list[float]:
    counts = mask.sum(axis=1)
    rows = np.flatnonzero(counts > 24)
    if len(rows) == 0:
        raise ValueError("orange benchmark ink is absent")
    groups: list[list[int]] = [[int(rows[0])]]
    for row in rows[1:]:
        if int(row) - groups[-1][-1] > 3:
            groups.append([])
        groups[-1].append(int(row))
    centers = [float(np.average(g, weights=counts[g])) for g in groups if len(g) <= 8]
    if len(centers) != 3:
        raise ValueError(f"expected three benchmark rows, recovered {len(centers)}")
    return centers


def _curve_width(mask: np.ndarray, target_y: float) -> float:
    xs = np.flatnonzero(mask.sum(axis=0) >= 2)
    if len(xs) < 200:
        raise ValueError("colored curve is missing or fragmented")
    above: list[int] = []
    for x in xs:
        ys = np.flatnonzero(mask[:, x])
        if len(ys) and float(np.median(ys)) < target_y - 1.0:
            above.append(int(x))
    if len(above) < 30:
        raise ValueError("curve has no resolved above-benchmark excursion")
    # The solid trace produces one contiguous excursion; isolated antialias specks
    # are removed by keeping only columns supported by a neighboring column.
    aset = set(above)
    stable = [x for x in above if x - 1 in aset or x + 1 in aset]
    if not stable:
        raise ValueError("crossings are unresolved")
    return float(max(stable) - min(stable))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    targets = _row_clusters(_mask_near(arr, ORANGE, 64.0))
    measured: list[tuple[float, str]] = []
    for name, rgb in COLORS.items():
        mask = _mask_near(arr, rgb)
        ys = np.flatnonzero(mask.sum(axis=1) > 0)
        if len(ys) == 0:
            raise ValueError(f"{name} curve is absent")
        curve_y = float(np.median(ys))
        target_y = min(targets, key=lambda y: abs(y - curve_y))
        measured.append((_curve_width(mask, target_y), name))
    measured.sort(reverse=True)
    if measured[0][0] - measured[1][0] < 18.0:
        raise ValueError("pixel decision margin is ambiguous")
    return measured[0][1]
