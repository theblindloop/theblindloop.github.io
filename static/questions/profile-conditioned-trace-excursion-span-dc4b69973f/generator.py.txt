"""Latent scene sampler and raster renderer for trace excursion spans."""

from __future__ import annotations

import copy
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 640, 440
SCALE = 3
PALETTE = (
    ("red", (205, 52, 52)),
    ("blue", (40, 105, 190)),
    ("green", (45, 150, 75)),
)
ORANGE = (230, 126, 34)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    winner = int(seed) % 3
    remaining = [i for i in range(3) if i != winner]
    rng.shuffle(remaining)
    ranks = {winner: rng.uniform(248.0, 278.0), remaining[0]: rng.uniform(182.0, 211.0), remaining[1]: rng.uniform(118.0, 154.0)}
    panels = [0, 1, 2]
    rng.shuffle(panels)
    target_rows = [120.0, 235.0, 350.0]
    curves = []
    for color_index, (name, rgb) in enumerate(PALETTE):
        span = ranks[color_index]
        center = rng.uniform(286.0, 354.0)
        left = center - span / 2.0
        right = center + span / 2.0
        panel = panels[color_index]
        curves.append(
            {
                "color": name,
                "rgb": list(rgb),
                "panel": panel,
                "crossing_left": round(left, 3),
                "crossing_right": round(right, 3),
                "peak_x": round(rng.uniform(left + 0.30 * span, left + 0.70 * span), 3),
                "peak_height": round(rng.uniform(25.0, 39.0), 3),
                "left_depth": round(rng.uniform(21.0, 34.0), 3),
                "right_depth": round(rng.uniform(21.0, 34.0), 3),
            }
        )
    return {
        "curves": curves,
        "target_rows": target_rows,
        "style_variant": rng.randrange(4),
        "audit_tag": rng.randrange(1_000_000),
    }


def _scaled(points: list[tuple[float, float]]) -> list[tuple[int, int]]:
    return [(round(x * SCALE), round(y * SCALE)) for x, y in points]


def render(scene: dict) -> Image.Image:
    curves = scene["curves"]
    targets = scene["target_rows"]
    variant = int(scene["style_variant"])
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), (250, 251, 253))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    draw.text((66 * SCALE, 18 * SCALE), "ABOVE-BENCHMARK EXCURSION", fill=(55, 61, 70), font=font, stroke_width=0)
    draw.text((66 * SCALE, 35 * SCALE), "Compare the distance between each curve's two crossings", fill=(105, 111, 120), font=font)
    panel_tops = (67.0, 182.0, 297.0)
    for panel, top in enumerate(panel_tops):
        bottom = top + 91.0
        shade = 255 if (panel + variant) % 2 == 0 else 253
        draw.rectangle((58 * SCALE, round(top * SCALE), 590 * SCALE, round(bottom * SCALE)), fill=(shade, shade, shade), outline=(215, 219, 224), width=1 * SCALE)
        draw.text((70 * SCALE, round((top + 8) * SCALE)), f"PANEL {panel + 1}", fill=(135, 140, 148), font=font)
        y = targets[panel]
        for x0 in range(78, 574, 18):
            draw.line(_scaled([(x0, y), (min(x0 + 10, 574), y)]), fill=ORANGE, width=2 * SCALE)

    for curve in curves:
        y0 = targets[int(curve["panel"])]
        points = [
            (78.0, y0 + float(curve["left_depth"])),
            (float(curve["crossing_left"]), y0),
            (float(curve["peak_x"]), y0 - float(curve["peak_height"])),
            (float(curve["crossing_right"]), y0),
            (574.0, y0 + float(curve["right_depth"])),
        ]
        draw.line(_scaled(points), fill=tuple(curve["rgb"]), width=5 * SCALE, joint="curve")
        # Small neutral endpoint marks emphasize the common horizontal domain.
        for x, y in (points[0], points[-1]):
            r = 2.0
            draw.ellipse((round((x-r)*SCALE), round((y-r)*SCALE), round((x+r)*SCALE), round((y+r)*SCALE)), fill=tuple(curve["rgb"]))

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def _spans(scene: dict) -> list[tuple[float, str]]:
    return [
        (float(curve["crossing_right"]) - float(curve["crossing_left"]), str(curve["color"]))
        for curve in scene["curves"]
    ]


def analytic_gold(scene: dict) -> str:
    return max(_spans(scene))[1]


def margin(scene: dict) -> float:
    ordered = sorted((span for span, _ in _spans(scene)), reverse=True)
    return ordered[0] - ordered[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 30.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["audit_tag"] = int(twin["audit_tag"]) + 1
    return [("audit_tag_alias", twin)]
