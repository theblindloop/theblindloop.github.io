"""Deterministic renderer and analytic label for the mosaic correspondence world."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "multipart_halo_binding_correspondence-0.1.0"
CANVAS_SIZE = (1260, 560)
SUPERSAMPLE = 4
GRID_STEP = 48.0
BASE_COORDS = ((0, 0), (1, 0), (2, 0), (0, 1), (0, 2))
TILE_COLORS = ("red", "blue", "green", "gold", "purple")
TILE_PALETTE = {
    "red": (220, 50, 47),
    "blue": (38, 91, 185),
    "green": (40, 151, 77),
    "gold": (235, 157, 24),
    "purple": (135, 70, 170),
}
HALO_COLORS = ("amber", "cyan", "violet")
HALO_PALETTE = {
    "amber": (238, 150, 24),
    "cyan": (20, 165, 185),
    "violet": (143, 77, 184),
}
PANEL_BOXES = {
    "reference": (24, 54, 386, 506),
    "A": (446, 54, 808, 506),
    "B": (870, 54, 1232, 506),
}
PANEL_CENTERS = {
    "reference": (205.0, 300.0),
    "A": (627.0, 300.0),
    "B": (1051.0, 300.0),
}
CELL_SIZE_CHOICES = (30.0, 34.0, 38.0)


def _turn(point: tuple[int, int], turns: int) -> tuple[int, int]:
    x, y = point
    for _ in range(turns % 4):
        x, y = -y, x
    return x, y


def _normalise_cells(cells: Iterable[dict]) -> list[dict]:
    raw = [(int(cell["x"]), int(cell["y"]), str(cell["color"])) for cell in cells]
    min_x = min(item[0] for item in raw)
    min_y = min(item[1] for item in raw)
    return [
        {"x": x - min_x, "y": y - min_y, "color": color}
        for x, y, color in sorted(raw, key=lambda item: (item[1], item[0], item[2]))
    ]


def _make_cells(colors: list[str], turns: int) -> list[dict]:
    transformed = []
    for coord, color in zip(BASE_COORDS, colors, strict=True):
        x, y = _turn(coord, turns)
        transformed.append({"x": x, "y": y, "color": color})
    return _normalise_cells(transformed)


def _rotated_map(cells: list[dict], turns: int) -> dict[tuple[int, int], str]:
    rotated = []
    for cell in cells:
        x, y = _turn((int(cell["x"]), int(cell["y"])), turns)
        rotated.append({"x": x, "y": y, "color": str(cell["color"])})
    normal = _normalise_cells(rotated)
    return {(int(cell["x"]), int(cell["y"])): str(cell["color"]) for cell in normal}


def _pattern_mismatch(reference: list[dict], candidate: list[dict]) -> int:
    """Return the least number of local tile assignments that disagree."""

    reference_map = _rotated_map(reference, 0)
    best = 10**6
    for turns in range(4):
        candidate_map = _rotated_map(candidate, turns)
        coordinates = set(reference_map) | set(candidate_map)
        mismatch = sum(reference_map.get(coord) != candidate_map.get(coord) for coord in coordinates)
        best = min(best, mismatch)
    return int(best)


def _candidate_scores(scene: dict) -> tuple[int, int]:
    reference = scene["reference"]
    return (
        _pattern_mismatch(reference["cells"], scene["candidate_a"]["cells"]),
        _pattern_mismatch(reference["cells"], scene["candidate_b"]["cells"]),
    )


def sample_scene(seed: int) -> dict:
    """Sample a balanced reference/decoy pair with independent visual nuisances."""

    rng = random.Random(int(seed))
    colors = list(TILE_COLORS)
    rng.shuffle(colors)
    reference_cells = _make_cells(colors, rng.randrange(4))

    wrong_colors = list(colors)
    first, second = rng.sample(range(len(wrong_colors)), 2)
    wrong_colors[first], wrong_colors[second] = wrong_colors[second], wrong_colors[first]
    correct_cells = _make_cells(colors, rng.randrange(4))
    wrong_cells = _make_cells(wrong_colors, rng.randrange(4))

    halos = list(HALO_COLORS)
    rng.shuffle(halos)

    def centre(panel: str) -> list[float]:
        cx, cy = PANEL_CENTERS[panel]
        return [round(cx + rng.uniform(-24.0, 24.0), 3), round(cy + rng.uniform(-18.0, 18.0), 3)]

    def obj(cells: list[dict], halo: str | None, panel: str) -> dict:
        return {
            "cells": cells,
            "halo": halo,
            "center": centre(panel),
            "cell_size": rng.choice(CELL_SIZE_CHOICES),
        }

    reference = obj(reference_cells, None, "reference")
    correct = obj(correct_cells, halos[0], "A")
    wrong = obj(wrong_cells, halos[1], "B")
    correct_is_a = int(seed) % 2 == 0
    if correct_is_a:
        candidate_a, candidate_b = correct, wrong
    else:
        candidate_a, candidate_b = wrong, correct
    return {
        "reference": reference,
        "candidate_a": candidate_a,
        "candidate_b": candidate_b,
    }


def _cell_centres(obj: dict) -> list[tuple[float, float, dict]]:
    cells = obj["cells"]
    center_x, center_y = (float(value) for value in obj["center"])
    min_x = min(int(cell["x"]) for cell in cells)
    max_x = max(int(cell["x"]) for cell in cells)
    min_y = min(int(cell["y"]) for cell in cells)
    max_y = max(int(cell["y"]) for cell in cells)
    origin_x = center_x - (min_x + max_x) * GRID_STEP / 2.0
    origin_y = center_y - (min_y + max_y) * GRID_STEP / 2.0
    return [
        (origin_x + int(cell["x"]) * GRID_STEP, origin_y + int(cell["y"]) * GRID_STEP, cell)
        for cell in cells
    ]


def _font(size: int):
    try:
        return ImageFont.load_default(size=size)
    except TypeError:
        return ImageFont.load_default()


def _draw_object(draw: ImageDraw.ImageDraw, obj: dict, scale: int, neutral: bool = False) -> None:
    centres = _cell_centres(obj)
    tile_size = float(obj["cell_size"])
    min_cx = min(item[0] for item in centres)
    max_cx = max(item[0] for item in centres)
    min_cy = min(item[1] for item in centres)
    max_cy = max(item[1] for item in centres)
    pad = 17.0 if neutral else 20.0
    left = (min_cx - tile_size / 2.0 - pad) * scale
    top = (min_cy - tile_size / 2.0 - pad) * scale
    right = (max_cx + tile_size / 2.0 + pad) * scale
    bottom = (max_cy + tile_size / 2.0 + pad) * scale
    if neutral:
        outline = (95, 101, 113)
        width = 4 * scale
    else:
        outline = HALO_PALETTE[str(obj["halo"])]
        width = 9 * scale
    draw.rounded_rectangle(
        (round(left), round(top), round(right), round(bottom)),
        radius=14 * scale,
        outline=outline,
        width=width,
    )
    half = tile_size / 2.0
    for cx, cy, cell in centres:
        box = (
            round((cx - half) * scale),
            round((cy - half) * scale),
            round((cx + half) * scale),
            round((cy + half) * scale),
        )
        draw.rectangle(
            box,
            fill=TILE_PALETTE[str(cell["color"])],
            outline=(35, 39, 48),
            width=2 * scale,
        )


def render(scene: dict) -> Image.Image:
    """Render only the public visual fields of the scene."""

    width, height = CANVAS_SIZE
    scale = SUPERSAMPLE
    image = Image.new("RGB", (width * scale, height * scale), (247, 248, 251))
    draw = ImageDraw.Draw(image)
    labels = {"reference": "REFERENCE", "A": "A", "B": "B"}
    for panel, box in PANEL_BOXES.items():
        left, top, right, bottom = box
        draw.rounded_rectangle(
            (left * scale, top * scale, right * scale, bottom * scale),
            radius=14 * scale,
            fill=(255, 255, 255),
            outline=(194, 198, 207),
            width=2 * scale,
        )
        draw.text(
            ((left + right) * scale / 2.0, (top + 21) * scale),
            labels[panel],
            fill=(54, 59, 70),
            anchor="mm",
            font=_font(18 * scale),
        )
    _draw_object(draw, scene["reference"], scale, neutral=True)
    _draw_object(draw, scene["candidate_a"], scale)
    _draw_object(draw, scene["candidate_b"], scale)
    return image.resize((width, height), Image.Resampling.LANCZOS).convert("RGB")


def analytic_gold(scene: dict) -> str:
    """Select a unique structural match, then read its attached halo attribute."""

    scores = _candidate_scores(scene)
    if scores.count(0) != 1 or scores[0] == scores[1]:
        raise ValueError(f"ambiguous mosaic correspondence: scores={scores}")
    index = 0 if scores[0] == 0 else 1
    answer = scene["candidate_a"]["halo"] if index == 0 else scene["candidate_b"]["halo"]
    if answer not in HALO_COLORS:
        raise ValueError(f"invalid halo attribute {answer!r}")
    return str(answer)


def margin(scene: dict) -> float:
    """Runner-up mismatch gap; larger means a clearer discrete assignment."""

    scores = _candidate_scores(scene)
    return float(abs(scores[0] - scores[1]))


def is_quarantined(scene: dict) -> bool:
    try:
        scores = _candidate_scores(scene)
        return scores.count(0) != 1 or scores[0] == scores[1] or margin(scene) < 1.0
    except (KeyError, TypeError, ValueError):
        return True

    """No latent aliases: all label-bearing fields are directly rendered."""
