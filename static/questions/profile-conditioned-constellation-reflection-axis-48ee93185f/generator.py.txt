"""Deterministic latent sampler, analytic arm, and raster renderer."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

ORIENTATIONS = ("vertical", "horizontal", "diag_down", "diag_up")
COLOR_NAMES = ("red", "blue", "green", "orange")
AXIS_RGB = {
    "red": (211, 55, 48),
    "blue": (42, 104, 177),
    "green": (34, 139, 78),
    "orange": (225, 126, 35),
}


def _reflect_relative(point: tuple[int, int], orientation: str) -> tuple[int, int]:
    x, y = point
    if orientation == "vertical":
        return -x, y
    if orientation == "horizontal":
        return x, -y
    if orientation == "diag_down":
        return y, x
    if orientation == "diag_up":
        return -y, -x
    raise ValueError(f"unknown orientation {orientation!r}")


def _point_scores(points: list[list[int]], center: list[int]) -> dict[str, float]:
    cx, cy = center
    relative = {(int(x) - cx, int(y) - cy) for x, y in points}
    scores: dict[str, float] = {}
    for orientation in ORIENTATIONS:
        reflected = {_reflect_relative(point, orientation) for point in relative}
        union = relative | reflected
        scores[orientation] = len(relative ^ reflected) / max(1, len(union))
    return scores


def _canonical_half(point: tuple[int, int], orientation: str) -> bool:
    x, y = point
    if orientation == "vertical":
        return x > 0
    if orientation == "horizontal":
        return y > 0
    if orientation == "diag_down":
        return y > x
    return y > -x


def sample_scene(seed: int) -> dict:
    """Sample a balanced, randomized constellation with one exact reflection axis."""
    rng = random.Random(int(seed))
    target = ORIENTATIONS[int(seed) % len(ORIENTATIONS)]
    center = [256 + rng.randint(-12, 12), 256 + rng.randint(-12, 12)]
    step = rng.choice((22, 24, 26))

    for _attempt in range(500):
        relative: set[tuple[int, int]] = set()
        tries = 0
        while len(relative) < 14 and tries < 500:
            tries += 1
            point = (rng.randint(-7, 7) * step, rng.randint(-7, 7) * step)
            if point == (0, 0) or not _canonical_half(point, target):
                continue
            twin = _reflect_relative(point, target)
            if point == twin or point in relative or twin in relative:
                continue
            if any(math.hypot(point[0] - q[0], point[1] - q[1]) < 1.5 * step for q in relative):
                continue
            relative.update((point, twin))
        if len(relative) != 14:
            continue
        points = [[center[0] + x, center[1] + y] for x, y in relative]
        scores = _point_scores(points, center)
        ordered = sorted(scores.values())
        if scores[target] == 0.0 and ordered[1] >= 0.55:
            break
    else:
        raise RuntimeError("could not sample a well-separated constellation")

    rng.shuffle(points)
    shift = (int(seed) // 4) % 4
    axis_colors = {
        orientation: COLOR_NAMES[(index + shift) % 4]
        for index, orientation in enumerate(ORIENTATIONS)
    }
    return {
        "width": 512,
        "height": 512,
        "center": center,
        "points": points,
        "radius": rng.choice((8, 9, 10)),
        "axis_colors": axis_colors,
        "axis_width": 3,
        "background": [248, 247, 242],
        "dot_color": [27, 32, 43],
        "frame_color": [190, 190, 184],
    }


def _line_endpoints(width: int, height: int, center: list[int], orientation: str):
    left, top, right, bottom = 28, 28, width - 29, height - 29
    cx, cy = center
    if orientation == "vertical":
        return [(cx, top), (cx, bottom)]
    if orientation == "horizontal":
        return [(left, cy), (right, cy)]
    slope = 1 if orientation == "diag_down" else -1
    hits: list[tuple[int, int]] = []
    for x in (left, right):
        y = cy + slope * (x - cx)
        if top <= y <= bottom:
            hits.append((x, y))
    for y in (top, bottom):
        x = cx + (y - cy) // slope
        if left <= x <= right:
            hits.append((x, y))
    unique = list(dict.fromkeys(hits))
    if len(unique) < 2:
        raise ValueError("guide does not cross frame")
    return [unique[0], unique[-1]]


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    center = [int(value) for value in scene["center"]]
    points = scene["points"]
    radius = int(scene["radius"])
    axis_colors = scene["axis_colors"]
    axis_width = int(scene["axis_width"])
    background = tuple(scene["background"])
    dot_color = tuple(scene["dot_color"])
    frame_color = tuple(scene["frame_color"])

    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)
    draw.rectangle((28, 28, width - 29, height - 29), outline=frame_color, width=2)
    for orientation in ORIENTATIONS:
        draw.line(
            _line_endpoints(width, height, center, orientation),
            fill=AXIS_RGB[axis_colors[orientation]],
            width=axis_width,
        )
    for x, y in points:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=dot_color)
    return image


def analytic_gold(scene: dict) -> str:
    scores = _point_scores(scene["points"], scene["center"])
    winning_orientation = min(ORIENTATIONS, key=lambda name: (scores[name], name))
    return scene["axis_colors"][winning_orientation]


def margin(scene: dict) -> float:
    scores = sorted(_point_scores(scene["points"], scene["center"]).values())
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.20


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["points"] = list(reversed(twin["points"]))
    return [("reverse_point_order", twin)]
