"""Fold-and-punch world: unfold a punched quarter and probe it with dots.

A square sheet is ruled into an 8 by 8 lattice of cells and carries two drawn
creases, the vertical and the horizontal midline.  Exactly one quarter of the
sheet holds solid black squares, the punched holes.  Folding the sheet along
both creases stacks the four quarters, so a hole punched through the stack
reappears in every quarter at the *mirror image* of its drawn position, never
at a translated copy of it.

Four colored dots sit in the other quarters.  Exactly one of them lands on a
hole of the unfolded sheet; the decision is that dot's color.  The unfolded
hole pattern itself is drawn nowhere: it has to be generated as the orbit of
the drawn holes under the group generated by the two drawn mirrors, and the
four dots have to be tested for membership in that orbit.

Only axis-aligned filled rectangles and circles are drawn, so the raster holds
a small set of exact colors with no antialiasing, and rendering a stored scene
reproduces its PNG bit for bit.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 780
HEIGHT = 780

GRID = 8                 # cells across the whole sheet
HALF = GRID // 2         # cells across one quarter
PITCH = 86               # cell pitch in pixels
SHEET = GRID * PITCH     # 688
ORIGIN = 46              # nominal sheet origin before jitter
JITTER = 8

BORDER_W = 4             # sheet border, drawn inward
GRID_W = 2               # interior rule lines
CREASE_W = 6             # crease stroke
DASH_ON = 24
DASH_OFF = 16

MARK = 52                # hole square side and dot diameter

QUARANTINE_PX = 11

WHITE = (255, 255, 255)
BLACK = (0, 0, 0)
RULE = (198, 198, 198)

COLOR_NAMES = ("blue", "green", "red", "yellow")
COLORS = {
    "blue": (30, 90, 220),
    "green": (20, 150, 60),
    "red": (220, 40, 40),
    "yellow": (245, 200, 30),
}

MIN_HOLES = 3
MAX_HOLES = 5


# --------------------------------------------------------------------------
# scene sampling
# --------------------------------------------------------------------------


def _quadrant(cell: tuple[int, int]) -> tuple[int, int]:
    r, c = cell
    return (r // HALF, c // HALF)


def unfolded_cells(holes) -> set[tuple[int, int]]:
    """Orbit of the drawn holes under the two drawn mirrors."""

    out: set[tuple[int, int]] = set()
    for r, c in holes:
        out.add((r, c))
        out.add((r, GRID - 1 - c))
        out.add((GRID - 1 - r, c))
        out.add((GRID - 1 - r, GRID - 1 - c))
    return out


def translated_cells(holes) -> set[tuple[int, int]]:
    """Where the punched cells would land if the fold *translated* them.

    This is the policy of a solver who copies the punched quarter into the
    other quarters instead of mirroring it; it is drawn nowhere and is never
    the gold.
    """

    out: set[tuple[int, int]] = set()
    for r, c in holes:
        a, b = r % HALF, c % HALF
        for qr in range(2):
            for qc in range(2):
                out.add((qr * HALF + a, qc * HALF + b))
    return out


def half_mirrored_cells(holes) -> set[tuple[int, int]]:
    """Diagonal-quarter cells reached by applying only one of the two mirrors."""

    out: set[tuple[int, int]] = set()
    for r, c in holes:
        qr, qc = _quadrant((r, c))
        a, b = r % HALF, c % HALF
        dr, dc = 1 - qr, 1 - qc
        out.add((dr * HALF + a, dc * HALF + (HALF - 1 - b)))
        out.add((dr * HALF + (HALF - 1 - a), dc * HALF + b))
    return out


def _image_in(cell: tuple[int, int], target: tuple[int, int]) -> tuple[int, int]:
    """Where a hole lands inside quadrant ``target`` after unfolding."""

    r, c = cell
    qr, qc = _quadrant(cell)
    tr, tc = target
    if tr != qr:
        r = GRID - 1 - r
    if tc != qc:
        c = GRID - 1 - c
    return (r, c)


def _sample_once(rng: random.Random) -> dict | None:
    source = (rng.randrange(2), rng.randrange(2))
    others = [(a, b) for a in range(2) for b in range(2) if (a, b) != source]

    n_holes = rng.randint(MIN_HOLES, MAX_HOLES)
    quarter_cells = [
        (source[0] * HALF + a, source[1] * HALF + b)
        for a in range(HALF)
        for b in range(HALF)
    ]
    holes = sorted(rng.sample(quarter_cells, n_holes))
    unfolded = unfolded_cells(holes)

    # the answer dot: the image of a drawn hole in one of the other quarters
    target = others[rng.randrange(len(others))]
    answer_cell = _image_in(holes[rng.randrange(len(holes))], target)

    # a solver who translates the punched quarter, or who applies only one of
    # the two mirrors, must never land on the answer by accident.
    wrong_policy = translated_cells(holes) | half_mirrored_cells(holes)
    if answer_cell in wrong_policy:
        return None

    used = {answer_cell}

    # decoy 1 - translated copy: same offset inside its quarter as some hole.
    hole = holes[rng.randrange(len(holes))]
    tq = others[rng.randrange(len(others))]
    a, b = hole[0] % HALF, hole[1] % HALF
    translate_cell = (tq[0] * HALF + a, tq[1] * HALF + b)
    if translate_cell in unfolded or translate_cell in used:
        return None
    used.add(translate_cell)

    # decoy 2 - only one of the two mirrors applied, in the diagonal quarter.
    diag = (1 - source[0], 1 - source[1])
    hole = holes[rng.randrange(len(holes))]
    a, b = hole[0] % HALF, hole[1] % HALF
    if rng.random() < 0.5:
        half_cell = (diag[0] * HALF + a, diag[1] * HALF + (HALF - 1 - b))
    else:
        half_cell = (diag[0] * HALF + (HALF - 1 - a), diag[1] * HALF + b)
    if half_cell in unfolded or half_cell in used:
        return None
    used.add(half_cell)

    # decoy 3 - a plain empty cell somewhere outside the punched quarter.
    pool = [
        (r, c)
        for r in range(GRID)
        for c in range(GRID)
        if _quadrant((r, c)) != source and (r, c) not in unfolded and (r, c) not in used
    ]
    if not pool:
        return None
    plain_cell = pool[rng.randrange(len(pool))]
    used.add(plain_cell)

    cells = [answer_cell, translate_cell, half_cell, plain_cell]
    if len(set(cells)) != 4:
        return None
    if any(_quadrant(cell) == source for cell in cells):
        return None
    if sum(1 for cell in cells if cell in unfolded) != 1:
        return None
    # the four dots must not all sit in one quarter
    if len({_quadrant(cell) for cell in cells}) < 2:
        return None

    colors = list(COLOR_NAMES)
    rng.shuffle(colors)
    dots = [
        {"cell": [int(cell[0]), int(cell[1])], "color": color}
        for cell, color in zip(cells, colors)
    ]
    rng.shuffle(dots)

    return {
        "source": [int(source[0]), int(source[1])],
        "holes": [[int(r), int(c)] for r, c in holes],
        "dots": dots,
        "jitter": [rng.randint(-JITTER, JITTER), rng.randint(-JITTER, JITTER)],
    }


def sample_scene(seed: int) -> dict:
    for attempt in range(4096):
        rng = random.Random((int(seed) * 1000003 + attempt * 7919) & 0xFFFFFFFF)
        scene = _sample_once(rng)
        if scene is None:
            continue
        if is_quarantined(scene):
            continue
        return scene
    raise RuntimeError(f"no admissible scene for seed {seed}")


# --------------------------------------------------------------------------
# gold, margin, quarantine
# --------------------------------------------------------------------------


def analytic_gold(scene: dict) -> str:
    holes = [tuple(cell) for cell in scene["holes"]]
    unfolded = unfolded_cells(holes)
    hits = [dot["color"] for dot in scene["dots"] if tuple(dot["cell"]) in unfolded]
    if len(hits) != 1:
        raise ValueError("scene does not carry exactly one probed hole")
    return hits[0]


def _origin(scene: dict) -> tuple[int, int]:
    dx, dy = scene["jitter"]
    return (ORIGIN + int(dx), ORIGIN + int(dy))


def margin(scene: dict) -> float:
    holes = [tuple(cell) for cell in scene["holes"]]
    unfolded = unfolded_cells(holes)

    inset = (PITCH - MARK) / 2.0
    mark_clear = min(inset - CREASE_W / 2.0, inset - BORDER_W)
    ox, oy = _origin(scene)
    edge_clear = float(min(ox, oy, WIDTH - ox - SHEET, HEIGHT - oy - SHEET))

    decisive = float(GRID) * PITCH
    for dot in scene["dots"]:
        cell = tuple(dot["cell"])
        if cell in unfolded:
            continue
        best = min(
            max(abs(cell[0] - r), abs(cell[1] - c)) for r, c in unfolded
        )
        decisive = min(decisive, float(best) * PITCH)

    return float(min(mark_clear, edge_clear, decisive))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """``not_applicable``: rendering is injective over sampled scenes.

    Every sampled field is drawn directly - the punched cells, the four dot
    cells, their colors and the sheet offset - so no edit of the scene can
    leave the raster untouched and there is no alias to test.
    """

    return []


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------


def _cell_box(ox: int, oy: int, r: int, c: int) -> tuple[int, int, int, int]:
    x0 = ox + c * PITCH
    y0 = oy + r * PITCH
    return (x0, y0, x0 + PITCH, y0 + PITCH)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), WHITE)
    draw = ImageDraw.Draw(image)
    ox, oy = _origin(scene)

    # interior rule lines
    for i in range(1, GRID):
        x = ox + i * PITCH
        draw.rectangle([x - GRID_W // 2, oy, x - GRID_W // 2 + GRID_W - 1, oy + SHEET - 1], fill=RULE)
        y = oy + i * PITCH
        draw.rectangle([ox, y - GRID_W // 2, ox + SHEET - 1, y - GRID_W // 2 + GRID_W - 1], fill=RULE)

    # dashed creases on the two midlines
    xm = ox + HALF * PITCH
    ym = oy + HALF * PITCH
    half_w = CREASE_W // 2
    pos = oy
    while pos < oy + SHEET:
        end = min(pos + DASH_ON, oy + SHEET)
        draw.rectangle([xm - half_w, pos, xm + half_w - 1, end - 1], fill=BLACK)
        pos = end + DASH_OFF
    pos = ox
    while pos < ox + SHEET:
        end = min(pos + DASH_ON, ox + SHEET)
        draw.rectangle([pos, ym - half_w, end - 1, ym + half_w - 1], fill=BLACK)
        pos = end + DASH_OFF

    # punched holes
    inset = (PITCH - MARK) // 2
    for r, c in scene["holes"]:
        x0, y0, _, _ = _cell_box(ox, oy, r, c)
        draw.rectangle([x0 + inset, y0 + inset, x0 + inset + MARK - 1, y0 + inset + MARK - 1], fill=BLACK)

    # probe dots
    for dot in scene["dots"]:
        r, c = dot["cell"]
        x0, y0, _, _ = _cell_box(ox, oy, r, c)
        draw.ellipse(
            [x0 + inset, y0 + inset, x0 + inset + MARK - 1, y0 + inset + MARK - 1],
            fill=COLORS[dot["color"]],
        )

    # sheet border, drawn inward last so it stays the outermost black ink
    for i in range(BORDER_W):
        draw.rectangle([ox + i, oy + i, ox + SHEET - 1 - i, oy + SHEET - 1 - i], outline=BLACK)

    return image
