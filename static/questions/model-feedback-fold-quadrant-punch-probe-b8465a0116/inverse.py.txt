"""Image-only inverse arm for the fold-and-punch world.

Receives a rendered PNG and nothing else.  It locates the sheet from its black
border, reads the 8 by 8 lattice of cells, classifies each cell as a punched
hole, a colored probe dot or empty, generates the unfolded hole set as the
orbit of the punched cells under the two midline mirrors, and reports the color
of the single dot that lands on that set.

It imports no renderer, prompt, generator, verifier or analytic-gold code and
never sees latent scene data.
"""

from __future__ import annotations

from PIL import Image

_GRID = 8
_PROBE = 11  # half-width of the square sampled at each cell centre

_PALETTE = {
    "blue": (30, 90, 220),
    "green": (20, 150, 60),
    "red": (220, 40, 40),
    "yellow": (245, 200, 30),
}
_FALLBACK = "blue"


def _is_black(pixel) -> bool:
    return max(pixel[:3]) <= 60


def _sheet_box(pixels, width: int, height: int):
    """Bounding box of the black ink; the sheet border is its outermost part."""

    x0, y0, x1, y1 = width, height, -1, -1
    for y in range(0, height, 2):
        row = None
        for x in range(0, width, 2):
            if _is_black(pixels[x, y]):
                if row is None:
                    row = x
                    if x < x0:
                        x0 = x
                if x > x1:
                    x1 = x
        if row is not None:
            if y < y0:
                y0 = y
            if y > y1:
                y1 = y
    if x1 < x0 or y1 < y0:
        return None
    return (x0, y0, x1, y1)


def _classify(pixels, cx: int, cy: int, width: int, height: int) -> str | None:
    counts: dict[str, int] = {}
    black = 0
    total = 0
    for dy in range(-_PROBE, _PROBE + 1, 2):
        for dx in range(-_PROBE, _PROBE + 1, 2):
            x = cx + dx
            y = cy + dy
            if not (0 <= x < width and 0 <= y < height):
                continue
            total += 1
            pixel = pixels[x, y][:3]
            if _is_black(pixel):
                black += 1
                continue
            for name, rgb in _PALETTE.items():
                if abs(pixel[0] - rgb[0]) <= 24 and abs(pixel[1] - rgb[1]) <= 24 and abs(pixel[2] - rgb[2]) <= 24:
                    counts[name] = counts.get(name, 0) + 1
                    break
    if not total:
        return None
    if black * 2 > total:
        return "hole"
    if counts:
        name, hits = max(counts.items(), key=lambda item: item[1])
        if hits * 2 > total:
            return name
    return None


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    width, height = rgb.size
    pixels = rgb.load()

    box = _sheet_box(pixels, width, height)
    if box is None:
        return _FALLBACK
    x0, y0, x1, y1 = box
    span_x = (x1 - x0 + 1) / _GRID
    span_y = (y1 - y0 + 1) / _GRID
    if span_x < 4 * _PROBE or span_y < 4 * _PROBE:
        return _FALLBACK

    holes: set[tuple[int, int]] = set()
    dots: list[tuple[tuple[int, int], str]] = []
    for r in range(_GRID):
        for c in range(_GRID):
            cx = int(round(x0 + (c + 0.5) * span_x))
            cy = int(round(y0 + (r + 0.5) * span_y))
            kind = _classify(pixels, cx, cy, width, height)
            if kind == "hole":
                holes.add((r, c))
            elif kind is not None:
                dots.append(((r, c), kind))

    if not holes or not dots:
        return _FALLBACK

    unfolded: set[tuple[int, int]] = set()
    for r, c in holes:
        unfolded.add((r, c))
        unfolded.add((r, _GRID - 1 - c))
        unfolded.add((_GRID - 1 - r, c))
        unfolded.add((_GRID - 1 - r, _GRID - 1 - c))

    hits = [name for cell, name in dots if cell in unfolded]
    if len(hits) != 1:
        return _FALLBACK
    return hits[0]
