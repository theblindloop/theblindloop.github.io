"""Deterministic raster renderer for the visible two-hand clock world."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "clock_face_time_reading-renderer-0.1.0"
WIDTH = 512
HEIGHT = 512
SUPERSAMPLE = 4

BACKGROUND_PALETTE = (
    (250, 247, 239),
    (239, 248, 250),
    (247, 241, 250),
    (242, 249, 240),
    (250, 242, 236),
)
FACE_PALETTE = (
    (231, 239, 239),
    (242, 233, 215),
    (231, 235, 247),
    (242, 226, 237),
    (226, 241, 229),
)
NEUTRAL_PALETTE = (
    (36, 45, 55),
    (52, 48, 61),
    (44, 61, 52),
    (64, 48, 42),
)
HAND_PALETTE = (
    (194, 48, 52),
    (31, 91, 190),
    (22, 143, 98),
    (151, 58, 176),
    (222, 103, 28),
    (23, 145, 157),
    (177, 39, 117),
    (75, 78, 186),
)


def _circular_distance(first: float, second: float) -> float:
    return abs((float(first) - float(second) + 180.0) % 360.0 - 180.0)


def _nominal_hour_angle(hour_index: int, minute_quarter: int) -> float:
    # Clock angle is degrees clockwise from 12 o'clock.  The short hand moves
    # one quarter of a 30-degree hour step for each quarter hour.
    return (float(hour_index) * 30.0 + float(minute_quarter) * 7.5) % 360.0


def _make_valid_times() -> tuple[tuple[int, int], ...]:
    # Keep the two visible strokes separated enough for a raster-only inverse,
    # while retaining all ordinary clock directions that are not near overlap.
    values = []
    for hour_index in range(12):
        for minute_quarter in range(4):
            minute_angle = float(minute_quarter) * 90.0
            if _circular_distance(
                _nominal_hour_angle(hour_index, minute_quarter), minute_angle
            ) >= 30.0:
                values.append((hour_index, minute_quarter))
    return tuple(values)


VALID_TIME_PAIRS = _make_valid_times()
VALID_TIME_STRINGS = tuple(
    f"{hour_index + 1}:{minute_quarter * 15:02d}"
    for hour_index, minute_quarter in VALID_TIME_PAIRS
)


def _rgb(value: Any) -> tuple[int, int, int]:
    return tuple(int(channel) for channel in value)  # type: ignore[return-value]


def _box(cx: float, cy: float, radius: float, scale: int) -> tuple[float, ...]:
    return (
        (cx - radius) * scale,
        (cy - radius) * scale,
        (cx + radius) * scale,
        (cy + radius) * scale,
    )


def _point(
    cx: float, cy: float, distance: float, clock_angle_deg: float, scale: int
) -> tuple[float, float]:
    angle = math.radians(clock_angle_deg)
    return (
        (cx + distance * math.sin(angle)) * scale,
        (cy - distance * math.cos(angle)) * scale,
    )


def _hand_angles(scene: dict[str, Any]) -> tuple[float, float]:
    hour_angle = (
        _nominal_hour_angle(int(scene["hour_index"]), int(scene["minute_quarter"]))
        + float(scene["hour_jitter_deg"])
    ) % 360.0
    minute_angle = (
        float(scene["minute_quarter"]) * 90.0 + float(scene["minute_jitter_deg"])
    ) % 360.0
    return hour_angle, minute_angle


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one balanced, JSON-friendly clock scene from an integer seed."""

    seed = int(seed)
    hour_index, minute_quarter = VALID_TIME_PAIRS[seed % len(VALID_TIME_PAIRS)]
    rng = random.Random(seed * 1_000_003 + 43_771)
    if seed % 19 == 0:
        hour_jitter = rng.choice((-10.5, 10.5))
    else:
        hour_jitter = rng.uniform(-4.0, 4.0)
    minute_jitter = rng.uniform(-3.0, 3.0)
    radius = round(rng.uniform(118.0, 148.0), 4)
    clearance = radius + 34.0
    background = rng.choice(BACKGROUND_PALETTE)
    face = rng.choice(FACE_PALETTE)
    hand_indices = rng.sample(range(len(HAND_PALETTE)), 2)
    while float(
        math.dist(HAND_PALETTE[hand_indices[0]], HAND_PALETTE[hand_indices[1]])
    ) < 100.0:
        hand_indices = rng.sample(range(len(HAND_PALETTE)), 2)
    return {
        "hour_index": hour_index,
        "minute_quarter": minute_quarter,
        "center_x": round(rng.uniform(clearance, WIDTH - clearance), 4),
        "center_y": round(rng.uniform(clearance, HEIGHT - clearance), 4),
        "radius": radius,
        "hour_jitter_deg": round(hour_jitter, 4),
        "minute_jitter_deg": round(minute_jitter, 4),
        "hour_length_ratio": round(rng.uniform(0.52, 0.60), 5),
        "minute_length_ratio": round(rng.uniform(0.77, 0.87), 5),
        "hour_width": rng.choice((8, 9, 10)),
        "minute_width": rng.choice((5, 6, 7)),
        "tick_width": rng.choice((3, 4, 5)),
        "face_color": list(face),
        "background_color": list(background),
        "outline_color": list(rng.choice(NEUTRAL_PALETTE)),
        "tick_color": list(rng.choice(NEUTRAL_PALETTE)),
        "hour_color": list(HAND_PALETTE[hand_indices[0]]),
        "minute_color": list(HAND_PALETTE[hand_indices[1]]),
        "hub_color": list(rng.choice(NEUTRAL_PALETTE)),
        "outline_width": rng.choice((3, 4, 5)),
        "tick_count": 12,
        "width": WIDTH,
        "height": HEIGHT,
        "antialias": True,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the clock face and its two visible, differently colored hands."""

    width = int(scene["width"])
    height = int(scene["height"])
    supersample = SUPERSAMPLE if bool(scene["antialias"]) else 1
    image = Image.new(
        "RGB", (width * supersample, height * supersample), _rgb(scene["background_color"])
    )
    draw = ImageDraw.Draw(image)
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    radius = float(scene["radius"])
    face = _rgb(scene["face_color"])
    outline = _rgb(scene["outline_color"])
    tick = _rgb(scene["tick_color"])
    draw.ellipse(
        _box(cx, cy, radius, supersample),
        fill=face,
        outline=outline,
        width=max(1, int(scene["outline_width"])) * supersample,
    )

    tick_count = int(scene["tick_count"])
    for index in range(tick_count):
        angle = index * 360.0 / tick_count
        outer = _point(cx, cy, radius - 9.0, angle, supersample)
        inner_gap = 18.0 if index % 3 == 0 else 22.0
        inner = _point(cx, cy, radius - inner_gap, angle, supersample)
        draw.line(
            [tuple(round(value) for value in inner), tuple(round(value) for value in outer)],
            fill=tick,
            width=max(1, int(scene["tick_width"])) * supersample,
        )

    hour_angle, minute_angle = _hand_angles(scene)
    hour_end = _point(
        cx,
        cy,
        radius * float(scene["hour_length_ratio"]),
        hour_angle,
        supersample,
    )
    minute_end = _point(
        cx,
        cy,
        radius * float(scene["minute_length_ratio"]),
        minute_angle,
        supersample,
    )
    centre = (round(cx * supersample), round(cy * supersample))
    draw.line(
        [centre, tuple(round(value) for value in hour_end)],
        fill=_rgb(scene["hour_color"]),
        width=max(1, int(scene["hour_width"])) * supersample,
    )
    draw.line(
        [centre, tuple(round(value) for value in minute_end)],
        fill=_rgb(scene["minute_color"]),
        width=max(1, int(scene["minute_width"])) * supersample,
    )
    hub_radius = 8.0 * supersample
    draw.ellipse(
        (centre[0] - hub_radius, centre[1] - hub_radius,
         centre[0] + hub_radius, centre[1] + hub_radius),
        fill=_rgb(scene["hub_color"]),
    )

    if supersample > 1:
        image = image.resize((width, height), Image.Resampling.LANCZOS)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the time represented by the two label-bearing hand fields."""

    hour = int(scene["hour_index"]) + 1
    minute = int(scene["minute_quarter"]) * 15
    return f"{hour}:{minute:02d}"


def margin(scene: dict[str, Any]) -> float:
    """Return the smallest visible angular clearance from a decision boundary."""

    hour_clearance = 15.0 - abs(float(scene["hour_jitter_deg"]))
    minute_clearance = 45.0 - abs(float(scene["minute_jitter_deg"]))
    hour_angle, minute_angle = _hand_angles(scene)
    separation_clearance = _circular_distance(hour_angle, minute_angle) - 12.0
    return round(min(hour_clearance, minute_clearance, separation_clearance), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Exclude near-boundary hand poses from headline claims."""

    return margin(scene) < 6.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No latent aliases are declared; every label input is visibly rendered."""

    return []
