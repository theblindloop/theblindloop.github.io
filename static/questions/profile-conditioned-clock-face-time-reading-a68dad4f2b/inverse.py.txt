"""Independent image-only inverse for the two-hand clock face.

The inverse locates the visible face from the raster, extracts the two strongly
chromatic hand strokes, identifies the long stroke by measured radial extent,
and quantizes their fitted directions.  It has no access to latent scene data.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "clock_face_time_reading-oracle-0.1.0"


def _face_center(image: Image.Image) -> tuple[float, float]:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    corner = array[0, 0].astype(np.int16)
    difference = np.max(np.abs(array - corner), axis=2)
    visible = difference > 12
    ys, xs = np.nonzero(visible)
    if len(xs) < 1000:
        raise ValueError("clock face is not visibly bounded")
    return (float(xs.min() + xs.max()) / 2.0, float(ys.min() + ys.max()) / 2.0)


def _hand_candidates(image: Image.Image, centre: tuple[float, float]) -> list[tuple[float, float]]:
    array = np.asarray(image.convert("RGB"), dtype=np.float64)
    chroma = array.max(axis=2) - array.min(axis=2)
    saturated = (chroma >= 55.0) & (array.max(axis=2) >= 100.0)
    pixels = array[saturated]
    if len(pixels) < 250:
        raise ValueError("two chromatic hand strokes are not measurable")

    # Quantized dominant colors are used only to split the two visible strokes;
    # no renderer palette or answer-specific color is assumed.
    quantized = (pixels.astype(np.int32) // 16) * 16 + 8
    unique, counts = np.unique(quantized, axis=0, return_counts=True)
    order = np.argsort(-counts)
    prototypes: list[np.ndarray] = []
    for index in order:
        prototype = unique[int(index)].astype(np.float64)
        if all(float(np.linalg.norm(prototype - other)) > 62.0 for other in prototypes):
            prototypes.append(prototype)
        if len(prototypes) == 2:
            break
    if len(prototypes) != 2:
        raise ValueError("could not separate two visible hand colors")

    cx, cy = centre
    candidates: list[tuple[float, float]] = []
    for prototype in prototypes:
        distances = np.linalg.norm(array - prototype.reshape(1, 1, 3), axis=2)
        mask = saturated & (distances <= 66.0)
        ys, xs = np.nonzero(mask)
        if len(xs) < 80:
            continue
        radial = np.hypot(xs.astype(float) - cx, ys.astype(float) - cy)
        usable = radial >= 15.0
        if int(np.count_nonzero(usable)) < 40:
            continue
        vectors = np.column_stack(
            (xs[usable].astype(float) - cx, ys[usable].astype(float) - cy)
        )
        vector = np.mean(vectors, axis=0)
        angle = math.degrees(math.atan2(float(vector[0]), float(-vector[1]))) % 360.0
        extent = float(np.percentile(radial, 96.0))
        candidates.append((extent, angle))
    if len(candidates) < 2:
        raise ValueError("two hand directions were not recoverable")
    return sorted(candidates, key=lambda item: item[0], reverse=True)[:2]


def _angular_distance(first: float, second: float) -> float:
    return abs((first - second + 180.0) % 360.0 - 180.0)


def _estimate(image: Image.Image) -> tuple[str, float]:
    centre = _face_center(image)
    hands = _hand_candidates(image, centre)
    minute_extent, minute_angle = hands[0]
    hour_extent, hour_angle = hands[1]
    del minute_extent, hour_extent
    minute_index = int(round(minute_angle / 90.0)) % 4
    minute_target = minute_index * 90.0
    adjusted_hour = (hour_angle - minute_index * 7.5) % 360.0
    hour_index = int(round(adjusted_hour / 30.0)) % 12
    minute_error = _angular_distance(minute_angle, minute_target)
    hour_error = _angular_distance(adjusted_hour, hour_index * 30.0)
    confidence = round(min(45.0 - minute_error, 15.0 - hour_error), 6)
    return f"{hour_index + 1}:{minute_index * 15:02d}", confidence


def decision_from_image(image: Image.Image) -> str:
    """Return the time recovered from the supplied raster only."""

    return _estimate(image)[0]


def decision_margin(image: Image.Image) -> float:
    """Return the inverse's nearest-boundary angular clearance."""

    return _estimate(image)[1]
