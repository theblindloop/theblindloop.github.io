"""Independent pixel-only inverse arm for the state-fifo-queue world.

Receives only the final PNG. From the raster alone it:
  1. labels pixels against the fixed palette;
  2. detects the five tall panel boxes from their full-height border columns;
  3. treats the leftmost panel as the START queue and reads its disc colors
     left to right (front at left);
  4. for each operator panel in left-to-right order, decides ENQUEUE (a colored
     disc is present -> enqueue that color) vs DEQUEUE (no colored disc -> pop
     the front);
  5. replays the operators from the initial queue and returns the final front
     color, or 'empty' if the queue is exhausted.

No latent scene, renderer, prompt, or gold code is accessed.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

BG = (255, 255, 255)
BORDER = (95, 95, 95)
ARROW = (60, 60, 60)
RING = (210, 0, 210)
TOKEN_COLORS = [
    (210, 40, 40),
    (40, 170, 40),
    (50, 80, 220),
    (240, 160, 30),
]
COLOR_LABELS = ["red", "green", "blue", "orange"]

PALETTE = [BG, BORDER, ARROW, RING, *TOKEN_COLORS]
_BG, _BRD, _ARROW, _RING, _TC0, _TC1, _TC2, _TC3 = range(8)

N_PANEL = 5
PANEL_W = 210
PANEL_H = 150
PANEL_GAP = 55
_NOP = 4


class UnreadableImageError(ValueError):
    """The raster does not expose the expected panel structure."""


def _label(arr: np.ndarray) -> np.ndarray:
    dists = np.stack(
        [np.linalg.norm(arr - np.array(c, dtype=np.float32), axis=-1) for c in PALETTE],
        axis=-1,
    )
    return dists.argmin(axis=-1)


def _cluster(sorted_vals: list[int], tol: int) -> list[int]:
    if not sorted_vals:
        return []
    out = [[sorted_vals[0]]]
    for v in sorted_vals[1:]:
        if v - out[-1][-1] > tol:
            out.append([v])
        else:
            out[-1].append(v)
    return [r[len(r) // 2] for r in out]


def _panel_x_edges(labels: np.ndarray) -> list[tuple[int, int]]:
    border_ys, border_xs = np.nonzero(labels == _BRD)
    if len(border_xs) == 0:
        raise UnreadableImageError("no panel borders found")
    w = labels.shape[1]
    counts = np.bincount(border_xs, minlength=w)
    # Vertical panel edges contain a full-height run of border pixels (roughly
    # PANEL_H rows); horizontal top/bottom edges contribute only a few rows per
    # column. Thresholding on count separates the two cleanly.
    candidates = [int(x) for x in range(w) if counts[x] > 40]
    edges = _cluster(sorted(set(candidates)), 8)
    if len(edges) < 2 * N_PANEL:
        raise UnreadableImageError(
            f"expected {2 * N_PANEL} panel-edge columns, found {len(edges)}"
        )
    return [(edges[2 * k], edges[2 * k + 1]) for k in range(N_PANEL)]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label(arr)
    h, w = labels.shape

    panels = _panel_x_edges(labels)

    queue = []
    px0, px1 = panels[0]
    token_idx = [_TC0, _TC1, _TC2, _TC3]
    token_x = []
    for idx, color_label in enumerate(COLOR_LABELS):
        ys, xs = np.nonzero(labels[:, px0:px1] == token_idx[idx])
        ax = (px0 + xs).tolist()
        if ax:
            token_x.append((float(np.mean(ax)), color_label))
    for _, color_label in sorted(token_x):
        queue.append(color_label)
    if not queue:
        raise UnreadableImageError("START queue has no discs")

    for k in range(1, N_PANEL):
        p0, p1 = panels[k]
        found = None
        for idx in range(len(TOKEN_COLORS)):
            if np.count_nonzero(labels[:, p0:p1] == token_idx[idx]) > 0:
                found = COLOR_LABELS[idx]
                break
        if found is not None:
            queue.append(found)
        else:
            if not queue:
                raise UnreadableImageError("dequeue on empty queue in operator panel")
            queue.pop(0)

    return queue[0] if queue else "empty"
