"""FIFO queue front-register world: read the front color after ordered ops.

The image is a horizontal timeline of five panels read left to right. The
leftmost panel is the START queue: a row of colored discs (red/green/blue/
orange), read left to right, with the FRONT of the queue at the very left. The
four panels to the right are OPERATOR panels read left to right in order:

  * a panel showing a colored disc      -> ENQUEUE that color (append to back)
  * a panel showing only a dark arrow   -> DEQUEUE (remove the front element)

The queue is never drawn again; the decision is the color at the FRONT of the
final queue after replaying the operators in order, or 'empty' if no disc
remains.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

N_COLOR = 4
N_PANEL = 5
N_OP = 4
MAX_LEN = 5

TOKEN_COLORS = [
    (210, 40, 40),
    (40, 170, 40),
    (50, 80, 220),
    (240, 160, 30),
]
COLOR_LABELS = ["red", "green", "blue", "orange"]
BORDER = (95, 95, 95)
BG = (255, 255, 255)
ARROW = (60, 60, 60)
RING = (210, 0, 210)

LEFT_M = 50
TOP_M = 80
BOTTOM_M = 60
PANEL_W = 210
PANEL_H = 150
PANEL_GAP = 55
SLOT_W = 30
SLOT_GAP = 14
TOKEN_R = 12
OP_DISC_R = 17

WIDTH = LEFT_M + N_PANEL * PANEL_W + (N_PANEL - 1) * PANEL_GAP + LEFT_M
HEIGHT = TOP_M + PANEL_H + BOTTOM_M


def _panel_x0(k: int) -> int:
    return LEFT_M + k * (PANEL_W + PANEL_GAP)


def _slot_x(i: int) -> int:
    return _panel_x0(0) + 14 + SLOT_W // 2 + i * (SLOT_W + SLOT_GAP)


def _color_index(label: str) -> int:
    return COLOR_LABELS.index(label)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    palette = list(range(N_COLOR))
    rng.shuffle(palette)
    length = rng.randint(1, 3)
    initial = [COLOR_LABELS[c] for c in palette[:length]]
    queue = list(initial)
    ops = []
    for _ in range(N_OP):
        if len(queue) == 0:
            ops.append({"kind": "enqueue", "color": rng.choice(COLOR_LABELS)})
        elif len(queue) >= MAX_LEN:
            ops.append({"kind": "dequeue"})
        else:
            ops.append(
                {"kind": "dequeue", "color": None}
                if rng.random() < 0.45
                else {"kind": "enqueue", "color": rng.choice(COLOR_LABELS)}
            )
        if ops[-1]["kind"] == "enqueue":
            queue.append(ops[-1]["color"])
        else:
            queue.pop(0)
    return {"initial": initial, "ops": ops}


def _draw_arrow(draw, cx, cy, direction, color, size=16, width=5):
    if direction == "right":
        draw.line([cx, cy, cx + size, cy], fill=color, width=width)
        draw.polygon(
            [(cx + size, cy), (cx + size - 8, cy - 7), (cx + size - 8, cy + 7)],
            fill=color,
        )
    else:
        draw.line([cx, cy, cx - size, cy], fill=color, width=width)
        draw.polygon(
            [(cx - size, cy), (cx - size + 8, cy - 7), (cx - size + 8, cy + 7)],
            fill=color,
        )


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)
    initial = scene["initial"]
    ops = scene["ops"]

    panel_mid_y = TOP_M + PANEL_H // 2

    for k in range(N_PANEL):
        x0 = _panel_x0(k)
        draw.rectangle([x0, TOP_M, x0 + PANEL_W, TOP_M + PANEL_H], outline=BORDER, width=3)

    x0 = _panel_x0(0)
    for i in range(MAX_LEN):
        sx = _slot_x(i)
        draw.rectangle(
            [sx - SLOT_W // 2, panel_mid_y - SLOT_W // 2, sx + SLOT_W // 2, panel_mid_y + SLOT_W // 2],
            outline=BORDER,
            width=2,
        )
    for i, label in enumerate(initial):
        sx = _slot_x(i)
        color = TOKEN_COLORS[_color_index(label)]
        draw.ellipse(
            [sx - TOKEN_R, panel_mid_y - TOKEN_R, sx + TOKEN_R, panel_mid_y + TOKEN_R],
            fill=color,
        )
    draw.rectangle(
        [x0 - 6, TOP_M - 6, x0 + PANEL_W + 6, TOP_M + PANEL_H + 6],
        outline=RING,
        width=4,
    )
    draw.text((x0, TOP_M - 40), "START", fill=RING)

    for k, op in enumerate(ops, start=1):
        px0 = _panel_x0(k)
        cx = px0 + PANEL_W // 2
        cy = panel_mid_y
        if op["kind"] == "enqueue":
            color = TOKEN_COLORS[_color_index(op["color"])]
            disc_cx = px0 + int(PANEL_W * 0.35)
            draw.ellipse(
                [disc_cx - OP_DISC_R, cy - OP_DISC_R, disc_cx + OP_DISC_R, cy + OP_DISC_R],
                fill=color,
            )
            _draw_arrow(draw, disc_cx + OP_DISC_R + 10, cy, "right", ARROW)
            draw.text((disc_cx - 14, TOP_M + 6), "ENQ", fill=ARROW)
        else:
            arrow_cx = px0 + int(PANEL_W * 0.35)
            _draw_arrow(draw, arrow_cx, cy, "left", ARROW, size=22)
            draw.text((arrow_cx - 16, TOP_M + 6), "DEQ", fill=ARROW)
    return img


def analytic_gold(scene: dict) -> str:
    queue = list(scene["initial"])
    for op in scene["ops"]:
        if op["kind"] == "enqueue":
            queue.append(op["color"])
        else:
            queue.pop(0)
    return queue[0] if queue else "empty"


def margin(scene: dict) -> float:
    return 10.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    """No nontrivial render-invariant latent symmetry: ``initial`` fixes the
    drawn disc colors and their left-to-right order, and ``ops`` fixes each
    operator glyph. Any change reorders discs or swaps an enqueue/dequeue and
    changes the raster as well as the gold."""
    return []
