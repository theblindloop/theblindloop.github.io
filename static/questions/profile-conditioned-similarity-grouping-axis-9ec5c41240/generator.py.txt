"""Latent scene sampling, rendering, analytic gold and margin for
`similarity_grouping_axis`.

A square 8x8 lattice of equally spaced marks is painted. Each mark is either a
solid dot or a hollow ring of the same ink area. Same-kind marks agree along one
axis of the lattice more than along the other, but the agreement is noisy, so no
local window settles it: the answer is the axis whose average purity is larger.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "similarity_grouping_axis-renderer-0.1.0"

N = 8
SUPERSAMPLE = 2
QUARANTINE_MARGIN = 10.0


# --------------------------------------------------------------------------
# scene sampling
# --------------------------------------------------------------------------
def _scores(grid: list[list[int]]) -> tuple[float, float]:
    """Row score and column score of a realised grid of 0/1 mark kinds."""
    size = len(grid)
    row_purities = []
    for row in grid:
        ones = sum(row)
        row_purities.append(abs(2 * ones - size) / size)
    col_purities = []
    for index in range(size):
        ones = sum(grid[r][index] for r in range(size))
        col_purities.append(abs(2 * ones - size) / size)
    return sum(row_purities) / size, sum(col_purities) / size


def _draw_grid(rng: random.Random, axis: str, flip: float) -> list[list[int]]:
    """Stripe base labels along `axis`, then flip individual cells."""
    base = [rng.randint(0, 1) for _ in range(N)]
    grid = []
    for r in range(N):
        row = []
        for c in range(N):
            label = base[r] if axis == "rows" else base[c]
            if rng.random() < flip:
                label = 1 - label
            row.append(label)
        grid.append(row)
    return grid


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    axis_intent = "rows" if seed % 2 == 0 else "columns"
    # every fourth scene deliberately targets the near-boundary band
    near_boundary = seed % 4 == 3
    low, high = (3.0, QUARANTINE_MARGIN) if near_boundary else (14.0, 60.0)

    grid = None
    for attempt in range(4000):
        flip = 0.20 + 0.16 * ((attempt % 7) / 6.0)
        candidate = _draw_grid(rng, axis_intent, flip)
        row_score, col_score = _scores(candidate)
        winner = "rows" if row_score > col_score else "columns"
        gap = abs(row_score - col_score) * 100.0
        if winner == axis_intent and low <= gap < high:
            grid = candidate
            break
    if grid is None:  # pragma: no cover - construction always converges
        raise RuntimeError(f"could not sample a scene for seed {seed}")

    pitch = rng.randint(52, 58)
    radius = rng.randint(10, 12)
    inner = max(7, round(0.72 * radius))
    # equal ink: pi*R^2 - pi*inner^2 == pi*radius^2
    outer = round((radius * radius + inner * inner) ** 0.5)
    width = height = 520
    span = pitch * (N - 1)
    origin_x = (width - span) // 2 + rng.randint(-6, 6)
    origin_y = (height - span) // 2 + rng.randint(-6, 6)
    shade = rng.randint(216, 242)
    tint = rng.randint(-8, 8)
    dark = rng.randint(26, 62)
    return {
        "width": width,
        "height": height,
        "bg_color": [shade, shade, max(200, min(250, shade + tint))],
        "ink_color": [dark, dark, min(90, dark + max(0, tint))],
        "pitch": pitch,
        "origin_x": origin_x,
        "origin_y": origin_y,
        "radius": radius,
        "inner": inner,
        "outer": outer,
        "grid": grid,
    }


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def render(scene: dict) -> Image.Image:
    scale = SUPERSAMPLE
    width = int(scene["width"]) * scale
    height = int(scene["height"]) * scale
    bg = tuple(int(v) for v in scene["bg_color"])
    ink = tuple(int(v) for v in scene["ink_color"])
    image = Image.new("RGB", (width, height), bg)
    draw = ImageDraw.Draw(image)
    pitch = int(scene["pitch"]) * scale
    ox = int(scene["origin_x"]) * scale
    oy = int(scene["origin_y"]) * scale
    radius = int(scene["radius"]) * scale
    inner = int(scene["inner"]) * scale
    outer = int(scene["outer"]) * scale
    grid = scene["grid"]
    for r, row in enumerate(grid):
        for c, kind in enumerate(row):
            cx = ox + c * pitch
            cy = oy + r * pitch
            if int(kind) == 1:
                draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=ink)
            else:
                draw.ellipse((cx - outer, cy - outer, cx + outer, cy + outer), fill=ink)
                draw.ellipse((cx - inner, cy - inner, cx + inner, cy + inner), fill=bg)
    return image.resize(
        (int(scene["width"]), int(scene["height"])), Image.Resampling.BOX
    )


# --------------------------------------------------------------------------
# analytic consequences (grid only)
# --------------------------------------------------------------------------
def analytic_gold(scene: dict) -> str:
    row_score, col_score = _scores(scene["grid"])
    return "rows" if row_score > col_score else "columns"


def margin(scene: dict) -> float:
    row_score, col_score = _scores(scene["grid"])
    return round(abs(row_score - col_score) * 100.0, 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases: every scene field is painted and the label input is
    the painted grid itself, so no pixel-identical relabelling exists."""
    return []
