"""Independent pixel-only inverse for the half-turn mosaic search.

No renderer state or coordinates are available here.  The lattice is located
from color-bearing row/column runs, every one of its 144 cell centers is
classified, and all four candidate quadrants are scored before a unique answer
is returned.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

COLOR_PROTOTYPES = np.asarray(
    [(214, 67, 73), (42, 113, 190), (239, 181, 45), (53, 157, 99)],
    dtype=np.float32,
)
GRID_SIDE = 12
QUADRANT_SIDE = 6
LABELS = ("A", "B", "C", "D")


def _runs(active: np.ndarray) -> list[tuple[int, int]]:
    groups: list[tuple[int, int]] = []
    start: int | None = None
    for index, value in enumerate(active.tolist() + [False]):
        if value and start is None:
            start = index
        elif not value and start is not None:
            groups.append((start, index - 1))
            start = None
    return groups


def _classify_cells(image: Image.Image) -> list[list[int]]:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    distances = np.linalg.norm(
        array[:, :, None, :] - COLOR_PROTOTYPES[None, None, :, :], axis=3
    )
    nearest = np.argmin(distances, axis=2)
    colored = np.min(distances, axis=2) < 18.0

    # Each colored cell interior is a long projection run; the dark lattice
    # separates adjacent runs. A local erasure cannot manufacture a new run.
    min_projection = max(40, int(0.16 * min(array.shape[:2])))
    x_runs = _runs(np.sum(colored, axis=0) >= min_projection)
    y_runs = _runs(np.sum(colored, axis=1) >= min_projection)
    if len(x_runs) != GRID_SIDE or len(y_runs) != GRID_SIDE:
        raise ValueError(
            f"could not recover a complete 12x12 lattice: {len(x_runs)} by {len(y_runs)}"
        )

    x_centers = [(first + last) // 2 for first, last in x_runs]
    y_centers = [(first + last) // 2 for first, last in y_runs]
    cells: list[list[int]] = []
    radius = 5
    for cy in y_centers:
        row: list[int] = []
        for cx in x_centers:
            y0, y1 = max(0, cy - radius), min(array.shape[0], cy + radius + 1)
            x0, x1 = max(0, cx - radius), min(array.shape[1], cx + radius + 1)
            patch_colored = colored[y0:y1, x0:x1]
            patch_labels = nearest[y0:y1, x0:x1]
            total = patch_colored.size
            present = int(patch_colored.sum())
            if present < int(0.82 * total):
                raise ValueError("a mosaic cell is erased or visually ambiguous")
            counts = np.bincount(patch_labels[patch_colored], minlength=4)
            winner = int(np.argmax(counts))
            if int(counts[winner]) < int(0.90 * present):
                raise ValueError("a mosaic cell has no unique visible color")
            row.append(winner)
        cells.append(row)
    return cells


def _quadrant_scores(cells: list[list[int]]) -> list[int]:
    scores = []
    for index in range(4):
        row0 = QUADRANT_SIDE if index >= 2 else 0
        col0 = QUADRANT_SIDE if index % 2 else 0
        mismatches = 0
        for row in range(QUADRANT_SIDE):
            for col in range(QUADRANT_SIDE):
                other_row = QUADRANT_SIDE - 1 - row
                other_col = QUADRANT_SIDE - 1 - col
                if (row, col) < (other_row, other_col):
                    mismatches += (
                        cells[row0 + row][col0 + col]
                        != cells[row0 + other_row][col0 + other_col]
                    )
        scores.append(int(mismatches))
    return scores


def analyze_image(image: Image.Image) -> dict:
    scores = _quadrant_scores(_classify_cells(image))
    best = min(scores)
    winners = [index for index, value in enumerate(scores) if value == best]
    ordered = sorted(scores)
    runner_up_gap = ordered[1] - ordered[0]
    if len(winners) != 1 or best != 0 or runner_up_gap < 2:
        raise ValueError(
            "no unique half-turn-invariant quadrant with a two-pair runner-up margin"
        )
    return {
        "decision": LABELS[winners[0]],
        "margin": float(runner_up_gap),
        "mismatched_pairs": scores,
    }


def decision_from_image(image: Image.Image) -> str:
    return str(analyze_image(image)["decision"])
