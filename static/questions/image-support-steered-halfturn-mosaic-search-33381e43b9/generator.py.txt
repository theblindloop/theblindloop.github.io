"""Analytic scene sampler and raster renderer for halfturn_mosaic_search.

The gold arm works on cell identities.  It never stores a target flag: the
winning label is recomputed from the same visible equality relation posed in
the question.
"""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

CANVAS = 512
CELL_SIZE = 32
SIDE = 12
QUAD_SIDE = 6
BACKGROUND = (248, 247, 242)
INK = (31, 35, 40)
BASE_PALETTE = (
    (214, 67, 73),
    (42, 113, 190),
    (239, 181, 45),
    (53, 157, 99),
)
LABELS = ("A", "B", "C", "D")


def _opposite_pairs() -> list[tuple[tuple[int, int], tuple[int, int]]]:
    pairs = []
    for row in range(QUAD_SIDE):
        for col in range(QUAD_SIDE):
            other = (QUAD_SIDE - 1 - row, QUAD_SIDE - 1 - col)
            if (row, col) < other:
                pairs.append(((row, col), other))
    return pairs


PAIRS = _opposite_pairs()


def _symmetric_quadrant(rng: random.Random) -> list[list[int]]:
    # Each quadrant has the same near-balanced histogram: 8, 8, 10, 10.
    pair_colors = [0] * 4 + [1] * 4 + [2] * 5 + [3] * 5
    rng.shuffle(pair_colors)
    quadrant = [[-1] * QUAD_SIDE for _ in range(QUAD_SIDE)]
    for (first, second), color in zip(PAIRS, pair_colors, strict=True):
        quadrant[first[0]][first[1]] = color
        quadrant[second[0]][second[1]] = color
    return quadrant


def _mismatch_pairs(quadrant: list[list[int]]) -> int:
    return sum(
        quadrant[a[0]][a[1]] != quadrant[b[0]][b[1]] for a, b in PAIRS
    )


def _make_distractor(
    symmetric: list[list[int]], rng: random.Random, swap_count: int
) -> list[list[int]]:
    """Break 2*swap_count opposite pairs while preserving the histogram."""
    quadrant = copy.deepcopy(symmetric)
    unused = list(range(len(PAIRS)))
    rng.shuffle(unused)
    used: set[int] = set()
    for _ in range(swap_count):
        choices = [index for index in unused if index not in used]
        found: tuple[int, int] | None = None
        for left in choices:
            left_pos = PAIRS[left][0]
            left_color = quadrant[left_pos[0]][left_pos[1]]
            for right in choices:
                if right == left:
                    continue
                right_pos = PAIRS[right][0]
                right_color = quadrant[right_pos[0]][right_pos[1]]
                if right_color != left_color:
                    found = (left, right)
                    break
            if found is not None:
                break
        if found is None:
            raise RuntimeError("could not construct histogram-matched distractor")
        left, right = found
        used.update((left, right))
        lp, rp = PAIRS[left][0], PAIRS[right][0]
        quadrant[lp[0]][lp[1]], quadrant[rp[0]][rp[1]] = (
            quadrant[rp[0]][rp[1]],
            quadrant[lp[0]][lp[1]],
        )
    expected = 2 * swap_count
    if _mismatch_pairs(quadrant) != expected:
        raise AssertionError("distractor construction lost its mismatch margin")
    return quadrant


def _join_quadrants(quadrants: list[list[list[int]]]) -> list[list[int]]:
    grid = [[0] * SIDE for _ in range(SIDE)]
    for index, quadrant in enumerate(quadrants):
        row_offset = QUAD_SIDE if index >= 2 else 0
        col_offset = QUAD_SIDE if index % 2 else 0
        for row in range(QUAD_SIDE):
            for col in range(QUAD_SIDE):
                grid[row_offset + row][col_offset + col] = quadrant[row][col]
    return grid


def _split_quadrants(grid: list[list[int]]) -> list[list[list[int]]]:
    quadrants = []
    for index in range(4):
        row_offset = QUAD_SIDE if index >= 2 else 0
        col_offset = QUAD_SIDE if index % 2 else 0
        quadrants.append(
            [
                row[col_offset : col_offset + QUAD_SIDE]
                for row in grid[row_offset : row_offset + QUAD_SIDE]
            ]
        )
    return quadrants


def _scores(scene: dict) -> list[int]:
    return [_mismatch_pairs(part) for part in _split_quadrants(scene["grid"])]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target_index = int(seed) % 4
    symmetric = _symmetric_quadrant(rng)
    base_swaps = 1 + ((int(seed) // 4) % 3)
    swap_counts = [base_swaps, base_swaps + 1, base_swaps + 2]
    rng.shuffle(swap_counts)
    quadrants: list[list[list[int]]] = []
    distractor_number = 0
    for index in range(4):
        if index == target_index:
            quadrants.append(copy.deepcopy(symmetric))
        else:
            quadrants.append(
                _make_distractor(symmetric, rng, swap_counts[distractor_number])
            )
            distractor_number += 1

    palette = [list(color) for color in BASE_PALETTE]
    rng.shuffle(palette)
    x0 = (56, 64, 72)[rng.randrange(3)]
    y0 = (56, 64, 72)[rng.randrange(3)]
    board_size = SIDE * CELL_SIZE
    return {
        "grid": _join_quadrants(quadrants),
        "palette": palette,
        "labels": list(LABELS),
        "width": CANVAS,
        "height": CANVAS,
        "board_box": [x0, y0, x0 + board_size, y0 + board_size],
        "cell_size": CELL_SIZE,
    }


_GLYPHS = {
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "B": ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    "C": ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    "D": ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
}


def _draw_label(draw: ImageDraw.ImageDraw, center_x: int, top: int, label: str) -> None:
    draw.rectangle((center_x - 16, top, center_x + 16, top + 27), fill=INK)
    pattern = _GLYPHS[label]
    scale = 3
    origin_x = center_x - (5 * scale) // 2
    origin_y = top + 3
    for row, bits in enumerate(pattern):
        for col, bit in enumerate(bits):
            if bit == "1":
                draw.rectangle(
                    (
                        origin_x + col * scale,
                        origin_y + row * scale,
                        origin_x + col * scale + scale - 1,
                        origin_y + row * scale + scale - 1,
                    ),
                    fill=(255, 255, 255),
                )


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    cell_size = int(scene["cell_size"])
    x0, y0, x1, y1 = (int(value) for value in scene["board_box"])
    grid = scene["grid"]
    palette = [tuple(int(channel) for channel in color) for color in scene["palette"]]
    labels = [str(value) for value in scene["labels"]]
    if len(grid) != SIDE or any(len(row) != SIDE for row in grid):
        raise ValueError("grid must be 12 by 12")
    if (x1 - x0, y1 - y0) != (SIDE * cell_size, SIDE * cell_size):
        raise ValueError("board dimensions disagree with the cell grid")

    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for row in range(SIDE):
        for col in range(SIDE):
            color = palette[int(grid[row][col])]
            draw.rectangle(
                (
                    x0 + col * cell_size,
                    y0 + row * cell_size,
                    x0 + (col + 1) * cell_size - 1,
                    y0 + (row + 1) * cell_size - 1,
                ),
                fill=color,
            )

    # A dark lattice makes the candidates explicit while the colored carpet
    # remains one high-fill rectangular visual structure.
    for offset in range(1, SIDE):
        line_width = 7 if offset == QUAD_SIDE else 2
        x = x0 + offset * cell_size
        y = y0 + offset * cell_size
        draw.line((x, y0, x, y1 - 1), fill=INK, width=line_width)
        draw.line((x0, y, x1 - 1, y), fill=INK, width=line_width)
    draw.rectangle((x0, y0, x1 - 1, y1 - 1), outline=INK, width=5)

    centers = (x0 + 3 * cell_size, x0 + 9 * cell_size)
    _draw_label(draw, centers[0], y0 - 27, labels[0])
    _draw_label(draw, centers[1], y0 - 27, labels[1])
    _draw_label(draw, centers[0], y1, labels[2])
    _draw_label(draw, centers[1], y1, labels[3])
    return image


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    winners = [index for index, score in enumerate(scores) if score == min(scores)]
    if len(winners) != 1 or scores[winners[0]] != 0:
        raise ValueError("scene has no unique half-turn-invariant quadrant")
    return str(scene["labels"][winners[0]])


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene))
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    scores = sorted(_scores(scene))
    return scores[0] != 0 or scores[1] - scores[0] < 2


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # Jointly rename palette indices and palette slots.  The latent encoding is
    # different, but every rendered RGB pixel and every equality relation is
    # identical.  This is a genuine representational alias, not a nuisance tag.
    permutation = (1, 0, 3, 2)  # new slot -> old slot; self-inverse
    twin = copy.deepcopy(scene)
    twin["palette"] = [copy.deepcopy(scene["palette"][old]) for old in permutation]
    twin["grid"] = [
        [permutation[int(value)] for value in row] for row in scene["grid"]
    ]
    return [("palette_index_relabel", twin)]
