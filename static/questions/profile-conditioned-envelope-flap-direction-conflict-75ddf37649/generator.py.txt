"""Deterministic renderer and analytic arm for envelope-flap direction."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

DIRECTIONS = ("up", "right", "down", "left")
FLAP_COLORS = ("#1677B8", "#19866F", "#7952B3", "#D66A24")
OUTLINE_COLORS = ("#27313A", "#30303A", "#26352F")
STAMP_COLORS = ("#D84A4A", "#2A78C5", "#E0A526", "#3A9B67")
STAMP_CORNERS = ("top-left", "top-right", "bottom-right", "bottom-left")
SUPER_SAMPLE = 4


def sample_scene(seed: int) -> dict:
    """Return a deterministic scene; each four-seed block covers every label."""

    seed = int(seed)
    rng = random.Random(seed ^ 0xE17A9B)
    direction_index = seed % 4
    block = seed // 4
    narrow = block % 6 == 0
    return {
        "flap_direction": DIRECTIONS[direction_index],
        "center_x": rng.randint(188, 198),
        "center_y": rng.randint(188, 198),
        "envelope_width": rng.randint(258, 282),
        "envelope_height": rng.randint(184, 206),
        "flap_depth": round(
            rng.uniform(58.0, 74.0) if narrow else rng.uniform(108.0, 126.0), 4
        ),
        "flap_base_span": round(rng.uniform(132.0, 154.0), 4),
        "flap_color": FLAP_COLORS[(3 * block + block // 4) % len(FLAP_COLORS)],
        "outline_color": OUTLINE_COLORS[block % len(OUTLINE_COLORS)],
        "stamp_corner": STAMP_CORNERS[(3 * block + 1) % len(STAMP_CORNERS)],
        "stamp_phase": rng.randrange(4),
        "canvas_size": 384,
        "background": "#F3EFE6",
    }


def _envelope_bounds(scene: dict, scale: int) -> tuple[float, float, float, float]:
    cx = float(scene["center_x"]) * scale
    cy = float(scene["center_y"]) * scale
    half_w = float(scene["envelope_width"]) * scale / 2.0
    half_h = float(scene["envelope_height"]) * scale / 2.0
    return cx - half_w, cy - half_h, cx + half_w, cy + half_h


def _flap_points(scene: dict, scale: int) -> list[tuple[float, float]]:
    left, top, right, bottom = _envelope_bounds(scene, scale)
    cx = float(scene["center_x"]) * scale
    cy = float(scene["center_y"]) * scale
    depth = float(scene["flap_depth"]) * scale
    half_base = float(scene["flap_base_span"]) * scale / 2.0
    inset = 9.0 * scale
    direction = str(scene["flap_direction"])
    if direction == "down":
        base_y = top + inset
        return [(cx - half_base, base_y), (cx + half_base, base_y), (cx, base_y + depth)]
    if direction == "up":
        base_y = bottom - inset
        return [(cx - half_base, base_y), (cx + half_base, base_y), (cx, base_y - depth)]
    if direction == "right":
        base_x = left + inset
        return [(base_x, cy - half_base), (base_x, cy + half_base), (base_x + depth, cy)]
    if direction == "left":
        base_x = right - inset
        return [(base_x, cy - half_base), (base_x, cy + half_base), (base_x - depth, cy)]
    raise ValueError(f"unknown flap direction: {direction}")


def _draw_stamp(scene: dict, image: Image.Image, scale: int) -> None:
    draw = ImageDraw.Draw(image)
    left, top, right, bottom = _envelope_bounds(scene, scale)
    stamp_w = 38.0 * scale
    stamp_h = 31.0 * scale
    pad = 20.0 * scale
    corner = str(scene["stamp_corner"])
    x0 = left + pad if corner.endswith("left") else right - pad - stamp_w
    y0 = top + pad if corner.startswith("top") else bottom - pad - stamp_h
    x1, y1 = x0 + stamp_w, y0 + stamp_h
    draw.rectangle((x0, y0, x1, y1), fill="#EFE6D4", outline=str(scene["outline_color"]), width=2 * scale)
    phase = int(scene["stamp_phase"]) % 4
    centers = (
        (x0 + 8 * scale, y0 + 8 * scale),
        (x1 - 8 * scale, y0 + 8 * scale),
        (x1 - 8 * scale, y1 - 8 * scale),
        (x0 + 8 * scale, y1 - 8 * scale),
    )
    radius = 3.2 * scale
    for index, (x, y) in enumerate(centers):
        color = STAMP_COLORS[(index + phase) % 4]
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)


def render(scene: dict) -> Image.Image:
    """Render an RGB raster from public scene fields only."""

    scale = SUPER_SAMPLE
    size = int(scene["canvas_size"])
    image = Image.new("RGB", (size * scale, size * scale), str(scene["background"]))
    draw = ImageDraw.Draw(image)
    bounds = _envelope_bounds(scene, scale)
    outline = str(scene["outline_color"])
    line_width = 7 * scale
    radius = 12 * scale
    draw.rounded_rectangle(bounds, radius=radius, fill="#FFFDF7", outline=outline, width=line_width)

    left, top, right, bottom = bounds
    cx = float(scene["center_x"]) * scale
    cy = float(scene["center_y"]) * scale
    seam = "#C9C4B9"
    inset = 8 * scale
    for corner in ((left + inset, top + inset), (right - inset, top + inset),
                   (right - inset, bottom - inset), (left + inset, bottom - inset)):
        draw.line((corner, (cx, cy)), fill=seam, width=3 * scale)

    draw.polygon(_flap_points(scene, scale), fill=str(scene["flap_color"]))
    draw.rounded_rectangle(bounds, radius=radius, outline=outline, width=line_width)
    _draw_stamp(scene, image, scale)
    return image.resize((size, size), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["flap_direction"])


def margin(scene: dict) -> float:
    """Ideal apex-axis centroid advantage over the perpendicular axis."""

    return float(scene["flap_depth"]) / 6.0


def is_quarantined(scene: dict) -> bool:
    return float(scene["flap_depth"]) < 84.0 or float(scene["flap_base_span"]) < 120.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Exercise the genuine period-four alias in stamp color indexing."""

    twin = dict(scene)
    twin["stamp_phase"] = int(scene["stamp_phase"]) + 4
    return [("stamp_phase_plus_four", twin)]
