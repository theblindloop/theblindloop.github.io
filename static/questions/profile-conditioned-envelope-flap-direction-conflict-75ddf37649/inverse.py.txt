"""Independent inverse arm that receives only the final envelope PNG."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _largest_component(mask: np.ndarray) -> np.ndarray:
    """Keep only the largest 4-connected region without renderer assumptions."""

    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    largest: list[tuple[int, int]] = []
    for start_y, start_x in np.argwhere(mask):
        y0, x0 = int(start_y), int(start_x)
        if visited[y0, x0]:
            continue
        visited[y0, x0] = True
        queue: deque[tuple[int, int]] = deque([(y0, x0)])
        component: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            component.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    queue.append((ny, nx))
        if len(component) > len(largest):
            largest = component
    result = np.zeros_like(mask, dtype=bool)
    if largest:
        coordinates = np.asarray(largest, dtype=np.int32)
        result[coordinates[:, 0], coordinates[:, 1]] = True
    return result


def _dominant_flap_mask(image: Image.Image) -> tuple[np.ndarray, np.ndarray]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    colors, counts = np.unique(arr.reshape(-1, 3), axis=0, return_counts=True)
    candidates: list[tuple[int, np.ndarray]] = []
    for color, count in zip(colors, counts):
        chroma = int(color.max()) - int(color.min())
        if count >= 1400 and chroma >= 50 and int(color.max()) <= 240:
            candidates.append((int(count), color))
    if not candidates:
        raise ValueError("no large colored triangular flap is visible")
    _, flap_color = max(candidates, key=lambda item: item[0])
    distance = np.sqrt(
        ((arr.astype(np.float64) - flap_color.astype(np.float64)) ** 2).sum(axis=2)
    )
    mask = _largest_component(distance <= 24.0)
    if int(mask.sum()) < 1800:
        raise ValueError("the candidate flap has too little visible ink")
    return arr, mask


def _envelope_body_bounds(arr: np.ndarray, flap_mask: np.ndarray) -> tuple[int, int, int, int]:
    dark = (arr.max(axis=2) <= 92) & ((arr.max(axis=2) - arr.min(axis=2)) <= 35)
    ys, xs = np.nonzero(dark)
    if len(xs) < 1800:
        raise ValueError("no complete dark envelope outline is visible")
    left, right = int(xs.min()), int(xs.max())
    top, bottom = int(ys.min()), int(ys.max())
    height, width = arr.shape[:2]
    if right - left < 0.48 * width or bottom - top < 0.35 * height:
        raise ValueError("the visible envelope body is too small or incomplete")
    fy, fx = np.nonzero(flap_mask)
    if int(fx.min()) < left - 3 or int(fx.max()) > right + 3 or int(fy.min()) < top - 3 or int(fy.max()) > bottom + 3:
        raise ValueError("the colored region is not contained by the envelope body")
    return left, top, right, bottom


def measurement_from_image(image: Image.Image) -> dict:
    """Return apex direction and the image-space runner-up margin."""

    arr, mask = _dominant_flap_mask(image)
    body_bounds = _envelope_body_bounds(arr, mask)
    ys, xs = np.nonzero(mask)
    left, right = float(xs.min()), float(xs.max())
    top, bottom = float(ys.min()), float(ys.max())
    if right - left < 42.0 or bottom - top < 42.0:
        raise ValueError("the colored flap lacks two-dimensional triangular extent")

    centroid_x = float(xs.mean())
    centroid_y = float(ys.mean())
    box_center_x = (left + right) / 2.0
    box_center_y = (top + bottom) / 2.0
    delta_x = centroid_x - box_center_x
    delta_y = centroid_y - box_center_y
    horizontal = abs(delta_x)
    vertical = abs(delta_y)
    runner_up_margin = abs(horizontal - vertical)
    threshold = max(5.0, 0.03 * max(right - left, bottom - top))
    if runner_up_margin < threshold:
        raise ValueError("no unique triangular flap apex direction is recoverable")

    if horizontal > vertical:
        decision = "right" if delta_x < 0.0 else "left"
    else:
        decision = "down" if delta_y < 0.0 else "up"
    return {
        "decision": decision,
        "runner_up_margin": float(runner_up_margin),
        "centroid_offset": [round(delta_x, 4), round(delta_y, 4)],
        "flap_bounds": [int(left), int(top), int(right), int(bottom)],
        "envelope_bounds": list(body_bounds),
    }


def decision_from_image(image: Image.Image) -> str:
    return str(measurement_from_image(image)["decision"])
