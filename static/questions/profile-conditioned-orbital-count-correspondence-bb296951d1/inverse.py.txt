"""Pixel-only inverse arm for the orbital-count correspondence world.

Receives only a final PNG. Reads the probe square, the probed disk's colour,
matches that colour into the reference (left) panel, and counts the black dots
orbiting that reference disk. Imports no renderer, prompts, generator, verifier
or analytic-gold code.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE_COLORS = {
    "blue": (0, 90, 200),
    "red": (200, 30, 30),
    "green": (30, 150, 60),
    "orange": (240, 140, 20),
}
COUNT_WORDS = {2: "two", 3: "three", 4: "four"}
PANEL_SPLIT = 500.0
R_DOTS = 44.0
COLOR_TOL = 70.0


def _classify(arr: np.ndarray):
    """Return array of palette-name strings (or None for background)."""
    h, w, _ = arr.shape
    out = np.empty((h, w), dtype=object)
    names = list(PALETTE_COLORS)
    cols = np.array([PALETTE_COLORS[n] for n in names], dtype=np.float64)
    flat = arr.reshape(-1, 3).astype(np.float64)
    dist = np.linalg.norm(flat[:, None, :] - cols[None, :, :], axis=2)
    nearest = dist.argmin(axis=1)
    best = dist[np.arange(len(flat)), nearest]
    labels = np.array(names)[nearest]
    labels[best > COLOR_TOL] = "background"
    out[:] = labels.reshape(h, w)
    return out


def _close_enough(a, b):
    return abs(np.linalg.norm(np.asarray(a, dtype=float) - np.asarray(b, dtype=float)))


def _color_match(color_value, name):
    return _close_enough(color_value, PALETTE_COLORS[name]) <= COLOR_TOL


def _components(mask: np.ndarray):
    """Return list of (area, centroid (cy,cx)) for 4-connected components."""
    h, w = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    comps = []
    for sy in range(h):
        for sx in range(w):
            if not mask[sy, sx] or visited[sy, sx]:
                continue
            q = deque()
            q.append((sy, sx))
            visited[sy, sx] = True
            pixels = []
            while q:
                y, x = q.popleft()
                pixels.append((y, x))
                for dy, dx in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < h and 0 <= nx < w and not visited[ny, nx] and mask[ny, nx]:
                        visited[ny, nx] = True
                        q.append((ny, nx))
            ys = [p[0] for p in pixels]
            xs = [p[1] for p in pixels]
            comps.append((len(pixels), (sum(ys) / len(ys), sum(xs) / len(xs))))
    return comps


def decision_from_image(image: Image.Image) -> str:
    img = image.convert("RGB")
    arr = np.asarray(img)
    h, w, _ = arr.shape
    labels = _classify(arr)

    black = np.all(arr == 0, axis=2)

    candidate_mask = np.zeros_like(black, dtype=bool)
    candidate_mask[:, int(PANEL_SPLIT):] = black[:, int(PANEL_SPLIT):]
    probe_comps = [c for c in _components(candidate_mask) if c[0] >= 10]
    if not probe_comps:
        return "abstain"
    probe_cy, probe_cx = probe_comps[0][1]

    disks = {}  # name -> list of (centroid cy, cx) in BOTH panels, but we key by panel
    left_comps = {}
    right_comps = {}
    for name in PALETTE_COLORS:
        mask = labels == name
        comps = _components(mask)
        big = [c for c in comps if c[0] >= 800]
        left = []
        right = []
        for area, (cy, cx) in big:
            if cx < PANEL_SPLIT:
                left.append((cy, cx))
            else:
                right.append((cy, cx))
        if len(left) == 1 and len(right) == 1:
            left_comps[name] = left[0]
            right_comps[name] = right[0]

    if len(left_comps) != 4 or len(right_comps) != 4:
        return "abstain"

    # Find probed candidate disk: the disk whose column is nearest the probe.
    best_name = None
    best_dx = None
    for name, (cy, cx) in right_comps.items():
        dx = abs(cx - probe_cx)
        if best_dx is None or dx < best_dx:
            best_dx = dx
            best_name = name
    probe_color = best_name

    ref_cy, ref_cx = left_comps[probe_color]

    ref_black = np.zeros_like(black, dtype=bool)
    ref_black[:, : int(PANEL_SPLIT)] = black[:, : int(PANEL_SPLIT)]
    dot_count = 0
    for area, (cy, cx) in _components(ref_black):
        if area >= 10 and _close_enough((cy, cx), (ref_cy, ref_cx)) <= R_DOTS + 8:
            dot_count += 1
    return COUNT_WORDS.get(dot_count, "abstain")
