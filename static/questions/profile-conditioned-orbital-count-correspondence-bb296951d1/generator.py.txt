"""Orbital-count correspondence world.

Two horizontal panels share the same four colored disks (each color exactly once
per panel). The LEFT (reference) panel annotates every disk with a small orbiting
"dot count" (2, 3 or 4 black dots around it). The RIGHT (candidate) panel shows
the same colors in a freshly permuted order, with no dot annotations. A hollow
black probe square sits directly under one candidate disk.

The decision (the dot count carried by the reference counterpart) can only be
recovered by (1) identifying the color of the probed candidate disk and
(2) matching that color into the reference panel — a genuine cross-panel
correspondence. The dot count exists ONLY in the reference, so there is no
attribute leakage visible at the probed location.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

W = 1000
H = 470
PALETTE = ("blue", "red", "green", "orange")
RGB = {
    "blue": (0, 90, 200),
    "red": (200, 30, 30),
    "green": (30, 150, 60),
    "orange": (240, 140, 20),
}
COUNT_WORDS = {2: "two", 3: "three", 4: "four"}
DISK_R = 26
DOT_R = 6
R_DOTS = 44
COL_LEFT = (90, 190, 290, 390)
COL_RIGHT = (610, 710, 810, 910)
CY = 235
PROBE_Y = CY + DISK_R + 40
PROBE_HALF = 11


def _draw_disk(draw, cx, cy, color):
    box = (cx - DISK_R, cy - DISK_R, cx + DISK_R, cy + DISK_R)
    draw.ellipse(box, fill=RGB[color])


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    colors = list(PALETTE)
    rng.shuffle(colors)
    counts = {c: rng.choice((2, 3, 4)) for c in colors}
    while len(set(counts.values())) < 2:
        counts = {c: rng.choice((2, 3, 4)) for c in colors}
    rng.shuffle(colors)
    lorder = list(colors)
    rng.shuffle(colors)
    probe_color = rng.choice(PALETTE)
    # ensure probe colour is present in both panels (it always is) and keep the
    # candidate order different from the reference order in most cases is not
    # required; determinism is what matters.
    rorder = list(rng.sample(list(PALETTE), 4))
    return {
        "colors": list(PALETTE),
        "lorder": list(lorder),
        "rorder": list(rorder),
        "counts": counts,
        "probe_color": probe_color,
    }


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (W, H), (255, 255, 255))
    draw = ImageDraw.Draw(img)
    lorder = scene["lorder"]
    rorder = scene["rorder"]
    counts = scene["counts"]
    probe_color = scene["probe_color"]
    for color, cx in zip(lorder, COL_LEFT):
        _draw_disk(draw, cx, CY, color)
        k = counts[color]
        rng = random.Random(repr((color, cx, counts[color], scene["colors"])))
        base = rng.uniform(0.0, 360.0)
        for idx in range(k):
            angle = math.radians(base + idx * (360.0 / k))
            dx = int(round(R_DOTS * math.cos(angle)))
            dy = int(round(R_DOTS * math.sin(angle)))
            draw.ellipse(
                (cx + dx - DOT_R, CY + dy - DOT_R, cx + dx + DOT_R, CY + dy + DOT_R),
                fill=(0, 0, 0),
            )
    for color, cx in zip(rorder, COL_RIGHT):
        _draw_disk(draw, cx, CY, color)
    probe_cx = COL_RIGHT[rorder.index(probe_color)]
    sx0, sy0 = probe_cx - PROBE_HALF, PROBE_Y - PROBE_HALF
    draw.rectangle(
        (sx0, sy0, sx0 + 2 * PROBE_HALF, sy0 + 2 * PROBE_HALF),
        outline=(0, 0, 0),
        width=4,
    )
    return img


def _answer(scene: dict) -> str:
    return COUNT_WORDS[scene["counts"][scene["probe_color"]]]


def analytic_gold(scene: dict) -> str:
    counts = scene["counts"]
    probe_color = scene["probe_color"]
    return COUNT_WORDS[counts[probe_color]]


def margin(scene: dict) -> float:
    return float(scene["counts"][scene["probe_color"]])


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict):
    return iter(())
