"""Pixel-only inverse arm for the key-swatch tone-band stencil world.

This module receives a rendered image and nothing else. It locates the coloured
fiducials (the two outlined key squares and the panel frame), measures the two
key greys, recovers the 9x9 cell lattice from the frame interior, measures each
cell's flat grey, executes the declared two-sided keep rule against the key
band, and names the closest of the five public symbols under the declared
sliding match. It imports only PIL and shares no code or constants with the
renderer.
"""

from __future__ import annotations

from PIL import Image

ABSTAIN = "abstain"
_NAMES = ("plus", "square", "chevron", "ladder", "arrow")
_SIDE = 9
_CORE = 7
_SHIFTS = _SIDE - _CORE + 1
_KEEP_TOL = 12
_DROP_TOL = 26
_MAX_BEST = 6


def _templates() -> dict[str, frozenset]:
    """Independently written copies of the five public 7x7 symbol templates."""

    out: dict[str, frozenset] = {}
    for name in _NAMES:
        cells = set()
        for row in range(_CORE):
            for col in range(_CORE):
                arm = abs(col - 3)
                if name == "plus":
                    hit = col == 3 or row == 3 or (row in (2, 4) and arm <= 1)
                elif name == "square":
                    hit = (1 <= row <= 5 and 1 <= col <= 5 and (row in (1, 5) or col in (1, 5))) or (
                        row == 3 and col == 3
                    )
                elif name == "chevron":
                    if row <= 3:
                        hit = arm in (3 - row, 4 - row)
                    elif row == 5:
                        hit = arm == 1
                    else:
                        hit = arm == 0
                elif name == "ladder":
                    hit = col in (1, 5) or (row == 3 and 1 <= col <= 5)
                else:
                    hit = (row <= 3 and arm <= row) or (row == 4 and arm == 0)
                if hit:
                    cells.add((row, col))
        out[name] = frozenset(cells)
    return out


def decision_from_image(image: Image.Image) -> str:
    return _decide(image)[0]


def decision_with_margin(image: Image.Image) -> tuple[str, float]:
    """Reported decision plus the runner-up gap in disagreeing cells."""

    return _decide(image)


def _bands(rows: list[int]) -> list[tuple[int, int]]:
    out: list[tuple[int, int]] = []
    for y in rows:
        if out and y - out[-1][1] <= 3:
            out[-1] = (out[-1][0], y)
        else:
            out.append((y, y))
    return out


def _runs(values: list[int], gap: int) -> list[tuple[int, int]]:
    out: list[tuple[int, int]] = []
    for v in values:
        if out and v - out[-1][1] <= gap:
            out[-1] = (out[-1][0], v)
        else:
            out.append((v, v))
    return out


def _patch(pixels, cx: int, cy: int, reach: int, width: int, height: int):
    values = []
    for y in range(cy - reach, cy + reach + 1):
        if y < 0 or y >= height:
            return None
        for x in range(cx - reach, cx + reach + 1):
            if x < 0 or x >= width:
                return None
            r, g, b = pixels[x, y]
            if max(r, g, b) - min(r, g, b) > 20:
                return None
            values.append((r + g + b) / 3.0)
    values.sort()
    return values[len(values) // 2]


def _decide(image: Image.Image) -> tuple[str, float]:
    rgb = image.convert("RGB")
    width, height = rgb.size
    if width < 200 or height < 200:
        return ABSTAIN, 0.0
    pixels = rgb.load()

    row_min: dict[int, int] = {}
    row_max: dict[int, int] = {}
    for y in range(height):
        lo = -1
        hi = -1
        for x in range(width):
            r, g, b = pixels[x, y]
            if b - max(r, g) > 60:
                if lo < 0:
                    lo = x
                hi = x
        if lo >= 0:
            row_min[y] = lo
            row_max[y] = hi
    if len(row_min) < 40:
        return ABSTAIN, 0.0

    bands = _bands(sorted(row_min))
    if len(bands) < 2:
        return ABSTAIN, 0.0
    panel = max(bands, key=lambda band: band[1] - band[0])
    keys = [band for band in bands if band[1] < panel[0]]
    if len(keys) != 1:
        return ABSTAIN, 0.0
    key_band = keys[0]

    key_xs = sorted(
        {
            x
            for y in range(key_band[0], key_band[1] + 1)
            for x in range(row_min.get(y, 1), row_max.get(y, 0) + 1)
            if b_at(pixels, x, y)
        }
    )
    key_runs = _runs(key_xs, 6)
    if len(key_runs) != 2:
        return ABSTAIN, 0.0
    key_greys = []
    for x0, x1 in key_runs:
        if x1 - x0 < 20 or key_band[1] - key_band[0] < 20:
            return ABSTAIN, 0.0
        value = _patch(
            pixels,
            (x0 + x1) // 2,
            (key_band[0] + key_band[1]) // 2,
            8,
            width,
            height,
        )
        if value is None:
            return ABSTAIN, 0.0
        key_greys.append(value)
    lo_key = min(key_greys)
    hi_key = max(key_greys)
    if hi_key - lo_key < 16 or hi_key - lo_key > 60:
        return ABSTAIN, 0.0

    px0 = min(row_min[y] for y in range(panel[0], panel[1] + 1) if y in row_min)
    px1 = max(row_max[y] for y in range(panel[0], panel[1] + 1) if y in row_min)
    py0, py1 = panel
    if (px1 - px0) != (py1 - py0):
        return ABSTAIN, 0.0
    thickness = 0
    mid_x = (px0 + px1) // 2
    while thickness < 40 and b_at(pixels, mid_x, py0 + thickness):
        thickness += 1
    if thickness < 2:
        return ABSTAIN, 0.0
    span = (px1 - px0 + 1) - 2 * thickness
    if span < 9 * _SIDE or span % _SIDE != 0:
        return ABSTAIN, 0.0
    quantum = span // _SIDE
    reach = max(3, quantum // 4)
    ix = px0 + thickness
    iy = py0 + thickness

    kept = set()
    for row in range(_SIDE):
        for col in range(_SIDE):
            value = _patch(
                pixels,
                ix + col * quantum + quantum // 2,
                iy + row * quantum + quantum // 2,
                reach,
                width,
                height,
            )
            if value is None:
                return ABSTAIN, 0.0
            if lo_key - _KEEP_TOL <= value <= hi_key + _KEEP_TOL:
                kept.add((row, col))
            elif not (value < lo_key - _DROP_TOL or value > hi_key + _DROP_TOL):
                return ABSTAIN, 0.0

    templates = _templates()
    scored = []
    for index, name in enumerate(_NAMES):
        best = min(
            len(kept ^ frozenset((r + dr, c + dc) for r, c in templates[name]))
            for dr in range(_SHIFTS)
            for dc in range(_SHIFTS)
        )
        scored.append((best, index, name))
    scored.sort()
    if scored[0][0] > _MAX_BEST:
        return ABSTAIN, 0.0
    return scored[0][2], float(scored[1][0] - scored[0][0])


def b_at(pixels, x: int, y: int) -> bool:
    r, g, b = pixels[x, y]
    return b - max(r, g) > 60
