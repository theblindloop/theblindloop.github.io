"""Renderer and analytic gold for the key-swatch tone-band stencil world.

One 9x9 panel of flat grey cells is printed inside a coloured frame. Two key
squares above the panel are printed in two of the six evenly spaced greys used
by the panel. The declared transform keeps exactly the cells whose grey lies in
the closed band spanned by the two key greys and discards every other cell,
whether it is lighter or darker; the kept cells spell one of five named symbols
on a 7x7 block that may sit anywhere inside the panel.

Analytic gold is a function of the rendered grey field and the rendered key
band alone. The generative bookkeeping (`symbol`, `decoy`, `flips`, `offsets`)
is never read by `render` or by `analytic_gold`.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

GRID = 9
CORE = 7
PLACEMENTS = GRID - CORE + 1
LEVELS = (44, 76, 108, 140, 172, 204)
SYMBOLS = ("plus", "square", "chevron", "ladder", "arrow")
CELL = 44
PAD = 2
FRAME = 6
SWATCH = 60
SWATCH_GAP = 40
CANVAS = (620, 700)
PAGE = (250, 250, 250)
FRAME_COLOR = (25, 60, 190)
QUARANTINE_MARGIN = 2.0

_PATTERNS = {
    "plus": ("...#...", "...#...", "..###..", "#######", "..###..", "...#...", "...#..."),
    "square": (".......", ".#####.", ".#...#.", ".#.#.#.", ".#...#.", ".#####.", "......."),
    "chevron": ("#.....#", "##...##", ".##.##.", "..###..", "...#...", "..#.#..", "...#..."),
    "ladder": (".#...#.", ".#...#.", ".#...#.", ".#####.", ".#...#.", ".#...#.", ".#...#."),
    "arrow": ("...#...", "..###..", ".#####.", "#######", "...#...", ".......", "......."),
}

TEMPLATES = {
    name: frozenset(
        (row, col)
        for row, line in enumerate(rows)
        for col, mark in enumerate(line)
        if mark == "#"
    )
    for name, rows in _PATTERNS.items()
}
INK_CELLS = {name: len(cells) for name, cells in TEMPLATES.items()}
assert set(TEMPLATES) == set(SYMBOLS)
assert len(set(INK_CELLS.values())) == 1, INK_CELLS


def placed(name: str, dr: int, dc: int) -> frozenset:
    return frozenset((r + dr, c + dc) for r, c in TEMPLATES[name])


def _distance(mask: frozenset, other: frozenset) -> int:
    return len(mask ^ other)


def _best_distance(mask: frozenset, name: str) -> int:
    return min(
        _distance(mask, placed(name, dr, dc))
        for dr in range(PLACEMENTS)
        for dc in range(PLACEMENTS)
    )


def _scores(mask: frozenset) -> list[tuple[int, int, str]]:
    return sorted(
        (_best_distance(mask, name), index, name) for index, name in enumerate(SYMBOLS)
    )


def separation() -> int:
    """Smallest sliding distance between two different placed symbols."""

    worst = 10**6
    for i, first in enumerate(SYMBOLS):
        for second in SYMBOLS[i + 1 :]:
            for dr in range(PLACEMENTS):
                for dc in range(PLACEMENTS):
                    worst = min(worst, _best_distance(placed(first, dr, dc), second))
    return worst


def _band_mask(scene: dict) -> frozenset:
    band = int(scene["band"])
    levels = scene["levels"]
    return frozenset(
        (r, c)
        for r in range(GRID)
        for c in range(GRID)
        if int(levels[r][c]) in (band, band + 1)
    )


def _decoy_mask(scene: dict) -> frozenset:
    band = int(scene["band"])
    levels = scene["levels"]
    return frozenset(
        (r, c) for r in range(GRID) for c in range(GRID) if int(levels[r][c]) == band - 1
    )


def analytic_gold(scene: dict) -> str:
    return _scores(_band_mask(scene))[0][2]


def margin(scene: dict) -> float:
    scores = _scores(_band_mask(scene))
    return float(scores[1][0] - scores[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= QUARANTINE_MARGIN


def _naive_decoy_symbol(scene: dict) -> str:
    """The symbol a reader gets by taking the single darkest-of-the-key tone."""

    return _scores(_decoy_mask(scene))[0][2]


def _one_sided_scores(scene: dict) -> list[tuple[int, int, str]]:
    band = int(scene["band"])
    levels = scene["levels"]
    mask = frozenset(
        (r, c) for r in range(GRID) for c in range(GRID) if int(levels[r][c]) <= band + 1
    )
    return _scores(mask)


def _naive_one_sided_symbol(scene: dict) -> str:
    """Nearest symbol under the one-sided reading 'not lighter than the light key'."""

    return _one_sided_scores(scene)[0][2]


def _naive_one_sided_margin(scene: dict) -> float:
    """Runner-up gap of that one-sided reading; zero means it decides nothing."""

    scores = _one_sided_scores(scene)
    return float(scores[1][0] - scores[0][0])


def one_sided_is_decisive_and_correct(scene: dict) -> bool:
    """True when discarding only the light side would still hand over the answer."""

    return (
        _naive_one_sided_symbol(scene) == analytic_gold(scene)
        and _naive_one_sided_margin(scene) > QUARANTINE_MARGIN
    )


def _levels_for(symbol_cells, decoy_cells, band: int, rng: random.Random) -> list[list[int]]:
    # The background is printed strictly lighter than the band and the decoy tone
    # is printed strictly darker, so a one-sided 'not lighter than the light key'
    # reading collects exactly the band cells plus the decoy cells.
    background = [lv for lv in range(band + 2, len(LEVELS))]
    grid = []
    for r in range(GRID):
        row = []
        for c in range(GRID):
            if (r, c) in symbol_cells:
                row.append(rng.choice((band, band + 1)))
            elif (r, c) in decoy_cells:
                row.append(band - 1)
            else:
                row.append(rng.choice(background))
        grid.append(row)
    return grid


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    symbol = rng.choice(SYMBOLS)
    decoy = rng.choice([name for name in SYMBOLS if name != symbol])
    band = rng.choice((1, 2))
    o1 = (rng.randrange(PLACEMENTS), rng.randrange(PLACEMENTS))
    o2 = (rng.randrange(PLACEMENTS), rng.randrange(PLACEMENTS))
    mode = rng.choice(("clean", "clean", "random", "random", "toward_rival", "toward_rival"))

    base = placed(symbol, *o1)
    decoy_cells = placed(decoy, *o2) - base
    cells = [(r, c) for r in range(GRID) for c in range(GRID)]
    rival = min(
        (_distance(base, placed(name, dr, dc)), index, name, dr, dc)
        for index, name in enumerate(SYMBOLS)
        if name != symbol
        for dr in range(PLACEMENTS)
        for dc in range(PLACEMENTS)
    )
    toward = sorted(base ^ placed(rival[2], rival[3], rival[4]))
    target = 0 if mode == "clean" else rng.choice((1, 2)) if mode == "random" else rng.choice((2, 3))

    chosen: list[tuple[int, int]] = []
    levels = _levels_for(base, decoy_cells, band, rng)
    accepted = False
    for attempt in range(64):
        if mode == "toward_rival":
            flips = rng.sample(toward, min(target, len(toward)))
        else:
            flips = rng.sample(cells, target) if target else []
        stencil = set(base)
        for cell in flips:
            if cell in stencil:
                stencil.discard(cell)
            else:
                stencil.add(cell)
        # The background tones are redrawn on every attempt: they never touch the
        # band mask, so gold is unchanged, but they decide whether the darkest-tone
        # and one-sided-threshold misreadings collide with the gold symbol.
        trial_rng = random.Random(seed * 104729 + attempt * 7717 + 1)
        if attempt:
            decoy = trial_rng.choice([name for name in SYMBOLS if name != symbol])
            o2 = (trial_rng.randrange(PLACEMENTS), trial_rng.randrange(PLACEMENTS))
            decoy_cells = placed(decoy, *o2) - base
        trial = _levels_for(stencil, decoy_cells - stencil, band, trial_rng)
        probe = {"band": band, "levels": trial}
        if analytic_gold(probe) != symbol:
            continue
        if _naive_decoy_symbol(probe) == symbol:
            continue
        if one_sided_is_decisive_and_correct(probe):
            continue
        chosen = [list(cell) for cell in flips]
        levels = trial
        accepted = True
        break
    if not accepted:  # pragma: no cover - not reached by any sampled seed
        raise RuntimeError(f"no admissible scene for seed {seed}")

    return {
        "levels": levels,
        "band": band,
        "symbol": symbol,
        "decoy": decoy,
        "offsets": [list(o1), list(o2)],
        "flips": chosen,
        "panel_x": 96 + rng.randrange(0, 21),
        "panel_y": 200 + rng.randrange(0, 17),
        "key_x": 120 + rng.randrange(0, 61),
        "key_y": 56 + rng.randrange(0, 21),
        "key_swap": bool(rng.getrandbits(1)),
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS, PAGE)
    draw = ImageDraw.Draw(image)

    px = int(scene["panel_x"])
    py = int(scene["panel_y"])
    span = GRID * CELL + 2 * FRAME
    draw.rectangle([px, py, px + span - 1, py + span - 1], outline=FRAME_COLOR, width=FRAME)
    ix = px + FRAME
    iy = py + FRAME
    levels = scene["levels"]
    for r in range(GRID):
        for c in range(GRID):
            grey = LEVELS[int(levels[r][c])]
            x = ix + c * CELL
            y = iy + r * CELL
            draw.rectangle(
                [x + PAD, y + PAD, x + CELL - PAD - 1, y + CELL - PAD - 1],
                fill=(grey, grey, grey),
            )

    band = int(scene["band"])
    pair = (LEVELS[band], LEVELS[band + 1])
    if scene["key_swap"]:
        pair = (pair[1], pair[0])
    kx = int(scene["key_x"])
    ky = int(scene["key_y"])
    outer = SWATCH + 2 * FRAME
    for index, grey in enumerate(pair):
        sx = kx + index * (outer + SWATCH_GAP)
        draw.rectangle([sx, ky, sx + outer - 1, ky + outer - 1], outline=FRAME_COLOR, width=FRAME)
        draw.rectangle(
            [sx + FRAME, ky + FRAME, sx + FRAME + SWATCH - 1, ky + FRAME + SWATCH - 1],
            fill=(grey, grey, grey),
        )
    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical rewrites of scene fields the renderer never reads."""

    first = dict(scene)
    first["symbol"] = "rewritten"
    first["decoy"] = "rewritten"
    first["offsets"] = [[0, 0], [0, 0]]
    first["flips"] = []

    second = dict(scene)
    second["flips"] = list(reversed(scene["flips"]))
    second["offsets"] = list(reversed(scene["offsets"]))
    return [("rewrite_generative_metadata", first), ("reverse_flip_listing", second)]
