"""value_modulo_band: value-domain modulo-residue band reveal.

Declared transform: the image is a 32x32 lattice of flat gray blocks, each
block painted a uniform random gray level. Every block is a single random
lumination drawn uniformly in [0, 255], so the raw field looks like uniform
flat noise; no local patch, row, column, or brightness cue exposes the hidden
band. Only inside a coarse vertical band (a left-hand block band or a
right-hand block band) is each block's gray level forced to be an exact
multiple of a declared prime M=7 (residue 0); outside the band the levels are
arbitrary random residues. The declared operation takes each block's gray
level MODULO M (a pure value-domain residue decimation, independent of spatial
position) and lights up the residue-0 blocks, which reconstructs a clean
vertical band whose left (L) or right (R) placement is the answer.

The mechanism is a value-domain residue decimation of each cell's own gray
level, not a positional parity/decimation (which samples by row/col phase) and
not a second-moment texture-energy filter; it reads the residue class of each
block's luminance value. In the raw raster the only distinguishing cue is the
value residue itself, so the band is unrecoverable without the declared modulo
step.

Latent scene z = {seedc (class/geometry seed), side (answer: L or R), seedv
(value seed)}. label = side.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RENDERER_VERSION = "value_modulo_band-0.1.0"

W = 512
H = 512
NGRID = 32          # blocks per side
CELL = W // NGRID   # 16
M = 7               # declared modulus (prime) for value-domain residues
L0, L1 = 3, 9       # left band block columns [3, 9)
R0, R1 = 23, 29     # right band block columns [23, 29)
Q_TOL = 0.15


def band_masks() -> dict[str, np.ndarray]:
    """(NGRID, NGRID) bool masks: L left vertical band, R right vertical band."""
    r, c = np.indices((NGRID, NGRID))
    mask_l = (c >= L0) & (c < L1)
    mask_r = (c >= R0) & (c < R1)
    return {"L": mask_l, "R": mask_r}


def build_scene(seedc: int, side: str, seedv: int) -> dict:
    return {"seedc": int(seedc), "side": side, "seedv": int(seedv)}


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(int(seed) & 0xFFFFFFFF)
    side = "L" if rng.integers(0, 2) else "R"
    return build_scene(seedc=int(seed), side=side, seedv=int(seed) + 37)


def _band_mask(side: str) -> np.ndarray:
    return band_masks()[side]


def render_matrix(scene: dict) -> np.ndarray:
    """(H, W) int luminance block lattice on which the residue is later read.

    Every block is painted a single uniform random gray level, so the whole
    field is flat per-block noise whose value distribution is identical inside
    and outside the band. Only inside the hidden band is a block's level forced
    to a multiple of M (residue 0). The declared modulo-M read then lights up
    exactly the band blocks.
    """
    rng = np.random.default_rng(int(scene["seedv"]) & 0xFFFFFFFF)
    vals = rng.integers(0, 256, size=(NGRID, NGRID)).astype(np.int64)
    mask = _band_mask(scene["side"])
    mult = rng.integers(0, 256 // M + 1, size=(NGRID, NGRID)).astype(np.int64)
    vals = np.where(mask, (mult * M), vals)
    # expand each block to CELL x CELL pixels of that flat level
    arr = np.repeat(np.repeat(vals, CELL, axis=0), CELL, axis=1)
    return arr[:W, :H]


def render(scene: dict) -> Image.Image:
    arr = render_matrix(scene)
    arr = np.clip(arr, 0, 255)
    canvas = np.repeat(arr[:, :, None], 3, axis=2).astype(np.uint8)
    canvas = canvas[:W, :H]
    return Image.fromarray(canvas, "RGB")


def margin(scene: dict) -> float:
    chosen = _band_mask(scene["side"])
    other = _band_mask("R" if scene["side"] == "L" else "L")
    o_filled = int(other.sum())
    corr_other = float((chosen & other).sum()) / max(1, o_filled)
    return round(1.0 - corr_other, 6)


def analytic_gold(scene: dict) -> str:
    return scene["side"]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < Q_TOL


def latent_symmetries(scene: dict):
    """not_applicable: every latent field (class seed, answer side, value seed)
    directly changes the rendered pixels, so no distinct latent vector
    reproduces the same image. Declared empty."""
    return iter(())
