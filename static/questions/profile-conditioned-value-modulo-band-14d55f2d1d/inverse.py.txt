"""Independent pixel-only inverse arm for value_modulo_band.

Receives ONLY the final PNG. It:
  1. reads the flat gray level of every 16x16 block (top-left sample);
  2. computes each block's residue modulo the declared prime M=7;
  3. lights up residue-0 blocks to form a binary band grid;
  4. correlates against the two declared vertical-band templates L and R and
     selects whose inside region carries the higher lit contrast.

No scene data, no analytic gold, no renderer imports.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "value_modulo_band-oracle-0.1.0"

W = 512
NGRID = 32
CELL = W // NGRID
M = 7
L0, L1 = 3, 9
R0, R1 = 23, 29


def _band_masks() -> dict[str, np.ndarray]:
    r, c = np.indices((NGRID, NGRID))
    mask_l = (c >= L0) & (c < L1)
    mask_r = (c >= R0) & (c < R1)
    return {"L": mask_l, "R": mask_r}


def _cell_levels(image: Image.Image) -> np.ndarray:
    """(NGRID, NGRID) flat gray level of every block (all inner pixels equal)."""
    arr = np.asarray(image.convert("L"), dtype=np.int64)
    return arr[::CELL, ::CELL]


def _contrast(lit: np.ndarray, mask: np.ndarray) -> float:
    inside = lit[mask].mean()
    outside = lit[~mask].mean()
    return float(inside - outside)


def decision_from_image(image: Image.Image) -> str:
    levels = _cell_levels(image)
    lit = (levels % M) == 0
    masks = _band_masks()
    sl = _contrast(lit, masks["L"])
    sr = _contrast(lit, masks["R"])
    return "L" if sl >= sr else "R"
