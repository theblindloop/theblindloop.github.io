"""Tracing world: how many black waypoints lie BEFORE the green terminal on-path.

A single winding blue path (one connected, non-self-crossing polyline) is drawn.
An orange port ring sits near the start. A green terminal dot sits part-way along
the path; a red terminal sits farther along. Several black waypoint dots are
placed at separated on-path positions, some strictly BEFORE the green dot (the
answer) and some after it (distractors). Walking from the orange ring along the
blue path toward the green terminal, the solution is the number of black
waypoints encountered strictly before reaching the green dot. Because the path
winds back on itself in a serpentine, straight-line order need not equal on-path
order, so the task cannot be solved by Euclidean projection.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "count_markers_before_terminal-0.1.0"

CANVAS = (480, 480)
BG = (255, 255, 255)
COLOR_BLUE = (40, 90, 210)
COLOR_RED = (225, 40, 40)
COLOR_GREEN = (40, 170, 70)
COLOR_BLACK = (30, 30, 30)
COLOR_ORANGE = (245, 150, 20)

STROKE = 20
PORT_R = 5
MARK_R = 4
MARGIN_PAD = 46
LANE_GAP = 30.0
ARC_SEP = 30.0
END_MARGIN = 40.0

ANSWER_VALUES = [0, 1, 2, 3]


def _pts(points: list) -> list[tuple[float, float]]:
    return [(float(p[0]), float(p[1])) for p in points]


def _path_length(points: list[tuple[float, float]]) -> float:
    return sum(math.dist(a, b) for a, b in zip(points, points[1:]))


def _point_at(points: list[tuple[float, float]], s: float) -> tuple[float, float]:
    remaining = s
    for a, b in zip(points, points[1:]):
        seg = math.dist(a, b)
        if remaining <= seg or seg == 0:
            t = 0.0 if seg == 0 else remaining / seg
            return (a[0] + (b[0] - a[0]) * t, a[1] + (b[1] - a[1]) * t)
        remaining -= seg
    return points[-1]


def _rejection_loop(rng: random.Random) -> dict:
    w, h = CANVAS
    lo, hi = MARGIN_PAD, w - MARGIN_PAD
    for _ in range(20000):
        lane_vert_gap = LANE_GAP
        n_lanes = rng.randint(4, 7)
        y_lo = lo + 10
        y_hi = hi - 10
        usable_h = y_hi - y_lo
        span_h = min((n_lanes - 1) * lane_vert_gap, usable_h)
        step_h = span_h / (n_lanes - 1) if n_lanes > 1 else 0.0
        x_start = rng.uniform(lo + 15, lo + 50)
        x_end = rng.uniform(hi - 50, hi - 15)
        path = []
        for lane in range(n_lanes):
            y = y_lo + lane * step_h
            if lane % 2 == 0:
                path.append((x_start, y))
                path.append((x_end, y))
            else:
                path.append((x_end, y))
                path.append((x_start, y))
        path = _pts(path)
        total = _path_length(path)
        if total < END_MARGIN * 4:
            continue

        s_port = END_MARGIN * rng.uniform(1.0, 1.5)
        port = _point_at(path, s_port)

        lo_s = END_MARGIN + 20
        hi_s = total - END_MARGIN - 20
        if hi_s - lo_s < 6 * ARC_SEP:
            continue
        s_green = rng.uniform(lo_s + 2 * ARC_SEP, lo_s + (hi_s - lo_s) * 0.55)
        s_red = s_green + rng.uniform(2.5 * ARC_SEP, hi_s - s_green)

        k_before = rng.choice(ANSWER_VALUES)
        before_gap = (s_green - s_port) / (k_before + 1)
        if before_gap < ARC_SEP:
            continue
        s_black_before = [s_port + before_gap * (i + 1) for i in range(k_before)]

        k_after = rng.choice([0, 1, 2])
        s_black_after = []
        after_lo = s_red + ARC_SEP
        for _ in range(k_after):
            if after_lo + ARC_SEP <= hi_s:
                s_black_after.append(rng.uniform(after_lo, hi_s))

        arc_positions = [s_port, s_green, s_red] + s_black_before + s_black_after
        arc_positions.sort()
        ok = True
        for a, b in zip(arc_positions, arc_positions[1:]):
            if b - a < ARC_SEP:
                ok = False
                break
        if not ok:
            continue

        marks = []
        marks.append({"x": _point_at(path, s_green)[0], "y": _point_at(path, s_green)[1], "color": "green"})
        marks.append({"x": _point_at(path, s_red)[0], "y": _point_at(path, s_red)[1], "color": "red"})
        for s in s_black_before:
            p = _point_at(path, s)
            marks.append({"x": p[0], "y": p[1], "color": "black"})
        for s in s_black_after:
            p = _point_at(path, s)
            marks.append({"x": p[0], "y": p[1], "color": "black"})

        allgood = True
        for m in marks:
            if not (20 < m["x"] < w - 20 and 20 < m["y"] < h - 20):
                allgood = False
                break
        if not allgood:
            continue

        scene = {
            "path": path,
            "port": [float(port[0]), float(port[1])],
            "marks": marks,
            "answer": str(k_before),
        }
        return scene
    raise RuntimeError("could not satisfy path constraints")


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    return _rejection_loop(rng)


def _draw_polyline(draw: ImageDraw.ImageDraw, pts, color, width):
    for a, b in zip(pts, pts[1:]):
        draw.line([a, b], fill=color, width=width, joint="curve")


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", CANVAS, BG)
    draw = ImageDraw.Draw(img)
    path = _pts(scene["path"])
    _draw_polyline(draw, path, COLOR_BLUE, STROKE)
    colors = {"red": COLOR_RED, "green": COLOR_GREEN, "black": COLOR_BLACK}
    for mark in scene["marks"]:
        c = mark["color"]
        r = MARK_R
        px, py = float(mark["x"]), float(mark["y"])
        draw.ellipse([px - r, py - r, px + r, py + r], fill=colors[c])
    px, py = float(scene["port"][0]), float(scene["port"][1])
    draw.ellipse([px - PORT_R, py - PORT_R, px + PORT_R, py + PORT_R], fill=COLOR_ORANGE)
    return img


def _arc_position(scene: dict, pt: tuple[float, float]) -> float:
    best = None
    best_d = float("inf")
    cum = 0.0
    for a, b in zip(_pts(scene["path"]), _pts(scene["path"])[1:]):
        abx, aby = b[0] - a[0], b[1] - a[1]
        seg2 = abx * abx + aby * aby
        t = 0.0 if seg2 == 0 else ((pt[0] - a[0]) * abx + (pt[1] - a[1]) * aby) / seg2
        t = max(0.0, min(1.0, t))
        proj = (a[0] + abx * t, a[1] + aby * t)
        d = math.dist(pt, proj)
        if d < best_d:
            best_d = d
            best = cum + math.dist(a, proj)
        cum += math.dist(a, b)
    return best


def analytic_gold(scene: dict) -> str:
    green = [m for m in scene["marks"] if m["color"] == "green"][0]
    ag = _arc_position(scene, (green["x"], green["y"]))
    cnt = 0
    for m in scene["marks"]:
        if m["color"] == "black":
            am = _arc_position(scene, (m["x"], m["y"]))
            if am < ag:
                cnt += 1
    return str(cnt)


def margin(scene: dict) -> float:
    pos = [scene["port"]] + [[m["x"], m["y"]] for m in scene["marks"]]
    arc = sorted(_arc_position(scene, (p[0], p[1])) for p in pos)
    return min(b - a for a, b in zip(arc, arc[1:]))


def is_quarantined(scene: dict) -> bool:
    w, h = CANVAS
    if margin(scene) < ARC_SEP * 0.8:
        return True
    for m in scene["marks"]:
        if not (20 <= m["x"] <= w - 20 and 20 <= m["y"] <= h - 20):
            return True
    return False


def latent_symmetries(scene: dict) -> list:
    return []
