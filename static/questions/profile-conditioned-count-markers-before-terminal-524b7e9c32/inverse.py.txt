"""Pixel-only inverse arm for the count-markers-before-terminal world.

Reads only the final RGB PNG. Segments the blue path stroke by nearest-colour
labelling, labels its single 4-connected component, then computes a geodesic
distance field from the orange port along the path by a breadth-first search
over the blue pixels. Each marker is mapped to the BFS distance of its nearest
blue pixel (its position along the path). The green terminal's distance is the
cutoff; the decision is the number of black markers whose distance is strictly
less than the green terminal's distance (i.e. encountered before reaching it).

No renderer, prompt, generator, verifier or scene data is imported or read.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_PALETTE = {
    "blue": (40, 90, 210),
    "red": (225, 40, 40),
    "green": (40, 170, 70),
    "black": (30, 30, 30),
    "orange": (245, 150, 20),
    "white": (255, 255, 255),
}
_NAMES = ["blue", "red", "green", "black", "orange", "white"]
_NEAR_R = 8


def _label_pixels(arr: np.ndarray) -> np.ndarray:
    palette = np.array([*_PALETTE.values()], dtype=np.float32)
    dists = np.stack([np.linalg.norm(arr - c, axis=-1) for c in palette], axis=-1)
    return dists.argmin(axis=-1)


def _bfs_distances(blue_mask: np.ndarray, seed_pts: list) -> np.ndarray:
    h, w = blue_mask.shape
    dist = np.full((h, w), -1, dtype=np.int64)
    dq = deque()
    for (y, x) in seed_pts:
        if 0 <= y < h and 0 <= x < w and blue_mask[y, x]:
            dist[y, x] = 0
            dq.append((y, x))
    while dq:
        y, x = dq.popleft()
        nd = dist[y, x] + 1
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < h and 0 <= nx < w and blue_mask[ny, nx] and dist[ny, nx] == -1:
                dist[ny, nx] = nd
                dq.append((ny, nx))
    return dist


def _port_seed_blue(labels, idx):
    orange = labels == idx["orange"]
    h, w = orange.shape
    seeds = []
    for y in range(h):
        for x in range(w):
            if orange[y, x]:
                for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                    if 0 <= ny < h and 0 <= nx < w and labels[ny, nx] == idx["blue"]:
                        seeds.append((ny, nx))
    return seeds


def _marker_nearest_blue(labels, blue, name):
    mask = labels == name
    ys, xs = np.nonzero(mask)
    if len(ys) == 0:
        return None
    cy, cx = int(round(float(ys.mean()))), int(round(float(xs.mean())))
    h, w = labels.shape
    best = None
    best_d = float("inf")
    tys, txs = np.nonzero(blue)
    for yy, xx in zip(tys.tolist(), txs.tolist()):
        d = (yy - cy) ** 2 + (xx - cx) ** 2
        if d < best_d:
            best_d = d
            best = (yy, xx)
    return best


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label_pixels(arr)
    name = {n: i for i, n in enumerate(_NAMES)}
    blue = labels == name["blue"]

    seeds = _port_seed_blue(labels, name)
    if not seeds:
        raise ValueError("port does not touch the blue path")
    dist = _bfs_distances(blue, seeds)
    if not (dist >= 0).any():
        raise ValueError("no blue reachable from port")

    green_near = _marker_nearest_blue(labels, blue, name["green"])
    if green_near is None:
        raise ValueError("green terminal not found")
    yy, xx = green_near
    if dist[yy, xx] < 0:
        raise ValueError("green terminal not on reachable path")
    g_dist = dist[yy, xx]

    black = labels == name["black"]
    h, w = blue.shape
    count = 0
    for comp_y, comp_x in _label_components_blobs(black):
        best_d = float("inf")
        best = None
        for yy, xx in zip(np.asarray(comp_y).tolist(), np.asarray(comp_x).tolist()):
            for dy in range(-_NEAR_R, _NEAR_R + 1):
                for dx in range(-_NEAR_R, _NEAR_R + 1):
                    ny, nx = yy + dy, xx + dx
                    if 0 <= ny < h and 0 <= nx < w and blue[ny, nx]:
                        dd = (ny - yy) ** 2 + (nx - xx) ** 2
                        if dd < best_d:
                            best_d = dd
                            best = (ny, nx)
        if best is None:
            continue
        yy, xx = best
        if 0 < dist[yy, xx] < g_dist:
            count += 1
    return str(count)


def _label_components_blobs(mask: np.ndarray):
    h, w = mask.shape
    seen = np.zeros((h, w), dtype=bool)
    blobs = []
    for y in range(h):
        for x in range(w):
            if mask[y, x] and not seen[y, x]:
                stack = [(y, x)]
                seen[y, x] = True
                ys, xs = [], []
                while stack:
                    yy, xx = stack.pop()
                    ys.append(yy)
                    xs.append(xx)
                    for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                        if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                            seen[ny, nx] = True
                            stack.append((ny, nx))
                blobs.append((ys, xs))
    return blobs
