"""Independent image-only inverse for the three-chamber itinerary world."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

REFERENCES = {
    "cyan": np.asarray((24, 169, 201), dtype=np.int16),
    "amber": np.asarray((230, 151, 35), dtype=np.int16),
    "rose": np.asarray((216, 71, 118), dtype=np.int16),
}
TARGET = (0, 1, 2)


def _components(mask: np.ndarray) -> list[tuple[int, float, float]]:
    """Return (area, x centroid, y centroid) for four-connected components."""
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[tuple[int, float, float]] = []
    for start_y, start_x in np.argwhere(mask):
        sy, sx = int(start_y), int(start_x)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= 70:
            output.append((len(xs), float(np.mean(xs)), float(np.mean(ys))))
    return output


def _lane_levels(points: list[tuple[float, float]]) -> list[float]:
    clusters: list[list[float]] = []
    for value in sorted(y for _, y in points):
        if not clusters or value - float(np.mean(clusters[-1])) > 45.0:
            clusters.append([value])
        else:
            clusters[-1].append(value)
    if len(clusters) != 3:
        raise ValueError(f"could not recover three visible chamber levels: {len(clusters)}")
    levels = [float(np.mean(group)) for group in clusters]
    if min(np.diff(levels)) < 80.0:
        raise ValueError("chamber levels are not cleanly separated")
    return levels


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    by_color: dict[str, list[tuple[float, float]]] = {}
    all_points: list[tuple[float, float]] = []
    for color, reference in REFERENCES.items():
        distance = np.max(np.abs(array - reference), axis=2)
        components = _components(distance <= 3)
        points = [(x, y) for area, x, y in components if area >= 70]
        by_color[color] = points
        all_points.extend(points)

    if len(all_points) < 3:
        raise ValueError("too few colored token marks")
    levels = _lane_levels(all_points)
    matches: list[str] = []
    for color, points in by_color.items():
        if len(points) != 3:
            continue
        ordered = sorted(points, key=lambda point: point[0])
        if min(np.diff([point[0] for point in ordered])) < 80.0:
            continue
        path = tuple(int(np.argmin([abs(y - level) for level in levels])) for _, y in ordered)
        if path == TARGET:
            matches.append(color)
    if len(matches) != 1:
        raise ValueError(f"image does not expose one complete target itinerary: {matches}")
    return matches[0]
