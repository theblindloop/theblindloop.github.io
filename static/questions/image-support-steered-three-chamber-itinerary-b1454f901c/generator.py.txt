"""Deterministic renderer and analytic arm for three-chamber itineraries."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 768, 576
COLORS = ("cyan", "amber", "rose")
RGB = {
    "cyan": (24, 169, 201),
    "amber": (230, 151, 35),
    "rose": (216, 71, 118),
}
TARGET = (0, 1, 2)
PANEL_CENTERS = (240, 432, 624)
LANE_CENTERS = (180, 324, 468)
SLOT_OFFSETS = (-20, 0, 20)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced scene with one unique TOP -> MIDDLE -> BOTTOM token."""
    rng = random.Random(int(seed))
    answer = COLORS[int(seed) % len(COLORS)]
    distractors = [color for color in COLORS if color != answer]
    rng.shuffle(distractors)

    paths: dict[str, list[int]] = {answer: list(TARGET)}
    # Every individual panel is locally ambiguous: distractor 1 shares the
    # answer's first and final chamber; distractor 2 shares its middle chamber.
    paths[distractors[0]] = [0, rng.choice((0, 2)), 2]
    paths[distractors[1]] = [rng.choice((1, 2)), 1, rng.choice((0, 1))]

    slot_order = list(COLORS)
    rng.shuffle(slot_order)
    return {
        "paths": paths,
        "slot_order": slot_order,
        "theme": rng.randrange(2),
        "offset_x": rng.choice((-3, 0, 3)),
        "offset_y": rng.choice((-3, 0, 3)),
    }


def _matching_colors(paths: dict[str, list[int]]) -> list[str]:
    return sorted(color for color, path in paths.items() if tuple(path) == TARGET)


def analytic_gold(scene: dict[str, Any]) -> str:
    matches = _matching_colors(scene["paths"])
    if len(matches) != 1:
        raise ValueError(f"expected one matching itinerary, found {matches}")
    return matches[0]


def margin(scene: dict[str, Any]) -> float:
    distances = sorted(
        sum(int(value != wanted) for value, wanted in zip(path, TARGET, strict=True))
        for path in scene["paths"].values()
    )
    return float(distances[1] - distances[0]) if len(distances) >= 2 else 0.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    paths = scene["paths"]
    return (
        set(paths) != set(COLORS)
        or any(len(path) != 3 or any(value not in (0, 1, 2) for value in path) for path in paths.values())
        or len(_matching_colors(paths)) != 1
        or margin(scene) <= 0.0
    )


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # All scene fields are directly rendered; there is no pixel-identical alias.
    return []


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    path = (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
        if bold
        else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
    )
    try:
        return ImageFont.truetype(path, size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict[str, Any]) -> Image.Image:
    theme = int(scene["theme"])
    ox, oy = int(scene["offset_x"]), int(scene["offset_y"])
    paper = (248, 246, 239) if theme == 0 else (242, 247, 250)
    card = (255, 255, 255)
    lanes = (
        ((239, 245, 248), (247, 242, 232), (243, 239, 247))
        if theme == 0
        else ((245, 241, 235), (235, 245, 241), (244, 238, 240))
    )
    image = Image.new("RGB", (WIDTH, HEIGHT), paper)
    draw = ImageDraw.Draw(image)
    title_font = _font(20, bold=True)
    label_font = _font(12, bold=True)
    draw.text((28, 24), "THREE SNAPSHOTS", fill=(39, 47, 54), font=title_font)
    draw.text((550, 30), "READ LEFT TO RIGHT", fill=(65, 72, 79), font=label_font)

    slots = {color: SLOT_OFFSETS[scene["slot_order"].index(color)] for color in COLORS}
    lane_names = ("TOP", "MIDDLE", "BOTTOM")
    for panel_index, base_cx in enumerate(PANEL_CENTERS):
        cx = base_cx + ox
        left, right = cx - 62, cx + 62
        top, bottom = 74 + oy, 542 + oy
        draw.rounded_rectangle((left, top, right, bottom), radius=10, fill=card, outline=(54, 62, 69), width=3)
        # One, two, and three header pips visibly reinforce panel order.
        pip_y = 101 + oy
        for pip_index in range(panel_index + 1):
            px = cx + (pip_index - panel_index / 2.0) * 13
            draw.ellipse((px - 4, pip_y - 4, px + 4, pip_y + 4), fill=(45, 52, 59))

        for lane_index, lane_y in enumerate(LANE_CENTERS):
            y = lane_y + oy
            draw.rounded_rectangle(
                (left + 7, y - 52, right - 7, y + 52),
                radius=7,
                fill=lanes[lane_index],
                outline=(151, 158, 164),
                width=2,
            )
            label = lane_names[lane_index]
            draw.text((left + 11, y - 48), label, fill=(91, 98, 104), font=label_font)

        for color in COLORS:
            chamber = int(scene["paths"][color][panel_index])
            x = cx + slots[color]
            y = LANE_CENTERS[chamber] + oy + 15
            radius = 9
            draw.ellipse(
                (x - radius, y - radius, x + radius, y + radius),
                fill=RGB[color],
                outline=(32, 37, 42),
                width=2,
            )
    return image
