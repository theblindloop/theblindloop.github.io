"""Independent image-only inverse for same-face marker separation."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _marker_center(mask: np.ndarray, label: str) -> tuple[int, int]:
    points = np.argwhere(mask)
    if len(points) < 30:
        raise ValueError(f"{label} marker is not recoverable from pixels")
    y, x = points.mean(axis=0)
    iy, ix = int(round(float(y))), int(round(float(x)))
    if not mask[iy, ix]:
        distances = (points[:, 0] - y) ** 2 + (points[:, 1] - x) ** 2
        iy, ix = points[int(np.argmin(distances))]
    return int(iy), int(ix)


def _same_open_component(wall: np.ndarray, start: tuple[int, int], goal: tuple[int, int]) -> bool:
    height, width = wall.shape
    start_index = start[0] * width + start[1]
    goal_index = goal[0] * width + goal[1]
    if wall[start] or wall[goal]:
        raise ValueError("marker touches a wall in the raster")
    visited = bytearray(height * width)
    visited[start_index] = 1
    queue = deque([start_index])
    while queue:
        index = queue.popleft()
        if index == goal_index:
            return True
        row, column = divmod(index, width)
        if row and not wall[row - 1, column]:
            other = index - width
            if not visited[other]:
                visited[other] = 1
                queue.append(other)
        if row + 1 < height and not wall[row + 1, column]:
            other = index + width
            if not visited[other]:
                visited[other] = 1
                queue.append(other)
        if column and not wall[row, column - 1]:
            other = index - 1
            if not visited[other]:
                visited[other] = 1
                queue.append(other)
        if column + 1 < width and not wall[row, column + 1]:
            other = index + 1
            if not visited[other]:
                visited[other] = 1
                queue.append(other)
    return False


def decision_from_image(image: Image.Image) -> str:
    """Recover the yes/no answer from only the final PNG."""
    # Use a signed type before adding thresholds; uint8 arithmetic would wrap
    # the warm background values and mistake large parts of the canvas for ink.
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    channels = array.astype(np.int16)
    red, green, blue = (channels[..., 0], channels[..., 1], channels[..., 2])
    red_marker = (red > 125) & (red > green + 40) & (red > blue + 30)
    blue_marker = (blue > 125) & (blue > red + 35) & (blue > green + 15)
    # Walls are dark and nearly neutral; marker colors and the warm background
    # are intentionally outside this mask.  The thick core survives LANCZOS.
    wall = (array.max(axis=2) < 135) & ((array.max(axis=2) - array.min(axis=2)) < 38)
    first = _marker_center(red_marker, "red")
    second = _marker_center(blue_marker, "blue")
    return "yes" if _same_open_component(wall, first, second) else "no"
