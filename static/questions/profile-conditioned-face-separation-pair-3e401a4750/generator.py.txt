"""Renderer and analytic arm for the same-face marker question.

The scene contains only visible ingredients: closed polygonal walls, two marker
centres, and rendering style.  Walls are laminar (nested or disjoint), so the
connected component of a marker in their complement is represented analytically
by the ordered tuple of contours containing it.  The public inverse does not
use this representation; it flood-fills the rasterized wall complement.
"""

from __future__ import annotations

import copy
import math
import random
from typing import Iterable, Sequence

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 384
SUPERSAMPLE = 3
BACKGROUND = (249, 247, 242)
DEFAULT_WALL = (38, 46, 55)
RED = (215, 55, 70)
BLUE = (48, 98, 214)
WALL_PALETTE = ((32, 40, 49), (38, 46, 55), (47, 53, 62), (28, 38, 48))


def _rect(x0: float, y0: float, x1: float, y1: float) -> list[list[float]]:
    return [[float(x0), float(y0)], [float(x1), float(y0)], [float(x1), float(y1)], [float(x0), float(y1)]]


def _transform(polygons: list[list[list[float]]], rng: random.Random) -> list[list[list[float]]]:
    sx = rng.uniform(0.94, 1.055)
    sy = rng.uniform(0.94, 1.055)
    dx = rng.uniform(-8.0, 8.0)
    dy = rng.uniform(-7.0, 7.0)
    cx, cy = WIDTH / 2.0, HEIGHT / 2.0
    result = []
    for polygon in polygons:
        result.append([
            [round(cx + (float(x) - cx) * sx + dx, 2), round(cy + (float(y) - cy) * sy + dy, 2)]
            for x, y in polygon
        ])
    return result


def _make_barriers(rng: random.Random) -> list[list[list[float]]]:
    """Create three non-touching rectangles in one of four laminar layouts."""

    layout = rng.randrange(4)
    if layout == 0:
        # Three nested faces, with deliberately unequal margins.
        base = [_rect(45, 36, 467, 348), _rect(126, 91, 393, 300), _rect(211, 145, 316, 249)]
    elif layout == 1:
        # Two disjoint outer contours, with one child in the left contour.
        base = [_rect(31, 64, 220, 331), _rect(292, 53, 481, 340), _rect(83, 136, 170, 263)]
    elif layout == 2:
        # One outer face with two separate child contours.
        base = [_rect(39, 32, 472, 351), _rect(83, 92, 222, 299), _rect(287, 79, 429, 311)]
    else:
        # An offset nested chain, visually unlike concentric rings.
        base = [_rect(59, 47, 454, 338), _rect(119, 77, 390, 310), _rect(184, 128, 325, 263)]
    return _transform(base, rng)


def _segment_distance(point: Sequence[float], a: Sequence[float], b: Sequence[float]) -> float:
    px, py = float(point[0]), float(point[1])
    ax, ay = float(a[0]), float(a[1])
    bx, by = float(b[0]), float(b[1])
    dx, dy = bx - ax, by - ay
    denom = dx * dx + dy * dy
    if denom == 0:
        return math.hypot(px - ax, py - ay)
    t = max(0.0, min(1.0, ((px - ax) * dx + (py - ay) * dy) / denom))
    return math.hypot(px - (ax + t * dx), py - (ay + t * dy))


def _point_in_polygon(point: Sequence[float], polygon: Sequence[Sequence[float]]) -> bool:
    x, y = float(point[0]), float(point[1])
    inside = False
    previous = polygon[-1]
    for current in polygon:
        x0, y0 = float(previous[0]), float(previous[1])
        x1, y1 = float(current[0]), float(current[1])
        if (y0 > y) != (y1 > y):
            crossing_x = (x1 - x0) * (y - y0) / (y1 - y0) + x0
            if x < crossing_x:
                inside = not inside
        previous = current
    return inside


def _face_signature(point: Sequence[float], barriers: Sequence[Sequence[Sequence[float]]]) -> tuple[bool, ...]:
    return tuple(_point_in_polygon(point, polygon) for polygon in barriers)


def _clearance(point: Sequence[float], barriers: Sequence[Sequence[Sequence[float]]]) -> float:
    distances = []
    for polygon in barriers:
        for index, vertex in enumerate(polygon):
            distances.append(_segment_distance(point, vertex, polygon[(index + 1) % len(polygon)]))
    distances.extend((float(point[0]), float(point[1]), WIDTH - float(point[0]), HEIGHT - float(point[1])))
    return min(distances)


def _choose_markers(
    barriers: list[list[list[float]]], rng: random.Random, same: bool
) -> tuple[list[float], list[float]]:
    groups: dict[tuple[bool, ...], list[list[float]]] = {}
    for y in range(24, HEIGHT - 23, 8):
        for x in range(24, WIDTH - 23, 8):
            point = [float(x), float(y)]
            if _clearance(point, barriers) >= 27.0:
                groups.setdefault(_face_signature(point, barriers), []).append(point)

    choices = [values for values in groups.values() if len(values) >= 2]
    if same:
        rng.shuffle(choices)
        for values in choices:
            for _ in range(80):
                first, second = rng.sample(values, 2)
                if math.dist(first, second) >= 72.0:
                    return first, second
    else:
        keys = list(groups)
        rng.shuffle(keys)
        for _ in range(300):
            key_a, key_b = rng.sample(keys, 2)
            first = rng.choice(groups[key_a])
            second = rng.choice(groups[key_b])
            if math.dist(first, second) >= 72.0:
                return first, second

    # The layouts above have generous open faces; these are only defensive
    # fallbacks if a future raster-sized change makes a grid choice impossible.
    if same:
        return [24.0, 24.0], [WIDTH - 24.0, HEIGHT - 24.0]
    return [24.0, 24.0], [WIDTH / 2.0, HEIGHT / 2.0]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    barriers = _make_barriers(rng)
    same = (int(seed) % 2 == 0)
    marker_a, marker_b = _choose_markers(barriers, rng, same)
    if rng.random() < 0.5:
        palette = {"a": list(RED), "b": list(BLUE)}
    else:
        palette = {"a": list(BLUE), "b": list(RED)}
    return {
        "barriers": barriers,
        "marker_a": marker_a,
        "marker_b": marker_b,
        "palette": palette,
        "wall_rgb": list(rng.choice(WALL_PALETTE)),
        "wall_width": rng.choice([9, 10, 11, 12]),
        "marker_radius": rng.choice([10, 11, 12, 13]),
    }


def _scaled_points(polygon: Sequence[Sequence[float]]) -> list[tuple[int, int]]:
    return [(round(float(x) * SUPERSAMPLE), round(float(y) * SUPERSAMPLE)) for x, y in polygon]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH * SUPERSAMPLE, HEIGHT * SUPERSAMPLE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    wall = tuple(int(value) for value in scene.get("wall_rgb", DEFAULT_WALL))
    width = max(1, int(scene.get("wall_width", 10)) * SUPERSAMPLE)
    for polygon in scene["barriers"]:
        points = _scaled_points(polygon)
        draw.line(points + [points[0]], fill=wall, width=width, joint="curve")

    radius = float(scene.get("marker_radius", 11)) * SUPERSAMPLE
    palette = scene.get("palette", {"a": list(RED), "b": list(BLUE)})
    for key, point in (("a", scene["marker_a"]), ("b", scene["marker_b"])):
        cx, cy = float(point[0]) * SUPERSAMPLE, float(point[1]) * SUPERSAMPLE
        color = tuple(int(value) for value in palette[key])
        draw.ellipse((round(cx - radius), round(cy - radius), round(cx + radius), round(cy + radius)), fill=color)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    first = _face_signature(scene["marker_a"], scene["barriers"])
    second = _face_signature(scene["marker_b"], scene["barriers"])
    return "yes" if first == second else "no"


def margin(scene: dict) -> float:
    clearance = min(
        _clearance(scene["marker_a"], scene["barriers"]),
        _clearance(scene["marker_b"], scene["barriers"]),
    )
    # This is the visible slack after allowing for the marker radius and half
    # the wall stroke.  It is also the runner-up/ambiguity margin in the ledger.
    slack = clearance - float(scene.get("marker_radius", 11)) - float(scene.get("wall_width", 10)) / 2.0
    return round(float(slack), 3)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["barriers"] = list(reversed(twin["barriers"]))
    return [("reverse_same_color_contour_order", twin)]
