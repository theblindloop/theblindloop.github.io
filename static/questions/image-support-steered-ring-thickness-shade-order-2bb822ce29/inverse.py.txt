"""Pixel-only inverse arm for the ring-thickness / ring-shade binding world.

Input is a single PIL image and nothing else: no latent scene, no renderer, no
prompt text, no manifest, no file name.  The arm

1. segments the near-black ink and locates the one small round centre pip,
2. walks many rays outwards from the pip centre and records the flat colour
   runs it crosses,
3. reconstructs the four concentric rings as (median radial thickness, modal
   fill colour) pairs, and
4. reports whether the thickest ring is nearer the centre than the darkest one.

Anything it cannot resolve - a missing or damaged pip, rays that disagree about
the ring structure, a thickness tie, a shade tie or a coincident pair - is
reported as ``unreadable`` rather than guessed.
"""

from __future__ import annotations

import math
from collections import Counter

import numpy as np
from PIL import Image

ORACLE_VERSION = "ring_thickness_shade_order-oracle-0.1.0"

UNREADABLE = "unreadable"
ALLOWED = ("nearer", "farther")

BLACK_MAX_CHANNEL = 60      # a pixel is pip ink only if every channel is darker
PIP_AREA_MIN, PIP_AREA_MAX = 180, 1400
PIP_ROUNDNESS = 0.45        # area / bbox area lower bound for a round pip
MAX_BLACK_PX = 4000         # a clean sheet carries only the ~450 px pip
RAYS = 96
MIN_GOOD_RAYS = 40
EXPECTED_RINGS = 4
MIN_RING_PX = 5
MAX_SCAN = 220              # ray length: comfortably past any legible button
TIE_PX = 1.0                # thickness ties are refused, not guessed
TIE_LUM = 6.0               # shade ties are refused, not guessed


def _luminance(color) -> float:
    r, g, b = (float(channel) for channel in color)
    return 0.299 * r + 0.587 * g + 0.114 * b


def _pip_center(mask: np.ndarray) -> tuple[float, float] | None:
    """Centre of the unique small round near-black component, if there is one."""
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[float, float]] = []
    for sy, sx in zip(*np.nonzero(mask)):
        if seen[sy, sx]:
            continue
        stack = [(int(sy), int(sx))]
        seen[sy, sx] = True
        pixels = []
        while stack:
            cy, cx = stack.pop()
            pixels.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        area = len(pixels)
        if not PIP_AREA_MIN <= area <= PIP_AREA_MAX:
            continue
        ys = [p[0] for p in pixels]
        xs = [p[1] for p in pixels]
        bh = max(ys) - min(ys) + 1
        bw = max(xs) - min(xs) + 1
        if abs(bh - bw) > max(4, 0.25 * max(bh, bw)):
            continue
        if area < PIP_ROUNDNESS * bh * bw:
            continue
        found.append((sum(ys) / area, sum(xs) / area))
        if len(found) > 1:
            return None
    if len(found) != 1:
        return None
    return found[0]


def _ray_runs(array: np.ndarray, cy: float, cx: float, angle: float, background) -> list | None:
    """Flat colour runs met along one ray, after the pip and before the paper."""
    height, width = array.shape[:2]
    dy, dx = math.sin(angle), math.cos(angle)
    runs: list[list] = []
    left_pip = False
    for step in range(MAX_SCAN):
        y = int(round(cy + dy * step))
        x = int(round(cx + dx * step))
        if not (0 <= y < height and 0 <= x < width):
            return None
        color = tuple(int(v) for v in array[y, x])
        dark = max(color) < BLACK_MAX_CHANNEL
        if not left_pip:
            if dark:
                continue
            left_pip = True
        if dark:
            return None                          # ink where a ring should be
        if color == background:
            return runs if runs else None        # reached the paper
        if runs and runs[-1][0] == color:
            runs[-1][1] += 1
        else:
            runs.append([color, 1])
    return None


def decision_from_image(image: Image.Image) -> str:
    """Recover the radial order of the thickest and the darkest ring."""
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or min(array.shape[:2]) < 64:
        return UNREADABLE
    black = array.max(axis=2) < BLACK_MAX_CHANNEL
    total_black = int(black.sum())
    if not PIP_AREA_MIN <= total_black <= MAX_BLACK_PX:
        return UNREADABLE
    center = _pip_center(black)
    if center is None:
        return UNREADABLE
    cy, cx = center

    border = np.concatenate(
        (
            array[0, :].reshape(-1, 3),
            array[-1, :].reshape(-1, 3),
            array[:, 0].reshape(-1, 3),
            array[:, -1].reshape(-1, 3),
        )
    )
    values, counts = np.unique(border, axis=0, return_counts=True)
    background = tuple(int(v) for v in values[int(np.argmax(counts))])

    good: list[list] = []
    for index in range(RAYS):
        runs = _ray_runs(array, cy, cx, 2.0 * math.pi * index / RAYS, background)
        if runs is None or len(runs) != EXPECTED_RINGS:
            continue
        if min(run[1] for run in runs) < MIN_RING_PX:
            continue
        if len({run[0] for run in runs}) != EXPECTED_RINGS:
            continue
        good.append(runs)
    if len(good) < MIN_GOOD_RAYS:
        return UNREADABLE

    colors: list[tuple[int, int, int]] = []
    for band in range(EXPECTED_RINGS):
        tally = Counter(runs[band][0] for runs in good)
        color, hits = tally.most_common(1)[0]
        if hits < 0.75 * len(good):
            return UNREADABLE
        colors.append(color)
    if len(set(colors)) != EXPECTED_RINGS:
        return UNREADABLE

    # Refine the radial thicknesses from filled areas inside the button's
    # neighbourhood.  The area enclosed by the outer edge of ring i is the pip
    # plus every ring up to i, so its outer radius is sqrt(area / pi) and the
    # thickness of ring i is the difference of consecutive outer radii.  This
    # uses thousands of pixels per ring instead of one run per ray, and the same
    # half-pixel edge bias appears in every radius, so it cancels in the
    # thickness comparison.
    y0 = max(0, int(cy) - MAX_SCAN)
    y1 = min(array.shape[0], int(cy) + MAX_SCAN + 1)
    x0 = max(0, int(cx) - MAX_SCAN)
    x1 = min(array.shape[1], int(cx) + MAX_SCAN + 1)
    patch = array[y0:y1, x0:x1].astype(np.int32)
    flat = patch[:, :, 0] * 65536 + patch[:, :, 1] * 256 + patch[:, :, 2]
    enclosed = float(total_black)
    radii: list[float] = [math.sqrt(enclosed / math.pi)]
    for color in colors:
        key = color[0] * 65536 + color[1] * 256 + color[2]
        area = float(np.count_nonzero(flat == key))
        if area < 40.0:
            return UNREADABLE
        enclosed += area
        radii.append(math.sqrt(enclosed / math.pi))
    widths = [radii[index + 1] - radii[index] for index in range(EXPECTED_RINGS)]
    if min(widths) < MIN_RING_PX:
        return UNREADABLE

    order = sorted(range(EXPECTED_RINGS), key=lambda i: widths[i], reverse=True)
    if widths[order[0]] - widths[order[1]] < TIE_PX:
        return UNREADABLE
    thick = order[0]

    lums = [_luminance(color) for color in colors]
    shade_order = sorted(range(EXPECTED_RINGS), key=lambda i: lums[i])
    if lums[shade_order[1]] - lums[shade_order[0]] < TIE_LUM:
        return UNREADABLE
    dark = shade_order[0]

    if thick == dark:
        return UNREADABLE
    answer = "nearer" if thick < dark else "farther"
    return answer if answer in ALLOWED else UNREADABLE
