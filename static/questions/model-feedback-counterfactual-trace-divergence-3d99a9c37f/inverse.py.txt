"""Independent pixel-only inverse arm for the trace-divergence world.

Given only the final PNG this module segments every slanted bar by exact ink
colour, reads each bar's slant from the sign of its principal axis, recovers
the lattice by one-dimensional clustering of bar centroids, locates the black
release dot, and then replays the drop twice: once on the board as segmented
and once with the red bar's slant inverted. It returns the colour name of the
single tagged bar touched by the second run and not by the first.

It imports no renderer / prompts / generate / verify / analytic-gold code and
never receives the latent scene.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_TAGS = {
    "blue": (35, 85, 205),
    "green": (25, 145, 65),
    "orange": (245, 150, 25),
    "purple": (135, 55, 190),
}
_PLAIN_RGB = (60, 66, 78)
_RED_RGB = (200, 40, 50)
_BALL_RGB = (0, 0, 0)
_TOL = 40.0
_FALLBACK = "blue"


def _near(arr: np.ndarray, rgb, tol: float = _TOL) -> np.ndarray:
    diff = arr - np.asarray(rgb, dtype=float)
    return np.sqrt((diff * diff).sum(axis=2)) < tol


def _components(mask: np.ndarray, min_pixels: int = 40) -> list:
    """Label 8-connected components of a boolean mask without SciPy."""
    height, width = mask.shape
    seen = np.zeros((height, width), dtype=bool)
    out = []
    ys, xs = np.nonzero(mask)
    for sy, sx in zip(ys.tolist(), xs.tolist()):
        if seen[sy, sx]:
            continue
        stack = [(sy, sx)]
        seen[sy, sx] = True
        pixels = []
        while stack:
            y, x = stack.pop()
            pixels.append((y, x))
            for ny in (y - 1, y, y + 1):
                if ny < 0 or ny >= height:
                    continue
                for nx in (x - 1, x, x + 1):
                    if 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        if len(pixels) >= min_pixels:
            out.append(np.asarray(pixels, dtype=float))
    return out


def _slant(pixels: np.ndarray) -> int:
    """-1 for a bar leaning like '/', +1 for a bar leaning like '\\'."""
    centred = pixels - pixels.mean(axis=0)
    cov = np.cov(centred, rowvar=False)
    eigvals, eigvecs = np.linalg.eigh(np.atleast_2d(cov))
    vec = eigvecs[:, int(np.argmax(eigvals))]
    dy, dx = float(vec[0]), float(vec[1])
    if dx < 0:
        dx, dy = -dx, -dy
    return -1 if dy < 0 else 1


def _cluster(values: list, gap: float) -> list:
    """Sorted cluster centres of 1-D values split at gaps wider than ``gap``."""
    ordered = sorted(values)
    groups = [[ordered[0]]]
    for value in ordered[1:]:
        if value - groups[-1][-1] > gap:
            groups.append([value])
        else:
            groups[-1].append(value)
    return [float(sum(g) / len(g)) for g in groups]


def _run(bars: list, cols: int, rows: int, start: int, flip: int) -> list:
    at_cell = {(b["row"], b["col"]): i for i, b in enumerate(bars)}
    col = int(start)
    met = []
    for row in range(rows):
        index = at_cell.get((row, col))
        if index is None:
            continue
        delta = bars[index]["slant"]
        if index == flip:
            delta = -delta
        met.append(index)
        col += delta
        if col < 0 or col >= cols:
            break
    return met


def decision_from_image(image: Image.Image) -> str:
    """Return the tag colour touched only by the edited (red-flipped) run."""
    arr = np.asarray(image.convert("RGB")).astype(float)

    pieces = []
    for label, rgb in [("plain", _PLAIN_RGB), ("red", _RED_RGB)] + [
        (name, value) for name, value in _TAGS.items()
    ]:
        for pixels in _components(_near(arr, rgb)):
            pieces.append((label, pixels))
    if len(pieces) < 6:
        return _FALLBACK

    raw = []
    for label, pixels in pieces:
        raw.append(
            {
                "label": label,
                "cx": float(pixels[:, 1].mean()),
                "cy": float(pixels[:, 0].mean()),
                "slant": _slant(pixels),
            }
        )

    col_centres = _cluster([item["cx"] for item in raw], 45.0)
    row_centres = _cluster([item["cy"] for item in raw], 40.0)
    if len(col_centres) < 2 or len(row_centres) < 2:
        return _FALLBACK

    bars = []
    for item in raw:
        bars.append(
            {
                "label": item["label"],
                "slant": item["slant"],
                "col": int(np.argmin([abs(item["cx"] - c) for c in col_centres])),
                "row": int(np.argmin([abs(item["cy"] - c) for c in row_centres])),
            }
        )

    ball = _near(arr, _BALL_RGB)
    ys, xs = np.nonzero(ball)
    if xs.size < 50:
        return _FALLBACK
    ball_x = float(xs.mean())
    start = int(np.argmin([abs(ball_x - c) for c in col_centres]))

    red = [i for i, b in enumerate(bars) if b["label"] == "red"]
    if len(red) != 1:
        return _FALLBACK

    n_cols = len(col_centres)
    n_rows = len(row_centres)
    factual = set(_run(bars, n_cols, n_rows, start, -1))
    edited = set(_run(bars, n_cols, n_rows, start, red[0]))
    gained = edited - factual

    hits = sorted(bars[i]["label"] for i in gained if bars[i]["label"] in _TAGS)
    if len(hits) != 1:
        return _FALLBACK
    return hits[0]
