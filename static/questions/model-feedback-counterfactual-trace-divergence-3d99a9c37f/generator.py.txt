"""Counterfactual trace-divergence world: the answer is a *contrast between two
runs of the same mechanism*, not the outcome of either run.

A ball is released from a black dot at the top of a lattice of slanted bars.
Falling straight down it slides to the lower end of every slanted bar whose
horizontal span it meets, and passes straight through empty lattice cells.

Exactly one bar is red; four other bars are drawn in four distinct hues. The
decision is not where the ball ends up in any world. It is the colour of the
single tagged bar that belongs to the *symmetric difference* of two
trajectories: the run on the board as drawn, and the run on the board in which
the red bar alone is re-leaned the opposite way. Precisely, the answer is the
tagged bar that the second run touches and the first run does not.

Every scene draws one tagged bar in each of the four contrast cells:

  gained  -- touched only by the edited run                (the gold)
  lost    -- touched only by the drawn run                 (contrast reversed)
  both    -- touched by both runs                          (one-world reading)
  neither -- touched by no run                             (mis-trace)

so each wrong option is the endpoint of one specific reasoning error, and no
single simulation of either board can select an answer.

The renderer owns scene sampling, rasterisation and the analytic gold. The
independent pixel-only inverse arm (``oracle.py``) recovers the same colour
name from the final PNG alone.
"""

from __future__ import annotations

from PIL import Image, ImageDraw

WIDTH = 580
HEIGHT = 662

COLS = 5
ROWS = 6
X0 = 90
XSTEP = 100
Y0 = 118
YSTEP = 88

BAR_W = 7
BALL_Y = 44
BALL_R = 12
RAIL_TOP = 614
RAIL_H = 14

BACKGROUND = (255, 255, 255)
BAR_COLOR = (60, 66, 78)
RED_COLOR = (200, 40, 50)
BALL_COLOR = (0, 0, 0)
RAIL_COLOR = (150, 150, 150)

TAG_PALETTE = {
    "blue": (35, 85, 205),
    "green": (25, 145, 65),
    "orange": (245, 150, 25),
    "purple": (135, 55, 190),
}
TAG_NAMES = ("blue", "green", "orange", "purple")
ROLES = ("gained", "lost", "both", "neither")

FILL_PROB = 0.5
DX_JITTER = 10
DY_JITTER = 6
HALF_LO = 20
HALF_HI = 26
MIN_VISITED = 3
QUARANTINE_PX = 11.0


def _col_x(col: int) -> float:
    return float(X0 + XSTEP * col)


def _row_y(row: int) -> float:
    return float(Y0 + YSTEP * row)


def _grid_of(bars: list) -> dict:
    return {(int(b[0]), int(b[1])): i for i, b in enumerate(bars)}


def _walk(bars: list, start: int, flip_index: int):
    """Drop the ball, flipping the bar at ``flip_index`` (-1 for none).

    Returns ``(met, ok)`` where ``met`` is the list of bar indices touched in
    order. ``ok`` is False when the ball leaves the lattice or when a
    deflection would land it on another bar inside the same row.
    """
    at_cell = _grid_of(bars)
    col = int(start)
    met: list[int] = []
    if col < 0 or col >= COLS:
        return met, False
    for row in range(ROWS):
        index = at_cell.get((row, col))
        if index is None:
            continue
        delta = int(bars[index][2])
        if index == flip_index:
            delta = -delta
        met.append(index)
        col += delta
        if col < 0 or col >= COLS:
            return met, False
        if (row, col) in at_cell:
            return met, False
    return met, True


def _walk_grid(grid: dict, start: int, flip_cell):
    """Sampling-time walk over a cell dictionary of ``cell -> params``."""
    col = int(start)
    visited = []
    for row in range(ROWS):
        cell = (row, col)
        bar = grid.get(cell)
        if bar is None:
            continue
        delta = -bar[0] if cell == flip_cell else bar[0]
        visited.append(cell)
        col += delta
        if col < 0 or col >= COLS:
            return visited, -1, "out", None
        if (row, col) in grid:
            return visited, -1, "collide", (row, col)
    return visited, col, "ok", None


def _repair(grid: dict, ball_col: int, flip_cell, protect: set):
    """Delete bars that make a deflection land on a same-row bar."""
    for _ in range(40):
        _v, _b, status, offender = _walk_grid(grid, ball_col, flip_cell)
        if status == "out":
            return False
        if status == "ok":
            return True
        if offender in protect:
            return False
        del grid[offender]
    return False


def _covers_lattice(grid: dict) -> bool:
    rows = {cell[0] for cell in grid}
    cols = {cell[1] for cell in grid}
    return len(rows) == ROWS and len(cols) == COLS


def _try_scene(rng) -> dict | None:
    grid = {}
    for row in range(ROWS):
        for col in range(COLS):
            if rng.random() >= FILL_PROB:
                continue
            slant = -1 if rng.random() < 0.5 else 1
            grid[(row, col)] = [
                slant,
                rng.randint(-DX_JITTER, DX_JITTER),
                rng.randint(-DY_JITTER, DY_JITTER),
                rng.randint(HALF_LO, HALF_HI),
            ]

    ball_col = rng.randrange(COLS)
    if not _repair(grid, ball_col, None, set()):
        return None

    factual_cells, _fb, status, _ = _walk_grid(grid, ball_col, None)
    if status != "ok" or len(factual_cells) < MIN_VISITED:
        return None

    red_cell = factual_cells[rng.randrange(len(factual_cells))]
    if not _repair(grid, ball_col, red_cell, {red_cell}):
        return None

    factual_cells, _fb, status, _ = _walk_grid(grid, ball_col, None)
    if status != "ok" or len(factual_cells) < MIN_VISITED:
        return None
    if red_cell not in factual_cells:
        return None
    edited_cells, _cb, status, _ = _walk_grid(grid, ball_col, red_cell)
    if status != "ok":
        return None
    if not _covers_lattice(grid):
        return None

    fset, eset = set(factual_cells), set(edited_cells)
    gained = sorted(eset - fset)
    lost = sorted(fset - eset)
    both = sorted((fset & eset) - {red_cell})
    neither = sorted(set(grid) - fset - eset)
    if not (gained and lost and both and neither):
        return None

    gained_cell = gained[rng.randrange(len(gained))]
    # Prefer a shared bar that the two runs meet *after* they have re-converged,
    # and below the gained bar, so that "the lowest tagged bar on the edited
    # run" cannot stand in for the contrast between the two runs.
    deep = [cell for cell in both if cell[0] > gained_cell[0]]
    pool = deep if deep else both
    picks = {
        "gained": gained_cell,
        "lost": lost[rng.randrange(len(lost))],
        "both": pool[rng.randrange(len(pool))],
        "neither": neither[rng.randrange(len(neither))],
    }
    if len(set(picks.values())) != 4:
        return None

    hues = list(TAG_NAMES)
    for i in range(len(hues) - 1, 0, -1):
        j = rng.randrange(i + 1)
        hues[i], hues[j] = hues[j], hues[i]

    cells = sorted(grid)
    bars = [[row, col, *grid[(row, col)]] for row, col in cells]
    tags = {hue: cells.index(picks[role]) for hue, role in zip(hues, ROLES)}

    return {
        "bars": bars,
        "red": int(cells.index(red_cell)),
        "ball_col": int(ball_col),
        "tags": {hue: int(tags[hue]) for hue in sorted(tags)},
    }


def sample_scene(seed: int) -> dict:
    """Sample one latent scene deterministically from ``seed``."""
    import random

    for attempt in range(6000):
        rng = random.Random((int(seed) * 1000003 + attempt * 7919 + 11) & 0x7FFFFFFF)
        scene = _try_scene(rng)
        if scene is not None:
            return scene
    raise RuntimeError(f"no admissible scene for seed {seed}")


def render(scene: dict) -> Image.Image:
    """Rasterise the latent scene (a deterministic PNG)."""
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(img)

    red_index = int(scene["red"])
    hue_of = {int(v): k for k, v in scene["tags"].items()}
    for index, bar in enumerate(scene["bars"]):
        row, col, slant, dx, dy, half = (int(v) for v in bar)
        cx = _col_x(col) + dx
        cy = _row_y(row) + dy
        if slant < 0:
            xy = [cx - half, cy + half, cx + half, cy - half]
        else:
            xy = [cx - half, cy - half, cx + half, cy + half]
        if index == red_index:
            color = RED_COLOR
        elif index in hue_of:
            color = TAG_PALETTE[hue_of[index]]
        else:
            color = BAR_COLOR
        draw.line(xy, fill=color, width=BAR_W)

    ball_x = _col_x(int(scene["ball_col"]))
    draw.ellipse(
        [ball_x - BALL_R, BALL_Y - BALL_R, ball_x + BALL_R, BALL_Y + BALL_R],
        fill=BALL_COLOR,
    )
    draw.line(
        [ball_x, BALL_Y + BALL_R, ball_x, BALL_Y + BALL_R + 14], fill=BALL_COLOR, width=5
    )
    draw.rectangle([40, RAIL_TOP, WIDTH - 40, RAIL_TOP + RAIL_H], fill=RAIL_COLOR)
    return img


def _contrast(scene: dict):
    """(factual indices, edited indices) for the drawn and re-leaned boards."""
    bars = scene["bars"]
    start = int(scene["ball_col"])
    factual, ok_f = _walk(bars, start, -1)
    edited, ok_e = _walk(bars, start, int(scene["red"]))
    if not (ok_f and ok_e):
        raise ValueError("scene trajectory leaves the lattice")
    return factual, edited


def analytic_gold(scene: dict) -> str:
    """Colour of the tagged bar met by the edited run but not the drawn run."""
    factual, edited = _contrast(scene)
    gained = set(edited) - set(factual)
    hits = sorted(hue for hue, idx in scene["tags"].items() if int(idx) in gained)
    if len(hits) != 1:
        raise ValueError("scene does not have a unique gained tag")
    return str(hits[0])


def _insets(scene: dict) -> list:
    """Legibility clearances that decide every bar-meeting event."""
    bars = scene["bars"]
    values = []
    for flip in (-1, int(scene["red"])):
        col = int(scene["ball_col"])
        met, ok = _walk(bars, col, flip)
        if not ok:
            return [0.0]
        pos = col
        met_set = set(met)
        for row in range(ROWS):
            column_x = _col_x(pos)
            for index, bar in enumerate(bars):
                if int(bar[0]) != row:
                    continue
                cx = _col_x(int(bar[1])) + int(bar[3])
                half = int(bar[5])
                gap = abs(column_x - cx)
                if index in met_set and int(bar[1]) == pos:
                    values.append(float(half - gap))
                    delta = int(bar[2])
                    if index == flip:
                        delta = -delta
                    pos += delta
                else:
                    values.append(float(gap - half))
    return values or [0.0]


def margin(scene: dict) -> float:
    """Commitment magnitude: minimum decisive bar/column clearance in pixels."""
    return float(min(_insets(scene)))


def is_quarantined(scene: dict) -> bool:
    """Quarantine boards whose bar-meeting events are not cleanly separated."""
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list:
    """The scene is injectively rendered; no latent aliases are declared."""
    return []
