"""Pixel-only inverse arm for the path-crossings world.

Reads only the final RGB PNG. Segments the non-white ink into the grey dashed
guide line and the blue path by nearest-colour labelling. It recovers the guide
line's vertical position from the grey dash pixels, then counts how many
separate connected blobs of blue path cross a thin horizontal strip centred on
that guide line. Each side-to-side crossing of the single continuous blue path
produces exactly one such blob, so that count is the number of crossings.

The oracle imports no renderer, prompt, generator, verifier or scene state and
consumes only the final pixels plus the public question contract.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_PALETTE = {
    "blue": (40, 90, 210),
    "gray": (150, 150, 150),
    "white": (255, 255, 255),
}
_NAMES = ["blue", "gray", "white"]
_STRIP = 8            # half-height (px) of the strip around the guide line


def _label_pixels(arr: np.ndarray) -> np.ndarray:
    palette = np.array([*_PALETTE.values()], dtype=np.float32)
    dists = np.stack([np.linalg.norm(arr - c, axis=-1) for c in palette], axis=-1)
    return dists.argmin(axis=-1)


def _components(mask: np.ndarray) -> int:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    count = 0
    stack: list[tuple[int, int]] = []
    for start in zip(*np.argwhere(mask).T, strict=False):
        sy, sx = int(start[0]), int(start[1])
        if seen[sy, sx]:
            continue
        count += 1
        stack.append((sy, sx))
        seen[sy, sx] = True
        while stack:
            y, x = stack.pop()
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
    return count


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label_pixels(arr)
    idx = {name: i for i, name in enumerate(_NAMES)}
    blue = labels == idx["blue"]
    gray = labels == idx["gray"]
    if not blue.any():
        raise ValueError("no blue path found")
    gray_rows = np.argwhere(gray)[:, 0]
    if gray_rows.size == 0:
        raise ValueError("no grey guide line found")
    y0 = float(np.mean(gray_rows))

    strip_rows = np.abs(np.arange(labels.shape[0]) - y0) <= _STRIP
    blue_strip = blue & strip_rows[:, None]
    if not blue_strip.any():
        raise ValueError("no blue path crossings found on the guide line")
    crossings = _components(blue_strip)
    return str(crossings)
