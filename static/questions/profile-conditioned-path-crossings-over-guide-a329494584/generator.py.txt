"""Tracing world: how many times does the single blue path cross the dashed guide line.

A single connected blue polyline weaves back and forth across a fixed dashed
horizontal grey guide line. At each of its bends the stroke crosses from one
side of the guide line to the other, so the path crosses the guide line a
specific whole number of times. The answer is the number of such crossings.

This is a tracing / intersection-event tally measured along the drawn stroke:
the count depends on how many times the single continuous route passes over the
reference line, not on a fold count of a serpentine (no hairpin U-turns), not on
a geodesic distance, and not on a count of disconnected components.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "path_crossings_over_guide-0.1.0"

CANVAS = (480, 480)
BG = (255, 255, 255)
COLOR_BLUE = (40, 90, 210)
COLOR_GRAY = (150, 150, 150)

STROKE = 14            # blue path stroke thickness (px)
GUIDE_WIDTH = 6        # grey guide line thickness (px)
GUIDE_DASH = 18        # dash length for the guide line (px)
GUIDE_GAP = 12         # gap between dashes (px)
D_MIN, D_MAX = 70.0, 110.0   # connector clearance amplitude from the guide line
A = 26.0               # horizontal half-span of each crossing diagonal
X_LO, X_HI = 60.0, 420.0    # crossing position range on the guide line
MIN_GAP, MAX_GAP = 60.0, 170.0  # min/max spacing between adjacent crossings
MIN_CLEARANCE = 20.0   # quarantine below this margin (px)


def _pts(points: list) -> list[tuple[float, float]]:
    return [(float(p[0]), float(p[1])) for p in points]


def _sample_positions(rng: random.Random, k: int) -> list[float]:
    if k <= 1:
        return [rng.uniform(X_LO + 25.0, X_HI - 25.0)]
    n = k - 1
    for _ in range(4000):
        hi_total = min(n * MAX_GAP, (X_HI - X_LO) - 70.0)
        total = rng.uniform(n * MIN_GAP, max(n * MIN_GAP, hi_total))
        weights = [rng.uniform(0.0, 1.0) for _ in range(n)]
        sw = sum(weights)
        gaps = [w / sw * total for w in weights]
        if any(g < MIN_GAP or g > MAX_GAP for g in gaps):
            continue
        x0 = rng.uniform(X_LO + 25.0, X_HI - 25.0 - total)
        xs = [x0]
        for g in gaps:
            xs.append(xs[-1] + g)
        if xs[-1] > X_HI - 25.0:
            continue
        return xs
    raise RuntimeError("could not place crossings")


def _rejection_loop(rng: random.Random) -> dict:
    w, h = CANVAS
    for _ in range(20000):
        k = rng.randint(2, 5)                      # crossings in {2,3,4,5}
        y0 = rng.uniform(150, 330)                 # guide line row
        D = rng.uniform(D_MIN, D_MAX)              # connector amplitude
        if not (20 < y0 - D - STROKE and y0 + D + STROKE < h - 20):
            continue
        xs = _sample_positions(rng, k)
        if not (30 < xs[0] - A and xs[-1] + A < w - 30):
            continue
        return {"crossings": xs, "y0": y0, "d": D, "a": A}
    raise RuntimeError("could not satisfy path constraints")


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    return _rejection_loop(rng)


def _polyline_points(scene: dict) -> list[tuple[float, float]]:
    xs = [float(x) for x in scene["crossings"]]
    y0 = float(scene["y0"])
    D = float(scene["d"])
    A = float(scene["a"])
    pts = [(xs[0] - A, y0 + D)]          # start below the guide line
    side = "below"
    for i, x in enumerate(xs):
        if i == 0:
            pts.append((x + A, y0 - D))  # first diagonal up through y0
            side = "above"
        else:
            prev_y = y0 - D if side == "above" else y0 + D
            pts.append((x - A, prev_y))  # horizontal connector on current side
            if side == "above":          # diagonal down through y0
                pts.append((x + A, y0 + D))
                side = "below"
            else:                        # diagonal up through y0
                pts.append((x + A, y0 - D))
                side = "above"
    return _pts(pts)


def _draw_polyline(draw: ImageDraw.ImageDraw, pts, color, width):
    for a, b in zip(pts, pts[1:]):
        draw.line([a, b], fill=color, width=width, joint="curve")


def _draw_guide_line(draw: ImageDraw.ImageDraw, y0: float):
    dash_on = True
    x = 30.0
    while x < CANVAS[0] - 30:
        if dash_on:
            draw.line([(x, y0), (min(x + GUIDE_DASH, CANVAS[0] - 30), y0)],
                      fill=COLOR_GRAY, width=GUIDE_WIDTH)
        x += GUIDE_DASH if dash_on else GUIDE_GAP
        dash_on = not dash_on


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", CANVAS, BG)
    draw = ImageDraw.Draw(img)
    _draw_guide_line(draw, float(scene["y0"]))
    _draw_polyline(draw, _polyline_points(scene), COLOR_BLUE, STROKE)
    return img


def analytic_gold(scene: dict) -> str:
    """Number of times the single blue path crosses the guide line."""
    return str(len(scene["crossings"]))


def margin(scene: dict) -> float:
    """Minimum clearance that protects the crossing count from ambiguity (px)."""
    D = float(scene["d"])
    xs = sorted(float(x) for x in scene["crossings"])
    gaps = [xs[i + 1] - xs[i] for i in range(len(xs) - 1)]
    blob_sep = (min(gaps) if gaps else 1e9) - STROKE
    connector_clear = D - STROKE          # connector stays away from the line
    return float(min(blob_sep, connector_clear))


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < MIN_CLEARANCE)


def latent_symmetries(scene: dict) -> list:
    return []
