from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

W = H = 384
TRAIL_PALETTES = [
    ("#185FA5", "#F5F2EA"),
    ("#146B8C", "#F7F1E3"),
    ("#2456A6", "#F3EFE5"),
]


def _cross(a, b, c) -> float:
    return (b[0] - a[0]) * (c[1] - b[1]) - (b[1] - a[1]) * (c[0] - b[0])


def _turns(points):
    # Image y increases downward, so negative raster cross product is a visual left turn.
    return [1 if _cross(a, b, c) < 0 else -1 for a, b, c in zip(points, points[1:], points[2:])]


def _segments_intersect(a, b, c, d):
    def orient(p, q, r):
        v = (q[0] - p[0]) * (r[1] - p[1]) - (q[1] - p[1]) * (r[0] - p[0])
        return (v > 0) - (v < 0)
    return orient(a, b, c) * orient(a, b, d) <= 0 and orient(c, d, a) * orient(c, d, b) <= 0


def _valid(points):
    if max(x for x, _ in points) - min(x for x, _ in points) > 145:
        return False
    if max(y for _, y in points) - min(y for _, y in points) > 145:
        return False
    for i in range(len(points) - 1):
        for j in range(i + 2, len(points) - 1):
            if j == i + 1:
                continue
            if _segments_intersect(points[i], points[i + 1], points[j], points[j + 1]):
                return False
    for i, p in enumerate(points):
        for q in points[i + 2:]:
            if math.dist(p, q) < 7:
                return False
    return True


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    target = "left" if seed % 2 == 0 else "right"
    desired = 1 if target == "left" else -1
    for _ in range(20000):
        # Sample a monotone narrow route rather than a fixed turn template. Its
        # seven slope changes carry the answer while the whole route stays focal.
        minority=1
        signs=[desired]*(7-minority)+[-desired]*minority
        rng.shuffle(signs)
        lengths=[rng.randrange(10,13) for _j in range(8)]
        slopes=[0.0]
        for sign in signs:
            delta = rng.uniform(0.26,0.34) if sign == desired else rng.uniform(1.68,1.96)
            slopes.append(slopes[-1]-sign*delta)
        weighted_mean=sum(s*n for s,n in zip(slopes,lengths))/sum(lengths)
        slopes=[s-weighted_mean for s in slopes]
        pts=[(0.0,0.0)]
        for length,slope in zip(lengths,slopes):
            pts.append((pts[-1][0]+length,pts[-1][1]+length*slope))
        ys=[p[1] for p in pts]
        if max(ys)-min(ys) <= 24 and _valid(pts):
            break
    else:
        raise RuntimeError("could not sample compact simple trail")
    quarter = rng.randrange(4)
    for _ in range(quarter):
        pts = [(-y, x) for x, y in pts]
    minx, maxx = min(x for x, _ in pts), max(x for x, _ in pts)
    miny, maxy = min(y for _, y in pts), max(y for _, y in pts)
    horizontal = (maxx-minx) >= (maxy-miny)
    if horizontal:
        target_x=rng.randint(145,239)
        target_y=rng.choice([120,168,216,264])
    else:
        target_x=rng.choice([120,168,216,264])
        target_y=rng.randint(145,239)
    cx = target_x-(minx+maxx)/2
    cy = target_y-(miny+maxy)/2
    pts = [[x + cx, y + cy] for x, y in pts]
    trail_color, background = rng.choice(TRAIL_PALETTES)
    return {
        "points": pts,
        "start_color": "#D1495B",
        "finish_color": "#238B57",
        "trail_color": trail_color,
        "background": background,
        "width": W,
        "height": H,
        "nuisance_tag": rng.randrange(1_000_000),
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), scene["background"])
    draw = ImageDraw.Draw(image)
    points = [tuple(p) for p in scene["points"]]
    draw.rounded_rectangle((18, 18, image.width - 19, image.height - 19), radius=14, outline="#C9C4B8", width=2)
    # Remote gray marks vary but carry no label information.
    for x, y in ((45, 47), (image.width - 46, 48), (46, image.height - 47), (image.width - 45, image.height - 46)):
        draw.ellipse((x - 2, y - 2, x + 2, y + 2), fill="#B9B4AA")
    draw.line(points, fill=scene["trail_color"], width=7, joint="curve")
    for x, y in points[1:-1]:
        draw.ellipse((x - 3, y - 3, x + 3, y + 3), fill=scene["trail_color"])
    for (x, y), color in ((points[0], scene["start_color"]), (points[-1], scene["finish_color"])):
        draw.ellipse((x - 6, y - 6, x + 6, y + 6), fill=color)
        draw.ellipse((x - 2, y - 2, x + 2, y + 2), fill="#FFFFFF")
    return image


def analytic_gold(scene: dict) -> str:
    votes = sum(_turns(scene["points"]))
    return "left" if votes > 0 else "right"


def margin(scene: dict) -> float:
    return float(abs(sum(_turns(scene["points"]))))


def is_quarantined(scene: dict) -> bool:
    points = scene["points"]
    return not _valid(points) or margin(scene) < 1


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["nuisance_tag"] = (int(scene.get("nuisance_tag", 0)) + 1) % 1_000_000
    return [("unused_sampling_tag", twin)]
