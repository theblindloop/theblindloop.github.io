"""Renderer for ``door_handle_hinge_side``.

One door leaf is drawn as an axis-aligned rectangle. A small group of dark slate
blocks straddles exactly one of its four edges: that edge is the hinged edge.
One solid black disc is the handle. The gold decision is whether the handle's
centre lies in the half of the door that touches the hinged edge (``same``) or in
the far half (``opposite``). The familiar prior says a handle sits opposite its
hinges; canonical and counterfactual scenes are sampled in exactly equal numbers
and are matched in wording, palette, scale, layout and ink.
"""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

RENDERER_VERSION = "door_handle_hinge_side-renderer-0.1.0"
CANONICAL_PRIOR_ANSWER = "opposite"

WHITE = (255, 255, 255)
FRAME_INK = (150, 150, 150)
HINGE_INK = (72, 72, 80)
HANDLE_INK = (16, 16, 16)
PANEL_INK = (255, 255, 255)

BODY_COLORS = [
    (196, 132, 96), (150, 176, 214), (176, 196, 140), (206, 160, 186),
    (210, 190, 120), (140, 190, 184), (188, 156, 210), (200, 142, 132),
]
EDGES = ("left", "right", "top", "bottom")
QUARANTINE_PX = 14.0


def _edge_sign(edge: str) -> int:
    return -1 if edge in ("left", "top") else 1


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    width, height = 560, 420
    edge = EDGES[(seed // 2) % 4]
    want_same = seed % 2 == 1

    hw = rng.randint(74, 108)
    hh = rng.randint(96, 140)
    cx = rng.randint(hw + 46, width - hw - 46)
    cy = rng.randint(hh + 40, height - hh - 40)

    handle_r = rng.randint(9, 13)
    horizontal = edge in ("left", "right")
    half_along = hw if horizontal else hh
    half_cross = hh if horizontal else hw

    lo = 10
    hi = half_along - handle_r - 20
    mag = rng.randint(lo, max(lo + 1, hi))
    sign = _edge_sign(edge) if want_same else -_edge_sign(edge)
    offset_along = sign * mag
    offset_cross = rng.randint(-(half_cross - handle_r - 22), half_cross - handle_r - 22)

    n_hinges = rng.choice((2, 3))
    hinge_len = rng.randint(20, 30)
    hinge_thick = rng.randint(9, 13)
    hinge_gap = rng.randint(26, 40)

    n_panels = rng.choice((0, 2, 3))
    panels = sorted(rng.sample(range(-half_cross + 24, half_cross - 24), n_panels)) if n_panels else []

    return {
        "width": width,
        "height": height,
        "cx": cx,
        "cy": cy,
        "hw": hw,
        "hh": hh,
        "hinge_edge": edge,
        "n_hinges": n_hinges,
        "hinge_len": hinge_len,
        "hinge_thick": hinge_thick,
        "hinge_gap": hinge_gap,
        "handle_r": handle_r,
        "handle_offset_px": offset_along,
        "handle_cross_px": offset_cross,
        "body_color": list(BODY_COLORS[rng.randrange(len(BODY_COLORS))]),
        "panels": panels,
        "decor": [rng.randint(10, 20) for _ in range(4)],
    }


def _handle_xy(scene: dict) -> tuple[float, float]:
    if scene["hinge_edge"] in ("left", "right"):
        return scene["cx"] + scene["handle_offset_px"], scene["cy"] + scene["handle_cross_px"]
    return scene["cx"] + scene["handle_cross_px"], scene["cy"] + scene["handle_offset_px"]


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (scene["width"], scene["height"]), WHITE)
    d = ImageDraw.Draw(img)
    w, h = scene["width"], scene["height"]
    d.rectangle([12, 12, w - 13, h - 13], outline=FRAME_INK, width=3)
    t = scene["decor"]
    d.line([26, 26, 26 + t[0], 26], fill=FRAME_INK, width=3)
    d.line([w - 27, 26, w - 27 - t[1], 26], fill=FRAME_INK, width=3)
    d.line([26, h - 27, 26 + t[2], h - 27], fill=FRAME_INK, width=3)
    d.line([w - 27, h - 27, w - 27 - t[3], h - 27], fill=FRAME_INK, width=3)

    cx, cy, hw, hh = scene["cx"], scene["cy"], scene["hw"], scene["hh"]
    x0, y0, x1, y1 = cx - hw, cy - hh, cx + hw, cy + hh
    d.rectangle([x0, y0, x1, y1], fill=tuple(scene["body_color"]))

    horizontal = scene["hinge_edge"] in ("left", "right")
    for p in scene["panels"]:
        if horizontal:
            d.rectangle([x0 + 14, cy + p - 2, x1 - 14, cy + p + 2], fill=PANEL_INK)
        else:
            d.rectangle([cx + p - 2, y0 + 14, cx + p + 2, y1 - 14], fill=PANEL_INK)

    n, gap, hl, ht = scene["n_hinges"], scene["hinge_gap"], scene["hinge_len"], scene["hinge_thick"]
    starts = [(i - (n - 1) / 2.0) * gap for i in range(n)]
    for s in starts:
        if horizontal:
            ex = x0 if scene["hinge_edge"] == "left" else x1
            d.rectangle([ex - ht, cy + s - hl / 2, ex + ht, cy + s + hl / 2], fill=HINGE_INK)
        else:
            ey = y0 if scene["hinge_edge"] == "top" else y1
            d.rectangle([cx + s - hl / 2, ey - ht, cx + s + hl / 2, ey + ht], fill=HINGE_INK)

    hx, hy = _handle_xy(scene)
    r = scene["handle_r"]
    d.ellipse([hx - r, hy - r, hx + r, hy + r], fill=HANDLE_INK)
    return img


def analytic_gold(scene: dict) -> str:
    same = (scene["handle_offset_px"] > 0) == (_edge_sign(scene["hinge_edge"]) > 0)
    return "same" if same else "opposite"


def margin(scene: dict) -> float:
    return float(abs(scene["handle_offset_px"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list:
    return []


def scene_targets(scene: dict) -> dict:
    return {
        "relation": analytic_gold(scene),
        "hinge_edge": scene["hinge_edge"],
        "handle_offset_px": scene["handle_offset_px"],
        "continuous": margin(scene),
        "canonical_prior_answer": CANONICAL_PRIOR_ANSWER,
        "is_canonical": analytic_gold(scene) == CANONICAL_PRIOR_ANSWER,
    }
