"""Image-only decision arm for ``door_handle_hinge_side``.

Receives one rendered RGB image and nothing else. It segments the two fixed
inks (the slate hinge blocks and the black handle disc), recovers the door leaf
as the remaining non-white non-decoration patch, decides which of the leaf's four
edges the hinge blocks straddle, and reports whether the handle centre lies in
the half of the leaf touching that edge. No renderer, prompt, generator,
verifier or gold code is imported and no latent scene is accepted.
"""

from __future__ import annotations

from dataclasses import dataclass

import numpy as np

ORACLE_VERSION = "door_handle_hinge_side-oracle-0.1.0"

_WHITE = (255, 255, 255)
_FRAME = (150, 150, 150)
_HINGE = (72, 72, 80)
_HANDLE = (16, 16, 16)
_MIN_MARGIN_PX = 6.0
_MIN_EDGE_SEPARATION_PX = 12.0
ABSTAIN = "abstain"


@dataclass(frozen=True)
class Measurement:
    decision: str
    runner_up_margin: float
    hinge_edge: str
    offset_px: float


def _mask(arr: np.ndarray, rgb) -> np.ndarray:
    return (arr[:, :, 0] == rgb[0]) & (arr[:, :, 1] == rgb[1]) & (arr[:, :, 2] == rgb[2])


def _centroid(mask: np.ndarray) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    return float(xs.mean()), float(ys.mean())


def measure(image) -> Measurement:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    hinge = _mask(arr, _HINGE)
    handle = _mask(arr, _HANDLE)
    body = ~(_mask(arr, _WHITE) | _mask(arr, _FRAME) | hinge | handle)
    if hinge.sum() < 120 or handle.sum() < 120 or body.sum() < 2000:
        return Measurement(ABSTAIN, 0.0, "unknown", 0.0)

    ys, xs = np.nonzero(body)
    x0, x1 = float(xs.min()), float(xs.max())
    y0, y1 = float(ys.min()), float(ys.max())
    if (x1 - x0) < 40 or (y1 - y0) < 40:
        return Measurement(ABSTAIN, 0.0, "unknown", 0.0)

    hx, hy = _centroid(hinge)
    dists = {"left": abs(hx - x0), "right": abs(x1 - hx), "top": abs(hy - y0), "bottom": abs(y1 - hy)}
    order = sorted(dists.items(), key=lambda kv: kv[1])
    edge, best = order[0]
    if order[1][1] - best < _MIN_EDGE_SEPARATION_PX:
        return Measurement(ABSTAIN, 0.0, "unknown", 0.0)

    gx, gy = _centroid(handle)
    if edge in ("left", "right"):
        mid = 0.5 * (x0 + x1)
        offset = gx - mid
    else:
        mid = 0.5 * (y0 + y1)
        offset = gy - mid
    if abs(offset) < _MIN_MARGIN_PX:
        return Measurement(ABSTAIN, abs(offset), edge, offset)

    edge_sign = -1.0 if edge in ("left", "top") else 1.0
    same = (offset > 0) == (edge_sign > 0)
    return Measurement("same" if same else "opposite", abs(offset), edge, offset)


def decision_from_image(image) -> str:
    return measure(image).decision
