"""Independent inverse arm: recover the answer from raster pixels alone."""

from __future__ import annotations

import numpy as np
from PIL import Image

BACKGROUND = np.array((250, 250, 247), dtype=np.float32)
PATTERN_COLORS = np.array(
    ((48, 105, 178), (218, 82, 78), (71, 157, 101), (235, 188, 62)),
    dtype=np.float32,
)
TAG_NAMES = ("amber", "cyan", "violet")
TAG_COLORS = np.array(((244, 137, 32), (35, 187, 207), (151, 83, 190)), dtype=np.float32)
SOURCE_TAG = np.array((105, 113, 122), dtype=np.float32)
ALL_FILLS = np.vstack((PATTERN_COLORS, TAG_COLORS, SOURCE_TAG[None, :]))


def _board_bounds(array: np.ndarray) -> tuple[int, int, int, int]:
    delta = np.linalg.norm(array.astype(np.float32) - BACKGROUND, axis=2)
    ys, xs = np.where(delta > 35.0)
    if len(xs) < 100000:
        raise ValueError("the filled carpet is missing or too incomplete")
    left, right = int(xs.min()), int(xs.max())
    top, bottom = int(ys.min()), int(ys.max())
    width = right - left + 1
    height = bottom - top + 1
    if width < 900 or height < 500 or not 1.68 < width / height < 1.90:
        raise ValueError("could not recover the 16-by-9 carpet boundary")
    return left, top, right + 1, bottom + 1


def _cell_labels(array: np.ndarray, bounds: tuple[int, int, int, int]) -> np.ndarray:
    left, top, right, bottom = bounds
    width = right - left
    height = bottom - top
    labels = np.empty((9, 16), dtype=np.int16)
    for row in range(9):
        y0 = int(round(top + (row + 0.24) * height / 9.0))
        y1 = int(round(top + (row + 0.76) * height / 9.0))
        for col in range(16):
            x0 = int(round(left + (col + 0.24) * width / 16.0))
            x1 = int(round(left + (col + 0.76) * width / 16.0))
            patch = array[y0:y1, x0:x1].astype(np.float32).reshape(-1, 3)
            if len(patch) < 100:
                raise ValueError("cell interior is too small")
            distances = np.linalg.norm(patch[:, None, :] - ALL_FILLS[None, :, :], axis=2)
            nearest = distances.argmin(axis=1)
            winner_counts = np.bincount(nearest, minlength=len(ALL_FILLS))
            winner = int(winner_counts.argmax())
            purity = float(winner_counts[winner] / len(patch))
            close = float(np.mean(distances[:, winner] < 16.0))
            if purity < 0.94 or close < 0.94:
                raise ValueError(f"cell ({row},{col}) is erased, mixed, or off-palette")
            labels[row, col] = winner
    return labels


def _uniform(values: np.ndarray, expected: set[int], what: str) -> int:
    unique = set(int(value) for value in values.flat)
    if len(unique) != 1:
        raise ValueError(f"{what} is not uniform")
    value = unique.pop()
    if value not in expected:
        raise ValueError(f"{what} has an invalid fill")
    return value


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    labels = _cell_labels(array, _board_bounds(array))

    mapping: dict[int, int] = {}
    for group in range(4):
        cols = slice(group * 4, group * 4 + 4)
        source_color = _uniform(labels[0:1, cols], set(range(4)), f"key source {group}")
        target_color = _uniform(labels[1:2, cols], set(range(4)), f"key target {group}")
        mapping[source_color] = target_color
    if set(mapping) != set(range(4)) or set(mapping.values()) != set(range(4)):
        raise ValueError("the visible key is not a one-to-one four-color mapping")

    source = labels[2:8, 0:4]
    if np.any(source >= 4):
        raise ValueError("source mosaic contains a non-pattern fill")
    transformed = np.vectorize(mapping.__getitem__, otypes=[np.int16])(source)
    matches: list[int] = []
    for candidate_index in range(3):
        candidate = labels[2:8, 4 + 4 * candidate_index : 8 + 4 * candidate_index]
        if np.any(candidate >= 4):
            raise ValueError("candidate mosaic contains a non-pattern fill")
        if np.array_equal(candidate, transformed):
            matches.append(candidate_index)
    if len(matches) != 1:
        raise ValueError(f"expected one pixel-level mosaic match, found {len(matches)}")

    _uniform(labels[8:9, 0:4], {7}, "source tag")
    tags: list[str] = []
    for candidate_index in range(3):
        value = _uniform(
            labels[8:9, 4 + 4 * candidate_index : 8 + 4 * candidate_index],
            {4, 5, 6},
            f"candidate tag {candidate_index}",
        )
        tags.append(TAG_NAMES[value - 4])
    if sorted(tags) != sorted(TAG_NAMES):
        raise ValueError("candidate answer tags are not a permutation")
    return tags[matches[0]]
