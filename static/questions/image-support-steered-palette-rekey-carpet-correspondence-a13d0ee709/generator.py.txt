"""Deterministic latent sampling and rendering for one unified rekey carpet."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 1024
HEIGHT = 640
BOARD_X = 32
BOARD_Y = 50
CELL = 60
ROWS = 9
COLS = 16

BACKGROUND = (250, 250, 247)
GRID = (29, 35, 44)
PATTERN_COLORS = (
    (48, 105, 178),
    (218, 82, 78),
    (71, 157, 101),
    (235, 188, 62),
)
TAG_COLORS = {
    "amber": (244, 137, 32),
    "cyan": (35, 187, 207),
    "violet": (151, 83, 190),
}
SOURCE_TAG = (105, 113, 122)
TAG_NAMES = ("amber", "cyan", "violet")


def _flat(grid: list[list[int]]) -> list[int]:
    return [value for row in grid for value in row]


def _grid(values: Iterable[int]) -> list[list[int]]:
    data = list(values)
    if len(data) != 24:
        raise ValueError("a mosaic must contain exactly 24 cells")
    return [data[offset : offset + 4] for offset in range(0, 24, 4)]


def _transform(source: list[list[int]], mapping: list[int]) -> list[list[int]]:
    return [[mapping[value] for value in row] for row in source]


def _hamming(left: list[list[int]], right: list[list[int]]) -> int:
    return sum(a != b for a, b in zip(_flat(left), _flat(right), strict=True))


def _balanced_source(rng: random.Random) -> list[list[int]]:
    while True:
        values = [value for value in range(4) for _ in range(6)]
        rng.shuffle(values)
        candidate = _grid(values)
        if all(len(set(row)) >= 2 for row in candidate):
            return candidate


def _distractor(target: list[list[int]], rng: random.Random, minimum: int) -> list[list[int]]:
    base = _flat(target)
    for _ in range(1000):
        changed = base.copy()
        swap_count = 2 if minimum <= 4 else 3
        available = list(range(24))
        rng.shuffle(available)
        pairs: list[tuple[int, int]] = []
        while available and len(pairs) < swap_count:
            first = available.pop()
            choices = [index for index, other in enumerate(available) if base[other] != base[first]]
            if not choices:
                break
            second = available.pop(choices[rng.randrange(len(choices))])
            pairs.append((first, second))
        if len(pairs) < swap_count:
            continue
        for a, b in pairs:
            changed[a], changed[b] = changed[b], changed[a]
        result = _grid(changed)
        if _hamming(result, target) >= minimum:
            return result
    raise RuntimeError("could not construct an equal-histogram distractor")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    source = _balanced_source(rng)
    mapping = list(range(4))
    while mapping == list(range(4)) or sum(a != b for a, b in enumerate(mapping)) < 3:
        rng.shuffle(mapping)
    target = _transform(source, mapping)
    wrong_a = _distractor(target, rng, 4)
    wrong_b = _distractor(target, rng, 6)
    while wrong_b == wrong_a:
        wrong_b = _distractor(target, rng, 6)

    candidate_tags = list(TAG_NAMES)
    rng.shuffle(candidate_tags)
    desired_tag = TAG_NAMES[int(seed) % len(TAG_NAMES)]
    gold_index = candidate_tags.index(desired_tag)
    candidates: list[list[list[int]]] = [wrong_a, wrong_b]
    candidates.insert(gold_index, target)
    return {
        "source": source,
        "mapping": mapping,
        "candidates": candidates,
        "candidate_tags": candidate_tags,
    }


def analytic_gold(scene: dict) -> str:
    source = scene["source"]
    mapping = scene["mapping"]
    candidates = scene["candidates"]
    candidate_tags = scene["candidate_tags"]
    target = _transform(source, mapping)
    matches = [index for index, candidate in enumerate(candidates) if candidate == target]
    if len(matches) != 1:
        raise ValueError(f"expected one transformed-source match, found {len(matches)}")
    return candidate_tags[matches[0]]


def margin(scene: dict) -> float:
    target = _transform(scene["source"], scene["mapping"])
    distances = sorted(_hamming(target, candidate) for candidate in scene["candidates"])
    if not distances or distances[0] != 0:
        return 0.0
    return float(distances[1]) if len(distances) > 1 else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        source = scene["source"]
        mapping = scene["mapping"]
        candidates = scene["candidates"]
        tags = scene["candidate_tags"]
        valid = (
            len(source) == 6
            and all(len(row) == 4 for row in source)
            and sorted(_flat(source)) == sorted([value for value in range(4) for _ in range(6)])
            and sorted(mapping) == list(range(4))
            and len(candidates) == 3
            and all(len(item) == 6 and all(len(row) == 4 for row in item) for item in candidates)
            and sorted(tags) == sorted(TAG_NAMES)
        )
        if not valid:
            return True
        target = _transform(source, mapping)
        matches = sum(candidate == target for candidate in candidates)
        return matches != 1 or margin(scene) < 4.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _fill_cell(draw: ImageDraw.ImageDraw, col: int, row: int, color: tuple[int, int, int]) -> None:
    x0 = BOARD_X + col * CELL
    y0 = BOARD_Y + row * CELL
    draw.rectangle((x0, y0, x0 + CELL - 1, y0 + CELL - 1), fill=color)


def render(scene: dict) -> Image.Image:
    source = scene["source"]
    mapping = scene["mapping"]
    candidates = scene["candidates"]
    candidate_tags = scene["candidate_tags"]
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    for key_index in range(4):
        for col in range(key_index * 4, key_index * 4 + 4):
            _fill_cell(draw, col, 0, PATTERN_COLORS[key_index])
            _fill_cell(draw, col, 1, PATTERN_COLORS[mapping[key_index]])

    panels = [source, *candidates]
    for panel_index, panel in enumerate(panels):
        for local_row, values in enumerate(panel):
            for local_col, value in enumerate(values):
                _fill_cell(
                    draw,
                    panel_index * 4 + local_col,
                    2 + local_row,
                    PATTERN_COLORS[value],
                )

    for col in range(4):
        _fill_cell(draw, col, 8, SOURCE_TAG)
    for panel_index, tag in enumerate(candidate_tags, start=1):
        for col in range(panel_index * 4, panel_index * 4 + 4):
            _fill_cell(draw, col, 8, TAG_COLORS[tag])

    right = BOARD_X + COLS * CELL
    bottom = BOARD_Y + ROWS * CELL
    for col in range(COLS + 1):
        x = BOARD_X + col * CELL
        draw.line((x, BOARD_Y, x, bottom), fill=GRID, width=4)
    for row in range(ROWS + 1):
        y = BOARD_Y + row * CELL
        draw.line((BOARD_X, y, right, y), fill=GRID, width=4)
    for col in (4, 8, 12):
        x = BOARD_X + col * CELL
        draw.line((x, BOARD_Y, x, bottom), fill=GRID, width=8)
    draw.rectangle((BOARD_X, BOARD_Y, right, bottom), outline=GRID, width=7)
    return image
