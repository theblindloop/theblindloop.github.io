"""Deterministic renderer for a ruler-calibrated horizontal box blur world."""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw

CANVAS_SIZE = (512, 420)
FIELD_SIZE = 288
FRAME_THICKNESS = 5
CYAN = (18, 188, 207)
GOLD = (236, 176, 38)
LABELS = ("up arrow", "right arrow", "down arrow", "left arrow")
RULER_LENGTHS = (12, 16, 20, 24)
BACKGROUNDS = ((244, 241, 234), (238, 244, 242), (244, 239, 244))

# (base luminance, carrier amplitude, signal amplitude, RGB tint)
PALETTES = (
    (128, 88, 8, (0, 0, 0)),
    (126, 86, 10, (7, -2, -5)),
    (130, 84, 12, (-5, 6, -1)),
    (127, 87, 9, (5, -5, 4)),
)


def _arrow_mask(symbol_id: str, size: int = FIELD_SIZE) -> np.ndarray:
    """Rasterize an equal-ink base arrow, then rotate it in quarter turns."""
    yy, xx = np.mgrid[0:size, 0:size]
    x = (xx + 0.5) / size * 2.0 - 1.0
    y = (yy + 0.5) / size * 2.0 - 1.0
    head = (y >= -0.80) & (y <= -0.08) & (np.abs(x) <= 0.88 * (y + 0.80))
    shaft = (np.abs(x) <= 0.18) & (y >= -0.08) & (y <= 0.78)
    up = head | shaft
    turns = {
        "up arrow": 0,
        "left arrow": 1,
        "down arrow": 2,
        "right arrow": 3,
    }
    if symbol_id not in turns:
        raise ValueError(f"unknown symbol_id {symbol_id!r}")
    return np.rot90(up, turns[symbol_id]).copy()


def _carrier(scene: dict) -> np.ndarray:
    """Balanced period-L row textures; every length-L horizontal mean is zero."""
    length = int(scene["ruler_length"])
    phase = int(scene["carrier_phase"])
    texture_seed = int(scene["texture_seed"])
    output = np.empty((FIELD_SIZE, FIELD_SIZE), dtype=np.int16)
    columns = np.arange(FIELD_SIZE)
    for row in range(FIELD_SIZE):
        values = [1] * (length // 2) + [-1] * (length // 2)
        random.Random(texture_seed + row * 104_729).shuffle(values)
        row_phase = (phase + row * 3 + (row // 9) * 5) % length
        output[row] = np.asarray(values, dtype=np.int16)[(columns + row_phase) % length]
    return output


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    ruler_length = rng.choice(RULER_LENGTHS)
    return {
        "symbol_id": rng.choice(LABELS),
        "frame_x": rng.randint(66, 142),
        "frame_y": rng.randint(62, 90),
        "ruler_length": ruler_length,
        "ruler_offset": rng.randint(18, FIELD_SIZE - ruler_length - 18),
        "carrier_phase": rng.randrange(ruler_length),
        "texture_seed": rng.randrange(1, 2_000_000_000),
        "palette_index": rng.randrange(len(PALETTES)),
        "contrast_level": rng.randrange(3),
        "distractor_variant": rng.randrange(4),
        "background_variant": rng.randrange(len(BACKGROUNDS)),
    }


def _field_rgb(scene: dict) -> np.ndarray:
    palette_index = int(scene["palette_index"]) % len(PALETTES)
    base, carrier_amp, nominal_signal, tint = PALETTES[palette_index]
    signal_amp = nominal_signal + int(scene["contrast_level"])
    carrier = _carrier(scene)
    signal = np.where(_arrow_mask(str(scene["symbol_id"])), signal_amp, -signal_amp)
    luminance = base + carrier_amp * carrier + signal
    rgb = luminance[..., None] + np.asarray(tint, dtype=np.int16)
    return np.clip(rgb, 0, 255).astype(np.uint8)


def render(scene: dict) -> Image.Image:
    width, height = CANVAS_SIZE
    background = BACKGROUNDS[int(scene["background_variant"]) % len(BACKGROUNDS)]
    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)

    # Answer-independent peripheral marks defeat whole-canvas crop assumptions.
    variant = int(scene["distractor_variant"])
    ink = (116, 121, 126)
    for index in range(5):
        x = 20 + ((variant * 47 + index * 83) % 458)
        y = 20 + ((variant * 31 + index * 61) % 370)
        if index % 2:
            draw.ellipse((x, y, x + 5, y + 5), fill=ink)
        else:
            draw.line((x, y, x + 9, y + 4), fill=ink, width=2)

    outer_x = int(scene["frame_x"])
    outer_y = int(scene["frame_y"])
    outer_size = FIELD_SIZE + 2 * FRAME_THICKNESS
    draw.rectangle(
        (outer_x, outer_y, outer_x + outer_size - 1, outer_y + outer_size - 1),
        fill=CYAN,
    )
    image.paste(
        Image.fromarray(_field_rgb(scene), mode="RGB"),
        (outer_x + FRAME_THICKNESS, outer_y + FRAME_THICKNESS),
    )

    ruler_x = outer_x + int(scene["ruler_offset"])
    ruler_y = outer_y - 23
    ruler_length = int(scene["ruler_length"])
    draw.rectangle(
        (ruler_x, ruler_y, ruler_x + ruler_length - 1, ruler_y + 6),
        fill=GOLD,
    )
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol_id"])


def margin(scene: dict) -> float:
    palette_index = int(scene["palette_index"]) % len(PALETTES)
    return float(PALETTES[palette_index][2] + int(scene["contrast_level"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 7.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["palette_index"] = int(scene["palette_index"]) + len(PALETTES)
    return [("palette_period", twin)]
