"""Independent inverse arm using only visible pixels and public transform rules."""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw

CYAN = np.array((18, 188, 207), dtype=np.float64)
GOLD = np.array((236, 176, 38), dtype=np.float64)
LABELS = ("up arrow", "right arrow", "down arrow", "left arrow")


class OracleAbstention(ValueError):
    """The declared transform cannot be executed reliably from this raster."""


def _near_color(arr: np.ndarray, target: np.ndarray, tolerance: float = 18.0) -> np.ndarray:
    return np.linalg.norm(arr.astype(np.float64) - target, axis=2) <= tolerance


def _frame_crop(arr: np.ndarray) -> tuple[np.ndarray, int]:
    cyan = _near_color(arr, CYAN)
    ys, xs = np.nonzero(cyan)
    if len(xs) < 1000:
        raise OracleAbstention("cyan frame missing")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    sub = cyan[y0 : y1 + 1, x0 : x1 + 1]
    height, width = sub.shape
    if not (250 <= width <= 340 and 250 <= height <= 340):
        raise OracleAbstention("cyan frame has implausible bounds")

    col_full = sub.sum(axis=0) > 0.80 * height
    row_full = sub.sum(axis=1) > 0.80 * width
    left = 0
    while left < width and col_full[left]:
        left += 1
    right = 0
    while right < width and col_full[width - 1 - right]:
        right += 1
    top = 0
    while top < height and row_full[top]:
        top += 1
    bottom = 0
    while bottom < height and row_full[height - 1 - bottom]:
        bottom += 1
    if min(left, right, top, bottom) < 3 or max(left, right, top, bottom) > 9:
        raise OracleAbstention("cyan frame thickness is unstable")
    crop = arr[y0 + top : y1 - bottom + 1, x0 + left : x1 - right + 1]
    if crop.shape[0] != crop.shape[1] or not 250 <= crop.shape[0] <= 310:
        raise OracleAbstention("framed field is malformed")

    gold = _near_color(arr, GOLD)
    gy, gx = np.nonzero(gold)
    if len(gx) < 60:
        raise OracleAbstention("gold calibration bar missing")
    ruler = int(gx.max() - gx.min() + 1)
    ruler_height = int(gy.max() - gy.min() + 1)
    if ruler not in range(10, 27) or ruler % 2 or not 5 <= ruler_height <= 9:
        raise OracleAbstention("gold calibration bar is ambiguous")
    if int(gy.max()) >= y0:
        raise OracleAbstention("gold calibration bar is not above the frame")
    return crop, ruler


def _horizontal_box(values: np.ndarray, width: int) -> np.ndarray:
    cumulative = np.pad(np.cumsum(values, axis=1, dtype=np.float64), ((0, 0), (1, 0)))
    return (cumulative[:, width:] - cumulative[:, :-width]) / float(width)


def _reference(label: str, height: int, width: int) -> np.ndarray:
    base = Image.new("L", (width, height), 0)
    draw = ImageDraw.Draw(base)
    points = [
        (0.50 * width, 0.10 * height),
        (0.18 * width, 0.46 * height),
        (0.39 * width, 0.46 * height),
        (0.39 * width, 0.88 * height),
        (0.61 * width, 0.88 * height),
        (0.61 * width, 0.46 * height),
        (0.82 * width, 0.46 * height),
    ]
    draw.polygon(points, fill=255)
    turns = {
        "up arrow": 0,
        "left arrow": 1,
        "down arrow": 2,
        "right arrow": 3,
    }
    arr = np.asarray(base, dtype=np.float64) / 255.0
    return np.rot90(arr, turns[label]).copy()


def _standardize(values: np.ndarray) -> np.ndarray:
    centered = values - values.mean()
    norm = float(np.linalg.norm(centered))
    if norm < 1e-6:
        raise OracleAbstention("transformed evidence has no contrast")
    return centered / norm


def analyze_image(
    image: Image.Image, *, kernel_override: int | None = None
) -> tuple[str, float, tuple[float, ...]]:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    crop, visible_ruler = _frame_crop(arr)
    kernel = visible_ruler if kernel_override is None else int(kernel_override)
    if kernel < 1 or kernel > crop.shape[1] // 3:
        raise OracleAbstention("invalid averaging span")
    luma = (
        0.2126 * crop[..., 0].astype(np.float64)
        + 0.7152 * crop[..., 1].astype(np.float64)
        + 0.0722 * crop[..., 2].astype(np.float64)
    )
    transformed = _standardize(_horizontal_box(luma, kernel))
    scores: list[float] = []
    for label in LABELS:
        reference = _horizontal_box(_reference(label, *crop.shape[:2]), kernel)
        scores.append(float(np.sum(transformed * _standardize(reference))))
    order = np.argsort(scores)[::-1]
    best, second = int(order[0]), int(order[1])
    gap = 100.0 * (scores[best] - scores[second])
    if scores[best] < 0.42 or gap < 8.0:
        raise OracleAbstention(
            f"ambiguous transformed symbol: top={scores[best]:.3f}, gap={gap:.2f}"
        )
    return LABELS[best], round(gap, 6), tuple(round(value, 6) for value in scores)


def decision_from_image(image: Image.Image) -> str:
    return analyze_image(image)[0]
