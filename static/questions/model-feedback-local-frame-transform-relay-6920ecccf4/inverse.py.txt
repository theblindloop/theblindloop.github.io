"""Independent PNG-only inverse for local-frame coordinate transport."""

from __future__ import annotations

import numpy as np
from PIL import Image

EXPECTED_SIZE = (1400, 720)
SIZE = 7
START_X, START_Y, START_CELL, START_GAP = 18, 145, 24, 3
CARD_XS = (225, 615, 1005)
CARD_FROM_DX, CARD_TO_DX = 18, 204
FRAME_Y, FRAME_CELL, FRAME_GAP = 151, 20, 3
OPTION_XS = (68, 348, 628, 908, 1188)
OPTION_Y, OPTION_CELL, OPTION_GAP = 548, 17, 3
PALETTE = {
    "tail": np.array((32, 54, 74), dtype=np.float32),
    "head": np.array((211, 47, 47), dtype=np.float32),
    "origin": np.array((226, 153, 0), dtype=np.float32),
    "right": np.array((0, 137, 180), dtype=np.float32),
    "up": np.array((123, 31, 162), dtype=np.float32),
}


class OracleError(ValueError):
    pass


def _classify(array: np.ndarray, cx: int, cy: int, allowed: tuple[str, ...]) -> str | None:
    patch = array[cy - 2:cy + 3, cx - 2:cx + 3].astype(np.float32)
    mean = patch.mean(axis=(0, 1))
    ranked = sorted((float(np.linalg.norm(mean - PALETTE[name])), name) for name in allowed)
    if ranked[0][0] < 12.0 and (len(ranked) == 1 or ranked[1][0] - ranked[0][0] > 25.0):
        return ranked[0][1]
    if float(mean.mean()) > 190.0:
        return None
    raise OracleError("a cell center is neither clean paper nor an allowed marker")


def _scan(array, x, y, cell, gap, allowed):
    found = {}
    for row in range(SIZE):
        for col in range(SIZE):
            cx = x + col * (cell + gap) + cell // 2
            cy = y + row * (cell + gap) + cell // 2
            name = _classify(array, cx, cy, allowed)
            if name is not None:
                if name in found:
                    raise OracleError(f"duplicate {name} marker")
                found[name] = (row, col)
    if set(found) != set(allowed):
        raise OracleError("a marker grid is incomplete")
    return found


def _token(array, x, y, cell, gap):
    found = _scan(array, x, y, cell, gap, ("tail", "head"))
    token = found["tail"], found["head"]
    if abs(token[0][0] - token[1][0]) + abs(token[0][1] - token[1][1]) != 1:
        raise OracleError("token dots are not adjacent")
    return token


def _frame(array, x, y):
    frame = _scan(array, x, y, FRAME_CELL, FRAME_GAP, ("origin", "right", "up"))
    o = frame["origin"]
    u = (frame["right"][0] - o[0], frame["right"][1] - o[1])
    v = (frame["up"][0] - o[0], frame["up"][1] - o[1])
    if sum(q * q for q in u) != 1 or sum(q * q for q in v) != 1 or u[0] * v[0] + u[1] * v[1] != 0:
        raise OracleError("frame axes are not perpendicular unit steps")
    return frame


def _transport(point, source, target):
    so = source["origin"]
    su = (source["right"][0] - so[0], source["right"][1] - so[1])
    sv = (source["up"][0] - so[0], source["up"][1] - so[1])
    delta = (point[0] - so[0], point[1] - so[1])
    a = delta[0] * su[0] + delta[1] * su[1]
    b = delta[0] * sv[0] + delta[1] * sv[1]
    to = target["origin"]
    tu = (target["right"][0] - to[0], target["right"][1] - to[1])
    tv = (target["up"][0] - to[0], target["up"][1] - to[1])
    return to[0] + a * tu[0] + b * tv[0], to[1] + a * tu[1] + b * tv[1]


def _validate_labels(array):
    ink = PALETTE["tail"]
    for x in OPTION_XS:
        cx, cy = x + 68, 515
        patch = array[cy - 19:cy + 20, cx - 19:cx + 20].astype(np.float32)
        if int((np.linalg.norm(patch - ink, axis=2) < 12.0).sum()) < 850:
            raise OracleError("an option label disk is missing")


def decision_from_image(image: Image.Image) -> str:
    if image.size != EXPECTED_SIZE:
        raise OracleError(f"unexpected raster size {image.size}")
    array = np.asarray(image.convert("RGB"))
    _validate_labels(array)
    state = _token(array, START_X, START_Y, START_CELL, START_GAP)
    for x in CARD_XS:
        source = _frame(array, x + CARD_FROM_DX, FRAME_Y)
        target = _frame(array, x + CARD_TO_DX, FRAME_Y)
        state = _transport(state[0], source, target), _transport(state[1], source, target)
        if not all(0 <= r < SIZE and 0 <= c < SIZE for r, c in state):
            raise OracleError("transported token leaves the board")
    options = [_token(array, x, OPTION_Y, OPTION_CELL, OPTION_GAP) for x in OPTION_XS]
    if len(set(options)) != 5:
        raise OracleError("options are not distinct")
    matches = [i for i, option in enumerate(options) if option == state]
    if len(matches) != 1:
        raise OracleError("transported token does not match exactly one option")
    return "ABCDE"[matches[0]]
