"""Deterministic relay through landmark-defined local coordinate frames."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1400, 720
SIZE = 7
START_X, START_Y, START_CELL, START_GAP = 18, 145, 24, 3
CARD_XS, CARD_Y = (225, 615, 1005), 64
CARD_FROM_DX, CARD_TO_DX = 18, 204
FRAME_Y, FRAME_CELL, FRAME_GAP = 151, 20, 3
OPTION_XS = (68, 348, 628, 908, 1188)
OPTION_Y, OPTION_CELL, OPTION_GAP = 548, 17, 3

BACKGROUND = "#ffffff"
INK = "#20364a"
PAPER = "#f7f3e8"
BORDER = "#6d7e8b"
PALE = "#dce4e8"
RED = "#d32f2f"
GOLD = "#e29900"
CYAN = "#0089b4"
VIOLET = "#7b1fa2"
FRAME_NAMES = ("origin", "right", "up")
FRAME_COLORS = {"origin": GOLD, "right": CYAN, "up": VIOLET}
FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"

# Ordered (local +RIGHT, local +UP) screen-grid unit vectors. Both handednesses
# are allowed, so a card can encode a rotation or a reflection plus translation.
ORIENTATIONS = tuple(
    (u, v)
    for u in ((0, 1), (1, 0), (0, -1), (-1, 0))
    for v in ((-u[1], u[0]), (u[1], -u[0]))
)


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(FONT_PATH, size=size)


def _point(raw: object) -> tuple[int, int]:
    if not isinstance(raw, list) or len(raw) != 2:
        raise ValueError("a cell must be a two-integer list")
    row, col = int(raw[0]), int(raw[1])
    if [row, col] != raw or not (0 <= row < SIZE and 0 <= col < SIZE):
        raise ValueError("cell coordinate is invalid")
    return row, col


def _token(raw: object) -> tuple[tuple[int, int], tuple[int, int]]:
    if not isinstance(raw, list) or len(raw) != 2:
        raise ValueError("a token needs navy tail then red head")
    tail, head = _point(raw[0]), _point(raw[1])
    if abs(tail[0] - head[0]) + abs(tail[1] - head[1]) != 1:
        raise ValueError("token dots must occupy adjacent cells")
    return tail, head


def _frame(raw: object) -> dict[str, tuple[int, int]]:
    if not isinstance(raw, dict) or set(raw) != set(FRAME_NAMES):
        raise ValueError("a frame needs origin, right, and up landmarks")
    frame = {name: _point(raw[name]) for name in FRAME_NAMES}
    if len(set(frame.values())) != 3:
        raise ValueError("frame landmarks must be distinct")
    o = frame["origin"]
    u = (frame["right"][0] - o[0], frame["right"][1] - o[1])
    v = (frame["up"][0] - o[0], frame["up"][1] - o[1])
    if (u, v) not in ORIENTATIONS:
        raise ValueError("axis landmarks must define perpendicular unit directions")
    return frame


def _basis(frame: dict[str, tuple[int, int]]) -> tuple[tuple[int, int], tuple[int, int]]:
    # Index through the declared ordered landmark names so the static
    # rendered-field audit treats the nested frame as part of frame_cards,
    # rather than mistaking its member names for top-level latent fields.
    o = frame[FRAME_NAMES[0]]
    return (
        (frame[FRAME_NAMES[1]][0] - o[0], frame[FRAME_NAMES[1]][1] - o[1]),
        (frame[FRAME_NAMES[2]][0] - o[0], frame[FRAME_NAMES[2]][1] - o[1]),
    )


def _transport_point(point: tuple[int, int], source: dict, target: dict) -> tuple[int, int]:
    su, sv = _basis(source)
    source_origin = source[FRAME_NAMES[0]]
    delta = (point[0] - source_origin[0], point[1] - source_origin[1])
    right_offset = delta[0] * su[0] + delta[1] * su[1]
    up_offset = delta[0] * sv[0] + delta[1] * sv[1]
    tu, tv = _basis(target)
    return (
        target[FRAME_NAMES[0]][0] + right_offset * tu[0] + up_offset * tv[0],
        target[FRAME_NAMES[0]][1] + right_offset * tu[1] + up_offset * tv[1],
    )


def _transport_token(token, source: dict, target: dict):
    return _transport_point(token[0], source, target), _transport_point(token[1], source, target)


def _valid_token(token) -> bool:
    return all(0 <= row < SIZE and 0 <= col < SIZE for row, col in token)


def _parse_scene(scene: dict):
    if set(scene) != {"start_token", "frame_cards", "options"}:
        raise ValueError("scene fields must be start_token, frame_cards, and options")
    start = _token(scene["start_token"])
    raw_cards = scene["frame_cards"]
    if not isinstance(raw_cards, list) or len(raw_cards) != 3:
        raise ValueError("exactly three frame cards are required")
    cards = []
    for raw in raw_cards:
        if not isinstance(raw, list) or len(raw) != 2:
            raise ValueError("a card needs FROM then TO frame")
        source, target = _frame(raw[0]), _frame(raw[1])
        if source == target:
            raise ValueError("a frame card must be nonidentity")
        cards.append((source, target))
    raw_options = scene["options"]
    if not isinstance(raw_options, list) or len(raw_options) != 5:
        raise ValueError("exactly five options are required")
    options = [_token(raw) for raw in raw_options]
    if len(set(options)) != 5:
        raise ValueError("options must be distinct")
    return start, cards, options


def _evolve(token, cards: Iterable[tuple[dict, dict]]):
    state = token
    for source, target in cards:
        state = _transport_token(state, source, target)
        if not _valid_token(state):
            raise ValueError("transported token leaves the board")
    return state


def _json_token(token) -> list[list[int]]:
    return [list(token[0]), list(token[1])]


def _json_frame(origin, orientation) -> dict[str, list[int]]:
    u, v = orientation
    return {
        "origin": list(origin),
        "right": [origin[0] + u[0], origin[1] + u[1]],
        "up": [origin[0] + v[0], origin[1] + v[1]],
    }


def _all_frames() -> list[dict[str, tuple[int, int]]]:
    frames = []
    for orientation in ORIENTATIONS:
        u, v = orientation
        for row in range(SIZE):
            for col in range(SIZE):
                points = ((row, col), (row + u[0], col + u[1]), (row + v[0], col + v[1]))
                if all(0 <= r < SIZE and 0 <= c < SIZE for r, c in points):
                    frames.append(_frame(_json_frame((row, col), orientation)))
    return frames


ALL_FRAMES = _all_frames()


def _det(frame: dict) -> int:
    u, v = _basis(frame)
    return u[0] * v[1] - u[1] * v[0]


def _sample_card(rng: random.Random, state):
    for _ in range(1200):
        source, target = rng.sample(ALL_FRAMES, 2)
        if source["origin"] == target["origin"] or _basis(source) == _basis(target):
            continue
        result = _transport_token(state, source, target)
        if _valid_token(result) and result != state:
            return (source, target), result
    raise RuntimeError("could not sample a usable local-frame map")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    oriented_tokens = []
    for row in range(SIZE):
        for col in range(SIZE):
            for dr, dc in ((-1, 0), (0, 1), (1, 0), (0, -1)):
                head = (row + dr, col + dc)
                token = ((row, col), head)
                if _valid_token(token):
                    oriented_tokens.append(token)
    for _ in range(1200):
        start = rng.choice(oriented_tokens)
        cards = []
        states = [start]
        try:
            for _step in range(3):
                card, state = _sample_card(rng, states[-1])
                cards.append(card)
                states.append(state)
        except RuntimeError:
            continue
        chiralities = {_det(a) * _det(b) for a, b in cards}
        if chiralities != {-1, 1} or len(set(states)) != 4:
            continue
        target = states[-1]
        preferred = [states[-2], states[1], (target[1], target[0])]
        try:
            reverse_order = _evolve(start, reversed(cards))
            preferred.append(reverse_order)
        except ValueError:
            pass
        try:
            inverse_state = start
            for source, destination in cards:
                inverse_state = _transport_token(inverse_state, destination, source)
                if not _valid_token(inverse_state):
                    raise ValueError
            preferred.append(inverse_state)
        except ValueError:
            pass
        distractors = []
        for token in preferred:
            if _valid_token(token) and token != target and token not in distractors:
                distractors.append(token)
        fallback = oriented_tokens[:]
        rng.shuffle(fallback)
        for token in fallback:
            if token != target and token not in distractors:
                distractors.append(token)
            if len(distractors) >= 4:
                break
        rng.shuffle(distractors)
        raw_cards = [[{k: list(v) for k, v in source.items()}, {k: list(v) for k, v in target_frame.items()}]
                     for source, target_frame in cards]
        options = [_json_token(token) for token in distractors[:4]]
        options.insert(int(seed) % 5, _json_token(target))
        scene = {"start_token": _json_token(start), "frame_cards": raw_cards, "options": options}
        _parse_scene(scene)
        return scene
    raise RuntimeError("could not sample a nondegenerate local-frame relay")


def analytic_gold(scene: dict) -> str:
    start, cards, options = _parse_scene(scene)
    target = _evolve(start, cards)
    matches = [i for i, option in enumerate(options) if option == target]
    if len(matches) != 1:
        raise ValueError("final token must match exactly one option")
    return "ABCDE"[matches[0]]


def margin(scene: dict) -> float:
    start, cards, options = _parse_scene(scene)
    target = _evolve(start, cards)
    return min(float(sum(a != b for a, b in zip(target, option))) for option in options if option != target)


def is_quarantined(scene: dict) -> bool:
    try:
        start, cards, options = _parse_scene(scene)
        target = _evolve(start, cards)
        return sum(option == target for option in options) != 1 or margin(scene) < 1.0
    except (TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    _parse_scene(scene)
    return []


def _center(point, x: int, y: int, cell: int, gap: int) -> tuple[int, int]:
    row, col = point
    return x + col * (cell + gap) + cell // 2, y + row * (cell + gap) + cell // 2


def _grid(draw: ImageDraw.ImageDraw, x: int, y: int, cell: int, gap: int) -> None:
    for row in range(SIZE):
        for col in range(SIZE):
            left, top = x + col * (cell + gap), y + row * (cell + gap)
            draw.rectangle((left, top, left + cell, top + cell), fill=PAPER, outline=BORDER, width=2)


def _dot(draw, point, color, x, y, cell, gap) -> None:
    cx, cy = _center(point, x, y, cell, gap)
    radius = max(6, cell // 3)
    draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=color, outline="white", width=2)


def _draw_token(draw, token, x, y, cell, gap) -> None:
    _grid(draw, x, y, cell, gap)
    _dot(draw, token[0], INK, x, y, cell, gap)
    _dot(draw, token[1], RED, x, y, cell, gap)


def _draw_frame(draw, frame, x, y) -> None:
    _grid(draw, x, y, FRAME_CELL, FRAME_GAP)
    oc = _center(frame["origin"], x, y, FRAME_CELL, FRAME_GAP)
    rc = _center(frame["right"], x, y, FRAME_CELL, FRAME_GAP)
    uc = _center(frame["up"], x, y, FRAME_CELL, FRAME_GAP)
    draw.line((oc, rc), fill=CYAN, width=5)
    draw.line((oc, uc), fill=VIOLET, width=5)
    _dot(draw, frame["right"], CYAN, x, y, FRAME_CELL, FRAME_GAP)
    _dot(draw, frame["up"], VIOLET, x, y, FRAME_CELL, FRAME_GAP)
    cx, cy = oc
    radius = 7
    draw.rectangle((cx - radius, cy - radius, cx + radius, cy + radius), fill=GOLD, outline="white", width=2)


def render(scene: dict) -> Image.Image:
    start, cards, options = _parse_scene(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((WIDTH // 2, 25), "KEEP LOCAL OFFSETS  •  APPLY 1 → 2 → 3", fill=INK, font=_font(25), anchor="mm")
    draw.rounded_rectangle((10, CARD_Y, 210, 410), radius=12, fill="#f8fafb", outline=PALE, width=3)
    draw.text((110, 91), "START", fill=INK, font=_font(21), anchor="mm")
    draw.text((110, 120), "NAVY TAIL • RED HEAD", fill=BORDER, font=_font(12), anchor="mm")
    _draw_token(draw, start, START_X, START_Y, START_CELL, START_GAP)
    for number, (x, card) in enumerate(zip(CARD_XS, cards), 1):
        draw.rounded_rectangle((x, CARD_Y, x + 380, 410), radius=12, fill="#f8fafb", outline=PALE, width=3)
        draw.text((x + 190, 89), f"CARD {number}", fill=INK, font=_font(20), anchor="mm")
        draw.text((x + 97, 121), "FROM", fill=BORDER, font=_font(14), anchor="mm")
        draw.text((x + 283, 121), "TO", fill=BORDER, font=_font(14), anchor="mm")
        draw.text((x + 190, 230), "→", fill=INK, font=_font(25), anchor="mm")
        _draw_frame(draw, card[0], x + CARD_FROM_DX, FRAME_Y)
        _draw_frame(draw, card[1], x + CARD_TO_DX, FRAME_Y)
        draw.text((x + 190, 376), "GOLD origin  •  CYAN +RIGHT  •  VIOLET +UP", fill=BORDER, font=_font(11), anchor="mm")
    draw.line((25, 435, WIDTH - 25, 435), fill=PALE, width=2)
    draw.text((WIDTH // 2, 465), "FINAL TOKEN", fill=BORDER, font=_font(20), anchor="mm")
    for label, token, x in zip("ABCDE", options, OPTION_XS):
        cx, cy = x + 68, 515
        draw.ellipse((cx - 21, cy - 21, cx + 21, cy + 21), fill=INK)
        draw.text((cx, cy), label, fill="white", font=_font(23), anchor="mm")
        _draw_token(draw, token, x, OPTION_Y, OPTION_CELL, OPTION_GAP)
    return image
