"""Deterministic scene sampler, raster renderer, and analytic arm.

Four ordered interval codes live on consecutive distant sections of one long
thread. A code is a permutation of 1,2,3,4; multiplying all entries by a
section-specific scale changes absolute size while preserving correspondence.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 1000, 420
BACKGROUND = (248, 247, 242)
THREAD = (38, 57, 76)
CORE = (0, 0, 0)
CHECKPOINT = (159, 214, 207)
GOLD = (226, 174, 43)
TAG_COLORS = {
    "red": (210, 67, 62),
    "green": (48, 151, 101),
    "blue": (55, 103, 184),
}

CODES = (
    (1, 2, 3, 4), (1, 2, 4, 3), (1, 3, 2, 4), (1, 3, 4, 2),
    (1, 4, 2, 3), (1, 4, 3, 2), (2, 1, 3, 4), (2, 1, 4, 3),
    (2, 3, 1, 4), (2, 4, 1, 3), (3, 1, 2, 4), (4, 1, 2, 3),
)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    reference = list(CODES[int(seed) % len(CODES)])
    answer_color = ("red", "green", "blue")[int(seed) % 3]
    match_slot = (int(seed) // 3) % 3

    distractors = [list(code) for code in CODES if list(code) != reference]
    rng.shuffle(distractors)
    candidate_codes = [reference[:] if slot == match_slot else distractors.pop() for slot in range(3)]

    other_colors = [name for name in ("red", "green", "blue") if name != answer_color]
    rng.shuffle(other_colors)
    candidate_colors = [answer_color if slot == match_slot else other_colors.pop() for slot in range(3)]

    return {
        "reference_code": reference,
        "candidate_codes": candidate_codes,
        "candidate_colors": candidate_colors,
        "section_scales": [rng.randint(17, 19) for _ in range(4)],
        "section_gap": rng.randint(22, 25),
        "y_offset": rng.randint(-18, 18),
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
        "thread_color": list(THREAD),
        "checkpoint_color": list(CHECKPOINT),
    }


def _diamond(draw: ImageDraw.ImageDraw, center: tuple[int, int], radius: int, color: tuple[int, int, int]) -> None:
    x, y = center
    draw.polygon(((x, y - radius), (x + radius, y), (x, y + radius), (x - radius, y)), fill=color)


def _section_markers(code: list[int], scale: int, x: int, y: int) -> list[tuple[int, int]]:
    points = [(x, y)]
    for unit in code:
        x += int(unit) * int(scale)
        points.append((x, y))
    if not all(50 <= px <= 925 for px, _ in points):
        raise ValueError("section does not fit the visible thread")
    return points


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    if (width, height) != (WIDTH, HEIGHT):
        raise ValueError("unsupported canvas")
    background = tuple(int(v) for v in scene["background"])
    thread_color = tuple(int(v) for v in scene["thread_color"])
    checkpoint_color = tuple(int(v) for v in scene["checkpoint_color"])
    codes = [scene["reference_code"], *scene["candidate_codes"]]
    colors = ["gold", *scene["candidate_colors"]]
    scales = [int(v) for v in scene["section_scales"]]
    gap = int(scene["section_gap"])
    y = 210 + int(scene["y_offset"])
    marker_rows = []
    next_x = 65
    for index in range(4):
        points = _section_markers(list(codes[index]), scales[index], next_x, y)
        marker_rows.append(points)
        next_x = points[-1][0] + gap
    if marker_rows[-1][-1][0] > 925:
        raise ValueError("all four rhythms must fit the long visible thread")

    route: list[tuple[int, int]] = [(40, y), (960, y)]

    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)
    draw.line(route, fill=thread_color, width=9, joint="curve")

    for row, points in enumerate(marker_rows):
        color = GOLD if colors[row] == "gold" else TAG_COLORS[str(colors[row])]
        sx, sy = points[0]
        draw.rounded_rectangle((sx - 7, sy - 7, sx + 7, sy + 7), radius=2, fill=color)
        for px, py in points[1:4]:
            draw.ellipse((px - 5, py - 5, px + 5, py + 5), fill=checkpoint_color)
        _diamond(draw, points[4], 7, color)

    # Repaint a one-pixel centerline after landmarks.  It makes connectivity a
    # pixel-observable prerequisite rather than an assumption about the scene.
    draw.line(route, fill=CORE, width=1)
    return image


def _code_dist(a: tuple[int, ...], b: tuple[int, ...]) -> int:
    return sum(abs(x - y) for x, y in zip(a, b))


def analytic_gold(scene: dict) -> str:
    reference = tuple(int(v) for v in scene["reference_code"])
    candidates = [tuple(int(v) for v in code) for code in scene["candidate_codes"]]
    colors = [str(value) for value in scene["candidate_colors"]]
    matches = [index for index, code in enumerate(candidates) if code == reference]
    if len(matches) != 1:
        raise ValueError("scene lacks one unique interval-rhythm match")
    return colors[matches[0]]


def margin(scene: dict) -> float:
    reference = tuple(int(v) for v in scene["reference_code"])
    candidates = [tuple(int(v) for v in code) for code in scene["candidate_codes"]]
    distances = sorted(_code_dist(reference, candidate) for candidate in candidates)
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    reference = tuple(int(v) for v in scene["reference_code"])
    candidates = [tuple(int(v) for v in code) for code in scene["candidate_codes"]]
    distances = sorted(_code_dist(reference, candidate) for candidate in candidates)
    return distances.count(0) != 1 or distances[1] <= 0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
