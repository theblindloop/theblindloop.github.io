"""Independent raster-only inverse arm for center-heart counting."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    seen = np.zeros(mask.shape, dtype=bool)
    out = []
    height, width = mask.shape
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        area = 0
        min_x = max_x = int(x)
        min_y = max_y = int(y)
        while queue:
            cy, cx = queue.popleft()
            area += 1
            min_x, max_x = min(min_x, cx), max(max_x, cx)
            min_y, max_y = min(min_y, cy), max(max_y, cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        out.append((area, min_x, min_y, max_x, max_y))
    return out


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    white = (arr[:, :, 0] > 235) & (arr[:, :, 1] > 235) & (arr[:, :, 2] > 230)
    ys, xs = np.nonzero(white)
    if len(xs) < 40000:
        raise ValueError("card face is missing or too corrupted")
    left, right = int(np.percentile(xs, 1)), int(np.percentile(xs, 99))
    top, bottom = int(np.percentile(ys, 1)), int(np.percentile(ys, 99))
    if right - left < 250 or bottom - top < 340:
        raise ValueError("could not recover a full playing-card face")
    red = (arr[:, :, 0] > 155) & (arr[:, :, 1] < 105) & (arr[:, :, 2] < 125)
    roi = np.zeros_like(red)
    roi[top + 52 : bottom - 52, left + 66 : right - 66] = True
    comps = [c for c in _components(red & roi) if c[0] >= 170 and c[3] - c[1] >= 18 and c[4] - c[2] >= 18]
    count = len(comps)
    if count not in (4, 5, 6, 7):
        raise ValueError(f"ambiguous center-heart component count: {count}")
    areas = [c[0] for c in comps]
    if min(areas) < 0.55 * max(areas):
        raise ValueError("center-heart component areas are inconsistent")
    return str(count)
