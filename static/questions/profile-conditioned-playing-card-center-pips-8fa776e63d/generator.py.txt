"""Deterministic playing-card prior-conflict renderer."""

from __future__ import annotations

import math
import random
from PIL import Image, ImageDraw, ImageFont

WIDTH = 512
HEIGHT = 512
CARD_W = 320
CARD_H = 420
SS = 3


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    pip_count = 4 + (int(seed) % 4)
    # Alternate matched controls and counterfactuals, while crossing every rank
    # with multiple answers over sequential deterministic samples.
    if (int(seed) // 4) % 2 == 0:
        rank = pip_count
    else:
        rank = 4 + ((pip_count - 4 + 1 + rng.randrange(3)) % 4)
    ink_budget = (3800, 4400, 5000)[rng.randrange(3)]
    pip_size = round(math.sqrt(ink_budget / (2.05 * pip_count)), 2)
    return {
        "pip_count": pip_count,
        "rank": rank,
        "pip_size": pip_size,
        "layout": rng.randrange(4),
        "card_x": 96 + rng.randrange(-16, 17),
        "card_y": 46 + rng.randrange(-10, 11),
        "accent": rng.choice(["plain", "classic"]),
    }


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default(size=size)


def _heart(draw: ImageDraw.ImageDraw, cx: float, cy: float, size: float, fill: tuple[int, int, int]) -> None:
    """Draw one connected heart from two lobes and a lower diamond."""
    r = size * 0.34
    draw.ellipse((cx - 0.66 * size, cy - 0.46 * size, cx + 0.02 * size, cy + 0.22 * size), fill=fill)
    draw.ellipse((cx - 0.02 * size, cy - 0.46 * size, cx + 0.66 * size, cy + 0.22 * size), fill=fill)
    draw.polygon(
        [(cx - 0.64 * size, cy - 0.04 * size), (cx + 0.64 * size, cy - 0.04 * size), (cx, cy + 0.88 * size)],
        fill=fill,
    )


def _pip_positions(count: int, layout: int) -> list[tuple[float, float]]:
    # Normalized positions inside the card. All layouts keep wide visible gaps.
    xs = (0.36, 0.64)
    if count == 4:
        pts = [(xs[0], 0.34), (xs[1], 0.34), (xs[0], 0.66), (xs[1], 0.66)]
    elif count == 5:
        pts = [(xs[0], 0.31), (xs[1], 0.31), (0.50, 0.50), (xs[0], 0.69), (xs[1], 0.69)]
    elif count == 6:
        pts = [(x, y) for y in (0.29, 0.50, 0.71) for x in xs]
    else:
        pts = [(x, y) for y in (0.27, 0.50, 0.73) for x in xs] + [(0.50, 0.50)]
    dx = (-0.012, 0.012, 0.0, 0.0)[layout]
    dy = (0.0, 0.0, -0.012, 0.012)[layout]
    return [(x + dx, y + dy) for x, y in pts]


def render(scene: dict) -> Image.Image:
    scale = SS
    backgrounds = ((42, 73, 98), (63, 77, 101), (48, 91, 88), (76, 65, 91))
    bg = backgrounds[int(scene["layout"]) % len(backgrounds)]
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), bg)
    draw = ImageDraw.Draw(image)
    x0 = int(scene["card_x"]) * scale
    y0 = int(scene["card_y"]) * scale
    x1 = x0 + CARD_W * scale
    y1 = y0 + CARD_H * scale
    radius = 18 * scale
    draw.rounded_rectangle((x0, y0, x1, y1), radius=radius, fill=(253, 252, 248), outline=(25, 25, 28), width=4 * scale)

    red = (199, 35, 55)
    rank = str(scene["rank"])
    font = _font(34 * scale)
    draw.text((x0 + 22 * scale, y0 + 17 * scale), rank, font=font, fill=(25, 25, 28), anchor="la")
    _heart(draw, x0 + 39 * scale, y0 + 76 * scale, 13 * scale, red)
    draw.text((x1 - 22 * scale, y1 - 17 * scale), rank, font=font, fill=(25, 25, 28), anchor="rs")
    _heart(draw, x1 - 39 * scale, y1 - 76 * scale, 13 * scale, red)

    size = float(scene["pip_size"]) * scale
    for nx, ny in _pip_positions(int(scene["pip_count"]), int(scene["layout"])):
        _heart(draw, x0 + nx * CARD_W * scale, y0 + ny * CARD_H * scale, size, red)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["pip_count"])


def margin(scene: dict) -> float:
    pts = _pip_positions(int(scene["pip_count"]), int(scene["layout"]))
    size = float(scene["pip_size"])
    centers = [(x * CARD_W, y * CARD_H) for x, y in pts]
    pair_gap = min(
        math.dist(a, b) - 1.35 * size
        for i, a in enumerate(centers)
        for b in centers[i + 1 :]
    )
    edge_gap = min(
        min(x - 68, CARD_W - 68 - x, y - 54, CARD_H - 54 - y) - 0.9 * size
        for x, y in centers
    )
    return round(min(pair_gap, edge_gap), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["accent"] = "classic" if scene["accent"] == "plain" else "plain"
    return [("unused_accent_alias", twin)]
