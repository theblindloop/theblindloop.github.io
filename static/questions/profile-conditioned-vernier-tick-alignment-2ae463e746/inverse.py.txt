"""Independent pixel-only arm for ``vernier_tick_alignment``.

Input is the final RGB raster and nothing else.  This module imports no
renderer, prompt, generator, verifier or analytic-gold code, receives no latent
scene, and knows no scene parameter: every quantity below (rail rows, tick
centres, pitches, the coincidence) is rediscovered from the image.

Method: locate the two long horizontal rails as the image rows that are almost
entirely inked; take the band strictly above the upper rail and the band
strictly below the lower rail; in each band accumulate an intensity-weighted
column profile and split it into connected runs, one per tick; take each run's
intensity-weighted centroid as a sub-pixel tick centre; then report the lower
tick whose distance to the nearest upper tick is smallest.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "vernier_tick_alignment-oracle-1.0.0"

DARK = 128.0  # a pixel this dark or darker counts as ink for rail finding
RAIL_COVERAGE = 0.5  # a rail row is inked across at least this fraction of the width
COLUMN_EPS = 2.0  # column profile mass below this is background
ABSTAIN_PX = 0.8  # refuse to decide below this runner-up margin, in pixels
MIN_LOWER_TICKS = 3
MIN_UPPER_TICKS = 2

ABSTAIN = "abstain"


def _rail_bands(dark_rows: np.ndarray) -> list[tuple[int, int]]:
    bands: list[tuple[int, int]] = []
    start = None
    for row, flag in enumerate(dark_rows):
        if flag and start is None:
            start = row
        elif not flag and start is not None:
            bands.append((start, row - 1))
            start = None
    if start is not None:
        bands.append((start, len(dark_rows) - 1))
    return bands


def _centres(profile: np.ndarray) -> list[float]:
    """Sub-pixel centroids of the connected runs of a column profile."""
    centres: list[float] = []
    active = profile > COLUMN_EPS
    start = None
    for column, flag in enumerate(active):
        if flag and start is None:
            start = column
        elif not flag and start is not None:
            centres.append(_centroid(profile, start, column))
            start = None
    if start is not None:
        centres.append(_centroid(profile, start, len(active)))
    return centres


def _centroid(profile: np.ndarray, start: int, stop: int) -> float:
    weights = profile[start:stop]
    columns = np.arange(start, stop, dtype=float) + 0.5
    return float((columns * weights).sum() / weights.sum())


def decision_from_image(image: Image.Image) -> str:
    gray = np.asarray(image.convert("L"), dtype=float)
    height, width = gray.shape
    ink = 255.0 - gray

    dark_rows = (gray <= DARK).sum(axis=1) >= RAIL_COVERAGE * width
    rails = [band for band in _rail_bands(dark_rows) if band[1] - band[0] >= 1]
    if len(rails) != 2:
        return ABSTAIN
    (upper_top, _), (_, lower_bottom) = rails[0], rails[1]

    above = ink[: max(upper_top - 1, 0), :]
    below = ink[min(lower_bottom + 2, height) :, :]
    if above.size == 0 or below.size == 0:
        return ABSTAIN

    upper = _centres(above.sum(axis=0))
    lower = _centres(below.sum(axis=0))
    if len(upper) < MIN_UPPER_TICKS or len(lower) < MIN_LOWER_TICKS:
        return ABSTAIN

    residuals = [min(abs(x - u) for u in upper) for x in lower]
    order = sorted(range(len(residuals)), key=lambda j: residuals[j])
    if residuals[order[1]] - residuals[order[0]] < ABSTAIN_PX:
        return ABSTAIN
    return str(order[0] + 1)
