"""Scene sampler, renderer and analytic gold for ``vernier_tick_alignment``.

Two horizontal rails carry two rows of evenly spaced ticks with *different*
pitches: the upper row points up, the lower row points down.  Because the two
pitches differ, at most one lower tick can sit vertically above/below an upper
tick; every other lower tick is off by a multiple of the pitch difference.  The
answer names that coincident lower tick by its rank from the left.

All geometry is sampled and stored in supersample units (``SS`` sub-pixels per
final pixel) as integers, so the analytic gold is computed on exactly the
coordinates that are painted: there is no rounding between scene and raster.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "vernier_tick_alignment-renderer-1.0.0"

SS = 3  # supersample factor; scene coordinates are integers in these units
WIDTH, HEIGHT = 820, 320
W3, H3 = WIDTH * SS, HEIGHT * SS

LOWER_TICKS = 7  # number of ticks on the lower (reading) rail

UPPER_RAIL_TOP, UPPER_RAIL_BOT = 128, 130  # final-pixel rows, inclusive
LOWER_RAIL_TOP, LOWER_RAIL_BOT = 190, 192
RAIL_X0, RAIL_X1 = 18, WIDTH - 18
TICK_HALF_W3 = 4  # ticks are 9 sub-pixels = 3 final pixels wide
TICK_MIN_H, TICK_MAX_H = 26, 48

INK = (18, 18, 18)
DECOR = (186, 186, 186)
BACKGROUND = (255, 255, 255)

QUARANTINE_PX = 3.0  # a scene is quarantined below this margin, in final pixels
CLEAN_MIN_PX = 4.0  # off-quarantine scenes are sampled at or above this margin
NEAR_BAND_SUB = (5, 8)  # deliberate near-boundary band, in sub-pixels of margin
NEAR_BOUNDARY_RATE = 0.22


def _upper_positions(scene: dict) -> list[int]:
    """Sub-pixel x centres of the upper ticks, left to right."""
    return [scene["upper_phase"] + i * scene["upper_pitch"] for i in range(scene["upper_count"])]


def _lower_positions(scene: dict) -> list[int]:
    """Sub-pixel x centres of the lower ticks, left to right."""
    return [scene["lower_start"] + j * scene["lower_pitch"] for j in range(LOWER_TICKS)]


def _residuals(scene: dict) -> list[int]:
    """For every lower tick, the sub-pixel distance to the nearest upper tick."""
    uppers = _upper_positions(scene)
    return [min(abs(x - u) for u in uppers) for x in _lower_positions(scene)]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 2654435761 + 12345)

    upper_pitch = rng.randrange(56 * SS, 72 * SS + 1)
    # The pitch difference must stay below upper_pitch / LOWER_TICKS so that the
    # residuals grow strictly with the distance from the coincident tick and no
    # second coincidence can appear inside the lower row.
    delta_hi = min(upper_pitch // LOWER_TICKS - 2, 9 * SS)
    delta = rng.randrange(5 * SS, delta_hi + 1)
    sign = rng.choice((-1, 1))  # the lower row may be finer or coarser
    lower_pitch = upper_pitch + sign * delta

    near_boundary = rng.random() < NEAR_BOUNDARY_RATE
    if near_boundary:
        lo = max((delta - NEAR_BAND_SUB[1] + 1) // 2, 0)
        hi = max((delta - NEAR_BAND_SUB[0]) // 2, 0)
        offset_abs = rng.randint(min(lo, hi), hi)
    else:
        offset_abs = rng.randint(0, max((delta - int(CLEAN_MIN_PX * SS)) // 2, 0))
    offset = offset_abs * rng.choice((-1, 1))

    winner = rng.randrange(LOWER_TICKS)  # 0-based rank from the left

    span = (LOWER_TICKS - 1) * lower_pitch
    slack = (W3 - 2 * 46 * SS) - span
    lower_start = 46 * SS + rng.randint(0, max(slack, 0))

    # Pin the upper grid so that the winning lower tick misses an upper tick by
    # exactly ``offset`` sub-pixels; every other lower tick misses by more.
    anchor = lower_start + winner * lower_pitch - offset
    phase = anchor % upper_pitch
    while phase < 10 * SS:
        phase += upper_pitch
    upper_count = (W3 - 10 * SS - phase) // upper_pitch + 1

    heights_u = [rng.randrange(TICK_MIN_H, TICK_MAX_H + 1) for _ in range(upper_count)]
    heights_l = [rng.randrange(TICK_MIN_H, TICK_MAX_H + 1) for _ in range(LOWER_TICKS)]
    decor = [
        (rng.randrange(40 * SS, (WIDTH - 40) * SS), rng.randrange(8, 20))
        for _ in range(rng.randrange(2, 6))
    ]

    return {
        "upper_pitch": upper_pitch,
        "upper_phase": phase,
        "upper_count": upper_count,
        "lower_pitch": lower_pitch,
        "lower_start": lower_start,
        "upper_heights": heights_u,
        "lower_heights": heights_l,
        "decor": decor,
    }


def render(scene: dict) -> Image.Image:
    big = Image.new("RGB", (W3, H3), BACKGROUND)
    draw = ImageDraw.Draw(big)

    for top, bottom in ((UPPER_RAIL_TOP, UPPER_RAIL_BOT), (LOWER_RAIL_TOP, LOWER_RAIL_BOT)):
        draw.rectangle(
            [RAIL_X0 * SS, top * SS, RAIL_X1 * SS + SS - 1, bottom * SS + SS - 1], fill=INK
        )

    for x, height in zip(_upper_positions(scene), scene["upper_heights"]):
        draw.rectangle(
            [
                x - TICK_HALF_W3,
                (UPPER_RAIL_TOP - height) * SS,
                x + TICK_HALF_W3,
                UPPER_RAIL_TOP * SS - 1,
            ],
            fill=INK,
        )
    for x, height in zip(_lower_positions(scene), scene["lower_heights"]):
        draw.rectangle(
            [
                x - TICK_HALF_W3,
                (LOWER_RAIL_BOT + 1) * SS,
                x + TICK_HALF_W3,
                (LOWER_RAIL_BOT + 1 + height) * SS - 1,
            ],
            fill=INK,
        )

    for x, height in scene["decor"]:
        mid = (UPPER_RAIL_BOT + LOWER_RAIL_TOP) // 2
        draw.rectangle(
            [x - TICK_HALF_W3, (mid - height) * SS, x + TICK_HALF_W3, (mid + height) * SS - 1],
            fill=DECOR,
        )

    return big.resize((WIDTH, HEIGHT), Image.BOX)


def analytic_gold(scene: dict) -> str:
    residuals = _residuals(scene)
    return str(min(range(LOWER_TICKS), key=lambda j: residuals[j]) + 1)


def margin(scene: dict) -> float:
    """Runner-up decision margin, in final pixels."""
    ordered = sorted(_residuals(scene))
    return (ordered[1] - ordered[0]) / SS


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist: every field is painted, so this list is empty."""
    return []


# --- counterbalance diagnostics (not used by gold) ---------------------------


def tallest_lower_index(scene: dict) -> str:
    heights = scene["lower_heights"]
    return str(max(range(LOWER_TICKS), key=lambda j: (heights[j], -j)) + 1)


def lower_is_finer(scene: dict) -> bool:
    return scene["lower_pitch"] < scene["upper_pitch"]


def nearest_decor_index(scene: dict) -> str:
    """Which lower tick is closest to a decorative grey mark (a cheap cue check)."""
    positions = _lower_positions(scene)
    if not scene["decor"]:
        return "0"
    best = min(
        range(LOWER_TICKS), key=lambda j: min(abs(positions[j] - d[0]) for d in scene["decor"])
    )
    return str(best + 1)
