"""Deterministic line-ring incidence sampler and Pillow renderer."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 760, 280
PANEL_CENTERS = (105, 285, 465, 645)
PANEL_TOP, PANEL_BOTTOM = 57, 235
RADIUS = 13
LINE_COLORS = ("#2457a6", "#18856f", "#9b3568")
AMBER = "#e3a018"
BACKGROUND = "#f4f1e8"


def _distance(line: dict, point: list[float]) -> float:
    # Line is y = intercept + slope * (x - cx), expressed in panel-local x.
    x, y = point
    return abs(y - (line["intercept"] + line["slope"] * x)) / math.sqrt(1.0 + line["slope"] ** 2)


def _incidence(panel: dict) -> list[list[bool]]:
    return [[_distance(line, ring) < RADIUS for ring in panel["rings"]] for line in panel["lines"]]


def _is_bijection(panel: dict) -> bool:
    relation = _incidence(panel)
    row_degrees = [sum(row) for row in relation]
    col_degrees = [sum(relation[i][j] for i in range(3)) for j in range(3)]
    return row_degrees == [1, 1, 1] and col_degrees == [1, 1, 1]


def _panel(rng: random.Random, assignment: list[int], near: bool) -> dict:
    lines = []
    bases = [88 + rng.randint(-3, 3), 145 + rng.randint(-3, 3), 202 + rng.randint(-3, 3)]
    slopes = [rng.uniform(-0.18, 0.18) for _ in range(3)]
    for intercept, slope, color in zip(bases, slopes, LINE_COLORS):
        lines.append({"intercept": intercept, "slope": round(slope, 6), "color": color})
    xs = [-45 + rng.randint(-2, 2), rng.randint(-2, 2), 45 + rng.randint(-2, 2)]
    rng.shuffle(xs)
    rings = []
    for j, (x, line_index) in enumerate(zip(xs, assignment)):
        line = lines[line_index]
        offset = 6.0 if near and j == 0 else rng.uniform(-1.5, 1.5)
        y = line["intercept"] + line["slope"] * x + offset
        rings.append([round(x, 3), round(y, 3)])
    return {"lines": lines, "rings": rings}


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    target = seed % 4
    # Every panel has exactly three rings and three full lines. Every ring is
    # crossed by one line; distractors duplicate one line degree and omit another.
    distractors = [[0, 0, 2], [0, 2, 2], [1, 1, 2]]
    rng.shuffle(distractors)
    panels = []
    di = 0
    for index in range(4):
        if index == target:
            assignment = [0, 1, 2]
            rng.shuffle(assignment)
            panel = _panel(rng, assignment, near=(seed % 7 == 0))
        else:
            assignment = list(distractors[di])
            di += 1
            # Relabel line identities so no fixed color is privileged.
            perm = [0, 1, 2]
            rng.shuffle(perm)
            assignment = [perm[value] for value in assignment]
            panel = _panel(rng, assignment, near=False)
        panels.append(panel)
    scene = {"panels": panels, "serialization_alias": seed % 2}
    if sum(_is_bijection(panel) for panel in panels) != 1 or is_quarantined(scene):
        raise RuntimeError(f"unsafe sampled scene for seed {seed}")
    return scene


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    label_font = ImageFont.load_default(size=18)
    title_font = ImageFont.load_default(size=17)
    draw.text((WIDTH // 2, 25), "ONE LINE PER RING — ONE RING PER LINE", fill="#263238", anchor="mm", font=title_font)
    for index, panel in enumerate(scene["panels"]):
        cx = PANEL_CENTERS[index]
        draw.rounded_rectangle((cx - 82, PANEL_TOP, cx + 82, PANEL_BOTTOM), radius=9, fill="#ffffff", outline="#66707b", width=2)
        for line in panel["lines"]:
            x0, x1 = -70, 70
            y0 = line["intercept"] + line["slope"] * x0
            y1 = line["intercept"] + line["slope"] * x1
            draw.line((cx + x0, y0, cx + x1, y1), fill=line["color"], width=5)
        # Rings are painted last so their full circular boundary stays visible.
        for x, y in panel["rings"]:
            draw.ellipse((cx + x - RADIUS, y - RADIUS, cx + x + RADIUS, y + RADIUS), outline=AMBER, width=5)
        draw.text((cx, 257), "ABCD"[index], fill="#263238", anchor="mm", font=label_font)
    return image


def analytic_gold(scene: dict) -> str:
    # Keep the gold entrypoint's scene access explicit for the rendered-field
    # audit; nested line and ring records are all part of the rendered panels.
    panels = scene["panels"]
    winners = []
    for panel_index, panel in enumerate(panels):
        relation = []
        for line in panel["lines"]:
            row = []
            for ring in panel["rings"]:
                x, y = ring
                distance = abs(y - (line["intercept"] + line["slope"] * x)) / math.sqrt(1.0 + line["slope"] ** 2)
                row.append(distance < RADIUS)
            relation.append(row)
        rows = [sum(row) for row in relation]
        cols = [sum(relation[i][j] for i in range(3)) for j in range(3)]
        if rows == [1, 1, 1] and cols == [1, 1, 1]:
            winners.append(panel_index)
    if len(winners) != 1:
        raise ValueError(f"expected one bijection, found {winners}")
    return "ABCD"[winners[0]]


def margin(scene: dict) -> float:
    clearances = []
    for panel in scene["panels"]:
        for line in panel["lines"]:
            for ring in panel["rings"]:
                clearances.append(abs(RADIUS - _distance(line, ring)))
    return min(clearances)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 5.0 or sum(_is_bijection(panel) for panel in scene["panels"]) != 1


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    for panel in twin["panels"]:
        panel["rings"].reverse()
        panel["lines"].reverse()
    twin["serialization_alias"] = 1 - int(scene["serialization_alias"])
    return [("reverse_storage_orders_and_flip_alias", twin)]
