"""Independent image-only inverse for line-ring bijection search."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

AMBER = np.array((227, 160, 24), dtype=np.int16)
LINE_RGB = (
    np.array((36, 87, 166), dtype=np.int16),
    np.array((24, 133, 111), dtype=np.int16),
    np.array((155, 53, 104), dtype=np.int16),
)


def _components(mask: np.ndarray) -> list[tuple[float, float, float, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]:
            continue
        q = deque([(y0, x0)])
        seen[y0, x0] = True
        xs, ys = [], []
        while q:
            y, x = q.popleft()
            xs.append(x); ys.append(y)
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = y + dy, x + dx
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    q.append((ny, nx))
        if len(xs) >= 100:
            radius = 0.25 * ((max(xs) - min(xs)) + (max(ys) - min(ys)))
            found.append((sum(xs) / len(xs), sum(ys) / len(ys), radius, len(xs)))
    return found


def _fit_line(mask: np.ndarray, left: float, right: float) -> tuple[np.ndarray, np.ndarray]:
    ys, xs = np.nonzero(mask)
    keep = (xs >= left) & (xs <= right)
    pts = np.column_stack((xs[keep], ys[keep])).astype(float)
    if len(pts) < 220:
        raise ValueError("insufficient colored line pixels")
    center = pts.mean(axis=0)
    _, _, vt = np.linalg.svd(pts - center, full_matrices=False)
    return center, vt[0]


def _point_line_distance(point: tuple[float, float], center: np.ndarray, direction: np.ndarray) -> float:
    delta = np.array(point) - center
    return abs(float(delta[0] * direction[1] - delta[1] * direction[0]))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    amber_mask = np.linalg.norm(arr - AMBER, axis=2) < 45.0
    rings = sorted(_components(amber_mask), key=lambda item: item[0])
    if len(rings) != 12:
        raise ValueError(f"expected 12 amber rings, found {len(rings)}")
    panels = [rings[i:i + 3] for i in range(0, 12, 3)]
    line_masks = [np.linalg.norm(arr - rgb, axis=2) < 45.0 for rgb in LINE_RGB]
    valid = []
    for panel in panels:
        xs = [ring[0] for ring in panel]
        left, right = min(xs) - 31.0, max(xs) + 31.0
        fits = [_fit_line(mask, left, right) for mask in line_masks]
        relation = []
        for center, direction in fits:
            relation.append([
                _point_line_distance((x, y), center, direction) < radius - 2.0
                for x, y, radius, _ in panel
            ])
        rows = [sum(row) for row in relation]
        cols = [sum(relation[i][j] for i in range(3)) for j in range(3)]
        valid.append(rows == [1, 1, 1] and cols == [1, 1, 1])
    winners = [i for i, value in enumerate(valid) if value]
    if len(winners) != 1:
        raise ValueError(f"ambiguous pixel incidence patterns: {valid}")
    return "ABCD"[winners[0]]
