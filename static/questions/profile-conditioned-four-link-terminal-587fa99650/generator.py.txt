"""Deterministic latent sampler and raster renderer for four_link_terminal."""

from __future__ import annotations

import copy
import math
import random
from collections import deque

from PIL import Image, ImageDraw, ImageFont

TARGET_HOPS = 4
TERMINAL_COLORS = ("red", "blue", "orange")

PALETTES = (
    {
        "background": "#faf9f5", "ink": "#263238", "neutral": "#eadfbe",
        "green": "#28a35c", "red": "#df4650", "blue": "#347fd5", "orange": "#ee8b2c",
    },
    {
        "background": "#f7fafc", "ink": "#30343b", "neutral": "#dce7dc",
        "green": "#218c57", "red": "#c94b55", "blue": "#356eb7", "orange": "#d77a25",
    },
    {
        "background": "#fffaf2", "ink": "#202a35", "neutral": "#eee0c7",
        "green": "#36a66a", "red": "#e2555e", "blue": "#437fd0", "orange": "#ed963c",
    },
)


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _smooth(t: float) -> float:
    return t * t * (3.0 - 2.0 * t)


def sample_scene(seed: int) -> dict:
    """Sample a balanced three-terminal tree with one unique four-hop branch."""
    rng = random.Random(int(seed))
    width, height = 640, 480
    dx = rng.randint(-14, 14)
    dy = rng.randint(-7, 7)
    root = (80.0 + dx, 242.0 + dy)
    rows = [102.0 + dy + rng.randint(-8, 8), 242.0 + dy + rng.randint(-8, 8), 382.0 + dy + rng.randint(-8, 8)]
    terminal_x = [558.0 + dx + rng.randint(-8, 8) for _ in rows]

    target_color = TERMINAL_COLORS[int(seed) % 3]
    target_row = (int(seed) // 3) % 3
    remaining_colors = [color for color in TERMINAL_COLORS if color != target_color]
    rng.shuffle(remaining_colors)
    row_colors = [None, None, None]
    row_colors[target_row] = target_color
    for row, color in zip((r for r in range(3) if r != target_row), remaining_colors):
        row_colors[row] = color

    distractor_pairs = ((3, 5), (2, 5), (3, 6), (2, 6))
    distractors = list(distractor_pairs[(int(seed) // 9) % len(distractor_pairs)])
    rng.shuffle(distractors)
    row_hops = [0, 0, 0]
    row_hops[target_row] = TARGET_HOPS
    for row, hops in zip((r for r in range(3) if r != target_row), distractors):
        row_hops[row] = hops

    nodes = [{"id": "start", "x": root[0], "y": root[1], "role": "start", "color": "green"}]
    edges: list[list[str]] = []
    for row in range(3):
        hops = row_hops[row]
        end = (terminal_x[row], rows[row])
        prior = "start"
        direction = -1.0 if row == 0 else (1.0 if row == 2 else rng.choice((-1.0, 1.0)))
        wave = rng.uniform(10.0, 22.0)
        for step in range(1, hops + 1):
            t = step / hops
            x = root[0] + (end[0] - root[0]) * t
            # Linear vertical separation makes the three spokes diverge immediately;
            # the sinusoid adds shape without letting early nodes from adjacent
            # branches become visually interchangeable.
            y = root[1] + (end[1] - root[1]) * t
            if step < hops:
                y += direction * wave * math.sin(math.pi * t)
                x += rng.uniform(-3.0, 3.0)
                y += rng.uniform(-4.0, 4.0)
                node_id = f"r{row}_n{step}"
                nodes.append({"id": node_id, "x": round(x, 3), "y": round(y, 3), "role": "junction", "color": "neutral"})
            else:
                node_id = f"terminal_{row_colors[row]}"
                nodes.append({"id": node_id, "x": round(end[0], 3), "y": round(end[1], 3), "role": "terminal", "color": row_colors[row]})
            edges.append([prior, node_id])
            prior = node_id

    return {
        "width": width,
        "height": height,
        "nodes": nodes,
        "edges": edges,
        "palette": copy.deepcopy(PALETTES[(int(seed) // 3) % len(PALETTES)]),
        "stroke_width": 5,
        "node_radius": 13,
    }


def _distances(scene: dict) -> dict[str, int]:
    nodes = {node["id"]: node for node in scene["nodes"]}
    adjacency = {node_id: [] for node_id in nodes}
    for left, right in scene["edges"]:
        adjacency[left].append(right)
        adjacency[right].append(left)
    distances = {"start": 0}
    queue = deque(["start"])
    while queue:
        here = queue.popleft()
        for nxt in adjacency[here]:
            if nxt not in distances:
                distances[nxt] = distances[here] + 1
                queue.append(nxt)
    return {
        node["color"]: distances[node["id"]]
        for node in nodes.values()
        if node["role"] == "terminal" and node["id"] in distances
    }


def analytic_gold(scene: dict) -> str:
    """Return the uniquely rendered terminal color at four visible links."""
    nodes = scene["nodes"]
    edges = scene["edges"]
    local = {"nodes": nodes, "edges": edges}
    matches = [color for color, distance in _distances(local).items() if distance == TARGET_HOPS]
    return matches[0] if len(matches) == 1 else "ambiguous"


def margin(scene: dict) -> float:
    """Runner-up score gap in whole hops for the exact-four selection."""
    distances = _distances(scene)
    errors = sorted(abs(distance - TARGET_HOPS) for distance in distances.values())
    if len(errors) < 2 or errors[0] != 0:
        return 0.0
    return float(errors[1] - errors[0])


def is_quarantined(scene: dict) -> bool:
    return analytic_gold(scene) == "ambiguous" or margin(scene) < 1.0


def render(scene: dict) -> Image.Image:
    """Render only the declared scene fields, at 3x supersampling."""
    width = int(scene["width"])
    height = int(scene["height"])
    nodes = scene["nodes"]
    edges = scene["edges"]
    palette = scene["palette"]
    stroke_width = int(scene["stroke_width"])
    node_radius = int(scene["node_radius"])
    ss = 3
    image = Image.new("RGB", (width * ss, height * ss), palette["background"])
    draw = ImageDraw.Draw(image)
    by_id = {node["id"]: node for node in nodes}
    for left, right in edges:
        a, b = by_id[left], by_id[right]
        draw.line(
            [(a["x"] * ss, a["y"] * ss), (b["x"] * ss, b["y"] * ss)],
            fill=palette["ink"], width=stroke_width * ss,
        )
    outline = 3 * ss
    for node in nodes:
        x, y = node["x"] * ss, node["y"] * ss
        radius = node_radius * ss
        draw.ellipse(
            (x - radius, y - radius, x + radius, y + radius),
            fill=palette[node["color"]], outline=palette["ink"], width=outline,
        )
    start = next(node for node in nodes if node["role"] == "start")
    draw.text(
        ((start["x"] - 31) * ss, (start["y"] + 20) * ss),
        "START", fill=palette["ink"], font=_font(13 * ss),
    )
    draw.text(
        (18 * ss, 16 * ss), "COUNT NODE-TO-NODE LINKS",
        fill=palette["ink"], font=_font(17 * ss),
    )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["nodes"] = list(reversed(twin["nodes"]))
    twin["edges"] = [[right, left] for left, right in reversed(twin["edges"])]
    return [("reverse_node_and_undirected_edge_storage", twin)]
