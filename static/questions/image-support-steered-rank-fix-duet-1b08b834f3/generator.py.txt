"""Rank-fixed-point duet: two far-apart cords, one shared colour triple.

Latent scene z = {canvas, cord_a, cord_b}.

Each cord is an x-monotone (optionally transposed) zigzag polyline of constant
stroke width, confined to its own widely separated box. One free end of every
cord carries a black start square; the other end is bare. *Both* cords carry
exactly the same three colours - one red, one green and one blue disk sitting
on the stroke - but generally in different along-cord orders.

Counting from its own black start square, every disk has a position number
1, 2 or 3 on its own cord. Exactly one colour receives the same position
number on both cords; the other two colours have their two position numbers
exchanged. The decision is that fixed-point colour.

The two cords play completely symmetric roles: neither one is a reference and
neither one alone constrains the answer, because any single cord admits all
three colours as the fixed point.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "rank_fix_duet-0.1.0"

CANVAS = (900, 620)
STROKE = 7
DISK_R = 11
SQUARE_HALF = 8

BACKGROUND = (255, 255, 255)
CORD_RGB = (34, 46, 74)
START_RGB = (0, 0, 0)
DISK_RGB = {
    "red": (214, 40, 40),
    "green": (26, 148, 62),
    "blue": (60, 110, 226),
}
COLORS = ("red", "green", "blue")

# Two widely separated boxes, used as a diagonal pair.
BOX_TL = (20, 20, 440, 280)
BOX_BR = (460, 340, 880, 600)
BOX_TR = (460, 20, 880, 280)
BOX_BL = (20, 340, 440, 600)

MARGIN_QUARANTINE = 8.0  # percent of the cord's own length

# The three permutations of {0,1,2} with exactly one fixed point, keyed by the
# rank that is held fixed. Each maps cord-a rank -> cord-b rank.
_ONE_FIXED = {
    0: {0: 0, 1: 2, 2: 1},
    1: {0: 2, 1: 1, 2: 0},
    2: {0: 1, 1: 0, 2: 2},
}


def _polyline(rng: random.Random, box: tuple[int, int, int, int]) -> list[list[float]]:
    """x-monotone zigzag inside ``box``; optionally transposed to run vertically.

    x-monotonicity makes self-intersection impossible and forces non-adjacent
    segments at least one x-step apart, so the stroke never touches itself.
    """
    x0, y0, x1, y1 = box
    transpose = rng.random() < 0.5
    if transpose:
        # Build in a transposed frame, then map back into the real box.
        w = y1 - y0
        h = x1 - x0
        ax0, ay0 = 0.0, 0.0
        ax1, ay1 = float(w), float(h)
    else:
        ax0, ay0, ax1, ay1 = float(x0), float(y0), float(x1), float(y1)

    pad = SQUARE_HALF + 4
    n = rng.randint(4, 6)
    left = ax0 + pad + rng.uniform(0.0, 22.0)
    right = ax1 - pad - rng.uniform(0.0, 22.0)
    weights = [rng.uniform(0.82, 1.18) for _ in range(n)]
    total = sum(weights)
    xs = [left]
    for w_i in weights:
        xs.append(xs[-1] + (right - left) * w_i / total)

    lo_band = (ay0 + pad, ay0 + pad + 0.22 * (ay1 - ay0))
    hi_band = (ay1 - pad - 0.22 * (ay1 - ay0), ay1 - pad)
    start_low = rng.random() < 0.5
    pts = []
    for i, x in enumerate(xs):
        band = lo_band if (i % 2 == 0) == start_low else hi_band
        pts.append([x, rng.uniform(*band)])

    if transpose:
        # (a, b) in the transposed frame -> (x0 + b, y0 + a) in the real box.
        pts = [[x0 + b, y0 + a] for a, b in pts]
    if rng.random() < 0.5:
        pts.reverse()
    return [[round(px, 3), round(py, 3)] for px, py in pts]


def _seglens(pts: list[list[float]]) -> list[float]:
    return [math.dist(pts[i], pts[i + 1]) for i in range(len(pts) - 1)]


def _total_len(pts: list[list[float]]) -> float:
    return sum(_seglens(pts))


def _point_at(pts: list[list[float]], frac: float) -> tuple[float, float]:
    lens = _seglens(pts)
    target = frac * sum(lens)
    acc = 0.0
    for i, seg in enumerate(lens):
        if acc + seg >= target or i == len(lens) - 1:
            t = 0.0 if seg <= 0 else (target - acc) / seg
            t = min(max(t, 0.0), 1.0)
            ax, ay = pts[i]
            bx, by = pts[i + 1]
            return (ax + t * (bx - ax), ay + t * (by - ay))
        acc += seg
    raise AssertionError("unreachable")


def _three_depths(rng: random.Random, gap_lo: float) -> list[float]:
    """Three increasing depths with all adjacent gaps >= ``gap_lo``.

    The triple is drawn inside a randomly placed sub-window of [0.08, 0.92]
    rather than the whole cord, so the two cords can carry their disks in very
    different absolute stretches of their own lengths. That decorrelates
    same-position depth from same-position rank, which is what makes the
    metric depth shortcut unreliable.
    """
    lo, hi = 0.08, 0.92
    width = min(2.0 * gap_lo + rng.uniform(0.02, 0.34), hi - lo)
    base = rng.uniform(lo, hi - width)
    for _ in range(6000):
        vals = sorted(rng.uniform(base, base + width) for _ in range(3))
        if vals[1] - vals[0] < gap_lo or vals[2] - vals[1] < gap_lo:
            continue
        return [round(v, 5) for v in vals]
    return None


def sample_scene(seed: int) -> dict:
    rng = random.Random(f"rank_fix_duet::{int(seed)}")
    boxes = (BOX_TL, BOX_BR) if rng.random() < 0.5 else (BOX_TR, BOX_BL)
    if rng.random() < 0.5:
        box_a, box_b = boxes
    else:
        box_b, box_a = boxes

    fixed_rank = rng.randrange(3)
    perm = _ONE_FIXED[fixed_rank]
    colors = list(COLORS)
    rng.shuffle(colors)
    # colors[i] sits at rank i on cord a, and at rank perm[i] on cord b.
    gold = colors[fixed_rank]

    near_boundary = rng.random() < 0.25
    gap_lo = rng.uniform(0.085, 0.125) if near_boundary else rng.uniform(0.13, 0.22)
    # The "colour whose two normalized depths are most alike" shortcut is
    # explicitly counterbalanced: it is forced to fail on a majority of scenes
    # and forced to succeed on a substantial minority, so neither the cue nor
    # its negation is usable and ordinal rank binding is the only route.
    roll = rng.random()
    want_cue = None
    if roll < 0.55:
        want_cue = False
    elif roll < 0.9:
        want_cue = True

    for _attempt in range(1200):
        depths_a = _three_depths(rng, gap_lo)
        depths_b = _three_depths(rng, gap_lo)
        if depths_a is None or depths_b is None:
            continue
        start_a = bool(rng.random() < 0.5)
        start_b = bool(rng.random() < 0.5)
        poly_a = _polyline(rng, box_a)
        poly_b = _polyline(rng, box_b)

        def stored(depth: float, start_at_first: bool) -> float:
            # ``depth`` is measured from the black start square; convert it to a
            # fraction of the stored vertex order.
            return round(depth if start_at_first else 1.0 - depth, 5)

        disks_a = [
            {"color": colors[i], "frac": stored(depths_a[i], start_a)}
            for i in range(3)
        ]
        disks_b = [
            {"color": colors[i], "frac": stored(depths_b[perm[i]], start_b)}
            for i in range(3)
        ]
        rng.shuffle(disks_a)
        rng.shuffle(disks_b)
        scene = {
            "canvas": list(CANVAS),
            "cord_a": {
                "poly": poly_a,
                "start_at_first": start_a,
                "disks": disks_a,
            },
            "cord_b": {
                "poly": poly_b,
                "start_at_first": start_b,
                "disks": disks_b,
            },
        }
        if is_quarantined(scene):
            continue
        if want_cue is not None and (depth_cue_answer(scene) == gold) != want_cue:
            continue
        # Sampling screen: the public quarantine rule below still defines the
        # flag, but quarantined scenes are never emitted, so every published
        # scene carries a rank ordering the pixel arm can certify.
        return scene
    raise RuntimeError(f"could not sample scene for seed {seed}")


# --- analytic gold -----------------------------------------------------------


def _ranks(scene: dict):
    """Per-cord depth-from-start and 0-based rank for each colour.

    Reads only the two declared label-input fields, ``cord_a`` and ``cord_b``.
    """
    labelled = {"cord_a": scene["cord_a"], "cord_b": scene["cord_b"]}
    out = {}
    for key, part in labelled.items():
        start_at_first = bool(part["start_at_first"])
        depths = {}
        for disk in part["disks"]:
            f = float(disk["frac"])
            depths[disk["color"]] = f if start_at_first else 1.0 - f
        order = sorted(depths.items(), key=lambda kv: (kv[1], kv[0]))
        ranks = {color: i for i, (color, _d) in enumerate(order)}
        out[key] = (depths, ranks)
    return out


def analytic_gold(scene: dict) -> str:
    table = _ranks(scene)
    ranks_a = table["cord_a"][1]
    ranks_b = table["cord_b"][1]
    fixed = sorted(c for c in ranks_a if ranks_a[c] == ranks_b.get(c))
    if len(fixed) != 1:
        raise ValueError("scene does not have exactly one rank fixed point")
    return fixed[0]


def margin(scene: dict) -> float:
    """Distance to the nearest rank swap, in percent of a cord's own length.

    Exchanging any two along-cord-adjacent disks on either cord changes the
    permutation and therefore the fixed point, so the smallest adjacent
    along-cord gap on either cord is exactly the distance to a decision flip.
    """
    table = _ranks(scene)
    gaps = []
    for key in ("cord_a", "cord_b"):
        depths = sorted(table[key][0].values())
        gaps.extend(depths[i + 1] - depths[i] for i in range(len(depths) - 1))
    return round(100.0 * min(gaps), 4)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < MARGIN_QUARANTINE


def latent_symmetries(scene: dict):
    """Nontrivial pixel-identical latent aliases.

    A cord's disk list is an unordered set of pairwise disjoint disks, so
    rotating the list only changes draw order between non-overlapping shapes and
    the raster is byte-identical. Reversing a cord's vertex list while flipping
    its start flag and complementing every stored fraction re-describes the same
    drawn cord, the same start square and the same disk positions, so it is also
    byte-identical; gold depends only on the rendered geometry, never on the
    stored vertex or list ordering.
    """
    out = []
    rotated = dict(scene)
    for key in ("cord_a", "cord_b"):
        part = dict(scene[key])
        disks = list(part["disks"])
        part["disks"] = disks[1:] + disks[:1]
        rotated[key] = part
    out.append(("rotate_disk_lists", rotated))

    reparam = dict(scene)
    for key in ("cord_a", "cord_b"):
        part = dict(scene[key])
        part["poly"] = [list(p) for p in reversed(part["poly"])]
        part["start_at_first"] = not part["start_at_first"]
        part["disks"] = [
            {"color": d["color"], "frac": round(1.0 - float(d["frac"]), 5)}
            for d in part["disks"]
        ]
        reparam[key] = part
    out.append(("reverse_cord_parametrization", reparam))
    return out


# --- rendering ---------------------------------------------------------------


def render(scene: dict) -> Image.Image:
    w, h = scene["canvas"]
    img = Image.new("RGB", (int(w), int(h)), BACKGROUND)
    draw = ImageDraw.Draw(img)

    parts = [scene["cord_a"], scene["cord_b"]]

    for part in parts:
        flat = [(float(x), float(y)) for x, y in part["poly"]]
        draw.line(flat, fill=CORD_RGB, width=STROKE, joint="curve")

    # Markers on top of the stroke; the oracle unions them back into the cord.
    for part in parts:
        poly = part["poly"]
        sx, sy = poly[0] if part["start_at_first"] else poly[-1]
        draw.rectangle(
            [sx - SQUARE_HALF, sy - SQUARE_HALF, sx + SQUARE_HALF, sy + SQUARE_HALF],
            fill=START_RGB,
        )

    for part in parts:
        pts = [list(p) for p in part["poly"]]
        for disk in part["disks"]:
            dx, dy = _point_at(pts, float(disk["frac"]))
            draw.ellipse(
                [dx - DISK_R, dy - DISK_R, dx + DISK_R, dy + DISK_R],
                fill=DISK_RGB[disk["color"]],
            )
    return img


# --- diagnostics -------------------------------------------------------------


def depth_cue_answer(scene: dict) -> str:
    """Answer produced by the metric shortcut, not by rank binding.

    A solver that skips the ordinal step and instead names the colour whose two
    normalized along-cord depths are most alike gets this. Reported per scene so
    its disagreement rate with gold is auditable.
    """
    table = _ranks(scene)
    depths_a = table["cord_a"][0]
    depths_b = table["cord_b"][0]
    ranked = sorted(
        depths_a.items(), key=lambda kv: (abs(kv[1] - depths_b[kv[0]]), kv[0])
    )
    return ranked[0][0]


def euclidean_cue_answer(scene: dict) -> str:
    """Answer from the cheapest layout shortcut: smallest straight-line distance
    between a colour's two disks, ignoring both cords entirely."""
    best = None
    best_d = math.inf
    for key_pair in [("cord_a", "cord_b")]:
        pa = scene[key_pair[0]]
        pb = scene[key_pair[1]]
        pts_a = [list(p) for p in pa["poly"]]
        pts_b = [list(p) for p in pb["poly"]]
        for disk in pa["disks"]:
            qa = _point_at(pts_a, float(disk["frac"]))
            other = next(d for d in pb["disks"] if d["color"] == disk["color"])
            qb = _point_at(pts_b, float(other["frac"]))
            d = math.dist(qa, qb)
            if d < best_d:
                best_d = d
                best = disk["color"]
    return best


def fixed_rank_of_answer(scene: dict) -> int:
    """0-based position number that the gold colour holds on both cords."""
    table = _ranks(scene)
    return table["cord_a"][1][analytic_gold(scene)]


def cord_length_ratio(scene: dict) -> float:
    la = _total_len([list(p) for p in scene["cord_a"]["poly"]])
    lb = _total_len([list(p) for p in scene["cord_b"]["poly"]])
    return round(max(la, lb) / max(min(la, lb), 1e-6), 4)
