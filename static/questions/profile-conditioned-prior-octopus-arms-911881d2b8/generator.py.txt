"""Prior-conflict octopus world: count visible arms.

The latent scene holds a stylized octopus with ``n`` tentacles dangling from a
dome-shaped head. A familiar canonical prior asserts that an octopus has EIGHT
arms; the counterfactual renders a non-canonical count (6 or 10) and the
decision must be read from the visible arms alone, never from the prior. The
pixel oracle segments the arm colour and counts the connected arm components.

Latent alias mode is ``not_applicable``: every latent field is directly
reflected in the visible geometry (head position, arm spacing, curvature,
extent, count), so the rendering is injective in the latent scene and there is
no render-invariant alias. The canonical-prior name "octopus" is a
matched-control/counterfactual manipulation of the *question*, not of any
latent field.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = 512
CX = 256.0
HEAD_CY = 150.0
HEAD_RX = 125.0
HEAD_RY = 95.0
ARM_START_Y = 236.0
MID_Y = 345.0
ARM_RADIUS = 9.0
# Steps are chosen large enough that neighbouring tentacles keep a positive
# pixel clearance for every supported count.
STEP_MIN = 46.0
STEP_MAX = 52.0
BEND_MAX = 4.0

# Supported off-quarantine counts. The canonical prior is 8; 6 and 10 are the
# counterfactuals that contradict it. 6/8/10 all keep clearances above the
# quarantine threshold at canvas width ~512.
SUPPORTED_N = (8, 6, 10)

# A count is quarantined when the tentacles are so crowded that adjacent arms
# can touch and be miscounted at pixel scale.
QUAR_MIN_CLEARANCE = 16.0

BACKGROUND = (255, 255, 255)
HEAD_COLOR = (204, 60, 60)   # red dome head
ARM_COLOR = (24, 28, 66)     # dark navy tentacles


def sample_scene(seed: int) -> dict:
    """Deterministic scene dict from an integer seed."""
    rng = random.Random(seed)
    n = SUPPORTED_N[rng.randrange(len(SUPPORTED_N))]
    # step varies independently of n within a range that keeps every supported
    # count well separated; this decouples spacing from the count (cheap-cue
    # counterbalance against "more arms = tighter spacing").
    step = rng.uniform(STEP_MIN, STEP_MAX)
    x0 = CX - (n - 1) * step / 2.0 + rng.uniform(-10.0, 10.0)
    arm_x = [round(x0 + k * step, 3) for k in range(n)]
    bend = rng.uniform(-BEND_MAX, BEND_MAX)
    arm_len = (CANVAS - ARM_START_Y) - 8.0 - rng.uniform(0.0, 30.0)
    mid_y = ARM_START_Y + arm_len * 0.45
    head_cx = CX + rng.uniform(-35.0, 35.0)
    return {
        "n": n,
        "arm_x": arm_x,
        "step": round(step, 6),
        "bend": round(bend, 6),
        "mid_y": round(mid_y, 3),
        "arm_len": round(arm_len, 3),
        "head_cx": round(head_cx, 3),
        "head_cy": HEAD_CY,
        "canvas": CANVAS,
    }


def _bezier_points(p0, pc, p1, steps: int = 28):
    """Sample a quadratic bezier so a thick curve is drawn as a polyline."""
    pts = []
    for i in range(steps + 1):
        t = i / steps
        mt = 1.0 - t
        x = mt * mt * p0[0] + 2 * mt * t * pc[0] + t * t * p1[0]
        y = mt * mt * p0[1] + 2 * mt * t * pc[1] + t * t * p1[1]
        pts.append((x, y))
    return pts


def render(scene: dict) -> Image.Image:
    """Render the stylized octopus to an RGB image."""
    canvas = int(scene["canvas"])
    img = Image.new("RGB", (canvas, canvas), BACKGROUND)
    draw = ImageDraw.Draw(img)
    head_cx = float(scene["head_cx"])
    head_cy = float(scene["head_cy"])
    draw.ellipse(
        [head_cx - HEAD_RX, head_cy - HEAD_RY, head_cx + HEAD_RX, head_cy + HEAD_RY],
        fill=HEAD_COLOR,
    )
    arm_x = scene["arm_x"]
    n = int(scene["n"])
    assert len(arm_x) == n
    start = ARM_START_Y
    end = start + float(scene["arm_len"])
    mid = float(scene["mid_y"])
    bend = float(scene["bend"])
    for k, x in enumerate(arm_x):
        s = bend if k % 2 == 0 else -bend
        pts = _bezier_points((x, start), (x + s, mid), (x, end))
        draw.line(pts, fill=ARM_COLOR, width=int(2 * ARM_RADIUS), joint="curve")
    return img


def analytic_gold(scene: dict) -> str:
    """Decision = the visible number of arms, as a string."""
    return str(int(scene["n"]))


def margin(scene: dict) -> float:
    """Verifiability margin = worst-case horizontal clearance between arms."""
    step = float(scene["step"])
    bend = abs(float(scene.get("bend", 0.0)))
    return step - 2 * ARM_RADIUS - 2 * bend


def is_quarantined(scene: dict) -> bool:
    """Quarantine crowded counts that are ambiguous at pixel scale."""
    if int(scene["n"]) not in SUPPORTED_N:
        return True
    return margin(scene) < QUAR_MIN_CLEARANCE


def latent_symmetries(scene: dict):
    """not_applicable: empty, the latent rendering is injective."""
    return []
