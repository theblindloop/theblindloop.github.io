"""Deterministic renderer for a prior-conflicting music-staff line count.

Seeds are arranged in matched triples.  Within a triple all visible styling and
all distractors are identical; only the number of long, bracket-connected staff
rails changes among four, five, and six.  Five is the familiar control.
"""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

CANVAS_SIZE = 512
STAFF_X0 = 29
STAFF_X1 = 483
STAFF_CENTER_Y = 256

BACKGROUNDS = (
    (250, 248, 242),
    (245, 249, 248),
    (249, 246, 250),
)
STAFF_COLORS = (
    (31, 78, 128),
    (126, 47, 72),
    (27, 112, 105),
)
NOTE_GRAYS = (
    (104, 108, 114),
    (114, 105, 111),
    (101, 112, 109),
)


def sample_scene(seed: int) -> dict:
    """Return a deterministic member of a style-matched 4/5/6 triple."""

    seed = int(seed)
    group = seed // 3
    rng = random.Random((group ^ 0x5A17F4FF) * 1_000_003)
    return {
        "background_code": rng.randrange(len(BACKGROUNDS)),
        "canvas_size": CANVAS_SIZE,
        "decor_seed": rng.randrange(1, 2**31),
        "line_count": 4 + seed % 3,
        "offset_x": rng.randrange(-4, 5),
        "offset_y": rng.randrange(-2, 3),
        "spacing": rng.choice((17, 18)),
        "staff_color_code": rng.randrange(len(STAFF_COLORS)),
        "stroke_width": rng.choice((4, 5)),
    }


def _validate(scene: dict) -> None:
    if int(scene["canvas_size"]) != CANVAS_SIZE:
        raise ValueError("the public world uses a fixed 512px canvas")
    if int(scene["line_count"]) not in (4, 5, 6):
        raise ValueError("line_count must be 4, 5, or 6")
    if int(scene["spacing"]) not in (17, 18):
        raise ValueError("spacing must be 17 or 18 pixels")
    if int(scene["stroke_width"]) not in (4, 5):
        raise ValueError("stroke_width must be 4 or 5 pixels")
    if not -8 <= int(scene["offset_x"]) <= 8 or not -5 <= int(scene["offset_y"]) <= 5:
        raise ValueError("translation is outside the supported visible range")


def _rail_centers(scene: dict) -> list[int]:
    count = int(scene["line_count"])
    spacing = int(scene["spacing"])
    center = STAFF_CENTER_Y + int(scene["offset_y"])
    start = center - ((count - 1) * spacing) // 2
    return [start + index * spacing for index in range(count)]


def _staff_bounds(scene: dict) -> tuple[int, int, int, int]:
    centers = _rail_centers(scene)
    half = int(scene["stroke_width"]) // 2
    return (
        STAFF_X0 + int(scene["offset_x"]),
        centers[0] - half,
        STAFF_X1 + int(scene["offset_x"]),
        centers[-1] + half,
    )


def _draw_note(draw: ImageDraw.ImageDraw, x: int, y: int, scale: int, gray: tuple[int, int, int]) -> None:
    """Draw an answer-neutral gray note glyph with an optional short ledger."""

    head_w = 7 + scale
    head_h = 5 + scale
    draw.ellipse((x - head_w, y - head_h, x + head_w, y + head_h), fill=gray)
    direction = -1 if (x + y) % 2 == 0 else 1
    stem_x = x + direction * head_w
    draw.rectangle((stem_x - 1, y - 29, stem_x + 1, y + 1), fill=gray)
    if y < 216 or y > 299:
        draw.rectangle((x - 21, y - 1, x + 21, y + 1), fill=gray)


def render(scene: dict) -> Image.Image:
    _validate(scene)
    background = BACKGROUNDS[int(scene["background_code"]) % len(BACKGROUNDS)]
    staff = STAFF_COLORS[int(scene["staff_color_code"]) % len(STAFF_COLORS)]
    gray = NOTE_GRAYS[int(scene["background_code"]) % len(NOTE_GRAYS)]
    image = Image.new("RGB", (CANVAS_SIZE, CANVAS_SIZE), background)
    draw = ImageDraw.Draw(image)

    # Musical context is answer-independent within every matched triple.  Gray
    # note glyphs and short ledgers are deliberately outside the staff-color
    # mask used by the independent inverse arm.
    decor = random.Random(int(scene["decor_seed"]))
    note_count = decor.randrange(7, 15)
    columns = list(range(78, 454, 28))
    decor.shuffle(columns)
    for index, x in enumerate(columns[:note_count]):
        band = decor.randrange(7)
        y = 205 + band * 17 + decor.randrange(-3, 4)
        _draw_note(draw, x, y, index % 3, gray)

    # Small remote gray proof marks broaden total-ink variation without carrying
    # the answer and make crop-based global-ink proxies unreliable.
    for _ in range(decor.randrange(2, 8)):
        x = decor.randrange(58, 455)
        y = decor.choice((decor.randrange(45, 125), decor.randrange(388, 468)))
        radius = decor.randrange(3, 9)
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=gray)

    x0, _, x1, _ = _staff_bounds(scene)
    centers = _rail_centers(scene)
    width = int(scene["stroke_width"])
    low = (width - 1) // 2
    high = width // 2
    for y in centers:
        draw.rectangle((x0, y - low, x1, y + high), fill=staff)

    # One left barline joins every rail into a single connected thin component.
    draw.rectangle((x0, centers[0] - low, x0 + width - 1, centers[-1] + high), fill=staff)
    return image


def analytic_gold(scene: dict) -> str:
    count = int(scene["line_count"])
    if count not in (4, 5, 6):
        raise ValueError("line_count must be 4, 5, or 6")
    return str(count)


def margin(scene: dict) -> float:
    """Visible whitespace separating adjacent full-width rails, in pixels."""

    return float(int(scene["spacing"]) - int(scene["stroke_width"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["staff_color_code"] = int(twin["staff_color_code"]) + len(STAFF_COLORS)
    return [("one_full_staff_palette_cycle", twin)]
