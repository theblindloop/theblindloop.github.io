"""Independent pixel-only inverse for ordered lobe-side alternation."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_MIN_CHROMA = 58
_MIN_FRAGMENT_AREA = 16
_MIN_TOTAL_AREA = 650
_MAX_TOTAL_AREA = 1800
_MIN_TRACE_WIDTH = 47
_MAX_TRACE_WIDTH = 64
_MIN_TRACE_HEIGHT = 76
_MAX_TRACE_HEIGHT = 102
_SIDE_THRESHOLD = 8.0
_MIN_RUN_ROWS = 7
_MIN_PEAK_OFFSET = 19.0


class OracleAbstention(ValueError):
    """The raster does not contain one structurally complete supported trace."""


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dy or dx):
                        continue
                    ny, nx = cy + dy, cx + dx
                    if (
                        0 <= ny < height
                        and 0 <= nx < width
                        and mask[ny, nx]
                        and not seen[ny, nx]
                    ):
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        output.append(np.asarray(points, dtype=np.int32))
    return output


def _trace_mask(array: np.ndarray) -> np.ndarray:
    high = array.max(axis=2).astype(np.int16)
    low = array.min(axis=2).astype(np.int16)
    return ((high - low) >= _MIN_CHROMA) & (high >= 105)


def _side_runs(mask: np.ndarray, axis_x: float, ymin: int, ymax: int) -> list[tuple[int, float, int]]:
    active: list[tuple[int, int, float]] = []
    for y in range(ymin, ymax + 1):
        xs = np.flatnonzero(mask[y])
        if not len(xs):
            continue
        offset = float(np.median(xs)) - axis_x
        if abs(offset) >= _SIDE_THRESHOLD:
            active.append((y, -1 if offset < 0 else 1, abs(offset)))

    groups: list[list[tuple[int, int, float]]] = []
    for item in active:
        if not groups or item[0] - groups[-1][-1][0] > 1 or item[1] != groups[-1][-1][1]:
            groups.append([item])
        else:
            groups[-1].append(item)

    runs: list[tuple[int, float, int]] = []
    for group in groups:
        if len(group) < _MIN_RUN_ROWS:
            continue
        peak = max(value[2] for value in group)
        if peak < _MIN_PEAK_OFFSET:
            continue
        runs.append((group[0][1], peak, len(group)))
    return runs


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 300 or array.shape[1] < 400:
        raise OracleAbstention("raster is too small")
    raw_mask = _trace_mask(array)
    pieces = []
    speckle = 0
    for points in _components(raw_mask):
        if len(points) < _MIN_FRAGMENT_AREA:
            speckle += len(points)
        else:
            pieces.append(points)
    if speckle > 30:
        raise OracleAbstention(f"too much chromatic speckle ({speckle} pixels)")
    if len(pieces) != 1:
        raise OracleAbstention(f"expected one connected colored trace, found {len(pieces)}")

    points = pieces[0]
    area = len(points)
    ys, xs = points[:, 0], points[:, 1]
    width = int(xs.max() - xs.min() + 1)
    height = int(ys.max() - ys.min() + 1)
    if not _MIN_TOTAL_AREA <= area <= _MAX_TOTAL_AREA:
        raise OracleAbstention(f"unsupported trace area {area}")
    if not _MIN_TRACE_WIDTH <= width <= _MAX_TRACE_WIDTH:
        raise OracleAbstention(f"unsupported trace width {width}")
    if not _MIN_TRACE_HEIGHT <= height <= _MAX_TRACE_HEIGHT or height / width < 1.2:
        raise OracleAbstention(f"trace is not a short vertical route ({width}x{height})")

    # Equal-amplitude, two-left/two-right construction makes the midpoint of the
    # visible extrema the staff axis. This is recovered from pixels, not scene state.
    axis_x = 0.5 * (float(xs.min()) + float(xs.max()))
    runs = _side_runs(raw_mask, axis_x, int(ys.min()), int(ys.max()))
    if len(runs) != 4:
        raise OracleAbstention(f"expected four recoverable side arcs, found {len(runs)}")
    signs = [item[0] for item in runs]
    return "yes" if all(a != b for a, b in zip(signs, signs[1:])) else "no"
