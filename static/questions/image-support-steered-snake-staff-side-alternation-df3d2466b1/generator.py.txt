"""Deterministic renderer for loop-side alternation on a one-snake staff."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

CANVAS_WIDTH = 960
CANVAS_HEIGHT = 480
CELL_WIDTH = 120
CELL_HEIGHT = 60
SUPERSAMPLE = 4

TRACE_PALETTE = (
    (28, 105, 219),
    (11, 151, 118),
    (190, 62, 172),
    (224, 82, 35),
)
BACKGROUND_TONES = ((248, 247, 243), (244, 247, 249), (249, 245, 248))
STAFF_TONES = ((82, 86, 89), (91, 86, 94), (77, 88, 91))


def sample_scene(seed: int) -> dict:
    """Return a matched alternating/nonalternating pair on adjacent seeds."""

    seed = int(seed)
    pair = seed // 2
    rng = random.Random(pair ^ 0x51A7F04D)
    first = -1 if rng.randrange(2) == 0 else 1
    if seed % 2 == 0:
        sides = [first, -first, first, -first]
    else:
        # Same two-left/two-right balance and same four congruent path lobes.
        sides = [first, -first, -first, first]
    return {
        "loop_sides": sides,
        "canvas_width": CANVAS_WIDTH,
        "canvas_height": CANVAS_HEIGHT,
        "center_col": rng.randrange(1, 7),
        "center_row": rng.randrange(2, 7),
        "jitter_x": rng.randrange(-4, 5),
        "loop_height": 16 + rng.randrange(3),
        "amplitude": 24 + rng.randrange(4),
        "stroke_width": 5 + rng.randrange(2),
        "trace_color_code": pair % len(TRACE_PALETTE),
        "staff_tone": rng.randrange(len(STAFF_TONES)),
        "staff_width": 5 + rng.randrange(2),
        "head_radius": 6 + rng.randrange(2),
        "decor_variant": rng.randrange(32),
    }


def _scaled_points(points: list[tuple[float, float]], scale: int) -> list[tuple[int, int]]:
    return [(round(x * scale), round(y * scale)) for x, y in points]


def _box(values: tuple[float, float, float, float], scale: int) -> tuple[int, int, int, int]:
    return tuple(round(value * scale) for value in values)


def _trace_points(scene: dict, cx: float, center_y: float) -> tuple[list[tuple[float, float]], float, float]:
    height = float(scene["loop_height"])
    amplitude = float(scene["amplitude"])
    sides = [int(value) for value in scene["loop_sides"]]
    y0 = center_y - 2.0 * height
    points: list[tuple[float, float]] = []
    samples_per_lobe = 28
    for lobe, side in enumerate(sides):
        for step in range(samples_per_lobe + 1):
            if lobe and step == 0:
                continue
            u = step / samples_per_lobe
            x = cx + side * amplitude * math.sin(math.pi * u)
            y = y0 + (lobe + u) * height
            points.append((x, y))
    return points, y0, y0 + len(sides) * height


def render(scene: dict) -> Image.Image:
    width = int(scene["canvas_width"])
    height = int(scene["canvas_height"])
    cx = int(scene["center_col"]) * CELL_WIDTH + CELL_WIDTH / 2 + int(scene["jitter_x"])
    center_y = int(scene["center_row"]) * CELL_HEIGHT
    color = TRACE_PALETTE[int(scene["trace_color_code"]) % len(TRACE_PALETTE)]
    tone_code = int(scene["staff_tone"]) % len(STAFF_TONES)
    background = BACKGROUND_TONES[tone_code]
    staff = STAFF_TONES[tone_code]
    pale = tuple(min(255, value + 139) for value in staff)
    remote = tuple(max(0, value - 8) for value in pale)
    trace_width = int(scene["stroke_width"])
    staff_width = int(scene["staff_width"])
    head_radius = int(scene["head_radius"])
    decor = int(scene["decor_variant"])

    points, y0, y4 = _trace_points(scene, cx, center_y)
    ss = SUPERSAMPLE
    image = Image.new("RGB", (width * ss, height * ss), background)
    draw = ImageDraw.Draw(image)

    # Label-independent, achromatic proof marks occupy remote cells.
    for index in range(4):
        x = 34 + ((decor * 61 + index * 149) % (width - 68))
        y = 31 if (decor + index) % 2 == 0 else height - 32
        draw.line(
            _scaled_points([(x - 8, y), (x + 8, y)], ss),
            fill=remote,
            width=2 * ss,
        )

    # A restrained medical-emblem badge supplies the familiar prior but carries no label.
    radius = 58
    draw.ellipse(
        _box((cx - radius, center_y - radius, cx + radius, center_y + radius), ss),
        fill=pale,
        outline=remote,
        width=2 * ss,
    )
    draw.line(
        _scaled_points([(cx, y0 - 18), (cx, y4 + 18)], ss),
        fill=staff,
        width=staff_width * ss,
    )
    draw.ellipse(
        _box((cx - 7, y0 - 23, cx + 7, y0 - 9), ss),
        fill=staff,
    )
    draw.ellipse(
        _box((cx - 9, y4 + 11, cx + 9, y4 + 17), ss),
        fill=staff,
    )

    # The single saturated path is drawn last so all four axis crossings remain visible.
    scaled = _scaled_points(points, ss)
    draw.line(scaled, fill=color, width=trace_width * ss, joint="curve")
    cap_radius = trace_width * ss / 2
    for x, y in (scaled[0], scaled[-1]):
        draw.ellipse(
            (round(x - cap_radius), round(y - cap_radius), round(x + cap_radius), round(y + cap_radius)),
            fill=color,
        )
    # Centered head and tail establish top-to-bottom reading without encoding a side.
    draw.ellipse(
        _box((cx - head_radius, y0 - 2 * head_radius, cx + head_radius, y0 + 1), ss),
        fill=color,
    )
    draw.line(
        _scaled_points([(cx, y4), (cx, y4 + 9)], ss),
        fill=color,
        width=trace_width * ss,
    )

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    sides = [int(value) for value in scene["loop_sides"]]
    return "yes" if len(sides) == 4 and all(a != b for a, b in zip(sides, sides[1:])) else "no"


def margin(scene: dict) -> float:
    return float(scene["amplitude"]) - 0.5 * (
        float(scene["stroke_width"]) + float(scene["staff_width"])
    )


def is_quarantined(scene: dict) -> bool:
    sides = [int(value) for value in scene["loop_sides"]]
    return (
        len(sides) != 4
        or any(value not in (-1, 1) for value in sides)
        or sides.count(-1) != 2
        or sides.count(1) != 2
        or int(scene["loop_height"]) < 14
        or margin(scene) <= 12.0
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["trace_color_code"] = int(twin["trace_color_code"]) + len(TRACE_PALETTE)
    return [("one_full_trace_palette_cycle", twin)]
