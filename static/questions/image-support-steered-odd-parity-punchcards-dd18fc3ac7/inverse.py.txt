"""Independent image-only inverse for odd-parity punch cards."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


_TEMPLATE_ROWS = {
    "X": ("1000001", "0100010", "0010100", "0001000", "0010100", "0100010", "1000001"),
    "plus": ("0001000", "0001000", "0001000", "1111111", "0001000", "0001000", "0001000"),
    "T": ("1111111", "0001000", "0001000", "0001000", "0001000", "0001000", "0001000"),
    "L": ("1000000", "1000000", "1000000", "1000000", "1000000", "1000000", "1111111"),
}


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        xs, ys = [], []
        while queue:
            yy, xx = queue.popleft()
            xs.append(xx); ys.append(yy)
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        result.append((min(xs), min(ys), max(xs), max(ys), len(xs)))
    return result


def _templates() -> dict[str, np.ndarray]:
    return {name: np.array([[c == "1" for c in row] for row in rows], dtype=np.uint8) for name, rows in _TEMPLATE_ROWS.items()}


def _read_card(card: np.ndarray) -> np.ndarray:
    # Marker location determines which multiple of 90 degrees restores top-left.
    magenta = (card[:, :, 0] > 170) & (card[:, :, 2] > 130) & (card[:, :, 1] < 90)
    yy, xx = np.nonzero(magenta)
    if len(xx) < 40:
        raise ValueError("missing orientation marker")
    cx, cy = float(xx.mean()), float(yy.mean())
    corner = (cy >= card.shape[0] / 2) * 2 + (cx >= card.shape[1] / 2)
    # observed corners TL, TR, BL, BR correspond to rotations 0,1,3,2.
    rotations = {0: 0, 1: 1, 2: 3, 3: 2}
    upright = np.rot90(card, rotations[int(corner)])
    start = 27
    cell = 18
    bits = np.zeros((7, 7), dtype=np.uint8)
    confidence = []
    for row in range(7):
        for col in range(7):
            cy0 = start + row * cell + cell // 2
            cx0 = start + col * cell + cell // 2
            patch = upright[cy0 - 3:cy0 + 4, cx0 - 3:cx0 + 4].astype(float)
            lum = float(patch.mean())
            bits[row, col] = lum < 120.0
            confidence.append(abs(lum - 120.0))
    if min(confidence) < 35.0:
        raise ValueError("ambiguous cell sample")
    return bits


def decision_with_margin(image: Image.Image) -> tuple[str, float]:
    arr = np.asarray(image.convert("RGB"))
    dark = (arr[:, :, 0] < 90) & (arr[:, :, 1] < 90) & (arr[:, :, 2] < 90)
    boxes = []
    for x0, y0, x1, y1, area in _components(dark):
        w, h = x1 - x0 + 1, y1 - y0 + 1
        if 168 <= w <= 184 and 168 <= h <= 184 and area >= 1800:
            # The detected contour is the 178 px dark border inside a 180 px card.
            boxes.append((x0 - 1, y0 - 1, x1 + 1, y1 + 1))
    if len(boxes) != 4:
        return "abstain", 0.0
    cards = []
    for x0, y0, x1, y1 in sorted(boxes, key=lambda b: (b[1], b[0])):
        crop = arr[y0:y1 + 1, x0:x1 + 1]
        if crop.shape[:2] != (180, 180):
            return "abstain", 0.0
        try:
            cards.append(_read_card(crop))
        except ValueError:
            return "abstain", 0.0
    parity = np.bitwise_xor.reduce(np.asarray(cards), axis=0)
    ranked = sorted((int(np.count_nonzero(parity != target)), name) for name, target in _templates().items())
    gap = float(ranked[1][0] - ranked[0][0])
    if ranked[0][0] != 0 or gap < 6:
        return "abstain", gap
    return ranked[0][1], gap


def decision_from_image(image: Image.Image) -> str:
    return decision_with_margin(image)[0]
