"""Deterministic renderer for four remote odd-parity punch cards."""

from __future__ import annotations

import copy
import random

import numpy as np
from PIL import Image, ImageDraw


TEMPLATES = {
    "X": (
        "1000001", "0100010", "0010100", "0001000", "0010100", "0100010", "1000001"
    ),
    "plus": (
        "0001000", "0001000", "0001000", "1111111", "0001000", "0001000", "0001000"
    ),
    "T": (
        "1111111", "0001000", "0001000", "0001000", "0001000", "0001000", "0001000"
    ),
    "L": (
        "1000000", "1000000", "1000000", "1000000", "1000000", "1000000", "1111111"
    ),
}
SYMBOLS = tuple(TEMPLATES)
PALETTES = (
    ((252, 249, 239), (218, 225, 232)),
    ((244, 251, 246), (225, 216, 231)),
    ((250, 244, 250), (215, 228, 219)),
    ((243, 248, 252), (232, 221, 211)),
)


def _template(name: str) -> np.ndarray:
    return np.array([[ch == "1" for ch in row] for row in TEMPLATES[name]], dtype=np.uint8)


def _decision(cards: list[list[list[int]]]) -> str:
    parity = np.bitwise_xor.reduce(np.asarray(cards, dtype=np.uint8), axis=0)
    distances = sorted((int(np.count_nonzero(parity != _template(name))), name) for name in SYMBOLS)
    return distances[0][1]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    symbol = SYMBOLS[int(seed) % len(SYMBOLS)]
    shares = [np.array([[rng.randrange(2) for _ in range(7)] for _ in range(7)], dtype=np.uint8) for _ in range(3)]
    target = _template(symbol)
    fourth = target ^ shares[0] ^ shares[1] ^ shares[2]
    cards = [share.tolist() for share in (*shares, fourth)]

    # Card identities are shuffled independently of the target. XOR is commutative.
    order = list(range(4))
    rng.shuffle(order)
    cards = [cards[i] for i in order]
    rotations = [rng.randrange(4) for _ in range(4)]
    base_positions = [(20, 20), (312, 20), (20, 312), (312, 312)]
    positions = [(x + rng.randint(-4, 4), y + rng.randint(-4, 4)) for x, y in base_positions]
    rng.shuffle(positions)
    palette = list(PALETTES)
    rng.shuffle(palette)
    return {
        "cards": cards,
        "rotations": rotations,
        "positions": [list(p) for p in positions],
        "palette": [[list(a), list(b)] for a, b in palette],
        "symbol": symbol,
        "cell_size": 18,
        "canvas_size": [512, 512],
    }


def _card_image(bits: list[list[int]], rotation: int, cell_size: int, palette: list[list[int]]) -> Image.Image:
    size = 180
    card = Image.new("RGB", (size, size), tuple(palette[0]))
    draw = ImageDraw.Draw(card)
    draw.rectangle((1, 1, size - 2, size - 2), outline=(52, 55, 60), width=4)
    start = 27
    extent = 7 * cell_size
    draw.rectangle((start - 2, start - 2, start + extent + 1, start + extent + 1), fill=tuple(palette[1]))
    draw.rectangle((start, start, start + extent - 1, start + extent - 1), fill=(249, 249, 246))
    for row in range(7):
        for col in range(7):
            x0 = start + col * cell_size
            y0 = start + row * cell_size
            draw.rectangle((x0, y0, x0 + cell_size - 1, y0 + cell_size - 1), outline=(184, 184, 180), width=1)
            if bits[row][col]:
                draw.rectangle((x0 + 4, y0 + 4, x0 + cell_size - 5, y0 + cell_size - 5), fill=(30, 32, 35))
    # Asymmetric visible orientation fiducial: rotate until this is top-left.
    draw.ellipse((7, 7, 20, 20), fill=(225, 20, 180), outline=(110, 0, 85), width=2)
    return card.rotate(-90 * int(rotation), expand=False)


def render(scene: dict) -> Image.Image:
    cards = scene["cards"]
    rotations = scene["rotations"]
    positions = scene["positions"]
    palette = scene["palette"]
    _symbol = scene["symbol"]
    cell_size = int(scene["cell_size"])
    canvas_size = tuple(scene["canvas_size"])
    image = Image.new("RGB", canvas_size, (238, 236, 229))
    for bits, rotation, position, colors in zip(cards, rotations, positions, palette):
        image.paste(_card_image(bits, rotation, cell_size, colors), tuple(position))
    return image


def analytic_gold(scene: dict) -> str:
    return _decision(scene["cards"])


def margin(scene: dict) -> float:
    parity = np.bitwise_xor.reduce(np.asarray(scene["cards"], dtype=np.uint8), axis=0)
    distances = sorted(int(np.count_nonzero(parity != _template(name))) for name in SYMBOLS)
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # Reordering the parallel card records changes latent indices but not pixels.
    order = (1, 0, 2, 3)
    twin = copy.deepcopy(scene)
    for key in ("cards", "rotations", "positions", "palette"):
        twin[key] = [copy.deepcopy(scene[key][i]) for i in order]
    return [("swap_first_two_card_records", twin)]
