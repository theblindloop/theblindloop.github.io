"""Deterministic renderer and analytic arm for straightest-branch routing."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 680
HEIGHT = 540
BACKGROUND = [248, 247, 242]
INK = (34, 37, 42)
GREEN = (35, 171, 96)
PURPLE = (137, 86, 205)
PALETTE = {
    "red": (220, 65, 75),
    "blue": (50, 105, 220),
    "orange": (238, 145, 35),
    "cyan": (25, 170, 190),
}
TERMINALS = ("T0", "T1", "T2", "T3")
EDGES = (
    ("S", "J0"),
    ("J0", "J1"),
    ("J0", "J2"),
    ("J1", "T0"),
    ("J1", "T1"),
    ("J2", "T2"),
    ("J2", "T3"),
)
QUARANTINE_DEGREES = 12.0


def sample_scene(seed: int) -> dict:
    """Sample a balanced visible tree without storing an answer field.

    Consecutive seeds cycle answer position. Answer color changes on a different
    four-scene cadence, covering every position/color pairing over 16 scenes.
    """

    rng = random.Random(int(seed))
    target = int(seed) % 4
    desired_color = (int(seed) // 4) % 4
    spread = 72 if int(seed) % 6 == 0 else 120
    terminal_y = [90.0, 90.0 + spread, 450.0 - spread, 450.0]
    child_y = [(terminal_y[0] + terminal_y[1]) / 2, (terminal_y[2] + terminal_y[3]) / 2]

    x_start, x_root, x_child, x_terminal = 70.0, 225.0, 410.0, 610.0
    selected_child_y = child_y[0 if target < 2 else 1]
    selected_terminal_y = terminal_y[target]
    # Make the desired child-to-terminal continuation nearly straight.
    root_y = selected_child_y - (selected_terminal_y - selected_child_y) * (
        (x_child - x_root) / (x_terminal - x_child)
    )
    # Likewise align START -> root with root -> desired child.
    start_y = root_y - (selected_child_y - root_y) * (
        (x_root - x_start) / (x_child - x_root)
    )

    dx = rng.randint(-6, 6)
    dy = rng.randint(-4, 4)
    raw = {
        "S": [x_start, start_y],
        "J0": [x_root, root_y],
        "J1": [x_child, child_y[0]],
        "J2": [x_child, child_y[1]],
        "T0": [x_terminal, terminal_y[0]],
        "T1": [x_terminal, terminal_y[1]],
        "T2": [x_terminal, terminal_y[2]],
        "T3": [x_terminal, terminal_y[3]],
    }
    nodes: dict[str, list[float]] = {}
    for name, (x, y) in raw.items():
        # Small independent perturbations prevent a single pixel template while
        # remaining far from the declared angular quarantine band.
        jx = rng.randint(-2, 2) if name not in {"S", "T0", "T1", "T2", "T3"} else 0
        jy = rng.randint(-2, 2)
        nodes[name] = [round(x + dx + jx, 3), round(y + dy + jy, 3)]

    colors = list(PALETTE)
    answer_color = colors[desired_color]
    remaining = [color for color in colors if color != answer_color]
    rng.shuffle(remaining)
    terminal_colors: dict[str, str] = {}
    for index, terminal in enumerate(TERMINALS):
        terminal_colors[terminal] = answer_color if index == target else remaining.pop()

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
        "nodes": nodes,
        "edges": [list(edge) for edge in EDGES],
        "terminal_colors": terminal_colors,
    }


def _adjacency(scene: dict) -> dict[str, list[str]]:
    nodes = scene["nodes"]
    adjacency = {name: [] for name in nodes}
    for left, right in scene["edges"]:
        adjacency[left].append(right)
        adjacency[right].append(left)
    return adjacency


def _deviation(incoming: tuple[float, float], outgoing: tuple[float, float]) -> float:
    a = math.hypot(*incoming)
    b = math.hypot(*outgoing)
    cosine = max(-1.0, min(1.0, (incoming[0] * outgoing[0] + incoming[1] * outgoing[1]) / (a * b)))
    return math.degrees(math.acos(cosine))


def _route(scene: dict) -> tuple[str, list[float]]:
    nodes = scene["nodes"]
    adjacency = _adjacency(scene)
    terminals = set(scene["terminal_colors"])
    previous: str | None = None
    current = "S"
    gaps: list[float] = []
    for _ in range(len(nodes) + 2):
        if current in terminals:
            return current, gaps
        options = [name for name in adjacency[current] if name != previous]
        if previous is None:
            if len(options) != 1:
                raise ValueError("START must have exactly one link")
            following = options[0]
        else:
            if len(options) < 2:
                raise ValueError("a junction must expose at least two forward choices")
            incoming = (
                nodes[current][0] - nodes[previous][0],
                nodes[current][1] - nodes[previous][1],
            )
            scored = []
            for name in options:
                outgoing = (
                    nodes[name][0] - nodes[current][0],
                    nodes[name][1] - nodes[current][1],
                )
                scored.append((_deviation(incoming, outgoing), name))
            scored.sort()
            gaps.append(scored[1][0] - scored[0][0])
            following = scored[0][1]
        previous, current = current, following
    raise ValueError("routing did not terminate")


def analytic_gold(scene: dict) -> str:
    terminal, _ = _route(scene)
    return scene["terminal_colors"][terminal]


def margin(scene: dict) -> float:
    _, gaps = _route(scene)
    return min(gaps)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_DEGREES


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # The serialized scene is a minimal, directly rendered parametrization.
    return []


def render(scene: dict) -> Image.Image:
    """Render edges behind colored circular nodes with 2x antialiasing."""

    scale = 2
    width, height = int(scene["width"]), int(scene["height"])
    background = tuple(scene["background"])
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    nodes = scene["nodes"]
    terminal_colors = scene["terminal_colors"]

    def point(name: str) -> tuple[int, int]:
        return (round(nodes[name][0] * scale), round(nodes[name][1] * scale))

    for left, right in scene["edges"]:
        draw.line([point(left), point(right)], fill=INK, width=8 * scale)

    for name, (x, y) in nodes.items():
        if name == "S":
            fill, outer, inner = GREEN, 20, 16
        elif name in terminal_colors:
            fill, outer, inner = PALETTE[terminal_colors[name]], 20, 16
        else:
            fill, outer, inner = PURPLE, 18, 14
        box_outer = [
            (x - outer) * scale,
            (y - outer) * scale,
            (x + outer) * scale,
            (y + outer) * scale,
        ]
        box_inner = [
            (x - inner) * scale,
            (y - inner) * scale,
            (x + inner) * scale,
            (y + inner) * scale,
        ]
        draw.ellipse(box_outer, fill=INK)
        draw.ellipse(box_inner, fill=fill)

    return image.resize((width, height), Image.Resampling.LANCZOS)
