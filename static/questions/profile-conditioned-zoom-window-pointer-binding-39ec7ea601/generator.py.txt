"""Scene sampling, rasterization, and analytic gold for the zoom-window world.

A horizontal tape of colored cells is shown once at unit scale. Below it, a
framed panel shows exactly three consecutive tape cells reproduced at twice the
linear size. Four down-pointing triangles sit above the tape; every triangle
points at a cell whose color equals the center color of the magnified window, so
the center color alone never localizes the window. Exactly one triangle points
at the cell that is the center of the magnified window.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

# ---------------------------------------------------------------- palettes

CELL_COLORS = {
    "crimson": (200, 30, 40),
    "azure": (20, 90, 210),
    "leaf": (30, 140, 60),
    "gold": (235, 180, 30),
    "violet": (130, 40, 170),
}
CELL_NAMES = tuple(sorted(CELL_COLORS))

POINTER_COLORS = {
    "orange": (255, 140, 0),
    "brown": (140, 70, 20),
    "teal": (0, 160, 160),
    "navy": (16, 32, 128),
}
POINTER_NAMES = ("orange", "brown", "teal", "navy")
WARM = ("orange", "brown")
COOL = ("teal", "navy")

# ---------------------------------------------------------------- geometry

N_CELLS = 14
CELL_W = 58
CELL_H = 72
TAPE_X0 = 18
TAPE_Y0 = 150
TAPE_Y1 = TAPE_Y0 + CELL_H

WIDTH = TAPE_X0 * 2 + N_CELLS * CELL_W
HEIGHT = 440

PTR_TOP = 64
PTR_BOT = 112
PTR_HALF = 22

MAG_SCALE = 2
MAG_W = CELL_W * MAG_SCALE
MAG_H = CELL_H * MAG_SCALE
MAG_X0 = (WIDTH - 3 * MAG_W) // 2
MAG_Y0 = 268
MAG_Y1 = MAG_Y0 + MAG_H

BLACK = (0, 0, 0)
WHITE = (255, 255, 255)

N_POINTERS = 4
MIN_POINTER_GAP = 3
QUARANTINE_MARGIN = 60.0


def _dist(a: tuple, b: tuple) -> float:
    return float(sum((int(x) - int(y)) ** 2 for x, y in zip(a, b))) ** 0.5


# ---------------------------------------------------------------- sampling


def _pointer_cells(rng: random.Random) -> list:
    while True:
        picks = sorted(rng.sample(range(1, N_CELLS - 1), N_POINTERS))
        gaps = [b - a for a, b in zip(picks, picks[1:])]
        if min(gaps) >= MIN_POINTER_GAP:
            return picks


def _windows(rng: random.Random, center: str) -> list:
    """Four distinct (left, center, right) triples sharing one center color.

    The true triple is index 0. One decoy shares the left cell, one shares the
    right cell, one shares neither, so no single flanking cell decides.
    """
    others = [c for c in CELL_NAMES if c != center]
    left, right = rng.sample(others, 2) if len(others) >= 2 else (others[0], others[0])
    right_alt = rng.choice([c for c in others if c != right])
    left_alt = rng.choice([c for c in others if c != left])
    left_alt2 = rng.choice([c for c in others if c != left])
    right_alt2 = rng.choice([c for c in others if c != right])
    return [
        (left, center, right),
        (left, center, right_alt),
        (left_alt, center, right),
        (left_alt2, center, right_alt2),
    ]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    for _ in range(4000):
        centers = _pointer_cells(rng)
        center_color = rng.choice(CELL_NAMES)
        windows = _windows(rng, center_color)
        if len({w for w in windows}) != N_POINTERS:
            continue
        slots = list(range(N_POINTERS))
        rng.shuffle(slots)
        # slots[k] is the window index placed at pointer position k
        cells = [None] * N_CELLS
        ok = True
        for pos, widx in zip(centers, slots):
            triple = windows[widx]
            for off, color in zip((-1, 0, 1), triple):
                if cells[pos + off] not in (None, color):
                    ok = False
                cells[pos + off] = color
        if not ok:
            continue
        for i in range(N_CELLS):
            if cells[i] is not None:
                continue
            choices = [
                c
                for c in CELL_NAMES
                if (i == 0 or cells[i - 1] != c)
                and (i + 1 >= N_CELLS or cells[i + 1] != c)
            ]
            if not choices:
                ok = False
                break
            cells[i] = rng.choice(choices)
        if not ok:
            continue
        if any(cells[i] == cells[i + 1] for i in range(N_CELLS - 1)):
            continue
        true_triple = windows[0]
        hits = [
            i
            for i in range(1, N_CELLS - 1)
            if tuple(cells[i - 1 : i + 2]) == true_triple
        ]
        if len(hits) != 1:
            continue
        target_slot = slots.index(0)
        if hits[0] != centers[target_slot]:
            continue
        # Counterbalance: the winning triangle's color is fixed by the seed and
        # is therefore independent of the tape, the window, and the position.
        winner = POINTER_NAMES[seed % N_POINTERS]
        rest = [c for c in POINTER_NAMES if c != winner]
        rng.shuffle(rest)
        pointer_colors = list(rest)
        pointer_colors.insert(target_slot, winner)
        return {
            "cells": list(cells),
            "pointer_cells": centers,
            "pointer_colors": pointer_colors,
            "target_slot": target_slot,
        }
    raise RuntimeError(f"scene sampling failed for seed {seed}")


# ---------------------------------------------------------------- rendering


def _cell_x(index: int) -> int:
    return TAPE_X0 + index * CELL_W


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), WHITE)
    draw = ImageDraw.Draw(image)

    for index, name in enumerate(scene["cells"]):
        x0 = _cell_x(index)
        draw.rectangle(
            [x0, TAPE_Y0, x0 + CELL_W - 1, TAPE_Y1 - 1],
            fill=CELL_COLORS[name],
            outline=BLACK,
            width=2,
        )

    for pos, name in zip(scene["pointer_cells"], scene["pointer_colors"]):
        cx = _cell_x(pos) + CELL_W // 2
        draw.polygon(
            [(cx - PTR_HALF, PTR_TOP), (cx + PTR_HALF, PTR_TOP), (cx, PTR_BOT)],
            fill=POINTER_COLORS[name],
        )

    center = scene["pointer_cells"][scene["target_slot"]]
    triple = scene["cells"][center - 1 : center + 2]
    for k, name in enumerate(triple):
        x0 = MAG_X0 + k * MAG_W
        draw.rectangle(
            [x0, MAG_Y0, x0 + MAG_W - 1, MAG_Y1 - 1],
            fill=CELL_COLORS[name],
            outline=BLACK,
            width=4,
        )
    draw.rectangle(
        [MAG_X0 - 6, MAG_Y0 - 6, MAG_X0 + 3 * MAG_W + 5, MAG_Y1 + 5],
        outline=BLACK,
        width=5,
    )
    return image


# ---------------------------------------------------------------- gold


def analytic_gold(scene: dict) -> str:
    return scene["pointer_colors"][scene["target_slot"]]


def _window_of(scene: dict, slot: int) -> tuple:
    pos = scene["pointer_cells"][slot]
    return tuple(scene["cells"][pos - 1 : pos + 2])


def margin(scene: dict) -> float:
    """Smallest RGB mismatch between the true window and any decoy window."""
    truth = _window_of(scene, scene["target_slot"])
    best = None
    for slot in range(N_POINTERS):
        if slot == scene["target_slot"]:
            continue
        cost = sum(
            _dist(CELL_COLORS[a], CELL_COLORS[b])
            for a, b in zip(_window_of(scene, slot), truth)
        )
        best = cost if best is None else min(best, cost)
    return float(best)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list:
    """No latent aliases exist: every label input is drawn."""
    return []
