"""Pixel-only inverse arm.

Receives a rendered PNG and nothing else. It rediscovers the three horizontal
bands, reads the tape's ordered cell colors, reads the magnified window's three
colors, localizes the window by an exact ordered-triple match over the tape, and
only then reports the color of the triangle that points at the matched cell.
Any ambiguity anywhere returns "abstain".
"""

from __future__ import annotations

from collections import Counter

from PIL import Image

ABSTAIN = "abstain"

_TAPE_REF = {
    "crimson": (200, 30, 40),
    "azure": (20, 90, 210),
    "leaf": (30, 140, 60),
    "gold": (235, 180, 30),
    "violet": (130, 40, 170),
}
_PTR_REF = {
    "orange": (255, 140, 0),
    "brown": (140, 70, 20),
    "teal": (0, 160, 160),
    "navy": (16, 32, 128),
}

_TOL = 40.0
_MIN_BAND_H = 16
_MIN_GAP = 12


def _d2(a, b) -> float:
    return sum((int(x) - int(y)) ** 2 for x, y in zip(a, b))


def _classify(pixel, reference):
    best_name, best = None, None
    for name, ref in reference.items():
        value = _d2(pixel, ref)
        if best is None or value < best:
            best_name, best = name, value
    if best is None or best > _TOL * _TOL:
        return None
    return best_name


def _is_ink(pixel) -> bool:
    return _d2(pixel, (255, 255, 255)) > 900


def _bands(pixels, width, height):
    rows = []
    for y in range(height):
        rows.append(any(_is_ink(pixels[x, y]) for x in range(0, width, 2)))
    groups, start = [], None
    blanks = 0
    for y in range(height):
        if rows[y]:
            if start is None:
                start = y
            blanks = 0
        elif start is not None:
            blanks += 1
            if blanks >= _MIN_GAP:
                groups.append((start, y - blanks))
                start = None
    if start is not None:
        groups.append((start, height - 1))
    return [g for g in groups if g[1] - g[0] + 1 >= _MIN_BAND_H]


def _runs(pixels, width, y, reference, min_len):
    out = []
    current, start = None, 0
    for x in range(width + 1):
        name = _classify(pixels[x, y], reference) if x < width else None
        if name != current:
            if current is not None and x - start >= min_len:
                out.append((current, (start + x - 1) / 2.0))
            current, start = name, x
    return out


def _column_groups(pixels, width, top, bottom):
    cols = []
    for x in range(width):
        cols.append(any(_is_ink(pixels[x, y]) for y in range(top, bottom + 1)))
    groups, start = [], None
    for x in range(width + 1):
        filled = cols[x] if x < width else False
        if filled and start is None:
            start = x
        elif not filled and start is not None:
            groups.append((start, x - 1))
            start = None
    return [g for g in groups if g[1] - g[0] + 1 >= 8]


def decision_from_image(image: "Image.Image") -> str:
    image = image.convert("RGB")
    width, height = image.size
    pixels = image.load()

    bands = _bands(pixels, width, height)
    if len(bands) != 3:
        return ABSTAIN
    (ptr_top, ptr_bot), (tape_top, tape_bot), (mag_top, mag_bot) = bands

    tape = _runs(pixels, width, (tape_top + tape_bot) // 2, _TAPE_REF, 15)
    if len(tape) < 5:
        return ABSTAIN
    mag = _runs(pixels, width, (mag_top + mag_bot) // 2, _TAPE_REF, 40)
    if len(mag) != 3:
        return ABSTAIN

    target = tuple(name for name, _ in mag)
    sequence = [name for name, _ in tape]
    centers = [cx for _, cx in tape]
    hits = [
        i
        for i in range(1, len(sequence) - 1)
        if tuple(sequence[i - 1 : i + 2]) == target
    ]
    if len(hits) != 1:
        return ABSTAIN
    matched_x = centers[hits[0]]

    pointers = []
    for x0, x1 in _column_groups(pixels, width, ptr_top, ptr_bot):
        tally = Counter()
        for x in range(x0, x1 + 1):
            for y in range(ptr_top, ptr_bot + 1):
                pixel = pixels[x, y]
                if not _is_ink(pixel):
                    continue
                name = _classify(pixel, _PTR_REF)
                if name is not None:
                    tally[name] += 1
        if not tally:
            return ABSTAIN
        pointers.append((name_of(tally), (x0 + x1) / 2.0))
    if len(pointers) != 4:
        return ABSTAIN
    if len({name for name, _ in pointers}) != 4:
        return ABSTAIN

    ranked = sorted(pointers, key=lambda item: abs(item[1] - matched_x))
    if abs(ranked[0][1] - matched_x) > 0.5 * _spacing(centers):
        return ABSTAIN
    if abs(ranked[1][1] - matched_x) <= abs(ranked[0][1] - matched_x):
        return ABSTAIN
    return ranked[0][0]


def name_of(tally: Counter) -> str:
    return tally.most_common(1)[0][0]


def _spacing(centers) -> float:
    if len(centers) < 2:
        return 1.0
    gaps = [b - a for a, b in zip(centers, centers[1:])]
    return sorted(gaps)[len(gaps) // 2]
