"""Concentric ring count world.

A field of identical dark dots is arranged on K nested circles that all share
one common center. Every dot on a single circle sits on the same ring, and the
circles are separated from each other by a conspicuously large radial gap (much
bigger than the arc spacing between neighbouring dots on one circle). The
decision is: how many rings of dots are there? Answer set {2, 3, 4}.

This is a global Gestalt structure decision: the whole dot set organizes into
concentric layers purely by its radial arrangement around one latent center,
revealed by the *nesting* of the surrounding dots rather than by any single
dot's size, colour, or absolute position.

The renderer owns scene sampling + rasterization + the structural analytic
gold. The independent pixel-only inverse arm (``oracle.py``) recovers every dot
centroid from the final PNG alone, estimates the common center as the centroid
of the whole symmetric constellation, measures each dot's radius, and counts
how many concentric radius layers (rings) the large radial-gap boundaries
divide the constellation into.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
CENTER = (WIDTH / 2.0, HEIGHT / 2.0)
BACKGROUND = (255, 255, 255)
INK = (30, 30, 30)

RING_COUNTS = [2, 3, 4]

# Nominal ring radii for the first up-to-4 rings; spacing between rings is a
# conspicuously large radial gap (~46 px), far larger than dot size or arc gap.
NOMINAL_RADII = [56.0, 102.0, 148.0, 194.0]
RADIUS_JITTER = 4.0

# Number of dots per ring and their dot radius.
DOTS_PER_RING = (5, 8)
DOT_RADIUS = (6, 8)

# Independent per-ring angular offset so the whole constellation is never
# axis-aligned; dots on one ring are otherwise equally spaced (symmetric).
TRANSLATION_SPAN = 28.0

# Quarantine when the between-ring radial gap is not clearly separated from the
# (near-zero) within-ring radius spread.
QUARANTINE_CLEARANCE_PX = 25.0


def sample_scene(seed: int) -> dict:
    """Sample one latent scene deterministically from ``seed``."""
    rng = np.random.default_rng(seed)
    k = int(rng.integers(0, len(RING_COUNTS)))
    return {"ring_count": RING_COUNTS[k], "layout_seed": int(seed)}


def _layout(scene: dict) -> dict:
    """Deterministic per-ring geometry from the layout seed.

    Returns per-ring radius, dot count, per-ring dot radius, and the shared
    constellation translation. Every dot on a ring sits exactly at that ring's
    radius so the radius feature is degenerate within a ring.
    """
    rng = np.random.default_rng(scene["layout_seed"])
    k = scene["ring_count"]
    tx = float(rng.uniform(-TRANSLATION_SPAN, TRANSLATION_SPAN))
    ty = float(rng.uniform(-TRANSLATION_SPAN, TRANSLATION_SPAN))
    rings = []
    for i in range(k):
        r = NOMINAL_RADII[i] + float(rng.uniform(-RADIUS_JITTER, RADIUS_JITTER))
        n = int(rng.integers(*DOTS_PER_RING))
        dot_r = int(rng.integers(*DOT_RADIUS))
        offset = float(rng.uniform(0.0, 2.0 * math.pi))
        rings.append(
            {
                "radius": r,
                "count": n,
                "dot_radius": dot_r,
                "offset": offset,
            }
        )
    return {"tx": tx, "ty": ty, "rings": rings, "ring_count": k}


def _dot_positions(layout: dict) -> list[tuple[float, float, float]]:
    """Return (x, y, dot_radius) for every dot in the constellation."""
    cx, cy = CENTER
    tx, ty = layout["tx"], layout["ty"]
    positions: list[tuple[float, float, float]] = []
    for ring in layout["rings"]:
        r = ring["radius"]
        n = ring["count"]
        dr = ring["dot_radius"]
        offset = ring["offset"]
        for idx in range(n):
            angle = offset + 2.0 * math.pi * idx / n
            x = cx + tx + r * math.cos(angle)
            y = cy + ty + r * math.sin(angle)
            positions.append((x, y, float(dr)))
    return positions


def _draw_scene(scene: dict) -> Image.Image:
    layout = _layout(scene)
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    d = ImageDraw.Draw(img)
    for (x, y, dr) in _dot_positions(layout):
        d.ellipse([x - dr, y - dr, x + dr, y + dr], fill=INK)
    return img


def render(scene: dict) -> Image.Image:
    """Rasterize the latent scene (a deterministic PNG)."""
    # _layout reads ring_count too, so rendering is a function of the declared
    # latent fields and the raster stays injective in the decision variable.
    _ = scene["ring_count"]
    return _draw_scene(scene)


def analytic_gold(scene: dict) -> str:
    """Structural gold: the number of concentric rings is the declared count."""
    return str(scene["ring_count"])


def _ring_radii(scene: dict) -> list[float]:
    layout = _layout(scene)
    return [ring["radius"] for ring in layout["rings"]]


def margin(scene: dict) -> float:
    """Commitment: min between-ring radial gap (the large-gap boundary clearance).

    Larger indicates the concentric layers are more decisively separated, given
    the within-ring radius spread is essentially zero (all dots on a circle).
    """
    radii = sorted(_ring_radii(scene))
    if len(radii) < 2:
        return 0.0
    return float(min(radii[i + 1] - radii[i] for i in range(len(radii) - 1)))


def is_quarantined(scene: dict) -> bool:
    """Quarantine when the between-ring radial gaps are not cleanly separated."""
    return margin(scene) < QUARANTINE_CLEARANCE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The constellation is injected from the seed; no latent aliases declared."""
    return []
