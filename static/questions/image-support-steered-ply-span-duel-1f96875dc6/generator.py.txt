"""Ply-span duel: two distant folded cords, one along-cord separation each.

Two serpentine cords of the same dark-teal ink sit in diagonally opposite
corner boxes with a wide empty gap between them.  Each cord carries exactly two
small square beads of its own colour (one cord orange, the other purple), both
beads sitting in the same column of the fold so that the *straight-line* gap
between a cord's two beads is deliberately (almost) the same on both sides.
What differs is the distance travelled *along the cord* between the two beads,
because the two cords are folded with different run lengths and different fold
pitches.  The answer is the colour whose bead pair is farther apart along its
own cord.

Geometry.  A cord with `rows` horizontal runs of centre-line length ``L`` and
fold pitch ``p`` is traversed boustrophedon-style.  Two beads placed in the same
column ``a`` of runs ``i`` and ``i + k`` with **even** ``k`` are separated by
exactly ``k * (L + p)`` of cord and by exactly ``k * p`` of straight line, the
column offset cancelling out.  Both quantities are therefore closed-form, and
the sampler equalises ``k * p`` across the two sides while forcing ``k*(L+p)``
apart.

Every drawn element is an axis-aligned rectangle, so the raster is crisp and
byte-deterministic.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 720
HEIGHT = 560

BG = (250, 250, 248)
CORD = (22, 96, 112)
ORANGE = (232, 124, 28)
PURPLE = (124, 56, 182)
GRAY = (156, 158, 164)

_RGB = {"orange": ORANGE, "purple": PURPLE}

HALF = 3            # half stroke width (stroke width 7)
BEAD = 5            # half bead width (bead 11 x 11)
BOX_W = 312
BOX_H = 252
PAD = 20

CORNERS = {
    "TL": (PAD, PAD),
    "TR": (WIDTH - PAD - BOX_W, PAD),
    "BL": (PAD, HEIGHT - PAD - BOX_H),
    "BR": (WIDTH - PAD - BOX_W, HEIGHT - PAD - BOX_H),
}
DIAGONALS = (("TL", "BR"), ("BR", "TL"), ("TR", "BL"), ("BL", "TR"))
UNUSED = {
    ("TL", "BR"): ("TR", "BL"),
    ("BR", "TL"): ("BL", "TR"),
    ("TR", "BL"): ("TL", "BR"),
    ("BL", "TR"): ("BR", "TL"),
}

QUARANTINE_MARGIN = 36.0
MAX_ROW_SPAN = BOX_H - 44
MAX_RUN = BOX_W - 44


def _even(value: float) -> int:
    return int(round(value / 2.0)) * 2


def _cord_geometry(side: dict) -> tuple[list[tuple[int, int, int, int]], list[tuple[int, int]]]:
    """Return the stroke rectangles and the two bead centres of one cord."""

    rows = side["rows"]
    run = side["run"]
    pitch = side["pitch"]
    x0, y0 = side["origin"]
    rects: list[tuple[int, int, int, int]] = []
    for index in range(rows):
        y = y0 + index * pitch
        rects.append((x0 - HALF, y - HALF, x0 + run + HALF, y + HALF))
        if index + 1 < rows:
            x = x0 + run if index % 2 == 0 else x0
            rects.append((x - HALF, y - HALF, x + HALF, y + pitch + HALF))
    first = side["bead_row"]
    beads = [
        (x0 + side["column"], y0 + first * pitch),
        (x0 + side["column"], y0 + (first + side["span_rows"]) * pitch),
    ]
    return rects, beads


def _decoy_rects(scene: dict) -> list[tuple[int, int, int, int]]:
    rects: list[tuple[int, int, int, int]] = []
    for blob in scene["decoys"]:
        if blob["kind"] == "square":
            cx, cy, half = blob["x"], blob["y"], blob["half"]
            rects.append((cx - half, cy - half, cx + half, cy + half))
        else:
            rects.extend(_cord_geometry(blob)[0])
    return rects


def _elements(scene: dict) -> list[dict]:
    """Every drawn rectangle with its semantic role (used by erasure tests)."""

    items: list[dict] = []
    for rect in _decoy_rects(scene):
        items.append({"role": "decoy", "rect": rect, "color": GRAY})
    for tag in ("a", "b"):
        side = scene["sides"][tag]
        rects, beads = _cord_geometry(side)
        for rect in rects:
            items.append({"role": f"cord_{tag}", "rect": rect, "color": CORD})
        for order, (cx, cy) in enumerate(beads):
            items.append(
                {
                    "role": f"bead_{tag}_{order}",
                    "rect": (cx - BEAD, cy - BEAD, cx + BEAD, cy + BEAD),
                    "color": _RGB[side["color"]],
                }
            )
    return items


def _fits(rows: int, run: int, pitch: int, span: int) -> bool:
    return (
        3 <= rows <= 10
        and span + 1 <= rows
        and (rows - 1) * pitch <= MAX_ROW_SPAN
        and 56 <= run <= MAX_RUN
        and 24 <= pitch <= 62
    )


def _ink(rows: int, run: int, pitch: int) -> int:
    return rows * run + (rows - 1) * pitch


def sample_scene(seed: int) -> dict:
    """Sample one balanced scene; both cords are always jointly necessary."""

    rng = random.Random(seed * 7919 + 13)
    for _ in range(20000):
        span_a = rng.choice((2, 2, 4))
        pitch_a = rng.randrange(24, 62, 2)
        euclid = span_a * pitch_a
        span_b = rng.choice((2, 4))
        pitch_b = _even(euclid / span_b)
        if abs(span_b * pitch_b - euclid) > 6:
            continue
        run_a = rng.randrange(56, MAX_RUN + 1, 2)
        geo_a = span_a * (run_a + pitch_a)
        tight = rng.random() < 0.28
        ratio = rng.uniform(1.02, 1.20) if tight else rng.uniform(1.32, 1.80)
        if rng.random() < 0.5:
            ratio = 1.0 / ratio
        run_b = _even(geo_a * ratio / span_b - pitch_b)
        geo_b = span_b * (run_b + pitch_b)
        if abs(geo_a - geo_b) < 6:
            continue
        rows_a = rng.randint(span_a + 1, 10)
        if not _fits(rows_a, run_a, pitch_a, span_a):
            continue
        ink_a = _ink(rows_a, run_a, pitch_a)
        rows_b = int(round((ink_a + pitch_b) / float(run_b + pitch_b)))
        if not _fits(rows_b, run_b, pitch_b, span_b):
            continue
        if abs(_ink(rows_b, run_b, pitch_b) - ink_a) > 16:
            continue
        column_a = rng.randrange(22, run_a - 21, 2)
        column_b = rng.randrange(22, run_b - 21, 2)
        pair = DIAGONALS[rng.randrange(4)]
        first_color = "orange" if rng.random() < 0.5 else "purple"

        sides = {}
        for tag, corner, rows, run, pitch, span, column, color in (
            ("a", pair[0], rows_a, run_a, pitch_a, span_a, column_a, first_color),
            (
                "b",
                pair[1],
                rows_b,
                run_b,
                pitch_b,
                span_b,
                column_b,
                "purple" if first_color == "orange" else "orange",
            ),
        ):
            bx, by = CORNERS[corner]
            origin = (
                bx + (BOX_W - run) // 2,
                by + (BOX_H - (rows - 1) * pitch) // 2,
            )
            sides[tag] = {
                "corner": corner,
                "rows": rows,
                "run": run,
                "pitch": pitch,
                "span_rows": span,
                "bead_row": rng.randrange(0, rows - span),
                "column": column,
                "origin": origin,
                "color": color,
            }

        decoys = _sample_decoys(rng, pair)
        return {"sides": sides, "decoys": decoys, "pair": list(pair)}
    raise RuntimeError("scene sampling failed")


def _sample_decoys(rng: random.Random, pair: tuple[str, str]) -> list[dict]:
    """Gray shapes: one gray serpentine plus loose squares, never decisive."""

    free = UNUSED[pair]
    rows = rng.randint(3, 6)
    pitch = rng.randrange(26, 50, 2)
    run = rng.randrange(60, 200, 2)
    while (rows - 1) * pitch > MAX_ROW_SPAN:
        rows -= 1
    bx, by = CORNERS[free[0]]
    decoys: list[dict] = [
        {
            "kind": "cord",
            "rows": rows,
            "run": run,
            "pitch": pitch,
            "span_rows": 2,
            "bead_row": 0,
            "column": run // 2,
            "origin": (bx + (BOX_W - run) // 2, by + (BOX_H - (rows - 1) * pitch) // 2),
            "color": "gray",
        }
    ]
    cx0, cy0 = CORNERS[free[1]]
    for _ in range(rng.randint(3, 6)):
        decoys.append(
            {
                "kind": "square",
                "x": cx0 + rng.randrange(24, BOX_W - 24),
                "y": cy0 + rng.randrange(24, BOX_H - 24),
                "half": rng.choice((6, 7, 8)),
            }
        )
    return decoys


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    for item in _elements(scene):
        draw.rectangle(item["rect"], fill=tuple(item["color"]))
    return image


def _spans(scene: dict) -> dict:
    out = {}
    for tag in ("a", "b"):
        side = scene["sides"][tag]
        out[side["color"]] = {
            "geodesic": float(side["span_rows"] * (side["run"] + side["pitch"])),
            "euclid": float(side["span_rows"] * side["pitch"]),
        }
    return out


def analytic_gold(scene: dict) -> str:
    spans = _spans(scene)
    return "orange" if spans["orange"]["geodesic"] > spans["purple"]["geodesic"] else "purple"


def margin(scene: dict) -> float:
    """Decisive along-cord gap, capped by the weakest rendering clearance."""

    spans = _spans(scene)
    decisive = abs(spans["orange"]["geodesic"] - spans["purple"]["geodesic"])
    clearances = []
    for tag in ("a", "b"):
        side = scene["sides"][tag]
        clearances.append(side["column"] - BEAD - HALF)
        clearances.append(side["run"] - side["column"] - BEAD - HALF)
        clearances.append(side["pitch"] - 2 * BEAD - 2 * HALF)
    gap = _component_gap(scene)
    return float(round(min(decisive, 4.0 * min(clearances), 2.0 * gap), 6))


def _component_gap(scene: dict) -> float:
    """Smallest background gap between the two decisive cords."""

    boxes = []
    for tag in ("a", "b"):
        rects, beads = _cord_geometry(scene["sides"][tag])
        xs = [r[0] for r in rects] + [r[2] for r in rects]
        ys = [r[1] for r in rects] + [r[3] for r in rects]
        boxes.append((min(xs), min(ys), max(xs), max(ys)))
    left, right = boxes
    dx = max(left[0] - right[2], right[0] - left[2], 0.0)
    dy = max(left[1] - right[3], right[1] - left[3], 0.0)
    return float(math.hypot(dx, dy))


def is_quarantined(scene: dict) -> bool:
    spans = _spans(scene)
    euclid_skew = abs(spans["orange"]["euclid"] - spans["purple"]["euclid"])
    return bool(margin(scene) < QUARANTINE_MARGIN or euclid_skew > 6.0)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical latent rewrites that must preserve the answer."""

    out: list[tuple[str, dict]] = []
    reordered = dict(scene)
    reordered["sides"] = {"b": scene["sides"]["b"], "a": scene["sides"]["a"]}
    out.append(("side_record_order_swapped", reordered))

    shuffled = dict(scene)
    shuffled["decoys"] = list(reversed(scene["decoys"]))
    out.append(("decoy_order_reversed", shuffled))
    return out
