"""Latent-scene sampler and Pillow renderer for Column Drift Correspondence."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw


CANVAS = 840
BACKGROUND = (248, 246, 238)
CHARCOAL = (43, 47, 52)
KEY_BLUE = (45, 105, 218)
TILE_COLORS = (
    (218, 73, 83),
    (32, 151, 137),
    (237, 190, 56),
    (102, 91, 181),
    (103, 170, 78),
)
TAG_RGB = {
    "amber": (244, 151, 28),
    "cyan": (33, 186, 222),
    "violet": (166, 69, 190),
}
ANSWERS = ("amber", "cyan", "violet")
GRID_SIZE = 7


def _drift(source: list[list[int]], drifts: list[int]) -> list[list[int]]:
    height = len(source)
    width = len(source[0])
    output = [[0 for _ in range(width)] for _ in range(height)]
    for column, amount in enumerate(drifts):
        for row in range(height):
            output[(row + amount) % height][column] = source[row][column]
    return output


def _mismatch(left: list[list[int]], right: list[list[int]]) -> int:
    return sum(a != b for row_a, row_b in zip(left, right) for a, b in zip(row_a, row_b))


def _distractor(
    target: list[list[int]], rng: random.Random, pair_count: int
) -> list[list[int]]:
    """Swap disjoint unequal cells so color counts are exactly preserved."""

    flat = [value for row in target for value in row]
    for _ in range(200):
        result = flat[:]
        available = list(range(len(flat)))
        rng.shuffle(available)
        used: set[int] = set()
        swaps = 0
        for first in available:
            if first in used:
                continue
            choices = [
                second
                for second in available
                if second not in used and second != first and result[second] != result[first]
            ]
            if not choices:
                continue
            second = choices[rng.randrange(len(choices))]
            result[first], result[second] = result[second], result[first]
            used.update((first, second))
            swaps += 1
            if swaps == pair_count:
                break
        if swaps == pair_count:
            grid = [result[index : index + GRID_SIZE] for index in range(0, 49, GRID_SIZE)]
            if _mismatch(grid, target) >= 2 * pair_count:
                return grid
    raise RuntimeError("could not construct a separated equal-histogram distractor")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0xC011D21F)
    source_values = [value for value, count in enumerate((10, 10, 10, 10, 9)) for _ in range(count)]
    rng.shuffle(source_values)
    source = [
        source_values[index : index + GRID_SIZE]
        for index in range(0, GRID_SIZE * GRID_SIZE, GRID_SIZE)
    ]
    drifts = [1 + rng.randrange(3) for _ in range(GRID_SIZE)]
    target = _drift(source, drifts)

    first = _distractor(target, rng, 2 + (abs(int(seed)) % 3))
    second = _distractor(target, rng, 3 + ((abs(int(seed)) // 3) % 3))
    while second == first:
        second = _distractor(target, rng, 4)

    correct_position = (abs(int(seed)) // 3) % 3
    proposals = [first, second]
    proposals.insert(correct_position, target)

    desired_answer = ANSWERS[abs(int(seed)) % 3]
    remaining = [value for value in ANSWERS if value != desired_answer]
    rng.shuffle(remaining)
    tag_colors = remaining[:]
    tag_colors.insert(correct_position, desired_answer)
    return {
        "source": source,
        "drifts": drifts,
        "proposals": proposals,
        "tag_colors": tag_colors,
    }


def _matching_index(
    source: list[list[int]], drifts: list[int], proposals: list[list[list[int]]]
) -> tuple[int, list[int]]:
    target = _drift(source, drifts)
    distances = [_mismatch(target, proposal) for proposal in proposals]
    matches = [index for index, distance in enumerate(distances) if distance == 0]
    if len(matches) != 1:
        raise ValueError("scene does not contain exactly one matching proposal")
    return matches[0], distances


def analytic_gold(scene: dict) -> str:
    source = scene["source"]
    drifts = scene["drifts"]
    proposals = scene["proposals"]
    tag_colors = scene["tag_colors"]
    winner, _ = _matching_index(source, drifts, proposals)
    return str(tag_colors[winner])


def margin(scene: dict) -> float:
    source = scene["source"]
    drifts = scene["drifts"]
    proposals = scene["proposals"]
    _, distances = _matching_index(source, drifts, proposals)
    return float(sorted(distances)[1])


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 4.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _draw_grid(draw: ImageDraw.ImageDraw, grid: list[list[int]], origin: tuple[int, int]) -> None:
    x0, y0 = origin
    cell = 48
    inset = 4
    for row in range(GRID_SIZE):
        for column in range(GRID_SIZE):
            left = x0 + column * cell + inset
            top = y0 + row * cell + inset
            right = x0 + (column + 1) * cell - inset - 1
            bottom = y0 + (row + 1) * cell - inset - 1
            draw.rectangle((left, top, right, bottom), fill=TILE_COLORS[grid[row][column]])


def render(scene: dict) -> Image.Image:
    source = scene["source"]
    drifts = scene["drifts"]
    proposals = scene["proposals"]
    tag_colors = scene["tag_colors"]

    image = Image.new("RGB", (CANVAS, CANVAS), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((12, 12, 827, 827), radius=20, fill=CHARCOAL)

    positions = ((36, 92), (468, 92), (36, 492), (468, 492))
    _draw_grid(draw, source, positions[0])
    for proposal, position in zip(proposals, positions[1:]):
        _draw_grid(draw, proposal, position)

    for column, count in enumerate(drifts):
        center_x = positions[0][0] + column * 48 + 24
        start_y = 54 - (count - 1) * 8
        for index in range(count):
            center_y = start_y + index * 16
            draw.ellipse(
                (center_x - 5, center_y - 5, center_x + 5, center_y + 5),
                fill=KEY_BLUE,
            )

    tag_boxes = ((476, 30, 795, 72), (44, 442, 363, 480), (476, 442, 795, 480))
    for color_name, box in zip(tag_colors, tag_boxes):
        draw.rounded_rectangle(box, radius=10, fill=TAG_RGB[color_name])
    return image
