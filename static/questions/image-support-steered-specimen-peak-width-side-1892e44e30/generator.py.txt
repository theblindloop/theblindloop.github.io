"""Deterministic latent sampler, renderer, and analytic arm."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

PALETTE = ("#087f8c", "#7b2cbf", "#c8553d", "#2a6f3b")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    winner_above = (int(seed) % 2) == 0
    narrow = rng.randint(94, 108)
    gap = rng.randint(18, 42)
    wide = narrow + gap
    upper = wide if winner_above else narrow
    lower = narrow if winner_above else wide
    # A wider peak is deliberately shorter vertically, preventing total lobe ink
    # from being a reliable substitute for the requested extremal measurement.
    wide_height = rng.randint(16, 24)
    narrow_height = rng.randint(40, 52)
    upper_h = wide_height if winner_above else narrow_height
    lower_h = narrow_height if winner_above else wide_height
    return {
        "canvas_size": 512,
        "center_x": rng.randint(236, 276),
        "center_y": rng.randint(244, 268),
        "upper_halfwidth": upper,
        "lower_halfwidth": lower,
        "upper_plateau_height": upper_h,
        "lower_plateau_height": lower_h,
        "waist_halfwidth": rng.randint(48, 56),
        "color": PALETTE[rng.randrange(len(PALETTE))],
    }


def _profile(scene: dict) -> list[tuple[int, int]]:
    cy = int(scene["center_y"])
    upper = int(scene["upper_halfwidth"])
    lower = int(scene["lower_halfwidth"])
    upper_h = int(scene["upper_plateau_height"])
    lower_h = int(scene["lower_plateau_height"])
    waist = int(scene["waist_halfwidth"])
    return [
        (cy - 126, 72),
        (cy - 108, 84),
        (cy - 70 - upper_h // 2, upper),
        (cy - 70 + upper_h // 2, upper),
        (cy - 22, 74),
        (cy - 8, waist + 5),
        (cy, waist),
        (cy + 8, waist + 5),
        (cy + 22, 74),
        (cy + 70 - lower_h // 2, lower),
        (cy + 70 + lower_h // 2, lower),
        (cy + 108, 84),
        (cy + 126, 72),
    ]


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas_size"])
    cx = int(scene["center_x"])
    color = str(scene["color"])
    profile = _profile(scene)
    left = [(cx - halfwidth, y) for y, halfwidth in profile]
    right = [(cx + halfwidth, y) for y, halfwidth in reversed(profile)]
    image = Image.new("RGB", (size, size), "white")
    ImageDraw.Draw(image).polygon(left + right, fill=color)
    return image


def analytic_gold(scene: dict) -> str:
    return "above" if int(scene["upper_halfwidth"]) > int(scene["lower_halfwidth"]) else "below"


def margin(scene: dict) -> float:
    return float(2 * abs(int(scene["upper_halfwidth"]) - int(scene["lower_halfwidth"])))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 14.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
