"""Independent image-only inverse arm for a single filled specimen."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _mask_from_image(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    border = np.concatenate((arr[:3].reshape(-1, 3), arr[-3:].reshape(-1, 3), arr[:, :3].reshape(-1, 3), arr[:, -3:].reshape(-1, 3)))
    background = np.median(border, axis=0)
    distance = np.linalg.norm(arr - background, axis=2)
    return distance > 70.0


def _row_profile(mask: np.ndarray) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    ys = np.flatnonzero(mask.any(axis=1))
    if len(ys) < 180 or mask.sum() < 15000:
        raise ValueError("no intact regional specimen")
    if ys[0] < 20 or ys[-1] >= mask.shape[0] - 20:
        raise ValueError("specimen touches image boundary")
    lefts = []
    rights = []
    for y in ys:
        xs = np.flatnonzero(mask[y])
        if len(xs) == 0 or len(xs) != int(xs[-1] - xs[0] + 1):
            raise ValueError("specimen row is split or contains erased evidence")
        lefts.append(int(xs[0]))
        rights.append(int(xs[-1]))
    left = np.asarray(lefts)
    right = np.asarray(rights)
    if int(right.max() - left.min() + 1) < 150:
        raise ValueError("specimen is too narrow")
    if np.max(np.abs(np.diff(left))) > 8 or np.max(np.abs(np.diff(right))) > 8:
        raise ValueError("specimen boundary has a destructive discontinuity")
    return ys, left, right


def decision_from_image(image: Image.Image) -> str:
    mask = _mask_from_image(image)
    ys, left, right = _row_profile(mask)
    widths = right - left + 1
    # Endpoint taper is excluded; the unique interior minimum is the visible waist.
    trim = max(20, len(widths) // 8)
    interior = widths[trim:-trim]
    waist_index = trim + int(np.argmin(interior))
    if waist_index < len(widths) * 0.35 or waist_index > len(widths) * 0.65:
        raise ValueError("no central narrow waist")
    guard = 8
    upper = widths[trim : waist_index - guard]
    lower = widths[waist_index + guard : len(widths) - trim]
    if len(upper) < 30 or len(lower) < 30:
        raise ValueError("insufficient rows around waist")
    upper_max = int(upper.max())
    lower_max = int(lower.max())
    if abs(upper_max - lower_max) < 14:
        raise ValueError("peak-width comparison is raster-ambiguous")
    return "above" if upper_max > lower_max else "below"
