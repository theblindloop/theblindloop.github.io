"""Anchor-row realignment world.

Each row of a 7x8 cell grid has been cyclically slid sideways. One cell per row
carries a bright anchor colour; the amount that row was slid is exactly the
column its anchor now occupies. Undoing the seven declared slides reveals a 7x7
pattern of dark cells that matches one of five named abstract symbols.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

ROWS = 7
COLS = 8
SPAN = COLS - 1  # symbol columns after the anchor column
SYMBOLS = ("plus", "ex", "box", "tee", "diamond")
QUARANTINE_MARGIN = 2.0

_BACKGROUNDS = ([255, 255, 255], [248, 248, 246], [250, 250, 252])
_LINES = ([150, 150, 150], [160, 158, 155], [140, 145, 150])
_INKS = ([30, 30, 34], [45, 45, 45], [25, 32, 40], [52, 48, 44])
_ANCHORS = ([214, 44, 44], [224, 120, 20], [190, 30, 120], [28, 138, 62])
_CELLS = (30, 34, 38)
_WIDTH = 372
_HEIGHT = 328


def _template(name: str) -> list[list[int]]:
    grid = [[0] * SPAN for _ in range(ROWS)]
    for r in range(ROWS):
        for c in range(SPAN):
            if name == "plus":
                on = r == 3 or c == 3
            elif name == "ex":
                on = r == c or r + c == 6
            elif name == "box":
                on = 1 <= r <= 5 and 1 <= c <= 5 and (r in (1, 5) or c in (1, 5))
            elif name == "tee":
                on = r == 0 or (c == 3 and r >= 1)
            else:
                on = abs(r - 3) + abs(c - 3) == 3
            grid[r][c] = 1 if on else 0
    return grid


TEMPLATES = {name: _template(name) for name in SYMBOLS}


def _distance(pattern: list[list[int]], template: list[list[int]]) -> int:
    return sum(
        1 for r in range(ROWS) for c in range(SPAN) if pattern[r][c] != template[r][c]
    )


def _scored(pattern: list[list[int]]) -> list[tuple[int, int, str]]:
    """Distances with a fixed, public tie-break by declared symbol order."""

    return sorted(
        (_distance(pattern, TEMPLATES[name]), SYMBOLS.index(name), name) for name in SYMBOLS
    )


def _aligned_grid(scene: dict) -> list[list[int]]:
    """The 7x7 dark-cell pattern that the declared realignment reveals."""

    grid = [row[:] for row in TEMPLATES[scene["symbol"]]]
    for cell in scene["flips"]:
        r = int(cell[0])
        c = int(cell[1])
        grid[r][c] = 1 - grid[r][c]
    return grid


def _displayed(scene: dict) -> tuple[list[list[int]], list[int]]:
    """Anchor columns and the displayed ink map after the per-row slides."""

    pattern = _aligned_grid(scene)
    shifts = [int(value) % COLS for value in scene["shifts"]]
    ink = [[0] * COLS for _ in range(ROWS)]
    for r in range(ROWS):
        for c in range(SPAN):
            if pattern[r][c]:
                ink[r][(1 + c + shifts[r]) % COLS] = 1
    return ink, shifts


def _naive_readable(scene: dict) -> int:
    """Smallest mismatch a solver reaches WITHOUT the declared realignment."""

    ink, _ = _displayed(scene)
    best = ROWS * SPAN
    for g in range(COLS):
        naive = [[ink[r][(1 + c + g) % COLS] for c in range(SPAN)] for r in range(ROWS)]
        for name in SYMBOLS:
            best = min(best, _distance(naive, TEMPLATES[name]))
    return best


def _shift_ok(shifts: list[int]) -> bool:
    distinct = len(set(shifts))
    crowded = max(shifts.count(value) for value in set(shifts))
    return distinct >= 5 and crowded <= 2


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    for _ in range(400):
        symbol = SYMBOLS[rng.randrange(len(SYMBOLS))]
        budget = rng.choice([0, 0, 1, 1, 2, 2, 3, 4, 5, 6, 7, 8])
        cells = [[r, c] for r in range(ROWS) for c in range(SPAN)]
        if rng.random() < 0.5:
            rival = SYMBOLS[rng.randrange(len(SYMBOLS))]
            if rival == symbol:
                rival = SYMBOLS[(SYMBOLS.index(symbol) + 1) % len(SYMBOLS)]
            pool = [
                cell
                for cell in cells
                if TEMPLATES[symbol][cell[0]][cell[1]] != TEMPLATES[rival][cell[0]][cell[1]]
            ]
        else:
            pool = list(cells)
        rng.shuffle(pool)
        flips = sorted(pool[:budget])
        shifts = [rng.randrange(COLS) for _ in range(ROWS)]
        if not _shift_ok(shifts):
            continue
        scene = {
            "symbol": symbol,
            "flips": flips,
            "shifts": shifts,
            "cell": _CELLS[rng.randrange(len(_CELLS))],
            "jitter_x": rng.randint(-6, 6),
            "jitter_y": rng.randint(-5, 5),
            "bg": list(_BACKGROUNDS[rng.randrange(len(_BACKGROUNDS))]),
            "line": list(_LINES[rng.randrange(len(_LINES))]),
            "ink": list(_INKS[rng.randrange(len(_INKS))]),
            "anchor": list(_ANCHORS[rng.randrange(len(_ANCHORS))]),
        }
        if _naive_readable(scene) < 8:
            continue
        return scene
    raise RuntimeError("scene sampling failed to satisfy the declared constraints")


def _origin(scene: dict) -> tuple[int, int]:
    cell = int(scene["cell"])
    ox = (_WIDTH - COLS * cell) // 2 + int(scene["jitter_x"])
    oy = (_HEIGHT - ROWS * cell) // 2 + int(scene["jitter_y"])
    return ox, oy


def render(scene: dict) -> Image.Image:
    ink_map, shifts = _displayed(scene)
    cell = int(scene["cell"])
    ox, oy = _origin(scene)
    image = Image.new("RGB", (_WIDTH, _HEIGHT), tuple(scene["bg"]))
    draw = ImageDraw.Draw(image)
    ink_rgb = tuple(scene["ink"])
    anchor_rgb = tuple(scene["anchor"])
    line_rgb = tuple(scene["line"])
    for r in range(ROWS):
        for c in range(COLS):
            x0 = ox + c * cell
            y0 = oy + r * cell
            if c == shifts[r] % COLS:
                fill = anchor_rgb
            elif ink_map[r][c]:
                fill = ink_rgb
            else:
                fill = None
            if fill is not None:
                draw.rectangle([x0 + 2, y0 + 2, x0 + cell - 2, y0 + cell - 2], fill=fill)
    for c in range(COLS + 1):
        x = ox + c * cell
        draw.line([x, oy, x, oy + ROWS * cell], fill=line_rgb, width=1)
    for r in range(ROWS + 1):
        y = oy + r * cell
        draw.line([ox, y, ox + COLS * cell, y], fill=line_rgb, width=1)
    return image


def analytic_gold(scene: dict) -> str:
    return _scored(_aligned_grid(scene))[0][2]


def margin(scene: dict) -> float:
    ranking = _scored(_aligned_grid(scene))
    return float(ranking[1][0] - ranking[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical relabelings of the same scene."""

    modular = dict(scene)
    modular["shifts"] = [int(value) + COLS for value in scene["shifts"]]
    reordered = dict(scene)
    reordered["flips"] = list(reversed(list(scene["flips"])))
    return [("shift_modulus", modular), ("flip_listing_order", reordered)]
