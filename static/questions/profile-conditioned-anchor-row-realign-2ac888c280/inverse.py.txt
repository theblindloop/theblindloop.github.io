"""Image-only decision arm.

Receives a rendered PNG and nothing else. It locates the printed lattice, reads
each cell's role from colour alone, undoes the per-row cyclic slide declared by
the public question, and names the nearest of the five public symbols.
"""

from __future__ import annotations

from PIL import Image

ROWS = 7
COLS = 8
SPAN = COLS - 1
SYMBOLS = ("plus", "ex", "box", "tee", "diamond")
ABSTAIN = "abstain"


def _template(name: str) -> list[list[int]]:
    grid = [[0] * SPAN for _ in range(ROWS)]
    for r in range(ROWS):
        for c in range(SPAN):
            if name == "plus":
                on = r == 3 or c == 3
            elif name == "ex":
                on = r == c or r + c == 6
            elif name == "box":
                on = 1 <= r <= 5 and 1 <= c <= 5 and (r in (1, 5) or c in (1, 5))
            elif name == "tee":
                on = r == 0 or (c == 3 and r >= 1)
            else:
                on = abs(r - 3) + abs(c - 3) == 3
            grid[r][c] = 1 if on else 0
    return grid


_TEMPLATES = {name: _template(name) for name in SYMBOLS}


def _content_box(image: Image.Image) -> tuple[int, int, int, int] | None:
    pixels = image.load()
    width, height = image.size
    xs_min, ys_min, xs_max, ys_max = width, height, -1, -1
    for y in range(height):
        for x in range(width):
            r, g, b = pixels[x, y]
            if min(r, g, b) < 220:
                if x < xs_min:
                    xs_min = x
                if x > xs_max:
                    xs_max = x
                if y < ys_min:
                    ys_min = y
                if y > ys_max:
                    ys_max = y
    if xs_max < 0 or xs_max - xs_min < 16 or ys_max - ys_min < 16:
        return None
    return xs_min, ys_min, xs_max, ys_max


def _cell_role(image: Image.Image, box: tuple[float, float, float, float]) -> str:
    pixels = image.load()
    x0, y0, x1, y1 = box
    px0 = int(round(x0 + 0.28 * (x1 - x0)))
    px1 = int(round(x0 + 0.72 * (x1 - x0)))
    py0 = int(round(y0 + 0.28 * (y1 - y0)))
    py1 = int(round(y0 + 0.72 * (y1 - y0)))
    total_r = total_g = total_b = 0
    count = 0
    for y in range(py0, max(py0 + 1, py1)):
        for x in range(px0, max(px0 + 1, px1)):
            r, g, b = pixels[x, y]
            total_r += r
            total_g += g
            total_b += b
            count += 1
    if count == 0:
        return "empty"
    r = total_r / count
    g = total_g / count
    b = total_b / count
    if max(r, g, b) - min(r, g, b) > 55:
        return "anchor"
    if (r + g + b) / 3 < 130:
        return "ink"
    return "empty"


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    box = _content_box(rgb)
    if box is None:
        return ABSTAIN
    x0, y0, x1, y1 = box
    cell_w = (x1 - x0 + 1) / COLS
    cell_h = (y1 - y0 + 1) / ROWS
    roles = [
        [
            _cell_role(
                rgb,
                (x0 + c * cell_w, y0 + r * cell_h, x0 + (c + 1) * cell_w, y0 + (r + 1) * cell_h),
            )
            for c in range(COLS)
        ]
        for r in range(ROWS)
    ]
    anchors = []
    for r in range(ROWS):
        found = [c for c in range(COLS) if roles[r][c] == "anchor"]
        if len(found) != 1:
            return ABSTAIN
        anchors.append(found[0])
    pattern = [
        [1 if roles[r][(anchors[r] + 1 + c) % COLS] == "ink" else 0 for c in range(SPAN)]
        for r in range(ROWS)
    ]
    ranking = sorted(
        (
            sum(
                1
                for r in range(ROWS)
                for c in range(SPAN)
                if pattern[r][c] != _TEMPLATES[name][r][c]
            ),
            SYMBOLS.index(name),
            name,
        )
        for name in SYMBOLS
    )
    return ranking[0][2]
