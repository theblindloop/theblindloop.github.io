"""Deterministic raster world for a compact chamber face-degree question."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw


RENDERER_VERSION = "focal_face_degree-0.1.0"
CANVAS = (512, 512)
SUPERSAMPLE = 4
BACKGROUND = (255, 255, 255)
REMOTE_INK = (224, 231, 237)
NEUTRAL = (137, 151, 164)
PALETTE = {
    "red": (220, 58, 70),
    "green": (46, 164, 91),
    "blue": (54, 104, 214),
}
WALLS = ((29, 41, 55), (38, 49, 62), (31, 47, 59))


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced hub/leaf graph and visible nuisance variation."""

    value = int(seed)
    rng = random.Random(value)
    hub_index = value % 4
    hub_color = ("red", "green", "blue")[value % 3]
    remaining_colors = [name for name in ("red", "green", "blue") if name != hub_color]
    remaining_colors.append("neutral")
    rng.shuffle(remaining_colors)
    assignment = ["neutral"] * 4
    assignment[hub_index] = hub_color
    leaf_indices = [index for index in range(4) if index != hub_index]
    for index, color in zip(leaf_indices, remaining_colors, strict=True):
        assignment[index] = color

    tightness = "tight" if value % 5 == 0 else "clear"
    if tightness == "tight":
        node_half = 18.0
        stroke_width = 10.0
        marker_radius = 9.0
        link_width = 11.0
    else:
        node_half = 21.0
        stroke_width = 8.0
        marker_radius = 8.0
        link_width = 10.0

    return {
        "width": CANVAS[0],
        "height": CANVAS[1],
        "hub_index": hub_index,
        "color_assignment": assignment,
        "orientation": rng.randrange(4),
        "origin_x": round(rng.uniform(185.0, 327.0), 3),
        "origin_y": round(rng.uniform(185.0, 327.0), 3),
        "glyph_scale": round(rng.choice((0.92, 1.0, 1.06)), 3),
        "node_half": node_half,
        "stroke_width": stroke_width,
        "marker_radius": marker_radius,
        "link_width": link_width,
        "wall_color": list(WALLS[rng.randrange(len(WALLS))]),
        "background": list(BACKGROUND),
        "neutral_color": list(NEUTRAL),
        "tightness": tightness,
        "decor_seed": rng.randrange(1, 2**31 - 1),
    }


def _rotate(point: tuple[float, float], orientation: int) -> tuple[float, float]:
    x, y = point
    turn = int(orientation) % 4
    if turn == 0:
        return x, y
    if turn == 1:
        return -y, x
    if turn == 2:
        return -x, -y
    return y, -x


def _template_nodes(hub_index: int, spacing: float) -> list[tuple[float, float]]:
    """Return four node centers; exactly one is incident to all three links."""

    s = float(spacing)
    templates = {
        0: [(0.0, 0.0), (-s, 0.0), (s, 0.0), (0.0, -s)],
        1: [(s / 2.0, 0.0), (-s / 2.0, 0.0), (-s / 2.0, -s), (-s / 2.0, s)],
        2: [(0.0, s / 2.0), (-s, -s / 2.0), (0.0, -s / 2.0), (s, -s / 2.0)],
        3: [(-s / 2.0, 0.0), (s / 2.0, -s), (s / 2.0, s), (s / 2.0, 0.0)],
    }
    return templates[int(hub_index) % 4]


def _node_centers(scene: dict[str, Any]) -> list[tuple[float, float]]:
    scale = float(scene["glyph_scale"])
    spacing = 50.0 * scale
    centers = []
    for point in _template_nodes(int(scene["hub_index"]), spacing):
        rx, ry = _rotate(point, int(scene["orientation"]))
        centers.append((float(scene["origin_x"]) + rx, float(scene["origin_y"]) + ry))
    return centers


def _scaled_point(point: tuple[float, float], scale: int) -> tuple[int, int]:
    return int(round(point[0] * scale)), int(round(point[1] * scale))


def _draw_remote_marks(draw: ImageDraw.ImageDraw, scene: dict[str, Any], scale: int) -> None:
    rng = random.Random(int(scene["decor_seed"]))
    for base_x, base_y in ((32, 34), (480, 34), (34, 478), (478, 478)):
        x = base_x + rng.randint(-8, 8)
        y = base_y + rng.randint(-8, 8)
        radius = (2 + rng.randrange(3)) * scale
        color = tuple(max(0, min(255, value + rng.randint(-3, 3))) for value in REMOTE_INK)
        draw.ellipse((x * scale - radius, y * scale - radius, x * scale + radius, y * scale + radius), fill=color)


def render(scene: dict[str, Any]) -> Image.Image:
    """Render one connected star of bounded white chambers and wall links."""

    width = int(scene["width"])
    height = int(scene["height"])
    scale = SUPERSAMPLE
    image = Image.new("RGB", (width * scale, height * scale), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    _draw_remote_marks(draw, scene, scale)

    centers = _node_centers(scene)
    hub = int(scene["hub_index"])
    wall = tuple(int(value) for value in scene["wall_color"])
    link_width = max(1, int(round(float(scene["link_width"]) * scale)))
    for index, center in enumerate(centers):
        if index == hub:
            continue
        draw.line(
            [_scaled_point(centers[hub], scale), _scaled_point(center, scale)],
            fill=wall,
            width=link_width,
        )

    half = float(scene["node_half"]) * float(scene["glyph_scale"])
    stroke = max(1, int(round(float(scene["stroke_width"]) * scale)))
    for center in centers:
        cx, cy = _scaled_point(center, scale)
        box = (cx - int(round(half * scale)), cy - int(round(half * scale)), cx + int(round(half * scale)), cy + int(round(half * scale)))
        draw.rounded_rectangle(
            box,
            radius=max(2, int(round(6.0 * scale))),
            fill=BACKGROUND,
            outline=wall,
            width=stroke,
        )

    marker_radius = int(round(float(scene["marker_radius"]) * scale))
    neutral = tuple(int(value) for value in scene["neutral_color"])
    for center, color_name in zip(centers, scene["color_assignment"], strict=True):
        cx, cy = _scaled_point(center, scale)
        color = neutral if color_name == "neutral" else PALETTE[str(color_name)]
        draw.ellipse((cx - marker_radius, cy - marker_radius, cx + marker_radius, cy + marker_radius), fill=color)

    return image.resize((width, height), Image.Resampling.LANCZOS).convert("RGB")


def _clearance(scene: dict[str, Any]) -> float:
    half = float(scene["node_half"]) * float(scene["glyph_scale"])
    return half - float(scene["stroke_width"]) / 2.0 - float(scene["marker_radius"])


def analytic_gold(scene: dict[str, Any]) -> str:
    """Select the color assigned to the unique degree-three chamber."""

    return str(scene["color_assignment"][int(scene["hub_index"])])


def margin(scene: dict[str, Any]) -> float:
    """Top-two degree gap; tight scenes have a smaller raster clearance margin."""

    return 1.0 if is_quarantined(scene) else 2.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    return bool(scene.get("tightness") == "tight" or _clearance(scene) < 7.0)


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No intentionally pixel-identical alternate latent transforms are declared."""

    del scene
    return []
