"""Shared circuit pair: which two coloured rooms share one round trip.

The scene is a 6 by 6 lattice of square rooms.  Every room boundary carries a
thick wall; a boundary that is a *doorway* is drawn with a central gap instead
of a solid run.  The doorway graph is a spanning tree over all thirty-six rooms
plus exactly three extra doorways whose fundamental cycles are pairwise
vertex-disjoint, so the free space is one connected region holding three
separate corridor loops.  Four rooms carry a flat coloured floor and *every one
of them stands on a loop*, so being on a round trip discriminates nothing.  Two
of the four sit on the same loop and the other two sit on loops of their own,
so the label is the unique pair of coloured rooms that one closed walk can
visit without ever reusing a doorway.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = 1000
CELLS = 6
CELL_PX = 150
WALL_PX = 12

BACKGROUND = (247, 246, 242)
WALL = (28, 30, 34)

COLOURS: dict[str, tuple[int, int, int]] = {
    "orange": (235, 130, 25),
    "green": (26, 150, 60),
    "blue": (32, 82, 214),
    "pink": (240, 105, 185),
}
COLOUR_ORDER = ("orange", "green", "blue", "pink")

GAP_MIN = 48
GAP_MAX = 78
QUARANTINE_GAP = 56
MIN_LOOP = 6
MAX_LOOP = 14
LOOPS = 3


# --------------------------------------------------------------------------- #
# lattice helpers
# --------------------------------------------------------------------------- #
def _neighbours(cell: tuple[int, int]) -> list[tuple[int, int]]:
    row, col = cell
    out = []
    for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
        nr, nc = row + dr, col + dc
        if 0 <= nr < CELLS and 0 <= nc < CELLS:
            out.append((nr, nc))
    return out


def _edge_key(a: tuple[int, int], b: tuple[int, int]) -> tuple[int, int, int, int]:
    first, second = sorted((a, b))
    return (first[0], first[1], second[0], second[1])


def _graph(doorways: set[tuple[int, int, int, int]]) -> dict:
    adjacency: dict[tuple[int, int], list[tuple[int, int]]] = {}
    for row in range(CELLS):
        for col in range(CELLS):
            cell = (row, col)
            adjacency[cell] = [
                n for n in _neighbours(cell) if _edge_key(cell, n) in doorways
            ]
    return adjacency


def _reach(adjacency: dict, source, blocked=None) -> set:
    blocked = blocked or set()
    seen = {source}
    stack = [source]
    while stack:
        current = stack.pop()
        for n in adjacency[current]:
            if n in seen or _edge_key(current, n) in blocked:
                continue
            seen.add(n)
            stack.append(n)
    return seen


def loop_components(doorways: set) -> list[set]:
    """Components of the free space once every bridge doorway is deleted.

    Two rooms admit a closed walk through both that repeats no doorway exactly
    when they land in the same component here.
    """
    adjacency = _graph(doorways)
    bridges = set()
    for edge in doorways:
        a = (edge[0], edge[1])
        if len(_reach(adjacency, a, blocked={edge})) != CELLS * CELLS:
            bridges.add(edge)
    kept = doorways - bridges
    core = _graph(kept)
    seen: set = set()
    out: list[set] = []
    for cell in sorted(core):
        if cell in seen:
            continue
        block = _reach(core, cell)
        seen |= block
        if len(block) > 1:
            out.append(block)
    return out


def _spanning_tree(rng: random.Random) -> list[tuple[int, int, int, int]]:
    cells = [(r, c) for r in range(CELLS) for c in range(CELLS)]
    start = cells[rng.randrange(len(cells))]
    visited = {start}
    frontier = [start]
    edges: list[tuple[int, int, int, int]] = []
    while frontier:
        index = rng.randrange(len(frontier)) if rng.random() < 0.35 else len(frontier) - 1
        current = frontier[index]
        options = [n for n in _neighbours(current) if n not in visited]
        if not options:
            frontier.pop(index)
            continue
        nxt = options[rng.randrange(len(options))]
        visited.add(nxt)
        edges.append(_edge_key(current, nxt))
        frontier.append(nxt)
    return sorted(set(edges))


def _tree_path(parent: dict, depth: dict, a, b) -> list:
    left, right = a, b
    path_left, path_right = [a], [b]
    while depth[left] > depth[right]:
        left = parent[left]
        path_left.append(left)
    while depth[right] > depth[left]:
        right = parent[right]
        path_right.append(right)
    while left != right:
        left = parent[left]
        right = parent[right]
        path_left.append(left)
        path_right.append(right)
    return path_left + list(reversed(path_right[:-1]))


def _manhattan(a, b) -> int:
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


# --------------------------------------------------------------------------- #
# sampling
# --------------------------------------------------------------------------- #
def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) * 7919 + 13)
    all_cells = [(r, c) for r in range(CELLS) for c in range(CELLS)]
    for _ in range(8000):
        tree = _spanning_tree(rng)
        tree_set = set(tree)
        tree_adj = _graph(tree_set)
        root = (0, 0)
        parent: dict = {root: root}
        depth = {root: 0}
        order = [root]
        head = 0
        while head < len(order):
            current = order[head]
            head += 1
            for n in tree_adj[current]:
                if n not in depth:
                    parent[n] = current
                    depth[n] = depth[current] + 1
                    order.append(n)
        solid = [
            _edge_key(a, b)
            for a in all_cells
            for b in _neighbours(a)
            if a < b and _edge_key(a, b) not in tree_set
        ]
        options = []
        for edge in solid:
            cycle = set(_tree_path(parent, depth, (edge[0], edge[1]), (edge[2], edge[3])))
            if MIN_LOOP <= len(cycle) <= MAX_LOOP:
                options.append((edge, cycle))
        rng.shuffle(options)
        chosen: list = []
        used: set = set()
        for edge, cycle in options:
            if used & cycle:
                continue
            chosen.append((edge, cycle))
            used |= cycle
            if len(chosen) == LOOPS:
                break
        if len(chosen) != LOOPS:
            continue
        doorways = sorted(tree_set | {edge for edge, _ in chosen})
        door_set = set(doorways)
        adjacency = _graph(door_set)
        blocks = loop_components(door_set)
        if len(blocks) != LOOPS:
            continue
        if sorted(len(b) for b in blocks) != sorted(len(c) for _, c in chosen):
            continue

        # the loop that carries two coloured rooms must not be the largest one
        indices = list(range(LOOPS))
        rng.shuffle(indices)
        placed = None
        for host in indices:
            others = [i for i in indices if i != host]
            if len(blocks[host]) > max(len(blocks[i]) for i in others):
                continue
            pool = {
                i: [c for c in sorted(blocks[i]) if len(adjacency[c]) == 2]
                for i in indices
            }
            if len(pool[host]) < 2 or any(not pool[i] for i in others):
                continue
            pairs = [
                (a, b)
                for k, a in enumerate(pool[host])
                for b in pool[host][k + 1:]
                if _manhattan(a, b) >= 3
            ]
            if not pairs:
                continue
            rng.shuffle(pairs)
            singles = [pool[i][rng.randrange(len(pool[i]))] for i in others]
            for pair in pairs:
                rooms = [pair[0], pair[1], singles[0], singles[1]]
                if len(set(rooms)) != 4:
                    continue
                answer_span = _manhattan(pair[0], pair[1])
                decoy_span = min(
                    _manhattan(a, b)
                    for k, a in enumerate(rooms)
                    for b in rooms[k + 1:]
                    if {a, b} != {pair[0], pair[1]}
                )
                # some non-answer pair must sit strictly closer than the answer
                if decoy_span >= answer_span or decoy_span < 2:
                    continue
                placed = (rooms, pair)
                break
            if placed:
                break
        if not placed:
            continue
        rooms, _pair = placed
        colours = list(COLOUR_ORDER)
        rng.shuffle(colours)
        assignment = dict(zip(colours, rooms))
        gap = rng.randrange(GAP_MIN, GAP_MAX + 1, 2)
        scene = {
            "canvas": CANVAS,
            "cells": CELLS,
            "cell_px": CELL_PX,
            "wall_px": WALL_PX,
            "gap_px": int(gap),
            "loop_sizes": sorted(len(b) for b in blocks),
            "doorways": [list(edge) for edge in doorways],
            "rooms": {
                colour: [int(assignment[colour][0]), int(assignment[colour][1])]
                for colour in COLOUR_ORDER
            },
            "draw_order": [COLOUR_ORDER.index(c) for c in colours],
        }
        try:
            analytic_gold(scene)
        except ValueError:
            continue
        return scene
    raise RuntimeError(f"failed to sample a scene for seed {seed}")


# --------------------------------------------------------------------------- #
# rendering
# --------------------------------------------------------------------------- #
def _origin(scene: dict) -> int:
    return (int(scene["canvas"]) - int(scene["cells"]) * int(scene["cell_px"])) // 2


def render(scene: dict) -> Image.Image:
    canvas = int(scene["canvas"])
    cells = int(scene["cells"])
    step = int(scene["cell_px"])
    half = int(scene["wall_px"]) // 2
    gap = int(scene["gap_px"])
    off = _origin(scene)
    doorways = {tuple(int(v) for v in edge) for edge in scene["doorways"]}

    image = Image.new("RGB", (canvas, canvas), BACKGROUND)
    draw = ImageDraw.Draw(image)

    for index in scene["draw_order"]:
        colour = COLOUR_ORDER[int(index)]
        row, col = (int(v) for v in scene["rooms"][colour])
        x0 = off + col * step + half
        y0 = off + row * step + half
        draw.rectangle(
            [x0, y0, x0 + step - 2 * half - 1, y0 + step - 2 * half - 1],
            fill=COLOURS[colour],
        )

    def horizontal(row: int, col: int, open_gap: bool) -> None:
        y = off + row * step
        x0 = off + col * step
        x1 = x0 + step
        if open_gap:
            mid = (x0 + x1) // 2
            draw.rectangle([x0, y - half, mid - gap // 2, y + half - 1], fill=WALL)
            draw.rectangle([mid + gap // 2, y - half, x1 - 1, y + half - 1], fill=WALL)
        else:
            draw.rectangle([x0, y - half, x1 - 1, y + half - 1], fill=WALL)

    def vertical(row: int, col: int, open_gap: bool) -> None:
        x = off + col * step
        y0 = off + row * step
        y1 = y0 + step
        if open_gap:
            mid = (y0 + y1) // 2
            draw.rectangle([x - half, y0, x + half - 1, mid - gap // 2], fill=WALL)
            draw.rectangle([x - half, mid + gap // 2, x + half - 1, y1 - 1], fill=WALL)
        else:
            draw.rectangle([x - half, y0, x + half - 1, y1 - 1], fill=WALL)

    for row in range(cells):
        for col in range(cells + 1):
            if col == 0 or col == cells:
                vertical(row, col, False)
            else:
                vertical(row, col, _edge_key((row, col - 1), (row, col)) in doorways)
    for col in range(cells):
        for row in range(cells + 1):
            if row == 0 or row == cells:
                horizontal(row, col, False)
            else:
                horizontal(row, col, _edge_key((row - 1, col), (row, col)) in doorways)
    for row in range(cells + 1):
        for col in range(cells + 1):
            x = off + col * step
            y = off + row * step
            draw.rectangle([x - half, y - half, x + half - 1, y + half - 1], fill=WALL)
    return image


# --------------------------------------------------------------------------- #
# analytic gold
# --------------------------------------------------------------------------- #
def pair_name(first: str, second: str) -> str:
    ordered = sorted((first, second), key=COLOUR_ORDER.index)
    return f"{ordered[0]}+{ordered[1]}"


def analytic_gold(scene: dict) -> str:
    door_set = {tuple(int(v) for v in edge) for edge in scene["doorways"]}
    adjacency = _graph(door_set)
    if len(_reach(adjacency, (0, 0))) != CELLS * CELLS:
        raise ValueError("the free space is not one connected region")
    blocks = loop_components(door_set)
    if len(blocks) != LOOPS or min(len(b) for b in blocks) < MIN_LOOP:
        raise ValueError("the scene does not carry three readable loops")
    rooms = {
        colour: (int(scene["rooms"][colour][0]), int(scene["rooms"][colour][1]))
        for colour in COLOUR_ORDER
    }
    if len(set(rooms.values())) != 4:
        raise ValueError("two coloured floors share a room")
    home: dict[str, int] = {}
    for colour, room in rooms.items():
        on = [i for i, block in enumerate(blocks) if room in block]
        if len(on) != 1:
            raise ValueError(f"the {colour} room does not stand on exactly one loop")
        home[colour] = on[0]
    shared = [
        (a, b)
        for i, a in enumerate(COLOUR_ORDER)
        for b in COLOUR_ORDER[i + 1:]
        if home[a] == home[b]
    ]
    if len(shared) != 1:
        raise ValueError(f"scene does not have exactly one shared pair: {shared}")
    return pair_name(*shared[0])


def margin(scene: dict) -> float:
    gap = int(scene["gap_px"])
    stub = (int(scene["cell_px"]) - gap) // 2
    return float(min(gap, stub * 2))


def is_quarantined(scene: dict) -> bool:
    return int(scene["gap_px"]) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    reversed_doors = dict(scene)
    reversed_doors["doorways"] = list(reversed(scene["doorways"]))
    rotated = dict(scene)
    rotated["doorways"] = list(scene["doorways"][1:]) + [scene["doorways"][0]]
    reordered = dict(scene)
    reordered["draw_order"] = list(reversed(scene["draw_order"]))
    return [
        ("reverse_doorway_serialisation", reversed_doors),
        ("rotate_doorway_serialisation", rotated),
        ("reverse_floor_draw_order", reordered),
    ]
