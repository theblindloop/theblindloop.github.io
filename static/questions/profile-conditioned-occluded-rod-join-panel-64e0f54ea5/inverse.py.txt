"""Pixel-only decision arm for the occluded-rod join world.

Receives a rendered RGB image and nothing else. It locates the two dark gray
divider lines by column and row coverage, splits the raster into four
quadrants, keeps only chromatic (coloured) pixels as rod ink, finds the
occluding gap that separates each rod into a left and a right piece, measures
each piece's centre line column by column, extrapolates the left piece's fitted
line across the gap, and returns the quadrant whose right piece departs least
from that continuation.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

CHROMA = 55.0
ACHROMATIC_TOLERANCE = 20.0
DIVIDER_LUMINANCE = 130.0
DIVIDER_COVERAGE = 0.75
MIN_GAP = 12
MAX_RIVAL_GAP = 6
MIN_PIECE_COLUMNS = 26
TRIM = 4
MATCH_TOLERANCE = 2.5
DECISION_MARGIN = 4.0
AMBIGUOUS = "ambiguous"
QUADRANTS = ("top_left", "top_right", "bottom_left", "bottom_right")


def _masks(image: Image.Image):
    arr = np.asarray(image.convert("RGB"), dtype=np.float64)
    high = arr.max(axis=2)
    low = arr.min(axis=2)
    chroma = high - low
    colored = chroma > CHROMA
    dark_gray = (chroma <= ACHROMATIC_TOLERANCE) & (high < DIVIDER_LUMINANCE)
    return colored, dark_gray


def _divider(counts: np.ndarray, extent: int):
    best = int(np.argmax(counts))
    if counts[best] < DIVIDER_COVERAGE * extent:
        return None
    return best


def _piece_profile(mask: np.ndarray, columns: range):
    """Mean row index of the rod ink in each column of a piece."""
    xs = []
    ys = []
    for x in columns:
        rows = np.flatnonzero(mask[:, x])
        if rows.size == 0:
            return None
        xs.append(float(x))
        ys.append(float(rows.mean()))
    return np.asarray(xs), np.asarray(ys)


def _quadrant_departure(mask: np.ndarray):
    columns = np.flatnonzero(mask.any(axis=0))
    if columns.size < 2 * MIN_PIECE_COLUMNS:
        return None
    breaks = np.flatnonzero(np.diff(columns) > 1)
    if breaks.size == 0:
        return None
    gaps = [(int(columns[i + 1] - columns[i]), int(i)) for i in breaks]
    gaps.sort(reverse=True)
    if gaps[0][0] < MIN_GAP:
        return None
    if len(gaps) > 1 and gaps[1][0] > MAX_RIVAL_GAP:
        return None
    split = gaps[0][1]
    left = range(int(columns[0]) + TRIM, int(columns[split]) - TRIM + 1)
    right = range(int(columns[split + 1]) + TRIM, int(columns[-1]) - TRIM + 1)
    if len(left) < MIN_PIECE_COLUMNS or len(right) < MIN_PIECE_COLUMNS:
        return None
    left_profile = _piece_profile(mask, left)
    right_profile = _piece_profile(mask, right)
    if left_profile is None or right_profile is None:
        return None
    slope, intercept = np.polyfit(left_profile[0], left_profile[1], 1)
    predicted = slope * right_profile[0] + intercept
    return float(np.abs(right_profile[1] - predicted).mean())


def decision_from_image(image: Image.Image) -> str:
    colored, dark_gray = _masks(image)
    height, width = colored.shape
    split_x = _divider(dark_gray.sum(axis=0), height)
    split_y = _divider(dark_gray.sum(axis=1), width)
    if split_x is None or split_y is None:
        return AMBIGUOUS
    scored = []
    for index, name in enumerate(QUADRANTS):
        row, col = divmod(index, 2)
        y0, y1 = (0, split_y) if row == 0 else (split_y + 1, height)
        x0, x1 = (0, split_x) if col == 0 else (split_x + 1, width)
        window = np.zeros_like(colored)
        window[y0:y1, x0:x1] = colored[y0:y1, x0:x1]
        value = _quadrant_departure(window)
        if value is None:
            return AMBIGUOUS
        scored.append((value, name))
    scored.sort()
    if scored[0][0] > MATCH_TOLERANCE:
        return AMBIGUOUS
    if scored[1][0] - scored[0][0] < DECISION_MARGIN:
        return AMBIGUOUS
    return scored[0][1]
