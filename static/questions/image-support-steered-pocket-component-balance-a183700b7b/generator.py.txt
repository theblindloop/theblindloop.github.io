"""Deterministic renderer and analytic arm for Pocket–Component Balance."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

CANVAS = 512
CELL = 64
FRAME_SIZE = (40, 48)
ONE_HOLE_SIZE = (20, 20)
TWO_HOLE_SIZE = (20, 10)
POCKET_OFFSET = 8
ONE_TILE_SIZE = (20, 20)
TWO_TILE_SIZE = (20, 10)
TILE_OFFSET = 9
MIN_CLEARANCE = 6.0


def _rect(draw: ImageDraw.ImageDraw, center: tuple[int, int], size: tuple[int, int], fill: str) -> None:
    """Draw a rectangle with exactly width*height pixels."""

    cx, cy = center
    width, height = size
    x0 = cx - width // 2
    y0 = cy - height // 2
    draw.rectangle((x0, y0, x0 + width - 1, y0 + height - 1), fill=fill)


def _sample_decorations(
    rng: random.Random,
    frame_center: tuple[int, int],
    tile_center: tuple[int, int],
) -> list[list[int]]:
    decorations: list[list[int]] = []
    attempts = 0
    while len(decorations) < 7 and attempts < 400:
        attempts += 1
        x = rng.randrange(28, CANVAS - 28)
        y = rng.randrange(28, CANVAS - 28)
        if min(math.dist((x, y), frame_center), math.dist((x, y), tile_center)) < 105:
            continue
        if any(math.dist((x, y), (old_x, old_y)) < 25 for old_x, old_y, _ in decorations):
            continue
        decorations.append([x, y, rng.randrange(2)])
    return decorations


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x51A7C0DE)
    combinations = ((1, 1), (1, 2), (2, 1), (2, 2))
    frame_holes, tile_count = combinations[int(seed) % len(combinations)]
    pair_axis = "horizontal" if rng.randrange(2) == 0 else "vertical"
    if pair_axis == "horizontal":
        row = rng.randrange(1, 7)
        first = rng.randrange(1, 5)
        centers = [(first * CELL + CELL // 2, row * CELL + CELL // 2),
                   ((first + 2) * CELL + CELL // 2, row * CELL + CELL // 2)]
    else:
        column = rng.randrange(1, 7)
        first = rng.randrange(1, 5)
        centers = [(column * CELL + CELL // 2, first * CELL + CELL // 2),
                   (column * CELL + CELL // 2, (first + 2) * CELL + CELL // 2)]
    if rng.randrange(2):
        centers.reverse()
    frame_center, tile_center = centers
    ink = rng.choice(("#173f5f", "#214b67", "#284f6a"))
    decoration_color = rng.choice(("#d8dee2", "#d2dadd", "#dde1e4"))
    return {
        "width": CANVAS,
        "height": CANVAS,
        "background": "#ffffff",
        "ink": ink,
        "decoration_color": decoration_color,
        "frame_holes": frame_holes,
        "tile_count": tile_count,
        "frame_center": list(frame_center),
        "tile_center": list(tile_center),
        "pair_axis": pair_axis,
        "decorations": _sample_decorations(rng, frame_center, tile_center),
    }


def _draw_frame(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    holes: int,
    axis: str,
    ink: str,
    background: str,
) -> None:
    _rect(draw, center, FRAME_SIZE, ink)
    cx, cy = center
    if holes == 1:
        _rect(draw, center, ONE_HOLE_SIZE, background)
        return
    if holes != 2:
        raise ValueError(f"unsupported frame_holes={holes!r}")
    if axis == "horizontal":
        pocket_centers = ((cx, cy - POCKET_OFFSET), (cx, cy + POCKET_OFFSET))
        pocket_size = TWO_HOLE_SIZE
    else:
        pocket_centers = ((cx - POCKET_OFFSET, cy), (cx + POCKET_OFFSET, cy))
        pocket_size = (TWO_HOLE_SIZE[1], TWO_HOLE_SIZE[0])
    for pocket_center in pocket_centers:
        _rect(draw, pocket_center, pocket_size, background)


def _draw_tiles(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    count: int,
    axis: str,
    ink: str,
) -> None:
    cx, cy = center
    if count == 1:
        _rect(draw, center, ONE_TILE_SIZE, ink)
        return
    if count != 2:
        raise ValueError(f"unsupported tile_count={count!r}")
    if axis == "horizontal":
        tile_centers = ((cx, cy - TILE_OFFSET), (cx, cy + TILE_OFFSET))
        tile_size = TWO_TILE_SIZE
    else:
        tile_centers = ((cx - TILE_OFFSET, cy), (cx + TILE_OFFSET, cy))
        tile_size = (TWO_TILE_SIZE[1], TWO_TILE_SIZE[0])
    for tile_center in tile_centers:
        _rect(draw, tile_center, tile_size, ink)


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    background = str(scene["background"])
    ink = str(scene["ink"])
    decoration_color = str(scene["decoration_color"])
    frame_holes = int(scene["frame_holes"])
    tile_count = int(scene["tile_count"])
    frame_center = tuple(int(value) for value in scene["frame_center"])
    tile_center = tuple(int(value) for value in scene["tile_center"])
    pair_axis = str(scene["pair_axis"])
    decorations = list(scene["decorations"])

    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)
    for x, y, kind in decorations:
        point = (int(x), int(y))
        if int(kind) == 0:
            cx, cy = point
            draw.rectangle((cx - 4, cy - 4, cx + 3, cy + 3), outline=decoration_color, width=2)
        else:
            cx, cy = point
            draw.rectangle((cx - 1, cy - 5, cx, cy + 4), fill=decoration_color)
            draw.rectangle((cx - 5, cy - 1, cx + 4, cy), fill=decoration_color)
    _draw_frame(draw, frame_center, frame_holes, pair_axis, ink, background)
    _draw_tiles(draw, tile_center, tile_count, pair_axis, ink)
    return image


def analytic_gold(scene: dict) -> str:
    frame_holes = int(scene["frame_holes"])
    tile_count = int(scene["tile_count"])
    return "yes" if frame_holes == tile_count else "no"


def margin(scene: dict) -> float:
    """Minimum visible clearance before either topological count can change."""

    frame_holes = int(scene["frame_holes"])
    tile_count = int(scene["tile_count"])
    frame_center = tuple(int(value) for value in scene["frame_center"])
    tile_center = tuple(int(value) for value in scene["tile_center"])
    width = int(scene["width"])
    height = int(scene["height"])
    pair_axis = str(scene["pair_axis"])
    wall_or_separator = 6.0 if frame_holes == 2 else 10.0
    inter_tile_gap = 8.0 if tile_count == 2 else 10.0
    pair_clearance = math.dist(frame_center, tile_center) - 48.0
    border_clearance = min(
        frame_center[0] - 20,
        width - (frame_center[0] + 20),
        frame_center[1] - 24,
        height - (frame_center[1] + 24),
        tile_center[0] - 20,
        width - (tile_center[0] + 20),
        tile_center[1] - 24,
        height - (tile_center[1] + 24),
    )
    axis_check = 12.0 if pair_axis in {"horizontal", "vertical"} else 0.0
    return float(min(wall_or_separator, inter_tile_gap, pair_clearance, border_clearance, axis_check))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < MIN_CLEARANCE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["decorations"] = list(reversed(twin["decorations"]))
    return [("reverse_disjoint_decoration_order", twin)]
