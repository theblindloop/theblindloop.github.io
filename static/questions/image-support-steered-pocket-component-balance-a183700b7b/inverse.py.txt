"""Independent image-only topology oracle.

It segments dark navy ink, labels foreground components, identifies the unique
large hole-bearing frame, counts its bounded white complement components, and
compares that count with the remaining detached foreground components.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    pass


def _navy_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[..., 0], array[..., 1], array[..., 2]
    return (red < 105) & (green < 135) & (blue < 165) & ((blue - red) > 18)


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    remaining = {tuple(int(value) for value in point) for point in np.argwhere(mask)}
    components: list[list[tuple[int, int]]] = []
    while remaining:
        start = remaining.pop()
        queue = [start]
        component = [start]
        while queue:
            y, x = queue.pop()
            for neighbor in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                ny, nx = neighbor
                if 0 <= ny < height and 0 <= nx < width and neighbor in remaining:
                    remaining.remove(neighbor)
                    queue.append(neighbor)
                    component.append(neighbor)
        components.append(component)
    return components


def _hole_count(component: list[tuple[int, int]]) -> int:
    ys = [point[0] for point in component]
    xs = [point[1] for point in component]
    y0, y1 = min(ys) - 1, max(ys) + 2
    x0, x1 = min(xs) - 1, max(xs) + 2
    if y0 < 0 or x0 < 0:
        raise OracleAbstention("frame touches image border")
    local = np.zeros((y1 - y0, x1 - x0), dtype=bool)
    for y, x in component:
        local[y - y0, x - x0] = True
    complement_components = _components(~local)
    holes = 0
    height, width = local.shape
    for region in complement_components:
        touches_border = any(y in {0, height - 1} or x in {0, width - 1} for y, x in region)
        if not touches_border and len(region) >= 40:
            holes += 1
    return holes


def decision_from_image(image: Image.Image) -> str:
    mask = _navy_mask(image)
    foreground = [component for component in _components(mask) if len(component) >= 80]
    if len(foreground) not in {2, 3}:
        raise OracleAbstention(f"expected two or three navy components, found {len(foreground)}")
    foreground.sort(key=len, reverse=True)
    frame = foreground[0]
    if len(frame) < 900 or len(foreground[1]) > 600:
        raise OracleAbstention("no unique large frame component")
    pockets = _hole_count(frame)
    tiles = len(foreground) - 1
    if pockets not in {1, 2} or tiles not in {1, 2}:
        raise OracleAbstention(f"unsupported topology pockets={pockets}, tiles={tiles}")
    return "yes" if pockets == tiles else "no"
