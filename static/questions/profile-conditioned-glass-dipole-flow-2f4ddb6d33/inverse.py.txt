"""Image-only inverse arm: recover the global dipole flow type from pixels.

Receives one rendered RGB image and nothing else. It segments the red centre
mark and the dark dots, pairs each dot with its mutual nearest neighbour, and
classifies the mean undirected angle between each pair's own axis and the
direction from the centre mark to that pair.
"""

from __future__ import annotations

import math
from collections import deque

from PIL import Image

LOW_BOUNDARY_DEG = 22.5
HIGH_BOUNDARY_DEG = 67.5
MIN_BLOB_AREA = 8
ABSTAIN = "abstain"


def _components(mask: list[list[bool]], width: int, height: int) -> list[tuple[float, float, int]]:
    seen = [[False] * width for _ in range(height)]
    out: list[tuple[float, float, int]] = []
    for y0 in range(height):
        row = mask[y0]
        for x0 in range(width):
            if not row[x0] or seen[y0][x0]:
                continue
            queue = deque([(x0, y0)])
            seen[y0][x0] = True
            sx = sy = 0.0
            area = 0
            while queue:
                x, y = queue.popleft()
                sx += x
                sy += y
                area += 1
                for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nx, ny = x + dx, y + dy
                    if 0 <= nx < width and 0 <= ny < height and mask[ny][nx] and not seen[ny][nx]:
                        seen[ny][nx] = True
                        queue.append((nx, ny))
            if area >= MIN_BLOB_AREA:
                out.append((sx / area, sy / area, area))
    return out


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    width, height = rgb.size
    px = rgb.load()

    dark = [[False] * width for _ in range(height)]
    red = [[False] * width for _ in range(height)]
    for y in range(height):
        drow = dark[y]
        rrow = red[y]
        for x in range(width):
            r, g, b = px[x, y]
            if max(r, g, b) < 120:
                drow[x] = True
            elif r > 140 and r - g > 60 and r - b > 60:
                rrow[x] = True

    marks = _components(red, width, height)
    if len(marks) != 1:
        return ABSTAIN
    cx, cy, _ = marks[0]

    dots = _components(dark, width, height)
    if len(dots) < 8 or len(dots) % 2 != 0:
        return ABSTAIN

    points = [(x, y) for x, y, _ in dots]
    nearest: list[int] = []
    for i, (xi, yi) in enumerate(points):
        best = -1
        best_d = float("inf")
        for j, (xj, yj) in enumerate(points):
            if i == j:
                continue
            d = (xi - xj) ** 2 + (yi - yj) ** 2
            if d < best_d:
                best_d = d
                best = j
        nearest.append(best)

    angles: list[float] = []
    used = [False] * len(points)
    for i, j in enumerate(nearest):
        if used[i] or nearest[j] != i or j <= i:
            continue
        used[i] = used[j] = True
        xi, yi = points[i]
        xj, yj = points[j]
        vx, vy = xj - xi, yj - yi
        mx, my = 0.5 * (xi + xj), 0.5 * (yi + yj)
        rx, ry = mx - cx, my - cy
        if (vx == 0 and vy == 0) or (rx == 0 and ry == 0):
            return ABSTAIN
        dot = abs(vx * rx + vy * ry)
        cross = abs(vx * ry - vy * rx)
        angles.append(math.degrees(math.atan2(cross, dot)))

    if any(not flag for flag in used) or len(angles) * 2 != len(points):
        return ABSTAIN

    mean_deg = sum(angles) / len(angles)
    if mean_deg < LOW_BOUNDARY_DEG:
        return "radial"
    if mean_deg > HIGH_BOUNDARY_DEG:
        return "circular"
    return "spiral"
