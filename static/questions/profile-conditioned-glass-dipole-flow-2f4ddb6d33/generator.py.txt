"""Glass-pattern dipole field: sampling, rendering and analytic gold.

Every scene is a field of identical dark dots arranged in short two-dot pairs.
Each pair's own orientation is set by rotating the direction that points from a
drawn red cross to that pair by one shared angle, so the whole field organises
into a radial, spiral or circular flow. The decision is that global flow type.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BG_COLOR = (246, 245, 240)
DOT_COLOR = (52, 52, 58)
CROSS_COLOR = (198, 38, 38)

N_PAIRS = 36
DOT_RADIUS = 3
CROSS_ARM = 9
CROSS_HALF_THICK = 1

MIN_PAIR_SPACING = 40.0
MIN_CENTRE_CLEAR = 62.0
BORDER_MARGIN = 26.0

BOUNDARY_LOW = 22.5
BOUNDARY_HIGH = 67.5
QUARANTINE_DEG = 4.0
SAMPLING_FLOOR_DEG = 6.0

CLASS_NAMES = ("radial", "spiral", "circular")
CLASS_RANGES = {"radial": (0.0, 15.0), "spiral": (31.0, 59.0), "circular": (75.0, 90.0)}
JITTER_DEG = 6.0


def _acute_angle_deg(vx: float, vy: float, rx: float, ry: float) -> float:
    """Undirected angle in [0, 90] between two nonzero plane vectors."""
    dot = abs(vx * rx + vy * ry)
    cross = abs(vx * ry - vy * rx)
    return math.degrees(math.atan2(cross, dot))


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    target_class = CLASS_NAMES[seed % 3]
    lo, hi = CLASS_RANGES[target_class]
    chirality = rng.choice((-1.0, 1.0))
    dipole_len = rng.uniform(10.0, 14.0)
    cx = rng.uniform(0.36 * WIDTH, 0.64 * WIDTH)
    cy = rng.uniform(0.36 * HEIGHT, 0.64 * HEIGHT)

    while True:
        base_phi = rng.uniform(lo, hi)
        mids: list[tuple[float, float]] = []
        attempts = 0
        while len(mids) < N_PAIRS and attempts < 40000:
            attempts += 1
            x = rng.uniform(BORDER_MARGIN, WIDTH - BORDER_MARGIN)
            y = rng.uniform(BORDER_MARGIN, HEIGHT - BORDER_MARGIN)
            if math.hypot(x - cx, y - cy) < MIN_CENTRE_CLEAR:
                continue
            if any(math.hypot(x - px, y - py) < MIN_PAIR_SPACING for px, py in mids):
                continue
            mids.append((x, y))
        if len(mids) < N_PAIRS:
            continue

        pairs = []
        for x, y in mids:
            phi = base_phi + rng.uniform(-JITTER_DEG, JITTER_DEG)
            rad = math.atan2(y - cy, x - cx)
            theta = rad + chirality * math.radians(phi)
            hx = 0.5 * dipole_len * math.cos(theta)
            hy = 0.5 * dipole_len * math.sin(theta)
            pairs.append(
                {
                    "ax": round(x - hx, 4),
                    "ay": round(y - hy, 4),
                    "bx": round(x + hx, 4),
                    "by": round(y + hy, 4),
                }
            )
        scene = {
            "width": WIDTH,
            "height": HEIGHT,
            "bg_color": list(BG_COLOR),
            "dot_color": list(DOT_COLOR),
            "cross_color": list(CROSS_COLOR),
            "dot_radius": DOT_RADIUS,
            "cross": {"x": round(cx, 4), "y": round(cy, 4), "arm": CROSS_ARM},
            "pairs": pairs,
        }
        if margin(scene) >= SAMPLING_FLOOR_DEG:
            return scene


def _mean_angle(scene: dict) -> float:
    cx = scene["cross"]["x"]
    cy = scene["cross"]["y"]
    total = 0.0
    for pair in scene["pairs"]:
        mx = 0.5 * (pair["ax"] + pair["bx"])
        my = 0.5 * (pair["ay"] + pair["by"])
        total += _acute_angle_deg(
            pair["bx"] - pair["ax"], pair["by"] - pair["ay"], mx - cx, my - cy
        )
    return total / len(scene["pairs"])


def _classify(mean_deg: float) -> str:
    if mean_deg < BOUNDARY_LOW:
        return "radial"
    if mean_deg > BOUNDARY_HIGH:
        return "circular"
    return "spiral"


def analytic_gold(scene: dict) -> str:
    return _classify(_mean_angle(scene))


def margin(scene: dict) -> float:
    mean_deg = _mean_angle(scene)
    return float(min(abs(mean_deg - BOUNDARY_LOW), abs(mean_deg - BOUNDARY_HIGH)))


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_DEG)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (scene["width"], scene["height"]), tuple(scene["bg_color"]))
    draw = ImageDraw.Draw(image)
    cross = scene["cross"]
    cx, cy, arm = cross["x"], cross["y"], cross["arm"]
    colour = tuple(scene["cross_color"])
    draw.rectangle(
        [cx - arm, cy - CROSS_HALF_THICK, cx + arm, cy + CROSS_HALF_THICK], fill=colour
    )
    draw.rectangle(
        [cx - CROSS_HALF_THICK, cy - arm, cx + CROSS_HALF_THICK, cy + arm], fill=colour
    )
    r = scene["dot_radius"]
    dot_colour = tuple(scene["dot_color"])
    for pair in scene["pairs"]:
        for x, y in ((pair["ax"], pair["ay"]), (pair["bx"], pair["by"])):
            draw.ellipse([x - r, y - r, x + r, y + r], fill=dot_colour)
    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical relabelings of the unordered pair field."""
    pairs = list(scene["pairs"])
    shifted = dict(scene)
    shifted["pairs"] = pairs[7:] + pairs[:7]
    reversed_scene = dict(scene)
    reversed_scene["pairs"] = list(reversed(pairs))
    swapped = dict(scene)
    swapped["pairs"] = [
        {"ax": p["bx"], "ay": p["by"], "bx": p["ax"], "by": p["ay"]} for p in pairs
    ]
    return [
        ("pair_order_cyclic_shift", shifted),
        ("pair_order_reversal", reversed_scene),
        ("within_pair_endpoint_swap", swapped),
    ]
