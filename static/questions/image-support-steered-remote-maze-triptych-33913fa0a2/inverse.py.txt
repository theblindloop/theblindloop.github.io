"""Independent image-only oracle for the remote maze triptych."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


PANEL_BOXES = {
    "left": (18, 12, 142, 112),
    "center": (188, 246, 312, 346),
    "right": (498, 12, 622, 112),
}


def _crop(array: np.ndarray, box: tuple[int, int, int, int]) -> np.ndarray:
    x0, y0, x1, y1 = box
    return array[y0 + 4 : y1 - 4, x0 + 4 : x1 - 4]


def _wall_mask(crop: np.ndarray) -> np.ndarray:
    values = crop.astype(np.int16)
    gray = values.mean(axis=2)
    chroma = values.max(axis=2) - values.min(axis=2)
    # Walls are dark and nearly neutral. Colored marker pixels are excluded by
    # the chroma guard, and pale card edges are excluded by the luminance cut.
    return (gray < 165) & (chroma < 62)


def _marker_center(crop: np.ndarray, kind: str) -> tuple[int, int]:
    values = crop.astype(np.int16)
    red, green, blue = values[..., 0], values[..., 1], values[..., 2]
    if kind == "green":
        mask = (green > 85) & (green > red + 28) & (green > blue + 10)
    elif kind == "red":
        mask = (red > 110) & (red > green + 45) & (red > blue + 35)
    else:
        raise ValueError(f"unknown marker kind {kind!r}")
    points = np.argwhere(mask)
    if len(points) < 20:
        raise ValueError(f"could not segment {kind} marker")
    y, x = np.rint(points.mean(axis=0)).astype(int)
    return int(y), int(x)


def _reachable(walls: np.ndarray, start: tuple[int, int], goal: tuple[int, int]) -> bool:
    height, width = walls.shape
    sy, sx = start
    gy, gx = goal
    if not (0 <= sy < height and 0 <= sx < width and 0 <= gy < height and 0 <= gx < width):
        raise ValueError("marker centroid outside panel crop")
    if walls[sy, sx] or walls[gy, gx]:
        raise ValueError("marker touches a decoded wall")
    seen = np.zeros_like(walls, dtype=bool)
    queue: deque[tuple[int, int]] = deque([(sy, sx)])
    seen[sy, sx] = True
    while queue:
        y, x = queue.popleft()
        if (y, x) == (gy, gx):
            return True
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < height and 0 <= nx < width and not walls[ny, nx] and not seen[ny, nx]:
                seen[ny, nx] = True
                queue.append((ny, nx))
    return False


def _panel_connected(array: np.ndarray, box: tuple[int, int, int, int]) -> bool:
    crop = _crop(array, box)
    walls = _wall_mask(crop)
    green = _marker_center(crop, "green")
    red = _marker_center(crop, "red")
    return _reachable(walls, green, red)


def decision_from_image(image: Image.Image) -> str:
    """Return the unique connected panel from final PNG pixels only."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    connected = {
        role: _panel_connected(array, box) for role, box in PANEL_BOXES.items()
    }
    winners = [role for role in ("left", "center", "right") if connected[role]]
    if len(winners) != 1:
        raise ValueError(f"expected one connected maze panel, got {connected}")
    return winners[0]
