"""Renderer and analytic label for the remote maze triptych world."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw


RENDERER_VERSION = "remote_maze_triptych-0.1.0"
BACKGROUND = (255, 255, 255)
WALL = (31, 44, 58)
PALE = (226, 232, 240)
GREEN = (38, 160, 91)
RED = (211, 62, 70)
PANEL_ORDER = ("left", "center", "right")

# The panels intentionally occupy separated cells of the 4x4 support probe:
# upper-left, lower-middle, and upper-right.
PANEL_BOXES: dict[str, tuple[int, int, int, int]] = {
    "left": (18, 12, 142, 112),
    "center": (188, 246, 312, 346),
    "right": (498, 12, 622, 112),
}


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one balanced, one-winner triptych scene."""

    rng = random.Random(int(seed))
    winner = PANEL_ORDER[int(seed) % len(PANEL_ORDER)]
    clearance = rng.choice((6, 8, 10, 12, 16))
    scene: dict[str, Any] = {
        "width": 640,
        "height": 480,
        "clearance": int(clearance),
        "wall_width": rng.choice((7, 8, 9)),
    }
    for role in PANEL_ORDER:
        scene[f"{role}_connected"] = role == winner
        scene[f"{role}_axis"] = rng.choice(("vertical", "horizontal"))
        scene[f"{role}_door_shift"] = rng.choice((-12, -8, 0, 8, 12))
        scene[f"{role}_marker_shift"] = rng.choice((-14, -8, 8, 14))
        scene[f"{role}_variant"] = rng.randrange(4)
    return scene


def _scaled(value: float, scale: int) -> int:
    return int(round(float(value) * scale))


def _line(
    draw: ImageDraw.ImageDraw,
    points: list[tuple[float, float]],
    color: tuple[int, int, int],
    width: int,
    scale: int,
) -> None:
    draw.line(
        [(_scaled(x, scale), _scaled(y, scale)) for x, y in points],
        fill=color,
        width=max(1, _scaled(width, scale)),
        joint="curve",
    )


def _rectangle(
    draw: ImageDraw.ImageDraw,
    box: tuple[float, float, float, float],
    *,
    outline: tuple[int, int, int],
    width: int,
    scale: int,
) -> None:
    draw.rectangle(
        tuple(_scaled(value, scale) for value in box),
        outline=outline,
        width=max(1, _scaled(width, scale)),
    )


def _circle(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    radius: float,
    fill: tuple[int, int, int],
    scale: int,
) -> None:
    cx, cy = center
    box = (
        _scaled(cx - radius, scale),
        _scaled(cy - radius, scale),
        _scaled(cx + radius, scale),
        _scaled(cy + radius, scale),
    )
    draw.ellipse(box, fill=fill)


def _divider(
    draw: ImageDraw.ImageDraw,
    *,
    x0: float,
    y0: float,
    x1: float,
    y1: float,
    axis: str,
    connected: bool,
    door_shift: int,
    wall_width: int,
    clearance: int,
    scale: int,
) -> None:
    mid_x = (x0 + x1) / 2.0
    mid_y = (y0 + y1) / 2.0
    door_half = 9.0 + min(4.0, float(clearance) / 4.0)
    if axis == "vertical":
        start, end = y0 + wall_width / 2.0, y1 - wall_width / 2.0
        door = mid_y + float(door_shift)
        if connected:
            _line(draw, [(mid_x, start), (mid_x, door - door_half)], WALL, wall_width, scale)
            _line(draw, [(mid_x, door + door_half), (mid_x, end)], WALL, wall_width, scale)
        else:
            _line(draw, [(mid_x, start), (mid_x, end)], WALL, wall_width, scale)
    else:
        start, end = x0 + wall_width / 2.0, x1 - wall_width / 2.0
        door = mid_x + float(door_shift)
        if connected:
            _line(draw, [(start, mid_y), (door - door_half, mid_y)], WALL, wall_width, scale)
            _line(draw, [(door + door_half, mid_y), (end, mid_y)], WALL, wall_width, scale)
        else:
            _line(draw, [(start, mid_y), (end, mid_y)], WALL, wall_width, scale)


def _spurs(
    draw: ImageDraw.ImageDraw,
    *,
    x0: float,
    y0: float,
    x1: float,
    y1: float,
    axis: str,
    variant: int,
    wall_width: int,
    scale: int,
) -> None:
    """Add dead-end teeth so the panels are small mazes, not bare symbols."""

    mid_x = (x0 + x1) / 2.0
    mid_y = (y0 + y1) / 2.0
    if axis == "vertical":
        y_a = y0 + 20.0 + 9.0 * (variant % 2)
        y_b = y1 - 20.0 - 9.0 * ((variant // 2) % 2)
        _line(draw, [(x0, y_a), (mid_x - 15.0, y_a)], WALL, wall_width, scale)
        _line(draw, [(mid_x + 15.0, y_b), (x1, y_b)], WALL, wall_width, scale)
    else:
        x_a = x0 + 20.0 + 9.0 * (variant % 2)
        x_b = x1 - 20.0 - 9.0 * ((variant // 2) % 2)
        _line(draw, [(x_a, y0), (x_a, mid_y - 15.0)], WALL, wall_width, scale)
        _line(draw, [(x_b, mid_y + 15.0), (x_b, y1)], WALL, wall_width, scale)


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    box: tuple[int, int, int, int],
    *,
    connected: bool,
    axis: str,
    door_shift: int,
    marker_shift: int,
    variant: int,
    wall_width: int,
    clearance: int,
    scale: int,
) -> None:
    x_left, y_top, x_right, y_bottom = box
    # A pale card boundary is a visual frame, not a traversable wall.
    draw.rounded_rectangle(
        tuple(_scaled(value, scale) for value in box),
        radius=_scaled(5, scale),
        fill=BACKGROUND,
        outline=PALE,
        width=_scaled(2, scale),
    )
    outer = (x_left + 9, y_top + 9, x_right - 9, y_bottom - 9)
    _rectangle(draw, outer, outline=WALL, width=wall_width, scale=scale)
    ox0, oy0, ox1, oy1 = outer
    inner_x0 = ox0 + wall_width / 2.0
    inner_y0 = oy0 + wall_width / 2.0
    inner_x1 = ox1 - wall_width / 2.0
    inner_y1 = oy1 - wall_width / 2.0
    _divider(
        draw,
        x0=inner_x0,
        y0=inner_y0,
        x1=inner_x1,
        y1=inner_y1,
        axis=axis,
        connected=connected,
        door_shift=door_shift,
        wall_width=wall_width,
        clearance=clearance,
        scale=scale,
    )
    _spurs(
        draw,
        x0=inner_x0,
        y0=inner_y0,
        x1=inner_x1,
        y1=inner_y1,
        axis=axis,
        variant=variant,
        wall_width=wall_width,
        scale=scale,
    )

    mid_x = (inner_x0 + inner_x1) / 2.0
    mid_y = (inner_y0 + inner_y1) / 2.0
    shift = float(marker_shift)
    if axis == "vertical":
        green_center = (inner_x0 + 20.0, mid_y + shift)
        red_center = (inner_x1 - 20.0, mid_y - shift)
    else:
        green_center = (mid_x + shift, inner_y0 + 20.0)
        red_center = (mid_x - shift, inner_y1 - 20.0)
    # A small white halo makes marker segmentation stable without encoding the
    # answer; it is applied in the same way to both colors in every panel.
    for center, color in ((green_center, GREEN), (red_center, RED)):
        _circle(draw, center, 9.0, BACKGROUND, scale)
        _circle(draw, center, 6.5, color, scale)


def render(scene: dict[str, Any]) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    wall_width = int(scene["wall_width"])
    clearance = int(scene["clearance"])
    _draw_panel(
        draw,
        PANEL_BOXES["left"],
        connected=bool(scene["left_connected"]),
        axis=str(scene["left_axis"]),
        door_shift=int(scene["left_door_shift"]),
        marker_shift=int(scene["left_marker_shift"]),
        variant=int(scene["left_variant"]),
        wall_width=wall_width,
        clearance=clearance,
        scale=scale,
    )
    _draw_panel(
        draw,
        PANEL_BOXES["center"],
        connected=bool(scene["center_connected"]),
        axis=str(scene["center_axis"]),
        door_shift=int(scene["center_door_shift"]),
        marker_shift=int(scene["center_marker_shift"]),
        variant=int(scene["center_variant"]),
        wall_width=wall_width,
        clearance=clearance,
        scale=scale,
    )
    _draw_panel(
        draw,
        PANEL_BOXES["right"],
        connected=bool(scene["right_connected"]),
        axis=str(scene["right_axis"]),
        door_shift=int(scene["right_door_shift"]),
        marker_shift=int(scene["right_marker_shift"]),
        variant=int(scene["right_variant"]),
        wall_width=wall_width,
        clearance=clearance,
        scale=scale,
    )
    return image.resize((width, height), Image.Resampling.LANCZOS).convert("RGB")


def analytic_gold(scene: dict[str, Any]) -> str:
    statuses = (
        bool(scene["left_connected"]),
        bool(scene["center_connected"]),
        bool(scene["right_connected"]),
    )
    if sum(statuses) != 1:
        raise ValueError(f"expected exactly one connected panel, got {statuses}")
    return PANEL_ORDER[statuses.index(True)]


def margin(scene: dict[str, Any]) -> float:
    """One-sided raster robustness margin in pixels."""

    return float(scene["clearance"])


def is_quarantined(scene: dict[str, Any]) -> bool:
    return float(scene["clearance"]) < 8.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    del scene
    return []
