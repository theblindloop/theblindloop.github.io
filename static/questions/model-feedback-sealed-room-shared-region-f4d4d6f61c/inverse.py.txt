"""Independent image-only decision recovery.

The oracle never sees a latent scene. It segments the ink, treats every
non-ink pixel as walkable background, and propagates walkability outward from
the blue star until the wavefront stops. Exactly one coloured disk must be
swallowed by that wavefront.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

SCALE = 4
TOLERANCE = 30

PALETTE = {
    "wall": (0, 0, 0),
    "start": (0, 90, 220),
    "red": (220, 30, 30),
    "gold": (235, 170, 0),
    "magenta": (190, 0, 150),
}
DISK_COLORS = ("red", "gold", "magenta")


class OracleError(RuntimeError):
    """Raised when the raster does not expose exactly one reachable disk."""


def _mask(arr: np.ndarray, colour) -> np.ndarray:
    target = np.asarray(colour, dtype=np.int16)
    return np.max(np.abs(arr.astype(np.int16) - target), axis=2) <= TOLERANCE


def _coarse_all(mask: np.ndarray) -> np.ndarray:
    """Down-sample by SCALE keeping a coarse cell True only if fully True."""
    h, w = mask.shape
    h -= h % SCALE
    w -= w % SCALE
    block = mask[:h, :w].reshape(h // SCALE, SCALE, w // SCALE, SCALE)
    return block.all(axis=(1, 3))


def _coarse_any(mask: np.ndarray) -> np.ndarray:
    h, w = mask.shape
    h -= h % SCALE
    w -= w % SCALE
    block = mask[:h, :w].reshape(h // SCALE, SCALE, w // SCALE, SCALE)
    return block.any(axis=(1, 3))


def _flood(free: np.ndarray, seeds: np.ndarray) -> np.ndarray:
    reached = seeds & free
    if not reached.any():
        raise OracleError("start marker does not sit in walkable background")
    while True:
        grown = reached.copy()
        grown[1:, :] |= reached[:-1, :]
        grown[:-1, :] |= reached[1:, :]
        grown[:, 1:] |= reached[:, :-1]
        grown[:, :-1] |= reached[:, 1:]
        grown &= free
        if grown.sum() == reached.sum():
            return reached
        reached = grown


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    if arr.ndim != 3 or min(arr.shape[:2]) < 4 * SCALE:
        raise OracleError("unusable raster")

    wall = _mask(arr, PALETTE["wall"])
    if wall.sum() < 2000:
        raise OracleError("no wall structure found")

    start = _mask(arr, PALETTE["start"])
    if start.sum() < 200:
        raise OracleError("start marker missing")

    free = _coarse_all(~wall)
    seeds = _coarse_all(start)
    if not seeds.any():
        seeds = _coarse_any(start) & free
    reached = _flood(free, seeds)

    hits = []
    for colour in DISK_COLORS:
        disk = _mask(arr, PALETTE[colour])
        if disk.sum() < 200:
            continue
        coarse = _coarse_any(disk)
        if (coarse & reached).any():
            hits.append(colour)
        elif not (coarse & free).any():
            raise OracleError(f"{colour} disk is not embedded in background")
    if len(hits) != 1:
        raise OracleError(f"expected exactly one reachable disk, found {hits}")
    return hits[0]
