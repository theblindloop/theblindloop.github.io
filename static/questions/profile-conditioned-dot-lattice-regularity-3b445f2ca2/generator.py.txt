"""Dot-lattice-regularity world: does the dot field form a regular grid or an irregular scatter?

A field of equal-sized dark dots is drawn on a white canvas. In one arrangement
the dots sit on a regular square lattice: straight, equally spaced rows and
columns, every dot's nearest neighbours all at essentially the same distance. In
the other arrangement the same number of dots is scattered uniformly and
irregularly, so nearest-neighbour spacings vary widely. The decision is whether
the emergent global periodicity of the whole field is a regular grid (yes) or a
degenerate irregular scatter (no).

The renderer owns latent-scene sampling, rasterization, and the structural
analytic gold. The independent pixel-only inverse arm (``oracle.py``) recovers
the same decision from the final PNG alone: it segments every dark dot, takes
each dot's centroid, measures the coefficient of variation of nearest-neighbour
distances across the whole set, and classifies the field as a regular grid when
that spacing regularity is high and an irregular scatter when it is low.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (255, 255, 255)
DOT_COLOR = (40, 40, 40)

DOT_RADIUS = 6
CENTER = (WIDTH / 2.0, HEIGHT / 2.0)

N_GRID_ROWS = 5
N_GRID_COLS = 5
N_DOTS = N_GRID_ROWS * N_GRID_COLS  # 25 dots in both arrangements
GRID_SPACING = 70.0
GRID_JITTER = 2.0  # small positional noise on the lattice, keeps it visibly regular

# Scatter dots are drawn uniformly in the same square bounding extent as the
# grid, so total ink, dot count, and overall spread do not separate the classes.
BOX_HALF = (N_GRID_COLS - 1) * GRID_SPACING / 2.0  # 140 pixels

# Minimum centre distance between scatter dots so the raster never merges two
# disks into a single connected component (dot diameter is 2*DOT_RADIUS = 12).
MIN_SCATTER_DIST = 16.0

# Spacing-regularity threshold on the coefficient of variation (CV) of the
# nearest-neighbour distances: CV below this is a regular grid, above is scatter.
CV_THRESHOLD = 0.16
# Minimal separation from the threshold before a scene is quarantined.
QUAR_MIN = 0.05


def _lattice_points(rng) -> list[list[float]]:
    cx, cy = CENTER
    points = []
    for r in range(N_GRID_ROWS):
        for c in range(N_GRID_COLS):
            x = cx + (c - (N_GRID_COLS - 1) / 2.0) * GRID_SPACING + float(rng.normal(0.0, GRID_JITTER))
            y = cy + (r - (N_GRID_ROWS - 1) / 2.0) * GRID_SPACING + float(rng.normal(0.0, GRID_JITTER))
            points.append([x, y])
    return points


def _scatter_points(rng) -> list[list[float]]:
    cx, cy = CENTER
    points = []
    attempts = 0
    while len(points) < N_DOTS and attempts < 20000:
        attempts += 1
        x = float(rng.uniform(cx - BOX_HALF, cx + BOX_HALF))
        y = float(rng.uniform(cy - BOX_HALF, cy + BOX_HALF))
        if all(math.hypot(x - px, y - py) >= MIN_SCATTER_DIST for (px, py) in points):
            points.append([x, y])
    return points


def sample_scene(seed: int) -> dict:
    """Sample one latent scene deterministically from ``seed``."""
    rng = np.random.default_rng(seed)
    layout = "grid" if float(rng.random()) < 0.5 else "scatter"
    positions = _lattice_points(rng) if layout == "grid" else _scatter_points(rng)
    return {"layout": layout, "positions": positions}


def _spacing_regularity(positions: list[list[float]]) -> float:
    """Coefficient of variation of nearest-neighbour distances across the whole set."""
    pts = np.asarray(positions, dtype=np.float64)
    nn = np.zeros(len(pts), dtype=np.float64)
    for i, (px, py) in enumerate(pts):
        d = np.hypot(pts[:, 0] - px, pts[:, 1] - py)
        d = d[d > 1e-9]
        nn[i] = float(d.min())
    mean = float(nn.mean())
    if mean <= 1e-9:
        return 0.0
    return float(nn.std()) / mean


def _metric(positions: list[list[float]]) -> float:
    return _spacing_regularity(positions)


def render(scene: dict) -> Image.Image:
    """Rasterize the latent scene (a deterministic RGB PNG)."""
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    d = ImageDraw.Draw(img)
    for (px, py) in scene["positions"]:
        d.ellipse(
            [px - DOT_RADIUS, py - DOT_RADIUS, px + DOT_RADIUS, py + DOT_RADIUS],
            fill=DOT_COLOR,
        )
    return img


def analytic_gold(scene: dict) -> str:
    """Structural gold: 'yes' iff the dots form a regular lattice (low spacing CV)."""
    return "yes" if _metric(scene["positions"]) < CV_THRESHOLD else "no"


def margin(scene: dict) -> float:
    """Decision margin: separation of the spacing CV from the lattice threshold."""
    return abs(_metric(scene["positions"]) - CV_THRESHOLD)


def is_quarantined(scene: dict) -> bool:
    """Quarantine scenes whose spacing regularity is in the ambiguous band."""
    return margin(scene) < QUAR_MIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The scene is injectively rendered; no latent aliases are declared."""
    return []
