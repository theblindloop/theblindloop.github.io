"""Independent pixel-only inverse arm for the dot-lattice-regularity world.

Recovers whether a field of dark dots forms a regular grid or an irregular
scatter from the final PNG alone. It never sees the latent scene, analytic gold,
or renderer code. It thresholds the dark dots, labels each connected component,
takes each dot's centroid, measures the coefficient of variation of the
nearest-neighbour distances over the whole set, and answers 'yes' (regular
grid) when that spacing regularity is high and 'no' (irregular scatter) when it
is low.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

DARK = (40, 40, 40)
_BACKGROUND = 255
_TH = 60
CV_THRESHOLD = 0.16
MIN_COMPONENT_PIXELS = 8


def _decision_from_centroids(centers: list[tuple[float, float]]) -> str:
    if len(centers) < 2:
        return "no"
    pts = np.asarray(centers, dtype=np.float64)
    nn = np.zeros(len(pts), dtype=np.float64)
    for i, (px, py) in enumerate(pts):
        d = np.hypot(pts[:, 0] - px, pts[:, 1] - py)
        d = d[d > 1e-9]
        nn[i] = float(d.min())
    mean = float(nn.mean())
    if mean <= 1e-9:
        return "no"
    cv = float(nn.std()) / mean
    return "yes" if cv < CV_THRESHOLD else "no"


def decision_from_image(image: Image.Image) -> str:
    """'yes' if the dark dot field forms a regular lattice, else 'no'."""
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    r = arr[..., 0]
    foreground = (np.abs(r - DARK[0]) < _TH) & (
        (np.abs(arr[..., 1] - DARK[1]) < _TH) & (np.abs(arr[..., 2] - DARK[2]) < _TH)
    )
    h, w = foreground.shape

    # 4-connected-component labelling over foreground pixels only.
    lab = np.zeros((h, w), dtype=np.int32)
    next_label = 1
    stack = []
    ys, xs = np.nonzero(foreground)
    for y, x in zip(ys.tolist(), xs.tolist()):
        if lab[y, x] != 0:
            continue
        lab[y, x] = next_label
        stack = [(y, x)]
        while stack:
            cy, cx = stack.pop()
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < h and 0 <= nx < w and foreground[ny, nx] and lab[ny, nx] == 0:
                    lab[ny, nx] = next_label
                    stack.append((ny, nx))
        next_label += 1

    centers = []
    for label in range(1, next_label):
        mask = lab == label
        count = int(mask.sum())
        if count < MIN_COMPONENT_PIXELS:
            continue
        coords = np.argwhere(mask)
        centers.append((float(coords[:, 1].mean()), float(coords[:, 0].mean())))
    return _decision_from_centroids(centers)
