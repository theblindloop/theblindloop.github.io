"""Independent pixel-only inverse for the colored route world.

This module intentionally contains its own palette, marker measurements, and
segment-support geometry. It receives only the final RGB image.
"""

from __future__ import annotations

import itertools
import math
from collections import deque

import numpy as np
from PIL import Image

PALETTE = {
    "green": np.array([32, 163, 125], dtype=np.float32),
    "red": np.array([214, 73, 66], dtype=np.float32),
    "blue": np.array([52, 111, 207], dtype=np.float32),
    "gold": np.array([232, 166, 40], dtype=np.float32),
}
INK = np.array([44, 58, 80], dtype=np.float32)


def _center_for_color(rgb: np.ndarray, color: np.ndarray) -> tuple[float, float]:
    distance = np.sqrt(np.sum((rgb - color) ** 2, axis=2))
    mask = distance < 24.0
    ys, xs = np.nonzero(mask)
    if len(xs) < 400:
        raise ValueError("colored node marker is missing or too small")
    return float(xs.mean()), float(ys.mean())


def _line_support(rgb: np.ndarray, first: tuple[float, float], second: tuple[float, float]) -> float:
    dark = np.sqrt(np.sum((rgb - INK) ** 2, axis=2)) < 72.0
    distance = math.hypot(second[0] - first[0], second[1] - first[1])
    count = max(50, int(distance * 0.72))
    hits = 0
    height, width = dark.shape
    for fraction in np.linspace(0.14, 0.86, count):
        x = int(round(first[0] + fraction * (second[0] - first[0])))
        y = int(round(first[1] + fraction * (second[1] - first[1])))
        x0, x1 = max(0, x - 3), min(width, x + 4)
        y0, y1 = max(0, y - 3), min(height, y + 4)
        if bool(np.any(dark[y0:y1, x0:x1])):
            hits += 1
    return hits / float(count)


def _path(adjacency: dict[str, set[str]]) -> list[str]:
    parent: dict[str, str | None] = {"green": None}
    queue: deque[str] = deque(["green"])
    while queue:
        current = queue.popleft()
        if current == "red":
            break
        for neighbor in sorted(adjacency[current]):
            if neighbor not in parent:
                parent[neighbor] = current
                queue.append(neighbor)
    if "red" not in parent:
        return []
    result: list[str] = []
    current: str | None = "red"
    while current is not None:
        result.append(current)
        current = parent[current]
    return list(reversed(result))


def decision_from_image(image: Image.Image) -> str:
    """Return the candidate color on the dark green-to-red path."""

    rgb = np.asarray(image.convert("RGB"), dtype=np.float32)
    centers = {name: _center_for_color(rgb, color) for name, color in PALETTE.items()}
    adjacency: dict[str, set[str]] = {name: set() for name in PALETTE}
    for left, right in itertools.combinations(PALETTE, 2):
        support = _line_support(rgb, centers[left], centers[right])
        if support >= 0.62:
            adjacency[left].add(right)
            adjacency[right].add(left)
    route = _path(adjacency)
    candidates = [color for color in ("blue", "gold") if color in route]
    if len(candidates) != 1:
        raise ValueError("pixel graph does not expose exactly one colored route candidate")
    return candidates[0]
