"""Deterministic renderer and analytic graph oracle for the candidate world."""

from __future__ import annotations

import math
import random
from collections import deque

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 900, 620
BACKGROUND = (250, 247, 239)
INK = (44, 58, 80)
DISTRACTOR = (211, 215, 220)
PALETTE = {
    "green": (32, 163, 125),
    "red": (214, 73, 66),
    "blue": (52, 111, 207),
    "gold": (232, 166, 40),
}


def _transform(point: tuple[float, float], angle: float, shift: tuple[float, float]) -> tuple[int, int]:
    cx, cy = WIDTH / 2.0, HEIGHT / 2.0
    x, y = point
    ca, sa = math.cos(angle), math.sin(angle)
    tx = cx + ca * (x - cx) - sa * (y - cy) + shift[0]
    ty = cy + sa * (x - cx) + ca * (y - cy) + shift[1]
    return (int(round(tx)), int(round(ty)))


def sample_scene(seed: int) -> dict:
    """Sample a balanced, non-crossing four-node tree."""

    rng = random.Random(int(seed))
    seed_value = int(seed)
    route_slot = "slot_a" if seed_value % 4 in (0, 1) else "slot_b"
    blue_on_a = seed_value % 2 == 0
    mirror = bool(rng.randrange(2))
    dy = rng.randint(145, 178)
    mid_x = rng.randint(405, 465)
    base = {
        "green": (112, 310),
        "red": (790, 310),
        "slot_a": (mid_x, 310 - dy),
        "slot_b": (mid_x, 310 + dy),
    }
    if mirror:
        base["slot_a"], base["slot_b"] = base["slot_b"], base["slot_a"]
    angle = rng.uniform(-0.105, 0.105)
    shift = (rng.randint(-24, 24), rng.randint(-18, 18))
    positions = {key: _transform(value, angle, shift) for key, value in base.items()}

    a_color = "blue" if blue_on_a else "gold"
    b_color = "gold" if blue_on_a else "blue"
    nodes = [
        {"id": "green", "color": "green", "x": positions["green"][0], "y": positions["green"][1]},
        {"id": "red", "color": "red", "x": positions["red"][0], "y": positions["red"][1]},
        {"id": "slot_a", "color": a_color, "x": positions["slot_a"][0], "y": positions["slot_a"][1]},
        {"id": "slot_b", "color": b_color, "x": positions["slot_b"][0], "y": positions["slot_b"][1]},
    ]
    if route_slot == "slot_a":
        edges = [["green", "slot_a"], ["slot_a", "red"], ["green", "slot_b"]]
    else:
        edges = [["green", "slot_b"], ["slot_b", "red"], ["green", "slot_a"]]

    distractor_base = [
        ((65, 92), (270, 72)),
        ((650, 74), (846, 111)),
        ((74, 540), (290, 514)),
        ((642, 539), (850, 501)),
    ]
    distractors = [
        [_transform(first, angle, shift), _transform(second, angle, shift)]
        for first, second in distractor_base
    ]
    return {
        "canvas": [WIDTH, HEIGHT],
        "background": list(BACKGROUND),
        "ink": list(INK),
        "stroke_width": 13,
        "marker_radius": 22,
        "halo_radius": 31,
        "nodes": nodes,
        "edges": edges,
        "distractor_segments": distractors,
    }


def _node_map(scene: dict) -> dict[str, dict]:
    return {node["id"]: node for node in scene["nodes"]}


def _route_ids(scene: dict) -> list[str]:
    graph: dict[str, list[str]] = {node["id"]: [] for node in scene["nodes"]}
    for left, right in scene["edges"]:
        graph[left].append(right)
        graph[right].append(left)
    parent: dict[str, str | None] = {"green": None}
    queue: deque[str] = deque(["green"])
    while queue:
        current = queue.popleft()
        if current == "red":
            break
        for neighbor in graph.get(current, []):
            if neighbor not in parent:
                parent[neighbor] = current
                queue.append(neighbor)
    if "red" not in parent:
        return []
    path: list[str] = []
    current: str | None = "red"
    while current is not None:
        path.append(current)
        current = parent[current]
    return list(reversed(path))


def render(scene: dict) -> Image.Image:
    width, height = (int(value) for value in scene["canvas"])
    image = Image.new("RGB", (width, height), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    for first, second in scene["distractor_segments"]:
        draw.line([tuple(first), tuple(second)], fill=tuple(DISTRACTOR), width=5)

    nodes = _node_map(scene)
    for left, right in scene["edges"]:
        p = (nodes[left]["x"], nodes[left]["y"])
        q = (nodes[right]["x"], nodes[right]["y"])
        draw.line([p, q], fill=tuple(scene["ink"]), width=int(scene["stroke_width"]), joint="curve")

    halo = int(scene["halo_radius"])
    radius = int(scene["marker_radius"])
    for node in scene["nodes"]:
        center = (int(node["x"]), int(node["y"]))
        draw.ellipse(
            [center[0] - halo, center[1] - halo, center[0] + halo, center[1] + halo],
            fill=tuple(scene["background"]),
        )
        draw.ellipse(
            [center[0] - radius - 3, center[1] - radius - 3, center[0] + radius + 3, center[1] + radius + 3],
            fill=tuple(scene["ink"]),
        )
        fill = PALETTE[node["color"]]
        draw.ellipse(
            [center[0] - radius, center[1] - radius, center[0] + radius, center[1] + radius],
            fill=tuple(fill),
        )
    return image


def analytic_gold(scene: dict) -> str:
    path = _route_ids(scene)
    nodes = _node_map(scene)
    candidate_ids = ("slot_a", "slot_b")
    on_route = [node_id for node_id in candidate_ids if node_id in path]
    if len(on_route) != 1:
        raise ValueError("scene does not contain exactly one candidate on the route")
    return str(nodes[on_route[0]]["color"])


def margin(scene: dict) -> float:
    path = _route_ids(scene)
    count = sum(node_id in path for node_id in ("slot_a", "slot_b"))
    return 1.0 if count == 1 else 0.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
