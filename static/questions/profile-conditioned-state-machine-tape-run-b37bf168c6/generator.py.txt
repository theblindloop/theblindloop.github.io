"""Scene sampler, renderer and analytic gold for state_machine_tape_run.

The image shows a transition table (four coloured state discs down the left,
three coloured symbol squares along the top, and one coloured square in every
cell naming the state reached from that row's state on that column's symbol),
a black ring marking the starting state, and an ordered tape of coloured symbol
squares below the table.  The answer is the state the marker ends on after the
tape is consumed left to right, together with how many tape steps moved it.

Nothing about the run is drawn: the table and the tape are drawn, the trajectory
is not.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "state_machine_tape_run-renderer-0.1.0"

W, H = 780, 560
BG = (255, 255, 255)
INK = (90, 90, 96)          # slate rule work: grid lines, arrow, borders
RING = (0, 0, 0)            # reserved: the start marker only

# four state colours and three symbol colours, disjoint and far apart
STATE_PALETTE = {
    "red": (205, 35, 45),
    "blue": (35, 75, 200),
    "green": (30, 145, 70),
    "orange": (240, 150, 25),
}
SYMBOL_PALETTE = {
    "violet": (150, 40, 190),
    "teal": (25, 185, 195),
    "yellow": (245, 225, 55),
}
STATE_NAMES = ("red", "blue", "green", "orange")
SYMBOL_NAMES = ("violet", "teal", "yellow")

N_STATES = 4
N_SYMBOLS = 3
TAPE_LEN = 5

DISC_X = 90
ROW_Y0, ROW_DY = 150, 80
COL_X0, COL_DX = 190, 90
HEAD_Y = 60
CELL = 48
HEAD = 40
DISC_R = 20
RING_X = 40
RING_R = 16
RING_WIDTH = 5
TAPE_Y = 490
TAPE_X0, TAPE_DX, TAPE_SIDE = 120, 80, 44

QUARANTINE_MARGIN = 6.0
MARGIN_CAP = 25.0


# --------------------------------------------------------------------------
# geometry helpers shared by gold and renderer (never by the pixel oracle)
# --------------------------------------------------------------------------
def _row_centre(i: int) -> tuple[float, float]:
    return float(DISC_X), float(ROW_Y0 + i * ROW_DY)


def _cell_centre(i: int, j: int) -> tuple[float, float]:
    return float(COL_X0 + j * COL_DX), float(ROW_Y0 + i * ROW_DY)


def _ring_centre(scene: dict) -> tuple[float, float]:
    _, cy = _row_centre(scene["ring"]["row"])
    return RING_X + scene["ring"]["dx"], cy + scene["ring"]["dy"]


def _ring_state(scene: dict) -> int:
    """The marked state: the row disc whose centre is nearest the ring centre.

    This is exactly the rule the question states and the rule the pixel arm
    applies; the stored ``ring['row']`` is only where the ring was drawn from.
    """
    rx, ry = _ring_centre(scene)
    best, best_d = 0, None
    for i in range(N_STATES):
        cx, cy = _row_centre(i)
        d = math.hypot(rx - cx, ry - cy)
        if best_d is None or d < best_d:
            best, best_d = i, d
    return best


def _table(scene: dict) -> list[list[int]]:
    """Rebuild the transition matrix from the unordered cell bag."""
    tbl = [[0] * N_SYMBOLS for _ in range(N_STATES)]
    for cell in scene["table"]:
        tbl[cell["r"]][cell["c"]] = cell["to"]
    return tbl


def _tape(scene: dict) -> list[int]:
    """Symbol column indices in drawn left-to-right order."""
    return [t["sym"] for t in sorted(scene["tape"], key=lambda t: t["slot"])]


def _state_colour(scene: dict, row: int) -> str:
    for s in scene["states"]:
        if s["row"] == row:
            return s["colour"]
    raise KeyError(row)


def _run(scene: dict, tape: list[int] | None = None) -> tuple[int, int]:
    """Replay the tape from the marked state; return (final row, moves)."""
    tbl = _table(scene)
    cur = _ring_state(scene)
    moves = 0
    for sym in (_tape(scene) if tape is None else tape):
        nxt = tbl[cur][sym]
        if nxt != cur:
            moves += 1
        cur = nxt
    return cur, moves


# --------------------------------------------------------------------------
# required interface
# --------------------------------------------------------------------------
def analytic_gold(scene: dict) -> str:
    final, moves = _run(scene)
    return f"{_state_colour(scene, final)}|{moves}"


def margin(scene: dict) -> float:
    """Clearance of the drawn ring from the bisector between two state discs.

    The only continuous reading in the image is which state disc the ring
    marks; every other reading (which colour fills a cell, which symbol a tape
    square carries, what order the tape is in) is discrete and separated by
    whole tens of pixels.  The margin is half the gap between the ring's
    distance to the nearest disc and to the runner-up disc, capped by the
    smallest drawn separation elsewhere in the figure.
    """
    rx, ry = _ring_centre(scene)
    ds = sorted(math.hypot(rx - _row_centre(i)[0], ry - _row_centre(i)[1]) for i in range(N_STATES))
    return round(min((ds[1] - ds[0]) / 2.0, MARGIN_CAP), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Re-orderings of the unordered bags: pixel-identical, answer-preserving."""
    out: list[tuple[str, dict]] = []
    for key in ("table", "tape", "states", "symbols"):
        out.append((f"{key}_bag_reversed", dict(scene, **{key: list(reversed(scene[key]))})))
    bag = list(scene["table"])
    out.append(("table_bag_rotated", dict(scene, table=bag[3:] + bag[:3])))
    return out


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    target = STATE_NAMES[seed % N_STATES]
    # the tight-marker mode is spread evenly over the four answer classes
    tight = (seed % 32) in (5, 14, 23, 28)

    for _ in range(4000):
        state_cols = list(STATE_NAMES)
        rng.shuffle(state_cols)
        sym_cols = list(SYMBOL_NAMES)
        rng.shuffle(sym_cols)

        table = [
            {"r": i, "c": j, "to": rng.randrange(N_STATES)}
            for i in range(N_STATES)
            for j in range(N_SYMBOLS)
        ]
        tape = [{"slot": k, "sym": rng.randrange(N_SYMBOLS)} for k in range(TAPE_LEN)]

        if tight:
            step = rng.uniform(32.0, 39.0)
        else:
            step = rng.uniform(0.0, 26.0)
        toward = rng.choice([-1, 1])
        row = rng.randrange(N_STATES)
        if row == 0 and toward < 0:
            row = 1
        if row == N_STATES - 1 and toward > 0:
            row = N_STATES - 2
        ring = {"row": row, "dx": round(rng.uniform(-5.0, 5.0), 3), "dy": round(step * toward, 3)}

        scene = {
            "canvas": [W, H],
            "states": [{"row": i, "colour": state_cols[i]} for i in range(N_STATES)],
            "symbols": [{"col": j, "colour": sym_cols[j]} for j in range(N_SYMBOLS)],
            "table": table,
            "tape": tape,
            "ring": ring,
        }

        syms = _tape(scene)
        if len(set(syms)) < 2:
            continue
        final, _ = _run(scene)
        if _state_colour(scene, final) != target:
            continue
        # the drawn program must be load bearing, and its order must matter
        if _run(scene, syms[:-1])[0] == final:
            continue
        if _run(scene, list(reversed(syms)))[0] == final:
            continue
        rng.shuffle(scene["table"])
        rng.shuffle(scene["tape"])
        rng.shuffle(scene["states"])
        rng.shuffle(scene["symbols"])
        return scene
    raise RuntimeError(f"failed to sample a scene for seed {seed}")


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", tuple(scene["canvas"]), BG)
    d = ImageDraw.Draw(img)

    # table frame and rules
    left = COL_X0 - CELL // 2 - 14
    right = COL_X0 + (N_SYMBOLS - 1) * COL_DX + CELL // 2 + 14
    top = ROW_Y0 - CELL // 2 - 14
    bot = ROW_Y0 + (N_STATES - 1) * ROW_DY + CELL // 2 + 14
    d.rectangle([left, top, right, bot], outline=INK, width=3)

    # column headers: the symbol each column responds to
    for s in scene["symbols"]:
        cx = COL_X0 + s["col"] * COL_DX
        d.rectangle(
            [cx - HEAD // 2, HEAD_Y - HEAD // 2, cx + HEAD // 2, HEAD_Y + HEAD // 2],
            fill=SYMBOL_PALETTE[s["colour"]],
            outline=INK,
            width=2,
        )

    # row headers: the state discs
    for s in scene["states"]:
        cx, cy = _row_centre(s["row"])
        d.ellipse(
            [cx - DISC_R, cy - DISC_R, cx + DISC_R, cy + DISC_R],
            fill=STATE_PALETTE[s["colour"]],
            outline=INK,
            width=2,
        )

    # transition cells
    for cell in scene["table"]:
        cx, cy = _cell_centre(cell["r"], cell["c"])
        d.rectangle(
            [cx - CELL // 2, cy - CELL // 2, cx + CELL // 2, cy + CELL // 2],
            fill=STATE_PALETTE[_state_colour(scene, cell["to"])],
            outline=INK,
            width=2,
        )

    # tape: the ordered input, left to right
    for t in scene["tape"]:
        cx = TAPE_X0 + t["slot"] * TAPE_DX
        d.rectangle(
            [cx - TAPE_SIDE // 2, TAPE_Y - TAPE_SIDE // 2, cx + TAPE_SIDE // 2, TAPE_Y + TAPE_SIDE // 2],
            fill=SYMBOL_PALETTE[
                next(s["colour"] for s in scene["symbols"] if s["col"] == t["sym"])
            ],
            outline=INK,
            width=2,
        )
    ay = TAPE_Y + TAPE_SIDE // 2 + 22
    ax0 = TAPE_X0 - TAPE_SIDE // 2 - 10
    ax1 = TAPE_X0 + (TAPE_LEN - 1) * TAPE_DX + TAPE_SIDE // 2 + 10
    d.line([ax0, ay, ax1, ay], fill=INK, width=4)
    d.polygon([(ax1 + 14, ay), (ax1 - 4, ay - 11), (ax1 - 4, ay + 11)], fill=INK)

    # the start marker, drawn last and in the reserved colour
    rx, ry = _ring_centre(scene)
    d.ellipse([rx - RING_R, ry - RING_R, rx + RING_R, ry + RING_R], outline=RING, width=RING_WIDTH)
    return img
