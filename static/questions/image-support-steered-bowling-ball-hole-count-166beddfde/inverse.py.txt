"""Independent pixel-only inverse for the compact bowling-ball icon."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]:
            continue
        seen[y0, x0] = True
        queue = deque([(y0, x0)])
        points = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        result.append(np.asarray(points, dtype=np.int32))
    return result


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    if array.ndim != 3 or min(array.shape[:2]) < 192:
        raise ValueError("image is too small")
    border = np.concatenate((array[:4].reshape(-1, 3), array[-4:].reshape(-1, 3),
                             array[:, :4].reshape(-1, 3), array[:, -4:].reshape(-1, 3)))
    background = np.median(border, axis=0)
    foreground = np.linalg.norm(array - background, axis=2) > 42.0
    objects = [part for part in _components(foreground) if len(part) > 100]
    if len(objects) != 1:
        raise ValueError("expected one compact foreground icon")
    host = objects[0]
    ys, xs = host[:, 0], host[:, 1]
    if not (9000 <= len(host) <= 30000):
        raise ValueError("foreground host has invalid area")

    dark = np.max(array, axis=2) < 72.0
    holes = []
    host_cx, host_cy = float(xs.mean()), float(ys.mean())
    for part in _components(dark):
        area = len(part)
        if area < 90:
            continue
        py, px = part[:, 0], part[:, 1]
        bw, bh = int(px.max() - px.min() + 1), int(py.max() - py.min() + 1)
        fill = area / float(bw * bh)
        if not (15 <= min(bw, bh) <= 30 and abs(bw - bh) <= 3 and 0.70 <= fill <= 0.88):
            raise ValueError("dark component is not an intact circular hole")
        pcx, pcy = float(px.mean()), float(py.mean())
        if np.hypot(pcx - host_cx, pcy - host_cy) > 100.0:
            raise ValueError("dark circle is not enclosed near the ball center")
        holes.append(part)
    if len(holes) not in range(2, 6):
        raise ValueError("ambiguous visible finger-hole count")
    return str(len(holes))
