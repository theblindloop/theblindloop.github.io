"""Deterministic renderer for visible bowling-ball finger-hole counting."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 512
SUPERSAMPLE = 4
HOLE_COLOR = (18, 21, 27)
PALETTES = (
    ((36, 112, 184), (248, 244, 231)),
    ((197, 66, 91), (239, 247, 242)),
    ((39, 151, 118), (250, 238, 232)),
    ((128, 75, 181), (246, 245, 226)),
    ((220, 132, 35), (234, 244, 250)),
)


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    ball_color, background = PALETTES[rng.randrange(len(PALETTES))]
    count = 2 + (int(seed) % 4)
    ball_radius = rng.uniform(88.0, 92.0)
    # Randomize absolute location while retaining a stable raster sampling phase.
    # The two decisive lobes then occupy grid cells with one blank cell between.
    cx = float(rng.choice((160, 224, 288, 352)))
    cy = float(rng.choice((160, 224, 288, 352)))
    x_gap = 64.0
    # Approximately normalize total dark ink across answer identities.
    hole_radius = 10.5 * math.sqrt(3.0 / count) * rng.uniform(0.96, 1.04)
    layouts = {
        2: [(-1.0, 0.0), (1.0, 0.0)],
        3: [(-1.0, 0.0), (1.0, -0.34), (1.0, 0.34)],
        4: [(-1.0, -0.32), (-1.0, 0.32), (1.0, -0.32), (1.0, 0.32)],
        5: [(-1.0, -0.34), (-1.0, 0.34), (1.0, -0.42), (1.0, 0.0), (1.0, 0.42)],
    }
    centers = [[cx + sx * x_gap, cy + sy * x_gap] for sx, sy in layouts[count]]
    return {
        "hole_centers": centers,
        "hole_radius": hole_radius,
        "ball_center": [cx, cy],
        "ball_radius": ball_radius,
        "ball_color": list(ball_color),
        "background": list(background),
        "width": WIDTH,
        "height": HEIGHT,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    scale = SUPERSAMPLE
    width, height = int(scene["width"]), int(scene["height"])
    background = tuple(int(v) for v in scene["background"])
    ball_color = tuple(int(v) for v in scene["ball_color"])
    large = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(large)
    cx, cy = (float(v) * scale for v in scene["ball_center"])
    radius = float(scene["ball_radius"]) * scale
    draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=ball_color)
    hole_radius = float(scene["hole_radius"]) * scale
    for hx0, hy0 in scene["hole_centers"]:
        hx, hy = float(hx0) * scale, float(hy0) * scale
        draw.ellipse((hx - hole_radius, hy - hole_radius, hx + hole_radius, hy + hole_radius), fill=HOLE_COLOR)
    return large.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(len(scene["hole_centers"]))


def margin(scene: dict[str, Any]) -> float:
    centers = [(float(x), float(y)) for x, y in scene["hole_centers"]]
    hr = float(scene["hole_radius"])
    cx, cy = (float(v) for v in scene["ball_center"])
    br = float(scene["ball_radius"])
    gaps = [math.hypot(x - cx, y - cy) for x, y in centers]
    boundary_gap = min(br - distance - hr for distance in gaps)
    pair_gap = min(
        (math.hypot(x1 - x2, y1 - y2) - 2.0 * hr
         for index, (x1, y1) in enumerate(centers)
         for x2, y2 in centers[index + 1:]),
        default=boundary_gap,
    )
    return min(boundary_gap, pair_gap, 2.0 * hr)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 8.0 or float(scene["hole_radius"]) < 7.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    transformed = dict(scene)
    transformed["hole_centers"] = list(reversed([list(pair) for pair in scene["hole_centers"]]))
    return [("reverse_hole_list", transformed)]
