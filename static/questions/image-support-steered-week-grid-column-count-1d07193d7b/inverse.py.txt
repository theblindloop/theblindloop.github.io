"""Independent image-only inverse for separated WEEK-grid column boxes."""

from __future__ import annotations

import numpy as np
from PIL import Image

_VALID = {5, 6, 7, 8, 9}


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    """Return (x0,y0,x1,y1,area) for 4-connected foreground components."""
    work = mask.copy()
    height, width = work.shape
    out = []
    for y in range(height):
        for x in range(width):
            if not work[y, x]:
                continue
            stack = [(y, x)]
            work[y, x] = False
            x0 = x1 = x
            y0 = y1 = y
            area = 0
            while stack:
                cy, cx = stack.pop()
                area += 1
                x0, x1 = min(x0, cx), max(x1, cx)
                y0, y1 = min(y0, cy), max(y1, cy)
                for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                    if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                        work[ny, nx] = False
                        stack.append((ny, nx))
            out.append((x0, y0, x1, y1, area))
    return out


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    if array.shape[:2] != (480, 640):
        raise ValueError("unexpected canvas")
    distance = np.linalg.norm(array.astype(np.float32) - 255.0, axis=2)
    components = _components(distance > 14.0)
    boxes = []
    for x0, y0, x1, y1, area in components:
        width, height = x1 - x0 + 1, y1 - y0 + 1
        fill = area / float(width * height)
        if 16 <= width <= 55 and 125 <= height <= 165 and area >= 1900 and fill >= 0.72:
            boxes.append((x0, y0, x1, y1, area))
    count = len(boxes)
    if count not in _VALID:
        raise ValueError("visible planner box count is outside the answer set")
    boxes.sort()
    heights = [b[3] - b[1] + 1 for b in boxes]
    tops = [b[1] for b in boxes]
    if max(heights) - min(heights) > 2 or max(tops) - min(tops) > 2:
        raise ValueError("candidate boxes do not form one aligned grid")
    gaps = [boxes[i + 1][0] - boxes[i][2] - 1 for i in range(count - 1)]
    wide = [gap for gap in gaps if gap > 70]
    ordinary = [gap for gap in gaps if gap <= 70]
    if len(wide) != 1 or not 190 <= wide[0] <= 215:
        raise ValueError("grid must have exactly one visible center break")
    if not ordinary or min(ordinary) < 7 or max(ordinary) > 15:
        raise ValueError("grid boxes are merged or implausibly separated")
    span = boxes[-1][2] - boxes[0][0] + 1
    if not 440 <= span <= 470:
        raise ValueError("grid region has an unexpected span")
    return str(count)
