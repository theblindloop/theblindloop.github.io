"""Analytic sampler and raster renderer for visible WEEK-grid columns."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 640, 480
BACKGROUND = (255, 255, 255)
COUNTS = (5, 6, 7, 8, 9)
PALETTES = (
    ((38, 67, 91), (225, 236, 244), (241, 228, 205)),
    ((73, 54, 96), (235, 228, 244), (224, 241, 230)),
    ((94, 55, 42), (244, 230, 218), (226, 237, 245)),
)
MIN_GAP = 9.0
MIN_WIDTH = 14.0


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x31C07A5E)
    count = COUNTS[int(seed) % len(COUNTS)]
    decoration_seed = rng.randrange(10_000)
    gap = 9 + decoration_seed % 4
    center_extra = 185 + ((decoration_seed // 4) % 5) * 3
    target_width = rng.randrange(448, 467)
    cell_width = (target_width - center_extra - (count - 1) * gap) // count
    cell_height = rng.randrange(132, 159)
    actual_width = count * cell_width + (count - 1) * gap + center_extra
    x = (WIDTH - actual_width) // 2 + rng.randrange(-20, 21)
    y = rng.randrange(190, 213)
    ink, a, b = PALETTES[rng.randrange(len(PALETTES))]
    if rng.randrange(2):
        a, b = b, a
    return {
        "column_count": count,
        "x": x,
        "y": y,
        "cell_width": cell_width,
        "cell_height": cell_height,
        "ink_color": list(ink),
        "fill_a": list(a),
        "fill_b": list(b),
        "phase": rng.randrange(12),
        "decoration_seed": decoration_seed,
    }


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size)


def render(scene: dict) -> Image.Image:
    count = int(scene["column_count"])
    x, y = int(scene["x"]), int(scene["y"])
    cw, ch = int(scene["cell_width"]), int(scene["cell_height"])
    ink = tuple(int(v) for v in scene["ink_color"])
    fill_a = tuple(int(v) for v in scene["fill_a"])
    fill_b = tuple(int(v) for v in scene["fill_b"])
    phase = int(scene["phase"]) % 2
    decoration_seed = int(scene["decoration_seed"])
    gap = 9 + decoration_seed % 4
    center_extra = 185 + ((decoration_seed // 4) % 5) * 3
    split = (count + 1) // 2

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title = "WEEK"
    title_font = _font(31)
    box = draw.textbbox((0, 0), title, font=title_font)
    total_width = count * cw + (count - 1) * gap + center_extra
    tx = x + (total_width - (box[2] - box[0])) // 2
    draw.text((tx, y - 52), title, fill=ink, font=title_font)

    for index in range(count):
        left = x + index * (cw + gap) + (center_extra if index >= split else 0)
        right = left + cw - 1
        bottom = y + ch - 1
        fill = fill_a if (index + phase) % 2 == 0 else fill_b
        draw.rectangle((left, y, right, bottom), fill=fill, outline=ink, width=4)
        # Neutral planner-writing rules are identical in role across counts.
        rule_count = 3 + ((decoration_seed + index) % 2)
        for row in range(rule_count):
            ry = y + 28 + row * 23
            if ry < bottom - 10 and right - left >= 22:
                draw.line((left + 8, ry, right - 8, ry), fill=ink, width=2)
    return image


def analytic_gold(scene: dict) -> str:
    return str(int(scene["column_count"]))


def margin(scene: dict) -> float:
    decoration_seed = int(scene["decoration_seed"])
    gap = 9 + decoration_seed % 4
    center_extra = 185 + ((decoration_seed // 4) % 5) * 3
    return float(min(gap - 4.0, int(scene["cell_width"]) - MIN_WIDTH, center_extra - 180.0))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["phase"] = int(scene["phase"]) + 2
    return [("phase_plus_two", twin)]
