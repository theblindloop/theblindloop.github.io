"""Pixel-only decision arm.

Receives a rendered RGB image and nothing else. It re-derives the tile masks,
the rotation and reflection orbits, and the bound dot colors directly from the
raster. It imports no renderer, prompt, generator, verifier or gold code and
never sees latent scene data.
"""

import numpy as np

_PALETTE = {
    "red": (214, 40, 40),
    "blue": (30, 80, 200),
    "green": (20, 150, 60),
    "orange": (240, 140, 20),
}

ABSTAIN = "abstain"


def _runs(flags, min_len=4):
    out = []
    start = None
    for i, f in enumerate(flags):
        if f and start is None:
            start = i
        elif not f and start is not None:
            if i - start >= min_len:
                out.append((start, i - 1))
            start = None
    if start is not None and len(flags) - start >= min_len:
        out.append((start, len(flags) - 1))
    return out


def _crop(mask):
    ys, xs = np.nonzero(mask)
    if ys.size == 0:
        return None
    return mask[ys.min(): ys.max() + 1, xs.min(): xs.max() + 1]


def _orbit(mask):
    rots = [np.rot90(mask, k) for k in range(4)]
    refl = [np.rot90(np.fliplr(mask), k) for k in range(4)]
    return rots, refl


def _matches(target, variants):
    for v in variants:
        if v.shape == target.shape and np.array_equal(v, target):
            return True
    return False


def _color_name(rgb):
    best, best_d = None, None
    for name, ref in _PALETTE.items():
        d = sum((int(a) - int(b)) ** 2 for a, b in zip(rgb, ref))
        if best_d is None or d < best_d:
            best, best_d = name, d
    if best_d is not None and best_d > 9000:
        return None
    return best


def decision_from_image(image):
    arr = np.asarray(image.convert("RGB")).astype(np.int16)
    h, w, _ = arr.shape
    r, g, b = arr[:, :, 0], arr[:, :, 1], arr[:, :, 2]

    mx = arr.max(axis=2)
    mn = arr.min(axis=2)
    neutral = (mx - mn) <= 30
    tile = neutral & (mn >= 80) & (mx <= 145)
    dark = neutral & (mx <= 70)

    rule_rows = np.nonzero(dark.sum(axis=1) > 0.8 * w)[0]
    if rule_rows.size == 0:
        return ABSTAIN
    rule_y = int(rule_rows.mean())

    ref_mask = _crop(tile[:rule_y, :])
    if ref_mask is None or ref_mask.size == 0:
        return ABSTAIN

    lower = tile[rule_y + 1:, :]
    col_runs = _runs(lower.any(axis=0), min_len=8)
    if len(col_runs) != 4:
        return ABSTAIN

    colored = (mx - mn) > 55
    dot_runs = _runs(colored[rule_y + 1:, :].any(axis=0), min_len=8)
    if len(dot_runs) != 4:
        return ABSTAIN

    rots, refl = _orbit(ref_mask)

    picks = []
    for idx, (c0, c1) in enumerate(col_runs):
        sub = _crop(lower[:, c0: c1 + 1])
        if sub is None:
            return ABSTAIN
        if sub.sum() != ref_mask.sum():
            continue
        if _matches(sub, rots):
            continue
        if _matches(sub, refl):
            picks.append(idx)

    if len(picks) != 1:
        return ABSTAIN
    slot = picks[0]

    d0, d1 = dot_runs[slot]
    if not (col_runs[slot][0] - 80 <= (d0 + d1) // 2 <= col_runs[slot][1] + 80):
        return ABSTAIN
    band = colored[rule_y + 1:, d0: d1 + 1]
    ys, xs = np.nonzero(band)
    if ys.size < 100:
        return ABSTAIN
    pix = arr[rule_y + 1:, d0: d1 + 1][ys, xs]
    rgb = tuple(int(np.median(pix[:, k])) for k in range(3))
    name = _color_name(rgb)
    if name is None:
        return ABSTAIN
    return "slot%d_%s" % (slot + 1, name)
