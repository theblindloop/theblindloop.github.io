"""Enantiomer selection world: pick the mirror-only twin of a reference tile.

The scene stores drawn cell occupancy for five tiles plus the dot color bound to
each candidate slot. Analytic gold separates the rotation orbit of the reference
from its reflection orbit and selects the unique candidate that is a reflection
but not a rotation.
"""

import random

from PIL import Image, ImageDraw

CELL = 20
WIDTH = 680
HEIGHT = 430
RULE_Y = 200
REF_CENTER = (340, 100)
SLOT_X = (100, 260, 420, 580)
SLOT_SHAPE_Y = 290
SLOT_DOT_Y = 385
DOT_R = 13

BG = (255, 255, 255)
TILE = (110, 110, 110)
INK = (0, 0, 0)

DOT_RGB = {
    "red": (214, 40, 40),
    "blue": (30, 80, 200),
    "green": (20, 150, 60),
    "orange": (240, 140, 20),
}
DOT_NAMES = ("red", "blue", "green", "orange")

# Three chiral pentominoes, all with a 4x2 bounding box and five cells, so that
# bounding box, cell count and total ink are constant across every tile.
BASE_SHAPES = {
    "L": ((0, 0), (1, 0), (2, 0), (3, 0), (3, 1)),
    "N": ((0, 1), (1, 1), (2, 1), (2, 0), (3, 0)),
    "Y": ((0, 1), (1, 0), (1, 1), (2, 1), (3, 1)),
}
FAMILIES = ("L", "N", "Y")


def _norm(cells):
    r0 = min(r for r, _ in cells)
    c0 = min(c for _, c in cells)
    return tuple(sorted((r - r0, c - c0) for r, c in cells))


def _rot(cells):
    """Rotate the cell set by 90 degrees clockwise."""
    return _norm([(c, -r) for r, c in cells])


def _mirror(cells):
    return _norm([(r, -c) for r, c in cells])


def _rot_k(cells, k):
    out = _norm(cells)
    for _ in range(k % 4):
        out = _rot(out)
    return out


def rotation_orbit(cells):
    return {_rot_k(cells, k) for k in range(4)}


def reflection_orbit(cells):
    return {_rot_k(_mirror(cells), k) for k in range(4)}


def _to_mask(cells, pad=6):
    grid = [[False] * pad for _ in range(pad)]
    for r, c in _norm(cells):
        grid[r][c] = True
    return grid


def cell_mismatch(a, b):
    """Minimum number of differing unit squares between tile ``a`` and tile
    ``b`` over turns in the plane and slides, with flipping forbidden.

    Flipping is excluded on purpose: it is exactly the operation the question
    forbids, so this metric measures how far a non-answer tile is from being
    the mirror twin under the rules stated to the solver.
    """
    tb = set(_norm(b))
    best = None
    for k in range(4):
        var = _rot_k(a, k)
        rows = max(r for r, _ in var) + 1
        cols = max(c for _, c in var) + 1
        for dr in range(-6, 7 - rows):
            for dc in range(-6, 7 - cols):
                placed = {(r + dr, c + dc) for r, c in var}
                diff = len(placed) + len(tb) - 2 * len(placed & tb)
                if best is None or diff < best:
                    best = diff
    return best


def _shape_variant(rng, family):
    cells = BASE_SHAPES[family]
    if rng.random() < 0.5:
        cells = _mirror(cells)
    return _rot_k(cells, rng.randrange(4))


def sample_scene(seed):
    rng = random.Random(1_000_003 + int(seed) * 7919)

    ref_family = rng.choice(FAMILIES)
    ref_cells = _shape_variant(rng, ref_family)

    answer_cells = _rot_k(_mirror(ref_cells), rng.randrange(4))

    rot_choices = [k for k in (1, 2, 3) if _rot_k(ref_cells, k) != _norm(ref_cells)]
    trap_cells = _rot_k(ref_cells, rng.choice(rot_choices))

    others = [f for f in FAMILIES if f != ref_family]
    rng.shuffle(others)
    distractors = [_shape_variant(rng, f) for f in others]

    tiles = [answer_cells, trap_cells] + distractors
    order = [0, 1, 2, 3]
    rng.shuffle(order)
    cand_cells = [tiles[i] for i in order]

    colors = list(DOT_NAMES)
    rng.shuffle(colors)

    return {
        "ref_cells": [list(c) for c in ref_cells],
        "cand_cells": [[list(c) for c in t] for t in cand_cells],
        "dot_colors": colors,
    }


def _ref(scene):
    return _norm([tuple(c) for c in scene["ref_cells"]])


def _cands(scene):
    return [_norm([tuple(c) for c in t]) for t in scene["cand_cells"]]


def _answer_slots(scene):
    ref = _ref(scene)
    rots = rotation_orbit(ref)
    refls = reflection_orbit(ref)
    return [i for i, t in enumerate(_cands(scene)) if t in refls and t not in rots]


def analytic_gold(scene):
    slots = _answer_slots(scene)
    if len(slots) != 1:
        return "abstain"
    slot = slots[0]
    return "slot%d_%s" % (slot + 1, scene["dot_colors"][slot])


def margin(scene):
    """Smallest number of unit squares by which a non-answer tile fails to be
    the mirror twin of the reference."""
    slots = _answer_slots(scene)
    if len(slots) != 1:
        return 0.0
    cands = _cands(scene)
    target = cands[slots[0]]
    worst = None
    for i, t in enumerate(cands):
        if i == slots[0]:
            continue
        d = cell_mismatch(t, target)
        if worst is None or d < worst:
            worst = d
    return float(worst)


def is_quarantined(scene):
    ref = _ref(scene)
    cands = _cands(scene)
    rots = rotation_orbit(ref)
    if len(_answer_slots(scene)) != 1:
        return True
    if not any(t in rots for t in cands):
        return True
    return margin(scene) < 2.0


def latent_symmetries(scene):
    """No latent aliases exist: every scene field is drawn (see candidate.json
    latent_alias.justification), so the declared transform list is empty."""
    return []


def _draw_tile(draw, cells, center):
    rows = max(r for r, _ in cells) + 1
    cols = max(c for _, c in cells) + 1
    x0 = center[0] - cols * CELL // 2
    y0 = center[1] - rows * CELL // 2
    for r, c in cells:
        x = x0 + c * CELL
        y = y0 + r * CELL
        draw.rectangle([x, y, x + CELL - 1, y + CELL - 1], fill=TILE)


def render(scene):
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)

    _draw_tile(draw, _ref(scene), REF_CENTER)

    draw.rectangle([0, RULE_Y, WIDTH - 1, RULE_Y + 2], fill=INK)

    for i, cells in enumerate(_cands(scene)):
        cx = SLOT_X[i]
        _draw_tile(draw, cells, (cx, SLOT_SHAPE_Y))
        color = DOT_RGB[scene["dot_colors"][i]]
        draw.ellipse(
            [cx - DOT_R, SLOT_DOT_Y - DOT_R, cx + DOT_R, SLOT_DOT_Y + DOT_R],
            fill=color,
        )
    return img
