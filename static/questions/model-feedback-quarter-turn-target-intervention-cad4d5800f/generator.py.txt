"""Deterministic renderer and analytic arm for the quarter-turn intervention world."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 960
HEIGHT = 700
PANEL_RECTS = (
    (35, 100, 455, 375),
    (505, 100, 925, 375),
    (35, 400, 455, 675),
    (505, 400, 925, 675),
)
HUB_OFFSET = (210, 140)
CUE_OFFSET = (365, 45)
CUE_RADIUS = 25
POINTER_INNER = 17
POINTER_WIDTH = 9
TARGET_RADIUS = 20.0
QUARANTINE_MARGIN = 6.0

BACKGROUND = (238, 242, 246)
PANEL_FILL = (250, 251, 253)
FRAME = (112, 126, 140)
FRAME_ACCENT = (205, 213, 220)
POINTER = (24, 35, 48)
HUB = (24, 168, 183)
TURN_CUE = (43, 111, 184)
TARGET = (207, 57, 68)
WHITE = (255, 255, 255)
LABEL = (52, 67, 82)

PANEL_LABELS = ("A", "B", "C", "D")
_BITMAPS = {
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "B": ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    "C": ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    "D": ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
}


def _point(center: tuple[float, float], radius: float, angle_deg: float) -> tuple[float, float]:
    angle = math.radians(angle_deg)
    return center[0] + radius * math.cos(angle), center[1] + radius * math.sin(angle)


def _hub_for(index: int) -> tuple[float, float]:
    rect = PANEL_RECTS[index]
    return float(rect[0] + HUB_OFFSET[0]), float(rect[1] + HUB_OFFSET[1])


def _cue_center(index: int) -> tuple[float, float]:
    rect = PANEL_RECTS[index]
    return float(rect[0] + CUE_OFFSET[0]), float(rect[1] + CUE_OFFSET[1])


def _post_angle(panel: dict) -> float:
    return float(panel["angle_deg"]) + (90.0 if int(panel["direction"]) > 0 else -90.0)


def _score(panel: dict) -> float:
    """Signed distance from the counterfactual endpoint to the target disk edge."""

    hub = (float(panel["hub_x"]), float(panel["hub_y"]))
    endpoint = _point(hub, float(panel["length"]), _post_angle(panel))
    target = (float(panel["target_x"]), float(panel["target_y"]))
    distance = math.hypot(endpoint[0] - target[0], endpoint[1] - target[1])
    return distance - float(panel["target_radius"])


def sample_scene(seed: int) -> dict:
    """Sample four clear panels with exactly one post-intervention bullseye hit."""

    rng = random.Random(int(seed))
    answer_index = int(seed) % 4
    panels = []
    offsets = (-62.0, 54.0, -50.0, 63.0)
    for index in range(4):
        hub_x, hub_y = _hub_for(index)
        angle = rng.uniform(-165.0, 165.0)
        length = rng.uniform(100.0, 112.0)
        direction = 1 if rng.randrange(2) == 0 else -1
        post = angle + (90.0 if direction > 0 else -90.0)
        target_angle = post if index == answer_index else post + offsets[index]
        target_x, target_y = _point((hub_x, hub_y), length, target_angle)
        panels.append(
            {
                "hub_x": round(hub_x, 3),
                "hub_y": round(hub_y, 3),
                "angle_deg": round(angle, 6),
                "length": round(length, 6),
                "direction": direction,
                "target_angle_deg": round(target_angle, 6),
                "target_x": round(target_x, 3),
                "target_y": round(target_y, 3),
                "target_radius": TARGET_RADIUS,
            }
        )
    return {
        "canvas_width": WIDTH,
        "canvas_height": HEIGHT,
        "panels": panels,
    }


def _draw_bitmap(draw: ImageDraw.ImageDraw, glyph: str, xy: tuple[int, int], scale: int = 4) -> None:
    bitmap = _BITMAPS[glyph]
    x0, y0 = xy
    for row, bits in enumerate(bitmap):
        for col, bit in enumerate(bits):
            if bit == "1":
                draw.rectangle(
                    (x0 + col * scale, y0 + row * scale,
                     x0 + (col + 1) * scale - 1, y0 + (row + 1) * scale - 1),
                    fill=LABEL,
                )


def _draw_arrowhead(
    draw: ImageDraw.ImageDraw,
    tip: tuple[float, float],
    direction: tuple[float, float],
    color: tuple[int, int, int],
    length: float,
    width: float,
) -> None:
    dx, dy = direction
    norm = math.hypot(dx, dy) or 1.0
    ux, uy = dx / norm, dy / norm
    px, py = -uy, ux
    back = (tip[0] - ux * length, tip[1] - uy * length)
    left = (back[0] + px * width, back[1] + py * width)
    right = (back[0] - px * width, back[1] - py * width)
    draw.polygon((tip, left, right), fill=color)


def _draw_pointer(draw: ImageDraw.ImageDraw, hub: tuple[float, float], panel: dict) -> None:
    angle = math.radians(float(panel["angle_deg"]))
    ux, uy = math.cos(angle), math.sin(angle)
    start = (hub[0] + POINTER_INNER * ux, hub[1] + POINTER_INNER * uy)
    end = _point(hub, float(panel["length"]), float(panel["angle_deg"]))
    draw.line((start, end), fill=POINTER, width=POINTER_WIDTH)
    _draw_arrowhead(draw, end, (ux, uy), POINTER, length=17.0, width=9.0)


def _draw_turn_cue(draw: ImageDraw.ImageDraw, center: tuple[float, float], direction: int) -> None:
    """Draw a blue quarter-circle arrow; its top/bottom endpoint shows the sign."""

    start = 180.0
    end = 270.0 if direction > 0 else 90.0
    steps = 18
    points = [_point(center, CUE_RADIUS, start + (end - start) * i / steps) for i in range(steps + 1)]
    draw.line(points, fill=TURN_CUE, width=7, joint="curve")
    last_angle = math.radians(end)
    if direction > 0:
        tangent = (-math.sin(last_angle), math.cos(last_angle))
    else:
        tangent = (math.sin(last_angle), -math.cos(last_angle))
    _draw_arrowhead(draw, points[-1], tangent, TURN_CUE, length=13.0, width=7.0)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    # A quiet top rule establishes the two-by-two reading order without adding
    # any color used by the inverse arm.
    draw.rectangle((35, 42, 925, 45), fill=FRAME_ACCENT)
    for index, panel in enumerate(scene["panels"]):
        x0, y0, x1, y1 = PANEL_RECTS[index]
        draw.rounded_rectangle((x0, y0, x1, y1), radius=12, fill=PANEL_FILL, outline=FRAME, width=3)
        draw.rectangle((x0 + 14, y0 + 14, x0 + 44, y0 + 44), fill=WHITE, outline=FRAME_ACCENT, width=1)
        _draw_bitmap(draw, PANEL_LABELS[index], (x0 + 19, y0 + 17), scale=4)

        hub = (float(panel["hub_x"]), float(panel["hub_y"]))
        _draw_turn_cue(draw, _cue_center(index), int(panel["direction"]))
        target = (float(panel["target_x"]), float(panel["target_y"]))
        radius = float(panel["target_radius"])
        draw.ellipse(
            (round(target[0] - radius), round(target[1] - radius),
             round(target[0] + radius), round(target[1] + radius)),
            fill=TARGET,
        )
        draw.ellipse(
            (round(target[0] - 5), round(target[1] - 5),
             round(target[0] + 5), round(target[1] + 5)),
            fill=WHITE,
        )
        _draw_pointer(draw, hub, panel)
        draw.ellipse(
            (round(hub[0] - 13), round(hub[1] - 13),
             round(hub[0] + 13), round(hub[1] + 13)),
            fill=HUB,
        )
        draw.ellipse(
            (round(hub[0] - 3), round(hub[1] - 3),
             round(hub[0] + 3), round(hub[1] + 3)),
            fill=WHITE,
        )
    return image


def analytic_gold(scene: dict) -> str:
    scores = []
    for panel in scene["panels"]:
        # Keep the decision computation explicit and scene-keyed for the
        # rendered-field audit; each panel's fields are rendered above.
        hub = (float(panel["hub_x"]), float(panel["hub_y"]))
        angle = float(panel["angle_deg"]) + (90.0 if int(panel["direction"]) > 0 else -90.0)
        endpoint = _point(hub, float(panel["length"]), angle)
        target = (float(panel["target_x"]), float(panel["target_y"]))
        scores.append(
            math.hypot(endpoint[0] - target[0], endpoint[1] - target[1])
            - float(panel["target_radius"])
        )
    inside = [score <= 0.0 for score in scores]
    if sum(inside) == 1:
        return f"panel-{chr(65 + inside.index(True))}"
    return f"panel-{chr(65 + min(range(4), key=lambda idx: scores[idx]))}"


def margin(scene: dict) -> float:
    scores = []
    for panel in scene["panels"]:
        hub = (float(panel["hub_x"]), float(panel["hub_y"]))
        angle = float(panel["angle_deg"]) + (90.0 if int(panel["direction"]) > 0 else -90.0)
        endpoint = _point(hub, float(panel["length"]), angle)
        target = (float(panel["target_x"]), float(panel["target_y"]))
        scores.append(
            math.hypot(endpoint[0] - target[0], endpoint[1] - target[1])
            - float(panel["target_radius"])
        )
    return float(min(abs(score) for score in scores))


def is_quarantined(scene: dict) -> bool:
    scores = []
    for panel in scene["panels"]:
        hub = (float(panel["hub_x"]), float(panel["hub_y"]))
        angle = float(panel["angle_deg"]) + (90.0 if int(panel["direction"]) > 0 else -90.0)
        endpoint = _point(hub, float(panel["length"]), angle)
        target = (float(panel["target_x"]), float(panel["target_y"]))
        scores.append(
            math.hypot(endpoint[0] - target[0], endpoint[1] - target[1])
            - float(panel["target_radius"])
        )
    return sum(score <= 0.0 for score in scores) != 1 or min(abs(score) for score in scores) < QUARANTINE_MARGIN
