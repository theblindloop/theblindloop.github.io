"""Independent pixel-only inverse for the quarter-turn pointer world."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

WIDTH = 960
HEIGHT = 700
PANEL_RECTS = (
    (35, 100, 455, 375),
    (505, 100, 925, 375),
    (35, 400, 455, 675),
    (505, 400, 925, 675),
)
HUB_OFF_X = 210
HUB_OFF_Y = 140
CUE_OFF_X = 365
CUE_OFF_Y = 45
TARGET_EDGE_RADIUS = 20.0
POINTER_COLOR = np.array([24, 35, 48], dtype=np.uint8)
HUB_COLOR = np.array([24, 168, 183], dtype=np.uint8)
CUE_COLOR = np.array([43, 111, 184], dtype=np.uint8)
TARGET_COLOR = np.array([207, 57, 68], dtype=np.uint8)


def _mask(arr: np.ndarray, color: np.ndarray) -> np.ndarray:
    return np.all(arr == color, axis=2)


def _centroid(points: np.ndarray) -> tuple[float, float]:
    if len(points) == 0:
        raise ValueError("missing colored component")
    # points are in (row, column) order.
    return float(points[:, 1].mean()), float(points[:, 0].mean())


def _direction_from_pointer(points: np.ndarray, hub: tuple[float, float]) -> tuple[float, float, float]:
    hx, hy = hub
    xy = np.column_stack((points[:, 1], points[:, 0])).astype(float)
    relative = xy - np.array([hx, hy])
    radial = np.linalg.norm(relative, axis=1)
    keep = (radial > 12.0) & (radial < 155.0)
    relative = relative[keep]
    if len(relative) < 20:
        raise ValueError("pointer mask is too small")
    covariance = np.cov(relative, rowvar=False)
    eigenvalues, eigenvectors = np.linalg.eigh(covariance)
    vector = eigenvectors[:, int(np.argmax(eigenvalues))]
    if float(np.dot(vector, relative.mean(axis=0))) < 0:
        vector = -vector
    vector = vector / (np.linalg.norm(vector) or 1.0)
    return float(vector[0]), float(vector[1]), float(np.linalg.norm(relative.mean(axis=0)))


def _panel_crop(mask: np.ndarray, rect: tuple[int, int, int, int]) -> np.ndarray:
    x0, y0, x1, y1 = rect
    return mask[y0:y1 + 1, x0:x1 + 1]


def _recover_panel(arr: np.ndarray, index: int) -> float:
    rect = PANEL_RECTS[index]
    x0, y0, x1, y1 = rect
    hub = (float(x0 + HUB_OFF_X), float(y0 + HUB_OFF_Y))

    pointer_points = np.argwhere(_panel_crop(_mask(arr, POINTER_COLOR), rect))
    if len(pointer_points) == 0:
        raise ValueError(f"panel {index} has no pointer")
    pointer_points[:, 0] += y0
    pointer_points[:, 1] += x0
    ux, uy, _ = _direction_from_pointer(pointer_points, hub)
    pointer_angle = math.atan2(uy, ux)

    cue_points = np.argwhere(_panel_crop(_mask(arr, CUE_COLOR), rect))
    if len(cue_points) < 20:
        raise ValueError(f"panel {index} has no turn cue")
    cue_y = y0 + CUE_OFF_Y
    cue_absolute_y = cue_points[:, 0].astype(float) + y0
    # The clockwise icon ends above its center; the counterclockwise icon
    # ends below it. The arc is large enough that this sign is stable under
    # ordinary pixel corruption but not dependent on any latent metadata.
    turn_sign = 1.0 if float(cue_absolute_y.min()) < cue_y - 10.0 else -1.0

    red_points = np.argwhere(_panel_crop(_mask(arr, TARGET_COLOR), rect))
    if len(red_points) < 100:
        raise ValueError(f"panel {index} has no bullseye")
    red_points[:, 0] += y0
    red_points[:, 1] += x0
    target_x, target_y = _centroid(red_points)

    # The target's observed radial distance supplies the visible pointer
    # length. Only its angle is counterfactually changed.
    dx = target_x - hub[0]
    dy = target_y - hub[1]
    target_radius = math.hypot(dx, dy)
    post_angle = pointer_angle + turn_sign * math.pi / 2.0
    predicted_x = hub[0] + target_radius * math.cos(post_angle)
    predicted_y = hub[1] + target_radius * math.sin(post_angle)
    return math.hypot(predicted_x - target_x, predicted_y - target_y) - TARGET_EDGE_RADIUS


def decision_from_image(image: Image.Image) -> str:
    """Return the panel whose visible, imagined quarter-turn enters its target."""

    rgb = image.convert("RGB")
    if rgb.size != (WIDTH, HEIGHT):
        return "ambiguous"
    arr = np.asarray(rgb)
    scores = [_recover_panel(arr, index) for index in range(4)]
    inside = [score <= 0.0 for score in scores]
    if sum(inside) != 1:
        return "ambiguous"
    return f"panel-{chr(65 + inside.index(True))}"
