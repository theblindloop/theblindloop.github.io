"""Deterministic analytic and visual world for Rim-Glued Escape."""

from __future__ import annotations

from collections import Counter, deque
import copy
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1200, 800
N, CELL = 7, 42
BACKGROUND = (245, 242, 235)
CARD = (255, 255, 252)
INK = (42, 47, 57)
MUTED = (84, 91, 104)
GRID = (203, 209, 216)
CARD_EDGE = (91, 100, 114)
DECISIONS = ("A", "B", "C", "D")
TERMINALS = {"A": (0, 0), "B": (0, 6), "C": (6, 0), "D": (6, 6)}
START = (3, 3)
PORTAL_CELLS = {"P": ((1, 2), (5, 4)), "Q": ((1, 4), (5, 2))}
DIRECTIONS = ((-1, 0), (0, 1), (1, 0), (0, -1))  # N,E,S,W
RIM_NAMES = ("red", "gold", "green", "blue")
RIM_COLORS = {
    "red": (224, 68, 75), "gold": (244, 177, 43),
    "green": (50, 166, 101), "blue": (54, 122, 214),
}
PORTAL_COLORS = {"P": (25, 151, 171), "Q": (146, 79, 190)}
MARK_COLORS = {
    "S": (215, 70, 65), "A": (247, 174, 47), "B": (45, 119, 204),
    "C": (76, 166, 88), "D": (150, 82, 191),
}
PANEL_BOXES = ((20, 181, 386, 755), (417, 181, 783, 755), (814, 181, 1180, 755))
GRID_OFFSET = (36, 137)
FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"


def _font(size: int):
    return ImageFont.truetype(FONT_PATH, size)


def _canon(edge):
    return tuple(sorted((tuple(edge[0]), tuple(edge[1]))))


def _all_edges():
    blocked = {cell for cells in PORTAL_CELLS.values() for cell in cells}
    out = []
    for r in range(N):
        for c in range(N):
            if (r, c) in blocked:
                continue
            for dr, dc in ((0, 1), (1, 0)):
                nxt = (r + dr, c + dc)
                if 0 <= nxt[0] < N and 0 <= nxt[1] < N and nxt not in blocked:
                    out.append(((r, c), nxt))
    return out


ALL_EDGES = _all_edges()


def _odd_walls(plans):
    counts = Counter(_canon(edge) for plan in plans for edge in plan)
    return {edge for edge, count in counts.items() if count % 2}


def _portal_map(portals):
    """Return seam edges between ordinary cells for equal-colored paired rims."""
    links = []
    for symbol, entries in portals.items():
        if len(entries) != 2:
            continue
        by_tag = []
        for entry in entries:
            cell = tuple(entry["cell"])
            sides = {}
            for index, tag in enumerate(entry["rims"]):
                dr, dc = DIRECTIONS[index]
                sides[tag] = (cell[0] + dr, cell[1] + dc)
            by_tag.append(sides)
        for tag in RIM_NAMES:
            links.append((by_tag[0][tag], by_tag[1][tag]))
    return links


def _reachable(walls, portals=None, hub=False):
    blocked = {cell for cells in PORTAL_CELLS.values() for cell in cells}
    graph = {cell: set() for r in range(N) for c in range(N)
             if (cell := (r, c)) not in blocked}
    for edge in ALL_EDGES:
        edge = _canon(edge)
        if edge not in walls:
            a, b = edge
            graph[a].add(b)
            graph[b].add(a)
    if portals:
        if hub:
            for entries in portals.values():
                mouths = []
                for entry in entries:
                    r, c = entry["cell"]
                    mouths.extend((r + dr, c + dc) for dr, dc in DIRECTIONS)
                mouths = [cell for cell in mouths if cell in graph]
                for a in mouths:
                    graph[a].update(b for b in mouths if b != a)
        else:
            for a, b in _portal_map(portals):
                if a in graph and b in graph:
                    graph[a].add(b)
                    graph[b].add(a)
    seen = {START}
    queue = deque([START])
    while queue:
        cell = queue.popleft()
        for nxt in graph[cell]:
            if nxt not in seen:
                seen.add(nxt)
                queue.append(nxt)
    return seen


def _partition(rng):
    blocked = {cell for cells in PORTAL_CELLS.values() for cell in cells}
    seeds = {START: "S", **{cell: label for label, cell in TERMINALS.items()},
             (0, 3): "X", (6, 3): "Y", (3, 0): "Z", (3, 6): "W"}
    owner = dict(seeds)
    while len(owner) < N * N - len(blocked):
        frontier = []
        for cell, label in owner.items():
            r, c = cell
            for dr, dc in DIRECTIONS:
                nxt = (r + dr, c + dc)
                if (0 <= nxt[0] < N and 0 <= nxt[1] < N and
                        nxt not in blocked and nxt not in owner):
                    frontier.append((nxt, label))
        if not frontier:
            raise RuntimeError("partition stalled")
        nxt, label = rng.choice(frontier)
        owner[nxt] = label
    return owner


def _hits(seen):
    return [label for label, cell in TERMINALS.items() if cell in seen]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = DECISIONS[int(seed) % 4]
    for _ in range(6000):
        owner = _partition(rng)
        final_walls = [_canon(edge) for edge in ALL_EDGES if owner[edge[0]] != owner[edge[1]]]
        if not (24 <= len(final_walls) <= 44):
            continue
        portals = {}
        for symbol, cells in PORTAL_CELLS.items():
            entries = []
            for cell in cells:
                rims = list(RIM_NAMES)
                rng.shuffle(rims)
                entries.append({"cell": list(cell), "rims": rims})
            portals[symbol] = entries
        walls = set(final_walls)
        if _hits(_reachable(walls)):
            continue
        exact_hits = _hits(_reachable(walls, portals))
        hub_hits = _hits(_reachable(walls, portals, hub=True))
        if exact_hits != [target] or len(hub_hits) < 2 or target not in hub_hits:
            continue
        nonwalls = [_canon(edge) for edge in ALL_EDGES if edge not in walls]
        rng.shuffle(final_walls)
        rng.shuffle(nonwalls)
        decoys = nonwalls[:18]
        plans = [[], [], []]
        for i, edge in enumerate(final_walls):
            plans[i % 3].append(edge)
        pairs = [(0, 1), (0, 2), (1, 2)]
        rng.shuffle(pairs)
        for i, edge in enumerate(decoys):
            for panel in pairs[i % 3]:
                plans[panel].append(edge)
        rng.shuffle(plans)
        encoded = []
        for plan in plans:
            rng.shuffle(plan)
            encoded.append([[list(edge[1]), list(edge[0])] if rng.getrandbits(1)
                            else [list(edge[0]), list(edge[1])] for edge in plan])
        scene = {"plans": encoded, "portals": portals}
        if not is_quarantined(scene):
            return scene
    raise RuntimeError(f"could not sample accepted scene for seed {seed}")


def analytic_gold(scene):
    hits = _hits(_reachable(_odd_walls(scene["plans"]), scene["portals"]))
    return hits[0] if len(hits) == 1 else "abstain"


def margin(scene):
    return 1.0 if analytic_gold(scene) in DECISIONS else 0.0


def is_quarantined(scene):
    try:
        if len(scene["plans"]) != 3 or set(scene["portals"]) != set(PORTAL_CELLS):
            return True
        occurrences = [_canon(edge) for plan in scene["plans"] for edge in plan]
        if any(edge not in ALL_EDGES for edge in occurrences):
            return True
        counts = Counter(occurrences)
        if any(count not in (1, 2) for count in counts.values()):
            return True
        if sum(count == 2 for count in counts.values()) < 14:
            return True
        for symbol, entries in scene["portals"].items():
            if len(entries) != 2:
                return True
            if tuple(tuple(e["cell"]) for e in entries) != PORTAL_CELLS[symbol]:
                return True
            if any(sorted(e["rims"]) != sorted(RIM_NAMES) for e in entries):
                return True
        walls = _odd_walls(scene["plans"])
        if _hits(_reachable(walls)):
            return True
        exact = _hits(_reachable(walls, scene["portals"]))
        hub = _hits(_reachable(walls, scene["portals"], hub=True))
        return len(exact) != 1 or len(hub) < 2 or exact[0] not in hub
    except Exception:
        return True


def latent_symmetries(scene):
    transformed = copy.deepcopy(scene)
    transformed["plans"] = [[[edge[1], edge[0]] for edge in plan]
                            for plan in transformed["plans"]]
    return [("reverse_all_stored_wall_endpoints", transformed)]


def _wall_line(edge, gx, gy):
    (r1, c1), (r2, c2) = _canon(edge)
    if r1 == r2:
        x = gx + max(c1, c2) * CELL
        return (x, gy + r1 * CELL, x, gy + (r1 + 1) * CELL)
    y = gy + max(r1, r2) * CELL
    return (gx + c1 * CELL, y, gx + (c1 + 1) * CELL, y)


def _draw_portal(draw, cx, cy, symbol, rims):
    draw.rounded_rectangle((cx - 18, cy - 18, cx + 18, cy + 18), radius=5,
                           fill=(236, 238, 241), outline=INK, width=2)
    boxes = ((cx - 8, cy - 18, cx + 8, cy - 10),
             (cx + 10, cy - 8, cx + 18, cy + 8),
             (cx - 8, cy + 10, cx + 8, cy + 18),
             (cx - 18, cy - 8, cx - 10, cy + 8))
    for tag, box in zip(rims, boxes):
        draw.rectangle(box, fill=RIM_COLORS[tag], outline=INK, width=1)
    draw.ellipse((cx - 8, cy - 8, cx + 8, cy + 8), fill=PORTAL_COLORS[symbol], outline=INK, width=1)
    draw.text((cx, cy), symbol, fill=(255, 255, 255), font=_font(11), anchor="mm")


def render(scene):
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((WIDTH // 2, 24), "RIM-GLUED ESCAPE", fill=INK, font=_font(30), anchor="mm")
    draw.text((WIDTH // 2, 61), "same grid edge = same wall", fill=MUTED, font=_font(18), anchor="mm")
    draw.rounded_rectangle((99, 87, 1101, 151), radius=18, fill=(235, 238, 242), outline=CARD_EDGE, width=3)
    draw.text((WIDTH // 2, 108), "shown once = WALL     shown twice = OPEN", fill=INK, font=_font(19), anchor="mm")
    draw.text((WIDTH // 2, 132), "same LETTER + same RIM COLOR = JOIN     •     NO TURN inside a hole",
              fill=INK, font=_font(18), anchor="mm")
    for panel_index, box in enumerate(PANEL_BOXES):
        x0, y0, x1, y1 = box
        draw.rounded_rectangle(box, radius=18, fill=CARD, outline=CARD_EDGE, width=4)
        draw.text(((x0 + x1) // 2, y0 + 35), f"PLAN {panel_index + 1}", fill=INK, font=_font(22), anchor="mm")
        gx, gy = x0 + GRID_OFFSET[0], y0 + GRID_OFFSET[1]
        for k in range(N + 1):
            draw.line((gx + k * CELL, gy, gx + k * CELL, gy + N * CELL), fill=GRID, width=2)
            draw.line((gx, gy + k * CELL, gx + N * CELL, gy + k * CELL), fill=GRID, width=2)
        draw.rectangle((gx, gy, gx + N * CELL, gy + N * CELL), outline=CARD_EDGE, width=4)
        for edge in scene["plans"][panel_index]:
            draw.line(_wall_line(edge, gx, gy), fill=INK, width=10)
        for symbol, entries in scene["portals"].items():
            for entry in entries:
                r, c = entry["cell"]
                _draw_portal(draw, gx + c * CELL + CELL // 2, gy + r * CELL + CELL // 2,
                             symbol, entry["rims"])
        for label, (r, c) in {"S": START, **TERMINALS}.items():
            cx, cy = gx + c * CELL + CELL // 2, gy + r * CELL + CELL // 2
            draw.ellipse((cx - 15, cy - 15, cx + 15, cy + 15), fill=MARK_COLORS[label], outline=INK, width=2)
            fill = (255, 255, 255) if label in {"S", "B", "D"} else (24, 27, 33)
            draw.text((cx, cy), label, fill=fill, font=_font(16), anchor="mm")
        draw.text(((x0 + x1) // 2, y1 - 22), "S = START    colored bars are rims", fill=MUTED, font=_font(14), anchor="mm")
    return image
