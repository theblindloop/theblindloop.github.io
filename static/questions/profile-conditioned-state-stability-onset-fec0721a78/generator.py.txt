"""Deterministic renderer and analytic side for state-stability onset scenes."""

from __future__ import annotations

import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont


LABELS = ("A", "B", "C", "D")
STATE_LABELS = ("CALM", "ACTIVE", "PAUSED")
CANVAS = (1800, 960)

BACKGROUND = (247, 249, 252)
INK = (35, 45, 58)
MUTED = (91, 104, 119)
GRID = (185, 195, 207)
CARD = (255, 255, 255)
RING = (221, 107, 45)

IDENTITY_BASE_COLORS = (
    (207, 73, 88),
    (38, 126, 180),
    (111, 82, 169),
    (43, 148, 105),
)
STATE_BASE_COLORS = (
    (226, 124, 54),
    (40, 145, 185),
    (157, 89, 163),
)
PANEL_TINTS = (
    (255, 255, 255),
    (247, 251, 254),
    (253, 249, 244),
)

PANEL_LEFTS = (55, 625, 1195)
PANEL_TOPS = (255, 605)
PANEL_W = 520
PANEL_H = 300
TILE_W = 220
TILE_H = 80
IDENTITY_STRIP_W = 52
SLOT_OFFSETS = ((22, 70), (266, 70), (22, 174), (266, 174))

IDENTITY_KEY_X = 70
IDENTITY_KEY_Y = 132
IDENTITY_KEY_STEP = 118
STATE_KEY_X = 625
STATE_KEY_Y = 132
STATE_KEY_STEP = 185
KEY_RADIUS = 14

_FONT_PATHS = (
    "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
    "/usr/share/fonts/dejavu/DejaVuSans.ttf",
)


def _font(size: int):
    for candidate in _FONT_PATHS:
        if Path(candidate).is_file():
            return ImageFont.truetype(candidate, size=size)
    return ImageFont.load_default()


def _centered_text(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font, fill):
    box = draw.textbbox((0, 0), text, font=font)
    width = box[2] - box[0]
    height = box[3] - box[1]
    draw.text(
        (xy[0] - width / 2, xy[1] - height / 2 - box[1]),
        text,
        font=font,
        fill=fill,
    )


def _identity_color(scene: dict, label: int) -> tuple[int, int, int]:
    return IDENTITY_BASE_COLORS[int(scene["label_palette_order"][int(label)])]


def _state_color(scene: dict, state: int) -> tuple[int, int, int]:
    return STATE_BASE_COLORS[int(scene["state_palette_order"][int(state)])]


def _onsets(states: list[list[int]]) -> list[int]:
    if len(states) != 6 or any(len(row) != 4 for row in states):
        raise ValueError("states must contain six panels and four labels")
    onsets: list[int] = []
    for label in range(4):
        onset = None
        for panel in range(6):
            if all(int(states[later][label]) == int(states[panel][label]) for later in range(panel, 6)):
                onset = panel
                break
        if onset is None:
            raise ValueError("every trajectory must have a final unchanged suffix")
        onsets.append(int(onset))
    return onsets


def sample_scene(seed: int) -> dict:
    """Sample balanced trajectories and independent visible style nuisances."""

    logic_rng = random.Random(int(seed) * 1000003 + 17)
    style_rng = random.Random(int(seed) * 1000033 + 31)
    desired_winner = int(seed) % 4
    winner_onset = 1 + logic_rng.randrange(3)
    desired_onsets = [winner_onset]
    desired_onsets.extend(
        [
            winner_onset + 1,
            min(winner_onset + 2, 5),
            5,
        ]
    )
    onset_by_label = [0] * 4
    other_labels = [label for label in range(4) if label != desired_winner]
    logic_rng.shuffle(other_labels)
    onset_by_label[desired_winner] = winner_onset
    for label, onset in zip(other_labels, desired_onsets[1:]):
        onset_by_label[label] = onset

    states = [[0 for _ in range(4)] for _ in range(6)]
    for label, onset in enumerate(onset_by_label):
        final_state = logic_rng.randrange(3)
        trajectory = [logic_rng.randrange(3) for _ in range(onset)]
        if onset > 0 and trajectory[-1] == final_state:
            trajectory[-1] = (final_state + 1 + logic_rng.randrange(2)) % 3
        trajectory.append(final_state)
        trajectory.extend([final_state] * (5 - onset))
        for panel, state in enumerate(trajectory):
            states[panel][label] = int(state)

    label_palette_order = list(range(4))
    state_palette_order = list(range(3))
    style_rng.shuffle(label_palette_order)
    style_rng.shuffle(state_palette_order)
    slot_orders = []
    for _ in range(6):
        order = list(range(4))
        style_rng.shuffle(order)
        slot_orders.append(order)
    panel_tints = [style_rng.randrange(len(PANEL_TINTS)) for _ in range(6)]
    glyph_radius = 7 if int(seed) % 11 == 0 else 23
    storage_order = list(range(4))
    style_rng.shuffle(storage_order)
    scene = {
        "canvas": list(CANVAS),
        "states": states,
        "label_palette_order": label_palette_order,
        "state_palette_order": state_palette_order,
        "slot_orders": slot_orders,
        "panel_tints": panel_tints,
        "glyph_radius": glyph_radius,
        "storage_order": storage_order,
    }
    computed = _onsets(states)
    if computed.count(min(computed)) != 1 or LABELS[computed.index(min(computed))] != LABELS[desired_winner]:
        raise AssertionError("sampler failed to create the requested unique winner")
    return scene


def _draw_state_shape(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    state: int,
    color: tuple[int, int, int],
    radius: int,
) -> None:
    cx, cy = center
    radius = max(2, int(radius))
    if int(state) == 0:
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=color, outline=INK, width=2)
    elif int(state) == 1:
        draw.rounded_rectangle(
            (cx - radius, cy - radius, cx + radius, cy + radius),
            radius=max(2, radius // 4),
            fill=color,
            outline=INK,
            width=2,
        )
    else:
        draw.polygon(
            ((cx, cy - radius), (cx + radius, cy), (cx, cy + radius), (cx - radius, cy)),
            fill=color,
            outline=INK,
        )


def _draw_identity_key(draw: ImageDraw.ImageDraw, scene: dict, label_font, small_font) -> None:
    draw.text((IDENTITY_KEY_X, IDENTITY_KEY_Y - 34), "IDENTITY KEY", font=small_font, fill=MUTED)
    for label, name in enumerate(LABELS):
        cx = IDENTITY_KEY_X + label * IDENTITY_KEY_STEP + 20
        cy = IDENTITY_KEY_Y + 16
        draw.rectangle((cx - 14, cy - 14, cx + 14, cy + 14), fill=_identity_color(scene, label), outline=INK, width=1)
        _centered_text(draw, (cx + 44, cy), name, label_font, INK)


def _draw_state_key(draw: ImageDraw.ImageDraw, scene: dict, label_font, small_font) -> None:
    draw.text((STATE_KEY_X, STATE_KEY_Y - 34), "STATE KEY", font=small_font, fill=MUTED)
    for state, name in enumerate(STATE_LABELS):
        cx = STATE_KEY_X + state * STATE_KEY_STEP + 18
        cy = STATE_KEY_Y + 16
        _draw_state_shape(draw, (cx, cy), state, _state_color(scene, state), KEY_RADIUS)
        _centered_text(draw, (cx + 72, cy), name, label_font, INK)


def _draw_tile(
    draw: ImageDraw.ImageDraw,
    scene: dict,
    left: int,
    top: int,
    label: int,
    state: int,
    label_font,
    state_font,
) -> None:
    draw.rounded_rectangle(
        (left, top, left + TILE_W, top + TILE_H),
        radius=8,
        fill=CARD,
        outline=GRID,
        width=2,
    )
    identity_color = _identity_color(scene, label)
    draw.rounded_rectangle(
        (left, top, left + IDENTITY_STRIP_W + 4, top + TILE_H),
        radius=8,
        fill=identity_color,
        outline=INK,
        width=1,
    )
    draw.rectangle((left + IDENTITY_STRIP_W - 3, top + 1, left + IDENTITY_STRIP_W + 4, top + TILE_H - 1), fill=identity_color)
    _centered_text(draw, (left + IDENTITY_STRIP_W / 2, top + TILE_H / 2), LABELS[label], label_font, (255, 255, 255))
    _draw_state_shape(
        draw,
        (left + 145, top + TILE_H / 2),
        state,
        _state_color(scene, state),
        int(scene["glyph_radius"]),
    )
    _centered_text(draw, (left + 190, top + TILE_H / 2), STATE_LABELS[state], state_font, MUTED)


def _draw_arrow(draw: ImageDraw.ImageDraw, start: tuple[int, int], end: tuple[int, int]) -> None:
    draw.line((*start, *end), fill=MUTED, width=3)
    dx = end[0] - start[0]
    dy = end[1] - start[1]
    if abs(dx) >= abs(dy):
        direction = 1 if dx >= 0 else -1
        x, y = end
        draw.polygon(((x, y), (x - 10 * direction, y - 6), (x - 10 * direction, y + 6)), fill=MUTED)
    else:
        direction = 1 if dy >= 0 else -1
        x, y = end
        draw.polygon(((x, y), (x - 6, y - 10 * direction), (x + 6, y - 10 * direction)), fill=MUTED)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", tuple(scene["canvas"]), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(25)
    section_font = _font(18)
    label_font = _font(18)
    state_font = _font(13)
    small_font = _font(14)

    draw.text((58, 28), "STATE STABILITY ONSET", font=title_font, fill=INK)
    draw.text(
        (58, 72),
        "Follow each identity through the numbered panels; the final state is not printed as a summary.",
        font=small_font,
        fill=MUTED,
    )
    _draw_identity_key(draw, scene, label_font, small_font)
    _draw_state_key(draw, scene, label_font, small_font)
    draw.text((1240, 113), "PANEL ORDER", font=small_font, fill=MUTED)
    draw.text((1240, 140), "P1 → P2 → P3 → P4 → P5 → P6", font=section_font, fill=INK)
    draw.text((1240, 176), "A tile is permanent only when every later panel keeps its state.", font=small_font, fill=MUTED)

    for panel in range(6):
        row = panel // 3
        column = panel % 3
        left = PANEL_LEFTS[column]
        top = PANEL_TOPS[row]
        draw.rounded_rectangle(
            (left, top, left + PANEL_W, top + PANEL_H),
            radius=12,
            fill=PANEL_TINTS[int(scene["panel_tints"][panel])],
            outline=INK,
            width=2,
        )
        _centered_text(draw, (left + PANEL_W / 2, top + 26), f"P{panel + 1}", section_font, INK)
        draw.line((left + 20, top + 49, left + PANEL_W - 20, top + 49), fill=GRID, width=1)
        for slot, (dx, dy) in enumerate(SLOT_OFFSETS):
            label = int(scene["slot_orders"][panel][slot])
            _draw_tile(
                draw,
                scene,
                left + dx,
                top + dy,
                label,
                int(scene["states"][panel][label]),
                label_font,
                state_font,
            )

    _draw_arrow(draw, (PANEL_LEFTS[0] + PANEL_W + 8, PANEL_TOPS[0] + 150), (PANEL_LEFTS[1] - 8, PANEL_TOPS[0] + 150))
    _draw_arrow(draw, (PANEL_LEFTS[1] + PANEL_W + 8, PANEL_TOPS[0] + 150), (PANEL_LEFTS[2] - 8, PANEL_TOPS[0] + 150))
    _draw_arrow(draw, (PANEL_LEFTS[2] + PANEL_W / 2, PANEL_TOPS[0] + PANEL_H + 8), (PANEL_LEFTS[2] + PANEL_W / 2, PANEL_TOPS[1] - 8))
    _draw_arrow(draw, (PANEL_LEFTS[0] + PANEL_W + 8, PANEL_TOPS[1] + 150), (PANEL_LEFTS[1] - 8, PANEL_TOPS[1] + 150))
    _draw_arrow(draw, (PANEL_LEFTS[1] + PANEL_W + 8, PANEL_TOPS[1] + 150), (PANEL_LEFTS[2] - 8, PANEL_TOPS[1] + 150))
    return image


def analytic_gold(scene: dict) -> str:
    onsets = _onsets(scene["states"])
    best = min(onsets)
    if onsets.count(best) != 1:
        raise ValueError("stability onset is tied")
    return LABELS[onsets.index(best)]


def margin(scene: dict) -> float:
    onsets = sorted(_onsets(scene["states"]))
    if len(onsets) < 2:
        return 0.0
    return float(onsets[1] - onsets[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 1.0 or int(scene["glyph_radius"]) < 12


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["storage_order"] = list(reversed(scene["storage_order"]))
    return [("reverse_orderless_storage", twin)]
