"""Independent pixel-only inverse for state-stability onset.

This module intentionally duplicates only the public raster geometry. It does
not import renderer, prompts, generator, verifier, analytic gold, or scene
data.
"""

from __future__ import annotations

from PIL import Image


CANVAS = (1800, 960)
LABELS = ("A", "B", "C", "D")
STATE_COUNT = 3
IDENTITY_BASE_COLORS = (
    (207, 73, 88),
    (38, 126, 180),
    (111, 82, 169),
    (43, 148, 105),
)
STATE_BASE_COLORS = (
    (226, 124, 54),
    (40, 145, 185),
    (157, 89, 163),
)

PANEL_LEFTS = (55, 625, 1195)
PANEL_TOPS = (255, 605)
TILE_W = 220
TILE_H = 80
IDENTITY_STRIP_W = 52
SLOT_OFFSETS = ((22, 70), (266, 70), (22, 174), (266, 174))
IDENTITY_KEY_X = 70
IDENTITY_KEY_Y = 132
IDENTITY_KEY_STEP = 118
STATE_KEY_X = 625
STATE_KEY_Y = 132
STATE_KEY_STEP = 185


def _read_key_color(frame: Image.Image, x: int, y: int, choices: tuple[tuple[int, int, int], ...]) -> int:
    color = frame.getpixel((x, y))
    matches = [index for index, candidate in enumerate(choices) if color == candidate]
    if len(matches) != 1:
        raise ValueError(f"unrecognized key color {color!r}")
    return matches[0]


def _palette(frame: Image.Image) -> tuple[dict[tuple[int, int, int], int], dict[tuple[int, int, int], int]]:
    identity_colors: dict[tuple[int, int, int], int] = {}
    for label in range(4):
        x = IDENTITY_KEY_X + label * IDENTITY_KEY_STEP + 20
        y = IDENTITY_KEY_Y + 16
        color = frame.getpixel((x, y))
        _read_key_color(frame, x, y, IDENTITY_BASE_COLORS)
        if color in identity_colors:
            raise ValueError("identity key reuses a color")
        identity_colors[color] = label
    state_colors: dict[tuple[int, int, int], int] = {}
    for state in range(STATE_COUNT):
        x = STATE_KEY_X + state * STATE_KEY_STEP + 18
        y = STATE_KEY_Y + 16
        color = frame.getpixel((x, y))
        _read_key_color(frame, x, y, STATE_BASE_COLORS)
        if color in state_colors:
            raise ValueError("state key reuses a color")
        state_colors[color] = state
    return identity_colors, state_colors


def _panel_trajectories(frame: Image.Image, identity_colors: dict, state_colors: dict) -> list[list[int]]:
    trajectories = [[None for _ in range(4)] for _ in range(6)]
    for panel in range(6):
        row = panel // 3
        column = panel % 3
        panel_left = PANEL_LEFTS[column]
        panel_top = PANEL_TOPS[row]
        seen: set[int] = set()
        for dx, dy in SLOT_OFFSETS:
            identity_sample = frame.getpixel((panel_left + dx + 10, panel_top + dy + TILE_H // 2))
            if identity_sample not in identity_colors:
                raise ValueError("panel tile has no readable identity stripe")
            label = identity_colors[identity_sample]
            if label in seen:
                raise ValueError("panel repeats an identity")
            seen.add(label)
            state_sample = frame.getpixel((panel_left + dx + 145, panel_top + dy + TILE_H // 2))
            if state_sample not in state_colors:
                raise ValueError("panel tile has no readable state glyph")
            trajectories[panel][label] = state_colors[state_sample]
        if seen != set(range(4)) or any(value is None for value in trajectories[panel]):
            raise ValueError("panel does not expose exactly four identities")
    return [[int(value) for value in row] for row in trajectories]


def _stability_onsets(trajectories: list[list[int]]) -> list[int]:
    onsets = []
    for label in range(4):
        onset = None
        for panel in range(6):
            if all(trajectories[later][label] == trajectories[panel][label] for later in range(panel, 6)):
                onset = panel
                break
        if onset is None:
            raise ValueError("trajectory has no final unchanged suffix")
        onsets.append(onset)
    return onsets


def decision_from_image(image: Image.Image) -> str:
    """Decode all visible panel events and return the earliest stable label."""

    frame = image.convert("RGB")
    if frame.size != CANVAS:
        raise ValueError(f"unexpected raster size {frame.size}")
    identity_colors, state_colors = _palette(frame)
    trajectories = _panel_trajectories(frame, identity_colors, state_colors)
    onsets = _stability_onsets(trajectories)
    best = min(onsets)
    if onsets.count(best) != 1:
        raise ValueError("stability onset is tied")
    return LABELS[onsets.index(best)]
