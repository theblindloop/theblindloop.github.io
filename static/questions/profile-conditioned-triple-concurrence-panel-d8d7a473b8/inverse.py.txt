"""Image-only inverse arm for the triple-concurrence panel world.

Receives one rasterised PNG and the public question contract, and nothing else:
no latent scene, no renderer state, no analytic gold. It locates the panel
dividers from grey columns, segments the three coloured strokes of each panel by
channel dominance, fits each stroke by its principal axis, intersects the three
fitted lines pairwise, and names the panel whose crossing triple is tightest.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

PANEL_NAMES = ("panel_1", "panel_2", "panel_3", "panel_4")
EXPECTED_PANELS = 4


def _divider_columns(rgb: np.ndarray) -> list[int]:
    height, width, _ = rgb.shape
    channels = rgb.astype(np.int16)
    spread = channels.max(axis=2) - channels.min(axis=2)
    mean = channels.mean(axis=2)
    grey = (spread <= 14) & (mean > 110) & (mean < 220)
    fraction = grey.mean(axis=0)
    columns = [x for x in range(width) if fraction[x] > 0.6]
    runs: list[list[int]] = []
    for x in columns:
        if runs and x - runs[-1][-1] <= 2:
            runs[-1].append(x)
        else:
            runs.append([x])
    centres = [
        int(round(sum(run) / len(run)))
        for run in runs
        if 4 < sum(run) / len(run) < width - 5
    ]
    return centres


def _panel_bounds(rgb: np.ndarray) -> list[tuple[int, int]]:
    height, width, _ = rgb.shape
    centres = _divider_columns(rgb)
    if len(centres) != EXPECTED_PANELS - 1:
        step = width / EXPECTED_PANELS
        centres = [int(round(step * k)) for k in range(1, EXPECTED_PANELS)]
    edges = [0] + centres + [width]
    return [(edges[k] + 3, edges[k + 1] - 3) for k in range(EXPECTED_PANELS)]


def _colour_masks(patch: np.ndarray) -> list[np.ndarray]:
    channels = patch.astype(np.int16)
    red, green, blue = channels[:, :, 0], channels[:, :, 1], channels[:, :, 2]
    return [
        red - np.maximum(green, blue) > 30,
        green - np.maximum(red, blue) > 25,
        blue - np.maximum(red, green) > 30,
    ]


def _fit_line(mask: np.ndarray) -> tuple[np.ndarray, np.ndarray] | None:
    ys, xs = np.nonzero(mask)
    if xs.size < 40:
        return None
    points = np.stack([xs.astype(float), ys.astype(float)], axis=1)
    centre = points.mean(axis=0)
    centred = points - centre
    if float(np.abs(centred).max()) < 12.0:
        return None
    _, _, vectors = np.linalg.svd(centred, full_matrices=False)
    return centre, vectors[0]


def _intersect(a, b) -> np.ndarray | None:
    (p1, d1), (p2, d2) = a, b
    den = d1[0] * d2[1] - d1[1] * d2[0]
    if abs(den) < 1e-6:
        return None
    delta = p2 - p1
    t = (delta[0] * d2[1] - delta[1] * d2[0]) / den
    return p1 + t * d1


def _panel_spread(patch: np.ndarray) -> float:
    fits = [_fit_line(mask) for mask in _colour_masks(patch)]
    if any(fit is None for fit in fits):
        return float("inf")
    points = []
    for i in range(3):
        for j in range(i + 1, 3):
            point = _intersect(fits[i], fits[j])
            if point is None:
                return float("inf")
            points.append(point)
    worst = 0.0
    for i in range(len(points)):
        for j in range(i + 1, len(points)):
            worst = max(worst, float(np.linalg.norm(points[i] - points[j])))
    return worst


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"))
    spreads = []
    for x0, x1 in _panel_bounds(rgb):
        spreads.append(_panel_spread(rgb[:, max(0, x0) : x1]))
    best = min(range(len(spreads)), key=lambda k: spreads[k])
    return PANEL_NAMES[best]
