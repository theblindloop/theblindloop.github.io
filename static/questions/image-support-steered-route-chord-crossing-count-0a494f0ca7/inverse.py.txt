from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_TRACE = np.array((23, 52, 93), dtype=np.int16)
_START = np.array((24, 165, 88), dtype=np.int16)
_FINISH = np.array((220, 63, 69), dtype=np.int16)


def _mask(array: np.ndarray, color: np.ndarray, tolerance: int = 8) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _centroid(mask: np.ndarray, name: str) -> tuple[float, float]:
    yy, xx = np.nonzero(mask)
    if not 150 <= len(xx) <= 260:
        raise ValueError(f"missing or malformed {name} marker")
    return float(xx.mean()), float(yy.mean())


def _component_count(mask: np.ndarray) -> int:
    points = {tuple(item) for item in np.argwhere(mask).tolist()}
    count = 0
    while points:
        count += 1
        stack = [points.pop()]
        while stack:
            y, x = stack.pop()
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    neighbor = (y + dy, x + dx)
                    if neighbor in points:
                        points.remove(neighbor)
                        stack.append(neighbor)
    return count


def _signed_profile(trace: np.ndarray, start: tuple[float, float], finish: tuple[float, float]) -> list[float]:
    sy, sx = np.nonzero(trace)
    if not 200 <= len(sx) <= 900 or _component_count(trace) != 1:
        raise ValueError("missing, broken, or extraneous navy centerline")
    dx = finish[0] - start[0]
    dy = finish[1] - start[1]
    length = float(np.hypot(dx, dy))
    if not 95.0 <= length <= 122.0:
        raise ValueError("malformed endpoint chord")
    ux, uy = dx / length, dy / length
    nx, ny = -uy, ux
    px = sx.astype(float) - start[0]
    py = sy.astype(float) - start[1]
    along = px * ux + py * uy
    side = px * nx + py * ny
    profile = []
    for center in np.arange(10.0, length - 9.9, 3.0):
        selected = np.abs(along - center) <= 1.8
        if int(selected.sum()) >= 2:
            profile.append(float(np.median(side[selected])))
    if len(profile) < 24:
        raise ValueError("centerline profile is incomplete")
    return profile


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 64 or array.shape[1] < 64:
        raise ValueError("unsupported raster")
    start = _centroid(_mask(array, _START), "green")
    finish = _centroid(_mask(array, _FINISH), "red")
    profile = _signed_profile(_mask(array, _TRACE), start, finish)
    signs = []
    for value in profile:
        sign = 1 if value > 4.0 else -1 if value < -4.0 else 0
        if sign and (not signs or signs[-1] != sign):
            signs.append(sign)
    crossings = len(signs) - 1
    if crossings not in (2, 3):
        raise ValueError(f"ambiguous crossing count {crossings}")
    if max(abs(value) for value in profile) < 11.0:
        raise ValueError("route stays too close to its endpoint chord")
    return str(crossings)
