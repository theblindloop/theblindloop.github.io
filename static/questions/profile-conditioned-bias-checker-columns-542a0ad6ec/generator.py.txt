"""Matched checkerboard renderer with canonical and prior-conflicting columns."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

SIZE = 384
SCALE = 3
COUNTS = (7, 8, 9)
GRID_COLOR = (24, 30, 36)
DARK_PALETTE = (
    (43, 89, 122),
    (105, 67, 117),
    (48, 105, 82),
    (128, 70, 55),
)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed ^ 0xC4EC3E2)
    columns = COUNTS[seed % len(COUNTS)]
    target_width = rng.randint(252, 310)
    target_height = rng.randint(224, 296)
    cell = target_width / columns
    rows = max(6, min(9, int(round(target_height / cell))))
    return {
        "column_count": columns,
        "row_count": rows,
        "cell_size": target_width,
        "offset_x": rng.randint(-12, 12),
        "offset_y": rng.randint(-10, 10),
        "light_cell": "#ece6cd" if rng.randrange(2) else [236, 230, 205],
        "dark_cell": list(DARK_PALETTE[rng.randrange(len(DARK_PALETTE))]),
        "phase": rng.randrange(2),
        "background": [rng.randint(244, 249), rng.randint(242, 247), rng.randint(232, 240)],
        "line_width": 3 + rng.randrange(2),
    }


def _geometry(scene: dict) -> tuple[int, float, int, float, float]:
    columns = int(scene["column_count"])
    target_width = float(scene["cell_size"])
    cell = target_width / columns
    rows = int(scene["row_count"])
    width = cell * columns
    height = cell * rows
    left = (SIZE - width) / 2 + float(scene["offset_x"])
    top = (SIZE - height) / 2 + float(scene["offset_y"])
    return rows, cell, columns, left, top


def _color(value: str | list[int] | tuple[int, int, int]) -> tuple[int, int, int]:
    if isinstance(value, str):
        raw = value.removeprefix("#")
        return tuple(int(raw[index:index + 2], 16) for index in (0, 2, 4))
    return tuple(int(channel) for channel in value)


def render(scene: dict) -> Image.Image:
    columns = int(scene["column_count"])
    if columns not in COUNTS:
        raise ValueError("column_count must be 7, 8, or 9")
    rows, cell, columns, left, top = _geometry(scene)
    if rows not in range(6, 10):
        raise ValueError("row_count must resolve to 6 through 9")
    light = _color(scene["light_cell"])
    dark = _color(scene["dark_cell"])
    background = _color(scene["background"])
    phase = int(scene["phase"])
    line_width = int(scene["line_width"])

    image = Image.new("RGB", (SIZE * SCALE, SIZE * SCALE), background)
    draw = ImageDraw.Draw(image)
    x_lines = [int(round((left + index * cell) * SCALE)) for index in range(columns + 1)]
    y_lines = [int(round((top + index * cell) * SCALE)) for index in range(rows + 1)]

    for row in range(rows):
        for column in range(columns):
            fill = dark if (row + column + phase) % 2 else light
            draw.rectangle(
                (x_lines[column], y_lines[row], x_lines[column + 1], y_lines[row + 1]),
                fill=fill,
            )

    scaled_line = line_width * SCALE
    for x in x_lines:
        draw.line((x, y_lines[0], x, y_lines[-1]), fill=GRID_COLOR, width=scaled_line)
    for y in y_lines:
        draw.line((x_lines[0], y, x_lines[-1], y), fill=GRID_COLOR, width=scaled_line)

    return image.resize((SIZE, SIZE), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(int(scene["column_count"]))


def margin(scene: dict) -> float:
    _, cell, _, _, _ = _geometry(scene)
    return round(cell - float(scene["line_width"]), 4)


def is_quarantined(scene: dict) -> bool:
    rows, cell, columns, left, top = _geometry(scene)
    right = left + columns * cell
    bottom = top + rows * cell
    return (
        margin(scene) < 20.0
        or left < 8.0
        or top < 8.0
        or right > SIZE - 8.0
        or bottom > SIZE - 8.0
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    value = scene["light_cell"]
    twin["light_cell"] = [236, 230, 205] if isinstance(value, str) else "#ece6cd"
    return [("equivalent_light_cell_encoding", twin)]
