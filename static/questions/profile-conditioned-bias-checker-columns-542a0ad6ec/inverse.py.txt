"""Independent PNG-only inverse arm for counting checkerboard columns."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _runs(mask: np.ndarray) -> list[tuple[int, int]]:
    padded = np.pad(mask.astype(np.int8), (1, 1))
    changes = np.diff(padded)
    starts = np.flatnonzero(changes == 1)
    ends = np.flatnonzero(changes == -1) - 1
    return [(int(start), int(end)) for start, end in zip(starts, ends)]


def decision_from_image(image: Image.Image) -> str:
    if image.width < 192 or image.height < 192:
        raise ValueError("image resolution is too low")
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    near_black = pixels.max(axis=2) < 66
    vertical_ink = near_black.sum(axis=0)
    peak = int(vertical_ink.max(initial=0))
    if peak < image.height * 0.45:
        raise ValueError("continuous checkerboard separators are absent")

    separator_columns = vertical_ink >= peak * 0.72
    runs = _runs(separator_columns)
    centers = np.array([(start + end) / 2.0 for start, end in runs], dtype=float)
    widths = np.array([end - start + 1 for start, end in runs], dtype=float)
    if len(runs) not in (8, 9, 10):
        raise ValueError("visible vertical separator count is out of contract")
    spacing = np.diff(centers)
    if spacing.min(initial=999.0) < 20.0 or spacing.max() / spacing.min() > 1.08:
        raise ValueError("vertical separators do not form a regular cell grid")
    if widths.max(initial=0.0) > 9.0 or widths.max() - widths.min() > 3.0:
        raise ValueError("separator evidence is fragmented or inconsistent")
    return str(len(runs) - 1)
