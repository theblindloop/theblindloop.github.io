"""Independent fixed-layout RGB decoder for the coupled-ledger raster."""

from __future__ import annotations

from PIL import Image

N = 5
WIDTH = 1280
HEIGHT = 920
CARD_POSITIONS = ((24, 128), (444, 128), (864, 128), (234, 502), (654, 502))
ROW_OFFSET = 96
ROW_STEP = 44

SOURCE_COLORS = (
    (214, 78, 74), (52, 115, 178), (67, 151, 96), (143, 91, 176), (226, 145, 48),
)
TOKEN_COLORS = (
    (24, 148, 143), (196, 78, 132), (56, 72, 126), (129, 153, 52), (38, 160, 201),
)
TARGET_COLORS = (
    (240, 108, 97), (82, 139, 211), (94, 181, 116), (171, 116, 203), (241, 176, 70),
)


def _classify(pixels, cx: int, cy: int, palette: tuple[tuple[int, int, int], ...], radius: int = 18) -> int:
    counts = []
    for color in palette:
        count = 0
        for y in range(cy - radius, cy + radius + 1):
            for x in range(cx - radius, cx + radius + 1):
                pixel = pixels[x, y]
                if max(abs(int(pixel[channel]) - color[channel]) for channel in range(3)) <= 8:
                    count += 1
        counts.append(count)
    best = max(range(len(counts)), key=counts.__getitem__)
    ordered = sorted(counts, reverse=True)
    if counts[best] < 180 or (len(ordered) > 1 and ordered[1] > counts[best] // 3):
        raise ValueError("identity badge is missing or ambiguous")
    return best


def _decode_board(pixels, origin: tuple[int, int]) -> tuple[list[int], list[int]]:
    x, y = origin
    first = [-1] * N
    second = [-1] * N
    for row in range(N):
        cy = y + ROW_OFFSET + row * ROW_STEP
        source = _classify(pixels, x + 34, cy, SOURCE_COLORS)
        token = _classify(pixels, x + 154, cy, TOKEN_COLORS)
        if first[source] != -1:
            raise ValueError("duplicate visible source row")
        first[source] = token

        right_token = _classify(pixels, x + 230, cy, TOKEN_COLORS)
        target = _classify(pixels, x + 351, cy, TARGET_COLORS)
        if second[right_token] != -1:
            raise ValueError("duplicate visible token row")
        second[right_token] = target
    if sorted(first) != list(range(N)) or sorted(second) != list(range(N)):
        raise ValueError("visible ledger is not two complete bijections")
    return first, second


def decision_from_image(image: Image.Image) -> str:
    raster = image.convert("RGB")
    if raster.size != (WIDTH, HEIGHT):
        raise ValueError("unexpected canvas size")
    pixels = raster.load()
    boards = [_decode_board(pixels, position) for position in CARD_POSITIONS]
    composites = [
        [second[first[source]] for source in range(N)]
        for first, second in boards
    ]
    invariant = [
        source
        for source in range(N)
        if len({board[source] for board in composites}) == 1
    ]
    if len(invariant) != 1:
        raise ValueError("visible boards do not identify exactly one invariant letter")
    if any(len({first[source] for first, _second in boards}) != N for source in range(N)):
        raise ValueError("a visible first-step binding does not vary on every board")
    if any(len({second[token] for _first, second in boards}) < 3 for token in range(N)):
        raise ValueError("visible second-step evidence is degenerate")
    return f"letter-{chr(ord('A') + invariant[0])}"
