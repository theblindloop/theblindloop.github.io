"""Render five coupled two-stage correspondence ledgers."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont

N = 5
WIDTH = 1280
HEIGHT = 920
CARD_W = 392
CARD_H = 338
CARD_POSITIONS = ((24, 128), (444, 128), (864, 128), (234, 502), (654, 502))
ROW_OFFSET = 96
ROW_STEP = 44

BG = (230, 237, 241)
PAPER = (255, 254, 249)
CARD = (249, 251, 250)
INK = (31, 42, 52)
MUTED = (83, 96, 106)
RULE = (177, 188, 196)
ROW_TINT = (238, 243, 244)

SOURCE_COLORS = (
    (214, 78, 74),
    (52, 115, 178),
    (67, 151, 96),
    (143, 91, 176),
    (226, 145, 48),
)
TOKEN_COLORS = (
    (24, 148, 143),
    (196, 78, 132),
    (56, 72, 126),
    (129, 153, 52),
    (38, 160, 201),
)
TARGET_COLORS = (
    (240, 108, 97),
    (82, 139, 211),
    (94, 181, 116),
    (171, 116, 203),
    (241, 176, 70),
)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{name}", size)


def _inverse(permutation: list[int]) -> list[int]:
    result = [0] * N
    for index, value in enumerate(permutation):
        result[value] = index
    return result


def _base_boards(rng: random.Random) -> list[tuple[list[int], list[int]]]:
    """Make five coupled alternatives with cancellation only after composition."""

    shifts = list(range(N))
    rng.shuffle(shifts)
    for _ in range(20000):
        composites: list[list[int]] = []
        for _board in range(N):
            tail = list(range(1, N))
            rng.shuffle(tail)
            composites.append([0, *tail])
        if len({tuple(item) for item in composites}) != N:
            continue
        if any(len({item[source] for item in composites}) < 3 for source in range(1, N)):
            continue

        boards: list[tuple[list[int], list[int]]] = []
        for shift, composite in zip(shifts, composites):
            first = [(source + shift) % N for source in range(N)]
            inv_first = _inverse(first)
            second = [composite[inv_first[token]] for token in range(N)]
            boards.append((first, second))
        if all(len({second[token] for _first, second in boards}) >= 3 for token in range(N)):
            return boards
    raise RuntimeError("could not construct coupled ledgers")


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed ^ 0xC0A91ED)
    target_source = seed % N

    visible_to_base = [None] * N
    visible_to_base[target_source] = 0
    remaining_base = list(range(1, N))
    rng.shuffle(remaining_base)
    for visible in range(N):
        if visible != target_source:
            visible_to_base[visible] = remaining_base.pop()

    token_permutation = list(range(N))
    target_permutation = list(range(N))
    rng.shuffle(token_permutation)
    rng.shuffle(target_permutation)
    inverse_token = _inverse(token_permutation)

    boards = []
    for first_base, second_base in _base_boards(rng):
        first_visible = [
            token_permutation[first_base[int(visible_to_base[source])]]
            for source in range(N)
        ]
        second_visible = [
            target_permutation[second_base[inverse_token[token]]]
            for token in range(N)
        ]
        left_order = list(range(N))
        right_order = list(range(N))
        rng.shuffle(left_order)
        rng.shuffle(right_order)
        boards.append({
            "letter_to_token": first_visible,
            "token_to_number": second_visible,
            "left_order": left_order,
            "right_order": right_order,
        })
    rng.shuffle(boards)
    scene = {"width": WIDTH, "height": HEIGHT, "boards": boards}
    if analytic_gold(scene) != f"letter-{chr(ord('A') + target_source)}":
        raise RuntimeError("construction lost its intended invariant")
    return scene


def _composites(scene: dict) -> list[list[int]]:
    results = []
    for board in scene["boards"]:
        first = board["letter_to_token"]
        second = board["token_to_number"]
        results.append([int(second[int(first[source])]) for source in range(N)])
    return results


def _invariant_source(scene: dict) -> int:
    composed = _composites(scene)
    invariant = [
        source
        for source in range(N)
        if len({board[source] for board in composed}) == 1
    ]
    if len(invariant) != 1:
        raise ValueError("scene must have exactly one invariant composite destination")
    return invariant[0]


def _draw_token(draw: ImageDraw.ImageDraw, center: tuple[int, int], token: int) -> None:
    cx, cy = center
    color = TOKEN_COLORS[token]
    radius = 15
    if token == 0:
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=color, outline=INK, width=2)
    elif token == 1:
        draw.rounded_rectangle((cx - radius, cy - radius, cx + radius, cy + radius), radius=4, fill=color, outline=INK, width=2)
    elif token == 2:
        draw.polygon(((cx, cy - 18), (cx + 18, cy), (cx, cy + 18), (cx - 18, cy)), fill=color, outline=INK)
        draw.line(((cx, cy - 18), (cx + 18, cy), (cx, cy + 18), (cx - 18, cy), (cx, cy - 18)), fill=INK, width=2)
    elif token == 3:
        draw.polygon(((cx, cy - 18), (cx + 18, cy + 15), (cx - 18, cy + 15)), fill=color, outline=INK)
        draw.line(((cx, cy - 18), (cx + 18, cy + 15), (cx - 18, cy + 15), (cx, cy - 18)), fill=INK, width=2)
    else:
        draw.polygon(((cx - 15, cy - 12), (cx, cy - 18), (cx + 15, cy - 12),
                      (cx + 15, cy + 12), (cx, cy + 18), (cx - 15, cy + 12)), fill=color, outline=INK)
        draw.line(((cx - 15, cy - 12), (cx, cy - 18), (cx + 15, cy - 12),
                   (cx + 15, cy + 12), (cx, cy + 18), (cx - 15, cy + 12),
                   (cx - 15, cy - 12)), fill=INK, width=2)


def _draw_source(draw: ImageDraw.ImageDraw, center: tuple[int, int], source: int) -> None:
    cx, cy = center
    draw.rounded_rectangle((cx - 17, cy - 17, cx + 17, cy + 17), radius=8,
                           fill=SOURCE_COLORS[source], outline=INK, width=2)
    draw.text((cx, cy), chr(ord("A") + source), font=_font(19, True), fill=(255, 255, 255), anchor="mm")


def _draw_target(draw: ImageDraw.ImageDraw, center: tuple[int, int], target: int) -> None:
    cx, cy = center
    draw.ellipse((cx - 17, cy - 17, cx + 17, cy + 17), fill=TARGET_COLORS[target], outline=INK, width=2)
    draw.text((cx, cy), str(target + 1), font=_font(18, True), fill=INK, anchor="mm")


def _draw_arrow(draw: ImageDraw.ImageDraw, start: tuple[int, int], end: tuple[int, int]) -> None:
    x0, y0 = start
    x1, y1 = end
    draw.line((x0, y0, x1 - 6, y1), fill=MUTED, width=3)
    draw.polygon(((x1, y1), (x1 - 9, y1 - 6), (x1 - 9, y1 + 6)), fill=MUTED)


def _draw_board(draw: ImageDraw.ImageDraw, board: dict, board_index: int, origin: tuple[int, int]) -> None:
    x, y = origin
    draw.rounded_rectangle((x, y, x + CARD_W, y + CARD_H), radius=16, fill=CARD, outline=INK, width=3)
    draw.text((x + CARD_W // 2, y + 21), f"BOARD {board_index + 1}", font=_font(18, True), fill=INK, anchor="mm")
    draw.line((x + 14, y + 42, x + CARD_W - 14, y + 42), fill=RULE, width=2)
    draw.text((x + 97, y + 62), "LETTER  →  TOKEN", font=_font(14, True), fill=MUTED, anchor="mm")
    draw.text((x + 292, y + 62), "TOKEN  →  NUMBER", font=_font(14, True), fill=MUTED, anchor="mm")
    draw.line((x + 195, y + 50, x + 195, y + CARD_H - 14), fill=RULE, width=2)

    for row in range(N):
        cy = y + ROW_OFFSET + row * ROW_STEP
        if row % 2 == 0:
            draw.rounded_rectangle((x + 12, cy - 19, x + 186, cy + 19), radius=8, fill=ROW_TINT)
            draw.rounded_rectangle((x + 204, cy - 19, x + 380, cy + 19), radius=8, fill=ROW_TINT)

        source = int(board["left_order"][row])
        token = int(board["letter_to_token"][source])
        _draw_source(draw, (x + 34, cy), source)
        _draw_arrow(draw, (x + 57, cy), (x + 123, cy))
        _draw_token(draw, (x + 154, cy), token)

        right_token = int(board["right_order"][row])
        target = int(board["token_to_number"][right_token])
        _draw_token(draw, (x + 230, cy), right_token)
        _draw_arrow(draw, (x + 253, cy), (x + 319, cy))
        _draw_target(draw, (x + 351, cy), target)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BG)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((10, 10, WIDTH - 10, HEIGHT - 10), radius=24, fill=PAPER, outline=INK, width=3)
    draw.text((WIDTH // 2, 35), "FIVE POSSIBLE WIRINGS", font=_font(27, True), fill=INK, anchor="mm")
    draw.text((WIDTH // 2, 72), "On each board: follow a LETTER to its TOKEN, then that TOKEN to its NUMBER.",
              font=_font(19, True), fill=MUTED, anchor="mm")
    draw.text((WIDTH // 2, 103), "Which letter reaches the same number on every board?",
              font=_font(21, True), fill=(172, 61, 63), anchor="mm")
    for board_index, (board, position) in enumerate(zip(scene["boards"], CARD_POSITIONS)):
        _draw_board(draw, board, board_index, position)
    draw.text((WIDTH // 2, 884), "ANSWER:  LETTER A   •   LETTER B   •   LETTER C   •   LETTER D   •   LETTER E",
              font=_font(18, True), fill=INK, anchor="mm")
    return image


def analytic_gold(scene: dict) -> str:
    return f"letter-{chr(ord('A') + _invariant_source(scene))}"


def margin(scene: dict) -> float:
    composed = _composites(scene)
    answer = _invariant_source(scene)
    return float(min(len({board[source] for board in composed}) for source in range(N) if source != answer) - 1)


def is_quarantined(scene: dict) -> bool:
    try:
        if int(scene["width"]) != WIDTH or int(scene["height"]) != HEIGHT or len(scene["boards"]) != N:
            return True
        for board in scene["boards"]:
            if sorted(map(int, board["letter_to_token"])) != list(range(N)):
                return True
            if sorted(map(int, board["token_to_number"])) != list(range(N)):
                return True
            if sorted(map(int, board["left_order"])) != list(range(N)):
                return True
            if sorted(map(int, board["right_order"])) != list(range(N)):
                return True
        answer = _invariant_source(scene)
        if any(len({board["letter_to_token"][source] for board in scene["boards"]}) != N for source in range(N)):
            return True
        if any(len({board["token_to_number"][token] for board in scene["boards"]}) < 3 for token in range(N)):
            return True
        composed = _composites(scene)
        if len({board[answer] for board in composed}) != 1:
            return True
        return margin(scene) < 2.0
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
