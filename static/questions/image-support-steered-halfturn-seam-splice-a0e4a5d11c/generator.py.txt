from __future__ import annotations

import itertools
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 1024
HEIGHT = 1024
BG = (250, 248, 242)
NAVY = (31, 48, 68)
PALE = (176, 195, 204)
GOLD = (225, 168, 38)
TAG_RGB = {
    "red": (210, 62, 64),
    "green": (47, 151, 91),
    "blue": (52, 105, 202),
}
DX = 8
DY = 6


def _valid_codes(length: int) -> list[tuple[int, ...]]:
    output = []
    for code in itertools.product((-1, 1), repeat=length):
        levels = [0]
        for step in code:
            levels.append(levels[-1] + step)
        if max(levels) - min(levels) <= 3 and abs(levels[-1]) <= 1:
            output.append(code)
    return output


REFERENCE_CODES = _valid_codes(7)
PATCH_CODES = list(itertools.product((-1, 1), repeat=3))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    reference = rng.choice(REFERENCE_CODES)
    wanted = tuple(reversed(reference))[2:5]
    wrong = [code for code in PATCH_CODES if code != wanted]
    rng.shuffle(wrong)
    correct_slot = int(seed) % 3
    candidates = [wrong[0], wrong[1], wrong[2]]
    candidates[correct_slot] = wanted
    # Rotate the visible tag palette independently of the correct slot. Over
    # consecutive generator seeds, every answer color occurs equally often.
    colors = ["red", "green", "blue"]
    offset = (int(seed) // 3) % 3
    tag_order = colors[offset:] + colors[:offset]
    origin = [128 * rng.randint(1, 5) + 8, 128 * rng.randint(1, 6) + 64]
    return {
        "reference_steps": list(reference),
        "candidate_steps": [list(code) for code in candidates],
        "tag_order": tag_order,
        "origin": origin,
    }


def _points(x: int, y: int, steps: Iterable[int]) -> list[tuple[int, int]]:
    pts = [(x, y)]
    for step in steps:
        x += DX
        y += int(step) * DY
        pts.append((x, y))
    return pts


def _draw_trace(draw: ImageDraw.ImageDraw, pts: list[tuple[int, int]], *, nodes: bool) -> None:
    draw.line(pts, fill=NAVY, width=5, joint="curve")
    if nodes:
        for x, y in pts:
            draw.ellipse((x - 3, y - 3, x + 3, y + 3), fill=PALE, outline=NAVY, width=1)


def _draw_tag(draw: ImageDraw.ImageDraw, x: int, y: int, color: tuple[int, int, int]) -> None:
    draw.ellipse((x - 5, y - 5, x + 5, y + 5), fill=color, outline=NAVY, width=1)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    ox, oy = map(int, scene["origin"])
    reference = tuple(int(v) for v in scene["reference_steps"])
    candidates = [tuple(int(v) for v in code) for code in scene["candidate_steps"]]
    tags = list(scene["tag_order"])

    # One small, translation-randomized plaque. Its long horizontal footprint
    # is deliberate: destructive support should be focal and pathlike.
    draw.rounded_rectangle((ox - 10, oy - 24, ox + 195, oy + 57), radius=9,
                           fill=(242, 239, 231), outline=(202, 199, 190), width=2)

    _draw_tag(draw, ox, oy, GOLD)
    ref_pts = _points(ox + 9, oy, reference)
    _draw_trace(draw, ref_pts, nodes=True)

    # The lower trace exposes the two outer pieces of the required half-turn
    # result. The three central steps are the visible missing seam.
    transformed = tuple(reversed(reference))
    target_pts = _points(ox + 9, oy + 35, transformed)
    _draw_trace(draw, target_pts[:3], nodes=False)
    _draw_trace(draw, target_pts[5:], nodes=False)
    for index in (2, 5):
        x, y = target_pts[index]
        draw.line((x, y - 5, x, y + 5), fill=GOLD, width=2)

    starts = (ox + 76, ox + 116, ox + 156)
    for start, code, tag in zip(starts, candidates, tags, strict=True):
        _draw_tag(draw, start, oy, TAG_RGB[tag])
        _draw_trace(draw, _points(start + 9, oy, code), nodes=True)
    return image


def analytic_gold(scene: dict) -> str:
    wanted = tuple(reversed(tuple(int(v) for v in scene["reference_steps"])))[2:5]
    candidates = [tuple(int(v) for v in code) for code in scene["candidate_steps"]]
    matches = [index for index, code in enumerate(candidates) if code == wanted]
    if len(matches) != 1:
        raise ValueError("scene does not contain one unique splice")
    return str(scene["tag_order"][matches[0]])


def margin(scene: dict) -> float:
    wanted = tuple(reversed(tuple(int(v) for v in scene["reference_steps"])))[2:5]
    distances = sorted(
        sum(a != b for a, b in zip(wanted, tuple(int(v) for v in code), strict=True))
        for code in scene["candidate_steps"]
    )
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    try:
        reference = tuple(int(v) for v in scene["reference_steps"])
        wanted = tuple(reversed(reference))[2:5]
        candidates = [tuple(int(v) for v in code) for code in scene["candidate_steps"]]
        tags = list(scene["tag_order"])
    except Exception:
        return True
    return not (
        len(reference) == 7
        and all(value in (-1, 1) for value in reference)
        and len(candidates) == 3
        and len(set(candidates)) == 3
        and sum(code == wanted for code in candidates) == 1
        and sorted(tags) == ["blue", "green", "red"]
        and margin(scene) >= 1.0
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
