from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALE = np.array((176, 195, 204), dtype=np.int16)
COLORS = {
    "gold": np.array((225, 168, 38), dtype=np.int16),
    "red": np.array((210, 62, 64), dtype=np.int16),
    "green": np.array((47, 151, 91), dtype=np.int16),
    "blue": np.array((52, 105, 202), dtype=np.int16),
}


def _mask(array: np.ndarray, color: np.ndarray, tolerance: float = 18.0) -> np.ndarray:
    delta = array.astype(np.int32) - color.astype(np.int32)
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _components(mask: np.ndarray, minimum: int = 8) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y0, x0 in np.argwhere(mask):
        if seen[y0, x0]:
            continue
        queue = deque([(int(y0), int(x0))])
        seen[y0, x0] = True
        points = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny in range(max(0, y - 1), min(height, y + 2)):
                for nx in range(max(0, x - 1), min(width, x + 2)):
                    if mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        if len(points) >= minimum:
            ys, xs = zip(*points)
            output.append((float(np.mean(xs)), float(np.mean(ys)), len(points)))
    return output


def _one_tag(array: np.ndarray, name: str) -> tuple[float, float]:
    components = _components(_mask(array, COLORS[name]), minimum=30)
    if len(components) != 1:
        raise ValueError(f"expected one intact {name} tag")
    return components[0][0], components[0][1]


def _code_after(tag: tuple[float, float], dots: list[tuple[float, float, int]], count: int) -> tuple[int, ...]:
    tx, ty = tag
    reach = 75.0 if count == 8 else 38.0
    nearby = [dot for dot in dots if tx + 5.0 < dot[0] < tx + reach and abs(dot[1] - ty) < 24.0]
    nearby.sort(key=lambda dot: dot[0])
    if len(nearby) != count:
        raise ValueError(f"expected {count} ordered contour landmarks, found {len(nearby)}")
    xs = [dot[0] for dot in nearby]
    if any(not 4.5 < b - a < 11.0 for a, b in zip(xs, xs[1:])):
        raise ValueError("contour landmarks are not one uniform thin trace")
    code = []
    for left, right in zip(nearby, nearby[1:]):
        delta = right[1] - left[1]
        if abs(delta) < 3.0:
            raise ValueError("contour step direction is ambiguous")
        code.append(1 if delta > 0 else -1)
    return tuple(code)


def decision_from_image(image: Image.Image) -> str:
    full = np.asarray(image.convert("RGB"), dtype=np.uint8)
    border = np.concatenate((full[:2].reshape(-1, 3), full[-2:].reshape(-1, 3),
                             full[:, :2].reshape(-1, 3), full[:, -2:].reshape(-1, 3)))
    background = np.median(border, axis=0)
    foreground = np.linalg.norm(full.astype(np.float32) - background, axis=2) > 24.0
    locations = np.argwhere(foreground)
    if not len(locations):
        raise ValueError("image has no visible seam plaque")
    y0, x0 = locations.min(axis=0)
    y1, x1 = locations.max(axis=0) + 1
    # All subsequent measurements operate on the visually discovered plaque,
    # not on canvas coordinates. This also keeps ablation probes inexpensive.
    array = full[max(0, y0 - 4):min(full.shape[0], y1 + 4),
                 max(0, x0 - 4):min(full.shape[1], x1 + 4)]
    dots = _components(_mask(array, PALE), minimum=14)
    gold = _one_tag(array, "gold")
    reference = _code_after(gold, dots, 8)
    wanted = tuple(reversed(reference))[2:5]
    matches = []
    for name in ("red", "green", "blue"):
        tag = _one_tag(array, name)
        if _code_after(tag, dots, 4) == wanted:
            matches.append(name)
    if len(matches) != 1:
        raise ValueError("image does not expose one unique half-turn splice")
    return matches[0]
