"""Deterministic raster renderer and analytic consequence for the jigsaw world."""

from __future__ import annotations

import copy
import random
from typing import Any

from PIL import Image, ImageDraw

FIT_TOLERANCE = 7.0
PROFILE_LIBRARY = (
    (-18, 20, 4, -22, 16, 0),
    (16, -16, 0, 18, -20, 10),
    (-12, 22, -20, 14, 2, -18),
    (20, -10, 18, -18, -4, 16),
    (-20, 8, 20, -6, -18, 22),
    (10, -22, -8, 20, -16, 6),
)
PALETTE = (
    (210, 67, 72),
    (48, 119, 204),
    (44, 157, 111),
    (168, 87, 181),
    (226, 135, 42),
    (31, 157, 164),
)


def _sorted_pieces(scene: dict[str, Any]) -> list[dict[str, Any]]:
    return sorted(scene["pieces"], key=lambda piece: int(piece["x"]))


def _gap_profile(scene: dict[str, Any]) -> list[float]:
    left, right = _sorted_pieces(scene)
    left_profile = [float(value) for value in left["profile"]]
    right_profile = [float(value) for value in right["profile"]]
    bands = len(left_profile)
    if len(left_profile) != bands or len(right_profile) != bands:
        raise ValueError("each piece must expose exactly band_count profile values")
    left_width = float(left["width"])
    return [
        float(right["x"]) - right_profile[index]
        - (float(left["x"]) + left_width + left_profile[index])
        for index in range(bands)
    ]


def analytic_gold(scene: dict[str, Any]) -> str:
    """Compute the visible edge-fit consequence from the latent geometry."""

    gaps = _gap_profile(scene)
    median_gap = sorted(gaps)[len(gaps) // 2]
    deviation = max(abs(gap - median_gap) for gap in gaps)
    return "yes" if deviation <= FIT_TOLERANCE else "no"


def margin(scene: dict[str, Any]) -> float:
    gaps = _gap_profile(scene)
    median_gap = sorted(gaps)[len(gaps) // 2]
    deviation = max(abs(gap - median_gap) for gap in gaps)
    return float(FIT_TOLERANCE - deviation)


def is_quarantined(scene: dict[str, Any]) -> bool:
    pieces = _sorted_pieces(scene)
    if len(pieces) != 2:
        return True
    bands = int(scene["band_count"])
    if bands != 6:
        return True
    if any(len(piece["profile"]) != bands for piece in pieces):
        return True
    top = int(scene["top"])
    bottom = top + min(int(piece["height"]) for piece in pieces)
    if top < 24 or bottom > int(scene["height"]) - 24:
        return True
    if pieces[0]["x"] + pieces[0]["width"] + 35 >= pieces[1]["x"]:
        return True
    if min(int(piece["width"]) for piece in pieces) < 110:
        return True
    if min(int(piece["height"]) for piece in pieces) < 210:
        return True
    return abs(margin(scene)) < 3.0


def _piece_polygon(scene: dict[str, Any], piece: dict[str, Any]) -> list[tuple[int, int]]:
    bands = int(scene["band_count"])
    top = int(scene["top"])
    height = int(piece["height"])
    width = int(piece["width"])
    x0 = int(piece["x"])
    x1 = x0 + width
    profile = [int(value) for value in piece["profile"]]
    if len(profile) != bands:
        raise ValueError("profile length does not match band_count")
    points: list[tuple[int, int]] = []
    if piece["side"] == "left":
        points = [(x0, top), (x1 + profile[0], top)]
        for index, offset in enumerate(profile):
            y_next = round(top + (index + 1) * height / bands)
            points.append((x1 + offset, y_next))
            if index + 1 < bands:
                points.append((x1 + profile[index + 1], y_next))
        points.append((x0, top + height))
    else:
        points = [(x0 - profile[0], top)]
        for index, offset in enumerate(profile):
            y_next = round(top + (index + 1) * height / bands)
            points.append((x0 - offset, y_next))
            if index + 1 < bands:
                points.append((x0 - profile[index + 1], y_next))
        points.extend([(x1, top + height), (x1, top)])
    return points


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new(
        "RGB",
        (int(scene["width"]), int(scene["height"])),
        tuple(int(value) for value in scene["background"]),
    )
    draw = ImageDraw.Draw(image)
    for piece in _sorted_pieces(scene):
        polygon = _piece_polygon(scene, piece)
        draw.polygon(
            polygon,
            fill=tuple(int(value) for value in piece["color"]),
            outline=tuple(int(value) for value in scene["outline_color"]),
            width=int(scene["outline_width"]),
        )
    return image


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    width, height = 800, 480
    band_count = 6
    piece_width = rng.randint(158, 178)
    piece_height = rng.randint(300, 326)
    top = rng.randint(68, 86)
    left_x = rng.randint(18, 32)
    right_x = rng.randint(608, 626)
    base = list(PROFILE_LIBRARY[rng.randrange(len(PROFILE_LIBRARY))])
    fit = int(seed) % 2 == 0
    if fit:
        right_profile = [-value for value in base]
    else:
        mismatch = (int(seed) + rng.randrange(6)) % band_count
        right_profile = [-value for value in base]
        # Alternate signed perturbations across every band: the counterfactual
        # is not recoverable from one local mismatch or from a cheap color cue.
        for index in range(band_count):
            right_profile[index] += 23 if index % 2 == 0 else -23
        right_profile[mismatch] += 5
    color_indices = list(range(len(PALETTE)))
    rng.shuffle(color_indices)
    left_color = PALETTE[color_indices[0]]
    right_color = PALETTE[color_indices[1]]
    pieces = [
        {
            "side": "left",
            "x": left_x,
            "width": piece_width,
            "height": piece_height,
            "profile": base,
            "color": left_color,
        },
        {
            "side": "right",
            "x": right_x,
            "width": piece_width,
            "height": piece_height,
            "profile": right_profile,
            "color": right_color,
        },
    ]
    return {
        "pieces": pieces,
        "band_count": band_count,
        "top": top,
        "outline_color": (27, 33, 45),
        "outline_width": 3,
        "width": width,
        "height": height,
        "background": (247, 245, 239),
    }


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    twin = copy.deepcopy(scene)
    twin["pieces"] = list(reversed(twin["pieces"]))
    return [("unordered_piece_record_permutation", twin)]
