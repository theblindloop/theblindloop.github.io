"""Independent image-only inverse for the visible jigsaw edge comparison."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

FIT_TOLERANCE = 7.0
BAND_COUNT = 6


def _foreground_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    border = np.concatenate(
        (
            array[:4].reshape(-1, 3),
            array[-4:].reshape(-1, 3),
            array[:, :4].reshape(-1, 3),
            array[:, -4:].reshape(-1, 3),
        ),
        axis=0,
    )
    background = np.median(border, axis=0)
    distance = np.sqrt(np.sum((array - background) ** 2, axis=2))
    return distance > 22.0


def _x_runs(mask: np.ndarray) -> list[tuple[int, int]]:
    occupied = np.any(mask, axis=0)
    starts = np.flatnonzero(occupied & ~np.r_[False, occupied[:-1]])
    ends = np.flatnonzero(occupied & ~np.r_[occupied[1:], False])
    return [(int(start), int(end) + 1) for start, end in zip(starts, ends, strict=True)]


def _bbox(mask: np.ndarray) -> tuple[int, int, int, int]:
    ys, xs = np.nonzero(mask)
    if len(xs) == 0:
        raise ValueError("empty puzzle component")
    return int(xs.min()), int(ys.min()), int(xs.max()) + 1, int(ys.max()) + 1


def _two_components(mask: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    runs = _x_runs(mask)
    if len(runs) != 2:
        raise ValueError(f"expected two separated puzzle components, found {len(runs)}")
    components = []
    for start, end in runs:
        component = np.zeros_like(mask)
        component[:, start:end] = mask[:, start:end]
        if int(component.sum()) < 5000:
            raise ValueError("puzzle component is too small")
        components.append(component)
    return components[0], components[1]


def _edge_gaps(left: np.ndarray, right: np.ndarray) -> list[float]:
    left_box = _bbox(left)
    right_box = _bbox(right)
    if left_box[0] >= right_box[0]:
        raise ValueError("components are not ordered left to right")
    common_top = max(left_box[1], right_box[1])
    common_bottom = min(left_box[3], right_box[3])
    common_height = common_bottom - common_top
    if common_height < 210:
        raise ValueError("facing edges do not have enough common vertical support")
    if left_box[2] >= right_box[0]:
        raise ValueError("pieces touch or overlap")
    gaps: list[float] = []
    for band in range(BAND_COUNT):
        y = int(round(common_top + (band + 0.5) * common_height / BAND_COUNT))
        rows = []
        for component, box in ((left, left_box), (right, right_box)):
            row = np.flatnonzero(component[y])
            if len(row) == 0:
                raise ValueError("a facing-edge band is missing from the raster")
            # A clean filled silhouette has one continuous horizontal run. A
            # destructive crop may leave a fragment that still has an apparent
            # edge; reject that fragment so the inverse cannot answer from one
            # side of a damaged component.
            if len(row) != int(row[-1] - row[0] + 1):
                raise ValueError("a band contains a raster hole")
            if len(row) < 0.55 * float(box[2] - box[0]):
                raise ValueError("a band contains only a partial piece")
            rows.append(row)
        left_row, right_row = rows
        left_edge = int(left_row[-1])
        right_edge = int(right_row[0])
        gaps.append(float(right_edge - left_edge))
    return gaps


def decision_from_image(image: Image.Image) -> str:
    """Return yes/no using only foreground pixels in the final RGB image."""

    if image.width < 200 or image.height < 200:
        raise ValueError("image is too small for the six-band contract")
    left, right = _two_components(_foreground_mask(image))
    gaps = _edge_gaps(left, right)
    median_gap = sorted(gaps)[len(gaps) // 2]
    deviation = max(abs(gap - median_gap) for gap in gaps)
    if not math.isfinite(deviation):
        raise ValueError("non-finite edge measurement")
    return "yes" if deviation <= FIT_TOLERANCE else "no"
