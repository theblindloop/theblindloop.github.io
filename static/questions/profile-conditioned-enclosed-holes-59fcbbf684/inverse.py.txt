"""Independent pixel-only inverse arm for the enclosed-holes world.

receives ONLY the final PNG. It never imports or reads the renderer, the
analytic scene, prompts, or any renderer state. It recovers the decision by:

1. marking "white-like" pixels (the background and any holes);
2. flood-filling the 4-connected white region reachable from the image border;
3. counting the connected components of white pixels that the border flood
   does NOT reach.

Every such unreachable white component is a hole fully enclosed by shape ink,
so its count equals the analytic hole count exactly on robust scenes.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "enclosed_holes-oracle-0.4.0"

_WHITE_THRESHOLD = 240


def _is_white(arr: np.ndarray, threshold: int = _WHITE_THRESHOLD) -> np.ndarray:
    return arr.min(axis=-1) > threshold


def _border_component_mask(white: np.ndarray) -> np.ndarray:
    """Boolean mask of white pixels 4-connected to any border white pixel."""
    h, w = white.shape
    reached = np.zeros_like(white, dtype=bool)
    stack: list[tuple[int, int]] = []
    for y in (0, h - 1):
        for x in range(w):
            if white[y, x]:
                reached[y, x] = True
                stack.append((x, y))
    for y in range(h):
        for x in (0, w - 1):
            if white[y, x] and not reached[y, x]:
                reached[y, x] = True
                stack.append((x, y))
    while stack:
        x, y = stack.pop()
        for nx, ny in ((x + 1, y), (x - 1, y), (x, y + 1), (x, y - 1)):
            if 0 <= nx < w and 0 <= ny < h and white[ny, nx] and not reached[ny, nx]:
                reached[ny, nx] = True
                stack.append((nx, ny))
    return reached


def _count_components(mask: np.ndarray) -> int:
    """Count 4-connected components of a boolean mask (BFS over true pixels)."""
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    count = 0
    for y in range(h):
        for x in range(w):
            if mask[y, x] and not seen[y, x]:
                count += 1
                seen[y, x] = True
                stack = [(x, y)]
                while stack:
                    cx, cy = stack.pop()
                    for nx, ny in ((cx + 1, cy), (cx - 1, cy), (cx, cy + 1), (cx, cy - 1)):
                        if (
                            0 <= nx < w
                            and 0 <= ny < h
                            and mask[ny, nx]
                            and not seen[ny, nx]
                        ):
                            seen[ny, nx] = True
                            stack.append((nx, ny))
    return count


def decision_from_image(image: Image.Image) -> int:
    """Number of enclosed white holes recovered from the final PNG alone."""
    arr = np.asarray(image.convert("RGB")).astype(np.int16)
    white = _is_white(arr)
    border = _border_component_mask(white)
    holes = white & ~border
    return _count_components(holes)
