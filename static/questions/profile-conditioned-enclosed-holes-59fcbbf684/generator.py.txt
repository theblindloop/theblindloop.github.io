"""Enclosed-holes world renderer.

A single fully-closed filled disk (the "shape") on a white background may
contain a small number of white "holes": pockets of background-colored pixels
fully surrounded by shape ink, with no opening to the outside white background.

Decision : the number of such enclosed white holes, recovered as a pure count of
background components that are not connected to the image border.

The scene is a plain dict so the verifier's AST reachability audit is free of
side effects.

    z = {
        "center": [cx, cy],
        "radius": R,
        "color": "blue",          # saturated, clearly non-white
        "holes": [[hx, hy, hr], ...],   # white holes punched into the disk
        "width": 400,
        "height": 400,
        "background": "white",
    }
"""

from __future__ import annotations

import math
import random
from collections.abc import Iterator

from PIL import Image, ImageDraw

RENDERER_VERSION = "enclosed_holes-0.4.0"

PALETTE = [
    "blue",
    "red",
    "green",
    "purple",
    "teal",
    "crimson",
    "mediumblue",
    "darkgreen",
    "maroon",
    "midnightblue",
]

# Robustness margins (pixels). Everything is kept comfortably above these so
# ordinary antialiasing can never open a wall or pinch a hole shut.
WALL_MIN = 9.0        # ink separating a hole from the outer rim or another hole
HOLE_RADIUS_MIN = 9.0
QUARANTINE_MIN = 6.0  # below this wall thickness the scene is ambiguous/quarantined
MAX_HOLES = 3


def sample_scene(
    rng: random.Random,
    n_holes: int | None = None,
    width: int = 400,
    height: int = 400,
) -> dict:
    """Sample a scene with 0..MAX_HOLES enclosed holes, robustly spaced."""
    color = rng.choice(PALETTE)
    radius = rng.uniform(70.0, 130.0)
    cx = rng.uniform(radius + 14, width - radius - 14)
    cy = rng.uniform(radius + 14, height - radius - 14)
    if n_holes is None:
        n_holes = rng.randint(0, MAX_HOLES)

    holes: list[list[float]] = []
    for _ in range(400):
        if len(holes) >= n_holes:
            break
        hr = rng.uniform(HOLE_RADIUS_MIN, 18.0)
        max_d = radius - hr - WALL_MIN
        if max_d < 4.0:
            break
        angle = rng.uniform(0.0, 2.0 * math.pi)
        d = rng.uniform(0.0, max_d)
        hx = cx + d * math.cos(angle)
        hy = cy + d * math.sin(angle)
        ok = True
        for (ox, oy, orad) in holes:
            if math.hypot(hx - ox, hy - oy) - hr - orad < WALL_MIN - 1.0:
                ok = False
                break
        if ok:
            holes.append([hx, hy, hr])
    return {
        "center": [cx, cy],
        "radius": radius,
        "color": color,
        "holes": holes,
        "width": width,
        "height": height,
        "background": "white",
    }


def render(scene: dict, supersample: int = 4) -> Image.Image:
    """Render I = R(z). Deterministic pure function of the scene dict."""
    width = int(scene["width"])
    height = int(scene["height"])
    background = scene["background"]
    (cx, cy) = (float(scene["center"][0]), float(scene["center"][1]))
    radius = float(scene["radius"])
    color = scene["color"]
    holes = scene["holes"]

    ss = supersample
    img = Image.new("RGB", (width * ss, height * ss), background)
    draw = ImageDraw.Draw(img)
    box = [(cx - radius) * ss, (cy - radius) * ss, (cx + radius) * ss, (cy + radius) * ss]
    draw.ellipse(box, fill=color)
    for (hx, hy, hr) in holes:
        hbox = [(hx - hr) * ss, (hy - hr) * ss, (hx + hr) * ss, (hy + hr) * ss]
        draw.ellipse(hbox, fill=background)
    if ss > 1:
        img = img.resize((width, height), Image.LANCZOS)
    return img


def analytic_gold(scene: dict) -> int:
    """Number of enclosed holes = number of background components in the scene."""
    return len(scene["holes"])


def margin(scene: dict) -> float:
    """Robustness margin: the thinnest ink wall guarding any hole.

    Continuous and deterministic; small values denote scenes whose hole count is
    fragile at pixel scale (quarantined by ``is_quarantined``).
    """
    (cx, cy) = (float(scene["center"][0]), float(scene["center"][1]))
    radius = float(scene["radius"])
    holes = scene["holes"]
    if not holes:
        return float(radius)
    best: float = float("inf")
    for (hx, hy, hr) in holes:
        outer = radius - (math.hypot(hx - cx, hy - cy) + hr)
        best = min(best, outer)
    for i in range(len(holes)):
        for j in range(i + 1, len(holes)):
            (ax, ay, ar) = holes[i]
            (bx, by, br) = holes[j]
            sep = math.hypot(ax - bx, ay - by) - ar - br
            best = min(best, sep)
    return float(best)


def is_quarantined(scene: dict) -> bool:
    """Quarantine scenes whose distinguishing walls are too thin to trust."""
    if margin(scene) < QUARANTINE_MIN:
        return True
    return False


def latent_symmetries(scene: dict) -> Iterator[dict]:
    """No nontrivial render symmetry: every field directly changes the picture.

    Reordering the hole list is a numerically identical parametrization (same
    pixels, same count) but the world keeps a canonical order; declared
    not_applicable.
    """
    return iter(())
