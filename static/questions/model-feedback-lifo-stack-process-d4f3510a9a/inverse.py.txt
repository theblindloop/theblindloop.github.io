"""Independent image-only inverse for the LIFO token stack."""

from __future__ import annotations

from PIL import Image

WIDTH = 2000
HEIGHT = 620
TOKEN_COLORS = (
    (229, 157, 30),
    (27, 164, 201),
    (128, 75, 177),
    (101, 171, 73),
)
ACTION_COLORS = (
    (208, 77, 66),
    (47, 113, 197),
    (122, 71, 171),
)
STACK_CENTERS = ((177, 285), (177, 365), (177, 445))
CARD_CENTERS = tuple(430 + 160 * index for index in range(10))
ACTION_Y = 240
OPERAND_Y = 390

PUSH_OP = 0
POP_OP = 1
ROTATE_OP = 2

LABELS = {
    (0, 1, 2): "amber-cyan-violet",
    (1, 2, 3): "cyan-violet-lime",
    (2, 3, 0): "violet-lime-amber",
    (3, 0, 1): "lime-amber-cyan",
}


def _read_exact(pixel: tuple[int, int, int], palette: tuple[tuple[int, int, int], ...]) -> int:
    for index, color in enumerate(palette):
        if tuple(pixel) == color:
            return index
    raise ValueError(f"unreadable color {pixel}")


def _apply(stack: list[int], op: int, color: int | None = None) -> None:
    if op == PUSH_OP:
        if color not in range(4):
            raise ValueError("PUSH operand is unreadable")
        stack.insert(0, int(color))
    elif op == POP_OP:
        if not stack:
            raise ValueError("POP on empty stack")
        stack.pop(0)
    elif op == ROTATE_OP:
        if not stack:
            raise ValueError("ROTATE on empty stack")
        stack.append(stack.pop(0))
    else:
        raise ValueError("unknown visible action")


def decision_from_image(image: Image.Image) -> str:
    """Read all visible stack inputs, then replay the numbered commands."""
    rgb = image.convert("RGB")
    if rgb.size != (WIDTH, HEIGHT):
        raise ValueError("unexpected image size")
    stack = [
        _read_exact(rgb.getpixel(center), TOKEN_COLORS)
        for center in STACK_CENTERS
    ]
    for x in CARD_CENTERS:
        op = _read_exact(rgb.getpixel((x - 32, ACTION_Y)), ACTION_COLORS)
        if op == PUSH_OP:
            color = _read_exact(rgb.getpixel((x, OPERAND_Y)), TOKEN_COLORS)
            _apply(stack, op, color)
        else:
            _apply(stack, op)
    if tuple(stack) not in LABELS:
        raise ValueError("final stack is not one of the declared patterns")
    return LABELS[tuple(stack)]
