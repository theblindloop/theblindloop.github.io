"""Renderer and analytic arm for a visible LIFO token-stack process."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 2000
HEIGHT = 620

BACKGROUND = (245, 248, 251)
PANEL = (255, 255, 255)
FRAME = (102, 117, 133)
INK = (35, 45, 60)
MUTED = (132, 145, 158)
GRID = (213, 221, 229)
TOKEN_COLORS = (
    (229, 157, 30),   # amber
    (27, 164, 201),   # cyan
    (128, 75, 177),   # violet
    (101, 171, 73),   # lime
)
TOKEN_NAMES = ("amber", "cyan", "violet", "lime")
PUSH = (208, 77, 66)
POP = (47, 113, 197)
ROTATE = (122, 71, 171)
TOP = (210, 137, 39)
CONNECTOR = (157, 169, 181)
WHITE = (255, 255, 255)

PUSH_OP = 0
POP_OP = 1
ROTATE_OP = 2
ACTION_COLORS = (PUSH, POP, ROTATE)
ACTION_NAMES = ("PUSH", "POP", "ROTATE")

START_RECT = (24, 120, 330, 500)
CARD_RECTS = tuple((360 + 160 * index, 120, 500 + 160 * index, 500) for index in range(10))
STACK_CENTERS = ((177, 285), (177, 365), (177, 445))
ACTION_Y = 240
OPERAND_Y = 390
CARD_CENTERS = tuple((x0 + x1) // 2 for x0, _, x1, _ in CARD_RECTS)

PATTERN_LABELS = {
    (0, 1, 2): "amber-cyan-violet",
    (1, 2, 3): "cyan-violet-lime",
    (2, 3, 0): "violet-lime-amber",
    (3, 0, 1): "lime-amber-cyan",
}
TARGET_PATTERNS = tuple(PATTERN_LABELS)


def _font(size: int = 18, bold: bool = False) -> ImageFont.ImageFont:
    filename = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{filename}", size)
    except OSError:
        return ImageFont.load_default()


def _center_text(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str, font, fill) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text(
        (xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2),
        text,
        font=font,
        fill=fill,
    )


def _apply(stack: list[int], command: dict) -> None:
    op = int(command["op"])
    if op == PUSH_OP:
        color = int(command["color"])
        if color not in range(4):
            raise ValueError("invalid PUSH color")
        stack.insert(0, color)
    elif op == POP_OP:
        if not stack:
            raise ValueError("POP on empty stack")
        stack.pop(0)
    elif op == ROTATE_OP:
        if not stack:
            raise ValueError("ROTATE on empty stack")
        stack.append(stack.pop(0))
    else:
        raise ValueError("invalid stack command")


def _simulate(initial_stack: list[int] | tuple[int, ...], commands: list[dict]) -> tuple[int, ...]:
    stack = [int(value) for value in initial_stack]
    for command in commands:
        _apply(stack, command)
    return tuple(stack)


def _random_commands(rng: random.Random) -> list[dict]:
    actions = [PUSH_OP] * 4 + [POP_OP] * 4 + [ROTATE_OP] * 2
    rng.shuffle(actions)
    commands: list[dict] = []
    depth = 3
    for op in actions:
        if op == POP_OP and depth == 1:
            return []
        if op == PUSH_OP:
            commands.append({"op": PUSH_OP, "color": rng.randrange(4)})
            depth += 1
        elif op == POP_OP:
            commands.append({"op": POP_OP})
            depth -= 1
        else:
            commands.append({"op": ROTATE_OP})
    return commands


def sample_scene(seed: int) -> dict:
    """Sample varied stack histories while balancing the four declared outputs."""
    rng = random.Random(int(seed))
    desired = TARGET_PATTERNS[int(seed) % len(TARGET_PATTERNS)]
    for _attempt in range(20000):
        initial = [rng.randrange(4) for _ in range(3)]
        if len(set(initial)) != 3:
            continue
        commands = _random_commands(rng)
        if len(commands) != 10:
            continue
        push_colors = [int(item["color"]) for item in commands if int(item["op"]) == PUSH_OP]
        if len(set(push_colors)) < 3:
            continue
        if _simulate(initial, commands) == desired:
            return {"initial_stack": initial, "commands": commands}
    raise RuntimeError("could not sample a balanced stack scene")


def _draw_token(draw: ImageDraw.ImageDraw, center: tuple[int, int], color: int, size: int = 44) -> None:
    cx, cy = center
    draw.rounded_rectangle(
        (cx - size, cy - size, cx + size, cy + size),
        radius=12,
        fill=TOKEN_COLORS[int(color)],
        outline=INK,
        width=3,
    )


def _arrowhead(draw: ImageDraw.ImageDraw, tip: tuple[int, int], color, direction: int = 1) -> None:
    x, y = tip
    if direction > 0:
        points = ((x, y), (x - 16, y - 10), (x - 16, y + 10))
    else:
        points = ((x, y), (x + 16, y - 10), (x + 16, y + 10))
    draw.polygon(points, fill=color)


def _draw_action_badge(draw: ImageDraw.ImageDraw, center: tuple[int, int], op: int) -> None:
    cx, cy = center
    color = ACTION_COLORS[int(op)]
    draw.rounded_rectangle((cx - 42, cy - 40, cx + 42, cy + 40), radius=12, fill=color, outline=INK, width=2)
    if op == PUSH_OP:
        draw.line((cx - 23, cy, cx + 23, cy), fill=WHITE, width=8)
        draw.line((cx, cy - 23, cx, cy + 23), fill=WHITE, width=8)
    elif op == POP_OP:
        draw.line((cx - 24, cy, cx + 24, cy), fill=WHITE, width=8)
    else:
        draw.arc((cx - 25, cy - 24, cx + 25, cy + 24), start=35, end=310, fill=WHITE, width=7)
        _arrowhead(draw, (cx + 22, cy - 18), WHITE, direction=1)


def _draw_connector(draw: ImageDraw.ImageDraw, start_x: int, end_x: int) -> None:
    y = 320
    draw.line((start_x, y, end_x, y), fill=CONNECTOR, width=4)
    _arrowhead(draw, (end_x, y), CONNECTOR, direction=1)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(22, bold=True)
    label_font = _font(17)
    small_font = _font(15)
    card_font = _font(15, bold=True)

    draw.text((28, 43), "LIFO STACK — APPLY CARDS LEFT TO RIGHT", fill=INK, font=title_font)
    draw.text((1630, 47), "the gold arrow marks TOP", fill=MUTED, font=small_font)
    draw.line((28, 84, 1970, 84), fill=(210, 218, 226), width=4)

    draw.rounded_rectangle(START_RECT, radius=18, fill=PANEL, outline=FRAME, width=3)
    draw.text((48, 151), "START", fill=INK, font=_font(20, bold=True))
    draw.text((48, 184), "top to bottom", fill=MUTED, font=small_font)
    draw.text((48, 454), "3 TOKENS", fill=INK, font=small_font)
    draw.line((177, 218, 177, 245), fill=TOP, width=7)
    draw.polygon(((177, 208), (165, 225), (189, 225)), fill=TOP)
    draw.text((195, 198), "TOP", fill=TOP, font=small_font)
    for index, center in enumerate(STACK_CENTERS):
        _draw_token(draw, center, int(scene["initial_stack"][index]))
        _center_text(draw, (278, center[1]), str(index + 1), label_font, FRAME)

    _draw_connector(draw, START_RECT[2] + 6, CARD_RECTS[0][0] - 6)
    for index, rect in enumerate(CARD_RECTS):
        x0, y0, x1, y1 = rect
        cx = (x0 + x1) // 2
        draw.rounded_rectangle(rect, radius=16, fill=PANEL, outline=FRAME, width=3)
        draw.text((x0 + 13, y0 + 17), str(index + 1), fill=FRAME, font=small_font)
        command = scene["commands"][index]
        op = int(command["op"])
        _draw_action_badge(draw, (cx, ACTION_Y), op)
        _center_text(draw, (cx, 305), ACTION_NAMES[op], card_font, INK)
        if op == PUSH_OP:
            _draw_token(draw, (cx, OPERAND_Y), int(command["color"]), size=30)
            _center_text(draw, (cx, 462), TOKEN_NAMES[int(command["color"])], small_font, MUTED)
        elif op == POP_OP:
            draw.line((cx - 24, OPERAND_Y, cx + 24, OPERAND_Y), fill=INK, width=4)
            _center_text(draw, (cx, 430), "top out", small_font, MUTED)
        else:
            draw.arc((cx - 27, OPERAND_Y - 27, cx + 27, OPERAND_Y + 27), start=30, end=330, fill=INK, width=4)
            _arrowhead(draw, (cx + 25, OPERAND_Y - 14), INK, direction=1)
            _center_text(draw, (cx, 462), "top → bottom", small_font, MUTED)
        if index < len(CARD_RECTS) - 1:
            _draw_connector(draw, x1 + 4, CARD_RECTS[index + 1][0] - 4)
    _draw_connector(draw, CARD_RECTS[-1][2] + 4, 1965)
    draw.text((1920, 350), "END", fill=FRAME, font=small_font)
    return image


def _final_stack(scene: dict) -> tuple[int, ...]:
    return _simulate(scene["initial_stack"], scene["commands"])


def analytic_gold(scene: dict) -> str:
    return PATTERN_LABELS[_final_stack(scene)]


def margin(scene: dict) -> float:
    return 30.0


def is_quarantined(scene: dict) -> bool:
    if not isinstance(scene, dict):
        return True
    initial = scene.get("initial_stack")
    commands = scene.get("commands")
    if not isinstance(initial, list) or len(initial) != 3 or any(int(value) not in range(4) for value in initial):
        return True
    if not isinstance(commands, list) or len(commands) != 10:
        return True
    stack = [int(value) for value in initial]
    try:
        for command in commands:
            if not isinstance(command, dict) or "op" not in command:
                return True
            _apply(stack, command)
    except (KeyError, TypeError, ValueError):
        return True
    return len(stack) != 3 or tuple(stack) not in PATTERN_LABELS


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
