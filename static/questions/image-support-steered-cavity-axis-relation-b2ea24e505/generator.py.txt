"""Deterministic renderer and analytic label for the two-chamber world."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 800
HEIGHT = 600
BACKGROUND = (248, 246, 240)

_PALETTES = (
    ((42, 70, 105), (20, 36, 59)),
    ((72, 94, 73), (34, 54, 38)),
    ((111, 66, 77), (65, 32, 43)),
    ((115, 81, 43), (68, 43, 19)),
    ((76, 66, 112), (40, 32, 70)),
)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced, well-separated pair of interior chambers."""

    seed = int(seed)
    rng = random.Random(seed)
    horizontal = seed % 2 == 0

    # The shell nearly fills the page, while the two chambers are broad and
    # close enough that an overlapping ablation can break their shared
    # enclosing structure. The shell remains a single thick foreground mass.
    shell_width = rng.randint(740, 760)
    shell_height = rng.randint(550, 570)
    translation_x = rng.randint(-12, 12)
    translation_y = rng.randint(-8, 8)

    if horizontal:
        separation_x = rng.randint(160, 180)
        separation_y = rng.randint(34, 78)
        radius_x_a = rng.randint(58, 66)
        radius_x_b = rng.randint(58, 66)
        radius_y_a = rng.randint(52, 64)
        radius_y_b = rng.randint(52, 64)
    else:
        separation_x = rng.randint(34, 78)
        separation_y = rng.randint(160, 180)
        radius_x_a = rng.randint(52, 64)
        radius_x_b = rng.randint(52, 64)
        radius_y_a = rng.randint(58, 66)
        radius_y_b = rng.randint(58, 66)

    # Independent jitter prevents the answer from being encoded by a fixed
    # template while keeping both chambers comfortably inside the shell.
    a_x = -separation_x // 2 + rng.randint(-9, 9)
    b_x = separation_x // 2 + rng.randint(-9, 9)
    a_y = -separation_y // 2 + rng.randint(-8, 8)
    b_y = separation_y // 2 + rng.randint(-8, 8)
    if horizontal:
        # Keep the small vertical offset genuinely small without tying it to
        # the answer identity or to a fixed top/bottom location.
        a_y = rng.randint(-separation_y // 2, separation_y // 2)
        b_y = rng.randint(-separation_y // 2, separation_y // 2)
    else:
        # Keep the vertical pair close enough for a shared ablation support,
        # but leave an explicit foreground neck between the two cavities.
        a_y = -separation_y // 2
        b_y = separation_y // 2
        a_x = rng.randint(-separation_x // 2, separation_x // 2)
        b_x = rng.randint(-separation_x // 2, separation_x // 2)

    hole_a_rx = radius_x_a
    hole_a_ry = radius_y_a
    hole_b_rx = radius_x_b
    hole_b_ry = radius_y_b

    fill_color, edge_color = _PALETTES[(seed * 7 + 3) % len(_PALETTES)]
    return {
        "hole_a_x": int(a_x),
        "hole_a_y": int(a_y),
        "hole_b_x": int(b_x),
        "hole_b_y": int(b_y),
        "hole_a_rx": int(hole_a_rx),
        "hole_a_ry": int(hole_a_ry),
        "hole_b_rx": int(hole_b_rx),
        "hole_b_ry": int(hole_b_ry),
        "shell_width": int(shell_width),
        "shell_height": int(shell_height),
        "translation_x": int(translation_x),
        "translation_y": int(translation_y),
        "fill_color": list(fill_color),
        "edge_color": list(edge_color),
    }


def _center(scene: dict[str, Any]) -> tuple[float, float]:
    local_x = (float(scene["hole_a_x"]) + float(scene["hole_b_x"])) / 2.0
    local_y = (float(scene["hole_a_y"]) + float(scene["hole_b_y"])) / 2.0
    return (
        WIDTH / 2.0 + float(scene["translation_x"]) + local_x,
        HEIGHT / 2.0 + float(scene["translation_y"]) + local_y,
    )


def _hole_center(scene: dict[str, Any], prefix: str) -> tuple[int, int]:
    center_x, center_y = _center(scene)
    local_center_x = float(scene[f"{prefix}_x"])
    local_center_y = float(scene[f"{prefix}_y"])
    local_mid_x = (float(scene["hole_a_x"]) + float(scene["hole_b_x"])) / 2.0
    local_mid_y = (float(scene["hole_a_y"]) + float(scene["hole_b_y"])) / 2.0
    return (
        int(round(center_x + local_center_x - local_mid_x)),
        int(round(center_y + local_center_y - local_mid_y)),
    )


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    center_x, center_y = _center(scene)
    shell_width = int(scene["shell_width"])
    shell_height = int(scene["shell_height"])
    left = int(round(center_x - shell_width / 2.0))
    top = int(round(center_y - shell_height / 2.0))
    right = int(round(center_x + shell_width / 2.0))
    bottom = int(round(center_y + shell_height / 2.0))
    radius = min(92, shell_height // 3)
    edge_color = tuple(int(value) for value in scene["edge_color"])
    fill_color = tuple(int(value) for value in scene["fill_color"])

    # A dark rim and inset fill make the single shell legible without adding
    # any disconnected labels or answer-specific marks.
    draw.rounded_rectangle(
        (left, top, right, bottom), radius=radius, fill=edge_color
    )
    inset = 6
    draw.rounded_rectangle(
        (left + inset, top + inset, right - inset, bottom - inset),
        radius=max(12, radius - inset),
        fill=fill_color,
    )

    for prefix in ("hole_a", "hole_b"):
        hole_x, hole_y = _hole_center(scene, prefix)
        rx = int(scene[f"{prefix}_rx"])
        ry = int(scene[f"{prefix}_ry"])
        draw.ellipse((hole_x - rx, hole_y - ry, hole_x + rx, hole_y + ry), fill=BACKGROUND)

    return image


def margin(scene: dict[str, Any]) -> float:
    horizontal = abs(float(scene["hole_a_x"]) - float(scene["hole_b_x"]))
    vertical = abs(float(scene["hole_a_y"]) - float(scene["hole_b_y"]))
    return horizontal - vertical


def analytic_gold(scene: dict[str, Any]) -> str:
    horizontal = abs(float(scene["hole_a_x"]) - float(scene["hole_b_x"]))
    vertical = abs(float(scene["hole_a_y"]) - float(scene["hole_b_y"]))
    return "horizontal" if horizontal > vertical else "vertical"


def is_quarantined(scene: dict[str, Any]) -> bool:
    return abs(margin(scene)) <= 18.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # The exposed center coordinates are label inputs, so this world declares
    # no latent aliases. Every sampled transformation that changes them is
    # observable in the final pixels.
    return []
