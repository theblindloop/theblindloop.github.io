"""Independent pixel-only oracle for enclosed chamber centroid separation."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _small_rgb(image: Image.Image) -> np.ndarray:
    rgb = image.convert("RGB")
    if rgb.width >= 400 and rgb.height >= 300:
        rgb = rgb.resize((rgb.width // 4, rgb.height // 4), Image.Resampling.BOX)
    return np.asarray(rgb, dtype=np.uint8)


def _foreground(rgb: np.ndarray) -> np.ndarray:
    # The shell can occupy most of the page, so use corner patches rather
    # than a long border whose median could be contaminated by the shell.
    corner = 6
    border = np.concatenate(
        (
            rgb[:corner, :corner].reshape(-1, 3),
            rgb[:corner, -corner:].reshape(-1, 3),
            rgb[-corner:, :corner].reshape(-1, 3),
            rgb[-corner:, -corner:].reshape(-1, 3),
        ),
        axis=0,
    )
    background = np.median(border, axis=0).astype(np.float32)
    distance = np.linalg.norm(rgb.astype(np.float32) - background, axis=2)
    return distance > 24.0


def _outside_background(background: np.ndarray) -> np.ndarray:
    height, width = background.shape
    outside = np.zeros_like(background, dtype=bool)
    queue: deque[tuple[int, int]] = deque()
    for x in range(width):
        for y in (0, height - 1):
            if background[y, x] and not outside[y, x]:
                outside[y, x] = True
                queue.append((y, x))
    for y in range(height):
        for x in (0, width - 1):
            if background[y, x] and not outside[y, x]:
                outside[y, x] = True
                queue.append((y, x))
    while queue:
        y, x = queue.popleft()
        for next_y, next_x in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if (
                0 <= next_y < height
                and 0 <= next_x < width
                and background[next_y, next_x]
                and not outside[next_y, next_x]
            ):
                outside[next_y, next_x] = True
                queue.append((next_y, next_x))
    return outside


def _enclosed_centroids(mask: np.ndarray) -> list[tuple[float, float, int]]:
    # The public question assumes one intact, high-fill connected shell. A
    # broad ablation that materially removes that shell is therefore an
    # abstention, not a locally guessed answer.
    if float(mask.mean()) < 0.795:
        raise ValueError("connected shell is materially underfilled")
    height, width = mask.shape
    outside = _outside_background(~mask)
    remaining = (~mask) & ~outside
    seen = np.zeros_like(remaining, dtype=bool)
    components: list[tuple[float, float, int]] = []
    for y, x in np.argwhere(remaining):
        y = int(y)
        x = int(x)
        if seen[y, x]:
            continue
        queue: deque[tuple[int, int]] = deque([(y, x)])
        seen[y, x] = True
        sum_x = 0
        sum_y = 0
        count = 0
        while queue:
            current_y, current_x = queue.popleft()
            sum_x += current_x
            sum_y += current_y
            count += 1
            for next_y, next_x in (
                (current_y - 1, current_x),
                (current_y + 1, current_x),
                (current_y, current_x - 1),
                (current_y, current_x + 1),
            ):
                if (
                    0 <= next_y < height
                    and 0 <= next_x < width
                    and remaining[next_y, next_x]
                    and not seen[next_y, next_x]
                ):
                    seen[next_y, next_x] = True
                    queue.append((next_y, next_x))
        if count >= 20:
            components.append((sum_x / count, sum_y / count, count))
    return components


def decision_from_image(image: Image.Image) -> str:
    rgb = _small_rgb(image)
    foreground = _foreground(rgb)
    chambers = _enclosed_centroids(foreground)
    if len(chambers) != 2:
        raise ValueError(f"expected exactly two enclosed chambers, found {len(chambers)}")
    first, second = chambers
    horizontal = abs(first[0] - second[0])
    vertical = abs(first[1] - second[1])
    if abs(horizontal - vertical) <= 0.5:
        raise ValueError("chamber separation axes are too close to distinguish")
    return "horizontal" if horizontal > vertical else "vertical"
