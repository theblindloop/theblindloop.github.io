"""Tightest-arc measurement world: which colored semicircular arc is tightest.

Latent scene z carries three concentric, same-color-distinct colored semicircular
arcs that all share one common center and one common angular span (a semicircle),
but each has a different radius (elliptic bbox half-size). The decision is a pure
measurement: recover each arc's radius from the raster by fitting a circle to its
colored annulus, then select the color of the arc with the SMALLEST radius
(i.e. the tightest bend / strongest curvature).

The three arcs are concentric so their colored masks never overlap: each color
occupies its own annular band. Which color owns the smallest radius is
counterbalanced by shuffling colors and by randomising the semicircle opening
direction, so neither absolute position, opening, nor color alone predicts the
answer; only the recovered scalar radius does.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "tightest_arc-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
BACKGROUND = (250, 250, 250)

WIDTH = 480
HEIGHT = 340
CENTER = (240, 205)
ARC_WIDTH = 8                 # stroke width of every arc (identical)
OPENINGS = [(180.0, 360.0), (0.0, 180.0), (90.0, 270.0)]
RADIUS_MIN = 50
RADIUS_MAX = 110
MIN_RADIUS_GAP = 20           # pairwise radius gap required for a decidable selection
QUARANTINE_PX = 12            # below this smallest-vs-runner gap the scene is quarantined


def sample_scene(seed: int) -> dict:
    """Sample a deterministic scene with a guaranteed, counterbalanced extremum."""
    rng = random.Random(seed * 1049 + 11)
    ncols = len(PALETTE)
    # Three distinct integer radii with pairwise gap at least MIN_RADIUS_GAP.
    while True:
        radii = sorted(rng.sample(range(RADIUS_MIN, RADIUS_MAX + 1), ncols))
        if all(radii[i + 1] - radii[i] >= MIN_RADIUS_GAP for i in range(ncols - 1)):
            break
    colors = list(PALETTE)
    rng.shuffle(colors)
    start, end = rng.choice(OPENINGS)
    cx, cy = CENTER
    arcs = []
    for color, radius in zip(colors, radii):
        arcs.append(
            {
                "color": color,
                "radius": int(radius),
                "cx": cx,
                "cy": cy,
                "start": float(start),
                "end": float(end),
                "width": int(ARC_WIDTH),
            }
        )
    return {
        "arcs": arcs,
        "center": [cx, cy],
        "width": WIDTH,
        "height": HEIGHT,
        "background": "light",
    }


def render(scene: dict) -> Image.Image:
    """Render I = R(z): three concentric colored semicircular arcs."""
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    for arc in scene["arcs"]:
        r = arc["radius"]
        box = [arc["cx"] - r, arc["cy"] - r, arc["cx"] + r, arc["cy"] + r]
        draw.arc(box, arc["start"], arc["end"], fill=COLOR_RGB[arc["color"]], width=arc["width"])
    return img


def _smallest_arc(arcs: list[dict]) -> tuple[dict, float, float]:
    """Return (smallest-radius arc, its radius, smallest-vs-runner radius gap)."""
    keyed = sorted(arcs, key=lambda arc: arc["radius"])
    top, second = keyed[0], keyed[1]
    return top, float(top["radius"]), float(second["radius"] - top["radius"])


def analytic_gold(scene: dict) -> str:
    """The color of the arc with the smallest radius (tightest curvature)."""
    top, _radius, _gap = _smallest_arc(scene["arcs"])
    return top["color"]


def margin(scene: dict) -> float:
    """Verifiability margin: smallest-vs-runner radius gap in pixels."""
    _top, _radius, gap = _smallest_arc(scene["arcs"])
    return float(gap)


def is_quarantined(scene: dict) -> bool:
    """Near-boundary: the smallest-vs-runner radius gap is finite-resolution small."""
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list:
    """not_applicable: no nontrivial pixel-identical latent symmetry.

    Every latent field (each arc's color label, radius, and the shared center,
    angular span, and stroke width) is directly rendered as visible ink. Swapping
    two arcs' radii changes which color is correct, and changing the common center,
    opening, or stroke width turns the raster into a different image. There is no
    hidden field that could change the gold without changing the raster.
    """
    return []
