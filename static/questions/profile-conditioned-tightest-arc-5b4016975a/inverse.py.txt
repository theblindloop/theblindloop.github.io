"""Independent pixel-only inverse arm for the tightest-arc world.

This module receives ONLY the rasterized PNG (plus the public palette, which is
the world's fixed naming convention) and re-measures which colored arc has the
smallest radius / tightest curvature. It shares no geometry with renderer.render()
and never imports the renderer, prompts, generator, verifier, or any analytical
gold code.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "tightest_arc-oracle-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
BACKGROUND = (250, 250, 250)


def _nearest_color(arr: np.ndarray) -> np.ndarray:
    """Label every pixel to its nearest palette entry (arc colors + background)."""
    refs = np.stack(
        [np.array(COLOR_RGB[c], dtype=np.float32) for c in PALETTE]
        + [np.array(BACKGROUND, dtype=np.float32)],
        axis=0,
    )
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def _fit_circle_mask(mask: np.ndarray) -> float:
    """Return the radius of a colored arc measured from the recovered center.

    The arc's circle center is recovered from a centroid-shifted Kasa least
    squares on the annulus mask (the linear terms give the centre directly),
    then the arc radius is measured as the mean distance of its pixels from that
    centre. This is numerically stable for a partial (semicircular) annulus where
    the raw Kasa radius estimate is biased, and it returns the true mean radius
    whether or not the arc is a full circle.
    """
    ys, xs = np.nonzero(mask)
    if len(ys) < 24:
        return -1.0
    x = xs.astype(np.float64)
    y = ys.astype(np.float64)
    cx = x.mean()
    cy = y.mean()
    a = np.stack([x - cx, y - cy, np.ones_like(x)], axis=1)
    b = (x - cx) ** 2 + (y - cy) ** 2
    center_r2 = -1.0
    try:
        sol, *_ = np.linalg.lstsq(a, b, rcond=None)
        center_x = cx - sol[0] / 2.0
        center_y = cy - sol[1] / 2.0
    except np.linalg.LinAlgError:
        center_x, center_y = cx, cy
    dist = np.hypot(x - center_x, y - center_y)
    if dist.size == 0:
        return -1.0
    return float(dist.mean())


def decision_from_image(image: Image.Image) -> str:
    """Return the color of the smallest-radius (tightest) arc from pixels alone."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _nearest_color(arr)
    best_color = PALETTE[0]
    best_radius = float("inf")
    for idx, color in enumerate(PALETTE):
        mask = labels == idx
        radius = _fit_circle_mask(mask)
        if radius >= 0.0 and radius < best_radius:
            best_radius = radius
            best_color = color
    return best_color
