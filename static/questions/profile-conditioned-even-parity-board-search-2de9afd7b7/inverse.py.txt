from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[tuple[int, int, int, int, int]] = []
    for y, x in zip(*np.nonzero(mask & ~seen)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            yy, xx = queue.popleft()
            xs.append(xx)
            ys.append(yy)
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        output.append((min(xs), min(ys), max(xs), max(ys), len(xs)))
    return output


def _read_board(arr: np.ndarray, box: tuple[int, int, int, int, int]) -> list[list[int]]:
    x0, y0, x1, y1, _ = box
    if not (110 <= x1 - x0 <= 170 and 110 <= y1 - y0 <= 170):
        raise ValueError("grid component has an implausible extent")
    board: list[list[int]] = []
    for row in range(4):
        values: list[int] = []
        for col in range(4):
            cx = x0 + (col + 0.5) * (x1 - x0) / 4.0
            cy = y0 + (row + 0.5) * (y1 - y0) / 4.0
            radius = max(3, int(min(x1 - x0, y1 - y0) / 18))
            patch = arr[int(cy) - radius : int(cy) + radius + 1, int(cx) - radius : int(cx) + radius + 1]
            blue = (patch[:, :, 2] > 130) & (patch[:, :, 0] < 100) & (patch[:, :, 1] > 70)
            values.append(int(float(blue.mean()) > 0.55))
        board.append(values)
    return board


def _violations(board: list[list[int]]) -> int:
    return sum(sum(row) % 2 for row in board) + sum(
        sum(board[row][col] for row in range(4)) % 2 for col in range(4)
    )


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    dark = (arr[:, :, 0] < 80) & (arr[:, :, 1] < 90) & (arr[:, :, 2] < 110)
    grids = [box for box in _components(dark) if box[4] > 900 and 110 <= box[2] - box[0] <= 170 and 110 <= box[3] - box[1] <= 170]
    grids.sort(key=lambda item: item[0])
    if len(grids) != 4:
        raise ValueError(f"expected four visible board grids, found {len(grids)}")
    boards = [_read_board(arr, box) for box in grids]
    winners = [index for index, board in enumerate(boards) if _violations(board) == 0]
    if len(winners) != 1:
        raise ValueError(f"pixel evidence does not identify a unique parity winner: {winners}")
    return ("A", "B", "C", "D")[winners[0]]
