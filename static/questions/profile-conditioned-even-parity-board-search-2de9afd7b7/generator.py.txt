from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 760, 430
BLUE = (35, 111, 190)
DARK = (45, 53, 68)
LIGHT = (246, 248, 251)
PANEL_X = (32, 216, 400, 584)
BOARD_Y = 175
CELL = 34
LABELS = ("A", "B", "C", "D")


def _violations_for_board(board: list[list[int]]) -> int:
    return sum(sum(row) % 2 for row in board) + sum(
        sum(board[r][c] for r in range(4)) % 2 for c in range(4)
    )


def _all_equal_fill_boards() -> tuple[list[list[list[int]]], list[list[list[int]]]]:
    valid: list[list[list[int]]] = []
    invalid: list[list[list[int]]] = []
    for chosen in itertools.combinations(range(16), 8):
        bits = [0] * 16
        for index in chosen:
            bits[index] = 1
        board = [bits[row * 4 : row * 4 + 4] for row in range(4)]
        (valid if _violations_for_board(board) == 0 else invalid).append(board)
    return valid, invalid


_VALID, _INVALID = _all_equal_fill_boards()


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    winner = int(seed) % 4
    boards: list[list[list[int]]] = []
    used: set[tuple[int, ...]] = set()
    for panel in range(4):
        pool = _VALID if panel == winner else _INVALID
        while True:
            board = rng.choice(pool)
            key = tuple(value for row in board for value in row)
            if key not in used:
                used.add(key)
                boards.append([list(row) for row in board])
                break
    return {"boards": boards}


def render(scene: dict) -> Image.Image:
    boards = scene["boards"]
    image = Image.new("RGB", (WIDTH, HEIGHT), LIGHT)
    draw = ImageDraw.Draw(image)
    font_path = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
    bold_path = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
    title_font = ImageFont.truetype(bold_path, 25)
    body_font = ImageFont.truetype(font_path, 18)
    label_font = ImageFont.truetype(bold_path, 23)
    draw.text((28, 24), "EVEN ROW-AND-COLUMN SEARCH", fill=DARK, font=title_font)
    draw.text(
        (28, 66),
        "Which board has an even number of blue cells in every row",
        fill=DARK,
        font=body_font,
    )
    draw.text((28, 92), "and every column?  A-D run left to right.", fill=DARK, font=body_font)
    for panel, board in enumerate(boards):
        x0 = PANEL_X[panel]
        draw.text((x0 + 58, 137), LABELS[panel], fill=DARK, font=label_font)
        for row in range(4):
            for col in range(4):
                left = x0 + col * CELL
                top = BOARD_Y + row * CELL
                if int(board[row][col]):
                    draw.rectangle((left + 3, top + 3, left + CELL - 3, top + CELL - 3), fill=BLUE)
        for k in range(5):
            xx = x0 + k * CELL
            yy = BOARD_Y + k * CELL
            draw.line((xx, BOARD_Y, xx, BOARD_Y + 4 * CELL), fill=DARK, width=2)
            draw.line((x0, yy, x0 + 4 * CELL, yy), fill=DARK, width=2)
    draw.text((28, 350), "Count blue cells separately in each visible row and column.", fill=DARK, font=body_font)
    return image


def analytic_gold(scene: dict) -> str:
    boards = scene["boards"]
    winners = [i for i, board in enumerate(boards) if _violations_for_board(board) == 0]
    if len(winners) != 1:
        raise ValueError(f"expected one parity winner, got {winners}")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    boards = scene["boards"]
    scores = sorted(_violations_for_board(board) for board in boards)
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    boards = scene["boards"]
    return len(boards) != 4 or sum(_violations_for_board(board) == 0 for board in boards) != 1 or margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
