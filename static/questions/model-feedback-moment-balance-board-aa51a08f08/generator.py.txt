"""Renderer and analytic inverse specification for the weighted-pivot world.

The scene contains two visually separated rows of five disk loads.  The same
five numbered distances and the same multiset of size-coded weights occur on
both sides of a central pivot, but the assignments differ.  The answer is the
sign of the difference between the two sums of (weight * distance).
"""

from __future__ import annotations

import itertools
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

RENDERER_VERSION = "moment_balance_board-renderer-0.1.0"

WIDTH = 1100
HEIGHT = 760
PIVOT_X = 550
BEAM_Y = 430
PITCH = 72
LOAD_Y = 292
DISTANCE_CHOICES = tuple(range(1, 7))
LOAD_COUNT = 5
WEIGHT_MULTISET = (1, 1, 2, 2, 3)
WEIGHT_RADII = {1: 13, 2: 20, 3: 27}
WEIGHT_DIAMETERS = {weight: 2 * radius + 1 for weight, radius in WEIGHT_RADII.items()}

BACKGROUND = (249, 249, 246)
INK = (38, 49, 58)
SUBTLE = (155, 164, 168)
GUIDE = (211, 218, 219)
LOAD_COLOR = (25, 91, 113)
KEY_COLOR = (92, 119, 126)
PIVOT_COLOR = (220, 151, 35)
WHITE = (255, 255, 255)


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _centered_text(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font, fill) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2), text, font=font, fill=fill)


def _moment(distances: Iterable[int], weights: Iterable[int]) -> int:
    return sum(int(distance) * int(weight) for distance, weight in zip(distances, weights, strict=True))


def _validate_scene_arrays(scene: dict) -> tuple[list[int], list[int], list[int]]:
    distances = [int(value) for value in scene["distances"]]
    left_weights = [int(value) for value in scene["left_weights"]]
    right_weights = [int(value) for value in scene["right_weights"]]
    if len(distances) != LOAD_COUNT or len(left_weights) != LOAD_COUNT or len(right_weights) != LOAD_COUNT:
        raise ValueError("the board needs five distances and five weights on each side")
    if distances != sorted(set(distances)) or any(value not in DISTANCE_CHOICES for value in distances):
        raise ValueError("distances must be five distinct numbered ticks")
    if any(value not in WEIGHT_RADII for value in left_weights + right_weights):
        raise ValueError("weights must be 1, 2, or 3")
    if sorted(left_weights) != list(WEIGHT_MULTISET) or sorted(right_weights) != list(WEIGHT_MULTISET):
        raise ValueError("both sides must carry the declared size multiset")
    return distances, left_weights, right_weights


def _candidate_pairs(
    distances: list[int], target: str, max_gap: int = 3
) -> list[tuple[tuple[int, ...], tuple[int, ...]]]:
    permutations = sorted(set(itertools.permutations(WEIGHT_MULTISET)))
    pairs: list[tuple[tuple[int, ...], tuple[int, ...]]] = []
    for left in permutations:
        left_total = _moment(distances, left)
        for right in permutations:
            right_total = _moment(distances, right)
            delta = left_total - right_total
            if target == "left" and 1 <= delta <= max_gap:
                pairs.append((left, right))
            elif target == "right" and -max_gap <= delta <= -1:
                pairs.append((left, right))
    return pairs


def sample_scene(seed: int) -> dict:
    """Sample a balanced, deterministic board from an integer seed.

    Odd/even seeds alternate the two headline classes.  A sparse set of seeds
    produces a true equality boundary for the quarantined tie class.  The
    target is used only to choose a pair; it is not stored in the returned
    scene and cannot become a gold-only field.
    """

    rng = random.Random(int(seed))
    distances = sorted(rng.sample(list(DISTANCE_CHOICES), LOAD_COUNT))
    if int(seed) % 23 == 0:
        base = tuple(rng.choice(sorted(set(itertools.permutations(WEIGHT_MULTISET)))))
        left_weights = list(base)
        right_weights = list(base)
    else:
        target = "left" if int(seed) % 2 == 0 else "right"
        pairs = _candidate_pairs(distances, target)
        if not pairs:
            raise RuntimeError("no near-boundary permutation pair exists")
        left_weights, right_weights = rng.choice(pairs)
        left_weights = list(left_weights)
        right_weights = list(right_weights)
    return {
        "distances": distances,
        "left_weights": left_weights,
        "right_weights": right_weights,
    }


def render(scene: dict) -> Image.Image:
    distances, left_weights, right_weights = _validate_scene_arrays(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(28)
    body_font = _font(18)
    small_font = _font(15)
    number_font = _font(20)

    draw.text((42, 24), "WEIGHTED PIVOT BOARD", font=title_font, fill=INK)
    draw.text(
        (42, 61),
        "Pair each disk's size with its numbered distance from the center.",
        font=body_font,
        fill=INK,
    )

    # A separate-color size key makes the numeric operand of every disk
    # visible, while keeping key marks out of the load mask used by oracle.py.
    draw.text((42, 105), "SIZE KEY", font=small_font, fill=INK)
    key_x = 150
    for weight, label in ((1, "small = 1"), (2, "medium = 2"), (3, "large = 3")):
        radius = WEIGHT_RADII[weight]
        draw.ellipse(
            (key_x - radius, 123 - radius, key_x + radius, 123 + radius),
            fill=KEY_COLOR,
        )
        draw.text((key_x + 38, 112), label, font=small_font, fill=INK)
        key_x += 182

    draw.text((42, 205), "LOADS", font=small_font, fill=INK)
    draw.line((90, BEAM_Y, 1010, BEAM_Y), fill=INK, width=16)
    draw.line((90, BEAM_Y - 10, 1010, BEAM_Y - 10), fill=WHITE, width=2)

    for distance in DISTANCE_CHOICES:
        for x in (PIVOT_X - distance * PITCH, PIVOT_X + distance * PITCH):
            draw.line((x, LOAD_Y + 34, x, BEAM_Y - 17), fill=GUIDE, width=2)
            draw.line((x, BEAM_Y - 20, x, BEAM_Y + 24), fill=SUBTLE, width=4)
            _centered_text(draw, (x, BEAM_Y + 49), str(distance), number_font, INK)

    # Draw the pivot after the beam and ticks so it is a unique, recoverable
    # marker rather than an implicit coordinate in the analytic code.
    draw.polygon(
        [(PIVOT_X, BEAM_Y - 55), (PIVOT_X - 28, BEAM_Y - 16), (PIVOT_X, BEAM_Y + 22), (PIVOT_X + 28, BEAM_Y - 16)],
        fill=PIVOT_COLOR,
    )
    _centered_text(draw, (PIVOT_X, BEAM_Y + 49), "0", number_font, INK)
    _centered_text(draw, (PIVOT_X - 250, 505), "LEFT", title_font, INK)
    _centered_text(draw, (PIVOT_X + 250, 505), "RIGHT", title_font, INK)

    # All load circles share one exact fill and remain separate from the beam;
    # their radii are the three visible size values from the key.
    for side, weights in ((-1, left_weights), (1, right_weights)):
        for distance, weight in zip(distances, weights, strict=True):
            x = PIVOT_X + side * distance * PITCH
            radius = WEIGHT_RADII[weight]
            draw.ellipse((x - radius, LOAD_Y - radius, x + radius, LOAD_Y + radius), fill=LOAD_COLOR)

    draw.text(
        (42, 676),
        "For each side: add (size value × distance). The beam drawing itself is not the answer.",
        font=body_font,
        fill=INK,
    )
    return image


def analytic_gold(scene: dict) -> str:
    distances, left_weights, right_weights = _validate_scene_arrays(scene)
    left_total = _moment(distances, left_weights)
    right_total = _moment(distances, right_weights)
    if left_total > right_total:
        return "left"
    if right_total > left_total:
        return "right"
    return "tie"


def margin(scene: dict) -> float:
    distances, left_weights, right_weights = _validate_scene_arrays(scene)
    return float(_moment(distances, left_weights) - _moment(distances, right_weights))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) == 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # No latent aliases are declared.  Every label input is visibly rendered,
    # and swapping or changing any one of them changes pixels.
    return []
