"""Independent image-only inverse for the weighted-pivot board.

This module deliberately duplicates the small amount of scene geometry needed
to interpret pixels.  It does not import renderer, prompts, generate, verify,
or any analytic-gold code.  It finds the exact-color load components, classifies
their visible diameters using the published size key, locates the gold pivot,
recovers numbered distances from pixel coordinates, and computes the two
weighted sums from those measurements alone.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "moment_balance_board-oracle-0.1.0"

_WIDTH = 1100
_HEIGHT = 760
_PITCH = 72.0
_LOAD_COLOR = np.array((25, 91, 113), dtype=np.uint8)
_PIVOT_COLOR = np.array((220, 151, 35), dtype=np.uint8)
_EXPECTED_DIAMETERS = {1: 27, 2: 41, 3: 55}
_LOAD_COUNT_PER_SIDE = 5


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Return 8-connected components of a boolean mask as (y, x) arrays."""

    ys, xs = np.nonzero(mask)
    seen = np.zeros(mask.shape, dtype=bool)
    components: list[np.ndarray] = []
    h, w = mask.shape
    for y0, x0 in zip(ys.tolist(), xs.tolist()):
        if seen[y0, x0]:
            continue
        queue = deque([(y0, x0)])
        seen[y0, x0] = True
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dy == 0 and dx == 0:
                        continue
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        components.append(np.asarray(points, dtype=np.int32))
    return components


def _best_component(mask: np.ndarray, label: str) -> np.ndarray:
    components = _components(mask)
    if not components:
        raise ValueError(f"missing {label} marker")
    return max(components, key=len)


def _load_records(arr: np.ndarray) -> list[tuple[float, float, int]]:
    mask = np.all(arr == _LOAD_COLOR, axis=2)
    components = [component for component in _components(mask) if len(component) >= 200]
    if len(components) != 2 * _LOAD_COUNT_PER_SIDE:
        raise ValueError(f"expected ten isolated load disks, found {len(components)}")
    records: list[tuple[float, float, int]] = []
    for component in components:
        ys = component[:, 0]
        xs = component[:, 1]
        diameter = int(max(xs) - min(xs) + 1)
        weight, expected = min(_EXPECTED_DIAMETERS.items(), key=lambda item: abs(item[1] - diameter))
        if abs(expected - diameter) > 4:
            raise ValueError(f"load diameter {diameter} does not match the size key")
        records.append((float(xs.mean()), float(ys.mean()), int(weight)))
    return records


def decision_from_image(image: Image.Image) -> str:
    """Return ``left``, ``right``, or ``tie`` from the PNG alone."""

    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.shape != (_HEIGHT, _WIDTH, 3):
        raise ValueError(f"unexpected image shape {arr.shape}")

    pivot_mask = np.all(arr == _PIVOT_COLOR, axis=2)
    pivot = _best_component(pivot_mask, "gold pivot")
    pivot_x = float(pivot[:, 1].mean())
    records = _load_records(arr)

    left: list[tuple[int, int]] = []
    right: list[tuple[int, int]] = []
    distances_seen: dict[str, set[int]] = {"left": set(), "right": set()}
    for cx, cy, weight in records:
        if abs(cy - 292.0) > 3.0:
            raise ValueError("load center is not on the load row")
        delta = cx - pivot_x
        if abs(delta) < 20:
            raise ValueError("load is too close to the pivot")
        side = "left" if delta < 0 else "right"
        distance = int(round(abs(delta) / _PITCH))
        if distance < 1 or distance > 6 or abs(abs(delta) - distance * _PITCH) > 4.0:
            raise ValueError("load is not aligned to a numbered tick")
        if distance in distances_seen[side]:
            raise ValueError("two loads occupy one side's tick")
        distances_seen[side].add(distance)
        (left if side == "left" else right).append((distance, weight))

    if len(left) != _LOAD_COUNT_PER_SIDE or len(right) != _LOAD_COUNT_PER_SIDE:
        raise ValueError("the image does not contain five loads on each side")
    if distances_seen["left"] != distances_seen["right"]:
        raise ValueError("the two sides do not share one visible distance set")
    if sorted(weight for _, weight in left) != [1, 1, 2, 2, 3] or sorted(weight for _, weight in right) != [1, 1, 2, 2, 3]:
        raise ValueError("the image does not contain the declared weight multiset")

    left_total = sum(distance * weight for distance, weight in left)
    right_total = sum(distance * weight for distance, weight in right)
    if left_total > right_total:
        return "left"
    if right_total > left_total:
        return "right"
    return "tie"
