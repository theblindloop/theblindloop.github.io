"""Independent PNG-only inverse arm for cross-panel rank retention."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_PALETTE = {
    "coral": np.array((225, 79, 86), dtype=np.float32),
    "blue": np.array((45, 119, 181), dtype=np.float32),
    "green": np.array((46, 160, 92), dtype=np.float32),
}
_MAX_COLOR_DISTANCE = 62.0
_MIN_AREA = 850
_MAX_AREA = 2300
_MIN_DIAMETER = 31


def _components(mask: np.ndarray) -> list[tuple[int, float, float, int, int]]:
    """Return (area,cx,cy,width,height) for 4-connected components."""
    ys, xs = np.nonzero(mask)
    if not len(xs):
        return []
    x0, x1 = int(xs.min()), int(xs.max()) + 1
    y0, y1 = int(ys.min()), int(ys.max()) + 1
    sub = mask[y0:y1, x0:x1]
    seen = np.zeros_like(sub, dtype=bool)
    output: list[tuple[int, float, float, int, int]] = []
    height, width = sub.shape
    for sy, sx in np.argwhere(sub):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and sub[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) < 20:
            continue
        py = np.fromiter((point[0] + y0 for point in points), dtype=np.float64)
        px = np.fromiter((point[1] + x0 for point in points), dtype=np.float64)
        output.append(
            (len(points), float(px.mean()), float(py.mean()), int(px.max() - px.min() + 1), int(py.max() - py.min() + 1))
        )
    return output


def _measure_markers(image: Image.Image) -> list[tuple[str, float, float]]:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    distances = np.stack(
        [np.linalg.norm(array - rgb, axis=2) for rgb in _PALETTE.values()], axis=2
    )
    labels = distances.argmin(axis=2)
    nearest = distances.min(axis=2)
    measured: list[tuple[str, float, float]] = []
    for index, color in enumerate(_PALETTE):
        comps = _components((labels == index) & (nearest < _MAX_COLOR_DISTANCE))
        valid = []
        for area, cx, cy, width, height in comps:
            aspect = width / max(1, height)
            if (
                _MIN_AREA <= area <= _MAX_AREA
                and width >= _MIN_DIAMETER
                and height >= _MIN_DIAMETER
                and 0.74 <= aspect <= 1.35
            ):
                valid.append((area, cx, cy, width, height))
        if len(valid) != 2:
            raise ValueError(f"{color}: expected two complete markers, found {len(valid)}")
        measured.extend((color, comp[1], comp[2]) for comp in valid)
    return measured


def decision_from_image(image: Image.Image) -> str:
    try:
        markers = _measure_markers(image)
        ordered_x = sorted(marker[1] for marker in markers)
        gaps = [ordered_x[index + 1] - ordered_x[index] for index in range(5)]
        split_index = int(np.argmax(gaps))
        if split_index != 2 or gaps[split_index] < image.width * 0.12:
            return "abstain"
        split_x = (ordered_x[2] + ordered_x[3]) / 2.0
        left = [marker for marker in markers if marker[1] < split_x]
        right = [marker for marker in markers if marker[1] > split_x]
        if len(left) != 3 or len(right) != 3:
            return "abstain"

        def ranks(group: list[tuple[str, float, float]]) -> dict[str, int]:
            group = sorted(group, key=lambda marker: marker[1])
            local_gaps = [group[index + 1][1] - group[index][1] for index in range(2)]
            if min(local_gaps) < 45.0 or len({marker[0] for marker in group}) != 3:
                raise ValueError("ambiguous panel ordering")
            return {marker[0]: rank for rank, marker in enumerate(group)}

        left_rank = ranks(left)
        right_rank = ranks(right)
        fixed = [color for color in _PALETTE if left_rank[color] == right_rank[color]]
        return fixed[0] if len(fixed) == 1 else "abstain"
    except (ValueError, IndexError, KeyError):
        return "abstain"
