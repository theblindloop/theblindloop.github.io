"""Deterministic renderer and latent analytic arm for cross-panel rank retention."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 768, 512
COLORS = ("coral", "blue", "green")
BASE_RGB = {
    "coral": (225, 79, 86),
    "blue": (45, 119, 181),
    "green": (46, 160, 92),
}


def _shade(rgb: tuple[int, int, int], delta: tuple[int, int, int]) -> list[int]:
    return [max(0, min(255, value + shift)) for value, shift in zip(rgb, delta)]


def sample_scene(seed: int) -> dict:
    """Sample one balanced, well-separated pair of three-color permutations."""
    rng = random.Random(int(seed) ^ 0x5A17C3)
    left_order = list(COLORS)
    rng.shuffle(left_order)
    preserved = COLORS[int(seed) % len(COLORS)]
    fixed_rank = left_order.index(preserved)
    other_ranks = [index for index in range(3) if index != fixed_rank]
    right_order = list(left_order)
    right_order[other_ranks[0]], right_order[other_ranks[1]] = (
        right_order[other_ranks[1]],
        right_order[other_ranks[0]],
    )

    dx = rng.randint(-10, 10)
    panel_y = rng.randint(38, 50)
    base_left = [108 + dx, 194 + dx, 280 + dx]
    base_right = [488 + dx, 574 + dx, 660 + dx]
    radius = rng.randint(20, 24)
    palette = {
        color: _shade(BASE_RGB[color], tuple(rng.randint(-9, 9) for _ in range(3)))
        for color in COLORS
    }

    left_markers = [
        {
            "color": color,
            "x": base_left[rank] + rng.randint(-5, 5),
            "y": panel_y + 112 + rng.randint(-24, 24),
            "radius": radius + rng.randint(-1, 1),
        }
        for rank, color in enumerate(left_order)
    ]
    right_markers = [
        {
            "color": color,
            "x": base_right[rank] + rng.randint(-5, 5),
            "y": panel_y + 348 + rng.randint(-24, 24),
            "radius": radius + rng.randint(-1, 1),
        }
        for rank, color in enumerate(right_order)
    ]
    return {
        "left_markers": left_markers,
        "right_markers": right_markers,
        "panel_y": panel_y,
        "background": [247, 249, 252],
        "palette": palette,
        "width": WIDTH,
        "height": HEIGHT,
    }


def _rank_map(markers: list[dict]) -> dict[str, int]:
    ordered = sorted(markers, key=lambda marker: float(marker["x"]))
    return {str(marker["color"]): rank for rank, marker in enumerate(ordered)}


def analytic_gold(scene: dict) -> str:
    left = _rank_map(scene["left_markers"])
    right = _rank_map(scene["right_markers"])
    matches = [color for color in COLORS if left.get(color) == right.get(color)]
    return matches[0] if len(matches) == 1 else "abstain"


def margin(scene: dict) -> float:
    gaps: list[float] = []
    for markers in (scene["left_markers"], scene["right_markers"]):
        xs = sorted(float(marker["x"]) for marker in markers)
        gaps.extend(xs[index + 1] - xs[index] for index in range(len(xs) - 1))
    return min(gaps) if gaps else 0.0


def is_quarantined(scene: dict) -> bool:
    for markers in (scene["left_markers"], scene["right_markers"]):
        if len(markers) != 3 or sorted(str(marker.get("color")) for marker in markers) != sorted(COLORS):
            return True
    return analytic_gold(scene) == "abstain" or margin(scene) < 60.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Serialization order of each marker list is a render-identical latent alias."""
    twin = copy.deepcopy(scene)
    twin["left_markers"] = list(reversed(twin["left_markers"]))
    twin["right_markers"] = [
        twin["right_markers"][1], twin["right_markers"][2], twin["right_markers"][0]
    ]
    return [("permute_marker_serialization", twin)]


def _font(size: int) -> ImageFont.ImageFont:
    for path in (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf",
        "/usr/share/fonts/truetype/liberation2/LiberationSans-Bold.ttf",
    ):
        try:
            return ImageFont.truetype(path, size)
        except OSError:
            continue
    return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    width = int(scene.get("width", WIDTH))
    height = int(scene.get("height", HEIGHT))
    scale = 3
    background = tuple(scene["background"])
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    panel_y = int(scene["panel_y"])

    # The two cards clarify panel membership but carry no answer-specific styling.
    cards = (
        (38, panel_y, 348, panel_y + 202),
        (420, panel_y + 234, 730, panel_y + 436),
    )
    for box in cards:
        x0, y0, x1, y1 = (value * scale for value in box)
        draw.rounded_rectangle(
            (x0 + 4 * scale, y0 + 5 * scale, x1 + 4 * scale, y1 + 5 * scale),
            radius=17 * scale,
            fill=(222, 227, 235),
        )
        draw.rounded_rectangle(
            (x0, y0, x1, y1),
            radius=17 * scale,
            fill=(255, 255, 255),
            outline=(170, 179, 192),
            width=2 * scale,
        )

    font = _font(17 * scale)
    for label, x, y in (
        ("PANEL A", 62, panel_y + 20),
        ("PANEL B", 444, panel_y + 254),
    ):
        draw.text((x * scale, y * scale), label, font=font, fill=(74, 84, 101))

    for x0, x1, baseline_y in (
        (76, 310, panel_y + 112),
        (458, 692, panel_y + 348),
    ):
        for x in range(x0, x1, 15):
            draw.line(
                ((x * scale, baseline_y * scale), (min(x + 7, x1) * scale, baseline_y * scale)),
                fill=(216, 221, 229),
                width=2 * scale,
            )

    palette = scene.get("palette", BASE_RGB)
    for marker in list(scene["left_markers"]) + list(scene["right_markers"]):
        x = float(marker["x"]) * scale
        y = float(marker["y"]) * scale
        radius = float(marker["radius"]) * scale
        rgb = tuple(palette[str(marker["color"])])
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=rgb)
        # A neutral highlight gives the marks visible dimensionality while leaving
        # their colored component connected and easily measurable.
        shine = max(3.0 * scale, radius * 0.18)
        draw.ellipse(
            (x - radius * 0.43 - shine, y - radius * 0.43 - shine,
             x - radius * 0.43 + shine, y - radius * 0.43 + shine),
            fill=(255, 255, 255),
        )

    return image.resize((width, height), Image.Resampling.LANCZOS)
