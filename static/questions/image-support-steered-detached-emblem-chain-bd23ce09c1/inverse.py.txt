"""Independent PNG-only inverse arm for Detached Emblem Transit."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_NAVY = np.array((29, 43, 70), dtype=np.int16)
_PURPLE = np.array((131, 79, 181), dtype=np.int16)
_GREEN = np.array((38, 151, 112), dtype=np.int16)
_ICON_COLORS = tuple(np.array(color, dtype=np.int16) for color in (
    (232, 104, 74), (55, 137, 201), (224, 169, 55)
))


class OracleAbstention(ValueError):
    pass


def _near(array: np.ndarray, color: np.ndarray, radius: int = 42) -> np.ndarray:
    # int32 avoids overflow when squared (255**2 exceeds signed int16).
    delta = array.astype(np.int32) - color.astype(np.int32)
    return np.sum(delta * delta, axis=2) <= radius * radius


def _components(mask: np.ndarray, minimum: int = 1) -> list[dict]:
    """Return generic 4-connected component measurements from one color mask."""
    seen = np.zeros(mask.shape, dtype=bool)
    height, width = mask.shape
    result: list[dict] = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        area = 0
        sum_x = 0
        sum_y = 0
        x0 = x1 = sx
        y0 = y1 = sy
        while queue:
            y, x = queue.popleft()
            area += 1
            sum_x += x
            sum_y += y
            x0, x1 = min(x0, x), max(x1, x)
            y0, y1 = min(y0, y), max(y1, y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if area >= minimum:
            result.append({
                "area": area,
                "bbox": (x0, y0, x1, y1),
                "cx": sum_x / area,
                "cy": sum_y / area,
            })
    return result


def _inside(component: dict, box: tuple[int, int, int, int], inset: int = 0) -> bool:
    x0, y0, x1, y1 = box
    return x0 + inset < component["cx"] < x1 - inset and y0 + inset < component["cy"] < y1 - inset


def _shape(component: dict) -> str:
    x0, y0, x1, y1 = component["bbox"]
    width = x1 - x0 + 1
    height = y1 - y0 + 1
    area = int(component["area"])
    if not (34 <= width <= 64 and 34 <= height <= 64 and 850 <= area <= 2400):
        raise OracleAbstention("emblem scale or area is outside the certified band")
    fill = area / float(width * height)
    aspect = width / float(height)
    if not 0.78 <= aspect <= 1.22:
        raise OracleAbstention("emblem bounding box is malformed")
    if 0.43 <= fill <= 0.61:
        return "triangle"
    if 0.69 <= fill <= 0.86:
        return "circle"
    if 0.91 <= fill <= 1.01:
        return "square"
    raise OracleAbstention("emblem has no unambiguous shape class")


def decision_from_image(image: Image.Image) -> str:
    """Parse all visible cues, bind ticket endpoints, and replay from pixels."""
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 400 or array.shape[1] < 600:
        raise OracleAbstention("canvas is missing")

    navy_mask = _near(array, _NAVY)
    frame_candidates = []
    for component in _components(navy_mask, minimum=800):
        x0, y0, x1, y1 = component["bbox"]
        width, height = x1 - x0 + 1, y1 - y0 + 1
        if 160 <= width <= 205 and 105 <= height <= 145 and component["area"] >= 1800:
            frame_candidates.append(component)
    if len(frame_candidates) != 4:
        raise OracleAbstention(f"expected four intact ticket frames, found {len(frame_candidates)}")
    frames = [component["bbox"] for component in frame_candidates]

    green_components = []
    for component in _components(_near(array, _GREEN), minimum=350):
        x0, y0, x1, y1 = component["bbox"]
        width, height = x1 - x0 + 1, y1 - y0 + 1
        if 95 <= width <= 145 and 95 <= height <= 145 and component["area"] >= 1200:
            green_components.append(component)
    if len(green_components) != 1:
        raise OracleAbstention("starting ring is missing or malformed")
    start_box = green_components[0]["bbox"]

    purple_components = []
    for component in _components(_near(array, _PURPLE), minimum=35):
        x0, y0, x1, y1 = component["bbox"]
        width, height = x1 - x0 + 1, y1 - y0 + 1
        if 7 <= width <= 18 and 7 <= height <= 18 and component["area"] <= 220:
            purple_components.append(component)

    icon_mask = np.zeros(array.shape[:2], dtype=bool)
    for color in _ICON_COLORS:
        icon_mask |= _near(array, color)
    icons = _components(icon_mask, minimum=700)

    start_icons = [component for component in icons if _inside(component, start_box)]
    if len(start_icons) != 1:
        raise OracleAbstention("start ring does not contain exactly one emblem")
    current = _shape(start_icons[0])

    ordered: dict[int, list[str]] = {}
    for frame in frames:
        pips = [component for component in purple_components if _inside(component, frame)]
        count = len(pips)
        if count not in (1, 2, 3, 4) or count in ordered:
            raise OracleAbstention("ticket dot counts do not uniquely encode one through four")
        inside_icons = [component for component in icons if _inside(component, frame, inset=4)]
        if len(inside_icons) != 2:
            raise OracleAbstention("ticket does not contain exactly two certified emblems")
        pair = [_shape(component) for component in inside_icons]
        if pair[0] == pair[1]:
            raise OracleAbstention("ticket endpoints are not distinct")
        ordered[count] = pair
    if set(ordered) != {1, 2, 3, 4}:
        raise OracleAbstention("ticket order is incomplete")

    for order in range(1, 5):
        pair = ordered[order]
        if pair.count(current) != 1:
            raise OracleAbstention("visible transition chain has no unique replay")
        current = pair[1] if pair[0] == current else pair[0]
    return current
