"""Analytic scene sampler and renderer for Detached Emblem Transit."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 960
HEIGHT = 720
BACKGROUND = (248, 246, 240)
NAVY = (29, 43, 70)
PURPLE = (131, 79, 181)
GREEN = (38, 151, 112)
PALETTE = ((232, 104, 74), (55, 137, 201), (224, 169, 55))
SHAPES = ("circle", "triangle", "square")

# Four widely separated, disconnected ticket centers.
CARD_CENTERS = ((130, 105), (830, 105), (130, 615), (830, 615))
FILLER_CENTERS = (
    (330, 105), (480, 105), (630, 105),
    (300, 250), (660, 250),
    (300, 470), (660, 470),
    (400, 615), (560, 615),
)


def _font(size: int) -> ImageFont.ImageFont:
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def sample_scene(seed: int) -> dict:
    """Sample one balanced valid four-transition scene from an integer seed."""
    rng = random.Random(int(seed))
    final = SHAPES[int(seed) % len(SHAPES)]
    states = [""] * 5
    states[4] = final
    for index in range(3, -1, -1):
        states[index] = rng.choice([shape for shape in SHAPES if shape != states[index + 1]])
    pairs = [[states[index], states[index + 1]] for index in range(4)]

    card_slots = list(range(4))
    rng.shuffle(card_slots)
    side_reversed = [bool(rng.getrandbits(1)) for _ in range(4)]
    main_colors = [rng.randrange(3) for _ in range(9)]

    main_shapes = [states[0]] + [shape for pair in pairs for shape in pair]
    counts = {shape: main_shapes.count(shape) for shape in SHAPES}
    filler_shapes = [shape for shape in SHAPES for _ in range(6 - counts[shape])]
    # Nine main emblems plus nine fillers gives exactly six visible copies of
    # each identity, independent of the answer.
    if len(filler_shapes) != len(FILLER_CENTERS):
        raise AssertionError("emblem balancing invariant changed")
    rng.shuffle(filler_shapes)
    filler_slots = list(range(len(FILLER_CENTERS)))
    rng.shuffle(filler_slots)
    fillers = [
        {"shape": shape, "slot": slot, "color": rng.randrange(3)}
        for shape, slot in zip(filler_shapes, filler_slots, strict=True)
    ]
    return {
        "start": states[0],
        "pairs": pairs,
        "card_slots": card_slots,
        "side_reversed": side_reversed,
        "main_colors": main_colors,
        "fillers": fillers,
        "palette_phase": rng.randrange(3),
        "icon_size": rng.choice((26, 27, 28)),
        "dx": rng.randint(-5, 5),
        "dy": rng.randint(-5, 5),
    }


def _replay(scene: dict) -> str:
    current = str(scene["start"])
    pairs = scene["pairs"]
    if current not in SHAPES or not isinstance(pairs, list) or len(pairs) != 4:
        raise ValueError("malformed transition program")
    for pair in pairs:
        if not isinstance(pair, list) or len(pair) != 2 or pair[0] == pair[1]:
            raise ValueError("each ticket needs two distinct endpoints")
        matches = [endpoint for endpoint in pair if endpoint == current]
        if len(matches) != 1:
            raise ValueError("carried emblem does not uniquely bind to ticket")
        current = pair[1] if pair[0] == current else pair[0]
    if current not in SHAPES:
        raise ValueError("unknown terminal emblem")
    return current


def analytic_gold(scene: dict) -> str:
    return _replay(scene)


def margin(scene: dict) -> float:
    """Runner-up margin: icon half-size above the minimum certified scale."""
    return float(scene["icon_size"] - 17)


def is_quarantined(scene: dict) -> bool:
    try:
        _replay(scene)
        return margin(scene) < 8.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["palette_phase"] = int(twin["palette_phase"]) + 3
    return [("palette_period_plus_three", twin)]


def _draw_emblem(draw: ImageDraw.ImageDraw, center: tuple[int, int], shape: str,
                  fill: tuple[int, int, int], size: int, scale: int) -> None:
    cx, cy = center
    cx *= scale
    cy *= scale
    r = size * scale
    outline_width = 3 * scale
    if shape == "circle":
        box = (cx - r + 2 * scale, cy - r + 2 * scale,
               cx + r - 2 * scale, cy + r - 2 * scale)
        draw.ellipse(box, fill=fill, outline=NAVY, width=outline_width)
    elif shape == "square":
        half = int(round(size * 0.82)) * scale
        draw.rounded_rectangle(
            (cx - half, cy - half, cx + half, cy + half),
            radius=2 * scale, fill=fill, outline=NAVY, width=outline_width,
        )
    elif shape == "triangle":
        points = (
            (cx, cy - int(round(size * 1.04)) * scale),
            (cx - int(round(size * 1.08)) * scale, cy + int(round(size * 0.88)) * scale),
            (cx + int(round(size * 1.08)) * scale, cy + int(round(size * 0.88)) * scale),
        )
        draw.polygon(points, fill=fill, outline=NAVY, width=outline_width)
    else:
        raise ValueError(f"unknown emblem {shape!r}")


def _color(code: int, phase: int) -> tuple[int, int, int]:
    return PALETTE[(int(code) + int(phase)) % len(PALETTE)]


def render(scene: dict) -> Image.Image:
    """Render from scene fields only; the analytic decision is never consulted."""
    scale = 2
    dx = int(scene["dx"])
    dy = int(scene["dy"])
    size = int(scene["icon_size"])
    phase = int(scene["palette_phase"])
    pairs = scene["pairs"]
    card_slots = scene["card_slots"]
    reversals = scene["side_reversed"]
    main_colors = scene["main_colors"]
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)

    # Pale corner brackets establish the canvas without joining decisive cards.
    pale = (220, 216, 205)
    for x, y in ((20, 20), (WIDTH - 20, 20), (20, HEIGHT - 20), (WIDTH - 20, HEIGHT - 20)):
        draw.ellipse(((x - 3) * scale, (y - 3) * scale,
                      (x + 3) * scale, (y + 3) * scale), fill=pale)

    # Free-floating distractors equalize global shape counts and colors.
    for filler in scene["fillers"]:
        fx, fy = FILLER_CENTERS[int(filler["slot"])]
        _draw_emblem(
            draw, (fx + dx, fy + dy), str(filler["shape"]),
            _color(int(filler["color"]), phase), size, scale,
        )

    # Green start ring is deliberately isolated from every ticket.
    start_center = (480 + dx, 360 + dy)
    scx, scy = start_center[0] * scale, start_center[1] * scale
    ring_r = 62 * scale
    draw.ellipse((scx - ring_r, scy - ring_r, scx + ring_r, scy + ring_r),
                 outline=GREEN, width=8 * scale)
    draw.text((scx, scy - 48 * scale), "START", fill=GREEN,
              font=_font(13 * scale), anchor="mm")
    _draw_emblem(draw, (start_center[0], start_center[1] + 8), str(scene["start"]),
                  _color(int(main_colors[0]), phase), size, scale)

    color_index = 1
    for logical_index in range(4):
        slot = int(card_slots[logical_index])
        cx0, cy0 = CARD_CENTERS[slot]
        cx, cy = cx0 + dx, cy0 + dy
        left, right = list(pairs[logical_index])
        left_color = int(main_colors[color_index])
        right_color = int(main_colors[color_index + 1])
        color_index += 2
        if bool(reversals[logical_index]):
            left, right = right, left
            left_color, right_color = right_color, left_color

        box = ((cx - 92) * scale, (cy - 62) * scale,
               (cx + 92) * scale, (cy + 62) * scale)
        draw.rounded_rectangle(box, radius=13 * scale, outline=NAVY, width=6 * scale)
        draw.line((cx * scale, (cy - 20) * scale, cx * scale, (cy + 47) * scale),
                  fill=NAVY, width=4 * scale)

        pip_count = logical_index + 1
        pip_spacing = 18
        first_x = cx - (pip_count - 1) * pip_spacing / 2.0
        for pip_index in range(pip_count):
            px = int(round(first_x + pip_index * pip_spacing))
            py = cy - 43
            pr = 6
            draw.ellipse(((px - pr) * scale, (py - pr) * scale,
                          (px + pr) * scale, (py + pr) * scale), fill=PURPLE)

        _draw_emblem(draw, (cx - 45, cy + 14), str(left),
                      _color(left_color, phase), size, scale)
        _draw_emblem(draw, (cx + 45, cy + 14), str(right),
                      _color(right_color, phase), size, scale)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
