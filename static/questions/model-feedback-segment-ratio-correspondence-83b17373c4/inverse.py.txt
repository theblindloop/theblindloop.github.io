"""Independent pixel-only inverse arm for the segment-ratio world.

Recovers the decision (which candidate reproduces the reference's segment ratio?)
from the final rasterized PNG alone. It detects the engineered three-section bars
by their exact indigo, turquoise, and amber colours, measures, within the
reference panel and each candidate panel, the width of the indigo (left) section
and the width of the turquoise (middle) section in columns, forms the derived
middle/left ratio, and returns the label of the candidate whose measured ratio
matches the reference's.

The three panels occupy fixed rows (renderer geometry), so the arm reads the
decision from the image pixels alone and never receives the latent scene. No
renderer / prompts / generate / verify / analytic-gold code is imported.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

INDIGO = (63, 72, 204)
TURQUOISE = (0, 188, 168)
AMBER = (255, 176, 32)
BAR_HEIGHT = 38
PANEL_TOPS = {"reference": 150, "A": 320, "B": 460}


def _ratio_in_band(arr: np.ndarray, top: int) -> tuple[float, bool]:
    """Derived middle/left ratio of the three-section bar within a row band."""
    band = arr[top : top + BAR_HEIGHT]
    indigo = (band == INDIGO).all(axis=-1)
    turquoise = (band == TURQUOISE).all(axis=-1)
    amber = (band == AMBER).all(axis=-1)
    ink = indigo | turquoise | amber
    cols = np.any(ink, axis=0)
    xs = np.nonzero(cols)[0]
    if xs.size == 0:
        return 1.0, False
    min_x = int(xs[0])
    max_x = int(xs[-1])
    left_w = int(np.count_nonzero(np.any(indigo[:, min_x : max_x + 1], axis=0)))
    mid_w = int(np.count_nonzero(np.any(turquoise[:, min_x : max_x + 1], axis=0)))
    if left_w <= 0:
        return 1.0, False
    return float(mid_w / left_w), True


def decision_from_image(image: Image.Image) -> str:
    """Return A or B: which candidate reproduces the reference's segment ratio."""
    arr = np.asarray(image.convert("RGB"))
    ratio_ref, ok_ref = _ratio_in_band(arr, PANEL_TOPS["reference"])
    ratio_a, ok_a = _ratio_in_band(arr, PANEL_TOPS["A"])
    ratio_b, ok_b = _ratio_in_band(arr, PANEL_TOPS["B"])
    if not (ok_ref and ok_a and ok_b):
        return "A"
    da = abs(ratio_a - ratio_ref)
    db = abs(ratio_b - ratio_ref)
    if da <= db:
        return "A"
    return "B"
