"""Independent pixel-only inverse arm for curved_stroke_oddball.

Receives only a PIL image. It separates the four quadrant regions, extracts the
dark stroke in each, measures each stroke's deviation from its own endpoint
chord (its bow / sagitta), and selects the quadrant whose stroke is the odd one
out: the stroke whose bow is farthest from the other three, which share the
same straight-or-bowed form.

Imports nothing from the renderer, prompts, generator, or verifier.
"""

from __future__ import annotations

import numpy as np


def _endpoints(xs: np.ndarray, ys: np.ndarray):
    """Return two extreme points along the stroke's dominant axis."""
    if xs.size < 30:
        return None
    pts = np.stack([xs.astype(float), ys.astype(float)], axis=1)
    mean = pts.mean(axis=0)
    centered = pts - mean
    cov = centered.T @ centered
    eigvals, eigvecs = np.linalg.eigh(cov)
    axis = eigvecs[:, np.argmax(eigvals)]
    proj = pts @ axis
    i0 = int(np.argmin(proj))
    i1 = int(np.argmax(proj))
    return pts[i0], pts[i1]


def _sagitta(xs: np.ndarray, ys: np.ndarray) -> float | None:
    ends = _endpoints(xs, ys)
    if ends is None:
        return None
    (ax, ay), (bx, by) = ends
    vx, vy = bx - ax, by - ay
    length = (vx * vx + vy * vy) ** 0.5
    if length < 1e-6:
        return None
    pts = np.stack([xs.astype(float), ys.astype(float)], axis=1)
    devs = np.abs(vx * (pts[:, 1] - ay) - vy * (pts[:, 0] - ax)) / length
    return float(devs.max())


def decision_from_image(image) -> str:
    w, h = image.size
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    dark = arr.sum(axis=2) < 150

    ys, xs = np.nonzero(dark)
    if xs.size == 0:
        return "TL"

    x_split = w * 0.5
    y_split = h * 0.645
    saggit: dict[str, float] = {}
    for sector in ("TL", "TR", "BL", "BR"):
        if sector in ("TL", "BL"):
            sel_x = xs < x_split
        else:
            sel_x = xs >= x_split
        if sector in ("TL", "TR"):
            sel_y = ys < y_split
        else:
            sel_y = ys >= y_split
        sel = sel_x & sel_y
        value = _sagitta(xs[sel], ys[sel])
        if value is None:
            continue
        saggit[sector] = value

    if not saggit:
        return "TL"
    if len(saggit) == 1:
        return next(iter(saggit))

    best_sector = None
    best_score = -1.0
    for sector, value in saggit.items():
        score = sum(abs(value - other) for other in saggit.values() if other != value)
        if score > best_score:
            best_score = score
            best_sector = sector
    return best_sector if best_sector is not None else "TL"
