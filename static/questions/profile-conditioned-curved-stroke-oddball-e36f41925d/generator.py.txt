"""Straight-versus-bowed stroke oddball search world.

The image is split into four quadrants (top-left, top-right, bottom-left,
bottom-right). Each quadrant holds one dark stroke. A \"bowed\" stroke bulges
out of the straight chord joining its two ends; a \"straight\" stroke stays on
its chord. Exactly three of the four strokes share the SAME form (all bowed or
all straight); the fourth stroke has the opposite form and is the odd one out.
No external reference is shown: the answer is found by internal uniqueness of
straight-versus-bowed morphology. All rendering is deterministic and the form
of every stroke is visible in the raster.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

SIZE = 480
BG = (255, 255, 255)
SEG = (20, 20, 20)
GRAY = (196, 196, 196)

SECTORS = ("TL", "TR", "BL", "BR")
SECTOR_CENTERS = {
    "TL": (SIZE // 4, 250),
    "TR": (3 * SIZE // 4, 250),
    "BL": (SIZE // 4, 410),
    "BR": (3 * SIZE // 4, 410),
}

STROKE_HALF = 62
STROKE_THICK = 13
BOW_LOW = 14.0
BOW_HIGH = 30.0
MAJORITY_BOW = 24.0
QUARANTINE_THRESH = 18.0


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(seed)
    majority_curved = bool(rng.integers(0, 2))
    winner_sector = SECTORS[int(rng.integers(0, len(SECTORS)))]

    if majority_curved:
        majority_bow = MAJORITY_BOW
        odd_bow = 0.0
    else:
        majority_bow = 0.0
        odd_bow = float(round(BOW_LOW + rng.uniform(0.0, BOW_HIGH - BOW_LOW), 1))

    cx = float(round(rng.uniform(-14.0, 14.0), 1))
    cy = float(round(rng.uniform(-8.0, 8.0), 1))
    angle_deg = float(round(rng.uniform(-24.0, 24.0), 1))
    strokes = []
    for sector in SECTORS:
        c = SECTOR_CENTERS[sector]
        bow = odd_bow if sector == winner_sector else majority_bow
        strokes.append(
            {
                "cx": int(c[0] + cx),
                "cy": int(c[1] + cy),
                "angle_deg": angle_deg,
                "bow_amp": float(round(bow, 1)),
                "sector": sector,
            }
        )
    return {"strokes": strokes}


def _stroke_points(cx: int, cy: int, angle_deg: float, bow_amp: float, n: int = 160):
    theta = np.deg2rad(angle_deg)
    t = np.linspace(-1.0, 1.0, n)
    xs = t * STROKE_HALF
    ys = -bow_amp * np.cos(0.5 * np.pi * t)
    px = xs * np.cos(theta) - ys * np.sin(theta) + cx
    py = xs * np.sin(theta) + ys * np.cos(theta) + cy
    return px, py


def _stamp_point(arr: np.ndarray, x: int, y: int, radius: float) -> None:
    r2 = radius * radius
    r = int(radius) + 2
    y0 = max(0, y - r)
    y1 = min(arr.shape[0], y + r + 1)
    x0 = max(0, x - r)
    x1 = min(arr.shape[1], x + r + 1)
    yy, xx = np.ogrid[y0:y1, x0:x1]
    mask = (xx - x) ** 2 + (yy - y) ** 2 <= r2
    arr[y0:y1, x0:x1][mask] = SEG


def render(scene: dict) -> Image.Image:
    arr = np.empty((SIZE, SIZE, 3), dtype=np.uint8)
    arr[:] = BG
    arr[309:311, :] = GRAY
    arr[:, 240:242] = GRAY
    radius = STROKE_THICK * 0.5
    for stroke in scene["strokes"]:
        px, py = _stroke_points(stroke["cx"], stroke["cy"], stroke["angle_deg"], stroke["bow_amp"])
        for x, y in zip(px.astype(int), py.astype(int)):
            _stamp_point(arr, int(x), int(y), radius)
    return Image.fromarray(arr, "RGB")


def _sagitta(cx: int, cy: int, angle_deg: float, bow_amp: float) -> float:
    px, py = _stroke_points(cx, cy, angle_deg, bow_amp)
    ax, ay = float(px[0]), float(py[0])
    bx, by = float(px[-1]), float(py[-1])
    vx, vy = bx - ax, by - ay
    length = (vx * vx + vy * vy) ** 0.5
    if length < 1e-9:
        return 0.0
    max_dev = 0.0
    for x, y in zip(px, py):
        dev = abs(vx * (y - ay) - vy * (x - ax)) / length
        if dev > max_dev:
            max_dev = dev
    return float(max_dev)


def _decision(scene: dict) -> tuple[str, float]:
    strokes = scene["strokes"]
    sag = {s["sector"]: _sagitta(s["cx"], s["cy"], s["angle_deg"], s["bow_amp"]) for s in strokes}
    total = {}
    for sector, value in sag.items():
        total[sector] = sum(abs(value - other) for other in sag.values() if other != value)
    winner = max(total, key=lambda k: total[k])
    gap = float(total[winner] / 3.0)
    return winner, gap


def analytic_gold(scene: dict) -> str:
    sector, _ = _decision(scene)
    return sector


def margin(scene: dict) -> float:
    _, m = _decision(scene)
    return m


def is_quarantined(scene: dict) -> bool:
    _, m = _decision(scene)
    return bool(m < QUARANTINE_THRESH)


def latent_symmetries(scene: dict) -> list:
    return []
