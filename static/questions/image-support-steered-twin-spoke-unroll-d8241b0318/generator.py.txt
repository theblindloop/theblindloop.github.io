"""Deterministic renderer for Twin Spoke Unroll."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 512
SYMBOLS = ("box", "D", "backward D", "oval")

LEFT_FLAT = (
    "1111", "1000", "1000", "1000", "1000", "1000", "1000", "1111",
)
RIGHT_FLAT = tuple(row[::-1] for row in LEFT_FLAT)
LEFT_CURVE = (
    "0011", "0110", "1100", "1000", "1000", "1100", "0110", "0011",
)
RIGHT_CURVE = tuple(row[::-1] for row in LEFT_CURVE)


def _templates() -> dict[str, tuple[str, ...]]:
    return {
        "box": tuple(a + b for a, b in zip(LEFT_FLAT, RIGHT_FLAT)),
        "D": tuple(a + b for a, b in zip(LEFT_FLAT, RIGHT_CURVE)),
        "backward D": tuple(a + b for a, b in zip(LEFT_CURVE, RIGHT_FLAT)),
        "oval": tuple(a + b for a, b in zip(LEFT_CURVE, RIGHT_CURVE)),
    }


def _bits(scene: dict) -> list[list[int]]:
    bits = [[int(value) for value in row] for row in _templates()[scene["symbol"]]]
    for row, column in scene["flips"]:
        bits[row][column] ^= 1
    return bits


def _distances(bits: list[list[int]]) -> list[tuple[int, str]]:
    scored = []
    for name, rows in _templates().items():
        distance = sum(
            bits[r][c] != int(rows[r][c]) for r in range(8) for c in range(8)
        )
        scored.append((distance, name))
    return sorted(scored)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    symbol = SYMBOLS[int(seed) % len(SYMBOLS)]
    left_center = [rng.randint(163, 173), rng.randint(244, 268)]
    right_center = [rng.randint(339, 349), rng.randint(244, 268)]
    if rng.randrange(2):
        center_one, center_two = right_center, left_center
    else:
        center_one, center_two = left_center, right_center
    palettes = ["violet", "teal"]
    rng.shuffle(palettes)
    scene = {
        "width": WIDTH,
        "height": HEIGHT,
        "symbol": symbol,
        "center_one": center_one,
        "center_two": center_two,
        "radius_one": rng.randint(74, 78),
        "radius_two": rng.randint(74, 78),
        "phase_one": rng.uniform(-math.pi, math.pi),
        "phase_two": rng.uniform(-math.pi, math.pi),
        "palette_one": palettes[0],
        "palette_two": palettes[1],
        "flips": [],
    }

    # Balanced dark/light swaps vary difficulty without changing total ink.
    bits = _bits(scene)
    pair_count = 2 if seed % 13 == 0 else (1 if seed % 5 == 0 else 0)
    dark = [(r, c) for r in range(8) for c in range(8) if bits[r][c]]
    light = [(r, c) for r in range(8) for c in range(8) if not bits[r][c]]
    rng.shuffle(dark)
    rng.shuffle(light)
    for index in range(pair_count):
        scene["flips"].extend([list(dark[index]), list(light[index])])
    # Never emit a tie or a label-changing corruption.
    scored = _distances(_bits(scene))
    if scored[0][1] != symbol or scored[0][0] == scored[1][0]:
        scene["flips"] = []
    return scene


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def margin(scene: dict) -> float:
    scored = _distances(_bits(scene))
    return float(scored[1][0] - scored[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _polar(center: list[int], radius: float, angle: float) -> tuple[float, float]:
    return (center[0] + radius * math.cos(angle), center[1] + radius * math.sin(angle))


def _draw_reel(
    draw: ImageDraw.ImageDraw,
    *,
    center: list[int],
    radius: int,
    phase: float,
    role: int,
    palette: str,
    columns: list[list[int]],
) -> None:
    frame = (112, 67, 171) if palette == "violet" else (29, 130, 145)
    dark = (25, 31, 43)
    pale = (225, 230, 235)
    guide = (190, 198, 207)
    amber = (238, 158, 42)
    green = (54, 176, 91)
    cx, cy = center
    draw.ellipse((cx-radius, cy-radius, cx+radius, cy+radius), outline=frame, width=4)

    # Four visible spokes; clockwise from the green bead is column order.
    for column_index in range(4):
        angle = phase + column_index * math.pi / 2.0
        inner = _polar(center, 0.22 * radius, angle)
        outer = _polar(center, 0.88 * radius, angle)
        draw.line((inner, outer), fill=guide, width=2)
        for row in range(8):
            radial = radius * (0.29 + row * (0.53 / 7.0))
            x, y = _polar(center, radial, angle)
            bead = max(3, round(radius * 0.038))
            fill = dark if columns[column_index][row] else pale
            draw.ellipse((x-bead, y-bead, x+bead, y+bead), fill=fill, outline=(145, 153, 163))

    # The green bead just inside the rim fixes angular phase.
    mx, my = _polar(center, radius * 0.93, phase)
    draw.ellipse((mx-4, my-4, mx+4, my+4), fill=green, outline=(25, 100, 55), width=1)

    # One/two amber dots bind the reel to the output half, independent of position.
    dot_centers = [(cx, cy)] if role == 1 else [(cx-6, cy), (cx+6, cy)]
    for x, y in dot_centers:
        draw.ellipse((x-3, y-3, x+3, y+3), fill=amber)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), (250, 248, 243))
    draw = ImageDraw.Draw(image)
    bits = _bits(scene)
    one_columns = [[bits[row][column] for row in range(8)] for column in range(4)]
    two_columns = [[bits[row][column] for row in range(8)] for column in range(4, 8)]
    _draw_reel(
        draw,
        center=scene["center_one"], radius=int(scene["radius_one"]),
        phase=float(scene["phase_one"]), role=1, palette=scene["palette_one"],
        columns=one_columns,
    )
    _draw_reel(
        draw,
        center=scene["center_two"], radius=int(scene["radius_two"]),
        phase=float(scene["phase_two"]), role=2, palette=scene["palette_two"],
        columns=two_columns,
    )
    return image
