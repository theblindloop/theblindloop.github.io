"""Deterministic raster world for directed route reachability.

The latent drawing is a four-edge star: a green source connects to one
junction, which connects to three colored terminals.  All four edges are
visible as the same dark stroke.  Orange triangular arrowheads are the only
thing that makes the graph directed; exactly one terminal branch points away
from the junction, so exactly one colored terminal is reachable from green.

The image-only inverse in oracle.py deliberately reconstructs this structure
from color masks, the dark ink density, and the orange triangle geometry.  It
does not import this module or receive a scene.
"""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

CANVAS = (1000, 900)
BASE_NODES = {
    "source": (-360.0, 0.0),
    "junction": (0.0, 0.0),
    "top": (300.0, -205.0),
    "middle": (365.0, 0.0),
    "bottom": (300.0, 205.0),
}
EDGE_NODES = (
    ("source", "junction"),
    ("junction", "top"),
    ("junction", "middle"),
    ("junction", "bottom"),
)
TERMINAL_SLOTS = ("top", "middle", "bottom")
PALETTE = {
    "navy": (31, 52, 82),
    "orange": (235, 112, 38),
    "green": (48, 157, 96),
    "red": (208, 70, 76),
    "blue": (57, 111, 187),
    "gold": (218, 163, 47),
    "pale": (192, 199, 211),
}


def _rotate(point: tuple[float, float], angle_degrees: float) -> tuple[float, float]:
    radians = math.radians(angle_degrees)
    cosine, sine = math.cos(radians), math.sin(radians)
    x, y = point
    return cosine * x - sine * y, sine * x + cosine * y


def _transform(point: tuple[float, float], scene: dict) -> tuple[float, float]:
    rotated = _rotate(point, float(scene["rotation"]))
    translation = scene["translation"]
    return rotated[0] + float(translation[0]), rotated[1] + float(translation[1])


def _node_point(name: str, scene: dict) -> tuple[float, float]:
    return _transform(BASE_NODES[name], scene)


def _scaled(point: tuple[float, float], scale: int) -> tuple[int, int]:
    return int(round(point[0] * scale)), int(round(point[1] * scale))


def _line_points(points: Iterable[tuple[float, float]], scale: int) -> list[tuple[int, int]]:
    return [_scaled(point, scale) for point in points]


def sample_scene(seed: int) -> dict:
    """Sample a balanced, safely rasterizable scene from one integer seed."""

    rng = random.Random(int(seed) * 1009 + 17)
    # The reachable slot cycles independently of the six terminal-color
    # permutations.  This makes answer identity independent of geometry and
    # gives balanced classes for every 24-scene stress sweep.
    reachable_slot = int(seed) % 3
    color_order = ["red", "blue", "gold"]
    permutation_index = int(seed) % 6
    permutations = (
        ("red", "blue", "gold"),
        ("red", "gold", "blue"),
        ("blue", "red", "gold"),
        ("blue", "gold", "red"),
        ("gold", "red", "blue"),
        ("gold", "blue", "red"),
    )
    terminal_colors = list(permutations[permutation_index])
    branch_directions = [-1, -1, -1]
    branch_directions[reachable_slot] = 1

    stroke_width = 8 + (int(seed) % 3)
    arrow_length = 30 + (int(seed) % 4) * 3
    arrow_half_width = 14 + (int(seed) % 3)
    return {
        "width": CANVAS[0],
        "height": CANVAS[1],
        "background": [248, 247, 242],
        "rotation": round(rng.uniform(0.0, 360.0), 6),
        "translation": [
            round(500.0 + rng.uniform(-24.0, 24.0), 6),
            round(450.0 + rng.uniform(-24.0, 24.0), 6),
        ],
        "directions": [1, *branch_directions],
        "terminal_colors": terminal_colors,
        "edge_color": "navy",
        "arrow_color": "orange",
        "source_color": "green",
        "terminal_radius": 22,
        "source_radius": 23,
        "stroke_width": stroke_width,
        "arrow_length": arrow_length,
        "arrow_half_width": arrow_half_width,
        "outline_width": 5,
        "distractor_variant": int(seed) % 4,
        # Used only as a deterministic diagnostic in manifests; it has no
        # effect on analytic gold and is not consulted by the pixel oracle.
        "seed_tag": int(seed),
    }


def _arrow_polygon(
    start: tuple[float, float],
    end: tuple[float, float],
    direction: int,
    length: float,
    half_width: float,
) -> list[tuple[float, float]]:
    a, b = (start, end) if direction > 0 else (end, start)
    dx, dy = b[0] - a[0], b[1] - a[1]
    norm = math.hypot(dx, dy)
    ux, uy = dx / norm, dy / norm
    px, py = -uy, ux
    center = ((a[0] + b[0]) / 2.0, (a[1] + b[1]) / 2.0)
    tip = (center[0] + ux * length / 2.0, center[1] + uy * length / 2.0)
    base = (center[0] - ux * length / 2.0, center[1] - uy * length / 2.0)
    return [
        tip,
        (base[0] + px * half_width, base[1] + py * half_width),
        (base[0] - px * half_width, base[1] - py * half_width),
    ]


def _draw_marker(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    radius: float,
    fill: tuple[int, int, int],
    outline: tuple[int, int, int],
    scale: int,
    outline_width: int,
) -> None:
    cx, cy = center
    outer = radius + outline_width
    draw.ellipse(
        (int(round((cx - outer) * scale)), int(round((cy - outer) * scale)),
         int(round((cx + outer) * scale)), int(round((cy + outer) * scale))),
        fill=outline,
    )
    draw.ellipse(
        (int(round((cx - radius) * scale)), int(round((cy - radius) * scale)),
         int(round((cx + radius) * scale)), int(round((cy + radius) * scale))),
        fill=fill,
    )


def render(scene: dict) -> Image.Image:
    """Render the scene as a deterministic RGB PNG-sized image."""

    scale = 4
    width, height = int(scene["width"]), int(scene["height"])
    background = tuple(int(v) for v in scene["background"])
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)

    edge_color = PALETTE[str(scene["edge_color"])]
    arrow_color = PALETTE[str(scene["arrow_color"])]
    pale_color = PALETTE["pale"]
    stroke = int(scene["stroke_width"] * scale)
    pale_stroke = max(scale * 3, stroke // 2)

    # Two low-contrast disconnected strokes exercise distractor rejection and
    # are deliberately generated from a field independent of the answer.
    variant = int(scene["distractor_variant"])
    distractors = (
        ((-285.0, -292.0), (-92.0, -292.0)),
        ((150.0, 292.0), (352.0, 292.0)),
    )
    if variant % 2:
        distractors = tuple(reversed(distractors))
    for left, right in distractors:
        draw.line(
            _line_points((_transform(left, scene), _transform(right, scene)), scale),
            fill=pale_color,
            width=pale_stroke,
        )

    # The four undirected route strokes are identical for all direction
    # assignments; only the arrowheads carry edge orientation.
    for left_name, right_name in EDGE_NODES:
        draw.line(
            _line_points(
                (_node_point(left_name, scene), _node_point(right_name, scene)),
                scale,
            ),
            fill=edge_color,
            width=stroke,
        )

    for edge_index, (left_name, right_name) in enumerate(EDGE_NODES):
        start = _node_point(left_name, scene)
        end = _node_point(right_name, scene)
        polygon = _arrow_polygon(
            start,
            end,
            int(scene["directions"][edge_index]),
            float(scene["arrow_length"]),
            float(scene["arrow_half_width"]),
        )
        draw.polygon(_line_points(polygon, scale), fill=arrow_color)

    outline = edge_color
    _draw_marker(
        draw,
        _node_point("source", scene),
        float(scene["source_radius"]),
        PALETTE[str(scene["source_color"])],
        outline,
        scale,
        int(scene["outline_width"]),
    )
    for slot, color_name in zip(TERMINAL_SLOTS, scene["terminal_colors"]):
        _draw_marker(
            draw,
            _node_point(slot, scene),
            float(scene["terminal_radius"]),
            PALETTE[str(color_name)],
            outline,
            scale,
            int(scene["outline_width"]),
        )

    return image.resize((width, height), Image.Resampling.LANCZOS).convert("RGB")


def analytic_gold(scene: dict) -> str:
    """Return the terminal color on the unique directed source path."""

    directions = scene["directions"]
    terminal_colors = scene["terminal_colors"]
    if int(directions[0]) != 1:
        raise ValueError("source edge must point from source to junction")
    outgoing = [index for index, value in enumerate(directions[1:]) if int(value) == 1]
    if len(outgoing) != 1:
        raise ValueError("scene must have exactly one outward terminal branch")
    return str(terminal_colors[outgoing[0]])


def margin(scene: dict) -> float:
    """Pixel clearance for recovering an arrowhead orientation."""

    arrow_length = float(scene["arrow_length"])
    arrow_half_width = float(scene["arrow_half_width"])
    stroke_width = float(scene["stroke_width"])
    return round(min(arrow_length - 1.5 * stroke_width, arrow_half_width - 0.5 * stroke_width), 6)


def is_quarantined(scene: dict) -> bool:
    """Exclude tiny arrowheads whose visible orientation has little clearance."""

    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """There are no declared pixel-identical latent aliases in this renderer."""

    return []
