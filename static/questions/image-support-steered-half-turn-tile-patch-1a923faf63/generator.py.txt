"""Latent sampler and renderer for the half-turn tile-patch world."""

from __future__ import annotations

import random
from collections import deque

import numpy as np
from PIL import Image

CANVAS = 256
CELL_COUNT = 36
PATCH_COLORS = ("#168C96", "#336CB3", "#6E5AA8", "#33845B")
BACKGROUNDS = ("#F7F3E8", "#F1F5F4", "#F6F0F4")
PIN_COLOR = "#D93636"


def _connected(cells: set[tuple[int, int]]) -> bool:
    if not cells:
        return False
    seen = {next(iter(cells))}
    queue = deque(seen)
    while queue:
        x, y = queue.popleft()
        for point in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
            if point in cells and point not in seen:
                seen.add(point)
                queue.append(point)
    return len(seen) == len(cells)


def _pair_key(cell: tuple[int, int]) -> tuple[int, int]:
    twin = (-cell[0], -cell[1])
    return min(cell, twin)


def _symmetric_patch(rng: random.Random) -> set[tuple[int, int]]:
    cells = {(x, y) for y in range(-3, 4) for x in range(-3, 4) if (x, y) != (0, 0)}
    pairs = sorted({_pair_key(cell) for cell in cells})
    rng.shuffle(pairs)
    removed = 0
    for first in pairs:
        if removed == 6:
            break
        pair = {first, (-first[0], -first[1])}
        trial = cells - pair
        xs = {x for x, _ in trial}
        ys = {y for _, y in trial}
        if _connected(trial) and xs == set(range(-3, 4)) and ys == set(range(-3, 4)):
            cells = trial
            removed += 1
    if removed != 6 or len(cells) != CELL_COUNT:
        raise RuntimeError("could not construct a dense connected symmetric patch")
    return cells


def _asymmetric_twin(cells: set[tuple[int, int]], rng: random.Random) -> set[tuple[int, int]]:
    occupied = list(cells)
    missing = [
        (x, y)
        for y in range(-3, 4)
        for x in range(-3, 4)
        if (x, y) != (0, 0) and (x, y) not in cells
    ]
    rng.shuffle(occupied)
    rng.shuffle(missing)
    for removed in occupied:
        # Removing only one member of a present rotational pair exposes a mismatch.
        if (-removed[0], -removed[1]) not in cells:
            continue
        for added in missing:
            # Adding only one member of an absent pair exposes a second mismatch.
            if (-added[0], -added[1]) in cells or added == (-removed[0], -removed[1]):
                continue
            trial = (cells - {removed}) | {added}
            xs = {x for x, _ in trial}
            ys = {y for _, y in trial}
            if _connected(trial) and xs == set(range(-3, 4)) and ys == set(range(-3, 4)):
                if trial != {(-x, -y) for x, y in trial}:
                    return trial
    raise RuntimeError("could not make a connected asymmetric counterpart")


def sample_scene(seed: int) -> dict:
    # Adjacent seeds form matched yes/no pairs with identical nuisance variables.
    pair_seed = seed // 2
    rng = random.Random(pair_seed ^ 0x5A17C3)
    cells = _symmetric_patch(rng)
    if seed % 2:
        cells = _asymmetric_twin(cells, rng)
    tile_size = (14, 16)[rng.randrange(2)]
    span = 7 * tile_size
    origin_x = rng.randint(72, 184 - span)
    origin_y = rng.randint(72, 184 - span)
    return {
        "cells": [[x, y] for x, y in sorted(cells, key=lambda p: (p[1], p[0]))],
        "origin_x": origin_x,
        "origin_y": origin_y,
        "tile_size": tile_size,
        "patch_color": PATCH_COLORS[rng.randrange(len(PATCH_COLORS))],
        "pin_color": PIN_COLOR,
        "background": BACKGROUNDS[rng.randrange(len(BACKGROUNDS))],
    }


def _cell_set(scene: dict) -> set[tuple[int, int]]:
    return {(int(x), int(y)) for x, y in scene["cells"]}


def render(scene: dict) -> Image.Image:
    tile = int(scene["tile_size"])
    origin_x = int(scene["origin_x"])
    origin_y = int(scene["origin_y"])
    array = np.empty((CANVAS, CANVAS, 3), dtype=np.uint8)
    array[:] = tuple(int(scene["background"][i : i + 2], 16) for i in (1, 3, 5))
    patch = tuple(int(scene["patch_color"][i : i + 2], 16) for i in (1, 3, 5))
    for x, y in _cell_set(scene):
        x0 = origin_x + (x + 3) * tile
        y0 = origin_y + (y + 3) * tile
        array[y0 : y0 + tile, x0 : x0 + tile] = patch
    cx = origin_x + 3.5 * tile - 0.5
    cy = origin_y + 3.5 * tile - 0.5
    radius = max(4, tile // 3)
    pin = tuple(int(scene["pin_color"][i : i + 2], 16) for i in (1, 3, 5))
    yy, xx = np.ogrid[:CANVAS, :CANVAS]
    array[(xx - cx) ** 2 + (yy - cy) ** 2 <= radius**2] = pin
    return Image.fromarray(array, mode="RGB")


def analytic_gold(scene: dict) -> str:
    cells = _cell_set(scene)
    return "yes" if cells == {(-x, -y) for x, y in cells} else "no"


def _mismatch_count(scene: dict) -> int:
    cells = _cell_set(scene)
    return len(cells.symmetric_difference({(-x, -y) for x, y in cells}))


def margin(scene: dict) -> float:
    mismatches = _mismatch_count(scene)
    # Gap to the nearest rival: one rotational pair for yes, observed mismatch pairs for no.
    return 2.0 if mismatches == 0 else float(mismatches)


def is_quarantined(scene: dict) -> bool:
    cells = _cell_set(scene)
    return len(cells) != CELL_COUNT or not _connected(cells) or _mismatch_count(scene) == 1


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["cells"] = list(reversed(scene["cells"]))
    return [("reverse_tile_serialization", twin)]
