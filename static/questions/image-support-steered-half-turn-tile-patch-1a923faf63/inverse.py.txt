"""Independent pixel-only inverse for half-turn patch symmetry."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    """The raster does not expose one valid 36-tile patch and center pin."""


def _masks(image: Image.Image) -> tuple[np.ndarray, np.ndarray]:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red = (
        (array[:, :, 0] >= 170)
        & (array[:, :, 0] >= array[:, :, 1] + 70)
        & (array[:, :, 0] >= array[:, :, 2] + 55)
    )
    chroma = array.max(axis=2) - array.min(axis=2)
    colored = (chroma >= 55) & (array.min(axis=2) < 190)
    return colored & ~red, red


def _tile_size_from_area(area: int) -> int:
    estimate = math.sqrt(area / 36.0)
    tile = int(round(estimate))
    if tile not in (14, 16) or area != 36 * tile * tile:
        raise OracleAbstention("colored area is not exactly 36 equal raster tiles")
    return tile


def decision_from_image(image: Image.Image) -> str:
    patch, pin = _masks(image)
    _tile_size_from_area(int(patch.sum()))
    py, px = np.nonzero(pin)
    if len(px) < 40:
        raise OracleAbstention("visible red center pin is missing or damaged")
    cx = float(px.mean())
    cy = float(py.mean())
    # Rotate pixel centers around the pin centroid without assuming a fixed placement.
    ys, xs = np.nonzero(patch)
    rx = np.rint(2.0 * cx - xs).astype(int)
    ry = np.rint(2.0 * cy - ys).astype(int)
    if (rx < 0).any() or (rx >= patch.shape[1]).any() or (ry < 0).any() or (ry >= patch.shape[0]).any():
        raise OracleAbstention("rotated patch would leave the canvas")
    matched = int(patch[ry, rx].sum())
    mismatched_pixels = len(xs) - matched
    return "yes" if mismatched_pixels == 0 else "no"
