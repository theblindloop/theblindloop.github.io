"""Deterministic raster world for counting a STOP sign's outer sides.

The scene is deliberately familiar but the visible outer silhouette is sampled
from five through ten sides.  The label is the integer side count itself; the
STOP bitmap and all style fields are rendered nuisances.  The independent
inverse in ``oracle.py`` recovers the chromatic silhouette from the PNG and
counts its convex-hull turns without importing this module.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "stop_sign_sides-0.1.0"
WIDTH = 512
HEIGHT = 512
SIDE_VALUES = (5, 6, 7, 8, 9, 10)

# Saturated fills are separated from the pale backgrounds and neutral outline
# by the oracle's chroma segmentation.  The palette is intentionally varied so
# red is not a reliable proxy for the answer or the canonical condition.
FILL_PALETTE = (
    (214, 43, 51),    # red
    (48, 101, 213),   # blue
    (38, 157, 91),    # green
    (226, 126, 35),   # orange
    (143, 67, 188),   # purple
    (22, 157, 166),   # teal
)
BACKGROUND_PALETTE = (
    (255, 255, 255),
    (248, 244, 236),
    (238, 245, 248),
)
OUTLINE = (25, 30, 40)
LETTER = (255, 246, 226)

_STOP_BITMAP = {
    "S": ("01111", "10000", "10000", "01110", "00001", "00001", "11110"),
    "T": ("11111", "00100", "00100", "00100", "00100", "00100", "00100"),
    "O": ("01110", "10001", "10001", "10001", "10001", "10001", "01110"),
    "P": ("11110", "10001", "10001", "11110", "10000", "10000", "10000"),
}


def _as_rgb(value: Any) -> tuple[int, int, int]:
    return tuple(int(channel) for channel in value)  # type: ignore[return-value]


def _regular_vertices(scene: dict, scale: int = 1) -> list[tuple[int, int]]:
    n = int(scene["outer_sides"])
    cx, cy = (float(value) for value in scene["center"])
    radius = float(scene["radius"])
    rotation = math.radians(float(scene["rotation_deg"]))
    return [
        (
            int(round((cx + radius * math.cos(rotation + 2.0 * math.pi * index / n)) * scale)),
            int(round((cy + radius * math.sin(rotation + 2.0 * math.pi * index / n)) * scale)),
        )
        for index in range(n)
    ]


def _draw_stop_bitmap(draw: ImageDraw.ImageDraw, scene: dict, scale: int) -> None:
    cx, cy = (float(value) for value in scene["center"])
    radius = float(scene["radius"])
    cell = max(2, min(7, int(round(radius * 0.055))))
    gap = cell
    word = "STOP"
    width_units = sum(len(_STOP_BITMAP[letter][0]) for letter in word) + len(word) - 1
    total_width = width_units * cell
    total_height = 7 * cell
    left = int(round(cx - total_width / 2.0))
    top = int(round(cy - total_height / 2.0))
    for letter_index, letter in enumerate(word):
        bitmap = _STOP_BITMAP[letter]
        offset_units = sum(len(_STOP_BITMAP[item][0]) for item in word[:letter_index]) + letter_index
        for row, bits in enumerate(bitmap):
            for column, bit in enumerate(bits):
                if bit == "1":
                    x0 = (left + (offset_units + column) * cell) * scale
                    y0 = (top + row * cell) * scale
                    x1 = x0 + cell * scale - 1
                    y1 = y0 + cell * scale - 1
                    draw.rectangle((x0, y0, x1, y1), fill=LETTER)


def sample_scene(seed: int) -> dict:
    """Sample one balanced, JSON-friendly scene from an integer seed."""

    rng = random.Random(int(seed))
    outer_sides = SIDE_VALUES[int(seed) % len(SIDE_VALUES)]
    # A small fraction deliberately approaches the observable pixel boundary;
    # those scenes remain renderable but are quarantined by is_quarantined().
    near_boundary = int(seed) % 13 == 0
    radius = rng.uniform(33.0, 45.0) if near_boundary else rng.uniform(68.0, 124.0)
    pad = radius + 30.0
    center = [
        round(rng.uniform(pad, WIDTH - pad), 4),
        round(rng.uniform(pad, HEIGHT - pad), 4),
    ]
    return {
        "outer_sides": outer_sides,
        "center": center,
        "radius": round(radius, 4),
        "rotation_deg": round(rng.uniform(-180.0, 180.0), 4),
        "fill_color": list(rng.choice(FILL_PALETTE)),
        "background": list(rng.choice(BACKGROUND_PALETTE)),
        "outline_width": rng.choice((5, 6, 7)),
        "width": WIDTH,
        "height": HEIGHT,
        "antialias": True,
    }


def render(scene: dict) -> Image.Image:
    """Render a scene to the final RGB PNG raster."""

    width = int(scene["width"])
    height = int(scene["height"])
    supersample = 4 if bool(scene["antialias"]) else 1
    image = Image.new("RGB", (width * supersample, height * supersample), _as_rgb(scene["background"]))
    draw = ImageDraw.Draw(image)
    vertices = _regular_vertices(scene, supersample)
    draw.polygon(
        vertices,
        fill=_as_rgb(scene["fill_color"]),
        outline=OUTLINE,
        width=max(1, int(scene["outline_width"]) * supersample),
    )
    _draw_stop_bitmap(draw, scene, supersample)
    if supersample > 1:
        image = image.resize((width, height), Image.Resampling.LANCZOS)
    return image


def analytic_gold(scene: dict) -> str:
    """Gold is the visible polygon's declared side cardinality."""

    return str(int(scene["outer_sides"]))


def margin(scene: dict) -> float:
    """One-sided pixel margin: shortest straight run above a 24 px floor."""

    n = int(scene["outer_sides"])
    radius = float(scene["radius"])
    shortest_edge = 2.0 * radius * math.sin(math.pi / n)
    return round(shortest_edge - 24.0, 6)


def is_quarantined(scene: dict) -> bool:
    """Quarantine silhouettes whose shortest edge is near the raster floor."""

    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases are declared; every label input is visibly rendered."""

    return []
