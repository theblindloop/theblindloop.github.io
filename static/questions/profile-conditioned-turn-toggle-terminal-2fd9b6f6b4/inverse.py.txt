"""Independent PNG-only inverse arm for turn_toggle_terminal.

The oracle segments colored marker interiors, infers straight black links between
their centroids, then executes the public alternating egocentric branch rule.
It receives no latent state and imports no candidate-side rendering or gold code.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

PALETTE = {
    "black": (28, 31, 36),
    "green": (35, 170, 85),
    "purple": (135, 82, 200),
    "red": (220, 55, 55),
    "blue": (45, 105, 220),
    "orange": (240, 145, 35),
    "cyan": (20, 175, 190),
}
TERMINAL_COLORS = ("red", "blue", "orange", "cyan")


def _color_mask(array: np.ndarray, rgb: tuple[int, int, int], tolerance: int = 34) -> np.ndarray:
    # int32 prevents signed overflow when a channel is nearly 255 levels away.
    delta = array.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _components(mask: np.ndarray, minimum: int = 30) -> list[tuple[float, float, int]]:
    """Return 8-connected component centroids using only the sparse true pixels."""
    pixels = {tuple(value) for value in np.argwhere(mask)}
    found: list[tuple[float, float, int]] = []
    while pixels:
        seed = pixels.pop()
        stack = [seed]
        ys: list[int] = []
        xs: list[int] = []
        while stack:
            y, x = stack.pop()
            ys.append(y)
            xs.append(x)
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    neighbor = (y + dy, x + dx)
                    if neighbor in pixels:
                        pixels.remove(neighbor)
                        stack.append(neighbor)
        if len(xs) >= minimum:
            found.append((float(np.mean(xs)), float(np.mean(ys)), len(xs)))
    return sorted(found)


def _extract_nodes(array: np.ndarray) -> list[dict]:
    nodes: list[dict] = []
    expected = {"green": 1, "purple": 3, "red": 1, "blue": 1, "orange": 1, "cyan": 1}
    for color, count in expected.items():
        components = _components(_color_mask(array, PALETTE[color]))
        if len(components) != count:
            raise ValueError(f"expected {count} {color} marker components, found {len(components)}")
        for x, y, area in components:
            kind = "start" if color == "green" else ("junction" if color == "purple" else "terminal")
            nodes.append({"x": x, "y": y, "kind": kind, "color": color, "area": area})
    return nodes


def _line_coverage(mask: np.ndarray, left: dict, right: dict) -> float:
    """Measure black support through the open interval between two marker rims."""
    dx, dy = right["x"] - left["x"], right["y"] - left["y"]
    length = math.hypot(dx, dy)
    if length < 55:
        return 0.0
    endpoint_fraction = min(0.26, 19.0 / length)
    values = []
    height, width = mask.shape
    for t in np.linspace(endpoint_fraction, 1.0 - endpoint_fraction, 72):
        x = int(round(left["x"] + t * dx))
        y = int(round(left["y"] + t * dy))
        y0, y1 = max(0, y - 2), min(height, y + 3)
        x0, x1 = max(0, x - 2), min(width, x + 3)
        values.append(bool(mask[y0:y1, x0:x1].any()))
    return float(np.mean(values))


def _recover_graph(array: np.ndarray, nodes: list[dict]) -> dict[int, list[int]]:
    black = _color_mask(array, PALETTE["black"], tolerance=62)
    scored: list[tuple[float, int, int]] = []
    for left in range(len(nodes)):
        for right in range(left + 1, len(nodes)):
            score = _line_coverage(black, nodes[left], nodes[right])
            if score >= 0.82:
                scored.append((score, left, right))
    graph = {index: [] for index in range(len(nodes))}
    for _, left, right in scored:
        graph[left].append(right)
        graph[right].append(left)
    degrees = sorted(len(neighbors) for neighbors in graph.values())
    if len(scored) != 7 or degrees != [1, 1, 1, 1, 1, 3, 3, 3]:
        raise ValueError(f"recovered graph is not the declared binary tree: edges={len(scored)}, degrees={degrees}")
    return graph


def _turn(previous: dict, current: dict, nxt: dict) -> float:
    incoming = (current["x"] - previous["x"], previous["y"] - current["y"])
    outgoing = (nxt["x"] - current["x"], current["y"] - nxt["y"])
    cross = incoming[0] * outgoing[1] - incoming[1] * outgoing[0]
    dot = incoming[0] * outgoing[0] + incoming[1] * outgoing[1]
    if dot <= 0:
        raise ValueError("a recovered candidate branch is not forward")
    return math.atan2(cross, dot)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    if array.shape[:2] != (512, 512):
        raise ValueError(f"unexpected raster size {array.shape[:2]}")
    nodes = _extract_nodes(array)
    graph = _recover_graph(array, nodes)
    starts = [index for index, node in enumerate(nodes) if node["kind"] == "start"]
    if len(starts) != 1 or len(graph[starts[0]]) != 1:
        raise ValueError("START recovery failed")
    previous = starts[0]
    current = graph[previous][0]
    choose_left = True
    visited = {previous}
    while nodes[current]["kind"] == "junction":
        if current in visited:
            raise ValueError("recovered traversal loop")
        visited.add(current)
        candidates = [index for index in graph[current] if index != previous]
        if len(candidates) != 2:
            raise ValueError("visited junction does not have two forward branches")
        turns = [_turn(nodes[previous], nodes[current], nodes[index]) for index in candidates]
        if abs(turns[0] - turns[1]) < math.radians(12):
            raise ValueError("branch order is raster-ambiguous")
        chosen = max(range(2), key=turns.__getitem__) if choose_left else min(
            range(2), key=turns.__getitem__
        )
        previous, current = current, candidates[chosen]
        choose_left = not choose_left
    result = nodes[current]
    if result["kind"] != "terminal" or result["color"] not in TERMINAL_COLORS:
        raise ValueError("traversal did not reach one colored terminal")
    return result["color"]
