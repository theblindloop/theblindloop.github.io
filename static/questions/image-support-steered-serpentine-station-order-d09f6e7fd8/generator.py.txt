"""Analytic sampler and raster renderer for one directed winding trail."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

CANVAS = (640, 320)

_TEMPLATES = (
    ((42, 160), (105, 122), (230, 122), (175, 198), (350, 198), (290, 122), (495, 122), (445, 198), (598, 160)),
    ((42, 145), (135, 202), (260, 202), (210, 118), (380, 118), (330, 202), (520, 202), (470, 118), (598, 175)),
    ((42, 175), (115, 116), (245, 116), (190, 204), (365, 204), (305, 116), (500, 116), (450, 204), (598, 145)),
    ((42, 132), (128, 208), (252, 208), (198, 112), (372, 112), (318, 208), (510, 208), (458, 112), (598, 182)),
)

_PALETTES = (
    {
        "background": (249, 248, 244),
        "trail": (26, 53, 78),
        "green": (39, 151, 91),
        "orange": (231, 126, 37),
        "purple": (139, 78, 176),
    },
    {
        "background": (247, 249, 246),
        "trail": (34, 58, 84),
        "green": (25, 143, 92),
        "orange": (222, 112, 31),
        "purple": (151, 70, 169),
    },
    {
        "background": (250, 247, 245),
        "trail": (23, 49, 75),
        "green": (53, 139, 82),
        "orange": (238, 138, 43),
        "purple": (124, 82, 184),
    },
)


def _transform(points: Iterable[tuple[int, int]], mirror_x: bool, mirror_y: bool, dx: int, dy: int) -> list[list[int]]:
    transformed = []
    for x, y in points:
        if mirror_x:
            x = CANVAS[0] - x
        if mirror_y:
            y = CANVAS[1] - y
        transformed.append([x + dx, y + dy])
    return transformed


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x5E71A710)
    template = _TEMPLATES[(int(seed) // 4) % len(_TEMPLATES)]
    mirror_x = bool((int(seed) // 2) & 1)
    mirror_y = bool((int(seed) // 8) & 1)
    dx = rng.randint(-7, 7)
    dy = rng.randint(-7, 7)
    return {
        "canvas_size": list(CANVAS),
        "path_points": _transform(template, mirror_x, mirror_y, dx, dy),
        "orange_first": bool(int(seed) & 1),
        "station_fractions": [0.29, 0.71],
        "line_width": 12 + 2 * ((int(seed) // 3) % 3),
        "marker_radius": 15 + ((int(seed) // 5) % 3),
        "palette_variant": (int(seed) // 7) % len(_PALETTES),
    }


def _polyline_lengths(points: list[list[int]]) -> tuple[list[float], float]:
    lengths = [0.0]
    for left, right in zip(points, points[1:]):
        lengths.append(lengths[-1] + math.hypot(right[0] - left[0], right[1] - left[1]))
    return lengths, lengths[-1]


def _point_at_fraction(points: list[list[int]], fraction: float) -> tuple[float, float]:
    lengths, total = _polyline_lengths(points)
    target = total * float(fraction)
    for index in range(len(points) - 1):
        if target <= lengths[index + 1]:
            span = lengths[index + 1] - lengths[index]
            ratio = 0.0 if span == 0 else (target - lengths[index]) / span
            x = points[index][0] + ratio * (points[index + 1][0] - points[index][0])
            y = points[index][1] + ratio * (points[index + 1][1] - points[index][1])
            return x, y
    return float(points[-1][0]), float(points[-1][1])


def render(scene: dict) -> Image.Image:
    width, height = map(int, scene["canvas_size"])
    points = [[int(value) for value in point] for point in scene["path_points"]]
    orange_first = bool(scene["orange_first"])
    fractions = [float(value) for value in scene["station_fractions"]]
    line_width = int(scene["line_width"])
    radius = int(scene["marker_radius"])
    palette = _PALETTES[int(scene["palette_variant"])]

    scale = 4
    image = Image.new("RGB", (width * scale, height * scale), palette["background"])
    draw = ImageDraw.Draw(image)
    scaled_points = [(x * scale, y * scale) for x, y in points]
    draw.line(
        scaled_points,
        fill=palette["trail"],
        width=line_width * scale,
        joint="curve",
    )
    cap_r = line_width * scale // 2
    for x, y in (scaled_points[0], scaled_points[-1]):
        draw.ellipse((x - cap_r, y - cap_r, x + cap_r, y + cap_r), fill=palette["trail"])

    first_color, second_color = (
        ("orange", "purple") if orange_first else ("purple", "orange")
    )
    stations = (
        (_point_at_fraction(points, fractions[0]), first_color),
        (_point_at_fraction(points, fractions[1]), second_color),
    )
    outline = max(2, line_width // 4) * scale
    for (x, y), color_name in stations:
        cx, cy = round(x * scale), round(y * scale)
        rr = radius * scale
        draw.ellipse(
            (cx - rr, cy - rr, cx + rr, cy + rr),
            fill=palette[color_name],
            outline=palette["trail"],
            width=outline,
        )

    sx, sy = scaled_points[0]
    sr = (radius + 2) * scale
    draw.ellipse(
        (sx - sr, sy - sr, sx + sr, sy + sr),
        fill=palette["green"],
        outline=palette["trail"],
        width=outline,
    )

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "yes" if bool(scene["orange_first"]) else "no"


def margin(scene: dict) -> float:
    points = scene["path_points"]
    fractions = scene["station_fractions"]
    _, total = _polyline_lengths(points)
    return float(total * abs(float(fractions[1]) - float(fractions[0])) - 2 * int(scene["marker_radius"]))


def is_quarantined(scene: dict) -> bool:
    width, height = scene["canvas_size"]
    points = scene["path_points"]
    radius = int(scene["marker_radius"])
    border = min(min(x, width - x, y, height - y) for x, y in points)
    return bool(
        len(points) < 4
        or int(scene["line_width"]) < 12
        or radius < 14
        or border < radius + 8
        or margin(scene) < 180.0
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
