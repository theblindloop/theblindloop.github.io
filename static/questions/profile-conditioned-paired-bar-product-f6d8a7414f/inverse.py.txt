"""Independent inverse arm that receives only the final RGB raster."""

from __future__ import annotations

import numpy as np
from PIL import Image

LABELS = "ABCD"


def _active_runs(values: np.ndarray) -> list[tuple[int, int]]:
    indices = np.flatnonzero(values)
    if len(indices) == 0:
        return []
    runs: list[tuple[int, int]] = []
    start = previous = int(indices[0])
    for value in indices[1:]:
        current = int(value)
        if current != previous + 1:
            runs.append((start, previous))
            start = current
        previous = current
    runs.append((start, previous))
    return runs


def decision_from_image(image: Image.Image) -> str:
    """Segment eight saturated rectangles and compare paired height products."""
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    spread = array.max(axis=2) - array.min(axis=2)
    mask = (spread > 65) & (array.max(axis=2) < 246)
    runs = _active_runs(mask.any(axis=0))
    if len(runs) != 8:
        raise ValueError(f"expected eight separated colored bars, found {len(runs)}")

    components: list[tuple[int, int, int, int]] = []
    for x0, x1 in runs:
        ys, xs = np.nonzero(mask[:, x0 : x1 + 1])
        if not 25 <= x1 - x0 + 1 <= 35 or len(xs) < 700:
            raise ValueError("colored component is not a complete bar")
        components.append((x0, x1, int(ys.min()), int(ys.max())))

    baselines = [component[3] for component in components]
    if max(baselines) - min(baselines) > 1:
        raise ValueError("bars do not share one visible baseline")
    starts = [component[0] for component in components]
    within_gaps = [starts[2 * row + 1] - starts[2 * row] for row in range(4)]
    between_gaps = [starts[2 * row + 2] - starts[2 * row + 1] for row in range(3)]
    if max(within_gaps) >= min(between_gaps):
        raise ValueError("colored bars cannot be paired into four groups")

    heights = [component[3] - component[2] + 1 for component in components]
    scores = [heights[2 * row] * heights[2 * row + 1] for row in range(4)]
    order = sorted(range(4), key=lambda index: scores[index], reverse=True)
    if scores[order[0]] - scores[order[1]] < 5:
        raise ValueError("product evidence is tied at raster resolution")
    return LABELS[order[0]]
