"""Deterministic scene sampler and Pillow raster renderer."""

from __future__ import annotations

import copy
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 720, 430
LABELS = "ABCD"
FONT_REGULAR = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
FONT_BOLD = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"

PALETTES = (
    {"first": [38, 104, 196], "second": [226, 112, 38]},
    {"first": [35, 132, 108], "second": [184, 72, 128]},
    {"first": [67, 97, 194], "second": [230, 140, 38]},
)

# In every safe template, the product winner is neither the largest-sum pair,
# the pair containing the tallest bar, nor the most equal pair. These competing
# proxies are deliberately assigned to different distractors.
SAFE_TEMPLATES = (
    ((74, 76), (108, 46), (126, 30), (66, 67)),
    ((78, 71), (104, 47), (124, 31), (64, 65)),
    ((69, 82), (105, 46), (118, 36), (66, 67)),
    ((80, 73), (110, 42), (126, 29), (68, 69)),
)

# Visible near-boundary cases exercise quarantine without becoming raster ties.
NEAR_TEMPLATE = ((74, 75), (76, 72), (110, 44), (66, 66))


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    path = FONT_BOLD if bold else FONT_REGULAR
    if not Path(path).is_file():
        raise RuntimeError(f"required deterministic font is missing: {path}")
    return ImageFont.truetype(path, size=size)


def _product(pair) -> int:
    return int(pair[0]) * int(pair[1])


def sample_scene(seed: int) -> dict:
    """Return a balanced scene; ``seed % 4`` fixes the winning group label."""
    rng = random.Random(int(seed))
    source = NEAR_TEMPLATE if int(seed) % 13 == 0 else SAFE_TEMPLATES[int(seed) % 4]
    pairs = [tuple(pair) for pair in source]
    rng.shuffle(pairs)
    winner = max(range(4), key=lambda index: _product(pairs[index]))
    target = int(seed) % 4
    pairs[winner], pairs[target] = pairs[target], pairs[winner]
    pairs = [tuple(reversed(pair)) if rng.getrandbits(1) else pair for pair in pairs]
    draw_order = list(range(8))
    rng.shuffle(draw_order)
    return {
        "pairs": [list(pair) for pair in pairs],
        "palette": copy.deepcopy(PALETTES[rng.randrange(len(PALETTES))]),
        "offset_x": [-8, 0, 8][rng.randrange(3)],
        "offset_y": [-4, 0, 4][rng.randrange(3)],
        "draw_order": draw_order,
    }


def analytic_gold(scene: dict) -> str:
    scores = [_product(pair) for pair in scene["pairs"]]
    return LABELS[max(range(4), key=lambda index: scores[index])]


def margin(scene: dict) -> float:
    scores = sorted((_product(pair) for pair in scene["pairs"]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 400.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["draw_order"] = list(reversed(twin["draw_order"]))
    return [("reverse_disjoint_draw_order", twin)]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (248, 249, 247))
    draw = ImageDraw.Draw(image)
    offset_x = int(scene["offset_x"])
    offset_y = int(scene["offset_y"])
    chart_left = 126 + offset_x
    baseline = 342 + offset_y
    group_pitch = 132
    bar_width = 30
    bar_gap = 10

    draw.text((34, 24), "PAIRED BAR PRODUCT", font=_font(23, True), fill=(36, 39, 43))
    draw.text(
        (35, 58),
        "multiply the two heights in each group",
        font=_font(14),
        fill=(83, 88, 94),
    )
    draw.rounded_rectangle(
        (chart_left - 34, 180 + offset_y, chart_left + 3 * group_pitch + 104, baseline + 48),
        radius=9,
        fill=(255, 255, 255),
        outline=(184, 188, 193),
        width=2,
    )
    for value in (0, 32, 64, 96, 128):
        y = baseline - value
        color = (113, 118, 124) if value == 0 else (225, 227, 229)
        width = 3 if value == 0 else 1
        draw.line(
            (chart_left - 12, y, chart_left + 3 * group_pitch + 82, y),
            fill=color,
            width=width,
        )
        label = str(value)
        box = draw.textbbox((0, 0), label, font=_font(10))
        draw.text(
            (chart_left - 18 - (box[2] - box[0]), y - 6),
            label,
            font=_font(10),
            fill=(104, 109, 114),
        )

    bars = []
    for group, pair in enumerate(scene["pairs"]):
        group_x = chart_left + group * group_pitch
        for within, height in enumerate(pair):
            x0 = group_x + within * (bar_width + bar_gap)
            bars.append(
                (
                    x0,
                    baseline - int(height),
                    x0 + bar_width - 1,
                    baseline - 1,
                    tuple(scene["palette"]["first" if within == 0 else "second"]),
                )
            )
    for bar_index in scene["draw_order"]:
        x0, y0, x1, y1, color = bars[int(bar_index)]
        draw.rectangle((x0, y0, x1, y1), fill=color)

    for group, label in enumerate(LABELS):
        center = chart_left + group * group_pitch + (2 * bar_width + bar_gap) / 2
        box = draw.textbbox((0, 0), label, font=_font(19, True))
        draw.text(
            (center - (box[2] - box[0]) / 2, baseline + 13),
            label,
            font=_font(19, True),
            fill=(36, 39, 43),
        )
    return image
