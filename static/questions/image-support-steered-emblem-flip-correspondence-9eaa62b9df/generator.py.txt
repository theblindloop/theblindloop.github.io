"""Deterministic Pillow renderer and analytic arm for two reordered snapshots."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 800
HEIGHT = 800
IDENTITIES = ("circle", "triangle", "square")

_BACKGROUND = (255, 255, 255)
_INK = (30, 36, 48)
_FRAME = (218, 224, 232)
_CAPTION = (105, 114, 128)
_DISTRACTOR = (235, 238, 243)
_PALETTES = (
    ((61, 154, 222), (231, 126, 71), (101, 181, 112)),
    ((144, 112, 214), (222, 145, 55), (64, 170, 164)),
    ((215, 102, 137), (76, 150, 211), (117, 173, 76)),
)


def sample_scene(seed: int) -> dict:
    """Balance the changed identity and flip direction before varying appearance."""
    seed = int(seed)
    target_index = seed % len(IDENTITIES)
    quotient = seed // len(IDENTITIES)
    states_before = {
        identity: int((quotient >> index) & 1)
        for index, identity in enumerate(IDENTITIES)
    }
    states_after = dict(states_before)
    target = IDENTITIES[target_index]
    states_after[target] = 1 - states_after[target]

    order_rng = random.Random(seed * 1009 + 41)
    order_before = list(IDENTITIES)
    order_after = list(IDENTITIES)
    order_rng.shuffle(order_before)
    order_rng.shuffle(order_after)

    style_rng = random.Random(seed * 7919 + 233)
    identity_colors = list(range(3))
    style_rng.shuffle(identity_colors)
    return {
        "states_before": states_before,
        "states_after": states_after,
        "order_before": order_before,
        "order_after": order_after,
        "palette_variant": style_rng.randrange(len(_PALETTES)),
        "identity_colors": identity_colors,
        "anchor_dx": style_rng.choice((-8, 0, 8)),
        "anchor_dy": style_rng.choice((-8, 0, 8)),
        "card_gap": style_rng.choice((332, 340, 348)),
        "token_size": style_rng.choice((20, 21, 22)),
        "distractor_variant": style_rng.randrange(4),
    }


def _draw_distractors(draw: ImageDraw.ImageDraw, variant: int) -> None:
    layouts = (
        ((72, 72), (728, 728)),
        ((728, 72), (72, 728)),
        ((72, 72), (728, 72), (72, 728)),
        ((728, 728), (728, 72), (72, 728)),
    )
    for x, y in layouts[int(variant)]:
        draw.ellipse((x - 7, y - 7, x + 7, y + 7), outline=_DISTRACTOR, width=2)
        draw.line((x - 4, y, x + 4, y), fill=_DISTRACTOR, width=2)


def _draw_emblem(
    draw: ImageDraw.ImageDraw,
    identity: str,
    center: tuple[int, int],
    size: int,
    color: tuple[int, int, int],
    state: int,
) -> None:
    x, y = center
    if identity == "circle":
        draw.ellipse((x - size, y - size, x + size, y + size), fill=color)
    elif identity == "triangle":
        draw.polygon(
            ((x, y - size - 2), (x + size + 2, y + size), (x - size - 2, y + size)),
            fill=color,
        )
    elif identity == "square":
        draw.rectangle((x - size, y - size, x + size, y + size), fill=color)
    else:
        raise ValueError(f"unknown emblem identity {identity!r}")

    half_length = 8
    half_thickness = 2
    if int(state):
        box = (x - half_thickness, y - half_length, x + half_thickness, y + half_length)
    else:
        box = (x - half_length, y - half_thickness, x + half_length, y + half_thickness)
    draw.rectangle(box, fill=_INK)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), _BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_distractors(draw, int(scene["distractor_variant"]))

    dx = int(scene["anchor_dx"])
    dy = int(scene["anchor_dy"])
    gap = int(scene["card_gap"])
    size = int(scene["token_size"])
    card_centers = (WIDTH // 2 - gap // 2 + dx, WIDTH // 2 + gap // 2 + dx)
    slot_ys = (310 + dy, 400 + dy, 490 + dy)
    palette = _PALETTES[int(scene["palette_variant"])]
    color_indices = list(scene["identity_colors"])
    color_for = {
        identity: palette[int(color_indices[index])]
        for index, identity in enumerate(IDENTITIES)
    }
    font = ImageFont.load_default(size=16)

    for card_index, (card_x, order, states, caption) in enumerate(
        (
            (card_centers[0], list(scene["order_before"]), scene["states_before"], "BEFORE"),
            (card_centers[1], list(scene["order_after"]), scene["states_after"], "AFTER"),
        )
    ):
        draw.rounded_rectangle(
            (card_x - 50, 252 + dy, card_x + 50, 548 + dy),
            radius=12,
            outline=_FRAME,
            width=3,
        )
        draw.text((card_x, 269 + dy), caption, fill=_CAPTION, anchor="mm", font=font)
        for slot, identity in enumerate(order):
            _draw_emblem(
                draw,
                identity,
                (card_x, slot_ys[slot]),
                size,
                color_for[identity],
                int(states[identity]),
            )
        if card_index == 0:
            draw.polygon(
                ((card_x - 8, 531 + dy), (card_x + 8, 531 + dy), (card_x, 540 + dy)),
                fill=_FRAME,
            )
        else:
            draw.polygon(
                ((card_x - 8, 540 + dy), (card_x + 8, 540 + dy), (card_x, 531 + dy)),
                fill=_FRAME,
            )
    return image


def analytic_gold(scene: dict) -> str:
    before = scene["states_before"]
    after = scene["states_after"]
    changed = [
        identity
        for identity in IDENTITIES
        if int(before[identity]) != int(after[identity])
    ]
    if len(changed) != 1:
        raise ValueError(f"expected exactly one changed emblem, found {changed}")
    return changed[0]


def margin(scene: dict) -> float:
    """Conservative runner-up clearance for shape and bar-orientation readings."""
    return float(min(int(scene["token_size"]) - 8, int(scene["card_gap"]) - 300, 12))


def is_quarantined(scene: dict) -> bool:
    try:
        before = scene["states_before"]
        after = scene["states_after"]
        changed = [
            identity
            for identity in IDENTITIES
            if int(before[identity]) != int(after[identity])
        ]
        return (
            set(before) != set(IDENTITIES)
            or set(after) != set(IDENTITIES)
            or any(int(before[item]) not in (0, 1) for item in IDENTITIES)
            or any(int(after[item]) not in (0, 1) for item in IDENTITIES)
            or len(changed) != 1
            or sorted(scene["order_before"]) != sorted(IDENTITIES)
            or sorted(scene["order_after"]) != sorted(IDENTITIES)
            or sorted(int(value) for value in scene["identity_colors"]) != [0, 1, 2]
            or int(scene["palette_variant"]) not in range(len(_PALETTES))
            or int(scene["anchor_dx"]) not in (-8, 0, 8)
            or int(scene["anchor_dy"]) not in (-8, 0, 8)
            or int(scene["card_gap"]) not in (332, 340, 348)
            or int(scene["token_size"]) not in (20, 21, 22)
            or int(scene["distractor_variant"]) not in range(4)
            or margin(scene) < 10.0
        )
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
