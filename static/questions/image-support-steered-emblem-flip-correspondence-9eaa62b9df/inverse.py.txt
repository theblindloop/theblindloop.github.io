"""Independent PNG-only inverse for two reordered emblem snapshots.

The inverse locates six chromatic components, separates them into left and
right groups, classifies shape identity from silhouette fill, reads the black
bar orientation inside each silhouette, and compares matching identities.
It receives no latent scene and imports no candidate implementation modules.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int = 1) -> list[np.ndarray]:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if visited[y0, x0]:
            continue
        queue = deque([(y0, x0)])
        visited[y0, x0] = True
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            output.append(np.asarray(points, dtype=np.int32))
    return output


def _bounds(points: np.ndarray) -> tuple[int, int, int, int]:
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    return int(x0), int(y0), int(x1), int(y1)


def _identity(points: np.ndarray) -> tuple[str, tuple[int, int, int, int]]:
    bounds = _bounds(points)
    x0, y0, x1, y1 = bounds
    width, height = x1 - x0 + 1, y1 - y0 + 1
    if not (38 <= width <= 53 and 38 <= height <= 49):
        raise ValueError(f"emblem silhouette has invalid bounds {width}x{height}")
    fill = len(points) / float(width * height)
    if 0.36 <= fill <= 0.56:
        identity = "triangle"
    elif 0.62 <= fill <= 0.79:
        identity = "circle"
    elif 0.86 <= fill <= 0.96:
        identity = "square"
    else:
        raise ValueError(f"ambiguous emblem silhouette fill {fill:.3f}")
    return identity, bounds


def _state(array: np.ndarray, bounds: tuple[int, int, int, int]) -> int:
    x0, y0, x1, y1 = bounds
    cx = round((x0 + x1) / 2.0)
    cy = round((y0 + y1) / 2.0)
    crop = array[cy - 13 : cy + 14, cx - 13 : cx + 14]
    if crop.shape[:2] != (27, 27):
        raise ValueError("state mark is clipped")
    dark = np.max(crop, axis=2) < 80
    ys, xs = np.where(dark)
    if not (85 <= len(xs) <= 125):
        raise ValueError(f"state mark has invalid ink area {len(xs)}")
    width = int(xs.max() - xs.min() + 1)
    height = int(ys.max() - ys.min() + 1)
    if height >= 3 * width:
        return 1
    if width >= 3 * height:
        return 0
    raise ValueError(f"state mark orientation is ambiguous ({width}x{height})")


def _decode_card(
    array: np.ndarray, parts: list[np.ndarray]
) -> dict[str, int]:
    decoded: dict[str, int] = {}
    for part in parts:
        identity, bounds = _identity(part)
        if identity in decoded:
            raise ValueError(f"card repeats {identity}")
        decoded[identity] = _state(array, bounds)
    if set(decoded) != {"circle", "triangle", "square"}:
        raise ValueError(f"card identities are incomplete: {sorted(decoded)}")
    return decoded


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.shape != (800, 800, 3):
        raise ValueError(f"expected an 800x800 RGB raster, found {array.shape}")
    values = array.astype(np.int16)
    chroma = values.max(axis=2) - values.min(axis=2)
    colored = (chroma > 55) & (values.mean(axis=2) < 235)
    parts = _components(colored, minimum=500)
    if len(parts) != 6:
        raise ValueError(f"expected six intact chromatic emblems, found {len(parts)}")

    measured = []
    for part in parts:
        x0, y0, x1, y1 = _bounds(part)
        measured.append(((x0 + x1) / 2.0, (y0 + y1) / 2.0, part))
    measured.sort(key=lambda item: item[0])
    left = measured[:3]
    right = measured[3:]
    left_x = sum(item[0] for item in left) / 3.0
    right_x = sum(item[0] for item in right) / 3.0
    if not (300 <= right_x - left_x <= 380):
        raise ValueError("the two snapshot groups are not distinctly separated")
    if max(abs(item[0] - left_x) for item in left) > 3:
        raise ValueError("left snapshot emblems do not share one card column")
    if max(abs(item[0] - right_x) for item in right) > 3:
        raise ValueError("right snapshot emblems do not share one card column")

    before = _decode_card(array, [item[2] for item in left])
    after = _decode_card(array, [item[2] for item in right])
    changed = [
        identity
        for identity in ("circle", "triangle", "square")
        if before[identity] != after[identity]
    ]
    if len(changed) != 1:
        raise ValueError(f"expected one visible state change, found {changed}")
    return changed[0]
