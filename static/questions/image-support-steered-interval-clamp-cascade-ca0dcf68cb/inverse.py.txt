"""Independent image-only inverse for the nearest-band clamp cascade.

Only the final raster and the public visual contract are used here.  This file
does not import renderer, prompts, generate, verify, scenes, or analytic gold.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_LANES = 5
_STEPS = 6
_CELL_WIDTH = 76
_HEADER_HEIGHT = 52
_ROW_HEIGHT = 56
_PAD = 5
_GRID = np.array((37, 44, 53), dtype=np.int16)
_OFF = np.array((250, 248, 241), dtype=np.float64)
_MARKER = np.array((236, 137, 43), dtype=np.int16)
_LABELS = ("A", "B", "C", "D", "E")
_LENGTHS = [1, 1, 2, 2, 3, 3]


class PixelParseError(ValueError):
    pass


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[np.ndarray] = []
    for start_y, start_x in np.argwhere(mask):
        if seen[start_y, start_x]:
            continue
        queue = deque([(int(start_y), int(start_x))])
        seen[start_y, start_x] = True
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for yy, xx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= yy < height and 0 <= xx < width and mask[yy, xx] and not seen[yy, xx]:
                    seen[yy, xx] = True
                    queue.append((yy, xx))
        groups.append(np.asarray(points, dtype=np.int32))
    return groups


def _parse(image: Image.Image) -> tuple[int, list[tuple[int, int]]]:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[2] != 3 or min(array.shape[:2]) < 400:
        raise PixelParseError("unsupported raster dimensions")
    values = array.astype(np.int32)
    dark = np.sqrt(np.sum((values - _GRID) ** 2, axis=2)) < 12.0
    groups = _components(dark)
    if not groups:
        raise PixelParseError("connected board is missing")
    board = max(groups, key=len)
    y_min, x_min = board.min(axis=0)
    y_max, x_max = board.max(axis=0)
    board_width = int(x_max - x_min + 1)
    board_height = int(y_max - y_min + 1)
    expected_width = _LANES * _CELL_WIDTH
    expected_height = _HEADER_HEIGHT + _STEPS * _ROW_HEIGHT
    if board_width != expected_width or board_height != expected_height or len(board) < 30000:
        raise PixelParseError("connected board geometry is incomplete")

    local_dark = dark[y_min : y_max + 1, x_min : x_max + 1]
    expected_gutter = np.ones((expected_height, expected_width), dtype=bool)
    for lane in range(_LANES):
        left = lane * _CELL_WIDTH + _PAD
        right = (lane + 1) * _CELL_WIDTH - _PAD
        expected_gutter[_PAD : _HEADER_HEIGHT - _PAD, left:right] = False
    for row in range(_STEPS):
        top = _HEADER_HEIGHT + row * _ROW_HEIGHT + _PAD
        bottom = _HEADER_HEIGHT + (row + 1) * _ROW_HEIGHT - _PAD
        for lane in range(_LANES):
            left = lane * _CELL_WIDTH + _PAD
            right = (lane + 1) * _CELL_WIDTH - _PAD
            expected_gutter[top:bottom, left:right] = False
    if float(local_dark[expected_gutter].mean()) < 0.992:
        raise PixelParseError("board gutters are damaged or occluded")

    marker_mask = np.sqrt(np.sum((values - _MARKER) ** 2, axis=2)) < 14.0
    marker_groups = [group for group in _components(marker_mask) if len(group) >= 100]
    header_groups = []
    for group in marker_groups:
        cy, cx = group.mean(axis=0)
        if y_min + 4 <= cy < y_min + _HEADER_HEIGHT - 4 and x_min <= cx <= x_max:
            header_groups.append(group)
    if len(header_groups) != 1:
        raise PixelParseError("exactly one start ring is required")
    marker_y, marker_x = header_groups[0].mean(axis=0)
    start_float = (marker_x - x_min) / _CELL_WIDTH
    start_lane = int(start_float)
    expected_x = x_min + start_lane * _CELL_WIDTH + _CELL_WIDTH / 2.0
    if not 0 <= start_lane < _LANES or abs(marker_x - expected_x) > 2.0:
        raise PixelParseError("start ring is not centered in a lane")

    intervals: list[tuple[int, int]] = []
    for row in range(_STEPS):
        active: list[int] = []
        cy = y_min + _HEADER_HEIGHT + row * _ROW_HEIGHT + _ROW_HEIGHT // 2
        for lane in range(_LANES):
            cx = x_min + lane * _CELL_WIDTH + _CELL_WIDTH // 2
            sample = array[cy - 7 : cy + 8, cx - 12 : cx + 13].astype(np.float64)
            flat = sample.reshape(-1, 3)
            mean = flat.mean(axis=0)
            if float(flat.std(axis=0).max()) > 9.0:
                raise PixelParseError("cell interior is visually ambiguous")
            distance = float(np.linalg.norm(mean - _OFF))
            if distance < 18.0:
                continue
            if distance <= 80.0:
                raise PixelParseError("active/off color separation is ambiguous")
            active.append(lane)
        if not active or active != list(range(active[0], active[-1] + 1)):
            raise PixelParseError("each row must contain one contiguous highlighted band")
        intervals.append((active[0], active[-1]))
    if sorted(upper - lower + 1 for lower, upper in intervals) != _LENGTHS:
        raise PixelParseError("interval-length balance is missing")
    return start_lane, intervals


def decision_from_image(image: Image.Image) -> str:
    position, intervals = _parse(image)
    for lower, upper in intervals:
        position = max(lower, min(upper, position))
    return _LABELS[position]
