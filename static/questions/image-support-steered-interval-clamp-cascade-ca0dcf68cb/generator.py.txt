"""Deterministic scene sampler, analytic arm, and Pillow renderer."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont

CANVAS = 512
LANES = 5
STEPS = 6
CELL_WIDTH = 76
HEADER_HEIGHT = 52
ROW_HEIGHT = 56
PAD = 5
BACKGROUND = (246, 243, 235)
GRID = (37, 44, 53)
OFF = (250, 248, 241)
MARKER = (236, 137, 43)
ACTIVE_COLORS = (
    (42, 139, 135),
    (80, 116, 181),
    (112, 91, 170),
    (38, 151, 111),
)
LABELS = ("A", "B", "C", "D", "E")
LENGTH_MULTISET = (1, 1, 2, 2, 3, 3)


def _clamp(value: int, interval: list[int] | tuple[int, int]) -> int:
    lower, upper = int(interval[0]), int(interval[1])
    return max(lower, min(upper, value))


def _replay(scene: dict) -> int:
    position = int(scene["start_lane"])
    for interval in scene["intervals"]:
        position = _clamp(position, interval)
    return position


def sample_scene(seed: int) -> dict:
    seed = int(seed)
    desired = seed % LANES
    rng = random.Random(seed ^ 0x5A17C1A5)
    chosen: tuple[int, list[list[int]]] | None = None
    for _ in range(20000):
        start = rng.randrange(LANES)
        lengths = list(LENGTH_MULTISET)
        rng.shuffle(lengths)
        intervals: list[list[int]] = []
        for length in lengths:
            lower = rng.randrange(LANES - length + 1)
            intervals.append([lower, lower + length - 1])
        states = [start]
        for interval in intervals:
            states.append(_clamp(states[-1], interval))
        changes = sum(left != right for left, right in zip(states, states[1:]))
        if states[-1] == desired and changes >= 4 and states[0] != states[1] and states[-2] != states[-1]:
            chosen = (start, intervals)
            break
    if chosen is None:
        raise RuntimeError(f"could not sample balanced clamp scene for seed {seed}")

    style = seed // LANES
    return {
        "start_lane": chosen[0],
        "intervals": chosen[1],
        "board_x": 66 + (style % 3) - 1,
        "board_y": 68 + ((style // 3) % 3) - 1,
        "cell_width": CELL_WIDTH,
        "header_height": HEADER_HEIGHT,
        "row_height": ROW_HEIGHT,
        "active_color": list(ACTIVE_COLORS[style % len(ACTIVE_COLORS)]),
        "background": list(BACKGROUND),
        "grid_color": list(GRID),
        "off_color": list(OFF),
        "marker_color": list(MARKER),
    }


def analytic_gold(scene: dict) -> str:
    return LABELS[_replay(scene)]


def margin(scene: dict) -> float:
    """One cell-state edit is the nearest discrete competing parse."""

    return 1.0


def is_quarantined(scene: dict) -> bool:
    try:
        intervals = [[int(pair[0]), int(pair[1])] for pair in scene["intervals"]]
        lengths = sorted(upper - lower + 1 for lower, upper in intervals)
        valid_intervals = (
            len(intervals) == STEPS
            and lengths == sorted(LENGTH_MULTISET)
            and all(0 <= lower <= upper < LANES for lower, upper in intervals)
        )
        start = int(scene["start_lane"])
        active = tuple(int(value) for value in scene["active_color"])
        off = tuple(int(value) for value in scene["off_color"])
        clearance_sq = sum((left - right) ** 2 for left, right in zip(active, off, strict=True))
        board_ok = (
            int(scene["cell_width"]) == CELL_WIDTH
            and int(scene["header_height"]) == HEADER_HEIGHT
            and int(scene["row_height"]) == ROW_HEIGHT
        )
        return not (valid_intervals and 0 <= start < LANES and clearance_sq >= 80 * 80 and board_ok)
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int, *, bold: bool = False) -> ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    start_lane = int(scene["start_lane"])
    intervals = scene["intervals"]
    x0 = int(scene["board_x"])
    y0 = int(scene["board_y"])
    cell_width = int(scene["cell_width"])
    header_height = int(scene["header_height"])
    row_height = int(scene["row_height"])
    active_color = tuple(scene["active_color"])
    background = tuple(scene["background"])
    grid_color = tuple(scene["grid_color"])
    off_color = tuple(scene["off_color"])
    marker_color = tuple(scene["marker_color"])

    image = Image.new("RGB", (CANVAS, CANVAS), background)
    draw = ImageDraw.Draw(image)
    board_width = LANES * cell_width
    board_height = header_height + STEPS * row_height

    draw.text((132, 22), "CLAMP CASCADE", font=_font(22, bold=True), fill=grid_color)
    draw.text((349, 28), "TOP TO BOTTOM", font=_font(10, bold=True), fill=marker_color)
    for lane, label in enumerate(LABELS):
        cx = x0 + lane * cell_width + cell_width // 2
        draw.text((cx - 7, y0 - 24), label, font=_font(17, bold=True), fill=grid_color)

    draw.rectangle((x0, y0, x0 + board_width - 1, y0 + board_height - 1), fill=grid_color)

    for lane in range(LANES):
        left = x0 + lane * cell_width + PAD
        right = x0 + (lane + 1) * cell_width - PAD - 1
        draw.rounded_rectangle(
            (left, y0 + PAD, right, y0 + header_height - PAD - 1),
            radius=5,
            fill=off_color,
        )

    start_cx = x0 + start_lane * cell_width + cell_width // 2
    start_cy = y0 + header_height // 2
    draw.ellipse(
        (start_cx - 15, start_cy - 15, start_cx + 15, start_cy + 15),
        outline=marker_color,
        width=5,
    )
    draw.ellipse((start_cx - 3, start_cy - 3, start_cx + 3, start_cy + 3), fill=marker_color)

    for row, interval in enumerate(intervals):
        lower, upper = int(interval[0]), int(interval[1])
        top = y0 + header_height + row * row_height + PAD
        bottom = y0 + header_height + (row + 1) * row_height - PAD - 1
        for lane in range(LANES):
            left = x0 + lane * cell_width + PAD
            right = x0 + (lane + 1) * cell_width - PAD - 1
            fill = active_color if lower <= lane <= upper else off_color
            draw.rounded_rectangle((left, top, right, bottom), radius=5, fill=fill)

        # Paired white ticks visibly identify the highlighted band's two ends.
        tick_y = (top + bottom) // 2
        left_edge = x0 + lower * cell_width + PAD + 8
        right_edge = x0 + (upper + 1) * cell_width - PAD - 9
        draw.line((left_edge, tick_y - 8, left_edge, tick_y + 8), fill=(255, 255, 255), width=3)
        draw.line((right_edge, tick_y - 8, right_edge, tick_y + 8), fill=(255, 255, 255), width=3)

    draw.text((12, y0 + 14), "START", font=_font(11, bold=True), fill=grid_color)
    for row in range(STEPS):
        cy = y0 + header_height + row * row_height + row_height // 2
        draw.text((31, cy - 8), str(row + 1), font=_font(13, bold=True), fill=grid_color)
        if row < STEPS - 1:
            draw.line((48, cy + 12, 48, cy + row_height - 12), fill=marker_color, width=3)
    return image
