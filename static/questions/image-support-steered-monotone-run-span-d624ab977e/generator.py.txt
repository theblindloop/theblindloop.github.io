"""Latent sampler, analytic arm, and deterministic raster renderer."""

from __future__ import annotations

import copy
import random
from typing import Iterable

from PIL import Image, ImageDraw

CANVAS = 512
SEGMENTS = 12
TOTAL_WIDTH = 84
TRACE_RADIUS = 2
PALE = (224, 229, 231)
PALE_2 = (237, 240, 241)
BACKGROUND = (250, 249, 246)

_INKS = {
    "lagoon": (16, 150, 158),
    "lagoon_alias": (16, 150, 158),
    "deep_teal": (22, 126, 151),
    "deep_teal_alias": (22, 126, 151),
    "aqua": (12, 166, 153),
    "aqua_alias": (12, 166, 153),
}


def _rotated_runs(rng: random.Random) -> list[int]:
    runs = [4, 1, 1, 2, 1, 3]
    shift = 2 * rng.randrange(3)
    return runs[shift:] + runs[:shift]


def _bounded_fill(rng: random.Random, count: int, total: int) -> list[int]:
    """Random composition with every part in [5, 9]."""
    values = [5] * count
    remaining = total - 5 * count
    order = list(range(count))
    while remaining:
        rng.shuffle(order)
        progressed = False
        for index in order:
            if values[index] < 9 and remaining:
                values[index] += 1
                remaining -= 1
                progressed = True
        if not progressed:
            raise ValueError("bounded width composition is impossible")
    return values


def _make_widths(rng: random.Random, runs: list[int]) -> tuple[list[int], list[int]]:
    signs: list[int] = []
    run_indices: list[list[int]] = []
    cursor = 0
    for run_number, count in enumerate(runs):
        indices = list(range(cursor, cursor + count))
        run_indices.append(indices)
        signs.extend(([1] if run_number % 2 == 0 else [-1]) * count)
        cursor += count

    major = run_indices[runs.index(4)]
    minor = run_indices[runs.index(3)]
    widths = [0] * SEGMENTS
    for index in major:
        widths[index] = rng.choice((8, 9))
    for index in minor:
        widths[index] = rng.choice((5, 6))
    free = [index for index in range(SEGMENTS) if not widths[index]]
    remainder = TOTAL_WIDTH - sum(widths)
    fill = _bounded_fill(rng, len(free), remainder)
    for index, value in zip(free, fill, strict=True):
        widths[index] = value
    return widths, signs


def _run_spans(widths: Iterable[int], signs: Iterable[int]) -> dict[int, int]:
    best = {1: 0, -1: 0}
    active_sign = 0
    active_span = 0
    for width, sign in zip(widths, signs, strict=True):
        if sign not in (-1, 1) or int(width) <= 0:
            raise ValueError("invalid signed segment")
        if sign == active_sign:
            active_span += int(width)
        else:
            if active_sign:
                best[active_sign] = max(best[active_sign], active_span)
            active_sign = int(sign)
            active_span = int(width)
    if active_sign:
        best[active_sign] = max(best[active_sign], active_span)
    return best


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    desired = "RISE" if int(seed) % 2 == 0 else "FALL"
    for _ in range(200):
        runs = _rotated_runs(rng)
        widths, signs = _make_widths(rng, runs)
        spans = _run_spans(widths, signs)
        if spans[1] - spans[-1] >= 10:
            break
    else:
        raise RuntimeError("could not sample a separated run-span scene")

    if desired == "FALL":
        signs = [-value for value in signs]
    step = rng.choice((3, 4))
    vertical_steps = [step] * SEGMENTS
    offsets = [0]
    for sign, amount in zip(signs, vertical_steps, strict=True):
        offsets.append(offsets[-1] - sign * amount)

    center_x = rng.choice((160, 224, 288, 352))
    center_y = rng.choice((160, 224, 288, 352))
    origin_x = center_x - TOTAL_WIDTH // 2
    origin_y = center_y - round((min(offsets) + max(offsets)) / 2)
    return {
        "origin_x": origin_x,
        "origin_y": origin_y,
        "widths": widths,
        "signs": signs,
        "vertical_steps": vertical_steps,
        "trace_color": rng.choice(("lagoon", "deep_teal", "aqua")),
        "decor_variant": rng.randrange(4),
    }


def _points(scene: dict) -> list[tuple[int, int]]:
    x = int(scene["origin_x"])
    y = int(scene["origin_y"])
    points = [(x, y)]
    for width, sign, vertical in zip(
        scene["widths"], scene["signs"], scene["vertical_steps"], strict=True
    ):
        x += int(width)
        y -= int(sign) * int(vertical)
        points.append((x, y))
    return points


def _draw_paper_context(draw: ImageDraw.ImageDraw, points: list[tuple[int, int]], variant: int) -> None:
    xs = [point[0] for point in points]
    ys = [point[1] for point in points]
    left, right = min(xs) - 13, max(xs) + 13
    top, bottom = min(ys) - 12, max(ys) + 12
    draw.rounded_rectangle((left, top, right, bottom), radius=7, outline=PALE, width=1)
    if variant % 2 == 0:
        draw.line((left + 8, bottom + 7, right - 8, bottom + 7), fill=PALE_2, width=1)
    else:
        draw.line((left - 7, top + 6, left - 7, bottom - 6), fill=PALE_2, width=1)
    if variant >= 2:
        draw.ellipse((36, 43, 41, 48), outline=PALE, width=1)
        draw.line((48, 45, 78, 45), fill=PALE_2, width=2)


def _draw_trace(image: Image.Image, points: list[tuple[int, int]], ink: tuple[int, int, int]) -> None:
    pixels = image.load()
    for (x0, y0), (x1, y1) in zip(points, points[1:]):
        width = x1 - x0
        if width <= 0:
            raise ValueError("trace must move strictly left to right")
        for x in range(x0, x1 + 1):
            y = round(y0 + (y1 - y0) * (x - x0) / width)
            for yy in range(y - TRACE_RADIUS, y + TRACE_RADIUS + 1):
                pixels[x, yy] = ink


def render(scene: dict) -> Image.Image:
    points = _points(scene)
    image = Image.new("RGB", (CANVAS, CANVAS), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_paper_context(draw, points, int(scene["decor_variant"]))
    ink = _INKS[str(scene["trace_color"])]
    _draw_trace(image, points, ink)
    return image


def analytic_gold(scene: dict) -> str:
    spans = _run_spans(scene["widths"], scene["signs"])
    if spans[1] == spans[-1]:
        raise ValueError("run-span tie has no analytic answer")
    return "RISE" if spans[1] > spans[-1] else "FALL"


def margin(scene: dict) -> float:
    spans = _run_spans(scene["widths"], scene["signs"])
    return float(abs(spans[1] - spans[-1]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    token = str(scene["trace_color"])
    twin = copy.deepcopy(scene)
    twin["trace_color"] = token[:-6] if token.endswith("_alias") else token + "_alias"
    return [("equivalent_palette_token", twin)]
