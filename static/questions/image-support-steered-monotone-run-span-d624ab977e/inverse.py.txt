"""Independent final-PNG inverse for the monotone-run-span decision."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _ink_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[..., 0], array[..., 1], array[..., 2]
    mask = (
        (red < 70)
        & (green > 105)
        & (blue > 115)
        & ((green - red) > 55)
        & ((blue - red) > 65)
        & (np.abs(green - blue) < 60)
    )
    if int(mask.sum()) < 240:
        raise ValueError("teal trace is absent or materially erased")
    return mask


def _component_count(mask: np.ndarray) -> int:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    count = 0
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        count += 1
        queue = deque([(y, x)])
        seen[y, x] = True
        while queue:
            cy, cx = queue.popleft()
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
    return count


def _centerline(mask: np.ndarray) -> tuple[np.ndarray, int]:
    if _component_count(mask) != 1:
        raise ValueError("trace is broken into multiple components")
    ys, xs = np.nonzero(mask)
    left, right = int(xs.min()), int(xs.max())
    width = right - left + 1
    if not 70 <= width <= 160:
        raise ValueError("trace horizontal extent is incomplete or implausible")
    if int(ys.max() - ys.min() + 1) > 70:
        raise ValueError("trace is not a local mini-trace")
    centers = []
    for x in range(left, right + 1):
        column = np.flatnonzero(mask[:, x])
        if len(column) < 3:
            raise ValueError("trace has an erased or too-thin column")
        centers.append(float(np.median(column)))
    return np.asarray(centers, dtype=np.float64), left


def _direction_runs(center: np.ndarray) -> list[tuple[str, float]]:
    plateaus: list[tuple[float, float]] = []
    start = 0
    for index in range(1, len(center) + 1):
        if index == len(center) or center[index] != center[start]:
            plateaus.append(((start + index - 1) / 2.0, float(center[start])))
            start = index
    if len(plateaus) < 8:
        raise ValueError("too few recoverable trace changes")

    edge_directions: list[str] = []
    for previous, current in zip(plateaus, plateaus[1:]):
        delta = current[1] - previous[1]
        if delta == 0:
            continue
        edge_directions.append("RISE" if delta < 0 else "FALL")
    if not edge_directions:
        raise ValueError("trace has no directional runs")

    runs: list[tuple[str, float]] = []
    active = edge_directions[0]
    boundary = plateaus[0][0]
    for edge_index in range(1, len(edge_directions)):
        direction = edge_directions[edge_index]
        if direction != active:
            turn_x = plateaus[edge_index][0]
            runs.append((active, turn_x - boundary))
            active = direction
            boundary = turn_x
    runs.append((active, plateaus[-1][0] - boundary))
    if not 4 <= len(runs) <= 8 or {kind for kind, _ in runs} != {"RISE", "FALL"}:
        raise ValueError("trace reversal structure is unresolved")
    return runs


def decision_from_image(image: Image.Image) -> str:
    """Return RISE or FALL using only visible pixels, otherwise abstain by exception."""
    mask = _ink_mask(image)
    center, _ = _centerline(mask)
    runs = _direction_runs(center)
    longest = {
        kind: max(span for candidate, span in runs if candidate == kind)
        for kind in ("RISE", "FALL")
    }
    gap = abs(longest["RISE"] - longest["FALL"])
    if gap < 5.0:
        raise ValueError("pixel runner-up margin is ambiguous")
    return "RISE" if longest["RISE"] > longest["FALL"] else "FALL"
