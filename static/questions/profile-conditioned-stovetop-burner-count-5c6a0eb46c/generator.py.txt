"""Deterministic Pillow renderer and analytic arm."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH, HEIGHT, SCALE = 512, 420, 3
BLUE = [30, 112, 210]


def sample_scene(seed: int) -> dict[str, Any]:
    # Consecutive sample seeds cycle through all four answers. Other styling is
    # generated independently after a fixed xor, so canonicality has no style bit.
    count = 3 + (int(seed) % 4)
    rng = random.Random(int(seed) ^ 0x5A17C9E3)
    x0 = rng.randint(42, 62)
    y0 = rng.randint(38, 54)
    x1 = x0 + rng.randint(388, 408)
    y1 = y0 + rng.randint(302, 320)
    body_box = [x0, y0, x1, y1]

    # Six plausible locations form two staggerable rows. A random subset creates
    # controls and counterfactuals without changing the surrounding appliance.
    usable_top = y0 + 50
    usable_bottom = y1 - 80
    xs = [x0 + 78, (x0 + x1) // 2, x1 - 78]
    ys = [usable_top + 38, usable_bottom - 38]
    cells = [(x, y) for y in ys for x in xs]
    rng.shuffle(cells)
    chosen = sorted(cells[:count], key=lambda p: (p[1], p[0]))
    burners = []
    for cx, cy in chosen:
        burners.append([
            cx + rng.randint(-5, 5),
            cy + rng.randint(-4, 4),
            rng.randint(31, 38),
            rng.randint(6, 9),
        ])

    knob_y = y1 - 27
    knobs = [[x0 + 95 + i * ((x1 - x0 - 190) // 3), knob_y, rng.randint(-25, 25)] for i in range(4)]
    body_tones = ([55, 61, 70], [68, 72, 78], [47, 54, 61], [76, 78, 82])
    bg_tones = ([244, 239, 225], [235, 241, 242], [246, 235, 230])
    return {
        "burners": burners,
        "body_box": body_box,
        "knobs": knobs,
        "palette": {
            "burner": BLUE,
            "body": list(rng.choice(body_tones)),
            "rim": [25, 28, 33],
            "panel": [105, 108, 112],
            "knob": [224, 174, 65],
            "highlight": [224, 228, 232],
        },
        "background": list(rng.choice(bg_tones)),
    }


def _clearance(scene: dict[str, Any]) -> float:
    burners = scene["burners"]
    x0, y0, x1, y1 = scene["body_box"]
    values: list[float] = []
    for i, (cx, cy, radius, _width) in enumerate(burners):
        values.extend([cx - radius - x0, x1 - cx - radius, cy - radius - y0, y1 - cy - radius])
        for ox, oy, other_radius, _ in burners[i + 1:]:
            values.append(math.hypot(cx - ox, cy - oy) - radius - other_radius)
    return min(values)


def render(scene: dict[str, Any]) -> Image.Image:
    palette = scene["palette"]
    canvas = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), tuple(scene["background"]))
    draw = ImageDraw.Draw(canvas)

    def box(coords: list[int] | tuple[int, ...], **kwargs: Any) -> None:
        draw.rounded_rectangle(tuple(v * SCALE for v in coords), radius=16 * SCALE, **kwargs)

    x0, y0, x1, y1 = scene["body_box"]
    box([x0 - 5, y0 - 5, x1 + 5, y1 + 5], fill=tuple(palette["rim"]))
    box([x0, y0, x1, y1], fill=tuple(palette["body"]))
    # A control rail and four fixed gold knobs make the silhouette unmistakably
    # appliance-like but never add blue target pixels.
    draw.rounded_rectangle(
        (x0 * SCALE, (y1 - 54) * SCALE, x1 * SCALE, y1 * SCALE),
        radius=12 * SCALE,
        fill=tuple(palette["panel"]),
    )
    for kx, ky, angle in scene["knobs"]:
        rr = 13
        draw.ellipse(((kx - rr) * SCALE, (ky - rr) * SCALE, (kx + rr) * SCALE, (ky + rr) * SCALE), fill=tuple(palette["knob"]), outline=tuple(palette["rim"]), width=2 * SCALE)
        rad = math.radians(angle - 90)
        draw.line((kx * SCALE, ky * SCALE, (kx + 9 * math.cos(rad)) * SCALE, (ky + 9 * math.sin(rad)) * SCALE), fill=tuple(palette["rim"]), width=2 * SCALE)

    for cx, cy, radius, width in scene["burners"]:
        draw.ellipse(((cx - radius) * SCALE, (cy - radius) * SCALE, (cx + radius) * SCALE, (cy + radius) * SCALE), fill=tuple(palette["burner"]))
        inner = radius - width
        draw.ellipse(((cx - inner) * SCALE, (cy - inner) * SCALE, (cx + inner) * SCALE, (cy + inner) * SCALE), fill=tuple(palette["body"]))
        # Small neutral center cap reinforces burner structure without affecting
        # the blue ring component count.
        cap = 5
        draw.ellipse(((cx - cap) * SCALE, (cy - cap) * SCALE, (cx + cap) * SCALE, (cy + cap) * SCALE), fill=tuple(palette["highlight"]))

    return canvas.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(len(scene["burners"]))


def margin(scene: dict[str, Any]) -> float:
    return round(_clearance(scene), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return _clearance(scene) < 8.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    twin = dict(scene)
    twin["burners"] = list(reversed(scene["burners"]))
    return [("reverse_burner_list", twin)]
