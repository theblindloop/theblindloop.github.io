"""Independent image-only inverse arm.

This module deliberately duplicates the public blue-color criterion and uses no
renderer state, coordinates, filenames, metadata, or latent scene fields.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

MIN_COMPONENT_PIXELS = 650


def _components(mask: np.ndarray) -> list[int]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    sizes: list[int] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        size = 0
        while queue:
            yy, xx = queue.popleft()
            size += 1
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        sizes.append(size)
    return sizes


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    r, g, b = arr[..., 0], arr[..., 1], arr[..., 2]
    # The named blue ink is saturated: this generous chromatic rule survives
    # antialiasing while excluding every neutral appliance/background pixel.
    blue = (b - r > 70) & (b - g > 35) & (b > 130) & (r < 100)
    components = [size for size in _components(blue) if size >= MIN_COMPONENT_PIXELS]
    if not 3 <= len(components) <= 6:
        raise ValueError(f"ambiguous or erased burner evidence: {len(components)} rings")
    if min(components) < MIN_COMPONENT_PIXELS:
        raise ValueError("undersized burner evidence")
    return str(len(components))
