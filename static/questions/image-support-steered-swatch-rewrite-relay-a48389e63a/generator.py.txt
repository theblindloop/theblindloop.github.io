"""Renderer for the swatch_rewrite_relay world.

A small notice board is painted somewhere on the canvas.  It holds six separated
tiles arranged in two horizontal bands of three, with a wide band of empty
background between the bands.  The leftmost tile of the upper band is the badge:
a single coloured square inside a dark rectangular ring, and its colour is the
value the relay starts from.  The other five tiles are rewrite cards.  A card is
two coloured squares with a background gap between them: the left square is the
card's "from" colour and the right square is its "to" colour.

Cards act in reading order (upper band left to right, then lower band left to
right).  A card acts only when the value carried at that moment equals its
"from" colour, and then the value becomes its "to" colour; every other card is
inert at that moment.  The decision is the colour name of the value after the
last card.

The decisive evidence is therefore a handful of small separated swatches spread
over one medium notice board: the ringed badge and the guard/result squares of
the cards.  Grey marks outside the board are inert decoration.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "swatch_rewrite_relay-renderer-0.1.0"

CANVAS = 448
BACKGROUND = (246, 246, 242)
INK = (28, 28, 34)
DECOR = (201, 201, 196)

CARDS = 5
BANDS = 2
TILES_PER_BAND = 3

SWATCH_COLOURS: dict[str, tuple[int, int, int]] = {
    "red": (198, 42, 42),
    "green": (32, 138, 66),
    "blue": (40, 84, 200),
    "purple": (142, 48, 168),
    "orange": (226, 132, 22),
}
SWATCH_NAMES = ("red", "green", "blue", "purple", "orange")

QUARANTINE_MARGIN = 3.0


# ---------------------------------------------------------------- geometry ---
def _tile_slots(scene: dict) -> list[tuple[int, int]]:
    """Top-left corner of every tile, in reading order (band major, then x)."""
    x0 = scene["x0"]
    y0 = scene["y0"]
    slots: list[tuple[int, int]] = []
    for band in range(BANDS):
        x = x0
        y = y0 + band * scene["band_gap"]
        for column in range(TILES_PER_BAND):
            slots.append((x, y))
            x += scene["tile_width"] + scene["tile_gaps"][band][column]
    return slots


def _swatch_boxes(scene: dict) -> list[tuple[int, int, int, int, str]]:
    """Every painted swatch as (left, top, right, bottom, colour name)."""
    size = scene["swatch"]
    gap = scene["hgap"]
    boxes: list[tuple[int, int, int, int, str]] = []
    slots = _tile_slots(scene)
    bx, by = slots[0]
    boxes.append((bx, by, bx + size, by + size, scene["start"]))
    for index, (left, top) in enumerate(slots[1:]):
        card = scene["cards"][index]
        boxes.append((left, top, left + size, top + size, card[0]))
        second = left + size + gap
        boxes.append((second, top, second + size, top + size, card[1]))
    return boxes


def _ring_box(scene: dict) -> tuple[int, int, int, int]:
    size = scene["swatch"]
    gap = scene["ring_gap"]
    x, y = _tile_slots(scene)[0]
    return (x - gap - 2, y - gap - 2, x + size + gap + 1, y + size + gap + 1)


def _board_bbox(scene: dict) -> tuple[int, int, int, int]:
    boxes = _swatch_boxes(scene)
    ring = _ring_box(scene)
    x_lo = min([box[0] for box in boxes] + [ring[0]])
    y_lo = min([box[1] for box in boxes] + [ring[1]])
    x_hi = max([box[2] for box in boxes] + [ring[2]])
    y_hi = max([box[3] for box in boxes] + [ring[3]])
    return x_lo, y_lo, x_hi, y_hi


# --------------------------------------------------------------- semantics ---
def _trace(scene: dict) -> list[str]:
    """Carried colour before each card, plus the final carried colour."""
    value = scene["start"]
    states = [value]
    for source, target in scene["cards"]:
        if value == source:
            value = target
        states.append(value)
    return states


def _fired(scene: dict) -> list[bool]:
    states = _trace(scene)
    return [states[index] != states[index + 1] for index in range(CARDS)]


# ---------------------------------------------------------------- sampling ---
def _sample_once(rng: random.Random) -> dict | None:
    tier = rng.choice([3, 4, 5, 7, 9, 11])
    swatch = rng.randint(12, 16)
    hgap = tier + rng.randint(1, 3)
    ring_gap = max(3, tier - 1)
    band_gap = rng.randint(184, 194)
    tile_width = 2 * swatch + hgap
    tile_gaps = [[rng.randint(26, 34) for _ in range(TILES_PER_BAND - 1)] + [0] for _ in range(BANDS)]

    start = rng.choice(SWATCH_NAMES)
    cards: list[tuple[str, str]] = []
    for _ in range(CARDS):
        source = rng.choice(SWATCH_NAMES)
        target = rng.choice([name for name in SWATCH_NAMES if name != source])
        cards.append((source, target))

    stub = {"start": start, "cards": cards}
    states = _trace(stub)
    fired = _fired(stub)
    final = states[-1]

    if final == start:
        return None
    if sum(fired) < 2:
        return None
    if fired[CARDS - 1]:
        return None
    if not any(fired[:2]) or not any(fired[2:4]):
        return None
    if final == cards[0][1] or final == cards[CARDS - 1][1]:
        return None
    visited = set(states)
    if not any(not fired[index] and cards[index][0] in visited for index in range(CARDS)):
        return None

    counts: dict[str, int] = {}
    for name in [start] + [name for card in cards for name in card]:
        counts[name] = counts.get(name, 0) + 1
    answer_count = counts[final]
    if sum(1 for value in counts.values() if value > answer_count) == 0:
        return None
    if sum(1 for value in counts.values() if value < answer_count) == 0:
        return None

    width = 3 * tile_width + sum(tile_gaps[0][:-1])
    width = max(width, 3 * tile_width + sum(tile_gaps[1][:-1]))
    height = band_gap + swatch
    pad = ring_gap + 16
    if width + 2 * pad + 12 > CANVAS or height + 2 * pad + 12 > CANVAS:
        return None
    x0 = rng.randint(pad, CANVAS - width - pad)
    y0 = rng.randint(pad, CANVAS - height - pad)

    scene = {
        "swatch": swatch,
        "hgap": hgap,
        "ring_gap": ring_gap,
        "band_gap": band_gap,
        "tile_width": tile_width,
        "tile_gaps": tile_gaps,
        "margin_tier": tier,
        "start": start,
        "cards": [list(card) for card in cards],
        "x0": x0,
        "y0": y0,
        "decor": [],
    }
    scene["decor"] = _sample_decor(rng, scene)
    if margin(scene) < QUARANTINE_MARGIN:
        return None
    return scene


def _sample_decor(rng: random.Random, scene: dict) -> list[dict]:
    """Inert grey marks, kept well clear of the notice board."""
    x_lo, y_lo, x_hi, y_hi = _board_bbox(scene)
    marks: list[dict] = []
    attempts = 0
    wanted = rng.randint(6, 10)
    while len(marks) < wanted and attempts < 400:
        attempts += 1
        kind = rng.choice(["dot", "box", "bar"])
        size = rng.randint(4, 10)
        length = rng.randint(9, 24)
        cx = rng.randint(10, CANVAS - 10)
        cy = rng.randint(10, CANVAS - 10)
        w = length if kind == "bar" else size
        h = size if kind != "bar" else rng.randint(3, 5)
        left, top = cx - w // 2, cy - h // 2
        if left < 6 or top < 6 or left + w > CANVAS - 6 or top + h > CANVAS - 6:
            continue
        if left - 18 < x_hi and x_lo - 18 < left + w and top - 18 < y_hi and y_lo - 18 < top + h:
            continue
        marks.append({"kind": kind, "left": left, "top": top, "w": w, "h": h})
    return marks


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 6151 + 29)
    for _ in range(4000):
        scene = _sample_once(rng)
        if scene is not None:
            return scene
    raise RuntimeError(f"could not sample a legible relay scene for seed {seed}")


# ------------------------------------------------------------------ answers ---
def analytic_gold(scene: dict) -> str:
    """Colour name carried by the relay after the last card has been read."""
    return _trace({"start": scene["start"], "cards": [tuple(c) for c in scene["cards"]]})[-1]


def margin(scene: dict) -> float:
    """Smallest visible clearance the replay depends on, in pixels."""
    clearances = [
        float(scene["hgap"]),
        float(scene["ring_gap"]),
        float(min(gap for band in scene["tile_gaps"] for gap in band if gap)),
        float(scene["band_gap"] - scene["swatch"]),
    ]
    return float(min(clearances))


def is_quarantined(scene: dict) -> bool:
    """True when a decisive clearance is too small to resolve on the raster."""
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist: every sampled field is painted (see candidate.json)."""
    return []


# ------------------------------------------------------------------ drawing ---
def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (CANVAS, CANVAS), BACKGROUND)
    draw = ImageDraw.Draw(image)

    for mark in scene["decor"]:
        box = [mark["left"], mark["top"], mark["left"] + mark["w"], mark["top"] + mark["h"]]
        if mark["kind"] == "dot":
            draw.ellipse(box, fill=DECOR)
        else:
            draw.rectangle(box, fill=DECOR)

    ring = _ring_box(scene)
    draw.rectangle(list(ring), outline=INK, width=2)

    for left, top, right, bottom, name in _swatch_boxes(scene):
        draw.rectangle([left, top, right - 1, bottom - 1], fill=SWATCH_COLOURS[name])
    return image
