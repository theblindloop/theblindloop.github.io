"""Cloud elongation axis world.

A large cloud of identical grey dots is drawn on a plain background together
with four straight coloured bars parked in the four border zones.  Every dot is
the same size and colour and carries no individual information; the only signal
in the picture is the *aggregate* shape of the cloud, i.e. the direction along
which the whole scatter is stretched furthest.  The decision selects the bar
whose drawn direction is closest to that global axis.

The latent scene is sampled so that the dot cloud's sample second-moment matrix
is exactly the intended anisotropic ellipse (a whitening step removes sampling
noise), which makes the visible long axis and the analytic long axis identical.
"""

from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image, ImageDraw

SIZE = 640
CENTER = (320.0, 320.0)
CLOUD_RADIUS_LIMIT = 165.0
DOT_RADIUS = 4
BAR_WIDTH = 9
BAR_HALF_LENGTH = 58.0
QUARANTINE_MARGIN_DEG = 7.0

BACKGROUND = (247, 247, 243)
DOT_COLOR = (105, 105, 105)
BAR_COLORS = {
    "red": (214, 32, 32),
    "green": (26, 150, 60),
    "blue": (36, 72, 214),
    "orange": (240, 140, 10),
}
COLOR_ORDER = ("red", "green", "blue", "orange")
ZONE_CENTERS = ((320.0, 78.0), (320.0, 562.0), (78.0, 320.0), (562.0, 320.0))


def _angle_gap(first: float, second: float) -> float:
    """Undirected angular distance in degrees, in [0, 90]."""
    delta = abs(first - second) % 180.0
    return min(delta, 180.0 - delta)


def _principal_angle(points: np.ndarray) -> float:
    """Direction of the largest second moment of a point set, in [0, 180)."""
    centered = points - points.mean(axis=0)
    cov = centered.T @ centered / max(len(centered) - 1, 1)
    values, vectors = np.linalg.eigh(cov)
    major = vectors[:, int(np.argmax(values))]
    return math.degrees(math.atan2(float(major[1]), float(major[0]))) % 180.0


def _cloud_angle(scene: dict) -> float:
    return _principal_angle(np.asarray(scene["dots"], dtype=float))


def _bar_angles(scene: dict) -> list[tuple[str, float]]:
    angles = []
    for bar in scene["bars"]:
        dx = bar["x2"] - bar["x1"]
        dy = bar["y2"] - bar["y1"]
        angles.append((bar["color"], math.degrees(math.atan2(dy, dx)) % 180.0))
    return angles


def _ranked_gaps(scene: dict) -> list[tuple[float, str]]:
    axis = _cloud_angle(scene)
    return sorted((_angle_gap(angle, axis), color) for color, angle in _bar_angles(scene))


def analytic_gold(scene: dict) -> str:
    """Colour of the bar whose direction is closest to the cloud's long axis."""
    return _ranked_gaps(scene)[0][1]


def margin(scene: dict) -> float:
    """Degrees separating the winning bar from the best runner-up bar."""
    ranked = _ranked_gaps(scene)
    return float(ranked[1][0] - ranked[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN_DEG


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist: every labelling field is drawn (see candidate.json)."""
    return []


def _sample_cloud(rng: random.Random, theta: float, ratio: float, sigma: float, count: int):
    generator = np.random.default_rng(rng.getrandbits(63))
    raw = generator.standard_normal((count, 2))
    raw = raw - raw.mean(axis=0)
    cov = raw.T @ raw / (count - 1)
    values, vectors = np.linalg.eigh(cov)
    whitener = vectors @ np.diag(1.0 / np.sqrt(np.maximum(values, 1e-12))) @ vectors.T
    unit = raw @ whitener
    scale = np.diag([sigma, sigma / ratio])
    radians = math.radians(theta)
    rotation = np.array(
        [
            [math.cos(radians), -math.sin(radians)],
            [math.sin(radians), math.cos(radians)],
        ]
    )
    points = unit @ scale @ rotation.T
    reach = float(np.max(np.hypot(points[:, 0], points[:, 1])))
    if reach > CLOUD_RADIUS_LIMIT:
        points = points * (CLOUD_RADIUS_LIMIT / reach)
    points = points + np.asarray(CENTER)
    return [[round(float(x), 3), round(float(y), 3)] for x, y in points]


def sample_scene(seed: int) -> dict:
    rng = random.Random(0x9E3779B9 ^ (seed * 2654435761))
    theta = rng.uniform(0.0, 180.0)
    ratio = rng.uniform(3.0, 4.2)
    sigma = rng.uniform(44.0, 54.0)
    count = rng.randint(130, 175)
    dots = _sample_cloud(rng, theta, ratio, sigma, count)

    if rng.random() < 0.25:
        target = rng.uniform(2.2, 6.0)
    else:
        target = rng.uniform(9.5, 52.0)
    winner_gap = rng.uniform(0.0, 4.0)
    gaps = [winner_gap, winner_gap + target]
    while len(gaps) < 4:
        gaps.append(min(88.0, winner_gap + target + rng.uniform(4.0, 34.0)))

    # The winning colour cycles with the seed, so answer identity is decorrelated
    # from every drawn nuisance cue and both prompt families stay balanced.
    winner = COLOR_ORDER[seed % 4]
    others = [color for color in COLOR_ORDER if color != winner]
    rng.shuffle(others)
    ordered_colors = [winner, *others]

    zones = list(ZONE_CENTERS)
    rng.shuffle(zones)
    bars = []
    for color, gap, (zx, zy) in zip(ordered_colors, gaps, zones):
        angle = (theta + gap * rng.choice((-1.0, 1.0))) % 180.0
        radians = math.radians(angle)
        if zy < 160.0 or zy > 480.0:
            zx += rng.uniform(-60.0, 60.0)
        else:
            zy += rng.uniform(-60.0, 60.0)
        dx = BAR_HALF_LENGTH * math.cos(radians)
        dy = BAR_HALF_LENGTH * math.sin(radians)
        bars.append(
            {
                "color": color,
                "x1": round(zx - dx, 3),
                "y1": round(zy - dy, 3),
                "x2": round(zx + dx, 3),
                "y2": round(zy + dy, 3),
            }
        )
    rng.shuffle(bars)
    return {
        "size": SIZE,
        "bg": list(BACKGROUND),
        "dot_radius": DOT_RADIUS,
        "bar_width": BAR_WIDTH,
        "dots": dots,
        "bars": bars,
    }


def render(scene: dict) -> Image.Image:
    size = int(scene["size"])
    image = Image.new("RGB", (size, size), tuple(scene["bg"]))
    draw = ImageDraw.Draw(image)
    radius = float(scene["dot_radius"])
    for x, y in scene["dots"]:
        draw.ellipse([x - radius, y - radius, x + radius, y + radius], fill=DOT_COLOR)
    for bar in scene["bars"]:
        draw.line(
            [bar["x1"], bar["y1"], bar["x2"], bar["y2"]],
            fill=BAR_COLORS[bar["color"]],
            width=int(scene["bar_width"]),
        )
    return image
