"""Image-only decision path for the cloud elongation axis world.

Receives a rendered RGB image and nothing else.  It rebuilds the declared global
evidence directly from pixels: the second-moment (inertia) tensor of the grey
dot field gives the cloud's long axis, the second-moment tensor of each coloured
bar's pixels gives that bar's direction, and the decision is the colour with the
smallest undirected angular distance.  Individual dot identities are never used;
only the aggregate orientation field is.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

REFERENCES = {
    "background": (247, 247, 243),
    "cloud": (105, 105, 105),
    "red": (214, 32, 32),
    "green": (26, 150, 60),
    "blue": (36, 72, 214),
    "orange": (240, 140, 10),
}
COLOR_KEYS = ("red", "green", "blue", "orange")
MATCH_TOLERANCE = 60.0
MIN_CLOUD_PIXELS = 800
MIN_BAR_PIXELS = 250
MIN_ANISOTROPY = 1.25
ABSTAIN = "abstain"


def _principal_angle(mask: np.ndarray) -> tuple[float, float]:
    """Return (orientation in [0,180), axis ratio) of a boolean pixel mask."""
    ys, xs = np.nonzero(mask)
    coords = np.stack([xs.astype(float), ys.astype(float)], axis=1)
    coords = coords - coords.mean(axis=0)
    cov = coords.T @ coords / max(len(coords) - 1, 1)
    values, vectors = np.linalg.eigh(cov)
    major = vectors[:, int(np.argmax(values))]
    angle = math.degrees(math.atan2(float(major[1]), float(major[0]))) % 180.0
    small = max(float(np.min(values)), 1e-9)
    ratio = math.sqrt(float(np.max(values)) / small)
    return angle, ratio


def _angle_gap(first: float, second: float) -> float:
    delta = abs(first - second) % 180.0
    return min(delta, 180.0 - delta)


def _label_masks(image: Image.Image) -> dict:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    names = list(REFERENCES)
    palette = np.asarray([REFERENCES[name] for name in names], dtype=np.int16)
    distances = np.linalg.norm(
        pixels[:, :, None, :].astype(np.float32) - palette[None, None, :, :].astype(np.float32),
        axis=3,
    )
    nearest = np.argmin(distances, axis=2)
    close = np.min(distances, axis=2) <= MATCH_TOLERANCE
    return {name: (nearest == index) & close for index, name in enumerate(names)}


def decision_from_image(image):
    masks = _label_masks(image)
    cloud = masks["cloud"]
    if int(cloud.sum()) < MIN_CLOUD_PIXELS:
        return ABSTAIN
    cloud_angle, cloud_ratio = _principal_angle(cloud)
    if cloud_ratio < MIN_ANISOTROPY:
        return ABSTAIN

    gaps = []
    for key in COLOR_KEYS:
        mask = masks[key]
        if int(mask.sum()) < MIN_BAR_PIXELS:
            return ABSTAIN
        bar_angle, bar_ratio = _principal_angle(mask)
        if bar_ratio < 2.0:
            return ABSTAIN
        gaps.append((_angle_gap(bar_angle, cloud_angle), key))
    gaps.sort()
    return gaps[0][1]
