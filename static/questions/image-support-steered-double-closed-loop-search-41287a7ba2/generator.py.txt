"""Deterministic scene sampler and raster renderer for paired contour closure search."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw


COLORS = ("red", "blue", "green", "purple")
PALETTE = {
    "red": [216, 54, 72],
    "blue": [42, 104, 196],
    "green": [28, 146, 90],
    "purple": [132, 72, 178],
}
BACKGROUND = [250, 249, 246]
CANVAS = 640
CELL = CANVAS // 8
STROKE = 5
QUARANTINE_APERTURE_PX = 6.0

# Every slot is two grid cells apart. All four colors therefore have the same
# coarse pair span, while shuffling colors over slots randomizes target location.
PAIR_LAYOUTS = (
    (
        ((1, 1), (1, 3)),
        ((1, 5), (3, 5)),
        ((4, 1), (6, 1)),
        ((6, 4), (6, 6)),
    ),
    (
        ((1, 2), (3, 2)),
        ((1, 5), (1, 7)),
        ((4, 6), (6, 6)),
        ((6, 1), (6, 3)),
    ),
    (
        ((1, 1), (3, 1)),
        ((1, 4), (1, 6)),
        ((4, 6), (6, 6)),
        ((6, 2), (6, 4)),
    ),
    (
        ((1, 1), (1, 3)),
        ((1, 6), (3, 6)),
        ((4, 1), (6, 1)),
        ((6, 4), (6, 6)),
    ),
)


def _center(cell: tuple[int, int]) -> tuple[int, int]:
    row, column = cell
    return column * CELL + CELL // 2, row * CELL + CELL // 2


def _open_aperture(mark: dict) -> float:
    if mark["closed"]:
        return float("inf")
    chord = 2.0 * mark["radius"] * math.sin(math.radians(mark["gap_degrees"]) / 2.0)
    return chord - mark["stroke_width"]


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; answer identity cycles deterministically by seed."""

    rng = random.Random(int(seed))
    winner = COLORS[int(seed) % len(COLORS)]
    layout = list(PAIR_LAYOUTS[rng.randrange(len(PAIR_LAYOUTS))])
    rng.shuffle(layout)
    color_order = list(COLORS)
    rng.shuffle(color_order)

    marks: list[dict] = []
    for color, pair in zip(color_order, layout, strict=True):
        budget = rng.choice((25, 26, 27))
        first_radius = rng.choice((12, 13, 14))
        gap_degrees = rng.choice((52, 60, 68, 76, 84))
        gap_fraction = gap_degrees / 360.0
        if color == winner:
            radii = (first_radius, budget - first_radius)
            closed_flags = (True, True)
        else:
            open_index = rng.randrange(2)
            closed_radius = first_radius
            open_radius = int(round((budget - closed_radius) / (1.0 - gap_fraction)))
            open_radius = max(12, min(16, open_radius))
            radii = (
                open_radius if open_index == 0 else closed_radius,
                open_radius if open_index == 1 else closed_radius,
            )
            closed_flags = (open_index != 0, open_index != 1)
        for cell, radius, closed in zip(pair, radii, closed_flags, strict=True):
            cx, cy = _center(cell)
            marks.append(
                {
                    "color": color,
                    "cx": cx,
                    "cy": cy,
                    "radius": int(radius),
                    "closed": bool(closed),
                    "gap_start": rng.randrange(0, 360, 15),
                    "gap_degrees": int(gap_degrees),
                    "stroke_width": STROKE,
                }
            )
    rng.shuffle(marks)
    return {
        "width": CANVAS,
        "height": CANVAS,
        "background": list(BACKGROUND),
        "palette": deepcopy(PALETTE),
        "marks": marks,
    }


def render(scene: dict) -> Image.Image:
    """Render only the declared scene fields to a crisp RGB PNG."""

    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(scene["background"])
    palette = scene["palette"]
    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)
    for mark in scene["marks"]:
        radius = int(mark["radius"])
        cx, cy = int(mark["cx"]), int(mark["cy"])
        box = (cx - radius, cy - radius, cx + radius, cy + radius)
        ink = tuple(palette[mark["color"]])
        stroke = int(mark["stroke_width"])
        if mark["closed"]:
            draw.ellipse(box, outline=ink, width=stroke)
        else:
            gap_start = float(mark["gap_start"])
            gap = float(mark["gap_degrees"])
            draw.arc(box, start=gap_start + gap, end=gap_start + 360.0, fill=ink, width=stroke)
    return image


def _scores(scene: dict) -> dict[str, int]:
    scores = {color: 0 for color in COLORS}
    for mark in scene["marks"]:
        scores[mark["color"]] += int(bool(mark["closed"]))
    return scores


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    best = max(scores.values())
    winners = [color for color, value in scores.items() if value == best]
    if best != 2 or len(winners) != 1:
        raise ValueError(f"scene has no unique two-closed-contour winner: {scores}")
    return winners[0]


def margin(scene: dict) -> float:
    values = sorted(_scores(scene).values(), reverse=True)
    return float(values[0] - values[1])


def is_quarantined(scene: dict) -> bool:
    apertures = [_open_aperture(mark) for mark in scene["marks"] if not mark["closed"]]
    return not apertures or min(apertures) < QUARANTINE_APERTURE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["marks"] = list(reversed(twin["marks"]))
    return [("reverse_unordered_mark_serialization", twin)]
