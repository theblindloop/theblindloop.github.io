"""Independent pixel-only inverse arm for paired contour closure.

The oracle knows only the public answer palette. It segments each color,
enumerates disconnected stroke components, and counts components that enclose a
background region. It receives neither scene data nor analytic labels.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


COLORS = ("red", "blue", "green", "purple")
RGB = {
    "red": (216, 54, 72),
    "blue": (42, 104, 196),
    "green": (28, 146, 90),
    "purple": (132, 72, 178),
}
COLOR_TOLERANCE_SQ = 28 * 28
MIN_HOLE_AREA = 28


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    points = {(int(y), int(x)) for y, x in np.argwhere(mask)}
    output: list[list[tuple[int, int]]] = []
    while points:
        start = points.pop()
        component = [start]
        stack = [start]
        while stack:
            y, x = stack.pop()
            for neighbor in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if neighbor in points:
                    points.remove(neighbor)
                    component.append(neighbor)
                    stack.append(neighbor)
        output.append(component)
    return output


def _encloses_hole(component: list[tuple[int, int]]) -> bool:
    ys = [point[0] for point in component]
    xs = [point[1] for point in component]
    y0, y1 = min(ys) - 1, max(ys) + 2
    x0, x1 = min(xs) - 1, max(xs) + 2
    height, width = y1 - y0, x1 - x0
    stroke = np.zeros((height, width), dtype=bool)
    for y, x in component:
        stroke[y - y0, x - x0] = True

    outside = np.zeros_like(stroke)
    queue: deque[tuple[int, int]] = deque()
    for x in range(width):
        if not stroke[0, x]:
            outside[0, x] = True
            queue.append((0, x))
        if not stroke[height - 1, x] and not outside[height - 1, x]:
            outside[height - 1, x] = True
            queue.append((height - 1, x))
    for y in range(height):
        if not stroke[y, 0] and not outside[y, 0]:
            outside[y, 0] = True
            queue.append((y, 0))
        if not stroke[y, width - 1] and not outside[y, width - 1]:
            outside[y, width - 1] = True
            queue.append((y, width - 1))
    while queue:
        y, x = queue.popleft()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if (
                0 <= ny < height
                and 0 <= nx < width
                and not stroke[ny, nx]
                and not outside[ny, nx]
            ):
                outside[ny, nx] = True
                queue.append((ny, nx))
    enclosed = (~stroke) & (~outside)
    return int(enclosed.sum()) >= MIN_HOLE_AREA


def _closed_count(array: np.ndarray, color: str) -> int:
    target = np.asarray(RGB[color], dtype=np.int32)
    delta = array.astype(np.int32) - target
    mask = np.sum(delta * delta, axis=2, dtype=np.int32) <= COLOR_TOLERANCE_SQ
    components = [item for item in _components(mask) if len(item) >= 12]
    return sum(_encloses_hole(component) for component in components)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    scores = {color: _closed_count(array, color) for color in COLORS}
    best = max(scores.values())
    winners = [color for color, score in scores.items() if score == best]
    if best != 2 or len(winners) != 1:
        raise ValueError(f"pixel evidence does not expose a unique 2-hole winner: {scores}")
    return winners[0]
