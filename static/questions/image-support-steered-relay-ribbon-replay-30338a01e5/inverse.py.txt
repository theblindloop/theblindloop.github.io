"""Independent RGB-only inverse arm for Relay Ribbon Replay."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_DARK = np.array((32, 44, 62), dtype=np.int16)
_CYAN = np.array((19, 178, 201), dtype=np.int16)
_VIOLET = np.array((132, 73, 188), dtype=np.int16)
_STATE = {
    "red": np.array((220, 62, 67), dtype=np.int16),
    "green": np.array((45, 158, 91), dtype=np.int16),
    "blue": np.array((47, 105, 196), dtype=np.int16),
}
_STATES = ("red", "green", "blue")


def _mask(array: np.ndarray, color: np.ndarray, tolerance: int = 8) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y = int(y)
        x = int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        output.append(np.asarray(points, dtype=np.int32))
    return output


def _large_components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    return [component for component in _components(mask) if len(component) >= minimum]


def _bbox(component: np.ndarray) -> tuple[int, int, int, int]:
    ys = component[:, 0]
    xs = component[:, 1]
    return int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.shape[:2] != (512, 512):
        raise ValueError("unexpected canvas size")

    dark = _mask(array, _DARK)
    cyan = _mask(array, _CYAN)
    violet = _mask(array, _VIOLET)
    state_masks = {name: _mask(array, color) for name, color in _STATE.items()}
    relevant = dark | cyan | violet
    for value in state_masks.values():
        relevant |= value

    connected = _large_components(relevant, 500)
    if len(connected) != 1:
        raise ValueError("the decisive ribbon is missing or disconnected")
    x0, y0, x1, y1 = _bbox(connected[0])
    if not 82 <= x1 - x0 + 1 <= 150 or not 24 <= y1 - y0 + 1 <= 30:
        raise ValueError("ribbon extent is ambiguous")

    state_hits = []
    for name, mask in state_masks.items():
        components = _large_components(mask, 150)
        for component in components:
            bx0, by0, bx1, by1 = _bbox(component)
            width = bx1 - bx0 + 1
            height = by1 - by0 + 1
            if 14 <= width <= 18 and 14 <= height <= 18:
                state_hits.append((name, float(component[:, 1].mean()), float(component[:, 0].mean())))
    if len(state_hits) != 1:
        raise ValueError("round starting state is ambiguous")
    state, start_x, start_y = state_hits[0]

    spans = []
    for operation, mask in (("advance", cyan), ("swap", violet)):
        for component in _large_components(mask, 150):
            bx0, by0, bx1, by1 = _bbox(component)
            width = bx1 - bx0 + 1
            height = by1 - by0 + 1
            if not 18 <= width <= 24 or not 10 <= height <= 11:
                raise ValueError("operation span geometry is ambiguous")
            spans.append((float(component[:, 1].mean()), float(component[:, 0].mean()), operation))
    if len(spans) != 4:
        raise ValueError("expected four visible operation spans")
    if any(abs(y - start_y) > 2.0 for _, y, _ in spans):
        raise ValueError("operation spans do not lie on the starting trace")

    span_center = sum(x for x, _, _ in spans) / len(spans)
    direction = 1 if span_center > start_x else -1
    spans.sort(key=lambda item: direction * item[0])
    operations = [operation for _, _, operation in spans]
    if any(operations[i] == operations[i + 1] for i in range(3)):
        raise ValueError("span boundaries are not uniquely readable")

    for operation in operations:
        if operation == "advance":
            state = _STATES[(_STATES.index(state) + 1) % 3]
        else:
            state = {"red": "blue", "green": "green", "blue": "red"}[state]
    return state
