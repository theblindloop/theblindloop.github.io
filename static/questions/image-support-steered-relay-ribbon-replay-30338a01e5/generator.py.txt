"""Latent sampler, analytic arm, and renderer for Relay Ribbon Replay."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (255, 255, 255)
DARK = (32, 44, 62)
CYAN = (19, 178, 201)
VIOLET = (132, 73, 188)
STATE_COLORS = {
    "red": (220, 62, 67),
    "green": (45, 158, 91),
    "blue": (47, 105, 196),
}
PALE = (225, 230, 236)
STATES = ("red", "green", "blue")
OPERATIONS = ("advance", "swap")


def _apply(state: str, operation: str) -> str:
    if operation == "advance":
        return STATES[(STATES.index(state) + 1) % 3]
    if operation == "swap":
        return {"red": "blue", "green": "green", "blue": "red"}[state]
    raise ValueError(f"unknown operation {operation!r}")


def _start_for_target(target: str, operations: list[str]) -> str:
    for state in STATES:
        current = state
        for operation in operations:
            current = _apply(current, operation)
        if current == target:
            return state
    raise AssertionError("operation sequence is not bijective")


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    first = OPERATIONS[rng.randrange(2)]
    second = OPERATIONS[1 - OPERATIONS.index(first)]
    operations = [first, second, first, second]
    target = STATES[int(seed) % 3]
    direction = 1 if rng.randrange(2) == 0 else -1
    lengths = [rng.randint(18, 24) for _ in range(4)]
    total = 11 + 7 + sum(lengths) + 7 + 13
    if direction == 1:
        anchor_x = rng.randint(82, WIDTH - total - 54)
    else:
        anchor_x = rng.randint(total + 54, WIDTH - 82)
    # Keep the thin trace centered inside a randomly selected 64 px band. This
    # makes destructive support follow one horizontal chain instead of being
    # split across two grid rows by arbitrary raster-cell boundaries.
    anchor_y = rng.choice([96, 160, 224, 288, 352, 416]) + rng.randint(-2, 2)
    decorations = []
    for _ in range(9):
        for _attempt in range(40):
            x = rng.randint(28, WIDTH - 28)
            y = rng.randint(28, HEIGHT - 28)
            if abs(y - anchor_y) > 55 or abs(x - anchor_x) > 180:
                decorations.append([x, y, rng.choice([2, 3])])
                break
    return {
        "anchor_x": anchor_x,
        "anchor_y": anchor_y,
        "direction": direction,
        "start_state": _start_for_target(target, operations),
        "operations": operations,
        "segment_lengths": lengths,
        "decorations": decorations,
    }


def _span_box(cursor: int, length: int, y: int, direction: int) -> tuple[int, int, int, int]:
    if direction == 1:
        return cursor, y - 5, cursor + length - 1, y + 5
    return cursor - length + 1, y - 5, cursor, y + 5


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x, y, radius in scene["decorations"]:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=PALE)

    x = int(scene["anchor_x"])
    y = int(scene["anchor_y"])
    direction = int(scene["direction"])
    operations = list(scene["operations"])
    lengths = list(scene["segment_lengths"])
    state_color = STATE_COLORS[str(scene["start_state"])]

    # The filled state bead and its dark rim form the round start of the trace.
    draw.ellipse((x - 12, y - 12, x + 12, y + 12), fill=DARK)
    draw.ellipse((x - 8, y - 8, x + 8, y + 8), fill=state_color)

    cursor = x + direction * 11
    draw.rectangle(_span_box(cursor, 8, y, direction), fill=DARK)
    cursor += direction * 8
    for operation, length in zip(operations, lengths, strict=True):
        color = CYAN if operation == "advance" else VIOLET
        draw.rectangle(_span_box(cursor, int(length), y, direction), fill=color)
        cursor += direction * int(length)
    draw.rectangle(_span_box(cursor, 8, y, direction), fill=DARK)
    cursor += direction * 7
    if direction == 1:
        arrow = [(cursor, y - 13), (cursor + 14, y), (cursor, y + 13)]
    else:
        arrow = [(cursor, y - 13), (cursor - 14, y), (cursor, y + 13)]
    draw.polygon(arrow, fill=DARK)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    state = str(scene["start_state"])
    for operation in scene["operations"]:
        state = _apply(state, str(operation))
    return state


def margin(scene: dict[str, Any]) -> float:
    return float(min(int(value) for value in scene["segment_lengths"]) - 5)


def is_quarantined(scene: dict[str, Any]) -> bool:
    operations = list(scene["operations"])
    lengths = list(scene["segment_lengths"])
    return bool(
        len(operations) != 4
        or len(lengths) != 4
        or any(operation not in OPERATIONS for operation in operations)
        or any(operations[i] == operations[i + 1] for i in range(3))
        or any(int(length) < 18 for length in lengths)
        or int(scene["direction"]) not in (-1, 1)
        or str(scene["start_state"]) not in STATES
    )


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
