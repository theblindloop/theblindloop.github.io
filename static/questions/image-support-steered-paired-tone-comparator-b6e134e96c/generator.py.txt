"""Deterministic scene sampler and renderer for the paired tone comparator."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 480
GRID = 7
CELL = 13
PANEL = 109
BACKGROUND = (248, 247, 242)
FRAME_PALETTES = (
    ((20, 145, 166), (229, 116, 30)),
    ((229, 116, 30), (20, 145, 166)),
    ((100, 79, 168), (35, 158, 104)),
    ((35, 158, 104), (100, 79, 168)),
)
SYMBOLS = ("X", "plus", "T", "L")


def _template(name: str) -> list[list[int]]:
    cells = [[0 for _ in range(GRID)] for _ in range(GRID)]
    for row in range(GRID):
        for col in range(GRID):
            if name == "X":
                on = row == col or row + col == GRID - 1
            elif name == "plus":
                on = row == GRID // 2 or col == GRID // 2
            elif name == "T":
                on = row == 0 or col == GRID // 2
            elif name == "L":
                on = col == 0 or row == GRID - 1
            else:
                raise ValueError(f"unknown symbol {name!r}")
            cells[row][col] = int(on)
    return cells


def _classify(bits: list[list[int]]) -> tuple[str, int]:
    scores = []
    for name in SYMBOLS:
        target = _template(name)
        distance = sum(bits[r][c] != target[r][c] for r in range(GRID) for c in range(GRID))
        scores.append((distance, name))
    scores.sort()
    return scores[0][1], int(scores[1][0] - scores[0][0])


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    symbol = SYMBOLS[int(seed) % len(SYMBOLS)]
    bits = _template(symbol)

    # Every glyph contains exactly 13 dark output cells. Within both the 13-cell
    # and 36-cell groups, use balanced tone-gap multisets so total panel ink is
    # answer-independent. Only pairwise ordering carries the label.
    on_indices = [(r, c) for r in range(GRID) for c in range(GRID) if bits[r][c]]
    off_indices = [(r, c) for r in range(GRID) for c in range(GRID) if not bits[r][c]]
    rng.shuffle(on_indices)
    rng.shuffle(off_indices)
    strong = set(on_indices[:6] + off_indices[:18])
    left = [[0] * GRID for _ in range(GRID)]
    right = [[0] * GRID for _ in range(GRID)]
    for row in range(GRID):
        for col in range(GRID):
            low, high = ((70, 230) if (row, col) in strong else (110, 190))
            if bits[row][col]:
                left[row][col], right[row][col] = low, high
            else:
                left[row][col], right[row][col] = high, low

    palette = FRAME_PALETTES[rng.randrange(len(FRAME_PALETTES))]
    return {
        "left_values": left,
        "right_values": right,
        "frame_colors": [list(palette[0]), list(palette[1])],
        "offset_x": rng.randint(-2, 2),
        "offset_y": rng.randint(-2, 2),
    }


def _draw_panel(draw: ImageDraw.ImageDraw, x: int, y: int, values: list, color: Iterable[int]) -> None:
    color = tuple(int(v) for v in color)
    draw.rectangle((x, y, x + PANEL - 1, y + PANEL - 1), fill=color)
    draw.rectangle((x + 8, y + 8, x + PANEL - 9, y + PANEL - 9), fill=(238, 236, 229))
    for row in range(GRID):
        for col in range(GRID):
            x0 = x + 9 + col * CELL
            y0 = y + 9 + row * CELL
            value = int(values[row][col])
            draw.rectangle((x0, y0, x0 + 11, y0 + 11), fill=(value, value, value))


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    ox, oy = int(scene["offset_x"]), int(scene["offset_y"])
    colors = scene["frame_colors"]
    _draw_panel(draw, 188 + ox, 100 + oy, scene["left_values"], colors[0])
    _draw_panel(draw, 352 + ox, 277 + oy, scene["right_values"], colors[1])
    return image


def _comparison_bits(scene: dict) -> list[list[int]]:
    left, right = scene["left_values"], scene["right_values"]
    return [[int(left[r][c] < right[r][c]) for c in range(GRID)] for r in range(GRID)]


def analytic_gold(scene: dict) -> str:
    return _classify(_comparison_bits(scene))[0]


def margin(scene: dict) -> float:
    _, runner_up_gap = _classify(_comparison_bits(scene))
    tone_gap = min(
        abs(int(scene["left_values"][r][c]) - int(scene["right_values"][r][c]))
        for r in range(GRID)
        for c in range(GRID)
    )
    return float(min(runner_up_gap, tone_gap / 10.0))


def is_quarantined(scene: dict) -> bool:
    gaps = [
        abs(int(scene["left_values"][r][c]) - int(scene["right_values"][r][c]))
        for r in range(GRID)
        for c in range(GRID)
    ]
    return min(gaps) < 45 or _classify(_comparison_bits(scene))[1] < 3


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
