"""Independent pixel-only inverse for the paired tone comparator."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

GRID = 7
SYMBOLS = ("X", "plus", "T", "L")


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        xs, ys = [], []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= 500:
            found.append((min(xs), min(ys), max(xs), max(ys), len(xs)))
    return found


def _read_panel(array: np.ndarray, box: tuple[int, int, int, int, int]) -> list[list[float]]:
    x0, y0, x1, y1, count = box
    if not (107 <= x1 - x0 <= 110 and 107 <= y1 - y0 <= 110 and count >= 1800):
        raise ValueError("missing or damaged panel fiducial")
    pitch_x = (x1 - x0 + 1 - 18) / GRID
    pitch_y = (y1 - y0 + 1 - 18) / GRID
    values = []
    for row in range(GRID):
        out_row = []
        for col in range(GRID):
            cx = int(round(x0 + 9 + col * pitch_x + pitch_x / 2 - 0.5))
            cy = int(round(y0 + 9 + row * pitch_y + pitch_y / 2 - 0.5))
            patch = array[cy - 3 : cy + 4, cx - 3 : cx + 4].astype(np.float32)
            if patch.shape != (7, 7, 3):
                raise ValueError("cell sample outside raster")
            channel_spread = float(np.max(np.ptp(patch, axis=2)))
            tone = float(np.median(np.mean(patch, axis=2)))
            if channel_spread > 8 or not 55 <= tone <= 238:
                raise ValueError("missing or non-gray cell evidence")
            out_row.append(tone)
        values.append(out_row)
    return values


def _template(name: str) -> list[list[int]]:
    result = []
    for row in range(GRID):
        line = []
        for col in range(GRID):
            if name == "X":
                on = row == col or row + col == GRID - 1
            elif name == "plus":
                on = row == GRID // 2 or col == GRID // 2
            elif name == "T":
                on = row == 0 or col == GRID // 2
            else:
                on = col == 0 or row == GRID - 1
            line.append(int(on))
        result.append(line)
    return result


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    spread = array.max(axis=2).astype(np.int16) - array.min(axis=2).astype(np.int16)
    colored = spread > 45
    boxes = _components(colored)
    if len(boxes) != 2:
        raise ValueError(f"expected two visible panel fiducials, found {len(boxes)}")
    boxes.sort(key=lambda item: item[0])
    left = _read_panel(array, boxes[0])
    right = _read_panel(array, boxes[1])
    bits = [[int(left[r][c] < right[r][c]) for c in range(GRID)] for r in range(GRID)]
    min_gap = min(abs(left[r][c] - right[r][c]) for r in range(GRID) for c in range(GRID))
    if min_gap < 45:
        raise ValueError("ambiguous paired tone gap")
    scored = []
    for name in SYMBOLS:
        target = _template(name)
        distance = sum(bits[r][c] != target[r][c] for r in range(GRID) for c in range(GRID))
        scored.append((distance, name))
    scored.sort()
    if scored[1][0] - scored[0][0] < 3:
        raise ValueError("ambiguous symbol classification")
    return scored[0][1]
