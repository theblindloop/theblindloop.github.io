"""Reflection (mirror-fold) register accumulator (state tracking).

The image shows one static scene. Across the top, a horizontal track of NOSLOT=5
slot boxes is drawn left to right (slot 1 = leftmost ... slot 5 = rightmost).
Exactly one slot box holds a single red marker disk, and that box is also
surrounded by an orange ring: this marks the TRACKED START slot.

To the right (in a row below the track) sit 4 operator boxes read left to
right. Each operator box holds exactly one colored disk that declares a step
applied to the tracked register:

  - a GREEN disk  = advance the register one slot to the right (+1);
  - a BLUE disk   = retreat the register one slot to the left  (-1);
  - a VIOLET disk = MIRROR the register, reflecting the current position about
                    the track's central vertical axis (slot index p maps to
                    (NOSLOT-1)-p, i.e. slot 1 <-> slot 5, slot 2 <-> slot 4,
                    and slot 3 stays fixed as the fold line).

The operator COLOR alone carries the operator type; no arrow geometry is
needed. The register is never drawn except at its ringed start, so the FINAL
slot is obtained only by replaying the declared operators in order from the
ringed start. Reflection is a non-additive, order-sensitive fold: it inverts
the current index about the centre of the rail, so the same multiset of
operators in a different order can reach a different final slot.

The start slot and every replay step are constrained to keep the register inside
the five-slot rail (a mirror always does; +1/-1 steps are sampled with an
in-range guard), so every scene has exactly one consistent replay. The decision
is the final slot label "slot 1".."slot 5".

Latent fields: ``start`` (0-based index 0..4 of the ringed red-marker slot) and
``ops`` (the ordered length-4 list of "+1"/"-1"/"MIRROR" operator types drawn
as colored disks). Every field is render-evident and feeds the analytic gold.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 940
HEIGHT = 320

NOSLOT = 5
SLOT_W = 64
SLOT_H = 64
SLOT_Y0 = 50
SLOT_X0 = 40
SLOT_GAP = 74
MARKER_R = 16

NOPS = 4
OPS_W = 120
OPS_H = 54
OPS_Y = 200
OPS_X0 = 46
OPS_SPACING = 195
DISK_R = 13

BG = (255, 255, 255)
BOX_OUTLINE = (90, 90, 90)
LIGHT = (190, 190, 190)
MARKER = (205, 40, 40)
RING = (240, 150, 20)
GREEN = (35, 160, 70)
BLUE = (45, 70, 210)
VIOLET = (150, 60, 175)

OP_COLOR = {"+1": GREEN, "-1": BLUE, "MIRROR": VIOLET}
SLOT_LABELS = [f"slot {i + 1}" for i in range(NOSLOT)]


def slot_x0s():
    return [SLOT_X0 + i * SLOT_GAP for i in range(NOSLOT)]


def op_x0s():
    return [OPS_X0 + i * OPS_SPACING for i in range(NOPS)]


def slot_center(i: int) -> tuple[int, int]:
    x0 = SLOT_X0 + i * SLOT_GAP
    return (int(x0 + SLOT_W * 0.5), int(SLOT_Y0 + SLOT_H * 0.5))


def op_center(i: int) -> tuple[int, int]:
    x0 = OPS_X0 + i * OPS_SPACING
    return (int(x0 + OPS_W * 0.5), int(OPS_Y + OPS_H * 0.5))


def _in_range(start: int, ops) -> bool:
    pos = start
    for op in ops:
        if op == "+1":
            pos += 1
        elif op == "-1":
            pos -= 1
        else:
            pos = NOSLOT - 1 - pos
        if not 0 <= pos < NOSLOT:
            return False
    return True


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    for _ in range(50_000):
        start = rng.randrange(NOSLOT)
        ops = [rng.choice(["+1", "-1", "MIRROR"]) for _ in range(NOPS)]
        if _in_range(start, ops) and "MIRROR" in ops:
            return {"start": start, "ops": ops}
    raise RuntimeError("could not sample a valid mirror-register scene")


def _replay(start: int, ops) -> int:
    pos = start
    for op in ops:
        if op == "+1":
            pos += 1
        elif op == "-1":
            pos -= 1
        else:
            pos = NOSLOT - 1 - pos
    return pos


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)
    start = scene["start"]
    ops = scene["ops"]

    for i in range(NOSLOT):
        x0 = SLOT_X0 + i * SLOT_GAP
        draw.rectangle([x0, SLOT_Y0, x0 + SLOT_W, SLOT_Y0 + SLOT_H], outline=LIGHT, width=3)

    mx, my = slot_center(start)
    draw.rectangle(
        [mx - MARKER_R, my - MARKER_R, mx + MARKER_R, my + MARKER_R],
        fill=MARKER,
    )
    inset = 7
    sx = SLOT_X0 + start * SLOT_GAP
    draw.rectangle(
        [sx + inset, SLOT_Y0 + inset, sx + SLOT_W - inset, SLOT_Y0 + SLOT_H - inset],
        outline=RING,
        width=5,
    )

    for i, op in enumerate(ops):
        x0 = OPS_X0 + i * OPS_SPACING
        draw.rectangle([x0, OPS_Y, x0 + OPS_W, OPS_Y + OPS_H], outline=LIGHT, width=3)
        ox, oy = op_center(i)
        draw.ellipse(
            [ox - DISK_R, oy - DISK_R, ox + DISK_R, oy + DISK_R], fill=OP_COLOR[op]
        )
    return img


def analytic_gold(scene: dict) -> str:
    pos = _replay(scene["start"], scene["ops"])
    return SLOT_LABELS[pos]


def margin(scene: dict) -> float:
    return 12.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    """No nontrivial render-invariant latent symmetry.

    Every latent field is directly render-evident and feeds the analytic gold:
    ``start`` fixes which slot box holds the red marker (and is ringed), and
    ``ops`` fixes the colours of the four drawn operator disks (green/blue/
    violet). No reparametrization yields a pixel-identical render with a
    different final slot, because the ringed start slot and every operator disk
    colour are explicitly visible, so there is no render-invariant alias.
    """
    return []
