"""Independent PNG-only inverse for the three conditional rewrite badges.

The inverse finds saturated badge interiors without knowing their coordinates or
palette, classifies the three silhouettes by fill geometry, counts compact dark
pip components, binds left/right rule groups, and replays the two conditions.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_WORDS = ("one", "two", "three")


def _components(mask: np.ndarray, minimum: int = 1) -> list[np.ndarray]:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if visited[y0, x0]:
            continue
        queue = deque([(y0, x0)])
        visited[y0, x0] = True
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            output.append(np.asarray(points, dtype=np.int32))
    return output


def _badge_parts(array: np.ndarray) -> dict[str, tuple[np.ndarray, tuple[int, int, int, int]]]:
    values = array.astype(np.int16)
    chroma = values.max(axis=2) - values.min(axis=2)
    distance = np.linalg.norm(values - 255, axis=2)
    colored = (chroma > 28) & (values.mean(axis=2) > 105) & (distance > 55)
    parts = [component for component in _components(colored, minimum=450)]
    if len(parts) != 3:
        raise ValueError(f"expected three intact badge interiors, found {len(parts)}")

    measured = []
    for points in parts:
        y0, x0 = points.min(axis=0)
        y1, x1 = points.max(axis=0)
        width = int(x1 - x0 + 1)
        height = int(y1 - y0 + 1)
        fill = float(len(points) / (width * height))
        measured.append((fill, points, (int(x0), int(y0), int(x1), int(y1))))
    measured.sort(key=lambda item: item[0])
    if not (measured[0][0] < 0.55 and 0.65 < measured[1][0] < 0.82 and measured[2][0] > 0.84):
        raise ValueError("badge silhouettes are incomplete or ambiguous")
    return {
        "diamond": (measured[0][1], measured[0][2]),
        "round": (measured[1][1], measured[1][2]),
        "square": (measured[2][1], measured[2][2]),
    }


def _pip_centers(array: np.ndarray, bounds: tuple[int, int, int, int]) -> list[tuple[float, float]]:
    x0, y0, x1, y1 = bounds
    pad = 5
    xa, ya = max(0, x0 - pad), max(0, y0 - pad)
    xb, yb = min(array.shape[1] - 1, x1 + pad), min(array.shape[0] - 1, y1 + pad)
    crop = array[ya : yb + 1, xa : xb + 1]
    dark = np.max(crop, axis=2) < 85
    centers: list[tuple[float, float]] = []
    for points in _components(dark, minimum=9):
        py0, px0 = points.min(axis=0)
        py1, px1 = points.max(axis=0)
        width = int(px1 - px0 + 1)
        height = int(py1 - py0 + 1)
        area = len(points)
        if 3 <= width <= 9 and 3 <= height <= 9 and 9 <= area <= 90:
            centers.append((float(points[:, 1].mean() + xa), float(points[:, 0].mean() + ya)))
    return centers


def _read_start(array: np.ndarray, bounds: tuple[int, int, int, int]) -> int:
    count = len(_pip_centers(array, bounds))
    if count not in (1, 2, 3):
        raise ValueError(f"round start badge has ambiguous pip count {count}")
    return count


def _read_rule(array: np.ndarray, bounds: tuple[int, int, int, int]) -> tuple[int, int]:
    pips = _pip_centers(array, bounds)
    center_x = (bounds[0] + bounds[2]) / 2.0
    left = sum(x < center_x - 2.0 for x, _ in pips)
    right = sum(x > center_x + 2.0 for x, _ in pips)
    if left not in (1, 2, 3) or right not in (1, 2, 3) or left + right != len(pips):
        raise ValueError(f"rule badge has ambiguous pip groups {left}|{right}")
    return left, right


def _rewrite(value: int, pair: tuple[int, int]) -> int:
    source, target = pair
    return target if value == source else value


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 32 or array.shape[1] < 32:
        raise ValueError("raster is too small")
    badges = _badge_parts(array)
    value = _read_start(array, badges["round"][1])
    value = _rewrite(value, _read_rule(array, badges["square"][1]))
    value = _rewrite(value, _read_rule(array, badges["diamond"][1]))
    return _WORDS[value - 1]
