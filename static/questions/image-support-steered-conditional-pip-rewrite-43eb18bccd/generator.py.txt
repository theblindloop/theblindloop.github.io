"""Deterministic Pillow renderer and analytic arm for conditional pip rewrites."""

from __future__ import annotations

import random
from typing import Callable

from PIL import Image, ImageDraw

WIDTH = 800
HEIGHT = 800
CELL = 100
STATE_WORDS = ("one", "two", "three")

_BACKGROUND = (255, 255, 255)
_INK = (28, 34, 45)
_DISTRACTOR = (226, 229, 233)
_PALETTES = (
    ((244, 180, 195), (248, 208, 108), (125, 219, 207)),
    ((158, 202, 245), (215, 178, 242), (246, 186, 132)),
    ((166, 225, 184), (238, 158, 151), (188, 180, 239)),
)


def _other(rng: random.Random, value: int) -> int:
    return rng.choice([item for item in (1, 2, 3) if item != value])


def _apply(value: int, source: int, target: int) -> int:
    return target if value == source else value


def sample_scene(seed: int) -> dict:
    """Balance final state and causal mode independently of styling and placement."""
    rng = random.Random(int(seed))
    final = int(seed) % 3 + 1
    mode = int(seed) % 4

    if mode == 0:  # both rules apply
        middle = _other(rng, final)
        start = _other(rng, middle)
        square_from, square_to = start, middle
        diamond_from, diamond_to = middle, final
    elif mode == 1:  # only the square rule applies
        start = _other(rng, final)
        square_from, square_to = start, final
        diamond_from = _other(rng, final)
        diamond_to = _other(rng, diamond_from)
    elif mode == 2:  # both rules visibly fail their match tests
        start = final
        square_from = _other(rng, start)
        square_to = _other(rng, square_from)
        diamond_from = _other(rng, start)
        diamond_to = _other(rng, diamond_from)
    else:  # square misses, diamond applies
        start = _other(rng, final)
        square_from = _other(rng, start)
        square_to = _other(rng, square_from)
        diamond_from, diamond_to = start, final

    return {
        "start_count": start,
        "square_from": square_from,
        "square_to": square_to,
        "diamond_from": diamond_from,
        "diamond_to": diamond_to,
        "anchor_row": rng.randint(1, 4),
        "anchor_col": rng.randint(1, 4),
        "palette_variant": rng.randrange(len(_PALETTES)),
        "badge_size": rng.choice((46, 48)),
        "pip_radius": rng.choice((2, 3)),
        "distractor_variant": rng.randrange(4),
    }


def _centers(scene: dict) -> tuple[tuple[int, int], tuple[int, int], tuple[int, int]]:
    row = int(scene["anchor_row"])
    col = int(scene["anchor_col"])
    start = (col * CELL + CELL // 2, row * CELL + CELL // 2)
    square = ((col + 2) * CELL + CELL // 2, row * CELL + CELL // 2)
    diamond = ((col + 1) * CELL + CELL // 2, (row + 2) * CELL + CELL // 2)
    return start, square, diamond


def _pip_offsets(count: int) -> tuple[int, ...]:
    if count == 1:
        return (0,)
    if count == 2:
        return (-5, 5)
    if count == 3:
        return (-8, 0, 8)
    raise ValueError("pip count must be one, two, or three")


def _draw_pip_group(
    draw: ImageDraw.ImageDraw, center: tuple[int, int], count: int, radius: int
) -> None:
    cx, cy = center
    for offset in _pip_offsets(count):
        draw.ellipse(
            (cx - radius, cy + offset - radius, cx + radius, cy + offset + radius),
            fill=_INK,
        )


def _draw_start(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    count: int,
    size: int,
    radius: int,
    fill: tuple[int, int, int],
) -> None:
    cx, cy = center
    half = size // 2
    box = (cx - half, cy - half, cx + half, cy + half)
    draw.ellipse(box, fill=fill, outline=_INK, width=2)
    _draw_pip_group(draw, center, count, radius)


def _draw_rule(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    source: int,
    target: int,
    size: int,
    radius: int,
    fill: tuple[int, int, int],
    polygon: Callable[[int, int, int], list[tuple[int, int]]],
) -> None:
    cx, cy = center
    points = polygon(cx, cy, size // 2)
    draw.polygon(points, fill=fill, outline=_INK, width=2)
    draw.line((cx, cy - 13, cx, cy + 13), fill=_INK, width=2)
    # Keep both groups clear of the sloping diamond outline at the three-pip
    # extremes while retaining a visible gap around the central divider.
    group_offset = max(7, size // 6)
    _draw_pip_group(draw, (cx - group_offset, cy), source, radius)
    _draw_pip_group(draw, (cx + group_offset, cy), target, radius)


def _square(cx: int, cy: int, half: int) -> list[tuple[int, int]]:
    return [(cx - half, cy - half), (cx + half, cy - half), (cx + half, cy + half), (cx - half, cy + half)]


def _diamond(cx: int, cy: int, half: int) -> list[tuple[int, int]]:
    return [(cx, cy - half), (cx + half, cy), (cx, cy + half), (cx - half, cy)]


def _draw_distractors(draw: ImageDraw.ImageDraw, variant: int) -> None:
    layouts = (
        ((60, 60), (740, 740)),
        ((740, 60), (60, 740)),
        ((60, 60), (740, 60), (60, 740)),
        ((740, 740), (740, 60), (60, 740)),
    )
    for cx, cy in layouts[variant]:
        draw.line((cx - 5, cy, cx + 5, cy), fill=_DISTRACTOR, width=2)
        draw.line((cx, cy - 5, cx, cy + 5), fill=_DISTRACTOR, width=2)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), _BACKGROUND)
    draw = ImageDraw.Draw(image)
    palette = _PALETTES[int(scene["palette_variant"])]
    size = int(scene["badge_size"])
    radius = int(scene["pip_radius"])
    start_center, square_center, diamond_center = _centers(scene)
    _draw_distractors(draw, int(scene["distractor_variant"]))
    _draw_start(draw, start_center, int(scene["start_count"]), size, radius, palette[0])
    _draw_rule(
        draw,
        square_center,
        int(scene["square_from"]),
        int(scene["square_to"]),
        size,
        radius,
        palette[1],
        _square,
    )
    _draw_rule(
        draw,
        diamond_center,
        int(scene["diamond_from"]),
        int(scene["diamond_to"]),
        size,
        radius,
        palette[2],
        _diamond,
    )
    return image


def analytic_gold(scene: dict) -> str:
    value = int(scene["start_count"])
    value = _apply(value, int(scene["square_from"]), int(scene["square_to"]))
    value = _apply(value, int(scene["diamond_from"]), int(scene["diamond_to"]))
    return STATE_WORDS[value - 1]


def margin(scene: dict) -> float:
    """Pixel clearance between a pip and its nearest non-group stroke."""
    half = int(scene["badge_size"]) // 2
    radius = int(scene["pip_radius"])
    return float(min(half - 13, int(scene["badge_size"]) // 4 - radius - 2))


def is_quarantined(scene: dict) -> bool:
    return (
        int(scene["badge_size"]) < 46
        or int(scene["pip_radius"]) < 2
        or margin(scene) < 6.0
        or int(scene["palette_variant"]) not in range(len(_PALETTES))
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
