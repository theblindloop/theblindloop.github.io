"""Independent pixel-only inverse arm for widest_empty_moat.

This module intentionally contains its own palette, segmentation, component,
centroid, and distance code. It receives no latent scene and imports no world
module.
"""

from __future__ import annotations

import math
from dataclasses import dataclass

import numpy as np
from PIL import Image

DARK = (35, 39, 45)
COLORS = {
    "red": (205, 51, 61),
    "blue": (42, 102, 184),
    "green": (31, 142, 84),
    "orange": (225, 126, 34),
}


@dataclass(frozen=True)
class PixelMeasurement:
    decision: str
    margin: float
    clearances: dict[str, float]
    dark_dot_count: int


def _near_color(array: np.ndarray, rgb: tuple[int, int, int], radius: float) -> np.ndarray:
    # int32 avoids overflow when squaring full-range RGB differences.
    delta = array.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2) <= radius * radius


def _components(mask: np.ndarray, minimum_area: int) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components: list[np.ndarray] = []
    for start_y, start_x in np.argwhere(mask):
        if seen[start_y, start_x]:
            continue
        stack = [(int(start_y), int(start_x))]
        seen[start_y, start_x] = True
        pixels: list[tuple[int, int]] = []
        while stack:
            y, x = stack.pop()
            pixels.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(pixels) >= minimum_area:
            components.append(np.asarray(pixels, dtype=np.float64))
    return components


def _centroid(component: np.ndarray) -> tuple[float, float]:
    # Pixel centers are used consistently for both rings and disks.
    return (float(component[:, 1].mean() + 0.5), float(component[:, 0].mean() + 0.5))


def measure_from_image(image: Image.Image) -> PixelMeasurement:
    array = np.asarray(image.convert("RGB"))
    dark_components = _components(_near_color(array, DARK, 62.0), minimum_area=35)
    if len(dark_components) < 8:
        raise ValueError("insufficient dark-dot evidence")
    dark_centers = [_centroid(component) for component in dark_components]

    candidate_centers: dict[str, tuple[float, float]] = {}
    for name, rgb in COLORS.items():
        components = _components(_near_color(array, rgb, 58.0), minimum_area=45)
        if len(components) != 1:
            raise ValueError(f"expected one visible {name} ring, found {len(components)}")
        candidate_centers[name] = _centroid(components[0])

    clearances = {}
    for name, (cx, cy) in candidate_centers.items():
        clearances[name] = min(
            math.hypot(cx - point_x, cy - point_y) for point_x, point_y in dark_centers
        )
    ranking = sorted(clearances, key=clearances.get, reverse=True)
    gap = clearances[ranking[0]] - clearances[ranking[1]]
    if gap < 1.25:
        raise ValueError(f"pixel evidence is ambiguous (top-two gap {gap:.3f}px)")
    return PixelMeasurement(ranking[0], float(gap), clearances, len(dark_centers))


def decision_from_image(image: Image.Image) -> str:
    return measure_from_image(image).decision
