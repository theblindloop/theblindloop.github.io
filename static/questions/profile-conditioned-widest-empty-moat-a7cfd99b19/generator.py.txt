"""Deterministic latent sampler and Pillow renderer for widest_empty_moat."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
SUPERSAMPLE = 4
BACKGROUND = (248, 247, 242)
DARK = (35, 39, 45)
COLORS = {
    "red": (205, 51, 61),
    "blue": (42, 102, 184),
    "green": (31, 142, 84),
    "orange": (225, 126, 34),
}
COLOR_NAMES = tuple(COLORS)
DOT_RADIUS = 5.5
RING_RADIUS = 11.5
RING_WIDTH = 4.5
QUARANTINE_GAP = 7.0


def _distance(a: Iterable[float], b: Iterable[float]) -> float:
    ax, ay = a
    bx, by = b
    return math.hypot(ax - bx, ay - by)


def _clearances(points: list[list[float]], candidates: list[dict]) -> list[float]:
    return [
        min(_distance((candidate["x"], candidate["y"]), point) for point in points)
        for candidate in candidates
    ]


def _candidate_positions(rng: random.Random) -> list[list[float]]:
    # One visually equivalent candidate per broad quadrant. Jitter and a random
    # global shear prevent four fixed coordinates from becoming a template.
    slots = ((155, 155), (357, 155), (155, 357), (357, 357))
    shear_x = rng.uniform(-19.0, 19.0)
    shear_y = rng.uniform(-19.0, 19.0)
    result = []
    for index, (base_x, base_y) in enumerate(slots):
        row = index // 2
        col = index % 2
        x = base_x + rng.uniform(-45.0, 45.0) + (row - 0.5) * shear_x
        y = base_y + rng.uniform(-45.0, 45.0) + (col - 0.5) * shear_y
        result.append([round(x, 3), round(y, 3)])
    return result


def _sample_points(rng: random.Random, centers: list[list[float]]) -> list[list[float]] | None:
    points: list[list[float]] = []
    for _ in range(36):
        for _attempt in range(800):
            point = [rng.uniform(38.0, 474.0), rng.uniform(38.0, 474.0)]
            if min(_distance(point, center) for center in centers) < 22.0:
                continue
            if points and min(_distance(point, other) for other in points) < 15.0:
                continue
            points.append([round(point[0], 3), round(point[1], 3)])
            break
        else:
            return None
    return points


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene.

    Consecutive groups of 24 seeds contain six winners of every color and six
    winners in every broad candidate-position slot. Every sixth seed is a
    deliberate, still-decodable near tie for quarantine coverage.
    """

    rng = random.Random(int(seed))
    desired_color_index = int(seed) % 4
    desired_slot = ((int(seed) // 4) + int(seed)) % 4
    near_boundary = int(seed) % 6 == 0

    for _scene_attempt in range(12000):
        centers = _candidate_positions(rng)
        points = _sample_points(rng, centers)
        if points is None:
            continue
        provisional = [
            {"x": center[0], "y": center[1], "color": COLOR_NAMES[index]}
            for index, center in enumerate(centers)
        ]
        scores = _clearances(points, provisional)
        order = sorted(range(4), key=lambda index: scores[index], reverse=True)
        gap = scores[order[0]] - scores[order[1]]
        if order[0] != desired_slot:
            continue
        if near_boundary:
            if not 3.0 <= gap <= 5.8:
                continue
        elif not 10.0 <= gap <= 30.0:
            continue

        colors = list(COLOR_NAMES)
        desired_color = COLOR_NAMES[desired_color_index]
        colors.remove(desired_color)
        rng.shuffle(colors)
        assigned = []
        for index, center in enumerate(centers):
            color = desired_color if index == desired_slot else colors.pop()
            assigned.append({"x": center[0], "y": center[1], "color": color})
        return {
            "width": WIDTH,
            "height": HEIGHT,
            "points": points,
            "candidates": assigned,
        }
    raise RuntimeError(f"could not sample a separated clearance scene for seed {seed}")


def render(scene: dict) -> Image.Image:
    scale = SUPERSAMPLE
    width = int(scene["width"])
    height = int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)

    for x, y in scene["points"]:
        radius = DOT_RADIUS
        draw.ellipse(
            ((x - radius) * scale, (y - radius) * scale,
             (x + radius) * scale, (y + radius) * scale),
            fill=DARK,
        )

    for candidate in scene["candidates"]:
        x = float(candidate["x"])
        y = float(candidate["y"])
        radius = RING_RADIUS
        draw.ellipse(
            ((x - radius) * scale, (y - radius) * scale,
             (x + radius) * scale, (y + radius) * scale),
            outline=COLORS[candidate["color"]],
            width=round(RING_WIDTH * scale),
        )

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    points = scene["points"]
    candidates = scene["candidates"]
    scores = _clearances(points, candidates)
    winner = max(range(len(candidates)), key=lambda index: scores[index])
    return str(candidates[winner]["color"])


def margin(scene: dict) -> float:
    points = scene["points"]
    candidates = scene["candidates"]
    scores = sorted(_clearances(points, candidates), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = {
        "width": scene["width"],
        "height": scene["height"],
        "points": list(reversed(scene["points"])),
        "candidates": list(reversed(scene["candidates"])),
    }
    return [("reverse_unordered_drawing_sets", twin)]
