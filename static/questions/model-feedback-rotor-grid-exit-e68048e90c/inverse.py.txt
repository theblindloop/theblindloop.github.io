"""Independent pixel-only inverse arm for the rotor-grid-exit world.

From the final PNG alone it

1. isolates the arrow ink by colour and splits it into four row bands and four
   column bands (the arrows live in disjoint cells, so the projections have
   four clean runs each);
2. reads every arrow's direction from the offset between its centroid and the
   centre of its bounding box (a solid triangle's mass sits behind its apex);
3. locates the pure-black start border and snaps it to one band pair;
4. runs the rotor walk -- use the current arrow, then turn it a quarter turn
   clockwise -- and reports the side the token crosses.

It imports no renderer / prompts / generate / verify / analytic-gold code and
never receives the latent scene.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_CW = {"N": "E", "E": "S", "S": "W", "W": "N"}
_STEP = {"N": (-1, 0), "E": (0, 1), "S": (1, 0), "W": (0, -1)}


def _near(arr: np.ndarray, color: tuple[int, int, int], tol: int) -> np.ndarray:
    diff = np.abs(arr.astype(int) - np.array(color, dtype=int))
    return (diff <= tol).all(axis=-1)


def _bands(flags: np.ndarray) -> list[tuple[int, int]]:
    """Inclusive (lo, hi) ranges of the contiguous True runs."""
    out: list[tuple[int, int]] = []
    start = None
    for i, f in enumerate(flags):
        if f and start is None:
            start = i
        elif not f and start is not None:
            out.append((start, i - 1))
            start = None
    if start is not None:
        out.append((start, len(flags) - 1))
    return out


def _direction(sub: np.ndarray) -> str:
    ys, xs = np.nonzero(sub)
    if xs.size == 0:
        return "N"
    ox = xs.mean() - (xs.min() + xs.max()) / 2.0
    oy = ys.mean() - (ys.min() + ys.max()) / 2.0
    if abs(ox) >= abs(oy):
        return "E" if ox < 0 else "W"
    return "N" if oy > 0 else "S"


def decision_from_image(image: Image.Image) -> str:
    """Return 'N', 'E', 'S' or 'W': the side the token leaves from."""
    arr = np.asarray(image.convert("RGB"))
    blue = _near(arr, (38, 92, 176), 45)
    black = arr.astype(int).sum(axis=-1) <= 40

    rows = [b for b in _bands(blue.any(axis=1)) if b[1] - b[0] >= 6]
    cols = [b for b in _bands(blue.any(axis=0)) if b[1] - b[0] >= 6]
    n = len(rows)
    if n < 3 or len(cols) != n:
        return "N"

    field: list[str] = []
    for r0, r1 in rows:
        for c0, c1 in cols:
            field.append(_direction(blue[r0 : r1 + 1, c0 : c1 + 1]))

    ys, xs = np.nonzero(black)
    if ys.size == 0:
        return "N"
    my = (ys.min() + ys.max()) / 2.0
    mx = (xs.min() + xs.max()) / 2.0
    r = min(range(n), key=lambda i: abs((rows[i][0] + rows[i][1]) / 2.0 - my))
    c = min(range(n), key=lambda i: abs((cols[i][0] + cols[i][1]) / 2.0 - mx))

    for _ in range(4000):
        idx = r * n + c
        d = field[idx]
        field[idx] = _CW[d]
        dr, dc = _STEP[d]
        r, c = r + dr, c + dc
        if r < 0:
            return "N"
        if r > n - 1:
            return "S"
        if c < 0:
            return "W"
        if c > n - 1:
            return "E"
    return "N"
