"""Independent pixel-only inverse arm for local reference bracketing."""

from __future__ import annotations

import numpy as np
from PIL import Image

TARGETS = {
    "coral": np.asarray((224, 72, 86), dtype=np.int16),
    "blue": np.asarray((45, 105, 205), dtype=np.int16),
    "gold": np.asarray((224, 158, 24), dtype=np.int16),
}


def _centroid(array: np.ndarray, target: np.ndarray) -> tuple[float, float]:
    delta = array.astype(np.int32) - target.astype(np.int32)[None, None, :]
    mask = np.sum(delta * delta, axis=2) <= 18 * 18
    points = np.argwhere(mask)
    if len(points) < 120:
        raise ValueError("required colored measurement marker is missing or corrupted")

    # Reject merged or wildly spread masks without consulting any latent scene.
    y_span = int(points[:, 0].max() - points[:, 0].min())
    x_span = int(points[:, 1].max() - points[:, 1].min())
    if y_span > 24 or x_span > 24:
        raise ValueError("colored marker mask is not one compact component")
    y, x = points.mean(axis=0)
    return float(x), float(y)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    _, coral_y = _centroid(array, TARGETS["coral"])
    _, blue_y = _centroid(array, TARGETS["blue"])
    _, reference_y = _centroid(array, TARGETS["gold"])
    coral_offset = coral_y - reference_y
    blue_offset = blue_y - reference_y
    if min(abs(coral_offset), abs(blue_offset)) < 12.0:
        raise ValueError("raster lies too close to the visible level boundary")
    return "yes" if coral_offset * blue_offset < 0 else "no"
