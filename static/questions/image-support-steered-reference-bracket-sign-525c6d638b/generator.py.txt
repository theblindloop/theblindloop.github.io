"""Latent sampler, Pillow renderer, and analytic arm for local bracketing."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (250, 249, 246)
CORAL = (224, 72, 86)
BLUE = (45, 105, 205)
GOLD = (224, 158, 24)


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene with three separated, compact markers."""

    rng = random.Random(int(seed))
    base_x = rng.choice((224, 288))
    base_y = rng.choice((224, 288))
    radius = rng.choice((9, 10, 11))
    yes_case = int(seed) % 2 == 0

    if yes_case:
        # Diagonal endpoints keep all three evidence components in separate
        # 64-pixel grid neighborhoods while remaining a focal triad.
        if rng.randrange(2):
            coral_x, coral_y = base_x - 64, base_y - 64
            blue_x, blue_y = base_x + 64, base_y + 64
        else:
            coral_x, coral_y = base_x + 64, base_y + 64
            blue_x, blue_y = base_x - 64, base_y - 64
        reference_x, reference_y = base_x, base_y
    else:
        side = -1 if rng.randrange(2) else 1
        coral_x, coral_y = base_x - 64, base_y + side * 64
        blue_x, blue_y = base_x + 64, base_y + side * 64
        reference_x, reference_y = base_x, base_y - side * 64

    return {
        "coral_y": coral_y,
        "blue_y": blue_y,
        "reference_y": reference_y,
        "coral_x": coral_x,
        "blue_x": blue_x,
        "reference_x": reference_x,
        "radius": radius,
        "draw_order": rng.randrange(2),
    }


def _draw_marker(draw: ImageDraw.ImageDraw, x: int, y: int, radius: int, color: tuple) -> None:
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    # Low-contrast, fixed calibration-paper marks provide visual context but
    # carry no scene or answer information and are ignored by the oracle.
    pale = (235, 232, 224)
    for x, y in ((80, 88), (430, 94), (72, 418), (438, 424)):
        draw.line((x - 7, y, x + 7, y), fill=pale, width=2)
        draw.line((x, y - 7, x, y + 7), fill=pale, width=2)

    radius = int(scene["radius"])
    markers = [
        (int(scene["coral_x"]), int(scene["coral_y"]), CORAL),
        (int(scene["blue_x"]), int(scene["blue_y"]), BLUE),
        (int(scene["reference_x"]), int(scene["reference_y"]), GOLD),
    ]
    if int(scene["draw_order"]) == 1:
        markers.reverse()
    for x, y, color in markers:
        _draw_marker(draw, x, y, radius, color)
    return image


def analytic_gold(scene: dict) -> str:
    coral_offset = int(scene["coral_y"]) - int(scene["reference_y"])
    blue_offset = int(scene["blue_y"]) - int(scene["reference_y"])
    return "yes" if coral_offset * blue_offset < 0 else "no"


def margin(scene: dict) -> float:
    coral_offset = abs(int(scene["coral_y"]) - int(scene["reference_y"]))
    blue_offset = abs(int(scene["blue_y"]) - int(scene["reference_y"]))
    return float(min(coral_offset, blue_offset))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 24.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["draw_order"] = 1 - int(scene["draw_order"])
    return [("reverse_disconnected_draw_order", twin)]
