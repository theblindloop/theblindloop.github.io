"""Deterministic renderer and analytic arm for paired disjoint routing.

The latent graph is a subgraph of a visibly complete 4x4 set of junction
positions.  Exactly 18 orthogonal links are present.  Four colored junctions
name two terminal pairs.  Gold is whether both pairs admit simultaneous
vertex-disjoint routes.
"""

from __future__ import annotations

import random
from collections import deque

from PIL import Image, ImageDraw


GRID = 4
WIDTH = 512
HEIGHT = 512
POTENTIAL_EDGES = tuple(
    (row * GRID + col, row * GRID + col + 1)
    for row in range(GRID)
    for col in range(GRID - 1)
) + tuple(
    (row * GRID + col, (row + 1) * GRID + col)
    for row in range(GRID - 1)
    for col in range(GRID)
)

PALETTES = (
    {
        "background": (250, 249, 246),
        "edge": (42, 46, 55),
        "neutral": (174, 181, 192),
        "green": (25, 158, 92),
        "purple": (146, 82, 190),
        "blue": (35, 118, 207),
        "orange": (232, 128, 31),
    },
    {
        "background": (255, 253, 247),
        "edge": (38, 43, 51),
        "neutral": (188, 181, 168),
        "green": (18, 145, 82),
        "purple": (133, 72, 178),
        "blue": (25, 105, 192),
        "orange": (219, 111, 24),
    },
)


def _adjacency(edges: list[list[int]] | tuple[tuple[int, int], ...]) -> list[set[int]]:
    adjacency = [set() for _ in range(GRID * GRID)]
    for left, right in edges:
        adjacency[int(left)].add(int(right))
        adjacency[int(right)].add(int(left))
    return adjacency


def _connected(adjacency: list[set[int]], start: int, goal: int, blocked: set[int]) -> bool:
    if start in blocked or goal in blocked:
        return False
    queue = deque([start])
    seen = {start}
    while queue:
        node = queue.popleft()
        if node == goal:
            return True
        for neighbor in adjacency[node]:
            if neighbor not in blocked and neighbor not in seen:
                seen.add(neighbor)
                queue.append(neighbor)
    return False


def _graph_is_connected(edges: list[list[int]]) -> bool:
    adjacency = _adjacency(edges)
    queue = deque([0])
    seen = {0}
    while queue:
        node = queue.popleft()
        for neighbor in adjacency[node]:
            if neighbor not in seen:
                seen.add(neighbor)
                queue.append(neighbor)
    return len(seen) == GRID * GRID


def _paired_routes_exist(edges: list[list[int]], terminals: dict[str, int]) -> bool:
    """Search green-purple paths and test blue-orange in each residual graph."""

    adjacency = _adjacency(edges)
    green = int(terminals["green"])
    purple = int(terminals["purple"])
    blue = int(terminals["blue"])
    orange = int(terminals["orange"])
    forbidden_on_first = {blue, orange}

    def search(node: int, path: list[int], used: set[int]) -> bool:
        if node == purple:
            return _connected(adjacency, blue, orange, set(path))
        for neighbor in sorted(adjacency[node]):
            if neighbor in used or neighbor in forbidden_on_first:
                continue
            used.add(neighbor)
            path.append(neighbor)
            if search(neighbor, path, used):
                return True
            path.pop()
            used.remove(neighbor)
        return False

    return search(green, [green], {green})


def _decision_from_fields(edges: list[list[int]], terminals: dict[str, int]) -> str:
    return "yes" if _paired_routes_exist(edges, terminals) else "no"


def _single_link_flip_exists(edges: list[list[int]], terminals: dict[str, int]) -> bool:
    base = _paired_routes_exist(edges, terminals)
    present = {tuple(sorted((int(left), int(right)))) for left, right in edges}
    for edge in POTENTIAL_EDGES:
        normalized = tuple(sorted(edge))
        changed = set(present)
        if normalized in changed:
            changed.remove(normalized)
        else:
            changed.add(normalized)
        candidate = [list(item) for item in sorted(changed)]
        if _paired_routes_exist(candidate, terminals) != base:
            return True
    return False


def _find_edges(rng: random.Random, terminals: dict[str, int], target: str) -> list[list[int]] | None:
    for _ in range(8_000):
        edges = [list(edge) for edge in sorted(rng.sample(POTENTIAL_EDGES, 18))]
        if not _graph_is_connected(edges):
            continue
        adjacency = _adjacency(edges)
        if not _connected(adjacency, terminals["green"], terminals["purple"], set()):
            continue
        if not _connected(adjacency, terminals["blue"], terminals["orange"], set()):
            continue
        if _decision_from_fields(edges, terminals) != target:
            continue
        if not _single_link_flip_exists(edges, terminals):
            continue
        return edges
    return None


def sample_scene(seed: int) -> dict:
    """Sample a class-paired, connected, fixed-ink scene.

    Even seeds request ``yes`` and odd seeds request ``no``.  Neighboring seeds
    share terminal positions and every style field; only their 18-link topology
    differs.  This makes color, position, size, palette, translation, spacing,
    and total link count exactly counterbalanced across answer classes.
    """

    integer_seed = int(seed)
    pair_seed = integer_seed // 2
    target = "yes" if integer_seed % 2 == 0 else "no"
    layout_rng = random.Random(7_919 + pair_seed * 104_729)
    for terminal_attempt in range(256):
        chosen = layout_rng.sample(range(GRID * GRID), 4)
        terminals = dict(zip(("green", "purple", "blue", "orange"), chosen))
        yes_rng = random.Random(11_003 + pair_seed * 1_000_003 + terminal_attempt * 97)
        no_rng = random.Random(19_999 + pair_seed * 1_000_003 + terminal_attempt * 193)
        yes_edges = _find_edges(yes_rng, terminals, "yes")
        if yes_edges is None:
            continue
        no_edges = _find_edges(no_rng, terminals, "no")
        if no_edges is None:
            continue
        edges = yes_edges if target == "yes" else no_edges
        return {
            "edges": edges,
            "terminals": terminals,
            "origin": [layout_rng.randint(92, 108), layout_rng.randint(92, 108)],
            "step": layout_rng.choice([96, 100, 104]),
            "canvas": [WIDTH, HEIGHT],
            "palette_variant": layout_rng.randrange(len(PALETTES)),
            "edge_width": layout_rng.choice([10, 11]),
            "node_radius": layout_rng.choice([16, 17]),
        }
    raise RuntimeError(f"could not sample requested class for seed {seed}")


def _node_position(node: int, origin: list[int], step: int) -> tuple[int, int]:
    row, col = divmod(int(node), GRID)
    return int(origin[0]) + col * int(step), int(origin[1]) + row * int(step)


def render(scene: dict) -> Image.Image:
    """Render links first, then all visible circular junctions."""

    width, height = (int(value) for value in scene["canvas"])
    palette = PALETTES[int(scene["palette_variant"])]
    origin = [int(value) for value in scene["origin"]]
    step = int(scene["step"])
    edge_width = int(scene["edge_width"])
    radius = int(scene["node_radius"])
    terminals = {str(key): int(value) for key, value in scene["terminals"].items()}
    edges = [[int(left), int(right)] for left, right in scene["edges"]]

    scale = 4
    image = Image.new("RGB", (width * scale, height * scale), palette["background"])
    draw = ImageDraw.Draw(image)
    for left, right in edges:
        p1 = _node_position(left, origin, step)
        p2 = _node_position(right, origin, step)
        draw.line(
            tuple(value * scale for point in (p1, p2) for value in point),
            fill=palette["edge"],
            width=edge_width * scale,
        )

    role_for_node = {node: role for role, node in terminals.items()}
    for node in range(GRID * GRID):
        cx, cy = _node_position(node, origin, step)
        fill = palette[role_for_node[node]] if node in role_for_node else palette["neutral"]
        box = (
            (cx - radius) * scale,
            (cy - radius) * scale,
            (cx + radius) * scale,
            (cy + radius) * scale,
        )
        draw.ellipse(box, fill=fill, outline=palette["edge"], width=3 * scale)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return _decision_from_fields(scene["edges"], scene["terminals"])


def margin(scene: dict) -> float:
    """Whole-link edit distance to the opposite class, capped above at two."""

    return 1.0 if _single_link_flip_exists(scene["edges"], scene["terminals"]) else 2.0


def is_quarantined(scene: dict) -> bool:
    """Thick complete links have no finite-resolution boundary cases."""

    _ = scene["edge_width"]
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reversing same-colored link draw order is pixel- and label-invariant."""

    twin = dict(scene)
    twin["edges"] = list(reversed(scene["edges"]))
    return [("reverse_link_draw_order", twin)]
