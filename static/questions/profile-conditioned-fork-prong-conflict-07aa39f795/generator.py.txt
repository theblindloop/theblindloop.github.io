"""Deterministic analytic scene arm and antialiased fork renderer."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS = 512
SUPERSAMPLE = 4
QUARANTINE_PX = 6.0
FORK_PALETTE = (
    (28, 75, 138),
    (139, 43, 57),
    (35, 112, 82),
    (92, 55, 135),
    (42, 91, 105),
)
BACKGROUNDS = ((250, 248, 242), (246, 249, 251), (250, 246, 249), (247, 250, 245))


def _style(seed: int, tight: bool = False) -> dict[str, Any]:
    rng = random.Random(int(seed))
    top_y = rng.randint(54, 88)
    prong_height = rng.randint(118, 158)
    shoulder_y = top_y + prong_height
    neck_y = shoulder_y + rng.randint(62, 88)
    bottom_y = rng.randint(432, 470)
    if tight:
        overall_width = 140
        total_prong_width = 110
    else:
        overall_width = rng.randint(154, 238)
        fraction = rng.uniform(0.61, 0.75)
        total_prong_width = int(round(overall_width * fraction))
    center_x = rng.randint(32 + overall_width // 2, CANVAS - 32 - overall_width // 2)
    return {
        "width": CANVAS,
        "height": CANVAS,
        "background_rgb": list(rng.choice(BACKGROUNDS)),
        "fork_rgb": list(rng.choice(FORK_PALETTE)),
        "center_x": center_x,
        "top_y": top_y,
        "shoulder_y": shoulder_y,
        "neck_y": neck_y,
        "bottom_y": bottom_y,
        "overall_width": overall_width,
        "total_prong_width": total_prong_width,
        "handle_width": rng.randint(34, 54),
        "head_depth": rng.randint(18, 28),
    }


def matched_scene(seed: int, prong_count: int, tight: bool = False) -> dict[str, Any]:
    if prong_count not in (3, 4, 5):
        raise ValueError("prong_count must be 3, 4, or 5")
    scene = _style(seed, tight=tight)
    scene["prong_count"] = int(prong_count)
    return scene


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one deterministic scene; consecutive residues balance all answers."""
    count = (3, 4, 5)[int(seed) % 3]
    return matched_scene(int(seed) * 104729 + 17, count)


def _geometry(scene: dict) -> tuple[float, float, float, float]:
    n = int(scene["prong_count"])
    width = float(scene["overall_width"])
    total = float(scene["total_prong_width"])
    prong_width = total / n
    gap = (width - total) / (n - 1)
    left = float(scene["center_x"]) - width / 2.0
    return left, prong_width, gap, width


def render(scene: dict) -> Image.Image:
    """Render one upright filled fork. Every declared scene field affects pixels."""
    scale = SUPERSAMPLE
    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(int(v) for v in scene["background_rgb"])
    foreground = tuple(int(v) for v in scene["fork_rgb"])
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)

    def box(values: tuple[float, float, float, float]) -> tuple[int, int, int, int]:
        return tuple(int(round(value * scale)) for value in values)

    left, prong_width, gap, overall_width = _geometry(scene)
    n = int(scene["prong_count"])
    top = float(scene["top_y"])
    shoulder = float(scene["shoulder_y"])
    head_depth = float(scene["head_depth"])
    center = float(scene["center_x"])
    neck_y = float(scene["neck_y"])
    bottom = float(scene["bottom_y"])
    handle_width = float(scene["handle_width"])

    # Rounded tips remain separated until they enter the shared solid shoulder.
    for index in range(n):
        x0 = left + index * (prong_width + gap)
        x1 = x0 + prong_width
        radius = max(2, int(round(min(prong_width * 0.44, 10.0) * scale)))
        draw.rounded_rectangle(box((x0, top, x1, shoulder + head_depth)), radius=radius, fill=foreground)

    right = left + overall_width
    draw.rectangle(box((left, shoulder, right, shoulder + head_depth)), fill=foreground)
    draw.polygon(
        [
            (int(round(left * scale)), int(round((shoulder + head_depth) * scale))),
            (int(round(right * scale)), int(round((shoulder + head_depth) * scale))),
            (int(round((center + handle_width / 2) * scale)), int(round(neck_y * scale))),
            (int(round((center - handle_width / 2) * scale)), int(round(neck_y * scale))),
        ],
        fill=foreground,
    )
    handle_radius = int(round(handle_width * 0.48 * scale))
    draw.rounded_rectangle(
        box((center - handle_width / 2, neck_y - 5, center + handle_width / 2, bottom)),
        radius=handle_radius,
        fill=foreground,
    )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["prong_count"])


def margin(scene: dict) -> float:
    _left, prong_width, gap, _width = _geometry(scene)
    return float(min(prong_width, gap))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # No pixel-identical latent aliases are intentionally introduced.
    return []
