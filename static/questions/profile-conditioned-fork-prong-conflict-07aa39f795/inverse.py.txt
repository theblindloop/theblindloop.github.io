"""Independent PNG-only inverse for fork upper-branch topology.

This module deliberately contains no imports from the scene, rendering, prompt,
generation, verification, or analytic-gold arms.
"""

from __future__ import annotations

from dataclasses import dataclass

import numpy as np
from PIL import Image


class PixelEvidenceError(ValueError):
    """The raster does not contain a measurable fork silhouette."""


@dataclass(frozen=True)
class Measurement:
    count: int
    min_gap_px: float
    support_rows: int
    foreground_bbox: tuple[int, int, int, int]


def _runs(row: np.ndarray, minimum: int) -> list[tuple[int, int]]:
    padded = np.pad(row.astype(np.int8), (1, 1))
    changes = np.diff(padded)
    starts = np.flatnonzero(changes == 1)
    ends = np.flatnonzero(changes == -1)
    return [(int(a), int(b)) for a, b in zip(starts, ends) if int(b - a) >= minimum]


def measure(image: Image.Image) -> Measurement:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    if arr.ndim != 3 or min(arr.shape[:2]) < 64:
        raise PixelEvidenceError("image is too small")
    h, w = arr.shape[:2]
    patch = max(4, min(h, w) // 32)
    corners = np.concatenate(
        [
            arr[:patch, :patch].reshape(-1, 3),
            arr[:patch, -patch:].reshape(-1, 3),
            arr[-patch:, :patch].reshape(-1, 3),
            arr[-patch:, -patch:].reshape(-1, 3),
        ],
        axis=0,
    )
    background = np.median(corners, axis=0)
    distance = np.linalg.norm(arr - background, axis=2)
    mask = distance > 55.0
    ys, xs = np.nonzero(mask)
    if len(xs) < 900:
        raise PixelEvidenceError("insufficient foreground ink")
    x0, x1 = int(xs.min()), int(xs.max()) + 1
    y0, y1 = int(ys.min()), int(ys.max()) + 1
    bw, bh = x1 - x0, y1 - y0
    if bw < 70 or bh < 150 or bh <= bw:
        raise PixelEvidenceError("foreground is not a supported upright fork")

    cropped = mask[y0:y1, x0:x1]
    minimum_run = max(4, int(round(bw * 0.035)))
    row_runs = [_runs(cropped[y], minimum_run) for y in range(cropped.shape[0])]
    row_counts = [len(items) for items in row_runs]
    candidates = sorted({value for value in row_counts if 3 <= value <= 5}, reverse=True)
    for count in candidates:
        best_start = -1
        best_length = 0
        start = None
        for index, value in enumerate(row_counts + [-1]):
            if value == count and start is None:
                start = index
            elif value != count and start is not None:
                if index - start > best_length:
                    best_start, best_length = start, index - start
                start = None
        required = max(8, int(round(bh * 0.035)))
        if best_length < required:
            continue
        gaps: list[int] = []
        for items in row_runs[best_start : best_start + best_length]:
            gaps.extend(items[k + 1][0] - items[k][1] for k in range(len(items) - 1))
        if not gaps:
            continue
        min_gap = float(min(gaps))
        if min_gap < 3.0:
            raise PixelEvidenceError("prong gaps are below pixel resolution")
        return Measurement(count, min_gap, best_length, (x0, y0, x1, y1))
    raise PixelEvidenceError("no stable three-to-five-prong upper topology")


def decision_from_image(image: Image.Image) -> str:
    return str(measure(image).count)
