"""Renderer and analytic arm for aggregate constellation distance."""

from __future__ import annotations

import math
import random
from PIL import Image, ImageDraw

WIDTH = HEIGHT = 512
DOT_RADIUS = 7
MARKER_RADIUS = 13
PALETTE = {"red": (224, 55, 66), "blue": (35, 112, 214)}
QUARANTINE_GAP = 70.0


def _distance_sums(dots: list[list[float]], markers: list[dict]) -> list[float]:
    return [
        sum(math.hypot(dot[0] - marker["center"][0], dot[1] - marker["center"][1]) for dot in dots)
        for marker in markers
    ]


def sample_scene(seed: int) -> dict:
    """Sample separated dots; each marker is closer to exactly half the dots."""
    rng = random.Random(int(seed))
    angle = rng.uniform(0.0, math.tau)
    separation = rng.uniform(125.0, 155.0)
    mx = rng.uniform(244.0, 268.0)
    my = rng.uniform(244.0, 268.0)
    ux, uy = math.cos(angle), math.sin(angle)
    centers = [
        [mx - ux * separation / 2, my - uy * separation / 2],
        [mx + ux * separation / 2, my + uy * separation / 2],
    ]

    for _ in range(20000):
        dots: list[list[float]] = []
        while len(dots) < 18:
            p = [rng.uniform(48, WIDTH - 48), rng.uniform(48, HEIGHT - 48)]
            if any(math.hypot(p[0] - q[0], p[1] - q[1]) < 27 for q in dots):
                continue
            if any(math.hypot(p[0] - c[0], p[1] - c[1]) < 30 for c in centers):
                continue
            dots.append(p)
        bare = [{"center": centers[0], "color": "red"}, {"center": centers[1], "color": "blue"}]
        sums = _distance_sums(dots, bare)
        closer0 = sum(
            math.hypot(p[0] - centers[0][0], p[1] - centers[0][1])
            < math.hypot(p[0] - centers[1][0], p[1] - centers[1][1])
            for p in dots
        )
        gap = abs(sums[0] - sums[1])
        if closer0 == 9 and 80.0 <= gap <= 260.0:
            preferred = 0 if sums[0] < sums[1] else 1
            preferred_color = "red" if int(seed) % 2 == 0 else "blue"
            other_color = "blue" if preferred_color == "red" else "red"
            colors = [other_color, other_color]
            colors[preferred] = preferred_color
            colors[1 - preferred] = other_color
            markers = [
                {"center": [round(c[0], 3), round(c[1], 3)], "color": colors[i]}
                for i, c in enumerate(centers)
            ]
            return {
                "dots": [[round(p[0], 3), round(p[1], 3)] for p in dots],
                "markers": markers,
            }
    raise RuntimeError("could not sample a clear, locally balanced constellation")


def render(scene: dict) -> Image.Image:
    scale = 4
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), (250, 248, 243))
    draw = ImageDraw.Draw(image)
    for x, y in scene["dots"]:
        box = ((x - DOT_RADIUS) * scale, (y - DOT_RADIUS) * scale,
               (x + DOT_RADIUS) * scale, (y + DOT_RADIUS) * scale)
        draw.ellipse(box, fill=(27, 29, 33))
    for marker in scene["markers"]:
        x, y = marker["center"]
        color = PALETTE[marker["color"]]
        box = ((x - MARKER_RADIUS) * scale, (y - MARKER_RADIUS) * scale,
               (x + MARKER_RADIUS) * scale, (y + MARKER_RADIUS) * scale)
        draw.ellipse(box, fill=color)
        inner = ((x - 5) * scale, (y - 5) * scale, (x + 5) * scale, (y + 5) * scale)
        draw.ellipse(inner, fill=(250, 248, 243))
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    sums = _distance_sums(scene["dots"], scene["markers"])
    return scene["markers"][0 if sums[0] < sums[1] else 1]["color"]


def margin(scene: dict) -> float:
    sums = _distance_sums(scene["dots"], scene["markers"])
    return abs(sums[0] - sums[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
