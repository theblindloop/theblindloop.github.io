"""Independent pixel-only inverse arm. This module imports no world code."""

from __future__ import annotations

import math
import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int) -> list[tuple[float, float, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[float, float, int]] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        xs: list[int] = []
        ys: list[int] = []
        while stack:
            cy, cx = stack.pop()
            xs.append(cx); ys.append(cy)
            for ny, nx in ((cy-1,cx),(cy+1,cx),(cy,cx-1),(cy,cx+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(xs) >= minimum:
            found.append((float(sum(xs))/len(xs), float(sum(ys))/len(ys), len(xs)))
    return found


def _color_center(arr: np.ndarray, target: tuple[int, int, int]) -> tuple[float, float]:
    delta = np.linalg.norm(arr.astype(np.float32) - np.array(target, dtype=np.float32), axis=2)
    comps = _components(delta < 65.0, 150)
    if len(comps) != 1:
        raise ValueError(f"expected one colored marker, found {len(comps)}")
    return comps[0][0], comps[0][1]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    red = _color_center(arr, (224, 55, 66))
    blue = _color_center(arr, (35, 112, 214))
    darkness = arr.astype(np.int16).mean(axis=2)
    neutrality = arr.max(axis=2).astype(np.int16) - arr.min(axis=2).astype(np.int16)
    dots = _components((darkness < 85) & (neutrality < 35), 80)
    if len(dots) != 18:
        raise ValueError(f"expected 18 black dots, found {len(dots)}")
    centers = [(x, y) for x, y, _ in dots]
    red_sum = sum(math.hypot(x-red[0], y-red[1]) for x, y in centers)
    blue_sum = sum(math.hypot(x-blue[0], y-blue[1]) for x, y in centers)
    if abs(red_sum - blue_sum) < 35.0:
        raise ValueError("pixel decision is ambiguous")
    return "red" if red_sum < blue_sum else "blue"
