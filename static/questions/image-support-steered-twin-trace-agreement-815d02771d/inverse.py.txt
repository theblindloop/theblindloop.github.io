"""Independent image-only inverse for the twin trace task.

The inverse segments visible palette classes, reconstructs a small bipartite
connectivity graph from blue starts through navy stroke components to orange
terminal components, classifies terminal shapes from their raster occupancy,
and compares the two recovered attributes. It receives no scene data.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    pass


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    points = {tuple(map(int, point)) for point in np.argwhere(mask)}
    output: list[np.ndarray] = []
    while points:
        start = points.pop()
        stack = [start]
        component = [start]
        while stack:
            y, x = stack.pop()
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    neighbor = (y + dy, x + dx)
                    if neighbor in points:
                        points.remove(neighbor)
                        stack.append(neighbor)
                        component.append(neighbor)
        if len(component) >= minimum:
            output.append(np.asarray(component, dtype=np.float64))
    return output


def _distance(left: np.ndarray, right: np.ndarray) -> float:
    # Components are tiny; chunking avoids any full-canvas operation.
    best = float("inf")
    for start in range(0, len(left), 64):
        delta = left[start : start + 64, None, :] - right[None, :, :]
        best = min(best, float(np.sqrt(np.sum(delta * delta, axis=2)).min()))
    return best


def _centroid(component: np.ndarray) -> tuple[float, float]:
    y, x = component.mean(axis=0)
    return float(y), float(x)


def _shape(component: np.ndarray) -> str:
    ys = component[:, 0]
    xs = component[:, 1]
    height = int(ys.max() - ys.min() + 1)
    width = int(xs.max() - xs.min() + 1)
    fill = len(component) / float(height * width)
    # At this deliberately compact scale the high-chroma core of the circle
    # can itself have a square bounding box.  Its occupied area is nonetheless
    # stably smaller than the 10x10 square core (81 versus 100 pixels in clean
    # renders), while the fill check rejects grossly damaged components.
    if fill < 0.70 or len(component) < 55:
        raise OracleAbstention("terminal symbol is materially corrupted")
    return "circle" if len(component) < 91 else "square"


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[:, :, 0], array[:, :, 1], array[:, :, 2]
    blue_mask = (red < 105) & (green > 65) & (green < 175) & (blue > 125) & (blue - green > 22)
    navy_mask = (red < 85) & (green < 95) & (blue < 120) & (blue - green < 30)
    orange_mask = (red > 155) & (green > 55) & (green < 190) & (blue < 135) & (red - green > 45)

    starts = _components(blue_mask, 18)
    routes = _components(navy_mask, 24)
    terminals = _components(orange_mask, 30)
    if len(starts) != 2 or len(routes) != 2 or len(terminals) != 4:
        raise OracleAbstention(
            f"expected 2 starts, 2 routes, and 4 terminals; found {len(starts)}, {len(routes)}, {len(terminals)}"
        )

    starts.sort(key=lambda item: _centroid(item))
    available = set(range(len(routes)))
    recovered: list[str] = []
    association_margins: list[float] = []
    for start in starts:
        ranked_routes = sorted(((_distance(start, routes[index]), index) for index in available))
        if not ranked_routes or ranked_routes[0][0] > 5.0:
            raise OracleAbstention("a blue start has no unique adjacent navy route")
        route_index = ranked_routes[0][1]
        available.remove(route_index)
        ranked_terminals = sorted(
            (_distance(routes[route_index], terminal), index)
            for index, terminal in enumerate(terminals)
        )
        if ranked_terminals[0][0] > 5.0:
            raise OracleAbstention("a navy route reaches no orange terminal")
        gap = ranked_terminals[1][0] - ranked_terminals[0][0]
        if gap < 8.0:
            raise OracleAbstention("terminal association margin is ambiguous")
        association_margins.append(gap)
        recovered.append(_shape(terminals[ranked_terminals[0][1]]))

    if len(recovered) != 2 or min(association_margins) < 8.0:
        raise OracleAbstention("incomplete route-to-terminal graph")
    return "same" if recovered[0] == recovered[1] else "different"
