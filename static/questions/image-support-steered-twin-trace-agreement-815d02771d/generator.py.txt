"""Deterministic renderer for two compact, disconnected route islands."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

BG = (250, 249, 245)
NAVY = (27, 42, 58)
BLUE = (20, 118, 189)
ORANGE = (232, 119, 45)
PALE = (188, 194, 196)
FRAME = (220, 217, 207)
DECOR = (229, 226, 216)
QUARANTINE_MARGIN = 12.0


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    # Cell centers keep each 46-pixel island out of the neighboring 8x8
    # ablation halos while the pair shifts across the canvas.
    left_cell = rng.randrange(1, 5)
    row_cell = rng.randrange(1, 7)
    centers = [
        [left_cell * 100 + 50, row_cell * 100 + 50],
        [(left_cell + 2) * 100 + 50, row_cell * 100 + 50],
    ]
    first = "circle" if rng.randrange(2) == 0 else "square"
    same = int(seed) % 2 == 0
    second = first if same else ("square" if first == "circle" else "circle")
    chosen = [first, second]
    upper = [
        "circle" if rng.randrange(2) == 0 else "square",
        "circle" if rng.randrange(2) == 0 else "square",
    ]
    rotations = [rng.randrange(4), rng.randrange(4)]
    offsets = [8 + rng.randrange(4), 8 + rng.randrange(4)]
    decorations: list[list[int]] = []
    while len(decorations) < 10:
        point = [rng.randrange(35, 766), rng.randrange(35, 766)]
        if all(math.dist(point, center) > 105 for center in centers):
            decorations.append(point)
    return {
        "canvas_size": 800,
        "centers": centers,
        "chosen_symbols": chosen,
        "upper_symbols": upper,
        "rotations": rotations,
        "terminal_offsets": offsets,
        "decorations": decorations,
    }


def _rot(point: tuple[float, float], quarter_turns: int) -> tuple[float, float]:
    x, y = point
    for _ in range(int(quarter_turns) % 4):
        x, y = -y, x
    return x, y


def _xy(center: list[int], point: tuple[float, float], rotation: int, scale: int) -> tuple[int, int]:
    x, y = _rot(point, rotation)
    return (round((center[0] + x) * scale), round((center[1] + y) * scale))


def _draw_symbol(
    draw: ImageDraw.ImageDraw,
    center: list[int],
    point: tuple[float, float],
    rotation: int,
    symbol: str,
    scale: int,
) -> None:
    cx, cy = _xy(center, point, rotation, scale)
    radius = 5 * scale
    box = [cx - radius, cy - radius, cx + radius, cy + radius]
    if symbol == "circle":
        draw.ellipse(box, fill=ORANGE)
    elif symbol == "square":
        draw.rectangle(box, fill=ORANGE)
    else:
        raise ValueError(symbol)


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    center: list[int],
    chosen_symbol: str,
    upper_symbol: str,
    rotation: int,
    offset: int,
    scale: int,
) -> None:
    lower_symbol = "square" if upper_symbol == "circle" else "circle"
    chosen_y = -offset if chosen_symbol == upper_symbol else offset
    other_y = offset if chosen_y < 0 else -offset

    # Pale decoy path has equal status and symmetric geometry but no blue start.
    pale_points = [(-18, other_y), (-12, other_y), (-7, other_y), (-7, other_y / 2), (12, other_y)]
    draw.line(
        [_xy(center, point, rotation, scale) for point in pale_points],
        fill=PALE,
        width=3 * scale,
        joint="curve",
    )
    px, py = _xy(center, (-18, other_y), rotation, scale)
    pr = 3 * scale
    draw.ellipse([px - pr, py - pr, px + pr, py + pr], outline=PALE, width=2 * scale)

    dark_points = [(-12, 0), (-7, 0), (-7, chosen_y), (5, chosen_y), (5, chosen_y), (12, chosen_y)]
    draw.line(
        [_xy(center, point, rotation, scale) for point in dark_points],
        fill=NAVY,
        width=4 * scale,
        joint="curve",
    )
    sx, sy = _xy(center, (-18, 0), rotation, scale)
    sr = 5 * scale
    draw.ellipse([sx - sr, sy - sr, sx + sr, sy + sr], outline=BLUE, width=3 * scale)

    _draw_symbol(draw, center, (18, -offset), rotation, upper_symbol, scale)
    _draw_symbol(draw, center, (18, offset), rotation, lower_symbol, scale)


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas_size"])
    centers = scene["centers"]
    chosen_symbols = scene["chosen_symbols"]
    upper_symbols = scene["upper_symbols"]
    rotations = scene["rotations"]
    offsets = scene["terminal_offsets"]
    decorations = scene["decorations"]
    scale = 4
    image = Image.new("RGB", (size * scale, size * scale), BG)
    draw = ImageDraw.Draw(image)

    # A common faint inset visually groups the two independent islands. It is
    # deliberately ignored by the oracle and is not decisive evidence.
    x0 = min(point[0] for point in centers) - 42
    x1 = max(point[0] for point in centers) + 42
    y0 = centers[0][1] - 42
    y1 = centers[0][1] + 42
    dash = 10
    for x in range(x0, x1, dash * 2):
        draw.line([(x * scale, y0 * scale), (min(x + dash, x1) * scale, y0 * scale)], fill=FRAME, width=2 * scale)
        draw.line([(x * scale, y1 * scale), (min(x + dash, x1) * scale, y1 * scale)], fill=FRAME, width=2 * scale)
    for y in range(y0, y1, dash * 2):
        draw.line([(x0 * scale, y * scale), (x0 * scale, min(y + dash, y1) * scale)], fill=FRAME, width=2 * scale)
        draw.line([(x1 * scale, y * scale), (x1 * scale, min(y + dash, y1) * scale)], fill=FRAME, width=2 * scale)

    for x, y in decorations:
        r = 2 * scale
        draw.ellipse([x * scale - r, y * scale - r, x * scale + r, y * scale + r], fill=DECOR)
    for index in range(2):
        _draw_panel(
            draw,
            centers[index],
            chosen_symbols[index],
            upper_symbols[index],
            int(rotations[index]),
            int(offsets[index]),
            scale,
        )
    return image.resize((size, size), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    symbols = scene["chosen_symbols"]
    return "same" if symbols[0] == symbols[1] else "different"


def margin(scene: dict) -> float:
    return float(2 * min(int(value) for value in scene["terminal_offsets"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
