"""Deterministic renderer and analytic arm for Pocket Inside a Pocket.

Every scene contains three navy components.  The large component and exactly
one small component are square annuli; the other small component is solid.  In
the positive class the small annulus occupies the large annulus's white pocket.
In the negative class the equal-sized solid and annulus swap locations.  Thus
both classes have the same foreground-component count, hole count, and navy ink.
"""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

CANVAS = 512
LARGE_OUTER = 176
LARGE_WALL = 32
SMALL_OUTER = 56
SMALL_WALL = 16
EXTERNAL_GAP = 12
MIN_CLEARANCE = 10.0
PALETTES = ("#173f5f", "#214b67", "#284f6a")
DECORATIONS = ("#d8dee2", "#d2dadd", "#dde1e4")


def _box(center: tuple[int, int], size: int) -> tuple[int, int, int, int]:
    cx, cy = center
    half = size // 2
    return cx - half, cy - half, cx + half - 1, cy + half - 1


def _solid(draw: ImageDraw.ImageDraw, center: tuple[int, int], size: int, ink: str) -> None:
    draw.rectangle(_box(center, size), fill=ink)


def _ring(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    size: int,
    wall: int,
    ink: str,
) -> None:
    _solid(draw, center, size, ink)
    draw.rectangle(_box(center, size - 2 * wall), fill="#ffffff")


def _rotate(vector: tuple[int, int], quarter_turns: int) -> tuple[int, int]:
    x, y = vector
    for _ in range(quarter_turns % 4):
        x, y = -y, x
    return x, y


def _component_centers(
    center: tuple[int, int], orientation: int, outer_size: int, inner_size: int
) -> tuple[tuple[int, int], tuple[int, int], tuple[int, int]]:
    cx, cy = center
    distance = outer_size // 2 + inner_size // 2 + EXTERNAL_GAP
    dx, dy = _rotate((distance, 0), orientation)
    return (cx, cy), (cx, cy), (cx + dx, cy + dy)


def _sample_decorations(
    rng: random.Random,
    center: tuple[int, int],
    orientation: int,
    outer_size: int,
    inner_size: int,
) -> list[list[int]]:
    big, _inside, outside = _component_centers(center, orientation, outer_size, inner_size)
    decorations: list[list[int]] = []
    attempts = 0
    while len(decorations) < 8 and attempts < 600:
        attempts += 1
        x = rng.randrange(24, CANVAS - 24)
        y = rng.randrange(24, CANVAS - 24)
        if abs(x - big[0]) < outer_size // 2 + 34 and abs(y - big[1]) < outer_size // 2 + 34:
            continue
        if abs(x - outside[0]) < inner_size // 2 + 34 and abs(y - outside[1]) < inner_size // 2 + 34:
            continue
        if any(math.dist((x, y), (old_x, old_y)) < 26 for old_x, old_y, _ in decorations):
            continue
        decorations.append([x, y, rng.randrange(2)])
    return decorations


def sample_scene(seed: int) -> dict:
    seed = int(seed)
    pair_index = seed // 2
    rng = random.Random(pair_index ^ 0x6E57A11)
    nested = seed % 2 == 0
    orientation = pair_index % 4
    # Paired consecutive labels share pose, dimensions, palette and decoration
    # layout.  Only which equal-sized small component is hollow changes.
    cx = rng.randrange(220, 293, 12)
    cy = rng.randrange(220, 293, 12)
    center = (cx, cy)
    palette_variant = (pair_index // 4) % len(PALETTES)
    return {
        "canvas_size": [CANVAS, CANVAS],
        "nested": nested,
        "center": list(center),
        "orientation": orientation,
        "outer_size": LARGE_OUTER,
        "inner_size": SMALL_OUTER,
        "wall": [LARGE_WALL, SMALL_WALL],
        "palette_variant": palette_variant,
        "decorations": _sample_decorations(
            rng, center, orientation, LARGE_OUTER, SMALL_OUTER
        ),
    }


def render(scene: dict) -> Image.Image:
    canvas_size = tuple(int(value) for value in scene["canvas_size"])
    nested = bool(scene["nested"])
    center = tuple(int(value) for value in scene["center"])
    orientation = int(scene["orientation"])
    outer_size = int(scene["outer_size"])
    inner_size = int(scene["inner_size"])
    large_wall, small_wall = (int(value) for value in scene["wall"])
    palette_variant = int(scene["palette_variant"])
    decorations = list(scene["decorations"])
    ink = PALETTES[palette_variant % len(PALETTES)]
    decoration_color = DECORATIONS[palette_variant % len(DECORATIONS)]

    image = Image.new("RGB", canvas_size, "#ffffff")
    draw = ImageDraw.Draw(image)
    for x, y, kind in decorations:
        x, y = int(x), int(y)
        if int(kind) == 0:
            draw.rectangle((x - 5, y - 5, x + 4, y + 4), outline=decoration_color, width=2)
        else:
            draw.rectangle((x - 1, y - 6, x, y + 5), fill=decoration_color)
            draw.rectangle((x - 6, y - 1, x + 5, y), fill=decoration_color)

    big, inside, outside = _component_centers(center, orientation, outer_size, inner_size)
    _ring(draw, big, outer_size, large_wall, ink)
    if nested:
        _ring(draw, inside, inner_size, small_wall, ink)
        _solid(draw, outside, inner_size, ink)
    else:
        _solid(draw, inside, inner_size, ink)
        _ring(draw, outside, inner_size, small_wall, ink)
    return image


def analytic_gold(scene: dict) -> str:
    return "yes" if bool(scene["nested"]) else "no"


def margin(scene: dict) -> float:
    canvas_width, canvas_height = (int(value) for value in scene["canvas_size"])
    center = tuple(int(value) for value in scene["center"])
    orientation = int(scene["orientation"])
    outer_size = int(scene["outer_size"])
    inner_size = int(scene["inner_size"])
    large_wall, small_wall = (int(value) for value in scene["wall"])
    big, _inside, outside = _component_centers(center, orientation, outer_size, inner_size)
    inner_clearance = (outer_size - 2 * large_wall - inner_size) / 2.0
    external_clearance = float(EXTERNAL_GAP)
    hole_clearance = (inner_size - 2 * small_wall) / 2.0
    boxes = (_box(big, outer_size), _box(outside, inner_size))
    border_clearance = min(
        min(x0, y0, canvas_width - 1 - x1, canvas_height - 1 - y1)
        for x0, y0, x1, y1 in boxes
    )
    return float(
        min(
            large_wall,
            small_wall,
            inner_clearance,
            external_clearance,
            hole_clearance,
            border_clearance,
        )
    )


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < MIN_CLEARANCE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["decorations"] = list(reversed(twin["decorations"]))
    return [("reverse_disjoint_decoration_order", twin)]
