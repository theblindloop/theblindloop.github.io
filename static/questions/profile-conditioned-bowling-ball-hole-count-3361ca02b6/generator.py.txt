"""Renderer and analytic labels for the bowling-ball hole-count world."""

from __future__ import annotations

import math
import random
from PIL import Image, ImageDraw

RENDERER_VERSION = "bowling_ball_hole_count-renderer-0.1.0"
WIDTH = 640
HEIGHT = 512
SUPERSAMPLE = 4
MIN_SEPARATION = 8.0
SAFE_SEPARATION = 24.0
BACKGROUND = (242, 239, 231)
BALL_COLORS = (
    (28, 49, 79),
    (71, 35, 62),
    (32, 76, 67),
    (74, 47, 92),
    (50, 53, 60),
)


def _as_tuple(color) -> tuple[int, int, int]:
    return tuple(int(value) for value in color)


def _hole_clearance(scene: dict) -> float:
    """Smallest dark-pixel clearance between an opening and its constraints."""

    cx = float(scene["ball_cx"])
    cy = float(scene["ball_cy"])
    radius = float(scene["ball_radius"])
    holes = scene["holes"]
    values: list[float] = []
    for hole in holes:
        distance = math.hypot(float(hole["x"]) - cx, float(hole["y"]) - cy)
        values.append(radius - distance - float(hole["r"]))
    for index, left in enumerate(holes):
        for right in holes[index + 1 :]:
            distance = math.hypot(
                float(left["x"]) - float(right["x"]),
                float(left["y"]) - float(right["y"]),
            )
            values.append(distance - float(left["r"]) - float(right["r"]))
    return min(values) if values else 0.0


def sample_scene(seed: int) -> dict:
    """Sample a balanced, fully visible ball and a count of enclosed openings.

    The count cycles with the public scene seed, while all placement, color,
    radius, hole size, and hole order choices come from an independent RNG.
    A small subset of seeds deliberately approaches, but does not cross, the
    connected-component boundary and is quarantined by ``is_quarantined``.
    """

    rng = random.Random(int(seed))
    count = 1 + (int(seed) % 4)
    width, height = WIDTH, HEIGHT
    ball_radius = rng.uniform(142.0, 168.0)
    ball_cx = rng.uniform(ball_radius + 38.0, width - ball_radius - 38.0)
    ball_cy = rng.uniform(ball_radius + 34.0, height - ball_radius - 34.0)
    ball_color = BALL_COLORS[rng.randrange(len(BALL_COLORS))]

    # Keep total removed light area roughly constant across counts so total
    # dark ink is not a reliable count cue. The individual radii still jitter.
    target_sum_sq = 3.0 * 30.0 * 30.0 * rng.uniform(0.90, 1.10)
    near_boundary = int(seed) % 13 == 0
    required_clearance = 8.5 if near_boundary else SAFE_SEPARATION

    for _ in range(12000):
        radii = []
        for _hole_index in range(count):
            base = math.sqrt(target_sum_sq / count)
            radii.append(base * rng.uniform(0.92, 1.08))
        scale = math.sqrt(target_sum_sq / sum(value * value for value in radii))
        radii = [value * scale for value in radii]
        holes: list[dict[str, float]] = []
        for radius in radii:
            placed = False
            for _placement in range(250):
                angle = rng.uniform(0.0, 2.0 * math.pi)
                max_distance = ball_radius - radius - required_clearance
                distance = math.sqrt(rng.uniform(0.0, max_distance * max_distance))
                x = ball_cx + distance * math.cos(angle)
                y = ball_cy + distance * math.sin(angle)
                candidate = {"x": x, "y": y, "r": radius}
                if all(
                    math.hypot(x - other["x"], y - other["y"])
                    >= radius + other["r"] + required_clearance
                    for other in holes
                ):
                    holes.append(candidate)
                    placed = True
                    break
            if not placed:
                break
        if len(holes) == count and _hole_clearance(
            {
                "ball_cx": ball_cx,
                "ball_cy": ball_cy,
                "ball_radius": ball_radius,
                "holes": holes,
            }
        ) >= required_clearance - 1e-6:
            if near_boundary:
                # Deliberately place one opening close to the silhouette for
                # gallery/quarantine coverage. The dark ring remains several
                # pixels wide, so the pixel inverse still has a determinate
                # answer even though the scene is excluded from headline
                # claims.
                for _edge_try in range(80):
                    edge_angle = rng.uniform(0.0, 2.0 * math.pi)
                    edge_clearance = 6.0
                    edge_distance = ball_radius - holes[0]["r"] - edge_clearance
                    edge_x = ball_cx + edge_distance * math.cos(edge_angle)
                    edge_y = ball_cy + edge_distance * math.sin(edge_angle)
                    candidate = {"x": edge_x, "y": edge_y, "r": holes[0]["r"]}
                    if all(
                        math.hypot(edge_x - other["x"], edge_y - other["y"])
                        >= candidate["r"] + other["r"] + 5.0
                        for other in holes[1:]
                    ):
                        holes[0] = candidate
                        break
                if _hole_clearance(
                    {
                        "ball_cx": ball_cx,
                        "ball_cy": ball_cy,
                        "ball_radius": ball_radius,
                        "holes": holes,
                    }
                ) < 4.5:
                    continue
            # Store JSON-native lists for deterministic scene records.
            return {
                "width": width,
                "height": height,
                "background": list(BACKGROUND),
                "ball_color": list(ball_color),
                "ball_cx": round(ball_cx, 4),
                "ball_cy": round(ball_cy, 4),
                "ball_radius": round(ball_radius, 4),
                "holes": [
                    {"x": round(hole["x"], 4), "y": round(hole["y"], 4), "r": round(hole["r"], 4)}
                    for hole in holes
                ],
            }
    raise RuntimeError(f"could not place {count} holes for seed {seed}")


def render(scene: dict) -> Image.Image:
    """Render the ball and cut out the light openings with antialiased edges."""

    ss = SUPERSAMPLE
    background = _as_tuple(scene["background"])
    image = Image.new("RGB", (int(scene["width"]) * ss, int(scene["height"]) * ss), background)
    draw = ImageDraw.Draw(image)

    def box(cx: float, cy: float, radius: float) -> tuple[int, int, int, int]:
        return (
            round((cx - radius) * ss),
            round((cy - radius) * ss),
            round((cx + radius) * ss),
            round((cy + radius) * ss),
        )

    draw.ellipse(
        box(float(scene["ball_cx"]), float(scene["ball_cy"]), float(scene["ball_radius"])),
        fill=_as_tuple(scene["ball_color"]),
    )
    for hole in scene["holes"]:
        draw.ellipse(box(float(hole["x"]), float(hole["y"]), float(hole["r"])), fill=background)
    return image.resize((int(scene["width"]), int(scene["height"])), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """The visible truth is exactly the number of hole records rendered."""

    return str(len(scene["holes"]))


def margin(scene: dict) -> float:
    """Return the smallest visible clearance protecting separate openings."""

    return round(_hole_clearance(scene), 6)


def is_quarantined(scene: dict) -> bool:
    """Exclude openings whose dark separating boundary is near pixel ambiguity."""

    return margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse same-colored hole records: a nontrivial render-preserving alias."""

    if len(scene["holes"]) < 2:
        return []
    twin = dict(scene)
    twin["holes"] = [dict(hole) for hole in reversed(scene["holes"])]
    return [("reverse_hole_record_order", twin)]
