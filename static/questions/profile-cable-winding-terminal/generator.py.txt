"""Renderer for the clockwise-winding cable world."""

from __future__ import annotations

import itertools
import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 400
BACKGROUND = (248, 247, 243)
CABLE = (47, 55, 64)
HUB = (12, 15, 18)
START = (24, 157, 91)
PALETTE = {
    "orange": (230, 126, 34),
    "purple": (137, 78, 193),
    "blue": (39, 112, 205),
}
QUESTION = (
    "Starting at each ring, follow its dark cable to the colored square. "
    "Which cable makes exactly one complete clockwise turn around the black hub? "
    "Answer orange, purple, or blue."
)


def _polyline(cx: float, cy: float, radius0: float, start: float, deltas: Iterable[float]) -> list[list[float]]:
    deltas = list(deltas)
    points: list[list[float]] = []
    for index, delta in enumerate(deltas):
        t = index / (len(deltas) - 1)
        radius = radius0 + 20.0 * t
        angle = start + delta
        points.append([round(cx + radius * math.cos(angle), 3), round(cy + radius * math.sin(angle), 3)])
    return points


def _role_deltas(role: str, decoy_net: float, samples: int = 161) -> list[float]:
    if role == "clockwise":
        return [-2.0 * math.pi * i / (samples - 1) for i in range(samples)]
    if role == "counterclockwise":
        return [2.0 * math.pi * i / (samples - 1) for i in range(samples)]
    # The decoy reverses once. Its total angular travel is one revolution, like
    # both monotone spirals, but its signed displacement is not one revolution.
    outward = (2.0 * math.pi + decoy_net) / 2.0
    backward = (2.0 * math.pi - decoy_net) / 2.0
    result = []
    for i in range(samples):
        t = i / (samples - 1)
        if t <= 0.5:
            result.append(outward * (t / 0.5))
        else:
            result.append(outward - backward * ((t - 0.5) / 0.5))
    return result


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    cx = 200 + rng.randint(-9, 9)
    cy = 200 + rng.randint(-9, 9)
    rotation = rng.uniform(0.0, 2.0 * math.pi)
    roles = list(itertools.permutations(("clockwise", "counterclockwise", "reversal")))[(int(seed) // 3) % 6]
    target_color = ("orange", "purple", "blue")[int(seed) % 3]
    remaining = [color for color in ("orange", "purple", "blue") if color != target_color]
    if (int(seed) // 3) % 2:
        remaining.reverse()
    role_colors = {"clockwise": target_color, "counterclockwise": remaining[0], "reversal": remaining[1]}
    decoy_net = (-4.25, -3.75, -3.25, -2.75, 0.4, 2.8)[int(seed) % 6]
    radii = (50.0, 94.0, 138.0)
    paths = []
    terminal_colors = []
    for band, role in enumerate(roles):
        start = rotation + band * 2.0 * math.pi / 3.0 + rng.uniform(-0.08, 0.08)
        deltas = _role_deltas(role, decoy_net)
        paths.append(_polyline(cx, cy, radii[band], start, deltas))
        terminal_colors.append(role_colors[role])
    return {
        "paths": paths,
        "terminal_colors": terminal_colors,
        "hub": [cx, cy],
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
    }


def _signed_winding(points: list[list[float]], hub: list[float]) -> float:
    angles = [math.atan2(point[1] - hub[1], point[0] - hub[0]) for point in points]
    total = 0.0
    for first, second in zip(angles, angles[1:]):
        step = second - first
        while step <= -math.pi:
            step += 2.0 * math.pi
        while step > math.pi:
            step -= 2.0 * math.pi
        total += step
    return total


def _errors(scene: dict) -> list[float]:
    hub = scene["hub"]
    return [abs(_signed_winding(points, hub) + 2.0 * math.pi) for points in scene["paths"]]


def analytic_gold(scene: dict) -> str:
    errors = _errors(scene)
    winner = min(range(len(errors)), key=errors.__getitem__)
    return scene["terminal_colors"][winner]


def margin(scene: dict) -> float:
    errors = sorted(_errors(scene))
    return float(errors[1] - errors[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _scale_points(points: list[list[float]], scale: int) -> list[tuple[int, int]]:
    return [(round(x * scale), round(y * scale)) for x, y in points]


def render(scene: dict) -> Image.Image:
    scale = 2
    width, height = int(scene["width"]), int(scene["height"])
    background = tuple(scene["background"])
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    hub = scene["hub"]
    # Draw cables first, then endpoint markers. Rounded joints make every trace
    # a single robust component without exposing any latent path data.
    for points in scene["paths"]:
        scaled = _scale_points(points, scale)
        draw.line(scaled, fill=CABLE, width=7 * scale, joint="curve")
        for x, y in scaled[::8]:
            draw.ellipse((x - 3 * scale, y - 3 * scale, x + 3 * scale, y + 3 * scale), fill=CABLE)
    hx, hy = hub
    draw.ellipse(((hx - 23) * scale, (hy - 23) * scale, (hx + 23) * scale, (hy + 23) * scale), fill=HUB)
    # The shared green color intentionally prevents a cheap answer cue; each
    # ring merely establishes the direction in which that cable is followed.
    for points, color_name in zip(scene["paths"], scene["terminal_colors"]):
        sx, sy = points[0]
        ex, ey = points[-1]
        draw.ellipse(((sx - 10) * scale, (sy - 10) * scale, (sx + 10) * scale, (sy + 10) * scale), fill=background, outline=START, width=5 * scale)
        color = PALETTE[color_name]
        draw.rounded_rectangle(((ex - 10) * scale, (ey - 10) * scale, (ex + 10) * scale, (ey + 10) * scale), radius=2 * scale, fill=color, outline=HUB, width=2 * scale)
    return image.resize((width, height), Image.Resampling.LANCZOS)
