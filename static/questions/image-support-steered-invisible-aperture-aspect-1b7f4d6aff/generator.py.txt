"""Deterministic latent sampler and renderer for the invisible aperture task."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
SUPERSAMPLE = 4
QUARANTINE_PX = 20.0

PALETTES = (
    ("#173f5f", "#f7f3e8"),
    ("#6b2d5c", "#f6f1e9"),
    ("#155e63", "#f8f1df"),
    ("#7a341f", "#f4f0e8"),
)


def sample_scene(seed: int) -> dict:
    """Sample balanced scenes; consecutive integer seeds alternate the answer."""

    rng = random.Random(int(seed))
    center_x = rng.randint(246, 266)
    center_y = rng.randint(246, 266)
    short_span = rng.randint(226, 252)
    # Both answer classes receive near-boundary examples (seed residues 0 and 1).
    if int(seed) % 8 in (0, 1):
        delta = rng.randint(14, 18)
    else:
        delta = rng.randint(58, 94)
    long_span = short_span + delta
    if int(seed) % 2 == 0:
        gap_width, gap_height = long_span, short_span
    else:
        gap_width, gap_height = short_span, long_span
    pointer_depth = rng.randint(58, 72)
    pointer_half_base = rng.randint(22, 29)
    foreground, background = PALETTES[rng.randrange(len(PALETTES))]
    return {
        "center_x": center_x,
        "center_y": center_y,
        "gap_width": gap_width,
        "gap_height": gap_height,
        "pointer_depth": pointer_depth,
        "pointer_half_base": pointer_half_base,
        "foreground": foreground,
        "background": background,
    }


def _triangle(tip: tuple[float, float], outward: tuple[int, int], depth: float, half: float):
    ox, oy = outward
    scale = 1.0 / math.sqrt(2.0)
    ux, uy = ox * scale, oy * scale
    px, py = -uy, ux
    bx, by = tip[0] + depth * ux, tip[1] + depth * uy
    return [tip, (bx + half * px, by + half * py), (bx - half * px, by - half * py)]


def render(scene: dict) -> Image.Image:
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    gap_width = float(scene["gap_width"])
    gap_height = float(scene["gap_height"])
    depth = float(scene["pointer_depth"])
    half = float(scene["pointer_half_base"])
    foreground = str(scene["foreground"])
    background = str(scene["background"])

    left, right = cx - gap_width / 2.0, cx + gap_width / 2.0
    top, bottom = cy - gap_height / 2.0, cy + gap_height / 2.0
    tips_and_directions = (
        ((left, top), (-1, -1)),
        ((right, top), (1, -1)),
        ((left, bottom), (-1, 1)),
        ((right, bottom), (1, 1)),
    )

    image = Image.new("RGB", (WIDTH * SUPERSAMPLE, HEIGHT * SUPERSAMPLE), background)
    draw = ImageDraw.Draw(image)
    for tip, outward in tips_and_directions:
        polygon = _triangle(tip, outward, depth, half)
        scaled = [(round(x * SUPERSAMPLE), round(y * SUPERSAMPLE)) for x, y in polygon]
        draw.polygon(scaled, fill=foreground)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "wider" if float(scene["gap_width"]) > float(scene["gap_height"]) else "taller"


def margin(scene: dict) -> float:
    return abs(float(scene["gap_width"]) - float(scene["gap_height"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
