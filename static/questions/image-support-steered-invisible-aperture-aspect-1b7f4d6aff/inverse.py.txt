"""Independent pixel-only inverse arm for the invisible aperture task."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    """Raised when four intact detached pointers cannot be recovered."""


def _foreground_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    border = np.concatenate(
        (array[:4].reshape(-1, 3), array[-4:].reshape(-1, 3),
         array[:, :4].reshape(-1, 3), array[:, -4:].reshape(-1, 3)),
        axis=0,
    )
    background = np.median(border, axis=0)
    distance = np.linalg.norm(array - background, axis=2)
    return distance > 45.0


def _components(mask: np.ndarray) -> list[np.ndarray]:
    points = {tuple(map(int, point)) for point in np.argwhere(mask)}
    groups: list[np.ndarray] = []
    while points:
        start = points.pop()
        queue = deque([start])
        group = [start]
        while queue:
            y, x = queue.popleft()
            for neighbor in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if neighbor in points:
                    points.remove(neighbor)
                    queue.append(neighbor)
                    group.append(neighbor)
        if len(group) >= 40:
            groups.append(np.asarray(group, dtype=np.float64))
    return groups


def _recover_spans(image: Image.Image) -> tuple[float, float]:
    groups = _components(_foreground_mask(image))
    if len(groups) != 4:
        raise OracleAbstention(f"expected four pointer components, recovered {len(groups)}")
    sizes = np.asarray([len(group) for group in groups], dtype=np.float64)
    if sizes.min() < 900 or sizes.min() / sizes.max() < 0.82:
        raise OracleAbstention("a pointer is missing or materially clipped")

    groups.sort(key=lambda group: float(group[:, 0].mean()))
    top = sorted(groups[:2], key=lambda group: float(group[:, 1].mean()))
    bottom = sorted(groups[2:], key=lambda group: float(group[:, 1].mean()))
    tl, tr = top
    bl, br = bottom

    # Arrays are (y, x). For each corner, its inward point is the appropriate
    # pair of extrema. Median over the outermost two raster rows/columns reduces
    # antialias sensitivity without consulting renderer coordinates.
    left_x = (tl[:, 1].max() + bl[:, 1].max()) / 2.0
    right_x = (tr[:, 1].min() + br[:, 1].min()) / 2.0
    top_y = (tl[:, 0].max() + tr[:, 0].max()) / 2.0
    bottom_y = (bl[:, 0].min() + br[:, 0].min()) / 2.0
    width = right_x - left_x
    height = bottom_y - top_y
    if min(width, height) < 80:
        raise OracleAbstention("recovered aperture is implausibly small")
    return float(width), float(height)


def decision_from_image(image: Image.Image) -> str:
    width, height = _recover_spans(image)
    if abs(width - height) < 4.0:
        raise OracleAbstention("pixel evidence is too close to the square boundary")
    return "wider" if width > height else "taller"
