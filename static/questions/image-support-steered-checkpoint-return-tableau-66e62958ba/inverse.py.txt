"""Independent image-only inverse arm for Checkpoint Return Tableau."""

from __future__ import annotations

import numpy as np
from PIL import Image

_ON = np.array((38, 101, 154), dtype=np.float32)
_OFF = np.array((235, 181, 62), dtype=np.float32)
_MAX_COLOR_DISTANCE = 38.0


def _recover_matrix(image: Image.Image) -> list[list[int]]:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    if arr.ndim != 3 or arr.shape[0] < 200 or arr.shape[1] < 200:
        raise ValueError("image is too small")
    d_on = np.linalg.norm(arr - _ON, axis=2)
    d_off = np.linalg.norm(arr - _OFF, axis=2)
    board_mask = np.minimum(d_on, d_off) < _MAX_COLOR_DISTANCE
    ys, xs = np.nonzero(board_mask)
    if len(xs) < 100000:
        raise ValueError("filled tableau is missing or too sparse")
    x0, x1 = int(xs.min()), int(xs.max()) + 1
    y0, y1 = int(ys.min()), int(ys.max()) + 1
    width, height = x1 - x0, y1 - y0
    if width < arr.shape[1] * 0.72 or height < arr.shape[0] * 0.60:
        raise ValueError("tableau extent is incomplete")
    if not 1.30 < width / height < 1.75:
        raise ValueError("tableau proportions are malformed")

    states: list[list[int]] = []
    for row in range(4):
        values = []
        for col in range(4):
            # Sample the broad interior while excluding grid bars and the central glyph.
            xa = int(x0 + width * (col + 0.13) / 4)
            xb = int(x0 + width * (col + 0.87) / 4)
            ya = int(y0 + height * (row + 0.13) / 4)
            yb = int(y0 + height * (row + 0.87) / 4)
            patch_on = d_on[ya:yb, xa:xb]
            patch_off = d_off[ya:yb, xa:xb]
            near_on = patch_on < _MAX_COLOR_DISTANCE
            near_off = patch_off < _MAX_COLOR_DISTANCE
            colored_fraction = float(np.mean(near_on | near_off))
            on_fraction = float(np.mean(near_on))
            off_fraction = float(np.mean(near_off))
            if colored_fraction < 0.72:
                raise ValueError("a tableau cell is erased or impure")
            if abs(on_fraction - off_fraction) < 0.35:
                raise ValueError("a tableau cell has ambiguous state")
            values.append(1 if on_fraction > off_fraction else 0)
        states.append(values)
    return states


def decision_from_image(image: Image.Image) -> str:
    states = _recover_matrix(image)
    counts = [sum(row[col] == row[0] for row in states) for col in range(1, 4)]
    best = max(counts)
    if counts.count(best) != 1:
        raise ValueError("return-count decision is tied")
    return str(counts.index(best) + 1)
