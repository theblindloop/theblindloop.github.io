"""Analytic sampler and renderer for Checkpoint Return Tableau."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 640, 512
BACKGROUND = (248, 246, 238)
INK = (25, 32, 45)
ON = (38, 101, 154)
OFF = (235, 181, 62)
PAPER = (255, 253, 247)


def _font(size: int):
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _matrix(scene: dict) -> list[list[int]]:
    states = scene["states"]
    if not isinstance(states, list) or len(states) != 4:
        raise ValueError("states must have four lamp rows")
    matrix = []
    for row in states:
        if not isinstance(row, list) or len(row) != 4:
            raise ValueError("each lamp row must have START and three checkpoints")
        values = [int(value) for value in row]
        if any(value not in (0, 1) for value in values):
            raise ValueError("states must be binary")
        matrix.append(values)
    return matrix


def sample_scene(seed: int) -> dict:
    """Choose a balanced winning checkpoint from seed modulo three."""
    rng = random.Random(int(seed))
    winner = int(seed) % 3
    count_templates = ([4, 2, 1], [2, 4, 1], [1, 2, 4], [3, 2, 0], [2, 3, 0], [0, 2, 3])
    eligible = [item for item in count_templates if item.index(max(item)) == winner]
    counts = list(rng.choice(eligible))
    start = [rng.randrange(2) for _ in range(4)]
    # Avoid a monochrome START column, reducing cheap total-ink regularities.
    if len(set(start)) == 1:
        start[rng.randrange(4)] ^= 1
    columns = [start]
    for matches in counts:
        matching_rows = set(rng.sample(range(4), matches))
        columns.append([start[row] if row in matching_rows else 1 - start[row] for row in range(4)])
    states = [[columns[col][row] for col in range(4)] for row in range(4)]
    return {"states": states}


def _return_counts(scene: dict) -> list[int]:
    states = _matrix(scene)
    return [sum(row[col] == row[0] for row in states) for col in range(1, 4)]


def analytic_gold(scene: dict) -> str:
    counts = _return_counts(scene)
    if counts.count(max(counts)) != 1:
        raise ValueError("highest return count is tied")
    return str(counts.index(max(counts)) + 1)


def margin(scene: dict) -> float:
    counts = sorted(_return_counts(scene), reverse=True)
    return float(counts[0] - counts[1])


def is_quarantined(scene: dict) -> bool:
    try:
        counts = _return_counts(scene)
    except (TypeError, ValueError, KeyError):
        return True
    return counts.count(max(counts)) != 1


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["states"] = [[bool(value) for value in row] for row in _matrix(scene)]
    return [("boolean_state_encoding", twin)]


def render(scene: dict) -> Image.Image:
    states = _matrix(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(22)
    header_font = _font(20)
    small_font = _font(15)

    title = "RETURN CHECKPOINTS"
    box = draw.textbbox((0, 0), title, font=title_font)
    draw.text(((WIDTH - (box[2] - box[0])) / 2, 12), title, fill=INK, font=title_font)

    bx0, by0, bx1, by1 = 36, 100, 604, 476
    cell_w, cell_h = (bx1 - bx0) // 4, (by1 - by0) // 4
    headers = ("START", "1", "2", "3")
    for col, text in enumerate(headers):
        center_x = bx0 + col * cell_w + cell_w / 2
        tbox = draw.textbbox((0, 0), text, font=header_font)
        draw.rounded_rectangle(
            (center_x - 49, 52, center_x + 49, 88), radius=10,
            fill=PAPER, outline=INK, width=3,
        )
        draw.text((center_x - (tbox[2] - tbox[0]) / 2, 57), text, fill=INK, font=header_font)

    draw.rectangle((bx0, by0, bx1, by1), fill=INK)
    for row in range(4):
        for col in range(4):
            x0 = bx0 + col * cell_w + 6
            y0 = by0 + row * cell_h + 6
            x1 = bx0 + (col + 1) * cell_w - 6
            y1 = by0 + (row + 1) * cell_h - 6
            draw.rectangle((x0, y0, x1, y1), fill=ON if states[row][col] else OFF)
            # A repeated lamp glyph makes each state visible without relying on color names alone.
            cx, cy = (x0 + x1) / 2, (y0 + y1) / 2
            radius = 18
            if states[row][col]:
                draw.ellipse((cx-radius, cy-radius, cx+radius, cy+radius), fill=PAPER, outline=PAPER, width=3)
                draw.ellipse((cx-7, cy-7, cx+7, cy+7), fill=ON)
            else:
                draw.ellipse((cx-radius, cy-radius, cx+radius, cy+radius), outline=INK, width=4)

    legend = "blue + dot = ON     gold + ring = OFF     each row is one lamp"
    lbox = draw.textbbox((0, 0), legend, font=small_font)
    draw.text(((WIDTH - (lbox[2] - lbox[0])) / 2, 486), legend, fill=INK, font=small_font)
    return image
