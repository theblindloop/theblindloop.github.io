"""Deterministic renderer for a radius-ordered winding constellation."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 512, 384
BACKGROUND = (248, 246, 240)
INK = (24, 28, 34)
CENTER = (142, 147, 156)
DOT_RADIUS = 7
CENTER_RADIUS = 8
DOT_COUNT = 14


def _signed_step(scene: dict) -> float:
    cx, cy = (float(value) for value in scene["center"])
    polar = []
    for x, y in scene["dots"]:
        dx, dy = float(x) - cx, float(y) - cy
        polar.append((math.hypot(dx, dy), math.atan2(dy, dx)))
    polar.sort()
    angles = [angle for _, angle in polar]
    unwrapped = [angles[0]]
    for angle in angles[1:]:
        delta = (angle - unwrapped[-1] + math.pi) % (2.0 * math.pi) - math.pi
        unwrapped.append(unwrapped[-1] + delta)
    n = len(unwrapped)
    mean_i = (n - 1) / 2.0
    mean_a = sum(unwrapped) / n
    denominator = sum((i - mean_i) ** 2 for i in range(n))
    return sum((i - mean_i) * (a - mean_a) for i, a in enumerate(unwrapped)) / denominator


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    direction = 1 if int(seed) % 2 == 0 else -1
    cx = 256 + rng.randint(-12, 12)
    cy = 192 + rng.randint(-10, 10)
    phase = rng.uniform(-math.pi, math.pi)
    angular_step = math.radians(rng.uniform(32.0, 40.0))
    base_radius = rng.uniform(27.0, 31.0)
    radial_step = rng.uniform(8.4, 9.0)
    dots: list[list[int]] = []
    last_radius = 0.0
    for index in range(DOT_COUNT):
        radius = base_radius + index * radial_step + rng.uniform(-0.8, 0.8)
        radius = max(radius, last_radius + 6.5)
        last_radius = radius
        angle = phase + direction * index * angular_step + math.radians(rng.uniform(-1.8, 1.8))
        dots.append([round(cx + radius * math.cos(angle)), round(cy + radius * math.sin(angle))])
    return {"center": [cx, cy], "dots": dots}


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x, y in scene["dots"]:
        x, y = int(x), int(y)
        draw.ellipse(
            (x - DOT_RADIUS, y - DOT_RADIUS, x + DOT_RADIUS, y + DOT_RADIUS),
            fill=INK,
        )
    cx, cy = (int(value) for value in scene["center"])
    draw.ellipse(
        (cx - CENTER_RADIUS, cy - CENTER_RADIUS, cx + CENTER_RADIUS, cy + CENTER_RADIUS),
        fill=CENTER,
    )
    return image


def analytic_gold(scene: dict) -> str:
    return "clockwise" if _signed_step(scene) > 0.0 else "counterclockwise"


def margin(scene: dict) -> float:
    # Difference between the two signed-direction scores, in degrees per step.
    return float(2.0 * abs(math.degrees(_signed_step(scene))))


def is_quarantined(scene: dict) -> bool:
    cx, cy = (float(value) for value in scene["center"])
    radii = sorted(math.hypot(float(x) - cx, float(y) - cy) for x, y in scene["dots"])
    radial_clearance = min(b - a for a, b in zip(radii, radii[1:]))
    return margin(scene) < 40.0 or radial_clearance < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
