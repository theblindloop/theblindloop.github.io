"""Independent pixel-only inverse for radius-ordered winding direction."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[float, float, int]] = []
    for y0, x0 in zip(*np.nonzero(mask)):
        if seen[y0, x0]:
            continue
        queue = deque([(int(y0), int(x0))])
        seen[y0, x0] = True
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= minimum:
            found.append((sum(xs) / len(xs), sum(ys) / len(ys), len(xs)))
    return found


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    black = (array[:, :, 0] < 65) & (array[:, :, 1] < 65) & (array[:, :, 2] < 75)
    gray = (
        (array[:, :, 0] >= 105)
        & (array[:, :, 0] <= 185)
        & (np.max(array, axis=2) - np.min(array, axis=2) < 25)
    )
    dots = _components(black, 80)
    centers = _components(gray, 100)
    if len(dots) != 14 or len(centers) != 1:
        raise ValueError("image must contain exactly fourteen black dots and one gray center")
    cx, cy, _ = centers[0]
    polar = sorted(
        (math.hypot(x - cx, y - cy), math.atan2(y - cy, x - cx))
        for x, y, _ in dots
    )
    radial_gaps = [b[0] - a[0] for a, b in zip(polar, polar[1:])]
    if not radial_gaps or min(radial_gaps) < 3.0:
        raise ValueError("ambiguous radial order")
    unwrapped = [polar[0][1]]
    for _, angle in polar[1:]:
        delta = (angle - unwrapped[-1] + math.pi) % (2.0 * math.pi) - math.pi
        unwrapped.append(unwrapped[-1] + delta)
    n = len(unwrapped)
    mean_i = (n - 1) / 2.0
    mean_a = sum(unwrapped) / n
    denominator = sum((i - mean_i) ** 2 for i in range(n))
    slope = sum((i - mean_i) * (a - mean_a) for i, a in enumerate(unwrapped)) / denominator
    if abs(math.degrees(slope)) < 20.0:
        raise ValueError("winding trend is ambiguous")
    return "clockwise" if slope > 0.0 else "counterclockwise"
