"""Deterministic renderer for cyclic pairing of broad petals in negative space."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

CANVAS_SIZE = 512
SUPERSAMPLE = 3
BACKGROUND = (247, 245, 238)
PALETTE = (
    (28, 116, 151),
    (116, 67, 169),
    (190, 73, 43),
    (38, 135, 94),
    (177, 55, 119),
)
DISTRACTOR_RGB = (158, 156, 150)


def sample_scene(seed: int) -> dict:
    """Sample balanced labels while randomizing pose, palette, and remote clutter."""
    value = int(seed)
    rng = random.Random(value ^ 0xB40A_D5)
    relation = "adjacent" if value % 2 == 0 else "opposite"
    anchor = rng.randrange(4)
    if relation == "adjacent":
        broad = [anchor, (anchor + rng.choice((-1, 1))) % 4]
    else:
        broad = [anchor, (anchor + 2) % 4]
    broad = sorted(broad)

    narrow_halfwidth = (8.0, 9.0, 10.0)[(value // 2) % 3]
    broad_halfwidth = narrow_halfwidth + (9.0, 11.0, 13.0)[(value // 6) % 3]
    widths = [broad_halfwidth if index in broad else narrow_halfwidth for index in range(4)]

    center_x = rng.choice((160, 352))
    center_y = rng.choice((160, 352))
    marks: list[dict] = []
    while len(marks) < 8:
        x = rng.randrange(30, CANVAS_SIZE - 30)
        y = rng.randrange(30, CANVAS_SIZE - 30)
        if abs(x - center_x) < 88 and abs(y - center_y) < 88:
            continue
        marks.append(
            {
                "x": x,
                "y": y,
                "kind": rng.choice(("dot", "square", "dash")),
                "size": rng.choice((3, 4, 5)),
            }
        )

    order = list(range(4))
    rng.shuffle(order)
    return {
        "canvas_size": CANVAS_SIZE,
        "center_x": center_x,
        "center_y": center_y,
        "rotation_deg": rng.randrange(0, 90),
        "broad_indices": broad,
        "petal_halfwidths_deg": widths,
        "petal_draw_order": order,
        "inner_radius": rng.choice((10, 11, 12)),
        "outer_radius": rng.choice((36, 37, 38)),
        "seal_radius": rng.choice((51, 52, 53)),
        "rosette_rgb": list(rng.choice(PALETTE)),
        "background_rgb": list(BACKGROUND),
        "distractor_marks": marks,
    }


def _scaled_point(x: float, y: float) -> tuple[int, int]:
    return round(x * SUPERSAMPLE), round(y * SUPERSAMPLE)


def _petal_polygon(scene: dict, index: int) -> list[tuple[int, int]]:
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    outer = float(scene["outer_radius"])
    center_angle = float(scene["rotation_deg"]) + 90.0 * int(index)
    halfwidths = [float(value) for value in scene["petal_halfwidths_deg"]]
    broad_indices = {int(value) for value in scene["broad_indices"]}
    ranked = sorted(range(4), key=lambda item: halfwidths[item], reverse=True)[:2]
    if set(ranked) != broad_indices:
        raise ValueError("broad_indices must identify the two widest rendered petals")
    halfwidth = halfwidths[int(index)]
    points = [_scaled_point(cx, cy)]
    for step in range(25):
        angle = math.radians(center_angle - halfwidth + 2.0 * halfwidth * step / 24.0)
        points.append(_scaled_point(cx + outer * math.cos(angle), cy + outer * math.sin(angle)))
    return points


def render(scene: dict) -> Image.Image:
    """Render one chromatic seal whose connected pale cavity is the evidence."""
    canvas = int(scene["canvas_size"])
    background = tuple(int(value) for value in scene["background_rgb"])
    image = Image.new("RGB", (canvas * SUPERSAMPLE, canvas * SUPERSAMPLE), background)
    draw = ImageDraw.Draw(image)

    for mark in scene["distractor_marks"]:
        x = int(mark["x"]) * SUPERSAMPLE
        y = int(mark["y"]) * SUPERSAMPLE
        radius = int(mark["size"]) * SUPERSAMPLE
        bounds = (x - radius, y - radius, x + radius, y + radius)
        if str(mark["kind"]) == "dot":
            draw.ellipse(bounds, fill=DISTRACTOR_RGB)
        elif str(mark["kind"]) == "square":
            draw.rectangle(bounds, fill=DISTRACTOR_RGB)
        else:
            draw.rounded_rectangle(
                (x - 2 * radius, y - radius // 2, x + 2 * radius, y + radius // 2),
                radius=max(1, radius // 2),
                fill=DISTRACTOR_RGB,
            )

    cx, cy = int(scene["center_x"]), int(scene["center_y"])
    seal_radius = int(scene["seal_radius"])
    if min(cx - seal_radius, cy - seal_radius) < 20 or max(
        cx + seal_radius, cy + seal_radius
    ) >= canvas - 20:
        raise ValueError("seal is too close to the canvas edge")
    ink = tuple(int(value) for value in scene["rosette_rgb"])
    draw.ellipse(
        (
            (cx - seal_radius) * SUPERSAMPLE,
            (cy - seal_radius) * SUPERSAMPLE,
            (cx + seal_radius) * SUPERSAMPLE,
            (cy + seal_radius) * SUPERSAMPLE,
        ),
        fill=ink,
    )

    for index in [int(value) for value in scene["petal_draw_order"]]:
        draw.polygon(_petal_polygon(scene, index), fill=background)
    inner = int(scene["inner_radius"])
    draw.ellipse(
        (
            (cx - inner) * SUPERSAMPLE,
            (cy - inner) * SUPERSAMPLE,
            (cx + inner) * SUPERSAMPLE,
            (cy + inner) * SUPERSAMPLE,
        ),
        fill=background,
    )
    return image.resize((canvas, canvas), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    first, second = sorted(int(value) for value in scene["broad_indices"])
    separation = (second - first) % 4
    return "opposite" if separation == 2 else "adjacent"


def margin(scene: dict) -> float:
    widths = sorted(
        (float(value) for value in scene["petal_halfwidths_deg"]), reverse=True
    )
    return widths[1] - widths[2]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["petal_draw_order"] = list(reversed([int(value) for value in scene["petal_draw_order"]]))
    return [("reverse_same_color_petal_draw_order", twin)]
