"""Code-native renderer and analytic world for two-center point rotations."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1240
HEIGHT = 820
BACKGROUND = (246, 248, 251)
PANEL = (255, 255, 255)
INK = (25, 33, 49)
MUTED = (83, 95, 115)
GRID = (188, 199, 214)
GOLD = (232, 171, 24)
P1_COLOR = (14, 151, 157)
P2_COLOR = (220, 96, 42)

BOARD_X = 72
BOARD_Y = 190
CELL = 54
BOARD_N = 9
BOARD_RIGHT = BOARD_X + BOARD_N * CELL
BOARD_BOTTOM = BOARD_Y + BOARD_N * CELL

TURN_BANDS = {
    "cw": (35, 111, 211),
    "ccw": (119, 70, 184),
}

TARGET_COLORS = {
    "red": (211, 61, 70),
    "blue": (42, 111, 205),
    "green": (31, 157, 100),
    "violet": (133, 75, 183),
}

# These are deliberately asymmetric: the answer is an exact port lookup, not
# a quadrant label or a board-axis predicate.
TARGETS = {
    "red": (1, 2),
    "blue": (7, 1),
    "green": (2, 7),
    "violet": (7, 6),
}
TARGET_BY_CELL = {cell: name for name, cell in TARGETS.items()}
TARGET_NAMES = tuple(TARGETS)
ALL_CELLS = [(x, y) for y in range(BOARD_N) for x in range(BOARD_N)]
NON_TARGET_CELLS = [cell for cell in ALL_CELLS if cell not in TARGET_BY_CELL]


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    path = (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
        if bold
        else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
    )
    try:
        return ImageFont.truetype(path, size=size)
    except OSError:
        return ImageFont.load_default()


def _cell_center(cell: tuple[int, int]) -> tuple[int, int]:
    x, y = cell
    return BOARD_X + x * CELL + CELL // 2, BOARD_Y + y * CELL + CELL // 2


def _on_board(cell: tuple[int, int]) -> bool:
    return 0 <= cell[0] < BOARD_N and 0 <= cell[1] < BOARD_N


def _opposite(direction: str) -> str:
    return "ccw" if direction == "cw" else "cw"


def _rotate(cell: tuple[int, int], pivot: tuple[int, int], direction: str) -> tuple[int, int]:
    """Rotate a board cell by a visual 90-degree turn about a cell pivot."""
    x, y = cell
    px, py = pivot
    dx, dy = x - px, y - py
    if direction == "cw":
        return px - dy, py + dx
    if direction == "ccw":
        return px + dy, py - dx
    raise ValueError(f"unknown turn direction: {direction!r}")


def _validate_scene(scene: dict[str, Any]) -> tuple[tuple[int, int], tuple[int, int], tuple[int, int], str, str]:
    # Keep the label-bearing scene reads explicit.  Besides making the
    # contract auditable, this prevents a dynamic-key helper from obscuring
    # which rendered fields analytic gold consumes.
    start_value = scene["start_cell"]
    pivot1_value = scene["pivot1_cell"]
    pivot2_value = scene["pivot2_cell"]
    cells: list[tuple[int, int]] = []
    for field, value in (
        ("start_cell", start_value),
        ("pivot1_cell", pivot1_value),
        ("pivot2_cell", pivot2_value),
    ):
        if not isinstance(value, (list, tuple)) or len(value) != 2:
            raise ValueError(f"{field} must contain two coordinates")
        cell = (int(value[0]), int(value[1]))
        if not _on_board(cell):
            raise ValueError(f"{field} is outside the 9 by 9 board")
        if cell in TARGET_BY_CELL:
            raise ValueError(f"{field} overlaps a landing port")
        cells.append(cell)
    if len(set(cells)) != len(cells):
        raise ValueError("start and pivot pins must occupy distinct cells")
    turn1 = str(scene["turn1_direction"])
    turn2 = str(scene["turn2_direction"])
    if turn1 not in TURN_BANDS or turn2 not in TURN_BANDS:
        raise ValueError("turn directions must be cw or ccw")
    intermediate = _rotate(cells[0], cells[1], turn1)
    final = _rotate(intermediate, cells[2], turn2)
    if not _on_board(intermediate) or not _on_board(final):
        raise ValueError("both transformed cells must remain on the board")
    return cells[0], cells[1], cells[2], turn1, turn2


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced, readable scene whose final cell is a fixed port."""
    seed = int(seed)
    rng = random.Random(seed * 1_000_003 + 97)
    wanted = TARGET_NAMES[seed % len(TARGET_NAMES)]
    target = TARGETS[wanted]
    for _ in range(20000):
        turn2 = rng.choice(tuple(TURN_BANDS))
        pivot2 = rng.choice(NON_TARGET_CELLS)
        intermediate = _rotate(target, pivot2, _opposite(turn2))
        if not _on_board(intermediate) or intermediate in TARGET_BY_CELL or intermediate == pivot2:
            continue
        turn1 = rng.choice(tuple(TURN_BANDS))
        pivot1 = rng.choice(NON_TARGET_CELLS)
        if pivot1 == pivot2 or pivot1 == intermediate:
            continue
        start = _rotate(intermediate, pivot1, _opposite(turn1))
        if not _on_board(start) or start in TARGET_BY_CELL:
            continue
        if start in {pivot1, pivot2, intermediate}:
            continue
        scene = {
            "start_cell": [start[0], start[1]],
            "pivot1_cell": [pivot1[0], pivot1[1]],
            "pivot2_cell": [pivot2[0], pivot2[1]],
            "turn1_direction": turn1,
            "turn2_direction": turn2,
        }
        # Validate the forward construction and the intended port class.
        if analytic_gold(scene) == wanted:
            return scene
    raise RuntimeError(f"could not sample a valid two-center scene for seed {seed}")


def _panel(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], radius: int = 18) -> None:
    draw.rounded_rectangle(box, radius=radius, fill=PANEL, outline=(207, 214, 224), width=2)


def _draw_turn_icon(draw: ImageDraw.ImageDraw, center: tuple[int, int], direction: str) -> None:
    cx, cy = center
    box = (cx - 54, cy - 54, cx + 54, cy + 54)
    draw.arc(box, start=35, end=315, fill=INK, width=9)
    if direction == "cw":
        draw.polygon([(cx + 39, cy - 49), (cx + 69, cy - 35), (cx + 40, cy - 22)], fill=INK)
    else:
        draw.polygon([(cx - 39, cy + 49), (cx - 69, cy + 35), (cx - 40, cy + 22)], fill=INK)
    draw.ellipse((cx - 8, cy - 8, cx + 8, cy + 8), fill=INK)


def _draw_pin(draw: ImageDraw.ImageDraw, cell: tuple[int, int], color: tuple[int, int, int], label: str) -> None:
    cx, cy = _cell_center(cell)
    draw.ellipse((cx - 18, cy - 18, cx + 18, cy + 18), fill=(255, 255, 255), outline=color, width=6)
    draw.line((cx - 23, cy, cx + 23, cy), fill=color, width=6)
    draw.line((cx, cy - 23, cx, cy + 23), fill=color, width=6)
    draw.text((cx, cy - 1), label, font=_font(12, True), fill=INK, anchor="mm")


def _draw_port(draw: ImageDraw.ImageDraw, name: str, cell: tuple[int, int]) -> None:
    x, y = cell
    left = BOARD_X + x * CELL + 7
    top = BOARD_Y + y * CELL + 7
    right = BOARD_X + (x + 1) * CELL - 7
    bottom = BOARD_Y + (y + 1) * CELL - 7
    color = TARGET_COLORS[name]
    draw.rounded_rectangle((left, top, right, bottom), radius=10, fill=(255, 255, 255), outline=color, width=6)
    draw.ellipse((left + 13, top + 13, right - 13, bottom - 13), fill=color)
    draw.text(((left + right) // 2, (top + bottom) // 2), name[0].upper(), font=_font(15, True), fill=(255, 255, 255), anchor="mm")


def _draw_turn_card(
    draw: ImageDraw.ImageDraw,
    box: tuple[int, int, int, int],
    number: int,
    pivot_label: str,
    direction: str,
) -> None:
    x0, y0, x1, y1 = box
    _panel(draw, box)
    band = TURN_BANDS[direction]
    draw.rectangle((x0 + 21, y0 + 22, x1 - 21, y0 + 59), fill=band)
    draw.text((x0 + 39, y0 + 40), f"TURN {number}  ·  ABOUT {pivot_label}", font=_font(16, True), fill=(255, 255, 255), anchor="lm")
    draw.text((x0 + 39, y0 + 86), "Rotate the bead by 90°", font=_font(18, True), fill=INK)
    draw.text((x0 + 39, y0 + 116), f"around the {pivot_label} pin", font=_font(15), fill=MUTED)
    _draw_turn_icon(draw, (x0 + 125, y0 + 184), direction)
    word = "CLOCKWISE" if direction == "cw" else "COUNTERCLOCKWISE"
    short = "CW" if direction == "cw" else "CCW"
    draw.text((x0 + 208, y0 + 172), short, font=_font(28, True), fill=INK, anchor="lm")
    draw.text((x0 + 208, y0 + 211), word, font=_font(13, True), fill=MUTED, anchor="lm")


def render(scene: dict[str, Any]) -> Image.Image:
    start, pivot1, pivot2, turn1, turn2 = _validate_scene(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    draw.text((38, 23), "TWO-CENTER TURN LOCATOR", font=_font(30, True), fill=INK)
    draw.text((40, 63), "Carry the gold bead through the two turns in order; the final bead is not drawn.", font=_font(17), fill=MUTED)

    _panel(draw, (30, 100, 650, 790))
    draw.text((55, 119), "9 BY 9 TRANSFORM BOARD", font=_font(19, True), fill=INK)
    draw.text((55, 148), "P1 and P2 are the two rotation centers.", font=_font(14), fill=MUTED)

    for index in range(BOARD_N + 1):
        x = BOARD_X + index * CELL
        y = BOARD_Y + index * CELL
        draw.line((x, BOARD_Y, x, BOARD_BOTTOM), fill=GRID, width=2)
        draw.line((BOARD_X, y, BOARD_RIGHT, y), fill=GRID, width=2)
    # Small row and column indices make exact cell binding countable without
    # introducing a second computational cue.
    for index in range(BOARD_N):
        draw.text((BOARD_X + index * CELL + CELL // 2, BOARD_Y - 14), str(index + 1), font=_font(11, True), fill=MUTED, anchor="ms")
        draw.text((BOARD_X - 14, BOARD_Y + index * CELL + CELL // 2), str(index + 1), font=_font(11, True), fill=MUTED, anchor="rm")

    for name, cell in TARGETS.items():
        _draw_port(draw, name, cell)
    _draw_pin(draw, pivot1, P1_COLOR, "P1")
    _draw_pin(draw, pivot2, P2_COLOR, "P2")
    sx, sy = _cell_center(start)
    draw.ellipse((sx - 20, sy - 20, sx + 20, sy + 20), fill=GOLD, outline=INK, width=4)
    draw.ellipse((sx - 7, sy - 10, sx - 1, sy - 4), fill=(255, 239, 164))

    draw.text((55, 700), "PORT KEY", font=_font(14, True), fill=MUTED)
    port_x = 138
    for name in TARGET_NAMES:
        draw.rounded_rectangle((port_x, 687, port_x + 22, 709), radius=5, fill=TARGET_COLORS[name])
        draw.text((port_x + 30, 698), name, font=_font(13, True), fill=INK, anchor="lm")
        port_x += 112

    _draw_turn_card(draw, (690, 100, 1205, 350), 1, "P1", turn1)
    _draw_turn_card(draw, (690, 380, 1205, 630), 2, "P2", turn2)
    _panel(draw, (690, 660, 1205, 790), radius=14)
    draw.text((718, 683), "READOUT", font=_font(15, True), fill=MUTED)
    draw.text((718, 714), "Which colored port occupies the final cell?", font=_font(17, True), fill=INK)
    draw.text((718, 748), "Answer: red · blue · green · violet", font=_font(14), fill=MUTED)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    start, pivot1, pivot2, turn1, turn2 = _validate_scene(scene)
    intermediate = _rotate(start, pivot1, turn1)
    final = _rotate(intermediate, pivot2, turn2)
    try:
        return TARGET_BY_CELL[final]
    except KeyError as exc:
        raise ValueError(f"final cell is not a landing port: {final}") from exc


def margin(scene: dict[str, Any]) -> float:
    start, pivot1, pivot2, turn1, turn2 = _validate_scene(scene)
    final = _rotate(_rotate(start, pivot1, turn1), pivot2, turn2)
    if final not in TARGET_BY_CELL:
        return 0.0
    distances = [abs(final[0] - x) + abs(final[1] - y) for x, y in TARGET_BY_CELL if (x, y) != final]
    return float(min(distances))


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        start, pivot1, pivot2, turn1, turn2 = _validate_scene(scene)
        final = _rotate(_rotate(start, pivot1, turn1), pivot2, turn2)
    except (KeyError, TypeError, ValueError):
        return True
    return final not in TARGET_BY_CELL


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    _validate_scene(scene)
    return []
