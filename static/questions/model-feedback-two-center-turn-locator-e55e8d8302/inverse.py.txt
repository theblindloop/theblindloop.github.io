"""Independent image-only inverse for the two-center turn world."""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH = 1240
HEIGHT = 820
BOARD_X = 72
BOARD_Y = 190
CELL = 54
BOARD_N = 9
GOLD = np.array((232, 171, 24), dtype=np.int16)
P1_COLOR = np.array((14, 151, 157), dtype=np.int16)
P2_COLOR = np.array((220, 96, 42), dtype=np.int16)
TURN_BANDS = {
    "cw": np.array((35, 111, 211), dtype=np.int16),
    "ccw": np.array((119, 70, 184), dtype=np.int16),
}
TARGET_COLORS = {
    "red": np.array((211, 61, 70), dtype=np.int16),
    "blue": np.array((42, 111, 205), dtype=np.int16),
    "green": np.array((31, 157, 100), dtype=np.int16),
    "violet": np.array((133, 75, 183), dtype=np.int16),
}
TARGETS = {
    "red": (1, 2),
    "blue": (7, 1),
    "green": (2, 7),
    "violet": (7, 6),
}


def _near(arr: np.ndarray, color: np.ndarray, tolerance: int) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - color), axis=2) <= tolerance


def _centroid(mask: np.ndarray, minimum: int, label: str) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if len(xs) < minimum:
        raise ValueError(f"{label} is not recoverable")
    return float(xs.mean()), float(ys.mean())


def _cell_from_xy(x: float, y: float) -> tuple[int, int]:
    col = int(round((x - (BOARD_X + CELL / 2)) / CELL))
    row = int(round((y - (BOARD_Y + CELL / 2)) / CELL))
    if not (0 <= col < BOARD_N and 0 <= row < BOARD_N):
        raise ValueError(f"mark is outside the board: {(col, row)}")
    return col, row


def _recover_pin(arr: np.ndarray, color: np.ndarray, label: str) -> tuple[int, int]:
    board = arr[BOARD_Y : BOARD_Y + BOARD_N * CELL, BOARD_X : BOARD_X + BOARD_N * CELL]
    ys, xs = np.nonzero(_near(board, color, 10))
    if len(xs) < 500:
        raise ValueError(f"{label} pin is not recoverable")
    return _cell_from_xy(float(xs.mean() + BOARD_X), float(ys.mean() + BOARD_Y))


def _card_value(arr: np.ndarray, bounds: tuple[int, int, int, int]) -> str:
    x0, y0, x1, y1 = bounds
    crop = arr[y0:y1, x0:x1]
    counts = {name: int(_near(crop, color, 8).sum()) for name, color in TURN_BANDS.items()}
    name, count = max(counts.items(), key=lambda item: item[1])
    other = max(value for key, value in counts.items() if key != name)
    if count < 3000 or count <= other * 2:
        raise ValueError(f"turn card is unreadable: {counts}")
    return name


def _rotate(cell: tuple[int, int], pivot: tuple[int, int], direction: str) -> tuple[int, int]:
    x, y = cell
    px, py = pivot
    dx, dy = x - px, y - py
    if direction == "cw":
        return px - dy, py + dx
    if direction == "ccw":
        return px + dy, py - dx
    raise ValueError(f"unknown direction {direction!r}")


def _port_at(arr: np.ndarray, cell: tuple[int, int]) -> str:
    x, y = cell
    left = BOARD_X + x * CELL + 7
    top = BOARD_Y + y * CELL + 7
    right = BOARD_X + (x + 1) * CELL - 7
    bottom = BOARD_Y + (y + 1) * CELL - 7
    crop = arr[top : bottom + 1, left : right + 1]
    counts = {name: int(_near(crop, color, 10).sum()) for name, color in TARGET_COLORS.items()}
    name, count = max(counts.items(), key=lambda item: item[1])
    if count < 250:
        raise ValueError(f"final cell has no visible landing port: {cell}, {counts}")
    expected = TARGETS.get(name)
    if expected != cell:
        raise ValueError(f"visible port/color map is inconsistent at {cell}: {name}")
    return name


def decision_from_image(image: Image.Image) -> str:
    """Recover the port color from the final RGB PNG only."""
    rgb = image.convert("RGB")
    if rgb.size != (WIDTH, HEIGHT):
        raise ValueError(f"unexpected canvas size {rgb.size}")
    arr = np.asarray(rgb, dtype=np.int16)
    gx, gy = _centroid(_near(arr, GOLD, 12), 500, "gold start bead")
    start = _cell_from_xy(gx, gy)
    pivot1 = _recover_pin(arr, P1_COLOR, "P1")
    pivot2 = _recover_pin(arr, P2_COLOR, "P2")
    turn1 = _card_value(arr, (711, 122, 1185, 160))
    turn2 = _card_value(arr, (711, 402, 1185, 440))
    intermediate = _rotate(start, pivot1, turn1)
    final = _rotate(intermediate, pivot2, turn2)
    return _port_at(arr, final)
