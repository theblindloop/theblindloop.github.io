"""Renderer and analytic arm for the deferred-toggle pulse world."""

from __future__ import annotations

import math
import random
from typing import Any, Iterable

from PIL import Image, ImageDraw, ImageFont


WIDTH = 1160
HEIGHT = 560
CARD_COUNT = 8
LAMP_COUNT = 4

BACKGROUND = (249, 249, 246)
PANEL = (255, 255, 255)
INK = (30, 36, 43)
MUTED = (91, 100, 109)
BORDER = (150, 159, 167)
DECORATION = (179, 184, 188)

# Shape, not hue, carries operation identity.  Roles rotate through high-separation
# palettes, preventing a fixed color from predicting an operation or answer.
PALETTES = (
    ((12, 145, 166), (230, 119, 30), (125, 72, 184)),
    ((125, 72, 184), (12, 145, 166), (230, 119, 30)),
    ((230, 119, 30), (125, 72, 184), (12, 145, 166)),
    ((18, 126, 91), (211, 78, 91), (55, 112, 190)),
)

_INITIAL_COUNT_OPTIONS = {
    0: (1, 2),
    1: (0, 2, 3),
    2: (0, 1, 3, 4),
    3: (1, 2, 4),
    4: (2, 3),
}


def _font(size: int, *, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _replay(initial: Iterable[bool], operations: Iterable[str]) -> list[bool]:
    """Apply tickets only at pulse barriers; discard nothing until a pulse."""

    state = [bool(value) for value in initial]
    pending = [False] * LAMP_COUNT
    for operation in operations:
        if operation.startswith("ticket_"):
            index = int(operation.split("_", 1)[1]) - 1
            if not 0 <= index < LAMP_COUNT:
                raise ValueError(f"invalid TICKET target in {operation!r}")
            pending[index] ^= True
        elif operation == "pulse":
            state = [value ^ queued for value, queued in zip(state, pending, strict=True)]
            pending = [False] * LAMP_COUNT
        else:
            raise ValueError(f"unknown operation {operation!r}")
    return state


def _counterfactual(initial: list[bool], operations: list[str], mode: str) -> list[bool]:
    state = list(initial)
    pending = [False] * LAMP_COUNT
    for operation in operations:
        if operation.startswith("ticket_"):
            index = int(operation.split("_", 1)[1]) - 1
            if mode == "immediate":
                state[index] ^= True
            elif mode == "dedupe":
                pending[index] = True
            else:
                pending[index] ^= True
        elif operation == "pulse" and mode != "immediate":
            state = [value ^ queued for value, queued in zip(state, pending, strict=True)]
            if mode in {"correct", "dedupe"}:
                pending = [False] * LAMP_COUNT
    return state


def _batch_targets(operations: list[str]) -> tuple[list[list[int]], list[int]]:
    committed: list[list[int]] = []
    current: list[int] = []
    for operation in operations:
        if operation.startswith("ticket_"):
            current.append(int(operation.split("_", 1)[1]))
        else:
            committed.append(current)
            current = []
    return committed, current


def _balanced_program(rng: random.Random, initial: list[bool], desired_count: int) -> list[str]:
    """Find a fixed-budget program whose barriers, clears, and suffix all matter."""

    for _ in range(100_000):
        operations = ["pulse", "pulse"] + [f"ticket_{rng.randrange(1, 5)}" for _ in range(6)]
        rng.shuffle(operations)
        pulse_positions = [i for i, operation in enumerate(operations) if operation == "pulse"]
        if pulse_positions[0] == 0 or pulse_positions[1] >= CARD_COUNT - 1:
            continue
        committed, suffix = _batch_targets(operations)
        if any(not batch for batch in committed) or not suffix:
            continue
        if not any(
            any(batch.count(target) >= 2 and batch.count(target) % 2 == 0 for target in set(batch))
            for batch in committed
        ):
            continue
        # Each pulse must commit at least one odd-parity target.
        if any(all(batch.count(target) % 2 == 0 for target in set(batch)) for batch in committed):
            continue
        final = _replay(initial, operations)
        if sum(final) != desired_count:
            continue
        if any(
            sum(_counterfactual(initial, operations, mode)) == desired_count
            for mode in ("immediate", "no_clear", "dedupe")
        ):
            continue
        if sum(_replay(initial, reversed(operations))) == desired_count:
            continue
        return operations
    raise RuntimeError(f"could not construct deferred-toggle program for final count {desired_count}")


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample deterministically; consecutive seeds exactly balance five answers."""

    seed = int(seed)
    rng = random.Random(seed)
    desired_count = seed % (LAMP_COUNT + 1)
    choices = _INITIAL_COUNT_OPTIONS[desired_count]
    initial_count = choices[(seed // (LAMP_COUNT + 1)) % len(choices)]
    initial = [True] * initial_count + [False] * (LAMP_COUNT - initial_count)
    rng.shuffle(initial)
    operations = _balanced_program(rng, initial, desired_count)

    display_order = list(range(LAMP_COUNT))
    rng.shuffle(display_order)
    decorations = list(range(CARD_COUNT))
    rng.shuffle(decorations)
    palette = [list(color) for color in PALETTES[rng.randrange(len(PALETTES))]]
    if rng.randrange(2):
        palette = palette[1:] + palette[:1]
    return {
        "initial_lamps": initial,
        "operations": operations,
        "display_order": display_order,
        "palette": palette,
        "offset_x": rng.randint(-5, 5),
        "offset_y": rng.randint(-4, 4),
        "card_gap": rng.randint(-1, 1),
        "icon_radius": rng.randint(24, 27),
        "ring_width": rng.randint(5, 7),
        "decorations": decorations,
    }


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(sum(_replay(scene["initial_lamps"], scene["operations"])))


def margin(scene: dict[str, Any]) -> float:
    """Conservative runner-up lead for every parsed raster distinction."""

    radius = float(scene["icon_radius"])
    ring_width = float(scene["ring_width"])
    saturation = min(max(color) - min(color) - 70.0 for color in scene["palette"])
    ring_ratio = math.pi * (radius**2 - (radius - ring_width) ** 2) / (4.0 * radius**2)
    ring_lead = 100.0 * (0.52 - ring_ratio)
    triangle_lead = 100.0 * (0.61 - 0.50)
    circle_lead = 100.0 * min(math.pi / 4.0 - 0.68, 0.87 - math.pi / 4.0)
    pip_clearance = 9.0
    separation = 16.0 + float(scene["card_gap"])
    return round(min(saturation, ring_lead, triangle_lead, circle_lead, pip_clearance, separation), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        initial = scene["initial_lamps"]
        operations = scene["operations"]
        order = scene["display_order"]
        decorations = scene["decorations"]
        palette = scene["palette"]
        tickets = [operation for operation in operations if operation.startswith("ticket_")]
        valid = (
            len(initial) == LAMP_COUNT
            and all(isinstance(value, bool) for value in initial)
            and len(operations) == CARD_COUNT
            and operations.count("pulse") == 2
            and len(tickets) == 6
            and all(operation in {"ticket_1", "ticket_2", "ticket_3", "ticket_4"} for operation in tickets)
            and operations[-1] != "pulse"
            and sorted(order) == list(range(LAMP_COUNT))
            and sorted(decorations) == list(range(CARD_COUNT))
            and len(palette) == 3
            and all(len(color) == 3 for color in palette)
            and -5 <= int(scene["offset_x"]) <= 5
            and -4 <= int(scene["offset_y"]) <= 4
            and -1 <= int(scene["card_gap"]) <= 1
            and 24 <= int(scene["icon_radius"]) <= 27
            and 5 <= int(scene["ring_width"]) <= 7
        )
        return not valid or margin(scene) < 8.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []


def _centered(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font, fill) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2 - box[1]), text, font=font, fill=fill)


def _draw_pips(draw: ImageDraw.ImageDraw, center_x: float, center_y: float, count: int, color: tuple[int, int, int]) -> None:
    for index in range(count):
        x = center_x + (index - (count - 1) / 2) * 13
        draw.ellipse((x - 4, center_y - 4, x + 4, center_y + 4), fill=color)


def _draw_decoration(draw: ImageDraw.ImageDraw, center_x: float, y: float, value: int) -> None:
    for bit in range(3):
        x = center_x + (bit - 1) * 9
        box = (x - 2, y - 2, x + 2, y + 2)
        if value & (1 << bit):
            draw.ellipse(box, fill=DECORATION)
        else:
            draw.ellipse(box, outline=DECORATION, width=1)


def render(scene: dict[str, Any]) -> Image.Image:
    initial = list(scene["initial_lamps"])
    operations = list(scene["operations"])
    display_order = list(scene["display_order"])
    palette = [tuple(int(channel) for channel in color) for color in scene["palette"]]
    offset_x = int(scene["offset_x"])
    offset_y = int(scene["offset_y"])
    card_gap = int(scene["card_gap"])
    icon_radius = int(scene["icon_radius"])
    ring_width = int(scene["ring_width"])
    decorations = list(scene["decorations"])

    if is_quarantined(scene):
        raise ValueError("cannot render malformed or quarantined scene")

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(25, bold=True)
    label_font = _font(14, bold=True)
    small_font = _font(12, bold=True)

    _centered(draw, (WIDTH / 2, 27), "DEFERRED TOGGLE BOARD", title_font, INK)
    _centered(draw, (WIDTH / 2, 57), "TICKET: queue a toggle     PULSE: apply all queued toggles, then clear", _font(14), MUTED)

    top = 79 + offset_y
    bottom = 225 + offset_y
    draw.rounded_rectangle((31 + offset_x, top, 1129 + offset_x, bottom), radius=16, fill=PANEL, outline=BORDER, width=3)
    draw.text((49 + offset_x, top + 13), "START", font=label_font, fill=INK)
    lamp_color = palette[0]
    for slot, logical_index in enumerate(display_order):
        x = 210 + slot * 248 + offset_x
        y = 139 + offset_y
        bounds = (x - icon_radius, y - icon_radius, x + icon_radius, y + icon_radius)
        if initial[logical_index]:
            draw.ellipse(bounds, fill=lamp_color)
        else:
            draw.ellipse(bounds, outline=lamp_color, width=ring_width)
        _draw_pips(draw, x, y + 51, logical_index + 1, lamp_color)
        _centered(draw, (x, y + 76), "ON" if initial[logical_index] else "OFF", small_font, MUTED)

    draw.text((31, 243), "READ CARDS LEFT TO RIGHT", font=label_font, fill=INK)
    card_top = 270 + offset_y
    card_bottom = 484 + offset_y
    card_width = 126
    step = 141 + card_gap
    for index, (operation, decoration) in enumerate(zip(operations, decorations, strict=True)):
        left = 18 + offset_x + index * step
        right = left + card_width
        center_x = (left + right) / 2
        draw.rounded_rectangle((left, card_top, right, card_bottom), radius=12, fill=PANEL, outline=BORDER, width=3)
        _centered(draw, (center_x, card_top + 19), str(index + 1), small_font, MUTED)
        center_y = card_top + 73
        if operation.startswith("ticket_"):
            color = palette[1]
            draw.polygon(((center_x, center_y - icon_radius), (center_x - icon_radius, center_y + icon_radius), (center_x + icon_radius, center_y + icon_radius)), fill=color)
            _draw_pips(draw, center_x, center_y + 53, int(operation.split("_", 1)[1]), color)
            label = "TICKET"
        elif operation == "pulse":
            color = palette[2]
            draw.ellipse((center_x - icon_radius, center_y - icon_radius, center_x + icon_radius, center_y + icon_radius), fill=color)
            label = "PULSE"
        else:
            raise ValueError(f"unknown operation {operation!r}")
        _centered(draw, (center_x, card_top + 155), label, _font(11, bold=True), INK)
        _draw_decoration(draw, center_x, card_top + 190, int(decoration))

    _centered(draw, (WIDTH / 2, 528), "filled = ON   ring = OFF     triangle = TICKET   circle = PULSE", _font(13, bold=True), MUTED)
    return image
