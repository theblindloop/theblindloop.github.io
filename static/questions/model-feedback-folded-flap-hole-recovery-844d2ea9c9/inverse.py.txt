"""Independent image-only inverse for the folded-flap world."""

from __future__ import annotations

import numpy as np
from PIL import Image

CANVAS = (900, 640)
MAP_X, MAP_Y, MAP_CELL = 42, 62, 45
PACKET_X, PACKET_Y, PACKET_CELL = 402, 92, 45
CANDIDATE_X = (42, 336, 630)
CANDIDATE_Y, CANDIDATE_CELL = 370, 36
INK = (27, 32, 38)
RED = (213, 65, 70)
BLUE = (39, 113, 189)
PACKET_BORDER = (40, 121, 88)
TAG_RGB = {"amber": (226, 151, 28), "cyan": (20, 157, 181), "violet": (137, 78, 191)}
ALLOWED = {
    (255, 255, 255), (247, 249, 250), (118, 128, 138), INK, RED, BLUE,
    (255, 228, 229), (225, 239, 253), (235, 222, 244), (229, 244, 237),
    PACKET_BORDER, *TAG_RGB.values(),
}


class OracleError(ValueError):
    pass


def _read_cells(arr: np.ndarray, x: int, y: int, n: int, cell: int) -> set[tuple[int, int]]:
    result: set[tuple[int, int]] = set()
    for r in range(n):
        for c in range(n):
            cx, cy = x + c * cell + cell // 2, y + r * cell + cell // 2
            patch = arr[cy - 5:cy + 6, cx - 5:cx + 6]
            ink_count = int(np.all(patch == INK, axis=2).sum())
            if ink_count >= 70:
                result.add((r, c))
            elif ink_count != 0:
                raise OracleError("damaged or off-center punch mark")
    return result


def _expand(punches: set[tuple[int, int]]) -> set[tuple[int, int]]:
    # Packet columns 0..3 coincide with full-sheet columns 2..5.
    full = {(r, c + 2) for r, c in punches}
    # Undo the lower flap: row 2 came from 5, row 3 came from 4.
    for r, c in tuple(full):
        if r == 2:
            full.add((5, c))
        elif r == 3:
            full.add((4, c))
    # Undo the left flap: column 2 came from 1, column 3 came from 0.
    for r, c in tuple(full):
        if c == 2:
            full.add((r, 1))
        elif c == 3:
            full.add((r, 0))
    return full


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    if rgb.size != CANVAS:
        raise OracleError("wrong canvas size")
    arr = np.asarray(rgb)
    histogram = rgb.getcolors(maxcolors=64)
    if histogram is None:
        raise OracleError("too many raster colors")
    colors = {tuple(color) for _, color in histogram}
    if not colors <= ALLOWED:
        raise OracleError("unknown raster color")
    if int(np.all(arr == RED, axis=2).sum()) < 1700 or int(np.all(arr == BLUE, axis=2).sum()) < 1700:
        raise OracleError("fold cues are missing")

    packet_positions = []
    for shift in (-12, 0, 12):
        x = PACKET_X + shift
        left_edge = arr[PACKET_Y:PACKET_Y + 4 * PACKET_CELL + 1, x:x + 6]
        if int(np.all(left_edge == PACKET_BORDER, axis=2).sum()) > 800:
            packet_positions.append(x)
    if len(packet_positions) != 1:
        raise OracleError("folded packet border is missing or ambiguous")
    punches = _read_cells(arr, packet_positions[0], PACKET_Y, 4, PACKET_CELL)
    if len(punches) != 2:
        raise OracleError("expected exactly two packet punches")
    target = _expand(punches)

    matches: list[str] = []
    seen_colors: set[str] = set()
    for x in CANDIDATE_X:
        pixel = tuple(int(v) for v in arr[CANDIDATE_Y + 2, x + 2])
        tags = [name for name, value in TAG_RGB.items() if pixel == value]
        if len(tags) != 1 or tags[0] in seen_colors:
            raise OracleError("candidate border colors are invalid")
        color = tags[0]
        seen_colors.add(color)
        pattern = _read_cells(arr, x, CANDIDATE_Y, 6, CANDIDATE_CELL)
        if len(pattern) != len(target):
            raise OracleError("candidate hole counts differ from the reconstructed count")
        if pattern == target:
            matches.append(color)
    if seen_colors != set(TAG_RGB) or len(matches) != 1:
        raise OracleError("no unique unfolded candidate")
    return matches[0]
