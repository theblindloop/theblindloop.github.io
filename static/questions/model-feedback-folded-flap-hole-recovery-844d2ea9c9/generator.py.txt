"""Deterministic renderer for a two-flap punch-unfolding world."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = (900, 640)
MAP_X, MAP_Y, MAP_CELL = 42, 62, 45
PACKET_X, PACKET_Y, PACKET_CELL = 402, 92, 45
CANDIDATE_X = (42, 336, 630)
CANDIDATE_Y, CANDIDATE_CELL = 370, 36
ANSWERS = ("amber", "cyan", "violet")

RGB = {
    "background": (255, 255, 255),
    "paper": (247, 249, 250),
    "grid": (118, 128, 138),
    "ink": (27, 32, 38),
    "red": (213, 65, 70),
    "blue": (39, 113, 189),
    "red_flap": (255, 228, 229),
    "blue_flap": (225, 239, 253),
    "overlap": (235, 222, 244),
    "packet": (229, 244, 237),
    "packet_border": (40, 121, 88),
    "amber": (226, 151, 28),
    "cyan": (20, 157, 181),
    "violet": (137, 78, 191),
}


def _unfold(punches: list[list[int]], red_map: dict[int, int], blue_map: dict[int, int]) -> set[tuple[int, int]]:
    points = {(int(r), int(c) + 2) for r, c in punches}
    for r, c in list(points):
        if r in blue_map:
            points.add((blue_map[r], c))
    for r, c in list(points):
        if c in red_map:
            points.add((r, red_map[c]))
    return points


def _patterns(punches: list[list[int]]) -> tuple[set[tuple[int, int]], ...]:
    correct = _unfold(punches, {2: 1, 3: 0}, {2: 5, 3: 4})
    preserve_red_order = _unfold(punches, {2: 0, 3: 1}, {2: 5, 3: 4})
    preserve_blue_order = _unfold(punches, {2: 1, 3: 0}, {2: 4, 3: 5})
    return correct, preserve_red_order, preserve_blue_order


def sample_scene(seed: int) -> dict:
    rng = random.Random((int(seed) ^ 0xF01D6A9B) & 0xFFFFFFFFFFFFFFFF)
    both = [(r, c) for r in (2, 3) for c in (0, 1)]
    other = [(r, c) for r in range(4) for c in range(4) if (r, c) not in both]
    first = both[rng.randrange(len(both))]
    # Alternate between one-axis and zero-axis second punches without hiding this in gold.
    one_axis = [(r, c) for r, c in other if (r in (2, 3)) ^ (c in (0, 1))]
    zero_axis = [(r, c) for r, c in other if r not in (2, 3) and c not in (0, 1)]
    pool = one_axis if int(seed) % 2 == 0 else zero_axis
    second = pool[rng.randrange(len(pool))]
    punches = [[first[0], first[1]], [second[0], second[1]]]

    correct, wrong_red, wrong_blue = _patterns(punches)
    if len({frozenset(correct), frozenset(wrong_red), frozenset(wrong_blue)}) != 3:
        raise RuntimeError("non-unique fold patterns")
    target_slot = int(seed) % 3
    patterns = [wrong_red, wrong_blue]
    rng.shuffle(patterns)
    patterns.insert(target_slot, correct)

    answer_color = ANSWERS[(int(seed) // 3) % 3]
    remaining = [color for color in ANSWERS if color != answer_color]
    if rng.randrange(2):
        remaining.reverse()
    tag_colors: list[str | None] = [None, None, None]
    tag_colors[target_slot] = answer_color
    fill = iter(remaining)
    for index in range(3):
        if tag_colors[index] is None:
            tag_colors[index] = next(fill)

    return {
        "punches": punches,
        "candidates": [sorted([list(point) for point in pattern]) for pattern in patterns],
        "tag_colors": [str(color) for color in tag_colors],
        "packet_shift": (-12, 0, 12)[(int(seed) // 9) % 3],
    }


def _grid(draw: ImageDraw.ImageDraw, x: int, y: int, n: int, cell: int, fill: tuple[int, int, int], border: tuple[int, int, int], width: int = 3) -> None:
    size = n * cell
    draw.rectangle((x, y, x + size, y + size), fill=fill, outline=border, width=width)
    for k in range(1, n):
        draw.line((x + k * cell, y, x + k * cell, y + size), fill=RGB["grid"], width=2)
        draw.line((x, y + k * cell, x + size, y + k * cell), fill=RGB["grid"], width=2)


def _dot(draw: ImageDraw.ImageDraw, x: int, y: int, cell: int) -> None:
    radius = max(7, cell // 5)
    cx, cy = x + cell // 2, y + cell // 2
    draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=RGB["ink"])


def _arrow(draw: ImageDraw.ImageDraw, start: tuple[int, int], end: tuple[int, int], color: tuple[int, int, int]) -> None:
    draw.line((*start, *end), fill=color, width=7)
    ex, ey = end
    sx, sy = start
    if abs(ex - sx) > abs(ey - sy):
        sign = 1 if ex > sx else -1
        draw.polygon([(ex, ey), (ex - sign * 18, ey - 12), (ex - sign * 18, ey + 12)], fill=color)
    else:
        sign = 1 if ey > sy else -1
        draw.polygon([(ex, ey), (ex - 12, ey - sign * 18), (ex + 12, ey - sign * 18)], fill=color)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS, RGB["background"])
    draw = ImageDraw.Draw(image)

    # Original 6x6 fold map. The lower-left overlap is purple; all cues are visible.
    for r in range(6):
        for c in range(6):
            if c < 2 and r >= 4:
                fill = RGB["overlap"]
            elif c < 2:
                fill = RGB["red_flap"]
            elif r >= 4:
                fill = RGB["blue_flap"]
            else:
                fill = RGB["paper"]
            x, y = MAP_X + c * MAP_CELL, MAP_Y + r * MAP_CELL
            draw.rectangle((x, y, x + MAP_CELL, y + MAP_CELL), fill=fill)
    _grid(draw, MAP_X, MAP_Y, 6, MAP_CELL, RGB["paper"], RGB["grid"], width=3)
    # Restore flap shading after the common grid helper's background fill.
    for r in range(6):
        for c in range(6):
            if c >= 2 and r < 4:
                continue
            if c < 2 and r >= 4:
                fill = RGB["overlap"]
            elif c < 2:
                fill = RGB["red_flap"]
            else:
                fill = RGB["blue_flap"]
            x, y = MAP_X + c * MAP_CELL + 2, MAP_Y + r * MAP_CELL + 2
            draw.rectangle((x, y, x + MAP_CELL - 2, y + MAP_CELL - 2), fill=fill)
    draw.line((MAP_X + 2 * MAP_CELL, MAP_Y, MAP_X + 2 * MAP_CELL, MAP_Y + 6 * MAP_CELL), fill=RGB["red"], width=7)
    draw.line((MAP_X, MAP_Y + 4 * MAP_CELL, MAP_X + 6 * MAP_CELL, MAP_Y + 4 * MAP_CELL), fill=RGB["blue"], width=7)
    _arrow(draw, (MAP_X + 28, MAP_Y + 76), (MAP_X + 153, MAP_Y + 76), RGB["red"])
    _arrow(draw, (MAP_X + 224, MAP_Y + 252), (MAP_X + 224, MAP_Y + 154), RGB["blue"])

    shift = int(scene["packet_shift"])
    px = PACKET_X + shift
    _grid(draw, px, PACKET_Y, 4, PACKET_CELL, RGB["packet"], RGB["packet_border"], width=6)
    for r, c in scene["punches"]:
        _dot(draw, px + int(c) * PACKET_CELL, PACKET_Y + int(r) * PACKET_CELL, PACKET_CELL)

    for slot, x in enumerate(CANDIDATE_X):
        color = str(scene["tag_colors"][slot])
        _grid(draw, x, CANDIDATE_Y, 6, CANDIDATE_CELL, RGB["paper"], RGB[color], width=8)
        for r, c in scene["candidates"][slot]:
            _dot(draw, x + int(c) * CANDIDATE_CELL, CANDIDATE_Y + int(r) * CANDIDATE_CELL, CANDIDATE_CELL)
    return image


def analytic_gold(scene: dict) -> str:
    target = _patterns(scene["punches"])[0]
    matches = [i for i, pattern in enumerate(scene["candidates"]) if {tuple(p) for p in pattern} == target]
    if len(matches) != 1:
        raise ValueError("no unique unfolded match")
    return str(scene["tag_colors"][matches[0]])


def margin(scene: dict) -> float:
    target = _patterns(scene["punches"])[0]
    distances = sorted(len(target.symmetric_difference({tuple(p) for p in pattern})) for pattern in scene["candidates"])
    return float(distances[1]) if distances and distances[0] == 0 else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        target = _patterns(scene["punches"])[0]
        patterns = [{tuple(p) for p in pattern} for pattern in scene["candidates"]]
        return len(set(map(frozenset, patterns))) != 3 or sum(pattern == target for pattern in patterns) != 1 or margin(scene) < 2
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
