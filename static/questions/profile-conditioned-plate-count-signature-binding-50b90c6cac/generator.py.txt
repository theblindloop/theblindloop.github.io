"""Renderer for plate_count_signature_binding.

Two rows of four plates. The upper row is colored (red, blue, green, orange);
the lower row shows the same four plates in gray, re-ordered, re-sized and with
their black dots re-arranged. The only property carried across the rows is the
number of black dots on a plate, and the four counts are distinct, so the
cross-row assignment is unique. Exactly one lower plate is surrounded by a black
frame; the gold answer is the color of the matching upper plate.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw

RENDERER_VERSION = "plate_count_signature_binding-renderer-0.1.0"

WIDTH = 650
HEIGHT = 430

COLOR_NAMES = ("red", "blue", "green", "orange")
PALETTE = {
    "red": (214, 45, 45),
    "blue": (40, 80, 210),
    "green": (35, 150, 60),
    "orange": (240, 150, 20),
}
GRAY = (185, 185, 185)
BLACK = (15, 15, 15)
WHITE = (255, 255, 255)

N_PLATES = 4
CELL_X0 = 16
CELL_STEP = 158
CELL_W = 146
TOP_BAND = (26, 196)
BOTTOM_BAND = (246, 416)

FRAME_OFFSET = 6
FRAME_WIDTH = 3
FRAME_MARGIN = FRAME_OFFSET + FRAME_WIDTH

COUNT_POOL = (2, 3, 4, 5, 6, 7, 8)
QUARANTINE_CLEARANCE = 5.0


def _place_dots(
    rng: np.random.Generator,
    box: list[int],
    count: int,
    radius: int,
    gap: float,
) -> list[list[float]] | None:
    """Rejection-sample `count` non-overlapping dot centers inside `box`."""
    x0, y0, x1, y1 = box
    pad = radius + gap
    lo_x, hi_x = x0 + pad, x1 - pad
    lo_y, hi_y = y0 + pad, y1 - pad
    if hi_x <= lo_x or hi_y <= lo_y:
        return None
    need = 2.0 * radius + gap
    centers: list[list[float]] = []
    attempts = 0
    while len(centers) < count:
        attempts += 1
        if attempts > 4000:
            return None
        cx = float(rng.uniform(lo_x, hi_x))
        cy = float(rng.uniform(lo_y, hi_y))
        if all((cx - px) ** 2 + (cy - py) ** 2 >= need * need for px, py in centers):
            centers.append([round(cx, 3), round(cy, 3)])
    return centers


def _plate_box(rng: np.random.Generator, slot: int, band: tuple[int, int]) -> list[int]:
    width = int(rng.integers(110, 129))
    height = int(rng.integers(126, 151))
    cell_x = CELL_X0 + CELL_STEP * slot
    x_off = int(rng.integers(FRAME_MARGIN, CELL_W - width - FRAME_MARGIN + 1))
    band_h = band[1] - band[0]
    y_off = int(rng.integers(FRAME_MARGIN, band_h - height - FRAME_MARGIN + 1))
    x0 = cell_x + x_off
    y0 = band[0] + y_off
    return [x0, y0, x0 + width, y0 + height]


def _clearance(box: list[int], dots: list[list[float]], radius: int) -> float:
    """Smallest visible separation: dot-to-dot gap and dot-to-plate-border gap."""
    x0, y0, x1, y1 = box
    best = float("inf")
    for cx, cy in dots:
        best = min(best, cx - radius - x0, y1 - cy - radius, cy - radius - y0, x1 - cx - radius)
    for i in range(len(dots)):
        for j in range(i + 1, len(dots)):
            dx = dots[i][0] - dots[j][0]
            dy = dots[i][1] - dots[j][1]
            best = min(best, float(np.hypot(dx, dy)) - 2.0 * radius)
    return float(best)


def _build_row(
    rng: np.random.Generator,
    counts_by_slot: list[int],
    band: tuple[int, int],
    gap: float,
) -> list[dict] | None:
    row: list[dict] = []
    for slot, count in enumerate(counts_by_slot):
        radius = int(rng.integers(3, 6))
        placed = None
        box: list[int] = []
        for _ in range(12):
            box = _plate_box(rng, slot, band)
            placed = _place_dots(rng, box, count, radius, gap)
            if placed is not None:
                break
        if placed is None:
            return None
        row.append({"count": int(count), "box": box, "radius": radius, "dots": placed})
    return row


def sample_scene(seed: int) -> dict:
    """Sample one balanced scene. Answer color is drawn uniformly and first."""
    rng = np.random.default_rng(int(seed) + 90_001)
    for _ in range(64):
        gap = float(round(rng.uniform(2.5, 11.5), 3))
        counts = sorted(int(c) for c in rng.choice(COUNT_POOL, size=N_PLATES, replace=False))
        # counts_by_color[k] is the dot count of the plate colored COLOR_NAMES[k]
        perm = rng.permutation(N_PLATES)
        counts_by_color = [counts[int(p)] for p in perm]

        top_order = list(map(int, rng.permutation(N_PLATES)))
        bottom_order = list(map(int, rng.permutation(N_PLATES)))
        marked_slot = int(rng.integers(N_PLATES))

        top_counts = [counts_by_color[c] for c in top_order]
        bottom_counts = [counts_by_color[c] for c in bottom_order]

        top_row = _build_row(rng, top_counts, TOP_BAND, gap)
        bottom_row = _build_row(rng, bottom_counts, BOTTOM_BAND, gap)
        if top_row is None or bottom_row is None:
            continue
        for slot, entry in enumerate(top_row):
            entry["color"] = COLOR_NAMES[top_order[slot]]

        scene = {
            "seed": int(seed),
            "top": top_row,
            "bottom": bottom_row,
            "marked_slot": marked_slot,
            "gap_target": gap,
        }
        return scene
    raise RuntimeError(f"scene sampling failed for seed {seed}")


def _clearances(scene: dict) -> list[float]:
    values = []
    for key in ("top", "bottom"):
        for plate in scene[key]:
            values.append(_clearance(plate["box"], plate["dots"], int(plate["radius"])))
    return values


def render(scene: dict) -> Image.Image:
    """Deterministic raster; the only cross-row signal drawn is the dot count."""
    image = Image.new("RGB", (WIDTH, HEIGHT), WHITE)
    draw = ImageDraw.Draw(image)
    for plate in scene["top"]:
        _draw_plate(draw, plate, PALETTE[plate["color"]])
    for plate in scene["bottom"]:
        _draw_plate(draw, plate, GRAY)
    marked = scene["bottom"][int(scene["marked_slot"])]
    x0, y0, x1, y1 = marked["box"]
    draw.rectangle(
        [x0 - FRAME_OFFSET, y0 - FRAME_OFFSET, x1 + FRAME_OFFSET, y1 + FRAME_OFFSET],
        outline=BLACK,
        width=FRAME_WIDTH,
    )
    return image


def _draw_plate(draw: ImageDraw.ImageDraw, plate: dict, fill: tuple[int, int, int]) -> None:
    x0, y0, x1, y1 = plate["box"]
    draw.rectangle([x0, y0, x1, y1], fill=fill)
    radius = int(plate["radius"])
    for cx, cy in plate["dots"]:
        draw.ellipse([cx - radius, cy - radius, cx + radius, cy + radius], fill=BLACK)


def analytic_gold(scene: dict) -> str:
    """Color of the upper plate carrying the framed lower plate's dot count."""
    target = int(scene["bottom"][int(scene["marked_slot"])]["count"])
    matches = [p["color"] for p in scene["top"] if int(p["count"]) == target]
    if len(matches) != 1:
        raise ValueError("non-unique count assignment")
    return matches[0]


def margin(scene: dict) -> float:
    """Weakest visible dot separation in pixels: the legibility boundary."""
    return float(round(min(_clearances(scene)), 6))


def is_quarantined(scene: dict) -> bool:
    """Flag scenes whose weakest dot separation approaches raster resolution."""
    return bool(margin(scene) < QUARANTINE_CLEARANCE)


def runner_up_margin(scene: dict) -> int:
    """Smallest count difference between the answer plate and any other plate."""
    target = int(scene["bottom"][int(scene["marked_slot"])]["count"])
    others = [int(p["count"]) for p in scene["top"] if int(p["count"]) != target]
    return int(min(abs(target - c) for c in others))


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Label-preserving latent aliases that render to a byte-identical PNG."""
    rng = np.random.default_rng(int(scene["seed"]) + 5501)

    def _copy(row: list[dict]) -> list[dict]:
        return [
            {k: ([list(d) for d in v] if k == "dots" else (list(v) if k == "box" else v))
             for k, v in plate.items()}
            for plate in row
        ]

    # Dots on a plate form an unordered set: any listing order draws the same pixels.
    shuffled = {
        "seed": scene["seed"],
        "top": _copy(scene["top"]),
        "bottom": _copy(scene["bottom"]),
        "marked_slot": scene["marked_slot"],
        "gap_target": scene["gap_target"],
    }
    for key in ("top", "bottom"):
        for plate in shuffled[key]:
            order = rng.permutation(len(plate["dots"]))
            plate["dots"] = [plate["dots"][int(i)] for i in order]

    # A plate box is equally described by its two opposite corners in either
    # listing order once the renderer normalizes; expressing the same rectangle
    # with float-valued but pixel-identical corner values is a second alias.
    retyped = {
        "seed": scene["seed"],
        "top": _copy(scene["top"]),
        "bottom": _copy(scene["bottom"]),
        "marked_slot": scene["marked_slot"],
        "gap_target": scene["gap_target"],
    }
    for key in ("top", "bottom"):
        for plate in retyped[key]:
            plate["dots"] = [[float(c) for c in dot] for dot in plate["dots"]]
            plate["box"] = [float(v) for v in plate["box"]]

    return [
        ("dot_listing_order", shuffled),
        ("coordinate_retyping", retyped),
    ]
