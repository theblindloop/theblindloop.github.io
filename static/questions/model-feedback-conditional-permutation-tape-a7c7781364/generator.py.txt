"""Renderer and analytic state machine for a conditional permutation tape."""

from __future__ import annotations

import itertools
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1400
HEIGHT = 920
BACKGROUND = (248, 250, 252)
INK = (25, 31, 42)
MUTED = (88, 98, 113)
PANEL = (255, 255, 255)
LINE = (199, 206, 216)
TILE_COLORS = (
    (245, 158, 11),   # amber
    (6, 182, 212),    # cyan
    (139, 92, 246),   # violet
)
TILE_NAMES = ("A", "C", "V")
TILE_WORDS = ("AMBER", "CYAN", "VIOLET")
ORDERS = tuple(itertools.permutations(range(3)))
ORDER_CODES = tuple("".join(TILE_NAMES[value] for value in order) for order in ORDERS)

START_CENTERS = ((125, 232), (270, 232), (415, 232))
COMMAND_X = (140, 325, 510, 695, 880, 1065, 1250)
COMMAND_Y = 528
COMMAND_COLOR_OFFSET = -30
COMMAND_SIGN_OFFSET = 34


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    path = (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
        if bold
        else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
    )
    try:
        return ImageFont.truetype(path, size=size)
    except OSError:
        return ImageFont.load_default()


def _validate_scene(scene: dict[str, Any]) -> tuple[tuple[int, int, int], list[tuple[int, int]]]:
    try:
        start = tuple(int(value) for value in scene["start_order"])
        commands = [(int(item[0]), int(item[1])) for item in scene["commands"]]
    except (KeyError, TypeError, ValueError, IndexError) as exc:
        raise ValueError("scene has invalid state inputs") from exc
    if len(start) != 3 or sorted(start) != [0, 1, 2]:
        raise ValueError("start_order must be a permutation of amber, cyan, violet")
    if len(commands) != 7:
        raise ValueError("commands must contain seven cards")
    if any(color not in range(3) or direction not in (-1, 1) for color, direction in commands):
        raise ValueError("command colors or signs are invalid")
    return start, commands


def _step(order: tuple[int, int, int] | list[int], color: int, direction: int) -> tuple[int, int, int]:
    """Apply one visible command, comparing against the current middle tile."""

    values = list(order)
    if color == values[1]:
        if direction == 1:  # rotate one place to the right
            values = [values[2], values[0], values[1]]
        else:  # rotate one place to the left
            values = [values[1], values[2], values[0]]
    elif direction == 1:  # miss +: exchange middle and right
        values[1], values[2] = values[2], values[1]
    else:  # miss -: exchange left and middle
        values[0], values[1] = values[1], values[0]
    return tuple(values)


def _trace(start: tuple[int, int, int], commands: list[tuple[int, int]]) -> list[tuple[int, int, int]]:
    path = [tuple(start)]
    order = tuple(start)
    for color, direction in commands:
        order = _step(order, color, direction)
        path.append(order)
    return path


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample all six terminal orders with both branch types represented."""

    seed = int(seed)
    target = seed % len(ORDERS)
    for attempt in range(20000):
        rng = random.Random(seed * 1_000_003 + attempt * 91_879 + 47)
        start = tuple(rng.sample(range(3), 3))
        commands = [(rng.randrange(3), rng.choice((-1, 1))) for _ in range(7)]
        path = _trace(start, commands)
        matches = sum(color == before[1] for before, (color, _) in zip(path[:-1], commands, strict=True))
        final = path[-1]
        if (
            ORDERS.index(final) == target
            and len({color for color, _ in commands}) == 3
            and {-1, 1}.issubset({direction for _, direction in commands})
            and 2 <= matches <= 5
            and len(set(path)) >= 4
        ):
            return {
                "start_order": list(start),
                "commands": [[color, direction] for color, direction in commands],
            }
    raise RuntimeError(f"could not sample a balanced permutation trace for seed {seed}")


def _panel(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], radius: int = 18) -> None:
    draw.rounded_rectangle(box, radius=radius, fill=PANEL, outline=LINE, width=2)


def _tile(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    value: int,
    half: int,
    *,
    text_size: int | None = None,
) -> None:
    cx, cy = center
    draw.rounded_rectangle(
        (cx - half, cy - half, cx + half, cy + half),
        radius=max(6, half // 4),
        fill=TILE_COLORS[value],
        outline=INK,
        width=3,
    )
    draw.text(
        (cx, cy + 1),
        TILE_NAMES[value],
        font=_font(text_size or max(12, half // 2), True),
        fill=(255, 255, 255),
        anchor="mm",
    )


def _sign(draw: ImageDraw.ImageDraw, center: tuple[int, int], direction: int, half: int = 21) -> None:
    cx, cy = center
    draw.rectangle((cx - half, cy - 5, cx + half, cy + 5), fill=INK)
    if direction == 1:
        draw.rectangle((cx - 5, cy - half, cx + 5, cy + half), fill=INK)


def _arrow(draw: ImageDraw.ImageDraw, start: tuple[int, int], end: tuple[int, int]) -> None:
    draw.line((*start, *end), fill=LINE, width=5)
    ex, ey = end
    draw.polygon([(ex, ey), (ex - 14, ey - 8), (ex - 14, ey + 8)], fill=LINE)


def _key_cell(draw: ImageDraw.ImageDraw, center: tuple[int, int], order: tuple[int, int, int]) -> None:
    cx, cy = center
    draw.rounded_rectangle(
        (cx - 180, cy - 22, cx + 180, cy + 22),
        radius=12,
        fill=(247, 249, 252),
        outline=LINE,
        width=2,
    )
    for index, value in enumerate(order):
        _tile(draw, (cx - 112 + index * 48, cy), value, 16, text_size=14)
    draw.text((cx + 86, cy + 1), "".join(TILE_NAMES[value] for value in order), font=_font(21, True), fill=INK, anchor="mm")


def render(scene: dict[str, Any]) -> Image.Image:
    start, commands = _validate_scene(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    draw.text((38, 24), "CONDITIONAL PERMUTATION TAPE", font=_font(30, True), fill=INK)
    draw.text(
        (40, 64),
        "Carry the ordered tiles forward; each card chooses a visible permutation.",
        font=_font(17),
        fill=MUTED,
    )

    _panel(draw, (35, 105, 500, 360))
    draw.text((58, 130), "START ORDER", font=_font(20, True), fill=INK)
    draw.text((58, 165), "read the three tiles left to right", font=_font(15), fill=MUTED)
    for index, (center, value) in enumerate(zip(START_CENTERS, start, strict=True), start=1):
        _tile(draw, center, value, 48)
        draw.text((center[0], 306), TILE_WORDS[value], font=_font(14, True), fill=INK, anchor="mm")
    draw.text((58, 340), "the middle tile is the comparison target", font=_font(15), fill=MUTED)

    _panel(draw, (525, 105, 1365, 360))
    draw.text((550, 130), "PERMUTATION RULE", font=_font(20, True), fill=INK)
    draw.text((550, 166), "At each card, compare its color with the CURRENT MIDDLE tile.", font=_font(16), fill=INK)
    draw.text((550, 200), "MATCH  →  rotate all three:  + right,  − left", font=_font(16, True), fill=INK)
    draw.text((550, 234), "MISS   →  swap one pair:  + middle/right,  − left/middle", font=_font(16, True), fill=INK)
    draw.text((550, 278), "The sign is a direction:  + = right       − = left", font=_font(15), fill=MUTED)
    # A compact legend makes the two branches visually concrete without adding a second data source.
    for index, value in enumerate(range(3)):
        _tile(draw, (905 + index * 70, 321), value, 23, text_size=16)
    draw.text((1140, 321), "A amber   C cyan   V violet", font=_font(15, True), fill=INK, anchor="lm")

    _panel(draw, (35, 390, 1365, 635))
    draw.text((58, 414), "COMMAND ORDER", font=_font(20, True), fill=INK)
    draw.text((290, 417), "read cards 1 through 7 from left to right", font=_font(16), fill=MUTED)
    for number, (x, (color, direction)) in enumerate(zip(COMMAND_X, commands, strict=True), start=1):
        draw.rounded_rectangle((x - 78, 455, x + 78, 605), radius=14, fill=(247, 249, 252), outline=LINE, width=2)
        draw.ellipse((x - 62, 466, x - 34, 494), fill=INK)
        draw.text((x - 48, 480), str(number), font=_font(13, True), fill=(255, 255, 255), anchor="mm")
        _tile(draw, (x + COMMAND_COLOR_OFFSET, COMMAND_Y), color, 28)
        _sign(draw, (x + COMMAND_SIGN_OFFSET, COMMAND_Y), direction)
        draw.text((x, 585), "color + sign", font=_font(12), fill=MUTED, anchor="mm")

    _panel(draw, (35, 665, 1365, 890))
    draw.text((58, 690), "FINAL ORDER KEY", font=_font(20, True), fill=INK)
    draw.text((315, 692), "report the code beside the matching tile order", font=_font(16), fill=MUTED)
    key_centers = ((400, 748), (1000, 748), (400, 805), (1000, 805), (400, 862), (1000, 862))
    for center, order in zip(key_centers, ORDERS, strict=True):
        _key_cell(draw, center, order)
    draw.text((58, 878), "tile order is always left to right", font=_font(14), fill=MUTED)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    start, commands = _validate_scene(scene)
    final = _trace(start, commands)[-1]
    return "".join(TILE_NAMES[value] for value in final)


def margin(scene: dict[str, Any]) -> float:
    _validate_scene(scene)
    return 1.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    _validate_scene(scene)
    return False


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    _validate_scene(scene)
    return []
