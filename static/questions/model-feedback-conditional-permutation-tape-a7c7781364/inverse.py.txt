"""Independent image-only inverse for the conditional permutation board."""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH = 1400
HEIGHT = 920
NAMES = ("A", "C", "V")
COLORS = np.array(
    [
        (245, 158, 11),
        (6, 182, 212),
        (139, 92, 246),
    ],
    dtype=np.int16,
)
INK = np.array((25, 31, 42), dtype=np.int16)
START_X = (125, 270, 415)
START_Y = 232
COMMAND_X = (140, 325, 510, 695, 880, 1065, 1250)
COMMAND_Y = 528
COLOR_OFFSET = -30
SIGN_OFFSET = 34


def _color_at(arr: np.ndarray, x: int, y: int) -> int:
    # The renderer places a white letter at the geometric center of every
    # tile. Sample a stable colored interior pixel beside that letter.
    rgb = arr[y - 18, x - 18]
    distances = ((COLORS.astype(np.int32) - rgb.astype(np.int32)) ** 2).sum(axis=1)
    index = int(np.argmin(distances))
    if int(distances[index]) > 1800:
        raise ValueError("tile color is missing or unreadable")
    return index


def _card(arr: np.ndarray, x: int) -> tuple[int, int]:
    color = _color_at(arr, x + COLOR_OFFSET, COMMAND_Y)
    sx = x + SIGN_OFFSET
    crop = arr[COMMAND_Y - 24:COMMAND_Y + 25, sx - 24:sx + 25]
    ink = np.all(crop == INK, axis=2)
    vertical = int(ink[:, 20:29].sum())
    if vertical > 200:
        direction = 1
    elif int(ink.sum()) > 150:
        direction = -1
    else:
        raise ValueError("command sign is missing or unreadable")
    return color, direction


def _step(order: list[int], color: int, direction: int) -> list[int]:
    values = list(order)
    if color == values[1]:
        if direction == 1:
            return [values[2], values[0], values[1]]
        return [values[1], values[2], values[0]]
    if direction == 1:
        values[1], values[2] = values[2], values[1]
    else:
        values[0], values[1] = values[1], values[0]
    return values


def decision_from_image(image: Image.Image) -> str:
    """Recover the terminal tile order from raster pixels only."""

    image = image.convert("RGB")
    if image.size != (WIDTH, HEIGHT):
        raise ValueError(f"unexpected board size {image.size}")
    arr = np.asarray(image, dtype=np.int16)
    order = [_color_at(arr, x, START_Y) for x in START_X]
    if sorted(order) != [0, 1, 2]:
        raise ValueError("start row is not three distinct tiles")
    for x in COMMAND_X:
        color, direction = _card(arr, x)
        order = _step(order, color, direction)
    return "".join(NAMES[value] for value in order)
