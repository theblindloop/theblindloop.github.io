"""Independent pixel-only inverse arm for the mean-elevation comparison world.

This module receives ONLY the rasterized PNG and re-measures which color's dots
sit, on average, higher on the panel. It shares no geometry with
renderer.render() and never imports the renderer, prompts, generator, verifier,
or analytic gold.

Pipeline: (1) label every pixel to its nearest fixed color (red, blue, or light
background); (2) for each color, find the connected components of that color's
ink, i.e. the individual dots; (3) take each dot's centroid; (4) average each
color's dot centroid vertical positions and return the color whose dots are
higher on average (smaller mean vertical position).
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "mean_elevation_compare-oracle-0.1.0"

RED_RGB = (205, 28, 38)
BLUE_RGB = (20, 60, 200)
BACKGROUND = (249, 249, 249)

_REF = np.array(
    [
        BACKGROUND,
        RED_RGB,
        BLUE_RGB,
    ],
    dtype=np.float32,
)


def _label(arr: np.ndarray) -> np.ndarray:
    """Label every pixel 0=background, 1=red, 2=blue (nearest fixed color)."""
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - _REF[np.newaxis, np.newaxis, :, :]
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def _dot_centroids(labels: np.ndarray, color_label: int) -> list[tuple[float, float]]:
    """Centroid (x, y) of each connected dot of a color."""
    ys, xs = np.nonzero(labels == color_label)
    if len(ys) == 0:
        return []
    pixels = set(zip(xs.tolist(), ys.tolist()))
    centroids: list[tuple[float, float]] = []
    while pixels:
        seed = pixels.pop()
        stack = [seed]
        member = []
        while stack:
            px, py = stack.pop()
            member.append((px, py))
            for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                nb = (px + dx, py + dy)
                if nb in pixels:
                    pixels.discard(nb)
                    stack.append(nb)
        mx = sum(p[0] for p in member) / len(member)
        my = sum(p[1] for p in member) / len(member)
        centroids.append((mx, my))
    return centroids


def _mean_y(centroids: list[tuple[float, float]]) -> float:
    if not centroids:
        return 0.0
    return float(sum(my for (_mx, my) in centroids) / len(centroids))


def decision_from_image(image: Image.Image) -> str:
    """Return the color whose dots sit higher on average (smaller mean y)."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label(arr)
    mu_red = _mean_y(_dot_centroids(labels, 1))
    mu_blue = _mean_y(_dot_centroids(labels, 2))
    if mu_red < mu_blue:
        return "red"
    return "blue"
