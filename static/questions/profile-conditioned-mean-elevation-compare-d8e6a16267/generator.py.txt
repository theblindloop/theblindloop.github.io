"""Mean-elevation comparison world.

A light panel shows a scattered set of red dots and blue dots interleaved over a
shared canvas. The two colors hold independent (and possibly unequal) counts of
dots. The question asks which color's dots sit, on average, HIGHER on the panel
(closer to the top), i.e. which color has the smaller mean vertical position of
its dots.

This is an aggregate-location (measurement) decision: the pixel arm must locate
every dot of each color, average that color's dot heights, and compare the two
mean heights. Neither a single topmost dot, nor the color with more dots, nor
the total amount of ink determines the winner, because only the arithmetic mean
of each color's dot positions separates them.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "mean_elevation_compare-0.1.0"

RED_RGB = (205, 28, 38)
BLUE_RGB = (20, 60, 200)
BACKGROUND = (249, 249, 249)

WIDTH = 560
HEIGHT = 320

PAD = 40
RADIUS_LO = 7
RADIUS_HI = 12
MIN_DOT_SEPARATION = 30.0
COUNT_POOL = (3, 4)
MIN_MARGIN = 6.0
QUARANTINE_MARGIN = 12.0


def _mean(values: list[float]) -> float:
    return float(sum(values) / len(values))


def _mean_y(groups: dict[str, list[float]]) -> float:
    return _mean(groups)


def sample_scene(seed: int) -> dict:
    """Sample a deterministic scene with a clearly separated mean-elevation decision."""
    rng = random.Random(seed * 131 + 17)
    for _attempt in range(40000):
        counts = {"red": rng.choice(COUNT_POOL), "blue": rng.choice(COUNT_POOL)}
        dots: list[dict] = []
        placed: list[tuple[float, float, float]] = []
        for color in ("red", "blue"):
            for _ in range(counts[color]):
                for _try in range(500):
                    x = rng.uniform(PAD, WIDTH - PAD)
                    y = rng.uniform(PAD, HEIGHT - PAD)
                    r = rng.uniform(RADIUS_LO, RADIUS_HI)
                    if all(
                        ((x - px) ** 2 + (y - py) ** 2) ** 0.5 > MIN_DOT_SEPARATION
                        for (px, py, _pr) in placed
                    ):
                        dots.append({"color": color, "x": x, "y": y, "r": r})
                        placed.append((x, y, r))
                        break
                else:
                    raise RuntimeError("could not place a dot")
        groups: dict[str, list[float]] = {"red": [], "blue": []}
        for dot in dots:
            groups[dot["color"]].append(float(dot["y"]))
        mu_red = _mean(groups["red"])
        mu_blue = _mean(groups["blue"])
        if abs(mu_red - mu_blue) < MIN_MARGIN:
            continue
        scene = {
            "dots": dots,
            "width": WIDTH,
            "height": HEIGHT,
        }
        return scene
    raise RuntimeError("could not sample a separated mean-elevation scene")


def render(scene: dict) -> Image.Image:
    """Render I = R(z): colored filled dots over a light canvas."""
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    for dot in scene["dots"]:
        rgb = RED_RGB if dot["color"] == "red" else BLUE_RGB
        r = float(dot["r"])
        x = float(dot["x"])
        y = float(dot["y"])
        draw.ellipse([x - r, y - r, x + r, y + r], fill=rgb)
    return img


def _groups(scene: dict) -> dict[str, list[float]]:
    groups: dict[str, list[float]] = {"red": [], "blue": []}
    for dot in scene["dots"]:
        groups[dot["color"]].append(float(dot["y"]))
    return groups


def analytic_gold(scene: dict) -> str:
    """Color whose dots sit higher on average (smaller mean y)."""
    groups = _groups(scene)
    mu_red = _mean(groups["red"])
    mu_blue = _mean(groups["blue"])
    if mu_red < mu_blue:
        return "red"
    return "blue"


def margin(scene: dict) -> float:
    """Absolute difference between the two color mean vertical positions."""
    groups = _groups(scene)
    return abs(_mean(groups["red"]) - _mean(groups["blue"]))


def is_quarantined(scene: dict) -> bool:
    """Near-boundary: mean vertical positions are uncomfortably close together."""
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list:
    """not_applicable: every latent field is drawn verbatim as visible ink.

    Each dot's color, center position, and radius are rendered as a filled disk
    of that color, and the analytic decision reads only these directly rendered
    dot fields. There is no hidden scene field that could change the mean
    position without changing the raster, so no nontrivial pixel-identical,
    gold-preserving latent transformation exists.
    """
    return []
