"""Deterministic renderer and analytic arm for a 2-by-2 determinant comparison."""

from __future__ import annotations

import copy
import itertools
import random

from PIL import Image, ImageDraw

WIDTH = 640
HEIGHT = 500
PANEL_RECTS = (
    (25, 45, 305, 230),
    (335, 45, 615, 230),
    (25, 285, 305, 470),
    (335, 285, 615, 470),
)
LABELS = "ABCD"
BACKGROUND = (255, 255, 255)
PANEL_FILL = (248, 249, 250)
INK = (31, 38, 45)
BLUE = (35, 103, 180)
ORANGE = (220, 105, 25)
PALE_GRID = (208, 214, 220)
QUARANTINE_GAP = 4.0

_GLYPHS = {
    "1": ("00100", "01100", "00100", "00100", "00100", "00100", "11111"),
    "2": ("11110", "00001", "00001", "11110", "10000", "10000", "11111"),
    "3": ("11110", "00001", "00001", "01110", "00001", "00001", "11110"),
    "4": ("10001", "10001", "10001", "11111", "00001", "00001", "00001"),
    "5": ("11111", "10000", "10000", "11110", "00001", "00001", "11110"),
    "6": ("01111", "10000", "10000", "11110", "10001", "10001", "01110"),
    "7": ("11111", "00001", "00010", "00100", "01000", "01000", "01000"),
    "8": ("01110", "10001", "10001", "01110", "10001", "10001", "01110"),
    "9": ("01110", "10001", "10001", "01111", "00001", "00001", "11110"),
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "B": ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    "C": ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    "D": ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
}


def _determinant_cells(cells: list[dict]) -> int:
    values = {(int(cell["row"]), int(cell["col"])): int(cell["value"]) for cell in cells}
    if set(values) != {(0, 0), (0, 1), (1, 0), (1, 1)}:
        raise ValueError("matrix must contain each of four cell positions exactly once")
    return values[(0, 0)] * values[(1, 1)] - values[(0, 1)] * values[(1, 0)]


def _determinants(panels: list[dict]) -> list[int]:
    if len(panels) != 4:
        raise ValueError("scene must contain four panels")
    return [_determinant_cells(panel["cells"]) for panel in panels]


def sample_scene(seed: int) -> dict:
    """Sample equal-multiset matrices; ``seed % 4`` counterbalances the winner."""
    rng = random.Random(int(seed))
    winner_index = int(seed) % 4
    for _ in range(500):
        digits = rng.sample(range(1, 10), 4)
        by_value: dict[int, list[tuple[int, int, int, int]]] = {}
        for permutation in itertools.permutations(digits):
            determinant = permutation[0] * permutation[3] - permutation[1] * permutation[2]
            by_value.setdefault(determinant, []).append(permutation)
        winner_value = max(by_value)
        loser_values = [value for value in by_value if value <= winner_value - 6]
        if len(loser_values) < 3:
            continue
        rng.shuffle(loser_values)
        selected_values = loser_values[:3]
        arrangements = [rng.choice(by_value[value]) for value in selected_values]
        winner_arrangement = rng.choice(by_value[winner_value])
        rng.shuffle(arrangements)
        arrangements.insert(winner_index, winner_arrangement)
        panels = []
        for arrangement in arrangements:
            cells = [
                {"row": index // 2, "col": index % 2, "value": int(value)}
                for index, value in enumerate(arrangement)
            ]
            rng.shuffle(cells)
            panels.append({"cells": cells})
        return {"panels": panels, "digit_color": rng.choice(("blue", "orange"))}
    raise RuntimeError("could not sample a separated determinant scene")


def analytic_gold(scene: dict) -> str:
    values = _determinants(scene["panels"])
    return LABELS[max(range(4), key=values.__getitem__)]


def margin(scene: dict) -> float:
    values = sorted(_determinants(scene["panels"]), reverse=True)
    return float(values[0] - values[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Cell record order is latent because row/column fields determine placement."""
    twin = copy.deepcopy(scene)
    for panel in twin["panels"]:
        panel["cells"].reverse()
    return [("reverse_cell_record_order", twin)]


def transpose_all_matrices(scene: dict) -> dict:
    """Transpose every displayed matrix; determinants and the winner are unchanged."""
    twin = copy.deepcopy(scene)
    for panel in twin["panels"]:
        for cell in panel["cells"]:
            cell["row"], cell["col"] = cell["col"], cell["row"]
    return twin


def swap_digit_palette(scene: dict) -> dict:
    """Swap the common digit palette without changing numeric content."""
    twin = copy.deepcopy(scene)
    twin["digit_color"] = "orange" if twin["digit_color"] == "blue" else "blue"
    return twin


def permute_panels(scene: dict, order: list[int]) -> dict:
    """Apply a declared label-transforming panel permutation."""
    if sorted(order) != [0, 1, 2, 3]:
        raise ValueError("order must be a permutation of 0,1,2,3")
    twin = copy.deepcopy(scene)
    twin["panels"] = [twin["panels"][index] for index in order]
    return twin


def _draw_glyph(
    draw: ImageDraw.ImageDraw, glyph: str, x: int, y: int, color: tuple[int, int, int], scale: int
) -> None:
    for row, bits in enumerate(_GLYPHS[glyph]):
        for col, bit in enumerate(bits):
            if bit == "1":
                draw.rectangle(
                    (x + col * scale, y + row * scale,
                     x + (col + 1) * scale - 1, y + (row + 1) * scale - 1),
                    fill=color,
                )


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    panels = scene["panels"]
    _determinants(panels)
    color_name = scene["digit_color"]
    if color_name not in {"blue", "orange"}:
        raise ValueError("digit_color must be blue or orange")
    digit_color = BLUE if color_name == "blue" else ORANGE
    for index, (panel, rect) in enumerate(zip(panels, PANEL_RECTS, strict=True)):
        x0, y0, x1, y1 = rect
        draw.rectangle(rect, fill=PANEL_FILL, outline=INK, width=3)
        _draw_glyph(draw, LABELS[index], x0 + 7, y0 - 28, INK, 3)
        # A visible 2-by-2 table establishes the row/column binding independently of code.
        gx0, gy0, gx1, gy1 = x0 + 62, y0 + 20, x1 - 62, y1 - 20
        draw.rectangle((gx0, gy0, gx1, gy1), outline=INK, width=3)
        draw.line((gx0 + (gx1 - gx0) // 2, gy0, gx0 + (gx1 - gx0) // 2, gy1), fill=PALE_GRID, width=2)
        draw.line((gx0, gy0 + (gy1 - gy0) // 2, gx1, gy0 + (gy1 - gy0) // 2), fill=PALE_GRID, width=2)
        positions = {}
        for cell in panel["cells"]:
            key = (int(cell["row"]), int(cell["col"]))
            if key in positions:
                raise ValueError("duplicate matrix cell")
            positions[key] = str(int(cell["value"]))
        for (row, col), value in positions.items():
            cell_width = (gx1 - gx0) // 2
            cell_height = (gy1 - gy0) // 2
            px = gx0 + col * cell_width + (cell_width - 20) // 2
            py = gy0 + row * cell_height + (cell_height - 28) // 2
            _draw_glyph(draw, value, px, py, digit_color, 4)
    return image
