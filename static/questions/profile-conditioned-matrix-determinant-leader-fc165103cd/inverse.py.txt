"""Independent inverse arm: decode visible matrix glyphs from PNG pixels."""

from __future__ import annotations

import numpy as np
from PIL import Image

_BLUE = np.array((35, 103, 180), dtype=np.int32)
_ORANGE = np.array((220, 105, 25), dtype=np.int32)
_LABELS = "ABCD"
_DIGITS = {
    "1": ("00100", "01100", "00100", "00100", "00100", "00100", "11111"),
    "2": ("11110", "00001", "00001", "11110", "10000", "10000", "11111"),
    "3": ("11110", "00001", "00001", "01110", "00001", "00001", "11110"),
    "4": ("10001", "10001", "10001", "11111", "00001", "00001", "00001"),
    "5": ("11111", "10000", "10000", "11110", "00001", "00001", "11110"),
    "6": ("01111", "10000", "10000", "11110", "10001", "10001", "01110"),
    "7": ("11111", "00001", "00010", "00100", "01000", "01000", "01000"),
    "8": ("01110", "10001", "10001", "01110", "10001", "10001", "01110"),
    "9": ("01110", "10001", "10001", "01111", "00001", "00001", "11110"),
}
_LETTERS = {
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "B": ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    "C": ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    "D": ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
}


def _runs(values: np.ndarray) -> list[int]:
    if len(values) == 0:
        return []
    groups: list[list[int]] = [[int(values[0])]]
    for value in values[1:]:
        current = int(value)
        if current == groups[-1][-1] + 1:
            groups[-1].append(current)
        else:
            groups.append([current])
    return [int(round(sum(group) / len(group))) for group in groups]


def _panel_rectangles(arr: np.ndarray) -> list[tuple[int, int, int, int]]:
    dark = np.all(arr < 90, axis=2)
    xs = _runs(np.flatnonzero(dark.sum(axis=0) > arr.shape[0] * 0.68))
    ys = _runs(np.flatnonzero(dark.sum(axis=1) > arr.shape[1] * 0.68))
    if len(xs) != 4 or len(ys) != 4:
        raise ValueError("abstain: four matrix panel frames could not be recovered")
    rectangles = [
        (xs[0], ys[0], xs[1], ys[1]),
        (xs[2], ys[0], xs[3], ys[1]),
        (xs[0], ys[2], xs[1], ys[3]),
        (xs[2], ys[2], xs[3], ys[3]),
    ]
    if any(x1 - x0 < 120 or y1 - y0 < 100 for x0, y0, x1, y1 in rectangles):
        raise ValueError("abstain: recovered matrix panels are degenerate")
    return rectangles


def _decode_glyph(mask: np.ndarray, templates: dict[str, tuple[str, ...]], name: str) -> str:
    coordinates = np.argwhere(mask)
    if len(coordinates) == 0:
        raise ValueError(f"abstain: missing {name} glyph")
    y0, x0 = coordinates.min(axis=0)
    y1, x1 = coordinates.max(axis=0)
    crop = mask[y0:y1 + 1, x0:x1 + 1]
    height, width = crop.shape
    if width % 5 or height % 7 or width // 5 != height // 7 or width // 5 < 2:
        raise ValueError(f"abstain: malformed {name} glyph bounds")
    scale = width // 5
    observed = []
    for row in range(7):
        bits = []
        for col in range(5):
            block = crop[row * scale:(row + 1) * scale, col * scale:(col + 1) * scale]
            bits.append("1" if float(block.mean()) > 0.75 else "0")
        observed.append("".join(bits))
    matches = [glyph for glyph, pattern in templates.items() if tuple(observed) == pattern]
    if len(matches) != 1:
        raise ValueError(f"abstain: unrecognized {name} glyph")
    return matches[0]


def _color_mask(arr: np.ndarray, target: np.ndarray) -> np.ndarray:
    delta = arr.astype(np.int32) - target
    return np.sum(delta * delta, axis=2, dtype=np.int32) < 10 * 10


def _validate_label(arr: np.ndarray, rect: tuple[int, int, int, int], expected: str) -> None:
    x0, y0, _, _ = rect
    search = arr[max(0, y0 - 38):y0 - 3, x0 + 2:x0 + 35]
    actual = _decode_glyph(np.all(search < 90, axis=2), _LETTERS, "panel-label")
    if actual != expected:
        raise ValueError("abstain: panel labels are missing or out of reading order")


def _decode_panel(
    arr: np.ndarray, rect: tuple[int, int, int, int], target: np.ndarray
) -> tuple[int, int, int, int]:
    x0, y0, x1, y1 = rect
    xm, ym = (x0 + x1) // 2, (y0 + y1) // 2
    bounds = (
        (x0 + 4, y0 + 4, xm, ym),
        (xm, y0 + 4, x1 - 3, ym),
        (x0 + 4, ym, xm, y1 - 3),
        (xm, ym, x1 - 3, y1 - 3),
    )
    values = []
    for left, top, right, bottom in bounds:
        mask = _color_mask(arr[top:bottom, left:right], target)
        values.append(int(_decode_glyph(mask, _DIGITS, "digit")))
    return tuple(values)  # type: ignore[return-value]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    if arr.ndim != 3 or arr.shape[0] < 200 or arr.shape[1] < 250:
        raise ValueError("abstain: raster is too small")
    rectangles = _panel_rectangles(arr)
    for expected, rect in zip(_LABELS, rectangles, strict=True):
        _validate_label(arr, rect, expected)
    blue_count = int(_color_mask(arr, _BLUE).sum())
    orange_count = int(_color_mask(arr, _ORANGE).sum())
    if blue_count >= 100 and orange_count < 10:
        target = _BLUE
    elif orange_count >= 100 and blue_count < 10:
        target = _ORANGE
    else:
        raise ValueError("abstain: a single supported digit palette was not recovered")
    determinants = []
    for rect in rectangles:
        a, b, c, d = _decode_panel(arr, rect, target)
        determinants.append(a * d - b * c)
    order = sorted(range(4), key=determinants.__getitem__, reverse=True)
    if determinants[order[0]] - determinants[order[1]] < 4:
        raise ValueError("abstain: top two visible determinants are ambiguous")
    return _LABELS[order[0]]
