"""Independent image-only inverse arm.

The implementation knows only the public visual grammar: four reading-order
panels, two saturated colors forming rectangular shafts and triangular heads,
and the rule that a head's displacement from its shaft gives arrow direction.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")


def _components(mask: np.ndarray) -> list[dict]:
    height, width = mask.shape
    remaining = {tuple(point) for point in np.argwhere(mask)}
    found = []
    while remaining:
        start = remaining.pop()
        queue = deque([start])
        points = [start]
        while queue:
            y, x = queue.popleft()
            for neighbor in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                ny, nx = neighbor
                if 0 <= ny < height and 0 <= nx < width and neighbor in remaining:
                    remaining.remove(neighbor)
                    queue.append(neighbor)
                    points.append(neighbor)
        if len(points) >= 120:
            ys = np.fromiter((p[0] for p in points), dtype=float)
            xs = np.fromiter((p[1] for p in points), dtype=float)
            found.append({"area": len(points), "x": float(xs.mean()), "y": float(ys.mean())})
    return found


def _dominant_saturated_colors(array: np.ndarray) -> list[tuple[int, int, int]]:
    flat = array.reshape(-1, 3)
    spread = flat.max(axis=1) - flat.min(axis=1)
    vivid = flat[spread > 65]
    if len(vivid) < 1000:
        raise ValueError("insufficient visible arrow ink")
    colors, counts = np.unique(vivid, axis=0, return_counts=True)
    order = np.argsort(counts)[::-1]
    dominant = [tuple(int(v) for v in colors[index]) for index in order if counts[index] > 900]
    if len(dominant) != 2:
        raise ValueError("expected exactly two dominant arrow colors")
    return dominant


def _split_two(values: list[float]) -> tuple[float, float]:
    ordered = sorted(values)
    gaps = [ordered[i + 1] - ordered[i] for i in range(len(ordered) - 1)]
    cut = max(range(len(gaps)), key=lambda i: gaps[i])
    return max(ordered[: cut + 1]), min(ordered[cut + 1 :])


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    colors = _dominant_saturated_colors(array)
    groups = []
    for color in colors:
        mask = np.all(array == np.asarray(color, dtype=np.uint8), axis=2)
        groups.append(_components(mask))
    if sorted(len(group) for group in groups) != [8, 8]:
        raise ValueError("could not enumerate eight shafts and eight heads")
    # Rectangular shafts occupy more pixels than triangular heads. This remains
    # true when their two colors are permuted.
    groups.sort(key=lambda group: sum(item["area"] for item in group), reverse=True)
    shafts, heads = groups
    if sum(item["area"] for item in shafts) <= sum(item["area"] for item in heads) * 1.2:
        raise ValueError("shaft/head geometry is ambiguous")

    x_left_max, x_right_min = _split_two([item["x"] for item in shafts])
    y_top_max, y_bottom_min = _split_two([item["y"] for item in shafts])
    x_cut = (x_left_max + x_right_min) / 2.0
    y_cut = (y_top_max + y_bottom_min) / 2.0

    panels: list[list[tuple[dict, dict]]] = [[] for _ in range(4)]
    unused = set(range(len(heads)))
    for shaft in shafts:
        panel = (2 if shaft["y"] > y_cut else 0) + (1 if shaft["x"] > x_cut else 0)
        candidates = [
            index
            for index in unused
            if ((2 if heads[index]["y"] > y_cut else 0) + (1 if heads[index]["x"] > x_cut else 0))
            == panel
        ]
        if not candidates:
            raise ValueError("head/shaft panel assignment failed")
        head_index = min(
            candidates,
            key=lambda index: (heads[index]["x"] - shaft["x"]) ** 2
            + (heads[index]["y"] - shaft["y"]) ** 2,
        )
        unused.remove(head_index)
        panels[panel].append((shaft, heads[head_index]))
    if unused or any(len(panel) != 2 for panel in panels):
        raise ValueError("panel arrow enumeration failed")

    scores = []
    for panel in panels:
        panel.sort(key=lambda pair: pair[0]["x"])
        left_dx = panel[0][1]["x"] - panel[0][0]["x"]
        right_dx = panel[1][1]["x"] - panel[1][0]["x"]
        scores.append(min(left_dx, -right_dx))
    order = sorted(range(4), key=lambda index: scores[index], reverse=True)
    if scores[order[0]] <= 8.0 or scores[order[0]] - scores[order[1]] <= 8.0:
        raise ValueError("ambiguous convergence winner; abstaining")
    return LABELS[order[0]]
