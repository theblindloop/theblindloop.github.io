"""Deterministic scene sampler, renderer, and analytic arm."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 640, 480
LABELS = ("A", "B", "C", "D")
PALETTES = (
    ((26, 87, 138), (232, 126, 35)),
    ((109, 46, 145), (17, 155, 128)),
    ((183, 48, 72), (35, 118, 178)),
    ((36, 116, 102), (213, 74, 112)),
)
QUARANTINE_MARGIN = 10.0


def _vector(rng: random.Random, horizontal_sign: int) -> list[int]:
    dx = horizontal_sign * rng.randint(20, 31)
    dy = rng.randint(-9, 9)
    return [dx, dy]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    # Exactly one inward pair; all four direction-sign configurations have
    # identical object/color/ink inventories before harmless angular jitter.
    target_index = int(seed) % 4
    distractors = [(-1, 1), (1, 1), (-1, -1)]
    rng.shuffle(distractors)
    configurations = []
    for index in range(4):
        configurations.append((1, -1) if index == target_index else distractors.pop())
    vectors = [
        [_vector(rng, left_sign), _vector(rng, right_sign)]
        for left_sign, right_sign in configurations
    ]
    shaft, head = PALETTES[rng.randrange(len(PALETTES))]
    if rng.randrange(2):
        shaft, head = head, shaft
    return {
        "panel_vectors": vectors,
        "shaft_color": list(shaft),
        "head_color": list(head),
        "shift_x": rng.randint(-7, 7),
        "shift_y": rng.randint(-6, 6),
    }


def _panel_scores(scene: dict) -> list[float]:
    scores = []
    for pair in scene["panel_vectors"]:
        left_dx = float(pair[0][0])
        right_dx = float(pair[1][0])
        scores.append(min(left_dx, -right_dx))
    return scores


def analytic_gold(scene: dict) -> str:
    scores = _panel_scores(scene)
    best = max(range(4), key=lambda index: scores[index])
    if scores[best] <= 0 or sum(score > 0 for score in scores) != 1:
        raise ValueError("scene does not contain exactly one inward-facing pair")
    return LABELS[best]


def margin(scene: dict) -> float:
    ordered = sorted(_panel_scores(scene), reverse=True)
    return float(min(ordered[0], ordered[0] - ordered[1]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _arrow_polygon(center: tuple[float, float], vector: list[int]):
    dx, dy = float(vector[0]), float(vector[1])
    length = math.hypot(dx, dy)
    ux, uy = dx / length, dy / length
    px, py = -uy, ux
    cx, cy = center
    half_len, half_width = 22.0, 8.0
    shaft = [
        (cx - ux * half_len - px * half_width, cy - uy * half_len - py * half_width),
        (cx + ux * half_len - px * half_width, cy + uy * half_len - py * half_width),
        (cx + ux * half_len + px * half_width, cy + uy * half_len + py * half_width),
        (cx - ux * half_len + px * half_width, cy - uy * half_len + py * half_width),
    ]
    base_x, base_y = cx + ux * 21.0, cy + uy * 21.0
    tip_x, tip_y = cx + ux * 47.0, cy + uy * 47.0
    head = [
        (tip_x, tip_y),
        (base_x + px * 18.0, base_y + py * 18.0),
        (base_x - px * 18.0, base_y - py * 18.0),
    ]
    return shaft, head


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (247, 247, 243))
    draw = ImageDraw.Draw(image)
    sx, sy = int(scene["shift_x"]), int(scene["shift_y"])
    panel_centers = ((170, 133), (470, 133), (170, 353), (470, 353))
    shaft_color = tuple(int(value) for value in scene["shaft_color"])
    head_color = tuple(int(value) for value in scene["head_color"])
    font = ImageFont.load_default()
    for index, ((cx, cy), pair) in enumerate(zip(panel_centers, scene["panel_vectors"])):
        cx, cy = cx + sx, cy + sy
        draw.rounded_rectangle(
            (cx - 135, cy - 92, cx + 135, cy + 92),
            radius=13,
            fill=(255, 255, 255),
            outline=(158, 163, 168),
            width=3,
        )
        draw.text((cx - 125, cy - 80), LABELS[index], fill=(54, 57, 62), font=font)
        centers = ((cx - 58, cy + 5), (cx + 58, cy + 5))
        for arrow_center, vector in zip(centers, pair):
            shaft, head = _arrow_polygon(arrow_center, vector)
            draw.polygon(shaft, fill=shaft_color)
            draw.polygon(head, fill=head_color)
    return image


def translated(scene: dict, dx: int, dy: int) -> dict:
    twin = deepcopy(scene)
    twin["shift_x"] = int(twin["shift_x"]) + int(dx)
    twin["shift_y"] = int(twin["shift_y"]) + int(dy)
    return twin
