"""Orientation-guided count-comparison search world.

A thin dark reference bar sits in a box at the top of the image with one of four
canonical undirected orientations {0, 45, 90, 135} degrees. Below it, the image
is split by a faint vertical divider into a left and a right half. Each half
holds the same number K of thin dark bars, and every bar is drawn at one of the
same four canonical orientations.

A bar "matches" the reference when its undirected line orientation equals the
reference orientation (they are parallel, within a small tolerance). The
question asks which half, left or right, contains MORE bars matching the
reference orientation. The analytic decision counts the matching bars in each
half and reports the half with the strictly greater count.

Total bars per half is always equal, and every bar shares the same size, colour,
and stroke, so the answer cannot be recovered from total count, colour, size,
position, or any single-feature salience cue: only the reference-relative
orientation statistics of each half determine the decision. Rendering is fully
deterministic given the scene.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw

SIZE = 512
BG = (255, 255, 255)
INK = (30, 30, 30)
DIVIDER = (205, 205, 205)
SPLIT_X = 256

REF_CENTER = (256, 60)
REF_LENGTH = 100
BAR_WIDTH = 10

ORIENTS = (0.0, 45.0, 90.0, 135.0)
MATCH_TOL = 2.0
QUARANTINE_THRESH = 1.0

Y0, Y1 = 165, 440
LX0, LX1 = 55, 195
RX0, RX1 = 317, 457
K_BARS = 5
BAR_LEN = 55
MIN_SPACING = 75


def _angdiff(a: float, b: float) -> float:
    d = abs(a - b) % 180.0
    return min(d, 180.0 - d)


def _bar_points(cx: float, cy: float, length: float, angle: float) -> list[tuple[int, int]]:
    a = np.radians(angle)
    ux, uy = np.cos(a), np.sin(a)
    px, py = -np.sin(a), np.cos(a)
    hl, hw = length / 2.0, BAR_WIDTH / 2.0
    out = []
    for su, sv in ((1, 1), (1, -1), (-1, -1), (-1, 1)):
        x = cx + su * hl * ux + sv * hw * px
        y = cy + su * hl * uy + sv * hw * py
        out.append((int(round(x)), int(round(y))))
    return out


def _place(x0: float, x1: float, angles: list[float], rng) -> list[list[float]]:
    bars: list[list[float]] = []
    for angle in angles:
        for _ in range(20000):
            cx = float(rng.uniform(x0, x1))
            cy = float(rng.uniform(Y0, Y1))
            ok = True
            for (bx, by, *_rest) in bars:
                if np.hypot(cx - bx, cy - by) < MIN_SPACING:
                    ok = False
                    break
            if ok:
                bars.append([cx, cy, float(angle), BAR_LEN])
                break
        else:
            raise RuntimeError("placement failed")
    return bars


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(seed)
    ref_angle = float(ORIENTS[int(rng.integers(0, len(ORIENTS)))])
    winner_is_left = bool(int(rng.integers(0, 2)) == 0)
    n_win = int(rng.integers(2, K_BARS + 1))
    n_lose = int(rng.integers(0, n_win))
    others = [o for o in ORIENTS if abs(o - ref_angle) > 1e-9]

    def angles_list(n_match: int) -> list[float]:
        res = [ref_angle] * n_match
        for _ in range(K_BARS - n_match):
            res.append(float(others[int(rng.integers(0, len(others))) % len(others)]))
        rng.shuffle(res)
        return res

    left_angles = angles_list(n_win if winner_is_left else n_lose)
    right_angles = angles_list(n_lose if winner_is_left else n_win)
    left_bars = _place(LX0, LX1, left_angles, rng)
    right_bars = _place(RX0, RX1, right_angles, rng)
    return {"ref_angle": ref_angle, "left_bars": left_bars, "right_bars": right_bars}


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (SIZE, SIZE), BG)
    draw = ImageDraw.Draw(img)
    draw.rectangle([140, 8, 372, 112], outline=DIVIDER, width=2)
    draw.line([(SPLIT_X, 110), (SPLIT_X, SIZE)], fill=DIVIDER, width=2)
    cx, cy = REF_CENTER
    draw.polygon(_bar_points(cx, cy, REF_LENGTH, scene["ref_angle"]), fill=INK)
    for bar in scene["left_bars"] + scene["right_bars"]:
        bx, by, ang, ln = bar
        draw.polygon(_bar_points(bx, by, ln, ang), fill=INK)
    return img


def _counts(scene: dict) -> tuple[int, int]:
    ref = scene["ref_angle"]
    left_match = sum(1 for b in scene["left_bars"] if _angdiff(b[2], ref) < MATCH_TOL)
    right_match = sum(1 for b in scene["right_bars"] if _angdiff(b[2], ref) < MATCH_TOL)
    return left_match, right_match


def analytic_gold(scene: dict) -> str:
    left_match, right_match = _counts(scene)
    if left_match > right_match:
        return "left"
    if right_match > left_match:
        return "right"
    return "left"


def margin(scene: dict) -> float:
    left_match, right_match = _counts(scene)
    return float(abs(left_match - right_match))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_THRESH


def latent_symmetries(scene: dict) -> list:
    return []
