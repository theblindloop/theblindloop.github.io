"""Independent pixel-only inverse arm for orientation_match_count.

Receives only a final PNG (and the public question contract). It thresholds the
dark bar pixels, recovers the reference bar's undirected orientation from the
top strip, and then counts, in each half of the main area, the thin bars whose
undirected orientation lies within a tolerance of the reference orientation. It
returns the half ("left" or "right") with the greater number of matching bars,
or "NONE" when the reference cannot be measured or the counts are tied.

It imports nothing from the renderer, prompts, generator, or verifier, and it
accepts no latent scene data.
"""

from __future__ import annotations

import numpy as np

REF_STRIP_Y = 105
MAIN_Y = 115
SPLIT_X = 256
DARK_THRESH = 128
MATCH_TOL = 22.5
MIN_COMP = 30


def _angdiff(a: float, b: float) -> float:
    d = abs(a - b) % 180.0
    return min(d, 180.0 - d)


def _orientation(py: np.ndarray, px: np.ndarray) -> float:
    m = np.stack([px.astype(float), py.astype(float)], axis=1)
    c = m.mean(axis=0)
    mm = m - c
    cov = mm.T @ mm
    w, v = np.linalg.eigh(cov)
    vec = v[:, int(np.argmax(w))]
    ang = np.degrees(np.arctan2(vec[1], vec[0])) % 180.0
    return ang


def _components(dark: np.ndarray) -> list[tuple[np.ndarray, np.ndarray]]:
    ys, xs = np.nonzero(dark)
    n = len(xs)
    if n == 0:
        return []
    parent = list(range(n))

    def find(a: int) -> int:
        while parent[a] != a:
            parent[a] = parent[parent[a]]
            a = parent[a]
        return a

    idx: dict = {}
    for i in range(n):
        idx[(int(ys[i]), int(xs[i]))] = i
    for i in range(n):
        y = int(ys[i])
        x = int(xs[i])
        for dy in (-1, 0, 1):
            for dx in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                j = idx.get((y + dy, x + dx))
                if j is not None:
                    ra, rb = find(i), find(j)
                    if ra != rb:
                        parent[ra] = rb
    groups: dict[int, list[int]] = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(i)
    comps = []
    for ids in groups.values():
        comps.append((ys[list(ids)], xs[list(ids)]))
    return comps


def decision_from_image(image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    lum = arr.mean(axis=2)
    dark = lum < DARK_THRESH

    ref_y, ref_x = np.nonzero(dark[:REF_STRIP_Y, :])
    if len(ref_x) < MIN_COMP:
        return "NONE"
    ref_ang = _orientation(ref_y, ref_x)

    main = dark.copy()
    main[:MAIN_Y, :] = False
    counts = {"left": 0, "right": 0}
    for py, px in _components(main):
        if len(px) < MIN_COMP:
            continue
        cx = float(px.mean())
        side = "left" if cx < SPLIT_X else "right"
        ang = _orientation(py, px)
        if _angdiff(ang, ref_ang) < MATCH_TOL:
            counts[side] += 1

    if counts["left"] > counts["right"]:
        return "left"
    if counts["right"] > counts["left"]:
        return "right"
    return "NONE"
