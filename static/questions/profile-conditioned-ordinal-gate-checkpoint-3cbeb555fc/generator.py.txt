"""Analytic sampler and Pillow renderer for the repeated-gate tracing world."""

from __future__ import annotations

import copy
import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

COLORS = ("red", "blue", "orange")
RGB = {
    "red": (220, 45, 55),
    "blue": (45, 105, 220),
    "orange": (240, 145, 35),
    "green": (35, 175, 85),
    "purple": (145, 75, 200),
}
GATE_RGB = (150, 150, 150)
INK = (25, 28, 32)
WHITE = (252, 252, 250)
QUARANTINE_PX = 28.0


def _segment_length(a: list[float], b: list[float]) -> float:
    return math.hypot(b[0] - a[0], b[1] - a[1])


def _cumulative(route: list[list[float]]) -> list[float]:
    out = [0.0]
    for a, b in zip(route, route[1:]):
        out.append(out[-1] + _segment_length(a, b))
    return out


def _point_at(route: list[list[float]], distance: float) -> list[float]:
    cumulative = _cumulative(route)
    if not 0.0 <= distance <= cumulative[-1]:
        raise ValueError("route distance out of range")
    for i, (lo, hi) in enumerate(zip(cumulative, cumulative[1:])):
        if distance <= hi + 1e-9:
            t = 0.0 if hi == lo else (distance - lo) / (hi - lo)
            return [route[i][0] + t * (route[i + 1][0] - route[i][0]),
                    route[i][1] + t * (route[i + 1][1] - route[i][1])]
    return list(route[-1])


def _crossing_distances(route: list[list[float]], gate: dict) -> list[float]:
    axis = gate["axis"]
    coordinate = float(gate["coordinate"])
    cumulative = _cumulative(route)
    result = []
    dim = 0 if axis == "vertical" else 1
    for index, (a, b) in enumerate(zip(route, route[1:])):
        av, bv = float(a[dim]), float(b[dim])
        if (av - coordinate) * (bv - coordinate) < 0.0:
            t = (coordinate - av) / (bv - av)
            result.append(cumulative[index] + t * _segment_length(a, b))
    return result


def _route_measurements(scene: dict) -> tuple[list[float], list[tuple[str, float]]]:
    route = scene["route"]
    gate = scene["gate"]
    markers = scene["markers"]
    crossings = _crossing_distances(route, gate)
    ordered = [(str(item["color"]), float(item["distance"])) for item in markers]
    return crossings, ordered


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    width, height = 720, 480
    left = rng.randint(62, 92)
    right = rng.randint(628, 658)
    base_ys = [55, 145, 235, 325, 415]
    ys = [base + rng.randint(-9, 9) for base in base_ys]
    route = [
        [left, ys[0]], [right, ys[0]], [right, ys[1]], [left, ys[1]],
        [left, ys[2]], [right, ys[2]], [right, ys[3]], [left, ys[3]],
        [left, ys[4]], [right, ys[4]],
    ]
    gate = {"axis": "vertical", "coordinate": rng.randint(330, 390)}

    if rng.randrange(2):
        route = [[width - x, y] for x, y in route]
        gate["coordinate"] = width - gate["coordinate"]
    if rng.randrange(2):
        route = [[x, height - y] for x, y in route]

    crossings = _crossing_distances(route, gate)
    if len(crossings) != 5:
        raise AssertionError("sampler must create five crossings")
    before_gap = 46.0 + (seed % 5) * 11.0
    after_gap = 51.0 + ((seed // 5) % 5) * 10.0
    target_fraction = (0.14, 0.28, 0.50, 0.72, 0.86)[seed % 5]
    distances = [
        crossings[1] - before_gap,
        crossings[1] + target_fraction * (crossings[2] - crossings[1]),
        crossings[2] + after_gap,
    ]
    answer = COLORS[seed % len(COLORS)]
    decoys = [color for color in COLORS if color != answer]
    if rng.randrange(2):
        decoys.reverse()
    assigned = [decoys[0], answer, decoys[1]]
    markers = [
        {"color": color, "distance": distance, "point": _point_at(route, distance)}
        for color, distance in zip(assigned, distances)
    ]

    occupied = [tuple(point) for point in route] + [tuple(item["point"]) for item in markers]
    decorations = []
    for _ in range(3):
        for _attempt in range(200):
            point = [rng.randint(24, width - 24), rng.randint(24, height - 24)]
            if all(math.hypot(point[0] - x, point[1] - y) > 38 for x, y in occupied):
                decorations.append(point)
                occupied.append(tuple(point))
                break
    return {
        "width": width,
        "height": height,
        "route": route,
        "gate": gate,
        "markers": markers,
        "style": {"route_width": 11, "gate_width": 5, "checkpoint_radius": 14,
                  "endpoint_radius": 18},
        "decorations": decorations,
    }


def analytic_gold(scene: dict) -> str:
    crossings, markers = _route_measurements(scene)
    if len(crossings) < 3:
        raise ValueError("fewer than three proper gate crossings")
    inside = [color for color, distance in markers if crossings[1] < distance < crossings[2]]
    if len(inside) != 1:
        raise ValueError("scene has no unique checkpoint in the ordinal interval")
    return inside[0]


def margin(scene: dict) -> float:
    crossings, markers = _route_measurements(scene)
    relevant = crossings[1:3]
    return min(abs(distance - crossing) for _color, distance in markers for crossing in relevant)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["decorations"] = list(reversed(twin["decorations"]))
    return [("reverse_disjoint_decorations", twin)]


def _draw_centered(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str) -> None:
    font = ImageFont.load_default()
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2),
              text, fill="white", font=font)


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    route = scene["route"]
    gate = scene["gate"]
    markers = scene["markers"]
    style = scene["style"]
    decorations = scene["decorations"]
    image = Image.new("RGB", (width, height), WHITE)
    draw = ImageDraw.Draw(image)
    for x, y in decorations:
        draw.ellipse((x - 5, y - 5, x + 5, y + 5), fill=(220, 220, 216))
    if gate["axis"] == "vertical":
        x = float(gate["coordinate"])
        draw.line((x, 18, x, height - 18), fill=GATE_RGB, width=int(style["gate_width"]))
    else:
        y = float(gate["coordinate"])
        draw.line((18, y, width - 18, y), fill=GATE_RGB, width=int(style["gate_width"]))
    draw.line([tuple(point) for point in route], fill=INK, width=int(style["route_width"]), joint="curve")
    radius = int(style["checkpoint_radius"])
    for item in markers:
        x, y = item["point"]
        draw.ellipse((x - radius, y - radius, x + radius, y + radius),
                     fill=RGB[item["color"]], outline=INK, width=3)
    endpoint_radius = int(style["endpoint_radius"])
    for point, color, label in ((route[0], "green", "S"), (route[-1], "purple", "F")):
        x, y = point
        draw.ellipse((x - endpoint_radius, y - endpoint_radius,
                      x + endpoint_radius, y + endpoint_radius),
                     fill=RGB[color], outline=INK, width=3)
        _draw_centered(draw, (x, y), label)
    return image
