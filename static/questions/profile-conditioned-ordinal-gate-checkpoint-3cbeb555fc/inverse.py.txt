"""Independent PNG-only inverse arm for ordinal gate crossings."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PROTOTYPES = {
    "red": np.array((220, 45, 55), dtype=np.int32),
    "blue": np.array((45, 105, 220), dtype=np.int32),
    "orange": np.array((240, 145, 35), dtype=np.int32),
    "green": np.array((35, 175, 85), dtype=np.int32),
    "purple": np.array((145, 75, 200), dtype=np.int32),
}


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = [(int(y), int(x))]
        seen[y, x] = True
        pixels = []
        while queue:
            cy, cx = queue.pop()
            pixels.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        result.append(np.asarray(pixels, dtype=np.int32))
    return result


def _distance_map(mask: np.ndarray, sources: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    distance = np.full((height, width), -1, dtype=np.int32)
    queue = deque()
    for y, x in sources:
        if mask[y, x] and distance[y, x] < 0:
            distance[y, x] = 0
            queue.append((int(y), int(x)))
    while queue:
        y, x = queue.popleft()
        nd = int(distance[y, x]) + 1
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and distance[ny, nx] < 0:
                distance[ny, nx] = nd
                queue.append((ny, nx))
    return distance


def _median_distance(distance: np.ndarray, pixels: np.ndarray) -> float:
    values = distance[pixels[:, 0], pixels[:, 1]]
    values = values[values >= 0]
    if len(values) == 0:
        raise ValueError("visible feature is disconnected from START route")
    return float(np.median(values))


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int32)
    height, width, _ = array.shape
    color_masks = {
        name: np.sum((array - rgb) ** 2, axis=2) < 38 * 38
        for name, rgb in PROTOTYPES.items()
    }
    main_components = {}
    for name, mask in color_masks.items():
        components = [component for component in _components(mask) if len(component) > 120]
        if len(components) != 1:
            raise ValueError(f"expected one visible {name} marker")
        main_components[name] = components[0]

    dark = np.max(array, axis=2) < 75
    saturated = np.zeros((height, width), dtype=bool)
    for mask in color_masks.values():
        saturated |= mask
    traversable = dark | saturated
    distance = _distance_map(traversable, main_components["green"])
    if not np.any(distance[main_components["purple"][:, 0], main_components["purple"][:, 1]] >= 0):
        raise ValueError("START and FINISH are not joined by one visible route")

    gray = (np.max(array, axis=2) - np.min(array, axis=2) <= 2) & (array[:, :, 0] >= 135) & (array[:, :, 0] <= 165)
    column_counts = gray.sum(axis=0)
    row_counts = gray.sum(axis=1)
    if int(column_counts.max()) >= int(row_counts.max()):
        coordinate = int(np.argmax(column_counts))
        band = np.zeros_like(traversable)
        band[:, max(0, coordinate - 4):min(width, coordinate + 5)] = traversable[:, max(0, coordinate - 4):min(width, coordinate + 5)]
    else:
        coordinate = int(np.argmax(row_counts))
        band = np.zeros_like(traversable)
        band[max(0, coordinate - 4):min(height, coordinate + 5), :] = traversable[max(0, coordinate - 4):min(height, coordinate + 5), :]
    crossing_components = [component for component in _components(band) if len(component) >= 35]
    crossing_distances = sorted(_median_distance(distance, component) for component in crossing_components)
    if len(crossing_distances) != 5:
        raise ValueError(f"expected five visible proper gate crossings, found {len(crossing_distances)}")
    lower, upper = crossing_distances[1], crossing_distances[2]
    inside = []
    for color in ("red", "blue", "orange"):
        marker_distance = _median_distance(distance, main_components[color])
        if lower < marker_distance < upper:
            inside.append(color)
    if len(inside) != 1:
        raise ValueError("pixel route has no unique checkpoint in the ordinal interval")
    return inside[0]
