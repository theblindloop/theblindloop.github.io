"""Pixel-only oracle: rebuild the containment depth of every coloured dot.

The image is the only input.  The oracle segments the four declared dot colours,
then reconstructs each dot's containment depth from the ink crossings along many
rays that leave the image.  For closed convex outlines every enclosing outline is
crossed an odd number of times and every other outline an even number of times,
so the crossing count has fixed parity and its minimum over clean rays equals the
depth.  The parity vote removes tangential grazes.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

COLOUR_NAMES = ("red", "green", "blue", "orange")
REFERENCE_RGB = {
    "red": (214, 38, 38),
    "green": (26, 140, 62),
    "blue": (36, 84, 210),
    "orange": (236, 142, 18),
}
INK_MAX = 110          # a pixel darker than this in every channel is outline ink
COLOUR_TOL = 70.0      # euclidean RGB tolerance for dot segmentation
N_RAYS = 144
ABSTAIN = "abstain"


def _dot_centres(rgb: np.ndarray) -> dict[str, tuple[float, float]]:
    centres: dict[str, tuple[float, float]] = {}
    flat = rgb.reshape(-1, 3).astype(np.float64)
    for name, ref in REFERENCE_RGB.items():
        dist = np.linalg.norm(flat - np.asarray(ref, dtype=np.float64), axis=1)
        hit = dist < COLOUR_TOL
        if int(hit.sum()) < 40:
            continue
        idx = np.flatnonzero(hit)
        ys, xs = np.divmod(idx, rgb.shape[1])
        centres[name] = (float(xs.mean()), float(ys.mean()))
    return centres


def _crossings(ink: np.ndarray, cx: float, cy: float, angle: float, start: float) -> int:
    height, width = ink.shape
    reach = float(math.hypot(width, height)) + 4.0
    steps = np.arange(start, reach, 0.5)
    xs = np.rint(cx + steps * math.cos(angle)).astype(np.int64)
    ys = np.rint(cy + steps * math.sin(angle)).astype(np.int64)
    ok = (xs >= 0) & (xs < width) & (ys >= 0) & (ys < height)
    line = ink[ys[ok], xs[ok]]
    if line.size == 0:
        return 0
    return int(np.count_nonzero(line[1:] & ~line[:-1]) + int(line[0]))


def _depth(ink: np.ndarray, cx: float, cy: float) -> int | None:
    counts = [
        _crossings(ink, cx, cy, 2.0 * math.pi * k / N_RAYS + 0.013, 12.0)
        for k in range(N_RAYS)
    ]
    even = sum(1 for c in counts if c % 2 == 0)
    if even == len(counts) - even:
        return None
    parity = 0 if even > len(counts) - even else 1
    keep = [c for c in counts if c % 2 == parity]
    return min(keep) if keep else None


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    ink = np.all(rgb < INK_MAX, axis=2)
    centres = _dot_centres(rgb)
    if len(centres) != len(COLOUR_NAMES):
        return ABSTAIN
    depths: dict[str, int] = {}
    for name, (cx, cy) in centres.items():
        value = _depth(ink, cx, cy)
        if value is None:
            return ABSTAIN
        depths[name] = value
    ranked = sorted(depths.items(), key=lambda kv: kv[1], reverse=True)
    if ranked[0][1] == ranked[1][1]:
        return ABSTAIN
    return ranked[0][0]
