"""Independent PNG-only inverse for conjunctive compound-key addressing."""

from __future__ import annotations

from collections import Counter

import numpy as np
from PIL import Image

WIDTH = 820
HEIGHT = 610
KEY_Y = tuple(95 + 51 * index for index in range(8))
GRID_X = 500
GRID_Y = 118
CELL = 70
BOARD_CENTERS_X = (535, 605, 675, 745)
BOARD_CENTERS_Y = (153, 223, 293, 363)

ROW_SYMBOLS = {
    "1111100010001111",
    "1001011001101001",
    "1111011001100110",
    "1000100010001111",
}
COLUMN_STAMPS = {
    "1111100110011111",
    "0110111101100110",
    "0001001001001000",
    "1000110011101111",
}
BACKGROUND = (248, 247, 242)
STAMP_BG = (255, 255, 255)
INK = (29, 37, 51)
STAMP_INK = (27, 74, 105)
GUIDE = (89, 101, 116)
ROW_RGB = {
    "red": (214, 68, 62),
    "blue": (48, 105, 196),
    "green": (42, 149, 91),
    "orange": (226, 126, 31),
}
DISK_RGB = {
    "magenta": (187, 52, 139),
    "teal": (22, 151, 158),
    "gold": (220, 174, 34),
    "purple": (118, 72, 184),
}
ANSWER_RGB = {
    "amber": (243, 164, 33),
    "cyan": (36, 181, 202),
    "violet": (143, 88, 203),
    "lime": (119, 184, 59),
}


class OracleError(ValueError):
    """Raised when the raster does not encode one intact worksheet."""


def _require_uniform(
    array: np.ndarray,
    left: int,
    top: int,
    right: int,
    bottom: int,
    color: tuple[int, int, int],
    what: str,
) -> None:
    patch = array[top:bottom, left:right]
    if patch.size == 0 or not np.all(patch == np.asarray(color, dtype=np.uint8)):
        raise OracleError(f"{what} is missing or damaged")


def _identify_uniform(
    array: np.ndarray,
    box: tuple[int, int, int, int],
    palette: dict[str, tuple[int, int, int]],
    what: str,
) -> str:
    left, top, right, bottom = box
    patch = array[top:bottom, left:right]
    matches = [name for name, color in palette.items() if np.all(patch == np.asarray(color, dtype=np.uint8))]
    if len(matches) != 1:
        raise OracleError(f"{what} has no unique declared color")
    return matches[0]


def _read_stamp(
    array: np.ndarray,
    center_x: int,
    center_y: int,
    ink: tuple[int, int, int],
    alphabet: set[str],
    what: str,
) -> str:
    _require_uniform(array, center_x - 18, center_y - 22, center_x + 19, center_y - 20, GUIDE, f"{what} top frame")
    _require_uniform(array, center_x - 18, center_y + 21, center_x + 19, center_y + 23, GUIDE, f"{what} bottom frame")
    values: list[str] = []
    start_x = center_x - 16
    start_y = center_y - 16
    ink_array = np.asarray(ink, dtype=np.uint8)
    white_array = np.asarray(STAMP_BG, dtype=np.uint8)
    for row in range(4):
        for column in range(4):
            left = start_x + column * 9
            top = start_y + row * 9
            patch = array[top : top + 7, left : left + 7]
            if np.all(patch == ink_array):
                values.append("1")
            elif np.all(patch == white_array):
                values.append("0")
            else:
                raise OracleError(f"{what} contains an ambiguous stamp cell")
    bits = "".join(values)
    if bits not in alphabet:
        raise OracleError(f"{what} is not a member of its visible alphabet")
    return bits


def _validate_board(array: np.ndarray) -> None:
    for x, y in ((0, 0), (WIDTH - 1, 0), (0, HEIGHT - 1), (WIDTH - 1, HEIGHT - 1)):
        if tuple(int(value) for value in array[y, x]) != BACKGROUND:
            raise OracleError("canvas corners do not contain the declared background")
    for row in range(4):
        for column in range(4):
            left = GRID_X + column * CELL
            top = GRID_Y + row * CELL
            _require_uniform(array, left, top, left + CELL, top + 4, INK, "grid horizontal frame")
            _require_uniform(array, left, top, left + 4, top + CELL, INK, "grid vertical frame")


def decision_from_image(image: Image.Image) -> str:
    """Recover the compound-key match and its two address bindings from pixels."""

    if image.size != (WIDTH, HEIGHT):
        raise OracleError(f"expected a {WIDTH} by {HEIGHT} raster")
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.shape != (HEIGHT, WIDTH, 3):
        raise OracleError("expected one RGB worksheet")
    _validate_board(array)

    pair_rows: list[tuple[str, str, str, str]] = []
    for index, center_y in enumerate(KEY_Y):
        symbol = _read_stamp(array, 65, center_y, INK, ROW_SYMBOLS, f"key symbol {index}")
        disk = _identify_uniform(array, (126, center_y - 6, 137, center_y + 7), DISK_RGB, f"key disk {index}")
        row_tag = _identify_uniform(array, (236, center_y - 8, 269, center_y + 9), ROW_RGB, f"key row output {index}")
        stamp = _read_stamp(array, 335, center_y, STAMP_INK, COLUMN_STAMPS, f"key column output {index}")
        pair_rows.append((symbol, disk, row_tag, stamp))
    if Counter(symbol for symbol, _, _, _ in pair_rows) != Counter({symbol: 2 for symbol in ROW_SYMBOLS}):
        raise OracleError("each input symbol must occur exactly twice")
    if Counter(disk for _, disk, _, _ in pair_rows) != Counter({disk: 2 for disk in DISK_RGB}):
        raise OracleError("each input disk must occur exactly twice")
    if len({(symbol, disk) for symbol, disk, _, _ in pair_rows}) != 8:
        raise OracleError("compound cue pairs are not unique")
    for symbol in ROW_SYMBOLS:
        matches = [row for row in pair_rows if row[0] == symbol]
        if matches[0][2] == matches[1][2] or matches[0][3] == matches[1][3]:
            raise OracleError("a symbol projection is not visibly ambiguous")
    for disk in DISK_RGB:
        matches = [row for row in pair_rows if row[1] == disk]
        if matches[0][2] == matches[1][2] or matches[0][3] == matches[1][3]:
            raise OracleError("a disk projection is not visibly ambiguous")

    board_rows = [
        _identify_uniform(array, (461, center_y - 12, 481, center_y + 13), ROW_RGB, f"board row {index}")
        for index, center_y in enumerate(BOARD_CENTERS_Y)
    ]
    if set(board_rows) != set(ROW_RGB):
        raise OracleError("board row headers are not a permutation")
    board_columns = [
        _read_stamp(array, center_x, 86, STAMP_INK, COLUMN_STAMPS, f"board column {index}")
        for index, center_x in enumerate(BOARD_CENTERS_X)
    ]
    if set(board_columns) != COLUMN_STAMPS:
        raise OracleError("board column headers are not a permutation")

    cells: list[list[str]] = []
    for row, center_y in enumerate(BOARD_CENTERS_Y):
        cells.append(
            [
                _identify_uniform(
                    array,
                    (center_x - 18, center_y - 18, center_x + 19, center_y + 19),
                    ANSWER_RGB,
                    f"grid cell {row},{column}",
                )
                for column, center_x in enumerate(BOARD_CENTERS_X)
            ]
        )
    if any(set(row) != set(ANSWER_RGB) for row in cells):
        raise OracleError("each grid row must visibly contain all answer colors")

    query_symbol = _read_stamp(array, 565, 525, INK, ROW_SYMBOLS, "query symbol")
    query_disk = _identify_uniform(array, (684, 518, 699, 533), DISK_RGB, "query disk")
    matches = [row for row in pair_rows if row[:2] == (query_symbol, query_disk)]
    if len(matches) != 1:
        raise OracleError("query pair does not select exactly one visible key row")
    row_tag, column_stamp = matches[0][2], matches[0][3]
    return cells[board_rows.index(row_tag)][board_columns.index(column_stamp)]
