"""Deterministic renderer for conjunctive compound-key addressing."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 820
HEIGHT = 610
KEY_Y = tuple(95 + 51 * index for index in range(8))
GRID_X = 500
GRID_Y = 118
CELL = 70
BOARD_CENTERS_X = tuple(GRID_X + CELL * column + CELL // 2 for column in range(4))
BOARD_CENTERS_Y = tuple(GRID_Y + CELL * row + CELL // 2 for row in range(4))

ROW_SYMBOLS = (
    "1111100010001111",  # squared C
    "1001011001101001",  # X
    "1111011001100110",  # T
    "1000100010001111",  # L
)
COLUMN_STAMPS = (
    "1111100110011111",  # ring
    "0110111101100110",  # plus
    "0001001001001000",  # rising diagonal
    "1000110011101111",  # wedge
)
ROW_TAGS = ("red", "blue", "green", "orange")
DISK_TAGS = ("magenta", "teal", "gold", "purple")
ANSWERS = ("amber", "cyan", "violet", "lime")

RGB = {
    "background": (248, 247, 242),
    "panel": (232, 236, 240),
    "panel_edge": (124, 137, 151),
    "stamp_bg": (255, 255, 255),
    "ink": (29, 37, 51),
    "stamp_ink": (27, 74, 105),
    "guide": (89, 101, 116),
    "red": (214, 68, 62),
    "blue": (48, 105, 196),
    "green": (42, 149, 91),
    "orange": (226, 126, 31),
    "magenta": (187, 52, 139),
    "teal": (22, 151, 158),
    "gold": (220, 174, 34),
    "purple": (118, 72, 184),
    "amber": (243, 164, 33),
    "cyan": (36, 181, 202),
    "violet": (143, 88, 203),
    "lime": (119, 184, 59),
}


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size=size)
    except OSError:
        return ImageFont.load_default()


def _shuffled(rng: random.Random, values: tuple[str, ...]) -> list[str]:
    result = list(values)
    rng.shuffle(result)
    return result


def sample_scene(seed: int) -> dict:
    """Sample an eight-edge degree-two cue relation and a balanced board."""

    seed = int(seed)
    rng = random.Random((seed ^ 0xC04A91E5) & 0xFFFFFFFFFFFFFFFF)
    symbols = _shuffled(rng, ROW_SYMBOLS)
    disks = _shuffled(rng, DISK_TAGS)
    board_rows = _shuffled(rng, ROW_TAGS)
    board_columns = _shuffled(rng, COLUMN_STAMPS)

    # These eight compound keys form one bipartite cycle. Every projected cue
    # occurs twice, but every ordered pair occurs once. The two records sharing
    # either cue have different row and column outputs.
    unshuffled_rows: list[list[str]] = []
    for index in range(4):
        unshuffled_rows.append(
            [symbols[index], disks[index], board_rows[index], board_columns[index]]
        )
        unshuffled_rows.append(
            [
                symbols[index],
                disks[(index + 1) % 4],
                board_rows[(index + 2) % 4],
                board_columns[(index + 3) % 4],
            ]
        )

    query_row = list(unshuffled_rows[(seed * 5 + 1) % 8])
    pair_rows = [list(row) for row in unshuffled_rows]
    rng.shuffle(pair_rows)

    target_row = board_rows.index(query_row[2])
    target_column = board_columns.index(query_row[3])
    palette = _shuffled(rng, ANSWERS)
    desired = ANSWERS[seed % 4]
    offset = (palette.index(desired) - target_row - target_column) % 4
    cells = [
        [palette[(row + column + offset) % 4] for column in range(4)]
        for row in range(4)
    ]

    return {
        "pair_rows": pair_rows,
        "board_rows": board_rows,
        "board_columns": board_columns,
        "cells": cells,
        "query_symbol": query_row[0],
        "query_disk": query_row[1],
    }


def _draw_stamp(
    draw: ImageDraw.ImageDraw,
    center_x: int,
    center_y: int,
    bits: str,
    ink: tuple[int, int, int],
) -> None:
    draw.rectangle(
        (center_x - 22, center_y - 22, center_x + 22, center_y + 22),
        fill=RGB["stamp_bg"],
        outline=RGB["guide"],
        width=2,
    )
    start_x = center_x - 16
    start_y = center_y - 16
    for row in range(4):
        for column in range(4):
            if bits[row * 4 + column] == "1":
                left = start_x + column * 9
                top = start_y + row * 9
                draw.rectangle((left, top, left + 6, top + 6), fill=ink)


def _draw_arrow(draw: ImageDraw.ImageDraw, x1: int, y: int, x2: int) -> None:
    draw.line((x1, y, x2, y), fill=RGB["guide"], width=3)
    draw.polygon(((x2, y), (x2 - 8, y - 5), (x2 - 8, y + 5)), fill=RGB["guide"])


def _validate_scene(scene: dict) -> None:
    if set(scene) != {
        "pair_rows",
        "board_rows",
        "board_columns",
        "cells",
        "query_symbol",
        "query_disk",
    }:
        raise ValueError("scene fields do not match the world schema")
    rows = scene["pair_rows"]
    if not isinstance(rows, list) or len(rows) != 8:
        raise ValueError("pair key must contain eight rows")
    if any(not isinstance(row, list) or len(row) != 4 for row in rows):
        raise ValueError("every pair-key row must have four fields")
    normalized = [tuple(str(value) for value in row) for row in rows]
    if any(symbol not in ROW_SYMBOLS for symbol, _, _, _ in normalized):
        raise ValueError("unknown input symbol")
    if any(disk not in DISK_TAGS for _, disk, _, _ in normalized):
        raise ValueError("unknown input disk")
    if any(tag not in ROW_TAGS for _, _, tag, _ in normalized):
        raise ValueError("unknown output row tag")
    if any(stamp not in COLUMN_STAMPS for _, _, _, stamp in normalized):
        raise ValueError("unknown output column stamp")
    if len({(symbol, disk) for symbol, disk, _, _ in normalized}) != 8:
        raise ValueError("compound cue pairs must be unique")
    for symbol in ROW_SYMBOLS:
        matches = [row for row in normalized if row[0] == symbol]
        if len(matches) != 2 or matches[0][2] == matches[1][2] or matches[0][3] == matches[1][3]:
            raise ValueError("each symbol must have two conflicting address records")
    for disk in DISK_TAGS:
        matches = [row for row in normalized if row[1] == disk]
        if len(matches) != 2 or matches[0][2] == matches[1][2] or matches[0][3] == matches[1][3]:
            raise ValueError("each disk must have two conflicting address records")
    if sorted(str(value) for value in scene["board_rows"]) != sorted(ROW_TAGS):
        raise ValueError("board row tags must be a permutation")
    if sorted(str(value) for value in scene["board_columns"]) != sorted(COLUMN_STAMPS):
        raise ValueError("board column stamps must be a permutation")
    cells = scene["cells"]
    if not isinstance(cells, list) or len(cells) != 4:
        raise ValueError("board must have four rows")
    if any(not isinstance(row, list) or len(row) != 4 for row in cells):
        raise ValueError("board rows must have four cells")
    if any({str(value) for value in row} != set(ANSWERS) for row in cells):
        raise ValueError("each board row must contain all answer colors")
    query = (str(scene["query_symbol"]), str(scene["query_disk"]))
    if sum((symbol, disk) == query for symbol, disk, _, _ in normalized) != 1:
        raise ValueError("query must match exactly one compound-key row")


def render(scene: dict) -> Image.Image:
    """Render one fixed-layout worksheet with large, repeated visible marks."""

    _validate_scene(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), RGB["background"])
    draw = ImageDraw.Draw(image)
    font_small = _font(13, bold=True)
    font_title = _font(18, bold=True)
    font_plus = _font(21, bold=True)

    draw.rounded_rectangle((16, 18, 420, 490), radius=10, fill=RGB["panel"], outline=RGB["panel_edge"], width=2)
    draw.text((30, 27), "PAIR KEY — match BOTH marks", fill=RGB["ink"], font=font_title)
    draw.text((46, 57), "cue pair", fill=RGB["ink"], font=font_small)
    draw.text((229, 57), "bound grid address", fill=RGB["ink"], font=font_small)
    for center_y, row in zip(KEY_Y, scene["pair_rows"], strict=True):
        symbol, disk, row_tag, column_stamp = (str(value) for value in row)
        _draw_stamp(draw, 65, center_y, symbol, RGB["ink"])
        draw.text((96, center_y - 13), "+", fill=RGB["ink"], font=font_plus)
        draw.ellipse((118, center_y - 13, 144, center_y + 13), fill=RGB[disk], outline=RGB["ink"], width=2)
        _draw_arrow(draw, 158, center_y, 210)
        draw.rectangle((230, center_y - 14, 274, center_y + 14), fill=RGB[row_tag], outline=RGB["ink"], width=2)
        draw.text((284, center_y - 13), "+", fill=RGB["ink"], font=font_plus)
        _draw_stamp(draw, 335, center_y, column_stamp, RGB["stamp_ink"])

    draw.text((500, 27), "ADDRESS GRID", fill=RGB["ink"], font=font_title)
    for column, stamp in enumerate(scene["board_columns"]):
        _draw_stamp(draw, BOARD_CENTERS_X[column], 86, str(stamp), RGB["stamp_ink"])
    for row, tag in enumerate(scene["board_rows"]):
        center_y = BOARD_CENTERS_Y[row]
        draw.rectangle((455, center_y - 18, 486, center_y + 18), fill=RGB[str(tag)], outline=RGB["ink"], width=2)
    for row in range(4):
        for column in range(4):
            left = GRID_X + column * CELL
            top = GRID_Y + row * CELL
            draw.rectangle((left, top, left + CELL - 1, top + CELL - 1), fill=RGB["ink"])
            draw.rectangle((left + 4, top + 4, left + CELL - 5, top + CELL - 5), fill=RGB[str(scene["cells"][row][column])])

    draw.rounded_rectangle((455, 435, 800, 590), radius=10, fill=RGB["panel"], outline=RGB["panel_edge"], width=2)
    draw.text((475, 448), "QUERY — keep as one pair", fill=RGB["ink"], font=font_title)
    _draw_stamp(draw, 565, 525, str(scene["query_symbol"]), RGB["ink"])
    draw.text((615, 511), "+", fill=RGB["ink"], font=font_plus)
    draw.ellipse((674, 508, 708, 542), fill=RGB[str(scene["query_disk"])], outline=RGB["ink"], width=2)
    return image


def analytic_gold(scene: dict) -> str:
    """Select by the intact cue tuple, then use both bound address outputs."""

    _validate_scene(scene)
    query = (str(scene["query_symbol"]), str(scene["query_disk"]))
    matching = [row for row in scene["pair_rows"] if (str(row[0]), str(row[1])) == query]
    row_tag = str(matching[0][2])
    column_stamp = str(matching[0][3])
    row = [str(value) for value in scene["board_rows"]].index(row_tag)
    column = [str(value) for value in scene["board_columns"]].index(column_stamp)
    return str(scene["cells"][row][column])


def _hamming(left: str, right: str) -> int:
    return sum(a != b for a, b in zip(left, right, strict=True))


def margin(scene: dict) -> float:
    """Smallest visible separation within any repeated cue/header alphabet."""

    _validate_scene(scene)
    symbol_gap = min(_hamming(a, b) for index, a in enumerate(ROW_SYMBOLS) for b in ROW_SYMBOLS[index + 1 :])
    stamp_gap = min(_hamming(a, b) for index, a in enumerate(COLUMN_STAMPS) for b in COLUMN_STAMPS[index + 1 :])
    color_gap = min(
        math.dist(RGB[a], RGB[b]) / 32.0
        for palette in (ROW_TAGS, DISK_TAGS, ANSWERS)
        for index, a in enumerate(palette)
        for b in palette[index + 1 :]
    )
    return float(min(symbol_gap, stamp_gap, color_gap))


def is_quarantined(scene: dict) -> bool:
    try:
        _validate_scene(scene)
        analytic_gold(scene)
        return margin(scene) < 2.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The explicit ordered worksheet has no intended pixel-identical aliases."""

    return []
