"""Independent pixel-only inverse for the causal dependency-pruning board."""
from __future__ import annotations

import numpy as np
from PIL import Image

CANVAS = (960, 620)
BOARD_W, BOARD_H = 920, 580
BORDER = (55, 64, 72)
CANCEL = (204, 43, 55)
INK = (37, 43, 49)
REGISTERS = ("A", "B", "C", "D")
BADGES = {(198, 82, 52): "A", (22, 139, 150): "B", (121, 75, 180): "C", (65, 104, 174): "D"}
SHAPES_BY_RGB = {(42, 124, 202): "circle", (38, 157, 91): "triangle", (230, 158, 26): "star", (205, 69, 119): "diamond"}
SHAPES = ("circle", "triangle", "star", "diamond")
PAIRS = tuple((a, b) for i, a in enumerate(REGISTERS) for b in REGISTERS[i+1:])


class OracleError(ValueError):
    pass


def _patch(array: np.ndarray, x: int, y: int, radius: int = 2) -> tuple[int, int, int]:
    patch = array[y-radius:y+radius+1, x-radius:x+radius+1]
    colors = {tuple(int(v) for v in row) for row in patch.reshape(-1, 3)}
    if len(colors) != 1:
        raise OracleError("a required marker is damaged")
    return next(iter(colors))


def _shape(array: np.ndarray, x: int, y: int) -> str:
    color = _patch(array, x, y)
    if color not in SHAPES_BY_RGB:
        raise OracleError("unknown shape marker")
    return SHAPES_BY_RGB[color]


def _badge(array: np.ndarray, x: int, y: int) -> str:
    color = _patch(array, x-18, y)
    if color not in BADGES:
        raise OracleError("unknown register badge")
    return BADGES[color]


def _dependencies(pairs: list[tuple[str, str]]) -> list[set[int]]:
    last: dict[str, int] = {}
    deps = []
    for index, pair in enumerate(pairs):
        parents = {last[slot] for slot in pair if slot in last}
        deps.append(parents)
        for slot in pair:
            last[slot] = index
    return deps


def _removed(pairs: list[tuple[str, str]], cancel: int) -> tuple[set[int], dict[int, int]]:
    deps = _dependencies(pairs)
    removed = {cancel}
    depth = {cancel: 0}
    for index in range(cancel+1, len(pairs)):
        hit = deps[index] & removed
        if hit:
            removed.add(index)
            depth[index] = 1 + max(depth[parent] for parent in hit)
    return removed, depth


def _run(initial: list[str], pairs: list[tuple[str, str]], skipped: set[int]) -> list[str]:
    state = list(initial)
    for index, pair in enumerate(pairs):
        if index in skipped:
            continue
        a, b = (REGISTERS.index(slot) for slot in pair)
        state[a], state[b] = state[b], state[a]
    return state


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if (array.shape[1], array.shape[0]) != CANVAS:
        raise OracleError("unexpected canvas size")
    mask = np.all(array == np.asarray(BORDER, dtype=np.uint8), axis=2)
    points = np.argwhere(mask)
    if not len(points):
        raise OracleError("board absent")
    top, left = points.min(axis=0); bottom, right = points.max(axis=0)
    if int(right-left+1) != BOARD_W or int(bottom-top+1) != BOARD_H:
        raise OracleError("wrong board extent")
    ax, ay = int(left), int(top)
    if not (np.all(mask[ay, ax:ax+BOARD_W]) and np.all(mask[ay+BOARD_H-1, ax:ax+BOARD_W])
            and np.all(mask[ay:ay+BOARD_H, ax]) and np.all(mask[ay:ay+BOARD_H, ax+BOARD_W-1])):
        raise OracleError("broken board border")
    initial = [_shape(array, ax+x+12, ay+102) for x in (185, 355, 525, 695)]
    if sorted(initial) != sorted(SHAPES):
        raise OracleError("invalid START inventory")
    target = _badge(array, ax+850, ay+96)
    pairs: list[tuple[str, str]] = []
    selectors = []
    for index in range(7):
        x, y = ax+20+index*128, ay+300
        selectors.append(_patch(array, x+43, y+28))
        pair = (_badge(array, x+56, y+88), _badge(array, x+56, y+178))
        if pair not in PAIRS:
            raise OracleError("malformed swap order")
        pairs.append(pair)
    selected = [i for i, color in enumerate(selectors) if color == CANCEL]
    if len(selected) != 1 or any(color not in {CANCEL, INK} for color in selectors):
        raise OracleError("malformed cancel selector")
    cancel = selected[0]
    if cancel not in range(1, 4):
        raise OracleError("cancel card outside supported range")
    removed, depth = _removed(pairs, cancel)
    survivors_after = [i for i in range(cancel+1, 7) if i not in removed]
    if not (3 <= len(removed) <= 5 and max(depth.values()) >= 2 and survivors_after):
        raise OracleError("cancellation is not causally deep")
    factual = _run(initial, pairs, set())
    naive = _run(initial, pairs, {cancel})
    pruned = _run(initial, pairs, removed)
    ti = REGISTERS.index(target)
    if factual[ti] == pruned[ti] or naive[ti] == pruned[ti]:
        raise OracleError("shortcut has same target outcome")
    if not any(target in pairs[i] for i in survivors_after):
        raise OracleError("target lacks later surviving cause")
    answer = pruned[ti]
    if answer not in SHAPES:
        raise OracleError("unknown answer")
    return answer
