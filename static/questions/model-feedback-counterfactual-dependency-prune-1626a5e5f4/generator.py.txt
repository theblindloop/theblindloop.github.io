"""Renderer for a causally pruned swap-event log."""
from __future__ import annotations

import math
import random
from PIL import Image, ImageDraw, ImageFont

CANVAS = (960, 620)
BOARD_W, BOARD_H = 920, 580
REGISTERS = ("A", "B", "C", "D")
SHAPES = ("circle", "triangle", "star", "diamond")
PAIRS = tuple((a, b) for i, a in enumerate(REGISTERS) for b in REGISTERS[i + 1 :])
RGB = {
    "white": (255, 255, 255), "board": (246, 248, 244), "card": (255, 255, 252),
    "ink": (37, 43, 49), "muted": (105, 112, 118), "border": (55, 64, 72),
    "cancel": (204, 43, 55), "target": (226, 158, 24),
    "A": (198, 82, 52), "B": (22, 139, 150), "C": (121, 75, 180), "D": (65, 104, 174),
    "circle": (42, 124, 202), "triangle": (38, 157, 91),
    "star": (230, 158, 26), "diamond": (205, 69, 119),
}
FONT = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
BOLD = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"


def _dependencies(pairs: list[list[str]]) -> list[set[int]]:
    last: dict[str, int] = {}
    deps: list[set[int]] = []
    for index, pair in enumerate(pairs):
        parents = {last[slot] for slot in pair if slot in last}
        deps.append(parents)
        for slot in pair:
            last[slot] = index
    return deps


def _removed(pairs: list[list[str]], cancel: int) -> tuple[set[int], dict[int, int]]:
    deps = _dependencies(pairs)
    removed = {cancel}
    depth = {cancel: 0}
    for index in range(cancel + 1, len(pairs)):
        hit = deps[index] & removed
        if hit:
            removed.add(index)
            depth[index] = 1 + max(depth[parent] for parent in hit)
    return removed, depth


def _run(initial: list[str], pairs: list[list[str]], skipped: set[int]) -> list[str]:
    state = list(initial)
    for index, pair in enumerate(pairs):
        if index in skipped:
            continue
        a, b = (REGISTERS.index(slot) for slot in pair)
        state[a], state[b] = state[b], state[a]
    return state


def sample_scene(seed: int) -> dict:
    rng = random.Random((int(seed) ^ 0xCA05A11C) & 0xFFFFFFFFFFFFFFFF)
    wanted = SHAPES[int(seed) % 4]
    for _ in range(100000):
        initial = list(SHAPES)
        rng.shuffle(initial)
        pairs = [list(rng.choice(PAIRS)) for _ in range(7)]
        cancel = rng.randrange(1, 4)
        removed, depth = _removed(pairs, cancel)
        survivors_after = [i for i in range(cancel + 1, 7) if i not in removed]
        if not (3 <= len(removed) <= 5 and max(depth.values()) >= 2 and survivors_after):
            continue
        factual = _run(initial, pairs, set())
        naive = _run(initial, pairs, {cancel})
        pruned = _run(initial, pairs, removed)
        target = REGISTERS[pruned.index(wanted)]
        ti = REGISTERS.index(target)
        if factual[ti] == wanted or naive[ti] == wanted:
            continue
        if not any(target in pairs[i] for i in survivors_after):
            continue
        return {
            "initial": initial, "pairs": pairs, "cancel_card": cancel,
            "target_slot": target, "anchor_x": rng.randint(18, 22),
            "anchor_y": rng.randint(18, 22),
        }
    raise RuntimeError("could not sample a deep dependency-pruning scene")


def _validate(scene: dict) -> None:
    initial = [str(v) for v in scene["initial"]]
    pairs = [[str(v) for v in pair] for pair in scene["pairs"]]
    cancel = int(scene["cancel_card"])
    target = str(scene["target_slot"])
    if sorted(initial) != sorted(SHAPES) or len(pairs) != 7:
        raise ValueError("invalid start or card count")
    if cancel not in range(1, 4) or target not in REGISTERS:
        raise ValueError("invalid cancellation or target")
    if any(len(pair) != 2 or tuple(pair) not in PAIRS for pair in pairs):
        raise ValueError("invalid swap pair")
    removed, depth = _removed(pairs, cancel)
    survivors_after = [i for i in range(cancel + 1, 7) if i not in removed]
    if not (3 <= len(removed) <= 5 and max(depth.values()) >= 2 and survivors_after):
        raise ValueError("cancellation is not causally deep")
    factual = _run(initial, pairs, set())
    naive = _run(initial, pairs, {cancel})
    pruned = _run(initial, pairs, removed)
    ti = REGISTERS.index(target)
    if factual[ti] == pruned[ti] or naive[ti] == pruned[ti]:
        raise ValueError("shortcut has same target outcome")
    if not any(target in pairs[i] for i in survivors_after):
        raise ValueError("target lacks a later surviving cause")


def analytic_gold(scene: dict) -> str:
    _validate(scene)
    removed, _ = _removed(scene["pairs"], int(scene["cancel_card"]))
    final = _run(scene["initial"], scene["pairs"], removed)
    answer = final[REGISTERS.index(str(scene["target_slot"]))]
    if answer not in SHAPES:
        raise ValueError("invalid answer")
    return answer


def margin(scene: dict) -> float:
    try:
        return 1.0 if analytic_gold(scene) in SHAPES else 0.0
    except (KeyError, TypeError, ValueError):
        return 0.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int, bold: bool = False):
    return ImageFont.truetype(BOLD if bold else FONT, size)


def _text(draw, xy, value, size, *, fill=None, bold=False, anchor="mm"):
    draw.text(xy, value, font=_font(size, bold), fill=fill or RGB["ink"], anchor=anchor)


def _shape(draw, center, kind: str, scale: int = 20) -> None:
    x, y = center
    color = RGB[kind]
    if kind == "circle":
        draw.ellipse((x-scale, y-scale, x+scale, y+scale), fill=color, outline=RGB["ink"], width=2)
    elif kind == "triangle":
        pts = ((x, y-scale-2), (x-scale-2, y+scale), (x+scale+2, y+scale))
        draw.polygon(pts, fill=color); draw.line(pts + (pts[0],), fill=RGB["ink"], width=2)
    elif kind == "diamond":
        pts = ((x, y-scale-2), (x-scale-2, y), (x, y+scale+2), (x+scale+2, y))
        draw.polygon(pts, fill=color); draw.line(pts + (pts[0],), fill=RGB["ink"], width=2)
    elif kind == "star":
        pts = []
        for i in range(10):
            radius = scale+2 if i % 2 == 0 else scale*0.45
            angle = -math.pi/2 + i*math.pi/5
            pts.append((x+radius*math.cos(angle), y+radius*math.sin(angle)))
        draw.polygon(pts, fill=color); draw.line(pts + [pts[0]], fill=RGB["ink"], width=2)
    else:
        raise ValueError(kind)


def _badge(draw, center, label: str, large: bool = False) -> None:
    x, y = center
    w, h = ((52, 38) if large else (42, 31))
    draw.rounded_rectangle((x-w//2, y-h//2, x+w//2, y+h//2), radius=6,
                           fill=RGB[label], outline=RGB["ink"], width=2)
    _text(draw, center, label, 20 if large else 17, fill=RGB["white"], bold=True)


def render(scene: dict) -> Image.Image:
    _validate(scene)
    ax, ay = int(scene["anchor_x"]), int(scene["anchor_y"])
    image = Image.new("RGB", CANVAS, RGB["white"])
    draw = ImageDraw.Draw(image)
    draw.rectangle((ax, ay, ax+BOARD_W-1, ay+BOARD_H-1), fill=RGB["board"], outline=RGB["border"], width=4)
    _text(draw, (ax+460, ay+24), "CANCEL CAUSAL DESCENDANTS", 22, bold=True)
    _text(draw, (ax+75, ay+55), "START", 17, bold=True)
    for label, kind, x in zip(REGISTERS, scene["initial"], (185, 355, 525, 695), strict=True):
        draw.rounded_rectangle((ax+x-67, ay+45, ax+x+67, ay+135), radius=9,
                               fill=RGB["card"], outline=RGB[label], width=4)
        _badge(draw, (ax+x-42, ay+68), label)
        _shape(draw, (ax+x+12, ay+102), str(kind), 21)
    _text(draw, (ax+790, ay+58), "TARGET", 16, fill=RGB["target"], bold=True)
    _badge(draw, (ax+850, ay+96), str(scene["target_slot"]), large=True)
    draw.rounded_rectangle((ax+38, ay+150, ax+882, ay+278), radius=10,
                           fill=(255, 252, 238), outline=RGB["target"], width=3)
    _text(draw, (ax+460, ay+174), "Use the FULL printed row to find dependencies.", 17, bold=True)
    _text(draw, (ax+460, ay+204), "A swap depends on the latest earlier swap touching either box.", 16)
    _text(draw, (ax+460, ay+234), "Cancel red X; repeat cancellation through dependencies.", 16, fill=RGB["cancel"], bold=True)
    _text(draw, (ax+460, ay+260), "Then restart and do only swaps left  →  read TARGET", 16, bold=True)
    cancel = int(scene["cancel_card"])
    for index, pair in enumerate(scene["pairs"]):
        x, y = ax+20+index*128, ay+300
        selected = index == cancel
        outline = RGB["cancel"] if selected else RGB["border"]
        draw.rounded_rectangle((x, y, x+112, y+244), radius=9, fill=RGB["card"], outline=outline, width=5 if selected else 3)
        draw.ellipse((x+38, y+10, x+74, y+46), fill=RGB["cancel"] if selected else RGB["ink"])
        _text(draw, (x+56, y+28), str(index+1), 17, fill=RGB["white"], bold=True)
        _badge(draw, (x+56, y+88), str(pair[0]), large=True)
        _text(draw, (x+56, y+133), "↕", 27, bold=True)
        _badge(draw, (x+56, y+178), str(pair[1]), large=True)
        if selected:
            _text(draw, (x+56, y+221), "X", 30, fill=RGB["cancel"], bold=True)
        else:
            _text(draw, (x+56, y+221), "SWAP", 13, fill=RGB["muted"], bold=True)
    return image
