"""Scene sampler and renderer for the descending-switchback counting world.

A single opaque black trace descends a survey panel from its top end to its
bottom end.  The trace never doubles back vertically: for every image row it
occupies, it has exactly one horizontal position.  Its horizontal direction of
travel, however, reverses a sampled number of times.  The analytic gold is that
reversal count, which is a property of the whole route rather than of any one
place on it.

All sampled fields are drawn.  The scene dictionary stores drawing primitives
only (node coordinates, widths, colours, decoration boxes), so no field can
carry the label without also changing pixels.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "descent_switchback_count-renderer-0.1.0"

CANVAS_W = 640
CANVAS_H = 480
BACKGROUND = (250, 250, 247)
ROUTE_COLOR = (0, 0, 0)
FRAME_COLOR = (176, 176, 172)
MARKER_COLOR = (104, 104, 100)
DISTRACTOR_COLORS = (
    (214, 92, 60),
    (72, 128, 206),
    (206, 158, 48),
    (96, 168, 108),
    (168, 104, 190),
    (86, 172, 178),
)

MIN_TURNS = 2
MAX_TURNS = 5
MIN_PROMINENCE = 20.0
QUARANTINE_PROMINENCE = 30.0
QUARANTINE_STEP = 22.0


# --------------------------------------------------------------------------
# route geometry
# --------------------------------------------------------------------------
def _zigzag_units(rng: random.Random, turns: int) -> list[float] | None:
    """Sample `turns` + 2 alternating node values normalised to span [0, 1].

    One long traverse plus shorter alternating backtracks keeps the total
    horizontal travel inside a band shared by every reversal count, so neither
    the route's bounding box nor its length betrays the answer.
    """
    segments = turns + 1
    magnitudes = [rng.uniform(0.30, 0.85) for _ in range(segments)]
    magnitudes[rng.randrange(segments)] = rng.uniform(1.1, 2.2)
    sign = 1.0 if rng.random() < 0.5 else -1.0
    values = [0.0]
    for magnitude in magnitudes:
        values.append(values[-1] + sign * magnitude)
        sign = -sign
    lo, hi = min(values), max(values)
    span = hi - lo
    if span < 1e-6:
        return None
    travel = sum(magnitudes) / span
    if not 1.6 <= travel <= 2.8:
        return None
    return [(value - lo) / span for value in values]


def _tighten(values: list[float], rng: random.Random, target: float) -> list[float]:
    """Pull one interior node inwards so its excursion equals `target` (unit)."""
    interior = list(range(1, len(values) - 1))
    if not interior:
        return values
    index = rng.choice(interior)
    left, right = values[index - 1], values[index + 1]
    base = right if abs(values[index] - right) < abs(values[index] - left) else left
    sign = 1.0 if values[index] > base else -1.0
    current = min(abs(values[index] - left), abs(values[index] - right))
    if current <= target:
        return values
    tightened = list(values)
    tightened[index] = base + sign * target
    return tightened


def sample_scene(seed: int) -> dict:
    """Sample one renderable scene deterministically from `seed`."""
    rng = random.Random(seed * 7919 + 13)
    turns = rng.randint(MIN_TURNS, MAX_TURNS)
    for _ in range(4000):
        units = _zigzag_units(rng, turns)
        if units is None:
            continue

        span = rng.uniform(150.0, 205.0)
        roll = rng.random()
        if roll < 0.10:
            target_prominence = rng.uniform(20.0, 29.0)
        elif roll < 0.35:
            target_prominence = rng.uniform(32.0, 56.0)
        else:
            target_prominence = rng.uniform(60.0, 400.0)
        units = _tighten(units, rng, target_prominence / span)

        lo, hi = min(units), max(units)
        if hi - lo < 0.6:
            continue
        units = [(value - lo) / (hi - lo) for value in units]
        deltas = [b - a for a, b in zip(units, units[1:])]
        if any(abs(delta) < 1e-6 for delta in deltas):
            continue
        if any(a * b >= 0 for a, b in zip(deltas, deltas[1:])):
            continue  # direction must alternate at every interior node

        xs_unit = units
        dx_pixels = [abs(delta) * span for delta in deltas]
        weights = [max(value / 2.4, 30.0) * rng.uniform(1.0, 1.3) for value in dx_pixels]
        extent = rng.uniform(240.0, 320.0)
        scale = extent / sum(weights)
        dy_pixels = [weight * scale for weight in weights]
        if min(dy_pixels) < QUARANTINE_STEP:
            continue
        if max(dx / dy for dx, dy in zip(dx_pixels, dy_pixels)) > 3.0:
            continue

        left = rng.uniform(96.0, CANVAS_W - span - 96.0)
        top = rng.uniform(74.0, CANVAS_H - extent - 74.0)
        xs = [round(left + value * span, 3) for value in xs_unit]
        ys = [round(top, 3)]
        for step in dy_pixels:
            ys.append(round(ys[-1] + step, 3))

        prominences = [
            min(abs(xs[i] - xs[i - 1]), abs(xs[i] - xs[i + 1]))
            for i in range(1, len(xs) - 1)
        ]
        if min(prominences) < MIN_PROMINENCE:
            continue  # the prompt promises at least this much swing at every turn

        nodes = [[x, y] for x, y in zip(xs, ys)]
        frame = [
            round(min(xs) - rng.uniform(38.0, 64.0), 3),
            round(top - rng.uniform(30.0, 52.0), 3),
            round(max(xs) + rng.uniform(38.0, 64.0), 3),
            round(ys[-1] + rng.uniform(30.0, 52.0), 3),
        ]
        if frame[0] < 8 or frame[1] < 8 or frame[2] > CANVAS_W - 8 or frame[3] > CANVAS_H - 8:
            continue

        ticks = []
        tick_count = rng.randint(4, 7)
        for index in range(tick_count):
            frac = (index + 1) / (tick_count + 1)
            ticks.append(round(frame[1] + frac * (frame[3] - frame[1]), 3))

        distractors = []
        palette = list(DISTRACTOR_COLORS)
        rng.shuffle(palette)
        for index in range(rng.randint(2, 4)):
            count = rng.randint(3, 6)
            dxs = [round(rng.uniform(frame[0] + 10, frame[2] - 10), 3) for _ in range(count)]
            dy0 = rng.uniform(frame[1] + 8, frame[1] + 60)
            dy1 = rng.uniform(frame[3] - 60, frame[3] - 8)
            dys = [
                round(dy0 + (dy1 - dy0) * step / (count - 1), 3) for step in range(count)
            ]
            distractors.append(
                {
                    "color": list(palette[index % len(palette)]),
                    "width": rng.choice([2, 3]),
                    "nodes": [[x, y] for x, y in zip(dxs, dys)],
                }
            )

        return {
            "width": CANVAS_W,
            "height": CANVAS_H,
            "route_nodes": nodes,
            "route_width": rng.choice([3, 4]),
            "marker_size": rng.choice([7, 9]),
            "frame": [round(value, 3) for value in frame],
            "tick_rows": ticks,
            "tick_length": rng.choice([8, 11]),
            "distractors": distractors,
        }
    raise RuntimeError(f"scene sampling failed for seed {seed}")


def _route_polyline(scene: dict) -> list[tuple[float, float]]:
    """Densely sample the cosine-eased route through its nodes."""
    nodes = [(float(x), float(y)) for x, y in scene["route_nodes"]]
    points: list[tuple[float, float]] = []
    for (x0, y0), (x1, y1) in zip(nodes, nodes[1:]):
        steps = max(8, int((y1 - y0) * 3))
        for step in range(steps + 1):
            t = step / steps
            ease = (1.0 - math.cos(math.pi * t)) / 2.0
            points.append((x0 + (x1 - x0) * ease, y0 + (y1 - y0) * t))
    return points


def _smooth_polyline(nodes: list[list[float]], samples: int = 96) -> list[tuple[float, float]]:
    """Cosine-eased polyline used for decorative distractor curves."""
    pts = [(float(x), float(y)) for x, y in nodes]
    out: list[tuple[float, float]] = []
    for (x0, y0), (x1, y1) in zip(pts, pts[1:]):
        for step in range(samples + 1):
            t = step / samples
            ease = (1.0 - math.cos(math.pi * t)) / 2.0
            out.append((x0 + (x1 - x0) * ease, y0 + (y1 - y0) * t))
    return out


def render(scene: dict) -> Image.Image:
    """Draw the scene; no anti-aliasing, so the black route stays pure black."""
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BACKGROUND)
    draw = ImageDraw.Draw(image)

    x0, y0, x1, y1 = scene["frame"]
    draw.rectangle([x0, y0, x1, y1], outline=FRAME_COLOR, width=2)
    for row in scene["tick_rows"]:
        draw.line([x0, row, x0 + scene["tick_length"], row], fill=FRAME_COLOR, width=2)
        draw.line([x1 - scene["tick_length"], row, x1, row], fill=FRAME_COLOR, width=2)

    for curve in scene["distractors"]:
        draw.line(
            _smooth_polyline(curve["nodes"]),
            fill=tuple(curve["color"]),
            width=int(curve["width"]),
            joint="curve",
        )

    head = scene["route_nodes"][0]
    tail = scene["route_nodes"][-1]
    half = scene["marker_size"] / 2.0
    draw.rectangle(
        [head[0] - half, head[1] - half, head[0] + half, head[1] + half],
        outline=MARKER_COLOR,
        width=2,
    )
    draw.rectangle(
        [tail[0] - half, tail[1] - half, tail[0] + half, tail[1] + half],
        outline=MARKER_COLOR,
        width=2,
    )

    draw.line(
        _route_polyline(scene),
        fill=ROUTE_COLOR,
        width=int(scene["route_width"]),
        joint="curve",
    )
    return image


# --------------------------------------------------------------------------
# analytic arm
# --------------------------------------------------------------------------
def _interior_prominences(scene: dict) -> list[float]:
    xs = [float(x) for x, _ in scene["route_nodes"]]
    return [
        min(abs(xs[i] - xs[i - 1]), abs(xs[i] - xs[i + 1]))
        for i in range(1, len(xs) - 1)
    ]


def analytic_gold(scene: dict) -> str:
    """Number of interior nodes where the horizontal direction reverses."""
    xs = [float(x) for x, _ in scene["route_nodes"]]
    deltas = [b - a for a, b in zip(xs, xs[1:])]
    reversals = sum(1 for a, b in zip(deltas, deltas[1:]) if a * b < 0)
    return str(reversals)


def margin(scene: dict) -> float:
    """Smallest horizontal excursion (pixels) of any counted reversal."""
    prominences = _interior_prominences(scene)
    return float(round(min(prominences), 6)) if prominences else 0.0


def min_vertical_step(scene: dict) -> float:
    ys = [float(y) for _, y in scene["route_nodes"]]
    return float(round(min(b - a for a, b in zip(ys, ys[1:])), 6))


def is_quarantined(scene: dict) -> bool:
    return bool(
        margin(scene) < QUARANTINE_PROMINENCE
        or min_vertical_step(scene) < QUARANTINE_STEP
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist: every sampled field is drawn."""
    return []
