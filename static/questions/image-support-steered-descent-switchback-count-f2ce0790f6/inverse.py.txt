"""Pixel-only inverse arm for the descending-switchback counting world.

It receives a rendered PNG and nothing else: no scene, no renderer code, no
analytic gold.  The procedure is

1. segment the opaque black route by colour;
2. recover its centre-line as one horizontal position per occupied image row,
   keeping only the single connected block of rows the route spans;
3. walk that centre-line from top to bottom with a hysteresis of
   `SWING_PX` pixels and count every reversal of horizontal travel.

The count is the returned decision.  Degenerate rasters return "unresolved".
"""

from __future__ import annotations

from PIL import Image

ORACLE_VERSION = "descent_switchback_count-oracle-0.1.0"

INK_MAX = 60          # a pixel is route ink when every channel is below this
SWING_PX = 12.0       # horizontal travel needed before a reversal is credited
MIN_ROWS = 60         # a shorter ink block is not a usable route
SMOOTH = 5            # centre-line moving-average window (odd)


def _centre_line(image: Image.Image) -> list[float] | None:
    rgb = image.convert("RGB")
    width, height = rgb.size
    pixels = rgb.load()

    per_row: list[float | None] = []
    for y in range(height):
        runs: list[tuple[int, int]] = []
        start = None
        for x in range(width):
            r, g, b = pixels[x, y]
            if r < INK_MAX and g < INK_MAX and b < INK_MAX:
                if start is None:
                    start = x
            elif start is not None:
                runs.append((start, x - 1))
                start = None
        if start is not None:
            runs.append((start, width - 1))
        if not runs:
            per_row.append(None)
        else:
            best = max(runs, key=lambda run: run[1] - run[0])
            per_row.append((best[0] + best[1]) / 2.0)

    best_block: list[float] = []
    current: list[float] = []
    for value in per_row:
        if value is None:
            if len(current) > len(best_block):
                best_block = current
            current = []
        else:
            current.append(value)
    if len(current) > len(best_block):
        best_block = current
    if len(best_block) < MIN_ROWS:
        return None

    half = SMOOTH // 2
    smoothed: list[float] = []
    for index in range(len(best_block)):
        lo = max(0, index - half)
        hi = min(len(best_block), index + half + 1)
        window = best_block[lo:hi]
        smoothed.append(sum(window) / len(window))
    return smoothed


def _count_reversals(trace: list[float]) -> int:
    """Hysteresis walk: a reversal needs `SWING_PX` of travel the other way."""
    direction = 0
    low = high = trace[0]
    anchor = trace[0]
    reversals = 0
    for value in trace[1:]:
        if direction == 0:
            low = min(low, value)
            high = max(high, value)
            if value - low > SWING_PX:
                direction = 1
                anchor = value
            elif high - value > SWING_PX:
                direction = -1
                anchor = value
        elif direction * (value - anchor) > 0:
            anchor = value
        elif direction * (anchor - value) > SWING_PX:
            reversals += 1
            direction = -direction
            anchor = value
    return reversals


def decision_from_image(image: Image.Image) -> str:
    """Return the number of horizontal reversals of the black route."""
    trace = _centre_line(image)
    if trace is None:
        return "unresolved"
    return str(_count_reversals(trace))
