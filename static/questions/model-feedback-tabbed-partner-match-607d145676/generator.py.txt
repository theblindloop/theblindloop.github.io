"""Deterministic renderer and analytic arm for the tabbed-partner-match world.

Two vertical columns of three square slots, left and right.  Every slot carries a
small coloured tab (cyan, magenta, or amber); each column uses all three colours
once, but the right column's top-to-bottom tab order is a different permutation
from the left, so the two decks are NOT aligned by position.  A left slot and a
right slot are PARTNERS when they share the same tab colour.  Every slot also
holds a marker that is either a SOLID dark square or a HOLLOW (outline-only)
square.  A pair is a TRUE match when the two partner slots carry markers of the
same kind.

The essential decision program is a correspondence binding: bind each left slot
to the right slot of the same tab colour (ignoring their differing vertical
positions), read the bound marker kinds, and decide how many bound pairs agree.
The prominent position-aligned reading (left-top with right-top) is a decoy that
binds the wrong partners whenever the two tab orders differ.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 780
HEIGHT = 900
COLORS = ["cyan", "magenta", "amber"]
RGB = {
    "cyan": (0, 150, 170),
    "magenta": (200, 40, 120),
    "amber": (232, 158, 32),
}
KINDS = ["solid", "hollow"]
DARK = (45, 45, 45)
BACKGROUND = (250, 250, 250)
SLOT_LINE = (200, 200, 200)
SLOT_SIZE = 170
TAB_SIZE = 48
MARKER_SIZE = 72
MARKER_BORDER = 12
SLOT_YS = [230, 460, 690]
LEFT_CX = 215
RIGHT_CX = 565
LEFT_TAB_X = 127
RIGHT_TAB_X = 655
MATCH_WORDS = ["zero", "one", "two", "three"]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    tabs_left = list(COLORS)
    rng.shuffle(tabs_left)
    tabs_right = list(COLORS)
    rng.shuffle(tabs_right)
    for _ in range(20):
        if tabs_right != tabs_left:
            break
        rng.shuffle(tabs_right)
    target = rng.randint(0, 3)
    partner = {color: j for j, color in enumerate(tabs_right)}
    pairs_match = [True, True, True]
    choose_false = rng.sample([0, 1, 2], 3 - target)
    for i in choose_false:
        pairs_match[i] = False
    markers_left = [None, None, None]
    markers_right = [None, None, None]
    for i, color in enumerate(tabs_left):
        j = partner[color]
        kind = rng.choice(KINDS)
        markers_left[i] = kind
        markers_right[j] = kind if pairs_match[i] else (
            "hollow" if kind == "solid" else "solid"
        )
    return {
        "canvas_size": [WIDTH, HEIGHT],
        "tabs_left": tabs_left,
        "tabs_right": tabs_right,
        "markers_left": markers_left,
        "markers_right": markers_right,
        "slot_size": SLOT_SIZE,
        "tab_size": TAB_SIZE,
        "marker_size": MARKER_SIZE,
    }


def _match_count(scene: dict) -> int:
    partner = {color: j for j, color in enumerate(scene["tabs_right"])}
    count = 0
    for i, color in enumerate(scene["tabs_left"]):
        j = partner[color]
        if scene["markers_left"][i] == scene["markers_right"][j]:
            count += 1
    return count


def _draw_slot_with_marker(
    draw: ImageDraw.ImageDraw,
    cx: int,
    cy: int,
    tab_color: str,
    tab_on_right: bool,
    kind: str,
) -> None:
    x0, y0, x1, y1 = cx - SLOT_SIZE // 2, cy - SLOT_SIZE // 2, cx + SLOT_SIZE // 2, cy + SLOT_SIZE // 2
    draw.rectangle((x0, y0, x1, y1), outline=SLOT_LINE, width=4)
    if tab_on_right:
        tx0, ty0, tx1, ty1 = x1 + 8, cy - TAB_SIZE // 2, x1 + 8 + TAB_SIZE, cy + TAB_SIZE // 2
    else:
        tx0, ty0, tx1, ty1 = x0 - 8 - TAB_SIZE, cy - TAB_SIZE // 2, x0 - 8, cy + TAB_SIZE // 2
    draw.rectangle((tx0, ty0, tx1, ty1), fill=RGB[tab_color], outline=SLOT_LINE)
    mx0, my0 = cx - MARKER_SIZE // 2, cy - MARKER_SIZE // 2
    mx1, my1 = cx + MARKER_SIZE // 2, cy + MARKER_SIZE // 2
    draw.rectangle((mx0, my0, mx1, my1), fill=BACKGROUND, outline=DARK, width=4)
    if kind == "solid":
        draw.rectangle(
            (mx0 + 4, my0 + 4, mx1 - 4, my1 - 4), fill=DARK, outline=DARK
        )
    else:
        draw.rectangle(
            (mx0 + MARKER_BORDER, my0 + MARKER_BORDER, mx1 - MARKER_BORDER, my1 - MARKER_BORDER),
            fill=DARK,
            outline=DARK,
        )
        draw.rectangle(
            (mx0 + MARKER_BORDER + 4, my0 + MARKER_BORDER + 4, mx1 - MARKER_BORDER - 4, my1 - MARKER_BORDER - 4),
            fill=BACKGROUND,
            outline=BACKGROUND,
        )


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for i, y in enumerate(SLOT_YS):
        _draw_slot_with_marker(
            draw, LEFT_CX, y, scene["tabs_left"][i], False, scene["markers_left"][i]
        )
        _draw_slot_with_marker(
            draw, RIGHT_CX, y, scene["tabs_right"][i], True, scene["markers_right"][i]
        )
    return image


def analytic_gold(scene: dict) -> str:
    return MATCH_WORDS[_match_count(scene)]


def margin(scene: dict) -> float:
    return 1.0


def is_quarantined(scene: dict) -> bool:
    try:
        if sorted(scene["tabs_left"]) != sorted(COLORS):
            return True
        if sorted(scene["tabs_right"]) != sorted(COLORS):
            return True
        if scene["tabs_left"] == scene["tabs_right"]:
            return True
        if len(scene["markers_left"]) != 3 or len(scene["markers_right"]) != 3:
            return True
        if any(k not in KINDS for k in scene["markers_left"] + scene["markers_right"]):
            return True
        return False
    except (KeyError, TypeError):
        return True


def latent_symmetries(scene: dict) -> list:
    return []
