"""Independent PNG-only inverse arm for the tabbed-partner-match world.

Recovers, from the raster alone, the coloured tab on every slot in the left and
right columns, establishes the partner correspondence by shared tab colour,
reads each slot's marker (solid vs hollow square) at the slot centre, and counts
how many tab-colour partners hold the same marker kind.  Uses no renderer
geometry, scene data, prompt, or gold code; only chromatic masks, connected
components, and centre-pixel brightness.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RGB = {
    "cyan": (0, 150, 170),
    "magenta": (200, 40, 120),
    "amber": (232, 158, 32),
}
WORDS = ["zero", "one", "two", "three"]
MID_X = 390
LEFT_MARKER_X = 215
RIGHT_MARKER_X = 565
TAB_MIN = 100
TOL = 45


def _color_mask(arr: np.ndarray, color: tuple, region: tuple[int, int, int, int]) -> np.ndarray:
    x0, x1, y0, y1 = region
    r, g, b = color
    mask = np.zeros(arr.shape[:2], dtype=bool)
    view = arr[y0:y1, x0:x1]
    sub = (
        (np.abs(view[:, :, 0] - r) <= TOL)
        & (np.abs(view[:, :, 1] - g) <= TOL)
        & (np.abs(view[:, :, 2] - b) <= TOL)
    )
    mask[y0:y1, x0:x1] = sub
    return mask


def _tabs(arr: np.ndarray, x0: int, x1: int) -> dict:
    region = (x0, x1, 0, arr.shape[0])
    found = {}
    for color in RGB:
        mask = _color_mask(arr, RGB[color], region)
        rows, cols = np.nonzero(mask)
        if rows.size == 0:
            raise ValueError(f"missing {color} tab")
        found[color] = (float(cols.mean()), float(rows.mean()))
    return found


def _marker_kind(arr: np.ndarray, cx: float, cy: float) -> str:
    y = int(round(cy))
    x = int(round(cx))
    mean = int(arr[y, x].mean())
    return "solid" if mean < 160 else "hollow"


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    left = _tabs(arr, 0, MID_X)
    right = _tabs(arr, MID_X, arr.shape[1])
    count = 0
    for color in RGB:
        lx, ly = left[color]
        rx, ry = right[color]
        lkind = _marker_kind(arr, LEFT_MARKER_X, ly)
        rkind = _marker_kind(arr, RIGHT_MARKER_X, ry)
        if lkind == rkind:
            count += 1
    if count > 3:
        raise ValueError(f"invalid match count {count}")
    return WORDS[count]
