"""Independent pixel-only inverse for the hinged-arm world."""

from __future__ import annotations

from collections import deque
import math

import numpy as np


ORACLE_VERSION = "counterfactual_hinge_proximity-oracle-0.1.0"
ABSTAIN = "abstain"

_ARM = np.array((22, 112, 120), dtype=np.int16)
_PIVOT = np.array((239, 178, 35), dtype=np.int16)
_RED = np.array((211, 67, 70), dtype=np.int16)
_BLUE = np.array((53, 108, 204), dtype=np.int16)


def _near(array: np.ndarray, colour: np.ndarray, tolerance: int = 8) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - colour), axis=2) <= tolerance


def _components(mask: np.ndarray, minimum: int = 100) -> list[tuple[int, float, float, int, int, int, int]]:
    """Return sizeable 8-connected blobs as size, centroid, and bounds."""

    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[tuple[int, float, float, int, int, int, int]] = []
    ys, xs = np.nonzero(mask)
    for start_y, start_x in zip(ys.tolist(), xs.tolist()):
        if seen[start_y, start_x]:
            continue
        queue = deque([(start_y, start_x)])
        seen[start_y, start_x] = True
        size = 0
        sum_x = 0
        sum_y = 0
        y0 = y1 = start_y
        x0 = x1 = start_x
        while queue:
            y, x = queue.popleft()
            size += 1
            sum_x += x
            sum_y += y
            y0 = min(y0, y)
            y1 = max(y1, y)
            x0 = min(x0, x)
            x1 = max(x1, x)
            for ny, nx in (
                (y - 1, x),
                (y + 1, x),
                (y, x - 1),
                (y, x + 1),
                (y - 1, x - 1),
                (y - 1, x + 1),
                (y + 1, x - 1),
                (y + 1, x + 1),
            ):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if size >= minimum:
            output.append((size, sum_x / size, sum_y / size, y0, x0, y1, x1))
    return output


def _largest_center(array: np.ndarray, colour: np.ndarray) -> tuple[float, float] | None:
    parts = _components(_near(array, colour), minimum=450)
    if not parts:
        return None
    part = max(parts, key=lambda item: item[0])
    return part[1], part[2]


def _arm_pose(array: np.ndarray, pivot: tuple[float, float]) -> tuple[float, float, float] | None:
    parts = _components(_near(array, _ARM), minimum=500)
    if not parts:
        return None
    part = max(parts, key=lambda item: item[0])
    mask = _near(array, _ARM)
    y0, x0, y1, x1 = part[3], part[4], part[5], part[6]
    crop = mask[y0:y1 + 1, x0:x1 + 1]
    ys, xs = np.nonzero(crop)
    xs = xs.astype(np.float64) + x0 - float(pivot[0])
    ys = ys.astype(np.float64) + y0 - float(pivot[1])
    if len(xs) < 50:
        return None
    points = np.column_stack((xs, ys))
    covariance = np.cov(points, rowvar=False)
    eigenvalues, eigenvectors = np.linalg.eigh(covariance)
    vector = eigenvectors[:, int(np.argmax(eigenvalues))]
    norm = float(np.hypot(vector[0], vector[1]))
    if norm <= 1e-6:
        return None
    vector = vector / norm
    projections = points @ vector
    if float(np.max(projections)) < abs(float(np.min(projections))):
        vector = -vector
        projections = -projections
    # The pivot hides the inner arm base.  The outer endpoint is the largest
    # projection and is stable even when the triangle tip is rasterized.
    length = float(np.percentile(projections, 99.5))
    if length < 60.0:
        return None
    return float(vector[0]), float(vector[1]), length


def decision_from_image(image) -> str:
    """Recover targets and arm geometry, then apply the clockwise turn in pixels."""

    array = np.asarray(image.convert("RGB"))
    if array.ndim != 3 or array.shape[0] < 200 or array.shape[1] < 300:
        return ABSTAIN
    pivot = _largest_center(array, _PIVOT)
    red = _largest_center(array, _RED)
    blue = _largest_center(array, _BLUE)
    if pivot is None or red is None or blue is None:
        return ABSTAIN
    pose = _arm_pose(array, pivot)
    if pose is None:
        return ABSTAIN
    ux, uy, length = pose
    # In image coordinates (x right, y down), this is a clockwise quarter-turn.
    post_ux, post_uy = -uy, ux
    tip = (pivot[0] + length * post_ux, pivot[1] + length * post_uy)
    red_distance = math.hypot(tip[0] - red[0], tip[1] - red[1])
    blue_distance = math.hypot(tip[0] - blue[0], tip[1] - blue[1])
    difference = red_distance - blue_distance
    if abs(difference) <= 18.0:
        return "tie"
    return "red" if difference < 0.0 else "blue"
