"""Renderer for a visible quarter-turn and post-turn target comparison."""

from __future__ import annotations

import math
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "counterfactual_hinge_proximity-0.1.0"

WIDTH = 900
HEIGHT = 640
BACKGROUND = (255, 255, 255)
BOARD = (246, 249, 253)
BOARD_EDGE = (99, 119, 143)
INK = (35, 48, 65)
MUTED = (132, 147, 165)
ARM = (22, 112, 120)
PIVOT = (239, 178, 35)
ACTION = (217, 128, 28)
RED = (211, 67, 70)
BLUE = (53, 108, 204)
TARGET_EDGE = (40, 52, 70)
WHITE = (255, 255, 255)

_FONT_PATH = Path("/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf")
_BOLD_FONT_PATH = Path("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf")


def _font(size: int, bold: bool = False):
    path = _BOLD_FONT_PATH if bold else _FONT_PATH
    try:
        return ImageFont.truetype(str(path), size=size)
    except OSError:
        return ImageFont.load_default()


def _point_on_orbit(pivot_x: int, pivot_y: int, radius: int, angle_deg: float) -> tuple[int, int]:
    angle = math.radians(float(angle_deg))
    return (
        int(round(float(pivot_x) + radius * math.cos(angle))),
        int(round(float(pivot_y) + radius * math.sin(angle))),
    )


def sample_scene(seed: int) -> dict:
    """Return a deterministic scene with a clear, non-tied post-turn answer."""

    rng = random.Random(
        (int(seed) * 6364136223846793005 + 1442695040888963407) & ((1 << 64) - 1)
    )
    pivot_x = 450 + rng.choice((-18, 0, 18))
    pivot_y = 350 + rng.choice((-16, 0, 16))
    arm_length = rng.choice((170, 180, 190))
    initial_angle_deg = rng.choice(tuple(range(0, 360, 45)))
    near_side = rng.choice((-1, 1))
    winner = rng.choice(("red", "blue"))
    target_orbit_radius = rng.choice((224, 236, 246))

    # In screen coordinates increasing angle is clockwise.  The near target is
    # placed away from the initial tip and the far target is near that initial
    # direction.  Thus a pre-action nearest-target shortcut is often wrong.
    post_angle = float(initial_angle_deg) + 90.0
    near_angle = post_angle + float(near_side * 58)
    far_angle = post_angle - 90.0
    near = _point_on_orbit(pivot_x, pivot_y, target_orbit_radius, near_angle)
    far = _point_on_orbit(pivot_x, pivot_y, target_orbit_radius, far_angle)
    red_target = near if winner == "red" else far
    blue_target = near if winner == "blue" else far

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "pivot_x": int(pivot_x),
        "pivot_y": int(pivot_y),
        "arm_length": int(arm_length),
        "initial_angle_deg": int(initial_angle_deg),
        "red_target_x": int(red_target[0]),
        "red_target_y": int(red_target[1]),
        "blue_target_x": int(blue_target[0]),
        "blue_target_y": int(blue_target[1]),
        "target_radius": 30,
        "layout_variant": int(rng.randrange(3)),
    }


def _post_tip(scene: dict) -> tuple[float, float]:
    angle = math.radians(float(scene["initial_angle_deg"]) + 90.0)
    return (
        float(scene["pivot_x"]) + float(scene["arm_length"]) * math.cos(angle),
        float(scene["pivot_y"]) + float(scene["arm_length"]) * math.sin(angle),
    )


def _initial_tip(scene: dict) -> tuple[float, float]:
    angle = math.radians(float(scene["initial_angle_deg"]))
    return (
        float(scene["pivot_x"]) + float(scene["arm_length"]) * math.cos(angle),
        float(scene["pivot_y"]) + float(scene["arm_length"]) * math.sin(angle),
    )


def _target(scene: dict, colour: str) -> tuple[float, float]:
    if colour == "red":
        return float(scene["red_target_x"]), float(scene["red_target_y"])
    if colour == "blue":
        return float(scene["blue_target_x"]), float(scene["blue_target_y"])
    raise ValueError(colour)


def _distance(a: tuple[float, float], b: tuple[float, float]) -> float:
    return math.hypot(a[0] - b[0], a[1] - b[1])


def _draw_centered_text(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, *, fill, font, anchor="mm") -> None:
    draw.text(xy, text, fill=fill, font=font, anchor=anchor)


def _draw_target(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    radius: int,
    colour: tuple[int, int, int],
    label: str,
) -> None:
    cx, cy = center
    draw.ellipse(
        (cx - radius - 3, cy - radius - 3, cx + radius + 3, cy + radius + 3),
        fill=TARGET_EDGE,
    )
    draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=colour)
    draw.ellipse((cx - 7, cy - 7, cx + 7, cy + 7), fill=WHITE)
    draw.ellipse((cx - 3, cy - 3, cx + 3, cy + 3), fill=TARGET_EDGE)
    _draw_centered_text(
        draw,
        (cx, cy + radius + 25),
        label,
        fill=colour,
        font=_font(18, bold=True),
    )


def _draw_action_arc(draw: ImageDraw.ImageDraw, pivot: tuple[int, int]) -> None:
    px, py = pivot
    radius = 64
    box = (px - radius, py - radius, px + radius, py + radius)
    draw.arc(box, start=205, end=323, fill=ACTION, width=7)
    end_angle = math.radians(323)
    ex = px + radius * math.cos(end_angle)
    ey = py + radius * math.sin(end_angle)
    tangent = (-math.sin(end_angle), math.cos(end_angle))
    tip = (ex, ey)
    base = (ex - 17 * tangent[0], ey - 17 * tangent[1])
    perp = (-tangent[1], tangent[0])
    draw.polygon(
        [
            tip,
            (base[0] + 10 * perp[0], base[1] + 10 * perp[1]),
            (base[0] - 10 * perp[0], base[1] - 10 * perp[1]),
        ],
        fill=ACTION,
    )


def render(scene: dict) -> Image.Image:
    """Render the initial hinged arm and a visible clockwise action cue."""

    width = int(scene.get("width", WIDTH))
    height = int(scene.get("height", HEIGHT))
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)

    draw.rounded_rectangle((38, 78, width - 38, height - 38), radius=24, fill=BOARD, outline=BOARD_EDGE, width=3)
    draw.rounded_rectangle((38, 18, width - 38, 66), radius=16, fill=WHITE, outline=BOARD_EDGE, width=2)
    _draw_centered_text(draw, (70, 42), "COUNTERFACTUAL TURN", fill=INK, font=_font(21, bold=True), anchor="lm")
    action_box = (width - 226, 27, width - 58, 57)
    draw.rounded_rectangle(action_box, radius=10, fill=ACTION)
    _draw_centered_text(draw, ((action_box[0] + action_box[2]) / 2, 42), "TURN 90 CW", fill=WHITE, font=_font(17, bold=True))

    px = int(scene["pivot_x"])
    py = int(scene["pivot_y"])
    length = float(scene["arm_length"])
    initial_tip = _initial_tip(scene)
    ux = (initial_tip[0] - px) / length
    uy = (initial_tip[1] - py) / length

    variant = int(scene.get("layout_variant", 0))
    if variant == 0:
        draw.line((84, 112, 84, height - 72), fill=(222, 230, 239), width=2)
        draw.line((width - 84, 112, width - 84, height - 72), fill=(222, 230, 239), width=2)
    elif variant == 1:
        draw.ellipse((88, 105, 155, 172), outline=(222, 230, 239), width=3)
        draw.ellipse((width - 155, 105, width - 88, 172), outline=(222, 230, 239), width=3)
    else:
        draw.line((88, 145, 150, 106), fill=(222, 230, 239), width=3)
        draw.line((width - 150, 106, width - 88, 145), fill=(222, 230, 239), width=3)
    _draw_centered_text(draw, (width / 2, 103), "HINGE BOARD", fill=MUTED, font=_font(16, bold=True))

    # A faint orbit gives the arm a spatial frame without showing the answer.
    draw.ellipse((px - 204, py - 204, px + 204, py + 204), outline=(224, 231, 239), width=2)

    # Draw the arm before the pivot so the gold hinge cleanly occludes its base.
    tip = initial_tip
    base = (tip[0] - 22 * ux, tip[1] - 22 * uy)
    perp = (-uy, ux)
    draw.line((px, py, tip[0], tip[1]), fill=ARM, width=20)
    draw.polygon(
        [
            (tip[0], tip[1]),
            (base[0] + 17 * perp[0], base[1] + 17 * perp[1]),
            (base[0] - 17 * perp[0], base[1] - 17 * perp[1]),
        ],
        fill=ARM,
    )
    arm_label_x = px - 68
    arm_label_y = py - 42
    draw.rounded_rectangle(
        (arm_label_x - 48, arm_label_y - 13, arm_label_x + 48, arm_label_y + 13),
        radius=7,
        fill=BOARD,
    )
    _draw_centered_text(draw, (arm_label_x, arm_label_y), "TEAL ARM", fill=ARM, font=_font(15, bold=True))

    _draw_target(
        draw,
        (int(scene["red_target_x"]), int(scene["red_target_y"])),
        int(scene["target_radius"]),
        RED,
        "RED TARGET",
    )
    _draw_target(
        draw,
        (int(scene["blue_target_x"]), int(scene["blue_target_y"])),
        int(scene["target_radius"]),
        BLUE,
        "BLUE TARGET",
    )

    _draw_action_arc(draw, (px, py))
    draw.ellipse((px - 25, py - 25, px + 25, py + 25), fill=PIVOT)
    draw.ellipse((px - 9, py - 9, px + 9, py + 9), fill=INK)
    draw.ellipse((px - 17, py - 17, px + 17, py + 17), outline=WHITE, width=3)
    _draw_centered_text(draw, (px, py + 43), "GOLD PIVOT", fill=INK, font=_font(15, bold=True))

    _draw_centered_text(
        draw,
        (width / 2, height - 17),
        "The arm is shown before the turn; compare the new tip with both targets.",
        fill=INK,
        font=_font(17),
    )
    return image


def analytic_gold(scene: dict) -> str:
    """Compare target distances after applying the clockwise quarter-turn."""

    tip = _post_tip(scene)
    red_distance = _distance(tip, _target(scene, "red"))
    blue_distance = _distance(tip, _target(scene, "blue"))
    if abs(red_distance - blue_distance) <= 1e-9:
        return "tie"
    return "red" if red_distance < blue_distance else "blue"


def margin(scene: dict) -> float:
    tip = _post_tip(scene)
    return abs(_distance(tip, _target(scene, "red")) - _distance(tip, _target(scene, "blue")))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 24.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases are declared; each semantic field is rendered directly."""

    return []
