"""Deterministic scene generator and independent analytic arm."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 840, 560
COLORS = ("red", "green", "blue")
PALETTES = (
    {"red": "#d32f2f", "green": "#2e7d32", "blue": "#1565c0"},
    {"red": "#c62828", "green": "#388e3c", "blue": "#1e5bb8"},
    {"red": "#e53935", "green": "#27843f", "blue": "#2457c5"},
)


def _slot_centers(rotation: float, dx: float, dy: float, rx: float, ry: float) -> list[tuple[float, float]]:
    return [
        (
            round(WIDTH / 2 + dx + rx * math.cos(rotation + index * math.pi / 3), 3),
            round(HEIGHT / 2 + dy + ry * math.sin(rotation + index * math.pi / 3), 3),
        )
        for index in range(6)
    ]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    answer = COLORS[int(seed) % 3]
    # The three matching edges cover the six equally spaced holes. Rotating their
    # phase and the whole layout prevents answer identity from fixing a location.
    phase = rng.randrange(2)
    paired_slots = [((phase + 2 * i) % 6, (phase + 2 * i + 1) % 6) for i in range(3)]
    target_edge = rng.randrange(3)
    remaining_colors = [color for color in COLORS if color != answer]
    rng.shuffle(remaining_colors)
    edge_colors = []
    cursor = 0
    for edge_index in range(3):
        if edge_index == target_edge:
            edge_colors.append(answer)
        else:
            edge_colors.append(remaining_colors[cursor])
            cursor += 1

    rotation = rng.choice((0.0, math.pi / 6, math.pi / 3, math.pi / 2)) + rng.uniform(-0.035, 0.035)
    centers = _slot_centers(
        rotation,
        rng.uniform(-15.0, 15.0),
        rng.uniform(-12.0, 12.0),
        rng.uniform(238.0, 258.0),
        rng.uniform(166.0, 182.0),
    )
    pieces = []
    for edge_index, ((a, b), color) in enumerate(zip(paired_slots, edge_colors)):
        holes = []
        for slot in (a, b):
            outer = rng.uniform(50.0, 57.0)
            wall = rng.uniform(18.0, 23.0)
            holes.append(
                {
                    "slot": slot,
                    "x": centers[slot][0],
                    "y": centers[slot][1],
                    "outer": round(outer, 3),
                    "inner": round(outer - wall, 3),
                    "marker": color,
                    "marker_radius": round(rng.uniform(11.0, 14.0), 3),
                }
            )
        if edge_index == target_edge:
            pieces.append({"holes": holes, "connected": True, "bridge_width": round(rng.uniform(31.0, 39.0), 3)})
        else:
            for hole in holes:
                pieces.append({"holes": [hole], "connected": False, "bridge_width": 0.0})
    rng.shuffle(pieces)
    return {"pieces": pieces, "palette": dict(PALETTES[rng.randrange(len(PALETTES))])}


def _ordered_holes(scene: dict) -> list[dict]:
    return sorted((hole for piece in scene["pieces"] for hole in piece["holes"]), key=lambda h: int(h["slot"]))


def render(scene: dict) -> Image.Image:
    supersample = 3
    image = Image.new("RGB", (WIDTH * supersample, HEIGHT * supersample), "white")
    draw = ImageDraw.Draw(image)
    pieces = sorted(scene["pieces"], key=lambda p: min(int(h["slot"]) for h in p["holes"]))
    # Connectors are drawn beneath rings, making foreground connectivity explicit.
    for piece in pieces:
        if piece["connected"]:
            if len(piece["holes"]) != 2:
                raise ValueError("connected piece must contain exactly two holes")
            a, b = piece["holes"]
            draw.line(
                ((a["x"] * supersample, a["y"] * supersample), (b["x"] * supersample, b["y"] * supersample)),
                fill="#111111",
                width=round(float(piece["bridge_width"]) * supersample),
            )
    holes = _ordered_holes(scene)
    for hole in holes:
        x, y, radius = float(hole["x"]) * supersample, float(hole["y"]) * supersample, float(hole["outer"]) * supersample
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill="#111111")
    for hole in holes:
        x, y, radius = float(hole["x"]) * supersample, float(hole["y"]) * supersample, float(hole["inner"]) * supersample
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill="white")
    for hole in holes:
        x, y, radius = float(hole["x"]) * supersample, float(hole["y"]) * supersample, float(hole["marker_radius"]) * supersample
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=scene["palette"][hole["marker"]])
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    connected = [piece for piece in scene["pieces"] if piece["connected"]]
    if len(connected) != 1 or len(connected[0]["holes"]) != 2:
        raise ValueError("scene must have one two-hole connected piece")
    labels = {hole["marker"] for hole in connected[0]["holes"]}
    if len(labels) != 1:
        raise ValueError("connected holes must carry a same-colored pair")
    return labels.pop()


def margin(scene: dict) -> float:
    holes = _ordered_holes(scene)
    wall = min(float(h["outer"]) - float(h["inner"]) for h in holes)
    marker_clearance = min(float(h["inner"]) - float(h["marker_radius"]) for h in holes)
    # For distinct pieces, use the visible Euclidean gap between outer disks.
    component_of = {int(h["slot"]): index for index, p in enumerate(scene["pieces"]) for h in p["holes"]}
    gaps = []
    for i, first in enumerate(holes):
        for second in holes[i + 1 :]:
            if component_of[int(first["slot"])] == component_of[int(second["slot"])]:
                continue
            distance = math.hypot(float(first["x"]) - float(second["x"]), float(first["y"]) - float(second["y"]))
            gaps.append(distance - float(first["outer"]) - float(second["outer"]))
    return round(min([wall, marker_clearance, *gaps]), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 14.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["pieces"] = list(reversed(twin["pieces"]))
    return [("reverse_unordered_piece_list", twin)]
