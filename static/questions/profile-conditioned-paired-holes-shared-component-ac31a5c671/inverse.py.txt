"""Pixel-only inverse arm; deliberately imports no candidate implementation."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _label(mask: np.ndarray, minimum_area: int) -> tuple[np.ndarray, list[dict]]:
    height, width = mask.shape
    labels = np.zeros((height, width), dtype=np.int32)
    components: list[dict] = []
    next_label = 0
    for y0, x0 in zip(*np.nonzero(mask)):
        if labels[y0, x0]:
            continue
        next_label += 1
        labels[y0, x0] = next_label
        queue = deque([(int(y0), int(x0))])
        pixels = []
        while queue:
            y, x = queue.popleft()
            pixels.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and labels[ny, nx] == 0:
                    labels[ny, nx] = next_label
                    queue.append((ny, nx))
        if len(pixels) >= minimum_area:
            ys = [p[0] for p in pixels]
            xs = [p[1] for p in pixels]
            components.append({"label": next_label, "area": len(pixels), "cx": sum(xs) / len(xs), "cy": sum(ys) / len(ys)})
        else:
            for y, x in pixels:
                labels[y, x] = 0
    return labels, components


def _marker_masks(rgb: np.ndarray) -> dict[str, np.ndarray]:
    red, green, blue = (rgb[:, :, i].astype(int) for i in range(3))
    return {
        "red": (red > 145) & (green < 115) & (blue < 125),
        "green": (green > 85) & (red < 115) & (blue < 150),
        "blue": (blue > 125) & (red < 115) & (green < 135),
    }


def _enclosing_black_label(labels: np.ndarray, cx: float, cy: float) -> int:
    height, width = labels.shape
    found = []
    # A marker is certified as lying in a bounded hole by encountering the same
    # black component in eight outward directions.
    for angle in np.linspace(0.0, 2.0 * np.pi, 8, endpoint=False):
        dx, dy = float(np.cos(angle)), float(np.sin(angle))
        hit = 0
        for radius in range(16, 70):
            x, y = round(cx + dx * radius), round(cy + dy * radius)
            if not (0 <= x < width and 0 <= y < height):
                break
            if labels[y, x] > 0:
                hit = int(labels[y, x])
                break
        if hit:
            found.append(hit)
    if len(found) < 7 or len(set(found)) != 1:
        raise ValueError(f"marker is not robustly enclosed by one black component: {found}")
    return found[0]


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    black = np.max(rgb, axis=2) < 80
    black_labels, black_components = _label(black, minimum_area=500)
    if len(black_components) != 5:
        raise ValueError(f"expected five connected black pieces, found {len(black_components)}")
    associations: dict[str, list[int]] = {}
    for color, mask in _marker_masks(rgb).items():
        _, markers = _label(mask, minimum_area=180)
        if len(markers) != 2:
            raise ValueError(f"expected two {color} markers, found {len(markers)}")
        associations[color] = [_enclosing_black_label(black_labels, item["cx"], item["cy"]) for item in markers]
    matches = [color for color, values in associations.items() if values[0] == values[1]]
    if len(matches) != 1:
        raise ValueError(f"ambiguous hole-component associations: {associations}")
    return matches[0]
