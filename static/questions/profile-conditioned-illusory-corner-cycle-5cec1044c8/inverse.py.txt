"""Image-only inverse arm: recover the closed ring of mutually pointing wedges.

Input is the final PNG plus the public question contract and nothing else.  The
arm segments the elements, recovers each element's centre, wedge bisector and
wedge opening from raw pixel moments, rebuilds the pointing graph, and reports
the size of the unique closed ring.
"""

from __future__ import annotations

import math
from collections import Counter, deque

import numpy as np
from PIL import Image

ABSTAIN = "abstain"
ANSWER_SET = ("3", "4", "5")
TOL_DEG = 9.0
MIN_BLOB_AREA = 900


def _wrap(angle: float) -> float:
    return (angle + math.pi) % (2.0 * math.pi) - math.pi


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Four-connected blobs of the foreground mask, as index arrays."""
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    blobs: list[np.ndarray] = []
    for sy, sx in np.argwhere(mask).tolist():
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        pixels = []
        while queue:
            y, x = queue.popleft()
            pixels.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(pixels) >= MIN_BLOB_AREA:
            blobs.append(np.array(pixels, dtype=np.int64))
    return blobs


def _elements(image: Image.Image) -> list[dict]:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    flat = array.reshape(-1, 3)
    packed = (flat[:, 0].astype(np.int64) << 16) | (flat[:, 1].astype(np.int64) << 8) | flat[:, 2]
    counts = Counter(packed.tolist())
    background = counts.most_common(1)[0][0]
    grid = packed.reshape(array.shape[:2])
    mask = grid != background

    found: list[dict] = []
    for pixels in _components(mask):
        ys = pixels[:, 0].astype(np.float64)
        xs = pixels[:, 1].astype(np.float64)
        area = float(len(pixels))
        cy, cx = float(ys.mean()), float(xs.mean())
        radius = math.sqrt(area / math.pi)
        colours = Counter(grid[pixels[:, 0], pixels[:, 1]].tolist())
        if len(colours) < 2:
            return []
        ranked = colours.most_common()
        wedge_colour = ranked[1][0]
        wedge_area = float(ranked[1][1])
        if wedge_area < 40.0 or wedge_area > 0.5 * area:
            return []
        sel = grid[pixels[:, 0], pixels[:, 1]] == wedge_colour
        wy = ys[sel] - cy
        wx = xs[sel] - cx
        # Bisector: mean direction of the wedge's own pixels about the centre.
        bisector = math.atan2(float(wy.mean()), float(wx.mean()))
        # Opening: a circular sector of half-angle h has area h * radius ** 2.
        half_angle = wedge_area / (radius * radius)
        if not (math.radians(12.0) <= half_angle <= math.radians(86.0)):
            return []
        found.append({"x": cx, "y": cy, "bisector": bisector, "half_angle": half_angle})
    return found


def _pointing_error(element: dict, tx: float, ty: float) -> float:
    target = math.atan2(ty - element["y"], tx - element["x"])
    best = math.pi
    for side in (
        element["bisector"] - element["half_angle"],
        element["bisector"] + element["half_angle"],
    ):
        best = min(best, abs(_wrap(side - target)))
    return math.degrees(best)


def decision_from_image(image: Image.Image) -> str:
    """Return the number of corners of the hidden polygon, or abstain."""
    elements = _elements(image)
    if len(elements) < 4:
        return ABSTAIN

    adjacency: dict[int, list[int]] = {}
    for i in range(len(elements)):
        for j in range(i + 1, len(elements)):
            a, b = elements[i], elements[j]
            slack = max(
                _pointing_error(a, b["x"], b["y"]),
                _pointing_error(b, a["x"], a["y"]),
            )
            if slack < TOL_DEG:
                adjacency.setdefault(i, []).append(j)
                adjacency.setdefault(j, []).append(i)
    if not adjacency:
        return ABSTAIN
    if any(len(neighbours) != 2 for neighbours in adjacency.values()):
        return ABSTAIN

    start = min(adjacency)
    ring = {start}
    previous, current = start, adjacency[start][0]
    while current != start:
        ring.add(current)
        nxt = [n for n in adjacency[current] if n != previous]
        if len(nxt) != 1:
            return ABSTAIN
        previous, current = current, nxt[0]
    if len(ring) != len(adjacency):
        return ABSTAIN
    answer = str(len(ring))
    return answer if answer in ANSWER_SET else ABSTAIN
