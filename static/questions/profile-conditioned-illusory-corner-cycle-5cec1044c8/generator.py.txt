"""Wedge-disc scenes whose hidden polygon emerges from mutual corner pointing.

Every disc is a filled dark circle carrying one pale wedge whose tip sits at the
disc centre.  The two straight sides of a wedge are rays from that centre.  A
subset of the discs are the corners of one convex polygon: for each of those the
two wedge sides point exactly at the two neighbouring corner discs and the wedge
opens toward the inside of the polygon, so the polygon's sides are never drawn -
they exist only as the global closure of the pointing relation.  The remaining
discs are decoys whose wedges point at nothing.  The decision is how many discs
belong to the unique closed ring.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 760
HEIGHT = 760
DISC_RADIUS = 30
N_DISCS = 10
CLASS_NAMES = ("3", "4", "5")
ABSTAIN = "abstain"

# Angular tolerance, in degrees, at which a wedge side counts as pointing at a
# disc centre.  It is the same finite-resolution tolerance the pixel arm uses.
TOL_DEG = 9.0
# Scenes whose nearest spurious mutual pointing is closer than this are ambiguous.
QUARANTINE_DEG = 14.0
# Rejection sampling keeps every generated scene clear of the quarantine band.
SAMPLING_FLOOR_DEG = 22.0

# Every scene spends exactly this much total wedge opening, in radians, so the
# total pale ink is identical for triangles, quadrilaterals and pentagons.
TOTAL_HALF_ANGLE = 8.0
MIN_HALF = math.radians(21.0)
MAX_HALF = math.radians(76.0)

MIN_CENTRE_GAP = 3.1 * DISC_RADIUS
BORDER = DISC_RADIUS + 14

_BG = (246, 244, 239)
_DISC = (34, 38, 48)
_WEDGE = (156, 160, 172)


def _wrap(angle: float) -> float:
    """Signed angular difference folded into (-pi, pi]."""
    return (angle + math.pi) % (2.0 * math.pi) - math.pi


def _pointing_error(
    x: float, y: float, bisector: float, half_angle: float, tx: float, ty: float
) -> float:
    """Degrees by which the better wedge side at (x, y) misses the point."""
    target = math.atan2(ty - y, tx - x)
    best = math.pi
    for side in (bisector - half_angle, bisector + half_angle):
        best = min(best, abs(_wrap(side - target)))
    return math.degrees(best)


def _mutual_slack(discs: list[dict], i: int, j: int) -> float:
    """How far the pair (i, j) is from being a mutual pointing, in degrees."""
    a, b = discs[i], discs[j]
    return max(
        _pointing_error(a["x"], a["y"], a["bisector"], a["half_angle"], b["x"], b["y"]),
        _pointing_error(b["x"], b["y"], b["bisector"], b["half_angle"], a["x"], a["y"]),
    )


def _links(discs: list[dict]) -> set[tuple[int, int]]:
    out: set[tuple[int, int]] = set()
    for i in range(len(discs)):
        for j in range(i + 1, len(discs)):
            if _mutual_slack(discs, i, j) < TOL_DEG:
                out.add((i, j))
    return out


def _cycle_size(discs: list[dict]) -> int | None:
    """Size of the unique closed ring of mutually pointing discs, else ``None``."""
    links = _links(discs)
    degree: dict[int, list[int]] = {}
    for i, j in links:
        degree.setdefault(i, []).append(j)
        degree.setdefault(j, []).append(i)
    if not degree:
        return None
    if any(len(neighbours) != 2 for neighbours in degree.values()):
        return None
    start = min(degree)
    seen = {start}
    previous, current = start, degree[start][0]
    while current != start:
        seen.add(current)
        nxt = [n for n in degree[current] if n != previous]
        if len(nxt) != 1:
            return None
        previous, current = current, nxt[0]
    if len(seen) != len(degree):
        return None
    return len(seen)


def _far_enough(x: float, y: float, placed: list[tuple[float, float]]) -> bool:
    return all(math.hypot(x - px, y - py) >= MIN_CENTRE_GAP for px, py in placed)


def _polygon(rng: random.Random, k: int) -> list[tuple[float, float]]:
    """A convex k-gon: jittered angles on a jittered circle, always in order."""
    cx = WIDTH / 2.0 + rng.uniform(-26.0, 26.0)
    cy = HEIGHT / 2.0 + rng.uniform(-26.0, 26.0)
    radius = rng.uniform(200.0, 258.0)
    base = rng.uniform(0.0, 2.0 * math.pi)
    step = 2.0 * math.pi / k
    spread = 0.30 * step
    angles = sorted(base + i * step + rng.uniform(-spread, spread) for i in range(k))
    points = []
    for angle in angles:
        scale = radius * rng.uniform(0.86, 1.0)
        points.append((cx + scale * math.cos(angle), cy + scale * math.sin(angle)))
    return points


def _corner_discs(points: list[tuple[float, float]]) -> list[dict] | None:
    """Turn polygon corners into discs whose wedge sides face both neighbours."""
    k = len(points)
    discs = []
    for index, (x, y) in enumerate(points):
        if not (BORDER <= x <= WIDTH - BORDER and BORDER <= y <= HEIGHT - BORDER):
            return None
        prev_pt = points[(index - 1) % k]
        next_pt = points[(index + 1) % k]
        to_prev = math.atan2(prev_pt[1] - y, prev_pt[0] - x)
        to_next = math.atan2(next_pt[1] - y, next_pt[0] - x)
        half = abs(_wrap(to_next - to_prev)) / 2.0
        if not (math.radians(21.0) <= half <= math.radians(76.0)):
            return None
        bisector = to_prev + _wrap(to_next - to_prev) / 2.0
        discs.append(
            {
                "x": round(x, 4),
                "y": round(y, 4),
                "bisector": round(bisector % (2.0 * math.pi), 6),
                "half_angle": round(half, 6),
            }
        )
    return discs


def sample_scene(seed: int) -> dict:
    """Deterministically sample one balanced, unambiguous scene."""
    rng = random.Random(seed * 7919 + 13)
    k = 3 + seed % 3
    for _ in range(4000):
        points = _polygon(rng, k)
        if min(
            math.hypot(a[0] - b[0], a[1] - b[1])
            for i, a in enumerate(points)
            for b in points[i + 1 :]
        ) < MIN_CENTRE_GAP:
            continue
        corners = _corner_discs(points)
        if corners is None:
            continue
        discs = list(corners)
        placed = [(d["x"], d["y"]) for d in discs]
        ok = True
        for _ in range(N_DISCS - k):
            for _attempt in range(600):
                x = rng.uniform(BORDER, WIDTH - BORDER)
                y = rng.uniform(BORDER, HEIGHT - BORDER)
                if not _far_enough(x, y, placed):
                    continue
                decoy = {
                    "x": round(x, 4),
                    "y": round(y, 4),
                    "bisector": round(rng.uniform(0.0, 2.0 * math.pi), 6),
                    "half_angle": rng.uniform(MIN_HALF, MAX_HALF),
                }
                discs.append(decoy)
                placed.append((x, y))
                break
            else:
                ok = False
                break
        if not ok or len(discs) != N_DISCS:
            continue
        # Ink balancing: shift the decoy openings so that every scene, whatever
        # its answer, spends the same total wedge opening.  Corner openings are
        # fixed by the polygon, so total pale ink cannot track the corner count.
        decoys = discs[k:]
        shift = (TOTAL_HALF_ANGLE - sum(d["half_angle"] for d in discs)) / len(decoys)
        if any(not MIN_HALF <= d["half_angle"] + shift <= MAX_HALF for d in decoys):
            continue
        for decoy in decoys:
            decoy["half_angle"] = round(decoy["half_angle"] + shift, 6)
        rng.shuffle(discs)
        scene = {
            "width": WIDTH,
            "height": HEIGHT,
            "bg_color": list(_BG),
            "disc_color": list(_DISC),
            "wedge_color": list(_WEDGE),
            "radius": DISC_RADIUS,
            "discs": discs,
        }
        if _cycle_size(discs) != k:
            continue
        if margin(scene) < SAMPLING_FLOOR_DEG:
            continue
        return scene
    raise RuntimeError(f"no admissible scene for seed {seed}")


def render(scene: dict) -> Image.Image:
    """Aliased Pillow raster: one dark disc plus one pale wedge per element."""
    image = Image.new("RGB", (scene["width"], scene["height"]), tuple(scene["bg_color"]))
    draw = ImageDraw.Draw(image)
    radius = scene["radius"]
    for disc in scene["discs"]:
        x, y = disc["x"], disc["y"]
        box = (x - radius, y - radius, x + radius, y + radius)
        draw.ellipse(box, fill=tuple(scene["disc_color"]))
        start = math.degrees(disc["bisector"] - disc["half_angle"])
        end = math.degrees(disc["bisector"] + disc["half_angle"])
        draw.pieslice(box, start, end, fill=tuple(scene["wedge_color"]))
    return image


def analytic_gold(scene: dict) -> str:
    """Recompute the closed ring from the rendered disc geometry alone."""
    size = _cycle_size(scene["discs"])
    if size is None or str(size) not in CLASS_NAMES:
        return ABSTAIN
    return str(size)


def margin(scene: dict) -> float:
    """Degrees of clearance of the nearest spurious mutual pointing."""
    discs = scene["discs"]
    links = _links(discs)
    slacks = [
        _mutual_slack(discs, i, j)
        for i in range(len(discs))
        for j in range(i + 1, len(discs))
        if (i, j) not in links
    ]
    worst_true = max((_mutual_slack(discs, i, j) for i, j in links), default=0.0)
    return round(min(slacks, default=180.0) - worst_true, 6)


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_DEG)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical relabellings of the unordered disc set."""
    discs = scene["discs"]
    out: list[tuple[str, dict]] = []
    rotated = discs[1:] + discs[:1]
    out.append(("cyclic_shift_of_disc_list", dict(scene, discs=rotated)))
    out.append(("reversed_disc_list", dict(scene, discs=list(reversed(discs)))))
    wound = [
        dict(d, bisector=d["bisector"] + 2.0 * math.pi * (1 if index % 2 == 0 else -1))
        for index, d in enumerate(discs)
    ]
    out.append(("bisector_winding_shift", dict(scene, discs=wound)))
    return out
