"""Independent pixel-only inverse for the visible button-hole count."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_INK_THRESHOLD = 245
_MIN_FACE_AREA = 3_000
_MIN_FACE_SPAN = 60
_MAX_FACE_SPAN = 120
_MIN_HOLE_AREA = 25
_MAX_HOLE_AREA = 260
_VALID_COUNTS = (3, 4, 5, 6)


class OracleAbstain(ValueError):
    """The raster does not expose one intact, countable button face."""


def _foreground(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    return np.any(array < _INK_THRESHOLD, axis=2)


def _component_areas(mask: np.ndarray) -> list[int]:
    """Return 4-connected component areas in a small boolean mask."""
    seen = np.zeros_like(mask, dtype=bool)
    areas: list[int] = []
    height, width = mask.shape
    for y, x in np.argwhere(mask):
        y = int(y)
        x = int(x)
        if seen[y, x]:
            continue
        queue: deque[tuple[int, int]] = deque([(y, x)])
        seen[y, x] = True
        area = 0
        while queue:
            cy, cx = queue.popleft()
            area += 1
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        areas.append(area)
    return areas


def _enclosed_void_areas(mask: np.ndarray) -> list[int]:
    """Find background components inside the foreground bounding box."""
    coordinates = np.argwhere(mask)
    if not len(coordinates):
        raise OracleAbstain("no foreground face")
    y0, x0 = coordinates.min(axis=0)
    y1, x1 = coordinates.max(axis=0) + 1
    crop = mask[int(y0) : int(y1), int(x0) : int(x1)]
    free = ~crop
    reachable = np.zeros_like(free, dtype=bool)
    height, width = free.shape
    queue: deque[tuple[int, int]] = deque()
    for x in range(width):
        if free[0, x] and not reachable[0, x]:
            reachable[0, x] = True
            queue.append((0, x))
        if free[height - 1, x] and not reachable[height - 1, x]:
            reachable[height - 1, x] = True
            queue.append((height - 1, x))
    for y in range(height):
        if free[y, 0] and not reachable[y, 0]:
            reachable[y, 0] = True
            queue.append((y, 0))
        if free[y, width - 1] and not reachable[y, width - 1]:
            reachable[y, width - 1] = True
            queue.append((y, width - 1))
    while queue:
        cy, cx = queue.popleft()
        for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
            if 0 <= ny < height and 0 <= nx < width and free[ny, nx] and not reachable[ny, nx]:
                reachable[ny, nx] = True
                queue.append((ny, nx))
    enclosed = free & ~reachable
    return [area for area in _component_areas(enclosed) if _MIN_HOLE_AREA <= area <= _MAX_HOLE_AREA]


def _measure(image: Image.Image) -> int:
    mask = _foreground(image)
    coordinates = np.argwhere(mask)
    if len(coordinates) < _MIN_FACE_AREA:
        raise OracleAbstain("no sufficiently large button face")
    y0, x0 = coordinates.min(axis=0)
    y1, x1 = coordinates.max(axis=0)
    span_y = int(y1 - y0 + 1)
    span_x = int(x1 - x0 + 1)
    if not (_MIN_FACE_SPAN <= span_x <= _MAX_FACE_SPAN and _MIN_FACE_SPAN <= span_y <= _MAX_FACE_SPAN):
        raise OracleAbstain("face span is outside the raster visibility band")
    aspect = span_x / span_y
    if not 0.82 <= aspect <= 1.22:
        raise OracleAbstain("foreground is not a compact button face")
    # Reject a severely fragmented body while allowing the separate dark rims
    # to remain part of the same connected face under antialiasing.
    if max(_component_areas(mask)) < 0.70 * len(coordinates):
        raise OracleAbstain("button face is fragmented")
    holes = _enclosed_void_areas(mask)
    if len(holes) not in _VALID_COUNTS:
        raise OracleAbstain(f"ambiguous enclosed-hole count {len(holes)}")
    return len(holes)


def decision_from_image(image: Image.Image) -> str:
    """Count enclosed background voids using only the final PNG."""
    return str(_measure(image))
