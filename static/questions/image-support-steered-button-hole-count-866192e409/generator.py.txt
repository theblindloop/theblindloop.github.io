"""Deterministic raster world for counting punched holes in a button face."""

from __future__ import annotations

import math
import random
from pathlib import Path

from PIL import Image, ImageDraw

RENDERER_VERSION = "button-hole-count-renderer-0.1.0"
WIDTH = 640
HEIGHT = 640
BACKGROUND = "#ffffff"
HOLE_COUNTS = (3, 4, 5, 6)
BODY_COLORS = ("#d94e5d", "#2878b8", "#4d9b6c", "#c28b2c", "#8b5bb4", "#d66d3d")
BORDER_COLORS = ("#20242b", "#252b36", "#29262f")
HOLE_RING_COLORS = ("#20242b", "#30343b")
FONT_PATHS = (
    "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
    "/usr/share/fonts/dejavu/DejaVuSans.ttf",
)


def sample_scene(seed: int) -> dict:
    """Sample one compact, fully visible button with a balanced hole count."""
    rng = random.Random(int(seed))
    count = HOLE_COUNTS[int(seed) % len(HOLE_COUNTS)]
    # Keep the face in the middle band but translate it substantially enough
    # that absolute position cannot identify the count.  The 90-ish pixel
    # diameter stays within at most three cells of the protected 8x8 ablation.
    center_x = rng.uniform(270.0, 370.0)
    center_y = rng.uniform(270.0, 370.0)
    body_radius = rng.uniform(43.0, 47.0)
    border_width = rng.uniform(5.0, 7.0)
    hole_radius = rng.uniform(5.4, 6.4)
    hole_ring_width = rng.uniform(2.0, 2.8)
    hole_orbit = rng.uniform(20.0, 22.0)
    hole_rotation = rng.uniform(0.0, 2.0 * math.pi)
    holes = [
        [
            round(center_x + hole_orbit * math.cos(hole_rotation + 2.0 * math.pi * i / count), 6),
            round(center_y + hole_orbit * math.sin(hole_rotation + 2.0 * math.pi * i / count), 6),
        ]
        for i in range(count)
    ]
    return {
        "holes": holes,
        "center_x": round(center_x, 6),
        "center_y": round(center_y, 6),
        "body_radius": round(body_radius, 6),
        "hole_radius": round(hole_radius, 6),
        "hole_rotation": round(hole_rotation, 12),
        "body_color": BODY_COLORS[(int(seed) * 13 + 7) % len(BODY_COLORS)],
        "border_color": BORDER_COLORS[(int(seed) * 5 + 1) % len(BORDER_COLORS)],
        "border_width": round(border_width, 6),
        "hole_ring_color": HOLE_RING_COLORS[(int(seed) * 3 + 1) % len(HOLE_RING_COLORS)],
        "hole_ring_width": round(hole_ring_width, 6),
        "width": WIDTH,
        "height": HEIGHT,
        "background": BACKGROUND,
        "antialias": True,
    }


def render(scene: dict) -> Image.Image:
    """Render only the serialized scene; no hidden count-specific decoration."""
    width = int(scene["width"])
    height = int(scene["height"])
    scale = 4 if bool(scene["antialias"]) else 1
    image = Image.new("RGB", (width * scale, height * scale), scene["background"])
    draw = ImageDraw.Draw(image)

    cx = float(scene["center_x"]) * scale
    cy = float(scene["center_y"]) * scale
    body_radius = float(scene["body_radius"]) * scale
    border_width = float(scene["border_width"]) * scale

    def box(x: float, y: float, radius: float) -> tuple[float, float, float, float]:
        return (x - radius, y - radius, x + radius, y + radius)

    draw.ellipse(
        box(cx, cy, body_radius + border_width / 2.0),
        fill=scene["border_color"],
    )
    draw.ellipse(
        box(cx, cy, max(1.0, body_radius - border_width / 2.0)),
        fill=scene["body_color"],
    )

    inner_radius = float(scene["hole_radius"]) * scale
    outer_radius = inner_radius + float(scene["hole_ring_width"]) * scale
    for hx, hy in scene["holes"]:
        x = float(hx) * scale
        y = float(hy) * scale
        draw.ellipse(box(x, y, outer_radius), fill=scene["hole_ring_color"])
        draw.ellipse(box(x, y, inner_radius), fill=scene["background"])

    if scale > 1:
        image = image.resize((width, height), Image.Resampling.LANCZOS)
    return image


def analytic_gold(scene: dict) -> str:
    """The only label input is the rendered list of punched holes."""
    return str(len(scene["holes"]))


def margin(scene: dict) -> float:
    """Return the minimum visible clearance above the hole-count boundary."""
    holes = [(float(x), float(y)) for x, y in scene["holes"]]
    body_radius = float(scene["body_radius"])
    border_width = float(scene["border_width"])
    hole_radius = float(scene["hole_radius"])
    ring_width = float(scene["hole_ring_width"])
    orbit_clearance = body_radius - border_width / 2.0 - max(
        math.dist((float(scene["center_x"]), float(scene["center_y"])), point)
        for point in holes
    ) - hole_radius - ring_width
    pair_clearance = min(
        math.dist(holes[i], holes[j]) - 2.0 * (hole_radius + ring_width)
        for i in range(len(holes))
        for j in range(i + 1, len(holes))
    )
    border_clearance = min(
        float(scene["center_x"]) - body_radius - border_width,
        float(scene["width"]) - float(scene["center_x"]) - body_radius - border_width,
        float(scene["center_y"]) - body_radius - border_width,
        float(scene["height"]) - float(scene["center_y"]) - body_radius - border_width,
    )
    return float(min(2.0 * hole_radius - 5.0, orbit_clearance - 4.0, pair_clearance - 2.0, border_clearance - 10.0))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reversing independent hole serialization preserves pixels and gold."""
    twin = dict(scene)
    twin["holes"] = list(reversed(scene["holes"]))
    return [("reversed_hole_list", twin)]
