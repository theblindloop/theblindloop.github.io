"""Renderer for `lagoon_breach_side_tally`.

One thick dark shelf spans the page from far left to far right. Six white
basins are cut into it, three nearer the left page edge and three nearer the
right page edge. Every basin carries exactly one straight white channel that
runs from a rim of the shelf towards it. Some channels break through into the
basin, which joins that basin to the outside water; the others stop short
behind a wall of dark ink, so that basin stays a sealed lagoon. The decision is
which page side holds more sealed lagoons.

Only `cavities` feeds the analytic gold, and every cavity is painted at the
place and size it declares. Nothing about the decision is stored in the scene.
"""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 1000
HEIGHT = 640

BAND_X0, BAND_X1 = 60, 940
BAND_Y0, BAND_Y1 = 140, 500

INK = (24, 24, 24)
BACKGROUND = (255, 255, 255)

DECOR_INKS = (
    (235, 120, 120),
    (120, 150, 235),
    (240, 175, 90),
    (185, 185, 185),
    (185, 140, 225),
    (130, 205, 140),
)

NOTCH_WIDTH = 24
NOTCH_OVERSHOOT = 10          # how far a channel runs past the rim into open water
BREACH_BITE = 6               # how far a breaking channel eats into its basin
MIN_CLEARANCE = 22.0          # never emitted below this
QUARANTINE_CLEARANCE = 26.0   # crowded, excluded from headline claims

LEFT_SLOTS = (150, 280, 410)
RIGHT_SLOTS = (590, 720, 850)
MIN_H, MAX_H = 92, 148
MIN_W, MAX_W = 64, 84
CENTER_Y = 320


def _rects_of(cavity: dict) -> list[tuple[int, int, int, int]]:
    """Every white rectangle that this cavity cuts out of the shelf."""
    basin = (cavity["x0"], cavity["y0"], cavity["x1"], cavity["y1"])
    nx0 = cavity["notch_x0"]
    nx1 = cavity["notch_x1"]
    if cavity["notch_side"] == "top":
        ny0 = BAND_Y0 - NOTCH_OVERSHOOT
        ny1 = cavity["y0"] + BREACH_BITE if cavity["breached"] else cavity["y0"] - cavity["gap"]
    else:
        ny0 = cavity["y1"] - BREACH_BITE if cavity["breached"] else cavity["y1"] + cavity["gap"]
        ny1 = BAND_Y1 + NOTCH_OVERSHOOT
    return [basin, (nx0, ny0, nx1, ny1)]


def _cut_mask(cavities: list[dict]) -> np.ndarray:
    mask = np.zeros((HEIGHT, WIDTH), dtype=bool)
    for cavity in cavities:
        for x0, y0, x1, y1 in _rects_of(cavity):
            mask[max(y0, 0):max(y1, 0), max(x0, 0):max(x1, 0)] = True
    return mask


def _side_ink(cavities: list[dict]) -> tuple[int, int]:
    """Dark pixel count of the shelf on each page side."""
    mask = _cut_mask(cavities)
    band = ~mask[BAND_Y0:BAND_Y1, BAND_X0:BAND_X1]
    split = WIDTH // 2 - BAND_X0
    return int(band[:, :split].sum()), int(band[:, split:].sum())


def _make_cavity(rng: random.Random, slot_x: int, side: str, breached: bool) -> dict:
    width = rng.randrange(MIN_W, MAX_W + 1, 2)
    height = rng.randrange(MIN_H, MAX_H + 1, 2)
    cx = slot_x + rng.randint(-6, 6)
    cy = CENTER_Y + rng.randint(-22, 22)
    notch_cx = cx + rng.randint(-(width // 2 - 18), width // 2 - 18)
    return {
        "side": side,
        "x0": cx - width // 2,
        "x1": cx + width // 2,
        "y0": cy - height // 2,
        "y1": cy + height // 2,
        "notch_side": rng.choice(("top", "bottom")),
        "notch_x0": notch_cx - NOTCH_WIDTH // 2,
        "notch_x1": notch_cx + NOTCH_WIDTH // 2,
        "breached": breached,
        "gap": rng.randrange(24, 62, 2),
    }


def _clearances(cavities: list[dict]) -> list[float]:
    """Every distance whose sign the reader has to resolve, in pixels."""
    values: list[float] = []
    midline = WIDTH / 2.0
    for cavity in cavities:
        values.append(abs((cavity["x0"] + cavity["x1"]) / 2.0 - midline))
        if not cavity["breached"]:
            values.append(float(cavity["gap"]))          # the sealing wall
        values.append(float(cavity["y0"] - BAND_Y0))     # shelf above the basin
        values.append(float(BAND_Y1 - cavity["y1"]))     # shelf below the basin
        values.append(float(cavity["x0"] - BAND_X0) if cavity["side"] == "left" else float(BAND_X1 - cavity["x1"]))
    ordered = sorted(cavities, key=lambda c: c["x0"])
    for left, right in zip(ordered, ordered[1:]):
        values.append(float(right["x0"] - left["x1"]))
        values.append(float(min(right["notch_x0"], right["x0"]) - max(left["notch_x1"], left["x1"])))
    return values


def _decor(rng: random.Random, cavities: list[dict]) -> list[dict]:
    items: list[dict] = []
    for cavity in cavities:
        if rng.random() < 0.55:
            w = min(cavity["x1"] - cavity["x0"] - 20, 34)
            h = min(cavity["y1"] - cavity["y0"] - 20, 34)
            if w >= 14 and h >= 14:
                cx = (cavity["x0"] + cavity["x1"]) // 2
                cy = (cavity["y0"] + cavity["y1"]) // 2
                items.append({
                    "shape": rng.choice(("box", "ring")),
                    "x0": cx - w // 2, "y0": cy - h // 2, "x1": cx + w // 2, "y1": cy + h // 2,
                    "color": rng.randrange(len(DECOR_INKS)),
                })
    for _ in range(rng.randint(4, 7)):
        w = rng.randrange(30, 80, 2)
        h = rng.randrange(24, 60, 2)
        above = rng.random() < 0.5
        x0 = rng.randint(70, WIDTH - 80 - w)
        y0 = rng.randint(24, BAND_Y0 - NOTCH_OVERSHOOT - 26 - h) if above \
            else rng.randint(BAND_Y1 + NOTCH_OVERSHOOT + 26, HEIGHT - 26 - h)
        items.append({
            "shape": rng.choice(("box", "ring")),
            "x0": x0, "y0": y0, "x1": x0 + w, "y1": y0 + h,
            "color": rng.randrange(len(DECOR_INKS)),
        })
    return items


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    for _ in range(7):            # decorrelate consecutive seeds
        rng.random()
    answer_side = "left" if rng.random() < 0.5 else "right"
    sealed_more = rng.choice((2, 3))
    sealed_less = sealed_more - 1
    sealed = {"left": sealed_more, "right": sealed_less} if answer_side == "left" \
        else {"left": sealed_less, "right": sealed_more}

    for _ in range(400):
        cavities: list[dict] = []
        for side, slots in (("left", LEFT_SLOTS), ("right", RIGHT_SLOTS)):
            flags = [False] * sealed[side] + [True] * (3 - sealed[side])
            rng.shuffle(flags)
            for slot_x, breached in zip(slots, flags):
                cavities.append(_make_cavity(rng, slot_x, side, breached))
        if min(_clearances(cavities)) < MIN_CLEARANCE:
            continue
        # counterbalance total dark ink across the two page sides by nudging the
        # height of one basin on the heavier side.
        for _ in range(200):
            left_ink, right_ink = _side_ink(cavities)
            diff = left_ink - right_ink
            if abs(diff) <= 130:
                break
            # the heavier side loses a little ink by growing one of its basins
            grow_side = "left" if diff > 0 else "right"
            target = [c for c in cavities if c["side"] == grow_side]
            victim = min(target, key=lambda c: c["y1"] - c["y0"])
            if (victim["y1"] - victim["y0"]) + 2 > MAX_H or victim["y1"] + 2 > BAND_Y1 - 40:
                break
            victim["y1"] += 2
        left_ink, right_ink = _side_ink(cavities)
        if abs(left_ink - right_ink) > 260:
            continue
        if min(_clearances(cavities)) < MIN_CLEARANCE:
            continue
        return {
            "body": [BAND_X0, BAND_Y0, BAND_X1, BAND_Y1],
            "cavities": cavities,
            "decor": _decor(rng, cavities),
        }
    raise RuntimeError(f"could not lay out a balanced shelf for seed {seed}")


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    bx0, by0, bx1, by1 = scene["body"]
    draw.rectangle([bx0, by0, bx1 - 1, by1 - 1], fill=INK)
    for cavity in scene["cavities"]:
        for x0, y0, x1, y1 in _rects_of(cavity):
            draw.rectangle([x0, y0, x1 - 1, y1 - 1], fill=BACKGROUND)
    for item in scene["decor"]:
        color = DECOR_INKS[item["color"]]
        box = [item["x0"], item["y0"], item["x1"] - 1, item["y1"] - 1]
        if item["shape"] == "box":
            draw.rectangle(box, fill=color)
        else:
            draw.rectangle(box, outline=color, width=5)
    return image


def _sealed_counts(scene: dict) -> tuple[int, int]:
    left = right = 0
    for cavity in scene["cavities"]:
        if cavity["breached"]:
            continue
        if (cavity["x0"] + cavity["x1"]) / 2.0 < WIDTH / 2.0:
            left += 1
        else:
            right += 1
    return left, right


def analytic_gold(scene: dict) -> str:
    left, right = _sealed_counts(scene)
    if left == right:
        raise ValueError("degenerate scene: equal sealed lagoon counts")
    return "left" if left > right else "right"


def margin(scene: dict) -> float:
    return float(min(_clearances(scene["cavities"])))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_CLEARANCE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases are declared: every scene field is painted as drawn."""
    return []
