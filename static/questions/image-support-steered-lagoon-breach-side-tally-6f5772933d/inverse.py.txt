"""Pixel-only inverse arm for `lagoon_breach_side_tally`.

Receives one RGB image and nothing else: no scene, no shelf rectangle, no
basin list, no channel widths, no analytic gold. It segments dark ink by
luminance, labels the four-connected white regions with a run-length union
pass, discards every white region that reaches the page border (that is the
open water), and reports which page side holds more of the remaining sealed
lagoons.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

DARK_THRESHOLD = 110.0
MIN_LAGOON_AREA = 700


def _dark_mask(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.float64)
    luminance = 0.299 * arr[:, :, 0] + 0.587 * arr[:, :, 1] + 0.114 * arr[:, :, 2]
    return luminance < DARK_THRESHOLD


def _row_runs(row: np.ndarray) -> list[tuple[int, int]]:
    padded = np.concatenate(([False], row, [False]))
    edges = np.diff(padded.astype(np.int8))
    starts = np.nonzero(edges == 1)[0]
    ends = np.nonzero(edges == -1)[0]
    return list(zip(starts.tolist(), ends.tolist()))


def _white_components(mask: np.ndarray) -> list[dict]:
    """Four-connected components of a boolean mask, as run-length groups."""
    height, width = mask.shape
    parent: list[int] = []

    def find(node: int) -> int:
        while parent[node] != node:
            parent[node] = parent[parent[node]]
            node = parent[node]
        return node

    def union(a: int, b: int) -> None:
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    runs: list[tuple[int, int, int]] = []          # (row, x0, x1_exclusive)
    per_row: list[list[int]] = []
    for y in range(height):
        indices = []
        for x0, x1 in _row_runs(mask[y]):
            indices.append(len(runs))
            parent.append(len(runs))
            runs.append((y, x0, x1))
        per_row.append(indices)

    for y in range(height - 1):
        upper, lower = per_row[y], per_row[y + 1]
        i = j = 0
        while i < len(upper) and j < len(lower):
            _, ax0, ax1 = runs[upper[i]]
            _, bx0, bx1 = runs[lower[j]]
            if ax1 > bx0 and bx1 > ax0:
                union(upper[i], lower[j])
            if ax1 <= bx1:
                i += 1
            else:
                j += 1

    groups: dict[int, dict] = {}
    for index, (y, x0, x1) in enumerate(runs):
        root = find(index)
        group = groups.setdefault(root, {"area": 0, "sum_x": 0.0, "border": False})
        length = x1 - x0
        group["area"] += length
        group["sum_x"] += (x0 + x1 - 1) * length / 2.0
        if y == 0 or y == height - 1 or x0 == 0 or x1 == width:
            group["border"] = True
    return list(groups.values())


def decision_from_image(image: Image.Image) -> str:
    dark = _dark_mask(image)
    width = dark.shape[1]
    left = right = 0
    for group in _white_components(~dark):
        if group["border"] or group["area"] < MIN_LAGOON_AREA:
            continue
        if group["sum_x"] / group["area"] < width / 2.0:
            left += 1
        else:
            right += 1
    if left == right:
        return "tie"
    return "left" if left > right else "right"
