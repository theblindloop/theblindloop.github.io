"""Deterministic scene sampler and renderer for Running Difference Reveal."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 800, 500
GRID = 7
LABELS = ("X", "plus", "T", "L")
PALE = (244, 240, 225)
DARK = (35, 39, 50)
BACKGROUND = (250, 248, 241)
GRIDLINE = (177, 175, 166)
PALETTES = (
    ((18, 145, 151), (232, 103, 73)),
    ((14, 132, 146), (218, 91, 82)),
    ((26, 151, 141), (225, 96, 67)),
)


def _template(label: str) -> list[int]:
    bits = [0] * (GRID * GRID)
    for row in range(GRID):
        for col in range(GRID):
            if label == "X":
                on = col == row or col == GRID - 1 - row
            elif label == "plus":
                on = row == GRID // 2 or col == GRID // 2
            elif label == "T":
                on = row == 0 or col == GRID // 2
            elif label == "L":
                on = col == 0 or row == GRID - 1
            else:
                raise ValueError(f"unknown glyph {label!r}")
            bits[row * GRID + col] = int(on)
    return bits


def _noisy_output(scene: dict) -> list[int]:
    output = _template(scene["glyph"])
    for index in scene["noise_indices"]:
        output[int(index)] ^= 1
    return output


def _derivative(bits: list[int]) -> list[int]:
    changes: list[int] = []
    for row in range(GRID):
        previous = 0
        for col in range(GRID):
            value = bits[row * GRID + col]
            changes.append(previous ^ value)
            previous = value
    return changes


def _distances(bits: list[int]) -> list[tuple[int, str]]:
    return sorted(
        (sum(a != b for a, b in zip(bits, _template(label))), label)
        for label in LABELS
    )


def _balanced_mask(rng: random.Random, changes: list[int]) -> list[int]:
    ones = [i for i, value in enumerate(changes) if value]
    zeros = [i for i, value in enumerate(changes) if not value]
    target_left = 24 + rng.randrange(2)
    choices = []
    for target_right in (24, 25):
        numerator = target_left + len(ones) - target_right
        if numerator % 2 == 0:
            overlap = numerator // 2
            if 0 <= overlap <= len(ones) and 0 <= target_left - overlap <= len(zeros):
                choices.append((target_right, overlap))
    if not choices:
        target_right, overlap = target_left, min(len(ones), target_left)
    else:
        target_right, overlap = choices[rng.randrange(len(choices))]
    selected = set(rng.sample(ones, overlap))
    selected.update(rng.sample(zeros, target_left - overlap))
    mask = [int(i in selected) for i in range(GRID * GRID)]
    assert sum(mask) == target_left
    assert sum(a ^ b for a, b in zip(mask, changes)) == target_right
    return mask


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    glyph = LABELS[int(seed) % len(LABELS)]
    noise_indices: list[int] = []
    if int(seed) % 11 in (0, 5):
        runner = LABELS[(LABELS.index(glyph) + 1 + (int(seed) // 11) % 3) % len(LABELS)]
        different = [
            i for i, pair in enumerate(zip(_template(glyph), _template(runner))) if pair[0] != pair[1]
        ]
        rng.shuffle(different)
        noise_indices = sorted(different[: max(1, (len(different) - 1) // 2)])
    provisional = {
        "glyph": glyph,
        "noise_indices": noise_indices,
    }
    changes = _derivative(_noisy_output(provisional))
    mask = _balanced_mask(rng, changes)
    return {
        "glyph": glyph,
        "noise_indices": noise_indices,
        "mask_bits": mask,
        "swap_positions": bool(rng.randrange(2)),
        "palette_index": rng.randrange(len(PALETTES)),
        "offset_x": rng.randrange(-9, 10),
        "offset_y": rng.randrange(-16, 17),
        "cell_size": 14 + rng.randrange(2),
    }


def _draw_panel(draw: ImageDraw.ImageDraw, bits: Iterable[int], x: int, y: int, cell: int, frame: tuple[int, int, int]) -> None:
    size = GRID * cell
    border = 6
    draw.rectangle((x - border, y - border, x + size + border - 1, y + size + border - 1), fill=frame)
    draw.rectangle((x, y, x + size - 1, y + size - 1), fill=PALE)
    values = list(bits)
    for row in range(GRID):
        for col in range(GRID):
            x0, y0 = x + col * cell, y + row * cell
            fill = DARK if values[row * GRID + col] else PALE
            draw.rectangle((x0 + 2, y0 + 2, x0 + cell - 3, y0 + cell - 3), fill=fill)
    for step in range(1, GRID):
        xx, yy = x + step * cell, y + step * cell
        draw.line((xx, y, xx, y + size - 1), fill=GRIDLINE, width=1)
        draw.line((x, yy, x + size - 1, yy), fill=GRIDLINE, width=1)


def render(scene: dict) -> Image.Image:
    glyph = scene["glyph"]
    noise_indices = scene["noise_indices"]
    mask = [int(v) for v in scene["mask_bits"]]
    changes = _derivative(_noisy_output({"glyph": glyph, "noise_indices": noise_indices}))
    other = [a ^ b for a, b in zip(mask, changes)]
    swap = bool(scene["swap_positions"])
    palette = PALETTES[int(scene["palette_index"])]
    offset_x = int(scene["offset_x"])
    offset_y = int(scene["offset_y"])
    cell = int(scene["cell_size"])
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    panel_y = 196 + offset_y
    left_x, right_x = 155 + offset_x, 535 + offset_x
    positions = (right_x, left_x) if swap else (left_x, right_x)
    _draw_panel(draw, mask, positions[0], panel_y, cell, palette[0])
    _draw_panel(draw, other, positions[1], panel_y, cell, palette[1])
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["glyph"])


def margin(scene: dict) -> float:
    distances = _distances(_noisy_output(scene))
    return float(distances[1][0] - distances[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
