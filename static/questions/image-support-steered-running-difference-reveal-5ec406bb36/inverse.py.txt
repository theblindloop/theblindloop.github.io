"""Independent final-PNG inverse for Running Difference Reveal."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

GRID = 7
LABELS = ("X", "plus", "T", "L")
TEAL_REFERENCES = np.asarray(((18, 145, 151), (14, 132, 146), (26, 151, 141)), dtype=np.int16)
CORAL_REFERENCES = np.asarray(((232, 103, 73), (218, 91, 82), (225, 96, 67)), dtype=np.int16)


def _template(label: str) -> list[int]:
    result = []
    for row in range(GRID):
        for col in range(GRID):
            if label == "X":
                result.append(int(col == row or col == GRID - 1 - row))
            elif label == "plus":
                result.append(int(row == GRID // 2 or col == GRID // 2))
            elif label == "T":
                result.append(int(row == 0 or col == GRID // 2))
            elif label == "L":
                result.append(int(col == 0 or row == GRID - 1))
    return result


def _color_mask(array: np.ndarray, references: np.ndarray) -> np.ndarray:
    pixels = array.astype(np.int32)
    refs = references.astype(np.int32)
    squared = ((pixels[:, :, None, :] - refs[None, None, :, :]) ** 2).sum(axis=3)
    return squared.min(axis=2) < 400


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    seen = np.zeros(mask.shape, dtype=bool)
    found: list[list[tuple[int, int]]] = []
    height, width = mask.shape
    for yy, xx in np.argwhere(mask):
        y, x = int(yy), int(xx)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        component = []
        while queue:
            cy, cx = queue.popleft()
            component.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        found.append(component)
    return found


def _panel_bits(array: np.ndarray, references: np.ndarray) -> list[int]:
    mask = _color_mask(array, references)
    components = [part for part in _components(mask) if len(part) >= 120]
    if len(components) != 1:
        raise ValueError("missing or fragmented colored panel fiducial")
    points = np.asarray(components[0])
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    outer_w, outer_h = int(x1 - x0 + 1), int(y1 - y0 + 1)
    if abs(outer_w - outer_h) > 2 or not 100 <= outer_w <= 123:
        raise ValueError("colored fiducial has invalid geometry")
    expected_frame = outer_w * outer_h - (outer_w - 12) * (outer_h - 12)
    if len(points) < 0.86 * expected_frame:
        raise ValueError("colored panel frame is incomplete")
    inner_x0, inner_y0 = int(x0 + 6), int(y0 + 6)
    inner_size = outer_w - 12
    cell = inner_size / GRID
    bits = []
    for row in range(GRID):
        for col in range(GRID):
            cx = inner_x0 + (col + 0.5) * cell
            cy = inner_y0 + (row + 0.5) * cell
            radius = max(2, int(cell * 0.23))
            patch = array[
                int(round(cy)) - radius : int(round(cy)) + radius,
                int(round(cx)) - radius : int(round(cx)) + radius,
            ]
            luminance = float(patch.astype(np.float32).mean())
            if 115.0 <= luminance <= 185.0:
                raise ValueError("ambiguous cell shade")
            bits.append(int(luminance < 115.0))
    return bits


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    teal = _panel_bits(array, TEAL_REFERENCES)
    coral = _panel_bits(array, CORAL_REFERENCES)
    changes = [a ^ b for a, b in zip(teal, coral)]
    output = []
    for row in range(GRID):
        state = 0
        for col in range(GRID):
            state ^= changes[row * GRID + col]
            output.append(state)
    scores = sorted((sum(a != b for a, b in zip(output, _template(label))), label) for label in LABELS)
    if scores[0][0] == scores[1][0]:
        raise ValueError("glyph classification tie")
    return scores[0][1]
