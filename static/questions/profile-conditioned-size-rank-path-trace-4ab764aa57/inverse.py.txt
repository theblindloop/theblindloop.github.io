"""Independent inverse: only raster pixels enter this module."""

from __future__ import annotations

from collections import deque
from PIL import Image

_TEMPLATES = {
    "inward spiral": (0, 1, 2, 5, 8, 7, 6, 3, 4),
    "row snake": (0, 1, 2, 5, 4, 3, 6, 7, 8),
    "crossing star": (1, 6, 5, 0, 7, 2, 3, 8, 4),
    "diagonal weave": (0, 2, 4, 6, 8, 5, 3, 1, 7),
}


def _components(image: Image.Image):
    im = image.convert("RGB")
    w, h = im.size
    px = im.load()
    mask = bytearray(w * h)
    for y in range(h):
        row = y * w
        for x in range(w):
            r, g, b = px[x, y]
            if max(r, g, b) - min(r, g, b) > 55 and max(r, g, b) < 230:
                mask[row + x] = 1
    seen = bytearray(w * h)
    found = []
    for start, ink in enumerate(mask):
        if not ink or seen[start]:
            continue
        seen[start] = 1
        queue = deque([start])
        area = sx = sy = 0
        while queue:
            p = queue.popleft()
            y, x = divmod(p, w)
            area += 1
            sx += x
            sy += y
            if x and mask[p-1] and not seen[p-1]:
                seen[p-1] = 1; queue.append(p-1)
            if x+1 < w and mask[p+1] and not seen[p+1]:
                seen[p+1] = 1; queue.append(p+1)
            if y and mask[p-w] and not seen[p-w]:
                seen[p-w] = 1; queue.append(p-w)
            if y+1 < h and mask[p+w] and not seen[p+w]:
                seen[p+w] = 1; queue.append(p+w)
        if area > 80:
            found.append((area, sx / area, sy / area))
    return found


def _cluster_axis(values: list[float]) -> list[float]:
    vals = sorted(values)
    # Three groups of three are guaranteed by the visible 3x3 layout, not coordinates.
    return [sum(vals[i:i+3]) / 3.0 for i in (0, 3, 6)]


def analyze_image(image: Image.Image) -> dict:
    comps = _components(image)
    if len(comps) != 9:
        raise ValueError(f"expected nine colored dot components, found {len(comps)}")
    comps.sort(key=lambda item: item[0])
    if any(comps[i+1][0] - comps[i][0] < 70 for i in range(8)):
        raise ValueError("dot areas do not expose a stable strict rank")
    xs = _cluster_axis([c[1] for c in comps])
    ys = _cluster_axis([c[2] for c in comps])
    sequence = []
    used = set()
    for _, x, y in comps:
        gx = min(range(3), key=lambda k: abs(x - xs[k]))
        gy = min(range(3), key=lambda k: abs(y - ys[k]))
        idx = gy * 3 + gx
        if idx in used:
            raise ValueError("grid assignment is ambiguous")
        used.add(idx)
        sequence.append(idx)
    scores = {name: sum(a != b for a, b in zip(sequence, template)) for name, template in _TEMPLATES.items()}
    ranked = sorted(scores.items(), key=lambda item: (item[1], item[0]))
    if ranked[0][1] != 0 or ranked[1][1] - ranked[0][1] < 3:
        raise ValueError("path template match is ambiguous")
    return {"decision": ranked[0][0], "margin": float(ranked[1][1] - ranked[0][1]), "sequence": sequence}


def decision_from_image(image: Image.Image) -> str:
    return str(analyze_image(image)["decision"])
