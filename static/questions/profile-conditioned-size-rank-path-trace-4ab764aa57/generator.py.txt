"""Deterministic analytic renderer for the size-ranked path world."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

CLASSES = ("inward spiral", "row snake", "crossing star", "diagonal weave")
PALETTES = ((30, 94, 168), (164, 55, 122), (22, 132, 92), (196, 79, 35))
RADII = (7, 9, 11, 13, 15, 17, 19, 21, 23)
GRID = (
    (0, 0), (1, 0), (2, 0),
    (0, 1), (1, 1), (2, 1),
    (0, 2), (1, 2), (2, 2),
)
# Each tuple gives grid-point indices from smallest dot to largest dot.
PATHS = {
    "inward spiral": (0, 1, 2, 5, 8, 7, 6, 3, 4),
    "row snake": (0, 1, 2, 5, 4, 3, 6, 7, 8),
    "crossing star": (1, 6, 5, 0, 7, 2, 3, 8, 4),
    "diagonal weave": (0, 2, 4, 6, 8, 5, 3, 1, 7),
}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    return {
        "path_class": CLASSES[rng.randrange(len(CLASSES))],
        "palette_index": rng.randrange(len(PALETTES)),
        "scale_index": rng.randrange(3),
        "offset_x": rng.randrange(-18, 19),
        "offset_y": rng.randrange(-18, 19),
    }


def _centers(scene: dict) -> list[tuple[int, int]]:
    spacing = (172, 184, 196)[int(scene["scale_index"])]
    left = 360 - spacing + int(scene["offset_x"])
    top = 374 - spacing + int(scene["offset_y"])
    return [(left + gx * spacing, top + gy * spacing) for gx, gy in GRID]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (720, 720), (250, 249, 246))
    draw = ImageDraw.Draw(image)
    # A neutral work area and a visible small-to-large instruction motif.
    draw.rounded_rectangle((46, 78, 674, 686), radius=18, fill=(255, 255, 255), outline=(205, 207, 210), width=3)
    draw.text((62, 28), "CONNECT DOT CENTERS: SMALLEST  ->  LARGEST", fill=(76, 79, 84))
    for x, r in ((530, 7), (566, 11), (606, 15)):
        draw.ellipse((x-r, 45-r, x+r, 45+r), fill=(154, 157, 162))
    color = PALETTES[int(scene["palette_index"])]
    centers = _centers(scene)
    order = PATHS[str(scene["path_class"])]
    radius_at_point = {}
    for rank, point_index in enumerate(order):
        radius_at_point[point_index] = RADII[rank]
    for point_index, (x, y) in enumerate(centers):
        r = radius_at_point[point_index]
        draw.ellipse((x-r, y-r, x+r, y+r), fill=color)
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["path_class"])


def margin(scene: dict) -> float:
    target = PATHS[str(scene["path_class"])]
    return float(min(sum(a != b for a, b in zip(target, other)) for name, other in PATHS.items() if name != scene["path_class"]))


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
