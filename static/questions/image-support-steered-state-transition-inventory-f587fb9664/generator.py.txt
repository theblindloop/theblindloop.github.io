"""Renderer and analytic arm for a connected directed-transition inventory."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "state-transition-inventory-0.1.0"
CANVAS = (800, 640)
STATION_COUNT = 10
STATE_LABELS = ("A", "B", "C")
EDGE_LABELS = ("AB", "BC", "CA", "AC", "CB", "BA")
PALETTE = {
    "background": (248, 250, 253),
    "route": (34, 58, 82),
    "operation": (213, 91, 57),
    "state_a": (193, 68, 67),
    "state_b": (39, 145, 91),
    "state_c": (61, 103, 190),
    "neutral": (164, 174, 185),
    "text": (28, 39, 52),
}


def _colors(scene: dict[str, Any]) -> dict[str, tuple[int, int, int]]:
    return {
        name: tuple(int(channel) for channel in value)
        for name, value in scene["palette"].items()
    }


def _trace(initial_state: int, operations: list[int]) -> list[int]:
    state = int(initial_state) % 3
    trace: list[int] = []
    for operation in operations:
        state = (state + int(operation)) % 3
        trace.append(state)
    return trace


def _edge_counts(initial_state: int, operations: list[int]) -> dict[str, int]:
    states = [int(initial_state) % 3, *_trace(initial_state, operations)]
    counts = {edge: 0 for edge in EDGE_LABELS}
    for source, target in zip(states, states[1:]):
        if source == target:
            continue
        edge = STATE_LABELS[source] + STATE_LABELS[target]
        counts[edge] += 1
    return counts


def _best_edge(initial_state: int, operations: list[int]) -> tuple[str, float]:
    counts = _edge_counts(initial_state, operations)
    ordered = sorted(EDGE_LABELS, key=lambda edge: (-counts[edge], edge))
    return ordered[0], float(counts[ordered[0]] - counts[ordered[1]])


def _sample_points(rng: random.Random) -> tuple[list[list[int]], int]:
    """Return a monotone-x route contained in a medium central region."""

    base_y = [350, 300, 425, 310, 430, 320, 418, 305, 422, 315, 390, 350]
    phase = rng.randrange(4)
    offsets = (-9, 0, 8, 4)
    xs = [170, 211, 252, 293, 334, 375, 416, 457, 498, 539, 580, 621]
    points: list[list[int]] = []
    for index, (x, y) in enumerate(zip(xs, base_y, strict=True)):
        jitter = 0 if index in (0, len(xs) - 1) else rng.randint(-3, 3)
        points.append([x + jitter, y + offsets[phase]])
    return points, phase


def _sample_distractors(rng: random.Random) -> list[dict[str, int]]:
    anchors = [(66, 94), (734, 91), (71, 548), (728, 548)]
    rng.shuffle(anchors)
    return [
        {
            "kind": index % 2,
            "x": x + rng.randint(-8, 8),
            "y": y + rng.randint(-8, 8),
            "size": rng.randint(9, 14),
        }
        for index, (x, y) in enumerate(anchors)
    ]


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a strict, class-balanced directed-transition inventory."""

    rng = random.Random(int(seed))
    route_points, route_phase = _sample_points(rng)
    desired_edge = EDGE_LABELS[int(seed) % len(EDGE_LABELS)]
    for _ in range(3000):
        operations = [rng.choice((-1, 0, 1)) for _ in range(STATION_COUNT)]
        initial_state = rng.randrange(3)
        edge, gap = _best_edge(initial_state, operations)
        counts = _edge_counts(initial_state, operations)
        if (
            edge == desired_edge
            and gap >= 1.0
            and counts[edge] >= 2
            and sum(counts.values()) >= 6
        ):
            break
    else:
        raise RuntimeError("could not sample a unique directed-transition leader")
    return {
        "initial_state": initial_state,
        "operations": operations,
        "route_points": route_points,
        "route_phase": route_phase,
        "distractors": _sample_distractors(rng),
        "route_width": 8,
        "marker_size": 16,
        "canvas": list(CANVAS),
        "palette": {name: list(value) for name, value in PALETTE.items()},
    }


def _draw_operation(
    draw: ImageDraw.ImageDraw,
    center: list[int],
    operation: int,
    radius: int,
    color: tuple[int, int, int],
) -> None:
    x, y = center
    if operation == 1:
        draw.polygon(
            [(x - radius, y - radius), (x - radius, y + radius), (x + radius, y)],
            fill=color,
        )
    elif operation == -1:
        draw.polygon(
            [(x + radius, y - radius), (x + radius, y + radius), (x - radius, y)],
            fill=color,
        )
    else:
        draw.rectangle(
            (x - radius, y - radius, x + radius, y + radius),
            outline=color,
            width=4,
        )


def _draw_state_badge(
    draw: ImageDraw.ImageDraw,
    center: list[int],
    state: int,
    radius: int,
    colors: dict[str, tuple[int, int, int]],
    font: ImageFont.ImageFont,
) -> None:
    x, y = center
    state_color = colors["state_a" if state == 0 else "state_b" if state == 1 else "state_c"]
    draw.ellipse(
        (x - radius, y - radius, x + radius, y + radius),
        fill=state_color,
        outline=colors["route"],
        width=4,
    )
    label = STATE_LABELS[state]
    bbox = draw.textbbox((0, 0), label, font=font)
    tw, th = bbox[2] - bbox[0], bbox[3] - bbox[1]
    draw.text((x - tw / 2, y - th / 2 - 1), label, fill=colors["background"], font=font)


def render(scene: dict[str, Any]) -> Image.Image:
    width, height = (int(value) for value in scene["canvas"])
    colors = _colors(scene)
    image = Image.new("RGB", (width, height), colors["background"])
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    for distractor in scene["distractors"]:
        x, y, size = (int(distractor[key]) for key in ("x", "y", "size"))
        if int(distractor["kind"]) == 0:
            draw.ellipse((x - size, y - size, x + size, y + size), outline=colors["neutral"], width=3)
        else:
            draw.rectangle((x - size, y - size, x + size, y + size), outline=colors["neutral"], width=3)

    points = [tuple(int(value) for value in point) for point in scene["route_points"]]
    draw.line(points, fill=colors["route"], width=int(scene["route_width"]), joint="curve")

    marker_radius = int(scene["marker_size"])
    for index, (point, operation) in enumerate(zip(points[1:-1], scene["operations"], strict=True), start=1):
        center = [int(point[0]), int(point[1])]
        _draw_operation(draw, center, int(operation), marker_radius, colors["operation"])
        label = str(index)
        bbox = draw.textbbox((0, 0), label, font=font)
        tw = bbox[2] - bbox[0]
        draw.text((center[0] - tw / 2, center[1] - marker_radius - 20), label, fill=colors["text"], font=font)

    _draw_state_badge(
        draw,
        [int(value) for value in points[0]],
        int(scene["initial_state"]),
        marker_radius + 8,
        colors,
        font,
    )
    end_x, end_y = points[-1]
    draw.ellipse((end_x - 11, end_y - 11, end_x + 11, end_y + 11), outline=colors["neutral"], width=3)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    edge, gap = _best_edge(int(scene["initial_state"]), [int(value) for value in scene["operations"]])
    if gap <= 0:
        raise ValueError("directed transition leader is not unique")
    return edge


def margin(scene: dict[str, Any]) -> float:
    _, gap = _best_edge(int(scene["initial_state"]), [int(value) for value in scene["operations"]])
    return round(float(gap), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    if len(scene.get("operations", [])) != STATION_COUNT:
        return True
    counts = _edge_counts(int(scene["initial_state"]), [int(value) for value in scene["operations"]])
    if sum(counts.values()) == 0:
        return True
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict]]:
    # Every answer-bearing field is explicitly visible. No hidden alias is claimed.
    return []
