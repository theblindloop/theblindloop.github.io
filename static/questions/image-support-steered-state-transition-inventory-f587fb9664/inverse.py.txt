"""Independent image-only inverse for a connected directed-transition route."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


_COLORS = {
    "background": (248, 250, 253),
    "route": (34, 58, 82),
    "operation": (213, 91, 57),
    "state_a": (193, 68, 67),
    "state_b": (39, 145, 91),
    "state_c": (61, 103, 190),
}
_STATE_LABELS = ("A", "B", "C")
_EDGE_LABELS = ("AB", "BC", "CA", "AC", "CB", "BA")
_STATION_COUNT = 10


def _masks(image: Image.Image) -> dict[str, np.ndarray]:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    return {
        name: np.all(array == np.asarray(color, dtype=np.uint8), axis=2)
        for name, color in _COLORS.items()
    }


def _components(mask: np.ndarray, minimum: int = 1) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue: deque[tuple[int, int]] = deque([(y, x)])
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dy or dx):
                        continue
                    ny, nx = cy + dy, cx + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        if len(points) >= minimum:
            output.append(np.asarray(points, dtype=np.int32))
    return output


def _centroid(component: np.ndarray) -> tuple[float, float]:
    y, x = component.mean(axis=0)
    return float(y), float(x)


def _bbox_stats(component: np.ndarray) -> tuple[float, float, float, float]:
    ys, xs = component[:, 0], component[:, 1]
    height = float(ys.max() - ys.min() + 1)
    width = float(xs.max() - xs.min() + 1)
    fill = len(component) / (width * height)
    offset = float(xs.mean() - (xs.min() + xs.max()) / 2.0)
    return fill, width, height, offset


def _operation(component: np.ndarray) -> int:
    """Classify right triangle, left triangle, or hollow square."""

    fill, width, height, offset = _bbox_stats(component)
    if not (29.0 <= width <= 37.0 and 29.0 <= height <= 37.0):
        raise ValueError("station glyph has invalid size")
    if fill < 0.30 or fill > 0.68:
        raise ValueError("station glyph has an ambiguous silhouette")
    if abs(offset) <= 2.3:
        return 0
    if offset < -2.3:
        return 1
    return -1


def _edge_counts(initial_state: int, operations: list[int]) -> dict[str, int]:
    state = int(initial_state) % 3
    states = [state]
    for operation in operations:
        state = (state + int(operation)) % 3
        states.append(state)
    counts = {edge: 0 for edge in _EDGE_LABELS}
    for source, target in zip(states, states[1:]):
        if source != target:
            counts[_STATE_LABELS[source] + _STATE_LABELS[target]] += 1
    return counts


def _best_edge(counts: dict[str, int]) -> str:
    ordered = sorted(_EDGE_LABELS, key=lambda edge: (-counts[edge], edge))
    if counts[ordered[0]] <= counts[ordered[1]]:
        raise ValueError("directed-transition leader is tied")
    return ordered[0]


def decision_from_image(image: Image.Image) -> str:
    """Recover the badge, every station operation, and the connected route from pixels."""

    masks = _masks(image)
    state_components: list[tuple[int, np.ndarray]] = []
    for state, name in enumerate(("state_a", "state_b", "state_c")):
        components = _components(masks[name], minimum=450)
        if len(components) > 1:
            raise ValueError("multiple components use one state color")
        if components:
            component = components[0]
            fill, width, height, _ = _bbox_stats(component)
            if not (36.0 <= width <= 58.0 and 36.0 <= height <= 58.0 and fill > 0.55):
                raise ValueError("start badge is ambiguous")
            state_components.append((state, component))
    if len(state_components) != 1:
        raise ValueError("expected exactly one colored start badge")
    initial_state = state_components[0][0]

    operation_components = _components(masks["operation"], minimum=100)
    if len(operation_components) != _STATION_COUNT:
        raise ValueError("expected ten visible station glyphs")
    ordered: list[tuple[float, np.ndarray]] = []
    for component in operation_components:
        _, center_x = _centroid(component)
        ordered.append((center_x, component))
    ordered.sort(key=lambda item: item[0])
    if any(right[0] - left[0] < 20.0 for left, right in zip(ordered, ordered[1:])):
        raise ValueError("station order is unresolved")
    operations = [_operation(component) for _, component in ordered]

    traversable = masks["route"] | masks["operation"] | masks["state_a"] | masks["state_b"] | masks["state_c"]
    route_components = _components(traversable, minimum=150)
    if len(route_components) != 1:
        raise ValueError("route is missing or disconnected")

    return _best_edge(_edge_counts(initial_state, operations))
