"""Independent PNG-only inverse arm.

The solver detects the five visibly colored grid components, derives every cell
occupancy from central pixels, performs the public transpose/XOR rule, and reads
the visible left-to-right result label.  It imports no world implementation.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

N = 4
_SOURCE = np.array((43, 116, 145), dtype=np.int16)
_CANDIDATE = np.array((190, 105, 45), dtype=np.int16)


class OracleAbstention(ValueError):
    pass


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[tuple[int, int, int, int, int]] = []
    for y, x in zip(*np.nonzero(mask), strict=True):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            cy, cx = queue.popleft()
            xs.append(cx)
            ys.append(cy)
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = cy + dy, cx + dx
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) > 300:
            output.append((min(xs), min(ys), max(xs), max(ys), len(xs)))
    return output


def _frame_components(arr: np.ndarray, color: np.ndarray) -> list[tuple[int, int, int, int]]:
    distance = np.sqrt(((arr.astype(np.int32) - color.astype(np.int32)) ** 2).sum(axis=2))
    components = _components(distance < 52.0)
    frames = []
    for x0, y0, x1, y1, _ in components:
        width, height = x1 - x0 + 1, y1 - y0 + 1
        if 140 <= width <= 240 and 140 <= height <= 240 and abs(width - height) <= 8:
            frames.append((x0, y0, x1, y1))
    return sorted(frames, key=lambda box: box[0])


def _read_grid(arr: np.ndarray, box: tuple[int, int, int, int]) -> set[tuple[int, int]]:
    x0, y0, x1, y1 = box
    width = x1 - x0 + 1
    height = y1 - y0 + 1
    occupied: set[tuple[int, int]] = set()
    for row in range(N):
        for column in range(N):
            cx = x0 + (column + 0.5) * width / N
            cy = y0 + (row + 0.5) * height / N
            radius = max(3, int(min(width, height) / 28))
            patch = arr[
                max(0, int(cy) - radius) : int(cy) + radius + 1,
                max(0, int(cx) - radius) : int(cx) + radius + 1,
            ].astype(np.float32)
            if patch.size == 0:
                raise OracleAbstention("empty cell sample")
            darkness = float(patch.mean(axis=2).mean())
            if darkness < 85.0:
                occupied.add((row, column))
            elif darkness < 185.0:
                raise OracleAbstention("uncertain cell occupancy")
    return occupied


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    source_boxes = _frame_components(arr, _SOURCE)
    candidate_boxes = _frame_components(arr, _CANDIDATE)
    if len(source_boxes) != 2 or len(candidate_boxes) != 3:
        raise OracleAbstention(
            f"expected two source and three result grids, found {len(source_boxes)} and {len(candidate_boxes)}"
        )
    left, right = (_read_grid(arr, box) for box in source_boxes)
    target = left.symmetric_difference({(column, row) for row, column in right})
    candidates = [_read_grid(arr, box) for box in candidate_boxes]
    distances = [len(target.symmetric_difference(value)) for value in candidates]
    best = min(distances)
    winners = [index for index, value in enumerate(distances) if value == best]
    ordered = sorted(distances)
    if best != 0 or len(winners) != 1 or ordered[1] - ordered[0] < 2:
        raise OracleAbstention(f"ambiguous or inexact match: {distances}")
    return ("A", "B", "C")[winners[0]]
