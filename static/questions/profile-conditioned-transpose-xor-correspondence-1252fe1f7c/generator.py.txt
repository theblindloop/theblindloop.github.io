"""Deterministic latent sampler and Pillow renderer."""

from __future__ import annotations

import copy
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

N = 4
WIDTH, HEIGHT = 960, 680
SCALE = 3
BACKGROUND = (250, 248, 242)
INK = (25, 29, 35)
SOURCE_FRAME = (43, 116, 145)
CANDIDATE_FRAME = (190, 105, 45)
EMPTY = (255, 255, 252)
LABEL = (55, 58, 64)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size * SCALE)
    except OSError:
        return ImageFont.load_default()


def _cell_set(cells: Iterable[Iterable[int]]) -> set[tuple[int, int]]:
    return {(int(cell[0]), int(cell[1])) for cell in cells}


def _transpose(cells: set[tuple[int, int]]) -> set[tuple[int, int]]:
    return {(column, row) for row, column in cells}


def _target(scene: dict) -> set[tuple[int, int]]:
    left = _cell_set(scene["left_cells"])
    right = _cell_set(scene["right_cells"])
    return left.symmetric_difference(_transpose(right))


def _swap_cells(
    base: set[tuple[int, int]], rng: random.Random, swaps: int
) -> set[tuple[int, int]]:
    filled = list(base)
    empty = [(r, c) for r in range(N) for c in range(N) if (r, c) not in base]
    removed = rng.sample(filled, swaps)
    added = rng.sample(empty, swaps)
    return (base - set(removed)) | set(added)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    all_cells = [(r, c) for r in range(N) for c in range(N)]
    for _ in range(1000):
        left = set(rng.sample(all_cells, rng.randint(5, 10)))
        right = set(rng.sample(all_cells, rng.randint(5, 10)))
        target = left.symmetric_difference(_transpose(right))
        if 5 <= len(target) <= 11:
            break
    else:
        raise RuntimeError("could not sample a balanced overlay")

    near = _swap_cells(target, rng, 1)
    far = _swap_cells(target, rng, 2)
    while far == near:
        far = _swap_cells(target, rng, 2)

    answer_index = int(seed) % 3
    distractors = [near, far]
    rng.shuffle(distractors)
    candidates: list[set[tuple[int, int]]] = []
    d = 0
    for index in range(3):
        if index == answer_index:
            candidates.append(target)
        else:
            candidates.append(distractors[d])
            d += 1

    # Coordinate ordering is deliberately semantically irrelevant and supplies
    # a real tested latent representation alias.
    def shuffled(values: set[tuple[int, int]]) -> list[list[int]]:
        ordered = list(values)
        rng.shuffle(ordered)
        return [[r, c] for r, c in ordered]

    return {
        "left_cells": shuffled(left),
        "right_cells": shuffled(right),
        "candidate_cells": [shuffled(value) for value in candidates],
    }


def analytic_gold(scene: dict) -> str:
    target = _target(scene)
    matches = [
        index
        for index, cells in enumerate(scene["candidate_cells"])
        if _cell_set(cells) == target
    ]
    if len(matches) != 1:
        raise ValueError("scene has no unique exact candidate")
    return ("A", "B", "C")[matches[0]]


def margin(scene: dict) -> float:
    target = _target(scene)
    distances = sorted(
        len(target.symmetric_difference(_cell_set(cells)))
        for cells in scene["candidate_cells"]
    )
    if not distances or distances[0] != 0:
        return 0.0
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 2.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["left_cells"] = list(reversed(twin["left_cells"]))
    twin["right_cells"] = list(reversed(twin["right_cells"]))
    twin["candidate_cells"] = [list(reversed(cells)) for cells in twin["candidate_cells"]]
    return [("reverse_coordinate_serialization", twin)]


def _draw_centered(
    draw: ImageDraw.ImageDraw,
    xy: tuple[int, int],
    text: str,
    font: ImageFont.ImageFont,
    fill: tuple[int, int, int],
) -> None:
    x, y = xy
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((x - (box[2] - box[0]) / 2, y - (box[3] - box[1]) / 2), text, font=font, fill=fill)


def _draw_card(
    draw: ImageDraw.ImageDraw,
    cells: Iterable[Iterable[int]],
    x: int,
    y: int,
    size: int,
    frame: tuple[int, int, int],
) -> None:
    x, y, size = x * SCALE, y * SCALE, size * SCALE
    cell = size // N
    occupied = _cell_set(cells)
    draw.rectangle((x, y, x + size, y + size), fill=EMPTY, outline=frame, width=7 * SCALE)
    for row in range(N):
        for column in range(N):
            if (row, column) in occupied:
                pad = 6 * SCALE
                x0, y0 = x + column * cell + pad, y + row * cell + pad
                x1, y1 = x + (column + 1) * cell - pad, y + (row + 1) * cell - pad
                draw.rectangle((x0, y0, x1, y1), fill=INK)
    for index in range(1, N):
        p = index * cell
        draw.line((x + p, y, x + p, y + size), fill=frame, width=3 * SCALE)
        draw.line((x, y + p, x + size, y + p), fill=frame, width=3 * SCALE)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(22, bold=True)
    label_font = _font(19, bold=True)
    small_font = _font(16, bold=False)

    _draw_centered(draw, (WIDTH * SCALE // 2, 31 * SCALE), "TRANSPOSE RIGHT, THEN XOR WITH LEFT", title_font, LABEL)
    _draw_centered(draw, (280 * SCALE, 76 * SCALE), "LEFT", label_font, SOURCE_FRAME)
    _draw_centered(draw, (680 * SCALE, 76 * SCALE), "RIGHT", label_font, SOURCE_FRAME)
    _draw_card(draw, scene["left_cells"], 180, 95, 200, SOURCE_FRAME)
    _draw_card(draw, scene["right_cells"], 580, 95, 200, SOURCE_FRAME)

    _draw_centered(draw, (WIDTH * SCALE // 2, 338 * SCALE), "RESULT CARDS", small_font, LABEL)
    for index, x in enumerate((90, 390, 690)):
        _draw_card(draw, scene["candidate_cells"][index], x, 375, 180, CANDIDATE_FRAME)
        _draw_centered(draw, ((x + 90) * SCALE, 590 * SCALE), ("A", "B", "C")[index], label_font, LABEL)

    # The diagonal icon visibly disambiguates which transpose is intended.
    draw.line((819 * SCALE, 107 * SCALE, 849 * SCALE, 137 * SCALE), fill=SOURCE_FRAME, width=4 * SCALE)
    draw.polygon(
        [(849 * SCALE, 137 * SCALE), (837 * SCALE, 133 * SCALE), (845 * SCALE, 125 * SCALE)],
        fill=SOURCE_FRAME,
    )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
