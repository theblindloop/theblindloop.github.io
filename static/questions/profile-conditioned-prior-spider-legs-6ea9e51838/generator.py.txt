"""Prior-conflict spider leg-count world renderer.

Latent state z = {n_legs, body_x, body_y, body_rx, body_ry, leg_len, leg_w,
angles_deg}.

A single dark-brown elliptical spider body is drawn, and `n_legs` black leg
segments radiate outward from the body edge at spread-out angles. Each leg is a
separate stroke, so on a light field every leg is its own black connected
component. The closed-form gold is the scene's leg count:

    leg_count = n_legs   (a scene parameter, not a pixel measurement)

The canonical prior -- that a spider has eight legs -- is recorded only in the
question narrative and is never used as ground truth. The visible truth is
whatever number of legs is actually drawn.

Continuous verifiability margin (px-equivalent, reported in degrees): the
smallest angular separation between two adjacent legs. When it stays above a
finite-resolution floor the legs remain resolvable as distinct black strokes.

Latent-alias mode: not_applicable. The leg count is pinned by n_legs, which is
rendered as `n_legs` separate black legs; different n_legs always produce a
different (and differently-counted) raster, so there is no render-invariant
latent alias and sample_scene returns no pixel-identical twins.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS_W = 384
CANVAS_H = 384
CANVAS_BG = (245, 245, 240)
BODY_COLOR = (96, 64, 48)    # dark brown, distinct from the black legs
LEG_COLOR = (28, 28, 28)     # near-black legs

N_LEGS_RANGE = (5, 9)        # inclusive leg count; canonical prior is 8
BODY_RX_RANGE = (26.0, 34.0)
BODY_RY_RANGE = (22.0, 30.0)
LEG_LEN_RANGE = (70.0, 95.0)
LEG_W = 3
JITTER_DEG = 5.0             # max deterministic per-leg angle jitter (deg)
CENTER_LO = 158.0
CENTER_HI = 226.0

RENDERER_VERSION = "prior_spider_legs-0.1.0"


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(seed)
    n_legs = rng.randint(*N_LEGS_RANGE)
    body_x = rng.uniform(CENTER_LO, CENTER_HI)
    body_y = rng.uniform(CENTER_LO, CENTER_HI)
    body_rx = rng.uniform(*BODY_RX_RANGE)
    body_ry = rng.uniform(*BODY_RY_RANGE)
    leg_len = rng.uniform(*LEG_LEN_RANGE)
    base = 360.0 / n_legs
    # Deterministically jitter so adjacent-leg spacing never falls below a safe
    # floor: base >= 40 deg (n<=9), jitter <= 5 deg keeps min gap >= 30 deg.
    angles = []
    for k in range(n_legs):
        jitter = rng.uniform(-JITTER_DEG, JITTER_DEG)
        angles.append(base * k + jitter)
    return {
        "n_legs": n_legs,
        "body_x": round(body_x, 3),
        "body_y": round(body_y, 3),
        "body_rx": round(body_rx, 3),
        "body_ry": round(body_ry, 3),
        "leg_len": round(leg_len, 3),
        "leg_w": LEG_W,
        "angles_deg": [round(a, 6) for a in angles],
    }


def _leg_geometry(scene: dict[str, Any]) -> list[tuple[float, float, float, float]]:
    """Return (x0, y0, x1, y1) for each leg on the final (unscaled) canvas."""
    cx = float(scene["body_x"])
    cy = float(scene["body_y"])
    rx = float(scene["body_rx"])
    leg_len = float(scene["leg_len"])
    lines = []
    for a_deg in scene["angles_deg"]:
        rad = math.radians(a_deg)
        dirx, diry = math.cos(rad), math.sin(rad)
        # Start slightly inside the body edge so the leg is visibly attached.
        start_r = 0.55 * rx
        end_r = rx + leg_len
        x0 = cx + dirx * start_r
        y0 = cy + diry * start_r
        x1 = cx + dirx * end_r
        y1 = cy + diry * end_r
        lines.append((x0, y0, x1, y1))
    return lines


def render(scene: dict[str, Any]) -> Image.Image:
    n = int(scene["n_legs"])
    if n < 1:
        raise ValueError("a spider needs at least one leg")
    scale = 2
    big_w = CANVAS_W * scale
    big_h = CANVAS_H * scale
    big = Image.new("RGB", (big_w, big_h), CANVAS_BG)
    draw = ImageDraw.Draw(big)

    cx = float(scene["body_x"])
    cy = float(scene["body_y"])
    rx = float(scene["body_rx"])
    ry = float(scene["body_ry"])
    # Draw legs first so the body covers their inner attachment stubs; the
    # outward portion of each leg stays visible as a separate black stroke.
    for (x0, y0, x1, y1) in _leg_geometry(scene):
        draw.line(
            [x0 * scale, y0 * scale, x1 * scale, y1 * scale],
            fill=LEG_COLOR,
            width=int(scene["leg_w"]) * scale,
        )
    draw.ellipse(
        [
            (cx - rx) * scale,
            (cy - ry) * scale,
            (cx + rx) * scale,
            (cy + ry) * scale,
        ],
        fill=BODY_COLOR,
    )
    return big.resize((CANVAS_W, CANVAS_H), Image.LANCZOS)


def _min_angular_gap(scene: dict[str, Any]) -> float:
    angles = [float(a) for a in scene["angles_deg"]]
    angles.sort()
    n = len(angles)
    if n < 2:
        return 360.0
    gaps = []
    for i in range(n):
        gaps.append((angles[(i + 1) % n] - angles[i]) % 360.0)
    return float(min(gaps))


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(int(scene["n_legs"]))


def margin(scene: dict[str, Any]) -> float:
    # The decision-relevant separation is the smallest angular gap between two
    # adjacent legs; below the finite-resolution floor the legs become ambiguous.
    return _min_angular_gap(scene)


def is_quarantined(scene: dict[str, Any]) -> bool:
    # Two adjacent legs are pixel-ambiguous when their angular separation drops
    # below the floor at which a 3px stroke pair can no longer be resolved.
    return margin(scene) < 22.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # not_applicable: the gold is pinned by the rendered n_legs; there is no
    # declared pixel-identical alias to exercise.
    return []
