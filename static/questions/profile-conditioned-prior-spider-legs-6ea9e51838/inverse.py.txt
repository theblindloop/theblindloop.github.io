"""Independent pixel-only inverse oracle for the spider leg-count world.

Receives only the final PNG and re-counts the spider's legs directly from the
raster. It imports nothing from the renderer, prompts, generator, verifier, or
analytic-gold code, and accepts no latent scene data.

Method: each leg is a near-black segment radiating outward from a dark-brown
spider body over a light field. Step 1 locates the body as the central brown
region and takes its centroid as the spider centre. Step 2 takes every
near-black (leg) pixel and its angle around that centre. Because every leg is a
straight radial stroke, all of a leg's pixels share the same angle, while
different legs sit at clearly separated angles. The number of angular clusters
of leg pixels (large circular gaps between consecutive angles) therefore equals
the number of legs.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

_LEG = 120        # near-black threshold for leg ink
_BROWN_MIN = 40   # brown body channel bounds
_BROWN_MAX = 130
_GAP_DEG = 13.0   # circular gap (deg) separating two legs' angular clusters


def _brown_mask(arr: np.ndarray) -> np.ndarray:
    r = arr[..., 0].astype(np.int16)
    g = arr[..., 1].astype(np.int16)
    b = arr[..., 2].astype(np.int16)
    return (
        (r >= _BROWN_MIN)
        & (r <= _BROWN_MAX)
        & (g >= 30)
        & (g <= 90)
        & (b >= 20)
        & (b <= 80)
        & (r > g)
        & (g > b)
    )


def _leg_mask(arr: np.ndarray) -> np.ndarray:
    r = arr[..., 0].astype(np.int16)
    g = arr[..., 1].astype(np.int16)
    b = arr[..., 2].astype(np.int16)
    span = np.maximum(np.maximum(r, g), b) - np.minimum(np.minimum(r, g), b)
    return (r < _LEG) & (g < _LEG) & (b < _LEG) & (span <= 10)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.int16)

    body = _brown_mask(arr)
    if body.sum() == 0:
        return "0"
    ys, xs = np.nonzero(body)
    cy = float(ys.mean())
    cx = float(xs.mean())

    legs = _leg_mask(arr)
    if legs.sum() == 0:
        return "0"
    lys, lxs = np.nonzero(legs)
    dx = lxs.astype(np.float64) - cx
    dy = lys.astype(np.float64) - cy
    angles = np.degrees(np.arctan2(dy, dx)) % 360.0
    angles.sort()

    diffs = np.diff(angles)
    wrap = (angles[0] + 360.0) - angles[-1]
    all_diffs = np.append(diffs, wrap)
    count = int(np.count_nonzero(all_diffs > _GAP_DEG))
    return str(count)
