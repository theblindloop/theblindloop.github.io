"""Deterministic renderer and analytic arm for Lamina Centroid Band."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw, ImageFont


CANVAS = (512, 512)
BACKGROUND_RGB = (246, 244, 238)
PANEL_RGB = (255, 255, 252)
FILL_PALETTE = (
    (30, 122, 171),
    (43, 139, 108),
    (111, 91, 176),
    (195, 78, 92),
)
GUIDE_PALETTE = (
    (224, 151, 28),
    (218, 119, 44),
    (197, 143, 33),
)
FRAME_PALETTE = ((48, 56, 64), (55, 61, 72), (62, 58, 70))
QUARANTINE_PX = 12.0


def _font(size: int, *, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _vertices(scene: dict) -> list[tuple[float, float]]:
    cx = float(scene["center_x"])
    top = float(scene["top_y"])
    step = float(scene["knot_step"])
    widths = [float(value) for value in scene["half_widths"]]
    left = [(cx - half, top + index * step) for index, half in enumerate(widths)]
    right = [(cx + half, top + index * step) for index, half in reversed(list(enumerate(widths)))]
    return left + right


def _centroid_y(scene: dict) -> float:
    vertices = _vertices(scene)
    cross_sum = 0.0
    moment_sum = 0.0
    for index, (x0, y0) in enumerate(vertices):
        x1, y1 = vertices[(index + 1) % len(vertices)]
        cross = x0 * y1 - x1 * y0
        cross_sum += cross
        moment_sum += (y0 + y1) * cross
    if abs(cross_sum) < 1e-9:
        raise ValueError("degenerate lamina")
    return moment_sum / (3.0 * cross_sum)


def sample_scene(seed: int) -> dict:
    """Sample one balanced, safely separated centroid-band scene."""

    rng = random.Random(int(seed))
    base = [28, 62, 88, 101, 104, 96, 78, 54, 26]
    perturb = [rng.randint(-4, 4) for _ in base]
    half_widths = [max(22, min(106, value + delta)) for value, delta in zip(base, perturb)]
    # Small paired compensation varies shape without making total ink a class cue.
    for left, right in ((1, 7), (2, 6), (3, 5)):
        shift = rng.randint(-5, 5)
        half_widths[left] += shift
        half_widths[right] -= shift

    fill_index = rng.randrange(len(FILL_PALETTE))
    guide_index = rng.randrange(len(GUIDE_PALETTE))
    if guide_index == fill_index % len(GUIDE_PALETTE):
        guide_index = (guide_index + 1) % len(GUIDE_PALETTE)
    scene = {
        "width": CANVAS[0],
        "height": CANVAS[1],
        "center_x": 256 + rng.randint(-3, 3),
        "top_y": 170 + rng.randint(-4, 4),
        "knot_step": 21,
        "half_widths": half_widths,
        "guide_ys": [210, 300],
        "fill_rgb": list(FILL_PALETTE[fill_index]),
        "guide_rgb": list(GUIDE_PALETTE[guide_index]),
        "frame_rgb": list(FRAME_PALETTE[rng.randrange(len(FRAME_PALETTE))]),
    }
    center_y = _centroid_y(scene)
    answer_index = int(seed) % 3
    near_offset = (14, 20, 28, 36)[(int(seed) // 3) % 4]
    far_gap = 64 + 2 * rng.randrange(8)
    if answer_index == 0:  # UPPER: center is above both rules.
        first = round(center_y + near_offset)
        guides = [first, first + far_gap]
    elif answer_index == 1:  # MIDDLE: center lies between the rules.
        upper = 18 + 2 * rng.randrange(8)
        lower = 18 + 2 * rng.randrange(8)
        guides = [round(center_y - upper), round(center_y + lower)]
    else:  # LOWER: center is below both rules.
        second = round(center_y - near_offset)
        guides = [second - far_gap, second]
    scene["guide_ys"] = guides
    return scene


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width, height), BACKGROUND_RGB)
    draw = ImageDraw.Draw(image)
    frame = tuple(scene["frame_rgb"])
    guide = tuple(scene["guide_rgb"])
    fill = tuple(scene["fill_rgb"])

    draw.rounded_rectangle((54, 48, 458, 458), radius=20, fill=PANEL_RGB, outline=frame, width=3)
    draw.text((76, 65), "CENTER-OF-AREA READOUT", font=_font(18, bold=True), fill=frame)
    draw.rectangle((82, 104, 430, 414), outline=frame, width=3)

    guide_ys = [int(value) for value in scene["guide_ys"]]
    for y in guide_ys:
        draw.line((84, y, 428, y), fill=guide, width=5)

    # Band labels are dark neutral ink and are deliberately irrelevant to the oracle.
    labels = (
        ("UPPER", (104 + guide_ys[0]) // 2),
        ("MIDDLE", (guide_ys[0] + guide_ys[1]) // 2),
        ("LOWER", (guide_ys[1] + 414) // 2),
    )
    label_font = _font(13, bold=True)
    for text_value, center_y in labels:
        draw.rounded_rectangle((88, center_y - 11, 145, center_y + 11), radius=5, fill=PANEL_RGB)
        draw.text((93, center_y - 8), text_value, font=label_font, fill=frame)

    vertices = [(round(x), round(y)) for x, y in _vertices(scene)]
    draw.polygon(vertices, fill=fill)

    draw.text((83, 430), "uniform lamina", font=_font(14), fill=frame)
    draw.text((337, 430), "gold guides", font=_font(14), fill=guide)
    return image


def analytic_gold(scene: dict) -> str:
    center_y = _centroid_y(scene)
    upper, lower = [float(value) for value in scene["guide_ys"]]
    if center_y < upper:
        return "UPPER"
    if center_y < lower:
        return "MIDDLE"
    return "LOWER"


def margin(scene: dict) -> float:
    center_y = _centroid_y(scene)
    return min(abs(center_y - float(value)) for value in scene["guide_ys"])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
