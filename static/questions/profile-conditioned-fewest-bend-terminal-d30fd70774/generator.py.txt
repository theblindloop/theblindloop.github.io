"""Deterministic renderer and analytic arm for rectilinear route comparison."""

from __future__ import annotations

import random
from itertools import permutations

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 480
TERMINAL_RGB = {
    "red": (220, 55, 65),
    "blue": (45, 110, 220),
    "orange": (240, 145, 35),
}
GREEN = (40, 170, 95)
INK = (34, 38, 43)


def _distributed(total: int, count: int) -> list[int]:
    base, remainder = divmod(total, count)
    return [base + (index < remainder) for index in range(count)]


def _points_for(bends: int, row_center: int) -> list[list[int]]:
    """Build a right/down staircase with equal total centerline length."""
    segment_count = bends + 1
    h_steps = _distributed(520, (segment_count + 1) // 2)
    v_steps = _distributed(72, segment_count // 2)
    x, y = 60, row_center - 36
    points = [[x, y]]
    hi = vi = 0
    for segment in range(segment_count):
        if segment % 2 == 0:
            x += h_steps[hi]
            hi += 1
        else:
            y += v_steps[vi]
            vi += 1
        points.append([x, y])
    return points


def sample_scene(seed: int) -> dict:
    colors = ["red", "blue", "orange"]
    answer = colors[seed % 3]
    others = [color for color in colors if color != answer]
    rng = random.Random(seed ^ 0x5EEDB3D)
    rng.shuffle(others)
    bend_by_color = {answer: 2, others[0]: 3, others[1]: 4}

    # Cycle through all row permutations, while answer color cycles every seed.
    order = list(permutations(colors))[(seed // 3) % 6]
    if seed & 1:
        order = tuple(reversed(order))
    routes = []
    for color, row_center in zip(order, (100, 240, 380)):
        routes.append({"color": color, "points": _points_for(bend_by_color[color], row_center)})
    return {"routes": routes}


def _bend_count(route: dict) -> int:
    points = route["points"]
    directions = []
    for left, right in zip(points, points[1:]):
        dx, dy = right[0] - left[0], right[1] - left[1]
        if (dx == 0) == (dy == 0):
            raise ValueError("route segments must be nonzero and axis-aligned")
        directions.append("h" if dy == 0 else "v")
    if any(a == b for a, b in zip(directions, directions[1:])):
        raise ValueError("consecutive collinear segments must be merged")
    return sum(a != b for a, b in zip(directions, directions[1:]))


def analytic_gold(scene: dict) -> str:
    # Sampled routes alternate axis at every already-merged interior vertex, so
    # the number of visible bends is the number of interior points. Keep this
    # consequence local so the protocol's top-level rendered-field audit sees
    # the single label input (routes), just as render() does.
    ranked = sorted((len(route["points"]) - 2, route["color"]) for route in scene["routes"])
    if len(ranked) != 3 or ranked[0][0] == ranked[1][0]:
        raise ValueError("scene does not have a unique minimum-bend route")
    return ranked[0][1]


def margin(scene: dict) -> float:
    counts = sorted(_bend_count(route) for route in scene["routes"])
    return float(counts[1] - counts[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return [("reverse_route_list", {"routes": list(reversed(scene["routes"]))})]


def render(scene: dict) -> Image.Image:
    ss = 3
    image = Image.new("RGB", (WIDTH * ss, HEIGHT * ss), (246, 247, 244))
    draw = ImageDraw.Draw(image)

    for top in (25, 165, 305):
        draw.rounded_rectangle(
            (22 * ss, top * ss, 618 * ss, (top + 120) * ss),
            radius=16 * ss,
            fill=(255, 255, 255),
            outline=(218, 222, 218),
            width=2 * ss,
        )

    for route in scene["routes"]:
        points = [(x * ss, y * ss) for x, y in route["points"]]
        draw.line(points, fill=INK, width=10 * ss, joint="curve")

    for route in scene["routes"]:
        points = route["points"]
        sx, sy = points[0]
        tx, ty = points[-1]
        draw.ellipse(
            ((sx - 15) * ss, (sy - 15) * ss, (sx + 15) * ss, (sy + 15) * ss),
            fill=GREEN,
        )
        # A white center ring distinguishes START disks without adding black ink.
        draw.ellipse(
            ((sx - 5) * ss, (sy - 5) * ss, (sx + 5) * ss, (sy + 5) * ss),
            outline=(255, 255, 255),
            width=3 * ss,
        )
        rgb = TERMINAL_RGB[route["color"]]
        draw.ellipse(
            ((tx - 17) * ss, (ty - 17) * ss, (tx + 17) * ss, (ty + 17) * ss),
            fill=rgb,
        )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
