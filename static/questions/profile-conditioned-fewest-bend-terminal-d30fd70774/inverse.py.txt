"""Independent image-only inverse arm for rectilinear route bends."""

from __future__ import annotations

import numpy as np
from PIL import Image

PALETTE = {
    "red": (220, 55, 65),
    "blue": (45, 110, 220),
    "orange": (240, 145, 35),
}
GREEN = (40, 170, 95)


def _clusters(active: np.ndarray) -> int:
    padded = np.pad(active.astype(np.int8), (1, 1))
    return int(np.count_nonzero((padded[1:-1] == 1) & (padded[:-2] == 0)))


def _terminal_centroids(arr: np.ndarray) -> dict[str, tuple[float, float]]:
    result = {}
    rgb = arr.astype(np.int32)
    for name, target in PALETTE.items():
        distance2 = ((rgb - np.asarray(target, dtype=np.int32)) ** 2).sum(axis=2)
        ys, xs = np.nonzero(distance2 < 55**2)
        if len(xs) < 250:
            raise ValueError(f"missing {name} terminal")
        result[name] = (float(xs.mean()), float(ys.mean()))
    return result


def _start_y_centroids(arr: np.ndarray) -> list[float]:
    rgb = arr.astype(np.int32)
    distance2 = ((rgb - np.asarray(GREEN, dtype=np.int32)) ** 2).sum(axis=2)
    mask = distance2 < 55**2
    active = mask.sum(axis=1) > 5
    padded = np.pad(active.astype(np.int8), (1, 1))
    starts = np.flatnonzero((padded[1:-1] == 1) & (padded[:-2] == 0))
    stops = np.flatnonzero((padded[1:-1] == 1) & (padded[2:] == 0))
    if len(starts) != 3 or len(stops) != 3:
        raise ValueError("expected three green START disks")
    values = []
    for lo, hi in zip(starts, stops):
        weights = mask[lo : hi + 1].sum(axis=1)
        values.append(float(np.average(np.arange(lo, hi + 1), weights=weights)))
    return values


def _bend_count(black: np.ndarray) -> int:
    # A long projection identifies each horizontal/vertical rectilinear run.
    horizontal = black.sum(axis=1) > max(45, black.shape[1] // 14)
    vertical = black.sum(axis=0) > max(15, black.shape[0] // 10)
    h_runs, v_runs = _clusters(horizontal), _clusters(vertical)
    if h_runs < 1 or h_runs > 4 or v_runs < 1 or v_runs > 3:
        raise ValueError(f"unresolved route geometry: {h_runs} horizontal/{v_runs} vertical")
    return h_runs + v_runs - 1


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    if arr.ndim != 3 or min(arr.shape[:2]) < 100:
        raise ValueError("image is too small")
    terminals = _terminal_centroids(arr)
    ordered = sorted((cy, color) for color, (_, cy) in terminals.items())
    start_ys = _start_y_centroids(arr)
    centers = [(start_y + terminal[0]) / 2 for start_y, terminal in zip(start_ys, ordered)]
    cuts = [0]
    cuts.extend(int(round((centers[i] + centers[i + 1]) / 2)) for i in range(2))
    cuts.append(arr.shape[0])
    black = arr.max(axis=2) < 100

    counts = {}
    for index, (_, color) in enumerate(ordered):
        lo, hi = cuts[index], cuts[index + 1]
        counts[color] = _bend_count(black[lo:hi, :])
    ranked = sorted((count, color) for color, count in counts.items())
    if ranked[0][0] == ranked[1][0]:
        raise ValueError("pixel bend counts are tied")
    return ranked[0][1]
