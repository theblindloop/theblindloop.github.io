"""Renderer for the half-turn shade-rank reveal world.

One large bordered panel fills the canvas.  Its 12x12 cells each carry exactly
one of three flat shades, and the rendered shade census is fixed: 48 light, 48
mid and 48 dark cells, with 24 of each shade in the upper half.  The declared
raster operation is a half turn of the panel about its own centre, marked by
four notches on the frame: every cell of the upper half is compared with the
cell diagonally opposite it through that centre, and it is marked when it is
strictly darker than that partner.  The 72 marks form one of six figures on a
12-wide, 6-tall lattice.

Nothing in the picture is the figure.  Each antipodal pair straddles both the
horizontal and the vertical centre line, so no half of the panel contains a
single complete pair and no half of the picture constrains the answer at all.
"""

from __future__ import annotations

import random
from typing import Any

import numpy as np
from PIL import Image

RENDERER_VERSION = "halfturn_shade_rank_reveal-renderer-0.1.0"

GRID = 12
HALF = GRID // 2
QUARANTINE_CLEARANCE = 24.0

# Answer alphabet: 6 rows x 12 columns.  '#' is a marked position.
SYMBOL_ART: dict[str, tuple[str, ...]] = {
    "cross": (
        ".....##.....",
        ".....##.....",
        "############",
        "############",
        ".....##.....",
        ".....##.....",
    ),
    "ex": (
        "##........##",
        ".##......##.",
        "..##....##..",
        "..##....##..",
        ".##......##.",
        "##........##",
    ),
    "ell": (
        "###.........",
        "###.........",
        "###.........",
        "###.........",
        "############",
        "############",
    ),
    "ring": (
        "############",
        "#..........#",
        "#..........#",
        "#..........#",
        "#..........#",
        "############",
    ),
    "tee": (
        "############",
        "############",
        ".....##.....",
        ".....##.....",
        ".....##.....",
        ".....##.....",
    ),
    "arrow": (
        ".##......##.",
        "..##....##..",
        "...##..##...",
        "....####....",
        ".....##.....",
        ".....##.....",
    ),
}

SYMBOLS = tuple(sorted(SYMBOL_ART))


def symbol_bits(name: str) -> tuple[tuple[int, ...], ...]:
    """Marked-position bitmap of one answer figure."""

    return tuple(
        tuple(1 if ch == "#" else 0 for ch in row) for row in SYMBOL_ART[name]
    )


BITS = {name: symbol_bits(name) for name in SYMBOLS}


def _choose_marked_split(rng: random.Random, marked: int) -> tuple[int, int, int]:
    """Split marked pairs into (mid/light, dark/light, dark/mid) ordered types.

    The split is constrained so that both panel halves keep an exactly uniform
    24/24/24 shade census, which makes shade area, ink and colour carry no
    information about the answer.
    """

    for _ in range(20000):
        x1 = rng.randint(0, min(24, marked))
        upper = min(24 - x1, marked - x1)
        if upper < 0:
            continue
        x2 = rng.randint(0, upper)
        x3 = marked - x1 - x2
        if 0 <= x3 <= 24 and x2 + x3 <= 24:
            return x1, x2, x3
    raise ValueError(f"no feasible shade split for {marked} marked pairs")


def _shade_levels(rng: random.Random, name: str) -> list[list[int]]:
    """Assign a shade level (0 light, 1 mid, 2 dark) to every one of 144 cells."""

    bits = BITS[name]
    marked_pos = [(r, c) for r in range(HALF) for c in range(GRID) if bits[r][c]]
    plain_pos = [(r, c) for r in range(HALF) for c in range(GRID) if not bits[r][c]]
    x1, x2, x3 = _choose_marked_split(rng, len(marked_pos))

    # Ordered (upper, lower) shade pairs with upper strictly darker than lower.
    marked_pairs = [(1, 0)] * x1 + [(2, 0)] * x2 + [(2, 1)] * x3
    rng.shuffle(marked_pairs)
    rng.shuffle(marked_pos)

    # Unmarked pairs: upper never darker than lower.  Sorting both multisets
    # ascending and pairing elementwise guarantees that relation.
    up = [0] * 24 + [1] * (24 - x1) + [2] * (24 - x2 - x3)
    low = [0] * (24 - x1 - x2) + [1] * (24 - x3) + [2] * 24
    if min(up) < 0 or min(low) < 0 or len(up) != len(plain_pos):
        raise ValueError("infeasible unmarked shade census")
    plain_pairs = list(zip(up, low))
    rng.shuffle(plain_pairs)
    rng.shuffle(plain_pos)

    levels = [[-1] * GRID for _ in range(GRID)]
    for (r, c), (hi, lo) in zip(marked_pos, marked_pairs):
        levels[r][c] = hi
        levels[GRID - 1 - r][GRID - 1 - c] = lo
    for (r, c), (hi, lo) in zip(plain_pos, plain_pairs):
        levels[r][c] = hi
        levels[GRID - 1 - r][GRID - 1 - c] = lo
    if any(v < 0 for row in levels for v in row):
        raise ValueError("incomplete shade assignment")
    return levels


def _saturated(rng: random.Random, base: tuple[int, int, int]) -> list[int]:
    """A clearly non-neutral colour: no cell shade can ever be mistaken for it."""

    out = [max(0, min(255, v + rng.randint(-10, 10))) for v in base]
    hi, lo = out.index(max(out)), out.index(min(out))
    need = 60 - (out[hi] - out[lo])
    if need > 0:
        out[hi] = max(0, min(255, out[hi] + (need + 1) // 2))
        out[lo] = max(0, min(255, out[lo] - (need + 1) // 2))
    if max(out) - min(out) < 55:
        raise ValueError("failed to sample a non-neutral colour")
    return out


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one latent scene.  Every field below is rendered."""

    rng = random.Random(seed * 7919 + 13)
    name = SYMBOLS[seed % len(SYMBOLS)]

    cell_px = 44
    border_px = 8
    rule_px = 2
    interior = GRID * cell_px
    canvas = 640
    span = interior + 2 * border_px
    slack = canvas - span
    origin_x = rng.randint(slack // 2 - 8, slack // 2 + 8)
    origin_y = rng.randint(slack // 2 - 8, slack // 2 + 8)

    dark = rng.randint(58, 78)
    step = rng.randint(44, 72)
    tint = [rng.randint(-5, 5) for _ in range(3)]
    frame = rng.randint(10, 24)

    scene: dict[str, Any] = {
        "canvas": canvas,
        "cell_px": cell_px,
        "border_px": border_px,
        "rule_px": rule_px,
        "origin": [origin_x, origin_y],
        "shade_dark": dark,
        "shade_mid": dark + step,
        "shade_light": dark + 2 * step,
        "tint": tint,
        "frame_value": frame,
        "background": _saturated(rng, (52, 104, 118)),
        "accent": _saturated(rng, (214, 122, 44)),
        "notch_px": rng.choice([10, 12, 14]),
        "symbol": name,
        "levels": _shade_levels(rng, name),
        "clearance": step / 2.0,
    }
    return scene


def _shade_rgb(scene: dict[str, Any], level: int) -> tuple[int, int, int]:
    value = (scene["shade_light"], scene["shade_mid"], scene["shade_dark"])[level]
    return tuple(
        int(max(0, min(255, value + scene["tint"][k]))) for k in range(3)
    )  # type: ignore[return-value]


def render(scene: dict[str, Any]) -> Image.Image:
    """Rasterise the scene with flat, hard-edged blocks (no antialiasing)."""

    n = int(scene["canvas"])
    cell = int(scene["cell_px"])
    border = int(scene["border_px"])
    rule = int(scene["rule_px"])
    ox, oy = (int(v) for v in scene["origin"])
    arr = np.zeros((n, n, 3), dtype=np.uint8)
    arr[:, :] = np.array(scene["background"], dtype=np.uint8)

    interior = GRID * cell
    span = interior + 2 * border
    frame_rgb = np.array([int(scene["frame_value"])] * 3, dtype=np.uint8)
    arr[oy : oy + span, ox : ox + span] = frame_rgb

    ix, iy = ox + border, oy + border
    levels = scene["levels"]
    for r in range(GRID):
        for c in range(GRID):
            rgb = np.array(_shade_rgb(scene, int(levels[r][c])), dtype=np.uint8)
            y0 = iy + r * cell
            x0 = ix + c * cell
            arr[y0 : y0 + cell, x0 : x0 + cell] = rgb

    accent = np.array(scene["accent"], dtype=np.uint8)
    for k in range(1, GRID):
        y = iy + k * cell - rule // 2
        arr[y : y + rule, ix : ix + interior] = accent
        x = ix + k * cell - rule // 2
        arr[iy : iy + interior, x : x + rule] = accent

    # Four centre notches on the frame name the half-turn centre.
    notch = int(scene["notch_px"])
    cx = ix + interior // 2
    cy = iy + interior // 2
    arr[oy : oy + border, cx - notch // 2 : cx + notch // 2] = accent
    arr[oy + span - border : oy + span, cx - notch // 2 : cx + notch // 2] = accent
    arr[cy - notch // 2 : cy + notch // 2, ox : ox + border] = accent
    arr[cy - notch // 2 : cy + notch // 2, ox + span - border : ox + span] = accent

    return Image.fromarray(arr, mode="RGB")


def _darkness(level: int) -> int:
    return level  # level 2 is the darkest shade


def marks(scene: dict[str, Any]) -> tuple[tuple[int, ...], ...]:
    """Execute the declared operation on the latent shade grid."""

    levels = scene["levels"]
    out = []
    for r in range(HALF):
        row = []
        for c in range(GRID):
            upper = _darkness(int(levels[r][c]))
            lower = _darkness(int(levels[GRID - 1 - r][GRID - 1 - c]))
            row.append(1 if upper > lower else 0)
        out.append(tuple(row))
    return tuple(out)


def analytic_gold(scene: dict[str, Any]) -> str:
    """Analytic answer, read only from the rendered shade grid."""

    found = marks(scene)
    for name in SYMBOLS:
        if BITS[name] == found:
            return name
    raise ValueError("marked positions do not form a declared figure")


def margin(scene: dict[str, Any]) -> float:
    """Half the rendered luma gap between neighbouring shades, in luma units."""

    light = np.array(_shade_rgb(scene, 0), dtype=float)
    mid = np.array(_shade_rgb(scene, 1), dtype=float)
    dark = np.array(_shade_rgb(scene, 2), dtype=float)
    weights = np.array([0.299, 0.587, 0.114])
    lumas = [float(v @ weights) for v in (light, mid, dark)]
    gap = min(lumas[0] - lumas[1], lumas[1] - lumas[2])
    return round(gap / 2.0, 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Flag scenes whose shade separation is not reliably separable by eye."""

    return bool(margin(scene) < QUARANTINE_CLEARANCE)


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No latent aliases exist: the raster fixes every field gold reads."""

    return []
