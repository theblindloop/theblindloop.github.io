"""Deterministic renderer for the cut-cell mosaic search world."""

from __future__ import annotations

import random
from collections import deque

from PIL import Image, ImageDraw

CANVAS = 512
BACKGROUND = (247, 244, 236)
CYAN_A = (49, 178, 194)
CYAN_B = (76, 196, 208)
MARKER_COLORS = {
    "red": (218, 69, 72),
    "gold": (231, 169, 45),
    "violet": (131, 83, 184),
    "green": (55, 151, 93),
}
COLOR_NAMES = tuple(MARKER_COLORS)


def _components(cells: set[tuple[int, int]]) -> int:
    unseen = set(cells)
    count = 0
    while unseen:
        count += 1
        start = unseen.pop()
        queue = deque([start])
        while queue:
            r, c = queue.popleft()
            for nxt in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
                if nxt in unseen:
                    unseen.remove(nxt)
                    queue.append(nxt)
    return count


def _articulation_scores(
    occupied_rows: list[list[int]], markers: dict[str, list[int]]
) -> dict[str, int]:
    occupied = {tuple(cell) for cell in occupied_rows}
    baseline = _components(occupied)
    return {
        color: _components(occupied - {tuple(cell)}) - baseline
        for color, cell in markers.items()
    }


def _transform(point: tuple[int, int], rotation: int, reflected: bool) -> tuple[int, int]:
    r, c = point
    if reflected:
        c = -c
    for _ in range(rotation):
        r, c = c, -r
    return r, c


def _normalize(points: set[tuple[int, int]]) -> tuple[set[tuple[int, int]], int, int]:
    min_r = min(r for r, _ in points)
    min_c = min(c for _, c in points)
    return {(r - min_r, c - min_c) for r, c in points}, min_r, min_c


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))

    # Two broad 6x4 lobes overlap vertically and meet through a one-cell neck.
    # The whole 49-cell shape is compact; the neck is a counterfactual cut cell.
    left = {(r, c) for r in range(0, 6) for c in range(0, 4)}
    right = {(r, c) for r in range(1, 7) for c in range(5, 9)}
    neck = (3, 4)
    base = left | right | {neck}

    rotation = rng.randrange(4)
    reflected = bool(rng.randrange(2))
    transformed = {_transform(p, rotation, reflected) for p in base}
    transformed_neck = _transform(neck, rotation, reflected)
    occupied, min_r, min_c = _normalize(transformed)
    neck_cell = (transformed_neck[0] - min_r, transformed_neck[1] - min_c)

    articulation_cells = {
        cell for cell in occupied if _components(occupied - {cell}) > 1
    }
    distractor_pool = [
        cell
        for cell in sorted(occupied)
        if cell not in articulation_cells
        and sum(
            neighbor in occupied
            for neighbor in (
                (cell[0] - 1, cell[1]),
                (cell[0] + 1, cell[1]),
                (cell[0], cell[1] - 1),
                (cell[0], cell[1] + 1),
            )
        )
        >= 2
    ]
    distractors = rng.sample(distractor_pool, 3)

    winner_color = COLOR_NAMES[int(seed) % len(COLOR_NAMES)]
    other_colors = [color for color in COLOR_NAMES if color != winner_color]
    rng.shuffle(other_colors)
    markers = {winner_color: list(neck_cell)}
    for color, cell in zip(other_colors, distractors, strict=True):
        markers[color] = list(cell)

    cell_size = 24
    height = (max(r for r, _ in occupied) + 1) * cell_size
    width = (max(c for _, c in occupied) + 1) * cell_size
    jitter_x = rng.randint(-8, 8)
    jitter_y = rng.randint(-8, 8)
    offset = [(CANVAS - width) // 2 + jitter_x, (CANVAS - height) // 2 + jitter_y]
    scene = {
        "occupied": [list(cell) for cell in sorted(occupied)],
        "markers": {color: markers[color] for color in COLOR_NAMES},
        "cell_size": cell_size,
        "offset": offset,
    }
    if is_quarantined(scene):
        raise RuntimeError("internal sampler produced an ambiguous scene")
    return scene


def render(scene: dict) -> Image.Image:
    occupied_rows = scene["occupied"]
    markers = scene["markers"]
    cell_size = int(scene["cell_size"])
    offset_x, offset_y = map(int, scene["offset"])

    image = Image.new("RGB", (CANVAS, CANVAS), BACKGROUND)
    draw = ImageDraw.Draw(image)
    marker_at = {tuple(cell): color for color, cell in markers.items()}
    for row, col in (tuple(cell) for cell in occupied_rows):
        x0 = offset_x + col * cell_size
        y0 = offset_y + row * cell_size
        fill = MARKER_COLORS[marker_at[(row, col)]] if (row, col) in marker_at else (
            CYAN_A if (row + col) % 2 == 0 else CYAN_B
        )
        draw.rectangle(
            (x0, y0, x0 + cell_size - 1, y0 + cell_size - 1),
            fill=fill,
        )
    return image


def analytic_gold(scene: dict) -> str:
    scores = _articulation_scores(scene["occupied"], scene["markers"])
    return max(COLOR_NAMES, key=lambda color: scores[color])


def margin(scene: dict) -> float:
    scores = sorted(
        _articulation_scores(scene["occupied"], scene["markers"]).values(), reverse=True
    )
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    try:
        occupied = {tuple(cell) for cell in scene["occupied"]}
        markers = scene["markers"]
        scores = _articulation_scores(scene["occupied"], markers)
        values = sorted(scores.values(), reverse=True)
        return (
            len(occupied) != 49
            or _components(occupied) != 1
            or set(markers) != set(COLOR_NAMES)
            or len({tuple(cell) for cell in markers.values()}) != 4
            or any(tuple(cell) not in occupied for cell in markers.values())
            or values[0] != 1
            or values[1] != 0
        )
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
