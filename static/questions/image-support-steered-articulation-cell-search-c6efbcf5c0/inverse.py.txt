"""Independent image-only oracle for the cut-cell mosaic search."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

BACKGROUND = np.array((247, 244, 236), dtype=np.int16)
CYAN_A = np.array((49, 178, 194), dtype=np.int16)
CYAN_B = np.array((76, 196, 208), dtype=np.int16)
MARKERS = {
    "red": np.array((218, 69, 72), dtype=np.int16),
    "gold": np.array((231, 169, 45), dtype=np.int16),
    "violet": np.array((131, 83, 184), dtype=np.int16),
    "green": np.array((55, 151, 93), dtype=np.int16),
}
INKS = (CYAN_A, CYAN_B, *MARKERS.values())


def _near(array: np.ndarray, color: np.ndarray, tolerance: int = 8) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _components(cells: set[tuple[int, int]]) -> int:
    unseen = set(cells)
    count = 0
    while unseen:
        count += 1
        queue = deque([unseen.pop()])
        while queue:
            r, c = queue.popleft()
            for nxt in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
                if nxt in unseen:
                    unseen.remove(nxt)
                    queue.append(nxt)
    return count


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    height, width = array.shape[:2]
    if height < 64 or width < 64:
        raise ValueError("raster is too small")

    marker_boxes: dict[str, tuple[int, int, int, int]] = {}
    sizes = []
    for name, rgb in MARKERS.items():
        mask = _near(array, rgb)
        ys, xs = np.nonzero(mask)
        if len(xs) == 0:
            raise ValueError(f"missing {name} marker")
        x0, x1 = int(xs.min()), int(xs.max())
        y0, y1 = int(ys.min()), int(ys.max())
        side_x, side_y = x1 - x0 + 1, y1 - y0 + 1
        if side_x != side_y or side_x < 12 or int(mask[y0 : y1 + 1, x0 : x1 + 1].sum()) != side_x * side_y:
            raise ValueError(f"damaged {name} marker")
        marker_boxes[name] = (x0, y0, x1, y1)
        sizes.append(side_x)
    if len(set(sizes)) != 1:
        raise ValueError("marker cells disagree in size")
    cell = sizes[0]

    ref_x, ref_y, _, _ = marker_boxes["red"]
    allowed = np.zeros((height, width), dtype=bool)
    for rgb in INKS:
        allowed |= _near(array, rgb)

    lattice: dict[tuple[int, int], tuple[int, int]] = {}
    for gy in range(-height // cell - 2, height // cell + 3):
        y0 = ref_y + gy * cell
        if y0 < 0 or y0 + cell > height:
            continue
        for gx in range(-width // cell - 2, width // cell + 3):
            x0 = ref_x + gx * cell
            if x0 < 0 or x0 + cell > width:
                continue
            cy, cx = y0 + cell // 2, x0 + cell // 2
            if allowed[cy, cx]:
                block = allowed[y0 : y0 + cell, x0 : x0 + cell]
                if int(block.sum()) != cell * cell:
                    raise ValueError("occupied cell is partially erased")
                lattice[(gy, gx)] = (x0, y0)

    occupied = set(lattice)
    if len(occupied) != 49 or _components(occupied) != 1:
        raise ValueError("mosaic is incomplete or disconnected")

    marker_cells: dict[str, tuple[int, int]] = {}
    for name, (x0, y0, _, _) in marker_boxes.items():
        delta_x, delta_y = x0 - ref_x, y0 - ref_y
        if delta_x % cell or delta_y % cell:
            raise ValueError("markers do not share one visible lattice")
        key = (delta_y // cell, delta_x // cell)
        if key not in occupied:
            raise ValueError("marker is not an occupied cell")
        marker_cells[name] = key
    if len(set(marker_cells.values())) != 4:
        raise ValueError("markers overlap")

    baseline = _components(occupied)
    scores = {
        name: _components(occupied - {cell_key}) - baseline
        for name, cell_key in marker_cells.items()
    }
    ranked = sorted(scores.items(), key=lambda item: item[1], reverse=True)
    if ranked[0][1] != 1 or ranked[1][1] != 0:
        raise ValueError("no unique cut-cell winner with runner-up gap")
    return ranked[0][0]
