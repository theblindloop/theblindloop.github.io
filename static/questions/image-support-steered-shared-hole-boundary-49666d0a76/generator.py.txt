"""Deterministic latent sampler and block-exact renderer."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw


PALETTES = (
    {"slab": (31, 40, 52), "cyan": (24, 181, 196), "orange": (241, 128, 36)},
    {"slab": (39, 45, 55), "cyan": (38, 151, 221), "orange": (226, 91, 66)},
    {"slab": (27, 43, 46), "cyan": (53, 190, 165), "orange": (238, 164, 38)},
)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    layout = rng.randrange(2)
    if layout == 0:
        cavities = [
            [104 + rng.randint(-3, 3), 104 + rng.randint(-3, 3), 158 + rng.randint(-3, 3), 158 + rng.randint(-3, 3)],
            [226 + rng.randint(-3, 3), 226 + rng.randint(-3, 3), 280 + rng.randint(-3, 3), 280 + rng.randint(-3, 3)],
        ]
    else:
        cavities = [
            [226 + rng.randint(-3, 3), 104 + rng.randint(-3, 3), 280 + rng.randint(-3, 3), 158 + rng.randint(-3, 3)],
            [104 + rng.randint(-3, 3), 226 + rng.randint(-3, 3), 158 + rng.randint(-3, 3), 280 + rng.randint(-3, 3)],
        ]
    # Even seeds are positive. Odd seeds alternate which color exposes the bay,
    # balancing the failing color and color position over every sequential sweep.
    if int(seed) & 1:
        cyan_enclosed = bool((int(seed) >> 1) & 1)
        orange_enclosed = not cyan_enclosed
    else:
        cyan_enclosed = True
        orange_enclosed = True
    return {
        "canvas": 384,
        "offset": [rng.randint(-8, 8), rng.randint(-8, 8)],
        "layout": layout,
        "cavities": cavities,
        "palette": rng.randrange(len(PALETTES)),
        "cyan_slot": rng.randrange(2),
        "cyan_enclosed": cyan_enclosed,
        "orange_enclosed": orange_enclosed,
        "channel_width": 18 + 2 * rng.randrange(4),
        "patch_depth": 11 + rng.randrange(3),
        "patch_length": 26 + 2 * rng.randrange(5),
    }


def _shift(rect: list[int], offset: list[int]) -> list[int]:
    return [rect[0] + offset[0], rect[1] + offset[1], rect[2] + offset[0], rect[3] + offset[1]]


def _patch_rect(cavity: list[int], side: str, depth: int, length: int) -> list[int]:
    x0, y0, x1, y1 = cavity
    cx, cy = (x0 + x1) // 2, (y0 + y1) // 2
    a, b = length // 2, length - length // 2 - 1
    if side == "top":
        return [cx - a, y0 - depth, cx + b, y0 - 1]
    if side == "bottom":
        return [cx - a, y1 + 1, cx + b, y1 + depth]
    if side == "left":
        return [x0 - depth, cy - a, x0 - 1, cy + b]
    return [x1 + 1, cy - a, x1 + depth, cy + b]


def _channel_rect(cavity: list[int], side: str, slab: list[int], width: int) -> list[int]:
    x0, y0, x1, y1 = cavity
    cx, cy = (x0 + x1) // 2, (y0 + y1) // 2
    a, b = width // 2, width - width // 2 - 1
    if side == "left":
        return [slab[0], cy - a, x0 - 1, cy + b]
    if side == "right":
        return [x1 + 1, cy - a, slab[2], cy + b]
    if side == "top":
        return [cx - a, slab[1], cx + b, y0 - 1]
    return [cx - a, y1 + 1, cx + b, slab[3]]


def render(scene: dict) -> Image.Image:
    canvas = int(scene["canvas"])
    offset = [int(v) for v in scene["offset"]]
    layout = int(scene["layout"])
    cavities = [[int(v) for v in rect] for rect in scene["cavities"]]
    palette = PALETTES[int(scene["palette"])]
    cyan_slot = int(scene["cyan_slot"])
    cyan_enclosed = bool(scene["cyan_enclosed"])
    orange_enclosed = bool(scene["orange_enclosed"])
    channel_width = int(scene["channel_width"])
    depth = int(scene["patch_depth"])
    length = int(scene["patch_length"])
    slab = [68, 68, 316, 316]
    shifted_slab = _shift(slab, offset)
    shifted_cavities = [_shift(rect, offset) for rect in cavities]
    image = Image.new("RGB", (canvas, canvas), (255, 255, 255))
    draw = ImageDraw.Draw(image)
    draw.rectangle(shifted_slab, fill=palette["slab"])
    for cavity in shifted_cavities:
        draw.rectangle(cavity, fill=(255, 255, 255))

    colors = [palette["orange"], palette["orange"]]
    colors[cyan_slot] = palette["cyan"]
    states = [orange_enclosed, orange_enclosed]
    states[cyan_slot] = cyan_enclosed
    if layout == 0:
        patch_sides, channel_sides = ["left", "right"], ["top", "bottom"]
    else:
        patch_sides, channel_sides = ["right", "left"], ["top", "bottom"]
    for slot, cavity in enumerate(shifted_cavities):
        if not states[slot]:
            draw.rectangle(_channel_rect(cavity, channel_sides[slot], shifted_slab, channel_width), fill=(255, 255, 255))
        draw.rectangle(_patch_rect(cavity, patch_sides[slot], depth, length), fill=colors[slot])
    return image


def analytic_gold(scene: dict) -> str:
    return "yes" if bool(scene["cyan_enclosed"]) and bool(scene["orange_enclosed"]) else "no"


def margin(scene: dict) -> float:
    return float(min(int(scene["channel_width"]), int(scene["patch_depth"]), int(scene["patch_length"]), 20))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
