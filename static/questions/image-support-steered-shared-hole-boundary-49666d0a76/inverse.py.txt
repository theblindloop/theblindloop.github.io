"""Independent PNG-only inverse for enclosed-component boundary incidence."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> tuple[np.ndarray, dict[int, bool]]:
    h, w = mask.shape
    labels = np.zeros((h, w), dtype=np.int32)
    border: dict[int, bool] = {}
    label = 0
    for y, x in zip(*np.nonzero(mask)):
        if labels[y, x]:
            continue
        label += 1
        labels[y, x] = label
        q = deque([(int(y), int(x))])
        touches = y == 0 or x == 0 or y == h - 1 or x == w - 1
        while q:
            yy, xx = q.popleft()
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and labels[ny, nx] == 0:
                    labels[ny, nx] = label
                    q.append((ny, nx))
                    touches = touches or ny == 0 or nx == 0 or ny == h - 1 or nx == w - 1
        border[label] = bool(touches)
    return labels, border


def _adjacent_white(color_mask: np.ndarray, labels: np.ndarray) -> set[int]:
    adjacent = set()
    for y, x in zip(*np.nonzero(color_mask)):
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < labels.shape[0] and 0 <= nx < labels.shape[1]:
                value = int(labels[ny, nx])
                if value:
                    adjacent.add(value)
    return adjacent


def decision_from_image(image: Image.Image) -> str:
    a = np.asarray(image.convert("RGB"), dtype=np.uint8)
    white = np.all(a >= 248, axis=2)
    cyan = (a[:, :, 0] < 90) & (a[:, :, 1] > 130) & (a[:, :, 2] > 135)
    orange = (a[:, :, 0] > 190) & (a[:, :, 1] > 65) & (a[:, :, 1] < 190) & (a[:, :, 2] < 100)
    if int(cyan.sum()) < 60 or int(orange.sum()) < 60:
        raise ValueError("missing colored boundary patch")
    labels, border = _components(white)
    cyan_neighbors = _adjacent_white(cyan, labels)
    orange_neighbors = _adjacent_white(orange, labels)
    if len(cyan_neighbors) != 1 or len(orange_neighbors) != 1:
        raise ValueError("patch contact is ambiguous")
    cyan_label = next(iter(cyan_neighbors))
    orange_label = next(iter(orange_neighbors))
    return "yes" if not border[cyan_label] and not border[orange_label] else "no"
