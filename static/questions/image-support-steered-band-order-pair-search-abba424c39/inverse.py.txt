"""Independent RGB-only inverse for the reversed-band visual search."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    found: list[list[tuple[int, int]]] = []
    for y, x in np.argwhere(mask):
        iy, ix = int(y), int(x)
        if seen[iy, ix]:
            continue
        queue = deque([(iy, ix)])
        seen[iy, ix] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= 500:
            found.append(points)
    return found


def _ribbon_info(points: list[tuple[int, int]], pixels: np.ndarray) -> tuple[float, float, tuple[tuple[int, int, int], ...]]:
    ys = [point[0] for point in points]
    xs = [point[1] for point in points]
    y0, y1 = min(ys), max(ys)
    x0, x1 = min(xs), max(xs)
    width = x1 - x0 + 1
    height = y1 - y0 + 1
    area = len(points)
    if not (145 <= width <= 195 and 48 <= height <= 66):
        raise ValueError("foreground component is not a complete three-band ribbon")
    if area < int(0.97 * width * height):
        raise ValueError("ribbon component has erased pixels")
    center_y = (y0 + y1) // 2
    centers = (x0 + width // 6, x0 + width // 2, x0 + (5 * width) // 6)
    sequence = tuple(tuple(int(value) for value in pixels[center_y, center_x]) for center_x in centers)
    if len(set(sequence)) != 3:
        raise ValueError("ribbon does not expose three distinct bands")
    return (float(sum(xs) / area), float(sum(ys) / area), sequence)


def decision_from_image(image: Image.Image) -> str:
    """Recover the quadrant from only the final RGB raster."""

    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    if pixels.ndim != 3 or pixels.shape[2] != 3 or pixels.shape[0] < 100 or pixels.shape[1] < 100:
        raise ValueError("unexpected image shape")
    background = pixels[0, 0]
    difference = np.max(np.abs(pixels - background), axis=2)
    mask = difference > 20
    ribbons = _components(mask)

    grouped: dict[str, list[tuple[float, float, tuple[tuple[int, int, int], ...]]]] = {
        "NW": [],
        "NE": [],
        "SW": [],
        "SE": [],
    }
    half_width = pixels.shape[1] / 2.0
    half_height = pixels.shape[0] / 2.0
    for component in ribbons:
        try:
            info = _ribbon_info(component, pixels)
        except ValueError:
            # A grid ablation may erase a distractor or damage one of its
            # bands.  Such a component is not evidence for a new winner; the
            # decisive pair must still be independently certified below.
            continue
        x_center, y_center, _ = info
        quadrant = ("N" if y_center < half_height else "S") + ("W" if x_center < half_width else "E")
        grouped[quadrant].append(info)

    winners: list[str] = []
    mismatch_scores: dict[str, int] = {}
    for quadrant, pair in grouped.items():
        if len(pair) != 2:
            continue
        left = pair[0][2]
        right = pair[1][2]
        mismatch_scores[quadrant] = sum(left[index] != right[2 - index] for index in range(3))
        if mismatch_scores[quadrant] == 0:
            winners.append(quadrant)
    if len(winners) != 1:
        raise ValueError("raster has no unique reverse-order winner")
    return winners[0]
