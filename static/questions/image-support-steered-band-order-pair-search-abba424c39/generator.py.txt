"""Deterministic four-quadrant reversed-band visual-search renderer."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 800
HEIGHT = 800
BAND_WIDTH = 56
BAND_HEIGHT = 56
QUADRANTS = ("NW", "NE", "SW", "SE")

# Each triplet is high-contrast against the light backgrounds and its members
# are distinct.  A scene uses one triplet, but shuffles its identity.
PALETTES = (
    ((214, 62, 70), (52, 104, 190), (48, 157, 91)),
    ((228, 137, 42), (111, 76, 181), (24, 151, 164)),
    ((198, 70, 151), (39, 130, 190), (224, 173, 44)),
    ((52, 142, 105), (206, 73, 64), (116, 85, 191)),
)
BACKGROUNDS = ((250, 250, 247), (247, 250, 252))


def _permutations(values: list[tuple[int, int, int]]) -> list[list[tuple[int, int, int]]]:
    """Return the six permutations without depending on itertools ordering."""

    result: list[list[tuple[int, int, int]]] = []
    for i in range(3):
        for j in range(3):
            if j == i:
                continue
            for k in range(3):
                if k in (i, j):
                    continue
                result.append([values[i], values[j], values[k]])
    return result


def _quadrant_bases(quadrant: str) -> tuple[int, int]:
    x_base = 0 if quadrant in ("NW", "SW") else 400
    y_base = 0 if quadrant in ("NW", "NE") else 400
    return x_base, y_base


def _record(quadrant: str, x: int, y: int, colors: Iterable[tuple[int, int, int]]) -> dict:
    return {
        "quadrant": quadrant,
        "x": int(x),
        "y": int(y),
        "colors": [list(color) for color in colors],
    }


def sample_scene(seed: int) -> dict:
    """Sample one valid scene; the seed also counterbalances the answer quadrant."""

    rng = random.Random(int(seed))
    palette = [tuple(color) for color in PALETTES[rng.randrange(len(PALETTES))]]
    rng.shuffle(palette)
    target = QUADRANTS[int(seed) % len(QUADRANTS)]
    records: list[dict] = []

    for quadrant in QUADRANTS:
        first = list(palette)
        rng.shuffle(first)
        reverse = list(reversed(first))
        if quadrant == target:
            second = reverse
        else:
            options = [candidate for candidate in _permutations(first) if candidate != first and candidate != reverse]
            second = list(rng.choice(options))
        if rng.randrange(2):
            first, second = second, first

        x_base, y_base = _quadrant_bases(quadrant)
        jitter_top = rng.randrange(-6, 7)
        jitter_bottom = rng.randrange(-6, 7)
        top_y = y_base + 20 + rng.randrange(-6, 7)
        bottom_y = y_base + 300 + rng.randrange(-6, 7)
        left_x = x_base + 20 + jitter_top
        right_x = x_base + 220 + jitter_bottom

        # Randomize whether the upper ribbon is the left or right member.  The
        # two possible diagonals have the same support and silhouette.
        if rng.randrange(2):
            top_x, bottom_x = right_x, left_x
        else:
            top_x, bottom_x = left_x, right_x
        records.append(_record(quadrant, top_x, top_y, first))
        records.append(_record(quadrant, bottom_x, bottom_y, second))

    rng.shuffle(records)
    return {"ribbons": records, "background": list(BACKGROUNDS[rng.randrange(len(BACKGROUNDS))])}


def render(scene: dict) -> Image.Image:
    """Render flat, contiguous RGB bands with no hidden guides or metadata."""

    background = tuple(int(value) for value in scene["background"])
    image = Image.new("RGB", (WIDTH, HEIGHT), background)
    draw = ImageDraw.Draw(image)
    for ribbon in scene["ribbons"]:
        x = int(ribbon["x"])
        y = int(ribbon["y"])
        for index, color in enumerate(ribbon["colors"]):
            x0 = x + index * BAND_WIDTH
            draw.rectangle(
                (x0, y, x0 + BAND_WIDTH - 1, y + BAND_HEIGHT - 1),
                fill=tuple(int(value) for value in color),
            )
    return image


def _grouped_ribbons(scene: dict) -> dict[str, list[dict]]:
    groups = {quadrant: [] for quadrant in QUADRANTS}
    for ribbon in scene["ribbons"]:
        quadrant = str(ribbon["quadrant"])
        if quadrant in groups:
            groups[quadrant].append(ribbon)
    return groups


def _sequence(colors: Iterable[Iterable[int]]) -> tuple[tuple[int, int, int], ...]:
    return tuple(tuple(int(value) for value in color) for color in colors)


def _mismatch(pair: list[dict]) -> int:
    if len(pair) != 2:
        return 99
    left = _sequence(pair[0]["colors"])
    right = _sequence(pair[1]["colors"])
    if len(left) != 3 or len(right) != 3:
        return 99
    return sum(left[index] != right[2 - index] for index in range(3))


def _pair_mismatches(scene: dict) -> dict[str, int]:
    groups = _grouped_ribbons(scene)
    return {quadrant: _mismatch(groups[quadrant]) for quadrant in QUADRANTS}


def analytic_gold(scene: dict) -> str:
    """Select the unique quadrant with zero reverse-position mismatches."""

    scores = _pair_mismatches(scene)
    winners = [quadrant for quadrant, score in scores.items() if score == min(scores.values())]
    if len(winners) != 1 or scores[winners[0]] != 0:
        raise ValueError("scene has no unique visible reverse-order winner")
    return winners[0]


def margin(scene: dict) -> float:
    """Return the gap between the best and second-best reverse mismatches."""

    values = sorted(_pair_mismatches(scene).values())
    if len(values) < 2:
        return 0.0
    return float(values[1] - values[0])


def is_quarantined(scene: dict) -> bool:
    """Reject incomplete, tied, or non-distinct-band scenes."""

    groups = _grouped_ribbons(scene)
    if any(len(groups[quadrant]) != 2 for quadrant in QUADRANTS):
        return True
    for ribbons in groups.values():
        for ribbon in ribbons:
            colors = _sequence(ribbon["colors"])
            if len(colors) != 3 or len(set(colors)) != 3:
                return True
    scores = sorted(_pair_mismatches(scene).values())
    return len(scores) != 4 or scores[0] != 0 or scores[1] - scores[0] < 1


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Expose a nontrivial record-order alias with identical raster output."""

    twin = dict(scene)
    twin["ribbons"] = list(reversed(scene["ribbons"]))
    return [("reverse_ribbon_record_order", twin)]
