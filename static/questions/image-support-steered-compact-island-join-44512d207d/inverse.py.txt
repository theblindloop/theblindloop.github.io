"""Independent image-only connected-component oracle."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int = 1) -> list[np.ndarray]:
    points = np.argwhere(mask)
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    found: list[np.ndarray] = []
    for y0, x0 in points:
        y0 = int(y0)
        x0 = int(x0)
        if seen[y0, x0]:
            continue
        stack = [(y0, x0)]
        seen[y0, x0] = True
        values: list[tuple[int, int]] = []
        while stack:
            y, x = stack.pop()
            values.append((y, x))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    ny, nx = y + dy, x + dx
                    if (
                        0 <= ny < height
                        and 0 <= nx < width
                        and mask[ny, nx]
                        and not seen[ny, nx]
                    ):
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        if len(values) >= minimum:
            found.append(np.asarray(values, dtype=np.int32))
    return found


def _saturated_mask(array: np.ndarray) -> np.ndarray:
    values = array.astype(np.int16)
    high = values.max(axis=2)
    low = values.min(axis=2)
    return (high - low >= 55) & (high >= 100) & (low <= 242)


def _dark_mask(array: np.ndarray) -> np.ndarray:
    values = array.astype(np.int16)
    spread = values.max(axis=2) - values.min(axis=2)
    intensity = values.mean(axis=2)
    return (spread <= 48) & (intensity < 205)


def _nearest_component(
    point: tuple[float, float], components: list[np.ndarray]
) -> tuple[int, float]:
    y, x = point
    best_index = -1
    best_distance = float("inf")
    for index, component in enumerate(components):
        dy = component[:, 0].astype(np.float64) - y
        dx = component[:, 1].astype(np.float64) - x
        distance = float(np.sqrt(np.min(dy * dy + dx * dx)))
        if distance < best_distance:
            best_index = index
            best_distance = distance
    return best_index, best_distance


def decision_from_image(image: Image.Image) -> str:
    """Recover same-component membership from final PNG pixels only."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    saturated = _saturated_mask(array)
    markers = _components(saturated, minimum=30)
    if len(markers) != 2:
        raise ValueError(f"expected two colored terminal marks, found {len(markers)}")

    centers = [
        (float(component[:, 0].mean()), float(component[:, 1].mean()))
        for component in markers
    ]
    ys = [point[0] for point in centers]
    xs = [point[1] for point in centers]
    height, width = array.shape[:2]
    y0 = max(0, int(np.floor(min(ys) - 32)))
    y1 = min(height, int(np.ceil(max(ys) + 33)))
    x0 = max(0, int(np.floor(min(xs) - 32)))
    x1 = min(width, int(np.ceil(max(xs) + 33)))
    crop = array[y0:y1, x0:x1]
    dark = _dark_mask(crop)
    components = _components(dark, minimum=18)
    if len(components) not in {1, 2}:
        raise ValueError(f"expected one or two dark terminal components, found {len(components)}")

    assignments = []
    for y, x in centers:
        index, distance = _nearest_component((y - y0, x - x0), components)
        if index < 0 or distance > 18.0:
            raise ValueError("colored terminal is not visibly attached to dark foreground")
        assignments.append(index)
    return "joined" if assignments[0] == assignments[1] else "separate"
