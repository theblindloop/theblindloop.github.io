"""Deterministic raster world for a compact marked-region connectivity question."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw


RENDERER_VERSION = "compact_island_join-0.1.0"
CANVAS = (512, 512)
SUPERSAMPLE = 4
DEFAULT_BACKGROUND = (255, 255, 255)
DEFAULT_DARK = (42, 53, 65)
REMOTE_INK = (226, 232, 238)
MARKER_PALETTE = (
    (220, 64, 58),
    (34, 126, 211),
    (20, 157, 130),
    (177, 62, 143),
)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample balanced joined/separate glyphs with randomized local placement."""

    rng = random.Random(int(seed))
    bucket = int(seed) % 8
    joined = bucket in {0, 1, 4, 5}
    tight = bucket in {0, 3, 6}
    cell_x = rng.randint(1, 6)
    cell_y = rng.randint(1, 6)
    origin_x = (cell_x + 0.5) * 64.0 + rng.uniform(-1.5, 1.5)
    origin_y = (cell_y + 0.5) * 64.0 + rng.uniform(-1.5, 1.5)
    left_color, right_color = rng.sample(MARKER_PALETTE, 2)
    return {
        "width": CANVAS[0],
        "height": CANVAS[1],
        "origin_x": round(origin_x, 3),
        "origin_y": round(origin_y, 3),
        "joined": joined,
        "orientation": rng.randrange(4),
        "terminal_offset": 16.0 if tight else 18.0,
        "lobe_half": 11.5 if tight else 13.0,
        "weld_half_width": 6.5 if tight else 7.5,
        "weld_half_height": 7.0 if tight else 8.5,
        "marker_radius": 7.5 if tight else 6.5,
        "left_color": left_color,
        "right_color": right_color,
        "dark_color": DEFAULT_DARK,
        "background": DEFAULT_BACKGROUND,
        "decor_seed": rng.randrange(1, 2**31 - 1),
    }


def _rotate(point: tuple[float, float], orientation: int) -> tuple[float, float]:
    x, y = point
    turn = int(orientation) % 4
    if turn == 0:
        return x, y
    if turn == 1:
        return -y, x
    if turn == 2:
        return -x, -y
    return y, -x


def _point(
    scene: dict[str, Any], local: tuple[float, float], scale: int
) -> tuple[int, int]:
    rx, ry = _rotate(local, int(scene["orientation"]))
    return (
        int(round((float(scene["origin_x"]) + rx) * scale)),
        int(round((float(scene["origin_y"]) + ry) * scale)),
    )


def _disk(
    draw: ImageDraw.ImageDraw,
    scene: dict[str, Any],
    center: tuple[float, float],
    radius: float,
    fill: tuple[int, int, int],
    scale: int,
) -> None:
    cx, cy = _point(scene, center, scale)
    rr = int(round(float(radius) * scale))
    draw.ellipse((cx - rr, cy - rr, cx + rr, cy + rr), fill=fill)


def _line(
    draw: ImageDraw.ImageDraw,
    scene: dict[str, Any],
    start: tuple[float, float],
    end: tuple[float, float],
    width: float,
    fill: tuple[int, int, int],
    scale: int,
) -> None:
    draw.line(
        [_point(scene, start, scale), _point(scene, end, scale)],
        fill=fill,
        width=max(1, int(round(float(width) * scale))),
    )


def _draw_remote_marks(
    draw: ImageDraw.ImageDraw, scene: dict[str, Any], scale: int
) -> None:
    """Draw pale remote marks that are not part of the foreground island."""

    rng = random.Random(int(scene["decor_seed"]))
    for base_x, base_y in ((34, 34), (478, 34), (34, 478), (478, 478)):
        x = base_x + rng.randint(-7, 7)
        y = base_y + rng.randint(-7, 7)
        radius = (2 + rng.randrange(2)) * scale
        draw.ellipse(
            (x * scale - radius, y * scale - radius, x * scale + radius, y * scale + radius),
            fill=tuple(int(value) for value in scene["background"])
            if False
            else REMOTE_INK,
        )


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the final RGB PNG from visible scene fields."""

    width = int(scene["width"])
    height = int(scene["height"])
    scale = SUPERSAMPLE
    background = tuple(int(value) for value in scene["background"])
    dark = tuple(int(value) for value in scene["dark_color"])
    left_color = tuple(int(value) for value in scene["left_color"])
    right_color = tuple(int(value) for value in scene["right_color"])
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    _draw_remote_marks(draw, scene, scale)

    offset = float(scene["terminal_offset"])
    radius = float(scene["lobe_half"])
    weld_width = float(scene["weld_half_width"]) * 2.0
    weld_height = float(scene["weld_half_height"]) * 2.0
    left = (-offset, 0.0)
    right = (offset, 0.0)

    _disk(draw, scene, left, radius, dark, scale)
    _disk(draw, scene, right, radius, dark, scale)
    if bool(scene["joined"]):
        _line(draw, scene, left, right, weld_height, dark, scale)

    marker_radius = float(scene["marker_radius"])
    _disk(draw, scene, left, marker_radius, left_color, scale)
    _disk(draw, scene, right, marker_radius, right_color, scale)
    # Keep weld_width as a rendered nuisance field even though the central weld
    # is represented by a round-capped line whose height controls its raster.
    if bool(scene["joined"]) and weld_width > 0.0:
        _line(draw, scene, left, right, min(weld_width, weld_height), dark, scale)
        _disk(draw, scene, left, marker_radius, left_color, scale)
        _disk(draw, scene, right, marker_radius, right_color, scale)

    return image.resize((width, height), Image.Resampling.LANCZOS).convert("RGB")


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the sole label input: whether the terminal lobes are joined."""

    return "joined" if bool(scene["joined"]) else "separate"


def margin(scene: dict[str, Any]) -> float:
    """Signed visible-clearance margin; positive means a clear weld."""

    radius = float(scene["lobe_half"])
    marker = float(scene["marker_radius"])
    gap_half = max(0.0, float(scene["terminal_offset"]) - radius)
    clearance = min(radius - marker, gap_half)
    return clearance if analytic_gold(scene) == "joined" else -clearance


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Exclude tight marker/gap scenes from headline balance claims."""

    return float(scene["marker_radius"]) >= 7.0 or float(scene["lobe_half"]) < 12.5


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No intentionally pixel-identical latent aliases are declared."""

    del scene
    return []
