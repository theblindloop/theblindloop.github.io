"""Independent RGB-only inverse for the connected terrain profile question.

This module contains its own panel geometry and measurement routine. It imports
no renderer, prompt, generator, verifier, or latent scene code.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

TEAL = np.asarray((71, 157, 163), dtype=np.float32)
LEVELS = np.asarray((0.22, 0.36, 0.50, 0.64, 0.78), dtype=np.float64)
HEIGHT_SCALE = 380.0


def _teal_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    distance = np.sqrt(np.sum((array - TEAL) ** 2, axis=2))
    # Interior pixels retain the exact fill; the threshold tolerates
    # antialiased upper edges without accepting the paper or navy frame.
    return distance < 48.0


def _measure_height(mask: np.ndarray, center_x: int) -> float:
    height, width = mask.shape
    half_window = 10
    left = max(0, center_x - half_window)
    right = min(width, center_x + half_window + 1)
    column = np.mean(mask[:, left:right], axis=1) > 0.55

    # Find a long teal run ending at the public common floor. The measurement
    # is derived from the input raster rather than from a latent scene.
    runs: list[tuple[int, int]] = []
    start = None
    for y, active in enumerate(column):
        if active and start is None:
            start = y
        if start is not None and (not active or y == height - 1):
            end = y if active and y == height - 1 else y - 1
            runs.append((start, end))
            start = None
    # A clean station has at least ~83 px of teal (the lowest profile plus the
    # floor).  Requiring a substantial run rejects the 30 px floor remnant
    # left when an ablation erases a lower vertical slice, making the entire
    # thick support area causally necessary to the inverse.
    candidates = [run for run in runs if 390 <= run[1] <= 455 and run[1] - run[0] >= 70]
    if not candidates:
        raise ValueError("no connected terrain run reaches the common floor")
    top, bottom = max(candidates, key=lambda run: (run[1] - run[0], run[1]))
    variable_height = max(1.0, float(bottom - top + 1) - 30.0)
    return variable_height / HEIGHT_SCALE


def _profile(mask: np.ndarray, x0: int) -> list[float]:
    centers = [x0 + 45 + 50 * index for index in range(5)]
    measured = [_measure_height(mask, center) for center in centers]
    snapped = [float(LEVELS[np.argmin(np.abs(LEVELS - value))]) for value in measured]
    if any(abs(value - chosen) > 0.055 for value, chosen in zip(measured, snapped)):
        raise ValueError("station height is not one of the visible level bands")
    if sorted(snapped) != sorted(float(value) for value in LEVELS):
        raise ValueError("slot does not contain the five distinct visible bands")
    return snapped


def _error(reference: list[float], panel: list[float]) -> float:
    observed = list(reversed(panel))
    return sum((a - b) ** 2 for a, b in zip(reference, observed)) / len(reference)


def decision_from_image(image: Image.Image) -> str:
    """Return ``A`` or ``B`` using only the supplied final raster."""

    mask = _teal_mask(image)
    if mask.ndim != 2 or mask.shape[1] < 900 or mask.shape[0] < 500:
        raise ValueError("unexpected raster dimensions")
    reference = _profile(mask, 40)
    candidate_a = _profile(mask, 330)
    candidate_b = _profile(mask, 620)
    error_a = _error(reference, candidate_a)
    error_b = _error(reference, candidate_b)
    if not math.isfinite(error_a + error_b):
        raise ValueError("non-finite profile measurement")
    return "A" if error_a < error_b else "B"
