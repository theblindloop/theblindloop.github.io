"""Renderer for a connected-terrain profile correspondence question.

The visible object is a single filled terrain: three adjacent slots share a
continuous floor, and each slot has five broad stations whose upper edge
encodes a height profile.  The question compares the reference profile with
each candidate after reversing the candidate's reading direction.
"""

from __future__ import annotations

import copy
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

RENDERER_VERSION = "ribbon-profile-correspondence-0.1.0"
WIDTH, HEIGHT = 960, 560
PANEL_BOUNDS = ((40, 320), (330, 610), (620, 900))
BASE_Y = 450
BASE_TOP = 420
# The tall silhouette makes every vertical ablation row intersect a station,
# while the profile levels remain comfortably separated in the raster.
PROFILE_SCALE = 380
STATION_INSET = 20
STATION_WIDTH = 50
STATION_COUNT = 5
TERRAIN = (71, 157, 163)
INK = (35, 54, 75)
PAPER = (247, 244, 236)
QUARANTINE_MARGIN = 0.012


def _font(size: int):
    """Use a fixed installed font, with a deterministic Pillow fallback."""

    for name in (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
        "/usr/share/fonts/truetype/liberation2/LiberationSans-Regular.ttf",
    ):
        if Path(name).is_file():
            return ImageFont.truetype(name, size=size)
    return ImageFont.load_default()


def _profile_error(reference: list[float], panel: list[float]) -> float:
    """Mean squared normalized height error after the declared reversal."""

    read_right_to_left = list(reversed(panel))
    return sum((float(a) - float(b)) ** 2 for a, b in zip(reference, read_right_to_left)) / len(reference)


def _slot_profiles(scene: dict) -> dict[int, list[float]]:
    """Return candidates by visible slot, independent of list order."""

    return {
        int(record["slot"]): [float(value) for value in record["profile"]]
        for record in scene["candidate_profiles"]
    }


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene with equalized candidate ink and position cues."""

    rng = random.Random(int(seed))
    near = (int(seed) % 7) == 0
    if near:
        levels = [0.28, 0.33, 0.48, 0.64, 0.78]
    else:
        levels = [0.22, 0.36, 0.50, 0.64, 0.78]
    reference = list(levels)
    rng.shuffle(reference)

    matched_reading = list(reference)
    matched_panel = list(reversed(matched_reading))
    wrong_reading = list(matched_reading)
    if near:
        # Swap the closest pair. This is a visible but finite-resolution
        # boundary case, and keeps the candidate area multiset identical.
        low, high = sorted(range(len(levels)), key=lambda index: levels[index])[:2]
        positions = [matched_reading.index(levels[low]), matched_reading.index(levels[high])]
        wrong_reading[positions[0]], wrong_reading[positions[1]] = (
            wrong_reading[positions[1]],
            wrong_reading[positions[0]],
        )
    else:
        wrong_reading = matched_reading[1:] + matched_reading[:1]
    wrong_panel = list(reversed(wrong_reading))

    match_slot = 1 if rng.random() < 0.5 else 2
    records = [
        {"slot": match_slot, "profile": matched_panel},
        {"slot": 3 - match_slot, "profile": wrong_panel},
    ]
    rng.shuffle(records)  # representational order, not visible panel order
    return {
        "reference_profile": [round(float(value), 6) for value in reference],
        "candidate_profiles": records,
    }


def analytic_gold(scene: dict) -> str:
    """Select the visible slot with the smaller transformed profile error."""

    reference = [float(value) for value in scene["reference_profile"]]
    profiles = _slot_profiles(scene)
    errors = {slot: _profile_error(reference, profile) for slot, profile in profiles.items()}
    return "A" if errors[1] < errors[2] else "B"


def margin(scene: dict) -> float:
    """Runner-up gap between the two aggregate profile errors."""

    reference = [float(value) for value in scene["reference_profile"]]
    profiles = _slot_profiles(scene)
    error_a = _profile_error(reference, profiles[1])
    error_b = _profile_error(reference, profiles[2])
    return abs(error_a - error_b)


def is_quarantined(scene: dict) -> bool:
    """Exclude profiles whose runner-up gap is too small for raster certainty."""

    return margin(scene) < QUARANTINE_MARGIN


def _scaled_box(box: tuple[float, float, float, float], scale: int):
    return tuple(int(round(value * scale)) for value in box)


def _draw_text(draw: ImageDraw.ImageDraw, xy, text: str, font, fill, scale: int) -> None:
    draw.text((int(xy[0] * scale), int(xy[1] * scale)), text, font=font, fill=fill)


def _draw_terrain(draw: ImageDraw.ImageDraw, x0: int, profile: list[float], scale: int) -> None:
    # Station rectangles touch the shared floor and each other, making the
    # decisive evidence one area rather than a collection of marks.
    for index, value in enumerate(profile):
        left = x0 + STATION_INSET + index * STATION_WIDTH
        right = left + STATION_WIDTH
        top = BASE_TOP - float(value) * PROFILE_SCALE
        draw.rectangle(_scaled_box((left, top, right, BASE_Y), scale), fill=TERRAIN)


def render(scene: dict) -> Image.Image:
    """Render one deterministic antialiased RGB PNG."""

    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), PAPER)
    draw = ImageDraw.Draw(image)
    title_font = _font(25 * scale)
    label_font = _font(18 * scale)
    small_font = _font(15 * scale)

    _draw_text(draw, (42, 28), "PROFILE CORRESPONDENCE", title_font, INK, scale)
    _draw_text(draw, (42, 67), "Match the five heights; read candidates from right to left.", small_font, INK, scale)

    # A broad dark board makes the three visual slots one framed world while
    # the teal terrain remains the pixel cue measured by the inverse arm.
    draw.rounded_rectangle(_scaled_box((30, 92, 910, 461), scale), radius=18 * scale, fill=INK)
    draw.rectangle(_scaled_box((39, 101, 901, 452), scale), fill=PAPER)

    labels = ("REFERENCE  L TO R", "CANDIDATE A  R TO L", "CANDIDATE B  R TO L")
    for panel_index, ((x0, x1), label) in enumerate(zip(PANEL_BOUNDS, labels)):
        bbox = draw.textbbox((0, 0), label, font=label_font)
        text_width = bbox[2] - bbox[0]
        _draw_text(draw, ((x0 + x1) / 2 - text_width / (2 * scale), 105), label, label_font, INK, scale)
        if panel_index:
            divider = PANEL_BOUNDS[panel_index - 1][1] + 5
            draw.rectangle(_scaled_box((divider, 101, divider + 6, 416), scale), fill=INK)

    draw.rectangle(_scaled_box((40, BASE_TOP, 900, BASE_Y), scale), fill=TERRAIN)
    profiles_by_slot = _slot_profiles(scene)
    _draw_terrain(draw, PANEL_BOUNDS[0][0], scene["reference_profile"], scale)
    _draw_terrain(draw, PANEL_BOUNDS[1][0], profiles_by_slot[1], scale)
    _draw_terrain(draw, PANEL_BOUNDS[2][0], profiles_by_slot[2], scale)

    # Station guides make the five sampling positions legible without
    # changing the connected fill. They terminate above the common floor.
    for x0, _ in PANEL_BOUNDS:
        for index in range(STATION_COUNT + 1):
            x = x0 + STATION_INSET + index * STATION_WIDTH
            draw.rectangle(_scaled_box((x, 128, x + 2, 416), scale), fill=INK)

    # Repaint the shared floor so all guide lines visibly terminate into it.
    draw.rectangle(_scaled_box((40, 420, 900, 450), scale), fill=TERRAIN)
    draw.rectangle(_scaled_box((40, 447, 900, 451), scale), fill=INK)
    draw.rectangle(_scaled_box((40, 451, 900, 454), scale), fill=PAPER)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Exercise the harmless permutation of slot-addressed records."""

    twin = copy.deepcopy(scene)
    twin["candidate_profiles"] = list(reversed(twin["candidate_profiles"]))
    return [("candidate_record_order", twin)]
