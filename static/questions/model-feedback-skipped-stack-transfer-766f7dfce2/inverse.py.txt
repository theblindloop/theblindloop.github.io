"""Independent pixel-only inverse for the skipped stack-transfer board."""

from __future__ import annotations

import numpy as np
from PIL import Image

CANVAS = (760, 600)
BOARD_W, BOARD_H = 700, 560
BOARD = (247, 248, 244)
CARD = (255, 255, 252)
BORDER = (55, 64, 72)
SKIP = (211, 49, 57)
STACKS = ("A", "B", "C")
BADGES = {
    (206, 91, 57): "A",
    (28, 144, 148): "B",
    (126, 79, 181): "C",
}
SHAPES = {
    (54, 118, 196): "circle",
    (42, 153, 91): "triangle",
    (231, 162, 34): "star",
}


class OracleError(ValueError):
    """Raised when a raster is not one intact, decidable transfer board."""


def _patch_color(array: np.ndarray, x: int, y: int, radius: int = 2) -> tuple[int, int, int]:
    patch = array[y - radius:y + radius + 1, x - radius:x + radius + 1]
    colors = {tuple(int(v) for v in row) for row in patch.reshape(-1, 3)}
    if len(colors) != 1:
        raise OracleError("a required visual marker is damaged or ambiguous")
    return next(iter(colors))


def _simulate(stacks: list[list[str]], moves: list[tuple[str, str]], skip: int | None) -> tuple[list[list[str]], list[str | None]]:
    state = [list(stack) for stack in stacks]
    moved: list[str | None] = []
    for index, (source, destination) in enumerate(moves):
        if index == skip:
            moved.append(None)
            continue
        si, di = STACKS.index(source), STACKS.index(destination)
        if not state[si]:
            raise OracleError("a visible transfer tries to use an empty stack")
        tile = state[si].pop()
        state[di].append(tile)
        moved.append(tile)
    return state, moved


def decision_from_image(image: Image.Image) -> str:
    """Decode visible stacks and cards, delete the marked event, and replay."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if (array.shape[1], array.shape[0]) != CANVAS:
        raise OracleError("unexpected canvas size")
    mask = np.all(array == np.asarray(BORDER, dtype=np.uint8), axis=2)
    points = np.argwhere(mask)
    if not len(points):
        raise OracleError("transfer board is absent")
    top, left = points.min(axis=0)
    bottom, right = points.max(axis=0)
    if int(right - left + 1) != BOARD_W or int(bottom - top + 1) != BOARD_H:
        raise OracleError("transfer board has the wrong extent")
    ax, ay = int(left), int(top)
    if not (
        np.all(mask[ay, ax:ax + BOARD_W])
        and np.all(mask[ay + BOARD_H - 1, ax:ax + BOARD_W])
        and np.all(mask[ay:ay + BOARD_H, ax])
        and np.all(mask[ay:ay + BOARD_H, ax + BOARD_W - 1])
    ):
        raise OracleError("transfer board border is broken")

    stacks: list[list[str]] = []
    for xoff in (145, 350, 555):
        stack = []
        for level in range(3):
            cy = ay + 207 - 39 * level
            color = _patch_color(array, ax + xoff, cy)
            if color not in SHAPES:
                raise OracleError("a stack tile has an unknown shape marker")
            stack.append(SHAPES[color])
        stacks.append(stack)
    if sorted(tile for stack in stacks for tile in stack) != sorted(list(SHAPES.values()) * 3):
        raise OracleError("the initial shape inventory is invalid")

    moves: list[tuple[str, str]] = []
    skip_hits: list[int] = []
    card_xs = (34, 260, 486)
    card_ys = (278, 401)
    for index in range(6):
        x = ax + card_xs[index % 3]
        y = ay + card_ys[index // 3]
        source_color = _patch_color(array, x + 30, y + 66)
        destination_color = _patch_color(array, x + 113, y + 66)
        if source_color not in BADGES or destination_color not in BADGES:
            raise OracleError("a transfer endpoint badge is missing")
        source, destination = BADGES[source_color], BADGES[destination_color]
        if source == destination:
            raise OracleError("a transfer must use different stacks")
        moves.append((source, destination))
        skip_color = _patch_color(array, x + 114, y + 8, radius=1)
        if skip_color == SKIP:
            skip_hits.append(index)
        elif skip_color != CARD:
            raise OracleError("a skip marker is damaged")
    if len(skip_hits) != 1:
        raise OracleError("exactly one transfer card must be skipped")
    skip_index = skip_hits[0]

    factual, factual_moved = _simulate(stacks, moves, None)
    counterfactual, cf_moved = _simulate(stacks, moves, skip_index)
    if not factual[2] or not counterfactual[2]:
        raise OracleError("stack C has no final top tile")
    if factual[2][-1] == counterfactual[2][-1]:
        raise OracleError("the skip does not change the requested result")
    if sum(factual_moved[i] != cf_moved[i] for i in range(skip_index + 1, 6)) < 2:
        raise OracleError("the skip does not alter two later moved shapes")
    answer = counterfactual[2][-1]
    if answer not in SHAPES.values():
        raise OracleError("unknown final shape")
    return answer
