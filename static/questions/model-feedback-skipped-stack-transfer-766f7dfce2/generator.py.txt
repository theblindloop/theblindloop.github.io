"""Deterministic renderer for an intervened top-tile transfer program."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw, ImageFont

CANVAS = (760, 600)
BOARD_W, BOARD_H = 700, 560
ANSWERS = ("circle", "triangle", "star")
STACKS = ("A", "B", "C")

RGB = {
    "white": (255, 255, 255),
    "board": (247, 248, 244),
    "card": (255, 255, 252),
    "ink": (37, 43, 49),
    "muted": (103, 111, 117),
    "border": (55, 64, 72),
    "skip": (211, 49, 57),
    "A": (206, 91, 57),
    "B": (28, 144, 148),
    "C": (126, 79, 181),
    "circle": (54, 118, 196),
    "triangle": (42, 153, 91),
    "star": (231, 162, 34),
}

FONT = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
BOLD = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"


def _simulate(stacks: list[list[str]], moves: list[list[str]], skip: int | None) -> tuple[list[list[str]], list[str | None]]:
    state = [list(stack) for stack in stacks]
    moved: list[str | None] = []
    for index, (source, destination) in enumerate(moves):
        if index == skip:
            moved.append(None)
            continue
        si, di = STACKS.index(source), STACKS.index(destination)
        if not state[si]:
            raise ValueError("a transfer source is empty")
        tile = state[si].pop()
        state[di].append(tile)
        moved.append(tile)
    return state, moved


def _causal(scene: dict) -> tuple[list[list[str]], list[list[str]], list[str | None], list[str | None]]:
    stacks = [[str(tile) for tile in stack] for stack in scene["stacks"]]
    moves = [[str(value) for value in move] for move in scene["moves"]]
    skip_index = int(scene["skip_index"])
    factual, factual_moved = _simulate(stacks, moves, None)
    counterfactual, counterfactual_moved = _simulate(stacks, moves, skip_index)
    return factual, counterfactual, factual_moved, counterfactual_moved


def sample_scene(seed: int) -> dict:
    """Sample a balanced, visibly causal six-card stack program."""

    rng = random.Random((int(seed) ^ 0x57ACCA11) & 0xFFFFFFFFFFFFFFFF)
    wanted = ANSWERS[int(seed) % len(ANSWERS)]
    pairs = [(a, b) for a in STACKS for b in STACKS if a != b]
    for _ in range(50000):
        tiles = list(ANSWERS) * 3
        rng.shuffle(tiles)
        stacks = [tiles[0:3], tiles[3:6], tiles[6:9]]
        moves = [list(pairs[rng.randrange(len(pairs))]) for _ in range(6)]
        skip_index = rng.randrange(3)
        probe = {
            "stacks": stacks,
            "moves": moves,
            "skip_index": skip_index,
            "anchor_x": 0,
            "anchor_y": 0,
        }
        try:
            factual, counterfactual, factual_moved, cf_moved = _causal(probe)
        except ValueError:
            continue
        if not counterfactual[2] or counterfactual[2][-1] != wanted:
            continue
        if not factual[2] or factual[2][-1] == counterfactual[2][-1]:
            continue
        later_differences = sum(
            factual_moved[index] != cf_moved[index]
            for index in range(skip_index + 1, 6)
        )
        if later_differences < 2:
            continue
        return {
            "stacks": stacks,
            "moves": moves,
            "skip_index": skip_index,
            "anchor_x": rng.randint(24, 36),
            "anchor_y": rng.randint(18, 22),
        }
    raise RuntimeError("could not sample a causally deep stack program")


def analytic_gold(scene: dict) -> str:
    factual, counterfactual, factual_moved, cf_moved = _causal(scene)
    skip_index = int(scene["skip_index"])
    if not counterfactual[2]:
        raise ValueError("stack C is empty")
    answer = counterfactual[2][-1]
    if answer not in ANSWERS or factual[2][-1] == answer:
        raise ValueError("the intervention does not change the requested result")
    if sum(factual_moved[i] != cf_moved[i] for i in range(skip_index + 1, 6)) < 2:
        raise ValueError("the intervention does not propagate through two later transfers")
    return answer


def margin(scene: dict) -> float:
    try:
        return 1.0 if analytic_gold(scene) in ANSWERS else 0.0
    except (KeyError, TypeError, ValueError):
        return 0.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(BOLD if bold else FONT, size)


def _text(draw: ImageDraw.ImageDraw, xy: tuple[int, int], value: str, size: int, *,
          fill: tuple[int, int, int] | None = None, bold: bool = False,
          anchor: str = "mm") -> None:
    draw.text(xy, value, font=_font(size, bold), fill=fill or RGB["ink"], anchor=anchor)


def _shape(draw: ImageDraw.ImageDraw, center: tuple[int, int], kind: str) -> None:
    x, y = center
    color = RGB[kind]
    if kind == "circle":
        draw.ellipse((x - 13, y - 13, x + 13, y + 13), fill=color, outline=RGB["ink"], width=2)
    elif kind == "triangle":
        draw.polygon(((x, y - 15), (x - 15, y + 12), (x + 15, y + 12)), fill=color)
        draw.line(((x, y - 15), (x - 15, y + 12), (x + 15, y + 12), (x, y - 15)), fill=RGB["ink"], width=2)
    elif kind == "star":
        points = []
        for index in range(10):
            radius = 16 if index % 2 == 0 else 7
            angle = -math.pi / 2 + index * math.pi / 5
            points.append((x + radius * math.cos(angle), y + radius * math.sin(angle)))
        draw.polygon(points, fill=color)
        draw.line(points + [points[0]], fill=RGB["ink"], width=2, joint="curve")
    else:
        raise ValueError(f"unknown shape {kind!r}")


def _badge(draw: ImageDraw.ImageDraw, center: tuple[int, int], label: str) -> None:
    x, y = center
    draw.rounded_rectangle((x - 27, y - 19, x + 27, y + 19), radius=7,
                           fill=RGB[label], outline=RGB["ink"], width=2)
    _text(draw, center, label, 21, fill=RGB["white"], bold=True)


def render(scene: dict) -> Image.Image:
    stacks = [[str(tile) for tile in stack] for stack in scene["stacks"]]
    moves = [[str(value) for value in move] for move in scene["moves"]]
    skip_index = int(scene["skip_index"])
    if len(stacks) != 3 or any(len(stack) != 3 for stack in stacks):
        raise ValueError("exactly three stacks of three tiles are required")
    if len(moves) != 6 or not 0 <= skip_index < 6:
        raise ValueError("six moves and one skip are required")
    analytic_gold(scene)
    ax, ay = int(scene["anchor_x"]), int(scene["anchor_y"])
    image = Image.new("RGB", CANVAS, RGB["white"])
    draw = ImageDraw.Draw(image)
    draw.rectangle((ax, ay, ax + BOARD_W - 1, ay + BOARD_H - 1),
                   fill=RGB["board"], outline=RGB["border"], width=4)
    _text(draw, (ax + 350, ay + 25), "TOP-TILE TRANSFER", 21, bold=True)
    _text(draw, (ax + 350, ay + 52), "Start here. Each card moves one TOP tile.", 16, fill=RGB["muted"])

    stack_xs = (145, 350, 555)
    for stack_index, (label, xoff) in enumerate(zip(STACKS, stack_xs, strict=True)):
        _text(draw, (ax + xoff, ay + 78), f"STACK {label}", 17, bold=True)
        draw.rounded_rectangle((ax + xoff - 62, ay + 91, ax + xoff + 62, ay + 231),
                               radius=8, fill=RGB["card"], outline=RGB[label], width=4)
        for level, kind in enumerate(stacks[stack_index]):
            cy = ay + 207 - 39 * level
            draw.rounded_rectangle((ax + xoff - 39, cy - 17, ax + xoff + 39, cy + 17),
                                   radius=5, fill=RGB["white"], outline=RGB["ink"], width=2)
            _shape(draw, (ax + xoff, cy), kind)
        _text(draw, (ax + xoff + 50, ay + 112), "TOP", 12, fill=RGB["muted"], bold=True)

    _text(draw, (ax + 350, ay + 254), "DO CARDS 1 -> 6     RED X = SKIP", 17, bold=True)
    card_xs = (34, 260, 486)
    card_ys = (278, 401)
    for index, (source, destination) in enumerate(moves):
        x = ax + card_xs[index % 3]
        y = ay + card_ys[index // 3]
        draw.rounded_rectangle((x, y, x + 180, y + 100), radius=9,
                               fill=RGB["card"], outline=RGB["border"], width=3)
        draw.ellipse((x + 9, y + 8, x + 39, y + 38), fill=RGB["ink"])
        _text(draw, (x + 24, y + 23), str(index + 1), 16, fill=RGB["white"], bold=True)
        _badge(draw, (x + 50, y + 66), source)
        _text(draw, (x + 90, y + 66), "->", 24, bold=True)
        _badge(draw, (x + 133, y + 66), destination)
        if index == skip_index:
            draw.rounded_rectangle((x + 111, y + 5, x + 174, y + 31), radius=5, fill=RGB["skip"])
            _text(draw, (x + 143, y + 18), "SKIP", 13, fill=RGB["white"], bold=True)
            draw.line((x + 103, y + 4, x + 177, y + 36), fill=RGB["skip"], width=6)
            draw.line((x + 177, y + 4, x + 103, y + 36), fill=RGB["skip"], width=6)
    return image
