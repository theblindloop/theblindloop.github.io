"""Scene sampler and raster renderer for fragmented curve inflection."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

SCALE = 3
N_FRAGMENTS = 13


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    model = "C" if seed % 2 == 0 else "S"
    black_level = rng.randint(20, 52)
    gray_level = rng.randint(142, 178)
    distractors = []
    for _ in range(11):
        distractors.append({
            "x": rng.uniform(45, 467),
            "y": rng.uniform(45, 467),
            "angle_deg": rng.uniform(0, 180),
            "length": rng.uniform(10, 16),
        })
    return {
        "seed": seed,
        "model": model,
        "amplitude": rng.uniform(54.0, 72.0),
        "bend_sign": -1 if rng.random() < 0.5 else 1,
        "rotation_deg": rng.uniform(0, 360),
        "center_x": rng.uniform(238, 274),
        "center_y": rng.uniform(238, 274),
        "dash_length": rng.uniform(12.0, 15.0),
        "dash_width": rng.choice([5, 6]),
        "black_rgb": [black_level, black_level, black_level + rng.randint(0, 5)],
        "gray_rgb": [gray_level, gray_level, gray_level + rng.randint(-3, 3)],
        "distractors": distractors,
    }


def _centerline(model: str, amplitude: float, bend_sign: int, u: float) -> tuple[float, float, float, float]:
    x = 162.0 * u
    if model == "C":
        y = bend_sign * amplitude * (u * u - 0.38)
        dx, dy = 162.0, bend_sign * 2.0 * amplitude * u
    elif model == "S":
        y = bend_sign * amplitude * (1.55 * u**3 - 0.58 * u)
        dx, dy = 162.0, bend_sign * amplitude * (4.65 * u * u - 0.58)
    else:
        raise ValueError("unknown model")
    return x, y, dx, dy


def _rotate(x: float, y: float, degrees: float) -> tuple[float, float]:
    a = math.radians(degrees)
    return x * math.cos(a) - y * math.sin(a), x * math.sin(a) + y * math.cos(a)


def _draw_dash(draw: ImageDraw.ImageDraw, x: float, y: float, angle: float, length: float, width: float, fill: tuple[int, int, int]) -> None:
    ux, uy = math.cos(angle), math.sin(angle)
    p1 = ((x - ux * length / 2) * SCALE, (y - uy * length / 2) * SCALE)
    p2 = ((x + ux * length / 2) * SCALE, (y + uy * length / 2) * SCALE)
    draw.line([p1, p2], fill=fill, width=max(1, round(width * SCALE)))


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (512 * SCALE, 512 * SCALE), (250, 249, 246))
    draw = ImageDraw.Draw(image)
    gray = tuple(scene["gray_rgb"])
    for item in scene["distractors"]:
        _draw_dash(draw, item["x"], item["y"], math.radians(item["angle_deg"]), item["length"], scene["dash_width"], gray)
    black = tuple(scene["black_rgb"])
    for index in range(N_FRAGMENTS):
        u = -1.0 + 2.0 * index / (N_FRAGMENTS - 1)
        x, y, dx, dy = _centerline(scene["model"], scene["amplitude"], scene["bend_sign"], u)
        x, y = _rotate(x, y, scene["rotation_deg"])
        tx, ty = _rotate(dx, dy, scene["rotation_deg"])
        _draw_dash(draw, x + scene["center_x"], y + scene["center_y"], math.atan2(ty, tx), scene["dash_length"], scene["dash_width"], black)
    return image.resize((512, 512), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return scene["model"]


def margin(scene: dict) -> float:
    # Normalized curvature-product separation from the zero-inflection boundary.
    amplitude = float(scene["amplitude"])
    if scene["model"] == "C":
        return round((2.0 * amplitude / (162.0**2)) ** 2 * 1_000_000.0, 9)
    return round((7.2 * amplitude / (162.0**2)) ** 2 * 1_000_000.0, 9)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.08


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
