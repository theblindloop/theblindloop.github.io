"""Independent pixel-only inverse arm."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    seen = np.zeros(mask.shape, dtype=bool)
    result = []
    h, w = mask.shape
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        pts = []
        while stack:
            yy, xx = stack.pop()
            pts.append((yy, xx))
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if 22 <= len(pts) <= 180:
            result.append(np.asarray(pts, dtype=float))
    return result


def _black_centers(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    mask = arr.mean(axis=2) < 92
    comps = _components(mask)
    if len(comps) != 13:
        raise ValueError(f"expected 13 black fragments, found {len(comps)}")
    return np.asarray([[c[:, 1].mean(), c[:, 0].mean()] for c in comps])


def decision_from_image(image: Image.Image) -> str:
    points = _black_centers(image)
    centered = points - points.mean(axis=0)
    _, _, vh = np.linalg.svd(centered, full_matrices=False)
    x = centered @ vh[0]
    y = centered @ vh[1]
    order = np.argsort(x)
    x, y = x[order], y[order]
    scale = max(abs(x[0]), abs(x[-1]))
    if scale < 80:
        raise ValueError("fragment span is too small")
    xn = x / scale
    coef = np.polyfit(xn, y, 3)
    left_curvature = 6.0 * coef[0] * -0.72 + 2.0 * coef[1]
    right_curvature = 6.0 * coef[0] * 0.72 + 2.0 * coef[1]
    product = left_curvature * right_curvature
    if abs(product) < 20.0:
        raise ValueError("ambiguous fitted curvature product")
    return "S" if product < 0 else "C"
