"""Deterministic renderer and analytic arm for a cross-coupled tile walk."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

CANVAS = (660, 660)
N, CELL, X0, Y0, STEPS = 5, 100, 80, 80, 6
DIRS = ((0, -1), (1, 0), (0, 1), (-1, 0))
TOKEN_COLORS = ((220, 45, 55), (25, 105, 220))  # red, blue
GOAL_COLORS = {"amber": (235, 150, 20), "cyan": (0, 165, 185), "violet": (125, 70, 195)}


def _inside(cell):
    return 0 <= cell[0] < N and 0 <= cell[1] < N


def _step(cell, direction):
    dx, dy = DIRS[direction]
    return cell[0] + dx, cell[1] + dy


def _trace(instructions, red_start, blue_start, rule="correct"):
    red, blue = tuple(red_start), tuple(blue_start)
    states, reds = [(red, blue)], [red]
    for _ in range(STEPS):
        old_red, old_blue = red, blue
        if rule == "self_read":
            red = _step(old_red, instructions[old_red[1]][old_red[0]][0])
            blue = _step(old_blue, instructions[old_blue[1]][old_blue[0]][1])
        else:
            red = _step(old_red, instructions[old_blue[1]][old_blue[0]][0])
            red_for_blue = red if rule == "red_first" else old_red
            blue = _step(old_blue, instructions[red_for_blue[1]][red_for_blue[0]][1])
        if not _inside(red) or not _inside(blue):
            raise ValueError("walk leaves grid")
        if (red, blue) in states:
            raise ValueError("joint state repeats")
        states.append((red, blue)); reds.append(red)
    return red, blue, states, reds


def _construct(rng):
    """Assign a causal joint path, then retain scenes with two distinct error endpoints."""
    for _attempt in range(12000):
        red = (rng.randrange(1, N-1), rng.randrange(1, N-1))
        blue = (rng.randrange(1, N-1), rng.randrange(1, N-1))
        if red == blue:
            continue
        starts = (red, blue)
        assignments = {}
        states, reds = [(red, blue)], [red]

        def visit(r, b, depth):
            if depth == STEPS:
                return True
            kr, kb = (b, 0), (r, 1)
            rdirs = [assignments[kr]] if kr in assignments else list(range(4))
            bdirs = [assignments[kb]] if kb in assignments else list(range(4))
            rng.shuffle(rdirs); rng.shuffle(bdirs)
            for dr in rdirs:
                nr = _step(r, dr)
                if not _inside(nr):
                    continue
                for db in bdirs:
                    nb = _step(b, db)
                    if not _inside(nb) or (nr, nb) in states:
                        continue
                    prior_r, prior_b = assignments.get(kr), assignments.get(kb)
                    assignments[kr], assignments[kb] = dr, db
                    states.append((nr, nb)); reds.append(nr)
                    if visit(nr, nb, depth + 1):
                        return True
                    states.pop(); reds.pop()
                    if prior_r is None: assignments.pop(kr, None)
                    else: assignments[kr] = prior_r
                    if prior_b is None: assignments.pop(kb, None)
                    else: assignments[kb] = prior_b
            return False

        if not visit(red, blue, 0):
            continue
        instructions = [[[rng.randrange(4), rng.randrange(4)] for _ in range(N)] for _ in range(N)]
        for ((x, y), token), direction in assignments.items():
            instructions[y][x][token] = direction
        try:
            correct = _trace(instructions, *starts)
            self_read = _trace(instructions, *starts, rule="self_read")
            red_first = _trace(instructions, *starts, rule="red_first")
        except (ValueError, IndexError):
            continue
        endpoints = [correct[0], self_read[0], red_first[0]]
        if len(set(endpoints)) != 3:
            continue
        if correct[0] in correct[3][:-1] or any(p in correct[3][:-1] for p in endpoints[1:]):
            continue
        return instructions, starts, endpoints
    raise RuntimeError("could not construct cross-coupled walk")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0xC20A771E)
    instructions, (red_start, blue_start), endpoints = _construct(rng)
    answer = ("amber", "cyan", "violet")[int(seed) % 3]
    others = [name for name in GOAL_COLORS if name != answer]
    rng.shuffle(others)
    goals = {answer: list(endpoints[0]), others[0]: list(endpoints[1]), others[1]: list(endpoints[2])}
    return {"instructions": instructions, "red_start": list(red_start),
            "blue_start": list(blue_start), "goals": goals}


def _simulate(scene, rule="correct"):
    return _trace(scene["instructions"], scene["red_start"], scene["blue_start"], rule)


def analytic_gold(scene):
    endpoint = _simulate(scene)[0]
    matches = [name for name, cell in scene["goals"].items() if tuple(cell) == endpoint]
    if len(matches) != 1:
        raise ValueError("red endpoint does not have exactly one goal")
    return matches[0]


def margin(scene):
    return 1.0 if not is_quarantined(scene) else 0.0


def is_quarantined(scene):
    try:
        ins = scene["instructions"]
        if len(ins) != N or any(len(row) != N for row in ins): return True
        if any(len(pair) != 2 or any(d not in range(4) for d in pair) for row in ins for pair in row): return True
        starts = tuple(scene["red_start"]), tuple(scene["blue_start"])
        if any(not _inside(s) for s in starts) or starts[0] == starts[1]: return True
        correct = _simulate(scene)
        wrong_a = _simulate(scene, "self_read")
        wrong_b = _simulate(scene, "red_first")
        endpoints = [correct[0], wrong_a[0], wrong_b[0]]
        if len(set(endpoints)) != 3: return True
        if correct[0] in correct[3][:-1] or any(p in correct[3][:-1] for p in endpoints[1:]): return True
        goal_cells = [tuple(cell) for cell in scene["goals"].values()]
        return set(scene["goals"]) != set(GOAL_COLORS) or set(goal_cells) != set(endpoints)
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene):
    return []


def _arrow(draw, center, direction, color):
    cx, cy = center
    if direction == 0:
        draw.line((cx, cy+12, cx, cy-5), fill=color, width=8); draw.polygon(((cx,cy-17),(cx-10,cy-3),(cx+10,cy-3)), fill=color)
    elif direction == 1:
        draw.line((cx-12,cy,cx+5,cy), fill=color, width=8); draw.polygon(((cx+17,cy),(cx+3,cy-10),(cx+3,cy+10)), fill=color)
    elif direction == 2:
        draw.line((cx,cy-12,cx,cy+5), fill=color, width=8); draw.polygon(((cx,cy+17),(cx-10,cy+3),(cx+10,cy+3)), fill=color)
    else:
        draw.line((cx+12,cy,cx-5,cy), fill=color, width=8); draw.polygon(((cx-17,cy),(cx-3,cy-10),(cx-3,cy+10)), fill=color)


def render(scene):
    image = Image.new("RGB", CANVAS, (248, 248, 246)); draw = ImageDraw.Draw(image)
    for y in range(N):
        for x in range(N):
            fill = (255,255,255) if (x+y)%2 == 0 else (243,246,248)
            draw.rectangle((X0+x*CELL,Y0+y*CELL,X0+(x+1)*CELL,Y0+(y+1)*CELL), fill=fill)
    for name, cell in scene["goals"].items():
        x, y = cell
        draw.rectangle((X0+x*CELL+5,Y0+y*CELL+5,X0+(x+1)*CELL-5,Y0+(y+1)*CELL-5), outline=GOAL_COLORS[name], width=8)
    for i in range(N+1):
        draw.line((X0+i*CELL,Y0,X0+i*CELL,Y0+N*CELL), fill=(45,50,55), width=3)
        draw.line((X0,Y0+i*CELL,X0+N*CELL,Y0+i*CELL), fill=(45,50,55), width=3)
    for y in range(N):
        for x in range(N):
            cx, cy = X0+x*CELL+CELL//2, Y0+y*CELL+CELL//2
            _arrow(draw, (cx,cy-22), scene["instructions"][y][x][0], TOKEN_COLORS[0])
            _arrow(draw, (cx,cy+22), scene["instructions"][y][x][1], TOKEN_COLORS[1])
    rx, ry = scene["red_start"]
    bx, by = scene["blue_start"]
    draw.ellipse((X0+rx*CELL+7,Y0+ry*CELL+7,X0+rx*CELL+29,Y0+ry*CELL+29), fill=TOKEN_COLORS[0], outline="black", width=3)
    draw.ellipse((X0+bx*CELL+71,Y0+by*CELL+7,X0+bx*CELL+93,Y0+by*CELL+29), fill=TOKEN_COLORS[1], outline="black", width=3)
    return image
