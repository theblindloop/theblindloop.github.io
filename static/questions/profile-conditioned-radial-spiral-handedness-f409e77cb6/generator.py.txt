"""Deterministic latent sampler and Pillow renderer for a radial spiral constellation."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw


WIDTH = 512
HEIGHT = 512
QUARANTINE_GAP = 0.18

PALETTES = (
    ((249, 247, 240), (29, 35, 43), (145, 151, 158)),
    ((245, 249, 250), (22, 43, 52), (142, 153, 158)),
    ((250, 246, 248), (47, 30, 42), (158, 145, 154)),
    ((247, 249, 244), (32, 43, 28), (149, 157, 145)),
)


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; consecutive seeds alternate answer identity."""
    rng = random.Random(int(seed) * 1_000_003 + 97)
    direction = 1 if int(seed) % 2 else -1  # screen-polar positive is clockwise
    count = (16, 18, 20)[(int(seed) // 2) % 3]
    near_boundary = int(seed) % 7 == 0
    step = rng.uniform(0.055, 0.070) if near_boundary else rng.uniform(0.235, 0.305)
    phase = rng.uniform(-math.pi, math.pi)
    center = [256 + rng.randint(-13, 13), 256 + rng.randint(-13, 13)]
    radial_start = rng.uniform(38.0, 43.0)
    radial_end = rng.uniform(190.0, 199.0)
    angular_jitter = 0.006 if near_boundary else 0.018
    points: list[list[int]] = []
    for index in range(count):
        fraction = index / (count - 1)
        radius = radial_start + fraction * (radial_end - radial_start)
        radius += rng.uniform(-0.55, 0.55)
        angle = phase + direction * step * index + rng.uniform(-angular_jitter, angular_jitter)
        x = int(round(center[0] + radius * math.cos(angle)))
        y = int(round(center[1] + radius * math.sin(angle)))
        points.append([x, y])
    background, ink, marker = PALETTES[(int(seed) // 6) % len(PALETTES)]
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "center": center,
        "points": points,
        "dot_radius": 3 if near_boundary else 5,
        "marker_radius": 12,
        "background": list(background),
        "ink_color": list(ink),
        "marker_color": list(marker),
    }


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(int(value) for value in scene["background"])
    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)
    cx, cy = (int(value) for value in scene["center"])
    marker_radius = int(scene["marker_radius"])
    marker_color = tuple(int(value) for value in scene["marker_color"])
    draw.ellipse(
        (cx - marker_radius, cy - marker_radius, cx + marker_radius, cy + marker_radius),
        fill=marker_color,
    )
    inner = marker_radius - 5
    draw.ellipse((cx - inner, cy - inner, cx + inner, cy + inner), fill=background)
    dot_radius = int(scene["dot_radius"])
    ink = tuple(int(value) for value in scene["ink_color"])
    for point in scene["points"]:
        x, y = (int(value) for value in point)
        draw.ellipse(
            (x - dot_radius, y - dot_radius, x + dot_radius, y + dot_radius),
            fill=ink,
        )
    return image


def _angular_slope(scene: dict) -> float:
    cx, cy = (float(value) for value in scene["center"])
    ordered = sorted(
        ((float(x), float(y)) for x, y in scene["points"]),
        key=lambda point: math.hypot(point[0] - cx, point[1] - cy),
    )
    unwrapped: list[float] = []
    for x, y in ordered:
        angle = math.atan2(y - cy, x - cx)
        if unwrapped:
            while angle - unwrapped[-1] > math.pi:
                angle -= 2.0 * math.pi
            while angle - unwrapped[-1] < -math.pi:
                angle += 2.0 * math.pi
        unwrapped.append(angle)
    mean_index = (len(unwrapped) - 1) / 2.0
    mean_angle = sum(unwrapped) / len(unwrapped)
    numerator = sum(
        (index - mean_index) * (angle - mean_angle)
        for index, angle in enumerate(unwrapped)
    )
    denominator = sum((index - mean_index) ** 2 for index in range(len(unwrapped)))
    return numerator / denominator


def analytic_gold(scene: dict) -> str:
    return "clockwise" if _angular_slope(scene) > 0.0 else "counterclockwise"


def margin(scene: dict) -> float:
    """Top-minus-runner-up directional score gap, in radians per radial step."""
    return 2.0 * abs(_angular_slope(scene))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["points"] = list(reversed(scene["points"]))
    return [("reverse_unordered_point_list", twin)]
