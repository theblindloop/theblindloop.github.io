"""Independent PNG-only inverse arm for radial spiral handedness."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


def _luminance(rgb: np.ndarray) -> np.ndarray:
    return 0.2126 * rgb[..., 0] + 0.7152 * rgb[..., 1] + 0.0722 * rgb[..., 2]


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Return 8-connected foreground components as arrays of (y, x) pixels."""
    remaining = {tuple(value) for value in np.argwhere(mask)}
    components: list[np.ndarray] = []
    neighbors = tuple(
        (dy, dx)
        for dy in (-1, 0, 1)
        for dx in (-1, 0, 1)
        if dy or dx
    )
    while remaining:
        start = remaining.pop()
        stack = [start]
        pixels = [start]
        while stack:
            y, x = stack.pop()
            for dy, dx in neighbors:
                candidate = (y + dy, x + dx)
                if candidate in remaining:
                    remaining.remove(candidate)
                    stack.append(candidate)
                    pixels.append(candidate)
        components.append(np.asarray(pixels, dtype=np.float64))
    return components


def _measure(image: Image.Image) -> tuple[np.ndarray, list[np.ndarray]]:
    rgb = np.asarray(image.convert("RGB"), dtype=np.float64)
    light = _luminance(rgb)
    dark_mask = light < 90.0
    marker_mask = (light > 110.0) & (light < 195.0)
    if int(marker_mask.sum()) < 80:
        raise ValueError("gray center marker is absent")
    marker_yx = np.argwhere(marker_mask).mean(axis=0)
    dots = []
    for component in _components(dark_mask):
        if 20 <= len(component) <= 140:
            dots.append(component.mean(axis=0))
    if len(dots) < 10:
        raise ValueError(f"only {len(dots)} dark dot components were recovered")
    return marker_yx, dots


def decision_from_image(image: Image.Image) -> str:
    center_yx, dot_centers_yx = _measure(image)
    cy, cx = center_yx
    ordered = sorted(
        dot_centers_yx,
        key=lambda point: math.hypot(float(point[1]) - cx, float(point[0]) - cy),
    )
    angles = np.asarray(
        [math.atan2(float(point[0]) - cy, float(point[1]) - cx) for point in ordered],
        dtype=np.float64,
    )
    unwrapped = np.unwrap(angles)
    indices = np.arange(len(unwrapped), dtype=np.float64)
    slope = float(np.polyfit(indices, unwrapped, 1)[0])
    if abs(slope) < 0.01:
        raise ValueError("angular trend is pixel-ambiguous")
    return "clockwise" if slope > 0.0 else "counterclockwise"
