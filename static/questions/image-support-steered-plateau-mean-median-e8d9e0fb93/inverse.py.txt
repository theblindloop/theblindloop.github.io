"""Independent final-PNG-only inverse for plateau mean versus median.

The oracle finds the sole broad saturated region, validates a solid common-base
lamina, divides its recovered extent into twelve equal-width plateaus, measures
one colored run height per plateau, and compares the arithmetic mean with the
even-sample median.  It uses no scene data, renderer values, file metadata, or
fixed raster coordinates.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

N_PLATEAUS = 12


def _filled_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    high = array.max(axis=2)
    low = array.min(axis=2)
    chroma = high - low
    # The evidence is a moderately dark saturated fill; paper, rules, type,
    # and ticks are neutral.  This supports every declared palette twin.
    return (chroma >= 48) & (high >= 105) & (low <= 175)


def _measure_heights(image: Image.Image) -> list[int]:
    mask = _filled_mask(image)
    ys, xs = np.nonzero(mask)
    if len(xs) < image.width * image.height * 0.30:
        raise ValueError("filled plateau lamina is missing or materially erased")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    width = x1 - x0 + 1
    height = y1 - y0 + 1
    if width < image.width * 0.76 or height < image.height * 0.58:
        raise ValueError("filled lamina has implausible extent")
    if width % N_PLATEAUS:
        raise ValueError("filled width cannot be divided into twelve equal plateaus")

    sub = mask[:, x0 : x1 + 1]
    if not bool(sub.any(axis=0).all()):
        raise ValueError("filled lamina has a severed horizontal span")
    tops = np.argmax(sub, axis=0)
    bottoms = sub.shape[0] - 1 - np.argmax(sub[::-1], axis=0)
    counts = sub.sum(axis=0).astype(np.int32)
    full_runs = bottoms - tops + 1
    if int(np.max(full_runs - counts)) > 1:
        raise ValueError("filled lamina contains an internal erasure")
    if int(bottoms.max() - bottoms.min()) > 1:
        raise ValueError("common baseline is incomplete")

    plateau_width = width // N_PLATEAUS
    if plateau_width < 28:
        raise ValueError("plateaus are too narrow for stable measurement")
    heights: list[int] = []
    for index in range(N_PLATEAUS):
        values = counts[index * plateau_width : (index + 1) * plateau_width]
        if int(values.max() - values.min()) > 1:
            raise ValueError("a plateau top is damaged or not level")
        heights.append(int(np.rint(np.median(values))))
    return heights


def decision_from_image(image: Image.Image) -> str:
    heights = _measure_heights(image)
    ordered = sorted(heights)
    mean = sum(heights) / N_PLATEAUS
    median = (ordered[5] + ordered[6]) / 2.0
    if mean == median:
        raise ValueError("pixel-derived mean and median tie")
    return "mean" if mean > median else "median"
