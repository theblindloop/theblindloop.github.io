"""Deterministic renderer and analytic arm for plateau mean versus median."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 640
HEIGHT = 480
N_PLATEAUS = 12
PLATEAU_WIDTH = 44
LAMINA_WIDTH = N_PLATEAUS * PLATEAU_WIDTH
CENTER_HEIGHT = 280
EXTREME_DEVIATION = 70
QUARANTINE_PX = 3.0
BACKGROUND = (247, 245, 239)
INK = (37, 43, 49)
GRID = (213, 212, 205)
PALETTE = (
    (20, 132, 158),
    (38, 132, 96),
    (70, 91, 170),
    (190, 88, 53),
    (151, 74, 145),
    (187, 119, 24),
)


def _font(size: int, *, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _four_parts(total: int, rng: random.Random) -> list[int]:
    """Return four balanced positive integers summing to ``total``."""

    base, remainder = divmod(total, 4)
    parts = [base + (1 if i < remainder else 0) for i in range(4)]
    rng.shuffle(parts)
    for _ in range(7):
        donor, receiver = rng.sample(range(4), 2)
        maximum = min(4, parts[donor] - 4, 68 - parts[receiver])
        if maximum > 0:
            amount = rng.randint(1, maximum)
            parts[donor] -= amount
            parts[receiver] += amount
    assert sum(parts) == total and min(parts) > 0 and max(parts) < EXTREME_DEVIATION
    return parts


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample cue-balanced complementary scene pairs.

    Both members have mean height 280.  The even member has seven negative
    deviations and five positive deviations, so its median is below its mean.
    The odd member negates every deviation in place.  This flips the decision
    while preserving total colored area, range, absolute deviations, palette,
    translation, and plateau count.
    """

    pair_seed = int(seed) // 2
    rng = random.Random(0x4D4D_2026 + pair_seed * 1_000_003)
    gap = 2 if pair_seed % 6 == 0 else (10, 14, 18, 22, 26)[pair_seed % 5]

    # Exactly seven negatives.  The two closest to zero are -gap, so the
    # even-sample median is exactly CENTER_HEIGHT-gap.
    intermediate = [gap + rng.randint(5, 18) for _ in range(4)]
    negatives = [-EXTREME_DEVIATION, *(-value for value in intermediate), -gap, -gap]
    remaining_positive_total = sum(intermediate) + 2 * gap
    positives = [EXTREME_DEVIATION, *_four_parts(remaining_positive_total, rng)]
    deviations = negatives + positives
    rng.shuffle(deviations)
    if int(seed) % 2:
        deviations = [-value for value in deviations]
    heights = [CENTER_HEIGHT + value for value in deviations]

    return {
        "heights": heights,
        "fill_rgb": list(PALETTE[pair_seed % len(PALETTE)]),
        "x0": 48 + 3 * (pair_seed % 5),
        "baseline_y": 418 + 2 * (pair_seed % 4),
    }


def _statistics(scene: dict[str, Any]) -> tuple[float, float]:
    heights = [int(value) for value in scene["heights"]]
    if len(heights) != N_PLATEAUS or any(value <= 0 for value in heights):
        raise ValueError("heights must contain twelve positive values")
    ordered = sorted(heights)
    mean = sum(heights) / N_PLATEAUS
    median = (ordered[5] + ordered[6]) / 2.0
    return mean, median


def analytic_gold(scene: dict[str, Any]) -> str:
    mean, median = _statistics(scene)
    if mean == median:
        raise ValueError("mean and median tie has no unique answer")
    return "mean" if mean > median else "median"


def margin(scene: dict[str, Any]) -> float:
    """Runner-up margin: absolute mean-minus-median height, in pixels."""

    mean, median = _statistics(scene)
    return abs(mean - median)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) <= QUARANTINE_PX


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # All represented fields affect pixels; ordinary palette and translation
    # invariances change the raster and are tested as label-preserving twins.
    return []


def render(scene: dict[str, Any]) -> Image.Image:
    heights = [int(value) for value in scene["heights"]]
    fill = tuple(int(value) for value in scene["fill_rgb"])
    x0 = int(scene["x0"])
    baseline_y = int(scene["baseline_y"])
    if len(heights) != N_PLATEAUS:
        raise ValueError("renderer requires twelve plateau heights")
    if not all(0 <= channel <= 255 for channel in fill):
        raise ValueError("fill_rgb is outside RGB range")
    if x0 < 30 or x0 + LAMINA_WIDTH >= WIDTH - 30:
        raise ValueError("lamina does not fit horizontally")
    if baseline_y >= HEIGHT - 48 or baseline_y - max(heights) < 54:
        raise ValueError("lamina does not fit vertically")

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    # Neutral reference rules sit behind the saturated evidence.
    for y in range(92, baseline_y, 54):
        draw.line((x0 - 8, y, x0 + LAMINA_WIDTH + 8, y), fill=GRID, width=1)
    draw.line((x0, baseline_y, x0 + LAMINA_WIDTH - 1, baseline_y), fill=INK, width=3)

    # Adjacent rectangles share a broad common base, making a single filled
    # component.  Each upper edge has the same two-pixel neutral rim, which
    # changes all measured heights by the same constant and preserves order.
    for index, height in enumerate(heights):
        left = x0 + index * PLATEAU_WIDTH
        right = left + PLATEAU_WIDTH - 1
        top = baseline_y - height
        draw.rectangle((left, top, right, baseline_y - 2), fill=fill)
        draw.line((left, top, right, top), fill=INK, width=2)

    # Thirteen ticks expose the twelve equal-width sampling units without
    # cutting the filled component into bars.
    tick_font = _font(11, bold=True)
    for boundary in range(N_PLATEAUS + 1):
        x = x0 + boundary * PLATEAU_WIDTH
        draw.line((x, baseline_y + 2, x, baseline_y + 10), fill=INK, width=2)
    for index in range(N_PLATEAUS):
        label = str(index + 1)
        cx = x0 + index * PLATEAU_WIDTH + PLATEAU_WIDTH // 2
        box = draw.textbbox((0, 0), label, font=tick_font)
        draw.text((cx - (box[2] - box[0]) / 2, baseline_y + 12), label, font=tick_font, fill=INK)

    title_font = _font(19, bold=True)
    label_font = _font(13)
    draw.text((x0, 17), "12-PLATEAU HEIGHT PROFILE", font=title_font, fill=INK)
    draw.text((x0, baseline_y + 31), "equal-width plateaus • common baseline", font=label_font, fill=INK)
    return image
