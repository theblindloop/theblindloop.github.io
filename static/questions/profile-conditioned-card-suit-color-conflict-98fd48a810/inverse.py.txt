"""Independent final-PNG-only inverse for red playing-card suit identity."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _red_mask(image: Image.Image) -> np.ndarray:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    return (
        (pixels[..., 0] > 135)
        & (pixels[..., 0] > pixels[..., 1] * 1.75)
        & (pixels[..., 0] > pixels[..., 2] * 1.55)
    )


def _shape_score(mask: np.ndarray) -> float:
    ys, xs = np.nonzero(mask)
    if xs.size < 2400:
        raise ValueError("red suit evidence is absent or materially erased")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    width, height = x1 - x0 + 1, y1 - y0 + 1
    if width < 65 or height < 75 or width > 110 or height > 125:
        raise ValueError("red component does not have suit-like dimensions")

    # Heart has broad paired lobes in the upper quarter and a narrow lower
    # point. Spade begins with a narrow upper point. The statistic uses only
    # normalized rows inside the recovered component, never absolute position.
    rows = []
    for frac in (0.20, 0.24, 0.28):
        y = min(y1, y0 + int(round(frac * (height - 1))))
        row_x = np.nonzero(mask[y])[0]
        rows.append(0 if row_x.size == 0 else int(row_x.max() - row_x.min() + 1))
    upper_width = float(np.median(rows))
    max_width = max(int(np.count_nonzero(mask[y])) for y in range(y0, y1 + 1))
    if max_width < 55:
        raise ValueError("red silhouette is too narrow")
    return upper_width / float(max_width)


def decision_from_image(image: Image.Image) -> str:
    score = _shape_score(_red_mask(image))
    if abs(score - 0.52) < 0.08:
        raise ValueError("heart-versus-spade geometry is ambiguous")
    return "heart" if score > 0.52 else "spade"
