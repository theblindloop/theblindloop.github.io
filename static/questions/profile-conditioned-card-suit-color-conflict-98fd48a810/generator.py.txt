"""Deterministic matched heart/spade scenes for visible-prior conflict."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

SIZE = 384
SUPERSAMPLE = 4
RED = (202, 43, 60)
DARK = (34, 39, 47)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed ^ 0xCA4D51)
    return {
        "red_suit": "heart" if seed % 2 == 0 else "spade",
        "heart_side": "left" if (seed // 2) % 2 == 0 else "right",
        "center_x": 192 + rng.randint(-9, 9),
        "center_y": 194 + rng.randint(-8, 8),
        "scale": round(rng.uniform(0.92, 1.06), 4),
        "card_tint": [255, 253, 247] if rng.randrange(2) else [247, 251, 255],
        "draw_order": ["heart", "spade"] if rng.randrange(2) else ["spade", "heart"],
    }


def _heart(draw: ImageDraw.ImageDraw, cx: float, cy: float, unit: float, fill: tuple[int, int, int]) -> None:
    # Two lobes plus a broad, downward-pointing kite. All coordinates are
    # relative and visible; no text or metadata identifies the shape.
    r = 25.0 * unit
    draw.ellipse((cx - 42 * unit, cy - 37 * unit, cx + 8 * unit, cy + 13 * unit), fill=fill)
    draw.ellipse((cx - 8 * unit, cy - 37 * unit, cx + 42 * unit, cy + 13 * unit), fill=fill)
    draw.polygon([
        (cx - 42 * unit, cy - 10 * unit),
        (cx, cy + 54 * unit),
        (cx + 42 * unit, cy - 10 * unit),
        (cx, cy - 25 * unit),
    ], fill=fill)


def _spade(draw: ImageDraw.ImageDraw, cx: float, cy: float, unit: float, fill: tuple[int, int, int]) -> None:
    # A vertically reflected heart body plus a narrow stem. The stem makes the
    # suit identity explicit while remaining part of the same filled component.
    draw.ellipse((cx - 42 * unit, cy - 7 * unit, cx + 8 * unit, cy + 43 * unit), fill=fill)
    draw.ellipse((cx - 8 * unit, cy - 7 * unit, cx + 42 * unit, cy + 43 * unit), fill=fill)
    draw.polygon([
        (cx - 42 * unit, cy + 16 * unit),
        (cx, cy - 52 * unit),
        (cx + 42 * unit, cy + 16 * unit),
        (cx, cy + 31 * unit),
    ], fill=fill)
    draw.polygon([
        (cx - 10 * unit, cy + 24 * unit),
        (cx + 10 * unit, cy + 24 * unit),
        (cx + 12 * unit, cy + 53 * unit),
        (cx + 25 * unit, cy + 58 * unit),
        (cx - 25 * unit, cy + 58 * unit),
        (cx - 12 * unit, cy + 53 * unit),
    ], fill=fill)


def render(scene: dict) -> Image.Image:
    factor = SUPERSAMPLE
    image = Image.new("RGB", (SIZE * factor, SIZE * factor), (228, 232, 237))
    draw = ImageDraw.Draw(image)
    cx = float(scene["center_x"]) * factor
    cy = float(scene["center_y"]) * factor
    scale = float(scene["scale"]) * factor

    tint = tuple(int(v) for v in scene["card_tint"])
    card = (cx - 164 * factor, cy - 158 * factor, cx + 164 * factor, cy + 154 * factor)
    draw.rounded_rectangle(card, radius=18 * factor, fill=tint, outline=(78, 84, 94), width=4 * factor)

    sides = {scene["heart_side"]: "heart", "right" if scene["heart_side"] == "left" else "left": "spade"}
    centers = {"left": cx - 82 * factor, "right": cx + 82 * factor}
    colors = {scene["red_suit"]: RED, "spade" if scene["red_suit"] == "heart" else "heart": DARK}
    shape_side = {shape: side for side, shape in sides.items()}
    for shape in scene["draw_order"]:
        sx = centers[shape_side[shape]]
        if shape == "heart":
            _heart(draw, sx, cy, scale, colors[shape])
        else:
            _spade(draw, sx, cy, scale, colors[shape])

    # Neutral divider prevents perceptual grouping without carrying the answer.
    draw.line((cx, cy - 91 * factor, cx, cy + 91 * factor), fill=(176, 181, 189), width=2 * factor)
    return image.resize((SIZE, SIZE), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["red_suit"])


def margin(scene: dict) -> float:
    # Conservative designed distance of the row-width shape statistic from its
    # 0.52 decision boundary; measured stress margins are larger.
    return 0.16


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.08


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["draw_order"] = list(reversed(scene["draw_order"]))
    return [("reverse_disjoint_symbol_draw_order", twin)]
