"""Independent pixel-only inverse arm for the peak-count argmax world.

This module receives ONLY the rasterized PNG (plus the public palette, which is
the world's fixed naming convention) and re-measures which colored band contains
the most distinct peaks (mountain tops). It shares no geometry with
renderer.render() and never imports the renderer, prompts, generator, verifier,
or any analytical gold code.
"""

from __future__ import annotations

from typing import Any

import numpy as np
from PIL import Image

ORACLE_VERSION = "peak_count_argmax-oracle-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
BACKGROUND = (250, 250, 250)

# Columns whose topmost pixel is within this many rows of a band's highest point
# are treated as apex columns of that band's peaks.
_APEX_TOLERANCE_PX = 2


def _nearest_color(arr: np.ndarray) -> np.ndarray:
    """Label every pixel to its nearest palette entry (segment colors + background)."""
    refs = np.stack(
        [np.array(COLOR_RGB[c], dtype=np.float32) for c in PALETTE]
        + [np.array(BACKGROUND, dtype=np.float32)],
        axis=0,
    )
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def _count_peaks(mask: np.ndarray) -> int:
    """Count distinct peaks in one colored band from its raster silhouette.

    The topmost (smallest row) pixel of every column defines the band's height
    silhouette. All peaks are equal-height sharp apexes, so a column is an apex
    column when its topmost pixel reaches the band's global top. Each contiguous
    run of apex columns is one peak; the number of runs is the peak count.
    """
    ys, xs = np.nonzero(mask)
    if len(ys) < 10:
        return -1
    top_by_col: dict[int, int] = {}
    cols = np.unique(xs)
    for x in cols:
        col_rows = ys[xs == x]
        top_by_col[int(x)] = int(col_rows.min())
    global_top = min(top_by_col.values())
    xs_sorted = sorted(top_by_col)
    runs = 0
    in_run = False
    for x in xs_sorted:
        is_apex = top_by_col[x] <= global_top + _APEX_TOLERANCE_PX
        if is_apex:
            if not in_run:
                runs += 1
                in_run = True
        else:
            in_run = False
    return runs


def decision_from_image(image: Image.Image) -> str:
    """Return the color of the band with the most peaks from pixels alone."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _nearest_color(arr)
    best_color = PALETTE[0]
    best_count = -1
    for idx, color in enumerate(PALETTE):
        mask = labels == idx
        count = _count_peaks(mask)
        if count > best_count:
            best_count = count
            best_color = color
    return best_color
