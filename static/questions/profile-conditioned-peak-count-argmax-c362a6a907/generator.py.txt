"""Peak-count argmax world: which colored waveform has the most peaks.

Latent scene z carries a set of three colored waveforms, each drawn as a band
(fixed horizontal lane) of equal-height triangular peaks standing on a shared
baseline for that lane. The task is a pure measurement: count the number of
distinct peaks (mountain tops) in each colored band and select the color with
the most peaks. All peaks in a band have the same apex height, so the count is
read off the raster by finding how many separated apex columns each colored band
contains.

This world is a *measurement* (count of recovered local maxima / peaks from
plotted strokes) and a set-argmax selection, counterbalancing which color holds
the largest peak count and which spatial lane each color occupies.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "peak_count_argmax-0.1.0"

# Fixed palette shared with oracle.py (plain list so both modules agree on the
# named answer set without importing each other).
PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
BACKGROUND = (250, 250, 250)
BASELINE_RGB = (168, 168, 168)

WIDTH = 480
HEIGHT = 340
X_START = 44
X_END = 436
LANE_BASELINES = [78, 170, 262]
PEAK_HEIGHT = 52
BASELINE_WIDTH = 2

MIN_PEAK_COUNT = 2
MAX_PEAK_COUNT = 5
# Margin (in peak-count units) required for an unambiguous selection.
MIN_MARGIN = 1


def sample_scene(seed: int) -> dict:
    """Sample a deterministic scene with a guaranteed, counterbalanced argmax.

    Returns a plain dict (JSON-serialisable). ``traces`` is a list of
    {"color", "peak_count", "lane"} where lane indexes LANE_BASELINES.
    """
    rng = random.Random(seed * 131 + 17)
    pool = list(range(MIN_PEAK_COUNT, MAX_PEAK_COUNT + 1))  # 2..5
    max_count = rng.choice([3, 4, 5])
    others = [c for c in pool if c != max_count]
    two_others = rng.sample(others, 2)
    counts = [max_count] + two_others
    rng.shuffle(counts)
    colors = list(PALETTE)
    rng.shuffle(colors)
    lanes = list(range(len(PALETTE)))
    rng.shuffle(lanes)
    traces = [
        {"color": color, "peak_count": count, "lane": lane}
        for color, count, lane in zip(colors, counts, lanes)
    ]
    return {
        "traces": traces,
        "width": WIDTH,
        "height": HEIGHT,
        "background": "light",
    }


def _render_band(draw: ImageDraw.ImageDraw, scene: dict, trace: dict) -> None:
    lane = trace["lane"]
    base_y = LANE_BASELINES[lane]
    apex_y = base_y - PEAK_HEIGHT
    n = trace["peak_count"]
    color = COLOR_RGB[trace["color"]]
    width = X_END - X_START
    base_w = width / n
    # baseline of the lane (light gray plot baseline)
    draw.line([(X_START, base_y), (X_END, base_y)], fill=BASELINE_RGB, width=BASELINE_WIDTH)
    # triangular peaks standing on the baseline
    for i in range(n):
        cx = X_START + (i + 0.5) * base_w
        draw.polygon(
            [(cx - base_w / 2, base_y), (cx, apex_y), (cx + base_w / 2, base_y)],
            fill=color,
        )


def render(scene: dict) -> Image.Image:
    """Render I = R(z): three colored peak-bands, one per lane."""
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    for trace in scene["traces"]:
        _render_band(draw, scene, trace)
    return img


def _peak_argmax(scene: dict) -> tuple[dict, float]:
    """Return (trace with most peaks, its count margin) from the analytic scene."""
    keyed = sorted(scene["traces"], key=lambda t: t["peak_count"], reverse=True)
    top, second = keyed[0], keyed[1]
    margin = float(top["peak_count"] - second["peak_count"])
    return top, margin


def analytic_gold(scene: dict) -> str:
    """The color of the waveform with the most peaks."""
    top, _margin = _peak_argmax(scene)
    return top["color"]


def margin(scene: dict) -> float:
    """Verifiability margin: the top-two peak-count gap (in peaks)."""
    _top, margin_peaks = _peak_argmax(scene)
    return float(margin_peaks)


def is_quarantined(scene: dict) -> bool:
    """Near-boundary: the top-two peak counts are tied (a finite-resolution case).

    Ties are never emitted because sampling enforces distinct peak counts, so the
    decision boundary (two bands with an equal peak count) is explicitly excluded.
    Every emitted scene has a peak-count margin of at least one.
    """
    return margin(scene) < MIN_MARGIN


def latent_symmetries(scene: dict) -> list:
    """not_applicable: no nontrivial pixel-identical latent symmetry.

    Swapping two bands' color labels changes which color is the answer, and any
    global flip/rotation turns the raster into a distinct image. Every latent
    field (the color label, peak count, and lane of each band) is directly
    visible in the render, so the parametrisation has no hidden label alias.
    """
    return []
