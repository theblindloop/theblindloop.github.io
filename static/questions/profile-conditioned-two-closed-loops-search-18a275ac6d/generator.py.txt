"""Deterministic latent sampler and Pillow renderer."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 640, 480
PANEL_BOXES = ((24, 52, 312, 240), (328, 52, 616, 240),
               (24, 260, 312, 448), (328, 260, 616, 448))
LABELS = ("A", "B", "C", "D")


def _balanced_dashes(radii: list[int], closed: list[bool], gaps: list[int], rng: random.Random) -> list[int]:
    contour_length = sum(
        2.0 * math.pi * radius * (1.0 if shut else (1.0 - gap / 360.0))
        for radius, shut, gap in zip(radii, closed, gaps)
    )
    total = max(66, int(round(650.0 - contour_length)))
    first = rng.randint(18, max(19, total - 42))
    second = rng.randint(18, max(19, total - first - 20))
    third = max(18, total - first - second)
    return [first, second, third]


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    panel_counts = [0, 1, 2, 3]
    rng.shuffle(panel_counts)
    centers: list[list[list[int]]] = []
    radii: list[list[int]] = []
    closed_slots: list[list[bool]] = []
    gap_angles: list[list[int]] = []
    gap_degrees: list[list[int]] = []
    balance_lengths: list[list[int]] = []
    for count in panel_counts:
        order = list(range(3))
        rng.shuffle(order)
        shut = [index in order[:count] for index in range(3)]
        panel_centers = [[58 + 85 * index + rng.randint(-4, 4), 91 + rng.randint(-5, 5)] for index in range(3)]
        panel_radii = [rng.randint(25, 30) for _ in range(3)]
        bearings = [rng.randint(0, 359) for _ in range(3)]
        gaps = [rng.randint(38, 48) for _ in range(3)]
        centers.append(panel_centers)
        radii.append(panel_radii)
        closed_slots.append(shut)
        gap_angles.append(bearings)
        gap_degrees.append(gaps)
        balance_lengths.append(_balanced_dashes(panel_radii, shut, gaps, rng))
    palettes = [
        {"background": [248, 247, 243], "panel": [255, 255, 252], "border": [73, 111, 137], "ink": [35, 39, 42], "accent": [238, 183, 55]},
        {"background": [245, 248, 247], "panel": [255, 255, 253], "border": [67, 112, 132], "ink": [40, 38, 43], "accent": [232, 171, 58]},
        {"background": [249, 246, 245], "panel": [255, 254, 251], "border": [80, 108, 139], "ink": [32, 41, 43], "accent": [242, 177, 48]},
    ]
    return {
        "panel_counts": panel_counts,
        "centers": centers,
        "radii": radii,
        "closed_slots": closed_slots,
        "gap_angles": gap_angles,
        "gap_degrees": gap_degrees,
        "balance_lengths": balance_lengths,
        "stroke_width": rng.choice([7, 8]),
        "palette": palettes[rng.randrange(len(palettes))],
    }


def render(scene: dict[str, Any]) -> Image.Image:
    panel_counts = scene["panel_counts"]
    centers = scene["centers"]
    radii = scene["radii"]
    closed_slots = scene["closed_slots"]
    gap_angles = scene["gap_angles"]
    gap_degrees = scene["gap_degrees"]
    balance_lengths = scene["balance_lengths"]
    stroke_width = int(scene["stroke_width"])
    palette = scene["palette"]
    image = Image.new("RGB", (WIDTH, HEIGHT), tuple(palette["background"]))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    draw.text((24, 19), "CLOSED-LOOP SEARCH", fill=tuple(palette["ink"]), font=font)
    draw.text((470, 19), "exactly 2", fill=tuple(palette["border"]), font=font)
    for panel_index, (left, top, right, bottom) in enumerate(PANEL_BOXES):
        draw.rounded_rectangle((left, top, right, bottom), radius=9,
                               fill=tuple(palette["panel"]), outline=tuple(palette["border"]), width=3)
        draw.rounded_rectangle((left + 10, top + 7, left + 34, top + 29), radius=5,
                               fill=tuple(palette["accent"]))
        draw.text((left + 18, top + 12), LABELS[panel_index], fill=tuple(palette["ink"]), font=font, anchor="mm")
        # Read the count here as well as in analytic_gold: it visibly determines
        # the number of closed strokes and is not a gold-only field.
        expected_count = int(panel_counts[panel_index])
        assert expected_count == sum(bool(value) for value in closed_slots[panel_index])
        for slot in range(3):
            cx = left + int(centers[panel_index][slot][0])
            cy = top + int(centers[panel_index][slot][1])
            radius = int(radii[panel_index][slot])
            box = (cx - radius, cy - radius, cx + radius, cy + radius)
            if closed_slots[panel_index][slot]:
                draw.ellipse(box, outline=tuple(palette["ink"]), width=stroke_width)
            else:
                bearing = int(gap_angles[panel_index][slot])
                gap = int(gap_degrees[panel_index][slot])
                draw.arc(box, start=bearing + gap // 2, end=bearing + 360 - gap // 2,
                         fill=tuple(palette["ink"]), width=stroke_width)
        # Three constant-count detached strokes absorb contour-ink differences.
        dash_y = top + 154
        cursor = left + 22
        for slot, length in enumerate(balance_lengths[panel_index]):
            y = dash_y + (slot % 2) * 9
            draw.line((cursor, y, cursor + int(length), y), fill=tuple(palette["ink"]), width=3)
            cursor += int(length) + 10
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    counts = [int(value) for value in scene["panel_counts"]]
    winners = [index for index, value in enumerate(counts) if value == 2]
    return LABELS[winners[0]] if len(winners) == 1 else "abstain"


def margin(scene: dict[str, Any]) -> float:
    counts = [int(value) for value in scene["panel_counts"]]
    scores = sorted((-abs(value - 2) for value in counts), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict[str, Any]) -> bool:
    if int(scene["stroke_width"]) < 6:
        return True
    for panel_radii, panel_closed, panel_gaps in zip(scene["radii"], scene["closed_slots"], scene["gap_degrees"]):
        for radius, shut, degrees in zip(panel_radii, panel_closed, panel_gaps):
            if not shut and 2.0 * math.pi * float(radius) * float(degrees) / 360.0 < 14.0:
                return True
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict]]:
    return []
