"""Independent image-only flood-fill oracle."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int, bool]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[int, int, int, int, int, bool]] = []
    for y0, x0 in zip(*np.nonzero(mask & ~seen)):
        if seen[y0, x0]:
            continue
        queue = deque([(int(y0), int(x0))])
        seen[y0, x0] = True
        area = 0
        min_x = max_x = int(x0)
        min_y = max_y = int(y0)
        touches = False
        while queue:
            y, x = queue.popleft()
            area += 1
            min_x, max_x = min(min_x, x), max(max_x, x)
            min_y, max_y = min(min_y, y), max(max_y, y)
            touches = touches or x == 0 or y == 0 or x == width - 1 or y == height - 1
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        found.append((min_x, min_y, max_x, max_y, area, touches))
    return found


def _panel_boxes(rgb: np.ndarray) -> list[tuple[int, int, int, int]]:
    red, green, blue = rgb[:, :, 0], rgb[:, :, 1], rgb[:, :, 2]
    border = (blue > red + 30) & (blue > green + 10) & (red < 120) & (green < 150)
    boxes = []
    for min_x, min_y, max_x, max_y, area, _ in _components(border):
        if area > 700 and max_x - min_x > 220 and max_y - min_y > 140:
            boxes.append((min_x, min_y, max_x, max_y))
    boxes.sort(key=lambda box: (box[1], box[0]))
    return boxes


def _hole_count(rgb: np.ndarray, box: tuple[int, int, int, int]) -> int:
    left, top, right, bottom = box
    # The colored frame determines the crop. The label band is excluded; no
    # renderer coordinates or latent state enter this computation.
    crop = rgb[top + 31:bottom - 8, left + 7:right - 7]
    dark = np.max(crop, axis=2) < 105
    white_space = ~dark
    pockets = 0
    for min_x, min_y, max_x, max_y, area, touches in _components(white_space):
        if not touches and area > 240 and (max_x - min_x) > 12 and (max_y - min_y) > 12:
            pockets += 1
    return pockets


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    boxes = _panel_boxes(rgb)
    if len(boxes) != 4:
        return "abstain"
    counts = [_hole_count(rgb, box) for box in boxes]
    winners = [index for index, value in enumerate(counts) if value == 2]
    return LABELS[winners[0]] if len(winners) == 1 else "abstain"
