"""Scene sampler, renderer and analytic gold for `button_hole_count`.

A single button is painted on white paper.  The decision is how many openings
are cut through it - white gaps through which the page shows - against the
canonical prior that a button has four holes.  Coloured spots printed on the
button are solid ink and are declared not to be openings, so the answer is a
cardinality of *enclosed background regions*, not of round marks.

Every latent field is painted into the raster; only `holes` feeds analytic gold.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "button_hole_count-renderer-0.1.0"

WIDTH = 480
HEIGHT = 480

COUNTS = (2, 3, 4, 5, 6)
CANONICAL_COUNT = 4
CANONICAL_DECISION = "4"

# Quarantine threshold, in pixels of the smallest visible separation that would
# have to be crossed for the count to change.
QUARANTINE_MARGIN = 11.0
MIN_SAMPLED_MARGIN = 7.0

FRAME_COLOR = (150, 150, 150)
TICK_COLOR = (150, 150, 150)
WHITE = (255, 255, 255)

PALETTE = (
    (196, 58, 64),
    (32, 124, 132),
    (214, 148, 32),
    (76, 74, 168),
    (104, 132, 52),
    (150, 64, 140),
)


def _darker(rgb, f=0.60):
    return tuple(max(0, int(round(c * f))) for c in rgb)


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------
def _hole_ring(cx, cy, rr, rh, k, phase, jitter):
    """k openings evenly spaced on a ring, with small per-hole jitter."""
    holes = []
    for i in range(k):
        a = phase + 2.0 * math.pi * i / k + jitter[i % len(jitter)][0]
        d = rr + jitter[i % len(jitter)][1]
        holes.append([round(cx + d * math.cos(a), 3), round(cy + d * math.sin(a), 3), float(rh)])
    return holes


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    for _ in range(400):
        canonical = rng.random() < 0.5
        k = CANONICAL_COUNT if canonical else rng.choice((2, 3, 5, 6))

        big_r = rng.randint(112, 154)
        cx = rng.randint(int(0.5 * WIDTH) - 26, int(0.5 * WIDTH) + 26)
        cy = rng.randint(int(0.5 * HEIGHT) - 26, int(0.5 * HEIGHT) + 26)

        rr = rng.uniform(0.32, 0.44) * big_r
        # keep every count in COUNTS geometrically legal for this ring
        rh_cap = min((rr - 9.0) / 2.0, 0.58 * big_r - rr, 0.13 * big_r)
        if rh_cap < 4.0:
            continue
        rh = round(rng.uniform(3.6, rh_cap), 3)

        phase = rng.uniform(0.0, 2.0 * math.pi)
        jitter = [(rng.uniform(-0.055, 0.055), rng.uniform(-2.0, 2.0)) for _ in range(7)]
        holes = _hole_ring(cx, cy, rr, rh, k, phase, jitter)

        b_i = rng.randrange(len(PALETTE))
        s_i = (b_i + rng.randrange(1, len(PALETTE))) % len(PALETTE)
        button_color = PALETTE[b_i]
        spot_color = PALETTE[s_i]

        # printed spots live in a reserved annulus that no opening can reach,
        # and their number is sampled independently of the answer
        n_spots = rng.choice((0, 1, 2, 3))
        spots = []
        for j in range(n_spots):
            a = rng.uniform(0.0, 2.0 * math.pi)
            d = rng.uniform(0.66, 0.76) * big_r
            sr = round(rng.uniform(4.0, 0.055 * big_r + 4.0), 3)
            spots.append([round(cx + d * math.cos(a), 3), round(cy + d * math.sin(a), 3), sr])

        scene = {
            "width": WIDTH,
            "height": HEIGHT,
            "cx": cx,
            "cy": cy,
            "big_r": big_r,
            "holes": holes,
            "spots": spots,
            "ring_r": round(0.90 * big_r, 3),
            "ring_w": rng.choice((3, 4)),
            "button_color": list(button_color),
            "spot_color": list(spot_color),
        }
        if _legal(scene) and margin(scene) >= MIN_SAMPLED_MARGIN:
            return scene
    raise RuntimeError(f"sampler failed for seed {seed}")


def _legal(scene: dict) -> bool:
    cx, cy, big_r = scene["cx"], scene["cy"], scene["big_r"]
    if cx - big_r < 40 or cy - big_r < 40:
        return False
    if cx + big_r > WIDTH - 40 or cy + big_r > HEIGHT - 40:
        return False
    for x, y, r in scene["holes"]:
        if math.hypot(x - cx, y - cy) + r > 0.60 * big_r:
            return False
        if r < 3.4:
            return False
    for x, y, r in scene["spots"]:
        d = math.hypot(x - cx, y - cy)
        if d - r < 0.62 * big_r or d + r > 0.845 * big_r:
            return False
    return True


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def _disk(draw, x, y, r, color):
    draw.ellipse([x - r, y - r, x + r, y + r], fill=color)


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (int(scene["width"]), int(scene["height"])), WHITE)
    d = ImageDraw.Draw(img)

    # fixed decoration: identical in every image of this set
    d.rectangle([10, 10, int(scene["width"]) - 11, int(scene["height"]) - 11],
                outline=FRAME_COLOR, width=2)
    for (x0, y0, x1, y1) in ((22, 22, 22, 46), (22, 22, 46, 22),
                             (457, 457, 457, 433), (457, 457, 433, 457)):
        d.line([x0, y0, x1, y1], fill=TICK_COLOR, width=2)

    cx, cy = scene["cx"], scene["cy"]
    _disk(d, cx, cy, scene["big_r"], tuple(scene["button_color"]))
    rr = scene["ring_r"]
    d.ellipse([cx - rr, cy - rr, cx + rr, cy + rr],
              outline=_darker(tuple(scene["button_color"])), width=int(scene["ring_w"]))
    for x, y, r in scene["spots"]:
        _disk(d, x, y, r, tuple(scene["spot_color"]))
    for x, y, r in scene["holes"]:
        _disk(d, x, y, r, WHITE)
    return img


# --------------------------------------------------------------------------
# gold, margin, quarantine
# --------------------------------------------------------------------------
def analytic_gold(scene: dict) -> str:
    return str(len(scene["holes"]))


def margin(scene: dict) -> float:
    """Smallest visible separation, in pixels, whose loss would change the count.

    An opening vanishes when its own width falls to zero, two openings become
    one when the gap between their painted edges closes, and an opening stops
    being an enclosed gap when it reaches the button's painted edge.
    """
    holes = scene["holes"]
    cx, cy, big_r = scene["cx"], scene["cy"], scene["big_r"]
    vals = []
    for x, y, r in holes:
        vals.append(2.0 * r)
        vals.append(big_r - (math.hypot(x - cx, y - cy) + r))
    for i in range(len(holes)):
        for j in range(i + 1, len(holes)):
            xi, yi, ri = holes[i]
            xj, yj, rj = holes[j]
            vals.append(math.hypot(xi - xj, yi - yj) - ri - rj)
    return round(max(0.0, min(vals)), 6) if vals else 0.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list:
    """No latent aliases are declared: every field is painted and recoverable."""
    return []


# --------------------------------------------------------------------------
# transforms used by verify.py and the tests
# --------------------------------------------------------------------------
def _copy(scene: dict) -> dict:
    out = dict(scene)
    out["holes"] = [list(h) for h in scene["holes"]]
    out["spots"] = [list(s) for s in scene["spots"]]
    out["button_color"] = list(scene["button_color"])
    out["spot_color"] = list(scene["spot_color"])
    return out


def translate(scene: dict, dx: int, dy: int) -> dict:
    out = _copy(scene)
    out["cx"] = scene["cx"] + dx
    out["cy"] = scene["cy"] + dy
    for h in out["holes"]:
        h[0] = round(h[0] + dx, 3)
        h[1] = round(h[1] + dy, 3)
    for s in out["spots"]:
        s[0] = round(s[0] + dx, 3)
        s[1] = round(s[1] + dy, 3)
    return out


def recolor(scene: dict) -> dict:
    out = _copy(scene)
    b = PALETTE.index(tuple(scene["button_color"]))
    s = PALETTE.index(tuple(scene["spot_color"]))
    out["button_color"] = list(PALETTE[(b + 2) % len(PALETTE)])
    out["spot_color"] = list(PALETTE[(s + 3) % len(PALETTE)])
    if out["button_color"] == out["spot_color"]:
        out["spot_color"] = list(PALETTE[(s + 4) % len(PALETTE)])
    return out


def permute_decor(scene: dict, seed: int) -> dict:
    """Rigidly rotate the whole printed interior; the count cannot change."""
    out = _copy(scene)
    ang = 0.31 + 0.017 * (seed % 61)
    cx, cy = scene["cx"], scene["cy"]
    ca, sa = math.cos(ang), math.sin(ang)
    for group in ("holes", "spots"):
        for item in out[group]:
            dx, dy = item[0] - cx, item[1] - cy
            item[0] = round(cx + ca * dx - sa * dy, 3)
            item[1] = round(cy + sa * dx + ca * dy, 3)
    return out


def rescale(scene: dict, factor: float = 0.9) -> dict:
    out = _copy(scene)
    cx, cy = scene["cx"], scene["cy"]
    out["big_r"] = round(scene["big_r"] * factor, 3)
    out["ring_r"] = round(scene["ring_r"] * factor, 3)
    for group in ("holes", "spots"):
        for item in out[group]:
            item[0] = round(cx + (item[0] - cx) * factor, 3)
            item[1] = round(cy + (item[1] - cy) * factor, 3)
            item[2] = round(item[2] * factor, 3)
    return out


def set_hole_count(scene: dict, k: int) -> dict:
    """Declared label-transforming edit: repaint the button with k openings."""
    if k not in COUNTS:
        raise ValueError(f"unsupported count: {k}")
    out = _copy(scene)
    cx, cy = scene["cx"], scene["cy"]
    x0, y0, r0 = scene["holes"][0]
    rr = math.hypot(x0 - cx, y0 - cy)
    phase = math.atan2(y0 - cy, x0 - cx)
    out["holes"] = _hole_ring(cx, cy, rr, r0, k, phase, [(0.0, 0.0)])
    return out


def collapse_margin(scene: dict) -> dict:
    """Slide two openings together until their painted edges meet."""
    out = _copy(scene)
    x0, y0, r0 = out["holes"][0]
    out["holes"][1] = [x0 + 0.5 * r0, y0, out["holes"][1][2]]
    return out


def ink_pixels(image: Image.Image) -> int:
    import numpy as np

    arr = np.asarray(image.convert("RGB"))
    return int((arr < 250).any(axis=-1).sum())


def hole_box(scene: dict) -> tuple:
    """Bounding box of all openings, used for evidence-erasure tests."""
    xs0 = min(h[0] - h[2] for h in scene["holes"])
    xs1 = max(h[0] + h[2] for h in scene["holes"])
    ys0 = min(h[1] - h[2] for h in scene["holes"])
    ys1 = max(h[1] + h[2] for h in scene["holes"])
    return (xs0, ys0, xs1, ys1)


def runner_up(scene: dict) -> str:
    """Count the image would show if its smallest visible separation closed.

    Every binding term in `margin` removes exactly one enclosed opening: the
    smallest opening closing up, two openings merging into one, or an opening
    breaking out through the button's edge.
    """
    return str(len(scene["holes"]) - 1)
