"""Pixel-only inverse arm for `button_hole_count`.

Receives one rasterised PNG and nothing else: no scene data and no code from
the forward arm.  It segments the coloured button by chroma, marks the pixels
the button encloses, and counts the connected white regions inside it - the
openings through which the page shows.  Solid coloured spots are never white,
so they are never counted.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "button_hole_count-oracle-0.1.0"
ABSTAIN = "abstain"

CHROMA_MIN = 40
WHITE_MIN = 245
MIN_OPENING_AREA = 8
MIN_BUTTON_AREA = 2000
VALID_COUNTS = (2, 3, 4, 5, 6)


def _components(mask: np.ndarray) -> list:
    """Areas of 4-connected components of a small boolean mask."""
    h, w = mask.shape
    flat = mask.reshape(-1).copy()
    areas = []
    idx = np.flatnonzero(flat)
    for start in idx:
        if not flat[start]:
            continue
        stack = [int(start)]
        flat[start] = False
        area = 0
        while stack:
            p = stack.pop()
            area += 1
            r, c = divmod(p, w)
            if c > 0 and flat[p - 1]:
                flat[p - 1] = False
                stack.append(p - 1)
            if c < w - 1 and flat[p + 1]:
                flat[p + 1] = False
                stack.append(p + 1)
            if r > 0 and flat[p - w]:
                flat[p - w] = False
                stack.append(p - w)
            if r < h - 1 and flat[p + w]:
                flat[p + w] = False
                stack.append(p + w)
        areas.append(area)
    return areas


def decision_from_image(image: "Image.Image") -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    chroma = arr.max(axis=-1) - arr.min(axis=-1)
    body = chroma >= CHROMA_MIN
    if int(body.sum()) < MIN_BUTTON_AREA:
        return ABSTAIN

    rows = np.flatnonzero(body.any(axis=1))
    cols = np.flatnonzero(body.any(axis=0))
    r0, r1 = int(rows[0]), int(rows[-1]) + 1
    c0, c1 = int(cols[0]), int(cols[-1]) + 1
    if (r1 - r0) < 24 or (c1 - c0) < 24:
        return ABSTAIN

    sub_body = body[r0:r1, c0:c1]
    sub_white = (arr[r0:r1, c0:c1] >= WHITE_MIN).all(axis=-1)

    # A pixel is enclosed by the button when painted body pixels lie on all four
    # sides of it in its own row and column.  For the convex painted disk this
    # is exactly its interior, and needs no flood fill of the page.
    left = np.cumsum(sub_body, axis=1) > 0
    right = np.cumsum(sub_body[:, ::-1], axis=1)[:, ::-1] > 0
    up = np.cumsum(sub_body, axis=0) > 0
    down = np.cumsum(sub_body[::-1, :], axis=0)[::-1, :] > 0
    enclosed = sub_white & left & right & up & down

    areas = _components(enclosed)
    if any(a < MIN_OPENING_AREA for a in areas):
        return ABSTAIN  # a mark too small to read as an opening
    count = len(areas)
    if count not in VALID_COUNTS:
        return ABSTAIN
    return str(count)
