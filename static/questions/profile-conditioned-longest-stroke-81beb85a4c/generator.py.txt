"""Longest-stroke (true Euclidean length) selection world.

Latent scene z carries a set of colored line segments, each at an *arbitrary*
orientation and position on a shared light panel. The task is a genuine 2D
measurement: find the segment whose actual planar (Euclidean) length is
largest. Because every segment has a different orientation, no single
coordinate projection (its vertical rise, its horizontal span, or its slope)
deterministically orders the strokes by length; the length must be recovered by
combining both the horizontal and vertical extent of each stroke from the
raster ink. This isolates the capability of measuring a true planar distance of
a stroke, distinct from measuring a single-axis rise (``steepest_rise``) or an
axis-parallel bar length (``relative_length`` / ``longest_bar``).

The inverse arm recovers each stroke's length from the raster by taking the
diameter (most distant point-pair) of each color's labelled ink mask, which for
a straight thick stroke equals its planar length plus a sub-pixel cap
contribution.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "longest_stroke-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
BACKGROUND = (250, 250, 250)

WIDTH = 480
HEIGHT = 340
SEGMENT_WIDTH_PX = 6
NCOLUMNS = 3
COLUMN_WIDTH = WIDTH // NCOLUMNS

# Each stroke is drawn in its own horizontal third so colour masks never
# occlude one another; a stroke's cap and the panel margin stay clean.
EDGE_MARGIN = 8

# Minimum top-two planar-length gap required for a decidable selection (px).
# The raster diameter measure is accurate to well under 3px, so a gap of 8px
# is far above the measurement noise on the independent pixel oracle.
MIN_MARGIN_PX = 8.0
# Below this top-two length gap the scene is quarantined as a finite-resolution
# near-boundary case.
QUARANTINE_PX = 12.0


def _sample_segment(rng, color: str, length: float, center_x: float) -> dict | None:
    """Sample one oblique segment of the given planar length centred on center_x."""
    angle = rng.uniform(0.3, math.pi - 0.3)
    dx = length * math.cos(angle)
    dy = length * math.sin(angle)
    half_x = abs(dx) / 2.0 + EDGE_MARGIN
    if half_x > COLUMN_WIDTH / 2.0 - EDGE_MARGIN:
        return None
    half_y = abs(dy) / 2.0 + EDGE_MARGIN
    x0 = rng.uniform(center_x - half_x + EDGE_MARGIN, center_x + half_x - EDGE_MARGIN)
    y0 = rng.uniform(half_y, HEIGHT - half_y)
    x1 = x0 + dx
    y1 = y0 + dy
    within = (
        EDGE_MARGIN <= x0 <= WIDTH - EDGE_MARGIN
        and EDGE_MARGIN <= x1 <= WIDTH - EDGE_MARGIN
        and EDGE_MARGIN <= y0 <= HEIGHT - EDGE_MARGIN
        and EDGE_MARGIN <= y1 <= HEIGHT - EDGE_MARGIN
    )
    if not within:
        return None
    return {"color": color, "x0": x0, "y0": y0, "x1": x1, "y1": y1}


def sample_scene(seed: int) -> dict:
    """Sample a deterministic scene with a guaranteed, counterbalanced longest stroke."""
    rng = random.Random(seed * 1009 + 7)
    while True:
        lengths = [rng.uniform(90.0, 190.0) for _ in range(NCOLUMNS)]
        lengths_sorted = sorted(lengths, reverse=True)
        if lengths_sorted[0] - lengths_sorted[1] >= MIN_MARGIN_PX:
            break
    colors = list(PALETTE)
    rng.shuffle(colors)
    segments: list[dict] = []
    while True:
        segments = []
        valid = True
        for idx, (color, length) in enumerate(zip(colors, lengths)):
            center_x = idx * COLUMN_WIDTH + COLUMN_WIDTH / 2.0
            seg = _sample_segment(rng, color, length, center_x)
            if seg is None:
                valid = False
                break
            segments.append(seg)
        if valid:
            break
    return {
        "segments": segments,
        "width": WIDTH,
        "height": HEIGHT,
        "background": "light",
    }


def render(scene: dict) -> Image.Image:
    """Render I = R(z): three oblique strokes, one per colour, in three columns."""
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    for seg in scene["segments"]:
        draw.line(
            [(seg["x0"], seg["y0"]), (seg["x1"], seg["y1"])],
            fill=COLOR_RGB[seg["color"]],
            width=SEGMENT_WIDTH_PX,
        )
    return img


def _true_length(x0: float, y0: float, x1: float, y1: float) -> float:
    return math.hypot(x1 - x0, y1 - y0)


def _longest(segments: list[dict]) -> tuple[dict, float, float]:
    """Return (longest segment, its planar length, the top-two length gap)."""
    keyed = [
        (seg, _true_length(seg["x0"], seg["y0"], seg["x1"], seg["y1"]))
        for seg in segments
    ]
    keyed.sort(key=lambda item: item[1], reverse=True)
    top, second = keyed[0], keyed[1]
    return top[0], top[1], top[1] - second[1]


def analytic_gold(scene: dict) -> str:
    """The colour of the segment with the largest true planar (Euclidean) length."""
    top, _length, _margin = _longest(scene["segments"])
    return top["color"]


def margin(scene: dict) -> float:
    """Verifiability margin: the top-two planar-length gap in pixels."""
    _top, _length, margin_px = _longest(scene["segments"])
    return float(margin_px)


def is_quarantined(scene: dict) -> bool:
    """Near-boundary: the top-two planar-length gap is finite-resolution small."""
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list:
    """not_applicable: no nontrivial pixel-identical latent symmetry.

    Every latent field (each stroke's colour label and its two endpoints) is
    directly rendered as visible ink. Swapping two strokes' colour labels
    changes which colour is the longest, and any global flip/rotation or
    permutation turns the raster into a distinct image, so there is no hidden
    alias transformation that is both pixel-identical and gold-preserving.
    """
    return []
