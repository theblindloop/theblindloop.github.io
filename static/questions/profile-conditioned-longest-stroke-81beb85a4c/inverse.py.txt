"""Independent pixel-only inverse arm for the longest-stroke world.

This module receives ONLY the rasterized PNG (plus the world's fixed public
palette) and re-measures which coloured stroke has the largest true planar
(Euclidean) length. It shares no geometry with renderer.render() and never
imports the renderer, prompts, generator, verifier, or any analytical-gold
code. It cannot be fed latent scene data.

For each colour it labels its ink mask, then computes the stroke's length as
the diameter of that mask: the distance between its two farthest pixels. For a
straight thick stroke this diameter equals the planar length (a sub-pixel
contribution comes from the caps, which is identical for every stroke and does
not disturb the ranking).
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "longest_stroke-oracle-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
BACKGROUND = (250, 250, 250)

_MIN_PIXELS = 10


def _nearest_color(arr: np.ndarray) -> np.ndarray:
    """Label every pixel to its nearest palette entry (segment colours + background).

    Edge antialiasing splits at ~50% toward the background, so each labelled
    stroke mask stays clean regardless of any anti-aliasing the renderer added.
    """
    refs = np.stack(
        [np.array(COLOR_RGB[c], dtype=np.float32) for c in PALETTE]
        + [np.array(BACKGROUND, dtype=np.float32)],
        axis=0,
    )
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def _mask_diameter(ys: np.ndarray, xs: np.ndarray) -> float:
    """Planar length-ish scalar: the diameter of a stroke's ink mask.

    Two farthest-pixel sweeps give the diameter of the thick stroke, which for
    a straight segment equals its planar length plus a tiny shared cap offset.
    """
    pts = np.stack([xs.astype(np.float64), ys.astype(np.float64)], axis=1)
    anchor = pts[0]
    dists = np.sum((pts - anchor) ** 2, axis=1)
    far_a = pts[dists.argmax()]
    dists = np.sum((pts - far_a) ** 2, axis=1)
    far_b = pts[dists.argmax()]
    return math.hypot(float(far_b[0] - far_a[0]), float(far_b[1] - far_a[1]))


def _length_from_mask(mask: np.ndarray) -> float:
    ys, xs = np.nonzero(mask)
    if len(ys) < _MIN_PIXELS:
        return -1.0
    return _mask_diameter(ys, xs)


def decision_from_image(image: Image.Image) -> str:
    """Return the colour of the longest (largest planar-length) stroke from pixels."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _nearest_color(arr)
    best_color = PALETTE[0]
    best_length = -1.0
    for idx, color in enumerate(PALETTE):
        length = _length_from_mask(labels == idx)
        if length > best_length:
            best_length = length
            best_color = color
    return best_color
