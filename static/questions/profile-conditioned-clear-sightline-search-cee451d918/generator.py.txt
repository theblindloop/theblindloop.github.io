"""Deterministic sampler, analytic arm, and raster renderer."""

from __future__ import annotations

import copy
import math
import random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 720
BG = (246, 245, 240)
PANEL = (255, 255, 255)
BORDER = (177, 181, 185)
INK = (43, 47, 52)
BLOCK = (61, 65, 71)
GUIDE = (116, 157, 190)
TEAL = (18, 151, 151)
AMBER = (232, 166, 38)

PANEL_BOXES = (
    (35, 82, 465, 365),
    (495, 82, 925, 365),
    (35, 415, 465, 698),
    (495, 415, 925, 698),
)
BLOCK_W, BLOCK_H = 38, 52


def _block_rect(center):
    x, y = center
    return [
        int(round(x - BLOCK_W / 2)),
        int(round(y - BLOCK_H / 2)),
        int(round(x + BLOCK_W / 2)),
        int(round(y + BLOCK_H / 2)),
    ]


def _panel_spec(rng: random.Random, blocked: bool) -> dict:
    start = [58, rng.randint(75, 208)]
    end = [372, rng.randint(75, 208)]
    dx, dy = end[0] - start[0], end[1] - start[1]
    length = math.hypot(dx, dy)
    nx, ny = -dy / length, dx / length
    fractions = [0.29, 0.50, 0.71]
    rng.shuffle(fractions)
    hit_slot = rng.randrange(3) if blocked else -1
    blockers = []
    for index, fraction in enumerate(fractions):
        bx = start[0] + fraction * dx
        by = start[1] + fraction * dy
        if index == hit_slot:
            offset = rng.uniform(-4.0, 4.0)
        else:
            offset = rng.choice((-1, 1)) * rng.uniform(62.0, 78.0)
        blockers.append(_block_rect((bx + nx * offset, by + ny * offset)))
    return {"teal": start, "amber": end, "blockers": blockers}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    answer_index = int(seed) % 4
    panels = [_panel_spec(rng, index != answer_index) for index in range(4)]
    return {"panels": panels, "width": WIDTH, "height": HEIGHT}


def _segment_hits_rect(a, b, rect) -> bool:
    """Closed segment versus closed axis-aligned rectangle (Liang-Barsky)."""
    x0, y0 = float(a[0]), float(a[1])
    dx, dy = float(b[0]) - x0, float(b[1]) - y0
    left, top, right, bottom = map(float, rect)
    t0, t1 = 0.0, 1.0
    for p, q in ((-dx, x0 - left), (dx, right - x0), (-dy, y0 - top), (dy, bottom - y0)):
        if abs(p) < 1e-12:
            if q < 0:
                return False
            continue
        ratio = q / p
        if p < 0:
            t0 = max(t0, ratio)
        else:
            t1 = min(t1, ratio)
        if t0 > t1:
            return False
    return True


def _scores(scene: dict) -> list[int]:
    scores = []
    for panel in scene["panels"]:
        hits = sum(
            _segment_hits_rect(panel["teal"], panel["amber"], rect)
            for rect in panel["blockers"]
        )
        scores.append(1 if hits == 0 else 0)
    return scores


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    winners = [index for index, score in enumerate(scores) if score == 1]
    if len(winners) != 1:
        raise ValueError("scene does not contain one unobstructed guide")
    return "ABCD"[winners[0]]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    scores = sorted(_scores(scene), reverse=True)
    return scores.count(1) != 1 or scores[0] - scores[1] != 1


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    transformed = copy.deepcopy(scene)
    for panel in transformed["panels"]:
        panel["blockers"] = list(reversed(panel["blockers"]))
    return [("reverse_obstacle_list_order", transformed)]


def _to_canvas(box, point):
    return (box[0] + int(point[0]), box[1] + int(point[1]))


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width, height), BG)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    draw.text((width // 2, 25), "FIND THE GUIDE WITH NO DARK BLOCK CROSSING", fill=INK, font=font, anchor="ma")
    for index, (box, panel) in enumerate(zip(PANEL_BOXES, scene["panels"])):
        draw.rounded_rectangle(box, radius=14, fill=PANEL, outline=BORDER, width=3)
        draw.text((box[0] + 18, box[1] + 16), "ABCD"[index], fill=INK, font=font, anchor="mm")
        teal = _to_canvas(box, panel["teal"])
        amber = _to_canvas(box, panel["amber"])
        draw.line((teal, amber), fill=GUIDE, width=7)
        for rect in panel["blockers"]:
            shifted = (box[0] + rect[0], box[1] + rect[1], box[0] + rect[2], box[1] + rect[3])
            draw.rectangle(shifted, fill=BLOCK)
        radius = 18
        draw.ellipse((teal[0] - radius, teal[1] - radius, teal[0] + radius, teal[1] + radius), fill=TEAL)
        draw.ellipse((amber[0] - radius, amber[1] - radius, amber[0] + radius, amber[1] + radius), fill=AMBER)
    return image
