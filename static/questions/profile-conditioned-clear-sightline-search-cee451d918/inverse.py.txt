"""Independent final-PNG inverse arm using only visible pixel evidence."""

from __future__ import annotations

import numpy as np
from PIL import Image

BLOCK = (61, 65, 71)
TEAL = (18, 151, 151)
AMBER = (232, 166, 38)


def _components(mask: np.ndarray, minimum: int = 40):
    height, width = mask.shape
    seen = np.zeros((height, width), dtype=bool)
    found = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        points = []
        for cy, cx in stack:
            points.append((cy, cx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = cy + dy, cx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= minimum:
            found.append(points)
    return found


def _centroid(points):
    return (
        sum(point[1] for point in points) / len(points),
        sum(point[0] for point in points) / len(points),
    )


def _terminal_pairs(arr: np.ndarray):
    teal_mask = np.all(arr == np.array(TEAL, dtype=np.uint8), axis=2)
    amber_mask = np.all(arr == np.array(AMBER, dtype=np.uint8), axis=2)
    teals = [_centroid(component) for component in _components(teal_mask, 200)]
    ambers = [_centroid(component) for component in _components(amber_mask, 200)]
    if len(teals) != 4 or len(ambers) != 4:
        raise ValueError(f"expected four terminal pairs, found {len(teals)} teal and {len(ambers)} amber")
    unused = set(range(4))
    pairs = []
    for teal in sorted(teals, key=lambda point: (-point[0], point[1])):
        choices = [index for index in unused if ambers[index][0] > teal[0]]
        if not choices:
            raise ValueError("could not pair a teal terminal with an amber terminal to its right")
        chosen = min(choices, key=lambda index: (ambers[index][0] - teal[0]) ** 2 + (ambers[index][1] - teal[1]) ** 2)
        unused.remove(chosen)
        pairs.append((teal, ambers[chosen]))
    pairs.sort(key=lambda pair: (pair[0][1] + pair[1][1]) / 2)
    top = sorted(pairs[:2], key=lambda pair: (pair[0][0] + pair[1][0]) / 2)
    bottom = sorted(pairs[2:], key=lambda pair: (pair[0][0] + pair[1][0]) / 2)
    return top + bottom


def _dark_crosses_segment(block_mask: np.ndarray, start, end) -> bool:
    height, width = block_mask.shape
    dx, dy = end[0] - start[0], end[1] - start[1]
    steps = max(1, int(round((dx * dx + dy * dy) ** 0.5)))
    for index in range(int(steps * 0.08), int(steps * 0.92) + 1):
        t = index / steps
        x = int(round(start[0] + t * dx))
        y = int(round(start[1] + t * dy))
        x0, x1 = max(0, x - 3), min(width, x + 4)
        y0, y1 = max(0, y - 3), min(height, y + 4)
        if block_mask[y0:y1, x0:x1].any():
            return True
    return False


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    pairs = _terminal_pairs(arr)
    block_mask = np.all(arr == np.array(BLOCK, dtype=np.uint8), axis=2)
    clear = [not _dark_crosses_segment(block_mask, teal, amber) for teal, amber in pairs]
    winners = [index for index, value in enumerate(clear) if value]
    if len(winners) != 1:
        raise ValueError(f"pixel evidence yields {len(winners)} unobstructed guides")
    return "ABCD"[winners[0]]
