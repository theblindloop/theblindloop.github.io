"""Independent PNG-only inverse for the signed gap-change decision."""

from __future__ import annotations

from collections.abc import Callable

import numpy as np
from PIL import Image


class OracleError(ValueError):
    """Raised when the required marker structure cannot be recovered."""


def _runs(values: np.ndarray) -> list[np.ndarray]:
    unique = np.unique(values.astype(np.int32))
    if len(unique) == 0:
        return []
    splits = np.where(np.diff(unique) > 1)[0] + 1
    return [part for part in np.split(unique, splits) if len(part)]


def _centroid_rows(mask: np.ndarray) -> dict[tuple[int, int, int], float]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 500:
        raise OracleError("too little colored marker ink")

    x_runs = _runs(xs)
    if len(x_runs) != 4:
        raise OracleError(f"expected four checkpoint columns, found {len(x_runs)}")
    x_centers = [float(run.mean()) for run in x_runs]
    x_gaps = np.diff(x_centers)
    split = int(np.argmax(x_gaps)) + 1
    if split != 2 or x_gaps[split - 1] < 1.45 * max(x_gaps[0], x_gaps[2]):
        raise OracleError("checkpoint columns do not form two separated panel columns")

    result: dict[tuple[int, int, int], float] = {}
    for x_index, run in enumerate(x_runs):
        in_x = np.isin(xs, run)
        local_y = ys[in_x]
        local_x = xs[in_x]
        y_runs = _runs(local_y)
        if len(y_runs) != 2:
            raise OracleError("each checkpoint column must contain one marker in each panel row")
        col = 0 if x_index < 2 else 1
        time = x_index % 2
        for row, y_run in enumerate(y_runs):
            component = np.isin(local_y, y_run)
            if int(component.sum()) < 120:
                raise OracleError("marker component is too small")
            # Color-threshold pixels are symmetric around the rendered disk center.
            result[(row, col, time)] = float(local_y[component].mean())
    if len(result) != 8:
        raise OracleError("incomplete marker map")
    return result


def _measure(image: Image.Image, predicate: Callable[[np.ndarray], np.ndarray]) -> dict[tuple[int, int, int], float]:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    return _centroid_rows(predicate(array))


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[..., 0], array[..., 1], array[..., 2]
    blue_mask = (blue > 95) & (blue - red > 45) & (blue - green > 18)
    orange_mask = (red > 130) & (red - blue > 80) & (red - green > 55)

    blue_y = _centroid_rows(blue_mask)
    orange_y = _centroid_rows(orange_mask)
    scores: list[float] = []
    for panel in range(4):
        row, col = divmod(panel, 2)
        before_gap = blue_y[(row, col, 0)] - orange_y[(row, col, 0)]
        after_gap = blue_y[(row, col, 1)] - orange_y[(row, col, 1)]
        scores.append(after_gap - before_gap)

    order = np.argsort(np.asarray(scores))[::-1]
    if scores[int(order[0])] - scores[int(order[1])] < 1.5:
        raise OracleError("raster decision is ambiguous")
    return "ABCD"[int(order[0])]
