"""Deterministic latent sampler and raster renderer for difference-in-differences."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 900, 620
SCALE = 3
QUARANTINE_PX = 8.0

PALETTES = (
    {"blue": "#1769aa", "orange": "#e66b15"},
    {"blue": "#225ea8", "orange": "#d97706"},
    {"blue": "#2166ac", "orange": "#e08214"},
)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size * SCALE)
    except OSError:
        return ImageFont.load_default()


def _scores(values: list[list[int]]) -> list[int]:
    # Each row is blue_before, orange_before, blue_after, orange_after.
    return [
        (orange_after - blue_after) - (orange_before - blue_before)
        for blue_before, orange_before, blue_after, orange_after in values
    ]


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced, safely separated scene.

    Consecutive seeds cycle the winning panel. About one scene in eleven is in
    the declared near-boundary quarantine; all others have a 22-pixel top gap.
    """

    rng = random.Random(int(seed))
    winner = int(seed) % 4
    runner = (winner + 1 + (int(seed) // 4) % 3) % 4
    near = int(seed) % 11 == 0

    # (before orange-blue gap, after orange-blue gap). Their differences are
    # 84 (or 68), 62, 15, and -25. Every same-checkpoint color gap is >= 23 px.
    patterns = [(-34, 34) if near else (-42, 42), (-30, 32), (25, 40), (48, 23)]
    assigned: list[tuple[int, int] | None] = [None] * 4
    assigned[winner] = patterns[0]
    assigned[runner] = patterns[1]
    leftovers = [patterns[2], patterns[3]]
    rng.shuffle(leftovers)
    for panel in range(4):
        if assigned[panel] is None:
            assigned[panel] = leftovers.pop()

    values: list[list[int]] = []
    for before_gap, after_gap in assigned:  # type: ignore[misc]
        before_low = max(18, 18 - before_gap)
        before_high = min(142, 142 - before_gap)
        after_low = max(18, 18 - after_gap)
        after_high = min(142, 142 - after_gap)
        blue_before = rng.randint(before_low, before_high)
        blue_after = rng.randint(after_low, after_high)
        values.append(
            [
                blue_before,
                blue_before + before_gap,
                blue_after,
                blue_after + after_gap,
            ]
        )

    return {
        "values": values,
        "palette": int(seed) % len(PALETTES),
        "offset": [rng.randint(-8, 8), rng.randint(-6, 6)],
    }


def analytic_gold(scene: dict[str, Any]) -> str:
    scores = _scores(scene["values"])
    return "ABCD"[max(range(4), key=lambda index: scores[index])]


def margin(scene: dict[str, Any]) -> float:
    scores = sorted(_scores(scene["values"]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # The sampled representation is raster-injective; non-pixel-identical
    # palette and translation interventions are tested separately.
    return []


def _point_layout(panel: int, offset: list[int]) -> tuple[int, int, int, int, int, int]:
    row, col = divmod(panel, 2)
    ox, oy = offset
    panel_left = 55 + 420 * col + ox
    panel_top = 36 + 275 * row + oy
    plot_left = panel_left + 55
    plot_right = panel_left + 350
    plot_bottom = panel_top + 205
    before_x = plot_left + 80
    after_x = plot_right - 60
    return panel_left, panel_top, plot_left, plot_right, plot_bottom, before_x, after_x


def render(scene: dict[str, Any]) -> Image.Image:
    values = scene["values"]
    palette = PALETTES[int(scene["palette"])]
    offset = scene["offset"]

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), "#ffffff")
    draw = ImageDraw.Draw(image)

    def line(points: list[tuple[int, int]], fill: str, width: int = 1) -> None:
        draw.line([(x * SCALE, y * SCALE) for x, y in points], fill=fill, width=width * SCALE)

    def text(x: int, y: int, value: str, *, size: int = 13, bold: bool = False, fill: str = "#222222") -> None:
        draw.text((x * SCALE, y * SCALE), value, font=_font(size, bold), fill=fill)

    text(450, 7, "SIGNED CHANGE OF THE ORANGE-BLUE GAP", size=15, bold=True, fill="#333333")

    for panel, panel_values in enumerate(values):
        panel_left, panel_top, plot_left, plot_right, plot_bottom, before_x, after_x = _point_layout(panel, offset)
        plot_top = plot_bottom - 160
        label = "ABCD"[panel]
        text(panel_left + 3, panel_top + 1, label, size=17, bold=True)

        for tick in (0, 40, 80, 120, 160):
            y = plot_bottom - tick
            line([(plot_left, y), (plot_right, y)], "#e2e5e8", 1)
        line([(plot_left, plot_top), (plot_left, plot_bottom), (plot_right, plot_bottom)], "#6b7280", 2)
        line([(plot_right, plot_bottom), (plot_right, plot_top), (plot_left, plot_top)], "#b8bec5", 1)
        text(before_x - 29, plot_bottom + 8, "BEFORE", size=11, fill="#4b5563")
        text(after_x - 23, plot_bottom + 8, "AFTER", size=11, fill="#4b5563")

        blue_before, orange_before, blue_after, orange_after = panel_values
        for x, value, color in (
            (before_x, blue_before, palette["blue"]),
            (before_x, orange_before, palette["orange"]),
            (after_x, blue_after, palette["blue"]),
            (after_x, orange_after, palette["orange"]),
        ):
            y = plot_bottom - int(value)
            radius = 9
            draw.ellipse(
                ((x - radius) * SCALE, (y - radius) * SCALE, (x + radius) * SCALE, (y + radius) * SCALE),
                fill=color,
                outline=color,
                width=1 * SCALE,
            )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
