"""Independent image-only continuation-graph oracle; receives pixels only."""
from __future__ import annotations
from collections import deque
import numpy as np
from PIL import Image

def _components(mask: np.ndarray, minimum: int = 20):
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]: continue
        q = [(int(y), int(x))]; seen[y, x] = True; pts = []
        while q:
            yy, xx = q.pop(); pts.append((yy, xx))
            for dy in (-1,0,1):
                for dx in (-1,0,1):
                    ny,nx=yy+dy,xx+dx
                    if 0<=ny<h and 0<=nx<w and not seen[ny,nx] and mask[ny,nx]:
                        seen[ny,nx]=True; q.append((ny,nx))
        if len(pts) >= minimum:
            a=np.asarray(pts); out.append({"pts":a,"y":float(a[:,0].mean()),"x":float(a[:,1].mean()),"n":len(pts)})
    return out

def _line_score(mask, a, b):
    ax,ay=a; bx,by=b
    ts=np.linspace(.10,.90,160)
    xs=np.rint(ax+(bx-ax)*ts).astype(int); ys=np.rint(ay+(by-ay)*ts).astype(int)
    h,w=mask.shape; scores=[]
    for x,y in zip(xs,ys):
        y0=max(0,y-4);y1=min(h,y+5);x0=max(0,x-4);x1=min(w,x+5)
        scores.append(bool(mask[y0:y1,x0:x1].any()))
    return float(np.mean(scores))

def decision_from_image(image: Image.Image) -> str:
    a=np.asarray(image.convert("RGB"),dtype=np.int16)
    r,g,b=a[:,:,0],a[:,:,1],a[:,:,2]
    red=(r>160)&(r>g*1.8)&(r>b*1.8)
    blue=(b>110)&(b>r*1.35)&(b>g*1.05)
    mag=(r>110)&(b>100)&(r>g*1.7)&(b>g*1.5)
    dark=(r<90)&(g<90)&(b<100)
    roots=_components(red,50); leaves=_components(mag,100); nodes=_components(dark,50)
    if len(roots)!=1 or len(leaves)<8 or len(nodes)<7:
        raise ValueError("visible continuation graph is too incomplete")
    root=roots[0]; all_nodes=[root]+nodes+leaves
    # Infer the five visible node columns, then reconstruct supported edges only
    # between consecutive columns.  No renderer coordinates are assumed.
    levels=[]
    for item in sorted(all_nodes,key=lambda z:z["x"]):
        if not levels or abs(item["x"]-np.mean([q["x"] for q in levels[-1]]))>20:
            levels.append([item])
        else: levels[-1].append(item)
    if len(levels)!=5 or len(levels[0])!=1:
        raise ValueError("continuation levels are incomplete")
    cur=root
    rules=("upper","lower","upper","lower")
    for rule in rules:
        # Count the already crossed x-columns from the current node.
        depth=min(range(len(levels)),key=lambda k:abs(np.mean([q["x"] for q in levels[k]])-cur["x"]))
        forward=[v for v in levels[depth+1] if _line_score(blue,(cur["x"],cur["y"]),(v["x"],v["y"]))>.88]
        if len(forward)!=2: raise ValueError("a routed fork was erased or ambiguous")
        forward.sort(key=lambda v: v["y"])
        cur=forward[0] if rule=="upper" else forward[1]
    target=cur
    pts=target["pts"]; yy,xx=pts[:,0],pts[:,1]
    box=(yy.max()-yy.min()+1)*(xx.max()-xx.min()+1)
    fill=len(pts)/float(box)
    return "square" if fill>.88 else "circle"
