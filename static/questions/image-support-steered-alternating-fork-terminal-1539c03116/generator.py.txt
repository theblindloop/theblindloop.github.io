"""Deterministic renderer for a branching-track navigation world."""
from __future__ import annotations

import random
from PIL import Image, ImageDraw

PALETTES = [
    {"track": (35, 91, 181), "root": (220, 48, 55), "node": (35, 38, 45), "leaf": (178, 45, 155), "bg": (247, 244, 235)},
    {"track": (24, 118, 143), "root": (218, 67, 42), "node": (43, 35, 52), "leaf": (156, 50, 184), "bg": (250, 246, 238)},
]

def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    w, h = 640, 512
    xs = [48, 176, 304, 432, 584]
    levels = [[256.0]]
    # Every generation changes the embedding while maintaining vertical order.
    for depth in range(1, 5):
        n = 2 ** depth
        base = [(i + 0.5) * h / n for i in range(n)]
        jitter = min(10.0, h / n * 0.12)
        levels.append([round(y + rng.uniform(-jitter, jitter), 2) for y in base])
    target = 5  # upper, lower, upper, lower in a top-first binary tree
    wanted = "circle" if int(seed) % 2 == 0 else "square"
    shapes = ["circle"] * 8 + ["square"] * 8
    rng.shuffle(shapes)
    other = "square" if wanted == "circle" else "circle"
    if shapes[target] != wanted:
        swap = next(i for i, value in enumerate(shapes) if value == wanted and i != target)
        shapes[target], shapes[swap] = shapes[swap], shapes[target]
    assert shapes.count("circle") == shapes.count("square") == 8
    return {"width": w, "height": h, "level_x": xs, "node_y": levels, "leaf_shapes": shapes, "palette": int(seed) % len(PALETTES)}

def analytic_gold(scene: dict) -> str:
    return str(scene["leaf_shapes"][5])

def margin(scene: dict) -> float:
    return 0.18

def is_quarantined(scene: dict) -> bool:
    return False

def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []

def render(scene: dict) -> Image.Image:
    w, h = int(scene["width"]), int(scene["height"])
    xs = [float(x) for x in scene["level_x"]]
    ys = [[float(y) for y in level] for level in scene["node_y"]]
    shapes = list(scene["leaf_shapes"])
    p = PALETTES[int(scene["palette"])]
    im = Image.new("RGB", (w, h), p["bg"])
    d = ImageDraw.Draw(im)
    # The entire tree is one continuous thin track; lines are drawn first under nodes.
    for depth in range(4):
        for i, y in enumerate(ys[depth]):
            for child in (2 * i, 2 * i + 1):
                d.line((xs[depth], y, xs[depth + 1], ys[depth + 1][child]), fill=p["track"], width=7)
    for depth in range(1, 4):
        for y in ys[depth]:
            x = xs[depth]
            d.ellipse((x-7, y-7, x+7, y+7), fill=p["node"])
    x0, y0 = xs[0], ys[0][0]
    d.ellipse((x0-10, y0-10, x0+10, y0+10), fill=p["root"])
    for i, y in enumerate(ys[4]):
        x = xs[4]
        if shapes[i] == "circle":
            d.ellipse((x-11, y-11, x+11, y+11), fill=p["leaf"])
        else:
            d.rectangle((x-10, y-10, x+10, y+10), fill=p["leaf"])
    return im
