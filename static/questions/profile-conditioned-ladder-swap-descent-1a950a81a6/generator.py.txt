"""Scene sampler, renderer and analytic gold for the panel identity relay world.

Four ordered panels show the same square scene at successive times. Five discs
move a little from panel to panel, and the five colours are re-dealt in every
panel, so colour carries no identity. A black ring in the first panel marks one
disc. The question is which colour that same disc wears in the final panel; the
only channel that carries identity across panels is motion continuity.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "ladder_swap_descent-renderer-0.1.0"
QUARANTINE_PX = 8.0

W, H = 900, 330
PANEL = 200
GAP = 12
LEFT = 22
TOP = 26
N_PANELS = 4
N_OBJ = 5
R = 13
RING_R = 20
PAD = 32
MIN_SEP = 54.0
DISP_LO, DISP_HI = 17.0, 32.0
CORR_MIN = 12.0

FRAME = (170, 175, 185)
ARROW = (150, 155, 165)
RING = (20, 20, 20)
WHITE = (255, 255, 255)

NAMES = ("blue", "green", "orange", "purple", "yellow")
COLOURS = {
    "blue": (30, 80, 200),
    "green": (30, 140, 60),
    "orange": (240, 150, 20),
    "purple": (140, 60, 180),
    "yellow": (235, 200, 40),
}


def _origin(k: int) -> tuple[int, int]:
    return LEFT + k * (PANEL + GAP), TOP


def _d(a, b) -> float:
    return math.hypot(a[0] - b[0], a[1] - b[1])


def _corr_margin(tracks: list[list[list[float]]]) -> float:
    """Smallest drawn clearance of the nearest-successor matching, in pixels."""
    best = float("inf")
    for k in range(len(tracks[0]) - 1):
        p = [t[k] for t in tracks]
        q = [t[k + 1] for t in tracks]
        for j in range(len(p)):
            true_d = _d(q[j], p[j])
            fwd = min(_d(q[i], p[j]) for i in range(len(p)) if i != j)
            bwd = min(_d(q[j], p[i]) for i in range(len(p)) if i != j)
            best = min(best, fwd - true_d, bwd - true_d)
    return best


def _spread(rng: random.Random) -> list[list[float]]:
    lo, hi = PAD, PANEL - PAD
    for _ in range(4000):
        pts: list[list[float]] = []
        for _ in range(N_OBJ):
            for _ in range(400):
                c = [rng.uniform(lo, hi), rng.uniform(lo, hi)]
                if all(_d(c, o) >= MIN_SEP for o in pts):
                    pts.append(c)
                    break
            else:
                break
        if len(pts) == N_OBJ:
            return pts
    raise RuntimeError("could not place the discs")


def _step(rng: random.Random, prev: list[list[float]]) -> list[list[float]] | None:
    lo, hi = PAD, PANEL - PAD
    for _ in range(600):
        nxt = []
        for p in prev:
            for _ in range(200):
                a = rng.uniform(0, 2 * math.pi)
                r = rng.uniform(DISP_LO, DISP_HI)
                c = [p[0] + r * math.cos(a), p[1] + r * math.sin(a)]
                if lo <= c[0] <= hi and lo <= c[1] <= hi:
                    nxt.append(c)
                    break
            else:
                break
        if len(nxt) != N_OBJ:
            continue
        if min(_d(nxt[i], nxt[j]) for i in range(N_OBJ) for j in range(i + 1, N_OBJ)) < MIN_SEP:
            continue
        if _corr_margin([[p, n] for p, n in zip(prev, nxt)]) < CORR_MIN:
            continue
        return nxt
    return None


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    mode = seed % 8
    tight = mode == 3
    near = mode == 6

    while True:
        tracks = [[p] for p in _spread(rng)]
        ok = True
        for _ in range(N_PANELS - 1):
            nxt = _step(rng, [t[-1] for t in tracks])
            if nxt is None:
                ok = False
                break
            for t, c in zip(tracks, nxt):
                t.append(c)
        if not ok:
            continue

        # the answer must not be readable by matching the marked object's
        # first-panel place straight to the final panel, so only objects whose
        # own final place is not the nearest one to their start may be marked
        first = [t[0] for t in tracks]
        last = [t[-1] for t in tracks]
        eligible = [
            j
            for j in range(N_OBJ)
            if min(range(N_OBJ), key=lambda i: _d(last[i], first[j])) != j
        ]
        if not eligible:
            continue
        break

    tracked = rng.choice(eligible)
    target = NAMES[seed % 5]

    # the ring is drawn displaced from its disc towards a neighbouring disc, so
    # the nearest-disc reading has an explicit, drawn clearance
    p0 = [t[0] for t in tracks]
    other = min((j for j in range(N_OBJ) if j != tracked), key=lambda j: _d(p0[j], p0[tracked]))
    span = _d(p0[other], p0[tracked])
    if tight:
        want = rng.uniform(2.5, 5.5)
    elif near:
        want = rng.uniform(8.6, 10.4)
    else:
        want = rng.uniform(span - 12.0, span - 2.0)
    shift = (span - want) / 2.0
    ux = (p0[other][0] - p0[tracked][0]) / span
    uy = (p0[other][1] - p0[tracked][1]) / span
    ring = [p0[tracked][0] + shift * ux, p0[tracked][1] + shift * uy]

    # colours are re-dealt independently in every panel; the last deal is chosen
    # so the tracked disc wears the balanced target colour
    deals = []
    for k in range(N_PANELS):
        deal = list(NAMES)
        rng.shuffle(deal)
        if k == N_PANELS - 1:
            j = deal.index(target)
            deal[j], deal[tracked] = deal[tracked], deal[j]
        deals.append(deal)
    # block the static shortcut: the disc that ends nearest the tracked disc's
    # first-panel place must not wear the answer colour
    last = [t[-1] for t in tracks]
    naive = min(range(N_OBJ), key=lambda j: _d(last[j], p0[tracked]))
    if naive != tracked and deals[-1][naive] == target:  # pragma: no cover - defensive
        raise RuntimeError("colour deal collision")
    if deals[0][tracked] == target:
        alt = next(n for n in NAMES if n != target and n != deals[0][naive])
        j = deals[0].index(alt)
        deals[0][j], deals[0][tracked] = deals[0][tracked], deals[0][j]

    objects = [
        {
            "pos": [[round(x, 3), round(y, 3)] for x, y in tracks[j]],
            "colours": [deals[k][j] for k in range(N_PANELS)],
        }
        for j in range(N_OBJ)
    ]
    rng.shuffle(objects)  # the bag order carries no information
    return {
        "canvas": [W, H],
        "objects": objects,
        "ring": [round(ring[0], 3), round(ring[1], 3)],
        "tight": bool(tight),
        "near": bool(near),
    }


def _tracked_index(objects: list[dict], ring: list[float]) -> int:
    return min(range(len(objects)), key=lambda j: _d(objects[j]["pos"][0], ring))


def analytic_gold(scene: dict) -> str:
    objects = scene["objects"]
    cur = _tracked_index(objects, scene["ring"])
    n = len(objects[0]["pos"])
    place = objects[cur]["pos"][0]
    for k in range(1, n):
        cur = min(range(len(objects)), key=lambda j: _d(objects[j]["pos"][k], place))
        place = objects[cur]["pos"][k]
    return objects[cur]["colours"][-1]


def margin(scene: dict) -> float:
    objects = scene["objects"]
    ring = scene["ring"]
    first = sorted(_d(o["pos"][0], ring) for o in objects)
    corr = _corr_margin([o["pos"] for o in objects])
    return round(min(corr, first[1] - first[0]), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    objs = scene["objects"]
    out = []
    rev = dict(scene)
    rev["objects"] = list(reversed(objs))
    out.append(("object_bag_reversed", rev))
    rot = dict(scene)
    rot["objects"] = objs[2:] + objs[:2]
    out.append(("object_bag_rotated", rot))
    return out


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (W, H), WHITE)
    draw = ImageDraw.Draw(image)
    objects = scene["objects"]
    n = len(objects[0]["pos"])
    for k in range(n):
        ox, oy = _origin(k)
        draw.rectangle([ox, oy, ox + PANEL, oy + PANEL], outline=FRAME, width=3)
        for obj in objects:
            x, y = obj["pos"][k]
            cx, cy = int(round(ox + x)), int(round(oy + y))
            draw.ellipse([cx - R, cy - R, cx + R, cy + R], fill=COLOURS[obj["colours"][k]])
    ox, oy = _origin(0)
    rx, ry = scene["ring"]
    cx, cy = int(round(ox + rx)), int(round(oy + ry))
    draw.ellipse([cx - RING_R, cy - RING_R, cx + RING_R, cy + RING_R], outline=RING, width=5)

    ay = TOP + PANEL + 40
    x0 = LEFT
    x1 = LEFT + (N_PANELS - 1) * (PANEL + GAP) + PANEL
    draw.line([x0, ay, x1 - 26, ay], fill=ARROW, width=5)
    draw.polygon([(x1, ay), (x1 - 28, ay - 13), (x1 - 28, ay + 13)], fill=ARROW)
    return image
