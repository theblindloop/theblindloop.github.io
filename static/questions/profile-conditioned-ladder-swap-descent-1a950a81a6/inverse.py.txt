"""Scene-free pixel inverse arm for the panel identity relay world.

Input is one rasterised PNG and nothing else. The arm recovers the panel row and
its left-to-right order from the drawn frames, the discs from their drawn
colours, the marked disc from the drawn black ring, then replays the declared
nearest-place matching from panel to panel and reports the colour the marked
disc wears in the final panel. It never sees the latent scene.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "ladder_swap_descent-oracle-0.1.0"

_FRAME = (170, 175, 185)
_RING = (20, 20, 20)
_PALETTE = {
    "blue": (30, 80, 200),
    "green": (30, 140, 60),
    "orange": (240, 150, 20),
    "purple": (140, 60, 180),
    "yellow": (235, 200, 40),
}


class OracleError(RuntimeError):
    """Raised when the image does not present a readable panel row."""


def _mask(arr: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    return (arr[:, :, 0] == rgb[0]) & (arr[:, :, 1] == rgb[1]) & (arr[:, :, 2] == rgb[2])


def _runs(profile: np.ndarray, thresh: float) -> list[tuple[int, int]]:
    out: list[tuple[int, int]] = []
    start = None
    for i, v in enumerate(profile):
        if v >= thresh and start is None:
            start = i
        elif v < thresh and start is not None:
            out.append((start, i - 1))
            start = None
    if start is not None:
        out.append((start, len(profile) - 1))
    return out


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    frame = _mask(arr, _FRAME)
    if frame.sum() < 400:
        raise OracleError("no panel frames found")

    cols = frame.sum(axis=0).astype(float)
    edges = _runs(cols, 0.5 * cols.max())
    if len(edges) % 2 or len(edges) < 4:
        raise OracleError(f"expected paired vertical frame edges, saw {len(edges)}")
    panels = [
        (edges[i][0], edges[i + 1][1]) for i in range(0, len(edges), 2)
    ]
    panels.sort(key=lambda b: b[0])

    rows = frame.sum(axis=1).astype(float)
    hedges = _runs(rows, 0.5 * rows.max())
    if len(hedges) != 2:
        raise OracleError(f"expected one top and one bottom frame edge, saw {len(hedges)}")
    y0 = hedges[0][0]

    # discs: exactly one of each palette colour inside each panel
    discs: list[dict[str, tuple[float, float]]] = []
    for x0, x1 in panels:
        here: dict[str, tuple[float, float]] = {}
        for name, rgb in _PALETTE.items():
            sub = _mask(arr[:, x0 : x1 + 1], rgb)
            if sub.sum() < 120:
                raise OracleError(f"panel at x={x0} is missing the {name} disc")
            ys, xs = np.nonzero(sub)
            here[name] = (float(xs.mean()) + x0 - x0, float(ys.mean()) - y0)
        discs.append(here)
    if len(discs) < 2:
        raise OracleError("need at least two panels to replay")

    ring = _mask(arr, _RING)
    if ring.sum() < 150:
        raise OracleError("no ring found")
    rys, rxs = np.nonzero(ring)
    rcx, rcy = float(rxs.mean()), float(rys.mean())
    start = None
    for idx, (x0, x1) in enumerate(panels):
        if x0 <= rcx <= x1:
            start = idx
            break
    if start is None:
        raise OracleError("the ring lies outside every panel")

    local = (rcx - panels[start][0], rcy - y0)
    name = min(
        discs[start],
        key=lambda n: (discs[start][n][0] - local[0]) ** 2 + (discs[start][n][1] - local[1]) ** 2,
    )
    place = discs[start][name]
    for k in range(start + 1, len(panels)):
        name = min(
            discs[k],
            key=lambda n: (discs[k][n][0] - place[0]) ** 2 + (discs[k][n][1] - place[1]) ** 2,
        )
        place = discs[k][name]
    return name
