"""Independent PNG-only inverse for Odd Link Network."""

from __future__ import annotations

from collections import Counter, deque

import numpy as np
from PIL import Image

WIDTH, HEIGHT = 1200, 800
DECISIONS = ("A", "B", "C", "D")
INK = np.array((42, 47, 57), dtype=np.int16)
NODE_COLORS = {
    "S": (218, 65, 65), "A": (247, 174, 47), "B": (60, 132, 222),
    "C": (99, 181, 75), "D": (153, 88, 205), "E": (42, 172, 170),
    "F": (231, 95, 148), "G": (163, 105, 61), "H": (60, 79, 151),
    "J": (72, 190, 222), "K": (205, 151, 37), "L": (38, 147, 91),
    "M": (121, 72, 153),
}
PANEL_ORIGINS = ((25, 155), (420, 155), (815, 155))
ROW_OFFSETS = (105, 205, 305, 405, 505)


def _badge_id(arr: np.ndarray, cx: int, cy: int) -> str | None:
    patch = arr[cy - 21:cy + 22, cx - 21:cx + 22]
    counts = {}
    for node, rgb in NODE_COLORS.items():
        ref = np.array(rgb, dtype=np.int16)
        counts[node] = int((np.abs(patch.astype(np.int16) - ref).max(axis=2) <= 3).sum())
    node, count = max(counts.items(), key=lambda item: item[1])
    ordered = sorted(counts.values(), reverse=True)
    if count < 650 or ordered[1] > 70:
        return None
    return node


def _link_present(arr: np.ndarray, lx: int, rx: int, cy: int) -> bool:
    strip = arr[cy - 5:cy + 6, lx + 36:rx - 36].astype(np.int16)
    inkish = np.abs(strip - INK).max(axis=2) <= 3
    return float(inkish.mean()) >= 0.86


def _read_occurrences(image: Image.Image) -> list[tuple[str, str]] | None:
    if image.size != (WIDTH, HEIGHT):
        return None
    arr = np.asarray(image.convert("RGB"))
    out = []
    for x0, y0 in PANEL_ORIGINS:
        for dy in ROW_OFFSETS:
            cy, lx, rx = y0 + dy, x0 + 82, x0 + 278
            left = _badge_id(arr, lx, cy)
            right = _badge_id(arr, rx, cy)
            if left is None or right is None or left == right:
                return None
            if not _link_present(arr, lx, rx, cy):
                return None
            out.append(tuple(sorted((left, right))))
    return out


def decision_from_image(image: Image.Image) -> str:
    occurrences = _read_occurrences(image)
    if occurrences is None:
        return "abstain"
    counts = Counter(occurrences)
    graph = {node: set() for node in NODE_COLORS}
    for (left, right), count in counts.items():
        if count % 2:
            graph[left].add(right)
            graph[right].add(left)
    seen = {"S"}
    queue = deque(["S"])
    while queue:
        cur = queue.popleft()
        for nxt in graph[cur]:
            if nxt not in seen:
                seen.add(nxt)
                queue.append(nxt)
    matches = [node for node in DECISIONS if node in seen]
    return matches[0] if len(matches) == 1 else "abstain"
