"""Deterministic renderer and analytic arm for Odd Link Network."""

from __future__ import annotations

from collections import Counter, deque
import copy
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1200, 800
BACKGROUND = (245, 242, 235)
INK = (42, 47, 57)
MUTED = (84, 91, 104)
CARD = (255, 255, 252)
CARD_EDGE = (91, 100, 114)
DECISIONS = ("A", "B", "C", "D")
NODE_COLORS = {
    "S": (218, 65, 65), "A": (247, 174, 47), "B": (60, 132, 222),
    "C": (99, 181, 75), "D": (153, 88, 205), "E": (42, 172, 170),
    "F": (231, 95, 148), "G": (163, 105, 61), "H": (60, 79, 151),
    "J": (72, 190, 222), "K": (205, 151, 37), "L": (38, 147, 91),
    "M": (121, 72, 153),
}
PANEL_BOXES = ((25, 155, 385, 745), (420, 155, 780, 745), (815, 155, 1175, 745))
ROW_OFFSETS = (105, 205, 305, 405, 505)
FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(FONT_PATH, size)


def _canon(edge: list[str] | tuple[str, str]) -> tuple[str, str]:
    return tuple(sorted(edge))


def _surviving_edges(maps: list[list[list[str]]]) -> list[tuple[str, str]]:
    counts = Counter(_canon(edge) for card in maps for edge in card)
    return sorted(edge for edge, count in counts.items() if count % 2 == 1)


def _component(edges: list[tuple[str, str]], start: str = "S") -> set[str]:
    graph = {node: set() for node in NODE_COLORS}
    for left, right in edges:
        graph[left].add(right)
        graph[right].add(left)
    seen = {start}
    queue = deque([start])
    while queue:
        cur = queue.popleft()
        for nxt in graph[cur]:
            if nxt not in seen:
                seen.add(nxt)
                queue.append(nxt)
    return seen


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = DECISIONS[int(seed) % 4]
    excluded = [node for node in DECISIONS if node != target]
    rng.shuffle(excluded)
    x0, x1, x2 = excluded

    # Five odd edges form the answer path. Each other edge is shown on exactly
    # two different maps, so it is present in the tempting raw union but absent
    # from the parity graph. Every map has the same visible density.
    maps = [
        [["S", "E"], ["G", "H"], [x0, "E"], [x2, "G"], ["J", "K"]],
        [["E", "F"], ["H", target], [x0, "E"], [x1, "F"], ["L", "M"]],
        [["F", "G"], [x1, "F"], [x2, "G"], ["J", "K"], ["L", "M"]],
    ]
    row_order = []
    endpoint_flips = []
    for _ in range(3):
        order = list(range(5))
        rng.shuffle(order)
        row_order.append(order)
        endpoint_flips.append([bool(rng.getrandbits(1)) for _ in range(5)])
    map_order = [0, 1, 2]
    rng.shuffle(map_order)
    scene = {
        "maps": maps,
        "map_order": map_order,
        "row_order": row_order,
        "endpoint_flips": endpoint_flips,
    }
    assert not is_quarantined(scene)
    return scene


def analytic_gold(scene: dict) -> str:
    seen = _component(_surviving_edges(scene["maps"]))
    matches = [node for node in DECISIONS if node in seen]
    return matches[0] if len(matches) == 1 else "abstain"


def margin(scene: dict) -> float:
    return 1.0 if analytic_gold(scene) in DECISIONS else 0.0


def is_quarantined(scene: dict) -> bool:
    answer = analytic_gold(scene)
    if answer not in DECISIONS:
        return True
    raw_union = sorted({_canon(edge) for card in scene["maps"] for edge in card})
    raw_seen = _component(raw_union)
    if not all(node in raw_seen for node in DECISIONS):
        return True
    counts = Counter(_canon(edge) for card in scene["maps"] for edge in card)
    return sorted(counts.values()).count(2) < 5


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    transformed = copy.deepcopy(scene)
    transformed["endpoint_flips"] = [
        [not flag for flag in card] for card in transformed["endpoint_flips"]
    ]
    return [("reverse_all_undirected_link_records", transformed)]


def _visible_endpoints(edge: list[str], flip: bool) -> tuple[str, str]:
    endpoints = list(reversed(edge)) if flip else list(edge)
    return tuple(sorted(endpoints))


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((WIDTH // 2, 28), "ODD LINK OVERLAY", fill=INK, font=_font(31), anchor="mm")
    draw.text((WIDTH // 2, 66), "same endpoint pair = same link", fill=MUTED,
              font=_font(19), anchor="mm")
    draw.rounded_rectangle((250, 89, 950, 137), radius=18, fill=(235, 238, 242),
                           outline=CARD_EDGE, width=3)
    draw.text((WIDTH // 2, 113), "shown once = KEEP     shown twice = CANCEL",
              fill=INK, font=_font(21), anchor="mm")

    for visible_map, box in enumerate(PANEL_BOXES):
        x0, y0, x1, y1 = box
        draw.rounded_rectangle(box, radius=18, fill=CARD, outline=CARD_EDGE, width=4)
        draw.text(((x0 + x1) // 2, y0 + 37), f"MAP {visible_map + 1}",
                  fill=INK, font=_font(23), anchor="mm")
        logical_map = scene["map_order"][visible_map]
        for visible_row, dy in enumerate(ROW_OFFSETS):
            edge_index = scene["row_order"][logical_map][visible_row]
            edge = scene["maps"][logical_map][edge_index]
            flip = scene["endpoint_flips"][logical_map][edge_index]
            left, right = _visible_endpoints(edge, flip)
            cy = y0 + dy
            lx, rx = x0 + 82, x0 + 278
            draw.line((lx, cy, rx, cy), fill=INK, width=13)
            for node, cx in ((left, lx), (right, rx)):
                draw.ellipse((cx - 29, cy - 29, cx + 29, cy + 29),
                             fill=NODE_COLORS[node], outline=INK, width=4)
                text_fill = (255, 255, 255) if node in {"S", "H", "M"} else (24, 27, 33)
                draw.text((cx, cy + 1), node, fill=text_fill, font=_font(25), anchor="mm")
            if "S" in (left, right):
                sx = lx if left == "S" else rx
                draw.text((sx, cy - 42), "START", fill=(177, 47, 47), font=_font(13), anchor="mm")
    return image
