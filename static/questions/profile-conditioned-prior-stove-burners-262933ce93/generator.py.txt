"""Prior-conflict stovetop world: count the visible burners.

The latent scene holds a domestic stovetop with ``n`` burners. A familiar
canonical prior asserts that a standard home range has FOUR burners; the
counterfactual renders a non-canonical count (three or five) and the decision
must be read from the visible discs alone, never from the prior. The pixel
oracle counts the actual burner discs.

Latent alias mode is ``not_applicable``: every latent field is directly
reflected in the visible discs (count, radius, position, canvas), so the
rendering is injective in the latent scene and there is no render-invariant
alias. The canonical-prior name "stovetop" is a matched-control/counterfactual
manipulation of the *question*, not of any latent field.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

CANVAS = 512
CX = 256.0
CY = 256.0
# Slot grid spacing (columns twice as wide as rows so burners never touch).
SX = 116.0
SY = 104.0
BURNER_RADIUS_MIN = 30.0
BURNER_RADIUS_MAX = 38.0
JITTER = 5.0

# Supported off-quarantine burner counts. The canonical prior is 4; 3 and 5
# are the counterfactuals that contradict it. In every supported layout the
# minimum centre-to-centre spacing is ~104 px, far above 2*radius, so each
# burner is an isolated disc.
SUPPORTED_N = (4, 3, 5)

# A count is quarantined when the burner discs crowd so closely that adjacent
# discs merge into one blob at pixel scale (gap < 4 px). Supported layouts
# never approach this.
QUAR_MIN_CLEARANCE = 4.0

SURFACE_COLOR = (72, 74, 80)   # dark range top
RING_COLOR = (168, 170, 176)   # light metal trim around each burner
BURNER_COLOR = (30, 31, 36)    # near-black burner disc


def _slot_centers(scene: dict) -> list[tuple[float, float]]:
    """Absolute pixel centres of the first n slots of a 2x3 burner grid."""
    n = int(scene["n"])
    gx = float(scene["grid_offset_x"])
    gy = float(scene["grid_offset_y"])
    centres = []
    for i in range(n):
        col = i % 2
        row = i // 2
        x = CX + gx + (SX if col else -SX)
        y = CY + gy + (SY * (row - 1))
        centres.append((x, y))
    return centres


def sample_scene(seed: int) -> dict:
    """Deterministic scene dict from an integer seed."""
    rng = random.Random(seed)
    n = SUPPORTED_N[rng.randrange(len(SUPPORTED_N))]
    # burner_radius, grid_offset_x, grid_offset_y vary independently of n so
    # that neither size nor position predicts the count (cheap-cue counterbalance).
    burner_radius = rng.uniform(BURNER_RADIUS_MIN, BURNER_RADIUS_MAX)
    grid_offset_x = rng.uniform(-JITTER, JITTER)
    grid_offset_y = rng.uniform(-JITTER, JITTER)
    return {
        "n": n,
        "burner_radius": round(burner_radius, 3),
        "grid_offset_x": round(grid_offset_x, 3),
        "grid_offset_y": round(grid_offset_y, 3),
        "canvas": CANVAS,
    }


def render(scene: dict) -> Image.Image:
    """Render the stovetop with its burners to an RGB image."""
    canvas = int(scene["canvas"])
    img = Image.new("RGB", (canvas, canvas), SURFACE_COLOR)
    draw = ImageDraw.Draw(img)
    radius = float(scene["burner_radius"])
    n = int(scene["n"])
    centres = _slot_centers(scene)
    for k in range(n):
        x, y = centres[k]
        # light trim ring, then the dark burner disc on top
        draw.ellipse(
            [x - radius - 6, y - radius - 6, x + radius + 6, y + radius + 6],
            fill=RING_COLOR,
        )
        draw.ellipse([x - radius, y - radius, x + radius, y + radius], fill=BURNER_COLOR)
    return img


def analytic_gold(scene: dict) -> str:
    """Decision = the visible count of burner discs, as a string."""
    return str(scene["n"])


def _min_pairwise(centres: list[tuple[float, float]]) -> float:
    best = float("inf")
    for a in range(len(centres)):
        for b in range(a + 1, len(centres)):
            dx = centres[a][0] - centres[b][0]
            dy = centres[a][1] - centres[b][1]
            best = min(best, math.hypot(dx, dy))
    return best


def margin(scene: dict) -> float:
    """Verifiability margin = clearance between the closest burner discs."""
    centres = _slot_centers(scene)
    if len(centres) < 2:
        return float("inf")
    return _min_pairwise(centres) - 2.0 * float(scene["burner_radius"])


def is_quarantined(scene: dict) -> bool:
    """Quarantine crowded layouts where disc edges can merge at pixel scale."""
    if int(scene["n"]) not in SUPPORTED_N:
        return True
    return margin(scene) < QUAR_MIN_CLEARANCE


def latent_symmetries(scene: dict):
    """not_applicable: empty, the latent rendering is injective."""
    return []
