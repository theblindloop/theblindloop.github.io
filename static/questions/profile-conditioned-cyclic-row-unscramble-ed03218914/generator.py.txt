"""Deterministic renderer and analytic arm for the cyclic-row world."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw


WIDTH, HEIGHT = 520, 420
GRID = 9
SYMBOLS = ("diamond", "square", "plus", "x")
QUARANTINE_MARGIN = 0.10


def _points(*pairs: tuple[int, int]) -> set[tuple[int, int]]:
    return set(pairs)


TEMPLATES = {
    "diamond": _points(
        (4, 0), (3, 1), (5, 1), (2, 2), (6, 2), (1, 3), (7, 3), (0, 4),
        (8, 4), (1, 5), (7, 5), (2, 6), (6, 6), (3, 7), (5, 7), (4, 8),
    ),
    "square": _points(
        (2, 2), (3, 2), (4, 2), (5, 2), (6, 2), (2, 3), (6, 3), (2, 4),
        (6, 4), (2, 5), (6, 5), (2, 6), (3, 6), (4, 6), (5, 6), (6, 6),
    ),
    "plus": _points(
        (4, 0), (4, 1), (3, 1), (4, 2), (4, 3), (0, 4), (1, 4), (2, 4),
        (3, 4), (4, 4), (5, 4), (6, 4), (7, 4), (8, 4), (4, 5), (4, 6),
    ),
    "x": _points(
        (0, 0), (8, 0), (1, 1), (7, 1), (2, 2), (6, 2), (3, 3), (5, 3),
        (4, 4), (3, 5), (5, 5), (2, 6), (6, 6), (1, 7), (7, 7), (0, 8),
    ),
}


PALETTES = (
    {"background": (248, 246, 240), "frame": (0, 174, 213), "tab": (229, 37, 139),
     "ink": (25, 28, 31), "grid": (205, 207, 205)},
    {"background": (243, 247, 250), "frame": (20, 160, 210), "tab": (210, 45, 150),
     "ink": (34, 30, 38), "grid": (198, 205, 212)},
)


def _noisy_grid(scene: dict) -> list[list[int]]:
    symbol = scene["symbol"]
    grid = [[int((column, row) in TEMPLATES[symbol]) for column in range(GRID)] for row in range(GRID)]
    for row, column in scene["flips"]:
        grid[row][column] = 1 - grid[row][column]
    return grid


def sample_scene(seed: int) -> dict:
    symbol = SYMBOLS[seed % len(SYMBOLS)]
    block = seed // len(SYMBOLS)
    rng = random.Random(block * 104729 + 8191)
    row_shifts = [rng.randrange(GRID) for _ in range(GRID)]
    cell_size = 32
    panel_x = 104 + rng.randrange(-30, 31)
    panel_y = 70 + rng.randrange(-15, 16)
    palette = block % len(PALETTES)

    occupied = sorted((row, column) for column, row in TEMPLATES[symbol])
    empty = sorted((row, column) for row in range(GRID) for column in range(GRID)
                   if (column, row) not in TEMPLATES[symbol])
    noise_pairs = block % 3
    flips: list[list[int]] = []
    if noise_pairs:
        off = rng.sample(occupied, noise_pairs)
        on = rng.sample(empty, noise_pairs)
        flips = [list(item) for item in off + on]

    distractors = []
    for index in range(6):
        side = index % 2
        x = rng.randrange(18, 64) if side == 0 else rng.randrange(456, 502)
        y = rng.randrange(44, 382)
        radius = rng.randrange(5, 11)
        tone = rng.randrange(0, 2)
        distractors.append([x, y, radius, tone])
    return {
        "symbol": symbol,
        "row_shifts": row_shifts,
        "flips": flips,
        "panel_x": panel_x,
        "panel_y": panel_y,
        "cell_size": cell_size,
        "palette": palette,
        "alias_turns": 0 if block % 2 == 0 else GRID,
        "distractors": distractors,
    }


def render(scene: dict) -> Image.Image:
    palette = PALETTES[int(scene["palette"]) % len(PALETTES)]
    image = Image.new("RGB", (WIDTH, HEIGHT), palette["background"])
    draw = ImageDraw.Draw(image)
    for x, y, radius, tone in scene["distractors"]:
        fill = (225, 184, 92) if tone == 0 else (139, 178, 122)
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=fill)

    x0, y0 = int(scene["panel_x"]), int(scene["panel_y"])
    cell = int(scene["cell_size"])
    size = GRID * cell
    for index in range(1, GRID):
        x = x0 + index * cell
        y = y0 + index * cell
        draw.line((x, y0, x, y0 + size), fill=palette["grid"], width=1)
        draw.line((x0, y, x0 + size, y), fill=palette["grid"], width=1)
    draw.rectangle((x0, y0, x0 + size, y0 + size), outline=palette["frame"], width=5)

    grid = _noisy_grid(scene)
    alias = int(scene["alias_turns"]) % GRID
    inset = max(5, cell // 6)
    for row in range(GRID):
        shift = (int(scene["row_shifts"][row]) + alias) % GRID
        for column in range(GRID):
            if not grid[row][column]:
                continue
            display_column = (column + shift) % GRID
            left = x0 + display_column * cell + inset
            top = y0 + row * cell + inset
            draw.rectangle((left, top, left + cell - 2 * inset, top + cell - 2 * inset), fill=palette["ink"])
        marker_x = x0 + shift * cell + cell // 2
        marker_y = y0 + row * cell + cell // 2
        tab_half = 4
        draw.rectangle((marker_x - tab_half, marker_y - tab_half,
                        marker_x + tab_half, marker_y + tab_half), fill=palette["tab"])
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def _template_gap(scene: dict) -> float:
    grid = _noisy_grid(scene)
    observed = {(column, row) for row in range(GRID) for column in range(GRID) if grid[row][column]}
    disagreements = sorted(len(observed.symmetric_difference(template)) / float(GRID * GRID)
                           for template in TEMPLATES.values())
    return disagreements[1] - disagreements[0]


def margin(scene: dict) -> float:
    return _template_gap(scene)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["alias_turns"] = int(twin["alias_turns"]) + GRID
    return [("add_full_cyclic_turn", twin)]
