"""Independent image-only inverse for the cyclic-row world."""

from __future__ import annotations

import numpy as np
from PIL import Image


GRID = 9


def _points(*pairs: tuple[int, int]) -> set[tuple[int, int]]:
    return set(pairs)


TEMPLATES = {
    "diamond": _points(
        (4, 0), (3, 1), (5, 1), (2, 2), (6, 2), (1, 3), (7, 3), (0, 4),
        (8, 4), (1, 5), (7, 5), (2, 6), (6, 6), (3, 7), (5, 7), (4, 8),
    ),
    "square": _points(
        (2, 2), (3, 2), (4, 2), (5, 2), (6, 2), (2, 3), (6, 3), (2, 4),
        (6, 4), (2, 5), (6, 5), (2, 6), (3, 6), (4, 6), (5, 6), (6, 6),
    ),
    "plus": _points(
        (4, 0), (4, 1), (3, 1), (4, 2), (4, 3), (0, 4), (1, 4), (2, 4),
        (3, 4), (4, 4), (5, 4), (6, 4), (7, 4), (8, 4), (4, 5), (4, 6),
    ),
    "x": _points(
        (0, 0), (8, 0), (1, 1), (7, 1), (2, 2), (6, 2), (3, 3), (5, 3),
        (4, 4), (3, 5), (5, 5), (2, 6), (6, 6), (1, 7), (7, 7), (0, 8),
    ),
}


def _decode(image: Image.Image) -> tuple[set[tuple[int, int]], tuple[float, float, float, float]]:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[:, :, 0], array[:, :, 1], array[:, :, 2]
    cyan = (red < 80) & (green > 115) & (blue > 170) & (blue > red + 90)
    ys, xs = np.where(cyan)
    if len(xs) < 100:
        raise ValueError("cyan frame is missing")
    x0, x1 = float(xs.min()), float(xs.max())
    y0, y1 = float(ys.min()), float(ys.max())
    cell_x, cell_y = (x1 - x0) / GRID, (y1 - y0) / GRID
    if not 29 <= cell_x <= 35 or not 29 <= cell_y <= 35:
        raise ValueError("cyan frame geometry is invalid")

    magenta = (red > 175) & (green < 100) & (blue > 100) & (red > green + 100)
    dark = (red < 80) & (green < 80) & (blue < 80)
    observed = [[False] * GRID for _ in range(GRID)]
    shifts = []
    for row in range(GRID):
        cy = y0 + (row + 0.5) * cell_y
        row_band = magenta[max(0, int(cy - cell_y * 0.38)):int(cy + cell_y * 0.38) + 1,
                            max(0, int(x0)):int(x1) + 1]
        marker_y, marker_x = np.where(row_band)
        if len(marker_x) < 20:
            raise ValueError(f"magenta tab missing in row {row}")
        marker_center_x = x0 + float(marker_x.mean())
        shift = int(round((marker_center_x - x0) / cell_x - 0.5))
        if not 0 <= shift < GRID:
            raise ValueError("tab lies outside the frame")
        shifts.append(shift)
        for display_column in range(GRID):
            cx = x0 + (display_column + 0.5) * cell_x
            half = int(min(cell_x, cell_y) * 0.34)
            patch = dark[max(0, int(cy) - half):int(cy) + half + 1,
                         max(0, int(cx) - half):int(cx) + half + 1]
            observed[row][display_column] = int(patch.sum()) > 100

    canonical = set()
    for row in range(GRID):
        for column in range(GRID):
            if observed[row][(column + shifts[row]) % GRID]:
                canonical.add((column, row))
    return canonical, (x0, y0, x1, y1)


def decision_with_margin(image: Image.Image) -> tuple[str, float]:
    observed, _ = _decode(image)
    scores = sorted(
        (len(observed.symmetric_difference(template)) / float(GRID * GRID), label)
        for label, template in TEMPLATES.items()
    )
    margin = scores[1][0] - scores[0][0]
    if margin < 0.10:
        raise ValueError(f"ambiguous reconstruction (margin={margin:.3f})")
    return scores[0][1], margin


def decision_from_image(image: Image.Image) -> str:
    return decision_with_margin(image)[0]
