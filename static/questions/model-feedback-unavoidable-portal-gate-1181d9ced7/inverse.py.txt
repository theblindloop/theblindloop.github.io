"""Independent image-only inverse for the portal-maze cell grammar."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

WIDTH = 820
HEIGHT = 650
ROWS = 11
COLS = 15
CELL = 42
ORIGIN_X = 95
ORIGIN_Y = 150

OPEN_RGB = (248, 250, 252)
WALL_RGB = (36, 52, 71)
START_RGB = (18, 150, 94)
EXIT_RGB = (219, 68, 68)
GATE_RGB = {
    "amber": (230, 146, 27),
    "cyan": (8, 145, 178),
    "violet": (132, 70, 190),
    "lime": (87, 164, 57),
}
PORTAL_RGB = {
    "A": (52, 105, 200),
    "B": (210, 72, 132),
}


class OracleError(ValueError):
    pass


def _near(patch: np.ndarray, rgb: tuple[int, int, int], radius: float = 30.0) -> int:
    delta = patch.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return int((np.sum(delta * delta, axis=2, dtype=np.int32) <= radius * radius).sum())


def _recover(image: Image.Image) -> tuple[set[tuple[int, int]], tuple[int, int], tuple[int, int], dict[str, tuple[tuple[int, int], tuple[int, int]]], dict[str, tuple[int, int]]]:
    if image.size != (WIDTH, HEIGHT):
        raise OracleError(f"expected {(WIDTH, HEIGHT)} image, got {image.size}")
    arr = np.asarray(image.convert("RGB"))
    special_palette = {
        "start": START_RGB,
        "exit": EXIT_RGB,
        **{f"gate:{name}": rgb for name, rgb in GATE_RGB.items()},
        **{f"portal:{name}": rgb for name, rgb in PORTAL_RGB.items()},
    }
    open_cells: set[tuple[int, int]] = set()
    found: dict[str, list[tuple[int, int]]] = {name: [] for name in special_palette}

    for row in range(ROWS):
        for col in range(COLS):
            cx = ORIGIN_X + col * CELL + CELL // 2
            cy = ORIGIN_Y + row * CELL + CELL // 2
            patch = arr[cy - 12 : cy + 13, cx - 12 : cx + 13]
            scores = {name: _near(patch, rgb) for name, rgb in special_palette.items()}
            best = max(scores, key=scores.get)
            ranked = sorted(scores.values(), reverse=True)
            if scores[best] >= 150 and scores[best] >= 2 * max(1, ranked[1]):
                found[best].append((row, col))
                open_cells.add((row, col))
                continue
            open_score = _near(patch, OPEN_RGB)
            wall_score = _near(patch, WALL_RGB)
            if open_score >= 520 and open_score >= 5 * max(1, wall_score):
                open_cells.add((row, col))
            elif wall_score >= 520 and wall_score >= 5 * max(1, open_score):
                pass
            else:
                raise OracleError(f"cell {(row, col)} has no stable interior class")

    if len(found["start"]) != 1 or len(found["exit"]) != 1:
        raise OracleError("expected exactly one START and one EXIT marker")
    gates: dict[str, tuple[int, int]] = {}
    for color in GATE_RGB:
        values = found[f"gate:{color}"]
        if len(values) != 1:
            raise OracleError(f"expected one {color} gate, found {len(values)}")
        gates[color] = values[0]
    portals: dict[str, tuple[tuple[int, int], tuple[int, int]]] = {}
    for letter in PORTAL_RGB:
        values = found[f"portal:{letter}"]
        if len(values) != 2:
            raise OracleError(f"expected two {letter} portals, found {len(values)}")
        portals[letter] = (values[0], values[1])
    return open_cells, found["start"][0], found["exit"][0], portals, gates


def _reachable(
    open_cells: set[tuple[int, int]],
    start: tuple[int, int],
    target: tuple[int, int],
    portals: dict[str, tuple[tuple[int, int], tuple[int, int]]],
    blocked: tuple[int, int] | None = None,
) -> bool:
    if start == blocked or target == blocked:
        return False
    jumps: dict[tuple[int, int], tuple[int, int]] = {}
    for left, right in portals.values():
        jumps[left] = right
        jumps[right] = left
    seen = {start}
    queue = deque([start])
    while queue:
        row, col = queue.popleft()
        if (row, col) == target:
            return True
        neighbors = [(row - 1, col), (row + 1, col), (row, col - 1), (row, col + 1)]
        if (row, col) in jumps:
            neighbors.append(jumps[(row, col)])
        for neighbor in neighbors:
            if neighbor in open_cells and neighbor != blocked and neighbor not in seen:
                seen.add(neighbor)
                queue.append(neighbor)
    return False


def decision_from_image(image: Image.Image) -> str:
    open_cells, start, exit_cell, portals, gates = _recover(image)
    if not _reachable(open_cells, start, exit_cell, portals):
        raise OracleError("START and EXIT are disconnected before any gate is blocked")
    answers = [
        color
        for color, cell in gates.items()
        if not _reachable(open_cells, start, exit_cell, portals, blocked=cell)
    ]
    if len(answers) != 1:
        raise OracleError(f"expected one unavoidable gate, found {answers}")
    return answers[0]
