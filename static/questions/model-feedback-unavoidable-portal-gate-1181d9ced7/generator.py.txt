"""Deterministic portal-corridor world with a unique relative separator."""

from __future__ import annotations

import random
from collections import deque
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 820
HEIGHT = 650
SCALE = 3
ROWS = 11
COLS = 15
CELL = 42
ORIGIN_X = 95
ORIGIN_Y = 150

OPEN_RGB = (248, 250, 252)
WALL_RGB = (36, 52, 71)
GRID_RGB = (91, 107, 126)
INK_RGB = (24, 34, 47)
MUTED_RGB = (92, 106, 123)
START_RGB = (18, 150, 94)
EXIT_RGB = (219, 68, 68)
GATE_RGB = {
    "amber": (230, 146, 27),
    "cyan": (8, 145, 178),
    "violet": (132, 70, 190),
    "lime": (87, 164, 57),
}
PORTAL_RGB = {
    "A": (52, 105, 200),
    "B": (210, 72, 132),
}
COLORS = tuple(GATE_RGB)


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size * SCALE
    )


def _transform(cell: tuple[int, int], variant: int) -> tuple[int, int]:
    row, col = cell
    if variant & 1:
        col = COLS - 1 - col
    if variant & 2:
        row = ROWS - 1 - row
    return row, col


def _base_geometry() -> tuple[set[tuple[int, int]], tuple[int, int], tuple[int, int], dict[str, list[tuple[int, int]]], list[tuple[int, int]]]:
    paths = [
        # START to the first branch.
        [(5, 1), (5, 2), (5, 3)],
        # Upper left route, containing candidate gate role 1.
        [(5, 3), (4, 3), (3, 3), (3, 4), (3, 5), (4, 5), (5, 5), (5, 6), (5, 7)],
        # Lower left route, containing candidate gate role 3 and portal A.
        [(5, 3), (6, 3), (7, 3), (7, 4), (7, 5)],
        [(6, 7), (5, 7)],
        # The sole separator between the left and right route systems.
        [(5, 7), (4, 7), (3, 7), (2, 7), (1, 7)],
        # Upper/right route, containing candidate gate role 2.
        [(1, 7), (1, 8), (1, 9), (1, 10), (2, 10), (3, 10), (3, 11), (4, 11), (5, 11), (5, 12), (5, 13)],
        # Portal B bypass around role 2.
        [(1, 7), (1, 6), (1, 5)],
        [(4, 9), (5, 9), (5, 10), (5, 11)],
        # Broad, connected cul-de-sacs make local geometry non-diagnostic.
        [(3, 3), (2, 3), (1, 3), (1, 2), (0, 2)],
        [(7, 3), (8, 3), (9, 3), (9, 2), (10, 2)],
        [(3, 11), (3, 12), (2, 12), (2, 13)],
        [(5, 9), (6, 9), (7, 9), (7, 10), (8, 10)],
    ]
    open_cells = {cell for path in paths for cell in path}
    start = (5, 1)
    exit_cell = (5, 13)
    portals = {"A": [(7, 5), (6, 7)], "B": [(1, 5), (4, 9)]}
    # role 0 is the unique unavoidable gate; roles 1..3 are bypassable.
    gate_roles = [(3, 7), (3, 4), (1, 9), (7, 4)]
    return open_cells, start, exit_cell, portals, gate_roles


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    open_cells, start, exit_cell, portals, gate_roles = _base_geometry()
    variant = (int(seed) // 4) % 4
    transform = lambda cell: _transform(cell, variant)

    target_color = COLORS[int(seed) % len(COLORS)]
    other_colors = [color for color in COLORS if color != target_color]
    rng.shuffle(other_colors)
    role_colors = [target_color, *other_colors]

    return {
        "open_cells": [list(transform(cell)) for cell in sorted(open_cells)],
        "start": list(transform(start)),
        "exit": list(transform(exit_cell)),
        "portals": {
            letter: [list(transform(cell)) for cell in cells]
            for letter, cells in portals.items()
        },
        "gates": {
            color: list(transform(cell))
            for color, cell in zip(role_colors, gate_roles)
        },
    }


def _cell(value: object) -> tuple[int, int] | None:
    if not isinstance(value, list) or len(value) != 2:
        return None
    if any(isinstance(item, bool) or not isinstance(item, int) for item in value):
        return None
    row, col = int(value[0]), int(value[1])
    if not (0 <= row < ROWS and 0 <= col < COLS):
        return None
    return row, col


def _decode(scene: dict[str, Any]) -> tuple[set[tuple[int, int]], tuple[int, int], tuple[int, int], dict[str, tuple[tuple[int, int], tuple[int, int]]], dict[str, tuple[int, int]]] | None:
    try:
        raw_open = scene["open_cells"]
        start = _cell(scene["start"])
        exit_cell = _cell(scene["exit"])
        raw_portals = scene["portals"]
        raw_gates = scene["gates"]
    except (KeyError, TypeError):
        return None
    if not isinstance(raw_open, list) or not isinstance(raw_portals, dict) or not isinstance(raw_gates, dict):
        return None
    converted = [_cell(value) for value in raw_open]
    if any(value is None for value in converted):
        return None
    open_cells = {value for value in converted if value is not None}
    if len(open_cells) != len(converted) or start is None or exit_cell is None:
        return None
    if set(raw_portals) != set(PORTAL_RGB) or set(raw_gates) != set(GATE_RGB):
        return None
    portals: dict[str, tuple[tuple[int, int], tuple[int, int]]] = {}
    for letter in PORTAL_RGB:
        values = raw_portals[letter]
        if not isinstance(values, list) or len(values) != 2:
            return None
        pair = tuple(_cell(value) for value in values)
        if any(value is None for value in pair) or pair[0] == pair[1]:
            return None
        portals[letter] = (pair[0], pair[1])  # type: ignore[arg-type]
    gates = {color: _cell(raw_gates[color]) for color in GATE_RGB}
    if any(value is None for value in gates.values()):
        return None
    clean_gates = {color: value for color, value in gates.items() if value is not None}
    special = [start, exit_cell, *clean_gates.values(), *(cell for pair in portals.values() for cell in pair)]
    if len(special) != len(set(special)) or any(cell not in open_cells for cell in special):
        return None
    return open_cells, start, exit_cell, portals, clean_gates


def _reachable(
    open_cells: set[tuple[int, int]],
    start: tuple[int, int],
    target: tuple[int, int],
    portals: dict[str, tuple[tuple[int, int], tuple[int, int]]],
    blocked: tuple[int, int] | None = None,
) -> bool:
    if start == blocked or target == blocked:
        return False
    portal_jump: dict[tuple[int, int], tuple[int, int]] = {}
    for left, right in portals.values():
        portal_jump[left] = right
        portal_jump[right] = left
    seen = {start}
    queue = deque([start])
    while queue:
        row, col = queue.popleft()
        if (row, col) == target:
            return True
        neighbors = [(row - 1, col), (row + 1, col), (row, col - 1), (row, col + 1)]
        if (row, col) in portal_jump:
            neighbors.append(portal_jump[(row, col)])
        for neighbor in neighbors:
            if neighbor in open_cells and neighbor != blocked and neighbor not in seen:
                seen.add(neighbor)
                queue.append(neighbor)
    return False


def _unavoidable(scene: dict[str, Any]) -> list[str]:
    decoded = _decode(scene)
    if decoded is None:
        return []
    open_cells, start, exit_cell, portals, gates = decoded
    if not _reachable(open_cells, start, exit_cell, portals):
        return []
    return [
        color
        for color, cell in gates.items()
        if not _reachable(open_cells, start, exit_cell, portals, blocked=cell)
    ]


def analytic_gold(scene: dict[str, Any]) -> str:
    # The helper deliberately consumes all five declared scene fields.
    answers = _unavoidable(scene)
    if len(answers) != 1:
        raise ValueError(f"scene has {len(answers)} unavoidable gates")
    return answers[0]


def margin(scene: dict[str, Any]) -> float:
    return 15.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    decoded = _decode(scene)
    if decoded is None:
        return True
    open_cells, start, exit_cell, portals, gates = decoded
    answers = _unavoidable(scene)
    if len(answers) != 1:
        return True
    # Every candidate gate is reachable from both endpoints in the unblocked graph.
    return any(
        not _reachable(open_cells, start, cell, portals)
        or not _reachable(open_cells, cell, exit_cell, portals)
        for cell in gates.values()
    )


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict]]:
    return []


def _centered_text(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    text: str,
    font: ImageFont.FreeTypeFont,
    fill: tuple[int, int, int] | str,
) -> None:
    x, y = center
    box = draw.textbbox((0, 0), text, font=font)
    width = box[2] - box[0]
    height = box[3] - box[1]
    draw.text((x * SCALE - width / 2, y * SCALE - height / 2 - 2 * SCALE), text, font=font, fill=fill)


def render(scene: dict[str, Any]) -> Image.Image:
    decoded = _decode(scene)
    if decoded is None:
        raise ValueError("cannot render malformed scene")
    open_cells, start, exit_cell, portals, gates = decoded

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), (238, 243, 248))
    draw = ImageDraw.Draw(image)
    draw.text((34 * SCALE, 20 * SCALE), "PORTAL MAZE  •  FIND THE GATE ON EVERY ROUTE", font=_font(22), fill=INK_RGB)
    draw.text((35 * SCALE, 58 * SCALE), "Touching cells connect  |  equal letters jump", font=_font(14), fill=MUTED_RGB)

    # Compact legend repeats the semantic role of each mark.
    legend_y = 105
    legend = [
        (START_RGB, "S  START"),
        (EXIT_RGB, "E  EXIT"),
        (PORTAL_RGB["A"], "A=A portal"),
        (GATE_RGB["amber"], "colored cells = gates"),
    ]
    legend_x = (56, 235, 410, 585)
    for x, (color, label) in zip(legend_x, legend):
        draw.ellipse(((x - 12) * SCALE, (legend_y - 12) * SCALE, (x + 12) * SCALE, (legend_y + 12) * SCALE), fill=color, outline=INK_RGB, width=2 * SCALE)
        draw.text(((x + 19) * SCALE, (legend_y - 10) * SCALE), label, font=_font(11), fill=INK_RGB)

    left = ORIGIN_X * SCALE
    top = ORIGIN_Y * SCALE
    right = (ORIGIN_X + COLS * CELL) * SCALE
    bottom = (ORIGIN_Y + ROWS * CELL) * SCALE
    draw.rectangle((left - 5 * SCALE, top - 5 * SCALE, right + 5 * SCALE, bottom + 5 * SCALE), fill=WALL_RGB, outline=INK_RGB, width=2 * SCALE)

    for row in range(ROWS):
        for col in range(COLS):
            x0 = (ORIGIN_X + col * CELL) * SCALE
            y0 = (ORIGIN_Y + row * CELL) * SCALE
            x1 = x0 + CELL * SCALE
            y1 = y0 + CELL * SCALE
            fill = OPEN_RGB if (row, col) in open_cells else WALL_RGB
            draw.rectangle((x0, y0, x1, y1), fill=fill)

    # Candidate gates occupy broad cell interiors and remain passable.
    for color, (row, col) in gates.items():
        x0 = (ORIGIN_X + col * CELL + 3) * SCALE
        y0 = (ORIGIN_Y + row * CELL + 3) * SCALE
        x1 = (ORIGIN_X + (col + 1) * CELL - 3) * SCALE
        y1 = (ORIGIN_Y + (row + 1) * CELL - 3) * SCALE
        draw.rounded_rectangle((x0, y0, x1, y1), radius=6 * SCALE, fill=GATE_RGB[color], outline=INK_RGB, width=2 * SCALE)

    # Portal disks use a stable letter-specific color as redundant inverse evidence.
    for letter, pair in portals.items():
        for row, col in pair:
            cx = ORIGIN_X + col * CELL + CELL // 2
            cy = ORIGIN_Y + row * CELL + CELL // 2
            radius = 16
            draw.ellipse(((cx - radius) * SCALE, (cy - radius) * SCALE, (cx + radius) * SCALE, (cy + radius) * SCALE), fill=PORTAL_RGB[letter], outline=INK_RGB, width=2 * SCALE)
            _centered_text(draw, (cx, cy), letter, _font(18), "white")

    for cell, color, label in ((start, START_RGB, "S"), (exit_cell, EXIT_RGB, "E")):
        row, col = cell
        cx = ORIGIN_X + col * CELL + CELL // 2
        cy = ORIGIN_Y + row * CELL + CELL // 2
        radius = 17
        draw.ellipse(((cx - radius) * SCALE, (cy - radius) * SCALE, (cx + radius) * SCALE, (cy + radius) * SCALE), fill=color, outline=INK_RGB, width=2 * SCALE)
        _centered_text(draw, (cx, cy), label, _font(18), "white")

    # Cell boundaries make the four-neighbor movement convention explicit.
    for col in range(COLS + 1):
        x = (ORIGIN_X + col * CELL) * SCALE
        draw.line((x, top, x, bottom), fill=GRID_RGB, width=1 * SCALE)
    for row in range(ROWS + 1):
        y = (ORIGIN_Y + row * CELL) * SCALE
        draw.line((left, y, right, y), fill=GRID_RGB, width=1 * SCALE)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
