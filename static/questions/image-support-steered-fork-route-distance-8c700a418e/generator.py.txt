"""Deterministic renderer and analytic arm for fork_route_distance."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

BACKGROUND = (248, 247, 243)
TRACE_HALO = (73, 169, 176)
TRACE_CORE = (12, 92, 103)
CORAL = (220, 82, 75)
BLUE = (62, 105, 207)
PALE = (224, 222, 215)
CANVAS = 640
ROUTE_RADIUS = 64


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    depths = [8, 14, 20]
    left_depth = rng.choice(depths)
    right_depth = rng.choice([value for value in depths if value != left_depth])
    colors = ["coral", "blue"]
    rng.shuffle(colors)
    # Randomize the absolute neighborhood while keeping the tiny trace away
    # from arbitrary ablation-grid boundaries at this fixed raster size.
    centers = [200, 280, 360, 440]
    fork = [rng.choice(centers), rng.choice(centers)]
    rotation = rng.randrange(4)
    distractors = []
    for _ in range(8):
        x = rng.randint(34, CANVAS - 35)
        y = rng.randint(80, CANVAS - 35)
        length = rng.randint(8, 20)
        vertical = bool(rng.randrange(2))
        distractors.append([x, y, length, vertical])
    return {
        "canvas_size": CANVAS,
        "fork": fork,
        "left_depth": left_depth,
        "right_depth": right_depth,
        "left_color": colors[0],
        "right_color": colors[1],
        "rotation": rotation,
        "distractors": distractors,
    }


def _rotate(point: tuple[int, int], quarter_turns: int) -> tuple[int, int]:
    x, y = point
    for _ in range(int(quarter_turns) % 4):
        x, y = -y, x
    return x, y


def _place(points: Iterable[tuple[int, int]], fork: list[int], rotation: int) -> list[tuple[int, int]]:
    fx, fy = int(fork[0]), int(fork[1])
    return [(fx + x, fy + y) for x, y in (_rotate(point, rotation) for point in points)]


def _routes(scene: dict) -> tuple[list[tuple[int, int]], list[tuple[int, int]], list[tuple[int, int]]]:
    # The colored tips remain equally far from the fork in Euclidean distance.
    # Only each visible rectangular detour changes its graph-geodesic length.
    left_depth = int(scene["left_depth"])
    right_depth = int(scene["right_depth"])
    left = [(0, 0), (-18, 0), (-18, -left_depth), (-36, -left_depth), (-36, 0), (-ROUTE_RADIUS, 0)]
    right = [(0, 0), (18, 0), (18, -right_depth), (36, -right_depth), (36, 0), (ROUTE_RADIUS, 0)]
    stem = [(0, 0), (0, 28)]
    return (
        _place(left, scene["fork"], int(scene["rotation"])),
        _place(right, scene["fork"], int(scene["rotation"])),
        _place(stem, scene["fork"], int(scene["rotation"])),
    )


def render(scene: dict) -> Image.Image:
    # Explicit accesses keep the rendered-field audit honest.
    canvas_size = int(scene["canvas_size"])
    _ = (scene["left_depth"], scene["right_depth"], scene["left_color"], scene["right_color"])
    rotation = int(scene["rotation"])
    fork = scene["fork"]
    distractors = scene["distractors"]
    image = Image.new("RGB", (canvas_size, canvas_size), BACKGROUND)
    draw = ImageDraw.Draw(image)

    # Pale remote strokes vary independently and are ignored by both arms.
    for x, y, length, vertical in distractors:
        end = (int(x), int(y) + int(length)) if vertical else (int(x) + int(length), int(y))
        draw.line([(int(x), int(y)), end], fill=PALE, width=2)

    left, right, stem = _routes(scene)
    for route in (left, right, stem):
        draw.line(route, fill=TRACE_HALO, width=5, joint="curve")
    # A one-pixel connected centerline is intentionally recoverable from pixels.
    for route in (left, right, stem):
        draw.line(route, fill=TRACE_CORE, width=1)

    palette = {"coral": CORAL, "blue": BLUE}
    for endpoint, color_name in ((left[-1], scene["left_color"]), (right[-1], scene["right_color"])):
        x, y = endpoint
        draw.ellipse((x - 7, y - 7, x + 7, y + 7), fill=palette[color_name], outline=BACKGROUND, width=2)

    # Neutral cap distinguishes the unused stem from the two answer branches.
    sx, sy = stem[-1]
    draw.rectangle((sx - 5, sy - 5, sx + 5, sy + 5), fill=(135, 130, 119))
    return image


def analytic_gold(scene: dict) -> str:
    left_depth = int(scene["left_depth"])
    right_depth = int(scene["right_depth"])
    left_color = str(scene["left_color"])
    right_color = str(scene["right_color"])
    return left_color if left_depth > right_depth else right_color


def margin(scene: dict) -> float:
    return float(2 * abs(int(scene["left_depth"]) - int(scene["right_depth"])))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
