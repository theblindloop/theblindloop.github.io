"""Independent image-only inverse arm for fork_route_distance."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

TRACE_CORE = np.array((12, 92, 103), dtype=np.uint8)
CORAL = np.array((220, 82, 75), dtype=np.uint8)
BLUE = np.array((62, 105, 207), dtype=np.uint8)


def _centroid(mask: np.ndarray) -> tuple[float, float]:
    points = np.argwhere(mask)
    if len(points) < 20:
        raise ValueError("colored endpoint marker is missing")
    y, x = points.mean(axis=0)
    return float(y), float(x)


def _neighbors(point: tuple[int, int], pixels: set[tuple[int, int]]):
    y, x = point
    for neighbor in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
        if neighbor in pixels:
            yield neighbor


def _component(start: tuple[int, int], pixels: set[tuple[int, int]]) -> set[tuple[int, int]]:
    found = {start}
    queue = [start]
    while queue:
        point = queue.pop()
        for neighbor in _neighbors(point, pixels):
            if neighbor not in found:
                found.add(neighbor)
                queue.append(neighbor)
    return found


def _distances(start: tuple[int, int], pixels: set[tuple[int, int]]) -> dict[tuple[int, int], int]:
    distances = {start: 0}
    queue = deque([start])
    while queue:
        point = queue.popleft()
        for neighbor in _neighbors(point, pixels):
            if neighbor not in distances:
                distances[neighbor] = distances[point] + 1
                queue.append(neighbor)
    return distances


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    trace_mask = np.all(array == TRACE_CORE, axis=2)
    coral_center = _centroid(np.all(array == CORAL, axis=2))
    blue_center = _centroid(np.all(array == BLUE, axis=2))
    pixels = {tuple(map(int, point)) for point in np.argwhere(trace_mask)}
    if len(pixels) < 100:
        raise ValueError("trace centerline is incomplete")
    first = next(iter(pixels))
    connected = _component(first, pixels)
    if connected != pixels:
        raise ValueError("trace centerline is disconnected")
    forks = [point for point in pixels if sum(1 for _ in _neighbors(point, pixels)) >= 3]
    if len(forks) != 1:
        raise ValueError("a unique fork is not visible")
    distances = _distances(forks[0], pixels)
    endpoints = [point for point in pixels if sum(1 for _ in _neighbors(point, pixels)) == 1]
    if len(endpoints) != 3:
        raise ValueError("the fork must expose exactly three trace ends")

    def endpoint_near(center: tuple[float, float]) -> tuple[int, int]:
        ordered = sorted(endpoints, key=lambda p: (p[0] - center[0]) ** 2 + (p[1] - center[1]) ** 2)
        if (ordered[0][0] - center[0]) ** 2 + (ordered[0][1] - center[1]) ** 2 > 225:
            raise ValueError("endpoint marker is detached from its route")
        return ordered[0]

    coral_tip = endpoint_near(coral_center)
    blue_tip = endpoint_near(blue_center)
    if coral_tip == blue_tip:
        raise ValueError("colored markers do not identify distinct branches")
    coral_distance = distances[coral_tip]
    blue_distance = distances[blue_tip]
    if abs(coral_distance - blue_distance) < 10:
        raise ValueError("finite-resolution route comparison is ambiguous")
    return "coral" if coral_distance > blue_distance else "blue"
