"""Deterministic latent sampler and Pillow renderer for room/junction incidence."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

SIZE = 400
XS = (50, 150, 250, 350)
YS = (100, 200, 300)

_PALETTES = (
    ((35, 43, 55), (250, 249, 246), (28, 158, 88)),
    ((48, 39, 63), (248, 250, 247), (36, 150, 82)),
    ((28, 50, 57), (250, 248, 244), (43, 168, 96)),
)

_TRANSPOSE = (
    None,
    Image.Transpose.ROTATE_90,
    Image.Transpose.ROTATE_180,
    Image.Transpose.ROTATE_270,
    Image.Transpose.FLIP_LEFT_RIGHT,
    Image.Transpose.FLIP_TOP_BOTTOM,
    Image.Transpose.TRANSPOSE,
    Image.Transpose.TRANSVERSE,
)


def _adjacent_cells(marker_col: int) -> set[int]:
    """Cells touching the internal junction at (XS[marker_col], YS[1])."""

    return {
        marker_col - 1,
        marker_col,
        3 + marker_col - 1,
        3 + marker_col,
    }


def sample_scene(seed: int) -> dict:
    """Sample one robust scene; consecutive seeds are exactly class-balanced."""

    pair = seed // 2
    rng = random.Random((seed * 0x9E3779B1) ^ 0xA11FACE)
    marker_col = 1 + (pair % 2)
    adjacent = sorted(_adjacent_cells(marker_col))
    far = sorted(set(range(6)) - set(adjacent))
    positive = seed % 2 == 0

    if positive:
        # Three of the four cells around the marked cross are enclosed.
        omitted = adjacent[rng.randrange(len(adjacent))]
        closed = [cell for cell in adjacent if cell != omitted]
    else:
        # Keep all gross counts fixed, but place one room beyond the marked cross.
        chosen_adjacent = rng.sample(adjacent, 2)
        closed = chosen_adjacent + [far[rng.randrange(len(far))]]

    wall_rgb, paper_rgb, marker_rgb = _PALETTES[pair % len(_PALETTES)]
    return {
        "closed_cells": sorted(closed),
        "marker_col": marker_col,
        "transform": pair % len(_TRANSPOSE),
        "wall_width": (10, 12, 14)[pair % 3],
        "gap_width": (44, 50, 56)[(pair // 3) % 3],
        "marker_radius": (17, 18, 19)[(pair // 5) % 3],
        "wall_rgb": list(wall_rgb),
        "paper_rgb": list(paper_rgb),
        "marker_rgb": list(marker_rgb),
    }


def _base_render(scene: dict) -> Image.Image:
    paper = tuple(int(v) for v in scene["paper_rgb"])
    wall = tuple(int(v) for v in scene["wall_rgb"])
    marker = tuple(int(v) for v in scene["marker_rgb"])
    wall_width = int(scene["wall_width"])
    gap_width = int(scene["gap_width"])
    marker_radius = int(scene["marker_radius"])
    marker_col = int(scene["marker_col"])
    closed = {int(v) for v in scene["closed_cells"]}

    image = Image.new("RGB", (SIZE, SIZE), paper)
    draw = ImageDraw.Draw(image)
    for x in XS:
        draw.line((x, YS[0], x, YS[2]), fill=wall, width=wall_width)
    for y in YS:
        draw.line((XS[0], y, XS[3], y), fill=wall, width=wall_width)

    # Every open cell has a wide, directly visible doorway to the outer white area.
    half_gap = gap_width // 2
    half_wall = wall_width // 2 + 2
    for cell in range(6):
        if cell in closed:
            continue
        row, col = divmod(cell, 3)
        center_x = (XS[col] + XS[col + 1]) // 2
        outer_y = YS[0] if row == 0 else YS[2]
        draw.rectangle(
            (
                center_x - half_gap,
                outer_y - half_wall,
                center_x + half_gap,
                outer_y + half_wall,
            ),
            fill=paper,
        )

    cx, cy = XS[marker_col], YS[1]
    draw.ellipse(
        (cx - marker_radius, cy - marker_radius, cx + marker_radius, cy + marker_radius),
        fill=marker,
    )
    return image


def render(scene: dict) -> Image.Image:
    image = _base_render(scene)
    operation = _TRANSPOSE[int(scene["transform"])]
    return image if operation is None else image.transpose(operation)


def _incident_count(scene: dict) -> int:
    closed = {int(v) for v in scene["closed_cells"]}
    adjacent = _adjacent_cells(int(scene["marker_col"]))
    return len(closed & adjacent)


def analytic_gold(scene: dict) -> str:
    """Analytic consequence of the declared enclosed cells and marker slot."""

    closed = {int(v) for v in scene["closed_cells"]}
    marker_col = int(scene["marker_col"])
    adjacent = _adjacent_cells(marker_col)
    return "yes" if len(closed) == 3 and closed <= adjacent else "no"


def margin(scene: dict) -> float:
    """Runner-up margin in room incidences around the universal threshold 2.5."""

    return float(abs(_incident_count(scene) - 2.5))


def is_quarantined(scene: dict) -> bool:
    return bool(
        len(set(scene["closed_cells"])) != 3
        or int(scene["wall_width"]) < 10
        or int(scene["gap_width"]) < 40
        or int(scene["marker_radius"]) - int(scene["wall_width"]) / 2 < 10
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Exercise the genuine list-order alias of the enclosed-cell set."""

    twin = copy.deepcopy(scene)
    twin["closed_cells"] = list(reversed(twin["closed_cells"]))
    return [("reverse_unordered_closed_cells", twin)]
