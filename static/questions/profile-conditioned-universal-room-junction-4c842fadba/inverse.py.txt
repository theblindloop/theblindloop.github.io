"""Independent pixel-only inverse arm for universal room/junction incidence.

This module intentionally knows no latent grid coordinates, templates, scene data,
renderer, prompt mapping, or analytic answer. It segments pixels, floods the outer
paper component, labels enclosed paper components, and measures marker adjacency.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class AmbiguousRasterError(ValueError):
    """The visible raster does not satisfy the public robust-scene contract."""


def _flood_exterior(free: np.ndarray) -> np.ndarray:
    height, width = free.shape
    exterior = np.zeros_like(free, dtype=bool)
    queue: deque[tuple[int, int]] = deque()

    def add(y: int, x: int) -> None:
        if free[y, x] and not exterior[y, x]:
            exterior[y, x] = True
            queue.append((y, x))

    for x in range(width):
        add(0, x)
        add(height - 1, x)
    for y in range(1, height - 1):
        add(y, 0)
        add(y, width - 1)

    while queue:
        y, x = queue.popleft()
        if y and free[y - 1, x] and not exterior[y - 1, x]:
            exterior[y - 1, x] = True
            queue.append((y - 1, x))
        if y + 1 < height and free[y + 1, x] and not exterior[y + 1, x]:
            exterior[y + 1, x] = True
            queue.append((y + 1, x))
        if x and free[y, x - 1] and not exterior[y, x - 1]:
            exterior[y, x - 1] = True
            queue.append((y, x - 1))
        if x + 1 < width and free[y, x + 1] and not exterior[y, x + 1]:
            exterior[y, x + 1] = True
            queue.append((y, x + 1))
    return exterior


def _label_rooms(free: np.ndarray, exterior: np.ndarray) -> tuple[np.ndarray, list[int]]:
    height, width = free.shape
    labels = np.zeros((height, width), dtype=np.int16)
    available = free & ~exterior
    sizes: list[int] = []
    label = 0
    for start_y, start_x in np.argwhere(available):
        if labels[start_y, start_x]:
            continue
        label += 1
        labels[start_y, start_x] = label
        queue: deque[tuple[int, int]] = deque([(int(start_y), int(start_x))])
        size = 0
        while queue:
            y, x = queue.popleft()
            size += 1
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if (
                    0 <= ny < height
                    and 0 <= nx < width
                    and available[ny, nx]
                    and labels[ny, nx] == 0
                ):
                    labels[ny, nx] = label
                    queue.append((ny, nx))
        sizes.append(size)
    return labels, sizes


def _marker_neighbors(marker: np.ndarray) -> np.ndarray:
    neighbors = np.zeros_like(marker, dtype=bool)
    neighbors[1:] |= marker[:-1]
    neighbors[:-1] |= marker[1:]
    neighbors[:, 1:] |= marker[:, :-1]
    neighbors[:, :-1] |= marker[:, 1:]
    return neighbors & ~marker


def decision_from_image(image: Image.Image) -> str:
    """Return yes iff every one of the three enclosed white regions touches green."""

    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    if rgb.shape[0] < 128 or rgb.shape[1] < 128:
        raise AmbiguousRasterError("image is too small for stable topology")

    # All supplied paper palettes are very light; every barrier pigment is dark.
    free = np.min(rgb, axis=2) > 220
    marker = (
        (rgb[:, :, 1] > 115)
        & (rgb[:, :, 1] - rgb[:, :, 0] > 65)
        & (rgb[:, :, 1] - rgb[:, :, 2] > 35)
    )
    marker_area = int(marker.sum())
    if not 650 <= marker_area <= 1400:
        raise AmbiguousRasterError(f"green marker area is ambiguous: {marker_area}")

    exterior = _flood_exterior(free)
    labels, sizes = _label_rooms(free, exterior)
    if len(sizes) != 3 or min(sizes, default=0) < 3500:
        raise AmbiguousRasterError(f"expected three robust enclosed rooms, found {sizes}")

    touching = np.unique(labels[_marker_neighbors(marker) & (labels > 0)])
    if len(touching) not in (1, 2, 3):
        raise AmbiguousRasterError(f"marker/room incidence is ambiguous: {touching.tolist()}")
    return "yes" if len(touching) == 3 else "no"
