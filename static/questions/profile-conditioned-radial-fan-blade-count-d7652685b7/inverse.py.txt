"""Independent image-only inverse for separated saturated fan blades.

Only the final RGB raster is accepted.  The inverse finds large connected
components of strongly chromatic ink, then counts those components.  Neutral
cage, hub, stand, label, and background pixels are deliberately not used as
renderer state.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


ORACLE_VERSION = "radial_fan_blade_count-oracle-0.1.0"
MIN_COMPONENT_AREA = 900


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Return 8-connected component coordinates for a boolean mask."""

    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[np.ndarray] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue: deque[tuple[int, int]] = deque([(int(y), int(x))])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dy or dx):
                        continue
                    ny, nx = cy + dy, cx + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        if len(points) >= MIN_COMPONENT_AREA:
            found.append(np.asarray(points, dtype=np.int32))
    return found


def decision_from_image(image: Image.Image) -> str:
    """Count the separated high-chroma vane components in the supplied PNG."""

    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    if array.ndim != 3 or array.shape[2] != 3:
        raise ValueError("expected an RGB raster")
    chroma = array.max(axis=2) - array.min(axis=2)
    # Neutral cage and typography have low chroma; this broad mask accepts all
    # blade hues without encoding the renderer palette or the expected answer.
    saturated = (chroma >= 62) & (array.max(axis=2) >= 120)
    components = _components(saturated)
    if not 3 <= len(components) <= 7:
        raise ValueError(f"expected 3-7 separated blade regions, found {len(components)}")
    areas = sorted((int(component.shape[0]) for component in components))
    if areas[0] < MIN_COMPONENT_AREA:
        raise ValueError("blade components are below the visible-area safety floor")
    return str(len(components))
