"""Deterministic raster renderer for the radial fan blade-count world."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "radial_fan_blade_count-0.1.0"
WIDTH = 512
HEIGHT = 512
SUPERSAMPLE = 4
COUNTS = (3, 4, 5, 6, 7)

# All blade colours are strongly chromatic and well separated from the neutral
# cage, hub, stand, label, and light backgrounds.  The inverse deliberately
# does not import or name this palette.
BLADE_PALETTE = (
    (218, 62, 67),
    (35, 126, 214),
    (31, 158, 111),
    (164, 69, 193),
    (231, 133, 35),
    (20, 157, 172),
)
BACKGROUND_PALETTE = (
    (248, 248, 244),
    (245, 249, 252),
    (252, 246, 240),
    (247, 245, 251),
)
NEUTRAL_PALETTE = (
    (61, 69, 78),
    (79, 84, 91),
    (54, 76, 83),
    (89, 72, 69),
)


def _rgb(value: list[int] | tuple[int, int, int]) -> tuple[int, int, int]:
    return tuple(int(channel) for channel in value)


def _point(cx: float, cy: float, radius: float, angle: float, scale: float) -> tuple[int, int]:
    radians = math.radians(angle)
    return (
        round((cx + radius * math.sin(radians)) * scale),
        round((cy - radius * math.cos(radians)) * scale),
    )


def _ellipse_box(cx: float, cy: float, radius: float, scale: float) -> tuple[int, int, int, int]:
    x = cx * scale
    y = cy * scale
    r = radius * scale
    return (round(x - r), round(y - r), round(x + r), round(y + r))


def _blade_polygon(
    cx: float,
    cy: float,
    inner_radius: float,
    outer_radius: float,
    centre_angle: float,
    span: float,
    sweep: float,
    scale: float,
) -> list[tuple[int, int]]:
    """Return one broad swept vane as a six-vertex, separated region."""

    length = outer_radius - inner_radius
    points = (
        (inner_radius, centre_angle - span * 0.22),
        (inner_radius + length * 0.43, centre_angle - span * 0.54),
        (outer_radius - 3.0, centre_angle - span * 0.20),
        (outer_radius, centre_angle + sweep + span * 0.12),
        (inner_radius + length * 0.43, centre_angle + sweep + span * 0.53),
        (inner_radius, centre_angle + sweep + span * 0.24),
    )
    return [_point(cx, cy, radius, angle, scale) for radius, angle in points]


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced fan scene from one integer seed."""

    seed = int(seed)
    rng = random.Random(seed * 1_000_003 + 71_119)
    blade_count = COUNTS[seed % len(COUNTS)]
    outer_radius = round(rng.uniform(139.0, 166.0), 4)
    hub_radius = round(rng.uniform(24.0, 32.0), 4)
    inner_radius = round(hub_radius + rng.uniform(14.0, 21.0), 4)
    cage_radius = round(outer_radius + rng.uniform(13.0, 21.0), 4)
    centre_x = round(rng.uniform(238.0, 274.0), 4)
    centre_y = round(rng.uniform(269.0, 302.0), 4)
    # The total angular coverage is randomized separately from the count.  This
    # keeps per-blade width from being a reliable count proxy.
    span_budget = rng.uniform(105.0, 148.0)
    blade_span = round(max(14.0, min(25.0, span_budget / blade_count)), 4)
    sweep = round(rng.uniform(2.0, 7.0), 4)
    blade_color = BLADE_PALETTE[(seed // len(COUNTS) + 2 * (seed // 31)) % len(BLADE_PALETTE)]
    background_color = BACKGROUND_PALETTE[(seed // 11 + seed // 47) % len(BACKGROUND_PALETTE)]
    neutral = NEUTRAL_PALETTE[(seed // 7 + seed // 29) % len(NEUTRAL_PALETTE)]
    return {
        "blade_count": blade_count,
        "center_x": centre_x,
        "center_y": centre_y,
        "outer_radius": outer_radius,
        "hub_radius": hub_radius,
        "inner_radius": inner_radius,
        "cage_radius": cage_radius,
        "blade_span_deg": blade_span,
        "blade_sweep_deg": sweep,
        "blade_color": list(blade_color),
        "background_color": list(background_color),
        "cage_color": list(neutral),
        "hub_color": list(neutral),
        "label_color": list(neutral),
        "cage_width": rng.choice((3, 4, 5)),
        "hub_outline_width": rng.choice((2, 3, 4)),
        "angular_offset_deg": round(rng.uniform(0.0, 360.0), 4),
        "width": WIDTH,
        "height": HEIGHT,
        "antialias": True,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only the visible fan, cage, stand, and short label."""

    width = int(scene["width"])
    height = int(scene["height"])
    scale = SUPERSAMPLE if bool(scene.get("antialias", True)) else 1
    image = Image.new("RGB", (width * scale, height * scale), _rgb(scene["background_color"]))
    draw = ImageDraw.Draw(image)
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    outer = float(scene["outer_radius"])
    cage = float(scene["cage_radius"])
    hub = float(scene["hub_radius"])
    inner = float(scene["inner_radius"])
    cage_color = _rgb(scene["cage_color"])
    blade_color = _rgb(scene["blade_color"])

    # The neutral cage is drawn first.  Its many thin strokes are intentionally
    # distinct from the saturated blade regions used by the image-only inverse.
    cage_width = max(1, int(scene["cage_width"])) * scale
    draw.ellipse(_ellipse_box(cx, cy, cage, scale), outline=cage_color, width=cage_width)
    draw.ellipse(_ellipse_box(cx, cy, cage - 9.0, scale), outline=cage_color, width=max(1, scale))
    for angle in range(0, 360, 45):
        draw.line(
            [
                _point(cx, cy, hub + 3.0, angle, scale),
                _point(cx, cy, cage - 3.0, angle, scale),
            ],
            fill=cage_color,
            width=max(1, scale),
        )

    blade_count = int(scene["blade_count"])
    base_angle = float(scene["angular_offset_deg"])
    span = float(scene["blade_span_deg"])
    sweep = float(scene["blade_sweep_deg"])
    for index in range(blade_count):
        angle = base_angle + index * 360.0 / blade_count
        polygon = _blade_polygon(cx, cy, inner, outer, angle, span, sweep, scale)
        draw.polygon(polygon, fill=blade_color)
        # A restrained same-colour highlight makes the blade read as a solid
        # vane without creating a second connected component.
        highlight = _blade_polygon(
            cx,
            cy,
            inner + 6.0,
            outer - 8.0,
            angle - 1.0,
            max(8.0, span * 0.25),
            sweep * 0.15,
            scale,
        )
        draw.polygon(highlight, fill=tuple(min(255, channel + 22) for channel in blade_color))

    draw.ellipse(_ellipse_box(cx, cy, hub, scale), fill=_rgb(scene["hub_color"]), outline=cage_color, width=max(1, int(scene["hub_outline_width"])) * scale)
    draw.ellipse(_ellipse_box(cx, cy, hub * 0.30, scale), fill=_rgb(scene["cage_color"]))

    # A simple stand makes the object recognizable but remains neutral and is
    # never part of the blade-color component mask.
    stem_top = _point(cx, cy, cage + 2.0, 180.0, scale)
    stem_bottom = (round(cx * scale), round(470.0 * scale))
    draw.line([stem_top, stem_bottom], fill=cage_color, width=5 * scale)
    draw.line(
        [(round((cx - 70.0) * scale), round(470.0 * scale)), (round((cx + 70.0) * scale), round(470.0 * scale))],
        fill=cage_color,
        width=7 * scale,
    )

    if scale > 1:
        image = image.resize((width, height), Image.Resampling.LANCZOS)
    label_draw = ImageDraw.Draw(image)
    try:
        font = ImageFont.load_default()
    except Exception:
        font = None
    label = "ELECTRIC FAN"
    bbox = label_draw.textbbox((0, 0), label, font=font)
    label_x = (width - (bbox[2] - bbox[0])) // 2
    label_draw.text((label_x, 20), label, fill=_rgb(scene["label_color"]), font=font)
    return image.convert("RGB")


def analytic_gold(scene: dict[str, Any]) -> str:
    """The label-bearing field is the number of visible radial blades."""

    return str(int(scene["blade_count"]))


def margin(scene: dict[str, Any]) -> float:
    """Return the smallest continuous clearance between visible parts."""

    count = int(scene["blade_count"])
    outer = float(scene["outer_radius"])
    span = float(scene["blade_span_deg"]) + float(scene["blade_sweep_deg"])
    angular_clearance = outer * math.radians(360.0 / count - span)
    hub_clearance = float(scene["inner_radius"]) - float(scene["hub_radius"])
    cage_clearance = float(scene["cage_radius"]) - outer
    return round(min(angular_clearance, hub_clearance, cage_clearance), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Exclude a finite-resolution pose with less than 8 px clearance."""

    return margin(scene) < 8.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """There are no undeclared pixel-identical aliases in this scene model."""

    return []
