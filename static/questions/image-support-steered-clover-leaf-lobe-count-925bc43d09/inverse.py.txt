from __future__ import annotations

import math

import numpy as np
from PIL import Image


def _radial_profile(mask: np.ndarray, cx: float, cy: float) -> tuple[np.ndarray, float]:
    height, width = mask.shape
    max_radius = int(math.hypot(width, height))
    angles = np.linspace(0.0, 2.0 * math.pi, 720, endpoint=False)
    radii = np.arange(max_radius, dtype=np.float64)
    profile = np.zeros(angles.size, dtype=np.float64)
    missing_inside = 0
    sampled_inside = 0
    for index, angle in enumerate(angles):
        xs = np.rint(cx + radii * math.cos(float(angle))).astype(np.int32)
        ys = np.rint(cy + radii * math.sin(float(angle))).astype(np.int32)
        valid = (xs >= 0) & (xs < width) & (ys >= 0) & (ys < height)
        values = mask[ys[valid], xs[valid]]
        hit_indices = np.flatnonzero(values)
        hit = radii[valid][values]
        if hit.size:
            profile[index] = float(hit.max())
            last = int(hit_indices[-1])
            missing_inside += int((~values[: last + 1]).sum())
            sampled_inside += last + 1
    gap_fraction = missing_inside / max(sampled_inside, 1)
    return profile, float(gap_fraction)


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    chroma = rgb.max(axis=2) - rgb.min(axis=2)
    mask = chroma > 42
    if int(mask.sum()) < int(mask.size * 0.12):
        raise ValueError("no substantial colored silhouette")
    ys, xs = np.nonzero(mask)
    visible_width = int(xs.max() - xs.min() + 1)
    visible_height = int(ys.max() - ys.min() + 1)
    if visible_width < 0.78 * mask.shape[1] or visible_height < 0.81 * mask.shape[0]:
        raise ValueError("clover silhouette is clipped or materially incomplete")
    row_widths = np.bincount(ys, minlength=mask.shape[0])
    broad_rows = np.flatnonzero(row_widths > 0.34 * row_widths.max())
    if broad_rows.size < 20:
        raise ValueError("silhouette has no broad connected body")
    cx = float(xs.mean())
    cy = float(ys.mean())
    center_y, center_x = int(round(cy)), int(round(cx))
    if not mask[center_y, center_x]:
        raise ValueError("colored center of silhouette is missing")
    profile, gap_fraction = _radial_profile(mask, cx, cy)
    if gap_fraction > 0.002:
        raise ValueError("silhouette is not a solid radial body")
    centered_profile = profile - profile.mean()
    spectrum = np.abs(np.fft.rfft(centered_profile))
    score3 = float(spectrum[3])
    score5 = float(spectrum[5])
    gap = abs(score3 - score5) / max(score3 + score5, 1e-9)
    if gap < 0.16:
        raise ValueError("ambiguous radial periodicity")
    count = 3 if score3 > score5 else 5
    angles = np.linspace(0.0, 2.0 * math.pi, profile.size, endpoint=False)
    design = np.column_stack((
        np.ones(profile.size),
        np.cos(angles),
        np.sin(angles),
        np.cos(count * angles),
        np.sin(count * angles),
    ))
    fitted = design @ np.linalg.lstsq(design, profile, rcond=None)[0]
    residual = np.abs(profile - fitted) / max(float(profile.mean()), 1.0)
    if float(np.quantile(residual, 0.90)) > 0.095:
        raise ValueError("rounded lobe boundary is materially corrupted")
    return str(count)
