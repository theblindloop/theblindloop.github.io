from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

SIZE = 384
AA = 3
PALETTES = (
    ((38, 150, 82), (248, 246, 238)),
    ((23, 132, 112), (246, 244, 250)),
    ((64, 128, 196), (250, 247, 238)),
    ((156, 92, 184), (246, 249, 243)),
    ((202, 92, 62), (247, 247, 242)),
    ((188, 126, 28), (246, 244, 250)),
)


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    foreground, background = PALETTES[rng.randrange(len(PALETTES))]
    return {
        "leaf_count": 3 if int(seed) % 2 == 0 else 5,
        "rotation_deg": round(rng.uniform(-14.0, 14.0), 4),
        "center_x": rng.randint(190, 194),
        "center_y": rng.randint(186, 190),
        "radius": rng.randint(132, 135),
        "lobe_depth": round(rng.uniform(0.265, 0.315), 4),
        "stem_length": rng.randint(144, 151),
        "stem_width": rng.randint(20, 27),
        "foreground_rgb": list(foreground),
        "background_rgb": list(background),
    }


def _body_points(scene: dict[str, Any]) -> list[tuple[int, int]]:
    count = int(scene["leaf_count"])
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    radius = float(scene["radius"])
    depth = float(scene["lobe_depth"])
    stem_angle = math.pi / 2.0 + math.radians(float(scene["rotation_deg"]))
    phase = stem_angle - math.pi / count
    points = []
    for index in range(720):
        theta = 2.0 * math.pi * index / 720.0
        r = radius * (1.0 + depth * math.cos(count * (theta - phase)))
        points.append((round(AA * (cx + r * math.cos(theta))), round(AA * (cy + r * math.sin(theta)))))
    return points


def render(scene: dict[str, Any]) -> Image.Image:
    background = tuple(int(v) for v in scene["background_rgb"])
    foreground = tuple(int(v) for v in scene["foreground_rgb"])
    image = Image.new("RGB", (SIZE * AA, SIZE * AA), background)
    draw = ImageDraw.Draw(image)
    cx = AA * float(scene["center_x"])
    cy = AA * float(scene["center_y"])
    angle = math.pi / 2.0 + math.radians(float(scene["rotation_deg"]))
    start = 0.35 * float(scene["radius"])
    end = float(scene["stem_length"])
    line = (
        cx + AA * start * math.cos(angle),
        cy + AA * start * math.sin(angle),
        cx + AA * end * math.cos(angle),
        cy + AA * end * math.sin(angle),
    )
    draw.line(line, fill=foreground, width=AA * int(scene["stem_width"]), joint="curve")
    endpoint_r = AA * int(scene["stem_width"]) / 2
    ex, ey = line[2], line[3]
    draw.ellipse((ex-endpoint_r, ey-endpoint_r, ex+endpoint_r, ey+endpoint_r), fill=foreground)
    draw.polygon(_body_points(scene), fill=foreground)
    return image.resize((SIZE, SIZE), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(int(scene["leaf_count"]))


def margin(scene: dict[str, Any]) -> float:
    return 2.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    return False


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
