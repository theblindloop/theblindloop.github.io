"""Two distant isolated rings, same-enclosure-state comparison world.

Latent scene z describes two physically disconnected black rings placed in two
opposite far corners of the canvas (an island in the upper-left and an island in
the lower-right). Each ring is independently either "sealed" (a full closed band
that encloses exactly one interior white region, a hole) or "open" (a single
radial gap that fuses its interior with the exterior so it encloses no hole),
while remaining one connected ink component in both cases.

The decision asks whether the two distant rings are in the SAME enclosure state
(both sealed or both open) or DIFFERENT states. The answer therefore binds the
state of both distant islands: neither island alone determines it.

The decision is recovered purely from the raster by labelling the two black
connected components and counting the total number of enclosed white regions
(holes) across the whole image. A sealed ring contributes one hole, an open ring
contributes none, so an even hole total (0 or 2) means the islands agree and an
odd total (1) means they differ.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS = (512, 512)
W, H = CANVAS
GRID = 8
CELL = W / GRID

# Two far-apart island cells on the abstraction grid (index 0..7 each axis).
RING_A_CELL = (1, 1)  # upper-left
RING_B_CELL = (6, 6)  # lower-right

JITTER = 16.0

OUTER_MIN, OUTER_MAX = 46.0, 58.0
THICK_MIN, THICK_MAX = 20.0, 30.0
GAP_WIDTH_DEG_MIN, GAP_WIDTH_DEG_MAX = 24.0, 44.0
MIN_GAP_LINEAR_PX = 15.0
MIN_THICK = 18.0


def _sample_ring(rng: random.Random, cell: tuple[int, int]) -> dict:
    cx0 = (cell[1] + 0.5) * CELL
    cy0 = (cell[0] + 0.5) * CELL
    cx = cx0 + rng.uniform(-JITTER, JITTER)
    cy = cy0 + rng.uniform(-JITTER, JITTER)
    sealed = rng.random() < 0.5
    outer = rng.uniform(OUTER_MIN, OUTER_MAX)
    thickness = rng.uniform(THICK_MIN, THICK_MAX)
    gap_width_deg = rng.uniform(GAP_WIDTH_DEG_MIN, GAP_WIDTH_DEG_MAX)
    gap_dir = rng.uniform(0.0, 360.0)
    return {
        "center": [cx, cy],
        "outer": outer,
        "thickness": thickness,
        "sealed": sealed,
        "gap_width_deg": gap_width_deg,
        "gap_dir": gap_dir,
    }


def sample_scene(seed: int) -> dict:
    """Deterministically sample a two-ring scene dict from a seed."""
    rng = random.Random(seed)
    ring_a = _sample_ring(rng, RING_A_CELL)
    ring_b = _sample_ring(rng, RING_B_CELL)
    return {"ring_a": ring_a, "ring_b": ring_b}


def _mid_radius(ring: dict) -> float:
    return ring["outer"] - ring["thickness"] / 2.0


def _gap_linear_width(ring: dict) -> float:
    return _mid_radius(ring) * math.radians(ring["gap_width_deg"])


def _draw_ring(draw: ImageDraw.ImageDraw, ring: dict) -> None:
    cx, cy = ring["center"]
    outer = ring["outer"]
    thickness = ring["thickness"]
    inner = outer - thickness
    draw.ellipse(
        [cx - outer, cy - outer, cx + outer, cy + outer], fill="black"
    )
    draw.ellipse(
        [cx - inner, cy - inner, cx + inner, cy + inner], fill="white"
    )
    if not ring["sealed"]:
        gap_rad = math.radians(ring["gap_width_deg"])
        ga = math.radians(ring["gap_dir"])
        far = outer + 50.0
        a0 = ga - gap_rad / 2.0
        a1 = ga + gap_rad / 2.0
        draw.polygon(
            [
                (cx, cy),
                (cx + far * math.cos(a0), cy + far * math.sin(a0)),
                (cx + far * math.cos(a1), cy + far * math.sin(a1)),
            ],
            fill="white",
        )


def render(scene: dict) -> Image.Image:
    """Render the raster I = R(z). Deterministic, no noise, no text."""
    img = Image.new("RGB", CANVAS, "white")
    d = ImageDraw.Draw(img)
    _draw_ring(d, scene["ring_a"])
    _draw_ring(d, scene["ring_b"])
    return img


def analytic_gold(scene: dict) -> str:
    """'same' iff the two islands are in the same enclosure state."""
    return "same" if scene["ring_a"]["sealed"] == scene["ring_b"]["sealed"] else "different"


def margin(scene: dict) -> float:
    """Smallest per-ring closure margin: thickness (sealed) or gap width (open)."""
    margins = []
    for ring in (scene["ring_a"], scene["ring_b"]):
        if ring["sealed"]:
            margins.append(ring["thickness"])
        else:
            margins.append(_gap_linear_width(ring))
    return float(min(margins))


def is_quarantined(scene: dict) -> bool:
    if scene["ring_a"]["thickness"] < MIN_THICK - 1e-9:
        return True
    if scene["ring_b"]["thickness"] < MIN_THICK - 1e-9:
        return True
    for ring in (scene["ring_a"], scene["ring_b"]):
        if not ring["sealed"] and _gap_linear_width(ring) < MIN_GAP_LINEAR_PX:
            return True
    return False


def latent_symmetries(scene: dict) -> list:
    """Declared empty: the raster fully determines the gold (not_applicable)."""
    return []
