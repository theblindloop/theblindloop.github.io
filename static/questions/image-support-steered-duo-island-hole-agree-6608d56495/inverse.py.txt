"""Independent pixel-only inverse arm for duo_island_hole_agree.

Recovers the same/different enclosure decision from the raster alone. It labels
the black (ink) connected components, requires exactly two distant rings, then
floods the exterior white in from the image border and counts the total number
of enclosed white regions (holes) across the whole image. An even hole total (0
or 2) means the two islands agree ("same"); an odd total (1) means they differ
("different"). When fewer or more than two ink components are present the oracle
abstains (raises), because the decision requires both distant islands.

This module imports neither the renderer nor any scene data; it reads only the
pixels.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class AbstentionError(Exception):
    pass


def _mask_black(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    return (arr[:, :, 0] < 128) & (arr[:, :, 1] < 128) & (arr[:, :, 2] < 128)


def _mask_white(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    return (arr[:, :, 0] > 128) & (arr[:, :, 1] > 128) & (arr[:, :, 2] > 128)


def _flood(size: tuple[int, int], starts, allow) -> np.ndarray:
    """4-connected flood fill through True pixels of allow from the starts."""
    h, w = size
    visited = np.zeros(h * w, dtype=bool)
    frontier = deque()
    for r, c in starts:
        idx = r * w + c
        if allow[idx] and not visited[idx]:
            visited[idx] = True
            frontier.append(idx)
    while frontier:
        idx = frontier.popleft()
        r, c = divmod(idx, w)
        if r > 0:
            ni = idx - w
            if allow[ni] and not visited[ni]:
                visited[ni] = True
                frontier.append(ni)
        if r < h - 1:
            ni = idx + w
            if allow[ni] and not visited[ni]:
                visited[ni] = True
                frontier.append(ni)
        if c > 0:
            ni = idx - 1
            if allow[ni] and not visited[ni]:
                visited[ni] = True
                frontier.append(ni)
        if c < w - 1:
            ni = idx + 1
            if allow[ni] and not visited[ni]:
                visited[ni] = True
                frontier.append(ni)
    return visited


def _count_components(size: tuple[int, int], mask_flat: np.ndarray) -> int:
    h, w = size
    unseen = np.nonzero(mask_flat)[0]
    labels = np.zeros(h * w, dtype=bool)
    count = 0
    for start_idx in unseen:
        if labels[start_idx]:
            continue
        count += 1
        labels[start_idx] = True
        frontier = deque([start_idx])
        while frontier:
            idx = frontier.popleft()
            r, c = divmod(idx, w)
            if r > 0:
                ni = idx - w
                if mask_flat[ni] and not labels[ni]:
                    labels[ni] = True
                    frontier.append(ni)
            if r < h - 1:
                ni = idx + w
                if mask_flat[ni] and not labels[ni]:
                    labels[ni] = True
                    frontier.append(ni)
            if c > 0:
                ni = idx - 1
                if mask_flat[ni] and not labels[ni]:
                    labels[ni] = True
                    frontier.append(ni)
            if c < w - 1:
                ni = idx + 1
                if mask_flat[ni] and not labels[ni]:
                    labels[ni] = True
                    frontier.append(ni)
    return count


def decision_from_image(image: Image.Image) -> str:
    black = _mask_black(image)
    h, w = black.shape
    n_islands = _count_components((h, w), black.reshape(-1))
    if n_islands != 2:
        raise AbstentionError("decision requires exactly two distant islands")

    white = _mask_white(image)
    white_flat = white.reshape(-1)

    border_starts = []
    for c in range(w):
        if white[0, c]:
            border_starts.append((0, c))
        if white[h - 1, c]:
            border_starts.append((h - 1, c))
    for r in range(1, h - 1):
        if white[r, 0]:
            border_starts.append((r, 0))
        if white[r, w - 1]:
            border_starts.append((r, w - 1))

    exterior = _flood((h, w), border_starts, white_flat)
    enclosed_white_flat = white_flat & ~exterior
    n_holes = _count_components((h, w), enclosed_white_flat)
    return "same" if n_holes in (0, 2) else "different"
