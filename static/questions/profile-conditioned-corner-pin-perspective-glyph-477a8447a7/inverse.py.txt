"""Independent raster-only inverse for corner-pin projective rectification."""

from __future__ import annotations

import numpy as np
from PIL import Image

GRID = 9
LABELS = ("fork", "hook", "zigzag", "ladder")
MARKERS = (
    (241, 184, 48),
    (232, 112, 42),
    (139, 92, 246),
    (21, 184, 210),
)


def _references() -> dict[str, set[tuple[int, int]]]:
    fork = (
        {(4, y) for y in range(3, 8)}
        | {(3, 2), (2, 1), (1, 0), (5, 2), (6, 1), (7, 0)}
        | {(x, 7) for x in range(1, 8)}
    )
    hook = (
        {(2, y) for y in range(1, 8)}
        | {(x, 1) for x in range(3, 6)}
        | {(x, 7) for x in range(3, 7)}
        | {(6, y) for y in range(4, 7)}
    )
    zigzag = (
        {(x, 1) for x in range(1, 7)}
        | {(6, 2), (5, 3), (4, 4), (3, 5), (2, 6)}
        | {(x, 7) for x in range(1, 7)}
    )
    ladder = (
        {(2, y) for y in range(1, 8)}
        | {(6, y) for y in range(1, 8)}
        | {(x, 4) for x in range(3, 6)}
    )
    return {"fork": fork, "hook": hook, "zigzag": zigzag, "ladder": ladder}


REFERENCES = _references()


def _marker_centroid(pixels: np.ndarray, rgb: tuple[int, int, int]) -> tuple[float, float]:
    delta = pixels.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    mask = np.sum(delta * delta, axis=2) < 42 * 42
    ys, xs = np.nonzero(mask)
    if len(xs) < 80:
        raise ValueError("a required colored corner pin is missing")
    return float(xs.mean()), float(ys.mean())


def _fit_frame(corners: list[tuple[float, float]]) -> np.ndarray:
    """DLT fit written independently from the renderer's square mapper."""
    canonical = np.asarray(((0, 0), (1, 0), (1, 1), (0, 1)), dtype=float)
    rows = []
    target = []
    for source, destination in zip(canonical, corners):
        u, v = source
        x, y = destination
        rows.extend(
            (
                (u, v, 1, 0, 0, 0, -x * u, -x * v),
                (0, 0, 0, u, v, 1, -y * u, -y * v),
            )
        )
        target.extend((x, y))
    coefficients, _, _, _ = np.linalg.lstsq(np.asarray(rows), np.asarray(target), rcond=None)
    return np.concatenate((coefficients, (1.0,))).reshape(3, 3)


def _map(frame: np.ndarray, u: float, v: float) -> tuple[float, float]:
    raw = frame.dot(np.asarray((u, v, 1.0)))
    return float(raw[0] / raw[2]), float(raw[1] / raw[2])


def _recover_cells(image: Image.Image) -> set[tuple[int, int]]:
    pixels = np.asarray(image.convert("RGB"))
    corners = [_marker_centroid(pixels, color) for color in MARKERS]
    frame = _fit_frame(corners)
    occupied: set[tuple[int, int]] = set()
    height, width = pixels.shape[:2]
    for row in range(GRID):
        for column in range(GRID):
            # Median a compact 3x3 canonical sub-sample, safely away from grid lines.
            samples = []
            for du, dv in ((0.0, 0.0), (-0.12, -0.12), (0.12, -0.12), (-0.12, 0.12), (0.12, 0.12)):
                u = (column + 0.5 + du) / GRID
                v = (row + 0.5 + dv) / GRID
                x, y = _map(frame, u, v)
                ix = min(width - 1, max(0, int(round(x))))
                iy = min(height - 1, max(0, int(round(y))))
                samples.append(pixels[iy, ix].astype(float))
            color = np.median(np.asarray(samples), axis=0)
            luminance = 0.2126 * color[0] + 0.7152 * color[1] + 0.0722 * color[2]
            if luminance < 155.0:
                occupied.add((column, row))
    return occupied


def analyze_image(image: Image.Image) -> tuple[str, float]:
    occupied = _recover_cells(image)
    if not 10 <= len(occupied) <= 24:
        raise ValueError("rectified dark-tile count is outside the declared glyph regime")
    ranked = sorted((len(occupied ^ REFERENCES[label]), label) for label in LABELS)
    return ranked[0][1], float(ranked[1][0] - ranked[0][0])


def decision_from_image(image: Image.Image) -> str:
    return analyze_image(image)[0]
