"""Deterministic renderer for the corner-pin projective rectification world."""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw

SIZE = 9
CANVAS = (560, 460)
NAMES = ("fork", "hook", "zigzag", "ladder")


def _glyphs() -> dict[str, set[tuple[int, int]]]:
    return {
        "fork": (
            {(4, y) for y in range(3, 8)}
            | {(3, 2), (2, 1), (1, 0), (5, 2), (6, 1), (7, 0)}
            | {(x, 7) for x in range(1, 8)}
        ),
        "hook": (
            {(2, y) for y in range(1, 8)}
            | {(x, 1) for x in range(3, 6)}
            | {(x, 7) for x in range(3, 7)}
            | {(6, y) for y in range(4, 7)}
        ),
        "zigzag": (
            {(x, 1) for x in range(1, 7)}
            | {(6, 2), (5, 3), (4, 4), (3, 5), (2, 6)}
            | {(x, 7) for x in range(1, 7)}
        ),
        "ladder": (
            {(2, y) for y in range(1, 8)}
            | {(6, y) for y in range(1, 8)}
            | {(x, 4) for x in range(3, 6)}
        ),
    }


GLYPHS = _glyphs()
PALETTES = (
    ((25, 48, 78), (239, 235, 220), (248, 247, 242)),
    ((91, 39, 54), (237, 232, 213), (247, 245, 238)),
    ((33, 75, 58), (235, 234, 219), (249, 247, 240)),
    ((61, 52, 91), (238, 232, 216), (247, 246, 241)),
)
PIN_COLORS = (
    (241, 184, 48),   # gold: canonical upper-left
    (232, 112, 42),   # orange: canonical upper-right
    (139, 92, 246),   # violet: canonical lower-right
    (21, 184, 210),   # cyan: canonical lower-left
)


def _grid_from_points(points: set[tuple[int, int]]) -> list[list[int]]:
    return [[int((x, y) in points) for x in range(SIZE)] for y in range(SIZE)]


def _distances(cells: list[list[int]]) -> list[tuple[int, str]]:
    observed = {(x, y) for y, row in enumerate(cells) for x, value in enumerate(row) if value}
    return sorted((len(observed ^ GLYPHS[name]), name) for name in NAMES)


def analytic_gold(scene: dict) -> str:
    """Classify solely from the rendered cell field; no hidden class token exists."""
    return _distances(scene["cells"])[0][1]


def margin(scene: dict) -> float:
    ranked = _distances(scene["cells"])
    return float(ranked[1][0] - ranked[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 6.0


def sample_scene(seed: int) -> dict:
    """Cross answer identity with palette, pose, and orientation deterministically."""
    rng = random.Random(int(seed) ^ 0x6A09E667)
    target_index = int(seed) % len(NAMES)
    target = NAMES[target_index]
    runner = NAMES[(target_index + 1) % len(NAMES)]
    points = set(GLYPHS[target])

    # Paired removals/additions keep exactly 17 dark cells.  One balanced block
    # in every six is deliberately close to the runner-up boundary.
    target_only = sorted(GLYPHS[target] - GLYPHS[runner])
    runner_only = sorted(GLYPHS[runner] - GLYPHS[target])
    pair_limit = len(target_only)
    if (int(seed) // 4) % 6 == 0:
        pair_count = max(0, (2 * pair_limit - 6) // 4)
    else:
        pair_count = (int(seed) // 7) % 2
    pair_count = min(pair_count, pair_limit - 1)
    for point in rng.sample(target_only, pair_count):
        points.remove(point)
    for point in rng.sample(runner_only, pair_count):
        points.add(point)

    palette = PALETTES[(int(seed) // 4) % len(PALETTES)]
    spin = ((int(seed) // 4) + 2 * (int(seed) % 4)) % 4
    quad = [
        [rng.randint(55, 125), rng.randint(45, 105)],
        [rng.randint(410, 520), rng.randint(55, 140)],
        [rng.randint(400, 520), rng.randint(325, 420)],
        [rng.randint(45, 145), rng.randint(315, 425)],
    ]
    scene = {
        "cells": _grid_from_points(points),
        "quad": quad,
        "spin": spin,
        "dark_rgb": list(palette[0]),
        "light_rgb": list(palette[1]),
        "background_rgb": list(palette[2]),
    }
    # The construction always stays on the intended side of the boundary.
    if analytic_gold(scene) != target:
        raise RuntimeError("balanced corruption crossed the template boundary")
    return scene


def _homography(quad: list[list[int]]) -> np.ndarray:
    """Map unit-square coordinates to the supplied quadrilateral."""
    source = ((0.0, 0.0), (1.0, 0.0), (1.0, 1.0), (0.0, 1.0))
    matrix: list[list[float]] = []
    values: list[float] = []
    for (u, v), (x, y) in zip(source, quad):
        matrix.append([u, v, 1.0, 0.0, 0.0, 0.0, -u * x, -v * x])
        values.append(float(x))
        matrix.append([0.0, 0.0, 0.0, u, v, 1.0, -u * y, -v * y])
        values.append(float(y))
    solved = np.linalg.solve(np.asarray(matrix), np.asarray(values))
    return np.append(solved, 1.0).reshape(3, 3)


def _project(matrix: np.ndarray, u: float, v: float) -> tuple[float, float]:
    point = matrix @ np.asarray([u, v, 1.0])
    return float(point[0] / point[2]), float(point[1] / point[2])


def _rotate(cells: list[list[int]], turns: int) -> list[list[int]]:
    result = [list(row) for row in cells]
    for _ in range(turns % 4):
        result = [list(row) for row in zip(*result)][::-1]
    return result


def render(scene: dict) -> Image.Image:
    """Render the projective tile carpet and four visible semantic corner pins."""
    scale = 2
    image = Image.new(
        "RGB", (CANVAS[0] * scale, CANVAS[1] * scale), tuple(scene["background_rgb"])
    )
    draw = ImageDraw.Draw(image)
    quad = scene["quad"]
    transform = _homography(quad)
    cells = _rotate(scene["cells"], int(scene["spin"]))

    def scaled(points: list[tuple[float, float]]) -> list[tuple[float, float]]:
        return [(x * scale, y * scale) for x, y in points]

    for row in range(SIZE):
        for column in range(SIZE):
            u0, u1 = column / SIZE, (column + 1) / SIZE
            v0, v1 = row / SIZE, (row + 1) / SIZE
            polygon = [
                _project(transform, u0, v0),
                _project(transform, u1, v0),
                _project(transform, u1, v1),
                _project(transform, u0, v1),
            ]
            color = scene["dark_rgb"] if cells[row][column] else scene["light_rgb"]
            draw.polygon(scaled(polygon), fill=tuple(color))

    line_color = (132, 139, 151)
    for index in range(SIZE + 1):
        t = index / SIZE
        draw.line(
            scaled([_project(transform, t, 0.0), _project(transform, t, 1.0)]),
            fill=line_color,
            width=2 * scale,
        )
        draw.line(
            scaled([_project(transform, 0.0, t), _project(transform, 1.0, t)]),
            fill=line_color,
            width=2 * scale,
        )

    # Rotating the cell field rotates the pin assignment with it.  Consequently
    # screen orientation alone is independent of the answer; color order is needed.
    turns = int(scene["spin"]) % 4
    screen_pin_colors = PIN_COLORS[turns:] + PIN_COLORS[:turns]
    radius = 9 * scale
    for (x, y), color in zip(quad, screen_pin_colors):
        cx, cy = x * scale, y * scale
        draw.ellipse(
            (cx - radius, cy - radius, cx + radius, cy + radius),
            fill=color,
            outline=(255, 255, 255),
            width=2 * scale,
        )

    return image.resize(CANVAS, Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
