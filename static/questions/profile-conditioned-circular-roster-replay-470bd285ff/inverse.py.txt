"""Independent inverse arm using only raster colors and components."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

INK = (28, 35, 48)
TEAL = (48, 174, 160)
MAGENTA = (190, 55, 120)
RED = (222, 78, 76)
BLUE = (64, 105, 210)


class Abstain(ValueError):
    pass


def _runs(values: np.ndarray) -> list[tuple[int, int]]:
    indices = np.flatnonzero(values)
    if len(indices) == 0:
        return []
    result = []
    start = previous = int(indices[0])
    for raw in indices[1:]:
        value = int(raw)
        if value > previous + 1:
            result.append((start, previous + 1))
            start = value
        previous = value
    result.append((start, previous + 1))
    return result


def _component_areas(mask: np.ndarray) -> list[int]:
    work = mask.copy()
    areas: list[int] = []
    height, width = work.shape
    for y, x in zip(*np.nonzero(work), strict=True):
        if not work[y, x]:
            continue
        work[y, x] = False
        queue = deque([(int(y), int(x))])
        area = 0
        while queue:
            yy, xx = queue.popleft()
            area += 1
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        areas.append(area)
    return areas


def _dot_count(ink: np.ndarray, box: tuple[int, int, int, int]) -> int:
    x0, y0, x1, y1 = box
    areas = _component_areas(ink[y0:y1, x0:x1])
    return sum(70 <= area <= 240 for area in areas)


def _next(active: list[int], holder: int) -> int:
    later = [value for value in active if value > holder]
    return min(later) if later else min(active)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    if arr.ndim != 3 or arr.shape[0] < 300 or arr.shape[1] < 600:
        raise Abstain("unexpected canvas")
    teal = np.all(arr == TEAL, axis=2)
    magenta = np.all(arr == MAGENTA, axis=2)
    red = np.all(arr == RED, axis=2)
    blue = np.all(arr == BLUE, axis=2)
    ink = np.all(arr == INK, axis=2)

    token_runs = _runs(teal.sum(axis=0) > 20)
    if len(token_runs) != 5:
        raise Abstain(f"expected five seats, found {len(token_runs)}")
    token_centers: list[float] = []
    identities: list[int] = []
    for x0, x1 in token_runs:
        yy, xx = np.nonzero(teal[:, x0:x1])
        if len(yy) < 3000:
            raise Abstain("seat fill too small")
        y0, y1 = int(yy.min()), int(yy.max()) + 1
        identity = _dot_count(ink, (x0, y0, x1, y1))
        identities.append(identity)
        token_centers.append(float(x0 + xx.mean()))
    if identities != [1, 2, 3, 4, 5]:
        raise Abstain(f"seat identities malformed: {identities}")

    my, mx = np.nonzero(magenta)
    if len(mx) < 1000:
        raise Abstain("starting ring missing")
    ring_x = float(mx.mean())
    distances = [abs(center - ring_x) for center in token_centers]
    if sorted(distances)[1] - sorted(distances)[0] < 30:
        raise Abstain("starting ring ambiguous")
    holder = int(np.argmin(distances)) + 1

    card_mask = red | blue
    card_runs = _runs(card_mask.sum(axis=0) > 30)
    if len(card_runs) != 6:
        raise Abstain(f"expected six operation cards, found {len(card_runs)}")
    active = [1, 2, 3, 4, 5]
    for x0, x1 in card_runs:
        red_count = int(red[:, x0:x1].sum())
        blue_count = int(blue[:, x0:x1].sum())
        if max(red_count, blue_count) < 10000 or min(red_count, blue_count) != 0:
            raise Abstain("card color ambiguous")
        color_mask = red[:, x0:x1] if red_count else blue[:, x0:x1]
        yy, _ = np.nonzero(color_mask)
        y0, y1 = int(yy.min()), int(yy.max()) + 1
        if blue_count:
            holder = _next(active, holder)
        else:
            target = _dot_count(ink, (x0, y0, x1, y1))
            if target not in range(1, 6):
                raise Abstain(f"remove target malformed: {target}")
            if target in active and len(active) > 1:
                replacement = _next(active, target)
                active.remove(target)
                if holder == target:
                    holder = replacement
    return str(holder)
