"""Deterministic Pillow renderer and analytic state machine."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 930, 550
BG = (248, 247, 242)
INK = (28, 35, 48)
TEAL = (48, 174, 160)
MAGENTA = (190, 55, 120)
RED = (222, 78, 76)
BLUE = (64, 105, 210)
WHITE = (255, 255, 255)


def _next(active: list[int], holder: int) -> int:
    greater = [value for value in active if value > holder]
    return min(greater) if greater else min(active)


def _replay(initial_holder: int, ops: list[dict]) -> int:
    active = [1, 2, 3, 4, 5]
    holder = int(initial_holder)
    for op in ops:
        if op["kind"] == "pass":
            holder = _next(active, holder)
        elif op["kind"] == "remove":
            target = int(op["target"])
            if target in active and len(active) > 1:
                replacement = _next(active, target)
                active.remove(target)
                if holder == target:
                    holder = replacement
        else:
            raise ValueError("unknown operation")
    return holder


def sample_scene(seed: int) -> dict:
    desired = seed % 5 + 1
    for attempt in range(20000):
        rng = random.Random(seed * 100003 + attempt)
        initial = rng.randint(1, 5)
        active = [1, 2, 3, 4, 5]
        ops: list[dict] = []
        n_remove = rng.choice((1, 2, 2, 3))
        remove_slots = set(rng.sample(range(6), n_remove))
        for index in range(6):
            if index in remove_slots and len(active) > 2:
                target = rng.choice(active)
                active.remove(target)
                ops.append({"kind": "remove", "target": target})
            else:
                ops.append({"kind": "pass", "target": 0})
        scene = {"initial_holder": initial, "ops": ops}
        if desired in active and _replay(initial, ops) == desired:
            return scene
    raise RuntimeError("could not sample balanced scene")


def analytic_gold(scene: dict) -> str:
    return str(_replay(int(scene["initial_holder"]), list(scene["ops"])))


def margin(scene: dict) -> float:
    # Minimum nominal clearance (pixels) between dot components and card borders.
    if not scene["ops"] or int(scene["initial_holder"]) not in range(1, 6):
        return 0.0
    return 14.0


def is_quarantined(scene: dict) -> bool:
    removes = {int(op["target"]) for op in scene["ops"] if op["kind"] == "remove"}
    return len({1, 2, 3, 4, 5} - removes) < 2


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int):
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _dots(draw: ImageDraw.ImageDraw, cx: int, cy: int, count: int, radius: int = 6) -> None:
    spots = [(-12, -13), (12, -13), (-12, 7), (12, 7), (0, 25)]
    for dx, dy in spots[:count]:
        draw.ellipse((cx + dx - radius, cy + dy - radius, cx + dx + radius, cy + dy + radius), fill=INK)


def render(scene: dict) -> Image.Image:
    initial = int(scene["initial_holder"])
    ops = list(scene["ops"])
    if len(ops) != 6:
        raise ValueError("exactly six operations required")
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    draw.text((36, 24), "STARTING CIRCULAR ROSTER", font=_font(25), fill=INK)
    draw.text((36, 242), "READ CARDS LEFT TO RIGHT", font=_font(23), fill=INK)
    draw.text((555, 30), "BLUE = PASS   RED = REMOVE", font=_font(15), fill=INK)
    xs = [125, 295, 465, 635, 805]
    cy = 155
    for seat, cx in enumerate(xs, start=1):
        if seat == initial:
            draw.ellipse((cx - 54, cy - 54, cx + 54, cy + 54), fill=MAGENTA)
        draw.ellipse((cx - 43, cy - 43, cx + 43, cy + 43), fill=TEAL)
        _dots(draw, cx, cy - 5, seat)
    draw.line((125, 218, 805, 218), fill=INK, width=3)
    draw.polygon([(805, 218), (788, 209), (788, 227)], fill=INK)
    card_y0, card_y1 = 310, 478
    for index, op in enumerate(ops):
        x0 = 28 + index * 150
        x1 = x0 + 124
        color = BLUE if op["kind"] == "pass" else RED
        draw.rounded_rectangle((x0, card_y0, x1, card_y1), radius=13, fill=color, outline=INK, width=3)
        cx = (x0 + x1) // 2
        if op["kind"] == "pass":
            draw.line((cx - 30, 394, cx + 25, 394), fill=INK, width=12)
            draw.polygon([(cx + 34, 394), (cx + 10, 374), (cx + 10, 414)], fill=INK)
        else:
            _dots(draw, cx, 382, int(op["target"]), radius=7)
        draw.text((x0 + 51, 444), str(index + 1), font=_font(17), fill=WHITE)
    return image
