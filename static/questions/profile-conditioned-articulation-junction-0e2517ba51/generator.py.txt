"""Deterministic renderer and analytic answer for articulation junctions."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw


WIDTH = 1200
HEIGHT = 900
SCALE = 3
CENTER = (600.0, 450.0)

# Two cyclic blocks are connected by a two-edge path.  All three queried
# vertices have degree two; only the middle path vertex is an articulation
# point.  The other six vertices are neutral distractors.
BASE_NODES = (
    (290.0, 300.0),
    (420.0, 300.0),
    (440.0, 570.0),
    (270.0, 650.0),
    (780.0, 300.0),
    (910.0, 300.0),
    (760.0, 570.0),
    (930.0, 650.0),
    (600.0, 520.0),
)
EDGES = (
    (0, 1),
    (1, 2),
    (2, 3),
    (3, 0),
    (4, 5),
    (5, 7),
    (7, 6),
    (6, 4),
    (2, 8),
    (8, 6),
)
CANDIDATE_NODES = (0, 4, 8)
COLORS = {
    "red": (214, 66, 78),
    "blue": (51, 116, 211),
    "green": (48, 158, 101),
}
PAGE = (248, 250, 252)
PANEL = (255, 255, 255)
PANEL_EDGE = (218, 225, 232)
EDGE_INK = (53, 64, 77)
NODE_FILL = (213, 228, 239)
NODE_RADIUS = 28.0
EDGE_WIDTH = 16


def _transform(point: tuple[float, float], rotation: int, mirror_x: bool, mirror_y: bool, scale: float) -> tuple[float, float]:
    x, y = point
    cx, cy = CENTER
    x = cx + (x - cx) * scale
    y = cy + (y - cy) * scale
    if mirror_x:
        x = cx - (x - cx)
    if mirror_y:
        y = cy - (y - cy)
    for _ in range(rotation % 4):
        x, y = cx - (y - cy), cy + (x - cx)
    return x, y


def sample_scene(seed: int) -> dict:
    """Sample one clean embedding and independently permute the labels."""

    rng = random.Random(int(seed))
    rotation = rng.randrange(4)
    mirror_x = bool(rng.randrange(2))
    mirror_y = bool(rng.randrange(2))
    scale = rng.uniform(0.94, 1.03)
    nodes = []
    for base_x, base_y in BASE_NODES:
        jitter_x = rng.uniform(-11.0, 11.0)
        jitter_y = rng.uniform(-11.0, 11.0)
        x, y = _transform(
            (base_x + jitter_x, base_y + jitter_y),
            rotation,
            mirror_x,
            mirror_y,
            scale,
        )
        nodes.append([round(x, 3), round(y, 3)])

    labels = ["red", "blue", "green"]
    rng.shuffle(labels)
    label_map = {
        str(node): label for node, label in zip(CANDIDATE_NODES, labels)
    }
    return {
        "nodes": nodes,
        "edges": [list(edge) for edge in EDGES],
        "candidate_nodes": list(CANDIDATE_NODES),
        "label_map": label_map,
    }


def _component_count(node_count: int, edges: list[list[int]] | tuple[tuple[int, int], ...], removed: int | None = None) -> int:
    adjacency = [[] for _ in range(node_count)]
    for left, right in edges:
        if left == removed or right == removed:
            continue
        adjacency[left].append(right)
        adjacency[right].append(left)
    seen = set()
    components = 0
    for start in range(node_count):
        if start == removed or start in seen:
            continue
        components += 1
        stack = [start]
        seen.add(start)
        while stack:
            node = stack.pop()
            for neighbor in adjacency[node]:
                if neighbor not in seen:
                    seen.add(neighbor)
                    stack.append(neighbor)
    return components


def render(scene: dict) -> Image.Image:
    """Render only visible topology, node markers, and a neutral frame."""

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), PAGE)
    draw = ImageDraw.Draw(image)
    nodes = scene["nodes"]
    for left, right in scene["edges"]:
        x0, y0 = (int(round(value * SCALE)) for value in nodes[left])
        x1, y1 = (int(round(value * SCALE)) for value in nodes[right])
        width = EDGE_WIDTH * SCALE
        draw.line((x0, y0, x1, y1), fill=EDGE_INK, width=width)
        cap = width // 2
        for x, y in ((x0, y0), (x1, y1)):
            draw.ellipse((x - cap, y - cap, x + cap, y + cap), fill=EDGE_INK)

    labels = scene["label_map"]
    candidate_nodes = set(scene["candidate_nodes"])
    for index, (x_value, y_value) in enumerate(nodes):
        x = int(round(float(x_value) * SCALE))
        y = int(round(float(y_value) * SCALE))
        radius = int(round(NODE_RADIUS * SCALE))
        fill = COLORS[labels[str(index)]] if index in candidate_nodes else NODE_FILL
        draw.ellipse(
            (x - radius, y - radius, x + radius, y + radius),
            fill=fill,
            outline=EDGE_INK,
            width=6 * SCALE,
        )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """Return the label of the unique candidate whose deletion splits the graph."""

    nodes = scene["nodes"]
    edges = scene["edges"]
    candidates = scene["candidate_nodes"]
    labels = scene["label_map"]
    baseline = _component_count(len(nodes), edges)
    changes = [
        _component_count(len(nodes), edges, removed=node) - baseline
        for node in candidates
    ]
    winners = [node for node, change in zip(candidates, changes) if change > 0]
    if len(winners) != 1:
        raise ValueError(f"expected one articulation candidate, got {changes}")
    return str(labels[str(winners[0])])


def margin(scene: dict) -> float:
    """Discrete separation between the unique cut candidate and its rivals."""

    if is_quarantined(scene):
        return 0.0
    nodes = scene["nodes"]
    edges = scene["edges"]
    candidates = scene["candidate_nodes"]
    baseline = _component_count(len(nodes), edges)
    changes = sorted(
        (_component_count(len(nodes), edges, removed=node) - baseline for node in candidates),
        reverse=True,
    )
    return float(changes[0] - changes[1])


def _orientation(a: tuple[float, float], b: tuple[float, float], c: tuple[float, float]) -> float:
    return (b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0])


def _segments_intersect(a, b, c, d) -> bool:
    o1 = _orientation(a, b, c)
    o2 = _orientation(a, b, d)
    o3 = _orientation(c, d, a)
    o4 = _orientation(c, d, b)
    eps = 1e-7
    return ((o1 > eps and o2 < -eps) or (o1 < -eps and o2 > eps)) and ((o3 > eps and o4 < -eps) or (o3 < -eps and o4 > eps))


def _point_segment_distance(point, a, b) -> float:
    dx = b[0] - a[0]
    dy = b[1] - a[1]
    denom = dx * dx + dy * dy
    if denom <= 0.0:
        return math.hypot(point[0] - a[0], point[1] - a[1])
    t = max(0.0, min(1.0, ((point[0] - a[0]) * dx + (point[1] - a[1]) * dy) / denom))
    return math.hypot(point[0] - (a[0] + t * dx), point[1] - (a[1] + t * dy))


def is_quarantined(scene: dict) -> bool:
    """Reject malformed, clipped, crossed, or raster-unstable embeddings."""

    try:
        nodes = scene["nodes"]
        edges = scene["edges"]
        candidates = scene["candidate_nodes"]
        labels = scene["label_map"]
        if len(nodes) != len(BASE_NODES) or len(edges) != len(EDGES):
            return True
        if tuple(tuple(edge) for edge in edges) != EDGES:
            return True
        if tuple(candidates) != CANDIDATE_NODES:
            return True
        if set(labels) != {str(index) for index in CANDIDATE_NODES} or set(labels.values()) != set(COLORS):
            return True
        points = [(float(point[0]), float(point[1])) for point in nodes]
        if any(not (80.0 < x < WIDTH - 80.0 and 80.0 < y < HEIGHT - 80.0) for x, y in points):
            return True
        if any(math.dist(first, second) < 88.0 for i, first in enumerate(points) for second in points[i + 1 :]):
            return True
        for left, right in edges:
            if math.dist(points[left], points[right]) < 96.0:
                return True
        for first_index, (left, right) in enumerate(edges):
            a, b = points[left], points[right]
            for second_index, (other_left, other_right) in enumerate(edges):
                if second_index <= first_index or {left, right} & {other_left, other_right}:
                    continue
                c, d = points[other_left], points[other_right]
                if _segments_intersect(a, b, c, d):
                    return True
                if min(_point_segment_distance(c, a, b), _point_segment_distance(d, a, b), _point_segment_distance(a, c, d), _point_segment_distance(b, c, d)) < 24.0:
                    return True
        if _component_count(len(points), edges) != 1:
            return True
        baseline = _component_count(len(points), edges)
        changes = [
            _component_count(len(points), edges, removed=node) - baseline
            for node in candidates
        ]
        return sorted(changes) != [0, 0, 1]
    except (KeyError, TypeError, ValueError, IndexError, ZeroDivisionError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No hidden render-invariant aliases are used by this direct scene encoding."""

    return []
