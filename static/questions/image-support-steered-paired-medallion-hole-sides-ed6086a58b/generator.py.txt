"""Deterministic raster renderer for the paired-medallion question.

The only answer-bearing visual facts are the signed horizontal offsets of the
two enclosed holes.  The faces are deliberately separate, medium-sized,
left/right-balanced components; no stroke connects them.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (248, 245, 237)
HOLE_RADIUS_CHOICES = (9, 10, 11)
FACE_PALETTE = (
    (37, 91, 128),
    (151, 67, 70),
    (53, 116, 83),
    (118, 76, 145),
    (181, 111, 39),
    (42, 126, 131),
)


def _ordered(scene: dict[str, Any]) -> list[dict[str, Any]]:
    return sorted(scene["medallions"], key=lambda item: int(item["slot"]))


def _face_polygon(record: dict[str, Any]) -> list[tuple[int, int]]:
    cx, cy = (int(value) for value in record["center"])
    half_w = int(record["width"]) // 2
    half_h = int(record["height"]) // 2
    # A left-right symmetric octagonal seal keeps the visible bounding-box
    # midpoint a stable, directly observable face center.
    fractions = (
        (-0.72, -1.00),
        (0.72, -1.00),
        (1.00, -0.60),
        (1.00, 0.60),
        (0.72, 1.00),
        (-0.72, 1.00),
        (-1.00, 0.60),
        (-1.00, -0.60),
    )
    return [
        (round(cx + fx * half_w), round(cy + fy * half_h)) for fx, fy in fractions
    ]


def _draw_medallion(draw: ImageDraw.ImageDraw, record: dict[str, Any], hole_radius: int) -> None:
    draw.polygon(_face_polygon(record), fill=tuple(record["face_color"]))
    cx, cy = (int(value) for value in record["center"])
    hx = cx + int(record["hole_offset"])
    hy = cy
    draw.ellipse(
        (hx - hole_radius, hy - hole_radius, hx + hole_radius, hy + hole_radius),
        fill=BACKGROUND,
    )


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced, visible pair with the requested prior-conflict nuisance variation."""

    rng = random.Random((int(seed) * 0x9E3779B1) ^ 0x4D454441)
    # Consecutive generator seeds alternate in two-scene blocks between the
    # same-side and opposite-side classes, giving every stress run both labels.
    same_side = ((int(seed) // 2) % 2) == 0
    left_sign = -1 if ((int(seed) * 1103515245 + 12345) & 1) == 0 else 1
    right_sign = left_sign if same_side else -left_sign
    magnitude = rng.choice((16, 18, 20))
    hole_radius = rng.choice(HOLE_RADIUS_CHOICES)
    width = rng.choice((74, 76, 78))
    height = rng.choice((104, 108, 112))

    colors = list(FACE_PALETTE)
    rng.shuffle(colors)
    left_center = (128 + rng.randint(-7, 7), 256 + rng.randint(-10, 10))
    right_center = (384 + rng.randint(-7, 7), 256 + rng.randint(-10, 10))
    records = [
        {
            "slot": 0,
            "center": [left_center[0], left_center[1]],
            "width": width,
            "height": height,
            "hole_offset": left_sign * magnitude,
            "face_color": list(colors[0]),
        },
        {
            "slot": 1,
            "center": [right_center[0], right_center[1]],
            "width": width,
            "height": height,
            "hole_offset": right_sign * magnitude,
            "face_color": list(colors[1]),
        },
    ]
    # Serialization order is nuisance-only and is tested as a pixel-identical
    # latent alias by latent_symmetries().
    if rng.randrange(2):
        records.reverse()
    return {
        "medallions": records,
        "canvas_size": [WIDTH, HEIGHT],
        "background": list(BACKGROUND),
        "hole_radius": hole_radius,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render a scene as a crisp RGB PNG-compatible image."""

    width, height = (int(value) for value in scene["canvas_size"])
    background = tuple(scene["background"])
    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)
    for record in _ordered(scene):
        _draw_medallion(draw, record, int(scene["hole_radius"]))
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return whether the two visible hole-side signs agree."""

    records = _ordered(scene)
    if len(records) != 2:
        raise ValueError("expected exactly two medallions")
    left = int(records[0]["hole_offset"])
    right = int(records[1]["hole_offset"])
    if left == 0 or right == 0:
        raise ValueError("a hole on the midpoint is not a classifiable side")
    return "yes" if (left > 0) == (right > 0) else "no"


def margin(scene: dict[str, Any]) -> float:
    """One-sided pixel verifiability margin for the visible side decision."""

    records = _ordered(scene)
    if len(records) != 2:
        return -1.0
    radius = float(scene["hole_radius"])
    clearances = []
    for record in records:
        half_width = float(record["width"]) / 2.0
        half_height = float(record["height"]) / 2.0
        offset = abs(float(record["hole_offset"]))
        clearances.extend(
            (
                offset - 8.0,
                half_width - offset - radius - 3.0,
                half_height - radius - 6.0,
            )
        )
    x0 = float(records[0]["center"][0]) + float(records[0]["width"]) / 2.0
    x1 = float(records[1]["center"][0]) - float(records[1]["width"]) / 2.0
    clearances.append((x1 - x0) / 2.0 - 4.0)
    return round(min(clearances), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Exclude midpoint, clipped, merged, or otherwise low-clearance scenes."""

    records = _ordered(scene)
    if len(records) != 2:
        return True
    width, height = (int(value) for value in scene["canvas_size"])
    for record in records:
        cx, cy = (int(value) for value in record["center"])
        half_w = int(record["width"]) // 2
        half_h = int(record["height"]) // 2
        if not (half_w < cx < width - half_w and half_h < cy < height - half_h):
            return True
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """Expose a nontrivial serialization alias with identical pixels and gold."""

    twin = dict(scene)
    twin["medallions"] = [dict(record) for record in reversed(scene["medallions"])]
    return [("reverse_medallion_serialization", twin)]
