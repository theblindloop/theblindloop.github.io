"""Independent pixel-only inverse for the paired-medallion raster."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[tuple[np.ndarray, np.ndarray]]:
    """Return 4-connected component coordinates without using renderer state."""

    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    components: list[tuple[np.ndarray, np.ndarray]] = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if visited[sy, sx]:
            continue
        queue = [(sy, sx)]
        visited[sy, sx] = True
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.pop()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    queue.append((ny, nx))
        if points:
            array = np.asarray(points, dtype=np.int32)
            components.append((array[:, 0], array[:, 1]))
    return components


def _enclosed_void_centroid(face_y: np.ndarray, face_x: np.ndarray, background: np.ndarray) -> tuple[float, float]:
    """Find the centroid of the one background component enclosed by a face."""

    y0, y1 = int(face_y.min()), int(face_y.max())
    x0, x1 = int(face_x.min()), int(face_x.max())
    sub = background[y0 : y1 + 1, x0 : x1 + 1]
    local_h, local_w = sub.shape
    outside = np.zeros_like(sub, dtype=bool)
    queue: deque[tuple[int, int]] = deque()
    for x in range(local_w):
        if sub[0, x] and not outside[0, x]:
            outside[0, x] = True
            queue.append((0, x))
        if sub[local_h - 1, x] and not outside[local_h - 1, x]:
            outside[local_h - 1, x] = True
            queue.append((local_h - 1, x))
    for y in range(local_h):
        if sub[y, 0] and not outside[y, 0]:
            outside[y, 0] = True
            queue.append((y, 0))
        if sub[y, local_w - 1] and not outside[y, local_w - 1]:
            outside[y, local_w - 1] = True
            queue.append((y, local_w - 1))
    while queue:
        y, x = queue.popleft()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < local_h and 0 <= nx < local_w and sub[ny, nx] and not outside[ny, nx]:
                outside[ny, nx] = True
                queue.append((ny, nx))
    enclosed = sub & ~outside
    voids = [item for item in _components(enclosed) if len(item[0]) >= 80]
    if len(voids) != 1:
        raise ValueError(f"expected one enclosed hole, found {len(voids)}")
    void_y, void_x = voids[0]
    return float(void_x.mean() + x0), float(void_y.mean() + y0)


def decision_from_image(image: Image.Image) -> str:
    """Recover the same-side decision from the final RGB image alone.

    Missing faces or holes intentionally raise instead of guessing; destructive
    ablations therefore expose the two disconnected evidence components.
    """

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[2] != 3 or array.shape[0] < 32 or array.shape[1] < 32:
        raise ValueError("unsupported image")
    background_rgb = array[0, 0]
    background = np.all(array == background_rgb, axis=2)
    foreground = ~background
    faces = [item for item in _components(foreground) if len(item[0]) >= 1200]
    if len(faces) != 2:
        raise ValueError(f"expected two separated medallion faces, found {len(faces)}")

    side_signs: list[int] = []
    for face_y, face_x in sorted(faces, key=lambda item: float(item[1].mean())):
        hole_x, _ = _enclosed_void_centroid(face_y, face_x, background)
        midpoint = (float(face_x.min()) + float(face_x.max())) / 2.0
        delta = hole_x - midpoint
        if abs(delta) < 4.0:
            raise ValueError("hole is too close to face midpoint")
        side_signs.append(1 if delta > 0 else -1)
    return "yes" if side_signs[0] == side_signs[1] else "no"
