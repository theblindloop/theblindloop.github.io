"""Scene sampling, rendering and analytic gold for `seal_pair_ring_binding`.

Three small red "seals" sit close together inside one local patch of a busy
page: one seal above, two seals below. Each seal is one or two nested red
square outlines. A ring is *sealed* when its outline is unbroken all the way
around; a ring carrying a visible mouth (a gap cut through one side) is not.
The decision binds the three disconnected seals: it compares the number of
sealed rings in the upper seal against the combined number in the lower pair.

Only red ink is decisive. Black, blue and grey page furniture is decoration.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = 1152
GRID = 8
CELL = CANVAS // GRID          # 144
HALO = CELL // 4               # keeps a seal inside a single ablation window
SAFE = HALO + 4                # inset of the placement box inside its cell

WHITE = (255, 255, 255)
RED = (204, 0, 0)
BLACK = (30, 30, 30)
BLUE = (30, 60, 180)
GREY = (140, 140, 140)
DECOR_COLORS = (BLACK, BLUE, GREY)

OUTER_CHOICES = (46, 50, 54)
STROKE_CHOICES = (5, 6)
CLEAR_CHOICES = (6, 7, 8)
MOUTH_CHOICES = (7, 8, 9, 10, 11)
SIDES = ("top", "bottom", "left", "right")

QUARANTINE_MARGIN = 5.0

# (upper_sealed, lower_a_sealed, lower_b_sealed) options per decision class.
_CLASS_OPTIONS = {
    "equal": ((1, 1, 0), (1, 0, 1), (2, 1, 1), (2, 2, 0), (2, 0, 2)),
    "upper_more": ((1, 0, 0), (2, 0, 0), (2, 1, 0), (2, 0, 1)),
    "lower_more": ((0, 1, 0), (0, 0, 1), (0, 1, 1), (0, 2, 0), (1, 2, 0),
                   (1, 1, 1), (0, 2, 1), (0, 1, 2), (1, 0, 2)),
}
DECISION_CLASSES = ("equal", "upper_more", "lower_more")


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------
def _sample_seal(rng: random.Random, sealed: int, tight: bool) -> dict:
    """One seal with `sealed` unbroken rings among one or two drawn rings."""
    for _ in range(400):
        n_rings = rng.choice((1, 2)) if sealed <= 1 else 2
        if n_rings < sealed:
            continue
        outer = rng.choice(OUTER_CHOICES)
        stroke = rng.choice(STROKE_CHOICES)
        clear = rng.choice(CLEAR_CHOICES)
        if tight:
            stroke = 5
            clear = 6
        sizes = [outer]
        if n_rings == 2:
            sizes.append(outer - 2 * (stroke + clear))
        if sizes[-1] - 2 * stroke < (4 if tight else 6):
            continue
        # which drawn rings stay unbroken
        idx = list(range(n_rings))
        rng.shuffle(idx)
        open_idx = set(idx[: n_rings - sealed])
        rings = []
        ok = True
        for i, size in enumerate(sizes):
            span = size - 2 * stroke - 4
            if i in open_idx:
                width = min(rng.choice(MOUTH_CHOICES), span)
                if tight:
                    width = min(width, 4)
                if width < (4 if tight else 6):
                    ok = False
                    break
                rings.append({"size": size, "stroke": stroke,
                              "mouth_side": rng.choice(SIDES),
                              "mouth_width": width})
            else:
                rings.append({"size": size, "stroke": stroke,
                              "mouth_side": None, "mouth_width": 0})
        if ok:
            return {"rings": rings}
    raise RuntimeError("seal sampling failed")


def _place(rng: random.Random, cell_row: int, cell_col: int, size: int) -> tuple[int, int]:
    """Centre of a seal of `size` pixels, jittered inside one grid cell."""
    lo = cell_col * CELL + SAFE
    hi = cell_col * CELL + CELL - SAFE - size
    x0 = rng.randint(lo, max(lo, hi))
    lo = cell_row * CELL + SAFE
    hi = cell_row * CELL + CELL - SAFE - size
    y0 = rng.randint(lo, max(lo, hi))
    return x0 + size // 2, y0 + size // 2


def _sample_decor(rng: random.Random, blocked: set[tuple[int, int]]) -> list[dict]:
    """Non-overlapping page furniture, one item per free grid cell at most."""
    free = [(r, c) for r in range(GRID) for c in range(GRID) if (r, c) not in blocked]
    rng.shuffle(free)
    decor = []
    for row, col in free[: rng.randint(11, 16)]:
        kind = rng.choice(("ring", "ring", "bar", "block"))
        color = rng.choice(DECOR_COLORS)
        x0 = col * CELL + rng.randint(10, 26)
        y0 = row * CELL + rng.randint(10, 26)
        if kind == "ring":
            size = rng.choice((44, 52, 60, 68))
            stroke = rng.choice((5, 6, 7))
            nested = rng.choice((1, 1, 2))
            item = {"kind": "ring", "color": list(color), "x": x0, "y": y0,
                    "size": size, "stroke": stroke, "nested": nested,
                    "mouth_side": rng.choice(SIDES + (None,)),
                    "mouth_width": rng.choice((8, 10, 12))}
        elif kind == "bar":
            item = {"kind": "bar", "color": list(color), "x": x0, "y": y0,
                    "w": rng.choice((70, 90, 104)), "h": rng.choice((6, 9, 12)),
                    "size": 0, "stroke": 0, "nested": 0,
                    "mouth_side": None, "mouth_width": 0}
        else:
            item = {"kind": "block", "color": list(color), "x": x0, "y": y0,
                    "w": rng.choice((26, 34, 46)), "h": rng.choice((26, 34, 46)),
                    "size": 0, "stroke": 0, "nested": 0,
                    "mouth_side": None, "mouth_width": 0}
        decor.append(item)
    return decor


def sample_scene(seed: int) -> dict:
    """Deterministically sample one scene from an integer seed."""
    rng = random.Random(0x5EA1 * 7 + seed)
    target = DECISION_CLASSES[seed % 3]
    counts = rng.choice(_CLASS_OPTIONS[target])
    tight = (seed % 7 == 3)

    row = rng.randint(0, GRID - 3)
    col = rng.randint(0, GRID - 3)
    upper_col_offset = rng.choice((1,))
    seals = []
    # upper seal, then the two lower seals; list order carries no information.
    specs = [
        (row, col + upper_col_offset, counts[0]),
        (row + 2, col, counts[1]),
        (row + 2, col + 2, counts[2]),
    ]
    order = [0, 1, 2]
    rng.shuffle(order)
    for index in order:
        cell_row, cell_col, sealed = specs[index]
        seal = _sample_seal(rng, sealed, tight)
        cx, cy = _place(rng, cell_row, cell_col, seal["rings"][0]["size"])
        seal["cx"] = cx
        seal["cy"] = cy
        seals.append(seal)

    blocked = {(r, c) for r in range(row, row + 3) for c in range(col, col + 3)}
    decor = _sample_decor(rng, blocked)
    return {"canvas": CANVAS, "cluster_cell": [row, col], "seals": seals,
            "decor": decor, "tight": tight}


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def _draw_ring(draw: ImageDraw.ImageDraw, cx: int, cy: int, size: int, stroke: int,
               color, mouth_side, mouth_width: int) -> None:
    half = size // 2
    x0, y0, x1, y1 = cx - half, cy - half, cx - half + size - 1, cy - half + size - 1
    draw.rectangle([x0, y0, x1, y1], outline=tuple(color), width=stroke)
    if not mouth_side or mouth_width <= 0:
        return
    half_m = mouth_width // 2
    if mouth_side == "top":
        box = [cx - half_m, y0, cx - half_m + mouth_width - 1, y0 + stroke - 1]
    elif mouth_side == "bottom":
        box = [cx - half_m, y1 - stroke + 1, cx - half_m + mouth_width - 1, y1]
    elif mouth_side == "left":
        box = [x0, cy - half_m, x0 + stroke - 1, cy - half_m + mouth_width - 1]
    else:
        box = [x1 - stroke + 1, cy - half_m, x1, cy - half_m + mouth_width - 1]
    draw.rectangle(box, fill=WHITE)


def render(scene: dict) -> Image.Image:
    """Rasterise the scene. Axis-aligned, unantialiased, byte-deterministic."""
    size = int(scene["canvas"])
    image = Image.new("RGB", (size, size), WHITE)
    draw = ImageDraw.Draw(image)
    for item in sorted(scene["decor"], key=lambda d: (d["y"], d["x"])):
        color = tuple(item["color"])
        if item["kind"] == "ring":
            outer = item["size"]
            stroke = item["stroke"]
            for level in range(item["nested"]):
                cur = outer - 2 * level * (stroke + 8)
                if cur - 2 * stroke < 6:
                    break
                _draw_ring(draw, item["x"] + outer // 2, item["y"] + outer // 2,
                           cur, stroke, color,
                           item["mouth_side"] if level == 0 else None,
                           item["mouth_width"])
        else:
            draw.rectangle([item["x"], item["y"],
                            item["x"] + item["w"] - 1, item["y"] + item["h"] - 1],
                           fill=color)
    for seal in sorted(scene["seals"], key=lambda s: (s["cy"], s["cx"])):
        for ring in sorted(seal["rings"], key=lambda r: -r["size"]):
            _draw_ring(draw, seal["cx"], seal["cy"], ring["size"], ring["stroke"],
                       RED, ring["mouth_side"], ring["mouth_width"])
    return image


# --------------------------------------------------------------------------
# analytic consequences of the scene
# --------------------------------------------------------------------------
def _roles(scene: dict) -> tuple[dict, list[dict]]:
    ordered = sorted(scene["seals"], key=lambda s: s["cy"])
    return ordered[0], ordered[1:]


def _sealed(rings: list) -> int:
    """Number of unbroken outlines in one rendered ring stack."""
    return sum(1 for ring in rings if not ring["mouth_side"])


def analytic_gold(scene: dict) -> str:
    upper, lower = _roles(scene)
    up = _sealed(upper["rings"])
    low = sum(_sealed(seal["rings"]) for seal in lower)
    if up == low:
        return "equal"
    return "upper_more" if up > low else "lower_more"


def margin(scene: dict) -> float:
    """Weakest decisive width in pixels over all three seals."""
    widths = []
    for seal in scene["seals"]:
        strokes = [ring["stroke"] for ring in seal["rings"]]
        widths.extend(strokes)
        sizes = sorted((ring["size"] for ring in seal["rings"]), reverse=True)
        widths.append(sizes[-1] - 2 * strokes[-1])          # innermost hole
        if len(sizes) == 2:
            widths.append((sizes[0] - sizes[1]) // 2 - strokes[0])  # clearance
        for ring in seal["rings"]:
            if ring["mouth_side"]:
                widths.append(ring["mouth_width"])
    return float(min(widths))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical relabelings of the latent scene."""
    import copy

    out = []
    permuted = copy.deepcopy(scene)
    permuted["decor"] = list(reversed(permuted["decor"]))
    out.append(("reverse_decor_order", permuted))

    reordered = copy.deepcopy(scene)
    reordered["seals"] = [reordered["seals"][i] for i in (2, 0, 1)]
    out.append(("rotate_seal_list_order", reordered))

    rings = copy.deepcopy(scene)
    for seal in rings["seals"]:
        seal["rings"] = list(reversed(seal["rings"]))
    out.append(("reverse_ring_order", rings))
    return out
