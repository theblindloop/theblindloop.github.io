"""Deterministic latent sampler and raster renderer.

The digits are deliberately drawn as crisp seven-segment glyphs so that the
independent inverse arm can recover symbol identity from the final PNG without
font files, OCR services, renderer state, or latent data.
"""

from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

CANVAS_WIDTH = 920
CANVAS_HEIGHT = 380
BACKGROUND = (250, 250, 247)
INK = (22, 24, 28)
REFERENCE_COLOR = (190, 45, 180)
CARD_COLORS = {
    "red": (220, 55, 55),
    "green": (40, 155, 82),
    "blue": (45, 105, 215),
}

_SEGMENTS: dict[int, frozenset[str]] = {
    2: frozenset("abdeg"),
    3: frozenset("abcdg"),
    4: frozenset("bcfg"),
    5: frozenset("acdfg"),
    6: frozenset("acdefg"),
    7: frozenset("abc"),
    8: frozenset("abcdefg"),
    9: frozenset("abcdfg"),
}


def sample_scene(seed: int) -> dict:
    """Sample a balanced, unique reciprocal match.

    Sequential seeds cycle answer color every scene and answer position every
    three scenes. The two one-role distractors have the same total card/border
    ink as the target and each agrees with exactly one of the required roles.
    """

    rng = random.Random(int(seed) * 7919 + 104729)
    digits = list(_SEGMENTS)
    rng.shuffle(digits)
    reference_top, reference_bottom, wrong_bottom, wrong_top = digits[:4]

    target_color = ("red", "green", "blue")[int(seed) % 3]
    target_position = (int(seed) // 3) % 3
    by_color = {
        target_color: {
            "color": target_color,
            "top": reference_bottom,
            "bottom": reference_top,
        }
    }
    remaining_colors = [c for c in ("red", "green", "blue") if c != target_color]
    if rng.randrange(2):
        remaining_colors.reverse()
    by_color[remaining_colors[0]] = {
        "color": remaining_colors[0],
        "top": reference_bottom,
        "bottom": wrong_bottom,
    }
    by_color[remaining_colors[1]] = {
        "color": remaining_colors[1],
        "top": wrong_top,
        "bottom": reference_top,
    }

    other_cards = [by_color[c] for c in remaining_colors]
    cards: list[dict] = []
    other_index = 0
    for position in range(3):
        card = by_color[target_color] if position == target_position else other_cards[other_index]
        if position != target_position:
            other_index += 1
        cards.append(dict(card))

    return {
        "reference_top": reference_top,
        "reference_bottom": reference_bottom,
        "cards": cards,
        "canvas_width": CANVAS_WIDTH,
        "canvas_height": CANVAS_HEIGHT,
    }


def _draw_digit(draw: ImageDraw.ImageDraw, center_x: int, top_y: int, digit: int) -> None:
    width, height, thick = 60, 80, 10
    left = center_x - width // 2
    mid = height // 2
    boxes = {
        "a": (left + thick, top_y, left + width - thick, top_y + thick),
        "b": (left + width - thick, top_y + thick, left + width, top_y + mid),
        "c": (left + width - thick, top_y + mid, left + width, top_y + height - thick),
        "d": (left + thick, top_y + height - thick, left + width - thick, top_y + height),
        "e": (left, top_y + mid, left + thick, top_y + height - thick),
        "f": (left, top_y + thick, left + thick, top_y + mid),
        "g": (left + thick, top_y + mid - thick // 2, left + width - thick, top_y + mid + thick // 2),
    }
    for name in _SEGMENTS[int(digit)]:
        draw.rectangle(boxes[name], fill=INK)


def _draw_fraction_card(
    draw: ImageDraw.ImageDraw,
    box: tuple[int, int, int, int],
    border: tuple[int, int, int],
    top_digit: int,
    bottom_digit: int,
) -> None:
    x0, y0, x1, y1 = box
    draw.rectangle(box, fill=(255, 255, 255), outline=border, width=8)
    center_x = (x0 + x1) // 2
    fraction_y = (y0 + y1) // 2
    draw.rectangle((center_x - 58, fraction_y - 3, center_x + 58, fraction_y + 3), fill=INK)

    upper_start, upper_end = y0 + 10, fraction_y - 8
    lower_start, lower_end = fraction_y + 8, y1 - 10
    _draw_digit(draw, center_x, (upper_start + upper_end - 80) // 2, int(top_digit))
    _draw_digit(draw, center_x, (lower_start + lower_end - 80) // 2, int(bottom_digit))


def render(scene: dict) -> Image.Image:
    width = int(scene["canvas_width"])
    height = int(scene["canvas_height"])
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)

    try:
        font = ImageFont.load_default(size=19)
        small = ImageFont.load_default(size=15)
    except TypeError:  # older Pillow still yields deterministic bitmap text
        font = ImageFont.load_default()
        small = font
    draw.text((124, 24), "REFERENCE", fill=INK, anchor="mm", font=font)
    draw.text((590, 24), "CANDIDATES", fill=INK, anchor="mm", font=font)
    draw.text((124, 50), "numerator / denominator", fill=(70, 72, 78), anchor="mm", font=small)
    draw.text((590, 50), "find the reciprocal", fill=(70, 72, 78), anchor="mm", font=small)

    card_y0, card_y1 = 82, 342
    _draw_fraction_card(
        draw,
        (34, card_y0, 214, card_y1),
        REFERENCE_COLOR,
        int(scene["reference_top"]),
        int(scene["reference_bottom"]),
    )
    candidate_x = (270, 485, 700)
    cards = scene["cards"]
    for x0, card in zip(candidate_x, cards, strict=True):
        _draw_fraction_card(
            draw,
            (x0, card_y0, x0 + 180, card_y1),
            CARD_COLORS[str(card["color"])],
            int(card["top"]),
            int(card["bottom"]),
        )
    return image


def _scores(scene: dict) -> list[tuple[int, str]]:
    wanted_top = int(scene["reference_bottom"])
    wanted_bottom = int(scene["reference_top"])
    return [
        (
            int(int(card["top"]) == wanted_top) + int(int(card["bottom"]) == wanted_bottom),
            str(card["color"]),
        )
        for card in scene["cards"]
    ]


def analytic_gold(scene: dict) -> str:
    wanted_top = int(scene["reference_bottom"])
    wanted_bottom = int(scene["reference_top"])
    matches = [
        str(card["color"])
        for card in scene["cards"]
        if int(card["top"]) == wanted_top and int(card["bottom"]) == wanted_bottom
    ]
    if len(matches) != 1:
        raise ValueError(f"expected one reciprocal card, found {len(matches)}")
    return matches[0]


def margin(scene: dict) -> float:
    ranked = sorted((score for score, _ in _scores(scene)), reverse=True)
    return float(ranked[0] - ranked[1]) if len(ranked) >= 2 else 0.0


def is_quarantined(scene: dict) -> bool:
    scores = [score for score, _ in _scores(scene)]
    return scores.count(2) != 1 or margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
