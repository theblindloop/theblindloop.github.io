"""Latent sampler and raster renderer for Refractory Pulse Plaque."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 512
BACKGROUND = (248, 247, 242)
PLAQUE = (37, 52, 70)
GOLD = (245, 184, 46)
CYAN = (0, 183, 199)       # PULSE
CORAL = (242, 100, 90)     # RECHARGE
LIGHT = (245, 244, 238)
DECOR = (207, 211, 210)
TILE = 14
GAP = 2
PAD = 4
PLAQUE_W = PLAQUE_H = PAD * 2 + TILE * 3 + GAP * 2


def _replay(initial_on: bool, events: list[bool]) -> bool:
    """True event is PULSE; false event is one RECHARGE step."""
    lamp = bool(initial_on)
    cooldown = 0  # 0 READY, 1 WARMING, 2 EXHAUSTED
    for pulse in events:
        if pulse:
            if cooldown == 0:
                lamp = not lamp
                cooldown = 2
        else:
            cooldown = max(0, cooldown - 1)
    return lamp


def sample_scene(seed: int) -> dict:
    """Balance answer, initial state, last event, and total color area."""
    rng = random.Random(int(seed))
    desired = bool(seed & 1)
    initial = bool((seed >> 1) & 1)
    terminal_pulse = bool((seed >> 2) & 1)
    pool = []
    for bits in itertools.product((False, True), repeat=6):
        events = list(bits)
        if sum(events) != 3 or events[-1] != terminal_pulse:
            continue
        if _replay(initial, events) == desired:
            pool.append(events)
    if not pool:
        raise RuntimeError("balanced event catalog unexpectedly empty")
    pool.sort()
    events = list(pool[rng.randrange(len(pool))])

    anchors = [(96, 96), (256, 96), (416, 96),
               (96, 256), (256, 256), (416, 256),
               (96, 416), (256, 416), (416, 416)]
    ax, ay = anchors[rng.randrange(len(anchors))]
    cx, cy = ax + rng.randint(-12, 12), ay + rng.randint(-12, 12)
    decor = []
    for _ in range(9):
        for _attempt in range(100):
            x, y = rng.randint(20, 492), rng.randint(20, 492)
            if abs(x - cx) > 70 or abs(y - cy) > 70:
                decor.append([x, y, rng.randint(2, 4)])
                break
    return {
        "initial_on": initial,
        "events": events,
        "position": [cx, cy],
        "decor": decor,
        "alias_nonce": rng.randrange(1_000_000),
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x, y, radius in scene["decor"]:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=DECOR)

    cx, cy = scene["position"]
    left = int(cx - PLAQUE_W // 2)
    top = int(cy - PLAQUE_H // 2)
    draw.rounded_rectangle(
        (left, top, left + PLAQUE_W - 1, top + PLAQUE_H - 1),
        radius=7,
        fill=PLAQUE,
    )
    values = [None] + list(scene["events"])
    for index, value in enumerate(values):
        row, col = divmod(index, 3)
        x0 = left + PAD + col * (TILE + GAP)
        y0 = top + PAD + row * (TILE + GAP)
        x1, y1 = x0 + TILE - 1, y0 + TILE - 1
        if index == 0:
            draw.rectangle((x0, y0, x1, y1), fill=GOLD)
            inset = 4
            fill = PLAQUE if scene["initial_on"] else LIGHT
            draw.rectangle((x0 + inset, y0 + inset, x1 - inset, y1 - inset), fill=fill)
        else:
            draw.rectangle((x0, y0, x1, y1), fill=CYAN if value else CORAL)
    return image


def analytic_gold(scene: dict) -> str:
    return "yes" if _replay(scene["initial_on"], scene["events"]) else "no"


def margin(scene: dict) -> float:
    return 10.0


def is_quarantined(scene: dict) -> bool:
    events = scene.get("events", [])
    return len(events) != 6 or sum(bool(value) for value in events) != 3


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["alias_nonce"] = int(scene.get("alias_nonce", 0)) + 1
    return [("unused_nonce_increment", twin)]
