"""Independent final-PNG-only inverse for Refractory Pulse Plaque."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

CYAN = np.array((0, 183, 199), dtype=np.int16)
CORAL = np.array((242, 100, 90), dtype=np.int16)
GOLD = np.array((245, 184, 46), dtype=np.int16)
DARK = np.array((37, 52, 70), dtype=np.int16)
LIGHT = np.array((245, 244, 238), dtype=np.int16)


def _mask(array: np.ndarray, color: np.ndarray, tolerance: float = 18.0) -> np.ndarray:
    delta = array.astype(np.int32) - color.astype(np.int32)
    return np.sum(delta * delta, axis=2, dtype=np.int32) <= tolerance * tolerance


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for sy, sx in np.argwhere(mask):
        if seen[sy, sx]:
            continue
        queue = deque([(int(sy), int(sx))])
        seen[sy, sx] = True
        xs, ys = [], []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        output.append((len(xs), min(xs), min(ys), max(xs), max(ys)))
    return output


def _solid_tiles(array: np.ndarray, color: np.ndarray, pulse: bool) -> list[tuple[float, float, bool]]:
    items = []
    for area, x0, y0, x1, y1 in _components(_mask(array, color)):
        if area < 20:
            continue
        width, height = x1 - x0 + 1, y1 - y0 + 1
        if not (13 <= width <= 15 and 13 <= height <= 15 and 180 <= area <= 220):
            raise ValueError("damaged event tile")
        items.append(((x0 + x1) / 2, (y0 + y1) / 2, pulse))
    return items


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 64 or array.shape[1] < 64:
        raise ValueError("invalid raster")

    events = _solid_tiles(array, CYAN, True) + _solid_tiles(array, CORAL, False)
    if len(events) != 6 or sum(item[2] for item in events) != 3:
        raise ValueError("expected six balanced event tiles")
    golds = [item for item in _components(_mask(array, GOLD)) if item[0] >= 20]
    if len(golds) != 1:
        raise ValueError("expected one START frame")
    area, x0, y0, x1, y1 = golds[0]
    if not (145 <= area <= 175 and 13 <= x1 - x0 + 1 <= 15 and 13 <= y1 - y0 + 1 <= 15):
        raise ValueError("damaged START frame")
    start = ((x0 + x1) / 2, (y0 + y1) / 2)

    centers = [(start[0], start[1], None)] + events
    xs = sorted({round(item[0]) for item in centers})
    ys = sorted({round(item[1]) for item in centers})
    if len(xs) != 3 or len(ys) != 3:
        raise ValueError("plaque grid is incomplete")
    if not all(15 <= b - a <= 17 for a, b in zip(xs, xs[1:])):
        raise ValueError("ambiguous plaque columns")
    if not all(15 <= b - a <= 17 for a, b in zip(ys, ys[1:])):
        raise ValueError("ambiguous plaque rows")
    ordered = sorted(centers, key=lambda item: (item[1], item[0]))
    if ordered[0][2] is not None:
        raise ValueError("START must be first in row order")
    # The seven occupied cells are exactly the first seven cells of a 3x3 grid.
    expected = [(xs[i % 3], ys[i // 3]) for i in range(7)]
    observed = [(round(item[0]), round(item[1])) for item in ordered]
    if observed != expected:
        raise ValueError("event order is not uniquely recoverable")

    patch = array[int(start[1]) - 2:int(start[1]) + 3, int(start[0]) - 2:int(start[0]) + 3]
    mean = patch.astype(np.float64).mean(axis=(0, 1))
    d_dark = float(np.linalg.norm(mean - DARK))
    d_light = float(np.linalg.norm(mean - LIGHT))
    if abs(d_dark - d_light) < 80:
        raise ValueError("ambiguous START fill")
    lamp = d_dark < d_light
    cooldown = 0
    for _, _, pulse in ordered[1:]:
        if pulse:
            if cooldown == 0:
                lamp = not lamp
                cooldown = 2
        else:
            cooldown = max(0, cooldown - 1)
    return "yes" if lamp else "no"
