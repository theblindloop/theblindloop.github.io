"""Deterministic latent scene and Pillow renderer for Streak Summit Quilt."""
from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

W, H = 720, 620
ROWS, COLS = 6, 8
CW, CH = 80, 65
BASE_X, BASE_Y = 40, 166
PALETTES = (
    {"a": (38, 164, 185), "b": (231, 105, 78), "gold": (246, 190, 43), "ink": (27, 34, 45), "paper": (249, 247, 239), "panel": (237, 233, 222)},
    {"a": (78, 91, 174), "b": (222, 151, 53), "gold": (244, 192, 47), "ink": (30, 35, 45), "paper": (250, 248, 241), "panel": (238, 235, 226)},
)


def _maximum_run(tiles: list[int]) -> int:
    current = best = 1
    for previous, value in zip(tiles, tiles[1:]):
        if value == previous:
            current += 1
        else:
            current = 1
        best = max(best, current)
    return best


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    target = 3 + (seed % 3)
    for _ in range(20000):
        tiles = [rng.randrange(2) for _ in range(ROWS * COLS)]
        ones = sum(tiles)
        # Both colors remain abundant, and both halves contain transitions, so
        # neither total ink nor one local patch predicts the answer.
        left_changes = sum(a != b for a, b in zip(tiles[:23], tiles[1:24]))
        right_changes = sum(a != b for a, b in zip(tiles[24:-1], tiles[25:]))
        if 18 <= ones <= 30 and left_changes >= 7 and right_changes >= 7 and _maximum_run(tiles) == target:
            return {
                "rows": ROWS,
                "cols": COLS,
                "tiles": tiles,
                "palette": rng.randrange(len(PALETTES)),
                "offset_x": rng.choice((-8, 0, 8)),
                "offset_y": rng.choice((-6, 0, 6)),
            }
    raise RuntimeError("balanced scene sampling failed")


def analytic_gold(scene: dict) -> str:
    return str(_maximum_run([int(v) for v in scene["tiles"]]))


def margin(scene: dict) -> float:
    p = PALETTES[int(scene["palette"])]
    colors = [p[k] for k in ("a", "b", "gold", "ink", "paper")]
    return float(min(sum((a - b) ** 2 for a, b in zip(x, y)) ** 0.5 for i, x in enumerate(colors) for y in colors[i + 1 :]))


def is_quarantined(scene: dict) -> bool:
    return analytic_gold(scene) not in ("3", "4", "5")


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int, bold: bool = False):
    path = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf" if bold else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
    try:
        return ImageFont.truetype(path, size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    p = PALETTES[int(scene["palette"])]
    im = Image.new("RGB", (W, H), p["paper"])
    d = ImageDraw.Draw(im)
    ink = p["ink"]
    d.text((W // 2, 22), "STREAK SUMMIT QUILT", anchor="ma", font=_font(25, True), fill=ink)
    d.rounded_rectangle((38, 60, 682, 142), radius=12, fill=p["panel"], outline=ink, width=3)
    d.text((W // 2, 74), "MATCH PREVIOUS COLOR:  STREAK +1", anchor="ma", font=_font(16, True), fill=ink)
    d.text((W // 2, 104), "COLOR CHANGES:  RESET STREAK TO 1", anchor="ma", font=_font(16, True), fill=ink)
    d.text((W // 2, 120), "KEEP THE LARGEST STREAK EVER REACHED", anchor="ma", font=_font(14), fill=ink)

    x0 = BASE_X + int(scene["offset_x"])
    y0 = BASE_Y + int(scene["offset_y"])
    for index, value in enumerate(scene["tiles"]):
        r, c = divmod(index, COLS)
        xa, ya = x0 + c * CW, y0 + r * CH
        d.rectangle((xa, ya, xa + CW, ya + CH), fill=p["a" if int(value) == 0 else "b"], outline=ink, width=3)

    # The inset outline marks the first tile but leaves a broad flat-color center.
    d.rounded_rectangle((x0 + 8, y0 + 8, x0 + CW - 8, y0 + CH - 8), radius=8, outline=p["gold"], width=7)
    d.text((W // 2, 584), "READ LEFT TO RIGHT, THEN TOP TO BOTTOM", anchor="mm", font=_font(16, True), fill=ink)
    return im
