"""Independent pixel-only inverse arm; imports no candidate implementation."""
from __future__ import annotations

import numpy as np
from PIL import Image

ROWS, COLS = 6, 8
CW, CH = 80, 65
PALETTES = (
    {"a": (38, 164, 185), "b": (231, 105, 78), "gold": (246, 190, 43)},
    {"a": (78, 91, 174), "b": (222, 151, 53), "gold": (244, 192, 47)},
)


def _distance(array, color):
    return np.linalg.norm(array.astype(float) - np.asarray(color, dtype=float), axis=-1)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    if arr.shape != (620, 720, 3):
        raise ValueError("unexpected raster size")

    # Infer palette and the START tile's footprint from the only gold component.
    palette_scores = [
        int(np.count_nonzero(_distance(arr, p["a"]) < 14))
        + int(np.count_nonzero(_distance(arr, p["b"]) < 14))
        for p in PALETTES
    ]
    palette_index = int(np.argmax(palette_scores))
    p = PALETTES[palette_index]
    ys, xs = np.where(_distance(arr, p["gold"]) < 18)
    if not (900 <= len(xs) <= 1700):
        raise ValueError("missing or corrupted START outline")
    gx0, gx1, gy0, gy1 = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    if not (58 <= gx1 - gx0 <= 66 and 43 <= gy1 - gy0 <= 51):
        raise ValueError("ambiguous START outline")
    x0, y0 = gx0 - 8, gy0 - 8
    if not (30 <= x0 <= 50 and 154 <= y0 <= 178):
        raise ValueError("invalid quilt placement")

    tiles: list[int] = []
    for r in range(ROWS):
        for c in range(COLS):
            cx, cy = x0 + c * CW + CW // 2, y0 + r * CH + CH // 2
            # The first center contains the START word, so sample two broad
            # off-center patches; all other cells use the same redundant probes.
            probes = np.concatenate(
                [arr[cy - 10 : cy - 3, cx - 24 : cx - 14].reshape(-1, 3),
                 arr[cy + 4 : cy + 11, cx + 14 : cx + 24].reshape(-1, 3)],
                axis=0,
            )
            a_fraction = float(np.mean(_distance(probes, p["a"]) < 16))
            b_fraction = float(np.mean(_distance(probes, p["b"]) < 16))
            if a_fraction > 0.96 and b_fraction < 0.02:
                tiles.append(0)
            elif b_fraction > 0.96 and a_fraction < 0.02:
                tiles.append(1)
            else:
                raise ValueError("missing or ambiguous quilt tile")

    if len(tiles) != ROWS * COLS or not (18 <= sum(tiles) <= 30):
        raise ValueError("incomplete or degenerate quilt")
    if sum(a != b for a, b in zip(tiles[:23], tiles[1:24])) < 7:
        raise ValueError("corrupted first half")
    if sum(a != b for a, b in zip(tiles[24:-1], tiles[25:])) < 7:
        raise ValueError("corrupted second half")

    current = best = 1
    for previous, value in zip(tiles, tiles[1:]):
        current = current + 1 if value == previous else 1
        best = max(best, current)
    if best not in (3, 4, 5):
        raise ValueError("answer outside declared boundary")
    return str(best)
