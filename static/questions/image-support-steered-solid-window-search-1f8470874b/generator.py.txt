"""Deterministic renderer and analytic arm for solid-window search."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

GRID = 6
INK_CELLS = 18
CELL = 14
CANVAS = 512
CYAN = (24, 166, 188)
NAVY = (24, 40, 61)
PAPER = (249, 247, 240)
PANEL = (224, 221, 211)
LABELS = ("A", "B", "C", "D")


def _window_scores(cells: Iterable[Iterable[int]]) -> list[int]:
    occupied = {tuple(p) for p in cells}
    return [
        sum((r + dr, c + dc) in occupied for dr in (0, 1) for dc in (0, 1))
        for r in range(GRID - 1)
        for c in range(GRID - 1)
    ]


def _score(cells: Iterable[Iterable[int]]) -> int:
    return max(_window_scores(cells), default=0)


_TARGET_BASES = (
    "04 10 13 14 15 20 21 22 23 32 33 34 35 43 45 52 53 55",
    "02 03 04 14 15 22 24 32 34 40 41 42 43 44 45 51 52 55",
    "02 12 13 14 20 21 22 24 25 32 33 34 35 41 42 44 54 55",
    "00 01 02 03 04 12 14 22 23 24 25 31 33 34 41 42 43 51",
)
_DISTRACTOR_BASES = (
    "00 01 02 04 10 12 13 14 15 20 21 23 25 32 33 34 43 53",
    "00 02 04 10 11 12 13 14 15 21 24 31 32 33 34 41 44 51",
    "00 01 02 04 12 13 14 15 22 24 32 33 34 35 44 52 53 54",
)


def _decode(spec: str) -> set[tuple[int, int]]:
    return {(int(token[0]), int(token[1])) for token in spec.split()}


def _transform(cells: set[tuple[int, int]], variant: int) -> list[list[int]]:
    out = set(cells)
    if variant & 4:
        out = {(r, GRID - 1 - c) for r, c in out}
    for _ in range(variant & 3):
        out = {(c, GRID - 1 - r) for r, c in out}
    return [list(p) for p in sorted(out)]


def _sample_mask(rng: random.Random, target: bool) -> list[list[int]]:
    pool = _TARGET_BASES if target else _DISTRACTOR_BASES
    cells = _transform(_decode(rng.choice(pool)), rng.randrange(8))
    assert len(cells) == INK_CELLS
    assert _score(cells) == (4 if target else 3)
    assert not target or _window_scores(cells).count(4) == 1
    return cells


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    answer_index = int(seed) % 4
    tiles = [_sample_mask(rng, i == answer_index) for i in range(4)]
    base = [(72, 82), (356, 82), (72, 346), (356, 346)]
    # Small scene-dependent translations randomize absolute cue position while
    # retaining the four visibly labeled regions.
    origins = [[x + rng.randrange(-14, 15, 14), y + rng.randrange(-14, 15, 14)] for x, y in base]
    target_cells = {tuple(p) for p in tiles[answer_index]}
    solid_anchors = [
        (r, c)
        for r in range(GRID - 1)
        for c in range(GRID - 1)
        if all((r + dr, c + dc) in target_cells for dr in (0, 1) for dc in (0, 1))
    ]
    assert len(solid_anchors) == 1
    r0, c0 = solid_anchors[0]
    # Put the small decisive square over a local 64-pixel intersection. Across
    # balanced answers these intersections occupy all four canvas quadrants.
    cue_x = 128 if answer_index % 2 == 0 else 384
    cue_y = 128 if answer_index < 2 else 384
    origins[answer_index] = [cue_x - (c0 + 1) * CELL, cue_y - (r0 + 1) * CELL]
    return {
        "tiles": tiles,
        "cell_size": CELL,
        "origins": origins,
        "canvas_size": CANVAS,
        "palette": {"background": list(PAPER), "cyan": list(CYAN), "ink": list(NAVY)},
    }


def analytic_gold(scene: dict) -> str:
    scores = [_score(tile) for tile in scene["tiles"]]
    best = max(scores)
    winners = [i for i, value in enumerate(scores) if value == best]
    return LABELS[winners[0]] if len(winners) == 1 else "abstain"


def margin(scene: dict) -> float:
    scores = sorted((_score(tile) for tile in scene["tiles"]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    scores = [_score(tile) for tile in scene["tiles"]]
    return scores.count(max(scores)) != 1 or max(scores) != 4 or margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas_size"])
    palette = scene["palette"]
    image = Image.new("RGB", (size, size), tuple(palette["background"]))
    draw = ImageDraw.Draw(image)
    cell = int(scene["cell_size"])
    font = ImageFont.load_default(size=18)
    for index, (tile, origin) in enumerate(zip(scene["tiles"], scene["origins"], strict=True)):
        ox, oy = map(int, origin)
        # A quiet backing card and label orient the human solver; neither encodes
        # the winner, and all cards have equal cyan area.
        draw.rounded_rectangle(
            (ox - 18, oy - 34, ox + GRID * cell + 18, oy + GRID * cell + 18),
            radius=10,
            fill=tuple(palette["background"]),
            outline=PANEL,
            width=3,
        )
        draw.text((ox + GRID * cell / 2, oy - 24), LABELS[index], fill=tuple(palette["ink"]), font=font, anchor="mm")
        for r, c in tile:
            x0, y0 = ox + int(c) * cell, oy + int(r) * cell
            draw.rectangle((x0, y0, x0 + cell - 1, y0 + cell - 1), fill=tuple(palette["cyan"]))
    return image
