"""Independent image-only inverse arm for solid-window search."""

from __future__ import annotations

from collections import defaultdict

import numpy as np
from PIL import Image

_CYAN = np.array((24, 166, 188), dtype=np.int16)
_CELL = 14
_LABELS = ("A", "B", "C", "D")


def _best_lattice(mask: np.ndarray) -> set[tuple[int, int]]:
    """Recover occupied cells while deriving lattice phase from visible pixels."""
    ys, xs = np.nonzero(mask)
    if len(xs) < 200:
        return set()
    # Filled-cell edges make projection changes at the hidden lattice phase.
    # Vote by change magnitude, which is robust to a small erased patch.
    xdiff = np.abs(np.diff(mask.sum(axis=0).astype(np.int32)))
    ydiff = np.abs(np.diff(mask.sum(axis=1).astype(np.int32)))
    xvotes = np.bincount((np.nonzero(xdiff)[0] + 1) % _CELL, weights=xdiff[xdiff > 0], minlength=_CELL)
    yvotes = np.bincount((np.nonzero(ydiff)[0] + 1) % _CELL, weights=ydiff[ydiff > 0], minlength=_CELL)
    px, py = int(np.argmax(xvotes)), int(np.argmax(yvotes))
    rows = (ys - py) // _CELL
    cols = (xs - px) // _CELL
    keys = rows.astype(np.int64) * 1000 + cols.astype(np.int64)
    unique, counts = np.unique(keys, return_counts=True)
    return {
        (int(key // 1000), int(key % 1000))
        for key, count in zip(unique, counts, strict=True)
        if count >= 0.62 * _CELL * _CELL
    }


def _tile_score(mask: np.ndarray) -> int:
    cells = _best_lattice(mask)
    if len(cells) < 6:
        return -1
    return max(
        (
            sum((r + dr, c + dc) in cells for dr in (0, 1) for dc in (0, 1))
            for r, c in cells
        ),
        default=-1,
    )


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    cyan = np.max(np.abs(arr - _CYAN), axis=2) <= 12
    h, w = cyan.shape
    # Enumerate the four relative candidate regions. The inverse uses neither
    # renderer state nor stored scene coordinates; small translations are benign.
    regions = (
        cyan[: h // 2, : w // 2],
        cyan[: h // 2, w // 2 :],
        cyan[h // 2 :, : w // 2],
        cyan[h // 2 :, w // 2 :],
    )
    scores = [_tile_score(region) for region in regions]
    best = max(scores)
    winners = [i for i, score in enumerate(scores) if score == best]
    if best != 4 or len(winners) != 1:
        return "abstain"
    return _LABELS[winners[0]]
