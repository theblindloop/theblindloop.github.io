"""Deterministic solar-system diagram renderer and analytic arm."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

SIZE = 512
SCALE = 4
CENTER = (256, 256)
PLANET_RADIUS = 13
SUN_RADIUS = 25

THEMES = {
    "cool": {
        "background": (248, 250, 253),
        "ring": (118, 118, 118),
        "other": [(220, 72, 62), (48, 164, 91), (155, 82, 190), (231, 129, 38)],
    },
    "warm": {
        "background": (253, 249, 242),
        "ring": (118, 118, 118),
        "other": [(201, 65, 72), (48, 151, 99), (143, 75, 180), (225, 120, 35)],
    },
}
BLUE = (28, 104, 218)
YELLOW = (246, 190, 24)
OUTLINE = (35, 39, 46)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    blue_orbit = int(seed) % 5 + 1
    inner = rng.randint(58, 64)
    gaps = [rng.randint(34, 40) for _ in range(4)]
    orbit_radii = [inner]
    for gap in gaps:
        orbit_radii.append(orbit_radii[-1] + gap)
    base_rotation = rng.randrange(0, 360)
    planet_angles = [base_rotation + rng.randrange(0, 360) for _ in range(5)]
    theme = rng.choice(sorted(THEMES))
    other_names = ["red", "green", "purple", "orange"]
    rng.shuffle(other_names)
    planet_colors: list[str] = []
    other_index = 0
    for orbit in range(1, 6):
        if orbit == blue_orbit:
            planet_colors.append("blue")
        else:
            planet_colors.append(other_names[other_index])
            other_index += 1
    return {
        "orbit_radii": orbit_radii,
        "planet_angles": planet_angles,
        "planet_colors": planet_colors,
        "palette_theme": theme,
        "blue_orbit": blue_orbit,
    }


def _planet_rgb(name: str, theme: str) -> tuple[int, int, int]:
    if name == "blue":
        return BLUE
    order = {"red": 0, "green": 1, "purple": 2, "orange": 3}
    return THEMES[theme]["other"][order[name]]


def render(scene: dict) -> Image.Image:
    theme_name = str(scene["palette_theme"])
    theme = THEMES[theme_name]
    radii = [int(value) for value in scene["orbit_radii"]]
    angles = [int(value) % 360 for value in scene["planet_angles"]]
    colors = list(scene["planet_colors"])
    blue_orbit = int(scene["blue_orbit"])
    if len(radii) != 5 or len(angles) != 5 or len(colors) != 5:
        raise ValueError("scene must contain five orbits and five planets")
    if colors.count("blue") != 1 or colors[blue_orbit - 1] != "blue":
        raise ValueError("blue_orbit must point to the uniquely blue planet")

    image = Image.new("RGB", (SIZE * SCALE, SIZE * SCALE), theme["background"])
    draw = ImageDraw.Draw(image)
    cx, cy = (CENTER[0] * SCALE, CENTER[1] * SCALE)
    for radius in radii:
        rr = radius * SCALE
        draw.ellipse(
            (cx - rr, cy - rr, cx + rr, cy + rr),
            outline=theme["ring"],
            width=3 * SCALE,
        )

    sr = SUN_RADIUS * SCALE
    draw.ellipse(
        (cx - sr, cy - sr, cx + sr, cy + sr),
        fill=YELLOW,
        outline=OUTLINE,
        width=2 * SCALE,
    )
    for radius, angle, color_name in zip(radii, angles, colors, strict=True):
        theta = math.radians(angle)
        px = int(round((CENTER[0] + radius * math.cos(theta)) * SCALE))
        py = int(round((CENTER[1] + radius * math.sin(theta)) * SCALE))
        pr = PLANET_RADIUS * SCALE
        draw.ellipse(
            (px - pr, py - pr, px + pr, py + pr),
            fill=_planet_rgb(color_name, theme_name),
            outline=OUTLINE,
            width=2 * SCALE,
        )
    return image.resize((SIZE, SIZE), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(int(scene["blue_orbit"]))


def margin(scene: dict) -> float:
    radii = sorted(float(value) for value in scene["orbit_radii"])
    target = radii[int(scene["blue_orbit"]) - 1]
    return min(abs(target - radius) for radius in radii if radius != target)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 24.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["planet_angles"] = [int(value) + 360 for value in scene["planet_angles"]]
    return [("full_turn_angle_alias", twin)]
