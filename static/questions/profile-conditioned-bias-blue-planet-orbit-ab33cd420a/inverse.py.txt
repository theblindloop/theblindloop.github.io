"""Scene-free inverse arm using only pixels from the final PNG."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _centroid(mask: np.ndarray, label: str) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 180:
        raise ValueError(f"missing or too-small {label} region")
    return float(xs.mean()), float(ys.mean())


def _ring_radii(arr: np.ndarray, cx: float, cy: float) -> list[float]:
    maximum = arr.max(axis=2)
    minimum = arr.min(axis=2)
    mean = arr.mean(axis=2)
    gray = (maximum - minimum < 8) & (mean > 95) & (mean < 155)
    ys, xs = np.nonzero(gray)
    if len(xs) < 500:
        raise ValueError("insufficient gray orbit evidence")
    radial = np.hypot(xs.astype(float) - cx, ys.astype(float) - cy)
    histogram = np.bincount(np.rint(radial).astype(int), minlength=260)
    active = np.flatnonzero(histogram > 90)
    groups: list[list[int]] = []
    for radius in active.tolist():
        if not groups or radius - groups[-1][-1] > 3:
            groups.append([radius])
        else:
            groups[-1].append(radius)
    peaks = []
    for group in groups:
        weights = histogram[group].astype(float)
        peaks.append(float(np.average(np.asarray(group, dtype=float), weights=weights)))
    peaks = [value for value in peaks if 40.0 < value < 235.0]
    if len(peaks) != 5:
        raise ValueError(f"expected five visible orbit rings, recovered {len(peaks)}")
    return sorted(peaks)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = arr[:, :, 0], arr[:, :, 1], arr[:, :, 2]
    blue_mask = (blue > 145) & (blue - red > 75) & (blue - green > 35)
    yellow_mask = (red > 190) & (green > 160) & (blue < 95) & (red - blue > 100)
    bx, by = _centroid(blue_mask, "blue planet")
    sx, sy = _centroid(yellow_mask, "yellow Sun")
    rings = _ring_radii(arr, sx, sy)
    blue_radius = float(np.hypot(bx - sx, by - sy))
    errors = [abs(blue_radius - radius) for radius in rings]
    order = np.argsort(errors)
    if errors[int(order[0])] > 5.0:
        raise ValueError("blue planet center is not on a visible orbit")
    if errors[int(order[1])] - errors[int(order[0])] < 18.0:
        raise ValueError("ambiguous orbit assignment")
    return str(int(order[0]) + 1)
