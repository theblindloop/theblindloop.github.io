"""Latent sampler and renderer for Sampling Aperture Capture Balance."""

from __future__ import annotations

import math
import random
from typing import Iterable

import numpy as np
from PIL import Image, ImageDraw, ImageFont

WIDTH = 512
HEIGHT = 512
BACKGROUND = (248, 248, 244)
TEAL = (18, 139, 145)
GOLD = (226, 161, 43)
INK = (48, 55, 60)
RING_WIDTH = 7
QUARANTINE_MARGIN = 1500


def _masks(polygon: Iterable[Iterable[int]], aperture: Iterable[int]) -> tuple[np.ndarray, np.ndarray]:
    """Construct latent occupancy masks without inspecting the final image."""
    poly = [tuple(map(int, point)) for point in polygon]
    cx, cy, radius = map(int, aperture)
    specimen_image = Image.new("1", (WIDTH, HEIGHT), 0)
    ImageDraw.Draw(specimen_image).polygon(poly, fill=1)
    ring_image = Image.new("1", (WIDTH, HEIGHT), 0)
    ImageDraw.Draw(ring_image).ellipse(
        (cx - radius, cy - radius, cx + radius, cy + radius),
        outline=1,
        width=RING_WIDTH,
    )
    return np.asarray(specimen_image, dtype=bool), np.asarray(ring_image, dtype=bool)


def _visible_counts(polygon: Iterable[Iterable[int]], aperture: Iterable[int]) -> tuple[int, int]:
    specimen, ring = _masks(polygon, aperture)
    cx, cy, radius = map(int, aperture)
    yy, xx = np.indices(specimen.shape)
    visible = specimen & ~ring
    interior = (xx - cx) ** 2 + (yy - cy) ** 2 < radius**2
    return int(np.count_nonzero(visible & interior)), int(np.count_nonzero(visible & ~interior))


def sample_scene(seed: int) -> dict:
    """Sample a balanced, well-separated one-component specimen scene."""
    rng = random.Random(int(seed))
    desired = "INSIDE" if int(seed) % 2 == 0 else "OUTSIDE"
    center_x = 256 + rng.randint(-14, 14)
    center_y = 267 + rng.randint(-14, 14)
    aperture = [
        center_x + rng.randint(-7, 7),
        center_y + rng.randint(-7, 7),
        rng.randint(78, 84),
    ]
    phase = rng.uniform(0.0, 2.0 * math.pi)

    best: tuple[float, list[list[int]]] | None = None
    for _ in range(4000):
        points: list[list[int]] = []
        # Sorted polar vertices make a simple, connected, star-shaped polygon.
        harmonic_phase = rng.uniform(0.0, 2.0 * math.pi)
        for index in range(20):
            angle = phase + 2.0 * math.pi * index / 20.0
            radius = (
                106.0
                + rng.uniform(-37.0, 37.0)
                + 14.0 * math.sin(3.0 * angle + harmonic_phase)
            )
            radius = max(58.0, min(151.0, radius))
            x = round(center_x + radius * math.cos(angle))
            y = round(center_y + radius * math.sin(angle))
            points.append([x, y])
        inside, outside = _visible_counts(points, aperture)
        total = inside + outside
        signed = inside - outside
        decision = "INSIDE" if signed > 0 else "OUTSIDE"
        separation = abs(signed)
        # Both answers share the same broad ink and aperture-size ranges.
        if 30000 <= total <= 40500 and decision == desired and 1550 <= separation <= 9000:
            best = (separation, points)
            break
    if best is None:
        raise RuntimeError(f"could not sample a separated {desired} scene for seed {seed}")
    return {
        "polygon": best[1],
        "aperture": aperture,
        "specimen_color": list(TEAL),
        "aperture_color": list(GOLD),
        "background": list(BACKGROUND),
        "width": WIDTH,
        "height": HEIGHT,
    }


def render(scene: dict) -> Image.Image:
    """Render a clean scientific sampling plate at a fixed pixel scale."""
    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(scene["background"])
    specimen_color = tuple(scene["specimen_color"])
    aperture_color = tuple(scene["aperture_color"])
    polygon = [tuple(point) for point in scene["polygon"]]
    cx, cy, radius = map(int, scene["aperture"])
    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((82, 88, 430, 446), radius=13, outline=(184, 188, 184), width=2)
    draw.polygon(polygon, fill=specimen_color)
    draw.ellipse(
        (cx - radius, cy - radius, cx + radius, cy + radius),
        outline=aperture_color,
        width=RING_WIDTH,
    )
    font = ImageFont.load_default(size=18)
    draw.text((256, 55), "SAMPLING APERTURE", fill=INK, anchor="mm", font=font)
    return image


def analytic_gold(scene: dict) -> str:
    inside, outside = _visible_counts(scene["polygon"], scene["aperture"])
    return "INSIDE" if inside > outside else "OUTSIDE"


def margin(scene: dict) -> float:
    inside, outside = _visible_counts(scene["polygon"], scene["aperture"])
    return float(abs(inside - outside))


def is_quarantined(scene: dict) -> bool:
    inside, outside = _visible_counts(scene["polygon"], scene["aperture"])
    return abs(inside - outside) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Exercise the cyclic-first-vertex alias of a closed polygon."""
    polygon = list(scene["polygon"])
    twin = dict(scene)
    twin["polygon"] = polygon[1:] + polygon[:1]
    return [("cyclic_polygon_vertex_origin", twin)]
