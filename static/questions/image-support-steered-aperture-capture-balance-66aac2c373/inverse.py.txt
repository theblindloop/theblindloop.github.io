"""Independent PNG-only inverse arm for aperture capture balance."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

TEAL = np.array((18, 139, 145), dtype=np.int32)
GOLD = np.array((226, 161, 43), dtype=np.int32)
MIN_SPECIMEN_PIXELS = 12000
MIN_DECISION_MARGIN = 350


def _near(array: np.ndarray, color: np.ndarray, tolerance: float = 22.0) -> np.ndarray:
    delta = array.astype(np.int32) - color
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _recover_aperture(gold: np.ndarray) -> tuple[float, float, float]:
    ys, xs = np.nonzero(gold)
    if len(xs) < 650:
        raise ValueError("gold aperture boundary is missing or incomplete")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    diameter_x = x1 - x0
    diameter_y = y1 - y0
    if not (150 <= diameter_x <= 170 and abs(diameter_x - diameter_y) <= 2):
        raise ValueError("gold aperture does not have one plausible circular extent")
    cx = (x0 + x1) / 2.0
    cy = (y0 + y1) / 2.0
    radius = (diameter_x + diameter_y) / 4.0

    # A destructive crop may preserve the bounding box while deleting an arc.
    # Every angular sector must retain visible gold boundary support.
    angles = (np.arctan2(ys - cy, xs - cx) + 2.0 * math.pi) % (2.0 * math.pi)
    bins = np.floor(angles * 48.0 / (2.0 * math.pi)).astype(int) % 48
    occupancy = np.bincount(bins, minlength=48)
    if int(occupancy.min()) < 5:
        raise ValueError("gold aperture boundary has a missing angular sector")
    return cx, cy, radius


def decision_from_image(image: Image.Image) -> str:
    """Recover the comparison from final RGB pixels and no scene information."""
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    teal = _near(array, TEAL)
    gold = _near(array, GOLD)
    cx, cy, radius = _recover_aperture(gold)
    if int(np.count_nonzero(teal)) < MIN_SPECIMEN_PIXELS:
        raise ValueError("visible teal specimen is missing too much area")
    yy, xx = np.indices(teal.shape)
    interior = (xx - cx) ** 2 + (yy - cy) ** 2 < radius**2
    inside = int(np.count_nonzero(teal & interior))
    outside = int(np.count_nonzero(teal & ~interior))
    if abs(inside - outside) < MIN_DECISION_MARGIN:
        raise ValueError("visible inside/outside areas are too close to resolve")
    return "INSIDE" if inside > outside else "OUTSIDE"
