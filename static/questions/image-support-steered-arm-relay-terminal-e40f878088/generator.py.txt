"""Two-hop arm relay: disconnected sign pads route to one of two colored pads.

The scene places one router pad at the centre of a medium region, three
axis-neighbour pads around it (the entry pad plus two colored terminals) and a
few off-axis gray decoys.  Every element is an axis-aligned rectangle, so the
raster is byte-deterministic and free of antialiasing.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 640
HEIGHT = 460

BG = (250, 250, 248)
GRAY = (140, 143, 148)
ORANGE = (233, 124, 30)
PURPLE = (126, 58, 180)
TEAL = (20, 145, 145)
MARK = (32, 34, 40)

PAD = 40
ARM_W = 10
DOT = 12

_DIRS = ("L", "R", "U", "D")
_STEP = {"L": (-1, 0), "R": (1, 0), "U": (0, -1), "D": (0, 1)}

QUARANTINE_MARGIN = 12.0
ALIGN_TOL = 16.0


def _rect(center: tuple[int, int], w: int, h: int) -> tuple[int, int, int, int]:
    cx, cy = center
    return (cx - w // 2, cy - h // 2, cx - w // 2 + w - 1, cy - h // 2 + h - 1)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)

    h_off = rng.choice([144, 152, 160])
    v_off = rng.choice([120, 128, 136])
    span_x = 2 * h_off + PAD
    span_y = 2 * v_off + PAD
    slack_x = (WIDTH - span_x) // 2 - 8
    slack_y = (HEIGHT - span_y) // 2 - 8
    cx = WIDTH // 2 + rng.randint(-slack_x, slack_x)
    cy = HEIGHT // 2 + rng.randint(-slack_y, slack_y)

    dirs = list(_DIRS)
    rng.shuffle(dirs)
    entry_dir, orange_dir, purple_dir = dirs[0], dirs[1], dirs[2]
    arm_dir = rng.choice([orange_dir, purple_dir])

    entry_arm_len = rng.choice([22, 26, 30])
    router_arm_len = rng.choice([22, 26, 30])

    # Off-axis decoys: diagonal fractions of the lattice offsets, so a decoy is
    # never aligned with a lattice row or column unless the jitter pushes it
    # close, which is exactly the near-boundary case the margin reports.
    decoys = []
    for index in range(rng.choice([1, 2, 3])):
        sx = rng.choice([-1, 1])
        sy = rng.choice([-1, 1])
        fx = rng.choice([0.52, 0.60, 0.68, 0.76])
        fy = rng.choice([0.40, 0.50, 0.62, 0.74])
        dx = int(round(sx * fx * h_off)) + rng.randint(-6, 6)
        dy = int(round(sy * fy * v_off)) + rng.randint(-6, 6)
        px = cx + dx
        py = cy + dy
        if not (PAD <= px <= WIDTH - PAD and PAD <= py <= HEIGHT - PAD):
            continue
        if all(abs(px - qx) > PAD + 14 or abs(py - qy) > PAD + 14 for qx, qy in decoys):
            decoys.append([px, py])
        del index

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "center": [cx, cy],
        "h_off": h_off,
        "v_off": v_off,
        "entry_dir": entry_dir,
        "orange_dir": orange_dir,
        "purple_dir": purple_dir,
        "arm_dir": arm_dir,
        "entry_arm_len": entry_arm_len,
        "router_arm_len": router_arm_len,
        "pad": PAD,
        "decoys": decoys,
    }


def _site(scene: dict, direction: str) -> tuple[int, int]:
    cx, cy = scene["center"]
    sx, sy = _STEP[direction]
    return (cx + sx * scene["h_off"], cy + sy * scene["v_off"])


def _arm_rect(pad_center: tuple[int, int], direction: str, length: int) -> tuple[int, int, int, int]:
    """Arm attached flush to one side of the pad, extending outward."""

    cx, cy = pad_center
    half = PAD // 2
    if direction in ("L", "R"):
        sign = -1 if direction == "L" else 1
        base = cx + sign * half
        tip = base + sign * length
        x0, x1 = (min(base, tip), max(base, tip))
        return (x0, cy - ARM_W // 2, x1, cy - ARM_W // 2 + ARM_W - 1)
    sign = -1 if direction == "U" else 1
    base = cy + sign * half
    tip = base + sign * length
    y0, y1 = (min(base, tip), max(base, tip))
    return (cx - ARM_W // 2, y0, cx - ARM_W // 2 + ARM_W - 1, y1)


def _elements(scene: dict) -> list[dict]:
    """Every drawn element as a typed rectangle, in painting order."""

    cx, cy = scene["center"]
    router = (cx, cy)
    entry = _site(scene, scene["entry_dir"])
    items = [
        {"kind": "pad", "color": GRAY, "role": "router", "rect": _rect(router, PAD, PAD), "center": router},
        {"kind": "pad", "color": GRAY, "role": "entry", "rect": _rect(entry, PAD, PAD), "center": entry},
    ]
    for site_dir, color, role in (
        (scene["orange_dir"], ORANGE, "orange"),
        (scene["purple_dir"], PURPLE, "purple"),
    ):
        center = _site(scene, site_dir)
        items.append(
            {"kind": "pad", "color": color, "role": role, "rect": _rect(center, PAD, PAD), "center": center}
        )
    for px, py in scene["decoys"]:
        items.append(
            {"kind": "pad", "color": GRAY, "role": "decoy", "rect": _rect((px, py), PAD, PAD), "center": (px, py)}
        )
    items.append(
        {
            "kind": "arm",
            "color": TEAL,
            "role": "entry_arm",
            "rect": _arm_rect(entry, _opposite(scene["entry_dir"]), scene["entry_arm_len"]),
            "center": entry,
        }
    )
    items.append(
        {
            "kind": "arm",
            "color": TEAL,
            "role": "router_arm",
            "rect": _arm_rect(router, scene["arm_dir"], scene["router_arm_len"]),
            "center": router,
        }
    )
    items.append({"kind": "mark", "color": MARK, "role": "entry_dot", "rect": _rect(entry, DOT, DOT), "center": entry})
    return items


def _opposite(direction: str) -> str:
    return {"L": "R", "R": "L", "U": "D", "D": "U"}[direction]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (scene["width"], scene["height"]), BG)
    draw = ImageDraw.Draw(image)
    for item in _elements(scene):
        draw.rectangle(item["rect"], fill=item["color"])
    return image


def analytic_gold(scene: dict) -> str:
    """The second hop lands on the pad the router arm points at."""

    return "orange" if scene["arm_dir"] == scene["orange_dir"] else "purple"


def _gap(a: tuple[int, int, int, int], b: tuple[int, int, int, int]) -> float:
    dx = max(a[0] - b[2], b[0] - a[2], 0)
    dy = max(a[1] - b[3], b[1] - a[3], 0)
    if dx == 0 and dy == 0:
        return 0.0
    if dx and dy:
        return float((dx * dx + dy * dy) ** 0.5) - 1.0
    return float(dx + dy) - 1.0


def _clearances(scene: dict) -> list[float]:
    items = _elements(scene)
    values = []
    for i, first in enumerate(items):
        for second in items[i + 1 :]:
            if first["kind"] == "mark" or second["kind"] == "mark":
                continue
            if first["kind"] == "arm" and first["center"] == second["center"]:
                continue
            if second["kind"] == "arm" and second["center"] == first["center"]:
                continue
            values.append(_gap(first["rect"], second["rect"]))
    # inset of the dark entry marker inside its gray pad
    values.append(float((PAD - DOT) // 2) - 1.0)
    return values


def _alignment_discrimination(scene: dict) -> float:
    """How far the scene is from a decoy stealing a hop.

    For each of the two hops, the intended target is perfectly aligned with the
    arm axis.  Every other pad must be either misaligned or farther away; the
    reported value is the smallest misalignment among pads that lie in the hop
    direction and are not the intended target.
    """

    cx, cy = scene["center"]
    router = (cx, cy)
    entry = _site(scene, scene["entry_dir"])
    hops = [
        (entry, _opposite(scene["entry_dir"]), router),
        (router, scene["arm_dir"], _site(scene, scene["arm_dir"])),
    ]
    others = [tuple(d) for d in scene["decoys"]]
    others.append(_site(scene, scene["orange_dir"]))
    others.append(_site(scene, scene["purple_dir"]))
    others.append(entry)
    others.append(router)

    worst = 1.0e9
    for origin, direction, target in hops:
        sx, sy = _STEP[direction]
        for other in others:
            if other == target or other == origin:
                continue
            along = (other[0] - origin[0]) * sx + (other[1] - origin[1]) * sy
            if along <= 0:
                continue
            perp = abs((other[0] - origin[0]) * sy - (other[1] - origin[1]) * sx)
            along_target = (target[0] - origin[0]) * sx + (target[1] - origin[1]) * sy
            if along > along_target and perp <= ALIGN_TOL:
                # a farther aligned pad is harmless: the hop stops at the first one
                continue
            worst = min(worst, float(perp))
    return worst


def margin(scene: dict) -> float:
    return round(min(min(_clearances(scene)), _alignment_discrimination(scene)), 3)


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_MARGIN)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Latent rewrites that are pixel-identical and answer-preserving."""

    out: list[tuple[str, dict]] = []
    reversed_scene = dict(scene)
    reversed_scene["decoys"] = list(reversed(scene["decoys"]))
    out.append(("decoy_order_reversed", reversed_scene))

    rotated = dict(scene)
    decoys = list(scene["decoys"])
    if decoys:
        decoys = decoys[1:] + decoys[:1]
    rotated["decoys"] = decoys
    out.append(("decoy_order_rotated", rotated))
    return out
