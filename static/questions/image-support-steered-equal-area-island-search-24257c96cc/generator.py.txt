"""Deterministic Pillow renderer for equal-area island search."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

LABELS = ("A", "B", "C", "D")
CYAN = (20, 166, 184)
AMBER = (232, 151, 24)
INK = (30, 42, 52)
FRAME = (174, 184, 190)
BG = (248, 247, 242)

# Connected polyomino silhouettes, normalized inside a 5x5 local lattice.
BASE = ((0, 0), (1, 0), (0, 1), (1, 1), (0, 2), (1, 2), (2, 1),
        (2, 2), (1, 3), (2, 3), (3, 2), (3, 3), (2, 4), (3, 4))


def _connected_shape(n: int, code: int) -> list[tuple[int, int]]:
    """Grow a deterministic connected shape of n cells, then apply a dihedral view."""
    rng = random.Random(7919 * n + code)
    cells = {(2, 2)}
    while len(cells) < n:
        frontier = sorted({(x + dx, y + dy) for x, y in cells
                           for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1))
                           if (x + dx, y + dy) not in cells and 0 <= x + dx < 5 and 0 <= y + dy < 5})
        cells.add(frontier[rng.randrange(len(frontier))])
    pts = list(cells)
    if code & 1:
        pts = [(4 - x, y) for x, y in pts]
    if code & 2:
        pts = [(x, 4 - y) for x, y in pts]
    if code & 4:
        pts = [(y, x) for x, y in pts]
    minx = min(x for x, _ in pts)
    miny = min(y for _, y in pts)
    return sorted((x - minx, y - miny) for x, y in pts)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    splits = [(7, 7), (6, 8), (5, 9), (4, 10)]
    rng.shuffle(splits)
    codes = [[rng.randrange(8), rng.randrange(8)] for _ in range(4)]
    return {"splits": [list(pair) for pair in splits], "shape_codes": codes}


def _draw_shape(draw: ImageDraw.ImageDraw, cells: list[tuple[int, int]], box: tuple[int, int, int, int], color: tuple[int, int, int]) -> None:
    x0, y0, x1, y1 = box
    width = max(x for x, _ in cells) + 1
    height = max(y for _, y in cells) + 1
    cell = min((x1 - x0) // max(width, 1), (y1 - y0) // max(height, 1), 11)
    ox = x0 + ((x1 - x0) - width * cell) // 2
    oy = y0 + ((y1 - y0) - height * cell) // 2
    for x, y in cells:
        draw.rectangle((ox + x * cell, oy + y * cell, ox + (x + 1) * cell - 1, oy + (y + 1) * cell - 1), fill=color)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (512, 512), BG)
    draw = ImageDraw.Draw(image)
    draw.text((139, 48), "EQUAL-AREA ISLAND SEARCH", fill=INK)
    draw.text((118, 70), "Which card splits its island area equally?", fill=INK)
    origins = ((104, 120), (264, 120), (104, 282), (264, 282))
    for index, ((x, y), split, codes) in enumerate(zip(origins, scene["splits"], scene["shape_codes"], strict=True)):
        draw.rounded_rectangle((x, y, x + 144, y + 140), radius=8, outline=FRAME, width=3)
        draw.text((x + 67, y + 8), LABELS[index], fill=INK)
        left = _connected_shape(int(split[0]), int(codes[0]))
        right = _connected_shape(int(split[1]), int(codes[1]))
        _draw_shape(draw, left, (x + 10, y + 36, x + 70, y + 128), CYAN)
        _draw_shape(draw, right, (x + 75, y + 36, x + 134, y + 128), AMBER)
    return image


def analytic_gold(scene: dict) -> str:
    winners = [i for i, pair in enumerate(scene["splits"]) if int(pair[0]) == int(pair[1])]
    if len(winners) != 1:
        raise ValueError("scene does not have one equal-area card")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    diffs = sorted(abs(int(a) - int(b)) for a, b in scene["splits"])
    return float(diffs[1] - diffs[0])


def is_quarantined(scene: dict) -> bool:
    diffs = sorted(abs(int(a) - int(b)) for a, b in scene["splits"])
    return not (diffs[0] == 0 and diffs[1] >= 2)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
