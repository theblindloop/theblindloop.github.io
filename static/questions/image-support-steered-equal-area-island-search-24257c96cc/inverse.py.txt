"""Independent image-only inverse arm for the equal-area search."""

from __future__ import annotations

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")
_CYAN = np.array((20, 166, 184), dtype=np.int16)
_AMBER = np.array((232, 151, 24), dtype=np.int16)


def _mask(array: np.ndarray, color: np.ndarray) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= 8


def _components(mask: np.ndarray) -> list[tuple[int, float, float]]:
    seen = np.zeros(mask.shape, dtype=bool)
    found: list[tuple[int, float, float]] = []
    height, width = mask.shape
    for sy, sx in np.argwhere(mask):
        if seen[sy, sx]:
            continue
        stack = [(int(sy), int(sx))]
        seen[sy, sx] = True
        points = []
        while stack:
            y, x = stack.pop()
            points.append((y, x))
            for ny, nx in ((y-1, x), (y+1, x), (y, x-1), (y, x+1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= 150:
            ys = np.fromiter((p[0] for p in points), dtype=float)
            xs = np.fromiter((p[1] for p in points), dtype=float)
            found.append((len(points), float(xs.mean()), float(ys.mean())))
    return found


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    cyan = _components(_mask(array, _CYAN))
    amber = _components(_mask(array, _AMBER))
    if len(cyan) != 4 or len(amber) != 4:
        raise ValueError("expected four cyan and four amber islands")
    # Bind islands into cards by mutual proximity, then derive the visible 2x2
    # reading order from pair centroids. No renderer coordinates or scene data
    # are available to this arm.
    remaining = set(range(4))
    pairs = []
    for c in sorted(cyan, key=lambda item: (item[2], item[1])):
        j = min(remaining, key=lambda k: (c[1] - amber[k][1]) ** 2 + (c[2] - amber[k][2]) ** 2)
        remaining.remove(j)
        a = amber[j]
        pairs.append((abs(c[0] - a[0]), (c[1] + a[1]) / 2, (c[2] + a[2]) / 2))
    if remaining:
        raise ValueError("could not bind island pairs")
    by_y = sorted(pairs, key=lambda p: p[2])
    pairs = sorted(by_y[:2], key=lambda p: p[1]) + sorted(by_y[2:], key=lambda p: p[1])
    scores = [p[0] for p in pairs]
    order = sorted(range(4), key=lambda i: scores[i])
    if scores[order[0]] != 0 or scores[order[1]] - scores[order[0]] < 100:
        raise ValueError("no unique equal-area winner with margin")
    return LABELS[order[0]]
