"""Independent image-only inverse for the paired marker world."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

PALETTE = {
    "coral": np.array([224, 82, 72], dtype=np.int16),
    "blue": np.array([48, 108, 202], dtype=np.int16),
}


def _components(mask: np.ndarray) -> list[tuple[int, float, float, int, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        points = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= 40:
            ys = np.fromiter((p[0] for p in points), dtype=np.float64)
            xs = np.fromiter((p[1] for p in points), dtype=np.float64)
            found.append((len(points), float(xs.mean()), float(ys.mean()), int(xs.max()-xs.min()+1), int(ys.max()-ys.min()+1)))
    return found


def _pair(array: np.ndarray, target: np.ndarray) -> tuple[tuple[float, float], tuple[float, float]]:
    # int32 is required here: squared RGB distances can exceed int16.
    distance2 = np.sum((array.astype(np.int32) - target.astype(np.int32)) ** 2, axis=2)
    components = _components(distance2 < 52 ** 2)
    valid = [item for item in components if 360 <= item[0] <= 820 and 21 <= item[3] <= 32 and 21 <= item[4] <= 32 and abs(item[3]-item[4]) <= 4]
    if len(valid) != 2:
        raise ValueError("cannot recover exactly two intact colored markers")
    valid.sort(key=lambda item: item[1])
    left, right = valid
    if right[1] - left[1] < 110:
        raise ValueError("measurement columns are not separable")
    return (left[1], left[2]), (right[1], right[2])


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[2] != 3:
        raise ValueError("expected RGB raster")
    coral_left, coral_right = _pair(array, PALETTE["coral"])
    blue_left, blue_right = _pair(array, PALETTE["blue"])
    coral_change = coral_left[1] - coral_right[1]
    blue_change = blue_left[1] - blue_right[1]
    if abs(coral_change - blue_change) < 14.0:
        raise ValueError("pixel-measured changes are too close")
    return "coral" if coral_change > blue_change else "blue"
