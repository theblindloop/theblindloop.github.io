"""Renderer for paired before/after measurement markers."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

WIDTH = 512
HEIGHT = 512
CORAL = (224, 82, 72)
BLUE = (48, 108, 202)
INK = (39, 45, 55)
GRID = (205, 211, 220)
BACKGROUND = (250, 249, 246)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    winner = "coral" if int(seed) % 2 == 0 else "blue"
    for _ in range(10000):
        small = rng.randint(22, 66)
        large = small + rng.randint(28, 72)
        coral_change, blue_change = ((large, small) if winner == "coral" else (small, large))
        coral_before = rng.randint(260, 390)
        blue_before = rng.randint(260, 390)
        coral_after = coral_before - coral_change
        blue_after = blue_before - blue_change
        if min(coral_after, blue_after) < 118:
            continue
        if abs(coral_before - blue_before) < 46:
            continue
        if abs(coral_after - blue_after) < 46:
            continue
        return {
            "coral_before": coral_before,
            "coral_after": coral_after,
            "blue_before": blue_before,
            "blue_after": blue_after,
            "column_left": rng.randint(132, 144),
            "column_right": rng.randint(368, 380),
            "top": 104 + rng.randint(-4, 4),
            "bottom": 412 + rng.randint(-4, 4),
            "marker_radius": rng.randint(13, 15),
            "tick_step": rng.choice([38, 42, 46]),
            "tick_phase": rng.randint(0, 11),
        }
    raise RuntimeError("could not sample a separated paired-marker scene")


def _font(size: int, bold: bool = False):
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(name, size)


def render(scene: dict) -> Image.Image:
    # Read every declared latent field into the visible construction.
    cb = scene["coral_before"]
    ca = scene["coral_after"]
    bb = scene["blue_before"]
    ba = scene["blue_after"]
    xl = scene["column_left"]
    xr = scene["column_right"]
    top = scene["top"]
    bottom = scene["bottom"]
    radius = scene["marker_radius"]
    tick_step = scene["tick_step"]
    tick_phase = scene["tick_phase"]

    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)

    def line(points, fill, width=1):
        draw.line([(int(x * scale), int(y * scale)) for x, y in points], fill=fill, width=width * scale)

    # A compact, visibly shared measurement scale. It contextualizes the marks
    # but does not join any of the decisive colored components.
    for x in (xl, xr):
        line([(x, top), (x, bottom)], INK, 2)
        y = top + tick_phase
        while y <= bottom:
            line([(x - 9, y), (x + 9, y)], GRID, 2)
            y += tick_step
        line([(x - 13, top), (x + 13, top)], INK, 2)
        line([(x - 13, bottom), (x + 13, bottom)], INK, 2)

    draw.text((xl * scale, 70 * scale), "BEFORE", font=_font(17 * scale, True), fill=INK, anchor="mm")
    draw.text((xr * scale, 70 * scale), "AFTER", font=_font(17 * scale, True), fill=INK, anchor="mm")
    draw.text((256 * scale, 31 * scale), "PAIRED MEASUREMENTS", font=_font(20 * scale, True), fill=INK, anchor="mm")
    draw.text((256 * scale, 461 * scale), "upward = increase", font=_font(15 * scale), fill=(86, 91, 100), anchor="mm")

    def marker(x: int, y: int, color: tuple[int, int, int]):
        r = radius * scale
        cx, cy = x * scale, y * scale
        draw.ellipse((cx - r, cy - r, cx + r, cy + r), fill=color, outline=(255, 255, 255), width=2 * scale)

    marker(xl, cb, CORAL)
    marker(xr, ca, CORAL)
    marker(xl, bb, BLUE)
    marker(xr, ba, BLUE)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    coral_change = scene["coral_before"] - scene["coral_after"]
    blue_change = scene["blue_before"] - scene["blue_after"]
    return "coral" if coral_change > blue_change else "blue"


def margin(scene: dict) -> float:
    coral_change = scene["coral_before"] - scene["coral_after"]
    blue_change = scene["blue_before"] - scene["blue_after"]
    return float(abs(coral_change - blue_change))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 18.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
