"""Deterministic renderer and analytic arm for the reciprocal diagonal seal."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS = 512
GRID_N = 5
TILE = 96
CELL = 14
GRID_OFFSET = 13
NAVY = (24, 43, 61)
CYAN = (53, 203, 207)
IVORY = (241, 235, 211)
GOLD = (245, 183, 49)
PALE = (225, 231, 234)
SHADOW = (205, 213, 217)
WHITE = (255, 255, 255)


def _base_cells(rng: random.Random, fill_pairs: int) -> list[list[int]]:
    cells = [[0 for _ in range(GRID_N)] for _ in range(GRID_N)]
    pairs = [(r, c) for r in range(GRID_N) for c in range(r + 1, GRID_N)]
    rng.shuffle(pairs)
    for r, c in pairs[:fill_pairs]:
        cells[r][c] = cells[c][r] = 1
    return cells


def sample_scene(seed: int) -> dict[str, Any]:
    """Return paired, count-matched yes/no scenes for adjacent integer seeds."""
    pair_seed = int(seed) // 2
    rng = random.Random(0x5EA1 + pair_seed * 7919)
    fill_pairs = 3 + pair_seed % 5
    cells = _base_cells(rng, fill_pairs)
    # Sample all nuisance geometry before the class-specific structural edit so
    # adjacent paired scenes match exactly on position and background marks.
    origin_x = rng.randrange(92, 325)
    origin_y = rng.randrange(92, 325)
    decorations: list[list[int]] = []
    while len(decorations) < 18:
        x, y = rng.randrange(24, 488), rng.randrange(24, 488)
        if origin_x - 24 <= x <= origin_x + TILE + 24 and origin_y - 24 <= y <= origin_y + TILE + 24:
            continue
        decorations.append([x, y, 2 + rng.randrange(3)])
    if int(seed) % 2:
        occupied = [(r, c) for r in range(GRID_N) for c in range(r + 1, GRID_N) if cells[r][c]]
        empty = [(r, c) for r in range(GRID_N) for c in range(r + 1, GRID_N) if not cells[r][c]]
        old_r, old_c = rng.choice(occupied)
        new_r, new_c = rng.choice(empty)
        # Break two different reciprocal pairs while preserving total cyan area.
        if rng.randrange(2):
            old_r, old_c = old_c, old_r
        if rng.randrange(2):
            new_r, new_c = new_c, new_r
        cells[old_r][old_c] = 0
        cells[new_r][new_c] = 1

    return {
        "cells": cells,
        "origin_x": origin_x,
        "origin_y": origin_y,
        "decorations": decorations,
    }


def _mismatch_pairs(scene: dict[str, Any]) -> int:
    cells = scene["cells"]
    return sum(cells[r][c] != cells[c][r] for r in range(GRID_N) for c in range(r + 1, GRID_N))


def analytic_gold(scene: dict[str, Any]) -> str:
    return "yes" if _mismatch_pairs(scene) == 0 else "no"


def margin(scene: dict[str, Any]) -> float:
    mismatches = _mismatch_pairs(scene)
    return 1.0 if mismatches == 0 else float(mismatches)


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        cells = scene["cells"]
        ox, oy = int(scene["origin_x"]), int(scene["origin_y"])
        decorations = scene["decorations"]
        valid_cells = (
            len(cells) == GRID_N
            and all(len(row) == GRID_N for row in cells)
            and all(value in (0, 1) for row in cells for value in row)
        )
        valid_position = 32 <= ox <= CANVAS - TILE - 32 and 32 <= oy <= CANVAS - TILE - 32
        valid_decorations = isinstance(decorations, list) and all(len(dot) == 3 for dot in decorations)
        return not (valid_cells and valid_position and valid_decorations)
    except (KeyError, TypeError, ValueError):
        return True


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (CANVAS, CANVAS), WHITE)
    draw = ImageDraw.Draw(image)
    for x, y, radius in scene["decorations"]:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=PALE)

    ox, oy = int(scene["origin_x"]), int(scene["origin_y"])
    draw.rounded_rectangle((ox + 5, oy + 6, ox + TILE + 5, oy + TILE + 6), radius=8, fill=SHADOW)
    draw.rectangle((ox, oy, ox + TILE - 1, oy + TILE - 1), fill=NAVY)
    cells = scene["cells"]
    for r in range(GRID_N):
        for c in range(GRID_N):
            x0 = ox + GRID_OFFSET + c * CELL
            y0 = oy + GRID_OFFSET + r * CELL
            color = GOLD if r == c else (CYAN if cells[r][c] else IVORY)
            draw.rectangle((x0, y0, x0 + 11, y0 + 11), fill=color)
    return image


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
