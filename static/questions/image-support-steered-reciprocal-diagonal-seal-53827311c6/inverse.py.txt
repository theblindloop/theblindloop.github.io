"""Independent pixel-only inverse arm for the reciprocal diagonal seal.

This module intentionally duplicates the public raster specification and does
not import any candidate module. It locates the one navy seal, validates its
area and gold diagonal, samples the cell interiors, then compares reciprocal
row/column positions.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_NAVY = np.array((24, 43, 61), dtype=np.int16)
_CYAN = np.array((53, 203, 207), dtype=np.int16)
_IVORY = np.array((241, 235, 211), dtype=np.int16)
_GOLD = np.array((245, 183, 49), dtype=np.int16)
_N = 5
_CELL = 14
_OFFSET = 13


def _near(arr: np.ndarray, color: np.ndarray, radius: float = 18.0) -> np.ndarray:
    delta = arr.astype(np.int32) - color.astype(np.int32)
    return np.sqrt(np.sum(delta * delta, axis=2, dtype=np.int32)) <= radius


def _patch_color(arr: np.ndarray, cx: int, cy: int) -> str:
    patch = arr[cy - 3 : cy + 4, cx - 3 : cx + 4].astype(np.float32)
    if patch.shape != (7, 7, 3):
        raise ValueError("cell patch falls outside image")
    mean = patch.mean(axis=(0, 1))
    distances = {
        "cyan": float(np.linalg.norm(mean - _CYAN)),
        "ivory": float(np.linalg.norm(mean - _IVORY)),
        "gold": float(np.linalg.norm(mean - _GOLD)),
    }
    label = min(distances, key=distances.get)
    if distances[label] > 22.0:
        raise ValueError("cell interior is missing or corrupted")
    return label


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    navy = _near(arr, _NAVY)
    ys, xs = np.nonzero(navy)
    if len(xs) < 5400:
        raise ValueError("navy seal support is missing")
    x0, x1, y0, y1 = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    if not (94 <= x1 - x0 + 1 <= 98 and 94 <= y1 - y0 + 1 <= 98):
        raise ValueError("seal bounds are malformed")
    if len(xs) > 5750:
        raise ValueError("unexpected navy region")

    cells: list[list[int]] = [[0] * _N for _ in range(_N)]
    for r in range(_N):
        for c in range(_N):
            cx = x0 + _OFFSET + c * _CELL + 5
            cy = y0 + _OFFSET + r * _CELL + 5
            label = _patch_color(arr, cx, cy)
            if r == c:
                if label != "gold":
                    raise ValueError("visible gold diagonal is incomplete")
            elif label == "cyan":
                cells[r][c] = 1
            elif label != "ivory":
                raise ValueError("off-diagonal cell has unsupported color")
    mirrored = all(cells[r][c] == cells[c][r] for r in range(_N) for c in range(r + 1, _N))
    return "yes" if mirrored else "no"
