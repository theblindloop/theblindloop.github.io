"""Deterministic scene sampler and renderer for shared-transversal search."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont


LABELS = ("A", "B", "C", "D")
PANEL_ORIGINS = ((34, 38), (326, 38), (34, 338), (326, 338))
PANEL_SIZE = (280, 264)
BAR_HEIGHT = 24
BAR_ROWS = (76, 132, 188)
SUPER_SAMPLE = 3

THEMES = (
    {
        "background": "#f3f1ea",
        "panel_fill": "#ffffff",
        "frame_color": "#c8c5ba",
        "badge_color": "#e58a3b",
        "bar_color": "#173f5f",
    },
    {
        "background": "#f2f4ef",
        "panel_fill": "#ffffff",
        "frame_color": "#c5cbc1",
        "badge_color": "#d97845",
        "bar_color": "#315d43",
    },
    {
        "background": "#f5f1f5",
        "panel_fill": "#ffffff",
        "frame_color": "#cec4ce",
        "badge_color": "#dc8540",
        "bar_color": "#573b68",
    },
)


def _bars_from_intervals(intervals: list[tuple[int, int]], rng: random.Random) -> list[dict[str, int]]:
    rng.shuffle(intervals)
    return [
        {"x0": x0, "x1": x1, "y0": y, "y1": y + BAR_HEIGHT}
        for (x0, x1), y in zip(intervals, BAR_ROWS, strict=True)
    ]


def _target_panel(lengths: list[int], rng: random.Random) -> dict[str, Any]:
    common_left = rng.randint(108, 116)
    common_right = common_left + rng.randint(34, 42)
    intervals: list[tuple[int, int]] = []
    shuffled = list(lengths)
    rng.shuffle(shuffled)
    for length in shuffled:
        low = max(8, common_right - length)
        high = min(common_left, 256 - length)
        start = rng.randint(low, high)
        intervals.append((start, start + length))
    shift = rng.randint(-3, 3)
    intervals = [(left + shift, right + shift) for left, right in intervals]
    return {"bars": _bars_from_intervals(intervals, rng)}


def _distractor_panel(lengths: list[int], gap: int, rng: random.Random) -> dict[str, Any]:
    """Make two extreme intervals disjoint while a third visibly bridges them."""
    shuffled = list(lengths)
    rng.shuffle(shuffled)
    left_length, right_length, bridge_length = shuffled
    left_end = 130 - gap // 2
    right_start = left_end + gap
    bridge_start = left_end - rng.randint(10, 18)
    intervals = [
        (left_end - left_length, left_end),
        (right_start, right_start + right_length),
        (bridge_start, bridge_start + bridge_length),
    ]
    shift = rng.randint(-3, 3)
    intervals = [(left + shift, right + shift) for left, right in intervals]
    return {"bars": _bars_from_intervals(intervals, rng)}


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene. Consecutive seeds cycle the answer A/B/C/D."""
    rng = random.Random(int(seed))
    answer_index = int(seed) % 4
    lengths = [80, 92, 104]
    gaps = [20, 28, 36]
    rng.shuffle(gaps)
    panels: list[dict[str, Any]] = []
    gap_iter = iter(gaps)
    for index in range(4):
        if index == answer_index:
            panels.append(_target_panel(lengths, rng))
        else:
            panels.append(_distractor_panel(lengths, next(gap_iter), rng))
    theme = THEMES[(int(seed) // 4) % len(THEMES)]
    canvas = {"width": 640, "height": 640, **theme}
    return {"canvas": canvas, "panels": panels}


def _panel_scores(scene: dict) -> list[float]:
    scores: list[float] = []
    for panel in scene["panels"]:
        bars = panel["bars"]
        scores.append(float(min(bar["x1"] for bar in bars) - max(bar["x0"] for bar in bars)))
    return scores


def analytic_gold(scene: dict) -> str:
    scores = _panel_scores(scene)
    winners = [index for index, score in enumerate(scores) if score >= 0.0]
    if len(winners) != 1:
        raise ValueError(f"expected one common-transversal panel, found {winners} from {scores}")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    ordered = sorted(_panel_scores(scene), reverse=True)
    return float(ordered[0] - ordered[1])


def is_quarantined(scene: dict) -> bool:
    scores = _panel_scores(scene)
    positive = [score for score in scores if score >= 0.0]
    if len(positive) != 1:
        return True
    target = positive[0]
    runner_up = max(score for score in scores if score != target)
    return target < 18.0 or runner_up > -10.0 or target - runner_up < 32.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The scene encoding is direct and has no pixel-identical aliases."""
    return []


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    path = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
    try:
        return ImageFont.truetype(path, size=size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    canvas = scene["canvas"]
    panels = scene["panels"]
    ss = SUPER_SAMPLE
    width, height = int(canvas["width"]), int(canvas["height"])
    image = Image.new("RGB", (width * ss, height * ss), canvas["background"])
    draw = ImageDraw.Draw(image)
    label_font = _font(25 * ss)

    for index, panel in enumerate(panels):
        ox, oy = PANEL_ORIGINS[index]
        pw, ph = PANEL_SIZE
        draw.rounded_rectangle(
            (ox * ss, oy * ss, (ox + pw) * ss, (oy + ph) * ss),
            radius=14 * ss,
            fill=canvas["panel_fill"],
            outline=canvas["frame_color"],
            width=3 * ss,
        )
        badge = (ox + 28, oy + 28)
        draw.ellipse(
            ((badge[0] - 19) * ss, (badge[1] - 19) * ss,
             (badge[0] + 19) * ss, (badge[1] + 19) * ss),
            fill=canvas["badge_color"],
        )
        draw.text(
            (badge[0] * ss, (badge[1] - 1) * ss),
            LABELS[index],
            font=label_font,
            fill="white",
            anchor="mm",
        )
        for bar in panel["bars"]:
            x0 = (ox + int(bar["x0"])) * ss
            x1 = (ox + int(bar["x1"])) * ss
            y0 = (oy + int(bar["y0"])) * ss
            y1 = (oy + int(bar["y1"])) * ss
            draw.rounded_rectangle(
                (x0, y0, x1, y1),
                radius=5 * ss,
                fill=canvas["bar_color"],
            )

    return image.resize((width, height), Image.Resampling.LANCZOS)
