"""Independent image-only inverse arm for shared-transversal search.

The oracle discovers dark rectangular components, groups them into the four
visible cards, measures every left/right endpoint, and evaluates the declared
three-interval intersection. It imports no candidate-side renderer or gold code.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


LABELS = ("A", "B", "C", "D")


def _connected_boxes(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    work = mask.copy()
    height, width = work.shape
    boxes: list[tuple[int, int, int, int, int]] = []
    while True:
        points = np.argwhere(work)
        if len(points) == 0:
            break
        start_y, start_x = (int(value) for value in points[0])
        queue = deque([(start_y, start_x)])
        work[start_y, start_x] = False
        min_x = max_x = start_x
        min_y = max_y = start_y
        area = 0
        while queue:
            y, x = queue.popleft()
            area += 1
            min_x, max_x = min(min_x, x), max(max_x, x)
            min_y, max_y = min(min_y, y), max(max_y, y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        boxes.append((min_x, min_y, max_x + 1, max_y + 1, area))
    return boxes


def _bar_boxes(image: Image.Image) -> list[tuple[int, int, int, int]]:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    # Every allowed bar palette is dark in all channels; frames, cards, badges,
    # and lettering are deliberately outside this public, scene-independent cut.
    dark = arr.max(axis=2) < 145
    found = []
    for x0, y0, x1, y1, area in _connected_boxes(dark):
        width, height = x1 - x0, y1 - y0
        if area >= 700 and width >= 60 and 14 <= height <= 32 and width / height >= 2.2:
            found.append((x0, y0, x1, y1))
    if len(found) != 12:
        raise ValueError(f"expected 12 dark bars, found {len(found)}")
    return found


def _split_at_largest_gap(values: list[float]) -> float:
    ordered = sorted(values)
    if len(ordered) < 2:
        raise ValueError("cannot split fewer than two positions")
    gaps = [ordered[index + 1] - ordered[index] for index in range(len(ordered) - 1)]
    cut = int(np.argmax(np.asarray(gaps)))
    if gaps[cut] < 80.0:
        raise ValueError(f"panel grouping gap is too small: {gaps[cut]}")
    return (ordered[cut] + ordered[cut + 1]) / 2.0


def _groups(image: Image.Image) -> list[list[tuple[int, int, int, int]]]:
    boxes = _bar_boxes(image)
    centers_y = [(box[1] + box[3]) / 2.0 for box in boxes]
    y_cut = _split_at_largest_gap(centers_y)
    rows = [
        [box for box in boxes if (box[1] + box[3]) / 2.0 < y_cut],
        [box for box in boxes if (box[1] + box[3]) / 2.0 >= y_cut],
    ]
    groups: list[list[tuple[int, int, int, int]]] = []
    for row in rows:
        x_cut = _split_at_largest_gap([(box[0] + box[2]) / 2.0 for box in row])
        groups.extend(
            [
                [box for box in row if (box[0] + box[2]) / 2.0 < x_cut],
                [box for box in row if (box[0] + box[2]) / 2.0 >= x_cut],
            ]
        )
    if any(len(group) != 3 for group in groups):
        raise ValueError(f"expected three bars in each panel, got {[len(group) for group in groups]}")
    return groups


def _scores(image: Image.Image) -> list[float]:
    return [
        float(min(box[2] for box in group) - max(box[0] for box in group))
        for group in _groups(image)
    ]


def decision_from_image(image: Image.Image) -> str:
    scores = _scores(image)
    winners = [index for index, score in enumerate(scores) if score >= 0.0]
    if len(winners) != 1:
        raise ValueError(f"image does not show one common-transversal panel: {scores}")
    ordered = sorted(scores, reverse=True)
    if ordered[0] < 12.0 or ordered[0] - ordered[1] < 24.0:
        raise ValueError(f"image-only decision margin is ambiguous: {scores}")
    return LABELS[winners[0]]
