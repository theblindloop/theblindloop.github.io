"""chroma_difference_reveal: red-minus-green channel difference glyph reveal.

Declared transform: view the image as an RGB chroma field and form the
red-minus-green channel difference (R - G) at every pixel.

A single block glyph (the letter A or the letter B) is carried entirely in the
relative level of the red versus the green channel. Inside the glyph the red
channel is pushed clearly above green (R = m + DA, G = m + DB with DA and DB of
opposite sign chosen so the weighted luminance contribution cancels); outside
the glyph red and green are identically balanced (R = G, up to per-pixel noise).
Each pixel also carries an independent decorrelated chromatic-noise triple, and
the base luminance m is uniform across the whole field. Consequently the
weighted grayscale image is a uniform salt-and-pepper field in which the glyph
is invisible: the symbol is recoverable only through the declared operation,
reading the R - G channel difference, which is large and uniformly positive
inside the glyph and zero-mean everywhere else.

The surviving R - G silhouette is the coarse diagonal-band glyph; its rising
"/" (A) versus falling "\\" (B) orientation is the answer.

Latent scene z = {seed, glyph, scale, pos_r, pos_c}.
label = glyph.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RENDERER_VERSION = "chroma_difference_reveal-0.1.0"

W = 256
H = 256

GR = 9          # template rows
GC = 7          # template cols

Q_TOL = 0.15    # quarantine threshold on the analytic A/B correlation margin

DA = 70.0       # red offset inside the glyph
DB = -36.0      # green offset inside the glyph (chosen so weighted luminance cancels)
NOISE = 20.0    # half-width of decorrelated per-channel speckle
MID_LO = 60.0   # base luminance lower bound
MID_HI = 160.0  # base luminance upper bound


def template_array(glyph: str) -> np.ndarray:
    """Block templates on a GR x GC grid.

    A = a thick rising diagonal slash "/",
    B = a thick falling diagonal slash "\\".
    """
    grid = np.zeros((GR, GC), dtype=bool)
    for row in range(GR):
        col_main = (GC - 1) * (GR - 1 - row) // (GR - 1)
        for t in range(-1, 2):
            c = col_main + t
            if 0 <= c < GC:
                if glyph == "A":
                    grid[row, c] = True
                else:
                    grid[row, GC - 1 - c] = True
    return grid


def build_scene(seed, glyph, scale, pos_r, pos_c) -> dict:
    return {
        "seed": int(seed),
        "glyph": glyph,
        "scale": int(scale),
        "pos_r": int(pos_r),
        "pos_c": int(pos_c),
    }


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(int(seed) & 0xFFFFFFFF)
    glyph = "A" if rng.integers(0, 2) else "B"
    scale = int(rng.choice([12, 14, 16]))
    glyph_h = GR * scale
    glyph_w = GC * scale
    max_r = max(H - glyph_h - 24, 19)
    max_c = max(W - glyph_w - 24, 19)
    pos_r = int(rng.integers(18, max_r))
    pos_c = int(rng.integers(18, max_c))
    return build_scene(seed=int(seed), glyph=glyph, scale=scale, pos_r=pos_r, pos_c=pos_c)


def glyph_mask(scene: dict) -> np.ndarray:
    scale = int(scene.get("scale", 14))
    row0 = int(scene.get("pos_r", 18))
    col0 = int(scene.get("pos_c", 18))
    cells = template_array(scene["glyph"])
    block = np.repeat(np.repeat(cells, scale, axis=0), scale, axis=1)
    mask = np.zeros((H, W), dtype=bool)
    r1 = min(H, row0 + block.shape[0])
    c1 = min(W, col0 + block.shape[1])
    mask[row0:r1, col0:c1] = block[: r1 - row0, : c1 - col0]
    return mask


def render(scene: dict) -> Image.Image:
    scene = dict(scene)
    rng = np.random.default_rng(int(scene.get("seed", 0)) & 0xFFFFFFFF)
    ink = glyph_mask(scene)
    m = rng.uniform(MID_LO, MID_HI, size=(H, W))
    n1 = rng.uniform(-NOISE, NOISE, size=(H, W))
    n2 = rng.uniform(-NOISE, NOISE, size=(H, W))
    n3 = rng.uniform(-NOISE, NOISE, size=(H, W))
    r_ch = m + np.where(ink, DA, 0.0) + n1
    g_ch = m + np.where(ink, DB, 0.0) + n2
    b_ch = m + n3
    rgb = np.clip(np.stack([r_ch, g_ch, b_ch], axis=-1), 0, 255)
    return Image.fromarray(rgb.astype(np.uint8), "RGB")


def margin(scene: dict) -> float:
    chosen = template_array(scene["glyph"])
    other = template_array("B" if scene["glyph"] == "A" else "A")
    corr_chosen = float((chosen & chosen).sum()) / max(1, int(chosen.sum()))
    corr_other = float((chosen & other).sum()) / max(1, int(other.sum()))
    return round(corr_chosen - corr_other, 6)


def analytic_gold(scene: dict) -> str:
    return scene["glyph"]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < Q_TOL


def latent_symmetries(scene: dict):
    """not_applicable: the decorrelated chromatic-noise seed, the answer glyph,
    the block scale and the placement row and column each directly change
    rendered pixels, and both color channels and the (hidden, uniform) luminance
    field are explicit and visible, so the latent-to-raster map is injective and
    declared empty."""
    return iter(())
