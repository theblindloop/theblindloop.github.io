"""Independent pixel-only inverse arm for chroma_difference_reveal.

Receives ONLY the final PNG. It re-discovers the revealed symbol by applying the
declared transform: it computes the red-minus-green channel difference at every
pixel. The decorrelated per-channel noise has zero mean in R - G, while the
glyph region carries a large uniform positive R - G offset, so thresholding the
difference yields the clean coarse diagonal-band silhouette. That silhouette is
cropped, resampled to the public grid and its rising "/" (A) versus falling
"\\" (B) orientation is read to select the answer.

No scene data, no analytic gold, no renderer import.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "chroma_difference_reveal-oracle-0.1.0"

GR = 9
GC = 7

DIFF_THRESHOLD = 55.0


def _orientation_scores(binary: np.ndarray) -> dict[str, float]:
    rows, cols = binary.shape
    main = np.zeros((rows, cols), dtype=np.float64)
    anti = np.zeros((rows, cols), dtype=np.float64)
    for r in range(rows):
        c_main = (cols - 1) * (rows - 1 - r) // (rows - 1)
        for t in range(-1, 2):
            c = c_main + t
            if 0 <= c < cols:
                main[r, c] = 1.0
                anti[r, cols - 1 - c] = 1.0
    mass = float(binary.sum())
    if mass <= 0:
        return {"A": 0.0, "B": 0.0}
    return {
        "A": float((binary * main).sum()) / mass,
        "B": float((binary * anti).sum()) / mass,
    }


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    diff = arr[..., 0] - arr[..., 1]
    ink = diff > DIFF_THRESHOLD
    if ink.sum() == 0:
        return "A"
    coords = np.argwhere(ink)
    r0, c0 = coords.min(axis=0)
    r1, c1 = coords.max(axis=0)
    crop = ink[r0 : r1 + 1, c0 : c1 + 1].astype(np.uint8) * 255
    if crop.size == 0:
        return "A"
    resized = np.asarray(
        Image.fromarray(crop).resize((GC, GR), Image.Resampling.BILINEAR), dtype=np.float32
    )
    binary = resized > 127.5
    scores = _orientation_scores(binary)
    return "A" if scores["A"] >= scores["B"] else "B"
