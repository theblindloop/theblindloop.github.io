"""Independent PNG-only inverse arm."""

from __future__ import annotations

import numpy as np
from PIL import Image


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    gray = arr.mean(axis=2)
    ink = gray < 245
    ys, xs = np.nonzero(ink)
    if len(xs) == 0:
        return "abstain"
    x0, x1 = int(xs.min()), int(xs.max()) + 1
    y0, y1 = int(ys.min()), int(ys.max()) + 1
    if not (410 <= x1 - x0 <= 440 and 340 <= y1 - y0 <= 370):
        return "abstain"
    field = gray[y0:y1, x0:x1]
    if float((field >= 245).mean()) > 0.001:
        return "abstain"
    mid = field.shape[1] // 2
    left = field[8:-8, 8:mid - 7]
    right = field[8:-8, mid + 7:-8]
    if left.size == 0 or right.size == 0:
        return "abstain"
    delta = float(left.mean() - right.mean())
    if abs(delta) < 8.0:
        return "abstain"
    return "left" if delta < 0 else "right"
