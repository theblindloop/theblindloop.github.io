"""Renderer and analytic arm for a contiguous gray measurement field."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

WIDTH = HEIGHT = 512
FIELD_W, FIELD_H = 428, 356
ROWS, COLS = 8, 12


def _balanced_noise(rng: random.Random, n: int) -> list[int]:
    vals: list[int] = []
    for _ in range(n // 2):
        value = rng.randint(3, 25)
        vals.extend((value, -value))
    rng.shuffle(vals)
    return vals


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    darker = "left" if seed % 2 == 0 else "right"
    gap = rng.randint(26, 62)
    dark_base = 130 - gap // 2
    light_base = 130 + (gap - gap // 2)
    half_n = ROWS * (COLS // 2)
    left_base, right_base = (dark_base, light_base) if darker == "left" else (light_base, dark_base)
    left = [left_base + v for v in _balanced_noise(rng, half_n)]
    right = [right_base + v for v in _balanced_noise(rng, half_n)]
    matrix = [left[r * 6:(r + 1) * 6] + right[r * 6:(r + 1) * 6] for r in range(ROWS)]
    return {"x0": rng.randint(34, 50), "y0": rng.randint(70, 86), "intensities": matrix}


def _half_means(scene: dict) -> tuple[float, float]:
    values = scene["intensities"]
    left = [float(row[c]) for row in values for c in range(6)]
    right = [float(row[c]) for row in values for c in range(6, 12)]
    return sum(left) / len(left), sum(right) / len(right)


def render(scene: dict) -> Image.Image:
    x0, y0 = int(scene["x0"]), int(scene["y0"])
    values = scene["intensities"]
    image = Image.new("RGB", (WIDTH, HEIGHT), "white")
    draw = ImageDraw.Draw(image)
    x_edges = [x0 + round(i * FIELD_W / COLS) for i in range(COLS + 1)]
    y_edges = [y0 + round(i * FIELD_H / ROWS) for i in range(ROWS + 1)]
    for r in range(ROWS):
        for c in range(COLS):
            value = int(values[r][c])
            draw.rectangle((x_edges[c], y_edges[r], x_edges[c + 1] - 1, y_edges[r + 1] - 1), fill=(value, value, value))
    draw.rectangle((x0, y0, x0 + FIELD_W - 1, y0 + FIELD_H - 1), outline=(12, 12, 12), width=6)
    mid = x0 + FIELD_W // 2
    draw.rectangle((mid - 2, y0, mid + 2, y0 + FIELD_H - 1), fill=(12, 12, 12))
    return image


def analytic_gold(scene: dict) -> str:
    left, right = _half_means(scene)
    return "left" if left < right else "right"


def margin(scene: dict) -> float:
    left, right = _half_means(scene)
    return abs(left - right)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
