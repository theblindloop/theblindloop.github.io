"""Pixel-only inverse for the offset-waist silhouette."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _largest_component(mask: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    best: list[tuple[int, int]] = []
    for yy, xx in np.argwhere(mask):
        y, x = int(yy), int(xx)
        if seen[y, x]:
            continue
        points: list[tuple[int, int]] = []
        queue = deque([(y, x)])
        seen[y, x] = True
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) > len(best):
            best = points
    component = np.zeros_like(mask, dtype=bool)
    if best:
        yy, xx = zip(*best)
        component[np.asarray(yy), np.asarray(xx)] = True
    return component


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    border = np.concatenate(
        (pixels[:3].reshape(-1, 3), pixels[-3:].reshape(-1, 3),
         pixels[:, :3].reshape(-1, 3), pixels[:, -3:].reshape(-1, 3)), axis=0
    )
    background = np.median(border, axis=0)
    distance = np.linalg.norm(pixels.astype(np.float32) - background, axis=2)
    component = _largest_component(distance > 58.0)
    points = np.argwhere(component)
    if len(points) < 1800:
        raise ValueError("no intact sample region")
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    height = int(y1 - y0 + 1)
    width = int(x1 - x0 + 1)
    if not (88 <= height <= 142 and 70 <= width <= 120):
        raise ValueError("sample bounds are outside the declared visual regime")
    row_widths = component[y0:y1 + 1].sum(axis=1).astype(float)
    positive = row_widths[row_widths > 0]
    if len(positive) < 88:
        raise ValueError("sample has too few occupied rows")
    outer_width = float(np.percentile(positive, 90))
    narrow = (row_widths > 0.28 * outer_width) & (row_widths < 0.72 * outer_width)
    narrow_indices = np.flatnonzero(narrow)
    if len(narrow_indices) < 18:
        raise ValueError("narrow waist is missing")
    if np.any(np.diff(narrow_indices) != 1):
        raise ValueError("narrow waist is not one contiguous band")
    waist_start = int(narrow_indices[0])
    waist_end = int(narrow_indices[-1])
    if waist_start < 14 or len(row_widths) - 1 - waist_end < 14:
        raise ValueError("waist lacks two visible outer shoulders")
    above = row_widths[:waist_start]
    below = row_widths[waist_end + 1:]
    if np.mean(above > 0.82 * outer_width) < 0.85 or np.mean(below > 0.82 * outer_width) < 0.85:
        raise ValueError("outer flanges are incomplete")
    waist_center = (waist_start + waist_end) / 2.0
    sample_center = (len(row_widths) - 1) / 2.0
    delta = sample_center - waist_center
    if abs(delta) < 3.5:
        raise ValueError("waist is too close to the raster equality boundary")
    return "top" if delta > 0 else "bottom"
