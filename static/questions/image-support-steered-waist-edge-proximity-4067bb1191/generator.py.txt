"""Render a compact connected sample whose narrow waist is vertically offset."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

PALETTES = (
    ((20, 145, 150), (250, 249, 245)),
    ((213, 82, 104), (250, 250, 252)),
    ((65, 105, 190), (249, 248, 243)),
    ((132, 87, 181), (251, 250, 246)),
    ((215, 121, 35), (249, 250, 252)),
)
QUARANTINE_PX = 8.0


def sample_scene(seed: int) -> dict:
    """Use paired nuisance sampling; parity selects the reflected answer class."""
    rng = random.Random(seed // 2)
    outer_width = rng.randrange(84, 107, 2)
    height = rng.randrange(104, 129, 2)
    waist_width = rng.randrange(34, min(51, outer_width - 31), 2)
    waist_height = rng.randrange(26, 37, 2)
    neutral_flange = (height - waist_height) // 2
    shift = rng.choice((5, 7, 9, 12, 15, 18))
    top_flange_height = neutral_flange - shift if seed % 2 == 0 else neutral_flange + shift
    color, background = rng.choice(PALETTES)
    x = rng.randint(52, 512 - outer_width - 52)
    y = rng.randint(52, 512 - height - 52)
    return {
        "width": 512,
        "height": height,
        "outer_width": outer_width,
        "waist_width": waist_width,
        "waist_height": waist_height,
        "top_flange_height": top_flange_height,
        "x": x,
        "y": y,
        "color": list(color),
        "background": list(background),
    }


def _vertices(scene: dict, scale: int) -> list[tuple[int, int]]:
    x = scene["x"] * scale
    y = scene["y"] * scale
    outer = scene["outer_width"] * scale
    waist = scene["waist_width"] * scale
    full_h = scene["height"] * scale
    waist_h = scene["waist_height"] * scale
    upper = scene["top_flange_height"] * scale
    inset = (outer - waist) // 2
    lower_y = y + upper + waist_h
    return [
        (x, y),
        (x + outer, y),
        (x + outer, y + upper),
        (x + outer - inset, y + upper),
        (x + outer - inset, lower_y),
        (x + outer, lower_y),
        (x + outer, y + full_h),
        (x, y + full_h),
        (x, lower_y),
        (x + inset, lower_y),
        (x + inset, y + upper),
        (x, y + upper),
    ]


def render(scene: dict) -> Image.Image:
    scale = 4
    canvas_width = scene["width"]
    canvas_height = 512
    color = tuple(scene["color"])
    background = tuple(scene["background"])
    image = Image.new("RGB", (canvas_width * scale, canvas_height * scale), background)
    draw = ImageDraw.Draw(image)
    draw.polygon(_vertices(scene, scale), fill=color)
    return image.resize((canvas_width, canvas_height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    signed = scene["height"] - scene["waist_height"] - 2 * scene["top_flange_height"]
    if signed == 0:
        raise ValueError("waist is exactly centered")
    return "top" if signed > 0 else "bottom"


def margin(scene: dict) -> float:
    return float(abs(scene["height"] - scene["waist_height"] - 2 * scene["top_flange_height"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
