"""Independent pixel-only inverse for paired marked-link resilience."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

INK = np.array((31, 41, 55), dtype=np.int16)
YELLOW = np.array((245, 179, 1), dtype=np.int16)


def _near(arr: np.ndarray, color: np.ndarray, tolerance: int = 24) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - color), axis=2) <= tolerance


def _component_sizes(mask: np.ndarray) -> list[int]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    sizes: list[int] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = deque([(y, x)])
        size = 0
        while queue:
            cy, cx = queue.popleft()
            size += 1
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        sizes.append(size)
    return sorted(sizes, reverse=True)


def _panel_effect(ink: np.ndarray, yellow: np.ndarray) -> bool:
    yellow_sizes = _component_sizes(yellow)
    if len([size for size in yellow_sizes if size >= 250]) != 1:
        raise ValueError("panel does not contain exactly one robust yellow link")
    black_sizes = [size for size in _component_sizes(ink) if size >= 300]
    if len(black_sizes) not in (1, 2):
        raise ValueError("black rail remainder is missing or ambiguous")
    if sum(black_sizes) < 4500:
        raise ValueError("too much rail evidence is missing")
    return len(black_sizes) == 2


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    if arr.ndim != 3 or arr.shape[1] < 200:
        raise ValueError("invalid raster")
    ink = _near(arr, INK)
    yellow = _near(arr, YELLOW)
    # The visibly separated halves are recovered from the raster dimensions;
    # no latent graph, filename, metadata, or renderer state is consulted.
    midpoint = arr.shape[1] // 2
    effects = [
        _panel_effect(ink[:, :midpoint], yellow[:, :midpoint]),
        _panel_effect(ink[:, midpoint:], yellow[:, midpoint:]),
    ]
    return "yes" if effects[0] == effects[1] else "no"
