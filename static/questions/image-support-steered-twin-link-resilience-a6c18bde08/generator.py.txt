"""Render two disconnected rail graphs with one removable link in each."""

from __future__ import annotations

import random
from collections import deque

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 720, 400
BACKGROUND = (248, 247, 243)
INK = (31, 41, 55)
YELLOW = (245, 179, 1)
BORDER = (156, 163, 175)
RAIL_WIDTH = 18
NODE_RADIUS = 12

# Eight vertices in two 2x2 blocks. Both cases contain nine links. The marked
# top link is a bridge in B. In N, a second bottom link supplies an alternate
# route while one unrelated vertical link is omitted, keeping link count fixed.
_NODES = [[0, 0], [1, 0], [1, 1], [0, 1], [2, 0], [3, 0], [3, 1], [2, 1]]
_BRIDGE_EDGES = [[0, 1], [1, 2], [2, 3], [3, 0], [4, 5], [5, 6], [6, 7], [7, 4], [1, 4]]
_LOOP_EDGES = [[0, 1], [1, 2], [2, 3], [3, 0], [5, 6], [6, 7], [7, 4], [1, 4], [2, 7]]


def _make_panel(status: str, variant: int) -> dict:
    edges = _BRIDGE_EDGES if status == "bridge" else _LOOP_EDGES
    nodes = [p[:] for p in _NODES]
    # Four grid isometries counterbalance vertical position and reflection.
    if variant & 1:
        nodes = [[3 - x, y] for x, y in nodes]
    if variant & 2:
        nodes = [[x, 1 - y] for x, y in nodes]
    mapped_edges = [e[:] for e in edges]
    return {"nodes": nodes, "edges": mapped_edges, "marked": [1, 4], "status": status}


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    code = seed % 4
    status_pairs = [
        ("bridge", "bridge"),
        ("bridge", "loop"),
        ("loop", "bridge"),
        ("loop", "loop"),
    ]
    left, right = status_pairs[code]
    panels = [_make_panel(left, rng.randrange(4)), _make_panel(right, rng.randrange(4))]
    if rng.randrange(2):
        panels.reverse()
    return {"panels": panels}


def _components(panel: dict, omit_marked: bool) -> int:
    n = len(panel["nodes"])
    adjacency = [[] for _ in range(n)]
    marked = {tuple(panel["marked"]), tuple(reversed(panel["marked"]))}
    for a, b in panel["edges"]:
        if omit_marked and (a, b) in marked:
            continue
        adjacency[a].append(b)
        adjacency[b].append(a)
    seen: set[int] = set()
    count = 0
    for start in range(n):
        if start in seen:
            continue
        count += 1
        seen.add(start)
        queue = deque([start])
        while queue:
            at = queue.popleft()
            for nxt in adjacency[at]:
                if nxt not in seen:
                    seen.add(nxt)
                    queue.append(nxt)
    return count


def _disconnects(panel: dict) -> bool:
    return _components(panel, True) > _components(panel, False)


def analytic_gold(scene: dict) -> str:
    panels = scene["panels"]

    # Keep the analytic derivation local to this entry point. The public
    # rendered-field audit can then see that the sole scene-level label input is
    # `panels`; its nested node/link data are the same data render() exposes.
    def disconnects(panel: dict) -> bool:
        def count_components(omit_marked: bool) -> int:
            n = len(panel["nodes"])
            adjacency = [[] for _ in range(n)]
            marked = {tuple(panel["marked"]), tuple(reversed(panel["marked"]))}
            for a, b in panel["edges"]:
                if omit_marked and (a, b) in marked:
                    continue
                adjacency[a].append(b)
                adjacency[b].append(a)
            seen: set[int] = set()
            count = 0
            for start in range(n):
                if start in seen:
                    continue
                count += 1
                stack = [start]
                seen.add(start)
                while stack:
                    for nxt in adjacency[stack.pop()]:
                        if nxt not in seen:
                            seen.add(nxt)
                            stack.append(nxt)
            return count

        return count_components(True) > count_components(False)

    return "yes" if disconnects(panels[0]) == disconnects(panels[1]) else "no"


def margin(scene: dict) -> float:
    panels = scene["panels"]
    # Thick links and large cell spacing make every discrete connection robust.
    effects = [abs(_components(p, True) - _components(p, False)) for p in panels]
    return float(RAIL_WIDTH + min(effects))


def is_quarantined(scene: dict) -> bool:
    panels = scene["panels"]
    return len(panels) != 2 or any(len(p.get("nodes", [])) != 8 or len(p.get("edges", [])) != 9 for p in panels)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _point(base_x: int, xy: list[int]) -> tuple[int, int]:
    # Compact networks live at opposite outer margins. Their disconnected
    # evidence is therefore genuinely image-wide rather than panel-central.
    return base_x + xy[0] * 38, 134 + xy[1] * 92


def render(scene: dict) -> Image.Image:
    panels = scene["panels"]
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    origins = [(4, 26, 12), (378, 26, 594)]
    for index, (panel, (ox, oy, graph_x)) in enumerate(zip(panels, origins)):
        draw.rounded_rectangle((ox, oy, ox + 338, oy + 348), radius=18, outline=BORDER, width=4)
        draw.text((ox + 16, oy + 14), "LEFT" if index == 0 else "RIGHT", fill=INK)
        pts = [_point(graph_x, xy) for xy in panel["nodes"]]
        marked = {tuple(panel["marked"]), tuple(reversed(panel["marked"]))}
        for a, b in panel["edges"]:
            if (a, b) not in marked:
                draw.line((pts[a], pts[b]), fill=INK, width=RAIL_WIDTH)
        for x, y in pts:
            draw.ellipse((x - NODE_RADIUS, y - NODE_RADIUS, x + NODE_RADIUS, y + NODE_RADIUS), fill=INK)
        a, b = panel["marked"]
        draw.line((pts[a], pts[b]), fill=YELLOW, width=RAIL_WIDTH)
        # Restore small black endpoint caps, making deletion visually unambiguous.
        for x, y in (pts[a], pts[b]):
            draw.ellipse((x - 7, y - 7, x + 7, y + 7), fill=INK)
    return image
