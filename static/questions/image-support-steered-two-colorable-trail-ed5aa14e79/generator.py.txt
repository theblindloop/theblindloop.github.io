"""Latent graph sampler and raster renderer for Two-Colorable Trail."""

from __future__ import annotations

from collections import deque
import math
import random

from PIL import Image, ImageDraw

CANVAS = 512

_PALETTES = (
    {
        "background": (249, 247, 242),
        "navy": (25, 48, 70),
        "station": (207, 221, 226),
    },
    {
        "background": (246, 249, 247),
        "navy": (34, 57, 79),
        "station": (216, 224, 226),
    },
    {
        "background": (250, 246, 244),
        "navy": (22, 45, 72),
        "station": (202, 216, 224),
    },
)


def _adjacency(edges: list[list[int]] | tuple[tuple[int, int], ...], count: int) -> list[list[int]]:
    adjacency = [[] for _ in range(count)]
    for raw_left, raw_right in edges:
        left, right = int(raw_left), int(raw_right)
        if left == right or not (0 <= left < count and 0 <= right < count):
            raise ValueError("invalid undirected link")
        adjacency[left].append(right)
        adjacency[right].append(left)
    return adjacency


def _is_bipartite(edges: list[list[int]] | tuple[tuple[int, int], ...], count: int) -> bool:
    adjacency = _adjacency(edges, count)
    colors = [-1] * count
    for start in range(count):
        if colors[start] >= 0:
            continue
        colors[start] = 0
        queue = deque([start])
        while queue:
            node = queue.popleft()
            for neighbor in adjacency[node]:
                if colors[neighbor] < 0:
                    colors[neighbor] = 1 - colors[node]
                    queue.append(neighbor)
                elif colors[neighbor] == colors[node]:
                    return False
    return True


def _theta_graph(path_lengths: list[int], rng: random.Random) -> tuple[list[list[int]], list[list[int]]]:
    """Embed three internally disjoint left-to-right paths without crossings."""
    centers: list[list[float]] = [[54.0, 256.0], [458.0, 256.0]]
    edges: list[list[int]] = []
    # Three clearly separated but vertically compact routes make the decisive
    # support one long connected band rather than an area-filling oval.
    lane_offsets = (-94.0, 0.0, 94.0)
    lane_wave = (0.0, 0.0, 0.0)
    for lane, length in enumerate(path_lengths):
        previous = 0
        for step in range(1, length):
            t = step / length
            x = 54.0 + 404.0 * t
            y = (
                256.0
                + lane_offsets[lane] * math.sin(math.pi * t)
                + lane_wave[lane] * math.sin(2.0 * math.pi * t)
            )
            index = len(centers)
            centers.append([x, y])
            edges.append([previous, index])
            previous = index
        edges.append([previous, 1])

    # Small independent perturbations suppress accidental collinearity while
    # preserving the three separated route bands.
    for index in range(2, len(centers)):
        centers[index][0] += rng.uniform(-3.0, 3.0)
        centers[index][1] += rng.uniform(-3.0, 3.0)
    return [[round(x), round(y)] for x, y in centers], edges


def sample_scene(seed: int) -> dict:
    value = int(seed)
    rng = random.Random(value ^ 0x2C010A81)
    want_yes = value % 2 == 0
    path_lengths = [3, 3, 5] if want_yes else [2, 4, 5]
    rng.shuffle(path_lengths)
    centers, edges = _theta_graph(path_lengths, rng)

    mirror_x = bool((value // 2) & 1)
    mirror_y = bool((value // 4) & 1)
    scale_x = rng.choice((0.94, 0.97, 1.0))
    scale_y = rng.choice((0.93, 0.97, 1.0))
    dx, dy = rng.randint(-4, 4), rng.randint(-4, 4)
    transformed: list[list[int]] = []
    for x, y in centers:
        x = 256.0 + (x - 256.0) * scale_x
        y = 256.0 + (y - 256.0) * scale_y
        if mirror_x:
            x = CANVAS - x
        if mirror_y:
            y = CANVAS - y
        transformed.append([round(x + dx), round(y + dy)])
    if (value // 8) & 1:
        edges.reverse()
    return {
        "canvas_size": CANVAS,
        "station_centers": transformed,
        "edges": edges,
        "stroke_width": 10 + ((value // 3) % 2),
        "station_radius": 14,
        "palette_variant": (value // 5) % len(_PALETTES),
    }


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas_size"])
    centers = [[int(value) for value in point] for point in scene["station_centers"]]
    edges = [[int(value) for value in edge] for edge in scene["edges"]]
    stroke = int(scene["stroke_width"])
    radius = int(scene["station_radius"])
    palette = _PALETTES[int(scene["palette_variant"])]
    scale = 4
    image = Image.new("RGB", (size * scale, size * scale), palette["background"])
    draw = ImageDraw.Draw(image)
    for left, right in edges:
        draw.line(
            [
                tuple(value * scale for value in centers[left]),
                tuple(value * scale for value in centers[right]),
            ],
            fill=palette["navy"],
            width=stroke * scale,
        )
    outline = 4 * scale
    for x, y in centers:
        box = (
            (x - radius) * scale,
            (y - radius) * scale,
            (x + radius) * scale,
            (y + radius) * scale,
        )
        draw.ellipse(box, fill=palette["station"], outline=palette["navy"], width=outline)
    return image.resize((size, size), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "yes" if _is_bipartite(scene["edges"], len(scene["station_centers"])) else "no"


def margin(scene: dict) -> float:
    # The smallest answer-changing intervention is one complete visible link:
    # deleting an odd-cycle edge repairs a no scene, while adding a forbidden
    # same-color-class link breaks a yes scene.
    _ = _is_bipartite(scene["edges"], len(scene["station_centers"]))
    return 1.0


def _segments_cross(a: list[int], b: list[int], c: list[int], d: list[int]) -> bool:
    def orient(p: list[int], q: list[int], r: list[int]) -> int:
        value = (q[0] - p[0]) * (r[1] - p[1]) - (q[1] - p[1]) * (r[0] - p[0])
        return (value > 0) - (value < 0)

    return orient(a, b, c) * orient(a, b, d) < 0 and orient(c, d, a) * orient(c, d, b) < 0


def is_quarantined(scene: dict) -> bool:
    centers = [[int(value) for value in point] for point in scene["station_centers"]]
    edges = [[int(value) for value in edge] for edge in scene["edges"]]
    radius = int(scene["station_radius"])
    stroke = int(scene["stroke_width"])
    size = int(scene["canvas_size"])
    if len(centers) != 10 or len(edges) != 11:
        return True
    try:
        adjacency = _adjacency(edges, len(centers))
    except ValueError:
        return True
    if sorted(len(items) for items in adjacency) != [2] * 8 + [3, 3]:
        return True
    seen = {0}
    queue = deque([0])
    while queue:
        node = queue.popleft()
        for neighbor in adjacency[node]:
            if neighbor not in seen:
                seen.add(neighbor)
                queue.append(neighbor)
    if len(seen) != 10:
        return True
    if min(min(x, size - x, y, size - y) for x, y in centers) < radius + 12:
        return True
    if min(math.dist(centers[i], centers[j]) for i in range(10) for j in range(i + 1, 10)) < 42:
        return True
    if min(math.dist(centers[left], centers[right]) for left, right in edges) < 2 * radius + stroke + 14:
        return True
    edge_set = [(left, right) for left, right in edges]
    for index, (left, right) in enumerate(edge_set):
        for other_left, other_right in edge_set[index + 1:]:
            if {left, right} & {other_left, other_right}:
                continue
            if _segments_cross(centers[left], centers[right], centers[other_left], centers[other_right]):
                return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["edges"] = [list(edge) for edge in reversed(scene["edges"])]
    return [("reverse_undirected_link_storage", twin)]
