"""Deterministic point-constellation world and analytic arm."""

from __future__ import annotations

import copy
import math
import random
from collections import deque

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
DOT_RADIUS = 8

PALETTES = (
    {"background": [248, 247, 243], "black": [30, 34, 39], "red": [218, 55, 68], "blue": [42, 105, 201], "gray": [132, 136, 140]},
    {"background": [245, 248, 246], "black": [45, 38, 52], "red": [194, 48, 80], "blue": [36, 126, 166], "gray": [139, 135, 143]},
    {"background": [250, 246, 242], "black": [25, 39, 42], "red": [230, 72, 47], "blue": [65, 88, 202], "gray": [128, 133, 136]},
)


def _graph(scene: dict) -> list[list[int]]:
    points = scene["points"]
    threshold = float(scene["scale_length"])
    graph = [[] for _ in points]
    for i in range(len(points)):
        for j in range(i + 1, len(points)):
            d = math.hypot(points[i]["x"] - points[j]["x"], points[i]["y"] - points[j]["y"])
            if d < threshold:
                graph[i].append(j)
                graph[j].append(i)
    return graph


def _route_stats(scene: dict) -> tuple[int | None, int]:
    points = scene["points"]
    red = next(i for i, p in enumerate(points) if p["color"] == "red")
    blue = next(i for i, p in enumerate(points) if p["color"] == "blue")
    graph = _graph(scene)
    dist = [-1] * len(points)
    ways = [0] * len(points)
    dist[red] = 0
    ways[red] = 1
    queue = deque([red])
    while queue:
        u = queue.popleft()
        for v in graph[u]:
            if dist[v] < 0:
                dist[v] = dist[u] + 1
                ways[v] = ways[u]
                queue.append(v)
            elif dist[v] == dist[u] + 1:
                ways[v] = min(2, ways[v] + ways[u])
    return (None if dist[blue] < 0 else dist[blue], ways[blue])


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target_links = 5 if seed % 2 else 6
    n_points = 16
    site_step = 2.0 * math.pi / 18.0

    # Sixteen consecutive sites from an eighteen-site oval make an induced
    # open chain: neighboring sites are below the bar threshold, all other
    # pairs are well above it, including the three-step terminal gap.
    for _ in range(100):
        cx = 256.0 + rng.uniform(-15.0, 15.0)
        cy = 244.0 + rng.uniform(-10.0, 10.0)
        axis_x = rng.uniform(142.0, 156.0)
        axis_y = rng.uniform(138.0, 154.0)
        phase = rng.uniform(0.0, 2.0 * math.pi)
        rotation = rng.uniform(0.0, 2.0 * math.pi)
        direction = rng.choice((-1.0, 1.0))
        scale_length = rng.randint(62, 67)
        cr, sr = math.cos(rotation), math.sin(rotation)
        raw = []
        for k in range(n_points):
            theta = phase + direction * k * site_step + rng.uniform(-0.012, 0.012)
            radial = rng.uniform(-2.0, 2.0)
            lx = (axis_x + radial) * math.cos(theta)
            ly = (axis_y + radial) * math.sin(theta)
            x = cx + cr * lx - sr * ly
            y = cy + sr * lx + cr * ly
            raw.append({"x": round(x, 3), "y": round(y, 3), "color": "black"})

        endpoint_a = rng.randrange(0, n_points - target_links)
        endpoint_b = endpoint_a + target_links
        if rng.randrange(2):
            endpoint_a, endpoint_b = endpoint_b, endpoint_a
        raw[endpoint_a]["color"] = "red"
        raw[endpoint_b]["color"] = "blue"
        rng.shuffle(raw)
        scene = {
            "points": raw,
            "scale_length": scale_length,
            "scale_bar": {"x": rng.randint(30, 410 - scale_length), "y": rng.randint(466, 480)},
            "palette": copy.deepcopy(PALETTES[rng.randrange(len(PALETTES))]),
        }
        route, ways = _route_stats(scene)
        if route == target_links and ways == 1 and margin(scene) >= 9.0:
            return scene
    raise RuntimeError("could not sample a well-separated route scene")


def render(scene: dict) -> Image.Image:
    palette = scene["palette"]
    image = Image.new("RGB", (WIDTH, HEIGHT), tuple(palette["background"]))
    draw = ImageDraw.Draw(image)
    radius = DOT_RADIUS
    for point in scene["points"]:
        x, y = float(point["x"]), float(point["y"])
        color = tuple(palette[point["color"]])
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)

    bar = scene["scale_bar"]
    length = int(scene["scale_length"])
    x, y = int(bar["x"]), int(bar["y"])
    gray = tuple(palette["gray"])
    draw.line((x, y, x + length, y), fill=gray, width=3)
    draw.line((x, y - 5, x, y + 5), fill=gray, width=3)
    draw.line((x + length, y - 5, x + length, y + 5), fill=gray, width=3)
    return image


def analytic_gold(scene: dict) -> str:
    # Directly depends only on the rendered point centers and bar length.
    route, ways = _route_stats(scene)
    if route is None or ways != 1:
        raise ValueError("scene does not have one unique colored route")
    return "odd" if route % 2 else "even"


def margin(scene: dict) -> float:
    points = scene["points"]
    threshold = float(scene["scale_length"])
    clearances = []
    for i in range(len(points)):
        for j in range(i + 1, len(points)):
            d = math.hypot(points[i]["x"] - points[j]["x"], points[i]["y"] - points[j]["y"])
            clearances.append(abs(d - threshold))
    return round(min(clearances), 6)


def is_quarantined(scene: dict) -> bool:
    route, ways = _route_stats(scene)
    return margin(scene) < 8.0 or route is None or ways != 1


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["points"] = list(reversed(twin["points"]))
    return [("reverse_unordered_point_serialization", twin)]
