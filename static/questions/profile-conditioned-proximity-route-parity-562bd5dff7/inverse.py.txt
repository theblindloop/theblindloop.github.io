"""Independent image-only inverse arm for the proximity route task."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int = 20) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        xs, ys = [], []
        while stack:
            yy, xx = stack.pop()
            xs.append(xx)
            ys.append(yy)
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(xs) >= minimum:
            found.append((float(np.mean(xs)), float(np.mean(ys)), len(xs)))
    return found


def _one(mask: np.ndarray, name: str) -> tuple[float, float]:
    items = _components(mask)
    if len(items) != 1:
        raise ValueError(f"expected one {name} dot, found {len(items)}")
    return items[0][0], items[0][1]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    r, g, b = arr[..., 0], arr[..., 1], arr[..., 2]
    black_mask = (r < 85) & (g < 85) & (b < 85)
    red_mask = (r > 150) & (r > g + 70) & (r > b + 60) & (g < 130)
    blue_mask = (b > 135) & (b > r + 55) & (g > 60) & (r < 120)
    gray_mask = (
        (np.abs(r - g) < 16)
        & (np.abs(g - b) < 16)
        & (r >= 105)
        & (r <= 170)
        & (np.indices(r.shape)[0] > 430)
    )

    black = [(x, y, "black") for x, y, _ in _components(black_mask)]
    rx, ry = _one(red_mask, "red")
    bx, by = _one(blue_mask, "blue")
    points = black + [(rx, ry, "red"), (bx, by, "blue")]
    if len(points) != 16:
        raise ValueError(f"expected sixteen dots, found {len(points)}")

    gray_pixels = np.argwhere(gray_mask)
    if len(gray_pixels) < 30:
        raise ValueError("visible scale bar is missing")
    threshold = float(gray_pixels[:, 1].max() - gray_pixels[:, 1].min())
    graph = [[] for _ in points]
    for i in range(len(points)):
        for j in range(i + 1, len(points)):
            if math.hypot(points[i][0] - points[j][0], points[i][1] - points[j][1]) < threshold:
                graph[i].append(j)
                graph[j].append(i)

    source, target = len(points) - 2, len(points) - 1
    dist = [-1] * len(points)
    ways = [0] * len(points)
    dist[source], ways[source] = 0, 1
    queue = deque([source])
    while queue:
        u = queue.popleft()
        for v in graph[u]:
            if dist[v] < 0:
                dist[v], ways[v] = dist[u] + 1, ways[u]
                queue.append(v)
            elif dist[v] == dist[u] + 1:
                ways[v] = min(2, ways[v] + ways[u])
    if dist[target] < 0 or ways[target] != 1:
        raise ValueError("colored dots do not have one unique route")
    return "odd" if dist[target] % 2 else "even"
