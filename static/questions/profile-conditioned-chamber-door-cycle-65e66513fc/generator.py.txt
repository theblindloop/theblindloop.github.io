"""Deterministic room-map renderer and independent analytic topology arm."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

SIZE = 512
DOOR_COUNT = 12
PALETTES = (
    {"background": [249, 248, 244], "wall": [48, 56, 66], "door": [232, 126, 30]},
    {"background": [247, 250, 246], "wall": [60, 62, 72], "door": [222, 105, 34]},
    {"background": [250, 247, 243], "wall": [42, 52, 62], "door": [238, 144, 38]},
    {"background": [246, 248, 250], "wall": [54, 58, 68], "door": [226, 116, 25]},
)


def _parse_edge(edge: str) -> tuple[str, int, int]:
    kind, first, second = edge.split(":")
    return kind, int(first), int(second)


def _all_edges(rows: int, cols: int) -> tuple[list[str], list[str]]:
    outer: list[str] = []
    internal: list[str] = []
    for r in range(rows + 1):
        for c in range(cols):
            edge = f"h:{r}:{c}"
            (outer if r in {0, rows} else internal).append(edge)
    for r in range(rows):
        for c in range(cols + 1):
            edge = f"v:{r}:{c}"
            (outer if c in {0, cols} else internal).append(edge)
    return outer, internal


def _incident_pair(edge: str, rows: int, cols: int) -> tuple[tuple[int, int], tuple[int, int]]:
    kind, r, c = _parse_edge(edge)
    if kind == "h" and 0 < r < rows:
        return (r - 1, c), (r, c)
    if kind == "v" and 0 < c < cols:
        return (r, c - 1), (r, c)
    raise ValueError(f"door is not an internal edge: {edge}")


def _door_pairs(scene: dict) -> list[tuple[tuple[int, int], tuple[int, int]]]:
    rows = int(scene["rows"])
    cols = int(scene["cols"])
    return [_incident_pair(str(edge), rows, cols) for edge in scene["doors"]]


def _cycle_rank(rows: int, cols: int, doors: list[str]) -> int:
    parent = {(r, c): (r, c) for r in range(rows) for c in range(cols)}

    def find(item: tuple[int, int]) -> tuple[int, int]:
        while parent[item] != item:
            parent[item] = parent[parent[item]]
            item = parent[item]
        return item

    components = rows * cols
    rank = 0
    for edge in doors:
        left, right = _incident_pair(edge, rows, cols)
        a, b = find(left), find(right)
        if a == b:
            rank += 1
        else:
            parent[b] = a
            components -= 1
    return rank


def _has_cycle(pairs: list[tuple[tuple[int, int], tuple[int, int]]]) -> bool:
    parent: dict[tuple[int, int], tuple[int, int]] = {}

    def find(item: tuple[int, int]) -> tuple[int, int]:
        parent.setdefault(item, item)
        while parent[item] != item:
            parent[item] = parent[parent[item]]
            item = parent[item]
        return item

    for left, right in pairs:
        a, b = find(left), find(right)
        if a == b:
            return True
        parent[b] = a
    return False


def _jittered_axis(rng: random.Random, count: int) -> list[int]:
    start = rng.randint(54, 66)
    end = rng.randint(446, 458)
    span = end - start
    values = [start]
    for index in range(1, count):
        nominal = start + round(span * index / count)
        values.append(nominal + rng.randint(-6, 6))
    values.append(end)
    return values


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    rows = cols = 4
    want_cycle = int(seed) % 2 == 0
    _outer, internal = _all_edges(rows, cols)
    for _attempt in range(5000):
        doors = sorted(rng.sample(internal, DOOR_COUNT))
        rank = _cycle_rank(rows, cols, doors)
        if (rank == 1) != want_cycle or rank > 1:
            continue
        xs = _jittered_axis(rng, cols)
        ys = _jittered_axis(rng, rows)
        scene = {
            "rows": rows,
            "cols": cols,
            "xs": xs,
            "ys": ys,
            "doors": doors,
            "wall_width": rng.choice([14, 16]),
            "door_length": rng.choice([38, 42, 44]),
            "palette": dict(PALETTES[rng.randrange(len(PALETTES))]),
        }
        if not is_quarantined(scene):
            return scene
    raise RuntimeError(f"could not sample requested cycle class for seed {seed}")


def _edge_box(edge: str, xs: list[int], ys: list[int], width: int) -> tuple[int, int, int, int]:
    kind, r, c = _parse_edge(edge)
    half = width // 2
    if kind == "h":
        return xs[c] - half, ys[r] - half, xs[c + 1] + half, ys[r] + half
    return xs[c] - half, ys[r] - half, xs[c] + half, ys[r + 1] + half


def _door_box(edge: str, xs: list[int], ys: list[int], width: int, length: int) -> tuple[int, int, int, int]:
    kind, r, c = _parse_edge(edge)
    half_width = width // 2
    half_length = length // 2
    if kind == "h":
        center = (xs[c] + xs[c + 1]) // 2
        return center - half_length, ys[r] - half_width, center + half_length, ys[r] + half_width
    center = (ys[r] + ys[r + 1]) // 2
    return xs[c] - half_width, center - half_length, xs[c] + half_width, center + half_length


def render(scene: dict) -> Image.Image:
    rows = int(scene["rows"])
    cols = int(scene["cols"])
    xs = [int(value) for value in scene["xs"]]
    ys = [int(value) for value in scene["ys"]]
    doors = [str(value) for value in scene["doors"]]
    wall_width = int(scene["wall_width"])
    door_length = int(scene["door_length"])
    palette = scene["palette"]
    if len(xs) != cols + 1 or len(ys) != rows + 1:
        raise ValueError("axis lengths disagree with grid dimensions")
    outer, internal = _all_edges(rows, cols)
    image = Image.new("RGB", (SIZE, SIZE), tuple(palette["background"]))
    draw = ImageDraw.Draw(image)
    for edge in outer + internal:
        draw.rectangle(_edge_box(edge, xs, ys, wall_width), fill=tuple(palette["wall"]))
    for edge in doors:
        draw.rectangle(
            _door_box(edge, xs, ys, wall_width, door_length), fill=tuple(palette["door"])
        )
    return image


def analytic_gold(scene: dict) -> str:
    return "yes" if _has_cycle(_door_pairs(scene)) else "no"


def margin(scene: dict) -> float:
    xs = [int(value) for value in scene["xs"]]
    ys = [int(value) for value in scene["ys"]]
    wall_width = float(scene["wall_width"])
    door_length = float(scene["door_length"])
    shortest = min(
        min(right - left for left, right in zip(xs, xs[1:])),
        min(right - left for left, right in zip(ys, ys[1:])),
    )
    endpoint_clearance = (shortest - door_length) / 2.0 - wall_width / 2.0
    return float(min(wall_width / 2.0, endpoint_clearance))


def is_quarantined(scene: dict) -> bool:
    return len(scene["doors"]) != DOOR_COUNT or margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
