"""Independent image-only inverse arm for cycle detection in a room-door dual graph."""

from __future__ import annotations

from collections import deque
from functools import lru_cache

import numpy as np
from PIL import Image

SCALE = 4
EXPECTED_ROOMS = 16
EXPECTED_DOORS = 12


def _component_labels(mask: np.ndarray) -> tuple[np.ndarray, set[int]]:
    height, width = mask.shape
    labels = np.full((height, width), -1, dtype=np.int32)
    exterior: set[int] = set()
    next_label = 0
    for y in range(height):
        for x in range(width):
            if not mask[y, x] or labels[y, x] >= 0:
                continue
            queue = deque([(y, x)])
            labels[y, x] = next_label
            touches_border = False
            while queue:
                cy, cx = queue.popleft()
                if cy in {0, height - 1} or cx in {0, width - 1}:
                    touches_border = True
                for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and labels[ny, nx] < 0:
                        labels[ny, nx] = next_label
                        queue.append((ny, nx))
            if touches_border:
                exterior.add(next_label)
            next_label += 1
    return labels, exterior


def _colored_boxes(mask: np.ndarray) -> list[tuple[int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    boxes: list[tuple[int, int, int, int]] = []
    for y in range(height):
        for x in range(width):
            if not mask[y, x] or seen[y, x]:
                continue
            queue = deque([(y, x)])
            seen[y, x] = True
            xs: list[int] = []
            ys: list[int] = []
            while queue:
                cy, cx = queue.popleft()
                xs.append(cx)
                ys.append(cy)
                for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
            if len(xs) < 300:
                raise ValueError("abstain: a doorway component is too small or fragmented")
            box = min(xs), min(ys), max(xs), max(ys)
            long_side = max(box[2] - box[0], box[3] - box[1])
            short_side = min(box[2] - box[0], box[3] - box[1])
            if long_side < 34 or short_side < 11 or long_side < 2 * short_side:
                raise ValueError("abstain: doorway geometry is ambiguous")
            boxes.append(box)
    if len(boxes) != EXPECTED_DOORS:
        raise ValueError(f"abstain: expected {EXPECTED_DOORS} doorway components, found {len(boxes)}")
    return boxes


def _side_labels(
    box: tuple[int, int, int, int], labels: np.ndarray, exterior: set[int]
) -> tuple[int, int]:
    x0, y0, x1, y1 = box
    height, width = labels.shape
    side_sets: list[set[int]] = []
    if x1 - x0 > y1 - y0:
        tangents = range(x0 + 8, x1 - 7, 4)
        normal_groups = (
            [y0 - offset for offset in range(3, 22, 3)],
            [y1 + offset for offset in range(3, 22, 3)],
        )
        sample_groups = [((normal // SCALE, tangent // SCALE) for normal in normals for tangent in tangents) for normals in normal_groups]
    else:
        tangents = range(y0 + 8, y1 - 7, 4)
        normal_groups = (
            [x0 - offset for offset in range(3, 22, 3)],
            [x1 + offset for offset in range(3, 22, 3)],
        )
        sample_groups = [((tangent // SCALE, normal // SCALE) for normal in normals for tangent in tangents) for normals in normal_groups]
    for samples in sample_groups:
        found: set[int] = set()
        for sy, sx in samples:
            if 0 <= sy < height and 0 <= sx < width:
                label = int(labels[sy, sx])
                if label >= 0 and label not in exterior:
                    found.add(label)
        side_sets.append(found)
    if any(len(found) != 1 for found in side_sets) or side_sets[0] == side_sets[1]:
        raise ValueError(f"abstain: doorway does not resolve to exactly two distinct room sides: {side_sets}")
    return next(iter(side_sets[0])), next(iter(side_sets[1]))


def _contains_cycle(edges: list[tuple[int, int]]) -> bool:
    parent: dict[int, int] = {}

    def find(item: int) -> int:
        parent.setdefault(item, item)
        while parent[item] != item:
            parent[item] = parent[parent[item]]
            item = parent[item]
        return item

    for left, right in edges:
        a, b = find(left), find(right)
        if a == b:
            return True
        parent[b] = a
    return False


@lru_cache(maxsize=128)
def _decision_from_bytes(width: int, height: int, payload: bytes) -> str:
    if width < 128 or height < 128 or width % SCALE or height % SCALE:
        raise ValueError("abstain: unsupported raster dimensions")
    array = np.frombuffer(payload, dtype=np.uint8).reshape((height, width, 3))
    red = array[:, :, 0].astype(np.int16)
    green = array[:, :, 1].astype(np.int16)
    blue = array[:, :, 2].astype(np.int16)
    orange = (red > 180) & (red > green + 60) & (green > blue + 40)
    boxes = _colored_boxes(orange)
    white = (red > 225) & (green > 225) & (blue > 225)
    small_white = white.reshape(height // SCALE, SCALE, width // SCALE, SCALE).mean(axis=(1, 3)) > 0.82
    labels, exterior = _component_labels(small_white)
    all_labels = set(int(value) for value in np.unique(labels) if value >= 0)
    enclosed = all_labels - exterior
    if len(enclosed) != EXPECTED_ROOMS:
        raise ValueError(f"abstain: expected {EXPECTED_ROOMS} enclosed rooms, found {len(enclosed)}")
    edges = [_side_labels(box, labels, exterior) for box in boxes]
    if len({tuple(sorted(edge)) for edge in edges}) != EXPECTED_DOORS:
        raise ValueError("abstain: duplicate doorway incidence is ambiguous")
    return "yes" if _contains_cycle(edges) else "no"


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    return _decision_from_bytes(rgb.width, rgb.height, rgb.tobytes())
