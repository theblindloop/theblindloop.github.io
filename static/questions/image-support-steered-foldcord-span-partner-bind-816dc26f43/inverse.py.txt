"""Pixel-only inverse arm for the fold-cord span partner world.

Input: the final PNG only.  The public question contract fixes the palette
names, states that the cord is one thin two-coloured route whose teal stretch
and tan stretch meet at the single large black CLAMP, that beads are the large
coloured discs sitting on the cord, that exactly one bead on the teal stretch
wears a grey collar, and that small squares lying off the cord are decoration.
Those contract facts are hardcoded here.  No renderer code, latent scene,
coordinate table, filename or metadata is read.

Procedure: segment inks; locate the clamp as the one large black blob; rebuild
each arm from its one-pixel cord core; measure travel distance along each core
by a breadth-first search rooted at the clamp; locate the coloured bead discs
and attach each to the arm whose core runs through it; read the collared bead's
travel distance; report the colour of the tan-arm bead whose travel distance
matches it.  Anything unreadable raises.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

CORE_NEAR = (16, 70, 80)
CORE_FAR = (110, 58, 18)
CLAMP_INK = (18, 18, 18)
COLLAR_INK = (104, 104, 110)
PALETTE = {
    "red": (198, 48, 44),
    "orange": (226, 132, 32),
    "yellow": (218, 190, 40),
    "green": (52, 146, 72),
    "blue": (46, 94, 200),
    "purple": (134, 60, 178),
}

MIN_CLAMP_AREA = 70        # decorative black squares are far smaller
MIN_BEAD_AREA = 120        # decorative coloured squares are far smaller
MIN_CORE_AREA = 300        # decorative core-coloured squares are far smaller
MIN_COLLAR_PIXELS = 120
MATCH_TOLERANCE = 12       # a true fold partner matches to the pixel
TIE_TOLERANCE = 12         # two rivals this close are not separable


class Abstain(ValueError):
    """Raised when the picture does not determine an answer."""


def _mask(arr: np.ndarray, rgb) -> np.ndarray:
    return np.all(arr == np.asarray(rgb, dtype=np.uint8), axis=-1)


def _dilate(mask: np.ndarray, rounds: int) -> np.ndarray:
    out = mask.copy()
    for _ in range(rounds):
        grown = out.copy()
        grown[1:, :] |= out[:-1, :]
        grown[:-1, :] |= out[1:, :]
        grown[:, 1:] |= out[:, :-1]
        grown[:, :-1] |= out[:, 1:]
        out = grown
    return out


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Connected components as boolean arrays, four-connected."""
    seen = np.zeros(mask.shape, dtype=bool)
    out: list[np.ndarray] = []
    ys, xs = np.nonzero(mask)
    for sy, sx in zip(ys.tolist(), xs.tolist()):
        if seen[sy, sx]:
            continue
        stack = [(sy, sx)]
        seen[sy, sx] = True
        blob = np.zeros(mask.shape, dtype=bool)
        while stack:
            y, x = stack.pop()
            blob[y, x] = True
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < mask.shape[0] and 0 <= nx < mask.shape[1] \
                        and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        out.append(blob)
    return out


def _centroid(blob: np.ndarray) -> tuple[int, int]:
    ys, xs = np.nonzero(blob)
    return int(round(float(ys.mean()))), int(round(float(xs.mean())))


def _largest(mask: np.ndarray, min_area: int, what: str) -> np.ndarray:
    blobs = [b for b in _components(mask) if b.sum() >= min_area]
    if not blobs:
        raise Abstain(f"no {what} found")
    blobs.sort(key=lambda b: -int(b.sum()))
    return blobs[0]


def _geodesic(core: np.ndarray, start: tuple[int, int]) -> np.ndarray:
    """Travel distance in pixels along a one-pixel-wide rectilinear core."""
    dist = np.full(core.shape, -1, dtype=np.int32)
    sy, sx = start
    dist[sy, sx] = 0
    queue = deque([(sy, sx)])
    height, width = core.shape
    while queue:
        y, x = queue.popleft()
        d = dist[y, x] + 1
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < height and 0 <= nx < width and core[ny, nx] and dist[ny, nx] < 0:
                dist[ny, nx] = d
                queue.append((ny, nx))
    return dist


def _root(core: np.ndarray, clamp_yx: tuple[int, int]) -> tuple[int, int]:
    ys, xs = np.nonzero(core)
    d2 = (ys - clamp_yx[0]) ** 2 + (xs - clamp_yx[1]) ** 2
    which = int(np.argmin(d2))
    if d2[which] > 24 ** 2:
        raise Abstain("a cord arm does not reach the clamp")
    return int(ys[which]), int(xs[which])


def _sample_distance(dist: np.ndarray, yx: tuple[int, int]) -> int | None:
    best = None
    cy, cx = yx
    for dy in range(-3, 4):
        for dx in range(-3, 4):
            y, x = cy + dy, cx + dx
            if 0 <= y < dist.shape[0] and 0 <= x < dist.shape[1] and dist[y, x] >= 0:
                value = int(dist[y, x])
                if best is None or value < best:
                    best = value
    return best


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)

    clamps = [b for b in _components(_mask(arr, CLAMP_INK)) if b.sum() >= MIN_CLAMP_AREA]
    if len(clamps) != 1:
        raise Abstain(f"expected exactly one clamp, found {len(clamps)}")
    clamp_yx = _centroid(clamps[0])

    near_core = _largest(_mask(arr, CORE_NEAR), MIN_CORE_AREA, "teal cord core")
    far_core = _largest(_mask(arr, CORE_FAR), MIN_CORE_AREA, "tan cord core")
    near_dist = _geodesic(near_core, _root(near_core, clamp_yx))
    far_dist = _geodesic(far_core, _root(far_core, clamp_yx))

    collar = _mask(arr, COLLAR_INK)
    near_beads: list[tuple[int, str, tuple[int, int], bool]] = []
    far_beads: list[tuple[int, str, tuple[int, int], bool]] = []
    for name, rgb in PALETTE.items():
        raw = _mask(arr, rgb)
        for blob in _components(_dilate(raw, 2)):
            body = blob & raw
            if body.sum() < MIN_BEAD_AREA:
                continue
            yx = _centroid(body)
            collared = bool(collar[max(0, yx[0] - 15):yx[0] + 16,
                                   max(0, yx[1] - 15):yx[1] + 16].sum() >= MIN_COLLAR_PIXELS)
            near_d = _sample_distance(near_dist, yx)
            far_d = _sample_distance(far_dist, yx)
            if near_d is not None and far_d is None:
                near_beads.append((near_d, name, yx, collared))
            elif far_d is not None and near_d is None:
                far_beads.append((far_d, name, yx, collared))
            elif near_d is not None and far_d is not None:
                raise Abstain("a bead sits on both cord arms")

    collared_near = [bead for bead in near_beads if bead[3]]
    if len(collared_near) != 1:
        raise Abstain(f"expected one collared teal-arm bead, found {len(collared_near)}")
    if len(far_beads) < 2:
        raise Abstain("fewer than two beads on the tan arm")

    query_d = collared_near[0][0]
    scored = sorted((abs(bead[0] - query_d), bead[1]) for bead in far_beads)
    if scored[0][0] > MATCH_TOLERANCE:
        raise Abstain("no tan-arm bead matches the collared bead's travel distance")
    if scored[1][0] - scored[0][0] < TIE_TOLERANCE:
        raise Abstain("two tan-arm beads are equally good fold partners")
    return scored[0][1]
