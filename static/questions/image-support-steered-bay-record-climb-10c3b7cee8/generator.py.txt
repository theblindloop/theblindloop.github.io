"""Bay-record climb: one thick unbranched corridor whose bays are read in walk order.

The corridor is a single connected shape: six long horizontal bays, each a solid
rectangle of its own thickness, stitched into one unbranched chain by narrow
vertical connectors.  A hollow green frame parked beyond one free end of one end
bay declares where the walk starts.  The decision counts how many bays after the
start bay are thicker than every bay met earlier on that walk, so the answer is
an order-dependent aggregate over the whole corridor body.

Every drawn element is an axis-aligned rectangle, so the raster is
byte-deterministic and free of antialiasing.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 720
HEIGHT = 470

BG = (252, 252, 250)
CORRIDOR = (46, 52, 66)
DECOY = (178, 182, 190)
RING = (28, 152, 74)

THICKNESSES = (14, 24, 34, 44, 54, 64)
N_BAYS = 6

LANES = tuple(100 + 90 * i for i in range(7))
LINK_HALF = 8
BAY_OVERHANG = 22
RING_GAP = 16
RING_WIDTH = 24
RING_STROKE = 4
DECOY_CLEAR = 14

QUARANTINE_MARGIN = 8.0


def _band_of_lane(bay: dict) -> tuple[int, int]:
    return (bay["a"], bay["b"])


def _order_family(rng: random.Random) -> list[int]:
    """Identity band order perturbed by one or two disjoint adjacent swaps."""

    order = list(range(N_BAYS))
    positions = list(range(N_BAYS - 1))
    rng.shuffle(positions)
    picked: list[int] = []
    for pos in positions:
        if all(abs(pos - other) >= 2 for other in picked):
            picked.append(pos)
        if len(picked) == rng.choice([1, 1, 2]):
            break
    for pos in picked:
        order[pos], order[pos + 1] = order[pos + 1], order[pos]
    return order


def _records(sequence: list[int]) -> int:
    """Bays after the first that beat every earlier bay on the walk."""

    best = sequence[0]
    count = 0
    for value in sequence[1:]:
        if value > best:
            count += 1
            best = value
    return count


def _try_scene(rng: random.Random, target: int) -> dict | None:
    gaps = [rng.choice([10, 12, 14, 16, 18, 20, 22, 24, 26]) for _ in range(N_BAYS - 1)]
    if rng.random() < 0.12:
        gaps[rng.randrange(N_BAYS - 1)] = rng.choice([6, 7])

    thick = list(THICKNESSES)
    rng.shuffle(thick)

    top = 20 + rng.randrange(0, 13)
    bays: list[dict] = []
    y = top
    for index in range(N_BAYS):
        a = rng.randrange(0, len(LANES) - 3)
        span = rng.choice([3, 3, 4, 5])
        b = min(a + span, len(LANES) - 1)
        bays.append(
            {
                "y0": y,
                "y1": y + thick[index] - 1,
                "t": thick[index],
                "a": a,
                "b": b,
                "x0": LANES[a] - BAY_OVERHANG,
                "x1": LANES[b] + BAY_OVERHANG,
            }
        )
        y += thick[index]
        if index < N_BAYS - 1:
            y += gaps[index]
    if y > HEIGHT - 100:
        return None

    order = _order_family(rng)
    start_at_first = bool(rng.getrandbits(1))
    walk = order if start_at_first else list(reversed(order))
    if _records([bays[i]["t"] for i in walk]) != target:
        return None

    link_lanes: list[int] = []
    used_lanes: set[int] = set()
    for step in range(N_BAYS - 1):
        u, v = order[step], order[step + 1]
        lo, hi = min(u, v), max(u, v)
        crossed = range(lo + 1, hi)
        low = max(bays[u]["a"], bays[v]["a"])
        high = min(bays[u]["b"], bays[v]["b"])
        options = []
        for lane in range(low, high + 1):
            if lane in used_lanes:
                continue
            if any(bays[w]["a"] <= lane <= bays[w]["b"] for w in crossed):
                continue
            options.append(lane)
        if not options:
            return None
        lane = rng.choice(options)
        used_lanes.add(lane)
        link_lanes.append(lane)

    degree = [0] * N_BAYS
    for step in range(N_BAYS - 1):
        degree[order[step]] += 1
        degree[order[step + 1]] += 1
    if sorted(degree) != [1, 1, 2, 2, 2, 2]:
        return None

    draft = {
        "width": WIDTH,
        "height": HEIGHT,
        "bays": bays,
        "order": order,
        "start_at_first": start_at_first,
        "link_lanes": link_lanes,
    }
    ring = _ring_rect(draft)
    if ring is None:
        return None

    solid = [(b["x0"], b["y0"], b["x1"], b["y1"]) for b in bays]
    solid.extend(_link_rects(draft))
    solid.append(tuple(ring))

    strip_top = y + 22
    decoys: list[list[int]] = []
    for _ in range(rng.choice([3, 4, 5, 6, 7])):
        for _attempt in range(30):
            size = rng.choice([14, 16, 18, 20])
            dx = rng.randrange(20, WIDTH - 20 - size)
            dy = rng.randrange(strip_top, max(strip_top + 1, HEIGHT - 18 - size))
            box = [dx, dy, dx + size - 1, dy + size - 1]
            if all(_gap(box, other) >= DECOY_CLEAR for other in solid + decoys):
                decoys.append(box)
                break

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "bays": bays,
        "order": order,
        "start_at_first": start_at_first,
        "link_lanes": link_lanes,
        "decoys": decoys,
        "gaps": gaps,
        "top": top,
    }


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 17)
    target = 1 + (seed % 3)
    for _ in range(4000):
        scene = _try_scene(rng, target)
        if scene is not None:
            return scene
    raise RuntimeError(f"no valid scene for seed {seed}")


def _link_rects(scene: dict) -> list[tuple[int, int, int, int]]:
    """Connector rectangles, derived from the chain order and its lanes."""

    bays = scene["bays"]
    order = scene["order"]
    rects = []
    for step, lane in enumerate(scene["link_lanes"]):
        cx = LANES[lane]
        top_bay, bottom_bay = sorted((order[step], order[step + 1]))
        rects.append(
            (
                cx - LINK_HALF,
                (bays[top_bay]["y0"] + bays[top_bay]["y1"]) // 2,
                cx + LINK_HALF,
                (bays[bottom_bay]["y0"] + bays[bottom_bay]["y1"]) // 2,
            )
        )
    return rects


def _ring_rect(scene: dict) -> list[int] | None:
    """The hollow frame, parked beyond the free end of the declared start bay."""

    order = scene["order"]
    lanes = scene["link_lanes"]
    start = order[0] if scene["start_at_first"] else order[-1]
    lane = lanes[0] if scene["start_at_first"] else lanes[-1]
    bay = scene["bays"][start]
    lane_x = LANES[lane]
    if abs(lane_x - bay["x0"]) < abs(lane_x - bay["x1"]):
        x0 = bay["x1"] + RING_GAP
    else:
        x0 = bay["x0"] - RING_GAP - RING_WIDTH
    if x0 < 8 or x0 + RING_WIDTH > scene["width"] - 9:
        return None
    return [x0, bay["y0"], x0 + RING_WIDTH - 1, bay["y1"]]


def _walk(scene: dict) -> list[int]:
    order = list(scene["order"])
    return order if scene["start_at_first"] else list(reversed(order))


def _elements(scene: dict) -> list[dict]:
    items: list[dict] = []
    bays = scene["bays"]
    for index, bay in enumerate(bays):
        items.append(
            {
                "kind": "bay",
                "color": CORRIDOR,
                "index": index,
                "rect": (bay["x0"], bay["y0"], bay["x1"], bay["y1"]),
            }
        )
    for rect in _link_rects(scene):
        items.append({"kind": "link", "color": CORRIDOR, "index": None, "rect": rect})
    for box in scene["decoys"]:
        items.append({"kind": "decoy", "color": DECOY, "index": None, "rect": tuple(box)})
    return items


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (scene["width"], scene["height"]), BG)
    draw = ImageDraw.Draw(image)
    for item in _elements(scene):
        draw.rectangle(item["rect"], fill=item["color"])
    x0, y0, x1, y1 = _ring_rect(scene)
    stroke = min(RING_STROKE, max(1, (y1 - y0) // 3))
    draw.rectangle((x0, y0, x1, y0 + stroke - 1), fill=RING)
    draw.rectangle((x0, y1 - stroke + 1, x1, y1), fill=RING)
    draw.rectangle((x0, y0, x0 + stroke - 1, y1), fill=RING)
    draw.rectangle((x1 - stroke + 1, y0, x1, y1), fill=RING)
    return image


def analytic_gold(scene: dict) -> str:
    """Count the walk-order thickness records after the start bay."""

    bays = scene["bays"]
    order = list(scene["order"])
    if not scene["start_at_first"]:
        order = list(reversed(order))
    return str(_records([bays[index]["t"] for index in order]))


def _gap(a, b) -> float:
    dx = max(a[0] - b[2], b[0] - a[2], 0)
    dy = max(a[1] - b[3], b[1] - a[3], 0)
    if dx == 0 and dy == 0:
        return 0.0
    if dx and dy:
        return float((dx * dx + dy * dy) ** 0.5) - 1.0
    return float(dx + dy) - 1.0


def _clearances(scene: dict) -> list[float]:
    bays = scene["bays"]
    values = [float(g) for g in scene["gaps"]]
    values.append(float(RING_GAP))
    boxes = [(b["x0"], b["y0"], b["x1"], b["y1"]) for b in bays]
    order = scene["order"]
    for step, rect in enumerate(_link_rects(scene)):
        joined = (order[step], order[step + 1])
        for index, box in enumerate(boxes):
            if index in joined:
                continue
            values.append(_gap(rect, box))
    for box in scene["decoys"]:
        for other in boxes:
            values.append(_gap(box, other))
        values.append(_gap(box, tuple(_ring_rect(scene))))
    return values


def _thickness_discrimination(scene: dict) -> float:
    """Smallest pixel gap that any walk-order record comparison turns on."""

    sequence = [scene["bays"][index]["t"] for index in _walk(scene)]
    best = sequence[0]
    worst = float(max(THICKNESSES))
    for value in sequence[1:]:
        worst = min(worst, float(abs(value - best)))
        best = max(best, value)
    return worst


def margin(scene: dict) -> float:
    return round(min(min(_clearances(scene)), _thickness_discrimination(scene)), 3)


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_MARGIN)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical latent rewrites that must leave the answer alone."""

    out: list[tuple[str, dict]] = []

    flipped = dict(scene)
    flipped["order"] = list(reversed(scene["order"]))
    flipped["link_lanes"] = list(reversed(scene["link_lanes"]))
    flipped["start_at_first"] = not scene["start_at_first"]
    out.append(("chain_written_backwards_with_start_flag_flipped", flipped))

    rotated = dict(scene)
    decoys = list(scene["decoys"])
    if decoys:
        decoys = decoys[1:] + decoys[:1]
    rotated["decoys"] = decoys
    out.append(("decoy_order_rotated", rotated))

    return out
