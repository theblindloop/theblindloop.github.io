"""Independent PNG-only inverse arm for well-nested checkpoint traces."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

GREEN = (34, 166, 96)
CYAN = (26, 184, 202)
MAGENTA = (207, 56, 135)
TERMINALS = {
    "orange": (238, 135, 35),
    "purple": (135, 82, 190),
    "blue": (51, 105, 210),
}


def _components(mask: np.ndarray, minimum: int = 12) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        xs, ys = [], []
        while queue:
            cy, cx = queue.popleft()
            xs.append(cx)
            ys.append(cy)
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = cy + dy, cx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= minimum:
            found.append((float(sum(xs)) / len(xs), float(sum(ys)) / len(ys), len(xs)))
    return found


def _color_components(array: np.ndarray, rgb: tuple[int, int, int], minimum: int = 12):
    return _components(np.all(array == np.asarray(rgb, dtype=np.uint8), axis=2), minimum)


def _well_nested(tokens: list[tuple[float, str]]) -> bool:
    balance = 0
    for _, token in tokens:
        balance += 1 if token == "O" else -1
        if balance < 0:
            return False
    return balance == 0


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    starts = _color_components(array, GREEN, 40)
    opens = _color_components(array, CYAN, 80)
    closes = _color_components(array, MAGENTA, 80)
    terminals = []
    for name, rgb in TERMINALS.items():
        for x, y, size in _color_components(array, rgb, 100):
            terminals.append((x, y, size, name))

    if len(starts) != 3 or len(terminals) != 3 or len(opens) != 9 or len(closes) != 9:
        raise ValueError("expected three starts, three terminals, and eighteen checkpoints")

    winners = []
    for tx, ty, _, terminal_name in sorted(terminals, key=lambda item: item[1]):
        sx, sy, _ = min(starts, key=lambda item: abs(item[1] - ty))
        if abs(sy - ty) > 55:
            raise ValueError("start and terminal row association is ambiguous")
        tokens = [(x, "O") for x, y, _ in opens if abs(y - ty) < 55]
        tokens += [(x, "C") for x, y, _ in closes if abs(y - ty) < 55]
        if len(tokens) != 6:
            raise ValueError("checkpoint row association is ambiguous")
        tokens.sort(key=lambda item: item[0], reverse=sx > tx)
        if _well_nested(tokens):
            winners.append(terminal_name)
    if len(winners) != 1:
        raise ValueError("image does not contain exactly one well-nested route")
    return winners[0]
