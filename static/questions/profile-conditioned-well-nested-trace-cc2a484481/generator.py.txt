"""Deterministic renderer and analytic arm for well-nested checkpoint traces."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 480
INK = (46, 52, 59)
GREEN = (34, 166, 96)
ORANGE = (238, 135, 35)
PURPLE = (135, 82, 190)
BLUE = (51, 105, 210)
CYAN = (26, 184, 202)
MAGENTA = (207, 56, 135)
TERMINAL_RGB = {"orange": ORANGE, "purple": PURPLE, "blue": BLUE}

VALID = [
    "OOOCCC",
    "OOCOCC",
    "OCOCOC",
    "OCOOCC",
    "OOCCOC",
]
INVALID = [
    "COOCOC",
    "OCCOOC",
    "CCOOOC",
    "OCCOCO",
    "COCOOC",
]


def _defect(sequence: str) -> int:
    balance = 0
    minimum = 0
    for token in sequence:
        balance += 1 if token == "O" else -1
        minimum = min(minimum, balance)
    return abs(balance) + max(0, -minimum)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    winner = int(seed) % 3
    sequences = []
    invalid_pool = INVALID[:]
    rng.shuffle(invalid_pool)
    for index in range(3):
        sequences.append(rng.choice(VALID) if index == winner else invalid_pool.pop())

    terminal_colors = ["orange", "purple", "blue"]
    rng.shuffle(terminal_colors)
    row_order = [0, 1, 2]
    rng.shuffle(row_order)
    starts_left = [bool(rng.getrandbits(1)) for _ in range(3)]
    bend_patterns = []
    patterns = [
        [0, -18, 14, -10, 18, -14, 12, 0],
        [0, 16, -12, 18, -16, 10, -18, 0],
        [0, -12, -20, 10, 20, 12, -10, 0],
        [0, 12, 20, -10, -20, -12, 10, 0],
    ]
    for _ in range(3):
        bend_patterns.append(deepcopy(rng.choice(patterns)))
    return {
        "sequences": sequences,
        "terminal_colors": terminal_colors,
        "starts_left": starts_left,
        "row_order": row_order,
        "bend_patterns": bend_patterns,
    }


def _route_points(row_y: int, starts_left: bool, offsets: list[int]) -> list[tuple[int, int]]:
    xs = [72, 144, 216, 288, 360, 432, 504, 568]
    if not starts_left:
        xs.reverse()
    return [(x, row_y + int(offset)) for x, offset in zip(xs, offsets)]


def render(scene: dict) -> Image.Image:
    sequences = scene["sequences"]
    terminal_colors = scene["terminal_colors"]
    starts_left = scene["starts_left"]
    row_order = scene["row_order"]
    bend_patterns = scene["bend_patterns"]

    image = Image.new("RGB", (WIDTH, HEIGHT), (248, 247, 243))
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((25, 31, 615, 449), radius=18, fill=(255, 255, 255), outline=(215, 218, 220), width=2)
    row_ys = [120, 240, 360]

    for display_row, logical in enumerate(row_order):
        row_y = row_ys[display_row]
        points = _route_points(row_y, bool(starts_left[logical]), bend_patterns[logical])
        draw.line(points, fill=INK, width=7, joint="curve")

        for token, (x, y) in zip(sequences[logical], points[1:7]):
            if token == "O":
                draw.ellipse((x - 10, y - 10, x + 10, y + 10), fill=CYAN, outline=(255, 255, 255), width=2)
            else:
                draw.polygon([(x, y - 12), (x + 12, y), (x, y + 12), (x - 12, y)], fill=MAGENTA)
                draw.line([(x, y - 10), (x + 10, y), (x, y + 10), (x - 10, y), (x, y - 10)], fill=(255, 255, 255), width=2)

        sx, sy = points[0]
        tx, ty = points[-1]
        draw.ellipse((sx - 16, sy - 16, sx + 16, sy + 16), fill=(255, 255, 255), outline=GREEN, width=7)
        color = TERMINAL_RGB[terminal_colors[logical]]
        draw.rounded_rectangle((tx - 17, ty - 17, tx + 17, ty + 17), radius=4, fill=color, outline=(255, 255, 255), width=3)

    return image


def analytic_gold(scene: dict) -> str:
    sequences = scene["sequences"]
    terminal_colors = scene["terminal_colors"]
    winners = [index for index, sequence in enumerate(sequences) if _defect(sequence) == 0]
    if len(winners) != 1:
        raise ValueError("scene does not contain exactly one well-nested sequence")
    return terminal_colors[winners[0]]


def margin(scene: dict) -> float:
    scores = sorted(_defect(sequence) for sequence in scene["sequences"])
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    try:
        sequences = scene["sequences"]
        if len(sequences) != 3 or any(len(s) != 6 or s.count("O") != 3 or s.count("C") != 3 for s in sequences):
            return True
        if len([s for s in sequences if _defect(s) == 0]) != 1:
            return True
        return margin(scene) < 1.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
