"""Independent inverse arm; consumes only pixels from the final PNG."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image


class AmbiguousImage(ValueError):
    pass


def _components(mask: np.ndarray, minimum: int = 20) -> list[np.ndarray]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y, x in zip(*np.nonzero(mask), strict=True):
        if seen[y, x]:
            continue
        q = deque([(int(y), int(x))])
        seen[y, x] = True
        pts = []
        while q:
            yy, xx = q.popleft()
            pts.append((yy, xx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = yy + dy, xx + dx
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    q.append((ny, nx))
        if len(pts) >= minimum:
            result.append(np.asarray(pts, dtype=np.int32))
    return result


def _bay_centers(arr: np.ndarray) -> list[float]:
    pale = ((arr[..., 0] >= 205) & (arr[..., 0] <= 238) &
            (arr[..., 1] >= 207) & (arr[..., 1] <= 240) &
            (arr[..., 2] >= 212) & (arr[..., 2] <= 243))
    items = []
    for pts in _components(pale, 500):
        ys, xs = pts[:, 0], pts[:, 1]
        if xs.max() - xs.min() > 90 and ys.max() - ys.min() > 55:
            items.append(float((xs.min() + xs.max()) / 2))
    items.sort()
    if len(items) != 5 or min(np.diff(items)) < 80:
        raise AmbiguousImage(f"expected five separated bay frames, found {len(items)}")
    return items


def _nearest_bay(x: float, centers: list[float]) -> int:
    distances = [abs(x - center) for center in centers]
    order = np.argsort(distances)
    if distances[int(order[0])] > 35 or distances[int(order[1])] - distances[int(order[0])] < 25:
        raise AmbiguousImage("marker is not clearly bound to one bay")
    return int(order[0])


def _start_state(arr: np.ndarray, bays: list[float]) -> tuple[int, int]:
    blue = (arr[..., 0] < 90) & (arr[..., 1] > 80) & (arr[..., 1] < 165) & (arr[..., 2] > 135)
    # Score each recovered bay locally so the short connector between the two
    # halves cannot be mistaken for a third component.
    occupied = []
    for index, center in enumerate(bays):
        x0, x1 = max(0, int(center - 52)), min(arr.shape[1], int(center + 53))
        if int(np.count_nonzero(blue[:, x0:x1])) > 2500:
            occupied.append(index)
    if len(occupied) != 2 or occupied[1] != occupied[0] + 1:
        raise AmbiguousImage(f"could not recover one adjacent domino span: {occupied}")
    magenta = (arr[..., 0] > 150) & (arr[..., 1] < 95) & (arr[..., 2] > 85) & (arr[..., 2] < 175)
    pts = np.argwhere(magenta)
    if len(pts) < 300:
        raise AmbiguousImage("marked end is missing")
    marked = _nearest_bay(float(pts[:, 1].mean()), bays)
    if marked not in occupied:
        raise AmbiguousImage("mark is outside the domino")
    return occupied[0], marked - occupied[0]


def _moves(arr: np.ndarray) -> list[str]:
    orange = (arr[..., 0] > 180) & (arr[..., 1] > 75) & (arr[..., 1] < 165) & (arr[..., 2] < 85)
    found = []
    for pts in _components(orange, 350):
        ys, xs = pts[:, 0], pts[:, 1]
        width, height = xs.max() - xs.min() + 1, ys.max() - ys.min() + 1
        if width < 55 or height < 25:
            continue
        x0, x1 = int(xs.min()), int(xs.max())
        midpoint = (x0 + x1) / 2
        left = int(np.count_nonzero(xs < midpoint))
        right = int(np.count_nonzero(xs > midpoint))
        gap = abs(left - right) / len(xs)
        if gap < 0.08:
            raise AmbiguousImage("arrow direction has insufficient pixel margin")
        found.append((float(xs.mean()), "L" if left > right else "R"))
    found.sort()
    if len(found) != 5:
        raise AmbiguousImage(f"expected five arrows, found {len(found)}")
    return [move for _, move in found]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.ndim != 3 or min(arr.shape[:2]) < 200:
        raise AmbiguousImage("unsupported raster")
    bays = _bay_centers(arr)
    left, side = _start_state(arr, bays)
    for move in _moves(arr):
        step = -1 if move == "L" else 1
        if 0 <= left + step <= 3:
            left += step
            side = 1 - side
    return str(left + side + 1)
