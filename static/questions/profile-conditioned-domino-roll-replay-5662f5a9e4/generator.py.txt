"""Renderer and analytic replay for bounded end-over-end domino rolls."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 520
BG = (248, 247, 243)
DARK = (39, 44, 52)
MUTED = (112, 120, 131)
BAY = (224, 227, 231)
BLUE = (42, 119, 181)
ORANGE = (226, 126, 34)
MAGENTA = (199, 44, 126)


def _replay(start_left: int, marked_side: int, moves: list[str]) -> tuple[int, int]:
    left, side = int(start_left), int(marked_side)
    for move in moves:
        step = -1 if move == "L" else 1
        if 0 <= left + step <= 3:
            left += step
            side = 1 - side
    return left, side


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0xD041A0)
    target = int(seed) % 5
    for _ in range(1000):
        moves = ["L" if rng.getrandbits(1) else "R" for _ in range(5)]
        starts = []
        for left in range(4):
            for side in range(2):
                end_left, end_side = _replay(left, side, moves)
                if end_left + end_side == target:
                    starts.append((left, side))
        if starts:
            start_left, marked_side = rng.choice(starts)
            return {
                "start_left": start_left,
                "marked_side": marked_side,
                "moves": moves,
                "card_tilts": [round(rng.uniform(-1.8, 1.8), 3) for _ in moves],
            }
    raise RuntimeError("failed to sample balanced scene")


def _font(size: int, bold: bool = False):
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _center(draw, xy, text, font, fill):
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2), text, font=font, fill=fill)


def _arrow_glyph(direction: str, tilt: float) -> Image.Image:
    glyph = Image.new("RGBA", (92, 62), (0, 0, 0, 0))
    d = ImageDraw.Draw(glyph)
    if direction == "R":
        d.rectangle((12, 25, 58, 37), fill=(*ORANGE, 255))
        d.polygon([(48, 9), (82, 31), (48, 53)], fill=(*ORANGE, 255))
    else:
        d.rectangle((34, 25, 80, 37), fill=(*ORANGE, 255))
        d.polygon([(44, 9), (10, 31), (44, 53)], fill=(*ORANGE, 255))
    return glyph.rotate(tilt, resample=Image.Resampling.BICUBIC, expand=False)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGBA", (WIDTH, HEIGHT), (*BG, 255))
    draw = ImageDraw.Draw(image)
    draw.text((28, 18), "DOMINO END-OVER-END REPLAY", font=_font(25, True), fill=DARK)
    draw.text((28, 54), "Successful roll: shift one bay + swap ends.  Edge-blocked roll: no change.", font=_font(16), fill=MUTED)

    centers = [150 + 155 * i for i in range(5)]
    y = 190
    for i, x in enumerate(centers):
        draw.rounded_rectangle((x - 62, y - 48, x + 62, y + 48), radius=12, fill=BAY, outline=DARK, width=4)
        _center(draw, (x, y + 76), str(i + 1), _font(21, True), DARK)

    left = int(scene["start_left"])
    for index in (left, left + 1):
        x = centers[index]
        draw.rounded_rectangle((x - 48, y - 34, x + 48, y + 34), radius=8, fill=BLUE)
    draw.line((centers[left] + 52, y, centers[left + 1] - 52, y), fill=BLUE, width=18)
    marked = centers[left + int(scene["marked_side"])]
    draw.ellipse((marked - 23, y - 23, marked + 23, y + 23), outline=MAGENTA, width=9)
    draw.ellipse((marked - 7, y - 7, marked + 7, y + 7), fill=MAGENTA)
    draw.text((28, 300), "ROLL CARDS  —  READ LEFT TO RIGHT", font=_font(18, True), fill=DARK)

    card_xs = [145 + 165 * i for i in range(5)]
    card_y = 397
    for i, (x, move, tilt) in enumerate(zip(card_xs, scene["moves"], scene["card_tilts"], strict=True)):
        draw.rounded_rectangle((x - 58, card_y - 55, x + 58, card_y + 55), radius=11, fill=(255, 255, 255), outline=DARK, width=3)
        glyph = _arrow_glyph(move, float(tilt))
        image.alpha_composite(glyph, (x - 46, card_y - 31))
        _center(draw, (x, card_y + 72), str(i + 1), _font(15, True), MUTED)
    return image.convert("RGB")


def analytic_gold(scene: dict) -> str:
    left, side = _replay(scene["start_left"], scene["marked_side"], list(scene["moves"]))
    return str(left + side + 1)


def margin(scene: dict) -> float:
    return round(24.0 - max(abs(float(v)) for v in scene["card_tilts"]), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
