"""Deterministic latent sampler and raster renderer for connected intersections."""

from __future__ import annotations

import random
from collections import deque

import numpy as np
from PIL import Image, ImageDraw

GRID = 14
CELL = 32
CANVAS = 512
BOARD_LEFT = 32
FRAME_LEFT = 26

WHITE = (248, 248, 244)
BLACK = (25, 31, 34)
CYAN = (22, 164, 183)
ORANGE = (235, 112, 42)
YELLOW = (247, 204, 55)
COLORS = {1: CYAN, 2: ORANGE, 3: YELLOW}


def _component_sizes(cells: list[list[int]], allowed: set[int]) -> list[int]:
    rows, cols = len(cells), len(cells[0])
    unseen = {(y, x) for y in range(rows) for x in range(cols) if cells[y][x] in allowed}
    sizes: list[int] = []
    while unseen:
        start = unseen.pop()
        queue = deque([start])
        size = 1
        while queue:
            y, x = queue.popleft()
            for point in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if point in unseen:
                    unseen.remove(point)
                    queue.append(point)
                    size += 1
        sizes.append(size)
    return sorted(sizes, reverse=True)


def _transform(cells: list[list[int]], turns: int, reflect: bool) -> list[list[int]]:
    array = np.asarray(cells, dtype=np.uint8)
    array = np.rot90(array, turns)
    if reflect:
        array = np.fliplr(array)
    return array.tolist()


def sample_scene(seed: int) -> dict:
    """Sample a full tiling whose two reconstructed source regions stay connected."""
    rng = random.Random(int(seed))
    split = 5 + rng.randrange(4)
    boundary = []
    for _ in range(GRID):
        split = max(4, min(8, split + rng.choice((-1, 0, 0, 1))))
        boundary.append(split)

    cells = [[1 if x <= boundary[y] else 2 for x in range(GRID)] for y in range(GRID)]
    two_components = bool(int(seed) % 2)
    if two_components:
        starts = (rng.choice((1, 2, 3)), rng.choice((9, 10, 11)))
        bands = ((starts[0], starts[0] + 2), (starts[1], starts[1] + 2))
    else:
        start = rng.choice((4, 5, 6))
        bands = ((start, start + 4),)
    for y0, y1 in bands:
        for y in range(y0, y1):
            cells[y][boundary[y]] = 3
            cells[y][boundary[y] + 1] = 3

    cells = _transform(cells, rng.randrange(4), bool(rng.randrange(2)))
    scene = {"cells": cells}
    if is_quarantined(scene):
        raise RuntimeError(f"sampler produced an invalid scene for seed {seed}")
    return scene


def render(scene: dict) -> Image.Image:
    cells = scene["cells"]
    image = Image.new("RGB", (CANVAS, CANVAS), WHITE)
    draw = ImageDraw.Draw(image)
    board_right = BOARD_LEFT + GRID * CELL
    draw.rectangle(
        (FRAME_LEFT, FRAME_LEFT, CANVAS - FRAME_LEFT - 1, CANVAS - FRAME_LEFT - 1),
        fill=BLACK,
    )
    for y, row in enumerate(cells):
        for x, code in enumerate(row):
            x0 = BOARD_LEFT + x * CELL
            y0 = BOARD_LEFT + y * CELL
            draw.rectangle((x0, y0, x0 + CELL - 1, y0 + CELL - 1), fill=COLORS[int(code)])
    assert board_right == CANVAS - BOARD_LEFT
    return image


def analytic_gold(scene: dict) -> str:
    components = _component_sizes(scene["cells"], {3})
    return "yes" if len(components) == 2 else "no"


def margin(scene: dict) -> float:
    """One-sided discrete robustness: smallest yellow component in grid cells."""
    components = _component_sizes(scene["cells"], {3})
    return float(min(components))


def is_quarantined(scene: dict) -> bool:
    try:
        cells = scene["cells"]
        if len(cells) != GRID or any(len(row) != GRID for row in cells):
            return True
        if any(code not in (1, 2, 3) for row in cells for code in row):
            return True
        cyan_region = _component_sizes(cells, {1, 3})
        orange_region = _component_sizes(cells, {2, 3})
        yellow = _component_sizes(cells, {3})
        if len(cyan_region) != 1 or len(orange_region) != 1:
            return True
        if len(yellow) not in (1, 2) or sum(yellow) != 8 or min(yellow) < 4:
            return True
        return False
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
