"""Independent pixel-only inverse arm.

The implementation locates the framed color board from pixels, checks that it
is a complete 14-by-14 tiling, reconstructs both advertised source regions,
and only then decides the connectivity of their yellow intersection.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

GRID = 14
WHITE = (248, 248, 244)
BLACK = (25, 31, 34)
CYAN = (22, 164, 183)
ORANGE = (235, 112, 42)
YELLOW = (247, 204, 55)
PALETTE = (CYAN, ORANGE, YELLOW)


class OracleError(ValueError):
    pass


def _components(labels: np.ndarray, allowed: set[int]) -> list[int]:
    height, width = labels.shape
    unseen = {(int(y), int(x)) for y, x in np.argwhere(np.isin(labels, list(allowed)))}
    sizes: list[int] = []
    while unseen:
        start = unseen.pop()
        queue = deque([start])
        size = 1
        while queue:
            y, x = queue.popleft()
            for point in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if point in unseen:
                    unseen.remove(point)
                    queue.append(point)
                    size += 1
        sizes.append(size)
    return sorted(sizes, reverse=True)


def _bbox(mask: np.ndarray, what: str) -> tuple[int, int, int, int]:
    points = np.argwhere(mask)
    if not len(points):
        raise OracleError(f"missing {what}")
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    return int(x0), int(y0), int(x1), int(y1)


def _decode(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[2] != 3 or min(array.shape[:2]) < 128:
        raise OracleError("unexpected raster dimensions")
    black = np.all(array == np.asarray(BLACK, dtype=np.uint8), axis=2)
    color_masks = [np.all(array == np.asarray(rgb, dtype=np.uint8), axis=2) for rgb in PALETTE]
    colored = np.logical_or.reduce(color_masks)
    bx0, by0, bx1, by1 = _bbox(black | colored, "framed board")
    cx0, cy0, cx1, cy1 = _bbox(colored, "color board")
    board_width, board_height = cx1 - cx0 + 1, cy1 - cy0 + 1
    if board_width != board_height or board_width % GRID:
        raise OracleError("color board is not a square 14-cell tiling")
    inset = cx0 - bx0
    if inset < 4 or cy0 - by0 != inset or bx1 - cx1 != inset or by1 - cy1 != inset:
        raise OracleError("frame is not uniformly inset around the board")
    frame_crop = array[by0 : by1 + 1, bx0 : bx1 + 1]
    inner = np.zeros(frame_crop.shape[:2], dtype=bool)
    inner[inset : inset + board_height, inset : inset + board_width] = True
    if not np.all(frame_crop[~inner] == np.asarray(BLACK, dtype=np.uint8)):
        raise OracleError("outer frame is incomplete")
    board = array[cy0 : cy1 + 1, cx0 : cx1 + 1]
    valid = np.logical_or.reduce(
        [np.all(board == np.asarray(rgb, dtype=np.uint8), axis=2) for rgb in PALETTE]
    )
    if not bool(valid.all()):
        raise OracleError("board contains erased or unknown pixels")
    cell = board_width // GRID
    labels = np.empty((GRID, GRID), dtype=np.uint8)
    for y in range(GRID):
        for x in range(GRID):
            tile = board[y * cell : (y + 1) * cell, x * cell : (x + 1) * cell]
            code = 0
            for index, rgb in enumerate(PALETTE, start=1):
                if bool(np.all(tile == np.asarray(rgb, dtype=np.uint8))):
                    code = index
                    break
            if not code:
                raise OracleError("a grid cell is not uniformly colored")
            labels[y, x] = code
    return labels


def decision_from_image(image: Image.Image) -> str:
    labels = _decode(image)
    cyan_region = _components(labels, {1, 3})
    orange_region = _components(labels, {2, 3})
    yellow = _components(labels, {3})
    if len(cyan_region) != 1 or len(orange_region) != 1:
        raise OracleError("one advertised source region is disconnected")
    if len(yellow) not in (1, 2) or sum(yellow) != 8 or min(yellow) < 4:
        raise OracleError("yellow overlap is outside the declared observable regime")
    return "yes" if len(yellow) == 2 else "no"
