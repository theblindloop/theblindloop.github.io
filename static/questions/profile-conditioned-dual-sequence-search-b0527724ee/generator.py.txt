"""Deterministic latent scenes and a Pillow raster for dual-sequence search."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

LABELS = ("A", "B", "C", "D")
TARGET = ("LC", "SC", "LQ", "SQ")
SIZE_ONLY = ("LC", "SQ", "LQ", "SC")
SHAPE_ONLY = ("SC", "LC", "LQ", "SQ")
NEITHER = ("SC", "LQ", "LC", "SQ")
PATTERNS = (TARGET, SIZE_ONLY, SHAPE_ONLY, NEITHER)
THEMES = (
    ((28, 104, 128), (248, 247, 243), (235, 237, 234)),
    ((76, 62, 150), (249, 247, 242), (237, 235, 239)),
    ((145, 52, 78), (247, 248, 244), (239, 235, 235)),
    ((42, 119, 72), (249, 247, 243), (235, 238, 234)),
)


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; consecutive seeds cycle the answer A/B/C/D."""
    rng = random.Random(int(seed) ^ 0x5EEDB17)
    target_lane = int(seed) % 4
    distractors = [list(value) for value in PATTERNS[1:]]
    rng.shuffle(distractors)
    lanes: list[list[str]] = []
    for lane in range(4):
        lanes.append(list(TARGET) if lane == target_lane else distractors.pop())
    small = rng.randint(17, 21)
    large = small + rng.randint(14, 19)
    ink, background, card = THEMES[(int(seed) // 4) % len(THEMES)]
    return {
        "width": 840,
        "height": 620,
        "lane_patterns": lanes,
        "small_size": small,
        "large_size": large,
        "ink_rgb": list(ink),
        "background_rgb": list(background),
        "card_rgb": list(card),
        "x_offset": rng.randint(-9, 9),
        "x_gap": rng.randint(112, 122),
        "y_offset": rng.randint(-4, 4),
    }


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _score(pattern: Iterable[str]) -> int:
    values = tuple(pattern)
    if len(values) != 4:
        return -100
    expected_sizes = ("L", "S", "L", "S")
    expected_shapes = ("C", "C", "Q", "Q")
    return sum(value[0] == wanted for value, wanted in zip(values, expected_sizes)) + sum(
        value[1] == wanted for value, wanted in zip(values, expected_shapes)
    )


def render(scene: dict) -> Image.Image:
    """Render a single RGB page; every lane has identical object inventory."""
    scale = 3
    width = int(scene["width"])
    height = int(scene["height"])
    lanes = scene["lane_patterns"]
    small = int(scene["small_size"])
    large = int(scene["large_size"])
    ink = tuple(scene["ink_rgb"])
    background = tuple(scene["background_rgb"])
    card = tuple(scene["card_rgb"])
    x_offset = int(scene["x_offset"])
    x_gap = int(scene["x_gap"])
    y_offset = int(scene["y_offset"])

    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)

    def box(coords: tuple[float, float, float, float], **kwargs) -> None:
        draw.rounded_rectangle(tuple(int(v * scale) for v in coords), radius=14 * scale, **kwargs)

    neutral = (42, 43, 46)
    rule = (122, 124, 124)
    draw.text((width * scale // 2, 23 * scale), "MATCH BOTH ORDERS", fill=neutral,
              font=_font(25 * scale), anchor="mm")
    guide = "SIZE: LARGE - SMALL - LARGE - SMALL     SHAPE: CIRCLE - CIRCLE - SQUARE - SQUARE"
    draw.text((width * scale // 2, 57 * scale), guide, fill=neutral,
              font=_font(15 * scale), anchor="mm")
    draw.line((38 * scale, 78 * scale, (width - 38) * scale, 78 * scale), fill=rule, width=2 * scale)

    for lane_index, pattern in enumerate(lanes):
        cy = 132 + lane_index * 126 + y_offset
        box((112, cy - 49, width - 35, cy + 49), fill=card, outline=rule, width=2 * scale)
        draw.ellipse(((47) * scale, (cy - 24) * scale, (95) * scale, (cy + 24) * scale),
                     fill=(255, 255, 255), outline=neutral, width=2 * scale)
        draw.text((71 * scale, cy * scale), LABELS[lane_index], fill=neutral,
                  font=_font(28 * scale), anchor="mm")
        for column, code in enumerate(pattern):
            cx = 286 + x_offset + column * x_gap
            half = large if code[0] == "L" else small
            xy = ((cx - half) * scale, (cy - half) * scale,
                  (cx + half) * scale, (cy + half) * scale)
            if code[1] == "C":
                draw.ellipse(xy, fill=ink)
            else:
                draw.rectangle(xy, fill=ink)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    scores = [_score(pattern) for pattern in scene["lane_patterns"]]
    winners = [index for index, score in enumerate(scores) if score == max(scores)]
    if len(winners) != 1 or scores[winners[0]] != 8:
        raise ValueError("scene does not have one exact dual-sequence match")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    scores = sorted((_score(pattern) for pattern in scene["lane_patterns"]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    return int(scene["large_size"]) - int(scene["small_size"]) < 8


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
