"""Independent inverse arm using only pixels from the final PNG."""

from __future__ import annotations

from dataclasses import dataclass

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")


class OracleAbstention(ValueError):
    """The raster does not expose one well-separated dual-pattern winner."""


@dataclass(frozen=True)
class Component:
    cx: float
    cy: float
    width: int
    height: int
    area: int
    fill_ratio: float


def _components(mask: np.ndarray) -> list[Component]:
    height, width = mask.shape
    ys, xs = np.nonzero(mask)
    unseen = set((ys * width + xs).tolist())
    found: list[Component] = []
    while unseen:
        start = unseen.pop()
        stack = [start]
        pixels = [start]
        while stack:
            value = stack.pop()
            y, x = divmod(value, width)
            for neighbor in (value - 1, value + 1, value - width, value + width):
                if neighbor not in unseen:
                    continue
                ny, nx = divmod(neighbor, width)
                if abs(nx - x) + abs(ny - y) != 1:
                    continue
                unseen.remove(neighbor)
                stack.append(neighbor)
                pixels.append(neighbor)
        if len(pixels) < 250:
            continue
        py = np.fromiter((value // width for value in pixels), dtype=np.int32)
        px = np.fromiter((value % width for value in pixels), dtype=np.int32)
        x0, x1 = int(px.min()), int(px.max())
        y0, y1 = int(py.min()), int(py.max())
        box_area = (x1 - x0 + 1) * (y1 - y0 + 1)
        found.append(Component(float(px.mean()), float(py.mean()), x1 - x0 + 1,
                               y1 - y0 + 1, len(pixels), len(pixels) / box_area))
    return found


def _score(codes: list[str]) -> int:
    if len(codes) != 4:
        return -100
    sizes = ("L", "S", "L", "S")
    shapes = ("C", "C", "Q", "Q")
    return sum(code[0] == expected for code, expected in zip(codes, sizes)) + sum(
        code[1] == expected for code, expected in zip(codes, shapes)
    )


def analyze_image(image: Image.Image) -> dict:
    """Enumerate lanes and return their measured codes, scores, and top gap."""
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    channel_max = arr.max(axis=2)
    channel_min = arr.min(axis=2)
    # Symbols are visibly saturated; all typography, cards, and rules are neutral.
    mask = ((channel_max - channel_min) >= 42) & (channel_max <= 190)
    components = _components(mask)
    if len(components) != 16:
        raise OracleAbstention(f"expected 16 visible symbols, measured {len(components)}")
    components.sort(key=lambda item: (item.cy, item.cx))
    lanes = [sorted(components[index:index + 4], key=lambda item: item.cx)
             for index in range(0, 16, 4)]
    if any(max(item.cy for item in lane) - min(item.cy for item in lane) > 3.0 for lane in lanes):
        raise OracleAbstention("symbol components do not form four horizontal lanes")
    extents = [0.5 * (item.width + item.height) for item in components]
    ordered = sorted(extents)
    small_max, large_min = ordered[7], ordered[8]
    if large_min - small_max < 6.0:
        raise OracleAbstention("large and small symbols are not pixel-separable")
    size_cut = 0.5 * (small_max + large_min)
    measured: list[list[str]] = []
    scores: list[int] = []
    for lane in lanes:
        codes = []
        for item in lane:
            size = "L" if 0.5 * (item.width + item.height) > size_cut else "S"
            shape = "C" if item.fill_ratio < 0.90 else "Q"
            codes.append(size + shape)
        measured.append(codes)
        scores.append(_score(codes))
    ranking = sorted(range(4), key=lambda index: scores[index], reverse=True)
    top_gap = scores[ranking[0]] - scores[ranking[1]]
    if scores[ranking[0]] != 8 or top_gap < 2:
        raise OracleAbstention("no unique exact lane with a runner-up margin")
    return {
        "decision": LABELS[ranking[0]],
        "codes": measured,
        "scores": scores,
        "runner_up_margin": float(top_gap),
        "size_extent_gap": float(large_min - small_max),
    }


def decision_from_image(image: Image.Image) -> str:
    return str(analyze_image(image)["decision"])
