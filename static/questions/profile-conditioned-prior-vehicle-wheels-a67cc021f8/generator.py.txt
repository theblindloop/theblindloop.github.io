"""Prior-conflict vehicle-wheel world: count the visible wheels.

The latent scene holds a schematic vehicle body with ``n`` separate dark wheels
arranged in a row beneath it. A familiar canonical prior asserts that a car has
FOUR wheels; the counterfactual renders a non-canonical count (three, six) and
the decision must be read from the visible wheels alone, never from the prior.
The pixel oracle counts the actual wheel blobs by color segmentation.

Every wheel is a separate, fully opaque dark disk that never touches the body or
another wheel, so the wheel count is a hard discrete label stable over a wide
range of geometry and is directly visible in the raster.

Latent alias mode is ``not_applicable``: the wheel count, wheel radius, and all
body geometry are directly reflected in the pixels, so the rendering is injective
in the latent scene and there is no render-invariant alias. The canonical-prior
name "vehicle" is a matched-control/counterfactual manipulation of the
*question*, not of any latent field.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

CANVAS = 512
# Geometry in px on a 512x512 canvas.
EDGE = 40.0            # horizontal margin from canvas left edge to first wheel centre
WHEEL_Y = 402.0        # vertical centre of every wheel
BODY_Y0_MIN = 232.0    # body top range
BODY_Y0_MAX = 248.0
BODY_BOTTOM = 330.0    # body bottom edge (well above the wheel tops)

# Supported off-quarantine wheel counts. The canonical prior is 4 (a car); the
# counterfactuals 3 (a tricycle / three-wheeler) and 6 (a six-wheeled truck)
# contradict it. Every supported count leaves large, unambiguous gaps between
# adjacent wheel rims at the chosen radius range.
SUPPORTED_N = (3, 4, 6)

# Wheel radius range. Bounded by the tightest layout (N=6): step=432/6=72px, so
# the rim gap is 72 - 2*R; keeping R <= 31 keeps the gap >= 10px.
RADIUS_MIN = 24.0
RADIUS_MAX = 31.0

# A count is quarantined when adjacent wheel rims crowd below this gap and can
# fuse into one blob at pixel scale. Supported counts never reach it.
QUAR_GAP_PX = 10.0

BACKGROUND = (255, 255, 255)
BODY_COLOR = (46, 128, 134)      # teal body (well separated from bg and wheels)
WHEEL_COLOR = (22, 20, 34)       # near-black wheels


def sample_scene(seed: int) -> dict:
    """Deterministic scene dict from an integer seed."""
    rng = random.Random(seed)
    n = SUPPORTED_N[rng.randrange(len(SUPPORTED_N))]
    wheel_r = rng.uniform(RADIUS_MIN, RADIUS_MAX)
    step = (CANVAS - 2.0 * EDGE) / float(n)
    wheels = []
    for k in range(n):
        cx = EDGE + step * (k + 0.5)
        wheels.append([round(cx, 3), WHEEL_Y, round(wheel_r, 3)])
    body_len = n * step + rng.uniform(-8.0, 10.0)
    body_y0 = rng.uniform(BODY_Y0_MIN, BODY_Y0_MAX)
    body_x0 = CANVAS / 2.0 - body_len / 2.0
    body_x1 = CANVAS / 2.0 + body_len / 2.0
    return {
        "n": n,
        "canvas": CANVAS,
        "wheels": wheels,
        "wheel_r": round(wheel_r, 3),
        "body_x0": round(body_x0, 3),
        "body_x1": round(body_x1, 3),
        "body_y0": round(body_y0, 3),
    }


def render(scene: dict) -> Image.Image:
    """Render the vehicle (body + wheels) to an RGB image."""
    img = Image.new("RGB", (int(scene["canvas"]), int(scene["canvas"])), BACKGROUND)
    draw = ImageDraw.Draw(img)
    body_x0 = float(scene["body_x0"])
    body_x1 = float(scene["body_x1"])
    body_y0 = float(scene["body_y0"])
    # Rounded body ("cabin") so the vehicle reads as a familiar object.
    draw.rounded_rectangle(
        [body_x0, body_y0, body_x1, BODY_BOTTOM],
        radius=18.0,
        fill=BODY_COLOR,
    )
    for cx, cy, r in scene["wheels"]:
        draw.ellipse([cx - r, cy - r, cx + r, cy + r], fill=WHEEL_COLOR)
    return img


def analytic_gold(scene: dict) -> str:
    """Decision = the visible number of wheels, as a string."""
    return str(len(scene["wheels"]))


def margin(scene: dict) -> float:
    """Verifiability margin = smallest rim gap between adjacent wheels (px)."""
    wheels = scene["wheels"]
    step = (CANVAS - 2.0 * EDGE) / float(len(wheels))
    r = float(scene["wheel_r"])
    return step - 2.0 * r


def is_quarantined(scene: dict) -> bool:
    """Quarantine counts whose adjacent wheel rims could fuse at pixel scale."""
    if int(scene["n"]) not in SUPPORTED_N:
        return True
    return margin(scene) < QUAR_GAP_PX


def latent_symmetries(scene: dict):
    """not_applicable: empty, the latent rendering is injective."""
    return []
