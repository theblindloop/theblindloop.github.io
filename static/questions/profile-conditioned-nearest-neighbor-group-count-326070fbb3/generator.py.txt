"""Render and analytically score nearest-neighbor constellation scenes."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

CANVAS = 512
BACKGROUND = (248, 247, 243)
INK = (28, 34, 42)
DOT_RADIUS = 10


def _graph_stats(points: list[list[float]] | list[tuple[float, float]]) -> tuple[int, float]:
    """Return weak component count and minimum nearest/runner-up gap."""
    n = len(points)
    nearest: list[int] = []
    gaps: list[float] = []
    for i, point in enumerate(points):
        distances = sorted(
            (math.dist(point, other), j) for j, other in enumerate(points) if j != i
        )
        nearest.append(distances[0][1])
        gaps.append(distances[1][0] - distances[0][0])
    adjacency = [set() for _ in range(n)]
    for i, j in enumerate(nearest):
        adjacency[i].add(j)
        adjacency[j].add(i)
    seen: set[int] = set()
    components = 0
    for start in range(n):
        if start in seen:
            continue
        components += 1
        stack = [start]
        seen.add(start)
        while stack:
            for neighbor in adjacency[stack.pop()]:
                if neighbor not in seen:
                    seen.add(neighbor)
                    stack.append(neighbor)
    return components, min(gaps)


def _normalize(raw: list[tuple[float, float]], angle: float) -> list[list[float]]:
    """Rotate, isotropically scale to a common extent, and center the constellation."""
    ca, sa = math.cos(angle), math.sin(angle)
    rotated = [(ca * x - sa * y, sa * x + ca * y) for x, y in raw]
    min_x = min(x for x, _ in rotated)
    max_x = max(x for x, _ in rotated)
    min_y = min(y for _, y in rotated)
    max_y = max(y for _, y in rotated)
    scale = 336.0 / max(max_x - min_x, max_y - min_y)
    cx = (min_x + max_x) / 2
    cy = (min_y + max_y) / 2
    return [[256 + scale * (x - cx), 256 + scale * (y - cy)] for x, y in rotated]


def sample_scene(seed: int) -> dict:
    """Sample eight dots, conditioned on one or two robust NN components."""
    rng = random.Random(int(seed))
    target = 1 if int(seed) % 2 == 0 else 2
    for _ in range(250_000):
        raw = [(rng.random(), rng.random()) for _ in range(8)]
        points = _normalize(raw, rng.uniform(0.0, math.tau))
        count, gap = _graph_stats(points)
        pair_clearance = min(
            math.dist(a, b) for i, a in enumerate(points) for b in points[i + 1 :]
        )
        if count == target and gap >= 7.5 and pair_clearance >= 31.0:
            # Quantization is part of the declared latent scene. Recheck after it.
            quantized = [[round(x, 2), round(y, 2)] for x, y in points]
            q_count, q_gap = _graph_stats(quantized)
            if q_count == target and q_gap >= 7.4:
                return {"points": quantized, "radius": DOT_RADIUS, "canvas": CANVAS}
    raise RuntimeError(f"could not sample a robust {target}-group scene")


def render(scene: dict) -> Image.Image:
    """Draw only the evidence: identical black disks on a neutral square canvas."""
    scale = 3
    size = int(scene.get("canvas", CANVAS))
    radius = float(scene.get("radius", DOT_RADIUS))
    image = Image.new("RGB", (size * scale, size * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x, y in scene["points"]:
        draw.ellipse(
            ((x - radius) * scale, (y - radius) * scale,
             (x + radius) * scale, (y + radius) * scale),
            fill=INK,
        )
    return image.resize((size, size), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    count, _ = _graph_stats(scene["points"])
    if count not in (1, 2):
        raise ValueError(f"scene has unsupported group count {count}")
    return "one" if count == 1 else "two"


def margin(scene: dict) -> float:
    """Smallest per-dot gap between its nearest and second-nearest centers."""
    return float(_graph_stats(scene["points"])[1])


def is_quarantined(scene: dict) -> bool:
    count, gap = _graph_stats(scene["points"])
    return count not in (1, 2) or gap < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Point-list order is a genuine render-invariant latent alias."""
    twin = dict(scene)
    twin["points"] = list(reversed(scene["points"]))
    return [("reverse_point_storage_order", twin)]
