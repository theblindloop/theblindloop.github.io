"""Independent PNG-only inverse for the nearest-neighbor constellation."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


def _centers_from_pixels(image: Image.Image) -> list[tuple[float, float]]:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    # Neutral background is near 248 in every channel; antialiased ink is much darker.
    mask = np.max(rgb, axis=2) < 125
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    centers: list[tuple[float, float]] = []
    for y in range(height):
        for x in range(width):
            if not mask[y, x] or visited[y, x]:
                continue
            stack = [(x, y)]
            visited[y, x] = True
            xs: list[int] = []
            ys: list[int] = []
            while stack:
                px, py = stack.pop()
                xs.append(px)
                ys.append(py)
                for nx, ny in ((px - 1, py), (px + 1, py), (px, py - 1), (px, py + 1)):
                    if 0 <= nx < width and 0 <= ny < height and mask[ny, nx] and not visited[ny, nx]:
                        visited[ny, nx] = True
                        stack.append((nx, ny))
            if len(xs) >= 80:
                centers.append((sum(xs) / len(xs), sum(ys) / len(ys)))
    if len(centers) != 8:
        raise ValueError(f"expected 8 visible dot components, recovered {len(centers)}")
    return centers


def decision_from_image(image: Image.Image) -> str:
    points = _centers_from_pixels(image)
    nearest: list[int] = []
    for i, point in enumerate(points):
        distances = sorted(
            (math.dist(point, other), j) for j, other in enumerate(points) if j != i
        )
        if distances[1][0] - distances[0][0] < 4.0:
            raise ValueError("pixel reconstruction has an ambiguous nearest-neighbor choice")
        nearest.append(distances[0][1])
    adjacency = [set() for _ in points]
    for i, j in enumerate(nearest):
        adjacency[i].add(j)
        adjacency[j].add(i)
    seen: set[int] = set()
    groups = 0
    for start in range(len(points)):
        if start in seen:
            continue
        groups += 1
        stack = [start]
        seen.add(start)
        while stack:
            for neighbor in adjacency[stack.pop()]:
                if neighbor not in seen:
                    seen.add(neighbor)
                    stack.append(neighbor)
    if groups not in (1, 2):
        raise ValueError(f"unsupported reconstructed group count {groups}")
    return "one" if groups == 1 else "two"
