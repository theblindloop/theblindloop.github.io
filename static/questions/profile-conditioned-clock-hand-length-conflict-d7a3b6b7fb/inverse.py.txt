"""Independent PNG-only inverse for the colored clock-hand comparison."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _masks(image: Image.Image) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    p = np.asarray(image.convert("RGB"), dtype=np.int16)
    blue = (p[..., 2] > 115) & (p[..., 2] > p[..., 0] * 1.35) & (p[..., 2] > p[..., 1] * 1.25)
    red = (p[..., 0] > 145) & (p[..., 0] > p[..., 1] * 1.70) & (p[..., 0] > p[..., 2] * 1.55)
    yellow = (p[..., 0] > 180) & (p[..., 1] > 120) & (p[..., 1] < p[..., 0] * 0.90) & (p[..., 2] < 100)
    return blue, red, yellow


def _extent(mask: np.ndarray, cx: float, cy: float) -> float:
    ys, xs = np.nonzero(mask)
    if xs.size < 300:
        raise ValueError("colored hand evidence is absent")
    distances = np.hypot(xs.astype(float) - cx, ys.astype(float) - cy)
    return float(np.percentile(distances, 99.5))


def decision_from_image(image: Image.Image) -> str:
    blue, red, yellow = _masks(image)
    hub_y, hub_x = np.nonzero(yellow)
    if hub_x.size < 350:
        raise ValueError("center pivot evidence is absent")
    cx = float(np.median(hub_x))
    cy = float(np.median(hub_y))
    scores = {"blue": _extent(blue, cx, cy), "red": _extent(red, cx, cy)}
    ranked = sorted(scores, key=scores.get, reverse=True)
    if scores[ranked[0]] - scores[ranked[1]] < 20.0:
        raise ValueError("hand-length decision is ambiguous at pixel resolution")
    return ranked[0]
