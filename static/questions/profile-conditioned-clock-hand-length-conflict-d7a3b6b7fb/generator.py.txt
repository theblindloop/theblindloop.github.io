"""Deterministic matched clock scenes with visible hand-length conflicts."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

SIZE = 384
SCALE = 4
COLORS = {"blue": (34, 104, 190), "red": (205, 49, 58)}


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed ^ 0xC10C4)
    shorter = rng.randint(80, 94)
    longer = 220 - shorter
    longer_color = "blue" if seed % 2 == 0 else "red"
    base = (seed // 2 % 8) * 45.0 + rng.uniform(-8.0, 8.0)
    separation = rng.choice((92.0, 118.0, 142.0))
    if rng.randrange(2):
        separation = -separation
    lengths = {longer_color: longer, "red" if longer_color == "blue" else "blue": shorter}
    return {
        "blue_length": lengths["blue"],
        "red_length": lengths["red"],
        "blue_angle_deg": round(base % 360.0, 4),
        "red_angle_deg": round((base + separation) % 360.0, 4),
        "center_x": 192 + rng.randint(-10, 10),
        "center_y": 192 + rng.randint(-10, 10),
        "radius": rng.randint(148, 156),
        "face_tint": "#fbfaf5" if rng.randrange(2) else [251, 250, 245],
        "draw_order": ["blue", "red"] if rng.randrange(2) else ["red", "blue"],
    }


def _point(cx: float, cy: float, distance: float, degrees: float) -> tuple[float, float]:
    angle = math.radians(degrees)
    return cx + distance * math.cos(angle), cy + distance * math.sin(angle)


def render(scene: dict) -> Image.Image:
    cx = float(scene["center_x"]) * SCALE
    cy = float(scene["center_y"]) * SCALE
    radius = float(scene["radius"]) * SCALE
    face_tint = scene["face_tint"]
    if isinstance(face_tint, list):
        face_tint = tuple(face_tint)
    hand_lengths = {
        "blue": float(scene["blue_length"]),
        "red": float(scene["red_length"]),
    }
    hand_angles = {
        "blue": float(scene["blue_angle_deg"]),
        "red": float(scene["red_angle_deg"]),
    }

    image = Image.new("RGB", (SIZE * SCALE, SIZE * SCALE), (236, 239, 242))
    draw = ImageDraw.Draw(image)
    face_box = (cx - radius, cy - radius, cx + radius, cy + radius)
    draw.ellipse(face_box, fill=face_tint, outline=(47, 55, 66), width=6 * SCALE)

    # A fixed neutral dial makes the object unmistakably a clock without encoding
    # which hand is longer.
    for index in range(60):
        angle = index * 6.0 - 90.0
        outer = _point(cx, cy, radius - 10 * SCALE, angle)
        tick = (13 if index % 5 == 0 else 7) * SCALE
        inner = _point(cx, cy, radius - 10 * SCALE - tick, angle)
        width = (4 if index % 5 == 0 else 2) * SCALE
        draw.line((inner, outer), fill=(67, 73, 82), width=width)

    # The two equal-width hands are drawn in a latent order. Their angular
    # separation keeps any overlap inside the hub, which is painted afterward.
    for color in scene["draw_order"]:
        length = hand_lengths[color] * SCALE
        angle = hand_angles[color]
        start = _point(cx, cy, 6 * SCALE, angle)
        tip = _point(cx, cy, length, angle)
        draw.line((start, tip), fill=COLORS[color], width=11 * SCALE)
        arrow_left = _point(*tip, 14 * SCALE, angle + 150)
        arrow_right = _point(*tip, 14 * SCALE, angle - 150)
        draw.polygon((tip, arrow_left, arrow_right), fill=COLORS[color])

    hub_r = 17 * SCALE
    draw.ellipse(
        (cx - hub_r, cy - hub_r, cx + hub_r, cy + hub_r),
        fill=(245, 185, 38),
        outline=(87, 70, 28),
        width=3 * SCALE,
    )
    return image.resize((SIZE, SIZE), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    blue = float(scene["blue_length"])
    red = float(scene["red_length"])
    return "blue" if blue > red else "red"


def margin(scene: dict) -> float:
    return abs(float(scene["blue_length"]) - float(scene["red_length"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 24.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["draw_order"] = list(reversed(scene["draw_order"]))
    return [("reverse_hidden_hand_draw_order", twin)]
