"""Renderer, analytic gold and sampler for the tick-twin-run world.

One self-avoiding wire of constant thickness winds across the whole frame.
Eight small square ticks (four orange, four blue) are printed on the wire.
Read in the order in which a walk along the wire's own ink meets them, exactly
one consecutive pair carries the same colour; the decision is that colour.
"""

from __future__ import annotations

import random
from itertools import combinations

from PIL import Image, ImageDraw

W, H = 900, 620
PITCH = 48
OX, OY = 60, 60
COLS, ROWS = 17, 11
THICK = 13          # wire stroke width, half-width 6
TICK = 7            # tick square side, strictly narrower than the wire
HALF_W = THICK // 2
HALF_T = TICK // 2

BG = (255, 255, 255)
WIRE = (38, 38, 44)
ORANGE = (214, 102, 20)
BLUE = (36, 120, 206)
GRAY = (176, 176, 176)

COLORS = {"orange": ORANGE, "blue": BLUE}

QUARANTINE_MARGIN = 22.0
MIN_CLEAR = 30.0            # normal scenes: clear wire between tick edges
TIGHT_CLEAR = (11.0, 18.0)  # deliberately tight draws
TWIN_SPAN = (230.0, 520.0)  # shared straight-line band for the twin pair


# --------------------------------------------------------------------------
# route construction
# --------------------------------------------------------------------------
def _cx(i: int) -> int:
    return OX + PITCH * i


def _cy(j: int) -> int:
    return OY + PITCH * j


def _neighbours(cell):
    i, j = cell
    for di, dj in ((1, 0), (-1, 0), (0, 1), (0, -1)):
        ni, nj = i + di, j + dj
        if 0 <= ni < COLS and 0 <= nj < ROWS:
            yield (ni, nj)


def _grow_route(rng: random.Random):
    """A folded ribbon along one image diagonal: thin, self-avoiding, corner to corner.

    The route zigzags across a narrow diagonal band, so it stays one thin
    elongated component at coarse scale while still folding tightly enough that
    page proximity says nothing about proximity along the walk.
    """
    width = rng.choice((2, 3))
    down = rng.random() < 0.5
    spine = []
    for i in range(COLS):
        t = i / (COLS - 1)
        centre = t * (ROWS - 1 - (width - 1)) if down else (ROWS - 1 - (width - 1)) * (1.0 - t)
        base = int(round(centre))
        spine.append(base)

    route = []
    entry = spine[0]
    for i, base in enumerate(spine):
        band = [base + k for k in range(width)]
        if entry not in band:
            entry = min(band, key=lambda r: abs(r - entry))
        far = min(band) if (entry - min(band)) >= (max(band) - entry) else max(band)
        step = 1 if far >= entry else -1
        for row in range(entry, far + step, step):
            cell = (i, row)
            if cell not in route:
                route.append(cell)
        entry = far
    return route


def _route(rng: random.Random):
    for _ in range(200):
        route = _grow_route(rng)
        if len(route) < 30:
            continue
        if len({i for i, _ in route}) != COLS:
            continue
        if max(j for _, j in route) - min(j for _, j in route) < ROWS - 3:
            continue
        if any(
            abs(a[0] - b[0]) + abs(a[1] - b[1]) != 1 for a, b in zip(route, route[1:])
        ):
            continue
        if len(set(route)) != len(route):
            continue
        if rng.random() < 0.5:
            route = list(reversed(route))
        return route
    raise RuntimeError("no frame-spanning ribbon")


def _segments(route):
    pts = [(_cx(i), _cy(j)) for i, j in route]
    segs = []
    s = 0.0
    for a, b in zip(pts, pts[1:]):
        length = float(abs(b[0] - a[0]) + abs(b[1] - a[1]))
        segs.append((s, a, b, length))
        s += length
    return segs


def _point_at(segs, s: float):
    for s0, a, b, length in segs:
        if s0 <= s <= s0 + length:
            t = (s - s0) / length
            return (
                int(round(a[0] + (b[0] - a[0]) * t)),
                int(round(a[1] + (b[1] - a[1]) * t)),
            )
    raise ValueError("arc position off route")


def _straight_positions(segs):
    """Arc positions at least 14 px from any vertex, so the wire is locally straight."""
    out = []
    for k in range(2, len(segs) - 2):
        s0, _a, _b, length = segs[k]
        for off in range(14, int(length) - 13):
            out.append(float(s0 + off))
    return out


# --------------------------------------------------------------------------
# tick colouring
# --------------------------------------------------------------------------
def _twin_sequences():
    """Balanced 8-tick colourings with exactly one same-coloured neighbour pair."""
    out = {"orange": [], "blue": []}
    names = ("orange", "blue")
    for bits in combinations(range(8), 4):
        seq = [1] * 8
        for b in bits:
            seq[b] = 0
        twins = [i for i in range(7) if seq[i] == seq[i + 1]]
        if len(twins) != 1:
            continue
        out[names[seq[twins[0]]]].append((tuple(seq), twins[0]))
    return out


SEQUENCES = _twin_sequences()


# --------------------------------------------------------------------------
# scene sampling
# --------------------------------------------------------------------------
def _pick_positions(rng, allowed, tight):
    allowed_set = set(allowed)
    for _ in range(600):
        picks = sorted(rng.sample(allowed, 7))
        gaps = [picks[i + 1] - picks[i] for i in range(6)]
        if gaps and min(gaps) < MIN_CLEAR + TICK:
            continue
        if not tight:
            extra = rng.choice(allowed)
            cand = sorted(picks + [extra])
            if len(set(cand)) != 8:
                continue
            gaps = [cand[i + 1] - cand[i] for i in range(7)]
            if min(gaps) < MIN_CLEAR + TICK:
                continue
            return cand
        anchor = rng.choice(picks)
        for _try in range(40):
            clear = rng.uniform(*TIGHT_CLEAR)
            step = round(clear + TICK)
            extra = anchor + rng.choice((-1.0, 1.0)) * step
            if extra not in allowed_set:
                continue
            cand = sorted(picks + [extra])
            gaps = sorted(cand[i + 1] - cand[i] for i in range(7))
            if gaps[0] - TICK < TIGHT_CLEAR[0] or gaps[0] - TICK > TIGHT_CLEAR[1]:
                continue
            if gaps[1] < MIN_CLEAR + TICK:
                continue
            return cand
    return None


def _decoys(rng):
    out = []
    for _ in range(rng.randrange(2, 5)):
        size = rng.randrange(8, 15)
        side = rng.choice(("left", "right", "top"))
        if side == "left":
            x0 = rng.randrange(4, 11)
            y0 = rng.randrange(24, H - 24 - size)
        elif side == "right":
            x0 = rng.randrange(W - 32, W - 16)
            y0 = rng.randrange(24, H - 24 - size)
        else:
            x0 = rng.randrange(40, W - 40 - size)
            y0 = rng.randrange(4, 11)
        out.append([int(x0), int(y0), int(x0 + size), int(y0 + size)])
    return sorted(out)


def _euclid(a, b):
    return ((a["x"] - b["x"]) ** 2 + (a["y"] - b["y"]) ** 2) ** 0.5


def _cheap_cues_defeated(ticks, twin_index, answer):
    twin = _euclid(ticks[twin_index], ticks[twin_index + 1])
    if not TWIN_SPAN[0] <= twin <= TWIN_SPAN[1]:
        return False
    same_pairs = [
        (_euclid(a, b), a["color"])
        for a, b in combinations(ticks, 2)
        if a["color"] == b["color"]
    ]
    closest = min(same_pairs)
    if closest[1] == answer:
        return False        # nearest same-coloured pair must be the decoy colour
    if closest[0] >= twin:
        return False        # and it must really look closer than the twin pair
    gaps = [(ticks[i + 1]["s"] - ticks[i]["s"], i) for i in range(7)]
    if min(gaps)[1] == twin_index:
        return False        # the twin pair is never the tightest along-path pair
    return True


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    answer = ("orange", "blue")[seed % 2]
    tight = (seed % 9) == 0
    for _attempt in range(60):
        route = _route(rng)
        segs = _segments(route)
        allowed = _straight_positions(segs)
        positions = _pick_positions(rng, allowed, tight)
        if positions is None:
            continue
        sequences = list(SEQUENCES[answer])
        rng.shuffle(sequences)
        for seq, twin_index in sequences:
            names = ("orange", "blue")
            ticks = []
            for s, bit in zip(positions, seq):
                x, y = _point_at(segs, s)
                ticks.append({"s": float(s), "x": x, "y": y, "color": names[bit]})
            if _cheap_cues_defeated(ticks, twin_index, answer):
                return {
                    "layout": {"w": W, "h": H, "pitch": PITCH, "origin": [OX, OY]},
                    "path": [[int(i), int(j)] for i, j in route],
                    "ticks": ticks,
                    "decoys": _decoys(rng),
                    "tight": bool(tight),
                }
    raise RuntimeError(f"scene sampling failed for seed {seed}")


# --------------------------------------------------------------------------
# gold, margin, quarantine
# --------------------------------------------------------------------------
def analytic_gold(scene: dict) -> str:
    """Colour of the unique same-coloured neighbour pair in along-path order."""
    ticks = sorted(scene["ticks"], key=lambda t: t["s"])
    twins = [t["color"] for a, t in zip(ticks, ticks[1:]) if a["color"] == t["color"]]
    if len(twins) != 1:
        raise ValueError("scene does not carry exactly one same-coloured pair")
    return twins[0]


def margin(scene: dict) -> float:
    ticks = sorted(scene["ticks"], key=lambda t: t["s"])
    return float(
        round(min(b["s"] - a["s"] - TICK for a, b in zip(ticks, ticks[1:])), 6)
    )


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_MARGIN)


def latent_symmetries(scene: dict):
    """Record-order rewrites: pixel-identical and answer-preserving by construction."""
    out = []
    reversed_ticks = dict(scene)
    reversed_ticks["ticks"] = list(reversed(scene["ticks"]))
    out.append(("reverse_tick_records", reversed_ticks))

    if len(scene["ticks"]) > 3:
        swapped = dict(scene)
        order = list(scene["ticks"])
        order[0], order[2] = order[2], order[0]
        swapped["ticks"] = order
        out.append(("swap_two_tick_records", swapped))

    if len(scene["decoys"]) > 1:
        rotated = dict(scene)
        rotated["decoys"] = scene["decoys"][1:] + scene["decoys"][:1]
        out.append(("rotate_decoy_records", rotated))
    return out


# --------------------------------------------------------------------------
# raster
# --------------------------------------------------------------------------
def render(scene: dict) -> Image.Image:
    layout = scene["layout"]
    image = Image.new("RGB", (int(layout["w"]), int(layout["h"])), BG)
    draw = ImageDraw.Draw(image)

    for x0, y0, x1, y1 in sorted(scene["decoys"]):
        draw.rectangle([x0, y0, x1, y1], fill=GRAY)

    pts = [(_cx(i), _cy(j)) for i, j in scene["path"]]
    for (x, y) in pts:
        draw.rectangle([x - HALF_W, y - HALF_W, x + HALF_W, y + HALF_W], fill=WIRE)
    for a, b in zip(pts, pts[1:]):
        x0, x1 = sorted((a[0], b[0]))
        y0, y1 = sorted((a[1], b[1]))
        draw.rectangle([x0 - HALF_W, y0 - HALF_W, x1 + HALF_W, y1 + HALF_W], fill=WIRE)

    for tick in sorted(scene["ticks"], key=lambda t: t["s"]):
        x, y = int(tick["x"]), int(tick["y"])
        draw.rectangle(
            [x - HALF_T, y - HALF_T, x + HALF_T, y + HALF_T],
            fill=COLORS[tick["color"]],
        )
    return image
