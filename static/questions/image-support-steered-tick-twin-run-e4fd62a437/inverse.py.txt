"""Independent pixel-only arm for the tick-twin-run world.

Receives the final PNG and nothing else: no scene, no renderer, no gold. It
segments the wire ink and the two tick colours, checks that the ink is one
connected stroke, orders the ticks by geodesic distance measured along the ink
itself, and returns the colour of the unique same-coloured neighbour pair.
Any structural violation abstains.
"""

from __future__ import annotations

from collections import deque

import numpy as np

INK = (38, 38, 44)
TINT = {"orange": (214, 102, 20), "blue": (36, 120, 206)}

STEP = 3            # ink is decimated before the geodesic walk
MIN_INK = 3000      # decimated ink pixels
TICK_AREA = (24, 90)
WINDOW = 3          # decimated radius of the ink window read around a tick
MIN_ORDER_GAP = 3   # decimated geodesic separation required between neighbours
ABSTAIN = "abstain"


def _mask(arr: np.ndarray, rgb) -> np.ndarray:
    return (
        (arr[:, :, 0] == rgb[0])
        & (arr[:, :, 1] == rgb[1])
        & (arr[:, :, 2] == rgb[2])
    )


def _blobs(mask: np.ndarray):
    """Four-connected components of a small mask, as (area, centroid) pairs."""
    pixels = {(int(y), int(x)) for y, x in zip(*np.nonzero(mask))}
    out = []
    while pixels:
        seed = pixels.pop()
        queue = deque([seed])
        blob = [seed]
        while queue:
            y, x = queue.popleft()
            for ny, nx in ((y + 1, x), (y - 1, x), (y, x + 1), (y, x - 1)):
                if (ny, nx) in pixels:
                    pixels.discard((ny, nx))
                    queue.append((ny, nx))
                    blob.append((ny, nx))
        ys = [p[0] for p in blob]
        xs = [p[1] for p in blob]
        out.append((len(blob), (sum(ys) / len(blob), sum(xs) / len(blob))))
    return out


def _geodesic(ink: np.ndarray, seed):
    dist = np.full(ink.shape, -1, dtype=np.int32)
    dist[seed] = 0
    queue = deque([seed])
    far, far_d = seed, 0
    rows, cols = ink.shape
    while queue:
        y, x = queue.popleft()
        d = dist[y, x] + 1
        if d - 1 > far_d:
            far, far_d = (y, x), d - 1
        for ny, nx in ((y + 1, x), (y - 1, x), (y, x + 1), (y, x - 1)):
            if 0 <= ny < rows and 0 <= nx < cols and ink[ny, nx] and dist[ny, nx] < 0:
                dist[ny, nx] = d
                queue.append((ny, nx))
    return dist, far


def decision_from_image(image) -> str:
    arr = np.asarray(image.convert("RGB"))

    ticks = []
    for name, rgb in TINT.items():
        for area, centroid in _blobs(_mask(arr, rgb)):
            if not TICK_AREA[0] <= area <= TICK_AREA[1]:
                return ABSTAIN
            ticks.append((name, centroid))
    if len(ticks) != 8:
        return ABSTAIN
    if sum(1 for name, _ in ticks if name == "orange") != 4:
        return ABSTAIN

    full = _mask(arr, INK)
    # a clean render has no one-pixel pinholes inside the stroke; scattered
    # raster damage does, and must not be traced as if the stroke were intact
    inner = (
        full[:-2, 1:-1] & full[2:, 1:-1] & full[1:-1, :-2] & full[1:-1, 2:]
    ) & ~full[1:-1, 1:-1]
    if bool(inner.any()):
        return ABSTAIN

    ink = full[::STEP, ::STEP]
    total = int(ink.sum())
    if total < MIN_INK:
        return ABSTAIN

    seeds = np.nonzero(ink)
    first = (int(seeds[0][0]), int(seeds[1][0]))
    _reach, end = _geodesic(ink, first)
    dist, _other = _geodesic(ink, end)
    if int((dist >= 0).sum()) != total:
        return ABSTAIN     # the ink is not one connected stroke

    ordered = []
    for name, (cy, cx) in ticks:
        gy, gx = int(round(cy / STEP)), int(round(cx / STEP))
        y0, y1 = max(gy - WINDOW, 0), min(gy + WINDOW + 1, ink.shape[0])
        x0, x1 = max(gx - WINDOW, 0), min(gx + WINDOW + 1, ink.shape[1])
        window = dist[y0:y1, x0:x1]
        values = window[window >= 0]
        if values.size < 6:
            return ABSTAIN     # the tick does not sit on the traced stroke
        ordered.append((float(np.median(values)), name))

    ordered.sort()
    for (a, _), (b, _) in zip(ordered, ordered[1:]):
        if b - a < MIN_ORDER_GAP:
            return ABSTAIN     # along-path order not resolvable at this resolution

    twins = [b for (_, a), (_, b) in zip(ordered, ordered[1:]) if a == b]
    if len(twins) != 1:
        return ABSTAIN
    return twins[0]
