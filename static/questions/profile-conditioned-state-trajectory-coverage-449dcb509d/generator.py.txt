"""Trajectory-coverage state tracker over a 3x3 grid.

A single ringed token moves on a 3x3 grid. Four panels are stacked vertically
and read top to bottom (P1, P2, P3, P4). Each panel shows the static 3x3 grid
and a filled red disk (the token) in its current cell. A magenta ring in the
top panel (P1) marks the token's starting cell. Between each pair of consecutive
panels a strip shows a single navy direction dot that declares the token's
one-cell move (up, down, left, or right) to the next panel.

Reading the panels in order and replaying the declared moves from the ringed
start cell yields the trajectory of four occupied cells. The decision is the
number of distinct grid cells the token occupies across the whole run
(counting P1's start cell and every cell it lands on in P2, P3, and P4).

The scene stores only the ordered list of occupied cells; the arrows drawn in
the strips are derived from the pairwise differences, so every latent field is
render-evident.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

GRID_N = 3
PANELS = 4

CELL = 60
GAP = 6
SIDE = GRID_N * CELL + (GRID_N - 1) * GAP
LEFT_M = 100
TOP_M = 50
STRIP_H = 50
TOKEN_R = 18
RING_R = 26
DIR_DIST = 30
DIR_R = 9

WIDTH = LEFT_M + SIDE + LEFT_M
HEIGHT = TOP_M + PANELS * SIDE + (PANELS - 1) * STRIP_H + 30

BG = (255, 255, 255)
BORDER = (120, 120, 120)
TOKEN = (210, 40, 40)
RING = (210, 0, 210)
DIR = (40, 40, 180)


def _col_x(i: int) -> int:
    return LEFT_M + GAP // 2 + i * (CELL + GAP) + CELL // 2


def _panel_top(k: int) -> int:
    return TOP_M + k * (SIDE + STRIP_H)


def _cell_center(k: int, r: int, c: int) -> tuple[int, int]:
    top = _panel_top(k)
    return (_col_x(c), top + GAP // 2 + r * (CELL + GAP) + CELL // 2)


def _grid_center_x() -> int:
    return LEFT_M + SIDE // 2


def _strip_cy(k: int) -> int:
    return _panel_top(k) + SIDE + STRIP_H // 2


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    cells = []
    r = rng.randint(0, GRID_N - 1)
    c = rng.randint(0, GRID_N - 1)
    cells.append((r, c))
    for _ in range(PANELS - 1):
        options = []
        if r > 0:
            options.append((r - 1, c))
        if r < GRID_N - 1:
            options.append((r + 1, c))
        if c > 0:
            options.append((r, c - 1))
        if c < GRID_N - 1:
            options.append((r, c + 1))
        r, c = options[rng.randrange(len(options))]
        cells.append((r, c))
    return {"cells": cells}


def _moves_from(cells: list[tuple[int, int]]) -> list[tuple[int, int]]:
    moves = []
    for (r0, c0), (r1, c1) in zip(cells, cells[1:]):
        moves.append((r1 - r0, c1 - c0))
    return moves


def render(scene: dict) -> Image.Image:
    cells = scene["cells"]
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)

    for k in range(PANELS):
        top = _panel_top(k)
        for r in range(GRID_N):
            for c in range(GRID_N):
                x0 = LEFT_M + GAP // 2 + c * (CELL + GAP)
                y0 = top + GAP // 2 + r * (CELL + GAP)
                draw.rectangle(
                    [x0, y0, x0 + CELL, y0 + CELL], outline=BORDER, width=2
                )
        cx, cy = _cell_center(k, cells[k][0], cells[k][1])
        draw.ellipse(
            [cx - TOKEN_R, cy - TOKEN_R, cx + TOKEN_R, cy + TOKEN_R], fill=TOKEN
        )
        if k == 0:
            sx, sy = _cell_center(0, cells[0][0], cells[0][1])
            draw.ellipse(
                [sx - RING_R, sy - RING_R, sx + RING_R, sy + RING_R],
                outline=RING,
                width=4,
            )

    gx = _grid_center_x()
    for k, (dr, dc) in enumerate(_moves_from(cells)):
        sy = _strip_cy(k)
        if dr == -1:
            pos = (gx, sy - DIR_DIST)
        elif dr == 1:
            pos = (gx, sy + DIR_DIST)
        elif dc == -1:
            pos = (gx - DIR_DIST, sy)
        else:
            pos = (gx + DIR_DIST, sy)
        draw.ellipse(
            [pos[0] - DIR_R, pos[1] - DIR_R, pos[0] + DIR_R, pos[1] + DIR_R],
            fill=DIR,
        )
    return img


def analytic_gold(scene: dict) -> str:
    cells = scene["cells"]
    return str(len({(r, c) for r, c in cells}))


def margin(scene: dict) -> float:
    return 10.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    """no nontrivial render-invariant latent symmetry: ``cells`` fixes the drawn
    disk position in every panel, the ringed start cell, and the derived
    direction dots, so any change to the trajectory changes the raster."""
    return []
