"""Independent scene-free pixel oracle for the trajectory-coverage world.

Receives only the final PNG. From the raster alone it:
  1. labels pixels against the fixed palette and verifies the panel/grid
     structure (border color present);
  2. finds the magenta ring in the top panel and locates its cell using the
     fixed 3x3 grid geometry;
  3. reads each navy direction dot in the strips to recover the per-step moves
     in panel order;
  4. replays the moves from the ringed start cell to obtain the four occupied
     cells and returns the number of distinct cells visited.

The grid geometry is fixed and deterministic. No latent scene, renderer, prompt,
or gold code is accessed.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

BG = (255, 255, 255)
BORDER = (120, 120, 120)
TOKEN = (210, 40, 40)
RING = (210, 0, 210)
DIR = (40, 40, 180)

PALETTE = [BG, BORDER, TOKEN, RING, DIR]
_BG, _BRD, _TOK, _RING, _DIR = range(5)

GRID_N = 3
PANELS = 4
CELL = 60
GAP = 6
SIDE = GRID_N * CELL + (GRID_N - 1) * GAP
LEFT_M = 100
TOP_M = 50
STRIP_H = 50
DIR_DIST = 30

COL_C = [LEFT_M + GAP // 2 + i * (CELL + GAP) + CELL // 2 for i in range(GRID_N)]
ROW_C = [TOP_M + GAP // 2 + r * (CELL + GAP) + CELL // 2 for r in range(GRID_N)]
GX = LEFT_M + SIDE // 2
STRIP_LO = [TOP_M + k * (SIDE + STRIP_H) + SIDE for k in range(PANELS - 1)]


class UnreadableImageError(ValueError):
    """The raster does not expose the expected grid-panel structure."""


def _label(arr: np.ndarray) -> np.ndarray:
    dists = np.stack(
        [np.linalg.norm(arr - np.array(c, dtype=np.float32), axis=-1) for c in PALETTE],
        axis=-1,
    )
    return dists.argmin(axis=-1)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label(arr)
    if (labels == _BRD).sum() == 0:
        raise UnreadableImageError("no grid borders found")

    ring_pts = list(zip(*np.nonzero(labels == _RING)))
    if not ring_pts:
        raise UnreadableImageError("no magenta ring found")
    ring_y = sum(p[0] for p in ring_pts) / len(ring_pts)
    ring_x = sum(p[1] for p in ring_pts) / len(ring_pts)
    start_r = min(range(GRID_N), key=lambda r: abs(ROW_C[r] - ring_y))
    start_c = min(range(GRID_N), key=lambda i: abs(COL_C[i] - ring_x))

    dir_ys, dir_xs = np.nonzero(labels == _DIR)
    if len(dir_xs) == 0:
        raise UnreadableImageError("no direction dots found")

    moves = []
    for k in range(PANELS - 1):
        lo = STRIP_LO[k]
        hi = lo + STRIP_H
        ys = [int(y) for (y, x) in zip(dir_ys.tolist(), dir_xs.tolist()) if lo <= y <= hi]
        xs = [int(x) for (y, x) in zip(dir_ys.tolist(), dir_xs.tolist()) if lo <= y <= hi]
        if not ys:
            raise UnreadableImageError(f"strip {k} has no direction dot")
        off_x = int(round(float(np.mean(xs)))) - GX
        off_y = int(round(float(np.mean(ys)))) - (lo + STRIP_H // 2)
        if abs(off_x) >= abs(off_y):
            moves.append((0, -1) if off_x < 0 else (0, 1))
        else:
            moves.append((-1, 0) if off_y < 0 else (1, 0))

    occupied = {(start_r, start_c)}
    r, c = start_r, start_c
    for dr, dc in moves:
        r += dr
        c += dc
        occupied.add((r, c))
    return str(len(occupied))
