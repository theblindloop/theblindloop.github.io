"""Independent image-only inverse arm for the contour patch world."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _runs(values: np.ndarray, gap: int) -> list[tuple[int, int]]:
    if values.size == 0:
        return []
    values = np.unique(values.astype(int))
    cuts = np.flatnonzero(np.diff(values) > gap)
    starts = np.r_[0, cuts + 1]
    ends = np.r_[cuts, len(values) - 1]
    return [(int(values[a]), int(values[b])) for a, b in zip(starts, ends, strict=True)]


def _signature(mask: np.ndarray, x0: int, x1: int, side: str) -> tuple[float, float]:
    ys, xs = np.nonzero(mask[:, x0 : x1 + 1])
    if xs.size < 30:
        raise ValueError("insufficient contour pixels")
    xs = xs + x0
    edge = int(xs.min() if side == "left" else xs.max())
    if side == "left":
        keep = (xs >= edge) & (xs <= edge + 23)
    else:
        keep = (xs <= edge) & (xs >= edge - 23)
    local_x = xs[keep]
    local_y = ys[keep]
    med_x = np.unique(local_x)
    med_y = np.array([np.median(local_y[local_x == x]) for x in med_x], dtype=float)
    if med_x.size < 12:
        raise ValueError("endpoint direction is not visible")
    slope, intercept = np.polyfit(med_x.astype(float), med_y, 1)
    endpoint = slope * edge + intercept
    return float(endpoint), float(slope)


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.uint8)
    flat = pixels.reshape(-1, 3)
    colors, counts = np.unique(flat, axis=0, return_counts=True)
    ranked = []
    for color, count in zip(colors, counts, strict=True):
        spread = int(color.max()) - int(color.min())
        if spread >= 70 and int(count) >= 250:
            ranked.append((int(count), tuple(int(v) for v in color)))
    ranked.sort(reverse=True)
    selected = sorted(color for _, color in ranked[:3])
    if len(selected) != 3:
        raise ValueError("three repeated contour colors were not found")

    masks = [np.all(pixels == np.array(color, dtype=np.uint8), axis=2) for color in selected]
    combined = np.logical_or.reduce(masks)
    occupied_y = np.flatnonzero(combined.any(axis=1))
    if occupied_y.size < 2:
        raise ValueError("no contour rows")
    y_gaps = np.diff(occupied_y)
    split_index = int(np.argmax(y_gaps))
    if int(y_gaps[split_index]) < 35:
        raise ValueError("context and candidates are not separable")
    split_y = (int(occupied_y[split_index]) + int(occupied_y[split_index + 1])) // 2

    top = combined.copy()
    top[split_y + 1 :, :] = False
    bottom = combined.copy()
    bottom[: split_y + 1, :] = False
    top_masks = []
    bottom_masks = []
    for mask in masks:
        top_mask = mask.copy()
        top_mask[split_y + 1 :, :] = False
        bottom_mask = mask.copy()
        bottom_mask[: split_y + 1, :] = False
        top_masks.append(top_mask)
        bottom_masks.append(bottom_mask)
    top_runs = sorted(_runs(np.flatnonzero(top.any(axis=0)), 10), key=lambda r: r[0])
    bottom_runs = sorted(_runs(np.flatnonzero(bottom.any(axis=0)), 10), key=lambda r: r[0])
    if len(top_runs) != 2 or len(bottom_runs) != 3:
        raise ValueError("expected two context fragments and three candidate patches")

    context_left = [_signature(mask, *top_runs[0], "right") for mask in top_masks]
    context_right = [_signature(mask, *top_runs[1], "left") for mask in top_masks]
    ctx_left_y = np.array([item[0] for item in context_left])
    ctx_right_y = np.array([item[0] for item in context_right])
    ctx_left_slope = np.array([item[1] for item in context_left])
    ctx_right_slope = np.array([item[1] for item in context_right])
    ctx_left_y -= ctx_left_y.mean()
    ctx_right_y -= ctx_right_y.mean()

    scores = []
    for run in bottom_runs:
        left = [_signature(mask, *run, "left") for mask in bottom_masks]
        right = [_signature(mask, *run, "right") for mask in bottom_masks]
        left_y = np.array([item[0] for item in left])
        right_y = np.array([item[0] for item in right])
        left_slope = np.array([item[1] for item in left])
        right_slope = np.array([item[1] for item in right])
        left_y -= left_y.mean()
        right_y -= right_y.mean()
        position_cost = np.abs(left_y - ctx_left_y).sum() + np.abs(right_y - ctx_right_y).sum()
        tangent_cost = np.abs(left_slope - ctx_left_slope).sum() + np.abs(right_slope - ctx_right_slope).sum()
        scores.append(float(position_cost + 24.0 * tangent_cost))

    order = np.argsort(np.asarray(scores))
    if not np.isfinite(scores).all() or scores[int(order[1])] - scores[int(order[0])] < 15.0:
        raise ValueError("ambiguous seam completion")
    return "ABC"[int(order[0])]
