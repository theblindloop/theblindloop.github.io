"""Deterministic Pillow renderer and analytic arm for contour seam completion."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw, ImageFont


RGB_CHOICES = [
    (222, 61, 69),
    (43, 157, 91),
    (48, 102, 211),
]
LOGICAL_COLORS = ("coral", "leaf", "blue")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    width, height = 900, 650

    rgb_values = RGB_CHOICES[:]
    rng.shuffle(rgb_values)
    palette = {
        name: list(rgb) for name, rgb in zip(LOGICAL_COLORS, rgb_values, strict=True)
    }

    lane_jitter = [rng.randint(-5, 5) for _ in range(3)]
    left_heights = [42 + lane_jitter[0], 90 + lane_jitter[1], 138 + lane_jitter[2]]
    right_heights = [
        left_heights[0] + rng.randint(-8, 8),
        left_heights[1] + rng.randint(-8, 8),
        left_heights[2] + rng.randint(-8, 8),
    ]
    slope_choices = (-0.30, -0.20, 0.20, 0.30)
    contours = []
    for index, name in enumerate(LOGICAL_COLORS):
        contours.append(
            {
                "color": name,
                "y_left": left_heights[index],
                "y_right": right_heights[index],
                "slope_left": rng.choice(slope_choices),
                "slope_right": rng.choice(slope_choices),
                "outer_offset": rng.randint(-6, 6),
            }
        )

    position_left_delta = [14, -10, -4]
    position_right_delta = [-12, 15, -3]
    rng.shuffle(position_left_delta)
    rng.shuffle(position_right_delta)
    tangent_delta_left = [rng.choice((-0.48, 0.48)) for _ in range(3)]
    tangent_delta_right = [rng.choice((-0.48, 0.48)) for _ in range(3)]

    kinds = ("correct", "position_near_miss", "tangent_near_miss")
    correct_slot = int(seed) % 3
    remaining_slots = [slot for slot in range(3) if slot != correct_slot]
    if int(seed) % 2:
        remaining_slots.reverse()
    slots = {
        "correct": correct_slot,
        "position_near_miss": remaining_slots[0],
        "tangent_near_miss": remaining_slots[1],
    }
    candidate_specs = []
    for kind in kinds:
        curves = []
        for index, contour in enumerate(contours):
            curve = {
                "color": contour["color"],
                "y_left": contour["y_left"],
                "y_right": contour["y_right"],
                "slope_left": contour["slope_left"],
                "slope_right": contour["slope_right"],
                "mid_offset": rng.randint(-7, 7),
            }
            if kind == "position_near_miss":
                curve["y_left"] += position_left_delta[index]
                curve["y_right"] += position_right_delta[index]
            elif kind == "tangent_near_miss":
                curve["slope_left"] += tangent_delta_left[index]
                curve["slope_right"] += tangent_delta_right[index]
            curves.append(curve)
        candidate_specs.append({"kind": kind, "slot": slots[kind], "curves": curves})

    context_x = 62 + rng.randint(-7, 7)
    context_y = 68 + rng.randint(-5, 5)
    gap_left = 356 + rng.randint(-8, 8)
    gap_width = 188
    patch_y = 423 + rng.randint(-5, 5)
    patch_xs = [70 + rng.randint(-4, 4), 356 + rng.randint(-4, 4), 642 + rng.randint(-4, 4)]
    layout = {
        "context_box": [context_x, context_y, 776, 238],
        "gap_box": [gap_left, context_y + 29, gap_width, 180],
        "candidate_boxes": [[x, patch_y, gap_width, 180] for x in patch_xs],
    }

    return {
        "width": width,
        "height": height,
        "contours": contours,
        "candidate_specs": candidate_specs,
        "candidate_labels": [
            {"slot": slot, "label": "ABC"[slot]} for slot in range(3)
        ],
        "layout": layout,
        "line_width": 7,
        "palette": palette,
    }


def _font(size: int):
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _draw_polyline(draw: ImageDraw.ImageDraw, points: list[tuple[int, int]], fill, width: int):
    draw.line(points, fill=fill, width=width, joint="curve")


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    contours = scene["contours"]
    candidate_specs = scene["candidate_specs"]
    candidate_labels = scene["candidate_labels"]
    layout = scene["layout"]
    line_width = int(scene["line_width"])
    palette = scene["palette"]

    image = Image.new("RGB", (width, height), (250, 250, 247))
    draw = ImageDraw.Draw(image)
    ink = (33, 38, 47)
    muted = (112, 119, 128)
    paper = (255, 255, 255)

    cx, cy, cw, ch = [int(v) for v in layout["context_box"]]
    gx, gy, gw, gh = [int(v) for v in layout["gap_box"]]
    candidate_boxes = [[int(v) for v in box] for box in layout["candidate_boxes"]]

    draw.text((cx, cy - 35), "CONTEXT", fill=ink, font=_font(22))
    draw.rounded_rectangle((cx, cy, cx + cw, cy + ch), radius=12, fill=paper, outline=ink, width=3)
    draw.rectangle((gx, gy, gx + gw, gy + gh), fill=(243, 244, 246), outline=muted, width=3)
    draw.text((gx + 51, gy + 74), "MISSING", fill=(150, 154, 162), font=_font(16))

    for contour in contours:
        color = tuple(int(v) for v in palette[contour["color"]])
        y_left = gy + float(contour["y_left"])
        y_right = gy + float(contour["y_right"])
        slope_left = float(contour["slope_left"])
        slope_right = float(contour["slope_right"])
        outer = float(contour["outer_offset"])
        left_points = [
            (cx + 18, round(y_left - slope_left * 32 + outer)),
            (gx - 88, round(y_left - slope_left * 32 + outer / 2)),
            (gx - 32, round(y_left - slope_left * 32)),
            (gx, round(y_left)),
        ]
        right_points = [
            (gx + gw, round(y_right)),
            (gx + gw + 32, round(y_right + slope_right * 32)),
            (gx + gw + 88, round(y_right + slope_right * 32 + outer / 2)),
            (cx + cw - 18, round(y_right + slope_right * 32 + outer)),
        ]
        _draw_polyline(draw, left_points, color, line_width)
        _draw_polyline(draw, right_points, color, line_width)

    draw.text((cx + 4, cy + ch + 8), "Match color, seam height, and local direction.", fill=muted, font=_font(15))

    for box in candidate_boxes:
        x, y, w, h = box
        draw.rounded_rectangle((x, y, x + w, y + h), radius=9, fill=paper, outline=ink, width=3)

    for spec in candidate_specs:
        x, y, w, h = candidate_boxes[int(spec["slot"])]
        for curve in spec["curves"]:
            color = tuple(int(v) for v in palette[curve["color"]])
            y_left = y + float(curve["y_left"])
            y_right = y + float(curve["y_right"])
            slope_left = float(curve["slope_left"])
            slope_right = float(curve["slope_right"])
            points = [
                (x, round(y_left)),
                (x + 32, round(y_left + slope_left * 32)),
                (x + w // 2, round(y + (float(curve["y_left"]) + float(curve["y_right"])) / 2 + float(curve["mid_offset"]))),
                (x + w - 32, round(y_right - slope_right * 32)),
                (x + w, round(y_right)),
            ]
            _draw_polyline(draw, points, color, line_width)

    for entry in candidate_labels:
        slot = int(entry["slot"])
        label = str(entry["label"])
        x, y, w, _ = candidate_boxes[slot]
        draw.ellipse((x + w // 2 - 19, y - 47, x + w // 2 + 19, y - 9), fill=ink)
        draw.text((x + w // 2, y - 28), label, fill=paper, font=_font(24), anchor="mm")

    return image


def _scene_scores(scene: dict) -> list[tuple[float, str]]:
    contours = scene["contours"]
    candidate_specs = scene["candidate_specs"]
    candidate_labels = scene["candidate_labels"]
    label_by_slot = {int(item["slot"]): str(item["label"]) for item in candidate_labels}
    context_by_color = {item["color"]: item for item in contours}
    scores = []
    for spec in candidate_specs:
        position_cost = 0.0
        tangent_cost = 0.0
        for curve in spec["curves"]:
            context = context_by_color[curve["color"]]
            position_cost += abs(float(curve["y_left"]) - float(context["y_left"]))
            position_cost += abs(float(curve["y_right"]) - float(context["y_right"]))
            tangent_cost += abs(float(curve["slope_left"]) - float(context["slope_left"]))
            tangent_cost += abs(float(curve["slope_right"]) - float(context["slope_right"]))
        score = position_cost + 24.0 * tangent_cost
        scores.append((score, label_by_slot[int(spec["slot"])]))
    return scores


def analytic_gold(scene: dict) -> str:
    return min(_scene_scores(scene), key=lambda item: (item[0], item[1]))[1]


def margin(scene: dict) -> float:
    ordered = sorted(score for score, _ in _scene_scores(scene))
    return float(ordered[1] - ordered[0])


def is_quarantined(scene: dict) -> bool:
    ordered = sorted(score for score, _ in _scene_scores(scene))
    return bool(ordered[0] > 1e-9 or ordered[1] - ordered[0] < 45.0)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["candidate_specs"] = list(reversed(twin["candidate_specs"]))
    twin["candidate_labels"] = list(reversed(twin["candidate_labels"]))
    return [("reverse_slot_addressed_patch_records", twin)]
