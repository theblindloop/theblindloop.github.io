"""Independent pixel-only inverse for arrow-chain reflow."""

from __future__ import annotations

from PIL import Image

N = 8
CYAN = (18, 164, 190)
MAGENTA = (219, 42, 137)
YELLOW = (244, 187, 35)
TEMPLATES = {
    "anchor": ("...##...", "...##...", ".######.", "...##...", "#..##..#", ".#.##.#.", "..####..", "........"),
    "boot": ("..##....", "..##....", "..##....", "..##....", "..##....", "..####..", "..####..", "..######"),
    "chair": (".##.....", ".##.....", ".##.....", ".######.", ".######.", ".##..##.", ".##.....", "........"),
    "sail": ("...#....", "..##....", ".###....", "####....", "#####...", "...##...", "...#####", "...##..."),
}


def _near(pixel: tuple[int, int, int], target: tuple[int, int, int], tol: int = 8) -> bool:
    return max(abs(pixel[i] - target[i]) for i in range(3)) <= tol


def _colored_centroid(pixels, width: int, height: int, target):
    points = [(x, y) for y in range(height) for x in range(width) if _near(pixels[x, y], target)]
    if len(points) < 20:
        raise ValueError("required fiducial is missing")
    return sum(x for x, _ in points) / len(points), sum(y for _, y in points) / len(points)


def _decode(image: Image.Image) -> tuple[str, float]:
    rgb = image.convert("RGB")
    pixels = rgb.load()
    cyan_points = [(x, y) for y in range(rgb.height) for x in range(rgb.width) if _near(pixels[x, y], CYAN)]
    if len(cyan_points) < 500:
        raise ValueError("cyan frame or arrows missing")
    minx, maxx = min(x for x, _ in cyan_points), max(x for x, _ in cyan_points)
    miny, maxy = min(y for _, y in cyan_points), max(y for _, y in cyan_points)
    # The six-pixel outer cyan outline bounds the visible board plus a 12-pixel pad.
    x0, y0 = minx + 12, miny + 12
    span_x, span_y = maxx - minx - 24, maxy - miny - 24
    if abs(span_x - span_y) > 3 or not 420 <= span_x <= 440:
        raise ValueError("ambiguous board frame")
    cell = span_x / N

    mx, my = _colored_centroid(pixels, rgb.width, rgb.height, MAGENTA)
    yx, yy = _colored_centroid(pixels, rgb.width, rgb.height, YELLOW)
    start = (int((my - y0) // cell), int((mx - x0) // cell))
    end = (int((yy - y0) // cell), int((yx - x0) // cell))
    if any(not 0 <= value < N for point in (start, end) for value in point):
        raise ValueError("endpoint outside board")

    values = {}
    directions = {}
    for r in range(N):
        for c in range(N):
            cx = x0 + (c + 0.5) * cell
            cy = y0 + (r + 0.5) * cell
            samples = []
            for y in range(round(cy - 4), round(cy + 5)):
                for x in range(round(cx - 4), round(cx + 5)):
                    samples.append(sum(pixels[x, y]) / 3)
            values[(r, c)] = int(sum(samples) / len(samples) < 150)
            arrow = []
            for y in range(round(cy - 24), round(cy + 25)):
                for x in range(round(cx - 24), round(cx + 25)):
                    if _near(pixels[x, y], CYAN):
                        arrow.append((x, y))
            if arrow:
                ax = sum(x for x, _ in arrow) / len(arrow) - cx
                ay = sum(y for _, y in arrow) / len(arrow) - cy
                if abs(ax) > abs(ay):
                    directions[(r, c)] = (0, 1 if ax > 0 else -1)
                else:
                    directions[(r, c)] = (1 if ay > 0 else -1, 0)

    route = []
    current = start
    while True:
        if current in route:
            raise ValueError("arrow chain loops")
        route.append(current)
        if current == end:
            break
        if current not in directions:
            raise ValueError("missing successor arrow")
        dr, dc = directions[current]
        current = current[0] + dr, current[1] + dc
        if not (0 <= current[0] < N and 0 <= current[1] < N):
            raise ValueError("arrow leaves board")
    if len(route) != N * N:
        raise ValueError("chain does not visit all 64 cells")
    bits = [values[cell_pos] for cell_pos in route]
    scored = []
    for name, rows in TEMPLATES.items():
        template = [int(ch == "#") for row in rows for ch in row]
        scored.append((sum(a == b for a, b in zip(bits, template)), name))
    scored.sort(reverse=True)
    gap = scored[0][0] - scored[1][0]
    if gap < 6:
        raise ValueError("ambiguous symbol classification")
    return scored[0][1], float(gap)


def decision_from_image(image: Image.Image) -> str:
    return _decode(image)[0]
