"""Renderer for the arrow-chain reflow world."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

N = 8
CELL = 54
CANVAS = (560, 570)
CYAN = (18, 164, 190)
MAGENTA = (219, 42, 137)
YELLOW = (244, 187, 35)

TEMPLATES = {
    "anchor": (
        "...##...", "...##...", ".######.", "...##...",
        "#..##..#", ".#.##.#.", "..####..", "........",
    ),
    "boot": (
        "..##....", "..##....", "..##....", "..##....",
        "..##....", "..####..", "..####..", "..######",
    ),
    "chair": (
        ".##.....", ".##.....", ".##.....", ".######.",
        ".######.", ".##..##.", ".##.....", "........",
    ),
    "sail": (
        "...#....", "..##....", ".###....", "####....",
        "#####...", "...##...", "...#####", "...##...",
    ),
}
SYMBOLS = tuple(TEMPLATES)
DARKS = ((31, 37, 47), (52, 35, 57), (35, 49, 43), (48, 43, 31))
LIGHTS = ((235, 238, 240), (228, 235, 232), (239, 232, 237), (236, 234, 225))


def _base_route() -> list[tuple[int, int]]:
    return [(r, c) for r in range(N) for c in (range(N) if r % 2 == 0 else range(N - 1, -1, -1))]


def _route(variant: int) -> list[tuple[int, int]]:
    route = _base_route()
    transform = (variant // 2) % 8
    reverse = variant % 2
    out = []
    for r, c in route:
        operation = transform
        if operation >= 4:
            c = N - 1 - c
            operation -= 4
        for _ in range(operation):
            r, c = c, N - 1 - r
        out.append((r, c))
    return list(reversed(out)) if reverse else out


def _template_bits(symbol: str) -> list[int]:
    return [int(ch == "#") for row in TEMPLATES[symbol] for ch in row]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed ^ 0xA77C31)
    return {
        "symbol": SYMBOLS[seed % len(SYMBOLS)],
        "route_variant": (seed * 11 + 3) % 16,
        "dark_rgb": list(DARKS[rng.randrange(len(DARKS))]),
        "light_rgb": list(LIGHTS[rng.randrange(len(LIGHTS))]),
        "canvas_dx": rng.randrange(-10, 11),
        "canvas_dy": rng.randrange(-8, 9),
    }


def _board_origin(scene: dict) -> tuple[int, int]:
    return 64 + int(scene["canvas_dx"]), 65 + int(scene["canvas_dy"])


def render(scene: dict) -> Image.Image:
    symbol = scene["symbol"]
    route = _route(int(scene["route_variant"]))
    dark = tuple(scene["dark_rgb"])
    light = tuple(scene["light_rgb"])
    x0, y0 = _board_origin(scene)
    image = Image.new("RGB", CANVAS, (250, 249, 246))
    draw = ImageDraw.Draw(image)
    bits = _template_bits(symbol)
    values = {cell: bits[index] for index, cell in enumerate(route)}

    draw.rounded_rectangle((x0 - 12, y0 - 12, x0 + N * CELL + 12, y0 + N * CELL + 12), radius=9, fill=(245, 247, 246), outline=CYAN, width=6)
    for r in range(N):
        for c in range(N):
            left, top = x0 + c * CELL, y0 + r * CELL
            draw.rectangle((left, top, left + CELL, top + CELL), outline=(179, 184, 184), width=2)
            cx, cy = left + CELL // 2, top + CELL // 2
            fill = dark if values[(r, c)] else light
            draw.rounded_rectangle((cx - 6, cy - 6, cx + 6, cy + 6), radius=2, fill=fill, outline=(92, 96, 99), width=1)

    for index, (r, c) in enumerate(route[:-1]):
        nr, nc = route[index + 1]
        cx, cy = x0 + c * CELL + CELL // 2, y0 + r * CELL + CELL // 2
        dr, dc = nr - r, nc - c
        if dc == 1:
            polygon = ((cx + 22, cy), (cx + 11, cy - 7), (cx + 11, cy + 7))
        elif dc == -1:
            polygon = ((cx - 22, cy), (cx - 11, cy - 7), (cx - 11, cy + 7))
        elif dr == 1:
            polygon = ((cx, cy + 22), (cx - 7, cy + 11), (cx + 7, cy + 11))
        else:
            polygon = ((cx, cy - 22), (cx - 7, cy - 11), (cx + 7, cy - 11))
        draw.polygon(polygon, fill=CYAN)

    sr, sc = route[0]
    er, ec = route[-1]
    for (r, c), color in (((sr, sc), MAGENTA), ((er, ec), YELLOW)):
        cx, cy = x0 + c * CELL + 10, y0 + r * CELL + 10
        draw.ellipse((cx - 5, cy - 5, cx + 5, cy + 5), fill=color, outline=(75, 70, 70), width=1)
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def margin(scene: dict) -> float:
    target = _template_bits(scene["symbol"])
    scores = []
    for name in SYMBOLS:
        candidate = _template_bits(name)
        scores.append(sum(a == b for a, b in zip(target, candidate)))
    scores.sort(reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    route = _route(int(scene["route_variant"]))
    valid = len(route) == N * N and len(set(route)) == N * N and all(
        abs(a[0] - b[0]) + abs(a[1] - b[1]) == 1 for a, b in zip(route, route[1:])
    )
    return (not valid) or margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
