"""Deterministic latent sampler, renderer, and analytic replay."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 768, 460
BACKGROUND = (248, 250, 252)
INK = (35, 43, 55)
PANEL = (255, 255, 255)
REGISTERS = ((220, 68, 74), (42, 166, 101), (55, 112, 220))
ON = (245, 184, 28)
OFF = (145, 78, 190)


def _replay(initial: list[bool], operations: list[list[int]]) -> list[bool]:
    values = list(initial)
    for source, destination in operations:
        values[destination] = values[source]
    return values


def sample_scene(seed: int) -> dict:
    desired = (seed % 2) == 0
    for attempt in range(10000):
        rng = random.Random(seed * 10007 + attempt * 97)
        initial = [bool(rng.randrange(2)) for _ in range(3)]
        if len(set(initial)) < 2:
            continue
        operations = []
        for _ in range(6):
            source = rng.randrange(3)
            destination = rng.randrange(2)
            if destination >= source:
                destination += 1
            operations.append([source, destination])
        target = rng.randrange(3)
        if not any(destination == target for _, destination in operations):
            continue
        if _replay(initial, operations)[target] == desired:
            return {"initial": initial, "operations": operations, "target": target}
    raise RuntimeError("could not sample balanced scene")


def analytic_gold(scene: dict) -> str:
    values = list(scene["initial"])
    for source, destination in scene["operations"]:
        values[destination] = values[source]
    return "ON" if values[scene["target"]] else "OFF"


def margin(scene: dict) -> float:
    # Discrete glyph margin: changing the decision requires changing at least one
    # full state lamp or one complete register chip on a causally relevant card.
    initial = scene["initial"]
    operations = scene["operations"]
    target = scene["target"]
    return 24.0 if len(initial) == 3 and len(operations) == 6 and 0 <= target < 3 else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        return not (
            len(scene["initial"]) == 3
            and len(scene["operations"]) == 6
            and 0 <= scene["target"] < 3
            and all(len(op) == 2 and op[0] != op[1] and 0 <= op[0] < 3 and 0 <= op[1] < 3 for op in scene["operations"])
        )
    except (KeyError, TypeError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _circle(draw: ImageDraw.ImageDraw, xy: tuple[int, int], radius: int, fill, outline=INK, width=3):
    x, y = xy
    draw.ellipse((x-radius, y-radius, x+radius, y+radius), fill=fill, outline=outline, width=width)


def render(scene: dict) -> Image.Image:
    initial = scene["initial"]
    operations = scene["operations"]
    target = scene["target"]
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    draw.rounded_rectangle((28, 28, 740, 193), radius=18, fill=PANEL, outline=INK, width=3)
    draw.text((48, 43), "STARTING REGISTERS", fill=INK, font=font)
    start_xs = (150, 384, 618)
    for index, x in enumerate(start_xs):
        _circle(draw, (x, 92), 18, REGISTERS[index])
        _circle(draw, (x, 148), 15, ON if initial[index] else OFF)
        state = "ON" if initial[index] else "OFF"
        draw.text((x - 10, 171), state, fill=INK, font=font)

    draw.rounded_rectangle((28, 210, 740, 350), radius=18, fill=PANEL, outline=INK, width=3)
    draw.text((48, 224), "COPY LEFT TO RIGHT  (READ CARDS LEFT TO RIGHT)", fill=INK, font=font)
    card_centers = (82, 203, 324, 445, 566, 687)
    for number, (source, destination) in enumerate(operations):
        cx = card_centers[number]
        draw.rounded_rectangle((cx-50, 251, cx+50, 330), radius=10, fill=(242, 246, 250), outline=(135, 145, 158), width=2)
        _circle(draw, (cx-22, 287), 11, REGISTERS[source], width=2)
        _circle(draw, (cx+22, 287), 11, REGISTERS[destination], width=2)
        draw.line((cx-8, 287, cx+7, 287), fill=INK, width=4)
        draw.polygon(((cx+9, 287), (cx+3, 282), (cx+3, 292)), fill=INK)
        draw.text((cx-4, 309), str(number+1), fill=INK, font=font)

    draw.rounded_rectangle((248, 372, 520, 443), radius=16, fill=PANEL, outline=INK, width=3)
    draw.text((278, 397), "QUERY", fill=INK, font=font)
    _circle(draw, (448, 407), 18, REGISTERS[target])
    return image
