"""Independent image-only inverse arm for signal-copy replay."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

REGISTER_COLORS = ((220, 68, 74), (42, 166, 101), (55, 112, 220))
STATE_COLORS = {True: (245, 184, 28), False: (145, 78, 190)}


def _components(mask: np.ndarray, minimum: int = 80) -> list[tuple[float, float, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = [(int(y), int(x))]
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.pop()
            points.append((cy, cx))
            for ny, nx in ((cy-1,cx),(cy+1,cx),(cy,cx-1),(cy,cx+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            result.append((sum(p[1] for p in points)/len(points), sum(p[0] for p in points)/len(points), len(points)))
    return result


def _mask_near(arr: np.ndarray, color: tuple[int, int, int], radius: float = 18.0) -> np.ndarray:
    # int32 avoids overflow when squaring channel differences near 255.
    delta = arr.astype(np.int32) - np.array(color, dtype=np.int32)
    return np.sum(delta * delta, axis=2) <= radius * radius


def _cluster_y(items: list[tuple[float, float, int, int]]) -> list[list[tuple[float, float, int, int]]]:
    groups: list[list[tuple[float, float, int, int]]] = []
    for item in sorted(items, key=lambda value: value[1]):
        if not groups or item[1] - sum(v[1] for v in groups[-1]) / len(groups[-1]) > 30:
            groups.append([item])
        else:
            groups[-1].append(item)
    return groups


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    chips = []
    for identity, color in enumerate(REGISTER_COLORS):
        for x, y, area in _components(_mask_near(arr, color), 180):
            chips.append((x, y, area, identity))
    bands = _cluster_y(chips)
    if len(bands) != 3 or sorted(len(band) for band in bands) != [1, 3, 12]:
        raise ValueError("register-chip structure is unreadable")
    top, middle, bottom = sorted(bands, key=lambda band: sum(v[1] for v in band)/len(band))
    if len(top) != 3 or len(middle) != 12 or len(bottom) != 1:
        raise ValueError("unexpected register bands")

    state_marks = []
    for state, color in STATE_COLORS.items():
        for x, y, area in _components(_mask_near(arr, color), 180):
            state_marks.append((x, y, state))
    if len(state_marks) != 3:
        raise ValueError("initial states are unreadable")

    top = sorted(top, key=lambda item: item[0])
    state_marks.sort(key=lambda item: item[0])
    values = [False, False, False]
    for chip, mark in zip(top, state_marks):
        if abs(chip[0] - mark[0]) > 20:
            raise ValueError("state-to-register alignment is ambiguous")
        values[chip[3]] = mark[2]

    ordered = sorted(middle, key=lambda item: item[0])
    for index in range(0, 12, 2):
        left, right = ordered[index], ordered[index+1]
        if right[0] - left[0] > 55:
            raise ValueError("copy-card pairing is ambiguous")
        values[right[3]] = values[left[3]]
    target = bottom[0][3]
    return "ON" if values[target] else "OFF"
