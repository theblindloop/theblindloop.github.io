"""Cut strand versus rung frame: pieces on one side, enclosed loops on the other.

One panel holds a single black strand that snakes across the panel and is
interrupted by visible cuts, so it falls into (cuts + 1) separate pieces. The
other panel holds a black frame of two long rails crossed by rungs; a rung that
reaches both rails binds them and closes a loop, a rung that stops short binds
nothing. The scene decision compares the number of strand pieces with the
number of enclosed loops. Neither panel decides anything alone.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = (960, 540)
WHITE = (255, 255, 255)
INK = (18, 18, 20)
DECOR = (168, 168, 172)

STROKE = 9
CUT_GAP = 16
CORD_ROW_GAP = 62
QUARANTINE_PX = 6.0
DECISIONS = ("pieces", "loops", "equal")


def _counts(rng: random.Random, decision: str) -> tuple[int, int]:
    """Pieces and loops with the requested order and a one-step boundary."""
    if decision == "equal":
        n = rng.randint(2, 5)
        return n, n
    if decision == "pieces":
        loops = rng.randint(1, 4)
        return loops + 1, loops
    pieces = rng.randint(2, 5)
    return pieces, pieces + 1


def sample_scene(seed: int) -> dict:
    rng = random.Random(0x5721 ^ (seed * 2654435761 % (1 << 32)))
    decision = DECISIONS[seed % 3]
    pieces, loops = _counts(rng, decision)

    # Side placement is drawn from an independent stream, so page side never
    # tracks the answer.
    strand_left = rng.random() < 0.5

    left_box = (46, 74, 428, 470)
    right_box = (532, 74, 914, 470)
    strand_box = left_box if strand_left else right_box
    frame_box = right_box if strand_left else left_box

    # Stroke width is an independent nuisance stream: it sets the weakest
    # decisive clearance of the page, so the dataset carries near-boundary and
    # quarantined scenes without ever touching the counts that decide the page.
    tier = rng.random()
    stroke = 5 if tier < 0.09 else 6 if tier < 0.22 else rng.choice((7, 8, 9, 10, 11))
    gap = rng.choice((14, 16, 20))

    rows = rng.randint(max(3, (pieces + 2) // 2), 5)
    x0, y0, x1, _y1 = strand_box
    inset = rng.choice((0, 10, 20))
    strand = {
        "box": [x0 + inset, y0 + rng.choice((0, 12)), x1 - inset],
        "rows": rows,
        "row_gap": CORD_ROW_GAP,
        "stroke": stroke,
        "cuts": _cuts(rng, pieces - 1, rows),
        "gap": gap,
    }

    fx0, fy0, fx1, fy1 = frame_box
    rails = loops + 1
    stubs = rng.randint(1, 3)
    span = rng.choice((300, 330, 360))
    frame = {
        "box": [fx0 + rng.choice((6, 18, 30)), fy0 + rng.choice((0, 14)), fx0 + rng.choice((6, 18, 30)) + span],
        "height": rng.choice((330, 360, 390)),
        "stroke": stroke,
        "rungs": rails,
        "stubs": stubs,
        "stub_clear": rng.choice((26, 34, 42)),
        "slots": rails + stubs,
    }
    frame["box"][2] = min(frame["box"][2], fx1)
    frame["height"] = min(frame["height"], fy1 - frame["box"][1])

    decor = []
    for i in range(rng.randint(4, 7)):
        w = rng.randint(26, 54)
        h = rng.randint(18, 44)
        cx = rng.choice((452, 470, 490, 508))
        cy = rng.randint(96, 452)
        decor.append(["ring" if i % 2 else "bar", cx - w // 2, cy - h // 2, w, h])
    for i in range(rng.randint(2, 4)):
        w = rng.randint(30, 70)
        cx = rng.randint(70, 880)
        cy = rng.choice((26, 40, 500, 514))
        decor.append(["bar", cx, cy, w, 14])

    return {
        "canvas": list(CANVAS),
        "strand": strand,
        "frame": frame,
        "decor": decor,
        "strand_left": strand_left,
    }


def _cuts(rng: random.Random, count: int, rows: int) -> list[list[float]]:
    """Distinct (row, fraction) cut sites kept clear of the strand bends."""
    slots: list[tuple[int, float]] = []
    for row in range(rows):
        for frac in (0.26, 0.42, 0.58, 0.74):
            slots.append((row, frac))
    rng.shuffle(slots)
    chosen: list[tuple[int, float]] = []
    for row, frac in slots:
        if len(chosen) == count:
            break
        if any(r == row and abs(f - frac) < 0.15 for r, f in chosen):
            continue
        chosen.append((row, frac))
    chosen.sort()
    return [[r, f] for r, f in chosen]


def _strand_geometry(scene: dict) -> tuple[list[tuple[int, int, int, int]], list[tuple[int, int, int, int]]]:
    s = scene["strand"]
    xa, ytop, xb = s["box"]
    t = s["stroke"]
    half = t // 2
    bars: list[tuple[int, int, int, int]] = []
    for row in range(s["rows"]):
        y = ytop + row * s["row_gap"]
        bars.append((xa, y - half, xb, y + half))
        if row + 1 < s["rows"]:
            y2 = ytop + (row + 1) * s["row_gap"]
            x = xb if row % 2 == 0 else xa
            bars.append((x - half, y - half, x + half, y2 + half))
    cuts: list[tuple[int, int, int, int]] = []
    for row, frac in s["cuts"]:
        y = ytop + int(row) * s["row_gap"]
        cx = xa + int(round((xb - xa) * frac))
        g = s["gap"]
        cuts.append((cx - g // 2, y - half - 3, cx + g // 2, y + half + 3))
    return bars, cuts


def _frame_geometry(scene: dict) -> list[tuple[int, int, int, int]]:
    f = scene["frame"]
    xa, ytop, xb = f["box"]
    h = f["height"]
    t = f["stroke"]
    bars = [(xa, ytop, xa + t, ytop + h), (xb - t, ytop, xb, ytop + h)]
    step = h / (f["slots"] + 1)
    kinds = ["rung"] * f["rungs"] + ["stub"] * f["stubs"]
    # Complete rungs and stubs alternate deterministically from the scene, and
    # the stub tips never reach the far rail.
    order: list[str] = []
    stub_positions = {1 + 2 * i for i in range(f["stubs"])}
    si = 0
    ri = 0
    for k in range(f["slots"]):
        if k in stub_positions and si < f["stubs"] and ri >= 1:
            order.append("stub")
            si += 1
        elif ri < f["rungs"]:
            order.append("rung")
            ri += 1
        else:
            order.append("stub")
            si += 1
    del kinds
    for k, kind in enumerate(order):
        y = int(ytop + step * (k + 1))
        if kind == "rung":
            bars.append((xa, y - t // 2, xb, y + t // 2))
        else:
            clear = f["stub_clear"]
            if k % 4 == 1:
                bars.append((xa, y - t // 2, xb - t - clear, y + t // 2))
            else:
                bars.append((xa + t + clear, y - t // 2, xb, y + t // 2))
    return bars


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", tuple(scene["canvas"]), WHITE)
    draw = ImageDraw.Draw(img)

    for kind, x, y, w, h in scene["decor"]:
        if kind == "ring":
            draw.rectangle([x, y, x + w, y + h], outline=DECOR, width=5)
        else:
            draw.rectangle([x, y, x + w, y + h], fill=DECOR)

    for bar in _frame_geometry(scene):
        draw.rectangle(list(bar), fill=INK)
    bars, cuts = _strand_geometry(scene)
    for bar in bars:
        draw.rectangle(list(bar), fill=INK)
    for cut in cuts:
        draw.rectangle(list(cut), fill=WHITE)
    return img


def piece_count(scene: dict) -> int:
    return len(scene["strand"]["cuts"]) + 1


def loop_count(scene: dict) -> int:
    return scene["frame"]["rungs"] - 1


def analytic_gold(scene: dict) -> str:
    p = piece_count(scene)
    loops = loop_count(scene)
    if p == loops:
        return "equal"
    return "pieces" if p > loops else "loops"


def margin(scene: dict) -> float:
    """Smallest decisive clearance in pixels of the whole page."""
    s = scene["strand"]
    f = scene["frame"]
    xa, _ytop, xb = s["box"]
    widths = [
        float(s["stroke"]),
        float(s["gap"]),
        float(s["row_gap"] - s["stroke"]),
        float(f["stroke"]),
        float(f["stub_clear"]),
        float(f["height"] / (f["slots"] + 1) - f["stroke"]),
    ]
    # Cuts must stay clear of the strand bends and of each other.
    for row, frac in s["cuts"]:
        cx = xa + (xb - xa) * frac
        widths.append(float(min(cx - xa, xb - cx) - s["gap"] / 2 - s["stroke"]))
    for i, (r1, f1) in enumerate(s["cuts"]):
        for r2, f2 in s["cuts"][i + 1:]:
            if r1 == r2:
                widths.append(abs(f1 - f2) * (xb - xa) - s["gap"])
    return round(min(widths), 3)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def runner_up(scene: dict) -> str:
    """Nearest competing decision, one cut or one complete bar away."""
    p = piece_count(scene)
    loops = loop_count(scene)
    if p == loops:
        return "pieces"
    return "equal"
