"""Render three converted flow lines with batches merging at each stage."""
from __future__ import annotations
from fractions import Fraction
import math, random
from PIL import Image, ImageDraw

CANVAS = (1200, 760)
ANSWERS = ("amber", "cyan", "violet")
RGB = {"background": (255,255,255), "panel": (247,249,252), "ink": (31,38,46),
       "card": (232,237,242), "add": (255,246,210), "amber": (226,151,28),
       "cyan": (18,157,181), "violet": (141,77,190)}
ROW_Y, ROW_H = (25,270,515), 220
STAGE_X, RATE_X = (105,600,1095), (350,850)
DIGIT_SEGMENTS = {0:"abcdef",1:"bc",2:"abdeg",3:"abcdg",4:"bcfg",5:"acdfg",6:"acdefg",7:"abc",8:"abcdefg",9:"abcdfg"}
BLOCK_FONT = {
 "M":("10001","11011","10101","10101","10001","10001","10001"),
 "A":("01110","10001","10001","11111","10001","10001","10001"),
 "X":("10001","10001","01010","00100","01010","10001","10001"),
 "D":("11110","10001","10001","10001","10001","10001","11110")}
CONVERSIONS = ((2,1),(3,2),(4,3),(5,4),(3,1),(5,3),(4,1),(5,2))

def _loads(adds: list[int], rates: list[list[int]]) -> tuple[Fraction, Fraction, Fraction]:
    if len(adds) != 3 or len(rates) != 2: raise ValueError("bad line")
    a = Fraction(int(adds[0]))
    b = a * Fraction(int(rates[0][1]), int(rates[0][0])) + int(adds[1])
    c = b * Fraction(int(rates[1][1]), int(rates[1][0])) + int(adds[2])
    return a,b,c

def _feasible(adds: list[int], caps: list[int], rates: list[list[int]]) -> bool:
    return all(load <= cap for load,cap in zip(_loads(adds,rates),caps))

def _make_line(rng: random.Random, failure_stage: int | None) -> dict:
    for _ in range(1000):
        rates = [list(rng.choice(CONVERSIONS)),list(rng.choice(CONVERSIONS))]
        adds = [rng.randrange(14,37) for _ in range(3)]
        loads = _loads(adds,rates)
        if max(loads) > 82: continue
        caps = [math.ceil(float(load))+rng.randrange(5,12) for load in loads]
        if failure_stage is not None: caps[failure_stage] = math.floor(float(loads[failure_stage]))-rng.randrange(3,8)
        if min(caps)<10 or max(caps)>99: continue
        status = [load<=cap for load,cap in zip(loads,caps)]
        if (failure_stage is None and all(status)) or (failure_stage is not None and status.count(False)==1 and not status[failure_stage]):
            return {"adds":adds,"capacities":caps,"conversions":rates}
    raise RuntimeError("could not make line")

def sample_scene(seed: int) -> dict:
    rng = random.Random((int(seed)^0xA6610AD)&0xFFFFFFFFFFFFFFFF)
    kinds: list[int|None] = [None,1,2]; rng.shuffle(kinds)
    lines = [_make_line(rng,k) for k in kinds]
    winner = kinds.index(None); answer = ANSWERS[(int(seed)//3)%3]
    rest = [c for c in ANSWERS if c!=answer]; rng.shuffle(rest)
    colors: list[str|None] = [None,None,None]; colors[winner]=answer
    for i in range(3):
        if colors[i] is None: colors[i]=rest.pop()
    scene = {"adds":[x["adds"] for x in lines],"capacities":[x["capacities"] for x in lines],
             "conversions":[x["conversions"] for x in lines],"tag_colors":colors}
    if is_quarantined(scene): raise RuntimeError("constructed quarantine")
    return scene

def _draw_digit(draw: ImageDraw.ImageDraw,digit:int,x:int,y:int)->None:
    boxes={"a":(x+4,y,x+19,y+4),"b":(x+19,y+4,x+23,y+21),"c":(x+19,y+23,x+23,y+40),
           "d":(x+4,y+40,x+19,y+44),"e":(x,y+23,x+4,y+40),"f":(x,y+4,x+4,y+21),"g":(x+4,y+20,x+19,y+24)}
    for s in DIGIT_SEGMENTS[int(digit)]: draw.rectangle(boxes[s],fill=RGB["ink"])

def _draw_number(draw:ImageDraw.ImageDraw,value:int,cx:int,y:int)->None:
    if not 10<=int(value)<=99: raise ValueError("two digits required")
    a,b=divmod(int(value),10); _draw_digit(draw,a,cx-27,y); _draw_digit(draw,b,cx+4,y)

def _draw_rate(draw:ImageDraw.ImageDraw,cx:int,cy:int,pair:list[int])->None:
    draw.rounded_rectangle((cx-61,cy-36,cx+61,cy+36),radius=11,fill=RGB["background"],outline=RGB["ink"],width=4)
    y=cy-22; _draw_digit(draw,int(pair[0]),cx-37,y)
    draw.ellipse((cx-5,cy-13,cx+3,cy-5),fill=RGB["ink"]); draw.ellipse((cx-5,cy+5,cx+3,cy+13),fill=RGB["ink"])
    _draw_digit(draw,int(pair[1]),cx+12,y)

def _block_text(draw:ImageDraw.ImageDraw,text:str,cx:int,top:int,scale:int=2)->None:
    widths=[len(BLOCK_FONT[ch][0])*scale for ch in text]; x=cx-(sum(widths)+(len(text)-1)*2*scale)//2
    for ch,width in zip(text,widths):
        for row,bits in enumerate(BLOCK_FONT[ch]):
            for col,bit in enumerate(bits):
                if bit=="1": draw.rectangle((x+col*scale,top+row*scale,x+(col+1)*scale-1,top+(row+1)*scale-1),fill=RGB["ink"])
        x+=width+2*scale

def render(scene:dict)->Image.Image:
    image=Image.new("RGB",CANVAS,RGB["background"]); draw=ImageDraw.Draw(image)
    for y,adds,caps,rates,color in zip(ROW_Y,scene["adds"],scene["capacities"],scene["conversions"],scene["tag_colors"]):
        draw.rounded_rectangle((18,y,1182,y+ROW_H),radius=22,fill=RGB["panel"],outline=RGB[str(color)],width=10); cy=y+108
        for edge in range(2):
            left,right=STAGE_X[edge]+82,STAGE_X[edge+1]-82; draw.line((left,cy,right,cy),fill=RGB["ink"],width=8)
            draw.polygon(((right,cy),(right-22,cy-16),(right-22,cy+16)),fill=RGB["ink"])
        for cx,cap,add in zip(STAGE_X,caps,adds):
            draw.rounded_rectangle((cx-76,y+16,cx+76,y+204),radius=14,fill=RGB["card"],outline=RGB["ink"],width=5)
            _block_text(draw,"MAX",cx,y+27); _draw_number(draw,int(cap),cx,y+48)
            draw.rounded_rectangle((cx-66,y+104,cx+66,y+194),radius=10,fill=RGB["add"],outline=RGB["ink"],width=3)
            _block_text(draw,"ADD",cx,y+112); _draw_number(draw,int(add),cx,y+139)
        for cx,pair in zip(RATE_X,rates): _draw_rate(draw,cx,cy,pair)
    return image

def analytic_gold(scene:dict)->str:
    if len(scene["adds"])!=3 or len(scene["capacities"])!=3 or len(scene["conversions"])!=3 or set(scene["tag_colors"])!=set(ANSWERS): raise ValueError("malformed")
    ok=[_feasible(a,c,r) for a,c,r in zip(scene["adds"],scene["capacities"],scene["conversions"])]
    hits=[i for i,v in enumerate(ok) if v]
    if len(hits)!=1: raise ValueError("not unique")
    return str(scene["tag_colors"][hits[0]])

def margin(scene:dict)->float:
    gaps=[]
    for adds,caps,rates in zip(scene["adds"],scene["capacities"],scene["conversions"]):
        gaps.extend(abs(Fraction(cap)-load)/cap for load,cap in zip(_loads(adds,rates),caps))
    return float(min(gaps)) if gaps else 0.0

def is_quarantined(scene:dict)->bool:
    try:
        if any(len(v)!=3 for v in (scene["adds"],scene["capacities"],scene["conversions"],scene["tag_colors"])): return True
        if any(not all(10<=int(v)<=99 for v in adds+caps) for adds,caps in zip(scene["adds"],scene["capacities"])): return True
        return analytic_gold(scene) not in ANSWERS or margin(scene)<0.025
    except (KeyError,TypeError,ValueError,ZeroDivisionError): return True

def latent_symmetries(scene:dict)->list[tuple[str,dict]]: return []
