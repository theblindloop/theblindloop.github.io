"""Independent image-only inverse for merged load feasibility."""
from __future__ import annotations
from fractions import Fraction
import numpy as np
from PIL import Image

CANVAS=(1200,760); INK=(31,38,46); BACKGROUND=(255,255,255); PANEL=(247,249,252); CARD=(232,237,242); ADD=(255,246,210)
TAG_RGB={"amber":(226,151,28),"cyan":(18,157,181),"violet":(141,77,190)}
ALLOWED={INK,BACKGROUND,PANEL,CARD,ADD,*TAG_RGB.values()}; ROW_Y=(25,270,515); STAGE_X=(105,600,1095); RATE_X=(350,850)
SEGMENT_POINTS={"a":(11,2),"b":(21,12),"c":(21,32),"d":(11,42),"e":(2,32),"f":(2,12),"g":(11,22)}
SEGMENT_DIGIT={frozenset("abcdef"):0,frozenset("bc"):1,frozenset("abdeg"):2,frozenset("abcdg"):3,
 frozenset("bcfg"):4,frozenset("acdfg"):5,frozenset("acdefg"):6,frozenset("abc"):7,
 frozenset("abcdefg"):8,frozenset("abcdfg"):9}

class OracleError(ValueError): pass

def _digit(arr:np.ndarray,x:int,y:int)->int:
    active=set()
    for name,(dx,dy) in SEGMENT_POINTS.items():
        patch=arr[y+dy-1:y+dy+2,x+dx-1:x+dx+2]
        if patch.shape!=(3,3,3): raise OracleError("digit outside canvas")
        count=int(np.all(patch==INK,axis=2).sum())
        if count>=7: active.add(name)
        elif count!=0: raise OracleError("damaged digit")
    try: return SEGMENT_DIGIT[frozenset(active)]
    except KeyError as exc: raise OracleError(f"invalid digit {sorted(active)}") from exc

def _number(arr:np.ndarray,cx:int,y:int)->int:
    value=10*_digit(arr,cx-27,y)+_digit(arr,cx+4,y)
    if not 10<=value<=99: raise OracleError("number outside range")
    return value

def _stage(arr:np.ndarray,cx:int,y:int)->tuple[int,int]:
    card=arr[y+20:y+100,cx-70:cx+71]
    add_card=arr[y+108:y+193,cx-62:cx+63]
    if int(np.all(card==CARD,axis=2).sum())<5500: raise OracleError("MAX card damaged")
    if int(np.all(add_card==ADD,axis=2).sum())<4500: raise OracleError("ADD card damaged")
    return _number(arr,cx,y+48),_number(arr,cx,y+139)

def _rate(arr:np.ndarray,cx:int,cy:int)->tuple[int,int]:
    y=cy-22; p,q=_digit(arr,cx-37,y),_digit(arr,cx+12,y)
    if p<2 or q<1 or p<=q: raise OracleError("invalid conversion")
    for yy in (cy-9,cy+9):
        if not np.all(arr[yy-2:yy+3,cx-3:cx+2]==INK): raise OracleError("missing colon")
    return p,q

def _route_color(arr:np.ndarray,y:int)->str:
    counts={name:int(np.all(arr[y:y+12,200:1000]==rgb,axis=2).sum()) for name,rgb in TAG_RGB.items()}
    names=[name for name,count in counts.items() if count>6000]
    if len(names)!=1: raise OracleError("ambiguous border")
    return names[0]

def _validate_arrows(arr:np.ndarray,cy:int)->None:
    for x0,x1 in ((190,282),(418,515),(685,782),(918,1010)):
        if int(np.all(arr[cy-2:cy+3,x0:x1]==INK,axis=2).sum())<350: raise OracleError("broken arrow")

def decision_from_image(image:Image.Image)->str:
    rgb=image.convert("RGB")
    if rgb.size!=CANVAS: raise OracleError("wrong canvas")
    colors=rgb.getcolors(maxcolors=64)
    if colors is None or not {tuple(color) for _,color in colors}<=ALLOWED: raise OracleError("unknown raster color")
    arr=np.asarray(rgb); colors_seen=[]; feasible=[]
    for y in ROW_Y:
        cy=y+108; colors_seen.append(_route_color(arr,y)); _validate_arrows(arr,cy)
        stages=[_stage(arr,cx,y) for cx in STAGE_X]
        caps=[item[0] for item in stages]; adds=[item[1] for item in stages]
        rates=[_rate(arr,cx,cy) for cx in RATE_X]
        first=Fraction(adds[0]); middle=first*Fraction(rates[0][1],rates[0][0])+adds[1]
        finish=middle*Fraction(rates[1][1],rates[1][0])+adds[2]
        feasible.append(all(load<=cap for load,cap in zip((first,middle,finish),caps)))
    if set(colors_seen)!=set(TAG_RGB): raise OracleError("colors are not a permutation")
    hits=[i for i,ok in enumerate(feasible) if ok]
    if len(hits)!=1: raise OracleError("not exactly one feasible line")
    return colors_seen[hits[0]]
