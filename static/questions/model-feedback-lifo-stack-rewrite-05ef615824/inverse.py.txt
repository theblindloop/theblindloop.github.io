"""Independent image-only inverse for the LIFO color-stack board."""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH = 1560
HEIGHT = 1080
COLORS = np.array(
    [
        (224, 67, 76),
        (239, 151, 38),
        (14, 174, 205),
        (135, 88, 207),
    ],
    dtype=np.int16,
)
NAMES = ("R", "A", "C", "V")
START_X = (125, 245, 365, 485)
START_Y = 285
EVENT_X = (120, 285, 450, 615, 780, 945, 1110, 1275, 1440)
EVENT_Y = 650


def _color_at(arr: np.ndarray, x: int, y: int) -> int:
    # The sample point is well inside every large colored tile, away from its
    # white letter and dark outline.
    rgb = arr[y - 15, x - 15]
    distances = ((COLORS.astype(np.int32) - rgb.astype(np.int32)) ** 2).sum(axis=1)
    index = int(np.argmin(distances))
    if int(distances[index]) > 2200:
        raise ValueError("colored tile is missing or unreadable")
    return index


def decision_from_image(image: Image.Image) -> str:
    """Recover the terminal top pair from raster pixels only."""

    image = image.convert("RGB")
    if image.size != (WIDTH, HEIGHT):
        raise ValueError(f"unexpected board size {image.size}")
    arr = np.asarray(image, dtype=np.int16)
    stack = [_color_at(arr, x, START_Y) for x in START_X]
    if sorted(stack) != [0, 1, 2, 3]:
        raise ValueError("start stack is not four distinct colored tiles")
    for x in EVENT_X:
        color = _color_at(arr, x, EVENT_Y)
        if stack[-1] == color:
            if len(stack) <= 1:
                raise ValueError("event underflows the visible stack")
            stack.pop()
        else:
            stack.append(color)
    if len(stack) < 2 or stack[-1] == stack[-2]:
        raise ValueError("terminal pair is not a distinct ordered pair")
    return NAMES[stack[-1]] + NAMES[stack[-2]]
