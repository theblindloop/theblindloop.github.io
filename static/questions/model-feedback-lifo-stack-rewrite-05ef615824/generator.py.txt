"""Renderer and analytic state machine for a conditional LIFO color stack."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1560
HEIGHT = 1080
BACKGROUND = (247, 249, 252)
INK = (24, 31, 43)
MUTED = (87, 98, 115)
PANEL = (255, 255, 255)
LINE = (198, 207, 219)
STACK_COLORS = (
    (224, 67, 76),    # red
    (239, 151, 38),   # amber
    (14, 174, 205),  # cyan
    (135, 88, 207),  # violet
)
NAMES = ("R", "A", "C", "V")
WORDS = ("RED", "AMBER", "CYAN", "VIOLET")
PAIR_ORDER = tuple((top, below) for top in range(4) for below in range(4) if top != below)
PAIR_CODES = tuple(NAMES[top] + NAMES[below] for top, below in PAIR_ORDER)

START_CENTERS = (125, 245, 365, 485)
START_Y = 285
EVENT_X = (120, 285, 450, 615, 780, 945, 1110, 1275, 1440)
EVENT_Y = 650


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    path = (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
        if bold
        else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
    )
    try:
        return ImageFont.truetype(path, size=size)
    except OSError:
        return ImageFont.load_default()


def _validate_scene(scene: dict[str, Any]) -> tuple[tuple[int, ...], tuple[int, ...]]:
    try:
        start = tuple(int(value) for value in scene["start_stack"])
        events = tuple(int(value) for value in scene["events"])
    except (KeyError, TypeError, ValueError) as exc:
        raise ValueError("scene has invalid stack inputs") from exc
    if len(start) != 4 or sorted(start) != [0, 1, 2, 3]:
        raise ValueError("start_stack must contain each of the four colors once")
    if len(events) != 9 or any(value not in range(4) for value in events):
        raise ValueError("events must contain nine valid color values")
    return start, events


def _trace(start: tuple[int, ...], events: tuple[int, ...]) -> list[tuple[int, ...]]:
    states = [tuple(start)]
    stack = list(start)
    for color in events:
        if stack[-1] == color:
            if len(stack) <= 1:
                raise ValueError("event would underflow the stack")
            stack.pop()
        else:
            stack.append(color)
        states.append(tuple(stack))
    return states


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample balanced traces, targeting each ordered terminal top pair."""

    seed = int(seed)
    target = seed % len(PAIR_ORDER)
    for attempt in range(30000):
        rng = random.Random(seed * 1_000_003 + attempt * 91_879 + 73)
        start = tuple(rng.sample(range(4), 4))
        events = tuple(rng.randrange(4) for _ in range(9))
        try:
            path = _trace(start, events)
        except ValueError:
            continue
        pushes = sum(before[-1] != color for before, color in zip(path[:-1], events, strict=True))
        pops = len(events) - pushes
        final = path[-1]
        pair = (final[-1], final[-2]) if len(final) >= 2 else None
        if (
            pair == PAIR_ORDER[target]
            and all(len(state) >= 2 for state in path)
            and len(set(events)) == 4
            and 3 <= pushes <= 7
            and 2 <= pops <= 6
            and len({state for state in path}) >= 6
            and pair[0] != pair[1]
        ):
            return {"start_stack": list(start), "events": list(events)}
    raise RuntimeError(f"could not sample a balanced stack trace for seed {seed}")


def _panel(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], radius: int = 18) -> None:
    draw.rounded_rectangle(box, radius=radius, fill=PANEL, outline=LINE, width=2)


def _tile(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    value: int,
    half: int,
    *,
    text_size: int | None = None,
) -> None:
    cx, cy = center
    draw.rounded_rectangle(
        (cx - half, cy - half, cx + half, cy + half),
        radius=max(7, half // 4),
        fill=STACK_COLORS[value],
        outline=INK,
        width=3,
    )
    draw.text(
        (cx, cy + 1),
        NAMES[value],
        font=_font(text_size or max(14, half // 2), True),
        fill=(255, 255, 255),
        anchor="mm",
    )


def _down_arrow(draw: ImageDraw.ImageDraw, x: int, y: int) -> None:
    draw.line((x, y, x, y + 30), fill=INK, width=5)
    draw.polygon([(x, y + 40), (x - 11, y + 25), (x + 11, y + 25)], fill=INK)


def _key_cell(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    pair: tuple[int, int],
) -> None:
    cx, cy = center
    draw.rounded_rectangle(
        (cx - 168, cy - 27, cx + 168, cy + 27),
        radius=12,
        fill=(247, 249, 252),
        outline=LINE,
        width=2,
    )
    _tile(draw, (cx - 85, cy), pair[0], 19, text_size=15)
    _tile(draw, (cx - 35, cy), pair[1], 19, text_size=15)
    draw.text(
        (cx + 77, cy + 1),
        NAMES[pair[0]] + NAMES[pair[1]],
        font=_font(23, True),
        fill=INK,
        anchor="mm",
    )


def render(scene: dict[str, Any]) -> Image.Image:
    start, events = _validate_scene(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    draw.text((38, 22), "LIFO COLOR STACK", font=_font(31, True), fill=INK)
    draw.text(
        (40, 64),
        "Carry the changing stack through the numbered color cards.",
        font=_font(18),
        fill=MUTED,
    )

    _panel(draw, (35, 112, 550, 470))
    draw.text((60, 138), "START STACK", font=_font(21, True), fill=INK)
    draw.text((60, 176), "bottom → top, read left to right", font=_font(16), fill=MUTED)
    for x, value in zip(START_CENTERS, start, strict=True):
        _tile(draw, (x, START_Y), value, 45)
    for x, value in zip(START_CENTERS, start, strict=True):
        draw.text((x, 366), WORDS[value], font=_font(13, True), fill=INK, anchor="mm")
    _down_arrow(draw, START_CENTERS[-1], 208)
    draw.text((START_CENTERS[-1], 198), "TOP", font=_font(14, True), fill=INK, anchor="mb")
    draw.text((60, 414), "The rightmost tile is the current top.", font=_font(16, True), fill=INK)
    draw.text((60, 442), "A push adds a new tile above it.", font=_font(15), fill=MUTED)

    _panel(draw, (610, 112, 1525, 470))
    draw.text((638, 138), "STACK RULE", font=_font(21, True), fill=INK)
    draw.text(
        (638, 181),
        "At each card, compare its color with the CURRENT TOP.",
        font=_font(18, True),
        fill=INK,
    )
    draw.text(
        (638, 230),
        "MATCH  →  POP: remove the top tile.",
        font=_font(19, True),
        fill=INK,
    )
    draw.text(
        (638, 274),
        "MISS    →  PUSH: place a new card-colored tile on top.",
        font=_font(19, True),
        fill=INK,
    )
    draw.text(
        (638, 322),
        "A pop discards one tile; a push creates one.",
        font=_font(17),
        fill=MUTED,
    )
    draw.text(
        (638, 370),
        "After card 9, read the top tile first and the tile below it second.",
        font=_font(17, True),
        fill=INK,
    )
    for index, value in enumerate(range(4)):
        _tile(draw, (1190 + index * 70, 424), value, 24, text_size=16)
    draw.text((638, 424), "COLORS", font=_font(15, True), fill=MUTED, anchor="lm")

    _panel(draw, (35, 500, 1525, 775))
    draw.text((60, 524), "CARD ORDER", font=_font(21, True), fill=INK)
    draw.text(
        (250, 528),
        "read cards 1 through 9 from left to right",
        font=_font(17),
        fill=MUTED,
    )
    for number, (x, value) in enumerate(zip(EVENT_X, events, strict=True), start=1):
        draw.rounded_rectangle(
            (x - 68, 568, x + 68, 744),
            radius=14,
            fill=(247, 249, 252),
            outline=LINE,
            width=2,
        )
        draw.ellipse((x - 57, 581, x - 27, 611), fill=INK)
        draw.text((x - 42, 596), str(number), font=_font(13, True), fill=(255, 255, 255), anchor="mm")
        _tile(draw, (x, EVENT_Y), value, 31)
        draw.text((x, 713), WORDS[value], font=_font(12, True), fill=INK, anchor="mm")

    _panel(draw, (35, 805, 1525, 1055))
    draw.text((60, 829), "TOP-PAIR KEY", font=_font(21, True), fill=INK)
    draw.text(
        (285, 833),
        "top tile first, then the tile directly below it",
        font=_font(17),
        fill=MUTED,
    )
    key_x = (225, 600, 975, 1350)
    key_y = (900, 970, 1040)
    for index, pair in enumerate(PAIR_ORDER):
        _key_cell(draw, (key_x[index % 4], key_y[index // 4]), pair)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    start, events = _validate_scene(scene)
    final = _trace(start, events)[-1]
    if len(final) < 2 or final[-1] == final[-2]:
        raise ValueError("terminal top pair is not valid")
    return NAMES[final[-1]] + NAMES[final[-2]]


def margin(scene: dict[str, Any]) -> float:
    _validate_scene(scene)
    return 1.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    start, events = _validate_scene(scene)
    final = _trace(start, events)[-1]
    return len(final) < 2 or final[-1] == final[-2]


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    _validate_scene(scene)
    return []
