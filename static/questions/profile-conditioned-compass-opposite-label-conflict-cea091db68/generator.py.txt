"""Deterministic renderer and analytic arm for a shuffled compass dial."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw, ImageFont

FONT_NAME = "DejaVuSans-Bold.ttf"
LABELS = ("N", "E", "S", "W")
ANSWER_LABELS = ("S", "E", "W")
BACKGROUND_PALETTE = ("#fffdf8", "#f5f8fb", "#fbf7ef", "#f7fbf7")
RING_PALETTE = ("#27788a", "#4267a8", "#317a63", "#8064a2")
INK_PALETTE = ("#181d24", "#241c2d", "#172a25", "#29211a")
HUB_COLOR = "#e24434"


def _angular_error(a: float, b: float) -> float:
    return abs((a - b + 180.0) % 360.0 - 180.0)


def _ranking(scene: dict) -> list[tuple[float, str]]:
    labels = scene["label_order"]
    angles = scene["angles"]
    n_index = labels.index("N")
    antipode = (float(angles[n_index]) + 180.0) % 360.0
    ranked = [
        (_angular_error(float(angle), antipode), str(label))
        for label, angle in zip(labels, angles)
        if label != "N"
    ]
    return sorted(ranked, key=lambda item: (item[0], item[1]))


def sample_scene(seed: int) -> dict:
    """Sample balanced canonical/counterfactual variants with independent style."""
    rng = random.Random(int(seed))
    answer = ANSWER_LABELS[int(seed) % len(ANSWER_LABELS)]
    n_slot = rng.randrange(4)
    opposite_slot = (n_slot + 2) % 4
    labels: list[str | None] = [None] * 4
    labels[n_slot] = "N"
    labels[opposite_slot] = answer
    remaining = [label for label in LABELS if label not in {"N", answer}]
    rng.shuffle(remaining)
    for slot in range(4):
        if labels[slot] is None:
            labels[slot] = remaining.pop()

    # Every fourth difficulty cycle moves the opposite and adjacent slots toward
    # one another while maintaining readable spacing and a positive decision gap.
    base = rng.uniform(0.0, 360.0)
    offsets = [rng.uniform(-6.0, 6.0) for _ in range(4)]
    if (int(seed) // 3) % 4 == 0:
        direction = rng.choice((-1, 1))
        offsets[opposite_slot] = direction * rng.uniform(21.0, 25.0)
        neighbor = (opposite_slot + direction) % 4
        offsets[neighbor] = -direction * rng.uniform(20.0, 25.0)
    angles = [(base + slot * 90.0 + offsets[slot]) % 360.0 for slot in range(4)]

    scene = {
        "label_order": [str(label) for label in labels],
        "angles": [round(value, 6) for value in angles],
        "width": 512,
        "height": 512,
        "center_x": rng.randint(245, 267),
        "center_y": rng.randint(245, 267),
        "ring_radius": rng.randint(181, 196),
        "label_radius": rng.randint(118, 132),
        "font_size": rng.randint(62, 74),
        "ring_width": rng.randint(5, 8),
        "tick_length": rng.randint(12, 19),
        "tick_count": rng.choice((12, 16, 20)),
        "hub_radius": rng.randint(10, 14),
        "background_color": rng.choice(BACKGROUND_PALETTE),
        "ring_color": rng.choice(RING_PALETTE),
        "ink_color": rng.choice(INK_PALETTE),
    }
    if analytic_gold(scene) != answer or margin(scene) < 28.0:
        # This is a deterministic safety fallback for an unexpectedly tight
        # jitter realization; it retains the sampled pose and all style fields.
        clean_angles = [(base + slot * 90.0) % 360.0 for slot in range(4)]
        scene["angles"] = [round(value, 6) for value in clean_angles]
    return scene


def analytic_gold(scene: dict) -> str:
    return _ranking(scene)[0][1]


def margin(scene: dict) -> float:
    ranked = _ranking(scene)
    return float(ranked[1][0] - ranked[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 18.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(FONT_NAME, size=size)


def render(scene: dict) -> Image.Image:
    """Render from scene fields only, using 4x supersampling for clean glyphs."""
    ss = 4
    width = int(scene["width"])
    height = int(scene["height"])
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    ring_radius = float(scene["ring_radius"])
    label_radius = float(scene["label_radius"])
    font_size = int(scene["font_size"])
    ring_width = int(scene["ring_width"])
    tick_length = float(scene["tick_length"])
    tick_count = int(scene["tick_count"])
    hub_radius = float(scene["hub_radius"])
    background = str(scene["background_color"])
    ring_color = str(scene["ring_color"])
    ink_color = str(scene["ink_color"])
    labels = list(scene["label_order"])
    angles = [float(value) for value in scene["angles"]]

    image = Image.new("RGB", (width * ss, height * ss), background)
    draw = ImageDraw.Draw(image)
    center = (cx * ss, cy * ss)
    rr = ring_radius * ss
    draw.ellipse(
        (center[0] - rr, center[1] - rr, center[0] + rr, center[1] + rr),
        outline=ring_color,
        width=ring_width * ss,
    )
    # Symmetric ticks and faint spokes make the center-relative geometry explicit.
    for index in range(tick_count):
        theta = math.radians(index * 360.0 / tick_count)
        outer = ring_radius - ring_width
        inner = outer - tick_length * (1.35 if index % (tick_count // 4) == 0 else 0.65)
        p1 = ((cx + inner * math.cos(theta)) * ss, (cy + inner * math.sin(theta)) * ss)
        p2 = ((cx + outer * math.cos(theta)) * ss, (cy + outer * math.sin(theta)) * ss)
        draw.line((p1, p2), fill=ring_color, width=max(2, ring_width * ss // 2))
    for theta_deg in (0.0, 90.0, 180.0, 270.0):
        theta = math.radians(theta_deg)
        end = ((cx + 55.0 * math.cos(theta)) * ss, (cy + 55.0 * math.sin(theta)) * ss)
        draw.line((center, end), fill=ring_color, width=max(2, 2 * ss))

    font = _font(font_size * ss)
    for label, angle in zip(labels, angles):
        theta = math.radians(angle)
        x = (cx + label_radius * math.cos(theta)) * ss
        y = (cy + label_radius * math.sin(theta)) * ss
        box = draw.textbbox((0, 0), label, font=font)
        glyph_w = box[2] - box[0]
        glyph_h = box[3] - box[1]
        origin = (x - glyph_w / 2.0 - box[0], y - glyph_h / 2.0 - box[1])
        draw.text(origin, label, font=font, fill=ink_color)

    hr = hub_radius * ss
    draw.ellipse(
        (center[0] - hr, center[1] - hr, center[0] + hr, center[1] + hr),
        fill=HUB_COLOR,
        outline="#ffffff",
        width=2 * ss,
    )
    return image.resize((width, height), Image.Resampling.LANCZOS)
