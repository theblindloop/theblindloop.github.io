"""Border-breach-direction world: global figure-ground outflow reveal.

A single connected dark (navy) figure fills part of the canvas. It is a filled
disk whose position and size are sampled so that it bulges out to breach
**exactly one** border of the square canvas and stays clearly away from the
other three borders. The decision is the global figure-ground property of which
border the one connected region runs into and breaks.

The renderer owns latent sampling + rasterization + the structural analytic
gold. The independent pixel-only inverse arm (``oracle.py``) recovers the same
global outflow decision from the final PNG alone by detecting which border the
dark figure actually reaches.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (255, 255, 255)
FIGURE = (40, 80, 200)
# Radius of the single filled figure disk (px).
_R_LO = 150
_R_HI = 180
# How far the disk bulges beyond the breached border (px).
_D_LO = 16
_D_HI = 60
# Minimum clearance of the disk from a non-breached border (px). Being strictly
# above zero guarantees the figure never touches a second border.
_MIN_SIDE_CLR = 10
# Below this margin a scene is quarantined (the figure nearly breaches a second
# border and the outflow target becomes ambiguous at finite resolution).
MIN_GAP = 24

_SIDES = ("top", "bottom", "left", "right")


def sample_scene(seed: int, side: str | None = None) -> dict:
    """Sample one latent scene deterministically from ``seed``.

    ``side`` is primarily a scene-derived value; the generator pins it each scene
    to balance the four off-quarantine classes.
    """
    rng = np.random.default_rng(seed)
    if side is None:
        side = _SIDES[int(rng.integers(0, 4))]
    r = float(rng.integers(_R_LO, _R_HI + 1))
    depth = float(rng.integers(_D_LO, _D_HI + 1))
    free = float(rng.integers(_MIN_SIDE_CLR + r, WIDTH - _MIN_SIDE_CLR - r + 1))
    if side == "top":
        cx, cy = free, r - depth
    elif side == "bottom":
        cx, cy = free, (HEIGHT - 1) - (r - depth)
    elif side == "left":
        cy, cx = free, r - depth
    else:  # right
        cy, cx = free, (WIDTH - 1) - (r - depth)
    return {
        "side": side,
        "center": [round(float(cy), 6), round(float(cx), 6)],
        "radius": round(r, 6),
        "breach": round(depth, 6),
    }


def _draw_scene(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    d = ImageDraw.Draw(img)
    _ = scene["side"]
    cy, cx = scene["center"]
    r = scene["radius"]
    d.ellipse([cx - r, cy - r, cx + r, cy + r], fill=FIGURE)
    return img


def render(scene: dict) -> Image.Image:
    """Rasterize the latent scene (deterministic, boundary-free PNG)."""
    return _draw_scene(scene)


def analytic_gold(scene: dict) -> str:
    """Structural gold: the name of the single breached border."""
    return str(scene["side"])


def margin(scene: dict) -> float:
    """Commitment magnitude: min clearance of the disk to the non-breached borders."""
    cy, cx = scene["center"]
    r = scene["radius"]
    clearances = {
        "top": cy - r,
        "bottom": (HEIGHT - 1) - (cy + r),
        "left": cx - r,
        "right": (WIDTH - 1) - (cx + r),
    }
    del clearances[scene["side"]]
    return float(min(clearances.values()))


def is_quarantined(scene: dict) -> bool:
    """Quarantine when the disk nearly breaches a second border."""
    return margin(scene) < MIN_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The scene is injectively rendered; no latent aliases are declared."""
    return []
