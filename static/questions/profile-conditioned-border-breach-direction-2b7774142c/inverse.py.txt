"""Independent pixel-only inverse arm for the border-breach-direction world.

Recovers which canvas border the single connected dark figure breaches from the
final rasterized PNG alone: it threshold out the dark figure and report the
unique border row/column that the dark region actually reaches. It shares no
geometry, palette or thresholds with the renderer beyond the physical facts that
the figure is clearly darker than the white background and that the answer is
the border the figure touches.

This module never imports renderer / prompts / generate / verify / analytic-gold
code and never receives the latent scene.
"""

from __future__ import annotations

from functools import lru_cache

import numpy as np
from PIL import Image

_DARK = 120


@lru_cache(maxsize=None)
def _decide(arr_bytes: bytes, shape: tuple[int, int]) -> str:
    a = np.frombuffer(arr_bytes, dtype=np.uint8).reshape(shape)
    dark = a < _DARK
    h, w = dark.shape
    top = bool(dark[0, :].any())
    bottom = bool(dark[h - 1, :].any())
    left = bool(dark[:, 0].any())
    right = bool(dark[:, w - 1].any())
    touched = [name for name, flag in
               (("top", top), ("bottom", bottom), ("left", left), ("right", right)) if flag]
    return touched[0]


def decision_from_image(image: Image.Image) -> str:
    """Return the breached border ('top'/'bottom'/'left'/'right') from the PNG."""
    arr = np.asarray(image.convert("RGB"))
    gray = np.asarray(arr, dtype=np.uint8).min(axis=2)
    return _decide(gray.tobytes(), gray.shape)
