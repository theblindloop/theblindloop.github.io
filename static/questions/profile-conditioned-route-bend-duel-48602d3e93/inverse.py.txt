"""Scene-free inverse arm operating only on final RGB pixels."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

_INK = np.array([38, 42, 47], dtype=np.int16)
_ORANGE = np.array([242, 142, 43], dtype=np.int16)
_PURPLE = np.array([156, 106, 222], dtype=np.int16)


def _near(arr: np.ndarray, color: np.ndarray, radius: int) -> np.ndarray:
    delta = arr.astype(np.int32) - color.astype(np.int32)
    return (delta * delta).sum(axis=2) <= radius * radius


def _components(mask: np.ndarray) -> list[np.ndarray]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        pts = []
        while queue:
            cy, cx = queue.popleft()
            pts.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(pts) > 150:
            groups.append(np.asarray(pts, dtype=np.int32))
    return groups


def _vertical_run_count(component: np.ndarray) -> int:
    ys, xs = component[:, 0], component[:, 1]
    lo, hi = int(xs.min()), int(xs.max())
    spans = np.zeros(hi - lo + 1, dtype=np.int32)
    for x in range(lo, hi + 1):
        col = ys[xs == x]
        if len(col):
            spans[x - lo] = int(col.max() - col.min() + 1)
    active = spans >= 20
    starts = int(active[0]) + int(np.count_nonzero(active[1:] & ~active[:-1]))
    if starts not in (1, 2, 3):
        raise ValueError(f"could not recover a supported route: {starts} vertical runs")
    return starts


def _centroid(mask: np.ndarray, label: str) -> np.ndarray:
    pts = np.argwhere(mask)
    if len(pts) < 40:
        raise ValueError(f"missing {label} terminal")
    return pts.mean(axis=0)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    components = _components(_near(arr, _INK, 42))
    if len(components) != 2:
        raise ValueError(f"expected two dark routes, found {len(components)}")

    terminals = {
        "orange": _centroid(_near(arr, _ORANGE, 48), "orange"),
        "purple": _centroid(_near(arr, _PURPLE, 48), "purple"),
    }
    counts: dict[str, int] = {}
    unused = list(range(len(components)))
    for label, center in terminals.items():
        index = min(unused, key=lambda i: float(np.min(((components[i] - center) ** 2).sum(axis=1))))
        unused.remove(index)
        counts[label] = 2 * _vertical_run_count(components[index])
    if counts["orange"] == counts["purple"]:
        raise ValueError("pixel evidence has tied bend counts")
    return "orange" if counts["orange"] < counts["purple"] else "purple"
