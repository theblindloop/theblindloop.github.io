"""Deterministic renderer and analytic arm for the route-bend duel."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BG = "#f7f7f4"
INK = "#262a2f"
GREEN = "#28a86b"
ORANGE = "#f28e2b"
PURPLE = "#9c6ade"


def _route_points(bends: int, center_y: int, mirror: bool, rng: random.Random) -> list[list[int]]:
    """Make an x-monotone orthogonal route with 490 px total centerline length.

    All routes span 400 px horizontally and travel 90 px vertically, so neither
    endpoint distance nor total path ink predicts the bend-count answer.
    """
    verticals = bends // 2
    sign = -1 if mirror else 1
    if verticals == 1:
        start_y = center_y - sign * 45
        deltas = [sign * 90]
    elif verticals == 2:
        start_y = center_y
        deltas = [sign * 45, -sign * 45]
    elif verticals == 3:
        start_y = center_y - sign * 15
        deltas = [sign * 30, -sign * 30, sign * 30]
    else:
        raise ValueError("supported bends are 2, 4, or 6")

    # Jitter internal x positions while preserving wide, visually distinct runs.
    step = 400 / (verticals + 1)
    xs = [round(56 + step * (i + 1) + rng.randint(-12, 12)) for i in range(verticals)]
    xs = sorted(xs)
    points: list[list[int]] = [[56, start_y]]
    y = start_y
    for x, dy in zip(xs, deltas):
        points.append([x, y])
        y += dy
        points.append([x, y])
    points.append([456, y])
    return points


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    low, high = rng.choice([(2, 4), (2, 6), (4, 6)])
    fewer_color = "orange" if int(seed) % 2 == 0 else "purple"
    top_color = rng.choice(["orange", "purple"])
    bottom_color = "purple" if top_color == "orange" else "orange"

    def count_for(color: str) -> int:
        return low if color == fewer_color else high

    routes = []
    for panel, color, cy in (("top", top_color, 150), ("bottom", bottom_color, 362)):
        bends = count_for(color)
        routes.append(
            {
                "panel": panel,
                "terminal": color,
                "points": _route_points(bends, cy, bool(rng.getrandbits(1)), rng),
            }
        )
    return {"routes": routes}


def _bend_count(points: list[list[int]]) -> int:
    directions = []
    for a, b in zip(points, points[1:]):
        dx, dy = b[0] - a[0], b[1] - a[1]
        if (dx == 0) == (dy == 0):
            raise ValueError("route segment must be exactly horizontal or vertical")
        directions.append("h" if dy == 0 else "v")
    return sum(a != b for a, b in zip(directions, directions[1:]))


def analytic_gold(scene: dict) -> str:
    counts = {
        route["terminal"]: _bend_count(route["points"])
        for route in scene["routes"]
    }
    if counts["orange"] == counts["purple"]:
        raise ValueError("equal bend counts have no unique answer")
    return "orange" if counts["orange"] < counts["purple"] else "purple"


def margin(scene: dict) -> float:
    counts = {
        route["terminal"]: _bend_count(route["points"])
        for route in scene["routes"]
    }
    return float(abs(counts["orange"] - counts["purple"]))


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 2.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    scale = 2
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BG)
    draw = ImageDraw.Draw(image)

    def box(values):
        return tuple(int(v * scale) for v in values)

    # The paired cards make ownership unambiguous without naming top or bottom.
    for bounds in ((24, 48, 488, 244), (24, 268, 488, 464)):
        draw.rounded_rectangle(box(bounds), radius=18 * scale, fill="#ffffff", outline="#d8dadd", width=2 * scale)

    for route in scene["routes"]:
        pts = [tuple(int(v * scale) for v in point) for point in route["points"]]
        draw.line(pts, fill=INK, width=8 * scale, joint="curve")

        sx, sy = route["points"][0]
        ex, ey = route["points"][-1]
        # White-centered green ring denotes the directionally irrelevant start.
        draw.ellipse(box((sx - 13, sy - 13, sx + 13, sy + 13)), fill="#ffffff", outline=GREEN, width=6 * scale)
        color = ORANGE if route["terminal"] == "orange" else PURPLE
        draw.rounded_rectangle(box((ex - 13, ey - 13, ex + 13, ey + 13)), radius=4 * scale, fill=color, outline="#ffffff", width=3 * scale)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
