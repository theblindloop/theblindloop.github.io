"""Independent pixel-only inverse arm for the constellation-interior-dot world.

The oracle receives only the final rendered PNG and the public question
contract. It reconstructs the dark dot constellation from the raster alone,
recomputes the convex hull of the recovered dot centroids, and reports whether
any recovered dot lies strictly inside that convex outline. It imports only
numpy/PIL and never receives the latent scene or analytic gold.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

WIDTH = 512
HEIGHT = 512
INK_THRESHOLD = 128
INTERIOR_EPS = 3.0


def _dark_mask(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB")).astype(int)
    return (arr[..., 0] < INK_THRESHOLD) & (
        (arr[..., 1] < INK_THRESHOLD) | (arr[..., 2] < INK_THRESHOLD) | (arr[..., 0] < INK_THRESHOLD)
    )


def _centroids(mask: np.ndarray) -> list[tuple[float, float]]:
    """Recover each connected dark dot's centroid by BFS flood fill."""
    h, w = mask.shape
    visited = np.zeros((h, w), dtype=bool)
    centroids: list[tuple[float, float]] = []
    for y in range(h):
        for x in range(w):
            if mask[y, x] and not visited[y, x]:
                queue = deque([(y, x)])
                visited[y, x] = True
                sx = 0
                sy = 0
                count = 0
                while queue:
                    cy, cx = queue.popleft()
                    sx += cx
                    sy += cy
                    count += 1
                    for ny in (cy - 1, cy, cy + 1):
                        for nx in (cx - 1, cx, cx + 1):
                            if 0 <= ny < h and 0 <= nx < w:
                                if mask[ny, nx] and not visited[ny, nx]:
                                    visited[ny, nx] = True
                                    queue.append((ny, nx))
                if count > 0:
                    centroids.append((float(sx) / count, float(sy) / count))
    return centroids


def _convex_hull(points: list[tuple[float, float]]) -> list[tuple[float, float]]:
    pts = sorted(set(points))

    def cross(o, a, b):
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

    lower: list = []
    for p in pts:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    upper: list = []
    for p in reversed(pts):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    return lower[:-1] + upper[:-1]


def _seg_dist(p, a, b) -> float:
    px, py = p
    ax, ay = a
    bx, by = b
    dx, dy = bx - ax, by - ay
    length_sq = dx * dx + dy * dy
    if length_sq == 0.0:
        return float(np.hypot(px - ax, py - ay))
    t = max(0.0, min(1.0, ((px - ax) * dx + (py - ay) * dy) / length_sq))
    proj_x, proj_y = ax + t * dx, ay + t * dy
    return float(np.hypot(px - proj_x, py - proj_y))


def _boundary_distance(p, hull) -> float:
    best = float("inf")
    h = len(hull)
    for i in range(h):
        d = _seg_dist(p, hull[i], hull[(i + 1) % h])
        if d < best:
            best = d
    return best


def decision_from_image(image: Image.Image) -> str:
    """Image-only inverse: 'yes' if a recovered dot lies inside the hull."""
    mask = _dark_mask(image)
    centroids = _centroids(mask)
    if len(centroids) < 3:
        return "no"
    hull = _convex_hull(centroids)
    for p in centroids:
        if _boundary_distance(p, hull) >= INTERIOR_EPS:
            return "yes"
    return "no"
