"""Constellation-interior-dot world: does a dot sit inside the hull?

A set of small dark dots is drawn on a white canvas. Taken together the
outermost dots outline a convex figure (the convex hull of the whole dot
set). The decision is whether at least one dot lies strictly inside that
convex outline formed by the other dots, versus every dot resting on the
outer convex hull perimeter.

The renderer owns scene sampling + rasterization + the structural analytic
gold. The independent pixel-only inverse arm (``oracle.py``) reconstructs the
dot constellation from the final PNG alone and recomputes the same global
convex-hull interior decision from the recovered dot centroids.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
CENTER_X = WIDTH // 2
CENTER_Y = HEIGHT // 2
DOT_R = 10
INK = (30, 30, 30)
BACKGROUND = (255, 255, 255)

HULL_R = 110.0
MIN_GAP_RAD = np.deg2rad(50.0)
FRAC_MIN = 0.35
FRAC_MAX = 0.55
INTERIOR_EPS = 3.0
QUARANTINE_MARGIN = 4.0


def sample_scene(seed: int) -> dict:
    """Sample one latent scene deterministically from ``seed``.

    Scenes are regenerated (with a derived seed) until their decision margin is
    comfortably above the quarantine band, so headline classes stay decisive.
    """
    for attempt in range(200):
        candidate = _build(seed + attempt * 1000003)
        if _interior_clearance(candidate["dots"])[0] >= QUARANTINE_MARGIN:
            return candidate
    return _build(seed)


def _build(seed: int) -> dict:
    rng = np.random.default_rng(seed)
    n_ring = int(rng.integers(5, 8))
    interior = bool(rng.integers(0, 2))

    angles = _sample_angles(rng, n_ring, MIN_GAP_RAD)
    dots = []
    for a in angles:
        x = CENTER_X + HULL_R * np.cos(a)
        y = CENTER_Y + HULL_R * np.sin(a)
        dots.append([int(round(x)), int(round(y))])

    if interior:
        frac = float(rng.uniform(FRAC_MIN, FRAC_MAX))
        a2 = float(rng.uniform(0.0, 2.0 * np.pi))
        x = CENTER_X + HULL_R * frac * np.cos(a2)
        y = CENTER_Y + HULL_R * frac * np.sin(a2)
        dots.append([int(round(x)), int(round(y))])

    return {"dots": dots}


def _sample_angles(rng, n: int, min_gap: float) -> np.ndarray:
    """Return ``n`` angles in [0, 2pi) with pairwise circular gap >= min_gap."""
    for _attempt in range(1000):
        angles = np.sort(rng.uniform(0.0, 2.0 * np.pi, n))
        gaps = np.diff(np.concatenate([angles, angles[:1] + 2.0 * np.pi]))
        if gaps.min() >= min_gap:
            return angles
    # Fallback: uniform spacing with jitter that keeps the min gap.
    base = np.linspace(0.0, 2.0 * np.pi, n, endpoint=False)
    return base + min_gap * 0.1 * (rng.random(n) - 0.5)


def render(scene: dict) -> Image.Image:
    """Rasterize the latent scene (a deterministic PNG)."""
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    d = ImageDraw.Draw(img)
    for (x, y) in scene["dots"]:
        d.ellipse([x - DOT_R, y - DOT_R, x + DOT_R, y + DOT_R], fill=INK)
    return img


def _convex_hull(points: list[tuple[float, float]]) -> list[tuple[float, float]]:
    """Andrew's monotone-chain convex hull; returns hull vertices in order."""
    pts = sorted({(float(x), float(y)) for (x, y) in points})
    if len(pts) <= 2:
        return pts

    def cross(o, a, b):
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

    lower: list = []
    for p in pts:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    upper: list = []
    for p in reversed(pts):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    return lower[:-1] + upper[:-1]


def _seg_dist(p: tuple[float, float], a: tuple[float, float], b: tuple[float, float]) -> float:
    """Distance from point p to the finite segment a-b."""
    px, py = p
    ax, ay = a
    bx, by = b
    dx, dy = bx - ax, by - ay
    length_sq = dx * dx + dy * dy
    if length_sq == 0.0:
        return float(np.hypot(px - ax, py - ay))
    t = max(0.0, min(1.0, ((px - ax) * dx + (py - ay) * dy) / length_sq))
    proj_x, proj_y = ax + t * dx, ay + t * dy
    return float(np.hypot(px - proj_x, py - proj_y))


def _hull_boundary_distance(p: tuple[float, float], hull: list) -> float:
    """Minimum distance from point p to the convex-hull polygon boundary."""
    best = float("inf")
    h = len(hull)
    for i in range(h):
        d = _seg_dist(p, hull[i], hull[(i + 1) % h])
        if d < best:
            best = d
    return best


def _vertex_deflection(vertex, hull_list) -> float:
    """Perpendicular distance from a hull vertex to the line through its neighbours."""
    h = len(hull_list)
    for i, v in enumerate(hull_list):
        if v == vertex:
            a = hull_list[(i - 1) % h]
            b = hull_list[(i + 1) % h]
            ax, ay = a
            bx, by = b
            vx, vy = v
            num = abs((by - ay) * vx - (bx - ax) * vy + bx * ay - by * ax)
            den = float(np.hypot(by - ay, bx - ax))
            return num / den if den > 0 else 0.0
    return 0.0

def _interior_clearance(dots: list) -> tuple[float, list]:
    """Return (min interior clearance, list of interior dots)."""
    pts = [(float(x), float(y)) for (x, y) in dots]
    hull = _convex_hull(pts)
    interiors = []
    for p in pts:
        c = _hull_boundary_distance(p, hull)
        if c >= INTERIOR_EPS:
            interiors.append(p)
    if interiors:
        depth = min(_hull_boundary_distance(p, hull) for p in interiors)
        return float(depth), interiors
    # No interior dot: margin is the min convex deflection over hull vertices.
    if len(hull) < 3:
        return float("inf"), []
    min_def = min(_vertex_deflection(v, hull) for v in hull)
    return float(min_def), []


def analytic_gold(scene: dict) -> str:
    """Structural gold: does an interior dot exist (strictly inside the hull)?"""
    _clearance, interiors = _interior_clearance(scene["dots"])
    return "yes" if interiors else "no"


def margin(scene: dict) -> float:
    """Decision commitment: interior depth (yes) or hull convex deflection (no)."""
    clearance, _interiors = _interior_clearance(scene["dots"])
    return clearance


def is_quarantined(scene: dict) -> bool:
    """Quarantine near-boundary constellations whose interior read is ambiguous."""
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The scene is injectively rendered; no latent aliases are declared."""
    return []
