"""Deterministic scientific dot-and-whisker renderer and analytic arm."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 640
HEIGHT = 420
COLORS = ("red", "blue", "green", "orange")
RGB = {
    "red": (214, 55, 62),
    "blue": (45, 105, 205),
    "green": (36, 145, 82),
    "orange": (232, 132, 30),
}
ROWS = (78, 158, 238, 318)
PLOT_LEFT = 66
PLOT_RIGHT = 594
MARKER_RADIUS = 12
CAP_HALF_HEIGHT = 8
LINE_WIDTH = 5
QUARANTINE_GAP = 6.0
ENDPOINT_CLEARANCE = 28.0


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene from one integer seed.

    Answer colour cycles with the seed while row, interval center, interval
    width, marker direction, and distractor difficulty are independently
    shuffled. Every record is fully rendered; there is no gold-only field.
    """

    rng = random.Random(int(seed))
    winner = COLORS[int(seed) % len(COLORS)]
    runner = rng.choice([color for color in COLORS if color != winner])

    winner_offset = rng.randint(0, 9)
    if int(seed) % 6 == 0:
        runner_gap = rng.randint(7, 11)  # deliberate near-boundary coverage
    else:
        runner_gap = rng.randint(18, 42)
    offsets = {winner: winner_offset, runner: winner_offset + runner_gap}
    remaining = [color for color in COLORS if color not in offsets]
    rng.shuffle(remaining)
    offsets[remaining[0]] = offsets[runner] + rng.randint(16, 29)
    offsets[remaining[1]] = offsets[remaining[0]] + rng.randint(14, 25)

    row_order = list(ROWS)
    rng.shuffle(row_order)
    intervals = []
    for color, y in zip(COLORS, row_order, strict=True):
        half_width = rng.randint(165, 198)
        center = rng.randint(PLOT_LEFT + half_width, PLOT_RIGHT - half_width)
        direction = -1 if rng.random() < 0.5 else 1
        marker = center + direction * offsets[color]
        intervals.append(
            {
                "color": color,
                "y": y,
                "left": center - half_width,
                "right": center + half_width,
                "marker": marker,
            }
        )
    return {"width": WIDTH, "height": HEIGHT, "intervals": intervals}


def _offsets(intervals: list[dict]) -> list[tuple[float, str]]:
    measured = []
    for item in intervals:
        midpoint = (float(item["left"]) + float(item["right"])) / 2.0
        measured.append((abs(float(item["marker"]) - midpoint), str(item["color"])))
    return sorted(measured)


def analytic_gold(scene: dict) -> str:
    """Return the color with the smallest absolute marker-to-midpoint offset."""

    return _offsets(scene["intervals"])[0][1]


def margin(scene: dict) -> float:
    """Runner-up decision margin in final-canvas pixels."""

    ranked = _offsets(scene["intervals"])
    return float(ranked[1][0] - ranked[0][0])


def is_quarantined(scene: dict) -> bool:
    if margin(scene) < QUARANTINE_GAP:
        return True
    for item in scene["intervals"]:
        clearance = min(item["marker"] - item["left"], item["right"] - item["marker"])
        if clearance < ENDPOINT_CLEARANCE:
            return True
    return False


def render(scene: dict) -> Image.Image:
    """Render only visible scene fields to a deterministic antialiased RGB PNG."""

    ss = 3
    width = int(scene["width"])
    height = int(scene["height"])
    image = Image.new("RGB", (width * ss, height * ss), (250, 251, 252))
    draw = ImageDraw.Draw(image)

    # A quiet scientific plotting field. Grid and axes carry no answer identity.
    for x in range(80, 601, 80):
        draw.line((x * ss, 40 * ss, x * ss, 354 * ss), fill=(225, 229, 234), width=1 * ss)
    draw.line((52 * ss, 356 * ss, 608 * ss, 356 * ss), fill=(117, 124, 134), width=2 * ss)
    for x in range(80, 601, 80):
        draw.line((x * ss, 356 * ss, x * ss, 363 * ss), fill=(117, 124, 134), width=2 * ss)

    for item in scene["intervals"]:
        color = RGB[str(item["color"])]
        y = int(item["y"])
        left = int(item["left"])
        right = int(item["right"])
        marker = int(item["marker"])
        draw.line(
            (left * ss, y * ss, right * ss, y * ss),
            fill=color,
            width=LINE_WIDTH * ss,
        )
        draw.line(
            (left * ss, (y - CAP_HALF_HEIGHT) * ss, left * ss, (y + CAP_HALF_HEIGHT) * ss),
            fill=color,
            width=LINE_WIDTH * ss,
        )
        draw.line(
            (right * ss, (y - CAP_HALF_HEIGHT) * ss, right * ss, (y + CAP_HALF_HEIGHT) * ss),
            fill=color,
            width=LINE_WIDTH * ss,
        )
        r = MARKER_RADIUS * ss
        draw.ellipse(
            (marker * ss - r, y * ss - r, marker * ss + r, y * ss + r),
            fill=color,
        )

    return image.resize((width, height), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Exercise the render-invariant list-order alias of nonoverlapping rows."""

    twin = deepcopy(scene)
    twin["intervals"] = list(reversed(twin["intervals"]))
    return [("reverse_unordered_interval_records", twin)]
