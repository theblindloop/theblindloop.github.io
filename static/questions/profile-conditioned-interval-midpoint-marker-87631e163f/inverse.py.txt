"""Independent inverse arm for colored dot-and-whisker rasters.

This module intentionally duplicates the public palette and measures endpoint
and marker positions from RGB pixels. It imports no candidate renderer, latent
scene, prompt, generator, verifier, or analytic-gold code.
"""

from __future__ import annotations

from dataclasses import dataclass

import numpy as np
from PIL import Image

_PALETTE = {
    "red": (214, 55, 62),
    "blue": (45, 105, 205),
    "green": (36, 145, 82),
    "orange": (232, 132, 30),
}
_MAX_COLOR_DISTANCE = 72.0


@dataclass(frozen=True)
class RowMeasurement:
    color: str
    left: float
    right: float
    marker: float

    @property
    def offset(self) -> float:
        return abs(self.marker - (self.left + self.right) / 2.0)


def _color_mask(arr: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    delta = arr.astype(np.float32) - np.asarray(rgb, dtype=np.float32)
    return np.sum(delta * delta, axis=2) <= _MAX_COLOR_DISTANCE**2


def _largest_true_run(values: np.ndarray) -> tuple[int, int]:
    indices = np.flatnonzero(values)
    if len(indices) == 0:
        raise ValueError("no marker-like columns")
    splits = np.flatnonzero(np.diff(indices) > 1) + 1
    runs = np.split(indices, splits)
    run = max(runs, key=lambda item: (len(item), -int(item[0])))
    return int(run[0]), int(run[-1])


def _measure_color(arr: np.ndarray, color: str, rgb: tuple[int, int, int]) -> RowMeasurement:
    mask = _color_mask(arr, rgb)
    ys, xs = np.nonzero(mask)
    if len(xs) < 500:
        raise ValueError(f"{color}: insufficient colored ink")

    # Ignore isolated resampling speckles; a real interval spans hundreds of x columns.
    column_counts = mask.sum(axis=0)
    support = column_counts >= 3
    support_x = np.flatnonzero(support)
    if len(support_x) < 180:
        raise ValueError(f"{color}: interval support is missing or too short")
    left = float(support_x[0])
    right = float(support_x[-1])

    # A 24-pixel circular marker is visibly taller than the 16-pixel endpoint caps.
    # Select the contiguous columns whose vertical span clears the cap height.
    spans = np.zeros(mask.shape[1], dtype=np.int32)
    for x in support_x:
        y_at_x = np.flatnonzero(mask[:, x])
        if len(y_at_x):
            spans[x] = int(y_at_x[-1] - y_at_x[0] + 1)
    marker_columns = spans >= 21
    m0, m1 = _largest_true_run(marker_columns)
    if m1 - m0 + 1 < 7 or m1 - m0 + 1 > 30:
        raise ValueError(f"{color}: marker width is implausible")

    marker_region = mask[:, m0 : m1 + 1]
    marker_xs = np.nonzero(marker_region)[1].astype(np.float64) + m0
    if len(marker_xs) < 120:
        raise ValueError(f"{color}: marker body is incomplete")
    marker = float(marker_xs.mean())
    return RowMeasurement(color=color, left=left, right=right, marker=marker)


def measure(image: Image.Image) -> list[RowMeasurement]:
    arr = np.asarray(image.convert("RGB"))
    if arr.ndim != 3 or arr.shape[0] < 200 or arr.shape[1] < 300:
        raise ValueError("image is too small for the declared plot")
    rows = [_measure_color(arr, color, rgb) for color, rgb in _PALETTE.items()]
    if len({round(row.marker, 1) for row in rows}) < 2:
        raise ValueError("marker measurements collapsed")
    return rows


def decision_from_image(image: Image.Image) -> str:
    rows = measure(image)
    ranked = sorted((row.offset, row.color) for row in rows)
    if ranked[1][0] - ranked[0][0] < 2.0:
        raise ValueError("pixel decision is ambiguous")
    return ranked[0][1]
