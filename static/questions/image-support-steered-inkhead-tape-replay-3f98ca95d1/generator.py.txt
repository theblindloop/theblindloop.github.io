"""Renderer and analytic model for Inkhead Tape Replay."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

WIDTH = HEIGHT = 512
INKS = ((44, 61, 88), (83, 52, 91))
BLANKS = ((205, 226, 237), (231, 218, 193))
ARROW = (35, 157, 166)
GOLD = (232, 166, 38)
OUTLINE = (31, 38, 48)
PAPER = (250, 248, 241)
CARD = (242, 239, 228)


def _replay(tape: list[int], head: int, moves: list[int]) -> tuple[list[int], int]:
    cells = list(tape)
    position = int(head)
    for move in moves:
        cells[position] ^= 1
        position = max(0, min(3, position + int(move)))
    return cells, position


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x4A71C)
    head = rng.randrange(4)
    moves = [rng.choice((-1, 1)) for _ in range(3)]
    tape = [rng.randrange(2) for _ in range(4)]
    # Exact balance without tying the answer to any visible local feature.
    target = int(seed) & 1
    blank = [0, 0, 0, 0]
    visits = [0, 0, 0, 0]
    position = head
    for move in moves:
        visits[position] ^= 1
        position = max(0, min(3, position + move))
    tape[position] = target ^ visits[position]
    return {
        "tape": tape,
        "head": head,
        "moves": moves,
        "palette_variant": rng.randrange(2),
    }


def analytic_gold(scene: dict) -> str:
    cells, head = _replay(scene["tape"], scene["head"], scene["moves"])
    return "yes" if cells[head] else "no"


def margin(scene: dict) -> float:
    # Minimum radius/stroke clearance among the discrete visual marks.
    return 10.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["palette_variant"] = int(scene["palette_variant"]) + 2
    return [("equivalent_palette_code_plus_two", twin)]


def _arrow(draw: ImageDraw.ImageDraw, cx: int, cy: int, move: int, scale: int) -> None:
    if move == 1:
        points = [(cx - 20, cy - 7), (cx + 4, cy - 7), (cx + 4, cy - 15),
                  (cx + 24, cy), (cx + 4, cy + 15), (cx + 4, cy + 7), (cx - 20, cy + 7)]
    else:
        points = [(cx + 20, cy - 7), (cx - 4, cy - 7), (cx - 4, cy - 15),
                  (cx - 24, cy), (cx - 4, cy + 15), (cx - 4, cy + 7), (cx + 20, cy + 7)]
    draw.polygon([(x * scale, y * scale) for x, y in points], fill=ARROW)


def render(scene: dict) -> Image.Image:
    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), PAPER)
    draw = ImageDraw.Draw(image)
    # Two detached cards occupy one centered, medium-width region.
    draw.rounded_rectangle((112 * scale, 132 * scale, 208 * scale, 380 * scale),
                           radius=18 * scale, fill=CARD, outline=(205, 198, 180), width=2 * scale)
    draw.rounded_rectangle((304 * scale, 132 * scale, 400 * scale, 380 * scale),
                           radius=18 * scale, fill=CARD, outline=(205, 198, 180), width=2 * scale)

    ink = INKS[int(scene["palette_variant"]) % 2]
    blank = BLANKS[int(scene["palette_variant"]) % 2]
    ys = (178, 230, 282, 334)
    for index, (value, cy) in enumerate(zip(scene["tape"], ys)):
        cx = 160
        if index == int(scene["head"]):
            draw.ellipse(((cx - 24) * scale, (cy - 24) * scale,
                          (cx + 24) * scale, (cy + 24) * scale),
                         outline=GOLD, width=6 * scale)
        draw.ellipse(((cx - 16) * scale, (cy - 16) * scale,
                      (cx + 16) * scale, (cy + 16) * scale),
                     fill=ink if int(value) else blank, outline=OUTLINE, width=3 * scale)

    for ordinal, (move, cy) in enumerate(zip(scene["moves"], (190, 256, 322)), start=1):
        _arrow(draw, 356, cy, int(move), scale)
        # Unary dot ordinals make sequence order visible without OCR.
        start = 356 - (ordinal - 1) * 8
        for dot_index in range(ordinal):
            dx = start + dot_index * 16
            draw.ellipse(((dx - 4) * scale, (cy - 27) * scale,
                          (dx + 4) * scale, (cy - 19) * scale), fill=OUTLINE)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
