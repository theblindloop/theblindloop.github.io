"""Independent image-only inverse for Inkhead Tape Replay."""

from __future__ import annotations

import numpy as np
from PIL import Image

INK_PROTOTYPES = np.asarray(((44, 61, 88), (83, 52, 91)), dtype=np.int16)
BLANK_PROTOTYPES = np.asarray(((205, 226, 237), (231, 218, 193)), dtype=np.int16)
ARROW_PROTOTYPE = np.asarray((35, 157, 166), dtype=np.int16)
GOLD_PROTOTYPE = np.asarray((232, 166, 38), dtype=np.int16)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        points = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        found.append(np.asarray(points, dtype=np.int32))
    return found


def _near_any(array: np.ndarray, prototypes: np.ndarray, tolerance: float) -> np.ndarray:
    rgb = array.astype(np.int32)
    prototypes = prototypes.astype(np.int32)
    distances = np.sqrt(np.sum((rgb[:, :, None, :] - prototypes[None, None, :, :]) ** 2, axis=3))
    return distances.min(axis=2) < tolerance


def _color_blobs(array: np.ndarray, prototypes: np.ndarray, minimum: int = 180) -> list[np.ndarray]:
    return [part for part in _components(_near_any(array, prototypes, 42.0)) if len(part) >= minimum]


def _replay(tape: list[int], head: int, moves: list[int]) -> tuple[list[int], int]:
    values = list(tape)
    position = head
    for move in moves:
        values[position] ^= 1
        position = max(0, min(3, position + move))
    return values, position


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)

    classified = []
    for bit, prototypes in ((1, INK_PROTOTYPES), (0, BLANK_PROTOTYPES)):
        for part in _color_blobs(array, prototypes):
            y0, x0 = part.min(axis=0)
            y1, x1 = part.max(axis=0)
            height, width = int(y1 - y0 + 1), int(x1 - x0 + 1)
            if 20 <= width <= 38 and 20 <= height <= 38 and 0.72 <= width / height <= 1.38:
                classified.append((float(part[:, 0].mean()), float(part[:, 1].mean()), bit))
    if len(classified) != 4:
        raise ValueError("could not recover exactly four tape cells")
    classified.sort()
    xs = [item[1] for item in classified]
    if max(xs) - min(xs) > 9:
        raise ValueError("tape cells are not one aligned ordered stack")
    tape = [item[2] for item in classified]

    gold = _color_blobs(array, GOLD_PROTOTYPE[None, :], minimum=120)
    if len(gold) != 1:
        raise ValueError("could not recover unique head ring")
    head_y = float(gold[0][:, 0].mean())
    gaps = [abs(item[0] - head_y) for item in classified]
    head = int(np.argmin(gaps))
    if gaps[head] > 8:
        raise ValueError("head ring is not bound to a tape cell")

    arrow_parts = _color_blobs(array, ARROW_PROTOTYPE[None, :], minimum=220)
    if len(arrow_parts) != 3:
        raise ValueError("could not recover exactly three arrows")
    arrow_parts.sort(key=lambda part: float(part[:, 0].mean()))
    moves = []
    arrow_ys = []
    for part in arrow_parts:
        y0, x0 = part.min(axis=0)
        y1, x1 = part.max(axis=0)
        if not (37 <= x1 - x0 + 1 <= 55 and 22 <= y1 - y0 + 1 <= 36):
            raise ValueError("malformed arrow")
        left_tip = np.sum(part[:, 1] <= x0 + 4)
        right_tip = np.sum(part[:, 1] >= x1 - 4)
        if abs(int(left_tip) - int(right_tip)) < 8:
            raise ValueError("ambiguous arrow direction")
        moves.append(-1 if left_tip < right_tip else 1)
        arrow_ys.append(float(part[:, 0].mean()))

    # Independently validate the visible 1/2/3 dot ordinals.
    dark = np.mean(array.astype(np.float32), axis=2) < 80
    small = []
    for part in _components(dark):
        if 25 <= len(part) <= 100:
            y0, x0 = part.min(axis=0)
            y1, x1 = part.max(axis=0)
            if x1 - x0 <= 11 and y1 - y0 <= 11:
                small.append((float(part[:, 0].mean()), float(part[:, 1].mean())))
    counts = [sum(abs(y - (arrow_y - 23)) < 8 for y, _ in small) for arrow_y in arrow_ys]
    if counts != [1, 2, 3]:
        raise ValueError("ordinal dots do not uniquely declare arrow order")

    values, position = _replay(tape, head, moves)
    return "yes" if values[position] else "no"
