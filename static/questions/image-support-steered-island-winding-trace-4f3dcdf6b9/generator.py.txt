"""Deterministic compact route-around-island renderer.

The analytic arm knows the sampled traversal direction.  The image arm does not:
it must recover the endpoint attachments and validate the intervening dark route.
Image angles use screen coordinates, so increasing angle is clockwise.
"""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

CANVAS = 512
SUPERSAMPLE = 4

PALETTES = (
    {
        "background": (248, 246, 239),
        "route": (28, 48, 70),
        "start": (38, 112, 216),
        "finish": (231, 105, 56),
        "island": (244, 193, 66),
        "frame": (215, 217, 213),
    },
    {
        "background": (246, 244, 238),
        "route": (38, 51, 73),
        "start": (33, 105, 205),
        "finish": (222, 96, 48),
        "island": (239, 184, 55),
        "frame": (210, 214, 211),
    },
)


def sample_scene(seed: int) -> dict:
    """Sample one balanced, compact scene from a single integer seed."""

    rng = random.Random(int(seed))
    centers = (96, 160, 224, 288, 352, 416)
    # Because the multiplier used by generate.py is odd, consecutive scene
    # indices alternate the class even for the public odd stress seeds.
    direction = 1 if int(seed) % 2 else -1
    scene = {
        "center_x": rng.choice(centers),
        "center_y": rng.choice(centers),
        "start_angle": round(rng.uniform(0.0, 360.0), 6),
        "sweep_degrees": round(rng.uniform(286.0, 306.0), 6),
        "direction": direction,
        "route_radius": round(rng.uniform(31.0, 33.0), 6),
        "stroke_width": round(rng.uniform(11.0, 13.0), 6),
        "island_radius": round(rng.uniform(14.0, 16.0), 6),
        "palette_index": rng.randrange(len(PALETTES)),
    }
    return scene


def _route_points(scene: dict, scale: int = 1) -> list[tuple[float, float]]:
    cx = float(scene["center_x"]) * scale
    cy = float(scene["center_y"]) * scale
    radius = float(scene["route_radius"]) * scale
    start = float(scene["start_angle"])
    sweep = float(scene["sweep_degrees"])
    direction = int(scene["direction"])
    count = max(80, int(sweep * 1.5))
    points = []
    for index in range(count + 1):
        fraction = index / count
        angle = math.radians(start + direction * sweep * fraction)
        # A tiny, answer-balanced radial undulation makes this an actual trace
        # rather than a perfect-circle direction glyph while retaining clearance.
        radial = radius + 0.9 * scale * math.sin(4.0 * math.pi * fraction)
        points.append((cx + radial * math.cos(angle), cy + radial * math.sin(angle)))
    return points


def _ellipse(draw: ImageDraw.ImageDraw, center: tuple[float, float], radius: float, fill) -> None:
    x, y = center
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=fill)


def render(scene: dict) -> Image.Image:
    """Render a single 512px RGB PNG with a small translated tracing tile."""

    palette = PALETTES[int(scene["palette_index"])]
    # Access every visible latent directly or through _route_points; this also
    # keeps the protocol's rendered-field audit transparent.
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    island_radius = float(scene["island_radius"])
    stroke_width = float(scene["stroke_width"])
    route_radius = float(scene["route_radius"])
    start_angle = float(scene["start_angle"])
    sweep_degrees = float(scene["sweep_degrees"])
    direction = int(scene["direction"])
    _ = (route_radius, start_angle, sweep_degrees, direction)

    ss = SUPERSAMPLE
    image = Image.new("RGB", (CANVAS * ss, CANVAS * ss), palette["background"])
    draw = ImageDraw.Draw(image)

    # A faint compact frame helps a human locate the small tile but is ignored by
    # both gold and oracle.
    frame_radius = 43.0 * ss
    draw.rounded_rectangle(
        (
            cx * ss - frame_radius,
            cy * ss - frame_radius,
            cx * ss + frame_radius,
            cy * ss + frame_radius,
        ),
        radius=7 * ss,
        outline=palette["frame"],
        width=1 * ss,
    )

    points = _route_points(scene, ss)
    draw.line(
        points,
        fill=palette["route"],
        width=max(1, round(stroke_width * ss)),
        joint="curve",
    )

    # The island is a solid, area-like reference component with generous gap to
    # the route.  Endpoint rings overwrite only the caps of the route.
    _ellipse(draw, (cx * ss, cy * ss), island_radius * ss, palette["island"])
    marker_outer = 7.2 * ss
    marker_inner = 3.2 * ss
    start, finish = points[0], points[-1]
    _ellipse(draw, start, marker_outer, palette["start"])
    _ellipse(draw, start, marker_inner, palette["background"])
    _ellipse(draw, finish, marker_outer, palette["finish"])
    _ellipse(draw, finish, marker_inner, palette["background"])

    return image.resize((CANVAS, CANVAS), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "clockwise" if int(scene["direction"]) > 0 else "counterclockwise"


def margin(scene: dict) -> float:
    """Angular distance from the zero-winding runner-up boundary, in degrees."""

    return float(scene["sweep_degrees"])


def is_quarantined(scene: dict) -> bool:
    route_gap = float(scene["route_radius"]) - float(scene["island_radius"])
    return (
        float(scene["sweep_degrees"]) < 220.0
        or float(scene["stroke_width"]) < 8.0
        or route_gap - float(scene["stroke_width"]) / 2.0 < 7.0
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
