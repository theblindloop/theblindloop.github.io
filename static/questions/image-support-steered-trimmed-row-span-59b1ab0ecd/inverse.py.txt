"""Independent image-only inverse arm for trimmed-row spans."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[float, float, int]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            cy, cx = queue.popleft()
            xs.append(cx)
            ys.append(cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= 100:
            found.append((float(np.mean(xs)), float(np.mean(ys)), len(xs)))
    return found


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    # Dots are the only saturated regions; gray grouping halos have low channel range.
    saturation = arr.max(axis=2) - arr.min(axis=2)
    darkness = arr.mean(axis=2) < 190
    components = _components((saturation > 45) & darkness)
    if len(components) != 10:
        raise ValueError(f"expected 10 dot components, found {len(components)}")
    areas = [item[2] for item in components]
    if max(areas) - min(areas) > 24:
        raise ValueError("one or more dots is materially damaged")
    components.sort(key=lambda item: item[1])
    top = sorted(item[0] for item in components[:5])
    bottom = sorted(item[0] for item in components[5:])
    if max(item[1] for item in components[:5]) >= min(item[1] for item in components[5:]):
        raise ValueError("rows are not separable")
    top_span = top[3] - top[1]
    bottom_span = bottom[3] - bottom[1]
    if abs(top_span - bottom_span) < 12.0:
        raise ValueError("pixel measurement is too close to the boundary")
    return "top" if top_span > bottom_span else "bottom"
