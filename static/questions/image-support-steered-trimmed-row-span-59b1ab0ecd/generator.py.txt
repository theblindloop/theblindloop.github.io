"""Deterministic latent sampler and raster renderer."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 480
QUARANTINE_PX = 18.0
PALETTES = (
    ("#137f83", "#d05b4f"),
    ("#355fbd", "#b06b17"),
    ("#7a4ca5", "#27865f"),
)


def _row(rng: random.Random, span: int) -> list[list[int]]:
    center = rng.randint(300, 340)
    left = center - span // 2
    right = left + span
    middle = rng.randint(left + 31, right - 31)
    outer_left = left - rng.randint(32, 55)
    outer_right = right + rng.randint(32, 55)
    return [[outer_left, 0], [left, 0], [middle, 0], [right, 0], [outer_right, 0]]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    smaller = rng.randint(88, 118)
    difference = rng.randint(28, 72)
    larger = smaller + difference
    top_wider = int(seed) % 2 == 0
    top = _row(rng, larger if top_wider else smaller)
    bottom = _row(rng, smaller if top_wider else larger)
    shift_y = rng.randint(-8, 8)
    for point in top:
        point[1] = 145 + shift_y
    for point in bottom:
        point[1] = 335 + shift_y
    colors = list(rng.choice(PALETTES))
    if rng.random() < 0.5:
        colors.reverse()
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "top_points": top,
        "bottom_points": bottom,
        "radius": 13,
        "palette": colors,
    }


def _trimmed_span(points: list[list[int]]) -> float:
    xs = sorted(float(point[0]) for point in points)
    return xs[3] - xs[1]


def analytic_gold(scene: dict) -> str:
    top_span = _trimmed_span(scene["top_points"])
    bottom_span = _trimmed_span(scene["bottom_points"])
    return "top" if top_span > bottom_span else "bottom"


def margin(scene: dict) -> float:
    return abs(_trimmed_span(scene["top_points"]) - _trimmed_span(scene["bottom_points"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def render(scene: dict) -> Image.Image:
    scale = 4
    width = int(scene["width"])
    height = int(scene["height"])
    radius = int(scene["radius"])
    colors = scene["palette"]
    image = Image.new("RGB", (width * scale, height * scale), "#fafbf9")
    draw = ImageDraw.Draw(image)
    # Faint row halos aid visual grouping while remaining separate from every dot.
    for y in (scene["top_points"][0][1], scene["bottom_points"][0][1]):
        draw.rounded_rectangle(
            (130 * scale, (y - 28) * scale, 510 * scale, (y + 28) * scale),
            radius=18 * scale,
            fill="#f0f2ee",
        )
    for row, color in ((scene["top_points"], colors[0]), (scene["bottom_points"], colors[1])):
        for x, y in row:
            draw.ellipse(
                ((x - radius) * scale, (y - radius) * scale,
                 (x + radius) * scale, (y + radius) * scale),
                fill=color,
            )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["top_points"] = list(reversed(twin["top_points"]))
    twin["bottom_points"] = list(reversed(twin["bottom_points"]))
    return [("reverse_stored_point_order", twin)]
