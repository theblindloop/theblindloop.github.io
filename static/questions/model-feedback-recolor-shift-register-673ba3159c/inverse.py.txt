"""Independent pixel-only oracle for the recolor-shift register world.

The oracle receives a PNG and nothing else. It rediscovers the three drawn
bands from ink alone, reads the color chain, the initial register and the
operation program, executes the machine, and reports the color in slot 1.

It shares no code, no constants and no layout table with the renderer: bands,
marks and icon identities are recovered by projection profiles.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_REFERENCE = {
    "red": (214, 39, 40),
    "green": (44, 160, 44),
    "blue": (31, 119, 180),
    "yellow": (255, 200, 10),
}
_FALLBACK = "blue"


def _runs(flags: np.ndarray, min_len: int = 3) -> list[tuple[int, int]]:
    """Contiguous ``True`` runs of ``flags`` as inclusive-exclusive spans."""

    spans: list[tuple[int, int]] = []
    start = None
    for index, value in enumerate(flags):
        if value and start is None:
            start = index
        elif not value and start is not None:
            if index - start >= min_len:
                spans.append((start, index))
            start = None
    if start is not None and len(flags) - start >= min_len:
        spans.append((start, len(flags)))
    return spans


def _name_of(pixels: np.ndarray) -> str:
    mean = pixels.reshape(-1, 3).mean(axis=0)
    best, best_distance = _FALLBACK, None
    for name, rgb in sorted(_REFERENCE.items()):
        distance = float(((mean - np.array(rgb, dtype=float)) ** 2).sum())
        if best_distance is None or distance < best_distance:
            best, best_distance = name, distance
    return best


def _band_marks(
    arr: np.ndarray, mask: np.ndarray, span: tuple[int, int]
) -> list[tuple[int, int]]:
    sub = mask[span[0] : span[1], :]
    return _runs(sub.any(axis=0), min_len=6)


def _classify_icon(mask: np.ndarray, span: tuple[int, int], columns: tuple[int, int]) -> str:
    sub = mask[span[0] : span[1], columns[0] : columns[1]]
    heights = sub.sum(axis=0).astype(float)
    if heights.size < 5 or heights.max() <= 0:
        return "R"
    peak = heights.max()
    left = heights[: max(1, heights.size // 8)].max()
    right = heights[-max(1, heights.size // 8) :].max()
    middle = heights[heights.size // 2 - 1 : heights.size // 2 + 2].max()
    if middle >= 0.85 * peak and left <= 0.6 * peak and right <= 0.6 * peak:
        return "B"
    return "R" if left > right else "L"


def decision_from_image(image) -> str:
    """Color in slot 1 after executing the drawn program, from pixels alone."""

    if not isinstance(image, Image.Image):
        image = Image.open(image)
    arr = np.asarray(image.convert("RGB")).astype(np.int16)
    high = arr.max(axis=2)
    low = arr.min(axis=2)
    colored = (high - low) > 60
    black = high < 80
    ink = colored | black

    bands = _runs(ink.any(axis=1), min_len=8)
    if len(bands) < 3:
        return _FALLBACK
    bands = sorted(bands, key=lambda span: span[1] - span[0], reverse=True)[:3]
    bands = sorted(bands, key=lambda span: span[0])
    chain_band, register_band, program_band = bands

    chain_cols = _band_marks(arr, colored, chain_band)
    register_cols = _band_marks(arr, colored, register_band)
    program_cols = _band_marks(arr, black, program_band)
    if len(chain_cols) != 4 or len(register_cols) != 4 or not program_cols:
        return _FALLBACK

    def read_colors(span: tuple[int, int], columns: list[tuple[int, int]]) -> list[str]:
        names = []
        for c0, c1 in columns:
            window = arr[span[0] : span[1], c0:c1]
            window_mask = colored[span[0] : span[1], c0:c1]
            if not window_mask.any():
                return []
            names.append(_name_of(window[window_mask]))
        return names

    chain = read_colors(chain_band, chain_cols)
    register = read_colors(register_band, register_cols)
    if len(chain) != 4 or len(register) != 4 or len(set(chain)) != 4:
        return _FALLBACK

    ops = [_classify_icon(black, program_band, columns) for columns in program_cols]

    state = list(register)
    for op in ops:
        if op == "R":
            state = [state[-1]] + state[:-1]
        elif op == "L":
            state = state[1:] + [state[0]]
        else:
            index = chain.index(state[0])
            state = [chain[(index + 1) % len(chain)]] + state[1:]
    return state[0]
