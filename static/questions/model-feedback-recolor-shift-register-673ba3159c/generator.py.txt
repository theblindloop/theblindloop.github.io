"""Renderer and analytic gold for the recolor-shift register world.

The scene is a tiny sequential machine drawn in three horizontal bands:

* a *color chain* of four colored squares joined by black arrows (the chain
  order is permuted per scene, so it must be read, never assumed);
* a *register strip* of four colored disks occupying slots 1..4;
* a left-to-right row of *operation cards* whose black icons are the program.

Executing the program is a state transition: shift icons permute the disks,
the plus icon recolors whichever disk currently occupies slot 1 by one step
along the chain. Because the recolor acts on a position and the shifts move
disks through that position, no net-permutation shortcut exists; the machine
must be stepped.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 780
HEIGHT = 600

BACKGROUND = (255, 255, 255)
INK = (0, 0, 0)
GRAY = (170, 170, 170)

COLORS: dict[str, tuple[int, int, int]] = {
    "red": (214, 39, 40),
    "green": (44, 160, 44),
    "blue": (31, 119, 180),
    "yellow": (255, 200, 10),
}
COLOR_NAMES = ("blue", "green", "red", "yellow")

SLOTS = 4
OPS = 6
OP_CODES = ("L", "R", "B")

CHAIN_TOP = 46
CHAIN_SIDE = 58
CHAIN_STEP = 150
CHAIN_LEFT = 70
WRAP_Y = 146

REG_CY = 255
REG_D = 70
REG_STEP = 160
REG_LEFT = 120

CARD_TOP = 400
CARD_SIDE = 80
CARD_STEP = 118
CARD_LEFT = 50

MIN_ICON = 36
QUARANTINE_PX = 11


# ---------------------------------------------------------------- sampling


def sample_scene(seed: int) -> dict:
    """Sample one machine scene deterministically from ``seed``."""

    rng = random.Random(seed * 2654435761 % (2**61 - 1))

    chain = list(COLOR_NAMES)
    rng.shuffle(chain)

    while True:
        register = [rng.choice(chain) for _ in range(SLOTS)]
        if len(set(register)) >= 3:
            break

    while True:
        ops = [rng.choice(OP_CODES) for _ in range(OPS)]
        bumps = ops.count("B")
        if 2 <= bumps <= 3 and len(ops) - bumps >= 3:
            break

    jitter = {
        "chain_dx": [rng.randint(-6, 6) for _ in range(SLOTS)],
        "reg_dx": [rng.randint(-7, 7) for _ in range(SLOTS)],
        "card_dx": [rng.randint(-16, 16) for _ in range(OPS)],
        "icon": [rng.randint(48, 58) for _ in range(OPS)],
    }
    return {"chain": chain, "register": register, "ops": ops, "jitter": jitter}


# ---------------------------------------------------------------- machine


def _run_program(chain: list[str], register: list[str], ops: list[str]) -> list[str]:
    state = list(register)
    for op in ops:
        if op == "R":
            state = [state[-1]] + state[:-1]
        elif op == "L":
            state = state[1:] + [state[0]]
        elif op == "B":
            index = chain.index(state[0])
            state = [chain[(index + 1) % len(chain)]] + state[1:]
        else:  # pragma: no cover - guarded by sampling
            raise ValueError(f"unknown operation {op!r}")
    return state


def analytic_gold(scene: dict) -> str:
    """Color occupying slot 1 after the drawn program has been executed."""

    return _run_program(scene["chain"], scene["register"], scene["ops"])[0]


# ---------------------------------------------------------------- geometry


def _chain_boxes(scene: dict) -> list[tuple[int, int, int, int]]:
    dx = scene["jitter"]["chain_dx"]
    boxes = []
    for i in range(SLOTS):
        x0 = CHAIN_LEFT + i * CHAIN_STEP + dx[i]
        boxes.append((x0, CHAIN_TOP, x0 + CHAIN_SIDE, CHAIN_TOP + CHAIN_SIDE))
    return boxes


def _register_boxes(scene: dict) -> list[tuple[int, int, int, int]]:
    dx = scene["jitter"]["reg_dx"]
    boxes = []
    for i in range(SLOTS):
        cx = REG_LEFT + i * REG_STEP + dx[i]
        boxes.append((cx - REG_D // 2, REG_CY - REG_D // 2, cx + REG_D // 2, REG_CY + REG_D // 2))
    return boxes


def _icon_boxes(scene: dict) -> list[tuple[int, int, int, int]]:
    dx = scene["jitter"]["card_dx"]
    sizes = scene["jitter"]["icon"]
    boxes = []
    for i in range(OPS):
        cx = CARD_LEFT + i * CARD_STEP + CARD_SIDE // 2 + dx[i]
        cy = CARD_TOP + CARD_SIDE // 2
        half = sizes[i] // 2
        boxes.append((cx - half, cy - half, cx + half, cy + half))
    return boxes


def _card_boxes(scene: dict) -> list[tuple[int, int, int, int]]:
    dx = scene["jitter"]["card_dx"]
    boxes = []
    for i in range(OPS):
        x0 = CARD_LEFT + i * CARD_STEP + dx[i]
        boxes.append((x0, CARD_TOP, x0 + CARD_SIDE, CARD_TOP + CARD_SIDE))
    return boxes


def _gaps(boxes: list[tuple[int, int, int, int]]) -> list[int]:
    return [boxes[i + 1][0] - boxes[i][2] for i in range(len(boxes) - 1)]


def margin(scene: dict) -> float:
    """Smallest decisive separation, in pixels, in the drawn scene.

    The decision needs every chain square, every register disk and every
    operation icon to be individually resolvable, so the margin is the
    smallest horizontal clearance between neighbouring marks, the smallest
    vertical clearance between the three bands, and the amount by which the
    smallest icon exceeds the minimum legible icon size.
    """

    chain = _chain_boxes(scene)
    register = _register_boxes(scene)
    icons = _icon_boxes(scene)
    cards = _card_boxes(scene)

    values: list[float] = []
    values.extend(_gaps(chain))
    values.extend(_gaps(register))
    values.extend(_gaps(icons))
    values.extend(_gaps(cards))
    values.append(min(box[1] for box in register) - WRAP_Y)
    values.append(min(box[1] for box in icons) - max(box[3] for box in register))
    values.append(min(scene["jitter"]["icon"]) - MIN_ICON)
    return float(min(values))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list:
    """No render-invariant relabelling exists: every field is drawn."""

    _ = scene
    return []


# ---------------------------------------------------------------- drawing


def _arrow(draw: ImageDraw.ImageDraw, x0: int, y0: int, x1: int, y1: int, head: int = 11) -> None:
    draw.line((x0, y0, x1, y1), fill=INK, width=5)
    if y0 == y1:
        sign = 1 if x1 >= x0 else -1
        draw.polygon(
            [(x1, y1), (x1 - sign * head, y1 - head), (x1 - sign * head, y1 + head)], fill=INK
        )
    else:
        sign = 1 if y1 >= y0 else -1
        draw.polygon(
            [(x1, y1), (x1 - head, y1 - sign * head), (x1 + head, y1 - sign * head)], fill=INK
        )


def _draw_icon(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], code: str) -> None:
    x0, y0, x1, y1 = box
    cy = (y0 + y1) // 2
    if code == "R":
        draw.polygon([(x0, y0), (x0, y1), (x1, cy)], fill=INK)
    elif code == "L":
        draw.polygon([(x1, y0), (x1, y1), (x0, cy)], fill=INK)
    else:
        thickness = (x1 - x0) // 3
        cx = (x0 + x1) // 2
        half = thickness // 2
        draw.rectangle((x0, cy - half, x1, cy + half), fill=INK)
        draw.rectangle((cx - half, y0, cx + half, y1), fill=INK)


def render(scene: dict) -> Image.Image:
    """Rasterize the machine. Deterministic and free of antialiasing."""

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    chain = _chain_boxes(scene)
    for index, box in enumerate(chain):
        draw.rectangle(box, fill=COLORS[scene["chain"][index]])
    for index in range(SLOTS - 1):
        mid = (chain[index][1] + chain[index][3]) // 2
        _arrow(draw, chain[index][2] + 8, mid, chain[index + 1][0] - 8, mid)
    last, first = chain[-1], chain[0]
    lx = (last[0] + last[2]) // 2
    fx = (first[0] + first[2]) // 2
    draw.line((lx, last[3] - 2, lx, WRAP_Y), fill=INK, width=5)
    draw.line((lx, WRAP_Y, fx, WRAP_Y), fill=INK, width=5)
    _arrow(draw, fx, WRAP_Y, fx, first[3] + 8)

    register = _register_boxes(scene)
    frame = (
        min(box[0] for box in register) - 24,
        REG_CY - REG_D // 2 - 22,
        max(box[2] for box in register) + 24,
        REG_CY + REG_D // 2 + 22,
    )
    draw.rectangle(frame, outline=GRAY, width=4)
    for index, box in enumerate(register):
        draw.ellipse(box, fill=COLORS[scene["register"][index]])

    for index, card in enumerate(_card_boxes(scene)):
        draw.rectangle(card, outline=GRAY, width=4)
    for index, box in enumerate(_icon_boxes(scene)):
        _draw_icon(draw, box, scene["ops"][index])

    return image
