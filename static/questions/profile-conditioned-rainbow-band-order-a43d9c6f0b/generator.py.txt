"""Latent-scene sampler and RGB renderer for the rainbow band-order world.

Five concentric half-annulus bands are drawn above a grey ground line, in the
five fixed colours red, yellow, green, blue and violet. Every band colour is
present in every image exactly once: the ONLY thing that changes between answer
classes is which colour is assigned to which radial slot. The geometry of a
scene (centre, outer radius, the five band thicknesses, the gap width, and the
achromatic decoration) is sampled independently of that assignment, so the two
variants of a scene are matched in scale, layout, palette, total ink and the
multiset of band areas.

The familiar prior under test is that the outermost band of a rainbow is red.
The canonical variant satisfies it; the counterfactual variants contradict it
with byte-identical wording and matched geometry.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "rainbow_band_order-renderer-0.1.0"

WIDTH = 480
HEIGHT = 360
SS = 3                      # supersampling factor; BOX downsample is exact
N_BANDS = 5
QUARANTINE_MARGIN = 12.0    # outermost-band thickness (px) below which we quarantine
MIN_SAMPLED_MARGIN = 7      # sampler never emits a band thinner than this
MIN_HOLE = 26               # the innermost white hole stays clearly visible

WHITE = (255, 255, 255)
GROUND_GREY = (140, 140, 140)
TICK_GREY = (96, 96, 96)
FRAME_GREY = (178, 178, 178)

# Public band colours, in the order the question names them. Every hue is at
# least 40 degrees from every other, and every entry has chroma >= 120, so the
# pixel oracle can name a band from the raster without any renderer constant.
COLOR_NAMES = ("red", "yellow", "green", "blue", "violet")
PALETTE = (
    (214, 30, 30),      # red      hue   0
    (240, 190, 25),     # yellow   hue  46
    (28, 150, 66),      # green    hue 139
    (36, 84, 210),      # blue     hue 223
    (146, 46, 180),     # violet   hue 285
)


def sample_scene(seed: int) -> dict:
    """Sample one scene.

    The outermost colour class is ``seed % N_BANDS``. That makes any contiguous
    block of scene seeds cover all five classes evenly, and it is independent of
    every geometric field, so no size, position or ink cue predicts the class.
    """
    rng = random.Random(seed)

    outer_class = seed % N_BANDS
    rest = [c for c in range(N_BANDS) if c != outer_class]
    rng.shuffle(rest)
    order = [outer_class] + rest

    # Decided once, outside the rejection loop, so that the rejection sampler
    # cannot bias how often near-boundary scenes appear.
    near_boundary = rng.random() < 0.18
    while True:
        gap = rng.randint(5, 9)
        thick = [rng.randint(14, 30) for _ in range(N_BANDS)]
        if near_boundary:
            # Near-boundary draw: the outermost band is thin, so the visible
            # evidence that separates it from the runner-up band is small.
            thick[0] = rng.randint(MIN_SAMPLED_MARGIN, 13)
        r_outer = rng.randint(146, 172)
        if r_outer - sum(thick) - (N_BANDS - 1) * gap >= MIN_HOLE:
            break

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "cx": rng.randint(226, 254),
        "cy": rng.randint(292, 306),
        "r_outer": r_outer,
        "thick": thick,
        "gap": gap,
        "order": order,
        "decor": sorted(rng.sample(range(40, WIDTH - 40, 12), 6)),
    }


def analytic_gold(scene: dict) -> str:
    """Gold depends on exactly one scene field, and that field is drawn."""
    return COLOR_NAMES[int(scene["order"][0])]


def margin(scene: dict) -> float:
    """Visible decision margin: the drawn thickness of the outermost band.

    The runner-up decision is the colour of the second band. It wins exactly
    when the outermost band stops being visible, so the pixels of that band are
    the whole evidence and their radial extent is the margin.
    """
    return float(int(scene["thick"][0]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist: every field is rendered (see candidate.json)."""
    return []


def render(scene: dict) -> Image.Image:
    w, h = int(scene["width"]), int(scene["height"])
    cx, cy = int(scene["cx"]), int(scene["cy"])
    gap = int(scene["gap"])
    thick = [int(t) for t in scene["thick"]]
    order = [int(c) for c in scene["order"]]

    img = Image.new("RGB", (w * SS, h * SS), WHITE)
    d = ImageDraw.Draw(img)

    def half_disk(radius: int, fill: tuple[int, int, int]) -> None:
        if radius <= 0:
            return
        box = [(cx - radius) * SS, (cy - radius) * SS,
               (cx + radius) * SS - 1, (cy + radius) * SS - 1]
        d.pieslice(box, 180, 360, fill=fill)

    def rect(x0: int, y0: int, x1: int, y1: int, fill: tuple[int, int, int]) -> None:
        d.rectangle([x0 * SS, y0 * SS, x1 * SS - 1, y1 * SS - 1], fill=fill)

    # Bands, painted outermost first: each colour disk is immediately hollowed
    # out by a white disk, so the next band is drawn inside the hole.
    radius = int(scene["r_outer"])
    for slot in range(N_BANDS):
        half_disk(radius, PALETTE[order[slot]])
        radius -= thick[slot]
        half_disk(radius, WHITE)
        radius -= gap

    # Fixed achromatic decoration: ground line, ticks under it, page frame.
    rect(0, cy, w, cy + 3, GROUND_GREY)
    for off in scene["decor"]:
        off = int(off)
        rect(off, cy + 7, off + 5, cy + 20, TICK_GREY)
    rect(0, 0, w, 3, FRAME_GREY)
    rect(0, h - 3, w, h, FRAME_GREY)
    rect(0, 0, 3, h, FRAME_GREY)
    rect(w - 3, 0, w, h, FRAME_GREY)

    return img.resize((w, h), Image.BOX)


# --- gold-preserving nuisance transforms (used by verify.py and tests) --------

def translate(scene: dict, dx: int, dy: int) -> dict:
    """Move the whole arc, ground line and ticks together."""
    alt = dict(scene)
    alt["cx"] = int(scene["cx"]) + dx
    alt["cy"] = int(scene["cy"]) + dy
    alt["decor"] = [int(v) + dx for v in scene["decor"]]
    return alt


def permute_decor(scene: dict, seed: int) -> dict:
    """Redraw the achromatic distractor ticks somewhere else."""
    rng = random.Random(seed * 7919 + 13)
    alt = dict(scene)
    alt["decor"] = sorted(rng.sample(range(40, WIDTH - 40, 12), 6))
    return alt


def swap_inner_colors(scene: dict) -> dict:
    """Label-preserving palette permutation: swap two INNER bands' colours.

    The outermost slot is untouched, so the answer is unchanged while three of
    the five colours move to different radii.
    """
    order = [int(c) for c in scene["order"]]
    order[2], order[3] = order[3], order[2]
    alt = dict(scene)
    alt["order"] = order
    return alt


def rethicken(scene: dict) -> dict:
    """Label-preserving geometry change: thicken every non-outermost band."""
    thick = [int(t) for t in scene["thick"]]
    alt = dict(scene)
    alt["thick"] = [thick[0]] + [max(10, t - 3) for t in thick[1:]]
    return alt


# --- label-transforming transform --------------------------------------------

def rotate_order(scene: dict, k: int) -> dict:
    """Cyclically rotate the radial colour sequence by k slots outward."""
    order = [int(c) for c in scene["order"]]
    k %= N_BANDS
    alt = dict(scene)
    alt["order"] = order[k:] + order[:k]
    return alt


def expected_rotated_gold(scene: dict, k: int) -> str:
    order = [int(c) for c in scene["order"]]
    return COLOR_NAMES[order[k % N_BANDS]]
