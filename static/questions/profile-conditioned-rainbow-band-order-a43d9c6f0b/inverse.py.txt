"""Pixel-only inverse arm for the rainbow band-order world.

This module receives a PNG and nothing else. It never imports the renderer,
prompts, generator, verifier or any analytic-gold code, and it never accepts a
latent scene. It shares no geometry constants with the renderer beyond the
PUBLIC visual contract stated in the question text: an arch of five coloured
bands, one inside the other, over achromatic decoration, with the five colours
named red, yellow, green, blue and violet.

Method, independent of how the image was produced:

1. Label a pixel chromatic iff ``max(R,G,B) - min(R,G,B) >= SAT_MIN``. The white
   page, the grey ground line, the grey ticks and the grey frame all drop out;
   only the coloured bands survive.
2. Locate the arch from that mask alone: its flat bottom edge is the lowest
   chromatic row, and its centre sits midway between the leftmost and rightmost
   chromatic pixels, because the widest band spans the full diameter.
3. Cast rays outward from that centre at several angles. Along each ray the
   bands appear as runs of chromatic pixels; the run at the largest radius is
   the outermost band.
4. Name that run's median colour by hue, using only the five public colour
   names. Answer when every ray agrees and the winning run is thick enough to
   be a band rather than an antialiasing crumb; abstain otherwise.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "rainbow_band_order-oracle-0.1.0"

ABSTAIN = "abstain"

SAT_MIN = 60          # renderer palette chroma >= 120; grey decoration chroma 0
MIN_VALUE = 60        # reject near-black chromatic mush
MIN_INK = 2000        # smallest plausible amount of band ink
MIN_RUN = 4           # runs shorter than this are antialiasing crumbs
MIN_RUNS_ON_RAY = 4   # a five-band arch shows five runs on every ray
MIN_SPAN = 120        # the widest band spans at least this many pixels
ANGLES_DEG = (28.0, 52.0, 76.0, 90.0, 104.0, 128.0, 152.0)

# Public colour vocabulary, as hue windows. The windows are separated by unused
# guard bands, so an off-palette colour is named by nobody and forces abstention.
HUE_WINDOWS = (
    ("red", -22.0, 22.0),
    ("yellow", 26.0, 72.0),
    ("green", 108.0, 172.0),
    ("blue", 198.0, 252.0),
    ("violet", 262.0, 312.0),
)


def _hue(rgb: tuple[float, float, float]) -> float | None:
    r, g, b = rgb
    hi, lo = max(r, g, b), min(r, g, b)
    delta = hi - lo
    if delta < SAT_MIN or hi < MIN_VALUE:
        return None
    if hi == r:
        hue = 60.0 * (((g - b) / delta) % 6.0)
    elif hi == g:
        hue = 60.0 * (((b - r) / delta) + 2.0)
    else:
        hue = 60.0 * (((r - g) / delta) + 4.0)
    return hue % 360.0


def _color_name(rgb: tuple[float, float, float]) -> str | None:
    hue = _hue(rgb)
    if hue is None:
        return None
    for name, lo, hi in HUE_WINDOWS:
        h = hue - 360.0 if hue > 180.0 and lo < 0.0 else hue
        if lo <= h <= hi:
            return name
    return None


def _ray_runs(mask: np.ndarray, cx: int, cy: int, angle_deg: float,
              r_max: int) -> list[tuple[int, int]]:
    """Runs of chromatic pixels along one ray, listed from outside inward."""
    dx = float(np.cos(np.radians(angle_deg)))
    dy = -float(np.sin(np.radians(angle_deg)))
    h, w = mask.shape
    hits: list[bool] = []
    for r in range(r_max, 2, -1):
        x = int(round(cx + r * dx))
        y = int(round(cy + r * dy))
        hits.append(0 <= x < w and 0 <= y < h and bool(mask[y, x]))
    runs: list[tuple[int, int]] = []
    start = None
    for i, v in enumerate(hits):
        if v and start is None:
            start = i
        elif not v and start is not None:
            runs.append((start, i))
            start = None
    if start is not None:
        runs.append((start, len(hits)))
    return [(r_max - end + 1, r_max - start) for start, end in runs]


def _median_rgb(arr: np.ndarray, cx: int, cy: int, angle_deg: float,
                r_lo: int, r_hi: int) -> tuple[float, float, float]:
    dx = float(np.cos(np.radians(angle_deg)))
    dy = -float(np.sin(np.radians(angle_deg)))
    h, w = arr.shape[:2]
    picked = []
    for r in range(r_lo, r_hi + 1):
        x = int(round(cx + r * dx))
        y = int(round(cy + r * dy))
        if 0 <= x < w and 0 <= y < h:
            picked.append(arr[y, x])
    if not picked:
        return (0.0, 0.0, 0.0)
    stack = np.asarray(picked, dtype=np.float64)
    med = np.median(stack, axis=0)
    return (float(med[0]), float(med[1]), float(med[2]))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    hi = arr.max(axis=-1)
    lo = arr.min(axis=-1)
    mask = ((hi - lo) >= SAT_MIN) & (hi >= MIN_VALUE)
    if int(mask.sum()) < MIN_INK:
        return ABSTAIN

    ys, xs = np.nonzero(mask)
    x_lo, x_hi = int(xs.min()), int(xs.max())
    if x_hi - x_lo < MIN_SPAN:
        return ABSTAIN
    cx = (x_lo + x_hi) // 2
    cy = int(ys.max())

    h, w = mask.shape
    r_max = min(cx, w - 1 - cx, cy) - 2
    if r_max < MIN_SPAN // 2:
        return ABSTAIN

    votes: set[str] = set()
    for angle in ANGLES_DEG:
        runs = _ray_runs(mask, cx, cy, angle, r_max)
        runs = [(a, b) for a, b in runs if b - a + 1 >= MIN_RUN]
        if len(runs) < MIN_RUNS_ON_RAY:
            return ABSTAIN
        r_lo, r_hi = max(runs, key=lambda ab: ab[1])
        name = _color_name(_median_rgb(arr, cx, cy, angle, r_lo + 1, r_hi - 1))
        if name is None:
            return ABSTAIN
        votes.add(name)
    if len(votes) != 1:
        return ABSTAIN
    return votes.pop()
