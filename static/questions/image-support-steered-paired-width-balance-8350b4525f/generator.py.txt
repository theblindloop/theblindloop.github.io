"""Deterministic latent sampler and raster renderer."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

CORAL = (231, 88, 72)
BLUE = (42, 105, 184)
GRAY = (205, 207, 205)
BACKGROUND = (250, 249, 245)


def _display_pairs(scene: dict) -> tuple[list[int], list[int]]:
    coral = [int(v) for v in scene["coral_widths"]]
    blue = [int(v) for v in scene["blue_widths"]]
    if bool(scene["palette_swap"]):
        return blue, coral
    return coral, blue


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    desired = "coral" if seed % 2 == 0 else "blue"
    small = rng.randint(1, 3)
    # Every sixth scene approaches the visible equality boundary without entering
    # quarantine; the rest have broad margins.
    gap = 5 if seed % 6 == 0 else rng.randint(7, 11)
    large = small + gap

    def widths(diff: int) -> list[int]:
        base = rng.randint(14, 30 - diff)
        pair = [base, base + diff]
        if rng.random() < 0.5:
            pair.reverse()
        return pair

    display_coral = widths(small if desired == "coral" else large)
    display_blue = widths(small if desired == "blue" else large)
    palette_swap = bool(rng.getrandbits(1))
    latent_coral, latent_blue = (
        (display_blue, display_coral) if palette_swap else (display_coral, display_blue)
    )

    # Put the two decisive clusters one and a half 8x8-grid cells apart diagonally. Their
    # absolute location and diagonal direction vary while their joint support
    # remains focal. Small jitter avoids an exact-coordinate shortcut.
    sx = -1
    sy = -1
    cx = rng.choice((224, 288, 352, 416))
    cy = rng.choice((224, 288, 352, 416))
    bx, by = cx + 96 * sx, cy + 96 * sy
    if not (48 <= bx <= 464):
        sx = -sx
        bx = cx + 96 * sx
    if not (48 <= by <= 464):
        sy = -sy
        by = cy + 96 * sy
    origin = [cx, cy, bx, by]

    # Remote pale marks create a figure-like measurement field but carry no label
    # information. Reversing their latent order is a pixel-identical symmetry.
    context_marks = []
    for _ in range(8):
        for _attempt in range(100):
            x, y = rng.randint(25, 487), rng.randint(25, 487)
            if min((x - origin[0]) ** 2 + (y - origin[1]) ** 2,
                   (x - origin[2]) ** 2 + (y - origin[3]) ** 2) > 75 ** 2:
                context_marks.append([x, y, rng.choice((5, 7, 9))])
                break
    return {
        "coral_widths": latent_coral,
        "blue_widths": latent_blue,
        "bar_height": 9,
        "origin": origin,
        "palette_swap": palette_swap,
        "context_marks": context_marks,
        "width": 512,
        "height": 512,
        "background": list(BACKGROUND),
    }


def _draw_pair(draw: ImageDraw.ImageDraw, center: tuple[int, int], widths: list[int], color: tuple[int, int, int], height: int) -> None:
    cx, cy = center
    for y, width in zip((cy - 13, cy + 13), widths, strict=True):
        left = cx - width // 2
        right = left + width - 1
        draw.rectangle((left, y - height // 2, right, y + height // 2), fill=color)


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width, height), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    for x, y, radius in scene["context_marks"]:
        r = int(radius)
        draw.ellipse((int(x) - r, int(y) - r, int(x) + r, int(y) + r), outline=GRAY, width=2)
    coral_widths, blue_widths = _display_pairs(scene)
    ax, ay, bx, by = [int(v) for v in scene["origin"]]
    bar_height = int(scene["bar_height"])
    _draw_pair(draw, (ax, ay), coral_widths, CORAL, bar_height)
    _draw_pair(draw, (bx, by), blue_widths, BLUE, bar_height)
    return image


def analytic_gold(scene: dict) -> str:
    coral_widths, blue_widths = _display_pairs(scene)
    coral_mismatch = abs(coral_widths[0] - coral_widths[1])
    blue_mismatch = abs(blue_widths[0] - blue_widths[1])
    if coral_mismatch == blue_mismatch:
        raise ValueError("tied mismatch")
    return "coral" if coral_mismatch < blue_mismatch else "blue"


def margin(scene: dict) -> float:
    coral_widths, blue_widths = _display_pairs(scene)
    return float(abs(abs(coral_widths[0] - coral_widths[1]) - abs(blue_widths[0] - blue_widths[1])))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 5.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["context_marks"] = list(reversed(twin["context_marks"]))
    return [("reverse_context_order", twin)]
