"""Independent pixel-only inverse arm for paired width balance."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_COLORS = {
    "coral": np.array((231, 88, 72), dtype=np.int16),
    "blue": np.array((42, 105, 184), dtype=np.int16),
}


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]:
            continue
        queue = deque([(y0, x0)])
        seen[y0, x0] = True
        xs, ys = [], []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= 30:
            result.append((min(xs), min(ys), max(xs), max(ys), len(xs)))
    return result


def _pair_widths(array: np.ndarray, name: str) -> list[int]:
    color = _COLORS[name]
    distance = np.max(np.abs(array.astype(np.int16) - color), axis=2)
    comps = _components(distance <= 8)
    if len(comps) != 2:
        raise ValueError(f"expected two {name} bars, found {len(comps)}")
    widths = []
    for x0, y0, x1, y1, area in comps:
        width, height = x1 - x0 + 1, y1 - y0 + 1
        if height < 7 or height > 11 or width < 10 or width > 34:
            raise ValueError(f"malformed {name} bar")
        if area < 0.90 * width * height:
            raise ValueError(f"damaged {name} bar")
        widths.append(width)
    return widths


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    coral = _pair_widths(array, "coral")
    blue = _pair_widths(array, "blue")
    coral_mismatch = abs(coral[0] - coral[1])
    blue_mismatch = abs(blue[0] - blue[1])
    if coral_mismatch == blue_mismatch:
        raise ValueError("pixel mismatch tie")
    return "coral" if coral_mismatch < blue_mismatch else "blue"
