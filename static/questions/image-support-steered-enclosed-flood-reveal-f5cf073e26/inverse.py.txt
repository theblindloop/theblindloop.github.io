"""Independent raster-only flood-fill inverse; imports no world modules."""
from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

N = 11
CELL = 18
SYMBOLS = ("2", "3", "5", "6")
_ROWS = {
    "2": ("11111", "00001", "00001", "11111", "10000", "10000", "11111"),
    "3": ("11111", "00001", "00001", "11111", "00001", "00001", "11111"),
    "5": ("11111", "10000", "10000", "11111", "00001", "00001", "11111"),
    "6": ("11111", "10000", "10000", "11111", "10001", "10001", "11111"),
}
_PALETTES = (
    ((105, 70, 155), (31, 48, 78), (235, 220, 184), (235, 151, 42)),
    ((124, 64, 145), (35, 62, 85), (225, 219, 178), (244, 165, 45)),
    ((88, 76, 164), (28, 55, 73), (239, 224, 193), (224, 142, 38)),
    ((116, 73, 165), (39, 49, 88), (229, 215, 188), (242, 158, 52)),
)


def _template(symbol: str) -> set[tuple[int, int]]:
    return {(r + 2, c + 3) for r, row in enumerate(_ROWS[symbol]) for c, value in enumerate(row) if value == "1"}


def analyze(image: Image.Image) -> tuple[str, float]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    if arr.shape[:2] != (512, 512):
        raise ValueError("expected a 512 by 512 raster")
    # Locate the visible violet lattice without using latent placement.
    frame_colors = np.asarray([p[0] for p in _PALETTES], dtype=np.int16)
    d = np.linalg.norm(arr[:, :, None, :] - frame_colors[None, None, :, :], axis=3)
    frame = d.min(axis=2) < 8.0
    ys, xs = np.nonzero(frame)
    if len(xs) < 1500:
        raise ValueError("violet badge fiducial is missing")
    left, right, top, bottom = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    if not (208 <= right - left + 1 <= 214 and 208 <= bottom - top + 1 <= 214):
        raise ValueError("badge fiducial has invalid extent")
    x0, y0 = left + 7, top + 7
    samples = np.asarray([[arr[y0 + r * CELL + 8, x0 + c * CELL + 8] for c in range(N)] for r in range(N)])
    best = None
    for palette in _PALETTES:
        roles = np.asarray(palette[1:], dtype=np.int16)  # wall, pale, seed
        distances = np.linalg.norm(samples[:, :, None, :] - roles[None, None, :, :], axis=3)
        score = float(distances.min(axis=2).sum())
        if best is None or score < best[0]:
            best = (score, distances)
    assert best is not None
    mind = best[1].min(axis=2)
    if float(mind.max()) > 24.0:
        raise ValueError("one or more cell interiors are missing or corrupted")
    states = best[1].argmin(axis=2)  # 0 wall, 1 pale, 2 seed
    seeds = np.argwhere(states == 2)
    if len(seeds) != 1:
        raise ValueError("expected exactly one amber seed cell")
    start = tuple(map(int, seeds[0]))
    reached = {start}
    queue = deque([start])
    while queue:
        r, c = queue.popleft()
        for p in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
            if 0 <= p[0] < N and 0 <= p[1] < N and states[p] != 0 and p not in reached:
                reached.add(p)
                queue.append(p)
    output = {(r, c) for r in range(N) for c in range(N) if states[r, c] == 1 and (r, c) not in reached}
    if not output:
        raise ValueError("flood leaves no enclosed pale cells")
    ranked = sorted((len(output ^ _template(symbol)), symbol) for symbol in SYMBOLS)
    if ranked[0][0] > 3 or ranked[0][0] == ranked[1][0]:
        raise ValueError("revealed raster is not classifiable with a positive margin")
    return ranked[0][1], float(ranked[1][0] - ranked[0][0])


def decision_from_image(image: Image.Image) -> str:
    return analyze(image)[0]
