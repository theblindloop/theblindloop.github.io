"""Deterministic latent scene and raster renderer."""
from __future__ import annotations

import random
from collections import deque
from PIL import Image, ImageDraw

N = 11
CELL = 18
BOARD = N * CELL
SYMBOLS = ("2", "3", "5", "6")
PALETTES = (
    {"frame": (105, 70, 155), "wall": (31, 48, 78), "pale": (235, 220, 184), "seed": (235, 151, 42)},
    {"frame": (124, 64, 145), "wall": (35, 62, 85), "pale": (225, 219, 178), "seed": (244, 165, 45)},
    {"frame": (88, 76, 164), "wall": (28, 55, 73), "pale": (239, 224, 193), "seed": (224, 142, 38)},
    {"frame": (116, 73, 165), "wall": (39, 49, 88), "pale": (229, 215, 188), "seed": (242, 158, 52)},
)

_ROWS = {
    "2": ("11111", "00001", "00001", "11111", "10000", "10000", "11111"),
    "3": ("11111", "00001", "00001", "11111", "00001", "00001", "11111"),
    "5": ("11111", "10000", "10000", "11111", "00001", "00001", "11111"),
    "6": ("11111", "10000", "10000", "11111", "10001", "10001", "11111"),
}


def _template(symbol: str) -> set[tuple[int, int]]:
    return {(r + 2, c + 3) for r, row in enumerate(_ROWS[symbol]) for c, value in enumerate(row) if value == "1"}


def _flood(walls: set[tuple[int, int]]) -> set[tuple[int, int]]:
    seen = {(0, 0)}
    queue = deque(seen)
    while queue:
        r, c = queue.popleft()
        for p in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
            if 0 <= p[0] < N and 0 <= p[1] < N and p not in walls and p not in seen:
                seen.add(p)
                queue.append(p)
    return seen


def _dist(mask: set[tuple[int, int]], symbol: str) -> int:
    return len(mask ^ _template(symbol))


def _make_scene(seed: int) -> dict:
    rng = random.Random(seed)
    target = SYMBOLS[seed % len(SYMBOLS)]
    output = _template(target)
    noise: list[list[int]] = []
    # Zero, one, and two-cell perturbations expose a visible template margin.
    wanted = (seed // 4) % 3
    # Remove target cells only: this varies the classification margin without
    # creating an answer-specific remote island or enlarging decisive support.
    positions = list(output)
    rng.shuffle(positions)
    if wanted:
        for p in positions:
            trial = output ^ {p}
            distances = sorted((_dist(trial, s), s) for s in SYMBOLS)
            if distances[0][1] == target and distances[1][0] - distances[0][0] >= 1:
                output = trial
                noise.append([p[0], p[1]])
                if len(noise) == wanted:
                    break
    ring = set()
    for r, c in output:
        for p in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
            if 0 <= p[0] < N and 0 <= p[1] < N and p not in output:
                ring.add(p)
    walls = set(ring)
    # Any pale pocket created by the enclosing ring becomes barrier, leaving the
    # intended noisy digit as the complete unreachable passable set.
    reachable = _flood(walls)
    for p in {(r, c) for r in range(N) for c in range(N)} - reachable - output:
        walls.add(p)
    # Hold total barrier count fixed to defeat total-ink and wall-count cues.
    candidates = [(r, c) for r in range(N) for c in range(N) if (r, c) not in output | walls and (r, c) != (0, 0)]
    rng.shuffle(candidates)
    for p in candidates:
        if len(walls) >= 54:
            break
        trial = walls | {p}
        reach = _flood(trial)
        remaining = {(r, c) for r in range(N) for c in range(N)} - trial - output
        if remaining <= reach:
            walls = trial
    if len(walls) != 54 or _flood(walls) & output:
        raise RuntimeError("could not construct certified flood badge")
    return {
        "target": target,
        "noise": noise,
        "walls": [list(p) for p in sorted(walls)],
        "origin_x": 157 + rng.randint(-22, 22),
        "origin_y": 157 + rng.randint(-22, 22),
        "palette_id": rng.randrange(len(PALETTES)),
    }


def sample_scene(seed: int) -> dict:
    return _make_scene(int(seed))


def _output_mask(scene: dict) -> set[tuple[int, int]]:
    mask = _template(scene["target"])
    for p in scene["noise"]:
        mask ^= {(int(p[0]), int(p[1]))}
    return mask


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (512, 512), (250, 249, 246))
    draw = ImageDraw.Draw(image)
    x0, y0 = int(scene["origin_x"]), int(scene["origin_y"])
    palette = PALETTES[int(scene["palette_id"])]
    # Reading target here is intentional: the rendered enclosed cells visibly
    # depend on the sole analytic label input.
    output = _template(scene["target"])
    for p in scene["noise"]:
        output ^= {(int(p[0]), int(p[1]))}
    walls = {tuple(map(int, p)) for p in scene["walls"]}
    draw.rectangle((x0 - 7, y0 - 7, x0 + BOARD + 6, y0 + BOARD + 6), fill=palette["frame"])
    for r in range(N):
        for c in range(N):
            p = (r, c)
            role = "seed" if p == (0, 0) else "wall" if p in walls else "pale"
            xa, ya = x0 + c * CELL + 2, y0 + r * CELL + 2
            draw.rectangle((xa, ya, xa + CELL - 5, ya + CELL - 5), fill=palette[role])
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["target"])


def margin(scene: dict) -> float:
    mask = _output_mask(scene)
    distances = sorted(_dist(mask, symbol) for symbol in SYMBOLS)
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
