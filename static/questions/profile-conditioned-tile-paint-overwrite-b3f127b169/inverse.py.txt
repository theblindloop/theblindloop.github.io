"""Independent inverse arm using only raster pixels and relative geometry."""

from __future__ import annotations

from PIL import Image
import numpy as np

PALETTE = {
    "blue": (49, 111, 221),
    "green": (44, 160, 90),
    "orange": (241, 135, 35),
    "purple": (142, 75, 190),
}
INK = (28, 32, 38)
TARGET = (220, 40, 130)


def _components(image: Image.Image, color: tuple[int, int, int], tolerance: int = 4):
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    target = np.asarray(color, dtype=np.int16)
    mask = np.max(np.abs(pixels - target), axis=2) <= tolerance

    # Label horizontal runs, joining overlapping runs on adjacent rows. This is
    # a small independent connected-component implementation, not a renderer map.
    parents: list[int] = []
    runs: list[tuple[int, int, int, int]] = []

    def find(value: int) -> int:
        while parents[value] != value:
            parents[value] = parents[parents[value]]
            value = parents[value]
        return value

    def union(left: int, right: int) -> None:
        left, right = find(left), find(right)
        if left != right:
            parents[right] = left

    previous: list[tuple[int, int, int]] = []
    for y, row in enumerate(mask):
        padded = np.pad(row.astype(np.int8), (1, 1))
        changes = np.diff(padded)
        starts = np.flatnonzero(changes == 1)
        ends = np.flatnonzero(changes == -1) - 1
        current: list[tuple[int, int, int]] = []
        for x0, x1 in zip(starts.tolist(), ends.tolist()):
            label = len(parents)
            parents.append(label)
            for px0, px1, previous_label in previous:
                if px1 >= x0 and x1 >= px0:
                    union(label, previous_label)
            runs.append((y, x0, x1, label))
            current.append((x0, x1, label))
        previous = current

    aggregates = {}
    for y, x0, x1, label in runs:
        root = find(label)
        count = x1 - x0 + 1
        item = aggregates.setdefault(root, {
            "area": 0, "x0": x0, "x1": x1, "y0": y, "y1": y,
            "sum_x": 0, "sum_y": 0,
        })
        item["area"] += count
        item["x0"] = min(item["x0"], x0)
        item["x1"] = max(item["x1"], x1)
        item["y0"] = min(item["y0"], y)
        item["y1"] = max(item["y1"], y)
        item["sum_x"] += (x0 + x1) * count / 2
        item["sum_y"] += y * count
    found = []
    for item in aggregates.values():
        if item["area"] >= 20:
            item["cx"] = item.pop("sum_x") / item["area"]
            item["cy"] = item.pop("sum_y") / item["area"]
            found.append(item)
    return found


def decision_from_image(image: Image.Image) -> str:
    rings = _components(image, TARGET)
    if len(rings) != 1:
        raise ValueError("target ring is missing or ambiguous")
    ring = rings[0]
    pips = _components(image, INK)
    target_pips = [p for p in pips if ring["x0"] < p["cx"] < ring["x1"]
                   and ring["y0"] < p["cy"] < ring["y1"]]
    target_id = len(target_pips)
    if target_id not in (1, 2, 3, 4):
        raise ValueError("target identity is unreadable")

    swatches = []
    for name, color in PALETTE.items():
        for component in _components(image, color):
            height = component["y1"] - component["y0"] + 1
            width = component["x1"] - component["x0"] + 1
            if width >= 50 and height >= 28:
                swatches.append((component["cx"], component, name))
    swatches.sort(key=lambda item: item[0])
    if len(swatches) != 5:
        raise ValueError("paint-card swatches are missing or ambiguous")

    state = None
    # The initial bar is the only palette component inside the target ring.
    for name, color in PALETTE.items():
        for component in _components(image, color):
            if ring["x0"] < component["cx"] < ring["x1"] and ring["y0"] < component["cy"] < ring["y1"]:
                state = name
    if state is None:
        raise ValueError("target initial color is unreadable")

    for _, swatch, color_name in swatches:
        card_pips = [p for p in pips
                     if swatch["x0"] - 10 < p["cx"] < swatch["x1"] + 10
                     and 35 < swatch["cy"] - p["cy"] < 210]
        identity = len(card_pips)
        if identity not in (1, 2, 3, 4):
            raise ValueError("paint-card identity is unreadable")
        if identity == target_id:
            state = color_name
    return state
