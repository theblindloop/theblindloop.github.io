"""Deterministic Pillow renderer for an identity-addressed overwrite replay."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 1000, 600
COLORS = ("blue", "green", "orange", "purple")
RGB = {
    "blue": (49, 111, 221),
    "green": (44, 160, 90),
    "orange": (241, 135, 35),
    "purple": (142, 75, 190),
}
INK = (28, 32, 38)
FRAME = (116, 124, 136)
TARGET = (220, 40, 130)


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; seed modulo four fixes the final answer class."""
    rng = random.Random(int(seed))
    target_id = 1 + ((int(seed) // 4) % 4)
    desired = COLORS[int(seed) % 4]
    initial_colors = list(COLORS)
    rng.shuffle(initial_colors)
    if initial_colors[target_id - 1] == desired:
        swap = target_id % 4
        initial_colors[target_id - 1], initial_colors[swap] = (
            initial_colors[swap], initial_colors[target_id - 1]
        )

    final_target_index = 2 + (int(seed) % 3)
    operations = []
    earlier_color = COLORS[(COLORS.index(desired) + 1 + (int(seed) % 2)) % 4]
    for index in range(5):
        if index == 0:
            tile = target_id
            color = earlier_color
        elif index == final_target_index:
            tile = target_id
            color = desired
        else:
            choices = [value for value in range(1, 5) if value != target_id]
            tile = rng.choice(choices)
            color = rng.choice(COLORS)
        operations.append({"tile": tile, "color": color})
    return {
        "target_id": target_id,
        "initial_colors": initial_colors,
        "operations": operations,
        "pip_gap": (12, 14, 16, 18)[(int(seed) // 3) % 4],
    }


def _pip_offsets(count: int, gap: int) -> list[tuple[int, int]]:
    if count == 1:
        return [(0, 0)]
    if count == 2:
        return [(-gap, 0), (gap, 0)]
    if count == 3:
        return [(-gap, -gap), (gap, -gap), (0, gap)]
    return [(-gap, -gap), (gap, -gap), (-gap, gap), (gap, gap)]


def _draw_pips(draw: ImageDraw.ImageDraw, cx: int, cy: int, count: int, gap: int) -> None:
    radius = 6
    for dx, dy in _pip_offsets(count, gap):
        draw.ellipse((cx + dx - radius, cy + dy - radius,
                      cx + dx + radius, cy + dy + radius), fill=INK)


def render(scene: dict) -> Image.Image:
    target_id = int(scene["target_id"])
    initial_colors = list(scene["initial_colors"])
    operations = list(scene["operations"])
    pip_gap = int(scene["pip_gap"])
    image = Image.new("RGB", (WIDTH, HEIGHT), (250, 249, 246))
    draw = ImageDraw.Draw(image)

    # Four initial-state tiles. Identity is encoded only by visible pip count.
    tile_boxes = []
    for tile_id in range(1, 5):
        col, row = (tile_id - 1) % 2, (tile_id - 1) // 2
        x0, y0 = 55 + col * 145, 145 + row * 155
        box = (x0, y0, x0 + 125, y0 + 125)
        tile_boxes.append(box)
        draw.rounded_rectangle(box, radius=12, fill=(239, 241, 244), outline=FRAME, width=4)
        _draw_pips(draw, x0 + 62, y0 + 43, tile_id, pip_gap)
        draw.rounded_rectangle((x0 + 20, y0 + 88, x0 + 105, y0 + 110),
                               radius=5, fill=RGB[initial_colors[tile_id - 1]])
    x0, y0, x1, y1 = tile_boxes[target_id - 1]
    draw.rounded_rectangle((x0 - 9, y0 - 9, x1 + 9, y1 + 9),
                           radius=17, outline=TARGET, width=7)

    # Ordered paint cards. Their taller swatches distinguish them from initial bars.
    for index, operation in enumerate(operations):
        x0 = 375 + index * 120
        y0 = 145
        draw.rounded_rectangle((x0, y0, x0 + 100, y0 + 280), radius=14,
                               fill=(244, 246, 248), outline=FRAME, width=4)
        _draw_pips(draw, x0 + 50, y0 + 76, int(operation["tile"]), pip_gap)
        draw.polygon([(x0 + 43, y0 + 128), (x0 + 57, y0 + 128),
                      (x0 + 57, y0 + 168), (x0 + 70, y0 + 168),
                      (x0 + 50, y0 + 193), (x0 + 30, y0 + 168),
                      (x0 + 43, y0 + 168)], fill=(176, 181, 189))
        draw.rounded_rectangle((x0 + 12, y0 + 222, x0 + 88, y0 + 262),
                               radius=7, fill=RGB[str(operation["color"])])

    # A visible left-to-right rail reinforces order without text or OCR.
    draw.line((390, 475, 936, 475), fill=(116, 124, 136), width=5)
    draw.polygon([(936, 463), (960, 475), (936, 487)], fill=(116, 124, 136))
    return image


def analytic_gold(scene: dict) -> str:
    target_id = int(scene["target_id"])
    current = list(scene["initial_colors"])[target_id - 1]
    for operation in list(scene["operations"]):
        if int(operation["tile"]) == target_id:
            current = str(operation["color"])
    return current


def margin(scene: dict) -> float:
    """Visible minimum edge-to-edge pip clearance, in pixels."""
    return float(2 * int(scene["pip_gap"]) - 13)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 3.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
