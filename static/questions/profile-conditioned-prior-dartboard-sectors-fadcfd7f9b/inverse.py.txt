"""Independent pixel-only inverse oracle for the dartboard sector-count world.

Receives only the final PNG and re-counts the radial sectors directly from the
raster. It imports nothing from the renderer, prompts, generator, verifier, or
analytic-gold code, and accepts no latent scene data.

Method: the one object on the field is a solid circular disc split into radial
wedges that strictly alternate between a light and a dark sector color around a
small central bull. Step 1 locates the disc as the set of light-or-dark sector
pixels and takes its bounding-box centre and radius. Step 2 samples a ring well
inside the disc (outside the bull) and classifies each sample as light or dark.
Because adjacent wedges strictly alternate, the number of colour transitions
read once around that closed ring equals the number of radial sectors.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

_LIGHT_THRESH = 230.0  # luminance at/above this is a light wedge
_DARK_THRESH = 130.0   # luminance at/below this is a dark wedge
_SAMPLE_MID = 160.0    # classifier cut between light and dark at the sample ring
_SAMPLE_FRACTION = 0.66
_N_SAMPLES = 2048


def _luminance(arr: np.ndarray) -> np.ndarray:
    return (
        0.299 * arr[..., 0].astype(np.float64)
        + 0.587 * arr[..., 1].astype(np.float64)
        + 0.114 * arr[..., 2].astype(np.float64)
    )


def _disc_geometry(arr: np.ndarray) -> tuple[float, float, float]:
    """Return (cx, cy, radius) of the dartboard disc from sector pixels only."""
    lum = _luminance(arr)
    mask = (lum <= _DARK_THRESH) | (lum >= _LIGHT_THRESH)
    if mask.sum() == 0:
        return 0.0, 0.0, 0.0
    ys, xs = np.nonzero(mask)
    cx = float(xs.mean())
    cy = float(ys.mean())
    x_min, x_max = float(xs.min()), float(xs.max())
    y_min, y_max = float(ys.min()), float(ys.max())
    radius = 0.5 * max(x_max - x_min, y_max - y_min) + 1.0
    return cx, cy, radius


def _clean_isolated(seq: list[int]) -> list[int]:
    """Round isolated misclassified samples into their neighbours."""
    out = list(seq)
    n = len(out)
    if n < 3:
        return out
    for i in range(n):
        prev = out[(i - 1) % n]
        nxt = out[(i + 1) % n]
        if out[i] != prev and out[i] != nxt and prev == nxt:
            out[i] = prev
    return out


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.float64)
    lum = _luminance(arr)
    h, w = lum.shape
    if (lum <= _DARK_THRESH).sum() == 0 and (lum >= _LIGHT_THRESH).sum() == 0:
        return "0"
    cx, cy, radius = _disc_geometry(arr)
    if radius <= 0.0:
        return "0"
    ring = _SAMPLE_FRACTION * radius
    angles = np.linspace(0.0, 2.0 * math.pi, _N_SAMPLES, endpoint=False)
    xs = np.round(cx + ring * np.cos(angles)).astype(np.int64)
    ys = np.round(cy + ring * np.sin(angles)).astype(np.int64)
    xs = np.clip(xs, 0, w - 1)
    ys = np.clip(ys, 0, h - 1)
    sample_lum = lum[ys, xs]
    seq = [1 if v >= _SAMPLE_MID else 0 for v in sample_lum]
    seq = _clean_isolated(seq)
    n = len(seq)
    transitions = sum(1 for i in range(n) if seq[i] != seq[(i + 1) % n])
    return str(transitions)
