"""Prior-conflict dartboard sector-count renderer.

Latent state z = {n_sectors, center_x, center_y, radius, rot, bull_r}.

A dartboard is drawn as a solid circular disc on a flat field, split into
`n_sectors` equal radial wedges that alternate between a light and a dark
sector color around a small central bull. Every sector is an exact pie wedge,
so the whole disc is exactly tiled by alternating wedges.

The closed-form gold is the number of radial sectors drawn:

    n_sectors (a scene parameter)

The canonical prior -- that a regulation dartboard is divided into 20 numbered
sectors -- is recorded only in the question narrative and is never used as
ground truth. The visible truth is however many sectors are actually drawn.

Continuous verifiability margin (reported in px): the minimum angular arc width
of a sector measured along the oracle's sampled ring. As long as that arc stays
above a finite-resolution floor, each wedge remains a resolvable, countable
band around the ring.

Latent-alias mode: not_applicable. The sector count is pinned by n_sectors, which
renders as exactly `n_sectors` alternating radial wedges; different n_sectors
always produce a differently-counted raster, there is no render-invariant latent
alias, and sample_scene returns no pixel-identical twins.
"""

from __future__ import annotations

import math
import random
from typing import Any

import numpy as np
from PIL import Image

CANVAS_W = 384
CANVAS_H = 384
CANVAS_BG = (208, 213, 220)     # flat gray field
LIGHT_SECTOR = (248, 248, 248)  # light wedge
DARK_SECTOR = (36, 36, 36)      # dark wedge
BULL_COLOR = (216, 66, 44)      # central bull (not sampled by the oracle)

S_RANGE = (6, 8, 10, 12, 14, 16, 18, 20)   # even radial sectors; canonical prior is 20
RADIUS_LO = 92.0
RADIUS_HI = 148.0
CENTER_MARGIN = 8.0             # px clearance from canvas edge for the disc
BULL_FRACTION = 0.17            # bull radius as a fraction of the disc radius
SAMPLE_FRACTION = 0.66          # ring radius as a fraction of the disc radius
MIN_SECTOR_ARC = 6.0            # resolution floor below which wedge count is ambiguous

RENDERER_VERSION = "prior_dartboard_sectors-0.1.0"


def _arc_at_sample_ring(scene: dict[str, Any]) -> float:
    """Minimum sector arc width (px) along the oracle sample ring."""
    n = int(scene["n_sectors"])
    radius = float(scene["radius"])
    ring = SAMPLE_FRACTION * radius
    return 2.0 * math.pi * ring / float(n)


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(seed)
    n_sectors = rng.choice(S_RANGE)
    # Center with enough room that the largest radius still fits on the canvas.
    center_x = rng.uniform(CENTER_MARGIN + RADIUS_HI, CANVAS_W - CENTER_MARGIN - RADIUS_HI)
    center_y = rng.uniform(CENTER_MARGIN + RADIUS_HI, CANVAS_H - CENTER_MARGIN - RADIUS_HI)
    radius = rng.uniform(RADIUS_LO, RADIUS_HI)
    rot = rng.uniform(0.0, 2.0 * math.pi)
    bull_r = round(radius * BULL_FRACTION, 3)
    return {
        "n_sectors": int(n_sectors),
        "center_x": round(center_x, 3),
        "center_y": round(center_y, 3),
        "radius": round(radius, 3),
        "rot": round(rot, 6),
        "bull_r": bull_r,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    n = int(scene["n_sectors"])
    if n < 2:
        raise ValueError("a dartboard needs at least two sectors")
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    radius = float(scene["radius"])
    rot = float(scene["rot"])
    bull = float(scene["bull_r"])

    scale = 2
    big_w = CANVAS_W * scale
    big_h = CANVAS_H * scale
    xs, ys = np.meshgrid(np.arange(big_w), np.arange(big_h))
    x = (xs - cx * scale).astype(np.float64)
    y = (ys - cy * scale).astype(np.float64)
    r = np.sqrt(x * x + y * y)
    ang = np.arctan2(y, x)
    step = 2.0 * math.pi / float(n)
    idx = np.floor(((ang - rot) % (2.0 * math.pi)) / step).astype(np.int64) % n

    img = np.zeros((big_h, big_w, 3), dtype=np.uint8)
    img[..., 0] = CANVAS_BG[0]
    img[..., 1] = CANVAS_BG[1]
    img[..., 2] = CANVAS_BG[2]

    disc = r <= radius * scale
    outer = disc & (r > bull * scale)
    light = outer & (idx % 2 == 0)
    dark = outer & (idx % 2 == 1)
    inner = disc & (r <= bull * scale)

    img[light] = LIGHT_SECTOR
    img[dark] = DARK_SECTOR
    img[inner] = BULL_COLOR

    pil = Image.fromarray(img)
    return pil.resize((CANVAS_W, CANVAS_H), Image.LANCZOS)


def _margin(scene: dict[str, Any]) -> float:
    return _arc_at_sample_ring(scene)


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(int(scene["n_sectors"]))


def margin(scene: dict[str, Any]) -> float:
    return _margin(scene)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return _margin(scene) < MIN_SECTOR_ARC


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # not_applicable: the gold is pinned by the rendered n_sectors; there is no
    # declared pixel-identical alias to exercise.
    return []
