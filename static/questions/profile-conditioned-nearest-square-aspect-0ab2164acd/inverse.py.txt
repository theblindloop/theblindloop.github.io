"""Independent pixel-only inverse arm for the nearest-to-square aspect world.

Receives ONLY the rasterized PNG. It segments the red, blue and green
rectangles by nearest colour, takes each colour's bounding box to recover its
width and height, forms the side-length imbalance |w-h|/max(w,h), and decides
which colour is closest to a perfect square. Imports only numpy and PIL; shares
no geometry, coordinates, or code with the renderer and never reads latent scene
data or analytic gold.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "nearest_square_aspect-oracle-0.1.0"

COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
COLOR_ORDER = ("red", "blue", "green")
BACKGROUND = (250, 250, 250)


def _labels(arr: np.ndarray) -> np.ndarray:
    refs = np.stack(
        [
            np.array(COLOR_RGB["red"], dtype=np.float32),
            np.array(COLOR_RGB["blue"], dtype=np.float32),
            np.array(COLOR_RGB["green"], dtype=np.float32),
            np.array(BACKGROUND, dtype=np.float32),
        ],
        axis=0,
    )
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _labels(arr)
    best = None
    best_im = None
    for color_index, name in enumerate(COLOR_ORDER):
        ys, xs = np.nonzero(labels == color_index)
        if len(ys) == 0:
            continue
        w = int(xs.max() - xs.min() + 1)
        h = int(ys.max() - ys.min() + 1)
        m = max(w, h)
        im = abs(w - h) / m
        if best is None or im < best_im or (im == best_im and name < best):
            best = name
            best_im = im
    return best if best is not None else "red"
