"""Nearest-to-square aspect measurement world.

A light panel carries exactly three coloured filled rectangles (red, blue,
green), each placed in its own non-overlapping horizontal band. The decision is
a recovered measurement: for every rectangle recover its width and height from
the raster, form its side-length imbalance |width - height| / max(width,
height), and select the colour whose measured imbalance is smallest, i.e. whose
aspect ratio is closest to a perfect square.

The analysed quantity is a per-mark ratio of its two own extents. Neither the
total coloured ink, the largest rectangle, the absolute width or height of any
single mark, its position, nor the colour assignment discloses the winner: the
three colours are shuffled across bands and the widths and heights are sampled
independently, so only the recovered aspect-ratio proximity to a square
separates them. Ties between the two smallest imbalances are resolved by a clear
margin and any near-tie scene is quarantined. The oracle and analytic gold share
no geometry and never read latent scene data.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "nearest_square_aspect-0.1.0"

COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
BACKGROUND = (250, 250, 250)

WIDTH = 480
HEIGHT = 340
BANDS = 3
EDGE = 40
PAD = 8
MIN_SIDE = 30
MAX_W = 150
MAX_H = 98
QUARANTINE = 0.05
MIN_MARGIN = 0.09


def _imbalance(w: int, h: int) -> float:
    m = max(w, h)
    return abs(w - h) / m


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 1117 + 29)
    band_h = HEIGHT // BANDS
    while True:
        colors = ["red", "blue", "green"]
        rng.shuffle(colors)
        rects = []
        for b in range(BANDS):
            w = rng.randint(MIN_SIDE, MAX_W)
            h = rng.randint(MIN_SIDE, MAX_H)
            x = rng.randint(EDGE, WIDTH - EDGE - w)
            top = b * band_h + PAD
            bottom = (b + 1) * band_h - PAD - h
            if bottom < top:
                bottom = top
            y = rng.randint(top, bottom)
            rects.append(
                {
                    "color": colors[b],
                    "x": x,
                    "y": y,
                    "w": w,
                    "h": h,
                }
            )
        vals = sorted(
            (_imbalance(r["w"], r["h"]), r["color"]) for r in rects
        )
        margin = vals[1][0] - vals[0][0]
        if margin < MIN_MARGIN:
            continue
        scene = {
            "rects": rects,
            "width": WIDTH,
            "height": HEIGHT,
            "background": "light",
        }
        return scene


def _imbalances(scene: dict) -> list:
    return sorted((_imbalance(r["w"], r["h"]), r["color"]) for r in scene["rects"])


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    for r in scene["rects"]:
        draw.rectangle(
            [r["x"], r["y"], r["x"] + r["w"], r["y"] + r["h"]],
            fill=COLOR_RGB[r["color"]],
        )
    return img


def analytic_gold(scene: dict) -> str:
    best = None
    best_im = None
    for r in scene["rects"]:
        im = _imbalance(r["w"], r["h"])
        if best is None or im < best_im or (im == best_im and r["color"] < best):
            best = r["color"]
            best_im = im
    return best


def margin(scene: dict) -> float:
    vals = _imbalances(scene)
    return float(vals[1][0] - vals[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE


def latent_symmetries(scene: dict) -> list:
    return []
