"""Independent final-PNG inverse for Endpoint Weave Reveal.

This module intentionally duplicates the public palette and three answer
templates. It imports no renderer or latent-scene code. Card positions, scale,
panel order, and endpoints are all estimated from pixels.
"""

from __future__ import annotations

from collections import deque
from typing import Iterable

import numpy as np
from PIL import Image

CYAN = np.array((20, 170, 190), dtype=np.int16)
GOLD = np.array((230, 165, 30), dtype=np.int16)
DARK = np.array((31, 38, 51), dtype=np.int16)

REFERENCE_RUNS: dict[str, tuple[tuple[int, int], ...]] = {
    "hourglass": ((0, 6), (1, 5), (2, 4), (2, 3), (2, 3), (2, 4), (1, 5), (0, 6)),
    "diamond": ((2, 3), (2, 4), (1, 5), (0, 6), (0, 6), (1, 5), (2, 4), (2, 3)),
    "stair": ((0, 3), (0, 3), (1, 4), (1, 4), (2, 5), (2, 5), (2, 6), (2, 6)),
}


class PixelEvidenceError(ValueError):
    """The declared transform cannot be recovered from the visible evidence."""


def _near(array: np.ndarray, color: np.ndarray, tolerance: float = 34.0) -> np.ndarray:
    delta = array.astype(np.int32) - color.astype(np.int32)
    return np.sqrt(np.sum(delta * delta, axis=2, dtype=np.int64)) <= tolerance


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        output.append(np.asarray(points, dtype=np.int32))
    return output


def _locate_card(color_mask: np.ndarray, name: str) -> dict[str, int]:
    components = [component for component in _components(color_mask) if len(component) >= 80]
    if len(components) != 1:
        raise PixelEvidenceError(f"{name} card fiducial is missing or fragmented")
    points = components[0]
    ymin, xmin = points.min(axis=0)
    ymax, xmax = points.max(axis=0)
    width = int(xmax - xmin)
    scale = int(round(width / 10.0))
    if scale not in (10, 11) or abs(width - 10 * scale) > 1:
        raise PixelEvidenceError(f"{name} card has an invalid visible scale")
    y0 = int(ymax - 6 * scale)
    if not (ymin <= y0 - 7 and 67 <= ymax - ymin <= 75):
        raise PixelEvidenceError(f"{name} triangular top notch is incomplete")

    # Every visible frame side must survive. This rejects regional evidence
    # erasure instead of hallucinating endpoints from a damaged card.
    top = color_mask[y0 : y0 + 3, xmin : xmax + 1]
    bottom = color_mask[ymax - 2 : ymax + 1, xmin : xmax + 1]
    left = color_mask[y0 : ymax + 1, xmin : xmin + 3]
    right = color_mask[y0 : ymax + 1, xmax - 2 : xmax + 1]
    if min(float(side.mean()) for side in (top, bottom, left, right)) < 0.84:
        raise PixelEvidenceError(f"{name} card frame is incomplete")
    notch = color_mask[max(0, y0 - 8) : y0 + 1, xmin + width // 2 - 7 : xmin + width // 2 + 8]
    if notch.size == 0 or float(notch.mean()) < 0.40:
        raise PixelEvidenceError(f"{name} top-notch orientation is unreadable")
    return {"x0": int(xmin), "x1": int(xmax), "y0": y0, "y1": int(ymax), "scale": scale}


def _card_runs(dark_mask: np.ndarray, card: dict[str, int], name: str) -> list[list[int]]:
    x0, x1, y0, y1, scale = (card[key] for key in ("x0", "x1", "y0", "y1", "scale"))
    interior = np.zeros_like(dark_mask)
    interior[y0 + 2 : y1 - 1, x0 + 2 : x1 - 1] = dark_mask[y0 + 2 : y1 - 1, x0 + 2 : x1 - 1]
    points = []
    for component in _components(interior):
        if 20 <= len(component) <= 100:
            cy, cx = component.mean(axis=0)
            points.append((float(cx), float(cy)))
    if len(points) != 8:
        raise PixelEvidenceError(f"{name} card must expose exactly eight endpoint pegs")

    rows: list[list[int]] = [[] for _ in range(4)]
    for cx, cy in points:
        local_row = int(round((cy - (y0 + scale)) / scale))
        column = int(round((cx - (x0 + 2 * scale)) / scale))
        if not 0 <= local_row < 4 or not 0 <= column < 7:
            raise PixelEvidenceError(f"{name} endpoint falls outside the visible rack")
        expected_x = x0 + (2 + column) * scale
        expected_y = y0 + (1 + local_row) * scale
        if abs(cx - expected_x) > 1.25 or abs(cy - expected_y) > 1.25:
            raise PixelEvidenceError(f"{name} endpoint is not centered on a visible cell")
        rows[local_row].append(column)
    output = []
    for row in rows:
        if len(row) != 2 or row[0] == row[1]:
            raise PixelEvidenceError(f"{name} row does not contain two distinct endpoints")
        output.append(sorted(row))
    return output


def _filled(runs: Iterable[Iterable[int]]) -> tuple[tuple[int, ...], ...]:
    output = []
    for pair in runs:
        left, right = (int(value) for value in pair)
        output.append(tuple(int(left <= column <= right) for column in range(7)))
    return tuple(output)


def classify_from_image(image: Image.Image) -> dict:
    """Recover the declared transform and report the runner-up margin."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    cyan_card = _locate_card(_near(array, CYAN), "cyan")
    gold_card = _locate_card(_near(array, GOLD), "gold")
    if cyan_card["scale"] != gold_card["scale"]:
        raise PixelEvidenceError("the two visible card scales disagree")
    dark_mask = _near(array, DARK, 30.0)
    cyan_rows = _card_runs(dark_mask, cyan_card, "cyan")
    gold_rows = _card_runs(dark_mask, gold_card, "gold")
    runs: list[list[int]] = []
    for cyan_row, gold_row in zip(cyan_rows, gold_rows, strict=True):
        runs.extend((cyan_row, gold_row))
    observed = _filled(runs)
    distances: dict[str, int] = {}
    for name, reference_runs in REFERENCE_RUNS.items():
        reference = _filled(reference_runs)
        distances[name] = sum(
            int(a != b)
            for observed_row, reference_row in zip(observed, reference, strict=True)
            for a, b in zip(observed_row, reference_row, strict=True)
        )
    ranked = sorted((distance, name) for name, distance in distances.items())
    if ranked[0][0] == ranked[1][0]:
        decision = "ambiguous"
        margin = 0.0
    else:
        decision = ranked[0][1]
        margin = float(ranked[1][0] - ranked[0][0])
    return {"decision": decision, "margin": margin, "distances": distances, "runs": runs}


def decision_from_image(image: Image.Image) -> str:
    return str(classify_from_image(image)["decision"])
