"""Deterministic latent sampler and raster renderer for Endpoint Weave Reveal."""

from __future__ import annotations

import copy
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 768
HEIGHT = 512
BACKGROUND = (250, 248, 242)
CYAN = (20, 170, 190)
GOLD = (230, 165, 30)
DARK = (31, 38, 51)
GRID = (205, 209, 207)
DECOR = (171, 174, 170)

SYMBOLS = ("hourglass", "diamond", "stair")

# Each pair is an inclusive [left, right] span on one seven-cell output row.
TEMPLATES: dict[str, tuple[tuple[int, int], ...]] = {
    "hourglass": ((0, 6), (1, 5), (2, 4), (2, 3), (2, 3), (2, 4), (1, 5), (0, 6)),
    "diamond": ((2, 3), (2, 4), (1, 5), (0, 6), (0, 6), (1, 5), (2, 4), (2, 3)),
    "stair": ((0, 3), (0, 3), (1, 4), (1, 4), (2, 5), (2, 5), (2, 6), (2, 6)),
}

# Unique, low-margin examples. Each remains two cell changes closer to its
# named template than to its runner-up and makes the visible boundary concrete.
NEAR_VARIANTS: dict[str, tuple[tuple[int, int], ...]] = {
    "hourglass": ((0, 6), (1, 5), (2, 4), (2, 3), (2, 5), (2, 5), (2, 6), (2, 6)),
    "diamond": ((2, 3), (2, 4), (1, 5), (0, 6), (2, 5), (1, 5), (2, 6), (2, 6)),
    "stair": ((0, 3), (0, 3), (1, 4), (1, 4), (2, 3), (2, 4), (1, 5), (0, 6)),
}


def _grid_from_runs(runs: Iterable[Iterable[int]]) -> tuple[tuple[int, ...], ...]:
    rows = []
    for pair in runs:
        left, right = (int(value) for value in pair)
        rows.append(tuple(int(left <= column <= right) for column in range(7)))
    return tuple(rows)


def _template_distances(runs: Iterable[Iterable[int]]) -> dict[str, int]:
    observed = _grid_from_runs(runs)
    if len(observed) != 8:
        raise ValueError("exactly eight reconstructed rows are required")
    distances: dict[str, int] = {}
    for name, template_runs in TEMPLATES.items():
        template = _grid_from_runs(template_runs)
        distances[name] = sum(
            int(observed_cell != template_cell)
            for observed_row, template_row in zip(observed, template, strict=True)
            for observed_cell, template_cell in zip(observed_row, template_row, strict=True)
        )
    return distances


def _classification(runs: Iterable[Iterable[int]]) -> tuple[str, float]:
    ranked = sorted((distance, name) for name, distance in _template_distances(runs).items())
    if ranked[0][0] == ranked[1][0]:
        return "ambiguous", 0.0
    return ranked[0][1], float(ranked[1][0] - ranked[0][0])


def sample_scene(seed: int) -> dict:
    """Sample balanced labels plus position, scale, order, and decoration nuisances."""

    seed = int(seed)
    rng = random.Random(seed)
    target = SYMBOLS[seed % len(SYMBOLS)]
    source = NEAR_VARIANTS[target] if seed % 5 == 0 else TEMPLATES[target]
    panel_order = ["cyan", "gold"]
    rng.shuffle(panel_order)
    anchors = ((104, 116), (303, 106), (455, 111), (666, 119), (304, 402), (465, 410))
    decorations = [
        [x + rng.randint(-7, 7), y + rng.randint(-7, 7), rng.choice((3, 4, 5))]
        for x, y in anchors
    ]
    return {
        "runs": [list(pair) for pair in source],
        "panel_order": panel_order,
        "scale": rng.choice((10, 11)),
        "x_shift": rng.randint(-2, 2),
        "y_shift": rng.randint(-4, 4),
        "decorations": decorations,
    }


def _draw_card(
    draw: ImageDraw.ImageDraw,
    *,
    x0: int,
    y0: int,
    scale: int,
    color: tuple[int, int, int],
    four_runs: list[list[int]],
) -> None:
    width = 10 * scale
    height = 6 * scale
    stroke = 3
    draw.rectangle((x0, y0, x0 + width, y0 + height), outline=color, width=stroke)
    center = x0 + width // 2
    draw.polygon(
        ((center, y0 - 8), (center - 7, y0 + 2), (center + 7, y0 + 2)),
        fill=color,
    )

    # Pale seven-column row racks provide the visible discrete coordinate system.
    for column in range(7):
        x = x0 + (2 + column) * scale
        draw.line((x, y0 + scale // 2, x, y0 + 9 * scale // 2), fill=GRID, width=1)
    for local_row in range(4):
        y = y0 + (1 + local_row) * scale
        draw.line((x0 + 3 * scale // 2, y, x0 + 17 * scale // 2, y), fill=GRID, width=1)

    radius = 3 if scale == 10 else 4
    for local_row, pair in enumerate(four_runs):
        left, right = (int(value) for value in pair)
        y = y0 + (1 + local_row) * scale
        for column in (left, right):
            x = x0 + (2 + column) * scale
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=DARK)


def render(scene: dict) -> Image.Image:
    """Render endpoint evidence only; no filled output silhouette is drawn."""

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x, y, radius in scene["decorations"]:
        radius = int(radius)
        draw.ellipse((int(x) - radius, int(y) - radius, int(x) + radius, int(y) + radius), fill=DECOR)

    scale = int(scene["scale"])
    card_width = 10 * scale
    centers = (205 + int(scene["x_shift"]), 560 + int(scene["x_shift"]))
    y0 = 226 + int(scene["y_shift"])
    runs = scene["runs"]
    by_color = {
        "cyan": [runs[index] for index in (0, 2, 4, 6)],
        "gold": [runs[index] for index in (1, 3, 5, 7)],
    }
    colors = {"cyan": CYAN, "gold": GOLD}
    for center, name in zip(centers, scene["panel_order"], strict=True):
        _draw_card(
            draw,
            x0=int(center - card_width // 2),
            y0=y0,
            scale=scale,
            color=colors[name],
            four_runs=by_color[name],
        )
    return image


def analytic_gold(scene: dict) -> str:
    return _classification(scene["runs"])[0]


def margin(scene: dict) -> float:
    return _classification(scene["runs"])[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["decorations"] = list(reversed(twin["decorations"]))
    return [("reverse_nonoverlapping_decoration_draw_order", twin)]
