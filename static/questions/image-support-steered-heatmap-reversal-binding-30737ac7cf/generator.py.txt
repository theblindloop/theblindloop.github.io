"""Deterministic compact paired-heatmap renderer and analytic reference."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 512
HEIGHT = 512
CARD_W = 220
CARD_H = 180
CELL_W = 20
CELL_H = 25
ROW_STEP = 31
PANEL_LEFT = 16
PANEL_RIGHT = 144
ROW_TOP = 56
ROW_KEY_X = 2

LABELS = ("amber", "teal", "violet")
LABEL_COLORS = {
    "amber": (218, 145, 44),
    "teal": (32, 157, 148),
    "violet": (136, 78, 188),
}
HEAT_LIGHT = (245, 205, 185)
HEAT_DARK = (130, 30, 22)
CARD_BORDER = (104, 124, 139)
PANEL_BORDER = (195, 204, 211)
TEXT = (70, 84, 96)


def _heat_color(value: int) -> tuple[int, int, int]:
    """Map a temperature measurement to a stable dark-is-hot RGB fill."""

    t = max(0.0, min(1.0, (float(value) - 35.0) / 200.0))
    return tuple(
        int(round(light + (dark - light) * t))
        for light, dark in zip(HEAT_LIGHT, HEAT_DARK, strict=True)
    )


def _extreme_indices(values: list[list[int]]) -> tuple[int, int]:
    p = [int(pair[0]) for pair in values]
    q = [int(pair[1]) for pair in values]
    hottest = max(range(3), key=lambda index: p[index])
    coldest = min(range(3), key=lambda index: q[index])
    return hottest, coldest


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample balanced, locally supported heatmap measurements."""

    rng = random.Random(int(seed))
    winner = int(seed) % 3
    row_labels = list(LABELS)
    rng.shuffle(row_labels)

    near_boundary = int(seed) % 11 == 0
    if near_boundary:
        p_winner = rng.choice((166, 171, 176))
        p_other = [p_winner - 22, p_winner - 27]
        q_winner = rng.choice((78, 83, 88))
        q_other = [q_winner + 22, q_winner + 27]
    else:
        p_winner = rng.choice((205, 215, 225))
        p_other = [rng.choice((92, 102, 112)), rng.choice((142, 152, 162))]
        q_winner = rng.choice((42, 52, 62))
        q_other = [rng.choice((132, 142, 152)), rng.choice((177, 187, 197))]

    values = [[0, 0] for _ in range(3)]
    values[winner] = [int(p_winner), int(q_winner)]
    other_indices = [index for index in range(3) if index != winner]
    rng.shuffle(other_indices)
    for index, p_value, q_value in zip(other_indices, p_other, q_other, strict=True):
        values[index] = [int(p_value), int(q_value)]

    backgrounds = ((235, 241, 245), (239, 242, 244), (232, 239, 244))
    cards = ((249, 247, 241), (247, 249, 246), (248, 247, 244))
    return {
        "values": values,
        "row_labels": row_labels,
        # These translations keep each filled component inside one protected
        # ablation tile while still placing P and Q in separated local tiles.
        "card_x": int(rng.choice((78, 142, 206, 270))),
        "card_y": int(rng.choice((64, 128, 192, 256))),
        "background": list(backgrounds[rng.randrange(len(backgrounds))]),
        "card_background": list(cards[rng.randrange(len(cards))]),
        "remote_variant": int(rng.randrange(24)),
        "style": int(rng.randrange(3)),
    }


def _draw_remote_context(draw: ImageDraw.ImageDraw, variant: int, style: int) -> None:
    """Add pale, answer-independent scientific-chart texture away from the card."""

    pale = ((213, 222, 230), (220, 227, 233), (207, 219, 229))
    accent = pale[(variant + style) % len(pale)]
    for index in range(5):
        x = 14 + ((variant * 29 + index * 67) % 120)
        y = 18 + ((variant * 17 + index * 31) % 86)
        draw.line((x, y, min(WIDTH - 14, x + 22 + index * 4), y), fill=accent, width=2)
        draw.line((x, y + 8, min(WIDTH - 14, x + 13 + index * 3), y + 8), fill=pale[1], width=2)
    for index in range(5):
        x = 360 + ((variant * 23 + index * 31) % 130)
        y = 372 + ((variant * 13 + index * 23) % 116)
        draw.line((x, y, min(WIDTH - 12, x + 28 + index * 3), y), fill=pale[1], width=2)
        draw.line((x, y + 8, min(WIDTH - 12, x + 16 + index * 2), y + 8), fill=accent, width=2)


def _font() -> ImageFont.ImageFont:
    return ImageFont.load_default()


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the final RGB image from the public scene dictionary."""

    image = Image.new("RGB", (WIDTH, HEIGHT), tuple(int(value) for value in scene["background"]))
    draw = ImageDraw.Draw(image)
    _draw_remote_context(draw, int(scene["remote_variant"]), int(scene["style"]))

    left = int(scene["card_x"])
    top = int(scene["card_y"])
    card_background = tuple(int(value) for value in scene["card_background"])
    draw.rounded_rectangle(
        (left, top, left + CARD_W, top + CARD_H),
        radius=12,
        fill=card_background,
        outline=CARD_BORDER,
        width=3,
    )
    draw.text((left + 3, top + 12), "KEY", fill=TEXT, font=_font())

    for panel_name, panel_offset in (("P", PANEL_LEFT), ("Q", PANEL_RIGHT)):
        panel_x = left + panel_offset
        draw.text((panel_x + 8, top + 24), panel_name, fill=TEXT, font=_font())
        draw.rounded_rectangle(
            (panel_x - 7, top + ROW_TOP - 8, panel_x + CELL_W + 7, top + ROW_TOP + 2 * ROW_STEP + CELL_H + 8),
            radius=5,
            outline=PANEL_BORDER,
            width=2,
        )

    labels = list(scene["row_labels"])
    values = scene["values"]
    for row_index, label in enumerate(labels):
        row_top = top + ROW_TOP + row_index * ROW_STEP
        row_center = row_top + CELL_H // 2
        draw.rounded_rectangle(
            (left + ROW_KEY_X, row_top + 4, left + ROW_KEY_X + 12, row_top + 19),
            radius=3,
            fill=LABEL_COLORS[label],
        )
        # The one-letter key is deliberately inside the swatch: the card stays
        # compact while the visible color, not text rasterization, carries the
        # row identity used by the inverse arm.
        letter_color = (255, 255, 255) if label != "amber" else (88, 68, 28)
        draw.text((left + ROW_KEY_X + 4, row_top + 3), label[0].upper(), fill=letter_color, font=_font())
        for panel_index, panel_offset in enumerate((PANEL_LEFT, PANEL_RIGHT)):
            cell_x = left + panel_offset
            fill = _heat_color(int(values[row_index][panel_index]))
            draw.rectangle(
                (cell_x, row_top, cell_x + CELL_W - 1, row_top + CELL_H - 1),
                fill=fill,
                outline=PANEL_BORDER,
                width=2,
            )
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Select the row that is hottest in P and coldest in Q."""

    values = scene["values"]
    row_labels = scene["row_labels"]
    hottest, coldest = _extreme_indices(values)
    if hottest != coldest:
        raise ValueError("scene does not contain a unique heatmap reversal")
    return str(row_labels[hottest])


def margin(scene: dict[str, Any]) -> float:
    """Return the smaller separation of the two opposing extrema."""

    values = scene["values"]
    p = [int(pair[0]) for pair in values]
    q = [int(pair[1]) for pair in values]
    hottest = max(p)
    coldest = min(q)
    p_gap = hottest - sorted(p)[-2]
    q_gap = sorted(q)[1] - coldest
    return float(min(p_gap, q_gap))


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Quarantine finite-resolution near ties and malformed reversals."""

    try:
        analytic_gold(scene)
    except (KeyError, TypeError, ValueError, IndexError):
        return True
    return margin(scene) < 24.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No nontrivial render-invariant latent aliases are declared."""

    return []
