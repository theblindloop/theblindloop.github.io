"""Deterministic spiral-notebook scene sampler and raster renderer."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

CANVAS = 512
SS = 3

BACKGROUNDS = ["#f3eee5", "#e9f0f2", "#f2edf3", "#edf2e9"]
PAPERS = ["#fffdf5", "#f8fbff", "#fff8ee", "#f7fff9"]
COVERS = ["#c96b67", "#5d91b8", "#7aa36f", "#b487bd", "#d49a54"]
RULES = ["#86a9c4", "#b18aa2", "#83a99b", "#9c9ab8"]
BINDINGS = ["#242b34", "#382c25", "#233934", "#34283b"]


def sample_scene(seed: int) -> dict:
    """Sample a balanced label and independently varied nuisance styling."""
    rng = random.Random((int(seed) ^ 0x5A17C9E3) & 0xFFFFFFFFFFFFFFFF)
    # The odd stride used by generate.py makes adjacent scenes alternate labels.
    binding_side = "left" if int(seed) % 2 == 0 else "right"
    return {
        "binding_side": binding_side,
        "center_x": rng.randint(244, 268),
        "center_y": rng.randint(250, 266),
        "page_width": rng.randrange(276, 321, 4),
        "page_height": rng.randrange(324, 373, 4),
        "background_color": rng.choice(BACKGROUNDS),
        "paper_color": rng.choice(PAPERS),
        "cover_color": rng.choice(COVERS),
        "rule_color": rng.choice(RULES),
        "binding_color": rng.choice(BINDINGS),
        "loop_count": rng.randint(8, 12),
        "line_count": rng.randint(8, 11),
        "page_stack": rng.randint(2, 4),
    }


def analytic_gold(scene: dict) -> str:
    side = scene["binding_side"]
    if side not in {"left", "right"}:
        raise ValueError(f"invalid binding side {side!r}")
    return side


def margin(scene: dict) -> float:
    """Analytic winner-minus-runner-up visible loop count."""
    return float(scene["loop_count"])


def is_quarantined(scene: dict) -> bool:
    return int(scene["loop_count"]) < 6


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No pixel-identical aliases: every sampled nuisance is visibly rendered."""
    return []


def horizontal_counterfactual(scene: dict) -> dict:
    """Declared label-transforming reflection used by candidate tests."""
    twin = deepcopy(scene)
    twin["binding_side"] = "right" if scene["binding_side"] == "left" else "left"
    twin["center_x"] = CANVAS - int(scene["center_x"])
    return twin


def _scaled_box(box: tuple[float, float, float, float]) -> tuple[int, int, int, int]:
    return tuple(round(v * SS) for v in box)


def render(scene: dict) -> Image.Image:
    """Render a familiar ruled notebook with separated, measurable spiral loops."""
    side = analytic_gold(scene)
    cx = int(scene["center_x"])
    cy = int(scene["center_y"])
    pw = int(scene["page_width"])
    ph = int(scene["page_height"])
    x0 = cx - pw // 2
    y0 = cy - ph // 2
    x1 = x0 + pw
    y1 = y0 + ph

    image = Image.new("RGB", (CANVAS * SS, CANVAS * SS), scene["background_color"])
    draw = ImageDraw.Draw(image)

    # Soft block shadow, colored cover, and visible page stack establish the object body.
    draw.rounded_rectangle(
        _scaled_box((x0 + 10, y0 + 13, x1 + 15, y1 + 18)),
        radius=10 * SS,
        fill="#c8c5bd",
    )
    cover_shift = 7 if side == "left" else -7
    draw.rounded_rectangle(
        _scaled_box((x0 + cover_shift - 5, y0 + 7, x1 + cover_shift + 5, y1 + 10)),
        radius=10 * SS,
        fill=scene["cover_color"],
    )
    for layer in range(int(scene["page_stack"]), 0, -1):
        off = layer * 2
        draw.rounded_rectangle(
            _scaled_box((x0 + off, y0 + off, x1 + off, y1 + off)),
            radius=7 * SS,
            fill="#e4dfd2",
        )
    draw.rounded_rectangle(
        _scaled_box((x0, y0, x1, y1)),
        radius=7 * SS,
        fill=scene["paper_color"],
    )

    # Ruled paper and a colored header line vary independently of the answer.
    inner_left = x0 + 38
    inner_right = x1 - 28
    top_rule = y0 + 45
    spacing = (ph - 82) / max(1, int(scene["line_count"]) - 1)
    for idx in range(int(scene["line_count"])):
        yy = top_rule + idx * spacing
        draw.line(
            [(inner_left * SS, round(yy * SS)), (inner_right * SS, round(yy * SS))],
            fill=scene["rule_color"],
            width=2 * SS,
        )
    draw.line(
        [((x0 + 55) * SS, (y0 + 23) * SS), ((x1 - 38) * SS, (y0 + 23) * SS)],
        fill=scene["cover_color"],
        width=3 * SS,
    )

    # Each full ellipse is disconnected from the page and from every other loop.
    loops = int(scene["loop_count"])
    ys = [y0 + 31 + i * (ph - 62) / (loops - 1) for i in range(loops)]
    edge_x = x0 if side == "left" else x1
    for yy in ys:
        hole_x = edge_x + (10 if side == "left" else -10)
        draw.ellipse(
            _scaled_box((hole_x - 4, yy - 4, hole_x + 4, yy + 4)),
            fill="#c9c3b5",
        )
        loop_box = (edge_x - 19, yy - 8, edge_x + 15, yy + 8)
        if side == "right":
            loop_box = (edge_x - 15, yy - 8, edge_x + 19, yy + 8)
        draw.ellipse(
            _scaled_box(loop_box),
            outline=scene["binding_color"],
            width=4 * SS,
        )

    return image.resize((CANVAS, CANVAS), Image.Resampling.LANCZOS)
