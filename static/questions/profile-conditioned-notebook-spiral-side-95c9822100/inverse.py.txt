"""PNG-only inverse arm for locating a notebook's repeated spiral loops."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[dict]:
    seen = np.zeros(mask.shape, dtype=bool)
    height, width = mask.shape
    found: list[dict] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            yy, xx = queue.popleft()
            xs.append(xx)
            ys.append(yy)
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    ny, nx = yy + dy, xx + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        found.append(
            {
                "pixels": len(xs),
                "x0": min(xs),
                "x1": max(xs),
                "y0": min(ys),
                "y1": max(ys),
                "cx": float(sum(xs)) / len(xs),
            }
        )
    return found


def measure_binding_sides(image: Image.Image) -> dict:
    """Return candidate loop counts on each side using pixels and no scene hints."""
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    if arr.ndim != 3 or min(arr.shape[:2]) < 64:
        raise ValueError("image is too small")

    # Infer the background from all four corners, then locate the full object extent.
    corners = np.array([arr[0, 0], arr[0, -1], arr[-1, 0], arr[-1, -1]], dtype=np.int16)
    background = np.median(corners, axis=0)
    distance = np.sqrt(np.sum((arr - background) ** 2, axis=2))
    foreground = distance > 28.0
    fy, fx = np.nonzero(foreground)
    if len(fx) < 500:
        raise ValueError("no notebook body detected")
    body_mid_x = (float(fx.min()) + float(fx.max())) / 2.0

    # Binding palettes are deliberately near-black; all other marks have a bright channel.
    dark = (arr.max(axis=2) < 100) & (arr.sum(axis=2) < 235)
    candidates = []
    for item in _components(dark):
        component_width = item["x1"] - item["x0"] + 1
        component_height = item["y1"] - item["y0"] + 1
        if (
            item["pixels"] >= 28
            and 22 <= component_width <= 48
            and 8 <= component_height <= 24
        ):
            candidates.append(item)
    left = sum(item["cx"] < body_mid_x for item in candidates)
    right = sum(item["cx"] > body_mid_x for item in candidates)
    return {
        "left_count": int(left),
        "right_count": int(right),
        "runner_up_margin": float(abs(left - right)),
        "loop_components": len(candidates),
        "body_mid_x": body_mid_x,
    }


def decision_from_image(image: Image.Image) -> str:
    measurement = measure_binding_sides(image)
    left = measurement["left_count"]
    right = measurement["right_count"]
    if max(left, right) < 5 or abs(left - right) < 4:
        raise ValueError("binding evidence is absent or ambiguous")
    return "left" if left > right else "right"
