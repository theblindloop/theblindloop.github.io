"""Independent pixel-only inverse for skyline-matrix correspondence."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE = {
    "ink": (17, 24, 39),
    "grid": (203, 213, 225),
    "red": (220, 38, 38),
    "green": (22, 163, 74),
    "blue": (37, 99, 235),
    "orange": (234, 88, 12),
}


def _mask(arr: np.ndarray, rgb: tuple[int, int, int], tolerance: int = 4) -> np.ndarray:
    delta = arr.astype(np.int16) - np.asarray(rgb, dtype=np.int16)
    return np.max(np.abs(delta), axis=2) <= tolerance


def _components(mask: np.ndarray) -> list[dict]:
    work = mask.copy()
    height, width = work.shape
    components: list[dict] = []
    for y, x in np.argwhere(work):
        y, x = int(y), int(x)
        if not work[y, x]:
            continue
        work[y, x] = False
        queue = deque([(x, y)])
        count = 0
        sx = sy = 0
        min_x = max_x = x
        min_y = max_y = y
        while queue:
            px, py = queue.popleft()
            count += 1
            sx += px
            sy += py
            min_x, max_x = min(min_x, px), max(max_x, px)
            min_y, max_y = min(min_y, py), max(max_y, py)
            for nx, ny in ((px - 1, py), (px + 1, py), (px, py - 1), (px, py + 1)):
                if 0 <= nx < width and 0 <= ny < height and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((nx, ny))
        components.append(
            {
                "area": count,
                "bbox": (min_x, min_y, max_x, max_y),
                "cx": sx / count,
                "cy": sy / count,
                "width": max_x - min_x + 1,
                "height": max_y - min_y + 1,
            }
        )
    return components


def _three_group_counts(components: list[dict], axis: str) -> list[int]:
    values = sorted((item["cx"] if axis == "x" else item["cy"]) for item in components)
    groups: list[list[float]] = []
    for value in values:
        if not groups or abs(value - sum(groups[-1]) / len(groups[-1])) > 5:
            groups.append([value])
        else:
            groups[-1].append(value)
    if len(groups) != 3 or any(not 1 <= len(group) <= 3 for group in groups):
        raise ValueError(f"expected three readable clue stacks, got {[len(g) for g in groups]}")
    return [len(group) for group in groups]


def _card_boxes(arr: np.ndarray) -> list[tuple[float, str, tuple[int, int, int, int]]]:
    found = []
    for name in ("red", "green", "blue"):
        comps = _components(_mask(arr, PALETTE[name]))
        frames = [c for c in comps if c["width"] > 180 and c["height"] > 180 and c["area"] > 1500]
        if len(frames) != 1:
            raise ValueError(f"expected one {name} candidate frame, got {len(frames)}")
        found.append((frames[0]["cx"], name, frames[0]["bbox"]))
    return sorted(found)


def _decode_grid(arr: np.ndarray, box: tuple[int, int, int, int]) -> list[list[int]]:
    x0, y0, x1, y1 = box
    grid_comps = _components(_mask(arr, PALETTE["grid"]))
    inside_gray = [
        c
        for c in grid_comps
        if x0 < c["cx"] < x1 and y0 < c["cy"] < y1 and c["width"] > 150 and c["height"] > 150
    ]
    if len(inside_gray) != 1:
        raise ValueError("candidate cell lattice is missing or ambiguous")
    gx0, gy0, gx1, gy1 = inside_gray[0]["bbox"]
    blocks = [
        c
        for c in _components(_mask(arr, PALETTE["ink"]))
        if gx0 < c["cx"] < gx1
        and gy0 < c["cy"] < gy1
        and 80 <= c["area"] <= 250
        and c["width"] <= 20
        and c["height"] <= 20
    ]
    counts = [[0, 0, 0] for _ in range(3)]
    span_x = gx1 - gx0 + 1
    span_y = gy1 - gy0 + 1
    for block in blocks:
        col = min(2, max(0, int(3 * (block["cx"] - gx0) / span_x)))
        row = min(2, max(0, int(3 * (block["cy"] - gy0) / span_y)))
        counts[row][col] += 1
    if any(value not in (1, 2, 3) for row in counts for value in row):
        raise ValueError(f"candidate contains unreadable stack heights {counts}")
    return counts


def _signature(grid: list[list[int]]) -> tuple[list[int], list[int]]:
    return (
        [max(grid[r][c] for r in range(3)) for c in range(3)],
        [max(row) for row in grid],
    )


def decision_from_image(image: Image.Image) -> str:
    """Recover clue stacks, candidate matrices, and selected border from pixels only."""
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    blue_small = [
        c
        for c in _components(_mask(arr, PALETTE["blue"]))
        if 80 <= c["area"] <= 250 and c["width"] <= 20 and c["height"] <= 20
    ]
    orange_small = [
        c
        for c in _components(_mask(arr, PALETTE["orange"]))
        if 80 <= c["area"] <= 250 and c["width"] <= 20 and c["height"] <= 20
    ]
    top = _three_group_counts(blue_small, "x")
    left = _three_group_counts(orange_small, "y")
    cards = _card_boxes(arr)
    decoded = [(name, _decode_grid(arr, box)) for _, name, box in cards]
    winners = [name for name, grid in decoded if _signature(grid) == (top, left)]
    if len(winners) != 1:
        raise ValueError(f"pixel evidence yields {len(winners)} matching candidates")
    return winners[0]
