"""Deterministic scene model and rasterizer for skyline-matrix correspondence."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 900, 640
BACKGROUND = (248, 250, 252)
INK = (17, 24, 39)
GRID = (203, 213, 225)
CARD = (255, 255, 255)
CLUE_BLUE = (37, 99, 235)
CLUE_ORANGE = (234, 88, 12)
BORDER_RGB = {
    "red": (220, 38, 38),
    "green": (22, 163, 74),
    "blue": CLUE_BLUE,
}
BORDER_NAMES = ("red", "green", "blue")


def _signature(grid: list[list[int]]) -> tuple[list[int], list[int]]:
    top = [max(grid[r][c] for r in range(3)) for c in range(3)]
    left = [max(row) for row in grid]
    return top, left


def _mismatches(grid: list[list[int]], top: list[int], left: list[int]) -> int:
    got_top, got_left = _signature(grid)
    return sum(a != b for a, b in zip(got_top + got_left, top + left))


def _matrix(values: list[int]) -> list[list[int]]:
    return [values[0:3], values[3:6], values[6:9]]


def sample_scene(seed: int) -> dict:
    """Sample one unique-match scene with equal histograms in all candidate cards."""
    rng = random.Random(int(seed))
    correct = [[1, 1, 1], [2, 2, 2], [3, 3, 1]]
    rows = list(range(3))
    cols = list(range(3))
    rng.shuffle(rows)
    rng.shuffle(cols)
    correct = [[correct[r][c] for c in cols] for r in rows]
    if rng.randrange(2):
        correct = [list(row) for row in zip(*correct)]
    top, left = _signature(correct)

    flat = [value for row in correct for value in row]
    by_score: dict[int, list[list[list[int]]]] = {}
    seen: set[tuple[int, ...]] = {tuple(flat)}
    for _ in range(6000):
        values = flat[:]
        rng.shuffle(values)
        key = tuple(values)
        if key in seen:
            continue
        seen.add(key)
        grid = _matrix(values)
        score = _mismatches(grid, top, left)
        if score:
            by_score.setdefault(score, []).append(grid)
    if not by_score:
        raise RuntimeError("could not construct skyline distractors")

    desired = 1 + ((int(seed) // 3) % 3)
    first_score = min(by_score, key=lambda value: (abs(value - desired), value))
    first = rng.choice(by_score[first_score])
    alternatives = [
        grid
        for score, grids in by_score.items()
        for grid in grids
        if grid != first and score >= first_score
    ]
    if not alternatives:
        alternatives = [grid for grids in by_score.values() for grid in grids if grid != first]
    second = rng.choice(alternatives)

    answer_color = BORDER_NAMES[int(seed) % len(BORDER_NAMES)]
    other_colors = [name for name in BORDER_NAMES if name != answer_color]
    rng.shuffle(other_colors)
    pairs = [(correct, answer_color), (first, other_colors[0]), (second, other_colors[1])]
    rng.shuffle(pairs)
    return {
        "top_clues": top,
        "left_clues": left,
        "candidate_grids": [deepcopy(grid) for grid, _ in pairs],
        "border_colors": [color for _, color in pairs],
    }


def _draw_outline(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], color, width: int) -> None:
    x0, y0, x1, y1 = box
    draw.rectangle((x0, y0, x1, y0 + width - 1), fill=color)
    draw.rectangle((x0, y1 - width + 1, x1, y1), fill=color)
    draw.rectangle((x0, y0, x0 + width - 1, y1), fill=color)
    draw.rectangle((x1 - width + 1, y0, x1, y1), fill=color)


def _draw_stack(
    draw: ImageDraw.ImageDraw,
    center_x: int,
    bottom_y: int,
    height: int,
    color,
    *,
    block_w: int = 14,
    block_h: int = 11,
    gap: int = 4,
) -> None:
    for level in range(height):
        y1 = bottom_y - level * (block_h + gap)
        y0 = y1 - block_h + 1
        draw.rectangle((center_x - block_w // 2, y0, center_x + block_w // 2, y1), fill=color)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    # Large clue card. Its blank center makes the orthogonal projections explicit.
    clue_box = (260, 24, 640, 310)
    draw.rounded_rectangle(clue_box, radius=14, fill=CARD, outline=GRID, width=3)
    gx, gy, cell = 376, 126, 52
    draw.rectangle((gx, gy, gx + 3 * cell, gy + 3 * cell), fill=(252, 253, 255), outline=GRID, width=2)
    for k in (1, 2):
        draw.line((gx + k * cell, gy, gx + k * cell, gy + 3 * cell), fill=GRID, width=2)
        draw.line((gx, gy + k * cell, gx + 3 * cell, gy + k * cell), fill=GRID, width=2)
    for c, value in enumerate(scene["top_clues"]):
        _draw_stack(draw, gx + c * cell + cell // 2, gy - 13, int(value), CLUE_BLUE)
    for r, value in enumerate(scene["left_clues"]):
        center_y = gy + r * cell + cell // 2
        for level in range(int(value)):
            x1 = gx - 14 - level * 19
            draw.rectangle((x1 - 14, center_y - 6, x1, center_y + 5), fill=CLUE_ORANGE)

    # Candidate cards; only the joint six-clue signature distinguishes the match.
    card_y = 374
    for index, (grid, color_name) in enumerate(
        zip(scene["candidate_grids"], scene["border_colors"])
    ):
        x0 = 45 + index * 285
        y0 = card_y
        x1, y1 = x0 + 240, y0 + 240
        draw.rectangle((x0, y0, x1, y1), fill=CARD)
        _draw_outline(draw, (x0, y0, x1, y1), BORDER_RGB[color_name], 7)
        inner_x, inner_y, csize = x0 + 27, y0 + 27, 62
        draw.rectangle(
            (inner_x, inner_y, inner_x + 3 * csize, inner_y + 3 * csize),
            fill=(252, 253, 255),
            outline=GRID,
            width=2,
        )
        for k in (1, 2):
            draw.line(
                (inner_x + k * csize, inner_y, inner_x + k * csize, inner_y + 3 * csize),
                fill=GRID,
                width=2,
            )
            draw.line(
                (inner_x, inner_y + k * csize, inner_x + 3 * csize, inner_y + k * csize),
                fill=GRID,
                width=2,
            )
        for r in range(3):
            for c in range(3):
                _draw_stack(
                    draw,
                    inner_x + c * csize + csize // 2,
                    inner_y + (r + 1) * csize - 10,
                    int(grid[r][c]),
                    INK,
                )
    return image


def analytic_gold(scene: dict) -> str:
    top = scene["top_clues"]
    left = scene["left_clues"]
    grids = scene["candidate_grids"]
    colors = scene["border_colors"]
    winners = [i for i, grid in enumerate(grids) if _mismatches(grid, top, left) == 0]
    return colors[winners[0]] if len(winners) == 1 else "abstain"


def margin(scene: dict) -> float:
    scores = sorted(
        _mismatches(grid, scene["top_clues"], scene["left_clues"])
        for grid in scene["candidate_grids"]
    )
    return float(scores[1] - scores[0]) if len(scores) >= 2 else 0.0


def is_quarantined(scene: dict) -> bool:
    scores = [
        _mismatches(grid, scene["top_clues"], scene["left_clues"])
        for grid in scene["candidate_grids"]
    ]
    return scores.count(0) != 1 or sorted(scores)[1] <= sorted(scores)[0]


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
