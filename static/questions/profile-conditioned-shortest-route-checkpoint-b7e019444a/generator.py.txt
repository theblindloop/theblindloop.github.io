"""Render a metric node-link world with tied shortest routes and a longer bypass."""

from __future__ import annotations

import copy
import heapq
import math
import random

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 500
BACKGROUND = "#f8fafc"
DARK = "#263238"
GREEN = "#22a06b"
RED = "#d9485f"
GRAY = "#cbd5e1"
COLORS = {"orange": "#f28e2b", "purple": "#8b5cf6", "blue": "#3182ce"}

BASE = {
    "s": (60, 230),
    "t": (160, 230),
    "f": (240, 230),
    "u": (320, 145),
    "v": (320, 315),
    "g": (430, 230),
    "d": (125, 420),
    "e": (365, 420),
}
EDGES = [("s", "t"), ("t", "f"), ("f", "u"), ("u", "g"),
         ("f", "v"), ("v", "g"), ("s", "d"), ("d", "e"), ("e", "g")]


def _transform(point: tuple[int, int], quarter: int, mirror: bool, dx: int, dy: int):
    x, y = point
    x -= 250
    y -= 250
    if mirror:
        x = -x
    for _ in range(quarter):
        x, y = -y, x
    return [int(x + 250 + dx), int(y + 250 + dy)]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    answer_index = seed % 3
    labels = ["orange", "purple", "blue"]
    answer = labels[answer_index]
    others = [label for label in labels if label != answer]
    if rng.randrange(2):
        others.reverse()
    quarter = (seed // 3) % 4
    mirror = bool((seed // 12) % 2)
    dx = rng.randint(-7, 7)
    dy = rng.randint(-7, 7)
    positions = {name: _transform(pos, quarter, mirror, dx, dy) for name, pos in BASE.items()}
    roles = {
        "s": ("start", "green"), "g": ("goal", "red"),
        "t": ("checkpoint", answer), "u": ("checkpoint", others[0]),
        "v": ("checkpoint", others[1]), "f": ("junction", "gray"),
        "d": ("junction", "gray"), "e": ("junction", "gray"),
    }
    nodes = [
        {"id": name, "x": positions[name][0], "y": positions[name][1],
         "kind": roles[name][0], "color": roles[name][1]}
        for name in ("s", "t", "f", "u", "v", "g", "d", "e")
    ]
    return {
        "nodes": nodes,
        "edges": [list(edge) for edge in EDGES],
        "width": WIDTH,
        "height": HEIGHT,
        "background": BACKGROUND,
        "stroke": 7,
    }


def render(scene: dict) -> Image.Image:
    nodes = scene["nodes"]
    edges = scene["edges"]
    image = Image.new("RGB", (scene["width"], scene["height"]), scene["background"])
    draw = ImageDraw.Draw(image)
    by_id = {node["id"]: node for node in nodes}
    for left, right in edges:
        a, b = by_id[left], by_id[right]
        draw.line((a["x"], a["y"], b["x"], b["y"]), fill=DARK,
                  width=scene["stroke"])
    for node in nodes:
        x, y = node["x"], node["y"]
        kind, color = node["kind"], node["color"]
        if kind == "goal":
            draw.rectangle((x - 13, y - 13, x + 13, y + 13), fill="white", outline=DARK, width=2)
            draw.rectangle((x - 9, y - 9, x + 9, y + 9), fill=RED)
        else:
            draw.ellipse((x - 13, y - 13, x + 13, y + 13), fill="white", outline=DARK, width=2)
            fill = GREEN if color == "green" else GRAY if color == "gray" else COLORS[color]
            draw.ellipse((x - 9, y - 9, x + 9, y + 9), fill=fill)
    return image


def _distances(nodes: list[dict], edges: list[list[str]], start: str, removed: str | None = None):
    by_id = {node["id"]: node for node in nodes}
    graph = {name: [] for name in by_id if name != removed}
    for left, right in edges:
        if removed in (left, right):
            continue
        a, b = by_id[left], by_id[right]
        weight = math.hypot(a["x"] - b["x"], a["y"] - b["y"])
        graph[left].append((right, weight))
        graph[right].append((left, weight))
    distances = {name: math.inf for name in graph}
    distances[start] = 0.0
    queue = [(0.0, start)]
    while queue:
        distance, node = heapq.heappop(queue)
        if distance != distances[node]:
            continue
        for neighbor, weight in graph[node]:
            candidate = distance + weight
            if candidate < distances[neighbor]:
                distances[neighbor] = candidate
                heapq.heappush(queue, (candidate, neighbor))
    return distances


def _penalties(nodes: list[dict], edges: list[list[str]]):
    start = next(node["id"] for node in nodes if node["kind"] == "start")
    goal = next(node["id"] for node in nodes if node["kind"] == "goal")
    baseline = _distances(nodes, edges, start)[goal]
    result = []
    for node in nodes:
        if node["kind"] == "checkpoint":
            alternate = _distances(nodes, edges, start, node["id"])[goal]
            result.append((alternate - baseline, node["color"]))
    return sorted(result, reverse=True)


def analytic_gold(scene: dict) -> str:
    penalties = _penalties(scene["nodes"], scene["edges"])
    return penalties[0][1]


def margin(scene: dict) -> float:
    penalties = _penalties(scene["nodes"], scene["edges"])
    return float(penalties[0][0] - penalties[1][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 30.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["nodes"] = list(reversed(twin["nodes"]))
    twin["edges"] = [[right, left] for left, right in reversed(twin["edges"])]
    return [("reverse_storage_order", twin)]
