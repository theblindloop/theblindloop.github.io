"""Independent PNG-only graph recovery for shortest-route checkpoint selection."""

from __future__ import annotations

import heapq
import math
from collections import deque

import numpy as np

PALETTE = {
    "green": (34, 160, 107), "red": (217, 72, 95), "gray": (203, 213, 225),
    "orange": (242, 142, 43), "purple": (139, 92, 246), "blue": (49, 130, 206),
}
DARK = np.array((38, 50, 56), dtype=np.int16)


def _components(mask: np.ndarray):
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= 80:
            arr = np.asarray(points, dtype=float)
            output.append((float(arr[:, 1].mean()), float(arr[:, 0].mean())))
    return output


def _point_segment_distance(point, a, b):
    px, py = point
    ax, ay = a
    bx, by = b
    dx, dy = bx - ax, by - ay
    denominator = dx * dx + dy * dy
    if denominator == 0:
        return math.hypot(px - ax, py - ay), 0.0
    t = ((px - ax) * dx + (py - ay) * dy) / denominator
    qx, qy = ax + t * dx, ay + t * dy
    return math.hypot(px - qx, py - qy), t


def _has_link(dark: np.ndarray, a, b, all_centers):
    for center in all_centers:
        if center is a or center is b:
            continue
        distance, t = _point_segment_distance(center, a, b)
        if 0.08 < t < 0.92 and distance < 15.0:
            return False
    length = math.hypot(a[0] - b[0], a[1] - b[1])
    samples = max(20, int(length))
    hits = 0
    used = 0
    height, width = dark.shape
    for index in range(samples + 1):
        t = index / samples
        if t * length < 16 or (1 - t) * length < 16:
            continue
        x = int(round(a[0] + t * (b[0] - a[0])))
        y = int(round(a[1] + t * (b[1] - a[1])))
        x0, x1 = max(0, x - 3), min(width, x + 4)
        y0, y1 = max(0, y - 3), min(height, y + 4)
        used += 1
        hits += bool(dark[y0:y1, x0:x1].any())
    return used > 10 and hits / used > 0.94


def _shortest(graph, start, goal, removed=None):
    distances = {node: math.inf for node in graph if node != removed}
    if start == removed or goal == removed:
        return math.inf
    distances[start] = 0.0
    queue = [(0.0, start)]
    while queue:
        distance, node = heapq.heappop(queue)
        if distance != distances[node]:
            continue
        if node == goal:
            return distance
        for neighbor, weight in graph[node]:
            if neighbor == removed:
                continue
            candidate = distance + weight
            if candidate < distances[neighbor]:
                distances[neighbor] = candidate
                heapq.heappush(queue, (candidate, neighbor))
    return distances.get(goal, math.inf)


def decision_from_image(image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    nodes = []
    for label, rgb in PALETTE.items():
        distance = np.max(np.abs(pixels - np.array(rgb, dtype=np.int16)), axis=2)
        components = _components(distance <= 8)
        expected = 3 if label == "gray" else 1
        if len(components) != expected:
            raise ValueError(f"expected {expected} {label} marker(s), found {len(components)}")
        for center in components:
            nodes.append({"label": label, "center": center})
    if len(nodes) != 8:
        raise ValueError(f"expected eight visible nodes, found {len(nodes)}")
    dark = np.max(np.abs(pixels - DARK), axis=2) <= 10
    centers = [node["center"] for node in nodes]
    graph = {index: [] for index in range(len(nodes))}
    for left in range(len(nodes)):
        for right in range(left + 1, len(nodes)):
            if _has_link(dark, centers[left], centers[right], centers):
                weight = math.hypot(centers[left][0] - centers[right][0],
                                    centers[left][1] - centers[right][1])
                graph[left].append((right, weight))
                graph[right].append((left, weight))
    edge_count = sum(map(len, graph.values())) // 2
    if edge_count != 9:
        raise ValueError(f"expected nine visible links, recovered {edge_count}")
    start = next(i for i, node in enumerate(nodes) if node["label"] == "green")
    goal = next(i for i, node in enumerate(nodes) if node["label"] == "red")
    baseline = _shortest(graph, start, goal)
    penalties = []
    for index, node in enumerate(nodes):
        if node["label"] in ("orange", "purple", "blue"):
            penalties.append((_shortest(graph, start, goal, index) - baseline, node["label"]))
    penalties.sort(reverse=True)
    if not math.isfinite(penalties[0][0]) or penalties[0][0] - penalties[1][0] < 30.0:
        raise ValueError("ambiguous checkpoint removal margin")
    return penalties[0][1]
