"""Pixel-only decision arm for even_split_fraction.

Receives the final PNG and nothing else: no latent scene, no renderer module, no
analytic gold. It segments each coloured bar by hue, recovers that bar's two ends
and its own middle column, locates the black divider lying inside that bar's own
rows, and reports the colour with the smallest divider-to-middle distance
measured as a fraction of that bar's own length.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ABSTAIN = "abstain"

_REFERENCE = {
    "red": (214, 39, 40),
    "green": (44, 160, 44),
    "blue": (31, 119, 180),
    "orange": (255, 127, 14),
    "purple": (148, 60, 180),
}
_TOLERANCE = 40
_SATURATION = 45
_DARK = 60
_EXPECTED_COLORS = 5
_MIN_PIECE_AREA = 90
_MIN_BAR_LENGTH = 90
_BAR_ROWS = (17, 31)
_DIVIDER_WIDTH = (3, 8)
_MIN_CLEARANCE = 6.0


def _pieces(mask: np.ndarray) -> list[tuple[int, int, int, int]] | None:
    """Connected components of a boolean mask as (area, height, xmin, xmax)."""

    points = {(int(r), int(c)) for r, c in np.argwhere(mask)}
    out: list[tuple[int, int, int, int]] = []
    while points:
        stack = [points.pop()]
        blob = [stack[0]]
        while stack:
            row, col = stack.pop()
            for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                neighbour = (row + dr, col + dc)
                if neighbour in points:
                    points.discard(neighbour)
                    stack.append(neighbour)
                    blob.append(neighbour)
        rows = [point[0] for point in blob]
        cols = [point[1] for point in blob]
        out.append((len(blob), max(rows) - min(rows) + 1, min(cols), max(cols)))
        if len(out) > 24:
            return None
    return out


def decision_from_image(image: Image.Image) -> str:
    """Colour whose divider splits its own bar most evenly, from the raster alone."""

    data = np.asarray(image.convert("RGB")).astype(np.int16)
    saturation = data.max(axis=2) - data.min(axis=2)
    coloured = saturation > _SATURATION
    dark = data.max(axis=2) <= _DARK

    readings: dict[str, float] = {}
    lengths: dict[str, float] = {}
    for name, reference in _REFERENCE.items():
        distance = np.abs(data - np.array(reference, dtype=np.int16)).max(axis=2)
        mask = coloured & (distance <= _TOLERANCE)
        if not mask.any():
            continue
        pieces = _pieces(mask)
        if pieces is None:
            return ABSTAIN
        kept = [piece for piece in pieces if piece[0] >= _MIN_PIECE_AREA]
        if len(kept) != 2:
            return ABSTAIN
        for _, height, _, _ in kept:
            if not _BAR_ROWS[0] <= height <= _BAR_ROWS[1]:
                return ABSTAIN
        rows = np.flatnonzero(mask.any(axis=1))
        left = min(piece[2] for piece in kept)
        right = max(piece[3] for piece in kept)
        length = float(right - left + 1)
        if length < _MIN_BAR_LENGTH:
            return ABSTAIN

        band = dark[rows.min() : rows.max() + 1, left : right + 1]
        columns = np.flatnonzero(band.any(axis=0))
        if columns.size == 0:
            return ABSTAIN
        width = int(columns.max() - columns.min() + 1)
        if not _DIVIDER_WIDTH[0] <= width <= _DIVIDER_WIDTH[1]:
            return ABSTAIN
        if int(columns.size) != width:
            return ABSTAIN
        centre = left + (float(columns.min()) + float(columns.max())) / 2.0
        middle = (float(left) + float(right)) / 2.0
        readings[name] = abs(centre - middle) / length
        lengths[name] = length

    if len(readings) != _EXPECTED_COLORS:
        return ABSTAIN
    ranked = sorted(readings.items(), key=lambda item: item[1])
    slack = ranked[1][1] * lengths[ranked[0][0]] - ranked[0][1] * lengths[ranked[0][0]]
    if slack < _MIN_CLEARANCE:
        return ABSTAIN
    return ranked[0][0]
