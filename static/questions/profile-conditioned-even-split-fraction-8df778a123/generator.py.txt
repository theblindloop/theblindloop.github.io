"""Scene sampling, rasterisation and analytic gold for even_split_fraction.

Five colour-coded horizontal bars are drawn, one to a lane, each cut by a single
black divider. The decided quantity is a RATIO: how far a bar's divider sits from
that bar's own middle, expressed as a fraction of that bar's own length. The
answer is the colour whose divider splits its bar most nearly into two equal
halves. Because the deviation is normalised by the bar's own length, the raw
pixel distance from a divider to its bar's middle deliberately names a different
colour, as do the longest bar, the shortest bar, the largest left piece, the
largest right piece and the divider nearest the middle of the page.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

W = 780
H = 520
X0 = 30
X1 = 750
XMIN = 46
XMAX = 734
SLOT_Y = (72, 144, 216, 288, 360, 432)
BAR_HALF = 12
DIV_HALF_W = 2
DIV_OVER = 8
PAGE_CENTRE = (XMIN + XMAX) / 2.0

BG = (255, 255, 255)
RULE = (236, 236, 236)
FRAME = (150, 150, 150)
GREY_BAR = (158, 158, 158)
BLACK = (0, 0, 0)

PALETTE = {
    "red": (214, 39, 40),
    "green": (44, 160, 44),
    "blue": (31, 119, 180),
    "orange": (255, 127, 14),
    "purple": (148, 60, 180),
}
COLORS = tuple(sorted(PALETTE))

MIN_MARGIN = 12.0
QUARANTINE_MARGIN = 22.0
MAX_OFFSET_FRACTION = 0.38


def _length(bar: dict) -> float:
    """Drawn length of a bar in pixels, endpoints included."""

    return float(int(bar["x1"]) - int(bar["x0"]) + 1)


def _middle(bar: dict) -> float:
    """Pixel column half way between a bar's own two ends."""

    return (float(int(bar["x0"])) + float(int(bar["x1"]))) / 2.0


def _offset(bar: dict) -> float:
    """Unsigned pixel distance from a bar's divider to that bar's own middle."""

    return abs(float(int(bar["xdiv"])) - _middle(bar))


def _split_ratio(left: int, right: int, divider: int) -> float:
    """Divider-to-middle distance as a share of that bar's own length."""

    length = float(int(right) - int(left) + 1)
    middle = (float(int(left)) + float(int(right))) / 2.0
    return abs(float(int(divider)) - middle) / length


def _deviation(bar: dict) -> float:
    """Offset expressed as a fraction of that bar's own length."""

    return _offset(bar) / _length(bar)


def _ranked(scene: dict) -> list[tuple[float, str]]:
    """(relative deviation, colour) pairs, most nearly even split first."""

    return sorted(
        (_split_ratio(bar["x0"], bar["x1"], bar["xdiv"]), bar["color"])
        for bar in scene["bars"]
    )


def _winner_bar(scene: dict) -> dict:
    winner = _ranked(scene)[0][1]
    return next(bar for bar in scene["bars"] if bar["color"] == winner)


def _place(rng: random.Random, length: int, deviation: float) -> tuple[int, int, int]:
    """Choose a bar's two ends and its divider column for a target deviation."""

    x0 = rng.randint(XMIN, XMAX - length + 1)
    x1 = x0 + length - 1
    sign = 1 if rng.random() < 0.5 else -1
    xdiv = int(round((x0 + x1) / 2.0 + sign * deviation * length))
    return x0, x1, xdiv


def _cues_clear(scene: dict) -> bool:
    """Reject any scene in which a cheap surface reading names the answer."""

    bars = scene["bars"]
    if len(bars) != len(COLORS):
        return False
    ranked = _ranked(scene)
    winner = ranked[0][1]
    if ranked[1][0] <= ranked[0][0]:
        return False
    if margin(scene) < MIN_MARGIN:
        return False
    for bar in bars:
        if _deviation(bar) > MAX_OFFSET_FRACTION:
            return False
        if _length(bar) / 2.0 - _offset(bar) < 16.0:
            return False
        if not XMIN <= int(bar["x0"]) < int(bar["x1"]) <= XMAX:
            return False
    longest = max(bars, key=_length)["color"]
    shortest = min(bars, key=_length)["color"]
    nearest_raw = min(bars, key=_offset)["color"]
    widest_left = max(bars, key=lambda bar: int(bar["xdiv"]) - int(bar["x0"]))["color"]
    widest_right = max(bars, key=lambda bar: int(bar["x1"]) - int(bar["xdiv"]))["color"]
    page_centred = min(
        bars, key=lambda bar: abs(int(bar["xdiv"]) - PAGE_CENTRE)
    )["color"]
    highest_lane = min(bars, key=lambda bar: int(bar["slot"]))["color"]
    if winner in {
        longest,
        shortest,
        nearest_raw,
        widest_left,
        widest_right,
        page_centred,
        highest_lane,
    }:
        return False
    # The raw-pixel shortcut must be a genuine trap, not a coincidence.
    return _offset(nearest_bar(scene)) < _offset(_winner_bar(scene))


def nearest_bar(scene: dict) -> dict:
    """Bar whose divider is nearest its own middle in raw pixels (a decoy)."""

    return min(scene["bars"], key=_offset)


def sample_scene(seed: int) -> dict:
    """Deterministic latent scene for one integer seed."""

    rng = random.Random(seed)
    winner_colour = COLORS[seed % len(COLORS)]
    others = [colour for colour in COLORS if colour != winner_colour]
    for _ in range(20000):
        pool = list(others)
        rng.shuffle(pool)
        raw_decoy, long_decoy = pool[0], pool[1]
        fillers = pool[2:]

        win_len = rng.randint(330, 430)
        dev_win = rng.uniform(0.050, 0.105)
        win_off = dev_win * win_len

        long_len = rng.randint(win_len + 50, 500)
        short_len = rng.randint(108, 150)
        raw_off = win_off * rng.uniform(0.50, 0.80)
        dev_raw = raw_off / short_len
        if dev_raw < dev_win * 1.45 or dev_raw > MAX_OFFSET_FRACTION:
            continue

        plan = {
            winner_colour: (win_len, dev_win),
            raw_decoy: (short_len, dev_raw),
            long_decoy: (long_len, rng.uniform(dev_win * 1.9, MAX_OFFSET_FRACTION)),
        }
        for colour in fillers:
            plan[colour] = (
                rng.randint(160, 500),
                rng.uniform(dev_win * 1.9, MAX_OFFSET_FRACTION),
            )

        slots = list(range(len(SLOT_Y)))
        rng.shuffle(slots)
        bars = []
        for index, colour in enumerate(COLORS):
            length, deviation = plan[colour]
            x0, x1, xdiv = _place(rng, length, deviation)
            bars.append(
                {"color": colour, "slot": slots[index], "x0": x0, "x1": x1, "xdiv": xdiv}
            )
        rng.shuffle(bars)

        grey_len = rng.randint(160, 460)
        gx0, gx1, gxdiv = _place(rng, grey_len, rng.uniform(0.05, 0.34))
        decoy = {"slot": slots[len(COLORS)], "x0": gx0, "x1": gx1, "xdiv": gxdiv}
        rules = sorted(rng.sample(range(XMIN + 20, XMAX - 20), 4))
        scene = {"bars": bars, "decoy": decoy, "rules": rules}
        if _ranked(scene)[0][1] != winner_colour:
            continue
        if _cues_clear(scene):
            return scene
    raise RuntimeError(f"scene sampling failed for seed {seed}")


def _slot_of(bar: dict) -> int:
    return int(bar["slot"])


def _draw_bar(draw: ImageDraw.ImageDraw, bar: dict, fill: tuple[int, int, int]) -> None:
    y = SLOT_Y[int(bar["slot"])]
    draw.rectangle(
        [int(bar["x0"]), y - BAR_HALF, int(bar["x1"]), y + BAR_HALF], fill=fill
    )
    xdiv = int(bar["xdiv"])
    draw.rectangle(
        [
            xdiv - DIV_HALF_W,
            y - BAR_HALF - DIV_OVER,
            xdiv + DIV_HALF_W,
            y + BAR_HALF + DIV_OVER,
        ],
        fill=BLACK,
    )


def render(scene: dict) -> Image.Image:
    """Rasterise a latent scene; the bar list is drawn as an unordered set."""

    image = Image.new("RGB", (W, H), BG)
    draw = ImageDraw.Draw(image)
    for rule in scene["rules"]:
        x = int(rule)
        draw.line([(x, 40), (x, H - 34)], fill=RULE, width=2)
    draw.rectangle([X0, 26, X1, H - 26], outline=FRAME, width=2)
    for bar in sorted(scene["bars"], key=_slot_of):
        _draw_bar(draw, bar, PALETTE[bar["color"]])
    _draw_bar(draw, scene["decoy"], GREY_BAR)
    return image


def analytic_gold(scene: dict) -> str:
    """Colour whose divider splits its own bar most nearly into equal halves."""

    return _ranked(scene)[0][1]


def margin(scene: dict) -> float:
    """Pixels the winning divider must slide to hand the answer to the runner-up."""

    ranked = _ranked(scene)
    winner = _winner_bar(scene)
    slack = ranked[1][0] * _length(winner) - _offset(winner)
    return float(round(slack, 6))


def is_quarantined(scene: dict) -> bool:
    """Near-boundary scenes excluded from headline claims."""

    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical latent relabelings that must preserve the gold."""

    bars = scene["bars"]
    swapped_ends = [
        {
            "color": bar["color"],
            "slot": bar["slot"],
            "x0": bar["x0"],
            "x1": bar["x1"],
            "xdiv": bar["xdiv"],
        }
        for bar in reversed(bars)
    ]
    return [
        (
            "reversed_bar_list",
            {"bars": list(reversed(bars)), "decoy": dict(scene["decoy"]), "rules": list(scene["rules"])},
        ),
        (
            "rotated_bar_list",
            {"bars": bars[2:] + bars[:2], "decoy": dict(scene["decoy"]), "rules": list(scene["rules"])},
        ),
        (
            "rebuilt_bar_records",
            {"bars": swapped_ends, "decoy": dict(scene["decoy"]), "rules": list(scene["rules"])},
        ),
        (
            "reversed_rule_list",
            {"bars": list(bars), "decoy": dict(scene["decoy"]), "rules": list(reversed(scene["rules"]))},
        ),
    ]
