"""Deterministic Pillow renderer and analytic arm for a paired-lamp relay cord."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw

WIDTH = 800
HEIGHT = 800
CELL = 100

FLIP = "flip"
COPY = "copy"
OPERATIONS = (FLIP, COPY)
DECISIONS = ("neither", "left only", "right only", "both")
_TARGET_STATES = ((0, 0), (1, 0), (0, 1), (1, 1))

_BACKGROUND = (255, 255, 255)
_INK = (31, 38, 52)
_START_FILL = (247, 248, 250)
_DISTRACTOR = (226, 230, 236)
_PALETTES = (
    ((242, 174, 72), (83, 190, 178), (219, 133, 163), (112, 162, 231)),
    ((151, 126, 224), (239, 151, 89), (91, 184, 126), (225, 117, 117)),
    ((91, 175, 218), (236, 185, 73), (185, 127, 207), (94, 191, 153)),
)


def _apply(state: tuple[int, int], operation: str) -> tuple[int, int]:
    left, right = state
    if operation == FLIP:
        return 1 - left, right
    if operation == COPY:
        return left, left
    raise ValueError(f"unknown relay operation {operation!r}")


def _run(start: tuple[int, int], operations: list[str] | tuple[str, ...]) -> tuple[int, int]:
    state = start
    for operation in operations:
        state = _apply(state, operation)
    return state


def _candidate_programs(target: tuple[int, int]) -> tuple[tuple[tuple[int, int], tuple[str, ...]], ...]:
    candidates = []
    for start in _TARGET_STATES:
        for program in itertools.product(OPERATIONS, repeat=4):
            if len(set(program)) != 2:
                continue
            if _run(start, program) == target:
                candidates.append((start, program))
    return tuple(candidates)


_PROGRAMS = {target: _candidate_programs(target) for target in _TARGET_STATES}


def sample_scene(seed: int) -> dict:
    """Choose the final register first, then independently vary program and style."""
    seed = int(seed)
    target = _TARGET_STATES[seed % len(_TARGET_STATES)]
    program_rng = random.Random(seed * 1009 + 73)
    start, program = program_rng.choice(_PROGRAMS[target])
    style_rng = random.Random(seed * 7919 + 211)
    palette_variant = style_rng.randrange(len(_PALETTES))
    colors = list(range(4))
    style_rng.shuffle(colors)
    return {
        "start_left": start[0],
        "start_right": start[1],
        "operations": list(program),
        "anchor_row": style_rng.randint(1, 6),
        "anchor_col": style_rng.randint(1, 6),
        "palette_variant": palette_variant,
        "gate_colors": colors,
        "spacing": style_rng.choice((31, 32)),
        "lamp_radius": style_rng.choice((4, 5)),
        "distractor_variant": style_rng.randrange(4),
    }


def _center(scene: dict) -> tuple[int, int]:
    return (
        int(scene["anchor_col"]) * CELL + CELL // 2,
        int(scene["anchor_row"]) * CELL + CELL // 2,
    )


def _gate_centers(scene: dict) -> list[int]:
    cx, _ = _center(scene)
    spacing = int(scene["spacing"])
    return [cx + round(offset * spacing) for offset in (-1.5, -0.5, 0.5, 1.5)]


def _draw_distractors(draw: ImageDraw.ImageDraw, variant: int) -> None:
    layouts = (
        ((58, 58), (742, 742)),
        ((742, 58), (58, 742)),
        ((58, 58), (742, 58), (58, 742)),
        ((742, 742), (742, 58), (58, 742)),
    )
    for x, y in layouts[variant]:
        draw.line((x - 6, y, x + 6, y), fill=_DISTRACTOR, width=2)
        draw.line((x, y - 6, x, y + 6), fill=_DISTRACTOR, width=2)


def _draw_lamp(
    draw: ImageDraw.ImageDraw, center: tuple[int, int], radius: int, on: int
) -> None:
    x, y = center
    fill = _INK if int(on) else _BACKGROUND
    draw.ellipse(
        (x - radius, y - radius, x + radius, y + radius),
        fill=fill,
        outline=_INK,
        width=2,
    )


def _draw_triangle(
    draw: ImageDraw.ImageDraw, x: int, y: int, fill: tuple[int, int, int]
) -> None:
    draw.polygon(
        ((x, y - 9), (x + 8, y + 7), (x - 8, y + 7)),
        fill=fill,
        outline=_INK,
        width=2,
    )


def _draw_square(
    draw: ImageDraw.ImageDraw, x: int, y: int, fill: tuple[int, int, int]
) -> None:
    draw.rectangle((x - 8, y - 8, x + 8, y + 8), fill=fill, outline=_INK, width=2)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), _BACKGROUND)
    draw = ImageDraw.Draw(image)
    cx, cy = _center(scene)
    gate_xs = _gate_centers(scene)
    spacing = int(scene["spacing"])
    radius = int(scene["lamp_radius"])
    operations = list(scene["operations"])
    palette = _PALETTES[int(scene["palette_variant"])]
    gate_colors = list(scene["gate_colors"])

    _draw_distractors(draw, int(scene["distractor_variant"]))

    start_left = gate_xs[0] - spacing - 30
    start_right = start_left + 31
    draw.line((start_right, cy, gate_xs[-1] + spacing + 20, cy), fill=_INK, width=3)
    draw.rectangle(
        (start_left, cy - 14, start_right, cy + 14),
        fill=_START_FILL,
        outline=_INK,
        width=2,
    )
    _draw_lamp(draw, (start_left + 9, cy), radius, int(scene["start_left"]))
    _draw_lamp(draw, (start_left + 23, cy), radius, int(scene["start_right"]))

    for index, (x, operation) in enumerate(zip(gate_xs, operations, strict=True)):
        fill = palette[int(gate_colors[index])]
        if operation == FLIP:
            _draw_triangle(draw, x, cy, fill)
        elif operation == COPY:
            _draw_square(draw, x, cy, fill)
        else:
            raise ValueError(f"unknown relay operation {operation!r}")

    arrow_x = gate_xs[-1] + spacing + 20
    draw.polygon(
        ((arrow_x, cy), (arrow_x - 9, cy - 5), (arrow_x - 9, cy + 5)),
        fill=_INK,
    )
    return image


def analytic_gold(scene: dict) -> str:
    final = _run(
        (int(scene["start_left"]), int(scene["start_right"])),
        list(scene["operations"]),
    )
    return DECISIONS[_TARGET_STATES.index(final)]


def margin(scene: dict) -> float:
    """Conservative runner-up clearance for lamp fill and relay-shape readings."""
    return float(
        min(
            int(scene["spacing"]) - 18,
            int(scene["lamp_radius"]) * 2,
            12,
        )
    )


def is_quarantined(scene: dict) -> bool:
    operations = list(scene["operations"])
    gate_colors = list(scene["gate_colors"])
    return (
        len(operations) != 4
        or set(operations) != set(OPERATIONS)
        or len(gate_colors) != 4
        or sorted(gate_colors) != [0, 1, 2, 3]
        or int(scene["spacing"]) < 31
        or int(scene["lamp_radius"]) < 4
        or int(scene["palette_variant"]) not in range(len(_PALETTES))
        or int(scene["anchor_row"]) not in range(1, 7)
        or int(scene["anchor_col"]) not in range(1, 7)
        or margin(scene) < 8.0
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
