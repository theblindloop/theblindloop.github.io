"""Independent pixel-only inverse arm for the four-panel dot plots."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _colored_components(image: Image.Image) -> list[dict]:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    high = pixels.max(axis=2)
    low = pixels.min(axis=2)
    mask = (high - low >= 80) & (high >= 130) & (low <= 190)
    remaining = {tuple(item) for item in np.argwhere(mask)}
    components: list[dict] = []
    while remaining:
        start = remaining.pop()
        queue = deque([start])
        points = [start]
        while queue:
            y, x = queue.popleft()
            for neighbor in (
                (y - 1, x - 1), (y - 1, x), (y - 1, x + 1),
                (y, x - 1), (y, x + 1),
                (y + 1, x - 1), (y + 1, x), (y + 1, x + 1),
            ):
                if neighbor in remaining:
                    remaining.remove(neighbor)
                    queue.append(neighbor)
                    points.append(neighbor)
        if len(points) < 20:
            continue
        ys = np.fromiter((item[0] for item in points), dtype=float)
        xs = np.fromiter((item[1] for item in points), dtype=float)
        components.append(
            {
                "area": len(points),
                "x": float(xs.mean()),
                "y": float(ys.mean()),
                "width": int(xs.max() - xs.min() + 1),
                "height": int(ys.max() - ys.min() + 1),
            }
        )
    return components


def decision_from_image(image: Image.Image) -> str:
    """Return A-D using marker pixels and relative 2x2 reading order only."""
    components = _colored_components(image)
    disks = [
        component
        for component in components
        if 150 <= component["area"] <= 270
        and 15 <= component["width"] <= 19
        and 15 <= component["height"] <= 19
    ]
    if len(disks) != 12:
        raise ValueError(f"expected 12 intact colored marker disks, found {len(disks)}")

    ordered_y = sorted(disks, key=lambda component: component["y"])
    top, bottom = ordered_y[:6], ordered_y[6:]
    if min(item["y"] for item in bottom) - max(item["y"] for item in top) < 55:
        raise ValueError("the two panel rows are not cleanly separated")

    panels: list[list[dict]] = []
    for row in (top, bottom):
        ordered_x = sorted(row, key=lambda component: component["x"])
        panels.extend((ordered_x[:3], ordered_x[3:]))

    residuals: list[float] = []
    for panel in panels:
        panel = sorted(panel, key=lambda component: component["x"])
        gaps = [panel[1]["x"] - panel[0]["x"], panel[2]["x"] - panel[1]["x"]]
        if min(gaps) < 55 or max(gaps) / min(gaps) > 1.15:
            raise ValueError("checkpoint columns are missing or irregular")
        residuals.append(abs(panel[1]["y"] - (panel[0]["y"] + panel[2]["y"]) / 2.0))

    ranking = sorted(range(4), key=lambda index: residuals[index], reverse=True)
    if residuals[ranking[0]] - residuals[ranking[1]] < 3.0:
        raise ValueError("pixel runner-up margin is ambiguous")
    return ("A", "B", "C", "D")[ranking[0]]
