"""Deterministic renderer and analytic arm for midpoint chord residuals."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageColor, ImageDraw

WIDTH, HEIGHT = 720, 520
LABELS = ("A", "B", "C", "D")
PALETTE = ("#2563EB", "#EA580C", "#16A34A", "#C026D3")
PANEL_ORIGINS = ((35, 25), (375, 25), (35, 275), (375, 275))
PANEL_W, PANEL_H = 310, 220
PLOT_LEFT, PLOT_RIGHT = 52, 275
PLOT_TOP, PLOT_BOTTOM = 27, 182
MARKER_RADIUS = 8


def sample_scene(seed: int) -> dict:
    """Sample a counterbalanced, separated four-panel scene from one integer seed."""
    rng = random.Random(int(seed))
    winner = int(seed) % 4
    others = [index for index in range(4) if index != winner]
    runner = others[rng.randrange(len(others))]
    near_boundary = int(seed) % 12 == 0

    residuals = [rng.uniform(4.0, 9.0) for _ in range(4)]
    if near_boundary:
        residuals[runner] = rng.uniform(13.6, 14.0)
        residuals[winner] = residuals[runner] + rng.uniform(3.8, 4.4)
    else:
        residuals[runner] = rng.uniform(12.0, 16.0)
        residuals[winner] = rng.uniform(25.0, 30.0)

    values: list[list[float]] = []
    for residual in residuals:
        chord_mid = rng.uniform(41.0, 59.0)
        half_endpoint_change = rng.uniform(-17.0, 17.0)
        sign = -1.0 if rng.random() < 0.5 else 1.0
        values.append(
            [
                round(chord_mid - half_endpoint_change, 4),
                round(chord_mid + sign * residual, 4),
                round(chord_mid + half_endpoint_change, 4),
            ]
        )

    return {
        "values": values,
        "colors": rng.sample(list(PALETTE), k=4),
        "shift_x": rng.randint(-6, 6),
        "shift_y": rng.randint(-6, 6),
    }


def _residuals(scene: dict) -> list[float]:
    values = scene["values"]
    return [abs(row[1] - (row[0] + row[2]) / 2.0) for row in values]


def analytic_gold(scene: dict) -> str:
    residuals = _residuals(scene)
    return LABELS[max(range(4), key=lambda index: residuals[index])]


def margin(scene: dict) -> float:
    ranked = sorted(_residuals(scene), reverse=True)
    return ranked[0] - ranked[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 5.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _value_y(origin_y: int, value: float) -> int:
    span = PLOT_BOTTOM - PLOT_TOP
    return round(origin_y + PLOT_BOTTOM - (value / 100.0) * span)


def _dashed_line(
    draw: ImageDraw.ImageDraw,
    start: tuple[int, int],
    end: tuple[int, int],
    *,
    fill: str,
    width: int = 2,
) -> None:
    dx, dy = end[0] - start[0], end[1] - start[1]
    length = math.hypot(dx, dy)
    if length == 0:
        return
    ux, uy = dx / length, dy / length
    position = 0.0
    while position < length:
        finish = min(position + 7.0, length)
        draw.line(
            (
                round(start[0] + ux * position),
                round(start[1] + uy * position),
                round(start[0] + ux * finish),
                round(start[1] + uy * finish),
            ),
            fill=fill,
            width=width,
        )
        position += 12.0


def render(scene: dict) -> Image.Image:
    """Render the final RGB PNG world; every gold input appears as marker position."""
    values = scene["values"]
    colors = scene["colors"]
    shift_x = int(scene["shift_x"])
    shift_y = int(scene["shift_y"])

    image = Image.new("RGB", (WIDTH, HEIGHT), "#FFFFFF")
    draw = ImageDraw.Draw(image)
    for panel_index, ((base_x, base_y), row, color_name) in enumerate(
        zip(PANEL_ORIGINS, values, colors)
    ):
        origin_x = base_x + shift_x
        origin_y = base_y + shift_y
        draw.rounded_rectangle(
            (origin_x, origin_y, origin_x + PANEL_W, origin_y + PANEL_H),
            radius=10,
            fill="#FAFAFA",
            outline="#9CA3AF",
            width=2,
        )
        draw.text((origin_x + 13, origin_y + 8), LABELS[panel_index], fill="#111827")

        x_positions = (
            origin_x + PLOT_LEFT,
            origin_x + (PLOT_LEFT + PLOT_RIGHT) // 2,
            origin_x + PLOT_RIGHT,
        )
        for level in (0, 50, 100):
            y = _value_y(origin_y, float(level))
            draw.line((x_positions[0], y, x_positions[2], y), fill="#E5E7EB", width=1)
            draw.text((origin_x + 27, y - 5), str(level), fill="#6B7280")
        for checkpoint, x in enumerate(x_positions, start=1):
            draw.line(
                (x, origin_y + PLOT_TOP, x, origin_y + PLOT_BOTTOM),
                fill="#E5E7EB",
                width=1,
            )
            draw.text((x - 3, origin_y + 193), str(checkpoint), fill="#4B5563")

        y_positions = tuple(_value_y(origin_y, float(value)) for value in row)
        _dashed_line(
            draw,
            (x_positions[0], y_positions[0]),
            (x_positions[2], y_positions[2]),
            fill="#9CA3AF",
        )
        rgb = ImageColor.getrgb(str(color_name))
        for x, y in zip(x_positions, y_positions):
            draw.ellipse(
                (
                    x - MARKER_RADIUS,
                    y - MARKER_RADIUS,
                    x + MARKER_RADIUS,
                    y + MARKER_RADIUS,
                ),
                fill=rgb,
                outline=rgb,
            )
    return image
