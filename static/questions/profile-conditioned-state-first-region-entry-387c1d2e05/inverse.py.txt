"""Independent pixel-only inverse arm for the region-entry world.

Receives only the final PNG. From the raster alone it:
  1. labels pixels against the fixed palette and recovers the shaded band's
     horizontal extent (the region column) from the band-colour pixels;
  2. locates the top of the first panel box from the border pixels and, using
     the declared panel geometry, derives each panel's vertical span;
  3. for every panel, finds the centroid of each uniquely-coloured token and
     decides inside/outside the band;
  4. tracks each token color across the ordered panels to recover its first
     outside-to-inside entry step and returns the color of the earliest
     entrant.

No latent scene, renderer, prompt, or gold code is accessed.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

BG = (255, 255, 255)
BORDER = (95, 95, 95)
BAND = (175, 178, 188)
TOKEN_COLORS = [
    (205, 45, 45),
    (45, 165, 60),
    (55, 85, 215),
    (240, 150, 30),
]
COLOR_NAMES = ["red", "green", "blue", "orange"]

PALETTE = [BG, BORDER, BAND, *TOKEN_COLORS]
_BG, _BRD, _BAND, _T0, _T1, _T2, _T3 = range(7)

NCOL = 4
KPANEL = 4
SLOT_H = 80
STRIP_H = 26


class UnreadableImageError(ValueError):
    """The raster does not expose the expected panel structure."""


def _label(arr: np.ndarray) -> np.ndarray:
    dists = np.stack(
        [np.linalg.norm(arr - np.array(c, dtype=np.float32), axis=-1) for c in PALETTE],
        axis=-1,
    )
    return dists.argmin(axis=-1)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label(arr)

    band_ys, band_xs = np.nonzero(labels == _BAND)
    if len(band_xs) == 0:
        raise UnreadableImageError("no shaded region band found")
    band_min = int(band_xs.min())
    band_max = int(band_xs.max())

    border_ys = np.nonzero(labels == _BRD)[0]
    if len(border_ys) == 0:
        raise UnreadableImageError("no panel borders found")
    top_edge = int(border_ys.min())

    inside = {}
    for c in range(NCOL):
        prev = None
        for k in range(KPANEL):
            top = top_edge + k * (SLOT_H + STRIP_H)
            mask = (labels == (_T0 + c)) & (np.arange(labels.shape[0])[:, None] >= top) & (
                np.arange(labels.shape[0])[:, None] < top + SLOT_H
            )
            ys, xs = np.nonzero(mask)
            if len(xs) == 0:
                raise UnreadableImageError(f"token color {c} missing in panel {k}")
            cx = float(xs.mean())
            is_in = band_min <= cx <= band_max
            if prev is False and is_in:
                inside.setdefault(c, k)
            prev = is_in

    if not inside:
        raise UnreadableImageError("no token entered the region")
    earliest = min(inside, key=lambda c: (inside[c], c))
    return COLOR_NAMES[earliest]
