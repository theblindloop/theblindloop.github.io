"""Earliest outside-to-inside region entry across ordered panels.

A vertical timeline of KPANELS=4 panels reads top to bottom. Every panel shows
the same five slot columns; a shaded vertical band (the "region") always covers
the middle column (slot index REGION=2, the third slot) across every panel.
Four tokens, each with its own fixed unique color, occupy four of the five
slots in each panel; the token colors never change, so identity is carried by
color. Between consecutive panels a token may move to a different slot.

A token is "inside the region" when its center lies within the shaded band. The
decision is the color of the token that makes the FIRST (earliest, in panel
order) outside-to-inside transition into the band: a token that is inside the
band in the first (top) panel never "enters" during the shown sequence. The
answer is one of the four token colors.

The latent scene stores, per panel, the slot occupied by each of the four
colored tokens, and the fixed region slot index.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

NCOL = 4
KPANEL = 4
NSLOT = 5
REGION = 2

LEFT_M = 50
TOP_M = 55
GAP = 18
SLOT_W = 90
SLOT_H = 80
STRIP_H = 26
TOKEN_R = 18

BG = (255, 255, 255)
BORDER = (95, 95, 95)
BAND = (175, 178, 188)
TOKEN_COLORS = [
    (205, 45, 45),
    (45, 165, 60),
    (55, 85, 215),
    (240, 150, 30),
]
COLOR_NAMES = ["red", "green", "blue", "orange"]

WIDTH = LEFT_M + NSLOT * SLOT_W + (NSLOT - 1) * GAP + LEFT_M
HEIGHT = TOP_M + KPANEL * SLOT_H + (KPANEL - 1) * STRIP_H + 40


def _col_center(i: int) -> int:
    return LEFT_M + SLOT_W // 2 + i * (SLOT_W + GAP)


def _col_x0(i: int) -> int:
    return _col_center(i) - SLOT_W // 2


def _panel_top(k: int) -> int:
    return TOP_M + k * (SLOT_H + STRIP_H)


def _first_entries(scene: dict) -> dict[int, int]:
    """color_index -> earliest outside->inside panel step (1..KPANEL-1)."""
    panels = scene["slots"]
    region = scene["region"]
    entries: dict[int, int] = {}
    for c in range(NCOL):
        prev_inside = panels[0][c] == region
        for k in range(1, KPANEL):
            inside = panels[k][c] == region
            if inside and not prev_inside:
                entries[c] = k
                break
            prev_inside = inside
    return entries


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 1009 + 17)
    while True:
        panels = []
        for _k in range(KPANEL):
            panels.append(rng.sample(range(NSLOT), NCOL))
        scene = {"slots": panels, "region": REGION}
        entries = _first_entries(scene)
        if len(entries) < 2:
            continue
        by_step: dict[int, int] = {}
        for step in entries.values():
            by_step[step] = by_step.get(step, 0) + 1
        if by_step[min(entries.values())] != 1:
            continue
        return scene


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)
    panels = scene["slots"]
    region = scene["region"]

    for k in range(KPANEL):
        top = _panel_top(k)
        bx0 = _col_x0(region)
        draw.rectangle([bx0, top, bx0 + SLOT_W, top + SLOT_H], fill=BAND)
        for c in range(NCOL):
            slot = panels[k][c]
            cx = _col_center(slot)
            cy = top + SLOT_H // 2
            draw.ellipse(
                [cx - TOKEN_R, cy - TOKEN_R, cx + TOKEN_R, cy + TOKEN_R],
                fill=TOKEN_COLORS[c],
            )
        left = _col_x0(0)
        right = _col_x0(NSLOT - 1) + SLOT_W
        draw.rectangle([left, top, right, top + SLOT_H], outline=BORDER, width=3)
    return img


def analytic_gold(scene: dict) -> str:
    entries = _first_entries(scene)
    earliest = min(entries, key=lambda c: (entries[c], c))
    return COLOR_NAMES[earliest]


def margin(scene: dict) -> float:
    entries = _first_entries(scene)
    steps = sorted(entries.values())
    if len(steps) >= 2:
        return float(steps[1] - steps[0])
    return 10.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    """No nontrivial render-invariant latent symmetry exists.

    Every latent field directly changes the raster: ``slots`` fixes the exact
    drawn position (slot column) of every uniquely-colored token in every panel
    and ``region`` fixes which column the shaded band occupies. Because token
    identity is carried by a fixed unique color and the band is drawn at a fixed
    column, any perturbation of ``slots`` or ``region`` changes the pixels and
    any permutation of the token colors changes the raster. There is therefore
    no reparametrization yielding pixel-identical renders with a different gold,
    so no alias symmetry is declared.
    """
    return []
