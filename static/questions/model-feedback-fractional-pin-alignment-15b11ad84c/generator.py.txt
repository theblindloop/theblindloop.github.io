"""Renderer and analytic arm for a paired consecutive-gap ratio relation.

Every panel contains one red and one blue row.  Each row has three large pins,
so its two visible gaps are unambiguous.  The question asks for the panel in
which (left gap / right gap) is the same for the two rows.  Row origins and
scales vary independently; only the ratio is informative.
"""

from __future__ import annotations

import random
from fractions import Fraction
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1400
HEIGHT = 820

BACKGROUND = (247, 249, 252)
PANEL = (255, 255, 255)
FRAME = (102, 116, 134)
INK = (34, 43, 56)
TRACK = (185, 195, 207)
RED = (208, 60, 60)
BLUE = (49, 101, 194)

PANEL_LABELS = ("panel-A", "panel-B", "panel-C", "panel-D")
PANEL_BOXES = (
    (35, 100, 685, 410),
    (715, 100, 1365, 410),
    (35, 440, 685, 750),
    (715, 440, 1365, 750),
)

RED_Y_OFFSET = 145
BLUE_Y_OFFSET = 235
TRACK_MARGIN = 48
TRACK_LEFT_LIMIT = 62
TRACK_RIGHT_LIMIT = 588
MIN_GAP = 54
MAX_GAP = 170
MARKER_RADIUS = 17

TARGET_TOLERANCE = 0.035
DECOY_CLEARANCE = 0.16
TARGET_HEADROOM = 0.01

_RED_POINTS_KEY = "red_points"
_BLUE_POINTS_KEY = "blue_points"

# Each pair is the integer proportion of left gap to right gap.  The shared
# ratio template is scaled independently in the red and blue rows.
RATIO_TEMPLATES: tuple[tuple[int, int], ...] = (
    (2, 3),
    (3, 2),
    (3, 4),
    (4, 3),
    (2, 5),
    (5, 2),
    (4, 5),
    (5, 4),
    (3, 5),
    (5, 3),
)


def _font(size: int) -> ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _ratio_from_points(points: Iterable[int]) -> float:
    values = [int(value) for value in points]
    if len(values) != 3 or not (values[0] < values[1] < values[2]):
        raise ValueError("a row must contain three strictly ordered pin centers")
    left_gap = values[1] - values[0]
    right_gap = values[2] - values[1]
    if left_gap <= 0 or right_gap <= 0:
        raise ValueError("row gaps must be positive")
    return left_gap / right_gap


def _normalized_residual(panel: dict) -> float:
    red_ratio = _ratio_from_points(panel.get(_RED_POINTS_KEY))
    blue_ratio = _ratio_from_points(panel.get(_BLUE_POINTS_KEY))
    return (red_ratio - blue_ratio) / max(red_ratio, blue_ratio)


def _track_points(
    rng: random.Random,
    ratio_template: tuple[int, int],
) -> tuple[list[int], list[int]]:
    left_units, right_units = ratio_template
    for _ in range(5000):
        scale = rng.randint(28, 34)
        left_gap = left_units * scale
        right_gap = right_units * scale
        total = left_gap + right_gap
        if not (MIN_GAP <= left_gap <= MAX_GAP and MIN_GAP <= right_gap <= MAX_GAP):
            continue
        first = rng.randint(
            TRACK_LEFT_LIMIT + TRACK_MARGIN,
            TRACK_RIGHT_LIMIT - TRACK_MARGIN - total,
        )
        points = [first, first + left_gap, first + total]
        return points, [first - TRACK_MARGIN, first + total + TRACK_MARGIN]
    raise RuntimeError("could not sample a readable three-pin row")


def _panel_for_templates(
    rng: random.Random,
    red_template: tuple[int, int],
    blue_template: tuple[int, int],
) -> dict:
    red_points, red_track = _track_points(rng, red_template)
    blue_points, blue_track = _track_points(rng, blue_template)
    return {
        "red_points": red_points,
        "blue_points": blue_points,
        "red_track": red_track,
        "blue_track": blue_track,
    }


def _ratio_value(template: tuple[int, int]) -> Fraction:
    return Fraction(template[0], template[1])


def _decoy_template_pair(rng: random.Random) -> tuple[tuple[int, int], tuple[int, int]]:
    for _ in range(5000):
        red_template = rng.choice(RATIO_TEMPLATES)
        blue_template = rng.choice(RATIO_TEMPLATES)
        if red_template == blue_template:
            continue
        red_ratio = float(_ratio_value(red_template))
        blue_ratio = float(_ratio_value(blue_template))
        residual = abs(red_ratio - blue_ratio) / max(red_ratio, blue_ratio)
        if residual >= DECOY_CLEARANCE:
            return red_template, blue_template
    raise RuntimeError("could not sample a separated ratio decoy")


def sample_scene(seed: int) -> dict:
    """Sample four panels with one matching pair of scale-free ratios."""
    rng = random.Random(int(seed))
    target_index = rng.randrange(len(PANEL_LABELS))
    target_template = rng.choice(RATIO_TEMPLATES)
    panels: list[dict] = []
    for index in range(len(PANEL_LABELS)):
        if index == target_index:
            panels.append(_panel_for_templates(rng, target_template, target_template))
        else:
            red_template, blue_template = _decoy_template_pair(rng)
            panels.append(_panel_for_templates(rng, red_template, blue_template))
    scene = {"panels": panels}
    if is_quarantined(scene):
        raise AssertionError("sampler produced a quarantined ratio scene")
    return scene


def _matches(panels: Iterable[dict]) -> list[int]:
    return [
        index
        for index, panel in enumerate(panels)
        if abs(_normalized_residual(panel)) <= TARGET_TOLERANCE
    ]


def render(scene: dict) -> Image.Image:
    """Render four panels with two labeled rows of three colored pins."""
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(27)
    legend_font = _font(18)
    panel_font = _font(29)
    row_font = _font(18)

    draw.text((40, 18), "Which paired spacing ratio matches?", fill=INK, font=title_font)
    draw.text(
        (40, 59),
        "For each row, compare left gap ÷ right gap; find the panel where red and blue agree.",
        fill=FRAME,
        font=legend_font,
    )

    for index, (box, panel) in enumerate(zip(PANEL_BOXES, scene["panels"])):
        x0, y0, x1, y1 = box
        draw.rounded_rectangle(box, radius=18, fill=PANEL, outline=FRAME, width=3)
        draw.text((x0 + 25, y0 + 20), PANEL_LABELS[index][-1], fill=INK, font=panel_font)

        for row_name, key, color, y_offset in (
            ("R", "red_points", RED, RED_Y_OFFSET),
            ("B", "blue_points", BLUE, BLUE_Y_OFFSET),
        ):
            y = y0 + y_offset
            points = [int(value) for value in panel[key]]
            track_key = "red_track" if key == "red_points" else "blue_track"
            track_start, track_end = [int(value) for value in panel[track_key]]
            draw.text((x0 + 24, y - 12), row_name, fill=INK, font=row_font)
            draw.line(
                (x0 + track_start, y, x0 + track_end, y),
                fill=TRACK,
                width=12,
            )
            for endpoint in (track_start, track_end):
                draw.line(
                    (x0 + endpoint, y - 14, x0 + endpoint, y + 14),
                    fill=FRAME,
                    width=2,
                )
            for point in points:
                center_x = x0 + point
                draw.ellipse(
                    (
                        center_x - MARKER_RADIUS,
                        y - MARKER_RADIUS,
                        center_x + MARKER_RADIUS,
                        y + MARKER_RADIUS,
                    ),
                    fill=color,
                    outline=INK,
                    width=3,
                )

    return image


def analytic_gold(scene: dict) -> str:
    """Return the unique panel whose two consecutive-gap ratios agree."""
    matches = _matches(scene["panels"])
    if len(matches) != 1:
        raise ValueError(f"expected one matching ratio panel, found {matches}")
    return PANEL_LABELS[matches[0]]


def margin(scene: dict) -> float:
    """Return separation between the target band and the nearest decoy."""
    try:
        residuals = sorted(abs(_normalized_residual(panel)) for panel in scene["panels"])
    except (KeyError, TypeError, ValueError, ZeroDivisionError):
        return -1.0
    if len(residuals) != len(PANEL_LABELS):
        return -1.0
    return min(TARGET_TOLERANCE - residuals[0], residuals[1] - DECOY_CLEARANCE)


def is_quarantined(scene: dict) -> bool:
    """Exclude multiple matches, close decoys, and malformed rows."""
    try:
        residuals = sorted(abs(_normalized_residual(panel)) for panel in scene["panels"])
        for panel in scene["panels"]:
            for key in ("red_points", "blue_points"):
                points = [int(value) for value in panel[key]]
                if len(points) != 3 or any(b - a < MIN_GAP for a, b in zip(points, points[1:])):
                    return True
    except (KeyError, TypeError, ValueError, ZeroDivisionError):
        return True
    return not (
        len(residuals) == len(PANEL_LABELS)
        and residuals[0] <= TARGET_HEADROOM
        and residuals[1] >= DECOY_CLEARANCE
        and len(_matches(scene["panels"])) == 1
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases are declared; all label inputs are visibly rendered."""
    return []
