"""Independent image-only inverse for paired consecutive-gap ratios.

Only exact red and blue marker pixels are used.  The oracle does not import
renderer code and does not read latent track coordinates.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

WIDTH = 1400
HEIGHT = 820
PANEL_BOXES = (
    (35, 100, 685, 410),
    (715, 100, 1365, 410),
    (35, 440, 685, 750),
    (715, 440, 1365, 750),
)
PANEL_LABELS = ("panel-A", "panel-B", "panel-C", "panel-D")
RED = (208, 60, 60)
BLUE = (49, 101, 194)
RED_Y_OFFSET = 145
BLUE_Y_OFFSET = 235
TARGET_TOLERANCE = 0.035
MIN_MARKER_PIXELS = 400


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Return 4-connected component coordinates for a boolean crop mask."""
    visited = np.zeros(mask.shape, dtype=bool)
    height, width = mask.shape
    result: list[np.ndarray] = []
    for y, x in zip(*np.nonzero(mask)):
        if visited[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        visited[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    queue.append((ny, nx))
        result.append(np.asarray(points, dtype=np.int32))
    return result


def _point_centers(
    arr: np.ndarray,
    box: tuple[int, int, int, int],
    color: tuple[int, int, int],
    expected_y_offset: int,
) -> list[float]:
    x0, y0, x1, y1 = box
    crop = arr[y0 : y1 + 1, x0 : x1 + 1]
    mask = np.all(crop == np.asarray(color, dtype=np.uint8), axis=2)
    components = [component for component in _components(mask) if len(component) >= MIN_MARKER_PIXELS]
    if len(components) != 3:
        raise ValueError("each colored row must contain exactly three visible pins")
    centers: list[tuple[float, float]] = []
    for points in components:
        ys = points[:, 0]
        xs = points[:, 1]
        center_x = float(xs.mean()) + x0
        center_y = float(ys.mean()) + y0
        if abs(center_y - (y0 + expected_y_offset)) > 2.0:
            raise ValueError("pin is not on its expected row")
        centers.append((center_x, center_y))
    centers.sort()
    xs = [center_x for center_x, _ in centers]
    if any(right - left < 20 for left, right in zip(xs, xs[1:])):
        raise ValueError("adjacent pins are not separated")
    return xs


def _ratio(points: list[float]) -> float:
    left_gap = points[1] - points[0]
    right_gap = points[2] - points[1]
    if left_gap <= 0 or right_gap <= 0:
        raise ValueError("row gap is not positive")
    return left_gap / right_gap


def _residual(arr: np.ndarray, box: tuple[int, int, int, int]) -> float:
    red_points = _point_centers(arr, box, RED, RED_Y_OFFSET)
    blue_points = _point_centers(arr, box, BLUE, BLUE_Y_OFFSET)
    red_ratio = _ratio(red_points)
    blue_ratio = _ratio(blue_points)
    return (red_ratio - blue_ratio) / max(red_ratio, blue_ratio)


def decision_from_image(image: Image.Image) -> str:
    """Measure both colored gap ratios in every panel and select the match."""
    rgb = image.convert("RGB")
    if rgb.size != (WIDTH, HEIGHT):
        raise ValueError("unexpected image size")
    arr = np.asarray(rgb, dtype=np.uint8)
    residuals = [_residual(arr, box) for box in PANEL_BOXES]
    matches = [index for index, residual in enumerate(residuals) if abs(residual) <= TARGET_TOLERANCE]
    if len(matches) != 1:
        raise ValueError(f"expected one matching ratio panel, found {matches}")
    return PANEL_LABELS[matches[0]]
