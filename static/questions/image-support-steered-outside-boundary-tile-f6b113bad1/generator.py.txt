"""Deterministic raster world for bounded-versus-unbounded boundary incidence."""

from __future__ import annotations

import copy

import numpy as np
from PIL import Image


CANVAS_SIZE = 512
BACKGROUND = [248, 250, 252]
SHEET_COLORS = ([13, 116, 110], [12, 105, 135])
AMBER = [245, 158, 11]
PURPLE = [147, 51, 234]

SHEET_BOX = [24, 24, 488, 488]
NOTCH_BOX = [24, 216, 192, 296]
OUTSIDE_PATCH_BOX = [192, 228, 212, 284]
INSIDE_PATCH_BOX = [308, 96, 328, 152]
HOLE_BOXES = [
    [328, 80, 432, 164],
    [84, 84, 200, 172],
    [88, 344, 204, 432],
    [320, 336, 432, 432],
]


def sample_scene(seed: int) -> dict:
    """Return a paired, balanced scene with wide raster clearances."""

    value = int(seed)
    return {
        "canvas_size": CANVAS_SIZE,
        "outside_color": "amber" if value % 2 == 0 else "purple",
        "transform": (value // 2) % 8,
        "palette_variant": (value // 16) % len(SHEET_COLORS),
        "sheet_box": list(SHEET_BOX),
        "notch_box": list(NOTCH_BOX),
        "outside_patch_box": list(OUTSIDE_PATCH_BOX),
        "inside_patch_box": list(INSIDE_PATCH_BOX),
        "hole_boxes": copy.deepcopy(HOLE_BOXES),
        "min_clearance": 20.0,
    }


def _paint_box(array: np.ndarray, box: list[int], color: list[int]) -> None:
    x0, y0, x1, y1 = (int(value) for value in box)
    array[y0:y1, x0:x1] = np.asarray(color, dtype=np.uint8)


def _transform(array: np.ndarray, transform: int) -> np.ndarray:
    code = int(transform) % 8
    rotated = np.rot90(array, code % 4)
    if code >= 4:
        rotated = np.fliplr(rotated)
    return np.ascontiguousarray(rotated)


def render(scene: dict) -> Image.Image:
    """Render only public scene fields; no answer metadata is hidden."""

    size = int(scene["canvas_size"])
    if size != CANVAS_SIZE:
        raise ValueError("outside_boundary_tile requires a 512x512 canvas")
    palette_variant = int(scene["palette_variant"]) % len(SHEET_COLORS)
    array = np.empty((size, size, 3), dtype=np.uint8)
    array[:] = np.asarray(BACKGROUND, dtype=np.uint8)

    _paint_box(array, scene["sheet_box"], list(SHEET_COLORS[palette_variant]))
    _paint_box(array, scene["notch_box"], BACKGROUND)
    for box in scene["hole_boxes"]:
        _paint_box(array, box, BACKGROUND)

    outside_color = str(scene["outside_color"])
    if outside_color not in {"amber", "purple"}:
        raise ValueError("outside_color must be amber or purple")
    inside_color = "purple" if outside_color == "amber" else "amber"
    colors = {"amber": AMBER, "purple": PURPLE}
    _paint_box(array, scene["outside_patch_box"], colors[outside_color])
    _paint_box(array, scene["inside_patch_box"], colors[inside_color])
    return Image.fromarray(_transform(array, int(scene["transform"])), mode="RGB")


def analytic_gold(scene: dict) -> str:
    return str(scene["outside_color"])


def margin(scene: dict) -> float:
    return float(scene["min_clearance"])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 16.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["hole_boxes"] = list(reversed(twin["hole_boxes"]))
    return [("reverse_disjoint_pocket_draw_order", twin)]
