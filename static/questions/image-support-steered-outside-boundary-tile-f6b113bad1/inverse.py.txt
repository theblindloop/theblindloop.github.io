"""Independent PNG-only inverse for colored boundary-component incidence."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


_NAMES = ("background", "sheet_a", "sheet_b", "amber", "purple")
_COLORS = np.asarray(
    [
        [248, 250, 252],
        [13, 116, 110],
        [12, 105, 135],
        [245, 158, 11],
        [147, 51, 234],
    ],
    dtype=np.int32,
)
_INDEX = {name: index for index, name in enumerate(_NAMES)}
_FACTOR = 4


def _nearest_labels(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int32)
    if array.shape != (512, 512, 3):
        raise ValueError(f"expected a 512x512 RGB raster, got {array.shape}")
    distances = ((array[:, :, None, :] - _COLORS[None, None, :, :]) ** 2).sum(axis=3)
    labels = distances.argmin(axis=2).astype(np.uint8)
    residual = np.sqrt(distances.min(axis=2))
    if float(np.quantile(residual, 0.999)) > 36.0:
        raise ValueError("raster contains material colors outside the public palette")
    return labels


def _block_fraction(mask: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    blocks = mask.reshape(height // _FACTOR, _FACTOR, width // _FACTOR, _FACTOR)
    return blocks.mean(axis=(1, 3))


def _components(mask: np.ndarray) -> tuple[np.ndarray, dict[int, int]]:
    height, width = mask.shape
    labels = np.zeros((height, width), dtype=np.int32)
    areas: dict[int, int] = {}
    component = 0
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if labels[y, x]:
            continue
        component += 1
        labels[y, x] = component
        queue = deque([(y, x)])
        area = 0
        while queue:
            cy, cx = queue.popleft()
            area += 1
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if (
                    0 <= ny < height
                    and 0 <= nx < width
                    and mask[ny, nx]
                    and labels[ny, nx] == 0
                ):
                    labels[ny, nx] = component
                    queue.append((ny, nx))
        areas[component] = area
    return labels, areas


def _border_ids(labels: np.ndarray) -> set[int]:
    values = set(int(value) for value in labels[0, :])
    values.update(int(value) for value in labels[-1, :])
    values.update(int(value) for value in labels[:, 0])
    values.update(int(value) for value in labels[:, -1])
    values.discard(0)
    return values


def _adjacent_white_components(
    color_blocks: np.ndarray, solid: np.ndarray, white_labels: np.ndarray
) -> set[int]:
    if not 60 <= int(color_blocks.sum()) <= 80:
        raise ValueError("a colored boundary tile is missing, resized, or materially erased")
    components: set[int] = set()
    height, width = solid.shape
    for raw_y, raw_x in np.argwhere(color_blocks):
        y, x = int(raw_y), int(raw_x)
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < height and 0 <= nx < width and not solid[ny, nx]:
                label = int(white_labels[ny, nx])
                if label:
                    components.add(label)
    if len(components) != 1:
        raise ValueError("a colored tile does not border exactly one white component")
    return components


def decision_from_image(image: Image.Image) -> str:
    """Return the color incident to the border-connected white component."""

    labels = _nearest_labels(image)
    solid_pixels = labels != _INDEX["background"]
    solid = _block_fraction(solid_pixels) >= 0.5
    solid_labels, solid_areas = _components(solid)
    substantial = [area for area in solid_areas.values() if area >= 40]
    if len(substantial) != 1 or not 10050 <= substantial[0] <= 10200:
        raise ValueError("the broad fixed-ink solid union is missing, split, or corrupted")
    points = np.argwhere(solid)
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    if x0 > 7 or y0 > 7 or x1 < 120 or y1 < 120:
        raise ValueError("the solid sheet no longer spans the raster")

    white_labels, white_areas = _components(~solid)
    border = _border_ids(white_labels)
    if len(border) != 1:
        raise ValueError("the outside white background is not a unique border component")
    holes = {
        component: area
        for component, area in white_areas.items()
        if component not in border and area >= 40
    }
    if len(holes) != 4:
        raise ValueError(f"expected four enclosed white pockets, recovered {len(holes)}")
    if any(not 500 <= area <= 710 for area in holes.values()):
        raise ValueError("an enclosed pocket has an unsafe or corrupted area")

    incidences: dict[str, set[int]] = {}
    for name in ("amber", "purple"):
        color_blocks = _block_fraction(labels == _INDEX[name]) >= 0.5
        incidences[name] = _adjacent_white_components(color_blocks, solid, white_labels)
    outside = [name for name, adjacent in incidences.items() if not adjacent.isdisjoint(border)]
    bounded = [name for name, adjacent in incidences.items() if not adjacent.isdisjoint(holes)]
    if len(outside) != 1 or len(bounded) != 1 or outside[0] == bounded[0]:
        raise ValueError("pixels do not expose one exterior tile and one pocket tile")
    return outside[0]
