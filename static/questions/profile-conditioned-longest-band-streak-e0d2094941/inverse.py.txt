"""Independent PNG-only inverse arm for longest_band_streak.

The module intentionally duplicates the public palette constants and implements
its own connected-component and longest-run logic.  It imports no candidate
renderer, scene, prompt, generator, verifier, or analytic-gold code.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_MARKERS = ((214, 95, 0), (24, 122, 82), (135, 63, 80))
_BANDS = (
    (185, 220, 242), (52, 126, 172),
    (217, 200, 232), (128, 96, 157),
    (195, 228, 213), (75, 144, 117),
)
_LABELS = ("A", "B", "C", "D")


def _palette_mask(array: np.ndarray, colors) -> np.ndarray:
    mask = np.zeros(array.shape[:2], dtype=bool)
    for color in colors:
        mask |= np.all(array == np.asarray(color, dtype=np.uint8), axis=2)
    return mask


def _components(mask: np.ndarray) -> list[dict]:
    remaining = {tuple(value) for value in np.argwhere(mask)}
    components = []
    while remaining:
        start = remaining.pop()
        queue = [start]
        points = [start]
        while queue:
            y, x = queue.pop()
            for ny in (y - 1, y, y + 1):
                for nx in (x - 1, x, x + 1):
                    if ny == y and nx == x:
                        continue
                    key = (ny, nx)
                    if key in remaining:
                        remaining.remove(key)
                        queue.append(key)
                        points.append(key)
        ys = np.fromiter((p[0] for p in points), dtype=float)
        xs = np.fromiter((p[1] for p in points), dtype=float)
        components.append(
            {
                "size": len(points),
                "x": float(xs.mean()),
                "y": float(ys.mean()),
                "minx": int(xs.min()),
                "maxx": int(xs.max()),
                "miny": int(ys.min()),
                "maxy": int(ys.max()),
            }
        )
    return components


def _largest_gap_threshold(values: list[float]) -> float:
    ordered = sorted(values)
    if len(ordered) < 2:
        raise ValueError("not enough components to infer panel split")
    gaps = [(ordered[i + 1] - ordered[i], i) for i in range(len(ordered) - 1)]
    gap, index = max(gaps)
    if gap < 45:
        raise ValueError("panel groups are not visibly separated")
    return (ordered[index] + ordered[index + 1]) / 2.0


def _longest_run(bits: list[bool]) -> int:
    best = current = 0
    for value in bits:
        current = current + 1 if value else 0
        best = max(best, current)
    return best


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    marker_components = [
        item
        for item in _components(_palette_mask(array, _MARKERS))
        if 100 <= item["size"] <= 260
        and 12 <= item["maxx"] - item["minx"] + 1 <= 18
        and 12 <= item["maxy"] - item["miny"] + 1 <= 18
    ]
    band_components = [
        item
        for item in _components(_palette_mask(array, _BANDS))
        if item["size"] >= 280
        and 20 <= item["maxx"] - item["minx"] + 1 <= 28
        and item["maxy"] - item["miny"] + 1 >= 25
    ]
    if len(marker_components) != 28:
        raise ValueError(f"expected 28 marker components, found {len(marker_components)}")
    if len(band_components) != 28:
        raise ValueError(f"expected 28 band components, found {len(band_components)}")

    x_split = _largest_gap_threshold([item["x"] for item in marker_components])
    y_split = _largest_gap_threshold([item["y"] for item in marker_components])

    scores = []
    for row, column in ((0, 0), (0, 1), (1, 0), (1, 1)):
        markers = [
            item for item in marker_components
            if (item["y"] >= y_split) == bool(row) and (item["x"] >= x_split) == bool(column)
        ]
        bands = [
            item for item in band_components
            if (item["y"] >= y_split) == bool(row) and (item["x"] >= x_split) == bool(column)
        ]
        markers.sort(key=lambda item: item["x"])
        bands.sort(key=lambda item: item["x"])
        if len(markers) != 7 or len(bands) != 7:
            raise ValueError("each inferred panel must contain seven markers and seven bands")
        if any(abs(marker["x"] - band["x"]) > 4.0 for marker, band in zip(markers, bands)):
            raise ValueError("marker-to-band checkpoint alignment is corrupted")
        membership = [
            band["miny"] <= marker["y"] <= band["maxy"]
            for marker, band in zip(markers, bands)
        ]
        scores.append(_longest_run(membership))
    best = max(scores)
    if scores.count(best) != 1:
        raise ValueError(f"pixel decision is tied: {scores}")
    return _LABELS[scores.index(best)]
