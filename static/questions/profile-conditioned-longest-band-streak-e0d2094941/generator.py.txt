"""Deterministic renderer and analytic arm for longest_band_streak."""

from __future__ import annotations

import itertools
import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH = 900
HEIGHT = 700
PANEL_LABELS = ("A", "B", "C", "D")
THEMES = (
    {"band": "#b9dcf2", "edge": "#347eac", "marker": "#d65f00"},
    {"band": "#d9c8e8", "edge": "#80609d", "marker": "#187a52"},
    {"band": "#c3e4d5", "edge": "#4b9075", "marker": "#873f50"},
)


def _font(size: int, bold: bool = False):
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{name}", size)
    except OSError:
        return ImageFont.load_default()


def _longest_run(bits) -> int:
    best = current = 0
    for value in bits:
        current = current + 1 if value else 0
        best = max(best, current)
    return best


def _patterns(run_length: int) -> list[tuple[int, ...]]:
    values = []
    for bits in itertools.product((0, 1), repeat=7):
        if sum(bits) == 4 and _longest_run(bits) == run_length:
            values.append(bits)
    return values


PATTERNS = {run: _patterns(run) for run in (1, 2, 3, 4)}


def sample_scene(seed: int) -> dict:
    """Sample one balanced scene; the answer label cycles uniformly with seed."""
    rng = random.Random(int(seed))
    winner = int(seed) % 4
    loser_scores = [1, 2, 3]
    rng.shuffle(loser_scores)
    scores = []
    cursor = 0
    for panel_index in range(4):
        if panel_index == winner:
            scores.append(4)
        else:
            scores.append(loser_scores[cursor])
            cursor += 1

    panels = []
    near_checkpoint = int(seed) % 7
    near_boundary = int(seed) % 6 == 0
    for panel_index, desired_run in enumerate(scores):
        pattern = list(rng.choice(PATTERNS[desired_run]))
        centers = []
        center = rng.randint(43, 57)
        for _ in range(7):
            center = max(35, min(65, center + rng.randint(-8, 8)))
            centers.append(float(center))
        halves = [float(rng.randint(11, 15)) for _ in range(7)]
        outside_indices = [i for i, inside in enumerate(pattern) if not inside]
        outside_signs = [1, 1, -1]
        rng.shuffle(outside_signs)
        sign_for = dict(zip(outside_indices, outside_signs))
        markers = []
        for checkpoint, inside in enumerate(pattern):
            c = centers[checkpoint]
            h = halves[checkpoint]
            if inside:
                if near_boundary and panel_index == ((winner + 1) % 4) and checkpoint == near_checkpoint:
                    sign = -1 if rng.random() < 0.5 else 1
                    value = c + sign * (h - 4.0)
                else:
                    span = max(0, int(h - 7.0))
                    value = c + float(rng.randint(-span, span))
            else:
                value = c + sign_for[checkpoint] * (h + float(rng.randint(8, 13)))
            markers.append(value)
        panels.append(
            {
                "band_centers": centers,
                "band_halves": halves,
                "markers": markers,
            }
        )

    return {
        "panels": panels,
        "theme": (int(seed) // 4) % len(THEMES),
        "dx": rng.randint(-12, 12),
        "dy": rng.randint(-8, 8),
        "width": WIDTH,
        "height": HEIGHT,
    }


def analytic_gold(scene: dict) -> str:
    scores = []
    for panel in scene["panels"]:
        bits = [
            abs(marker - center) <= half + 1e-9
            for marker, center, half in zip(
                panel["markers"], panel["band_centers"], panel["band_halves"]
            )
        ]
        best = current = 0
        for value in bits:
            current = current + 1 if value else 0
            best = max(best, current)
        scores.append(best)
    return PANEL_LABELS[max(range(4), key=lambda index: scores[index])]


def _scores_and_clearance(scene: dict) -> tuple[list[int], float]:
    scores = []
    clearance = float("inf")
    for panel in scene["panels"]:
        bits = []
        for marker, center, half in zip(
            panel["markers"], panel["band_centers"], panel["band_halves"]
        ):
            bits.append(abs(marker - center) <= half + 1e-9)
            clearance = min(clearance, abs(abs(marker - center) - half))
        scores.append(_longest_run(bits))
    return scores, clearance


def margin(scene: dict) -> float:
    scores, _ = _scores_and_clearance(scene)
    ordered = sorted(scores, reverse=True)
    return float(ordered[0] - ordered[1])


def is_quarantined(scene: dict) -> bool:
    scores, clearance = _scores_and_clearance(scene)
    return scores.count(max(scores)) != 1 or clearance < 2.5


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def translated(scene: dict, dx: int, dy: int) -> dict:
    twin = deepcopy(scene)
    twin["dx"] = int(dx)
    twin["dy"] = int(dy)
    return twin


def recolored(scene: dict, theme: int) -> dict:
    twin = deepcopy(scene)
    twin["theme"] = int(theme) % len(THEMES)
    return twin


def _value_y(origin_y: int, value: float) -> int:
    return int(round(origin_y + 205 - 1.48 * value))


def _draw_panel(draw, panel: dict, origin: tuple[int, int], label: str, theme: dict) -> None:
    ox, oy = origin
    plot_left = ox + 47
    xs = [plot_left + 45 * i for i in range(7)]
    plot_top = oy + 48
    plot_bottom = oy + 205
    draw.rounded_rectangle(
        (ox, oy, ox + 365, oy + 250), radius=8, fill="#fbfcfd", outline="#aab2ba", width=2
    )
    draw.text((ox + 12, oy + 9), label, fill="#17212b", font=_font(24, True))
    for grid_value in (25, 50, 75):
        gy = _value_y(oy, grid_value)
        draw.line((plot_left - 14, gy, plot_left + 284, gy), fill="#e5e9ed", width=1)
    draw.rectangle(
        (plot_left - 14, plot_top, plot_left + 284, plot_bottom), outline="#7f8993", width=2
    )

    marker_points = [(x, _value_y(oy, value)) for x, value in zip(xs, panel["markers"])]
    draw.line(marker_points, fill="#59636d", width=3, joint="curve")
    for checkpoint, (x, center, half) in enumerate(
        zip(xs, panel["band_centers"], panel["band_halves"]), start=1
    ):
        y_top = _value_y(oy, center + half)
        y_bottom = _value_y(oy, center - half)
        draw.rectangle(
            (x - 12, y_top, x + 12, y_bottom), fill=theme["band"], outline=theme["edge"], width=2
        )
        text = str(checkpoint)
        bbox = draw.textbbox((0, 0), text, font=_font(13))
        draw.text((x - (bbox[2] - bbox[0]) / 2, oy + 218), text, fill="#37414a", font=_font(13))
    for x, y in marker_points:
        draw.ellipse((x - 7, y - 7, x + 7, y + 7), fill=theme["marker"], outline="#ffffff", width=1)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), "#ffffff")
    draw = ImageDraw.Draw(image)
    theme = THEMES[int(scene["theme"]) % len(THEMES)]
    dx = int(scene["dx"])
    dy = int(scene["dy"])
    draw.text(
        (225 + dx, 18 + dy),
        "CHECKPOINT BAND STREAKS",
        fill="#17212b",
        font=_font(26, True),
    )
    origins = ((62 + dx, 82 + dy), (474 + dx, 82 + dy), (62 + dx, 375 + dy), (474 + dx, 375 + dy))
    for panel, origin, label in zip(scene["panels"], origins, PANEL_LABELS):
        _draw_panel(draw, panel, origin, label, theme)
    return image
