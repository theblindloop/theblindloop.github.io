"""Pixel-only inverse arm for the patch-bay world.

Input: the final PNG only. The public question contract supplies the six palette
colours, the magenta frame, the dark sockets and the grey cables. No renderer
code, latent scene, cable list, coordinate table, or analytic gold is read.

Method: segment the cable ink, then trace the cable that leaves the framed
socket. Tracing is a flood over cable ink extended by straight run-continuation
jumps: at a run end the trace continues in the same row or column across a short
white break (an under-crossing), which is the only way a cable is ever
interrupted. Finally report the palette colour of the single pad the traced cable
reaches. Anything ambiguous raises, i.e. the arm abstains.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

CABLE_RGB = (110, 110, 110)
FRAME_RGB = (255, 0, 255)
CONTRACT_PALETTE = {
    "red": (220, 40, 40),
    "orange": (240, 150, 20),
    "purple": (140, 60, 190),
    "blue": (40, 70, 210),
    "green": (30, 150, 60),
    "teal": (20, 170, 170),
}
TOLERANCE = 30
MIN_CABLE_PIXELS = 200
MIN_PAD_PIXELS = 120
MAX_BREAK = 19  # widest readable under-crossing break, in pixels
RUN_EVIDENCE = 6  # a jump may only continue a straight run at least this long
DIRECTIONS = ((0, 1), (0, -1), (1, 0), (-1, 0))


class Abstain(RuntimeError):
    """Raised when the picture does not present one readable traced cable."""


def _near(array: np.ndarray, rgb: tuple[int, int, int], tol: int = TOLERANCE) -> np.ndarray:
    diff = np.abs(array.astype(np.int16) - np.array(rgb, dtype=np.int16))
    return diff.max(axis=2) <= tol


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    seen = np.zeros(mask.shape, dtype=bool)
    out = []
    ys, xs = np.nonzero(mask)
    for sy, sx in zip(ys.tolist(), xs.tolist()):
        if seen[sy, sx]:
            continue
        stack = deque([(sy, sx)])
        seen[sy, sx] = True
        blob = []
        while stack:
            y, x = stack.pop()
            blob.append((y, x))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < mask.shape[0] and 0 <= nx < mask.shape[1]:
                        if mask[ny, nx] and not seen[ny, nx]:
                            seen[ny, nx] = True
                            stack.append((ny, nx))
        out.append(blob)
    return out


def _dilated_bbox(blob: list[tuple[int, int]], pad: int, shape) -> tuple[int, int, int, int]:
    ys = [p[0] for p in blob]
    xs = [p[1] for p in blob]
    return (
        max(0, min(ys) - pad),
        min(shape[0] - 1, max(ys) + pad),
        max(0, min(xs) - pad),
        min(shape[1] - 1, max(xs) + pad),
    )


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3:
        raise Abstain("not an RGB raster")

    cable = _near(array, CABLE_RGB)
    if int(cable.sum()) < MIN_CABLE_PIXELS:
        raise Abstain("no cable ink")

    frame = _near(array, FRAME_RGB)
    if int(frame.sum()) < 40:
        raise Abstain("no framed socket")
    frame_blobs = [b for b in _components(frame) if len(b) >= 40]
    if len(frame_blobs) != 1:
        raise Abstain(f"expected exactly one frame, found {len(frame_blobs)}")
    fy0, fy1, fx0, fx1 = _dilated_bbox(frame_blobs[0], 0, array.shape)
    cy, cx = (fy0 + fy1) // 2, (fx0 + fx1) // 2

    # Cable ink that enters the framed socket's immediate neighbourhood.
    reach = 4 + (fy1 - fy0) // 2
    y0, y1 = max(0, cy - reach), min(array.shape[0] - 1, cy + reach)
    x0, x1 = max(0, cx - reach), min(array.shape[1] - 1, cx + reach)
    window = np.zeros_like(cable)
    window[y0 : y1 + 1, x0 : x1 + 1] = True
    seeds = np.nonzero(cable & window)
    if seeds[0].size == 0:
        raise Abstain("no cable reaches the framed socket")

    seen = np.zeros(cable.shape, dtype=bool)

    def flood(starts):
        stack = deque()
        for y, x in starts:
            if cable[y, x] and not seen[y, x]:
                seen[y, x] = True
                stack.append((y, x))
        while stack:
            y, x = stack.pop()
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < cable.shape[0] and 0 <= nx < cable.shape[1]:
                        if cable[ny, nx] and not seen[ny, nx]:
                            seen[ny, nx] = True
                            stack.append((ny, nx))

    flood(list(zip(seeds[0].tolist(), seeds[1].tolist())))
    for _ in range(64):
        jumped = []
        for dy, dx in DIRECTIONS:
            ahead1 = np.roll(np.roll(cable, -dy, 0), -dx, 1)
            ahead2 = np.roll(np.roll(cable, -2 * dy, 0), -2 * dx, 1)
            # Only continue a genuine straight run: the cable must extend well
            # behind this pixel along the jump direction. This refuses to hop off
            # the flank of a cable that merely passes across the trace.
            ends = seen & ~ahead1 & ~ahead2
            for back in range(1, RUN_EVIDENCE + 1):
                ends = ends & np.roll(np.roll(cable, back * dy, 0), back * dx, 1)
            ys, xs = np.nonzero(ends)
            for y, x in zip(ys.tolist(), xs.tolist()):
                for step in range(3, MAX_BREAK + 1):
                    ny, nx = y + dy * step, x + dx * step
                    if not (0 <= ny < cable.shape[0] and 0 <= nx < cable.shape[1]):
                        break
                    # Land only on a collinear continuation of the same run, so
                    # the cable that passes across the break is skipped, never
                    # entered.
                    ahead = [
                        (ny + dy * k, nx + dx * k) for k in range(RUN_EVIDENCE + 1)
                    ]
                    if all(
                        0 <= ay < cable.shape[0]
                        and 0 <= ax < cable.shape[1]
                        and cable[ay, ax]
                        for ay, ax in ahead
                    ):
                        if not seen[ny, nx]:
                            jumped.append((ny, nx))
                        break
        if not jumped:
            break
        flood(jumped)
    traced = list(zip(*np.nonzero(seen)))
    if len(traced) < MIN_CABLE_PIXELS:
        raise Abstain("traced cable is too short to be a full route")

    height, width = cable.shape
    span_x = max(p[1] for p in traced) - min(p[1] for p in traced)
    if span_x < width // 4:
        raise Abstain("traced cable does not span the bay")

    pads = []
    for name, rgb in CONTRACT_PALETTE.items():
        mask = _near(array, rgb, tol=24)
        for blob in _components(mask):
            if len(blob) >= MIN_PAD_PIXELS:
                pads.append((name, blob))
    if not pads:
        raise Abstain("no palette pads found")

    hits = []
    for name, blob in pads:
        py0, py1, px0, px1 = _dilated_bbox(blob, 3, array.shape)
        if seen[py0 : py1 + 1, px0 : px1 + 1].any():
            hits.append(name)
    unique = sorted(set(hits))
    if len(unique) != 1:
        raise Abstain(f"traced cable reaches {len(unique)} pads, not one")
    return unique[0]
