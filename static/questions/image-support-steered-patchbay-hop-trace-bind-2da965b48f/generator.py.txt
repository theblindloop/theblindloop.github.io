"""Patch-bay hop-trace binding world.

One medium-sized patch bay holds five coloured source pads on the left and five
identical dark sockets on the right. Five identical grey cables realise a
scrambled source-to-socket assignment. Cables cross; at each crossing one cable
passes over and the other passes under, shown by a small white break in the
lower cable. One socket is framed. The answer is the colour of the pad that the
framed socket's cable comes from, which can only be obtained by following that
one cable across its whole interior, carrying the run direction through every
under-crossing break and every corner.

Geometry is laid out so that no two cables share a row or a column, every
crossing is a proper transversal crossing far from any cable endpoint, and the
only near-boundary quantity is the visible break width at an under-crossing.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "patchbay_hop_trace_bind-0.1.0"

CANVAS_W = 840
CANVAS_H = 760
BACKGROUND = (255, 255, 255)

K = 5
X_L = 200
X_R = 650
SRC_Y = [150 + 100 * k for k in range(K)]
SINK_Y = [190 + 100 * k for k in range(K)]
MID_Y = [220 + 100 * k for k in range(K)]
LANES_A = [240 + 35 * j for j in range(K)]
LANES_B = [440 + 35 * j for j in range(K)]

CABLE = (110, 110, 110)
CABLE_WIDTH = 3
SOCKET = (25, 25, 25)
SOCKET_HALF = 11
FRAME = (255, 0, 255)
FRAME_HALF = 19
FRAME_WIDTH = 3
PAD_HALF = 12

COLOR_NAMES = ("red", "orange", "purple", "blue", "green", "teal")
PALETTE = {
    "red": (220, 40, 40),
    "orange": (240, 150, 20),
    "purple": (140, 60, 190),
    "blue": (40, 70, 210),
    "green": (30, 150, 60),
    "teal": (20, 170, 170),
}

GAPS_CLEAR = (4, 5, 6)
GAPS_TIGHT = (2, 3)
QUARANTINE_MARGIN = 3.5
MIN_DECISIVE_CROSSINGS = 2


def segments(wire: dict) -> list[tuple]:
    """The five alternating segments of one cable, in path order."""
    a, b = wire["lane_a"], wire["lane_b"]
    sy, my, ty = SRC_Y[wire["src"]], MID_Y[wire["mid"]], SINK_Y[wire["sink"]]
    return [
        ("h", sy, X_L, a),
        ("v", a, sy, my),
        ("h", my, a, b),
        ("v", b, my, ty),
        ("h", ty, b, X_R),
    ]


def _span(seg: tuple) -> tuple[float, float]:
    lo, hi = seg[2], seg[3]
    return (lo, hi) if lo <= hi else (hi, lo)


def _crosses(h: tuple, v: tuple) -> tuple[float, float] | None:
    hx0, hx1 = _span(h)
    vy0, vy1 = _span(v)
    if hx0 < v[1] < hx1 and vy0 < h[1] < vy1:
        return (float(v[1]), float(h[1]))
    return None


def crossings(wires: list[dict]) -> list[dict]:
    """All transversal cable crossings, deterministically ordered."""
    segs = [segments(w) for w in wires]
    found = []
    for i, si in enumerate(segs):
        for hi, h in enumerate(si):
            if h[0] != "h":
                continue
            for j, sj in enumerate(segs):
                if j == i:
                    continue
                for vi, v in enumerate(sj):
                    if v[0] != "v":
                        continue
                    point = _crosses(h, v)
                    if point is not None:
                        found.append(
                            {
                                "x": point[0],
                                "y": point[1],
                                "h_wire": i,
                                "h_seg": hi,
                                "v_wire": j,
                                "v_seg": vi,
                            }
                        )
    return found


def _seg_distance(p: tuple, q: tuple) -> float:
    """Axis-aligned segment distance; only called for non-crossing pairs."""
    def clamp(value, lo, hi):
        return max(lo, min(hi, value))

    def point_to_seg(px, py, seg):
        if seg[0] == "h":
            lo, hi = _span(seg)
            return abs(py - seg[1]) + max(0.0, lo - px, px - hi)
        lo, hi = _span(seg)
        return abs(px - seg[1]) + max(0.0, lo - py, py - hi)

    best = float("inf")
    for seg, other in ((p, q), (q, p)):
        if seg[0] == "h":
            lo, hi = _span(seg)
            xs = (lo, hi, clamp(other[1] if other[0] == "v" else lo, lo, hi))
            for x in xs:
                best = min(best, point_to_seg(x, seg[1], other))
        else:
            lo, hi = _span(seg)
            ys = (lo, hi, clamp(other[1] if other[0] == "h" else lo, lo, hi))
            for y in ys:
                best = min(best, point_to_seg(seg[1], y, other))
    return float(best)


def sample_scene(seed: int) -> dict:
    """Sample one patch bay. Answer colour is cycled; source row is uniform."""
    rng = random.Random(seed * 6151 + 29)
    target = COLOR_NAMES[seed % len(COLOR_NAMES)]
    near_boundary = seed % 4 == 0
    gap = rng.choice(GAPS_TIGHT if near_boundary else GAPS_CLEAR)

    for _ in range(400):
        sink_of = list(range(K))
        rng.shuffle(sink_of)  # sink_of[i] = socket row of source row i
        marked = rng.randrange(K)
        decisive = sink_of.index(marked)
        if decisive == marked:
            continue  # forbid the row-aligned shortcut for the decisive cable

        mids = list(range(K))
        rng.shuffle(mids)
        lanes_a = list(range(K))
        rng.shuffle(lanes_a)
        lanes_b = list(range(K))
        rng.shuffle(lanes_b)

        others = [name for name in COLOR_NAMES if name != target]
        rng.shuffle(others)
        colors = others[: K - 1]
        colors.insert(decisive, target)

        wires = [
            {
                "src": i,
                "mid": mids[i],
                "sink": sink_of[i],
                "lane_a": LANES_A[lanes_a[i]],
                "lane_b": LANES_B[lanes_b[i]],
                "color": colors[i],
            }
            for i in range(K)
        ]

        cross = crossings(wires)
        decisive_crossings = sum(
            1 for c in cross if decisive in (c["h_wire"], c["v_wire"])
        )
        if decisive_crossings < MIN_DECISIVE_CROSSINGS:
            continue

        order = list(range(K))
        rng.shuffle(order)  # order[p] = cable drawn at depth p; later is on top
        depth = {w: p for p, w in enumerate(order)}
        under = sum(
            1
            for c in cross
            if decisive in (c["h_wire"], c["v_wire"])
            and depth[decisive] < depth[c["v_wire"] if c["h_wire"] == decisive else c["h_wire"]]
        )
        if under < 1 or depth[decisive] == K - 1:
            continue  # the decisive cable must itself dip under, never be the free top cable

        return {
            "gap": int(gap),
            "marked_sink": int(marked),
            "draw_order": order,
            "wires": wires,
        }
    raise RuntimeError(f"no admissible patch bay for seed {seed}")


def _polyline(wire: dict) -> list[tuple[float, float]]:
    a, b = wire["lane_a"], wire["lane_b"]
    sy, my, ty = SRC_Y[wire["src"]], MID_Y[wire["mid"]], SINK_Y[wire["sink"]]
    return [(X_L, sy), (a, sy), (a, my), (b, my), (b, ty), (X_R, ty)]


def render(scene: dict) -> Image.Image:
    """I = R(z): five cased cables in depth order, then pads, sockets and frame."""
    image = Image.new("RGB", (CANVAS_W, CANVAS_H), BACKGROUND)
    draw = ImageDraw.Draw(image)
    gap = int(scene["gap"])
    casing = CABLE_WIDTH + 2 * gap

    for index in list(scene["draw_order"]):
        path = _polyline(scene["wires"][int(index)])
        draw.line(path, fill=BACKGROUND, width=casing, joint="curve")
        draw.line(path, fill=CABLE, width=CABLE_WIDTH, joint="curve")

    for k in range(K):
        y = SINK_Y[k]
        draw.rectangle(
            [X_R - SOCKET_HALF, y - SOCKET_HALF, X_R + SOCKET_HALF, y + SOCKET_HALF],
            fill=SOCKET,
        )
    for wire in scene["wires"]:
        y = SRC_Y[wire["src"]]
        draw.rectangle(
            [X_L - PAD_HALF, y - PAD_HALF, X_L + PAD_HALF, y + PAD_HALF],
            fill=PALETTE[wire["color"]],
        )

    y = SINK_Y[int(scene["marked_sink"])]
    draw.rectangle(
        [X_R - FRAME_HALF, y - FRAME_HALF, X_R + FRAME_HALF, y + FRAME_HALF],
        outline=FRAME,
        width=FRAME_WIDTH,
    )
    return image


def decisive_wire(scene: dict) -> dict:
    marked = int(scene["marked_sink"])
    for wire in scene["wires"]:
        if int(wire["sink"]) == marked:
            return wire
    raise ValueError("framed socket has no cable")


def analytic_gold(scene: dict) -> str:
    """Colour of the pad feeding the framed socket."""
    return str(decisive_wire(scene)["color"])


def margin(scene: dict) -> float:
    """Smallest visible clearance, in pixels, on the traced structure.

    Terms: the white break width shown on the lower cable at every crossing (the
    quantity that decides whether an under-crossing can be read at all), the
    clearance between non-crossing segments of different cables, and the
    clearance between a cable and a foreign pad or socket. The break width
    shrinks toward zero and dominates by construction.
    """
    gap = float(scene["gap"])
    wires = scene["wires"]
    half = CABLE_WIDTH / 2.0
    casing = CABLE_WIDTH + 2.0 * gap
    terms = [gap]

    segs = [segments(w) for w in wires]
    cross_pairs = {
        (c["h_wire"], c["h_seg"], c["v_wire"], c["v_seg"]) for c in crossings(wires)
    }
    for i in range(len(wires)):
        for j in range(i + 1, len(wires)):
            for si, p in enumerate(segs[i]):
                for sj, q in enumerate(segs[j]):
                    if (i, si, j, sj) in cross_pairs or (j, sj, i, si) in cross_pairs:
                        continue
                    terms.append(_seg_distance(p, q) - half - casing / 2.0)

    for i, wire in enumerate(wires):
        for seg in segs[i]:
            for k in range(K):
                if k != int(wire["src"]):
                    pad = ("h", SRC_Y[k], X_L - PAD_HALF, X_L + PAD_HALF)
                    terms.append(_seg_distance(seg, pad) - casing / 2.0)
                if k != int(wire["sink"]):
                    sock = ("h", SINK_Y[k], X_R - SOCKET_HALF, X_R + SOCKET_HALF)
                    terms.append(_seg_distance(seg, sock) - casing / 2.0)
    return float(round(min(terms), 6))


def is_quarantined(scene: dict) -> bool:
    """Breaks so narrow that an under-crossing is not visibly readable."""
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Cable indexing is a pure alias.

    The raster depends on the cables and on their depth order, not on the index
    each cable is stored under. Re-indexing the cable list while carrying the
    depth permutation along re-draws the identical raster and leaves the
    analytic answer unchanged.
    """
    wires = list(scene["wires"])
    order = [int(i) for i in scene["draw_order"]]
    n = len(wires)
    rot = {**scene, "wires": wires[1:] + wires[:1], "draw_order": [(i - 1) % n for i in order]}
    rev = {
        **scene,
        "wires": list(reversed(wires)),
        "draw_order": [n - 1 - i for i in order],
    }
    return [("cable_index_rotation", rot), ("cable_index_reversal", rev)]
