"""Deterministic latent sampler and renderer for Ruler-Keyed Reflections."""

from __future__ import annotations

import copy
import itertools
import random

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 800
CELL = 32
N = 7
BOARD = CELL * N
SYMBOLS = ("up", "right", "down", "left")
TEMPLATES = {
    "up": (
        "0001000", "0011100", "0111110", "0001000", "0001000", "0001000", "0001100",
    ),
    "right": (
        "0000000", "0000100", "0000110", "1111111", "1000110", "0000100", "0000000",
    ),
    "down": (
        "0011000", "0001000", "0001000", "0001000", "0111110", "0011100", "0001000",
    ),
    "left": (
        "0000000", "0010000", "0110001", "1111111", "0110000", "0010000", "0000000",
    ),
}
FRAME = (68, 72, 78)
BACKGROUND = (249, 248, 244)
PALETTE = (
    (20, 126, 153),
    (211, 82, 67),
    (119, 76, 169),
    (37, 116, 194),
    (207, 133, 25),
    (35, 146, 91),
)


def _template_bits(name: str) -> list[list[int]]:
    return [[int(ch) for ch in row] for row in TEMPLATES[name]]


def _reverse_rows(bits: list[list[int]], rows: list[int]) -> list[list[int]]:
    out = [list(row) for row in bits]
    for row in rows:
        out[int(row)] = list(reversed(out[int(row)]))
    return out


def _reverse_cols(bits: list[list[int]], cols: list[int]) -> list[list[int]]:
    out = [list(row) for row in bits]
    for col in cols:
        c = int(col)
        values = [out[r][c] for r in range(N)][::-1]
        for r, value in enumerate(values):
            out[r][c] = value
    return out


def _common_valid_keys() -> tuple[tuple[tuple[int, ...], tuple[int, ...]], ...]:
    """Keys equally available to every class and resistant to shortcut reads."""
    templates = [_template_bits(name) for name in SYMBOLS]
    encoded = {tuple(map(tuple, template)) for template in templates}
    valid = []
    for rows in itertools.combinations(range(N), 5):
        for cols in itertools.combinations(range(N), 5):
            works_for_all = True
            for target in templates:
                visible = _reverse_rows(_reverse_cols(target, list(cols)), list(rows))
                variants = (
                    visible,
                    _reverse_rows(visible, list(rows)),
                    _reverse_cols(visible, list(cols)),
                    _reverse_rows(_reverse_cols(visible, list(cols)), list(rows)),
                )
                distance = sum(visible[r][c] != target[r][c] for r in range(N) for c in range(N))
                if distance < 12 or any(tuple(map(tuple, item)) in encoded for item in variants):
                    works_for_all = False
                    break
            if works_for_all:
                valid.append((rows, cols))
    if not valid:
        raise RuntimeError("no counterbalanced reflection keys")
    return tuple(valid)


COMMON_VALID_KEYS = _common_valid_keys()


def transformed_bits(bits: list[list[int]], flip_rows: list[int], flip_cols: list[int]) -> list[list[int]]:
    """Execute the public rule: selected rows first, selected columns second."""
    return _reverse_cols(_reverse_rows(bits, flip_rows), flip_cols)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    symbol = SYMBOLS[int(seed) % len(SYMBOLS)]
    rows, cols = rng.choice(COMMON_VALID_KEYS)
    flip_rows = list(rows)
    flip_cols = list(cols)
    palette = list(PALETTE)
    rng.shuffle(palette)
    return {
        "symbol": symbol,
        "flip_rows": flip_rows,
        "flip_cols": flip_cols,
        "cluster_x": 300 + rng.randint(-22, 22),
        "cluster_y": 340 + rng.randint(-5, 5),
        "palette": [list(color) for color in palette[:3]],
    }


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def margin(scene: dict) -> float:
    target = _template_bits(str(scene["symbol"]))
    distances = []
    for name in SYMBOLS:
        if name == scene["symbol"]:
            continue
        other = _template_bits(name)
        distances.append(sum(target[r][c] != other[r][c] for r in range(N) for c in range(N)))
    return float(min(distances))


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["flip_rows"] = list(reversed(twin["flip_rows"]))
    twin["flip_cols"] = list(reversed(twin["flip_cols"]))
    return [("reverse_set_serializations", twin)]


def _board_bits(scene: dict) -> list[list[int]]:
    # Invert the declared row-then-column transform. Each conditional reversal
    # is its own inverse, so inverse order is columns first, then rows.
    target = _template_bits(str(scene["symbol"]))
    return _reverse_rows(_reverse_cols(target, scene["flip_cols"]), scene["flip_rows"])


def _draw_grid(draw: ImageDraw.ImageDraw, x0: int, y0: int, rows: int, cols: int) -> None:
    draw.rectangle((x0, y0, x0 + cols * CELL, y0 + rows * CELL), outline=FRAME, width=4)
    for index in range(1, cols):
        x = x0 + index * CELL
        draw.line((x, y0, x, y0 + rows * CELL), fill=FRAME, width=3)
    for index in range(1, rows):
        y = y0 + index * CELL
        draw.line((x0, y, x0 + cols * CELL, y), fill=FRAME, width=3)


def _fill_cell(draw: ImageDraw.ImageDraw, x0: int, y0: int, row: int, col: int, color: tuple[int, int, int]) -> None:
    pad = 7
    left = x0 + col * CELL + pad
    top = y0 + row * CELL + pad
    draw.rounded_rectangle((left, top, left + CELL - 2 * pad, top + CELL - 2 * pad), radius=3, fill=color)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    bx = int(scene["cluster_x"])
    by = int(scene["cluster_y"])
    # Detached rulers remain outside the ablation halo of the board frame.
    hx, hy = bx, by - 270
    vx, vy = bx - 58, by
    _draw_grid(draw, bx, by, N, N)
    _draw_grid(draw, hx, hy, 1, N)
    _draw_grid(draw, vx, vy, N, 1)
    board_color, col_color, row_color = (tuple(v) for v in scene["palette"])
    bits = _board_bits(scene)
    for r in range(N):
        for c in range(N):
            if bits[r][c]:
                _fill_cell(draw, bx, by, r, c, board_color)
    for c in scene["flip_cols"]:
        _fill_cell(draw, hx, hy, 0, int(c), col_color)
    for r in scene["flip_rows"]:
        _fill_cell(draw, vx, vy, int(r), 0, row_color)
    return image
