"""Deterministic four-donut renderer and analytic entropy decision."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 820
HEIGHT = 260
CENTERS_X = (110, 310, 510, 710)
CENTER_Y = 112
OUTER_RADIUS = 75
INNER_RADIUS = 34
PALETTE = ("#d64b4b", "#377eb8", "#2f9e67", "#e49b32")
BACKGROUND = "#ffffff"
INK = "#30343b"
QUARANTINE_BITS = 0.06


def _entropy(values: list[float]) -> float:
    return -sum(value * math.log2(value) for value in values if value > 0.0)


def _panel_entropies(scene: dict) -> list[float]:
    profiles = scene["proportions"]
    return [_entropy(profiles[index]) for index in scene["panel_order"]]


def _jittered(rng: random.Random, template: list[float], amount: float) -> list[float]:
    values = [max(0.035, value + rng.uniform(-amount, amount)) for value in template]
    total = sum(values)
    values = [value / total for value in values]
    rng.shuffle(values)
    return values


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; consecutive seeds cycle the winning panel."""

    rng = random.Random(int(seed))
    near_boundary = int(seed) % 11 == 0
    templates = [
        [0.28, 0.26, 0.24, 0.22],
        [0.33, 0.27, 0.22, 0.18] if near_boundary else [0.48, 0.22, 0.18, 0.12],
        [0.61, 0.18, 0.13, 0.08],
        [0.72, 0.13, 0.09, 0.06],
    ]
    amount = 0.0 if near_boundary else 0.008
    proportions = [_jittered(rng, template, amount) for template in templates]

    winner_panel = int(seed) % 4
    losers = [1, 2, 3]
    rng.shuffle(losers)
    panel_order = [None, None, None, None]
    panel_order[winner_panel] = 0
    for panel, profile in zip((i for i in range(4) if i != winner_panel), losers):
        panel_order[panel] = profile

    palette_order = []
    for _ in range(4):
        order = [0, 1, 2, 3]
        rng.shuffle(order)
        palette_order.append(order)

    return {
        "proportions": proportions,
        "sector_orders": [rng.randrange(4) for _ in range(4)],
        "palette_order": palette_order,
        "panel_order": panel_order,
        "start_angles": [rng.uniform(-180.0, 180.0) for _ in range(4)],
        "translation": [rng.randint(-5, 5), rng.randint(-4, 4)],
    }


def _font(size: int) -> ImageFont.ImageFont:
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    """Render only from the latent scene; no analytic decision is consulted."""

    scale = 4
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    profiles = scene["proportions"]
    panel_order = scene["panel_order"]
    sector_orders = scene["sector_orders"]
    palette_order = scene["palette_order"]
    start_angles = scene["start_angles"]
    dx, dy = scene["translation"]

    for panel in range(4):
        cx = (CENTERS_X[panel] + dx) * scale
        cy = (CENTER_Y + dy) * scale
        outer = OUTER_RADIUS * scale
        inner = INNER_RADIUS * scale
        box = (cx - outer, cy - outer, cx + outer, cy + outer)
        values = profiles[panel_order[panel]]
        offset = int(sector_orders[panel]) % 4
        ordered = [values[(slot + offset) % 4] for slot in range(4)]
        colors = [PALETTE[index] for index in palette_order[panel]]
        angle = float(start_angles[panel])
        for slot, value in enumerate(ordered):
            end = angle + 360.0 * value
            draw.pieslice(box, start=angle, end=end, fill=colors[slot])
            angle = end
        inner_box = (cx - inner, cy - inner, cx + inner, cy + inner)
        draw.ellipse(inner_box, fill=BACKGROUND, outline=INK, width=2 * scale)
        draw.ellipse(box, outline=INK, width=2 * scale)
        draw.text(
            (cx, (221 + dy) * scale),
            "ABCD"[panel],
            fill=INK,
            font=_font(21 * scale),
            anchor="mm",
        )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    entropies = _panel_entropies(scene)
    return "ABCD"[max(range(4), key=entropies.__getitem__)]


def margin(scene: dict) -> float:
    entropies = sorted(_panel_entropies(scene), reverse=True)
    return entropies[0] - entropies[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_BITS


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """A +4 cyclic offset changes latent integers but renders identically."""

    twin = copy.deepcopy(scene)
    twin["sector_orders"] = [int(value) + 4 for value in scene["sector_orders"]]
    return [("full_cycle_sector_offsets", twin)]
