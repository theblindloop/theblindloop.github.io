"""Image-only inverse for the four-donut entropy decision.

This module deliberately has no access to renderer state, latent scenes, or gold.
It segments the public palette, discovers four horizontal chart components, and
computes entropy from the recovered colored pixel counts.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

PALETTE = np.array(
    [
        (214, 75, 75),
        (55, 126, 184),
        (47, 158, 103),
        (228, 155, 50),
        (255, 255, 255),
        (48, 52, 59),
    ],
    dtype=np.int32,
)


def _runs(bits: np.ndarray) -> list[tuple[int, int]]:
    padded = np.pad(bits.astype(np.int8), (1, 1))
    changes = np.flatnonzero(np.diff(padded))
    return [(int(changes[i]), int(changes[i + 1])) for i in range(0, len(changes), 2)]


def _entropy(counts: np.ndarray) -> float:
    proportions = counts / counts.sum()
    return -sum(float(p) * math.log2(float(p)) for p in proportions if p > 0.0)


def decision_from_image(image: Image.Image) -> str:
    """Return A-D using only pixels; raise when the chart evidence is absent."""

    array = np.asarray(image.convert("RGB"), dtype=np.int32)
    distances = ((array[:, :, None, :] - PALETTE[None, None, :, :]) ** 2).sum(axis=3)
    labels = distances.argmin(axis=2)
    colored = labels < 4
    column_runs = [(left, right) for left, right in _runs(colored.any(axis=0)) if right - left > 25]
    if len(column_runs) != 4:
        raise ValueError(f"expected four colored donut components, found {len(column_runs)}")

    entropies = []
    for left, right in column_runs:
        local = labels[:, left:right]
        counts = np.array([(local == color).sum() for color in range(4)], dtype=np.float64)
        if np.any(counts < 120.0) or counts.sum() < 5000.0:
            raise ValueError("a donut is missing a measurable colored sector")
        entropies.append(_entropy(counts))
    return "ABCD"[max(range(4), key=entropies.__getitem__)]
