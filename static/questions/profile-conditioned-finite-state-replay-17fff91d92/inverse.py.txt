"""Independent pixel-only inverse for finite-state replay.

This module intentionally duplicates only the public raster geometry. It does
not import or inspect the renderer, prompts, generator, verifier, or scene.
"""

from __future__ import annotations

import math

from PIL import Image


CANVAS = (1800, 800)
LABELS = ("A", "B", "C")
STATE_BASE_COLORS = (
    (48, 112, 190),
    (45, 151, 104),
    (139, 82, 172),
)
INPUT_BASE_COLORS = (
    (234, 133, 49),
    (32, 157, 172),
)
RING = (224, 111, 42)

TABLE_LEFT = 80
TABLE_TOP = 230
ROW_LABEL_W = 110
CELL_W = 185
CELL_H = 78
START_Y = 625
START_XS = (160, 340, 520)
STATE_KEY_X = 700
STATE_KEY_Y = 155
STATE_KEY_STEP = 145
INPUT_KEY_X = 700
INPUT_KEY_Y = 245
INPUT_KEY_STEP = 230
SEQUENCE_X = 700
SEQUENCE_Y = 410
TILE_W = 105
TILE_STEP = 130


def _exact_color(value: tuple[int, int, int], choices: tuple[tuple[int, int, int], ...]) -> int:
    matches = [index for index, color in enumerate(choices) if value == color]
    if len(matches) != 1:
        raise ValueError(f"unrecognized or ambiguous marker color {value!r}")
    return matches[0]


def _palette(frame: Image.Image) -> tuple[dict[tuple[int, int, int], int], dict[tuple[int, int, int], int]]:
    pixels = frame.load()
    state_colors: dict[tuple[int, int, int], int] = {}
    input_colors: dict[tuple[int, int, int], int] = {}
    for state in range(3):
        x = STATE_KEY_X + state * STATE_KEY_STEP + 18
        y = STATE_KEY_Y + 22
        color = pixels[x, y]
        _exact_color(color, STATE_BASE_COLORS)
        if color in state_colors:
            raise ValueError("state key reuses a color")
        state_colors[color] = state
    for symbol in range(2):
        x = INPUT_KEY_X + symbol * INPUT_KEY_STEP + 18
        y = INPUT_KEY_Y + 22
        color = pixels[x, y]
        _exact_color(color, INPUT_BASE_COLORS)
        if color in input_colors:
            raise ValueError("input key reuses a color")
        input_colors[color] = symbol
    return state_colors, input_colors


def _start_state(frame: Image.Image, state_colors: dict[tuple[int, int, int], int]) -> int:
    pixels = frame.load()
    points: list[tuple[int, int]] = []
    for y in range(START_Y - 70, START_Y + 71):
        for x in range(80, 601):
            if pixels[x, y] == RING:
                points.append((x, y))
    if len(points) < 30:
        raise ValueError("START row does not expose a complete ring")
    center_x = sum(point[0] for point in points) / len(points)
    center_y = sum(point[1] for point in points) / len(points)
    position = min(range(3), key=lambda index: math.hypot(center_x - START_XS[index], center_y - START_Y))
    if math.hypot(center_x - START_XS[position], center_y - START_Y) > 8.0:
        raise ValueError("ring does not identify one start node")
    # The node label is drawn at the center; sample an inner side pixel instead.
    sample = pixels[START_XS[position] - 15, START_Y]
    if sample not in state_colors:
        raise ValueError("ringed node has no readable state fill")
    return state_colors[sample]


def _transition_table(frame: Image.Image, state_colors: dict[tuple[int, int, int], int]) -> list[list[int]]:
    pixels = frame.load()
    table: list[list[int]] = []
    for row in range(3):
        values: list[int] = []
        top = TABLE_TOP + row * CELL_H
        for column in range(2):
            left = TABLE_LEFT + ROW_LABEL_W + column * CELL_W
            sample = pixels[left + 50, top + CELL_H // 2]
            if sample not in state_colors:
                raise ValueError("transition cell is missing its destination marker")
            values.append(state_colors[sample])
        table.append(values)
    return table


def _input_sequence(frame: Image.Image, input_colors: dict[tuple[int, int, int], int]) -> list[int]:
    pixels = frame.load()
    sequence: list[int] = []
    for index in range(7):
        left = SEQUENCE_X + index * TILE_STEP
        sample = pixels[left + TILE_W // 2, SEQUENCE_Y + 72]
        if sample not in input_colors:
            raise ValueError(f"input tile {index + 1} has no readable symbol")
        sequence.append(input_colors[sample])
    if not sequence or set(sequence) != {0, 1}:
        raise ValueError("input sequence must contain both visible symbols")
    return sequence


def decision_from_image(image: Image.Image) -> str:
    """Decode the public raster and compose all visible transitions."""

    frame = image.convert("RGB")
    if frame.size != CANVAS:
        raise ValueError(f"unexpected raster size {frame.size}")
    state_colors, input_colors = _palette(frame)
    state = _start_state(frame, state_colors)
    table = _transition_table(frame, state_colors)
    sequence = _input_sequence(frame, input_colors)
    for symbol in sequence:
        state = int(table[state][symbol])
    if state not in range(3):
        raise ValueError("replayed state is invalid")
    return LABELS[state]
