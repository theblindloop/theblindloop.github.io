"""Deterministic renderer and analytic side for finite-state replay scenes."""

from __future__ import annotations

import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont


LABELS = ("A", "B", "C")
INPUT_LABELS = ("alpha", "beta")
CANVAS = (1800, 800)

BACKGROUND = (248, 250, 252)
CARD = (255, 255, 255)
INK = (38, 46, 58)
MUTED = (94, 106, 119)
GRID = (187, 197, 208)
RING = (224, 111, 42)
STATE_BASE_COLORS = (
    (48, 112, 190),
    (45, 151, 104),
    (139, 82, 172),
)
INPUT_BASE_COLORS = (
    (234, 133, 49),
    (32, 157, 172),
)

TABLE_LEFT = 80
TABLE_TOP = 230
ROW_LABEL_W = 110
CELL_W = 185
CELL_H = 78
TABLE_ROWS = 3
TABLE_COLS = 2

START_TITLE_Y = 535
START_Y = 625
START_XS = (160, 340, 520)

STATE_KEY_X = 700
STATE_KEY_Y = 155
STATE_KEY_STEP = 145
KEY_SWATCH_R = 15

INPUT_KEY_X = 700
INPUT_KEY_Y = 245
INPUT_KEY_STEP = 230

SEQUENCE_X = 700
SEQUENCE_Y = 410
TILE_W = 105
TILE_H = 120
TILE_STEP = 130

_FONT_PATHS = (
    "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
    "/usr/share/fonts/dejavu/DejaVuSans.ttf",
)


def _font(size: int):
    for candidate in _FONT_PATHS:
        if Path(candidate).is_file():
            return ImageFont.truetype(candidate, size=size)
    return ImageFont.load_default()


def _centered_text(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font, fill):
    box = draw.textbbox((0, 0), text, font=font)
    width = box[2] - box[0]
    height = box[3] - box[1]
    draw.text(
        (xy[0] - width / 2, xy[1] - height / 2 - box[1]),
        text,
        font=font,
        fill=fill,
    )


def _state_color(scene: dict, state: int) -> tuple[int, int, int]:
    order = scene["state_palette_order"]
    return STATE_BASE_COLORS[int(order[int(state)])]


def _input_color(scene: dict, symbol: int) -> tuple[int, int, int]:
    order = scene["input_palette_order"]
    return INPUT_BASE_COLORS[int(order[int(symbol)])]


def _replay(start_state: int, table: list[list[int]], sequence: list[int]) -> int:
    state = int(start_state)
    if state not in range(3):
        raise ValueError("start state must be A, B, or C")
    if len(table) != 3 or any(len(row) != 2 for row in table):
        raise ValueError("transition table must be three by two")
    for symbol in sequence:
        symbol = int(symbol)
        if symbol not in (0, 1):
            raise ValueError("input symbol must be alpha or beta")
        destination = int(table[state][symbol])
        if destination not in range(3):
            raise ValueError("transition destination must be A, B, or C")
        state = destination
    return state


def sample_scene(seed: int) -> dict:
    """Sample a balanced, styled three-state composition problem."""

    logic_rng = random.Random(int(seed) * 1000003 + 17)
    style_rng = random.Random(int(seed) * 1000033 + 31)

    # Each input column is a permutation, keeping every start state viable for
    # balancing while making both inputs genuinely state-changing.
    alpha = list(range(3))
    beta = list(range(3))
    while True:
        logic_rng.shuffle(alpha)
        logic_rng.shuffle(beta)
        if alpha != list(range(3)) or beta != list(range(3)):
            break
    sequence = [0, 1]
    sequence.extend(logic_rng.randrange(2) for _ in range(5))
    logic_rng.shuffle(sequence)

    table = [[alpha[state], beta[state]] for state in range(3)]
    desired = int(seed) % 3
    candidates = [
        start
        for start in range(3)
        if _replay(start, table, sequence) == desired
    ]
    if len(candidates) != 1:
        raise AssertionError("permutation composition must produce one start per class")
    start_state = candidates[0]

    state_palette_order = list(range(3))
    input_palette_order = list(range(2))
    node_order = list(range(3))
    style_rng.shuffle(state_palette_order)
    style_rng.shuffle(input_palette_order)
    style_rng.shuffle(node_order)

    # A small-marker variant is a declared finite-resolution quarantine probe.
    marker_radius = 5 if int(seed) % 11 == 0 else 13
    symbol_radius = 6 if int(seed) % 13 == 0 else 17
    node_radius = 22 if int(seed) % 17 == 0 else 34
    scene = {
        "canvas": list(CANVAS),
        "start_state": start_state,
        "transition_table": table,
        "input_sequence": sequence,
        "state_palette_order": state_palette_order,
        "input_palette_order": input_palette_order,
        "node_order": node_order,
        "node_radius": node_radius,
        "marker_radius": marker_radius,
        "symbol_radius": symbol_radius,
    }
    if LABELS[_replay(start_state, table, sequence)] != LABELS[desired]:
        raise AssertionError("sampler failed to balance the requested class")
    return scene


def _draw_input_icon(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    symbol: int,
    color: tuple[int, int, int],
    radius: int,
) -> None:
    cx, cy = center
    if symbol == 0:
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=color, outline=INK, width=2)
    else:
        draw.polygon(
            ((cx, cy - radius), (cx + radius, cy), (cx, cy + radius), (cx - radius, cy)),
            fill=color,
            outline=INK,
        )


def _draw_state_key(draw: ImageDraw.ImageDraw, scene: dict, label_font, small_font) -> None:
    draw.text((STATE_KEY_X, STATE_KEY_Y - 28), "STATE KEY", font=small_font, fill=MUTED)
    for state, label in enumerate(LABELS):
        cx = STATE_KEY_X + state * STATE_KEY_STEP + 18
        cy = STATE_KEY_Y + 22
        draw.ellipse(
            (cx - KEY_SWATCH_R, cy - KEY_SWATCH_R, cx + KEY_SWATCH_R, cy + KEY_SWATCH_R),
            fill=_state_color(scene, state),
            outline=INK,
            width=1,
        )
        _centered_text(draw, (cx + 42, cy), label, label_font, INK)


def _draw_input_key(draw: ImageDraw.ImageDraw, scene: dict, label_font, small_font) -> None:
    draw.text((INPUT_KEY_X, INPUT_KEY_Y - 28), "INPUT KEY", font=small_font, fill=MUTED)
    for symbol, label in enumerate(INPUT_LABELS):
        cx = INPUT_KEY_X + symbol * INPUT_KEY_STEP + 18
        cy = INPUT_KEY_Y + 22
        _draw_input_icon(draw, (cx, cy), symbol, _input_color(scene, symbol), 13)
        _centered_text(draw, (cx + 58, cy), label, label_font, INK)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", tuple(scene["canvas"]), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(24)
    section_font = _font(17)
    label_font = _font(18)
    small_font = _font(13)
    node_font = _font(22)

    draw.text((70, 28), "FINITE-STATE REPLAY", font=title_font, fill=INK)
    draw.text(
        (70, 70),
        "Use the visible transition table, then read the numbered input sequence left to right.",
        font=small_font,
        fill=MUTED,
    )
    _draw_state_key(draw, scene, label_font, small_font)
    _draw_input_key(draw, scene, label_font, small_font)

    draw.text((TABLE_LEFT, TABLE_TOP - 48), "TRANSITION TABLE", font=section_font, fill=INK)
    draw.text(
        (TABLE_LEFT, TABLE_TOP - 25),
        "row = current state; cell marker = destination",
        font=small_font,
        fill=MUTED,
    )

    header_y = TABLE_TOP - 2
    draw.rectangle(
        (TABLE_LEFT, header_y, TABLE_LEFT + ROW_LABEL_W + TABLE_COLS * CELL_W, TABLE_TOP + TABLE_ROWS * CELL_H),
        fill=CARD,
        outline=INK,
        width=2,
    )
    for column, label in enumerate(INPUT_LABELS):
        left = TABLE_LEFT + ROW_LABEL_W + column * CELL_W
        draw.line((left, header_y, left, TABLE_TOP + TABLE_ROWS * CELL_H), fill=GRID, width=1)
        _centered_text(draw, (left + CELL_W / 2, header_y + 22), label, label_font, INK)
    draw.line(
        (TABLE_LEFT, TABLE_TOP, TABLE_LEFT + ROW_LABEL_W + TABLE_COLS * CELL_W, TABLE_TOP),
        fill=GRID,
        width=1,
    )
    for row, label in enumerate(LABELS):
        top = TABLE_TOP + row * CELL_H
        bottom = top + CELL_H
        draw.line((TABLE_LEFT, top, TABLE_LEFT + ROW_LABEL_W + TABLE_COLS * CELL_W, top), fill=GRID, width=1)
        draw.line((TABLE_LEFT + ROW_LABEL_W, top, TABLE_LEFT + ROW_LABEL_W, bottom), fill=GRID, width=1)
        draw.ellipse(
            (TABLE_LEFT + 18, top + 25, TABLE_LEFT + 48, top + 55),
            fill=_state_color(scene, row),
            outline=INK,
            width=1,
        )
        _centered_text(draw, (TABLE_LEFT + 78, top + CELL_H / 2), label, label_font, INK)
        for column in range(TABLE_COLS):
            left = TABLE_LEFT + ROW_LABEL_W + column * CELL_W
            destination = int(scene["transition_table"][row][column])
            marker_x = left + 50
            marker_y = top + CELL_H / 2
            radius = int(scene["marker_radius"])
            draw.ellipse(
                (marker_x - radius, marker_y - radius, marker_x + radius, marker_y + radius),
                fill=_state_color(scene, destination),
                outline=INK,
                width=1,
            )
            _centered_text(draw, (left + 103, marker_y), LABELS[destination], label_font, INK)
    draw.line(
        (TABLE_LEFT, TABLE_TOP + TABLE_ROWS * CELL_H, TABLE_LEFT + ROW_LABEL_W + TABLE_COLS * CELL_W, TABLE_TOP + TABLE_ROWS * CELL_H),
        fill=INK,
        width=2,
    )

    draw.text((TABLE_LEFT, START_TITLE_Y), "START STATE", font=section_font, fill=INK)
    draw.text((TABLE_LEFT + 150, START_TITLE_Y + 2), "the ring marks the initial state", font=small_font, fill=MUTED)
    for position, state in enumerate(scene["node_order"]):
        cx = START_XS[position]
        cy = START_Y
        radius = int(scene["node_radius"])
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=_state_color(scene, state), outline=INK, width=2)
        _centered_text(draw, (cx, cy), LABELS[state], node_font, (255, 255, 255))
        if state == int(scene["start_state"]):
            ring_radius = radius + 10
            draw.ellipse(
                (cx - ring_radius, cy - ring_radius, cx + ring_radius, cy + ring_radius),
                outline=RING,
                width=4,
            )

    draw.text((SEQUENCE_X, SEQUENCE_Y - 48), "INPUT SEQUENCE", font=section_font, fill=INK)
    draw.text(
        (SEQUENCE_X, SEQUENCE_Y - 25),
        "apply in numbered order; the final state is not shown",
        font=small_font,
        fill=MUTED,
    )
    for index, symbol in enumerate(scene["input_sequence"], start=1):
        left = SEQUENCE_X + (index - 1) * TILE_STEP
        top = SEQUENCE_Y
        draw.rounded_rectangle((left, top, left + TILE_W, top + TILE_H), radius=8, fill=CARD, outline=INK, width=2)
        _centered_text(draw, (left + TILE_W / 2, top + 23), str(index), label_font, INK)
        _draw_input_icon(
            draw,
            (left + TILE_W / 2, top + 72),
            int(symbol),
            _input_color(scene, int(symbol)),
            int(scene["symbol_radius"]),
        )
        if index < len(scene["input_sequence"]):
            arrow_x0 = left + TILE_W + 8
            arrow_x1 = left + TILE_STEP - 8
            arrow_y = top + TILE_H / 2
            draw.line((arrow_x0, arrow_y, arrow_x1, arrow_y), fill=MUTED, width=2)
            draw.polygon(
                ((arrow_x1, arrow_y), (arrow_x1 - 7, arrow_y - 4), (arrow_x1 - 7, arrow_y + 4)),
                fill=MUTED,
            )

    draw.text(
        (SEQUENCE_X, 625),
        "At each step, use the current row and the tile's input column; continue from the destination.",
        font=small_font,
        fill=MUTED,
    )
    return image


def analytic_gold(scene: dict) -> str:
    state = _replay(
        int(scene["start_state"]),
        scene["transition_table"],
        scene["input_sequence"],
    )
    return LABELS[state]


def margin(scene: dict) -> float:
    """One-sided pixel clearance for the smallest evidence marker or icon."""

    return float(
        min(
            int(scene["marker_radius"]) - 1,
            int(scene["symbol_radius"]) - 1,
            int(scene["node_radius"]) - 20,
        )
    )


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # Every answer-bearing field is explicitly drawn. No latent alias is used.
    return []
