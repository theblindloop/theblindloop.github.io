"""Independent image-only decoder for directed inequality consequence."""

from __future__ import annotations

from fractions import Fraction

import numpy as np
from PIL import Image

CANVAS = (1200, 980)
INK = (31, 38, 46)
PANEL = (247, 249, 252)
DIVIDER = (181, 188, 197)
SYMBOL_RGB = ((210, 61, 67), (38, 111, 187), (47, 151, 91), (126, 73, 181), (218, 103, 44))
TAG_RGB = {"amber": (226, 151, 28), "cyan": (18, 157, 181), "violet": (141, 77, 190)}
ALLOWED = {(255, 255, 255), PANEL, INK, DIVIDER, *SYMBOL_RGB, *TAG_RGB.values()}
PREMISE_Y = (25, 170, 315)
PROPOSAL_Y = (535, 675, 815)


class OracleError(ValueError):
    pass


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]; seen[y, x] = True; xs, ys = [], []
        while stack:
            cy, cx = stack.pop(); xs.append(cx); ys.append(cy)
            for ny, nx in ((cy-1,cx),(cy+1,cx),(cy,cx-1),(cy,cx+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny,nx] and not seen[ny,nx]:
                    seen[ny,nx] = True; stack.append((ny,nx))
        out.append((min(xs), min(ys), max(xs), max(ys), len(xs)))
    return out


def _read_group(arr: np.ndarray, box: tuple[int, int, int, int]) -> list[int]:
    x0, y0, x1, y1 = box
    vector = []
    for color in SYMBOL_RGB:
        mask = np.all(arr[y0:y1, x0:x1] == color, axis=2)
        components = [c for c in _components(mask) if c[4] > 350]
        if any(not (430 <= c[4] <= 1400 and 28 <= c[2]-c[0]+1 <= 43
                    and 28 <= c[3]-c[1]+1 <= 44) for c in components):
            raise OracleError("malformed symbol")
        if int(mask.sum()) - sum(c[4] for c in components) > 10:
            raise OracleError("damaged symbol fragments")
        vector.append(len(components))
    if not 2 <= sum(vector) <= 6:
        raise OracleError("invalid group size")
    return vector


def _rank(rows: list[list[int]]) -> int:
    a = [[Fraction(v) for v in row] for row in rows]; rank = 0
    for col in range(5):
        pivot = next((i for i in range(rank, len(a)) if a[i][col]), None)
        if pivot is None: continue
        a[rank], a[pivot] = a[pivot], a[rank]
        z = a[rank][col]; a[rank] = [v/z for v in a[rank]]
        for i in range(len(a)):
            if i != rank and a[i][col]:
                z = a[i][col]; a[i] = [v-z*q for v,q in zip(a[i],a[rank])]
        rank += 1
    return rank


def _coordinates(basis: list[list[int]], vector: list[int]) -> tuple[Fraction, ...] | None:
    if _rank(basis) != len(basis): return None
    aug = [[Fraction(basis[c][r]) for c in range(len(basis))] + [Fraction(vector[r])] for r in range(5)]
    row = 0; pivots = []
    for col in range(len(basis)):
        pivot = next((i for i in range(row, len(aug)) if aug[i][col]), None)
        if pivot is None: return None
        aug[row], aug[pivot] = aug[pivot], aug[row]
        z = aug[row][col]; aug[row] = [v/z for v in aug[row]]
        for i in range(len(aug)):
            if i != row and aug[i][col]:
                z = aug[i][col]; aug[i] = [v-z*q for v,q in zip(aug[i],aug[row])]
        pivots.append(row); row += 1
    if any(all(not x for x in eq[:-1]) and eq[-1] for eq in aug): return None
    return tuple(aug[pivots[c]][-1] for c in range(len(basis)))


def _forced(premises: list[list[int]], proposal: list[int]) -> bool:
    co = _coordinates(premises, proposal)
    return co is not None and all(c >= 0 for c in co) and sum(c > 0 for c in co) >= 2


def _check_greater(arr: np.ndarray, y: int) -> None:
    cy = y + 57
    mask = np.all(arr[cy-32:cy+33, 560:642] == INK, axis=2)
    comps = [c for c in _components(mask) if c[4] > 100]
    if len(comps) != 1 or not (950 <= comps[0][4] <= 1200):
        raise OracleError("greater-than mark is missing or malformed")
    # The rightmost ink must be near the vertical centre, unlike a less-than mark.
    ys, xs = np.nonzero(mask)
    tip_y = ys[xs >= xs.max()-2]
    if (len(tip_y) < 3 or abs(float(tip_y.mean()) - 32.0) > 4.0
            or float(tip_y.std()) > 5.0):
        raise OracleError("comparison direction is invalid")


def _relation(left: list[int], right: list[int]) -> list[int]:
    return [a-b for a,b in zip(left,right)]


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    if rgb.size != CANVAS: raise OracleError("wrong canvas")
    colors = rgb.getcolors(maxcolors=32)
    if colors is None or not {tuple(color) for _,color in colors} <= ALLOWED:
        raise OracleError("unknown raster color")
    arr = np.asarray(rgb)
    premises = []
    for y in PREMISE_Y:
        left = _read_group(arr, (55,y+8,495,y+107)); right = _read_group(arr, (705,y+8,1145,y+107))
        _check_greater(arr,y); premises.append(_relation(left,right))
    if _rank(premises) != 3: raise OracleError("premises are not independent")
    matches, seen = [], set()
    for y in PROPOSAL_Y:
        left = _read_group(arr, (55,y+8,515,y+107)); right = _read_group(arr, (685,y+8,1145,y+107))
        _check_greater(arr,y)
        counts = {name:int(np.all(arr[y:y+11,350:850] == color,axis=2).sum()) for name,color in TAG_RGB.items()}
        names = [name for name,count in counts.items() if count > 2000]
        if len(names) != 1 or names[0] in seen: raise OracleError("proposal border is invalid")
        seen.add(names[0])
        if _forced(premises,_relation(left,right)): matches.append(names[0])
    if seen != set(TAG_RGB) or len(matches) != 1:
        raise OracleError("no unique direction-preserving consequence")
    return matches[0]
