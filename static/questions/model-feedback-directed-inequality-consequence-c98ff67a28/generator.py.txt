"""Render directed symbol-weight inequalities and candidate consequences."""

from __future__ import annotations

from fractions import Fraction
import random

from PIL import Image, ImageDraw

CANVAS = (1200, 980)
ANSWERS = ("amber", "cyan", "violet")
RGB = {
    "background": (255, 255, 255), "panel": (247, 249, 252), "ink": (31, 38, 46),
    "red": (210, 61, 67), "blue": (38, 111, 187), "green": (47, 151, 91),
    "purple": (126, 73, 181), "orange": (218, 103, 44),
    "amber": (226, 151, 28), "cyan": (18, 157, 181), "violet": (141, 77, 190),
}
SYMBOL_COLORS = ("red", "blue", "green", "purple", "orange")
PREMISE_Y = (25, 170, 315)
PROPOSAL_Y = (535, 675, 815)
ROW_H = 115

# Each relation r means r dot weights > 0, hence the left multiset is heavier.
# In every family target = p1+p2+p3. Each wrong proposal is a signed combination
# with a negative coefficient and has a positive integer counterexample (tests).
TEMPLATES = (
    (((1,-2,2,-1,0),(1,1,-2,3,-2),(0,1,-1,1,-1)), (2,0,-1,3,-3), ((0,2,-3,3,-1),(0,1,-2,2,0))),
    (((1,0,-2,2,-1),(1,0,3,-2,-2),(1,-2,-3,1,2)), (3,-2,-2,1,-1), ((1,-2,2,-3,1),(0,2,1,1,-3))),
    (((0,3,-3,-1,2),(1,-2,2,1,-1),(1,-3,3,1,-2)), (2,-2,2,1,-1), ((1,-1,1,0,-1),(1,2,-2,-1,1))),
    (((0,1,3,1,-3),(0,1,-2,0,1),(2,0,-2,-2,1)), (2,2,-1,-1,-1), ((2,0,3,-1,-3),(2,-2,2,-2,-1))),
    (((0,1,-3,1,1),(1,2,0,-1,-1),(2,-1,1,0,-1)), (3,2,-2,0,-1), ((1,1,3,-2,-2),(3,-3,-1,2,0))),
    (((1,1,2,-1,-3),(1,2,-3,-1,0),(1,0,-2,1,1)), (3,3,-3,-1,-2), ((1,-1,3,1,-2),(0,2,-1,-2,-1))),
    (((0,1,3,-1,-3),(0,2,-2,0,1),(3,0,-1,-1,-1)), (3,3,0,-2,-3), ((3,-3,-2,0,1),(3,-2,1,-1,-2))),
    (((0,1,-2,-1,2),(0,1,-1,-2,1),(2,-3,2,0,0)), (2,-1,-1,-3,3), ((2,-3,0,2,2),(2,-3,3,-1,-1))),
    (((0,0,3,-2,2),(0,1,-1,1,-2),(1,0,1,0,-3)), (1,1,3,-1,-3), ((1,-1,2,-1,-1),(1,-1,-1,1,-3))),
)


def _sides(relation: tuple[int, ...]) -> list[list[int]]:
    return [[max(v, 0) for v in relation], [max(-v, 0) for v in relation]]


def _relation(equation: list[list[int]]) -> tuple[int, ...]:
    if len(equation) != 2 or any(len(side) != 5 for side in equation):
        raise ValueError("malformed comparison")
    return tuple(int(a) - int(b) for a, b in zip(equation[0], equation[1]))


def _rank(rows: list[tuple[int, ...]]) -> int:
    a = [[Fraction(v) for v in row] for row in rows]
    rank = 0
    for col in range(5):
        pivot = next((i for i in range(rank, len(a)) if a[i][col]), None)
        if pivot is None:
            continue
        a[rank], a[pivot] = a[pivot], a[rank]
        scale = a[rank][col]
        a[rank] = [v / scale for v in a[rank]]
        for i in range(len(a)):
            if i != rank and a[i][col]:
                scale = a[i][col]
                a[i] = [v - scale * q for v, q in zip(a[i], a[rank])]
        rank += 1
    return rank


def _coordinates(basis: list[tuple[int, ...]], vector: tuple[int, ...]) -> tuple[Fraction, ...] | None:
    """Unique coordinates of vector in an independent row basis, if present."""
    if _rank(basis) != len(basis):
        return None
    # Solve B^T c = vector by reduced elimination of five equations in three vars.
    aug = [[Fraction(basis[c][r]) for c in range(len(basis))] + [Fraction(vector[r])]
           for r in range(5)]
    row = 0
    pivots: list[int] = []
    for col in range(len(basis)):
        pivot = next((i for i in range(row, len(aug)) if aug[i][col]), None)
        if pivot is None:
            return None
        aug[row], aug[pivot] = aug[pivot], aug[row]
        scale = aug[row][col]
        aug[row] = [v / scale for v in aug[row]]
        for i in range(len(aug)):
            if i != row and aug[i][col]:
                scale = aug[i][col]
                aug[i] = [v - scale * q for v, q in zip(aug[i], aug[row])]
        pivots.append(row)
        row += 1
    if any(all(not x for x in equation[:-1]) and equation[-1] for equation in aug):
        return None
    return tuple(aug[pivots[c]][-1] for c in range(len(basis)))


def _forced(premises: list[tuple[int, ...]], proposal: tuple[int, ...]) -> bool:
    coefficients = _coordinates(premises, proposal)
    return coefficients is not None and all(c >= 0 for c in coefficients) and sum(c > 0 for c in coefficients) >= 2


def sample_scene(seed: int) -> dict:
    rng = random.Random((int(seed) ^ 0xD1EC7ED) & 0xFFFFFFFFFFFFFFFF)
    raw_premises, target, wrong = TEMPLATES[int(seed) % len(TEMPLATES)]
    premises = [_sides(row) for row in raw_premises]
    rng.shuffle(premises)
    target_slot = int(seed) % 3
    proposals = [_sides(row) for row in wrong]
    proposals.insert(target_slot, _sides(target))
    answer_color = ANSWERS[(int(seed) // 3) % 3]
    other_colors = [color for color in ANSWERS if color != answer_color]
    rng.shuffle(other_colors)
    tag_colors: list[str | None] = [None, None, None]
    tag_colors[target_slot] = answer_color
    for slot in range(3):
        if tag_colors[slot] is None:
            tag_colors[slot] = other_colors.pop()
    scene = {"premises": premises, "proposals": proposals, "tag_colors": tag_colors}
    if analytic_gold(scene) != answer_color:
        raise RuntimeError("constructed answer binding is inconsistent")
    return scene


def _symbol(draw: ImageDraw.ImageDraw, kind: int, cx: int, cy: int) -> None:
    color = RGB[SYMBOL_COLORS[kind]]
    if kind == 0:
        draw.ellipse((cx - 19, cy - 19, cx + 19, cy + 19), fill=color, outline=RGB["ink"], width=3)
    elif kind == 1:
        draw.rectangle((cx - 18, cy - 18, cx + 18, cy + 18), fill=color, outline=RGB["ink"], width=3)
    elif kind == 2:
        pts = ((cx, cy - 22), (cx - 22, cy + 18), (cx + 22, cy + 18))
        draw.polygon(pts, fill=color); draw.line(pts + (pts[0],), fill=RGB["ink"], width=3, joint="curve")
    elif kind == 3:
        pts = ((cx, cy - 22), (cx - 22, cy), (cx, cy + 22), (cx + 22, cy))
        draw.polygon(pts, fill=color); draw.line(pts + (pts[0],), fill=RGB["ink"], width=3, joint="curve")
    else:
        pts = ((cx - 19, cy - 11), (cx, cy - 21), (cx + 19, cy - 11),
               (cx + 19, cy + 11), (cx, cy + 21), (cx - 19, cy + 11))
        draw.polygon(pts, fill=color); draw.line(pts + (pts[0],), fill=RGB["ink"], width=3, joint="curve")


def _group(draw: ImageDraw.ImageDraw, vector: list[int], box: tuple[int, int, int, int]) -> None:
    items = [kind for kind, count in enumerate(vector) for _ in range(int(count))]
    x0, y0, x1, y1 = box
    spacing = 61
    start = (x0 + x1 - spacing * (len(items) - 1)) // 2
    for index, kind in enumerate(items):
        _symbol(draw, kind, start + index * spacing, (y0 + y1) // 2)


def _greater(draw: ImageDraw.ImageDraw, y: int) -> None:
    cy = y + ROW_H // 2
    draw.line(((574, cy - 25), (628, cy), (574, cy + 25)), fill=RGB["ink"], width=9, joint="curve")


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS, RGB["background"])
    draw = ImageDraw.Draw(image)
    for y, comparison in zip(PREMISE_Y, scene["premises"]):
        for box in ((35, y, 515, y + ROW_H), (685, y, 1165, y + ROW_H)):
            draw.rounded_rectangle(box, radius=17, fill=RGB["panel"], outline=RGB["ink"], width=4)
        _group(draw, comparison[0], (55, y + 8, 495, y + ROW_H - 8))
        _group(draw, comparison[1], (705, y + 8, 1145, y + ROW_H - 8))
        _greater(draw, y)
    draw.line((90, 480, 1110, 480), fill=(181, 188, 197), width=4)
    for y, comparison, color in zip(PROPOSAL_Y, scene["proposals"], scene["tag_colors"]):
        draw.rounded_rectangle((35, y, 1165, y + ROW_H), radius=17,
                               fill=RGB["panel"], outline=RGB[str(color)], width=10)
        _group(draw, comparison[0], (55, y + 8, 515, y + ROW_H - 8))
        _group(draw, comparison[1], (685, y + 8, 1145, y + ROW_H - 8))
        _greater(draw, y)
    return image


def analytic_gold(scene: dict) -> str:
    premises = [_relation(comparison) for comparison in scene["premises"]]
    if _rank(premises) != 3:
        raise ValueError("premises must be independent")
    matches = [i for i, comparison in enumerate(scene["proposals"])
               if _forced(premises, _relation(comparison))]
    if len(matches) != 1 or set(scene["tag_colors"]) != set(ANSWERS):
        raise ValueError("scene has no unique direction-preserving consequence")
    return str(scene["tag_colors"][matches[0]])


def margin(scene: dict) -> float:
    premises = [_relation(comparison) for comparison in scene["premises"]]
    gaps: list[Fraction] = []
    for comparison in scene["proposals"]:
        coefficients = _coordinates(premises, _relation(comparison))
        if coefficients is None:
            return 0.0
        if not _forced(premises, _relation(comparison)):
            negative = [-c for c in coefficients if c < 0]
            if not negative:
                return 0.0
            gaps.append(max(negative))
    return float(min(gaps)) if len(gaps) == 2 else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 1 or analytic_gold(scene) not in ANSWERS
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
