"""Independent pixel-only inverse for pocket nesting.

The module deliberately knows no renderer state. It segments dark ink, checks
that the ink is one connected component, flood-fills its white complement, and
classifies the containment order of the two bounded components.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        points = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        output.append(np.asarray(points, dtype=np.int32))
    return output


def _bbox(points: np.ndarray) -> tuple[int, int, int, int]:
    ys, xs = points[:, 0], points[:, 1]
    return int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())


def _strictly_contains(outer: tuple[int, int, int, int], inner: tuple[int, int, int, int]) -> bool:
    ox0, oy0, ox1, oy1 = outer
    ix0, iy0, ix1, iy1 = inner
    return ox0 + 3 < ix0 and oy0 + 3 < iy0 and ox1 - 3 > ix1 and oy1 - 3 > iy1


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    dark = np.max(array, axis=2) < 90
    points = np.argwhere(dark)
    if len(points) < 320:
        return "abstain"
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    if x1 - x0 < 72 or y1 - y0 < 20 or x1 - x0 > 120 or y1 - y0 > 45:
        return "abstain"
    y0, x0 = max(0, int(y0) - 2), max(0, int(x0) - 2)
    y1 = min(array.shape[0] - 1, int(y1) + 2)
    x1 = min(array.shape[1] - 1, int(x1) + 2)
    ink = dark[y0 : y1 + 1, x0 : x1 + 1]
    ink_components = _components(ink)
    if len(ink_components) != 1:
        return "abstain"
    whites = _components(~ink)
    height, width = ink.shape
    pockets = []
    for component in whites:
        ys, xs = component[:, 0], component[:, 1]
        touches_border = bool(
            np.any(ys == 0) or np.any(xs == 0) or np.any(ys == height - 1) or np.any(xs == width - 1)
        )
        if not touches_border and len(component) >= 80:
            pockets.append(component)
    if len(pockets) != 2:
        return "abstain"
    first, second = _bbox(pockets[0]), _bbox(pockets[1])
    return "yes" if _strictly_contains(first, second) or _strictly_contains(second, first) else "no"
