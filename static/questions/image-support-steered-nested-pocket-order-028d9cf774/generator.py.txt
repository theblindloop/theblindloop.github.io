"""Deterministic renderer and analytic arm for nested pocket order."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = 512
BLACK = (24, 28, 32)
PALE = (224, 229, 234)
WHITE = (255, 255, 255)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    layout = "nested" if int(seed) % 2 == 0 else "side_by_side"
    trace_width = rng.choice((88, 92, 94))
    trace_height = 28
    stroke = 3
    x = rng.randint(72, CANVAS - trace_width - 72)
    # Place the short axis inside a randomly selected 64 px band. This varies
    # absolute position while keeping the thin cue from straddling a coarse
    # ablation boundary merely by accident.
    y = rng.choice((146, 210, 274, 338))
    decorations = []
    for _ in range(4):
        while True:
            dx = rng.randint(28, CANVAS - 29)
            dy = rng.randint(28, CANVAS - 29)
            if not (x - 28 <= dx <= x + trace_width + 28 and y - 28 <= dy <= y + trace_height + 28):
                decorations.append([dx, dy, rng.choice((3, 4, 5))])
                break
    return {
        "layout": layout,
        "x": x,
        "y": y,
        "trace_width": trace_width,
        "trace_height": trace_height,
        "stroke": stroke,
        "decorations": decorations,
    }


def _frame(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], stroke: int) -> None:
    x0, y0, x1, y1 = box
    draw.rectangle((x0, y0, x1, y0 + stroke - 1), fill=BLACK)
    draw.rectangle((x0, y1 - stroke + 1, x1, y1), fill=BLACK)
    draw.rectangle((x0, y0, x0 + stroke - 1, y1), fill=BLACK)
    draw.rectangle((x1 - stroke + 1, y0, x1, y1), fill=BLACK)


def render(scene: dict) -> Image.Image:
    layout = scene["layout"]
    x, y = int(scene["x"]), int(scene["y"])
    width, height = int(scene["trace_width"]), int(scene["trace_height"])
    stroke = int(scene["stroke"])
    image = Image.new("RGB", (CANVAS, CANVAS), WHITE)
    draw = ImageDraw.Draw(image)
    for dx, dy, radius in scene["decorations"]:
        draw.ellipse((dx - radius, dy - radius, dx + radius, dy + radius), fill=PALE)

    if layout == "nested":
        _frame(draw, (x, y, x + width, y + height), stroke)
        inner_width = width // 2
        inner_height = max(12, height // 2)
        ix = x + (width - inner_width) // 2
        iy = y + (height - inner_height) // 2
        _frame(draw, (ix, iy, ix + inner_width, iy + inner_height), stroke)
        cy = y + height // 2
        draw.rectangle((x, cy - stroke // 2, ix, cy + (stroke - 1) // 2), fill=BLACK)
    elif layout == "side_by_side":
        gap = max(18, width // 8)
        loop_width = (width - gap) // 2
        left = (x, y, x + loop_width, y + height)
        right_x = x + loop_width + gap
        right = (right_x, y, x + width, y + height)
        _frame(draw, left, stroke)
        _frame(draw, right, stroke)
        cy = y + height // 2
        draw.rectangle(
            (x + loop_width, cy - stroke // 2, right_x, cy + (stroke - 1) // 2),
            fill=BLACK,
        )
    else:
        raise ValueError(f"unknown layout: {layout}")
    return image


def analytic_gold(scene: dict) -> str:
    return "yes" if scene["layout"] == "nested" else "no"


def margin(scene: dict) -> float:
    # Minimum nominal separation between the two pocket extents or from a pocket
    # extent to its enclosing pocket extent.
    return float(max(10, int(scene["trace_height"]) // 4 - int(scene["stroke"])))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
