"""Independent pixel-only inverse arm for marked colored intervals."""

from __future__ import annotations

import numpy as np
from PIL import Image

PALETTE = {
    "red": (216, 58, 50),
    "blue": (40, 120, 200),
    "green": (39, 150, 87),
    "orange": (228, 134, 36),
}


def _components(mask: np.ndarray) -> list[np.ndarray]:
    seen = np.zeros(mask.shape, dtype=bool)
    found = []
    h, w = mask.shape
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        pts = []
        while stack:
            cy, cx = stack.pop()
            pts.append((cy, cx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = cy + dy, cx + dx
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(pts) >= 20:
            found.append(np.asarray(pts, dtype=float))
    return found


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    intervals = {}
    for name, rgb in PALETTE.items():
        distance = np.sqrt(np.sum((arr - np.asarray(rgb, dtype=np.int32)) ** 2, axis=2))
        ys, xs = np.nonzero(distance < 48)
        if len(xs) < 500:
            raise ValueError(f"missing colored interval: {name}")
        intervals[name] = (float(xs.min()), float(xs.max()), float(np.median(ys)))

    black = np.max(arr, axis=2) < 75
    components = [part for part in _components(black) if 90 <= len(part) <= 650]
    if len(components) != 4:
        raise ValueError(f"expected four marker components, found {len(components)}")
    centers = [(float(part[:, 1].mean()), float(part[:, 0].mean())) for part in components]

    scores = {}
    for name, (x0, x1, y) in intervals.items():
        marker_x, marker_y = min(centers, key=lambda center: abs(center[1] - y))
        if abs(marker_y - y) > 16 or x1 - x0 < 150:
            raise ValueError("marker-to-interval binding is not reliable")
        midpoint = (x0 + x1) / 2.0
        scores[name] = abs(marker_x - midpoint) / (x1 - x0)
    ordered = sorted(scores, key=scores.get)
    if scores[ordered[1]] - scores[ordered[0]] < 0.012:
        raise ValueError("pixel measurement is ambiguous")
    return ordered[0]
