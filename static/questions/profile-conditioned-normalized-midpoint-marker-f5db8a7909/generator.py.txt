"""Renderer for the normalized midpoint-marker measurement world."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

PALETTE = {
    "red": "#d83a32",
    "blue": "#2878c8",
    "green": "#279657",
    "orange": "#e48624",
}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    colors = list(PALETTE)
    rng.shuffle(colors)
    winner = int(seed) % 4
    others = [i for i in range(4) if i != winner]
    rng.shuffle(others)

    # Normalized offsets are the relevant quantities. Lengths and sides vary
    # independently so neither raw distance nor direction identifies the answer.
    if int(seed) % 7 == 0:
        offsets = [0.0] * 4
        offsets[winner] = rng.uniform(0.105, 0.120)
        offsets[others[0]] = offsets[winner] + rng.uniform(0.020, 0.029)
        offsets[others[1]] = rng.uniform(0.265, 0.320)
        offsets[others[2]] = rng.uniform(0.350, 0.405)
    else:
        offsets = [0.0] * 4
        offsets[winner] = rng.uniform(0.055, 0.115)
        offsets[others[0]] = rng.uniform(0.180, 0.235)
        offsets[others[1]] = rng.uniform(0.275, 0.330)
        offsets[others[2]] = rng.uniform(0.360, 0.420)

    fractions = []
    for offset in offsets:
        side = -1 if rng.random() < 0.5 else 1
        fractions.append(0.5 + side * offset)

    lengths = [rng.randint(185, 390) for _ in range(4)]
    lefts = [rng.randint(65, 535 - length) for length in lengths]
    rows = [64 + 94 * i + rng.randint(-5, 5) for i in range(4)]
    return {
        "colors": colors,
        "rows": rows,
        "lefts": lefts,
        "lengths": lengths,
        "fractions": fractions,
        "width": 600,
        "height": 420,
    }


def render(scene: dict) -> Image.Image:
    colors = scene["colors"]
    rows = scene["rows"]
    lefts = scene["lefts"]
    lengths = scene["lengths"]
    fractions = scene["fractions"]
    width = scene["width"]
    height = scene["height"]
    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), "white")
    draw = ImageDraw.Draw(image)
    for color, y, left, length, fraction in zip(colors, rows, lefts, lengths, fractions):
        x0, x1 = left, left + length
        fill = PALETTE[color]
        draw.rectangle((x0 * scale, (y - 7) * scale, x1 * scale, (y + 7) * scale), fill=fill)
        draw.rectangle(((x0 - 3) * scale, (y - 18) * scale, (x0 + 3) * scale, (y + 18) * scale), fill=fill)
        draw.rectangle(((x1 - 3) * scale, (y - 18) * scale, (x1 + 3) * scale, (y + 18) * scale), fill=fill)
        marker_x = x0 + fraction * length
        r = 11
        draw.ellipse(((marker_x - r) * scale, (y - r) * scale,
                      (marker_x + r) * scale, (y + r) * scale), fill="#151515")
        draw.ellipse(((marker_x - 4) * scale, (y - 4) * scale,
                      (marker_x + 4) * scale, (y + 4) * scale), fill="white")
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    fractions = scene["fractions"]
    colors = scene["colors"]
    offsets = [abs(value - 0.5) for value in fractions]
    return colors[min(range(len(offsets)), key=offsets.__getitem__)]


def margin(scene: dict) -> float:
    values = sorted(abs(value - 0.5) for value in scene["fractions"])
    return (values[1] - values[0]) * 100.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 3.5


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
