"""Pixel-only inverse arm for the rook-clear grid-panel search world.

The oracle receives a rasterized image and nothing else.  It recovers the row
of framed panels from strongly drawn ink, isolates the dark dots inside each
panel, groups the dot pixels into blobs, measures for every panel the smallest
per-pair alignment slack over its dot centres, and selects the panel whose
slack is largest.  It abstains when the layout or the blob count is not
recovered or when the runner-up gap is too small.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

AMBIGUOUS = "ambiguous"

N_PANELS = 4
DOTS_PER_PANEL = 5

DRAWN_THRESHOLD = 220  # frames and dots; the faint lattice stays below
INK_THRESHOLD = 430  # dots only
BLOB_RADIUS = 12.0
MIN_BLOB_PIXELS = 45
MAX_BLOB_PIXELS = 220
PANEL_INSET = 6
ABSTAIN_MARGIN_PX = 10.0


def _modal_colour(pixels: np.ndarray) -> np.ndarray:
    flat = pixels.reshape(-1, 3).astype(np.int64)
    keys = (flat[:, 0] << 16) | (flat[:, 1] << 8) | flat[:, 2]
    values, counts = np.unique(keys, return_counts=True)
    key = int(values[int(np.argmax(counts))])
    return np.array([(key >> 16) & 255, (key >> 8) & 255, key & 255], dtype=np.int64)


def _runs(flags: np.ndarray, min_length: int = 6) -> list:
    runs = []
    start = None
    for index, flag in enumerate(flags):
        if flag and start is None:
            start = index
        elif not flag and start is not None:
            if index - start >= min_length:
                runs.append((start, index))
            start = None
    if start is not None and len(flags) - start >= min_length:
        runs.append((start, len(flags)))
    return runs


def _blob_centres(mask: np.ndarray) -> list:
    """Group dot pixels into well-separated blobs and return their centroids."""
    coords = np.argwhere(mask)  # rows of (y, x)
    centres = []
    while coords.shape[0] > 0 and len(centres) <= DOTS_PER_PANEL:
        seed = coords[0].astype(np.float64)
        close = np.hypot(coords[:, 0] - seed[0], coords[:, 1] - seed[1]) <= BLOB_RADIUS
        blob = coords[close]
        if not (MIN_BLOB_PIXELS <= blob.shape[0] <= MAX_BLOB_PIXELS):
            return []
        centres.append((float(blob[:, 1].mean()), float(blob[:, 0].mean())))
        coords = coords[~close]
    if coords.shape[0] > 0 or len(centres) != DOTS_PER_PANEL:
        return []
    return centres


def _clearance(points) -> float:
    best = None
    for i in range(len(points)):
        for j in range(i + 1, len(points)):
            dx = abs(points[i][0] - points[j][0])
            dy = abs(points[i][1] - points[j][1])
            slack = dx if dx < dy else dy
            if best is None or slack < best:
                best = slack
    return float(best)


def analyse(image: Image.Image) -> dict:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int64)
    background = _modal_colour(pixels)
    distance = np.abs(pixels - background).sum(axis=2)
    drawn = distance > DRAWN_THRESHOLD
    ink = distance > INK_THRESHOLD

    empty = {"decision": AMBIGUOUS, "margin": 0.0, "clearances": []}
    bands = _runs(drawn.any(axis=1))
    if len(bands) != 1:
        return empty
    band_top, band_bottom = bands[0]
    columns = _runs(drawn[band_top:band_bottom, :].any(axis=0))
    if len(columns) != N_PANELS:
        return empty

    clearances = []
    for left, right in columns:
        window = ink[
            band_top + PANEL_INSET : band_bottom - PANEL_INSET,
            left + PANEL_INSET : right - PANEL_INSET,
        ]
        if window.size == 0:
            return empty
        centres = _blob_centres(window)
        if len(centres) != DOTS_PER_PANEL:
            return empty
        clearances.append(_clearance(centres))

    order = sorted(range(N_PANELS), key=lambda index: clearances[index], reverse=True)
    gap = clearances[order[0]] - clearances[order[1]]
    if gap < ABSTAIN_MARGIN_PX:
        return {
            "decision": AMBIGUOUS,
            "margin": round(float(gap), 9),
            "clearances": clearances,
        }
    return {
        "decision": f"panel-{order[0] + 1}",
        "margin": round(float(gap), 9),
        "clearances": clearances,
    }


def decision_from_image(image: Image.Image) -> str:
    return analyse(image)["decision"]


def margin_from_image(image: Image.Image) -> float:
    return round(float(analyse(image)["margin"]), 9)
