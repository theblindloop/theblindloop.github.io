"""Renderer and analytic arm for the rook-clear grid-panel search world.

Four framed square panels each carry a drawn lattice and five identical dots,
one dot per lattice cell.  In exactly one panel no two dots share a lattice row
and no two share a lattice column; in every other panel exactly one pair of
dots is aligned along a row or along a column.  The decision is the number of
the panel with no aligned pair, in left-to-right order.
"""

from __future__ import annotations

import json
import random

from PIL import Image, ImageDraw

N_PANELS = 4
DOTS_PER_PANEL = 5

PANEL_SIZE = 214
PANEL_GAP = 22
MARGIN = 22
CANVAS_WIDTH = 2 * MARGIN + N_PANELS * PANEL_SIZE + (N_PANELS - 1) * PANEL_GAP
CANVAS_HEIGHT = 2 * MARGIN + PANEL_SIZE

FRAME = 3
GRID_INSET = 10
GRID_SPAN = PANEL_SIZE - 2 * GRID_INSET
DOT_RADIUS = 5
JITTER = 2  # cell-centre wobble, far below a cell width

GRID_SIZES = (5, 6)
ALIGNED_CEILING = 0.5  # an aligned pair shares its coordinate exactly
MARGIN_FLOOR_PX = 20.0
QUARANTINE_MARGIN_PX = 10.0

ANSWERS = tuple(f"panel-{index + 1}" for index in range(N_PANELS))

# (background, frame, lattice, ink); ink is far from background, the lattice is
# faint, and the frames are mid-tone, so the three drawn layers stay separable.
PALETTES = (
    ((247, 245, 240), (132, 130, 138), (214, 212, 206), (26, 30, 44)),
    ((240, 246, 249), (126, 142, 152), (206, 218, 224), (16, 46, 62)),
    ((249, 243, 243), (146, 126, 128), (222, 208, 208), (58, 20, 28)),
    ((243, 247, 241), (128, 144, 126), (210, 220, 208), (24, 50, 26)),
)


def json_clone(scene: dict) -> dict:
    """Round-trip copy of a scene, matching what the manifest stores."""
    return json.loads(json.dumps(scene, sort_keys=True))


def panel_box(index: int) -> tuple:
    left = MARGIN + index * (PANEL_SIZE + PANEL_GAP)
    top = MARGIN
    return (left, top, left + PANEL_SIZE, top + PANEL_SIZE)


def cell_size(grid_n: int) -> float:
    return GRID_SPAN / float(grid_n)


def cell_centre(grid_n: int, col: int, row: int) -> tuple:
    step = cell_size(grid_n)
    return (GRID_INSET + (col + 0.5) * step, GRID_INSET + (row + 0.5) * step)


def alignment_clearance(points) -> float:
    """Smallest per-pair alignment slack: zero when a pair shares a row or column.

    For every unordered pair the slack is the smaller of the horizontal and the
    vertical centre separation, so a pair aligned along either axis scores zero.
    A panel is scored by its worst, i.e. smallest, pair.
    """
    best = None
    for i in range(len(points)):
        for j in range(i + 1, len(points)):
            dx = abs(float(points[i][0]) - float(points[j][0]))
            dy = abs(float(points[i][1]) - float(points[j][1]))
            slack = dx if dx < dy else dy
            if best is None or slack < best:
                best = slack
    return float(best)


def _clear_cells(rng: random.Random, grid_n: int) -> list:
    cols = rng.sample(range(grid_n), DOTS_PER_PANEL)
    rows = rng.sample(range(grid_n), DOTS_PER_PANEL)
    return [[cols[i], rows[i]] for i in range(DOTS_PER_PANEL)]


def _panel_points(rng: random.Random, grid_n: int, clear: bool) -> list:
    cells = _clear_cells(rng, grid_n)
    jitters = [
        [rng.randint(-JITTER, JITTER), rng.randint(-JITTER, JITTER)]
        for _ in range(DOTS_PER_PANEL)
    ]
    if not clear:
        mover, anchor = rng.sample(range(DOTS_PER_PANEL), 2)
        axis = rng.randrange(2)  # 0 aligns a column, 1 aligns a row
        cells[mover][1 - axis] = cells[anchor][1 - axis]
        jitters[mover][1 - axis] = jitters[anchor][1 - axis]
    points = []
    for (col, row), (jx, jy) in zip(cells, jitters):
        cx, cy = cell_centre(grid_n, col, row)
        points.append([int(round(cx)) + jx, int(round(cy)) + jy])
    return points


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    palette = rng.randrange(len(PALETTES))
    grid_n = rng.choice(GRID_SIZES)
    answer_index = rng.randrange(N_PANELS)
    panels = [
        _panel_points(rng, grid_n, clear=(index == answer_index))
        for index in range(N_PANELS)
    ]
    scene = {
        "width": CANVAS_WIDTH,
        "height": CANVAS_HEIGHT,
        "palette": palette,
        "grid_n": grid_n,
        "panels": panels,
    }
    clearances = _clearances(scene)
    distractor_peak = max(
        clearances[i] for i in range(N_PANELS) if i != answer_index
    )
    ok = (
        ANSWERS.index(analytic_gold(scene)) == answer_index
        and margin(scene) >= MARGIN_FLOOR_PX
        and distractor_peak <= ALIGNED_CEILING
    )
    if not ok:  # pragma: no cover - construction keeps this unreachable
        return sample_scene(seed + 7_919)
    return scene


def _clearances(scene: dict) -> list:
    return [alignment_clearance(points) for points in scene["panels"]]


def analytic_gold(scene: dict) -> str:
    clearances = _clearances(scene)
    best = max(range(N_PANELS), key=lambda index: clearances[index])
    return ANSWERS[best]


def margin(scene: dict) -> float:
    """Runner-up alignment-clearance gap, in pixels."""
    ordered = sorted(_clearances(scene), reverse=True)
    return round(ordered[0] - ordered[1], 9)


def is_quarantined(scene: dict) -> bool:
    return float(margin(scene)) < QUARANTINE_MARGIN_PX


def latent_symmetries(scene: dict) -> list:
    """Declared pixel-identical latent aliases that must preserve the answer."""
    period = json_clone(scene)
    period["palette"] = scene["palette"] + len(PALETTES)

    reordered = json_clone(scene)
    reordered["panels"] = [list(reversed(points)) for points in scene["panels"]]
    return [("palette_period", period), ("panel_dot_order", reordered)]


def render(scene: dict) -> Image.Image:
    background, frame, lattice, ink = PALETTES[scene["palette"] % len(PALETTES)]
    grid_n = int(scene["grid_n"])
    image = Image.new("RGB", (scene["width"], scene["height"]), background)
    draw = ImageDraw.Draw(image)
    step = cell_size(grid_n)
    for index, points in enumerate(scene["panels"]):
        left, top, right, bottom = panel_box(index)
        for k in range(grid_n + 1):
            offset = int(round(GRID_INSET + k * step))
            draw.line(
                [left + GRID_INSET, top + offset, right - GRID_INSET, top + offset],
                fill=lattice,
                width=1,
            )
            draw.line(
                [left + offset, top + GRID_INSET, left + offset, bottom - GRID_INSET],
                fill=lattice,
                width=1,
            )
        draw.rectangle([left, top, right, bottom], outline=frame, width=FRAME)
        for x, y in points:
            cx, cy = left + int(x), top + int(y)
            draw.ellipse(
                [cx - DOT_RADIUS, cy - DOT_RADIUS, cx + DOT_RADIUS, cy + DOT_RADIUS],
                fill=ink,
            )
    return image
