"""Pixel-only oracle for paddock_gate_gap.

Receives a PNG and nothing else.  It never imports the renderer, the prompts,
the generator, the verifier, or any analytic gold, and it is never handed a
latent scene.

Method: the picture is painted in flat inks, so the distinct non-white colours
are read straight off the raster.  The fence is the ink that lays down a large
thin structure spanning most of the page; the animal is the only ink whose mask is a round solid blob
(a disc covers about pi/4 of its bounding box, a rail net covers far less, and
a bar covers all of it).  The fence ink is then treated as a wall and the page
border is flooded through everything else; the animal can leave exactly when
the flood reaches it.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "paddock_gate_gap-oracle-0.1.0"
ABSTAIN = "abstain"

_INK_MAX = 235
_MIN_FENCE_AREA = 3000
_MIN_FENCE_DIAG = 250.0
_MIN_ANIMAL_AREA = 380


def _colour_masks(arr: np.ndarray) -> list:
    ink = arr.min(axis=-1) < _INK_MAX
    if not ink.any():
        return []
    flat = arr[ink].astype(np.int32)
    keys = flat[:, 0] * 65536 + flat[:, 1] * 256 + flat[:, 2]
    out = []
    for key, count in zip(*np.unique(keys, return_counts=True)):
        if count < 60:
            continue
        r, g, b = int(key) >> 16, (int(key) >> 8) & 255, int(key) & 255
        mask = ink & (arr[:, :, 0] == r) & (arr[:, :, 1] == g) & (arr[:, :, 2] == b)
        out.append((mask, int(count)))
    return out


def _bbox(mask: np.ndarray):
    ys, xs = np.nonzero(mask)
    return int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())


def _prop(reach: np.ndarray, free: np.ndarray) -> np.ndarray:
    """Propagate the reached set along every contiguous free run of each row."""
    h, w = free.shape
    pad = np.zeros((h, 1), dtype=bool)
    f = np.concatenate([free, pad], axis=1).ravel()
    r = np.concatenate([reach, pad], axis=1).ravel()
    starts = f & ~np.concatenate([[False], f[:-1]])
    run_id = np.cumsum(starts)
    n = int(run_id[-1]) if run_id.size else 0
    hit = np.zeros(n + 1, dtype=bool)
    sel = r & f
    if sel.any():
        hit[run_id[sel]] = True
    out = f & hit[run_id]
    return out.reshape(h, w + 1)[:, :w]


def _flood(free: np.ndarray, seeds: np.ndarray) -> np.ndarray:
    reach = free & seeds
    for _ in range(200):
        before = int(reach.sum())
        reach = _prop(reach, free)
        reach = np.ascontiguousarray(_prop(np.ascontiguousarray(reach.T), np.ascontiguousarray(free.T)).T)
        if int(reach.sum()) == before:
            break
    return reach


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    h, w = arr.shape[:2]
    masks = _colour_masks(arr)
    if not masks:
        return ABSTAIN

    fence = None
    fence_diag = 0.0
    animals = []
    for mask, count in masks:
        x0, y0, x1, y1 = _bbox(mask)
        bw, bh = x1 - x0 + 1, y1 - y0 + 1
        diag = float(np.hypot(bw, bh))
        fill = count / float(bw * bh)
        if count >= _MIN_FENCE_AREA and diag > fence_diag and fill < 0.45:
            fence, fence_diag = mask, diag
        aspect = bw / float(bh)
        if (count >= _MIN_ANIMAL_AREA and 0.80 <= aspect <= 1.25
                and 0.68 <= fill <= 0.88):
            animals.append((mask, (x0 + x1) // 2, (y0 + y1) // 2))

    if fence is None or fence_diag < _MIN_FENCE_DIAG:
        return ABSTAIN
    if len(animals) != 1:
        return ABSTAIN

    _, ax, ay = animals[0]
    fx0, fy0, fx1, fy1 = _bbox(fence)
    if not (fx0 < ax < fx1 and fy0 < ay < fy1):
        return ABSTAIN

    free = ~fence
    seeds = np.zeros((h, w), dtype=bool)
    seeds[0, :] = True
    seeds[-1, :] = True
    seeds[:, 0] = True
    seeds[:, -1] = True
    reach = _flood(free, seeds)
    if not free[ay, ax]:
        return ABSTAIN
    return "yes" if bool(reach[ay, ax]) else "no"
