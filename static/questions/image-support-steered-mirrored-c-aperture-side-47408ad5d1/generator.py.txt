"""Renderer for a compact, thick C-shaped annular region."""

from __future__ import annotations

import math
import random
from PIL import Image, ImageDraw

CANVAS = 512
SCALE = 3
INKS = ((25, 92, 190), (184, 42, 65), (17, 137, 104), (116, 61, 173))
BACKGROUNDS = ((250, 248, 242), (246, 249, 252))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0xC0A7E)
    outer = rng.randint(108, 122)
    thickness = rng.randint(38, 46)
    return {
        "open_side": "left" if int(seed) % 2 else "right",
        "center_x": rng.randint(194, 318),
        "center_y": rng.randint(194, 318),
        "outer_radius": outer,
        "inner_radius": outer - thickness,
        "ink_color": list(INKS[rng.randrange(len(INKS))]),
        "background_color": list(BACKGROUNDS[rng.randrange(len(BACKGROUNDS))]),
        "distractors": [
            [rng.randint(38, 474), rng.randint(38, 474), rng.randint(3, 6)]
            for _ in range(rng.randint(0, 3))
        ],
    }


def _arc_polygon(scene: dict, scale: int) -> list[tuple[float, float]]:
    cx = float(scene["center_x"]) * scale
    cy = float(scene["center_y"]) * scale
    outer = float(scene["outer_radius"]) * scale
    inner = float(scene["inner_radius"]) * scale
    # Image angles grow clockwise because y grows downward. A right-opening C
    # keeps the continuous 304-degree arc from 28 through 332 degrees.
    start = 28.0 if scene["open_side"] == "right" else 208.0
    stop = start + 304.0
    angles = [start + (stop - start) * i / 304 for i in range(305)]
    outer_points = [
        (cx + outer * math.cos(math.radians(a)), cy + outer * math.sin(math.radians(a)))
        for a in angles
    ]
    inner_points = [
        (cx + inner * math.cos(math.radians(a)), cy + inner * math.sin(math.radians(a)))
        for a in reversed(angles)
    ]
    return outer_points + inner_points


def render(scene: dict) -> Image.Image:
    background = tuple(scene["background_color"])
    ink = tuple(scene["ink_color"])
    image = Image.new("RGB", (CANVAS * SCALE, CANVAS * SCALE), background)
    draw = ImageDraw.Draw(image)
    # Neutral dots vary local clutter but are excluded by the chroma-based oracle.
    for x, y, radius in scene["distractors"]:
        box = tuple(SCALE * v for v in (x - radius, y - radius, x + radius, y + radius))
        draw.ellipse(box, fill=(172, 172, 172))
    draw.polygon(_arc_polygon(scene, SCALE), fill=ink)
    return image.resize((CANVAS, CANVAS), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["open_side"])


def margin(scene: dict) -> float:
    # Area removed from one horizontal half versus its mirrored runner-up.
    outer = float(scene["outer_radius"])
    inner = float(scene["inner_radius"])
    return round(math.pi * (outer * outer - inner * inner) * (56.0 / 360.0), 6)


def is_quarantined(scene: dict) -> bool:
    thickness = float(scene["outer_radius"]) - float(scene["inner_radius"])
    clearance = min(
        float(scene["center_x"]),
        CANVAS - float(scene["center_x"]),
        float(scene["center_y"]),
        CANVAS - float(scene["center_y"]),
    ) - float(scene["outer_radius"])
    return thickness < 28.0 or float(scene["inner_radius"]) < 48.0 or clearance < 24.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
