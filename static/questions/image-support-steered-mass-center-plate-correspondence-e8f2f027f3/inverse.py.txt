"""Independent pixel-only inverse for the connected mass-centroid plate."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_NAVY = np.asarray((31, 76, 101), dtype=np.float32)
_AMBER = np.asarray((236, 163, 54), dtype=np.float32)
_VIOLET = np.asarray((172, 103, 194), dtype=np.float32)
_NEUTRAL = np.asarray((119, 164, 153), dtype=np.float32)
_BAY_W = 64
_BAY_RELS = {"R": (10, 20), "A": (86, 20), "B": (162, 20)}
_BAY_H = 132
_PATCH_BOX = (0, 12, 64, 104)
_TOL = 82.0
_MIN_PLATE_INK = 18000
_MAX_PLATE_INK = 20000


def _mask(array: np.ndarray, color: np.ndarray, tolerance: float = _TOL) -> np.ndarray:
    return np.linalg.norm(array - color, axis=2) < tolerance


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue: deque[tuple[int, int]] = deque([(y, x)])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = cy + dy, cx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            component = np.zeros_like(mask, dtype=bool)
            ys, xs = zip(*points, strict=True)
            component[np.asarray(ys), np.asarray(xs)] = True
            found.append(component)
    return found


def _plate_bbox(array: np.ndarray) -> tuple[int, int]:
    components = _components(_mask(array, _NAVY), 7000)
    if len(components) != 1:
        raise ValueError("expected one connected navy plate")
    plate_ink = int(components[0].sum())
    # The carrier is deliberately part of the observable support. Removing a
    # grid cell from the central plate must abstain rather than silently let a
    # centroid-only partial image through.
    if not _MIN_PLATE_INK <= plate_ink <= _MAX_PLATE_INK:
        raise ValueError("connected plate area is incomplete")
    ys, xs = np.nonzero(components[0])
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    if not (220 <= x1 - x0 + 1 <= 245 and 185 <= y1 - y0 + 1 <= 210):
        raise ValueError("connected plate has unexpected dimensions")
    return x0, y0


def _centroid_x(mask: np.ndarray, x0: int, y0: int, x1: int, y1: int) -> float:
    crop = mask[y0:y1, x0:x1]
    ys, xs = np.nonzero(crop)
    if len(xs) < 250:
        raise ValueError("bay patch has too little filled area")
    if int(xs.max() - xs.min()) < 25 or int(ys.max() - ys.min()) < 30:
        raise ValueError("bay patch has insufficient regional extent")
    return float(xs.mean())


def _bay_centroid(array: np.ndarray, plate_x: int, plate_y: int, slot: str) -> float:
    rx, ry = _BAY_RELS[slot]
    bx, by = plate_x + rx, plate_y + ry
    x0, y0, x1, y1 = _PATCH_BOX
    return _centroid_x(_mask(array, _NAVY), bx + x0, by + y0, bx + x1, by + y1)


def _nearest_name(pixel: np.ndarray, palette: tuple[tuple[str, np.ndarray], ...]) -> str:
    name, distance = min(
        ((name, float(np.linalg.norm(pixel - color))) for name, color in palette),
        key=lambda item: item[1],
    )
    if distance >= 105.0:
        raise ValueError("badge color is unreadable")
    return name


def _badge(array: np.ndarray, plate_x: int, plate_y: int, slot: str) -> str:
    rx, ry = _BAY_RELS[slot]
    bx, by = plate_x + rx, plate_y + ry
    pixel = array[by + 11, bx + 52]
    return _nearest_name(pixel, (("amber", _AMBER), ("violet", _VIOLET)))


def decision_from_image(image: Image.Image) -> str:
    """Recover the matching badge from PNG pixels and no latent scene data."""

    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    plate_x, plate_y = _plate_bbox(array)
    reference_x = _bay_centroid(array, plate_x, plate_y, "R")
    candidate_x = {slot: _bay_centroid(array, plate_x, plate_y, slot) for slot in ("A", "B")}
    errors = {slot: abs(reference_x - (_BAY_W - candidate_x[slot])) for slot in ("A", "B")}
    if abs(errors["A"] - errors["B"]) <= 3.0:
        raise ValueError("reflected centroid errors are tied or too close")
    best = "A" if errors["A"] < errors["B"] else "B"
    return _badge(array, plate_x, plate_y, best)
