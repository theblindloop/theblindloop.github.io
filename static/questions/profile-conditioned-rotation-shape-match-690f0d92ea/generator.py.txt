"""Deterministic latent scenes and Pillow raster for rotation-only shape search."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont

LABELS = ("A", "B", "C", "D")

# Each pattern is a connected, acyclic eleven-cell polyomino spanning a 5x5 box.
# Every entry has perimeter 24 and sorted row/column counts (1, 1, 2, 3, 4).
# Entries are distinct even after rotations and reflections.
PATTERN_CATALOG = (
    ("..#..", ".###.", "...##", "####.", "...#."),
    ("..###", "...#.", "####.", "#.#..", "..#.."),
    ("#....", "#....", "#..##", "####.", "...##"),
    ("..#..", "#.#..", "####.", "#..##", "#...."),
    ("####.", ".#.##", ".#...", ".##..", "..#.."),
    ("...##", "...#.", "##.#.", ".####", ".#..."),
    ("...#.", "####.", "...##", "....#", "..###"),
    ("###..", "#.###", "..#..", ".##..", ".#..."),
    ("...#.", "##.#.", ".####", "##...", ".#..."),
    ("..#..", "..#..", ".####", ".#..#", "###.."),
    ("....#", "##..#", ".####", ".#...", "##..."),
    (".####", "..#..", "###..", "#.#..", "#...."),
    ("..#.#", "..###", "..#..", "####.", "...#."),
    ("....#", "..###", "###.#", "#...#", "#...."),
    ("..##.", "..#..", "..#..", "####.", "#..##"),
    ("##...", "#....", "##..#", ".####", ".#..."),
    ("#....", "####.", "#..##", "..##.", "...#."),
    ("...#.", "..##.", "...#.", "..###", "###.#"),
    ("..#..", "####.", "...#.", "..###", ".##.."),
    ("..###", "...#.", "####.", "..#..", ".##.."),
    ("#....", "#....", "##.##", ".###.", "##..."),
    (".#...", "##...", ".####", "..#..", "###.."),
    ("#....", "####.", ".#.#.", ".#.##", ".#..."),
    ("####.", ".#.##", "...#.", "...##", "....#"),
)

THEMES = (
    ((25, 112, 146), (249, 248, 244), (238, 241, 240)),
    ((101, 67, 163), (249, 247, 244), (240, 238, 242)),
    ((163, 53, 89), (249, 248, 244), (243, 237, 239)),
    ((31, 130, 82), (249, 248, 244), (237, 242, 238)),
)


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _as_set(pattern: list[str] | tuple[str, ...]) -> frozenset[tuple[int, int]]:
    if len(pattern) != 5 or any(len(row) != 5 for row in pattern):
        raise ValueError("patterns must be five by five")
    cells = frozenset(
        (row, column)
        for row, values in enumerate(pattern)
        for column, value in enumerate(values)
        if value == "#"
    )
    if any(value not in ".#" for row in pattern for value in row):
        raise ValueError("pattern contains an unknown cell token")
    return cells


def _as_rows(cells: frozenset[tuple[int, int]]) -> list[str]:
    return [
        "".join("#" if (row, column) in cells else "." for column in range(5))
        for row in range(5)
    ]


def _rotate_cells(cells: frozenset[tuple[int, int]], turns: int = 1) -> frozenset[tuple[int, int]]:
    result = cells
    for _ in range(int(turns) % 4):
        result = frozenset((column, 4 - row) for row, column in result)
    return result


def _rotate_pattern(pattern: list[str] | tuple[str, ...], turns: int) -> list[str]:
    return _as_rows(_rotate_cells(_as_set(pattern), turns))


def _mirror_pattern(pattern: list[str] | tuple[str, ...]) -> list[str]:
    return _as_rows(frozenset((row, 4 - column) for row, column in _as_set(pattern)))


def _match_score(reference: list[str] | tuple[str, ...], candidate: list[str] | tuple[str, ...]) -> int:
    wanted = _as_set(reference)
    observed = _as_set(candidate)
    return max(len(wanted & _rotate_cells(observed, turns)) for turns in range(4))


def _canonical_dihedral(pattern: list[str] | tuple[str, ...]) -> tuple[tuple[int, int], ...]:
    cells = _as_set(pattern)
    mirror = frozenset((row, 4 - column) for row, column in cells)
    forms = [_rotate_cells(cells, turn) for turn in range(4)]
    forms.extend(_rotate_cells(mirror, turn) for turn in range(4))
    return min(tuple(sorted(form)) for form in forms)


def sample_scene(seed: int) -> dict:
    """Sample a cue-balanced scene; consecutive seeds cycle answer A/B/C/D."""
    value = int(seed)
    rng = random.Random(value ^ 0xA17E5C3)
    answer_index = value % 4
    base = PATTERN_CATALOG[rng.randrange(len(PATTERN_CATALOG))]
    reference = _rotate_pattern(base, rng.randrange(4))
    reference_class = _canonical_dihedral(reference)

    eligible = [
        pattern
        for pattern in PATTERN_CATALOG
        if _canonical_dihedral(pattern) != reference_class
        and _match_score(reference, pattern) <= 9
    ]
    rng.shuffle(eligible)
    if len(eligible) < 2:
        raise RuntimeError("pattern catalog has insufficient balanced distractors")

    distractors = [
        _rotate_pattern(_mirror_pattern(reference), rng.randrange(4)),
        _rotate_pattern(eligible[0], rng.randrange(4)),
        _rotate_pattern(eligible[1], rng.randrange(4)),
    ]
    rng.shuffle(distractors)
    target = _rotate_pattern(reference, rng.randrange(4))
    candidates: list[list[str]] = []
    for index in range(4):
        candidates.append(target if index == answer_index else distractors.pop())

    ink, background, card = THEMES[(value // 4) % len(THEMES)]
    scene = {
        "width": 840,
        "height": 760,
        "reference_pattern": reference,
        "candidate_patterns": candidates,
        "cell_size": 22 + ((value // 16) % 5),
        "ink_rgb": list(ink),
        "background_rgb": list(background),
        "card_rgb": list(card),
        "x_shift": rng.randint(-8, 8),
        "y_shift": rng.randint(-3, 3),
    }
    if margin(scene) < 2:
        raise RuntimeError("sampled distractor violates the runner-up margin")
    return scene


def _draw_pattern(
    draw: ImageDraw.ImageDraw,
    pattern: list[str],
    origin: tuple[int, int],
    cell_size: int,
    ink: tuple[int, int, int],
    scale: int,
) -> None:
    gap = 5
    ox, oy = origin
    for row, values in enumerate(pattern):
        for column, value in enumerate(values):
            if value != "#":
                continue
            x0 = (ox + column * (cell_size + gap)) * scale
            y0 = (oy + row * (cell_size + gap)) * scale
            x1 = x0 + cell_size * scale
            y1 = y0 + cell_size * scale
            draw.rounded_rectangle((x0, y0, x1, y1), radius=3 * scale, fill=ink)


def render(scene: dict) -> Image.Image:
    """Render the visible reference and four equally styled candidate cards."""
    scale = 3
    width = int(scene["width"])
    height = int(scene["height"])
    reference = list(scene["reference_pattern"])
    candidates = [list(pattern) for pattern in scene["candidate_patterns"]]
    cell_size = int(scene["cell_size"])
    ink = tuple(scene["ink_rgb"])
    background = tuple(scene["background_rgb"])
    card = tuple(scene["card_rgb"])
    x_shift = int(scene["x_shift"])
    y_shift = int(scene["y_shift"])

    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    neutral = (45, 47, 49)
    rule = (135, 138, 138)

    draw.text((width * scale // 2, 25 * scale), "FIND THE ROTATED MATCH",
              fill=neutral, font=_font(25 * scale), anchor="mm")
    draw.text((width * scale // 2, 58 * scale), "ROTATIONS COUNT  •  MIRROR IMAGES DO NOT",
              fill=neutral, font=_font(15 * scale), anchor="mm")
    draw.line((40 * scale, 78 * scale, (width - 40) * scale, 78 * scale),
              fill=rule, width=2 * scale)

    def card_box(box: tuple[int, int, int, int]) -> None:
        draw.rounded_rectangle(tuple(value * scale for value in box), radius=14 * scale,
                               fill=card, outline=rule, width=2 * scale)

    card_box((280 + x_shift, 89 + y_shift, 560 + x_shift, 280 + y_shift))
    draw.text(((420 + x_shift) * scale, (105 + y_shift) * scale), "REFERENCE",
              fill=neutral, font=_font(14 * scale), anchor="mm")

    pattern_extent = 5 * cell_size + 4 * 5
    reference_origin = (
        420 + x_shift - pattern_extent // 2,
        121 + y_shift,
    )
    _draw_pattern(draw, reference, reference_origin, cell_size, ink, scale)

    boxes = (
        (35 + x_shift, 300 + y_shift, 405 + x_shift, 505 + y_shift),
        (435 + x_shift, 300 + y_shift, 805 + x_shift, 505 + y_shift),
        (35 + x_shift, 530 + y_shift, 405 + x_shift, 735 + y_shift),
        (435 + x_shift, 530 + y_shift, 805 + x_shift, 735 + y_shift),
    )
    centers = ((220, 400), (620, 400), (220, 630), (620, 630))
    for index, (pattern, box, center) in enumerate(zip(candidates, boxes, centers)):
        card_box(box)
        label_x = box[0] + 28
        label_y = box[1] + 24
        draw.ellipse(((label_x - 18) * scale, (label_y - 18) * scale,
                      (label_x + 18) * scale, (label_y + 18) * scale),
                     fill=(255, 255, 255), outline=neutral, width=2 * scale)
        draw.text((label_x * scale, label_y * scale), LABELS[index], fill=neutral,
                  font=_font(20 * scale), anchor="mm")
        origin = (
            center[0] + x_shift - pattern_extent // 2,
            center[1] + y_shift - pattern_extent // 2 + 8,
        )
        _draw_pattern(draw, pattern, origin, cell_size, ink, scale)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    scores = [
        _match_score(scene["reference_pattern"], pattern)
        for pattern in scene["candidate_patterns"]
    ]
    winners = [index for index, score in enumerate(scores) if score == max(scores)]
    if len(winners) != 1 or scores[winners[0]] != 11:
        raise ValueError("scene lacks one exact rotation-only match")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    scores = sorted(
        (_match_score(scene["reference_pattern"], pattern)
         for pattern in scene["candidate_patterns"]),
        reverse=True,
    )
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    return int(scene["cell_size"]) < 20 or margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
