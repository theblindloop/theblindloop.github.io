"""Deterministic long-trace renderer; analytic labels use latent turn order only."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

BACKGROUND = (252, 252, 248)
HALO = (164, 194, 220)
TRACE = (23, 52, 93)
START = (24, 165, 88)
FINISH = (220, 63, 69)
WHITE = (255, 255, 255)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    first = 1 if rng.randrange(2) else -1
    signs = [first if i % 2 == 0 else -first for i in range(8)]
    if int(seed) % 2 == 0:
        # Flip one non-edge sign. Its location varies, while every odd seed is
        # a fully alternating negative example.
        signs[rng.randrange(1, 7)] *= -1
    return {
        "canvas_size": 640,
        "center": [320 + rng.randint(-9, 9), 320 + rng.randint(-9, 9)],
        "rotation": rng.randrange(4),
        "reverse": bool(rng.randrange(2)),
        "mirror": bool(rng.randrange(2)),
        "turn_signs": signs,
        "shape_variant": rng.randrange(3),
    }


def _cross(a: tuple[float, float], b: tuple[float, float]) -> float:
    return a[0] * b[1] - a[1] * b[0]


def _template(signs: list[int], variant: int) -> list[tuple[float, float]]:
    dx = (55.0, 57.0, 59.0)[int(variant)]
    delta = (0.54, 0.58, 0.62)[int(variant)]
    slopes = [-int(signs[0]) * delta / 2.0]
    for sign in signs:
        slopes.append(slopes[-1] + int(sign) * delta)
    ys = [0.0]
    for slope in slopes:
        ys.append(ys[-1] + dx * slope)
    ymid = (max(ys) + min(ys)) / 2.0
    xs = [(i - 4.5) * dx for i in range(10)]
    points = [(x, y - ymid) for x, y in zip(xs, ys, strict=True)]
    observed = []
    for i in range(1, 9):
        u = (points[i][0] - points[i-1][0], points[i][1] - points[i-1][1])
        v = (points[i+1][0] - points[i][0], points[i+1][1] - points[i][1])
        observed.append(1 if _cross(u, v) > 0 else -1)
    if observed != [int(v) for v in signs]:
        raise AssertionError((observed, signs))
    return points


def _points(scene: dict) -> list[tuple[int, int]]:
    points = _template(scene["turn_signs"], scene["shape_variant"])
    if scene["mirror"]:
        points = [(x, -y) for x, y in points]
    if scene["reverse"]:
        points = list(reversed(points))
    turns = int(scene["rotation"]) % 4
    cx, cy = (int(v) for v in scene["center"])
    output = []
    for x, y in points:
        for _ in range(turns):
            x, y = -y, x
        output.append((int(round(cx + x)), int(round(cy + y))))
    return output


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas_size"])
    image = Image.new("RGB", (size, size), BACKGROUND)
    draw = ImageDraw.Draw(image)
    points = _points(scene)
    draw.line(points, fill=HALO, width=13, joint="curve")
    draw.line(points, fill=TRACE, width=5, joint="curve")
    for point, color in ((points[0], START), (points[-1], FINISH)):
        x, y = point
        draw.ellipse((x-12, y-12, x+12, y+12), fill=WHITE)
        draw.ellipse((x-9, y-9, x+9, y+9), fill=color)
    return image


def analytic_gold(scene: dict) -> str:
    signs = [int(v) for v in scene["turn_signs"]]
    return "yes" if any(a == b for a, b in zip(signs, signs[1:])) else "no"


def margin(scene: dict) -> float:
    points = _template(scene["turn_signs"], scene["shape_variant"])
    values = []
    for i in range(1, 9):
        u = (points[i][0]-points[i-1][0], points[i][1]-points[i-1][1])
        v = (points[i+1][0]-points[i][0], points[i+1][1]-points[i][1])
        values.append(abs(_cross(u, v)) / (math.hypot(*u) * math.hypot(*v)))
    return round(min(values), 12)


def is_quarantined(scene: dict) -> bool:
    size = int(scene["canvas_size"])
    edge = min(min(x, y, size-1-x, size-1-y) for x, y in _points(scene))
    return margin(scene) < 0.20 or edge < 30


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
