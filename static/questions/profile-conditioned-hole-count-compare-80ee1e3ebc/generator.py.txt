"""Hole-count-compare world renderer.

The image shows two separate filled colored shapes on a white background, one
on the left and one on the right. Each shape is a closed disk that may contain
a small number of white "holes": pockets of background-colored pixels fully
surrounded by that shape's ink, with no opening to the outside white background.

Decision : "left" if the left shape contains strictly more fully-enclosed white
holes than the right shape, else "right". The two shapes always have different
hole counts so the comparison is never ambiguous.

The scene is a plain dict so the verifier's AST reachability audit is free of
side effects. analytic_gold reads only the two nested shape records, whose
"holes" lists are the sole label-bearing content.

    z = {
        "left":   {"color": "red",   "center": [cx, cy], "radius": R, "holes": [[hx,hy,hr], ...]},
        "right":  {"color": "blue",  "center": [cx, cy], "radius": R, "holes": [[hx,hy,hr], ...]},
        "width": 400,
        "height": 400,
        "background": "white",
    }
"""

from __future__ import annotations

import math
import random
from collections.abc import Iterator

from PIL import Image, ImageDraw

RENDERER_VERSION = "hole_count_compare-0.4.0"

PALETTE = [
    "red",
    "mediumblue",
    "darkgreen",
    "purple",
    "teal",
    "crimson",
    "maroon",
    "darkorange",
]

# Robustness margins (pixels). Kept comfortably above ordinary antialiasing so
# a wall can never open a leak or pinch a hole shut at pixel scale.
WALL_MIN = 9.0
HOLE_RADIUS_MIN = 9.0
HOLES_PER_SHAPE_MAX = 3

WIDTH = 400
HEIGHT = 400
LEFT_X = 110.0
RIGHT_X = 290.0
CENTRE_Y = 200.0


def sample_scene(rng: random.Random) -> dict:
    """Sample a scene with two shapes of different hole counts."""
    left_color = rng.choice(PALETTE)
    right_color = rng.choice([c for c in PALETTE if c != left_color])
    left_radius = rng.uniform(55.0, 80.0)
    right_radius = rng.uniform(55.0, 80.0)

    left_holes = _sample_holes(rng, rng.randint(0, HOLES_PER_SHAPE_MAX), LEFT_X, left_radius)
    right_count = rng.randint(0, HOLES_PER_SHAPE_MAX)
    while right_count == len(left_holes):
        right_count = rng.randint(0, HOLES_PER_SHAPE_MAX)
    right_holes = _sample_holes(rng, right_count, RIGHT_X, right_radius)

    return {
        "left": {
            "color": left_color,
            "center": [LEFT_X, CENTRE_Y],
            "radius": left_radius,
            "holes": left_holes,
        },
        "right": {
            "color": right_color,
            "center": [RIGHT_X, CENTRE_Y],
            "radius": right_radius,
            "holes": right_holes,
        },
        "width": WIDTH,
        "height": HEIGHT,
        "background": "white",
    }


def _sample_holes(
    rng: random.Random, n: int, cx: float, radius: float
) -> list[list[float]]:
    holes: list[list[float]] = []
    for _ in range(400):
        if len(holes) >= n:
            break
        hr = rng.uniform(HOLE_RADIUS_MIN, 16.0)
        max_d = radius - hr - WALL_MIN
        if max_d < 4.0:
            break
        angle = rng.uniform(0.0, 2.0 * math.pi)
        d = rng.uniform(0.0, max_d)
        hx = cx + d * math.cos(angle)
        hy = CENTRE_Y + d * math.sin(angle)
        ok = True
        for (ox, oy, orad) in holes:
            if math.hypot(hx - ox, hy - oy) - hr - orad < WALL_MIN - 1.0:
                ok = False
                break
        if ok:
            holes.append([hx, hy, hr])
    return holes


def render(scene: dict, supersample: int = 4) -> Image.Image:
    """Render I = R(z). Deterministic pure function of the scene dict."""
    width = int(scene["width"])
    height = int(scene["height"])
    background = scene["background"]
    ss = supersample
    img = Image.new("RGB", (width * ss, height * ss), background)
    draw = ImageDraw.Draw(img)
    for rec in (scene["left"], scene["right"]):
        (cx, cy) = (float(rec["center"][0]), float(rec["center"][1]))
        radius = float(rec["radius"])
        color = rec["color"]
        box = [(cx - radius) * ss, (cy - radius) * ss, (cx + radius) * ss, (cy + radius) * ss]
        draw.ellipse(box, fill=color)
        for (hx, hy, hr) in rec["holes"]:
            hbox = [(hx - hr) * ss, (hy - hr) * ss, (hx + hr) * ss, (hy + hr) * ss]
            draw.ellipse(hbox, fill=background)
    if ss > 1:
        img = img.resize((width, height), Image.LANCZOS)
    return img


def analytic_gold(scene: dict) -> str:
    """Which shape has strictly more enclosed holes: left or right."""
    left_n = len(scene["left"]["holes"])
    right_n = len(scene["right"]["holes"])
    if left_n > right_n:
        return "left"
    return "right"


def margin(scene: dict) -> float:
    """Signed hole-count difference: left_count - right_count.

    Its sign is the decision; its magnitude is how many holes the two shapes
    differ by (a discrete verifiability margin). Zero never occurs by
    construction but is the quarantine trigger.
    """
    left_n = len(scene["left"]["holes"])
    right_n = len(scene["right"]["holes"])
    return float(left_n - right_n)


def is_quarantined(scene: dict) -> bool:
    """Quarantine the no-decision case where the counts are equal.

    Counts are forced to differ during sampling, so practically this is always
    False; it guards against any degenerate by-construction equal-count scene.
    """
    return margin(scene) == 0.0


def latent_symmetries(scene: dict) -> Iterator[dict]:
    """No nontrivial render symmetry: every field changes the picture.

    Reordering each shape's hole list is a numerically identical parametrization
    (same pixels, same counts) but the world keeps a canonical order; declared
    not_applicable.
    """
    return iter(())
