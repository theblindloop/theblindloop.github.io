"""Independent pixel-only inverse arm for the hole-count-compare world.

Receives ONLY the final PNG. It never imports or reads the renderer, the
analytic scene, prompts, or any renderer state. It recovers the decision by:

1. marking "white-like" pixels (the background and any holes);
2. flood-filling the 4-connected white region reachable from the image border;
3. counting, on the LEFT half and the RIGHT half separately, the connected
   white components that the border flood cannot reach (enclosed holes).

Each shape occupies a separated image half, and the accumulated enclosed-white
pockets on a side are the holes of that shape. The side whose enclosed-hole
count is larger wins, which equals the analytic comparison on robust scenes.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "hole_count_compare-oracle-0.4.0"

_WHITE_THRESHOLD = 240

# Border flood reachability uses only the four cardinal neighbours.
_NEIGHBOURS = ((1, 0), (-1, 0), (0, 1), (0, -1))


def _is_white(arr: np.ndarray, threshold: int = _WHITE_THRESHOLD) -> np.ndarray:
    return arr.min(axis=-1) > threshold


def _border_reached(white: np.ndarray) -> np.ndarray:
    """Boolean mask of white pixels 4-connected to any border white pixel."""
    h, w = white.shape
    reached = np.zeros_like(white, dtype=bool)
    stack: list[tuple[int, int]] = []
    for y in (0, h - 1):
        for x in range(w):
            if white[y, x] and not reached[y, x]:
                reached[y, x] = True
                stack.append((x, y))
    for y in range(h):
        for x in (0, w - 1):
            if white[y, x] and not reached[y, x]:
                reached[y, x] = True
                stack.append((x, y))
    while stack:
        x, y = stack.pop()
        for dx, dy in _NEIGHBOURS:
            nx, ny = x + dx, y + dy
            if 0 <= nx < w and 0 <= ny < h and white[ny, nx] and not reached[ny, nx]:
                reached[ny, nx] = True
                stack.append((nx, ny))
    return reached


def _count_components(mask: np.ndarray) -> int:
    """Count 4-connected components of a boolean mask (BFS over true pixels)."""
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    count = 0
    for y in range(h):
        for x in range(w):
            if mask[y, x] and not seen[y, x]:
                count += 1
                seen[y, x] = True
                stack = [(x, y)]
                while stack:
                    cx, cy = stack.pop()
                    for dx, dy in _NEIGHBOURS:
                        nx, ny = cx + dx, cy + dy
                        if 0 <= nx < w and 0 <= ny < h and mask[ny, nx] and not seen[ny, nx]:
                            seen[ny, nx] = True
                            stack.append((nx, ny))
    return count


def decision_from_image(image: Image.Image) -> str:
    """Which shape has more enclosed holes: 'left' or 'right'? From pixels only."""
    arr = np.asarray(image.convert("RGB"))
    white = _is_white(arr)
    border = _border_reached(white)
    holes = white & ~border
    _, w = holes.shape
    mid = w // 2
    left_holes = holes[:, :mid]
    right_holes = holes[:, mid:]
    left_count = _count_components(left_holes)
    right_count = _count_components(right_holes)
    if left_count > right_count:
        return "left"
    return "right"
