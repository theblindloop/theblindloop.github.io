"""Pixel-only inverse arm for the tri-roster pip-consensus world.

Input: one rasterised RGB image plus the public question contract. This module
imports no renderer, prompt, generator, verifier or analytic-gold code, receives
no latent scene, and reads no coordinates supplied from outside the image.

Pipeline: segment the light roster panels, extract the coloured tokens that lie
inside them, read each token's pip count from the dark blobs it contains, then
solve the cross-roster correspondence by colour and report the single colour
whose pip count is shared by all three rosters.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "triroster_pip_consensus_bind-oracle-0.1.0"

_PANEL_REF = (250, 250, 248)
_TOKEN_REF = {
    "red": (206, 48, 44),
    "orange": (228, 124, 28),
    "yellow": (222, 196, 44),
    "green": (44, 142, 70),
    "blue": (46, 84, 196),
    "purple": (132, 54, 170),
}
_TOL = 12
_PANEL_MIN_AREA = 6000
_TOKEN_AREA = (900, 2600)
_PIP_AREA = (40, 220)
_PIP_MAX_LEVEL = 70
_ROSTERS = 3


class OracleAbstention(ValueError):
    """Raised when the image does not present a readable tri-roster scene."""


def _components(mask: np.ndarray) -> list[dict]:
    """Connected components (4-connectivity) of a boolean mask via row runs."""
    height, width = mask.shape
    parent: list[int] = []
    runs: list[tuple[int, int, int]] = []  # (row, start, end)

    def find(a: int) -> int:
        while parent[a] != a:
            parent[a] = parent[parent[a]]
            a = parent[a]
        return a

    def union(a: int, b: int) -> None:
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    previous: list[int] = []
    for row in range(height):
        line = mask[row].astype(np.int8)
        edges = np.flatnonzero(np.diff(np.concatenate(([0], line, [0]))))
        current: list[int] = []
        for start, end in zip(edges[0::2], edges[1::2]):
            index = len(runs)
            runs.append((row, int(start), int(end)))
            parent.append(index)
            current.append(index)
            for other in previous:
                _, ostart, oend = runs[other]
                if ostart < end and start < oend:
                    union(index, other)
        previous = current

    groups: dict[int, dict] = {}
    for index, (row, start, end) in enumerate(runs):
        root = find(index)
        entry = groups.setdefault(
            root, {"area": 0, "x0": width, "y0": height, "x1": 0, "y1": 0}
        )
        entry["area"] += end - start
        entry["x0"] = min(entry["x0"], start)
        entry["x1"] = max(entry["x1"], end)
        entry["y0"] = min(entry["y0"], row)
        entry["y1"] = max(entry["y1"], row + 1)
    return list(groups.values())


def _near(arr: np.ndarray, ref: tuple[int, int, int], tol: int = _TOL) -> np.ndarray:
    return (np.abs(arr - np.array(ref, dtype=np.int16)).max(axis=2) <= tol)


def _centre(box: dict) -> tuple[float, float]:
    return ((box["x0"] + box["x1"]) / 2.0, (box["y0"] + box["y1"]) / 2.0)


def _inside(box: dict, outer: dict) -> bool:
    cx, cy = _centre(box)
    return outer["x0"] <= cx <= outer["x1"] and outer["y0"] <= cy <= outer["y1"]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)

    panels = [c for c in _components(_near(arr, _PANEL_REF)) if c["area"] >= _PANEL_MIN_AREA]
    if len(panels) != _ROSTERS:
        raise OracleAbstention(f"found {len(panels)} roster panels, expected {_ROSTERS}")
    panels.sort(key=lambda p: (p["y0"], p["x0"]))

    dark = _components(arr.max(axis=2) <= _PIP_MAX_LEVEL)
    pip_blobs = [b for b in dark if _PIP_AREA[0] <= b["area"] <= _PIP_AREA[1]]

    tables: list[dict[str, int]] = [{} for _ in panels]
    for colour, ref in _TOKEN_REF.items():
        for blob in _components(_near(arr, ref)):
            if not _TOKEN_AREA[0] <= blob["area"] <= _TOKEN_AREA[1]:
                continue
            hosts = [i for i, panel in enumerate(panels) if _inside(blob, panel)]
            if len(hosts) != 1:
                continue
            index = hosts[0]
            pips = sum(1 for pip in pip_blobs if _inside(pip, blob))
            if not 1 <= pips <= 4:
                raise OracleAbstention(f"{colour} token carries {pips} readable pips")
            if colour in tables[index]:
                raise OracleAbstention(f"roster {index} carries two {colour} tokens")
            tables[index][colour] = pips

    for index, table in enumerate(tables):
        if set(table) != set(_TOKEN_REF):
            missing = sorted(set(_TOKEN_REF) - set(table))
            raise OracleAbstention(f"roster {index} is missing tokens {missing}")

    hits = [
        (colour, tables[0][colour])
        for colour in sorted(_TOKEN_REF)
        if tables[0][colour] == tables[1][colour] == tables[2][colour]
    ]
    if len(hits) != 1:
        raise OracleAbstention(f"{len(hits)} colours agree on all three rosters")
    colour, pips = hits[0]
    return f"{colour}:{pips}"
