"""Renderer for the tri-roster pip-consensus world.

Three ROSTER panels sit at three widely separated corners of one board. Every
roster carries exactly one token of each of six colours, and every token carries
between one and four dark pips. Exactly one colour is given the same pip count
on all three rosters; that colour, and its pip count, is the answer.

The mechanism is a three-way consensus over an attribute vector indexed by
colour. Any two rosters agree on at least two colours, so no pair of rosters can
pick the answer: all three widely separated rosters are jointly necessary.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "triroster_pip_consensus_bind-0.1.0"

CANVAS = (640, 480)
BOARD_RGB = (226, 216, 196)
PANEL_RGB = (250, 250, 248)
BORDER_RGB = (52, 58, 66)
PIP_RGB = (20, 22, 26)

COLOURS: tuple[str, ...] = ("red", "orange", "yellow", "green", "blue", "purple")
COLOUR_RGB: dict[str, tuple[int, int, int]] = {
    "red": (206, 48, 44),
    "orange": (228, 124, 28),
    "yellow": (222, 196, 44),
    "green": (44, 142, 70),
    "blue": (46, 84, 196),
    "purple": (132, 54, 170),
}

PANEL_W, PANEL_H = 176, 120
PANEL_SLOTS: tuple[tuple[int, int], ...] = (
    (24, 24),
    (440, 24),
    (24, 336),
    (440, 336),
)
PAD = 12
TOKEN = 40
TOKEN_GAP = 16
PIP = 10
PIP_OFF = 7
PIP_STEP = 16

MARK = 12
MARK_COUNT = 18
SAFE_CLEARANCE = 14.0
MARK_SPACING = 8
QUARANTINE_MARGIN = 4.0
PIP_VALUES = (1, 2, 3, 4)


def _panel_rect(slot: int) -> tuple[int, int, int, int]:
    x, y = PANEL_SLOTS[slot]
    return (x, y, x + PANEL_W, y + PANEL_H)


def _token_rect(slot: int, cell: int) -> tuple[int, int, int, int]:
    x, y = PANEL_SLOTS[slot]
    col, row = cell % 3, cell // 3
    tx = x + PAD + col * (TOKEN + TOKEN_GAP)
    ty = y + PAD + row * (TOKEN + TOKEN_GAP)
    return (tx, ty, tx + TOKEN, ty + TOKEN)


def _rect_distance(a: tuple[int, int, int, int], b: tuple[int, int, int, int]) -> float:
    dx = max(b[0] - a[2], a[0] - b[2], 0)
    dy = max(b[1] - a[3], a[1] - b[3], 0)
    return math.hypot(dx, dy)


def _pip_tables(rng: random.Random, c_star: str, p_star: int) -> list[dict[str, int]] | None:
    """Three pip tables with a unique three-way consensus at ``c_star``."""
    others = [c for c in COLOURS if c != c_star]
    tables: list[dict[str, int]] = []
    for _ in range(3):
        pool = list(PIP_VALUES) + [rng.choice(PIP_VALUES), rng.choice(PIP_VALUES)]
        rng.shuffle(pool)
        pool.remove(p_star)
        order = list(others)
        rng.shuffle(order)
        table = {c_star: p_star}
        for colour, value in zip(order, pool):
            table[colour] = value
        tables.append(table)

    for colour in others:
        if tables[0][colour] == tables[1][colour] == tables[2][colour]:
            return None
    decoys = 0
    for i, j in ((0, 1), (0, 2), (1, 2)):
        agree = [c for c in others if tables[i][c] == tables[j][c]]
        if not 1 <= len(agree) <= 3:
            return None
        decoys += len(agree)
    counts = {v: 0 for v in PIP_VALUES}
    for table in tables:
        for value in table.values():
            counts[value] += 1
    rest = [counts[v] for v in PIP_VALUES if v != p_star]
    if not (max(rest) >= counts[p_star] and min(rest) <= counts[p_star]):
        return None
    for table in tables:
        if set(table.values()) != set(PIP_VALUES):
            return None
    return tables + [{"_decoys": decoys}]


def sample_scene(seed: int) -> dict:
    rng = random.Random((seed * 2654435761) % (2 ** 32) ^ 0x5AB1CE)
    c_star = COLOURS[seed % len(COLOURS)]
    p_star = 1 + ((seed // len(COLOURS)) % 4)

    tables = None
    for _ in range(6000):
        tables = _pip_tables(rng, c_star, p_star)
        if tables is not None:
            break
    if tables is None:  # pragma: no cover - exhaustive search always succeeds
        raise RuntimeError(f"could not sample pip tables for seed {seed}")
    decoys = int(tables[-1]["_decoys"])
    tables = tables[:3]

    slots = sorted(rng.sample(range(len(PANEL_SLOTS)), 3))
    placements: list[list[str]] = []
    while True:
        placements = []
        for _ in range(3):
            order = list(COLOURS)
            rng.shuffle(order)
            placements.append(order)
        star_cells = {order.index(c_star) for order in placements}
        if len(star_cells) > 1:
            break

    panels = []
    for slot, table, order in zip(slots, tables, placements):
        tokens = [
            {"cell": cell, "colour": colour, "pips": table[colour]}
            for cell, colour in enumerate(order)
        ]
        panels.append({"slot": slot, "tokens": tokens})

    rects = [_panel_rect(slot) for slot in slots]
    mode = ((seed * 2654435761) % (2 ** 32) >> 8) % 6
    special = {4: 6, 5: 3}.get(mode)

    marks: list[dict] = []
    if special is not None:
        for attempt in range(64):
            rect = rects[rng.randrange(3)]
            side = rng.randrange(4)
            if side == 0:
                mx, my = rect[0] - special - MARK, rng.randint(rect[1], rect[3] - MARK)
            elif side == 1:
                mx, my = rect[2] + special, rng.randint(rect[1], rect[3] - MARK)
            elif side == 2:
                mx, my = rng.randint(rect[0], rect[2] - MARK), rect[1] - special - MARK
            else:
                mx, my = rng.randint(rect[0], rect[2] - MARK), rect[3] + special
            box = (mx, my, mx + MARK, my + MARK)
            if not (8 <= mx and 8 <= my and box[2] <= CANVAS[0] - 8 and box[3] <= CANVAS[1] - 8):
                continue
            if min(_rect_distance(box, r) for r in rects) < special - 0.5:
                continue
            marks.append({"x": mx, "y": my})
            break

    grid = [
        (x, y)
        for x in range(8, CANVAS[0] - MARK - 7, 20)
        for y in range(8, CANVAS[1] - MARK - 7, 20)
    ]
    rng.shuffle(grid)
    for mx, my in grid:
        if len(marks) >= MARK_COUNT:
            break
        box = (mx, my, mx + MARK, my + MARK)
        if min(_rect_distance(box, r) for r in rects) < SAFE_CLEARANCE:
            continue
        if any(
            _rect_distance(box, (m["x"], m["y"], m["x"] + MARK, m["y"] + MARK)) < MARK_SPACING
            for m in marks
        ):
            continue
        marks.append({"x": mx, "y": my})

    marks.sort(key=lambda m: (m["y"], m["x"]))
    for index, mark in enumerate(marks):
        mark["colour"] = COLOURS[index % len(COLOURS)]

    clearance = min(
        min(
            _rect_distance((m["x"], m["y"], m["x"] + MARK, m["y"] + MARK), r)
            for r in rects
        )
        for m in marks
    )
    return {
        "canvas": list(CANVAS),
        "panels": panels,
        "clutter": marks,
        "clearance": float(clearance),
        "decoy_agreements": decoys,
    }


def _consensus(scene: dict) -> tuple[str, int]:
    tables = [{t["colour"]: t["pips"] for t in p["tokens"]} for p in scene["panels"]]
    hits = [
        (colour, tables[0][colour])
        for colour in COLOURS
        if tables[0][colour] == tables[1][colour] == tables[2][colour]
    ]
    if len(hits) != 1:
        raise ValueError(f"scene has {len(hits)} consensus colours, expected exactly one")
    return hits[0]


def analytic_gold(scene: dict) -> str:
    colour, pips = _consensus(scene)
    return f"{colour}:{pips}"


def margin(scene: dict) -> float:
    tables = [{t["colour"]: t["pips"] for t in p["tokens"]} for p in scene["panels"]]
    star, _ = _consensus(scene)
    runner_up = 1
    for colour in COLOURS:
        if colour == star:
            continue
        values = [table[colour] for table in tables]
        best = max(values.count(v) for v in set(values))
        runner_up = max(runner_up, best)
    structural = 8.0 * (3 - runner_up) - 0.5 * max(0, int(scene["decoy_agreements"]) - 3)
    return float(min(float(scene["clearance"]), structural, 12.0))


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_MARGIN)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    import copy

    out: list[tuple[str, dict]] = []

    reversed_panels = copy.deepcopy(scene)
    reversed_panels["panels"] = list(reversed(reversed_panels["panels"]))
    out.append(("roster_list_reversal", reversed_panels))

    rotated_tokens = copy.deepcopy(scene)
    for panel in rotated_tokens["panels"]:
        panel["tokens"] = panel["tokens"][2:] + panel["tokens"][:2]
    out.append(("token_list_rotation", rotated_tokens))

    reversed_clutter = copy.deepcopy(scene)
    reversed_clutter["clutter"] = list(reversed(reversed_clutter["clutter"]))
    out.append(("clutter_list_reversal", reversed_clutter))
    return out


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", tuple(scene["canvas"]), BOARD_RGB)
    draw = ImageDraw.Draw(image)

    for panel in sorted(scene["panels"], key=lambda p: p["slot"]):
        slot = panel["slot"]
        x0, y0, x1, y1 = _panel_rect(slot)
        draw.rectangle((x0, y0, x1 - 1, y1 - 1), fill=PANEL_RGB, outline=BORDER_RGB, width=3)
        for token in sorted(panel["tokens"], key=lambda t: t["cell"]):
            tx0, ty0, tx1, ty1 = _token_rect(slot, token["cell"])
            draw.rectangle((tx0, ty0, tx1 - 1, ty1 - 1), fill=COLOUR_RGB[token["colour"]])
            for index in range(int(token["pips"])):
                px = tx0 + PIP_OFF + (index % 2) * PIP_STEP
                py = ty0 + PIP_OFF + (index // 2) * PIP_STEP
                draw.rectangle((px, py, px + PIP - 1, py + PIP - 1), fill=PIP_RGB)

    for mark in sorted(scene["clutter"], key=lambda m: (m["y"], m["x"], m["colour"])):
        mx, my = mark["x"], mark["y"]
        draw.rectangle((mx, my, mx + MARK - 1, my + MARK - 1), fill=COLOUR_RGB[mark["colour"]])
    return image
