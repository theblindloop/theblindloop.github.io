"""Route-reach color world renderer.

Latent scene: a single continuous dark route (a connected polyline of waypoints)
begins at an orange start marker and ends exactly at one colored destination
marker. Two other colored destination markers are present but disconnected from
the route. The question asks which destination color the route connects to.

The decisive evidence is the connected route: the answer is the color of the
destination that lies in the same connected inked component as the start. The
analytic gold derives the answer from the visible route/destination geometry.

```text
I = R(z)
reach = color of destination whose center equals route[-1]
```
"""

from __future__ import annotations

import dataclasses
import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "route_reach_color-0.1.0"

CANVAS = (512, 512)
BG = (255, 255, 255)
ROUTE_COLOR = (25, 25, 25)
START_COLOR = (245, 150, 30)  # orange start marker
DEST_COLORS = {
    "red": (214, 48, 49),
    "blue": (41, 82, 214),
    "green": (38, 166, 91),
}
DEST_RADIUS = 13.0
START_RADIUS = 12.0
ROUTE_WIDTH = 6

# The route and all markers are confined to this central box so that the oracle's
# decisive support is a connected thin trace spanning a regional area of the
# canvas (never crossing opposite edges and never collapsing to a focal point).
# A 232px box (3.6/8 of the canvas) keeps the support's normalized span in the
# regional band (roughly 0.5) while the route is a thin near-diagonal trace.
BOX = (140, 140, 372, 372)

# Minimum center-to-center distance between the route centreline and any
# non-answer destination, guaranteeing the disconnected markers never touch the
# route even after anti-aliasing.
MIN_CLEARANCE = 42.0


@dataclasses.dataclass(frozen=True)
class RouteScene:
    start: tuple  # (x, y) start marker center
    route: list  # ordered list of (x, y): route[0]==start, route[-1]==reached dest center
    destinations: list  # list of dicts {"pos":(x,y), "color":str}
    width: int = 512
    height: int = 512


def _dist(p, q):
    return math.dist(p, q)


def _segment_clearance(a, b, center):
    """Distance from an infinite* segment endpoint-restricted to a point.

    Restricts to the finite segment [a, b].
    """
    ax, ay = a
    bx, by = b
    px, py = center
    dx, dy = bx - ax, by - ay
    length_sq = dx * dx + dy * dy
    if length_sq == 0:
        return math.dist(a, center)
    t = max(0.0, min(1.0, ((px - ax) * dx + (py - ay) * dy) / length_sq))
    nearest = (ax + t * dx, ay + t * dy)
    return math.dist(nearest, center)


def _sample_route(rng, start, dest_center, unreached_centers):
    """Sample a thin near-diagonal polyline from start to dest_center inside BOX.

    Interior knots are placed close to the start->destination axis with modest
    perpendicular offsets, so the rendered route stays a thin elongated trace
    (pathlike) spanning the regional box. Each segment keeps MIN_CLEARANCE from
    every unreached center.
    """
    x0, y0, x1, y1 = BOX
    sx, sy = start
    dx, dy = dest_center
    axis_len = math.dist(start, dest_center)
    for _ in range(8000):
        n_knots = rng.randint(2, 3)
        waypoints = [start]
        prev = start
        for _ in range(n_knots):
            # advance along the axis a random distance, then offset perpendicular
            t = rng.uniform(0.30, 0.75)
            bx = sx + (dx - sx) * t
            by = sy + (dy - sy) * t
            # perpendicular direction
            ax, ay = dx - sx, dy - sy
            perp = (-ay / axis_len, ax / axis_len)
            offset = rng.uniform(-0.12, 0.12) * axis_len
            kx = bx + perp[0] * offset
            ky = by + perp[1] * offset
            kx = min(max(kx, x0 + 8), x1 - 8)
            ky = min(max(ky, y0 + 8), y1 - 8)
            if math.dist(prev, (kx, ky)) < 24:
                continue
            waypoints.append((kx, ky))
            prev = (kx, ky)
        waypoints.append(dest_center)
        ok = True
        for i in range(len(waypoints) - 1):
            a, b = waypoints[i], waypoints[i + 1]
            for center in unreached_centers:
                if _segment_clearance(a, b, center) < MIN_CLEARANCE:
                    ok = False
                    break
            if not ok:
                break
        if ok:
            return waypoints
    raise RuntimeError("could not sample a clear route")


def _jitter_inside(rng, x, y, slack=14):
    return (x + rng.uniform(-slack, slack), y + rng.uniform(-slack, slack))


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    x0, y0, x1, y1 = BOX
    corners = [(x0, y0), (x1, y0), (x0, y1), (x1, y1)]
    first = rng.randrange(4)
    opposite = (first + 2) % 4
    start = _jitter_inside(rng, *corners[first])
    dest_center = _jitter_inside(rng, *corners[opposite])

    colors = ["red", "blue", "green"]
    rng.shuffle(colors)
    reached_color = colors[0]
    unreached_colors = colors[1:]

    # Place unreached destinations at the two remaining corners, jittered inward
    # so their discs stay inside the box and clear of the diagonal corridor.
    other_corners = [i for i in range(4) if i not in (first, opposite)]
    unreached_centers = [_jitter_inside(rng, *corners[i], slack=18) for i in other_corners]

    route = _sample_route(rng, start, dest_center, unreached_centers)

    destinations = [{"pos": (dest_center[0], dest_center[1]), "color": reached_color}]
    for center, color in zip(unreached_centers, unreached_colors, strict=True):
        destinations.append({"pos": (center[0], center[1]), "color": color})
    rng.shuffle(destinations)

    return {
        "width": 512,
        "height": 512,
        "start": start,
        "route": route,
        "destinations": destinations,
    }


def render(scene) -> Image.Image:
    w = int(scene["width"])
    h = int(scene["height"])
    ss = 2
    img = Image.new("RGB", (w * ss, h * ss), BG)
    draw = ImageDraw.Draw(img)

    def S(p):
        return (p[0] * ss, p[1] * ss)

    # start marker
    sx, sy = S(scene["start"])
    draw.ellipse(
        [sx - START_RADIUS * ss, sy - START_RADIUS * ss, sx + START_RADIUS * ss, sy + START_RADIUS * ss],
        fill=START_COLOR,
    )
    # destination markers (drawn before the route so the route can overshoot into
    # the reached disc and guarantee pixel connectivity despite anti-aliasing)
    for d in scene["destinations"]:
        cx, cy = S(d["pos"])
        color = DEST_COLORS[d["color"]]
        draw.ellipse(
            [cx - DEST_RADIUS * ss, cy - DEST_RADIUS * ss, cx + DEST_RADIUS * ss, cy + DEST_RADIUS * ss],
            fill=color,
        )

    # route polyline drawn on top so it overlaps the start disc and the reached
    # destination disc, keeping the whole decision support one connected ink path
    pts = [S(p) for p in scene["route"]]
    draw.line(pts, fill=ROUTE_COLOR, width=ROUTE_WIDTH * ss, joint="curve")

    img = img.resize((w, h), Image.LANCZOS)
    return img


def _reached_color(scene) -> str:
    end = scene["route"][-1]
    for d in scene["destinations"]:
        if tuple(d["pos"]) == tuple(end):
            return d["color"]
    raise ValueError("route endpoint does not match any destination")


def analytic_gold(scene) -> str:
    return _reached_color(scene)


def margin(scene) -> float:
    """Runner-up margin: minimum clearance from the route to any non-answer dest."""
    reached = _reached_color(scene)
    best = float("inf")
    for d in scene["destinations"]:
        if d["color"] == reached:
            continue
        center = d["pos"]
        seg_min = min(
            _segment_clearance(a, b, center)
            for a, b in zip(scene["route"][:-1], scene["route"][1:])
        )
        best = min(best, seg_min - DEST_RADIUS)
    return float(best if best != float("inf") else 0.0)


def is_quarantined(scene) -> bool:
    # Sampling guarantees a clear gap; every scene is decidable and non-marginal.
    return margin(scene) < 8.0


def latent_symmetries(scene):
    """not_applicable: the latent parametrization is minimal and injective.

    Every label-input field (route, destinations) is directly and visibly
    realised in the raster; there is no latent field hidden from rendering and
    no pixel-identical alternative scene that changes the answer.
    """
    return iter(())
