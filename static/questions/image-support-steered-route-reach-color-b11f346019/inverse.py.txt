"""Independent pixel-only oracle for the route-reach color world.

Receives only the final PNG. Never imports renderer/prompts/generate/verify and
never accepts latent scene data.

Algorithm: segment ink (dark route plus the colored markers), flood-fill the
component containing the orange start marker, and report the color of the
destination whose disc lies in that same connected component. If no destination
is reachable, or two destinations tie for overlap, returns "none".
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _bfs(mask, sy, sx):
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    if not mask[sy, sx]:
        return visited
    queue = deque([(sy, sx)])
    visited[sy, sx] = True
    while queue:
        y, x = queue.popleft()
        ylo, yhi = max(0, y - 1), min(height, y + 2)
        xlo, xhi = max(0, x - 1), min(width, x + 2)
        for ny in range(ylo, yhi):
            row = mask[ny]
            vrow = visited[ny]
            for nx in range(xlo, xhi):
                if row[nx] and not vrow[nx]:
                    vrow[nx] = True
                    queue.append((ny, nx))
    return visited


def decision_from_image(image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    if arr.ndim != 3:
        return "none"
    r = arr[..., 0]
    g = arr[..., 1]
    b = arr[..., 2]

    route = (r < 90) & (g < 90) & (b < 90)
    start = (r > 170) & (g > 100) & (g < 210) & (b < 130) & (r > g) & (g > b)
    red = (r > 150) & (g < 110) & (b < 110)
    green = (g > 100) & (r < 120) & (b < 120) & (g > r) & (g > b)
    blue = (b > 150) & (r < 120) & (g < 130)

    ink = route | start | red | green | blue

    seeds = np.argwhere(start)
    if not len(seeds):
        return "none"
    sy, sx = seeds[len(seeds) // 2]
    visited = _bfs(ink, int(sy), int(sx))

    def score(mask):
        return int(np.count_nonzero(mask & visited))

    scores = {
        "red": score(red),
        "blue": score(blue),
        "green": score(green),
    }
    ranked = sorted(scores.values(), reverse=True)
    if not ranked or ranked[0] == 0:
        return "none"
    if len(ranked) > 1 and ranked[0] == ranked[1]:
        return "none"
    return max(scores, key=scores.get)
