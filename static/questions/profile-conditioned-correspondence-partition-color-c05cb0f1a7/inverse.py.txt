"""Independent pixel-only inverse arm for partition-proportion correspondence.

Receives only the final PNG. It segments each bar by color, measures the dark
partition line's position within each bar's own bounding box, normalizes that
position by the bar width to recover the partition fraction, and selects the
candidate whose fraction best matches the gray probe bar's fraction, returning
that candidate's color name.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

GRAY = (160, 160, 160)
DARK = (50, 50, 50)
PALETTE = {
    "red": (215, 60, 60),
    "green": (55, 180, 75),
    "blue": (65, 100, 215),
}
TOL = 70


def _close(image: np.ndarray, color: tuple) -> np.ndarray:
    return np.abs(image.astype(int) - np.array(color)).sum(axis=2) < TOL


def _fraction_of_mask(mask: np.ndarray, dark: np.ndarray) -> float | None:
    ys, xs = np.nonzero(mask)
    if len(xs) == 0:
        return None
    left, right = int(xs.min()), int(xs.max())
    width = right - left
    if width <= 0:
        return None
    slice_mask = dark[ys.min() : ys.max() + 1, left : right + 1]
    mark_xs = np.nonzero(slice_mask)[1]
    if len(mark_xs) == 0:
        return None
    mark_x = left + float(mark_xs.mean())
    return (mark_x - left) / width


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"))
    dark = _close(rgb, DARK)
    probe_fraction = _fraction_of_mask(_close(rgb, GRAY), dark)
    if probe_fraction is None:
        return "unknown"
    best_color = "unknown"
    best_delta = float("inf")
    for name, color in PALETTE.items():
        fraction = _fraction_of_mask(_close(rgb, color), dark)
        if fraction is None:
            continue
        delta = abs(fraction - probe_fraction)
        if delta < best_delta:
            best_delta = delta
            best_color = name
    return best_color
