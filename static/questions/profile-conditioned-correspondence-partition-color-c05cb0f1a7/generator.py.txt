"""Cross-panel partition-proportion correspondence world.

The gray probe bar at the top is divided by a dark vertical line that sits at a
specific fraction f of the bar's length. Below, three colored bars are each
divided at a different fraction (and are a different color, with different
widths). The task is to find the candidate bar whose dark partition line sits at
exactly the same fraction of its own length as the gray probe's, then report that
bar's color.

The matching key is a dimensionless proportion (partition position normalized by
bar width), so it is scale-invariant: the same fraction can correspond to very
different absolute pixel positions when bar widths differ.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 620, 420

PROBE_TOP = 110
CAND_TOP = 300
BAR_H = 80

FRACTIONS = [0.25, 0.40, 0.55, 0.70]
FRACTION_GAP = min(
    abs(FRACTIONS[i + 1] - FRACTIONS[i]) for i in range(len(FRACTIONS) - 1)
)
MIN_GAP = 0.05

GRAY = (160, 160, 160)
DARK = (50, 50, 50)
PALETTE = {
    "red": (215, 60, 60),
    "green": (55, 180, 75),
    "blue": (65, 100, 215),
}
COLOR_NAMES = ["red", "green", "blue"]
MARK_HALF = 2


def _draw_bar(draw: ImageDraw.ImageDraw, left: int, top: int, width: int, color, fraction: float):
    right = left + width
    bottom = top + BAR_H
    draw.rectangle([left, top, right, bottom], fill=color)
    mark_x = left + int(round(fraction * width))
    draw.rectangle([mark_x - MARK_HALF, top, mark_x + MARK_HALF, bottom], fill=DARK)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    colors = list(COLOR_NAMES)
    rng.shuffle(colors)
    fractions = rng.sample(FRACTIONS, 3)
    probe_fraction = rng.choice(fractions)
    widths = [rng.choice([120, 150, 180]) for _ in range(3)]
    centers = [120, 310, 500]
    rng.shuffle(centers)
    return {
        "probe_partition": probe_fraction,
        "probe_width": 220,
        "candidate_colors": colors,
        "candidate_partitions": fractions,
        "candidate_widths": widths,
        "candidate_positions": centers,
    }


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), (252, 252, 252))
    draw = ImageDraw.Draw(img)
    probe_left = (WIDTH - scene["probe_width"]) // 2
    _draw_bar(
        draw,
        probe_left,
        PROBE_TOP,
        scene["probe_width"],
        GRAY,
        scene["probe_partition"],
    )
    for color_name, fraction, width, center_x in zip(
        scene["candidate_colors"],
        scene["candidate_partitions"],
        scene["candidate_widths"],
        scene["candidate_positions"],
    ):
        left = center_x - width // 2
        _draw_bar(draw, left, CAND_TOP, width, PALETTE[color_name], fraction)
    return img


def analytic_gold(scene: dict) -> str:
    pp = scene["probe_partition"]
    parts = scene["candidate_partitions"]
    cols = scene["candidate_colors"]
    return cols[parts.index(pp)]


def margin(scene: dict) -> float:
    pp = scene["probe_partition"]
    distances = sorted(
        abs(pp - part) for part in scene["candidate_partitions"] if part != pp
    )
    return float(distances[0]) if distances else float("inf")


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < MIN_GAP


def latent_symmetries(scene: dict) -> list:
    return []
