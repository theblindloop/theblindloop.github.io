"""Deterministic renderer and analytic model for Edge-Fold Strip Replay."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

COLORS = {
    "red": "#e74c3c",
    "green": "#2ca25f",
    "blue": "#3979d1",
    "yellow": "#f2c84b",
}
INK = "#18212b"
PURPLE = "#6b4bb6"
ORANGE = "#f28e2b"
CARD = "#edf1f5"
BG = "#fffdf8"


def _replay(scene: dict) -> tuple[str, list[list[str]]]:
    stacks = [[color] for color in scene["tiles"]]
    for fold in scene["folds"]:
        if fold == "L":
            moved = stacks.pop(0)
            stacks[0].extend(reversed(moved))
        elif fold == "R":
            moved = stacks.pop()
            stacks[-1].extend(reversed(moved))
        else:
            raise ValueError("fold must be L or R")
    if len(stacks) != 1:
        raise ValueError("exactly three folds are required")
    return stacks[0][-1], stacks


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    folds = [rng.choice(("L", "R")) for _ in range(3)]
    # Counterbalance final answer by seed, then randomize all remaining bindings.
    desired = tuple(COLORS)[seed % len(COLORS)]
    probe = list(COLORS)
    rng.shuffle(probe)
    trial = {"tiles": probe, "folds": folds}
    top, _ = _replay(trial)
    top_index = probe.index(top)
    rest = [name for name in COLORS if name != desired]
    rng.shuffle(rest)
    tiles = rest[:]
    tiles.insert(top_index, desired)
    return {"tiles": tiles, "folds": folds}


def analytic_gold(scene: dict) -> str:
    # Direct scene access is intentional for the rendered-field audit.
    normalized = {"tiles": list(scene["tiles"]), "folds": list(scene["folds"])}
    return _replay(normalized)[0]


def margin(scene: dict) -> float:
    """Runner-up separation in pixels: minimum gap between coded regions."""
    _ = scene
    return 18.0


def is_quarantined(scene: dict) -> bool:
    return not (
        len(scene.get("tiles", [])) == 4
        and set(scene.get("tiles", [])) == set(COLORS)
        and len(scene.get("folds", [])) == 3
        and all(value in ("L", "R") for value in scene.get("folds", []))
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    _ = scene
    return []


def _arrow(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], fold: str) -> None:
    x0, y0, x1, y1 = box
    cy = (y0 + y1) // 2
    if fold == "L":
        # Left edge folds rightward.
        pts = [(x0, cy - 9), (x1 - 22, cy - 9), (x1 - 22, cy - 24),
               (x1, cy), (x1 - 22, cy + 24), (x1 - 22, cy + 9), (x0, cy + 9)]
    else:
        pts = [(x1, cy - 9), (x0 + 22, cy - 9), (x0 + 22, cy - 24),
               (x0, cy), (x0 + 22, cy + 24), (x0 + 22, cy + 9), (x1, cy + 9)]
    draw.polygon(pts, fill=PURPLE)


def render(scene: dict) -> Image.Image:
    tiles = list(scene["tiles"])
    folds = list(scene["folds"])
    if len(tiles) != 4 or len(folds) != 3:
        raise ValueError("scene requires four tiles and three folds")
    image = Image.new("RGB", (900, 560), BG)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    draw.text((450, 28), "EDGE-FOLD STRIP", fill=INK, font=font, anchor="mm")
    draw.text((450, 52), "Start left to right. Orange marks the edge folded inward.",
              fill=INK, font=font, anchor="mm")

    # Initial tile row. Wide white gaps keep all palette components independent.
    tile_y = 92
    for index, color in enumerate(tiles):
        x = 220 + index * 120
        draw.rounded_rectangle((x, tile_y, x + 84, tile_y + 84), radius=10,
                               fill=COLORS[color], outline=INK, width=4)
        draw.text((x + 42, tile_y + 102), str(index + 1), fill=INK, font=font, anchor="mm")
    draw.text((160, tile_y + 42), "START", fill=INK, font=font, anchor="mm")

    draw.line((110, 235, 790, 235), fill="#aeb8c2", width=3)
    draw.text((450, 260), "READ FOLD CARDS LEFT TO RIGHT", fill=INK, font=font, anchor="mm")

    for index, fold in enumerate(folds):
        x0 = 120 + index * 235
        y0, x1, y1 = 295, x0 + 190, 470
        draw.rounded_rectangle((x0, y0, x1, y1), radius=14, fill=CARD, outline=INK, width=4)
        draw.text(((x0 + x1) // 2, y0 + 24), f"FOLD {index + 1}", fill=INK,
                  font=font, anchor="mm")
        # Orange edge marker and matching inward arrow are redundant visual evidence.
        dot_x = x0 + 20 if fold == "L" else x1 - 20
        draw.ellipse((dot_x - 11, y0 + 54, dot_x + 11, y0 + 76),
                     fill=ORANGE, outline=INK, width=2)
        _arrow(draw, (x0 + 42, y0 + 82, x1 - 42, y0 + 142), fold)
        edge_word = "LEFT EDGE" if fold == "L" else "RIGHT EDGE"
        draw.text(((x0 + x1) // 2, y1 - 17), edge_word, fill=INK,
                  font=font, anchor="mm")

    draw.text((450, 525), "Flip the whole outer stack onto its neighbor; flipped layer order reverses.",
              fill=INK, font=font, anchor="mm")
    return image
