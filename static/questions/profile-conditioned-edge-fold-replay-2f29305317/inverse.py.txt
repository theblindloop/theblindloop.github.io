"""Independent image-only inverse arm for Edge-Fold Strip Replay."""

from __future__ import annotations

from collections import deque
import numpy as np

PALETTE = {
    "red": (231, 76, 60),
    "green": (44, 162, 95),
    "blue": (57, 121, 209),
    "yellow": (242, 200, 75),
}
CARD_RGB = (237, 241, 245)
ORANGE_RGB = (242, 142, 43)


def _components(mask: np.ndarray) -> list[dict]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    values = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        ys = [p[0] for p in points]
        xs = [p[1] for p in points]
        values.append({
            "area": len(points), "x0": min(xs), "x1": max(xs),
            "y0": min(ys), "y1": max(ys),
            "cx": sum(xs) / len(xs), "cy": sum(ys) / len(ys),
        })
    return values


def _near(arr: np.ndarray, rgb: tuple[int, int, int], tolerance: int = 8) -> np.ndarray:
    target = np.asarray(rgb, dtype=np.int16)
    return np.max(np.abs(arr.astype(np.int16) - target), axis=2) <= tolerance


def _replay(tiles: list[str], folds: list[str]) -> str:
    stacks = [[color] for color in tiles]
    for fold in folds:
        if fold == "L":
            moved = stacks.pop(0)
            stacks[0].extend(reversed(moved))
        else:
            moved = stacks.pop()
            stacks[-1].extend(reversed(moved))
    return stacks[0][-1]


def decision_from_image(image) -> str:
    arr = np.asarray(image.convert("RGB"))

    # Recover four large tile fills from color components, then sort spatially.
    tile_hits = []
    for name, rgb in PALETTE.items():
        comps = [item for item in _components(_near(arr, rgb)) if item["area"] > 2000]
        if len(comps) != 1:
            raise ValueError(f"expected one large {name} tile, found {len(comps)}")
        tile_hits.append((comps[0]["cx"], name))
    tiles = [name for _, name in sorted(tile_hits)]
    if len(tiles) != 4:
        raise ValueError("could not recover four initial tiles")

    # Recover card bounds as large connected light-gray fields.
    cards = [item for item in _components(_near(arr, CARD_RGB)) if item["area"] > 12000]
    cards.sort(key=lambda item: item["cx"])
    if len(cards) != 3:
        raise ValueError(f"expected three fold cards, found {len(cards)}")

    orange = [item for item in _components(_near(arr, ORANGE_RGB)) if item["area"] > 150]
    if len(orange) != 3:
        raise ValueError(f"expected three edge markers, found {len(orange)}")
    folds = []
    for card in cards:
        hits = [dot for dot in orange if card["x0"] < dot["cx"] < card["x1"]
                and card["y0"] < dot["cy"] < card["y1"]]
        if len(hits) != 1:
            raise ValueError("edge marker is not uniquely bound to a card")
        folds.append("L" if hits[0]["cx"] < card["cx"] else "R")
    return _replay(tiles, folds)
