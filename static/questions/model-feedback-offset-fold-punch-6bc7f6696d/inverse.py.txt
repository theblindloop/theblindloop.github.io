"""Independent image-only inverse for offset fold-and-punch scenes."""

from __future__ import annotations

from PIL import Image


N = 6
HOLE = (38, 47, 62)
LABELS = ("A", "B", "C", "D")


def _is_hole(image: Image.Image, cx: int, cy: int) -> bool:
    pixels = image.load()
    hits = 0
    for y in range(cy - 5, cy + 6):
        for x in range(cx - 5, cx + 6):
            if pixels[x, y] == HOLE:
                hits += 1
    return hits >= 100


def _read_packet(image: Image.Image) -> set[tuple[int, int]]:
    x, y, cell = 420, 142, 58
    holes = set()
    for r in range(3):
        for local_c in range(4):
            if _is_hole(image, x + local_c * cell + cell // 2, y + r * cell + cell // 2):
                holes.add((r, local_c + 2))
    if len(holes) != 3:
        raise ValueError(f"expected three packet punches, found {len(holes)}")
    return holes


def _read_result(image: Image.Image, x: int) -> set[tuple[int, int]]:
    y, cell = 535, 38
    holes = set()
    for r in range(N):
        for c in range(N):
            if _is_hole(image, x + c * cell + cell // 2, y + r * cell + cell // 2):
                holes.add((r, c))
    return holes


def _unfold(punches: set[tuple[int, int]]) -> set[tuple[int, int]]:
    after_second = punches | {(N - 1 - r, c) for r, c in punches}
    return after_second | {(r, 3 - c) for r, c in after_second if c in (2, 3)}


def decision_from_image(image: Image.Image) -> str:
    image = image.convert("RGB")
    if image.size != (1200, 820):
        raise ValueError(f"unexpected image size {image.size}")
    expected = _unfold(_read_packet(image))
    observed = [_read_result(image, x) for x in (30, 323, 616, 909)]
    matches = [i for i, holes in enumerate(observed) if holes == expected]
    if len(matches) != 1:
        raise ValueError(f"expected one correct result, found {len(matches)}")
    return LABELS[matches[0]]
