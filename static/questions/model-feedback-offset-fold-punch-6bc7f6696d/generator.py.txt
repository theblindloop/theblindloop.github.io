"""Deterministic renderer and analytic program for offset fold-and-punch geometry."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont


WIDTH = 1200
HEIGHT = 820
N = 6
OPTION_LABELS = ("A", "B", "C", "D")
MODES = ("correct", "horizontal_only", "center_both", "vertical_only")
INK = (38, 47, 62)
GRID = (137, 148, 164)
AMBER = (224, 145, 36)
BLUE = (49, 117, 190)
PAPER = (253, 253, 250)
HOLE = (38, 47, 62)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype("DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf", size)


def _result_for_mode(punches: Iterable[Iterable[int]], mode: str) -> set[tuple[int, int]]:
    base = {(int(r), int(c)) for r, c in punches}
    if mode == "horizontal_only":
        return base | {(N - 1 - r, c) for r, c in base}
    if mode == "vertical_only":
        return base | {(r, 3 - c) for r, c in base if c in (2, 3)}
    if mode == "center_both":
        horizontal = base | {(N - 1 - r, c) for r, c in base}
        return horizontal | {(r, N - 1 - c) for r, c in horizontal}
    if mode == "correct":
        horizontal = base | {(N - 1 - r, c) for r, c in base}
        return horizontal | {(r, 3 - c) for r, c in horizontal if c in (2, 3)}
    raise ValueError(f"unknown mode {mode!r}")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    packet = [(r, c) for r in range(3) for c in range(2, 6)]
    for _ in range(100):
        left = rng.choice([(r, c) for r, c in packet if c in (2, 3)])
        right = rng.choice([(r, c) for r, c in packet if c in (4, 5)])
        third = rng.choice([cell for cell in packet if cell not in (left, right)])
        punches = sorted([left, right, third])
        sets = [_result_for_mode(punches, mode) for mode in MODES]
        if len({tuple(sorted(value)) for value in sets}) == len(MODES):
            break
    else:
        raise RuntimeError("could not sample distinct unfold outcomes")
    shift = int(seed) % 4
    option_modes = list(MODES[shift:] + MODES[:shift])
    return {"width": WIDTH, "height": HEIGHT, "punches": punches, "option_modes": option_modes}


def _grid(draw: ImageDraw.ImageDraw, x: int, y: int, cols: int, rows: int, cell: int) -> None:
    draw.rectangle((x, y, x + cols * cell, y + rows * cell), fill=PAPER, outline=INK, width=4)
    for c in range(1, cols):
        draw.line((x + c * cell, y, x + c * cell, y + rows * cell), fill=GRID, width=2)
    for r in range(1, rows):
        draw.line((x, y + r * cell, x + cols * cell, y + r * cell), fill=GRID, width=2)


def _disk(draw: ImageDraw.ImageDraw, cx: int, cy: int, radius: int, fill=HOLE) -> None:
    draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=fill)


def _arrow(draw: ImageDraw.ImageDraw, points: list[tuple[int, int]], color: tuple[int, int, int]) -> None:
    draw.line(points, fill=color, width=9, joint="curve")
    x1, y1 = points[-2]
    x2, y2 = points[-1]
    if abs(x2 - x1) > abs(y2 - y1):
        s = 1 if x2 > x1 else -1
        head = [(x2, y2), (x2 - 18 * s, y2 - 13), (x2 - 18 * s, y2 + 13)]
    else:
        s = 1 if y2 > y1 else -1
        head = [(x2, y2), (x2 - 13, y2 - 18 * s), (x2 + 13, y2 - 18 * s)]
    draw.polygon(head, fill=color)


def _draw_setup(draw: ImageDraw.ImageDraw) -> None:
    x, y, cell = 48, 118, 45
    draw.text((x, 77), "SHEET + FOLD ORDER", font=_font(23, True), fill=INK)
    _grid(draw, x, y, 6, 6, cell)
    # Tint the moved regions without obscuring the grid.
    draw.rectangle((x + 3, y + 3, x + 2 * cell - 3, y + 6 * cell - 3), fill=(255, 244, 216))
    for c in range(1, 6):
        draw.line((x + c * cell, y, x + c * cell, y + 6 * cell), fill=GRID, width=2)
    for r in range(1, 6):
        draw.line((x, y + r * cell, x + 6 * cell, y + r * cell), fill=GRID, width=2)
    draw.line((x + 2 * cell, y - 6, x + 2 * cell, y + 6 * cell + 6), fill=AMBER, width=7)
    _arrow(draw, [(x + 30, y + 70), (x + 105, y + 70)], AMBER)
    _disk(draw, x - 19, y + 34, 15, AMBER)
    draw.text((x - 19, y + 34), "1", font=_font(18, True), fill=(255, 255, 255), anchor="mm")
    draw.text((x + 18, y + 6 * cell + 16), "LEFT 2 COLUMNS  →", font=_font(17, True), fill=(137, 83, 15))

    draw.rectangle((x + 3, y + 3 * cell + 3, x + 6 * cell - 3, y + 6 * cell - 3), outline=(183, 215, 244), width=10)
    draw.line((x - 6, y + 3 * cell, x + 6 * cell + 6, y + 3 * cell), fill=BLUE, width=7)
    _arrow(draw, [(x + 220, y + 235), (x + 220, y + 165)], BLUE)
    _disk(draw, x - 19, y + 3 * cell + 34, 15, BLUE)
    draw.text((x - 19, y + 3 * cell + 34), "2", font=_font(18, True), fill=(255, 255, 255), anchor="mm")
    draw.text((x + 79, y + 6 * cell + 43), "BOTTOM 3 ROWS  ↑", font=_font(17, True), fill=(29, 83, 139))


def _draw_packet(draw: ImageDraw.ImageDraw, punches: Iterable[Iterable[int]]) -> None:
    x, y, cell = 420, 142, 58
    draw.text((x, 77), "FOLDED PACKET", font=_font(23, True), fill=INK)
    _grid(draw, x, y, 4, 3, cell)
    for r, c in punches:
        _disk(draw, x + (int(c) - 2) * cell + cell // 2, y + int(r) * cell + cell // 2, 14)
    draw.text((x + 2 * cell, y + 3 * cell + 24), "PUNCH 3 HOLES", font=_font(18, True), fill=INK, anchor="ma")
    draw.text((x + 2 * cell, y + 3 * cell + 52), "then unfold  2  →  1", font=_font(18, True), fill=(78, 87, 101), anchor="ma")


def _draw_rule_card(draw: ImageDraw.ImageDraw) -> None:
    x, y = 730, 112
    draw.rounded_rectangle((x, y, 1138, 385), radius=18, fill=(238, 244, 252), outline=(101, 126, 158), width=3)
    draw.text((x + 204, y + 24), "FOLD ONLY THE NAMED PART", font=_font(20, True), fill=INK, anchor="ma")
    draw.text((x + 30, y + 74), "1", font=_font(19, True), fill=(255, 255, 255), anchor="mm", stroke_width=9, stroke_fill=AMBER)
    draw.text((x + 58, y + 74), "left two columns move right", font=_font(18), fill=INK, anchor="lm")
    draw.text((x + 30, y + 121), "2", font=_font(19, True), fill=(255, 255, 255), anchor="mm", stroke_width=9, stroke_fill=BLUE)
    draw.text((x + 58, y + 121), "bottom three rows move up", font=_font(18), fill=INK, anchor="lm")
    draw.line((x + 26, y + 166, x + 382, y + 166), fill=(167, 177, 190), width=2)
    draw.text((x + 204, y + 196), "UNFOLD IN REVERSE", font=_font(20, True), fill=(65, 72, 84), anchor="ma")
    draw.text((x + 204, y + 232), "each sheet layer gets the hole", font=_font(18), fill=(65, 72, 84), anchor="ma")


def _draw_option(draw: ImageDraw.ImageDraw, x: int, y: int, label: str, holes: set[tuple[int, int]]) -> None:
    cell = 38
    draw.text((x + 3 * cell, y - 31), label, font=_font(27, True), fill=INK, anchor="ma")
    _grid(draw, x, y, 6, 6, cell)
    for r, c in holes:
        _disk(draw, x + c * cell + cell // 2, y + r * cell + cell // 2, 9)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), (246, 248, 252))
    draw = ImageDraw.Draw(image)
    draw.text((WIDTH // 2, 25), "OFFSET FOLD & PUNCH", font=_font(29, True), fill=INK, anchor="ma")
    _draw_setup(draw)
    _draw_packet(draw, scene["punches"])
    _draw_rule_card(draw)
    draw.text((44, 480), "RESULT OPTIONS — FULL 6 × 6 SHEET", font=_font(23, True), fill=INK)
    for label, mode, x in zip(OPTION_LABELS, scene["option_modes"], (30, 323, 616, 909)):
        _draw_option(draw, x, 535, label, _result_for_mode(scene["punches"], str(mode)))
    return image


def analytic_gold(scene: dict) -> str:
    matches = [i for i, mode in enumerate(scene["option_modes"]) if mode == "correct"]
    if len(matches) != 1:
        raise ValueError("scene must have exactly one correct panel")
    return OPTION_LABELS[matches[0]]


def margin(scene: dict) -> float:
    results = [_result_for_mode(scene["punches"], str(mode)) for mode in scene["option_modes"]]
    target = results[OPTION_LABELS.index(analytic_gold(scene))]
    return float(min(len(target.symmetric_difference(other)) for other in results if other != target))


def is_quarantined(scene: dict) -> bool:
    results = [tuple(sorted(_result_for_mode(scene["punches"], str(mode)))) for mode in scene["option_modes"]]
    return len(set(results)) != 4


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
