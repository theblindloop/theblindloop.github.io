"""Independent pixel-only inverse arm for the bridge-cut-count world.

receives ONLY the final PNG. It never imports or reads the renderer, the
analytic scene, prompts, or any renderer state. It recovers the decision by:

1. nearest-colour labelling of every pixel to one of the four node colours, the
   near-black link colour, or white;
2. finding the connected components of near-black pixels (each component is one
   visible link) and, for each, which node colours it is 4-adjacent to (a link
   touches exactly its two endpoint nodes);
3. building the node-graph and counting the number of links (cut-edges) whose
   removal disconnects the network.

This reproduces the analytic bridge count exactly on robust scenes.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "bridge_cut_count-oracle-0.4.0"

# Must mirror the renderer's palette/order so nearest-colour labelling and the
# tie-breaking are identical, but this module shares no code with the renderer.
_PALETTE = ["red", "green", "blue", "purple", "orange", "teal"]
_COLOR_RGBS = {
    "red": (220, 30, 60),
    "green": (40, 200, 60),
    "blue": (40, 90, 230),
    "purple": (160, 40, 210),
    "orange": (255, 150, 0),
    "teal": (0, 170, 170),
}
_LINK_RGB = (20, 20, 20)
_WHITE_RGB = (255, 255, 255)


def _nearest_labels(arr: np.ndarray) -> tuple[np.ndarray, int]:
    refs = [_WHITE_RGB] + [_COLOR_RGBS[c] for c in _PALETTE] + [_LINK_RGB]
    refs = np.array(refs, dtype=np.float32)
    labels = np.linalg.norm(arr[..., None, :] - refs[None, None, :, :], axis=-1).argmin(axis=-1)
    # label 0 = white, 1..len(_PALETTE) = node colours, len(_PALETTE)+1 = black.
    black_label = len(_PALETTE) + 1
    return labels, black_label


def _black_components_adjacent_blocks(labels: np.ndarray, black_label: int) -> list[set[int]]:
    black = labels == black_label
    h, w = black.shape
    visited = np.zeros((h, w), dtype=bool)
    blocks: list[set[int]] = []
    ys, xs = np.nonzero(black)
    for y, x in zip(ys.tolist(), xs.tolist()):
        if visited[y, x]:
            continue
        stack = [(y, x)]
        visited[y, x] = True
        node_hit: set[int] = set()
        while stack:
            cy, cx = stack.pop()
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < h and 0 <= nx < w:
                    if black[ny, nx]:
                        if not visited[ny, nx]:
                            visited[ny, nx] = True
                            stack.append((ny, nx))
                    else:
                        lbl = int(labels[ny, nx])
                        if 1 <= lbl <= len(_PALETTE):
                            node_hit.add(lbl - 1)
        if len(node_hit) == 2:
            blocks.append(node_hit)
    return blocks


def _bridge_count_from_blocks(blocks: list[set[int]]) -> int:
    adj: dict[int, set[int]] = {}
    for block in blocks:
        a, b = sorted(block)
        adj.setdefault(a, set()).add(b)
        adj.setdefault(b, set()).add(a)

    def cut(a: int, b: int) -> bool:
        seen = {a}
        stack = [a]
        while stack:
            u = stack.pop()
            for v in adj.get(u, ()):
                if {u, v} == {a, b}:
                    continue
                if v not in seen:
                    seen.add(v)
                    stack.append(v)
        return b not in seen

    count = 0
    for a, b in blocks:
        if cut(a, b):
            count += 1
    return count


def decision_from_image(image: Image.Image) -> str:
    """Number of visible bridges as a string, recovered from the PNG alone."""
    arr = np.asarray(image.convert("RGB")).astype(np.float32)
    labels, black_label = _nearest_labels(arr)
    blocks = _black_components_adjacent_blocks(labels, black_label)
    return str(_bridge_count_from_blocks(blocks))
