"""Pixel-only inverse arm for the dip-recovery counting world.

The arm receives one rendered PNG and nothing else: no scene, no renderer code,
no analytic gold, no file names.  The procedure is

1. segment the opaque black ink by colour and keep its single largest
   8-connected component (a second substantial component is a degenerate
   raster and returns "unresolved");
2. recover one centre height per occupied image column, requiring the column
   support to be gap-free and at least `MIN_COLUMNS` columns wide, so a trace
   that has been cut short or interrupted anywhere makes the arm abstain
   instead of answering from a surviving fragment;
3. read the guide height off the trace's own left end, then walk the centre
   line left to right with a hysteresis band and count every transition from
   below the guide height to above it.

The count is the returned decision.  Degenerate rasters return "unresolved".
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "dip_recovery_count-oracle-0.1.0"

INK_MAX = 70          # every channel below this counts as trace ink
MIN_INK = 240         # a smaller component cannot be a usable trace
STRAY_INK = 24        # a second component above this size is degenerate
MIN_COLUMNS = 160     # a usable trace spans at least this many image columns
HYST = 3.5            # pixels of hysteresis around the guide height
END_COLUMNS = 2       # columns averaged to read the left-end guide height


def _ink_pixels(image: Image.Image) -> list[tuple[int, int]]:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    mask = (array < INK_MAX).all(axis=2)
    return [(int(x), int(y)) for y, x in np.argwhere(mask)]


def _components(points: list[tuple[int, int]]) -> list[list[tuple[int, int]]]:
    remaining = set(points)
    groups: list[list[tuple[int, int]]] = []
    while remaining:
        start = next(iter(remaining))
        remaining.discard(start)
        queue = deque([start])
        group = [start]
        while queue:
            cx, cy = queue.popleft()
            for dx in (-1, 0, 1):
                for dy in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    neighbour = (cx + dx, cy + dy)
                    if neighbour in remaining:
                        remaining.discard(neighbour)
                        queue.append(neighbour)
                        group.append(neighbour)
        groups.append(group)
    groups.sort(key=len, reverse=True)
    return groups


def _centre_line(group: list[tuple[int, int]]) -> list[float] | None:
    columns: dict[int, list[int]] = {}
    for x, y in group:
        columns.setdefault(x, []).append(y)
    xs = sorted(columns)
    if len(xs) < MIN_COLUMNS:
        return None
    if xs[-1] - xs[0] + 1 != len(xs):  # a gap in the column support
        return None
    return [sum(columns[x]) / len(columns[x]) for x in xs]


def decision_from_image(image: Image.Image) -> str:
    """Return the number of upward returns as a string, or "unresolved"."""
    points = _ink_pixels(image)
    if len(points) < MIN_INK:
        return "unresolved"
    groups = _components(points)
    if len(groups) > 1 and len(groups[1]) > STRAY_INK:
        return "unresolved"
    if len(groups[0]) < MIN_INK:
        return "unresolved"
    line = _centre_line(groups[0])
    if line is None:
        return "unresolved"

    level = sum(line[:END_COLUMNS]) / END_COLUMNS
    state: int | None = None
    returns = 0
    for value in line:
        if value > level + HYST:
            here = -1              # below the guide height (image y grows down)
        elif value < level - HYST:
            here = 1               # above the guide height
        else:
            continue
        if state is None:
            if here != -1:
                return "unresolved"  # the trace must drop below the guide first
            state = here
            continue
        if state == -1 and here == 1:
            returns += 1
        state = here
    if state is None or returns < 1:
        return "unresolved"
    return str(returns)
