"""Independent pixel-only inverse arm for marker-region partitions."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _colored_components(mask: np.ndarray) -> list[tuple[int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components: list[tuple[int, int, int]] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        pixels: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            pixels.append((cy, cx))
            for ny in range(max(0, cy - 1), min(height, cy + 2)):
                for nx in range(max(0, cx - 1), min(width, cx + 2)):
                    if not seen[ny, nx] and mask[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        if len(pixels) >= 30:
            ys = [item[0] for item in pixels]
            xs = [item[1] for item in pixels]
            components.append((round(sum(ys) / len(ys)), round(sum(xs) / len(xs)), len(pixels)))
    return components


def _reachable_labels(free: np.ndarray, starts: list[tuple[int, int]]) -> list[int]:
    height, width = free.shape
    labels = np.full((height, width), -1, dtype=np.int16)
    start_labels: list[int] = []
    next_label = 0
    for sy, sx in starts:
        existing = int(labels[sy, sx])
        if existing >= 0:
            start_labels.append(existing)
            continue
        if not free[sy, sx]:
            raise ValueError("a marker centroid lies on a segmented wall")
        queue = [sy * width + sx]
        labels[sy, sx] = next_label
        while queue:
            value = queue.pop()
            y, x = divmod(value, width)
            if y and free[y - 1, x] and labels[y - 1, x] < 0:
                labels[y - 1, x] = next_label
                queue.append(value - width)
            if y + 1 < height and free[y + 1, x] and labels[y + 1, x] < 0:
                labels[y + 1, x] = next_label
                queue.append(value + width)
            if x and free[y, x - 1] and labels[y, x - 1] < 0:
                labels[y, x - 1] = next_label
                queue.append(value - 1)
            if x + 1 < width and free[y, x + 1] and labels[y, x + 1] < 0:
                labels[y, x + 1] = next_label
                queue.append(value + 1)
        start_labels.append(next_label)
        next_label += 1
    # Starts encountered before a later flood can retain a provisional label only
    # if already filled. Re-read the final label image for the authoritative IDs.
    return [int(labels[y, x]) for y, x in starts]


def occupied_region_count_from_image(image: Image.Image) -> int:
    """Segment walls/markers and count marker-bearing 4-connected free regions."""

    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if rgb.ndim != 3 or rgb.shape[0] < 64 or rgb.shape[1] < 64:
        raise ValueError("image is too small")
    # A generic 2x reduction is safe because submitted walls are >=8 px wide.
    rgb = rgb[::2, ::2]
    high = rgb.max(axis=2).astype(np.int16)
    low = rgb.min(axis=2).astype(np.int16)
    barrier = high < 90
    colored = (high - low > 55) & (high < 250) & ~barrier
    markers = _colored_components(colored)
    if len(markers) != 4:
        raise ValueError(f"expected four colored marker components, found {len(markers)}")
    starts = [(y, x) for y, x, _ in markers]
    labels = _reachable_labels(~barrier, starts)
    return len(set(labels))


def decision_from_image(image: Image.Image) -> str:
    return "yes" if occupied_region_count_from_image(image) == 2 else "no"
