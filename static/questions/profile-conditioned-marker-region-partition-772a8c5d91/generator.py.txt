"""Deterministic latent sampler and renderer for marker-region partitions."""

from __future__ import annotations

import copy
import itertools
import random
from collections import deque

from PIL import Image, ImageDraw

CANVAS = 512
GRID_N = 4
GRID_LEFT = 56
GRID_TOP = 56
CELL = 100
MARKER_RADIUS = 14
PALETTE = ("#2d70d2", "#1fa05a", "#8f4ebe", "#e67e22")
BACKGROUND = "#ffffff"
WALL = "#17191d"


def _all_edges() -> list[tuple[int, int, int, int]]:
    edges: list[tuple[int, int, int, int]] = []
    for row in range(GRID_N):
        for col in range(GRID_N):
            if col + 1 < GRID_N:
                edges.append((row, col, row, col + 1))
            if row + 1 < GRID_N:
                edges.append((row, col, row + 1, col))
    return edges


ALL_EDGES = tuple(_all_edges())


def _canonical_edge(edge: list[int] | tuple[int, ...]) -> tuple[int, int, int, int]:
    a = (int(edge[0]), int(edge[1]))
    b = (int(edge[2]), int(edge[3]))
    if a > b:
        a, b = b, a
    return (a[0], a[1], b[0], b[1])


def _components(walls: list[list[int]] | tuple[tuple[int, ...], ...]) -> list[list[tuple[int, int]]]:
    blocked = {_canonical_edge(edge) for edge in walls}
    adjacency: dict[tuple[int, int], list[tuple[int, int]]] = {
        (r, c): [] for r in range(GRID_N) for c in range(GRID_N)
    }
    for edge in ALL_EDGES:
        if edge in blocked:
            continue
        a, b = (edge[0], edge[1]), (edge[2], edge[3])
        adjacency[a].append(b)
        adjacency[b].append(a)
    unseen = set(adjacency)
    groups: list[list[tuple[int, int]]] = []
    while unseen:
        start = min(unseen)
        unseen.remove(start)
        queue = deque([start])
        group = [start]
        while queue:
            cell = queue.popleft()
            for neighbor in adjacency[cell]:
                if neighbor in unseen:
                    unseen.remove(neighbor)
                    queue.append(neighbor)
                    group.append(neighbor)
        groups.append(sorted(group))
    return groups


def _occupied_count(walls: list[list[int]], markers: list[dict]) -> int:
    groups = _components(walls)
    group_of = {cell: index for index, group in enumerate(groups) for cell in group}
    occupied = {
        group_of[(int(marker["cell"][0]), int(marker["cell"][1]))]
        for marker in markers
    }
    return len(occupied)


class _DisjointSet:
    def __init__(self) -> None:
        self.parent = list(range(GRID_N * GRID_N))
        self.count = GRID_N * GRID_N

    def find(self, value: int) -> int:
        while self.parent[value] != value:
            self.parent[value] = self.parent[self.parent[value]]
            value = self.parent[value]
        return value

    def union(self, left: int, right: int) -> bool:
        a, b = self.find(left), self.find(right)
        if a == b:
            return False
        self.parent[b] = a
        self.count -= 1
        return True


def _cell_id(row: int, col: int) -> int:
    return row * GRID_N + col


def _target_occupied_count(seed: int) -> int:
    # Consecutive seeds yield yes/no balance; both one- and three-region no cases occur.
    return (2, 1, 2, 3)[seed % 4]


def sample_scene(seed: int) -> dict:
    """Sample a 4x4 walled cell complex with three global regions.

    Every scene has nine internal walls and four dots. Only which global regions
    the dots occupy controls the label, counterbalancing total wall and marker ink.
    """

    rng = random.Random(int(seed))
    target = _target_occupied_count(int(seed))
    for _ in range(5000):
        shuffled = list(ALL_EDGES)
        rng.shuffle(shuffled)
        dsu = _DisjointSet()
        open_edges: set[tuple[int, int, int, int]] = set()
        for edge in shuffled:
            if dsu.count <= 3:
                break
            if dsu.union(_cell_id(edge[0], edge[1]), _cell_id(edge[2], edge[3])):
                open_edges.add(edge)

        roots: dict[int, list[tuple[int, int]]] = {}
        for row in range(GRID_N):
            for col in range(GRID_N):
                roots.setdefault(dsu.find(_cell_id(row, col)), []).append((row, col))
        groups = list(roots.values())
        if len(groups) != 3 or min(map(len, groups)) < 2 or max(map(len, groups)) < 4:
            continue

        redundant = [
            edge
            for edge in ALL_EDGES
            if edge not in open_edges
            and dsu.find(_cell_id(edge[0], edge[1]))
            == dsu.find(_cell_id(edge[2], edge[3]))
        ]
        if len(redundant) < 2:
            continue
        rng.shuffle(redundant)
        open_edges.update(redundant[:2])
        walls = [list(edge) for edge in ALL_EDGES if edge not in open_edges]
        final_groups = _components(walls)
        if len(walls) != 9 or len(final_groups) != 3:
            continue

        rng.shuffle(final_groups)
        if target == 1:
            eligible = [group for group in final_groups if len(group) >= 4]
            if not eligible:
                continue
            cells = rng.sample(rng.choice(eligible), 4)
        elif target == 2:
            pair_options = [
                pair
                for pair in itertools.combinations(final_groups, 2)
                if len(pair[0]) >= 2 and len(pair[1]) >= 2
            ]
            if not pair_options:
                continue
            first, second = rng.choice(pair_options)
            cells = rng.sample(first, 2) + rng.sample(second, 2)
        else:
            doubled = rng.choice([group for group in final_groups if len(group) >= 2])
            cells = rng.sample(doubled, 2)
            for group in final_groups:
                if group is not doubled:
                    cells.append(rng.choice(group))

        rng.shuffle(cells)
        colors = list(PALETTE)
        rng.shuffle(colors)
        markers = [
            {"cell": [cell[0], cell[1]], "color": color}
            for cell, color in zip(cells, colors, strict=True)
        ]
        scene = {
            "walls": walls,
            "markers": markers,
            "wall_width": (8, 10, 12, 14)[rng.randrange(4)],
        }
        if _occupied_count(scene["walls"], scene["markers"]) == target:
            return scene
    raise RuntimeError(f"could not sample a valid scene for seed {seed}")


def _wall_box(edge: list[int] | tuple[int, ...], width: int) -> tuple[int, int, int, int]:
    row, col, next_row, next_col = _canonical_edge(edge)
    half = width // 2
    if row == next_row:
        x = GRID_LEFT + max(col, next_col) * CELL
        y0, y1 = GRID_TOP + row * CELL, GRID_TOP + (row + 1) * CELL
        return (x - half, y0 - half, x + half - 1, y1 + half - 1)
    y = GRID_TOP + max(row, next_row) * CELL
    x0, x1 = GRID_LEFT + col * CELL, GRID_LEFT + (col + 1) * CELL
    return (x0 - half, y - half, x1 + half - 1, y + half - 1)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (CANVAS, CANVAS), BACKGROUND)
    draw = ImageDraw.Draw(image)
    width = int(scene["wall_width"])
    half = width // 2
    right = GRID_LEFT + GRID_N * CELL
    bottom = GRID_TOP + GRID_N * CELL
    outer = (
        (GRID_LEFT - half, GRID_TOP - half, right + half - 1, GRID_TOP + half - 1),
        (GRID_LEFT - half, bottom - half, right + half - 1, bottom + half - 1),
        (GRID_LEFT - half, GRID_TOP - half, GRID_LEFT + half - 1, bottom + half - 1),
        (right - half, GRID_TOP - half, right + half - 1, bottom + half - 1),
    )
    for box in outer:
        draw.rectangle(box, fill=WALL)
    for edge in scene["walls"]:
        draw.rectangle(_wall_box(edge, width), fill=WALL)
    for marker in scene["markers"]:
        row, col = int(marker["cell"][0]), int(marker["cell"][1])
        x = GRID_LEFT + col * CELL + CELL // 2
        y = GRID_TOP + row * CELL + CELL // 2
        r = MARKER_RADIUS
        draw.ellipse((x - r, y - r, x + r, y + r), fill=str(marker["color"]))
    return image


def analytic_gold(scene: dict) -> str:
    walls = scene["walls"]
    markers = scene["markers"]
    return "yes" if _occupied_count(walls, markers) == 2 else "no"


def _toggled_walls(walls: list[list[int]], edge: tuple[int, int, int, int]) -> list[list[int]]:
    blocked = {_canonical_edge(item) for item in walls}
    if edge in blocked:
        blocked.remove(edge)
    else:
        blocked.add(edge)
    return [list(item) for item in ALL_EDGES if item in blocked]


def _runner_up_wall_edits(walls: list[list[int]], markers: list[dict]) -> int:
    original = _occupied_count(walls, markers) == 2
    for edge in ALL_EDGES:
        if (_occupied_count(_toggled_walls(walls, edge), markers) == 2) != original:
            return 1
    for left, right in itertools.combinations(ALL_EDGES, 2):
        once = _toggled_walls(walls, left)
        twice = _toggled_walls(once, right)
        if (_occupied_count(twice, markers) == 2) != original:
            return 2
    return 3


def margin(scene: dict) -> float:
    """Runner-up margin: minimum visible internal-wall toggles that flip the answer."""

    return float(_runner_up_wall_edits(scene["walls"], scene["markers"]))


def is_quarantined(scene: dict) -> bool:
    return int(scene["wall_width"]) < 8


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["walls"] = [
        [edge[2], edge[3], edge[0], edge[1]] for edge in reversed(twin["walls"])
    ]
    twin["markers"] = list(reversed(twin["markers"]))
    return [("serialization_order", twin)]


def raster_clearance(scene: dict) -> float:
    """Minimum half-width of a blocking wall in output pixels."""

    return float(scene["wall_width"]) / 2.0
