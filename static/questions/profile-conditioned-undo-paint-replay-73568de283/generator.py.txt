"""Deterministic Pillow world for replaying SET and LIFO UNDO operations."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 540
PALETTE = ("red", "green", "blue", "orange")
COLORS = {
    "red": (221, 78, 72),
    "green": (63, 166, 112),
    "blue": (66, 121, 210),
    "orange": (236, 151, 53),
}
BACKGROUND = (247, 245, 239)
INK = (35, 39, 47)
MUTED = (104, 111, 122)
CARD = (255, 255, 255)
SET_HEADER = (35, 155, 184)
UNDO_HEADER = (151, 80, 178)


def _font(size: int):
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _replay(initial_color: str, operations: list[dict]) -> str:
    history = [initial_color]
    for operation in operations:
        if operation["kind"] == "set":
            history.append(operation["color"])
        elif operation["kind"] == "undo":
            if len(history) > 1:
                history.pop()
        else:
            raise ValueError("unknown operation")
    return history[-1]


def sample_scene(seed: int) -> dict:
    """Sample four SETs and two UNDOs, then balance answer color by label rotation."""
    rng = random.Random((int(seed) ^ 0x51A7E) * 1_000_003)
    target = PALETTE[int(seed) % len(PALETTE)]
    for _ in range(20_000):
        initial = rng.choice(PALETTE)
        set_colors = list(PALETTE)
        rng.shuffle(set_colors)
        kinds = ["set"] * 4 + ["undo"] * 2
        rng.shuffle(kinds)
        operations = []
        color_iter = iter(set_colors)
        for kind in kinds:
            operations.append(
                {"kind": "set", "color": next(color_iter)} if kind == "set" else {"kind": "undo"}
            )

        # Both rollback cards must pop real edits, and rollback must matter.
        depth = 0
        effective_undos = 0
        for operation in operations:
            if operation["kind"] == "set":
                depth += 1
            elif depth:
                depth -= 1
                effective_undos += 1
        rollback = _replay(initial, operations)
        last_set = next(op["color"] for op in reversed(operations) if op["kind"] == "set")
        if effective_undos != 2 or rollback == last_set:
            continue

        # Apply one global label rotation so answer identity is exactly balanced
        # without changing card geometry, operation order, or color counts.
        shift = (PALETTE.index(target) - PALETTE.index(rollback)) % len(PALETTE)
        mapping = {color: PALETTE[(i + shift) % len(PALETTE)] for i, color in enumerate(PALETTE)}
        return {
            "initial_color": mapping[initial],
            "operations": [
                ({"kind": "set", "color": mapping[op["color"]]} if op["kind"] == "set" else {"kind": "undo"})
                for op in operations
            ],
        }
    raise RuntimeError("could not sample a causally informative scene")


def render(scene: dict) -> Image.Image:
    initial_color = scene["initial_color"]
    operations = scene["operations"]
    if initial_color not in PALETTE or len(operations) != 6:
        raise ValueError("invalid scene")

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title = _font(28)
    small = _font(15)
    medium = _font(19)
    draw.text((40, 28), "PAINT HISTORY", fill=INK, font=title)
    draw.text((40, 67), "START, then read the cards from left to right", fill=MUTED, font=small)

    # Initial canvas is deliberately the largest answer-palette component.
    draw.rounded_rectangle((38, 124, 186, 391), radius=13, fill=CARD, outline=INK, width=3)
    draw.rectangle((40, 126, 184, 174), fill=INK)
    draw.text((80, 140), "START", fill=CARD, font=small)
    draw.rectangle((64, 212, 160, 308), fill=COLORS[initial_color])
    draw.rectangle((64, 212, 160, 308), outline=INK, width=3)
    draw.text((78, 332), "canvas", fill=MUTED, font=small)

    card_xs = [214 + 119 * index for index in range(6)]
    for index, (x, operation) in enumerate(zip(card_xs, operations), start=1):
        draw.rounded_rectangle((x, 124, x + 103, 391), radius=12, fill=CARD, outline=INK, width=3)
        header = SET_HEADER if operation["kind"] == "set" else UNDO_HEADER
        # Exact solid inset is the oracle's categorical evidence.
        draw.rectangle((x + 3, 127, x + 100, 174), fill=header)
        label = "SET" if operation["kind"] == "set" else "UNDO"
        tx = x + (35 if label == "SET" else 21)
        draw.text((tx, 141), label, fill=CARD, font=small)
        if operation["kind"] == "set":
            color = operation["color"]
            draw.rectangle((x + 24, 225, x + 79, 280), fill=COLORS[color])
            draw.rectangle((x + 24, 225, x + 79, 280), outline=INK, width=3)
            draw.text((x + 25, 311), "paint", fill=MUTED, font=small)
        else:
            # A large back-arrow reinforces the label without creating answer-color pixels.
            draw.arc((x + 24, 218, x + 80, 274), 35, 310, fill=INK, width=7)
            draw.polygon([(x + 21, 237), (x + 39, 225), (x + 38, 247)], fill=INK)
            draw.text((x + 27, 311), "pop", fill=MUTED, font=small)
        draw.ellipse((x + 42, 357, x + 61, 376), fill=INK)
        digit = str(index)
        bbox = draw.textbbox((0, 0), digit, font=small)
        draw.text((x + 51 - (bbox[2] - bbox[0]) / 2, 358), digit, fill=CARD, font=small)

    draw.rounded_rectangle((38, 424, 919, 500), radius=12, fill=(235, 231, 222))
    draw.text((60, 442), "SET pushes a new color", fill=INK, font=medium)
    draw.text((353, 442), "UNDO removes the latest SET", fill=INK, font=medium)
    draw.text((734, 442), "? final color", fill=INK, font=medium)
    return image


def analytic_gold(scene: dict) -> str:
    return _replay(scene["initial_color"], scene["operations"])


def margin(scene: dict) -> float:
    # Runner-up margin is the minimum categorical swatch/header width in pixels.
    _ = scene["initial_color"]
    _ = scene["operations"]
    return 47.0


def is_quarantined(scene: dict) -> bool:
    initial_color = scene["initial_color"]
    operations = scene["operations"]
    return (
        initial_color not in PALETTE
        or len(operations) != 6
        or sum(op.get("kind") == "set" for op in operations) != 4
        or sum(op.get("kind") == "undo" for op in operations) != 2
        or any(op.get("kind") == "set" and op.get("color") not in PALETTE for op in operations)
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    _ = deepcopy(scene)
    return []
