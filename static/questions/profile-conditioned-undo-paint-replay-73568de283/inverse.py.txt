"""Independent inverse arm: recover the rollback result from RGB pixels only."""

from __future__ import annotations

from collections import deque

import numpy as np

ANSWER_RGB = {
    "red": (221, 78, 72),
    "green": (63, 166, 112),
    "blue": (66, 121, 210),
    "orange": (236, 151, 53),
}
SET_RGB = (35, 155, 184)
UNDO_RGB = (151, 80, 178)


def _components(mask: np.ndarray) -> list[dict]:
    """Return 4-connected components without relying on layout coordinates."""
    work = mask.copy()
    height, width = work.shape
    found = []
    for y0, x0 in np.argwhere(work):
        y0, x0 = int(y0), int(x0)
        if not work[y0, x0]:
            continue
        work[y0, x0] = False
        queue = deque([(y0, x0)])
        area = 0
        min_x = max_x = x0
        min_y = max_y = y0
        while queue:
            y, x = queue.popleft()
            area += 1
            min_x, max_x = min(min_x, x), max(max_x, x)
            min_y, max_y = min(min_y, y), max(max_y, y)
            for yy, xx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= yy < height and 0 <= xx < width and work[yy, xx]:
                    work[yy, xx] = False
                    queue.append((yy, xx))
        found.append({"area": area, "bbox": (min_x, min_y, max_x, max_y), "cx": (min_x + max_x) / 2})
    return found


def _large_components(array: np.ndarray, rgb: tuple[int, int, int], minimum: int) -> list[dict]:
    mask = np.all(array == np.asarray(rgb, dtype=np.uint8), axis=2)
    return [component for component in _components(mask) if component["area"] >= minimum]


def decision_from_image(image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    headers = []
    for kind, rgb in (("set", SET_RGB), ("undo", UNDO_RGB)):
        for component in _large_components(array, rgb, 2_500):
            headers.append((component["cx"], kind, component["bbox"]))
    headers.sort()
    if len(headers) != 6 or any(headers[i + 1][0] - headers[i][0] < 60 for i in range(5)):
        raise ValueError("abstain: six distinct operation headers were not recovered")

    color_components = []
    for name, rgb in ANSWER_RGB.items():
        for component in _large_components(array, rgb, 1_500):
            color_components.append((name, component))
    initial_candidates = [(name, comp) for name, comp in color_components if comp["area"] >= 8_000]
    if len(initial_candidates) != 1:
        raise ValueError("abstain: unique starting canvas was not recovered")
    initial_color = initial_candidates[0][0]

    operations = []
    for cx, kind, bbox in headers:
        if kind == "undo":
            operations.append({"kind": "undo"})
            continue
        left, top, right, bottom = bbox
        matches = [
            name
            for name, comp in color_components
            if 1_500 <= comp["area"] < 8_000
            and left <= comp["cx"] <= right
            and comp["bbox"][1] > bottom
        ]
        if len(matches) != 1:
            raise ValueError("abstain: SET card does not bind one color swatch")
        operations.append({"kind": "set", "color": matches[0]})

    history = [initial_color]
    for operation in operations:
        if operation["kind"] == "set":
            history.append(operation["color"])
        elif len(history) > 1:
            history.pop()
    return history[-1]
