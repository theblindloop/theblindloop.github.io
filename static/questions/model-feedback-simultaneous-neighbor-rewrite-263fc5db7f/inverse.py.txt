"""Image-only inverse arm for the simultaneous rewrite board."""

from __future__ import annotations

import numpy as np
from PIL import Image

EXPECTED_SIZE = (680, 510)
CELL_SIZE, CELL_GAP, FIRST_X = 46, 7, 240
START_TOP = 50
OPTION_TOPS = (172, 236, 300, 364, 428)
INK = np.array((24, 59, 86), dtype=np.float32)
PAPER = np.array((248, 246, 238), dtype=np.float32)


class OracleError(ValueError):
    pass


def _classify_cell(array: np.ndarray, column: int, top: int) -> int:
    left = FIRST_X + column * (CELL_SIZE + CELL_GAP)
    patch = array[top + 12 : top + 34, left + 12 : left + 34].astype(np.float32)
    mean = patch.mean(axis=(0, 1))
    ink_distance = float(np.linalg.norm(mean - INK))
    paper_distance = float(np.linalg.norm(mean - PAPER))
    if ink_distance < 8.0 and ink_distance + 25.0 < paper_distance:
        return 1
    if paper_distance < 8.0 and paper_distance + 25.0 < ink_distance:
        return 0
    raise OracleError(f"cell at column {column}, y={top} has no clean binary fill")


def _read_row(array: np.ndarray, top: int) -> list[int]:
    return [_classify_cell(array, column, top) for column in range(7)]


def _validate_labels(array: np.ndarray) -> None:
    for top in OPTION_TOPS:
        cy = top + CELL_SIZE // 2
        navy = array[cy - 18 : cy + 19, 128:165].astype(np.float32)
        distance = np.linalg.norm(navy - INK, axis=2)
        # A large navy disk must surround a high-contrast white glyph.
        navy_count = int((distance < 18.0).sum())
        white_count = int((navy.mean(axis=2) > 235).sum())
        if navy_count < 700 or white_count < 35:
            raise OracleError("an option label disk or glyph is missing")


def _next_state(bits: list[int]) -> list[int]:
    if len(bits) != 7 or any(value not in (0, 1) for value in bits):
        raise OracleError("invalid recovered state")
    next_bits = []
    for position in range(7):
        left = bits[position - 1] if position > 0 else 0
        self_value = bits[position]
        right = bits[position + 1] if position < 6 else 0
        next_bits.append(int(left + self_value + right == 1))
    return next_bits


def decision_from_image(image: Image.Image) -> str:
    if image.size != EXPECTED_SIZE:
        raise OracleError(f"unexpected raster size {image.size}")
    array = np.asarray(image.convert("RGB"))
    _validate_labels(array)
    start = _read_row(array, START_TOP)
    options = [_read_row(array, top) for top in OPTION_TOPS]
    if len({tuple(row) for row in options}) != 5:
        raise OracleError("option rows are not distinct")
    state = start
    for _ in range(4):
        state = _next_state(state)
    matches = [index for index, option in enumerate(options) if option == state]
    if len(matches) != 1:
        raise OracleError("recovered fourth state does not match exactly one option")
    return "ABCDE"[matches[0]]
