"""Deterministic sampler, renderer, and analytic transition program."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 680, 510
CELL, GAP, GRID_X = 46, 7, 240
START_Y = 50
OPTION_YS = (172, 236, 300, 364, 428)
DARK = "#183b56"
LIGHT = "#f8f6ee"
BORDER = "#536879"
AMBER = "#c47716"
BACKGROUND = "#ffffff"
FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"


def _step(bits: Iterable[int]) -> list[int]:
    old = [int(value) for value in bits]
    if len(old) != 7 or any(value not in (0, 1) for value in old):
        raise ValueError("a state must contain seven binary cells")
    result = []
    for index in range(7):
        triple = (old[index - 1] if index else 0, old[index], old[index + 1] if index < 6 else 0)
        result.append(int(sum(triple) == 1))
    return result


def _evolve(bits: Iterable[int], steps: int) -> list[int]:
    state = list(bits)
    for _ in range(steps):
        state = _step(state)
    return state


def _asynchronous(bits: Iterable[int], steps: int) -> list[int]:
    state = [int(value) for value in bits]
    for _ in range(steps):
        for index in range(7):
            left = state[index - 1] if index else 0
            centre = state[index]
            right = state[index + 1] if index < 6 else 0
            state[index] = int(left + centre + right == 1)
    return state


def _at_least_one(bits: Iterable[int], steps: int) -> list[int]:
    state = [int(value) for value in bits]
    for _ in range(steps):
        old = state
        state = [
            int((old[index - 1] if index else 0) + old[index] + (old[index + 1] if index < 6 else 0) >= 1)
            for index in range(7)
        ]
    return state


def _valid_scene(scene: dict) -> tuple[list[int], list[list[int]]]:
    if set(scene) != {"start_bits", "options"}:
        raise ValueError("scene fields must be exactly start_bits and options")
    start = [int(value) for value in scene["start_bits"]]
    options = [[int(value) for value in row] for row in scene["options"]]
    if len(start) != 7 or any(value not in (0, 1) for value in start):
        raise ValueError("invalid START row")
    if len(options) != 5 or any(len(row) != 7 for row in options):
        raise ValueError("invalid option matrix")
    if any(value not in (0, 1) for row in options for value in row):
        raise ValueError("invalid option cell")
    if len({tuple(row) for row in options}) != 5:
        raise ValueError("options must be distinct")
    return start, options


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    for _ in range(400):
        start = [rng.randrange(2) for _ in range(7)]
        target = _evolve(start, 4)
        async_wrong = _asynchronous(start, 4)
        if not (2 <= sum(start) <= 5 and 2 <= sum(target) <= 5):
            continue
        preferred = [
            _evolve(start, 3),
            _evolve(start, 5),
            async_wrong,
            _at_least_one(start, 4),
            [1 - value for value in target],
        ]
        distractors: list[list[int]] = []
        for row in preferred:
            if row != target and row not in distractors:
                distractors.append(row)
        if async_wrong == target or len(distractors) < 3:
            continue
        pool = [[(mask >> (6 - bit)) & 1 for bit in range(7)] for mask in range(128)]
        rng.shuffle(pool)
        pool.sort(key=lambda row: (abs(sum(a != b for a, b in zip(row, target)) - 2), abs(sum(row) - sum(target))))
        for row in pool:
            if row != target and row not in distractors:
                distractors.append(row)
            if len(distractors) == 4:
                break
        distractors = distractors[:4]
        rng.shuffle(distractors)
        correct_position = int(seed) % 5
        options = list(distractors)
        options.insert(correct_position, target)
        return {"start_bits": start, "options": options}
    raise RuntimeError("could not sample a nondegenerate rewrite board")


def analytic_gold(scene: dict) -> str:
    start = [int(value) for value in scene["start_bits"]]
    options = [[int(value) for value in row] for row in scene["options"]]
    target = _evolve(start, 4)
    matches = [index for index, row in enumerate(options) if row == target]
    if len(matches) != 1:
        raise ValueError("the fourth state must occur in exactly one option")
    return "ABCDE"[matches[0]]


def margin(scene: dict) -> float:
    start = [int(value) for value in scene["start_bits"]]
    options = [[int(value) for value in row] for row in scene["options"]]
    target = _evolve(start, 4)
    distances = [sum(a != b for a, b in zip(target, row)) for row in options if row != target]
    return float(min(distances)) if distances else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        start, options = _valid_scene(scene)
        target = _evolve(start, 4)
        return sum(row == target for row in options) != 1 or margin(scene) < 1.0
    except (TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    _valid_scene(scene)
    return []


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(FONT_PATH, size=size)


def _draw_row(draw: ImageDraw.ImageDraw, bits: Iterable[int], y: int, *, start: bool = False) -> None:
    for index, value in enumerate(bits):
        x = GRID_X + index * (CELL + GAP)
        outline = AMBER if start else BORDER
        draw.rounded_rectangle(
            (x, y, x + CELL, y + CELL),
            radius=4,
            fill=DARK if value else LIGHT,
            outline=outline,
            width=3,
        )


def render(scene: dict) -> Image.Image:
    start = [int(value) for value in scene["start_bits"]]
    options = [[int(value) for value in row] for row in scene["options"]]
    _valid_scene({"start_bits": start, "options": options})
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((340, 20), "UPDATE ALL CELLS AT ONCE", fill=BORDER, font=_font(18), anchor="mm")
    draw.text((146, START_Y + CELL / 2), "START", fill=AMBER, font=_font(24), anchor="mm")
    _draw_row(draw, start, START_Y, start=True)
    draw.line((54, 121, 626, 121), fill="#d8dee3", width=2)
    draw.text((54, 142), "OPTIONS — fourth new row", fill=BORDER, font=_font(16), anchor="lm")
    for label, row, y in zip("ABCDE", options, OPTION_YS):
        centre = (146, y + CELL // 2)
        draw.ellipse((centre[0] - 22, centre[1] - 22, centre[0] + 22, centre[1] + 22), fill=DARK)
        draw.text(centre, label, fill="white", font=_font(25), anchor="mm", stroke_width=0)
        _draw_row(draw, row, y)
    return image
