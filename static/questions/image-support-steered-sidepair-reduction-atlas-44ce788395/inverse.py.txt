"""Independent image-only inverse arm based on visible color components."""

from __future__ import annotations

import numpy as np
from PIL import Image

NAVY = np.array((20, 43, 69), dtype=np.int16)
CYAN = np.array((69, 214, 208), dtype=np.int16)
CREAM = np.array((243, 229, 192), dtype=np.int16)
TAG_COLORS = {
    "red": np.array((233, 93, 98), dtype=np.int16),
    "green": np.array((84, 182, 106), dtype=np.int16),
    "blue": np.array((86, 133, 216), dtype=np.int16),
}


def _mask(arr: np.ndarray, color: np.ndarray, tolerance: int = 12) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - color), axis=2) <= tolerance


def _components(mask: np.ndarray, minimum: int = 20) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for start_y, start_x in np.argwhere(mask):
        if seen[start_y, start_x]:
            continue
        stack = [(int(start_y), int(start_x))]
        seen[start_y, start_x] = True
        xs, ys = [], []
        while stack:
            y, x = stack.pop()
            xs.append(x); ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(xs) >= minimum:
            output.append((min(xs), min(ys), max(xs), max(ys), len(xs)))
    return output


def _ordered_grid(items: list[tuple[float, float, int]], rows: int, cols: int) -> list[list[int]]:
    if len(items) != rows * cols:
        raise ValueError(f"expected {rows * cols} tiles, found {len(items)}")
    ordered = sorted(items, key=lambda item: item[1])
    row_groups = [ordered[index * cols:(index + 1) * cols] for index in range(rows)]
    if any(max(value[1] for value in group) - min(value[1] for value in group) > 4 for group in row_groups):
        raise ValueError("tile rows are not aligned")
    return [[value for _, _, value in sorted(group)] for group in row_groups]


def _reduce(source: list[list[int]]) -> list[list[int]]:
    result = []
    for br in range(3):
        row = []
        for bc in range(3):
            a = source[2 * br][2 * bc]
            b = source[2 * br][2 * bc + 1]
            c = source[2 * br + 1][2 * bc]
            d = source[2 * br + 1][2 * bc + 1]
            row.append(int(a + b + c + d == 2 and ((a and b) or (c and d) or (a and c) or (b and d))))
        result.append(row)
    return result


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    navy = _mask(arr, NAVY)
    navy_points = np.argwhere(navy)
    if len(navy_points) < arr.shape[0] * arr.shape[1] * 0.25:
        raise ValueError("uninterrupted atlas is absent")
    y0, x0 = navy_points.min(axis=0)
    y1, x1 = navy_points.max(axis=0)
    # The visible atlas explicitly has no white holes. This catches erasures in
    # gutters as well as erased evidence tiles without relying on latent layout.
    interior = arr[y0:y1 + 1, x0:x1 + 1]
    white = np.all(interior >= 245, axis=2)
    if float(white.mean()) > 0.002:
        raise ValueError("atlas fabric is broken")

    tile_components = []
    for value, color in ((1, CYAN), (0, CREAM)):
        for left, top, right, bottom, area in _components(_mask(arr, color), minimum=800):
            width, height = right - left + 1, bottom - top + 1
            if 30 <= width <= 46 and 30 <= height <= 46 and area >= width * height * 0.96:
                tile_components.append(((left + right) / 2, (top + bottom) / 2, value, area))
    if len(tile_components) != 63:
        raise ValueError(f"expected 63 visible tiles, found {len(tile_components)}")
    source_items = [(x, y, value) for x, y, value, area in tile_components if area > 1500]
    candidate_items = [(x, y, value) for x, y, value, area in tile_components if area <= 1500]
    source = _ordered_grid(source_items, 6, 6)
    if len(candidate_items) != 27:
        raise ValueError("candidate tile count is malformed")
    by_y = sorted(candidate_items, key=lambda item: item[1])
    candidates = [_ordered_grid(by_y[i * 9:(i + 1) * 9], 3, 3) for i in range(3)]

    tags = []
    for name, color in TAG_COLORS.items():
        components = _components(_mask(arr, color), minimum=500)
        if len(components) != 1:
            raise ValueError(f"expected one {name} tag")
        left, top, right, bottom, _ = components[0]
        tags.append(((top + bottom) / 2, name))
    tags = [name for _, name in sorted(tags)]
    target = _reduce(source)
    matches = [index for index, matrix in enumerate(candidates) if matrix == target]
    if len(matches) != 1:
        raise ValueError("no unique pixel-derived match")
    return tags[matches[0]]
