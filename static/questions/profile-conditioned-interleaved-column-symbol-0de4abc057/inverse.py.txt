"""Independent pixel-only inverse for the column-interleave raster.

This module deliberately duplicates the public transform contract instead of
importing any candidate renderer or analytic code.  It sees only the final
RGB image: saturated connected mark coordinates locate the square, the raster
columns are deinterleaved, a five-pixel box mean is applied, and four local
template distances produce the decision.
"""

from __future__ import annotations

import statistics
from collections import deque

from PIL import Image, ImageDraw, ImageFilter

_SYMBOLS = ("bell", "leaf", "ladder", "hook")


def _clusters(values: list[int], gap: int = 20) -> list[list[int]]:
    unique = sorted(set(values))
    if not unique:
        return []
    groups: list[list[int]] = [[unique[0]]]
    for value in unique[1:]:
        if value - groups[-1][-1] > gap:
            groups.append([value])
        else:
            groups[-1].append(value)
    return groups


def _marker_bounds(image: Image.Image) -> tuple[int, int, int]:
    saturated: list[tuple[int, int]] = []
    for y in range(image.height):
        for x in range(image.width):
            r, g, b = image.getpixel((x, y))
            if max(r, g, b) - min(r, g, b) > 45 and max(r, g, b) > 120:
                saturated.append((x, y))
    if len(saturated) < 40:
        raise ValueError("fewer than four colored fiducial regions were found")
    x_groups = _clusters([x for x, _ in saturated])
    y_groups = _clusters([y for _, y in saturated])
    if len(x_groups) != 2 or len(y_groups) != 2:
        raise ValueError("colored fiducials do not form two horizontal and vertical pairs")
    x_centers = []
    y_centers = []
    for group in x_groups:
        x_centers.append(sum(x for x, _ in saturated if x in group) / sum(1 for x, _ in saturated if x in group))
    for group in y_groups:
        y_centers.append(sum(y for _, y in saturated if y in group) / sum(1 for _, y in saturated if y in group))
    x0 = int(round(min(x_centers) + 0.5))
    y0 = int(round(min(y_centers) + 0.5))
    # The opposite marks are centered on the last pixel (x0+S-1), so their
    # pixel-centre separation is S-1 rather than S.
    side_x = int(round(max(x_centers) - min(x_centers))) + 1
    side_y = int(round(max(y_centers) - min(y_centers))) + 1
    if side_x != side_y or side_x < 64 or side_x % 2:
        raise ValueError(f"fiducials imply invalid square {side_x}x{side_y}")
    return x0, y0, side_x


def _extract_gray(image: Image.Image, x0: int, y0: int, side: int) -> Image.Image:
    rows: list[int] = []
    colored: list[int] = []
    for y in range(y0, y0 + side):
        for x in range(x0, x0 + side):
            r, g, b = image.getpixel((x, y))
            colored_flag = max(r, g, b) - min(r, g, b) > 45 and max(r, g, b) > 120
            gray = int(round((r + g + b) / 3.0))
            rows.append(gray)
            if not colored_flag:
                colored.append(gray)
    background = int(round(statistics.median(colored)))
    source = Image.new("L", (side, side))
    pixels = source.load()
    index = 0
    for y in range(side):
        for x in range(side):
            r, g, b = image.getpixel((x0 + x, y0 + y))
            if max(r, g, b) - min(r, g, b) > 45 and max(r, g, b) > 120:
                pixels[x, y] = background
            else:
                pixels[x, y] = rows[index]
            index += 1
    return source


def _undo_columns(scrambled: Image.Image) -> Image.Image:
    side = scrambled.width
    half = side // 2
    restored = Image.new("L", (side, side))
    source = scrambled.load()
    target = restored.load()
    for y in range(side):
        for x in range(half):
            target[x, y] = source[2 * x, y]
            target[x + half, y] = source[2 * x + 1, y]
    return restored


def _draw_symbol(draw: ImageDraw.ImageDraw, size: int, symbol: str, ink: int, width: int) -> None:
    def q(x: float, y: float) -> tuple[int, int]:
        return (int(round(x * size)), int(round(y * size)))

    if symbol == "bell":
        draw.arc((int(.25 * size), int(.22 * size), int(.75 * size), int(.77 * size)), 205, 335, fill=ink, width=width)
        draw.arc((int(.33 * size), int(.15 * size), int(.67 * size), int(.43 * size)), 180, 360, fill=ink, width=width)
        draw.line([q(.31, .70), q(.69, .70)], fill=ink, width=width)
        draw.line([q(.46, .17), q(.54, .17)], fill=ink, width=width)
        draw.ellipse((int(.46 * size), int(.74 * size), int(.54 * size), int(.82 * size)), fill=ink)
    elif symbol == "leaf":
        points = [q(.50, .16), q(.76, .43), q(.62, .72), q(.38, .82), q(.23, .57), q(.27, .34), q(.50, .16)]
        draw.line(points, fill=ink, width=width, joint="curve")
        draw.line([q(.27, .72), q(.68, .30)], fill=ink, width=width)
    elif symbol == "ladder":
        draw.line([q(.31, .23), q(.31, .78)], fill=ink, width=width)
        draw.line([q(.69, .23), q(.69, .78)], fill=ink, width=width)
        for y in (.32, .44, .56, .68):
            draw.line([q(.31, y), q(.69, y)], fill=ink, width=width)
    elif symbol == "hook":
        draw.line([q(.39, .20), q(.39, .66)], fill=ink, width=width)
        draw.arc((int(.35 * size), int(.53 * size), int(.74 * size), int(.84 * size)), 0, 180, fill=ink, width=width)
        draw.line([q(.56, .78), q(.72, .78)], fill=ink, width=width)
    else:
        raise ValueError(symbol)


def _template(side: int, symbol: str) -> Image.Image:
    background = 220
    template = Image.new("L", (side, side), background)
    draw = ImageDraw.Draw(template)
    width = max(6, int(round(side * 0.052)))
    _draw_symbol(draw, side, symbol, 56, width)
    return template.filter(ImageFilter.BoxBlur(2))


def _binary_crop(image: Image.Image, side: int) -> list[int]:
    pad = max(10, int(round(side * 0.10)))
    crop = image.crop((pad, pad, side - pad, side - pad)).resize((96, 96), Image.Resampling.BILINEAR)
    values = list(crop.getdata())
    background = statistics.median(values)
    threshold = background - 34
    return [1 if value < threshold else 0 for value in values]


def _distance(left: list[int], right: list[int]) -> int:
    return sum(a != b for a, b in zip(left, right))


def decision_from_image(image: Image.Image) -> str:
    """Return the restored symbol from the final PNG, without scene data."""

    rgb = image.convert("RGB")
    x0, y0, side = _marker_bounds(rgb)
    scrambled = _extract_gray(rgb, x0, y0, side)
    restored = _undo_columns(scrambled).filter(ImageFilter.BoxBlur(2))
    observed = _binary_crop(restored, side)
    distances = []
    for symbol in _SYMBOLS:
        distances.append((_distance(observed, _binary_crop(_template(side, symbol), side)), symbol))
    distances.sort()
    best_distance, best_symbol = distances[0]
    runner_distance = distances[1][0]
    # A weak ambiguity guard keeps the arm honest on materially corrupted or
    # near-tied images while all generated scenes remain comfortably decisive.
    if runner_distance - best_distance < max(8, len(observed) // 250):
        raise ValueError("restored symbol template margin is ambiguous")
    return best_symbol
