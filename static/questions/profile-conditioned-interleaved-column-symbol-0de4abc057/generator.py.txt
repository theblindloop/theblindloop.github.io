"""Renderer for the marked column-interleave symbol world.

The scene contains a grayscale square whose source columns are written in a
two-way perfect-shuffle order.  The image is intentionally a plain raster;
the declared inverse operation is the only route used by the independent
oracle to identify the symbol.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "interleaved-column-symbol-0.1.0"
CANVAS = (360, 360)
SYMBOLS = ("bell", "leaf", "ladder", "hook")
MARK_PALETTE = (
    (226, 74, 74),
    (67, 133, 225),
    (68, 177, 112),
    (230, 163, 59),
)


def _mix(value: int) -> int:
    """A small deterministic integer mixer used to separate labels and nuisances."""

    value = (value + 0x9E3779B97F4A7C15) & ((1 << 64) - 1)
    value = ((value ^ (value >> 30)) * 0xBF58476D1CE4E5B9) & ((1 << 64) - 1)
    value = ((value ^ (value >> 27)) * 0x94D049BB133111EB) & ((1 << 64) - 1)
    return (value ^ (value >> 31)) & ((1 << 64) - 1)


def _draw_symbol(draw: ImageDraw.ImageDraw, size: int, symbol: str, ink: int, width: int) -> None:
    """Draw one of four broad, answer-bearing line symbols."""

    def q(x: float, y: float) -> tuple[int, int]:
        return (int(round(x * size)), int(round(y * size)))

    if symbol == "bell":
        draw.arc((int(.25 * size), int(.22 * size), int(.75 * size), int(.77 * size)), 205, 335, fill=ink, width=width)
        draw.arc((int(.33 * size), int(.15 * size), int(.67 * size), int(.43 * size)), 180, 360, fill=ink, width=width)
        draw.line([q(.31, .70), q(.69, .70)], fill=ink, width=width)
        draw.line([q(.46, .17), q(.54, .17)], fill=ink, width=width)
        draw.ellipse((int(.46 * size), int(.74 * size), int(.54 * size), int(.82 * size)), fill=ink)
    elif symbol == "leaf":
        points = [q(.50, .16), q(.76, .43), q(.62, .72), q(.38, .82), q(.23, .57), q(.27, .34), q(.50, .16)]
        draw.line(points, fill=ink, width=width, joint="curve")
        draw.line([q(.27, .72), q(.68, .30)], fill=ink, width=width)
    elif symbol == "ladder":
        draw.line([q(.31, .23), q(.31, .78)], fill=ink, width=width)
        draw.line([q(.69, .23), q(.69, .78)], fill=ink, width=width)
        for y in (.32, .44, .56, .68):
            draw.line([q(.31, y), q(.69, y)], fill=ink, width=width)
    elif symbol == "hook":
        draw.line([q(.39, .20), q(.39, .66)], fill=ink, width=width)
        draw.arc((int(.35 * size), int(.53 * size), int(.74 * size), int(.84 * size)), 0, 180, fill=ink, width=width)
        draw.line([q(.56, .78), q(.72, .78)], fill=ink, width=width)
    else:
        raise ValueError(f"unknown symbol {symbol!r}")


def _source_tile(scene: dict[str, Any]) -> Image.Image:
    size = int(scene["tile_size"])
    background = int(scene["tile_background"])
    phase = int(scene["phase"])
    amplitude = int(scene["carrier_amplitude"])
    tile = Image.new("L", (size, size))
    pixels = tile.load()
    for y in range(size):
        for x in range(size):
            # A fine carrier is label-independent and is strongly reduced by
            # the declared mean after the columns have been restored.
            wobble = amplitude if ((x + 2 * y + phase) % 4 in (0, 1)) else -amplitude
            pixels[x, y] = max(0, min(255, background + wobble))
    draw = ImageDraw.Draw(tile)
    _draw_symbol(draw, size, str(scene["symbol"]), int(scene["ink"]), int(scene["line_width"]))
    return tile


def _interleave_columns(tile: Image.Image) -> Image.Image:
    size = tile.width
    if size % 2:
        raise ValueError("tile width must be even")
    half = size // 2
    source = tile.load()
    scrambled = Image.new("L", (size, size))
    target = scrambled.load()
    for y in range(size):
        for x in range(half):
            target[2 * x, y] = source[x, y]
            target[2 * x + 1, y] = source[x + half, y]
    return scrambled


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the final RGB raster seen by the pixel-only inverse arm."""

    # Keep each scene field visibly on the rendering path.  This also makes
    # the v0.4 rendered-field audit honest about the not_applicable alias mode.
    symbol = scene["symbol"]
    tile_x = int(scene["tile_x"])
    tile_y = int(scene["tile_y"])
    tile_size = int(scene["tile_size"])
    background_rgb = tuple(int(v) for v in scene["background_rgb"])
    tile_background = int(scene["tile_background"])
    phase = int(scene["phase"])
    ink = int(scene["ink"])
    mark_colors = [tuple(int(v) for v in color) for color in scene["mark_colors"]]
    mark_size = int(scene["mark_size"])
    line_width = int(scene["line_width"])
    carrier_amplitude = int(scene["carrier_amplitude"])
    border_rgb = tuple(int(v) for v in scene["border_rgb"])

    # The local dictionary is deliberately explicit so the source raster and
    # the outer framing share exactly the sampled nuisance settings.
    local_scene = {
        "symbol": symbol,
        "tile_size": tile_size,
        "tile_background": tile_background,
        "phase": phase,
        "ink": ink,
        "line_width": line_width,
        "carrier_amplitude": carrier_amplitude,
    }
    scrambled = _interleave_columns(_source_tile(local_scene))
    image = Image.new("RGB", CANVAS, background_rgb)
    image.paste(scrambled.convert("RGB"), (tile_x, tile_y))
    draw = ImageDraw.Draw(image)
    draw.rectangle(
        (tile_x, tile_y, tile_x + tile_size - 1, tile_y + tile_size - 1),
        outline=border_rgb,
        width=1,
    )
    half_mark = mark_size // 2
    centers = (
        (tile_x, tile_y),
        (tile_x + tile_size - 1, tile_y),
        (tile_x, tile_y + tile_size - 1),
        (tile_x + tile_size - 1, tile_y + tile_size - 1),
    )
    for (cx, cy), color in zip(centers, mark_colors):
        draw.rectangle(
            (cx - half_mark, cy - half_mark, cx + half_mark - 1, cy + half_mark - 1),
            fill=color,
        )
    return image


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced, deterministic scene from one integer seed."""

    label_hash = _mix(int(seed))
    rng = random.Random(_mix(int(seed) ^ 0xD1B54A32D192ED03))
    tile_size = rng.choice((144, 160, 176))
    tile_x = rng.randint(48, CANVAS[0] - tile_size - 48)
    tile_y = rng.randint(48, CANVAS[1] - tile_size - 48)
    colors = list(MARK_PALETTE)
    rng.shuffle(colors)
    background = rng.choice((242, 248, 250))
    tile_background = rng.choice((211, 217, 223))
    ink = rng.choice((48, 56, 64, 72))
    phase = rng.randrange(4)
    carrier_amplitude = rng.choice((6, 8, 10))
    line_width = max(6, int(round(tile_size * 0.052)))
    return {
        "symbol": SYMBOLS[label_hash % len(SYMBOLS)],
        "tile_x": tile_x,
        "tile_y": tile_y,
        "tile_size": tile_size,
        "background_rgb": [background, background, background],
        "tile_background": tile_background,
        "phase": phase,
        "ink": ink,
        "mark_colors": [list(color) for color in colors],
        "mark_size": 10,
        "line_width": line_width,
        "carrier_amplitude": carrier_amplitude,
        "border_rgb": [116, 116, 116],
    }


def analytic_gold(scene: dict[str, Any]) -> str:
    """Analytic answer; symbol is the sole label input."""

    return str(scene["symbol"])


def margin(scene: dict[str, Any]) -> float:
    """One-sided normalized separation margin for the rendered template bank."""

    # The generated symbols have generous separation.  The small size term
    # records the finite-resolution effect without becoming an answer cue.
    size = int(scene["tile_size"])
    return round(0.34 + 0.0005 * (size - 144), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    """All sampled scenes are above the explicit 0.08 finite-resolution bound."""

    return margin(scene) < 0.08


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No nontrivial latent aliases are declared for this injective scene map."""

    return []
