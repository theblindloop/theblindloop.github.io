"""Bicycle-spoke count world under a contradictory canonical prior.

Profile: paper-semantic@0.3.0:prior_conflict. Report visible truth (the actual
number of spokes) when a familiar canonical prior ("an ordinary bicycle wheel
has many spokes, traditionally about 32") is contradicted by the actual image,
which in the counterfactual scenes shows a wheel with far fewer spokes drawn.

Latent z = {n, start_angle, radius, hub_radius, spoke_width, width, height}.
``n`` is the number of spokes actually drawn on the wheel rim. ``start_angle``
rotates the whole wheel and is the only field that can be freely changed
without changing the pixel-identical raster under the wheel's n-fold rotational
symmetry (a declared, exercised latent symmetry). The closed form gold is

    decision = str(n)

a scene parameter, not a pixel measurement.

The wheel is drawn as a light-metal rim and hub outline rings with ``n`` dark
spoke line segments running from just outside the hub to just inside the rim.
The spokes are the only dark pixels, so the image unambiguously reads as a
bicycle-style spoked wheel. The pixel-only inverse arm counts the radial spoke
directions by sampling the dark spoke colour on concentric rings around the
wheel centre and counting the number of contiguous angular runs, which is
exactly the number of spokes.

The continuous verifiability margin is the angular separation between adjacent
spokes in degrees (360 / n). When that separation is too small the spokes crowd
into each other near the hub and the count becomes pixel-ambiguous, so such
scenes are quarantined.
"""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw

RENDERER_VERSION = "prior_bicycle_spokes-0.1.0"

CANVAS = (512, 512)
BG = (255, 255, 255)
METAL = (240, 242, 246)   # rim / hub outline (ignored by the oracle)
SPOKE = (18, 22, 46)      # the only dark, counted pixels

CX, CY = CANVAS[0] // 2, CANVAS[1] // 2

RADIUS = 200
HUB_RADIUS = 30
SPOKE_WIDTH = 6

SUPPORTED_COUNTS = (4, 5, 6, 7, 8, 9, 10, 11, 12, 14, 16, 18)
QUAR_MARGIN_DEG = 15.0


def margin(scene: dict) -> float:
    """Angular gap (degrees) between adjacent spokes; the count robustness."""
    return 360.0 / float(scene["n"])


def is_quarantined(scene: dict) -> bool:
    n = int(scene["n"])
    if n not in SUPPORTED_COUNTS:
        return True
    return margin(scene) < QUAR_MARGIN_DEG


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    while True:
        n = rng.choice(SUPPORTED_COUNTS)
        scene = {
            "n": n,
            "start_angle": rng.uniform(0.0, 360.0),
            "radius": RADIUS,
            "hub_radius": HUB_RADIUS,
            "spoke_width": SPOKE_WIDTH,
            "width": CANVAS[0],
            "height": CANVAS[1],
        }
        if not is_quarantined(scene):
            return scene


def render(scene: dict) -> Image.Image:
    w, h = scene["width"], scene["height"]
    n = int(scene["n"])
    start = float(scene["start_angle"])
    radius = int(scene["radius"])
    hub = int(scene["hub_radius"])
    spoke_width = int(scene["spoke_width"])

    img = Image.new("RGB", (w, h), BG)
    draw = ImageDraw.Draw(img)
    draw.ellipse(
        [CX - radius, CY - radius, CX + radius, CY + radius], outline=METAL, width=10
    )
    draw.ellipse([CX - hub, CY - hub, CX + hub, CY + hub], outline=METAL, width=6)
    inner = hub + 8
    outer = radius - 12
    for k in range(n):
        ang = np.deg2rad(start + 360.0 * k / n)
        x0 = CX + inner * np.cos(ang)
        y0 = CY + inner * np.sin(ang)
        x1 = CX + outer * np.cos(ang)
        y1 = CY + outer * np.sin(ang)
        draw.line([x0, y0, x1, y1], fill=SPOKE, width=spoke_width)
    return img


def analytic_gold(scene: dict) -> str:
    return str(scene["n"])


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """One declared, exercised rotational alias: rotate by one spoke pitch.

    A wheel with ``n`` spokes has an n-fold rotational symmetry: rotating the
    whole scene by 360/n degrees simply relabels the spokes and leaves the
    raster pixel-identical, while preserving the spoke count (gold).
    """
    n = int(scene["n"])
    pitch = 360.0 / n
    twin = dict(scene, start_angle=float(scene["start_angle"]) + pitch)
    return [("rotate_by_spoke_pitch", twin)]
