"""Independent pixel-only inverse arm for the bicycle-spoke world.

Receives ONLY the final PNG and returns the number of spokes drawn as a string.
It shares no geometry constants with renderer.render except the spoke colour,
which is part of the visible image. It never imports renderer, prompts,
generate, verify, or analytic-gold code, and accepts no latent scene data.

Method: the wheel's spokes are the only dark pixels. The oracle finds the
spoke-colour pixels, estimates the wheel centre as their centroid (the spoke
crown is roughly spherically symmetric), and samples the wheel's radial extent
on several concentric rings. On each ring it records, for a dense set of angles,
whether the ring pixel is spoke-coloured, giving a one-dimensional boolean
signal whose contiguous runs are individual spoke crossings. The number of runs
equals the number of spokes; taking the maximum across rings discards the
near-hub radii where crowded spokes can touch.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "prior_bicycle_spokes-oracle-0.1.0"

_SPOKE = np.array((18, 22, 46), dtype=np.float32)
_COLOR_DIST = 80.0
_RINGS = (0.4, 0.5, 0.6, 0.7)
_ANGLES = 2048


def _ring_signal(signal: np.ndarray) -> int:
    """Count contiguous runs in a boolean ring signal (assumes >= 2 runs)."""
    wrapped = np.concatenate([signal, signal[:1]])
    transitions = int(np.sum(np.abs(np.diff(wrapped.astype(np.int8)))))
    return transitions // 2


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    h, w = arr.shape[:2]
    spoke = np.linalg.norm(arr - _SPOKE, axis=-1) < _COLOR_DIST
    ys, xs = np.nonzero(spoke)
    if len(ys) == 0:
        return "0"
    cy = float(np.mean(ys))
    cx = float(np.mean(xs))
    r_max = float(np.max(np.sqrt((xs - cx) ** 2 + (ys - cy) ** 2)))
    best = 0
    angles = np.linspace(0.0, 2.0 * np.pi, _ANGLES, endpoint=False)
    cos_a = np.cos(angles)
    sin_a = np.sin(angles)
    for frac in _RINGS:
        r = r_max * frac
        xx = np.round(cx + r * cos_a).astype(np.int64)
        yy = np.round(cy + r * sin_a).astype(np.int64)
        valid = (xx >= 0) & (xx < w) & (yy >= 0) & (yy < h)
        sig = np.zeros(_ANGLES, dtype=np.int8)
        sig[valid] = spoke[yy[valid], xx[valid]].astype(np.int8)
        count = _ring_signal(sig)
        if count > best:
            best = count
    return str(best)


def measure(image: Image.Image) -> dict:
    """Diagnostic measurement (not used for the decision)."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    h, w = arr.shape[:2]
    spoke = np.linalg.norm(arr - _SPOKE, axis=-1) < _COLOR_DIST
    ys, xs = np.nonzero(spoke)
    cy = float(np.mean(ys)) if len(ys) else 0.0
    cx = float(np.mean(xs)) if len(xs) else 0.0
    r_max = (
        float(np.max(np.sqrt((xs - cx) ** 2 + (ys - cy) ** 2))) if len(ys) else 0.0
    )
    return {
        "spoke_px": int(spoke.sum()),
        "centre": (round(cx, 1), round(cy, 1)),
        "max_radius": round(r_max, 1),
        "size": image.size,
    }
