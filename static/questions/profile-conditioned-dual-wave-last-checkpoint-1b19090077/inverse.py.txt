"""Independent image-only inverse arm: color segment, rebuild links, run BFS."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE = {
    "background": (250, 249, 246),
    "dark": (42, 46, 52),
    "gray": (205, 207, 210),
    "green": (40, 170, 91),
    "red": (220, 67, 67),
    "orange": (242, 145, 35),
    "purple": (139, 84, 202),
    "blue": (47, 126, 216),
}
NODE_COLORS = ("gray", "green", "red", "orange", "purple", "blue")
ANSWER_COLORS = ("orange", "purple", "blue")


def _components(mask: np.ndarray, minimum: int = 35) -> list[tuple[float, float]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        pts = []
        while stack:
            cy, cx = stack.pop()
            pts.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(pts) >= minimum:
            ys = [p[0] for p in pts]
            xs = [p[1] for p in pts]
            result.append((float(sum(xs) / len(xs)), float(sum(ys) / len(ys))))
    return result


def _distances(graph: list[set[int]], start: int) -> list[int]:
    distances = [-1] * len(graph)
    distances[start] = 0
    queue = deque([start])
    while queue:
        node = queue.popleft()
        for nxt in graph[node]:
            if distances[nxt] < 0:
                distances[nxt] = distances[node] + 1
                queue.append(nxt)
    return distances


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    nodes: list[dict] = []
    for color in NODE_COLORS:
        target = np.asarray(PALETTE[color], dtype=np.float32)
        mask = np.linalg.norm(arr - target, axis=2) < 28.0
        for x, y in _components(mask):
            nodes.append({"x": x, "y": y, "color": color})
    if len(nodes) != 36:
        raise ValueError(f"expected 36 visible nodes, found {len(nodes)}")

    graph = [set() for _ in nodes]
    dark = np.linalg.norm(arr - np.asarray(PALETTE["dark"], dtype=np.float32), axis=2) < 35.0
    for i, left in enumerate(nodes):
        for j in range(i + 1, len(nodes)):
            right = nodes[j]
            dx, dy = right["x"] - left["x"], right["y"] - left["y"]
            distance = float(np.hypot(dx, dy))
            if not 45.0 <= distance <= 88.0:
                continue
            hits = 0
            total = 0
            for t in np.linspace(0.30, 0.70, 17):
                x = int(round(left["x"] + t * dx))
                y = int(round(left["y"] + t * dy))
                patch = dark[max(0, y - 2) : y + 3, max(0, x - 2) : x + 3]
                hits += int(bool(patch.any()))
                total += 1
            if hits / total >= 0.82:
                graph[i].add(j)
                graph[j].add(i)

    unseen = set(range(len(nodes)))
    groups = []
    while unseen:
        root = min(unseen)
        stack = [root]
        group = set()
        while stack:
            node = stack.pop()
            if node in group:
                continue
            group.add(node)
            stack.extend(graph[node] - group)
        unseen -= group
        groups.append(group)
    if sorted(map(len, groups)) != [12, 12, 12]:
        raise ValueError(f"link reconstruction produced component sizes {sorted(map(len, groups))}")

    scores: dict[str, int] = {}
    for group in groups:
        colors = {nodes[i]["color"]: i for i in group if nodes[i]["color"] != "gray"}
        candidates = [color for color in ANSWER_COLORS if color in colors]
        if "green" not in colors or "red" not in colors or len(candidates) != 1:
            raise ValueError("each chain must contain green, red, and one answer checkpoint")
        marker_color = candidates[0]
        dg = _distances(graph, colors["green"])
        dr = _distances(graph, colors["red"])
        marker = colors[marker_color]
        scores[marker_color] = min(dg[marker], dr[marker])
    ranked = sorted(scores.items(), key=lambda item: item[1], reverse=True)
    if len(ranked) != 3 or ranked[0][1] <= ranked[1][1]:
        raise ValueError(f"ambiguous wave arrival scores {scores}")
    return ranked[0][0]
