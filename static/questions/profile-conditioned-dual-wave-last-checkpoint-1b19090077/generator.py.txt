"""Deterministic latent sampler and raster renderer for folded dual-end chains."""

from __future__ import annotations

import random
from collections import deque

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 780, 330
SCALE = 3
COLORS = {
    "background": (250, 249, 246),
    "dark": (42, 46, 52),
    "gray": (205, 207, 210),
    "green": (40, 170, 91),
    "red": (220, 67, 67),
    "orange": (242, 145, 35),
    "purple": (139, 84, 202),
    "blue": (47, 126, 216),
}
ANSWER_COLORS = ("orange", "purple", "blue")


def _hamiltonian_path(rng: random.Random) -> list[tuple[int, int]]:
    """Sample a Hamiltonian path on a 4 by 3 grid without exposing a fixed fold."""
    cells = [(x, y) for y in range(3) for x in range(4)]
    for _ in range(80):
        start = rng.choice(cells)
        path = [start]
        used = {start}

        def search() -> bool:
            if len(path) == len(cells):
                return True
            x, y = path[-1]
            options = [
                (nx, ny)
                for nx, ny in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1))
                if 0 <= nx < 4 and 0 <= ny < 3 and (nx, ny) not in used
            ]
            rng.shuffle(options)
            options.sort(
                key=lambda p: sum(
                    (p[0] + dx, p[1] + dy) not in used
                    and 0 <= p[0] + dx < 4
                    and 0 <= p[1] + dy < 3
                    for dx, dy in ((-1, 0), (1, 0), (0, -1), (0, 1))
                )
            )
            for nxt in options:
                used.add(nxt)
                path.append(nxt)
                if search():
                    return True
                path.pop()
                used.remove(nxt)
            return False

        if search():
            if rng.randrange(2):
                path.reverse()
            return path
    raise RuntimeError("could not sample a folded chain")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    answer = ANSWER_COLORS[int(seed) % len(ANSWER_COLORS)]
    remaining = [c for c in ANSWER_COLORS if c != answer]
    rng.shuffle(remaining)
    score_by_color = {answer: 5, remaining[0]: 4, remaining[1]: 3}
    panel_order = list(ANSWER_COLORS)
    rng.shuffle(panel_order)
    points_by_chain = []
    colors_by_chain = []
    for panel, marker_color in enumerate(panel_order):
        grid_path = _hamiltonian_path(rng)
        ox = 35 + panel * 255
        oy = 73
        sx, sy = 56, 82
        points = [[ox + x * sx, oy + y * sy] for x, y in grid_path]
        score = score_by_color[marker_color]
        marker_index = rng.choice([score, 11 - score])
        node_colors = ["gray"] * 12
        if rng.randrange(2):
            node_colors[0], node_colors[-1] = "green", "red"
        else:
            node_colors[0], node_colors[-1] = "red", "green"
        node_colors[marker_index] = marker_color
        points_by_chain.append(points)
        colors_by_chain.append(node_colors)
    return {
        "points": points_by_chain,
        "node_colors": colors_by_chain,
        "width": WIDTH,
        "height": HEIGHT,
    }


def _chain_score(colors: list[str]) -> tuple[str, int]:
    green = colors.index("green")
    red = colors.index("red")
    marker_color = next(color for color in ANSWER_COLORS if color in colors)
    marker = colors.index(marker_color)
    return marker_color, min(abs(marker - green), abs(marker - red))


def _scores(node_colors: list[list[str]]) -> dict[str, int]:
    return dict(_chain_score(colors) for colors in node_colors)


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene["node_colors"])
    return max(scores, key=scores.get)


def margin(scene: dict) -> float:
    values = sorted(_scores(scene["node_colors"]).values(), reverse=True)
    return float(values[0] - values[1])


def is_quarantined(scene: dict) -> bool:
    node_colors = scene["node_colors"]
    if len(node_colors) != 3 or any(len(colors) != 12 for colors in node_colors):
        return True
    values = sorted(_scores(node_colors).values(), reverse=True)
    return values[0] == values[1]


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width * SCALE, height * SCALE), COLORS["background"])
    draw = ImageDraw.Draw(image)
    for raw_points, node_colors in zip(scene["points"], scene["node_colors"]):
        points = [(int(x) * SCALE, int(y) * SCALE) for x, y in raw_points]
        draw.line(points, fill=COLORS["dark"], width=8 * SCALE, joint="curve")
        for (x, y), color in zip(points, node_colors):
            radius = (15 if color in ANSWER_COLORS else 13) * SCALE
            box = (x - radius, y - radius, x + radius, y + radius)
            draw.ellipse(box, fill=COLORS[color], outline=COLORS["dark"], width=3 * SCALE)
            if color in ("green", "red"):
                inner = 6 * SCALE
                draw.ellipse((x - inner, y - inner, x + inner, y + inner), fill=COLORS["background"])
    return image.resize((width, height), Image.Resampling.LANCZOS)
