"""Deterministic sampler and renderer for synchronous neighbor-lamp evolution."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1280
HEIGHT = 600
N_LAMPS = 7

PALETTES = (
    {"lamp": (24, 126, 157), "round": (230, 126, 34)},
    {"lamp": (18, 111, 143), "round": (216, 112, 27)},
)
TARGET = (202, 48, 67)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{name}", size)
    except OSError:
        return ImageFont.load_default()


def _advance(bits: list[int]) -> list[int]:
    """Apply one simultaneous XOR-of-neighbors round."""
    old = list(bits)
    return [
        (old[index - 1] if index > 0 else 0)
        ^ (old[index + 1] if index + 1 < len(old) else 0)
        for index in range(len(old))
    ]


def _replay(bits: list[int], rounds: int) -> list[int]:
    current = [int(value) for value in bits]
    for _ in range(int(rounds)):
        current = _advance(current)
    return current


def sample_scene(seed: int) -> dict:
    """Return one answer-balanced scene with fixed initial total ink.

    Consecutive seeds form four-scene strata for one target: rounds 2 and 3,
    with OFF and ON answers for each round count. Every sampled initial row has
    exactly three ON lamps, blocking total-ink and object-count shortcuts.
    """
    seed = int(seed)
    desired = seed & 1
    rounds = 2 + ((seed // 2) & 1)
    target_index = (seed // 4) % N_LAMPS
    required_initial_target = (seed // 28) & 1
    rng = random.Random(seed ^ 0x51A7E)

    options: list[list[int]] = []
    for raw in itertools.product((0, 1), repeat=N_LAMPS):
        bits = list(raw)
        if sum(bits) != 3 or bits[target_index] != required_initial_target:
            continue
        if _replay(bits, rounds)[target_index] == desired:
            options.append(bits)
    if not options:
        raise RuntimeError("balanced lamp stratum unexpectedly has no scene")

    initial_bits = options[rng.randrange(len(options))]
    return {
        "initial_bits": initial_bits,
        "target_index": target_index,
        "rounds": rounds,
        "palette_index": (seed // 56) % len(PALETTES),
        "offset_x": rng.choice((-5, 0, 5)),
        "offset_y": rng.choice((-4, 0, 4)),
    }


def analytic_gold(scene: dict) -> str:
    """Return the marked lamp's state after all synchronous rounds."""
    bits = _replay(
        [int(value) for value in scene["initial_bits"]],
        int(scene["rounds"]),
    )
    return "ON" if bits[int(scene["target_index"])] else "OFF"


def margin(scene: dict) -> float:
    """Discrete runner-up margin: ON and OFF differ by one binary state."""
    return 1.0


def is_quarantined(scene: dict) -> bool:
    """Fixed, widely separated glyphs do not approach a raster boundary."""
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No distinct latent record is claimed to produce identical pixels."""
    return []


def _centered(
    draw: ImageDraw.ImageDraw,
    xy: tuple[float, float],
    text: str,
    font: ImageFont.ImageFont,
    fill: tuple[int, int, int],
) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text(
        (xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2 - box[1]),
        text,
        font=font,
        fill=fill,
    )


def render(scene: dict) -> Image.Image:
    """Render the initial row, tracked-cell marker, and visible round count."""
    initial_bits = [int(value) for value in scene["initial_bits"]]
    target_index = int(scene["target_index"])
    rounds = int(scene["rounds"])
    palette = PALETTES[int(scene["palette_index"])]
    ox = int(scene["offset_x"])
    oy = int(scene["offset_y"])
    if len(initial_bits) != N_LAMPS or any(value not in (0, 1) for value in initial_bits):
        raise ValueError("initial_bits must contain seven binary states")
    if not 0 <= target_index < N_LAMPS or rounds not in (2, 3):
        raise ValueError("scene has an invalid target or round count")

    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), (250, 249, 246))
    draw = ImageDraw.Draw(image)
    ink = (35, 39, 45)
    paper = (255, 255, 255)
    soft = (225, 227, 229)

    def rounded(box, radius, **kwargs):
        draw.rounded_rectangle(
            tuple(int(value * scale) for value in box),
            radius=int(radius * scale),
            **kwargs,
        )

    title_font = _font(25 * scale, bold=True)
    rule_font = _font(14 * scale, bold=True)
    label_font = _font(18 * scale, bold=True)
    small_font = _font(12 * scale, bold=True)

    _centered(
        draw,
        ((WIDTH / 2 + ox) * scale, (29 + oy) * scale),
        "SYNCHRONOUS NEIGHBOR LAMPS",
        title_font,
        ink,
    )
    _centered(
        draw,
        ((WIDTH / 2 + ox) * scale, (66 + oy) * scale),
        "Each lamp reads the PREVIOUS row: exactly one ON neighbor → ON; otherwise → OFF",
        rule_font,
        ink,
    )
    _centered(
        draw,
        ((WIDTH / 2 + ox) * scale, (91 + oy) * scale),
        "All lamps update together  •  beyond either end is OFF",
        rule_font,
        ink,
    )

    rounded(
        (47 + ox, 120 + oy, 1233 + ox, 337 + oy),
        14,
        fill=paper,
        outline=ink,
        width=2 * scale,
    )
    _centered(
        draw,
        ((WIDTH / 2 + ox) * scale, (145 + oy) * scale),
        "START ROW",
        rule_font,
        ink,
    )
    centers = [145 + ox + 165 * index for index in range(N_LAMPS)]
    lamp_y = 222 + oy
    radius = 31
    for index, (cx, bit) in enumerate(zip(centers, initial_bits)):
        box = tuple(
            int(value * scale)
            for value in (cx - radius, lamp_y - radius, cx + radius, lamp_y + radius)
        )
        if bit:
            draw.ellipse(box, fill=palette["lamp"], outline=palette["lamp"], width=2 * scale)
        else:
            draw.ellipse(box, fill=paper, outline=palette["lamp"], width=7 * scale)
        _centered(
            draw,
            (cx * scale, (276 + oy) * scale),
            "ABCDEFG"[index],
            label_font,
            ink,
        )
        _centered(
            draw,
            (cx * scale, (308 + oy) * scale),
            "ON" if bit else "OFF",
            small_font,
            ink,
        )

    target_x = centers[target_index] * scale
    draw.polygon(
        (
            (target_x, (251 + oy) * scale),
            ((centers[target_index] - 13) * scale, (270 + oy) * scale),
            ((centers[target_index] + 13) * scale, (270 + oy) * scale),
        ),
        fill=TARGET,
    )

    _centered(
        draw,
        ((WIDTH / 2 + ox) * scale, (365 + oy) * scale),
        "Apply the orange ROUND cards from left to right",
        rule_font,
        ink,
    )
    card_width = 210
    gap = 42
    total_width = rounds * card_width + (rounds - 1) * gap
    start_x = (WIDTH - total_width) / 2 + ox
    for index in range(rounds):
        left = start_x + index * (card_width + gap)
        right = left + card_width
        rounded(
            (left, 392 + oy, right, 505 + oy),
            13,
            fill=paper,
            outline=ink,
            width=2 * scale,
        )
        cx = (left + right) / 2
        cy = 431 + oy
        badge_radius = 22
        draw.ellipse(
            tuple(
                int(value * scale)
                for value in (
                    cx - badge_radius,
                    cy - badge_radius,
                    cx + badge_radius,
                    cy + badge_radius,
                )
            ),
            fill=palette["round"],
        )
        _centered(
            draw,
            (cx * scale, (478 + oy) * scale),
            f"ROUND {index + 1}",
            rule_font,
            ink,
        )

    draw.line(
        ((62 + ox) * scale, (539 + oy) * scale, (1218 + ox) * scale, (539 + oy) * scale),
        fill=soft,
        width=2 * scale,
    )
    _centered(
        draw,
        ((WIDTH / 2 + ox) * scale, (565 + oy) * scale),
        "Red triangle = lamp to track     •     filled = ON     •     ring = OFF",
        rule_font,
        ink,
    )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
