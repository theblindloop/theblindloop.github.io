"""Independent pixel-only inverse for synchronous neighbor-lamp evolution.

The inverse segments only raster colors, recovers seven filled/ring states, maps
the red target marker to a lamp component, counts orange round badges, and then
implements the public simultaneous-update rule independently.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


class OracleDecodeError(ValueError):
    """Raised when the raster does not encode one unambiguous scene."""


_LAMP_COLORS = np.array(((24, 126, 157), (18, 111, 143)), dtype=np.int32)
_ROUND_COLORS = np.array(((230, 126, 34), (216, 112, 27)), dtype=np.int32)
_TARGET_COLORS = np.array(((202, 48, 67),), dtype=np.int32)


def _mask_near(arr: np.ndarray, colors: np.ndarray, radius: int = 48) -> np.ndarray:
    pixels = arr.astype(np.int32)
    distances = ((pixels[:, :, None, :] - colors[None, None, :, :]) ** 2).sum(axis=3)
    return distances.min(axis=2) <= radius * radius


def _runs(active: np.ndarray, minimum_width: int) -> list[tuple[int, int]]:
    indices = np.flatnonzero(active)
    if not len(indices):
        return []
    cuts = np.flatnonzero(np.diff(indices) > 1)
    starts = np.r_[0, cuts + 1]
    ends = np.r_[cuts, len(indices) - 1]
    runs = [(int(indices[a]), int(indices[b]) + 1) for a, b in zip(starts, ends)]
    return [(left, right) for left, right in runs if right - left >= minimum_width]


def _x_components(mask: np.ndarray, minimum_width: int) -> list[tuple[int, int]]:
    return _runs(mask.sum(axis=0) >= 2, minimum_width)


def _decode_lamps(mask: np.ndarray) -> tuple[list[int], list[float]]:
    components = _x_components(mask, 36)
    if len(components) != 7:
        raise OracleDecodeError(f"found {len(components)} lamp glyphs, expected 7")
    bits: list[int] = []
    centers: list[float] = []
    for left, right in components:
        ys, local_xs = np.nonzero(mask[:, left:right])
        if len(ys) < 250:
            raise OracleDecodeError("a lamp glyph has too little colored ink")
        top = int(ys.min())
        bottom = int(ys.max()) + 1
        width = right - left
        height = bottom - top
        if not 48 <= width <= 72 or not 48 <= height <= 72:
            raise OracleDecodeError("a lamp glyph has invalid geometry")
        inner_left = left + int(width * 0.34)
        inner_right = left + int(width * 0.66)
        inner_top = top + int(height * 0.34)
        inner_bottom = top + int(height * 0.66)
        interior = mask[inner_top:inner_bottom, inner_left:inner_right]
        fill_fraction = float(interior.mean())
        if fill_fraction >= 0.75:
            bits.append(1)
        elif fill_fraction <= 0.10:
            bits.append(0)
        else:
            raise OracleDecodeError(f"ambiguous lamp fill ({fill_fraction:.3f})")
        centers.append((left + right - 1) / 2)
    if min(np.diff(centers)) < 120:
        raise OracleDecodeError("lamp glyphs are merged or out of order")
    return bits, centers


def _decode_target(mask: np.ndarray, lamp_centers: list[float]) -> int:
    components = _x_components(mask, 14)
    if len(components) != 1:
        raise OracleDecodeError(f"found {len(components)} target markers, expected 1")
    ys, xs = np.nonzero(mask)
    if len(xs) < 100:
        raise OracleDecodeError("target marker is incomplete")
    marker_x = float(xs.mean())
    distances = np.abs(np.asarray(lamp_centers) - marker_x)
    order = np.argsort(distances)
    if distances[order[0]] > 24 or distances[order[1]] - distances[order[0]] < 60:
        raise OracleDecodeError("target marker does not select one lamp")
    return int(order[0])


def _decode_rounds(mask: np.ndarray) -> int:
    components = _x_components(mask, 30)
    if len(components) not in (2, 3):
        raise OracleDecodeError(f"found {len(components)} round badges, expected 2 or 3")
    for left, right in components:
        area = int(mask[:, left:right].sum())
        if area < 900:
            raise OracleDecodeError("a round badge is incomplete")
    return len(components)


def decision_from_image(image: Image.Image) -> str:
    """Decode all visible evidence and return the marked final state."""
    rgb = image.convert("RGB")
    if rgb.width < 900 or rgb.height < 400:
        raise OracleDecodeError(f"raster is too small for this world: {rgb.size}")
    arr = np.asarray(rgb, dtype=np.uint8)
    lamp_mask = _mask_near(arr, _LAMP_COLORS)
    target_mask = _mask_near(arr, _TARGET_COLORS)
    round_mask = _mask_near(arr, _ROUND_COLORS)

    bits, centers = _decode_lamps(lamp_mask)
    target = _decode_target(target_mask, centers)
    rounds = _decode_rounds(round_mask)
    for _ in range(rounds):
        previous = list(bits)
        bits = [
            (previous[index - 1] if index > 0 else 0)
            ^ (previous[index + 1] if index + 1 < len(previous) else 0)
            for index in range(len(previous))
        ]
    return "ON" if bits[target] else "OFF"
