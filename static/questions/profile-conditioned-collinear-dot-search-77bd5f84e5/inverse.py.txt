"""Independent pixel-only inverse arm for collinear_dot_search.

Receives only a PIL image. It separates the two gray reference dots and the
four dark candidate dots by color, computes the straight line through the two
gray reference centers, measures each dark dot's perpendicular distance to that
line, and selects the quadrant whose dark dot lies closest to / on the line.
Imports nothing from the renderer, prompts, generator, or verifier.
"""

from __future__ import annotations

import math

import numpy as np


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Label 8-connected components and return a list of (ys, xs) index arrays."""
    ids = np.zeros(mask.shape, dtype=np.int32)
    label = 0
    ys, xs = np.nonzero(mask)
    for yy, xx in zip(ys.tolist(), xs.tolist()):
        if ids[yy, xx]:
            continue
        label += 1
        stack = [(yy, xx)]
        ids[yy, xx] = label
        while stack:
            cy, cx = stack.pop()
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    ny, nx = cy + dy, cx + dx
                    if 0 <= ny < mask.shape[0] and 0 <= nx < mask.shape[1]:
                        if mask[ny, nx] and not ids[ny, nx]:
                            ids[ny, nx] = label
                            stack.append((ny, nx))
    out = []
    for lab in range(1, label + 1):
        py, px = np.nonzero(ids == lab)
        out.append((py, px))
    return out


def _centroid(py, px):
    return (float(px.mean()), float(py.mean()))


def _dist_to_line(px, py, a, b, c):
    return abs(a * px + b * py + c) / math.hypot(a, b)


def decision_from_image(image) -> str:
    w, h = image.size
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    r, g, b = arr[..., 0], arr[..., 1], arr[..., 2]

    dark = (r < 120) & (g < 120) & (b < 120)
    ref = (r > 150) & (g > 150) & (b > 150) & (r < 250)

    ref_comps = _components(ref)
    if len(ref_comps) < 2:
        return "TL"
    # The two gray reference markers are the two largest gray components.
    ref_comps.sort(key=lambda c: c[0].size, reverse=True)
    c1 = _centroid(*ref_comps[0])
    c2 = _centroid(*ref_comps[1])
    a = c2[1] - c1[1]
    bb = c1[0] - c2[0]
    c = c2[0] * c1[1] - c1[0] * c2[1]

    dark_comps = _components(dark)
    if len(dark_comps) < 1:
        return "TL"

    best_quad = None
    best_dist = float("inf")
    for py, px in dark_comps:
        cx, cy = _centroid(py, px)
        dd = _dist_to_line(cx, cy, a, bb, c)
        # Map this dark component to its quadrant by centroid position.
        if cx < w * 0.5:
            quad = "TL" if cy < h * 0.5 else "BL"
        else:
            quad = "TR" if cy < h * 0.5 else "BR"
        if dd < best_dist:
            best_dist = dd
            best_quad = quad

    return best_quad if best_quad is not None else "TL"
