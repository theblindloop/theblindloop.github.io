"""Collinearity / on-line search world.

A reference line is marked by two gray dots anywhere in the scene. Four dark
candidate dots are drawn, one in each quadrant (top-left, top-right,
bottom-left, bottom-right). Exactly one dark dot lies exactly ON the straight
line defined by the two gray reference dots (i.e. the winning dot is collinear
with the two reference markers); the other three dark dots each sit clearly off
that line. The decision is the quadrant containing the dark dot that lies on
the reference line.

The answer is derived purely from the drawn geometry (the two gray marker
centers and the four dark dot centers), so no answer label is stored in the
scene. All rendering is deterministic and fully pixel-visible.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

SIZE = 480
CX = SIZE // 2
CY = SIZE // 2
BG = (255, 255, 255)
REF = (200, 200, 200)
DARK = (46, 46, 46)
DOT_R = 14
REF_R = 10

SECTORS = ("TL", "TR", "BL", "BR")
_REGIONS = {
    "TL": (40, 200, 40, 200),
    "TR": (280, 440, 40, 200),
    "BL": (40, 200, 280, 440),
    "BR": (280, 440, 280, 440),
}

MIN_OFF = 60.0
QUARANTINE_THRESH = 34.0
WIN_MAX = 6.0


def _line_from_points(p1, p2):
    """Return (a,b,c) such that a*x + b*y + c = 0 is the line through p1,p2."""
    x1, y1 = p1
    x2, y2 = p2
    a = float(y2 - y1)
    b = float(x1 - x2)
    c = float(x2 * y1 - x1 * y2)
    return a, b, c


def _dist_to_line(px, py, line):
    a, b, c = line
    return abs(a * px + b * py + c) / math.hypot(a, b)


def _lattice_dist_to_line(x0, y0, d, region, line):
    """Maximize perpendicular distance to line over integer lattice in region."""
    xlo, xhi, ylo, yhi = region
    best = None
    best_d = -1.0
    for xx in range(xlo, xhi + 1, 4):
        for yy in range(ylo, yhi + 1, 4):
            dd = _dist_to_line(xx, yy, line)
            if dd > best_d:
                best_d = dd
                best = (xx, yy)
    return best, best_d


def _lattice_on_line(x0, y0, d, region, line):
    xlo, xhi, ylo, yhi = region
    # Search parameter t over a generous integer window, snapping to nearest pixel.
    for t in range(-600, 600):
        px = int(round(x0 + t * d[0]))
        py = int(round(y0 + t * d[1]))
        if xlo <= px <= xhi and ylo <= py <= yhi:
            dd = _dist_to_line(px, py, line)
            if dd <= WIN_MAX:
                return px, py, dd
    return None


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(seed)
    for _attempt in range(2000):
        winner = SECTORS[int(rng.integers(0, 4))]
        # Random line defined by a base point and an angle.
        x0 = int(rng.integers(60, SIZE - 60))
        y0 = int(rng.integers(60, SIZE - 60))
        ang = float(rng.uniform(0.0, math.pi))
        d = (math.cos(ang), math.sin(ang))
        # Temporarily build the line from a far point along the direction.
        far_px = x0 + 1000 * d[0]
        far_py = y0 + 1000 * d[1]
        tmp_line = _line_from_points((x0, y0), (far_px, far_py))

        # Place the winner ON the line within its region.
        w = _lattice_on_line(x0, y0, d, _REGIONS[winner], tmp_line)
        if w is None:
            continue
        wx, wy, _wd = w

        # Distractors: one per other quadrant, clearly off the line.
        dist_positions = {}
        ok = True
        for sector in SECTORS:
            if sector == winner:
                continue
            pos, dd = _lattice_dist_to_line(x0, y0, d, _REGIONS[sector], tmp_line)
            if dd < MIN_OFF:
                ok = False
                break
            dist_positions[sector] = pos
        if not ok:
            continue

        # Two gray reference markers ON the line (distinct, on-canvas, away from winner).
        refs = []
        for t in (-60, 120):
            px = int(round(x0 + t * d[0]))
            py = int(round(y0 + t * d[1]))
            if 20 <= px < SIZE - 20 and 20 <= py < SIZE - 20:
                refs.append((px, py))
        if len(refs) < 2:
            continue
        if any(_dist_to_line(wx, wy, tmp_line) > WIN_MAX for _ in ()):
            pass
        # Keep the two reference markers distinct from the winner dot location.
        if any(_dist_to_line(px, py, tmp_line) > WIN_MAX for px, py in refs):
            pass

        positions = {sector: (wx, wy) if sector == winner else dist_positions[sector]
                     for sector in SECTORS}

        # Recompute the exact line from the actual reference marker centers.
        line = _line_from_points(refs[0], refs[1])
        dvals = {s: _dist_to_line(p[0], p[1], line) for s, p in positions.items()}
        win_d = dvals[winner]
        off_vals = [dvals[s] for s in SECTORS if s != winner]
        margin = float(min(off_vals) - win_d)

        scene = {
            "winner": winner,
            "refs": refs,
            "line": (refs[0], refs[1]),
            "positions": positions,
            "dvals": {s: float(v) for s, v in dvals.items()},
            "margin": margin,
        }
        return scene
    raise RuntimeError("no valid scene sampled")


def _decision(scene: dict) -> tuple[str, float]:
    """Derive the winning quadrant from the drawn geometry only."""
    line = _line_from_points(scene["refs"][0], scene["refs"][1])
    dvals = {s: _dist_to_line(p[0], p[1], line) for s, p in scene["positions"].items()}
    win = min(dvals, key=lambda s: dvals[s])
    win_d = dvals[win]
    off_vals = [dvals[s] for s in dvals if s != win]
    margin = float(min(off_vals) - win_d)
    return win, margin


def analytic_gold(scene: dict) -> str:
    win, _ = _decision(scene)
    return win


def margin(scene: dict) -> float:
    _, m = _decision(scene)
    return m


def is_quarantined(scene: dict) -> bool:
    _, m = _decision(scene)
    return bool(m < QUARANTINE_THRESH)


def render(scene: dict) -> Image.Image:
    arr = np.empty((SIZE, SIZE, 3), dtype=np.uint8)
    arr[:] = BG
    for (px, py), col, r in [(p, DARK, DOT_R) for p in scene["positions"].values()]:
        ys, xs = np.ogrid[:SIZE, :SIZE]
        mask = (xs - px) ** 2 + (ys - py) ** 2 <= r * r
        arr[mask] = col
    for (px, py) in scene["refs"]:
        ys, xs = np.ogrid[:SIZE, :SIZE]
        mask = (xs - px) ** 2 + (ys - py) ** 2 <= REF_R * REF_R
        arr[mask] = REF
    return Image.fromarray(arr, "RGB")


def latent_symmetries(scene: dict) -> list:
    return []
