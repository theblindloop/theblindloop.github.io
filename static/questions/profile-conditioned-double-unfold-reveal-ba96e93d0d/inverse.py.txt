"""Independent inverse arm. Only raster pixels enter this module."""

from __future__ import annotations

import numpy as np
from PIL import Image

N = 6
LABELS = ("ring", "plus", "X", "four blocks")


def _expected_masks() -> dict[str, np.ndarray]:
    # Independently specify the four completed 12x12 public answer shapes.
    masks: dict[str, np.ndarray] = {}
    ring = np.zeros((12, 12), dtype=np.uint8)
    ring[1, 1:11] = ring[10, 1:11] = 1
    ring[1:11, 1] = ring[1:11, 10] = 1
    masks["ring"] = ring
    plus = np.zeros((12, 12), dtype=np.uint8)
    plus[1:11, 5:7] = 1
    plus[5:7, 1:11] = 1
    masks["plus"] = plus
    xmask = np.zeros((12, 12), dtype=np.uint8)
    quarter_x = {(i, i) for i in range(1, 6)} | {(1, 2), (2, 1), (3, 4), (4, 3)}
    for r, c in quarter_x:
        for rr in (r, 11 - r):
            for cc in (c, 11 - c):
                xmask[rr, cc] = 1
    masks["X"] = xmask
    blocks = np.zeros((12, 12), dtype=np.uint8)
    blocks[1:4, 1:4] = blocks[1:4, 8:11] = 1
    blocks[8:11, 1:4] = blocks[8:11, 8:11] = 1
    masks["four blocks"] = blocks
    return masks


def _largest_bbox(mask: np.ndarray) -> tuple[int, int, int, int]:
    # Cyan exists only on the main frame; use its full connected outline bbox.
    ys, xs = np.nonzero(mask)
    if len(xs) < 200:
        raise ValueError("missing cyan frame")
    return int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())


def _recover_quarter(image: Image.Image) -> np.ndarray:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    r, g, b = rgb[..., 0], rgb[..., 1], rgb[..., 2]
    cyan = (r < 70) & (g > 95) & (b > 125) & (b > g)
    x0, y0, x1, y1 = _largest_bbox(cyan)
    sx, sy = (x1 - x0) / N, (y1 - y0) / N
    if not (17.5 <= sx <= 24.5 and abs(sx - sy) < 0.2):
        raise ValueError("invalid cell lattice")
    magenta = (r > 145) & (g < 105) & (b > 80) & (r > b)
    yellow = (r > 175) & (g > 125) & (b < 90)
    if int(magenta.sum()) < 250 or int(yellow.sum()) < 250:
        raise ValueError("missing colored hinge")
    my, mx = np.nonzero(magenta)
    yy, yx = np.nonzero(yellow)
    magenta_center = (float(mx.mean()), float(my.mean()))
    yellow_center = (float(yx.mean()), float(yy.mean()))
    if magenta_center[0] > x1:
        orientation = 0
    elif magenta_center[1] > y1:
        orientation = 1
    elif magenta_center[0] < x0:
        orientation = 2
    elif magenta_center[1] < y0:
        orientation = 3
    else:
        raise ValueError("magenta hinge is not outside frame")
    expected_yellow_side = ("bottom", "left", "top", "right")[orientation]
    observed = (
        "right" if yellow_center[0] > x1 else
        "bottom" if yellow_center[1] > y1 else
        "left" if yellow_center[0] < x0 else
        "top" if yellow_center[1] < y0 else "inside"
    )
    if observed != expected_yellow_side:
        raise ValueError("yellow hinge is not on the declared perpendicular edge")
    q = np.zeros((N, N), dtype=np.uint8)
    for row in range(N):
        for col in range(N):
            cx = int(round(x0 + (col + 0.5) * sx))
            cy = int(round(y0 + (row + 0.5) * sy))
            patch = rgb[cy - 2 : cy + 3, cx - 2 : cx + 3]
            q[row, col] = int(float(patch.mean()) < 100.0)
    if int(q.sum()) != 9:
        raise ValueError("black-square evidence count invalid")
    return np.rot90(q, k=orientation)


def _unfold(q: np.ndarray) -> np.ndarray:
    once = np.hstack((q, np.fliplr(q)))
    return np.vstack((once, np.flipud(once)))


def decision_and_margin_from_image(image: Image.Image) -> tuple[str, float]:
    completed = _unfold(_recover_quarter(image))
    scores = sorted((int(np.sum(completed != expected)), label) for label, expected in _expected_masks().items())
    gap = float(scores[1][0] - scores[0][0])
    if gap < 4.0:
        raise ValueError("ambiguous completed pattern")
    return scores[0][1], gap


def decision_from_image(image: Image.Image) -> str:
    return decision_and_margin_from_image(image)[0]
