"""Deterministic scene sampler and Pillow renderer."""

from __future__ import annotations

import random
from typing import Iterable

import numpy as np
from PIL import Image, ImageDraw

LABELS = ("ring", "plus", "X", "four blocks")
N = 6


def _base_quarter(label: str) -> set[tuple[int, int]]:
    if label == "ring":
        return {(1, c) for c in range(1, 6)} | {(r, 1) for r in range(2, 6)}
    if label == "plus":
        return {(5, c) for c in range(1, 6)} | {(r, 5) for r in range(1, 5)}
    if label == "X":
        return {(i, i) for i in range(1, 6)} | {(1, 2), (2, 1), (3, 4), (4, 3)}
    if label == "four blocks":
        return {(r, c) for r in range(1, 4) for c in range(1, 4)}
    raise ValueError(label)


def _rotate(coords: Iterable[tuple[int, int]], turns: int) -> list[list[int]]:
    out = list(coords)
    for _ in range(turns % 4):
        out = [(c, N - 1 - r) for r, c in out]
    return [[r, c] for r, c in sorted(out)]


def _unfold(quarter: Iterable[tuple[int, int]]) -> np.ndarray:
    q = np.zeros((N, N), dtype=np.uint8)
    for r, c in quarter:
        q[int(r), int(c)] = 1
    side = np.concatenate([q, q[:, ::-1]], axis=1)
    return np.concatenate([side, side[::-1, :]], axis=0)


def _templates() -> dict[str, np.ndarray]:
    return {label: _unfold(_base_quarter(label)) for label in LABELS}


def _classify(quarter: Iterable[tuple[int, int]], orientation: int = 0) -> tuple[str, float]:
    canonical = _rotate((tuple(x) for x in quarter), -int(orientation))
    got = _unfold(canonical)
    scored = sorted((int(np.count_nonzero(got != mask)), label) for label, mask in _templates().items())
    return scored[0][1], float(scored[1][0] - scored[0][0])


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    label = LABELS[seed % len(LABELS)]
    orientation = rng.randrange(4)
    coords = set(_base_quarter(label))
    # Every fourth scene receives one visible, non-ambiguous bit perturbation.
    if (seed // 4) % 4 == 3:
        options = [(r, c) for r in range(N) for c in range(N) if (r, c) not in coords]
        add = options[rng.randrange(len(options))]
        remove = sorted(coords)[rng.randrange(len(coords))]
        coords.remove(remove)
        coords.add(add)
    scene = {
        "quarter": _rotate(coords, orientation),
        "frame_x": rng.randrange(154, 207),
        "frame_y": rng.randrange(102, 139),
        "cell_px": (18, 20, 22, 24)[rng.randrange(4)],
        "orientation": orientation,
        "palette_variant": rng.randrange(2),
    }
    if is_quarantined(scene):
        raise RuntimeError("sampler produced ambiguous scene")
    return scene


def analytic_gold(scene: dict) -> str:
    return _classify(scene["quarter"], scene["orientation"])[0]


def margin(scene: dict) -> float:
    return _classify(scene["quarter"], scene["orientation"])[1]


def is_quarantined(scene: dict) -> bool:
    q = scene.get("quarter", [])
    return len(q) != 9 or len({tuple(x) for x in q}) != 9 or margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["quarter"] = list(reversed([list(x) for x in scene["quarter"]]))
    return [("reverse_quarter_storage_order", twin)]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (512, 384), (250, 248, 244))
    draw = ImageDraw.Draw(image)
    x0, y0, s = int(scene["frame_x"]), int(scene["frame_y"]), int(scene["cell_px"])
    size = N * s
    palette = (
        {"cyan": (0, 154, 190), "magenta": (204, 42, 135), "yellow": (239, 183, 28)},
        {"cyan": (0, 132, 176), "magenta": (186, 35, 151), "yellow": (226, 165, 20)},
    )[int(scene["palette_variant"])]

    # A light shadow and paper field make the folded object explicit.
    draw.rounded_rectangle((x0 + 8, y0 + 9, x0 + size + 8, y0 + size + 9), radius=5, fill=(224, 222, 217))
    draw.rectangle((x0, y0, x0 + size, y0 + size), fill=(255, 255, 255))
    occupied = {tuple(v) for v in scene["quarter"]}
    for r in range(N):
        for c in range(N):
            left, top = x0 + c * s, y0 + r * s
            fill = (28, 29, 32) if (r, c) in occupied else (247, 247, 245)
            draw.rectangle((left + 2, top + 2, left + s - 2, top + s - 2), fill=fill)
    for i in range(1, N):
        draw.line((x0 + i * s, y0, x0 + i * s, y0 + size), fill=(207, 209, 208), width=1)
        draw.line((x0, y0 + i * s, x0 + size, y0 + i * s), fill=(207, 209, 208), width=1)
    draw.rectangle((x0, y0, x0 + size, y0 + size), outline=palette["cyan"], width=4)

    # Base hinges are right (magenta, first) and bottom (yellow, second), rotated with the sheet.
    sides = ["right", "bottom", "left", "top"]
    magenta_side = sides[int(scene["orientation"]) % 4]
    yellow_side = sides[(int(scene["orientation"]) + 1) % 4]

    def hinge(side: str, color: tuple[int, int, int], inset: int) -> None:
        pad = 8
        if side == "right":
            draw.rounded_rectangle((x0 + size + pad, y0 + inset, x0 + size + pad + 9, y0 + size - inset), radius=3, fill=color)
        elif side == "bottom":
            draw.rounded_rectangle((x0 + inset, y0 + size + pad, x0 + size - inset, y0 + size + pad + 9), radius=3, fill=color)
        elif side == "left":
            draw.rounded_rectangle((x0 - pad - 9, y0 + inset, x0 - pad, y0 + size - inset), radius=3, fill=color)
        else:
            draw.rounded_rectangle((x0 + inset, y0 - pad - 9, x0 + size - inset, y0 - pad), radius=3, fill=color)

    hinge(magenta_side, palette["magenta"], max(14, s))
    hinge(yellow_side, palette["yellow"], max(24, s + 10))
    return image
