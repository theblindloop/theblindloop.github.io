"""Stripe-phase wedge world.

One small pale card carries evenly spaced dark bars of a single colour,
thickness and pitch.  Inside the card, one isoceles-triangular patch has its
bars shifted by exactly half a pitch, so inside the patch bars sit in the gaps
of the surrounding bars.  The patch has no outline, no colour of its own and no
extra ink: its silhouette exists only as a phase discontinuity of the periodic
carrier.  The question asks which way the patch's tip points.

The decisive evidence is therefore one connected, area-like region inside a
single small neighbourhood of the page: the phase-shifted patch together with
the collar of bars immediately around it.  Everything else -- the pale clutter
blobs, the page, the bar colour, the card position -- is nondiagnostic.
"""

from __future__ import annotations

import random

import numpy as np
from PIL import Image

RENDERER_VERSION = "stripe_phase_wedge-0.1.0"

WIDTH = 800
HEIGHT = 640

QUARANTINE_THRESHOLD = 20.0
CLEAR_BAND_MIN = 62.0
BLURRED_BAND_MAX = 8.0

READABLE_PITCHES = 4.0  # declared readability rule: four bar pitches across

DIRECTIONS = ("up", "down", "left", "right")

PAGE_COLORS = ((248, 247, 244), (245, 246, 248), (250, 248, 243))
CARD_COLORS = ((236, 233, 226), (231, 236, 238), (238, 234, 238), (233, 237, 231))
BAR_COLORS = ((34, 36, 41), (28, 42, 58), (46, 32, 38), (30, 48, 44))
CLUTTER_COLORS = ((214, 212, 206), (208, 214, 217), (216, 210, 214))

PITCHES = (10, 12, 14, 16)


def sample_scene(seed: int) -> dict:
    """Deterministically sample one scene from an integer seed."""
    rng = random.Random(seed * 7919 + 13)

    # Answer identity and the quarantine class cycle on the seed, so datasets
    # built from consecutive seeds are balanced by construction.  Nothing about
    # this cycle is visible in the raster.
    apex = DIRECTIONS[seed % 4]
    quarantined_target = ((seed // 4) % 6) == 4

    period = rng.choice(PITCHES)
    thickness = period // 2
    axis = rng.choice(("h", "v"))

    if quarantined_target:
        short = rng.uniform(2.0 * period, 4.0 * period + BLURRED_BAND_MAX)
    else:
        short = rng.uniform(
            4.0 * period + CLEAR_BAND_MIN, 4.0 * period + CLEAR_BAND_MIN + 34.0
        )
    long = short * rng.uniform(1.0, 1.34)
    if rng.random() < 0.5:
        base, height = float(short), float(long)
    else:
        base, height = float(long), float(short)

    if apex in ("up", "down"):
        box_w, box_h = base, height
    else:
        box_w, box_h = height, base

    # The card is large enough that the phase-shifted patch is a clear minority
    # of the carrier, so the surrounding bars define the reference phase.
    area = 0.5 * base * height
    pad = max(1.8 * period + 2.0, 12.0)
    while (box_w + 2 * pad) * (box_h + 2 * pad) < 3.2 * area:
        pad += 2.0
    pad += rng.uniform(0.0, 8.0)
    card_w = int(round(box_w + 2 * pad))
    card_h = int(round(box_h + 2 * pad))

    card_x = rng.randint(26, WIDTH - card_w - 26)
    card_y = rng.randint(26, HEIGHT - card_h - 26)

    jitter = max(0.0, pad - 1.8 * period - 2.0)
    cx = card_x + card_w / 2.0 + rng.uniform(-jitter, jitter)
    cy = card_y + card_h / 2.0 + rng.uniform(-jitter, jitter)

    origin = rng.randint(0, period - 1)

    clutter = []
    guard = 22
    for _ in range(rng.randint(3, 6)):
        for _attempt in range(80):
            rx = rng.randint(16, 46)
            ry = rng.randint(14, 40)
            bx = rng.randint(rx + 4, WIDTH - rx - 4)
            by = rng.randint(ry + 4, HEIGHT - ry - 4)
            if (
                bx + rx + guard < card_x
                or bx - rx - guard > card_x + card_w
                or by + ry + guard < card_y
                or by - ry - guard > card_y + card_h
            ):
                clutter.append([bx, by, rx, ry])
                break

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "page": list(rng.choice(PAGE_COLORS)),
        "card_color": list(rng.choice(CARD_COLORS)),
        "bar_color": list(rng.choice(BAR_COLORS)),
        "clutter_color": list(rng.choice(CLUTTER_COLORS)),
        "card": [card_x, card_y, card_w, card_h],
        "axis": axis,
        "period": period,
        "thickness": thickness,
        "origin": origin,
        "apex": apex,
        "tri_base": round(base, 4),
        "tri_height": round(height, 4),
        "center": [round(cx, 4), round(cy, 4)],
        "clutter": clutter,
    }


def _vertices(scene: dict) -> list[tuple[float, float]]:
    cx, cy = scene["center"]
    b = float(scene["tri_base"]) / 2.0
    h = float(scene["tri_height"]) / 2.0
    apex = scene["apex"]
    if apex == "up":
        return [(cx, cy - h), (cx - b, cy + h), (cx + b, cy + h)]
    if apex == "down":
        return [(cx, cy + h), (cx - b, cy - h), (cx + b, cy - h)]
    if apex == "left":
        return [(cx - h, cy), (cx + h, cy - b), (cx + h, cy + b)]
    return [(cx + h, cy), (cx - h, cy - b), (cx - h, cy + b)]


def _triangle_mask(scene: dict, xx: np.ndarray, yy: np.ndarray) -> np.ndarray:
    (ax, ay), (bx, by), (cx2, cy2) = _vertices(scene)

    def side(x0, y0, x1, y1):
        return (x1 - x0) * (yy - y0) - (y1 - y0) * (xx - x0)

    s1 = side(ax, ay, bx, by)
    s2 = side(bx, by, cx2, cy2)
    s3 = side(cx2, cy2, ax, ay)
    return ((s1 >= 0) & (s2 >= 0) & (s3 >= 0)) | ((s1 <= 0) & (s2 <= 0) & (s3 <= 0))


def render(scene: dict) -> Image.Image:
    w = int(scene["width"])
    h = int(scene["height"])
    canvas = np.zeros((h, w, 3), dtype=np.uint8)
    canvas[:, :] = np.array(scene["page"], dtype=np.uint8)

    yy, xx = np.mgrid[0:h, 0:w]
    yy = yy.astype(np.float64)
    xx = xx.astype(np.float64)

    clutter_color = np.array(scene["clutter_color"], dtype=np.uint8)
    for bx, by, rx, ry in scene["clutter"]:
        blob = ((xx - bx) / float(rx)) ** 2 + ((yy - by) / float(ry)) ** 2 <= 1.0
        canvas[blob] = clutter_color

    card_x, card_y, card_w, card_h = (int(v) for v in scene["card"])
    canvas[card_y : card_y + card_h, card_x : card_x + card_w] = np.array(
        scene["card_color"], dtype=np.uint8
    )

    period = int(scene["period"])
    thickness = int(scene["thickness"])
    origin = int(scene["origin"])
    shift = period // 2

    coord = yy if scene["axis"] == "h" else xx
    base_ink = ((coord - origin) % period) < thickness
    wedge_ink = ((coord - origin + shift) % period) < thickness

    tri = _triangle_mask(scene, xx, yy)
    ink = np.where(tri, wedge_ink, base_ink)

    inside_card = np.zeros((h, w), dtype=bool)
    inside_card[card_y : card_y + card_h, card_x : card_x + card_w] = True
    canvas[ink & inside_card] = np.array(scene["bar_color"], dtype=np.uint8)

    return Image.fromarray(canvas, mode="RGB")


def margin(scene: dict) -> float:
    """Pixels of readable span above the declared four-pitch readability rule."""
    span = min(float(scene["tri_base"]), float(scene["tri_height"]))
    return float(span - READABLE_PITCHES * float(scene["period"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_THRESHOLD


def analytic_gold(scene: dict) -> str:
    if is_quarantined(scene):
        return "unclear"
    return str(scene["apex"])


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical relabelings of the latent scene.

    The bar carrier is periodic with pitch ``period`` both inside and outside
    the patch, so translating the phase origin by a whole pitch is a pure
    relabeling.  Clutter blobs share one flat colour, so their list order is a
    relabeling as well.
    """
    out: list[tuple[str, dict]] = []
    period = int(scene["period"])

    for name, delta in (("phase_origin_plus_pitch", period), ("phase_origin_minus_pitch", -period)):
        variant = dict(scene)
        variant["origin"] = int(scene["origin"]) + delta
        out.append((name, variant))

    if len(scene["clutter"]) > 1:
        reversed_clutter = dict(scene)
        reversed_clutter["clutter"] = [list(b) for b in reversed(scene["clutter"])]
        out.append(("clutter_order_reversed", reversed_clutter))

        rotated = dict(scene)
        rotated["clutter"] = [list(b) for b in scene["clutter"][1:] + scene["clutter"][:1]]
        out.append(("clutter_order_rotated", rotated))

    return out


def scene_summary(scene: dict) -> str:
    return (
        f"apex={scene['apex']} axis={scene['axis']} pitch={scene['period']} "
        f"base={scene['tri_base']:.1f} height={scene['tri_height']:.1f} "
        f"margin={margin(scene):.1f}"
    )
