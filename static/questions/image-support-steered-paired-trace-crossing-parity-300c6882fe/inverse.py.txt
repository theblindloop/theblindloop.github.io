"""Independent raster-only oracle for paired trace crossing parity."""

from __future__ import annotations

from collections import deque
import math

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    points = {(int(y), int(x)) for y, x in np.argwhere(mask)}
    components: list[np.ndarray] = []
    while points:
        start = points.pop()
        queue = deque([start])
        found = [start]
        while queue:
            y, x = queue.popleft()
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dy or dx):
                        continue
                    neighbor = (y + dy, x + dx)
                    if neighbor in points:
                        points.remove(neighbor)
                        queue.append(neighbor)
                        found.append(neighbor)
        components.append(np.asarray(found, dtype=np.int32))
    return components


def _trace_components(array: np.ndarray) -> list[np.ndarray]:
    spread = array.max(axis=2).astype(np.int16) - array.min(axis=2).astype(np.int16)
    mask = (spread > 45) & (array.min(axis=2) < 245)
    components = [item for item in _components(mask) if len(item) >= 90]
    components.sort(key=len, reverse=True)
    if len(components) != 2:
        raise ValueError(f"expected two colored traces, recovered {len(components)}")
    return components


def _baseline_row(array: np.ndarray, component: np.ndarray) -> float:
    y_values = component[:, 0]
    x_values = component[:, 1]
    x0 = max(0, int(x_values.min()) - 10)
    x1 = min(array.shape[1], int(x_values.max()) + 11)
    spread = array.max(axis=2).astype(np.int16) - array.min(axis=2).astype(np.int16)
    intensity = array.mean(axis=2)
    dark_gray = (spread <= 30) & (intensity < 175)
    counts = dark_gray[:, x0:x1].sum(axis=1)
    row = int(np.argmax(counts))
    if int(counts[row]) < 12:
        raise ValueError("visible dashed baseline was not recovered")
    return float(row)


def _crossings(component: np.ndarray, baseline: float) -> int:
    by_x: dict[int, list[int]] = {}
    for y, x in component.tolist():
        by_x.setdefault(int(x), []).append(int(y))
    if len(by_x) < 12:
        raise ValueError("trace has too few horizontal samples")
    signs: list[int] = []
    for x in sorted(by_x):
        center = float(np.median(by_x[x]))
        distance = center - baseline
        if abs(distance) <= 2.0:
            continue
        signs.append(1 if distance > 0 else -1)
    if len(signs) < 8:
        raise ValueError("trace centerline is too close to its baseline")
    compressed: list[int] = []
    for sign in signs:
        if not compressed or compressed[-1] != sign:
            compressed.append(sign)
    return sum(a != b for a, b in zip(compressed, compressed[1:]))


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    traces = _trace_components(array)
    local_counts = []
    for component in traces:
        baseline = _baseline_row(array, component)
        local_counts.append(_crossings(component, baseline))
    if any(count < 1 or count > 5 for count in local_counts):
        raise ValueError("recovered crossing count is outside the declared scene range")
    return "odd" if sum(local_counts) % 2 else "even"
