"""Renderer and analytic reference for the paired trace parity world."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw


WIDTH = 512
HEIGHT = 512
TRACE_COLORS = ("#D94841", "#1D70A2", "#7A3E9D", "#0F766E")
BASELINE_COLORS = ("#263238", "#3B4652", "#344054")
PANEL_COLORS = ("#C7D2DE", "#D0D5DD", "#CBD5E1")
BACKGROUNDS = ("#FFFFFF", "#FAFAF7", "#F8FAFC")
DASH_PATTERNS = ((8, 6), (10, 7), (7, 6))


def _crossing_count(trace: list[list[int]], baseline: int) -> int:
    signs = [1 if int(point[1]) > baseline else -1 for point in trace]
    return sum(a != b for a, b in zip(signs, signs[1:]))


def _make_trace(
    *, panel_x: int, panel_y: int, panel_width: int, baseline: int,
    rng: random.Random, transitions: int
) -> list[list[int]]:
    n_points = 6
    x0 = panel_x + 8
    usable = max(42, panel_width - 16)
    step = usable / (n_points - 1)
    transition_slots = set(rng.sample(range(n_points - 1), transitions))
    sign = 1 if rng.randrange(2) else -1
    magnitudes = [rng.choice((17, 19, 22, 25, 28)) for _ in range(n_points)]
    points: list[list[int]] = []
    for index in range(n_points):
        if index - 1 in transition_slots:
            sign *= -1
        x = int(round(x0 + index * step))
        y = int(baseline + sign * magnitudes[index])
        points.append([x, y])
    return points


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a deterministic, balanced scene from one integer seed."""

    rng = random.Random(int(seed))
    left_panel_x = rng.choice((96, 98, 100))
    right_panel_x = rng.choice((340, 342, 344))
    left_panel_y = rng.choice((142, 144, 146))
    right_panel_y = rng.choice((142, 144, 146))
    panel_width = rng.choice((68, 70, 72))
    panel_height = rng.choice((106, 108, 110))
    left_baseline_y = left_panel_y + rng.choice((48, 52, 56))
    right_baseline_y = right_panel_y + rng.choice((46, 51, 56))

    desired_odd = bool(int(seed) & 1)
    left_count = rng.choice((1, 2, 3, 4))
    allowed_right = [value for value in (1, 2, 3, 4) if (value + left_count) % 2 == int(desired_odd)]
    right_count = rng.choice(allowed_right)

    left_trace = _make_trace(
        panel_x=left_panel_x,
        panel_y=left_panel_y,
        panel_width=panel_width,
        baseline=left_baseline_y,
        rng=rng,
        transitions=left_count,
    )
    right_trace = _make_trace(
        panel_x=right_panel_x,
        panel_y=right_panel_y,
        panel_width=panel_width,
        baseline=right_baseline_y,
        rng=rng,
        transitions=right_count,
    )
    return {
        "left_trace": left_trace,
        "right_trace": right_trace,
        "left_baseline_y": left_baseline_y,
        "right_baseline_y": right_baseline_y,
        "left_panel_x": left_panel_x,
        "left_panel_y": left_panel_y,
        "right_panel_x": right_panel_x,
        "right_panel_y": right_panel_y,
        "panel_width": panel_width,
        "panel_height": panel_height,
        "trace_color": rng.choice(TRACE_COLORS),
        "baseline_color": rng.choice(BASELINE_COLORS),
        "panel_color": rng.choice(PANEL_COLORS),
        "background": rng.choice(BACKGROUNDS),
        "stroke_width": rng.choice((4, 5)),
        "baseline_dash": list(rng.choice(DASH_PATTERNS)),
    }


def _draw_dashed_line(
    draw: ImageDraw.ImageDraw, x0: int, x1: int, y: int, color: str,
    width: int, dash: list[int], scale: int
) -> None:
    cursor = x0
    dash_index = 0
    while cursor < x1:
        length = int(dash[dash_index % len(dash)])
        end = min(x1, cursor + length)
        if dash_index % 2 == 0:
            draw.line(
                [(cursor * scale, y * scale), (end * scale, y * scale)],
                fill=color,
                width=width * scale,
            )
        cursor = end
        dash_index += 1


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only visible scene fields into a deterministic RGB PNG image."""

    supersample = 4
    image = Image.new(
        "RGB", (WIDTH * supersample, HEIGHT * supersample), scene["background"]
    )
    draw = ImageDraw.Draw(image)
    panel_color = scene["panel_color"]
    panel_width = int(scene["panel_width"])
    panel_height = int(scene["panel_height"])
    left_x = int(scene["left_panel_x"])
    left_y = int(scene["left_panel_y"])
    right_x = int(scene["right_panel_x"])
    right_y = int(scene["right_panel_y"])
    for px, py in ((left_x, left_y), (right_x, right_y)):
        draw.rectangle(
            [
                px * supersample,
                py * supersample,
                (px + panel_width) * supersample,
                (py + panel_height) * supersample,
            ],
            outline=panel_color,
            width=2 * supersample,
        )

    baseline_color = scene["baseline_color"]
    dash = list(scene["baseline_dash"])
    _draw_dashed_line(
        draw,
        left_x + 5,
        left_x + panel_width - 5,
        int(scene["left_baseline_y"]),
        baseline_color,
        2,
        dash,
        supersample,
    )
    _draw_dashed_line(
        draw,
        right_x + 5,
        right_x + panel_width - 5,
        int(scene["right_baseline_y"]),
        baseline_color,
        2,
        dash,
        supersample,
    )

    trace_color = scene["trace_color"]
    stroke_width = int(scene["stroke_width"])
    draw.line(
        [(int(point[0]) * supersample, int(point[1]) * supersample) for point in scene["left_trace"]],
        fill=trace_color,
        width=stroke_width * supersample,
        joint="curve",
    )
    draw.line(
        [(int(point[0]) * supersample, int(point[1]) * supersample) for point in scene["right_trace"]],
        fill=trace_color,
        width=stroke_width * supersample,
        joint="curve",
    )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    left = _crossing_count(scene["left_trace"], int(scene["left_baseline_y"]))
    right = _crossing_count(scene["right_trace"], int(scene["right_baseline_y"]))
    return "odd" if (left + right) % 2 else "even"


def margin(scene: dict[str, Any]) -> float:
    clearances = [
        abs(int(point[1]) - int(scene["left_baseline_y"]))
        for point in scene["left_trace"]
    ] + [
        abs(int(point[1]) - int(scene["right_baseline_y"]))
        for point in scene["right_trace"]
    ]
    return float(min(clearances))


def is_quarantined(scene: dict[str, Any]) -> bool:
    if margin(scene) < 14.0:
        return True
    for trace, baseline in (
        (scene["left_trace"], scene["left_baseline_y"]),
        (scene["right_trace"], scene["right_baseline_y"]),
    ):
        if any(int(point[1]) == int(baseline) for point in trace):
            return True
    return False


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
