"""Analytic sampler and rasterizer for the fourfold broad-arm sweep fan."""

from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image

QUESTION = (
    "The single colored shape has four matching broad arms around its center. "
    "As the arms extend outward, do they sweep clockwise or counterclockwise? "
    "Answer clockwise or counterclockwise."
)

_PALETTES = (
    ("#1d4ed8", "#f8fafc"),
    ("#be123c", "#fff7ed"),
    ("#047857", "#f0fdfa"),
    ("#6d28d9", "#faf5ff"),
    ("#b45309", "#fffbeb"),
)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    foreground, background = _PALETTES[rng.randrange(len(_PALETTES))]
    sign = 1.0 if int(seed) % 2 == 0 else -1.0
    # Every eighth scene approaches the declared zero-drift boundary while
    # remaining raster-answerable; the rest have a clear broad sweep.
    magnitude = rng.uniform(0.32, 0.40) if int(seed) % 8 == 0 else rng.uniform(0.68, 0.92)
    return {
        "width": 384,
        "height": 384,
        "center_x": 192.0 + rng.uniform(-9.0, 9.0),
        "center_y": 192.0 + rng.uniform(-9.0, 9.0),
        "outer_radius": rng.uniform(154.0, 164.0),
        "core_radius": rng.uniform(54.0, 61.0),
        "arm_halfwidth": rng.uniform(0.305, 0.365),
        "sweep_radians": sign * magnitude,
        "base_angle": rng.uniform(0.0, math.pi / 2.0),
        "foreground": foreground,
        "background": background,
    }


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    outer = float(scene["outer_radius"])
    core = float(scene["core_radius"])
    halfwidth = float(scene["arm_halfwidth"])
    sweep = float(scene["sweep_radians"])
    base = float(scene["base_angle"]) % (math.pi / 2.0)
    foreground = str(scene["foreground"])
    background = str(scene["background"])

    scale = 3
    yy, xx = np.mgrid[0 : height * scale, 0 : width * scale]
    x = (xx + 0.5) / scale
    y = (yy + 0.5) / scale
    dx = x - cx
    dy = y - cy
    radius = np.hypot(dx, dy)
    theta = np.arctan2(dy, dx)  # positive angles run clockwise in image coordinates
    u = np.clip((radius - core) / (outer - core), 0.0, 1.0)
    arm_center = base + sweep * u
    period = math.pi / 2.0
    phase = (theta - arm_center + period / 2.0) % period - period / 2.0
    tapered_halfwidth = halfwidth * (1.0 - 0.10 * u)
    mask = (radius <= core) | ((radius <= outer) & (np.abs(phase) <= tapered_halfwidth))

    fg = np.asarray(Image.new("RGB", (1, 1), foreground), dtype=np.uint8)[0, 0]
    bg = np.asarray(Image.new("RGB", (1, 1), background), dtype=np.uint8)[0, 0]
    array = np.empty((height * scale, width * scale, 3), dtype=np.uint8)
    array[:] = bg
    array[mask] = fg
    image = Image.fromarray(array, mode="RGB")
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "clockwise" if float(scene["sweep_radians"]) > 0.0 else "counterclockwise"


def margin(scene: dict) -> float:
    return abs(float(scene["sweep_radians"]))


def is_quarantined(scene: dict) -> bool:
    return abs(float(scene["sweep_radians"])) < 0.42


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
