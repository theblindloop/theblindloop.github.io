"""Pixel-only inverse for a broad fourfold radial phase field.

The inverse estimates the border color, segments the one foreground component,
finds its mass centroid, reconstructs fourth-harmonic phase in radial bands,
and classifies the sign of the fitted phase slope.  It imports no world code and
receives only the final raster.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    pass


def _segment(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    border = np.concatenate(
        (array[:4].reshape(-1, 3), array[-4:].reshape(-1, 3),
         array[:, :4].reshape(-1, 3), array[:, -4:].reshape(-1, 3)), axis=0
    )
    background = np.median(border, axis=0)
    delta = np.linalg.norm(array - background, axis=2)
    mask = delta > 48.0
    if int(mask.sum()) < 8000:
        raise OracleAbstention("no substantial foreground region")
    if np.any(mask[:3]) or np.any(mask[-3:]) or np.any(mask[:, :3]) or np.any(mask[:, -3:]):
        raise OracleAbstention("foreground touches the image border")
    return mask


def _component_size(mask: np.ndarray, start: tuple[int, int]) -> int:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    stack = [start]
    seen[start] = True
    count = 0
    while stack:
        y, x = stack.pop()
        count += 1
        if y and mask[y - 1, x] and not seen[y - 1, x]:
            seen[y - 1, x] = True
            stack.append((y - 1, x))
        if y + 1 < height and mask[y + 1, x] and not seen[y + 1, x]:
            seen[y + 1, x] = True
            stack.append((y + 1, x))
        if x and mask[y, x - 1] and not seen[y, x - 1]:
            seen[y, x - 1] = True
            stack.append((y, x - 1))
        if x + 1 < width and mask[y, x + 1] and not seen[y, x + 1]:
            seen[y, x + 1] = True
            stack.append((y, x + 1))
    return count


def _validate_connected(mask: np.ndarray) -> None:
    points = np.argwhere(mask)
    size = _component_size(mask, tuple(int(v) for v in points[len(points) // 2]))
    if size < 0.985 * len(points):
        raise OracleAbstention("foreground is not one connected region")


def _fit(image: Image.Image) -> tuple[float, float]:
    mask = _segment(image)
    _validate_connected(mask)
    yy, xx = np.nonzero(mask)
    cx = float(xx.mean() + 0.5)
    cy = float(yy.mean() + 0.5)
    dx = xx.astype(np.float64) + 0.5 - cx
    dy = yy.astype(np.float64) + 0.5 - cy
    radius = np.hypot(dx, dy)
    theta = np.arctan2(dy, dx)
    outer = float(np.percentile(radius, 99.75))
    if not (120.0 <= outer <= 185.0):
        raise OracleAbstention("implausible silhouette scale")

    # The filled core is the connected bridge between all broad arms.
    inner = radius <= 0.29 * outer
    inner_area = math.pi * (0.29 * outer) ** 2
    if float(inner.sum()) < 0.91 * inner_area:
        raise OracleAbstention("central connecting core is incomplete")

    # Begin outside the largest allowed filled core; including the core would
    # correctly have no angular phase, but would dilute the first arm band.
    centers = np.linspace(0.46, 0.91, 9)
    phases: list[float] = []
    amplitudes: list[float] = []
    imbalances: list[float] = []
    for center in centers:
        band = (radius >= (center - 0.035) * outer) & (radius <= (center + 0.035) * outer)
        angles = theta[band]
        if len(angles) < 180:
            raise OracleAbstention("radial evidence band is incomplete")
        moment = np.mean(np.exp(4j * angles))
        phases.append(float(np.angle(moment)))
        amplitudes.append(float(abs(moment)))
        sectors = ((angles + math.pi) // (math.pi / 2.0)).astype(int).clip(0, 3)
        counts = np.bincount(sectors, minlength=4).astype(np.float64)
        imbalances.append(float((counts.max() - counts.min()) / counts.mean()))

    if min(amplitudes) < 0.46:
        raise OracleAbstention("fourfold arm phase is not coherent")
    if max(imbalances) > 0.115:
        raise OracleAbstention("the four matching arms do not all contribute")

    unwrapped = np.unwrap(np.asarray(phases, dtype=np.float64))
    xvals = centers - centers.mean()
    slope4 = float(np.dot(xvals, unwrapped - unwrapped.mean()) / np.dot(xvals, xvals))
    fitted = unwrapped.mean() + slope4 * xvals
    rms = float(np.sqrt(np.mean((unwrapped - fitted) ** 2)))
    slope = slope4 / 4.0
    if rms > 0.20 or abs(slope) < 0.20:
        raise OracleAbstention("radial phase drift is ambiguous")
    return slope, rms


def decision_from_image(image: Image.Image) -> str:
    slope, _ = _fit(image)
    return "clockwise" if slope > 0.0 else "counterclockwise"


def measurement_margin(image: Image.Image) -> float:
    slope, rms = _fit(image)
    return abs(slope) / max(0.01, rms)
