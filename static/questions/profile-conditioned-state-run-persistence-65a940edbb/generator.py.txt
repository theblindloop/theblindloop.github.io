"""Renderer and analytic answer for the ordered state-run persistence world."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw, ImageFont


CANVAS = (1536, 480)
PANEL_COUNT = 7
PANEL_X0 = 18
PANEL_Y = 110
PANEL_W = 202
PANEL_H = 338
PANEL_GAP = 12
CARD_X_OFFSET = 10
CARD_START_OFFSET = 56
CARD_W = 182
ROW_GAP = 8
TILE_HEIGHT = 52
BADGE_SIZE = 36
BADGE_X_OFFSET = 10
GLYPH_SCALE = 4
GLYPH_X_OFFSET = 22
GLYPH_Y_OFFSET = 16
STATE_CENTER_X_OFFSET = 144
STATE_TEXT_X_OFFSET = 158
STATE_NAMES = ("CALM", "ACTIVE", "PAUSE")
LABELS = ("A", "B", "C", "D")
STATE_COUNT = 3

BACKGROUND = (248, 246, 239)
PANEL_FILL = (255, 253, 248)
PANEL_BORDER = (70, 79, 92)
CARD_FILL = (255, 255, 255)
CARD_BORDER = (200, 205, 211)
BADGE_FILL = (43, 52, 67)
GLYPH_FILL = (245, 247, 250)
TEXT_FILL = (40, 46, 55)
MUTED_FILL = (91, 99, 111)

PALETTE_OPTIONS = (
    ((49, 117, 190), (221, 143, 47), (190, 70, 91)),
    ((47, 145, 107), (178, 80, 153), (211, 113, 42)),
    ((50, 125, 125), (198, 83, 71), (119, 91, 171)),
)

GLYPHS = {
    "A": ("010", "101", "111", "101", "101"),
    "B": ("110", "101", "110", "101", "110"),
    "C": ("011", "100", "100", "100", "011"),
    "D": ("110", "101", "101", "101", "110"),
}


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _centered_text(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font, fill) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2), text, font=font, fill=fill)


def _run_length(values: list[int] | tuple[int, ...]) -> int:
    if not values:
        return 0
    best = 1
    current = 1
    for left, right in zip(values, values[1:]):
        if left == right:
            current += 1
        else:
            current = 1
        best = max(best, current)
    return best


def _sequence_with_exact_max(rng: random.Random, target: int) -> list[int]:
    for _ in range(50_000):
        values = [rng.randrange(STATE_COUNT) for _ in range(PANEL_COUNT)]
        if _run_length(values) == target:
            return values
    raise RuntimeError(f"could not sample a state trajectory with max run {target}")


def _sequence_with_max_at_most(rng: random.Random, limit: int) -> list[int]:
    for _ in range(50_000):
        values = [rng.randrange(STATE_COUNT) for _ in range(PANEL_COUNT)]
        if _run_length(values) <= limit:
            return values
    raise RuntimeError(f"could not sample a state trajectory with max run <= {limit}")


def sample_scene(seed: int) -> dict:
    """Sample a balanced seven-panel state trajectory with a unique winner."""
    rng = random.Random(int(seed))
    near_boundary = int(seed) % 5 == 0 or int(seed) % 7 == 0
    gap_target = 1 if near_boundary else rng.choice((2, 3))
    max_target = rng.choice((4, 5, 6)) if gap_target == 3 else rng.choice((3, 4, 5))
    if max_target - gap_target < 1:
        max_target = gap_target + 1

    winner = LABELS[rng.randrange(len(LABELS))]
    competitor_choices = [label for label in LABELS if label != winner]
    exact_competitor = competitor_choices[rng.randrange(len(competitor_choices))]
    trajectories: dict[str, list[int]] = {}
    trajectories[winner] = _sequence_with_exact_max(rng, max_target)
    trajectories[exact_competitor] = _sequence_with_exact_max(rng, max_target - gap_target)
    for label in LABELS:
        if label in trajectories:
            continue
        trajectories[label] = _sequence_with_max_at_most(
            rng, max(1, max_target - gap_target - 1)
        )

    # The helper above deliberately receives the RNG last in the call site below;
    # keep the generated order independent of the label names.
    if any(not isinstance(values, list) for values in trajectories.values()):
        raise AssertionError("trajectory sampling failed")
    trajectories = {label: trajectories[label] for label in reversed(LABELS)}

    panel_rows: list[list[str]] = []
    for _ in range(PANEL_COUNT):
        row = list(LABELS)
        rng.shuffle(row)
        panel_rows.append(row)

    base_palette = PALETTE_OPTIONS[rng.randrange(len(PALETTE_OPTIONS))]
    permutation = list(range(STATE_COUNT))
    rng.shuffle(permutation)
    state_palette = [list(base_palette[index]) for index in permutation]
    state_names = [STATE_NAMES[index] for index in permutation]
    swatch_side = rng.choice((22, 24, 26, 28))

    return {
        "width": CANVAS[0],
        "height": CANVAS[1],
        "panel_count": PANEL_COUNT,
        "panel_order": [f"P{index + 1}" for index in range(PANEL_COUNT)],
        "panel_rows": panel_rows,
        "trajectories": trajectories,
        "state_palette": state_palette,
        "state_names": state_names,
        "swatch_side": swatch_side,
        "tile_height": TILE_HEIGHT,
    }


def _draw_identity_badge(draw: ImageDraw.ImageDraw, left: int, top: int, label: str) -> None:
    draw.rectangle(
        [left, top, left + BADGE_SIZE - 1, top + BADGE_SIZE - 1],
        fill=BADGE_FILL,
        outline=BADGE_FILL,
    )
    glyph = GLYPHS[label]
    for row_index, row in enumerate(glyph):
        for col_index, bit in enumerate(row):
            if bit == "1":
                x0 = left + GLYPH_X_OFFSET - BADGE_X_OFFSET + col_index * GLYPH_SCALE
                y0 = top + GLYPH_Y_OFFSET + row_index * GLYPH_SCALE
                draw.rectangle(
                    [x0, y0, x0 + GLYPH_SCALE - 1, y0 + GLYPH_SCALE - 1],
                    fill=GLYPH_FILL,
                )


def render(scene: dict) -> Image.Image:
    """Render only the visible board; all state marks are solid interior pixels."""
    width = int(scene["width"])
    height = int(scene["height"])
    panel_count = int(scene["panel_count"])
    panel_order = list(scene["panel_order"])
    panel_rows = scene["panel_rows"]
    trajectories = scene["trajectories"]
    state_palette = scene["state_palette"]
    state_names = scene["state_names"]
    swatch_side = int(scene["swatch_side"])
    tile_height = int(scene["tile_height"])

    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(20)
    panel_font = _font(15)
    small_font = _font(10)
    draw.text((18, 16), "STATE PERSISTENCE BOARD", font=title_font, fill=TEXT_FILL)
    draw.text((18, 48), "Longest uninterrupted run of one state", font=small_font, fill=MUTED_FILL)

    legend_x = 390
    draw.text((legend_x, 18), "STATE KEY", font=small_font, fill=MUTED_FILL)
    for state_index, (color, name) in enumerate(zip(state_palette, state_names)):
        x = legend_x + 78 + state_index * 126
        draw.rectangle([x, 17, x + 22, 39], fill=tuple(color), outline=PANEL_BORDER)
        draw.text((x + 30, 21), str(name), font=small_font, fill=TEXT_FILL)

    for panel_index in range(panel_count):
        panel_x = PANEL_X0 + panel_index * (PANEL_W + PANEL_GAP)
        draw.rectangle(
            [panel_x, PANEL_Y, panel_x + PANEL_W, PANEL_Y + PANEL_H],
            fill=PANEL_FILL,
            outline=PANEL_BORDER,
            width=2,
        )
        _centered_text(
            draw,
            (panel_x + PANEL_W / 2, PANEL_Y + 24),
            str(panel_order[panel_index]),
            panel_font,
            TEXT_FILL,
        )
        draw.line(
            [panel_x + 12, PANEL_Y + 43, panel_x + PANEL_W - 12, PANEL_Y + 43],
            fill=(221, 222, 220),
            width=1,
        )
        for row_index, label in enumerate(panel_rows[panel_index]):
            card_x = panel_x + CARD_X_OFFSET
            card_y = PANEL_Y + CARD_START_OFFSET + row_index * (tile_height + ROW_GAP)
            draw.rectangle(
                [card_x, card_y, card_x + CARD_W, card_y + tile_height],
                fill=CARD_FILL,
                outline=CARD_BORDER,
                width=1,
            )
            badge_top = card_y + (tile_height - BADGE_SIZE) // 2
            _draw_identity_badge(draw, card_x + BADGE_X_OFFSET, badge_top, label)
            state_index = int(trajectories[label][panel_index])
            center_x = panel_x + STATE_CENTER_X_OFFSET
            center_y = card_y + tile_height // 2
            half = swatch_side // 2
            draw.rectangle(
                [center_x - half, center_y - half, center_x + half - 1, center_y + half - 1],
                fill=tuple(state_palette[state_index]),
                outline=PANEL_BORDER,
            )
            draw.text(
                (panel_x + STATE_TEXT_X_OFFSET, center_y - 6),
                str(state_names[state_index]),
                font=small_font,
                fill=MUTED_FILL,
            )

        if panel_index < panel_count - 1:
            arrow_y = PANEL_Y + PANEL_H // 2
            start_x = panel_x + PANEL_W + 2
            end_x = panel_x + PANEL_W + PANEL_GAP - 2
            draw.line([start_x, arrow_y, end_x, arrow_y], fill=MUTED_FILL, width=2)
            draw.polygon(
                [(end_x, arrow_y), (end_x - 6, arrow_y - 4), (end_x - 6, arrow_y + 4)],
                fill=MUTED_FILL,
            )
    return image


def analytic_gold(scene: dict) -> str:
    """Return the unique label with the largest contiguous state run."""
    scores = {
        label: _run_length(values)
        for label, values in sorted(scene["trajectories"].items())
    }
    ranking = sorted(scores.items(), key=lambda item: (-item[1], item[0]))
    if len(ranking) < 2 or ranking[0][1] == ranking[1][1]:
        raise ValueError("ambiguous longest state run")
    return ranking[0][0]


def margin(scene: dict) -> float:
    scores = sorted(
        (_run_length(values) for values in scene["trajectories"].values()),
        reverse=True,
    )
    if len(scores) < 2:
        return 0.0
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse the storage order of the label-keyed trajectory mapping."""
    twin = copy.deepcopy(scene)
    twin["trajectories"] = {
        label: copy.deepcopy(values)
        for label, values in reversed(list(scene["trajectories"].items()))
    }
    return [("trajectory_mapping_storage_order", twin)]
