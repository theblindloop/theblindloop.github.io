"""Independent pixel-only inverse for the state-run persistence board."""

from __future__ import annotations

from PIL import Image


PANEL_COUNT = 7
PANEL_X0 = 18
PANEL_Y = 110
PANEL_W = 202
PANEL_GAP = 12
CARD_START_OFFSET = 56
CARD_X_OFFSET = 10
ROW_GAP = 8
TILE_HEIGHT = 52
BADGE_X_OFFSET = 10
GLYPH_X_OFFSET = 22
GLYPH_Y_OFFSET = 16
GLYPH_SCALE = 4
STATE_CENTER_X_OFFSET = 144
LABELS = ("A", "B", "C", "D")

GLYPHS = {
    "A": ("010", "101", "111", "101", "101"),
    "B": ("110", "101", "110", "101", "110"),
    "C": ("011", "100", "100", "100", "011"),
    "D": ("110", "101", "101", "101", "110"),
}
PATTERN_TO_LABEL = {tuple(row for row in pattern): label for label, pattern in GLYPHS.items()}


def _glyph_from_pixels(pixels, x: int, y: int) -> str:
    pattern = []
    for row_index in range(5):
        bits = []
        for col_index in range(3):
            rgb = pixels[x + col_index * GLYPH_SCALE + 2, y + row_index * GLYPH_SCALE + 2]
            bits.append("1" if sum(rgb) > 500 else "0")
        pattern.append("".join(bits))
    key = tuple(pattern)
    if key not in PATTERN_TO_LABEL:
        raise ValueError(f"unresolved identity glyph {key!r}")
    return PATTERN_TO_LABEL[key]


def _longest_run(values: list[tuple[int, int, int]]) -> int:
    if not values:
        return 0
    best = 1
    current = 1
    for left, right in zip(values, values[1:]):
        if left == right:
            current += 1
        else:
            current = 1
        best = max(best, current)
    return best


def decision_from_image(image: Image.Image) -> str:
    """Decode panel rows, identity glyphs, and state swatches from the PNG alone."""
    raster = image.convert("RGB")
    if raster.width < PANEL_X0 + PANEL_COUNT * PANEL_W:
        raise ValueError(f"unexpected board width {raster.width}")
    pixels = raster.load()
    trajectories: dict[str, list[tuple[int, int, int]]] = {label: [] for label in LABELS}

    for panel_index in range(PANEL_COUNT):
        panel_x = PANEL_X0 + panel_index * (PANEL_W + PANEL_GAP)
        seen: set[str] = set()
        for row_index in range(4):
            card_x = panel_x + CARD_X_OFFSET
            card_y = PANEL_Y + CARD_START_OFFSET + row_index * (TILE_HEIGHT + ROW_GAP)
            glyph_x = card_x + BADGE_X_OFFSET + GLYPH_X_OFFSET - BADGE_X_OFFSET
            glyph_y = card_y + (TILE_HEIGHT - 36) // 2 + GLYPH_Y_OFFSET
            label = _glyph_from_pixels(pixels, glyph_x, glyph_y)
            if label in seen:
                raise ValueError(f"duplicate identity {label} in panel {panel_index + 1}")
            seen.add(label)
            state_x = panel_x + STATE_CENTER_X_OFFSET
            state_y = card_y + TILE_HEIGHT // 2
            state = pixels[state_x, state_y]
            trajectories[label].append(tuple(state))
        if seen != set(LABELS):
            raise ValueError(f"panel {panel_index + 1} does not contain A-D")

    scores = {label: _longest_run(values) for label, values in trajectories.items()}
    ranking = sorted(scores.items(), key=lambda item: (-item[1], item[0]))
    if len(ranking) < 2 or ranking[0][1] == ranking[1][1]:
        raise ValueError("ambiguous pixel trajectory")
    return ranking[0][0]
