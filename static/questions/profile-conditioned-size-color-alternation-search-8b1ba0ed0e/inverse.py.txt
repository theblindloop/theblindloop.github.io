"""Independent pixel-only inverse arm; intentionally imports no world module."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")
COLOR_RGB = ((22, 140, 140), (227, 161, 38))
INK_RGB = (32, 38, 44)


def _near(rgb: np.ndarray, target: tuple[int, int, int], tolerance: float) -> np.ndarray:
    delta = rgb.astype(np.int32) - np.asarray(target, dtype=np.int32)
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _components(mask: np.ndarray, minimum: int = 30) -> list[tuple[np.ndarray, np.ndarray]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y0, x0 in zip(*np.nonzero(mask)):
        if seen[y0, x0]:
            continue
        stack = [(int(y0), int(x0))]
        seen[y0, x0] = True
        ys, xs = [], []
        while stack:
            y, x = stack.pop()
            ys.append(y); xs.append(x)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(xs) >= minimum:
            found.append((np.asarray(ys), np.asarray(xs)))
    return found


def _ring_radius(dark: np.ndarray, cx: float, cy: float) -> float:
    y0, y1 = max(0, int(cy - 48)), min(dark.shape[0], int(cy + 49))
    x0, x1 = max(0, int(cx - 48)), min(dark.shape[1], int(cx + 49))
    ys, xs = np.nonzero(dark[y0:y1, x0:x1])
    if len(xs) < 35:
        raise ValueError("ring missing")
    distances = np.hypot(xs + x0 - cx, ys + y0 - cy)
    distances = distances[(distances >= 12.0) & (distances <= 45.0)]
    if len(distances) < 30:
        raise ValueError("ring outline missing")
    bins = np.arange(11.5, 46.0, 1.0)
    histogram, edges = np.histogram(distances, bins=bins)
    peak = int(np.argmax(histogram))
    return float((edges[peak] + edges[peak + 1]) / 2.0)


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"))
    if rgb.shape[1] < 600 or rgb.shape[0] < 250:
        return "abstain"
    color_masks = [_near(rgb, color, 46.0) for color in COLOR_RGB]
    colored = color_masks[0] | color_masks[1]
    blobs = _components(colored, minimum=70)
    if len(blobs) != 16:
        return "abstain"
    dark = _near(rgb, INK_RGB, 55.0)
    objects = []
    for ys, xs in blobs:
        cx, cy = float(xs.mean()), float(ys.mean())
        pixels = rgb[ys, xs].astype(float)
        mean = pixels.mean(axis=0)
        color = int(np.argmin([np.linalg.norm(mean - np.asarray(target)) for target in COLOR_RGB]))
        try:
            radius = _ring_radius(dark, cx, cy)
        except ValueError:
            return "abstain"
        objects.append((cx, cy, radius, color))
    objects.sort(key=lambda item: item[0])
    panels = [objects[index * 4:(index + 1) * 4] for index in range(4)]
    if any(max(x for x, _, _, _ in panel) - min(x for x, _, _, _ in panel) > rgb.shape[1] * 0.24 for panel in panels):
        return "abstain"
    scores = []
    for panel in panels:
        ordered = sorted(panel, key=lambda item: item[2])
        radii = [item[2] for item in ordered]
        if min(radii[i + 1] - radii[i] for i in range(3)) < 4.0:
            return "abstain"
        sequence = [item[3] for item in ordered]
        scores.append(sum(sequence[i] != sequence[i + 1] for i in range(3)))
    best = max(scores)
    winners = [index for index, score in enumerate(scores) if score == best]
    if len(winners) != 1 or best != 3 or best - sorted(scores, reverse=True)[1] < 1:
        return "abstain"
    return LABELS[winners[0]]
