"""Independent image-only inverse for the debounced signal world."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class AmbiguousImage(ValueError):
    pass


def _components(mask: np.ndarray):
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        points = []
        while queue:
            yy, xx = queue.popleft()
            points.append((yy, xx))
            for ny, nx in ((yy-1,xx),(yy+1,xx),(yy,xx-1),(yy,xx+1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny,nx] and not seen[ny,nx]:
                    seen[ny,nx] = True
                    queue.append((ny,nx))
        if len(points) >= 180:
            output.append(points)
    return output


def _read_states(image: Image.Image) -> list[int]:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    high = array.max(axis=2)
    low = array.min(axis=2)
    mask = (high - low > 55) & (high > 80) & (low < 170)
    marks = []
    for points in _components(mask):
        ys = np.fromiter((point[0] for point in points), dtype=int)
        xs = np.fromiter((point[1] for point in points), dtype=int)
        box_width = int(xs.max() - xs.min() + 1)
        box_height = int(ys.max() - ys.min() + 1)
        if 32 <= box_width <= 55 and 32 <= box_height <= 55 and 0.75 <= box_width / box_height <= 1.3:
            fill_ratio = len(points) / float(box_width * box_height)
            if fill_ratio > 0.61:
                state = 1
            elif fill_ratio < 0.46:
                state = 0
            else:
                raise AmbiguousImage(f"signal fill ratio {fill_ratio:.3f} has no clear state")
            marks.append((float(xs.mean()), float(ys.mean()), state))
    if len(marks) != 8:
        raise AmbiguousImage(f"expected 8 signal marks, recovered {len(marks)}")
    marks.sort()
    if max(mark[1] for mark in marks) - min(mark[1] for mark in marks) > 12:
        raise AmbiguousImage("signal marks do not form one ordered row")
    gaps = [marks[index+1][0] - marks[index][0] for index in range(7)]
    if min(gaps) < 55:
        raise AmbiguousImage("signal order is unresolved")
    return [mark[2] for mark in marks]


def decision_from_image(image: Image.Image) -> str:
    states = _read_states(image)
    output = states[0]
    pending = None
    for sample in states[1:]:
        if sample == output:
            pending = None
        elif pending == sample:
            output = sample
            pending = None
        else:
            pending = sample
    return "ON" if output else "OFF"
