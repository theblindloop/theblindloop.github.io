"""Latent sampler and Pillow renderer for Two-Sample Debounced Signal."""

from __future__ import annotations

import itertools
import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1160, 430
N_SAMPLES = 7
PALETTE = ("#1769aa", "#167c5a", "#7546a8", "#b54b2a")


def _debounce(initial: int, samples: list[int]) -> int:
    output = int(initial)
    pending = None
    for sample in samples:
        sample = int(sample)
        if sample == output:
            pending = None
        elif pending == sample:
            output = sample
            pending = None
        else:
            pending = sample
    return output


def _visual_samples(scene: dict) -> list[int]:
    values = list(scene["samples"])
    if scene["display_reversed"]:
        values.reverse()
    return values


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = int(seed) % 2
    possibilities = []
    for initial in (0, 1):
        for values in itertools.product((0, 1), repeat=N_SAMPLES):
            if _debounce(initial, list(values)) == target:
                possibilities.append((initial, list(values)))
    initial, visual = rng.choice(possibilities)
    display_reversed = bool(rng.randrange(2))
    stored = list(reversed(visual)) if display_reversed else list(visual)
    return {
        "initial_output": initial,
        "samples": stored,
        "signal_color": PALETTE[rng.randrange(len(PALETTE))],
        "display_reversed": display_reversed,
    }


def analytic_gold(scene: dict) -> str:
    final = _debounce(int(scene["initial_output"]), _visual_samples(scene))
    return "ON" if final else "OFF"


def margin(scene: dict) -> float:
    outer, inner = 22.0, 15.0
    solid_fill = math.pi * outer * outer / ((2 * outer + 1) ** 2)
    ring_fill = math.pi * (outer * outer - inner * inner) / ((2 * outer + 1) ** 2)
    return round(min(solid_fill - 0.53, 0.53 - ring_fill), 6)


def is_quarantined(scene: dict) -> bool:
    values = [scene.get("initial_output"), *scene.get("samples", [])]
    return len(values) != N_SAMPLES + 1 or any(v not in (0, 1) for v in values)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["samples"] = list(reversed(twin["samples"]))
    twin["display_reversed"] = not bool(twin["display_reversed"])
    return [("reverse_storage_and_direction", twin)]


def _font(size: int, bold: bool = False):
    names = (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf" if bold
        else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
        "/usr/share/fonts/truetype/liberation2/LiberationSans-Bold.ttf" if bold
        else "/usr/share/fonts/truetype/liberation2/LiberationSans-Regular.ttf",
    )
    for name in names:
        try:
            return ImageFont.truetype(name, size)
        except OSError:
            pass
    return ImageFont.load_default()


def _center_text(draw: ImageDraw.ImageDraw, box, text: str, font, fill):
    bounds = draw.textbbox((0, 0), text, font=font)
    x = (box[0] + box[2] - (bounds[2] - bounds[0])) / 2
    y = (box[1] + box[3] - (bounds[3] - bounds[1])) / 2 - bounds[1]
    draw.text((x, y), text, font=font, fill=fill)


def _draw_signal(draw: ImageDraw.ImageDraw, cx: int, cy: int, state: int, color: str, scale: int):
    radius = 22 * scale
    box = (cx * scale - radius, cy * scale - radius, cx * scale + radius, cy * scale + radius)
    if state:
        draw.ellipse(box, fill=color)
    else:
        draw.ellipse(box, outline=color, width=7 * scale)


def render(scene: dict) -> Image.Image:
    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), "#fbfcfe")
    draw = ImageDraw.Draw(image)
    ink, muted, border = "#202733", "#687180", "#8b95a3"
    signal = scene["signal_color"]
    title_font = _font(25 * scale, True)
    label_font = _font(15 * scale, True)
    small_font = _font(13 * scale, False)

    _center_text(draw, (0, 14 * scale, WIDTH * scale, 56 * scale), "TWO-SAMPLE DEBOUNCED SIGNAL", title_font, ink)
    _center_text(draw, (0, 56 * scale, WIDTH * scale, 88 * scale), "Change output only after 2 matching new samples in a row", small_font, muted)

    y0, y1 = 128, 282
    start_x0, start_x1 = 35, 170
    draw.rounded_rectangle((start_x0*scale, y0*scale, start_x1*scale, y1*scale), radius=12*scale, fill="white", outline=border, width=2*scale)
    _center_text(draw, (start_x0*scale, (y0+10)*scale, start_x1*scale, (y0+42)*scale), "START", label_font, ink)
    _draw_signal(draw, (start_x0+start_x1)//2, 216, int(scene["initial_output"]), signal, scale)
    _center_text(draw, (start_x0*scale, 246*scale, start_x1*scale, 276*scale), "ON" if scene["initial_output"] else "OFF", small_font, muted)

    draw.line((183*scale, 205*scale, 224*scale, 205*scale), fill=muted, width=3*scale)
    draw.polygon(((224*scale,205*scale),(211*scale,197*scale),(211*scale,213*scale)), fill=muted)

    samples = _visual_samples(scene)
    card_w, gap, first_x = 105, 13, 238
    for index, state in enumerate(samples, start=1):
        x0 = first_x + (index - 1) * (card_w + gap)
        x1 = x0 + card_w
        draw.rounded_rectangle((x0*scale, y0*scale, x1*scale, y1*scale), radius=12*scale, fill="white", outline=border, width=2*scale)
        _center_text(draw, (x0*scale, (y0+10)*scale, x1*scale, (y0+42)*scale), f"SAMPLE {index}", label_font, ink)
        _draw_signal(draw, (x0+x1)//2, 216, int(state), signal, scale)
        _center_text(draw, (x0*scale, 246*scale, x1*scale, 276*scale), "ON" if state else "OFF", small_font, muted)

    _center_text(draw, (0, 307*scale, WIDTH*scale, 338*scale), "Read left to right", label_font, ink)
    _center_text(draw, (0, 340*scale, WIDTH*scale, 370*scale), "One opposite sample is pending; a matching current-state sample cancels it", small_font, muted)
    _center_text(draw, (0, 375*scale, WIDTH*scale, 406*scale), "Question: What is the final output — ON or OFF?", label_font, ink)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
