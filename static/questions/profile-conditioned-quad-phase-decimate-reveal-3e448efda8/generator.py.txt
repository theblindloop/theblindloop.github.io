"""quad_phase_decimate_reveal: two-axis 2x2-phase decimation glyph reveal world.

Declared transform: deinterlace the lattice by keeping only the even-even phase
cells (the top-left cell of every 2x2 block, i.e. a true decimation by two in
BOTH the row and column axes) and averaging within each kept cell. The raw
image is a statistically flat field of grey noise because every 2x2 block is
electrically balanced: its top-left even-even cell carries the full glyph bias
and the other three phase cells carry exactly -1/3 of that bias, so every block
(and therefore the whole field) sums to a uniform base. No single row, single
column, single diagonal parity, or local patch reveals the glyph; only keeping
one cell of every 2x2 block collapses the balanced phases to a coarse mosaic
that carries a solid triangle: upward (A) or downward (B).

Latent scene z = {noise, glyph, cell}. label = glyph.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RENDERER_VERSION = "quad_phase_decimate_reveal-0.1.0"

W = 512
H = 512
NGRID = 32          # lattice cells per side (multiple of 2)
CELL = W // NGRID   # 16
M = NGRID // 2      # mosaic cells per side after decimation (16)
BASE = 128.0
BIAS = 70.0         # full amplitude carried by the even-even phase cell
NOISE_SIGMA = 46.0
Q_TOL = 0.20        # quarantine threshold on the structural A/B separation


def glyph_masks() -> dict[str, np.ndarray]:
    """(M, M) bool masks: A upward triangle, B downward triangle."""
    r, c = np.indices((M, M))
    center = (M - 1) / 2.0
    maxhw = float(round(M * 0.40))  # ~6 -> base width ~13 of 16
    # Upward triangle: narrow at the top row, wide at the bottom row.
    hw_up = np.round((r / (M - 1)) * maxhw)
    mask_a = (c >= center - hw_up) & (c <= center + hw_up)
    # Downward triangle: wide at the top row, narrow at the bottom row.
    hw_down = np.round(((M - 1 - r) / (M - 1)) * maxhw)
    mask_b = (c >= center - hw_down) & (c <= center + hw_down)
    return {"A": mask_a, "B": mask_b}


def build_scene(noise: int, glyph: str, cell: int = CELL) -> dict:
    return {
        "noise": int(noise),
        "glyph": glyph,
        "cell": int(cell),
    }


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(int(seed) & 0xFFFFFFFF)
    glyph = "A" if rng.integers(0, 2) else "B"
    return build_scene(noise=int(seed), glyph=glyph)


def _glyph_mask(glyph: str) -> np.ndarray:
    return glyph_masks()[glyph]


def render_matrix(scene: dict) -> np.ndarray:
    """(NGRID, NGRID) mean-luminance of each lattice cell."""
    cell = int(scene.get("cell", CELL))
    n = W // cell                      # cells per side
    half = n // 2
    inside = _glyph_mask(scene["glyph"])[:half, :half]
    sign = np.where(inside, 1.0, -1.0)  # mosaic sign of the even-even phase
    mat = np.zeros((n, n), dtype=np.float64)
    # 2x2 blocks: even-even cell carries BIAS*sign, the other three -BIAS/3*sign.
    mat[0::2, 0::2] = BASE + BIAS * sign
    mat[0::2, 1::2] = BASE - (BIAS / 3.0) * sign
    mat[1::2, 0::2] = BASE - (BIAS / 3.0) * sign
    mat[1::2, 1::2] = BASE - (BIAS / 3.0) * sign
    rng = np.random.default_rng(int(scene["noise"]) & 0xFFFFFFFF)
    mat = mat + rng.normal(0.0, NOISE_SIGMA, size=(n, n))
    return mat


def render(scene: dict) -> Image.Image:
    cell = int(scene.get("cell", CELL))
    n = W // cell
    mat = render_matrix(scene)
    arr = np.repeat(np.repeat(mat, cell, axis=0), cell, axis=1)
    arr = np.clip(arr, 0, 255)
    canvas = np.repeat(arr[:, :, None], 3, axis=2).astype(np.uint8)
    canvas = canvas[:W, :H]
    return Image.fromarray(canvas, "RGB")


def margin(scene: dict) -> float:
    chosen = _glyph_mask(scene["glyph"])
    other = _glyph_mask("B" if scene["glyph"] == "A" else "A")
    c_filled = int(chosen.sum())
    o_filled = int(other.sum())
    corr_other = float((chosen & other).sum()) / max(1, o_filled)
    return round(1.0 - corr_other, 6)


def analytic_gold(scene: dict) -> str:
    return scene["glyph"]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < Q_TOL


def latent_symmetries(scene: dict):
    """not_applicable: every latent field (the per-cell noise seed, the answer
    glyph, the cell pitch) directly changes the rendered pixels, so no distinct
    latent vector reproduces the same image. Declared empty."""
    return iter(())
