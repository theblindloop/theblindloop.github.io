"""Independent pixel-only inverse arm for quad_phase_decimate_reveal.

Receives ONLY the final PNG. It re-discovers the revealed glyph by:
  1. reading the mean luminance of every even-even phase cell (the top-left cell
     of each 2x2 block) of the lattice into an (M, M) mosaic;
  2. correlating that decimated signal against the two declared triangle
     templates A (upward) and B (downward);
  3. reporting the template with the larger signed inside-vs-outside contrast.

No scene data, no analytic gold, no renderer imports.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "quad_phase_decimate_reveal-oracle-0.1.0"

W = 512
NGRID = 32
CELL = W // NGRID
M = NGRID // 2


def _glyph_masks() -> dict[str, np.ndarray]:
    r, c = np.indices((M, M))
    center = (M - 1) / 2.0
    maxhw = float(round(M * 0.40))
    hw_up = np.round((r / (M - 1)) * maxhw)
    mask_a = (c >= center - hw_up) & (c <= center + hw_up)
    hw_down = np.round(((M - 1 - r) / (M - 1)) * maxhw)
    mask_b = (c >= center - hw_down) & (c <= center + hw_down)
    return {"A": mask_a, "B": mask_b}


def _decimated_signal(image: Image.Image) -> np.ndarray:
    """(M, M) mean-luminance of each even-even phase cell."""
    arr = np.asarray(image.convert("L"), dtype=np.float64)
    sig = np.zeros((M, M), dtype=np.float64)
    for i in range(M):
        for j in range(M):
            top = 2 * i * CELL
            left = 2 * j * CELL
            cell = arr[top:top + CELL, left:left + CELL]
            sig[i, j] = cell.mean()
    return sig


def _contrast(sig: np.ndarray, mask: np.ndarray) -> float:
    inside = sig[mask].mean()
    outside = sig[~mask].mean()
    return float(inside - outside)


def decision_from_image(image: Image.Image) -> str:
    sig = _decimated_signal(image)
    masks = _glyph_masks()
    sa = _contrast(sig, masks["A"])
    sb = _contrast(sig, masks["B"])
    return "A" if abs(sa) >= abs(sb) else "B"


def reveal_margin(image: Image.Image) -> float:
    sig = _decimated_signal(image)
    masks = _glyph_masks()
    sa = _contrast(sig, masks["A"])
    sb = _contrast(sig, masks["B"])
    return abs(abs(sa) - abs(sb))
