"""Independent inverse arm using only the final raster image."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result: list[np.ndarray] = []
    for sy, sx in np.argwhere(mask):
        if seen[sy, sx]:
            continue
        queue = deque([(int(sy), int(sx))])
        seen[sy, sx] = True
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= 40:
            result.append(np.asarray(points, dtype=np.float64))
    return result


def _principal_line(points: np.ndarray) -> tuple[np.ndarray, np.ndarray, float]:
    xy = points[:, ::-1] + 0.5
    center = xy.mean(axis=0)
    values, vectors = np.linalg.eigh(np.cov(xy.T))
    direction = vectors[:, int(np.argmax(values))]
    elongation = float(values.max() / max(values.min(), 1e-6))
    return center, direction, elongation


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    border = np.concatenate(
        (array[:4].reshape(-1, 3), array[-4:].reshape(-1, 3),
         array[:, :4].reshape(-1, 3), array[:, -4:].reshape(-1, 3)), axis=0
    )
    background = np.median(border, axis=0)
    # The inner card is near white. Saturated/dark colored marks are far from
    # both the border and neutral gray, while card edges are excluded.
    chroma = array.max(axis=2) - array.min(axis=2)
    darkness = 255.0 - array.mean(axis=2)
    foreground = (chroma > 38.0) & (darkness > 35.0)
    components = _components(foreground)
    if len(components) != 6:
        raise ValueError(f"expected six colored components, found {len(components)}")

    ring_candidates = []
    dash_candidates = []
    for points in components:
        ys, xs = points[:, 0], points[:, 1]
        bw, bh = xs.max() - xs.min() + 1.0, ys.max() - ys.min() + 1.0
        fill = len(points) / (bw * bh)
        _, _, elongation = _principal_line(points)
        if 0.78 <= bw / bh <= 1.28 and 0.18 <= fill <= 0.62 and min(bw, bh) >= 60:
            ring_candidates.append(points)
        elif elongation >= 12.0 and max(bw, bh) >= 30.0:
            dash_candidates.append(points)
    if len(ring_candidates) != 1 or len(dash_candidates) != 5:
        raise ValueError("could not uniquely recover one ring and five elongated dashes")

    ring_xy = ring_candidates[0][:, ::-1] + 0.5
    ring_center = ring_xy.mean(axis=0)
    radial = np.linalg.norm(ring_xy - ring_center, axis=1)
    inner_radius = float(np.percentile(radial, 4.0))
    if not 20.0 <= inner_radius <= 34.0:
        raise ValueError("recovered ring geometry is outside the declared visual regime")

    distances = []
    for points in dash_candidates:
        center, direction, _ = _principal_line(points)
        delta = ring_center - center
        distances.append(abs(direction[0] * delta[1] - direction[1] * delta[0]))
    worst = max(distances)
    if abs(worst - inner_radius) < 5.0:
        raise ValueError("raster is too close to the visible ring boundary")
    return "all" if worst < inner_radius else "not_all"
