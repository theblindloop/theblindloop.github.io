"""Deterministic scene sampler and renderer for target-ring concurrence."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw


PALETTES = (
    ("#176B87", "#C64B45"),
    ("#6046A8", "#D17B20"),
    ("#187A5B", "#B33A72"),
    ("#9A4E18", "#3066A5"),
)


def _line_distance(center: list[float], dash: list[list[float]]) -> float:
    (x1, y1), (x2, y2) = dash
    cx, cy = center
    dx, dy = x2 - x1, y2 - y1
    return abs(dx * (y1 - cy) - (x1 - cx) * dy) / math.hypot(dx, dy)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    width = height = 512
    cx = rng.uniform(242.0, 270.0)
    cy = rng.uniform(242.0, 270.0)
    inner_radius = rng.uniform(25.0, 29.0)
    outer_radius = inner_radius + rng.uniform(6.0, 8.0)
    ink, target = PALETTES[rng.randrange(len(PALETTES))]
    label = "all" if int(seed) % 2 == 0 else "not_all"
    global_angle = rng.choice((0.0, math.pi / 2.0, math.pi, 3.0 * math.pi / 2.0))
    global_angle += rng.uniform(-0.035, 0.035)
    # Two separated dash banks keep the decisive components disconnected even
    # under the public overlapping-grid ablations; the ring remains a third
    # isolated component, all within one medium-sized central region.
    bank_angles = (-0.28, 0.0, 0.28, math.pi - 0.18, math.pi + 0.18)
    failing = set(rng.sample(range(5), 2)) if label == "not_all" else set()
    dashes: list[list[list[float]]] = []
    for index in range(5):
        radial = global_angle + bank_angles[index] + rng.uniform(-0.025, 0.025)
        radius = rng.uniform(153.0, 160.0)
        mx = cx + radius * math.cos(radial)
        my = cy + radius * math.sin(radial)
        if index in failing:
            offset = rng.choice((-1.0, 1.0)) * rng.uniform(43.0, 51.0)
        else:
            offset = rng.uniform(-10.0, 10.0)
        # A line through the dash midpoint whose perpendicular distance from
        # the ring center is exactly |offset|.
        delta = math.asin(max(-0.8, min(0.8, offset / radius)))
        direction = radial + math.pi + delta
        half = rng.uniform(20.0, 23.0)
        vx, vy = half * math.cos(direction), half * math.sin(direction)
        dashes.append([[mx - vx, my - vy], [mx + vx, my + vy]])
    return {
        "width": width,
        "height": height,
        "background": "#F4F0E8",
        "card": "#FFFDFC",
        "ink": ink,
        "target": target,
        "ring_center": [cx, cy],
        "outer_radius": outer_radius,
        "inner_radius": inner_radius,
        "line_width": 8,
        "dashes": dashes,
    }


def render(scene: dict) -> Image.Image:
    scale = 3
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), scene["background"])
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle(
        (44 * scale, 44 * scale, (width - 44) * scale, (height - 44) * scale),
        radius=20 * scale,
        fill=scene["card"],
    )
    cx, cy = scene["ring_center"]
    outer = float(scene["outer_radius"])
    inner = float(scene["inner_radius"])
    draw.ellipse(
        ((cx - outer) * scale, (cy - outer) * scale, (cx + outer) * scale, (cy + outer) * scale),
        fill=scene["target"],
    )
    draw.ellipse(
        ((cx - inner) * scale, (cy - inner) * scale, (cx + inner) * scale, (cy + inner) * scale),
        fill=scene["card"],
    )
    for dash in scene["dashes"]:
        points = [(point[0] * scale, point[1] * scale) for point in dash]
        draw.line(points, fill=scene["ink"], width=int(scene["line_width"]) * scale)
        radius = int(scene["line_width"]) * scale / 2
        for x, y in points:
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=scene["ink"])
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    center = scene["ring_center"]
    inner = float(scene["inner_radius"])
    distances = [_line_distance(center, dash) for dash in scene["dashes"]]
    return "all" if max(distances) < inner else "not_all"


def margin(scene: dict) -> float:
    center = scene["ring_center"]
    inner = float(scene["inner_radius"])
    worst = max(_line_distance(center, dash) for dash in scene["dashes"])
    return abs(inner - worst)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
