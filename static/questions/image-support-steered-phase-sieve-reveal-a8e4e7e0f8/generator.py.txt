"""Renderer for the phase-sieve reveal world.

Nine different 7x7 symbols are interleaved cell by cell inside one bordered
21x21 panel: the symbol living on the sub-lattice with row remainder ``a`` and
column remainder ``b`` (modulo three) is ``symbol_grid[3 * a + b]``.  The panel
alone therefore shows all nine symbols superimposed as noise.  Two solid
triangles drawn far apart in opposite margins point at one of the first three
rows and one of the first three columns; that pair of pointers, and nothing
else in the picture, selects which interleaved sub-lattice is the answer.

Everything the analytic gold reads is rendered: the panel cells and the two
pointer positions.
"""

from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image

RENDERER_VERSION = "phase_sieve_reveal-renderer-0.1.0"

GRID = 21
BLOCK = 3
SUB = GRID // BLOCK  # 7
CELL_PX = 14
FRAME_PX = 2
PAD_PX = 5
PANEL_PX = GRID * CELL_PX + 2 * FRAME_PX + 2 * PAD_PX  # 308
CANVAS = 468
MARKER_PX = 13  # odd; triangles taper by two pixels per step
# The two pointers are kept near-opposite corners of the canvas, so that the
# decisive support spans the picture.
MIN_POINTER_SEPARATION = 410.0

# A gap of at most two cells between the revealed pattern and its closest
# competing symbol is not reliably readable by eye, so such scenes are
# quarantined; they remain exactly decidable cell by cell.
QUARANTINE_GAP = 3.0

SYMBOLS: tuple[str, ...] = tuple(sorted(SYMBOL_ART))


def _template(name: str) -> np.ndarray:
    art = SYMBOL_ART[name]
    return np.array([[1 if ch == "#" else 0 for ch in row] for row in art], dtype=np.int8)


TEMPLATES: dict[str, np.ndarray] = {name: _template(name) for name in SYMBOLS}


def _nearest(pattern: np.ndarray) -> tuple[str, float]:
    """Nearest symbol and the cell gap to the closest competing symbol."""

    scored = sorted(
        (int(np.count_nonzero(pattern != TEMPLATES[name])), name) for name in SYMBOLS
    )
    best_d, best_name = scored[0]
    second_d, _ = scored[1]
    return best_name, float(second_d - best_d)


def _closest_rival(name: str) -> tuple[str, int]:
    """The symbol nearest to ``name`` and its cell distance."""

    scored = sorted(
        (int(np.count_nonzero(TEMPLATES[name] != TEMPLATES[other])), other)
        for other in SYMBOLS
        if other != name
    )
    return scored[0][1], scored[0][0]


def _patterns(scene: dict) -> dict[tuple[int, int], np.ndarray]:
    """The realised 7x7 pattern of every interleaved sub-lattice."""

    out: dict[tuple[int, int], np.ndarray] = {}
    for a in range(BLOCK):
        for b in range(BLOCK):
            pattern = TEMPLATES[scene["symbol_grid"][BLOCK * a + b]].copy()
            for fa, fb, r, c in scene["flips"]:
                if fa == a and fb == b:
                    pattern[r, c] = 1 - pattern[r, c]
            out[(a, b)] = pattern
    return out


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 17)
    for _ in range(400):
        answer = SYMBOLS[seed % len(SYMBOLS)]
        others = [name for name in SYMBOLS if name != answer]
        rng.shuffle(others)
        row_line = rng.randrange(GRID)
        col_line = rng.randrange(GRID)
        row_phase = row_line % BLOCK
        col_phase = col_line % BLOCK
        grid: list[str] = []
        pool = iter(others)
        for a in range(BLOCK):
            for b in range(BLOCK):
                grid.append(answer if (a, b) == (row_phase, col_phase) else next(pool))
        flip_count = rng.choice([0, 1, 1, 2, 2, 3, 3, 4])
        flips: list[list[int]] = []
        for a in range(BLOCK):
            for b in range(BLOCK):
                name = grid[BLOCK * a + b]
                rival, distance = _closest_rival(name)
                budget = min(flip_count, max(0, (distance - 2) // 2))
                cells = [
                    (int(r), int(c))
                    for r, c in zip(*np.nonzero(TEMPLATES[name] != TEMPLATES[rival]))
                ]
                rng.shuffle(cells)
                for r, c in cells[:budget]:
                    flips.append([a, b, r, c])
        scene = {
            "canvas": CANVAS,
            "cell_px": CELL_PX,
            "grid": GRID,
            "block": BLOCK,
            "frame_px": FRAME_PX,
            "pad_px": PAD_PX,
            "panel_x": 80 + rng.randrange(-7, 8),
            "panel_y": 80 + rng.randrange(-7, 8),
            "row_line": row_line,
            "col_line": col_line,
            "symbol_grid": grid,
            "flip_count": flip_count,
            "flips": sorted(flips),
            "row_marker_side": rng.choice(["left", "right"]),
            "col_marker_side": rng.choice(["top", "bottom"]),
            "row_marker_offset": rng.randrange(18, 49),
            "col_marker_offset": rng.randrange(18, 49),
            "marker_px": MARKER_PX,
            "ink_value": rng.randrange(26, 53),
            "pale_value": rng.randrange(226, 245),
            "background_value": rng.randrange(132, 169),
            "tint": rng.choice([(0, 0, 0), (6, 2, 0), (0, 5, 3), (2, 0, 6), (4, 4, 0)]),
            "clutter": [],
        }
        rx0, ry0, rx1, ry1 = _marker_box(scene, "row")
        cx0m, cy0m, cx1m, cy1m = _marker_box(scene, "col")
        separation = math.dist(
            ((rx0 + rx1) / 2.0, (ry0 + ry1) / 2.0),
            ((cx0m + cx1m) / 2.0, (cy0m + cy1m) / 2.0),
        )
        if separation < MIN_POINTER_SEPARATION:
            continue
        scene["clutter"] = _sample_clutter(scene, rng)
        realised = _patterns(scene)
        if any(
            _nearest(pattern)[0] != scene["symbol_grid"][BLOCK * a + b]
            or _nearest(pattern)[1] < 2.0
            for (a, b), pattern in realised.items()
        ):
            continue
        return scene
    raise RuntimeError(f"could not sample a clean scene for seed {seed}")


def _panel_box(scene: dict) -> tuple[int, int, int, int]:
    """Outer box (x0, y0, x1, y1) of the pale panel block, inclusive."""

    x0 = scene["panel_x"]
    y0 = scene["panel_y"]
    return x0, y0, x0 + PANEL_PX - 1, y0 + PANEL_PX - 1


def _cells_origin(scene: dict) -> tuple[int, int]:
    x0, y0, _, _ = _panel_box(scene)
    inset = scene["pad_px"] + scene["frame_px"]
    return x0 + inset, y0 + inset


def _marker_box(scene: dict, axis: str) -> tuple[int, int, int, int]:
    """Bounding box (x0, y0, x1, y1) of one pointer triangle."""

    px0, py0, px1, py1 = _panel_box(scene)
    cx0, cy0 = _cells_origin(scene)
    size = scene["marker_px"]
    half = size // 2
    if axis == "row":
        centre = cy0 + scene["cell_px"] * scene["row_line"] + scene["cell_px"] // 2
        if scene["row_marker_side"] == "left":
            x1 = px0 - scene["row_marker_offset"]
            x0 = x1 - size + 1
        else:
            x0 = px1 + scene["row_marker_offset"]
            x1 = x0 + size - 1
        return x0, centre - half, x1, centre + half
    centre = cx0 + scene["cell_px"] * scene["col_line"] + scene["cell_px"] // 2
    if scene["col_marker_side"] == "top":
        y1 = py0 - scene["col_marker_offset"]
        y0 = y1 - size + 1
    else:
        y0 = py1 + scene["col_marker_offset"]
        y1 = y0 + size - 1
    return centre - half, y0, centre + half, y1


def _keepouts(scene: dict) -> list[tuple[int, int, int, int]]:
    boxes = [_panel_box(scene), _marker_box(scene, "row"), _marker_box(scene, "col")]
    return [(x0 - 7, y0 - 7, x1 + 7, y1 + 7) for x0, y0, x1, y1 in boxes]


def _sample_clutter(scene: dict, rng: random.Random) -> list[list[int]]:
    """Thin rectangular clutter bars; they never taper, so they are not pointers."""

    boxes = _keepouts(scene)
    placed: list[list[int]] = []
    attempts = 0
    wanted = rng.randrange(3, 7)
    while len(placed) < wanted and attempts < 400:
        attempts += 1
        length = rng.randrange(15, 31)
        thick = 4
        if rng.random() < 0.5:
            w, h = length, thick
        else:
            w, h = thick, length
        x = rng.randrange(6, scene["canvas"] - w - 6)
        y = rng.randrange(6, scene["canvas"] - h - 6)
        box = (x - 7, y - 7, x + w + 6, y + h + 6)
        if any(
            box[0] <= ox1 and ox0 <= box[2] and box[1] <= oy1 and oy0 <= box[3]
            for ox0, oy0, ox1, oy1 in boxes
        ):
            continue
        boxes.append(box)
        placed.append([x, y, w, h])
    return sorted(placed)


def _tone(value: int, tint: tuple[int, int, int]) -> tuple[int, int, int]:
    return tuple(int(min(255, max(0, value + channel))) for channel in tint)


def render(scene: dict) -> Image.Image:
    size = scene["canvas"]
    tint = tuple(scene["tint"])
    ink = _tone(scene["ink_value"], tint)
    pale = _tone(scene["pale_value"], tint)
    canvas = np.zeros((size, size, 3), dtype=np.uint8)
    canvas[:, :] = _tone(scene["background_value"], tint)

    px0, py0, px1, py1 = _panel_box(scene)
    canvas[py0 : py1 + 1, px0 : px1 + 1] = pale
    fx0 = px0 + scene["pad_px"]
    fy0 = py0 + scene["pad_px"]
    fx1 = px1 - scene["pad_px"]
    fy1 = py1 - scene["pad_px"]
    frame = scene["frame_px"]
    canvas[fy0 : fy0 + frame, fx0 : fx1 + 1] = ink
    canvas[fy1 - frame + 1 : fy1 + 1, fx0 : fx1 + 1] = ink
    canvas[fy0 : fy1 + 1, fx0 : fx0 + frame] = ink
    canvas[fy0 : fy1 + 1, fx1 - frame + 1 : fx1 + 1] = ink

    cx0, cy0 = _cells_origin(scene)
    cell = scene["cell_px"]
    patterns = _patterns(scene)
    for r in range(GRID):
        for c in range(GRID):
            pattern = patterns[(r % BLOCK, c % BLOCK)]
            if pattern[r // BLOCK, c // BLOCK]:
                y = cy0 + r * cell
                x = cx0 + c * cell
                canvas[y : y + cell, x : x + cell] = ink

    for axis in ("row", "col"):
        mx0, my0, mx1, my1 = _marker_box(scene, axis)
        size_m = scene["marker_px"]
        half = size_m // 2
        if axis == "row":
            pointing_right = scene["row_marker_side"] == "left"
            for step in range(size_m):
                extent = size_m - 2 * step
                if extent <= 0:
                    break
                x = mx0 + step if pointing_right else mx1 - step
                y = my0 + half - (extent - 1) // 2
                canvas[y : y + extent, x : x + 1] = ink
        else:
            pointing_down = scene["col_marker_side"] == "top"
            for step in range(size_m):
                extent = size_m - 2 * step
                if extent <= 0:
                    break
                y = my0 + step if pointing_down else my1 - step
                x = mx0 + half - (extent - 1) // 2
                canvas[y : y + 1, x : x + extent] = ink

    for x, y, w, h in scene["clutter"]:
        canvas[y : y + h, x : x + w] = ink

    return Image.fromarray(canvas, mode="RGB")


def _revealed(scene: dict) -> np.ndarray:
    return _patterns(scene)[(scene["row_line"] % BLOCK, scene["col_line"] % BLOCK)]


def analytic_gold(scene: dict) -> str:
    return _nearest(_revealed(scene))[0]


def margin(scene: dict) -> float:
    return _nearest(_revealed(scene))[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist: every sampled field is rendered."""

    return []
