"""Independent image-only inverse for Snapshot flip pair.

It segments only public raster colors. It does not know sampler coordinates or
receive latent data. A valid answer requires the filled lamp, its disconnected
hollow counterpart in the other snapshot, and the nearby identity glyph.
"""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

CYAN = np.array((18, 111, 151), dtype=np.int16)
GOLD = np.array((242, 166, 35), dtype=np.int16)
MAGENTA = np.array((154, 51, 112), dtype=np.int16)


def _mask(arr: np.ndarray, color: np.ndarray, tolerance: int = 8) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - color), axis=2) <= tolerance


def _components(mask: np.ndarray, minimum: int) -> list[tuple[int, int, int, int, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for sy, sx in np.argwhere(mask):
        if seen[sy, sx]:
            continue
        queue = deque([(int(sy), int(sx))])
        seen[sy, sx] = True
        points = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y-1, x), (y+1, x), (y, x-1), (y, x+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            ys = [p[0] for p in points]
            xs = [p[1] for p in points]
            output.append((min(xs), min(ys), max(xs), max(ys), len(points)))
    return output


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.ndim != 3 or arr.shape[0] < 128 or arr.shape[1] < 128:
        raise ValueError("raster is too small")

    rings = []
    for x0, y0, x1, y1, area in _components(_mask(arr, CYAN), 140):
        if 27 <= x1-x0+1 <= 36 and 27 <= y1-y0+1 <= 36 and area >= 250:
            rings.append(((x0+x1)/2.0, (y0+y1)/2.0))
    if len(rings) < 2:
        raise ValueError("too few intact lamp outlines")

    fills = _components(_mask(arr, GOLD), 250)
    if len(fills) != 1:
        raise ValueError("expected one intact filled interior")
    fx0, fy0, fx1, fy1, _ = fills[0]
    if not (19 <= fx1-fx0+1 <= 25 and 19 <= fy1-fy0+1 <= 25):
        raise ValueError("filled interior is malformed")
    fx, fy = (fx0+fx1)/2.0, (fy0+fy1)/2.0

    same_column = [(x, y) for x, y in rings if abs(x-fx) <= 3 and abs(y-fy) >= 120]
    if len(same_column) != 1:
        raise ValueError("filled lamp has no unique other-snapshot counterpart")

    glyphs = []
    for x0, y0, x1, y1, area in _components(_mask(arr, MAGENTA), 80):
        width, height = x1-x0+1, y1-y0+1
        if 11 <= width <= 14 and 9 <= height <= 28:
            label = "A" if height <= 13 else ("B" if height <= 22 else "C")
            glyphs.append(((x0+x1)/2.0, label))
    # The public identity glyph is immediately left of its lamp. Looking for the
    # relative pairing (not an absolute coordinate) rejects a neighboring label.
    matches = [label for gx, label in glyphs if 25 <= fx-gx <= 43]
    if len(matches) != 1:
        raise ValueError("changed lamp identity glyph is missing or ambiguous")
    return matches[0]
