"""Deterministic latent sampler and raster renderer."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

W = H = 768
BG = (248, 248, 244)
INK = (75, 79, 88)
FRAME = (183, 187, 187)
CYAN = (18, 111, 151)
GOLD = (242, 166, 35)
MAGENTA = (154, 51, 112)
ACCENTS = ((225, 232, 230), (232, 226, 238), (238, 231, 218))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    labels = ["A", "B", "C"]
    rng.shuffle(labels)
    changed = rng.randrange(3)
    direction = rng.randrange(2)
    before = [0, 0, 0]
    before[changed] = direction
    # Grid-compatible translations keep the local evidence away from borders while
    # moving its absolute position substantially across scenes.
    return {
        "origin_x": rng.choice((210, 306, 402)),
        "origin_y": rng.choice((204, 300, 396)),
        "label_order": labels,
        "before_states": before,
        "changed_column": changed,
        "accent_index": rng.randrange(len(ACCENTS)),
    }


def _lamp(draw: ImageDraw.ImageDraw, cx: int, cy: int, filled: bool) -> None:
    r = 16
    draw.rectangle((cx-r, cy-r, cx+r, cy+r), fill=CYAN)
    draw.rectangle((cx-r+5, cy-r+5, cx+r-5, cy+r-5), fill=GOLD if filled else BG)


def _identity(draw: ImageDraw.ImageDraw, cx: int, cy: int, label: str) -> None:
    # Letter identities use three visibly distinct compact glyph heights. Their
    # permutation is independent of column and changed state.
    height = {"A": 10, "B": 18, "C": 26}[label]
    draw.rectangle((cx-6, cy-height//2, cx+6, cy+(height-1)//2), fill=MAGENTA)


def render(scene: dict) -> Image.Image:
    ox = int(scene["origin_x"])
    oy = int(scene["origin_y"])
    labels = list(scene["label_order"])
    before = [int(v) for v in scene["before_states"]]
    changed = int(scene["changed_column"])
    accent = ACCENTS[int(scene["accent_index"])]
    after = list(before)
    after[changed] = 1 - after[changed]

    image = Image.new("RGB", (W, H), BG)
    draw = ImageDraw.Draw(image)
    # Pale frame and headings make temporal order explicit while staying outside
    # the oracle palette and therefore nondiagnostic.
    draw.rounded_rectangle((ox-54, oy-48, ox+174, oy+256), radius=18, fill=accent, outline=FRAME, width=3)
    draw.rectangle((ox-34, oy-28, ox+154, oy-15), fill=FRAME)
    draw.rectangle((ox-34, oy+198, ox+154, oy+211), fill=FRAME)
    # A slim gray direction arrow between snapshots.
    draw.rectangle((ox+57, oy+92, ox+63, oy+144), fill=INK)
    draw.polygon(((ox+48, oy+138), (ox+72, oy+138), (ox+60, oy+154)), fill=INK)

    xs = [ox, ox+60, ox+120]
    top_y, bottom_y = oy+36, oy+228
    for idx, cx in enumerate(xs):
        _identity(draw, cx-34, top_y, labels[idx])
        _lamp(draw, cx, top_y, bool(before[idx]))
        _lamp(draw, cx, bottom_y, bool(after[idx]))
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["label_order"][int(scene["changed_column"])])


def margin(scene: dict) -> float:
    # Minimum pure-color thickness in the glyphs/rings, in pixels.
    _ = scene["label_order"], scene["changed_column"]
    return 5.0


def is_quarantined(scene: dict) -> bool:
    _ = (scene["origin_x"], scene["origin_y"], scene["label_order"], scene["before_states"], scene["changed_column"], scene["accent_index"])
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    _ = scene
    return []
