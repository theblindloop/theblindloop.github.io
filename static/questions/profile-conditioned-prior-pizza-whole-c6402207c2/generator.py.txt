"""Whole-pizza completeness world under a contradictory completeness prior.

Profile: paper-semantic@0.3.0:prior_conflict. Report visible truth (whether a
pizza is presented as a COMPLETE circle or has a MISSING slice) when a familiar
canonical prior ("a whole pizza is an unbroken, complete round disk, no slice
missing") is contradicted by the image, which in the counterfactual scenes shows
a wedge-shaped slice cut out of the pizza even though it is presented as a whole.

Latent z = {whole, gap_deg, radius, cx, cy, hue, a0_deg}. ``whole`` is a boolean:
True means the pizza is drawn as a complete circular disk (matches the prior, the
control variant); False means one wedge of angle ``gap_deg`` is removed from the
disk (counterfactual variant that contradicts the prior). ``radius`` sets how
large the pizza is, ``cx``/``cy`` its position, ``hue`` selects one of a fixed
palette, and ``a0_deg`` the angular start of the removed wedge (rotation
counterbalancing; irrelevant for a complete pizza). The closed-form gold is

    decision = "complete" if whole else "missing"

a scene parameter, not a pixel measurement.

The pizza is drawn as a round disk with a lighter crust ring and a saturated
cheese interior on a plain white background. When ``not whole`` the wedge
[a0_deg, a0_deg + gap_deg] is cut from both crust and cheese, exposing the
background so the missing slice is a clearly visible wedge gap.

The continuous verifiability margin is the angular gap between the present (or
removed) wedge angle and the decision threshold. When a counterfactual gap is too
close to the decision threshold the scene is quarantined because the gap becomes
pixel-ambiguous relative to the threshold.
"""

from __future__ import annotations

import random

import numpy as np
from PIL import Image

RENDERER_VERSION = "prior_pizza_whole-0.1.0"

BG = (255, 255, 255)
CRUST = (228, 178, 96)
PALETTE = [
    (206, 54, 54),     # tomato red cheese
    (46, 158, 78),     # green
    (46, 118, 200),    # blue
    (236, 190, 40),    # yellow
]

CANVAS = (512, 512)

DECISION_GAP_DEG = 25.0     # largest empty angular run above which a slice is "missing"
QUAR_MARGIN_DEG = 8.0
RADIUS_MIN, RADIUS_MAX = 120.0, 162.0
CRUST_WIDTH = 12.0
GAP_MIN, GAP_MAX = 40.0, 72.0


def margin(scene: dict) -> float:
    """Angular gap (deg) between the present wedge angle and the decision threshold."""
    if scene["whole"]:
        return DECISION_GAP_DEG
    return abs(float(scene["gap_deg"]) - DECISION_GAP_DEG)


def is_quarantined(scene: dict) -> bool:
    radius = float(scene["radius"])
    cx, cy = float(scene["cx"]), float(scene["cy"])
    if not (RADIUS_MIN <= radius <= RADIUS_MAX * 1.05):
        return True
    if cx - radius < 8 or cx + radius > CANVAS[0] - 8:
        return True
    if cy - radius < 8 or cy + radius > CANVAS[1] - 8:
        return True
    return margin(scene) < QUAR_MARGIN_DEG


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    while True:
        whole = rng.random() < 0.5
        radius = rng.uniform(RADIUS_MIN, RADIUS_MAX)
        cx = rng.uniform(200, 312)
        cy = rng.uniform(200, 312)
        hue = rng.randrange(len(PALETTE))
        if whole:
            gap_deg, a0_deg = 0.0, 0.0
        else:
            gap_deg = rng.uniform(GAP_MIN, GAP_MAX)
            a0_deg = rng.uniform(0.0, 360.0)
        scene = {
            "whole": whole,
            "gap_deg": gap_deg,
            "radius": radius,
            "cx": cx,
            "cy": cy,
            "hue": hue,
            "a0_deg": a0_deg,
        }
        if not is_quarantined(scene):
            return scene


def render(scene: dict) -> Image.Image:
    w, h = CANVAS
    img = np.full((h, w, 3), np.array(BG), dtype=np.uint8)
    cx, cy = float(scene["cx"]), float(scene["cy"])
    radius = float(scene["radius"])
    inner_r = radius - CRUST_WIDTH

    yy, xx = np.mgrid[0:h, 0:w].astype(np.float64)
    dy = yy - cy
    dx = xx - cx
    dist = np.sqrt(dx * dx + dy * dy)
    ang = (np.degrees(np.arctan2(dy, dx)) + 360.0) % 360.0

    inside = dist <= radius
    cheese = inside & (dist <= inner_r)
    crust = inside & (dist > inner_r)

    if not scene["whole"]:
        delta = (ang - float(scene["a0_deg"])) % 360.0
        removed = delta <= float(scene["gap_deg"])
        cheese = cheese & ~removed
        crust = crust & ~removed

    cheese_rgb = np.array(PALETTE[int(scene["hue"])], dtype=np.uint8)
    img[cheese] = cheese_rgb
    img[crust] = np.array(CRUST, dtype=np.uint8)
    return Image.fromarray(img, "RGB")


def analytic_gold(scene: dict) -> str:
    return "complete" if scene["whole"] else "missing"


def latent_symmetries(scene: dict) -> list:
    """not_applicable: no render-preserving latent alias exists.

    The pizza region's pixels are determined by ``whole`` (a full disk) and, in
    the counterfactual, by ``gap_deg`` and ``a0_deg`` (the removed wedge cuts a
    background-shaped arc out of the disk), plus ``radius``/``cx``/``cy``/``hue``.
    Changing the label input ``whole`` swaps a complete disk for one with a wedge
    cut out, changing the raster; the wedge angle and rotation, radius, position,
    and hue also change the raster. The raster is therefore injective in the
    rendered latent fields, so no label-identical render-preserving transform
    exists. We declare no latent aliases and return an empty list.
    """
    return []
