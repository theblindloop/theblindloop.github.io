"""Independent pixel-only inverse arm for the whole-pizza completeness world.

Receives ONLY the final PNG and returns whether the pizza is a COMPLETE circle
("complete") or has a MISSING slice ("missing"). It shares no geometry with
renderer.render or analytic_gold; it detects the coloured pizza region directly
from the raster and measures its angular coverage.

Method: white-background subtraction isolates all non-white pixels (both the
crust ring and the cheese interior form the pizza). The pizza centroid is the
mean of those pixels and its radius is the median pixel distance from the
centroid. A mid-radius band that lies inside the cheese interior is profiled by
angle; wherever a slice is missing, that band is empty over a contiguous angular
run equal to the gap width. The pizza is "missing" exactly when the largest
contiguous empty angular run in that band exceeds the decision threshold.

Import discipline: this module imports only PIL and numpy. It never imports
renderer, prompts, generate, verify, or any analytic-gold code.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "prior_pizza_whole-oracle-0.1.0"

_WHITE = np.array((255, 255, 255), dtype=np.float32)
_DECISION_GAP_DEG = 25.0
_MIN_PIXELS = 400
_BAND_LO, _BAND_HI = 0.42, 0.80
_BINS = 360


def _largest_empty_run(occupied: np.ndarray) -> int:
    n = len(occupied)
    doubl = np.concatenate([occupied, occupied])
    best = 0
    run = 0
    for value in doubl:
        if value == 0:
            run += 1
            best = max(best, run)
        else:
            run = 0
    return min(best, n)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    nonwhite = np.linalg.norm(arr - _WHITE, axis=-1) > 40.0
    ys, xs = np.nonzero(nonwhite)
    if len(ys) < _MIN_PIXELS:
        return "complete"
    cy = float(np.mean(ys))
    cx = float(np.mean(xs))
    dists = np.sqrt((xs - cx) ** 2 + (ys - cy) ** 2)
    radius = float(np.median(dists))
    band = (dists >= _BAND_LO * radius) & (dists <= _BAND_HI * radius)
    if band.sum() == 0:
        return "complete"
    angles = (np.degrees(np.arctan2(ys[band] - cy, xs[band] - cx)) + 360.0) % 360.0
    occupied = np.zeros(_BINS, dtype=np.int8)
    bins = np.clip(angles.astype(np.int64) % _BINS, 0, _BINS - 1)
    occupied[bins] = 1
    largest = _largest_empty_run(occupied)
    gap_deg = largest * (360.0 / _BINS)
    return "missing" if gap_deg > _DECISION_GAP_DEG else "complete"


def measure(image: Image.Image) -> dict:
    """Diagnostic measurement (not used for the decision)."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    nonwhite = np.linalg.norm(arr - _WHITE, axis=-1) > 40.0
    ys, xs = np.nonzero(nonwhite)
    if len(ys) == 0:
        return {"pixel_count": 0}
    cy, cx = float(np.mean(ys)), float(np.mean(xs))
    dists = np.sqrt((xs - cx) ** 2 + (ys - cy) ** 2)
    radius = float(np.median(dists))
    return {
        "pixel_count": int(nonwhite.sum()),
        "centroid": (round(cx, 1), round(cy, 1)),
        "radius_est": round(radius, 1),
        "size": image.size,
    }
