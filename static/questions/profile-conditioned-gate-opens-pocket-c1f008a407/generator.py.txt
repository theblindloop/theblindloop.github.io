"""Gate-opening (region-connectivity) world renderer.

A single thick black ring loop is drawn on a white canvas, sealing a white
pocket at its center that holds a small green dot. Two colored gate markers,
one red and one blue, sit on the circular path of the ring. At exactly one of
the two gates the ring band is actually broken (an open gap in the loop), so
that the sealed central pocket meets the outside white through that break; at
the other gate the ring is a continuous closed band. Removing (erasing) the
gate that sits at the break would open the pocket.

Decision: which colored gate (red or blue) is located at the real opening in
the ring.

The scene is a plain dict so the verifier's AST reachability audit is free of
side effects.

    z = {
        "width": 400,
        "height": 400,
        "background": "white",
        "center": [cx, cy],
        "radius": R,
        "ring_color": "black",
        "ring_width": 30.0,
        "green": {"color": "green", "x": cx, "y": cy},
        "green_radius": 7.0,
        "gates": [{"color": "red", "theta": ..}, {"color": "blue", "theta": ..}],
        "open_color": "red",            # the gate located at the ring's break
        "gap_half": 0.45,
        "gate_radius": 6.0,
    }
"""

from __future__ import annotations

import math
import random
from collections.abc import Iterator

from PIL import Image, ImageDraw

RENDERER_VERSION = "gate_opens_pocket-0.4.0"

GATE_COLORS = {
    "red": (255, 0, 0),
    "blue": (0, 0, 255),
}
GREEN_COLOR = (0, 128, 0)

WIDTH = 400
HEIGHT = 400
CENTER = (200.0, 200.0)
RING_WIDTH = 30.0
GREEN_RADIUS = 7.0
GATE_RADIUS = 6.0
GAP_HALF = 0.45             # half-angular-width of the ring's open gap (radians)
MIN_GATE_SEP = 1.25        # minimum angular separation of the decoy from gap center
QUARANTINE_MIN = 6.0        # below this the opening decision is fragile


def sample_scene(rng: random.Random, width: int = WIDTH, height: int = HEIGHT) -> dict:
    cx, cy = float(CENTER[0]), float(CENTER[1])
    for _attempt in range(240):
        radius = rng.uniform(95.0, 125.0)
        gap_center = rng.uniform(0.0, 2.0 * math.pi)
        open_theta = gap_center
        decoy_delta = rng.uniform(MIN_GATE_SEP, math.pi - 0.2)
        decoy_theta = gap_center + decoy_delta
        open_color = rng.choice(["red", "blue"])
        decoy_color = "blue" if open_color == "red" else "red"

        gates = [
            {"color": open_color, "theta": open_theta},
            {"color": decoy_color, "theta": decoy_theta},
        ]
        scene = {
            "width": width,
            "height": height,
            "background": "white",
            "center": [cx, cy],
            "radius": radius,
            "ring_color": "black",
            "ring_width": RING_WIDTH,
            "green": {"color": "green", "x": cx, "y": cy},
            "green_radius": GREEN_RADIUS,
            "gates": gates,
            "open_color": open_color,
            "gap_half": GAP_HALF,
            "gate_radius": GATE_RADIUS,
        }
        if not is_quarantined(scene):
            return scene
    return _fallback_scene()


def _fallback_scene() -> dict:
    cx, cy = float(CENTER[0]), float(CENTER[1])
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": "white",
        "center": [cx, cy],
        "radius": 110.0,
        "ring_color": "black",
        "ring_width": RING_WIDTH,
        "green": {"color": "green", "x": cx, "y": cy},
        "green_radius": GREEN_RADIUS,
        "gates": [
            {"color": "red", "theta": 3.0},
            {"color": "blue", "theta": 6.2},
        ],
        "open_color": "red",
        "gap_half": GAP_HALF,
        "gate_radius": GATE_RADIUS,
    }


def _point(cx: float, cy: float, radius: float, theta: float) -> tuple[float, float]:
    return (cx + radius * math.cos(theta), cy - radius * math.sin(theta))


def _draw_ring_arc(
    draw: ImageDraw.ImageDraw,
    cx: float,
    cy: float,
    radius: float,
    width_px: int,
    theta_lo: float,
    theta_hi: float,
) -> None:
    step = 0.02
    theta = theta_lo
    prev = _point(cx, cy, radius, theta)
    while theta < theta_hi:
        theta = min(theta + step, theta_hi)
        cur = _point(cx, cy, radius, theta)
        draw.line([prev, cur], fill="black", width=width_px)
        prev = cur


def render(scene: dict, supersample: int = 1) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    background = scene["background"]
    ss = supersample
    img = Image.new("RGB", (width * ss, height * ss), background)
    draw = ImageDraw.Draw(img)

    cx, cy = (float(v) * ss for v in scene["center"])
    radius = float(scene["radius"]) * ss
    ring_w = max(1, int(round(float(scene["ring_width"]) * ss)))
    gap_half = float(scene["gap_half"])
    _opening = scene["open_color"]

    open_theta = 0.0
    for gate in scene["gates"]:
        if gate["color"] == scene["open_color"]:
            open_theta = float(gate["theta"])
    theta_lo = open_theta + gap_half
    theta_hi = open_theta + 2.0 * math.pi - gap_half
    _draw_ring_arc(draw, cx, cy, radius, ring_w, theta_lo, theta_hi)

    green_r = float(scene["green_radius"]) * ss
    gx, gy = (float(v) * ss for v in (scene["green"]["x"], scene["green"]["y"]))
    draw.ellipse(
        [gx - green_r, gy - green_r, gx + green_r, gy + green_r],
        fill=GREEN_COLOR,
    )

    gate_r = float(scene["gate_radius"]) * ss
    for gate in scene["gates"]:
        color = gate["color"]
        px, py = _point(cx, cy, radius, float(gate["theta"]))
        draw.ellipse(
            [px - gate_r, py - gate_r, px + gate_r, py + gate_r],
            fill=GATE_COLORS[color],
        )

    if ss > 1:
        img = img.resize((width, height), Image.LANCZOS)
    return img


def _gate_clearance(scene: dict, gate: dict) -> float:
    """Min distance from a gate's centroid to the nearest ring ink minus the
    gate radius, i.e. how far the nearest black ring band is from the gate's
    own ink. An open-gate sits far from any ring ink; a decoy sits on it."""
    cx, cy = (float(v) for v in scene["center"])
    radius = float(scene["radius"])
    theta = float(gate["theta"])
    d = 0.0
    # Ring band only exists outside the gap: nearest ink is at the gap edge.
    gap_half = float(scene["gap_half"])
    if gate["color"] == scene["open_color"]:
        arc = radius * gap_half
        radial = float(scene["ring_width"]) / 2.0
        d = math.hypot(arc, radial)
    else:
        d = float(scene["ring_width"]) / 2.0
    gate_r = float(scene.get("gate_radius", GATE_RADIUS))
    return max(0.0, d - gate_r)


def margin(scene: dict) -> float:
    """Robustness margin: min clearance protecting the open/decoy reading."""
    cx, cy = (float(v) for v in scene["center"])
    green = scene["green"]
    green_dist = math.hypot(float(green["x"]) - cx, float(green["y"]) - cy)
    green_clear = (
        float(scene["radius"]) - float(scene["ring_width"]) / 2.0 - green_dist
    ) - float(scene.get("green_radius", GREEN_RADIUS))
    gate_clear = min(_gate_clearance(scene, gate) for gate in scene["gates"])
    margin = min(gate_clear, green_clear)
    return max(0.0, float(margin))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MIN


def analytic_gold(scene: dict) -> str:
    """Return the color of the gate located at the ring's opening."""
    return str(scene["open_color"])


def latent_symmetries(scene: dict) -> Iterator[dict]:
    return iter(())
