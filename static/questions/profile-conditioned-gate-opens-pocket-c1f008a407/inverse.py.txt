"""Independent pixel-only inverse arm for the gate_opens_pocket world.

Receives ONLY the final PNG. It never imports or reads the renderer, the
analytic scene, prompts, or any renderer state. It recovers the decision by:

1. marking the red and blue gate colors and taking each gate's centroid;
2. taking the centroid of all black ring ink as the ring center;
3. computing, for each gate centroid, the minimum distance to any black ring
   pixel -- a gate sitting on the continuous ring has ink essentially at its
   center, while the gate at the ring's opening is surrounded by white and its
   nearest ring ink lies across the break, far away;
4. returning the color of the gate whose nearest-ring-ink distance is large
   (the gate located at the ring's opening).
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "gate_opens_pocket-oracle-0.4.0"

_GATE_RGB = {
    "red": (255, 0, 0),
    "blue": (0, 0, 255),
}
_RING_MAX = 90
_OPEN_THRESHOLD = 15.0


def _centroid(mask: np.ndarray) -> tuple[float, float] | None:
    ys, xs = np.nonzero(mask)
    if len(xs) == 0:
        return None
    return (float(xs.mean()), float(ys.mean()))


def _min_dist(x: float, y: float, ys: np.ndarray, xs: np.ndarray) -> float:
    if len(xs) == 0:
        return float("inf")
    d2 = (xs.astype(np.float64) - x) ** 2 + (ys.astype(np.float64) - y) ** 2
    return float(np.sqrt(d2.min()))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.int16)

    gates: dict[str, tuple[float, float]] = {}
    for name, (r, g, b) in _GATE_RGB.items():
        mask = (arr[..., 0] == r) & (arr[..., 1] == g) & (arr[..., 2] == b)
        c = _centroid(mask)
        if c is not None:
            gates[name] = c

    ring = (arr[..., 0] <= _RING_MAX) & (arr[..., 1] <= _RING_MAX) & (arr[..., 2] <= _RING_MAX)
    ys, xs = np.nonzero(ring)
    if len(xs) == 0 or set(gates) != set(_GATE_RGB):
        return "red"

    dist: dict[str, float] = {}
    for name, (gx, gy) in gates.items():
        dist[name] = _min_dist(gx, gy, ys, xs)

    red_open = dist["red"] > _OPEN_THRESHOLD
    blue_open = dist["blue"] > _OPEN_THRESHOLD
    if red_open and not blue_open:
        return "red"
    return "blue" if blue_open else "red"
