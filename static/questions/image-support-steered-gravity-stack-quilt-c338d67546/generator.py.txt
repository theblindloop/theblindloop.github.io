from __future__ import annotations

import copy
import itertools
import random
from PIL import Image, ImageDraw

SIZE = 768
BOARD0 = 48
CELL = 56
N = 5
COLORS = {
    "white": (250, 250, 247),
    "navy": (28, 45, 66),
    "cream": (239, 224, 183),
    "cyan": (38, 190, 202),
    "gold": (236, 178, 44),
    "red": (214, 68, 68),
    "green": (52, 157, 91),
    "blue": (65, 109, 205),
}
ANSWER_COLORS = ("red", "green", "blue")


def _settle(source: list[list[int]]) -> list[list[int]]:
    counts = [sum(source[r][c] for r in range(N)) for c in range(N)]
    return [[int(r >= N - counts[c]) for c in range(N)] for r in range(N)]


def _stack(counts: tuple[int, ...] | list[int]) -> list[list[int]]:
    return [[int(r >= N - counts[c]) for c in range(N)] for r in range(N)]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    while True:
        counts = tuple(rng.randint(1, 4) for _ in range(N))
        if len(set(counts)) >= 3:
            break
    source = [[0] * N for _ in range(N)]
    for c, count in enumerate(counts):
        for r in rng.sample(range(N), count):
            source[r][c] = 1
    # All choices are valid bottom-packed stacks with identical total ink.
    permutations = sorted(set(itertools.permutations(counts)))
    wrong = [p for p in permutations if p != counts]
    rng.shuffle(wrong)
    answer = ANSWER_COLORS[int(seed) % len(ANSWER_COLORS)]
    candidates: dict[str, list[list[int]]] = {answer: _stack(counts)}
    for color, vector in zip((c for c in ANSWER_COLORS if c != answer), wrong[:2]):
        candidates[color] = _stack(vector)
    tags = ["gold", *ANSWER_COLORS]
    rng.shuffle(tags)
    return {"source": source, "candidates": candidates, "tags": tags}


def _board_for(scene: dict, tag: str) -> list[list[int]]:
    if tag == "gold":
        return scene["source"]
    return scene["candidates"][tag]


def render(scene: dict) -> Image.Image:
    source = scene["source"]
    candidates = scene["candidates"]
    tags = scene["tags"]
    if len(source) != N or set(candidates) != set(ANSWER_COLORS) or sorted(tags) != sorted(("gold", *ANSWER_COLORS)):
        raise ValueError("invalid scene")
    image = Image.new("RGB", (SIZE, SIZE), COLORS["white"])
    draw = ImageDraw.Draw(image)
    # One thick, continuous 12 x 12 quilt: four 6 x 6 quadrants.
    draw.rectangle((BOARD0 - 7, BOARD0 - 7, BOARD0 + 12 * CELL + 6, BOARD0 + 12 * CELL + 6), fill=COLORS["navy"])
    for q, tag in enumerate(tags):
        qr, qc = divmod(q, 2)
        base_r, base_c = qr * 6, qc * 6
        board = _board_for(scene, tag)
        for lr in range(6):
            for lc in range(6):
                r, c = base_r + lr, base_c + lc
                x0, y0 = BOARD0 + c * CELL, BOARD0 + r * CELL
                if lr == 0:
                    fill = COLORS[tag]
                elif lc == 0:
                    fill = COLORS["cream"]
                else:
                    fill = COLORS["cyan"] if board[lr - 1][lc - 1] else COLORS["cream"]
                draw.rectangle((x0 + 3, y0 + 3, x0 + CELL - 4, y0 + CELL - 4), fill=fill)
    # Grid strokes join every cell and make the evidence one connected area.
    for k in range(13):
        p = BOARD0 + k * CELL
        draw.line((BOARD0, p, BOARD0 + 12 * CELL, p), fill=COLORS["navy"], width=5)
        draw.line((p, BOARD0, p, BOARD0 + 12 * CELL), fill=COLORS["navy"], width=5)
    return image


def analytic_gold(scene: dict) -> str:
    source = scene["source"]
    candidates = scene["candidates"]
    tags = scene["tags"]
    target = _settle(source)
    matches = [tag for tag in tags if tag != "gold" and candidates[tag] == target]
    if len(matches) != 1:
        raise ValueError("non-unique match")
    return matches[0]


def margin(scene: dict) -> float:
    target = _settle(scene["source"])
    distances = []
    for color, board in scene["candidates"].items():
        d = sum(a != b for ra, rb in zip(target, board) for a, b in zip(ra, rb))
        if d:
            distances.append(d)
    return float(min(distances))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # JSON object insertion order is a representational alias, not a visual field.
    twin = copy.deepcopy(scene)
    twin["candidates"] = {k: twin["candidates"][k] for k in reversed(list(twin["candidates"]))}
    return [("reverse_candidate_map_insertion_order", twin)]
