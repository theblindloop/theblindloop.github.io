from __future__ import annotations

import numpy as np
from PIL import Image

WHITE = np.array((250, 250, 247), dtype=np.float32)
NAVY = np.array((28, 45, 66), dtype=np.float32)
CREAM = np.array((239, 224, 183), dtype=np.float32)
CYAN = np.array((38, 190, 202), dtype=np.float32)
TAG_RGB = {
    "gold": np.array((236, 178, 44), dtype=np.float32),
    "red": np.array((214, 68, 68), dtype=np.float32),
    "green": np.array((52, 157, 91), dtype=np.float32),
    "blue": np.array((65, 109, 205), dtype=np.float32),
}


def _nearest(pixel: np.ndarray, palette: dict[str, np.ndarray]) -> str:
    return min(palette, key=lambda key: float(np.linalg.norm(pixel - palette[key])))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    delta = np.linalg.norm(arr - WHITE, axis=2)
    ys, xs = np.nonzero(delta > 55.0)
    if len(xs) < 10000:
        raise ValueError("continuous quilt is missing")
    x0, x1 = int(xs.min()), int(xs.max()) + 1
    y0, y1 = int(ys.min()), int(ys.max()) + 1
    if not (0.90 <= (x1 - x0) / (y1 - y0) <= 1.10):
        raise ValueError("quilt is not square")

    def patch(r: int, c: int) -> np.ndarray:
        cx = x0 + (c + 0.5) * (x1 - x0) / 12.0
        cy = y0 + (r + 0.5) * (y1 - y0) / 12.0
        radius = max(4, int(min(x1 - x0, y1 - y0) / 12.0 * 0.23))
        return arr[int(cy) - radius:int(cy) + radius + 1, int(cx) - radius:int(cx) + radius + 1]

    # Every cell is semantic evidence: cream explicitly means OFF. Missing paint
    # is not silently interpreted as OFF, which also makes destructive erasure abstain.
    means: list[list[np.ndarray]] = []
    for r in range(12):
        row = []
        for c in range(12):
            p = patch(r, c)
            if p.size == 0 or float(np.mean(np.linalg.norm(p - WHITE, axis=2) < 35.0)) > 0.08:
                raise ValueError("a quilt cell was erased")
            row.append(p.mean(axis=(0, 1)))
        means.append(row)

    tags: list[str] = []
    boards: dict[str, list[list[int]]] = {}
    for q in range(4):
        qr, qc = divmod(q, 2)
        br, bc = qr * 6, qc * 6
        header = np.mean(np.stack(means[br][bc:bc + 6]), axis=0)
        tag = _nearest(header, TAG_RGB)
        if tag in tags or float(np.linalg.norm(header - TAG_RGB[tag])) > 45.0:
            raise ValueError("ambiguous chamber header")
        tags.append(tag)
        board = []
        for rr in range(1, 6):
            out = []
            for cc in range(1, 6):
                value = _nearest(means[br + rr][bc + cc], {"off": CREAM, "on": CYAN})
                out.append(1 if value == "on" else 0)
            board.append(out)
        boards[tag] = board
    if set(tags) != {"gold", "red", "green", "blue"}:
        raise ValueError("headers are incomplete")
    source = boards["gold"]
    counts = [sum(source[r][c] for r in range(5)) for c in range(5)]
    target = [[int(r >= 5 - counts[c]) for c in range(5)] for r in range(5)]
    matches = [color for color in ("red", "green", "blue") if boards[color] == target]
    if len(matches) != 1:
        raise ValueError("no unique settled match")
    return matches[0]
