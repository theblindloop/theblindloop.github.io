"""Independent image-only oracle for connected route pocket topology.

This module deliberately contains no renderer, scene, prompt, or generator
imports.  It segments the colored route, checks that it is one connected
foreground component, flood-fills the exterior white background, and counts
large enclosed white components.  Damaged routes are rejected rather than
guessed.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _background_color(array: np.ndarray) -> np.ndarray:
    border = np.concatenate(
        (
            array[:3].reshape(-1, 3),
            array[-3:].reshape(-1, 3),
            array[:, :3].reshape(-1, 3),
            array[:, -3:].reshape(-1, 3),
        ),
        axis=0,
    )
    return np.median(border, axis=0).astype(np.float32)


def _component_sizes(mask: np.ndarray) -> list[int]:
    remaining = mask.copy()
    height, width = remaining.shape
    sizes: list[int] = []
    for y, x in zip(*np.nonzero(remaining)):
        if not remaining[y, x]:
            continue
        remaining[y, x] = False
        queue: deque[tuple[int, int]] = deque([(int(y), int(x))])
        size = 0
        while queue:
            cy, cx = queue.popleft()
            size += 1
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and remaining[ny, nx]:
                    remaining[ny, nx] = False
                    queue.append((ny, nx))
        sizes.append(size)
    return sizes


def _exterior_background(background: np.ndarray) -> np.ndarray:
    height, width = background.shape
    exterior = np.zeros_like(background, dtype=bool)
    queue: deque[tuple[int, int]] = deque()
    for x in range(width):
        if background[0, x] and not exterior[0, x]:
            exterior[0, x] = True
            queue.append((0, x))
        if background[height - 1, x] and not exterior[height - 1, x]:
            exterior[height - 1, x] = True
            queue.append((height - 1, x))
    for y in range(1, height - 1):
        if background[y, 0] and not exterior[y, 0]:
            exterior[y, 0] = True
            queue.append((y, 0))
        if background[y, width - 1] and not exterior[y, width - 1]:
            exterior[y, width - 1] = True
            queue.append((y, width - 1))
    while queue:
        cy, cx = queue.popleft()
        for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
            if 0 <= ny < height and 0 <= nx < width:
                if background[ny, nx] and not exterior[ny, nx]:
                    exterior[ny, nx] = True
                    queue.append((ny, nx))
    return exterior


def _pocket_count(image: Image.Image) -> int:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    background_rgb = _background_color(array)
    distance = np.linalg.norm(array.astype(np.float32) - background_rgb, axis=2)
    foreground = distance > 30.0
    if int(foreground.sum()) < 2500:
        raise ValueError("route has too little foreground")
    ys, xs = np.nonzero(foreground)
    if int(xs.max() - xs.min()) < 150 or int(ys.max() - ys.min()) < 65:
        raise ValueError("route extent is incomplete")
    foreground_components = [size for size in _component_sizes(foreground) if size >= 80]
    if len(foreground_components) != 1:
        raise ValueError("route is not one connected foreground component")
    exterior = _exterior_background(~foreground)
    enclosed = (~foreground) & ~exterior
    pockets = [size for size in _component_sizes(enclosed) if size >= 700]
    if len(pockets) not in (1, 2):
        raise ValueError("route does not expose exactly one or two stable pockets")
    if min(pockets) < 2500:
        raise ValueError("pocket is too small for a confident topology decision")
    return len(pockets)


def decision_from_image(image: Image.Image) -> str:
    return "yes" if _pocket_count(image) == 2 else "no"
