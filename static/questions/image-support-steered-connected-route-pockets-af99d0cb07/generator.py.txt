"""Deterministic renderer and analytic gold for a connected route topology task."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 384
HEIGHT = 384
SUPER = 4
BACKGROUND = (255, 255, 255)
PALETTE: tuple[tuple[int, int, int], ...] = (
    (33, 78, 142),
    (163, 54, 63),
    (39, 119, 82),
    (112, 65, 148),
    (196, 103, 27),
)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample balanced topology classes and independent rendered nuisances."""

    rng = random.Random(int(seed))
    return {
        "pockets": 1 + rng.randrange(2),
        "cx": round(rng.uniform(184.0, 200.0), 4),
        "cy": round(rng.uniform(187.0, 197.0), 4),
        "scale": round(rng.uniform(0.94, 1.04), 5),
        "mirror": bool(rng.randrange(2)),
        "stroke": round(rng.uniform(7.0, 9.2), 4),
        "ink": list(rng.choice(PALETTE)),
        "spur_side": "top" if rng.randrange(2) == 0 else "bottom",
        "spur_x": round(rng.uniform(-0.18, 0.18), 5),
        "spur_hook": -1 if rng.randrange(2) == 0 else 1,
    }


def _route_paths(scene: dict[str, Any]) -> list[list[tuple[float, float]]]:
    cx = float(scene["cx"])
    cy = float(scene["cy"])
    scale = float(scene["scale"])
    # A long, shallow frame keeps the decisive ablation trace regional while
    # making its connected support visibly pathlike at the protected grid.
    width = 250.0 * scale
    height = 82.0 * scale
    x0, x1 = cx - width / 2.0, cx + width / 2.0
    y0, y1 = cy - height / 2.0, cy + height / 2.0
    frame = [(x0, y0), (x1, y0), (x1, y1), (x0, y1), (x0, y0)]
    paths = [frame]
    ratio = float(scene["spur_x"])
    if int(scene["pockets"]) == 2:
        divider_x = cx + ratio * width * 0.42
        paths.append([(divider_x, y0), (divider_x, y1)])
    else:
        spur_x = cx + ratio * width * 0.42
        hook = float(scene["spur_hook"])
        hook_length = 0.17 * width
        if scene["spur_side"] == "top":
            end_y = y0 + 0.72 * height
            paths.append(
                [(spur_x, y0), (spur_x, end_y), (spur_x + hook * hook_length, end_y)]
            )
        else:
            end_y = y1 - 0.72 * height
            paths.append(
                [(spur_x, y1), (spur_x, end_y), (spur_x + hook * hook_length, end_y)]
            )
    if bool(scene["mirror"]):
        paths = [[(2.0 * cx - x, y) for x, y in path] for path in paths]
    return paths


def _draw_route(draw: ImageDraw.ImageDraw, path: list[tuple[float, float]], color, width: int) -> None:
    points = [(round(x * SUPER), round(y * SUPER)) for x, y in path]
    draw.line(points, fill=color, width=width, joint="curve")
    radius = width / 2.0
    for x, y in points:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (WIDTH * SUPER, HEIGHT * SUPER), BACKGROUND)
    draw = ImageDraw.Draw(image)
    width = max(3, int(round(float(scene["stroke"]) * SUPER)))
    color = tuple(int(value) for value in scene["ink"])
    for path in _route_paths(scene):
        _draw_route(draw, path, color, width)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS).convert("RGB")


def analytic_gold(scene: dict[str, Any]) -> str:
    return "yes" if int(scene["pockets"]) == 2 else "no"


def margin(scene: dict[str, Any]) -> float:
    return float(scene["pockets"]) - 1.5


def is_quarantined(scene: dict[str, Any]) -> bool:
    return False


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """The compact parameterization declares no distinct pixel-identical aliases."""

    return []
