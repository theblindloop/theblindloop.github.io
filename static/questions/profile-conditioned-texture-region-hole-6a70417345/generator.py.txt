"""Deterministic slash-texture renderer and latent analytic arm."""

from __future__ import annotations

import copy
import random
from collections import deque

from PIL import Image, ImageDraw


ROWS = 13
COLS = 15


def _bounded_complement(mask: list[list[bool]], rows: int, cols: int) -> list[int]:
    seen: set[tuple[int, int]] = set()
    queue: deque[tuple[int, int]] = deque()
    for r in range(rows):
        for c in (0, cols - 1):
            if not mask[r][c] and (r, c) not in seen:
                seen.add((r, c)); queue.append((r, c))
    for c in range(cols):
        for r in (0, rows - 1):
            if not mask[r][c] and (r, c) not in seen:
                seen.add((r, c)); queue.append((r, c))
    while queue:
        r, c = queue.popleft()
        for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            rr, cc = r + dr, c + dc
            if 0 <= rr < rows and 0 <= cc < cols and not mask[rr][cc] and (rr, cc) not in seen:
                seen.add((rr, cc)); queue.append((rr, cc))
    sizes: list[int] = []
    remaining = {(r, c) for r in range(rows) for c in range(cols) if not mask[r][c] and (r, c) not in seen}
    while remaining:
        start = remaining.pop(); component = {start}; queue.append(start)
        while queue:
            r, c = queue.popleft()
            for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                p = (r + dr, c + dc)
                if p in remaining:
                    remaining.remove(p); component.add(p); queue.append(p)
        sizes.append(len(component))
    return sizes


def _mask_from_scene(grid: list[str], rows: int, cols: int) -> list[list[bool]]:
    if len(grid) != rows or any(len(row) != cols for row in grid):
        raise ValueError("grid dimensions do not match")
    return [[value == "1" for value in row] for row in grid]


def _target_connected(mask: list[list[bool]]) -> bool:
    pts = {(r, c) for r, row in enumerate(mask) for c, value in enumerate(row) if value}
    if not pts:
        return False
    seen = {next(iter(pts))}; queue = deque(seen)
    while queue:
        r, c = queue.popleft()
        for p in ((r + 1, c), (r - 1, c), (r, c + 1), (r, c - 1)):
            if p in pts and p not in seen:
                seen.add(p); queue.append(p)
    return seen == pts


def _transform(points: set[tuple[int, int]], width: int, height: int, turn: int, mirror: bool) -> tuple[set[tuple[int, int]], int, int]:
    pts = set(points)
    w, h = width, height
    if mirror:
        pts = {(r, w - 1 - c) for r, c in pts}
    for _ in range(turn % 4):
        pts = {(c, h - 1 - r) for r, c in pts}
        w, h = h, w
    return pts, w, h


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    decision = "yes" if seed % 2 == 0 else "no"
    variants = [(9, 7, 3, 3), (8, 8, 2, 5), (10, 7, 4, 4)]
    width, height, cut_w, cut_h = variants[(seed // 2) % len(variants)]
    target = {(r, c) for r in range(height) for c in range(width)}
    cut_x = (width - cut_w) // 2
    if decision == "yes":
        cut_y = (height - cut_h) // 2
    else:
        cut_y = 0
    target -= {(r, c) for r in range(cut_y, cut_y + cut_h) for c in range(cut_x, cut_x + cut_w)}
    turn = rng.randrange(4)
    target, width, height = _transform(target, width, height, turn, bool(rng.randrange(2)))
    top = rng.randint(1, ROWS - height - 1)
    left = rng.randint(1, COLS - width - 1)
    mask = [[False] * COLS for _ in range(ROWS)]
    for r, c in target:
        mask[top + r][left + c] = True
    if sum(map(sum, mask)) != 54 or not _target_connected(mask):
        raise AssertionError("invalid sampled target region")
    grid = ["".join("1" if value else "0" for value in row) for row in mask]
    spacing = rng.choice([27, 28, 29])
    length = rng.choice([16, 18, 20])
    order = list(range(ROWS * COLS)); rng.shuffle(order)
    return {
        "grid": grid,
        "rows": ROWS,
        "cols": COLS,
        "origin": [rng.randint(53, 67), rng.randint(42, 54)],
        "spacing": spacing,
        "dash_length": length,
        "dash_width": rng.choice([3, 4]),
        "canvas": [520, 430],
        "palette": {"background": "#f7f4ed", "ink": "#17191c"},
        "mark_order": order,
    }


def render(scene: dict) -> Image.Image:
    grid = scene["grid"]
    rows, cols = int(scene["rows"]), int(scene["cols"])
    ox, oy = map(int, scene["origin"])
    spacing = int(scene["spacing"])
    length, width = int(scene["dash_length"]), int(scene["dash_width"])
    canvas = tuple(map(int, scene["canvas"]))
    palette = scene["palette"]
    order = list(scene["mark_order"])
    image = Image.new("RGB", canvas, palette["background"])
    draw = ImageDraw.Draw(image)
    half = length // 2
    for index in order:
        r, c = divmod(int(index), cols)
        if not (0 <= r < rows):
            raise ValueError("mark order outside grid")
        x, y = ox + c * spacing, oy + r * spacing
        if grid[r][c] == "1":
            endpoints = (x - half, y + half, x + half, y - half)
        else:
            endpoints = (x - half, y - half, x + half, y + half)
        draw.line(endpoints, fill=palette["ink"], width=width)
    return image


def analytic_gold(scene: dict) -> str:
    mask = _mask_from_scene(scene["grid"], int(scene["rows"]), int(scene["cols"]))
    return "yes" if _bounded_complement(mask, int(scene["rows"]), int(scene["cols"])) else "no"


def margin(scene: dict) -> float:
    mask = _mask_from_scene(scene["grid"], int(scene["rows"]), int(scene["cols"]))
    holes = _bounded_complement(mask, int(scene["rows"]), int(scene["cols"]))
    if holes:
        return float(max(holes))
    # All no-scenes expose the potential pocket through a channel at least two cells wide.
    return -2.0


def is_quarantined(scene: dict) -> bool:
    mask = _mask_from_scene(scene["grid"], int(scene["rows"]), int(scene["cols"]))
    return not _target_connected(mask)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["mark_order"] = list(reversed(twin["mark_order"]))
    return [("reverse_nonoverlapping_painter_order", twin)]
