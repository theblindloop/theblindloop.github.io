"""Independent raster-only inverse for slash-texture topology."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))]); seen[y, x] = True; pts = []
        while queue:
            yy, xx = queue.popleft(); pts.append((yy, xx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    ny, nx = yy + dy, xx + dx
                    if (dy or dx) and 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True; queue.append((ny, nx))
        if len(pts) >= 12:
            result.append(np.asarray(pts, dtype=float))
    return result


def _cluster_levels(values: list[float]) -> list[list[int]]:
    order = sorted(range(len(values)), key=values.__getitem__)
    groups: list[list[int]] = []
    for index in order:
        if not groups or values[index] - np.mean([values[i] for i in groups[-1]]) > 10.0:
            groups.append([index])
        else:
            groups[-1].append(index)
    return groups


def _has_hole(mask: list[list[bool]]) -> bool:
    rows, cols = len(mask), len(mask[0])
    outside: set[tuple[int, int]] = set(); queue: deque[tuple[int, int]] = deque()
    for r in range(rows):
        for c in (0, cols - 1):
            if not mask[r][c] and (r, c) not in outside:
                outside.add((r, c)); queue.append((r, c))
    for c in range(cols):
        for r in (0, rows - 1):
            if not mask[r][c] and (r, c) not in outside:
                outside.add((r, c)); queue.append((r, c))
    while queue:
        r, c = queue.popleft()
        for rr, cc in ((r + 1, c), (r - 1, c), (r, c + 1), (r, c - 1)):
            if 0 <= rr < rows and 0 <= cc < cols and not mask[rr][cc] and (rr, cc) not in outside:
                outside.add((rr, cc)); queue.append((rr, cc))
    return any(not mask[r][c] and (r, c) not in outside for r in range(rows) for c in range(cols))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    ink = arr.mean(axis=2) < 90
    components = _components(ink)
    if len(components) < 20:
        raise ValueError("raster does not contain a complete dash lattice")
    centers = [(float(p[:, 1].mean()), float(p[:, 0].mean())) for p in components]
    row_groups = _cluster_levels([p[1] for p in centers])
    if len(row_groups) < 5 or len({len(group) for group in row_groups}) != 1:
        raise ValueError("could not infer a complete regular lattice")
    cols = len(row_groups[0])
    recovered: list[list[bool]] = []
    for group in row_groups:
        group.sort(key=lambda i: centers[i][0])
        if len(group) != cols:
            raise ValueError("incomplete lattice row")
        row = []
        for index in group:
            pts = components[index]
            x = pts[:, 1] - pts[:, 1].mean()
            y = pts[:, 0] - pts[:, 0].mean()
            covariance = float(np.dot(x, y))
            if abs(covariance) < 50:
                raise ValueError("uncertain dash orientation")
            row.append(covariance < 0.0)
        recovered.append(row)
    return "yes" if _has_hole(recovered) else "no"
