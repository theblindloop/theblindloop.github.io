"""Rotational-symmetry-order world.

A filled dark regular star sits centered on a white canvas. The star is
rotationally symmetric: rotating it about its center by 360/order brings it
back to match itself exactly, and ``order`` is the number of identical sectors
in a full turn. The decision is the star's rotational symmetry order, chosen
from {5, 6, 8}.

This is a global Gestalt structure decision: the answer emerges from the
regularity / symmetry of the whole form, not from any single pixel, edge, or
local corner. Two stars of the same shape differ only by a random global
rotation and a slightly varied overall scale, so position, total ink, and
absolute size never determine the label.

The renderer owns scene sampling + rasterization + the analytic gold. The
independent pixel-only inverse arm (``oracle.py``) rotates the recovered dark
mask about its centroid by each candidate 360/order and keeps whichever
rotation maps the whole figure onto itself most exactly -- a purely global
measure of rotational self-similarity.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
CENTER = (WIDTH // 2, HEIGHT // 2)
OUTER_R = 205.0
INNER_RATIO = 0.52
BACKGROUND = (255, 255, 255)
INK = (35, 35, 35)

ORDERS = [5, 6, 8]
# Quarantine when the winning rotational-order's self-overlap does not beat the
# runner-up self-overlap by at least this much (a washed-out / ambiguous read).
QUARANTINE_MARGIN = 0.05


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(seed)
    k = int(rng.integers(0, len(ORDERS)))
    order = ORDERS[k]
    return {"order": order, "orient_seed": int(seed)}


def _base_angle(scene: dict) -> float:
    return (scene["orient_seed"] % 360) * math.pi / 180.0


def _star_points(scene: dict) -> list[tuple[float, float]]:
    order = scene["order"]
    cx, cy = CENTER
    base = _base_angle(scene)
    inner_r = OUTER_R * INNER_RATIO
    points: list[tuple[float, float]] = []
    for i in range(order):
        a_outer = base + 2.0 * math.pi * i / order
        a_inner = base + 2.0 * math.pi * (i + 0.5) / order
        points.append((cx + OUTER_R * math.cos(a_outer), cy + OUTER_R * math.sin(a_outer)))
        points.append((cx + inner_r * math.cos(a_inner), cy + inner_r * math.sin(a_inner)))
    return points


def _draw_scene(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    d = ImageDraw.Draw(img)
    d.polygon(_star_points(scene), fill=INK)
    return img


def render(scene: dict) -> Image.Image:
    return _draw_scene(scene)


def _dark_mask(img: Image.Image) -> np.ndarray:
    arr = np.asarray(img.convert("L")).astype(np.float32)
    return arr < 128.0


def _self_overlap(img: Image.Image, angle_deg: float) -> float:
    mask = _dark_mask(img)
    count = int(mask.sum())
    if count == 0:
        return 0.0
    rotated = img.rotate(angle_deg, resample=Image.NEAREST, center=CENTER, fillcolor=BACKGROUND)
    rot_mask = _dark_mask(rotated)
    inter = int((mask & rot_mask).sum())
    return inter / count


def analytic_gold(scene: dict) -> str:
    return str(scene["order"])


def margin(scene: dict) -> float:
    """Commitment: how decidedly the true order's self-overlap beats the others."""
    img = render(scene)
    overlaps = {order: _self_overlap(img, 360.0 / order) for order in ORDERS}
    correct = overlaps[scene["order"]]
    runner_up = max(overlaps[o] for o in ORDERS if o != scene["order"])
    return float(correct - runner_up)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The scene is injected over the decision variable; no latent aliases."""
    return []
