"""Independent pixel-only inverse arm for the star-rotational-order world.

Recovers the star's dark mass from the final PNG alone (no scene, no gold),
then measures its rotational self-similarity: for each candidate ``order`` in
{5, 6, 8} it rotates the recovered dark mask about the figure centroid by
``360/order`` and measures what fraction of the dark mass maps back onto dark
pixels. The rotation that best returns the whole figure to itself identifies
the rotational symmetry order.

This module imports no renderer / prompts / generate / verify / analytic-gold
code and never receives the latent scene.
"""

from __future__ import annotations

from PIL import Image
import numpy as np

ORDERS = (5, 6, 8)
_BG = (255, 255, 255)


def _dark_mask(img: Image.Image) -> np.ndarray:
    arr = np.asarray(img.convert("L")).astype(np.float32)
    return arr < 128.0


def _self_overlap(img: Image.Image, angle_deg: float) -> float:
    """Fraction of the original dark mass that remains dark after a rotation.

    A perfect ``order`` rotation maps the whole symmetric star onto itself, so
    this fraction is near 1; a wrong rotation scatters the lobes and lowers it.
    """
    h, w = img.size
    center = (w // 2, h // 2)
    mask = _dark_mask(img)
    count = int(mask.sum())
    if count == 0:
        return 0.0
    rotated = img.rotate(angle_deg, resample=Image.NEAREST, center=center, fillcolor=_BG)
    rot_mask = _dark_mask(rotated)
    inter = int((mask & rot_mask).sum())
    return inter / count


def decision_from_image(image: Image.Image) -> str:
    """Return the recovered rotational symmetry order as a string in {5,6,8}."""
    img = image.convert("RGB")
    best_order = ORDERS[0]
    best_overlap = -1.0
    for order in ORDERS:
        overlap = _self_overlap(img, 360.0 / order)
        if overlap > best_overlap:
            best_overlap = overlap
            best_order = order
    return str(best_order)
