"""Independent pixel-only inverse arm for the threshold-count world.

This module receives ONLY the rasterized PNG and independently recovers the top
(y) extent of the gray reference bar and the top extent of every colored bar by
color-segmented bounding boxes, then returns the number of colored bars whose top
sits strictly above the gray reference bar's top. It shares no geometry or grid
with renderer.render() and never imports the renderer, prompts, generator,
verifier, or any analytic-gold code.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "taller_than_reference-oracle-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
GRAY_REF = (96, 96, 104)
BACKGROUND = (250, 250, 250)


def _labels(arr: np.ndarray) -> np.ndarray:
    """Label each pixel to its nearest of {red, blue, green, gray-reference, paper}."""
    refs = np.stack(
        [np.array(COLOR_RGB[c], dtype=np.float32) for c in PALETTE]
        + [np.array(GRAY_REF, dtype=np.float32), np.array(BACKGROUND, dtype=np.float32)],
        axis=0,
    )
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def _top_y(mask: np.ndarray) -> float:
    """Topmost inked row of a single-colour region (its min y)."""
    ys = np.nonzero(mask)[0]
    if len(ys) < 20:
        return float("nan")
    return float(ys.min())


def decision_from_image(image: Image.Image) -> str:
    """Return the count of colored bars whose top is above the gray reference top."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _labels(arr)
    gray = _top_y(labels == len(PALETTE))  # gray-reference label
    if np.isnan(gray):
        return "0"
    count = 0
    for idx in range(len(PALETTE)):
        top = _top_y(labels == idx)
        if not np.isnan(top) and top < gray:
            count += 1
    return str(count)
