"""Threshold-count measurement world.

Three colored vertical bars (red, blue, green) and one neutral gray reference
bar stand on a shared black baseline on a light panel. The decision is a pure
threshold measurement: recover the top height of the gray reference bar and the
top height of every colored bar, then count how many colored bars are TALLER
than the gray reference bar (a colored bar counts when its top sits strictly
above the gray bar's top). The answer is that integer count, between 0 and 3.

The count is counterbalanced over the independent random heights of all four
bars and over an arbitrary shuffle of the three color labels among the three
colored slots, so neither the tallest bar, the gray-bar height alone, any single
colored bar's height, nor any positional or colour ordering cue discloses the
count; only the recovered threshold comparison of each colored bar's top against
the gray reference top does.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "taller_than_reference-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
GRAY_REF = (96, 96, 104)     # neutral reference bar, distinct from palette + paper
BASELINE_RGB = (20, 20, 20)  # black baseline shared by all bars
BACKGROUND = (250, 250, 250)

WIDTH = 500
HEIGHT = 300
BASELINE = 250
BAR_W = 36
COLOR_SLOT_X = [70, 185, 300]   # x of the three colored bars (left edges)
REF_X = 415                     # x of the gray reference bar (left edge)
H_MIN = 45
H_MAX = 195
QUARANTINE_PX = 5.0             # below this |colored_h - gray_h| gaps are ambiguous
MIN_MARGIN = 6.0                # emitted scenes keep every colored bar this far from gray


def sample_scene(seed: int) -> dict:
    """Three colored-bar heights + a gray reference height, colors shuffled."""
    rng = random.Random(seed * 1049 + 17)
    while True:
        gray_h = rng.randint(H_MIN, H_MAX)
        heights = [rng.randint(H_MIN, H_MAX) for _ in PALETTE]
        if all(abs(h - gray_h) >= MIN_MARGIN for h in heights):
            break
    colors = list(PALETTE)
    rng.shuffle(colors)
    colored = []
    for color, height in zip(colors, heights):
        colored.append({"color": color, "height": height, "x": COLOR_SLOT_X[len(colored)]})
    return {
        "colored": colored,
        "reference": {"height": gray_h},
        "width": WIDTH,
        "height": HEIGHT,
        "baseline": BASELINE,
    }


def render(scene: dict) -> Image.Image:
    """Draw the gray reference bar and the three colored bars above the baseline."""
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    baseline = scene["baseline"]
    draw.rectangle([0, baseline, scene["width"], baseline + 3], fill=BASELINE_RGB)
    gray_h = scene["reference"]["height"]
    draw.rectangle(
        [REF_X, baseline - gray_h, REF_X + BAR_W, baseline], fill=GRAY_REF
    )
    for bar in scene["colored"]:
        x = bar["x"]
        h = bar["height"]
        draw.rectangle([x, baseline - h, x + BAR_W, baseline], fill=COLOR_RGB[bar["color"]])
    return img


def analytic_gold(scene: dict) -> str:
    """Count colored bars whose top is strictly above the gray reference top."""
    gray_h = scene["reference"]["height"]
    count = sum(1 for bar in scene["colored"] if bar["height"] > gray_h)
    return str(count)


def margin(scene: dict) -> float:
    """Smallest |colored_h - gray_h| gap; the decision's threshold margin."""
    gray_h = scene["reference"]["height"]
    return float(min(abs(bar["height"] - gray_h) for bar in scene["colored"]))


def is_quarantined(scene: dict) -> bool:
    """Ambiguous when a colored top is within QUARANTINE of the gray reference top."""
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list:
    """not_applicable: every latent field is directly rendered as visible ink."""
    return []
