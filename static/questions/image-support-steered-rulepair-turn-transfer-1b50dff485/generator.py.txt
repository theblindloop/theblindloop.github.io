"""Rule-pair turn transfer world.

A cream board carries seven framed 5x5 dot tiles that never touch each other.

Two tiles form the RULE PAIR and sit at opposite corners of the board: the
navy-framed BEFORE tile and the brown-framed AFTER tile show one and the same
pattern before and after a single hidden change taken from four declared
operations (quarter turn clockwise, half turn, left-right mirror, top-bottom
mirror). Exactly one of the four operations carries BEFORE onto AFTER, so the
rule in force is a latent variable that must be inferred from two widely
separated tiles.

A third graphite-framed QUERY tile sits at a further corner. The four remaining
tiles wear distinct palette frame colours and are precisely the four images of
the query under the four declared operations. Every candidate therefore carries
exactly the same number of dots as the query and as every other candidate, and
no candidate can be preferred by ink, count, size, colour or position: the only
way to choose is to infer the operation from the distant rule pair and transfer
it to the distant query.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "rulepair_turn_transfer-0.1.0"

CANVAS_W = 880
CANVAS_H = 880

GRID = 5
CELL = 24
STROKE = 7
TILE_SIDE = GRID * CELL + 2 * STROKE  # 134

BOARD = (244, 239, 228)
CELL_BG = (253, 252, 249)
GRIDLINE = (212, 207, 196)
INK = (28, 30, 36)
BEFORE_FRAME = (32, 62, 118)
AFTER_FRAME = (124, 86, 22)
QUERY_FRAME = (58, 60, 68)

COLOR_NAMES = ("red", "orange", "green", "teal", "blue", "purple")
COLORS = {
    "red": (204, 54, 48),
    "orange": (228, 130, 24),
    "green": (52, 148, 64),
    "teal": (28, 150, 158),
    "blue": (60, 94, 204),
    "purple": (138, 64, 176),
}

DECOR_COLORS = ((200, 190, 166), (180, 192, 180))
DECOR_SIDE = 11
N_DECOR = 15

N_CELLS = GRID * GRID
OPERATIONS = (
    "quarter_turn_clockwise",
    "half_turn",
    "mirror_left_right",
    "mirror_top_bottom",
)
N_CANDIDATES = len(OPERATIONS)

SLOT_XS = (150, 440, 730)
SLOT_YS = (150, 440, 730)
SLOTS = tuple((SLOT_XS[c], SLOT_YS[r]) for r in range(3) for c in range(3))
DIAGONALS = ((0, 8), (8, 0), (2, 6), (6, 2))

QUARANTINE_MARGIN = 4.0  # a runner-up within three cells is a near-boundary case


def _permutation(operation: str) -> tuple[int, ...]:
    """Cell index permutation: output cell i takes its state from input perm[i]."""
    perm = [0] * N_CELLS
    for row in range(GRID):
        for col in range(GRID):
            if operation == "quarter_turn_clockwise":
                new_row, new_col = col, GRID - 1 - row
            elif operation == "half_turn":
                new_row, new_col = GRID - 1 - row, GRID - 1 - col
            elif operation == "mirror_left_right":
                new_row, new_col = row, GRID - 1 - col
            elif operation == "mirror_top_bottom":
                new_row, new_col = GRID - 1 - row, col
            else:
                raise ValueError(f"unknown operation {operation!r}")
            perm[new_row * GRID + new_col] = row * GRID + col
    return tuple(perm)


PERMUTATIONS = {name: _permutation(name) for name in OPERATIONS}


def apply_operation(operation: str, cells: list[int]) -> list[int]:
    return [cells[source] for source in PERMUTATIONS[operation]]


def _hamming(left: list[int], right: list[int]) -> int:
    return sum(1 for a, b in zip(left, right) if a != b)


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------


def _random_pattern(rng: random.Random) -> list[int] | None:
    cells = [1 if rng.random() < 0.42 else 0 for _ in range(N_CELLS)]
    if not 8 <= sum(cells) <= 16:
        return None
    return cells


def _rule_margin(base: list[int], operation: str) -> int:
    """Cells by which the closest rival operation misses the AFTER tile."""
    target = apply_operation(operation, base)
    rivals = [
        _hamming(apply_operation(other, base), target)
        for other in OPERATIONS
        if other != operation
    ]
    return min(rivals)


def _transfer_margin(query: list[int], operation: str) -> int:
    """Cells by which the closest wrong candidate misses the correct candidate."""
    target = apply_operation(operation, query)
    rivals = [
        _hamming(apply_operation(other, query), target)
        for other in OPERATIONS
        if other != operation
    ]
    return min(rivals)


def _candidates_distinct(query: list[int]) -> bool:
    images = [tuple(apply_operation(name, query)) for name in OPERATIONS]
    return len(set(images)) == N_CANDIDATES


def sample_scene(seed: int) -> dict:
    """Sample one fully specified scene; deterministic in ``seed``."""
    rng = random.Random(seed * 6151 + 29)
    index = seed % 1000

    operation = OPERATIONS[index % N_CANDIDATES]
    tight = (index % 4) == 3

    for _ in range(6000):
        base = _random_pattern(rng)
        query = _random_pattern(rng)
        if base is None or query is None:
            continue
        if not _candidates_distinct(query):
            continue
        rule_gap = _rule_margin(base, operation)
        transfer_gap = _transfer_margin(query, operation)
        if rule_gap < 1 or transfer_gap < 1:
            continue
        gap = min(rule_gap, transfer_gap)
        if tight:
            if gap > 3:
                continue
        elif gap < 5:
            continue

        colours = [c for c in COLOR_NAMES]
        rng.shuffle(colours)
        answer_colour = COLOR_NAMES[index % len(COLOR_NAMES)]
        others = [c for c in colours if c != answer_colour][: N_CANDIDATES - 1]

        d0, d1 = DIAGONALS[index % len(DIAGONALS)]
        spare_corners = [s for s in (0, 2, 6, 8) if s not in (d0, d1)]
        query_slot = spare_corners[(index // N_CANDIDATES) % len(spare_corners)]
        free = [s for s in range(len(SLOTS)) if s not in (d0, d1, query_slot)]
        rng.shuffle(free)
        chosen = sorted(free[:N_CANDIDATES])
        rank = index % N_CANDIDATES

        def place(slot: int) -> tuple[int, int]:
            cx, cy = SLOTS[slot]
            return (
                cx + rng.randint(-12, 12) - TILE_SIDE // 2,
                cy + rng.randint(-12, 12) - TILE_SIDE // 2,
            )

        wrong_ops = [name for name in OPERATIONS if name != operation]
        tiles = []
        wrong_index = 0
        for position, slot in enumerate(chosen):
            x, y = place(slot)
            if position == rank:
                tiles.append(
                    {
                        "x": x,
                        "y": y,
                        "colour": answer_colour,
                        "cells": apply_operation(operation, query),
                    }
                )
            else:
                tiles.append(
                    {
                        "x": x,
                        "y": y,
                        "colour": others[wrong_index],
                        "cells": apply_operation(wrong_ops[wrong_index], query),
                    }
                )
                wrong_index += 1

        bx, by = place(d0)
        ax, ay = place(d1)
        qx, qy = place(query_slot)
        before = {"x": bx, "y": by, "cells": list(base)}
        after = {"x": ax, "y": ay, "cells": apply_operation(operation, base)}
        query_tile = {"x": qx, "y": qy, "cells": list(query)}

        boxes = [(t["x"], t["y"]) for t in tiles]
        boxes += [(bx, by), (ax, ay), (qx, qy)]
        decor = _sample_decor(rng, boxes)

        return {
            "before": before,
            "after": after,
            "query": query_tile,
            "tiles": tiles,
            "decor": decor,
            "tight": bool(tight),
        }
    raise RuntimeError(f"scene sampling failed for seed {seed}")


def _sample_decor(rng: random.Random, boxes: list[tuple[int, int]]) -> list[dict]:
    placed: list[dict] = []
    guard = 0
    while len(placed) < N_DECOR and guard < 5000:
        guard += 1
        x = rng.randint(14, CANVAS_W - 14 - DECOR_SIDE)
        y = rng.randint(14, CANVAS_H - 14 - DECOR_SIDE)
        clash = False
        for bx, by in boxes:
            if bx - 22 <= x <= bx + TILE_SIDE + 11 and by - 22 <= y <= by + TILE_SIDE + 11:
                clash = True
                break
        if clash:
            continue
        for other in placed:
            if abs(other["x"] - x) < DECOR_SIDE + 9 and abs(other["y"] - y) < DECOR_SIDE + 9:
                clash = True
                break
        if clash:
            continue
        placed.append({"x": x, "y": y, "tone": rng.randrange(len(DECOR_COLORS))})
    return placed


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------


def _draw_tile(draw: ImageDraw.ImageDraw, x: int, y: int, frame, cells) -> None:
    draw.rectangle([x, y, x + TILE_SIDE - 1, y + TILE_SIDE - 1], fill=frame)
    ix, iy = x + STROKE, y + STROKE
    inner = GRID * CELL
    draw.rectangle([ix, iy, ix + inner - 1, iy + inner - 1], fill=CELL_BG)
    for i in range(1, GRID):
        draw.line([ix + i * CELL, iy, ix + i * CELL, iy + inner - 1], fill=GRIDLINE)
        draw.line([ix, iy + i * CELL, ix + inner - 1, iy + i * CELL], fill=GRIDLINE)
    for idx in range(N_CELLS):
        row, col = divmod(idx, GRID)
        if cells[idx] != 1:
            continue
        cx0 = ix + col * CELL
        cy0 = iy + row * CELL
        draw.rectangle([cx0 + 3, cy0 + 3, cx0 + CELL - 4, cy0 + CELL - 4], fill=INK)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (CANVAS_W, CANVAS_H), BOARD)
    draw = ImageDraw.Draw(image)
    for spot in sorted(scene["decor"], key=lambda d: (d["y"], d["x"])):
        x, y = spot["x"], spot["y"]
        draw.rectangle(
            [x, y, x + DECOR_SIDE - 1, y + DECOR_SIDE - 1],
            fill=DECOR_COLORS[spot["tone"]],
        )
    fixed = (
        (scene["before"], BEFORE_FRAME),
        (scene["after"], AFTER_FRAME),
        (scene["query"], QUERY_FRAME),
    )
    for tile, frame in sorted(fixed, key=lambda pair: (pair[0]["y"], pair[0]["x"])):
        _draw_tile(draw, tile["x"], tile["y"], frame, tile["cells"])
    for tile in sorted(scene["tiles"], key=lambda t: (t["y"], t["x"])):
        _draw_tile(draw, tile["x"], tile["y"], COLORS[tile["colour"]], tile["cells"])
    return image


# --------------------------------------------------------------------------
# analytic gold
# --------------------------------------------------------------------------


def inferred_operation(scene: dict) -> str:
    """The unique declared operation carrying the BEFORE tile onto the AFTER tile."""
    base = scene["before"]["cells"]
    goal = scene["after"]["cells"]
    hits = [name for name in OPERATIONS if apply_operation(name, base) == goal]
    if len(hits) != 1:
        raise ValueError("the rule pair does not determine a unique operation")
    return hits[0]


def analytic_gold(scene: dict) -> str:
    """Frame colour of the candidate equal to the query under the inferred rule."""
    target = apply_operation(inferred_operation(scene), scene["query"]["cells"])
    winners = [t for t in scene["tiles"] if t["cells"] == target]
    if len(winners) != 1:
        raise ValueError("scene does not have a unique transferred candidate")
    return winners[0]["colour"]


def stage_margins(scene: dict) -> tuple[int, int]:
    """(rule-pair margin, transfer margin) in disagreeing cells."""
    operation = inferred_operation(scene)
    base = scene["before"]["cells"]
    goal = scene["after"]["cells"]
    rule = min(
        _hamming(apply_operation(other, base), goal)
        for other in OPERATIONS
        if other != operation
    )
    target = apply_operation(operation, scene["query"]["cells"])
    transfer = min(
        _hamming(t["cells"], target) for t in scene["tiles"] if t["cells"] != target
    )
    return rule, transfer


def margin(scene: dict) -> float:
    """Runner-up decision margin: the weaker of the two evidence stages, in cells."""
    rule, transfer = stage_margins(scene)
    return float(min(rule, transfer))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Stored list order is a pure alias.

    No two drawn objects overlap: tiles sit on a widely separated slot lattice
    and decorative squares are rejection-sampled away from every tile and from
    each other, so draw order is invisible. The gold searches the candidate
    tiles for the one equal to the transferred query and never consults list
    position, and the operation is recomputed from the two exemplar tiles rather
    than read from a label, so permuting the stored lists re-draws a
    byte-identical raster with an unchanged answer.
    """
    tiles = scene["tiles"]
    return [
        ("tile_list_rotation", {**scene, "tiles": tiles[1:] + tiles[:1]}),
        ("tile_list_reversal", {**scene, "tiles": list(reversed(tiles))}),
        ("tile_list_double_rotation", {**scene, "tiles": tiles[2:] + tiles[:2]}),
        ("decor_list_reversal", {**scene, "decor": list(reversed(scene["decor"]))}),
    ]
