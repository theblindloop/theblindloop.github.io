"""Independent image-only inverse for paired enclosed-pocket comparison."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


_RED = np.asarray([211.0, 76.0, 88.0], dtype=np.float32)
_BLUE = np.asarray([62.0, 117.0, 197.0], dtype=np.float32)
_WHITE = np.asarray([255.0, 255.0, 255.0], dtype=np.float32)
_MIN_COMPONENT = 1200
_MIN_HOLE = 180


def _components(mask: np.ndarray, minimum: int = 1) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y = int(y)
        x = int(x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = cy + dy, cx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= int(minimum):
            output.append(np.asarray(points, dtype=np.int32))
    return output


def _color_mask(image: Image.Image, color: np.ndarray) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    other = _BLUE if np.array_equal(color, _RED) else _RED
    # Nearest-color labeling is deliberately independent of renderer geometry.
    distance = np.linalg.norm(array - color, axis=2)
    white_distance = np.linalg.norm(array - _WHITE, axis=2)
    other_distance = np.linalg.norm(array - other, axis=2)
    return (distance < white_distance) & (distance < other_distance) & (distance < 190.0)


def _border_reachable(mask: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    queue: deque[tuple[int, int]] = deque()
    for x in range(width):
        if mask[0, x]:
            queue.append((0, x))
        if mask[height - 1, x]:
            queue.append((height - 1, x))
    for y in range(height):
        if mask[y, 0]:
            queue.append((y, 0))
        if mask[y, width - 1]:
            queue.append((y, width - 1))
    while queue:
        y, x = queue.popleft()
        if not (0 <= y < height and 0 <= x < width) or seen[y, x] or not mask[y, x]:
            continue
        seen[y, x] = True
        queue.extend(((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)))
    return seen


def _hole_count(mask: np.ndarray) -> int:
    components = _components(mask, minimum=_MIN_COMPONENT)
    if len(components) != 1:
        raise ValueError(f"expected one intact colored island, found {len(components)}")
    component = components[0]
    y0, x0 = component.min(axis=0)
    y1, x1 = component.max(axis=0) + 1
    local = mask[int(y0) : int(y1), int(x0) : int(x1)]
    background = ~local
    reachable = _border_reachable(background)
    enclosed = background & ~reachable
    holes = 0
    for component in _components(enclosed, minimum=_MIN_HOLE):
        if len(component) >= _MIN_HOLE:
            holes += 1
    if holes < 1 or holes > 3:
        raise ValueError(f"unstable enclosed-pocket count {holes}")
    return holes


def decision_from_image(image: Image.Image) -> str:
    """Return red or blue after counting enclosed white components in both masks."""

    red_count = _hole_count(_color_mask(image, _RED))
    blue_count = _hole_count(_color_mask(image, _BLUE))
    if red_count == blue_count:
        raise ValueError("pocket counts tie; no red/blue selection")
    return "red" if red_count > blue_count else "blue"
