"""Deterministic raster renderer for paired enclosed-pocket comparison."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw


RENDERER_VERSION = "dual_island_pocket_order-0.1.0"
WIDTH = 512
HEIGHT = 512
SUPER = 4
BACKGROUND = "#ffffff"
RED = "#d34c58"
BLUE = "#3e75c5"
DECOR = "#e7ebf0"


def _box(cx: float, cy: float, width: float, height: float) -> list[float]:
    return [cx - width / 2.0, cy - height / 2.0, cx + width / 2.0, cy + height / 2.0]


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced pair of separated islands with unequal pocket counts."""

    value = int(seed)
    rng = random.Random(value)
    # The six ordered pairs make both answers and all count differences recur
    # deterministically, independent of the sampled geometry.
    red_holes, blue_holes = (
        (1, 2), (2, 1), (1, 3), (3, 1), (2, 3), (3, 2)
    )[value % 6]
    left_red = ((value // 2) % 2) == 0
    outer_width = float(rng.choice((96, 100, 104)))
    outer_height = float(rng.choice((148, 152, 156)))
    stroke = float(rng.choice((15, 16, 17)))
    corner = float(rng.choice((15, 17, 19)))
    hole_radius = float(rng.choice((9, 10, 11)))
    hole_gap = float(rng.choice((4, 6, 8)))
    y_center = float(rng.choice((238, 256, 274)))
    left_box = _box(116.0, y_center, outer_width, outer_height)
    right_box = _box(396.0, y_center, outer_width, outer_height)
    if left_red:
        red_box, blue_box = left_box, right_box
    else:
        red_box, blue_box = right_box, left_box
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": BACKGROUND,
        "red": RED,
        "blue": BLUE,
        "red_box": red_box,
        "blue_box": blue_box,
        "red_holes": red_holes,
        "blue_holes": blue_holes,
        "stroke": stroke,
        "corner": corner,
        "hole_radius": hole_radius,
        "hole_gap": hole_gap,
        "decor_seed": int(rng.randrange(1, 2**31 - 1)),
    }


def _scaled_box(box: list[float] | tuple[float, ...]) -> tuple[int, int, int, int]:
    return tuple(int(round(float(value) * SUPER)) for value in box)  # type: ignore[return-value]


def _pocket_offsets(count: int, hole_gap: float) -> list[tuple[float, float]]:
    # Keep neighboring white pockets disconnected even at the largest sampled
    # radius; the outer wall still has ample visible clearance.
    distance = 24.0 + float(hole_gap)
    if int(count) == 1:
        return [(0.0, 0.0)]
    if int(count) == 2:
        return [(-distance, 0.0), (distance, 0.0)]
    return [(-distance, 0.0), (0.0, 0.0), (distance, 0.0)]


def _draw_island(
    draw: ImageDraw.ImageDraw,
    scene: dict[str, Any],
    box: list[float],
    color: str,
    holes: int,
) -> None:
    draw.rounded_rectangle(
        _scaled_box(box),
        radius=int(round(float(scene["corner"]) * SUPER)),
        fill=color,
    )
    cx = (float(box[0]) + float(box[2])) / 2.0
    cy = (float(box[1]) + float(box[3])) / 2.0
    radius = float(scene["hole_radius"])
    for dx, dy in _pocket_offsets(int(holes), float(scene["hole_gap"])):
        draw.ellipse(
            _scaled_box((cx + dx - radius, cy + dy - radius, cx + dx + radius, cy + dy + radius)),
            fill=str(scene["background"]),
        )


def _draw_decorations(draw: ImageDraw.ImageDraw, scene: dict[str, Any]) -> None:
    """Render small neutral marks far from both islands as nuisance variation."""

    rng = random.Random(int(scene["decor_seed"]))
    for base_x, base_y in ((28, 34), (484, 36), (30, 476), (482, 474)):
        x = base_x + rng.randint(-6, 6)
        y = base_y + rng.randint(-6, 6)
        radius = rng.choice((2, 3)) * SUPER
        draw.ellipse((x * SUPER - radius, y * SUPER - radius, x * SUPER + radius, y * SUPER + radius), fill=DECOR)
    draw.line(
        [(38 * SUPER, 104 * SUPER), (68 * SUPER, 104 * SUPER)],
        fill=DECOR,
        width=SUPER,
    )
    draw.line(
        [(444 * SUPER, 408 * SUPER), (474 * SUPER, 408 * SUPER)],
        fill=DECOR,
        width=SUPER,
    )


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the scene; no analytic answer is encoded outside the visible holes."""

    width = int(scene["width"])
    height = int(scene["height"])
    canvas = Image.new("RGB", (width * SUPER, height * SUPER), str(scene["background"]))
    draw = ImageDraw.Draw(canvas)
    _draw_decorations(draw, scene)
    _draw_island(draw, scene, list(scene["red_box"]), str(scene["red"]), int(scene["red_holes"]))
    _draw_island(draw, scene, list(scene["blue_box"]), str(scene["blue"]), int(scene["blue_holes"]))
    return canvas.resize((width, height), Image.Resampling.LANCZOS).convert("RGB")


def analytic_gold(scene: dict[str, Any]) -> str:
    """Select the visible color whose enclosed-pocket count is larger."""

    red_holes = int(scene["red_holes"])
    blue_holes = int(scene["blue_holes"])
    if red_holes == blue_holes:
        return "tie"
    return "red" if red_holes > blue_holes else "blue"


def margin(scene: dict[str, Any]) -> float:
    """Discrete comparison margin: the absolute visible count difference."""

    return float(abs(int(scene["red_holes"]) - int(scene["blue_holes"])))


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Ties have no valid red/blue selection and are excluded."""

    return margin(scene) <= 0.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No intentionally pixel-identical latent aliases are declared."""

    del scene
    return []
