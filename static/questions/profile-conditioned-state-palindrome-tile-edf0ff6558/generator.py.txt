"""Renderer and analytic side for the keyed state-trajectory palindrome world.

The image is a seven-panel serpentine strip.  Every panel contains one tile for
each identity, but tile rows are shuffled.  The analytic question is whether a
label's ordered categorical state path is equal to its reverse.
"""

from __future__ import annotations

import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont


LABELS = ("A", "B", "C", "D")
STATE_NAMES = ("LOW", "MID", "HIGH")
CANVAS = (1800, 1160)
BACKGROUND = (245, 248, 251)
INK = (35, 44, 58)
MUTED = (91, 104, 119)
GRID = (187, 197, 208)
CARD = (255, 255, 255)

IDENTITY_PALETTE = (
    (207, 73, 88),
    (38, 126, 180),
    (111, 82, 169),
    (43, 148, 105),
)
STATE_PALETTE = (
    (226, 124, 54),
    (40, 145, 185),
    (157, 89, 163),
)
PANEL_TINTS = (
    (255, 255, 255),
    (247, 251, 254),
    (253, 249, 244),
    (249, 253, 247),
)

# Four panels on the top row, then three panels from right to left on the
# bottom row.  Arrows make the order explicit even though the panels are static.
PANEL_LEFTS = (30, 470, 910, 1350, 1350, 910, 470)
PANEL_TOPS = (225, 225, 225, 225, 700, 700, 700)
PANEL_W = 400
PANEL_H = 390
SLOT_OFFSETS = ((18, 78), (18, 144), (18, 210), (18, 276))
TILE_W = 364
TILE_H = 56
IDENTITY_STRIP_W = 48
IDENTITY_SAMPLE_X = 6
STATE_MARKER_OFFSET_X = 280
STATE_MARKER_OFFSET_Y = TILE_H // 2

IDENTITY_KEY_X = 30
IDENTITY_KEY_Y = 130
IDENTITY_KEY_STEP = 105
STATE_KEY_X = 520
STATE_KEY_Y = 130
STATE_KEY_STEP = 200
KEY_RADIUS = 16

_FONT_PATHS = (
    "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
    "/usr/share/fonts/dejavu/DejaVuSans.ttf",
)


def _font(size: int):
    for candidate in _FONT_PATHS:
        if Path(candidate).is_file():
            return ImageFont.truetype(candidate, size=size)
    return ImageFont.load_default()


def _centered_text(
    draw: ImageDraw.ImageDraw,
    xy: tuple[float, float],
    text: str,
    font,
    fill,
) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    width = box[2] - box[0]
    height = box[3] - box[1]
    draw.text(
        (xy[0] - width / 2, xy[1] - height / 2 - box[1]),
        text,
        font=font,
        fill=fill,
    )


def _identity_color(scene: dict, label: int) -> tuple[int, int, int]:
    return IDENTITY_PALETTE[int(scene["label_palette_order"][int(label)])]


def _state_color(scene: dict, state: int) -> tuple[int, int, int]:
    return STATE_PALETTE[int(scene["state_palette_order"][int(state)])]


def _draw_state_marker(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    state: int,
    color: tuple[int, int, int],
    radius: int,
) -> None:
    cx, cy = center
    radius = max(5, int(radius))
    if int(state) == 0:
        draw.ellipse(
            (cx - radius, cy - radius, cx + radius, cy + radius),
            fill=color,
            outline=INK,
            width=2,
        )
    elif int(state) == 1:
        draw.rounded_rectangle(
            (cx - radius, cy - radius, cx + radius, cy + radius),
            radius=max(2, radius // 4),
            fill=color,
            outline=INK,
            width=2,
        )
    else:
        draw.polygon(
            ((cx, cy - radius), (cx + radius, cy), (cx, cy + radius), (cx - radius, cy)),
            fill=color,
            outline=INK,
        )


def _draw_identity_key(draw: ImageDraw.ImageDraw, scene: dict, label_font, small_font) -> None:
    draw.text((IDENTITY_KEY_X, IDENTITY_KEY_Y - 31), "IDENTITY KEY", font=small_font, fill=MUTED)
    for label, name in enumerate(LABELS):
        left = IDENTITY_KEY_X + label * IDENTITY_KEY_STEP
        draw.rectangle(
            (left, IDENTITY_KEY_Y, left + IDENTITY_STRIP_W, IDENTITY_KEY_Y + 34),
            fill=_identity_color(scene, label),
            outline=INK,
            width=1,
        )
        _centered_text(draw, (left + 76, IDENTITY_KEY_Y + 17), name, label_font, INK)


def _draw_state_key(draw: ImageDraw.ImageDraw, scene: dict, label_font, small_font) -> None:
    draw.text(
        (STATE_KEY_X, STATE_KEY_Y - 31),
        "STATE KEY  (ordered LOW < MID < HIGH)",
        font=small_font,
        fill=MUTED,
    )
    for state, name in enumerate(STATE_NAMES):
        cx = STATE_KEY_X + state * STATE_KEY_STEP + 18
        cy = STATE_KEY_Y + 18
        _draw_state_marker(draw, (cx, cy), state, _state_color(scene, state), KEY_RADIUS)
        _centered_text(draw, (cx + 76, cy), name, label_font, INK)


def _draw_tile(
    draw: ImageDraw.ImageDraw,
    scene: dict,
    left: int,
    top: int,
    label: int,
    state: int,
    label_font,
) -> None:
    draw.rounded_rectangle(
        (left, top, left + TILE_W, top + TILE_H),
        radius=10,
        fill=CARD,
        outline=GRID,
        width=2,
    )
    identity_color = _identity_color(scene, label)
    draw.rounded_rectangle(
        (left, top, left + IDENTITY_STRIP_W, top + TILE_H),
        radius=8,
        fill=identity_color,
    )
    draw.rectangle(
        (left + IDENTITY_STRIP_W - 8, top, left + IDENTITY_STRIP_W, top + TILE_H),
        fill=identity_color,
    )
    _centered_text(
        draw,
        (left + IDENTITY_STRIP_W / 2, top + TILE_H / 2),
        LABELS[label],
        label_font,
        (255, 255, 255),
    )
    _centered_text(draw, (left + 135, top + TILE_H / 2), "state", label_font, MUTED)
    _draw_state_marker(
        draw,
        (left + STATE_MARKER_OFFSET_X, top + STATE_MARKER_OFFSET_Y),
        state,
        _state_color(scene, state),
        int(scene["state_marker_radius"]),
    )


def _arrow(draw: ImageDraw.ImageDraw, start: tuple[int, int], end: tuple[int, int]) -> None:
    draw.line((*start, *end), fill=MUTED, width=4)
    dx, dy = end[0] - start[0], end[1] - start[1]
    length = max(1.0, (dx * dx + dy * dy) ** 0.5)
    ux, uy = dx / length, dy / length
    px, py = -uy, ux
    bx, by = end[0] - ux * 16, end[1] - uy * 16
    draw.polygon(
        ((end[0], end[1]), (bx + px * 8, by + py * 8), (bx - px * 8, by - py * 8)),
        fill=MUTED,
    )


def _draw_order_arrows(draw: ImageDraw.ImageDraw) -> None:
    centers = [
        (PANEL_LEFTS[index] + PANEL_W // 2, PANEL_TOPS[index] + PANEL_H // 2)
        for index in range(7)
    ]
    for before, after in ((0, 1), (1, 2), (2, 3), (4, 5), (5, 6)):
        x1, y1 = centers[before]
        x2, y2 = centers[after]
        if y1 == y2:
            direction = 1 if x2 > x1 else -1
            _arrow(draw, (x1 + direction * (PANEL_W // 2 + 8), y1), (x2 - direction * (PANEL_W // 2 + 8), y2))
        else:
            direction = 1 if y2 > y1 else -1
            _arrow(draw, (x1, y1 + direction * (PANEL_H // 2 + 8)), (x2, y2 - direction * (PANEL_H // 2 + 8)))
    # P4 is above P5; these are consecutive in the visible sequence.
    x, y1 = centers[3]
    _, y2 = centers[4]
    _arrow(draw, (x, y1 + PANEL_H // 2 + 8), (x, y2 - PANEL_H // 2 - 8))


def _palindrome_score(path: list[int]) -> int:
    return sum(int(path[index] == path[-1 - index]) for index in range(len(path) // 2))


def _winner_and_gap(panel_states: list[list[int]]) -> tuple[int, int, int]:
    if len(panel_states) != 7 or any(len(panel) != len(LABELS) for panel in panel_states):
        raise ValueError("seven ordered panels with four identity states are required")
    paths = [
        [int(panel_states[panel][label]) for panel in range(7)]
        for label in range(len(LABELS))
    ]
    scores = [_palindrome_score(path) for path in paths]
    ranking = sorted(range(len(LABELS)), key=lambda label: (-scores[label], label))
    return ranking[0], scores[ranking[0]] - scores[ranking[1]], scores[ranking[0]]


def _random_path(rng: random.Random) -> list[int]:
    return [rng.randrange(len(STATE_NAMES)) for _ in range(7)]


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene with one and only one palindromic path."""
    logic_rng = random.Random(int(seed) * 1000003 + 17)
    style_rng = random.Random(int(seed) * 1000033 + 31)
    target = logic_rng.randrange(len(LABELS))

    prefix = [logic_rng.randrange(len(STATE_NAMES)) for _ in range(3)]
    center = logic_rng.randrange(len(STATE_NAMES))
    paths: list[list[int]] = [[] for _ in LABELS]
    paths[target] = prefix + [center] + list(reversed(prefix))
    for label in range(len(LABELS)):
        if label == target:
            continue
        for _ in range(1000):
            path = _random_path(logic_rng)
            if _palindrome_score(path) < 3:
                paths[label] = path
                break
        else:
            raise RuntimeError("could not sample a non-palindromic distractor path")

    panel_states = [
        [int(paths[label][panel]) for label in range(len(LABELS))]
        for panel in range(7)
    ]
    label_palette_order = list(range(len(LABELS)))
    state_palette_order = list(range(len(STATE_NAMES)))
    style_rng.shuffle(label_palette_order)
    style_rng.shuffle(state_palette_order)
    slot_orders: list[list[int]] = []
    for _ in range(7):
        order = list(range(len(LABELS)))
        style_rng.shuffle(order)
        slot_orders.append(order)
    panel_tints = [style_rng.randrange(len(PANEL_TINTS)) for _ in range(7)]
    return {
        "canvas": list(CANVAS),
        "panel_states": panel_states,
        "label_palette_order": label_palette_order,
        "state_palette_order": state_palette_order,
        "slot_orders": slot_orders,
        "panel_tints": panel_tints,
        "state_marker_radius": 15 + style_rng.randrange(5),
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", tuple(scene["canvas"]), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(30)
    section_font = _font(21)
    label_font = _font(18)
    small_font = _font(15)

    draw.text((30, 20), "STATE TRAJECTORY MIRROR", font=title_font, fill=INK)
    draw.text(
        (30, 58),
        "Follow each keyed identity through the ordered panels; compare states at equal distances from the ends.",
        font=small_font,
        fill=MUTED,
    )
    _draw_identity_key(draw, scene, label_font, small_font)
    _draw_state_key(draw, scene, label_font, small_font)
    draw.text((1325, 82), "ORDER", font=small_font, fill=MUTED)
    draw.text((1325, 106), "P1 → P2 → P3 → P4", font=section_font, fill=INK)
    draw.text((1325, 140), "P7 ← P6 ← P5", font=section_font, fill=INK)

    for panel in range(7):
        left, top = PANEL_LEFTS[panel], PANEL_TOPS[panel]
        draw.rounded_rectangle(
            (left, top, left + PANEL_W, top + PANEL_H),
            radius=16,
            fill=PANEL_TINTS[int(scene["panel_tints"][panel])],
            outline=INK,
            width=2,
        )
        _centered_text(draw, (left + PANEL_W / 2, top + 28), f"P{panel + 1}", section_font, INK)
        draw.line((left + 22, top + 52, left + PANEL_W - 22, top + 52), fill=GRID, width=1)
        for slot, (dx, dy) in enumerate(SLOT_OFFSETS):
            label = int(scene["slot_orders"][panel][slot])
            state = int(scene["panel_states"][panel][label])
            _draw_tile(draw, scene, left + dx, top + dy, label, state, label_font)
    _draw_order_arrows(draw)
    return image


def analytic_gold(scene: dict) -> str:
    winner, gap, top_score = _winner_and_gap(scene["panel_states"])
    if gap <= 0 or top_score != 3:
        raise ValueError("state trajectories do not have one unique palindrome")
    return LABELS[winner]


def margin(scene: dict) -> float:
    _winner, gap, _top_score = _winner_and_gap(scene["panel_states"])
    return float(gap)


def is_quarantined(scene: dict) -> bool:
    _winner, gap, top_score = _winner_and_gap(scene["panel_states"])
    return gap <= 0 or top_score != 3


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases are claimed; all fields are rendered or answer-bearing."""
    return []
