"""Independent pixel-only inverse for state-trajectory mirror symmetry.

The inverse shares no renderer code.  It reads the visible identity key and
state key, samples the keyed strips and markers in each panel, reconstructs all
four trajectories, and compares the three mirrored pairs.  Missing evidence is
an explicit abstention via ValueError.
"""

from __future__ import annotations

from PIL import Image


LABELS = ("A", "B", "C", "D")
STATE_NAMES = ("LOW", "MID", "HIGH")
IDENTITY_PALETTE = (
    (207, 73, 88),
    (38, 126, 180),
    (111, 82, 169),
    (43, 148, 105),
)
STATE_PALETTE = (
    (226, 124, 54),
    (40, 145, 185),
    (157, 89, 163),
)
CANVAS = (1800, 1160)
PANEL_LEFTS = (30, 470, 910, 1350, 1350, 910, 470)
PANEL_TOPS = (225, 225, 225, 225, 700, 700, 700)
SLOT_OFFSETS = ((18, 78), (18, 144), (18, 210), (18, 276))
TILE_H = 56
IDENTITY_SAMPLE_X = 6
STATE_MARKER_OFFSET_X = 280
STATE_MARKER_OFFSET_Y = TILE_H // 2
IDENTITY_KEY_X = 30
IDENTITY_KEY_Y = 130
IDENTITY_KEY_STEP = 105
STATE_KEY_X = 520
STATE_KEY_Y = 130
STATE_KEY_STEP = 200


def _pixel(frame: Image.Image, xy: tuple[int, int]) -> tuple[int, int, int]:
    value = frame.getpixel(xy)
    if not isinstance(value, tuple) or len(value) != 3:
        raise ValueError("raster is not RGB")
    return tuple(int(channel) for channel in value)


def _read_keys(
    frame: Image.Image,
) -> tuple[dict[tuple[int, int, int], int], dict[tuple[int, int, int], int]]:
    identities: dict[tuple[int, int, int], int] = {}
    for label in range(len(LABELS)):
        left = IDENTITY_KEY_X + label * IDENTITY_KEY_STEP
        color = _pixel(frame, (left + IDENTITY_SAMPLE_X, IDENTITY_KEY_Y + 7))
        if color not in IDENTITY_PALETTE or color in identities:
            raise ValueError("identity key is unreadable")
        identities[color] = label

    states: dict[tuple[int, int, int], int] = {}
    for state in range(len(STATE_NAMES)):
        cx = STATE_KEY_X + state * STATE_KEY_STEP + 18
        color = _pixel(frame, (cx, STATE_KEY_Y + 18))
        if color not in STATE_PALETTE or color in states:
            raise ValueError("state key is unreadable")
        states[color] = state
    return identities, states


def _snapshot(
    frame: Image.Image,
    identities: dict[tuple[int, int, int], int],
    states: dict[tuple[int, int, int], int],
    panel: int,
) -> list[int]:
    values: list[int | None] = [None] * len(LABELS)
    seen: set[int] = set()
    left, top = PANEL_LEFTS[panel], PANEL_TOPS[panel]
    for dx, dy in SLOT_OFFSETS:
        identity_color = _pixel(
            frame,
            (left + dx + IDENTITY_SAMPLE_X, top + dy + 7),
        )
        if identity_color not in identities:
            raise ValueError("tile identity evidence is missing")
        label = identities[identity_color]
        if label in seen:
            raise ValueError("panel repeats an identity")
        seen.add(label)
        state_color = _pixel(
            frame,
            (left + dx + STATE_MARKER_OFFSET_X, top + dy + STATE_MARKER_OFFSET_Y),
        )
        if state_color not in states:
            raise ValueError("tile state evidence is missing")
        values[label] = states[state_color]
    if seen != set(range(len(LABELS))) or any(value is None for value in values):
        raise ValueError("panel does not expose all four identities")
    return [int(value) for value in values]


def _mirror_score(path: list[int]) -> int:
    return sum(int(path[index] == path[-1 - index]) for index in range(len(path) // 2))


def decision_from_image(image: Image.Image) -> str:
    """Decode the seven panels and return the unique complete palindrome."""
    frame = image.convert("RGB")
    if frame.size != CANVAS:
        raise ValueError(f"unexpected raster size {frame.size}")
    identities, states = _read_keys(frame)
    snapshots = [_snapshot(frame, identities, states, panel) for panel in range(7)]
    paths = [
        [snapshots[panel][label] for panel in range(7)]
        for label in range(len(LABELS))
    ]
    scores = [_mirror_score(path) for path in paths]
    ranking = sorted(range(len(LABELS)), key=lambda label: (-scores[label], label))
    if scores[ranking[0]] != 3 or scores[ranking[0]] == scores[ranking[1]]:
        raise ValueError("pixel evidence has no unique complete palindrome")
    return LABELS[ranking[0]]
