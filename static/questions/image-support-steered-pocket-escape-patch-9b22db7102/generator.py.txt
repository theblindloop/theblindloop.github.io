"""Deterministic latent renderer for the pocket-escape-patch world."""

from __future__ import annotations

import random

import numpy as np
from PIL import Image

CANVAS_SIZE = 512
BACKGROUND = (247, 245, 238)
PALETTE = (
    (17, 126, 132),
    (92, 70, 173),
    (201, 83, 37),
    (35, 91, 169),
    (173, 52, 111),
)


def sample_scene(seed: int) -> dict:
    """Return a balanced scene with one connected, compact colored patch."""
    value = int(seed)
    rng = random.Random(value ^ 0x50C0_E7)
    radius = rng.choice((84, 88, 92, 96))
    pocket = round(radius * rng.choice((0.34, 0.36, 0.38)))
    channel = max(20, round(radius * rng.choice((0.24, 0.26, 0.28))))
    plug_length = max(14, round(radius * rng.choice((0.17, 0.19, 0.21))))
    secondary_offset = pocket + channel + rng.choice((2, 4, 6))
    corner = round(radius * rng.choice((0.20, 0.24, 0.28)))
    order = ["pocket", "gateway", "decoy"]
    rng.shuffle(order)
    return {
        "canvas_size": CANVAS_SIZE,
        "center_x": CANVAS_SIZE // 2 + rng.choice((-12, -8, -4, 0, 4, 8, 12)),
        "center_y": CANVAS_SIZE // 2 + rng.choice((-12, -8, -4, 0, 4, 8, 12)),
        "outer_radius": radius,
        "pocket_radius": pocket,
        "corner_radius": corner,
        "channel_width": channel,
        "plug_length": plug_length,
        "secondary_offset": secondary_offset,
        "quarter_turns": rng.randrange(4),
        "offset_sign": rng.choice((-1, 1)),
        "plug_target": "gateway" if value % 2 == 0 else "decoy",
        "cutout_order": order,
        "fill_rgb": list(rng.choice(PALETTE)),
        "background_rgb": list(BACKGROUND),
    }


def _canonical_components(scene: dict) -> tuple[np.ndarray, dict[str, np.ndarray], dict[str, np.ndarray]]:
    radius = int(scene["outer_radius"])
    pocket_radius = int(scene["pocket_radius"])
    corner = int(scene["corner_radius"])
    width = int(scene["channel_width"])
    plug_length = int(scene["plug_length"])
    secondary = int(scene["secondary_offset"]) * int(scene["offset_sign"])

    axis = np.arange(-radius, radius + 1, dtype=np.int16)
    x, y = np.meshgrid(axis, axis)
    qx = np.maximum(np.abs(x) - (radius - corner), 0)
    qy = np.maximum(np.abs(y) - (radius - corner), 0)
    outer = (qx * qx + qy * qy) <= corner * corner

    half = width // 2
    gateway_start = pocket_radius - 2
    decoy_end = -pocket_radius + 2
    components = {
        "pocket": x * x + y * y <= pocket_radius * pocket_radius,
        "gateway": (x >= gateway_start) & (np.abs(y) <= half),
        "decoy": (x <= decoy_end) & (np.abs(y - secondary) <= half),
    }
    plugs = {
        "gateway": (
            (x >= gateway_start)
            & (x < gateway_start + plug_length)
            & (np.abs(y) <= half)
        ),
        "decoy": (
            (x <= decoy_end)
            & (x > decoy_end - plug_length)
            & (np.abs(y - secondary) <= half)
        ),
    }
    return outer, components, plugs


def _foreground_mask(scene: dict) -> np.ndarray:
    outer, components, plugs = _canonical_components(scene)
    foreground = outer.copy()
    for name in scene["cutout_order"]:
        foreground &= ~components[str(name)]
    foreground |= outer & plugs[str(scene["plug_target"])]
    turns = int(scene["quarter_turns"]) % 4
    return np.rot90(foreground, turns).copy()


def render(scene: dict) -> Image.Image:
    canvas = int(scene["canvas_size"])
    background = np.asarray(scene["background_rgb"], dtype=np.uint8)
    fill = np.asarray(scene["fill_rgb"], dtype=np.uint8)
    array = np.empty((canvas, canvas, 3), dtype=np.uint8)
    array[:] = background
    mask = _foreground_mask(scene)
    height, width = mask.shape
    x0 = int(scene["center_x"]) - width // 2
    y0 = int(scene["center_y"]) - height // 2
    if x0 < 3 or y0 < 3 or x0 + width > canvas - 3 or y0 + height > canvas - 3:
        raise ValueError("patch is too close to the canvas boundary")
    view = array[y0 : y0 + height, x0 : x0 + width]
    view[mask] = fill
    return Image.fromarray(array, mode="RGB")


def analytic_gold(scene: dict) -> str:
    return "yes" if str(scene["plug_target"]) == "gateway" else "no"


def margin(scene: dict) -> float:
    """Visible seal thickness or open-passage width, in raster pixels."""
    return float(min(int(scene["plug_length"]), int(scene["channel_width"])))


def is_quarantined(scene: dict) -> bool:
    return (
        margin(scene) < 8.0
        or int(scene["pocket_radius"]) < 24
        or int(scene["secondary_offset"]) + int(scene["channel_width"]) // 2
        > int(scene["outer_radius"]) - 5
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["cutout_order"] = list(reversed([str(name) for name in scene["cutout_order"]]))
    return [("reverse_commutative_cutout_order", twin)]
