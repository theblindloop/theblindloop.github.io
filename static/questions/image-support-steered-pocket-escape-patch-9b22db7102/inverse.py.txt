"""Independent pixel-only inverse for negative-space escape connectivity.

The inverse knows only the public visual contract.  It segments the sole
chromatic area without a named palette, validates that it is one substantial
compact component, floods pale pixels from the local frame, and classifies the
large central complement component as enclosed or border-reachable.
"""

from __future__ import annotations

from collections import deque
from dataclasses import dataclass

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    """The raster is outside the unambiguous public rendering contract."""


@dataclass(frozen=True)
class Measurement:
    decision: str
    foreground_pixels: int
    foreground_fill: float
    enclosed_components: int
    enclosed_pixels: int
    foreground_bbox: tuple[int, int, int, int]


def _foreground(array: np.ndarray) -> np.ndarray:
    border = np.concatenate(
        (array[:3].reshape(-1, 3), array[-3:].reshape(-1, 3), array[:, :3].reshape(-1, 3), array[:, -3:].reshape(-1, 3)),
        axis=0,
    )
    background = np.median(border.astype(np.float32), axis=0)
    delta = np.linalg.norm(array.astype(np.float32) - background, axis=2)
    chroma = array.max(axis=2).astype(np.int16) - array.min(axis=2).astype(np.int16)
    return (delta >= 62.0) & (chroma >= 45)


def _component_sizes(mask: np.ndarray) -> list[int]:
    work = mask.copy()
    sizes: list[int] = []
    height, width = work.shape
    for sy, sx in np.argwhere(work):
        sy, sx = int(sy), int(sx)
        if not work[sy, sx]:
            continue
        work[sy, sx] = False
        queue = deque([(sy, sx)])
        size = 0
        while queue:
            y, x = queue.popleft()
            size += 1
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        sizes.append(size)
    return sorted(sizes, reverse=True)


def _exterior_background(foreground: np.ndarray) -> np.ndarray:
    background = ~foreground
    outside = np.zeros_like(background)
    height, width = background.shape
    queue: deque[tuple[int, int]] = deque()
    for x in range(width):
        for y in (0, height - 1):
            if background[y, x] and not outside[y, x]:
                outside[y, x] = True
                queue.append((y, x))
    for y in range(height):
        for x in (0, width - 1):
            if background[y, x] and not outside[y, x]:
                outside[y, x] = True
                queue.append((y, x))
    while queue:
        y, x = queue.popleft()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < height and 0 <= nx < width and background[ny, nx] and not outside[ny, nx]:
                outside[ny, nx] = True
                queue.append((ny, nx))
    return outside


def measure_image(image: Image.Image) -> Measurement:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[2] != 3 or min(array.shape[:2]) < 256:
        raise OracleAbstention("unsupported raster dimensions")
    mask = _foreground(array)
    ys, xs = np.nonzero(mask)
    if len(xs) < 18000:
        raise OracleAbstention("too little colored area")
    x0, x1 = int(xs.min()), int(xs.max()) + 1
    y0, y1 = int(ys.min()), int(ys.max()) + 1
    width, height = x1 - x0, y1 - y0
    if not (160 <= width <= 200 and 160 <= height <= 200 and abs(width - height) <= 2):
        raise OracleAbstention("colored patch does not have the declared compact extent")

    local = mask[max(0, y0 - 2) : min(mask.shape[0], y1 + 2), max(0, x0 - 2) : min(mask.shape[1], x1 + 2)]
    components = _component_sizes(local)
    if len(components) != 1:
        raise OracleAbstention("colored evidence is not one connected patch")
    fill = float(components[0] / (width * height))
    if not 0.735 <= fill <= 0.86:
        raise OracleAbstention("colored-area fraction is inconsistent with the declared patch")

    outside = _exterior_background(local)
    enclosed = (~local) & (~outside)
    hole_sizes = _component_sizes(enclosed)
    large = [size for size in hole_sizes if size >= 900]
    small_total = sum(size for size in hole_sizes if size < 900)
    if len(large) > 1 or small_total > 24:
        raise OracleAbstention("unexpected extra enclosed background regions")
    if large:
        hy, hx = np.nonzero(enclosed)
        center_distance = ((float(hx.mean()) - local.shape[1] / 2) ** 2 + (float(hy.mean()) - local.shape[0] / 2) ** 2) ** 0.5
        if center_distance > 0.22 * max(width, height):
            raise OracleAbstention("enclosed background is not the center pocket")
    decision = "yes" if len(large) == 1 else "no"
    return Measurement(
        decision=decision,
        foreground_pixels=int(mask.sum()),
        foreground_fill=fill,
        enclosed_components=len(large),
        enclosed_pixels=sum(large),
        foreground_bbox=(x0, y0, x1, y1),
    )


def decision_from_image(image: Image.Image) -> str:
    return measure_image(image).decision
