"""Independent inverse arm: receives only raster pixels."""

from __future__ import annotations

from collections import deque
import numpy as np

PURPLE = np.array((104, 72, 170), dtype=np.int16)
ORANGE = np.array((224, 116, 45), dtype=np.int16)
CYAN = np.array((24, 169, 184), dtype=np.int16)


def _components(mask: np.ndarray, minimum: int) -> list[tuple[float, float, int, tuple[int,int,int,int]]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for sy, sx in np.argwhere(mask):
        if seen[sy, sx]:
            continue
        queue = deque([(int(sy), int(sx))])
        seen[sy, sx] = True
        xs, ys = [], []
        while queue:
            y, x = queue.popleft()
            xs.append(x); ys.append(y)
            for ny, nx in ((y-1,x),(y+1,x),(y,x-1),(y,x+1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny,nx] and not seen[ny,nx]:
                    seen[ny,nx] = True
                    queue.append((ny,nx))
        if len(xs) >= minimum:
            found.append((float(np.mean(xs)), float(np.mean(ys)), len(xs),
                          (min(xs), min(ys), max(xs), max(ys))))
    return sorted(found)


def _role_components(array: np.ndarray, color: np.ndarray, minimum: int):
    distance = np.linalg.norm(array.astype(np.int16) - color, axis=2)
    return _components(distance < 38.0, minimum)


def decision_from_image(image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 200 or array.shape[1] < 400:
        raise ValueError("unexpected raster")
    zeros = _role_components(array, PURPLE, 300)
    targets = _role_components(array, ORANGE, 300)
    readings = _role_components(array, CYAN, 350)
    if len(zeros) != 2 or len(targets) != 2 or len(readings) != 2:
        raise ValueError("required gauge marks are missing or corrupted")
    # Reject fragmented/short ticks and malformed reading markers.
    for component in zeros + targets:
        x0, y0, x1, y1 = component[3]
        if x1 - x0 < 70 or y1 - y0 > 16:
            raise ValueError("malformed calibration tick")
    for component in readings:
        x0, y0, x1, y1 = component[3]
        if not (23 <= x1 - x0 <= 32 and 23 <= y1 - y0 <= 32):
            raise ValueError("malformed reading marker")

    ratios = []
    for index in range(2):
        zx, zy, *_ = zeros[index]
        tx, ty, *_ = targets[index]
        rx, ry, *_ = readings[index]
        if max(abs(zx-tx), abs(zx-rx), abs(tx-rx)) > 12:
            raise ValueError("unbound gauge marks")
        span = abs(zy - ty)
        if span < 100:
            raise ValueError("calibration span too small")
        ratios.append(abs(ry - ty) / span)
    if abs(ratios[0] - ratios[1]) < 0.045:
        raise ValueError("ambiguous normalized comparison")
    return "left" if ratios[0] < ratios[1] else "right"
