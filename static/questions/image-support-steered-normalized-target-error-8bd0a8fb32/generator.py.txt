"""Deterministic visual world with two independently calibrated gauges."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 720, 400
SCALE = 3
BG = (248, 249, 246)
PURPLE = (104, 72, 170)
ORANGE = (224, 116, 45)
CYAN = (24, 169, 184)
INK = (45, 52, 58)
LIGHT = (215, 219, 216)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    winner = "left" if int(seed) % 2 == 0 else "right"
    # The normalized winner always has the larger raw pixel gap. This makes
    # raw distance a systematically wrong shortcut, not merely a weak cue.
    win_span = rng.randint(232, 250)
    lose_span = rng.randint(136, 150)
    win_error = rng.randint(300, 330) / 1000.0
    lose_error = rng.randint(455, 495) / 1000.0
    if round(win_span * win_error) <= round(lose_span * lose_error) + 4:
        win_span, win_error = 250, 0.33
        lose_span, lose_error = 136, 0.46
    target_left = rng.randint(58, 78)
    target_right = rng.randint(58, 78)
    if winner == "left":
        left_span, left_error = win_span, win_error
        right_span, right_error = lose_span, lose_error
    else:
        left_span, left_error = lose_span, lose_error
        right_span, right_error = win_span, win_error
    return {
        "left_x": 90,
        "right_x": 630,
        "left_zero_y": target_left + left_span,
        "left_target_y": target_left,
        "left_reading_y": target_left + round(left_span * left_error),
        "right_zero_y": target_right + right_span,
        "right_target_y": target_right,
        "right_reading_y": target_right + round(right_span * right_error),
    }


def _ratios(scene: dict) -> tuple[float, float]:
    left = abs(scene["left_reading_y"] - scene["left_target_y"]) / abs(
        scene["left_zero_y"] - scene["left_target_y"]
    )
    right = abs(scene["right_reading_y"] - scene["right_target_y"]) / abs(
        scene["right_zero_y"] - scene["right_target_y"]
    )
    return left, right


def analytic_gold(scene: dict) -> str:
    left, right = _ratios(scene)
    return "left" if left < right else "right"


def margin(scene: dict) -> float:
    left, right = _ratios(scene)
    return abs(left - right)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.06


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BG)
    draw = ImageDraw.Draw(image)

    def rect(box, fill):
        draw.rectangle(tuple(int(v * SCALE) for v in box), fill=fill)

    def ellipse(box, fill, outline=None, width=1):
        draw.ellipse(
            tuple(int(v * SCALE) for v in box), fill=fill, outline=outline,
            width=width * SCALE,
        )

    # A small central legend is descriptive only and stays far from both
    # decisive gauge regions. The inverse arm ignores it.
    rect((284, 178, 436, 222), (238, 240, 237))
    rect((304, 190, 334, 198), PURPLE)
    rect((345, 190, 375, 198), ORANGE)
    ellipse((393, 184, 409, 200), CYAN)

    gauges = (
        (
            scene["left_x"], scene["left_zero_y"],
            scene["left_target_y"], scene["left_reading_y"],
        ),
        (
            scene["right_x"], scene["right_zero_y"],
            scene["right_target_y"], scene["right_reading_y"],
        ),
    )
    for x, zero, target, reading in gauges:
        rect((x - 3, target - 12, x + 3, zero + 12), LIGHT)
        # Equal-length role ticks prevent total-ink and length shortcuts.
        rect((x - 42, zero - 6, x + 42, zero + 6), PURPLE)
        rect((x - 42, target - 6, x + 42, target + 6), ORANGE)
        ellipse((x - 15, reading - 15, x + 15, reading + 15), CYAN, INK, 2)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
