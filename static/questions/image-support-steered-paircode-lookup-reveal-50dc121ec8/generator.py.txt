from __future__ import annotations

import random
from PIL import Image, ImageDraw

W = H = 512
BG = (250, 248, 243)
INK = (30, 43, 62)
PALE = (232, 228, 217)
PURPLE = (111, 65, 170)
TEAL = (24, 139, 145)

TEMPLATES = {
    "X": ("10001", "01010", "00100", "01010", "10001"),
    "plus": ("00100", "00100", "11111", "00100", "00100"),
    "T": ("11111", "00100", "00100", "00100", "00100"),
    "L": ("10000", "10000", "10000", "10000", "11111"),
}


def _flat(rows):
    return [int(ch) for row in rows for ch in row]


def _decode(key_bits, source_bits):
    return [int(key_bits[source_bits[i] * 2 + source_bits[i + 1]]) for i in range(0, 50, 2)]


def _classify(bits):
    scores = {name: sum(a != b for a, b in zip(bits, _flat(rows))) for name, rows in TEMPLATES.items()}
    ordered = sorted(scores.items(), key=lambda item: (item[1], item[0]))
    return ordered[0][0], float(ordered[1][1] - ordered[0][1])


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    glyph = tuple(TEMPLATES)[seed % len(TEMPLATES)]
    dark_codes = set(rng.sample(range(4), 2))
    key_bits = [int(i in dark_codes) for i in range(4)]
    preimages = {value: [i for i, bit in enumerate(key_bits) if bit == value] for value in (0, 1)}
    source_bits = []
    for bit in _flat(TEMPLATES[glyph]):
        code = rng.choice(preimages[bit])
        source_bits.extend([code // 2, code % 2])
    return {"glyph": glyph, "key_bits": key_bits, "source_bits": source_bits}


def render(scene: dict) -> Image.Image:
    # Access every declared field through the raster path; glyph is a redundant audit label.
    _ = scene["glyph"]
    key_bits = [int(v) for v in scene["key_bits"]]
    source_bits = [int(v) for v in scene["source_bits"]]
    im = Image.new("RGB", (W, H), BG)
    d = ImageDraw.Draw(im)

    # A detached 5 x 10 source panel. Adjacent horizontal cells form 25 codes.
    x0, y0, cell = 164, 112, 18
    d.rectangle((x0 - 6, y0 - 6, x0 + 10 * cell + 5, y0 + 5 * cell + 5), fill=PURPLE)
    for r in range(5):
        for c in range(10):
            pad = 2
            xa, ya = x0 + c * cell + pad, y0 + r * cell + pad
            d.rectangle((xa, ya, x0 + (c + 1) * cell - pad, y0 + (r + 1) * cell - pad), fill=INK if source_bits[r * 10 + c] else PALE)
    # Small pair dividers reinforce the declared left-to-right grouping.
    for c in (2, 4, 6, 8):
        x = x0 + c * cell
        d.rectangle((x - 1, y0 - 5, x + 1, y0 + 5 * cell + 4), fill=BG)

    # Four disconnected lookup cards, one for each visible input code 00,01,10,11.
    card_y, card_w, card_h, gap = 350, 42, 48, 7
    start_x = 161
    for code in range(4):
        cx = start_x + code * (card_w + gap)
        d.rectangle((cx, card_y, cx + card_w, card_y + card_h), fill=TEAL)
        d.rectangle((cx + 4, card_y + 4, cx + card_w - 4, card_y + card_h - 4), fill=BG)
        inputs = (code // 2, code % 2)
        for j, value in enumerate(inputs):
            xa = cx + 7 + j * 13
            d.rectangle((xa, card_y + 9, xa + 9, card_y + 18), fill=INK if value else PALE)
        # A visible gap separates input code from its mapped output sample.
        d.rectangle((cx + 7, card_y + 29, cx + 34, card_y + 39), fill=INK if key_bits[code] else PALE)
    return im


def analytic_gold(scene: dict) -> str:
    decision, _ = _classify(_decode(scene["key_bits"], scene["source_bits"]))
    return decision


def margin(scene: dict) -> float:
    _, gap = _classify(_decode(scene["key_bits"], scene["source_bits"]))
    return gap


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 3.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
