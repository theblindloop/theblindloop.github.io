from __future__ import annotations

import numpy as np
from PIL import Image

PURPLE = np.array((111, 65, 170), dtype=np.int16)
TEAL = np.array((24, 139, 145), dtype=np.int16)
INK = np.array((30, 43, 62), dtype=np.int16)
PALE = np.array((232, 228, 217), dtype=np.int16)
TEMPLATES = {
    "X": ("10001", "01010", "00100", "01010", "10001"),
    "plus": ("00100", "00100", "11111", "00100", "00100"),
    "T": ("11111", "00100", "00100", "00100", "00100"),
    "L": ("10000", "10000", "10000", "10000", "11111"),
}


def _near(a, color, tolerance=8):
    return np.max(np.abs(a.astype(np.int16) - color), axis=2) <= tolerance


def _components(mask):
    seen = set()
    out = []
    h, w = mask.shape
    for yy, xx in np.argwhere(mask):
        p = (int(yy), int(xx))
        if p in seen:
            continue
        stack, pts = [p], []
        seen.add(p)
        while stack:
            y, x = stack.pop(); pts.append((y, x))
            for q in ((y-1,x),(y+1,x),(y,x-1),(y,x+1)):
                if 0 <= q[0] < h and 0 <= q[1] < w and mask[q] and q not in seen:
                    seen.add(q); stack.append(q)
        out.append(pts)
    return out


def _bbox(points):
    ys = [p[0] for p in points]; xs = [p[1] for p in points]
    return min(xs), min(ys), max(xs), max(ys)


def _bit(arr, x, y):
    patch = arr[max(0,y-2):y+3, max(0,x-2):x+3].astype(np.int16)
    if patch.size == 0:
        raise ValueError("missing cell sample")
    di = np.mean(np.linalg.norm(patch - INK, axis=2))
    dp = np.mean(np.linalg.norm(patch - PALE, axis=2))
    if abs(di - dp) < 35:
        raise ValueError("ambiguous cell shade")
    return int(di < dp)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    purple = [c for c in _components(_near(arr, PURPLE)) if len(c) > 500]
    if len(purple) != 1:
        raise ValueError("source fiducial missing or split")
    x0, y0, x1, y1 = _bbox(purple[0])
    if not (170 <= x1-x0 <= 210 and 85 <= y1-y0 <= 115):
        raise ValueError("source panel geometry invalid")
    sx0, sy0 = x0 + 6, y0 + 6
    sw, sh = (x1 - x0 + 1) - 12, (y1 - y0 + 1) - 12
    source = []
    for r in range(5):
        for c in range(10):
            source.append(_bit(arr, round(sx0 + (c + .5) * sw / 10), round(sy0 + (r + .5) * sh / 5)))

    cards = []
    for comp in _components(_near(arr, TEAL)):
        if len(comp) > 250:
            box = _bbox(comp)
            if 36 <= box[2]-box[0] <= 48 and 42 <= box[3]-box[1] <= 54:
                cards.append(box)
    cards.sort()
    if len(cards) != 4:
        raise ValueError("four lookup fiducials not recovered")
    key = [None] * 4
    for x0, y0, x1, y1 in cards:
        w, h = x1-x0+1, y1-y0+1
        a = _bit(arr, round(x0 + .28*w), round(y0 + .29*h))
        b = _bit(arr, round(x0 + .59*w), round(y0 + .29*h))
        out = _bit(arr, round(x0 + .50*w), round(y0 + .71*h))
        code = 2*a+b
        if key[code] is not None:
            raise ValueError("duplicate lookup code")
        key[code] = out
    if any(v is None for v in key) or sum(key) != 2:
        raise ValueError("lookup is incomplete or unbalanced")
    decoded = [key[source[i]*2 + source[i+1]] for i in range(0, 50, 2)]
    scores = {}
    for name, rows in TEMPLATES.items():
        target = [int(ch) for row in rows for ch in row]
        scores[name] = sum(a != b for a, b in zip(decoded, target))
    ordered = sorted(scores.items(), key=lambda item: (item[1], item[0]))
    if ordered[0][1] != 0 or ordered[1][1] - ordered[0][1] < 3:
        raise ValueError("template decision is ambiguous")
    return ordered[0][0]
