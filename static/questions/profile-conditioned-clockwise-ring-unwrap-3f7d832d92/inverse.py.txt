"""Independent pixel-only inverse arm for the annular glyph task."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

N = 9
LABELS = ("circle", "triangle", "plus", "x")


def _reference_bitmaps() -> dict[str, np.ndarray]:
    refs = {label: np.zeros((N, N), dtype=np.uint8) for label in LABELS}
    coordinates = {
        "circle": [(1, 3), (1, 4), (1, 5), (2, 2), (2, 6), (3, 1), (3, 7),
                   (4, 1), (4, 7), (5, 1), (5, 7), (6, 2), (6, 6),
                   (7, 3), (7, 4), (7, 5), (4, 4)],
        "triangle": [(0, 4), (1, 3), (1, 5), (2, 2), (2, 6), (3, 2), (3, 6),
                     (4, 1), (4, 7), (5, 1), (5, 7), (6, 1), (6, 7),
                     (7, 2), (7, 4), (7, 6), (4, 4)],
        "plus": [(4, c) for c in range(9)] + [(r, 4) for r in range(9) if r != 4],
        "x": [(r, r) for r in range(9)] + [(r, 8-r) for r in range(9) if r != 4],
    }
    for label, cells in coordinates.items():
        for row, col in cells:
            refs[label][row, col] = 1
    return refs


REFERENCES = _reference_bitmaps()


def _center_from_cyan(array: np.ndarray) -> tuple[float, float]:
    mask = ((array[:, :, 0] < 70) & (array[:, :, 1] > 145) & (array[:, :, 2] > 170))
    yy, xx = np.nonzero(mask)
    if len(xx) < 10:
        raise ValueError("missing cyan center fiducial")
    return float(np.median(xx)), float(np.median(yy))


def _seam_angle(array: np.ndarray, cx: float, cy: float) -> float:
    mask = ((array[:, :, 0] > 150) & (array[:, :, 1] < 105) & (array[:, :, 2] > 90))
    yy, xx = np.nonzero(mask)
    if len(xx) < 20:
        raise ValueError("missing magenta seam")
    dx, dy = xx - cx, yy - cy
    radial = np.hypot(dx, dy)
    keep = radial > np.percentile(radial, 65)
    return math.atan2(float(np.mean(dy[keep])), float(np.mean(dx[keep])))


def _ring_bounds(array: np.ndarray, cx: float, cy: float) -> tuple[float, float]:
    # A complete annulus has high angular occupancy; thin arrows and dots do not.
    good = []
    angles = np.linspace(0.0, 2.0 * math.pi, 720, endpoint=False)
    for radius in range(30, 180):
        xx = np.rint(cx + radius * np.cos(angles)).astype(int)
        yy = np.rint(cy + radius * np.sin(angles)).astype(int)
        pixels = array[yy, xx]
        non_background = np.any(pixels < 243, axis=1)
        if float(np.mean(non_background)) > 0.68:
            good.append(radius)
    if len(good) < 35:
        raise ValueError("annular raster is missing")
    # Grid outlines can create one-radius holes; the full supported span is the
    # robust inner/outer estimate and center sampling tolerates subpixel error.
    return float(min(good) - 1), float(max(good) + 1)


def _unwrap(array: np.ndarray, cx: float, cy: float, seam: float,
            inner: float, outer: float) -> np.ndarray:
    cell = (outer - inner) / N
    values = np.zeros((N, N), dtype=np.uint8)
    for row in range(N):
        radius = outer - (row + 0.5) * cell
        for col in range(N):
            angle = seam + (col + 0.5) * (2.0 * math.pi / N)
            x = int(round(cx + radius * math.cos(angle)))
            y = int(round(cy + radius * math.sin(angle)))
            patch = array[max(0, y-1):y+2, max(0, x-1):x+2]
            values[row, col] = int(float(np.mean(patch)) < 145.0)
    return values


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    cx, cy = _center_from_cyan(array)
    seam = _seam_angle(array, cx, cy)
    inner, outer = _ring_bounds(array, cx, cy)
    bitmap = _unwrap(array, cx, cy, seam, inner, outer)
    scored = sorted((1.0 - float(np.mean(bitmap != ref)), label)
                    for label, ref in REFERENCES.items())
    best_score, best_label = scored[-1]
    runner_score = scored[-2][0]
    if best_score - runner_score <= 0.0:
        raise ValueError("ambiguous unwrapped glyph")
    return best_label
