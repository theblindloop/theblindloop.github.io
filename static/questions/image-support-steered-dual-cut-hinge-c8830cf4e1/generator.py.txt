"""Deterministic dual-topology intervention on one thick connected sheet."""

from __future__ import annotations

import copy
import math
import random
from collections import deque

from PIL import Image, ImageDraw


CANVAS_SIZE = 512
PALETTES = (
    {
        "background": [248, 250, 252],
        "sheet": [13, 100, 110],
        "patch": [245, 158, 11],
        "decoration": [203, 213, 225],
    },
    {
        "background": [250, 248, 244],
        "sheet": [30, 84, 122],
        "patch": [226, 124, 40],
        "decoration": [211, 211, 216],
    },
)


def _base_cells(opens_pockets: bool, stays_joined: bool, bridge_at_bottom: bool) -> tuple[set[tuple[int, int]], set[tuple[int, int]]]:
    """Build two thick lobes and one common amber patch on a 16x12 grid."""

    teal = {
        (x, y)
        for y in range(12)
        for x in (*range(0, 6), *range(10, 16))
    }
    amber = {(x, y) for y in range(2, 10) for x in range(5, 11)}
    teal.difference_update(amber)

    if opens_pockets:
        left_pocket = {(x, y) for y in range(2, 10) for x in range(2, 5)}
        right_pocket = {(x, y) for y in range(2, 10) for x in range(11, 14)}
    else:
        left_pocket = {(x, y) for y in range(2, 10) for x in range(1, 4)}
        right_pocket = {(x, y) for y in range(2, 10) for x in range(12, 15)}
    teal.difference_update(left_pocket | right_pocket)

    if stays_joined:
        bridge_rows = (10, 11) if bridge_at_bottom else (0, 1)
        teal.update((x, y) for y in bridge_rows for x in range(6, 10))
        notch_y = 11 if bridge_at_bottom else 0
        teal.difference_update((x, notch_y) for x in range(0, 4))
        teal.difference_update((x, notch_y) for x in range(12, 16))

    if len(teal) != 80 or len(amber) != 48 or teal & amber:
        raise RuntimeError("internal equal-ink construction invariant failed")
    return teal, amber


def _transform_cells(
    cells: set[tuple[int, int]], quarter_turns: int, mirrored: bool
) -> tuple[set[tuple[int, int]], int, int]:
    width, height = 16, 12
    points = set(cells)
    if mirrored:
        points = {(width - 1 - x, y) for x, y in points}
    for _ in range(quarter_turns % 4):
        points = {(height - 1 - y, x) for x, y in points}
        width, height = height, width
    return points, width, height


def _decorations(rng: random.Random) -> list[dict]:
    anchors = [(10, 10), (256, 10), (501, 10), (10, 256), (501, 256),
               (10, 501), (256, 501), (501, 501)]
    selected = rng.sample(anchors, 6)
    return [
        {"center": [x, y], "radius": 4 + ((index + rng.randrange(2)) % 2)}
        for index, (x, y) in enumerate(selected)
    ]


def sample_scene(seed: int) -> dict:
    """Sample balanced joint effects without storing an analytic answer bit."""

    seed = int(seed)
    rng = random.Random(seed ^ 0xD0A1C07)
    positive = seed % 2 == 0
    if positive:
        opens_pockets, stays_joined = True, False
    else:
        subtype = (seed // 2) % 3
        opens_pockets, stays_joined = ((False, False), (True, True), (False, True))[subtype]
    bridge_at_bottom = bool((seed // 6) % 2)
    quarter_turns = (seed // 12) % 4
    mirrored = bool((seed // 48) % 2)
    teal, amber = _base_cells(opens_pockets, stays_joined, bridge_at_bottom)
    teal, width, height = _transform_cells(teal, quarter_turns, mirrored)
    amber, width2, height2 = _transform_cells(amber, quarter_turns, mirrored)
    if (width, height) != (width2, height2):
        raise RuntimeError("transformed grids disagree")
    cell_size = 24 if (seed // 8) % 2 == 0 else 28
    jitter_x = rng.choice((-8, -4, 0, 4, 8))
    jitter_y = rng.choice((-8, -4, 0, 4, 8))
    origin_x = (CANVAS_SIZE - width * cell_size) // 2 + jitter_x
    origin_y = (CANVAS_SIZE - height * cell_size) // 2 + jitter_y
    if min(origin_x, origin_y, CANVAS_SIZE - origin_x - width * cell_size,
           CANVAS_SIZE - origin_y - height * cell_size) < 18:
        raise RuntimeError("sampled sheet is too close to the canvas edge")
    return {
        "canvas_size": CANVAS_SIZE,
        "cell_size": cell_size,
        "origin": [origin_x, origin_y],
        "sheet_cells": [list(point) for point in sorted(teal)],
        "patch_cells": [list(point) for point in sorted(amber)],
        "decorations": _decorations(rng),
        "palette": copy.deepcopy(PALETTES[(seed // 24) % len(PALETTES)]),
    }


def _draw_cells(draw: ImageDraw.ImageDraw, cells: list[list[int]], origin: list[int], size: int, color: tuple[int, int, int]) -> None:
    ox, oy = map(int, origin)
    for x, y in cells:
        x0, y0 = ox + int(x) * size, oy + int(y) * size
        draw.rectangle((x0, y0, x0 + size - 1, y0 + size - 1), fill=color)


def render(scene: dict) -> Image.Image:
    """Render crisp, wide cells so ordinary raster perturbations cannot change topology."""

    canvas_size = int(scene["canvas_size"])
    cell_size = int(scene["cell_size"])
    origin = scene["origin"]
    sheet_cells = scene["sheet_cells"]
    patch_cells = scene["patch_cells"]
    palette = scene["palette"]
    decorations = scene["decorations"]
    image = Image.new("RGB", (canvas_size, canvas_size), tuple(palette["background"]))
    draw = ImageDraw.Draw(image)
    _draw_cells(draw, sheet_cells, origin, cell_size, tuple(palette["sheet"]))
    _draw_cells(draw, patch_cells, origin, cell_size, tuple(palette["patch"]))
    for item in decorations:
        x, y = map(int, item["center"])
        radius = int(item["radius"])
        draw.ellipse((x - radius, y - radius, x + radius, y + radius),
                     fill=tuple(palette["decoration"]))
    return image


def _components(cells: set[tuple[int, int]]) -> int:
    remaining = set(cells)
    count = 0
    while remaining:
        count += 1
        start = remaining.pop()
        queue = deque([start])
        while queue:
            x, y = queue.popleft()
            for neighbor in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
                if neighbor in remaining:
                    remaining.remove(neighbor)
                    queue.append(neighbor)
    return count


def _bounded_background_count(solid: set[tuple[int, int]]) -> int:
    max_x = max(x for x, _ in solid)
    max_y = max(y for _, y in solid)
    domain = {(x, y) for y in range(-1, max_y + 2) for x in range(-1, max_x + 2)}
    white = domain - solid
    remaining = set(white)
    bounded = 0
    while remaining:
        start = remaining.pop()
        queue = deque([start])
        touches_border = False
        while queue:
            x, y = queue.popleft()
            if x in {-1, max_x + 1} or y in {-1, max_y + 1}:
                touches_border = True
            for neighbor in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
                if neighbor in remaining:
                    remaining.remove(neighbor)
                    queue.append(neighbor)
        bounded += int(not touches_border)
    return bounded


def _topology(scene: dict) -> tuple[int, int, int, int]:
    sheet = {tuple(map(int, point)) for point in scene["sheet_cells"]}
    patch = {tuple(map(int, point)) for point in scene["patch_cells"]}
    solid = sheet | patch
    return (
        _components(solid),
        _components(sheet),
        _bounded_background_count(solid),
        _bounded_background_count(sheet),
    )


def analytic_gold(scene: dict) -> str:
    before_solid, after_solid, before_pockets, after_pockets = _topology(scene)
    return "yes" if after_solid > before_solid and after_pockets < before_pockets else "no"


def margin(scene: dict) -> float:
    return float(scene["cell_size"])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 22.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["decorations"] = list(reversed(twin["decorations"]))
    return [("reverse_disjoint_decoration_draw_order", twin)]
