"""Independent pixel-only inverse for the phase-slip raster question.

This module deliberately contains no import of the renderer or any latent-scene
code.  It locates the colored fiducials in the PNG, infers the enclosed raster,
undoes the public row registration rule, smooths the restored pixels, and uses
coarse visual template correlation with a reported runner-up margin.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image, ImageDraw

_TILE = 256
_MARK_GAP = 6
_MARK_SIZE = 18
_ROW_STEP = 3
_SYMBOLS = ("ring", "cross", "triangle", "zigzag")


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    boxes: list[tuple[int, int, int, int]] = []
    for y in range(height):
        for x in range(width):
            if not mask[y, x] or seen[y, x]:
                continue
            queue = deque([(y, x)])
            seen[y, x] = True
            min_x = max_x = x
            min_y = max_y = y
            count = 0
            while queue:
                cy, cx = queue.popleft()
                count += 1
                min_x, max_x = min(min_x, cx), max(max_x, cx)
                min_y, max_y = min(min_y, cy), max(max_y, cy)
                for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
            if count >= 80:
                boxes.append((min_x, min_y, max_x, max_y))
    return boxes


def _locate_tile(rgb: np.ndarray) -> tuple[int, int, int, int]:
    spread = rgb.max(axis=2).astype(np.int16) - rgb.min(axis=2).astype(np.int16)
    colorful = (spread > 50) & (rgb.max(axis=2) > 120)
    boxes = _components(colorful)
    if len(boxes) != 4:
        raise ValueError(f"expected four colored fiducials, found {len(boxes)}")
    centers = [((a + c) / 2.0, (b + d) / 2.0, a, b, c, d) for a, b, c, d in boxes]
    left = sorted(centers, key=lambda item: item[0])[:2]
    right = sorted(centers, key=lambda item: item[0])[2:]
    top = sorted(centers, key=lambda item: item[1])[:2]
    bottom = sorted(centers, key=lambda item: item[1])[2:]
    tile_x = max(int(item[4]) for item in left) + _MARK_GAP + 1
    tile_y = max(int(item[5]) for item in top) + _MARK_GAP + 1
    right_edge = min(int(item[2]) for item in right)
    bottom_edge = min(int(item[3]) for item in bottom)
    tile_w = right_edge - _MARK_GAP - tile_x
    tile_h = bottom_edge - _MARK_GAP - tile_y
    if (tile_w, tile_h) != (_TILE, _TILE):
        raise ValueError(f"fiducials enclose unexpected raster {(tile_w, tile_h)}")
    return tile_x, tile_y, tile_w, tile_h


def _restore_rows(raw: np.ndarray) -> np.ndarray:
    restored = np.empty_like(raw, dtype=np.float32)
    for row in range(_TILE):
        shift = (_ROW_STEP * row) % _TILE
        restored[row] = np.roll(raw[row], -shift)
    return restored


def _edge_box_mean(values: np.ndarray, size: int = 7) -> np.ndarray:
    radius = size // 2
    padded = np.pad(values, ((radius, radius), (radius, radius)), mode="edge")
    integral = np.pad(padded, ((1, 0), (1, 0)), mode="constant").cumsum(0).cumsum(1)
    total = (
        integral[size:, size:]
        - integral[:-size, size:]
        - integral[size:, :-size]
        + integral[:-size, :-size]
    )
    return total / float(size * size)


def _glyph_template(symbol: str, scale: float) -> np.ndarray:
    """Build a coarse visual template independently at 64-pixel resolution."""
    canvas = Image.new("L", (64, 64), 0)
    draw = ImageDraw.Draw(canvas)
    cx = cy = 32
    if symbol == "ring":
        outer = int(round(17.0 * scale))
        inner = int(round(11.0 * scale))
        draw.ellipse((cx - outer, cy - outer, cx + outer, cy + outer), fill=255)
        draw.ellipse((cx - inner, cy - inner, cx + inner, cy + inner), fill=0)
    elif symbol == "cross":
        half = int(round(20.0 * scale))
        half_width = max(3, int(round(3.25 * scale)))
        draw.rectangle((cx - half, cy - half_width, cx + half, cy + half_width), fill=255)
        draw.rectangle((cx - half_width, cy - half, cx + half_width, cy + half), fill=255)
    elif symbol == "triangle":
        half_width = int(round(17.5 * scale))
        top = int(round(17.5 * scale))
        bottom = int(round(13.75 * scale))
        draw.polygon(
            [(cx, cy - top), (cx - half_width, cy + bottom), (cx + half_width, cy + bottom)],
            fill=255,
        )
    elif symbol == "zigzag":
        span = int(round(20.5 * scale))
        rise = int(round(11.25 * scale))
        points = [
            (cx - span, cy + rise),
            (cx - span // 2, cy - rise),
            (cx, cy + rise),
            (cx + span // 2, cy - rise),
            (cx + span, cy + rise),
        ]
        draw.line(points, fill=255, width=max(3, int(round(5.0 * scale))), joint="curve")
    else:
        raise ValueError(symbol)
    return np.asarray(canvas, dtype=np.float32) / 255.0


def _shift_template(template: np.ndarray, dx: int, dy: int) -> np.ndarray:
    shifted = np.zeros_like(template)
    source_x0 = max(0, -dx)
    source_x1 = min(64, 64 - dx)
    source_y0 = max(0, -dy)
    source_y1 = min(64, 64 - dy)
    target_x0 = max(0, dx)
    target_x1 = target_x0 + max(0, source_x1 - source_x0)
    target_y0 = max(0, dy)
    target_y1 = target_y0 + max(0, source_y1 - source_y0)
    if source_x1 > source_x0 and source_y1 > source_y0:
        shifted[target_y0:target_y1, target_x0:target_x1] = template[source_y0:source_y1, source_x0:source_x1]
    return shifted


def _classify(restored: np.ndarray) -> tuple[str, float]:
    smooth = _edge_box_mean(restored, 7)
    low, high = np.percentile(smooth, (10.0, 90.0))
    contrast = float(high - low)
    # A few small glyphs at the quarantined 34-unit signal level have a lower
    # percentile spread even though their template margin is large.  Keep the
    # floor above the carrier-only blank-image spread while admitting those
    # explicitly quarantined but still decodable cases.
    if contrast < 10.0:
        raise ValueError("restored raster has insufficient tonal evidence")
    baseline = float(np.percentile(smooth, 78.0))
    darkness = np.clip((baseline - smooth) / max(20.0, contrast * 0.82), 0.0, 1.0)
    coarse = darkness.reshape(64, 4, 64, 4).mean(axis=(1, 3))
    energy = float(np.sqrt(np.square(coarse).sum()))
    if energy < 5.0:
        raise ValueError("restored raster has insufficient symbol energy")
    scores: list[tuple[float, str]] = []
    for symbol in _SYMBOLS:
        for scale in (0.90, 1.00, 1.10):
            template = _glyph_template(symbol, scale)
            for dy in range(-6, 7):
                for dx in range(-6, 7):
                    shifted = _shift_template(template, dx, dy)
                    denom = float(np.sqrt(np.square(shifted).sum()) * energy)
                    score = float(np.multiply(coarse, shifted).sum()) / denom if denom else -1.0
                    scores.append((score, symbol))
    scores.sort(reverse=True)
    best_score, best_symbol = scores[0]
    runner_score = max(score for score, symbol in scores if symbol != best_symbol)
    decision_margin = best_score - runner_score
    if best_score < 0.27:
        raise ValueError("restored symbol correlation is too weak")
    if decision_margin < 0.008:
        raise ValueError("restored symbol is ambiguous")
    return best_symbol, float(decision_margin)


def _measure(image: Image.Image) -> tuple[str, float]:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    tile_x, tile_y, width, height = _locate_tile(rgb)
    raw = rgb[tile_y : tile_y + height, tile_x : tile_x + width, 0].astype(np.float32)
    restored = _restore_rows(raw)
    return _classify(restored)


def decision_from_image(image: Image.Image) -> str:
    """Return the symbol recovered from the final PNG alone."""
    decision, _ = _measure(image)
    return decision


def confidence_margin(image: Image.Image) -> float:
    """Return the positive winner-versus-runner-up margin, or abstain."""
    _, margin = _measure(image)
    return margin
