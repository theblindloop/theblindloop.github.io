"""Renderer and analytic world definition for the phase-slip raster question."""

from __future__ import annotations

import math
import random
from pathlib import Path

import numpy as np
from PIL import Image, ImageDraw

RENDERER_VERSION = "phase-slip-raster-reveal-0.1.0"
WIDTH, HEIGHT = 640, 440
TILE_SIZE = 256
MARK_SIZE = 18
MARK_GAP = 6
ROW_STEP = 3
SMOOTH_SIZE = 7
SYMBOLS = ("ring", "cross", "triangle", "zigzag")
BACKGROUND = (247, 244, 236)
FRAME = (72, 70, 66)
MARKER_COLORS = (
    (224, 67, 104),
    (38, 145, 201),
    (62, 165, 93),
    (184, 72, 196),
    (238, 153, 39),
)
SIGNAL_LEVELS = (34, 40, 48, 56, 64)
GLYPH_SCALES = (0.90, 1.00, 1.10)


def _marker_boxes(tile_x: int, tile_y: int) -> list[tuple[int, int, int, int]]:
    left = tile_x - MARK_GAP - MARK_SIZE
    top = tile_y - MARK_GAP - MARK_SIZE
    right = tile_x + TILE_SIZE + MARK_GAP
    bottom = tile_y + TILE_SIZE + MARK_GAP
    return [
        (left, top, left + MARK_SIZE - 1, top + MARK_SIZE - 1),
        (right, top, right + MARK_SIZE - 1, top + MARK_SIZE - 1),
        (left, bottom, left + MARK_SIZE - 1, bottom + MARK_SIZE - 1),
        (right, bottom, right + MARK_SIZE - 1, bottom + MARK_SIZE - 1),
    ]


def _draw_glyph(symbol: str, dx: int, dy: int, scale: float) -> np.ndarray:
    """Draw one broad, non-textual glyph into a binary tile mask."""
    mask = Image.new("L", (TILE_SIZE, TILE_SIZE), 0)
    draw = ImageDraw.Draw(mask)
    cx = TILE_SIZE // 2 + int(dx)
    cy = TILE_SIZE // 2 + int(dy)
    if symbol == "ring":
        outer = int(round(68 * scale))
        inner = int(round(44 * scale))
        draw.ellipse((cx - outer, cy - outer, cx + outer, cy + outer), fill=255)
        draw.ellipse((cx - inner, cy - inner, cx + inner, cy + inner), fill=0)
    elif symbol == "cross":
        half = int(round(80 * scale))
        half_width = max(8, int(round(13 * scale)))
        draw.rectangle((cx - half, cy - half_width, cx + half, cy + half_width), fill=255)
        draw.rectangle((cx - half_width, cy - half, cx + half_width, cy + half), fill=255)
    elif symbol == "triangle":
        half_width = int(round(70 * scale))
        top = int(round(70 * scale))
        bottom = int(round(55 * scale))
        draw.polygon(
            [(cx, cy - top), (cx - half_width, cy + bottom), (cx + half_width, cy + bottom)],
            fill=255,
        )
    elif symbol == "zigzag":
        span = int(round(82 * scale))
        rise = int(round(45 * scale))
        points = [
            (cx - span, cy + rise),
            (cx - span // 2, cy - rise),
            (cx, cy + rise),
            (cx + span // 2, cy - rise),
            (cx + span, cy + rise),
        ]
        draw.line(points, fill=255, width=max(10, int(round(20 * scale))), joint="curve")
    else:
        raise ValueError(f"unknown symbol {symbol!r}")
    return np.asarray(mask, dtype=np.float32) / 255.0


def _carrier(scene: dict) -> np.ndarray:
    yy, xx = np.indices((TILE_SIZE, TILE_SIZE), dtype=np.int32)
    phase = int(scene["texture_phase"]) % 8
    mode = int(scene["carrier_mode"]) % 3
    if mode == 0:
        phase_field = xx + phase
    elif mode == 1:
        phase_field = xx + yy + phase
    else:
        phase_field = 2 * xx + yy + phase
    signs = np.where(np.mod(phase_field, 8) < 4, 1.0, -1.0)
    return signs * float(scene["carrier_amplitude"])


def sample_scene(seed: int) -> dict:
    """Sample a JSON-serializable scene with nuisance fields independent of label."""
    seed = int(seed)
    label_rng = random.Random(seed * 104729 + 17)
    nuisance_rng = random.Random(seed * 130363 + 7919)
    symbol = SYMBOLS[seed % len(SYMBOLS)]
    tile_x = 90 + nuisance_rng.randrange(0, 250)
    tile_y = 68 + nuisance_rng.randrange(0, 80)
    marker_color = list(MARKER_COLORS[nuisance_rng.randrange(len(MARKER_COLORS))])
    signal_strength = SIGNAL_LEVELS[nuisance_rng.randrange(len(SIGNAL_LEVELS))]
    glyph_scale = GLYPH_SCALES[nuisance_rng.randrange(len(GLYPH_SCALES))]
    return {
        "seed": seed,
        "symbol": symbol,
        "texture_phase": nuisance_rng.randrange(8),
        "carrier_mode": nuisance_rng.randrange(3),
        "carrier_amplitude": nuisance_rng.choice((20, 24, 28)),
        "base_level": nuisance_rng.choice((112, 128, 144)),
        "signal_strength": signal_strength,
        "marker_color": marker_color,
        "tile_x": tile_x,
        "tile_y": tile_y,
        "symbol_dx": nuisance_rng.randrange(-18, 19),
        "symbol_dy": nuisance_rng.randrange(-18, 19),
        "glyph_scale": glyph_scale,
    }


def _record_rows(canonical: np.ndarray) -> np.ndarray:
    recorded = np.empty_like(canonical, dtype=np.uint8)
    for row in range(TILE_SIZE):
        shift = (ROW_STEP * row) % TILE_SIZE
        recorded[row] = np.roll(canonical[row], shift)
    return recorded


def render(scene: dict) -> Image.Image:
    """Render only from the scene fields; output is a deterministic RGB PNG source."""
    symbol = str(scene["symbol"])
    base = float(scene["base_level"])
    signal = float(scene["signal_strength"])
    canonical = base + _carrier(scene)
    glyph = _draw_glyph(symbol, int(scene["symbol_dx"]), int(scene["symbol_dy"]), float(scene["glyph_scale"]))
    canonical = np.clip(canonical - signal * glyph, 0.0, 255.0)
    recorded = _record_rows(np.rint(canonical).astype(np.uint8))

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    tile_x, tile_y = int(scene["tile_x"]), int(scene["tile_y"])
    image.paste(Image.fromarray(recorded, mode="L").convert("RGB"), (tile_x, tile_y))
    draw = ImageDraw.Draw(image)
    draw.rectangle(
        (tile_x - 2, tile_y - 2, tile_x + TILE_SIZE + 1, tile_y + TILE_SIZE + 1),
        outline=FRAME,
        width=2,
    )
    color = tuple(int(value) for value in scene["marker_color"])
    for box in _marker_boxes(tile_x, tile_y):
        draw.rectangle(box, fill=color)
    return image


def analytic_gold(scene: dict) -> str:
    """The answer is the declared abstract symbol, not a renderer-side image feature."""
    return str(scene["symbol"])


def margin(scene: dict) -> float:
    """Signed signal-to-boundary margin used for gallery difficulty reporting."""
    return float(scene["signal_strength"]) - 36.0


def is_quarantined(scene: dict) -> bool:
    return float(scene["signal_strength"]) <= 36.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Expose the carrier's tested eight-pixel phase alias."""
    twin = dict(scene)
    twin["texture_phase"] = int(scene["texture_phase"]) + 8
    return [("carrier_period_alias", twin)]

    out.write_bytes(b"")
    render(sample_scene(101)).save(out)
