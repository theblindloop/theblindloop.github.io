"""Deterministic renderer and analytic world for Euler-trail selection."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 660, 320
CANVAS = (WIDTH, HEIGHT)
BACKGROUND = (255, 255, 255)
LINK = (47, 55, 67)
COLORS = {
    "red": (214, 75, 87),
    "blue": (61, 111, 208),
    "green": (50, 159, 103),
}
COLOR_NAMES = tuple(COLORS)
NODE_RADIUS = 10.0
LINK_WIDTH = 6.0
SUPERSAMPLE = 4
SLOT_CENTERS = ((110.0, 160.0), (330.0, 160.0), (550.0, 160.0))

# The two graph families use the same five positions and six links.  The first
# is a pentagon plus one chord (two odd nodes); the second is a four-edge path
# plus two chords (four odd nodes).  Both are planar in the convex-pentagon
# embedding and connected.
EULER_EDGES = ((0, 1), (1, 2), (2, 3), (3, 4), (4, 0), (0, 2))
NON_EULER_EDGES = ((0, 1), (1, 2), (2, 3), (3, 4), (0, 2), (0, 3))


def _base_nodes() -> tuple[tuple[float, float], ...]:
    return tuple(
        (
            math.cos(-math.pi / 2.0 + 2.0 * math.pi * index / 5.0),
            math.sin(-math.pi / 2.0 + 2.0 * math.pi * index / 5.0),
        )
        for index in range(5)
    )


def _edge_length(nodes: Iterable[tuple[float, float]], edge: tuple[int, int]) -> float:
    points = tuple(nodes)
    a, b = points[edge[0]], points[edge[1]]
    return math.hypot(a[0] - b[0], a[1] - b[1])


def _total_edge_length(nodes: tuple[tuple[float, float], ...], edges: tuple[tuple[int, int], ...]) -> float:
    return sum(_edge_length(nodes, edge) for edge in edges)


def _transformed_nodes(
    center: tuple[float, float],
    scale: float,
    rotation: float,
) -> list[list[float]]:
    base = _base_nodes()
    cos_theta, sin_theta = math.cos(rotation), math.sin(rotation)
    output: list[list[float]] = []
    for x, y in base:
        xr = cos_theta * x - sin_theta * y
        yr = sin_theta * x + cos_theta * y
        output.append([round(center[0] + scale * xr, 3), round(center[1] + scale * yr, 3)])
    return output


def _degrees(node_count: int, edges: Iterable[tuple[int, int]]) -> list[int]:
    degrees = [0] * node_count
    for left, right in edges:
        degrees[left] += 1
        degrees[right] += 1
    return degrees


def _connected(node_count: int, edges: Iterable[tuple[int, int]]) -> bool:
    adjacency = [[] for _ in range(node_count)]
    for left, right in edges:
        adjacency[left].append(right)
        adjacency[right].append(left)
    seen = {0}
    pending = [0]
    while pending:
        node = pending.pop()
        for neighbor in adjacency[node]:
            if neighbor not in seen:
                seen.add(neighbor)
                pending.append(neighbor)
    return len(seen) == node_count


def _euler_score(node_count: int, edge_values: Iterable[tuple[int, int]]) -> float:
    edges = [tuple(edge) for edge in edge_values]
    if node_count != 5 or len(edges) != 6:
        return 0.0
    odd = sum(degree % 2 for degree in _degrees(node_count, edges))
    return 1.0 if _connected(node_count, edges) and odd in (0, 2) else 0.0


def sample_scene(seed: int) -> dict:
    """Sample one balanced scene; the seed is the only source of randomness."""
    rng = random.Random(int(seed))
    base = _base_nodes()
    target_link_length = rng.uniform(300.0, 340.0)
    scales = {
        "euler": target_link_length / _total_edge_length(base, EULER_EDGES),
        "non_euler": target_link_length / _total_edge_length(base, NON_EULER_EDGES),
    }
    winner = COLOR_NAMES[int(seed) % len(COLOR_NAMES)]
    slots = list(SLOT_CENTERS)
    rng.shuffle(slots)
    graphs: list[dict] = []
    for color, center in zip(COLOR_NAMES, slots):
        dx = rng.uniform(-9.0, 9.0)
        dy = rng.uniform(-8.0, 8.0)
        rotation = rng.uniform(-math.pi, math.pi)
        kind = "euler" if color == winner else "non_euler"
        edges = EULER_EDGES if kind == "euler" else NON_EULER_EDGES
        graph_center = (center[0] + dx, center[1] + dy)
        graphs.append(
            {
                "color": color,
                "nodes": _transformed_nodes(graph_center, scales[kind], rotation),
                "edges": [list(edge) for edge in edges],
            }
        )
    rng.shuffle(graphs)
    return {"graphs": graphs}


def _draw_graph(draw: ImageDraw.ImageDraw, nodes: list[list[float]], edges: list[list[int]], color: tuple[int, int, int], ss: int) -> None:
    for left, right in edges:
        draw.line(
            [
                (round(nodes[left][0] * ss), round(nodes[left][1] * ss)),
                (round(nodes[right][0] * ss), round(nodes[right][1] * ss)),
            ],
            fill=LINK,
            width=round(LINK_WIDTH * ss),
        )
    radius = NODE_RADIUS * ss
    for x, y in nodes:
        cx, cy = x * ss, y * ss
        draw.ellipse(
            (round(cx - radius), round(cy - radius), round(cx + radius), round(cy + radius)),
            fill=color,
        )


def render(scene: dict) -> Image.Image:
    ss = SUPERSAMPLE
    image = Image.new("RGB", (WIDTH * ss, HEIGHT * ss), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for graph in scene["graphs"]:
        _draw_graph(draw, graph["nodes"], graph["edges"], COLORS[graph["color"]], ss)
    return image.resize(CANVAS, Image.Resampling.LANCZOS).convert("RGB")


def analytic_gold(scene: dict) -> str:
    scored = []
    for graph in scene["graphs"]:
        edges = [tuple(edge) for edge in graph["edges"]]
        scored.append((graph["color"], _euler_score(len(graph["nodes"]), edges)))
    winners = [color for color, score in scored if score == 1.0]
    if len(winners) != 1:
        raise ValueError(f"expected one Euler-trail winner, got {winners}")
    return winners[0]


def margin(scene: dict) -> float:
    scores = sorted(
        (
            _euler_score(len(graph["nodes"]), [tuple(edge) for edge in graph["edges"]])
            for graph in scene["graphs"]
        ),
        reverse=True,
    )
    if len(scores) < 2:
        return 0.0
    return float(scores[0] - scores[1])


def _minimum_node_separation(nodes: list[list[float]]) -> float:
    return min(
        math.dist(nodes[left], nodes[right])
        for left in range(len(nodes))
        for right in range(left + 1, len(nodes))
    )


def is_quarantined(scene: dict) -> bool:
    try:
        graphs = scene["graphs"]
        if len(graphs) != 3:
            return True
        colors = [graph["color"] for graph in graphs]
        if set(colors) != set(COLOR_NAMES):
            return True
        if margin(scene) < 1.0:
            return True
        for graph in graphs:
            nodes = graph["nodes"]
            edges = [tuple(edge) for edge in graph["edges"]]
            if len(nodes) != 5 or len(edges) != 6:
                return True
            if any(not (0.0 <= x < WIDTH and 0.0 <= y < HEIGHT) for x, y in nodes):
                return True
            if _minimum_node_separation(nodes) < 24.0:
                return True
            if any(math.dist(nodes[a], nodes[b]) < 27.0 for a, b in edges):
                return True
        return False
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases: all graph fields that affect gold are rendered."""
    return []
