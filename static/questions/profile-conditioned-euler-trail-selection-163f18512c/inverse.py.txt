"""Independent pixel-only inverse for the Euler-trail network world."""

from __future__ import annotations

import collections
import math

import numpy as np
from PIL import Image

_COLORS = {
    "red": np.array([214.0, 75.0, 87.0]),
    "blue": np.array([61.0, 111.0, 208.0]),
    "green": np.array([50.0, 159.0, 103.0]),
}
_LINK = np.array([47.0, 55.0, 67.0])
_NODE_RADIUS = 10.0


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    found: list[np.ndarray] = []
    for y, x in zip(*np.nonzero(mask)):
        if visited[y, x]:
            continue
        queue = collections.deque([(int(y), int(x))])
        visited[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not dx and not dy:
                        continue
                    ny, nx = cy + dy, cx + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                        visited[ny, nx] = True
                        queue.append((ny, nx))
        if len(points) >= 35:
            found.append(np.asarray(points, dtype=np.float64))
    return found


def _node_centers(arr: np.ndarray, color: np.ndarray) -> np.ndarray:
    distances = np.linalg.norm(arr - color, axis=2)
    # The dark link palette is close to green in RGB distance, so keep a
    # conservative color-core threshold; supersampled node interiors provide
    # ample area even after antialiasing.
    mask = distances < 70.0
    components = _components(mask)
    if len(components) != 5:
        raise ValueError(f"expected five node components, found {len(components)}")
    centers = []
    for component in components:
        centers.append([float(component[:, 1].mean()) + 0.5, float(component[:, 0].mean()) + 0.5])
    return np.asarray(centers, dtype=np.float64)


def _link_mask(arr: np.ndarray) -> np.ndarray:
    return np.linalg.norm(arr - _LINK, axis=2) < 108.0


def _line_support(mask: np.ndarray, first: np.ndarray, second: np.ndarray) -> float:
    delta = second - first
    length = float(np.linalg.norm(delta))
    if length < 25.0:
        return 0.0
    unit = delta / length
    normal = np.array([-unit[1], unit[0]])
    values = []
    for fraction in np.linspace(0.16, 0.84, 80):
        point = first + fraction * delta
        hit = False
        for offset in np.linspace(-5.0, 5.0, 9):
            probe = point + offset * normal
            x, y = int(round(probe[0])), int(round(probe[1]))
            if 0 <= x < mask.shape[1] and 0 <= y < mask.shape[0] and mask[y, x]:
                hit = True
                break
        values.append(hit)
    return float(np.mean(values))


def _recover_edges(centers: np.ndarray, links: np.ndarray) -> list[tuple[int, int]]:
    scored = []
    for first in range(len(centers)):
        for second in range(first + 1, len(centers)):
            scored.append((_line_support(links, centers[first], centers[second]), first, second))
    scored.sort(reverse=True)
    selected = [(first, second) for support, first, second in scored[:6] if support >= 0.52]
    if len(selected) != 6 or scored[5][0] - scored[6][0] < 0.10:
        raise ValueError("link support is ambiguous")
    return selected


def _is_euler(centers: np.ndarray, edges: list[tuple[int, int]]) -> bool:
    if len(centers) != 5 or len(edges) != 6:
        return False
    adjacency = [[] for _ in range(5)]
    degrees = [0] * 5
    for first, second in edges:
        adjacency[first].append(second)
        adjacency[second].append(first)
        degrees[first] += 1
        degrees[second] += 1
    seen = {0}
    pending = [0]
    while pending:
        node = pending.pop()
        for neighbor in adjacency[node]:
            if neighbor not in seen:
                seen.add(neighbor)
                pending.append(neighbor)
    return len(seen) == 5 and sum(degree % 2 for degree in degrees) in (0, 2)


def decision_from_image(image: Image.Image) -> str:
    """Recover the colored network with an Euler trail from pixels alone."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float64)
    links = _link_mask(arr)
    answers = []
    for name, color in _COLORS.items():
        centers = _node_centers(arr, color)
        edges = _recover_edges(centers, links)
        if _is_euler(centers, edges):
            answers.append(name)
    if len(answers) != 1:
        raise ValueError(f"expected one Euler-trail network, recovered {answers}")
    return answers[0]
