"""Independent pixel-only inverse for the split-panel identity handoff."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "state_split_identity-oracle-0.1.0"
IDENTITY_COLORS = np.asarray(
    ((35, 88, 150), (175, 62, 56), (48, 130, 94), (126, 72, 145)),
    dtype=np.float32,
)
STATE_COLORS = np.asarray(
    ((238, 190, 48), (236, 108, 48), (91, 180, 92), (58, 157, 216)),
    dtype=np.float32,
)
RING_COLOR = np.asarray((240, 50, 160), dtype=np.float32)


def _near(array: np.ndarray, color: np.ndarray, tolerance: float) -> np.ndarray:
    return np.linalg.norm(array - color, axis=2) <= tolerance


def _centroid(mask: np.ndarray) -> tuple[np.ndarray, int]:
    points = np.argwhere(mask)
    if len(points) == 0:
        raise ValueError("required visual component is absent")
    return points[:, ::-1].mean(axis=0), len(points)


def _identity_centers(array: np.ndarray, left: bool) -> list[tuple[int, np.ndarray, int]]:
    height, width = array.shape[:2]
    x0, x1 = (0, width // 2) if left else (width // 2, width)
    region = array.copy()
    region[:, :x0] = 255
    region[:, x1:] = 255
    found = []
    for identity, color in enumerate(IDENTITY_COLORS):
        mask = _near(region, color, 38.0)
        center, count = _centroid(mask)
        if count < 250:
            raise ValueError("identity disk is too damaged to bind")
        found.append((identity, center, count))
    return found


def decision_from_image(image: Image.Image) -> str:
    """Recover the selected identity on the left and its badge on the right."""
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    height, width = array.shape[:2]
    ring_mask = _near(array, RING_COLOR, 42.0)
    ring_mask[:, width // 2 :] = False
    ring_center, ring_count = _centroid(ring_mask)
    # A complete ring is the left-hand causal support. Requiring most of its
    # ink makes grid ablation of the marker produce an honest abstention rather
    # than letting a surviving arc stand in for the initial identity.
    if ring_count < 950:
        raise ValueError("initial identity ring is missing or ambiguous")

    initial = _identity_centers(array, left=True)
    identity, _, _ = min(initial, key=lambda item: math.hypot(*(item[1] - ring_center)))
    if math.hypot(*(min(initial, key=lambda item: math.hypot(*(item[1] - ring_center)))[1] - ring_center)) > 55.0:
        raise ValueError("ring does not enclose a visible identity disk")

    final = _identity_centers(array, left=False)
    target_center = next(center for item, center, _ in final if item == identity)
    local_y, local_x = np.indices((height, width))
    local = (local_x - target_center[0]) ** 2 + (local_y - target_center[1]) ** 2 <= 23.0**2
    scores = [int(np.count_nonzero(local & _near(array, color, 55.0))) for color in STATE_COLORS]
    state, score = max(enumerate(scores), key=lambda item: item[1])
    if score < 90 or score < max(scores) * 0.55:
        raise ValueError("terminal state badge is absent or ambiguous")
    return ("A", "B", "C", "D")[state]
