"""Renderer and analytic arm for the split-panel identity handoff world."""

from __future__ import annotations

import math
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

RENDERER_VERSION = "state_split_identity-renderer-0.1.0"
WIDTH = 512
HEIGHT = 384
IDENTITY_COLORS = (
    (35, 88, 150),
    (175, 62, 56),
    (48, 130, 94),
    (126, 72, 145),
)
STATE_COLORS = (
    (238, 190, 48),
    (236, 108, 48),
    (91, 180, 92),
    (58, 157, 216),
)
STATE_LABELS = ("A", "B", "C", "D")
RING_COLOR = (240, 50, 160)
INK = (46, 52, 60)
PANEL_FILL = (255, 253, 246)
BACKGROUND = (248, 248, 244)
SLOT_CENTERS_INITIAL = ((66, 105), (196, 105), (66, 275), (196, 275))
SLOT_CENTERS_FINAL = ((316, 105), (446, 105), (316, 275), (446, 275))


def _jittered_centers(rng: random.Random, bases: tuple[tuple[int, int], ...]) -> list[list[float]]:
    return [
        [round(x + rng.uniform(-7.0, 7.0), 3), round(y + rng.uniform(-7.0, 7.0), 3)]
        for x, y in bases
    ]


def sample_scene(seed: int) -> dict:
    """Sample a balanced, fully visible two-snapshot scene from one integer seed."""
    rng = random.Random(int(seed))
    identities = [0, 1, 2, 3]
    ringed_id = rng.randrange(4)
    initial_slot = rng.choice((0, 2))
    final_slot = 3 if initial_slot == 0 else 1
    remaining = [identity for identity in identities if identity != ringed_id]
    rng.shuffle(remaining)
    initial_order: list[int | None] = [None, None, None, None]
    initial_order[initial_slot] = ringed_id
    for slot in range(4):
        if initial_order[slot] is None:
            initial_order[slot] = remaining.pop()
    remaining = [identity for identity in identities if identity != ringed_id]
    rng.shuffle(remaining)
    final_order: list[int | None] = [None, None, None, None]
    final_order[final_slot] = ringed_id
    for slot in range(4):
        if final_order[slot] is None:
            final_order[slot] = remaining.pop()
    final_states = [rng.randrange(4) for _ in identities]
    return {
        "ringed_id": ringed_id,
        "initial_order": [int(value) for value in initial_order],
        "final_order": [int(value) for value in final_order],
        "final_states": final_states,
        "token_centers_initial": _jittered_centers(rng, SLOT_CENTERS_INITIAL),
        "token_centers_final": _jittered_centers(rng, SLOT_CENTERS_FINAL),
        "token_radius": round(rng.uniform(28.0, 32.0), 3),
        "badge_radius": round(rng.uniform(12.0, 16.0), 3),
        "ring_radius": round(rng.uniform(41.0, 45.0), 3),
        "background": list(BACKGROUND),
        "width": WIDTH,
        "height": HEIGHT,
    }


def _ellipse(box: tuple[float, float, float, float], scale: int) -> tuple[int, int, int, int]:
    return tuple(int(round(value * scale)) for value in box)


def _draw_token(
    draw: ImageDraw.ImageDraw,
    center: list[float],
    identity: int,
    radius: float,
    scale: int,
    *,
    badge: int | None = None,
    badge_radius: float = 0.0,
    ring: bool = False,
    ring_radius: float = 0.0,
) -> None:
    cx, cy = center
    outer = (cx - radius, cy - radius, cx + radius, cy + radius)
    draw.ellipse(_ellipse(outer, scale), fill=IDENTITY_COLORS[identity], outline=INK, width=2 * scale)
    if badge is not None:
        inner = (cx - badge_radius, cy - badge_radius, cx + badge_radius, cy + badge_radius)
        draw.ellipse(_ellipse(inner, scale), fill=STATE_COLORS[badge], outline=INK, width=2 * scale)
    else:
        neutral = max(6.0, radius * 0.28)
        inner = (cx - neutral, cy - neutral, cx + neutral, cy + neutral)
        draw.ellipse(_ellipse(inner, scale), fill=(255, 255, 255), outline=INK, width=scale)
    if ring:
        ring_box = (
            cx - ring_radius,
            cy - ring_radius,
            cx + ring_radius,
            cy + ring_radius,
        )
        draw.ellipse(_ellipse(ring_box, scale), outline=RING_COLOR, width=5 * scale)


def _font(scale: int) -> ImageFont.ImageFont:
    try:
        return ImageFont.truetype("DejaVuSans.ttf", 13 * scale)
    except OSError:
        return ImageFont.load_default()


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    rect: tuple[int, int, int, int],
    title: str,
    scale: int,
) -> None:
    x0, y0, x1, y1 = rect
    draw.rounded_rectangle(
        _ellipse((x0, y0, x1, y1), scale),
        radius=12 * scale,
        fill=PANEL_FILL,
        outline=INK,
        width=2 * scale,
    )
    draw.text((int((x0 + 12) * scale), int((y0 + 9) * scale)), title, fill=INK, font=_font(scale))
    draw.line(
        (int((x0 + 10) * scale), int((y0 + 35) * scale), int((x1 - 10) * scale), int((y0 + 35) * scale)),
        fill=(188, 190, 188),
        width=scale,
    )


def _draw_state_key(draw: ImageDraw.ImageDraw, scale: int) -> None:
    font = _font(scale)
    x0, y = 286, 321
    for index, label in enumerate(STATE_LABELS):
        x = x0 + index * 50
        draw.ellipse(_ellipse((x, y, x + 14, y + 14), scale), fill=STATE_COLORS[index], outline=INK, width=scale)
        draw.text((int((x + 18) * scale), int((y - 2) * scale)), label, fill=INK, font=font)
    draw.text((int(286 * scale), int(303 * scale)), "final badge key", fill=(88, 90, 92), font=font)


def render(scene: dict) -> Image.Image:
    """Render only the visible two-panel snapshots; no metadata is embedded."""
    scale = 3
    background = tuple(int(value) for value in scene["background"])
    image = Image.new("RGB", (scene["width"] * scale, scene["height"] * scale), background)
    draw = ImageDraw.Draw(image)
    _draw_panel(draw, (18, 48, 244, 350), "1  INITIAL", scale)
    _draw_panel(draw, (268, 48, 494, 350), "2  FINAL", scale)
    draw.text((int(202 * scale), int(20 * scale)), "identity handoff", fill=INK, font=_font(scale))

    radius = float(scene["token_radius"])
    for slot, identity in enumerate(scene["initial_order"]):
        _draw_token(
            draw,
            scene["token_centers_initial"][slot],
            int(identity),
            radius,
            scale,
            ring=int(identity) == int(scene["ringed_id"]),
            ring_radius=float(scene["ring_radius"]),
        )
    for slot, identity in enumerate(scene["final_order"]):
        _draw_token(
            draw,
            scene["token_centers_final"][slot],
            int(identity),
            radius,
            scale,
            badge=int(scene["final_states"][int(identity)]),
            badge_radius=float(scene["badge_radius"]),
        )
    _draw_state_key(draw, scale)
    return image.resize((scene["width"], scene["height"]), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """Read the final state assigned to the identity selected by the initial ring."""
    return STATE_LABELS[int(scene["final_states"][int(scene["ringed_id"])])]


def margin(scene: dict) -> float:
    """Raster clearance margin for the visible identity/badge binding."""
    radius = float(scene["token_radius"])
    badge = float(scene["badge_radius"])
    ring = float(scene["ring_radius"])
    centers = [*scene["token_centers_initial"], *scene["token_centers_final"]]
    pair_clearance = min(
        math.hypot(float(a[0]) - float(b[0]), float(a[1]) - float(b[1])) - 2.0 * radius
        for index, a in enumerate(centers)
        for b in centers[index + 1 :]
    )
    return round(min(badge - 8.0, ring - radius - 4.0, pair_clearance), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.5


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """There are no nontrivial latent aliases in this minimal raster parameterization."""
    return []
