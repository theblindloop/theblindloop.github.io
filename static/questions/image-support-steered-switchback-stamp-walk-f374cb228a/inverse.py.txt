"""Independent pixel-only parser for Switchback Stamp Walk."""
from __future__ import annotations
from collections import deque
import numpy as np
from PIL import Image

def _components(mask: np.ndarray, minimum: int=1) -> list[np.ndarray]:
    h,w=mask.shape; seen=np.zeros_like(mask,bool); out=[]
    for yy,xx in np.argwhere(mask):
        y,x=int(yy),int(xx)
        if seen[y,x]: continue
        q=deque([(y,x)]); seen[y,x]=1; pts=[]
        while q:
            cy,cx=q.popleft(); pts.append((cy,cx))
            for ny,nx in ((cy-1,cx),(cy+1,cx),(cy,cx-1),(cy,cx+1)):
                if 0<=ny<h and 0<=nx<w and mask[ny,nx] and not seen[ny,nx]: seen[ny,nx]=1; q.append((ny,nx))
        if len(pts)>=minimum: out.append(np.asarray(pts,np.int32))
    return out

def _cent(c): return float(c[:,0].mean()),float(c[:,1].mean())

def _decode_digit(ink: np.ndarray, cx: float, cy: float) -> int:
    x=int(round(cx))-12; y=int(round(cy))-11
    probes={"a":(x,y-7),"g":(x,y),"d":(x,y+7),"f":(x-7,y-4),"b":(x+7,y-4),"e":(x-7,y+4),"c":(x+7,y+4)}
    active=frozenset(k for k,(px,py) in probes.items() if np.any(ink[py-1:py+2,px-1:px+2]))
    patterns={frozenset(v):k for k,v in {
      1:("b","c"),2:("a","b","g","e","d"),3:("a","b","g","c","d"),4:("f","g","b","c"),
      5:("a","f","g","c","d"),6:("a","f","g","e","c","d"),7:("a","b","c"),
      8:("a","b","c","d","e","f","g"),9:("a","b","c","d","f","g")}.items()}
    if active not in patterns: raise ValueError("digit label is missing or ambiguous")
    return patterns[active]

def decision_from_image(image: Image.Image) -> str:
    a=np.asarray(image.convert("RGB"),dtype=np.int16); r,g,b=a[:,:,0],a[:,:,1],a[:,:,2]
    violet=(r>=80)&(r<=145)&(g>=45)&(g<=105)&(b>=145)&(b<=210)
    cyan=(r<=55)&(g>=130)&(g<=195)&(b>=155)&(b<=220)
    gold=(r>=200)&(g>=125)&(g<=200)&(b<=75)
    ink=(r>=20)&(r<=80)&(g>=20)&(g<=85)&(b>=25)&(b<=95)
    vc=_components(violet,30)
    if len(vc)!=1 or len(vc[0])<5000: raise ValueError("connected track is missing or broken")
    cc=_components(cyan,40)
    if not 7<=len(cc)<=14: raise ValueError("stamp components are missing")
    cents=sorted([_cent(c) for c in cc])
    groups=[]
    for cy,cx in sorted(cents,key=lambda p:p[1]):
        target=next((z for z in groups if abs(np.mean([p[1] for p in z])-cx)<20),None)
        if target is None: groups.append([(cy,cx)])
        else: target.append((cy,cx))
    if len(groups)!=7 or any(len(z) not in (1,2) for z in groups): raise ValueError("node stamps are ambiguous")
    nodes=[]
    for group in groups:
        stamp=len(group); doty=float(np.mean([p[0] for p in group])); dotx=float(np.mean([p[1] for p in group]))
        cx=dotx-(13.0 if stamp==1 else 10.0); cy=doty-(0.0 if stamp==1 else 1.5)
        nodes.append((cy,cx,stamp,_decode_digit(ink,cx,cy)))
    labels=[n[3] for n in nodes]
    if sorted(labels)!=list(range(1,8)): raise ValueError("digit labels are not a permutation")
    gp=np.argwhere(gold)
    if len(gp)<150: raise ValueError("start ring is missing")
    gy,gx=float(gp[:,0].mean()),float(gp[:,1].mean()); d=[(cy-gy)**2+(cx-gx)**2 for cy,cx,_,_ in nodes]
    order=np.argsort(d)
    if d[int(order[0])]>100 or d[int(order[1])]-d[int(order[0])]<1000: raise ValueError("start square is ambiguous")
    pos=int(order[0])
    if pos>=2: raise ValueError("start is outside declared initial section")
    for _ in range(3):
        pos += nodes[pos][2]
        if pos>=7: raise ValueError("walk exits visible route")
    return str(labels[pos])
