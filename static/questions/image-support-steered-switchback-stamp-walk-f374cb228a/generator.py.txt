"""Deterministic renderer and analytic arm for Switchback Stamp Walk."""
from __future__ import annotations
import random
from typing import Any
from PIL import Image, ImageDraw

WIDTH = HEIGHT = 512
BG = (250, 249, 245)
VIOLET = (112, 76, 176)
CYAN = (16, 164, 188)
GOLD = (232, 166, 35)
INK = (48, 52, 58)
PALE = (229, 226, 218)

SEGMENTS = {
    0: ("a","b","c","d","e","f"), 1: ("b","c"),
    2: ("a","b","g","e","d"), 3: ("a","b","g","c","d"),
    4: ("f","g","b","c"), 5: ("a","f","g","c","d"),
    6: ("a","f","g","e","c","d"), 7: ("a","b","c"),
    8: ("a","b","c","d","e","f","g"), 9: ("a","b","c","d","f","g"),
}

def _replay(start: int, stamps: list[int]) -> int:
    pos = start
    for _ in range(3):
        pos += stamps[pos]
        if pos >= 7:
            return 99
    return pos

def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    target_label = int(seed) % 7 + 1
    valid = []
    for start in range(2):
        for bits in range(128):
            stamps = [1 + ((bits >> i) & 1) for i in range(7)]
            final = _replay(start, stamps)
            if 3 <= final < 7:
                valid.append((start, stamps, final))
    start, stamps, final = rng.choice(valid)
    labels = list(range(1, 8)); rng.shuffle(labels)
    where = labels.index(target_label)
    labels[where], labels[final] = labels[final], labels[where]
    return {"start_index": start, "stamps": stamps, "labels": labels,
            "anchor_x": rng.choice((246, 256, 266)), "anchor_y": rng.choice((236, 246, 256)),
            "decoration_phase": rng.randrange(11)}

def _points(scene: dict[str, Any]) -> list[tuple[int,int]]:
    ax, ay = int(scene["anchor_x"]), int(scene["anchor_y"])
    offsets = (-24, -8, 8, 24, 8, -8, -24)
    return [(ax - 144 + 48*i, ay + offsets[i]) for i in range(7)]

def _digit(draw: ImageDraw.ImageDraw, cx: int, cy: int, digit: int) -> None:
    x, y = cx-12, cy-11
    seg = {"a":(x-5,y-8,x+5,y-5), "g":(x-5,y-1,x+5,y+2), "d":(x-5,y+6,x+5,y+9),
           "f":(x-8,y-7,x-5,y), "b":(x+5,y-7,x+8,y), "e":(x-8,y,x-5,y+7), "c":(x+5,y,x+8,y+7)}
    for name in SEGMENTS[digit]: draw.rectangle(seg[name], fill=INK)

def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BG); draw = ImageDraw.Draw(image)
    points = _points(scene)
    phase = int(scene["decoration_phase"])
    for i in range(12):
        x = 22 + ((i*97 + phase*29) % 468); y = 22 + ((i*61 + phase*43) % 468)
        if 105 < x < 407 and 125 < y < 375: continue
        draw.ellipse((x-3,y-3,x+3,y+3), outline=PALE, width=2)
    for cx,cy in points:
        draw.rectangle((cx-21,cy-21,cx+21,cy+21), fill=BG, outline=VIOLET, width=5)
    # Explicit overlapping connectors make every node border part of one route.
    for (x0,y0),(x1,y1) in zip(points,points[1:]):
        draw.line((x0,y0,x1,y1),fill=VIOLET,width=9)
    x0,y0 = points[0]; draw.polygon([(x0-31,y0),(x0-21,y0-9),(x0-21,y0+9)], fill=VIOLET)
    x1,y1 = points[-1]; draw.polygon([(x1+31,y1),(x1+21,y1-9),(x1+21,y1+9)], fill=VIOLET)
    for idx, ((cx,cy), stamp, label) in enumerate(zip(points, scene["stamps"], scene["labels"], strict=True)):
        _digit(draw,cx,cy,int(label))
        if int(stamp)==1:
            draw.ellipse((cx+8,cy-5,cx+18,cy+5), fill=CYAN)
        else:
            draw.ellipse((cx+5,cy-10,cx+15,cy), fill=CYAN); draw.ellipse((cx+5,cy+3,cx+15,cy+13), fill=CYAN)
        if idx == int(scene["start_index"]):
            # Four corner brackets mark the square without cutting the route at its edge midpoints.
            for sx,sy in ((-1,-1),(1,-1),(-1,1),(1,1)):
                draw.line((cx+sx*27,cy+sy*27,cx+sx*18,cy+sy*27),fill=GOLD,width=4)
                draw.line((cx+sx*27,cy+sy*27,cx+sx*27,cy+sy*18),fill=GOLD,width=4)
    return image

def analytic_gold(scene: dict[str, Any]) -> str:
    return str(int(scene["labels"][_replay(int(scene["start_index"]), list(map(int,scene["stamps"]))) ]))

def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        start=int(scene["start_index"]); stamps=list(map(int,scene["stamps"])); labels=list(map(int,scene["labels"]));
        ax=int(scene["anchor_x"]); ay=int(scene["anchor_y"]); int(scene["decoration_phase"])
    except (KeyError,TypeError,ValueError): return True
    return not (start in range(2) and len(stamps)==7 and all(v in (1,2) for v in stamps)
                and sorted(labels)==list(range(1,8)) and _replay(start,stamps) in range(7)
                and ax in (246,256,266) and ay in (236,246,256))

def margin(scene: dict[str, Any]) -> float:
    return 0.0 if is_quarantined(scene) else 10.0

def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str,dict[str,Any]]]:
    return []
