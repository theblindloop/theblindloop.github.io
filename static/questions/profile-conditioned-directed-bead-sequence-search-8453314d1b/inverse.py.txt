"""Independent final-PNG inverse arm.

Only visible palette masks and connectivity are used. No renderer or latent scene
module is imported, and no layout coordinate is supplied.
"""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

TEAL = (18, 151, 151)
NAVY = (39, 66, 125)
INK = (42, 45, 49)
BEADS = {
    (218, 70, 64): "red",
    (232, 174, 40): "gold",
    (55, 154, 91): "green",
    (139, 82, 181): "purple",
}


def _components(mask: np.ndarray, minimum: int = 20):
    h, w = mask.shape
    seen = np.zeros((h, w), dtype=bool)
    found = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = [(int(y), int(x))]
        seen[y, x] = True
        pts = []
        for cy, cx in queue:
            pts.append((cy, cx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = cy + dy, cx + dx
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(pts) >= minimum:
            found.append(pts)
    return found


def _centroid(points):
    return (sum(p[0] for p in points) / len(points), sum(p[1] for p in points) / len(points))


def _distance_map(foreground: np.ndarray, start):
    h, w = foreground.shape
    dist = np.full((h, w), -1, dtype=np.int32)
    sy, sx = start
    queue = deque([(sy, sx)])
    dist[sy, sx] = 0
    while queue:
        y, x = queue.popleft()
        nd = int(dist[y, x]) + 1
        for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            ny, nx = y + dy, x + dx
            if 0 <= ny < h and 0 <= nx < w and foreground[ny, nx] and dist[ny, nx] < 0:
                dist[ny, nx] = nd
                queue.append((ny, nx))
    return dist


def _read_route(arr: np.ndarray, start_points, bead_components):
    h, w, _ = arr.shape
    allowed = np.zeros((h, w), dtype=bool)
    for rgb in [INK, TEAL, NAVY, *BEADS]:
        allowed |= np.all(arr == np.array(rgb, dtype=np.uint8), axis=2)
    cy, cx = _centroid(start_points)
    sy, sx = int(round(cy)), int(round(cx))
    if not allowed[sy, sx]:
        raise ValueError("start marker is not connected to route ink")
    dist = _distance_map(allowed, (sy, sx))
    ordered = []
    for name, points in bead_components:
        reachable = [int(dist[y, x]) for y, x in points if dist[y, x] >= 0]
        if reachable:
            ordered.append((min(reachable), name))
    if len(ordered) != 4:
        raise ValueError(f"route exposes {len(ordered)} reachable beads, expected four")
    ordered.sort()
    if len({d for d, _ in ordered}) != 4:
        raise ValueError("bead traversal positions are ambiguous")
    return tuple(name for _, name in ordered)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    starts = _components(np.all(arr == np.array(TEAL, dtype=np.uint8), axis=2), minimum=80)
    if len(starts) != 5:
        raise ValueError(f"found {len(starts)} teal start markers, expected five")
    routes = []
    all_beads = []
    for rgb, name in BEADS.items():
        comps = _components(np.all(arr == np.array(rgb, dtype=np.uint8), axis=2), minimum=80)
        if len(comps) != 5:
            raise ValueError(f"found {len(comps)} {name} beads, expected five")
        all_beads.extend((name, pts, _centroid(pts)) for pts in comps)
    for start in starts:
        sy, sx = _centroid(start)
        nearby = []
        # Each route is an isolated foreground component, so the full bead list
        # can be offered and unreachable beads are ignored by _read_route.
        for name, pts, center in all_beads:
            nearby.append((name, pts))
        routes.append(((sy, sx), _read_route(arr, start, nearby)))
    routes.sort(key=lambda item: (item[0][0], item[0][1]))
    reference = routes[0][1]
    candidates = sorted(routes[1:], key=lambda item: item[0][1])
    matches = [i for i, item in enumerate(candidates) if item[1] == reference]
    if len(matches) != 1:
        raise ValueError("pixel evidence does not yield one matching candidate")
    return "ABCD"[matches[0]]
