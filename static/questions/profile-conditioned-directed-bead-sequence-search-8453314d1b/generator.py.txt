"""Deterministic scene sampler and raster renderer."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 720
BG = (248, 247, 242)
PANEL = (255, 255, 255)
BORDER = (170, 174, 178)
INK = (42, 45, 49)
TEAL = (18, 151, 151)
NAVY = (39, 66, 125)
BEAD_RGB = {
    "red": (218, 70, 64),
    "gold": (232, 174, 40),
    "green": (55, 154, 91),
    "purple": (139, 82, 181),
}

# Six-point simple polylines. Internal points carry the four beads. Dihedral
# variants make traversal horizontal, vertical, or reversed without changing the
# color-order rule.
TEMPLATES = (
    ((.08, .14), (.48, .14), (.48, .42), (.22, .42), (.22, .84), (.90, .84)),
    ((.10, .18), (.70, .18), (.70, .48), (.38, .48), (.38, .82), (.90, .82)),
    ((.10, .12), (.34, .12), (.34, .68), (.62, .68), (.62, .34), (.90, .34)),
    ((.12, .18), (.42, .18), (.42, .78), (.76, .78), (.76, .46), (.90, .46)),
)


def _transform(points, variant: int):
    pts = list(points)
    rot = (variant // 2) % 4
    flip = variant % 2
    out = []
    for x, y in pts:
        if flip:
            x = 1.0 - x
        for _ in range(rot):
            x, y = 1.0 - y, x
        out.append((x, y))
    return out


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    colors = list(BEAD_RGB)
    rng.shuffle(colors)
    reference = colors
    near = colors.copy()
    swap_at = rng.randrange(3)
    near[swap_at], near[swap_at + 1] = near[swap_at + 1], near[swap_at]
    rotate = colors[1:] + colors[:1]
    reverse = list(reversed(colors))
    distractors = [near, rotate, reverse]
    rng.shuffle(distractors)
    answer_index = int(seed) % 4
    candidates = []
    cursor = 0
    for index in range(4):
        if index == answer_index:
            candidates.append(reference.copy())
        else:
            candidates.append(distractors[cursor])
            cursor += 1
    styles = []
    for _ in range(5):
        styles.append([rng.randrange(len(TEMPLATES)), rng.randrange(8)])
    return {
        "reference_sequence": reference,
        "candidate_sequences": candidates,
        "route_styles": styles,
        "width": WIDTH,
        "height": HEIGHT,
    }


def _draw_marker_triangle(draw, center, radius=17):
    x, y = center
    draw.polygon([(x - radius, y + radius), (x - radius, y - radius), (x + radius, y)], fill=TEAL)


def _draw_route(draw, box, sequence, style):
    x0, y0, x1, y1 = box
    inset_x, inset_y = 25, 30
    template = TEMPLATES[int(style[0]) % len(TEMPLATES)]
    points = _transform(template, int(style[1]))
    coords = [
        (int(x0 + inset_x + px * (x1 - x0 - 2 * inset_x)),
         int(y0 + inset_y + py * (y1 - y0 - 2 * inset_y)))
        for px, py in points
    ]
    draw.line(coords, fill=INK, width=11, joint="curve")
    _draw_marker_triangle(draw, coords[0])
    ex, ey = coords[-1]
    draw.rectangle((ex - 15, ey - 15, ex + 15, ey + 15), fill=NAVY)
    for name, (x, y) in zip(sequence, coords[1:5]):
        draw.ellipse((x - 14, y - 14, x + 14, y + 14), fill=BEAD_RGB[name], outline=INK, width=2)


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width, height), BG)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    draw.text((width // 2, 18), "FOLLOW TRIANGLE  >  SQUARE", fill=INK, font=font, anchor="ma")
    ref_box = (240, 55, 720, 310)
    draw.rounded_rectangle(ref_box, radius=12, fill=PANEL, outline=BORDER, width=3)
    draw.text((260, 68), "REFERENCE", fill=INK, font=font)
    boxes = []
    for index in range(4):
        x0 = 20 + index * 235
        box = (x0, 385, x0 + 215, 680)
        boxes.append(box)
        draw.rounded_rectangle(box, radius=12, fill=PANEL, outline=BORDER, width=3)
        draw.text((x0 + 107, 365), "ABCD"[index], fill=INK, font=font, anchor="mm")
    _draw_route(draw, ref_box, scene["reference_sequence"], scene["route_styles"][0])
    for box, sequence, style in zip(boxes, scene["candidate_sequences"], scene["route_styles"][1:]):
        _draw_route(draw, box, sequence, style)
    return image


def _scores(scene: dict) -> list[int]:
    reference = scene["reference_sequence"]
    return [sum(a == b for a, b in zip(reference, seq)) for seq in scene["candidate_sequences"]]


def analytic_gold(scene: dict) -> str:
    reference = scene["reference_sequence"]
    candidates = scene["candidate_sequences"]
    matches = [index for index, seq in enumerate(candidates) if seq == reference]
    if len(matches) != 1:
        raise ValueError("scene does not have a unique exact sequence match")
    return "ABCD"[matches[0]]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    scores = sorted(_scores(scene), reverse=True)
    return scores[0] != 4 or scores[0] - scores[1] < 2


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
