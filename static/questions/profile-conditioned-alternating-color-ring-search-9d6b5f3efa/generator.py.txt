"""Deterministic latent sampler and raster renderer for cyclic color search."""

from __future__ import annotations

import copy
import itertools
import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 760, 560
SUPERSAMPLE = 3
LABELS = ("A", "B", "C", "D")
PANEL_CENTERS = ((205.0, 160.0), (555.0, 160.0), (205.0, 400.0), (555.0, 400.0))
PANEL_BOXES = ((35, 35, 375, 275), (385, 35, 725, 275), (35, 285, 375, 525), (385, 285, 725, 525))
ORBIT_RADIUS = 70.0
DOT_RADIUS = 10.0
QUARANTINE_MARGIN_CHANGES = 2.0
PALETTES = (
    ("#C1121F", "#2563EB"),
    ("#7B2CBF", "#00A896"),
    ("#D1495B", "#00798C"),
)


def _transition_count(sequence: list[int]) -> int:
    if len(sequence) != 6:
        raise ValueError("each ring must contain exactly six marks")
    if sequence.count(0) != 3 or sequence.count(1) != 3:
        raise ValueError("each ring must contain three marks of each color")
    return sum(sequence[index] != sequence[(index + 1) % 6] for index in range(6))


def _ring_sequence(ring: list[dict]) -> list[int]:
    ordered = sorted(ring, key=lambda mark: float(mark["angle"]) % 360.0)
    return [int(mark["color_class"]) for mark in ordered]


def _panel_scores(rings: list[list[dict]]) -> list[int]:
    if len(rings) != 4:
        raise ValueError("the world requires exactly four panels")
    return [_transition_count(_ring_sequence(ring)) for ring in rings]


def _balanced_sequences_by_score() -> dict[int, list[list[int]]]:
    pools: dict[int, list[list[int]]] = {2: [], 4: [], 6: []}
    for ones in itertools.combinations(range(6), 3):
        sequence = [0] * 6
        for index in ones:
            sequence[index] = 1
        pools[_transition_count(sequence)].append(sequence)
    return pools


BALANCED_SEQUENCES = _balanced_sequences_by_score()


def sample_scene(seed: int) -> dict:
    """Sample equal-inventory scenes; answer identity cycles exactly by seed."""

    rng = random.Random(int(seed))
    target = int(seed) % 4
    loser_scores = [4, 4, 2]
    rng.shuffle(loser_scores)
    rings: list[list[dict]] = []
    loser_index = 0
    for panel_index in range(4):
        if panel_index == target:
            sequence = list(rng.choice(BALANCED_SEQUENCES[6]))
        else:
            sequence = list(rng.choice(BALANCED_SEQUENCES[loser_scores[loser_index]]))
            loser_index += 1
        rotation = rng.uniform(0.0, 59.0)
        ring = [
            {
                "angle": (rotation + 60.0 * index) % 360.0,
                "color_class": sequence[index],
            }
            for index in range(6)
        ]
        rng.shuffle(ring)
        rings.append(ring)
    return {
        "rings": rings,
        "palette": list(rng.choice(PALETTES)),
        "offsets": [[rng.randint(-4, 4), rng.randint(-4, 4)] for _ in range(4)],
    }


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    path = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
    try:
        return ImageFont.truetype(path, size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    """Render only declared fields, with identical geometry and ink per panel."""

    rings = scene["rings"]
    palette = scene["palette"]
    offsets = scene["offsets"]
    if len(rings) != 4 or len(offsets) != 4 or len(palette) != 2:
        raise ValueError("malformed scene")
    scale = SUPERSAMPLE
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), "#FFFFFF")
    draw = ImageDraw.Draw(image)
    label_font = _font(24 * scale)
    for index, (box, base_center) in enumerate(zip(PANEL_BOXES, PANEL_CENTERS, strict=True)):
        draw.rounded_rectangle(
            tuple(int(value * scale) for value in box),
            radius=14 * scale,
            fill="#FAFBFC",
            outline="#AEB8C2",
            width=2 * scale,
        )
        draw.text(
            ((box[0] + 16) * scale, (box[1] + 12) * scale),
            LABELS[index],
            fill="#343A40",
            font=label_font,
        )
        cx = (base_center[0] + float(offsets[index][0])) * scale
        cy = (base_center[1] + float(offsets[index][1])) * scale
        for mark in rings[index]:
            radians = math.radians(float(mark["angle"]))
            x = cx + ORBIT_RADIUS * scale * math.cos(radians)
            y = cy + ORBIT_RADIUS * scale * math.sin(radians)
            radius = DOT_RADIUS * scale
            color = palette[int(mark["color_class"])]
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    scores = _panel_scores(scene["rings"])
    order = sorted(range(4), key=lambda index: scores[index], reverse=True)
    if scores[order[0]] - scores[order[1]] < 1:
        raise ValueError("scene has no unique alternating-ring winner")
    return LABELS[order[0]]


def margin(scene: dict) -> float:
    scores = sorted(_panel_scores(scene["rings"]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN_CHANGES


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Mark-list order is latent because dots never overlap."""

    twin = copy.deepcopy(scene)
    twin["rings"] = [list(reversed(ring)) for ring in scene["rings"]]
    return [("reverse_unordered_mark_lists", twin)]
