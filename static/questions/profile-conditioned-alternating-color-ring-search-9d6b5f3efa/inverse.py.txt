"""Independent final-PNG-only inverse arm for alternating-color ring search.

The oracle uses only the public visual contract: four labeled regions in reading
order, each containing six equal-size colored dots in a ring, with three of each
color. It discovers marks, panels, colors, and circular order from raster pixels.
It imports no renderer, latent scene, prompt, generator, verifier, or gold code.
"""

from __future__ import annotations

from collections import Counter, deque
import math

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")
MIN_COMPONENT_AREA = 80
ABSTAIN_MARGIN_CHANGES = 2


def _components(mask: np.ndarray, rgb: np.ndarray) -> list[dict]:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    components: list[dict] = []
    for start_y, start_x in np.argwhere(mask):
        sy, sx = int(start_y), int(start_x)
        if visited[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        visited[sy, sx] = True
        count = 0
        sum_x = 0.0
        sum_y = 0.0
        colors: Counter[tuple[int, int, int]] = Counter()
        while queue:
            y, x = queue.popleft()
            count += 1
            sum_x += x + 0.5
            sum_y += y + 0.5
            colors[tuple(int(value) for value in rgb[y, x])] += 1
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    queue.append((ny, nx))
        if count >= MIN_COMPONENT_AREA:
            components.append(
                {
                    "area": count,
                    "cx": sum_x / count,
                    "cy": sum_y / count,
                    "tone": colors.most_common(1)[0][0],
                }
            )
    return components


def _panel_index(mark: dict, width: int, height: int) -> int:
    return (2 if mark["cy"] >= height / 2 else 0) + (1 if mark["cx"] >= width / 2 else 0)


def _transition_count(marks: list[dict]) -> int:
    if len(marks) != 6:
        raise ValueError("abstain: every panel must contain six dots")
    tones = Counter(mark["tone"] for mark in marks)
    if len(tones) != 2 or sorted(tones.values()) != [3, 3]:
        raise ValueError("abstain: every panel must visibly contain three dots of each color")
    center_x = sum(mark["cx"] for mark in marks) / 6.0
    center_y = sum(mark["cy"] for mark in marks) / 6.0
    ordered = sorted(
        marks,
        key=lambda mark: math.atan2(mark["cy"] - center_y, mark["cx"] - center_x),
    )
    return sum(ordered[index]["tone"] != ordered[(index + 1) % 6]["tone"] for index in range(6))


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    channel_range = rgb.max(axis=2) - rgb.min(axis=2)
    colored = channel_range >= 60
    components = _components(colored, rgb)
    if len(components) != 24:
        raise ValueError(f"abstain: expected 24 colored dots, recovered {len(components)}")
    tones = Counter(component["tone"] for component in components)
    if len(tones) != 2 or sorted(tones.values()) != [12, 12]:
        raise ValueError("abstain: the two globally balanced dot colors were not recovered")
    height, width = colored.shape
    panels: list[list[dict]] = [[] for _ in LABELS]
    for component in components:
        panels[_panel_index(component, width, height)].append(component)
    scores = [_transition_count(panel) for panel in panels]
    order = sorted(range(4), key=lambda index: scores[index], reverse=True)
    if scores[order[0]] != 6:
        raise ValueError("abstain: no fully alternating ring was recovered")
    if scores[order[0]] - scores[order[1]] < ABSTAIN_MARGIN_CHANGES:
        raise ValueError("abstain: alternating-ring winner lacks a safe runner-up margin")
    return LABELS[order[0]]
