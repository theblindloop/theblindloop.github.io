"""Renderer, analytic gold and latent aliases for the arrow-aim stencil world."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

GRID = 5
PITCH = 88
CANVAS = 620
ARROW_R = 17
ARROW_BACK = 9
ARROW_HALF = 8
MARK = 15
DOT_R = 16

INKS = [(28, 28, 34), (34, 30, 26), (24, 32, 30), (32, 26, 32)]
PAPERS = [(246, 245, 240), (243, 244, 246), (247, 243, 238)]
DOTS = [(206, 40, 40), (196, 52, 30), (214, 34, 62)]
MARKS = [(38, 70, 190), (26, 84, 176), (46, 62, 200)]


def _plus() -> frozenset:
    return frozenset([(2, j) for j in range(5)] + [(i, 2) for i in range(5)])


def _ex() -> frozenset:
    return frozenset([(i, i) for i in range(5)] + [(i, 4 - i) for i in range(5)])


def _tee() -> frozenset:
    return frozenset([(0, j) for j in range(5)] + [(i, 2) for i in range(1, 5)])


def _ell() -> frozenset:
    return frozenset([(i, 0) for i in range(5)] + [(4, j) for j in range(1, 5)])


def _cee() -> frozenset:
    return frozenset([(i, 0) for i in range(5)] + [(0, j) for j in range(1, 5)])


SYMBOL_NAMES = ("cee", "ell", "ex", "plus", "tee")
SYMBOL_CELLS = {
    "cee": _cee(),
    "ell": _ell(),
    "ex": _ex(),
    "plus": _plus(),
    "tee": _tee(),
}


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    symbol_index = rng.randrange(len(SYMBOL_NAMES)) + len(SYMBOL_NAMES) * rng.randrange(2)
    flaws = [0, 0, 0, 1, 1, 2][seed % 6]
    order = list(range(GRID * GRID))
    rng.shuffle(order)
    flipped = sorted(order[:flaws])
    side = rng.randrange(4)
    along = rng.randrange(-140, 141)
    offsets = [rng.randrange(55, 306) for _ in range(GRID * GRID)]
    return {
        "symbol_index": symbol_index,
        "flipped": flipped,
        "side": side + 4 * rng.randrange(2),
        "along": along,
        "offsets": offsets,
        "palette": rng.randrange(len(INKS)) + len(INKS) * rng.randrange(2),
        "paper": rng.randrange(len(PAPERS)),
        "dot_hue": rng.randrange(len(DOTS)),
        "mark_hue": rng.randrange(len(MARKS)),
        "jitter_x": rng.randrange(-9, 10),
        "jitter_y": rng.randrange(-9, 10),
    }


def _symbol(scene: dict) -> str:
    return SYMBOL_NAMES[int(scene["symbol_index"]) % len(SYMBOL_NAMES)]


def _geometry(scene: dict) -> dict:
    span = (GRID - 1) * PITCH
    ox = (CANVAS - span) // 2 + int(scene["jitter_x"])
    oy = (CANVAS - span) // 2 + int(scene["jitter_y"])
    side = int(scene["side"]) % 4
    along = int(scene["along"])
    cx, cy = ox + span / 2.0, oy + span / 2.0
    edge = 48
    if side == 0:
        dot = (cx + along, edge)
    elif side == 1:
        dot = (CANVAS - edge, cy + along)
    elif side == 2:
        dot = (cx + along, CANVAS - edge)
    else:
        dot = (edge, cy + along)
    return {"ox": ox, "oy": oy, "dot": dot}


def _pattern(scene: dict) -> frozenset:
    cells = set(SYMBOL_CELLS[_symbol(scene)])
    for index in scene["flipped"]:
        cell = (int(index) // GRID, int(index) % GRID)
        cells.symmetric_difference_update({cell})
    return frozenset(cells)


def _distances(scene: dict) -> list:
    pattern = _pattern(scene)
    out = []
    for name in SYMBOL_NAMES:
        out.append((len(pattern.symmetric_difference(SYMBOL_CELLS[name])), name))
    out.sort()
    return out


def analytic_gold(scene: dict) -> str:
    return _distances(scene)[0][1]


def margin(scene: dict) -> float:
    ranked = _distances(scene)
    return float(ranked[1][0] - ranked[0][0])


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < 5.0)


def latent_symmetries(scene: dict) -> list:
    twins = []
    first = dict(scene)
    first["symbol_index"] = int(scene["symbol_index"]) + len(SYMBOL_NAMES)
    twins.append(("symbol_index_alias", first))
    second = dict(scene)
    second["palette"] = int(scene["palette"]) + len(INKS)
    twins.append(("palette_index_alias", second))
    third = dict(scene)
    third["side"] = int(scene["side"]) + 4
    twins.append(("side_index_alias", third))
    return twins


def render(scene: dict) -> Image.Image:
    geo = _geometry(scene)
    ox, oy = geo["ox"], geo["oy"]
    dot = geo["dot"]
    ink = INKS[int(scene["palette"]) % len(INKS)]
    paper = PAPERS[int(scene["paper"]) % len(PAPERS)]
    dot_colour = DOTS[int(scene["dot_hue"]) % len(DOTS)]
    mark_colour = MARKS[int(scene["mark_hue"]) % len(MARKS)]
    image = Image.new("RGB", (CANVAS, CANVAS), paper)
    draw = ImageDraw.Draw(image)
    half = PITCH // 2
    for iy in (oy - half, oy + (GRID - 1) * PITCH + half):
        for ix in (ox - half, ox + (GRID - 1) * PITCH + half):
            draw.rectangle(
                [ix - MARK // 2, iy - MARK // 2, ix + MARK // 2, iy + MARK // 2],
                fill=mark_colour,
            )
    draw.ellipse(
        [dot[0] - DOT_R, dot[1] - DOT_R, dot[0] + DOT_R, dot[1] + DOT_R], fill=dot_colour
    )
    pattern = _pattern(scene)
    for row in range(GRID):
        for col in range(GRID):
            cx = ox + col * PITCH
            cy = oy + row * PITCH
            aim = math.atan2(dot[1] - cy, dot[0] - cx)
            if (row, col) in pattern:
                theta = aim
            else:
                theta = aim + math.radians(scene["offsets"][row * GRID + col])
            ct, st = math.cos(theta), math.sin(theta)
            points = []
            for lx, ly in (
                (ARROW_R, 0),
                (-ARROW_BACK, ARROW_HALF),
                (-ARROW_BACK, -ARROW_HALF),
            ):
                points.append((cx + lx * ct - ly * st, cy + lx * st + ly * ct))
            draw.polygon(points, fill=ink)
    return image
