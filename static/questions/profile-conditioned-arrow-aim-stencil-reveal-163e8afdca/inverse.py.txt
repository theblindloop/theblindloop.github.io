"""Pixel-only arm: reads one PNG and names the stencilled symbol."""

from __future__ import annotations

import math

import numpy as np

GRID = 5
ABSTAIN = "abstain"
NAMES = ("cee", "ell", "ex", "plus", "tee")


def _templates() -> dict:
    plus = {(2, j) for j in range(5)} | {(i, 2) for i in range(5)}
    ex = {(i, i) for i in range(5)} | {(i, 4 - i) for i in range(5)}
    tee = {(0, j) for j in range(5)} | {(i, 2) for i in range(1, 5)}
    ell = {(i, 0) for i in range(5)} | {(4, j) for j in range(1, 5)}
    cee = {(i, 0) for i in range(5)} | {(0, j) for j in range(1, 5)}
    return {"plus": plus, "ex": ex, "tee": tee, "ell": ell, "cee": cee}


def _components(mask: np.ndarray) -> list:
    seen = np.zeros(mask.shape, dtype=bool)
    out = []
    ys, xs = np.nonzero(mask)
    for sy, sx in zip(ys.tolist(), xs.tolist()):
        if seen[sy, sx]:
            continue
        stack = [(sy, sx)]
        seen[sy, sx] = True
        pixels = []
        while stack:
            y, x = stack.pop()
            pixels.append((y, x))
            for dy, dx in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                ny, nx = y + dy, x + dx
                if 0 <= ny < mask.shape[0] and 0 <= nx < mask.shape[1]:
                    if mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        if len(pixels) >= 25:
            out.append(np.array(pixels, dtype=float))
    return out


def decision_from_image(image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(int)
    red = arr[:, :, 0]
    green = arr[:, :, 1]
    blue = arr[:, :, 2]
    dot_mask = (red > 150) & (red - green > 60) & (red - blue > 60)
    mark_mask = (blue - red > 60) & (blue - green > 40)
    ink_mask = arr.max(axis=2) < 120

    dots = _components(dot_mask)
    marks = _components(mark_mask)
    arrows = _components(ink_mask)
    if len(dots) != 1 or len(marks) != 4 or len(arrows) != GRID * GRID:
        return ABSTAIN
    dot_y, dot_x = dots[0].mean(axis=0)
    centres = np.array([m.mean(axis=0) for m in marks])
    y0, y1 = centres[:, 0].min(), centres[:, 0].max()
    x0, x1 = centres[:, 1].min(), centres[:, 1].max()
    if y1 - y0 < 100 or x1 - x0 < 100:
        return ABSTAIN

    lit = set()
    used = set()
    for pixels in arrows:
        cy, cx = pixels.mean(axis=0)
        col = int(round((cx - x0) / (x1 - x0) * GRID - 0.5))
        row = int(round((cy - y0) / (y1 - y0) * GRID - 0.5))
        if not (0 <= row < GRID and 0 <= col < GRID) or (row, col) in used:
            return ABSTAIN
        used.add((row, col))
        deltas = pixels - np.array([cy, cx])
        apex = pixels[int(np.argmax((deltas ** 2).sum(axis=1)))]
        heading = math.atan2(apex[0] - cy, apex[1] - cx)
        aim = math.atan2(dot_y - cy, dot_x - cx)
        error = abs(math.degrees(math.atan2(math.sin(heading - aim), math.cos(heading - aim))))
        if error < 25.0:
            lit.add((row, col))
    if len(used) != GRID * GRID:
        return ABSTAIN

    ranked = sorted((len(lit ^ cells), name) for name, cells in _templates().items())
    if ranked[0][0] == ranked[1][0]:
        return ABSTAIN
    return ranked[0][1]
