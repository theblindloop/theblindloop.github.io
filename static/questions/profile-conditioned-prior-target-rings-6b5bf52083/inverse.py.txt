"""Pixel-only inverse arm: count the painted rings from the raster alone.

This module receives only the rendered PNG and never imports renderer,
prompts, generate, verify, or analytic-gold code, and it accepts no latent
scene data. The target face is painted as ``n`` concentric rings of distinct
solid colors on a white background, so the number of painted rings equals the
number of distinct non-background colors present in the image.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

BACKGROUND = (255, 255, 255)


def decision_from_image(image: Image.Image) -> str:
    """Return the counted number of painted rings as a string."""
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    background = np.array(BACKGROUND, dtype=np.int32)
    # A pixel is a painted-ring pixel unless it equals the white background.
    diff = np.abs(arr - background).sum(axis=-1)
    painted = arr[diff > 0]
    if painted.size == 0:
        return "0"
    # The rendering uses only solid ring colors with no anti-aliased blends, so
    # each distinct painted color corresponds to exactly one concentric ring.
    unique = {tuple(int(v) for v in colors) for colors in painted.tolist()}
    return str(len(unique))
