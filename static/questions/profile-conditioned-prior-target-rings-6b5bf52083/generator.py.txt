"""Prior-conflict archery target ring-count world.

A concentric target face is painted with ``n`` evenly spaced colored rings. A
familiar canonical prior asserts that a scoring target has TEN rings; the
counterfactual renders an explicit non-canonical ring count and the decision
must be read from the visible colored bands alone, never from the prior. The
pixel oracle counts the distinct painted ring colors present in the target.

Latent alias mode is ``not_applicable``: ring count, ring color set, and face
radius are all directly reflected in the raster (number of visible color
bands, their exact colors, and the target extents), so the rendering is
injective in the latent scene and there is no render-invariant alias. The
canonical-prior name "target" is a matched-control/counterfactual manipulation
of the *question*, not of any latent field.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = 512
CX = 256.0
CY = 256.0
FACE_RADIUS = 200.0
MIN_RING_WIDTH = 8.0

# Supported off-quarantine ring counts. The canonical prior is 10; 6 and 8 are
# counterfactuals that contradict it. All remain far enough apart at face
# radius ~200px that each painted ring is a readable color band.
SUPPORTED_N = (10, 8, 6)

BACKGROUND = (255, 255, 255)
# Fixed large palette of well-separated ring colors. A scene uses a contiguous
# cycle of ``n`` distinct colors starting at a random offset, so the identity
# of the colors never reveals the ring count (cheap-cue counterbalance).
_PALETTE = [
    (190, 30, 30),   # crimson
    (235, 120, 25),  # orange
    (225, 205, 40),  # gold
    (90, 170, 60),   # green
    (30, 130, 190),  # blue
    (70, 60, 170),   # indigo
    (160, 40, 160),  # magenta
    (180, 90, 40),   # brown
    (30, 30, 30),    # near-black
    (120, 120, 140), # slate
    (0, 150, 150),   # teal
    (210, 210, 210), # silver
]

PALETTE_LEN = len(_PALETTE)


def _palette_for_random(rng: random.Random, n: int) -> list[tuple[int, int, int]]:
    """Return ``n`` distinct ring colors, one per concentric band."""
    offset = rng.randrange(PALETTE_LEN)
    return [_PALETTE[(offset + i) % PALETTE_LEN] for i in range(n)]


def sample_scene(seed: int) -> dict:
    """Deterministic scene dict from an integer seed."""
    rng = random.Random(seed)
    n = SUPPORTED_N[rng.randrange(len(SUPPORTED_N))]
    # Face radius varies independently of n so object size does not cue the
    # count; the whole disk is always filled so total ink is constant in n.
    face_radius = FACE_RADIUS + rng.uniform(-6.0, 8.0)
    colors = _palette_for_random(rng, n)
    return {
        "n": n,
        "face_radius": round(face_radius, 3),
        "colors": colors,
        "canvas": CANVAS,
    }


def render(scene: dict) -> Image.Image:
    """Render the concentric target face to an RGB image."""
    img = Image.new("RGB", (int(scene["canvas"]), int(scene["canvas"])), BACKGROUND)
    draw = ImageDraw.Draw(img)
    n = int(scene["n"])
    face_r = float(scene["face_radius"])
    colors = [tuple(c) for c in scene["colors"]]
    # Draw from the outermost band to the innermost disk so each successive,
    # smaller disk paints its own ring on top. Result: n concentric bands.
    for i in range(n, 0, -1):
        color = colors[n - i]
        radius = face_r * (i / n)
        draw.ellipse(
            [CX - radius, CY - radius, CX + radius, CY + radius], fill=color
        )
    return img


def analytic_gold(scene: dict) -> str:
    """Decision = the visible number of painted rings, as a string."""
    return str(scene["n"])


def margin(scene: dict) -> float:
    """Verifiability margin = the thinnest painted ring band in pixels."""
    radius = float(scene.get("face_radius", FACE_RADIUS))
    return radius / float(scene["n"])


def is_quarantined(scene: dict) -> bool:
    """Quarantine ring counts whose thinnest band is ambiguous at pixel scale."""
    if int(scene["n"]) not in SUPPORTED_N:
        return True
    return margin(scene) < MIN_RING_WIDTH


def latent_symmetries(scene: dict):
    """not_applicable: empty, the latent rendering is injective."""
    return []
