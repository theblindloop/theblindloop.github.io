"""Independent image-only inverse for coupled symbol-weight balances."""

from __future__ import annotations

from fractions import Fraction

import numpy as np
from PIL import Image

CANVAS = (1000, 760)
INK = (31, 38, 46)
PANEL = (247, 249, 252)
SYMBOL_RGB = ((210, 61, 67), (38, 111, 187), (47, 151, 91), (126, 73, 181))
TAG_RGB = {"amber": (226, 151, 28), "cyan": (18, 157, 181), "violet": (141, 77, 190)}
ALLOWED = {(255, 255, 255), PANEL, INK, *SYMBOL_RGB, *TAG_RGB.values()}
ROW_Y = (42, 185, 328)
LEFT_BOXES = [(50, y + 8, 430, y + 104) for y in ROW_Y]
RIGHT_BOXES = [(565, y + 8, 945, y + 104) for y in ROW_Y]
QUERY = (50, 530, 260, 690)
CANDIDATES = [(372, 530, 533, 690), (582, 530, 743, 690), (792, 530, 953, 690)]


class OracleError(ValueError):
    pass


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out = []
    for y, x in zip(*np.nonzero(mask & ~seen)):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        xs, ys = [], []
        while stack:
            cy, cx = stack.pop()
            xs.append(cx); ys.append(cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        out.append((min(xs), min(ys), max(xs), max(ys), len(xs)))
    return out


def _read_group(arr: np.ndarray, box: tuple[int, int, int, int]) -> list[int]:
    x0, y0, x1, y1 = box
    vector = []
    for color in SYMBOL_RGB:
        mask = np.all(arr[y0:y1, x0:x1] == color, axis=2)
        comps = [c for c in _components(mask) if c[4] > 500]
        if any(not (650 <= c[4] <= 1500 and 30 <= c[2] - c[0] + 1 <= 47 and 30 <= c[3] - c[1] + 1 <= 47) for c in comps):
            raise OracleError("malformed symbol")
        leftover = int(mask.sum()) - sum(c[4] for c in comps)
        if leftover > 12:
            raise OracleError("damaged symbol fragments")
        vector.append(len(comps))
    if not 2 <= sum(vector) <= 6:
        raise OracleError("invalid group size")
    return vector


def _solve(rows: list[list[int]]) -> tuple[Fraction, ...]:
    a = [[Fraction(v) for v in row] for row in rows]
    pivots = []
    r = 0
    for col in range(4):
        pivot = next((i for i in range(r, len(a)) if a[i][col]), None)
        if pivot is None:
            continue
        a[r], a[pivot] = a[pivot], a[r]
        scale = a[r][col]
        a[r] = [v / scale for v in a[r]]
        for i in range(len(a)):
            if i != r and a[i][col]:
                scale = a[i][col]
                a[i] = [v - scale * q for v, q in zip(a[i], a[r])]
        pivots.append(col)
        r += 1
    if r != 3:
        raise OracleError("balance equations are rank deficient")
    free = next((c for c in range(4) if c not in pivots), None)
    if free is None:
        raise OracleError("no free ratio scale")
    values = [Fraction(0) for _ in range(4)]
    values[free] = Fraction(1)
    for row_index in range(2, -1, -1):
        col = pivots[row_index]
        values[col] = -sum(a[row_index][j] * values[j] for j in range(col + 1, 4))
    if any(v <= 0 for v in values):
        raise OracleError("symbol weights are not positive")
    return tuple(values)


def _total(vector: list[int], weights: tuple[Fraction, ...]) -> Fraction:
    return sum((Fraction(n) * w for n, w in zip(vector, weights)), Fraction(0))


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    if rgb.size != CANVAS:
        raise OracleError("wrong canvas")
    colors = rgb.getcolors(maxcolors=32)
    if colors is None or not {tuple(c) for _, c in colors} <= ALLOWED:
        raise OracleError("unknown raster color")
    arr = np.asarray(rgb)
    equations = []
    for left_box, right_box, y in zip(LEFT_BOXES, RIGHT_BOXES, ROW_Y):
        left, right = _read_group(arr, left_box), _read_group(arr, right_box)
        if int(np.all(arr[y + 44:y + 69, 470:530] == INK, axis=2).sum()) < 400:
            raise OracleError("equality mark is missing")
        equations.append([a - b for a, b in zip(left, right)])
    weights = _solve(equations)
    query = _read_group(arr, QUERY)
    target = _total(query, weights)
    matches = []
    seen = set()
    for box in CANDIDATES:
        vector = _read_group(arr, box)
        x0, y0, _, _ = box
        sample = tuple(int(v) for v in arr[y0 - 6, x0 + 60])
        names = [name for name, color in TAG_RGB.items() if sample == color]
        if len(names) != 1 or names[0] in seen:
            raise OracleError("candidate borders are invalid")
        seen.add(names[0])
        if sum(vector) != sum(query):
            raise OracleError("candidate and query item counts differ")
        if _total(vector, weights) == target:
            matches.append(names[0])
    if seen != set(TAG_RGB) or len(matches) != 1:
        raise OracleError("no unique balancing candidate")
    return matches[0]
