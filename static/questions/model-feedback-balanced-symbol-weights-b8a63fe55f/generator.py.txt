"""Deterministic renderer for coupled symbol-weight balances."""

from __future__ import annotations

import math
import random
from itertools import combinations, product

from PIL import Image, ImageDraw

CANVAS = (1000, 760)
ANSWERS = ("amber", "cyan", "violet")
WEIGHT_FAMILIES = (
    (2, 3, 5, 7), (2, 4, 5, 9), (3, 4, 6, 7), (3, 5, 7, 8),
    (2, 5, 6, 9), (4, 5, 7, 9), (2, 3, 7, 8), (3, 4, 5, 9),
)
RGB = {
    "background": (255, 255, 255), "panel": (247, 249, 252), "ink": (31, 38, 46),
    "red": (210, 61, 67), "blue": (38, 111, 187), "green": (47, 151, 91),
    "purple": (126, 73, 181), "amber": (226, 151, 28),
    "cyan": (18, 157, 181), "violet": (141, 77, 190),
}
SYMBOL_COLORS = ("red", "blue", "green", "purple")
ROW_Y = (42, 185, 328)
LEFT_X, RIGHT_X, ROW_W, ROW_H = 35, 550, 410, 112
QUERY_BOX = (35, 520, 275, 700)
CANDIDATE_X = (360, 570, 780)
CANDIDATE_Y, CANDIDATE_W, CANDIDATE_H = 520, 185, 180


def _rank(rows: list[tuple[int, ...]]) -> int:
    a = [[float(x) for x in row] for row in rows]
    rank = 0
    for col in range(4):
        pivot = next((i for i in range(rank, len(a)) if abs(a[i][col]) > 1e-9), None)
        if pivot is None:
            continue
        a[rank], a[pivot] = a[pivot], a[rank]
        scale = a[rank][col]
        a[rank] = [x / scale for x in a[rank]]
        for i in range(len(a)):
            if i != rank and abs(a[i][col]) > 1e-9:
                scale = a[i][col]
                a[i] = [x - scale * y for x, y in zip(a[i], a[rank])]
        rank += 1
    return rank


def _det3(m: list[list[int]]) -> int:
    return (m[0][0] * (m[1][1] * m[2][2] - m[1][2] * m[2][1])
            - m[0][1] * (m[1][0] * m[2][2] - m[1][2] * m[2][0])
            + m[0][2] * (m[1][0] * m[2][1] - m[1][1] * m[2][0]))


def _infer_weights(equations: list[list[list[int]]]) -> tuple[int, ...]:
    rows = [[int(a) - int(b) for a, b in zip(left, right)] for left, right in equations]
    if _rank([tuple(row) for row in rows]) != 3:
        raise ValueError("equations do not determine one ratio")
    null = []
    for omitted in range(4):
        minor = [[row[j] for j in range(4) if j != omitted] for row in rows]
        null.append(((-1) ** omitted) * _det3(minor))
    if all(x < 0 for x in null):
        null = [-x for x in null]
    if any(x <= 0 for x in null):
        raise ValueError("inferred weights are not positive")
    divisor = math.gcd(*null)
    return tuple(x // divisor for x in null)


def _dot(v: tuple[int, ...] | list[int], w: tuple[int, ...]) -> int:
    return sum(int(a) * b for a, b in zip(v, w))


def sample_scene(seed: int) -> dict:
    rng = random.Random((int(seed) ^ 0xBA1A9CE5) & 0xFFFFFFFFFFFFFFFF)
    weights = WEIGHT_FAMILIES[int(seed) % len(WEIGHT_FAMILIES)]
    vectors = [v for v in product(range(4), repeat=4) if 2 <= sum(v) <= 6]
    by_total: dict[int, list[tuple[int, ...]]] = {}
    for vector in vectors:
        by_total.setdefault(_dot(vector, weights), []).append(vector)
    pairs = []
    for group in by_total.values():
        for left, right in combinations(group, 2):
            if all(not (a and b) for a, b in zip(left, right)):
                diff = tuple(a - b for a, b in zip(left, right))
                pairs.append((left, right, diff))
    rng.shuffle(pairs)
    chosen = []
    for left, right, diff in pairs:
        if _rank([item[2] for item in chosen] + [diff]) > len(chosen):
            chosen.append((left, right, diff))
        if len(chosen) == 3:
            break
    if len(chosen) != 3:
        raise RuntimeError("could not construct independent balances")
    equations = []
    for left, right, _ in chosen:
        if rng.randrange(2):
            left, right = right, left
        equations.append([list(left), list(right)])
    rng.shuffle(equations)

    query_pool = []
    for query in vectors:
        if sum(query) not in (4, 5):
            continue
        equal = [v for v in vectors if sum(v) == sum(query) and v != query and _dot(v, weights) == _dot(query, weights)]
        if equal:
            query_pool.append((query, equal))
    query, equal = query_pool[rng.randrange(len(query_pool))]
    correct = equal[rng.randrange(len(equal))]
    wrong_pool = sorted(
        (v for v in vectors if sum(v) == sum(query) and v != query and v != correct and _dot(v, weights) != _dot(query, weights)),
        key=lambda v: (abs(_dot(v, weights) - _dot(query, weights)), v),
    )
    nearest_gap = abs(_dot(wrong_pool[0], weights) - _dot(query, weights))
    near = [v for v in wrong_pool if abs(_dot(v, weights) - _dot(query, weights)) <= nearest_gap + 2]
    rng.shuffle(near)
    wrong = near[:2] if len(near) >= 2 else wrong_pool[:2]
    candidates = [wrong[0], wrong[1]]
    target_slot = int(seed) % 3
    candidates.insert(target_slot, correct)

    answer_color = ANSWERS[(int(seed) // 3) % 3]
    other_colors = [c for c in ANSWERS if c != answer_color]
    rng.shuffle(other_colors)
    tag_colors = [None, None, None]
    tag_colors[target_slot] = answer_color
    for slot in range(3):
        if tag_colors[slot] is None:
            tag_colors[slot] = other_colors.pop()
    scene = {
        "equations": equations,
        "query": list(query),
        "candidates": [list(v) for v in candidates],
        "tag_colors": tag_colors,
    }
    if _infer_weights(equations) != weights:
        raise RuntimeError("constructed balances infer the wrong ratios")
    return scene


def _symbol(draw: ImageDraw.ImageDraw, kind: int, cx: int, cy: int) -> None:
    color = RGB[SYMBOL_COLORS[kind]]
    if kind == 0:
        draw.ellipse((cx - 20, cy - 20, cx + 20, cy + 20), fill=color, outline=RGB["ink"], width=3)
    elif kind == 1:
        draw.rectangle((cx - 19, cy - 19, cx + 19, cy + 19), fill=color, outline=RGB["ink"], width=3)
    elif kind == 2:
        draw.polygon(((cx, cy - 23), (cx - 23, cy + 19), (cx + 23, cy + 19)), fill=color)
        draw.line(((cx, cy - 23), (cx - 23, cy + 19), (cx + 23, cy + 19), (cx, cy - 23)), fill=RGB["ink"], width=3, joint="curve")
    else:
        draw.polygon(((cx, cy - 24), (cx - 24, cy), (cx, cy + 24), (cx + 24, cy)), fill=color)
        draw.line(((cx, cy - 24), (cx - 24, cy), (cx, cy + 24), (cx + 24, cy), (cx, cy - 24)), fill=RGB["ink"], width=3, joint="curve")


def _group(draw: ImageDraw.ImageDraw, vector: list[int], box: tuple[int, int, int, int]) -> None:
    items = [kind for kind, count in enumerate(vector) for _ in range(int(count))]
    x0, y0, x1, y1 = box
    if len(items) <= 6 and y1 - y0 < 130:
        spacing = 58
        start = (x0 + x1 - spacing * (len(items) - 1)) // 2
        for i, kind in enumerate(items):
            _symbol(draw, kind, start + i * spacing, (y0 + y1) // 2)
    else:
        columns = 3
        spacing = min(65, (x1 - x0 - 58) // 2)
        start = (x0 + x1 - 2 * spacing) // 2
        for i, kind in enumerate(items):
            col, row = i % columns, i // columns
            cx = start + col * spacing
            cy = y0 + 48 + row * 72
            _symbol(draw, kind, cx, cy)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS, RGB["background"])
    draw = ImageDraw.Draw(image)
    for y, equation in zip(ROW_Y, scene["equations"]):
        draw.rounded_rectangle((LEFT_X, y, LEFT_X + ROW_W, y + ROW_H), radius=16, fill=RGB["panel"], outline=RGB["ink"], width=3)
        draw.rounded_rectangle((RIGHT_X, y, RIGHT_X + ROW_W, y + ROW_H), radius=16, fill=RGB["panel"], outline=RGB["ink"], width=3)
        _group(draw, equation[0], (LEFT_X + 15, y + 8, LEFT_X + ROW_W - 15, y + ROW_H - 8))
        _group(draw, equation[1], (RIGHT_X + 15, y + 8, RIGHT_X + ROW_W - 15, y + ROW_H - 8))
        cy = y + ROW_H // 2
        draw.line((472, cy - 11, 528, cy - 11), fill=RGB["ink"], width=7)
        draw.line((472, cy + 11, 528, cy + 11), fill=RGB["ink"], width=7)

    draw.rounded_rectangle(QUERY_BOX, radius=18, fill=RGB["panel"], outline=RGB["ink"], width=5)
    _group(draw, scene["query"], (QUERY_BOX[0] + 15, QUERY_BOX[1] + 10, QUERY_BOX[2] - 15, QUERY_BOX[3] - 10))
    draw.line((305, 576, 335, 576), fill=RGB["ink"], width=8)
    draw.line((305, 606, 335, 606), fill=RGB["ink"], width=8)
    draw.ellipse((314, 633, 326, 645), fill=RGB["ink"])
    for slot, x in enumerate(CANDIDATE_X):
        color = str(scene["tag_colors"][slot])
        box = (x, CANDIDATE_Y, x + CANDIDATE_W, CANDIDATE_Y + CANDIDATE_H)
        draw.rounded_rectangle(box, radius=18, fill=RGB["panel"], outline=RGB[color], width=9)
        _group(draw, scene["candidates"][slot], (x + 12, CANDIDATE_Y + 8, x + CANDIDATE_W - 12, CANDIDATE_Y + CANDIDATE_H - 8))
    return image


def analytic_gold(scene: dict) -> str:
    weights = _infer_weights(scene["equations"])
    target = _dot(scene["query"], weights)
    matches = [i for i, candidate in enumerate(scene["candidates"]) if _dot(candidate, weights) == target]
    if len(matches) != 1:
        raise ValueError("no unique balancing candidate")
    return str(scene["tag_colors"][matches[0]])


def margin(scene: dict) -> float:
    weights = _infer_weights(scene["equations"])
    target = _dot(scene["query"], weights)
    gaps = sorted(abs(_dot(candidate, weights) - target) for candidate in scene["candidates"])
    return float(gaps[1]) if gaps and gaps[0] == 0 else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 1 or analytic_gold(scene) not in ANSWERS or set(scene["tag_colors"]) != set(ANSWERS)
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
