"""Latent sampler, renderer, and analytic arm for balanced cut station."""

from __future__ import annotations

import random
from collections import deque
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 720, 400
BG = (250, 249, 245)
LINK = (105, 112, 120)
CYAN = (38, 176, 190)
OUTLINE = (30, 34, 38)
COLORS = {
    "orange": (232, 126, 32),
    "green": (35, 158, 78),
    "purple": (145, 76, 188),
}


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    # A 15-vertex caterpillar. Candidate vertices 2, 4, 6 all have degree two.
    edges = [[i, i + 1] for i in range(8)]
    edges += [[0, 9], [1, 10], [3, 11], [5, 12], [7, 13], [8, 14]]
    base = [[82 + 69 * i, 201 + rng.randint(-9, 9)] for i in range(9)]
    leaf_hubs = [0, 1, 3, 5, 7, 8]
    for j, hub in enumerate(leaf_hubs):
        direction = -1 if j % 2 == 0 else 1
        base.append([base[hub][0] + rng.randint(-15, 15), base[hub][1] + direction * rng.randint(82, 105)])
    # Reflections and modest translations counterbalance screen position.
    if rng.randrange(2):
        base = [[WIDTH - x, y] for x, y in base]
    if rng.randrange(2):
        base = [[x, HEIGHT - y] for x, y in base]
    dx, dy = rng.randint(-18, 18), rng.randint(-16, 16)
    base = [[x + dx, y + dy] for x, y in base]
    labels = ["orange", "green", "purple"]
    rng.shuffle(labels)
    candidate_labels = {str(node): label for node, label in zip((2, 4, 6), labels)}
    return {"edges": edges, "positions": base, "candidate_labels": candidate_labels}


def render(scene: dict) -> Image.Image:
    edges = scene["edges"]
    positions = scene["positions"]
    candidate_labels = scene["candidate_labels"]
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    for a, b in edges:
        draw.line([tuple(positions[a]), tuple(positions[b])], fill=OUTLINE, width=12)
    for a, b in edges:
        draw.line([tuple(positions[a]), tuple(positions[b])], fill=LINK, width=8)
    for node, (x, y) in enumerate(positions):
        label = candidate_labels.get(str(node))
        fill = COLORS[label] if label else CYAN
        draw.ellipse((x - 15, y - 15, x + 15, y + 15), fill=OUTLINE)
        draw.ellipse((x - 11, y - 11, x + 11, y + 11), fill=fill)
    return image


def _component_sizes(edges: list[list[int]], removed: int) -> list[int]:
    vertices = {v for edge in edges for v in edge} - {removed}
    adjacency = {v: set() for v in vertices}
    for a, b in edges:
        if removed not in (a, b):
            adjacency[a].add(b)
            adjacency[b].add(a)
    sizes = []
    unseen = set(vertices)
    while unseen:
        start = next(iter(unseen))
        queue = deque([start])
        unseen.remove(start)
        size = 0
        while queue:
            v = queue.popleft()
            size += 1
            for n in adjacency[v]:
                if n in unseen:
                    unseen.remove(n)
                    queue.append(n)
        sizes.append(size)
    return sorted(sizes)


def analytic_gold(scene: dict) -> str:
    edges = scene["edges"]
    candidate_labels = scene["candidate_labels"]
    winners = []
    for node_text, label in candidate_labels.items():
        sizes = _component_sizes(edges, int(node_text))
        if len(sizes) == 2 and sizes[0] == sizes[1]:
            winners.append(label)
    if len(winners) != 1:
        raise ValueError("scene does not have a unique balanced cut")
    return winners[0]


def margin(scene: dict) -> float:
    imbalances = []
    for node_text in scene["candidate_labels"]:
        sizes = _component_sizes(scene["edges"], int(node_text))
        if len(sizes) == 2:
            imbalances.append(abs(sizes[0] - sizes[1]))
    positive = [value for value in imbalances if value > 0]
    return float(min(positive)) if len(imbalances) == 3 and positive else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        if margin(scene) < 2 or len(scene["positions"]) != 15:
            return True
        return analytic_gold(scene) not in COLORS
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = {
        "edges": [list(edge) for edge in reversed(scene["edges"])],
        "positions": [list(p) for p in scene["positions"]],
        "candidate_labels": dict(scene["candidate_labels"]),
    }
    return [("reverse_undirected_edge_serialization", twin)]
