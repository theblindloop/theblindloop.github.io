"""Independent image-only inverse arm."""

from __future__ import annotations

from collections import deque
import numpy as np

PALETTES = {
    "orange": np.array([232, 126, 32]),
    "green": np.array([35, 158, 78]),
    "purple": np.array([145, 76, 188]),
    "cyan": np.array([38, 176, 190]),
}
LINK = np.array([105, 112, 120])


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y, x in zip(*np.nonzero(mask & ~seen)):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        xs, ys = [], []
        while stack:
            cy, cx = stack.pop()
            xs.append(cx); ys.append(cy)
            for ny, nx in ((cy-1,cx),(cy+1,cx),(cy,cx-1),(cy,cx+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(xs) >= 80:
            result.append((sum(xs) / len(xs), sum(ys) / len(ys), len(xs)))
    return result


def _has_link(rgb: np.ndarray, p: tuple[float, float], q: tuple[float, float]) -> bool:
    x1, y1 = p; x2, y2 = q
    distance = float(np.hypot(x2-x1, y2-y1))
    if distance < 38:
        return False
    samples = max(12, int(distance - 34))
    hits = 0
    total = 0
    for t in np.linspace(17.0/distance, 1.0-17.0/distance, samples):
        x = x1 + t*(x2-x1); y = y1 + t*(y2-y1)
        ix, iy = int(round(x)), int(round(y))
        patch = rgb[max(0,iy-2):iy+3, max(0,ix-2):ix+3]
        if patch.size:
            delta = np.max(np.abs(patch.astype(int) - LINK), axis=2)
            hits += int(np.any(delta <= 4))
            total += 1
    return total > 0 and hits / total > 0.90


def _sizes(adjacency: list[set[int]], removed: int) -> list[int]:
    unseen = set(range(len(adjacency))) - {removed}
    sizes = []
    while unseen:
        start = unseen.pop(); queue = deque([start]); size = 0
        while queue:
            v = queue.popleft(); size += 1
            for n in adjacency[v]:
                if n != removed and n in unseen:
                    unseen.remove(n); queue.append(n)
        sizes.append(size)
    return sorted(sizes)


def decision_from_image(image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    nodes = []
    labels = {}
    for name, color in PALETTES.items():
        delta = np.max(np.abs(rgb.astype(int) - color), axis=2)
        for x, y, area in _components(delta <= 5):
            index = len(nodes)
            nodes.append((x, y))
            if name != "cyan":
                labels[index] = name
    if len(nodes) != 15 or len(labels) != 3:
        raise ValueError("could not recover fifteen stations and three candidates")
    adjacency = [set() for _ in nodes]
    for i in range(len(nodes)):
        for j in range(i+1, len(nodes)):
            if _has_link(rgb, nodes[i], nodes[j]):
                adjacency[i].add(j); adjacency[j].add(i)
    if sum(map(len, adjacency)) // 2 != 14:
        raise ValueError("could not recover the fourteen-link tree")
    winners = []
    for node, label in labels.items():
        sizes = _sizes(adjacency, node)
        if len(sizes) == 2 and sizes[0] == sizes[1]:
            winners.append(label)
    if len(winners) != 1:
        raise ValueError("pixel graph has no unique balanced cut")
    return winners[0]
