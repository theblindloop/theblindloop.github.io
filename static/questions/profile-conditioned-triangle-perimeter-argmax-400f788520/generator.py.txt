"""Triangle perimeter selection world: which colored triangle has the largest perimeter.

Latent scene z carries three colored filled triangles (one red, one blue, one
green), each confined to its own separate horizontal column so their ink never
overlaps. Every triangle is scalene with three independently sampled integer
corner positions. The decision is a measurement: recover each triangle's three
corners, compute the three pairwise Euclidean side lengths, sum them into the
perimeter (the outline length), and select the color of the triangle with the
largest perimeter. The winner is counterbalanced over which color owns the
largest perimeter, the arbitrary corner geometry, the column of each triangle,
and its absolute position.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "triangle_perimeter_argmax-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
BACKGROUND = (250, 250, 250)

WIDTH = 480
HEIGHT = 340
# Horizontal column x-ranges (outer bounds within which corners are sampled).
COL_X = ((30, 150), (180, 300), (330, 450))
Y_RANGE = (40, 300)
MIN_SPREAD = 30        # minimum bounding-box extent (px) in both dimensions
MIN_AREA = 1200.0      # minimum signed-area magnitude to avoid near-degenerate triangles
MIN_MARGIN_PX = 16.0   # minimum top-two perimeter gap (px) for a decidable selection
QUARANTINE_PX = 8.0    # below this top-two perimeter gap the scene is quarantined


def _side(a, b) -> float:
    return math.hypot(a[0] - b[0], a[1] - b[1])


def _perimeter(corners) -> float:
    return _side(corners[0], corners[1]) + _side(corners[1], corners[2]) + _side(
        corners[2], corners[0]
    )


def _area2(corners) -> float:
    (x0, y0), (x1, y1), (x2, y2) = corners
    return 0.5 * abs((x1 - x0) * (y2 - y0) - (x2 - x0) * (y1 - y0))


def _valid_triangle(corners) -> bool:
    xs = [c[0] for c in corners]
    ys = [c[1] for c in corners]
    if (max(xs) - min(xs)) < MIN_SPREAD or (max(ys) - min(ys)) < MIN_SPREAD:
        return False
    return _area2(corners) >= MIN_AREA


def _sample_triangle(rng: random.Random, x0: int, x1: int) -> list:
    while True:
        corners = [
            (rng.randint(x0, x1), rng.randint(Y_RANGE[0], Y_RANGE[1])) for _ in range(3)
        ]
        if _valid_triangle(corners):
            return corners


def sample_scene(seed: int) -> dict:
    """Sample a deterministic scene with a guaranteed, counterbalanced extremum."""
    rng = random.Random(seed * 1009 + 7)
    colors = list(PALETTE)
    rng.shuffle(colors)
    columns = list(range(len(PALETTE)))
    rng.shuffle(columns)

    while True:
        triangles = []
        for color, col in zip(colors, columns):
            corners = _sample_triangle(rng, COL_X[col][0], COL_X[col][1])
            triangles.append({"color": color, "col": int(col), "corners": corners})
        perimeters = sorted(_perimeter(t["corners"]) for t in triangles)
        if perimeters[-1] - perimeters[-2] >= MIN_MARGIN_PX:
            break

    return {
        "triangles": triangles,
        "width": WIDTH,
        "height": HEIGHT,
        "background": "light",
    }


def render(scene: dict) -> Image.Image:
    """Render I = R(z): three colored triangles occupying separate columns."""
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    for tri in scene["triangles"]:
        draw.polygon(tri["corners"], fill=COLOR_RGB[tri["color"]])
    return img


def margin(scene: dict) -> float:
    """Verifiability margin: the top-two perimeter gap in pixels."""
    perimeters = sorted(_perimeter(t["corners"]) for t in scene["triangles"])
    return float(perimeters[-1] - perimeters[-2])


def is_quarantined(scene: dict) -> bool:
    """Near-boundary: the top-two perimeter gap is finite-resolution small."""
    return margin(scene) < QUARANTINE_PX


def analytic_gold(scene: dict) -> str:
    """The color of the largest-perimeter triangle (max summed outline length)."""
    best_color = PALETTE[0]
    best_perim = -1.0
    for tri in scene["triangles"]:
        p = _perimeter(tri["corners"])
        if p > best_perim:
            best_perim = p
            best_color = tri["color"]
    return best_color


def latent_symmetries(scene: dict) -> list:
    """not_applicable: no nontrivial pixel-identical latent symmetry.

    Every latent field of a triangle (color, column, and all three corner
    coordinates) is directly rendered as visible ink. Swapping two triangles'
    corner configurations changes which color holds the largest perimeter, and
    any global flip or translation turns the raster into a distinctly shifted
    image. There is no hidden field that could change the gold without changing
    the raster.
    """
    return []
