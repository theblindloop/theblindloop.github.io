"""Independent pixel-only inverse arm for the triangle perimeter world.

This module receives ONLY the rasterized PNG (plus the world's fixed public
palette) and re-measures, for each colored triangle, its three corner positions
from the color-segmented ink and composes them into a perimeter. It shares no
geometry with renderer.render() and never imports the renderer, prompts,
generator, verifier, or any analytical gold code.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "triangle_perimeter_argmax-oracle-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
BACKGROUND = (250, 250, 250)


def _nearest_color(arr: np.ndarray) -> np.ndarray:
    """Label every pixel to its nearest palette entry (segment colors + background)."""
    refs = np.stack(
        [np.array(COLOR_RGB[c], dtype=np.float32) for c in PALETTE]
        + [np.array(BACKGROUND, dtype=np.float32)],
        axis=0,
    )
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def _convex_hull(points):
    """Andrew monotone-chain convex hull of a point set (returns hull vertices)."""
    pts = sorted(set(points))
    if len(pts) < 3:
        return pts

    def cross(o, a, b):
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

    lower = []
    for p in pts:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    upper = []
    for p in reversed(pts):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    return lower[:-1] + upper[:-1]


def _polygon_length(vertices) -> float:
    if len(vertices) < 2:
        return 0.0
    total = 0.0
    for i in range(len(vertices)):
        x0, y0 = vertices[i]
        x1, y1 = vertices[(i + 1) % len(vertices)]
        total += math.hypot(x1 - x0, y1 - y0)
    return total


def decision_from_image(image: Image.Image) -> str:
    """Return the color of the largest-perimeter triangle from pixels alone."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _nearest_color(arr)
    best_color = PALETTE[0]
    best_perimeter = -1.0
    for idx, color in enumerate(PALETTE):
        ys, xs = np.nonzero(labels == idx)
        if xs.size == 0:
            continue
        points = [(int(x), int(y)) for x, y in zip(xs.tolist(), ys.tolist())]
        perimeter = _polygon_length(_convex_hull(points))
        if perimeter > best_perimeter:
            best_perimeter = perimeter
            best_color = color
    return best_color
