"""Independent raster-only inverse for guarded-skip replay.

The inverse locates saturated symbol columns, orders the components by their
observed centroids, classifies rings/triangles/disks by fill ratio, and replays
the decoded program.  It has no renderer import and uses no scene or metadata.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


EXPECTED_SYMBOLS = 9  # one start badge and eight instruction symbols


def _column_runs(active: np.ndarray) -> list[tuple[int, int]]:
    runs: list[tuple[int, int]] = []
    start: int | None = None
    for x, present in enumerate(active.tolist() + [False]):
        if present and start is None:
            start = x
        elif not present and start is not None:
            if x - start >= 18:
                runs.append((start, x))
            start = None
    return runs


def _fill_ratio(mask: np.ndarray, x0: int, x1: int) -> float:
    crop = mask[:, x0:x1]
    ys = np.flatnonzero(crop.any(axis=1))
    xs = np.flatnonzero(crop.any(axis=0))
    if len(xs) < 18 or len(ys) < 18:
        raise ValueError("symbol component is too small")
    tight = crop[ys[0] : ys[-1] + 1, xs[0] : xs[-1] + 1]
    return float(tight.mean())


def _decode(image: Image.Image) -> tuple[bool, list[str], float]:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    high = rgb.max(axis=2)
    low = rgb.min(axis=2)
    saturated = (high - low >= 70) & (high >= 95)

    # A shape contributes many saturated pixels in each occupied column.  Text,
    # borders, and nuisance codes are neutral and therefore absent from the mask.
    runs = _column_runs(saturated.sum(axis=0) >= 2)
    if len(runs) != EXPECTED_SYMBOLS:
        raise ValueError(f"expected {EXPECTED_SYMBOLS} visible symbols, found {len(runs)}")
    if any(runs[i + 1][0] - runs[i][1] < 18 for i in range(len(runs) - 1)):
        raise ValueError("symbol order is not visually separated")

    ratios = [_fill_ratio(saturated, x0, x1) for x0, x1 in runs]
    start_ratio = ratios[0]
    if start_ratio <= 0.46:
        state = False
        state_lead = 0.46 - start_ratio
    elif start_ratio >= 0.60:
        state = True
        state_lead = start_ratio - 0.60
    else:
        raise ValueError("start badge is ambiguous between ring and filled disk")

    operations: list[str] = []
    leads = [state_lead]
    for ratio in ratios[1:]:
        if ratio <= 0.59:
            operations.append("flip")
            leads.append(0.59 - ratio)
        elif ratio >= 0.66:
            operations.append("guard")
            leads.append(ratio - 0.66)
        else:
            raise ValueError("instruction symbol is ambiguous between triangle and circle")

    pc = 0
    while pc < len(operations):
        operation = operations[pc]
        if operation == "flip":
            state = not state
            pc += 1
        else:
            pc += 2 if state else 1
    return state, operations, 100.0 * min(leads)


def decision_from_image(image: Image.Image) -> str:
    state, _operations, _runner_up_margin = _decode(image)
    return "on" if state else "off"
