"""Renderer and analytic model for the guarded-skip replay world.

The scene shows one binary badge followed by eight ordered instruction cards.
Triangles flip the badge.  Circles are guards: when reached while the badge is
ON, they skip the next card.  Thus the running state affects the instruction
pointer, rather than merely deciding whether the current card is a no-op.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont


WIDTH = 1040
HEIGHT = 420
SCALE = 3
CARD_COUNT = 8

BACKGROUND = (249, 249, 246)
INK = (31, 35, 39)
MUTED = (105, 111, 117)
CARD_BORDER = (142, 147, 151)
DECORATION = (176, 181, 184)

# Each palette contains the same three saturated inks in a different order.
# Operation identity is carried by shape, not by hue.
PALETTES = (
    ((12, 145, 166), (230, 119, 30), (125, 72, 184)),
    ((125, 72, 184), (12, 145, 166), (230, 119, 30)),
    ((230, 119, 30), (125, 72, 184), (12, 145, 166)),
    ((18, 126, 91), (211, 78, 91), (55, 112, 190)),
)


def _font(size: int, *, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(name, size * SCALE)


def _program_result(initial_on: bool, operations: list[str]) -> bool:
    state = bool(initial_on)
    pc = 0
    while pc < len(operations):
        operation = operations[pc]
        if operation == "flip":
            state = not state
            pc += 1
        elif operation == "guard":
            pc += 2 if state else 1
        else:
            raise ValueError(f"unknown operation {operation!r}")
    return state


def _has_effectful_skip(initial_on: bool, operations: list[str]) -> bool:
    """True when an executed ON-guard visibly skips at least one FLIP card."""

    state = bool(initial_on)
    pc = 0
    skipped_flips = 0
    executed_guards_off = 0
    while pc < len(operations):
        operation = operations[pc]
        if operation == "flip":
            state = not state
            pc += 1
        elif operation == "guard":
            if state and pc + 1 < len(operations):
                skipped_flips += operations[pc + 1] == "flip"
                pc += 2
            else:
                executed_guards_off += not state
                pc += 1
        else:
            return False
    return skipped_flips > 0 and executed_guards_off > 0


def _balanced_program(rng: random.Random, initial_on: bool, target_on: bool) -> list[str]:
    """Choose four cards of each kind while forcing the requested result.

    Fixed visible card counts block raw symbol count and total icon ink as answer
    proxies.  Every accepted program exercises both outcomes of the guard rule.
    """

    base = ["flip"] * 4 + ["guard"] * 4
    for _ in range(2000):
        candidate = list(base)
        rng.shuffle(candidate)
        if (
            _program_result(initial_on, candidate) == target_on
            and _has_effectful_skip(initial_on, candidate)
        ):
            return candidate
    raise RuntimeError("could not sample a balanced guarded-skip program")


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a deterministic scene; consecutive seeds alternate final class."""

    rng = random.Random(int(seed))
    target_on = bool(int(seed) % 2)
    initial_on = bool((int(seed) // 2) % 2)
    operations = _balanced_program(rng, initial_on, target_on)

    palette = [list(color) for color in PALETTES[rng.randrange(len(PALETTES))]]
    if rng.randrange(2):
        palette = palette[1:] + palette[:1]
    decorations = list(range(CARD_COUNT))
    rng.shuffle(decorations)
    return {
        "initial_on": initial_on,
        "operations": operations,
        "offset_x": rng.randint(-8, 8),
        "offset_y": rng.randint(-5, 5),
        "card_gap": rng.randint(-2, 2),
        "icon_radius": rng.randint(24, 29),
        "ring_width": rng.randint(5, 7),
        "palette": palette,
        "decorations": decorations,
    }


def analytic_gold(scene: dict[str, Any]) -> str:
    state = _program_result(bool(scene["initial_on"]), list(scene["operations"]))
    return "on" if state else "off"


def margin(scene: dict[str, Any]) -> float:
    """Minimum visual decoding lead over the nearest rival interpretation.

    Terms are pixel/component-size clearance, horizontal separation, saturation
    clearance above the inverse threshold, and fill-ratio clearance between a
    ring, triangle, and disk.  The minimum is a conservative runner-up margin.
    """

    radius = float(scene["icon_radius"])
    ring_width = float(scene["ring_width"])
    spacing = 104.0 + float(scene["card_gap"])
    palette = scene["palette"]
    saturation_clearance = min(max(color) - min(color) - 70.0 for color in palette)
    component_clearance = 2.0 * radius - 38.0
    separation_clearance = spacing - 2.0 * radius - 24.0
    ring_ratio = math.pi * (2.0 * radius * ring_width - ring_width**2) / (4.0 * radius**2)
    ring_clearance = 100.0 * (0.50 - ring_ratio)
    triangle_clearance = 10.0
    disk_clearance = 100.0 * (math.pi / 4.0 - 0.66)
    return round(
        min(
            saturation_clearance,
            component_clearance,
            separation_clearance,
            ring_clearance,
            triangle_clearance,
            disk_clearance,
        ),
        6,
    )


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # All latent fields are visibly rendered, and no pixel-identical aliases are
    # claimed.  Pixel-changing nuisance transformations are tested separately.
    return []


def _centered_text(
    draw: ImageDraw.ImageDraw,
    xy: tuple[float, float],
    text: str,
    font: ImageFont.FreeTypeFont,
    fill: tuple[int, int, int],
) -> None:
    x, y = xy
    box = draw.textbbox((0, 0), text, font=font)
    width = box[2] - box[0]
    height = box[3] - box[1]
    draw.text((int(x * SCALE - width / 2), int(y * SCALE - height / 2 - box[1])), text, font=font, fill=fill)


def _draw_binary_decoration(
    draw: ImageDraw.ImageDraw, center_x: float, y: float, value: int
) -> None:
    """Render a small neutral three-bit nuisance code on a card."""

    for bit in range(3):
        x = center_x + (bit - 1) * 9
        if value & (1 << bit):
            draw.ellipse(
                ((x - 2) * SCALE, (y - 2) * SCALE, (x + 2) * SCALE, (y + 2) * SCALE),
                fill=DECORATION,
            )
        else:
            draw.ellipse(
                ((x - 2) * SCALE, (y - 2) * SCALE, (x + 2) * SCALE, (y + 2) * SCALE),
                outline=DECORATION,
                width=SCALE,
            )


def render(scene: dict[str, Any]) -> Image.Image:
    initial_on = bool(scene["initial_on"])
    operations = list(scene["operations"])
    offset_x = int(scene["offset_x"])
    offset_y = int(scene["offset_y"])
    card_gap = int(scene["card_gap"])
    icon_radius = int(scene["icon_radius"])
    ring_width = int(scene["ring_width"])
    palette = [tuple(int(channel) for channel in color) for color in scene["palette"]]
    decorations = list(scene["decorations"])

    if len(operations) != CARD_COUNT or sorted(operations) != ["flip"] * 4 + ["guard"] * 4:
        raise ValueError("scene must contain exactly four FLIP and four GUARD cards")
    if sorted(decorations) != list(range(CARD_COUNT)):
        raise ValueError("decorations must be a permutation of 0..7")
    if len(palette) != 3 or any(len(color) != 3 for color in palette):
        raise ValueError("palette must contain three RGB colors")

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(25, bold=True)
    label_font = _font(15, bold=True)
    small_font = _font(12, bold=True)

    _centered_text(draw, (WIDTH / 2, 34), "GUARDED SKIP PROGRAM", title_font, INK)
    _centered_text(
        draw,
        (WIDTH / 2, 73),
        "Read the numbered cards from left to right",
        _font(14),
        MUTED,
    )

    ox = offset_x
    oy = offset_y
    panel_y = 124 + oy
    panel_h = 212

    # START panel.
    start_left = 18 + ox
    start_right = 146 + ox
    draw.rounded_rectangle(
        (start_left * SCALE, panel_y * SCALE, start_right * SCALE, (panel_y + panel_h) * SCALE),
        radius=12 * SCALE,
        fill=(255, 255, 255),
        outline=CARD_BORDER,
        width=2 * SCALE,
    )
    _centered_text(draw, ((start_left + start_right) / 2, panel_y + 24), "START", small_font, MUTED)
    start_cx = (start_left + start_right) / 2
    icon_cy = panel_y + 84
    start_color = palette[0]
    bounds = (
        (start_cx - icon_radius) * SCALE,
        (icon_cy - icon_radius) * SCALE,
        (start_cx + icon_radius) * SCALE,
        (icon_cy + icon_radius) * SCALE,
    )
    if initial_on:
        draw.ellipse(bounds, fill=start_color)
    else:
        draw.ellipse(bounds, outline=start_color, width=ring_width * SCALE)
    _centered_text(
        draw,
        (start_cx, panel_y + 150),
        "ON" if initial_on else "OFF",
        label_font,
        INK,
    )
    _centered_text(draw, (start_cx, panel_y + 181), "badge", _font(12), MUTED)

    # Ordered operation cards.
    first_left = 168 + ox
    card_w = 82
    step = 104 + card_gap
    for index, (operation, decoration) in enumerate(zip(operations, decorations)):
        left = first_left + index * step
        right = left + card_w
        center_x = (left + right) / 2
        draw.rounded_rectangle(
            (left * SCALE, panel_y * SCALE, right * SCALE, (panel_y + panel_h) * SCALE),
            radius=10 * SCALE,
            fill=(255, 255, 255),
            outline=CARD_BORDER,
            width=2 * SCALE,
        )
        _centered_text(draw, (center_x, panel_y + 22), str(index + 1), small_font, MUTED)
        color = palette[1] if operation == "flip" else palette[2]
        if operation == "flip":
            draw.polygon(
                (
                    (center_x * SCALE, (icon_cy - icon_radius) * SCALE),
                    ((center_x - icon_radius) * SCALE, (icon_cy + icon_radius) * SCALE),
                    ((center_x + icon_radius) * SCALE, (icon_cy + icon_radius) * SCALE),
                ),
                fill=color,
            )
            label = "FLIP"
        else:
            draw.ellipse(
                (
                    (center_x - icon_radius) * SCALE,
                    (icon_cy - icon_radius) * SCALE,
                    (center_x + icon_radius) * SCALE,
                    (icon_cy + icon_radius) * SCALE,
                ),
                fill=color,
            )
            label = "GUARD"
        _centered_text(draw, (center_x, panel_y + 150), label, label_font, INK)
        _draw_binary_decoration(draw, center_x, panel_y + 186, int(decoration))

    _centered_text(
        draw,
        (WIDTH / 2, 378),
        "triangle = FLIP     circle = GUARD",
        _font(13, bold=True),
        MUTED,
    )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
