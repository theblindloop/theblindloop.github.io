"""Renderer for the rotor_window_dial world.

One compact rotor disc is drawn somewhere on an otherwise inert canvas. Eight
coloured tiles sit at eight evenly spaced stations of a ring. A solid black
bracket outside the ring marks the fixed read window. Inside the hub a short
stack of rotation glyphs gives the ordered turns: a triangle pointing the way
the ring turns, followed by pips giving how many stations that turn advances.

Replaying the turns from the printed ring brings a different tile under the
window; the decision is the colour of that tile.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "rotor_window_dial-renderer-0.1.0"

WIDTH = 760
HEIGHT = 560
BACKGROUND = (255, 255, 255)
INK = (0, 0, 0)
DECOR_A = (206, 206, 206)
DECOR_B = (223, 223, 223)

STATIONS = 8

COLOURS: tuple[tuple[str, tuple[int, int, int]], ...] = (
    ("red", (214, 38, 38)),
    ("orange", (232, 120, 20)),
    ("yellow", (222, 196, 20)),
    ("green", (34, 150, 54)),
    ("teal", (18, 158, 158)),
    ("blue", (36, 72, 204)),
    ("purple", (128, 44, 180)),
    ("pink", (226, 86, 158)),
)
COLOUR_NAMES = tuple(name for name, _ in COLOURS)
COLOUR_RGB = {name: rgb for name, rgb in COLOURS}

ROW_PITCH = 14
ARROW_X0 = -24
ARROW_HALF = 5
ARROW_LEN = 10
PIP_X0 = -8
PIP_PITCH = 9
PIP_HALF = 2

MARGIN_FLOOR = 6.0


# ---------------------------------------------------------------- sampling

def sample_scene(seed: int) -> dict:
    """Sample one rotor scene. Geometry is rejection sampled so that every
    scene is legible; the only quarantine bin that survives is the degenerate
    net turn."""
    rng = random.Random(seed)
    for _ in range(400):
        turns = [
            {"dir": rng.choice([1, -1]), "pips": rng.randint(1, 3)}
            for _ in range(rng.choice([3, 4]))
        ]
        scene = {
            "cx": rng.randint(200, 560),
            "cy": rng.randint(160, 400),
            "phase_deg": rng.uniform(0.0, 360.0 / STATIONS),
            "ring_r": rng.randint(50, 56),
            "tile_half": rng.choice([6, 7]),
            "window_r": rng.randint(70, 75),
            "window_index": rng.randrange(STATIONS),
            "tiles": _permutation(rng),
            "turns": turns,
            "decor": _decor(rng),
        }
        if _clearances(scene)["margin"] >= MARGIN_FLOOR:
            return scene
    raise RuntimeError("failed to sample a legible rotor scene")


def _permutation(rng: random.Random) -> list[str]:
    names = list(COLOUR_NAMES)
    rng.shuffle(names)
    return names


def _decor(rng: random.Random) -> list[dict]:
    """Inert pale grey furniture, kept well clear of the rotor disc."""
    items: list[dict] = []
    for _ in range(rng.randint(7, 11)):
        items.append(
            {
                "kind": rng.choice(["ring", "bar", "dot", "box"]),
                "x": rng.randint(30, WIDTH - 30),
                "y": rng.randint(30, HEIGHT - 30),
                "a": rng.randint(9, 26),
                "b": rng.randint(9, 26),
                "shade": rng.choice([0, 1]),
            }
        )
    return items


# ------------------------------------------------------------------ layout

def _station_angle(scene: dict, index: int) -> float:
    return math.radians(scene["phase_deg"] + 360.0 * index / STATIONS)


def _station_xy(scene: dict, index: int) -> tuple[float, float]:
    theta = _station_angle(scene, index)
    return (
        scene["cx"] + scene["ring_r"] * math.cos(theta),
        scene["cy"] + scene["ring_r"] * math.sin(theta),
    )


def _hub_reach(scene: dict) -> float:
    rows = len(scene["turns"])
    dy = ROW_PITCH * (rows - 1) / 2.0 + ARROW_HALF
    left = math.hypot(abs(ARROW_X0), dy)
    right = math.hypot(abs(PIP_X0) + 2 * PIP_PITCH + PIP_HALF + 2, dy)
    return max(left, right)


def _clearances(scene: dict) -> dict:
    tile_reach = scene["tile_half"] * math.sqrt(2.0)
    ring_r = scene["ring_r"]
    window_gap = scene["window_r"] - 3.0 - (ring_r + tile_reach)
    hub_gap = (ring_r - tile_reach) - _hub_reach(scene)
    chord = 2.0 * ring_r * math.sin(math.pi / STATIONS)
    tile_gap = chord - 2.0 * tile_reach
    return {
        "window_gap": window_gap,
        "hub_gap": hub_gap,
        "tile_gap": tile_gap,
        "margin": min(window_gap, hub_gap, tile_gap),
    }


def _net_turn(scene: dict) -> int:
    return sum(turn["dir"] * turn["pips"] for turn in scene["turns"])


def _replay(scene: dict) -> dict:
    net = _net_turn(scene)
    index = (scene["window_index"] - net) % STATIONS
    return {
        "net_turn": net,
        "landing_index": index,
        "colour": scene["tiles"][index],
        "resting_colour": scene["tiles"][scene["window_index"] % STATIONS],
    }


# ----------------------------------------------------------------- drawing

def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_decor(scene, draw)
    _draw_window(scene, draw)
    _draw_tiles(scene, draw)
    _draw_hub(scene, draw)
    return image


def _draw_decor(scene: dict, draw: ImageDraw.ImageDraw) -> None:
    keep_out = scene["window_r"] + 34
    for item in scene["decor"]:
        x, y = item["x"], item["y"]
        if math.hypot(x - scene["cx"], y - scene["cy"]) < keep_out:
            continue
        shade = DECOR_A if item["shade"] == 0 else DECOR_B
        a, b = item["a"], item["b"]
        if item["kind"] == "ring":
            draw.ellipse([x - a, y - a, x + a, y + a], outline=shade, width=3)
        elif item["kind"] == "bar":
            draw.rectangle([x - a, y - 4, x + a, y + 4], fill=shade)
        elif item["kind"] == "dot":
            draw.ellipse([x - 5, y - 5, x + 5, y + 5], fill=shade)
        else:
            draw.rectangle([x - a, y - b, x + a, y + b], outline=shade, width=3)


def _draw_window(scene: dict, draw: ImageDraw.ImageDraw) -> None:
    """A solid black bracket: an area-like annular wedge outside one station."""
    centre = math.radians(scene["phase_deg"] + 360.0 * scene["window_index"] / STATIONS)
    half = math.radians(15.0)
    r_in = float(scene["window_r"]) - 3.0
    r_out = float(scene["window_r"]) + 3.0
    steps = 16
    outer = []
    inner = []
    for i in range(steps + 1):
        theta = centre - half + 2.0 * half * i / steps
        outer.append((scene["cx"] + r_out * math.cos(theta), scene["cy"] + r_out * math.sin(theta)))
        inner.append((scene["cx"] + r_in * math.cos(theta), scene["cy"] + r_in * math.sin(theta)))
    draw.polygon(outer + list(reversed(inner)), fill=INK)


def _draw_tiles(scene: dict, draw: ImageDraw.ImageDraw) -> None:
    half = scene["tile_half"]
    for index, name in enumerate(scene["tiles"]):
        x, y = _station_xy(scene, index)
        cx, cy = int(round(x)), int(round(y))
        draw.rectangle([cx - half, cy - half, cx + half, cy + half], fill=COLOUR_RGB[name])


def _draw_hub(scene: dict, draw: ImageDraw.ImageDraw) -> None:
    rows = len(scene["turns"])
    cx, cy = scene["cx"], scene["cy"]
    for i, turn in enumerate(scene["turns"]):
        y = int(round(cy + ROW_PITCH * (i - (rows - 1) / 2.0)))
        x0 = cx + ARROW_X0
        if turn["dir"] > 0:
            tri = [(x0, y - ARROW_HALF), (x0, y + ARROW_HALF), (x0 + ARROW_LEN, y)]
        else:
            tri = [
                (x0 + ARROW_LEN, y - ARROW_HALF),
                (x0 + ARROW_LEN, y + ARROW_HALF),
                (x0, y),
            ]
        draw.polygon(tri, fill=INK)
        for j in range(turn["pips"]):
            px = cx + PIP_X0 + j * PIP_PITCH
            draw.rectangle([px - PIP_HALF, y - PIP_HALF, px + PIP_HALF, y + PIP_HALF], fill=INK)


# ------------------------------------------------------------------- gold

def analytic_gold(scene: dict) -> str:
    """Colour of the tile that finishes under the read window."""
    return _replay(scene)["colour"]


def margin(scene: dict) -> float:
    """Smallest legibility clearance in pixels between the parts that carry
    the decision: the window bracket, the ring of tiles, and the hub stack."""
    return round(float(_clearances(scene)["margin"]), 6)


def is_quarantined(scene: dict) -> bool:
    """Degenerate net turn (a whole number of full revolutions) leaves the
    resting tile under the window, which a non-replaying reader could copy;
    sub-floor clearances are illegible."""
    if _net_turn(scene) % STATIONS == 0:
        return True
    return _clearances(scene)["margin"] < MARGIN_FLOOR


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases are declared: every sampled field is painted."""
    return []
