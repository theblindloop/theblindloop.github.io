"""Independent image-only inverse for mirror-fold landmark images."""

from __future__ import annotations

import math
import numpy as np
from PIL import Image

PALETTE = {
    "red": (220, 55, 60),
    "blue": (55, 105, 220),
    "gold": (225, 165, 35),
    "magenta": (205, 45, 170),
    "fold": (82, 92, 104),
}


def _centroid(arr: np.ndarray, rgb: tuple[int, int, int], tolerance: float = 42.0) -> tuple[float, float]:
    delta = arr.astype(np.float32) - np.asarray(rgb, dtype=np.float32)
    mask = np.sqrt(np.sum(delta * delta, axis=2)) < tolerance
    ys, xs = np.nonzero(mask)
    if len(xs) < 25:
        raise ValueError("required visible color component is missing")
    return float(xs.mean()), float(ys.mean())


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    source = _centroid(arr, PALETTE["magenta"])
    # The fold has hundreds of exact interior pixels. A tight threshold avoids
    # admitting unrelated antialiased dark outlines from the landmarks.
    fold = _centroid(arr, PALETTE["fold"], tolerance=12.0)
    target = (2.0 * fold[0] - source[0], source[1])
    ranked = []
    for name in ("red", "blue", "gold"):
        center = _centroid(arr, PALETTE[name])
        distance = math.hypot(center[0] - target[0], center[1] - target[1])
        ranked.append((distance, name))
    ranked.sort()
    if ranked[1][0] - ranked[0][0] < 8.0:
        raise ValueError("ambiguous reflected landmark")
    return ranked[0][1]
