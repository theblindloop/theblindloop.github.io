"""Deterministic renderer for the mirror-fold landmark world."""

from __future__ import annotations

import math
import random
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 420
COLORS = ("red", "blue", "gold")
RGB = {
    "red": (220, 55, 60),
    "blue": (55, 105, 220),
    "gold": (225, 165, 35),
    "magenta": (205, 45, 170),
    "fold": (82, 92, 104),
}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    fold_x = rng.randint(285, 355)
    source_side = -1 if rng.randrange(2) == 0 else 1
    distance = rng.randint(72, 118)
    source_x = fold_x + source_side * distance
    source_y = rng.randint(105, 315)
    reflected_x = fold_x - source_side * distance

    # One lure preserves only height; the other preserves only fold distance.
    row_shift = rng.choice((-1, 1)) * rng.randint(48, 72)
    row_x = reflected_x + row_shift
    opposite_lo = fold_x + 28 if source_side < 0 else 45
    opposite_hi = 595 if source_side < 0 else fold_x - 28
    if not (opposite_lo <= row_x <= opposite_hi):
        row_x = reflected_x - row_shift
    vertical_shift = rng.choice((-1, 1)) * rng.randint(58, 88)
    distance_y = source_y + vertical_shift
    if not (55 <= distance_y <= 365):
        distance_y = source_y - vertical_shift

    centers = [(reflected_x, source_y), (row_x, source_y), (reflected_x, distance_y)]
    correct_color = COLORS[int(seed) % len(COLORS)]
    remaining = [c for c in COLORS if c != correct_color]
    rng.shuffle(remaining)
    assignment = [correct_color, *remaining]
    rng.shuffle(assignment)
    positions = [centers[0], centers[1], centers[2]]
    # Bind the correct color to the true reflected center after shuffling labels.
    options = []
    for color in assignment:
        index = 0 if color == correct_color else (1 if remaining.index(color) == 0 else 2)
        options.append({"color": color, "center": list(positions[index])})
    rng.shuffle(options)
    return {
        "fold_x": fold_x,
        "source": [source_x, source_y],
        "options": options,
        "canvas": [WIDTH, HEIGHT],
    }


def _dist(a: tuple[float, float] | list[float], b: tuple[float, float] | list[float]) -> float:
    return math.hypot(float(a[0]) - float(b[0]), float(a[1]) - float(b[1]))


def analytic_gold(scene: dict) -> str:
    target = (2 * float(scene["fold_x"]) - float(scene["source"][0]), float(scene["source"][1]))
    ranked = sorted(
        (_dist(item["center"], target), str(item["color"])) for item in scene["options"]
    )
    return ranked[0][1]


def margin(scene: dict) -> float:
    target = (2 * float(scene["fold_x"]) - float(scene["source"][0]), float(scene["source"][1]))
    distances = sorted(_dist(item["center"], target) for item in scene["options"])
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    width, height = [int(v) for v in scene.get("canvas", [WIDTH, HEIGHT])]
    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), (250, 249, 246))
    draw = ImageDraw.Draw(image)
    fold_x = int(scene["fold_x"])

    # Pale two-sided paper field and a high-contrast dashed fold.
    draw.rounded_rectangle(
        [18 * scale, 18 * scale, (width - 18) * scale, (height - 18) * scale],
        radius=14 * scale,
        fill=(255, 255, 255),
        outline=(215, 217, 220),
        width=2 * scale,
    )
    for y in range(36, height - 30, 24):
        draw.line(
            [(fold_x * scale, y * scale), (fold_x * scale, min(y + 13, height - 30) * scale)],
            fill=RGB["fold"],
            width=5 * scale,
        )

    sx, sy = [int(v) for v in scene["source"]]
    radius = 17
    diamond = [
        (sx * scale, (sy - radius) * scale),
        ((sx + radius) * scale, sy * scale),
        (sx * scale, (sy + radius) * scale),
        ((sx - radius) * scale, sy * scale),
    ]
    draw.polygon(diamond, fill=RGB["magenta"], outline=(110, 25, 95), width=2 * scale)
    draw.ellipse(
        [(sx - 4) * scale, (sy - 4) * scale, (sx + 4) * scale, (sy + 4) * scale],
        fill=(255, 255, 255),
    )

    for item in scene["options"]:
        x, y = [int(v) for v in item["center"]]
        color = RGB[str(item["color"])]
        r = 15
        draw.ellipse(
            [(x - r) * scale, (y - r) * scale, (x + r) * scale, (y + r) * scale],
            fill=color,
            outline=(45, 48, 52),
            width=2 * scale,
        )
        draw.ellipse(
            [(x - 3) * scale, (y - 3) * scale, (x + 3) * scale, (y + 3) * scale],
            fill=(255, 255, 255),
        )

    return image.resize((width, height), Image.Resampling.LANCZOS)
