"""Scene-free pixel oracle for corridor_farther_room.

Input is the final PNG and nothing else.  The oracle segments the traversable
white corridor region and the three coloured dots, propagates a breadth-first
wavefront from the green dot through corridor pixels only, and reports which of
the two target dots the wavefront reaches later, i.e. which one is farther when
distance is measured along the corridor.

It imports no renderer, prompt, generator, verifier or analytic-gold code and
never receives latent scene data.
"""

from __future__ import annotations

from collections import deque

from PIL import Image

ORACLE_VERSION = "corridor_farther_room-oracle-0.1.0"

ABSTAIN = "unknown"

# Reference hues used only to name a dot's colour once it has been segmented as
# a saturated blob; they are read from the pixels, not from any scene record.
_REFERENCES = {
    "green": (0, 150, 70),
    "blue": (30, 90, 235),
    "red": (220, 45, 45),
}

_CORRIDOR_MIN = 200          # a corridor pixel is bright in all channels
_WALL_MAX = 90               # a wall pixel is dark in all channels
_MIN_BLOB_PX = 40            # a dot is a solid blob, not a stray pixel
_MIN_STEP_MARGIN_PX = 40     # below this the pixel decision abstains


def _classify(px) -> str:
    r, g, b = px[0], px[1], px[2]
    if r >= _CORRIDOR_MIN and g >= _CORRIDOR_MIN and b >= _CORRIDOR_MIN:
        return "corridor"
    if r <= _WALL_MAX and g <= _WALL_MAX and b <= _WALL_MAX:
        return "wall"
    best, best_d = None, None
    for name, (rr, gg, bb) in _REFERENCES.items():
        d = (r - rr) ** 2 + (g - gg) ** 2 + (b - bb) ** 2
        if best_d is None or d < best_d:
            best, best_d = name, d
    if best_d is not None and best_d <= 120 * 120:
        return best
    return "wall"


def decision_from_image(image: Image.Image) -> str:
    img = image.convert("RGB")
    w, h = img.size
    px = img.load()

    passable = bytearray(w * h)
    dots: dict[str, list[int]] = {"green": [], "blue": [], "red": []}
    for y in range(h):
        row = y * w
        for x in range(w):
            kind = _classify(px[x, y])
            if kind == "wall":
                continue
            passable[row + x] = 1
            if kind != "corridor":
                dots[kind].append(row + x)

    for name, blob in dots.items():
        if len(blob) < _MIN_BLOB_PX:
            return ABSTAIN

    dist = [-1] * (w * h)
    queue: deque = deque()
    for idx in dots["green"]:
        dist[idx] = 0
        queue.append(idx)

    while queue:
        cur = queue.popleft()
        d = dist[cur] + 1
        y, x = divmod(cur, w)
        if x > 0 and passable[cur - 1] and dist[cur - 1] < 0:
            dist[cur - 1] = d
            queue.append(cur - 1)
        if x + 1 < w and passable[cur + 1] and dist[cur + 1] < 0:
            dist[cur + 1] = d
            queue.append(cur + 1)
        if y > 0 and passable[cur - w] and dist[cur - w] < 0:
            dist[cur - w] = d
            queue.append(cur - w)
        if y + 1 < h and passable[cur + w] and dist[cur + w] < 0:
            dist[cur + w] = d
            queue.append(cur + w)

    reach = {}
    for name in ("blue", "red"):
        seen = [dist[i] for i in dots[name] if dist[i] >= 0]
        if not seen:
            return ABSTAIN
        reach[name] = min(seen)

    gap = reach["blue"] - reach["red"]
    if abs(gap) < _MIN_STEP_MARGIN_PX:
        return ABSTAIN
    return "blue" if gap > 0 else "red"
