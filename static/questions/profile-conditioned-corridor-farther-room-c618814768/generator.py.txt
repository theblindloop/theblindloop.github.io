"""Scene sampler, renderer, analytic gold and margin for corridor_farther_room.

The world draws a loop-free corridor plan on a cell lattice in two colours:
white corridor on black wall.  Three dots sit in three distinct cells - one
green start dot and two target dots, one blue and one red.  The decision is
which target dot is farther from the green dot when distance is measured as
length walked along the corridor.

Sampling deliberately decorrelates the answer from every cheap local cue: in
half of the scenes the target that is nearer in a straight line across the page
is the one that is farther along the corridor, the answer colour is balanced,
and the two targets carry identical ink.
"""

from __future__ import annotations

import copy
import random
from collections import deque

from PIL import Image, ImageDraw

RENDERER_VERSION = "corridor_farther_room-renderer-0.1.0"

GRID = 9
CELL_PX = 7           # corridor stroke width
WALL_PX = 23          # black wall between two neighbouring corridor centres
MARGIN_PX = 16
DOT_RADIUS = 6
STEP_PX = CELL_PX + WALL_PX
CANVAS = 2 * MARGIN_PX + (GRID - 1) * STEP_PX + 1

WALL_RGB = (18, 18, 24)
CORRIDOR_RGB = (246, 246, 246)
GREEN_RGB = (0, 150, 70)
BLUE_RGB = (30, 90, 235)
RED_RGB = (220, 45, 45)

MIN_STEP_MARGIN = 4
MIN_START_DISTANCE = 2


# --------------------------------------------------------------------------
# lattice helpers
# --------------------------------------------------------------------------
def _carve(rng: random.Random, grid: int) -> list[list[list[int]]]:
    """Randomised depth-first carving of a loop-free (spanning-tree) plan."""
    visited = {(0, 0)}
    stack = [(0, 0)]
    passages: list[list[list[int]]] = []
    while stack:
        r, c = stack[-1]
        nbrs = []
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nr, nc = r + dr, c + dc
            if 0 <= nr < grid and 0 <= nc < grid and (nr, nc) not in visited:
                nbrs.append((nr, nc))
        if not nbrs:
            stack.pop()
            continue
        nr, nc = nbrs[rng.randrange(len(nbrs))]
        passages.append([[r, c], [nr, nc]])
        visited.add((nr, nc))
        stack.append((nr, nc))
    return passages


def _adjacency(grid: int, passages: list) -> dict:
    adj: dict = {(r, c): [] for r in range(grid) for c in range(grid)}
    for (ar, ac), (br, bc) in ((tuple(p[0]), tuple(p[1])) for p in passages):
        adj[(ar, ac)].append((br, bc))
        adj[(br, bc)].append((ar, ac))
    return adj


def _cell_distances(grid: int, passages: list, start: tuple) -> dict:
    adj = _adjacency(grid, passages)
    dist = {start: 0}
    queue = deque([start])
    while queue:
        cur = queue.popleft()
        for nxt in adj[cur]:
            if nxt not in dist:
                dist[nxt] = dist[cur] + 1
                queue.append(nxt)
    return dist


def _cell_center(cell) -> tuple[int, int]:
    r, c = int(cell[0]), int(cell[1])
    return MARGIN_PX + c * STEP_PX, MARGIN_PX + r * STEP_PX


def _plane_distance(cell_a, cell_b) -> float:
    ax, ay = _cell_center(cell_a)
    bx, by = _cell_center(cell_b)
    return ((ax - bx) ** 2 + (ay - by) ** 2) ** 0.5


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------
def sample_scene(seed: int) -> dict:
    """Sample one scene.

    Counterbalancing is driven by the seed itself so that any contiguous seed
    range is balanced: `want_conflict` decides whether the straight-line-nearer
    target is the corridor-farther one, `want_blue` decides the answer colour,
    and `want_margin` walks the accepted step margin through near-boundary and
    quarantined values.
    """
    rng = random.Random(int(seed) * 7919 + 13)
    want_conflict = (int(seed) % 2) == 0
    want_blue = ((int(seed) // 2) % 2) == 0
    phase = (int(seed) // 4) % 8
    if phase == 7:
        want_margin = 2 + (int(seed) // 32) % 2          # quarantined band
    elif phase in (3, 5):
        want_margin = MIN_STEP_MARGIN                     # near-boundary band
    else:
        want_margin = MIN_STEP_MARGIN + 1 + ((int(seed) // 4) % 5)

    for _ in range(400):
        passages = _carve(rng, GRID)
        cells = [(r, c) for r in range(GRID) for c in range(GRID)]
        for _ in range(60):
            start = cells[rng.randrange(len(cells))]
            dist = _cell_distances(GRID, passages, start)
            pool = [cell for cell, d in dist.items() if d >= MIN_START_DISTANCE]
            rng.shuffle(pool)
            for i in range(len(pool)):
                for j in range(i + 1, len(pool)):
                    a, b = pool[i], pool[j]
                    if abs(dist[a] - dist[b]) != want_margin:
                        continue
                    far, near = (a, b) if dist[a] > dist[b] else (b, a)
                    conflict = _plane_distance(start, far) < _plane_distance(start, near)
                    if conflict != want_conflict:
                        continue
                    far_color = "blue" if want_blue else "red"
                    near_color = "red" if want_blue else "blue"
                    dots = [
                        {"color": "green", "cell": [start[0], start[1]]},
                        {"color": far_color, "cell": [far[0], far[1]]},
                        {"color": near_color, "cell": [near[0], near[1]]},
                    ]
                    return {
                        "canvas": [CANVAS, CANVAS],
                        "grid": GRID,
                        "cell_px": CELL_PX,
                        "wall_px": WALL_PX,
                        "margin_px": MARGIN_PX,
                        "dot_radius": DOT_RADIUS,
                        "passages": passages,
                        "dots": dots,
                        "palette": {
                            "wall": list(WALL_RGB),
                            "corridor": list(CORRIDOR_RGB),
                            "green": list(GREEN_RGB),
                            "blue": list(BLUE_RGB),
                            "red": list(RED_RGB),
                        },
                    }
    raise RuntimeError(f"sampling failed for seed {seed}")


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def _dot_cell(scene: dict, color: str):
    for dot in scene["dots"]:
        if dot["color"] == color:
            return tuple(int(v) for v in dot["cell"])
    raise KeyError(color)


def render(scene: dict) -> Image.Image:
    w, h = int(scene["canvas"][0]), int(scene["canvas"][1])
    grid = int(scene["grid"])
    cell = int(scene["cell_px"])
    wall = int(scene["wall_px"])
    off = int(scene["margin_px"])
    step = cell + wall
    pal = scene["palette"]

    img = Image.new("RGB", (w, h), tuple(pal["wall"]))
    draw = ImageDraw.Draw(img)
    corridor = tuple(pal["corridor"])
    half = cell // 2

    # every carved passage is a straight corridor stroke joining two centres;
    # the stub squares keep corners and dead ends square-capped.
    for r in range(grid):
        for c in range(grid):
            x0 = off + c * step
            y0 = off + r * step
            draw.rectangle([x0 - half, y0 - half, x0 + half, y0 + half], fill=corridor)

    for pair in scene["passages"]:
        (ar, ac), (br, bc) = (int(pair[0][0]), int(pair[0][1])), (int(pair[1][0]), int(pair[1][1]))
        ax, ay = off + ac * step, off + ar * step
        bx, by = off + bc * step, off + br * step
        x0, x1 = min(ax, bx), max(ax, bx)
        y0, y1 = min(ay, by), max(ay, by)
        draw.rectangle([x0 - half, y0 - half, x1 + half, y1 + half], fill=corridor)

    radius = int(scene["dot_radius"])
    for dot in scene["dots"]:
        cx, cy = _cell_center(dot["cell"])
        draw.ellipse(
            [cx - radius, cy - radius, cx + radius, cy + radius],
            fill=tuple(pal[dot["color"]]),
        )
    return img


# --------------------------------------------------------------------------
# gold, margin, quarantine, symmetries
# --------------------------------------------------------------------------
def _target_steps(scene: dict) -> tuple[int, int]:
    grid = int(scene["grid"])
    start = _dot_cell(scene, "green")
    dist = _cell_distances(grid, scene["passages"], start)
    return dist[_dot_cell(scene, "blue")], dist[_dot_cell(scene, "red")]


def analytic_gold(scene: dict) -> str:
    d_blue, d_red = _target_steps(scene)
    if d_blue == d_red:
        return "tie"
    return "blue" if d_blue > d_red else "red"


def margin(scene: dict) -> float:
    d_blue, d_red = _target_steps(scene)
    return float(abs(d_blue - d_red) * (int(scene["cell_px"]) + int(scene["wall_px"])))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < MIN_STEP_MARGIN * (int(scene["cell_px"]) + int(scene["wall_px"]))


def latent_symmetries(scene: dict) -> list:
    """Label-preserving, pixel-identical relabellings of the latent record."""
    flipped = copy.deepcopy(scene)
    flipped["passages"] = [[list(p[1]), list(p[0])] for p in reversed(scene["passages"])]

    reordered = copy.deepcopy(scene)
    reordered["dots"] = list(reversed(copy.deepcopy(scene["dots"])))

    return [
        ("passage_list_reversed_and_endpoints_swapped", flipped),
        ("dot_record_order_reversed", reordered),
    ]
