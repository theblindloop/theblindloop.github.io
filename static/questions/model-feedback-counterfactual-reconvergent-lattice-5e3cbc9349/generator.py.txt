"""Deterministic renderer for a counterfactual reconvergent Boolean lattice."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 760
SCALE = 2
BACKGROUND = (246, 245, 240)
PANEL_FILL = (255, 255, 255)
INK = (35, 40, 47)
OFF = (255, 255, 255)
RED = (205, 54, 54)
WIRE = (158, 164, 172)
GATE_COLORS = {
    "and": (48, 112, 190),
    "or": (224, 148, 39),
    "xor": (132, 76, 183),
}
GATE_LABELS = {"and": "ALL", "or": "ANY", "xor": "ONE"}
ROW_Y = (142, 222, 302, 382, 462)
TOP_X0 = 84
CELL_STEP = 60
CELL_RADIUS = 16
GATE_X0, GATE_X1 = 650, 758
OPTION_BOXES = (
    (22, 574, 238, 728),
    (256, 574, 472, 728),
    (490, 574, 706, 728),
    (724, 574, 940, 728),
)


def _apply(kind: str, left: int, right: int) -> int:
    if kind == "and":
        return int(bool(left) and bool(right))
    if kind == "or":
        return int(bool(left) or bool(right))
    if kind == "xor":
        return int(bool(left) != bool(right))
    raise ValueError(f"unknown gate {kind!r}")


def _next_row(row: list[int], kind: str) -> list[int]:
    return [_apply(kind, row[index], row[index + 1]) for index in range(len(row) - 1)]


def _run(top: list[int], gates: list[str]) -> list[list[int]]:
    rows = [list(top)]
    for kind in gates:
        rows.append(_next_row(rows[-1], kind))
    return rows


def _counterfactual_bottom(top: list[int], gates: list[str], marked: int) -> list[int]:
    changed = list(top)
    changed[marked] = 1 - changed[marked]
    return _run(changed, gates)[-1]


def _plausible_errors(top: list[int], gates: list[str], marked: int) -> list[list[int]]:
    variants: list[list[int]] = []
    variants.append(_run(top, gates)[-1])
    variants.append(_counterfactual_bottom(top, list(reversed(gates)), marked))
    variants.append(_counterfactual_bottom(top, ["or" if gate == "xor" else gate for gate in gates], marked))
    variants.append(_counterfactual_bottom(top, ["or" if gate == "and" else "and" if gate == "or" else gate for gate in gates], marked))
    variants.append(_counterfactual_bottom(top, gates, marked - 1))
    variants.append(_counterfactual_bottom(top, gates, marked + 1))
    return variants


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target_slot = int(seed) % 4
    for _ in range(20000):
        top = [rng.randrange(2) for _ in range(9)]
        gates = [rng.choice(("and", "or", "xor")) for _ in range(4)]
        if len(set(gates)) < 3:
            continue
        marked = rng.randrange(1, 8)
        factual = _run(top, gates)[-1]
        target = _counterfactual_bottom(top, gates, marked)
        distance = sum(a != b for a, b in zip(factual, target, strict=True))
        if not 2 <= distance <= 4:
            continue

        # One near-boundary option models a single-cell bookkeeping slip.
        one_bit = list(target)
        one_bit[(int(seed) // 4) % len(target)] ^= 1
        distractors: list[list[int]] = []
        for pattern in [one_bit, *_plausible_errors(top, gates, marked)]:
            candidate = list(pattern)
            if candidate != target and candidate not in distractors:
                distractors.append(candidate)
            if len(distractors) == 3:
                break
        if len(distractors) < 3:
            continue
        rng.shuffle(distractors)
        options: list[list[int]] = []
        cursor = 0
        for slot in range(4):
            if slot == target_slot:
                options.append(target)
            else:
                options.append(distractors[cursor])
                cursor += 1
        return {
            "top": top,
            "gates": gates,
            "marked": marked,
            "options": options,
            "option_draw_order": [0, 1, 2, 3],
        }
    raise RuntimeError("could not sample a separated counterfactual lattice")


def analytic_gold(scene: dict) -> str:
    target = _counterfactual_bottom(
        [int(value) for value in scene["top"]],
        [str(value) for value in scene["gates"]],
        int(scene["marked"]),
    )
    matches = [index for index, option in enumerate(scene["options"]) if list(option) == target]
    if len(matches) != 1:
        raise ValueError(f"expected one matching option, got {matches}")
    return f"option-{matches[0] + 1}"


def margin(scene: dict) -> float:
    correct = int(analytic_gold(scene).split("-")[1]) - 1
    target = list(scene["options"][correct])
    distances = [
        sum(a != b for a, b in zip(target, option, strict=True))
        for index, option in enumerate(scene["options"])
        if index != correct
    ]
    return float(min(distances))


def is_quarantined(scene: dict) -> bool:
    try:
        top = scene["top"]
        gates = scene["gates"]
        options = scene["options"]
        if len(top) != 9 or any(value not in (0, 1) for value in top):
            return True
        if len(gates) != 4 or any(value not in GATE_COLORS for value in gates):
            return True
        if not 1 <= int(scene["marked"]) <= 7:
            return True
        if len(options) != 4 or any(len(option) != 5 for option in options):
            return True
        if any(value not in (0, 1) for option in options for value in option):
            return True
        if len({tuple(option) for option in options}) != 4:
            return True
        if sorted(scene["option_draw_order"]) != [0, 1, 2, 3]:
            return True
        analytic_gold(scene)
        return False
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["option_draw_order"] = list(reversed(twin["option_draw_order"]))
    return [("reverse_disjoint_option_draw_order", twin)]


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(name, size * SCALE)


def _pt(point: tuple[int, int]) -> tuple[int, int]:
    return point[0] * SCALE, point[1] * SCALE


def _box(box: tuple[int, int, int, int]) -> tuple[int, int, int, int]:
    return tuple(value * SCALE for value in box)  # type: ignore[return-value]


def _row_centers(row_index: int) -> list[tuple[int, int]]:
    length = 9 - row_index
    x0 = TOP_X0 + row_index * (CELL_STEP // 2)
    return [(x0 + index * CELL_STEP, ROW_Y[row_index]) for index in range(length)]


def _draw_cell(draw: ImageDraw.ImageDraw, center: tuple[int, int], value: int, radius: int) -> None:
    x, y = center
    draw.ellipse(
        _box((x - radius, y - radius, x + radius, y + radius)),
        fill=INK if value else OFF,
        outline=INK,
        width=3 * SCALE,
    )


def _draw_option(draw: ImageDraw.ImageDraw, slot: int, pattern: list[int]) -> None:
    x0, y0, x1, y1 = OPTION_BOXES[slot]
    draw.rounded_rectangle(_box((x0, y0, x1, y1)), radius=13 * SCALE,
                           fill=PANEL_FILL, outline=INK, width=3 * SCALE)
    draw.text(_pt((x0 + 16, y0 + 14)), f"OPTION {slot + 1}", fill=INK, font=_font(17, True))
    first = x0 + 38
    for index, value in enumerate(pattern):
        _draw_cell(draw, (first + index * 35, y0 + 101), int(value), 11)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text(_pt((28, 18)), "FLIP THE RED-RINGED TOP CELL", fill=INK, font=_font(28, True))
    draw.text(_pt((30, 58)), "then rebuild each whole row downward", fill=INK, font=_font(19))
    draw.text(_pt((660, 25)), "dark = ON", fill=INK, font=_font(18, True))

    rows = _run([int(value) for value in scene["top"]], [str(value) for value in scene["gates"]])
    centers = [_row_centers(index) for index in range(5)]
    # Draw the causal lattice first, keeping cells crisp above the wires.
    for row_index in range(1, 5):
        for index, child in enumerate(centers[row_index]):
            for parent in (centers[row_index - 1][index], centers[row_index - 1][index + 1]):
                draw.line([_pt(parent), _pt(child)], fill=WIRE, width=3 * SCALE)

    draw.text(_pt((18, ROW_Y[0] - 11)), "TOP", fill=INK, font=_font(15, True))
    for row_index, row in enumerate(rows):
        for index, value in enumerate(row):
            center = centers[row_index][index]
            if row_index == 0 and index == int(scene["marked"]):
                x, y = center
                draw.ellipse(_box((x - 26, y - 26, x + 26, y + 26)), outline=RED, width=6 * SCALE)
            _draw_cell(draw, center, int(value), CELL_RADIUS)
        if row_index:
            kind = str(scene["gates"][row_index - 1])
            y = ROW_Y[row_index]
            draw.rounded_rectangle(_box((GATE_X0, y - 23, GATE_X1, y + 23)), radius=9 * SCALE,
                                   fill=GATE_COLORS[kind], outline=INK, width=2 * SCALE)
            label = GATE_LABELS[kind]
            bbox = draw.textbbox((0, 0), label, font=_font(17, True))
            tw, th = bbox[2] - bbox[0], bbox[3] - bbox[1]
            draw.text((704 * SCALE - tw // 2, y * SCALE - th // 2 - SCALE), label,
                      fill=(255, 255, 255), font=_font(17, True))
            draw.text(_pt((776, y - 10)), f"ROW {row_index}", fill=INK, font=_font(15, True))

    draw.text(_pt((28, 528)), "COUNTERFACTUAL BOTTOM:", fill=INK, font=_font(19, True))
    for slot in scene["option_draw_order"]:
        _draw_option(draw, int(slot), [int(value) for value in scene["options"][int(slot)]])
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
