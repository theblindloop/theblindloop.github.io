"""Renderer and analytic gold for the ternary smear deconvolution world.

Latent scene: a hidden 7x7 binary glyph, printed only through the declared
smear  P[i][j] = (S[i][j] + S[i][j-1] + S[i][j-2]) mod 3, plus palette, tone
separation, cell pitch, placement jitter and decorative distractors.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "ternary_smear_deconvolve-0.1.0"

ROWS = 7
COLS = 7
KERNEL = 3
FRAME = 4
CANVAS = (560, 700)

GLYPHS: dict[str, tuple[str, ...]] = {
    "ring": (
        ".......",
        "..###..",
        ".#...#.",
        ".#...#.",
        ".#...#.",
        "..###..",
        ".......",
    ),
    "plus": (
        "...#...",
        "...#...",
        "...#...",
        "#######",
        "...#...",
        "...#...",
        ".......",
    ),
    "chevron": (
        "##...##",
        ".##.##.",
        "..###..",
        "...#...",
        ".......",
        ".......",
        ".......",
    ),
    "steps": (
        ".......",
        "##.....",
        ".##....",
        "..##...",
        "...##..",
        "....##.",
        ".....##",
    ),
    "bars": (
        ".####..",
        ".......",
        ".####..",
        ".......",
        ".####..",
        ".......",
        ".......",
    ),
}

SYMBOLS = tuple(sorted(GLYPHS))

TEMPLATES: dict[str, tuple[int, ...]] = {
    name: tuple(1 if ch == "#" else 0 for row in rows for ch in row)
    for name, rows in GLYPHS.items()
}

# Base ink colours; every base stays far from the black fiducial frame.
PALETTES = (
    {"bg": (246, 244, 238), "base": (32, 74, 168), "deco": (196, 116, 40)},
    {"bg": (240, 244, 240), "base": (26, 122, 92), "deco": (150, 74, 160)},
    {"bg": (247, 240, 240), "base": (158, 46, 52), "deco": (58, 112, 176)},
)
# Tone-separation steps: the three key colours are white blended with the base.
TINTS = (0.30, 0.36, 0.42, 0.20)
CELLS = (40, 44, 48)
GRID_GRAY = (150, 150, 150)


def _norm(scene: dict) -> dict:
    out = dict(scene)
    out["symbol"] = str(scene["symbol"]).strip().lower()
    return out


def bits(scene: dict) -> list[list[int]]:
    flat = TEMPLATES[_norm(scene)["symbol"]]
    return [list(flat[r * COLS : (r + 1) * COLS]) for r in range(ROWS)]


def smeared(scene: dict) -> list[list[int]]:
    """The printed tone levels: a modulo-three box smear of width KERNEL."""
    src = bits(scene)
    out = []
    for row in src:
        line = []
        for j in range(COLS):
            total = sum(row[max(0, j - k)] if j - k >= 0 else 0 for k in range(KERNEL))
            line.append(total % 3)
        out.append(line)
    return out


def tone_colors(scene: dict) -> list[tuple[int, int, int]]:
    pal = PALETTES[int(scene["palette"]) % len(PALETTES)]
    step = TINTS[int(scene["tint"]) % len(TINTS)]
    base = pal["base"]
    colors = []
    for k in range(3):
        a = 0.06 + step * k
        colors.append(tuple(int(round(255 + (c - 255) * a)) for c in base))
    return colors


def _layout(scene: dict) -> dict:
    cell = int(scene["cell"])
    ox = int(scene["jitter_x"])
    oy = int(scene["jitter_y"])
    panel_w = COLS * cell
    panel_h = ROWS * cell
    key_y = oy + panel_h + FRAME + cell
    sw = cell + 10
    sh = int(0.7 * cell)
    return {
        "cell": cell,
        "px": ox,
        "py": oy,
        "pw": panel_w,
        "ph": panel_h,
        "kx": ox,
        "ky": key_y,
        "sw": sw,
        "sh": sh,
    }


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    return {
        "symbol": rng.choice(SYMBOLS),
        "palette": rng.randrange(len(PALETTES)),
        "tint": rng.randrange(len(TINTS)),
        "cell": rng.choice(CELLS),
        "jitter_x": rng.randrange(24, 70),
        "jitter_y": rng.randrange(24, 64),
        "deco": [
            [rng.randrange(0, 4), rng.randrange(430, 520), rng.randrange(40, 620)]
            for _ in range(rng.randrange(2, 5))
        ],
    }


def render(scene: dict) -> Image.Image:
    scene = _norm(scene)
    pal = PALETTES[int(scene["palette"]) % len(PALETTES)]
    lay = _layout(scene)
    img = Image.new("RGB", CANVAS, pal["bg"])
    draw = ImageDraw.Draw(img)
    colors = tone_colors(scene)
    cell = lay["cell"]

    # decorative distractors: saturated, never black, never on the fiducials
    for kind, dx, dy in scene["deco"]:
        c = pal["deco"]
        if kind % 2:
            draw.ellipse([dx, dy, dx + 18, dy + 18], fill=c)
        else:
            draw.rectangle([dx, dy, dx + 20, dy + 12], fill=c)

    levels = smeared(scene)
    for i in range(ROWS):
        for j in range(COLS):
            x0 = lay["px"] + j * cell
            y0 = lay["py"] + i * cell
            draw.rectangle([x0, y0, x0 + cell, y0 + cell], fill=colors[levels[i][j]])
    for i in range(ROWS + 1):
        y = lay["py"] + i * cell
        draw.line([lay["px"], y, lay["px"] + lay["pw"], y], fill=GRID_GRAY, width=1)
    for j in range(COLS + 1):
        x = lay["px"] + j * cell
        draw.line([x, lay["py"], x, lay["py"] + lay["ph"]], fill=GRID_GRAY, width=1)
    draw.rectangle(
        [lay["px"], lay["py"], lay["px"] + lay["pw"], lay["py"] + lay["ph"]],
        outline=(0, 0, 0),
        width=FRAME,
    )

    # key strip: swatches for 0, 1, 2 from left to right
    for k in range(3):
        x0 = lay["kx"] + k * lay["sw"]
        draw.rectangle([x0, lay["ky"], x0 + lay["sw"], lay["ky"] + lay["sh"]], fill=colors[k])
        draw.rectangle(
            [x0, lay["ky"], x0 + lay["sw"], lay["ky"] + lay["sh"]],
            outline=(0, 0, 0),
            width=FRAME,
        )
    return img


def analytic_gold(scene: dict) -> str:
    return _norm(scene)["symbol"]


def _hamming(a: tuple[int, ...], b: tuple[int, ...]) -> int:
    return sum(1 for x, y in zip(a, b) if x != y)


def margin(scene: dict) -> float:
    scene = _norm(scene)
    truth = TEMPLATES[scene["symbol"]]
    hamming_gap = min(
        _hamming(truth, other) for name, other in TEMPLATES.items() if name != scene["symbol"]
    )
    colors = tone_colors(scene)
    color_gap = min(
        sum((p - q) ** 2 for p, q in zip(colors[a], colors[b])) ** 0.5
        for a in range(3)
        for b in range(a + 1, 3)
    )
    return float(round(min(float(hamming_gap), color_gap / 10.0), 6))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.5


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical latent aliases: both index fields are read modulo their
    table length and the symbol name is normalised, so none of these can move
    either the raster or the analytic gold."""
    out = []
    twin = dict(scene)
    twin["palette"] = int(scene["palette"]) + len(PALETTES)
    out.append(("palette_period", twin))
    twin = dict(scene)
    twin["tint"] = int(scene["tint"]) + len(TINTS)
    out.append(("tint_period", twin))
    twin = dict(scene)
    twin["symbol"] = str(scene["symbol"]).upper()
    out.append(("symbol_case", twin))
    return out
