"""Deterministic renderer and analytic answer for the side-tip graph world."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
ROUTE_RGB = (38, 54, 71)
GREEN_RGB = (38, 170, 104)
AMBER_RGB = (226, 151, 34)
VIOLET_RGB = (132, 86, 190)
TERMINAL_RGB = {"amber": AMBER_RGB, "violet": VIOLET_RGB}
ROUTE_WIDTH = 11
MARKER_RADIUS = 14.0
MIN_COUNT_GAP = 1.0


def _point(value: list[float] | tuple[float, float]) -> tuple[float, float]:
    return float(value[0]), float(value[1])


def _transformed(point: list[float] | tuple[float, float], layout: dict[str, Any]) -> tuple[float, float]:
    x, y = _point(point)
    if bool(layout["mirror_x"]):
        x = 438.0 - x
    return x + float(layout["dx"]), y + float(layout["dy"])


def _geometry(y: float, side_count: int, mirror_x: bool, branch_flip: bool) -> dict[str, Any]:
    """Make a clean tree whose only unmarked leaves are the requested side tips."""

    vertices: dict[str, list[float]] = {
        "start": [78.0, y],
        "j1": [176.0, y],
        "j2": [282.0, y],
        "terminal": [416.0, y],
    }
    edges: list[list[str]] = [["start", "j1"], ["j1", "j2"], ["j2", "terminal"]]
    branch_options = [
        ("j1", "tip_1", [176.0, y - 58.0 if not branch_flip else y + 58.0]),
        ("j2", "tip_2", [282.0, y + 58.0 if not branch_flip else y - 58.0]),
        ("j1", "tip_3", [176.0, y + 58.0 if not branch_flip else y - 58.0]),
    ]
    for parent, child, location in branch_options[:side_count]:
        vertices[child] = location
        edges.append([parent, child])
    # `mirror_x` is applied by the visible layout transform at render time.
    # Keeping this topology canonical also makes the analytic geometry easy to audit.
    _ = mirror_x
    return {"vertices": vertices, "edges": edges, "side_tip_count": side_count}


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    low = 1 + rng.randrange(2)
    high = low + 1 + rng.randrange(2)
    if high > 3:
        high = 3
    winner_index = rng.randrange(2)
    counts = [low, high] if winner_index == 1 else [high, low]
    y_slots = [142.0, 365.0]
    rng.shuffle(y_slots)
    mirror_x = bool(rng.randrange(2))
    branch_flip = bool(rng.randrange(2))
    layout = {
        "dx": float(rng.randrange(8, 25)),
        "dy": float(rng.randrange(-5, 6)),
        "mirror_x": mirror_x,
    }
    networks = []
    for index, (y, count) in enumerate(zip(y_slots, counts, strict=True)):
        topology = _geometry(y, count, mirror_x, branch_flip if index == 0 else not branch_flip)
        networks.append(
            {
                "network_id": f"network_{index}",
                "vertices": topology["vertices"],
                "edges": topology["edges"],
                "side_tip_count": count,
            }
        )
    terminal_assignment = {
        "network_0": "amber" if rng.randrange(2) == 0 else "violet",
        "network_1": "violet",
    }
    if terminal_assignment["network_0"] == "violet":
        terminal_assignment["network_1"] = "amber"
    return {
        "networks": networks,
        "terminal_assignment": terminal_assignment,
        "layout": layout,
    }


def _visible_vertices(network: dict[str, Any], layout: dict[str, Any]) -> dict[str, tuple[float, float]]:
    return {name: _transformed(value, layout) for name, value in network["vertices"].items()}


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only the visible graph and endpoint labels with fixed supersampling."""

    supersample = 4
    image = Image.new("RGB", (WIDTH * supersample, HEIGHT * supersample), (255, 255, 255))
    draw = ImageDraw.Draw(image)

    def scaled(value: tuple[float, float]) -> tuple[float, float]:
        return value[0] * supersample, value[1] * supersample

    def circle(center: tuple[float, float], radius: float, color: tuple[int, int, int]) -> None:
        cx, cy = scaled(center)
        rr = radius * supersample
        draw.ellipse((cx - rr, cy - rr, cx + rr, cy + rr), fill=color)

    for network in scene["networks"]:
        vertices = _visible_vertices(network, scene["layout"])
        for a, b in network["edges"]:
            draw.line(
                [scaled(vertices[a]), scaled(vertices[b])],
                fill=ROUTE_RGB,
                width=ROUTE_WIDTH * supersample,
                joint="curve",
            )
        # Small dark joint discs stabilize connectivity at orthogonal junctions.
        for name in vertices:
            if name.startswith("j"):
                circle(vertices[name], ROUTE_WIDTH / 2.0, ROUTE_RGB)

    for network in scene["networks"]:
        vertices = _visible_vertices(network, scene["layout"])
        circle(vertices["start"], MARKER_RADIUS, GREEN_RGB)
        terminal = scene["terminal_assignment"][network["network_id"]]
        circle(vertices["terminal"], MARKER_RADIUS, TERMINAL_RGB[terminal])

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def _counts(scene: dict[str, Any]) -> dict[str, int]:
    return {network["network_id"]: int(network["side_tip_count"]) for network in scene["networks"]}


def analytic_gold(scene: dict[str, Any]) -> str:
    counts = _counts(scene)
    winner = max(sorted(counts), key=lambda key: (counts[key], key))
    return scene["terminal_assignment"][winner]


def margin(scene: dict[str, Any]) -> float:
    values = sorted(_counts(scene).values())
    return float(values[-1] - values[0])


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < MIN_COUNT_GAP


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # No latent identity is omitted from the pixels: each topology and marker
    # assignment is explicitly rendered, so the v0.4 tri-state alias gate is N/A.
    return []
