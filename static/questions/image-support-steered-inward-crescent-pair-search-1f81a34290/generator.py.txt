"""Renderer and analytic world for the detached inward-crescent search."""

from __future__ import annotations

import copy
import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "inward-crescent-pair-search-0.1.0"
QUADRANTS = ("NW", "NE", "SW", "SE")
INK = "#1C4E94"
BACKGROUND = "white"
GUIDE = "#D9E0E8"
PAIR_SEPARATION = 64.0
CANVAS = (512, 512)


def _quadrant_center(quadrant: str) -> tuple[float, float]:
    return {
        "NW": (160.0, 160.0),
        "NE": (352.0, 160.0),
        "SW": (160.0, 352.0),
        "SE": (352.0, 352.0),
    }[quadrant]


def _pair_centers(
    quadrant: str, axis: str, offset: tuple[float, float]
) -> list[tuple[float, float]]:
    cx, cy = _quadrant_center(quadrant)
    ox, oy = offset
    cx += ox
    cy += oy
    if axis == "horizontal":
        return [(cx - PAIR_SEPARATION, cy), (cx + PAIR_SEPARATION, cy)]
    if axis == "vertical":
        return [(cx, cy - PAIR_SEPARATION), (cx, cy + PAIR_SEPARATION)]
    raise ValueError(f"unknown pair axis {axis!r}")


def _opening_signs(pattern: str) -> tuple[int, int]:
    """Signs along the pair axis; + means toward the second mark."""

    if pattern == "inward":
        return (1, -1)
    if pattern == "outward":
        return (-1, 1)
    if pattern == "mixed_first":
        return (1, 1)
    if pattern == "mixed_second":
        return (-1, -1)
    raise ValueError(f"unknown badge pattern {pattern!r}")


def _axis_vector(axis: str) -> tuple[float, float]:
    if axis == "horizontal":
        return (1.0, 0.0)
    if axis == "vertical":
        return (0.0, 1.0)
    raise ValueError(f"unknown pair axis {axis!r}")


def _rotate(point: tuple[float, float], angle: float) -> tuple[float, float]:
    x, y = point
    c, s = math.cos(angle), math.sin(angle)
    return (x * c - y * s, x * s + y * c)


def _draw_crescent(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    direction: tuple[float, float],
    half_size: float,
    stroke: float,
    color: str,
    supersample: int,
) -> None:
    """Draw a connected, squared C whose missing side faces ``direction``."""

    angle = math.atan2(direction[1], direction[0])
    # Canonical C: a closed left side and two arms extending to the right.
    segments = (
        ((-half_size, -half_size), (-half_size, half_size)),
        ((-half_size, -half_size), (half_size, -half_size)),
        ((-half_size, half_size), (half_size, half_size)),
    )
    width = max(1, round(stroke * supersample))
    for first, second in segments:
        p1 = _rotate(first, angle)
        p2 = _rotate(second, angle)
        draw.line(
            [
                ((center[0] + p1[0]) * supersample, (center[1] + p1[1]) * supersample),
                ((center[0] + p2[0]) * supersample, (center[1] + p2[1]) * supersample),
            ],
            fill=color,
            width=width,
        )
    # Round the three joins so thresholding cannot split the glyph into pieces.
    radius = stroke * supersample / 2.0
    for point in ((-half_size, -half_size), (-half_size, half_size)):
        px, py = _rotate(point, angle)
        draw.ellipse(
            [
                (center[0] * supersample + px * supersample - radius),
                (center[1] * supersample + py * supersample - radius),
                (center[0] * supersample + px * supersample + radius),
                (center[1] * supersample + py * supersample + radius),
            ],
            fill=color,
        )


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one balanced four-quadrant scene from a public integer seed."""

    rng = random.Random(int(seed))
    target = rng.choice(QUADRANTS)
    patterns = ["outward", "mixed_first", "mixed_second"]
    rng.shuffle(patterns)
    pair_axes: dict[str, str] = {}
    pair_offsets: dict[str, list[float]] = {}
    specs: list[dict[str, str]] = []
    pattern_index = 0
    for quadrant in QUADRANTS:
        pair_axes[quadrant] = rng.choice(("horizontal", "vertical"))
        # A perpendicular translation changes absolute position without moving
        # either component across the safe ablation-cell margins.
        jitter = round(rng.uniform(-6.0, 6.0), 3)
        if pair_axes[quadrant] == "horizontal":
            offset = [0.0, jitter]
        else:
            offset = [jitter, 0.0]
        pair_offsets[quadrant] = offset
        if quadrant == target:
            pattern = "inward"  # render derives this from target_quadrant
        else:
            pattern = patterns[pattern_index]
            pattern_index += 1
        specs.append({"quadrant": quadrant, "pattern": pattern})
    rng.shuffle(specs)
    return {
        "target_quadrant": target,
        "badge_specs": specs,
        "pair_axes": pair_axes,
        "pair_offsets": pair_offsets,
        "glyph_size": round(rng.uniform(11.5, 12.5), 3),
        "glyph_stroke": round(rng.uniform(5.5, 6.5), 3),
        "ink_color": INK,
        "background": BACKGROUND,
        "guide_color": GUIDE,
        "width": CANVAS[0],
        "height": CANVAS[1],
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only the visible fields of a sampled scene."""

    supersample = 4
    width = int(scene["width"])
    height = int(scene["height"])
    image = Image.new(
        "RGB",
        (width * supersample, height * supersample),
        scene["background"],
    )
    draw = ImageDraw.Draw(image)
    guide_width = max(1, round(2.0 * supersample))
    draw.line(
        [(width * supersample / 2.0, 0), (width * supersample / 2.0, height * supersample)],
        fill=scene["guide_color"],
        width=guide_width,
    )
    draw.line(
        [(0, height * supersample / 2.0), (width * supersample, height * supersample / 2.0)],
        fill=scene["guide_color"],
        width=guide_width,
    )

    for badge in scene["badge_specs"]:
        quadrant = badge["quadrant"]
        axis = scene["pair_axes"][quadrant]
        offset = tuple(scene["pair_offsets"][quadrant])
        centers = _pair_centers(quadrant, axis, offset)
        pattern = "inward" if quadrant == scene["target_quadrant"] else badge["pattern"]
        signs = _opening_signs(pattern)
        ux, uy = _axis_vector(axis)
        for center, sign in zip(centers, signs, strict=True):
            _draw_crescent(
                draw,
                center,
                (sign * ux, sign * uy),
                float(scene["glyph_size"]),
                float(scene["glyph_stroke"]),
                scene["ink_color"],
                supersample,
            )

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    """The selected quadrant is a visible relation rendered into the target badge."""

    return str(scene["target_quadrant"])


def margin(scene: dict[str, Any]) -> float:
    """Runner-up gap in the discrete pair-relation score."""

    target = str(scene["target_quadrant"])
    scores = {quadrant: 0 for quadrant in QUADRANTS}
    scores[target] = 2
    for badge in scene["badge_specs"]:
        if badge["quadrant"] != target:
            scores[badge["quadrant"]] = {
                "outward": 0,
                "mixed_first": 1,
                "mixed_second": 1,
            }[badge["pattern"]]
    ranked = sorted(scores.values(), reverse=True)
    return float(ranked[0] - ranked[1])


def is_quarantined(scene: dict[str, Any]) -> bool:
    """All sampled glyphs are well above the discrete ambiguity boundary."""

    return margin(scene) < 1.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """Reordering independent badge records is render-invariant."""

    twin = copy.deepcopy(scene)
    twin["badge_specs"] = list(reversed(twin["badge_specs"]))
    return [("reverse_badge_record_order", twin)]
