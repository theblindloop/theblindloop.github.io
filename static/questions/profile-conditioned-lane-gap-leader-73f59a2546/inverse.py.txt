"""Scene-free pixel oracle.

Receives only the rendered image. It segments the three saturated runner discs
and the dark finish posts, binds each runner to the post sharing its lane row,
measures the horizontal gap from each runner to its own post, and reports the
smallest-gap colour together with whether that runner is also the rightmost one
on the page. It abstains whenever the marks are not legible or the decision is
too close to call at finite resolution.
"""

from collections import deque

ABSTAIN = "abstain"

_MIN_BLOB = 260
_MIN_POST = 300
_MIN_GAP_MARGIN = 24
_MIN_X_MARGIN = 24


def _classify(px):
    r, g, b = px[0], px[1], px[2]
    mx, mn = max(r, g, b), min(r, g, b)
    if mx < 110 and mx - mn < 45:
        return "post"
    if mx - mn < 60:
        return None
    if r == mx and r > 150 and g < 115 and b < 115:
        return "red"
    if g == mx and g > 110 and r < 120 and b < 130:
        return "green"
    if b == mx and b > 150 and r < 120 and g < 130:
        return "blue"
    return None


def _components(coords):
    pts = set(coords)
    out = []
    while pts:
        start = pts.pop()
        blob = [start]
        q = deque([start])
        while q:
            x, y = q.popleft()
            for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                p = (x + dx, y + dy)
                if p in pts:
                    pts.discard(p)
                    blob.append(p)
                    q.append(p)
        out.append(blob)
    return out


def _centroid(blob):
    n = len(blob)
    return sum(p[0] for p in blob) / n, sum(p[1] for p in blob) / n


def decision_from_image(image) -> str:
    img = image.convert("RGB")
    w, h = img.size
    px = img.load()

    buckets = {"post": [], "red": [], "green": [], "blue": []}
    for y in range(h):
        for x in range(w):
            k = _classify(px[x, y])
            if k is not None:
                buckets[k].append((x, y))

    runners = {}
    for name in ("red", "green", "blue"):
        blobs = [b for b in _components(buckets[name]) if len(b) >= _MIN_BLOB]
        if len(blobs) != 1:
            return ABSTAIN
        runners[name] = _centroid(blobs[0])

    posts = [b for b in _components(buckets["post"]) if len(b) >= _MIN_POST]
    if len(posts) != 3:
        return ABSTAIN
    post_pts = [_centroid(b) for b in posts]

    gaps = {}
    used = set()
    for name, (rx, ry) in runners.items():
        best, best_dy = None, None
        for j, (pxc, pyc) in enumerate(post_pts):
            dy = abs(pyc - ry)
            if best_dy is None or dy < best_dy:
                best, best_dy = j, dy
        if best is None or best_dy > 46 or best in used:
            return ABSTAIN
        used.add(best)
        gap = post_pts[best][0] - rx
        if gap <= 0:
            return ABSTAIN
        gaps[name] = gap

    ordered = sorted(gaps.items(), key=lambda kv: kv[1])
    if ordered[1][1] - ordered[0][1] < _MIN_GAP_MARGIN:
        return ABSTAIN
    leader = ordered[0][0]

    by_x = sorted(runners.items(), key=lambda kv: kv[1][0], reverse=True)
    if by_x[0][1][0] - by_x[1][1][0] < _MIN_X_MARGIN:
        return ABSTAIN
    rightmost = by_x[0][0]

    return "%s_%s" % (leader, "yes" if leader == rightmost else "no")
