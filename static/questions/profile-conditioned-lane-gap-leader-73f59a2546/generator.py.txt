"""Latent scene sampling, deterministic raster rendering and analytic gold.

Scene: three horizontal lanes on white paper. Each lane carries one coloured
runner disc and one dark finish post that belongs to that lane only. The
reported quantity is which runner has the smallest gap to its OWN post, plus
whether that runner is also the one farthest to the right on the page.
"""

import random

from PIL import Image, ImageDraw

WIDTH = 660
HEIGHT = 430

COLOUR_NAMES = ("red", "green", "blue")
COLOUR_INKS = {
    "red": (214, 38, 38),
    "green": (26, 148, 62),
    "blue": (38, 74, 206),
}
POST_INK = (48, 48, 48)
FURNITURE_INK = (196, 196, 196)

MIN_GAP_MARGIN = 26
MIN_X_MARGIN = 26
QUARANTINE_MARGIN = 24


def _lane_ys():
    return [96, 214, 332]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    want_aligned = bool(rng.getrandbits(1))

    for _ in range(4000):
        gaps = sorted(rng.sample(range(34, 268), 3))
        if gaps[1] - gaps[0] < MIN_GAP_MARGIN + 8:
            continue
        if gaps[2] - gaps[1] < MIN_GAP_MARGIN + 8:
            continue
        order = [0, 1, 2]
        rng.shuffle(order)
        lane_gap = [0, 0, 0]
        for rank, lane in enumerate(order):
            lane_gap[lane] = gaps[rank]

        post_x = [rng.randrange(392, 616, 2) for _ in range(3)]
        runner_x = [post_x[i] - lane_gap[i] for i in range(3)]
        if min(runner_x) < 54:
            continue

        leader = min(range(3), key=lambda i: lane_gap[i])
        sx = sorted(runner_x, reverse=True)
        if sx[0] - sx[1] < MIN_X_MARGIN + 8:
            continue
        rightmost = max(range(3), key=lambda i: runner_x[i])
        if (leader == rightmost) != want_aligned:
            continue

        colours = list(COLOUR_NAMES)
        rng.shuffle(colours)
        radius = [rng.randrange(16, 27) for _ in range(3)]
        return {
            "width": WIDTH,
            "height": HEIGHT,
            "lane_y": _lane_ys(),
            "post_x": post_x,
            "runner_x": runner_x,
            "radius": radius,
            "colour": colours,
            "post_w": rng.randrange(10, 15, 2),
            "post_h": rng.randrange(54, 71, 2),
            "frame_inset": rng.randrange(12, 21),
            "tick_len": rng.randrange(14, 25),
        }

    raise RuntimeError("scene rejection sampling failed")


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (scene["width"], scene["height"]), (255, 255, 255))
    d = ImageDraw.Draw(img)

    inset = scene["frame_inset"]
    d.rectangle(
        [inset, inset, scene["width"] - 1 - inset, scene["height"] - 1 - inset],
        outline=FURNITURE_INK,
        width=2,
    )
    tick = scene["tick_len"]
    d.line([inset, inset + tick, inset + tick, inset], fill=FURNITURE_INK, width=2)
    d.line(
        [
            scene["width"] - 1 - inset - tick,
            scene["height"] - 1 - inset,
            scene["width"] - 1 - inset,
            scene["height"] - 1 - inset - tick,
        ],
        fill=FURNITURE_INK,
        width=2,
    )

    for y in scene["lane_y"]:
        d.line([inset + 8, y + 52, scene["width"] - 9 - inset, y + 52],
               fill=FURNITURE_INK, width=2)
        d.line([inset + 8, y - 52, scene["width"] - 9 - inset, y - 52],
               fill=FURNITURE_INK, width=2)

    for i in range(3):
        y = scene["lane_y"][i]
        px = scene["post_x"][i]
        hw = scene["post_w"] // 2
        hh = scene["post_h"] // 2
        d.rectangle([px - hw, y - hh, px + hw, y + hh], fill=POST_INK)

    for i in range(3):
        y = scene["lane_y"][i]
        rx = scene["runner_x"][i]
        r = scene["radius"][i]
        ink = COLOUR_INKS[scene["colour"][i]]
        d.ellipse([rx - r, y - r, rx + r, y + r], fill=ink)

    return img


def _leader_and_alignment(scene: dict):
    gaps = [scene["post_x"][i] - scene["runner_x"][i] for i in range(3)]
    leader = min(range(3), key=lambda i: gaps[i])
    rightmost = max(range(3), key=lambda i: scene["runner_x"][i])
    return leader, (leader == rightmost)


def analytic_gold(scene: dict) -> str:
    leader, aligned = _leader_and_alignment(scene)
    return "%s_%s" % (scene["colour"][leader], "yes" if aligned else "no")


def margin(scene: dict) -> float:
    gaps = sorted(scene["post_x"][i] - scene["runner_x"][i] for i in range(3))
    xs = sorted(scene["runner_x"], reverse=True)
    return float(min(gaps[1] - gaps[0], xs[0] - xs[1]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list:
    return []
