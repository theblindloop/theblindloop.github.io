"""Deterministic renderer and analytic label for the integrated-trace world."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

RENDERER_VERSION = "integrated_curve_area-0.1.0"

WIDTH = 960
HEIGHT = 560
PANEL_RECTS = ((90, 450), (510, 870))
PLOT_TOP = 120
BASELINE_Y = 450
PLOT_HEIGHT = BASELINE_Y - PLOT_TOP
CHECKPOINTS = 12
QUARANTINE_MARGIN = 0.055
PALETTE = (
    (197, 52, 52),
    (32, 104, 184),
    (0, 135, 118),
    (141, 63, 168),
    (222, 130, 31),
    (32, 145, 170),
)
BACKGROUND = (255, 255, 255)
GRID = (218, 222, 228)
AXIS = (44, 48, 54)
TEXT = (30, 34, 40)


def _trapezoid_area(values: Iterable[float]) -> float:
    values = list(values)
    if len(values) != CHECKPOINTS:
        raise ValueError("each trace must have twelve checkpoint values")
    return sum((values[i] + values[i + 1]) * 0.5 for i in range(CHECKPOINTS - 1)) / (
        CHECKPOINTS - 1
    )


def _balanced_shape(rng: random.Random, phase: float) -> list[float]:
    """Produce a varied but comfortably rasterizable shape around height .5."""

    values = []
    for index in range(CHECKPOINTS):
        wave = 0.075 * math.sin(phase + index * 0.83)
        secondary = 0.035 * math.cos(phase * 0.7 + index * 1.61)
        jitter = rng.uniform(-0.045, 0.045)
        values.append(0.5 + wave + secondary + jitter)
    mean = _trapezoid_area(values)
    return [value + (0.5 - mean) for value in values]


def _shift_to_area(values: list[float], target: float) -> list[float]:
    shift = target - _trapezoid_area(values)
    shifted = [value + shift for value in values]
    if min(shifted) <= 0.12 or max(shifted) >= 0.88:
        raise ValueError("sampled curve leaves the visible plot safety band")
    return shifted


def sample_scene(seed: int) -> dict:
    """Sample a balanced pair; answer identity is independent of trace colors."""

    rng = random.Random(int(seed))
    shape_rng = random.Random((int(seed) * 1000003) ^ 0x5EED5EED)
    shape_a = _balanced_shape(shape_rng, rng.uniform(0.0, math.tau))
    shape_b = _balanced_shape(shape_rng, rng.uniform(0.0, math.tau))
    low = 0.35 + rng.uniform(0.0, 0.075)
    gap = 0.065 + rng.uniform(0.0, 0.14)
    high = low + gap
    if int(seed) % 2 == 0:
        a_values = _shift_to_area(shape_a, high)
        b_values = _shift_to_area(shape_b, low)
    else:
        a_values = _shift_to_area(shape_a, low)
        b_values = _shift_to_area(shape_b, high)

    color_rng = random.Random((int(seed) * 9176) ^ 0xA11CE)
    colors = [list(color) for color in color_rng.sample(PALETTE, 2)]
    return {
        "a_values": [round(value, 8) for value in a_values],
        "b_values": [round(value, 8) for value in b_values],
        "colors": colors,
        "style_token": int(seed) % 7,
    }


def _x_positions(x0: int, x1: int) -> list[int]:
    left = x0 + 18
    right = x1 - 18
    return [round(left + i * (right - left) / (CHECKPOINTS - 1)) for i in range(CHECKPOINTS)]


def _draw_panel(draw: ImageDraw.ImageDraw, label: str, x0: int, x1: int, values: list[float], color: list[int]) -> None:
    draw.rectangle((x0, PLOT_TOP, x1, BASELINE_Y), outline=AXIS, width=2)
    for fraction in (0.25, 0.5, 0.75):
        y = round(BASELINE_Y - fraction * PLOT_HEIGHT)
        draw.line((x0 + 1, y, x1 - 1, y), fill=GRID, width=1)
    draw.line((x0, BASELINE_Y, x1, BASELINE_Y), fill=AXIS, width=3)
    draw.text((x0 + 2, 78), label, fill=TEXT, font=ImageFont.load_default())
    points = [
        (x, round(BASELINE_Y - value * PLOT_HEIGHT))
        for x, value in zip(_x_positions(x0, x1), values, strict=True)
    ]
    draw.line(points, fill=tuple(color), width=6, joint="curve")
    for x, y in points:
        draw.ellipse((x - 5, y - 5, x + 5, y + 5), fill=tuple(color))


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((90, 30), "Baseline-relative trace area", fill=TEXT, font=ImageFont.load_default())
    draw.text(
        (90, 52),
        "Compare the area above each horizontal baseline",
        fill=(88, 94, 102),
        font=ImageFont.load_default(),
    )
    _draw_panel(draw, "A", PANEL_RECTS[0][0], PANEL_RECTS[0][1], scene["a_values"], scene["colors"][0])
    _draw_panel(draw, "B", PANEL_RECTS[1][0], PANEL_RECTS[1][1], scene["b_values"], scene["colors"][1])
    return image


def analytic_gold(scene: dict) -> str:
    area_a = _trapezoid_area(scene["a_values"])
    area_b = _trapezoid_area(scene["b_values"])
    return "A" if area_a > area_b else "B"


def margin(scene: dict) -> float:
    return abs(_trapezoid_area(scene["a_values"]) - _trapezoid_area(scene["b_values"]))


def is_quarantined(scene: dict) -> bool:
    values = list(scene["a_values"]) + list(scene["b_values"])
    return (
        margin(scene) < QUARANTINE_MARGIN
        or min(values) <= 0.10
        or max(values) >= 0.90
        or len(scene["a_values"]) != CHECKPOINTS
        or len(scene["b_values"]) != CHECKPOINTS
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["style_token"] = (int(scene.get("style_token", 0)) + 1) % 7
    return [("nuisance_style_token_toggle", twin)]
