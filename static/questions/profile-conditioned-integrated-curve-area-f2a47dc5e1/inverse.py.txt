"""Independent pixel-only inverse for the integrated-trace world."""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH = 960
HEIGHT = 560
PANEL_RECTS = ((90, 450), (510, 870))
TOP_LIMIT = 105
BASELINE_SEARCH = (100, 485)
PLOT_HEIGHT = 330.0


def _baseline_from_pixels(arr: np.ndarray, x0: int, x1: int) -> float:
    gray = arr[:, :, 0].astype(np.int16)
    dark = (gray < 90) & (np.max(arr, axis=2) - np.min(arr, axis=2) < 20)
    start, stop = BASELINE_SEARCH
    counts = dark[start:stop, x0 : x1 + 1].sum(axis=1)
    candidates = np.flatnonzero(counts > (x1 - x0) * 0.55)
    if len(candidates) == 0:
        raise ValueError("horizontal baseline is not recoverable")
    return float(start + candidates[-1])


def _trace_height(arr: np.ndarray, x0: int, x1: int, baseline: float) -> float:
    spread = arr.max(axis=2).astype(np.int16) - arr.min(axis=2).astype(np.int16)
    # All trace inks are saturated; axes, labels, and grid lines are grayscale.
    colored = spread >= 45
    left = x0 + 14
    right = x1 - 14
    ys = []
    xs = []
    top = max(TOP_LIMIT, int(baseline - 330))
    bottom = min(arr.shape[0], int(baseline + 10))
    for x in range(left, right + 1):
        column = np.flatnonzero(colored[top:bottom, x]) + top
        if len(column):
            ys.append(float(np.median(column)))
            xs.append(float(x))
    if len(xs) < 0.75 * (right - left + 1):
        raise ValueError("colored trace is incomplete or unreadable")
    all_x = np.arange(left, right + 1, dtype=np.float64)
    center_y = np.interp(all_x, np.asarray(xs), np.asarray(ys))
    heights = baseline - center_y
    if float(np.max(heights)) <= 8.0 or float(np.min(heights)) < -3.0:
        raise ValueError("trace is not above the recovered baseline")
    return float(np.mean(heights) / PLOT_HEIGHT)


def decision_from_image(image: Image.Image) -> str:
    """Recover both baseline-relative integrals from the RGB raster alone."""

    arr = np.asarray(image.convert("RGB"))
    if arr.shape[0] != HEIGHT or arr.shape[1] != WIDTH:
        raise ValueError("unexpected raster dimensions")
    areas = []
    for x0, x1 in PANEL_RECTS:
        baseline = _baseline_from_pixels(arr, x0, x1)
        areas.append(_trace_height(arr, x0, x1, baseline))
    difference = areas[0] - areas[1]
    if abs(difference) < 0.025:
        raise ValueError("measured areas are too close to decide")
    return "A" if difference > 0.0 else "B"
