"""Independent image-only inverse arm for guarded pair depletion."""

from __future__ import annotations

from dataclasses import dataclass

import numpy as np
from PIL import Image

COLOR_NAMES = ("cyan", "gold", "purple")
RGB = {
    "cyan": (22, 138, 173),
    "gold": (233, 154, 53),
    "purple": (123, 44, 191),
}
RING_RGB = (214, 40, 40)
COLOR_THRESHOLD = 52.0
MIN_COMPONENT_AREA = 280
MAX_COMPONENT_AREA = 1500


@dataclass(frozen=True)
class Component:
    color: str
    area: int
    cx: float
    cy: float


def _components(mask: np.ndarray, color: str) -> list[Component]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result: list[Component] = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]:
            continue
        stack = [(y0, x0)]
        seen[y0, x0] = True
        area = 0
        sx = 0
        sy = 0
        while stack:
            y, x = stack.pop()
            area += 1
            sx += x
            sy += y
            for yy, xx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= yy < height and 0 <= xx < width and mask[yy, xx] and not seen[yy, xx]:
                    seen[yy, xx] = True
                    stack.append((yy, xx))
        if MIN_COMPONENT_AREA <= area <= MAX_COMPONENT_AREA:
            result.append(Component(color, area, sx / area, sy / area))
    return result


def _decode(image: Image.Image) -> tuple[dict[str, int], str, list[list[str]]]:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    colored: list[Component] = []
    for name in COLOR_NAMES:
        target = np.asarray(RGB[name], dtype=np.float32)
        mask = np.linalg.norm(arr - target, axis=2) < COLOR_THRESHOLD
        colored.extend(_components(mask, name))
    if len(colored) != 27:
        raise ValueError(f"expected 27 separable colored disks, recovered {len(colored)}")

    # Six cards contain the 12 lowest colored disks. This derives regions from
    # component order, not from renderer coordinates.
    ordered_y = sorted(colored, key=lambda item: item.cy)
    top, card_pips = ordered_y[:-12], ordered_y[-12:]
    if min(item.cy for item in card_pips) - max(item.cy for item in top) < 70:
        raise ValueError("inventory/card separation margin collapsed")

    headers = sorted(top, key=lambda item: item.cy)[:3]
    inventory = sorted(top, key=lambda item: item.cy)[3:]
    if len(inventory) != 12:
        raise ValueError("starting inventories are not uniquely countable")
    if min(item.cy for item in inventory) - max(item.cy for item in headers) < 25:
        raise ValueError("header/inventory separation margin collapsed")
    if {item.color for item in headers} != set(COLOR_NAMES):
        raise ValueError("one color header per inventory was not recovered")

    counts = {name: sum(item.color == name for item in inventory) for name in COLOR_NAMES}

    ring_target = np.asarray(RING_RGB, dtype=np.float32)
    ring_mask = np.linalg.norm(arr - ring_target, axis=2) < COLOR_THRESHOLD
    ring_pixels = np.argwhere(ring_mask)
    if len(ring_pixels) < 180:
        raise ValueError("tracking ring is missing or materially erased")
    ring_cy, ring_cx = ring_pixels.mean(axis=0)
    ranked = sorted(
        ((header.cx - ring_cx) ** 2 + (header.cy - ring_cy) ** 2, header)
        for header in headers
    )
    if ranked[0][0] > 20**2 or ranked[1][0] - ranked[0][0] < 80**2:
        raise ValueError("tracking ring does not identify one header with margin")
    tracked = ranked[0][1].color

    x_sorted = sorted(card_pips, key=lambda item: item.cx)
    operations: list[list[str]] = []
    previous_center = None
    for index in range(0, 12, 2):
        pair = x_sorted[index : index + 2]
        if pair[0].color == pair[1].color:
            raise ValueError("card does not show two distinct colors")
        spread = pair[1].cx - pair[0].cx
        center = (pair[0].cx + pair[1].cx) / 2
        if not 30 <= spread <= 75:
            raise ValueError("card-pair grouping margin collapsed")
        if previous_center is not None and center - previous_center < 100:
            raise ValueError("card order is not uniquely separated")
        previous_center = center
        operations.append([pair[0].color, pair[1].color])
    return counts, tracked, operations


def decision_from_image(image: Image.Image) -> str:
    counts, tracked, operations = _decode(image)
    for left, right in operations:
        if counts[left] > 0 and counts[right] > 0:
            counts[left] -= 1
            counts[right] -= 1
    return str(counts[tracked])
