"""Deterministic scene sampler and raster renderer for guarded pair depletion."""

from __future__ import annotations

import itertools
import math
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

COLOR_NAMES = ("cyan", "gold", "purple")
PALETTE = {
    "cyan": "#168AAD",
    "gold": "#E99A35",
    "purple": "#7B2CBF",
}
TRACK_RED = "#D62828"
CANVAS = (960, 620)
CARD_COUNT = 6
PIXEL_MARGIN = 24.0


def _replay(initial_counts: dict[str, int], operations: list[list[str]]) -> dict[str, int]:
    counts = {name: int(initial_counts[name]) for name in COLOR_NAMES}
    for pair in operations:
        left, right = pair
        if left == right or left not in counts or right not in counts:
            raise ValueError("each card must show two distinct public colors")
        if counts[left] > 0 and counts[right] > 0:
            counts[left] -= 1
            counts[right] -= 1
    return counts


def _abstract_final(order: tuple[tuple[int, int], ...]) -> int:
    counts = [4, 2, 6]
    for left, right in order:
        if counts[left] > 0 and counts[right] > 0:
            counts[left] -= 1
            counts[right] -= 1
    return counts[0]


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; answer is 0 for even seeds and 2 for odd seeds.

    Every scene has the same total starting ink (12 beads), the same tracked
    starting count (4), and the same multiset of cards (two of every color pair).
    Only card order determines which of the two balanced answers is reached.
    """
    seed = int(seed)
    rng = random.Random(seed ^ 0x5A17C9D3)
    desired = 0 if seed % 2 == 0 else 2

    tracked = COLOR_NAMES[(seed // 2) % len(COLOR_NAMES)]
    others = [name for name in COLOR_NAMES if name != tracked]
    if (seed // 6) % 2:
        others.reverse()
    low, high = others

    abstract = [(0, 1)] * 2 + [(0, 2)] * 2 + [(1, 2)] * 2
    choices = [
        order
        for order in sorted(set(itertools.permutations(abstract)))
        if _abstract_final(order) == desired
        and _abstract_final(tuple(reversed(order))) != desired
    ]
    order = choices[rng.randrange(len(choices))]
    role_to_color = {0: tracked, 1: low, 2: high}
    operations: list[list[str]] = []
    for a, b in order:
        pair = [role_to_color[a], role_to_color[b]]
        if rng.randrange(2):
            pair.reverse()
        operations.append(pair)

    scene = {
        "initial_counts": {tracked: 4, low: 2, high: 6},
        "tracked_color": tracked,
        "operations": operations,
    }
    assert int(analytic_gold(scene)) == desired
    assert analytic_gold({**scene, "operations": list(reversed(operations))}) != str(desired)
    return scene


def analytic_gold(scene: dict) -> str:
    final_counts = _replay(scene["initial_counts"], scene["operations"])
    return str(final_counts[scene["tracked_color"]])


def margin(scene: dict) -> float:
    """Runner-up pixel grouping margin in pixels for the fixed card layout."""
    return PIXEL_MARGIN


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int, *, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    filename = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    path = Path("/usr/share/fonts/truetype/dejavu") / filename
    try:
        return ImageFont.truetype(str(path), size=size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    initial_counts = scene["initial_counts"]
    tracked_color = scene["tracked_color"]
    operations = scene["operations"]
    if set(initial_counts) != set(COLOR_NAMES) or tracked_color not in COLOR_NAMES:
        raise ValueError("invalid public color inventory")
    if len(operations) != CARD_COUNT:
        raise ValueError("exactly six operation cards are required")

    scale = 3
    width, height = CANVAS
    image = Image.new("RGB", (width * scale, height * scale), "#F8FAFC")
    draw = ImageDraw.Draw(image)

    def rect(box, *, fill, outline=None, radius=0, line=1):
        b = tuple(int(v * scale) for v in box)
        if radius:
            draw.rounded_rectangle(
                b, radius=int(radius * scale), fill=fill, outline=outline,
                width=max(1, int(line * scale)),
            )
        else:
            draw.rectangle(b, fill=fill, outline=outline, width=max(1, int(line * scale)))

    def disk(cx, cy, radius, fill, *, outline=None, line=1):
        box = tuple(int(v * scale) for v in (cx - radius, cy - radius, cx + radius, cy + radius))
        draw.ellipse(box, fill=fill, outline=outline, width=max(1, int(line * scale)))

    def text(xy, value, size, *, fill="#172033", bold=False, anchor="mm"):
        draw.text(
            (int(xy[0] * scale), int(xy[1] * scale)), value,
            font=_font(size * scale, bold=bold), fill=fill, anchor=anchor,
        )

    text((width / 2, 28), "STARTING INVENTORIES", 22, bold=True)
    column_centers = (220, 480, 740)
    for name, cx in zip(COLOR_NAMES, column_centers, strict=True):
        rect((cx - 105, 58, cx + 105, 333), fill="#FFFFFF", outline="#CBD5E1", radius=18, line=3)
        disk(cx, 111, 20, PALETTE[name])
        if name == tracked_color:
            disk(cx, 111, 29, None, outline=TRACK_RED, line=5)
        text((cx, 150), f"{int(initial_counts[name])} BEADS", 15, bold=True)
        count = int(initial_counts[name])
        cols = 2 if count <= 4 else 3
        rows = math.ceil(count / cols)
        x_step = 48
        y_step = 47
        start_x = cx - (cols - 1) * x_step / 2
        start_y = 210 + (2 - rows) * 18
        for index in range(count):
            row, col = divmod(index, cols)
            disk(start_x + col * x_step, start_y + row * y_step, 14, PALETTE[name])

    text((width / 2, 367), "CONSUME BOTH IF AVAILABLE  •  READ LEFT TO RIGHT", 19, bold=True)
    for index, pair in enumerate(operations):
        x0 = 35 + index * 150
        x1 = x0 + 130
        rect((x0, 401, x1, 565), fill="#FFFFFF", outline="#64748B", radius=15, line=3)
        text(((x0 + x1) / 2, 426), str(index + 1), 16, fill="#475569", bold=True)
        disk(x0 + 39, 493, 16, PALETTE[pair[0]])
        disk(x0 + 91, 493, 16, PALETTE[pair[1]])
        text(((x0 + x1) / 2, 493), "+", 21, bold=True)
        text(((x0 + x1) / 2, 540), "−1 EACH", 13, fill="#475569", bold=True)

    text((width / 2, 596), "The red ring marks the inventory to report.", 16, fill="#475569")
    return image.resize(CANVAS, Image.Resampling.LANCZOS)
