"""Deterministic scene sampler and renderer for closed braid components."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 560, 640
SCALE = 4
PALETTES = (
    {"background": "#fffdf7", "ink": "#17233b"},
    {"background": "#f9fcff", "ink": "#291b35"},
    {"background": "#fffaf6", "ink": "#17342d"},
)


def _cycle_count(word: list[int]) -> int:
    # lane -> strand, then invert to the top-lane -> bottom-lane permutation
    lanes = list(range(4))
    for adjacent in word:
        lanes[adjacent], lanes[adjacent + 1] = lanes[adjacent + 1], lanes[adjacent]
    bottom_of = [0] * 4
    for bottom_lane, strand in enumerate(lanes):
        bottom_of[strand] = bottom_lane
    seen: set[int] = set()
    cycles = 0
    for start in range(4):
        if start in seen:
            continue
        cycles += 1
        here = start
        while here not in seen:
            seen.add(here)
            here = bottom_of[here]
    return cycles


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = 1 if seed % 2 == 0 else 3
    for _ in range(10000):
        word = [rng.randrange(3) for _ in range(7)]
        counts = [word.count(i) for i in range(3)]
        if min(counts) == 0 or max(counts) > 4:
            continue
        if any(a == b for a, b in zip(word, word[1:])):
            continue
        if _cycle_count(word) == target:
            break
    else:  # pragma: no cover - deterministic search has ample support
        raise RuntimeError("could not sample a balanced braid word")
    return {
        "word": word,
        "mirror": bool(rng.randrange(2)),
        "palette": rng.randrange(len(PALETTES)),
        "band_jitter": [rng.randint(-5, 5) for _ in word],
    }


def analytic_gold(scene: dict) -> str:
    return "one" if _cycle_count([int(v) for v in scene["word"]]) == 1 else "three"


def _centers(scene: dict) -> list[float]:
    return [120.0 + 57.0 * i + float(j) for i, j in enumerate(scene["band_jitter"])]


def margin(scene: dict) -> float:
    ys = _centers(scene)
    return min(b - a for a, b in zip(ys, ys[1:])) - 44.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _cubic(p0, p1, p2, p3, n=50):
    result = []
    for k in range(n + 1):
        t = k / n
        u = 1.0 - t
        result.append(
            (
                u**3 * p0[0] + 3 * u * u * t * p1[0] + 3 * u * t * t * p2[0] + t**3 * p3[0],
                u**3 * p0[1] + 3 * u * u * t * p1[1] + 3 * u * t * t * p2[1] + t**3 * p3[1],
            )
        )
    return result


def _scaled(points):
    return [(round(x * SCALE), round(y * SCALE)) for x, y in points]


def render(scene: dict) -> Image.Image:
    palette = PALETTES[int(scene["palette"]) % len(PALETTES)]
    bg, ink = palette["background"], palette["ink"]
    canvas = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), bg)
    draw = ImageDraw.Draw(canvas)
    lanes = [190.0, 250.0, 310.0, 370.0]
    if bool(scene["mirror"]):
        lanes = [WIDTH - x for x in lanes[::-1]]
    top, bottom = 76.0, 520.0
    centers = _centers(scene)
    word = [int(v) for v in scene["word"]]
    if bool(scene["mirror"]):
        word = [2 - v for v in word]

    # Four visible same-column closures, nested outside the braid body.
    side_xs = [42.0, 104.0, 456.0, 518.0]
    for lane_index, x in enumerate(lanes):
        side = side_xs[lane_index]
        if lane_index < 2:
            top_ctrl = min(x - 30, side + 35)
        else:
            top_ctrl = max(x + 30, side - 35)
        pts = _cubic((x, top), (top_ctrl, 28), (side, 28), (side, 78), 24)
        pts += _cubic((side, 78), (side, 230), (side, 410), (side, 560), 42)[1:]
        pts += _cubic((side, 560), (side, 612), (top_ctrl, 612), (x, bottom), 28)[1:]
        draw.line(_scaled(pts), fill=ink, width=7 * SCALE, joint="curve")

    # Build one polyline for each strand through localized adjacent crossings.
    strand_lane = list(range(4))
    paths = [[(lanes[i], top)] for i in range(4)]
    for yc, adjacent in zip(centers, word):
        before = strand_lane[:]
        for strand in range(4):
            lane_index = before.index(strand)
            paths[strand].append((lanes[lane_index], yc - 16.0))
        strand_lane[adjacent], strand_lane[adjacent + 1] = (
            strand_lane[adjacent + 1],
            strand_lane[adjacent],
        )
        after = strand_lane[:]
        for strand in range(4):
            lane_index = after.index(strand)
            paths[strand].append((lanes[lane_index], yc + 16.0))
    for strand in range(4):
        lane_index = strand_lane.index(strand)
        paths[strand].append((lanes[lane_index], bottom))
        draw.line(_scaled(paths[strand]), fill=ink, width=7 * SCALE, joint="curve")

    # Bridge cue: one diagonal remains continuous while the other has a short gap.
    for event, (yc, adjacent) in enumerate(zip(centers, word)):
        x0, x1 = lanes[adjacent], lanes[adjacent + 1]
        if (event + int(scene["palette"])) % 2:
            under = ((x0 + 19, yc - 10), (x1 - 19, yc + 10))
            over = ((x1 - 23, yc - 12), (x0 + 23, yc + 12))
        else:
            under = ((x1 - 19, yc - 10), (x0 + 19, yc + 10))
            over = ((x0 + 23, yc - 12), (x1 - 23, yc + 12))
        draw.line(_scaled(under), fill=bg, width=14 * SCALE)
        draw.line(_scaled(over), fill=ink, width=7 * SCALE)

    return canvas.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
