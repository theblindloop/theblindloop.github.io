"""Independent image-only inverse arm based only on raster structure."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _runs(row: np.ndarray) -> list[tuple[int, int]]:
    padded = np.pad(row.astype(np.int8), (1, 1))
    changes = np.diff(padded)
    starts = np.flatnonzero(changes == 1)
    stops = np.flatnonzero(changes == -1)
    return [(int(a), int(b - 1)) for a, b in zip(starts, stops) if b - a >= 2]


def _clusters(values: list[int], gap: int = 2) -> list[list[int]]:
    if not values:
        return []
    groups = [[values[0]]]
    for value in values[1:]:
        if value - groups[-1][-1] <= gap:
            groups[-1].append(value)
        else:
            groups.append([value])
    return groups


def _cycle_count(word: list[int]) -> int:
    lanes = list(range(4))
    for adjacent in word:
        lanes[adjacent], lanes[adjacent + 1] = lanes[adjacent + 1], lanes[adjacent]
    bottom_of = [0] * 4
    for bottom_lane, strand in enumerate(lanes):
        bottom_of[strand] = bottom_lane
    seen: set[int] = set()
    cycles = 0
    for start in range(4):
        if start in seen:
            continue
        cycles += 1
        here = start
        while here not in seen:
            seen.add(here)
            here = bottom_of[here]
    return cycles


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    h, w = arr.shape[:2]
    # Corners are guaranteed paper; distance from their median segments all ink palettes.
    corner = max(3, min(h, w) // 40)
    samples = np.concatenate(
        [arr[:corner, :corner].reshape(-1, 3), arr[:corner, -corner:].reshape(-1, 3),
         arr[-corner:, :corner].reshape(-1, 3), arr[-corner:, -corner:].reshape(-1, 3)]
    )
    paper = np.median(samples, axis=0)
    mask = np.linalg.norm(arr - paper, axis=2) > 55.0
    if int(mask.sum()) < h * w * 0.005:
        raise ValueError("no braid ink found")

    # Discover the four central body lanes from persistent vertical-ink peaks.
    y0, y1 = int(0.14 * h), int(0.82 * h)
    projection = mask[y0:y1].sum(axis=0)
    peak_cut = max(10, int(0.32 * (y1 - y0)))
    peak_groups = _clusters(np.flatnonzero(projection >= peak_cut).tolist(), gap=3)
    peaks = [float(np.mean(g)) for g in peak_groups if len(g) >= 2]
    if len(peaks) < 4:
        raise ValueError("could not recover four braid lanes")
    # The braid body is the compact four-peak block nearest the image center.
    choices = []
    for i in range(len(peaks) - 3):
        block = peaks[i:i + 4]
        gaps = np.diff(block)
        regularity = float(np.std(gaps) / max(np.mean(gaps), 1.0))
        centrality = abs(float(np.mean(block)) - w / 2) / w
        choices.append((regularity + centrality, block))
    lanes = min(choices, key=lambda item: item[0])[1]
    spacing = float(np.median(np.diff(lanes)))
    if not (0.06 * w < spacing < 0.18 * w):
        raise ValueError("recovered lane spacing is implausible")

    left = int(max(0, lanes[0] - 0.42 * spacing))
    right = int(min(w, lanes[-1] + 0.42 * spacing))
    candidate_rows: list[int] = []
    pair_for_row: dict[int, int] = {}
    for y in range(y0, y1):
        runs = _runs(mask[y, left:right])
        centers = [left + (a + b) / 2 for a, b in runs]
        # At a crossing center the two approaching lane strokes merge into one run.
        if len(centers) != 3:
            continue
        costs = []
        for pair in range(3):
            midpoint = (lanes[pair] + lanes[pair + 1]) / 2
            costs.append(min(abs(c - midpoint) for c in centers))
        pair = int(np.argmin(costs))
        if costs[pair] < 0.25 * spacing:
            candidate_rows.append(y)
            pair_for_row[y] = pair
    groups = [g for g in _clusters(candidate_rows, gap=3) if len(g) >= 2]
    word: list[int] = []
    for group in groups:
        votes = [pair_for_row[y] for y in group]
        word.append(max(range(3), key=votes.count))
    # A valid submitted world always has seven separated events; this also makes
    # evidence erasure or unrelated drawings abstain rather than guess.
    if len(word) != 7:
        raise ValueError(f"expected seven crossing bands, recovered {len(word)}")
    cycles = _cycle_count(word)
    if cycles not in (1, 3):
        raise ValueError(f"unexpected closure component count {cycles}")
    return "one" if cycles == 1 else "three"
