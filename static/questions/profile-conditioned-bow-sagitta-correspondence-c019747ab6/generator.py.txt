"""Cross-panel bow-deflection (sagitta) correspondence renderer.

A paper-semantic correspondence task whose cross-panel identity key is the
deflection (sagitta) of a bowed segment drawn between two fixed endpoints.

Visual world:

  * At the top is a neutral gray "bow": a thick gray arc (a circular segment)
    that rises from a horizontal chord between two dark endpoint hubs. The
    amount by which the arc deflects upward from the straight chord is its
    sagitta, drawn from a finite canonical set.
  * A dark divider separates it from a column of three colored bows A, B, C
    (red / green / blue). Each colored bow shares the same chord length and hub
    size as the reference but deflects a different amount.
  * Exactly one candidate has the same sagitta as the gray reference.

The question asks for the color of the candidate whose deflection equals the
reference's. The pixel-only inverse arm must, for every panel, locate the two
dark endpoint hubs (giving the straight chord baseline) and measure the arc's
maximum upward deflection from that baseline, solve the cross-panel assignment
by sagitta equality, and only then read out the bound color of the matched bow.

Latent state z (every field influencing the gold is rendered -- nothing hidden):

    ref_sag    : sagitta (px) of the gray reference bow from CANONICAL_SAG.
    cand_sags  : length-3 candidate sagittas; exactly one equals ``ref_sag``.
    cand_colors: the fixed 3-color palette bound one-to-one to the candidates.
    width, height, divider_y, chord_x0, chord_x1, chord_y (ref and candidates),
    end_r, arc_w : fixed geometry.

The analytic gold is the color of the candidate whose sagitta equals the
reference's. Because the reference sagitta, each candidate sagitta and the bound
colors are all rendered verbatim, the gold is recoverable from the pixels alone.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "bow_sagitta_correspondence-0.1.0"

BACKGROUND = "white"
DIVIDER = (70, 70, 70)
GRAY = (150, 150, 150)
DARK = (30, 30, 30)

PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (235, 40, 40),
    "green": (0, 170, 70),
    "blue": (40, 60, 235),
}
CAND_COLOR_ORDER = ["red", "green", "blue"]

# Canonical sagittas (px). Adjacent values are 16 px apart, far above the finite
# resolution / measurement tolerance of the pixel-only oracle.
CANONICAL_SAG = [12, 28, 44, 60, 76]

WIDTH = 520
HEIGHT = 820
DIVIDER_Y = 205
DIVIDER_THICK = 3
CHORD_X0 = 140
CHORD_X1 = 380
CHORD_CX = (CHORD_X0 + CHORD_X1) / 2.0
CHORD_HALF = (CHORD_X1 - CHORD_X0) / 2.0
REF_CHORD_Y = 120
CAND_CHORD_YS = [310, 470, 630]
END_R = 10
ARC_W = 8

_QUARANTINE_FLOOR = 14.0  # px


def _arc_points(chord_y: float, sag: float) -> list[tuple[float, float]]:
    """Sample a circular segment rising to a peak ``sag`` px above the chord."""
    a = CHORD_HALF
    s = sag
    r = (a * a + s * s) / (2.0 * s) if s > 0 else float("inf")
    c_y = chord_y - s + r
    pts = []
    n = 240
    for i in range(n + 1):
        x = CHORD_X0 + (CHORD_X1 - CHORD_X0) * i / n
        y = c_y - math.sqrt(max(r * r - (x - CHORD_CX) ** 2, 0.0))
        pts.append((x, y))
    return pts


def _draw_bow(draw, chord_y: float, sag: float, arc_color, endpoints: bool = True) -> None:
    draw.line(_arc_points(chord_y, sag), fill=arc_color, width=ARC_W, joint="curve")
    if endpoints:
        for x in (CHORD_X0, CHORD_X1):
            draw.ellipse(
                [x - END_R, chord_y - END_R, x + END_R, chord_y + END_R], fill=DARK
            )


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    ref_sag = rng.choice(CANONICAL_SAG)
    pool = [s for s in CANONICAL_SAG if s != ref_sag]
    rng.shuffle(pool)
    distractors = pool[:2]
    cand_sags = [ref_sag] + distractors
    rng.shuffle(cand_sags)

    return {
        "ref_sag": ref_sag,
        "cand_sags": cand_sags,
        "cand_colors": list(PALETTE.keys()),
        "width": WIDTH,
        "height": HEIGHT,
        "divider_y": DIVIDER_Y,
        "chord_x0": CHORD_X0,
        "chord_x1": CHORD_X1,
        "ref_chord_y": REF_CHORD_Y,
        "cand_chord_ys": list(CAND_CHORD_YS),
        "end_r": END_R,
        "arc_w": ARC_W,
    }


def render(scene: dict) -> Image.Image:
    w = int(scene["width"])
    h = int(scene["height"])
    img = Image.new("RGB", (w, h), BACKGROUND)
    draw = ImageDraw.Draw(img)
    _draw_bow(draw, float(scene["ref_chord_y"]), float(scene["ref_sag"]), GRAY)
    for i, center in enumerate(scene["cand_chord_ys"]):
        color = scene["cand_colors"][i]
        _draw_bow(draw, float(center), float(scene["cand_sags"][i]), PALETTE[color])
    draw.rectangle([0, int(scene["divider_y"]), w, int(scene["divider_y"]) + DIVIDER_THICK], fill=DIVIDER)
    return img


def analytic_gold(scene: dict) -> str:
    idx = scene["cand_sags"].index(scene["ref_sag"])
    return scene["cand_colors"][idx]


def margin(scene: dict) -> float:
    """Runner-up margin: minimum |reference sagitta - distractor sagitta|."""
    ref = scene["ref_sag"]
    diffs = [abs(ref - s) for s in scene["cand_sags"] if s != ref]
    return float(min(diffs)) if diffs else 0.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < _QUARANTINE_FLOOR


def latent_symmetries(scene: dict) -> list:
    """not_applicable: every label-influencing field (reference sagitta, each
    candidate sagitta, and the bound colors) is rendered verbatim, so the raster
    uniquely determines the gold; no pixel-identical transform exists."""
    return []
