"""Independent pixel-only inverse arm for the bow-deflection correspondence.

Receives ONLY the final rasterized PNG and the public question contract (this
file's own visual spec). It never imports the renderer, prompts, generator,
verifier, or analytic-gold code, and never reads scene data.

Public visual spec:

  * the top panel holds one neutral gray "bow": a thick gray arc segment that
    rises from a horizontal chord whose two ends are each marked by a small dark
    endpoint hub. The arc's deflection (sagitta) above the straight chord is one
    of {12,28,44,60,76} pixels;
  * a dark horizontal divider separates it from a column of three colored bows
    (red, green, blue), each a thick colored arc between two dark hubs, each
    deflecting a different amount;
  * the task: decide which bow color deflects exactly as much as (same sagitta
    as) the gray reference bow.

The oracle, from pixels alone: finds the divider, locates each panel's two dark
endpoint hubs (their shared row is the straight chord baseline), measures each
arc's maximum upward deflection from that baseline by scanning the top of the
colored arc stroke in the center of the chord, bins the deflection to the
nearest canonical sagitta, and returns the color of the candidate whose binned
sagitta equals the reference's.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "bow_sagitta_correspondence-oracle-0.1.0"

PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (235, 40, 40),
    "green": (0, 170, 70),
    "blue": (40, 60, 235),
}
GRAY = (150, 150, 150)
CANONICAL_SAG = [12, 28, 44, 60, 76]
CHORD_CX = 260.0
_DARK = 90.0
_COLOR_DIST = 70.0
_CENTRAL_HALF = 44.0  # half-width of the central scan band around the chord center


def _classify(arr, color) -> np.ndarray:
    c = np.asarray(color, dtype=np.float32)
    d = np.linalg.norm(arr - c[None, None, :], axis=-1)
    return d < _COLOR_DIST


def _label_blobs(mask: np.ndarray) -> list[dict]:
    h, w = mask.shape
    labels = np.zeros((h, w), dtype=np.int32)
    comps: dict[int, dict] = {}
    cur = 0
    for i in range(h):
        for j in range(w):
            if not mask[i, j] or labels[i, j]:
                continue
            cur += 1
            stack = [(i, j)]
            labels[i, j] = cur
            rows = []
            cols = []
            while stack:
                r, c = stack.pop()
                rows.append(r)
                cols.append(c)
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < h and 0 <= nc < w and mask[nr, nc] and labels[nr, nc] == 0:
                        labels[nr, nc] = cur
                        stack.append((nr, nc))
            comps[cur] = (rows, cols, len(rows))
    blobs = []
    for rows, cols, area in comps.values():
        rmin, rmax = min(rows), max(rows)
        cmin, cmax = min(cols), max(cols)
        if 40 < area < 1000 and (rmax - rmin) < 45 and (cmax - cmin) < 45:
            blobs.append((float(np.mean(rows)), float(np.mean(cols))))
    return blobs


def _deflection(arr, baseline_y: float, color_mask: np.ndarray) -> float:
    """Maximum upward deflection of the arc stroke above the chord baseline.

    Scan the top-most colored-row column by column within the central band of
    the chord and take the shallowest such row as the arc peak.
    """
    h, w = arr.shape[:2]
    j0 = int(round(CHORD_CX - _CENTRAL_HALF))
    j1 = int(round(CHORD_CX + _CENTRAL_HALF))
    peak = baseline_y
    for j in range(max(0, j0), min(w, j1 + 1)):
        col = color_mask[:, j]
        rows = np.where(col)[0]
        if rows.size:
            peak = min(peak, float(rows.min()))
    return float(baseline_y - peak)


def _bin(sag: float) -> int:
    return min(CANONICAL_SAG, key=lambda value: abs(value - sag))


def _panel_color(arr, baseline: float) -> str:
    """Dominant palette color in the bow region of a candidate panel."""
    h, w = arr.shape[:2]
    top = max(0, int(round(baseline - 96)))
    bot = min(h, int(round(baseline + 6)))
    j0 = int(round(CHORD_CX - 120))
    j1 = int(round(CHORD_CX + 120))
    region = arr[top:bot, max(0, j0):min(w, j1 + 1), :]
    counts = {}
    for name, rgb in PALETTE.items():
        counts[name] = int(_classify(region, rgb).sum())
    best = max(counts, key=counts.get)
    if counts[best] == 0:
        raise ValueError("candidate panel has no detectable bow color")
    return best


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.float32)
    h, w = arr.shape[:2]

    dark = arr.max(axis=2) < _DARK
    row_counts = dark.sum(axis=1)
    grid = row_counts > 0.45 * dark.shape[1]
    rows = np.where(grid)[0]
    if rows.size == 0:
        raise ValueError("no divider line found")
    divider_mid = float(rows.mean())

    blobs = _label_blobs(dark)
    if not blobs:
        raise ValueError("no endpoint hubs found")
    ref_blobs = [b for b in blobs if b[0] < divider_mid]
    below = [b for b in blobs if b[0] > divider_mid]
    if len(ref_blobs) != 2:
        raise ValueError(f"expected 2 reference hubs, got {len(ref_blobs)}")
    ref_baseline = float(np.mean([b[0] for b in ref_blobs]))

    # cluster candidate hubs into 3 panels (two hubs close in row each)
    cand = sorted(below, key=lambda b: b[0])
    groups = []
    for blob in cand:
        if not groups or abs(blob[0] - groups[-1][-1][0]) > 45:
            groups.append([blob])
        else:
            groups[-1].append(blob)
    panels = [g for g in groups if len(g) >= 2]
    if len(panels) != 3:
        raise ValueError(f"expected 3 candidate panels, got {len(panels)}")

    gray_mask = _classify(arr, GRAY)
    ref_def = _deflection(arr, ref_baseline, gray_mask)
    ref_key = _bin(ref_def)

    color_masks = {name: _classify(arr, rgb) for name, rgb in PALETTE.items()}

    for panel in panels:
        baseline = float(np.mean([b[0] for b in panel]))
        color = _panel_color(arr, baseline)
        defl = _deflection(arr, baseline, color_masks[color])
        if _bin(defl) == ref_key:
            return color
    raise ValueError(f"no candidate matched reference sagitta {ref_key}")


if __name__ == "__main__":
    import sys
    img = Image.open(sys.argv[1]).convert("RGB")
    print(decision_from_image(img))
