"""Independent pixel-only inverse for convex-position dot constellations.

The inverse sees only RGB pixels.  It segments each hue, recovers dot centers
from connected components, and independently computes the convex hull of each
center set.  It contains no renderer, scene, prompt, generator, or gold import.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


LABELS = ("red", "blue", "green")
EXPECTED_DOTS = 10
MIN_COMPONENT_AREA = 160


def _label_mask(arr: np.ndarray, label: str) -> np.ndarray:
    red = arr[:, :, 0].astype(np.int16)
    green = arr[:, :, 1].astype(np.int16)
    blue = arr[:, :, 2].astype(np.int16)
    if label == "red":
        return (red > green + 55) & (red > blue + 34)
    if label == "blue":
        return (blue > red + 50) & (blue > green + 18)
    return (green > red + 42) & (green > blue + 18)


def _component_centers(mask: np.ndarray) -> list[tuple[float, float]]:
    ys, xs = np.nonzero(mask)
    active = set(zip(ys.tolist(), xs.tolist()))
    centers: list[tuple[float, float]] = []
    while active:
        start = active.pop()
        pending = [start]
        total_x = start[1]
        total_y = start[0]
        count = 1
        while pending:
            y, x = pending.pop()
            for neighbor in (
                (y - 1, x - 1),
                (y - 1, x),
                (y - 1, x + 1),
                (y, x - 1),
                (y, x + 1),
                (y + 1, x - 1),
                (y + 1, x),
                (y + 1, x + 1),
            ):
                if neighbor in active:
                    active.remove(neighbor)
                    pending.append(neighbor)
                    count += 1
                    total_x += neighbor[1]
                    total_y += neighbor[0]
        if count >= MIN_COMPONENT_AREA:
            centers.append((total_x / count, total_y / count))
    return centers


def _cross(o: tuple[float, float], a: tuple[float, float], b: tuple[float, float]) -> float:
    return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])


def _hull(points: list[tuple[float, float]]) -> list[tuple[float, float]]:
    ordered = sorted(set((round(x, 7), round(y, 7)) for x, y in points))
    if len(ordered) <= 1:
        return ordered
    lower: list[tuple[float, float]] = []
    for point in ordered:
        while len(lower) >= 2 and _cross(lower[-2], lower[-1], point) <= 1e-7:
            lower.pop()
        lower.append(point)
    upper: list[tuple[float, float]] = []
    for point in reversed(ordered):
        while len(upper) >= 2 and _cross(upper[-2], upper[-1], point) <= 1e-7:
            upper.pop()
        upper.append(point)
    return lower[:-1] + upper[:-1]


def _hull_vertex_count(points: list[tuple[float, float]]) -> int:
    return len(_hull(points))


def decision_from_image(image: Image.Image) -> str:
    """Return the hue whose ten recovered centers all lie on their hull."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    centers: dict[str, list[tuple[float, float]]] = {}
    for label in LABELS:
        found = _component_centers(_label_mask(array, label))
        if len(found) != EXPECTED_DOTS:
            raise ValueError(f"expected {EXPECTED_DOTS} {label} dots, found {len(found)}")
        centers[label] = found
    scores = {label: _hull_vertex_count(centers[label]) for label in LABELS}
    best = max(scores.values())
    winners = [label for label in LABELS if scores[label] == best]
    if len(winners) != 1:
        raise ValueError(f"expected one convex-position color, found {winners} from {scores}")
    return winners[0]
