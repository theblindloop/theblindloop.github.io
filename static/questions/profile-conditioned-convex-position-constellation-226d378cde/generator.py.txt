"""Deterministic renderer and analytic contract for convex-position constellations."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw


RENDERER_VERSION = "convex_position_constellation-0.1.0"
COLORS = ("red", "blue", "green")
WIDTH = 1000
HEIGHT = 720
DOT_COUNT = 10
OUTER_X = 142.0
OUTER_Y = 102.0
DOT_RADIUS_MIN = 10.5
DOT_RADIUS_MAX = 12.5
INTERIOR_CLEARANCE_FLOOR = 20.0
MIN_CENTER_GAP = 35.0
BACKGROUND = (248, 247, 242)

PALETTE_STYLES = (
    {
        "red": (210, 58, 66),
        "blue": (49, 106, 184),
        "green": (52, 145, 86),
    },
    {
        "red": (194, 54, 82),
        "blue": (49, 94, 174),
        "green": (39, 139, 105),
    },
    {
        "red": (218, 72, 49),
        "blue": (44, 119, 187),
        "green": (73, 150, 72),
    },
)


def _sample_angles(rng: random.Random, count: int, minimum_gap: float) -> list[float]:
    """Draw a well-spaced cyclic set of angles with no scene template."""
    base = 2.0 * math.pi / count
    jitter = min(0.11, max(0.02, (base - minimum_gap) * 0.30))
    phase = rng.uniform(-math.pi, math.pi)
    angles = [phase + index * base + rng.uniform(-jitter, jitter) for index in range(count)]
    return sorted(angle if angle <= math.pi else angle - 2.0 * math.pi for angle in angles)


def _ellipse_points(rng: random.Random, count: int) -> list[list[float]]:
    angles = _sample_angles(rng, count, 0.36 if count == 10 else 0.48)
    return [
        [round(OUTER_X * math.cos(angle), 4), round(OUTER_Y * math.sin(angle), 4)]
        for angle in angles
    ]


def _patterns(rng: random.Random) -> dict[str, list[list[float]]]:
    convex = _ellipse_points(rng, 10)
    decoy_a = _ellipse_points(rng, 8)
    decoy_b = _ellipse_points(rng, 8)
    inner_a = [
        round(rng.uniform(-35.0, 32.0), 4),
        round(rng.uniform(-28.0, 26.0), 4),
    ]
    inner_b = [
        round(rng.uniform(-30.0, 34.0), 4),
        round(rng.uniform(-25.0, 29.0), 4),
    ]
    # The inner dots are deliberately separated and well inside the eight-point
    # ellipse, while their exact offsets are randomized per scene.
    if math.hypot(inner_a[0] - inner_b[0], inner_a[1] - inner_b[1]) < 22.0:
        inner_b[0] += 26.0
    decoy_a.extend([inner_a, inner_b])

    inner_c = [
        round(rng.uniform(-34.0, 31.0), 4),
        round(rng.uniform(-29.0, 25.0), 4),
    ]
    inner_d = [
        round(rng.uniform(-29.0, 35.0), 4),
        round(rng.uniform(-24.0, 30.0), 4),
    ]
    if math.hypot(inner_c[0] - inner_d[0], inner_c[1] - inner_d[1]) < 22.0:
        inner_d[1] -= 27.0
    decoy_b.extend([inner_c, inner_d])
    return {"convex": convex, "decoy_a": decoy_a, "decoy_b": decoy_b}


def _role_for_color(scene: dict, color: str) -> str:
    target = str(scene["target_color"])
    if color == target:
        return "convex"
    decoys = [name for name in COLORS if name != target]
    return "decoy_a" if color == decoys[0] else "decoy_b"


def _world_points(scene: dict, color: str) -> list[tuple[float, float]]:
    cx, cy, rotation, scale = (float(value) for value in scene["pose"][color])
    cosine = math.cos(rotation)
    sine = math.sin(rotation)
    points = []
    for local_x, local_y in scene["patterns"][_role_for_color(scene, color)]:
        x = float(local_x) * scale
        y = float(local_y) * scale
        points.append((cx + cosine * x - sine * y, cy + sine * x + cosine * y))
    return points


def _cross(o: tuple[float, float], a: tuple[float, float], b: tuple[float, float]) -> float:
    return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])


def _convex_hull(points: Iterable[tuple[float, float]]) -> list[tuple[float, float]]:
    unique = sorted(set((round(float(x), 8), round(float(y), 8)) for x, y in points))
    if len(unique) <= 1:
        return unique
    lower: list[tuple[float, float]] = []
    for point in unique:
        while len(lower) >= 2 and _cross(lower[-2], lower[-1], point) <= 1e-8:
            lower.pop()
        lower.append(point)
    upper: list[tuple[float, float]] = []
    for point in reversed(unique):
        while len(upper) >= 2 and _cross(upper[-2], upper[-1], point) <= 1e-8:
            upper.pop()
        upper.append(point)
    return lower[:-1] + upper[:-1]


def _point_segment_distance(
    point: tuple[float, float], left: tuple[float, float], right: tuple[float, float]
) -> float:
    vx = right[0] - left[0]
    vy = right[1] - left[1]
    length_sq = vx * vx + vy * vy
    if length_sq <= 1e-12:
        return math.hypot(point[0] - left[0], point[1] - left[1])
    t = max(0.0, min(1.0, ((point[0] - left[0]) * vx + (point[1] - left[1]) * vy) / length_sq))
    return math.hypot(point[0] - (left[0] + t * vx), point[1] - (left[1] + t * vy))


def _hull_score(points: list[tuple[float, float]]) -> tuple[int, float]:
    hull = _convex_hull(points)
    hull_keys = {(round(x, 7), round(y, 7)) for x, y in hull}
    interior = [point for point in points if (round(point[0], 7), round(point[1], 7)) not in hull_keys]
    if not interior or len(hull) < 3:
        return len(hull), float("inf")
    clearance = float("inf")
    for point in interior:
        for index, left in enumerate(hull):
            clearance = min(clearance, _point_segment_distance(point, left, hull[(index + 1) % len(hull)]))
    return len(hull), clearance


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene whose label is assigned independently of nuisances."""

    rng = random.Random(int(seed))
    target_color = COLORS[int(seed) % len(COLORS)]
    lane_order = list(COLORS)
    rng.shuffle(lane_order)
    lane_x = (170.0, 500.0, 830.0)
    palette = {name: list(rgb) for name, rgb in rng.choice(PALETTE_STYLES).items()}
    common_scale = round(rng.uniform(0.96, 1.04), 4)
    dot_radius = round(rng.uniform(DOT_RADIUS_MIN, DOT_RADIUS_MAX), 4)
    pose: dict[str, list[float]] = {}
    for index, color in enumerate(lane_order):
        pose[color] = [
            round(lane_x[index] + rng.uniform(-12.0, 12.0), 4),
            round(360.0 + rng.uniform(-15.0, 15.0), 4),
            round(rng.uniform(-math.pi, math.pi), 5),
            common_scale,
        ]
    scene = {
        "target_color": target_color,
        "lane_order": lane_order,
        "palette": palette,
        "patterns": _patterns(rng),
        "pose": pose,
        "dot_radius": dot_radius,
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
    }
    if is_quarantined(scene):
        return sample_scene(int(seed) + 104729)
    return scene


def render(scene: dict) -> Image.Image:
    """Render colored dots with deterministic supersampling."""

    supersample = 4
    width = int(scene["width"])
    height = int(scene["height"])
    image = Image.new(
        "RGB",
        (width * supersample, height * supersample),
        tuple(int(value) for value in scene["background"]),
    )
    draw = ImageDraw.Draw(image)
    radius = float(scene["dot_radius"]) * supersample
    for color in scene["lane_order"]:
        fill = tuple(int(value) for value in scene["palette"][color])
        for x, y in _world_points(scene, color):
            draw.ellipse(
                [x * supersample - radius, y * supersample - radius, x * supersample + radius, y * supersample + radius],
                fill=fill,
            )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """The label is the visible color assigned the all-hull pattern."""

    return str(scene["target_color"])


def margin(scene: dict) -> float:
    scores = [_hull_score(_world_points(scene, color))[0] for color in COLORS]
    ordered = sorted(scores, reverse=True)
    return float(ordered[0] - ordered[1])


def is_quarantined(scene: dict) -> bool:
    scores = []
    for color in COLORS:
        points = _world_points(scene, color)
        if len(points) != DOT_COUNT:
            return True
        for first_index, first in enumerate(points):
            for second in points[first_index + 1 :]:
                if math.hypot(first[0] - second[0], first[1] - second[1]) < MIN_CENTER_GAP:
                    return True
        score, clearance = _hull_score(points)
        scores.append(score)
        if score < 3 or (math.isfinite(clearance) and clearance < INTERIOR_CLEARANCE_FLOOR):
            return True
    ordered = sorted(scores, reverse=True)
    return ordered[0] - ordered[1] < 1.0 or ordered[0] != DOT_COUNT


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases are declared for this fully rendered parameterization."""

    return []
