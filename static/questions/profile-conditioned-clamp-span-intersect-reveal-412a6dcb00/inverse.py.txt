"""Pixel-only inverse arm for the clamp-span intersection world.

It receives a rendered PNG and the public question contract only: it locates
the printed 5x7 lattice, finds every colour that appears both in a lane above
the lattice and in a lane to its left, intersects each colour's column run with
its row run, unions the three rectangles, and names the nearest of the five
declared symbols. It imports nothing from the renderer, prompts, generator,
verifier, or analytic gold, and never sees a scene.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ROWS = 5
COLS = 7
SYMBOLS: tuple[str, ...] = ("corner", "cross", "flag", "goalpost", "tee")

_TEMPLATE_BARS: dict[str, tuple[tuple[int, int, int, int], ...]] = {
    "cross": ((2, 2, 0, 6), (0, 4, 3, 3)),
    "tee": ((0, 0, 0, 6), (0, 4, 3, 3)),
    "corner": ((0, 4, 0, 0), (4, 4, 0, 6)),
    "flag": ((0, 4, 0, 0), (0, 1, 1, 3)),
    "goalpost": ((1, 4, 1, 1), (1, 4, 5, 5), (4, 4, 2, 4)),
}


def _expand(bars) -> frozenset[tuple[int, int]]:
    out: set[tuple[int, int]] = set()
    for r0, r1, c0, c1 in bars:
        for r in range(r0, r1 + 1):
            for c in range(c0, c1 + 1):
                out.add((r, c))
    return frozenset(out)


TEMPLATES = {name: _expand(bars) for name, bars in _TEMPLATE_BARS.items()}

UNRESOLVED = "unresolved"


def _runs(flags: np.ndarray) -> list[tuple[int, int]]:
    out: list[tuple[int, int]] = []
    start = None
    for index, value in enumerate(flags):
        if value and start is None:
            start = index
        elif not value and start is not None:
            out.append((start, index - 1))
            start = None
    if start is not None:
        out.append((start, len(flags) - 1))
    return out


def _uniform(runs: list[tuple[int, int]]) -> bool:
    lengths = [hi - lo + 1 for lo, hi in runs]
    return bool(lengths) and max(lengths) - min(lengths) <= 2


def _find_lattice(packed: np.ndarray):
    colours, counts = np.unique(packed, return_counts=True)
    order = np.argsort(-counts)
    for index in order:
        colour = colours[index]
        if counts[index] < 35 * 100:
            continue
        mask = packed == colour
        xs = mask.any(axis=0)
        ys = mask.any(axis=1)
        col_runs = _runs(xs)
        row_runs = _runs(ys)
        if len(col_runs) != COLS or len(row_runs) != ROWS:
            continue
        if not (_uniform(col_runs) and _uniform(row_runs)):
            continue
        area = sum(hi - lo + 1 for lo, hi in col_runs) * sum(
            hi - lo + 1 for lo, hi in row_runs
        )
        if abs(int(mask.sum()) - area) > area * 0.05:
            continue
        return colour, col_runs, row_runs
    return None, None, None


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint32)
    packed = (arr[:, :, 0] << 16) | (arr[:, :, 1] << 8) | arr[:, :, 2]

    lattice, col_runs, row_runs = _find_lattice(packed)
    if lattice is None:
        return UNRESOLVED

    x0 = col_runs[0][0]
    x1 = col_runs[-1][1]
    y0 = row_runs[0][0]
    y1 = row_runs[-1][1]
    col_centres = [(lo + hi) / 2.0 for lo, hi in col_runs]
    row_centres = [(lo + hi) / 2.0 for lo, hi in row_runs]

    shaded: set[tuple[int, int]] = set()
    found = 0
    for colour in np.unique(packed):
        if colour == lattice:
            continue
        mask = packed == colour
        if int(mask.sum()) < 12:
            continue
        ys, xs = np.nonzero(mask)
        top = (ys < y0) & (xs >= x0) & (xs <= x1)
        left = (xs < x0) & (ys >= y0) & (ys <= y1)
        if not top.any() or not left.any():
            continue
        if not bool((top | left).all()):
            continue
        found += 1
        lo_x, hi_x = int(xs[top].min()), int(xs[top].max())
        lo_y, hi_y = int(ys[left].min()), int(ys[left].max())
        cols = [j for j, centre in enumerate(col_centres) if lo_x <= centre <= hi_x]
        rows = [i for i, centre in enumerate(row_centres) if lo_y <= centre <= hi_y]
        for r in rows:
            for c in cols:
                shaded.add((r, c))

    if found == 0:
        return UNRESOLVED
    pattern = frozenset(shaded)
    ranked = sorted((len(pattern ^ TEMPLATES[name]), name) for name in SYMBOLS)
    return ranked[0][1]
