"""Renderer for the clamp-span intersection world.

A blank 5x7 grid is printed with three coloured bars above it (each covering a
run of columns) and three matching coloured bars to its left (each covering a
run of rows). Each colour shades the rectangle where its column run meets its
row run; the union of the three rectangles is matched to the nearest of five
named symbols. Nothing on the sheet is pre-shaded: the pattern exists only
after the declared intersection is carried out.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

ROWS = 5
COLS = 7
CANVAS = (640, 480)

SYMBOLS: tuple[str, ...] = ("corner", "cross", "flag", "goalpost", "tee")

# Each template is described by three axis-aligned bars, every bar one cell
# thick along at least one axis.  Every template covers exactly 11 cells and no
# bar is redundant: dropping any one of the three changes the revealed pattern.
TEMPLATE_BARS: dict[str, tuple[tuple[int, int, int, int], ...]] = {
    # (row_lo, row_hi, col_lo, col_hi)
    "cross": ((2, 2, 0, 6), (0, 1, 3, 3), (3, 4, 3, 3)),
    "tee": ((0, 0, 0, 6), (1, 2, 3, 3), (3, 4, 3, 3)),
    "corner": ((4, 4, 0, 6), (0, 1, 0, 0), (2, 3, 0, 0)),
    "flag": ((0, 4, 0, 0), (0, 0, 1, 3), (1, 1, 1, 3)),
    "goalpost": ((1, 4, 1, 1), (1, 4, 5, 5), (4, 4, 2, 4)),
}

PALETTE = (
    (198, 44, 58),
    (36, 96, 190),
    (26, 140, 96),
    (206, 122, 20),
    (120, 62, 178),
    (24, 148, 176),
)


def _cells(bars) -> frozenset[tuple[int, int]]:
    out: set[tuple[int, int]] = set()
    for r0, r1, c0, c1 in bars:
        for r in range(r0, r1 + 1):
            for c in range(c0, c1 + 1):
                out.add((r, c))
    return frozenset(out)


TEMPLATES: dict[str, frozenset[tuple[int, int]]] = {
    name: _cells(bars) for name, bars in TEMPLATE_BARS.items()
}


def _distances(pattern: frozenset[tuple[int, int]]) -> list[tuple[int, str]]:
    return sorted((len(pattern ^ TEMPLATES[name]), name) for name in SYMBOLS)


def _pattern(scene: dict) -> frozenset[tuple[int, int]]:
    out: set[tuple[int, int]] = set()
    for bar in scene["bars"]:
        r0, r1 = bar["rows"]
        c0, c1 = bar["cols"]
        for r in range(r0, r1 + 1):
            for c in range(c0, c1 + 1):
                out.add((r, c))
    return frozenset(out)


def _perturb(bar: list[int], rng: random.Random) -> list[int] | None:
    """Move one end of a bar by one cell along its long axis."""
    r0, r1, c0, c1 = bar
    if r0 == r1 and c1 > c0:
        axis = "c"
    elif c0 == c1 and r1 > r0:
        axis = "r"
    elif r0 == r1 and c0 == c1:
        axis = rng.choice(("r", "c"))
    else:
        return None
    step = rng.choice((-1, 1))
    end = rng.choice((0, 1))
    r0, r1, c0, c1 = r0, r1, c0, c1
    if axis == "c":
        if end == 0:
            c0 += step
        else:
            c1 += step
        if not (0 <= c0 <= c1 <= COLS - 1):
            return None
    else:
        if end == 0:
            r0 += step
        else:
            r1 += step
        if not (0 <= r0 <= r1 <= ROWS - 1):
            return None
    return [r0, r1, c0, c1]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 6364136223846793005 % (2**63) + 17)
    # The target symbol cycles with the scene seed so that any contiguous block
    # of scenes is class balanced; nothing in the raster exposes the seed.
    target = SYMBOLS[seed % len(SYMBOLS)]

    for _ in range(400):
        bars = [list(bar) for bar in TEMPLATE_BARS[target]]
        moves = rng.choice((0, 0, 1, 1, 2, 2, 3, 4, 5, 6))
        for _ in range(moves):
            index = rng.randrange(len(bars))
            moved = _perturb(bars[index], rng)
            if moved is not None:
                bars[index] = moved
        pattern = _cells(tuple(tuple(bar) for bar in bars))
        ranked = _distances(pattern)
        if ranked[0][1] != target:
            continue
        break
    else:  # pragma: no cover - the exact decomposition always matches
        bars = [list(bar) for bar in TEMPLATE_BARS[target]]

    cell = rng.choice(
        [rng.randint(15, 21)] if rng.random() < 0.18 else [rng.randint(22, 34)]
    )
    thick = max(6, round(cell * 0.45))
    gap = max(3, round(cell * 0.22))
    lane_span = 3 * (thick + gap)
    grid_w = COLS * cell
    grid_h = ROWS * cell
    left = rng.randint(lane_span + 12, CANVAS[0] - grid_w - 34)
    top = rng.randint(lane_span + 12, CANVAS[1] - grid_h - 34)

    colours = rng.sample(PALETTE, 3)
    lanes = list(range(3))
    rng.shuffle(lanes)

    scene_bars = []
    for index, bar in enumerate(bars):
        r0, r1, c0, c1 = bar
        scene_bars.append(
            {
                "lane": lanes[index],
                "colour": list(colours[index]),
                "rows": [r0, r1],
                "cols": [c0, c1],
            }
        )

    marks = []
    for _ in range(rng.randint(3, 7)):
        size = rng.randint(5, 10)
        if rng.random() < 0.5:
            mx = rng.randint(left + grid_w + 12, CANVAS[0] - size - 8)
            my = rng.randint(10, CANVAS[1] - size - 10)
        else:
            mx = rng.randint(10, CANVAS[0] - size - 8)
            my = rng.randint(top + grid_h + 12, CANVAS[1] - size - 10)
        marks.append([mx, my, size])

    shade = rng.randint(232, 244)
    return {
        "bars": scene_bars,
        "marks": marks,
        "cell": cell,
        "thick": thick,
        "gap": gap,
        "left": left,
        "top": top,
        "bg": [shade, shade, rng.randint(228, 240)],
        "plate": [rng.randint(198, 214)] * 3,
        "mark": [rng.randint(150, 168)] * 3,
    }


def render(scene: dict) -> Image.Image:
    cell = int(scene["cell"])
    thick = int(scene["thick"])
    gap = int(scene["gap"])
    left = int(scene["left"])
    top = int(scene["top"])
    image = Image.new("RGB", CANVAS, tuple(scene["bg"]))
    draw = ImageDraw.Draw(image)

    plate = tuple(scene["plate"])
    for r in range(ROWS):
        for c in range(COLS):
            x0 = left + c * cell
            y0 = top + r * cell
            draw.rectangle([x0 + 1, y0 + 1, x0 + cell - 2, y0 + cell - 2], fill=plate)

    for x, y, size in scene["marks"]:
        draw.rectangle([x, y, x + size, y + size], fill=tuple(scene["mark"]))

    for bar in scene["bars"]:
        colour = tuple(bar["colour"])
        lane = int(bar["lane"])
        c0, c1 = bar["cols"]
        r0, r1 = bar["rows"]
        by = top - (lane + 1) * (thick + gap)
        draw.rectangle(
            [left + c0 * cell + 1, by, left + (c1 + 1) * cell - 2, by + thick - 1],
            fill=colour,
        )
        bx = left - (lane + 1) * (thick + gap)
        draw.rectangle(
            [bx, top + r0 * cell + 1, bx + thick - 1, top + (r1 + 1) * cell - 2],
            fill=colour,
        )
    return image


def analytic_gold(scene: dict) -> str:
    return _distances(_pattern(scene))[0][1]


def margin(scene: dict) -> float:
    ranked = _distances(_pattern(scene))
    gap_cells = float(ranked[1][0] - ranked[0][0])
    resolution = (float(scene["cell"]) - 12.0) / 2.5
    return round(min(gap_cells, resolution), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Declared pixel-identical rewrites of the latent scene."""
    reversed_bars = dict(scene)
    reversed_bars["bars"] = list(reversed(scene["bars"]))

    duplicated = dict(scene)
    duplicated["bars"] = list(scene["bars"]) + [dict(scene["bars"][0])]

    reordered_marks = dict(scene)
    reordered_marks["marks"] = list(reversed(scene["marks"]))
    return [
        ("reverse_bar_order", reversed_bars),
        ("duplicate_first_bar", duplicated),
        ("reverse_mark_order", reordered_marks),
    ]


def _best_distance(scene: dict) -> int:
    """Cell distance from the revealed pattern to its nearest symbol template."""
    return _distances(_pattern(scene))[0][0]


def _template_gap(scene: dict) -> int:
    """Uncapped cell gap between the best and runner-up symbol templates."""
    ranked = _distances(_pattern(scene))
    return ranked[1][0] - ranked[0][0]


def _shortcut_distance(scene: dict) -> int:
    """How far the column-only shortcut lands from the true pattern.

    A model that reads only the top bars and shades every covered column in
    full ignores the declared intersection; this reports the cell distance
    between that shortcut and the answer's template.
    """
    shortcut: set[tuple[int, int]] = set()
    for bar in scene["bars"]:
        c0, c1 = bar["cols"]
        for c in range(c0, c1 + 1):
            for r in range(ROWS):
                shortcut.add((r, c))
    return len(frozenset(shortcut) ^ TEMPLATES[analytic_gold(scene)])
