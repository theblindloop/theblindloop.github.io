"""Independent final-PNG-only inverse arm.

This module deliberately contains no imports from any candidate world module and
receives neither scene state nor analytic labels.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

_CENTER = np.array((236, 126, 45), dtype=np.int16)
_TAIL = np.array((29, 183, 195), dtype=np.int16)
_HEAD = np.array((37, 65, 125), dtype=np.int16)
_LABELS = ("A", "B", "C", "D")


def _mask(arr: np.ndarray, color: np.ndarray) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - color), axis=2) <= 25


def _components(mask: np.ndarray, min_area: int) -> list[tuple[float, float, int]]:
    remaining = {tuple(p) for p in np.argwhere(mask)}
    components: list[tuple[float, float, int]] = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        pixels = [start]
        while stack:
            y, x = stack.pop()
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    neighbor = (y + dy, x + dx)
                    if neighbor in remaining:
                        remaining.remove(neighbor)
                        stack.append(neighbor)
                        pixels.append(neighbor)
        if len(pixels) >= min_area:
            ys = sum(p[0] for p in pixels) / len(pixels)
            xs = sum(p[1] for p in pixels) / len(pixels)
            components.append((xs, ys, len(pixels)))
    return components


def _ordered_centers(points: list[tuple[float, float, int]]) -> list[tuple[float, float]]:
    if len(points) != 4:
        raise ValueError(f"expected four orange centers, found {len(points)}")
    by_y = sorted(points, key=lambda p: p[1])
    top = sorted(by_y[:2], key=lambda p: p[0])
    bottom = sorted(by_y[2:], key=lambda p: p[0])
    return [(p[0], p[1]) for p in top + bottom]


def _nearest_center(point: tuple[float, float, int], centers: list[tuple[float, float]]) -> int:
    return min(range(4), key=lambda i: math.hypot(point[0] - centers[i][0],
                                                   point[1] - centers[i][1]))


def _pair(tails: list[tuple[float, float, int]],
          heads: list[tuple[float, float, int]]) -> list[tuple[tuple[float, float, int],
                                                              tuple[float, float, int]]]:
    if len(tails) != 4 or len(heads) != 4:
        raise ValueError(f"panel inventory is {len(tails)} tails and {len(heads)} tips")
    choices = sorted((math.hypot(t[0] - h[0], t[1] - h[1]), ti, hi)
                     for ti, t in enumerate(tails) for hi, h in enumerate(heads))
    used_t: set[int] = set()
    used_h: set[int] = set()
    pairs = []
    for distance, ti, hi in choices:
        if ti in used_t or hi in used_h:
            continue
        if distance > 55.0:
            raise ValueError("tail-tip pairing is too distant")
        used_t.add(ti)
        used_h.add(hi)
        pairs.append((tails[ti], heads[hi]))
    if len(pairs) != 4:
        raise ValueError("could not form four one-to-one arrows")
    return pairs


def decision_from_image(image: Image.Image) -> str:
    """Enumerate panels and select the unique all-clockwise ring from RGB pixels."""
    arr = np.asarray(image.convert("RGB"))
    centers = _ordered_centers(_components(_mask(arr, _CENTER), 120))
    tails = _components(_mask(arr, _TAIL), 55)
    heads = _components(_mask(arr, _HEAD), 90)
    if len(tails) != 16 or len(heads) != 16:
        raise ValueError(f"expected 16 tails and tips, found {len(tails)} and {len(heads)}")
    panel_tails = [[] for _ in range(4)]
    panel_heads = [[] for _ in range(4)]
    for point in tails:
        panel_tails[_nearest_center(point, centers)].append(point)
    for point in heads:
        panel_heads[_nearest_center(point, centers)].append(point)

    scores = []
    for index, (cx, cy) in enumerate(centers):
        alignments = []
        for tail, head in _pair(panel_tails[index], panel_heads[index]):
            tx, ty = tail[:2]
            hx, hy = head[:2]
            mx, my = (tx + hx) / 2.0, (ty + hy) / 2.0
            rx, ry = mx - cx, my - cy
            vx, vy = hx - tx, hy - ty
            denom = math.hypot(rx, ry) * math.hypot(vx, vy)
            if denom == 0:
                raise ValueError("degenerate visible arrow")
            alignments.append((rx * vy - ry * vx) / denom)
        scores.append(min(alignments))
    order = sorted(range(4), key=lambda i: scores[i], reverse=True)
    if scores[order[0]] < 0.70 or scores[order[0]] - scores[order[1]] < 0.50:
        raise ValueError("no unique all-clockwise panel with sufficient pixel margin")
    return _LABELS[order[0]]
