"""Deterministic latent sampler and raster renderer for arrow-ring search."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 640, 500
SCALE = 3
LABELS = ("A", "B", "C", "D")
PANEL_BOXES = ((30, 28, 310, 242), (330, 28, 610, 242),
               (30, 258, 310, 472), (330, 258, 610, 472))
BASE_CENTERS = ((170, 137), (470, 137), (170, 367), (470, 367))

BACKGROUND = (250, 249, 246)
PANEL_FILL = (255, 255, 255)
PANEL_EDGE = (178, 183, 190)
TEXT = (58, 61, 66)
CONNECTOR = (132, 139, 148)
CENTER_COLOR = (236, 126, 45)
TAIL_COLOR = (29, 183, 195)
HEAD_COLOR = (37, 65, 125)


def _unit(x: float, y: float) -> tuple[float, float]:
    n = math.hypot(x, y)
    if n == 0:
        raise ValueError("zero-length vector")
    return x / n, y / n


def _panel_scores(scene: dict) -> list[float]:
    scores: list[float] = []
    for panel in scene["panels"]:
        cx, cy = panel["center"]
        alignments = []
        for arrow in panel["arrows"]:
            tx, ty = arrow["tail"]
            hx, hy = arrow["head"]
            mx, my = (tx + hx) / 2.0, (ty + hy) / 2.0
            rx, ry = mx - cx, my - cy
            vx, vy = hx - tx, hy - ty
            denom = math.hypot(rx, ry) * math.hypot(vx, vy)
            if denom == 0:
                alignments.append(-1.0)
            else:
                # With image y increasing downward, positive cross product is clockwise.
                alignments.append((rx * vy - ry * vx) / denom)
        scores.append(min(alignments) if alignments else -1.0)
    return scores


def sample_scene(seed: int) -> dict:
    """Sample four balanced panels; seed modulo four counterbalances the label."""
    rng = random.Random(int(seed))
    target = int(seed) % 4
    flip_counts = [1, 2, 3]
    rng.shuffle(flip_counts)
    distractor_number = 0
    panels = []
    for panel_index, (base_x, base_y) in enumerate(BASE_CENTERS):
        cx = base_x + rng.uniform(-4.0, 4.0)
        cy = base_y + rng.uniform(-4.0, 4.0)
        phase = rng.uniform(0.0, 2.0 * math.pi)
        if panel_index == target:
            flipped: set[int] = set()
        else:
            count = flip_counts[distractor_number]
            distractor_number += 1
            flipped = set(rng.sample(range(4), count))
        arrows = []
        for k in range(4):
            theta = phase + k * math.pi / 2.0 + rng.uniform(-0.035, 0.035)
            radius = rng.uniform(65.0, 70.0)
            mx = cx + radius * math.cos(theta)
            my = cy + radius * math.sin(theta)
            tangent_angle = theta + math.pi / 2.0 + rng.uniform(-0.045, 0.045)
            ux, uy = math.cos(tangent_angle), math.sin(tangent_angle)
            if k in flipped:
                ux, uy = -ux, -uy
            half_span = rng.uniform(15.0, 17.0)
            arrows.append({
                "tail": [round(mx - half_span * ux, 3), round(my - half_span * uy, 3)],
                "head": [round(mx + half_span * ux, 3), round(my + half_span * uy, 3)],
            })
        rng.shuffle(arrows)
        panels.append({
            "center": [round(cx, 3), round(cy, 3)],
            "arrows": arrows,
        })
    return {"panels": panels}


def analytic_gold(scene: dict) -> str:
    scores = _panel_scores(scene)
    order = sorted(range(4), key=lambda i: scores[i], reverse=True)
    if scores[order[0]] - scores[order[1]] <= 1e-9:
        raise ValueError("scene has no unique circulation winner")
    return LABELS[order[0]]


def margin(scene: dict) -> float:
    scores = sorted(_panel_scores(scene), reverse=True)
    return scores[0] - scores[1]


def is_quarantined(scene: dict) -> bool:
    scores = _panel_scores(scene)
    winners = [value for value in scores if value > 0.75]
    return len(winners) != 1 or margin(scene) < 0.5


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    for panel in twin["panels"]:
        panel["arrows"].reverse()
    return [("reverse_arrow_enumeration", twin)]


def _sc_point(point: tuple[float, float] | list[float]) -> tuple[int, int]:
    return int(round(point[0] * SCALE)), int(round(point[1] * SCALE))


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf", size * SCALE)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    """Render only public scene fields to a deterministic antialiased RGB PNG."""
    canvas = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(canvas)
    for index, panel in enumerate(scene["panels"]):
        x0, y0, x1, y1 = PANEL_BOXES[index]
        draw.rounded_rectangle(
            (x0 * SCALE, y0 * SCALE, x1 * SCALE, y1 * SCALE),
            radius=13 * SCALE,
            fill=PANEL_FILL,
            outline=PANEL_EDGE,
            width=2 * SCALE,
        )
        draw.text(((x0 + 13) * SCALE, (y0 + 9) * SCALE), LABELS[index],
                  fill=TEXT, font=_font(21))
        cx, cy = panel["center"]
        # Neutral connectors are drawn first; colored role markers remain intact.
        for arrow in panel["arrows"]:
            tail = arrow["tail"]
            head = arrow["head"]
            draw.line((_sc_point(tail), _sc_point(head)), fill=CONNECTOR,
                      width=4 * SCALE)
        for arrow in panel["arrows"]:
            tx, ty = arrow["tail"]
            hx, hy = arrow["head"]
            ux, uy = _unit(hx - tx, hy - ty)
            nx, ny = -uy, ux
            apex = (hx + 15.0 * ux, hy + 15.0 * uy)
            rear1 = (hx - 9.0 * ux + 11.0 * nx, hy - 9.0 * uy + 11.0 * ny)
            rear2 = (hx - 9.0 * ux - 11.0 * nx, hy - 9.0 * uy - 11.0 * ny)
            draw.polygon([_sc_point(apex), _sc_point(rear1), _sc_point(rear2)],
                         fill=HEAD_COLOR)
            r = 7.0 * SCALE
            stx, sty = _sc_point((tx, ty))
            draw.ellipse((stx - r, sty - r, stx + r, sty + r), fill=TAIL_COLOR)
        ccx, ccy = _sc_point((cx, cy))
        rr = 10.0 * SCALE
        draw.ellipse((ccx - rr, ccy - rr, ccx + rr, ccy + rr),
                     fill=CENTER_COLOR)
    return canvas.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
