"""Deterministic renderer and analytic arm for the row-zoned dual-rule cascade.

A 5-by-5 grid of square cells is drawn; each cell is filled either dark (an ON
cell) or pale (an OFF cell).  To the left of the grid a column of five large
coloured tags sits, one beside every row.  An amber tag marks a "two" row; a
violet tag marks a "three" row.  Below the grid a small key of teal dots shows
the number of transition rounds S to apply (one, two, or three dots).

The transition is a non-uniform whole-grid cellular rule: for every cell count
how many of its eight surrounding neighbours (the eight cells sharing an edge or
a corner with it) are dark.  A cell on an amber row becomes dark in the next
round if and only if that count is exactly two; a cell on a violet row becomes
dark if and only if that count is exactly three; in every other case the cell
becomes pale, and a dark cell does not automatically stay dark.  The rule is
applied S times, each round recomputing every cell from the current grid, with
the row tags fixed; the question asks whether the final number of dark cells is
odd.  The intermediate and terminal grids are never drawn; they must be
recovered by iterating the visible initial grid through S rounds of the rule.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = (1000, 920)
N = 5
CELL = 76
GAP = 16
GRID_LEFT = 360
GRID_TOP = 150
ZS = 60
ZONE_LEFT = 150
DOT_Y = 760
DOT_R = 22
DOT_SP = 56
BACKGROUND = (250, 250, 250)
OUTLINE = (128, 128, 128)
ON_FILL = (35, 35, 35)
OFF_FILL = (235, 235, 235)
ZONE_TWO = (235, 160, 35)      # amber tag -> "exactly two" rule
ZONE_THREE = (180, 190, 240)   # violet tag -> "exactly three" rule
DOT_COLOR = (0, 150, 150)
MAX_STEP = 3
DUTY = 0.45


def step(grid: list[list[int]], zones: list[int]) -> list[list[int]]:
    """One round of the row-zoned two/three-neighbourhood cascade.

    zones[i] == 1 selects the "exactly two" rule for row i, zones[i] == 0
    selects the "exactly three" rule for row i.
    """

    n = len(grid)
    m = len(grid[0])
    out = [[0] * m for _ in range(n)]
    for i in range(n):
        threshold = 2 if zones[i] == 1 else 3
        for j in range(m):
            dark = 0
            for di in (-1, 0, 1):
                for dj in (-1, 0, 1):
                    if di == 0 and dj == 0:
                        continue
                    ni, nj = i + di, j + dj
                    if 0 <= ni < n and 0 <= nj < m:
                        dark += grid[ni][nj]
            out[i][j] = 1 if dark == threshold else 0
    return out


def _simulate(grid: list[list[int]], zones: list[int], steps: int) -> list[list[int]]:
    current = [list(row) for row in grid]
    for _ in range(steps):
        current = step(current, zones)
    return current


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    grid = [[1 if rng.random() < DUTY else 0 for _ in range(N)] for _ in range(N)]
    zones = [1 if rng.random() < 0.5 else 0 for _ in range(N)]
    steps = rng.randint(1, MAX_STEP)
    return {
        "canvas_size": list(CANVAS),
        "grid": grid,
        "zones": zones,
        "steps": steps,
        "cell_px": CELL,
        "gap_px": GAP,
        "grid_left": GRID_LEFT,
        "grid_top": GRID_TOP,
        "zone_left": ZONE_LEFT,
        "zone_px": ZS,
        "dot_y": DOT_Y,
        "dot_r": DOT_R,
        "dot_spacing": DOT_SP,
    }


def render(scene: dict) -> Image.Image:
    width, height = (int(v) for v in scene["canvas_size"])
    grid = [[int(v) for v in row] for row in scene["grid"]]
    zones = [int(v) for v in scene["zones"]]
    steps = int(scene["steps"])
    cell = int(scene["cell_px"])
    gap = int(scene["gap_px"])
    gl = int(scene["grid_left"])
    gt = int(scene["grid_top"])
    zl = int(scene["zone_left"])
    zs = int(scene["zone_px"])
    dot_y = int(scene["dot_y"])
    dot_r = int(scene["dot_r"])
    dot_spacing = int(scene["dot_spacing"])
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for i, row in enumerate(grid):
        row_top = gt + i * (cell + gap)
        zone_fill = ZONE_TWO if zones[i] == 1 else ZONE_THREE
        draw.rectangle((zl, row_top + 8, zl + zs, row_top + 8 + zs),
                       fill=zone_fill)
        for j, value in enumerate(row):
            x = gl + j * (cell + gap)
            y = row_top
            draw.rectangle(
                (x, y, x + cell, y + cell),
                fill=ON_FILL if value == 1 else OFF_FILL,
                outline=OUTLINE,
                width=3,
            )
    total_dots = steps * dot_spacing
    x0 = (width - total_dots) // 2 + dot_spacing // 2
    for k in range(steps):
        cx = x0 + k * dot_spacing
        draw.ellipse(
            (cx - dot_r, dot_y - dot_r, cx + dot_r, dot_y + dot_r),
            fill=DOT_COLOR,
        )
    return image


def analytic_gold(scene: dict) -> str:
    grid = [[int(v) for v in row] for row in scene["grid"]]
    zones = [int(v) for v in scene["zones"]]
    steps = int(scene["steps"])
    final = _simulate(grid, zones, steps)
    return "yes" if sum(sum(row) for row in final) % 2 == 1 else "no"


def margin(scene: dict) -> float:
    return 1.0


def is_quarantined(scene: dict) -> bool:
    grid = scene["grid"]
    zones = scene["zones"]
    steps = int(scene["steps"])
    if len(grid) != N or any(len(row) != N for row in grid):
        return True
    if len(zones) != N:
        return True
    if steps < 1 or steps > MAX_STEP:
        return True
    if any(value not in (0, 1) for row in grid for value in row):
        return True
    if any(value not in (0, 1) for value in zones):
        return True
    return False


def latent_symmetries(scene: dict) -> list:
    return []
