"""Independent PNG-only inverse arm for the row-zoned dual-rule cascade.

Recovers the 5-by-5 cell ON/OFF states, the five row-zone tags (amber marks a
"two" row, violet marks a "three" row) and the number of transition rounds S
directly from the final raster, then iterates the non-uniform cascade rule S
times and returns whether the final dark-cell count is odd.  Uses no renderer
geometry, scene parameters, prompt, or gold code.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

N = 5
MAX_STEP = 3


def _structure_mask(arr: np.ndarray) -> np.ndarray:
    chroma = arr.max(axis=2).astype(np.int16) - arr.min(axis=2).astype(np.int16)
    grays = chroma <= 12
    notwhite = (255 - arr.mean(axis=2)) >= 10
    return grays & notwhite


def _teal_mask(arr: np.ndarray) -> np.ndarray:
    r = arr[:, :, 0].astype(np.int16)
    g = arr[:, :, 1].astype(np.int16)
    b = arr[:, :, 2].astype(np.int16)
    return (g > 90) & (r < 80) & (b > 90) & ((b - r) > 25)


def _amber_mask(arr: np.ndarray) -> np.ndarray:
    r = arr[:, :, 0].astype(np.int16)
    g = arr[:, :, 1].astype(np.int16)
    b = arr[:, :, 2].astype(np.int16)
    return (r > 200) & (g > 120) & (b < 90) & ((r - b) > 60)


def _violet_mask(arr: np.ndarray) -> np.ndarray:
    r = arr[:, :, 0].astype(np.int16)
    g = arr[:, :, 1].astype(np.int16)
    b = arr[:, :, 2].astype(np.int16)
    return (b > 200) & (r < 225) & ((b - r) > 20)


def _components(mask: np.ndarray, minsize: int) -> int:
    pixels = set(zip(*np.nonzero(mask)))
    count = 0
    while pixels:
        start = pixels.pop()
        stack = [start]
        size = 1
        while stack:
            y, x = stack.pop()
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if (ny, nx) in pixels:
                    pixels.remove((ny, nx))
                    size += 1
                    stack.append((ny, nx))
        if size >= minsize:
            count += 1
    return count


def _bands(colmask: np.ndarray, minw: int) -> list[tuple[int, int]]:
    trans = np.diff(colmask.astype(np.int8))
    starts = np.flatnonzero(trans == 1) + 1
    ends = np.flatnonzero(trans == -1) + 1
    if colmask[0]:
        starts = np.concatenate(([0], starts))
    if colmask[-1]:
        ends = np.concatenate((ends, [len(colmask)]))
    return [(int(s), int(e)) for s, e in zip(starts, ends) if int(e) - int(s) >= minw]


def _onestep(grid: list[list[int]], zones: list[int]) -> list[list[int]]:
    n = len(grid)
    m = len(grid[0])
    out = [[0] * m for _ in range(n)]
    for i in range(n):
        threshold = 2 if zones[i] == 1 else 3
        for j in range(m):
            dark = 0
            for di in (-1, 0, 1):
                for dj in (-1, 0, 1):
                    if di == 0 and dj == 0:
                        continue
                    ni, nj = i + di, j + dj
                    if 0 <= ni < n and 0 <= nj < m:
                        dark += grid[ni][nj]
            out[i][j] = 1 if dark == threshold else 0
    return out


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    struct = _structure_mask(arr)
    rowmask = struct.any(axis=1)
    rbands = _bands(rowmask, 25)
    if len(rbands) != N:
        raise ValueError(f"expected {N} grid row bands, found {len(rbands)}")
    block_top = rbands[0][0]
    block_bottom = rbands[-1][1]
    block = struct[block_top:block_bottom, :]
    subrow = block.any(axis=1)
    rowbands = _bands(subrow, 25)
    if len(rowbands) != N:
        raise ValueError(f"expected {N} cell-row bands, found {len(rowbands)}")
    cells = [[0] * N for _ in range(N)]
    for i, (ry0, ry1) in enumerate(rowbands):
        band = block[ry0:ry1, :]
        colmask = band.any(axis=0)
        cbands = _bands(colmask, 25)
        if len(cbands) != N:
            raise ValueError(f"row {i}: expected {N} column bands, found {len(cbands)}")
        for j, (cx0, cx1) in enumerate(cbands):
            x = (cx0 + cx1) // 2
            y = block_top + (ry0 + ry1) // 2
            px = arr[y, x].astype(np.int16)
            cells[i][j] = 1 if float(px.mean()) < 120 else 0
    amber = _amber_mask(arr)
    violet = _violet_mask(arr)
    zones = [0] * N
    for i, (ry0, ry1) in enumerate(rowbands):
        a = int(amber[block_top + ry0:block_top + ry1, :].sum())
        v = int(violet[block_top + ry0:block_top + ry1, :].sum())
        if a == 0 and v == 0:
            raise ValueError(f"row {i}: no zone tag found")
        zones[i] = 1 if a > v else 0
    steps = _components(_teal_mask(arr), 25)
    if steps < 1 or steps > MAX_STEP:
        raise ValueError(f"invalid round count {steps}")
    current = [list(row) for row in cells]
    for _ in range(steps):
        current = _onestep(current, zones)
    return "yes" if sum(sum(row) for row in current) % 2 == 1 else "no"
