"""Independent image-only oracle for paired local connectivity glyphs."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int = 1) -> list[np.ndarray]:
    points = np.argwhere(mask)
    seen = np.zeros(mask.shape, dtype=bool)
    height, width = mask.shape
    found: list[np.ndarray] = []
    for y, x in points:
        y = int(y)
        x = int(x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        values: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            values.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    ny, nx = cy + dy, cx + dx
                    if (
                        0 <= ny < height
                        and 0 <= nx < width
                        and mask[ny, nx]
                        and not seen[ny, nx]
                    ):
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        if len(values) >= minimum:
            found.append(np.asarray(values, dtype=np.int32))
    return found


def _dilate(mask: np.ndarray, radius: int = 3) -> np.ndarray:
    if radius <= 0:
        return mask.copy()
    height, width = mask.shape
    padded = np.pad(mask, radius, mode="constant")
    output = np.zeros_like(mask)
    for dy in range(2 * radius + 1):
        for dx in range(2 * radius + 1):
            output |= padded[dy : dy + height, dx : dx + width]
    return output


def _color_mask(array: np.ndarray) -> np.ndarray:
    values = array.astype(np.int16)
    high = values.max(axis=2)
    low = values.min(axis=2)
    # Network ink and remote decoration are near-gray. Saturation is the
    # public visual marker for exactly the two highlighted links.
    return (high - low >= 45) & (high >= 100) & (low <= 240)


def _status_for_component(array: np.ndarray, component: np.ndarray) -> bool:
    """Return True for bridge, False for loop from one highlighted link."""

    ys = component[:, 0]
    xs = component[:, 1]
    center_x = float(xs.mean())
    center_y = float(ys.mean())
    half = 28
    height, width = array.shape[:2]
    x0 = max(0, int(round(center_x)) - half)
    x1 = min(width, int(round(center_x)) + half + 1)
    y0 = max(0, int(round(center_y)) - half)
    y1 = min(height, int(round(center_y)) + half + 1)
    crop = array[y0:y1, x0:x1]
    colors = _color_mask(crop)
    values = crop.astype(np.int16)
    foreground = np.linalg.norm(values - 255, axis=2) >= 24
    # Removing the highlighted link leaves two endpoint components for a
    # bridge and one connected component for a loop.
    remaining = foreground & ~_dilate(colors, radius=3)
    graph_components = _components(remaining, minimum=8)
    if len(graph_components) < 1:
        raise ValueError("highlighted link has no recoverable local network")
    return len(graph_components) >= 2


def decision_from_image(image: Image.Image) -> str:
    """Recover ``same`` or ``different`` from the final PNG only."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    mask = _color_mask(array)
    components = _components(mask, minimum=12)
    if len(components) != 2:
        raise ValueError(f"expected two highlighted links, found {len(components)}")
    components.sort(key=lambda item: (float(item[:, 1].mean()), float(item[:, 0].mean())))
    statuses = [_status_for_component(array, item) for item in components]
    return "same" if statuses[0] == statuses[1] else "different"
