"""Deterministic raster world for paired local bridge-status comparison."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw


RENDERER_VERSION = "paired_bridge_signature-0.1.0"
PALETTE = ("#e4572e", "#2185d0", "#18a999", "#b23a8f")
BACKGROUND = "#ffffff"
NETWORK_INK = "#27313b"
DECOR_INK = "#d8dde4"
QUARANTINE_CLEARANCE = 12


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced scene; the seed encodes a deterministic four-cell mix."""

    rng = random.Random(int(seed))
    pattern = int(seed) % 4
    if pattern == 0:
        status_a, status_b = "bridge", "bridge"
    elif pattern == 1:
        status_a, status_b = "loop", "loop"
    elif pattern == 2:
        status_a, status_b = "bridge", "loop"
    else:
        status_a, status_b = "loop", "bridge"

    colors = rng.sample(list(PALETTE), 2)
    return {
        "width": 512,
        "height": 512,
        # Centers stay in separate ablation cells, with a small random
        # translation inside each cell.
        "origin_x": 224 + rng.choice((-2, -1, 0, 1, 2)),
        "origin_y": 256 + rng.choice((-1, 0, 1)),
        "card_gap": 128,
        "status_a": status_a,
        "status_b": status_b,
        "color_a": colors[0],
        "color_b": colors[1],
        "route_side_a": rng.choice((-1, 1)),
        "route_side_b": rng.choice((-1, 1)),
        # Lower-clearance samples are visible but explicitly quarantined.
        "clearance_a": rng.choice((9, 10, 14, 15, 16)),
        "clearance_b": rng.choice((9, 10, 14, 15, 16)),
        "stroke_width": 6,
        "marker_radius": 4,
        "decor_seed": rng.randrange(1, 2**31 - 1),
    }


def _point(point: tuple[float, float], scale: int) -> tuple[int, int]:
    return (int(round(point[0] * scale)), int(round(point[1] * scale)))


def _draw_graph(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    status: str,
    color: str,
    route_side: int,
    clearance: int,
    stroke_width: int,
    marker_radius: int,
    scale: int,
) -> None:
    """Draw one compact graph; a loop has a dark alternate route."""

    cx, cy = center
    left = (cx - 9, cy)
    right = (cx + 9, cy)
    if status == "loop":
        alternate_y = cy + int(route_side) * int(clearance)
        route = [left, (left[0], alternate_y), (right[0], alternate_y), right]
        draw.line(
            [_point(item, scale) for item in route],
            fill=NETWORK_INK,
            width=max(1, int(round(stroke_width * scale))),
            joint="curve",
        )

    # The highlighted link is drawn before dark endpoint markers. Its visible
    # middle is a saturated component, while markers remain in the graph.
    draw.line(
        [_point(left, scale), _point(right, scale)],
        fill=color,
        width=max(1, int(round(stroke_width * scale))),
    )
    for x, y in (left, right):
        radius = marker_radius * scale
        draw.ellipse(
            [
                int(round(x * scale - radius)),
                int(round(y * scale - radius)),
                int(round(x * scale + radius)),
                int(round(y * scale + radius)),
            ],
            fill=NETWORK_INK,
        )


def _draw_decoration(draw: ImageDraw.ImageDraw, seed: int, scale: int) -> None:
    """Add neutral remote marks that are deliberately nondiagnostic."""

    rng = random.Random(int(seed))
    positions = []
    for corner_x, corner_y in ((52, 56), (448, 58), (52, 448), (448, 448)):
        positions.append(
            (
                corner_x + rng.randint(-18, 18),
                corner_y + rng.randint(-18, 18),
            )
        )
    for x, y in positions:
        radius = 2 * scale
        draw.ellipse(
            [
                int(x * scale - radius),
                int(y * scale - radius),
                int(x * scale + radius),
                int(y * scale + radius),
            ],
            fill=DECOR_INK,
        )
    draw.line(
        [_point((24, 128), scale), _point((70, 128), scale)],
        fill=DECOR_INK,
        width=max(1, scale),
    )
    draw.line(
        [_point((442, 384), scale), _point((488, 384), scale)],
        fill=DECOR_INK,
        width=max(1, scale),
    )


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only from the public scene dictionary."""

    width = int(scene["width"])
    height = int(scene["height"])
    origin_x = float(scene["origin_x"])
    origin_y = float(scene["origin_y"])
    card_gap = float(scene["card_gap"])
    status_a = str(scene["status_a"])
    status_b = str(scene["status_b"])
    color_a = str(scene["color_a"])
    color_b = str(scene["color_b"])
    route_side_a = int(scene["route_side_a"])
    route_side_b = int(scene["route_side_b"])
    clearance_a = int(scene["clearance_a"])
    clearance_b = int(scene["clearance_b"])
    stroke_width = int(scene["stroke_width"])
    marker_radius = int(scene["marker_radius"])
    decor_seed = int(scene["decor_seed"])

    scale = 4
    image = Image.new("RGB", (width * scale, height * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_decoration(draw, decor_seed, scale)
    _draw_graph(
        draw,
        (origin_x, origin_y),
        status_a,
        color_a,
        route_side_a,
        clearance_a,
        stroke_width,
        marker_radius,
        scale,
    )
    _draw_graph(
        draw,
        (origin_x + card_gap, origin_y),
        status_b,
        color_b,
        route_side_b,
        clearance_b,
        stroke_width,
        marker_radius,
        scale,
    )
    return image.resize((width, height), Image.Resampling.LANCZOS).convert("RGB")


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the comparison label using only the two topology inputs."""

    return "same" if scene["status_a"] == scene["status_b"] else "different"


def margin(scene: dict[str, Any]) -> float:
    """Signed topological comparison margin, scaled by raster clearance."""

    clearance = float(min(scene["clearance_a"], scene["clearance_b"]))
    return clearance if analytic_gold(scene) == "same" else -clearance


def is_quarantined(scene: dict[str, Any]) -> bool:
    return min(int(scene["clearance_a"]), int(scene["clearance_b"])) < QUARANTINE_CLEARANCE


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No intentionally pixel-identical latent aliases are declared."""

    del scene
    return []
