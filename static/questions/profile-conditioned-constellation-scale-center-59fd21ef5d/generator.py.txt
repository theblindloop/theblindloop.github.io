"""Deterministic scale-constellation sampler, analytic arm, and RGB renderer."""

from __future__ import annotations

import copy
import itertools
import math
import random

from PIL import Image, ImageDraw

COLOR_NAMES = ("red", "blue", "green", "orange")
CANDIDATE_RGB = {
    "red": (211, 55, 48),
    "blue": (42, 104, 177),
    "green": (34, 139, 78),
    "orange": (225, 126, 35),
}


def _fit_residual(
    filled: list[list[int]] | list[tuple[float, float]],
    hollow: list[list[int]] | list[tuple[float, float]],
    center: tuple[float, float],
) -> tuple[float, float]:
    """Best no-rotation positive-scale RMS over all set correspondences."""
    cx, cy = center
    source = [(float(x) - cx, float(y) - cy) for x, y in filled]
    target = [(float(x) - cx, float(y) - cy) for x, y in hollow]
    denominator = sum(x * x + y * y for x, y in source)
    if len(source) != len(target) or not source or denominator <= 1e-9:
        return float("inf"), 0.0
    best_residual = float("inf")
    best_scale = 0.0
    for ordering in itertools.permutations(target):
        scale = sum(
            sx * tx + sy * ty
            for (sx, sy), (tx, ty) in zip(source, ordering, strict=True)
        ) / denominator
        if not 0.05 <= scale <= 5.0:
            continue
        squared = sum(
            (tx - scale * sx) ** 2 + (ty - scale * sy) ** 2
            for (sx, sy), (tx, ty) in zip(source, ordering, strict=True)
        )
        residual = math.sqrt(squared / len(source))
        if residual < best_residual:
            best_residual, best_scale = residual, scale
    return best_residual, best_scale


def _score_by_color(scene: dict) -> dict[str, float]:
    filled = scene["filled_points"]
    hollow = scene["hollow_points"]
    candidates = scene["candidates"]
    return {
        candidate["color"]: _fit_residual(
            filled, hollow, (float(candidate["x"]), float(candidate["y"]))
        )[0]
        for candidate in candidates
    }


def _far_enough(point: tuple[int, int], others: list[tuple[int, int]], floor: float) -> bool:
    return all(math.hypot(point[0] - x, point[1] - y) >= floor for x, y in others)


def sample_scene(seed: int) -> dict:
    """Sample balanced answer identities and an interleaved radial scale transform."""
    rng = random.Random(int(seed))
    answer_color = COLOR_NAMES[int(seed) % len(COLOR_NAMES)]
    width, height = 640, 560

    for _attempt in range(1200):
        slots = [
            (rng.randint(190, 275), rng.randint(155, 245)),
            (rng.randint(365, 450), rng.randint(155, 245)),
            (rng.randint(190, 275), rng.randint(315, 405)),
            (rng.randint(365, 450), rng.randint(315, 405)),
        ]
        rng.shuffle(slots)
        colors = list(COLOR_NAMES)
        rng.shuffle(colors)
        candidates = [
            {"color": color, "x": point[0], "y": point[1]}
            for color, point in zip(colors, slots, strict=True)
        ]
        center_item = next(item for item in candidates if item["color"] == answer_color)
        cx, cy = int(center_item["x"]), int(center_item["y"])
        scale = rng.uniform(1.52, 1.82)

        angles: list[float] = []
        filled_points: list[list[int]] = []
        hollow_points: list[list[int]] = []
        occupied: list[tuple[int, int]] = [(item["x"], item["y"]) for item in candidates]
        for _ in range(8000):
            if len(filled_points) == 6:
                break
            angle = rng.uniform(0.0, 2.0 * math.pi)
            if any(abs(math.atan2(math.sin(angle - old), math.cos(angle - old))) < 0.42 for old in angles):
                continue
            radius = rng.uniform(62.0, 112.0)
            fx = int(round(cx + radius * math.cos(angle)))
            fy = int(round(cy + radius * math.sin(angle)))
            hx = int(round(cx + scale * radius * math.cos(angle)))
            hy = int(round(cy + scale * radius * math.sin(angle)))
            filled = (fx, fy)
            hollow = (hx, hy)
            if not (35 <= fx <= width - 36 and 35 <= fy <= height - 36):
                continue
            if not (30 <= hx <= width - 31 and 30 <= hy <= height - 31):
                continue
            if not _far_enough(filled, occupied, 27.0):
                continue
            if not _far_enough(hollow, occupied, 29.0):
                continue
            if math.hypot(fx - hx, fy - hy) < 31.0:
                continue
            filled_points.append([fx, fy])
            hollow_points.append([hx, hy])
            occupied.extend((filled, hollow))
            angles.append(angle)
        if len(filled_points) != 6:
            continue

        rng.shuffle(filled_points)
        rng.shuffle(hollow_points)
        rng.shuffle(candidates)
        scene = {
            "width": width,
            "height": height,
            "background": [248, 247, 242],
            "dark_color": [27, 32, 43],
            "frame_color": [178, 180, 178],
            "filled_radius": 6,
            "hollow_radius": 11,
            "hollow_width": 3,
            "candidate_radius": 9,
            "candidate_width": 4,
            "filled_points": filled_points,
            "hollow_points": hollow_points,
            "candidates": candidates,
        }
        scores = _score_by_color(scene)
        ordered = sorted(scores, key=lambda color: (scores[color], color))
        if ordered[0] == answer_color and scores[ordered[0]] <= 1.2 and (
            scores[ordered[1]] - scores[ordered[0]] >= 12.0
        ):
            return scene
    raise RuntimeError("could not sample an unambiguous scale-center constellation")


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(scene["background"])
    dark = tuple(scene["dark_color"])
    frame = tuple(scene["frame_color"])
    filled_radius = int(scene["filled_radius"])
    hollow_radius = int(scene["hollow_radius"])
    hollow_width = int(scene["hollow_width"])
    candidate_radius = int(scene["candidate_radius"])
    candidate_width = int(scene["candidate_width"])
    filled_points = scene["filled_points"]
    hollow_points = scene["hollow_points"]
    candidates = scene["candidates"]

    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((18, 18, width - 19, height - 19), radius=12, outline=frame, width=2)
    for x, y in filled_points:
        draw.ellipse((x - filled_radius, y - filled_radius, x + filled_radius, y + filled_radius), fill=dark)
    for x, y in hollow_points:
        draw.ellipse(
            (x - hollow_radius, y - hollow_radius, x + hollow_radius, y + hollow_radius),
            outline=dark,
            width=hollow_width,
        )
    for candidate in candidates:
        x, y = int(candidate["x"]), int(candidate["y"])
        color = CANDIDATE_RGB[candidate["color"]]
        draw.ellipse(
            (x - candidate_radius, y - candidate_radius, x + candidate_radius, y + candidate_radius),
            outline=color,
            width=candidate_width,
        )
        draw.ellipse((x - 2, y - 2, x + 2, y + 2), fill=color)
    return image


def analytic_gold(scene: dict) -> str:
    scores = _score_by_color(scene)
    return min(scores, key=lambda color: (scores[color], color))


def margin(scene: dict) -> float:
    ordered = sorted(_score_by_color(scene).values())
    return float(ordered[1] - ordered[0])


def is_quarantined(scene: dict) -> bool:
    ordered = sorted(_score_by_color(scene).values())
    return ordered[0] > 2.5 or ordered[1] - ordered[0] < 10.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["filled_points"] = list(reversed(twin["filled_points"]))
    twin["hollow_points"] = list(reversed(twin["hollow_points"]))
    return [("reverse_unordered_mark_storage", twin)]
