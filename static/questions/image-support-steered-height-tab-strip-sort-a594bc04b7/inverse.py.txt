"""Independent image-only inverse for the declared height-tab strip sort."""

from __future__ import annotations

import numpy as np
from PIL import Image

_SYMBOLS = ("Z", "I", "H", "U")
_CORE = {
    "Z": ("11111", "00010", "00100", "01000", "11111"),
    "I": ("11111", "00100", "00100", "00100", "11111"),
    "H": ("10001", "10001", "11111", "10001", "10001"),
    "U": ("10001", "10001", "10001", "10001", "11111"),
}


def _template(symbol: str) -> list[list[int]]:
    out = [[0] * 7 for _ in range(7)]
    for r, row in enumerate(_CORE[symbol], start=1):
        for c, value in enumerate(row, start=1):
            out[r][c] = int(value)
    return out


def _badge_bbox(arr: np.ndarray) -> tuple[int, int, int, int]:
    r, g, b = arr[:, :, 0], arr[:, :, 1], arr[:, :, 2]
    violet = (r >= 85) & (r <= 150) & (g >= 50) & (g <= 95) & (b >= 135) & (b <= 185)
    ys, xs = np.nonzero(violet)
    if len(xs) < 900:
        raise ValueError("missing or damaged violet badge")
    x0, x1, y0, y1 = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    if not (90 <= x1 - x0 + 1 <= 92 and 90 <= y1 - y0 + 1 <= 92):
        raise ValueError("ambiguous violet badge bounds")
    return x0, y0, x1, y1


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.shape[:2] != (512, 512):
        raise ValueError("unexpected canvas")
    x, y, _, _ = _badge_bbox(arr)
    ranked_columns: list[tuple[int, list[int]]] = []
    for slot in range(7):
        sx = x + 7 + 11 * slot
        # Reject erased or badly recolored support anywhere in the data strip.
        strip = arr[y + 38 : y + 86, sx : sx + 10]
        if strip.shape != (48, 10, 3):
            raise ValueError("truncated strip")
        background_like = (
            (strip[:, :, 0] >= 215) & (strip[:, :, 0] <= 245)
            & (strip[:, :, 1] >= 210) & (strip[:, :, 1] <= 240)
            & (strip[:, :, 2] >= 225) & (strip[:, :, 2] <= 248)
        )
        dark_like = (strip[:, :, 0] < 70) & (strip[:, :, 1] < 70) & (strip[:, :, 2] < 85)
        outline_like = (
            (strip[:, :, 0] >= 165) & (strip[:, :, 0] <= 200)
            & (strip[:, :, 1] >= 155) & (strip[:, :, 1] <= 195)
            & (strip[:, :, 2] >= 175) & (strip[:, :, 2] <= 210)
        )
        if float((background_like | dark_like | outline_like).mean()) < 0.92:
            raise ValueError("damaged strip support")

        # Count amber rows in the fixed header and rank them by visible height.
        header = arr[y + 5 : y + 33, sx + 2 : sx + 9]
        amber = (
            (header[:, :, 0] > 190) & (header[:, :, 1] > 110)
            & (header[:, :, 1] < 190) & (header[:, :, 2] < 80)
        )
        row_fill = amber.mean(axis=1)
        height = int(np.count_nonzero(row_fill > 0.80))
        if height < 3:
            raise ValueError("missing height tab")

        column = []
        for row in range(7):
            patch = arr[y + 38 + 7 * row : y + 44 + 7 * row, sx : sx + 10]
            dark = (patch[:, :, 0] < 70) & (patch[:, :, 1] < 70) & (patch[:, :, 2] < 85)
            fraction = float(dark.mean())
            if fraction > 0.25:
                column.append(1)
            elif fraction < 0.04:
                column.append(0)
            else:
                raise ValueError("ambiguous raster cell")
        ranked_columns.append((height, column))

    ranked_columns.sort(key=lambda item: item[0])
    heights = [height for height, _ in ranked_columns]
    if any(b - a < 3 for a, b in zip(heights, heights[1:])):
        raise ValueError("height-tab ordering is ambiguous")
    bits = [[ranked_columns[col][1][row] for col in range(7)] for row in range(7)]
    scores = []
    for symbol in _SYMBOLS:
        template = _template(symbol)
        distance = sum(bits[r][c] != template[r][c] for r in range(7) for c in range(7))
        scores.append((distance, symbol))
    scores.sort()
    if scores[1][0] - scores[0][0] < 2:
        raise ValueError("ambiguous reconstructed glyph")
    return scores[0][1]
