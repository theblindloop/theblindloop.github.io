"""Deterministic latent sampler and renderer for height-tab strip sorting."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 512
SYMBOLS = ("Z", "I", "H", "U")
QUESTION = (
    "Inside the violet badge are seven vertical strips. Reorder whole strips from "
    "the shortest amber height tab to the tallest, keeping every strip upright. "
    "Which glyph appears: Z, I, H, or U?"
)

# Each public 5x5 glyph has exactly 13 dark cells. It is centered in a 7x7 field.
_CORE = {
    "Z": ("11111", "00010", "00100", "01000", "11111"),
    "I": ("11111", "00100", "00100", "00100", "11111"),
    "H": ("10001", "10001", "11111", "10001", "10001"),
    "U": ("10001", "10001", "10001", "10001", "11111"),
}

_PALETTES = (
    {"frame": (112, 71, 159), "panel": (231, 221, 239), "tab": (226, 155, 34)},
    {"frame": (132, 77, 166), "panel": (225, 232, 239), "tab": (210, 138, 28)},
    {"frame": (98, 75, 151), "panel": (235, 224, 232), "tab": (234, 166, 45)},
)
_INK = (27, 34, 48)
_BG = (249, 248, 244)


def _template(symbol: str) -> list[list[int]]:
    out = [[0 for _ in range(7)] for _ in range(7)]
    for r, row in enumerate(_CORE[symbol], start=1):
        for c, value in enumerate(row, start=1):
            out[r][c] = int(value)
    return out


def _dist(a: Iterable[Iterable[int]], b: Iterable[Iterable[int]]) -> int:
    return sum(int(x != y) for ar, br in zip(a, b) for x, y in zip(ar, br))


def _scores(bits: list[list[int]]) -> list[tuple[int, str]]:
    return sorted((_dist(bits, _template(symbol)), symbol) for symbol in SYMBOLS)


def _permuted(bits: list[list[int]], permutation: list[int]) -> list[list[int]]:
    return [[row[source] for source in permutation] for row in bits]


def sample_scene(seed: int) -> dict:
    seed = int(seed)
    rng = random.Random(seed)
    symbol = SYMBOLS[seed % len(SYMBOLS)]
    base = _template(symbol)
    swap_count = (seed // len(SYMBOLS)) % 3
    bits = [row[:] for row in base]
    for _ in range(200):
        trial = [row[:] for row in base]
        on = [(r, c) for r in range(7) for c in range(7) if trial[r][c]]
        off = [(r, c) for r in range(7) for c in range(7) if not trial[r][c]]
        rng.shuffle(on)
        rng.shuffle(off)
        for r, c in on[:swap_count]:
            trial[r][c] = 0
        for r, c in off[:swap_count]:
            trial[r][c] = 1
        scores = _scores(trial)
        if scores[0][1] == symbol and scores[1][0] - scores[0][0] >= 2:
            bits = trial
            break

    # Require a genuine, visually disruptive shuffle: reading current columns
    # without the declared height-tab sort cannot confidently reveal the gold.
    permutation = list(range(7))
    for _ in range(500):
        rng.shuffle(permutation)
        visible_scores = _scores(_permuted(bits, permutation))
        if permutation != list(range(7)) and not (
            visible_scores[0][1] == symbol
            and visible_scores[1][0] - visible_scores[0][0] >= 2
        ):
            break
    return {
        "bits": bits,
        "permutation": permutation,
        "badge_x": rng.randint(42, 374),
        "badge_y": rng.randint(34, 386),
        "palette_variant": rng.randrange(len(_PALETTES)),
    }


def render(scene: dict) -> Image.Image:
    bits = [[int(v) for v in row] for row in scene["bits"]]
    permutation = [int(v) for v in scene["permutation"]]
    x, y = int(scene["badge_x"]), int(scene["badge_y"])
    palette = _PALETTES[int(scene["palette_variant"]) % len(_PALETTES)]
    image = Image.new("RGB", (WIDTH, HEIGHT), _BG)
    draw = ImageDraw.Draw(image)

    # One 92x92 connected, area-like badge. All answer-bearing evidence and
    # ordering fiducials live inside it; absolute placement is nuisance only.
    draw.rounded_rectangle((x, y, x + 91, y + 91), radius=7, fill=palette["frame"])
    draw.rectangle((x + 4, y + 4, x + 87, y + 87), fill=palette["panel"])
    for slot, source_col in enumerate(permutation):
        sx = x + 7 + 11 * slot
        # A bottom-aligned unary height tab identifies the source column rank.
        tab_height = 4 + 4 * source_col
        draw.rectangle((sx + 2, y + 33 - tab_height, sx + 8, y + 32), fill=palette["tab"])
        # Upright seven-cell raster strip. Cell interiors remain panel-colored.
        for row in range(7):
            cy = y + 38 + 7 * row
            draw.rectangle((sx, cy, sx + 9, cy + 5), outline=(181, 174, 190), width=1)
            if bits[row][source_col]:
                draw.rectangle((sx + 2, cy + 1, sx + 8, cy + 4), fill=_INK)
    return image


def analytic_gold(scene: dict) -> str:
    scores = _scores([[int(v) for v in row] for row in scene["bits"]])
    if scores[0][0] == scores[1][0]:
        raise ValueError("ambiguous glyph")
    return scores[0][1]


def margin(scene: dict) -> float:
    scores = _scores([[int(v) for v in row] for row in scene["bits"]])
    return float(scores[1][0] - scores[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
