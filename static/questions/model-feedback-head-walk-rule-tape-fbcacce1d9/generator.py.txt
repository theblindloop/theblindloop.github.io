"""Head-walk rule tape: a machine whose own run decides which symbol it reads next.

The raster shows a rule table whose rows are labelled by state colors and whose
columns are labelled by black shapes.  Every cell holds one colored arrowhead:
its color is the next state and the side it points is the direction the reading
head then steps along the strip.  A black annulus marks the starting state and a
black bar under one strip cell marks where the head starts.

The strip is therefore not consumed left to right.  Which cell is read at step k
is produced by the run itself, cells are re-read from different states, and the
head reverses.  The reported decision is the state color after six steps.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

# ---------------------------------------------------------------- constants

WIDTH = 1080
HEIGHT = 980

WHITE = (255, 255, 255)
BLACK = (18, 18, 18)

STATES = ("amber", "teal", "violet", "crimson")
STATE_RGB = {
    "amber": (240, 170, 20),
    "teal": (20, 160, 160),
    "violet": (150, 60, 200),
    "crimson": (210, 40, 60),
}
SHAPES = ("square", "circle", "triangle")

DISK_R = 38
RING_INNER_R = 52
RING_OUTER_R = 66
ARROW_HALF = 40        # colored cell arrowhead half-extent
GLYPH_HALF = 30        # black silhouette half-extent
TRI_HALF_W = 34

HEADER_Y = 86
ROW_Y0 = 200
ROW_PITCH = 150
LABEL_X = 190
CELL_X0 = 430
CELL_PITCH = 200
TAPE_Y = 800
TAPE_X0 = 140
TAPE_PITCH = 104
TAPE_LEN = 9
BAR_Y = 866
BAR_H = 12
BAR_HALF_W = 30

STEPS = 6
MIN_MARGIN = 11.0


# ---------------------------------------------------------------- machine

def _run(table, start, tape, head, steps):
    """Return (final_state, visited_head_indices) or None if the head leaves."""
    cur = start
    pos = head
    visited = [pos]
    for _ in range(steps):
        if pos < 0 or pos >= len(tape):
            return None
        cell = table[cur][tape[pos]]
        cur = cell[0]
        pos = pos - 1 if cell[1] == "left" else pos + 1
        visited.append(pos)
    if pos < 0 or pos >= len(tape):
        return None
    return cur, visited


def _rightward(table, start, tape, head, steps):
    """Shortcut reading: ignore the arrow directions, always step right."""
    cur = start
    pos = head
    for _ in range(steps):
        cur = table[cur][tape[min(pos, len(tape) - 1)]][0]
        pos += 1
    return cur


def _left_to_right(table, start, tape, steps):
    """Shortcut reading: consume the drawn strip from its left end."""
    cur = start
    for k in range(steps):
        cur = table[cur][tape[k]][0]
    return cur


def _mode_cell_color(table):
    counts = {s: 0 for s in STATES}
    for row in STATES:
        for sym in SHAPES:
            counts[table[row][sym][0]] += 1
    best = max(counts.values())
    winners = [s for s in STATES if counts[s] == best]
    return winners[0] if len(winners) == 1 else None


# ---------------------------------------------------------------- sampling

def sample_scene(seed: int) -> dict:
    """Sample one latent scene.  Every field below is drawn in the raster."""
    rng = random.Random(seed * 7919 + 13)
    for _ in range(8000):
        table = {
            row: {
                sym: (rng.choice(STATES), rng.choice(("left", "right")))
                for sym in SHAPES
            }
            for row in STATES
        }
        dirs = [table[r][s][1] for r in STATES for s in SHAPES]
        if dirs.count("left") < 4 or dirs.count("right") < 4:
            continue
        # the successor color must depend on both the row and the symbol
        if any(len({table[r][s][0] for s in SHAPES}) < 2 for r in STATES):
            continue
        if any(len({table[r][s][0] for r in STATES}) < 3 for s in SHAPES):
            continue
        # the step direction must depend on both the row and the symbol
        if any(len({table[r][s][1] for s in SHAPES}) < 2 for r in STATES):
            continue
        if any(len({table[r][s][1] for r in STATES}) < 2 for s in SHAPES):
            continue

        start = rng.choice(STATES)
        tape = [rng.choice(SHAPES) for _ in range(TAPE_LEN)]
        if any(tape.count(s) < 2 for s in SHAPES):
            continue
        head = rng.randint(2, TAPE_LEN - 3)

        res = _run(table, start, tape, head, STEPS)
        if res is None:
            continue
        final, visited = res

        # the walk must genuinely wander: it reverses and re-reads cells
        steps_dir = [b - a for a, b in zip(visited, visited[1:])]
        reversals = sum(1 for a, b in zip(steps_dir, steps_dir[1:]) if a != b)
        if reversals < 2:
            continue
        read = visited[:-1]
        if len(set(read)) > STEPS - 2:      # at least two cells read twice
            continue
        states = [start]
        cur = start
        for pos in read:
            cur = table[cur][tape[pos]][0]
            states.append(cur)
        if len(set(states)) < 3:
            continue

        # every cheap reading must land on another color
        if final == start:
            continue
        if final == _rightward(table, start, tape, head, STEPS):
            continue
        if final == _left_to_right(table, start, tape, STEPS):
            continue
        if final == table[start][tape[head]][0]:
            continue
        short = _run(table, start, tape, head, STEPS - 1)
        if short is None or final == short[0]:
            continue
        longer = _run(table, start, tape, head, STEPS + 1)
        if longer is not None and final == longer[0]:
            continue
        mode = _mode_cell_color(table)
        if mode is not None and final == mode:
            continue

        rows = list(STATES)
        rng.shuffle(rows)
        cols = list(SHAPES)
        rng.shuffle(cols)
        jit = {
            "rows": [rng.randint(-4, 4) for _ in range(4)],
            "cols": [rng.randint(-4, 4) for _ in range(3)],
            "tape": [rng.randint(-3, 3) for _ in range(TAPE_LEN)],
            "header": rng.randint(-3, 3),
        }
        return {
            "seed": seed,
            "rows": rows,
            "cols": cols,
            "table": table,
            "start": start,
            "tape": tape,
            "head": head,
            "jitter": jit,
        }
    raise RuntimeError("rejection sampling failed for seed %r" % (seed,))


# ---------------------------------------------------------------- geometry

def _row_y(scene, i):
    return ROW_Y0 + ROW_PITCH * i + scene["jitter"]["rows"][i]


def _col_x(scene, j):
    return CELL_X0 + CELL_PITCH * j + scene["jitter"]["cols"][j]


def _tape_x(scene, k):
    return TAPE_X0 + TAPE_PITCH * k + scene["jitter"]["tape"][k]


# ---------------------------------------------------------------- drawing

def _disk(draw, cx, cy, r, rgb):
    draw.ellipse([cx - r, cy - r, cx + r, cy + r], fill=rgb)


def _arrowhead(draw, cx, cy, side, rgb):
    if side == "right":
        pts = [(cx - ARROW_HALF, cy - ARROW_HALF),
               (cx - ARROW_HALF, cy + ARROW_HALF),
               (cx + ARROW_HALF, cy)]
    else:
        pts = [(cx + ARROW_HALF, cy - ARROW_HALF),
               (cx + ARROW_HALF, cy + ARROW_HALF),
               (cx - ARROW_HALF, cy)]
    draw.polygon(pts, fill=rgb)


def _glyph(draw, cx, cy, shape):
    if shape == "square":
        draw.rectangle(
            [cx - GLYPH_HALF, cy - GLYPH_HALF, cx + GLYPH_HALF, cy + GLYPH_HALF],
            fill=BLACK,
        )
    elif shape == "circle":
        draw.ellipse(
            [cx - GLYPH_HALF, cy - GLYPH_HALF, cx + GLYPH_HALF, cy + GLYPH_HALF],
            fill=BLACK,
        )
    elif shape == "triangle":
        draw.polygon(
            [(cx - TRI_HALF_W, cy + GLYPH_HALF),
             (cx + TRI_HALF_W, cy + GLYPH_HALF),
             (cx, cy - GLYPH_HALF)],
            fill=BLACK,
        )
    else:  # pragma: no cover - defensive
        raise ValueError(shape)


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), WHITE)
    draw = ImageDraw.Draw(img)

    hy = HEADER_Y + scene["jitter"]["header"]
    for j, shape in enumerate(scene["cols"]):
        _glyph(draw, _col_x(scene, j), hy, shape)

    for i, row_state in enumerate(scene["rows"]):
        cy = _row_y(scene, i)
        if row_state == scene["start"]:
            _disk(draw, LABEL_X, cy, RING_OUTER_R, BLACK)
            _disk(draw, LABEL_X, cy, RING_INNER_R, WHITE)
        _disk(draw, LABEL_X, cy, DISK_R, STATE_RGB[row_state])
        for j, shape in enumerate(scene["cols"]):
            nxt, side = scene["table"][row_state][shape]
            _arrowhead(draw, _col_x(scene, j), cy, side, STATE_RGB[nxt])

    for k, shape in enumerate(scene["tape"]):
        _glyph(draw, _tape_x(scene, k), TAPE_Y, shape)

    bx = _tape_x(scene, scene["head"])
    draw.rectangle(
        [bx - BAR_HALF_W, BAR_Y, bx + BAR_HALF_W, BAR_Y + BAR_H], fill=BLACK
    )
    return img


# ---------------------------------------------------------------- contract

def analytic_gold(scene: dict) -> str:
    res = _run(scene["table"], scene["start"], scene["tape"], scene["head"], STEPS)
    if res is None:  # pragma: no cover - sampling forbids this
        raise ValueError("head left the strip")
    return res[0]


def margin(scene: dict) -> float:
    gaps = [
        float(RING_OUTER_R - RING_INNER_R),
        float(RING_INNER_R - DISK_R),
        float(BAR_H),
    ]
    rows = [_row_y(scene, i) for i in range(4)]
    for a, b in zip(rows, rows[1:]):
        gaps.append(float(b - a - 2 * ARROW_HALF))
        gaps.append(float(b - a - RING_OUTER_R - ARROW_HALF))
    cols = [_col_x(scene, j) for j in range(3)]
    for a, b in zip(cols, cols[1:]):
        gaps.append(float(b - a - 2 * ARROW_HALF))
        gaps.append(float(b - a - 2 * TRI_HALF_W))
    gaps.append(float(cols[0] - LABEL_X - RING_OUTER_R - ARROW_HALF))
    taps = [_tape_x(scene, k) for k in range(TAPE_LEN)]
    for a, b in zip(taps, taps[1:]):
        gaps.append(float(b - a - 2 * TRI_HALF_W))
        gaps.append(float(b - a - 2 * BAR_HALF_W))
    hy = HEADER_Y + scene["jitter"]["header"]
    gaps.append(float(rows[0] - ARROW_HALF - hy - GLYPH_HALF))
    gaps.append(float(TAPE_Y - GLYPH_HALF - rows[-1] - RING_OUTER_R))
    gaps.append(float(BAR_Y - TAPE_Y - GLYPH_HALF))
    gaps.append(float(TAPE_X0 - TRI_HALF_W))
    gaps.append(float(WIDTH - taps[-1] - TRI_HALF_W))
    gaps.append(float(HEIGHT - BAR_Y - BAR_H))
    gaps.append(float(2 * ARROW_HALF - 20))
    return float(min(gaps))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < MIN_MARGIN


def latent_symmetries(scene: dict) -> list:
    """Rendering is injective over sampled scenes; no nontrivial alias exists."""
    return []
