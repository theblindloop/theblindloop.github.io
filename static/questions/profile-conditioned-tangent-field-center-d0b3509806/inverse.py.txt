"""Independent PNG-only inverse arm for the tangent orientation field."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    groups: list[np.ndarray] = []
    for start_y, start_x in np.argwhere(mask):
        if seen[start_y, start_x]:
            continue
        stack = [(int(start_y), int(start_x))]
        seen[start_y, start_x] = True
        pixels = []
        while stack:
            y, x = stack.pop()
            pixels.append((y, x))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    yy, xx = y + dy, x + dx
                    if 0 <= yy < height and 0 <= xx < width and mask[yy, xx] and not seen[yy, xx]:
                        seen[yy, xx] = True
                        stack.append((yy, xx))
        if len(pixels) >= 25:
            groups.append(np.asarray(pixels, dtype=float))
    return groups


def _marker_center(array: np.ndarray, color: str) -> np.ndarray:
    red = array[:, :, 0]
    green = array[:, :, 1]
    blue = array[:, :, 2]
    if color == "red":
        mask = (red > 155) & (red > green * 1.65) & (red > blue * 1.45)
    else:
        mask = (blue > 135) & (blue > red * 1.65) & (blue > green * 1.25)
    coords = np.argwhere(mask)
    if len(coords) < 40:
        raise ValueError(f"missing {color} candidate marker")
    return coords[:, ::-1].mean(axis=0)


def _dash_axes(array: np.ndarray) -> tuple[list[np.ndarray], list[np.ndarray]]:
    dark = np.max(array, axis=2) < 105
    groups = _components(dark)
    centers: list[np.ndarray] = []
    axes: list[np.ndarray] = []
    for group_yx in groups:
        xy = group_yx[:, ::-1]
        center = xy.mean(axis=0)
        covariance = np.cov((xy - center).T)
        values, vectors = np.linalg.eigh(covariance)
        if values[1] < 4.0 or values[1] / max(values[0], 1e-6) < 3.0:
            continue
        centers.append(center)
        axes.append(vectors[:, 1])
    if len(centers) < 24:
        raise ValueError("too few separated black dashes")
    return centers, axes


def _score(point: np.ndarray, centers: list[np.ndarray], axes: list[np.ndarray]) -> float:
    residuals = []
    for center, axis in zip(centers, axes):
        radial = center - point
        radius = max(float(np.linalg.norm(radial)), 1e-9)
        residuals.append(abs(float(np.dot(axis, radial))) / radius)
    residuals.sort()
    trim = max(1, len(residuals) // 10)
    core = residuals[trim:-trim] if len(residuals) > 2 * trim else residuals
    return float(np.mean(core))


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=float)
    red_center = _marker_center(array, "red")
    blue_center = _marker_center(array, "blue")
    centers, axes = _dash_axes(array)
    red_score = _score(red_center, centers, axes)
    blue_score = _score(blue_center, centers, axes)
    if abs(red_score - blue_score) < 0.05:
        raise ValueError("ambiguous tangent-field center")
    return "red" if red_score < blue_score else "blue"
