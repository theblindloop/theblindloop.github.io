"""Deterministic scene sampler and raster renderer for tangent_field_center."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

CANVAS = 640
SCALE = 3
RED = (220, 55, 62)
BLUE = (42, 105, 210)
INK = (28, 31, 36)
PAPER = (248, 246, 239)
RING = (255, 255, 255)
QUARANTINE_MARGIN = 0.12


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    center_x = 320.0 + rng.uniform(-22.0, 22.0)
    center_y = 320.0 + rng.uniform(-22.0, 22.0)
    separation = rng.uniform(185.0, 235.0)
    axis = rng.uniform(0.0, math.tau)
    offset_x = 0.5 * separation * math.cos(axis)
    offset_y = 0.5 * separation * math.sin(axis)
    points = [
        [center_x - offset_x, center_y - offset_y],
        [center_x + offset_x, center_y + offset_y],
    ]
    colors = ["red", "blue"]
    target_index = int(seed) & 1
    target = points[target_index]

    dash_centers: list[list[float]] = []
    dash_lengths: list[float] = []
    dash_angles: list[float] = []
    for row in range(6):
        for column in range(6):
            x = 82.0 + 95.0 * column + rng.uniform(-15.0, 15.0)
            y = 82.0 + 95.0 * row + rng.uniform(-15.0, 15.0)
            if min(math.hypot(x - px, y - py) for px, py in points) < 42.0:
                continue
            radial_angle = math.atan2(y - target[1], x - target[0])
            jitter = rng.gauss(0.0, 0.027 + 0.004 * (int(seed) % 4))
            dash_centers.append([x, y])
            dash_lengths.append(rng.uniform(26.0, 36.0))
            dash_angles.append(radial_angle + math.pi / 2.0 + jitter)

    return {
        "canvas": [CANVAS, CANVAS],
        "candidate_points": points,
        "candidate_colors": colors,
        "dash_centers": dash_centers,
        "dash_lengths": dash_lengths,
        "dash_angles": dash_angles,
        "draw_order": list(range(len(dash_centers))),
    }


def _axis_scores(scene: dict) -> list[float]:
    points = scene["candidate_points"]
    centers = scene["dash_centers"]
    angles = scene["dash_angles"]
    scores: list[float] = []
    for px, py in points:
        residuals = []
        for (x, y), angle in zip(centers, angles):
            radial_x = x - px
            radial_y = y - py
            radius = max(math.hypot(radial_x, radial_y), 1e-9)
            axis_x = math.cos(angle)
            axis_y = math.sin(angle)
            residuals.append(abs((axis_x * radial_x + axis_y * radial_y) / radius))
        residuals.sort()
        trim = max(1, len(residuals) // 10)
        core = residuals[trim:-trim] if len(residuals) > 2 * trim else residuals
        scores.append(sum(core) / len(core))
    return scores


def analytic_gold(scene: dict) -> str:
    scores = _axis_scores(scene)
    colors = scene["candidate_colors"]
    return str(colors[0] if scores[0] < scores[1] else colors[1])


def margin(scene: dict) -> float:
    scores = _axis_scores(scene)
    return float(abs(scores[0] - scores[1]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["draw_order"] = list(reversed(twin["draw_order"]))
    return [("reverse_disjoint_dash_painter_order", twin)]


def render(scene: dict) -> Image.Image:
    width, height = (int(value) for value in scene["canvas"])
    points = scene["candidate_points"]
    colors = scene["candidate_colors"]
    centers = scene["dash_centers"]
    lengths = scene["dash_lengths"]
    angles = scene["dash_angles"]
    order = scene["draw_order"]

    image = Image.new("RGB", (width * SCALE, height * SCALE), PAPER)
    draw = ImageDraw.Draw(image)
    for index in order:
        x, y = centers[index]
        half = lengths[index] / 2.0
        dx = half * math.cos(angles[index])
        dy = half * math.sin(angles[index])
        draw.line(
            ((x - dx) * SCALE, (y - dy) * SCALE, (x + dx) * SCALE, (y + dy) * SCALE),
            fill=INK,
            width=7 * SCALE,
        )

    palette = {"red": RED, "blue": BLUE}
    for (x, y), color in zip(points, colors):
        outer = 15.0 * SCALE
        inner = 10.5 * SCALE
        cx, cy = x * SCALE, y * SCALE
        draw.ellipse((cx - outer, cy - outer, cx + outer, cy + outer), fill=RING)
        draw.ellipse((cx - inner, cy - inner, cx + inner, cy + inner), fill=palette[color])

    resampling = getattr(Image, "Resampling", Image)
    return image.resize((width, height), resampling.LANCZOS)
