"""Sample and render a dense plaque of color-partition correspondences."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw

WIDTH = 720
HEIGHT = 720
PANEL = 338
PANEL_POSITIONS = ((14, 14), (368, 14), (14, 368), (368, 368))
FRAME = 12
GRID_OFFSET = 22
CELL = 42
TILE = 38
ROWS = 7
COLS = 7

BACKGROUND = (248, 246, 240)
PLAQUE = (31, 38, 48)
REFERENCE_FRAME = (103, 111, 119)
REFERENCE_COLORS = (
    (42, 91, 158),
    (191, 63, 75),
    (49, 132, 91),
    (218, 174, 49),
)
CANDIDATE_COLORS = (
    (112, 73, 166),
    (194, 88, 36),
    (30, 142, 151),
    (202, 86, 139),
)
TAG_COLORS = {
    "amber": (246, 145, 35),
    "cyan": (26, 188, 212),
    "violet": (143, 77, 190),
}
TAG_NAMES = ("amber", "cyan", "violet")


def _flat(matrix: list[list[int]]) -> list[int]:
    return [value for row in matrix for value in row]


def _matrix(values: list[int]) -> list[list[int]]:
    if len(values) != ROWS * COLS:
        raise ValueError("a mosaic must contain exactly 49 cells")
    return [values[start : start + COLS] for start in range(0, len(values), COLS)]


def _best_relabel_distance(reference: list[list[int]], candidate: list[list[int]]) -> int:
    left = _flat(reference)
    right = _flat(candidate)
    if len(left) != len(right):
        return ROWS * COLS
    best = len(left)
    for permutation in itertools.permutations(range(4)):
        distance = sum(a != permutation[b] for a, b in zip(left, right, strict=True))
        best = min(best, distance)
    return best


def _balanced_reference(rng: random.Random) -> list[list[int]]:
    counts = (13, 12, 12, 12)
    for _ in range(1000):
        values = [label for label, count in enumerate(counts) for _ in range(count)]
        rng.shuffle(values)
        matrix = _matrix(values)
        columns = [[matrix[row][col] for row in range(ROWS)] for col in range(COLS)]
        if all(len(set(row)) >= 3 for row in matrix) and all(
            len(set(column)) >= 3 for column in columns
        ):
            return matrix
    raise RuntimeError("could not sample a suitably mixed reference")


def _relabel(matrix: list[list[int]], mapping: list[int]) -> list[list[int]]:
    return [[mapping[value] for value in row] for row in matrix]


def _distractor(target: list[list[int]], rng: random.Random, minimum: int) -> list[list[int]]:
    base = _flat(target)
    for _ in range(4000):
        changed = base.copy()
        indices = list(range(len(base)))
        rng.shuffle(indices)
        swap_total = rng.choice((3, 4, 5))
        swaps: list[tuple[int, int]] = []
        while len(indices) >= 2 and len(swaps) < swap_total:
            first = indices.pop()
            choices = [offset for offset, other in enumerate(indices) if base[other] != base[first]]
            if not choices:
                continue
            second = indices.pop(choices[rng.randrange(len(choices))])
            swaps.append((first, second))
        if len(swaps) != swap_total:
            continue
        for first, second in swaps:
            changed[first], changed[second] = changed[second], changed[first]
        result = _matrix(changed)
        if _best_relabel_distance(target, result) >= minimum:
            return result
    raise RuntimeError("could not construct an equal-histogram partition distractor")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    reference = _balanced_reference(rng)
    mapping = list(range(4))
    rng.shuffle(mapping)
    target = _relabel(reference, mapping)
    wrong_a = _distractor(target, rng, 4)
    wrong_b = _distractor(target, rng, 6)
    while wrong_b == wrong_a:
        wrong_b = _distractor(target, rng, 6)

    candidate_tags = list(TAG_NAMES)
    rng.shuffle(candidate_tags)
    desired_tag = TAG_NAMES[int(seed) % len(TAG_NAMES)]
    gold_index = candidate_tags.index(desired_tag)
    candidates = [wrong_a, wrong_b]
    candidates.insert(gold_index, target)
    return {
        "reference": reference,
        "candidates": candidates,
        "candidate_tags": candidate_tags,
    }


def _distances(scene: dict) -> list[int]:
    return [
        _best_relabel_distance(scene["reference"], candidate)
        for candidate in scene["candidates"]
    ]


def analytic_gold(scene: dict) -> str:
    distances = _distances(scene)
    matches = [index for index, distance in enumerate(distances) if distance == 0]
    if len(matches) != 1:
        raise ValueError(f"expected one partition-isomorphic candidate, found {len(matches)}")
    return scene["candidate_tags"][matches[0]]


def margin(scene: dict) -> float:
    distances = sorted(_distances(scene))
    if len(distances) != 3 or distances[0] != 0:
        return 0.0
    return float(distances[1])


def is_quarantined(scene: dict) -> bool:
    try:
        reference = scene["reference"]
        candidates = scene["candidates"]
        tags = scene["candidate_tags"]
        expected_histogram = [12, 12, 12, 13]
        valid = (
            len(reference) == ROWS
            and all(len(row) == COLS for row in reference)
            and sorted(_flat(reference)).count(0) >= 0
            and sorted(_flat(reference)) == sorted([0] * 13 + [1] * 12 + [2] * 12 + [3] * 12)
            and len(candidates) == 3
            and all(len(item) == ROWS and all(len(row) == COLS for row in item) for item in candidates)
            and all(sorted(_flat(item)).count(value) in expected_histogram for item in candidates for value in range(4))
            and all(sorted([_flat(item).count(value) for value in range(4)]) == expected_histogram for item in candidates)
            and sorted(tags) == sorted(TAG_NAMES)
        )
        if not valid:
            return True
        distances = _distances(scene)
        return distances.count(0) != 1 or margin(scene) < 4.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    position: tuple[int, int],
    matrix: list[list[int]],
    frame_color: tuple[int, int, int],
    palette: tuple[tuple[int, int, int], ...],
) -> None:
    left, top = position
    draw.rectangle((left, top, left + PANEL - 1, top + PANEL - 1), fill=frame_color)
    draw.rectangle(
        (left + FRAME, top + FRAME, left + PANEL - FRAME - 1, top + PANEL - FRAME - 1),
        fill=BACKGROUND,
    )
    for row, values in enumerate(matrix):
        for col, value in enumerate(values):
            x0 = left + GRID_OFFSET + col * CELL
            y0 = top + GRID_OFFSET + row * CELL
            draw.rectangle((x0, y0, x0 + TILE - 1, y0 + TILE - 1), fill=palette[value])


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rectangle((4, 4, WIDTH - 5, HEIGHT - 5), fill=PLAQUE)
    _draw_panel(
        draw,
        PANEL_POSITIONS[0],
        scene["reference"],
        REFERENCE_FRAME,
        REFERENCE_COLORS,
    )
    for index, (matrix, tag) in enumerate(
        zip(scene["candidates"], scene["candidate_tags"], strict=True), start=1
    ):
        _draw_panel(
            draw,
            PANEL_POSITIONS[index],
            matrix,
            TAG_COLORS[tag],
            CANDIDATE_COLORS,
        )
    return image
