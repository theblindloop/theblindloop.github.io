"""Deterministic renderer and analytic side of the ordered-entry world."""

from __future__ import annotations

import copy
import math
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont


LABELS = ("A", "B", "C", "D")
CANVAS = (1280, 420)
PANEL_X0 = 20
PANEL_STEP = 250
PANEL_TOP = 70
PANEL_W = 238
PANEL_H = 320
PANEL_ORDER = (1, 2, 3, 4, 5)

# The palette is deliberately sufficiently separated for a pixel-only inverse.
PALETTE = (
    (214, 70, 70),
    (45, 116, 193),
    (41, 151, 96),
    (145, 79, 174),
)
REGION_FILL = (224, 237, 247)
REGION_OUTLINE = (70, 119, 153)
BACKGROUND = (247, 249, 252)
PANEL_FILL = (255, 255, 255)
INK = (35, 44, 54)
ARROW = (100, 111, 122)

_FONT_PATHS = (
    "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
    "/usr/share/fonts/dejavu/DejaVuSans.ttf",
)


def _font(size: int):
    for candidate in _FONT_PATHS:
        if Path(candidate).is_file():
            return ImageFont.truetype(candidate, size=size)
    return ImageFont.load_default()


def _panel_left(index: int) -> int:
    return PANEL_X0 + index * PANEL_STEP


def _centered_text(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font, fill):
    box = draw.textbbox((0, 0), text, font=font)
    w = box[2] - box[0]
    h = box[3] - box[1]
    draw.text((xy[0] - w / 2, xy[1] - h / 2 - box[1]), text, font=font, fill=fill)


def _entry_order(state_table: dict[str, list[int]]) -> list[tuple[int, str]]:
    entries: list[tuple[int, str]] = []
    for label in LABELS:
        states = state_table[label]
        first = None
        for index in range(1, len(states)):
            if not states[index - 1] and states[index]:
                first = index
                break
        if first is None:
            raise ValueError(f"token {label} has no outside-to-inside transition")
        entries.append((first, label))
    return entries


def sample_scene(seed: int) -> dict:
    """Sample one balanced, deterministic five-panel scene."""

    event_rng = random.Random(seed * 1000003 + 17)
    style_rng = random.Random(seed * 1000033 + 31)
    motion_rng = random.Random(seed * 1000211 + 47)

    event_order = event_rng.sample(list(LABELS), len(LABELS))
    first_panel = {label: 2 + index for index, label in enumerate(event_order)}
    state_table: dict[str, list[int]] = {}
    for label in LABELS:
        states = [0]
        for panel_number in range(2, 6):
            if panel_number < first_panel[label]:
                states.append(0)
            elif panel_number == first_panel[label]:
                states.append(1)
            else:
                # Later exits/re-entries make the first transition, not final
                # membership, the quantity that must be recovered.
                states.append(1 if event_rng.random() < 0.53 else 0)
        state_table[label] = states

    color_indices = style_rng.sample(list(range(4)), 4)
    radii = style_rng.sample([15, 17, 19, 21], 4)
    tokens = [
        {"label": label, "color_index": color_indices[index], "radius": radii[index]}
        for index, label in enumerate(LABELS)
    ]

    region_center = [119 + style_rng.randint(-7, 7), 190 + style_rng.randint(-6, 6)]
    region_radius = 78
    region = {"center": region_center, "radius": region_radius}

    inside_offsets = [(0, -41), (41, 0), (0, 41), (-41, 0)]
    outside_offsets = [(-96, -67), (96, -67), (-96, 67), (96, 67)]
    centers: dict[str, list[list[float]]] = {label: [] for label in LABELS}
    near_boundary = (seed % 4 == 0)
    near_label = event_order[-1]  # It is inside at panel five by construction.

    for panel_index in range(5):
        active = [label for label in LABELS if state_table[label][panel_index]]
        inactive = [label for label in LABELS if not state_table[label][panel_index]]
        inside_slots = list(range(4))
        outside_slots = list(range(4))
        motion_rng.shuffle(inside_slots)
        motion_rng.shuffle(outside_slots)

        panel_positions: dict[str, list[float]] = {}
        for position, label in enumerate(sorted(active)):
            slot = inside_slots[position]
            dx, dy = inside_offsets[slot]
            jitter_x = motion_rng.uniform(-3.0, 3.0)
            jitter_y = motion_rng.uniform(-3.0, 3.0)
            panel_positions[label] = [region_center[0] + dx + jitter_x, region_center[1] + dy + jitter_y]

        for position, label in enumerate(sorted(inactive)):
            slot = outside_slots[position]
            dx, dy = outside_offsets[slot]
            jitter_x = motion_rng.uniform(-4.0, 4.0)
            jitter_y = motion_rng.uniform(-4.0, 4.0)
            panel_positions[label] = [region_center[0] + dx + jitter_x, region_center[1] + dy + jitter_y]

        # A small, still recoverable boundary example is included on a fixed
        # subset of seeds. It is explicitly quarantined by is_quarantined().
        if near_boundary and panel_index == 4:
            panel_positions[near_label] = [region_center[0] + region_radius - 5.0, region_center[1]]

        for label in LABELS:
            centers[label].append(panel_positions[label])

    clearances = []
    for label in LABELS:
        for position in centers[label]:
            distance = math.hypot(position[0] - region_center[0], position[1] - region_center[1])
            clearances.append(abs(distance - region_radius))

    scene = {
        "canvas": list(CANVAS),
        "panel_count": 5,
        "panel_order": list(PANEL_ORDER),
        "region": region,
        "tokens": tokens,
        "state_table": state_table,
        "centers": centers,
        "clearance": round(min(clearances), 6),
    }
    # Validate the sampler's own contract before exposing the scene.
    entries = _entry_order(state_table)
    if sorted(panel for panel, _ in entries) != [1, 2, 3, 4]:
        raise AssertionError("sampler did not create four unique entry panels")
    return scene


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", tuple(scene["canvas"]), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(15)
    panel_font = _font(16)
    small_font = _font(12)
    token_font = _font(13)

    draw.text((20, 15), "FIRST ENTRY", font=title_font, fill=INK)
    draw.text((20, 38), "Follow each letter through the numbered panels", font=small_font, fill=ARROW)
    draw.text((1040, 22), "ORDER", font=small_font, fill=ARROW)

    # A visible legend makes the identity binding explicit even when palette
    # assignments are permuted independently of the event order.
    draw.text((220, 17), "token identity:", font=small_font, fill=INK)
    token_by_label = {token["label"]: token for token in scene["tokens"]}
    for index, label in enumerate(LABELS):
        x = 330 + index * 62
        color = PALETTE[token_by_label[label]["color_index"]]
        draw.ellipse((x - 9, 25, x + 9, 43), fill=color, outline=INK, width=1)
        _centered_text(draw, (x, 34), label, small_font, (255, 255, 255))

    for index, panel_number in enumerate(scene["panel_order"]):
        left = _panel_left(index)
        top = PANEL_TOP
        right = left + PANEL_W
        bottom = top + PANEL_H
        draw.rounded_rectangle((left, top, right, bottom), radius=8, fill=PANEL_FILL, outline=INK, width=2)
        draw.text((left + 10, top + 8), f"P{panel_number}", font=panel_font, fill=INK)
        if index < len(scene["panel_order"]) - 1:
            next_left = _panel_left(index + 1)
            y = top - 17
            draw.line((right + 4, y, next_left - 6, y), fill=ARROW, width=2)
            draw.polygon(((next_left - 6, y), (next_left - 13, y - 4), (next_left - 13, y + 4)), fill=ARROW)

        center = scene["region"]["center"]
        radius = scene["region"]["radius"]
        cx = left + center[0]
        cy = top + center[1]
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=REGION_FILL, outline=REGION_OUTLINE, width=2)
        _centered_text(draw, (cx, cy - radius + 13), "inside", small_font, REGION_OUTLINE)

        for label in sorted(scene["state_table"]):
            token = token_by_label[label]
            x, y = scene["centers"][label][index]
            r = token["radius"]
            color = PALETTE[token["color_index"]]
            draw.ellipse((left + x - r, top + y - r, left + x + r, top + y + r), fill=color, outline=INK, width=2)
            _centered_text(draw, (left + x, top + y), label, token_font, (255, 255, 255))

    return image


def analytic_gold(scene: dict) -> str:
    entries = _entry_order(scene["state_table"])
    first_panel = min(panel for panel, _ in entries)
    winners = [label for panel, label in entries if panel == first_panel]
    if len(winners) != 1:
        raise ValueError("ambiguous first entry")
    return winners[0]


def margin(scene: dict) -> float:
    """Continuous distance of the closest token center from the visible boundary."""

    return float(scene["clearance"])


def is_quarantined(scene: dict) -> bool:
    entries = _entry_order(scene["state_table"])
    panels = [panel for panel, _ in entries]
    return float(scene["clearance"]) < 10.0 or len(set(panels)) != len(panels)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["tokens"] = list(reversed(twin["tokens"]))
    return [("token_record_order", twin)]
