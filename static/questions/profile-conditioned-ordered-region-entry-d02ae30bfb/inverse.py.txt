"""Independent pixel-only inverse for ordered_region_entry.

This module knows only the public raster contract: five left-to-right panels,
the visible legend, and the declared center-inside-circle rule. It never loads
latent scene data or any other candidate module.
"""

from __future__ import annotations

import math

from PIL import Image


LABELS = ("A", "B", "C", "D")
PALETTE = (
    (214, 70, 70),
    (45, 116, 193),
    (41, 151, 96),
    (145, 79, 174),
)
REGION_FILL = (224, 237, 247)
PANEL_X0 = 20
PANEL_STEP = 250
PANEL_TOP = 70
PANEL_W = 238
PANEL_H = 320


def _legend_color_map(frame: Image.Image) -> dict[int, str]:
    """Read the four colored legend chips at their public fixed locations."""

    pixels = frame.load()
    result: dict[int, str] = {}
    for index, label in enumerate(LABELS):
        center_x = 330 + index * 62
        counts = [0, 0, 0, 0]
        for y in range(24, 45):
            for x in range(center_x - 11, center_x + 12):
                pixel = pixels[x, y]
                for palette_index, color in enumerate(PALETTE):
                    if pixel == color:
                        counts[palette_index] += 1
        best = max(range(4), key=lambda item: counts[item])
        if counts[best] < 20 or best in result:
            raise ValueError("legend does not expose a unique color identity")
        result[best] = label
    if set(result) != {0, 1, 2, 3}:
        raise ValueError("legend is incomplete")
    return result


def _panel_boxes(frame: Image.Image, left: int) -> tuple[dict[int, tuple[int, int, int, int]], tuple[int, int, int, int]]:
    """Recover token and region bounding boxes by exact-color segmentation."""

    pixels = frame.load()
    extrema: dict[int, list[int]] = {
        index: [frame.width, frame.height, -1, -1, 0] for index in range(4)
    }
    region = [frame.width, frame.height, -1, -1, 0]
    x_start = left + 4
    x_stop = left + PANEL_W - 4
    y_start = PANEL_TOP + 36
    y_stop = PANEL_TOP + PANEL_H - 4
    lookup = {color: index for index, color in enumerate(PALETTE)}
    for y in range(y_start, y_stop + 1):
        for x in range(x_start, x_stop + 1):
            pixel = pixels[x, y]
            palette_index = lookup.get(pixel)
            if palette_index is not None:
                box = extrema[palette_index]
                box[0] = min(box[0], x)
                box[1] = min(box[1], y)
                box[2] = max(box[2], x)
                box[3] = max(box[3], y)
                box[4] += 1
            elif pixel == REGION_FILL:
                region[0] = min(region[0], x)
                region[1] = min(region[1], y)
                region[2] = max(region[2], x)
                region[3] = max(region[3], y)
                region[4] += 1
    token_boxes: dict[int, tuple[int, int, int, int]] = {}
    for palette_index, box in extrema.items():
        if box[4] < 80 or box[2] < box[0] or box[3] < box[1]:
            raise ValueError("a panel is missing a colored token")
        token_boxes[palette_index] = (box[0], box[1], box[2], box[3])
    if region[4] < 500 or region[2] < region[0] or region[3] < region[1]:
        raise ValueError("a panel is missing the shaded region")
    return token_boxes, (region[0], region[1], region[2], region[3])


def _center(box: tuple[int, int, int, int]) -> tuple[float, float]:
    return ((box[0] + box[2]) / 2.0, (box[1] + box[3]) / 2.0)


def decision_from_image(image: Image.Image) -> str:
    """Return the first-entry label using only pixels in ``image``."""

    frame = image.convert("RGB")
    if frame.size != (1280, 420):
        raise ValueError(f"unexpected raster size {frame.size}")

    color_to_label = _legend_color_map(frame)
    states: dict[str, list[bool]] = {label: [] for label in LABELS}
    for panel_index in range(5):
        boxes, region_box = _panel_boxes(frame, PANEL_X0 + panel_index * PANEL_STEP)
        region_center = _center(region_box)
        region_radius = max(region_box[2] - region_box[0], region_box[3] - region_box[1]) / 2.0 + 2.0
        for palette_index, label in color_to_label.items():
            token_center = _center(boxes[palette_index])
            distance = math.hypot(token_center[0] - region_center[0], token_center[1] - region_center[1])
            states[label].append(distance < region_radius)

    entries: list[tuple[int, str]] = []
    for label in LABELS:
        first = None
        for panel_index in range(1, 5):
            if not states[label][panel_index - 1] and states[label][panel_index]:
                first = panel_index
                break
        if first is None:
            raise ValueError(f"no recoverable entry for token {label}")
        entries.append((first, label))
    earliest = min(panel for panel, _ in entries)
    winners = [label for panel, label in entries if panel == earliest]
    if len(winners) != 1:
        raise ValueError("ambiguous earliest transition")
    return winners[0]
