"""Deterministic scene sampler and rasterizer for convergent broken rings."""

from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image, ImageDraw

CANVAS_SIZE = 768
INK_PALETTE = (
    (31, 92, 166),
    (166, 47, 82),
    (24, 126, 88),
    (111, 67, 166),
    (190, 82, 29),
)
BACKGROUNDS = (
    (250, 248, 242),
    (246, 249, 252),
    (251, 247, 250),
)
DECORATION = (220, 220, 216)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    modes = [0, 0, 0, 0]
    if int(seed) % 2:
        modes[(int(seed) // 2) % 4] = 1
    order = list(range(4))
    rng.shuffle(order)
    return {
        "canvas_size": CANVAS_SIZE,
        "center_x": 384 + rng.choice((-2, -1, 0, 1, 2)),
        "center_y": 384 + rng.choice((-2, -1, 0, 1, 2)),
        "half_separation": rng.choice((142, 144, 146)),
        "layout_rotation_deg": rng.choice((-2.0, -1.0, 0.0, 1.0, 2.0)),
        "ring_radius": rng.choice((28.0, 29.0, 30.0)),
        "stroke_width": rng.choice((11.0, 12.0, 13.0)),
        "gap_width_deg": rng.choice((76.0, 80.0, 84.0)),
        "gap_jitter_deg": [round(rng.uniform(-3.5, 3.5), 6) for _ in range(4)],
        "gap_modes": modes,
        "ink_rgb": list(rng.choice(INK_PALETTE)),
        "background_rgb": list(rng.choice(BACKGROUNDS)),
        "draw_order": order,
        "decoration_phase": rng.randrange(17),
    }


def _ring_centers(scene: dict) -> list[tuple[float, float]]:
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    half = float(scene["half_separation"])
    theta = math.radians(float(scene["layout_rotation_deg"]))
    cosine, sine = math.cos(theta), math.sin(theta)
    centers: list[tuple[float, float]] = []
    for dx, dy in ((-half, -half), (half, -half), (half, half), (-half, half)):
        centers.append((cx + dx * cosine - dy * sine, cy + dx * sine + dy * cosine))
    return centers


def _angle_delta_deg(angle: np.ndarray, center: float) -> np.ndarray:
    return np.abs((angle - center + 180.0) % 360.0 - 180.0)


def _ring_mask(scene: dict, index: int) -> tuple[slice, slice, np.ndarray]:
    center = _ring_centers(scene)[index]
    radius = float(scene["ring_radius"])
    stroke = float(scene["stroke_width"])
    gap_width = float(scene["gap_width_deg"])
    target = (float(scene["center_x"]), float(scene["center_y"]))
    inward = math.degrees(math.atan2(target[1] - center[1], target[0] - center[0]))
    gap_angle = inward + float(scene["gap_jitter_deg"][index])
    if int(scene["gap_modes"][index]):
        gap_angle += 180.0
    pad = int(math.ceil(radius + stroke)) + 2
    x0 = int(math.floor(center[0])) - pad
    x1 = int(math.floor(center[0])) + pad + 1
    y0 = int(math.floor(center[1])) - pad
    y1 = int(math.floor(center[1])) + pad + 1
    yy, xx = np.mgrid[y0:y1, x0:x1]
    dx = xx.astype(np.float64) - center[0]
    dy = yy.astype(np.float64) - center[1]
    distance = np.hypot(dx, dy)
    angle = np.degrees(np.arctan2(dy, dx))
    annulus = np.abs(distance - radius) <= stroke / 2.0
    visible = _angle_delta_deg(angle, gap_angle) >= gap_width / 2.0
    return slice(y0, y1), slice(x0, x1), annulus & visible


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas_size"])
    background = np.asarray(scene["background_rgb"], dtype=np.uint8)
    array = np.empty((size, size, 3), dtype=np.uint8)
    array[:, :] = background
    ink = np.asarray(scene["ink_rgb"], dtype=np.uint8)
    for raw_index in scene["draw_order"]:
        ys, xs, mask = _ring_mask(scene, int(raw_index))
        patch = array[ys, xs]
        patch[mask] = ink
    image = Image.fromarray(array, mode="RGB")
    draw = ImageDraw.Draw(image)
    phase = int(scene["decoration_phase"])
    # Neutral peripheral dots are visible nuisance marks but remain outside the
    # chromatic inverse mask and outside the decisive central region.
    for index in range(8):
        x = 42 + ((index * 227 + phase * 31) % 676)
        y = 39 + ((index * 311 + phase * 47) % 690)
        if 180 < x < 588 and 180 < y < 588:
            x = 52 if index % 2 == 0 else 716
        draw.ellipse((x - 2, y - 2, x + 2, y + 2), fill=DECORATION)
    return image


def analytic_gold(scene: dict) -> str:
    modes = [int(value) for value in scene["gap_modes"]]
    return "yes" if sum(modes) == 0 else "no"


def margin(scene: dict) -> float:
    """Angular runner-up clearance in degrees from the 55-degree cone boundary."""

    jitters = [abs(float(value)) for value in scene["gap_jitter_deg"]]
    return 55.0 - max(jitters)


def is_quarantined(scene: dict) -> bool:
    modes = [int(value) for value in scene["gap_modes"]]
    jitters = [abs(float(value)) for value in scene["gap_jitter_deg"]]
    return len(modes) != 4 or any(value not in (0, 1) for value in modes) or sum(modes) > 1 or max(jitters, default=99.0) >= 20.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["draw_order"] = list(reversed([int(value) for value in scene["draw_order"]]))
    return [("reverse_disjoint_ring_draw_order", twin)]
