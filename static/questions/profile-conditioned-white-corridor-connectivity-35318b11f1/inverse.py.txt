"""Pixel-only inverse arm for white-corridor connectivity."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    seen = np.zeros(mask.shape, dtype=bool)
    groups = []
    height, width = mask.shape
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        pending = [(int(y), int(x))]
        seen[y, x] = True
        points = []
        while pending:
            cy, cx = pending.pop()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    pending.append((ny, nx))
        if len(points) >= 40:
            groups.append(np.asarray(points, dtype=np.int32))
    return groups


def decision_from_image(image: Image.Image) -> str:
    """Recover marker locations and four-connected free space from RGB pixels only."""
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    dark = np.max(arr, axis=2) < 90
    ys, xs = np.nonzero(dark)
    if len(xs) < 100:
        raise ValueError("no dark maze walls found")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())

    chroma = np.max(arr, axis=2) - np.min(arr, axis=2)
    colorful = (chroma > 65) & (np.max(arr, axis=2) > 115) & ~dark
    markers = _components(colorful)
    if len(markers) != 2:
        raise ValueError(f"expected two colored terminal components, found {len(markers)}")
    centers = []
    for points in markers:
        centers.append((int(round(points[:, 0].mean())), int(round(points[:, 1].mean()))))

    passable = ~dark
    passable[:y0, :] = False
    passable[y1 + 1 :, :] = False
    passable[:, :x0] = False
    passable[:, x1 + 1 :] = False
    start, goal = centers
    if not passable[start] or not passable[goal]:
        raise ValueError("terminal center lies on a wall")
    seen = np.zeros(passable.shape, dtype=bool)
    pending = deque([start])
    seen[start] = True
    height, width = passable.shape
    while pending:
        y, x = pending.popleft()
        if (y, x) == goal:
            return "yes"
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < height and 0 <= nx < width and passable[ny, nx] and not seen[ny, nx]:
                seen[ny, nx] = True
                pending.append((ny, nx))
    return "no"
