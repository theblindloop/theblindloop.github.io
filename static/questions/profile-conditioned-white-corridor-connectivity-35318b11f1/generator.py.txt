"""Deterministic white-corridor connectivity world."""

from __future__ import annotations

import random
from collections import deque

from PIL import Image, ImageDraw

BACKGROUND = (238, 240, 237)
CORRIDOR = (250, 250, 247)
WALL = (30, 37, 43)
PALETTES = (
    ((0, 137, 123), (230, 126, 34)),
    ((52, 101, 164), (222, 72, 98)),
    ((117, 78, 158), (225, 154, 32)),
    ((0, 150, 199), (205, 72, 128)),
)


def _edge(a: tuple[int, int], b: tuple[int, int]) -> tuple[int, int, int, int]:
    return (*a, *b) if a < b else (*b, *a)


def _all_edges(n: int) -> list[tuple[int, int, int, int]]:
    edges = []
    for r in range(n):
        for c in range(n):
            if r + 1 < n:
                edges.append(_edge((r, c), (r + 1, c)))
            if c + 1 < n:
                edges.append(_edge((r, c), (r, c + 1)))
    return edges


def _adjacency(scene: dict) -> dict[tuple[int, int], set[tuple[int, int]]]:
    n = int(scene["grid_size"])
    graph = {(r, c): set() for r in range(n) for c in range(n)}
    for r1, c1, r2, c2 in scene["open_edges"]:
        graph[(r1, c1)].add((r2, c2))
        graph[(r2, c2)].add((r1, c1))
    return graph


def _connected(scene: dict) -> bool:
    graph = _adjacency(scene)
    start = tuple(scene["start"])
    goal = tuple(scene["goal"])
    seen = {start}
    pending = [start]
    while pending:
        node = pending.pop()
        if node == goal:
            return True
        for neighbor in graph[node] - seen:
            seen.add(neighbor)
            pending.append(neighbor)
    return False


def _perfect_maze(rng: random.Random, n: int) -> set[tuple[int, int, int, int]]:
    parent = list(range(n * n))

    def find(x: int) -> int:
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    edges = _all_edges(n)
    rng.shuffle(edges)
    opened: set[tuple[int, int, int, int]] = set()
    for edge in edges:
        a = edge[0] * n + edge[1]
        b = edge[2] * n + edge[3]
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[ra] = rb
            opened.add(edge)
    return opened


def _tree_path(
    opened: set[tuple[int, int, int, int]],
    start: tuple[int, int],
    goal: tuple[int, int],
    n: int,
) -> list[tuple[int, int, int, int]]:
    graph = {(r, c): [] for r in range(n) for c in range(n)}
    for e in opened:
        a, b = (e[0], e[1]), (e[2], e[3])
        graph[a].append(b)
        graph[b].append(a)
    parent = {start: None}
    pending = deque([start])
    while pending:
        node = pending.popleft()
        if node == goal:
            break
        for neighbor in graph[node]:
            if neighbor not in parent:
                parent[neighbor] = node
                pending.append(neighbor)
    path = []
    node = goal
    while parent[node] is not None:
        previous = parent[node]
        path.append(_edge(previous, node))
        node = previous
    path.reverse()
    return path


def _component(
    opened: set[tuple[int, int, int, int]], start: tuple[int, int], n: int
) -> set[tuple[int, int]]:
    graph = {(r, c): set() for r in range(n) for c in range(n)}
    for e in opened:
        a, b = (e[0], e[1]), (e[2], e[3])
        graph[a].add(b)
        graph[b].add(a)
    seen = {start}
    pending = [start]
    while pending:
        node = pending.pop()
        for neighbor in graph[node] - seen:
            seen.add(neighbor)
            pending.append(neighbor)
    return seen


def sample_scene(seed: int) -> dict:
    """Sample paired-balanced labels with equal numbers of open walls."""
    rng = random.Random(int(seed))
    n = 7
    cells = [(r, c) for r in range(n) for c in range(n)]
    for _ in range(100):
        start, goal = rng.sample(cells, 2)
        if abs(start[0] - goal[0]) + abs(start[1] - goal[1]) >= 6:
            break
    else:
        raise RuntimeError("could not place separated terminals")

    opened = _perfect_maze(rng, n)
    want_connected = (seed % 2 == 0)
    if not want_connected:
        path = _tree_path(opened, start, goal, n)
        cut = path[len(path) // 2]
        opened.remove(cut)
        side = _component(opened, start, n)
        candidates = []
        for e in _all_edges(n):
            a, b = (e[0], e[1]), (e[2], e[3])
            if e not in opened and e != cut and ((a in side) == (b in side)):
                candidates.append(e)
        if not candidates:
            raise RuntimeError("could not preserve wall count while splitting maze")
        opened.add(rng.choice(candidates))

    pair_index = seed // 2
    cell_size = (52, 56, 60)[pair_index % 3]
    wall_width = (4, 6, 8, 10)[pair_index % 4]
    span = n * cell_size
    jitter_x, jitter_y = rng.randint(-10, 10), rng.randint(-10, 10)
    offset_x = (512 - span) // 2 + jitter_x
    offset_y = (512 - span) // 2 + jitter_y
    colors = PALETTES[pair_index % len(PALETTES)]
    if rng.random() < 0.5:
        colors = colors[::-1]
    scene = {
        "grid_size": n,
        "cell_size": cell_size,
        "wall_width": wall_width,
        "offset_x": offset_x,
        "offset_y": offset_y,
        "open_edges": [list(e) for e in sorted(opened)],
        "start": list(start),
        "goal": list(goal),
        "marker_colors": [list(colors[0]), list(colors[1])],
        "marker_radius": 11 + (pair_index % 4),
        "canvas_size": 512,
    }
    if _connected(scene) != want_connected:
        raise AssertionError("sampled graph has wrong connectivity")
    return scene


def _strip(center: int, width: int) -> tuple[int, int]:
    lo = center - width // 2
    return lo, lo + width - 1


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas_size"])
    n = int(scene["grid_size"])
    cell = int(scene["cell_size"])
    wall = int(scene["wall_width"])
    left, top = int(scene["offset_x"]), int(scene["offset_y"])
    image = Image.new("RGB", (size, size), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rectangle((left, top, left + n * cell, top + n * cell), fill=CORRIDOR)

    # Draw the complete lattice first, then remove one equal-area throat per open edge.
    for k in range(n + 1):
        x0, x1 = _strip(left + k * cell, wall)
        draw.rectangle((x0, top - wall // 2, x1, top + n * cell + wall // 2), fill=WALL)
        y0, y1 = _strip(top + k * cell, wall)
        draw.rectangle((left - wall // 2, y0, left + n * cell + wall // 2, y1), fill=WALL)
    for r1, c1, r2, c2 in scene["open_edges"]:
        if r1 == r2:
            x0, x1 = _strip(left + max(c1, c2) * cell, wall)
            y0 = top + r1 * cell + wall
            y1 = top + (r1 + 1) * cell - wall
        else:
            y0, y1 = _strip(top + max(r1, r2) * cell, wall)
            x0 = left + c1 * cell + wall
            x1 = left + (c1 + 1) * cell - wall
        draw.rectangle((x0, y0, x1, y1), fill=CORRIDOR)

    radius = int(scene["marker_radius"])
    for cell_rc, color in zip((scene["start"], scene["goal"]), scene["marker_colors"]):
        r, c = cell_rc
        cx = left + c * cell + cell // 2
        cy = top + r * cell + cell // 2
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=tuple(color))
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), outline=(255, 255, 255), width=2)
    return image


def analytic_gold(scene: dict) -> str:
    return "yes" if _connected(scene) else "no"


def margin(scene: dict) -> float:
    return float(min(int(scene["wall_width"]), int(scene["cell_size"]) - 2 * int(scene["wall_width"])))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
