"""Independent pixel-only inverse for the calibration residual question."""

from __future__ import annotations

import numpy as np
from PIL import Image


# These are public chart-layout facts duplicated independently for the inverse;
# no renderer module or latent scene is imported.
PLOT_RECTS = ((82, 156, 350, 508), (475, 156, 743, 508), (868, 156, 1136, 508))
CHECKPOINTS = (0.22, 0.313, 0.407, 0.5, 0.593, 0.687, 0.78)
MODEL_LABELS = ("A", "B", "C")


def _colored_mask(image: Image.Image) -> np.ndarray:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    spread = pixels.max(axis=2) - pixels.min(axis=2)
    brightness = pixels.mean(axis=2)
    mask = (spread >= 48) & (brightness <= 247)
    mask[:130, :] = False
    mask[535:, :] = False
    return mask


def _checkpoint_y(mask: np.ndarray, plot: tuple[int, int, int, int], checkpoint: float) -> float:
    x0, y0, x1, y1 = plot
    center_x = round(x0 + checkpoint * (x1 - x0))
    left = max(x0, center_x - 9)
    right = min(x1 + 1, center_x + 10)
    ys = np.nonzero(mask[y0 : y1 + 1, left:right])[0] + y0
    if ys.size < 8:
        raise ValueError("colored checkpoint is not recoverable")
    return float(np.median(ys))


def _panel_residual(mask: np.ndarray, plot: tuple[int, int, int, int]) -> float:
    x0, y0, x1, y1 = plot
    height = float(y1 - y0)
    gaps = []
    for checkpoint in CHECKPOINTS:
        x = x0 + checkpoint * (x1 - x0)
        observed_y = _checkpoint_y(mask, plot, checkpoint)
        diagonal_y = y1 - checkpoint * height
        gaps.append(abs(observed_y - diagonal_y) / height)
    return max(gaps)


def decision_from_image(image: Image.Image) -> str:
    """Recover the model label using only the final RGB raster."""

    mask = _colored_mask(image)
    residuals = [_panel_residual(mask, plot) for plot in PLOT_RECTS]
    best = max(range(3), key=lambda index: residuals[index])
    if sorted(residuals, reverse=True)[0] - sorted(residuals, reverse=True)[1] < 0.004:
        raise ValueError("residual decision is too close at raster resolution")
    return MODEL_LABELS[best]
