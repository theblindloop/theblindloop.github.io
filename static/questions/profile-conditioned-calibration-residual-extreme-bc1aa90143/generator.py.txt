"""Deterministic three-panel calibration residual renderer and analytic gold."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw, ImageFont


CANVAS_SIZE = (1200, 620)
MODEL_LABELS = ("A", "B", "C")
PANEL_RECTS = ((28, 92, 386, 552), (421, 92, 779, 552), (814, 92, 1172, 552))
PLOT_RECTS = ((82, 156, 350, 508), (475, 156, 743, 508), (868, 156, 1136, 508))
CHECKPOINTS = (0.22, 0.313, 0.407, 0.5, 0.593, 0.687, 0.78)
QUARANTINE_GAP = 0.045
CURVE_WIDTH = 6
MARKER_RADIUS = 8

PALETTE = (
    (214, 65, 73),
    (39, 112, 203),
    (32, 153, 108),
    (140, 76, 190),
    (226, 129, 35),
    (194, 59, 145),
)


def _max_residual(values: list[float] | tuple[float, ...]) -> float:
    return max(abs(float(value) - checkpoint) for value, checkpoint in zip(values, CHECKPOINTS))


def _scores(scene: dict) -> list[float]:
    return [
        _max_residual(scene["a_values"]),
        _max_residual(scene["b_values"]),
        _max_residual(scene["c_values"]),
    ]


def _make_series(rng: random.Random, target: float) -> list[float]:
    residuals = [rng.uniform(-0.34 * target, 0.34 * target) for _ in CHECKPOINTS]
    peak = rng.randrange(len(CHECKPOINTS))
    residuals[peak] = target if rng.random() < 0.5 else -target
    return [round(checkpoint + residual, 6) for checkpoint, residual in zip(CHECKPOINTS, residuals)]


def sample_scene(seed: int) -> dict:
    """Sample three traces with a balanced winner and a visible score gap."""

    rng = random.Random(int(seed))
    winner = (int(seed) * 5 + 1) % 3
    near_boundary = int(seed) % 11 in (0, 1)
    winner_score = rng.uniform(0.13, 0.18)
    gap = rng.uniform(0.022, 0.035) if near_boundary else rng.uniform(0.060, 0.095)
    runner_score = winner_score - gap
    third_cap = min(0.075, runner_score - 0.035)
    third_score = rng.uniform(0.025, max(0.026, third_cap))
    target_by_index = [third_score, third_score, third_score]
    target_by_index[winner] = winner_score
    runner = (winner + (1 if rng.random() < 0.5 else 2)) % 3
    target_by_index[runner] = runner_score
    values = [_make_series(rng, target) for target in target_by_index]
    colors = [list(color) for color in rng.sample(PALETTE, 3)]
    return {
        "a_values": values[0],
        "b_values": values[1],
        "c_values": values[2],
        "colors": colors,
    }


def _font() -> ImageFont.ImageFont:
    return ImageFont.load_default()


def _point_xy(plot: tuple[int, int, int, int], checkpoint: float, value: float) -> tuple[int, int]:
    x0, y0, x1, y1 = plot
    x = round(x0 + checkpoint * (x1 - x0))
    y = round(y1 - value * (y1 - y0))
    return x, y


def _draw_dashed_diagonal(draw: ImageDraw.ImageDraw, plot: tuple[int, int, int, int]) -> None:
    x0, y0, x1, y1 = plot
    for index in range(0, 30, 2):
        start = index / 30.0
        end = min((index + 1) / 30.0, 1.0)
        draw.line(
            (
                x0 + (x1 - x0) * start,
                y1 - (y1 - y0) * start,
                x0 + (x1 - x0) * end,
                y1 - (y1 - y0) * end,
            ),
            fill=(54, 61, 72),
            width=3,
        )


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    plot: tuple[int, int, int, int],
    panel: tuple[int, int, int, int],
    label: str,
    values: list[float],
    color: tuple[int, int, int] | list[int],
    font: ImageFont.ImageFont,
) -> None:
    draw.rounded_rectangle(panel, radius=10, fill=(247, 249, 252), outline=(185, 193, 204), width=2)
    draw.text((panel[0] + 14, panel[1] + 13), f"Model {label}", fill=(29, 36, 47), font=font)
    x0, y0, x1, y1 = plot
    draw.rectangle(plot, fill=(255, 255, 255), outline=(190, 197, 207), width=2)
    for fraction in (0.25, 0.5, 0.75):
        gx = round(x0 + fraction * (x1 - x0))
        gy = round(y1 - fraction * (y1 - y0))
        draw.line((gx, y0, gx, y1), fill=(229, 233, 239), width=1)
        draw.line((x0, gy, x1, gy), fill=(229, 233, 239), width=1)
    draw.line((x0, y1, x1, y1), fill=(76, 84, 98), width=2)
    draw.line((x0, y0, x0, y1), fill=(76, 84, 98), width=2)
    _draw_dashed_diagonal(draw, plot)
    points = [_point_xy(plot, checkpoint, value) for checkpoint, value in zip(CHECKPOINTS, values)]
    draw.line(points, fill=tuple(color), width=CURVE_WIDTH, joint="curve")
    for x, y in points:
        draw.ellipse(
            (x - MARKER_RADIUS, y - MARKER_RADIUS, x + MARKER_RADIUS, y + MARKER_RADIUS),
            fill=tuple(color),
        )
    draw.text((x0 - 3, y1 + 10), "0", fill=(76, 84, 98), font=font)
    draw.text((x1 - 7, y1 + 10), "1", fill=(76, 84, 98), font=font)
    draw.text((x0 - 17, y0 - 2), "1", fill=(76, 84, 98), font=font)
    draw.text((x0 - 17, y1 - 7), "0", fill=(76, 84, 98), font=font)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS_SIZE, (250, 251, 253))
    draw = ImageDraw.Draw(image)
    font = _font()
    draw.text((34, 27), "calibration traces and common reference", fill=(31, 38, 50), font=font)
    draw.text((34, 50), "colored markers are sampled at the same seven checkpoints", fill=(87, 95, 108), font=font)
    draw.line((778, 43, 812, 43), fill=(54, 61, 72), width=3)
    draw.text((820, 36), "dashed diagonal: perfect calibration", fill=(66, 74, 87), font=font)
    value_sets = (scene["a_values"], scene["b_values"], scene["c_values"])
    for plot, panel, label, values, color in zip(
        PLOT_RECTS, PANEL_RECTS, MODEL_LABELS, value_sets, scene["colors"]
    ):
        _draw_panel(draw, plot, panel, label, values, color, font)
    return image


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    return MODEL_LABELS[max(range(3), key=lambda index: scores[index])]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
