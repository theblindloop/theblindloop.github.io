"""Content-addressed-swap world.

Latent scene z = {arrangement, program, arrow_size, paint_order}.

Six discs of six flat colours sit in six slots numbered 1..6 from the left.
Below them a program of 12 to 15 instructions is drawn, one per row, read top
to bottom. Each instruction names a colour (a small disc of that colour) and a
side (a solid triangle pointing left or right): find the disc of that colour in
the strip *as it stands at that moment* and swap it with its neighbour on that
side; if the named colour already sits in the end slot on that side the
instruction does nothing.

The operand of every instruction is content-addressed, not positional, so the
strip cannot be updated slot-wise and no single disc can be tracked on its own:
whether an instruction touches the red disc depends on where the *named* colour
currently is. The sampler forces at least three steps in which red moves as the
partner of an instruction that does not name it, so the "only red instructions
move red" decomposition is guaranteed wrong, and at least two instructions are
blocked no-ops.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "content_addressed_swaps-0.1.0"

N_SLOTS = 6
RED = 0

# palette: index 0 is the queried red disc
COLORS = (
    (214, 39, 40),    # red
    (31, 119, 180),   # blue
    (44, 160, 44),    # green
    (255, 140, 0),    # orange
    (148, 60, 190),   # purple
    (23, 190, 190),   # teal
)
TRI = (45, 45, 55)
FRAME = (170, 172, 178)
BG = (255, 255, 255)

WIDTH = 1000
HEIGHT = 1560

SLOT_X0 = 50
SLOT_PITCH = 150
SLOT_BOX = 140
SLOT_Y0 = 40
DISC_TOP = 110
FRAME_W = 3

RULE_Y = 230
RULE_H = 6

ROW_Y0 = 300
ROW_PITCH = 82
ROW_HALF = 34
LIST_DISC = 58
LIST_DISC_X = 400
TRI_X = 500

ARROW_MIN = 42
ARROW_MAX = 96
QUARANTINE_MARGIN = 8.0

PROGRAM_MIN = 12
PROGRAM_MAX = 15

_SIDES = ("L", "R")


# --------------------------------------------------------------------------
# dynamics
# --------------------------------------------------------------------------
def _run(arrangement: list[int], program: list[list], mode: str = "std") -> tuple[int, dict]:
    """Execute the program and return (final 1-based slot of red, stats).

    mode 'std'     -- the rule: swap the named colour with its neighbour on the
                      arrow's side; a named colour already in that end slot is a
                      no-op;
    mode 'mirror'  -- read every arrow the other way round;
    mode 'wrap'    -- treat the strip as circular, so a blocked instruction
                      swaps the two end slots instead of doing nothing;
    mode 'reverse' -- execute the instructions bottom to top;
    mode 'redonly' -- apply only the instructions that name red (the
                      decomposition shortcut: track one disc alone).
    """
    strip = list(arrangement)
    steps = program[::-1] if mode == "reverse" else program
    blocked = 0
    indirect_red = 0
    red_moves = 0
    for color, side in steps:
        if mode == "mirror":
            side = "L" if side == "R" else "R"
        if mode == "redonly" and color != RED:
            continue
        i = strip.index(color)
        j = i + 1 if side == "R" else i - 1
        before = strip.index(RED)
        if 0 <= j < N_SLOTS:
            strip[i], strip[j] = strip[j], strip[i]
        elif mode == "wrap":
            a, b = (N_SLOTS - 1, 0) if side == "R" else (0, N_SLOTS - 1)
            strip[a], strip[b] = strip[b], strip[a]
        else:
            blocked += 1
        after = strip.index(RED)
        if after != before:
            red_moves += 1
            if color != RED:
                indirect_red += 1
    stats = {
        "blocked": blocked,
        "indirect_red": indirect_red,
        "red_moves": red_moves,
        "final_strip": strip,
    }
    return strip.index(RED) + 1, stats


def _variant(scene: dict, mode: str) -> int:
    return _run(list(scene["arrangement"]), [list(p) for p in scene["program"]], mode)[0]


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------
def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 2654435761 % (2 ** 61 - 1))
    for _ in range(8000):
        arrangement = list(range(N_SLOTS))
        rng.shuffle(arrangement)
        k = rng.randint(PROGRAM_MIN, PROGRAM_MAX)
        program = [[rng.randrange(N_SLOTS), rng.choice(_SIDES)] for _ in range(k)]

        if sum(1 for c, _ in program if c == RED) > 5:
            continue
        gold, stats = _run(arrangement, program, "std")
        if stats["blocked"] < 2:
            continue
        if stats["indirect_red"] < 3:
            continue
        if stats["red_moves"] < 4:
            continue
        if gold == arrangement.index(RED) + 1:
            continue
        if any(gold == _run(arrangement, program, m)[0] for m in ("mirror", "wrap", "reverse", "redonly")):
            continue

        arrow_size = rng.randint(ARROW_MIN, ARROW_MAX)
        paint_order = list(range(k))
        rng.shuffle(paint_order)
        return {
            "arrangement": arrangement,
            "program": program,
            "arrow_size": arrow_size,
            "paint_order": paint_order,
        }
    raise RuntimeError(f"sampling failed for seed {seed}")


def analytic_gold(scene: dict) -> str:
    slot, _ = _run(list(scene["arrangement"]), [list(p) for p in scene["program"]], "std")
    return str(slot)


def margin(scene: dict) -> float:
    """Arrow orientation margin: the triangle's centroid-to-bbox-centre offset."""
    return round(int(scene["arrow_size"]) / 6.0, 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Paint order is a sampled but photometrically invisible scene field."""
    order = list(scene["paint_order"])
    out: list[tuple[str, dict]] = []
    if len(order) > 1:
        rotated = dict(scene)
        rotated["paint_order"] = order[1:] + order[:1]
        out.append(("rotated_paint_order", rotated))
        reversed_ = dict(scene)
        reversed_["paint_order"] = order[::-1]
        out.append(("reversed_paint_order", reversed_))
    return out


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def _tri_height(arrow_size: int) -> int:
    return 2 * max(14, min(32, arrow_size // 3))


def render(scene: dict) -> Image.Image:
    arrangement = [int(c) for c in scene["arrangement"]]
    program = [[int(p[0]), str(p[1])] for p in scene["program"]]
    arrow_size = int(scene["arrow_size"])
    order = [int(i) for i in scene["paint_order"]]

    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)

    # the strip of six slots
    for i, color in enumerate(arrangement):
        bx = SLOT_X0 + i * SLOT_PITCH
        draw.rectangle(
            [bx, SLOT_Y0, bx + SLOT_BOX, SLOT_Y0 + SLOT_BOX],
            outline=FRAME,
            width=FRAME_W,
        )
        inset = (SLOT_BOX - DISC_TOP) // 2
        dx = bx + inset
        dy = SLOT_Y0 + inset
        draw.ellipse([dx, dy, dx + DISC_TOP, dy + DISC_TOP], fill=COLORS[color])

    draw.rectangle([SLOT_X0, RULE_Y, WIDTH - SLOT_X0, RULE_Y + RULE_H], fill=FRAME)

    # the program, one instruction per row; paint order is invisible
    h = _tri_height(arrow_size)
    for j in order:
        color, side = program[j]
        yc = ROW_Y0 + j * ROW_PITCH + ROW_HALF
        r = LIST_DISC // 2
        draw.ellipse(
            [LIST_DISC_X, yc - r, LIST_DISC_X + LIST_DISC, yc + r],
            fill=COLORS[color],
        )
        if side == "R":
            pts = [
                (TRI_X, yc - h // 2),
                (TRI_X, yc + h // 2),
                (TRI_X + arrow_size, yc),
            ]
        else:
            pts = [
                (TRI_X + arrow_size, yc - h // 2),
                (TRI_X + arrow_size, yc + h // 2),
                (TRI_X, yc),
            ]
        draw.polygon(pts, fill=TRI)

    return image
