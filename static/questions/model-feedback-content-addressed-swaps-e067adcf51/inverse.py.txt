"""Independent pixel-only inverse arm for the content-addressed-swap world.

From the final PNG alone it

1. masks the six flat disc hues and the flat triangle ink by exact colour;
2. splits the combined ink into horizontal bands (every mark lives in its own
   band, so the y-projection has one clean run per band);
3. reads the topmost band as the strip: for each hue present, its mean x fixes
   which slot it occupies, and sorting the hues by x recovers the arrangement;
4. reads every later band as one instruction: the single hue present names the
   colour, and the triangle's centroid-to-bbox-centre offset gives the side it
   points to (a solid triangle's mass sits behind its apex);
5. executes the content-addressed swap program on the recovered strip and
   reports the 1-based slot holding the red disc.

It imports no renderer / prompts / generate / verify / analytic-gold code and
never receives the latent scene.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_PALETTE = (
    (214, 39, 40),
    (31, 119, 180),
    (44, 160, 44),
    (255, 140, 0),
    (148, 60, 190),
    (23, 190, 190),
)
_TRI = (45, 45, 55)
_RED = 0
_N_SLOTS = 6


def _near(arr: np.ndarray, color, tol: int) -> np.ndarray:
    diff = np.abs(arr.astype(int) - np.array(color, dtype=int))
    return (diff <= tol).all(axis=-1)


def _bands(flags: np.ndarray, min_len: int = 8) -> list[tuple[int, int]]:
    out: list[tuple[int, int]] = []
    start = None
    for i, f in enumerate(flags):
        if f and start is None:
            start = i
        elif not f and start is not None:
            if i - start >= min_len:
                out.append((start, i - 1))
            start = None
    if start is not None and len(flags) - start >= min_len:
        out.append((start, len(flags) - 1))
    return out


def _side(sub: np.ndarray) -> str:
    ys, xs = np.nonzero(sub)
    if xs.size == 0:
        return "R"
    ox = xs.mean() - (xs.min() + xs.max()) / 2.0
    return "R" if ox < 0 else "L"


def decision_from_image(image: Image.Image) -> str:
    """Return '1'..'6': the slot holding the red disc after the program runs."""
    arr = np.asarray(image.convert("RGB"))
    hue_masks = [_near(arr, c, 30) for c in _PALETTE]
    tri = _near(arr, _TRI, 30)

    ink = tri.copy()
    for m in hue_masks:
        ink |= m
    bands = _bands(ink.any(axis=1))
    if len(bands) < 3:
        return "1"

    top = bands[0]
    present = []
    for idx, m in enumerate(hue_masks):
        sub = m[top[0] : top[1] + 1]
        ys, xs = np.nonzero(sub)
        if xs.size:
            present.append((xs.mean(), idx))
    if len(present) != _N_SLOTS:
        return "1"
    strip = [idx for _, idx in sorted(present)]

    program: list[tuple[int, str]] = []
    for lo, hi in bands[1:]:
        color = None
        best = 0
        for idx, m in enumerate(hue_masks):
            count = int(m[lo : hi + 1].sum())
            if count > best:
                best = count
                color = idx
        if color is None:
            continue
        program.append((color, _side(tri[lo : hi + 1])))

    for color, side in program:
        if color not in strip:
            continue
        i = strip.index(color)
        j = i + 1 if side == "R" else i - 1
        if 0 <= j < _N_SLOTS:
            strip[i], strip[j] = strip[j], strip[i]

    return str(strip.index(_RED) + 1)
