"""Pixel-only inverse arm. It has no access to latent scene data."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

PALETTE = {"orange": np.array([232, 121, 36]), "blue": np.array([35, 116, 216])}
DOT = np.array([24, 34, 45])


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        points = []
        while stack:
            yy, xx = stack.pop()
            points.append((yy, xx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    ny, nx = yy + dy, xx + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        if len(points) >= 80:
            groups.append(np.asarray(points, dtype=float))
    return groups


def _fit_axis(arr: np.ndarray, rgb: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    distance = np.linalg.norm(arr - rgb, axis=2)
    points = np.argwhere(distance < 58.0)[:, ::-1].astype(float) + 0.5
    if len(points) < 100:
        raise ValueError("candidate axis is not visible")
    center = points.mean(axis=0)
    _, _, vh = np.linalg.svd(points - center, full_matrices=False)
    direction = vh[0]
    return center, direction / np.linalg.norm(direction)


def _reflection_error(points: np.ndarray, axis: tuple[np.ndarray, np.ndarray]) -> float:
    center, direction = axis
    delta = points - center
    reflected = 2 * (delta @ direction)[:, None] * direction[None, :] - delta + center
    distances = np.linalg.norm(reflected[:, None, :] - points[None, :, :], axis=2)
    return float(distances.min(axis=1).mean())


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=float)
    dot_mask = np.linalg.norm(arr - DOT, axis=2) < 62.0
    groups = _components(dot_mask)
    if len(groups) < 7:
        raise ValueError("complete dot constellation is not recoverable")
    centers = np.asarray([[group[:, 1].mean() + 0.5, group[:, 0].mean() + 0.5] for group in groups])
    errors = {
        name: _reflection_error(centers, _fit_axis(arr, rgb)) for name, rgb in PALETTE.items()
    }
    ordered = sorted(errors.items(), key=lambda item: item[1])
    if ordered[1][1] - ordered[0][1] < 4.0:
        raise ValueError("ambiguous pixel-scale symmetry margin")
    return ordered[0][0]
