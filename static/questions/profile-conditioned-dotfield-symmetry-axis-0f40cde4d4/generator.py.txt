"""Deterministic renderer and analytic arm for a global dot-symmetry task."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 440
HEIGHT = 440
SCALE = 3
BACKGROUND = "#F7F4ED"
BLACK = "#18222D"
COLORS = {"orange": "#E87924", "blue": "#2374D8"}
PALETTES = [
    {"background": "#F7F4ED", "dot": "#18222D"},
    {"background": "#F4F7FA", "dot": "#202A38"},
    {"background": "#FFF8EC", "dot": "#111C29"},
]
QUARANTINE_GAP = 8.0


def _reflect(point: tuple[float, float], axis: dict[str, float]) -> tuple[float, float]:
    cx, cy, angle = axis["cx"], axis["cy"], math.radians(axis["angle"])
    ux, uy = math.cos(angle), math.sin(angle)
    dx, dy = point[0] - cx, point[1] - cy
    along = dx * ux + dy * uy
    return (2 * (cx + along * ux) - point[0], 2 * (cy + along * uy) - point[1])


def _set_error(dots: list[list[float]], axis: dict[str, float]) -> float:
    points = [(float(dot[0]), float(dot[1])) for dot in dots]
    nearest = []
    for point in points:
        rx, ry = _reflect(point, axis)
        nearest.append(min(math.hypot(rx - x, ry - y) for x, y in points))
    return sum(nearest) / len(nearest)


def _make_dots(rng: random.Random, axis: dict[str, float]) -> list[list[float]]:
    angle = math.radians(axis["angle"])
    ux, uy = math.cos(angle), math.sin(angle)
    vx, vy = -uy, ux
    # Six independent correspondences plus one on-axis anchor. Avoid opposite
    # along-axis coordinates so the perpendicular distractor is visibly worse.
    us = [-126, -91, -57, -18, 43, 82, 121]
    rng.shuffle(us)
    us = us[:6]
    dots: list[list[float]] = []
    for index, u in enumerate(us):
        v = rng.uniform(38, 116)
        radius = rng.choice([8, 9, 10])
        for sign in (-1, 1):
            x = axis["cx"] + u * ux + sign * v * vx
            y = axis["cy"] + u * uy + sign * v * vy
            dots.append([round(x, 2), round(y, 2), radius])
    anchor_u = rng.choice([-108, -72, 28, 66, 104])
    dots.append([
        round(axis["cx"] + anchor_u * ux, 2),
        round(axis["cy"] + anchor_u * uy, 2),
        rng.choice([8, 9, 10]),
    ])
    rng.shuffle(dots)
    return dots


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    cx = 220 + rng.randint(-10, 10)
    cy = 220 + rng.randint(-10, 10)
    angle = rng.choice([19, 27, 35, 43, 51, 59, 67, 75]) + rng.uniform(-2, 2)
    winner = {"cx": float(cx), "cy": float(cy), "angle": round(angle, 3)}
    loser = {"cx": float(cx), "cy": float(cy), "angle": round(angle + 90.0, 3)}
    winner_color, loser_color = (("orange", "blue") if int(seed) % 2 == 0 else ("blue", "orange"))
    dots = _make_dots(rng, winner)
    # Draw order is independently counterbalanced so the tiny crossing junction
    # cannot reveal which axis wins.
    if (int(seed) // 2) % 2 == 0:
        axes, colors = [winner, loser], [winner_color, loser_color]
    else:
        axes, colors = [loser, winner], [loser_color, winner_color]
    return {"axes": axes, "axis_colors": colors, "dots": dots, "palette": PALETTES[int(seed) % len(PALETTES)]}


def _axis_endpoints(axis: dict[str, float], distance: float = 330) -> tuple[tuple[float, float], tuple[float, float]]:
    angle = math.radians(axis["angle"])
    ux, uy = math.cos(angle), math.sin(angle)
    return (
        (axis["cx"] - distance * ux, axis["cy"] - distance * uy),
        (axis["cx"] + distance * ux, axis["cy"] + distance * uy),
    )


def _dashed_line(draw: ImageDraw.ImageDraw, axis: dict[str, float], fill: str) -> None:
    angle = math.radians(axis["angle"])
    ux, uy = math.cos(angle), math.sin(angle)
    for start in range(-330, 330, 30):
        end = min(start + 18, 330)
        p1 = ((axis["cx"] + start * ux) * SCALE, (axis["cy"] + start * uy) * SCALE)
        p2 = ((axis["cx"] + end * ux) * SCALE, (axis["cy"] + end * uy) * SCALE)
        draw.line([p1, p2], fill=fill, width=5 * SCALE)


def render(scene: dict[str, Any]) -> Image.Image:
    axes = scene["axes"]
    axis_colors = scene["axis_colors"]
    dots = scene["dots"]
    palette = scene["palette"]
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), palette["background"])
    draw = ImageDraw.Draw(image)
    for axis, color_name in zip(axes, axis_colors):
        _dashed_line(draw, axis, COLORS[color_name])
    for x, y, radius in dots:
        box = ((x - radius) * SCALE, (y - radius) * SCALE, (x + radius) * SCALE, (y + radius) * SCALE)
        draw.ellipse(box, fill=palette["dot"])
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    axes = scene["axes"]
    axis_colors = scene["axis_colors"]
    dots = scene["dots"]
    errors = [_set_error(dots, axis) for axis in axes]
    return str(axis_colors[min(range(len(errors)), key=errors.__getitem__)])


def margin(scene: dict[str, Any]) -> float:
    axes = scene["axes"]
    _ = scene["axis_colors"]
    dots = scene["dots"]
    errors = sorted(_set_error(dots, axis) for axis in axes)
    return float(errors[1] - errors[0])


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict]]:
    return []
