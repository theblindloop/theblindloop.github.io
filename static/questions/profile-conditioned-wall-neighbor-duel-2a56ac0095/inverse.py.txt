"""Pixel-only inverse arm for the wall-neighbour duel.

Receives a rendered RGB image and nothing else.  It segments dark wall ink from
region fill, labels the connected areas of non-wall pixels, discards the face the
image border can reach (the paper outside the plate), finds the red and the blue
face by mean colour, and counts, for each of them, how many other faces share a
stretch of wall with it.  A shared stretch shows up as a contact band of
overlapping dilations whose area is far above what a bare corner crossing can
produce.  No renderer, prompt, generator, verifier, scene, or gold input is used.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

WALL_LEVEL = 90  # a wall pixel is dark in every channel
DILATE = 6  # half-width of the square dilation used to find contact bands
CONTACT_AREA = 90  # px^2: above a corner crossing, below the thinnest real stretch
MIN_FACE_AREA = 300
ABSTAIN = "abstain"


def _flood(seed: np.ndarray, allowed: np.ndarray) -> np.ndarray:
    """4-connected reachability inside `allowed`, seeded by `seed`."""
    current = seed & allowed
    height, width = allowed.shape
    while True:
        before = int(current.sum())
        for col in range(1, width):
            current[:, col] |= current[:, col - 1] & allowed[:, col]
        for col in range(width - 2, -1, -1):
            current[:, col] |= current[:, col + 1] & allowed[:, col]
        for row in range(1, height):
            current[row] |= current[row - 1] & allowed[row]
        for row in range(height - 2, -1, -1):
            current[row] |= current[row + 1] & allowed[row]
        if int(current.sum()) == before:
            return current


def _dilate(mask: np.ndarray, half: int) -> np.ndarray:
    out = mask.copy()
    for axis in (0, 1):
        grown = out.copy()
        for shift in range(1, half + 1):
            forward = np.zeros_like(out)
            backward = np.zeros_like(out)
            if axis == 0:
                forward[:-shift] = out[shift:]
                backward[shift:] = out[:-shift]
            else:
                forward[:, :-shift] = out[:, shift:]
                backward[:, shift:] = out[:, :-shift]
            grown |= forward | backward
        out = grown
    return out


def _faces(free: np.ndarray) -> list[np.ndarray]:
    remaining = free.copy()
    faces: list[np.ndarray] = []
    while True:
        hits = np.argwhere(remaining)
        if hits.size == 0:
            return faces
        seed = np.zeros_like(remaining)
        seed[hits[0][0], hits[0][1]] = True
        blob = _flood(seed, remaining)
        if int(blob.sum()) >= MIN_FACE_AREA:
            faces.append(blob)
        remaining &= ~blob


def _touches_border(mask: np.ndarray) -> bool:
    return bool(mask[0].any() or mask[-1].any() or mask[:, 0].any() or mask[:, -1].any())


def _mean_rgb(array: np.ndarray, mask: np.ndarray) -> tuple[float, float, float]:
    picked = array[mask]
    values = picked.mean(axis=0)
    return float(values[0]), float(values[1]), float(values[2])


def _degree(target: np.ndarray, others: list[np.ndarray]) -> int:
    grown = _dilate(target, DILATE)
    count = 0
    for face in others:
        band = grown & _dilate(face, DILATE)
        if int(band.sum()) >= CONTACT_AREA:
            count += 1
    return count


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB")).astype(np.int16)
    wall = array.max(axis=2) < WALL_LEVEL
    faces = [face for face in _faces(~wall) if not _touches_border(face)]
    if len(faces) < 3:
        return ABSTAIN

    red_face = None
    blue_face = None
    for face in faces:
        red, green, blue = _mean_rgb(array, face)
        if red > blue + 40 and red > green + 40:
            if red_face is not None:
                return ABSTAIN
            red_face = face
        elif blue > red + 40 and blue > green + 40:
            if blue_face is not None:
                return ABSTAIN
            blue_face = face
    if red_face is None or blue_face is None:
        return ABSTAIN

    red_degree = _degree(red_face, [face for face in faces if face is not red_face])
    blue_degree = _degree(blue_face, [face for face in faces if face is not blue_face])
    if red_degree > blue_degree:
        return "red"
    if blue_degree > red_degree:
        return "blue"
    return "same"
