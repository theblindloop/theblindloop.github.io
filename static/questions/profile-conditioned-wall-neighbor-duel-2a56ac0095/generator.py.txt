"""Wall-neighbour duel: a walled plate, two shaded regions, whose region borders more regions?

A rectangular plate is divided into regions by black walls.  Every region is a
rook-connected block of lattice cells; walls are painted on exactly those cell
boundaries whose two cells belong to different regions, so each region renders as
one single-coloured area.  Two regions are shaded (one red, one blue); the rest
stay paper white.  The decision is which shaded region shares a stretch of wall
with more other regions of the plate.  Pixel areas of the two shaded regions are
held close so that size cannot predict the answer.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "wall_neighbor_duel-renderer-0.1.0"

PLATE_W = 480
PLATE_H = 400
BORDER = 24
WALL = 7  # painted wall thickness in px (odd, centred on the lattice line)
HALF = WALL // 2

BACKGROUND = (255, 255, 255)
WALL_COLOR = (0, 0, 0)
PLAIN = (255, 255, 255)
RED = (206, 74, 74)
BLUE = (66, 104, 208)

MIN_SPAN = 26  # thinnest lattice span, hence the thinnest shared wall stretch
SAFE_SPAN = 34
QUARANTINE_MARGIN = 16.0
CLASSES = ("red", "blue", "same")


# --------------------------------------------------------------------------- #
# lattice partition sampling
# --------------------------------------------------------------------------- #
def _split(total: int, parts: int, minimum: int, rng: random.Random) -> list[int]:
    """Split `total` px into `parts` spans, each at least `minimum` px."""
    slack = total - parts * minimum
    if slack < 0:
        raise ValueError("lattice cannot satisfy the minimum span")
    cuts = sorted(rng.randint(0, slack) for _ in range(parts - 1))
    spans: list[int] = []
    previous = 0
    for cut in cuts + [slack]:
        spans.append(minimum + cut - previous)
        previous = cut
    return spans


def _grow(ncols: int, nrows: int, count: int, rng: random.Random) -> list[list[int]] | None:
    """Multi-source random growth into `count` rook-connected polyomino regions."""
    cells = [(r, c) for r in range(nrows) for c in range(ncols)]
    seeds = rng.sample(cells, count)
    grid = [[-1] * ncols for _ in range(nrows)]
    for region, (row, col) in enumerate(seeds):
        grid[row][col] = region
    unassigned = len(cells) - count
    while unassigned:
        moves = []
        for row in range(nrows):
            for col in range(ncols):
                if grid[row][col] != -1:
                    continue
                neighbours = {
                    grid[row + dr][col + dc]
                    for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1))
                    if 0 <= row + dr < nrows
                    and 0 <= col + dc < ncols
                    and grid[row + dr][col + dc] != -1
                }
                for region in sorted(neighbours):
                    moves.append((row, col, region))
        if not moves:
            return None
        row, col, region = moves[rng.randrange(len(moves))]
        grid[row][col] = region
        unassigned -= 1
    return grid


def _contacts(scene: dict) -> dict[tuple[int, int], int]:
    """Shared wall stretch in px for every pair of regions that meet along an edge."""
    grid = scene["grid"]
    widths = scene["col_widths"]
    heights = scene["row_heights"]
    shared: dict[tuple[int, int], int] = {}
    for row, line in enumerate(grid):
        for col, region in enumerate(line):
            if col + 1 < len(line) and line[col + 1] != region:
                key = (min(region, line[col + 1]), max(region, line[col + 1]))
                shared[key] = shared.get(key, 0) + heights[row]
            if row + 1 < len(grid) and grid[row + 1][col] != region:
                other = grid[row + 1][col]
                key = (min(region, other), max(region, other))
                shared[key] = shared.get(key, 0) + widths[col]
    return shared


def _neighbours(scene: dict, region: int) -> set[int]:
    found: set[int] = set()
    for (left, right), length in _contacts(scene).items():
        if length <= 0:
            continue
        if left == region:
            found.add(right)
        elif right == region:
            found.add(left)
    return found


def _areas(scene: dict) -> dict[int, int]:
    widths = scene["col_widths"]
    heights = scene["row_heights"]
    area: dict[int, int] = {}
    for row, line in enumerate(scene["grid"]):
        for col, region in enumerate(line):
            area[region] = area.get(region, 0) + widths[col] * heights[row]
    return area


def _attempt(rng: random.Random, wanted: str, thin: bool) -> dict | None:
    ncols = rng.choice((6, 7))
    nrows = rng.choice((5, 6))
    minimum = MIN_SPAN if thin else SAFE_SPAN
    widths = _split(PLATE_W, ncols, minimum, rng)
    heights = _split(PLATE_H, nrows, minimum, rng)
    grid = _grow(ncols, nrows, rng.randint(8, 11), rng)
    if grid is None:
        return None

    scene = {
        "canvas": [PLATE_W + 2 * BORDER, PLATE_H + 2 * BORDER],
        "border": BORDER,
        "wall_thickness": WALL,
        "col_widths": widths,
        "row_heights": heights,
        "grid": grid,
        "region_count": max(max(line) for line in grid) + 1,
        "red_region": 0,
        "blue_region": 1,
    }
    area = _areas(scene)
    degree = {region: len(_neighbours(scene, region)) for region in area}

    pairs = []
    for first in sorted(area):
        for second in sorted(area):
            if first >= second:
                continue
            if min(degree[first], degree[second]) < 2:
                continue
            mean = (area[first] + area[second]) / 2.0
            if abs(area[first] - area[second]) > 0.12 * mean:
                continue  # visible size must not predict the answer
            gap = degree[first] - degree[second]
            if wanted == "same" and gap != 0:
                continue
            if wanted != "same" and not 1 <= abs(gap) <= 3:
                continue
            pairs.append((first, second, gap))
    if not pairs:
        return None

    first, second, gap = pairs[rng.randrange(len(pairs))]
    if wanted == "same":
        red, blue = (first, second) if rng.random() < 0.5 else (second, first)
    elif wanted == "red":
        red, blue = (first, second) if gap > 0 else (second, first)
    else:
        red, blue = (second, first) if gap > 0 else (first, second)
    scene["red_region"] = red
    scene["blue_region"] = blue

    # every counted contact of a shaded region must stay clear of a corner touch
    lengths = [
        length
        for (left, right), length in _contacts(scene).items()
        if red in (left, right) or blue in (left, right)
    ]
    if not lengths or min(lengths) < MIN_SPAN:
        return None
    return scene


def sample_scene(seed: int) -> dict:
    """Deterministic balanced scene: the target class cycles with the seed."""
    wanted = CLASSES[int(seed) % 3]
    for attempt in range(400):
        rng = random.Random((int(seed) * 1000003) ^ (attempt * 7919 + 13))
        thin = rng.random() < 0.35
        scene = _attempt(rng, wanted, thin)
        if scene is not None:
            return scene
    raise RuntimeError(f"no balanced scene for seed {seed}")


# --------------------------------------------------------------------------- #
# rendering
# --------------------------------------------------------------------------- #
def _bounds(scene: dict) -> tuple[list[int], list[int]]:
    xs = [scene["border"]]
    for width in scene["col_widths"]:
        xs.append(xs[-1] + width)
    ys = [scene["border"]]
    for height in scene["row_heights"]:
        ys.append(ys[-1] + height)
    return xs, ys


def render(scene: dict) -> Image.Image:
    width, height = scene["canvas"]
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    xs, ys = _bounds(scene)
    grid = scene["grid"]
    half = scene["wall_thickness"] // 2

    for row, line in enumerate(grid):
        for col, region in enumerate(line):
            if region == scene["red_region"]:
                fill = RED
            elif region == scene["blue_region"]:
                fill = BLUE
            else:
                fill = PLAIN
            draw.rectangle([xs[col], ys[row], xs[col + 1] - 1, ys[row + 1] - 1], fill=fill)

    # plate outline
    draw.rectangle([xs[0] - half, ys[0] - half, xs[-1] - 1 + half, ys[0] - 1 + half], fill=WALL_COLOR)
    draw.rectangle([xs[0] - half, ys[-1] - half, xs[-1] - 1 + half, ys[-1] - 1 + half], fill=WALL_COLOR)
    draw.rectangle([xs[0] - half, ys[0] - half, xs[0] - 1 + half, ys[-1] - 1 + half], fill=WALL_COLOR)
    draw.rectangle([xs[-1] - half, ys[0] - half, xs[-1] - 1 + half, ys[-1] - 1 + half], fill=WALL_COLOR)

    # interior walls only between different regions
    for row, line in enumerate(grid):
        for col, region in enumerate(line):
            if col + 1 < len(line) and line[col + 1] != region:
                draw.rectangle(
                    [xs[col + 1] - half, ys[row], xs[col + 1] - 1 + half, ys[row + 1] - 1],
                    fill=WALL_COLOR,
                )
            if row + 1 < len(grid) and grid[row + 1][col] != region:
                draw.rectangle(
                    [xs[col], ys[row + 1] - half, xs[col + 1] - 1, ys[row + 1] - 1 + half],
                    fill=WALL_COLOR,
                )
    # seal every lattice crossing that separates different regions
    for row in range(1, len(grid)):
        for col in range(1, len(grid[0])):
            block = {
                grid[row - 1][col - 1],
                grid[row - 1][col],
                grid[row][col - 1],
                grid[row][col],
            }
            if len(block) > 1:
                draw.rectangle(
                    [xs[col] - half, ys[row] - half, xs[col] - 1 + half, ys[row] - 1 + half],
                    fill=WALL_COLOR,
                )
    return image


# --------------------------------------------------------------------------- #
# gold, margin, quarantine, symmetries
# --------------------------------------------------------------------------- #
def analytic_gold(scene: dict) -> str:
    """Which shaded region shares a wall stretch with more other plate regions."""
    red = len(_neighbours(scene, scene["red_region"]))
    blue = len(_neighbours(scene, scene["blue_region"]))
    if red > blue:
        return "red"
    if blue > red:
        return "blue"
    return "same"


def margin(scene: dict) -> float:
    """Shortest counted wall stretch of either shaded region, less a corner crossing."""
    red = scene["red_region"]
    blue = scene["blue_region"]
    lengths = [
        length
        for (left, right), length in _contacts(scene).items()
        if length > 0 and (red in (left, right) or blue in (left, right))
    ]
    return float(min(lengths) - 2 * scene["wall_thickness"])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical relabelings of the region bookkeeping that must preserve gold."""
    twins: list[tuple[str, dict]] = []
    count = scene["region_count"]

    rotation = {region: (region + 1) % count for region in range(count)}
    twin = dict(scene)
    twin["grid"] = [[rotation[region] for region in line] for line in scene["grid"]]
    twin["red_region"] = rotation[scene["red_region"]]
    twin["blue_region"] = rotation[scene["blue_region"]]
    twins.append(("region_id_rotation", twin))

    reversed_ids = dict(scene)
    reversed_ids["grid"] = [[count - 1 - region for region in line] for line in scene["grid"]]
    reversed_ids["red_region"] = count - 1 - scene["red_region"]
    reversed_ids["blue_region"] = count - 1 - scene["blue_region"]
    twins.append(("region_id_reversal", reversed_ids))
    return twins
