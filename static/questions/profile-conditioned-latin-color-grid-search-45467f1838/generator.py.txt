"""Deterministic renderer for the Latin color-grid visual search world."""

from __future__ import annotations

import copy
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH = 760
HEIGHT = 760
SCALE = 2
GRID_SIZE = 240
CELL_SIZE = GRID_SIZE // 4
BASE_POSITIONS = ((70, 95), (450, 95), (70, 440), (450, 440))
LABELS = ("A", "B", "C", "D")
VISIBLE_COLORS = ("#D1495B", "#3066BE", "#2A9D6F", "#E9A23B")
INK = "#263238"
BACKGROUND = "#FAFBFC"


def _latin(rng: random.Random) -> list[list[int]]:
    """Return a randomized order-four Latin square."""
    rows = list(range(4))
    cols = list(range(4))
    symbols = list(range(4))
    rng.shuffle(rows)
    rng.shuffle(cols)
    rng.shuffle(symbols)
    return [[symbols[(rows[r] + cols[c]) % 4] for c in range(4)] for r in range(4)]


def _invalid_latin(rng: random.Random, axis: str) -> list[list[int]]:
    """Preserve global inventory but break exactly two orthogonal lines."""
    panel = _latin(rng)
    if axis == "row":
        row = rng.randrange(4)
        c1, c2 = rng.sample(range(4), 2)
        panel[row][c1], panel[row][c2] = panel[row][c2], panel[row][c1]
    else:
        col = rng.randrange(4)
        r1, r2 = rng.sample(range(4), 2)
        panel[r1][col], panel[r2][col] = panel[r2][col], panel[r1][col]
    return panel


def _score_panel(panel: Iterable[Iterable[int]]) -> int:
    grid = [list(row) for row in panel]
    wanted = {0, 1, 2, 3}
    row_score = sum(set(row) == wanted for row in grid)
    col_score = sum({grid[r][c] for r in range(4)} == wanted for c in range(4))
    return int(row_score + col_score)


def _panel_scores(scene: dict) -> list[int]:
    return [_score_panel(panel) for panel in scene["panels"]]


def sample_scene(seed: int) -> dict:
    """Sample four balanced grids with a position-balanced unique winner."""
    rng = random.Random(int(seed))
    winner = int(seed) % 4
    panels: list[list[list[int]]] = []
    seen: set[tuple[tuple[int, ...], ...]] = set()
    for index in range(4):
        while True:
            if index == winner:
                panel = _latin(rng)
            else:
                # Alternating defect axes prevent one repeated spatial cue.
                panel = _invalid_latin(rng, "row" if (index + seed) % 2 else "column")
            frozen = tuple(tuple(row) for row in panel)
            if frozen not in seen:
                seen.add(frozen)
                panels.append(panel)
                break

    palette = list(VISIBLE_COLORS)
    rng.shuffle(palette)
    offsets = [[rng.randint(-7, 7), rng.randint(-7, 7)] for _ in range(4)]
    radius = rng.randint(14, 17)
    scene = {
        "panels": panels,
        "palette": palette,
        "panel_offsets": offsets,
        "disk_radius": radius,
    }
    assert _panel_scores(scene).count(8) == 1
    assert sorted(_panel_scores(scene)) == [6, 6, 6, 8]
    return scene


def analytic_gold(scene: dict) -> str:
    scores = _panel_scores(scene)
    best = max(scores)
    winners = [index for index, value in enumerate(scores) if value == best]
    if best != 8 or len(winners) != 1:
        raise ValueError("scene lacks a unique all-different grid")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    scores = sorted(_panel_scores(scene), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    scores = _panel_scores(scene)
    return scores.count(8) != 1 or margin(scene) < 2.0


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size
    )


def render(scene: dict) -> Image.Image:
    """Render from latent arrays; no analytic answer is consulted."""
    panels = scene["panels"]
    palette = scene["palette"]
    offsets = scene["panel_offsets"]
    radius = int(scene["disk_radius"])
    if len(panels) != 4 or len(palette) != 4 or len(offsets) != 4:
        raise ValueError("scene must contain four panels, four colors, and four offsets")

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    label_font = _font(28 * SCALE)
    for index, panel in enumerate(panels):
        if len(panel) != 4 or any(len(row) != 4 for row in panel):
            raise ValueError("every panel must be 4 by 4")
        dx, dy = map(int, offsets[index])
        gx = (BASE_POSITIONS[index][0] + dx) * SCALE
        gy = (BASE_POSITIONS[index][1] + dy) * SCALE
        extent = GRID_SIZE * SCALE
        line_width = 3 * SCALE

        # The visible frame allows the inverse arm to discover panel regions.
        for step in range(5):
            x = gx + step * CELL_SIZE * SCALE
            y = gy + step * CELL_SIZE * SCALE
            draw.line((x, gy, x, gy + extent), fill=INK, width=line_width)
            draw.line((gx, y, gx + extent, y), fill=INK, width=line_width)

        for row in range(4):
            for col in range(4):
                token = int(panel[row][col])
                cx = gx + (col * CELL_SIZE + CELL_SIZE / 2) * SCALE
                cy = gy + (row * CELL_SIZE + CELL_SIZE / 2) * SCALE
                rr = radius * SCALE
                draw.ellipse((cx - rr, cy - rr, cx + rr, cy + rr), fill=palette[token])

        label = LABELS[index]
        box = draw.textbbox((0, 0), label, font=label_font)
        tw = box[2] - box[0]
        draw.text(
            (gx + extent / 2 - tw / 2, gy - 43 * SCALE),
            label,
            fill=INK,
            font=label_font,
        )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Return a genuine representational alias with pixel-identical output.

    Token identifiers have no visible identity of their own. Relabeling every
    token while moving each color to its new identifier leaves every rendered
    disk unchanged and preserves every equality/all-different relation.
    """
    permutation = (1, 3, 0, 2)  # old token -> new token
    twin = copy.deepcopy(scene)
    twin["panels"] = [
        [[permutation[int(value)] for value in row] for row in panel]
        for panel in scene["panels"]
    ]
    new_palette = [""] * 4
    for old, new in enumerate(permutation):
        new_palette[new] = scene["palette"][old]
    twin["palette"] = new_palette
    return [("global_token_relabel", twin)]
