"""Independent pixel-only inverse for the Latin color-grid search world.

The implementation knows the public visual vocabulary (four ink colors and
four labels) but receives only a PIL image. It discovers the four grid frames
as connected dark components, classifies each cell-center patch by visible
color, evaluates every row and column, and requires a unique winner margin.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")
PALETTE = np.array(
    ((209, 73, 91), (48, 102, 190), (42, 157, 111), (233, 162, 59)),
    dtype=np.float64,
)


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    """Return (area, x0, y0, x1, y1) for 8-connected true components."""
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components: list[tuple[int, int, int, int, int]] = []
    for sy, sx in np.argwhere(mask):
        sy = int(sy)
        sx = int(sx)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        area = 0
        x0 = x1 = sx
        y0 = y1 = sy
        while queue:
            y, x = queue.popleft()
            area += 1
            x0 = min(x0, x)
            x1 = max(x1, x)
            y0 = min(y0, y)
            y1 = max(y1, y)
            for yy in range(max(0, y - 1), min(height, y + 2)):
                for xx in range(max(0, x - 1), min(width, x + 2)):
                    if mask[yy, xx] and not seen[yy, xx]:
                        seen[yy, xx] = True
                        queue.append((yy, xx))
        components.append((area, x0, y0, x1, y1))
    return components


def _grid_boxes(array: np.ndarray) -> list[tuple[int, int, int, int]]:
    dark = (array[:, :, 0] < 90) & (array[:, :, 1] < 92) & (array[:, :, 2] < 105)
    boxes = []
    for area, x0, y0, x1, y1 in _components(dark):
        width = x1 - x0 + 1
        height = y1 - y0 + 1
        if area > 2500 and 225 <= width <= 255 and 225 <= height <= 255:
            boxes.append((x0, y0, x1, y1))
    if len(boxes) != 4:
        raise ValueError(f"expected four visible grid frames, recovered {len(boxes)}")

    # Infer the two rows from geometry, then use left-to-right order. No latent
    # coordinate or renderer state is consulted.
    by_vertical = sorted(boxes, key=lambda box: ((box[1] + box[3]) / 2, box[0]))
    top = sorted(by_vertical[:2], key=lambda box: box[0])
    bottom = sorted(by_vertical[2:], key=lambda box: box[0])
    if max(box[3] for box in top) >= min(box[1] for box in bottom):
        raise ValueError("grid rows overlap or cannot be ordered")
    return top + bottom


def _classify_panel(array: np.ndarray, box: tuple[int, int, int, int]) -> list[list[int]]:
    x0, y0, x1, y1 = box
    span_x = x1 - x0
    span_y = y1 - y0
    if abs(span_x - span_y) > 5:
        raise ValueError("grid frame is not square enough")
    grid: list[list[int]] = []
    for row in range(4):
        values = []
        for col in range(4):
            cx = int(round(x0 + (col + 0.5) * span_x / 4.0))
            cy = int(round(y0 + (row + 0.5) * span_y / 4.0))
            patch = array[cy - 5 : cy + 6, cx - 5 : cx + 6].astype(np.float64)
            if patch.shape[:2] != (11, 11):
                raise ValueError("cell center lies outside the raster")
            mean = patch.reshape(-1, 3).mean(axis=0)
            distances = np.linalg.norm(PALETTE - mean, axis=1)
            nearest = int(distances.argmin())
            ordered = np.sort(distances)
            if ordered[0] > 20.0 or ordered[1] - ordered[0] < 45.0:
                raise ValueError("cell color is missing or ambiguous")
            values.append(nearest)
        grid.append(values)
    counts = np.bincount(np.asarray(grid).ravel(), minlength=4)
    if counts.tolist() != [4, 4, 4, 4]:
        raise ValueError(f"panel color inventory is corrupted: {counts.tolist()}")
    return grid


def _score(grid: list[list[int]]) -> int:
    wanted = {0, 1, 2, 3}
    rows = sum(set(row) == wanted for row in grid)
    cols = sum({grid[row][col] for row in range(4)} == wanted for col in range(4))
    return int(rows + cols)


def decision_from_image(image: Image.Image) -> str:
    """Recover A/B/C/D strictly from visible raster structure."""
    array = np.asarray(image.convert("RGB"))
    if array.ndim != 3 or array.shape[0] < 300 or array.shape[1] < 300:
        raise ValueError("image is too small for four grids")
    boxes = _grid_boxes(array)
    scores = [_score(_classify_panel(array, box)) for box in boxes]
    order = sorted(range(4), key=lambda index: scores[index], reverse=True)
    if scores[order[0]] != 8 or scores[order[0]] - scores[order[1]] < 2:
        raise ValueError(f"no unique all-different winner with margin 2: {scores}")
    return LABELS[order[0]]
