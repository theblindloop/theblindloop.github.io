"""Renderer and analytic scorer for the orientation-consensus search world."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont


WIDTH = 720
HEIGHT = 720
SUPERSAMPLE = 4
BACKGROUND = (255, 255, 255)
PANEL_FILL = (248, 250, 252)
PANEL_EDGE = (204, 213, 222)
LABEL_INK = (112, 124, 138)
BAR_INK = (24, 32, 44)
BAR_LENGTH = 96.0
BAR_WIDTH = 16.0

# The four equal panel rectangles are part of the public visual contract.  The
# inverse does not use these coordinates: it recovers panels from bar centroids.
PANEL_BOXES: tuple[tuple[int, int, int, int], ...] = (
    (28, 52, 330, 290),
    (362, 52, 330, 290),
    (28, 378, 330, 290),
    (362, 378, 330, 290),
)
PANEL_LABELS = ("A", "B", "C", "D")
SLOT_CENTERS: tuple[tuple[int, int], ...] = (
    (80, 92),
    (250, 92),
    (80, 232),
    (250, 232),
)


def _undirected_delta(first: float, second: float) -> float:
    """Return the smaller angular separation for line orientations."""

    raw = abs((float(first) - float(second)) % 180.0)
    return min(raw, 180.0 - raw)


def _angular_spread(angles: list[float]) -> float:
    if len(angles) != 4:
        raise ValueError("each panel must contain exactly four bars")
    return max(
        _undirected_delta(angles[left], angles[right])
        for left in range(len(angles))
        for right in range(left)
    )


def _panel_spread(panel: dict[str, Any]) -> float:
    angles = [float(bar["angle"]) for bar in panel["bars"]]
    return _angular_spread(angles)


def _panel_spreads(panels: list[dict[str, Any]]) -> list[float]:
    if len(panels) != 4:
        raise ValueError("scene must contain exactly four panels")
    return [_panel_spread(panel) for panel in panels]


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one balanced scene from a deterministic integer seed.

    The answer label cycles with the seed while all geometric nuisance values
    come from an independent mixed random stream.  A small fraction of scenes
    sit inside the declared near-parallel quarantine band.
    """

    # The arithmetic cycle balances panel identity in sequential generation;
    # the mixed stream keeps geometry independent of that answer cycle.
    target_panel = int(seed) % 4
    rng = random.Random((int(seed) ^ 0x6D2B79F5) & 0xFFFFFFFF)
    base_angle = rng.uniform(0.0, 180.0)
    if rng.random() < 0.18:
        separation = rng.uniform(22.0, 28.0)
    else:
        separation = rng.uniform(36.0, 58.0)
    direction = -1.0 if rng.randrange(2) == 0 else 1.0
    odd_angle = (base_angle + direction * separation) % 180.0
    odd_slot = rng.randrange(len(SLOT_CENTERS))

    panels: list[dict[str, Any]] = []
    for panel_index in range(4):
        bars = []
        for slot_index, center in enumerate(SLOT_CENTERS):
            angle = odd_angle if (panel_index == target_panel and slot_index == odd_slot) else base_angle
            bars.append(
                {
                    "center": [int(center[0]), int(center[1])],
                    "angle": round(float(angle), 6),
                }
            )
        # Record order is deliberately nuisance-only and independent of panel
        # identity.  Rendering is non-overlapping, so reversal is a true alias.
        rng.shuffle(bars)
        panels.append({"bars": bars})
    return {"panels": panels}


def _bar_polygon(center: tuple[float, float], angle: float) -> list[tuple[float, float]]:
    theta = math.radians(float(angle))
    ux, uy = math.cos(theta), math.sin(theta)
    nx, ny = -uy, ux
    half_length = BAR_LENGTH / 2.0
    half_width = BAR_WIDTH / 2.0
    cx, cy = center
    along = (ux * half_length, uy * half_length)
    across = (nx * half_width, ny * half_width)
    return [
        (cx - along[0] - across[0], cy - along[1] - across[1]),
        (cx + along[0] - across[0], cy + along[1] - across[1]),
        (cx + along[0] + across[0], cy + along[1] + across[1]),
        (cx - along[0] + across[0], cy - along[1] + across[1]),
    ]


def _label_font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the public scene as a deterministic antialiased RGB raster."""

    scale = SUPERSAMPLE
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    label_font = _label_font(18 * scale)

    for panel_index, (x0, y0, panel_width, panel_height) in enumerate(PANEL_BOXES):
        box = (x0 * scale, y0 * scale, (x0 + panel_width) * scale, (y0 + panel_height) * scale)
        draw.rectangle(box, fill=PANEL_FILL, outline=PANEL_EDGE, width=2 * scale)
        draw.text(
            ((x0 + 14) * scale, (y0 + 10) * scale),
            PANEL_LABELS[panel_index],
            font=label_font,
            fill=LABEL_INK,
        )
        for bar in scene["panels"][panel_index]["bars"]:
            center = (x0 + float(bar["center"][0]), y0 + float(bar["center"][1]))
            polygon = _bar_polygon(center, float(bar["angle"]))
            draw.polygon(
                [(round(x * scale), round(y * scale)) for x, y in polygon],
                fill=BAR_INK,
            )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    """Select the panel with the unique largest within-panel angular spread."""

    # Keep the scene boundary explicit here.  The protocol's rendered-field
    # audit can then see that analytic gold reads only the visible `panels`
    # field; helper arguments below are ordinary bar-angle lists, not latent
    # scene fields.
    panels = scene["panels"]
    spreads = [
        _angular_spread([float(bar["angle"]) for bar in panel["bars"]])
        for panel in panels
    ]
    order = sorted(range(4), key=lambda index: spreads[index], reverse=True)
    if spreads[order[0]] - spreads[order[1]] <= 1e-9:
        raise ValueError("panel spread is not uniquely maximal")
    return PANEL_LABELS[order[0]]


def margin(scene: dict[str, Any]) -> float:
    """Return the top-versus-runner-up angular spread gap in degrees."""

    spreads = sorted(_panel_spreads(scene["panels"]), reverse=True)
    return float(spreads[0] - spreads[1])


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Exclude near-parallel oddballs whose raster margin is deliberately small."""

    return margin(scene) < 30.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """Return a visible-field-preserving reversal of every bar record list."""

    twin = {
        "panels": [
            {"bars": [dict(bar) for bar in reversed(panel["bars"])]}
            for panel in scene["panels"]
        ]
    }
    return [("within_panel_record_reversal", twin)]
