"""Independent image-only inverse for the orientation-consensus world."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image


_MIN_COMPONENT_PIXELS = 500
_PANEL_LABELS = ("A", "B", "C", "D")


def _dark_components(image: Image.Image) -> list[np.ndarray]:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    # The public bars are near-black; panel borders and labels are deliberately
    # lighter.  No renderer state or scene object enters this measurement.
    mask = (
        (array[:, :, 0] < 78)
        & (array[:, :, 1] < 88)
        & (array[:, :, 2] < 102)
    )
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components: list[np.ndarray] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue: deque[tuple[int, int]] = deque([(int(y), int(x))])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in (
                (cy - 1, cx),
                (cy + 1, cx),
                (cy, cx - 1),
                (cy, cx + 1),
            ):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= _MIN_COMPONENT_PIXELS:
            component = np.asarray(points, dtype=np.float64)
            y_extent = float(component[:, 0].max() - component[:, 0].min() + 1.0)
            x_extent = float(component[:, 1].max() - component[:, 1].min() + 1.0)
            # A rotated rectangle can have a nearly square axis-aligned box,
            # so component aspect is not used as a gate.  The size threshold
            # and the PCA eigenvalue check below separate bars from text.
            if max(x_extent, y_extent) >= 30.0:
                components.append(component)
    return components


def _orientation(component: np.ndarray) -> float:
    xy = component[:, [1, 0]]
    centered = xy - xy.mean(axis=0, keepdims=True)
    covariance = centered.T @ centered / max(1, len(centered) - 1)
    eigenvalues, eigenvectors = np.linalg.eigh(covariance)
    if float(eigenvalues[-1]) <= float(eigenvalues[0]) * 1.5:
        raise ValueError("bar component has no stable long axis")
    direction = eigenvectors[:, int(np.argmax(eigenvalues))]
    return math.degrees(math.atan2(float(direction[1]), float(direction[0]))) % 180.0


def _line_delta(first: float, second: float) -> float:
    raw = abs((first - second) % 180.0)
    return min(raw, 180.0 - raw)


def _spread(angles: list[float]) -> float:
    return max(
        _line_delta(angles[left], angles[right])
        for left in range(len(angles))
        for right in range(left)
    )


def decision_from_image(image: Image.Image) -> str:
    """Recover the selected panel from the final PNG alone."""

    height, width = image.height, image.width
    components = _dark_components(image)
    if len(components) != 16:
        raise ValueError(f"expected 16 dark bar components, found {len(components)}")

    grouped: list[list[float]] = [[] for _ in range(4)]
    for component in components:
        centroid_y = float(component[:, 0].mean())
        centroid_x = float(component[:, 1].mean())
        row = 0 if centroid_y < height / 2.0 else 1
        column = 0 if centroid_x < width / 2.0 else 1
        panel_index = row * 2 + column
        grouped[panel_index].append(_orientation(component))
    if any(len(angles) != 4 for angles in grouped):
        raise ValueError(f"bar components did not partition into four panels: {[len(x) for x in grouped]}")

    spreads = [_spread(angles) for angles in grouped]
    order = sorted(range(4), key=lambda index: spreads[index], reverse=True)
    if spreads[order[0]] - spreads[order[1]] < 4.0:
        raise ValueError("pixel angular spread is ambiguous")
    return _PANEL_LABELS[order[0]]
