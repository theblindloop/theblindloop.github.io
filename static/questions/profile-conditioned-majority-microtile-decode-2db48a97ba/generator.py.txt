"""Deterministic Pillow world for local 3x3 majority reduction."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

SIZE = 620
N = 8
MICRO = 3
PITCH = 18
GRID = N * MICRO * PITCH

PALETTES = (
    {"ink": [20, 25, 34], "paper": [248, 246, 239], "frame": [176, 42, 144]},
    {"ink": [35, 25, 22], "paper": [241, 248, 245], "frame": [24, 136, 177]},
    {"ink": [18, 36, 30], "paper": [249, 243, 250], "frame": [221, 92, 28]},
    {"ink": [30, 28, 38], "paper": [245, 248, 235], "frame": [82, 72, 190]},
)


def _templates() -> dict[str, tuple[str, ...]]:
    # Each template contains exactly 20 black cells, preventing global-ink cues.
    diamond = [
        "00011000", "00100100", "01000010", "10000001",
        "10011001", "01011010", "00100100", "00011000",
    ]
    plus = [
        "00000000", "00011000", "00011000", "01111110",
        "01111110", "00011000", "00011000", "00000000",
    ]
    xmark = [
        "10000001", "01000010", "00100100", "00111100",
        "00111100", "00100100", "01000010", "10000001",
    ]
    ring = [
        "00000000", "01111110", "01000010", "01000010",
        "01000010", "01000010", "01111110", "00000000",
    ]
    result = {"diamond": diamond, "plus": plus, "X": xmark, "ring": ring}
    return {key: tuple(rows) for key, rows in result.items()}


TEMPLATES = _templates()


def _decode(microtiles: list[list[list[int]]]) -> tuple[str, ...]:
    rows: list[str] = []
    for macro_row in microtiles:
        rows.append("".join("1" if sum(tile) >= 5 else "0" for tile in macro_row))
    return tuple(rows)


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    names = tuple(TEMPLATES)
    target = names[int(seed) % len(names)]
    bitmap = TEMPLATES[target]
    microtiles: list[list[list[int]]] = []
    for row in bitmap:
        out_row = []
        for bit in row:
            count = 5 if bit == "1" else 4
            positions = list(range(9))
            rng.shuffle(positions)
            chosen = set(positions[:count])
            out_row.append([int(i in chosen) for i in range(9)])
        microtiles.append(out_row)
    palette = PALETTES[rng.randrange(len(PALETTES))]
    return {
        "microtiles": microtiles,
        "ink": palette["ink"],
        "paper": palette["paper"],
        "frame": palette["frame"],
        "offset_x": rng.randint(-28, 28),
        "offset_y": rng.randint(-28, 28),
    }


def render(scene: dict[str, Any]) -> Image.Image:
    ink = tuple(scene["ink"])
    paper = tuple(scene["paper"])
    frame = tuple(scene["frame"])
    gx = (SIZE - GRID) // 2 + int(scene["offset_x"])
    gy = (SIZE - GRID) // 2 + int(scene["offset_y"])
    image = Image.new("RGB", (SIZE, SIZE), (255, 255, 255))
    draw = ImageDraw.Draw(image)
    draw.rectangle((gx, gy, gx + GRID - 1, gy + GRID - 1), fill=paper)
    # Fine gutters delineate each microcell; heavier lines delineate 3x3 tiles.
    gutter = tuple((2 * c + 180) // 3 for c in paper)
    for r in range(N * MICRO + 1):
        y = gy + r * PITCH
        width = 3 if r % MICRO == 0 else 1
        draw.line((gx, y, gx + GRID - 1, y), fill=gutter, width=width)
    for c in range(N * MICRO + 1):
        x = gx + c * PITCH
        width = 3 if c % MICRO == 0 else 1
        draw.line((x, gy, x, gy + GRID - 1), fill=gutter, width=width)
    for mr, macro_row in enumerate(scene["microtiles"]):
        for mc, tile in enumerate(macro_row):
            for k, value in enumerate(tile):
                rr, cc = divmod(k, MICRO)
                x0 = gx + (mc * MICRO + cc) * PITCH + 3
                y0 = gy + (mr * MICRO + rr) * PITCH + 3
                draw.rectangle((x0, y0, x0 + 12, y0 + 12), fill=ink if value else paper)
    # Chromatic outer frame is the visible fiducial used by the inverse arm.
    draw.rectangle((gx - 7, gy - 7, gx + GRID + 6, gy + GRID + 6), outline=frame, width=5)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    decoded = _decode(scene["microtiles"])
    distances = {name: sum(a != b for ra, rb in zip(decoded, tmpl) for a, b in zip(ra, rb))
                 for name, tmpl in TEMPLATES.items()}
    return min(distances, key=distances.get)


def margin(scene: dict[str, Any]) -> float:
    decoded = _decode(scene["microtiles"])
    distances = sorted(sum(a != b for ra, rb in zip(decoded, tmpl) for a, b in zip(ra, rb))
                       for tmpl in TEMPLATES.values())
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        tiles = scene["microtiles"]
        if len(tiles) != N or any(len(row) != N for row in tiles):
            return True
        if any(len(tile) != 9 or sum(tile) not in (4, 5) for row in tiles for tile in row):
            return True
        return margin(scene) <= 0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
