"""Independent raster-only inverse for the majority-microtile world."""

from __future__ import annotations

import numpy as np
from PIL import Image


_GLYPHS = {
    "diamond": ("00011000", "00100100", "01000010", "10000001", "10011001", "01011010", "00100100", "00011000"),
    "plus": ("00000000", "00011000", "00011000", "01111110", "01111110", "00011000", "00011000", "00000000"),
    "X": ("10000001", "01000010", "00100100", "00111100", "00111100", "00100100", "01000010", "10000001"),
    "ring": ("00000000", "01111110", "01000010", "01000010", "01000010", "01000010", "01111110", "00000000"),
}


def _frame_bounds(rgb: np.ndarray) -> tuple[int, int, int, int]:
    hi = rgb.max(axis=2).astype(int)
    lo = rgb.min(axis=2).astype(int)
    chromatic = (hi - lo > 55) & (hi < 245) & (lo < 190)
    ys, xs = np.nonzero(chromatic)
    if len(xs) < 1000:
        raise ValueError("visible chromatic frame not found")
    return int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())


def _recover(image: Image.Image) -> tuple[tuple[str, ...], float]:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    x0, y0, x1, y1 = _frame_bounds(rgb)
    if abs((x1 - x0) - (y1 - y0)) > 2:
        raise ValueError("frame is not square")
    gx, gy = x0 + 7, y0 + 7
    span = x1 - x0 + 1 - 14
    pitch = span / 24.0
    if not (16.0 <= pitch <= 20.0):
        raise ValueError("unexpected grid scale")
    lum = rgb.astype(float) @ np.array([0.299, 0.587, 0.114])
    bits = np.zeros((8, 8), dtype=np.uint8)
    strengths = []
    for mr in range(8):
        for mc in range(8):
            micro = []
            for rr in range(3):
                for cc in range(3):
                    cx = gx + (mc * 3 + cc + 0.5) * pitch
                    cy = gy + (mr * 3 + rr + 0.5) * pitch
                    xa, xb = int(round(cx - pitch * .22)), int(round(cx + pitch * .22))
                    ya, yb = int(round(cy - pitch * .22)), int(round(cy + pitch * .22))
                    micro.append(float(np.median(lum[ya:yb + 1, xa:xb + 1])))
            # The rendered roles are deliberately separated into dark ink (<100)
            # and light paper (>230) across every palette.
            if any(105 <= value <= 210 for value in micro):
                raise ValueError("a microcell has ambiguous luminance")
            count = sum(value < 128 for value in micro)
            if count not in (4, 5):
                raise ValueError("microtile is malformed")
            bits[mr, mc] = int(count >= 5)
            strengths.append(abs(count - 4.5))
    rows = tuple("".join(str(int(v)) for v in row) for row in bits)
    return rows, min(strengths)


def decision_from_image(image: Image.Image) -> str:
    rows, _ = _recover(image)
    distances = {name: sum(a != b for ra, rb in zip(rows, tmpl) for a, b in zip(ra, rb))
                 for name, tmpl in _GLYPHS.items()}
    ordered = sorted(distances.items(), key=lambda item: (item[1], item[0]))
    if ordered[0][1] != 0 or ordered[1][1] == ordered[0][1]:
        raise ValueError("decoded raster is not an unambiguous declared symbol")
    return ordered[0][0]


def measured_margin(image: Image.Image) -> float:
    rows, _ = _recover(image)
    distances = sorted(sum(a != b for ra, rb in zip(rows, tmpl) for a, b in zip(ra, rb))
                       for tmpl in _GLYPHS.values())
    return float(distances[1] - distances[0])
