"""Deterministic renderer for the compact negative-space keystone world."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

W = H = 512
BG = (246, 244, 238)
INKS = ((28, 91, 153), (112, 62, 143), (25, 125, 105), (184, 78, 38))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    direction = "left" if int(seed) % 2 == 0 else "right"
    narrow = 8 + (int(seed) % 3)
    delta = 6 + 2 * (int(seed) % 10)
    distractors = []
    # Pale remote dots vary in number and location but are never part of the ink mask.
    for side in (-1, 1):
        if rng.random() < 0.75:
            distractors.append(
                [
                    rng.randint(45, 120) if side < 0 else rng.randint(392, 467),
                    rng.randint(60, 452),
                    rng.randint(4, 8),
                    rng.choice(((176, 180, 180), (190, 183, 167), (171, 179, 190))),
                ]
            )
    return {
        "direction": direction,
        # Four absolute positions keep the focal cue randomized while ensuring
        # its two end measurements do not straddle the same coarse support cell.
        "center_x": rng.choice((224, 288)),
        "center_y": rng.choice((224, 288)),
        "scale": round(rng.uniform(0.86, 0.94), 4),
        "half_height": rng.randint(48, 58),
        "narrow_half": narrow,
        "wide_half": narrow + delta,
        "ink": list(rng.choice(INKS)),
        "distractors": distractors,
    }


def _geometry(scene: dict) -> tuple[list[tuple[int, int]], list[tuple[int, int]], list[tuple[int, int]]]:
    cx = int(scene["center_x"])
    cy = int(scene["center_y"])
    scale = float(scene["scale"])
    outer = int(round(70 * scale))
    split = max(5, int(round(6 * scale)))
    height = int(scene["half_height"])
    narrow = int(scene["narrow_half"])
    wide = int(scene["wide_half"])
    left_half, right_half = (wide, narrow) if scene["direction"] == "left" else (narrow, wide)

    def half_at(x: int) -> int:
        t = (x - (cx - outer)) / float(2 * outer)
        return int(round(left_half + t * (right_half - left_half)))

    x0, x1 = cx - outer, cx + outer
    upper_left = [
        (x0, cy - height),
        (cx - split, cy - height),
        (cx - split, cy - half_at(cx - split)),
        (x0, cy - half_at(x0)),
    ]
    upper_right = [
        (cx + split, cy - height),
        (x1, cy - height),
        (x1, cy - half_at(x1)),
        (cx + split, cy - half_at(cx + split)),
    ]
    lower = [
        (x0, cy + half_at(x0)),
        (x1, cy + half_at(x1)),
        (x1, cy + height),
        (x0, cy + height),
    ]
    return upper_left, upper_right, lower


def render(scene: dict) -> Image.Image:
    ss = 3
    image = Image.new("RGB", (W * ss, H * ss), BG)
    draw = ImageDraw.Draw(image)
    for x, y, radius, color in scene["distractors"]:
        box = ((x - radius) * ss, (y - radius) * ss, (x + radius) * ss, (y + radius) * ss)
        draw.ellipse(box, fill=tuple(color))
    ink = tuple(scene["ink"])
    for polygon in _geometry(scene):
        draw.polygon([(x * ss, y * ss) for x, y in polygon], fill=ink)
    return image.resize((W, H), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["direction"])


def margin(scene: dict) -> float:
    return float(2 * (int(scene["wide_half"]) - int(scene["narrow_half"])))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 14.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
