"""Independent image-only inverse arm based on negative-space taper."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        pts = []
        while queue:
            yy, xx = queue.popleft()
            pts.append((yy, xx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = yy + dy, xx + dx
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(pts) >= 20:
            components.append(np.asarray(pts, dtype=np.int32))
    return components


def _ink_mask(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    high = arr.max(axis=2)
    low = arr.min(axis=2)
    saturation = high - low
    # Keystone inks are strongly chromatic; pale gray distractors and background are not.
    return (saturation > 55) & (low < 155)


def decision_from_image(image: Image.Image) -> str:
    mask = _ink_mask(image)
    if int(mask.sum()) < 2200:
        return "abstain"
    ys, xs = np.nonzero(mask)
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    if not (112 <= x1 - x0 <= 142 and 88 <= y1 - y0 <= 122):
        return "abstain"
    x_left, x_right = x0 + 8, x1 - 8

    def gap_width(x: int) -> float | None:
        widths = []
        for xx in range(x - 2, x + 3):
            ink_y = np.flatnonzero(mask[y0 : y1 + 1, xx]) + y0
            jumps = np.diff(ink_y)
            candidates = np.flatnonzero(jumps >= 6)
            if len(candidates):
                widest = candidates[int(np.argmax(jumps[candidates]))]
                widths.append(float(ink_y[widest + 1] - ink_y[widest] - 1))
        return float(np.median(widths)) if widths else None

    left_width = gap_width(x_left)
    right_width = gap_width(x_right)
    if left_width is None or right_width is None or abs(left_width - right_width) < 8:
        return "abstain"
    return "left" if left_width > right_width else "right"
