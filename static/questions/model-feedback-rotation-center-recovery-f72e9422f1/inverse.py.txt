"""Independent pixel-only oracle for the rotation-centre-recovery world.

Receives a rendered PNG and nothing else: no latent scene, no renderer import,
no analytic gold. It labels the black ink into two components, finds the unique
quarter or half turn that carries one component exactly onto the other, solves
that isometry for its fixed point in closed form, and names the colored disk
standing on it.

Pixel ``(x, y)`` is treated as the continuous square ``[x, x+1) x [y, y+1)``, so
each rotation acts on pixel indices by an exact integer map and the fixed point
follows from the integer translation part.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

# The disk palette is duplicated here on purpose; the oracle never reads scene
# data and classifies each disk by exact color match.
_PALETTE = {
    "blue": (40, 80, 220),
    "green": (30, 150, 60),
    "orange": (240, 140, 20),
    "purple": (150, 45, 190),
}
_FALLBACK = "blue"


def _components(mask):
    """Row-run connected-component labelling (4-connectivity)."""

    height, width = mask.shape
    parent = []

    def find(a):
        while parent[a] != a:
            parent[a] = parent[parent[a]]
            a = parent[a]
        return a

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    runs = []          # (y, start, end_inclusive, label)
    prev_runs = []
    for y in range(height):
        row = mask[y]
        if not row.any():
            prev_runs = []
            continue
        padded = np.concatenate(([0], row.view(np.uint8), [0]))
        edges = np.flatnonzero(np.diff(padded))
        starts = edges[0::2]
        ends = edges[1::2] - 1
        cur = []
        for s, e in zip(starts.tolist(), ends.tolist()):
            label = len(parent)
            parent.append(label)
            for ps, pe, plabel in prev_runs:
                if ps <= e and s <= pe:
                    union(label, plabel)
            cur.append((s, e, label))
            runs.append((y, s, e, label))
        prev_runs = cur

    groups = {}
    for y, s, e, label in runs:
        groups.setdefault(find(label), []).append((y, s, e))
    return list(groups.values())


def _coords(component):
    xs = []
    ys = []
    for y, s, e in component:
        span = e - s + 1
        xs.append(np.arange(s, e + 1, dtype=np.int64))
        ys.append(np.full(span, y, dtype=np.int64))
    return np.concatenate(xs), np.concatenate(ys)


def _key(xs, ys):
    return np.sort(xs.astype(np.int64) * 1000003 + ys.astype(np.int64))


def _rotate(xs, ys, kind):
    if kind == "r1":
        return ys, -xs - 1
    if kind == "r2":
        return -ys - 1, xs
    return -xs - 1, -ys - 1


def _fixed_point(kind, tx, ty):
    if kind == "r1":
        return ((tx + ty) / 2.0, (ty - tx) / 2.0)
    if kind == "r2":
        return ((tx - ty) / 2.0, (tx + ty) / 2.0)
    return (tx / 2.0, ty / 2.0)


def decision_from_image(image: Image.Image) -> str:
    arr = np.array(image.convert("RGB")).astype(np.int16)
    ink = np.all(arr < 60, axis=-1)
    if not ink.any():
        return _FALLBACK

    comps = _components(ink)
    comps = [c for c in comps if sum(e - s + 1 for _, s, e in c) >= 400]
    if len(comps) != 2:
        return _FALLBACK

    ax, ay = _coords(comps[0])
    bx, by = _coords(comps[1])
    if ax.size != bx.size:
        return _FALLBACK
    target = _key(bx, by)

    solutions = []
    for kind in ("r1", "r2", "r180"):
        rx, ry = _rotate(ax, ay, kind)
        tx = int(bx.min() - rx.min())
        ty = int(by.min() - ry.min())
        if np.array_equal(_key(rx + tx, ry + ty), target):
            solutions.append(_fixed_point(kind, tx, ty))
    if len(solutions) != 1:
        return _FALLBACK
    fx, fy = solutions[0]

    best = None
    for name, rgb in _PALETTE.items():
        hit = np.all(arr == np.array(rgb, dtype=np.int16), axis=-1)
        if hit.sum() < 60:
            continue
        ys, xs = np.nonzero(hit)
        cx = float(xs.mean()) + 0.5
        cy = float(ys.mean()) + 0.5
        d = (cx - fx) ** 2 + (cy - fy) ** 2
        if best is None or d < best[0]:
            best = (d, name)
    if best is None:
        return _FALLBACK
    return best[1]
