"""Deterministic latent sampler and renderer for equal-stride checkpoint search."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

CANVAS_SIZE = 800
BACKGROUND = (250, 248, 242)
DARK = (35, 43, 52)
GOLD = (224, 164, 35)
GRAY = (126, 137, 148)
COLORS = ("red", "blue", "green", "purple")
RGB = {
    "red": (210, 61, 69),
    "blue": (53, 105, 196),
    "green": (42, 145, 94),
    "purple": (139, 77, 177),
}

# Five center-to-center section lengths.  For target checkpoint t in 1..4,
# exactly lengths[t-1] and lengths[t] agree.  Every distractor mismatch is >= 9 px.
LENGTH_PATTERNS = {
    1: (18, 18, 32, 21, 30),
    2: (20, 32, 32, 18, 30),
    3: (30, 18, 32, 32, 20),
    4: (32, 20, 34, 22, 22),
}


def _rotate(point: tuple[int, int], quarter_turns: int) -> tuple[int, int]:
    x, y = point
    return ((x, y), (-y, x), (-x, -y), (y, -x))[quarter_turns % 4]


def sample_scene(seed: int) -> dict:
    """Sample a balanced compact route from an integer seed.

    Answer color cycles every seed and the winning checkpoint every four seeds,
    crossing all 16 color/position pairs in each consecutive block of 16.
    """

    rng = random.Random(int(seed) ^ 0x5EED51DE)
    target_index = (int(seed) // 4) % 4 + 1
    target_color = COLORS[int(seed) % 4]
    other_colors = [color for color in COLORS if color != target_color]
    rng.shuffle(other_colors)
    checkpoint_colors: list[str] = []
    for index in range(1, 5):
        checkpoint_colors.append(
            target_color if index == target_index else other_colors.pop()
        )

    lengths = LENGTH_PATTERNS[target_index]
    canonical = [(0, 0)]
    for length in lengths:
        canonical.append((canonical[-1][0] + length, 0))

    quarter_turns = rng.randrange(4)
    rotated = [_rotate(point, quarter_turns) for point in canonical]
    min_x = min(point[0] for point in rotated)
    min_y = min(point[1] for point in rotated)
    max_x = max(point[0] for point in rotated)
    max_y = max(point[1] for point in rotated)
    cell_x = rng.randrange(7)
    cell_y = rng.randrange(7)
    if max_x > min_x:  # horizontal route, safely inside two 100 px grid cells
        offset_x = 100 * cell_x + 34 - min_x
        offset_y = 100 * cell_y + 50 - min_y
    else:  # vertical route
        offset_x = 100 * cell_x + 50 - min_x
        offset_y = 100 * cell_y + 34 - min_y
    points = [[x + offset_x, y + offset_y] for x, y in rotated]

    return {
        "canvas_size": CANVAS_SIZE,
        "points": points,
        "checkpoint_colors": checkpoint_colors,
        "route_width": 5,
        "marker_radius": 7,
    }


def _section_lengths(points: Iterable[Iterable[float]]) -> list[float]:
    values = [tuple(map(float, point)) for point in points]
    return [
        math.hypot(values[index + 1][0] - values[index][0], values[index + 1][1] - values[index][1])
        for index in range(len(values) - 1)
    ]


def _mismatches(scene: dict) -> list[float]:
    lengths = _section_lengths(scene["points"])
    if len(lengths) != 5:
        raise ValueError("scene must contain exactly five route sections")
    return [abs(lengths[index - 1] - lengths[index]) for index in range(1, 5)]


def analytic_gold(scene: dict) -> str:
    mismatches = _mismatches(scene)
    winner = min(range(4), key=mismatches.__getitem__)
    if sum(abs(value - mismatches[winner]) < 1e-9 for value in mismatches) != 1:
        raise ValueError("analytic scene has no unique equality winner")
    return str(scene["checkpoint_colors"][winner])


def margin(scene: dict) -> float:
    ordered = sorted(_mismatches(scene))
    return float(ordered[1] - ordered[0])


def is_quarantined(scene: dict) -> bool:
    lengths = _section_lengths(scene["points"])
    mismatches = sorted(_mismatches(scene))
    return bool(
        len(scene["checkpoint_colors"]) != 4
        or mismatches[0] > 1e-9
        or mismatches[1] < 8.0
        or min(lengths) < 16.0
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    """Render the final RGB raster with fixed supersampling."""

    size = int(scene["canvas_size"])
    points = [tuple(map(int, point)) for point in scene["points"]]
    colors = [str(color) for color in scene["checkpoint_colors"]]
    route_width = int(scene["route_width"])
    radius = int(scene["marker_radius"])
    if len(points) != 6 or len(colors) != 4 or set(colors) != set(COLORS):
        raise ValueError("invalid route or checkpoint color permutation")

    supersample = 3
    image = Image.new("RGB", (size * supersample, size * supersample), BACKGROUND)
    draw = ImageDraw.Draw(image)
    scaled_points = [(x * supersample, y * supersample) for x, y in points]
    draw.line(
        scaled_points,
        fill=DARK,
        width=route_width * supersample,
        joint="curve",
    )

    marker_fills = [GOLD, *(RGB[color] for color in colors), GRAY]
    scaled_radius = radius * supersample
    for (x, y), fill in zip(scaled_points, marker_fills, strict=True):
        draw.ellipse(
            (x - scaled_radius, y - scaled_radius, x + scaled_radius, y + scaled_radius),
            fill=fill,
            outline=DARK,
            width=supersample,
        )
    return image.resize((size, size), Image.Resampling.LANCZOS)
