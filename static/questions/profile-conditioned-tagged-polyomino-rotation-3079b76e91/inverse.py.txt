"""Independent pixel-only inverse arm for tagged polyomino matching."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

INKS = {(32, 36, 43), (76, 82, 91)}
TAG_RGB = {
    "red": (207, 55, 59),
    "blue": (48, 96, 195),
    "gold": (220, 161, 28),
}


class OracleError(ValueError):
    pass


def _components(mask: np.ndarray):
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]:
            continue
        queue = deque([(y0, x0)])
        seen[y0, x0] = True
        pixels = []
        while queue:
            y, x = queue.popleft()
            pixels.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(pixels) >= 1000:
            ys = [p[0] for p in pixels]
            xs = [p[1] for p in pixels]
            result.append((min(xs), min(ys), max(xs), max(ys), pixels))
    return result


def _normalize(cells):
    x0 = min(x for x, _ in cells)
    y0 = min(y for _, y in cells)
    return tuple(sorted((x - x0, y - y0) for x, y in cells))


def _rotate(cells):
    return _normalize(tuple((-y, x) for x, y in cells))


def _canonical(cells):
    variants = []
    value = _normalize(cells)
    for _ in range(4):
        variants.append(value)
        value = _rotate(value)
    return min(variants)


def _signature(component) -> tuple:
    x0, y0, x1, y1, pixels = component
    local = np.zeros((y1 - y0 + 1, x1 - x0 + 1), dtype=bool)
    for y, x in pixels:
        local[y - y0, x - x0] = True
    runs = []
    for row in local:
        indices = np.flatnonzero(row)
        if len(indices):
            cuts = np.flatnonzero(np.diff(indices) > 1)
            starts = np.r_[0, cuts + 1]
            ends = np.r_[cuts, len(indices) - 1]
            runs.extend(int(indices[b] - indices[a] + 1) for a, b in zip(starts, ends))
    cell = 0
    for length in runs:
        cell = math.gcd(cell, length)
    if cell < 8 or local.shape[0] % cell or local.shape[1] % cell:
        raise OracleError("could not infer a regular square-cell lattice")
    cells = []
    for gy in range(local.shape[0] // cell):
        for gx in range(local.shape[1] // cell):
            block = local[gy * cell : (gy + 1) * cell, gx * cell : (gx + 1) * cell]
            if float(block.mean()) > 0.8:
                cells.append((gx, gy))
    if len(cells) != 7:
        raise OracleError("piece does not contain exactly seven occupied cells")
    return _canonical(tuple(cells))


def _tag_centers(arr: np.ndarray):
    centers = {}
    for name, rgb in TAG_RGB.items():
        ys, xs = np.nonzero(np.all(arr == np.asarray(rgb, dtype=np.uint8), axis=2))
        if len(xs) < 1000:
            raise OracleError(f"missing {name} tag")
        centers[name] = (float(xs.mean()), float(ys.mean()))
    return centers


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    dark = np.zeros(arr.shape[:2], dtype=bool)
    for rgb in INKS:
        dark |= np.all(arr == np.asarray(rgb, dtype=np.uint8), axis=2)
    pieces = _components(dark)
    if len(pieces) != 4:
        raise OracleError("expected exactly four seven-tile pieces")
    pieces.sort(key=lambda item: sum(y for y, _ in item[4]) / len(item[4]))
    sources, target = pieces[:3], pieces[3]
    sources.sort(key=lambda item: sum(x for _, x in item[4]) / len(item[4]))
    target_signature = _signature(target)
    matches = [component for component in sources if _signature(component) == target_signature]
    if len(matches) != 1:
        raise OracleError("raster has no unique rotation-normalized match")
    matched = matches[0]
    matched_x = sum(x for _, x in matched[4]) / len(matched[4])
    tags = _tag_centers(arr)
    ordered = sorted(tags, key=lambda name: tags[name][0])
    source_index = sources.index(matched)
    return ordered[source_index]
