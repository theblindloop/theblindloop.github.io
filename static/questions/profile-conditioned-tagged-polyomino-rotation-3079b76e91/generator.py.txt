"""Deterministic equal-area polyomino correspondence world."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 960, 640
CELL = 28
INK = (32, 36, 43)
GRID = (76, 82, 91)
BACKGROUND = (250, 248, 243)
PANEL = (228, 231, 234)
TAG_RGB = {
    "red": (207, 55, 59),
    "blue": (48, 96, 195),
    "gold": (220, 161, 28),
}
SOURCE_CENTERS = (190, 480, 770)

# All have seven cells, a 4x4 bounding box, one connected component, and
# different rotation orbits. Distractors therefore share tile count, ink, and size.
SHAPES = {
    "elbow": ((0, 0), (0, 1), (0, 2), (0, 3), (1, 0), (2, 0), (3, 0)),
    "stair_hook": ((0, 0), (1, 0), (2, 0), (2, 1), (3, 1), (3, 2), (3, 3)),
    "zigzag": ((0, 0), (1, 0), (1, 1), (2, 1), (2, 2), (3, 2), (3, 3)),
}


def _normalize(cells):
    x0 = min(x for x, _ in cells)
    y0 = min(y for _, y in cells)
    return tuple(sorted((x - x0, y - y0) for x, y in cells))


def _rotate(cells, turns: int):
    result = tuple((int(x), int(y)) for x, y in cells)
    for _ in range(int(turns) % 4):
        result = _normalize(tuple((-y, x) for x, y in result))
    return _normalize(result)


def _canonical(cells):
    return min(_rotate(cells, turns) for turns in range(4))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    names = list(SHAPES)
    target_shape = names[(int(seed) // 3) % 3]
    answer = ("red", "blue", "gold")[int(seed) % 3]

    remaining_colors = [color for color in TAG_RGB if color != answer]
    rng.shuffle(remaining_colors)
    tag_for = {target_shape: answer}
    for name, color in zip((name for name in names if name != target_shape), remaining_colors):
        tag_for[name] = color

    rng.shuffle(names)
    items = []
    source_rotation = 0
    for slot, name in enumerate(names):
        rotation = rng.randrange(4)
        if name == target_shape:
            source_rotation = rotation
        items.append({"slot": slot, "shape": name, "rotation": rotation, "tag": tag_for[name]})
    delta = 1 + rng.randrange(3)
    return {
        "items": items,
        "target_shape": target_shape,
        "target_rotation": (source_rotation + delta) % 4,
    }


def _draw_piece(draw: ImageDraw.ImageDraw, cells, cx: int, cy: int) -> None:
    cells = _normalize(cells)
    width = (max(x for x, _ in cells) + 1) * CELL
    height = (max(y for _, y in cells) + 1) * CELL
    x0, y0 = int(cx - width / 2), int(cy - height / 2)
    for x, y in cells:
        box = (x0 + x * CELL, y0 + y * CELL, x0 + (x + 1) * CELL - 1, y0 + (y + 1) * CELL - 1)
        draw.rectangle(box, fill=INK, outline=GRID, width=2)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((35, 35, 925, 355), radius=22, fill=(246, 246, 243), outline=PANEL, width=5)
    draw.rounded_rectangle((285, 395, 675, 605), radius=22, fill=(246, 246, 243), outline=PANEL, width=5)
    for item in sorted(scene["items"], key=lambda value: int(value["slot"])):
        cx = SOURCE_CENTERS[int(item["slot"])]
        cells = _rotate(SHAPES[str(item["shape"])], int(item["rotation"]))
        _draw_piece(draw, cells, cx, 185)
        color = TAG_RGB[str(item["tag"])]
        draw.line((cx, 250, cx, 274), fill=(174, 178, 182), width=5)
        draw.rounded_rectangle((cx - 45, 274, cx + 45, 316), radius=13, fill=color)
    target = _rotate(SHAPES[str(scene["target_shape"])], int(scene["target_rotation"]))
    _draw_piece(draw, target, 480, 500)
    return image


def analytic_gold(scene: dict) -> str:
    target_shape = str(scene["target_shape"])
    matches = [str(item["tag"]) for item in scene["items"] if str(item["shape"]) == target_shape]
    if len(matches) != 1:
        raise ValueError("scene must bind the target shape to exactly one tag")
    return matches[0]


def _cell_distance(left, right) -> int:
    left_set, right_set = set(_canonical(left)), set(_canonical(right))
    return len(left_set.symmetric_difference(right_set))


def margin(scene: dict) -> float:
    target = SHAPES[str(scene["target_shape"])]
    distances = sorted(_cell_distance(target, SHAPES[str(item["shape"])]) for item in scene["items"])
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    try:
        target = _canonical(SHAPES[str(scene["target_shape"])])
        matches = sum(_canonical(SHAPES[str(item["shape"])]) == target for item in scene["items"])
        return matches != 1 or margin(scene) < 2.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["items"] = list(reversed(twin["items"]))
    return [("source_record_serialization_reversal", twin)]
