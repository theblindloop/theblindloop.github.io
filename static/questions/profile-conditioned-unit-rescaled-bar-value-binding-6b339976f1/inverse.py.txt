"""Pixel-only inverse arm for `unit_rescaled_bar_value_binding`.

Receives a single PIL image and nothing else. It re-derives the panel frames,
each panel's unit ruler, the bar extents, and the marker position purely from
pixels, then solves the cross-panel assignment in unit space and reads the color
of the assigned reference bar.

This module imports no renderer, prompt, generator, verifier, or gold code.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_PALETTE = {
    "red": (220, 40, 40),
    "blue": (40, 70, 210),
    "green": (30, 150, 60),
    "orange": (240, 150, 20),
}
_GRAY = (130, 130, 130)
UNRESOLVED = "unresolved"


def _groups(indices, max_gap: int = 3):
    out = []
    for i in indices:
        if out and i - out[-1][-1] <= max_gap:
            out[-1].append(i)
        else:
            out.append([i])
    return out


def _boxes_by_y(mask: np.ndarray):
    rows = np.where(mask.any(axis=1))[0]
    boxes = []
    for grp in _groups(list(rows)):
        sub = mask[grp[0]: grp[-1] + 1]
        cols = np.where(sub.any(axis=0))[0]
        if cols.size == 0:
            continue
        boxes.append(
            {
                "y0": int(grp[0]),
                "y1": int(grp[-1]),
                "x0": int(cols[0]),
                "x1": int(cols[-1]),
                "yc": (grp[0] + grp[-1]) / 2.0,
                "w": int(cols[-1] - cols[0] + 1),
            }
        )
    return boxes


def _boxes_by_x(mask: np.ndarray):
    cols = np.where(mask.any(axis=0))[0]
    boxes = []
    for grp in _groups(list(cols)):
        sub = mask[:, grp[0]: grp[-1] + 1]
        rows = np.where(sub.any(axis=1))[0]
        if rows.size == 0:
            continue
        boxes.append(
            {
                "x0": int(grp[0]),
                "x1": int(grp[-1]),
                "y0": int(rows[0]),
                "y1": int(rows[-1]),
                "yc": (rows[0] + rows[-1]) / 2.0,
                "w": int(grp[-1] - grp[0] + 1),
            }
        )
    return boxes


def _exact(arr: np.ndarray, rgb) -> np.ndarray:
    return (
        (arr[:, :, 0] == rgb[0])
        & (arr[:, :, 1] == rgb[1])
        & (arr[:, :, 2] == rgb[2])
    )


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.int16)
    h, w, _ = arr.shape
    dark = arr.sum(axis=2) < 120

    # --- frames: long horizontal / vertical dark runs -----------------------
    row_runs = _groups(list(np.where(dark.sum(axis=1) > 0.5 * w)[0]))
    col_runs = _groups(list(np.where(dark.sum(axis=0) > 0.25 * h)[0]))
    if len(row_runs) != 4 or len(col_runs) != 2:
        return UNRESOLVED
    left = col_runs[0][-1] + 1
    right = col_runs[1][0] - 1
    panels = []
    for a, b in ((0, 1), (2, 3)):
        panels.append((row_runs[a][-1] + 1, row_runs[b][0] - 1))
    if right - left < 50:
        return UNRESOLVED

    def interior(mask, panel):
        out = np.zeros_like(mask)
        top, bot = panel
        out[top:bot + 1, left:right + 1] = mask[top:bot + 1, left:right + 1]
        return out

    # --- unit rulers and the marker ----------------------------------------
    black_a = _boxes_by_x(interior(dark, panels[0]))
    black_b = _boxes_by_x(interior(dark, panels[1]))
    if len(black_a) != 1 or len(black_b) != 2:
        return UNRESOLVED
    unit_a = black_a[0]["w"]
    ruler_b, marker = sorted(black_b, key=lambda b: b["w"])[::-1]
    unit_b = ruler_b["w"]
    if unit_a < 5 or unit_b < 5 or marker["w"] >= unit_b:
        return UNRESOLVED

    # --- reference bars -----------------------------------------------------
    ref = {}
    for name, rgb in _PALETTE.items():
        boxes = _boxes_by_y(interior(_exact(arr, rgb), panels[0]))
        if len(boxes) != 1:
            return UNRESOLVED
        ref[name] = boxes[0]["w"] / unit_a

    # --- gray bars ----------------------------------------------------------
    gray_boxes = _boxes_by_y(interior(_exact(arr, _GRAY), panels[1]))
    if len(gray_boxes) != 4:
        return UNRESOLVED
    marked = min(gray_boxes, key=lambda b: abs(b["yc"] - marker["yc"]))
    value = marked["w"] / unit_b

    ranked = sorted(ref.items(), key=lambda kv: abs(kv[1] - value))
    return ranked[0][0]


def margin_from_image(image: Image.Image) -> float:
    """Diagnostic runner-up margin in reference-panel pixels (image-only)."""
    arr = np.asarray(image.convert("RGB")).astype(np.int16)
    h, w, _ = arr.shape
    dark = arr.sum(axis=2) < 120
    row_runs = _groups(list(np.where(dark.sum(axis=1) > 0.5 * w)[0]))
    col_runs = _groups(list(np.where(dark.sum(axis=0) > 0.25 * h)[0]))
    if len(row_runs) != 4 or len(col_runs) != 2:
        return -1.0
    left, right = col_runs[0][-1] + 1, col_runs[1][0] - 1
    panels = [(row_runs[0][-1] + 1, row_runs[1][0] - 1),
              (row_runs[2][-1] + 1, row_runs[3][0] - 1)]

    def interior(mask, panel):
        out = np.zeros_like(mask)
        top, bot = panel
        out[top:bot + 1, left:right + 1] = mask[top:bot + 1, left:right + 1]
        return out

    black_a = _boxes_by_x(interior(dark, panels[0]))
    black_b = _boxes_by_x(interior(dark, panels[1]))
    if len(black_a) != 1 or len(black_b) != 2:
        return -1.0
    unit_a = black_a[0]["w"]
    ruler_b, marker = sorted(black_b, key=lambda b: b["w"])[::-1]
    unit_b = ruler_b["w"]
    vals = []
    for rgb in _PALETTE.values():
        boxes = _boxes_by_y(interior(_exact(arr, rgb), panels[0]))
        if len(boxes) != 1:
            return -1.0
        vals.append(boxes[0]["w"] / unit_a)
    gray_boxes = _boxes_by_y(interior(_exact(arr, _GRAY), panels[1]))
    if len(gray_boxes) != 4:
        return -1.0
    marked = min(gray_boxes, key=lambda b: abs(b["yc"] - marker["yc"]))
    value = marked["w"] / unit_b
    diffs = sorted(abs(value - v) for v in vals)
    return round((diffs[1] - diffs[0]) * unit_a, 4)
