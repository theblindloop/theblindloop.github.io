"""Renderer for `unit_rescaled_bar_value_binding`.

Two framed panels. Each panel carries its own black *unit ruler* at the bottom
left, and the two rulers have different pixel lengths. The upper panel holds
four colored bars, the lower panel holds four gray bars; both panels encode the
same four unit-values, so the gray bars are a bijective re-encoding of the
colored bars under a change of unit. One gray bar is marked; the decision is the
color of the reference bar that carries the same unit-value.

Scenes are sampled so that raw *pixel* length matching is actively misleading:
the colored bar whose pixel length is closest to the marked gray bar's pixel
length is never the correct answer.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

# ---------------------------------------------------------------- geometry --
WIDTH, HEIGHT = 760, 560

PANEL_A = (30, 30, 730, 265)   # left, top, right, bottom (inclusive frame box)
PANEL_B = (30, 295, 730, 530)
FRAME_WIDTH = 2

BAR_X0 = 70                    # every bar and every unit ruler starts here
BAR_HEIGHT = 22
ROW_PITCH = 44
ROW_A_TOP = 58
ROW_B_TOP = 323
GAUGE_A_TOP = 232
GAUGE_B_TOP = 497
GAUGE_HEIGHT = 14

MARKER_SIZE = 14
MARKER_X0 = 44

MAX_BAR_PX = 620
MIN_BAR_PX = 30

# ------------------------------------------------------------------ palette --
COLOR_NAMES = ("red", "blue", "green", "orange")
COLOR_RGB = {
    "red": (220, 40, 40),
    "blue": (40, 70, 210),
    "green": (30, 150, 60),
    "orange": (240, 150, 20),
}
GRAY_RGB = (130, 130, 130)
BLACK = (0, 0, 0)
WHITE = (255, 255, 255)

QUARANTINE_MARGIN_PX = 12.0

VALUE_LO, VALUE_HI = 1.5, 8.0


# ------------------------------------------------------------------ sampling --
def _sample_values(rng: random.Random) -> list[float]:
    """Four increasing unit-values with a controlled minimum gap.

    A `tight` draw deliberately places two reference values close together, which
    is the near-boundary regime for the correspondence decision.
    """
    tight = rng.random() < 0.34
    gmin = rng.uniform(0.13, 0.42) if tight else rng.uniform(0.55, 1.15)
    span = VALUE_HI - VALUE_LO
    extra = span - 3.0 * gmin
    cuts = sorted(rng.random() for _ in range(3))
    parts = [cuts[0], cuts[1] - cuts[0], cuts[2] - cuts[1], 1.0 - cuts[2]]
    parts = [p * extra for p in parts]
    gaps = [gmin + parts[1], gmin + parts[2], gmin + parts[3]]
    start = VALUE_LO + parts[0]
    if tight:
        k = rng.randrange(3)
        start += gaps[k] - gmin
        gaps[k] = gmin
    values = [start]
    for g in gaps:
        values.append(values[-1] + g)
    return [round(v, 6) for v in values]


def _sample_units(rng: random.Random, vmax: float) -> tuple[int, int] | None:
    unit_a = rng.randint(38, 74)
    if vmax * unit_a > MAX_BAR_PX:
        unit_a = int(MAX_BAR_PX // vmax)
    if unit_a < 30:
        return None
    if rng.random() < 0.5:
        ratio = rng.uniform(0.62, 0.80)
    else:
        ratio = rng.uniform(1.28, 1.62)
    unit_b = int(round(unit_a * ratio))
    if unit_b < 24 or abs(unit_b - unit_a) < 8:
        return None
    if vmax * unit_b > MAX_BAR_PX:
        return None
    if min(VALUE_LO * unit_a, VALUE_LO * unit_b) < MIN_BAR_PX:
        return None
    return unit_a, unit_b


def _lengths(values: list[float], unit: int) -> list[int]:
    return [int(round(v * unit)) for v in values]


def _pixel_cue_is_misleading(values, unit_a, unit_b, marked_value_index) -> bool:
    """True when nearest-pixel-length matching picks the wrong colored bar."""
    len_a = _lengths(values, unit_a)
    len_gray = _lengths(values, unit_b)[marked_value_index]
    diffs = [abs(len_gray - la) for la in len_a]
    return diffs.index(min(diffs)) != marked_value_index


def sample_scene(seed: int) -> dict:
    """Sample one fully-rendered latent scene.

    Answer color, marked row in the lower panel, and the answer's row in the
    upper panel are each cycled against `seed` so that color identity, lower-panel
    position, and upper-panel position are counterbalanced by construction.
    """
    rng = random.Random(seed)
    for _ in range(400):
        values = _sample_values(rng)
        units = _sample_units(rng, values[-1])
        if units is None:
            continue
        unit_a, unit_b = units

        order_b = [0, 1, 2, 3]
        rng.shuffle(order_b)
        marked_row_b = (seed // 4) % 4
        marked_value_index = order_b[marked_row_b]

        if not _pixel_cue_is_misleading(values, unit_a, unit_b, marked_value_index):
            continue

        # Upper-panel row assignment: the answer sits in a counterbalanced row.
        answer_row_a = (seed // 16) % 4
        rest = [i for i in range(4) if i != marked_value_index]
        rng.shuffle(rest)
        order_a = [None] * 4
        order_a[answer_row_a] = marked_value_index
        for row in range(4):
            if order_a[row] is None:
                order_a[row] = rest.pop()

        # Color assignment: the answer color is cycled against `seed`.
        answer_color = COLOR_NAMES[seed % 4]
        others = [c for c in COLOR_NAMES if c != answer_color]
        rng.shuffle(others)
        colors = [None] * 4
        colors[marked_value_index] = answer_color
        for i in range(4):
            if colors[i] is None:
                colors[i] = others.pop()

        scene = {
            "values": values,
            "unit_a": unit_a,
            "unit_b": unit_b,
            "order_a": order_a,
            "order_b": order_b,
            "colors": colors,
            "marked_row_b": marked_row_b,
        }
        if margin(scene) <= 0.0:
            continue
        return scene
    raise RuntimeError(f"scene sampling failed for seed {seed}")


# ------------------------------------------------------------------- render --
def _panel(draw: ImageDraw.ImageDraw, box) -> None:
    draw.rectangle(box, outline=BLACK, width=FRAME_WIDTH)


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), WHITE)
    draw = ImageDraw.Draw(img)
    _panel(draw, PANEL_A)
    _panel(draw, PANEL_B)

    values = scene["values"]
    unit_a, unit_b = scene["unit_a"], scene["unit_b"]
    len_a = _lengths(values, unit_a)
    len_b = _lengths(values, unit_b)

    for row, vi in enumerate(scene["order_a"]):
        top = ROW_A_TOP + row * ROW_PITCH
        rgb = COLOR_RGB[scene["colors"][vi]]
        draw.rectangle(
            [BAR_X0, top, BAR_X0 + len_a[vi] - 1, top + BAR_HEIGHT - 1], fill=rgb
        )

    for row, vi in enumerate(scene["order_b"]):
        top = ROW_B_TOP + row * ROW_PITCH
        draw.rectangle(
            [BAR_X0, top, BAR_X0 + len_b[vi] - 1, top + BAR_HEIGHT - 1], fill=GRAY_RGB
        )

    draw.rectangle(
        [BAR_X0, GAUGE_A_TOP, BAR_X0 + unit_a - 1, GAUGE_A_TOP + GAUGE_HEIGHT - 1],
        fill=BLACK,
    )
    draw.rectangle(
        [BAR_X0, GAUGE_B_TOP, BAR_X0 + unit_b - 1, GAUGE_B_TOP + GAUGE_HEIGHT - 1],
        fill=BLACK,
    )

    marked_top = ROW_B_TOP + scene["marked_row_b"] * ROW_PITCH
    center = marked_top + BAR_HEIGHT // 2
    half = MARKER_SIZE // 2
    draw.rectangle(
        [MARKER_X0, center - half, MARKER_X0 + MARKER_SIZE - 1, center + half - 1],
        fill=BLACK,
    )
    return img


# ---------------------------------------------------------- analytic answer --
def _measured(scene: dict):
    """Unit-values exactly as they are recoverable from the rendered pixels."""
    values = scene["values"]
    unit_a, unit_b = scene["unit_a"], scene["unit_b"]
    ref = [la / unit_a for la in _lengths(values, unit_a)]
    marked_value_index = scene["order_b"][scene["marked_row_b"]]
    gray = _lengths(values, unit_b)[marked_value_index] / unit_b
    return ref, gray, marked_value_index


def analytic_gold(scene: dict) -> str:
    ref, gray, _ = _measured(scene)
    diffs = [abs(gray - r) for r in ref]
    return scene["colors"][diffs.index(min(diffs))]


def margin(scene: dict) -> float:
    """Runner-up decision margin, in reference-panel pixels."""
    ref, gray, _ = _measured(scene)
    diffs = sorted(abs(gray - r) for r in ref)
    return round((diffs[1] - diffs[0]) * scene["unit_a"], 4)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases: every scene field is rendered and the raster determines
    the answer, so the latent-alias gate is declared `not_applicable`."""
    return []
