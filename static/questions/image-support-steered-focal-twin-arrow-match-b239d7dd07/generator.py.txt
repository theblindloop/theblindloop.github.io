"""Focal twin-glyph direction-correspondence world.

One small local inset (position randomized across scenes, well inside the
canvas) contains exactly two disconnected filled triangle glyphs pointing at one
of four cardinal directions (right, down, left, up), separated by visible white
background. The decision is a shape-orientation correspondence predicate: do the
two disconnected triangles point in the SAME direction?

Both triangles are jointly decisive and both lie within one small local span.
Removing either triangle leaves only one glyph, so the oracle abstains; the
decisive pixel support is exactly the two disconnected focal glyphs.

Latent state z = {dir_left, dir_right, color_left, color_right, ox, oy}. Only
dir_left/dir_right influence analytic gold. The colours are sampled independently
of direction (so colour never predicts the answer) and the inset position
(ox, oy) is painted into the raster so the decisive support sits in one focal
neighborhood and is randomized across scenes. The parametrisation is injective
over its label inputs (no nontrivial pixel-identical symmetry), declared
``not_applicable``.
"""

from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image, ImageDraw

RENDERER_VERSION = "focal_twin_arrow_match-0.1.0"

CANVAS = 512
CELL = 64

WHITE = (255, 255, 255)

PALETTE: dict[str, tuple[int, int, int]] = {
    "coral": (232, 93, 77),
    "teal": (31, 158, 141),
    "violet": (106, 81, 176),
}
PALETTE_NAMES = tuple(PALETTE)

# Direction -> nominal pointing unit vector (screen y grows downward).
DIRECTIONS = ("right", "down", "left", "up")
_DIRECTION_VECTOR = {
    "right": (1.0, 0.0),
    "down": (0.0, 1.0),
    "left": (-1.0, 0.0),
    "up": (0.0, -1.0),
}
_DIRECTION_ANGLE = {"right": 0.0, "down": 90.0, "left": 180.0, "up": 270.0}

# Triangle proportions measured from the glyph centre. The apex (pointing tip)
# is clearly the farthest point from the pixel centroid so an independent oracle
# can recover the direction from the raster alone.
APEX_R = 24.0
BASE_BACK = 10.0
BASE_HALF = 13.0

# Coarse cell-aligned origins so the two triangles deterministically occupy two
# grid-adjacent-but-disjoint cells with a full empty cell between them in the
# protected 8x8 ablation grid, preserving focal + multipart support everywhere.
ORIGIN_X = [CELL, 2 * CELL, 3 * CELL, 4 * CELL]
ORIGIN_Y = [CELL, 2 * CELL, 3 * CELL, 4 * CELL]

# Relative triangle centres inside the inset: 2 cells apart horizontally.
_TRI_1_OFF = (32, 32)
_TRI_2_OFF = (160, 32)


def _triangle_centres(scene: dict) -> tuple[tuple[int, int], tuple[int, int]]:
    ox, oy = scene["ox"], scene["oy"]
    return (ox + _TRI_1_OFF[0], oy + _TRI_1_OFF[1]), (
        ox + _TRI_2_OFF[0],
        oy + _TRI_2_OFF[1],
    )


def _triangle_vertices(center: tuple[int, int], direction: str) -> list[tuple[int, int]]:
    cx, cy = center
    ux, uy = _DIRECTION_VECTOR[direction]
    px, py = -uy, ux
    apex = (cx + ux * APEX_R, cy + uy * APEX_R)
    base_center = (cx - ux * BASE_BACK, cy - uy * BASE_BACK)
    base1 = (base_center[0] + px * BASE_HALF, base_center[1] + py * BASE_HALF)
    base2 = (base_center[0] - px * BASE_HALF, base_center[1] - py * BASE_HALF)
    return [apex, base1, base2]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    return {
        "dir_left": rng.choice(DIRECTIONS),
        "dir_right": rng.choice(DIRECTIONS),
        "color_left": rng.choice(PALETTE_NAMES),
        "color_right": rng.choice(PALETTE_NAMES),
        "ox": rng.choice(ORIGIN_X),
        "oy": rng.choice(ORIGIN_Y),
    }


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (CANVAS, CANVAS), WHITE)
    draw = ImageDraw.Draw(img)
    dirs = (scene["dir_left"], scene["dir_right"])
    colors = (scene["color_left"], scene["color_right"])
    for center, direction, color in zip(
        _triangle_centres(scene), dirs, colors, strict=True
    ):
        vertices = _triangle_vertices(center, direction)
        draw.polygon([(round(x), round(y)) for x, y in vertices], fill=PALETTE[color])
    return img


def analytic_gold(scene: dict) -> str:
    return "yes" if scene["dir_left"] == scene["dir_right"] else "no"


def margin(scene: dict) -> float:
    a = _DIRECTION_ANGLE[scene["dir_left"]]
    b = _DIRECTION_ANGLE[scene["dir_right"]]
    diff = abs(a - b) % 360.0
    return float(min(diff, 360.0 - diff))


def is_quarantined(scene: dict) -> bool:
    # Directions are four discrete, well-separated cardinal orientations and the
    # triangle glyphs are solid-filled, so the "same / different" identity is a
    # crisp finite-resolution event. Same-direction scenes (margin 0) are a
    # legitimate class, not an ambiguous boundary.
    return False


def latent_symmetries(scene: dict) -> list:
    return []


def scene_targets(scene: dict) -> dict:
    return {
        "dir_left": scene["dir_left"],
        "dir_right": scene["dir_right"],
        "color_left": scene["color_left"],
        "color_right": scene["color_right"],
        "ox": scene["ox"],
        "oy": scene["oy"],
    }
