"""Independent pixel-only oracle for the focal twin-glyph direction world.

Receives only the final PNG. It never sees the latent scene, renderer geometry
API, or analytic gold. It re-measures the raster independently:

1. It separates non-background pixels (white background) from the two glyphs.
2. It extracts the connected components of non-background pixels; a legitimate
   scene contains exactly the two small disconnected triangle glyphs.
3. If the number of such components is not exactly two (a triangle was removed,
   occluded, or replaced by background), the oracle abstains by raising. This is
   what makes each of the two disconnected components jointly decisive.
4. For each triangle it recovers the pointing direction purely from geometry:
   the apex (the component pixel farthest from the component pixel-centroid),
   whose bearing discretized to the nearest of the four cardinal directions
   yields that glyph's direction.
5. It answers whether the two recovered directions are identical.

The cardinal-direction mapping and the farthest-from-centroid apex recovery are
re-derived independently from the raster; no renderer/gold code is imported.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "focal_twin_arrow_match-oracle-0.1.0"

CANVAS = 512
WHITE = (255, 255, 255)

# Direction -> nominal pointing unit vector (screen y grows downward). The
# bearing of the recovered apex is rounded to the nearest cardinal angle.
_DIRECTION_VECTOR = {
    "right": (1.0, 0.0),
    "down": (0.0, 1.0),
    "left": (-1.0, 0.0),
    "up": (0.0, -1.0),
}


class TwinGlyphsCompromisedError(ValueError):
    """One or both of the two decisive disconnected glyphs are missing/damaged."""


def _component_labels(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = set()
    components = []
    for y in range(height):
        for x in range(width):
            if not mask[y, x] or (y, x) in seen:
                continue
            stack = [(y, x)]
            seen.add((y, x))
            cells = []
            while stack:
                cy, cx = stack.pop()
                cells.append((cy, cx))
                for neighbor in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                    ny, nx = neighbor
                    if (
                        0 <= ny < height
                        and 0 <= nx < width
                        and mask[ny, nx]
                        and neighbor not in seen
                    ):
                        seen.add(neighbor)
                        stack.append(neighbor)
            components.append(cells)
    return components


def _direction_of(cells: list[tuple[int, int]]) -> str:
    ys = np.asarray([c[0] for c in cells], dtype=np.float64)
    xs = np.asarray([c[1] for c in cells], dtype=np.float64)
    cx, cy = float(xs.mean()), float(ys.mean())
    dx_all = xs - cx
    dy_all = ys - cy
    dist2 = dx_all * dx_all + dy_all * dy_all
    apex = int(dist2.argmax())
    dx, dy = float(xs[apex] - cx), float(ys[apex] - cy)
    angle = math.degrees(math.atan2(dy, dx)) % 360.0
    nominal = round(angle / 90.0) % 4
    return tuple(_DIRECTION_VECTOR)[nominal]


def decision_from_image(image: Image.Image) -> str:
    image = image.convert("RGB")
    if image.size != (CANVAS, CANVAS):
        raise TwinGlyphsCompromisedError("unexpected image size")
    arr = np.asarray(image, dtype=np.uint8)
    flat = arr.reshape(-1, 3).astype(np.float32)
    white = np.linalg.norm(flat - np.array(WHITE, dtype=np.float32), axis=1) <= 24.0
    mask = (~white).reshape(arr.shape[:2])

    components = _component_labels(mask)
    if len(components) != 2:
        raise TwinGlyphsCompromisedError(
            f"expected exactly two disconnected glyphs, found {len(components)}"
        )

    directions = [_direction_of(cells) for cells in components]
    return "yes" if directions[0] == directions[1] else "no"
