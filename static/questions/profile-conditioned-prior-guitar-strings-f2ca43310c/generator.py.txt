"""Prior-conflict guitar-string world: count the visible strings.

The latent scene holds a guitar silhouette whose neck carries ``n`` evenly
spaced, parallel, unbroken vertical strings. A familiar canonical prior asserts
that a guitar has SIX strings; the counterfactual renders 4 or 8 strings and the
decision must be read from the visible strings alone, never from the prior. The
pixel oracle counts the actual string-colored components in the raster.

Latent alias mode is ``not_applicable``: every latent field is directly
reflected in the visible geometry (string count, spacing, neck width, body
placement, palette), so the rendering is injective in the latent scene and there
is no render-invariant alias. The canonical-prior name "guitar" is a
matched-control/counterfactual manipulation of the *question*, not of any latent
field.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS_W = 560
CANVAS_H = 640
NX_BASE = 280.0

# Supported off-quarantine counts. The canonical prior is 6; 4 and 8 are the
# counterfactuals that contradict it. Every supported count is spaced far enough
# that each string is an isolated unbroken vertical component (>= ~19px gap).
SUPPORTED_N = (4, 6, 8)

# A count is quarantined when adjacent strings are so crowded that their blobs
# can touch and be miscounted at pixel scale.
QUAR_MIN_GAP = 16.0

BACKGROUND = (255, 255, 255)
FRETBOARD = (72, 52, 38)     # dark brown neck / headstock
BODY = (150, 98, 54)         # warm brown guitar body
SOUNDHOLE = (38, 24, 14)     # very dark sound hole
STRING = (150, 196, 214)     # light steel-blue strings


def sample_scene(seed: int) -> dict:
    """Deterministic scene dict from an integer seed."""
    rng = random.Random(seed)
    n = SUPPORTED_N[rng.randrange(len(SUPPORTED_N))]
    # Cheap-cue counterbalance: position, neck width, string thickness, string
    # span, and body placement all vary independently of the count so that no
    # non-count cue predicts the answer.
    nx = round(NX_BASE + rng.uniform(-42.0, 42.0), 3)
    half_span = round(rng.uniform(92.0, 142.0), 3)
    neck_w = round(rng.uniform(42.0, 60.0), 3)
    string_w = round(rng.uniform(3.0, 7.0), 3)
    body_y = round(rng.uniform(412.0, 442.0), 3)
    return {
        "n": n,
        "nx": nx,
        "half_span": half_span,
        "neck_w": neck_w,
        "string_w": string_w,
        "body_y": body_y,
        "canvas_w": CANVAS_W,
        "canvas_h": CANVAS_H,
    }


def _string_x(scene: dict, k: int) -> float:
    """Horizontal position of the k-th string (0-indexed), evenly spaced."""
    n = int(scene["n"])
    if n == 1:
        return float(scene["nx"])
    step = 2.0 * float(scene["half_span"]) / (n - 1)
    return float(scene["nx"]) - float(scene["half_span"]) + k * step


def render(scene: dict) -> Image.Image:
    """Render an acoustic-guitar silhouette with its visible strings."""
    w = int(scene["canvas_w"])
    h = int(scene["canvas_h"])
    img = Image.new("RGB", (w, h), BACKGROUND)
    draw = ImageDraw.Draw(img)
    nx = float(scene["nx"])
    neck_w = float(scene["neck_w"])
    body_y = float(scene["body_y"])
    neck_left = nx - neck_w / 2.0
    neck_right = nx + neck_w / 2.0
    head_left = nx - neck_w / 2.0 + 2.0
    head_right = nx + neck_w / 2.0 - 2.0

    # Headstock above the neck.
    draw.rectangle([head_left, 44, head_right, 102], fill=FRETBOARD)
    # Tuning pegs (small circles, one per string on alternate sides).
    for k in range(int(scene["n"])):
        peg_y = 62 + (k % 2) * 12
        peg_x = head_left - 8 if k % 2 == 0 else head_right + 8
        draw.ellipse([peg_x - 5, peg_y - 5, peg_x + 5, peg_y + 5], fill=BODY)

    # Neck (fretboard) rectangle down to the body.
    draw.rectangle([neck_left, 102, neck_right, body_y], fill=FRETBOARD)
    # A few thin fret lines so the neck reads as a guitar.
    for f in range(1, 8):
        fy = 112 + f * 40
        if fy < body_y - 6:
            draw.line([neck_left + 3, fy, neck_right - 3, fy], fill=(120, 92, 62), width=2)

    # Body: upper bout + lower bout + sound hole.
    upper_cy = body_y + 34
    draw.ellipse(
        [nx - 76, upper_cy - 58, nx + 76, upper_cy + 52], fill=BODY
    )
    lower_cy = body_y + 118
    draw.ellipse(
        [nx - 104, lower_cy - 92, nx + 104, lower_cy + 92], fill=BODY
    )
    draw.ellipse([nx - 34, upper_cy - 34, nx + 34, upper_cy + 34], fill=SOUNDHOLE)

    # Strings drawn last, on top, as unbroken vertical lines of string color.
    string_w = float(scene["string_w"])
    top = 106.0
    bottom = h - 54.0
    for k in range(int(scene["n"])):
        sx = _string_x(scene, k)
        draw.line([sx, top, sx, bottom], fill=STRING, width=int(round(string_w)))
    return img


def analytic_gold(scene: dict) -> str:
    """Decision = the visible count of strings, as a string."""
    return str(int(scene["n"]))


def margin(scene: dict) -> float:
    """Verifiability margin = smallest horizontal gap between adjacent strings."""
    n = int(scene["n"])
    if n < 2:
        return float("inf")
    step = 2.0 * float(scene["half_span"]) / (n - 1)
    return round(step - float(scene["string_w"]), 6)


def is_quarantined(scene: dict) -> bool:
    """Quarantine crowded counts that are ambiguous at pixel scale."""
    n = int(scene["n"])
    if n not in SUPPORTED_N:
        return True
    return margin(scene) < QUAR_MIN_GAP


def latent_symmetries(scene: dict):
    """not_applicable: empty, the latent rendering is injective."""
    return []
