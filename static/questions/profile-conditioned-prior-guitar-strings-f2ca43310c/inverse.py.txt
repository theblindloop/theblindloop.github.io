"""Pixel-only inverse arm: count the guitar strings from the raster alone.

This module receives only the rendered PNG and never imports renderer, prompts,
generate, verify, or analytic-gold code, and it accepts no latent scene data. It
classifies each pixel to a fixed world palette by nearest Euclidean color and
counts the number of 4-connected string-colour components.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

# Fixed world palette (must match renderer constants).
BACKGROUND = (255, 255, 255)
FRETBOARD = (72, 52, 38)
BODY = (150, 98, 54)
SOUNDHOLE = (38, 24, 14)
STRING = (150, 196, 214)

_PALETTE = [BACKGROUND, FRETBOARD, BODY, SOUNDHOLE, STRING]
_STRING_LABEL = 4


def _label_pixels(arr: np.ndarray) -> np.ndarray:
    dists = np.stack(
        [np.linalg.norm(arr - np.array(c, dtype=np.float32), axis=-1) for c in _PALETTE],
        axis=-1,
    )
    return dists.argmin(axis=-1)


def _count_components(mask: np.ndarray) -> int:
    """Count 4-connected components of the boolean mask (BFS flood fill)."""
    h, w = mask.shape
    visited = np.zeros((h, w), dtype=bool)
    count = 0
    ys, xs = np.nonzero(mask)
    for y, x in zip(ys.tolist(), xs.tolist()):
        if visited[y, x]:
            continue
        count += 1
        stack = [(y, x)]
        visited[y, x] = True
        while stack:
            cy, cx = stack.pop()
            for ny, nx in (
                (cy - 1, cx),
                (cy + 1, cx),
                (cy, cx - 1),
                (cy, cx + 1),
            ):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    stack.append((ny, nx))
    return count


def decision_from_image(image: Image.Image) -> str:
    """Return the counted number of strings as a string."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label_pixels(arr)
    string_mask = labels == _STRING_LABEL
    n = _count_components(string_mask)
    return str(n)
