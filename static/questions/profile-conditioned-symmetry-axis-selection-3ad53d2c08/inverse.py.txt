"""Independent pixel-only inverse arm for the symmetry-axis-selection world.

Recovers which colored line is the reflection symmetry axis of the dark figure
from the final PNG alone. It never sees the latent scene, analytic gold, or
renderer code. It detects the three colored lines by palette segmentation,
fits each to a supporting line, detects the dark figure, reflects the whole
figure across each fitted line, and selects the line whose mirror maps the
figure onto itself the most exactly (largest mirror-overlap fraction).
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

FIGURE_COLOR = (45, 45, 58)
BACKGROUND = (255, 255, 255)
LINE_COLORS = {
    "red": (222, 40, 40),
    "green": (38, 178, 84),
    "blue": (44, 92, 224),
}
COLOR_NAMES = ["red", "green", "blue"]


def _fit_line(mask: np.ndarray) -> tuple[float, float, float] | None:
    """Principal-axis line through a color mask: returns (nx, ny, c) or None."""
    coords = np.argwhere(mask)
    if len(coords) < 20:
        return None
    ys = coords[:, 0].astype(float)
    xs = coords[:, 1].astype(float)
    mx, my = xs.mean(), ys.mean()
    cov = np.cov(np.stack([xs, ys]))
    eigvals, eigvecs = np.linalg.eigh(cov)
    d = eigvecs[:, int(np.argmax(eigvals))]
    norm = math.hypot(d[0], d[1])
    d = d / norm
    nx, ny = -d[1], d[0]
    c = -(nx * mx + ny * my)
    return (float(nx), float(ny), float(c))


def _overlap(fig_mask: np.ndarray, line: tuple[float, float, float]) -> float:
    """Fraction of figure pixels whose mirror across the line is also figure."""
    nx, ny, c = line
    rows, cols = np.nonzero(fig_mask)
    if len(rows) == 0:
        return 0.0
    xs = cols.astype(float)
    ys = rows.astype(float)
    s = nx * xs + ny * ys + c
    rx = xs - 2.0 * s * nx
    ry = ys - 2.0 * s * ny
    irx = np.rint(rx).astype(int)
    iry = np.rint(ry).astype(int)
    h, w = fig_mask.shape
    ok = (irx >= 0) & (irx < w) & (iry >= 0) & (iry < h)
    matched = np.zeros(len(rows), dtype=bool)
    matched[ok] = fig_mask[iry[ok], irx[ok]]
    return float(matched.mean())


def decision_from_image(image: Image.Image) -> str:
    """The color name of the colored line that is the figure's symmetry axis."""
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    r, g, b = arr[..., 0], arr[..., 1], arr[..., 2]

    fig = (
        (np.abs(r - FIGURE_COLOR[0]) < 40)
        & (np.abs(g - FIGURE_COLOR[1]) < 40)
        & (np.abs(b - FIGURE_COLOR[2]) < 40)
    )

    # Detect every present line color and its mirror overlap of the figure.
    results = {}
    lines_found = 0
    for color in COLOR_NAMES:
        rc, gc, bc = LINE_COLORS[color]
        mask = (np.abs(r - rc) < 48) & (np.abs(g - gc) < 48) & (np.abs(b - bc) < 48)
        line = _fit_line(mask)
        if line is None:
            continue
        lines_found += 1
        results[color] = _overlap(fig, line)

    if not results:
        return COLOR_NAMES[0]
    best = max(results, key=results.get)
    # Guard against a degenerate read: if no meaningful figure is present, a
    # consistent but arbitrary label keeps the oracle total and deterministic.
    if fig.sum() < 200:
        return COLOR_NAMES[0]
    if lines_found < 2:
        return best
    return best
