"""Reference-orientation visual search world.

A dark reference line segment is shown in a top panel. Four quadrants below
each contain one dark segment. Exactly one candidate segment has the same
undirected orientation (is parallel) to the reference; the decision is the
quadrant containing that segment. All rendering is deterministic and all
decision-relevant geometry is visible in the raster.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image, ImageDraw

SIZE = 480
BG = (255, 255, 255)
SEG = (20, 20, 20)
GRAY = (190, 190, 190)
BOX = (170, 170, 170)

REF_CX = SIZE // 2
REF_CY = 80
REF_LEN = 120
REF_THICK = 10
CAND_LEN = 96
CAND_THICK = 10

SECTORS = ("TL", "TR", "BL", "BR")
SECTOR_CENTERS = {
    "TL": (SIZE // 4, 265),
    "TR": (3 * SIZE // 4, 265),
    "BL": (SIZE // 4, 395),
    "BR": (3 * SIZE // 4, 395),
}

# Minimum undirected angular separation (degrees) of any distractor from the
# reference orientation and from every other candidate. Guarantees a runner-up
# decision margin and a unique winner.
MIN_SEP_FROM_REF = 30
MIN_SEP_ANY = 20
QUARANTINE_THRESH = 12.0


def _angdist(a: float, b: float) -> float:
    """Undirected angular distance between two line orientations (mod 180)."""
    d = abs((a - b) % 180.0)
    return min(d, 180.0 - d)


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(seed)
    ref_angle = float(rng.uniform(10.0, 170.0))

    angles: dict[str, float] = {}
    winner_sector = SECTORS[int(rng.integers(0, len(SECTORS)))]
    angles[winner_sector] = ref_angle

    for sector in SECTORS:
        if sector in angles:
            continue
        for _ in range(400):
            candidate = float(rng.uniform(0.0, 180.0))
            if _angdist(candidate, ref_angle) < MIN_SEP_FROM_REF:
                continue
            if any(
                _angdist(candidate, existing) < MIN_SEP_ANY
                for existing in angles.values()
            ):
                continue
            angles[sector] = candidate
            break
        else:  # pragma: no cover - extremely unlikely path
            angles[sector] = float((ref_angle + 60.0) % 180.0)

    candidates = [
        {
            "cx": SECTOR_CENTERS[sector][0],
            "cy": SECTOR_CENTERS[sector][1],
            "angle": float(angles[sector]),
            "sector": sector,
        }
        for sector in SECTORS
    ]
    return {
        "reference": {
            "cx": REF_CX,
            "cy": REF_CY,
            "angle": float(ref_angle),
            "length": REF_LEN,
            "thickness": REF_THICK,
        },
        "candidates": candidates,
    }


def _draw_segment(draw: ImageDraw.ImageDraw, cx: int, cy: int, angle: float,
                  length: float, thickness: int, color) -> None:
    rad = math.radians(angle)
    dx = (length / 2.0) * math.cos(rad)
    dy = (length / 2.0) * math.sin(rad)
    x0, y0 = cx - dx, cy - dy
    x1, y1 = cx + dx, cy + dy
    draw.line([x0, y0, x1, y1], fill=color, width=thickness)


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (SIZE, SIZE), BG)
    draw = ImageDraw.Draw(img)

    # Reference panel box outline.
    draw.rectangle([SIZE // 2 - 170, 20, SIZE // 2 + 170, 140],
                   outline=BOX, width=3)

    # Light-gray quadrant dividers (deeply lighter than SEG, never selected by
    # the oracle's dark threshold).
    draw.line([0, 330, SIZE, 330], fill=GRAY, width=2)
    draw.line([SIZE // 2, 200, SIZE // 2, SIZE], fill=GRAY, width=2)

    ref = scene["reference"]
    _draw_segment(draw, ref["cx"], ref["cy"], ref["angle"], ref["length"],
                  ref["thickness"], SEG)

    for cand in scene["candidates"]:
        _draw_segment(draw, cand["cx"], cand["cy"], cand["angle"], CAND_LEN,
                      CAND_THICK, SEG)

    return img


def _decision(scene: dict) -> tuple[str, float]:
    ref_angle = scene["reference"]["angle"]
    dists = []
    for cand in scene["candidates"]:
        d = _angdist(cand["angle"], ref_angle)
        dists.append((d, cand["sector"]))
    dists.sort(key=lambda item: (item[0], item[1]))
    best = dists[0]
    second = dists[1]
    return best[1], float(second[0] - best[0])


def analytic_gold(scene: dict) -> str:
    sector, _ = _decision(scene)
    return sector


def margin(scene: dict) -> float:
    _, m = _decision(scene)
    return m


def is_quarantined(scene: dict) -> bool:
    _, m = _decision(scene)
    return bool(m < QUARANTINE_THRESH)


def latent_symmetries(scene: dict) -> list:
    return []
