"""Independent pixel-only inverse arm for reference_orientation_search.

Receives only a PIL image and the public question contract. It recovers each
dark segment's orientation by PCA of its pixels, identifies the reference
segment (top panel) and the four quadrant segments, and selects the quadrant
whose segment has the undirected orientation closest to the reference's.

Imports nothing from the renderer, prompts, generator, or verifier.
"""

from __future__ import annotations

import math

import numpy as np


def _angdist(a: float, b: float) -> float:
    d = abs((a - b) % 180.0)
    return min(d, 180.0 - d)


def _orientation(points: np.ndarray) -> float:
    centered = points - points.mean(axis=0)
    if len(centered) < 2:
        raise ValueError("too few pixels for orientation")
    cov = centered.T @ centered
    vals, vecs = np.linalg.eigh(cov)
    v = vecs[:, int(np.argmax(vals))]
    return float(np.degrees(np.arctan2(v[1], v[0])) % 180.0)


def decision_from_image(image) -> str:
    w, h = image.size
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    dark = arr.sum(axis=2) < 150

    ys, xs = np.nonzero(dark)
    if xs.size == 0:
        return "TL"

    x_split = w * 0.5
    y_ref = h * 0.31
    y_upper = h * 0.416
    y_split = h * 0.6875

    ref_mask = ys < y_ref
    ref_orientation = _orientation(np.column_stack([xs[ref_mask], ys[ref_mask]]))

    def sector_orientation(sel) -> float:
        xsel, ysel = xs[sel], ys[sel]
        if xsel.size < 2:
            return float("nan")
        return _orientation(np.column_stack([xsel.astype(float), ysel.astype(float)]))

    best_sector = None
    best_dist = math.inf
    for sector in ("TL", "TR", "BL", "BR"):
        if sector in ("TL", "BL"):
            left = xs < x_split
        else:
            left = xs >= x_split
        if sector in ("TL", "TR"):
            band = (ys >= y_upper) & (ys < y_split)
        else:
            band = ys >= y_split
        sel = band & left
        ang = sector_orientation(sel)
        if math.isnan(ang):
            continue
        d = _angdist(ang, ref_orientation)
        if d < best_dist:
            best_dist = d
            best_sector = sector

    return best_sector if best_sector is not None else "TL"
