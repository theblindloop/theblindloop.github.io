"""Independent pixel-only inverse arm for column_integral_reveal.

Receives ONLY the final PNG. It re-discovers the hidden glyph by summing each
column of dark ink (the declared vertical projection) and reading the 1D
centroid-versus-flank shape of the surviving profile: a single central peak
selects A, two symmetric twin peaks select B. It reads no scene data and no
analytic gold.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "column_integral_reveal-oracle-0.1.0"

Q_TOL = 0.35


def _projection_margin(profile: np.ndarray) -> float:
    ln = len(profile)
    div = (ln - 1) // 4
    center = profile[ln // 2]
    side = 0.5 * (profile[div] + profile[ln - 1 - div])
    mean = profile.mean() + 1e-9
    return (center - side) / mean


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("L"), dtype=float)
    h, w = arr.shape
    dark = (255.0 - arr).sum(axis=0)  # per-column dark ink
    m = _projection_margin(dark)
    return "A" if m >= 0 else "B"


def reveal_margin(image: Image.Image) -> float:
    arr = np.asarray(image.convert("L"), dtype=float)
    dark = (255.0 - arr).sum(axis=0)
    return abs(_projection_margin(dark))
