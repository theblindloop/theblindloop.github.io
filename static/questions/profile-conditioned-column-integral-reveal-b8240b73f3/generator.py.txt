"""column_integral_reveal: vertical-projection (column-integral) glyph reveal.

Declared transform: the image is a 2D field of black speckle on white, in which
a coarse 1D glyph is hidden purely in the *column* of the ink distribution.
Every column holds an independent vertical scatter of dark pixels whose COUNT
follows a smooth 1D profile across the horizontal axis: a single central peak
(glyph A) or two symmetric twin peaks (glyph B). The raw 2D raster therefore
looks like uninformative random speckle with no readable shape, because the
ink is scattered at random vertical positions in every column. Sum (integrate)
each column vertically to collapse the speckle onto the horizontal axis: the
one-dimensional column-sum profile that survives reveals the hidden glyph.
A single central peak is glyph A; two symmetric twin peaks are glyph B.

Latent scene z = {glyph, seed}. label = glyph.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RENDERER_VERSION = "column_integral_reveal-0.1.0"

W = 64      # columns (horizontal axis of the projected profile)
H = 96      # rows (vertical scatter extent)
BASE = 6.0  # uniform pedestal of dark pixels per column
AMP = 40.0  # glyph amplitude above the pedestal
Q_TOL = 0.35


def _shape_A(width: int) -> np.ndarray:
    """Single central tent peak: 1 at centre, falls toward both edges."""
    j = np.arange(width, dtype=float)
    return np.clip(1.0 - np.abs(2.0 * j / (width - 1) - 1.0) ** 1.5, 0.0, 1.0)


def _shape_B(width: int) -> np.ndarray:
    """Two symmetric twin humps around the quarter and three-quarter columns."""
    j = np.arange(width, dtype=float)
    w = width * 0.08
    h = np.exp(-((j - (width - 1) * 0.25) ** 2) / (2.0 * w * w))
    h += np.exp(-((j - (width - 1) * 0.75) ** 2) / (2.0 * w * w))
    return h


def _profile(glyph: str) -> np.ndarray:
    return _shape_A(W) if glyph == "A" else _shape_B(W)


def build_scene(glyph: str, seed: int) -> dict:
    return {"glyph": glyph, "seed": int(seed)}


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(int(seed) & 0xFFFFFFFF)
    glyph = "A" if int(rng.integers(0, 2)) == 0 else "B"
    return build_scene(glyph, int(seed))


def _column_counts(glyph: str) -> np.ndarray:
    """Per-column dark-ink counts. A and B are area-normalized to equal ink."""
    a = _shape_A(W)
    b = _shape_B(W)
    ca = BASE + AMP * a
    cb = BASE + AMP * b
    ca *= (cb.sum() / ca.sum())  # equalize total ink against B
    return ca if glyph == "A" else cb


def render(scene: dict) -> Image.Image:
    """Render the 2D speckle field hiding the column profile."""
    glyph = scene["glyph"]
    seed = int(scene["seed"])
    rng = np.random.default_rng(seed & 0xFFFFFFFF)
    counts = _column_counts(glyph)
    field = np.full((H, W), 255.0, dtype=np.uint8)
    for j in range(W):
        c = int(round(counts[j]))
        c = min(c, H)
        rows = rng.choice(H, size=c, replace=False)
        field[rows, j] = 0
    return Image.fromarray(field, "L").convert("RGB")


def _profile_probe(glyph: str, seed: int) -> np.ndarray:
    """Column sums of dark ink of the deterministic render (analytic view)."""
    arr = np.asarray(render(build_scene(glyph, seed)).convert("L"), dtype=float)
    return (255.0 - arr).sum(axis=0)  # higher = more dark ink in that column


def _projection_margin(profile: np.ndarray) -> float:
    """Signed margin: (centre peak vs average of the two quarter flanks), scaled
    by the mean ink so it is scale-invariant. Positive => A, negative => B."""
    ln = len(profile)
    div = (ln - 1) // 4
    center = profile[ln // 2]
    side = 0.5 * (profile[div] + profile[ln - 1 - div])
    mean = profile.mean() + 1e-9
    return float(round((center - side) / mean, 6))


def analytic_gold(scene: dict) -> str:
    return scene["glyph"]


def margin(scene: dict) -> float:
    return _projection_margin(_profile_probe(scene["glyph"], int(scene["seed"])))


def is_quarantined(scene: dict) -> bool:
    return bool(abs(margin(scene)) < Q_TOL)


def latent_symmetries(scene: dict):
    """not_applicable: the random per-column scatter is injected by the signed
    seed and the glyph sets the exact integral, so no distinct latent vector
    reproduces the same raster. Declared empty."""
    return iter(())
