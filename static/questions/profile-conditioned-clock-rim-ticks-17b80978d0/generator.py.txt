"""Deterministic analog-clock world; gold reads only the visible tick count."""

from __future__ import annotations

import math
import random
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 480
COUNTS = (10, 11, 12, 13, 14)
BACKGROUNDS = ((220, 226, 231), (232, 224, 214), (220, 231, 222), (230, 222, 232))
FACES = ((250, 247, 235), (240, 247, 249), (248, 242, 249), (244, 249, 238))
RIMS = ((20, 24, 29), (31, 27, 25), (23, 29, 25), (29, 24, 34))
HANDS = ((181, 39, 48), (31, 91, 164), (29, 132, 91), (175, 74, 27))


def sample_scene(seed: int) -> dict:
    # Adjacent blocks of five share nuisance distributions and contain every class.
    rng = random.Random(int(seed) // len(COUNTS))
    radius = rng.randrange(142, 177, 4)
    cx = rng.randrange(radius + 32, WIDTH - radius - 31)
    cy = rng.randrange(radius + 28, HEIGHT - radius - 27)
    return {
        "tick_count": COUNTS[int(seed) % len(COUNTS)],
        "cx": cx,
        "cy": cy,
        "radius": radius,
        "phase": rng.uniform(-math.pi, math.pi),
        "face_color": list(FACES[rng.randrange(len(FACES))]),
        "rim_color": list(RIMS[rng.randrange(len(RIMS))]),
        "hand_color": list(HANDS[rng.randrange(len(HANDS))]),
        "hour_angle": rng.uniform(-math.pi, math.pi),
        "minute_angle": rng.uniform(-math.pi, math.pi),
    }


def _point(cx: float, cy: float, radius: float, angle: float) -> tuple[float, float]:
    return cx + radius * math.cos(angle), cy + radius * math.sin(angle)


def render(scene: dict) -> Image.Image:
    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), tuple(scene["face_color"]))
    draw = ImageDraw.Draw(image)
    cx, cy = float(scene["cx"]) * scale, float(scene["cy"]) * scale
    radius = float(scene["radius"]) * scale
    # Background is outside a solid clock face; its variation cannot predict the label.
    draw.rectangle((0, 0, WIDTH * scale, HEIGHT * scale), fill=tuple(scene["background"]) if "background" in scene else (224, 228, 230))
    face = tuple(scene["face_color"])
    rim = tuple(scene["rim_color"])
    draw.ellipse((cx-radius, cy-radius, cx+radius, cy+radius), fill=face, outline=rim, width=6*scale)
    count = int(scene["tick_count"])
    phase = float(scene["phase"])
    for i in range(count):
        angle = phase + 2 * math.pi * i / count
        p0 = _point(cx, cy, radius - 22*scale, angle)
        p1 = _point(cx, cy, radius - 9*scale, angle)
        draw.line((p0, p1), fill=rim, width=6*scale)
    hand = tuple(scene["hand_color"])
    for angle, length, width in ((float(scene["hour_angle"]), 0.48, 8), (float(scene["minute_angle"]), 0.70, 5)):
        tip = _point(cx, cy, radius * length, angle)
        draw.line((cx, cy, *tip), fill=hand, width=width*scale)
    hub = 9 * scale
    draw.ellipse((cx-hub, cy-hub, cx+hub, cy+hub), fill=hand)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(int(scene["tick_count"]))


def margin(scene: dict) -> float:
    return float(2 * math.pi * (float(scene["radius"]) - 15.0) / int(scene["tick_count"]) - 6.0)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 10.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
