"""Pixel-only inverse arm: count dark angular runs in a recovered clock annulus."""

from __future__ import annotations

import math
from PIL import Image


class OracleAbstention(ValueError):
    pass


def _dark(pixel: tuple[int, int, int]) -> bool:
    return max(pixel) < 90 and sum(pixel) < 220


def _circular_runs(bits: list[bool]) -> list[int]:
    if not any(bits) or all(bits):
        return []
    start = next(i for i, value in enumerate(bits) if not value)
    ordered = bits[start:] + bits[:start]
    runs, length = [], 0
    for value in ordered:
        if value:
            length += 1
        elif length:
            runs.append(length)
            length = 0
    if length:
        runs.append(length)
    return runs


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    w, h = rgb.size
    points = [(x, y) for y in range(h) for x in range(w) if _dark(rgb.getpixel((x, y)))]
    if len(points) < 700:
        raise OracleAbstention("no substantial dark clock structure")
    xs = [p[0] for p in points]
    ys = [p[1] for p in points]
    left, right, top, bottom = min(xs), max(xs), min(ys), max(ys)
    diameter_x, diameter_y = right - left, bottom - top
    if min(diameter_x, diameter_y) < 220 or abs(diameter_x - diameter_y) > 8:
        raise OracleAbstention("dark outline is not a large circular rim")
    cx, cy = (left + right) / 2.0, (top + bottom) / 2.0
    radius = (diameter_x + diameter_y) / 4.0
    decisions = []
    for inset in (13.0, 15.0, 17.0):
        sample_radius = radius - inset
        bits = []
        for i in range(1440):
            angle = 2 * math.pi * i / 1440
            x = int(round(cx + sample_radius * math.cos(angle)))
            y = int(round(cy + sample_radius * math.sin(angle)))
            bits.append(_dark(rgb.getpixel((x, y))))
        runs = _circular_runs(bits)
        clean = [length for length in runs if length >= 3]
        if len(clean) == len(runs) and clean and max(clean) <= 22:
            decisions.append(len(clean))
    if len(decisions) < 2 or len(set(decisions)) != 1:
        raise OracleAbstention("angular scans disagree or lack separated tick peaks")
    count = decisions[0]
    if count < 6 or count > 20:
        raise OracleAbstention("implausible rim-mark count")
    return str(count)
