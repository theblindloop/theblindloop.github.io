"""Deterministic renderer for a complementary-region topology search."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

BACKGROUND = (248, 246, 238)
CYAN = (22, 138, 173)
MARKERS = {
    "red": (211, 52, 55),
    "gold": (224, 151, 0),
    "violet": (132, 75, 176),
    "green": (24, 142, 78),
}
COLOR_NAMES = tuple(MARKERS)


def sample_scene(seed: int) -> dict:
    """Balance the sealed position and marker identity across consecutive seeds."""

    rng = random.Random(seed ^ 0x5EEDC0DE)
    sealed_index = seed % 4
    target_name = COLOR_NAMES[(seed // 4) % 4]
    remaining = [name for name in COLOR_NAMES if name != target_name]
    rng.shuffle(remaining)
    colors: list[str | None] = [None, None, None, None]
    colors[sealed_index] = target_name
    for index, name in zip((i for i in range(4) if i != sealed_index), remaining):
        colors[index] = name
    return {
        "width": 512,
        "height": 512,
        "colors": colors,
        "sealed_index": sealed_index,
        "door_variants": [rng.randrange(2) for _ in range(4)],
    }


def _geometry(scene: dict) -> tuple[tuple[int, int, int, int], list[tuple[int, int, int, int]], list[tuple[int, int, int, int] | None]]:
    width = int(scene["width"])
    height = int(scene["height"])
    variants = [int(value) for value in scene["door_variants"]]
    sealed = int(scene["sealed_index"])
    # Geometry scales from the visible canvas; the standard scene is 512 square.
    sx, sy = width / 512.0, height / 512.0
    plate = tuple(round(v * s) for v, s in zip((24, 40, 488, 472), (sx, sy, sx, sy)))
    bases = ((65, 85, 215, 215), (297, 85, 447, 215), (65, 297, 215, 427), (297, 297, 447, 427))
    courtyards = [
        (round(x0 * sx), round(y0 * sy), round(x1 * sx), round(y1 * sy))
        for x0, y0, x1, y1 in bases
    ]
    doors: list[tuple[int, int, int, int] | None] = []
    half = max(14, round(17 * min(sx, sy)))
    px0, py0, px1, py1 = plate
    for index, (x0, y0, x1, y1) in enumerate(courtyards):
        if index == sealed:
            doors.append(None)
            continue
        cx, cy = (x0 + x1) // 2, (y0 + y1) // 2
        if index == 0:
            door = (cx - half, py0, cx + half, y0) if variants[index] == 0 else (px0, cy - half, x0, cy + half)
        elif index == 1:
            door = (cx - half, py0, cx + half, y0) if variants[index] == 0 else (x1, cy - half, px1, cy + half)
        elif index == 2:
            door = (cx - half, y1, cx + half, py1) if variants[index] == 0 else (px0, cy - half, x0, cy + half)
        else:
            door = (cx - half, y1, cx + half, py1) if variants[index] == 0 else (x1, cy - half, px1, cy + half)
        doors.append(door)
    return plate, courtyards, doors


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    colors = list(scene["colors"])
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    plate, courtyards, doors = _geometry(scene)
    draw.rectangle(plate, fill=CYAN)
    for box, door in zip(courtyards, doors):
        draw.rectangle(box, fill=BACKGROUND)
        if door is not None:
            draw.rectangle(door, fill=BACKGROUND)
    radius = max(9, round(12 * min(width, height) / 512.0))
    for name, box in zip(colors, courtyards):
        if name not in MARKERS:
            raise ValueError(f"unknown marker color {name!r}")
        x0, y0, x1, y1 = box
        cx, cy = (x0 + x1) // 2, (y0 + y1) // 2
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=MARKERS[name])
    return image


def analytic_gold(scene: dict) -> str:
    return scene["colors"][scene["sealed_index"]]


def margin(scene: dict) -> float:
    # The visible runner-up gap is the minimum opening width: open regions have
    # a 34-pixel passage while the winner has no passage at all.
    return 34.0 * min(int(scene["width"]), int(scene["height"])) / 512.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 24.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
