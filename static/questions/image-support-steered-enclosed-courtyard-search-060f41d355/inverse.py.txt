"""Independent pixel-only inverse arm for the courtyard search."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

CYAN = np.asarray((22, 138, 173), dtype=np.int16)
MARKERS = {
    "red": np.asarray((211, 52, 55), dtype=np.int16),
    "gold": np.asarray((224, 151, 0), dtype=np.int16),
    "violet": np.asarray((132, 75, 176), dtype=np.int16),
    "green": np.asarray((24, 142, 78), dtype=np.int16),
}


def _components(mask: np.ndarray) -> tuple[np.ndarray, list[dict]]:
    height, width = mask.shape
    labels = np.full((height, width), -1, dtype=np.int32)
    records: list[dict] = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if labels[sy, sx] >= 0:
            continue
        label = len(records)
        queue = deque([(sy, sx)])
        labels[sy, sx] = label
        count = 0
        touches_border = False
        while queue:
            y, x = queue.popleft()
            count += 1
            touches_border |= y == 0 or x == 0 or y == height - 1 or x == width - 1
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and labels[ny, nx] < 0:
                    labels[ny, nx] = label
                    queue.append((ny, nx))
        records.append({"count": count, "touches_border": touches_border})
    return labels, records


def _marker_centers(array: np.ndarray) -> dict[str, tuple[int, int]]:
    centers = {}
    work = array.astype(np.int16)
    for name, rgb in MARKERS.items():
        distance = np.linalg.norm(work - rgb, axis=2)
        mask = distance < 18.0
        ys, xs = np.nonzero(mask)
        if len(xs) < 250:
            raise ValueError(f"missing or damaged {name} marker")
        # One compact disk, not scattered pixels.
        if (xs.max() - xs.min()) > 32 or (ys.max() - ys.min()) > 32:
            raise ValueError(f"noncompact {name} marker")
        centers[name] = (int(round(float(ys.mean()))), int(round(float(xs.mean()))))
    return centers


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or min(array.shape[:2]) < 128:
        raise ValueError("canvas is too small")
    distance = np.linalg.norm(array.astype(np.int16) - CYAN, axis=2)
    cyan = distance < 18.0
    cyan_fraction = float(cyan.mean())
    # All four layouts deliberately keep the broad plate at the same visible
    # area (minor differences come only from which equal-width door reaches a
    # nearby edge). This tight, coordinate-free band rejects erasure anywhere
    # on that single distributed evidence component.
    if not 0.447 < cyan_fraction < 0.451:
        raise ValueError("cyan plate area is outside the visible contract")
    _, cyan_records = _components(cyan)
    substantial_cyan = [record for record in cyan_records if record["count"] >= 500]
    if len(substantial_cyan) != 1:
        raise ValueError("cyan plate is not one connected component")

    centers = _marker_centers(array)
    labels, regions = _components(~cyan)
    interior = [index for index, record in enumerate(regions) if not record["touches_border"] and record["count"] >= 500]
    if len(interior) != 1:
        raise ValueError("image does not have exactly one enclosed courtyard")
    enclosed_label = interior[0]
    winners = []
    for name, (y, x) in centers.items():
        label = int(labels[y, x])
        if label < 0:
            raise ValueError("marker lies on cyan")
        if label == enclosed_label:
            winners.append(name)
    if len(winners) != 1:
        raise ValueError("enclosed-region winner is not unique")
    # All other marker regions must visibly reach the canvas border.
    if any(not regions[int(labels[y, x])]["touches_border"] for name, (y, x) in centers.items() if name != winners[0]):
        raise ValueError("a distractor courtyard is also enclosed")
    return winners[0]
