"""Pixel-only decision arm for decade_step_count.

Receives a rasterized chart and nothing else. It recovers the ten-fold step of
the logarithmic scale from the thick horizontal rules, segments the grey and
coloured bars, pairs each coloured bar with the grey bar of its own group, and
counts the groups whose coloured top stands more than two steps above its grey
partner. It imports no renderer, prompt, generator, verifier or gold code and
never sees a scene.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "decade_step_count-oracle-1.0.0"

ABSTAIN = "abstain"

_THICK = (70, 70, 80)
_GREY = (150, 150, 150)
_HUES = {
    "red": (215, 60, 50),
    "green": (40, 150, 70),
    "blue": (55, 95, 205),
    "orange": (235, 150, 25),
}


def _mask(array: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    return (
        (array[:, :, 0] == rgb[0])
        & (array[:, :, 1] == rgb[1])
        & (array[:, :, 2] == rgb[2])
    )


def _runs(flags: np.ndarray) -> list[tuple[int, int]]:
    runs: list[tuple[int, int]] = []
    start = None
    for index, flag in enumerate(flags):
        if flag and start is None:
            start = index
        elif not flag and start is not None:
            runs.append((start, index - 1))
            start = None
    if start is not None:
        runs.append((start, len(flags) - 1))
    return runs


def _bars(mask: np.ndarray) -> list[tuple[float, int, int]]:
    """Return (centre column, top row, column width) for each column run."""
    columns = mask.any(axis=0)
    found = []
    for lo, hi in _runs(columns):
        block = mask[:, lo : hi + 1]
        rows = np.flatnonzero(block.any(axis=1))
        if rows.size == 0:
            continue
        found.append(((lo + hi) / 2.0, int(rows[0]), hi - lo + 1))
    return found


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    height, width = array.shape[0], array.shape[1]

    thick_rows = _mask(array, _THICK).sum(axis=1)
    line_rows = thick_rows > 0.35 * width
    groups = _runs(line_rows)
    if len(groups) < 4:
        return ABSTAIN
    centres = [(lo + hi) / 2.0 for lo, hi in groups]
    steps = [centres[i + 1] - centres[i] for i in range(len(centres) - 1)]
    if not steps:
        return ABSTAIN
    step = float(np.median(steps))
    if step < 20.0 or max(abs(value - step) for value in steps) > 1.5:
        return ABSTAIN

    grey = [bar for bar in _bars(_mask(array, _GREY)) if bar[2] >= 8]
    if len(grey) != 4:
        return ABSTAIN

    coloured = []
    for name, rgb in _HUES.items():
        found = [bar for bar in _bars(_mask(array, rgb)) if bar[2] >= 8]
        if len(found) != 1:
            return ABSTAIN
        coloured.append(found[0])
    if len(coloured) != 4:
        return ABSTAIN

    used: set[int] = set()
    count = 0
    for centre, top, _width in coloured:
        order = sorted(range(4), key=lambda i: abs(grey[i][0] - centre))
        best = order[0]
        if best in used:
            return ABSTAIN
        if abs(grey[order[1]][0] - centre) - abs(grey[best][0] - centre) < 20.0:
            return ABSTAIN
        used.add(best)
        gap = grey[best][1] - top
        if abs(gap - 2.0 * step) < 2.0:
            return ABSTAIN
        if gap > 2.0 * step:
            count += 1
    if len(used) != 4:
        return ABSTAIN
    if height < 100:
        return ABSTAIN
    return str(count)
