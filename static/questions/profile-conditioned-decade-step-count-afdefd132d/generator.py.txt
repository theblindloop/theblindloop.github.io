"""Scene sampler and raster renderer for decade_step_count.

The picture is a single chart panel whose vertical scale is logarithmic. Thick
dark horizontal rules mark successive ten-fold steps; thin pale rules are the
intermediate multiples and carry no information. Four groups of bars stand on
the bottom rule. Each group holds one grey bar, one coloured bar and one black
distractor bar. A group COUNTS when its coloured bar's top lies more than two
thick-rule steps above its grey partner's top, i.e. when the coloured value is
more than one hundred times the grey value. The decision is how many of the
four groups count.

Everything the gold uses is an integer pixel row that is actually drawn: the
sampler stores the rendered bar-top rows and the rendered thick-rule spacing,
so the analytic arm and the pixel arm read the very same integers.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "decade_step_count-renderer-1.0.0"

WIDTH = 660
HEIGHT = 430
PLOT_LEFT = 70
PLOT_RIGHT = 630
N_DECADES = 4

WHITE = (255, 255, 255)
THICK = (70, 70, 80)
THIN = (205, 205, 215)
GREY_BAR = (150, 150, 150)
BLACK_BAR = (20, 20, 20)
HUES = {
    "red": (215, 60, 50),
    "green": (40, 150, 70),
    "blue": (55, 95, 205),
    "orange": (235, 150, 25),
}
COLOR_ORDER = ("red", "green", "blue", "orange")

BAR_W = 20
BLACK_W = 12
QUARANTINE_PX = 10.0
_REJECT_PX = 3.0
_LOG10 = {k: math.log10(k) for k in range(2, 10)}


def _tops(baseline: int, decade_px: int, exponent: float) -> int:
    return int(round(baseline - exponent * decade_px))


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    for _ in range(400):
        decade_px = rng.randint(74, 92)
        baseline = rng.randint(392, 400)
        top_line = baseline - N_DECADES * decade_px
        centers = [base + rng.randint(-10, 10) for base in (140, 262, 384, 506)]
        slot_colors = list(COLOR_ORDER)
        rng.shuffle(slot_colors)

        target = rng.choice([0, 1, 2, 3])
        want_conflict = rng.random() < 0.5
        qualifying = sorted(rng.sample(range(4), target))

        gaps: list[float] = []
        greys: list[float] = []
        for slot in range(4):
            if slot in qualifying:
                gap = rng.uniform(2.14, 3.35)
            elif slot == 0 and target < 4:
                gap = rng.uniform(1.25, 1.86)
            else:
                gap = rng.uniform(0.35, 1.86)
            high = min(1.75, 3.90 - gap)
            if high <= 0.10:
                break
            gaps.append(gap)
            greys.append(rng.uniform(0.10, high))
        if len(gaps) < 4:
            continue
        if rng.random() < 0.22:
            tight = rng.randrange(4)
            gaps[tight] = (
                rng.uniform(2.025, 2.075)
                if tight in qualifying
                else rng.uniform(1.925, 1.975)
            )
            greys[tight] = min(greys[tight], max(0.10, 3.90 - gaps[tight]))

        grey_tops = [_tops(baseline, decade_px, greys[i]) for i in range(4)]
        color_tops = [_tops(baseline, decade_px, greys[i] + gaps[i]) for i in range(4)]
        black_tops = [
            _tops(baseline, decade_px, rng.uniform(0.20, 3.88)) for _ in range(4)
        ]

        pixel_gaps = [grey_tops[i] - color_tops[i] for i in range(4)]
        threshold = 2 * decade_px
        counted = [gap > threshold for gap in pixel_gaps]
        if sum(counted) != target:
            continue
        if min(abs(gap - threshold) for gap in pixel_gaps) < _REJECT_PX:
            continue
        if min(color_tops) <= top_line + 3 or max(grey_tops) >= baseline - 6:
            continue

        tallest = min(range(4), key=lambda i: color_tops[i])
        congruent = counted[tallest]
        if want_conflict and congruent and target < 4:
            continue

        return {
            "baseline": baseline,
            "decade_px": decade_px,
            "top_line": top_line,
            "centers": centers,
            "slot_colors": slot_colors,
            "grey_tops": grey_tops,
            "color_tops": color_tops,
            "black_tops": black_tops,
        }
    raise RuntimeError(f"scene sampling failed for seed {seed}")


def _pixel_gaps(scene: dict) -> list[int]:
    return [
        int(scene["grey_tops"][i]) - int(scene["color_tops"][i]) for i in range(4)
    ]


def pixel_gaps(scene: dict) -> list[int]:
    return _pixel_gaps(scene)


def counted_slots(scene: dict) -> list[bool]:
    threshold = 2 * int(scene["decade_px"])
    return [gap > threshold for gap in _pixel_gaps(scene)]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), WHITE)
    draw = ImageDraw.Draw(image)
    baseline = int(scene["baseline"])
    decade_px = int(scene["decade_px"])
    top_line = int(scene["top_line"])

    for decade in range(N_DECADES):
        for k in range(2, 10):
            y = int(round(baseline - (decade + _LOG10[k]) * decade_px))
            if top_line <= y <= baseline:
                draw.rectangle([PLOT_LEFT, y, PLOT_RIGHT, y], fill=THIN)
    for decade in range(N_DECADES + 1):
        y = baseline - decade * decade_px
        draw.rectangle([PLOT_LEFT, y - 1, PLOT_RIGHT, y + 1], fill=THICK)

    for slot in range(4):
        cx = int(scene["centers"][slot])
        hue = HUES[scene["slot_colors"][slot]]
        draw.rectangle(
            [cx - 34, int(scene["grey_tops"][slot]), cx - 34 + BAR_W - 1, baseline],
            fill=GREY_BAR,
        )
        draw.rectangle(
            [cx - 8, int(scene["color_tops"][slot]), cx - 8 + BAR_W - 1, baseline],
            fill=hue,
        )
        draw.rectangle(
            [cx + 18, int(scene["black_tops"][slot]), cx + 18 + BLACK_W - 1, baseline],
            fill=BLACK_BAR,
        )
    return image


def analytic_gold(scene: dict) -> str:
    return str(sum(counted_slots(scene)))


def margin(scene: dict) -> float:
    threshold = 2 * int(scene["decade_px"])
    return float(min(abs(gap - threshold) for gap in _pixel_gaps(scene)))


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_PX)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def cue_congruence(scene: dict) -> dict:
    counted = counted_slots(scene)
    tallest = min(range(4), key=lambda i: int(scene["color_tops"][i]))
    blackest = min(range(4), key=lambda i: int(scene["black_tops"][i]))
    midline = int(scene["baseline"]) - 2 * int(scene["decade_px"])
    tall_colored = sum(1 for t in scene["color_tops"] if int(t) < midline)
    return {
        "tallest_colored_counts": bool(counted[tallest]),
        "tallest_black_group_counts": bool(counted[blackest]),
        "count_equals_bars_above_midline": bool(tall_colored == sum(counted)),
    }
