"""Independent inverse arm using only final RGB pixels."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int) -> list[dict]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        pixels = []
        while queue:
            y, x = queue.popleft()
            pixels.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(pixels) < minimum:
            continue
        ys = np.fromiter((p[0] for p in pixels), dtype=np.int32)
        xs = np.fromiter((p[1] for p in pixels), dtype=np.int32)
        output.append(
            {
                "area": int(len(pixels)),
                "cx": float(xs.mean()),
                "cy": float(ys.mean()),
                "x0": int(xs.min()),
                "x1": int(xs.max()),
                "y0": int(ys.min()),
                "y1": int(ys.max()),
                "pixels": pixels,
            }
        )
    return output


def _split_loads(beads: list[dict]) -> tuple[int, int]:
    if len(beads) != 10:
        raise ValueError(f"expected ten starting beads, found {len(beads)}")
    centers = sorted(component["cx"] for component in beads)
    gaps = [centers[index + 1] - centers[index] for index in range(len(centers) - 1)]
    split = int(np.argmax(gaps)) + 1
    other = [gap for index, gap in enumerate(gaps) if index != split - 1 and gap > 0.5]
    ordinary = float(np.median(other)) if other else 1.0
    if gaps[split - 1] < max(35.0, ordinary * 2.5) or split in (0, len(centers)):
        raise ValueError("starting beads do not form two separated bowls")
    return split, len(centers) - split


def _arrow_direction(component: dict) -> str:
    midpoint = (component["x0"] + component["x1"]) / 2.0
    left_ink = sum(1 for _y, x in component["pixels"] if x < midpoint)
    right_ink = sum(1 for _y, x in component["pixels"] if x > midpoint)
    if abs(left_ink - right_ink) < max(18, int(component["area"] * 0.08)):
        raise ValueError("arrow direction has insufficient pixel margin")
    return "to_left" if left_ink > right_ink else "to_right"


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    if array.ndim != 3 or array.shape[2] != 3:
        raise ValueError("RGB image required")
    red, green, blue = array[:, :, 0], array[:, :, 1], array[:, :, 2]

    bead_mask = (
        (red > 175)
        & (green > 60)
        & (green < 190)
        & (blue < 125)
        & (red > green + 42)
        & (green > blue + 18)
    )
    arrow_mask = (
        (red < 110)
        & (green > 105)
        & (blue > 115)
        & (blue > red + 55)
        & (green > red + 45)
    )
    pip_mask = (
        (red > 90)
        & (blue > 105)
        & (green < 125)
        & (red > green + 18)
        & (blue > green + 28)
    )

    beads = [item for item in _components(bead_mask, 35) if 45 <= item["area"] <= 220]
    arrows = [item for item in _components(arrow_mask, 120) if item["area"] <= 1200]
    pips = [item for item in _components(pip_mask, 12) if item["area"] <= 160]
    left, right = _split_loads(beads)
    if len(arrows) != 4:
        raise ValueError(f"expected four transfer arrows, found {len(arrows)}")
    if len(pips) != 10:
        raise ValueError(f"expected ten ordering dots, found {len(pips)}")

    groups: list[list[dict]] = [[] for _ in arrows]
    for pip in pips:
        nearest = min(
            range(len(arrows)),
            key=lambda index: (pip["cx"] - arrows[index]["cx"]) ** 2
            + (pip["cy"] - arrows[index]["cy"]) ** 2,
        )
        groups[nearest].append(pip)
    counts = [len(group) for group in groups]
    if sorted(counts) != [1, 2, 3, 4]:
        raise ValueError(f"ordering dots do not uniquely rank arrows: {counts}")

    ranked = sorted(
        ((_arrow_direction(arrows[index]), counts[index]) for index in range(4)),
        key=lambda item: item[1],
    )
    for direction, _rank in ranked:
        if direction == "to_left":
            if right <= 0:
                raise ValueError("transfer uses an empty right source")
            left, right = left + 1, right - 1
        else:
            if left <= 0:
                raise ValueError("transfer uses an empty left source")
            left, right = left - 1, right + 1
    if left > right:
        return "left"
    if right > left:
        return "right"
    return "same"
