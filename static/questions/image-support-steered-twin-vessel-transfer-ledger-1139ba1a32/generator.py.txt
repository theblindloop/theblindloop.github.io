"""Deterministic scene sampler, renderer, and analytic arm."""

from __future__ import annotations

import copy
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH = 512
HEIGHT = 512
SCALE = 2
TOTAL_BEADS = 10

_PALETTES = (
    {
        "bead": (235, 139, 52),
        "bead_edge": (124, 73, 30),
        "arrow": (28, 156, 183),
        "pip": (125, 72, 158),
    },
    {
        "bead": (224, 116, 45),
        "bead_edge": (116, 58, 28),
        "arrow": (48, 132, 196),
        "pip": (155, 73, 150),
    },
)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = ("left", "same", "right")[int(seed) % 3]
    desired_left = {"left": 6, "same": 5, "right": 4}[target]

    while True:
        transfers = [rng.choice(("to_left", "to_right")) for _ in range(4)]
        if len(set(transfers)) != 2:
            continue
        net_left = sum(1 if item == "to_left" else -1 for item in transfers)
        initial_left = desired_left - net_left
        initial_right = TOTAL_BEADS - initial_left
        if not (2 <= initial_left <= 8 and 2 <= initial_right <= 8):
            continue
        left, right = initial_left, initial_right
        valid = True
        for item in transfers:
            if item == "to_left":
                if right == 0:
                    valid = False
                    break
                left, right = left + 1, right - 1
            else:
                if left == 0:
                    valid = False
                    break
                left, right = left - 1, right + 1
        if valid and left == desired_left:
            break

    slots = list(range(4))
    rng.shuffle(slots)
    decorations = [
        ["triangle", 74 + rng.randrange(-8, 9), 238 + rng.randrange(-10, 11)],
        ["diamond", 438 + rng.randrange(-8, 9), 242 + rng.randrange(-10, 11)],
    ]
    return {
        "initial_counts": [initial_left, initial_right],
        "transfers": transfers,
        "card_slots": slots,
        "palette_id": rng.randrange(len(_PALETTES)),
        "decorations": decorations,
    }


def _final_counts(scene: dict) -> tuple[int, int]:
    initial = scene["initial_counts"]
    transfers = scene["transfers"]
    if not isinstance(initial, list) or len(initial) != 2:
        raise ValueError("initial_counts must contain two loads")
    if not isinstance(transfers, list) or len(transfers) != 4:
        raise ValueError("transfers must contain four commands")
    left, right = int(initial[0]), int(initial[1])
    if left + right != TOTAL_BEADS or min(left, right) < 0:
        raise ValueError("invalid conserved starting inventory")
    for item in transfers:
        if item == "to_left":
            if right <= 0:
                raise ValueError("empty right source")
            left, right = left + 1, right - 1
        elif item == "to_right":
            if left <= 0:
                raise ValueError("empty left source")
            left, right = left - 1, right + 1
        else:
            raise ValueError("unknown transfer")
    return left, right


def analytic_gold(scene: dict) -> str:
    left, right = _final_counts(scene)
    if left > right:
        return "left"
    if right > left:
        return "right"
    return "same"


def margin(scene: dict) -> float:
    left, right = _final_counts(scene)
    return float(abs(left - right) if left != right else 2)


def is_quarantined(scene: dict) -> bool:
    try:
        left, right = _final_counts(scene)
    except (KeyError, TypeError, ValueError):
        return True
    return left < 0 or right < 0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["decorations"] = list(reversed(twin["decorations"]))
    return [("reverse_decoration_storage", twin)]


def _font(size: int, *, bold: bool = False) -> ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size * SCALE)
    except OSError:
        return ImageFont.load_default()


def _xy(values: Iterable[float]) -> tuple[int, ...]:
    return tuple(int(round(value * SCALE)) for value in values)


def _center_text(
    draw: ImageDraw.ImageDraw,
    center_x: float,
    y: float,
    text: str,
    font: ImageFont.ImageFont,
    fill: tuple[int, int, int],
) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    width = box[2] - box[0]
    draw.text((int(center_x * SCALE - width / 2), int(y * SCALE)), text, font=font, fill=fill)


def _draw_bowl(
    draw: ImageDraw.ImageDraw,
    center_x: int,
    count: int,
    label: str,
    palette: dict,
) -> None:
    ink = (58, 62, 70)
    _center_text(draw, center_x, 71, label, _font(13, bold=True), ink)
    draw.rounded_rectangle(
        _xy((center_x - 58, 92, center_x + 58, 163)),
        radius=12 * SCALE,
        fill=(254, 253, 249),
        outline=(91, 96, 105),
        width=3 * SCALE,
    )
    rows = [count] if count <= 4 else [4, count - 4]
    ys = [126] if len(rows) == 1 else [113, 141]
    radius = 7
    for row_count, cy in zip(rows, ys, strict=True):
        spacing = 17
        start = center_x - spacing * (row_count - 1) / 2
        for index in range(row_count):
            cx = start + index * spacing
            draw.ellipse(
                _xy((cx - radius, cy - radius, cx + radius, cy + radius)),
                fill=palette["bead"],
                outline=palette["bead_edge"],
                width=2 * SCALE,
            )


def _draw_arrow(draw: ImageDraw.ImageDraw, cx: int, cy: int, direction: str, color: tuple) -> None:
    points = [(-22, -5), (6, -5), (6, -14), (24, 0), (6, 14), (6, 5), (-22, 5)]
    if direction == "to_left":
        points = [(-x, y) for x, y in points]
    draw.polygon([(int((cx + x) * SCALE), int((cy + y) * SCALE)) for x, y in points], fill=color)


def _draw_card(
    draw: ImageDraw.ImageDraw,
    cx: int,
    rank: int,
    direction: str,
    palette: dict,
) -> None:
    draw.rounded_rectangle(
        _xy((cx - 29, 278, cx + 29, 383)),
        radius=8 * SCALE,
        fill=(253, 252, 248),
        outline=(121, 125, 132),
        width=2 * SCALE,
    )
    spacing = 10
    start = cx - spacing * (rank - 1) / 2
    for index in range(rank):
        px = start + index * spacing
        draw.ellipse(_xy((px - 4, 293 - 4, px + 4, 293 + 4)), fill=palette["pip"])
    draw.line(_xy((cx - 20, 310, cx + 20, 310)), fill=(220, 218, 211), width=2 * SCALE)
    _draw_arrow(draw, cx, 344, direction, palette["arrow"])


def _draw_decoration(draw: ImageDraw.ImageDraw, item: list) -> None:
    shape, x, y = item
    fill = (222, 219, 210)
    if shape == "triangle":
        draw.polygon([_xy((x, y - 12)), _xy((x - 11, y + 9)), _xy((x + 11, y + 9))], fill=fill)
    elif shape == "diamond":
        draw.polygon([_xy((x, y - 12)), _xy((x - 11, y)), _xy((x, y + 12)), _xy((x + 11, y))], fill=fill)


def render(scene: dict) -> Image.Image:
    initial = scene["initial_counts"]
    transfers = scene["transfers"]
    slots = scene["card_slots"]
    palette_id = int(scene["palette_id"])
    decorations = scene["decorations"]
    if sorted(int(value) for value in slots) != [0, 1, 2, 3]:
        raise ValueError("card_slots must be a permutation")
    if len(transfers) != 4:
        raise ValueError("four transfers are required")
    palette = _PALETTES[palette_id]
    canvas = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), (248, 246, 240))
    draw = ImageDraw.Draw(canvas)
    ink = (48, 52, 60)
    _center_text(draw, WIDTH / 2, 24, "TWIN-VESSEL LEDGER", _font(20, bold=True), ink)
    _center_text(draw, WIDTH / 2, 51, "STARTING LOADS", _font(11, bold=True), (105, 107, 111))
    _draw_bowl(draw, 170, int(initial[0]), "LEFT", palette)
    _draw_bowl(draw, 342, int(initial[1]), "RIGHT", palette)

    _center_text(draw, WIDTH / 2, 207, "READ PURPLE DOTS  1  •  2  •  3  •  4", _font(12, bold=True), ink)
    for item in sorted(decorations, key=lambda value: (int(value[2]), int(value[1]), str(value[0]))):
        _draw_decoration(draw, item)

    centers = (148, 220, 292, 364)
    for sequence_index, direction in enumerate(transfers):
        slot = int(slots[sequence_index])
        _draw_card(draw, centers[slot], sequence_index + 1, direction, palette)

    _center_text(draw, WIDTH / 2, 414, "EACH ARROW MOVES ONE BEAD", _font(13, bold=True), ink)
    _center_text(draw, WIDTH / 2, 440, "tail bowl  →  head bowl", _font(13), (82, 86, 93))
    _center_text(draw, WIDTH / 2, 469, "FINAL:  LEFT   /   SAME   /   RIGHT", _font(14, bold=True), ink)
    return canvas.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
