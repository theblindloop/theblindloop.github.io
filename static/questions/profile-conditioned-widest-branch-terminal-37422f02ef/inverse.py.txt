"""Independent image-only graph reconstruction and width-guided traversal."""

from __future__ import annotations

import numpy as np
from PIL import Image

PALETTES = {
    "green": (35, 164, 88),
    "cyan": (25, 183, 194),
    "orange": (234, 116, 28),
    "purple": (137, 73, 188),
    "blue": (42, 105, 207),
    "red": (205, 55, 63),
}
TERMINALS = ("orange", "purple", "blue", "red")


def _components(mask: np.ndarray, minimum: int = 12) -> list[np.ndarray]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        points = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for ny, nx in ((cy-1,cx), (cy+1,cx), (cy,cx-1), (cy,cx+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= minimum:
            output.append(np.asarray(points, dtype=np.float64))
    return output


def _color_components(arr: np.ndarray, rgb: tuple[int, int, int]) -> list[np.ndarray]:
    delta = arr.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return _components((delta * delta).sum(axis=2) < 55 * 55, minimum=80)


def _centroid(component: np.ndarray) -> np.ndarray:
    return component[:, ::-1].mean(axis=0)


def _markers(arr: np.ndarray) -> list[dict]:
    markers = []
    starts = _color_components(arr, PALETTES["green"])
    if len(starts) != 1:
        raise ValueError("expected one green start")
    markers.append({"kind": "start", "center": _centroid(starts[0])})
    junctions = _color_components(arr, PALETTES["cyan"])
    if len(junctions) != 3:
        raise ValueError("expected three cyan junctions")
    markers.extend({"kind": "junction", "center": _centroid(c)} for c in junctions)
    for color in TERMINALS:
        found = _color_components(arr, PALETTES[color])
        if len(found) != 1:
            raise ValueError(f"expected one {color} terminal")
        markers.append({"kind": "terminal", "color": color, "center": _centroid(found[0])})
    return markers


def _graph(arr: np.ndarray, markers: list[dict]) -> dict[int, list[tuple[int, int]]]:
    spread = arr.max(axis=2).astype(int) - arr.min(axis=2).astype(int)
    dark = (arr.max(axis=2) < 105) & (spread < 35)
    links = _components(dark, minimum=80)
    graph: dict[int, list[tuple[int, int]]] = {i: [] for i in range(len(markers))}
    for pixels_yx in links:
        pixels = pixels_yx[:, ::-1]
        incident = []
        for index, marker in enumerate(markers):
            d2 = ((pixels - marker["center"]) ** 2).sum(axis=1)
            if float(d2.min()) < 31.0 ** 2:
                incident.append(index)
        if len(incident) == 2:
            a, b = incident
            ink = len(pixels_yx)
            graph[a].append((b, ink))
            graph[b].append((a, ink))
    if sorted(len(v) for v in graph.values()) != [1, 1, 1, 1, 1, 3, 3, 3]:
        raise ValueError("route graph reconstruction failed")
    return graph


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    markers = _markers(arr)
    graph = _graph(arr, markers)
    current = next(i for i, m in enumerate(markers) if m["kind"] == "start")
    previous = None
    for _ in range(4):
        marker = markers[current]
        if marker["kind"] == "terminal":
            return marker["color"]
        options = [(n, ink) for n, ink in graph[current] if n != previous]
        if previous is None:
            if len(options) != 1:
                raise ValueError("start must have one forward link")
            chosen = options[0][0]
        else:
            if marker["kind"] != "junction" or len(options) != 2:
                raise ValueError("expected a binary junction")
            options.sort(key=lambda item: item[1])
            if options[1][1] < options[0][1] * 1.55:
                raise ValueError("outgoing width comparison is ambiguous")
            chosen = options[1][0]
        previous, current = current, chosen
    raise ValueError("navigation did not reach a terminal")
