"""Analytic sampler and raster renderer for local-width-guided navigation."""

from __future__ import annotations

import random
from copy import deepcopy
from PIL import Image, ImageDraw

SIZE = 600
BG = (250, 249, 246)
DARK = (42, 47, 54)
GREEN = (35, 164, 88)
CYAN = (25, 183, 194)
TERMINAL_RGB = {
    "orange": (234, 116, 28),
    "purple": (137, 73, 188),
    "blue": (42, 105, 207),
    "red": (205, 55, 63),
}
COLORS = tuple(TERMINAL_RGB)
NARROW = 6
WIDE = 14

# start, root junction, upper/lower junction, then four terminals.
POINTS = [
    (60, 300), (180, 300), (340, 170), (340, 430),
    (540, 90), (540, 250), (540, 350), (540, 510),
]
EDGES = [
    (0, 1, [(60, 300), (180, 300)]),
    (1, 2, [(180, 300), (180, 170), (340, 170)]),
    (1, 3, [(180, 300), (180, 430), (340, 430)]),
    (2, 4, [(340, 170), (340, 90), (540, 90)]),
    (2, 5, [(340, 170), (340, 250), (540, 250)]),
    (3, 6, [(340, 430), (340, 350), (540, 350)]),
    (3, 7, [(340, 430), (340, 510), (540, 510)]),
]


def _rotate_point(point: tuple[int, int], turns: int) -> tuple[int, int]:
    x, y = point
    for _ in range(turns % 4):
        x, y = SIZE - y, x
    return x, y


def _winner(branch_widths: list[int]) -> int:
    upper = branch_widths[0] > branch_widths[1]
    if upper:
        return 0 if branch_widths[2] > branch_widths[3] else 1
    return 2 if branch_widths[4] > branch_widths[5] else 3


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    widths: list[int] = []
    for _ in range(3):
        pair = [NARROW, WIDE]
        rng.shuffle(pair)
        widths.extend(pair)
    terminal_colors = list(COLORS)
    rng.shuffle(terminal_colors)
    winner = _winner(widths)
    target = COLORS[seed % len(COLORS)]
    target_index = terminal_colors.index(target)
    terminal_colors[winner], terminal_colors[target_index] = (
        terminal_colors[target_index], terminal_colors[winner]
    )
    return {
        "rotation": (seed // 4) % 4,
        "branch_widths": widths,
        "terminal_colors": terminal_colors,
    }


def analytic_gold(scene: dict) -> str:
    return scene["terminal_colors"][_winner(list(scene["branch_widths"]))]


def margin(scene: dict) -> float:
    widths = list(scene["branch_widths"])
    first = abs(widths[0] - widths[1])
    second = abs(widths[2] - widths[3]) if widths[0] > widths[1] else abs(widths[4] - widths[5])
    return float(min(first, second))


def is_quarantined(scene: dict) -> bool:
    widths = list(scene["branch_widths"])
    return len(widths) != 6 or margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    rotation = int(scene["rotation"])
    widths = list(scene["branch_widths"])
    terminal_colors = list(scene["terminal_colors"])
    if len(widths) != 6 or len(terminal_colors) != 4:
        raise ValueError("malformed scene")
    scale = 4
    image = Image.new("RGB", (SIZE * scale, SIZE * scale), BG)
    draw = ImageDraw.Draw(image)

    def pts(values):
        return [tuple(v * scale for v in _rotate_point(p, rotation)) for p in values]

    draw.line(pts(EDGES[0][2]), fill=DARK, width=9 * scale, joint="curve")
    for index, (_a, _b, polyline) in enumerate(EDGES[1:]):
        draw.line(pts(polyline), fill=DARK, width=int(widths[index]) * scale, joint="curve")

    transformed = [_rotate_point(p, rotation) for p in POINTS]

    def ellipse(center, radius, **kwargs):
        x, y = center
        draw.ellipse(((x-radius)*scale, (y-radius)*scale, (x+radius)*scale, (y+radius)*scale), **kwargs)

    def rectangle(center, half, **kwargs):
        x, y = center
        draw.rectangle(((x-half)*scale, (y-half)*scale, (x+half)*scale, (y+half)*scale), **kwargs)

    ellipse(transformed[0], 24, fill=BG)
    ellipse(transformed[0], 17, fill=GREEN)
    for center in transformed[1:4]:
        ellipse(center, 24, fill=BG)
        ellipse(center, 16, fill=CYAN)
    for index, center in enumerate(transformed[4:]):
        rectangle(center, 24, fill=BG)
        rectangle(center, 17, fill=TERMINAL_RGB[terminal_colors[index]])
    return image.resize((SIZE, SIZE), Image.Resampling.LANCZOS)


def rotated_invariance(scene: dict) -> dict:
    twin = deepcopy(scene)
    twin["rotation"] = (int(twin["rotation"]) + 1) % 4
    return twin
