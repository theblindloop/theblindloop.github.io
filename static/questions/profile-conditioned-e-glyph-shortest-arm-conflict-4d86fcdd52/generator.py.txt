"""Deterministic renderer for a visible E-glyph proportion conflict."""

from __future__ import annotations

import hashlib
import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 520
HEIGHT = 360
ANSWERS = ("top", "middle", "bottom")
BACKGROUNDS = ("#ffffff", "#f7f5ef", "#f2f6fa", "#faf3f6")
INKS = ("#17233b", "#33203d", "#16382f", "#35291c")


def _rng(seed: int, channel: str) -> random.Random:
    payload = f"e-shortest-arm-v1|{int(seed)}|{channel}".encode("utf-8")
    return random.Random(int.from_bytes(hashlib.sha256(payload).digest()[:8], "big"))


def sample_scene(seed: int) -> dict:
    """Balance answer slot and sample styling independently of that slot."""
    seed = int(seed)
    answer_index = seed % 3
    style = _rng(seed, "style")
    near_boundary = (seed // 3) % 7 == 0

    if near_boundary:
        short_length = style.choice((142, 146, 150))
        long_length = short_length + style.choice((12, 14, 16))
    else:
        short_length = style.choice((104, 112, 120, 128))
        long_length = style.choice((184, 196, 208, 220))
    arm_lengths = [long_length, long_length, long_length]
    arm_lengths[answer_index] = short_length

    stem_width = style.choice((22, 26, 30))
    arm_thickness = style.choice((20, 24, 28))
    glyph_height = style.choice((210, 224, 238))
    x_offset = style.randint(72, 132)
    y_offset = style.randint(48, HEIGHT - glyph_height - 38)
    paint_order = [0, 1, 2, 3]
    _rng(seed, "paint").shuffle(paint_order)

    palette = _rng(seed, "palette")
    return {
        "arm_lengths": arm_lengths,
        "background": palette.choice(BACKGROUNDS),
        "ink": palette.choice(INKS),
        "x_offset": x_offset,
        "y_offset": y_offset,
        "stem_width": stem_width,
        "arm_thickness": arm_thickness,
        "glyph_height": glyph_height,
        "paint_order": paint_order,
    }


def _rectangles(scene: dict) -> list[tuple[int, int, int, int]]:
    lengths = [int(value) for value in scene["arm_lengths"]]
    x = int(scene["x_offset"])
    y = int(scene["y_offset"])
    stem_width = int(scene["stem_width"])
    thickness = int(scene["arm_thickness"])
    height = int(scene["glyph_height"])
    middle_y = y + (height - thickness) // 2
    arm_ys = (y, middle_y, y + height - thickness)
    stem = (x, y, x + stem_width - 1, y + height - 1)
    arms = [
        (x, arm_y, x + length - 1, arm_y + thickness - 1)
        for arm_y, length in zip(arm_ys, lengths, strict=True)
    ]
    return [stem, *arms]


def render(scene: dict) -> Image.Image:
    lengths = scene["arm_lengths"]
    if len(lengths) != 3 or min(int(value) for value in lengths) <= int(scene["stem_width"]):
        raise ValueError("scene must contain three arms extending beyond the stem")
    if lengths.count(min(lengths)) != 1:
        raise ValueError("scene must expose one uniquely shortest arm")
    image = Image.new("RGB", (WIDTH, HEIGHT), scene["background"])
    draw = ImageDraw.Draw(image)
    rectangles = _rectangles(scene)
    order = [int(value) for value in scene["paint_order"]]
    if sorted(order) != [0, 1, 2, 3]:
        raise ValueError("paint_order must be a permutation of four primitives")
    for index in order:
        draw.rectangle(rectangles[index], fill=scene["ink"])
    return image


def analytic_gold(scene: dict) -> str:
    lengths = [int(value) for value in scene["arm_lengths"]]
    if len(lengths) != 3 or lengths.count(min(lengths)) != 1:
        raise ValueError("analytic gold requires one uniquely shortest arm")
    return ANSWERS[lengths.index(min(lengths))]


def margin(scene: dict) -> float:
    ordered = sorted(int(value) for value in scene["arm_lengths"])
    return float(ordered[1] - ordered[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 16.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["paint_order"] = list(reversed(scene["paint_order"]))
    return [("reverse_same_ink_primitive_paint_order", twin)]
