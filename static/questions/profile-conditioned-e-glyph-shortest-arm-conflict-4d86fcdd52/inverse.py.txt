"""Independent PNG-only inverse arm for the connected E-like silhouette."""

from __future__ import annotations

import numpy as np
from PIL import Image


class AmbiguousRasterError(ValueError):
    """The raster does not expose exactly three comparable arm extents."""


def _runs(values: np.ndarray) -> list[tuple[int, int]]:
    padded = np.pad(values.astype(np.int8), (1, 1))
    changes = np.diff(padded)
    starts = np.flatnonzero(changes == 1)
    stops = np.flatnonzero(changes == -1)
    return [(int(a), int(b)) for a, b in zip(starts, stops, strict=True)]


def _visible_arm_lengths(image: Image.Image) -> tuple[int, int, int]:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    if arr.ndim != 3 or arr.shape[0] < 120 or arr.shape[1] < 180:
        raise AmbiguousRasterError("raster is too small for the public glyph contract")

    corners = np.concatenate(
        (
            arr[:8, :8].reshape(-1, 3),
            arr[:8, -8:].reshape(-1, 3),
            arr[-8:, :8].reshape(-1, 3),
            arr[-8:, -8:].reshape(-1, 3),
        ),
        axis=0,
    )
    background = np.median(corners, axis=0)
    foreground = np.linalg.norm(arr - background, axis=2) > 60.0
    ys, xs = np.nonzero(foreground)
    if len(xs) < 2000:
        raise AmbiguousRasterError("too little visible ink to form the glyph")
    x0, x1 = int(xs.min()), int(xs.max()) + 1
    y0, y1 = int(ys.min()), int(ys.max()) + 1
    mask = foreground[y0:y1, x0:x1]
    if mask.shape[0] < 150 or mask.shape[1] < 90:
        raise AmbiguousRasterError("visible glyph extent is below the supported scale")

    widths = np.zeros(mask.shape[0], dtype=np.int32)
    for row in range(mask.shape[0]):
        columns = np.flatnonzero(mask[row])
        if len(columns):
            widths[row] = int(columns.max() - columns.min() + 1)
    positive = widths[widths > 0]
    if len(positive) != mask.shape[0]:
        raise AmbiguousRasterError("the vertical stem is visibly broken")
    stem_width = int(np.percentile(positive, 25))
    arm_rows = widths > max(stem_width * 2, int(mask.shape[1] * 0.34))
    bands = [(a, b) for a, b in _runs(arm_rows) if b - a >= 12]
    if len(bands) != 3:
        raise AmbiguousRasterError(f"expected three visible horizontal arm bands, found {len(bands)}")

    lengths = tuple(int(np.median(widths[a:b])) for a, b in bands)
    ordered = sorted(lengths)
    if ordered[1] - ordered[0] < 4:
        raise AmbiguousRasterError("shortest arm is not separated from the runner-up")
    if lengths.count(ordered[0]) != 1:
        raise AmbiguousRasterError("shortest arm is not unique")
    return lengths


def decision_from_image(image: Image.Image) -> str:
    lengths = _visible_arm_lengths(image)
    return ("top", "middle", "bottom")[lengths.index(min(lengths))]
