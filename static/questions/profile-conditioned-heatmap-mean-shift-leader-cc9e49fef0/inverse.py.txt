"""Independent inverse arm: recover row mean shifts from RGB pixels only."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

PALETTE = (
    (239, 247, 255), (222, 235, 247), (198, 219, 239),
    (158, 202, 225), (107, 174, 214), (66, 146, 198),
    (33, 113, 181), (8, 81, 156), (8, 48, 107),
)
LABELS = {
    "red": (214, 62, 73),
    "green": (31, 145, 92),
    "orange": (230, 126, 34),
    "purple": (126, 77, 171),
}


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    """Return (area, x0, y0, x1, y1) for four-connected components."""
    work = mask.copy()
    height, width = work.shape
    found = []
    ys, xs = np.nonzero(work)
    for sy, sx in zip(ys.tolist(), xs.tolist()):
        if not work[sy, sx]:
            continue
        work[sy, sx] = False
        queue = deque([(sy, sx)])
        area = 0
        x0 = x1 = sx
        y0 = y1 = sy
        while queue:
            y, x = queue.popleft()
            area += 1
            x0, x1 = min(x0, x), max(x1, x)
            y0, y1 = min(y0, y), max(y1, y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        found.append((area, x0, y0, x1, y1))
    return found


def _cluster(values: list[float], tolerance: float = 7.0) -> list[float]:
    groups: list[list[float]] = []
    for value in sorted(values):
        if not groups or value - sum(groups[-1]) / len(groups[-1]) > tolerance:
            groups.append([value])
        else:
            groups[-1].append(value)
    return [sum(group) / len(group) for group in groups]


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"))
    cells: list[tuple[float, float, int]] = []
    for level, rgb in enumerate(PALETTE):
        mask = np.all(pixels == np.asarray(rgb, dtype=np.uint8), axis=2)
        for area, x0, y0, x1, y1 in _components(mask):
            width, height = x1 - x0 + 1, y1 - y0 + 1
            if area > 1200 and width > 35 and height > 30:
                cells.append(((x0 + x1) / 2, (y0 + y1) / 2, level))
    if len(cells) != 32:
        raise ValueError(f"expected 32 recoverable heatmap cells, found {len(cells)}")

    row_centers = _cluster([item[1] for item in cells])
    col_centers = _cluster([item[0] for item in cells])
    if len(row_centers) != 4 or len(col_centers) != 8:
        raise ValueError("could not recover the four-row, two-panel grid")

    matrix = np.full((4, 8), np.nan)
    for x, y, level in cells:
        r = min(range(4), key=lambda i: abs(y - row_centers[i]))
        c = min(range(8), key=lambda i: abs(x - col_centers[i]))
        matrix[r, c] = level
    if np.isnan(matrix).any():
        raise ValueError("incomplete cell grid")
    gains = matrix[:, 4:].mean(axis=1) - matrix[:, :4].mean(axis=1)
    order = np.argsort(gains)
    if gains[order[-1]] - gains[order[-2]] < 0.75:
        raise ValueError("ambiguous top-two row gain")
    winner_y = row_centers[int(order[-1])]

    recovered = []
    for name, rgb in LABELS.items():
        mask = np.all(pixels == np.asarray(rgb, dtype=np.uint8), axis=2)
        parts = [part for part in _components(mask) if part[0] > 150]
        if len(parts) != 1:
            raise ValueError(f"could not recover {name} row label")
        area, x0, y0, x1, y1 = parts[0]
        recovered.append((abs((y0 + y1) / 2 - winner_y), name))
    recovered.sort()
    if recovered[0][0] > 8:
        raise ValueError("winner row has no aligned color label")
    return recovered[0][1]
