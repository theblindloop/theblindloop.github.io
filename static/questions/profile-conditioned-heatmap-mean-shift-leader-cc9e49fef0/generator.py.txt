"""Deterministic renderer for paired heatmap row-mean changes."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 620, 310
ROWS, COLS = 4, 4
CELL_W, CELL_H, GAP = 46, 40, 4
PANEL_X = (78, 350)
GRID_Y = 58

PALETTE = (
    (239, 247, 255),
    (222, 235, 247),
    (198, 219, 239),
    (158, 202, 225),
    (107, 174, 214),
    (66, 146, 198),
    (33, 113, 181),
    (8, 81, 156),
    (8, 48, 107),
)
ROW_RGB = {
    "red": (214, 62, 73),
    "green": (31, 145, 92),
    "orange": (230, 126, 34),
    "purple": (126, 77, 171),
}
ROW_NAMES = tuple(ROW_RGB)


def _gains(scene: dict) -> list[float]:
    a = scene["values_a"]
    b = scene["values_b"]
    return [sum(b[r]) / COLS - sum(a[r]) / COLS for r in range(ROWS)]


def sample_scene(seed: int) -> dict:
    """Make a balanced scene with a stable, visibly recoverable winner."""
    rng = random.Random(int(seed))
    winner = int(seed) % ROWS
    runner = (winner + 1 + rng.randrange(ROWS - 1)) % ROWS

    # Rotations counterbalance answer color against physical row over consecutive seeds.
    shift = (int(seed) // ROWS) % ROWS
    row_colors = [ROW_NAMES[(r + shift) % ROWS] for r in range(ROWS)]

    values_a = [[rng.randint(2, 5) for _ in range(COLS)] for _ in range(ROWS)]
    change_patterns = [[-1, -1, 0, 0] for _ in range(ROWS)]
    change_patterns[runner] = [0, 1, 0, 1]
    change_patterns[winner] = [2, 2, 2, 2]
    # Permuting cells prevents a fixed column from carrying the row statistic.
    values_b = []
    for r in range(ROWS):
        pattern = list(change_patterns[r])
        rng.shuffle(pattern)
        values_b.append([values_a[r][c] + pattern[c] for c in range(COLS)])
    return {"values_a": values_a, "values_b": values_b, "row_colors": row_colors}


def render(scene: dict) -> Image.Image:
    values_a = scene["values_a"]
    values_b = scene["values_b"]
    row_colors = scene["row_colors"]
    image = Image.new("RGB", (WIDTH, HEIGHT), (250, 250, 248))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    draw.text((267, 13), "ROW MEAN CHANGE", fill=(28, 33, 40), font=font)
    for panel_index, (panel_x, values, title) in enumerate(
        ((PANEL_X[0], values_a, "A"), (PANEL_X[1], values_b, "B"))
    ):
        draw.text((panel_x + 92, 38), title, fill=(20, 25, 32), font=font)
        for r in range(ROWS):
            for c in range(COLS):
                x0 = panel_x + c * (CELL_W + GAP)
                y0 = GRID_Y + r * (CELL_H + GAP)
                value = int(values[r][c])
                draw.rectangle((x0, y0, x0 + CELL_W - 1, y0 + CELL_H - 1), fill=PALETTE[value])
        draw.rectangle(
            (panel_x - 2, GRID_Y - 2, panel_x + COLS * (CELL_W + GAP) - GAP + 1,
             GRID_Y + ROWS * (CELL_H + GAP) - GAP + 1),
            outline=(70, 76, 84), width=2,
        )

    for r, name in enumerate(row_colors):
        cy = GRID_Y + r * (CELL_H + GAP) + CELL_H // 2
        draw.ellipse((35, cy - 10, 55, cy + 10), fill=ROW_RGB[name], outline=(45, 45, 45), width=1)

    legend_y = 260
    draw.text((205, legend_y + 4), "LOW", fill=(40, 45, 52), font=font)
    for i, rgb in enumerate(PALETTE):
        x0 = 240 + i * 19
        draw.rectangle((x0, legend_y, x0 + 17, legend_y + 17), fill=rgb)
    draw.text((418, legend_y + 4), "HIGH", fill=(40, 45, 52), font=font)
    draw.text((215, 286), "darker blue = larger value", fill=(45, 50, 58), font=font)
    return image


def analytic_gold(scene: dict) -> str:
    gains = _gains(scene)
    winner = max(range(ROWS), key=lambda r: gains[r])
    return str(scene["row_colors"][winner])


def margin(scene: dict) -> float:
    ordered = sorted(_gains(scene), reverse=True)
    return float(ordered[0] - ordered[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
