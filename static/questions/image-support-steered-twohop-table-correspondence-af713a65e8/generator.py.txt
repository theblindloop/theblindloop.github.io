"""Deterministic visual world for composing two correspondence tables."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

CANVAS = 1200
BACKGROUND = (248, 250, 252)
INK = (35, 43, 54)
MIDDLE_BLUE = (37, 99, 235)
TARGET_GREEN = (22, 150, 82)
KEY_GRAY = (107, 114, 128)
PAIR_GRAY = (188, 194, 202)
TAG_RGB = {
    "amber": (217, 119, 6),
    "cyan": (8, 145, 178),
    "violet": (124, 58, 237),
}
TAGS = tuple(TAG_RGB)

# Five deliberately separated card centers. On the public 8x8 support probe
# these occupy (row, col) cells (2,2), (2,5), (5,1), (5,3), and (5,5).
KEY_CENTERS = ((375, 375), (825, 375))
PROPOSAL_CENTERS = ((225, 825), (525, 825), (825, 825))
CARD_HALF_W = 34
CARD_HALF_H = 36
ROW_OFFSETS = (-22, 0, 22)


def _permutation(rng: random.Random) -> list[int]:
    values = [0, 1, 2]
    rng.shuffle(values)
    return values


def sample_scene(seed: int) -> dict:
    """Sample balanced bijections, full-relation distractors, rows, and tags."""

    rng = random.Random(int(seed) ^ 0x74A5EED)
    left_map = _permutation(rng)  # source -> middle
    right_map = _permutation(rng)  # middle -> target
    composed = [right_map[left_map[source]] for source in range(3)]
    distractor_a = [composed[(source + 1) % 3] for source in range(3)]
    distractor_b = [composed[(source + 2) % 3] for source in range(3)]

    correct_position = (int(seed) // 3) % 3
    proposals: list[list[int] | None] = [None, None, None]
    proposals[correct_position] = composed
    remaining_positions = [index for index in range(3) if index != correct_position]
    if rng.randrange(2):
        distractor_a, distractor_b = distractor_b, distractor_a
    proposals[remaining_positions[0]] = distractor_a
    proposals[remaining_positions[1]] = distractor_b

    answer_tag = TAGS[int(seed) % 3]
    proposal_tags: list[str | None] = [None, None, None]
    proposal_tags[correct_position] = answer_tag
    remaining_tags = [tag for tag in TAGS if tag != answer_tag]
    rng.shuffle(remaining_tags)
    proposal_tags[remaining_positions[0]] = remaining_tags[0]
    proposal_tags[remaining_positions[1]] = remaining_tags[1]

    row_orders = {
        "left": _permutation(rng),
        "right": _permutation(rng),
        "proposals": [_permutation(rng) for _ in range(3)],
    }
    return {
        "left_map": left_map,
        "right_map": right_map,
        "proposals": proposals,
        "row_orders": row_orders,
        "proposal_tags": proposal_tags,
    }


def _draw_source(draw: ImageDraw.ImageDraw, code: int, x: int, y: int) -> None:
    if code == 0:  # circle
        draw.ellipse((x - 7, y - 7, x + 7, y + 7), fill=INK)
    elif code == 1:  # square
        draw.rectangle((x - 7, y - 7, x + 7, y + 7), fill=INK)
    elif code == 2:  # triangle
        draw.polygon(((x, y - 8), (x - 8, y + 7), (x + 8, y + 7)), fill=INK)
    else:
        raise ValueError("unknown source glyph")


def _draw_middle(draw: ImageDraw.ImageDraw, code: int, x: int, y: int) -> None:
    offsets = {
        0: ((0, 0),),
        1: ((-4, 0), (4, 0)),
        2: ((-4, 3), (4, 3), (0, -4)),
    }
    if code not in offsets:
        raise ValueError("unknown middle glyph")
    for dx, dy in offsets[code]:
        draw.ellipse((x + dx - 2, y + dy - 2, x + dx + 2, y + dy + 2), fill=MIDDLE_BLUE)


def _draw_target(draw: ImageDraw.ImageDraw, code: int, x: int, y: int) -> None:
    if code == 0:  # horizontal bar
        draw.rounded_rectangle((x - 8, y - 3, x + 8, y + 3), radius=2, fill=TARGET_GREEN)
    elif code == 1:  # vertical bar
        draw.rounded_rectangle((x - 3, y - 8, x + 3, y + 8), radius=2, fill=TARGET_GREEN)
    elif code == 2:  # diamond
        draw.polygon(((x, y - 8), (x - 8, y), (x, y + 8), (x + 8, y)), fill=TARGET_GREEN)
    else:
        raise ValueError("unknown target glyph")


def _frame(draw: ImageDraw.ImageDraw, center: tuple[int, int], color: tuple[int, int, int]) -> None:
    cx, cy = center
    draw.rectangle(
        (cx - CARD_HALF_W, cy - CARD_HALF_H, cx + CARD_HALF_W, cy + CARD_HALF_H),
        outline=color,
        width=3,
    )


def _pair_mark(draw: ImageDraw.ImageDraw, x: int, y: int) -> None:
    draw.line((x - 4, y, x + 3, y), fill=PAIR_GRAY, width=2)
    draw.line((x + 1, y - 2, x + 4, y), fill=PAIR_GRAY, width=2)
    draw.line((x + 1, y + 2, x + 4, y), fill=PAIR_GRAY, width=2)


def _draw_left_key(scene: dict, draw: ImageDraw.ImageDraw) -> None:
    cx, cy = KEY_CENTERS[0]
    _frame(draw, (cx, cy), KEY_GRAY)
    for row, source in enumerate(scene["row_orders"]["left"]):
        y = cy + ROW_OFFSETS[row]
        _draw_source(draw, source, cx - 20, y)
        _pair_mark(draw, cx, y)
        _draw_middle(draw, scene["left_map"][source], cx + 20, y)


def _draw_right_key(scene: dict, draw: ImageDraw.ImageDraw) -> None:
    cx, cy = KEY_CENTERS[1]
    _frame(draw, (cx, cy), KEY_GRAY)
    for row, middle in enumerate(scene["row_orders"]["right"]):
        y = cy + ROW_OFFSETS[row]
        _draw_middle(draw, middle, cx - 20, y)
        _pair_mark(draw, cx, y)
        _draw_target(draw, scene["right_map"][middle], cx + 20, y)


def _draw_proposal(scene: dict, draw: ImageDraw.ImageDraw, index: int) -> None:
    cx, cy = PROPOSAL_CENTERS[index]
    _frame(draw, (cx, cy), TAG_RGB[scene["proposal_tags"][index]])
    for row, source in enumerate(scene["row_orders"]["proposals"][index]):
        y = cy + ROW_OFFSETS[row]
        _draw_source(draw, source, cx - 20, y)
        _pair_mark(draw, cx, y)
        _draw_target(draw, scene["proposals"][index][source], cx + 20, y)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (CANVAS, CANVAS), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_left_key(scene, draw)
    _draw_right_key(scene, draw)
    for index in range(3):
        _draw_proposal(scene, draw, index)
    return image


def analytic_gold(scene: dict) -> str:
    left = scene["left_map"]
    right = scene["right_map"]
    proposals = scene["proposals"]
    tags = scene["proposal_tags"]
    composed = [right[left[source]] for source in range(3)]
    matches = [index for index, proposal in enumerate(proposals) if proposal == composed]
    if len(matches) != 1:
        raise ValueError("scene has no unique composed proposal")
    return tags[matches[0]]


def margin(scene: dict) -> float:
    left = scene["left_map"]
    right = scene["right_map"]
    composed = [right[left[source]] for source in range(3)]
    distances = [
        sum(int(proposal[source] != composed[source]) for source in range(3))
        for proposal in scene["proposals"]
        if proposal != composed
    ]
    return float(min(distances)) if distances else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        left = list(scene["left_map"])
        right = list(scene["right_map"])
        proposals = [list(value) for value in scene["proposals"]]
        tags = list(scene["proposal_tags"])
        orders = scene["row_orders"]
        if sorted(left) != [0, 1, 2] or sorted(right) != [0, 1, 2]:
            return True
        if len(proposals) != 3 or any(sorted(value) != [0, 1, 2] for value in proposals):
            return True
        if sorted(tags) != sorted(TAGS):
            return True
        if sorted(orders["left"]) != [0, 1, 2] or sorted(orders["right"]) != [0, 1, 2]:
            return True
        if len(orders["proposals"]) != 3 or any(
            sorted(value) != [0, 1, 2] for value in orders["proposals"]
        ):
            return True
        composed = [right[left[source]] for source in range(3)]
        if sum(proposal == composed for proposal in proposals) != 1:
            return True
        return margin(scene) < 2.0
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # Every scene field is visible and this world declares no pixel aliases.
    del scene
    return []
