"""Independent pixel-only inverse arm for Two-Hop Table Correspondence.

This module intentionally knows only the public visual vocabulary. It discovers
the five frames, extracts and classifies local glyphs, builds two relations,
joins them through blue-dot cardinality, and compares all proposal relations.
"""

from __future__ import annotations

from dataclasses import dataclass

import numpy as np
from PIL import Image

INK = np.array((35, 43, 54), dtype=np.int32)
MIDDLE_BLUE = np.array((37, 99, 235), dtype=np.int32)
TARGET_GREEN = np.array((22, 150, 82), dtype=np.int32)
KEY_GRAY = np.array((107, 114, 128), dtype=np.int32)
TAG_COLORS = {
    "amber": np.array((217, 119, 6), dtype=np.int32),
    "cyan": np.array((8, 145, 178), dtype=np.int32),
    "violet": np.array((124, 58, 237), dtype=np.int32),
}


@dataclass(frozen=True)
class Component:
    x0: int
    y0: int
    x1: int
    y1: int
    area: int
    cx: float
    cy: float

    @property
    def width(self) -> int:
        return self.x1 - self.x0 + 1

    @property
    def height(self) -> int:
        return self.y1 - self.y0 + 1


class OracleAbstention(ValueError):
    pass


def _mask(array: np.ndarray, color: np.ndarray, tolerance: float = 8.0) -> np.ndarray:
    delta = array.astype(np.int32) - color
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _components(mask: np.ndarray, minimum_area: int = 1) -> list[Component]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[Component] = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        stack = [(sy, sx)]
        seen[sy, sx] = True
        xs: list[int] = []
        ys: list[int] = []
        while stack:
            y, x = stack.pop()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(xs) >= minimum_area:
            output.append(
                Component(
                    min(xs), min(ys), max(xs), max(ys), len(xs),
                    float(sum(xs) / len(xs)), float(sum(ys) / len(ys)),
                )
            )
    return output


def _shifted_components(
    mask: np.ndarray, minimum_area: int, *, offset_x: int = 0, offset_y: int = 0
) -> list[Component]:
    """Run flood fill only inside the nonempty mask bounding box."""

    points = np.argwhere(mask)
    if not len(points):
        return []
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    local = _components(mask[y0 : y1 + 1, x0 : x1 + 1], minimum_area)
    return [
        Component(
            item.x0 + int(x0) + offset_x,
            item.y0 + int(y0) + offset_y,
            item.x1 + int(x0) + offset_x,
            item.y1 + int(y0) + offset_y,
            item.area,
            item.cx + int(x0) + offset_x,
            item.cy + int(y0) + offset_y,
        )
        for item in local
    ]


def _large_frames(array: np.ndarray, color: np.ndarray, expected: int) -> list[Component]:
    candidates = [
        item
        for item in _shifted_components(_mask(array, color), 120)
        if item.width > 35 and item.height > 35
    ]
    valid: list[Component] = []
    raw = _mask(array, color)
    for item in candidates:
        crop = raw[item.y0 : item.y1 + 1, item.x0 : item.x1 + 1]
        top = float(crop[:3].mean())
        bottom = float(crop[-3:].mean())
        left = float(crop[:, :3].mean())
        right = float(crop[:, -3:].mean())
        if min(top, bottom, left, right) >= 0.90:
            valid.append(item)
    if len(valid) != expected:
        raise OracleAbstention(f"expected {expected} intact frames, found {len(valid)}")
    return valid


def _inside(item: Component, frame: Component, inset: int = 4) -> bool:
    return (
        frame.x0 + inset < item.cx < frame.x1 - inset
        and frame.y0 + inset < item.cy < frame.y1 - inset
    )


def _glyph_components(array: np.ndarray, color: np.ndarray, frame: Component, minimum: int) -> list[Component]:
    x0, y0 = frame.x0 + 4, frame.y0 + 4
    x1, y1 = frame.x1 - 3, frame.y1 - 3
    crop = array[y0:y1, x0:x1]
    return _shifted_components(
        _mask(crop, color), minimum, offset_x=x0, offset_y=y0
    )


def _source_code(item: Component) -> int:
    fill = item.area / float(item.width * item.height)
    if fill > 0.94:
        return 1  # square
    if fill > 0.68:
        return 0  # circle
    if 0.42 < fill < 0.68:
        return 2  # triangle
    raise OracleAbstention("ambiguous dark source glyph")


def _target_code(item: Component) -> int:
    ratio = item.width / float(item.height)
    if ratio > 1.65:
        return 0  # horizontal
    if ratio < 0.61:
        return 1  # vertical
    fill = item.area / float(item.width * item.height)
    if 0.42 < fill < 0.68:
        return 2  # diamond
    raise OracleAbstention("ambiguous green target glyph")


def _require_inventory(mapping: dict[int, int]) -> dict[int, int]:
    if set(mapping) != {0, 1, 2} or set(mapping.values()) != {0, 1, 2}:
        raise OracleAbstention("table is not a complete bijection")
    return mapping


def _dot_identity_by_rows(dots: list[Component], rows: list[Component]) -> list[int]:
    if len(dots) != 6 or len(rows) != 3:
        raise OracleAbstention("incomplete blue-dot inventory")
    counts = [0, 0, 0]
    for dot in dots:
        distances = [abs(dot.cy - row.cy) for row in rows]
        nearest = int(np.argmin(distances))
        if distances[nearest] > 8.0:
            raise OracleAbstention("blue dots do not align with a table row")
        counts[nearest] += 1
    if sorted(counts) != [1, 2, 3]:
        raise OracleAbstention("blue-dot groups are not unique identities")
    return [count - 1 for count in counts]


def _parse_left_key(array: np.ndarray, frame: Component) -> dict[int, int]:
    sources = sorted(_glyph_components(array, INK, frame, 50), key=lambda item: item.cy)
    dots = _glyph_components(array, MIDDLE_BLUE, frame, 8)
    if len(sources) != 3:
        raise OracleAbstention("left key lacks three source shapes")
    middle_codes = _dot_identity_by_rows(dots, sources)
    mapping: dict[int, int] = {}
    for source, middle in zip(sources, middle_codes, strict=True):
        code = _source_code(source)
        if code in mapping:
            raise OracleAbstention("left key repeats a source")
        mapping[code] = middle
    return _require_inventory(mapping)


def _parse_right_key(array: np.ndarray, frame: Component) -> dict[int, int]:
    targets = sorted(_glyph_components(array, TARGET_GREEN, frame, 45), key=lambda item: item.cy)
    dots = _glyph_components(array, MIDDLE_BLUE, frame, 8)
    if len(targets) != 3:
        raise OracleAbstention("right key lacks three targets")
    middle_codes = _dot_identity_by_rows(dots, targets)
    mapping: dict[int, int] = {}
    for target, middle in zip(targets, middle_codes, strict=True):
        if middle in mapping:
            raise OracleAbstention("right key repeats a middle identity")
        mapping[middle] = _target_code(target)
    return _require_inventory(mapping)


def _parse_proposal(array: np.ndarray, frame: Component) -> dict[int, int]:
    sources = sorted(_glyph_components(array, INK, frame, 50), key=lambda item: item.cy)
    targets = sorted(_glyph_components(array, TARGET_GREEN, frame, 45), key=lambda item: item.cy)
    if len(sources) != 3 or len(targets) != 3:
        raise OracleAbstention("proposal lacks a complete glyph inventory")
    mapping: dict[int, int] = {}
    for source, target in zip(sources, targets, strict=True):
        if abs(source.cy - target.cy) > 5.0:
            raise OracleAbstention("proposal glyphs do not form aligned rows")
        code = _source_code(source)
        if code in mapping:
            raise OracleAbstention("proposal repeats a source")
        mapping[code] = _target_code(target)
    return _require_inventory(mapping)


def decision_from_image(image: Image.Image) -> str:
    """Recover the answer from the final raster alone or explicitly abstain."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    border = np.concatenate(
        (array[:3].reshape(-1, 3), array[-3:].reshape(-1, 3),
         array[:, :3].reshape(-1, 3), array[:, -3:].reshape(-1, 3)),
        axis=0,
    )
    background = np.median(border, axis=0).astype(np.int32)
    delta = array.astype(np.int32) - background
    foreground = np.sum(delta * delta, axis=2) > 12 * 12
    points = np.argwhere(foreground)
    if not len(points):
        raise OracleAbstention("image has no visible tables")
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    pad = 4
    y0, x0 = max(0, int(y0) - pad), max(0, int(x0) - pad)
    y1 = min(array.shape[0], int(y1) + pad + 1)
    x1 = min(array.shape[1], int(x1) + pad + 1)
    array = array[y0:y1, x0:x1]
    key_frames = sorted(_large_frames(array, KEY_GRAY, 2), key=lambda item: item.cx)
    if abs(key_frames[0].cy - key_frames[1].cy) > max(key_frames[0].height, key_frames[1].height):
        raise OracleAbstention("gray keys do not share a row")
    left_map = _parse_left_key(array, key_frames[0])
    right_map = _parse_right_key(array, key_frames[1])

    proposals: dict[str, dict[int, int]] = {}
    for tag, color in TAG_COLORS.items():
        frame = _large_frames(array, color, 1)[0]
        proposals[tag] = _parse_proposal(array, frame)

    composed = {source: right_map[left_map[source]] for source in (0, 1, 2)}
    matches = [tag for tag, proposal in proposals.items() if proposal == composed]
    if len(matches) != 1:
        raise OracleAbstention("composed relation has no unique proposal")
    return matches[0]
