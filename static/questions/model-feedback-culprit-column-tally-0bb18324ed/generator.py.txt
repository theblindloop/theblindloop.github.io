"""Culprit-column tally world: which unseen deletion produced this much water?

The picture draws a block landscape exactly as it stood before the rain, four
black tags naming four candidate columns, and a pale blue tally strip whose
squares count every block of water the landscape holds *after* one of those
four columns was taken away.  The deletion itself is never drawn, and the
edited landscape is never drawn.  The label is the deletion.

Geometry is flat, integer-aligned, non-anti-aliased Pillow ink, so the raster
is byte-deterministic and the pixel-only arm can segment by exact colour.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 1160
HEIGHT = 680
N_COLS = 13
MAX_H = 8
BLOCK = 56          # lattice pitch of the landscape
INSET = 3           # background gutter on each side of a block
GROUND_TOP = 576
GROUND_THICK = 12
N_TAGS = 4
TAG_W = 40
TAG_H = 36
TAG_GAP = 12        # clear background between ground bar and a tag

TALLY_TOP = 40
TALLY_SIDE = 28
TALLY_PITCH = 40    # 28 px square + 12 px clear background
TALLY_GROUP = 4
TALLY_GROUP_GAP = 20
MAX_TALLY = 24
MIN_TALLY = 6

BG = (255, 255, 255)
ROCK = (70, 74, 82)
GROUND = (30, 32, 36)
TAG = (0, 0, 0)
TALLY = (150, 196, 236)

MIN_ANSWER_GAP = 3          # drawn tally clears every decoy tally by 3 blocks
QUARANTINE_MIN_MARGIN = 4.0 * TALLY_PITCH   # headline scenes clear it by 4 blocks


# ---------------------------------------------------------------- water rule

def _water(heights: list[int]) -> list[int]:
    """Standing-water depth over each column of a landscape open at both ends."""
    n = len(heights)
    pref = [0] * n
    suf = [0] * n
    run = 0
    for i in range(n):
        run = max(run, heights[i])
        pref[i] = run
    run = 0
    for i in range(n - 1, -1, -1):
        run = max(run, heights[i])
        suf[i] = run
    return [max(0, min(pref[i], suf[i]) - heights[i]) for i in range(n)]


def _total_water(heights: list[int]) -> int:
    return sum(_water(heights))


def _removed_profile(heights: list[int], col: int) -> list[int]:
    edited = list(heights)
    edited[col] = 0
    return edited


def _totals_after_removal(heights: list[int], tags: list[int]) -> list[int]:
    return [_total_water(_removed_profile(heights, t)) for t in tags]


# ------------------------------------------------------------------ sampling

def _origin_x() -> int:
    return (WIDTH - N_COLS * BLOCK) // 2


def _tally_width(total: int) -> int:
    if total <= 0:
        return 0
    groups = (total - 1) // TALLY_GROUP
    return total * TALLY_PITCH - (TALLY_PITCH - TALLY_SIDE) + groups * TALLY_GROUP_GAP


def sample_scene(seed: int) -> dict:
    """Sample a landscape, four tagged columns, and the hidden deletion."""
    rng = random.Random(seed * 2654435761 + 12345)
    # answer position is balanced by rejection so that no tag order is a prior
    target = (seed * 2246822519 + 374761393) % N_TAGS
    for attempt in range(24000):
        strict = attempt < 20000
        heights = [rng.randrange(0, MAX_H + 1) for _ in range(N_COLS)]
        factual = _total_water(heights)
        if factual < 4:
            continue
        pool = [i for i, h in enumerate(heights) if h >= 2]
        if len(pool) < N_TAGS + 1:
            continue
        tags = sorted(rng.sample(pool, N_TAGS))
        if min(b - a for a, b in zip(tags, tags[1:])) < 2:
            continue
        totals = _totals_after_removal(heights, tags)
        ordered = sorted(totals)
        if min(b - a for a, b in zip(ordered, ordered[1:])) < 2:
            continue
        if any(abs(t - factual) < 2 for t in totals):
            continue

        tall = max(range(N_TAGS), key=lambda k: heights[tags[k]])
        middle = [k for k in range(N_TAGS) if ordered[1] <= totals[k] <= ordered[2]]
        rng.shuffle(middle)
        for k in middle:
            total = totals[k]
            if k == tall:
                continue
            if strict and k != target:
                continue
            if min(abs(total - totals[j]) for j in range(N_TAGS) if j != k) < MIN_ANSWER_GAP:
                continue
            if not MIN_TALLY <= total <= MAX_TALLY:
                continue
            if _tally_width(total) > WIDTH - 2 * 40:
                continue
            # the local shortcut "the removal only floods its own footprint"
            # must point at a decoy, never at the culprit
            naive = [abs(factual + heights[tags[j]] - total) for j in range(N_TAGS)]
            if min(range(N_TAGS), key=lambda j: naive[j]) == k:
                continue
            paint = list(range(N_COLS))
            rng.shuffle(paint)
            return {
                "n_cols": N_COLS,
                "heights": heights,
                "tags": tags,
                "culprit": tags[k],
                "paint_order": paint,
            }
    raise RuntimeError(f"sampler exhausted for seed {seed}")


# -------------------------------------------------------------------- labels

def _culprit_rank(scene: dict) -> int:
    return scene["tags"].index(scene["culprit"])


def analytic_gold(scene: dict) -> str:
    """Which tagged column, counting 1 to 4 from the left, was taken away."""
    return f"tag-{_culprit_rank(scene) + 1}"


def drawn_total(scene: dict) -> int:
    """Number of blue tally squares: total water of the edited landscape."""
    return _total_water(_removed_profile(scene["heights"], scene["culprit"]))


def margin(scene: dict) -> float:
    """Pixel gap between the drawn tally and the nearest decoy's tally."""
    target = drawn_total(scene)
    others = [
        _total_water(_removed_profile(scene["heights"], t))
        for t in scene["tags"]
        if t != scene["culprit"]
    ]
    return float(min(abs(target - o) for o in others) * TALLY_PITCH)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MIN_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Column paint order is sampled but photometrically invisible."""
    order = list(scene["paint_order"])
    out = []
    rotated = dict(scene)
    rotated["paint_order"] = order[1:] + order[:1]
    out.append(("rotated_paint_order", rotated))
    reversed_ = dict(scene)
    reversed_["paint_order"] = list(reversed(order))
    out.append(("reversed_paint_order", reversed_))
    return out


# ------------------------------------------------------------------ renderer

def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)

    ox = _origin_x()

    # ground bar spanning the whole landscape
    draw.rectangle(
        [ox, GROUND_TOP, ox + N_COLS * BLOCK - 1, GROUND_TOP + GROUND_THICK - 1],
        fill=GROUND,
    )

    # landscape columns, painted in the scene's invisible order
    for col in scene["paint_order"]:
        h = scene["heights"][col]
        x0 = ox + col * BLOCK + INSET
        x1 = ox + (col + 1) * BLOCK - INSET - 1
        for level in range(h):
            y1 = GROUND_TOP - level * BLOCK - INSET - 1
            y0 = GROUND_TOP - (level + 1) * BLOCK + INSET
            draw.rectangle([x0, y0, x1, y1], fill=ROCK)

    # black tags under the ground bar
    ty0 = GROUND_TOP + GROUND_THICK + TAG_GAP
    for col in scene["tags"]:
        cx = ox + col * BLOCK + BLOCK // 2
        draw.rectangle(
            [cx - TAG_W // 2, ty0, cx - TAG_W // 2 + TAG_W - 1, ty0 + TAG_H - 1],
            fill=TAG,
        )

    # pale blue tally strip: one square per block of water in the edited world
    total = drawn_total(scene)
    strip = _tally_width(total)
    x = (WIDTH - strip) // 2
    for i in range(total):
        draw.rectangle([x, TALLY_TOP, x + TALLY_SIDE - 1, TALLY_TOP + TALLY_SIDE - 1],
                       fill=TALLY)
        x += TALLY_PITCH
        if (i + 1) % TALLY_GROUP == 0 and i + 1 < total:
            x += TALLY_GROUP_GAP

    return image
