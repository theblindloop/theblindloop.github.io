"""Independent pixel-only inverse arm for the culprit-column tally world.

Receives the rendered PNG and nothing else: no latent scene, no renderer, no
prompt, generator, verifier or analytic-gold code.  The ground bar, the block
lattice, the four black tags and the blue tally strip are recovered from the
raster; the water rule is reimplemented here; each of the four candidate
deletions is simulated and the one whose total water equals the counted tally
is reported.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_ROCK = (70, 74, 82)
_GROUND = (30, 32, 36)
_TALLY = (150, 196, 236)


def _mask(arr: np.ndarray, ref) -> np.ndarray:
    return np.all(np.abs(arr.astype(np.int16) - np.array(ref, dtype=np.int16)) <= 10, axis=2)


def _runs(present: np.ndarray) -> list[tuple[int, int]]:
    out = []
    x = 0
    n = present.shape[0]
    while x < n:
        if present[x]:
            start = x
            while x < n and present[x]:
                x += 1
            out.append((start, x - 1))
        else:
            x += 1
    return out


def _standing(heights: list[int]) -> int:
    n = len(heights)
    pref = [0] * n
    suf = [0] * n
    run = 0
    for i in range(n):
        run = max(run, heights[i])
        pref[i] = run
    run = 0
    for i in range(n - 1, -1, -1):
        run = max(run, heights[i])
        suf[i] = run
    return sum(max(0, min(pref[i], suf[i]) - heights[i]) for i in range(n))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))

    ground = _mask(arr, _GROUND)
    rows = np.flatnonzero(ground.any(axis=1))
    if rows.size == 0:
        raise ValueError("no ground bar found")
    ground_top = int(rows.min())
    ground_bot = int(rows.max())
    gcols = np.flatnonzero(ground[ground_top])
    ox, ex = int(gcols.min()), int(gcols.max())

    rock = _mask(arr, _ROCK)
    rock[ground_top:] = False
    rock_runs = _runs(rock.any(axis=0))
    if not rock_runs:
        raise ValueError("no landscape columns found")

    # lattice pitch: the run width plus its two background gutters, recovered
    # from the span of the ground bar and the number of cells it covers.
    widths = [b - a + 1 for a, b in rock_runs]
    cell = max(widths) + 6
    span = ex - ox + 1
    n_cols = int(round(span / cell))
    pitch = span / float(n_cols)

    heights = [0] * n_cols
    for col in range(n_cols):
        cx = int(ox + (col + 0.5) * pitch)
        level = 0
        while True:
            cy = int(ground_top - level * pitch - pitch / 2)
            if cy < 0 or not rock[cy, cx]:
                break
            heights[col] += 1
            level += 1

    dark = np.all(arr <= 20, axis=2)
    dark[: ground_bot + 1] = False
    tag_runs = _runs(dark.any(axis=0))
    if not tag_runs:
        raise ValueError("no tags found")
    tags = []
    for a, b in tag_runs:
        centre = (a + b) / 2.0
        col = int(round((centre - ox) / pitch - 0.5))
        tags.append(max(0, min(n_cols - 1, col)))
    tags = sorted(set(tags))

    tally = _mask(arr, _TALLY)
    trows = np.flatnonzero(tally.any(axis=1))
    if trows.size == 0:
        raise ValueError("no tally strip found")
    band = tally[int(trows.min()): int(trows.max()) + 1]
    total = len(_runs(band.any(axis=0)))

    hits = []
    for rank, col in enumerate(tags):
        edited = list(heights)
        edited[col] = 0
        if _standing(edited) == total:
            hits.append(rank)
    if len(hits) != 1:
        raise ValueError(f"tally {total} matches {len(hits)} candidate deletions")
    return f"tag-{hits[0] + 1}"
