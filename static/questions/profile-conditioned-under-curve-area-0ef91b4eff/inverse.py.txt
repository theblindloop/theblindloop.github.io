"""Independent scene-free pixel oracle for the under_curve_area world.

Recovers the largest-area color from the raster alone. The oracle imports no
renderer, prompt, generator, verifier, or analytic-gold code and never sees the
scene. It labels every pixel by its nearest palette color (so antialiased edges
split at the ~50% mix), then returns the color whose fill mask holds the most
pixels.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

PALETTE = {
    "red": (200, 40, 40),
    "blue": (40, 80, 200),
    "green": (40, 180, 70),
    "gray": (150, 150, 150),
    "white": (255, 255, 255),
}
COLORS = ["red", "blue", "green"]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    rgb_list = [np.array(value, dtype=np.float32) for value in PALETTE.values()]
    labels = np.argmin(
        np.stack([np.linalg.norm(arr - color, axis=-1) for color in rgb_list], axis=-1),
        axis=-1,
    )
    counts = [int((labels == index).sum()) for index in range(len(COLORS))]
    winner = max(range(len(COLORS)), key=lambda index: (counts[index], -index))
    return COLORS[winner]
