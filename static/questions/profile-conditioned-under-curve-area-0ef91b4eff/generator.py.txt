"""Under-curve area selection world renderer.

A three-lane plot. Each lane holds one horizontal gray baseline and one colored
filled region shaped like a "hill": a trapezoid that rises from the baseline at
both ends up to a flat plateau. The plateaus have independent widths ``w`` and
heights ``h``, so a region's enclosed area is a genuine two-dimensional product
``area = h * (w + slope_run)`` that cannot be read off any single length.

The analytic decision is the color of the region with the largest filled area.
The independent inverse arm (oracle.py) recovers that same color by counting
colored fill pixels in the raster alone.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "under_curve_area-0.1.0"

WIDTH = 360
HEIGHT = 360
LANE_SPAN = 300
SLOPE_RUN = 40
LEFT_X = 40
RIGHT_X = LEFT_X + LANE_SPAN  # 340
BASELINES = [112, 222, 332]

COLOR_RGB = {
    "red": (200, 40, 40),
    "blue": (40, 80, 200),
    "green": (40, 180, 70),
}
GRAY = (150, 150, 150)
WHITE = (255, 255, 255)

QUARANTINE_MARGIN = 0.08

COLORS = ["red", "blue", "green"]


def _area(trace: dict) -> float:
    return float(trace["h"] * (trace["w"] + SLOPE_RUN))


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + (seed % 1000))
    winner = COLORS[seed % 3]
    panels = [0, 1, 2]
    rng.shuffle(panels)
    traces = []
    for color in COLORS:
        if color == winner:
            h = rng.randint(50, 66)
            w = rng.randint(120, 176)
        else:
            h = rng.randint(22, 44)
            w = rng.randint(48, 96)
        traces.append({"color": color, "h": h, "w": w})
    if _area(traces[1]) == _area(traces[2]):
        traces[2]["h"] = max(22, min(44, traces[2]["h"] + 4))
    for index, trace in enumerate(traces):
        trace["panel"] = panels[index]
    return {"traces": traces}


def _baseline(panel: int) -> int:
    return BASELINES[panel]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), WHITE)
    draw = ImageDraw.Draw(image)
    for trace in scene["traces"]:
        color = trace["color"]
        h = trace["h"]
        w = trace["w"]
        panel = trace["panel"]
        baseline = _baseline(panel)
        top = baseline - h
        x1 = LEFT_X + SLOPE_RUN
        x2 = LEFT_X + SLOPE_RUN + w
        x3 = LEFT_X + 2 * SLOPE_RUN + w
        polygon = [
            (LEFT_X, baseline),
            (x1, top),
            (x2, top),
            (x3, baseline),
            (RIGHT_X, baseline),
        ]
        draw.polygon(polygon, fill=COLOR_RGB[color])
        draw.line([(LEFT_X - 4, baseline), (RIGHT_X + 4, baseline)], fill=GRAY, width=2)
    return image


def analytic_gold(scene: dict) -> str:
    best = max(scene["traces"], key=_area)
    return best["color"]


def margin(scene: dict) -> float:
    areas = sorted(_area(trace) for trace in scene["traces"])
    top, second = areas[-1], areas[-2]
    return float((top - second) / top) if top > 0 else 0.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list:
    return []
