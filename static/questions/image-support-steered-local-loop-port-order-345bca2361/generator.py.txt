"""Renderer and analytic arm for the compact closed-loop port-order world."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
SCALE = 4
BACKGROUND = (255, 255, 255)
BODY = (35, 78, 108)
ROUTE = (239, 229, 187)
BLACK = (26, 31, 38)
GREEN = (39, 174, 106)
AMBER = (242, 157, 40)
VIOLET = (176, 73, 183)


def _angle_point(center: tuple[float, float], radius: float, angle: float) -> tuple[float, float]:
    return (
        center[0] + radius * math.cos(angle),
        center[1] + radius * math.sin(angle),
    )


def _loop_points(scene: dict, samples: int = 192) -> list[tuple[float, float]]:
    center = (float(scene["tile_position"][0]), float(scene["tile_position"][1]))
    radius = float(scene["loop_geometry"]["radius"])
    return [
        _angle_point(center, radius, 2.0 * math.pi * index / samples)
        for index in range(samples)
    ]


def _port_point(scene: dict, slot: str) -> tuple[float, float]:
    center = (float(scene["tile_position"][0]), float(scene["tile_position"][1]))
    radius = float(scene["loop_geometry"]["radius"])
    angle = float(scene["start_port"]["angle"]) + float(scene["colored_port_offsets"][slot])
    return _angle_point(center, radius, angle)


def sample_scene(seed: int) -> dict:
    """Sample balanced absolute positions, loop phases, and port identities."""

    rng = random.Random(int(seed))
    center = [round(rng.uniform(145.0, 367.0), 4), round(rng.uniform(145.0, 367.0), 4)]
    radius = round(rng.uniform(42.0, 47.0), 4)
    tile_size = round(rng.uniform(124.0, 132.0), 4)
    corner_radius = round(rng.uniform(20.0, 26.0), 4)
    start_angle = round(rng.uniform(-math.pi, math.pi), 6)

    # Some scenes are deliberately close to the visible equality boundary,
    # while all generated examples keep enough marker clearance for a stable
    # finite-resolution inverse.
    if rng.randrange(8) in (0, 1, 2):
        gap_px = rng.uniform(24.0, 38.0)
    else:
        gap_px = rng.uniform(62.0, 112.0)
    gap = gap_px / radius
    first = rng.uniform(0.78, 1.72)
    if first + gap > 5.25:
        first = 5.25 - gap
    second = first + gap
    first_color = "amber" if rng.randrange(2) == 0 else "violet"
    second_color = "violet" if first_color == "amber" else "amber"

    return {
        "tile_position": center,
        "tile_geometry": {
            "size": tile_size,
            "corner_radius": corner_radius,
        },
        "loop_geometry": {
            "radius": radius,
            "stroke_width": 13.0,
        },
        "start_port": {
            "angle": start_angle,
        },
        "colored_port_offsets": {
            "first": round(first, 6),
            "second": round(second, 6),
        },
        "port_color_assignment": {
            "first": first_color,
            "second": second_color,
        },
    }


def _scaled_points(points: Iterable[tuple[float, float]]) -> list[tuple[int, int]]:
    return [(round(x * SCALE), round(y * SCALE)) for x, y in points]


def _draw_port(draw: ImageDraw.ImageDraw, point: tuple[float, float], color: tuple[int, int, int], inner_radius: float) -> None:
    x, y = round(point[0] * SCALE), round(point[1] * SCALE)
    outer = round((inner_radius + 3.0) * SCALE)
    inner = round(inner_radius * SCALE)
    draw.ellipse((x - outer, y - outer, x + outer, y + outer), fill=BLACK)
    draw.ellipse((x - inner, y - inner, x + inner, y + inner), fill=color)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    center = (float(scene["tile_position"][0]), float(scene["tile_position"][1]))
    size = float(scene["tile_geometry"]["size"])
    corner_radius = float(scene["tile_geometry"]["corner_radius"])
    half = size / 2.0
    draw.rounded_rectangle(
        (
            round((center[0] - half) * SCALE),
            round((center[1] - half) * SCALE),
            round((center[0] + half) * SCALE),
            round((center[1] + half) * SCALE),
        ),
        radius=round(corner_radius * SCALE),
        fill=BODY,
    )

    loop = _scaled_points(_loop_points(scene))
    loop.append(loop[0])
    stroke_width = round(float(scene["loop_geometry"]["stroke_width"]) * SCALE)
    draw.line(loop, fill=ROUTE, width=stroke_width, joint="curve")

    start = _angle_point(center, float(scene["loop_geometry"]["radius"]), float(scene["start_port"]["angle"]))
    _draw_port(draw, start, GREEN, 8.0)
    colors = {"amber": AMBER, "violet": VIOLET}
    for slot in ("first", "second"):
        _draw_port(
            draw,
            _port_point(scene, slot),
            colors[str(scene["port_color_assignment"][slot])],
            8.0,
        )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    offsets = scene["colored_port_offsets"]
    assignment = scene["port_color_assignment"]
    first_slot = "first" if float(offsets["first"]) < float(offsets["second"]) else "second"
    return str(assignment[first_slot])


def margin(scene: dict) -> float:
    radius = float(scene["loop_geometry"]["radius"])
    offsets = scene["colored_port_offsets"]
    return abs(float(offsets["second"]) - float(offsets["first"])) * radius


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 22.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # All label-bearing fields are rendered directly.  This scene model has no
    # unused latent alias to expose as a pixel-identical transform.
    return []
