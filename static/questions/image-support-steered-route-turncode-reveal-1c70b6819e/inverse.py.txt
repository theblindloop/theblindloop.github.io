"""Independent pixel-only inverse for the connected route curvature code."""

from __future__ import annotations

import numpy as np
from PIL import Image

N_TEETH = 32
LOOP = tuple(int(bit) for bit in "11110000100110011001100111110000")
LACE = tuple(int(bit) for bit in "00001111011001100110011000001111")


def _mask_components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        pixels: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            pixels.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    ny, nx = cy + dy, cx + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        if len(pixels) >= minimum:
            components.append(np.asarray(pixels, dtype=np.float64))
    return components


def _centers(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    return [component.mean(axis=0)[::-1] for component in _mask_components(mask, minimum)]


def _color_masks(array: np.ndarray) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    red, green, blue = [array[:, :, index].astype(np.int16) for index in range(3)]
    navy = (red < 125) & (green < 140) & (blue < 175) & ((blue - red) > 18)
    red_mask = (red > 135) & (red - green > 42) & (red - blue > 30) & (green < 155)
    green_mask = (green > 92) & (green - red > 22) & (green - blue > 2) & (red < 150)
    return navy, red_mask, green_mask


def _column_center(mask: np.ndarray, x: float, radius: int = 2) -> float:
    center = int(round(float(x)))
    x0 = max(0, center - radius)
    x1 = min(mask.shape[1], center + radius + 1)
    ys = np.argwhere(mask[:, x0:x1])[:, 0] if mask[:, x0:x1].any() else np.asarray([])
    if len(ys) < 3:
        raise ValueError("route interval is missing from the raster")
    return float(np.median(ys))


def _baseline_center(mask: np.ndarray, left: float, right: float) -> float:
    # Endpoint caps cover one of the two joins in the first and last tooth.
    # Use whichever visible join samples succeed; interior teeth contribute
    # both joins and make the local baseline estimate more stable.
    values = []
    for x in (left, right):
        try:
            values.append(_column_center(mask, x, radius=1))
        except ValueError:
            pass
    if not values:
        for x in (left + 2.5, right - 2.5):
            try:
                values.append(_column_center(mask, x))
            except ValueError:
                pass
    if not values:
        raise ValueError("both joins of a route tooth are missing")
    return float(np.median(values))


def _hamming(code: tuple[int, ...], template: tuple[int, ...]) -> int:
    return sum(a != b for a, b in zip(code, template))


def decision_from_image(image: Image.Image) -> str:
    """Locate caps, certify one connected route, and decode all 32 teeth."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    navy, red_mask, green_mask = _color_masks(array)
    red = _centers(red_mask, 25)
    green = _centers(green_mask, 25)
    if len(red) != 1 or len(green) != 1:
        raise ValueError(f"expected one red and one green endpoint, got {len(red)} and {len(green)}")
    route_components = _mask_components(navy, 60)
    if len(route_components) != 1:
        raise ValueError("the route is absent or not one connected component")
    route = route_components[0]
    if len(route) < 700:
        raise ValueError("route evidence is incomplete")
    start_x = float(red[0][0])
    end_x = float(green[0][0])
    left = min(start_x, end_x)
    right = max(start_x, end_x)
    if right - left < 400.0:
        raise ValueError("endpoint span is too short for the declared route")
    start_right = start_x > end_x
    pitch = (right - left) / float(N_TEETH)
    bits: list[int] = []
    for index in range(N_TEETH):
        if start_right:
            apex_x = right - (index + 0.5) * pitch
            tooth_right = right - index * pitch
            tooth_left = tooth_right - pitch
        else:
            apex_x = left + (index + 0.5) * pitch
            tooth_left = left + index * pitch
            tooth_right = tooth_left + pitch
        baseline = _baseline_center(navy, tooth_left, tooth_right)
        apex = _column_center(navy, apex_x)
        delta = apex - baseline
        if abs(delta) < 2.5:
            raise ValueError("route tooth curvature is ambiguous")
        bits.append(1 if delta < 0.0 else 0)
    code = tuple(bits)
    distance_a = _hamming(code, LOOP)
    distance_b = _hamming(code, LACE)
    return "A" if distance_a <= distance_b else "B"
