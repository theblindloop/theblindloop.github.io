"""Renderer and analytic arm for a connected route curvature-code question."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "route-turncode-reveal-0.1.0"
WIDTH = 640
HEIGHT = 480
N_TEETH = 32
PITCH = 16.0
ROUTE_LENGTH = N_TEETH * PITCH
BACKGROUND = (248, 248, 245)

# The two templates have equal ink, equal up/down counts, and differ at every
# route station. The route geometry, rather than total ink or position, carries
# the label.
LOOP = tuple(int(bit) for bit in "11110000100110011001100111110000")
LACE = tuple(int(bit) for bit in "00001111011001100110011000001111")
TEMPLATES = {"A": LOOP, "B": LACE}
DIFF_POSITIONS = tuple(index for index, (a, b) in enumerate(zip(LOOP, LACE)) if a != b)

PALETTES = (
    ((18, 40, 83), (211, 57, 61), (41, 145, 88)),
    ((25, 47, 91), (204, 62, 57), (35, 139, 98)),
    ((20, 45, 78), (218, 68, 54), (49, 151, 82)),
)


def _hamming(code: tuple[int, ...], template: tuple[int, ...]) -> int:
    return sum(a != b for a, b in zip(code, template))


def _classify_code(code: tuple[int, ...]) -> str:
    distance_a = _hamming(code, LOOP)
    distance_b = _hamming(code, LACE)
    return "A" if distance_a <= distance_b else "B"


def sample_scene(seed: int) -> dict:
    """Return one balanced deterministic scene from an integer seed."""

    rng = random.Random(int(seed))
    base_label = "A" if int(seed) % 2 == 0 else "B"
    code = list(TEMPLATES[base_label])
    # A tie population makes the observable boundary explicit. Mutating half
    # of the 32 disagreements gives equal distances to the displayed templates.
    if int(seed) % 9 == 0:
        for index in rng.sample(DIFF_POSITIONS, len(DIFF_POSITIONS) // 2):
            code[index] = 1 - code[index]
    return {
        "code": code,
        "x0": int(rng.choice((48, 64, 80, 96))),
        "center_y": int(rng.choice((100, 160, 220, 280, 340, 380))),
        "start_right": bool(rng.randrange(2)),
        "stroke": int(rng.choice((6, 7, 8))),
        "amplitude": int(rng.choice((10, 11, 12))),
        "palette_variant": int(rng.randrange(len(PALETTES))),
    }


def _route_points(scene: dict) -> list[tuple[float, float]]:
    code = tuple(int(value) for value in scene["code"])
    x0 = float(scene["x0"])
    y0 = float(scene["center_y"])
    amplitude = float(scene["amplitude"])
    start_right = bool(scene["start_right"])
    points: list[tuple[float, float]] = []
    for index, bit in enumerate(code):
        cell = N_TEETH - 1 - index if start_right else index
        left = x0 + cell * PITCH
        right = left + PITCH
        apex_x = (left + right) / 2.0
        apex_y = y0 - amplitude if bit else y0 + amplitude
        if start_right:
            if index == 0:
                points.append((right, y0))
            points.extend(((apex_x, apex_y), (left, y0)))
        else:
            if index == 0:
                points.append((left, y0))
            points.extend(((apex_x, apex_y), (right, y0)))
    return points


def _scaled_points(points: list[tuple[float, float]], scale: int) -> list[tuple[float, float]]:
    return [(round(x * scale), round(y * scale)) for x, y in points]


def render(scene: dict) -> Image.Image:
    """Render one anti-aliased RGB image from visible route geometry."""

    scale = 4
    navy, red, green = PALETTES[int(scene["palette_variant"]) % len(PALETTES)]
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    points = _scaled_points(_route_points(scene), scale)
    width = int(scene["stroke"]) * scale
    draw.line(points, fill=navy, width=width, joint="curve")
    radius = 10 * scale
    start = points[0]
    end = points[-1]
    draw.ellipse((start[0] - radius, start[1] - radius, start[0] + radius, start[1] + radius), fill=red)
    draw.ellipse((end[0] - radius, end[1] - radius, end[0] + radius, end[1] + radius), fill=green)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    code = tuple(int(value) for value in scene["code"])
    if len(code) != N_TEETH or any(value not in (0, 1) for value in code):
        raise ValueError("route code must contain 32 binary teeth")
    return _classify_code(code)


def margin(scene: dict) -> float:
    code = tuple(int(value) for value in scene["code"])
    return float(abs(_hamming(code, LOOP) - _hamming(code, LACE)))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # No latent alias is claimed: the visible code and every nuisance field are
    # represented in the rendered pixels, and this parameterization has no
    # hidden permutation of route stations.
    return []
