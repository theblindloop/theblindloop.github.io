"""Scene sampler, rasterizer and analytic gold for pipe_turn_handedness.

One thick gray rectilinear pipe runs across a white page. A green disc covers
one of its two loose ends. Walking from the green end and facing the direction
of travel, every bend is a right-angle turn either toward the walker's left hand
or toward the walker's right hand. The decision is the handedness of the THIRD
bend met from the green end.

Everything that fixes the label -- the ordered polyline `nodes` and which end
carries the green disc (always `nodes[0]`) -- is painted into the raster.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "pipe_turn_handedness-renderer-0.1.0"

CANVAS = (640, 460)
BACKGROUND = (255, 255, 255)
PIPE_COLOR = (120, 120, 120)
GREEN = (30, 150, 60)
STROKE_WIDTH = 9
DISC_RADIUS = 12

PITCH = 64
ORIGIN = (64, 52)
GRID_N = (9, 7)
JITTER = 18
TURN_INDEX = 3  # the third bend, counted from the green end
MIN_MARGIN_PX = 34.0

_DIRS = {"E": (1, 0), "W": (-1, 0), "N": (0, -1), "S": (0, 1)}


def _handedness(d1: tuple[int, int], d2: tuple[int, int]) -> str:
    """Screen coordinates have y growing downward, so a walker heading `d1` who
    leaves along `d2` turned toward the left hand exactly when the 2-D cross
    product is negative."""
    cross = d1[0] * d2[1] - d1[1] * d2[0]
    return "left" if cross < 0 else "right"


def _walk(rng: random.Random, n_turns: int) -> list[tuple[int, int]] | None:
    """Self-avoiding rectilinear lattice walk whose visited cells keep a
    Chebyshev clearance of two lattice units from every non-consecutive cell."""
    gx, gy = GRID_N
    start = (rng.randrange(gx), rng.randrange(gy))
    d = _DIRS[rng.choice("EWNS")]
    cells = [start]
    nodes = [start]
    cur = start
    for seg in range(n_turns + 1):
        if seg:
            options = [(0, -1), (0, 1)] if d[1] == 0 else [(-1, 0), (1, 0)]
            rng.shuffle(options)
        else:
            options = [d]
        placed = False
        for nd in options:
            for run in rng.sample([1, 2, 3], 3):
                trial = []
                ok = True
                p = cur
                for _ in range(run):
                    p = (p[0] + nd[0], p[1] + nd[1])
                    if not (0 <= p[0] < gx and 0 <= p[1] < gy):
                        ok = False
                        break
                    known = cells + trial
                    for j, c in enumerate(known):
                        if j >= len(known) - 2:
                            continue  # the cell just left, and the corner behind it
                        if max(abs(c[0] - p[0]), abs(c[1] - p[1])) < 2:
                            ok = False
                            break
                    if not ok:
                        break
                    trial.append(p)
                if ok and len(trial) == run:
                    cells.extend(trial)
                    cur = trial[-1]
                    nodes.append(cur)
                    d = nd
                    placed = True
                    break
            if placed:
                break
        if not placed:
            return None
    return nodes


def _dirs_of(nodes: list[tuple[float, float]]) -> list[tuple[int, int]]:
    out = []
    for a, b in zip(nodes, nodes[1:]):
        dx, dy = b[0] - a[0], b[1] - a[1]
        out.append((0 if abs(dx) < 1e-9 else (1 if dx > 0 else -1),
                    0 if abs(dy) < 1e-9 else (1 if dy > 0 else -1)))
    return out


def _turns(nodes: list[tuple[float, float]]) -> list[str]:
    ds = _dirs_of(nodes)
    return [_handedness(a, b) for a, b in zip(ds, ds[1:])]


def _to_pixels(rng: random.Random, cells: list[tuple[int, int]]) -> list[list[float]]:
    """Lattice corners -> jittered pixel corners; segment orientations and the
    turn sequence are preserved because every run is at least one pitch long and
    each end moves by at most JITTER."""
    ox, oy = ORIGIN
    raw = [(ox + c[0] * PITCH, oy + c[1] * PITCH) for c in cells]
    n_seg = len(raw) - 1
    off = [rng.randint(-JITTER, JITTER) for _ in range(n_seg)]
    horiz = [abs(raw[i][1] - raw[i + 1][1]) < 1e-9 for i in range(n_seg)]
    ends = [rng.randint(-JITTER, JITTER), rng.randint(-JITTER, JITTER)]
    pts: list[list[float]] = []
    for i in range(len(raw)):
        x, y = float(raw[i][0]), float(raw[i][1])
        for seg in (i - 1, i):
            if 0 <= seg < n_seg:
                if horiz[seg]:
                    y = float(raw[i][1] + off[seg])
                else:
                    x = float(raw[i][0] + off[seg])
        pts.append([x, y])
    # free coordinate of each loose end, along its own segment
    if horiz[0]:
        pts[0][0] += ends[0]
    else:
        pts[0][1] += ends[0]
    if horiz[-1]:
        pts[-1][0] += ends[1]
    else:
        pts[-1][1] += ends[1]
    return pts


def _seg_gap(a: list[float], b: list[float], c: list[float], d: list[float]) -> float:
    """Exact free gap between two axis-aligned strokes of width STROKE_WIDTH."""
    ax0, ax1 = sorted((a[0], b[0]))
    ay0, ay1 = sorted((a[1], b[1]))
    bx0, bx1 = sorted((c[0], d[0]))
    by0, by1 = sorted((c[1], d[1]))
    dx = max(bx0 - ax1, ax0 - bx1, 0.0)
    dy = max(by0 - ay1, ay0 - by1, 0.0)
    return math.hypot(dx, dy) - STROKE_WIDTH


def _geometry_margin(nodes: list[list[float]]) -> float:
    segs = list(zip(nodes, nodes[1:]))
    best = min(math.hypot(b[0] - a[0], b[1] - a[1]) for a, b in segs)
    for i in range(len(segs)):
        for j in range(i + 2, len(segs)):
            best = min(best, _seg_gap(segs[i][0], segs[i][1], segs[j][0], segs[j][1]))
    return round(float(best), 4)


def _in_canvas(nodes: list[list[float]]) -> bool:
    pad = STROKE_WIDTH + DISC_RADIUS + 2
    return all(pad <= p[0] <= CANVAS[0] - pad and pad <= p[1] <= CANVAS[1] - pad
               for p in nodes)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 104729)
    want_third = rng.choice(["left", "right"])
    want_first = rng.choice(["left", "right"])
    want_major = rng.choice(["left", "right"])
    want_pair = rng.choice([True, False])  # does bend 2 turn the same way as bend 3?
    n_turns = rng.choice([4, 5, 6, 7])
    for attempt in range(6000):
        cells = _walk(rng, n_turns)
        if cells is None:
            continue
        pts = _to_pixels(rng, cells)
        if not _in_canvas(pts):
            continue
        turns = _turns(pts)
        if len(turns) != n_turns or turns[TURN_INDEX - 1] != want_third:
            continue
        if attempt < 5000 and (turns[TURN_INDEX - 2] == want_third) != want_pair:
            continue
        if attempt < 4000 and turns[0] != want_first:
            continue
        if attempt < 2500:
            major = "left" if turns.count("left") * 2 > len(turns) else "right"
            if major != want_major:
                continue
        return {
            "seed": int(seed),
            "canvas": list(CANVAS),
            "background": list(BACKGROUND),
            "pipe_color": list(PIPE_COLOR),
            "green_color": list(GREEN),
            "stroke_width": STROKE_WIDTH,
            "disc_radius": DISC_RADIUS,
            "nodes": [[round(p[0], 3), round(p[1], 3)] for p in pts],
        }
    raise RuntimeError(f"no admissible pipe for seed {seed}")


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", tuple(scene["canvas"]), tuple(scene["background"]))
    drw = ImageDraw.Draw(img)
    h = scene["stroke_width"] // 2
    pipe = tuple(scene["pipe_color"])
    nodes = [(round(p[0]), round(p[1])) for p in scene["nodes"]]
    for a, b in zip(nodes, nodes[1:]):
        x0, x1 = sorted((a[0], b[0]))
        y0, y1 = sorted((a[1], b[1]))
        drw.rectangle([x0 - h, y0 - h, x1 + h, y1 + h], fill=pipe)
    for p in nodes:  # square caps and mitred corners
        drw.rectangle([p[0] - h, p[1] - h, p[0] + h, p[1] + h], fill=pipe)
    r = scene["disc_radius"]
    sx, sy = nodes[0]
    drw.ellipse([sx - r, sy - r, sx + r, sy + r], fill=tuple(scene["green_color"]))
    return img


def analytic_gold(scene: dict) -> str:
    turns = _turns([tuple(p) for p in scene["nodes"]])
    return turns[TURN_INDEX - 1]


def margin(scene: dict) -> float:
    return _geometry_margin([list(p) for p in scene["nodes"]])


def is_quarantined(scene: dict) -> bool:
    nodes = [tuple(p) for p in scene["nodes"]]
    if len(_turns(nodes)) < TURN_INDEX:
        return True
    return margin(scene) < MIN_MARGIN_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist: every scene field is painted, and the label is a
    function of the drawn polyline alone."""
    return []
