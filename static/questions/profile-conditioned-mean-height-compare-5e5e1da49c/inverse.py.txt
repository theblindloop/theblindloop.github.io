"""Independent pixel-only inverse arm for the mean-height comparison world.

This module receives ONLY the rasterized PNG and re-measures which color's bars
have the greater average height. It shares no geometry with renderer.render()
and never imports the renderer, prompts, generator, verifier, or analytic gold.

Pipeline: (1) label every pixel to its nearest fixed color (red, blue, or
light background); (2) for each color find the connected components of that
color's ink, i.e. the individual bars; (3) measure each bar's height as the
vertical extent of its bounding box; (4) compute each color's mean bar height
(sum / count) and return the color with the larger mean.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "mean_height_compare-oracle-0.1.0"

RED_RGB = (205, 28, 38)
BLUE_RGB = (20, 60, 200)
BACKGROUND = (249, 249, 249)
BASELINE_RGB = (90, 90, 90)

_REF = np.array(
    [
        BACKGROUND,
        RED_RGB,
        BLUE_RGB,
    ],
    dtype=np.float32,
)


def _label(arr: np.ndarray) -> np.ndarray:
    """Label every pixel 0=background, 1=red, 2=blue (nearest fixed color)."""
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - _REF[np.newaxis, np.newaxis, :, :]
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def _bar_heights(labels: np.ndarray, color_label: int) -> list[float]:
    """Vertical extent (bounding-box height) of each connected bar of a color."""
    ys, xs = np.nonzero(labels == color_label)
    if len(ys) == 0:
        return []
    pixels = set(zip(xs.tolist(), ys.tolist()))
    heights: list[float] = []
    while pixels:
        seed = pixels.pop()
        stack = [seed]
        member = []
        while stack:
            px, py = stack.pop()
            member.append((px, py))
            for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                nb = (px + dx, py + dy)
                if nb in pixels:
                    pixels.discard(nb)
                    stack.append(nb)
        comp_ys = [y for (_x, y) in member]
        heights.append(float(int(np.max(comp_ys)) - int(np.min(comp_ys)) + 1))
    return heights


def _mean(values: list[float]) -> float:
    if not values:
        return 0.0
    return float(sum(values) / len(values))


def decision_from_image(image: Image.Image) -> str:
    """Return the color whose bars have the greater mean height."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label(arr)
    mean_red = _mean(_bar_heights(labels, 1))
    mean_blue = _mean(_bar_heights(labels, 2))
    if mean_red > mean_blue:
        return "red"
    return "blue"
