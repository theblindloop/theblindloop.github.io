"""Mean-height comparison world: which colored bar set has the greater average height.

A light panel shows two classes of vertical bars, one red and one blue, all
standing on a common light-gray baseline. The two color classes hold a
DIFFERENT number of bars (uneven set sizes). The question asks which color's
bars have the greater AVERAGE height, i.e. the larger per-bar mean height. This
is a *measurement/aggregate* decision: the pixel arm must count the bars of each
color and divide each color's total height by that count, then compare the two
means.

Because the two classes have unequal counts, the total ink (the sum of all a
color's bar heights) does not determine the winner: a color with few very tall
bars can lose to a color with many medium bars even though it has more total
ink in places, and vice versa. Only the division that yields the per-bar average
separates the two colors.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "mean_height_compare-0.1.0"

RED_RGB = (205, 28, 38)
BLUE_RGB = (20, 60, 200)
BACKGROUND = (249, 249, 249)
BASELINE_RGB = (216, 216, 216)

WIDTH = 560
HEIGHT = 320

BASELINE_Y = 290.0
BAR_WIDTH = 30.0
SLOT_EDGE_GAP = 10.0
BASELINE_THICKNESS = 3

HEIGHT_LO = 45.0
HEIGHT_HI = 205.0
MIN_BAR_HEIGHT = 42.0
MIN_MEAN_MARGIN = 5.0
QUARANTINE_MEAN_MARGIN = 12.0
COUNT_POOL = (2, 3, 4)


def _mean(bar_heights: list[float]) -> float:
    return float(sum(bar_heights) / len(bar_heights))


def sample_scene(seed: int) -> dict:
    """Sample a deterministic scene with a clearly separated mean decision."""
    rng = random.Random(seed * 131 + 17)
    for _attempt in range(20000):
        n_red = rng.choice(COUNT_POOL)
        n_blue = rng.choice(COUNT_POOL)
        if n_red == n_blue:
            continue
        heights = {
            "red": [float(rng.randint(int(HEIGHT_LO), int(HEIGHT_HI))) for _ in range(n_red)],
            "blue": [float(rng.randint(int(HEIGHT_LO), int(HEIGHT_HI))) for _ in range(n_blue)],
        }
        if min(_mean(heights["red"]), _mean(heights["blue"])) < MIN_BAR_HEIGHT:
            continue
        mv = abs(_mean(heights["red"]) - _mean(heights["blue"]))
        if mv < MIN_MEAN_MARGIN:
            continue
        # Randomize which color cluster sits on the left, so no positional cue
        # is coupled to a color label.
        order = rng.sample(["red", "blue"], 2)

        bars: list[dict] = []
        cursor = 24.0
        for color in order:
            for h in heights[color]:
                bars.append({"color": color, "x": cursor, "height": h})
                cursor += BAR_WIDTH + SLOT_EDGE_GAP

        scene = {
            "bars": bars,
            "width": WIDTH,
            "height": HEIGHT,
            "background": "light",
        }
        return scene
    raise RuntimeError("could not sample a separated mean-height scene")


def _slot_y(height: float) -> float:
    return BASELINE_Y - height


def render(scene: dict) -> Image.Image:
    """Render I = R(z): a full-width baseline plus colored vertical bar rectangles."""
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    draw.line(
        [(0, BASELINE_Y), (scene["width"], BASELINE_Y)],
        fill=BASELINE_RGB,
        width=BASELINE_THICKNESS,
    )
    for bar in scene["bars"]:
        color = bar["color"]
        rgb = RED_RGB if color == "red" else BLUE_RGB
        x = float(bar["x"])
        top = _slot_y(float(bar["height"]))
        bottom = BASELINE_Y
        draw.rectangle([x, top, x + BAR_WIDTH, bottom], fill=rgb)
    return img


def _grouped(scene: dict) -> dict:
    groups: dict[str, list[float]] = {"red": [], "blue": []}
    for bar in scene["bars"]:
        groups[bar["color"]].append(float(bar["height"]))
    return groups


def analytic_gold(scene: dict) -> str:
    """Color of the bar class with the greater MEAN height."""
    groups = _grouped(scene)
    mean_red = _mean(groups["red"])
    mean_blue = _mean(groups["blue"])
    if mean_red > mean_blue:
        return "red"
    return "blue"


def margin(scene: dict) -> float:
    """Absolute difference between the two recovered per-bar mean heights."""
    groups = _grouped(scene)
    return abs(_mean(groups["red"]) - _mean(groups["blue"]))


def is_quarantined(scene: dict) -> bool:
    """Near-boundary: mean-height difference is uncomfortably small."""
    return margin(scene) < QUARANTINE_MEAN_MARGIN


def latent_symmetries(scene: dict) -> list:
    """not_applicable: every latent field is drawn verbatim as visible ink.

    Each bar's color, x position, and height are all rendered as filled
    rectangles of that color, and the analytic decision reads only these
    directly rendered bar fields. There is no hidden scene field that could
    change the mean comparison without changing the raster, so no nontrivial
    pixel-identical, gold-preserving latent transformation exists.
    """
    return []
