"""Scene sampling, rendering and analytic gold for `unpaired_window_plate`.

Five coloured plates -- red, orange, green, blue and purple -- sit far apart on
a large page, one plate per widely separated grid cell. Every plate is a solid
slab with a number of rectangular windows punched clean through it, so a plate
carries as many holes as it has windows. Exactly four of the five plates split
into two pairs whose members have equally many windows; the remaining plate has
a window count shared with no other coloured plate. That unpaired plate, and
its own window count, are the decision.

The page is littered with grey slabs of the same construction. Grey ink is
decoration and never counts.

Nothing local decides the answer: a window count is only "unpaired" relative to
the four other counts, which live in distant corners of the page, so the answer
binds five disconnected components spread across the raster.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = 1024
GRID = 8
CELL = CANVAS // GRID              # 128
CELL_INSET = 10                    # keeps a plate inside one ablation window

WHITE = (255, 255, 255)
COLOR_NAMES = ("red", "orange", "green", "blue", "purple")
PLATE_INK = {
    "red": (204, 0, 0),
    "orange": (235, 132, 0),
    "green": (0, 138, 62),
    "blue": (24, 62, 196),
    "purple": (128, 40, 176),
}
DECOR_INK = ((140, 140, 140), (95, 95, 95), (185, 185, 185), (40, 40, 40))

# widely separated anchor cells: pairwise Chebyshev distance of three cells
ANCHOR_ROWS = (0, 3, 6)
ANCHOR_COLS = (0, 3, 6)

WALL_CHOICES = (8, 9, 10)
WIN_CHOICES = (12, 14, 16, 18)
GAP_CHOICES = (8, 10, 12)
SLOT_GRIDS = ((1, 1), (1, 2), (1, 3), (2, 2), (2, 3))
COUNT_CHOICES = (1, 2, 3, 4, 5)

QUARANTINE_MARGIN = 6.0


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------
def _plate_body(rng: random.Random, windows: int, scale: str) -> dict:
    """Geometry of one slab carrying `windows` punched windows."""
    if scale == "tight":
        wall, win, gap = 5, 5, 5
    elif scale == "snug":
        wall, win, gap = 7, 7, 7
    else:
        wall = rng.choice(WALL_CHOICES)
        win = rng.choice(WIN_CHOICES)
        gap = rng.choice(GAP_CHOICES)
    grids = [(r, c) for r, c in SLOT_GRIDS if r * c >= windows]
    rows, cols = rng.choice(grids)
    slots = list(range(rows * cols))
    rng.shuffle(slots)
    slots = sorted(slots[:windows])
    width = 2 * wall + cols * win + (cols - 1) * gap
    height = 2 * wall + rows * win + (rows - 1) * gap
    return {"wall": wall, "win": win, "gap": gap, "rows": rows, "cols": cols,
            "slots": slots, "w": width, "h": height}


def _place(rng: random.Random, cell_row: int, cell_col: int, w: int, h: int) -> tuple[int, int]:
    lo_x = cell_col * CELL + CELL_INSET
    hi_x = cell_col * CELL + CELL - CELL_INSET - w
    lo_y = cell_row * CELL + CELL_INSET
    hi_y = cell_row * CELL + CELL - CELL_INSET - h
    return rng.randint(lo_x, max(lo_x, hi_x)), rng.randint(lo_y, max(lo_y, hi_y))


def _sample_counts(rng: random.Random) -> tuple[int, int, int]:
    """Two paired window counts and one unpaired count, all distinct."""
    picked = rng.sample(COUNT_CHOICES, 3)
    rng.shuffle(picked)
    orphan = picked[0]
    pair_a, pair_b = sorted(picked[1:])
    return pair_a, pair_b, orphan


def _sample_decor(rng: random.Random, blocked: set[tuple[int, int]], scale: str) -> list[dict]:
    free = [(r, c) for r in range(GRID) for c in range(GRID) if (r, c) not in blocked]
    rng.shuffle(free)
    decor = []
    for row, col in free[: rng.randint(15, 21)]:
        windows = rng.choice((0, 1, 1, 2, 2, 3, 3, 4, 5))
        body = _plate_body(rng, max(windows, 1), scale if rng.random() < 0.3 else "loose")
        if windows == 0:
            body["slots"] = []
        x, y = _place(rng, row, col, body["w"], body["h"])
        item = dict(body)
        item.update({"x": x, "y": y, "ink": list(rng.choice(DECOR_INK))})
        decor.append(item)
    return decor


def sample_scene(seed: int) -> dict:
    """Deterministically sample one scene from an integer seed."""
    rng = random.Random(0x9F17 * 11 + seed * 7)
    orphan_color = COLOR_NAMES[seed % len(COLOR_NAMES)]
    scale = "tight" if seed % 7 == 3 else ("snug" if seed % 5 == 2 else "loose")

    pair_a, pair_b, orphan_count = _sample_counts(rng)
    others = [name for name in COLOR_NAMES if name != orphan_color]
    rng.shuffle(others)
    assignment = {orphan_color: orphan_count,
                  others[0]: pair_a, others[1]: pair_a,
                  others[2]: pair_b, others[3]: pair_b}

    cells = [(r, c) for r in ANCHOR_ROWS for c in ANCHOR_COLS]
    rng.shuffle(cells)
    chosen = cells[:5]
    order = list(COLOR_NAMES)
    rng.shuffle(order)

    plates = []
    for name, (cell_row, cell_col) in zip(order, chosen):
        body = _plate_body(rng, assignment[name], scale)
        x, y = _place(rng, cell_row, cell_col, body["w"], body["h"])
        plate = dict(body)
        plate.update({"color_name": name, "ink": list(PLATE_INK[name]), "x": x, "y": y})
        plates.append(plate)

    blocked = {cell for cell in chosen}
    decor = _sample_decor(rng, blocked, scale)
    return {"canvas": CANVAS, "plates": plates, "decor": decor}


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def _draw_plate(draw: ImageDraw.ImageDraw, item: dict) -> None:
    x, y, w, h = item["x"], item["y"], item["w"], item["h"]
    draw.rectangle([x, y, x + w - 1, y + h - 1], fill=tuple(item["ink"]))
    wall, win, gap, cols = item["wall"], item["win"], item["gap"], item["cols"]
    for slot in sorted(item["slots"]):
        row, col = divmod(int(slot), cols)
        wx = x + wall + col * (win + gap)
        wy = y + wall + row * (win + gap)
        draw.rectangle([wx, wy, wx + win - 1, wy + win - 1], fill=WHITE)


def render(scene: dict) -> Image.Image:
    """Rasterise the scene. Axis-aligned, unantialiased, byte-deterministic."""
    size = int(scene["canvas"])
    image = Image.new("RGB", (size, size), WHITE)
    draw = ImageDraw.Draw(image)
    for item in sorted(scene["decor"], key=lambda d: (d["y"], d["x"])):
        _draw_plate(draw, item)
    for item in sorted(scene["plates"], key=lambda d: (d["y"], d["x"])):
        _draw_plate(draw, item)
    return image


# --------------------------------------------------------------------------
# analytic consequences of the scene
# --------------------------------------------------------------------------
def _decide(counts: dict) -> str:
    """Decision from a colour -> window-count table."""
    if len(counts) != 5:
        return "abstain"
    tally: dict[int, int] = {}
    for value in counts.values():
        tally[value] = tally.get(value, 0) + 1
    singles = [value for value, hits in tally.items() if hits == 1]
    pairs = [value for value, hits in tally.items() if hits == 2]
    if len(singles) != 1 or len(pairs) != 2:
        return "abstain"
    orphan = singles[0]
    for name, value in counts.items():
        if value == orphan:
            return f"{name}_{orphan}"
    return "abstain"


def _window_counts(plates: list) -> dict:
    return {plate["color_name"]: len(plate["slots"]) for plate in plates}


def analytic_gold(scene: dict) -> str:
    return _decide(_window_counts(scene["plates"]))


def runner_up(scene: dict) -> str:
    """Nearest competing decision: one window added to or removed from one plate."""
    counts = _window_counts(scene["plates"])
    truth = _decide(counts)
    alternatives = set()
    for name in sorted(counts):
        for delta in (-1, 1):
            perturbed = dict(counts)
            perturbed[name] = counts[name] + delta
            if perturbed[name] < 0:
                continue
            other = _decide(perturbed)
            if other != truth:
                alternatives.add(other)
    return sorted(alternatives)[0] if alternatives else "abstain"


def margin(scene: dict) -> float:
    """Weakest decisive width in pixels over the five coloured plates."""
    widths = []
    for plate in scene["plates"]:
        widths.append(plate["wall"])
        widths.append(plate["win"])
        if plate["rows"] * plate["cols"] > 1:
            widths.append(plate["gap"])
    return float(min(widths))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list:
    """No latent aliases are declared: every latent field is drawn and gold is
    a function of the drawn window counts alone."""
    return []
