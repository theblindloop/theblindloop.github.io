"""Scene-free pixel inverse arm for mirror_axis_replay.

Receives one rendered RGB image and nothing else.  It re-finds the dial hub and
the arrow, re-finds the cards and orders them by drawn horizontal position,
fits each card's bar orientation from image moments, and replays the reflections
in that drawn order.  It imports no renderer, prompt, generator, verifier or
analytic-gold code and never sees a latent scene.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "mirror_axis_replay-oracle-0.1.0"

_HUB = (64, 64, 64)
_ARROW = (198, 32, 62)
_CARD = (246, 238, 213)
_BAR = (22, 72, 200)
_NAMES = {0: "right", 90: "up", 180: "left", 270: "down"}


def _mask(arr: np.ndarray, colour: tuple[int, int, int], tol: int = 24) -> np.ndarray:
    diff = np.abs(arr.astype(np.int16) - np.array(colour, dtype=np.int16))
    return (diff.max(axis=2) <= tol)


def _centroid(mask: np.ndarray) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if xs.size == 0:
        raise ValueError("empty mask")
    return float(xs.mean()), float(ys.mean())


def _runs(cols: np.ndarray) -> list[tuple[int, int]]:
    """Contiguous runs of True columns, left to right."""
    out: list[tuple[int, int]] = []
    start = None
    for x, flag in enumerate(cols):
        if flag and start is None:
            start = x
        elif not flag and start is not None:
            out.append((start, x - 1))
            start = None
    if start is not None:
        out.append((start, len(cols) - 1))
    return [run for run in out if run[1] - run[0] >= 20]


def _bar_angle(mask: np.ndarray) -> float:
    """Orientation of an elongated blob, in degrees, 0 = right, y flipped."""
    ys, xs = np.nonzero(mask)
    if xs.size < 20:
        raise ValueError("bar not found")
    x = xs - xs.mean()
    y = -(ys - ys.mean())
    sxx = float((x * x).mean())
    syy = float((y * y).mean())
    sxy = float((x * y).mean())
    return math.degrees(0.5 * math.atan2(2.0 * sxy, sxx - syy)) % 180.0


def _snap(value: float, step: int, modulus: int) -> int:
    return int(round(value / step) % (modulus // step)) * step


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))

    hub_x, hub_y = _centroid(_mask(arr, _HUB))
    arrow_x, arrow_y = _centroid(_mask(arr, _ARROW))
    heading = math.degrees(math.atan2(-(arrow_y - hub_y), arrow_x - hub_x)) % 360.0
    phi = _snap(heading, 90, 360)

    card_mask = _mask(arr, _CARD)
    columns = card_mask.sum(axis=0) > 8
    bar_mask = _mask(arr, _BAR)
    for x0, x1 in _runs(columns):
        window = np.zeros_like(bar_mask)
        window[:, x0 : x1 + 1] = bar_mask[:, x0 : x1 + 1]
        theta = _snap(_bar_angle(window), 45, 180)
        phi = (2 * theta - phi) % 360

    return _NAMES[phi % 360]
