"""Renderer and analytic gold for mirror_axis_replay.

A dial on the left carries an arrow drawn close to one of the four cardinal
directions.  To its right stands an ordered row of cream cards, each carrying a
single blue bar drawn close to one of four orientations (horizontal, rising
diagonal, vertical, falling diagonal).  Each card reflects the arrow's heading
across the line of its bar.  The cards act one at a time, left to right; no
intermediate heading and no final heading is ever drawn.  The answer is the
compass direction the arrow points after the last card.

All angles below are mathematical angles in degrees: 0 = right, 90 = up.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "mirror_axis_replay-renderer-0.1.0"

CANVAS = (900, 300)
BG = (255, 255, 255)
DIAL_FACE = (206, 206, 206)
HUB = (64, 64, 64)
ARROW = (198, 32, 62)
CARD_FILL = (246, 238, 213)
CARD_EDGE = (150, 150, 150)
AXIS = (22, 72, 200)
ORDER_ARROW = (120, 120, 120)

DIAL_CENTRE = (112, 150)
DIAL_RADIUS = 74
ARROW_LEN = 60
ARROW_WIDTH = 9
HUB_RADIUS = 10
HEAD_LEN = 26
HEAD_HALF = 15

CARD_X0 = 244
CARD_PITCH = 126
CARD_SIZE = 104
CARD_CY = 150
BAR_LEN = 78
BAR_WIDTH = 7

HEADINGS = (0, 90, 180, 270)          # east, north, west, south
NAMES = {0: "right", 90: "up", 180: "left", 270: "down"}
AXES = (0, 45, 90, 135)

QUARANTINE_THRESHOLD = 8.0
MAX_ARROW_JITTER = 30.0               # boundary between headings is 45 degrees
MAX_BAR_JITTER = 15.0                 # boundary between bar orientations is 22.5


# --------------------------------------------------------------------------
# scene sampling
# --------------------------------------------------------------------------
def _replay(start: int, thetas: list[int]) -> int:
    """Reflect the heading across each bar line in the given order."""
    phi = start % 360
    for theta in thetas:
        phi = (2 * theta - phi) % 360
    return phi


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    target = HEADINGS[seed % 4]                   # balanced final heading
    tight = (seed % 7) == 3                       # explicit near-boundary mode

    n_cards = rng.choice((3, 4, 5))
    start = rng.choice(HEADINGS)                  # sampled independently of target
    while True:
        thetas = [rng.choice(AXES) for _ in range(n_cards)]
        if _replay(start, thetas) == target:
            break

    if tight:
        arrow_jitter = rng.choice((37.0, 39.5, 41.0, 43.0))
    else:
        arrow_jitter = rng.choice((0.0, 6.0, 11.5, 17.0, 23.0, 27.5, MAX_ARROW_JITTER))
    arrow_jitter *= rng.choice((-1.0, 1.0))

    cards = []
    tight_card = rng.randrange(n_cards) if tight else -1
    for slot in range(n_cards):
        if slot == tight_card:
            jitter = rng.choice((18.5, 20.0, 21.5))
        else:
            jitter = rng.choice((0.0, 3.0, 6.5, 9.0, 12.0, MAX_BAR_JITTER))
        cards.append(
            {
                "slot": slot,
                "theta": thetas[slot],
                "jitter": jitter * rng.choice((-1.0, 1.0)),
            }
        )
    rng.shuffle(cards)                            # cards are an unordered bag

    return {
        "canvas": list(CANVAS),
        "n_cards": n_cards,
        "arrow": {"heading": start, "jitter": arrow_jitter},
        "cards": cards,
    }


# --------------------------------------------------------------------------
# analytic gold
# --------------------------------------------------------------------------
def _ordered(scene: dict) -> list[dict]:
    return sorted(scene["cards"], key=lambda card: card["slot"])


def analytic_gold(scene: dict) -> str:
    thetas = [card["theta"] for card in _ordered(scene)]
    return NAMES[_replay(scene["arrow"]["heading"], thetas)]


def margin(scene: dict) -> float:
    """Least angular clearance, in degrees, to a drawn reading boundary.

    The arrow's heading is read by nearest cardinal direction, so its boundary
    sits 45 degrees away; a bar's orientation is read by nearest multiple of 45,
    so its boundary sits 22.5 degrees away.  Bar clearances are doubled to put
    both readings on the same scale.
    """
    clearances = [45.0 - abs(scene["arrow"]["jitter"])]
    for card in scene["cards"]:
        clearances.append(2.0 * (22.5 - abs(card["jitter"])))
    return round(min(clearances), 3)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_THRESHOLD


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Re-orderings of the card bag: pixel-identical and answer-preserving."""
    cards = list(scene["cards"])
    out: list[tuple[str, dict]] = []
    for name, permuted in (
        ("card_bag_reversed", list(reversed(cards))),
        ("card_bag_rotated", cards[1:] + cards[:1]),
        ("card_bag_slot_sorted", sorted(cards, key=lambda c: c["slot"])),
    ):
        alt = {
            "canvas": list(scene["canvas"]),
            "n_cards": scene["n_cards"],
            "arrow": dict(scene["arrow"]),
            "cards": [dict(card) for card in permuted],
        }
        out.append((name, alt))
    return out


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def _unit(angle_deg: float) -> tuple[float, float]:
    rad = math.radians(angle_deg)
    return math.cos(rad), -math.sin(rad)          # screen y grows downward


def _thick_line(draw: ImageDraw.ImageDraw, cx, cy, dx, dy, half_len, width, colour):
    x0, y0 = cx - dx * half_len, cy - dy * half_len
    x1, y1 = cx + dx * half_len, cy + dy * half_len
    draw.line([(x0, y0), (x1, y1)], fill=colour, width=width)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", tuple(scene["canvas"]), BG)
    draw = ImageDraw.Draw(image)

    cx, cy = DIAL_CENTRE
    draw.ellipse(
        [cx - DIAL_RADIUS, cy - DIAL_RADIUS, cx + DIAL_RADIUS, cy + DIAL_RADIUS],
        outline=DIAL_FACE,
        width=3,
    )

    angle = scene["arrow"]["heading"] + scene["arrow"]["jitter"]
    dx, dy = _unit(angle)
    tip = (cx + dx * ARROW_LEN, cy + dy * ARROW_LEN)
    base = (cx + dx * (ARROW_LEN - HEAD_LEN), cy + dy * (ARROW_LEN - HEAD_LEN))
    draw.line([(cx, cy), base], fill=ARROW, width=ARROW_WIDTH)
    px, py = -dy, dx
    draw.polygon(
        [
            tip,
            (base[0] + px * HEAD_HALF, base[1] + py * HEAD_HALF),
            (base[0] - px * HEAD_HALF, base[1] - py * HEAD_HALF),
        ],
        fill=ARROW,
    )
    draw.ellipse(
        [cx - HUB_RADIUS, cy - HUB_RADIUS, cx + HUB_RADIUS, cy + HUB_RADIUS], fill=HUB
    )

    half = CARD_SIZE // 2
    for card in _ordered(scene):
        ccx = CARD_X0 + card["slot"] * CARD_PITCH + half
        draw.rectangle(
            [ccx - half, CARD_CY - half, ccx + half, CARD_CY + half],
            fill=CARD_FILL,
            outline=CARD_EDGE,
            width=3,
        )
        bx, by = _unit(card["theta"] + card["jitter"])
        _thick_line(draw, ccx, CARD_CY, bx, by, BAR_LEN / 2, BAR_WIDTH, AXIS)

    # drawn reading order for the card row
    y = CARD_CY + half + 26
    x_end = CARD_X0 + (scene["n_cards"] - 1) * CARD_PITCH + CARD_SIZE
    draw.line([(CARD_X0, y), (x_end, y)], fill=ORDER_ARROW, width=4)
    draw.polygon(
        [(x_end + 16, y), (x_end - 4, y - 11), (x_end - 4, y + 11)], fill=ORDER_ARROW
    )
    return image
