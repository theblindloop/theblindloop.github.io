"""Deterministic renderer and analytic arm for alternating_link_route."""

from __future__ import annotations

import itertools
import math
import random
from copy import deepcopy
from functools import lru_cache

from PIL import Image, ImageDraw

CANVAS = 448
NODE_RADIUS = 15
BASE_EDGES = tuple(
    [(row * 4 + col, row * 4 + col + 1) for row in range(3) for col in range(3)]
    + [(row * 4 + col, (row + 1) * 4 + col) for row in range(2) for col in range(4)]
)
BLUE_DEGREE_SIGNATURE = (0, 1, 1, 1, 1, 1, 1, 1, 2, 2, 2, 3)
PALETTES = (
    {
        "background": [249, 247, 240],
        "blue_link": [39, 107, 184],
        "orange_link": [235, 121, 35],
        "node": [84, 94, 107],
        "green": [43, 157, 96],
        "purple": [142, 76, 180],
    },
    {
        "background": [247, 246, 241],
        "blue_link": [45, 113, 190],
        "orange_link": [228, 113, 31],
        "node": [91, 100, 113],
        "green": [52, 165, 104],
        "purple": [149, 80, 185],
    },
    {
        "background": [250, 248, 242],
        "blue_link": [34, 101, 177],
        "orange_link": [239, 128, 39],
        "node": [77, 89, 103],
        "green": [38, 150, 90],
        "purple": [135, 71, 174],
    },
)


def _edge_key(edge: list | tuple) -> tuple[int, int]:
    first, second = int(edge[0]), int(edge[1])
    return (first, second) if first < second else (second, first)


def _alternating_exists(
    edges: list[list] | tuple[tuple[int, int, str], ...], start: int, target: int
) -> bool:
    adjacency: dict[int, list[tuple[int, str]]] = {}
    for edge in edges:
        first, second, color = int(edge[0]), int(edge[1]), str(edge[2])
        adjacency.setdefault(first, []).append((second, color))
        adjacency.setdefault(second, []).append((first, color))

    def visit(vertex: int, last_color: str | None, seen: frozenset[int]) -> bool:
        if vertex == target:
            return True
        for neighbor, color in adjacency.get(vertex, []):
            if neighbor in seen or color == last_color:
                continue
            if visit(neighbor, color, seen | {neighbor}):
                return True
        return False

    return visit(start, None, frozenset({start}))


def _blue_degree_signature(blue_indices: frozenset[int]) -> tuple[int, ...]:
    degrees = [0] * 12
    for index in blue_indices:
        first, second = BASE_EDGES[index]
        degrees[first] += 1
        degrees[second] += 1
    return tuple(sorted(degrees))


@lru_cache(maxsize=1)
def _coloring_pools() -> tuple[tuple[tuple[int, ...], ...], tuple[tuple[int, ...], ...]]:
    """Enumerate colorings with class-matched local and aggregate statistics."""
    pools: list[list[tuple[int, ...]]] = [[], []]
    terminal_edges = (
        tuple(index for index, edge in enumerate(BASE_EDGES) if 0 in edge),
        tuple(index for index, edge in enumerate(BASE_EDGES) if 11 in edge),
    )
    for chosen in itertools.combinations(range(len(BASE_EDGES)), 8):
        blue = frozenset(chosen)
        if sum(index in blue for index in range(9)) != 4:
            continue
        if any(sum(index in blue for index in incident) != 1 for incident in terminal_edges):
            continue
        if _blue_degree_signature(blue) != BLUE_DEGREE_SIGNATURE:
            continue
        labeled = tuple(
            (first, second, "blue" if index in blue else "orange")
            for index, (first, second) in enumerate(BASE_EDGES)
        )
        outcome = int(_alternating_exists(labeled, 0, 11))
        pools[outcome].append(chosen)
    if not pools[0] or not pools[1]:
        raise RuntimeError("counterbalanced coloring pools are incomplete")
    return tuple(pools[0]), tuple(pools[1])


def _transform_point(
    point: tuple[float, float], quarter_turns: int, mirror: bool
) -> list[int]:
    x, y = point[0] - 224.0, point[1] - 224.0
    if mirror:
        x = -x
    for _ in range(quarter_turns % 4):
        x, y = -y, x
    return [round(x + 224.0), round(y + 224.0)]


def sample_scene(seed: int) -> dict:
    """Sample one balanced labeled lattice; seed parity fixes only the answer class."""
    rng = random.Random(int(seed))
    expected = int(int(seed) % 2 == 0)
    pools = _coloring_pools()
    blue_indices = frozenset(rng.choice(pools[expected]))

    x_gap = rng.choice((86, 88, 90))
    y_gap = rng.choice((94, 97, 100))
    dx, dy = rng.randint(-8, 8), rng.randint(-8, 8)
    nodes = [
        [
            round(224 - 1.5 * x_gap + col * x_gap + dx),
            round(224 - y_gap + row * y_gap + dy),
        ]
        for row in range(3)
        for col in range(4)
    ]
    # Apply one shared dihedral transform after constructing the entire grid.
    quarter_turns = rng.randrange(4)
    mirror = bool(rng.randrange(2))
    nodes = [_transform_point(tuple(point), quarter_turns, mirror) for point in nodes]

    edges = [
        [first, second, "blue" if index in blue_indices else "orange"]
        for index, (first, second) in enumerate(BASE_EDGES)
    ]
    green_node, purple_node = 0, 11
    if rng.randrange(2):
        green_node, purple_node = purple_node, green_node

    # Relabel latent vertices and reorder links without changing visible geometry.
    order = list(range(12))
    rng.shuffle(order)
    inverse = {old: new for new, old in enumerate(order)}
    nodes = [nodes[old] for old in order]
    edges = [[inverse[first], inverse[second], color] for first, second, color in edges]
    green_node, purple_node = inverse[green_node], inverse[purple_node]
    rng.shuffle(edges)
    scene = {
        "nodes": nodes,
        "edges": edges,
        "green_node": green_node,
        "purple_node": purple_node,
        "palette": deepcopy(PALETTES[rng.randrange(len(PALETTES))]),
    }
    if analytic_gold(scene) != ("yes" if expected else "no"):
        raise RuntimeError("sampled coloring does not match requested class")
    return scene


def render(scene: dict) -> Image.Image:
    """Draw labeled links behind opaque, separated circular junction rings."""
    scale = 4
    nodes = scene["nodes"]
    edges = scene["edges"]
    green_node = int(scene["green_node"])
    purple_node = int(scene["purple_node"])
    palette = scene["palette"]
    image = Image.new(
        "RGB", (CANVAS * scale, CANVAS * scale), tuple(palette["background"])
    )
    draw = ImageDraw.Draw(image)
    for first, second, color in edges:
        points = [
            tuple(value * scale for value in nodes[int(first)]),
            tuple(value * scale for value in nodes[int(second)]),
        ]
        draw.line(points, fill=tuple(palette[f"{color}_link"]), width=9 * scale)

    radius = NODE_RADIUS * scale
    center_hole = 3 * scale
    for index, (x, y) in enumerate(nodes):
        if index == green_node:
            color = palette["green"]
        elif index == purple_node:
            color = palette["purple"]
        else:
            color = palette["node"]
        px, py = x * scale, y * scale
        draw.ellipse((px - radius, py - radius, px + radius, py + radius), fill=tuple(color))
        draw.ellipse(
            (px - center_hole, py - center_hole, px + center_hole, py + center_hole),
            fill=tuple(palette["background"]),
        )
    return image.resize((CANVAS, CANVAS), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return (
        "yes"
        if _alternating_exists(
            scene["edges"], int(scene["green_node"]), int(scene["purple_node"])
        )
        else "no"
    )


def margin(scene: dict) -> float:
    """Minimum number of whole-link recolorings required to flip the decision."""
    edges = scene["edges"]
    start, target = int(scene["green_node"]), int(scene["purple_node"])
    original = _alternating_exists(edges, start, target)
    for count in range(1, len(edges) + 1):
        for chosen in itertools.combinations(range(len(edges)), count):
            altered = [list(edge) for edge in edges]
            for index in chosen:
                altered[index][2] = "orange" if altered[index][2] == "blue" else "blue"
            if _alternating_exists(altered, start, target) != original:
                return float(count)
    return float(len(edges))


def _point_segment_distance(point: list[int], first: list[int], second: list[int]) -> float:
    px, py = point
    ax, ay = first
    bx, by = second
    vx, vy = bx - ax, by - ay
    denominator = vx * vx + vy * vy
    if denominator == 0:
        return math.inf
    fraction = max(0.0, min(1.0, ((px - ax) * vx + (py - ay) * vy) / denominator))
    return math.hypot(px - (ax + fraction * vx), py - (ay + fraction * vy))


def is_quarantined(scene: dict) -> bool:
    nodes = scene["nodes"]
    edges = scene["edges"]
    if len(nodes) != 12 or len(edges) != 17:
        return True
    if int(scene["green_node"]) == int(scene["purple_node"]):
        return True
    keys = [_edge_key(edge) for edge in edges]
    colors = [str(edge[2]) for edge in edges]
    if len(set(keys)) != 17 or set(colors) != {"blue", "orange"}:
        return True
    if colors.count("blue") != 8 or colors.count("orange") != 9:
        return True
    if min(
        math.dist(first, second)
        for index, first in enumerate(nodes)
        for second in nodes[index + 1 :]
    ) < 62:
        return True
    for edge in edges:
        first, second = int(edge[0]), int(edge[1])
        length = math.dist(nodes[first], nodes[second])
        if not 80 <= length <= 105:
            return True
        for index, point in enumerate(nodes):
            if index not in (first, second) and _point_segment_distance(
                point, nodes[first], nodes[second]
            ) < 35:
                return True
    return margin(scene) < 1


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["edges"] = list(reversed(scene["edges"]))
    return [("reverse_link_draw_list_order", twin)]
