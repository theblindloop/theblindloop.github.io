"""Deterministic scene sampler and Pillow renderer for Cross-Pair Ledger."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 800, 600
BACKGROUND = (250, 248, 244)
INK = (31, 36, 43)
FRAME = (166, 177, 189)
PAIR_FRAME = (194, 202, 211)
REFERENCE_FILL = (231, 234, 237)
LEDGER_FILL = (245, 243, 238)
TAGS = ("amber", "cyan", "violet")
TAG_RGB = {
    "amber": (239, 166, 35),
    "cyan": (26, 164, 190),
    "violet": (126, 87, 194),
}
GLYPHS = ("circle", "square", "triangle", "diamond", "plus", "x")


def _pair(a: str, b: str) -> tuple[str, str]:
    return tuple(sorted((a, b)))


def _target(ref_top: Iterable[str], ref_bottom: Iterable[str]) -> frozenset[tuple[str, str]]:
    return frozenset(_pair(a, b) for a, b in zip(ref_top, reversed(tuple(ref_bottom))))


def _matching_set(pairs: Iterable[Iterable[str]]) -> frozenset[tuple[str, str]]:
    return frozenset(_pair(str(pair[0]), str(pair[1])) for pair in pairs)


def _all_matchings(items: tuple[str, ...]) -> list[tuple[tuple[str, str], ...]]:
    if not items:
        return [tuple()]
    first = items[0]
    output: list[tuple[tuple[str, str], ...]] = []
    for index in range(1, len(items)):
        second = items[index]
        rest = items[1:index] + items[index + 1 :]
        for tail in _all_matchings(rest):
            output.append((_pair(first, second),) + tail)
    return output


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    identities = list(GLYPHS)
    rng.shuffle(identities)
    top = identities[:3]
    bottom = identities[3:]
    target = _target(top, bottom)

    alternatives = [
        matching
        for matching in _all_matchings(GLYPHS)
        if frozenset(matching) != target and len(frozenset(matching) & target) <= 1
    ]
    rng.shuffle(alternatives)
    ledgers: list[list[list[str]]] = [
        [list(pair) for pair in target],
        [list(pair) for pair in alternatives[0]],
        [list(pair) for pair in alternatives[1]],
    ]
    for ledger in ledgers:
        rng.shuffle(ledger)
        for pair in ledger:
            if rng.randrange(2):
                pair.reverse()

    correct_slot = (int(seed) // 3) % 3
    ordered_ledgers: list[list[list[str]] | None] = [None, None, None]
    ordered_ledgers[correct_slot] = ledgers[0]
    distractor_slots = [slot for slot in range(3) if slot != correct_slot]
    ordered_ledgers[distractor_slots[0]] = ledgers[1]
    ordered_ledgers[distractor_slots[1]] = ledgers[2]

    answer_tag = TAGS[int(seed) % 3]
    remaining_tags = [tag for tag in TAGS if tag != answer_tag]
    rng.shuffle(remaining_tags)
    tags: list[str | None] = [None, None, None]
    tags[correct_slot] = answer_tag
    for slot, tag in zip(distractor_slots, remaining_tags):
        tags[slot] = tag

    return {
        "ref_top": top,
        "ref_bottom": bottom,
        "candidate_pairs": ordered_ledgers,
        "candidate_tags": tags,
    }


def _draw_glyph(draw: ImageDraw.ImageDraw, name: str, cx: int, cy: int, r: int = 10) -> None:
    if name == "circle":
        draw.ellipse((cx - r, cy - r, cx + r, cy + r), fill=INK)
    elif name == "square":
        draw.rectangle((cx - r, cy - r, cx + r, cy + r), fill=INK)
    elif name == "triangle":
        draw.polygon(((cx, cy - r), (cx + r, cy + r), (cx - r, cy + r)), fill=INK)
    elif name == "diamond":
        draw.polygon(((cx, cy - r), (cx + r, cy), (cx, cy + r), (cx - r, cy)), fill=INK)
    elif name == "plus":
        arm = 4
        draw.rectangle((cx - arm, cy - r, cx + arm, cy + r), fill=INK)
        draw.rectangle((cx - r, cy - arm, cx + r, cy + arm), fill=INK)
    elif name == "x":
        draw.line((cx - r + 1, cy - r + 1, cx + r - 1, cy + r - 1), fill=INK, width=5)
        draw.line((cx - r + 1, cy + r - 1, cx + r - 1, cy - r + 1), fill=INK, width=5)
    else:
        raise ValueError(f"unknown glyph {name!r}")


def render(scene: dict) -> Image.Image:
    ref_top = scene["ref_top"]
    ref_bottom = scene["ref_bottom"]
    candidate_pairs = scene["candidate_pairs"]
    candidate_tags = scene["candidate_tags"]

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    draw.rounded_rectangle((145, 220, 295, 380), radius=16, fill=REFERENCE_FILL, outline=FRAME, width=4)
    draw.line((166, 300, 274, 300), fill=FRAME, width=2)
    for x, glyph in zip((180, 220, 260), ref_top):
        _draw_glyph(draw, str(glyph), x, 260)
    for x, glyph in zip((180, 220, 260), ref_bottom):
        _draw_glyph(draw, str(glyph), x, 340)

    draw.rounded_rectangle((475, 155, 700, 445), radius=18, fill=LEDGER_FILL, outline=FRAME, width=4)
    pair_centers = (550, 605, 660)
    row_centers = (210, 300, 390)
    for row_y, pairs, tag in zip(row_centers, candidate_pairs, candidate_tags):
        draw.rounded_rectangle((481, row_y - 15, 509, row_y + 15), radius=5, fill=TAG_RGB[str(tag)])
        for pair_x, pair_values in zip(pair_centers, pairs):
            draw.rounded_rectangle(
                (pair_x - 25, row_y - 25, pair_x + 25, row_y + 25),
                radius=8,
                outline=PAIR_FRAME,
                width=2,
            )
            _draw_glyph(draw, str(pair_values[0]), pair_x - 11, row_y)
            _draw_glyph(draw, str(pair_values[1]), pair_x + 11, row_y)
    return image


def analytic_gold(scene: dict) -> str:
    ref_top = scene["ref_top"]
    ref_bottom = scene["ref_bottom"]
    candidate_pairs = scene["candidate_pairs"]
    candidate_tags = scene["candidate_tags"]
    target = _target(ref_top, ref_bottom)
    matches = [index for index, pairs in enumerate(candidate_pairs) if _matching_set(pairs) == target]
    if len(matches) != 1:
        raise ValueError("scene has no unique exact ledger")
    return str(candidate_tags[matches[0]])


def margin(scene: dict) -> float:
    target = _target(scene["ref_top"], scene["ref_bottom"])
    scores = sorted(
        (len(_matching_set(pairs) & target) for pairs in scene["candidate_pairs"]), reverse=True
    )
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    try:
        target = _target(scene["ref_top"], scene["ref_bottom"])
        scores = [len(_matching_set(pairs) & target) for pairs in scene["candidate_pairs"]]
        return scores.count(max(scores)) != 1 or float(max(scores) - sorted(scores)[-2]) < 2.0
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    del scene
    return []
