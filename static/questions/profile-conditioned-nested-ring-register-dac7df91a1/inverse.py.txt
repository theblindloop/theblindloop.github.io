"""Independent pixel-only inverse for nested square-ring registration."""

from __future__ import annotations

from PIL import Image

GRID = 11


def _templates() -> dict[str, set[tuple[int, int]]]:
    arch = ({(r, c) for r in range(4, 10) for c in (3, 8)} | {(3, c) for c in range(4, 8)} | {(4, c) for c in (4, 7)} | {(10, c) for c in (2, 3, 4, 7, 8, 9)})
    cup = ({(r, c) for r in range(3, 9) for c in (3, 8)} | {(9, c) for c in range(4, 8)} | {(2, c) for c in (2, 3, 8, 9)} | {(10, c) for c in range(4, 8)})
    fork = ({(r, c) for r in range(5, 11) for c in (5, 6)} | {(r, c) for r in range(2, 5) for c in (2, 3, 8, 9)} | {(4, c) for c in (4, 7)})
    fork.remove((2, 3)); fork.remove((2, 8))
    bolt = ({(2, c) for c in range(4, 9)} | {(3, c) for c in range(3, 8)} | {(4, c) for c in range(4, 7)} | {(5, c) for c in range(3, 6)} | {(6, c) for c in range(2, 5)} | {(7, c) for c in range(3, 6)} | {(8, c) for c in (2, 3)})
    return {"arch": arch, "cup": cup, "fork": fork, "bolt": bolt}


TEMPLATES = _templates()


def _ring_path(layer: int) -> list[tuple[int, int]]:
    lo, hi = layer, GRID - 1 - layer
    path = [(lo, c) for c in range(lo, hi + 1)]
    path += [(r, hi) for r in range(lo + 1, hi + 1)]
    path += [(hi, c) for c in range(hi - 1, lo - 1, -1)]
    path += [(r, lo) for r in range(hi - 1, lo, -1)]
    return path


def _cyan(pixel: tuple[int, int, int]) -> bool:
    r, g, b = pixel
    return r < 70 and g > 115 and b > 150


def _magenta(pixel: tuple[int, int, int]) -> bool:
    r, g, b = pixel
    return r > 165 and g < 90 and b > 80 and r > b * 1.35


def _dark(pixel: tuple[int, int, int]) -> bool:
    r, g, b = pixel
    return max(r, g, b) < 90 and r < 65


def _bbox(image: Image.Image, predicate) -> tuple[int, int, int, int]:
    points = [(x, y) for y in range(image.height) for x in range(image.width) if predicate(image.getpixel((x, y)))]
    if len(points) < 100:
        raise ValueError("required fiducial missing")
    xs, ys = zip(*points)
    return min(xs), min(ys), max(xs), max(ys)


def decision_from_image(image: Image.Image) -> str:
    if not isinstance(image, Image.Image):
        raise TypeError("expected PIL image")
    raster = image.convert("RGB")
    if raster.size != (512, 512):
        raise ValueError("unexpected raster size")
    fx0, fy0, fx1, fy1 = _bbox(raster, _cyan)
    if abs((fx1 - fx0) - (fy1 - fy0)) > 2:
        raise ValueError("cyan frame is not square")
    left, top = fx0 + 4, fy0 + 4
    span = (fx1 - fx0) - 8
    cell = round(span / GRID)
    if cell not in (32, 34, 36) or abs(span - GRID * cell) > 1:
        raise ValueError("invalid grid geometry")
    occupied: set[tuple[int, int]] = set()
    markers: list[tuple[int, int]] = []
    for row in range(GRID):
        for col in range(GRID):
            x0, y0 = left + col * cell, top + row * cell
            probes = ((x0 + cell // 4, y0 + cell // 4), (x0 + 3 * cell // 4, y0 + cell // 4), (x0 + cell // 4, y0 + 3 * cell // 4), (x0 + 3 * cell // 4, y0 + 3 * cell // 4))
            if sum(_dark(raster.getpixel(point)) for point in probes) >= 3:
                occupied.add((row, col))
            cx, cy = x0 + cell // 2, y0 + cell // 2
            marker_votes = sum(_magenta(raster.getpixel((cx + dx, cy + dy))) for dx, dy in ((0, 0), (2, 0), (-2, 0), (0, 2), (0, -2)))
            if marker_votes >= 3:
                markers.append((row, col))
    if len(markers) != 5:
        raise ValueError("expected exactly one marker on each of five rings")
    decoded: set[tuple[int, int]] = set()
    for layer in range(5):
        path = _ring_path(layer)
        candidates = [cell_ for cell_ in markers if min(cell_[0], cell_[1], GRID - 1 - cell_[0], GRID - 1 - cell_[1]) == layer]
        if len(candidates) != 1:
            raise ValueError("ring marker missing or duplicated")
        anchor = path.index((layer, GRID // 2))
        shift = (path.index(candidates[0]) - anchor) % len(path)
        for index, canonical_cell in enumerate(path):
            if path[(index + shift) % len(path)] in occupied:
                decoded.add(canonical_cell)
    if (GRID // 2, GRID // 2) in occupied:
        decoded.add((GRID // 2, GRID // 2))
    scores = sorted((len(decoded.symmetric_difference(template)), label) for label, template in TEMPLATES.items())
    if scores[0][0] > 1 or scores[1][0] - scores[0][0] < 4:
        raise ValueError("ambiguous registered symbol")
    return scores[0][1]
