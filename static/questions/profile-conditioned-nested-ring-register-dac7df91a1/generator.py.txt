"""Deterministic sampler and renderer for nested square-ring registration."""

from __future__ import annotations

import copy
import random
from typing import Any

from PIL import Image, ImageDraw

GRID = 11
LABELS = ("arch", "cup", "fork", "bolt")


def _templates() -> dict[str, set[tuple[int, int]]]:
    arch = (
        {(r, c) for r in range(4, 10) for c in (3, 8)}
        | {(3, c) for c in range(4, 8)}
        | {(4, c) for c in (4, 7)}
        | {(10, c) for c in (2, 3, 4, 7, 8, 9)}
    )
    cup = (
        {(r, c) for r in range(3, 9) for c in (3, 8)}
        | {(9, c) for c in range(4, 8)}
        | {(2, c) for c in (2, 3, 8, 9)}
        | {(10, c) for c in range(4, 8)}
    )
    fork = (
        {(r, c) for r in range(5, 11) for c in (5, 6)}
        | {(r, c) for r in range(2, 5) for c in (2, 3, 8, 9)}
        | {(4, c) for c in (4, 7)}
    )
    fork.remove((2, 3))
    fork.remove((2, 8))
    bolt = (
        {(2, c) for c in range(4, 9)}
        | {(3, c) for c in range(3, 8)}
        | {(4, c) for c in range(4, 7)}
        | {(5, c) for c in range(3, 6)}
        | {(6, c) for c in range(2, 5)}
        | {(7, c) for c in range(3, 6)}
        | {(8, c) for c in (2, 3)}
    )
    result = {"arch": arch, "cup": cup, "fork": fork, "bolt": bolt}
    assert {len(value) for value in result.values()} == {24}
    return result


TEMPLATES = _templates()
PALETTES = (
    {"paper": [249, 248, 244], "empty": [237, 239, 236], "ink": [25, 29, 38], "cyan": [9, 158, 185], "magenta": [220, 35, 124], "grid": [191, 199, 201]},
    {"paper": [247, 250, 248], "empty": [238, 241, 243], "ink": [42, 31, 47], "cyan": [18, 145, 180], "magenta": [203, 30, 133], "grid": [198, 205, 208]},
    {"paper": [251, 247, 244], "empty": [241, 238, 234], "ink": [20, 39, 39], "cyan": [4, 166, 178], "magenta": [230, 43, 108], "grid": [201, 197, 193]},
)
DISTRACTOR_COLORS = ([231, 109, 55], [77, 151, 91], [231, 180, 35], [96, 104, 184])


def _ring_path(layer: int) -> list[tuple[int, int]]:
    lo, hi = layer, GRID - 1 - layer
    if lo == hi:
        return [(lo, lo)]
    path = [(lo, c) for c in range(lo, hi + 1)]
    path += [(r, hi) for r in range(lo + 1, hi + 1)]
    path += [(hi, c) for c in range(hi - 1, lo - 1, -1)]
    path += [(r, lo) for r in range(hi - 1, lo, -1)]
    return path


def _encode(target: set[tuple[int, int]], shifts: list[int]) -> tuple[set[tuple[int, int]], list[list[int]]]:
    encoded: set[tuple[int, int]] = set()
    markers: list[list[int]] = []
    for layer, shift in enumerate(shifts):
        path = _ring_path(layer)
        length = len(path)
        for index, cell in enumerate(path):
            if cell in target:
                encoded.add(path[(index + shift) % length])
        anchor_index = path.index((layer, GRID // 2))
        markers.append(list(path[(anchor_index + shift) % length]))
    if (GRID // 2, GRID // 2) in target:
        encoded.add((GRID // 2, GRID // 2))
    return encoded, markers


def _decode(encoded: set[tuple[int, int]], markers: list[list[int]]) -> set[tuple[int, int]]:
    result: set[tuple[int, int]] = set()
    for layer, marker in enumerate(markers):
        path = _ring_path(layer)
        anchor_index = path.index((layer, GRID // 2))
        shift = (path.index(tuple(marker)) - anchor_index) % len(path)
        for index, cell in enumerate(path):
            if path[(index + shift) % len(path)] in encoded:
                result.add(cell)
    if (GRID // 2, GRID // 2) in encoded:
        result.add((GRID // 2, GRID // 2))
    return result


def _scores(mask: set[tuple[int, int]]) -> list[tuple[int, str]]:
    return sorted((len(mask.symmetric_difference(template)), label) for label, template in TEMPLATES.items())


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    label = LABELS[int(seed) % len(LABELS)]
    shifts = [rng.randrange(1, len(_ring_path(layer))) for layer in range(5)]
    encoded, markers = _encode(TEMPLATES[label], shifts)
    cell = rng.choice((32, 34, 36))
    span = GRID * cell
    left = (512 - span) // 2 + rng.randint(-12, 12)
    top = (512 - span) // 2 + rng.randint(-12, 12)
    distractors = []
    for index in range(10):
        side = index % 2
        x = rng.randint(18, max(20, left - 18)) if side == 0 else rng.randint(left + span + 18, 494)
        distractors.append([x, rng.randint(24, 488), rng.randint(3, 7), index % 4])
    return {
        "target": label,
        "encoded_cells": [list(cell_) for cell_ in sorted(encoded)],
        "marker_cells": markers,
        "palette": rng.randrange(len(PALETTES)),
        "frame_x": left,
        "frame_y": top,
        "cell_px": cell,
        "distractors": distractors,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    palette = PALETTES[int(scene["palette"])]
    left, top, cell = int(scene["frame_x"]), int(scene["frame_y"]), int(scene["cell_px"])
    span = GRID * cell
    encoded = {tuple(item) for item in scene["encoded_cells"]}
    markers = [list(item) for item in scene["marker_cells"]]
    decoded = _decode(encoded, markers)
    if _scores(decoded)[0][1] != scene["target"] or _scores(decoded)[0][0] != 0:
        raise ValueError("encoded rings do not recover the declared target")
    image = Image.new("RGB", (512, 512), tuple(palette["paper"]))
    draw = ImageDraw.Draw(image)
    for x, y, radius, color_index in scene["distractors"]:
        color = tuple(DISTRACTOR_COLORS[int(color_index) % len(DISTRACTOR_COLORS)])
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)
    for row in range(GRID):
        for col in range(GRID):
            x0, y0 = left + col * cell, top + row * cell
            fill = palette["ink"] if (row, col) in encoded else palette["empty"]
            draw.rectangle((x0 + 2, y0 + 2, x0 + cell - 2, y0 + cell - 2), fill=tuple(fill))
    for index in range(GRID + 1):
        x, y = left + index * cell, top + index * cell
        draw.line((x, top, x, top + span), fill=tuple(palette["grid"]), width=1)
        draw.line((left, y, left + span, y), fill=tuple(palette["grid"]), width=1)
    draw.rectangle((left - 4, top - 4, left + span + 4, top + span + 4), outline=tuple(palette["cyan"]), width=4)
    radius = max(4, cell // 7)
    for row, col in markers:
        cx, cy = left + col * cell + cell // 2, top + row * cell + cell // 2
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=tuple(palette["magenta"]), outline=(255, 255, 255), width=1)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(scene["target"])


def margin(scene: dict[str, Any]) -> float:
    decoded = _decode({tuple(item) for item in scene["encoded_cells"]}, scene["marker_cells"])
    scores = _scores(decoded)
    return float(scores[1][0] - scores[0][0])


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 4.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
