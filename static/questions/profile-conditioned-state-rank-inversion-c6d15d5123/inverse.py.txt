"""Independent image-only parser for state-rank inversion panels."""

from __future__ import annotations

from collections import deque

from PIL import Image


PANEL_X0 = 18
PANEL_STEP = 216
KEY_X = 610
KEY_STEP = 108
KEY_Y = 42
CARD_Y0 = 176
CARD_STEP = 88
ARROW_NAMES = ("P1->P2", "P2->P3", "P3->P4", "P4->P5", "P5->P6")


def _nearest(color: tuple[int, int, int], palette: list[tuple[int, int, int]]) -> int:
    distances = [
        sum((int(a) - int(b)) ** 2 for a, b in zip(color, candidate))
        for candidate in palette
    ]
    best = min(range(len(distances)), key=distances.__getitem__)
    if distances[best] > 900:
        raise ValueError("state swatch is not close to the visible key")
    return best


def _dark_components(image: Image.Image, box: tuple[int, int, int, int]) -> int:
    pixels = image.load()
    left, top, right, bottom = box
    points: set[tuple[int, int]] = set()
    for y in range(top, bottom):
        for x in range(left, right):
            r, g, b = pixels[x, y]
            if r < 100 and g < 110 and b < 120:
                points.add((x, y))
    components = 0
    while points:
        components += 1
        start = points.pop()
        queue = deque([start])
        while queue:
            x, y = queue.popleft()
            for nx in (x - 1, x, x + 1):
                for ny in (y - 1, y, y + 1):
                    if (nx, ny) in points:
                        points.remove((nx, ny))
                        queue.append((nx, ny))
    return components


def _reversal_counts(rows: list[list[int]]) -> list[int]:
    counts = []
    for panel in range(5):
        total = 0
        before = rows[panel]
        after = rows[panel + 1]
        for first in range(4):
            for second in range(first + 1, 4):
                if (before[first] - before[second]) * (after[first] - after[second]) < 0:
                    total += 1
        counts.append(total)
    return counts


def decision_from_image(image: Image.Image) -> str:
    """Recover the winner using only rendered pixels and the public layout."""

    image = image.convert("RGB")
    pixels = image.load()
    palette = [
        tuple(pixels[KEY_X + state * KEY_STEP + 16, KEY_Y + 16])
        for state in range(4)
    ]
    if len(set(palette)) != 4:
        raise ValueError("visible state key is incomplete")

    rows: list[list[int]] = []
    for panel in range(6):
        px = PANEL_X0 + panel * PANEL_STEP
        row = [-1, -1, -1, -1]
        for slot in range(4):
            cy = CARD_Y0 + slot * CARD_STEP
            identity_count = _dark_components(image, (px + 25, cy + 14, px + 59, cy + 51))
            identity = identity_count - 1
            if identity not in range(4) or row[identity] != -1:
                raise ValueError("identity marker is missing, duplicated, or occluded")
            swatch = tuple(pixels[px + 166, cy + 32])
            row[identity] = _nearest(swatch, palette)
        if -1 in row or len(set(row)) != 4:
            raise ValueError("panel does not contain four distinct keyed states")
        rows.append(row)

    counts = _reversal_counts(rows)
    ordered = sorted(counts, reverse=True)
    if not counts or ordered[0] == ordered[1]:
        raise ValueError("reversal winner is tied")
    return ARROW_NAMES[max(range(5), key=lambda index: counts[index])]
