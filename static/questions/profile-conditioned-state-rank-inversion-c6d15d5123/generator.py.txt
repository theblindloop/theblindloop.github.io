"""Deterministic renderer and analytic world for state-rank inversion."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont


WIDTH = 1320
HEIGHT = 680
PANEL_X0 = 18
PANEL_STEP = 216
PANEL_WIDTH = 202
PANEL_TOP = 122
PANEL_HEIGHT = 492
CARD_X_OFFSET = 10
CARD_RIGHT_OFFSET = 192
CARD_Y0 = 176
CARD_STEP = 88
CARD_HEIGHT = 64
KEY_X = 610
KEY_STEP = 108
KEY_Y = 42
STATE_NAMES = ("LOW", "MID", "HIGH", "PEAK")
ARROW_NAMES = ("P1->P2", "P2->P3", "P3->P4", "P4->P5", "P5->P6")
BASE_PALETTE = (
    (220, 74, 79),
    (54, 139, 92),
    (67, 111, 193),
    (219, 145, 45),
)
BACKGROUND = (248, 249, 247)
PANEL_FILL = (255, 255, 255)
CARD_FILL = (235, 239, 242)
CARD_OUTLINE = (116, 128, 137)
DARK = (28, 43, 54)
MUTED = (83, 98, 108)


def _font(size: int) -> ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", size)
    except OSError:
        return ImageFont.load_default()


FONT_TITLE = _font(18)
FONT_LABEL = _font(13)
FONT_SMALL = _font(11)
FONT_TINY = _font(10)


def _pip_offsets(count: int) -> tuple[tuple[int, int], ...]:
    patterns = {
        1: ((0, 0),),
        2: ((-8, -8), (8, 8)),
        3: ((-8, -8), (0, 0), (8, 8)),
        4: ((-8, -8), (8, -8), (-8, 8), (8, 8)),
    }
    return patterns[count]


def _draw_identity(
    draw: ImageDraw.ImageDraw, cx: int, cy: int, identity: int, *, radius: int = 4
) -> None:
    for ox, oy in _pip_offsets(identity + 1):
        draw.ellipse(
            (cx + ox - radius, cy + oy - radius, cx + ox + radius, cy + oy + radius),
            fill=DARK,
        )


def _reversal_counts(state_rows: list[list[int]]) -> list[int]:
    counts: list[int] = []
    for panel in range(len(state_rows) - 1):
        total = 0
        left = state_rows[panel]
        right = state_rows[panel + 1]
        for first in range(4):
            for second in range(first + 1, 4):
                before = left[first] - left[second]
                after = right[first] - right[second]
                if before * after < 0:
                    total += 1
        counts.append(total)
    return counts


def _row_from_rank_order(rank_order: list[int]) -> list[int]:
    row = [0, 0, 0, 0]
    for rank, identity in enumerate(rank_order):
        row[identity] = rank
    return row


def sample_scene(seed: int) -> dict:
    """Sample a balanced, uniquely winning six-panel scene."""

    rng = random.Random(int(seed))
    target = int(seed) % len(ARROW_NAMES)
    state_rows: list[list[int]] = []
    counts: list[int] = []
    for _attempt in range(100000):
        candidate_rows: list[list[int]] = []
        for _panel in range(6):
            rank_order = [0, 1, 2, 3]
            rng.shuffle(rank_order)
            candidate_rows.append(_row_from_rank_order(rank_order))
        candidate_counts = _reversal_counts(candidate_rows)
        ordered = sorted(candidate_counts, reverse=True)
        if candidate_counts[target] == ordered[0] and ordered[0] - ordered[1] >= 2:
            state_rows = candidate_rows
            counts = candidate_counts
            break
    else:
        raise RuntimeError("could not sample a uniquely winning reversal arrow")

    slot_orders: list[list[int]] = []
    for _panel in range(6):
        order = [0, 1, 2, 3]
        rng.shuffle(order)
        slot_orders.append(order)

    palette = [list(color) for color in BASE_PALETTE]
    rng.shuffle(palette)
    if counts[target] != max(counts) or max(counts) - sorted(counts)[-2] < 2:
        raise AssertionError("sampled scene violates the unique-margin policy")
    return {
        "state_rows": state_rows,
        "slot_orders": slot_orders,
        "palette": palette,
    }


def render(scene: dict) -> Image.Image:
    """Render only the public scene fields as a single RGB page."""

    state_rows = scene["state_rows"]
    slot_orders = scene["slot_orders"]
    palette = scene["palette"]
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    draw.text((20, 13), "STATE-RANK REVERSALS", fill=DARK, font=FONT_TITLE)
    draw.text((20, 37), "ID KEY", fill=MUTED, font=FONT_SMALL)
    for identity, label in enumerate(("A", "B", "C", "D")):
        cx = 62 + identity * 62
        _draw_identity(draw, cx, 61, identity, radius=3)
        draw.text((cx + 16, 53), label, fill=DARK, font=FONT_LABEL)
    draw.text((20, 95), "Same identities; rows may move between panels", fill=MUTED, font=FONT_TINY)

    draw.text((KEY_X, 13), "STATE KEY (ordered)", fill=MUTED, font=FONT_SMALL)
    for state, name in enumerate(STATE_NAMES):
        x = KEY_X + state * KEY_STEP
        color = tuple(palette[state])
        draw.rectangle((x, KEY_Y, x + 32, KEY_Y + 32), fill=color, outline=DARK, width=1)
        draw.text((x - 1, KEY_Y + 38), name, fill=DARK, font=FONT_TINY)
    draw.text((KEY_X, 96), "LOW < MID < HIGH < PEAK", fill=DARK, font=FONT_SMALL)

    for panel in range(6):
        px = PANEL_X0 + panel * PANEL_STEP
        draw.rectangle(
            (px, PANEL_TOP, px + PANEL_WIDTH, PANEL_TOP + PANEL_HEIGHT),
            fill=PANEL_FILL,
            outline=(85, 104, 115),
            width=2,
        )
        draw.text((px + 10, PANEL_TOP + 10), f"P{panel + 1}", fill=DARK, font=FONT_LABEL)
        draw.line(
            (px + 10, PANEL_TOP + 39, px + PANEL_WIDTH - 10, PANEL_TOP + 39),
            fill=(204, 211, 214),
            width=1,
        )
        for slot, identity in enumerate(slot_orders[panel]):
            cy = CARD_Y0 + slot * CARD_STEP
            draw.rectangle(
                (px + CARD_X_OFFSET, cy, px + CARD_RIGHT_OFFSET, cy + CARD_HEIGHT),
                fill=CARD_FILL,
                outline=CARD_OUTLINE,
                width=1,
            )
            _draw_identity(draw, px + 42, cy + 32, identity)
            state = int(state_rows[panel][identity])
            draw.text((px + 70, cy + 24), STATE_NAMES[state], fill=DARK, font=FONT_SMALL)
            draw.rectangle(
                (px + 148, cy + 17, px + 184, cy + 47),
                fill=tuple(palette[state]),
                outline=DARK,
                width=1,
            )
        if panel < 5:
            ax = px + PANEL_WIDTH + 6
            ay = PANEL_TOP + PANEL_HEIGHT - 25
            draw.line((ax, ay, ax + 7, ay), fill=MUTED, width=2)
            draw.polygon(((ax + 7, ay - 4), (ax + 14, ay), (ax + 7, ay + 4)), fill=MUTED)
            draw.text((ax - 1, ay - 22), str(panel + 1), fill=MUTED, font=FONT_TINY)

    draw.text(
        (20, 635),
        "For each arrow, count pairs whose LOW-to-PEAK ordering reverses; choose the largest count.",
        fill=DARK,
        font=FONT_SMALL,
    )
    return image


def analytic_gold(scene: dict) -> str:
    state_rows = scene["state_rows"]
    counts = _reversal_counts(state_rows)
    return ARROW_NAMES[max(range(len(counts)), key=lambda index: counts[index])]


def margin(scene: dict) -> float:
    state_rows = scene["state_rows"]
    counts = _reversal_counts(state_rows)
    ordered = sorted(counts, reverse=True)
    return float(ordered[0] - ordered[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
