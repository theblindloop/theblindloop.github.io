"""Density-gradient world: reveal the global direction of a texture-energy field.

A monochrome field of equal-sized colored dots fills a white canvas. Across
the whole field the dot density increases monotonically along exactly one of
four principal directions (left, right, up, down). The decision is the global
Gestalt property recovered from the texture-energy field: the single direction
in which the field gets denser. The first spatial moment (the centroid of the
dot mass relative to the canvas centre) reveals that direction.

The renderer owns scene sampling + rasterization + the structural analytic
gold. The independent pixel-only inverse arm (``oracle.py``) recovers the same
direction from the final PNG alone.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (255, 255, 255)
# Saturated, mutually distinct ink colors, none of which is white.
PALETTE = [
    (215, 40, 40),
    (40, 110, 220),
    (40, 170, 80),
    (230, 150, 20),
    (150, 60, 180),
    (40, 190, 190),
]

NC = 14  # grid cells per axis
CELL = WIDTH / NC
P_MIN = 0.05
P_MAX = 0.95
DOT_R = 6
JITTER = 0.30  # fraction of a cell, applied to each dot centre
# Decision / quarantine threshold on the centroid-offset margin (px).
GAP_MIN = 28.0

_DIRECTIONS = ("left", "right", "up", "down")


def _color(cidx: int) -> tuple[int, int, int]:
    return PALETTE[cidx % len(PALETTE)]


def sample_scene(seed: int) -> dict:
    """Sample one latent scene deterministically from ``seed``."""
    rng = np.random.default_rng(seed)
    orient = _DIRECTIONS[int(rng.integers(0, len(_DIRECTIONS)))]
    cidx = int(rng.integers(0, len(PALETTE)))
    color = _color(cidx)

    dots: list[list[float]] = []
    for r in range(NC):
        for c in range(NC):
            if orient == "right":
                frac = c / (NC - 1)
            elif orient == "left":
                frac = (NC - 1 - c) / (NC - 1)
            elif orient == "down":
                frac = r / (NC - 1)
            else:  # up
                frac = (NC - 1 - r) / (NC - 1)
            p = P_MIN + (P_MAX - P_MIN) * frac
            if rng.random() < p:
                x = c * CELL + CELL / 2 + rng.uniform(-JITTER, JITTER) * CELL
                y = r * CELL + CELL / 2 + rng.uniform(-JITTER, JITTER) * CELL
                dots.append([float(round(x, 3)), float(round(y, 3))])

    return {"dots": dots, "color": color, "orient": orient}


def render(scene: dict) -> Image.Image:
    """Rasterize the latent scene (a deterministic PNG)."""
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    d = ImageDraw.Draw(img)
    color = tuple(scene["color"])
    for (x, y) in scene["dots"]:
        d.ellipse([x - DOT_R, y - DOT_R, x + DOT_R, y + DOT_R], fill=color)
    return img


def _dots_centroid(dots: list) -> tuple[float, float]:
    if not dots:
        return float(WIDTH) / 2.0, float(HEIGHT) / 2.0
    xs = [float(p[0]) for p in dots]
    ys = [float(p[1]) for p in dots]
    return float(np.mean(xs)), float(np.mean(ys))


def centroid_offset(scene: dict) -> tuple[float, float]:
    """Signed centroid offset from the canvas centre, in pixels."""
    cx, cy = _dots_centroid(scene["dots"])
    return cx - WIDTH / 2.0, cy - HEIGHT / 2.0


def analytic_gold(scene: dict) -> str:
    """Structural gold: the direction of the revealed density gradient."""
    ox, oy = centroid_offset(scene)
    if abs(ox) >= abs(oy):
        return "right" if ox > 0 else "left"
    return "down" if oy > 0 else "up"


def margin(scene: dict) -> float:
    """Commitment magnitude: gap between the best and second-best axis offsets."""
    ox, oy = centroid_offset(scene)
    best = max(abs(ox), abs(oy))
    worst = min(abs(ox), abs(oy))
    return float(best - worst)


def is_quarantined(scene: dict) -> bool:
    """Quarantine scenes whose winning axis offset is not clearly dominant."""
    return margin(scene) < GAP_MIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The scene is injectively rendered; no latent aliases are declared."""
    return []
