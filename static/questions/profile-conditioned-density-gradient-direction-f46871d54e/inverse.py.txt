"""Independent pixel-only inverse arm for the density-gradient world.

Recovers the decision (in which direction does the dot texture get denser?)
from the final rasterized PNG alone. It detects the ink pixels itself (any
sufficiently saturated, non-white pixel), measures their spatial centroid, and
reads the revealed gradient as the direction of the ink-mass offset from the
canvas centre. It shares no geometry, palette, sizes, or thresholds with
``renderer`` beyond the physical fact that the dots are coloured ink on a white
canvas.

This module imports no renderer / prompts / generate / verify / analytic-gold
code and never receives the latent scene.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

# Any pixel that is clearly not white (saturated or dark) counts as ink.
_BG_EDGE = 235
_MIN_INK = 12


def _ink_mask(arr: np.ndarray) -> np.ndarray:
    r = arr[..., 0].astype(int)
    g = arr[..., 1].astype(int)
    b = arr[..., 2].astype(int)
    not_white = (r < _BG_EDGE) & (g < _BG_EDGE) & (b < _BG_EDGE)
    strong = (np.abs(r - g) > 25) | (np.abs(g - b) > 25) | (np.abs(r - b) > 25)
    return not_white | strong


def decision_from_image(image: Image.Image) -> str:
    """Return the direction of increasing dot density among left/right/up/down."""
    arr = np.asarray(image.convert("RGB")).astype(int)
    mask = _ink_mask(arr)
    ys, xs = np.nonzero(mask)
    h, w = arr.shape[:2]
    if xs.size < _MIN_INK:
        return "no"
    cx = float(xs.mean()) - w / 2.0
    cy = float(ys.mean()) - h / 2.0
    if abs(cx) >= abs(cy):
        return "right" if cx > 0 else "left"
    return "down" if cy > 0 else "up"
