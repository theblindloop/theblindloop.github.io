"""Independent pixel-only inverse arm.

This module intentionally knows only the two public glyph colors and receives
one raster image. Missing or damaged evidence raises ValueError (abstention).
"""

from __future__ import annotations

import numpy as np
from PIL import Image

CORAL = np.array([222, 86, 78], dtype=np.int32)
BLUE = np.array([48, 119, 180], dtype=np.int32)


def _slope_sign(array: np.ndarray, target: np.ndarray) -> int:
    # int32 avoids overflow when squared RGB differences exceed 32767.
    rgb = array.astype(np.int32)
    distance2 = np.sum((rgb - target) ** 2, axis=2)
    points = np.argwhere(distance2 <= 24 * 24)
    if len(points) < 45:
        raise ValueError("required colored needle is missing")
    ys = points[:, 0].astype(np.float64)
    xs = points[:, 1].astype(np.float64)
    yspan = float(ys.max() - ys.min())
    xspan = float(xs.max() - xs.min())
    if yspan < 14.0 or xspan < 7.0:
        raise ValueError("needle evidence is too incomplete or too vertical")
    # cov(x,y) is negative in image coordinates when x moves right as the
    # needle rises (toward smaller y). Its sign alone is the measurement.
    covariance = float(np.sum((xs - xs.mean()) * (ys - ys.mean())))
    if abs(covariance) < 80.0:
        raise ValueError("needle slope sign is ambiguous")
    return 1 if covariance < 0.0 else -1


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    coral_sign = _slope_sign(array, CORAL)
    blue_sign = _slope_sign(array, BLUE)
    return "yes" if coral_sign == blue_sign else "no"
