"""Deterministic renderer and analytic arm for paired slope-sign agreement."""

from __future__ import annotations

import copy
import random
from PIL import Image, ImageDraw

WIDTH = 256
HEIGHT = 256
BACKGROUND = (248, 247, 243)
CORAL = (222, 86, 78)
BLUE = (48, 119, 180)
NEUTRAL = (198, 198, 194)
QUARANTINE_DX = 8.0


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    # The class is deterministically balanced for every consecutive seed range.
    agree = (int(seed) % 2 == 0)
    coral_sign = -1 if rng.randrange(2) == 0 else 1
    blue_sign = coral_sign if agree else -coral_sign
    # Cell-centered translations keep the two decisive components compact but
    # separated under the public 8x8 ablation grid.
    tx = rng.choice((0, 32))
    ty = rng.choice((-16, 16))
    coral_cx, blue_cx = 80 + tx, 144 + tx
    cy = 128 + ty

    def needle(cx: int, sign: int) -> list[list[int]]:
        dx = rng.choice((10, 12, 14))
        dy = rng.choice((20, 22, 24))
        # Stored in lower-to-upper order only for human-readable evidence; the
        # renderer treats the entries simply as endpoints.
        return [[cx - sign * dx // 2, cy + dy // 2], [cx + sign * (dx - dx // 2), cy - (dy - dy // 2)]]

    # Remote marks are intentionally nondiagnostic and never share target colors.
    distractors = [
        [[24, 30], [38, 30]],
        [[218, 220], [232, 220]],
        [[28, 222], [38, 216]],
    ]
    if rng.randrange(2):
        distractors.reverse()
    return {
        "canvas": [WIDTH, HEIGHT],
        "center": [112 + tx, cy],
        "coral": needle(coral_cx, coral_sign),
        "blue": needle(blue_cx, blue_sign),
        "needle_width": rng.choice((5, 6)),
        "distractors": distractors,
    }


def render(scene: dict) -> Image.Image:
    width, height = [int(v) for v in scene["canvas"]]
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for segment in scene["distractors"]:
        draw.line([tuple(segment[0]), tuple(segment[1])], fill=NEUTRAL, width=3)
    stroke = int(scene["needle_width"])
    draw.line([tuple(scene["coral"][0]), tuple(scene["coral"][1])], fill=CORAL, width=stroke)
    draw.line([tuple(scene["blue"][0]), tuple(scene["blue"][1])], fill=BLUE, width=stroke)
    # Round caps are visible endpoint cues and stay in the same color mask.
    radius = stroke // 2
    for key, color in (("coral", CORAL), ("blue", BLUE)):
        for x, y in scene[key]:
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)
    return image


def analytic_gold(scene: dict) -> str:
    coral = scene["coral"]
    blue = scene["blue"]
    coral_dx = int(coral[1][0]) - int(coral[0][0])
    blue_dx = int(blue[1][0]) - int(blue[0][0])
    return "yes" if (coral_dx > 0) == (blue_dx > 0) else "no"


def margin(scene: dict) -> float:
    coral = scene["coral"]
    blue = scene["blue"]
    coral_dx = abs(int(coral[1][0]) - int(coral[0][0]))
    blue_dx = abs(int(blue[1][0]) - int(blue[0][0]))
    # Distance from the closest constituent sign to its vertical boundary.
    return float(min(coral_dx, blue_dx))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_DX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["distractors"] = list(reversed(twin["distractors"]))
    return [("reverse_neutral_distractor_order", twin)]
