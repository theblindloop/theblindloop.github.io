"""Deterministic renderer and analytic arm for facing-apertures search."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

COLORS = {
    "red": (216, 55, 62),
    "blue": (45, 105, 210),
    "green": (18, 151, 92),
    "purple": (145, 63, 191),
}
DIRECTIONS = ("N", "E", "S", "W")
DIR_ANGLE = {"E": 0.0, "S": 90.0, "W": 180.0, "N": 270.0}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    color_names = list(COLORS)
    winner = color_names[int(seed) % len(color_names)]
    row_colors = color_names[:]
    rng.shuffle(row_colors)

    # Every distractor misses one or both of the two inward conditions. This
    # gives the unique winner a discrete runner-up margin of at least one.
    distractor_patterns = [
        ("E", "N"), ("E", "S"), ("N", "W"), ("S", "W"),
        ("N", "S"), ("S", "N"), ("W", "E"), ("W", "N"),
    ]
    rng.shuffle(distractor_patterns)
    pair_directions = {}
    for color in color_names:
        pair_directions[color] = ["E", "W"] if color == winner else list(distractor_patterns.pop())

    # Ring centers sit near 64-pixel grid-cell centers. The 128-pixel separation
    # keeps the two decisive components disconnected under the public 8x8
    # ablation probe while their joint normalized span stays focal.
    left_x = rng.choice((160, 224))
    base_y = rng.choice((166, 230))
    patch_origin = [left_x, base_y]
    decorations = []
    occupied = {(left_x, base_y + 26 * row) for row in range(4)}
    occupied |= {(left_x + 128, base_y + 26 * row) for row in range(4)}
    while len(decorations) < 12:
        x = rng.randrange(28, 485, 19)
        y = rng.randrange(28, 485, 19)
        if all((x - ox) ** 2 + (y - oy) ** 2 > 70 ** 2 for ox, oy in occupied):
            point = [x, y]
            if point not in decorations:
                decorations.append(point)
    return {
        "canvas_size": [512, 512],
        "patch_origin": patch_origin,
        "cell_size": [128, 26],
        "pair_directions": pair_directions,
        "row_colors": row_colors,
        "palette_index": int(seed) % 2,
        "decorations": decorations,
    }


def _draw_broken_ring(draw: ImageDraw.ImageDraw, cx: int, cy: int, color: tuple[int, int, int], gap: str, ss: int) -> None:
    radius = 11 * ss
    width = 5 * ss
    box = [cx * ss - radius, cy * ss - radius, cx * ss + radius, cy * ss + radius]
    center = DIR_ANGLE[gap]
    # Pillow angles increase clockwise in image coordinates. A 92-degree aperture
    # is broad enough to read but leaves a single connected 268-degree arc.
    draw.arc(box, start=center + 46.0, end=center + 314.0, fill=color, width=width)


def render(scene: dict) -> Image.Image:
    width, height = scene["canvas_size"]
    left_x, base_y = scene["patch_origin"]
    pair_dx, row_dy = scene["cell_size"]
    directions = scene["pair_directions"]
    rows = scene["row_colors"]
    palette_index = scene["palette_index"]
    decorations = scene["decorations"]
    background = (250, 249, 246) if palette_index == 0 else (247, 250, 249)
    ss = 4
    image = Image.new("RGB", (width * ss, height * ss), background)
    draw = ImageDraw.Draw(image)

    # Low-contrast remote marks are true nuisances and intentionally ignored by
    # both analytic and pixel decisions.
    decor_color = (218, 218, 214)
    for x, y in decorations:
        r = 2 * ss
        draw.ellipse([x * ss - r, y * ss - r, x * ss + r, y * ss + r], fill=decor_color)

    for row, color_name in enumerate(rows):
        y = base_y + row * row_dy
        left_gap, right_gap = directions[color_name]
        _draw_broken_ring(draw, left_x, y, COLORS[color_name], left_gap, ss)
        _draw_broken_ring(draw, left_x + pair_dx, y, COLORS[color_name], right_gap, ss)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def _scores(scene: dict) -> dict[str, int]:
    return {
        color: int(pair[0] == "E") + int(pair[1] == "W")
        for color, pair in scene["pair_directions"].items()
    }


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    winners = [color for color, value in scores.items() if value == max(scores.values())]
    if len(winners) != 1 or scores[winners[0]] != 2:
        raise ValueError("scene has no unique mutually facing pair")
    return winners[0]


def margin(scene: dict) -> float:
    values = sorted(_scores(scene).values(), reverse=True)
    return float(values[0] - values[1])


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 1.0 or analytic_gold(scene) not in COLORS
    except (ValueError, KeyError, TypeError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["decorations"] = list(reversed(twin["decorations"]))
    return [("reverse_remote_decoration_draw_order", twin)]
