"""Independent image-only inverse arm for the broken-ring search."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

PALETTE = {
    "red": (216, 55, 62),
    "blue": (45, 105, 210),
    "green": (18, 151, 92),
    "purple": (145, 63, 191),
}
CARDINALS = (("E", 1.0, 0.0), ("S", 0.0, 1.0), ("W", -1.0, 0.0), ("N", 0.0, -1.0))


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]:
            continue
        stack = [(y0, x0)]
        seen[y0, x0] = True
        points = []
        while stack:
            y, x = stack.pop()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= 45:
            output.append(np.asarray(points, dtype=np.float64))
    return output


def _fit_center(points_yx: np.ndarray) -> tuple[float, float]:
    ys = points_yx[:, 0]
    xs = points_yx[:, 1]
    # Fit the annulus center by minimizing radial spread. This uses only the
    # surviving colored pixels and makes no use of expected scene coordinates.
    guess_x = float((xs.min() + xs.max()) / 2.0)
    guess_y = float((ys.min() + ys.max()) / 2.0)
    best = (float("inf"), guess_x, guess_y)
    for cy in np.arange(guess_y - 7.0, guess_y + 7.01, 0.5):
        for cx in np.arange(guess_x - 7.0, guess_x + 7.01, 0.5):
            radii = np.hypot(xs - cx, ys - cy)
            score = float(np.median(np.abs(radii - np.median(radii))))
            candidate = (score, float(cx), float(cy))
            if candidate < best:
                best = candidate
    return best[1], best[2]


def _gap_direction(points_yx: np.ndarray) -> tuple[str, float]:
    cx, cy = _fit_center(points_yx)
    dx = points_yx[:, 1] - cx
    dy = points_yx[:, 0] - cy
    radii = np.hypot(dx, dy)
    valid = radii > 3.0
    ux = dx[valid] / radii[valid]
    uy = dy[valid] / radii[valid]
    # Ink's mean unit vector points away from the aperture, so negate it.
    gx = -float(ux.mean())
    gy = -float(uy.mean())
    strengths = sorted(
        ((gx * vx + gy * vy, name) for name, vx, vy in CARDINALS), reverse=True
    )
    confidence = strengths[0][0] - strengths[1][0]
    if confidence < 0.12:
        raise ValueError("ambiguous aperture direction")
    return strengths[0][1], float(confidence)


def _measure_pairs(image: Image.Image) -> dict[str, tuple[str, str, float]]:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    results = {}
    for color, rgb in PALETTE.items():
        distance = np.linalg.norm(array - np.asarray(rgb, dtype=np.float32), axis=2)
        comps = _components(distance < 72.0)
        valid = []
        for points in comps:
            ys, xs = points[:, 0], points[:, 1]
            if 75 <= len(points) <= 420 and xs.max() - xs.min() >= 10 and ys.max() - ys.min() >= 10:
                try:
                    direction, confidence = _gap_direction(points)
                except ValueError:
                    continue
                valid.append((float(xs.mean()), direction, confidence))
        if len(valid) == 2:
            valid.sort()
            results[color] = (valid[0][1], valid[1][1], min(valid[0][2], valid[1][2]))
    return results


def decision_from_image(image: Image.Image) -> str:
    pairs = _measure_pairs(image)
    scores = {
        color: int(left == "E") + int(right == "W")
        for color, (left, right, _confidence) in pairs.items()
    }
    winners = [color for color, score in scores.items() if score == 2]
    if len(winners) != 1:
        raise ValueError("image does not expose a unique mutually facing pair")
    winner = winners[0]
    runner_up = max((score for color, score in scores.items() if color != winner), default=0)
    if scores[winner] - runner_up < 1:
        raise ValueError("insufficient runner-up margin")
    return winner
