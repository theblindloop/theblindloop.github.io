"""Deterministic supersampled renderer and analytic gold for a wire-tracing task."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 384
HEIGHT = 384
SUPER = 4
BACKGROUND = (255, 255, 255)
MARKER_COLORS: dict[str, tuple[int, int, int]] = {
    "amber": (232, 154, 35),
    "teal": (26, 157, 151),
    "violet": (143, 77, 190),
}
WIRE_COLORS: tuple[tuple[int, int, int], ...] = (
    (76, 91, 111),
    (91, 82, 100),
    (72, 105, 92),
    (83, 87, 94),
)
ARROW_COLOR = (24, 29, 36)


def _template_points(template: int) -> list[tuple[float, float]]:
    templates = (
        (
            (0.04, 0.58),
            (0.16, 0.34),
            (0.29, 0.60),
            (0.42, 0.36),
            (0.56, 0.63),
            (0.70, 0.33),
            (0.84, 0.57),
            (0.96, 0.40),
        ),
        (
            (0.04, 0.60),
            (0.16, 0.38),
            (0.30, 0.64),
            (0.44, 0.31),
            (0.57, 0.56),
            (0.71, 0.39),
            (0.85, 0.66),
            (0.96, 0.35),
        ),
        (
            (0.04, 0.64),
            (0.17, 0.41),
            (0.31, 0.34),
            (0.44, 0.58),
            (0.58, 0.37),
            (0.72, 0.65),
            (0.85, 0.40),
            (0.96, 0.56),
        ),
        (
            (0.04, 0.57),
            (0.17, 0.33),
            (0.30, 0.61),
            (0.44, 0.39),
            (0.58, 0.58),
            (0.72, 0.34),
            (0.85, 0.63),
            (0.96, 0.42),
        ),
    )
    return [(float(x), float(y)) for x, y in templates[int(template) % len(templates)]]


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    marker_order = ["amber", "teal", "violet"]
    rng.shuffle(marker_order)
    t_values = sorted(
        [
            round(0.23 + rng.uniform(-0.018, 0.018), 5),
            round(0.50 + rng.uniform(-0.022, 0.022), 5),
            round(0.77 + rng.uniform(-0.018, 0.018), 5),
        ]
    )
    return {
        "marker_order": marker_order,
        "marker_t": t_values,
        "template": rng.randrange(4),
        "turn": rng.randrange(4),
        "mirror_x": bool(rng.randrange(2)),
        "mirror_y": bool(rng.randrange(2)),
        "scale": round(rng.uniform(0.90, 0.96), 5),
        "dx": round(rng.uniform(-7.0, 7.0), 4),
        "dy": round(rng.uniform(-7.0, 7.0), 4),
        "wire_rgb": list(rng.choice(WIRE_COLORS)),
        "wire_width": round(rng.uniform(6.2, 7.4), 4),
        "marker_radius": round(rng.uniform(13.5, 15.5), 4),
        "marker_width": round(rng.uniform(4.5, 5.8), 4),
    }


def _route_points(scene: dict[str, Any]) -> list[tuple[float, float]]:
    points: list[tuple[float, float]] = []
    turn = int(scene["turn"]) % 4
    mirror_x = bool(scene["mirror_x"])
    mirror_y = bool(scene["mirror_y"])
    scale = float(scene["scale"])
    dx = float(scene["dx"])
    dy = float(scene["dy"])
    for original_x, original_y in _template_points(int(scene["template"])):
        x, y = original_x, original_y
        if mirror_x:
            x = 1.0 - x
        if mirror_y:
            y = 1.0 - y
        if turn == 1:
            x, y = 1.0 - y, x
        elif turn == 2:
            x, y = 1.0 - x, 1.0 - y
        elif turn == 3:
            x, y = y, 1.0 - x
        points.append(
            (
                (x - 0.5) * WIDTH * scale + WIDTH / 2.0 + dx,
                (y - 0.5) * HEIGHT * scale + HEIGHT / 2.0 + dy,
            )
        )
    return points


def _polyline_lengths(points: list[tuple[float, float]]) -> tuple[list[float], float]:
    lengths = [0.0]
    total = 0.0
    for left, right in zip(points, points[1:]):
        total += math.hypot(right[0] - left[0], right[1] - left[1])
        lengths.append(total)
    return lengths, total


def _point_at_fraction(points: list[tuple[float, float]], fraction: float) -> tuple[float, float]:
    lengths, total = _polyline_lengths(points)
    target = max(0.0, min(1.0, float(fraction))) * total
    for index in range(len(points) - 1):
        left_length, right_length = lengths[index], lengths[index + 1]
        if target <= right_length or index == len(points) - 2:
            span = max(1e-9, right_length - left_length)
            weight = (target - left_length) / span
            left, right = points[index], points[index + 1]
            return (
                left[0] + (right[0] - left[0]) * weight,
                left[1] + (right[1] - left[1]) * weight,
            )
    return points[-1]


def _scaled_points(points: list[tuple[float, float]]) -> list[tuple[int, int]]:
    return [(round(x * SUPER), round(y * SUPER)) for x, y in points]


def _draw_smooth_line(
    draw: ImageDraw.ImageDraw,
    points: list[tuple[int, int]],
    color: tuple[int, int, int],
    width: int,
) -> None:
    draw.line(points, fill=color, width=width, joint="curve")
    radius = width / 2.0
    for x, y in points:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)


def _draw_arrow(
    draw: ImageDraw.ImageDraw,
    points: list[tuple[float, float]],
    width: float,
) -> None:
    start_x, start_y = points[0]
    next_x, next_y = points[1]
    vx, vy = next_x - start_x, next_y - start_y
    length = max(1e-6, math.hypot(vx, vy))
    ux, uy = vx / length, vy / length
    px, py = -uy, ux
    tip = (start_x + ux * 17.0, start_y + uy * 17.0)
    back = (start_x - ux * 7.0, start_y - uy * 7.0)
    half = max(8.0, width * 1.35)
    triangle = [
        (round(tip[0] * SUPER), round(tip[1] * SUPER)),
        (round((back[0] + px * half) * SUPER), round((back[1] + py * half) * SUPER)),
        (round((back[0] - px * half) * SUPER), round((back[1] - py * half) * SUPER)),
    ]
    draw.polygon(triangle, fill=ARROW_COLOR)


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (WIDTH * SUPER, HEIGHT * SUPER), BACKGROUND)
    draw = ImageDraw.Draw(image)
    points = _route_points(scene)
    wire_color = tuple(int(value) for value in scene["wire_rgb"])
    wire_width = max(3, round(float(scene["wire_width"]) * SUPER))
    _draw_smooth_line(draw, _scaled_points(points), wire_color, wire_width)
    _draw_arrow(draw, points, float(scene["wire_width"]))

    radius = float(scene["marker_radius"])
    ring_width = max(2, round(float(scene["marker_width"]) * SUPER))
    for fraction, name in zip(scene["marker_t"], scene["marker_order"]):
        center_x, center_y = _point_at_fraction(points, float(fraction))
        center = (center_x * SUPER, center_y * SUPER)
        bbox = (
            round(center[0] - radius * SUPER),
            round(center[1] - radius * SUPER),
            round(center[0] + radius * SUPER),
            round(center[1] + radius * SUPER),
        )
        draw.ellipse(
            bbox,
            outline=MARKER_COLORS[str(name)],
            width=ring_width,
        )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS).convert("RGB")


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(scene["marker_order"][1])


def margin(scene: dict[str, Any]) -> float:
    positions = [float(value) for value in scene["marker_t"]]
    return min(right - left for left, right in zip(positions, positions[1:])) - 0.13


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 0.035


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No distinct latent aliases are declared for this injective rendering."""

    return []
