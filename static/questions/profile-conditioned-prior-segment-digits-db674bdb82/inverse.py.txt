"""Independent pixel-only oracle for the seven-segment lit-count world.

Receives ONLY the final PNG and returns the number of SOLID amber segments.
It shares no geometry with renderer.render / analytic_gold. Its decision must
equal the analytic count for every generated scene.

Method: label every pixel by nearest palette colour {amber, ghost-gray, white}
(the palette is fixed and mutually well separated, so antialiased blend pixels
split cleanly at the ~50% point). Extract 4-connectivity connected components of
the amber mask; each lit segment is a separate solid bar, so the number of
disk-/bar-sized amber components is exactly the lit count. Sub-threshold
components are LANCZOS speckle and are discarded.

Import discipline: this module imports only PIL, numpy, ImageColor, math. It never
imports renderer, prompts, generate, verify, or any analytic-gold code.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "prior_segment_digits-oracle-0.1.0"

_AMBER = (255, 153, 0)
_GHOST = (200, 200, 200)
_WHITE = (255, 255, 255)
_PALETTE = [_AMBER, _GHOST, _WHITE]

# A real lit bar is >= ~18*150 = 1800 px; speckle from LANCZOS ringing is a few px.
MIN_SEGMENT_PX = 400
# Palette colours must be pairwise separable.
COLOR_MIN_DIST_PX = 60.0


def _label_pixels(arr: np.ndarray) -> np.ndarray:
    dists = np.stack(
        [np.linalg.norm(arr - np.array(c, dtype=np.float32), axis=-1) for c in _PALETTE],
        axis=-1,
    )
    return dists.argmin(axis=-1)


def _connected_components(mask: np.ndarray) -> list[np.ndarray]:
    ys, xs = np.nonzero(mask)
    if len(ys) == 0:
        return []
    y0, y1 = int(ys.min()), int(ys.max()) + 1
    x0, x1 = int(xs.min()), int(xs.max()) + 1
    sub = mask[y0:y1, x0:x1]
    h, w = sub.shape
    lin = np.arange(h * w).reshape(h, w)
    parent = list(range(h * w))

    def find(x: int) -> int:
        root = x
        while parent[root] != root:
            root = parent[root]
        while parent[x] != root:
            parent[x], x = root, parent[x]
        return root

    def union(a: int, b: int) -> None:
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    mr = sub[:, :-1] & sub[:, 1:]
    for a, b in zip(
        lin[:, :-1][mr].tolist(), lin[:, 1:][mr].tolist(), strict=True
    ):
        union(a, b)
    md = sub[:-1, :] & sub[1:, :]
    for a, b in zip(
        lin[:-1, :][md].tolist(), lin[1:, :][md].tolist(), strict=True
    ):
        union(a, b)

    groups: dict[int, list[int]] = {}
    for p in lin[sub].tolist():
        groups.setdefault(find(p), []).append(p)

    comps: list[np.ndarray] = []
    for pix in groups.values():
        arr = np.array(pix)
        yy = arr // w + y0
        xx = arr % w + x0
        full = np.zeros_like(mask)
        full[yy, xx] = True
        comps.append(full)
    return comps


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label_pixels(arr)
    amber = labels == 0
    count = 0
    for comp in _connected_components(amber):
        if int(comp.sum()) >= MIN_SEGMENT_PX:
            count += 1
    return str(count)


def measure(image: Image.Image) -> dict:
    """Diagnostic measurement (not used for the decision)."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label_pixels(arr)
    amber = labels == 0
    comps = _connected_components(amber)
    sizes = sorted((int(c.sum()) for c in comps), reverse=True)
    colored = int((labels == 0).sum())
    return {
        "amber_components": len(comps),
        "amber_component_sizes": sizes,
        "amber_px": colored,
        "size": image.size,
    }


def _color_dist(a: tuple[int, int, int], b: tuple[int, int, int]) -> float:
    return math.dist(a, b)


def assert_palette_separable() -> None:
    for i in range(len(_PALETTE)):
        for j in range(i + 1, len(_PALETTE)):
            if _color_dist(_PALETTE[i], _PALETTE[j]) < COLOR_MIN_DIST_PX:
                raise ValueError("palette colours are not separable")
