"""Seven-segment display lit-count world under a contradictory canonical prior.

Profile: paper-semantic@0.3.0:prior_conflict. Report visible truth (the number
of SOLIDLY LIT segments) when a familiar canonical prior ("a digital seven
segment display digit is made of seven segments") is contradicted by the actual
image, which lights only a strict subset of the seven segment slots.

Latent z = {lit_segments, width, height}. ``lit_segments`` is an ordered list of
segment indices (0..6) that are drawn SOLID in the amber palette; the remaining
segment slots are drawn only as faint gray ghost outlines so the image still reads
as a seven-segment display digit. The closed-form gold is

    count = len(lit_segments)

a scene parameter, not a pixel measurement.

Segments are laid out on a fixed seven-segment grid with clear separations, so
each lit segment is a disjoint solid bar. The continuous verifiability margin is
the minimum edge-to-edge clearance between two lit segments; when that is small
the lit bars approach each other and their component count becomes pixel-ambiguous
(fusion). The oracle counts amber connected components independently.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "prior_segment_digits-0.1.0"

AMBER = (255, 153, 0)
GHOST = (200, 200, 200)
BACKGROUND = (255, 255, 255)

CANVAS = (512, 512)

# Fixed seven-segment slot grid (x0, y0, x1, y1). Order: a,b,c,d,e,f,g.
SEGMENTS: tuple[tuple[int, int, int, int], ...] = (
    (180, 138, 332, 156),  # a top horizontal
    (314, 168, 332, 248),  # b top-right vertical
    (314, 284, 332, 364),  # c bottom-right vertical
    (180, 374, 332, 392),  # d bottom horizontal
    (180, 284, 198, 364),  # e bottom-left vertical
    (180, 168, 198, 248),  # f top-left vertical
    (180, 256, 332, 274),  # g middle horizontal
)

# Lit count range; the canonical prior contradictorily is that "all seven" are on.
LIT_COUNT_RANGE = (3, 7)
GHOST_OUTLINE_WIDTH = 2
QUARANTINE_CLEARANCE_PX = 6.0


def _clearance(a: tuple[int, int, int, int], b: tuple[int, int, int, int]) -> float:
    """Axis-aligned edge-to-edge separation between two segment rectangles."""
    return float(
        max(
            a[0] - b[2],
            b[0] - a[2],
            a[1] - b[3],
            b[1] - a[3],
        )
    )


def margin(scene: dict) -> float:
    """Min edge-to-edge clearance among the LIT segments (px)."""
    lit = scene["lit_segments"]
    segs = [SEGMENTS[i] for i in lit]
    if len(segs) < 2:
        return 400.0
    return min(
        _clearance(segs[i], segs[j])
        for i in range(len(segs))
        for j in range(i + 1, len(segs))
    )


def is_quarantined(scene: dict) -> bool:
    """A scene is quarantined only when two lit segments nearly touch (could fuse)."""
    return margin(scene) < QUARANTINE_CLEARANCE_PX


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    k = LIT_COUNT_RANGE[0] + (seed % (LIT_COUNT_RANGE[1] - LIT_COUNT_RANGE[0] + 1))
    lit = sorted(rng.sample(range(7), k))
    return {"lit_segments": lit, "width": CANVAS[0], "height": CANVAS[1]}


def render(scene: dict, supersample: int = 4) -> Image.Image:
    w, h = scene["width"], scene["height"]
    ss = supersample
    img = Image.new("RGB", (w * ss, h * ss), BACKGROUND)
    draw = ImageDraw.Draw(img)
    lit = set(scene["lit_segments"])
    for index, (x0, y0, x1, y1) in enumerate(SEGMENTS):
        box = (x0 * ss, y0 * ss, x1 * ss, y1 * ss)
        if index in lit:
            draw.rectangle(box, fill=AMBER)
        else:
            draw.rectangle(
                box,
                outline=GHOST,
                width=GHOST_OUTLINE_WIDTH * ss,
            )
    if ss > 1:
        img = img.resize((w, h), Image.LANCZOS)
    return img


def analytic_gold(scene: dict) -> str:
    return str(len(scene["lit_segments"]))


def latent_symmetries(scene: dict) -> list:
    """not_applicable: no render-fidelity-preserving nontrivial transform exists.

    Each lit segment is a distinct fixed-position bar; moving, recolouring, or
    reordering a lit segment changes the pixels. The image is injective in the
    lit set, so we declare no latent aliases and return an empty list.
    """
    return []
