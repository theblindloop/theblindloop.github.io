"""Independent pixel-only inverse arm for the room-neighbor atlas."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_WHITE = np.array((247, 245, 238), dtype=np.int16)
_ROOM = np.array(
    ((31, 174, 166), (244, 154, 55), (142, 92, 190), (231, 211, 93)),
    dtype=np.int16,
)
_TAGS = {
    "red": np.array((213, 61, 74), dtype=np.int16),
    "green": np.array((70, 170, 92), dtype=np.int16),
    "blue": np.array((52, 113, 199), dtype=np.int16),
    "gold": np.array((238, 187, 56), dtype=np.int16),
}


def _near(array: np.ndarray, color: np.ndarray, tolerance: float = 12.0) -> np.ndarray:
    return np.linalg.norm(array.astype(np.int16) - color, axis=2) <= tolerance


def _edges(grid: tuple[tuple[int, ...], ...]) -> frozenset[tuple[int, int]]:
    result: set[tuple[int, int]] = set()
    for row in range(4):
        for col in range(4):
            for rr, cc in ((row + 1, col), (row, col + 1)):
                if rr < 4 and cc < 4 and grid[row][col] != grid[rr][cc]:
                    result.add(tuple(sorted((grid[row][col], grid[rr][cc]))))
    return frozenset(result)


def _is_connected(grid: tuple[tuple[int, ...], ...], label: int) -> bool:
    cells = {(row, col) for row in range(4) for col in range(4) if grid[row][col] == label}
    if not cells:
        return False
    seen = {next(iter(cells))}
    queue = deque(seen)
    while queue:
        row, col = queue.popleft()
        for point in ((row - 1, col), (row + 1, col), (row, col - 1), (row, col + 1)):
            if point in cells and point not in seen:
                seen.add(point)
                queue.append(point)
    return seen == cells


def _room_grid(array: np.ndarray, quadrant: tuple[slice, slice]) -> tuple[tuple[int, ...], ...]:
    sub = array[quadrant]
    distances = np.stack([np.linalg.norm(sub.astype(np.int16) - color, axis=2) for color in _ROOM], axis=2)
    room_mask = distances.min(axis=2) <= 12.0
    ys, xs = np.nonzero(room_mask)
    if len(xs) < 10000:
        raise ValueError("mosaic room area is incomplete")
    left, right = int(xs.min()), int(xs.max())
    top, bottom = int(ys.min()), int(ys.max())
    if right - left < 120 or bottom - top < 120:
        raise ValueError("mosaic extent is too small")
    values: list[tuple[int, ...]] = []
    for row in range(4):
        line = []
        for col in range(4):
            x = round(left + (col + 0.5) * (right - left + 1) / 4.0)
            y = round(top + (row + 0.5) * (bottom - top + 1) / 4.0)
            pixel = sub[min(y, sub.shape[0] - 1), min(x, sub.shape[1] - 1)].astype(np.int16)
            nearest = int(np.linalg.norm(_ROOM - pixel, axis=1).argmin())
            if float(np.linalg.norm(_ROOM[nearest] - pixel)) > 12.0:
                raise ValueError("cell center has no recognized room color")
            line.append(nearest)
        values.append(tuple(line))
    grid = tuple(values)
    if {cell for line in grid for cell in line} != {0, 1, 2, 3}:
        raise ValueError("a mosaic does not expose all four room colors")
    if not all(_is_connected(grid, label) for label in range(4)):
        raise ValueError("a room color is not side-connected")
    return grid


def _tag_centers(array: np.ndarray, box: tuple[int, int, int, int]) -> dict[tuple[int, int], str]:
    left, top, right, bottom = box
    width, height = right - left + 1, bottom - top + 1
    found: dict[tuple[int, int], str] = {}
    for name, color in _TAGS.items():
        mask = _near(array, color)
        ys, xs = np.nonzero(mask)
        if len(xs) < 300:
            raise ValueError(f"missing {name} tag")
        cx, cy = float(xs.mean()), float(ys.mean())
        quadrant = (0 if cy < top + height / 2 else 1, 0 if cx < left + width / 2 else 1)
        if quadrant in found:
            raise ValueError("multiple tags map to one mosaic")
        found[quadrant] = name
    return found


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 128 or array.shape[1] < 128:
        raise ValueError("image is too small")
    foreground = ~_near(array, _WHITE)
    ys, xs = np.nonzero(foreground)
    if not len(xs):
        raise ValueError("atlas is absent")
    left, right = int(xs.min()), int(xs.max())
    top, bottom = int(ys.min()), int(ys.max())
    width, height = right - left + 1, bottom - top + 1
    if min(width, height) < 0.75 * min(array.shape[:2]) or abs(width - height) > 4:
        raise ValueError("connected atlas extent is invalid")
    # The public atlas is one solid, area-like field.  A white hole means decisive
    # visual evidence was erased, so the inverse arm abstains rather than guessing.
    interior_white = _near(array[top : bottom + 1, left : right + 1], _WHITE)
    if bool(interior_white.any()):
        raise ValueError("connected atlas has a missing area")

    mid_x, mid_y = left + width // 2, top + height // 2
    quadrants = (
        (slice(top, mid_y), slice(left, mid_x)),
        (slice(top, mid_y), slice(mid_x, right + 1)),
        (slice(mid_y, bottom + 1), slice(left, mid_x)),
        (slice(mid_y, bottom + 1), slice(mid_x, right + 1)),
    )
    grids = tuple(_room_grid(array, quadrant) for quadrant in quadrants)
    tags = _tag_centers(array, (left, top, right, bottom))
    if tags.get((0, 0)) != "gold" or set(tags.values()) != {"gold", "red", "green", "blue"}:
        raise ValueError("reference and candidate tag assignment is invalid")

    reference = _edges(grids[0])
    candidate_quadrants = ((0, 1), (1, 0), (1, 1))
    matches = [index for index in range(1, 4) if _edges(grids[index]) == reference]
    if len(matches) != 1:
        raise ValueError("pixel atlas has no unique adjacency-graph match")
    return tags[candidate_quadrants[matches[0] - 1]]
