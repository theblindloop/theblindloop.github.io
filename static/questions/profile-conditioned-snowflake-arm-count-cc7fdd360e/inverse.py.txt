"""Independent pixel-only inverse arm.

Receives the final PNG and nothing else: no latent scene, no renderer, no
prompts, no generator, no analytic gold.  It recovers the rotational symmetry
order of the painted snowflake by counting angular runs of chromatic ink on
several concentric measurement rings and abstaining unless the rings agree.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "snowflake_arm_count-oracle-0.1.0"

ABSTAIN = "uncertain"

SAT_MIN = 40  # chromatic-ink threshold; achromatic decoration is ignored
RING_FRACTIONS = (0.34, 0.40, 0.46, 0.52, 0.58)
N_SAMPLES = 2880
MIN_INK_PIXELS = 200
MIN_RUN_SAMPLES = 16
MIN_COUNT = 3
MAX_COUNT = 12


def _ink_mask(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    sat = arr.max(axis=-1) - arr.min(axis=-1)
    return sat >= SAT_MIN


def _despeckle(ring: np.ndarray, min_run: int) -> np.ndarray:
    """Fill circular runs shorter than `min_run`, gaps first then specks.

    Anti-aliased stroke edges can drop or add isolated samples along a ring;
    only runs far longer than that are treated as real arms and real wedges.
    """
    out = ring.copy()
    for value in (False, True):
        idx = 0
        n = out.size
        if out.all() or not out.any():
            break
        # rotate so that the ring starts at a run boundary
        start = int(np.argmax(out & ~np.roll(out, 1)))
        rolled = np.roll(out, -start)
        runs: list[tuple[int, int]] = []
        cur = rolled[0]
        begin = 0
        for i in range(1, n):
            if rolled[i] != cur:
                runs.append((begin, i))
                begin, cur = i, rolled[i]
        runs.append((begin, n))
        for a, b in runs:
            if rolled[a] == value and (b - a) < min_run:
                rolled[a:b] = not value
        out = np.roll(rolled, start)
        idx += 1
    return out


def _circular_runs(ring: np.ndarray) -> int:
    """Number of maximal circular runs of True in a boolean ring."""
    if not ring.any():
        return 0
    if ring.all():
        return 1
    prev = np.roll(ring, 1)
    return int(np.count_nonzero(ring & ~prev))


def decision_from_image(image: Image.Image) -> str:
    mask = _ink_mask(image)
    ys, xs = np.nonzero(mask)
    if xs.size < MIN_INK_PIXELS:
        return ABSTAIN

    cx = float(xs.mean())
    cy = float(ys.mean())
    radii = np.hypot(xs - cx, ys - cy)
    r_max = float(np.percentile(radii, 99.5))
    if r_max < 20.0:
        return ABSTAIN

    h, w = mask.shape
    angles = np.linspace(0.0, 2.0 * math.pi, N_SAMPLES, endpoint=False)
    cos_a, sin_a = np.cos(angles), np.sin(angles)

    counts: list[int] = []
    for frac in RING_FRACTIONS:
        r = frac * r_max
        if r < 8.0:
            return ABSTAIN
        ring = np.zeros(N_SAMPLES, dtype=bool)
        for dr in (-1.2, 0.0, 1.2):
            px = np.rint(cx + cos_a * (r + dr)).astype(int)
            py = np.rint(cy + sin_a * (r + dr)).astype(int)
            if px.min() < 0 or py.min() < 0 or px.max() >= w or py.max() >= h:
                return ABSTAIN
            ring |= mask[py, px]
        counts.append(_circular_runs(_despeckle(ring, MIN_RUN_SAMPLES)))

    if len(set(counts)) != 1:
        return ABSTAIN
    n = counts[0]
    if n < MIN_COUNT or n > MAX_COUNT:
        return ABSTAIN
    return str(n)
