"""Snowflake arm-count world.

A single connected snowflake is drawn: `n_arms` identical arms radiate from one
hub, evenly spaced by 360/n_arms degrees, each carrying two short side branches.
The analytic gold is the rotational symmetry order of the drawn shape, i.e. how
many arms are actually painted, against the canonical prior that a snowflake has
six arms.

Every geometric field (arm length, stroke half-width, rotation offset, hub
position, colour, decorative tick offsets) is sampled independently of the arm
count, and the arm length is solved so that the painted ink area tracks an
ink budget that is itself sampled independently of the arm count.  Nothing but
the number of painted arms distinguishes the classes.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "snowflake_arm_count-renderer-0.1.0"

WIDTH = 480
HEIGHT = 480
SS = 3  # supersampling factor

BG = (255, 255, 255)
DECOR = (176, 176, 176)

# Chromatic ink colours, sampled independently of the arm count.
COLORS: tuple[tuple[str, tuple[int, int, int]], ...] = (
    ("azure", (31, 90, 200)),
    ("teal", (0, 138, 138)),
    ("violet", (122, 58, 190)),
    ("crimson", (198, 40, 62)),
    ("amber", (206, 128, 12)),
    ("forest", (28, 130, 60)),
)

N_MIN = 4
N_MAX = 9
N_CLASSES = N_MAX - N_MIN + 1

R_IN_FRAC = 0.34  # inner measurement ring, as a fraction of the arm length
BRANCH_AT = 0.72  # branch attachment point along the arm
BRANCH_LEN = 0.22  # branch length, as a fraction of the arm length
BRANCH_DEG = 35.0  # branch opening angle from the arm axis

MIN_GAP_PX = 10.0  # sampler floor on the visible white wedge
QUARANTINE_GAP_PX = 16.0  # near-boundary flag


# ---------------------------------------------------------------- sampling


def _solve_geometry(n: int, gap_px: float, ink_target: float) -> tuple[int, int]:
    """Solve arm length R and stroke half-width hw.

    The white wedge between neighbouring arms at the inner ring is driven to
    `gap_px`, and R is driven so that the painted area tracks `ink_target`.
    Both targets are sampled independently of `n`.
    """
    r_len = 110.0
    hw = 6.0
    for _ in range(12):
        r_in = R_IN_FRAC * r_len
        gap_deg = math.degrees(gap_px / r_in)
        half_deg = (360.0 / n - gap_deg) / 2.0
        half_deg = max(1.2, min(half_deg, 40.0))
        s = math.sin(math.radians(half_deg))
        hw = r_in * s
        # ink ~ n arms * (2*hw) * R * (1 + 2*BRANCH_LEN)
        denom = 2.0 * n * R_IN_FRAC * s * (1.0 + 2.0 * BRANCH_LEN)
        r_len = math.sqrt(max(ink_target, 1.0) / max(denom, 1e-6))
        r_len = max(70.0, min(150.0, r_len))
    hw = max(4.0, min(hw, 0.22 * r_len))
    return int(round(r_len)), int(round(hw))


def sample_scene(seed: int) -> dict:
    """Deterministic latent scene.

    The answer class is `N_MIN + seed % N_CLASSES`, which is independent of
    every geometric field, so any contiguous block of scene seeds is balanced
    across all six classes.
    """
    rng = random.Random(seed * 7919 + 13)
    n = N_MIN + (seed % N_CLASSES)

    for _ in range(64):
        gap_px = rng.uniform(MIN_GAP_PX + 0.5, 34.0)
        ink_target = rng.uniform(9000.0, 15000.0)
        r_len, hw = _solve_geometry(n, gap_px, ink_target)
        r_in = R_IN_FRAC * r_len
        if hw >= r_in - 1.0:
            continue
        gap = _gap_px(n, r_len, hw)
        if gap >= MIN_GAP_PX and hw >= 4:
            break
    else:  # pragma: no cover - the search above always succeeds in practice
        r_len, hw = 120, 5
        gap = _gap_px(n, r_len, hw)

    color_name = COLORS[rng.randrange(len(COLORS))][0]
    scene = {
        "width": WIDTH,
        "height": HEIGHT,
        "cx": WIDTH // 2 + rng.randint(-20, 20),
        "cy": HEIGHT // 2 + rng.randint(-20, 20),
        "n_arms": n,
        "arm_len": r_len,
        "half_width": hw,
        "theta0": round(rng.uniform(0.0, 360.0), 3),
        "color": color_name,
        "decor": [rng.randint(0, 26) for _ in range(4)],
    }
    return scene


# ------------------------------------------------------------------ gold


def _gap_px(n: int, r_len: int, hw: int) -> float:
    """Visible white wedge between neighbouring arms at the inner ring, px."""
    r_in = R_IN_FRAC * float(r_len)
    ratio = min(1.0, float(hw) / r_in)
    half_deg = math.degrees(math.asin(ratio))
    gap_deg = 360.0 / n - 2.0 * half_deg
    return r_in * math.radians(gap_deg)


def analytic_gold(scene: dict) -> str:
    return str(int(scene["n_arms"]))


def margin(scene: dict) -> float:
    """Arc length in pixels of the smallest white wedge between two arms.

    This is exactly the evidence that separates `n` arms from the runner-up
    reading `n - 1`: when the wedge closes, two neighbouring arms merge into
    one painted lobe.
    """
    gap = _gap_px(int(scene["n_arms"]), int(scene["arm_len"]), int(scene["half_width"]))
    return round(float(gap), 4)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases are declared; see candidate.json latent_alias."""
    return []


# --------------------------------------------------- nuisance perturbations


def translate(scene: dict, dx: int, dy: int) -> dict:
    out = dict(scene)
    out["cx"] = int(scene["cx"]) + dx
    out["cy"] = int(scene["cy"]) + dy
    return out


def rotate_page(scene: dict, degrees: float) -> dict:
    out = dict(scene)
    out["theta0"] = round((float(scene["theta0"]) + degrees) % 360.0, 3)
    return out


def recolor(scene: dict) -> dict:
    out = dict(scene)
    names = [c[0] for c in COLORS]
    out["color"] = names[(names.index(scene["color"]) + 3) % len(names)]
    return out


def permute_decor(scene: dict, seed: int) -> dict:
    out = dict(scene)
    rng = random.Random(int(seed) + 5)
    out["decor"] = [rng.randint(0, 26) for _ in range(4)]
    return out


def rescale(scene: dict) -> dict:
    """Shrink the whole snowflake without touching the arm count."""
    out = dict(scene)
    r_len = max(70, int(round(int(scene["arm_len"]) * 0.85)))
    hw = max(4, int(round(int(scene["half_width"]) * 0.85)))
    out["arm_len"] = r_len
    out["half_width"] = hw
    return out


def set_arm_count(scene: dict, n: int) -> dict:
    """Label-transforming edit used only by tests: repaint with `n` arms."""
    out = dict(scene)
    out["n_arms"] = int(n)
    return out


# --------------------------------------------------------------- rendering


def _rect(draw: ImageDraw.ImageDraw, x0: float, y0: float, x1: float, y1: float,
          hw: float, fill: tuple[int, int, int]) -> None:
    """Stroke from (x0,y0) to (x1,y1) as a rotated rectangle of half-width hw."""
    dx, dy = x1 - x0, y1 - y0
    length = math.hypot(dx, dy)
    if length <= 0:
        return
    nx, ny = -dy / length * hw, dx / length * hw
    draw.polygon(
        [(x0 + nx, y0 + ny), (x1 + nx, y1 + ny), (x1 - nx, y1 - ny), (x0 - nx, y0 - ny)],
        fill=fill,
    )


def ink_pixels(image: Image.Image) -> int:
    """Diagnostic: painted chromatic area, reported so that reviewers can check
    that total ink does not predict the arm count."""
    import numpy as np

    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    return int(((arr.max(axis=-1) - arr.min(axis=-1)) >= 40).sum())


def render(scene: dict) -> Image.Image:
    w, h = int(scene["width"]), int(scene["height"])
    img = Image.new("RGB", (w * SS, h * SS), BG)
    d = ImageDraw.Draw(img)

    # Fixed achromatic decoration: page frame plus four corner ticks.
    d.rectangle([12 * SS, 12 * SS, w * SS - 12 * SS, h * SS - 12 * SS],
                outline=DECOR, width=2 * SS)
    for i, off in enumerate(scene["decor"]):
        t = 26 + int(off)
        if i == 0:
            d.line([(30 * SS, t * SS), (30 * SS + 18 * SS, t * SS)], fill=DECOR, width=2 * SS)
        elif i == 1:
            d.line([((w - 48) * SS, t * SS), ((w - 30) * SS, t * SS)], fill=DECOR, width=2 * SS)
        elif i == 2:
            d.line([(30 * SS, (h - t) * SS), (48 * SS, (h - t) * SS)], fill=DECOR, width=2 * SS)
        else:
            d.line([((w - 48) * SS, (h - t) * SS), ((w - 30) * SS, (h - t) * SS)],
                   fill=DECOR, width=2 * SS)

    color = dict(COLORS)[scene["color"]]
    cx, cy = float(scene["cx"]) * SS, float(scene["cy"]) * SS
    r_len = float(scene["arm_len"]) * SS
    hw = float(scene["half_width"]) * SS
    n = int(scene["n_arms"])
    theta0 = float(scene["theta0"])

    for k in range(n):
        a = math.radians(theta0 + 360.0 * k / n)
        ux, uy = math.cos(a), math.sin(a)
        tipx, tipy = cx + ux * r_len, cy + uy * r_len
        _rect(d, cx, cy, tipx, tipy, hw, color)
        bx, by = cx + ux * r_len * BRANCH_AT, cy + uy * r_len * BRANCH_AT
        for sign in (-1.0, 1.0):
            b = a + sign * math.radians(BRANCH_DEG)
            _rect(d, bx, by,
                  bx + math.cos(b) * r_len * BRANCH_LEN,
                  by + math.sin(b) * r_len * BRANCH_LEN,
                  hw * 0.8, color)
    d.ellipse([cx - hw, cy - hw, cx + hw, cy + hw], fill=color)

    return img.resize((w, h), Image.BOX)
