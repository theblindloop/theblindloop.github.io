"""Independent fixed-layout inverse arm for rescue-lock lattice rasters."""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH, HEIGHT = 960, 760
INK = np.array((35, 40, 47), dtype=np.float64)
OFF = np.array((255, 255, 255), dtype=np.float64)
RED = np.array((205, 54, 54), dtype=np.float64)
GATE_COLORS = {
    "and": np.array((48, 112, 190), dtype=np.float64),
    "or": np.array((224, 148, 39), dtype=np.float64),
    "xor": np.array((132, 76, 183), dtype=np.float64),
}
LOCK_COLORS = tuple(np.array(color, dtype=np.float64) for color in (
    (0, 143, 163), (43, 139, 72), (205, 91, 27), (179, 54, 132),
))
ROW_Y = (134, 229, 324, 419, 514)
TOP_X0 = 84
CELL_STEP = 60


class OracleAbstention(ValueError):
    pass


def _patch(arr: np.ndarray, x: int, y: int, radius: int = 4) -> np.ndarray:
    pixels = arr[y - radius:y + radius + 1, x - radius:x + radius + 1]
    return np.median(pixels.reshape(-1, 3), axis=0)


def _binary(arr: np.ndarray, x: int, y: int) -> int:
    rgb = _patch(arr, x, y)
    d_on = float(np.linalg.norm(rgb - INK))
    d_off = float(np.linalg.norm(rgb - OFF))
    if min(d_on, d_off) > 35 or abs(d_on - d_off) < 60:
        raise OracleAbstention(f"unclear binary cell at {(x, y)}: {rgb.tolist()}")
    return int(d_on < d_off)


def _gate(arr: np.ndarray, y: int) -> str:
    rgb = _patch(arr, 662, y, 5)
    distances = {kind: float(np.linalg.norm(rgb - color)) for kind, color in GATE_COLORS.items()}
    kind = min(distances, key=distances.get)  # type: ignore[arg-type]
    if distances[kind] > 45:
        raise OracleAbstention(f"unclear row rule at y={y}: {rgb.tolist()}")
    return kind


def _annulus(arr: np.ndarray, x: int, y: int) -> np.ndarray:
    crop = arr[y - 31:y + 32, x - 31:x + 32].astype(np.float64)
    yy, xx = np.ogrid[-31:32, -31:32]
    mask = ((xx * xx + yy * yy) >= 21 * 21) & ((xx * xx + yy * yy) <= 30 * 30)
    return crop[mask]


def _color_count(pixels: np.ndarray, color: np.ndarray, tolerance: float = 58.0) -> int:
    return int(np.count_nonzero(np.linalg.norm(pixels - color, axis=1) < tolerance))


def _has_source_ring(arr: np.ndarray, x: int, y: int) -> bool:
    return _color_count(_annulus(arr, x, y), RED) >= 100


def _lock_color_at(arr: np.ndarray, x: int, y: int) -> int | None:
    pixels = _annulus(arr, x, y)
    counts = [_color_count(pixels, color) for color in LOCK_COLORS]
    best = max(range(4), key=counts.__getitem__)
    ordered = sorted(counts, reverse=True)
    if counts[best] < 100:
        return None
    if ordered[0] - ordered[1] < 50:
        raise OracleAbstention(f"ambiguous lock ring at {(x, y)}: {counts}")
    return best


def _apply(kind: str, left: int, right: int) -> int:
    if kind == "and":
        return int(bool(left) and bool(right))
    if kind == "or":
        return int(bool(left) or bool(right))
    if kind == "xor":
        return int(bool(left) != bool(right))
    raise OracleAbstention("unknown gate")


def _reduce(row: list[int], kind: str) -> list[int]:
    return [_apply(kind, row[index], row[index + 1]) for index in range(len(row) - 1)]


def _run(
    top: list[int], gates: list[str], lock: tuple[int, int] | None,
    factual_rows: list[list[int]],
) -> list[list[int]]:
    rows = [list(top)]
    for row_index, kind in enumerate(gates, 1):
        row = _reduce(rows[-1], kind)
        if lock is not None and row_index == lock[0]:
            row[lock[1]] = factual_rows[row_index][lock[1]]
        rows.append(row)
    return rows


def decision_from_image(image: Image.Image) -> str:
    raster = image.convert("RGB")
    if raster.size != (WIDTH, HEIGHT):
        raise OracleAbstention(f"unexpected canvas {raster.size}")
    arr = np.asarray(raster)

    visible_rows: list[list[int]] = []
    source_marks: list[bool] = []
    recovered_locks: list[tuple[int, int] | None] = [None, None, None, None]
    for row_index, y in enumerate(ROW_Y):
        x0 = TOP_X0 + row_index * (CELL_STEP // 2)
        row: list[int] = []
        for cell_index in range(9 - row_index):
            x = x0 + cell_index * CELL_STEP
            row.append(_binary(arr, x, y))
            if row_index == 0:
                source_marks.append(_has_source_ring(arr, x, y))
            elif row_index <= 3:
                lock_index = _lock_color_at(arr, x, y)
                if lock_index is not None:
                    if recovered_locks[lock_index] is not None:
                        raise OracleAbstention(f"duplicate lock-{lock_index + 1} marker")
                    recovered_locks[lock_index] = (row_index, cell_index)
        visible_rows.append(row)

    if sum(source_marks) != 1:
        raise OracleAbstention(f"found {sum(source_marks)} source rings")
    marked = source_marks.index(True)
    if not 1 <= marked <= 7:
        raise OracleAbstention("source ring is on an unsupported edge cell")
    if any(lock is None for lock in recovered_locks):
        raise OracleAbstention(f"missing lock markers: {recovered_locks}")

    gates = [_gate(arr, ROW_Y[index]) for index in range(1, 5)]
    factual_rows = [list(visible_rows[0])]
    for row_index, kind in enumerate(gates, 1):
        factual_rows.append(_reduce(factual_rows[-1], kind))
        if factual_rows[row_index] != visible_rows[row_index]:
            raise OracleAbstention(f"visible factual row {row_index} contradicts its rule")

    counter_top = list(visible_rows[0])
    counter_top[marked] ^= 1
    outcomes = [
        _run(counter_top, gates, lock, factual_rows)[-1]
        for lock in recovered_locks
        if lock is not None
    ]
    matches = [index for index, outcome in enumerate(outcomes) if outcome == factual_rows[-1]]
    if len(matches) != 1:
        raise OracleAbstention(f"expected one restorative lock, recovered {matches}")
    return f"lock-{matches[0] + 1}"
