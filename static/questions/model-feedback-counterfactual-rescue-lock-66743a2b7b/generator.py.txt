"""Deterministic renderer for counterfactual rescue-lock interventions."""

from __future__ import annotations

import itertools
import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 760
SCALE = 2
BACKGROUND = (246, 245, 240)
PANEL_FILL = (255, 255, 255)
INK = (35, 40, 47)
OFF = (255, 255, 255)
RED = (205, 54, 54)
WIRE = (158, 164, 172)
GATE_COLORS = {
    "and": (48, 112, 190),
    "or": (224, 148, 39),
    "xor": (132, 76, 183),
}
GATE_LABELS = {"and": "ALL", "or": "ANY", "xor": "ONE"}
LOCK_COLORS = (
    (0, 143, 163),
    (43, 139, 72),
    (205, 91, 27),
    (179, 54, 132),
)
ROW_Y = (134, 229, 324, 419, 514)
TOP_X0 = 84
CELL_STEP = 60
CELL_RADIUS = 16
GATE_X0, GATE_X1 = 650, 758
LOCK_BOXES = (
    (38, 626, 238, 732),
    (266, 626, 466, 732),
    (494, 626, 694, 732),
    (722, 626, 922, 732),
)


def _apply(kind: str, left: int, right: int) -> int:
    if kind == "and":
        return int(bool(left) and bool(right))
    if kind == "or":
        return int(bool(left) or bool(right))
    if kind == "xor":
        return int(bool(left) != bool(right))
    raise ValueError(f"unknown gate {kind!r}")


def _next_row(row: list[int], kind: str) -> list[int]:
    return [_apply(kind, row[index], row[index + 1]) for index in range(len(row) - 1)]


def _run(
    top: list[int],
    gates: list[str],
    lock: tuple[int, int] | None = None,
    factual_rows: list[list[int]] | None = None,
) -> list[list[int]]:
    rows = [list(top)]
    for row_index, kind in enumerate(gates, 1):
        row = _next_row(rows[-1], kind)
        if lock is not None and row_index == lock[0]:
            if factual_rows is None:
                raise ValueError("a rescue lock needs factual rows")
            row[lock[1]] = int(factual_rows[row_index][lock[1]])
        rows.append(row)
    return rows


def _intervened_top(top: list[int], marked: int) -> list[int]:
    changed = list(top)
    changed[marked] = 1 - changed[marked]
    return changed


def _lock_bottom(
    top: list[int], gates: list[str], marked: int, lock: tuple[int, int]
) -> list[int]:
    factual_rows = _run(top, gates)
    return _run(_intervened_top(top, marked), gates, lock, factual_rows)[-1]


def _candidate_sets(
    top: list[int], gates: list[str], marked: int
) -> tuple[list[tuple[tuple[int, int], tuple[int, ...]]], list[tuple[tuple[int, int], tuple[int, ...]]]]:
    factual_rows = _run(top, gates)
    counter_rows = _run(_intervened_top(top, marked), gates)
    rescuers: list[tuple[tuple[int, int], tuple[int, ...]]] = []
    nonrescuers: list[tuple[tuple[int, int], tuple[int, ...]]] = []
    for row_index in range(1, 4):
        for cell_index in range(9 - row_index):
            # Every numbered lock must visibly matter in the un-clamped world.
            if counter_rows[row_index][cell_index] == factual_rows[row_index][cell_index]:
                continue
            coordinate = (row_index, cell_index)
            outcome = tuple(_run(
                _intervened_top(top, marked), gates, coordinate, factual_rows
            )[-1])
            item = (coordinate, outcome)
            if list(outcome) == factual_rows[-1]:
                rescuers.append(item)
            else:
                nonrescuers.append(item)
    return rescuers, nonrescuers


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target_slot = int(seed) % 4
    for _ in range(20000):
        top = [rng.randrange(2) for _ in range(9)]
        gates = [rng.choice(("and", "or", "xor")) for _ in range(4)]
        if len(set(gates)) < 3:
            continue
        marked = rng.randrange(1, 8)
        factual_rows = _run(top, gates)
        raw_counter = _run(_intervened_top(top, marked), gates)
        raw_distance = sum(
            left != right for left, right in zip(factual_rows[-1], raw_counter[-1], strict=True)
        )
        if not 2 <= raw_distance <= 4:
            continue

        rescuers, nonrescuers = _candidate_sets(top, gates, marked)
        if not rescuers:
            continue
        rng.shuffle(rescuers)
        rng.shuffle(nonrescuers)

        # The three wrong locks have distinct bottom consequences, and the four
        # numbered candidates collectively span all three internal depths.
        selected: tuple[
            tuple[tuple[int, int], tuple[int, ...]],
            tuple[tuple[tuple[int, int], tuple[int, ...]], ...],
        ] | None = None
        for rescuer in rescuers:
            for wrongs in itertools.combinations(nonrescuers, 3):
                if len({outcome for _coordinate, outcome in wrongs}) != 3:
                    continue
                rows_used = {rescuer[0][0], *(item[0][0] for item in wrongs)}
                if rows_used != {1, 2, 3}:
                    continue
                selected = (rescuer, wrongs)
                break
            if selected is not None:
                break
        if selected is None:
            continue

        rescuer, wrongs = selected
        wrong_list = list(wrongs)
        rng.shuffle(wrong_list)
        locks: list[list[int]] = []
        wrong_cursor = 0
        for slot in range(4):
            coordinate = rescuer[0] if slot == target_slot else wrong_list[wrong_cursor][0]
            if slot != target_slot:
                wrong_cursor += 1
            locks.append([coordinate[0], coordinate[1]])
        return {
            "top": top,
            "gates": gates,
            "marked": marked,
            "locks": locks,
            "lock_draw_order": [0, 1, 2, 3],
        }
    raise RuntimeError("could not sample a separated rescue-lock lattice")


def _lock_outcomes(scene: dict) -> list[list[int]]:
    top = [int(value) for value in scene["top"]]
    gates = [str(value) for value in scene["gates"]]
    marked = int(scene["marked"])
    return [
        _lock_bottom(top, gates, marked, (int(lock[0]), int(lock[1])))
        for lock in scene["locks"]
    ]


def analytic_gold(scene: dict) -> str:
    factual = _run(
        [int(value) for value in scene["top"]],
        [str(value) for value in scene["gates"]],
    )[-1]
    matches = [index for index, outcome in enumerate(_lock_outcomes(scene)) if outcome == factual]
    if len(matches) != 1:
        raise ValueError(f"expected one restorative lock, got {matches}")
    return f"lock-{matches[0] + 1}"


def margin(scene: dict) -> float:
    factual = _run(
        [int(value) for value in scene["top"]],
        [str(value) for value in scene["gates"]],
    )[-1]
    correct = int(analytic_gold(scene).split("-")[1]) - 1
    distances = [
        sum(left != right for left, right in zip(factual, outcome, strict=True))
        for index, outcome in enumerate(_lock_outcomes(scene))
        if index != correct
    ]
    return float(min(distances))


def is_quarantined(scene: dict) -> bool:
    try:
        top = scene["top"]
        gates = scene["gates"]
        locks = scene["locks"]
        if len(top) != 9 or any(value not in (0, 1) for value in top):
            return True
        if len(gates) != 4 or any(value not in GATE_COLORS for value in gates):
            return True
        if len(set(gates)) < 3 or not 1 <= int(scene["marked"]) <= 7:
            return True
        if len(locks) != 4 or any(len(lock) != 2 for lock in locks):
            return True
        coordinates = [(int(lock[0]), int(lock[1])) for lock in locks]
        if len(set(coordinates)) != 4:
            return True
        if any(not (1 <= row <= 3 and 0 <= cell < 9 - row) for row, cell in coordinates):
            return True
        if sorted(scene["lock_draw_order"]) != [0, 1, 2, 3]:
            return True
        factual_rows = _run(list(top), list(gates))
        raw_rows = _run(_intervened_top(list(top), int(scene["marked"])), list(gates))
        if {row for row, _cell in coordinates} != {1, 2, 3}:
            return True
        if any(raw_rows[row][cell] == factual_rows[row][cell] for row, cell in coordinates):
            return True
        outcomes = _lock_outcomes(scene)
        correct = int(analytic_gold(scene).split("-")[1]) - 1
        wrong_outcomes = [tuple(outcome) for index, outcome in enumerate(outcomes) if index != correct]
        if len(set(wrong_outcomes)) != 3 or margin(scene) < 1.0:
            return True
        return False
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["lock_draw_order"] = list(reversed(twin["lock_draw_order"]))
    return [("reverse_disjoint_lock_legend_draw_order", twin)]


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(name, size * SCALE)


def _pt(point: tuple[int, int]) -> tuple[int, int]:
    return point[0] * SCALE, point[1] * SCALE


def _box(box: tuple[int, int, int, int]) -> tuple[int, int, int, int]:
    return tuple(value * SCALE for value in box)  # type: ignore[return-value]


def _row_centers(row_index: int) -> list[tuple[int, int]]:
    length = 9 - row_index
    x0 = TOP_X0 + row_index * (CELL_STEP // 2)
    return [(x0 + index * CELL_STEP, ROW_Y[row_index]) for index in range(length)]


def _draw_cell(draw: ImageDraw.ImageDraw, center: tuple[int, int], value: int) -> None:
    x, y = center
    draw.ellipse(
        _box((x - CELL_RADIUS, y - CELL_RADIUS, x + CELL_RADIUS, y + CELL_RADIUS)),
        fill=INK if value else OFF,
        outline=INK,
        width=3 * SCALE,
    )


def _draw_lock_marker(
    draw: ImageDraw.ImageDraw, center: tuple[int, int], lock_index: int
) -> None:
    x, y = center
    color = LOCK_COLORS[lock_index]
    draw.ellipse(_box((x - 27, y - 27, x + 27, y + 27)), outline=color, width=6 * SCALE)
    draw.rounded_rectangle(
        _box((x - 13, y + 27, x + 13, y + 50)),
        radius=6 * SCALE,
        fill=color,
        outline=INK,
        width=2 * SCALE,
    )
    label = str(lock_index + 1)
    font = _font(15, True)
    bbox = draw.textbbox((0, 0), label, font=font)
    tw, th = bbox[2] - bbox[0], bbox[3] - bbox[1]
    draw.text((x * SCALE - tw // 2, (y + 37) * SCALE - th // 2 - SCALE), label,
              fill=(255, 255, 255), font=font)


def _draw_lock_box(draw: ImageDraw.ImageDraw, slot: int) -> None:
    x0, y0, x1, y1 = LOCK_BOXES[slot]
    color = LOCK_COLORS[slot]
    draw.rounded_rectangle(
        _box((x0, y0, x1, y1)), radius=13 * SCALE,
        fill=PANEL_FILL, outline=color, width=5 * SCALE,
    )
    label = f"LOCK {slot + 1}"
    font = _font(21, True)
    bbox = draw.textbbox((0, 0), label, font=font)
    tw, th = bbox[2] - bbox[0], bbox[3] - bbox[1]
    draw.text(((x0 + x1) * SCALE // 2 - tw // 2, (y0 + y1) * SCALE // 2 - th // 2 - SCALE),
              label, fill=INK, font=font)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text(_pt((28, 16)), "FLIP RED SOURCE  •  THEN TEST ONE LOCK", fill=INK, font=_font(26, True))
    draw.text(_pt((30, 53)), "A lock keeps its shown value; rebuild every row below it", fill=INK, font=_font(18))
    draw.text(_pt((750, 22)), "dark = ON", fill=INK, font=_font(17, True))

    top = [int(value) for value in scene["top"]]
    gates = [str(value) for value in scene["gates"]]
    rows = _run(top, gates)
    centers = [_row_centers(index) for index in range(5)]

    for row_index in range(1, 5):
        for index, child in enumerate(centers[row_index]):
            for parent in (centers[row_index - 1][index], centers[row_index - 1][index + 1]):
                draw.line([_pt(parent), _pt(child)], fill=WIRE, width=3 * SCALE)

    draw.text(_pt((17, ROW_Y[0] - 10)), "TOP", fill=INK, font=_font(14, True))
    for row_index, row in enumerate(rows):
        for index, value in enumerate(row):
            center = centers[row_index][index]
            if row_index == 0 and index == int(scene["marked"]):
                x, y = center
                draw.ellipse(_box((x - 27, y - 27, x + 27, y + 27)), outline=RED, width=6 * SCALE)
            _draw_cell(draw, center, int(value))
        if row_index:
            kind = gates[row_index - 1]
            y = ROW_Y[row_index]
            draw.rounded_rectangle(
                _box((GATE_X0, y - 23, GATE_X1, y + 23)), radius=9 * SCALE,
                fill=GATE_COLORS[kind], outline=INK, width=2 * SCALE,
            )
            label = GATE_LABELS[kind]
            font = _font(17, True)
            bbox = draw.textbbox((0, 0), label, font=font)
            tw, th = bbox[2] - bbox[0], bbox[3] - bbox[1]
            draw.text((704 * SCALE - tw // 2, y * SCALE - th // 2 - SCALE), label,
                      fill=(255, 255, 255), font=font)
            row_label = "SHOWN BOTTOM" if row_index == 4 else f"ROW {row_index}"
            draw.text(_pt((776, y - 10)), row_label, fill=INK, font=_font(14, True))

    for lock_index, lock in enumerate(scene["locks"]):
        row_index, cell_index = int(lock[0]), int(lock[1])
        _draw_lock_marker(draw, centers[row_index][cell_index], lock_index)

    draw.text(_pt((28, 583)), "WHICH LOCK RESTORES THE SHOWN BOTTOM?", fill=INK, font=_font(20, True))
    for slot in scene["lock_draw_order"]:
        _draw_lock_box(draw, int(slot))
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
