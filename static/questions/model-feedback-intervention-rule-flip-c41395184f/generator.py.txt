"""Counterfactual-intervention colour-rule world.

A single START disc holds one of four colours. Two legends are drawn side by
side. The ACTUAL RULE is a permutation over the four-colour alphabet: for each
source colour it shows the target colour it turns into. A NOW disc shows the
outcome the ACTUAL RULE would produce from the START colour. The IF RULE is an
intervention: a copy of the ACTUAL RULE in which exactly one entry (the row for
the START colour) is changed to point at a different target colour. The
decision is the counterfactual outcome: the colour that would have resulted if
the IF RULE (rather than the ACTUAL RULE) had produced the outcome from the
same START colour. It is not the ACTUAL RULE's outcome (the NOW disc) and not
any single legend target; it is the single changed entry applied to START.

The renderer owns scene sampling, rasterization and the structural analytic
gold (the counterfactual IF-rule outcome for the START colour). The independent
pixel-only inverse arm (``oracle.py``) recovers the same decision from the final
PNG alone by reading the IF-rule target squares, the START disc, and composing
one lookup.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw, ImageFont

WIDTH = 780
HEIGHT = 580
BACKGROUND = (255, 255, 255)
GREY = (85, 85, 85)
LABEL = (40, 40, 40)

COLORS = [
    (222, 55, 60),    # red
    (45, 95, 215),    # blue
    (50, 160, 72),    # green
    (242, 146, 30),   # orange
]
COLOR_NAMES = ["red", "blue", "green", "orange"]

# ACTUAL RULE legend geometry (four rows, source order = COLOR order).
KEYX_SRC = 90
KEYX_TGT = 270
KEYY0 = 150
ROW_STEP = 62
SQUARE = 34

# IF RULE legend geometry.
IFX_SRC = 480
IFX_TGT = 660

# START disc
START_CX = 150
START_CY = 500
START_R = 26

# NOW disc (outcome of the ACTUAL RULE) - a distractor, never the answer.
NOW_CX = 410
NOW_CY = 500
NOW_R = 26

_FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"


def sample_scene(seed: int) -> dict:
    """Sample one latent scene deterministically from ``seed``."""
    rng = np.random.default_rng(seed)
    perm = [int(value) for value in rng.permutation(4)]
    start = int(rng.integers(0, 4))
    actual_target = perm[start]
    options = [i for i in range(4) if i != actual_target]
    flip_dest = int(options[int(rng.integers(0, len(options)))])
    return {
        "perm": perm,
        "start": start,
        "flip_dest": flip_dest,
    }


def _if_rule(scene: dict) -> list[int]:
    rule = list(scene["perm"])
    rule[scene["start"]] = scene["flip_dest"]
    return rule


def analytic_gold(scene: dict) -> str:
    """Structural gold: the counterfactual IF-rule outcome for the START colour."""
    return COLOR_NAMES[scene["flip_dest"]]


def margin(scene: dict) -> float:
    """The counterfactual outcome is fully deterministic and unambiguous."""
    return 1.0


def is_quarantined(scene: dict) -> bool:
    """No illegibility: four pairwise-distinct fixed colours, large solid fills."""
    return False


def render(scene: dict) -> Image.Image:
    """Rasterize the latent scene (a deterministic PNG)."""
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(img)
    try:
        font_big = ImageFont.truetype(_FONT_PATH, 26)
        font_small = ImageFont.truetype(_FONT_PATH, 18)
    except OSError:
        font_big = ImageFont.load_default()
        font_small = ImageFont.load_default()

    draw.text((40, 50), "Colour rule picture", font=font_big, fill=LABEL)

    perm = scene["perm"]
    start = scene["start"]
    flip_dest = scene["flip_dest"]
    if_rule = list(perm)
    if_rule[start] = flip_dest

    # ACTUAL RULE legend.
    draw.text((60, 122), "ACTUAL RULE", font=font_small, fill=LABEL)
    for row in range(4):
        y = KEYY0 + row * ROW_STEP
        src_colour = COLORS[row]
        tgt_colour = COLORS[perm[row]]
        draw.rectangle([KEYX_SRC, y, KEYX_SRC + SQUARE, y + SQUARE], fill=src_colour)
        draw.rectangle([KEYX_TGT, y, KEYX_TGT + SQUARE, y + SQUARE], fill=tgt_colour)
        arrow_x0 = KEYX_SRC + SQUARE + 10
        arrow_x1 = KEYX_TGT - 10
        mid_y = y + SQUARE // 2
        draw.line([arrow_x0, mid_y, arrow_x1, mid_y], fill=GREY, width=4)
        draw.polygon(
            [(arrow_x1, mid_y), (arrow_x1 - 14, mid_y - 9), (arrow_x1 - 14, mid_y + 9)],
            fill=GREY,
        )

    # IF RULE legend (intervention).
    draw.text((450, 122), "IF RULE", font=font_small, fill=LABEL)
    for row in range(4):
        y = KEYY0 + row * ROW_STEP
        src_colour = COLORS[row]
        tgt_colour = COLORS[if_rule[row]]
        draw.rectangle([IFX_SRC, y, IFX_SRC + SQUARE, y + SQUARE], fill=src_colour)
        draw.rectangle([IFX_TGT, y, IFX_TGT + SQUARE, y + SQUARE], fill=tgt_colour)
        arrow_x0 = IFX_SRC + SQUARE + 10
        arrow_x1 = IFX_TGT - 10
        mid_y = y + SQUARE // 2
        draw.line([arrow_x0, mid_y, arrow_x1, mid_y], fill=GREY, width=4)
        draw.polygon(
            [(arrow_x1, mid_y), (arrow_x1 - 14, mid_y - 9), (arrow_x1 - 14, mid_y + 9)],
            fill=GREY,
        )

    # START disc.
    draw.text((60, 468), "START", font=font_big, fill=LABEL)
    draw.ellipse(
        [START_CX - START_R, START_CY - START_R, START_CX + START_R, START_CY + START_R],
        fill=COLORS[start],
    )

    # NOW disc: the ACTUAL RULE's outcome (distractor).
    draw.text((300, 468), "NOW", font=font_big, fill=LABEL)
    draw.ellipse(
        [NOW_CX - NOW_R, NOW_CY - NOW_R, NOW_CX + NOW_R, NOW_CY + NOW_R],
        fill=COLORS[perm[start]],
    )

    draw.text((560, 468), "Which colour would the IF RULE give?", font=font_small, fill=LABEL)

    return img


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The scene is injectively rendered; no latent aliases are declared."""
    return []
