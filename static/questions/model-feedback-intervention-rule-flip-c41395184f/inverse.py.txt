"""Independent pixel-only inverse arm for the counterfactual-intervention world.

Recovers the decision (the counterfactual IF-rule outcome for the START colour)
from the final rasterized PNG alone. It reads the four IF-rule rows by sampling
the fixed source/target squares, reads the START disc colour, looks up the IF
rule's target at the START row, and returns that colour name. It never reads
the ACTUAL-rule legend or the NOW disc, because the answer is the intervened
outcome, not the actual one. No renderer / prompts / generate / verify /
analytic-gold code is imported.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

COLORS = [
    (222, 55, 60),    # red
    (45, 95, 215),    # blue
    (50, 160, 72),    # green
    (242, 146, 30),   # orange
]
COLOR_NAMES = ["red", "blue", "green", "orange"]

IFX_TGT = 660
KEYY0 = 150
ROW_STEP = 62
SQUARE = 34
START_CX = 150
START_CY = 500


def _nearest_colour(rgb: tuple[int, int, int]) -> int:
    best = 0
    best_d = float("inf")
    for index, colour in enumerate(COLORS):
        d = sum((rgb[i] - colour[i]) ** 2 for i in range(3))
        if d < best_d:
            best_d = d
            best = index
    return best


def _pixel(arr: np.ndarray, x: int, y: int) -> tuple[int, int, int]:
    return (int(arr[y, x, 0]), int(arr[y, x, 1]), int(arr[y, x, 2]))


def decision_from_image(image: Image.Image) -> str:
    """Return the counterfactual IF-rule colour for the START colour."""
    arr = np.asarray(image.convert("RGB"))

    if_rule: list[int] = []
    for row in range(4):
        y = KEYY0 + row * ROW_STEP
        if_rule.append(_nearest_colour(_pixel(arr, IFX_TGT + SQUARE // 2, y + SQUARE // 2)))

    start_index = _nearest_colour(_pixel(arr, START_CX, START_CY))
    return COLOR_NAMES[if_rule[start_index]]
