"""Renderer and analytic arm for the zone-crossing cable world."""

from __future__ import annotations

from PIL import Image, ImageDraw

WIDTH, HEIGHT, SS = 560, 540, 3
BACKGROUND = "#f7f5ef"
DARK = "#263238"
CYAN = "#19b5c5"
GREEN = "#2ca25f"
COLORS = ("orange", "purple", "blue")
FILLS = {"orange": "#f28e2b", "purple": "#8e5ac8", "blue": "#377eb8"}


def _path_for(count: int, cy: int, zone: list[int], sign: int) -> list[list[int]]:
    x0, y0, x1, y1 = zone
    outside_y = cy + sign * 70
    if count == 0:
        return [[55, cy], [145, cy], [145, outside_y], [395, outside_y], [395, cy], [500, cy]]
    if count == 2:
        return [[55, cy], [115, cy], [115, outside_y], [155, outside_y], [155, cy], [500, cy]]
    # Enter at the left edge, exit through one horizontal edge, re-enter through
    # that edge, and finally exit at the right edge: four explicit transitions.
    loop_y = y1 + 20 if sign > 0 else y0 - 20
    return [[55, cy], [x0 + 40, cy], [x0 + 40, loop_y],
            [x1 - 40, loop_y], [x1 - 40, cy], [500, cy]]


def sample_scene(seed: int) -> dict:
    """Deterministically counterbalance target row, answer color, and detour side."""
    centers = [90, 270, 450]
    target_row = (seed // 3) % 3
    target_color = COLORS[seed % 3]
    other_counts = [0, 2]
    if (seed // 9) % 2:
        other_counts.reverse()
    counts = []
    for row in range(3):
        counts.append(4 if row == target_row else other_counts.pop(0))

    remaining_colors = [c for c in COLORS if c != target_color]
    if (seed // 27) % 2:
        remaining_colors.reverse()
    terminal_colors = []
    for row in range(3):
        terminal_colors.append(target_color if row == target_row else remaining_colors.pop(0))

    shift = (-10, 0, 10)[(seed // 81) % 3]
    zones = [[190 + shift, cy - 50, 350 + shift, cy + 50] for cy in centers]
    paths = []
    for row, (cy, count, zone) in enumerate(zip(centers, counts, zones)):
        sign = 1 if ((seed + row) % 2) else -1
        paths.append(_path_for(count, cy, zone, sign))
    return {"paths": paths, "zones": zones, "terminal_colors": terminal_colors}


def _segment_crossings(path: list[list[int]], zone: list[int]) -> int:
    x0, y0, x1, y1 = zone
    total = 0
    for (ax, ay), (bx, by) in zip(path, path[1:]):
        if ay == by:
            lo, hi = sorted((ax, bx))
            if y0 < ay < y1:
                total += int(lo < x0 < hi) + int(lo < x1 < hi)
        elif ax == bx:
            lo, hi = sorted((ay, by))
            if x0 < ax < x1:
                total += int(lo < y0 < hi) + int(lo < y1 < hi)
        else:
            raise ValueError("paths must be orthogonal")
    return total


def _crossing_counts(scene: dict) -> list[int]:
    paths = scene["paths"]
    zones = scene["zones"]
    # terminal_colors is deliberately accessed: it is the visible binding from
    # the selected cable row to the declared decision label.
    terminal_colors = scene["terminal_colors"]
    if not (len(paths) == len(zones) == len(terminal_colors) == 3):
        raise ValueError("scene must contain three paired rows")
    return [_segment_crossings(path, zone) for path, zone in zip(paths, zones)]


def analytic_gold(scene: dict) -> str:
    counts = _crossing_counts(scene)
    winners = [i for i, value in enumerate(counts) if value == 4]
    if len(winners) != 1:
        raise ValueError(f"expected one four-crossing cable, got {counts}")
    return scene["terminal_colors"][winners[0]]


def margin(scene: dict) -> float:
    counts = _crossing_counts(scene)
    ordered = sorted(counts, reverse=True)
    return float(ordered[0] - ordered[1])


def is_quarantined(scene: dict) -> bool:
    try:
        counts = _crossing_counts(scene)
    except (KeyError, TypeError, ValueError):
        return True
    if sorted(counts) != [0, 2, 4] or margin(scene) < 2:
        return True
    for path, zone in zip(scene["paths"], scene["zones"]):
        x0, y0, x1, y1 = zone
        for x, y in path:
            if min(abs(x - x0), abs(x - x1)) < 12 and min(abs(y - y0), abs(y - y1)) < 12:
                return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    paths = scene["paths"]
    zones = scene["zones"]
    terminal_colors = scene["terminal_colors"]
    if not (len(paths) == len(zones) == len(terminal_colors) == 3):
        raise ValueError("scene must contain three paired rows")
    image = Image.new("RGB", (WIDTH * SS, HEIGHT * SS), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x0, y0, x1, y1 in zones:
        draw.rectangle((x0 * SS, y0 * SS, x1 * SS, y1 * SS), outline=CYAN, width=6 * SS)
    for path in paths:
        draw.line([(x * SS, y * SS) for x, y in path], fill=DARK, width=8 * SS, joint="curve")
    for row, path in enumerate(paths):
        sx, sy = path[0]
        ex, ey = path[-1]
        draw.ellipse(((sx - 14) * SS, (sy - 14) * SS, (sx + 14) * SS, (sy + 14) * SS),
                     fill=BACKGROUND, outline=GREEN, width=7 * SS)
        draw.rectangle(((ex - 14) * SS, (ey - 14) * SS, (ex + 14) * SS, (ey + 14) * SS),
                       fill=FILLS[terminal_colors[row]], outline=DARK, width=3 * SS)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
