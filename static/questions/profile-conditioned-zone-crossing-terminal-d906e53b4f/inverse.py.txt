"""Independent inverse arm using only raster pixels and the public palette."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

CYAN = np.array([25, 181, 197], dtype=np.int16)
DARK = np.array([38, 50, 56], dtype=np.int16)
TERMINALS = {
    "orange": np.array([242, 142, 43], dtype=np.int16),
    "purple": np.array([142, 90, 200], dtype=np.int16),
    "blue": np.array([55, 126, 184], dtype=np.int16),
}


def _near(arr: np.ndarray, color: np.ndarray, radius: int = 48) -> np.ndarray:
    delta = arr.astype(np.int32) - color.astype(np.int32)
    return np.sum(delta * delta, axis=2) <= radius * radius


def _runs(values: np.ndarray, minimum: int = 3) -> int:
    padded = np.pad(values.astype(np.int8), (1, 1))
    starts = np.flatnonzero(np.diff(padded) == 1)
    ends = np.flatnonzero(np.diff(padded) == -1)
    return int(np.sum((ends - starts) >= minimum))


def _largest_component(mask: np.ndarray) -> np.ndarray:
    """Return the largest 8-connected dark component in one row band."""
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    best: list[tuple[int, int]] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        q = deque([(int(y), int(x))])
        seen[y, x] = True
        points = []
        while q:
            py, px = q.popleft()
            points.append((py, px))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    ny, nx = py + dy, px + dx
                    if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        q.append((ny, nx))
        if len(points) > len(best):
            best = points
    if len(best) < 500:
        raise ValueError("no connected cable component")
    out = np.zeros_like(mask, dtype=bool)
    yy, xx = zip(*best)
    out[np.array(yy), np.array(xx)] = True
    return out


def _zone_bands(cyan: np.ndarray) -> list[tuple[int, int]]:
    # Long horizontal outline strokes identify the six top/bottom edges even
    # when a cable simultaneously cuts both vertical sides at one y value.
    ys = np.flatnonzero(cyan.sum(axis=1) > 80)
    if not len(ys):
        raise ValueError("cyan zones are absent")
    groups = []
    start = previous = int(ys[0])
    for value in ys[1:]:
        value = int(value)
        if value > previous + 1:
            groups.append((start, previous))
            start = value
        previous = value
    groups.append((start, previous))
    if len(groups) != 6:
        raise ValueError(f"expected six horizontal zone edges, found {len(groups)}")
    centers = [(lo + hi) // 2 for lo, hi in groups]
    return [(centers[index], centers[index + 1]) for index in range(0, 6, 2)]


def _crossings_for_row(dark: np.ndarray, cyan: np.ndarray, band: tuple[int, int]) -> tuple[int, int]:
    y0, y1 = band
    sub_cyan = cyan[y0:y1 + 1]
    xs = np.flatnonzero(sub_cyan.sum(axis=0) > 10)
    if len(xs) < 8:
        raise ValueError("zone outline is incomplete")
    x0, x1 = int(xs[0]), int(xs[-1])
    cable_band = dark[max(0, y0 - 24):min(dark.shape[0], y1 + 25)]
    cable = _largest_component(cable_band)
    full = np.zeros_like(dark)
    full[max(0, y0 - 24):min(dark.shape[0], y1 + 25)] = cable

    # Each projection detects separated places where the reconstructed cable
    # component intersects a narrow strip centred on one visible zone edge.
    left = np.any(full[y0 + 10:y1 - 9, max(0, x0 - 6):x0 + 7], axis=1)
    right = np.any(full[y0 + 10:y1 - 9, x1 - 6:min(full.shape[1], x1 + 7)], axis=1)
    top = np.any(full[max(0, y0 - 6):y0 + 7, x0 + 10:x1 - 9], axis=0)
    bottom = np.any(full[y1 - 6:min(full.shape[0], y1 + 7), x0 + 10:x1 - 9], axis=0)
    count = _runs(left) + _runs(right) + _runs(top) + _runs(bottom)
    return count, (y0 + y1) // 2


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    if arr.ndim != 3 or arr.shape[0] < 300 or arr.shape[1] < 400:
        raise ValueError("unexpected raster")
    cyan = _near(arr, CYAN)
    dark = _near(arr, DARK)
    rows = []
    for band in _zone_bands(cyan):
        count, cy = _crossings_for_row(dark, cyan, band)
        rows.append((count, cy))
    winners = [cy for count, cy in rows if count == 4]
    if len(winners) != 1:
        raise ValueError(f"no unique four-crossing cable: {rows}")
    target_y = winners[0]

    best_name = None
    best_distance = float("inf")
    for name, color in TERMINALS.items():
        mask = _near(arr, color)
        points = np.argwhere(mask)
        if len(points) < 40:
            raise ValueError(f"missing {name} terminal")
        cy = float(points[:, 0].mean())
        distance = abs(cy - target_y)
        if distance < best_distance:
            best_name, best_distance = name, distance
    if best_name is None or best_distance > 45:
        raise ValueError("terminal binding is ambiguous")
    return best_name
