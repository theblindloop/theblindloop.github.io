"""Rainbow band-count world (prior_conflict profile).

The familiar canonical prior is that a rainbow has seven color bands
(red-orange-yellow-green-blue-indigo-violet, ROYGBIV). Here the image is a
painted arcing rainbow whose visible band count N may contradict that prior, and
the question asks to report the *visible* count. The closed-form gold is simply

    count = N        (the number of concentric spectral color bands drawn)

an exact scene parameter. The continuous verifiability margin is the minimum
pairwise RGB distance between adjacent band colors (how cleanly the bands can be
separated pixel-wise); the arc bands are thick (BW px) so they never fuse and the
count is always pixel-recoverable off-quarantine.

Latent state z = {canvas, arc center/radius, band thickness, ordered list of N
band colors}. Each band is one distinct, well-separated spectral color drawn as a
thick upper-semicircle arc; consecutive bands tile the annulus exactly. No two
bands share a color, so counting the number of distinct spectral colors present
in the raster recovers N exactly and independently of the geometry.
"""

from __future__ import annotations

import dataclasses
import math
import random
from dataclasses import dataclass

from PIL import Image, ImageDraw

RENDERER_VERSION = "prior_rainbow_bands-0.1.0"

# 8 well-separated spectral band colors, in canonical rainbow order. Consecutive
# RGB distances are all > 85 px so nearest-color labeling cleanly separates them.
BAND_PALETTE: tuple[tuple[int, int, int], ...] = (
    (216, 32, 32),    # red
    (240, 132, 20),   # orange
    (248, 216, 32),   # yellow
    (40, 160, 56),    # green
    (0, 184, 200),    # cyan
    (48, 64, 212),    # blue
    (130, 30, 190),   # violet
    (220, 40, 130),   # magenta
)
BACKGROUND = (255, 255, 255)

# Canvas and arc geometry.
CANVAS = (640, 400)
ARC_CENTER = (320, 360)  # below the top so only the upper semicircle shows
INNER_RADIUS = 120.0
BW = 20.0  # band thickness px

# Sampled band count range. The canonical prior is 7; counterfactual scenes use a
# different N so the visible truth can contradict the prior.
N_RANGE = (3, 8)  # inclusive

# Quarantine: bands thinner than this fuse under antialiasing. The sampler always
# uses BW = 20, so nothing is ever quarantined (off-quarantine only).
QUARANTINE_THICKNESS_PX = 6.0


def _rgb_distance(a: tuple[int, int, int], b: tuple[int, int, int]) -> float:
    return math.dist(a, b)


def min_adjacent_color_distance(n_bands: int) -> float:
    colors = list(BAND_PALETTE[:n_bands])
    return min(_rgb_distance(colors[i], colors[i + 1]) for i in range(len(colors) - 1))


@dataclass(frozen=True)
class RainbowScene:
    n_bands: int
    width: int = CANVAS[0]
    height: int = CANVAS[1]
    center: tuple[float, float] = ARC_CENTER
    inner_radius: float = INNER_RADIUS
    bandwidth: float = BW
    background: tuple[int, int, int] = BACKGROUND

    @property
    def count(self) -> int:
        return self.n_bands

    def band_colors(self) -> list[tuple[int, int, int]]:
        return list(BAND_PALETTE[: self.n_bands])

    def to_dict(self) -> dict:
        d = dataclasses.asdict(self)
        d["center"] = [float(v) for v in self.center]
        return d


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    n = rng.randint(*N_RANGE)
    scene = RainbowScene(n_bands=n)
    return scene.to_dict()


def scene_from_dict(d: dict) -> RainbowScene:
    return RainbowScene(
        n_bands=int(d["n_bands"]),
        width=int(d["width"]),
        height=int(d["height"]),
        center=(float(d["center"][0]), float(d["center"][1])),
        inner_radius=float(d["inner_radius"]),
        bandwidth=float(d["bandwidth"]),
        background=tuple(int(v) for v in d["background"]),
    )


def render(scene_dict: dict) -> Image.Image:
    scene = scene_from_dict(scene_dict)
    w, h = scene.width, scene.height
    img = Image.new("RGB", (w, h), scene.background)
    draw = ImageDraw.Draw(img)
    cx, cy = scene.center
    colors = scene.band_colors()
    for i, color in enumerate(colors):
        r = scene.inner_radius + i * scene.bandwidth
        bbox = [cx - r, cy - r, cx + r, cy + r]
        draw.arc(bbox, start=180, end=360, fill=color, width=int(round(scene.bandwidth)))
    return img


def analytic_gold(scene_dict: dict) -> int:
    scene = scene_from_dict(scene_dict)
    return scene.n_bands


def margin(scene_dict: dict) -> float:
    scene = scene_from_dict(scene_dict)
    return float(min_adjacent_color_distance(scene.n_bands))


def is_quarantined(scene_dict: dict) -> bool:
    scene = scene_from_dict(scene_dict)
    return scene.bandwidth < QUARANTINE_THICKNESS_PX


def latent_symmetries(scene_dict: dict) -> list[tuple[str, dict]]:
    """No nontrivial pixel-identical latent symmetry exists.

    The count is a scene parameter equal to the number of distinct band colors
    drawn, and every band color occupies a distinct positional concentric ring
    that directly shows in the raster. Permuting band colors would change the
    pixels; changing the count changes the gold. The parametrization is minimal.
    """
    return []
