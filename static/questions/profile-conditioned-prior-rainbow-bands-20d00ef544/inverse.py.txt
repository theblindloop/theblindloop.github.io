"""Pixel-only independent inverse arm for the rainbow band-count world.

`decision_from_image(image)` receives ONLY the final PNG. It shares no code with
the renderer, prompts, generator, or verifier and never sees any latent scene.
It labels every pixel by its nearest color (the eight band colors + white
background, duplicated here deliberately), then counts how many distinct band
colors are present as a substantial arc band. Each band is a distinct color, so
that count is exactly N and is independent of the geometry.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "prior_rainbow_bands-oracle-0.1.0"

# Public visual spec, duplicated deliberately (never imported from the renderer).
BAND_COLORS: tuple[tuple[int, int, int], ...] = (
    (216, 32, 32),
    (240, 132, 20),
    (248, 216, 32),
    (40, 160, 56),
    (0, 184, 200),
    (48, 64, 212),
    (130, 30, 190),
    (220, 40, 130),
)
BACKGROUND = (255, 255, 255)

# A genuine band is a thick arc: hundreds of pixels. Threshold well below the
# smallest band area and far above any antialiasing speckle.
_BAND_AREA_MIN = 800.0


def _nearest_labels(arr: np.ndarray, palette: list[tuple[int, int, int]]) -> np.ndarray:
    dists = np.stack(
        [np.linalg.norm(arr - np.array(c, dtype=np.float32), axis=-1) for c in palette], axis=-1
    )
    return dists.argmin(axis=-1)


def measure_bands(image: Image.Image) -> list[int]:
    """Count of distinct spectral bands present (indices into BAND_COLORS)."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    palette: list[tuple[int, int, int]] = list(BAND_COLORS) + [BACKGROUND]
    labels = _nearest_labels(arr, palette)
    present: list[int] = []
    for idx, _color in enumerate(BAND_COLORS):
        count = int(np.count_nonzero(labels == idx))
        if count >= _BAND_AREA_MIN:
            present.append(idx)
    return present


def decision_from_image(image: Image.Image) -> int:
    """Recover the visible band count N from the raster alone."""
    return len(measure_bands(image))
