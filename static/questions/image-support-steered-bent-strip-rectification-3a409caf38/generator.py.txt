"""Deterministic latent sampler and renderer for Bent Strip Rectification."""

from __future__ import annotations

import random
from typing import Iterable

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
COLS = 9
ROWS = 5
COL_W = 8
ROW_H = 4
STRIP_W = COLS * COL_W
STRIP_H = ROWS * ROW_H
BACKGROUND = (248, 247, 242)
RED = (210, 48, 67)
GOLD = (235, 170, 37)
DISTRACTOR = (184, 184, 178)

PALETTES = (
    ((32, 183, 180), (28, 38, 57)),
    ((63, 169, 214), (45, 31, 62)),
    ((48, 190, 142), (41, 43, 68)),
)

TEMPLATES: dict[str, tuple[str, ...]] = {
    "arch": (
        "011111110",
        "010000010",
        "010000010",
        "010000010",
        "010000010",
    ),
    "cup": (
        "010000010",
        "010000010",
        "010000010",
        "010000010",
        "011111110",
    ),
    "tee": (
        "011111110",
        "000111000",
        "000010000",
        "000010000",
        "000111000",
    ),
}
SYMBOLS = tuple(TEMPLATES)


def _matrix(rows: Iterable[str]) -> np.ndarray:
    values = np.asarray([[character == "1" for character in row] for row in rows], dtype=bool)
    if values.shape != (ROWS, COLS):
        raise ValueError(f"ink rows must form a {ROWS}-by-{COLS} raster")
    return values


def _distances(rows: Iterable[str]) -> list[tuple[int, str]]:
    observed = _matrix(rows)
    return sorted(
        (int(np.count_nonzero(observed != _matrix(template))), name)
        for name, template in TEMPLATES.items()
    )


def analytic_gold(scene: dict) -> str:
    distances = _distances(scene["rows"])
    return "ambiguous" if distances[0][0] == distances[1][0] else distances[0][1]


def margin(scene: dict) -> float:
    distances = _distances(scene["rows"])
    return float(distances[1][0] - distances[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _tie_rows() -> list[str]:
    left = _matrix(TEMPLATES["arch"])
    right = _matrix(TEMPLATES["cup"])
    differing = list(zip(*np.nonzero(left != right), strict=True))
    mixed = left.copy()
    for row, column in differing[len(differing) // 2 :]:
        mixed[row, column] = right[row, column]
    return ["".join("1" if value else "0" for value in row) for row in mixed]


def _noisy_rows(symbol: str, rng: random.Random, swaps: int) -> list[str]:
    values = _matrix(TEMPLATES[symbol]).copy()
    for _ in range(swaps):
        ink = [(r, c) for r, c in zip(*np.nonzero(values), strict=True) if c not in (0, COLS - 1)]
        blank = [
            (r, c)
            for r, c in zip(*np.nonzero(~values), strict=True)
            if c not in (0, COLS - 1)
        ]
        off = rng.choice(ink)
        on = rng.choice(blank)
        values[off] = False
        values[on] = True
    return ["".join("1" if value else "0" for value in row) for row in values]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    symbol = SYMBOLS[int(seed) % len(SYMBOLS)]
    rows = _tie_rows() if int(seed) % 13 == 0 else _noisy_rows(symbol, rng, (int(seed) // 3) % 2)

    offsets = [rng.randint(-2, 2)]
    for _ in range(1, COLS):
        offsets.append(max(-3, min(3, offsets[-1] + rng.choice((-2, -1, 0, 1, 2)))))

    quarter_turn = rng.randrange(4)
    if quarter_turn % 2 == 0:
        center_x = rng.choice((128, 192, 256, 320, 384))
        center_y = rng.choice((96, 160, 224, 288, 352, 416))
    else:
        center_x = rng.choice((96, 160, 224, 288, 352, 416))
        center_y = rng.choice((128, 192, 256, 320, 384))

    distractors: list[list[int]] = []
    while len(distractors) < 7:
        x, y = rng.randrange(28, WIDTH - 28), rng.randrange(28, HEIGHT - 28)
        if abs(x - center_x) + abs(y - center_y) > 150:
            distractors.append([x, y, rng.randrange(2)])

    return {
        "rows": rows,
        "offsets": offsets,
        "center_x": center_x,
        "center_y": center_y,
        "quarter_turn": quarter_turn,
        "palette_index": (int(seed) // 3) % len(PALETTES),
        "distractors": distractors,
        "width": WIDTH,
        "height": HEIGHT,
    }


def _local_strip(scene: dict) -> Image.Image:
    carrier, ink = PALETTES[int(scene["palette_index"])]
    array = np.zeros((64, 96, 4), dtype=np.uint8)
    bits = _matrix(scene["rows"])
    offsets = [int(value) for value in scene["offsets"]]
    if len(offsets) != COLS or any(abs(a - b) > 2 for a, b in zip(offsets, offsets[1:])):
        raise ValueError("offsets must be a nine-element connected walk")
    for column in range(COLS):
        x0 = 12 + column * COL_W
        y0 = 22 + offsets[column]
        array[y0 : y0 + STRIP_H, x0 : x0 + COL_W, :3] = carrier
        array[y0 : y0 + STRIP_H, x0 : x0 + COL_W, 3] = 255
        for row in range(ROWS):
            if bits[row, column]:
                array[
                    y0 + row * ROW_H : y0 + (row + 1) * ROW_H,
                    x0 + 1 : x0 + COL_W - 1,
                    :3,
                ] = ink

    local = Image.fromarray(array, mode="RGBA")
    draw = ImageDraw.Draw(local)
    start_y = 22 + offsets[0] + STRIP_H // 2
    end_y = 22 + offsets[-1] + STRIP_H // 2
    draw.ellipse((13, start_y - 3, 19, start_y + 3), outline=RED + (255,), width=2)
    draw.polygon(((80, end_y - 4), (83, end_y + 3), (76, end_y + 3)), fill=GOLD + (255,))

    turn = int(scene["quarter_turn"]) % 4
    for _ in range(turn):
        local = local.transpose(Image.Transpose.ROTATE_90)
    return local


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x, y, kind in scene["distractors"]:
        if int(kind) == 0:
            draw.line((x - 4, y, x + 4, y), fill=DISTRACTOR, width=1)
            draw.line((x, y - 4, x, y + 4), fill=DISTRACTOR, width=1)
        else:
            draw.ellipse((x - 3, y - 3, x + 3, y + 3), outline=DISTRACTOR, width=1)

    local = _local_strip(scene)
    left = int(scene["center_x"]) - local.width // 2
    top = int(scene["center_y"]) - local.height // 2
    image.paste(local, (left, top), local)
    return image
