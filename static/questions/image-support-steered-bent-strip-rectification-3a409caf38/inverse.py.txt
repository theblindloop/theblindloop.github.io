"""Independent final-PNG inverse for Bent Strip Rectification.

This module intentionally duplicates only the public palette and glyph contract.
It never imports or receives renderer state. It finds the two endpoint fiducials,
normalizes the raster direction, infers each crossbar's visible edge displacement,
and transports dark pixels into a straight five-by-nine raster.
"""

from __future__ import annotations

from dataclasses import dataclass

import numpy as np
from PIL import Image

COLS = 9
ROWS = 5
COL_W = 8
ROW_H = 4
RED = (210, 48, 67)
GOLD = (235, 170, 37)
PALETTES = (
    ((32, 183, 180), (28, 38, 57)),
    ((63, 169, 214), (45, 31, 62)),
    ((48, 190, 142), (41, 43, 68)),
)
GLYPHS: dict[str, tuple[str, ...]] = {
    "arch": ("011111110", "010000010", "010000010", "010000010", "010000010"),
    "cup": ("010000010", "010000010", "010000010", "010000010", "011111110"),
    "tee": ("011111110", "000111000", "000010000", "000010000", "000111000"),
}


class OracleError(ValueError):
    pass


@dataclass(frozen=True)
class Measurement:
    decision: str
    margin: float
    raster: tuple[str, ...]
    crossbar_offsets: tuple[int, ...]


def _mask(array: np.ndarray, color: tuple[int, int, int]) -> np.ndarray:
    return np.all(array == np.asarray(color, dtype=np.uint8), axis=2)


def _centroid(mask: np.ndarray, name: str) -> tuple[float, float]:
    points = np.argwhere(mask)
    if len(points) < 6:
        raise OracleError(f"missing {name} fiducial")
    return float(points[:, 1].mean()), float(points[:, 0].mean())


def _orient_red_to_gold(array: np.ndarray) -> np.ndarray:
    red = _centroid(_mask(array, RED), "red-ring")
    gold = _centroid(_mask(array, GOLD), "gold-triangle")
    dx, dy = gold[0] - red[0], gold[1] - red[1]
    if abs(dx) >= abs(dy):
        turns = 0 if dx > 0 else 2
    else:
        turns = 1 if dy > 0 else 3
    oriented = np.rot90(array, turns)
    red2 = _centroid(_mask(oriented, RED), "red-ring")
    gold2 = _centroid(_mask(oriented, GOLD), "gold-triangle")
    if gold2[0] <= red2[0] or abs(gold2[1] - red2[1]) > 12:
        raise OracleError("endpoint direction is not a rectifiable quarter turn")
    return oriented


def _template(name: str) -> np.ndarray:
    return np.asarray([[ch == "1" for ch in row] for row in GLYPHS[name]], dtype=bool)


def measure_from_image(image: Image.Image) -> Measurement:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    array = _orient_red_to_gold(array)

    palette_scores = []
    for carrier, ink in PALETTES:
        palette_scores.append((int(_mask(array, carrier).sum() + _mask(array, ink).sum()), carrier, ink))
    score, carrier_color, ink_color = max(palette_scores)
    if score < 900:
        raise OracleError("no complete supported cyan carrier palette")

    carrier = _mask(array, carrier_color)
    ink = _mask(array, ink_color)
    red = _mask(array, RED)
    gold = _mask(array, GOLD)
    active = carrier | ink | red | gold
    points = np.argwhere(active)
    if not len(points):
        raise OracleError("strip is absent")
    y_min, x_min = points.min(axis=0)
    y_max, x_max = points.max(axis=0) + 1
    if x_max - x_min != COLS * COL_W:
        raise OracleError(f"strip has invalid along-travel extent {x_max - x_min}")
    if not 20 <= y_max - y_min <= 26:
        raise OracleError(f"strip has invalid cross-travel extent {y_max - y_min}")

    decoded = np.zeros((ROWS, COLS), dtype=bool)
    offsets: list[int] = []
    for column in range(COLS):
        xa = x_min + column * COL_W
        xb = xa + COL_W
        top_values: list[int] = []
        for x in range(xa, xb):
            ys = np.flatnonzero(active[:, x])
            if len(ys) != ROWS * ROW_H or np.any(np.diff(ys) != 1):
                raise OracleError(f"crossbar {column} is incomplete")
            top_values.append(int(ys[0]))
        if len(set(top_values)) != 1:
            raise OracleError(f"crossbar {column} does not translate rigidly")
        top = top_values[0]
        offsets.append(top)
        for row in range(ROWS):
            cell = ink[top + row * ROW_H : top + (row + 1) * ROW_H, xa:xb]
            count = int(cell.sum())
            if count not in (0, 24):
                raise OracleError(f"ink cell ({row},{column}) is corrupted")
            decoded[row, column] = count == 24
    if any(abs(a - b) > 2 for a, b in zip(offsets, offsets[1:])):
        raise OracleError("carrier is not one connected gently bent strip")

    distances = sorted(
        (int(np.count_nonzero(decoded != _template(name))), name) for name in GLYPHS
    )
    decision = "ambiguous" if distances[0][0] == distances[1][0] else distances[0][1]
    margin = float(distances[1][0] - distances[0][0])
    raster = tuple("".join("1" if value else "0" for value in row) for row in decoded)
    normalized_offsets = tuple(value - min(offsets) for value in offsets)
    return Measurement(decision, margin, raster, normalized_offsets)


def decision_from_image(image: Image.Image) -> str:
    return measure_from_image(image).decision
