"""Deterministic raster world for the visible mitten-thumb-side question."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "mitten-thumb-side-renderer-0.1.0"
WIDTH = 640
HEIGHT = 640
QUARANTINE_CLEARANCE = 9.0

FILL_COLORS = (
    (210, 67, 72),
    (48, 119, 204),
    (44, 157, 111),
    (168, 87, 181),
    (226, 135, 42),
    (31, 157, 164),
)
BORDER_COLORS = (
    (29, 36, 46),
    (55, 43, 35),
    (37, 61, 59),
)
BACKGROUNDS = (
    (248, 247, 242),
    (239, 246, 250),
    (248, 241, 237),
    (241, 247, 241),
)


def _outline(scene: dict) -> list[tuple[float, float]]:
    """Build one closed, thick palm-plus-thumb boundary in image coordinates."""

    side = int(scene["thumb_side"])
    if side not in (-1, 1):
        raise ValueError("thumb_side must be -1 or 1")
    palm_width = float(scene["palm_width"])
    palm_height = float(scene["palm_height"])
    thumb_length = float(scene["thumb_length"])
    thumb_height = float(scene["thumb_height"])
    thumb_offset = float(scene["thumb_offset"])
    half_width = palm_width / 2.0
    top = -palm_height / 2.0
    bottom = palm_height / 2.0
    left = -half_width
    right = half_width
    # Clockwise from the crown, down the ordinary palm edge, around the
    # broad thumb lobe, and back to the crown. The lobe is area-like rather
    # than a thin appendage so its visible support spans several bands.
    points = [
        (-0.68 * half_width, top + 4.0),
        (-0.28 * half_width, top - 8.0),
        (0.24 * half_width, top - 5.0),
        (0.46 * half_width, top + 22.0),
        (0.98 * half_width, top + 72.0),
        (right + 4.0, thumb_offset + 0.52 * thumb_height),
        (right - 4.0, bottom - 42.0),
        (0.30 * half_width, bottom - 3.0),
        (-0.24 * half_width, bottom + 5.0),
        (-0.72 * half_width, bottom - 9.0),
        (left - 2.0, bottom - 56.0),
    ]
    thumb_points = [
        (left - 0.20 * thumb_length, thumb_offset + 0.48 * thumb_height),
        (left - 0.72 * thumb_length, thumb_offset + 0.36 * thumb_height),
        (left - 1.00 * thumb_length, thumb_offset + 0.08 * thumb_height),
        (left - 0.94 * thumb_length, thumb_offset - 0.20 * thumb_height),
        (left - 0.67 * thumb_length, thumb_offset - 0.43 * thumb_height),
        (left - 0.25 * thumb_length, thumb_offset - 0.53 * thumb_height),
        (left - 3.0, top + 74.0),
    ]
    points.extend(thumb_points)
    # The base outline is drawn with its thumb on the left (negative x).
    # Serialized -1 means left and +1 means right, so the mirror multiplier
    # is the negation of the label value.
    points = [(-side * x, y) for x, y in points]
    scale = float(scene["scale"])
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    transformed = [(cx + x * scale, cy + y * scale) for x, y in points]
    # vertex_order names the serialized starting index, but the renderer
    # canonicalizes that representation before rasterizing. This keeps the
    # declared representation alias byte-identical even for antialiased line
    # joins whose numerical walk order could otherwise differ by a pixel.
    order = int(scene["vertex_order"]) % len(transformed)
    rotated = transformed[order:] + transformed[:order]
    return rotated[-order:] + rotated[:-order] if order else rotated


def sample_scene(seed: int) -> dict:
    """Sample matched handedness classes with visible nuisance variation."""

    rng = random.Random(int(seed))
    thumb_side = -1 if int(seed) % 2 == 0 else 1
    return {
        "thumb_side": thumb_side,
        "center_x": round(WIDTH / 2.0 + rng.uniform(-34.0, 34.0), 6),
        "center_y": round(HEIGHT / 2.0 + rng.uniform(-28.0, 28.0), 6),
        "scale": round(rng.uniform(0.91, 1.06), 6),
        "palm_width": round(rng.uniform(190.0, 218.0), 6),
        "palm_height": round(rng.uniform(326.0, 366.0), 6),
        "thumb_length": round(rng.uniform(132.0, 158.0), 6),
        "thumb_height": round(rng.uniform(132.0, 160.0), 6),
        "thumb_offset": round(rng.uniform(-4.0, 22.0), 6),
        "fill_color": FILL_COLORS[(int(seed) * 17 + 3) % len(FILL_COLORS)],
        "border_color": BORDER_COLORS[(int(seed) * 11 + 1) % len(BORDER_COLORS)],
        "border_width": 7 + int(seed) % 4,
        "background": BACKGROUNDS[(int(seed) * 13 + 2) % len(BACKGROUNDS)],
        "width": WIDTH,
        "height": HEIGHT,
        "antialias": True,
        "vertex_order": 0,
    }


def render(scene: dict) -> Image.Image:
    """Render one filled, outlined mitten to a deterministic RGB image."""

    if int(scene["thumb_side"]) not in (-1, 1):
        raise ValueError("unsupported thumb side")
    width = int(scene["width"])
    height = int(scene["height"])
    supersample = 4 if bool(scene["antialias"]) else 1
    image = Image.new(
        "RGB",
        (width * supersample, height * supersample),
        tuple(int(value) for value in scene["background"]),
    )
    draw = ImageDraw.Draw(image)
    points = _outline(scene)
    scaled = [(round(x * supersample), round(y * supersample)) for x, y in points]
    draw.polygon(scaled, fill=tuple(int(value) for value in scene["fill_color"]))
    closed = scaled + [scaled[0]]
    draw.line(
        closed,
        fill=tuple(int(value) for value in scene["border_color"]),
        width=max(1, int(scene["border_width"]) * supersample),
        joint="curve",
    )
    if supersample > 1:
        image = image.resize((width, height), Image.Resampling.LANCZOS)
    return image


def analytic_gold(scene: dict) -> str:
    return "left" if int(scene["thumb_side"]) < 0 else "right"


def margin(scene: dict) -> float:
    """Clearance of the thumb lobe above a finite-resolution width floor."""

    return float(float(scene["thumb_length"]) - 120.0)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_CLEARANCE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Rotate the starting index of the same closed polygon representation."""

    twin = dict(scene)
    twin["vertex_order"] = (int(scene["vertex_order"]) + 1) % 18
    return [("cyclic_vertex_order", twin)]
