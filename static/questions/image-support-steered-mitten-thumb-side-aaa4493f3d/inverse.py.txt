"""Independent pixel-only inverse for the visible mitten thumb side."""

from __future__ import annotations

import numpy as np
from PIL import Image

_BACKGROUND_DELTA = 26.0
_MIN_FOREGROUND = 12000
_MIN_SPAN = 120.0
_MIN_SIDE_SCORE = 12.0
_MIN_FILL_RATIO = 0.63


class OracleAbstain(ValueError):
    """The raster does not expose a complete, sufficiently broad mitten."""


def _background(array: np.ndarray) -> np.ndarray:
    border = np.concatenate(
        (
            array[:4].reshape(-1, 3),
            array[-4:].reshape(-1, 3),
            array[:, :4].reshape(-1, 3),
            array[:, -4:].reshape(-1, 3),
        ),
        axis=0,
    )
    return np.median(border, axis=0).astype(np.float32)


def _foreground(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    background = _background(array)
    distance = np.linalg.norm(array - background, axis=2)
    mask = distance > _BACKGROUND_DELTA
    if int(mask.sum()) < _MIN_FOREGROUND:
        raise OracleAbstain("foreground mitten body is too small")
    ys, xs = np.nonzero(mask)
    bbox_area = float((ys.max() - ys.min() + 1) * (xs.max() - xs.min() + 1))
    if float(mask.sum()) / bbox_area < _MIN_FILL_RATIO:
        raise OracleAbstain("foreground mitten body is materially incomplete")
    return mask


def _row_edges(mask: np.ndarray, start: int, stop: int) -> tuple[float, float]:
    rows = mask[max(0, start) : min(mask.shape[0], stop)]
    occupied = np.flatnonzero(np.any(rows, axis=1))
    if len(occupied) < max(8, rows.shape[0] // 3):
        raise OracleAbstain("a required palm shoulder band is missing")
    lefts = []
    rights = []
    for row in rows[occupied]:
        xs = np.flatnonzero(row)
        if len(xs) < 25:
            continue
        lefts.append(float(xs[0]))
        rights.append(float(xs[-1]))
    if len(lefts) < 8:
        raise OracleAbstain("palm shoulder rows are too fragmented")
    return float(np.median(lefts)), float(np.median(rights))


def _side_score(mask: np.ndarray) -> float:
    ys, xs = np.nonzero(mask)
    if len(xs) < _MIN_FOREGROUND:
        raise OracleAbstain("foreground is too small")
    top = int(ys.min())
    bottom = int(ys.max()) + 1
    span = float(bottom - top)
    if span < _MIN_SPAN:
        raise OracleAbstain("mitten vertical span is too small")
    x_span = float(xs.max() - xs.min() + 1)
    if x_span < _MIN_SPAN:
        raise OracleAbstain("mitten horizontal span is too small")

    # The upper and lower palm bands do not contain the thumb. They define a
    # visible palm-side baseline. The middle shoulder bands contain the broad
    # lobe, so the larger side-specific excess identifies the thumb.
    upper = _row_edges(mask, top + int(0.15 * span), top + int(0.29 * span))
    lower = _row_edges(mask, top + int(0.72 * span), top + int(0.90 * span))
    middle = _row_edges(mask, top + int(0.34 * span), top + int(0.64 * span))
    base_left = float(np.median((upper[0], lower[0])))
    base_right = float(np.median((upper[1], lower[1])))
    left_excess = base_left - middle[0]
    right_excess = middle[1] - base_right
    return float(right_excess - left_excess)


def decision_from_image(image: Image.Image) -> str:
    """Return the visible thumb side from the final RGB PNG only."""

    score = _side_score(_foreground(image))
    if abs(score) < _MIN_SIDE_SCORE:
        raise OracleAbstain("left/right width evidence is too close")
    return "right" if score > 0.0 else "left"
