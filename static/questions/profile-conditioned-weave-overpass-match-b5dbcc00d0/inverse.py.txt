"""Independent pixel-only inverse for ribbon overpass cards."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

RIBBONS = {
    "cyan": np.array((24, 170, 196)),
    "orange": np.array((239, 126, 42)),
    "purple": np.array((139, 82, 190)),
}
BORDERS = {
    "red": np.array((215, 59, 62)),
    "blue": np.array((55, 104, 190)),
    "gold": np.array((211, 155, 34)),
    "reference": np.array((35, 35, 35)),
}
PAIRS = (("cyan", "orange"), ("cyan", "purple"), ("orange", "purple"))


def _mask(arr: np.ndarray, rgb: np.ndarray, tolerance: float = 8.0) -> np.ndarray:
    return np.linalg.norm(arr.astype(np.float32) - rgb, axis=2) <= tolerance


def _largest_box(mask: np.ndarray) -> tuple[int, int, int, int]:
    seen = np.zeros(mask.shape, dtype=bool)
    best = None
    height, width = mask.shape
    for start_y, start_x in np.argwhere(mask & ~seen):
        if seen[start_y, start_x]:
            continue
        queue = deque([(int(start_y), int(start_x))])
        seen[start_y, start_x] = True
        xs, ys = [], []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) > 500:
            box = (min(xs), min(ys), max(xs), max(ys), len(xs))
            if best is None or box[4] > best[4]:
                best = box
    if best is None or best[2] - best[0] < 180 or best[3] - best[1] < 180:
        raise ValueError("card border not found")
    return best[:4]


def _fit_line(points: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    # points arrive as (y, x); PCA obtains the infinite ribbon centerline.
    xy = points[:, ::-1].astype(float)
    center = xy.mean(axis=0)
    _, _, vh = np.linalg.svd(xy - center, full_matrices=False)
    return center, vh[0]


def _intersect(line_a: tuple[np.ndarray, np.ndarray], line_b: tuple[np.ndarray, np.ndarray]) -> np.ndarray:
    ca, da = line_a
    cb, db = line_b
    matrix = np.column_stack((da, -db))
    if abs(np.linalg.det(matrix)) < 0.25:
        raise ValueError("ribbon lines are too parallel")
    t, _ = np.linalg.solve(matrix, cb - ca)
    return ca + t * da


def _signature(arr: np.ndarray, box: tuple[int, int, int, int]) -> tuple[str, ...]:
    x0, y0, x1, y1 = box
    # Ignore the colored border itself with a visible inset.
    crop = arr[y0 + 12 : y1 - 11, x0 + 12 : x1 - 11]
    masks = {name: _mask(crop, rgb) for name, rgb in RIBBONS.items()}
    lines = {}
    for name, mask in masks.items():
        points = np.argwhere(mask)
        if len(points) < 250:
            raise ValueError(f"insufficient {name} ribbon ink")
        lines[name] = _fit_line(points)
    result = []
    for first, second in PAIRS:
        point = _intersect(lines[first], lines[second])
        distances = {}
        for name in (first, second):
            pixels = np.argwhere(masks[name])[:, ::-1].astype(float)
            distances[name] = float(np.linalg.norm(pixels - point, axis=1).min())
        ordered = sorted(distances, key=distances.get)
        if distances[ordered[1]] - distances[ordered[0]] < 5.0 or distances[ordered[0]] > 5.0:
            raise ValueError("ambiguous overpass evidence")
        result.append(ordered[0])
    return tuple(result)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    boxes = {name: _largest_box(_mask(arr, rgb)) for name, rgb in BORDERS.items()}
    reference = _signature(arr, boxes["reference"])
    matches = [name for name in ("red", "blue", "gold") if _signature(arr, boxes[name]) == reference]
    if len(matches) != 1:
        raise ValueError("image does not contain one unique matching card")
    return matches[0]
