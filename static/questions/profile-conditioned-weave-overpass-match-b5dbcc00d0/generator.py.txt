"""Deterministic renderer for colored-ribbon overpass correspondence."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw, ImageFont

RIBBON_COLORS = {
    "cyan": (24, 170, 196),
    "orange": (239, 126, 42),
    "purple": (139, 82, 190),
}
BORDER_COLORS = {
    "red": (215, 59, 62),
    "blue": (55, 104, 190),
    "gold": (211, 155, 34),
}
PAIR_ORDER = (("cyan", "orange"), ("cyan", "purple"), ("orange", "purple"))
CARD_BOXES = ((315, 28, 585, 298), (25, 352, 295, 622), (315, 352, 585, 622), (605, 352, 875, 622))


def _signature(bits: int) -> list[str]:
    return [pair[(bits >> i) & 1] for i, pair in enumerate(PAIR_ORDER)]


def _make_card(rng: random.Random, bits: int) -> dict:
    # Use the sides of a visibly large, randomly turned triangle as infinite
    # centerlines. Pairwise crossings are therefore its three separated
    # vertices rather than a nearly concurrent bundle.
    turn = rng.uniform(-0.35, 0.35)
    vertices = []
    for index in range(3):
        angle = turn - math.pi / 2 + index * 2 * math.pi / 3
        radius = rng.uniform(49, 57)
        vertices.append((135 + radius * math.cos(angle), 135 + radius * math.sin(angle)))
    line_vertices = ((vertices[0], vertices[1]), (vertices[0], vertices[2]), (vertices[1], vertices[2]))
    ribbons = []
    for color, (first, second) in zip(RIBBON_COLORS, line_vertices):
        dx, dy = second[0] - first[0], second[1] - first[1]
        length = math.hypot(dx, dy)
        dx, dy = dx / length, dy / length
        extension = 55
        ribbons.append(
            {
                "color": color,
                "p1": [round(first[0] - extension * dx, 4), round(first[1] - extension * dy, 4)],
                "p2": [round(second[0] + extension * dx, 4), round(second[1] + extension * dy, 4)],
            }
        )
    return {"ribbons": ribbons, "over": _signature(bits)}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    reference_bits = rng.randrange(8)
    distractors = [value for value in range(8) if value != reference_bits]
    rng.shuffle(distractors)
    option_bits = [reference_bits, distractors[0], distractors[1]]
    rng.shuffle(option_bits)
    cards = [_make_card(rng, reference_bits)] + [_make_card(rng, bits) for bits in option_bits]

    # Gold identity is deliberately cycled by seed, while position is shuffled.
    desired_border = ("red", "blue", "gold")[int(seed) % 3]
    matching_index = option_bits.index(reference_bits)
    remaining = [name for name in BORDER_COLORS if name != desired_border]
    rng.shuffle(remaining)
    option_borders = [None, None, None]
    option_borders[matching_index] = desired_border
    cursor = 0
    for index in range(3):
        if option_borders[index] is None:
            option_borders[index] = remaining[cursor]
            cursor += 1
    return {"cards": cards, "option_borders": option_borders}


def _line_intersection(first: dict, second: dict) -> tuple[float, float]:
    x1, y1 = first["p1"]
    x2, y2 = first["p2"]
    x3, y3 = second["p1"]
    x4, y4 = second["p2"]
    den = (x1 - x2) * (y3 - y4) - (y1 - y2) * (x3 - x4)
    px = ((x1 * y2 - y1 * x2) * (x3 - x4) - (x1 - x2) * (x3 * y4 - y3 * x4)) / den
    py = ((x1 * y2 - y1 * x2) * (y3 - y4) - (y1 - y2) * (x3 * y4 - y3 * x4)) / den
    return px, py


def _draw_card(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], card: dict, border: tuple[int, int, int]) -> None:
    x0, y0, x1, y1 = box
    draw.rounded_rectangle(box, radius=12, fill=(250, 249, 246), outline=border, width=5)
    by_color = {ribbon["color"]: ribbon for ribbon in card["ribbons"]}
    for ribbon in card["ribbons"]:
        p1 = (x0 + ribbon["p1"][0], y0 + ribbon["p1"][1])
        p2 = (x0 + ribbon["p2"][0], y0 + ribbon["p2"][1])
        draw.line((p1, p2), fill=RIBBON_COLORS[ribbon["color"]], width=11)

    # Independently paint each crossing: erase a disk, then restore only the
    # overpassing ribbon through it. Thus local evidence need not be globally
    # consistent with one painter's-order stack.
    for pair, top in zip(PAIR_ORDER, card["over"]):
        first, second = by_color[pair[0]], by_color[pair[1]]
        cx, cy = _line_intersection(first, second)
        center = (x0 + cx, y0 + cy)
        radius = 13
        draw.ellipse((center[0] - radius, center[1] - radius, center[0] + radius, center[1] + radius), fill=(250, 249, 246))
        ribbon = by_color[top]
        ax, ay = ribbon["p1"]
        bx, by = ribbon["p2"]
        length = math.hypot(bx - ax, by - ay)
        dx, dy = (bx - ax) / length, (by - ay) / length
        span = 17
        draw.line(
            ((center[0] - span * dx, center[1] - span * dy), (center[0] + span * dx, center[1] + span * dy)),
            fill=RIBBON_COLORS[top],
            width=11,
        )


def render(scene: dict) -> Image.Image:
    cards = scene["cards"]
    option_borders = scene["option_borders"]
    image = Image.new("RGB", (900, 650), (242, 240, 235))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    draw.text((414, 8), "REFERENCE", fill=(25, 25, 25), font=font)
    _draw_card(draw, CARD_BOXES[0], cards[0], (35, 35, 35))
    for index in range(3):
        _draw_card(draw, CARD_BOXES[index + 1], cards[index + 1], BORDER_COLORS[option_borders[index]])
    return image


def analytic_gold(scene: dict) -> str:
    cards = scene["cards"]
    option_borders = scene["option_borders"]
    reference = tuple(cards[0]["over"])
    matches = [index for index, card in enumerate(cards[1:]) if tuple(card["over"]) == reference]
    if len(matches) != 1:
        raise ValueError("scene must have exactly one matching option")
    return option_borders[matches[0]]


def margin(scene: dict) -> float:
    cards = scene["cards"]
    option_borders = scene["option_borders"]  # label binding is part of the audited scene input
    _ = option_borders
    reference = cards[0]["over"]
    gaps = [sum(a != b for a, b in zip(reference, card["over"])) for card in cards[1:] if card["over"] != reference]
    return float(min(gaps) * 10)


def is_quarantined(scene: dict) -> bool:
    cards = scene["cards"]
    option_borders = scene["option_borders"]
    return len(cards) != 4 or len(set(option_borders)) != 3 or margin(scene) < 9.5


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    for card in twin["cards"]:
        for ribbon in card["ribbons"]:
            ribbon["p1"], ribbon["p2"] = ribbon["p2"], ribbon["p1"]
    return [("reverse_undirected_ribbon_endpoints", twin)]
