"""Deterministic renderer for the cycle-hitting-station world."""
from __future__ import annotations

import random
from collections import defaultdict
from PIL import Image, ImageDraw

CANVAS = 640
BG = (250, 250, 247)
INK = (31, 41, 55)
LINK = (100, 116, 139)
PALETTE = {"red": (220, 55, 65), "green": (24, 158, 105), "blue": (45, 108, 220)}
QUESTION = ("A loop is a closed route following links that repeats no station except its "
            "start/end. Imagine removing one colored station and every link touching it. "
            "Which removal leaves no loop anywhere? Answer red, green, or blue.")

# Two triangles share t.  Removing t destroys both cycles; removing b or d only one.
# Tree decorations equalize the three candidate degrees at four.
BASE = {
    "t": (320, 205), "a": (235, 125), "b": (180, 220),
    "c": (405, 125), "d": (460, 220),
    "x": (105, 300), "y": (210, 325), "x1": (42, 270),
    "x2": (62, 365), "x3": (132, 400), "y1": (235, 415),
    "u": (535, 300), "v": (430, 325), "u1": (598, 270),
    "u2": (578, 365), "u3": (508, 400), "v1": (405, 415),
}
EDGES = [
    ("t", "a"), ("a", "b"), ("b", "t"),
    ("t", "c"), ("c", "d"), ("d", "t"),
    ("b", "x"), ("b", "y"), ("x", "x1"), ("x", "x2"),
    ("x", "x3"), ("y", "y1"),
    ("d", "u"), ("d", "v"), ("u", "u1"), ("u", "u2"),
    ("u", "u3"), ("v", "v1"),
]


def _transform(p, turns, mirror, dx, dy):
    x, y = p
    if mirror:
        x = CANVAS - x
    for _ in range(turns):
        x, y = CANVAS - y, x
    return [x + dx, y + dy]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    colors = ["red", "green", "blue"]
    # Exact cyclic answer balance across consecutive seeds; the other colors swap.
    target_color = colors[seed % 3]
    rest = [c for c in colors if c != target_color]
    if rng.randrange(2):
        rest.reverse()
    candidates = {target_color: "t", rest[0]: "b", rest[1]: "d"}
    turns = rng.randrange(4)
    mirror = bool(rng.randrange(2))
    dx, dy = rng.randint(-12, 12), rng.randint(-12, 12)
    nodes = {k: _transform(v, turns, mirror, dx, dy) for k, v in BASE.items()}
    edges = [list(e) for e in EDGES]
    rng.shuffle(edges)
    return {"canvas": CANVAS, "nodes": nodes, "edges": edges, "candidates": candidates}


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas"])
    scale = 3
    image = Image.new("RGB", (size * scale, size * scale), BG)
    draw = ImageDraw.Draw(image)
    nodes = scene["nodes"]
    for u, v in scene["edges"]:
        p = tuple(round(q * scale) for q in nodes[u])
        r = tuple(round(q * scale) for q in nodes[v])
        draw.line([p, r], fill=LINK, width=9 * scale)
    node_colors = {node: PALETTE[color] for color, node in scene["candidates"].items()}
    radius = 13 * scale
    for name in sorted(nodes):
        x, y = (round(q * scale) for q in nodes[name])
        fill = node_colors.get(name, INK)
        draw.ellipse((x-radius, y-radius, x+radius, y+radius), fill=fill,
                     outline=(255, 255, 255), width=2 * scale)
    return image.resize((size, size), Image.Resampling.LANCZOS)


def _has_cycle(node_names, edges, removed):
    adjacency = defaultdict(list)
    for u, v in edges:
        if removed not in (u, v):
            adjacency[u].append(v)
            adjacency[v].append(u)
    seen = set()
    def visit(node, parent):
        seen.add(node)
        for nxt in adjacency[node]:
            if nxt == parent:
                continue
            if nxt in seen or visit(nxt, node):
                return True
        return False
    for node in node_names:
        if node != removed and node not in seen and visit(node, None):
            return True
    return False


def analytic_gold(scene: dict) -> str:
    winners = [color for color, node in scene["candidates"].items()
               if not _has_cycle(scene["nodes"], scene["edges"], node)]
    if len(winners) != 1:
        raise ValueError(f"expected one cycle-hitting station, found {winners}")
    return winners[0]


def _cycle_rank(scene, removed):
    vertices = [n for n in scene["nodes"] if n != removed]
    edges = [(u, v) for u, v in scene["edges"] if removed not in (u, v)]
    adj = defaultdict(list)
    for u, v in edges:
        adj[u].append(v); adj[v].append(u)
    seen = set(); components = 0
    for start in vertices:
        if start in seen:
            continue
        components += 1; stack = [start]; seen.add(start)
        while stack:
            for nxt in adj[stack.pop()]:
                if nxt not in seen:
                    seen.add(nxt); stack.append(nxt)
    return len(edges) - len(vertices) + components


def margin(scene: dict) -> float:
    ranks = sorted(_cycle_rank(scene, node) for node in scene["candidates"].values())
    return float(ranks[1] - ranks[0])


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 1 or len(set(analytic_gold(scene) for _ in [0])) != 1
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["edges"] = [[v, u] for u, v in reversed(scene["edges"])]
    twin["nodes"] = dict(reversed(list(scene["nodes"].items())))
    return [("reverse_storage_order_and_edge_orientation", twin)]
