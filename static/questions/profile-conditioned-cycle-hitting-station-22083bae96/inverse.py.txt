"""Independent pixel-only graph recovery and feedback-vertex decision."""
from __future__ import annotations

import math
from collections import deque
import numpy as np

INK = np.array([31, 41, 55])
LINK = np.array([100, 116, 139])
COLORS = {
    "red": np.array([220, 55, 65]),
    "green": np.array([24, 158, 105]),
    "blue": np.array([45, 108, 220]),
}


def _components(mask, minimum=20):
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        q = [(int(y), int(x))]; seen[y, x] = True; pts = []
        while q:
            cy, cx = q.pop(); pts.append((cy, cx))
            for ny, nx in ((cy-1,cx),(cy+1,cx),(cy,cx-1),(cy,cx+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny,nx] and not seen[ny,nx]:
                    seen[ny,nx] = True; q.append((ny,nx))
        if len(pts) >= minimum:
            out.append(pts)
    return out


def _near(arr, color, tolerance=28):
    return np.max(np.abs(arr.astype(int) - color.astype(int)), axis=2) <= tolerance


def _center(component):
    ys = [p[0] for p in component]; xs = [p[1] for p in component]
    return (sum(xs)/len(xs), sum(ys)/len(ys))


def _link_present(arr, p, q):
    x1, y1 = p; x2, y2 = q
    dist = math.hypot(x2-x1, y2-y1)
    if dist > 145 or dist < 35:
        return False
    good = total = 0
    # Ignore station disks at both ends, sample the core of the thick gray link.
    for i in range(18, max(19, int(dist)-17)):
        t = i / dist
        x = int(round(x1 + t*(x2-x1))); y = int(round(y1 + t*(y2-y1)))
        patch = arr[max(0,y-2):y+3, max(0,x-2):x+3]
        total += 1
        if patch.size and np.any(np.max(np.abs(patch.astype(int)-LINK), axis=2) <= 32):
            good += 1
    return total > 10 and good / total > 0.88


def _has_cycle(n, edges, removed):
    adj = [[] for _ in range(n)]
    for u, v in edges:
        if removed not in (u, v):
            adj[u].append(v); adj[v].append(u)
    seen = set()
    def visit(u, parent):
        seen.add(u)
        for v in adj[u]:
            if v == parent: continue
            if v in seen or visit(v, u): return True
        return False
    return any(i != removed and i not in seen and visit(i, -1) for i in range(n))


def decision_from_image(image) -> str:
    arr = np.asarray(image.convert("RGB"))
    nodes = []
    candidate_index = {}
    for comp in _components(_near(arr, INK), 80):
        nodes.append(_center(comp))
    for name, rgb in COLORS.items():
        comps = _components(_near(arr, rgb), 80)
        if len(comps) != 1:
            raise ValueError(f"expected one {name} station")
        candidate_index[name] = len(nodes)
        nodes.append(_center(comps[0]))
    if len(nodes) != 17:
        raise ValueError(f"expected 17 visible stations, found {len(nodes)}")
    edges = []
    for i in range(len(nodes)):
        for j in range(i+1, len(nodes)):
            if _link_present(arr, nodes[i], nodes[j]):
                edges.append((i,j))
    winners = [name for name, idx in candidate_index.items()
               if not _has_cycle(len(nodes), edges, idx)]
    if len(winners) != 1:
        raise ValueError(f"pixel graph has non-unique answer {winners}; edges={len(edges)}")
    return winners[0]
