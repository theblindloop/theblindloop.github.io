"""Deterministic compact district-map renderer."""

from __future__ import annotations

import random
from collections import deque
from copy import deepcopy

from PIL import Image, ImageDraw

CANVAS = 384
CELL = 24
GRID_N = 7
ORIGIN = (CANVAS - CELL * GRID_N) // 2

# Both partitions have 49 cells, six connected districts, eleven yellow cells,
# and the same multiset of district sizes.  They differ only in whether vertex
# zero is an articulation of the region-contact graph.
YES_TEMPLATE = (
    "5555000",
    "3555500",
    "3354500",
    "2334400",
    "2244100",
    "2222111",
    "2222111",
)
NO_TEMPLATE = (
    "5555333",
    "5555333",
    "5552234",
    "0002244",
    "1000244",
    "1100044",
    "1100444",
)

YELLOW = (244, 190, 48)
OTHER_COLORS = [
    (45, 134, 183),
    (221, 92, 76),
    (79, 164, 108),
    (143, 99, 184),
    (42, 169, 166),
]


def _to_grid(rows: tuple[str, ...]) -> list[list[int]]:
    return [[int(value) for value in row] for row in rows]


def _adjacency(grid: list[list[int]]) -> dict[int, set[int]]:
    labels = {value for row in grid for value in row}
    graph = {value: set() for value in labels}
    for y, row in enumerate(grid):
        for x, value in enumerate(row):
            for dy, dx in ((1, 0), (0, 1)):
                yy, xx = y + dy, x + dx
                if yy < len(grid) and xx < len(row):
                    other = grid[yy][xx]
                    if other != value:
                        graph[value].add(other)
                        graph[other].add(value)
    return graph


def _remaining_connected(grid: list[list[int]]) -> bool:
    graph = _adjacency(grid)
    remaining = sorted(set(graph) - {0})
    seen: set[int] = set()
    queue = deque(remaining[:1])
    while queue:
        node = queue.popleft()
        if node in seen:
            continue
        seen.add(node)
        queue.extend(graph[node] - {0} - seen)
    return seen == set(remaining)


def _transform(grid: list[list[int]], code: int) -> list[list[int]]:
    value = [row[:] for row in grid]
    if code >= 4:
        value = [list(reversed(row)) for row in value]
    for _ in range(code % 4):
        value = [list(row) for row in zip(*value[::-1])]
    return value


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    desired_yes = (int(seed) % 2) == 0
    grid = _to_grid(YES_TEMPLATE if desired_yes else NO_TEMPLATE)

    # Relabel all non-yellow districts and independently permute their colors.
    labels = [1, 2, 3, 4, 5]
    shuffled = labels[:]
    rng.shuffle(shuffled)
    mapping = {0: 0, **dict(zip(labels, shuffled))}
    grid = [[mapping[value] for value in row] for row in grid]
    palette_tail = OTHER_COLORS[:]
    rng.shuffle(palette_tail)
    return {
        "grid": grid,
        "palette": [list(YELLOW), *[list(color) for color in palette_tail]],
        "transform": rng.randrange(8),
        "render_alias": rng.randrange(1_000_000),
    }


def render(scene: dict) -> Image.Image:
    # Accessing render_alias is intentional: it is a pixel-neutral encoding alias.
    _ = scene.get("render_alias")
    grid = _transform(scene["grid"], int(scene["transform"]))
    palette = [tuple(color) for color in scene["palette"]]
    image = Image.new("RGB", (CANVAS, CANVAS), (250, 250, 247))
    draw = ImageDraw.Draw(image)
    for y, row in enumerate(grid):
        for x, label in enumerate(row):
            x0, y0 = ORIGIN + x * CELL, ORIGIN + y * CELL
            draw.rectangle((x0, y0, x0 + CELL - 1, y0 + CELL - 1), fill=palette[label])
    # A dark frame makes the map extent explicit without separating districts.
    draw.rectangle(
        (ORIGIN - 4, ORIGIN - 4, ORIGIN + GRID_N * CELL + 3, ORIGIN + GRID_N * CELL + 3),
        outline=(35, 39, 45),
        width=4,
    )
    return image


def analytic_gold(scene: dict) -> str:
    return "yes" if _remaining_connected(scene["grid"]) else "no"


def margin(scene: dict) -> float:
    graph = _adjacency(scene["grid"])
    remaining = set(graph) - {0}
    seen: set[int] = set()
    component_count = 0
    for start in sorted(remaining):
        if start in seen:
            continue
        component_count += 1
        queue = deque([start])
        while queue:
            node = queue.popleft()
            if node in seen:
                continue
            seen.add(node)
            queue.extend((graph[node] & remaining) - seen)
    if component_count > 1:
        return float(component_count - 1)
    edge_count = sum(len(graph[node] & remaining) for node in remaining) // 2
    return float(max(1, edge_count - (len(remaining) - 1) + 1))


def is_quarantined(scene: dict) -> bool:
    grid = scene.get("grid")
    if not isinstance(grid, list) or len(grid) != GRID_N or any(len(row) != GRID_N for row in grid):
        return True
    if {value for row in grid for value in row} != set(range(6)):
        return True
    for label in range(6):
        cells = {(y, x) for y, row in enumerate(grid) for x, value in enumerate(row) if value == label}
        if len(cells) < 5:
            return True
        seen: set[tuple[int, int]] = set()
        queue = deque(cells and [next(iter(cells))])
        while queue:
            cell = queue.popleft()
            if cell in seen:
                continue
            seen.add(cell)
            y, x = cell
            queue.extend(
                (y + dy, x + dx)
                for dy, dx in ((1, 0), (-1, 0), (0, 1), (0, -1))
                if (y + dy, x + dx) in cells and (y + dy, x + dx) not in seen
            )
        if seen != cells:
            return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["render_alias"] = int(twin["render_alias"]) + 1
    return [("pixel_neutral_alias", twin)]
