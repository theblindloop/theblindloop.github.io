"""Independent pixel-only district adjacency oracle."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PROTOTYPES = np.array(
    [
        (244, 190, 48),
        (45, 134, 183),
        (221, 92, 76),
        (79, 164, 108),
        (143, 99, 184),
        (42, 169, 166),
    ],
    dtype=np.int16,
)


def _one_component(mask: np.ndarray) -> bool:
    points = np.argwhere(mask)
    if len(points) == 0:
        return False
    start = tuple(int(v) for v in points[0])
    seen = {start}
    queue = deque([start])
    height, width = mask.shape
    while queue:
        y, x = queue.popleft()
        for dy, dx in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            yy, xx = y + dy, x + dx
            point = (yy, xx)
            if 0 <= yy < height and 0 <= xx < width and mask[yy, xx] and point not in seen:
                seen.add(point)
                queue.append(point)
    return len(seen) == int(mask.sum())


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int32)
    distance = ((pixels[:, :, None, :] - PROTOTYPES[None, None, :, :]) ** 2).sum(axis=3)
    nearest = distance.argmin(axis=2)
    colored = distance.min(axis=2) <= 12 * 12
    if int(colored.sum()) < 20_000:
        return "abstain"

    # The dark frame visibly declares one completely filled square map.  Missing
    # interior color is therefore corruption/ambiguity, not a smaller district.
    ys, xs = np.where(colored)
    if len(xs) == 0:
        return "abstain"
    box_height = int(ys.max() - ys.min() + 1)
    box_width = int(xs.max() - xs.min() + 1)
    box_area = box_height * box_width
    if abs(box_height - box_width) > 2 or int(colored.sum()) < int(0.995 * box_area):
        return "abstain"

    masks = [(nearest == index) & colored for index in range(6)]
    # The region integrity check is part of the visible rule: each color is one district.
    if any(int(mask.sum()) < 2_600 or not _one_component(mask) for mask in masks):
        return "abstain"

    graph = {index: set() for index in range(6)}
    labels = np.full(nearest.shape, -1, dtype=np.int8)
    labels[colored] = nearest[colored].astype(np.int8)
    for left, right in ((labels[:, :-1], labels[:, 1:]), (labels[:-1, :], labels[1:, :])):
        valid = (left >= 0) & (right >= 0) & (left != right)
        for first, second in np.unique(np.stack((left[valid], right[valid]), axis=1), axis=0):
            graph[int(first)].add(int(second))
            graph[int(second)].add(int(first))

    remaining = set(range(1, 6))
    seen: set[int] = set()
    queue = deque([1])
    while queue:
        node = queue.popleft()
        if node in seen:
            continue
        seen.add(node)
        queue.extend((graph[node] & remaining) - seen)
    return "yes" if seen == remaining else "no"
