"""Independent image-only inverse for the staircase shear world."""

from __future__ import annotations

from PIL import Image

N = 12
NAMES = ("arrow", "cup", "bolt", "key")
CORES = {
    "arrow": ("..##....", ".####...", "######..", "..##....", ".####...", ".####...", "..##....", "########"),
    "cup": ("##....##", "##....##", "##....##", "##....##", "##....##", "##....##", ".######.", "...##..."),
    "bolt": ("...#####", "...####.", "..####..", ".#####..", "#####...", ".####...", "..###...", "..##...."),
    "key": ("..####..", ".######.", ".##..##.", ".######.", "..####..", "...##...", "...##...", "...####."),
}


def _reference(name: str) -> list[list[int]]:
    result = [[0] * N for _ in range(N)]
    for r, row in enumerate(CORES[name], 2):
        for c, ch in enumerate(row, 2):
            result[r][c] = int(ch == "#")
    return result


def _cyan(pixel: tuple[int, int, int]) -> bool:
    r, g, b = pixel
    return b > 130 and g > 115 and r < 80 and b + g > 2 * r + 180


def _extract(image: Image.Image) -> list[list[int]]:
    rgb = image.convert("RGB")
    pts = [(x, y) for y in range(rgb.height) for x in range(rgb.width) if _cyan(rgb.getpixel((x, y)))]
    if len(pts) < 100:
        raise ValueError("cyan frame missing")
    left, right = min(x for x, _ in pts), max(x for x, _ in pts)
    top, bottom = min(y for _, y in pts), max(y for _, y in pts)
    # Frame extends seven pixels past the cell field on each side.
    x0, x1 = left + 7, right - 6
    y0, y1 = top + 7, bottom - 6
    if min(x1 - x0, y1 - y0) < 200 or abs((x1 - x0) - (y1 - y0)) > 3:
        raise ValueError("ambiguous frame")
    grid = []
    for r in range(N):
        row = []
        for c in range(N):
            cx = round(x0 + (c + 0.5) * (x1 - x0) / N)
            cy = round(y0 + (r + 0.5) * (y1 - y0) / N)
            samples = [rgb.getpixel((cx + dx, cy + dy)) for dy in (-2, 0, 2) for dx in (-2, 0, 2)]
            luminance = sum(3 * p[0] + 4 * p[1] + p[2] for p in samples) / (8 * len(samples))
            row.append(int(luminance < 120))
        grid.append(row)
    return grid


def decision_from_image(image: Image.Image) -> str:
    encoded = _extract(image)
    fixed = [[row[(c + r) % N] for c in range(N)] for r, row in enumerate(encoded)]
    scores = sorted(
        (sum(a != b for ra, rb in zip(fixed, _reference(name)) for a, b in zip(ra, rb)), name)
        for name in NAMES
    )
    if scores[1][0] - scores[0][0] < 8:
        return "abstain"
    return scores[0][1]
