"""Renderer for a visibly framed, ordinal-coupled staircase shear."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

SYMBOLS = ("arrow", "cup", "bolt", "key")
N = 12

# Eight-by-eight equal-ink glyphs embedded in a twelve-by-twelve field.  Each has
# exactly 32 dark cells, so label identity cannot be read from total ink.
_CORE = {
    "arrow": (
        "..##....", ".####...", "######..", "..##....",
        ".####...", ".####...", "..##....", "########",
    ),
    "cup": (
        "##....##", "##....##", "##....##", "##....##",
        "##....##", "##....##", ".######.", "...##...",
    ),
    "bolt": (
        "...#####", "...####.", "..####..", ".#####..",
        "#####...", ".####...", "..###...", "..##....",
    ),
    "key": (
        "..####..", ".######.", ".##..##.", ".######.",
        "..####..", "...##...", "...##...", "...####.",
    ),
}


def template(name: str) -> list[list[int]]:
    core = _CORE[name]
    out = [[0] * N for _ in range(N)]
    for r, row in enumerate(core, 2):
        for c, value in enumerate(row, 2):
            out[r][c] = int(value == "#")
    return out


def _shear_encode(grid: list[list[int]]) -> list[list[int]]:
    # Solver shifts row r left by r.  Encoding therefore shifts it right by r.
    return [[row[(c - r) % N] for c in range(N)] for r, row in enumerate(grid)]


def _deshear(grid: list[list[int]]) -> list[list[int]]:
    return [[row[(c + r) % N] for c in range(N)] for r, row in enumerate(grid)]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    symbol = SYMBOLS[int(seed) % len(SYMBOLS)]
    cell_size = rng.choice((22, 23, 24, 25))
    span = N * cell_size
    origin_x = rng.randint(70, 480 - span - 70)
    origin_y = rng.randint(70, 480 - span - 70)
    palettes = (
        ((25, 28, 34), (0, 170, 205), (250, 248, 240)),
        ((48, 30, 62), (0, 155, 190), (248, 246, 252)),
        ((18, 48, 44), (10, 175, 200), (250, 250, 242)),
    )
    dark, frame, background = palettes[rng.randrange(len(palettes))]
    return {
        "symbol": symbol,
        "grid": _shear_encode(template(symbol)),
        "origin_x": origin_x,
        "origin_y": origin_y,
        "cell_size": cell_size,
        "dark_color": list(dark),
        "frame_color": list(frame),
        "background": list(background),
    }


def render(scene: dict) -> Image.Image:
    _ = scene["symbol"]  # visibly determined redundant name; gold audit is explicit.
    grid = scene["grid"]
    ox, oy = int(scene["origin_x"]), int(scene["origin_y"])
    cell = int(scene["cell_size"])
    dark = tuple(scene["dark_color"])
    cyan = tuple(scene["frame_color"])
    background = tuple(scene["background"])
    image = Image.new("RGB", (480, 480), background)
    draw = ImageDraw.Draw(image)
    span = N * cell
    # Thick, saturated frame is the oracle's visible fiducial.
    draw.rectangle((ox - 7, oy - 7, ox + span + 6, oy + span + 6), outline=cyan, width=5)
    gridline = tuple((3 * a + b) // 4 for a, b in zip(background, cyan))
    for k in range(N + 1):
        x, y = ox + k * cell, oy + k * cell
        draw.line((x, oy, x, oy + span), fill=gridline, width=1)
        draw.line((ox, y, ox + span, y), fill=gridline, width=1)
    inset = max(2, cell // 10)
    for r in range(N):
        for c in range(N):
            if grid[r][c]:
                x0, y0 = ox + c * cell + inset, oy + r * cell + inset
                x1, y1 = ox + (c + 1) * cell - inset - 1, oy + (r + 1) * cell - inset - 1
                draw.rectangle((x0, y0, x1, y1), fill=dark)
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def _scores(grid: list[list[int]]) -> list[tuple[int, str]]:
    fixed = _deshear(grid)
    return sorted(
        (sum(a != b for row_a, row_b in zip(fixed, template(name)) for a, b in zip(row_a, row_b)), name)
        for name in SYMBOLS
    )


def margin(scene: dict) -> float:
    scores = _scores(scene["grid"])
    return float(scores[1][0] - scores[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
