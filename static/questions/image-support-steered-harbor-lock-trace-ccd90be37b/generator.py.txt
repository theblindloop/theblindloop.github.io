"""Deterministic scene sampler and Pillow renderer."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 768, 320
BACKGROUND = (249, 247, 240)
DARK = (35, 45, 58)
NAVY = (25, 60, 95)
GOLD = (230, 174, 45)
PALE_GOLD = (255, 239, 184)
PALE_BLUE = (222, 237, 242)
PALETTES = (
    ((24, 174, 205), (235, 91, 96)),
    ((38, 183, 155), (240, 119, 62)),
)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{name}", size)


def _levels(scene: dict) -> list[int]:
    value = 0
    values = []
    for operation in scene["operations"]:
        value += int(operation)
        values.append(value)
    return values


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = "ABC"[int(seed) % 3]
    for _ in range(20000):
        operations = [1] * 6 + [-1] * 6
        rng.shuffle(operations)
        levels = []
        value = 0
        valid = True
        for operation in operations:
            value += operation
            levels.append(value)
            if value < 0:
                valid = False
        if not valid:
            continue
        buoy_indices = [rng.choice((3, 4)), rng.choice((6, 7, 8)), rng.choice((9, 10, 11))]
        buoy_levels = [levels[index - 1] for index in buoy_indices]
        order = sorted(range(3), key=lambda index: buoy_levels[index], reverse=True)
        if buoy_levels[order[0]] == buoy_levels[order[1]] or "ABC"[order[0]] != target:
            continue
        return {
            "operations": operations,
            "buoy_indices": buoy_indices,
            "route_variant": rng.randrange(4),
            "palette_variant": rng.randrange(2),
        }
    raise RuntimeError("could not sample a uniquely answerable scene")


def analytic_gold(scene: dict) -> str:
    operations = scene["operations"]
    buoy_indices = scene["buoy_indices"]
    levels = []
    value = 0
    for operation in operations:
        value += int(operation)
        levels.append(value)
    buoy_levels = [levels[int(index) - 1] for index in buoy_indices]
    best = max(range(3), key=lambda index: buoy_levels[index])
    return "ABC"[best]


def margin(scene: dict) -> float:
    levels = _levels(scene)
    values = sorted((levels[int(index) - 1] for index in scene["buoy_indices"]), reverse=True)
    return float(values[0] - values[1])


def is_quarantined(scene: dict) -> bool:
    levels = _levels(scene)
    values = [levels[int(index) - 1] for index in scene["buoy_indices"]]
    return len(set(values)) < 3 or margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["route_variant"] = int(scene["route_variant"]) + 2
    return [("route_variant_plus_two", twin)]


def _route_points(scene: dict) -> list[tuple[int, int]]:
    # Parity is visible; adding two is an intentional latent encoding alias.
    if int(scene["route_variant"]) % 2 == 0:
        ys = (180, 178, 181, 179, 182, 180, 178, 181, 179, 182, 180, 178, 181, 179)
    else:
        ys = (179, 182, 180, 178, 181, 179, 182, 180, 178, 181, 179, 182, 180, 178)
    xs = [22] + [72 + 52 * index for index in range(12)] + [746]
    return list(zip(xs, ys, strict=True))


def render(scene: dict) -> Image.Image:
    operations = scene["operations"]
    buoy_indices = scene["buoy_indices"]
    route_variant = scene["route_variant"]
    palette_variant = scene["palette_variant"]
    if len(operations) != 12 or len(buoy_indices) != 3:
        raise ValueError("scene must contain twelve locks and three buoys")
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    raise_color, lower_color = PALETTES[int(palette_variant) % len(PALETTES)]
    points = _route_points({"route_variant": route_variant})
    operation_points = points[1:13]

    draw.rounded_rectangle((18, 18, 750, 93), radius=18, fill=PALE_BLUE, outline=NAVY, width=2)
    draw.text((38, 30), "HARBOR LOCK TRACE", font=_font(23, True), fill=NAVY)
    draw.text((38, 62), "CYAN +1     CORAL -1     Compare the level at buoys A, B, and C", font=_font(16), fill=NAVY)

    # Event gates and buoy rings are laid beneath the uninterrupted channel centerline.
    for (x, y), operation in zip(operation_points, operations, strict=True):
        color = raise_color if int(operation) == 1 else lower_color
        draw.rounded_rectangle((x - 8, y - 19, x + 8, y + 19), radius=5, fill=color)
    for label, index in zip("ABC", buoy_indices, strict=True):
        left = operation_points[int(index) - 1]
        right = operation_points[int(index)]
        x = (left[0] + right[0]) // 2
        y = (left[1] + right[1]) // 2
        draw.ellipse((x - 13, y - 13, x + 13, y + 13), fill=PALE_GOLD, outline=GOLD, width=7)
        box = draw.textbbox((0, 0), label, font=_font(22, True))
        draw.text((x - (box[2] - box[0]) / 2, y - 55), label, font=_font(22, True), fill=NAVY)

    draw.line(points, fill=DARK, width=7, joint="curve")
    sx, sy = points[0]
    draw.ellipse((sx - 12, sy - 12, sx + 12, sy + 12), fill=PALE_GOLD, outline=GOLD, width=5)
    draw.line((sx - 5, sy, sx + 5, sy), fill=DARK, width=4)
    ex, ey = points[-1]
    draw.polygon(((ex, ey), (ex - 22, ey - 13), (ex - 22, ey + 13)), fill=DARK)
    draw.text((18, 217), "START AT 0", font=_font(16, True), fill=NAVY)

    draw.rounded_rectangle((154, 259, 614, 298), radius=12, fill=(255, 255, 255), outline=(170, 183, 190), width=2)
    draw.text((182, 269), "Which buoy has the greatest running level?", font=_font(17, True), fill=NAVY)
    return image
