"""Independent RGB-only inverse for Harbor Lock Trace."""

from __future__ import annotations

import numpy as np
from PIL import Image

DARK = np.array((35, 45, 58), dtype=np.uint8)
GOLD = np.array((230, 174, 45), dtype=np.uint8)
RAISE_COLORS = (np.array((24, 174, 205), dtype=np.uint8), np.array((38, 183, 155), dtype=np.uint8))
LOWER_COLORS = (np.array((235, 91, 96), dtype=np.uint8), np.array((240, 119, 62), dtype=np.uint8))


def _mask_for(array: np.ndarray, colors: tuple[np.ndarray, ...]) -> np.ndarray:
    result = np.zeros(array.shape[:2], dtype=bool)
    for color in colors:
        result |= np.all(array == color, axis=2)
    return result


def _column_groups(mask: np.ndarray, minimum_pixels: int) -> list[tuple[int, int, int]]:
    counts = mask.sum(axis=0)
    columns = np.flatnonzero(counts >= minimum_pixels)
    if not len(columns):
        return []
    groups = []
    start = previous = int(columns[0])
    for value in columns[1:]:
        value = int(value)
        if value > previous + 1:
            groups.append((start, previous, int(counts[start : previous + 1].sum())))
            start = value
        previous = value
    groups.append((start, previous, int(counts[start : previous + 1].sum())))
    return groups


def _centers(mask: np.ndarray, minimum_pixels: int, minimum_ink: int) -> list[float]:
    return [
        (left + right) / 2.0
        for left, right, ink in _column_groups(mask, minimum_pixels)
        if ink >= minimum_ink
    ]


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    height, width = array.shape[:2]
    if min(height, width) < 100:
        raise ValueError("raster is too small")

    # The channel is the only exact-dark, nearly full-width continuous stroke.
    dark = np.all(array == DARK, axis=2)
    occupied = np.flatnonzero(dark.any(axis=0))
    if not len(occupied) or (int(occupied[-1]) - int(occupied[0])) / width < 0.90:
        raise ValueError("the connected channel is missing or truncated")
    internal = dark[:, int(occupied[0]) : int(occupied[-1]) + 1].any(axis=0)
    missing = np.flatnonzero(~internal)
    if len(missing):
        longest = 1
        run = 1
        for current, previous in zip(missing[1:], missing[:-1], strict=True):
            run = run + 1 if int(current) == int(previous) + 1 else 1
            longest = max(longest, run)
        if longest > max(5, width // 100):
            raise ValueError("the connected channel has a destructive gap")

    raise_mask = _mask_for(array, RAISE_COLORS)
    lower_mask = _mask_for(array, LOWER_COLORS)
    raise_centers = _centers(raise_mask, minimum_pixels=8, minimum_ink=120)
    lower_centers = _centers(lower_mask, minimum_pixels=8, minimum_ink=120)
    operations = sorted([(x, 1) for x in raise_centers] + [(x, -1) for x in lower_centers])
    if len(operations) != 12:
        raise ValueError("exactly twelve visible lock gates are required")
    if any(operations[index + 1][0] - operations[index][0] < width / 30 for index in range(11)):
        raise ValueError("lock order is unresolved")

    gold_mask = np.all(array == GOLD, axis=2)
    gold_centers = _centers(gold_mask, minimum_pixels=2, minimum_ink=25)
    # Four outlined markers are visible: START followed by A, B, and C.
    if len(gold_centers) != 4:
        raise ValueError("START and three buoy markers are required")
    buoy_centers = sorted(gold_centers)[1:]
    levels = []
    for buoy_x in buoy_centers:
        level = sum(operation for x, operation in operations if x < buoy_x)
        levels.append(level)
    order = sorted(range(3), key=lambda index: levels[index], reverse=True)
    if levels[order[0]] == levels[order[1]]:
        raise ValueError("buoy maximum is ambiguous")
    return "ABC"[order[0]]
