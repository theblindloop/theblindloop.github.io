"""Deterministic renderer for correlated bindings across ambiguous bijections."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw, ImageFont

N = 8
WIDTH = 960
HEIGHT = 720
GRID_X = 346
GRID_Y = 142
CELL = 58

BG = (234, 239, 242)
PAPER = (255, 254, 250)
INK = (33, 42, 52)
GRID = (160, 171, 181)
DOT = (28, 112, 174)
DOT_HIGHLIGHT = (219, 239, 250)
BLOCKED = (218, 223, 226)
BRACE = (193, 72, 73)
BADGE = (250, 197, 72)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    filename = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{filename}", size)


def _perfect_matchings(allowed: list[list[bool]]) -> list[tuple[int, ...]]:
    return [
        assignment
        for assignment in itertools.permutations(range(N))
        if all(bool(allowed[row][assignment[row]]) for row in range(N))
    ]


def _pair_profiles(allowed: list[list[bool]]) -> tuple[list[tuple[int, ...]], list[set[tuple[int, int]]], list[set[int]]]:
    matches = _perfect_matchings(allowed)
    if len(matches) < 2:
        raise ValueError("matrix must admit several complete assignments")
    bundles: list[set[tuple[int, int]]] = []
    domains = [{assignment[row] for assignment in matches} for row in range(N)]
    for pair_index in range(4):
        first = 2 * pair_index
        second = first + 1
        bundles.append({tuple(sorted((assignment[first], assignment[second]))) for assignment in matches})
    return matches, bundles, domains


def _locked_pair(allowed: list[list[bool]]) -> int:
    _, bundles, domains = _pair_profiles(allowed)
    if any(len(domain) < 2 for domain in domains):
        raise ValueError("every individual row must change partners")
    locked = [index for index, values in enumerate(bundles) if len(values) == 1]
    if len(locked) != 1:
        raise ValueError("matrix must have exactly one locked row pair")
    pair_index = locked[0]
    first = 2 * pair_index
    locked_columns = next(iter(bundles[pair_index]))
    if set(domains[first]) != set(locked_columns) or set(domains[first + 1]) != set(locked_columns):
        raise ValueError("the locked rows must visibly swap their two columns")
    return pair_index


def sample_scene(seed: int) -> dict:
    """Build three matching blocks with only globally infeasible forward decoys.

    The middle two-row block is ambiguous internally but cannot exchange a
    column with either outer block.  Its two rows therefore preserve a joint
    two-column image while swapping individually.  The adjacent candidate
    pairs in the outer blocks all change their joint images.
    """

    rng = random.Random(seed ^ 0x51C0A7E2)
    base = [[False] * N for _ in range(N)]

    # Outer 3x3 blocks alternate between a complete relation and a cycle-like
    # relation. Both have several matchings and no invariant individual row.
    dense_left = bool((seed >> 1) & 1)
    dense_right = bool((seed >> 2) & 1)
    for row in range(3):
        for column in range(3):
            if dense_left or row != column:
                base[row][column] = True
    for row in range(5, 8):
        for column in range(5, 8):
            if dense_right or (row - 5) != (column - 5):
                base[row][column] = True

    # The two middle rows can swap the two middle columns.
    for row in (3, 4):
        for column in (3, 4):
            base[row][column] = True

    # Forward decoys are locally plausible but occur in no complete matching:
    # the first three rows must fill the first three columns, then the middle
    # rows must fill the middle columns. Vary them without changing the gold.
    offset = rng.randrange(6)
    for row in range(3):
        needed = 4 - sum(base[row])
        later = [3 + ((row + offset + step) % 5) for step in range(5)]
        for column in later[:needed]:
            base[row][column] = True
    for row in (3, 4):
        picks = [5 + ((row + offset + step) % 3) for step in range(3)]
        for column in picks[:2]:
            base[row][column] = True

    target_slot = (3 * seed + 1) % 4
    target_rows = [3, 4]
    rng.shuffle(target_rows)
    remaining_pairs = [[0, 1], [5, 6], [2, 7]]
    rng.shuffle(remaining_pairs)
    for pair in remaining_pairs:
        rng.shuffle(pair)
    visual_pairs: list[list[int]] = []
    remainder_index = 0
    for slot in range(4):
        if slot == target_slot:
            visual_pairs.append(target_rows)
        else:
            visual_pairs.append(remaining_pairs[remainder_index])
            remainder_index += 1
    row_order = [row for pair in visual_pairs for row in pair]
    column_order = list(range(N))
    rng.shuffle(column_order)
    allowed = [
        [bool(base[row_order[row]][column_order[column]]) for column in range(N)]
        for row in range(N)
    ]
    scene = {"width": WIDTH, "height": HEIGHT, "allowed": allowed}
    if _locked_pair(allowed) != target_slot:
        raise RuntimeError("construction did not preserve the intended locked pair")
    return scene


def _draw_brace(draw: ImageDraw.ImageDraw, pair_index: int) -> None:
    top = GRID_Y + pair_index * 2 * CELL + 8
    bottom = top + 2 * CELL - 16
    x_right = GRID_X - 47
    x_left = x_right - 25
    middle = (top + bottom) // 2
    draw.line((x_right, top, x_left, top, x_left, middle - 9, x_right - 9, middle,
               x_left, middle + 9, x_left, bottom, x_right, bottom), fill=BRACE, width=5, joint="curve")
    badge_x = x_left - 42
    draw.ellipse((badge_x - 21, middle - 21, badge_x + 21, middle + 21), fill=BADGE, outline=INK, width=2)
    draw.text((badge_x, middle), chr(ord("A") + pair_index), font=_font(22, True), fill=INK, anchor="mm")


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    allowed = scene["allowed"]
    image = Image.new("RGB", (width, height), BG)
    draw = ImageDraw.Draw(image)

    draw.rounded_rectangle((42, 20, width - 42, height - 22), radius=24, fill=PAPER, outline=INK, width=3)
    draw.text((width // 2, 49), "WHICH PAIR KEEPS THE SAME TWO COLUMNS?", font=_font(25, True), fill=INK, anchor="mm")
    draw.text((width // 2, 84), "BLUE = ALLOWED   •   USE EVERY COLUMN ONCE", font=_font(18, True), fill=(65, 77, 88), anchor="mm")
    for column in range(N):
        cx = GRID_X + column * CELL + CELL // 2
        draw.ellipse((cx - 20, GRID_Y - 42, cx + 20, GRID_Y - 2), fill=BADGE, outline=INK, width=2)
        draw.text((cx, GRID_Y - 22), str(column + 1), font=_font(21, True), fill=INK, anchor="mm")

    for pair_index in range(4):
        _draw_brace(draw, pair_index)

    for row in range(N):
        cy = GRID_Y + row * CELL + CELL // 2
        for column in range(N):
            x0 = GRID_X + column * CELL
            y0 = GRID_Y + row * CELL
            draw.rectangle((x0, y0, x0 + CELL, y0 + CELL), outline=GRID, width=2)
            cx = x0 + CELL // 2
            if bool(allowed[row][column]):
                draw.ellipse((cx - 15, cy - 15, cx + 15, cy + 15), fill=DOT, outline=INK, width=2)
                draw.ellipse((cx - 5, cy - 5, cx + 5, cy + 5), fill=DOT_HIGHLIGHT)
            else:
                draw.line((cx - 7, cy - 7, cx + 7, cy + 7), fill=BLOCKED, width=3)
                draw.line((cx - 7, cy + 7, cx + 7, cy - 7), fill=BLOCKED, width=3)

    draw.text((width // 2, 641), "Many full assignments work. Only one pair keeps its joint bundle.", font=_font(17), fill=(65, 77, 88), anchor="mm")
    draw.text((width // 2, 674), "PAIR A     PAIR B     PAIR C     PAIR D", font=_font(18, True), fill=BRACE, anchor="mm")
    return image


def analytic_gold(scene: dict) -> str:
    return f"pair-{chr(ord('A') + _locked_pair(scene['allowed']))}"


def margin(scene: dict) -> float:
    _, bundles, _ = _pair_profiles(scene["allowed"])
    locked = _locked_pair(scene["allowed"])
    return float(min(len(values) for index, values in enumerate(bundles) if index != locked) - 1)


def is_quarantined(scene: dict) -> bool:
    try:
        allowed = scene["allowed"]
        matches, bundles, domains = _pair_profiles(allowed)
        locked = _locked_pair(allowed)
        pair_rows = (2 * locked, 2 * locked + 1)
        row_degrees = [sum(bool(value) for value in row) for row in allowed]
        column_degrees = [sum(bool(allowed[row][column]) for row in range(N)) for column in range(N)]
        return not (
            len(matches) >= 8
            and all(len(domain) >= 2 for domain in domains)
            and len(bundles[locked]) == 1
            and all(row_degrees[row] >= 4 for row in pair_rows)
            and all(len(values) >= 2 for index, values in enumerate(bundles) if index != locked)
            and min(row_degrees) >= 2
            and min(column_degrees) >= 2
            and margin(scene) >= 1.0
        )
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
