"""Independent fixed-layout PNG decoder and correlated-matching solver."""

from __future__ import annotations

import itertools

from PIL import Image

N = 8
GRID_X = 346
GRID_Y = 142
CELL = 58


def _is_blue(pixel: tuple[int, int, int]) -> bool:
    red, green, blue = pixel
    return blue >= 125 and blue >= red + 45 and blue >= green + 18


def decision_from_image(image: Image.Image) -> str:
    raster = image.convert("RGB")
    if raster.size != (960, 720):
        raise ValueError("unexpected canvas size")
    pixels = raster.load()
    allowed: list[list[bool]] = []
    for row in range(N):
        cy = GRID_Y + row * CELL + CELL // 2
        decoded = []
        for column in range(N):
            cx = GRID_X + column * CELL + CELL // 2
            blue_count = sum(
                _is_blue(pixels[x, y])
                for y in range(cy - 11, cy + 12)
                for x in range(cx - 11, cx + 12)
            )
            decoded.append(blue_count > 220)
        allowed.append(decoded)

    assignments = [
        assignment
        for assignment in itertools.permutations(range(N))
        if all(allowed[row][assignment[row]] for row in range(N))
    ]
    if len(assignments) < 2:
        raise ValueError("visible matrix has fewer than two complete assignments")
    if any(len({assignment[row] for assignment in assignments}) < 2 for row in range(N)):
        raise ValueError("an individual visible row does not change partners")
    bundle_counts = []
    for pair_index in range(4):
        first = 2 * pair_index
        second = first + 1
        bundles = {
            tuple(sorted((assignment[first], assignment[second])))
            for assignment in assignments
        }
        bundle_counts.append(len(bundles))
    locked = [index for index, count in enumerate(bundle_counts) if count == 1]
    if len(locked) != 1:
        raise ValueError("visible matrix does not have exactly one locked pair")
    return f"pair-{chr(ord('A') + locked[0])}"
