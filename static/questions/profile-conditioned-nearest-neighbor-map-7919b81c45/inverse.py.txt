"""Independent PNG-only inverse arm."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

PALETTE = {
    "red": (220, 55, 55),
    "blue": (45, 105, 210),
    "green": (45, 165, 90),
    "purple": (145, 70, 190),
    "gold": (220, 165, 35),
}
POINT_COLORS = ("red", "blue", "green", "purple")


def _components(mask: np.ndarray) -> list[tuple[int, float, float, tuple[int, int, int, int]]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y, x in zip(*np.nonzero(mask & ~seen)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        coords = []
        while queue:
            yy, xx = queue.popleft()
            coords.append((yy, xx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = yy + dy, xx + dx
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        ys = [p[0] for p in coords]
        xs = [p[1] for p in coords]
        found.append((len(coords), sum(xs) / len(xs), sum(ys) / len(ys), (min(xs), min(ys), max(xs), max(ys))))
    return found


def _mask(rgb: np.ndarray, target: tuple[int, int, int], tolerance: int = 4) -> np.ndarray:
    return np.max(np.abs(rgb.astype(np.int16) - np.array(target, dtype=np.int16)), axis=2) <= tolerance


def _map(points: dict[str, tuple[float, float]]) -> dict[str, str]:
    result = {}
    for color, p in points.items():
        choices = [(math.dist(p, q), other) for other, q in points.items() if other != color]
        choices.sort()
        if len(choices) != 3 or choices[1][0] - choices[0][0] < 8:
            raise ValueError("ambiguous nearest neighbor")
        result[color] = choices[0][1]
    return result


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"))
    if rgb.shape[0] < 500 or rgb.shape[1] < 700:
        raise ValueError("unexpected canvas")
    points_by_color: dict[str, list[tuple[float, float]]] = {}
    for color in POINT_COLORS:
        comps = _components(_mask(rgb, PALETTE[color]))
        dots = [(cx, cy) for area, cx, cy, box in comps if 450 <= area <= 750 and 22 <= box[2] - box[0] <= 30 and 22 <= box[3] - box[1] <= 30]
        if len(dots) != 4:
            raise ValueError(f"expected four {color} dots, got {len(dots)}")
        points_by_color[color] = dots

    card_rows = sorted({round(y / 40) for dots in points_by_color.values() for _, y in dots})
    all_points = [(color, x, y) for color, dots in points_by_color.items() for x, y in dots]
    reference_raw = [(c, x, y) for c, x, y in all_points if y < 320]
    if len(reference_raw) != 4:
        raise ValueError("reference card not recoverable")
    reference = {c: (x, y) for c, x, y in reference_raw}

    border_boxes = {}
    for color in ("red", "blue", "gold"):
        comps = _components(_mask(rgb, PALETTE[color]))
        rectangles = [box for area, _, _, box in comps if area > 1800 and box[2] - box[0] > 240 and box[3] - box[1] > 220]
        if len(rectangles) != 1:
            raise ValueError(f"expected one {color} option border")
        border_boxes[color] = rectangles[0]

    target_map = _map(reference)
    matches = []
    for border_color, (x0, y0, x1, y1) in border_boxes.items():
        card = {c: (x, y) for c, x, y in all_points if x0 < x < x1 and y0 < y < y1}
        if len(card) != 4:
            raise ValueError("option card does not contain four colors")
        if _map(card) == target_map:
            matches.append(border_color)
    if len(matches) != 1:
        raise ValueError("pixel evidence does not identify one option")
    return matches[0]
