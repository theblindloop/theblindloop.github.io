"""Renderer and analytic arm for nearest-neighbor map correspondence."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 900, 680
COLORS = {
    "red": (220, 55, 55),
    "blue": (45, 105, 210),
    "green": (45, 165, 90),
    "purple": (145, 70, 190),
    "gold": (220, 165, 35),
}
POINT_COLORS = ("red", "blue", "green", "purple")
BORDER_COLORS = ("red", "blue", "gold")
REF_BOX = (250, 42, 650, 292)
OPTION_BOXES = ((25, 375, 295, 625), (315, 375, 585, 625), (605, 375, 875, 625))


def _dist(a: list[float], b: list[float]) -> float:
    return math.hypot(a[0] - b[0], a[1] - b[1])


def _nn_map(points: dict[str, list[float]]) -> dict[str, str]:
    result = {}
    for color in POINT_COLORS:
        others = [other for other in POINT_COLORS if other != color]
        result[color] = min(others, key=lambda other: _dist(points[color], points[other]))
    return result


def _gap(points: dict[str, list[float]]) -> float:
    gaps = []
    for color in POINT_COLORS:
        distances = sorted(_dist(points[color], points[other]) for other in POINT_COLORS if other != color)
        gaps.append(distances[1] - distances[0])
    return min(gaps)


def _random_points(rng: random.Random, width: int, height: int) -> dict[str, list[float]]:
    for _ in range(10000):
        pts = {}
        for color in POINT_COLORS:
            for _ in range(500):
                p = [rng.uniform(42, width - 42), rng.uniform(42, height - 42)]
                if all(_dist(p, q) >= 62 for q in pts.values()):
                    pts[color] = p
                    break
            if color not in pts:
                break
        if len(pts) == 4 and _gap(pts) >= 20:
            return pts
    raise RuntimeError("unable to sample separated points")


def _similarity(points: dict[str, list[float]], rng: random.Random, width: int, height: int) -> dict[str, list[float]]:
    values = list(points.values())
    cx = sum(p[0] for p in values) / 4
    cy = sum(p[1] for p in values) / 4
    angle = rng.uniform(0.55, 5.75)
    mirror = -1 if rng.random() < 0.5 else 1
    ca, sa = math.cos(angle), math.sin(angle)
    transformed = {}
    unscaled = []
    for color in POINT_COLORS:
        x = mirror * (points[color][0] - cx)
        y = points[color][1] - cy
        unscaled.append((color, x * ca - y * sa, x * sa + y * ca))
    spanx = max(v[1] for v in unscaled) - min(v[1] for v in unscaled)
    spany = max(v[2] for v in unscaled) - min(v[2] for v in unscaled)
    scale = min(rng.uniform(0.62, 0.82), (width - 90) / spanx, (height - 90) / spany)
    raw = [(color, x * scale, y * scale) for color, x, y in unscaled]
    minx, maxx = min(v[1] for v in raw), max(v[1] for v in raw)
    miny, maxy = min(v[2] for v in raw), max(v[2] for v in raw)
    room_x = width - 90 - (maxx - minx)
    room_y = height - 90 - (maxy - miny)
    ox = 45 - minx + rng.uniform(0, max(0.0, room_x))
    oy = 45 - miny + rng.uniform(0, max(0.0, room_y))
    for color, x, y in raw:
        transformed[color] = [x + ox, y + oy]
    return transformed


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    ref = _random_points(rng, 400, 250)
    ref_map = _nn_map(ref)
    match_slot = rng.randrange(3)
    cards: list[dict[str, list[float]]] = []
    used_maps = {tuple(ref_map[c] for c in POINT_COLORS)}
    for slot in range(3):
        if slot == match_slot:
            cards.append(_similarity(ref, rng, 270, 250))
            continue
        for _ in range(10000):
            pts = _random_points(rng, 270, 250)
            key = tuple(_nn_map(pts)[c] for c in POINT_COLORS)
            if key not in used_maps:
                used_maps.add(key)
                cards.append(pts)
                break
        else:
            raise RuntimeError("unable to sample unique distractor")
    border_colors = list(BORDER_COLORS)
    rng.shuffle(border_colors)
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "cards": [ref, *cards],
        "border_colors": border_colors,
    }


def _draw_card(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], border: tuple[int, int, int], points: dict[str, list[float]]) -> None:
    draw.rounded_rectangle(box, radius=14, fill=(250, 250, 247), outline=border, width=7)
    x0, y0, _, _ = box
    for color in POINT_COLORS:
        x, y = points[color]
        cx, cy = int(round(x0 + x)), int(round(y0 + y))
        draw.ellipse((cx - 14, cy - 14, cx + 14, cy + 14), fill=COLORS[color], outline=(35, 35, 35), width=2)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), (245, 244, 239))
    draw = ImageDraw.Draw(image)
    _draw_card(draw, REF_BOX, (70, 70, 70), scene["cards"][0])
    for box, border_color, points in zip(OPTION_BOXES, scene["border_colors"], scene["cards"][1:]):
        _draw_card(draw, box, COLORS[border_color], points)
    draw.text((385, 15), "REFERENCE", fill=(45, 45, 45), anchor="ma")
    draw.text((450, 335), "OPTIONS", fill=(45, 45, 45), anchor="ma")
    return image


def analytic_gold(scene: dict) -> str:
    cards = scene["cards"]
    border_colors = scene["border_colors"]
    reference = _nn_map(cards[0])
    matches = [i for i, card in enumerate(cards[1:]) if _nn_map(card) == reference]
    if len(matches) != 1:
        raise ValueError("scene has no unique matching map")
    return border_colors[matches[0]]


def margin(scene: dict) -> float:
    cards = scene["cards"]
    reference = _nn_map(cards[0])
    geometric = min(_gap(card) for card in cards)
    mismatches = [sum(_nn_map(card)[c] != reference[c] for c in POINT_COLORS) for card in cards[1:]]
    positive = [value for value in mismatches if value > 0]
    structural = 20.0 * min(positive) if positive else 0.0
    return float(min(geometric, structural))


def is_quarantined(scene: dict) -> bool:
    cards = scene["cards"]
    reference = _nn_map(cards[0])
    matches = sum(_nn_map(card) == reference for card in cards[1:])
    return matches != 1 or margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
