"""Renderer and analytic label for the colored-region adjacency hub world.

The latent scene contains only three visible colored rectangles. Their shared
side geometry forms a three-node path. The analytic answer is recomputed from
that geometry; it is not stored as a hidden role or answer field.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "region_adjacency_hub-renderer-0.1.0"
WIDTH, HEIGHT = 720, 460
SUPERSAMPLE = 4
BACKGROUND = (248, 247, 242)
FRAME = (35, 36, 42)
COLORS: dict[str, tuple[int, int, int]] = {
    "red": (211, 66, 72),
    "green": (54, 157, 93),
    "blue": (68, 105, 198),
}
CONTACT_RULE = 18.0
QUARANTINE_CLEARANCE = 10.0


def _overlap(lo1: float, hi1: float, lo2: float, hi2: float) -> float:
    return max(0.0, min(hi1, hi2) - max(lo1, lo2))


def _contact_length(first_rect: list[float], second_rect: list[float]) -> float:
    """Length of a shared axis-aligned side, excluding corner contacts."""

    ax0, ay0, ax1, ay1 = (float(value) for value in first_rect)
    bx0, by0, bx1, by1 = (float(value) for value in second_rect)
    horizontal = max(
        _overlap(ax0, ax1, bx0, bx1) if abs(ay1 - by0) < 1e-6 or abs(by1 - ay0) < 1e-6 else 0.0,
        _overlap(ax0, ax1, bx0, bx1) if abs(ay0 - by1) < 1e-6 or abs(by0 - ay1) < 1e-6 else 0.0,
    )
    vertical = max(
        _overlap(ay0, ay1, by0, by1) if abs(ax1 - bx0) < 1e-6 or abs(bx0 - ax1) < 1e-6 else 0.0,
        _overlap(ay0, ay1, by0, by1) if abs(ax0 - bx1) < 1e-6 or abs(bx1 - ax0) < 1e-6 else 0.0,
    )
    return max(horizontal, vertical)


def _contact_matrix(scene: dict[str, Any]) -> list[list[float]]:
    regions = scene["regions"]
    matrix = [[0.0 for _ in regions] for _ in regions]
    for i in range(len(regions)):
        for j in range(i + 1, len(regions)):
            value = _contact_length(regions[i]["rect"], regions[j]["rect"])
            matrix[i][j] = matrix[j][i] = value
    return matrix


def _rect(x0: float, y0: float, x1: float, y1: float) -> dict[str, Any]:
    return {"rect": [float(x0), float(y0), float(x1), float(y1)]}


def _inside_frame(regions: list[dict[str, Any]]) -> bool:
    for region in regions:
        x0, y0, x1, y1 = region["rect"]
        if x0 < 48.0 or y0 < 48.0 or x1 > WIDTH - 48.0 or y1 > HEIGHT - 48.0:
            return False
    return True


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced three-region path from one integer seed."""

    rng = random.Random(int(seed))
    directions = rng.sample(("left", "right", "top", "bottom"), 2)
    for _ in range(5000):
        cx = rng.uniform(220.0, 500.0)
        cy = rng.uniform(155.0, 305.0)
        cw = rng.uniform(112.0, 172.0)
        ch = rng.uniform(104.0, 154.0)
        hub = _rect(cx - cw / 2.0, cy - ch / 2.0, cx + cw / 2.0, cy + ch / 2.0)
        endpoints: list[dict[str, Any]] = []
        for direction in directions:
            if direction in ("left", "right"):
                shared = rng.uniform(0.30, 0.80) * ch
                offset = rng.uniform(-0.06, 0.06) * ch
                y0 = cy + offset - shared / 2.0
                y1 = cy + offset + shared / 2.0
                width = rng.uniform(76.0, 132.0)
                if direction == "left":
                    endpoints.append(_rect(hub["rect"][0] - width, y0, hub["rect"][0], y1))
                else:
                    endpoints.append(_rect(hub["rect"][2], y0, hub["rect"][2] + width, y1))
            else:
                shared = rng.uniform(0.30, 0.80) * cw
                offset = rng.uniform(-0.06, 0.06) * cw
                x0 = cx + offset - shared / 2.0
                x1 = cx + offset + shared / 2.0
                height = rng.uniform(74.0, 126.0)
                if direction == "top":
                    endpoints.append(_rect(x0, hub["rect"][1] - height, x1, hub["rect"][1]))
                else:
                    endpoints.append(_rect(x0, hub["rect"][3], x1, hub["rect"][3] + height))
        regions = [hub, *endpoints]
        if not _inside_frame(regions):
            continue
        if min(_contact_length(hub["rect"], endpoint["rect"]) for endpoint in endpoints) < 24.0:
            continue
        # The two endpoints must not share a positive-length side with one another.
        if _contact_length(endpoints[0]["rect"], endpoints[1]["rect"]) >= CONTACT_RULE:
            continue
        colors = list(COLORS)
        rng.shuffle(colors)
        for region, color in zip(regions, colors):
            region["color"] = color
        rng.shuffle(regions)
        return {"regions": regions}
    raise RuntimeError("could not sample a separated three-region path")


def _scaled_rect(rect: list[float], scale: int) -> tuple[int, int, int, int]:
    return tuple(int(round(float(value) * scale)) for value in rect)  # type: ignore[return-value]


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the public scene fields as a deterministic RGB image."""

    size = (WIDTH * SUPERSAMPLE, HEIGHT * SUPERSAMPLE)
    image = Image.new("RGB", size, BACKGROUND)
    draw = ImageDraw.Draw(image)
    for region in scene["regions"]:
        draw.rectangle(
            _scaled_rect(region["rect"], SUPERSAMPLE),
            fill=COLORS[region["color"]],
        )
    frame = 24 * SUPERSAMPLE
    draw.rectangle(
        (frame, frame, size[0] - frame - 1, size[1] - frame - 1),
        outline=FRAME,
        width=3 * SUPERSAMPLE,
    )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    """Select the unique region with two rule-sized shared sides."""

    matrix = _contact_matrix(scene)
    degrees = [sum(value >= CONTACT_RULE for value in row) for row in matrix]
    hubs = [index for index, degree in enumerate(degrees) if degree == 2]
    if len(hubs) != 1 or sorted(degrees) != [1, 1, 2]:
        raise ValueError(f"scene is not a three-region path: degrees={degrees}")
    return str(scene["regions"][hubs[0]]["color"])


def margin(scene: dict[str, Any]) -> float:
    """Smallest intended contact clearance above the visible adjacency rule."""

    matrix = _contact_matrix(scene)
    positive = [matrix[i][j] for i in range(3) for j in range(i + 1, 3) if matrix[i][j] > 0.0]
    if len(positive) != 2:
        raise ValueError("scene does not contain exactly two positive contacts")
    return float(min(positive) - CONTACT_RULE)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < QUARANTINE_CLEARANCE


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """This renderer declares no latent aliases; order is treated as public input."""

    return []
