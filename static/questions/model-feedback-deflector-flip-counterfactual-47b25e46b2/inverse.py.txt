"""Independent pixel-only inverse arm for the deflector-flip world.

Given only the final PNG, this module segments the five colored bins, the black
release dot, and every slanted bar; reads each bar's slant from the sign of its
principal axis; groups bars into rows by vertical clustering; and then replays
the drop with the single red bar's slant inverted. The returned decision is the
color name of the bin reached on that edited board.

It imports no renderer / prompts / generate / verify / analytic-gold code and
never receives the latent scene.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_BIN_PALETTE = {
    "blue": (35, 85, 205),
    "green": (25, 145, 65),
    "orange": (245, 150, 25),
    "purple": (135, 55, 190),
    "yellow": (240, 200, 30),
}
_BAR_RGB = (60, 66, 78)
_RED_RGB = (200, 40, 50)
_TOL = 60.0
_ROW_GAP = 35.0


def _near(arr: np.ndarray, rgb, tol: float = _TOL) -> np.ndarray:
    diff = arr - np.asarray(rgb, dtype=float)
    return np.sqrt((diff * diff).sum(axis=2)) < tol


def _components(mask: np.ndarray, min_pixels: int = 40) -> list:
    """Label 8-connected components of a boolean mask without SciPy."""
    height, width = mask.shape
    seen = np.zeros((height, width), dtype=bool)
    out = []
    ys, xs = np.nonzero(mask)
    for sy, sx in zip(ys.tolist(), xs.tolist()):
        if seen[sy, sx]:
            continue
        stack = [(sy, sx)]
        seen[sy, sx] = True
        pixels = []
        while stack:
            y, x = stack.pop()
            pixels.append((y, x))
            for ny in (y - 1, y, y + 1):
                if ny < 0 or ny >= height:
                    continue
                for nx in (x - 1, x, x + 1):
                    if 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        if len(pixels) >= min_pixels:
            out.append(np.asarray(pixels, dtype=float))
    return out


def _slant(pixels: np.ndarray) -> int:
    """-1 for a bar leaning like '/', +1 for a bar leaning like '\\'."""
    centred = pixels - pixels.mean(axis=0)
    cov = np.cov(centred, rowvar=False)
    eigvals, eigvecs = np.linalg.eigh(np.atleast_2d(cov))
    vec = eigvecs[:, int(np.argmax(eigvals))]
    dy, dx = float(vec[0]), float(vec[1])
    if dx < 0:
        dx, dy = -dx, -dy
    return -1 if dy < 0 else 1


def decision_from_image(image: Image.Image) -> str:
    """Return the bin color reached after the red bar is flipped."""
    arr = np.asarray(image.convert("RGB")).astype(float)

    bins = []
    for name, rgb in _BIN_PALETTE.items():
        mask = _near(arr, rgb, 40.0)
        ys, xs = np.nonzero(mask)
        if xs.size < 200:
            continue
        bins.append((float(xs.mean()), name, float(ys.min())))
    if len(bins) < 2:
        return "blue"
    bins.sort()
    bin_x = [item[0] for item in bins]
    bin_names = [item[1] for item in bins]
    lattice_bottom = min(item[2] for item in bins) - 8.0

    region = np.zeros(arr.shape[:2], dtype=bool)
    region[: int(lattice_bottom), :] = True

    ball_mask = _near(arr, (0, 0, 0), 50.0) & region
    ys, xs = np.nonzero(ball_mask)
    if xs.size < 50:
        return bin_names[0]
    ball_x = float(xs.mean())
    start = int(np.argmin([abs(ball_x - x) for x in bin_x]))

    bar_mask = (_near(arr, _BAR_RGB) | _near(arr, _RED_RGB)) & region
    pieces = _components(bar_mask)

    bars = []
    for pixels in pieces:
        cy = float(pixels[:, 0].mean())
        cx = float(pixels[:, 1].mean())
        rows = pixels[:, 0].astype(int)
        cols = pixels[:, 1].astype(int)
        mean_rgb = arr[rows, cols].mean(axis=0)
        is_red = float(np.linalg.norm(mean_rgb - np.asarray(_RED_RGB, dtype=float))) < _TOL
        col = int(np.argmin([abs(cx - x) for x in bin_x]))
        bars.append(
            {
                "y": cy,
                "col": col,
                "slant": _slant(pixels),
                "red": is_red,
                "span": (float(pixels[:, 1].min()), float(pixels[:, 1].max())),
            }
        )
    if not bars:
        return bin_names[start] if start < len(bin_names) else bin_names[0]

    bars.sort(key=lambda item: item["y"])
    bands = [[bars[0]]]
    for bar in bars[1:]:
        if bar["y"] - bands[-1][-1]["y"] > _ROW_GAP:
            bands.append([bar])
        else:
            bands[-1].append(bar)

    col = start
    for band in bands:
        for bar in band:
            if bar["col"] != col:
                continue
            delta = -bar["slant"] if bar["red"] else bar["slant"]
            col += delta
            break
        col = max(0, min(len(bin_names) - 1, col))
    return bin_names[col]
