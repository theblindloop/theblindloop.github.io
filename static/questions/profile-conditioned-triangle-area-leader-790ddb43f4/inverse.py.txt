"""Independent pixel-only inverse arm for the rendered scatter panels.

This module deliberately contains its own palette, segmentation, component,
grouping, centroid, and determinant logic.  It accepts no scene information.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

COLORS = {
    "red": (214, 55, 65),
    "blue": (48, 105, 210),
    "green": (38, 154, 92),
}
LABELS = ("A", "B", "C", "D")
COLOR_DISTANCE = 52
MIN_COMPONENT_AREA = 95


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    remaining = {(int(y), int(x)) for y, x in np.argwhere(mask)}
    found: list[tuple[float, float, int]] = []
    while remaining:
        start = remaining.pop()
        queue = deque([start])
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for neighbor in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                ny, nx = neighbor
                if 0 <= ny < height and 0 <= nx < width and neighbor in remaining:
                    remaining.remove(neighbor)
                    queue.append(neighbor)
        if len(xs) >= MIN_COMPONENT_AREA:
            found.append((float(np.mean(xs)), float(np.mean(ys)), len(xs)))
    return found


def _area(points: list[tuple[float, float]]) -> float:
    a, b, c = points
    return abs((b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0])) / 2.0


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int32)
    marks: list[tuple[float, float, str]] = []
    for name, rgb in COLORS.items():
        delta = pixels - np.asarray(rgb, dtype=np.int32)
        mask = np.sum(delta * delta, axis=2) <= COLOR_DISTANCE * COLOR_DISTANCE
        components = _components(mask)
        if len(components) != 4:
            raise ValueError(f"expected four {name} markers, found {len(components)}")
        marks.extend((x, y, name) for x, y, _ in components)

    marks.sort(key=lambda item: item[0])
    if len(marks) != 12:
        raise ValueError("expected twelve colored markers")
    panels = [marks[index:index + 3] for index in range(0, 12, 3)]
    areas: list[float] = []
    for panel in panels:
        if {mark[2] for mark in panel} != set(COLORS):
            raise ValueError("panel grouping does not contain one marker of each color")
        if max(mark[0] for mark in panel) - min(mark[0] for mark in panel) > 190:
            raise ValueError("colored markers do not form compact panel groups")
        areas.append(_area([(mark[0], mark[1]) for mark in panel]))
    ranking = sorted(range(4), key=lambda index: areas[index], reverse=True)
    if areas[ranking[0]] - areas[ranking[1]] < 80.0:
        raise ValueError("pixel measurement is ambiguous")
    return LABELS[ranking[0]]
