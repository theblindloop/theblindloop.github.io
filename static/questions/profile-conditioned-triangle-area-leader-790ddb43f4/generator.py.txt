"""Deterministic latent sampler, analytic arm, and Pillow renderer."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1020
HEIGHT = 420
SCALE = 3
PANEL_CENTERS_X = (135, 385, 635, 885)
PANEL_CENTER_Y = 225
PANEL_LABELS = ("A", "B", "C", "D")
PALETTE = {
    "red": (214, 55, 65),
    "blue": (48, 105, 210),
    "green": (38, 154, 92),
}
QUARANTINE_GAP = 400.0


def _triangle_area(points: list[dict]) -> float:
    a, b, c = points
    return abs(
        (b["x"] - a["x"]) * (c["y"] - a["y"])
        - (b["y"] - a["y"]) * (c["x"] - a["x"])
    ) / 2.0


def _make_triangle(rng: random.Random, target_area: float) -> list[dict]:
    """Construct a varied centered triangle with a controlled determinant area."""
    raw: list[tuple[float, float]] | None = None
    for _ in range(300):
        theta = rng.uniform(0.0, 2.0 * math.pi)
        length = rng.uniform(96.0, 122.0)
        height = 2.0 * target_area / length
        shear = rng.uniform(-28.0, 28.0)
        ux, uy = math.cos(theta), math.sin(theta)
        px, py = -uy, ux
        u = (length * ux, length * uy)
        v = (shear * ux + height * px, shear * uy + height * py)
        trial = [
            (-(u[0] + v[0]) / 3.0, -(u[1] + v[1]) / 3.0),
            ((2.0 * u[0] - v[0]) / 3.0, (2.0 * u[1] - v[1]) / 3.0),
            ((2.0 * v[0] - u[0]) / 3.0, (2.0 * v[1] - u[1]) / 3.0),
        ]
        if max(max(abs(x), abs(y)) for x, y in trial) <= 82.0:
            raw = trial
            break
    if raw is None:
        length = 112.0
        height = 2.0 * target_area / length
        raw = [
            (-length / 3.0, -height / 3.0),
            (2.0 * length / 3.0, -height / 3.0),
            (-length / 3.0, 2.0 * height / 3.0),
        ]

    colors = list(PALETTE)
    rng.shuffle(colors)
    return [
        {"x": round(x, 4), "y": round(y, 4), "color": color}
        for (x, y), color in zip(raw, colors)
    ]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    if int(seed) % 9 == 0:
        low = 1810.0 + rng.uniform(-45.0, 45.0)
        mid = 2690.0 + rng.uniform(-45.0, 45.0)
        runner_up = 3750.0 + rng.uniform(-35.0, 35.0)
        target_areas = [low, mid, runner_up, runner_up + 220.0]
    else:
        target_areas = [
            1800.0 + rng.uniform(-70.0, 70.0),
            2670.0 + rng.uniform(-70.0, 70.0),
            3510.0 + rng.uniform(-70.0, 70.0),
            4400.0 + rng.uniform(-70.0, 70.0),
        ]
    panels = [{"points": _make_triangle(rng, area)} for area in target_areas]
    panel_order = list(range(4))
    rng.shuffle(panel_order)
    return {
        "panels": panels,
        "panel_order": panel_order,
        "translation": [rng.randint(-6, 6), rng.randint(-4, 4)],
    }


def _visible_areas(scene: dict) -> list[float]:
    panels = scene["panels"]
    return [_triangle_area(panels[index]["points"]) for index in scene["panel_order"]]


def analytic_gold(scene: dict) -> str:
    areas = _visible_areas(scene)
    return PANEL_LABELS[max(range(4), key=lambda index: areas[index])]


def margin(scene: dict) -> float:
    areas = sorted(_visible_areas(scene), reverse=True)
    return areas[0] - areas[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    for panel in twin["panels"]:
        panel["points"].reverse()
    return [("reverse_unordered_point_records", twin)]


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size * SCALE)
    except OSError:
        return ImageFont.load_default()


def _scaled_box(box: tuple[float, float, float, float]) -> tuple[int, int, int, int]:
    return tuple(round(value * SCALE) for value in box)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), (250, 251, 253))
    draw = ImageDraw.Draw(image)
    dx, dy = scene["translation"]

    title_font = _font(19, bold=True)
    label_font = _font(21, bold=True)
    draw.text(
        (WIDTH * SCALE / 2, 18 * SCALE),
        "TRIANGLE AREA  |  EQUAL SCALE",
        fill=(42, 48, 58),
        font=title_font,
        anchor="ma",
    )

    panels = scene["panels"]
    for visible_index, canonical_index in enumerate(scene["panel_order"]):
        cx = PANEL_CENTERS_X[visible_index] + dx
        cy = PANEL_CENTER_Y + dy
        draw.rounded_rectangle(
            _scaled_box((cx - 106, 82 + dy, cx + 106, 354 + dy)),
            radius=9 * SCALE,
            fill=(255, 255, 255),
            outline=(184, 191, 201),
            width=2 * SCALE,
        )
        for offset in (-80, -40, 0, 40, 80):
            shade = (222, 226, 232) if offset else (171, 179, 189)
            width = SCALE if offset else 2 * SCALE
            draw.line(
                [(round((cx + offset) * SCALE), round((cy - 100) * SCALE)),
                 (round((cx + offset) * SCALE), round((cy + 100) * SCALE))],
                fill=shade,
                width=width,
            )
            draw.line(
                [(round((cx - 92) * SCALE), round((cy + offset) * SCALE)),
                 (round((cx + 92) * SCALE), round((cy + offset) * SCALE))],
                fill=shade,
                width=width,
            )
        draw.text(
            (cx * SCALE, 58 * SCALE),
            PANEL_LABELS[visible_index],
            fill=(36, 42, 51),
            font=label_font,
            anchor="mm",
        )

        by_color = {point["color"]: point for point in panels[canonical_index]["points"]}
        ordered = [by_color[name] for name in PALETTE]
        polygon = [
            (round((cx + point["x"]) * SCALE), round((cy + point["y"]) * SCALE))
            for point in ordered
        ]
        draw.line(polygon + [polygon[0]], fill=(112, 119, 130), width=2 * SCALE, joint="curve")
        radius = 8.5
        for point in ordered:
            x = cx + point["x"]
            y = cy + point["y"]
            draw.ellipse(
                _scaled_box((x - radius, y - radius, x + radius, y + radius)),
                fill=PALETTE[point["color"]],
                outline=(255, 255, 255),
                width=2 * SCALE,
            )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
