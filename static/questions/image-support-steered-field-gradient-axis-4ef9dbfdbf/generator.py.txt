"""Deterministic latent sampler, analytic arm, and raster renderer."""

from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image, ImageDraw

CANVAS_W = 640
CANVAS_H = 480
QUARANTINE_GAP = 18.0


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    # Consecutive seeds alternate the intended answer independently of geometry.
    horizontal_wins = int(seed) % 2 == 0
    near = int(seed) % 7 == 0
    small = rng.uniform(24.0, 39.0)
    gap = rng.uniform(8.0, 15.0) if near else rng.uniform(25.0, 43.0)
    large = small + gap
    horizontal_mag, vertical_mag = (large, small) if horizontal_wins else (small, large)
    horizontal_change = horizontal_mag * (-1.0 if rng.random() < 0.5 else 1.0)
    vertical_change = vertical_mag * (-1.0 if rng.random() < 0.5 else 1.0)
    return {
        "seed": int(seed),
        "x0": rng.randint(42, 70),
        "y0": rng.randint(52, 78),
        "width": rng.randint(505, 548),
        "height": rng.randint(322, 354),
        "base": rng.randint(128, 162),
        "horizontal_change": round(horizontal_change, 6),
        "vertical_change": round(vertical_change, 6),
        "texture_phase_x": rng.randint(0, 62),
        "texture_phase_y": rng.randint(0, 58),
    }


def render(scene: dict) -> Image.Image:
    seed = int(scene["seed"])
    x0 = int(scene["x0"])
    y0 = int(scene["y0"])
    width = int(scene["width"])
    height = int(scene["height"])
    base = float(scene["base"])
    horizontal_change = float(scene["horizontal_change"])
    vertical_change = float(scene["vertical_change"])
    phase_x = int(scene["texture_phase_x"])
    phase_y = int(scene["texture_phase_y"])

    yy, xx = np.mgrid[0:height, 0:width]
    u = xx / max(1, width - 1) - 0.5
    v = yy / max(1, height - 1) - 0.5
    # Low-amplitude deterministic texture prevents a trivial two-pixel lookup,
    # while least-squares recovery over the dense interior remains stable.
    texture = (
        2.2 * np.sin((xx + phase_x) * (2.0 * math.pi / 47.0))
        + 1.7 * np.cos((yy + phase_y) * (2.0 * math.pi / 43.0))
        + 0.55 * np.sin((xx + yy + seed % 31) * (2.0 * math.pi / 29.0))
    )
    gray = np.clip(
        np.rint(base + horizontal_change * u + vertical_change * v + texture),
        45,
        218,
    ).astype(np.uint8)
    # Neutral grayscale keeps gradient direction independent of a color label.
    field = np.stack([gray, gray, gray], axis=2)
    image = Image.new("RGB", (CANVAS_W, CANVAS_H), (249, 249, 247))
    image.paste(Image.fromarray(field, "RGB"), (x0, y0))
    draw = ImageDraw.Draw(image)
    draw.rectangle(
        (x0 - 4, y0 - 4, x0 + width + 3, y0 + height + 3),
        outline=(28, 32, 36),
        width=4,
    )
    # Small symmetric ticks make the two measurement axes explicit without
    # supplying an answer-specific cue.
    cx = x0 + width // 2
    cy = y0 + height // 2
    draw.line((x0 - 10, cy, x0 - 4, cy), fill=(28, 32, 36), width=3)
    draw.line((x0 + width + 3, cy, x0 + width + 9, cy), fill=(28, 32, 36), width=3)
    draw.line((cx, y0 - 10, cx, y0 - 4), fill=(28, 32, 36), width=3)
    draw.line((cx, y0 + height + 3, cx, y0 + height + 9), fill=(28, 32, 36), width=3)
    return image


def _changes(scene: dict) -> tuple[float, float]:
    return abs(float(scene["horizontal_change"])), abs(float(scene["vertical_change"]))


def analytic_gold(scene: dict) -> str:
    horizontal, vertical = _changes(scene)
    return "horizontal" if horizontal > vertical else "vertical"


def margin(scene: dict) -> float:
    horizontal, vertical = _changes(scene)
    return abs(horizontal - vertical)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
