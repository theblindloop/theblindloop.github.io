"""Independent image-only inverse arm for the planar brightness field."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _field_bounds(rgb: np.ndarray) -> tuple[int, int, int, int]:
    # The field and its dark frame are the sole large non-background component.
    dark = np.mean(rgb.astype(np.float64), axis=2) < 232.0
    ys, xs = np.nonzero(dark)
    if len(xs) < 5000:
        raise ValueError("measurement field not found")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    if x1 - x0 < 200 or y1 - y0 < 160:
        raise ValueError("measurement field is too small")
    return x0, y0, x1, y1


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    x0, y0, x1, y1 = _field_bounds(rgb)
    # Ignore frame and axis ticks. The regression uses only raster intensities.
    pad = 12
    crop = rgb[y0 + pad : y1 - pad + 1, x0 + pad : x1 - pad + 1]
    if crop.shape[0] < 120 or crop.shape[1] < 180:
        raise ValueError("field interior unavailable")
    lum = crop.astype(np.float64).mean(axis=2)
    # This question explicitly asks for the trend across the whole dense field.
    # Background-valued holes mean that evidence was erased, not that the hole
    # is an intensity sample. Abstain rather than impute missing raster support.
    if float(np.mean(lum > 235.0)) > 0.001:
        raise ValueError("measurement field has erased support")
    h, w = lum.shape
    # Robustly reduce the dense field to row/column means, then fit both slopes.
    col_mean = np.median(lum, axis=0)
    row_mean = np.median(lum, axis=1)
    x = np.linspace(-0.5, 0.5, w)
    y = np.linspace(-0.5, 0.5, h)
    horizontal_change = float(np.polyfit(x, col_mean, 1)[0])
    vertical_change = float(np.polyfit(y, row_mean, 1)[0])
    if not np.isfinite(horizontal_change + vertical_change):
        raise ValueError("invalid field fit")
    return "horizontal" if abs(horizontal_change) > abs(vertical_change) else "vertical"
