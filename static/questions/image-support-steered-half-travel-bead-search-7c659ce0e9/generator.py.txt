"""Renderer and analytic gold for the half-travel bead world.

One thin wire is folded into nine straight runs (alternating horizontal and
vertical) and laid out over a medium-sized region of a large plain page. Five
identical solid beads sit on the wire. Exactly one bead is reached after the
same travelled distance from either end of the wire, i.e. it splits the wire
into two halves of equal travel; every other bead is clearly off balance.

Only the travelled distances along the wire decide the answer. The bead that
balances the two straight-line distances to the ends is always a different bead,
the bead nearest the middle of the drawn region is always a different bead, ink
colour, page colour, page position and the background specks are randomised and
carry no information about the winner.

Everything is drawn with integer rectangles and no antialiasing.
"""

from __future__ import annotations

import random

import numpy as np
from PIL import Image

RENDERER_VERSION = "half_travel_bead_search-renderer-0.1.0"

WIDTH = 640
HEIGHT = 440
BACKGROUNDS = (
    (238, 238, 234),
    (233, 236, 241),
    (241, 237, 229),
)
INKS = (
    (26, 30, 38),
    (32, 62, 112),
    (122, 32, 42),
    (30, 92, 72),
    (92, 42, 112),
)

RUN_COUNT = 9              # nine straight runs, so eight right-angle corners
BEAD_COUNT = 5
BEAD_HALF = 6              # beads are thirteen by thirteen solid squares
SPECK_HALF = 1
MIN_RUN = 40
MAX_RUN = 96
MAX_SPAN_X = 330           # the wire covers a medium region, never the page
MAX_SPAN_Y = 250
MIN_SPAN_X = 190
MIN_SPAN_Y = 130
CANVAS_MARGIN = 24
MIN_SELF_CLEARANCE = 34    # non-adjacent runs never come close to touching
MIN_END_DX = 16            # the two bare ends are unambiguously left and right
MIN_END_GAP = 52           # no bead sits near either end of the wire
MIN_CORNER_GAP = 20        # every bead sits on a straight stretch
MIN_BEAD_GAP = 30          # consecutive beads never touch
CLEAR_SPECKS = 30          # keeps stray specks off the wire component
QUARANTINE_MARGIN = 76.0

WORDS = ("second", "third", "fourth")
CANDIDATE_ORDINALS = (2, 3, 4)

_DIRS = ((1, 0), (0, 1), (-1, 0), (0, -1))


# ---------------------------------------------------------------- geometry
def _polyline(first_dir: tuple[int, int], turns: list[int], lengths: list[int]):
    dirs = [first_dir]
    for sign in turns:
        dx, dy = dirs[-1]
        dirs.append((-dy, dx) if sign > 0 else (dy, -dx))
    points = [(0, 0)]
    for index in range(RUN_COUNT):
        dx, dy = dirs[index]
        x, y = points[-1]
        points.append((x + dx * lengths[index], y + dy * lengths[index]))
    return points


def _segment_gap(a1, a2, b1, b2) -> float:
    """Minimum distance between two axis-aligned segments, sampled on a grid."""

    def _points(p1, p2):
        (x1, y1), (x2, y2) = p1, p2
        if y1 == y2:
            step = 1 if x2 >= x1 else -1
            return [(x, y1) for x in range(x1, x2 + step, step * 4)] + [(x2, y2)]
        step = 1 if y2 >= y1 else -1
        return [(x1, y) for y in range(y1, y2 + step, step * 4)] + [(x2, y2)]

    best = float("inf")
    pts_a = _points(a1, a2)
    pts_b = _points(b1, b2)
    for x1, y1 in pts_a:
        for x2, y2 in pts_b:
            best = min(best, abs(x1 - x2) + abs(y1 - y2))
    return best


def _self_clear(points) -> bool:
    count = len(points) - 1
    for i in range(count):
        for j in range(i + 2, count):
            if _segment_gap(points[i], points[i + 1], points[j], points[j + 1]) < MIN_SELF_CLEARANCE:
                return False
    return True


def _cumulative(points) -> list[int]:
    out = [0]
    for index in range(len(points) - 1):
        (x1, y1), (x2, y2) = points[index], points[index + 1]
        out.append(out[-1] + abs(x2 - x1) + abs(y2 - y1))
    return out


def _point_at(points, cum, travel: int) -> tuple[int, int]:
    """Integer point reached after ``travel`` pixels of walk from ``points[0]``."""

    for index in range(len(points) - 1):
        if cum[index] <= travel <= cum[index + 1]:
            (x1, y1), (x2, y2) = points[index], points[index + 1]
            step = travel - cum[index]
            length = cum[index + 1] - cum[index]
            dx = (x2 - x1) // length
            dy = (y2 - y1) // length
            return (x1 + dx * step, y1 + dy * step)
    raise ValueError("travel outside the wire")


def _travel_of(points, cum, target) -> int:
    """Travelled distance of a point known to lie on the wire."""

    tx, ty = int(target[0]), int(target[1])
    for index in range(len(points) - 1):
        (x1, y1), (x2, y2) = points[index], points[index + 1]
        if y1 == y2 and ty == y1 and min(x1, x2) <= tx <= max(x1, x2):
            return cum[index] + abs(tx - x1)
        if x1 == x2 and tx == x1 and min(y1, y2) <= ty <= max(y1, y2):
            return cum[index] + abs(ty - y1)
    raise ValueError("bead is not on the wire")


def _corner_gap_ok(cum, travel: int) -> bool:
    return all(abs(travel - stop) >= MIN_CORNER_GAP for stop in cum[1:-1])


# ---------------------------------------------------------------- sampling
def _place_beads(rng, cum, total: int, target: int, tight: bool):
    """Bead travels: the winner sits at half travel, the others straddle it."""

    half = total // 2
    before = target - 1
    after = BEAD_COUNT - target
    for _ in range(160):
        near = rng.randrange(30, 46) if tight else rng.randrange(50, 92)
        near_side = rng.choice((-1, 1))       # which side holds the closest loser
        other = near + rng.randrange(MIN_BEAD_GAP // 2, 60)

        def _ladder(count: int, first: int) -> list[int]:
            out: list[int] = []
            value = first
            for _ in range(count):
                out.append(value)
                value += rng.randrange(MIN_BEAD_GAP, 76)
            return out

        gaps_before = _ladder(before, near if near_side < 0 else other)
        gaps_after = _ladder(after, near if near_side > 0 else other)
        offsets = gaps_before + gaps_after
        if not offsets or max(offsets) > half - MIN_END_GAP:
            continue
        travels = sorted([half - g for g in gaps_before] + [half] + [half + g for g in gaps_after])
        if any(t < MIN_END_GAP or t > total - MIN_END_GAP for t in travels):
            continue
        if any(travels[i + 1] - travels[i] < MIN_BEAD_GAP for i in range(BEAD_COUNT - 1)):
            continue
        if not all(_corner_gap_ok(cum, t) for t in travels):
            continue
        return travels
    return None


def sample_scene(seed: int) -> dict:
    """Deterministic scene; the balanced bead is cycled across consecutive seeds."""

    rng = random.Random(seed * 7907 + 29)
    target = CANDIDATE_ORDINALS[seed % len(CANDIDATE_ORDINALS)]
    tight = (seed // len(CANDIDATE_ORDINALS)) % 4 == 0
    for _ in range(6000):
        first_dir = _DIRS[rng.randrange(len(_DIRS))]
        turns = [rng.choice((-1, 1)) for _ in range(RUN_COUNT - 1)]
        lengths = [2 * rng.randrange(MIN_RUN // 2, MAX_RUN // 2 + 1) for _ in range(RUN_COUNT)]
        points = _polyline(first_dir, turns, lengths)
        xs = [p[0] for p in points]
        ys = [p[1] for p in points]
        span_x, span_y = max(xs) - min(xs), max(ys) - min(ys)
        if not (MIN_SPAN_X <= span_x <= MAX_SPAN_X and MIN_SPAN_Y <= span_y <= MAX_SPAN_Y):
            continue
        if abs(points[0][0] - points[-1][0]) < MIN_END_DX:
            continue
        if not _self_clear(points):
            continue
        # walk order: start from the bare end that lies further to the left
        if points[0][0] > points[-1][0]:
            points = points[::-1]
        cum = _cumulative(points)
        total = cum[-1]
        half = total // 2

        if not _corner_gap_ok(cum, half):
            continue
        travels = _place_beads(rng, cum, total, target, tight)
        if travels is None:
            continue

        origin_x = rng.randrange(CANVAS_MARGIN, WIDTH - CANVAS_MARGIN - span_x + 1)
        origin_y = rng.randrange(CANVAS_MARGIN, HEIGHT - CANVAS_MARGIN - span_y + 1)
        shift_x, shift_y = origin_x - min(xs), origin_y - min(ys)
        moved = [(x + shift_x, y + shift_y) for x, y in points]
        cum = _cumulative(moved)
        bead_points = [list(_point_at(moved, cum, t)) for t in travels]
        vertices = [[x, y] for x, y in moved]
        scene = {
            "width": WIDTH,
            "height": HEIGHT,
            "vertices": vertices,
            "bead_points": bead_points,
            "color": rng.randrange(len(INKS)),
            "background": rng.randrange(len(BACKGROUNDS)),
            "specks": _sample_specks(rng, vertices),
        }
        if _winner_index(scene) != target - 1:
            continue
        if not _counterbalanced(scene, target - 1):
            continue
        return scene
    raise RuntimeError(f"scene sampling failed for seed {seed}")


def _counterbalanced(scene: dict, winner: int) -> bool:
    """Cheap straight-line and centre cues must never point at the winner."""

    verts = [(int(x), int(y)) for x, y in scene["vertices"]]
    beads = [(int(x), int(y)) for x, y in scene["bead_points"]]
    end_a, end_b = verts[0], verts[-1]

    def _dist(p, q):
        return ((p[0] - q[0]) ** 2 + (p[1] - q[1]) ** 2) ** 0.5

    straight = [abs(_dist(b, end_a) - _dist(b, end_b)) for b in beads]
    if min(range(BEAD_COUNT), key=lambda i: straight[i]) == winner:
        return False
    xs = [v[0] for v in verts]
    ys = [v[1] for v in verts]
    centre = ((min(xs) + max(xs)) / 2.0, (min(ys) + max(ys)) / 2.0)
    if min(range(BEAD_COUNT), key=lambda i: _dist(beads[i], centre)) == winner:
        return False
    cloud = (
        sum(b[0] for b in beads) / float(BEAD_COUNT),
        sum(b[1] for b in beads) / float(BEAD_COUNT),
    )
    if min(range(BEAD_COUNT), key=lambda i: _dist(beads[i], cloud)) == winner:
        return False
    return True


def _sample_specks(rng: random.Random, vertices) -> list[list[int]]:
    xs = [v[0] for v in vertices]
    ys = [v[1] for v in vertices]
    lo_x, hi_x = min(xs) - CLEAR_SPECKS, max(xs) + CLEAR_SPECKS
    lo_y, hi_y = min(ys) - CLEAR_SPECKS, max(ys) + CLEAR_SPECKS
    specks: list[list[int]] = []
    wanted = rng.randrange(5, 10)
    for _ in range(3000):
        if len(specks) >= wanted:
            break
        x = rng.randrange(CANVAS_MARGIN // 2, WIDTH - CANVAS_MARGIN // 2)
        y = rng.randrange(CANVAS_MARGIN // 2, HEIGHT - CANVAS_MARGIN // 2)
        if lo_x <= x <= hi_x and lo_y <= y <= hi_y:
            continue
        if any((x - sx) ** 2 + (y - sy) ** 2 < 22 ** 2 for sx, sy in specks):
            continue
        specks.append([x, y])
    return specks


# ---------------------------------------------------------------- gold
def _scores(scene: dict) -> list[int]:
    """Absolute difference between the two travelled distances, per bead."""

    points = [(int(x), int(y)) for x, y in scene["vertices"]]
    if points[0][0] > points[-1][0]:
        points = points[::-1]
    cum = _cumulative(points)
    total = cum[-1]
    travels = sorted(_travel_of(points, cum, b) for b in scene["bead_points"])
    return [abs(2 * t - total) for t in travels]


def _winner_index(scene: dict) -> int:
    scores = _scores(scene)
    best = min(range(len(scores)), key=lambda i: scores[i])
    return best


def analytic_gold(scene: dict) -> str:
    index = _winner_index(scene)
    if index + 1 not in CANDIDATE_ORDINALS:
        raise ValueError("balanced bead is an end bead")
    return WORDS[index - 1]


def margin(scene: dict) -> float:
    """Runner-up travel imbalance, in pixels of travelled distance."""

    scores = _scores(scene)
    winner = _winner_index(scene)
    return float(min(s for i, s in enumerate(scores) if i != winner) - scores[winner])


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_MARGIN)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Latent rewrites that keep the raster identical and the gold unchanged."""

    specks = scene["specks"]
    out: list[tuple[str, dict]] = []
    if len(specks) >= 2:
        out.append(("specks_reversed", {**scene, "specks": specks[::-1]}))
        out.append(("specks_rotated", {**scene, "specks": specks[1:] + specks[:1]}))
    return out


# ---------------------------------------------------------------- rendering
def render(scene: dict) -> Image.Image:
    array = np.zeros((int(scene["height"]), int(scene["width"]), 3), dtype=np.uint8)
    array[:, :] = np.array(BACKGROUNDS[int(scene["background"]) % len(BACKGROUNDS)], dtype=np.uint8)
    ink = np.array(INKS[int(scene["color"]) % len(INKS)], dtype=np.uint8)
    points = [(int(x), int(y)) for x, y in scene["vertices"]]

    for index in range(len(points) - 1):
        (x1, y1), (x2, y2) = points[index], points[index + 1]
        if y1 == y2:
            array[y1, min(x1, x2) : max(x1, x2) + 1] = ink
        else:
            array[min(y1, y2) : max(y1, y2) + 1, x1] = ink

    for bx, by in scene["bead_points"]:
        bx, by = int(bx), int(by)
        array[by - BEAD_HALF : by + BEAD_HALF + 1, bx - BEAD_HALF : bx + BEAD_HALF + 1] = ink

    for sx, sy in scene["specks"]:
        sx, sy = int(sx), int(sy)
        array[sy - SPECK_HALF : sy + SPECK_HALF + 1, sx - SPECK_HALF : sx + SPECK_HALF + 1] = ink

    return Image.fromarray(array, mode="RGB")
