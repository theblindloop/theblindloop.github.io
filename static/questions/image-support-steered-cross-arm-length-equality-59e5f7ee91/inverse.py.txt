"""Independent inverse renderer. This module accepts pixels only."""

from __future__ import annotations

import numpy as np
from PIL import Image

EQUALITY_THRESHOLD_PX = 12


def _measure(image: Image.Image) -> tuple[list[int], float]:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    if array.shape[:2] != (512, 512):
        raise ValueError("unexpected raster dimensions")
    border = np.concatenate(
        (array[:4].reshape(-1, 3), array[-4:].reshape(-1, 3), array[:, :4].reshape(-1, 3), array[:, -4:].reshape(-1, 3)),
        axis=0,
    )
    background = np.median(border, axis=0)
    distance = np.linalg.norm(array - background, axis=2)
    # The public scene contract contains one colored foreground object, so the
    # complete chromatic mask is the object. Avoiding scene-specific location
    # assumptions also keeps translated and palette-permuted renders valid.
    component = distance > 55.0
    area = int(component.sum())
    if area < 22000:
        raise ValueError("colored cross is absent or too damaged")
    ys, xs = np.nonzero(component)
    ymin, ymax = int(ys.min()), int(ys.max())
    xmin, xmax = int(xs.min()), int(xs.max())
    row_counts = component.sum(axis=1)
    col_counts = component.sum(axis=0)
    row_peak = int(row_counts.max())
    col_peak = int(col_counts.max())
    band_rows = np.flatnonzero(row_counts >= max(20, int(0.72 * row_peak)))
    band_cols = np.flatnonzero(col_counts >= max(20, int(0.72 * col_peak)))
    if not len(band_rows) or not len(band_cols):
        raise ValueError("cross center bands cannot be recovered")
    cy = int(round(float(np.median(band_rows))))
    cx = int(round(float(np.median(band_cols))))
    if not (xmin < cx < xmax and ymin < cy < ymax):
        raise ValueError("invalid cross center")
    half_y = max(cy - int(band_rows.min()), int(band_rows.max()) - cy)
    half_x = max(cx - int(band_cols.min()), int(band_cols.max()) - cx)
    expected = np.zeros_like(component)
    expected[cy - half_y : cy + half_y + 1, xmin : xmax + 1] = True
    expected[ymin : ymax + 1, cx - half_x : cx + half_x + 1] = True
    missing = int(np.logical_and(expected, ~component).sum())
    excess = int(np.logical_and(component, ~expected).sum())
    mismatch_fraction = (missing + excess) / max(1, int(expected.sum()))
    # A one-percent loss is already a substantial missing patch on this solid
    # emblem; reject it so distant arm-tip evidence cannot be ignored.
    if mismatch_fraction > 0.008:
        raise ValueError("cross silhouette is materially damaged or ambiguous")
    extents = [cy - ymin, xmax - cx, ymax - cy, cx - xmin]
    return extents, mismatch_fraction


def decision_from_image(image: Image.Image) -> str:
    extents, _ = _measure(image)
    spread = max(extents) - min(extents)
    return "yes" if spread <= EQUALITY_THRESHOLD_PX else "no"
