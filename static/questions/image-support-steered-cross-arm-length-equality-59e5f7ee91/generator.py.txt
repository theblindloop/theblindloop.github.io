"""Deterministic analytic scene sampler and Pillow renderer."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (247, 245, 238)
PALETTE = (
    (32, 92, 190),
    (193, 51, 64),
    (21, 139, 104),
    (125, 67, 184),
    (211, 103, 28),
    (16, 145, 170),
)
EQUALITY_THRESHOLD_PX = 12
QUARANTINE_MARGIN_PX = 6


def sample_scene(seed: int) -> dict[str, Any]:
    # Consecutive even/odd seeds are a matched control/counterfactual pair.
    pair_key = int(seed) // 2
    rng = random.Random(pair_key ^ 0xC4A55EED)
    base = rng.randint(184, 192)
    center_x = rng.randint(252, 260)
    center_y = rng.randint(252, 260)
    half_thickness = rng.randint(58, 66)
    color = list(PALETTE[rng.randrange(len(PALETTE))])
    delta = rng.randint(24, 34)
    long_index = pair_key % 4
    short_index = (long_index + 1 + rng.randrange(3)) % 4
    equal = int(seed) % 2 == 0
    if equal:
        arms = [base, base, base, base]  # top, right, bottom, left
    else:
        arms = [base, base, base, base]
        arms[long_index] += delta
        arms[short_index] -= delta
    return {
        "arm_lengths": arms,
        "center_x": center_x,
        "center_y": center_y,
        "half_thickness": half_thickness,
        "color": color,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    arms = [int(value) for value in scene["arm_lengths"]]
    cx = int(scene["center_x"])
    cy = int(scene["center_y"])
    t = int(scene["half_thickness"])
    color = tuple(int(value) for value in scene["color"])
    top, right, bottom, left = arms
    if min(arms) < 130 or max(arms) > 235 or not (50 <= t <= 72):
        raise ValueError("cross geometry outside supported render range")
    if cx - left < 18 or cx + right >= WIDTH - 18 or cy - top < 18 or cy + bottom >= HEIGHT - 18:
        raise ValueError("cross would be clipped")
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    # The two rectangles overlap in the central square, forming one connected area.
    draw.rectangle((cx - left, cy - t, cx + right, cy + t), fill=color)
    draw.rectangle((cx - t, cy - top, cx + t, cy + bottom), fill=color)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    arms = [int(value) for value in scene["arm_lengths"]]
    return "yes" if max(arms) == min(arms) else "no"


def margin(scene: dict[str, Any]) -> float:
    arms = [int(value) for value in scene["arm_lengths"]]
    spread = max(arms) - min(arms)
    return float(EQUALITY_THRESHOLD_PX - spread if spread <= EQUALITY_THRESHOLD_PX else spread - EQUALITY_THRESHOLD_PX)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < QUARANTINE_MARGIN_PX


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict]]:
    return []
