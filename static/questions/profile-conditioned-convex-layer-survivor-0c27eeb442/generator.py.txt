"""Deterministic dot-field renderer and analytic convex-layer oracle."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

CANVAS = 512
DOT_RADIUS = 8


def _cross(o: tuple[float, float], a: tuple[float, float], b: tuple[float, float]) -> float:
    return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])


def _hull_indices(points: list[tuple[float, float]], indices: list[int]) -> list[int]:
    if len(indices) <= 2:
        return list(indices)
    ordered = sorted(indices, key=lambda i: (points[i][0], points[i][1], i))
    lower: list[int] = []
    for i in ordered:
        while len(lower) >= 2 and _cross(points[lower[-2]], points[lower[-1]], points[i]) <= 0:
            lower.pop()
        lower.append(i)
    upper: list[int] = []
    for i in reversed(ordered):
        while len(upper) >= 2 and _cross(points[upper[-2]], points[upper[-1]], points[i]) <= 0:
            upper.pop()
        upper.append(i)
    return list(dict.fromkeys(lower[:-1] + upper[:-1]))


def _onion_depths(points_raw: list[list[float]] | list[tuple[float, float]]) -> list[int]:
    points = [(float(p[0]), float(p[1])) for p in points_raw]
    remaining = list(range(len(points)))
    depths = [0] * len(points)
    layer = 1
    while remaining:
        hull = _hull_indices(points, remaining)
        if not hull:
            raise ValueError("convex layer extraction stalled")
        for i in hull:
            depths[i] = layer
        removed = set(hull)
        remaining = [i for i in remaining if i not in removed]
        layer += 1
    return depths


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    cx = 256.0 + rng.uniform(-15.0, 15.0)
    cy = 256.0 + rng.uniform(-15.0, 15.0)
    rotation = rng.uniform(0.0, 2.0 * math.pi)
    layer_specs = [
        (rng.randint(9, 11), 188.0, 163.0),
        (rng.randint(8, 10), 124.0, 106.0),
        (rng.randint(6, 8), 63.0, 54.0),
    ]
    points: list[list[float]] = []
    for n, rx, ry in layer_specs:
        phase = rng.uniform(0.0, 2.0 * math.pi)
        for k in range(n):
            theta = phase + 2.0 * math.pi * k / n + rng.uniform(-0.055, 0.055)
            scale = rng.uniform(0.965, 1.035)
            ex = rx * scale * math.cos(theta)
            ey = ry * scale * math.sin(theta)
            x = cx + ex * math.cos(rotation) - ey * math.sin(rotation)
            y = cy + ex * math.sin(rotation) + ey * math.cos(rotation)
            points.append([round(x, 3), round(y, 3)])

    depths = _onion_depths(points)
    unique = sorted(set(depths))
    if len(unique) < 3:
        raise RuntimeError("sample did not produce three convex layers")
    winner_pool = [i for i, d in enumerate(depths) if d == unique[-1]]
    runner_pool = [i for i, d in enumerate(depths) if d == unique[-2]]
    winner = rng.choice(winner_pool)
    runner = rng.choice(runner_pool)
    if int(seed) % 2 == 0:
        markers = {"red": winner, "blue": runner}
    else:
        markers = {"red": runner, "blue": winner}
    return {
        "canvas_size": CANVAS,
        "dot_radius": DOT_RADIUS,
        "points": points,
        "marker_indices": markers,
    }


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas_size"])
    radius = int(scene["dot_radius"])
    points = scene["points"]
    markers = scene["marker_indices"]
    scale = 4
    image = Image.new("RGB", (size * scale, size * scale), (248, 246, 240))
    draw = ImageDraw.Draw(image)
    colored = {int(markers["red"]), int(markers["blue"])}

    def disk(point: list[float], fill: tuple[int, int, int]) -> None:
        x, y = float(point[0]) * scale, float(point[1]) * scale
        r = radius * scale
        draw.ellipse((x - r, y - r, x + r, y + r), fill=fill)

    for i, point in enumerate(points):
        if i not in colored:
            disk(point, (25, 28, 31))
    disk(points[int(markers["red"])], (218, 55, 61))
    disk(points[int(markers["blue"])], (35, 104, 190))
    return image.resize((size, size), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    points = scene["points"]
    markers = scene["marker_indices"]
    depths = _onion_depths(points)
    red_depth = depths[int(markers["red"])]
    blue_depth = depths[int(markers["blue"])]
    if red_depth == blue_depth:
        return "tie"
    return "red" if red_depth > blue_depth else "blue"


def margin(scene: dict) -> float:
    depths = _onion_depths(scene["points"])
    markers = scene["marker_indices"]
    return float(abs(depths[int(markers["red"])] - depths[int(markers["blue"])]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse the unlabelled point order while updating colored indices."""
    points = scene["points"]
    markers = scene["marker_indices"]
    n = len(points)
    red = int(markers["red"])
    blue = int(markers["blue"])
    black = [i for i in range(n) if i not in {red, blue}]
    order = list(reversed(black)) + [red, blue]
    twin = copy.deepcopy(scene)
    twin["points"] = [copy.deepcopy(points[i]) for i in order]
    twin["marker_indices"] = {"red": len(order) - 2, "blue": len(order) - 1}
    return [("reverse_unlabelled_point_order", twin)]
