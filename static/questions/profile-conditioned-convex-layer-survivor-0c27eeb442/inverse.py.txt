"""Independent raster-only inverse for recursive convex-hull peeling."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    coords = set(zip(*np.nonzero(mask), strict=True))
    result: list[tuple[float, float, int]] = []
    while coords:
        start = coords.pop()
        queue = deque([start])
        ys: list[int] = []
        xs: list[int] = []
        while queue:
            y, x = queue.popleft()
            ys.append(y)
            xs.append(x)
            for nxt in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if nxt in coords:
                    coords.remove(nxt)
                    queue.append(nxt)
        if len(xs) >= 80:
            result.append((float(np.mean(xs)), float(np.mean(ys)), len(xs)))
    return result


def _cross(o: tuple[float, float], a: tuple[float, float], b: tuple[float, float]) -> float:
    return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])


def _hull(points: list[tuple[float, float]], indices: list[int]) -> list[int]:
    if len(indices) <= 2:
        return list(indices)
    ordered = sorted(indices, key=lambda i: (points[i][0], points[i][1], i))
    lo: list[int] = []
    for i in ordered:
        while len(lo) >= 2 and _cross(points[lo[-2]], points[lo[-1]], points[i]) <= 0:
            lo.pop()
        lo.append(i)
    hi: list[int] = []
    for i in reversed(ordered):
        while len(hi) >= 2 and _cross(points[hi[-2]], points[hi[-1]], points[i]) <= 0:
            hi.pop()
        hi.append(i)
    return list(dict.fromkeys(lo[:-1] + hi[:-1]))


def _depths(points: list[tuple[float, float]]) -> list[int]:
    remaining = list(range(len(points)))
    depths = [0] * len(points)
    layer = 1
    while remaining:
        hull = _hull(points, remaining)
        if not hull:
            raise ValueError("could not recover a convex layer")
        for i in hull:
            depths[i] = layer
        gone = set(hull)
        remaining = [i for i in remaining if i not in gone]
        layer += 1
    return depths


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    red_mask = (arr[:, :, 0] > 150) & (arr[:, :, 0] > arr[:, :, 1] * 1.45) & (arr[:, :, 0] > arr[:, :, 2] * 1.35)
    blue_mask = (arr[:, :, 2] > 120) & (arr[:, :, 2] > arr[:, :, 0] * 1.35) & (arr[:, :, 2] > arr[:, :, 1] * 1.15)
    black_mask = (arr[:, :, 0] < 90) & (arr[:, :, 1] < 90) & (arr[:, :, 2] < 90)
    red = _components(red_mask)
    blue = _components(blue_mask)
    black = _components(black_mask)
    if len(red) != 1 or len(blue) != 1 or len(black) < 15:
        raise ValueError(f"expected one red, one blue, and at least 15 black dots; got {len(red)}, {len(blue)}, {len(black)}")
    points = [(item[0], item[1]) for item in black] + [(red[0][0], red[0][1]), (blue[0][0], blue[0][1])]
    depths = _depths(points)
    red_depth, blue_depth = depths[-2], depths[-1]
    if red_depth == blue_depth:
        return "tie"
    return "red" if red_depth > blue_depth else "blue"
