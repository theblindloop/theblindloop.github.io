"""Deterministic renderer and analytic arm for signed arrow resultants."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 640
HEIGHT = 640
BACKGROUND = (249, 248, 244)
INK = (48, 54, 61)
GRID = (211, 215, 215)
PALETTES = (
    (16, 126, 164),
    (196, 77, 42),
    (55, 137, 74),
    (132, 70, 166),
    (188, 112, 12),
)
PANEL_TONES = (
    (237, 240, 238),
    (239, 237, 242),
    (242, 238, 233),
)
HEAD_HALF_WIDTH = 16
SHAFT_HALF_WIDTH = 6
HEAD_HEIGHT = 20
QUARANTINE_PX = 10


def _font(size: int, *, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def sample_scene(seed: int) -> dict[str, Any]:
    """Return pair-balanced scenes with three disconnected directed glyphs."""

    seed = int(seed)
    pair_seed = seed // 2
    rng = random.Random(0xA770_2026 + pair_seed * 1_000_003)

    # Types 0/2 have one upward arrow and types 1/3 have two; every type has
    # a positive base resultant. Odd seeds reverse all arrows. Thus each
    # answer occurs with either direction count, while pair mates preserve
    # lengths, placement, palette, and colored-pixel total.
    variant = pair_seed % 4
    if variant < 2:
        base_lengths = [48, 64, 118]
    else:
        base_lengths = [46, 68, 130]
    if variant in (0, 2):
        base_directions = ["down", "down", "up"]
    else:
        base_directions = ["down", "up", "up"]
    if seed % 2:
        base_directions = ["down" if value == "up" else "up" for value in base_directions]

    order = [0, 1, 2]
    rng.shuffle(order)
    lengths = [base_lengths[index] for index in order]
    directions = [base_directions[index] for index in order]
    dx = (-8, 0, 8, 4)[pair_seed % 4]
    dy = (-4, 3, 0)[(pair_seed // 4) % 3]
    centers = [[200 + dx, 180 + dy], [440 + dx, 180 + dy], [320 + dx, 460 + dy]]

    return {
        "arrow_centers": centers,
        "lengths": lengths,
        "directions": directions,
        "arrow_rgb": list(PALETTES[pair_seed % len(PALETTES)]),
        "panel_rgb": list(PANEL_TONES[(pair_seed // 3) % len(PANEL_TONES)]),
    }


def _measurements(scene: dict[str, Any]) -> tuple[list[tuple[int, int]], list[int], list[str]]:
    centers = [tuple(int(value) for value in point) for point in scene["arrow_centers"]]
    lengths = [int(value) for value in scene["lengths"]]
    directions = [str(value) for value in scene["directions"]]
    if len(centers) != 3 or len(lengths) != 3 or len(directions) != 3:
        raise ValueError("exactly three arrows are required")
    if len(set(centers)) != 3:
        raise ValueError("arrow centers must be distinct")
    if any(not 42 <= length <= 140 for length in lengths):
        raise ValueError("arrow length is outside the visible range")
    if any(direction not in {"up", "down"} for direction in directions):
        raise ValueError("arrow direction must be up or down")
    return centers, lengths, directions


def _signed_sum(scene: dict[str, Any]) -> int:
    lengths = [int(value) for value in scene["lengths"]]
    directions = [str(value) for value in scene["directions"]]
    if len(lengths) != 3 or len(directions) != 3:
        raise ValueError("exactly three signed measurements are required")
    if any(direction not in {"up", "down"} for direction in directions):
        raise ValueError("arrow direction must be up or down")
    return sum(length if direction == "up" else -length for length, direction in zip(lengths, directions))


def analytic_gold(scene: dict[str, Any]) -> str:
    signed = _signed_sum(scene)
    if signed == 0:
        raise ValueError("zero resultant lies on the decision boundary")
    return "up" if signed > 0 else "down"


def margin(scene: dict[str, Any]) -> float:
    """Runner-up margin: absolute signed length sum in final raster pixels."""

    return float(abs(_signed_sum(scene)))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) <= QUARANTINE_PX


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    transformed = dict(scene)
    for key in ("arrow_centers", "lengths", "directions"):
        transformed[key] = list(reversed(scene[key]))
    return [("reverse_arrow_storage_order", transformed)]


def _draw_arrow(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    length: int,
    direction: str,
    color: tuple[int, int, int],
) -> None:
    x, center_y = center
    top = center_y - length // 2
    bottom = top + length
    if direction == "up":
        points = (
            (x, top),
            (x + HEAD_HALF_WIDTH, top + HEAD_HEIGHT),
            (x + SHAFT_HALF_WIDTH, top + HEAD_HEIGHT),
            (x + SHAFT_HALF_WIDTH, bottom),
            (x - SHAFT_HALF_WIDTH, bottom),
            (x - SHAFT_HALF_WIDTH, top + HEAD_HEIGHT),
            (x - HEAD_HALF_WIDTH, top + HEAD_HEIGHT),
        )
    else:
        points = (
            (x - SHAFT_HALF_WIDTH, top),
            (x + SHAFT_HALF_WIDTH, top),
            (x + SHAFT_HALF_WIDTH, bottom - HEAD_HEIGHT),
            (x + HEAD_HALF_WIDTH, bottom - HEAD_HEIGHT),
            (x, bottom),
            (x - HEAD_HALF_WIDTH, bottom - HEAD_HEIGHT),
            (x - SHAFT_HALF_WIDTH, bottom - HEAD_HEIGHT),
        )
    draw.polygon(points, fill=color)


def render(scene: dict[str, Any]) -> Image.Image:
    centers, lengths, directions = _measurements(scene)
    arrow_rgb = tuple(int(value) for value in scene["arrow_rgb"])
    panel_rgb = tuple(int(value) for value in scene["panel_rgb"])
    if len(arrow_rgb) != 3 or len(panel_rgb) != 3:
        raise ValueError("RGB fields must have three channels")
    if any(not (90 <= x <= 550 and 80 <= y <= 560) for x, y in centers):
        raise ValueError("arrow center lies outside the bounded plate")

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    xs = [point[0] for point in centers]
    ys = [point[1] for point in centers]
    panel_box = (min(xs) - 72, min(ys) - 112, max(xs) + 72, max(ys) + 106)
    draw.rounded_rectangle(panel_box, radius=18, fill=panel_rgb, outline=GRID, width=2)
    draw.text((panel_box[0] + 20, panel_box[1] + 16), "SIGNED ADJUSTMENTS", font=_font(15, bold=True), fill=INK)
    draw.text((panel_box[0] + 20, panel_box[1] + 39), "UP +     DOWN −     add tip-to-tail lengths", font=_font(13), fill=INK)
    draw.line((panel_box[0] + 18, 322, panel_box[2] - 18, 322), fill=GRID, width=1)

    for center, length, direction in zip(centers, lengths, directions):
        _draw_arrow(draw, center, length, direction, arrow_rgb)
    return image
