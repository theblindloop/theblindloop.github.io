"""Independent final-PNG inverse arm for the signed arrow resultant.

The arm discovers saturated connected glyphs without using scene data or fixed
locations. For each polygon it recovers the visible bounding height, detects
which end contains the triangular arrowhead, validates the glyph profile, and
adds the resulting signed tip-to-tail lengths.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


def _colored_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    high = array.max(axis=2)
    low = array.min(axis=2)
    return ((high - low) >= 55) & (high >= 100) & (low <= 195)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components: list[np.ndarray] = []
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) < 80:
            raise ValueError("unexpected colored speckle or damaged arrow fragment")
        components.append(np.asarray(points, dtype=np.int32))
    return components


def _signed_length(component: np.ndarray, mask: np.ndarray) -> int:
    ys = component[:, 0]
    xs = component[:, 1]
    y0, y1 = int(ys.min()), int(ys.max())
    x0, x1 = int(xs.min()), int(xs.max())
    width = x1 - x0 + 1
    height = y1 - y0 + 1
    if width != 33 or not 43 <= height <= 141:
        raise ValueError("colored component has an invalid arrow extent")
    sub = mask[y0 : y1 + 1, x0 : x1 + 1]
    if int(sub.sum()) != int(component.shape[0]):
        raise ValueError("arrow component contains a disconnected colored defect")
    row_widths = sub.sum(axis=1).astype(int)
    if np.any(row_widths == 0) or int(row_widths.max()) < 31:
        raise ValueError("arrow shaft or head is incomplete")
    top_flare = int(row_widths[:21].sum())
    bottom_flare = int(row_widths[-21:].sum())
    if abs(top_flare - bottom_flare) < 50:
        raise ValueError("arrowhead direction is unresolved")
    direction = 1 if top_flare > bottom_flare else -1
    if direction > 0:
        tip_width, tail_width = int(row_widths[0]), int(row_widths[-1])
    else:
        tip_width, tail_width = int(row_widths[-1]), int(row_widths[0])
    if tip_width > 3 or not 11 <= tail_width <= 13:
        raise ValueError("arrow tip or tail profile is malformed")
    density = float(component.shape[0] / (width * height))
    if not 0.34 <= density <= 0.49:
        raise ValueError("arrow polygon has implausible fill density")
    return direction * (height - 1)


def decision_from_image(image: Image.Image) -> str:
    mask = _colored_mask(image)
    components = _components(mask)
    if len(components) != 3:
        raise ValueError("exactly three disconnected colored arrows are required")
    signed_total = sum(_signed_length(component, mask) for component in components)
    if signed_total == 0:
        raise ValueError("the recovered resultant lies on the decision boundary")
    return "up" if signed_total > 0 else "down"
