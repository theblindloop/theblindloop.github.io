"""Deterministic renderer and analytic arm for Fold-Cover Correspondence."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

CANVAS = 512
ROWS = 6
HALF_COLS = 3
CELL = 36
BOARD = ROWS * CELL
BASE_X = (CANVAS - BOARD) // 2
BASE_Y = (CANVAS - BOARD) // 2
COLORS = ("amber", "cyan", "violet")

BACKGROUND = (250, 248, 243)
PAPER = (255, 255, 255)
FRAME = (65, 76, 88)
HINGE = (178, 54, 104)
INK = (37, 45, 56)
PALE = (224, 232, 238)
TEXT = (50, 58, 68)

PALETTES = (
    {
        "amber": (232, 158, 38),
        "cyan": (34, 171, 190),
        "violet": (135, 86, 184),
    },
    {
        "amber": (218, 143, 26),
        "cyan": (26, 151, 174),
        "violet": (151, 91, 190),
    },
    {
        "amber": (241, 174, 53),
        "cyan": (45, 181, 198),
        "violet": (122, 75, 173),
    },
)


def _flat_index(row: int, col: int) -> int:
    return row * HALF_COLS + col


def _coverage_counts(scene: dict) -> dict[str, int]:
    left = scene["left_mask"]
    right = scene["right_cells"]
    counts = {color: 0 for color in COLORS}
    for row in range(ROWS):
        for right_col in range(HALF_COLS):
            left_col = HALF_COLS - 1 - right_col
            if bool(left[_flat_index(row, left_col)]):
                counts[str(right[_flat_index(row, right_col)])] += 1
    return counts


def sample_scene(seed: int) -> dict:
    """Sample one balanced scene; consecutive seeds cycle answer identity."""

    rng = random.Random(int(seed) ^ 0xF01DC0DE)
    target = COLORS[int(seed) % len(COLORS)]

    right_cells = [color for color in COLORS for _ in range(6)]
    rng.shuffle(right_cells)

    covered_right = [color == target for color in right_cells]
    distractors = [color for color in COLORS if color != target]
    rng.shuffle(distractors)
    near_boundary = int(seed) % 5 == 0
    requested = (5, 3) if near_boundary else (4, 4)
    for color, amount in zip(distractors, requested, strict=True):
        indices = [i for i, value in enumerate(right_cells) if value == color]
        rng.shuffle(indices)
        for index in indices[:amount]:
            covered_right[index] = True

    left_mask = [False] * (ROWS * HALF_COLS)
    for row in range(ROWS):
        for right_col in range(HALF_COLS):
            left_col = HALF_COLS - 1 - right_col
            left_mask[_flat_index(row, left_col)] = covered_right[
                _flat_index(row, right_col)
            ]

    return {
        "left_mask": left_mask,
        "right_cells": right_cells,
        "offset_x": rng.choice((-12, 0, 12)),
        "offset_y": rng.choice((-12, 0, 12)),
        "palette_variant": int(seed) % len(PALETTES),
    }


def analytic_gold(scene: dict) -> str:
    counts = _coverage_counts(scene)
    complete = [color for color in COLORS if counts[color] == 6]
    if len(complete) != 1:
        raise ValueError(f"scene has {len(complete)} fully covered colors")
    return complete[0]


def margin(scene: dict) -> float:
    counts = sorted(_coverage_counts(scene).values(), reverse=True)
    return float(counts[0] - counts[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _tile_box(x0: int, y0: int, row: int, col: int) -> tuple[int, int, int, int]:
    gap = 3
    left = x0 + col * CELL + gap
    top = y0 + row * CELL + gap
    return (left, top, left + CELL - 2 * gap - 1, top + CELL - 2 * gap - 1)


def _draw_dashed_vertical(
    draw: ImageDraw.ImageDraw, x: int, y0: int, y1: int, *, width: int = 4
) -> None:
    dash = 10
    gap = 7
    y = y0
    while y < y1:
        draw.line((x, y, x, min(y + dash, y1)), fill=HINGE, width=width)
        y += dash + gap


def _font(size: int) -> ImageFont.ImageFont:
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf", size=size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    left_mask = scene["left_mask"]
    right_cells = scene["right_cells"]
    x0 = BASE_X + int(scene["offset_x"])
    y0 = BASE_Y + int(scene["offset_y"])
    palette = PALETTES[int(scene["palette_variant"])]

    if len(left_mask) != ROWS * HALF_COLS or len(right_cells) != ROWS * HALF_COLS:
        raise ValueError("both half-grids must contain exactly eighteen tiles")

    image = Image.new("RGB", (CANVAS, CANVAS), BACKGROUND)
    draw = ImageDraw.Draw(image)

    draw.rounded_rectangle(
        (x0 - 10, y0 - 10, x0 + BOARD + 9, y0 + BOARD + 9),
        radius=12,
        fill=PAPER,
        outline=FRAME,
        width=3,
    )
    for row in range(ROWS):
        for left_col in range(HALF_COLS):
            value = bool(left_mask[_flat_index(row, left_col)])
            draw.rectangle(
                _tile_box(x0, y0, row, left_col),
                fill=INK if value else PALE,
            )
        for right_col in range(HALF_COLS):
            color = str(right_cells[_flat_index(row, right_col)])
            if color not in palette:
                raise ValueError(f"unknown answer color {color!r}")
            draw.rectangle(
                _tile_box(x0, y0, row, HALF_COLS + right_col),
                fill=palette[color],
            )

    hinge_x = x0 + HALF_COLS * CELL
    draw.rectangle((hinge_x - 3, y0 - 5, hinge_x + 3, y0 + BOARD + 4), fill=PAPER)
    _draw_dashed_vertical(draw, hinge_x, y0 - 5, y0 + BOARD + 5)

    label_font = _font(16)
    small_font = _font(13)
    draw.text(
        (x0 + HALF_COLS * CELL // 2, y0 - 31),
        "STENCIL",
        fill=TEXT,
        anchor="mm",
        font=label_font,
    )
    draw.text(
        (x0 + HALF_COLS * CELL + HALF_COLS * CELL // 2, y0 - 31),
        "COLOR SETS",
        fill=TEXT,
        anchor="mm",
        font=label_font,
    )
    draw.text(
        (hinge_x, y0 + BOARD + 27),
        "FOLD",
        fill=HINGE,
        anchor="mm",
        font=small_font,
    )
    draw.line(
        (x0 + 45, y0 + BOARD + 26, hinge_x - 20, y0 + BOARD + 26),
        fill=HINGE,
        width=3,
    )
    draw.polygon(
        (
            (hinge_x - 20, y0 + BOARD + 26),
            (hinge_x - 31, y0 + BOARD + 20),
            (hinge_x - 31, y0 + BOARD + 32),
        ),
        fill=HINGE,
    )
    return image


def coverage_counts(scene: dict) -> dict[str, int]:
    """Public analytic diagnostic used by generation and tests."""

    return _coverage_counts(scene)
