"""Latent scene sampler and renderer for a bend-marked orthogonal cable."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
PALETTE = {
    "orange": (230, 126, 34),
    "blue": (40, 105, 210),
    "green": (35, 158, 88),
}
START = (194, 48, 171)
DARK = (36, 42, 48)
BACKGROUND = (250, 249, 246)


def _transform(points: list[tuple[int, int]], code: int) -> list[tuple[int, int]]:
    """Apply one of the eight square symmetries about the canvas center."""
    out = []
    for x, y in points:
        u, v = x - 256, y - 256
        if code == 0:
            a, b = u, v
        elif code == 1:
            a, b = -u, v
        elif code == 2:
            a, b = u, -v
        elif code == 3:
            a, b = -u, -v
        elif code == 4:
            a, b = v, u
        elif code == 5:
            a, b = -v, u
        elif code == 6:
            a, b = v, -u
        else:
            a, b = -v, -u
        out.append((a + 256, b + 256))
    return out


def _valid(points: list[tuple[int, int]]) -> bool:
    if any(not (24 <= x <= WIDTH - 24 and 24 <= y <= HEIGHT - 24) for x, y in points):
        return False
    # Only adjacent vertices may share an x or y coordinate. This keeps the
    # visible node-link graph unambiguous for both people and the pixel arm.
    for i, (x1, y1) in enumerate(points):
        for j, (x2, y2) in enumerate(points):
            if j <= i + 1:
                continue
            if abs(x1 - x2) < 3 or abs(y1 - y2) < 3:
                return False
    return True


def _colors_for(seed: int, longest_endpoint: int, rng: random.Random) -> list[str]:
    """Two markers of each color, with a seed-balanced answer identity."""
    answer = ("orange", "blue", "green")[seed % 3]
    remaining = ["orange", "orange", "blue", "blue", "green", "green"]
    remaining.remove(answer)
    rng.shuffle(remaining)
    colors: list[str] = []
    for endpoint in range(1, 7):
        colors.append(answer if endpoint == longest_endpoint else remaining.pop())
    return colors


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    # Every 4th scene has a deliberately smaller but still safe runner-up gap.
    lengths = [62, 78, 96, 116, 145, 171] if seed % 4 == 0 else [62, 78, 96, 116, 142, 176]
    for _ in range(500):
        rng.shuffle(lengths)
        x, y = 44, 174
        points = [(x, y)]
        for index, length in enumerate(lengths):
            if index % 2 == 0:
                x += length
            elif index % 4 == 1:
                y += length
            else:
                y -= length
            points.append((x, y))
        points = _transform(points, rng.randrange(8))
        dx, dy = rng.randint(-8, 8), rng.randint(-8, 8)
        points = [(px + dx, py + dy) for px, py in points]
        if _valid(points):
            longest_endpoint = 1 + max(range(6), key=lambda i: lengths[i])
            return {
                "points": [[px, py] for px, py in points],
                "checkpoint_colors": _colors_for(seed, longest_endpoint, rng),
            }
    raise RuntimeError("could not sample an unambiguous cable")


def _segment_lengths(scene: dict) -> list[float]:
    points = scene["points"]
    return [abs(b[0] - a[0]) + abs(b[1] - a[1]) for a, b in zip(points, points[1:])]


def analytic_gold(scene: dict) -> str:
    points = scene["points"]
    colors = scene["checkpoint_colors"]
    lengths = [abs(b[0] - a[0]) + abs(b[1] - a[1]) for a, b in zip(points, points[1:])]
    return colors[max(range(len(lengths)), key=lengths.__getitem__)]


def margin(scene: dict) -> float:
    ranked = sorted(_segment_lengths(scene), reverse=True)
    return float(ranked[0] - ranked[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 24.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # No two different declared latents are intended to produce identical pixels.
    return []


def render(scene: dict) -> Image.Image:
    points = [tuple(point) for point in scene["points"]]
    colors = scene["checkpoint_colors"]
    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    scaled = [(x * scale, y * scale) for x, y in points]
    draw.line(scaled, fill=DARK, width=7 * scale, joint="curve")

    # Filled pale centers visibly separate the links at each marker, making the
    # node-link structure explicit while leaving every link fully measurable.
    radius = 10 * scale
    for (x, y), name in zip(scaled[1:], colors):
        box = (x - radius, y - radius, x + radius, y + radius)
        draw.ellipse(box, fill=BACKGROUND, outline=PALETTE[name], width=5 * scale)

    sx, sy = scaled[0]
    r = 11 * scale
    draw.rectangle((sx - r, sy - r, sx + r, sy + r), fill=START)
    inner = 4 * scale
    draw.rectangle((sx - inner, sy - inner, sx + inner, sy + inner), fill=BACKGROUND)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
