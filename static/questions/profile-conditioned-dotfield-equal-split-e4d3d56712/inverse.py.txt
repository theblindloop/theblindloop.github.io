"""Independent pixel-only inverse arm.

This module intentionally has no dependency on any candidate module and receives
only a raster image.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    found: list[tuple[float, float, int]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            yy, xx = queue.popleft()
            xs.append(xx)
            ys.append(yy)
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = yy + dy, xx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= minimum:
            found.append((float(np.mean(xs)), float(np.mean(ys)), len(xs)))
    return found


def _line(mask: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 120:
        raise ValueError("colored line evidence is missing")
    points = np.column_stack((xs.astype(float), ys.astype(float)))
    center = points.mean(axis=0)
    covariance = np.cov(points - center, rowvar=False)
    values, vectors = np.linalg.eigh(covariance)
    direction = vectors[:, int(np.argmax(values))]
    if float(values.max()) < 500.0:
        raise ValueError("colored evidence is not line-like")
    return center, direction


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[..., 0], array[..., 1], array[..., 2]
    black = (red < 70) & (green < 70) & (blue < 70)
    orange = (red > 180) & (green > 80) & (green < 190) & (blue < 90)
    blue_mask = (blue > 170) & (red < 100) & (green < 170)
    dots = _components(black, minimum=80)
    if not 12 <= len(dots) <= 24:
        raise ValueError(f"expected a complete dot constellation, found {len(dots)} marks")
    centers = np.array([[x, y] for x, y, _ in dots], dtype=float)
    residuals: dict[str, int] = {}
    for color, mask in (("orange", orange), ("blue", blue_mask)):
        center, direction = _line(mask)
        signed = direction[0] * (centers[:, 1] - center[1]) - direction[1] * (
            centers[:, 0] - center[0]
        )
        if np.min(np.abs(signed)) < 10.0:
            raise ValueError("a dot is too close to a candidate line")
        residuals[color] = abs(int(np.sum(signed > 0)) - int(np.sum(signed < 0)))
    winner = min(residuals, key=residuals.get)
    ordered = sorted(residuals.values())
    if ordered[0] != 0 or ordered[1] - ordered[0] < 4:
        raise ValueError(f"ambiguous split residuals {residuals}")
    return winner
