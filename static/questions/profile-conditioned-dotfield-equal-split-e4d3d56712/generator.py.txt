"""Deterministic renderer for a global equal-cardinality line split."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 512
PALETTE = {"orange": (242, 142, 43), "blue": (47, 111, 237)}


def _signed_distance(point: list[float], line: dict) -> float:
    x, y = point
    cx, cy = line["center"]
    theta = line["angle"]
    return -(x - cx) * math.sin(theta) + (y - cy) * math.cos(theta)


def _residuals(scene: dict) -> list[tuple[str, int]]:
    points = scene["dot_centers"]
    lines = scene["lines"]
    values = []
    for line in lines:
        positive = sum(_signed_distance(point, line) > 0 for point in points)
        negative = sum(_signed_distance(point, line) < 0 for point in points)
        values.append((line["color"], abs(positive - negative)))
    return values


def sample_scene(seed: int) -> dict:
    """Sample a balanced target split and a visibly imbalanced runner-up."""
    rng = random.Random(int(seed))
    target_color = "orange" if int(seed) % 2 == 0 else "blue"
    other_color = "blue" if target_color == "orange" else "orange"
    n = (14, 16, 18, 20)[rng.randrange(4)]
    cx = rng.uniform(238.0, 274.0)
    cy = rng.uniform(238.0, 274.0)
    target_angle = rng.uniform(0.0, math.pi)
    offset = rng.choice((-1.0, 1.0)) * rng.uniform(math.radians(28), math.radians(68))
    other_angle = (target_angle + offset) % math.pi
    target = {"color": target_color, "center": [cx, cy], "angle": target_angle}
    other = {"color": other_color, "center": [cx, cy], "angle": other_angle}

    # Reject complete point sets rather than encoding a hidden answer variable.
    # Both decisions are computed solely from the rendered centers and lines.
    for _ in range(20000):
        points: list[list[float]] = []
        for _index in range(n):
            for _attempt in range(2000):
                x = rng.uniform(70.0, 442.0)
                y = rng.uniform(70.0, 442.0)
                if ((x - cx) / 190.0) ** 2 + ((y - cy) / 190.0) ** 2 > 1.0:
                    continue
                point = [x, y]
                if abs(_signed_distance(point, target)) < 19.0:
                    continue
                if abs(_signed_distance(point, other)) < 19.0:
                    continue
                if any(math.dist(point, prior) < 27.0 for prior in points):
                    continue
                points.append(point)
                break
            else:
                break
        if len(points) != n:
            continue
        scene = {"dot_centers": points, "lines": [target, other]}
        residuals = dict(_residuals(scene))
        if residuals[target_color] == 0 and residuals[other_color] >= 4:
            # Random draw order prevents a stable first-slot cue in stored JSON.
            if rng.random() < 0.5:
                scene["lines"].reverse()
            return scene
    raise RuntimeError(f"could not sample certified scene for seed {seed}")


def _dash(draw: ImageDraw.ImageDraw, line: dict) -> None:
    cx, cy = line["center"]
    ux, uy = math.cos(line["angle"]), math.sin(line["angle"])
    color = PALETTE[line["color"]]
    for start in range(-220, 220, 32):
        end = min(start + 20, 220)
        draw.line(
            [(cx + start * ux, cy + start * uy), (cx + end * ux, cy + end * uy)],
            fill=color,
            width=6,
        )


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (248, 248, 245))
    draw = ImageDraw.Draw(image)
    # Candidate partitions sit behind the constellation, so all dot centers stay visible.
    for line in scene["lines"]:
        _dash(draw, line)
    for x, y in scene["dot_centers"]:
        draw.ellipse((x - 7, y - 7, x + 7, y + 7), fill=(24, 27, 31))
    return image


def analytic_gold(scene: dict) -> str:
    residuals = _residuals(scene)
    best_color, best = min(residuals, key=lambda item: item[1])
    if best != 0 or sum(value == 0 for _, value in residuals) != 1:
        raise ValueError("scene has no unique exact split")
    return best_color


def margin(scene: dict) -> float:
    ordered = sorted(value for _, value in _residuals(scene))
    return float(ordered[1] - ordered[0])


def is_quarantined(scene: dict) -> bool:
    clearance = min(
        abs(_signed_distance(point, line))
        for point in scene["dot_centers"]
        for line in scene["lines"]
    )
    return clearance < 15.0 or margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
