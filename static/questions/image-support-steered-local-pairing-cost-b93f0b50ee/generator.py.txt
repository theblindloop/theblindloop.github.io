"""Deterministic renderer and analytic arm for Local Pairing Cost."""

from __future__ import annotations

import copy
import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

CANVAS = 512
POINT_RADIUS = 9
QUARANTINE_PX = 5.0
PALETTES = (
    ((0, 140, 168), (226, 92, 0)),
    ((0, 128, 157), (215, 82, 0)),
    ((0, 151, 177), (238, 104, 0)),
)


def _distance(left: Iterable[float], right: Iterable[float]) -> float:
    lx, ly = left
    rx, ry = right
    return math.hypot(float(lx) - float(rx), float(ly) - float(ry))


def _matching_costs_from_scene(scene: dict) -> tuple[float, float]:
    cyan = scene["cyan_points"]
    orange = scene["orange_points"]
    within = _distance(cyan[0], cyan[1]) + _distance(orange[0], orange[1])
    cross_a = _distance(cyan[0], orange[0]) + _distance(cyan[1], orange[1])
    cross_b = _distance(cyan[0], orange[1]) + _distance(cyan[1], orange[0])
    return within, min(cross_a, cross_b)


def analytic_gold(scene: dict) -> str:
    within, cross = _matching_costs_from_scene(scene)
    return "WITHIN" if within < cross else "CROSS"


def margin(scene: dict) -> float:
    within, cross = _matching_costs_from_scene(scene)
    return abs(within - cross)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def _candidate_points(rng: random.Random) -> list[list[int]]:
    # Centers lie in four nonadjacent 8x8-ablation cells. The small jitter and
    # marker radius stay inside the cells' non-overlap cores.
    origin_x = rng.choice((96, 160, 224))
    origin_y = rng.choice((96, 160, 224))
    bases = (
        (origin_x, origin_y),
        (origin_x + 128, origin_y),
        (origin_x, origin_y + 128),
        (origin_x + 128, origin_y + 128),
    )
    return [[x + rng.randint(-6, 6), y + rng.randint(-6, 6)] for x, y in bases]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x51A7C0DE)
    desired = "WITHIN" if int(seed) % 2 == 0 else "CROSS"
    near_boundary = (int(seed) // 2) % 3 == 0

    for _ in range(50000):
        points = _candidate_points(rng)
        cyan_indices = set(rng.sample(range(4), 2))
        cyan = [point for index, point in enumerate(points) if index in cyan_indices]
        orange = [point for index, point in enumerate(points) if index not in cyan_indices]
        scene = {
            "canvas_size": [CANVAS, CANVAS],
            "radius": POINT_RADIUS,
            "cyan_points": cyan,
            "orange_points": orange,
            "palette_id": rng.randrange(len(PALETTES)),
        }
        observed = analytic_gold(scene)
        observed_margin = margin(scene)
        if observed != desired:
            continue
        if near_boundary and 6.0 <= observed_margin <= 14.0:
            return scene
        if not near_boundary and observed_margin >= 16.0:
            return scene
    raise RuntimeError(f"could not sample {desired} scene for seed {seed}")


def render(scene: dict) -> Image.Image:
    width, height = (int(value) for value in scene["canvas_size"])
    radius = int(scene["radius"])
    cyan, orange = PALETTES[int(scene["palette_id"])]
    scale = 4
    image = Image.new("RGB", (width * scale, height * scale), (255, 255, 255))
    draw = ImageDraw.Draw(image)
    for points, color in ((scene["cyan_points"], cyan), (scene["orange_points"], orange)):
        for x, y in points:
            box = (
                (int(x) - radius) * scale,
                (int(y) - radius) * scale,
                (int(x) + radius) * scale,
                (int(y) + radius) * scale,
            )
            draw.ellipse(box, fill=color)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["cyan_points"] = list(reversed(twin["cyan_points"]))
    twin["orange_points"] = list(reversed(twin["orange_points"]))
    return [("reverse_same_color_list_orders", twin)]
