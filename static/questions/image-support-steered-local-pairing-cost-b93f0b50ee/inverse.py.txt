"""Independent pixel-only inverse arm for Local Pairing Cost.

This module knows only the public marker color families and the question rule.
It does not import or inspect renderer state or latent scenes.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

CYAN_PROTOTYPES = np.asarray(((0, 140, 168), (0, 128, 157), (0, 151, 177)), dtype=np.float32)
ORANGE_PROTOTYPES = np.asarray(((226, 92, 0), (215, 82, 0), (238, 104, 0)), dtype=np.float32)
WHITE = np.asarray((255, 255, 255), dtype=np.float32)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[np.ndarray] = []
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        pixels: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            pixels.append((cy, cx))
            for ny in range(cy - 1, cy + 2):
                for nx in range(cx - 1, cx + 2):
                    if (
                        0 <= ny < height
                        and 0 <= nx < width
                        and mask[ny, nx]
                        and not seen[ny, nx]
                    ):
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        points = np.asarray(pixels, dtype=np.float64)
        if len(points) >= 140:
            groups.append(points)
    return groups


def _centers(mask: np.ndarray, label: str) -> list[tuple[float, float]]:
    groups = _components(mask)
    if len(groups) != 2:
        raise ValueError(f"expected two {label} markers, found {len(groups)}")
    centers: list[tuple[float, float]] = []
    for points in groups:
        height = float(points[:, 0].max() - points[:, 0].min() + 1)
        width = float(points[:, 1].max() - points[:, 1].min() + 1)
        if min(width, height) < 15.0 or max(width, height) > 22.0:
            raise ValueError(f"damaged {label} marker has bounding box {width}x{height}")
        centers.append((float(points[:, 1].mean()), float(points[:, 0].mean())))
    return centers


def _distance(left: tuple[float, float], right: tuple[float, float]) -> float:
    return math.hypot(left[0] - right[0], left[1] - right[1])


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    cyan_distance = ((array[:, :, None, :] - CYAN_PROTOTYPES[None, None, :, :]) ** 2).sum(axis=3).min(axis=2)
    orange_distance = ((array[:, :, None, :] - ORANGE_PROTOTYPES[None, None, :, :]) ** 2).sum(axis=3).min(axis=2)
    white_distance = ((array - WHITE) ** 2).sum(axis=2)
    labels = np.stack((cyan_distance, orange_distance, white_distance), axis=2).argmin(axis=2)
    cyan = _centers(labels == 0, "cyan")
    orange = _centers(labels == 1, "orange")

    within = _distance(cyan[0], cyan[1]) + _distance(orange[0], orange[1])
    cross_a = _distance(cyan[0], orange[0]) + _distance(cyan[1], orange[1])
    cross_b = _distance(cyan[0], orange[1]) + _distance(cyan[1], orange[0])
    cross = min(cross_a, cross_b)
    if abs(within - cross) < 3.0:
        raise ValueError("rasterized matching totals are too close to decide")
    return "WITHIN" if within < cross else "CROSS"
