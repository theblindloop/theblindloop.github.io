"""Independent pixel-only inverse arm for the hinge-hub question.

It receives the final PNG and nothing else: no latent scene, no renderer, no
analytic gold.  It segments the drawn ink, locates the three coloured hubs by
colour alone, and then applies the declared rule directly in image space: it
erases each coloured hub from the ink and counts how many separate pieces of
ink are left.  A hub is the hinge when erasing it leaves more than one piece.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw

_ABSTAIN = "abstain"

# The public question contract names the painted classes: black struts and
# round junctions, plus one red, one green and one blue hub.  The segmentation
# below uses only that, never a renderer constant.
_COLORS = ("red", "green", "blue")
_CHANNEL = {"red": 0, "green": 1, "blue": 2}

_INK_SUM = 700  # a pixel counts as drawn ink when its channels average < ~233
_CHROMA = 45  # neutral strut ink and its antialiased edges stay below this
_MIN_HUB_PIXELS = 240
_ERASE_PAD = 5.0


def _classify(image: Image.Image):
    """Split the drawn ink into neutral struts and the three coloured hubs.

    Neutral ink and its antialiased edges are grey, so they carry almost no
    chroma; hub paint and every blend of hub paint with paper or with strut
    black keeps a strong dominant channel.  Segmenting on chroma first and then
    on the dominant channel therefore needs no exact colour match.
    """

    a = np.asarray(image.convert("RGB")).astype(np.int16)
    ink = a.sum(axis=2) < _INK_SUM
    chroma = a.max(axis=2) - a.min(axis=2)
    dominant = a.argmax(axis=2)  # 0 red, 1 green, 2 blue
    coloured = ink & (chroma > _CHROMA)
    return ink, coloured, dominant


def _mask_image(ink: np.ndarray) -> Image.Image:
    return Image.fromarray((ink.astype(np.uint8) * 255), mode="L")


def _count_pieces(mask: Image.Image, limit: int = 12) -> int:
    """Count 4-connected pieces of drawn ink by repeated flood filling."""

    work = mask.copy()
    pieces = 0
    while pieces <= limit:
        arr = np.asarray(work)
        flat = np.flatnonzero(arr.ravel() == 255)
        if flat.size == 0:
            return pieces
        first = int(flat[0])
        y, x = divmod(first, arr.shape[1])
        ImageDraw.floodfill(work, (x, y), 128, thresh=0)
        pieces += 1
    return limit + 1


def decision_from_image(image: Image.Image) -> str:
    ink, coloured, dominant = _classify(image)
    if int(ink.sum()) < 2000:
        return _ABSTAIN

    hubs = {}
    for color in _COLORS:
        mask = coloured & (dominant == _CHANNEL[color])
        area = int(mask.sum())
        if area < _MIN_HUB_PIXELS:
            return _ABSTAIN
        ys, xs = np.nonzero(mask)
        radius = float(np.sqrt(area / np.pi))
        hubs[color] = (float(xs.mean()), float(ys.mean()), radius)

    base = _mask_image(ink)
    if _count_pieces(base) != 1:
        return _ABSTAIN  # the drawing is not one connected network

    work = base.copy()
    painter = ImageDraw.Draw(work)
    for color in _COLORS:
        cx, cy, radius = hubs[color]
        r = radius + _ERASE_PAD
        painter.ellipse([cx - r, cy - r, cx + r, cy + r], fill=0)

    pieces = _count_pieces(work)
    if pieces not in (4, 5, 6):
        return _ABSTAIN
    return str(pieces)
