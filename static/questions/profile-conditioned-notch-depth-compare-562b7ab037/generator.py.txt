"""Renderer and analytic contract for the paired inward-notch figure."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "notch_depth_compare-0.1.0"
WIDTH = 512
HEIGHT = 340
PANEL_TOP = 62
PANEL_BOTTOM = 316
PANEL_BOXES = ((24, PANEL_TOP, 248, PANEL_BOTTOM), (264, PANEL_TOP, 488, PANEL_BOTTOM))
QUARANTINE_GAP_PX = 12.0
SUPERSAMPLE = 4

# All pixels belonging to the two silhouettes share one color within a scene.
# The palette changes by scene, while every entry is strongly chromatic so the
# independent inverse can separate it from the nearly-white panel background.
FILL_PALETTE = (
    (38, 104, 164),
    (116, 68, 158),
    (35, 132, 103),
    (184, 84, 48),
)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced pair of equal-envelope, equal-notch-area silhouettes."""

    rng = random.Random(int(seed))
    shallow = rng.uniform(28.0, 42.0)
    if int(seed) % 17 == 0:
        separation = rng.uniform(4.5, 9.0)
    else:
        separation = rng.uniform(17.0, 32.0)
    deep = shallow + separation

    # Alternating seed parity keeps the answer identity counterbalanced even
    # for short deterministic stress runs.
    if int(seed) % 2 == 0:
        depth_a, depth_b = deep, shallow
    else:
        depth_a, depth_b = shallow, deep

    return {
        "depth_a": round(depth_a, 6),
        "depth_b": round(depth_b, 6),
        "notch_area": round(rng.uniform(1100.0, 1450.0), 6),
        "fill_color": list(rng.choice(FILL_PALETTE)),
        "vertex_phase": rng.randrange(2),
        "width": WIDTH,
        "height": HEIGHT,
    }


def _shape_points(
    panel_left: int,
    depth: float,
    notch_area: float,
    vertex_phase: int,
) -> list[tuple[float, float]]:
    """Return one simple polygon with a single top-edge V-notch.

    The notch triangle has area ``notch_area`` in both panels, so changing its
    depth changes its width inversely and does not expose a filled-area cue.
    """

    center_x = panel_left + 112.0
    base_y = 78.0
    shoulder_y = base_y + 48.0
    half_notch = notch_area / max(depth, 1e-6)
    points = [
        (panel_left + 44.0, shoulder_y),
        (center_x - half_notch, shoulder_y),
        (center_x, shoulder_y + depth),
        (center_x + half_notch, shoulder_y),
        (panel_left + 180.0, shoulder_y),
        (panel_left + 198.0, base_y + 66.0),
        (panel_left + 192.0, base_y + 176.0),
        (panel_left + 170.0, base_y + 194.0),
        (panel_left + 54.0, base_y + 194.0),
        (panel_left + 30.0, base_y + 176.0),
        (panel_left + 26.0, base_y + 66.0),
    ]
    if int(vertex_phase) % 2:
        # A cyclic start-point change is raster-invariant but exercises a
        # genuine latent alias rather than a cosmetic metadata field.
        points = points[3:] + points[:3]
    return points


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the two silhouettes to one deterministic RGB PNG."""

    width = int(scene["width"])
    height = int(scene["height"])
    scale = SUPERSAMPLE
    image = Image.new("RGB", (width * scale, height * scale), (255, 255, 255))
    draw = ImageDraw.Draw(image)

    panel_fill = (248, 250, 252)
    panel_edge = (190, 198, 207)
    fill = tuple(int(value) for value in scene["fill_color"])
    notch_area = float(scene["notch_area"])
    phase = int(scene["vertex_phase"])
    for panel_index, depth in enumerate(
        (float(scene["depth_a"]), float(scene["depth_b"]))
    ):
        x0, y0, x1, y1 = PANEL_BOXES[panel_index]
        draw.rounded_rectangle(
            (x0 * scale, y0 * scale, x1 * scale, y1 * scale),
            radius=8 * scale,
            fill=panel_fill,
            outline=panel_edge,
            width=2 * scale,
        )
        panel_left = PANEL_BOXES[panel_index][0]
        points = _shape_points(panel_left, depth, notch_area, phase)
        scaled = [(round(x * scale), round(y * scale)) for x, y in points]
        draw.polygon(scaled, fill=fill)

    image = image.resize((width, height), Image.Resampling.LANCZOS)
    label_draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    label_draw.text((PANEL_BOXES[0][0] + 108, 26), "A", fill=(28, 39, 52), font=font)
    label_draw.text((PANEL_BOXES[1][0] + 108, 26), "B", fill=(28, 39, 52), font=font)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the panel with the larger analytic shoulder-to-tip depth."""

    depth_a = float(scene["depth_a"])
    depth_b = float(scene["depth_b"])
    if math.isclose(depth_a, depth_b, abs_tol=1e-9):
        raise ValueError("an exact depth tie has no unique selection")
    return "A" if depth_a > depth_b else "B"


def margin(scene: dict[str, Any]) -> float:
    """Runner-up margin in rendered-pixel units before rasterization."""

    return abs(float(scene["depth_a"]) - float(scene["depth_b"]))


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Exclude near ties whose raster decision margin is too small."""

    depth_a = float(scene["depth_a"])
    depth_b = float(scene["depth_b"])
    return min(depth_a, depth_b) < 24.0 or margin(scene) < QUARANTINE_GAP_PX


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """Exercise the render-invariant cyclic vertex-start alias."""

    twin = dict(scene)
    twin["vertex_phase"] = 1 - int(scene["vertex_phase"])
    return [("cyclic_vertex_start", twin)]
