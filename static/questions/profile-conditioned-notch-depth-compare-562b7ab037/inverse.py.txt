"""Independent image-only inverse for the inward-notch figure.

The inverse knows only the public two-panel layout. It segments the chromatic
silhouette inside each panel, finds the connected foreground component, and
measures the top-boundary profile. Shoulder bands outside the possible notch
provide a baseline; the deepest top-profile row in the central band is the tip.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


_BOXES = ((24, 62, 248, 316), (264, 62, 488, 316))
_PAD = 10
_MIN_COMPONENT = 1000


def _largest_component(mask: np.ndarray) -> np.ndarray:
    """Return the largest 8-connected component in a small boolean mask."""

    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    best: list[tuple[int, int]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue: deque[tuple[int, int]] = deque([(int(y), int(x))])
        seen[y, x] = True
        component: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            component.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dy or dx):
                        continue
                    ny, nx = cy + dy, cx + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        if len(component) > len(best):
            best = component
    if len(best) < _MIN_COMPONENT:
        raise ValueError("no sufficiently large silhouette component")
    output = np.zeros(mask.shape, dtype=bool)
    ys, xs = zip(*best)
    output[np.asarray(ys), np.asarray(xs)] = True
    return output


def _measure_depth(image_array: np.ndarray, box: tuple[int, int, int, int]) -> float:
    x0, y0, x1, y1 = box
    crop = image_array[y0 + _PAD : y1 - _PAD, x0 + _PAD : x1 - _PAD]
    channel_spread = crop.max(axis=2) - crop.min(axis=2)
    # The panel background and border are nearly neutral; the filled shape is
    # deliberately chromatic. Edge antialias pixels remain attached to it.
    mask = (channel_spread > 24) & (crop.mean(axis=2) < 235)
    component = _largest_component(mask)
    top_by_x = np.full(component.shape[1], -1, dtype=np.int32)
    for x in range(component.shape[1]):
        ys = np.flatnonzero(component[:, x])
        if ys.size:
            top_by_x[x] = int(ys[0])

    center = component.shape[1] // 2
    # The maximum possible notch half-width is just below 52 px. These bands
    # are therefore guaranteed to lie on the two visible shoulder flats.
    shoulder_values = []
    for start, stop in ((center - 66, center - 58), (center + 58, center + 66)):
        shoulder_values.extend(int(value) for value in top_by_x[start:stop] if value >= 0)
    if len(shoulder_values) < 8:
        raise ValueError("shoulder profile is not recoverable")
    baseline = float(np.median(np.asarray(shoulder_values)))

    tip_values = [
        int(value)
        for value in top_by_x[center - 53 : center + 54]
        if value >= 0
    ]
    if len(tip_values) < 20:
        raise ValueError("notch tip profile is not recoverable")
    tip = float(max(tip_values))
    depth = tip - baseline
    if depth <= 8.0:
        raise ValueError("measured notch is too shallow for a certified raster")
    return depth


def decision_from_image(image: Image.Image) -> str:
    """Select A or B from the final PNG alone."""

    image_array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    depth_a = _measure_depth(image_array, _BOXES[0])
    depth_b = _measure_depth(image_array, _BOXES[1])
    if abs(depth_a - depth_b) < 1.0:
        raise ValueError("pixel depths are indistinguishable")
    return "A" if depth_a > depth_b else "B"
