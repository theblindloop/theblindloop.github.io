"""Independent pixel-only inverse for shoreline boundary-component identity."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_BACKGROUND = np.array((248, 247, 242), dtype=np.int16)
_NAVY = np.array((25, 52, 73), dtype=np.int16)
_MARKERS = {
    "blue": np.array((31, 119, 193), dtype=np.int16),
    "red": np.array((210, 62, 57), dtype=np.int16),
    "gold": np.array((225, 164, 36), dtype=np.int16),
}


class OracleAbstention(ValueError):
    pass


def _near(array: np.ndarray, color: np.ndarray, tolerance: float = 12.0) -> np.ndarray:
    delta = array.astype(np.int32) - color.astype(np.int32)
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _components(mask: np.ndarray) -> tuple[np.ndarray, list[dict]]:
    height, width = mask.shape
    labels = np.full((height, width), -1, dtype=np.int16)
    records: list[dict] = []
    for start_y, start_x in np.argwhere(mask):
        sy, sx = int(start_y), int(start_x)
        if labels[sy, sx] >= 0:
            continue
        label = len(records)
        labels[sy, sx] = label
        queue = deque([(sy, sx)])
        count = 0
        touches_border = False
        while queue:
            y, x = queue.popleft()
            count += 1
            touches_border = touches_border or y == 0 or x == 0 or y == height - 1 or x == width - 1
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and labels[ny, nx] < 0:
                    labels[ny, nx] = label
                    queue.append((ny, nx))
        records.append({"area": count, "outer": touches_border})
    return labels, records


def _marker_centers(array: np.ndarray) -> dict[str, tuple[float, float]]:
    centers = {}
    for name, color in _MARKERS.items():
        ys, xs = np.nonzero(_near(array, color))
        if not 65 <= len(xs) <= 105:
            raise OracleAbstention(f"{name} bead is missing or malformed")
        if float(xs.max() - xs.min()) > 11 or float(ys.max() - ys.min()) > 11:
            raise OracleAbstention(f"{name} bead is not one compact component")
        centers[name] = (float(xs.mean()), float(ys.mean()))
    return centers


def _attachment(
    center: tuple[float, float],
    labels: np.ndarray,
    records: list[dict],
    offset: tuple[int, int],
) -> str:
    cx = center[0] - offset[0]
    cy = center[1] - offset[1]
    yy, xx = np.indices(labels.shape)
    neighborhood = (xx - cx) ** 2 + (yy - cy) ** 2 <= 13.0**2
    values, counts = np.unique(labels[neighborhood & (labels >= 0)], return_counts=True)
    ranked = sorted(zip(counts.tolist(), values.tolist()), reverse=True)
    if not ranked or ranked[0][0] < 18:
        raise OracleAbstention("a bead is not visibly attached to water")
    if len(ranked) > 1 and ranked[0][0] - ranked[1][0] < 8:
        raise OracleAbstention("a bead has ambiguous shoreline attachment")
    return "outer" if records[ranked[0][1]]["outer"] else "lake"


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[2] != 3:
        raise OracleAbstention("expected an RGB raster")
    navy = _near(array, _NAVY)
    ys, xs = np.nonzero(navy)
    if not 4200 <= len(xs) <= 5000:
        raise OracleAbstention("compact island ink area is outside the supported range")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    width, height = x1 - x0 + 1, y1 - y0 + 1
    if not (78 <= width <= 88 and 78 <= height <= 88):
        raise OracleAbstention("island extent is missing or corrupted")

    pad = 15
    left, right = max(0, x0 - pad), min(array.shape[1], x1 + pad + 1)
    top, bottom = max(0, y0 - pad), min(array.shape[0], y1 + pad + 1)
    crop = array[top:bottom, left:right]
    water = _near(crop, _BACKGROUND)
    labels, records = _components(water)
    outer = [index for index, record in enumerate(records) if record["outer"] and record["area"] >= 100]
    lakes = [index for index, record in enumerate(records) if not record["outer"] and record["area"] >= 250]
    if len(outer) != 1 or len(lakes) != 1:
        raise OracleAbstention("shoreline topology is not exactly one outer water and one lake")

    centers = _marker_centers(array)
    attachments = {
        name: _attachment(center, labels, records, (left, top))
        for name, center in centers.items()
    }
    matches = [name for name in ("red", "gold") if attachments[name] == attachments["blue"]]
    if len(matches) != 1:
        raise OracleAbstention("the blue shoreline has no unique colored mate")
    return matches[0]
