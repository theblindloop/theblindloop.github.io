"""Deterministic latent sampler, analytic arm, and raster renderer."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = [248, 247, 242]
ISLAND = [25, 52, 73]
MARKER_COLORS = {
    "blue": (31, 119, 193),
    "red": (210, 62, 57),
    "gold": (225, 164, 36),
}
DECOY = (205, 207, 201)
QUARANTINE_PX = 10.0


def _point(center: tuple[float, float], radius: float, angle: float) -> list[int]:
    return [
        int(round(center[0] + radius * math.cos(angle))),
        int(round(center[1] + radius * math.sin(angle))),
    ]


def _contour(
    center: tuple[float, float],
    count: int,
    base_radius: float,
    rotation: float,
    phase: float,
) -> list[list[int]]:
    points = []
    for index in range(count):
        angle = rotation + 2.0 * math.pi * index / count
        radius = base_radius + 1.7 * math.sin(3.0 * angle + phase) + 0.9 * math.sin(
            5.0 * angle - 0.7 * phase
        )
        points.append(_point(center, radius, angle))
    return points


def _far_decoys(rng: random.Random, center: tuple[int, int]) -> list[list[int]]:
    decoys: list[list[int]] = []
    while len(decoys) < 9:
        x = rng.randint(35, WIDTH - 36)
        y = rng.randint(35, HEIGHT - 36)
        if math.hypot(x - center[0], y - center[1]) < 105:
            continue
        angle = rng.uniform(0, 2 * math.pi)
        length = rng.randint(10, 18)
        dx = int(round(length * math.cos(angle)))
        dy = int(round(length * math.sin(angle)))
        decoys.append([x - dx, y - dy, x + dx, y + dy])
    return decoys


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    centers = (160, 224, 288, 352)
    center = (rng.choice(centers), rng.choice(centers))
    rotation = rng.uniform(0.0, 2.0 * math.pi)
    outer = _contour(center, 20, 42.0, rotation, rng.uniform(0.0, 2.0 * math.pi))
    hole_center = (
        center[0] + rng.randint(-4, 4),
        center[1] + rng.randint(-4, 4),
    )
    hole = _contour(
        hole_center,
        16,
        16.5,
        -0.63 * rotation + 0.31,
        rng.uniform(0.0, 2.0 * math.pi),
    )

    blue_on_outer = bool(rng.randrange(2))
    matching_color = "red" if int(seed) % 2 == 0 else "gold"
    other_color = "gold" if matching_color == "red" else "red"
    outer_start = rng.randrange(len(outer))
    outer_far = (outer_start + len(outer) // 2 + rng.choice((-1, 0, 1))) % len(outer)
    hole_start = rng.randrange(len(hole))
    hole_far = (hole_start + len(hole) // 2 + rng.choice((-1, 0, 1))) % len(hole)
    if blue_on_outer:
        markers = [
            {"color": "blue", "xy": outer[outer_start]},
            {"color": matching_color, "xy": outer[outer_far]},
            {"color": other_color, "xy": hole[hole_start]},
        ]
    else:
        markers = [
            {"color": "blue", "xy": hole[hole_start]},
            {"color": matching_color, "xy": hole[hole_far]},
            {"color": other_color, "xy": outer[outer_start]},
        ]
    rng.shuffle(markers)
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
        "island_color": list(ISLAND),
        "outer": outer,
        "hole": hole,
        "markers": markers,
        "decoys": _far_decoys(rng, center),
    }


def _segments(points: list[list[int]]) -> Iterable[tuple[list[int], list[int]]]:
    for index, point in enumerate(points):
        yield point, points[(index + 1) % len(points)]


def _point_segment_distance(point: list[int], a: list[int], b: list[int]) -> float:
    px, py = point
    ax, ay = a
    bx, by = b
    dx, dy = bx - ax, by - ay
    denominator = dx * dx + dy * dy
    if denominator == 0:
        return math.hypot(px - ax, py - ay)
    t = max(0.0, min(1.0, ((px - ax) * dx + (py - ay) * dy) / denominator))
    return math.hypot(px - (ax + t * dx), py - (ay + t * dy))


def _contour_distance(point: list[int], contour: list[list[int]]) -> float:
    return min(_point_segment_distance(point, a, b) for a, b in _segments(contour))


def _nearest_loop(outer: list[list[int]], hole: list[list[int]], point: list[int]) -> int:
    outer_distance = _contour_distance(point, outer)
    hole_distance = _contour_distance(point, hole)
    return 0 if outer_distance < hole_distance else 1


def analytic_gold(scene: dict) -> str:
    markers = scene["markers"]
    blue = next(marker for marker in markers if marker["color"] == "blue")
    blue_loop = _nearest_loop(scene["outer"], scene["hole"], blue["xy"])
    matches = [
        marker["color"]
        for marker in markers
        if marker["color"] != "blue"
        and _nearest_loop(scene["outer"], scene["hole"], marker["xy"]) == blue_loop
    ]
    if len(matches) != 1:
        raise ValueError("scene does not expose exactly one shoreline mate")
    return matches[0]


def margin(scene: dict) -> float:
    values = []
    for marker in scene["markers"]:
        point = marker["xy"]
        outer_distance = _contour_distance(point, scene["outer"])
        hole_distance = _contour_distance(point, scene["hole"])
        values.append(abs(outer_distance - hole_distance))
    return float(min(values))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    image = Image.new(
        "RGB",
        (int(scene["width"]), int(scene["height"])),
        tuple(scene["background"]),
    )
    draw = ImageDraw.Draw(image)
    for x0, y0, x1, y1 in scene["decoys"]:
        draw.line((x0, y0, x1, y1), fill=DECOY, width=3)
        draw.ellipse((x0 - 2, y0 - 2, x0 + 2, y0 + 2), fill=DECOY)
    draw.polygon([tuple(point) for point in scene["outer"]], fill=tuple(scene["island_color"]))
    draw.polygon([tuple(point) for point in scene["hole"]], fill=tuple(scene["background"]))
    for marker in scene["markers"]:
        x, y = marker["xy"]
        draw.ellipse((x - 7, y - 7, x + 7, y + 7), fill=tuple(scene["background"]))
        draw.ellipse((x - 5, y - 5, x + 5, y + 5), fill=MARKER_COLORS[marker["color"]])
        draw.ellipse((x - 1, y - 1, x + 1, y + 1), fill=(255, 255, 255))
    return image
