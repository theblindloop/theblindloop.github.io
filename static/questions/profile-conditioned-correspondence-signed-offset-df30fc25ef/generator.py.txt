"""Signed-offset rail correspondence renderer (paper-semantic family).

A cross-panel signed-offset correspondence task. The top panel shows one
neutral gray horizontal "rail" (a straight line) with a small dark "rider" dot
drawn a fixed perpendicular distance away from the rail, either above or below
it. Below a light divider sit three colored rails (red / green / blue), each
with its own dark rider dot offset from its rail by a specific signed offset
(side and magnitude). Exactly one candidate's dot has the same signed offset as
the reference rider dot. The question asks for that candidate's color.

The pixel-only inverse arm must, for every panel, recover the rail as a visible
axis line, locate its rider dot, measure the signed perpendicular offset (side
and magnitude), establish the cross-panel correspondence by signed-offset
equality, and only then read out the bound color of the single matched rail.

Latent state z (every field that influences the gold is rendered -- nothing
hidden):

    ref_side   : which side ('above' or 'below') the reference rider dot is on.
    ref_mag    : perpendicular distance (px) of the reference rider dot.
    cand_sides : length-3 list of candidate rider sides, one per colored rail.
    cand_mags  : length-3 list of candidate rider magnitudes.
    colors     : the fixed 3-color palette bound one-to-one to the rails.
    (all geometry -- rail rows, column span, thickness, dot radius, canvas --
     are fixed constants so only side+magnitude, all rendered, decide gold.)

The analytic gold is the color of the candidate rail whose rider dot has the
same signed offset (side and magnitude) as the reference rider dot. Because the
reference side, reference magnitude, each candidate side, and each candidate
magnitude are all rendered verbatim, the gold is recoverable from pixels alone.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "correspondence_signed_offset-0.1.0"

BACKGROUND = (255, 255, 255)
DIVIDER = (235, 235, 235)
GRAY_RAIL = (150, 150, 150)
RIDER = (25, 25, 25)

PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (235, 40, 40),
    "green": (0, 170, 70),
    "blue": (40, 60, 235),
}
COLOR_ORDER = ["red", "green", "blue"]

# Canonical perpendicular magnitudes (px) a rider dot may be offset from its
# rail. Adjacent values are separated by >= 15 px, far above pixel tolerance.
CANONICAL_MAGS = [35, 52, 69, 86]

WIDTH = 540
HEIGHT = 900
DIVIDER_Y = 210
DIVIDER_THICK = 2
RAIL_X0 = 60
RAIL_X1 = 480
RAIL_THICK = 8
REF_RAIL_Y = 110
CAND_RAIL_YS = [360, 560, 760]
DOT_R = 11
DOT_CX = 270

_QUARANTINE_FLOOR = 12.0  # px


def sample_scene(seed: int) -> dict:
    """Sample a balanced, deterministic latent scene (dict)."""
    rng = random.Random(seed)
    ref_mag = rng.choice(CANONICAL_MAGS)
    ref_side = rng.choice(["above", "below"])
    pool = [(side, mag) for side in ("above", "below") for mag in CANONICAL_MAGS]
    pool.remove((ref_side, ref_mag))
    rng.shuffle(pool)
    distractors = pool[:2]
    pairings = [(ref_side, ref_mag)] + distractors
    rng.shuffle(pairings)
    return {
        "ref_side": ref_side,
        "ref_mag": ref_mag,
        "cand_sides": [p[0] for p in pairings],
        "cand_mags": [p[1] for p in pairings],
        "colors": list(COLOR_ORDER),
    }


def _draw_rail(draw, y: int, color) -> None:
    draw.rectangle([RAIL_X0, y - RAIL_THICK // 2, RAIL_X1, y + RAIL_THICK // 2], fill=color)


def _draw_dot(draw, y: int) -> None:
    draw.ellipse([DOT_CX - DOT_R, y - DOT_R, DOT_CX + DOT_R, y + DOT_R], fill=RIDER)


def render(scene: dict) -> Image.Image:
    """Render I = R(z) deterministically at pixel resolution."""
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(img)
    _draw_rail(draw, REF_RAIL_Y, GRAY_RAIL)
    ref_doty = REF_RAIL_Y - scene["ref_mag"] if scene["ref_side"] == "above" else REF_RAIL_Y + scene["ref_mag"]
    _draw_dot(draw, ref_doty)
    for i, y in enumerate(CAND_RAIL_YS):
        color = scene["colors"][i]
        _draw_rail(draw, y, PALETTE[color])
        doty = y - scene["cand_mags"][i] if scene["cand_sides"][i] == "above" else y + scene["cand_mags"][i]
        _draw_dot(draw, doty)
    draw.rectangle([0, DIVIDER_Y, WIDTH, DIVIDER_Y + DIVIDER_THICK], fill=DIVIDER)
    return img


def _signed_pair(side: str, mag: float) -> float:
    """Map a (side, magnitude) pair to a signed scalar so distance is a scalar."""
    return mag if side == "above" else -mag


def analytic_gold(scene: dict) -> str:
    """Decision: color (red/green/blue) of the rail whose rider has the same
    signed offset as the reference rider dot."""
    for i in range(len(scene["cand_sides"])):
        if (
            scene["cand_sides"][i] == scene["ref_side"]
            and scene["cand_mags"][i] == scene["ref_mag"]
        ):
            return scene["colors"][i]
    raise ValueError("no candidate matched the reference signed offset")


def margin(scene: dict) -> float:
    """Runner-up margin: minimum absolute signed-offset difference (px) between
    the reference rider and any distractor rider."""
    ref = _signed_pair(scene["ref_side"], float(scene["ref_mag"]))
    diffs = []
    for side, mag in zip(scene["cand_sides"], scene["cand_mags"]):
        cand = _signed_pair(side, float(mag))
        if cand != ref:
            diffs.append(abs(cand - ref))
    return float(min(diffs)) if diffs else 0.0


def is_quarantined(scene: dict) -> bool:
    """Quarantine an ambiguous scene where a distractor rider comes within
    _QUARANTINE_FLOOR px of the reference signed offset (never produced because
    canonical magnitudes are separated by >= 15 px and sides are discrete)."""
    return margin(scene) < _QUARANTINE_FLOOR


def latent_symmetries(scene: dict) -> list:
    """not_applicable: every label-influencing field (reference side, reference
    magnitude, each candidate side and magnitude, and the bound colors) is
    rendered verbatim, so the raster uniquely determines the gold; no
    pixel-identical transform exists."""
    return []
