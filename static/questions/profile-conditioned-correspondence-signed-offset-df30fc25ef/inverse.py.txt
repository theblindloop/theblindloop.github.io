"""Independent pixel-only inverse arm for the signed-offset rail correspondence.

Receives ONLY the final rasterized PNG and the public question contract (this
file's own visual spec). It never imports the renderer, prompts, generator,
verifier, or analytic-gold code, and never reads scene data.

Public visual spec:

  * the top panel holds one neutral gray horizontal rail (the reference axis)
    with a small dark "rider" dot offset from it by a fixed perpendicular
    distance, either above or below the rail;
  * a thin light divider separates it from a column of three colored rails,
    one red, one green, one blue, each with its own dark rider dot offset from
    its rail by a specific side and distance;
  * the task: decide which colored rail's rider dot is at the SAME signed
    offset (same side and same distance) from its rail as the gray reference
    rider dot.

The oracle, from pixels alone: finds the dark rider dots (small dark blobs),
finds the four horizontal rails by their colors (one gray, three colored), and
for each rail measures the signed perpendicular offset of the rider dot nearest
to it (vertical pixel distance, sign by above/below). It returns the color of
the candidate rail whose rider's signed offset equals the reference's within a
pixel tolerance.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "correspondence_signed_offset-oracle-0.1.0"

PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (235, 40, 40),
    "green": (0, 170, 70),
    "blue": (40, 60, 235),
}
GRAY = (150, 150, 150)
DIVIDER_Y = 210
RAIL_X0 = 60
RAIL_X1 = 480
_DARK = 110.0
_COLOR_DIST = 45.0
_TOL = 6.0


def _classify(arr: np.ndarray, color: tuple[int, int, int]) -> np.ndarray:
    c = np.asarray(color, dtype=np.float32)
    d = np.linalg.norm(arr - c[None, None, :], axis=-1)
    return d < _COLOR_DIST


def _label_dots(arr: np.ndarray) -> list[tuple[float, float]]:
    """Return centroids (y, x) of small dark blobs (the rider dots)."""
    dark = arr.max(axis=2) < _DARK
    h, w = dark.shape
    labels = np.zeros((h, w), dtype=np.int32)
    comps: dict[int, list] = {}
    cur = 0
    for i in range(h):
        for j in range(w):
            if not dark[i, j] or labels[i, j]:
                continue
            cur += 1
            labels[i, j] = cur
            stack = [(i, j)]
            rows = []
            cols = []
            while stack:
                r, c = stack.pop()
                rows.append(r)
                cols.append(c)
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < h and 0 <= nc < w and dark[nr, nc] and labels[nr, nc] == 0:
                        labels[nr, nc] = cur
                        stack.append((nr, nc))
            rmin, rmax = min(rows), max(rows)
            cmin, cmax = min(cols), max(cols)
            area = len(rows)
            if 120 < area < 600 and (rmax - rmin) < 40 and (cmax - cmin) < 40:
                comps[cur] = (float(np.mean(rows)), float(np.mean(cols)))
    return list(comps.values())


def _rail_row(mask: np.ndarray, min_col: int, max_col: int) -> float:
    """Row centroid of one horizontal rail (the modal row of colored pixels
    within the rail's horizontal span)."""
    cols = np.arange(mask.shape[1])
    center = (min_col + max_col) // 2
    sub = mask[:, center - 40 : center + 40]
    ys = np.where(sub.sum(axis=1) > 0)[0]
    if ys.size == 0:
        for x in range(center - 60, center + 60):
            m = mask[:, x]
            ys = np.where(m > 0)[0]
            if ys.size:
                break
    if ys.size == 0:
        raise ValueError("rail not found")
    return float(np.mean(ys))


def _find_rails(arr: np.ndarray) -> dict:
    gray_mask = _classify(arr, GRAY)
    # exactly the gray reference rail: take rows with a wide run of gray pixels
    gray_rows = np.where(gray_mask.sum(axis=1) > 0.5 * (RAIL_X1 - RAIL_X0))[0]
    if gray_rows.size == 0:
        raise ValueError("no gray reference rail found")
    ref_y = float(np.mean(gray_rows))
    rails = []
    for name, rgb in PALETTE.items():
        mask = _classify(arr, rgb)
        rows = np.where(mask.sum(axis=1) > 0.4 * (RAIL_X1 - RAIL_X0))[0]
        if rows.size == 0:
            raise ValueError(f"no {name} rail found")
        rails.append((name, float(np.mean(rows))))
    return {"ref_y": ref_y, "cands": rails}


def _signed_offset(dot_y: float, rail_y: float) -> float:
    return dot_y - rail_y


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.float32)
    dots = _label_dots(arr)
    if len(dots) != 4:
        raise ValueError(f"expected 4 rider dots, found {len(dots)}")
    rails = _find_rails(arr)

    def nearest_dot(target_y: float) -> float:
        return min(dots, key=lambda d: abs(d[0] - target_y))[0]

    ref_off = _signed_offset(nearest_dot(rails["ref_y"]), rails["ref_y"])

    matches = []
    for name, rail_y in rails["cands"]:
        dot_y = nearest_dot(rail_y)
        off = _signed_offset(dot_y, rail_y)
        if abs(off - ref_off) <= _TOL:
            matches.append(name)
    if len(matches) == 1:
        return matches[0]
    if len(matches) == 0:
        raise ValueError("no candidate rail matched the reference signed offset")
    raise ValueError(f"ambiguous: multiple candidates matched ({matches})")


if __name__ == "__main__":
    import sys

    img = Image.open(sys.argv[1]).convert("RGB")
    print(decision_from_image(img))
