"""Deterministic scene sampler, analytic arm, and Pillow renderer."""

from __future__ import annotations

import itertools
import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 700
COLOR_NAMES = ("red", "blue", "gold", "teal", "purple")
ANSWER_NAMES = ("red", "blue", "gold")
COLORS = {
    "red": (218, 65, 68),
    "blue": (55, 112, 204),
    "gold": (229, 164, 35),
    "teal": (40, 157, 147),
    "purple": (143, 88, 181),
}
LEFT_CENTER = (245.0, 417.0)
RIGHT_CENTER = (715.0, 417.0)
MARKER = (26, 40, 63)


def _font(size: int, bold: bool = False):
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _similarity_residual(source, target) -> float:
    """RMS error of the best translation/rotation/uniform-scale fit (no reflection)."""
    sx = sum(p[0] for p in source) / len(source)
    sy = sum(p[1] for p in source) / len(source)
    tx = sum(p[0] for p in target) / len(target)
    ty = sum(p[1] for p in target) / len(target)
    a = [complex(p[0] - sx, p[1] - sy) for p in source]
    b = [complex(p[0] - tx, p[1] - ty) for p in target]
    denom = sum(abs(z) ** 2 for z in a)
    coefficient = sum(z.conjugate() * w for z, w in zip(a, b)) / denom
    return math.sqrt(sum(abs(coefficient * z - w) ** 2 for z, w in zip(a, b)) / len(a))


def _runner_residual(points) -> float:
    scores = []
    for perm in itertools.permutations(range(5)):
        if perm == tuple(range(5)):
            continue
        scores.append(_similarity_residual(points, [points[i] for i in perm]))
    return min(scores)


def _sample_points(rng: random.Random) -> list[list[float]]:
    for _ in range(10000):
        raw = [(rng.uniform(-0.9, 0.9), rng.uniform(-0.9, 0.9)) for _ in range(5)]
        cx = sum(x for x, _ in raw) / 5
        cy = sum(y for _, y in raw) / 5
        centered = [(x - cx, y - cy) for x, y in raw]
        radius = max(math.hypot(x, y) for x, y in centered)
        points = [(x / radius, y / radius) for x, y in centered]
        separation = min(
            math.hypot(points[i][0] - points[j][0], points[i][1] - points[j][1])
            for i in range(5) for j in range(i + 1, 5)
        )
        if separation >= 0.34 and _runner_residual(points) >= 0.17:
            return [[round(x, 5), round(y, 5)] for x, y in points]
    raise RuntimeError("could not sample a well-separated asymmetric constellation")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    points = _sample_points(rng)
    color_order = list(COLOR_NAMES)
    rng.shuffle(color_order)
    desired = ANSWER_NAMES[int(seed) % 3]
    return {
        "points": points,
        "color_order": color_order,
        "target_index": color_order.index(desired),
        "left_angle": rng.uniform(-math.pi, math.pi),
        "right_angle": rng.uniform(-math.pi, math.pi),
        "left_scale": rng.uniform(140.0, 161.0),
        "right_scale": rng.uniform(132.0, 162.0),
    }


def analytic_gold(scene: dict) -> str:
    answer = str(scene["color_order"][int(scene["target_index"])])
    if answer not in ANSWER_NAMES:
        raise ValueError("queried landmark must carry an answer color")
    return answer


def margin(scene: dict) -> float:
    return float(_runner_residual(scene["points"]) * float(scene["right_scale"]))


def is_quarantined(scene: dict) -> bool:
    points = scene["points"]
    return (
        len(points) != 5
        or len(set(scene["color_order"])) != 5
        or not 0 <= int(scene["target_index"]) < 5
        or margin(scene) < 8.0
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _transform(points, center, scale: float, angle: float):
    cosine, sine = math.cos(angle), math.sin(angle)
    output = []
    for x, y in points:
        rx = cosine * x - sine * y
        ry = sine * x + cosine * y
        output.append((int(round(center[0] + scale * rx)), int(round(center[1] + scale * ry))))
    return output


def _center_text(draw, xy, text, font, fill):
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2), text, font=font, fill=fill)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (246, 244, 238))
    draw = ImageDraw.Draw(image)
    _center_text(draw, (WIDTH // 2, 31), "TURNED CONSTELLATION", _font(27, True), (35, 43, 55))
    _center_text(draw, (WIDTH // 2, 70), "Same five-dot pattern • turn + uniform resize • no reflection", _font(17), (75, 83, 94))
    left_box = (30, 120, 460, 650)
    right_box = (500, 120, 930, 650)
    for box in (left_box, right_box):
        draw.rounded_rectangle(box, radius=18, fill=(255, 255, 252), outline=(174, 181, 190), width=4)
    _center_text(draw, (245, 151), "COLOR MAP", _font(19, True), (72, 80, 92))
    _center_text(draw, (715, 151), "TURNED COPY", _font(19, True), (72, 80, 92))
    draw.line((480, 165, 480, 620), fill=(185, 190, 197), width=3)
    left = _transform(scene["points"], LEFT_CENTER, float(scene["left_scale"]), float(scene["left_angle"]))
    right = _transform(scene["points"], RIGHT_CENTER, float(scene["right_scale"]), float(scene["right_angle"]))
    for index, (x, y) in enumerate(left):
        color = COLORS[str(scene["color_order"][index])]
        draw.ellipse((x - 12, y - 12, x + 12, y + 12), fill=color, outline=(255, 255, 255), width=2)
    target = int(scene["target_index"])
    for index, (x, y) in enumerate(right):
        if index == target:
            draw.ellipse((x - 16, y - 16, x + 16, y + 16), fill=(255, 255, 252), outline=MARKER, width=6)
        else:
            draw.ellipse((x - 10, y - 10, x + 10, y + 10), fill=MARKER)
    _center_text(draw, (WIDTH // 2, 676), "Which left color matches the ringed dot?   red • blue • gold", _font(17, True), (46, 53, 64))
    return image
