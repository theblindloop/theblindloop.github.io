"""Independent pixel-only inverse arm for landmark correspondence."""

from __future__ import annotations

from collections import deque
import itertools
import math

import numpy as np
from PIL import Image

PALETTE = (
    ("red", (218, 65, 68)),
    ("blue", (55, 112, 204)),
    ("gold", (229, 164, 35)),
    ("teal", (40, 157, 147)),
    ("purple", (143, 88, 181)),
)
MARKER = (26, 40, 63)


def _mask(arr: np.ndarray, rgb, tolerance: float = 8.0) -> np.ndarray:
    delta = arr.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _centroid(mask: np.ndarray) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 150:
        raise ValueError("missing colored landmark")
    return float(xs.mean()), float(ys.mean())


def _components(mask: np.ndarray):
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points = []
        while queue:
            py, px = queue.popleft()
            points.append((py, px))
            for ny, nx in ((py - 1, px), (py + 1, px), (py, px - 1), (py, px + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= 150:
            data = np.asarray(points)
            x0, x1 = int(data[:, 1].min()), int(data[:, 1].max())
            y0, y1 = int(data[:, 0].min()), int(data[:, 0].max())
            if 17 <= x1 - x0 <= 36 and 17 <= y1 - y0 <= 36:
                found.append({
                    "center": (float(data[:, 1].mean()), float(data[:, 0].mean())),
                    "width": x1 - x0 + 1,
                    "height": y1 - y0 + 1,
                    "area": len(points),
                })
    return found


def _fit(source, target) -> float:
    sx = sum(p[0] for p in source) / len(source)
    sy = sum(p[1] for p in source) / len(source)
    tx = sum(p[0] for p in target) / len(target)
    ty = sum(p[1] for p in target) / len(target)
    a = [complex(p[0] - sx, p[1] - sy) for p in source]
    b = [complex(p[0] - tx, p[1] - ty) for p in target]
    denom = sum(abs(z) ** 2 for z in a)
    if denom < 100.0:
        raise ValueError("collapsed source constellation")
    coefficient = sum(z.conjugate() * w for z, w in zip(a, b)) / denom
    if not 0.65 <= abs(coefficient) <= 1.45:
        raise ValueError("implausible constellation scale")
    return math.sqrt(sum(abs(coefficient * z - w) ** 2 for z, w in zip(a, b)) / len(a))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    if arr.shape[:2] != (700, 960):
        raise ValueError("unexpected raster dimensions")
    roi_left = np.zeros(arr.shape[:2], dtype=bool)
    roi_left[190:630, 50:450] = True
    source = []
    for name, rgb in PALETTE:
        source.append((name, _centroid(_mask(arr, rgb) & roi_left)))

    roi_right = np.zeros(arr.shape[:2], dtype=bool)
    roi_right[190:630, 520:920] = True
    components = _components(_mask(arr, MARKER) & roi_right)
    if len(components) != 5:
        raise ValueError(f"expected five right landmarks, found {len(components)}")
    ringed = [i for i, item in enumerate(components) if item["width"] >= 29 and item["height"] >= 29]
    if len(ringed) != 1:
        raise ValueError("could not isolate one ringed target")
    target_right = ringed[0]
    right = [item["center"] for item in components]
    source_points = [point for _, point in source]

    scored = []
    for perm in itertools.permutations(range(5)):
        try:
            residual = _fit(source_points, [right[i] for i in perm])
        except ValueError:
            continue
        scored.append((residual, perm))
    if len(scored) < 2:
        raise ValueError("insufficient valid mappings")
    scored.sort(key=lambda item: item[0])
    best, runner = scored[0], scored[1]
    if best[0] > 2.0 or runner[0] - best[0] < 8.0:
        raise ValueError("ambiguous or inconsistent landmark mapping")
    source_index = best[1].index(target_right)
    answer = source[source_index][0]
    if answer not in ("red", "blue", "gold"):
        raise ValueError("ring maps to a non-answer anchor")
    return answer
