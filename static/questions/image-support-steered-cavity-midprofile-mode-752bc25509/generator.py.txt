"""Analytic scene generator and independent forward renderer."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
AA = 3
PALE = (248, 246, 238)
PALE_2 = (239, 237, 229)
PALETTES = (
    (32, 104, 143),
    (18, 126, 105),
    (105, 76, 145),
    (166, 75, 57),
)


def sample_scene(seed: int) -> dict:
    """Sample balanced modes while decorrelating pose, scale, and palette."""
    rng = random.Random(int(seed))
    marks = []
    for _ in range(13):
        marks.append([
            rng.randint(34, WIDTH - 34),
            rng.randint(34, HEIGHT - 34),
            rng.randint(2, 5),
            rng.randint(214, 232),
        ])
    return {
        "shape_mode": "waist" if int(seed) % 2 == 0 else "belly",
        "cx": rng.randint(176, 336),
        "cy": rng.randint(176, 336),
        "scale": round(rng.uniform(0.86, 1.14), 6),
        "palette": (int(seed) // 2) % len(PALETTES),
        "background_marks": marks,
    }


def _cavity_polygon(scene: dict) -> list[tuple[float, float]]:
    mode = scene["shape_mode"]
    scale = float(scene["scale"])
    cx, cy = float(scene["cx"]), float(scene["cy"])
    half_w = 57.0 * scale
    points_top = []
    points_bottom = []
    # Complementary smooth profiles have similar total area but opposite
    # middle curvature. Rounded endcaps come from tapering the last samples.
    for i in range(49):
        t = -1.0 + 2.0 * i / 48.0
        g = abs(t) ** 1.55
        if mode == "waist":
            half_h = (25.0 + 25.0 * g) * scale
        else:
            half_h = (45.0 - 25.0 * g) * scale
        cap = max(0.20, math.sqrt(max(0.0, 1.0 - abs(t) ** 10)))
        half_h *= cap
        x = cx + half_w * t
        points_top.append((x, cy - half_h))
        points_bottom.append((x, cy + half_h))
    return points_top + list(reversed(points_bottom))


def render(scene: dict) -> Image.Image:
    """Render one compact plaque; only final PNG pixels are oracle-visible."""
    canvas = Image.new("RGB", (WIDTH * AA, HEIGHT * AA), PALE)
    draw = ImageDraw.Draw(canvas)

    for x, y, radius, tone in scene["background_marks"]:
        box = [(x - radius) * AA, (y - radius) * AA, (x + radius) * AA, (y + radius) * AA]
        draw.ellipse(box, fill=(tone, tone, tone))

    cx, cy = float(scene["cx"]), float(scene["cy"])
    scale = float(scene["scale"])
    plaque_color = PALETTES[int(scene["palette"]) % len(PALETTES)]
    pw, ph, radius = 91.0 * scale, 78.0 * scale, 28.0 * scale
    plaque_box = [(cx - pw) * AA, (cy - ph) * AA, (cx + pw) * AA, (cy + ph) * AA]
    shadow = [v + 5 * AA for v in plaque_box]
    draw.rounded_rectangle(shadow, radius=radius * AA, fill=(213, 211, 204))
    draw.rounded_rectangle(plaque_box, radius=radius * AA, fill=plaque_color)

    cavity = [(x * AA, y * AA) for x, y in _cavity_polygon(scene)]
    draw.polygon(cavity, fill=PALE_2)

    return canvas.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["shape_mode"])


def margin(scene: dict) -> float:
    return 20.0 * float(scene["scale"])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # Pose and palette interventions change pixels, so they are tested in the
    # candidate suite rather than misdeclared as exact latent aliases.
    return []
