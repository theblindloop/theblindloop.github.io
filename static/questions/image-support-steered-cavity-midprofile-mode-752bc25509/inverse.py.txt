"""Pixel-only inverse arm for the compact enclosed-cavity profile."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Return 4-connected components as coordinate arrays, largest first."""
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[np.ndarray] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        q = deque([(int(y), int(x))])
        seen[y, x] = True
        pts = []
        while q:
            yy, xx = q.popleft()
            pts.append((yy, xx))
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    q.append((ny, nx))
        groups.append(np.asarray(pts, dtype=np.int32))
    groups.sort(key=len, reverse=True)
    return groups


def _recover_cavity(image: Image.Image) -> np.ndarray | None:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    chroma = arr.max(axis=2) - arr.min(axis=2)
    dark_enough = arr.mean(axis=2) < 220
    ink = (chroma > 42) & dark_enough
    ink_groups = _components(ink)
    if not ink_groups or len(ink_groups[0]) < 3500:
        return None
    plaque = np.zeros(ink.shape, dtype=bool)
    plaque[ink_groups[0][:, 0], ink_groups[0][:, 1]] = True

    # Flood-fill the complement from the canvas border. Any substantial
    # complement component not reached is an enclosed negative-space cavity.
    free = ~plaque
    h, w = free.shape
    outside = np.zeros_like(free)
    q: deque[tuple[int, int]] = deque()
    for x in range(w):
        if free[0, x]: q.append((0, x)); outside[0, x] = True
        if free[h - 1, x] and not outside[h - 1, x]: q.append((h - 1, x)); outside[h - 1, x] = True
    for y in range(h):
        if free[y, 0] and not outside[y, 0]: q.append((y, 0)); outside[y, 0] = True
        if free[y, w - 1] and not outside[y, w - 1]: q.append((y, w - 1)); outside[y, w - 1] = True
    while q:
        y, x = q.popleft()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < h and 0 <= nx < w and free[ny, nx] and not outside[ny, nx]:
                outside[ny, nx] = True
                q.append((ny, nx))
    enclosed = free & ~outside
    holes = _components(enclosed)
    if not holes or len(holes[0]) < 1800:
        return None
    cavity = np.zeros_like(enclosed)
    cavity[holes[0][:, 0], holes[0][:, 1]] = True
    return cavity


def _profile_score(cavity: np.ndarray) -> float | None:
    ys, xs = np.nonzero(cavity)
    if len(xs) == 0:
        return None
    x0, x1 = int(xs.min()), int(xs.max())
    width = x1 - x0 + 1
    if width < 70:
        return None
    counts = cavity[:, x0:x1 + 1].sum(axis=0).astype(float)
    u = np.linspace(-1.0, 1.0, len(counts))
    center = counts[np.abs(u) <= 0.16]
    left = counts[(u >= -0.72) & (u <= -0.44)]
    right = counts[(u >= 0.44) & (u <= 0.72)]
    if min(len(center), len(left), len(right)) < 5:
        return None
    flank = 0.5 * (float(np.median(left)) + float(np.median(right)))
    return float(np.median(center)) - flank


def decision_from_image(image: Image.Image) -> str:
    cavity = _recover_cavity(image)
    if cavity is None:
        return "abstain"
    score = _profile_score(cavity)
    if score is None or abs(score) < 8.0:
        return "abstain"
    return "belly" if score > 0 else "waist"
