"""Deterministic control-selected donor correspondence boards."""

from __future__ import annotations

import itertools
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1460, 800
N = 6
DECISIONS = tuple("ABCDEF")
INK = (35, 39, 48)
PAPER = (246, 247, 244)
WHITE = (255, 255, 255)
GRID = (116, 123, 132)
START = (198, 38, 139)
FILLED_HEAD = (48, 92, 150)
HOLLOW_HEAD = (141, 75, 145)
SOURCE_COLORS = ((220, 55, 62), (35, 105, 210), (20, 153, 100), (235, 137, 25), (126, 70, 190), (10, 157, 177))
TARGET_COLORS = ((119, 63, 45), (224, 91, 168), (88, 126, 35), (34, 65, 139), (191, 103, 26), (72, 72, 78))
CELL = 58
GRID_Y = 284
SOURCE_X = 76
TARGET_X = 1036
CHECK_X = 500
FILLED_DONOR_X = 632
HOLLOW_DONOR_X = 790
TARGET_BADGE_X = 929
ROUTE_Y = 302
ROUTE_STEP = 55
FONT_PATH = Path("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf")

_CANONICAL = tuple(tuple(1 if row in {col, (col + 1) % N, (col + 3) % N} else 0 for col in range(N)) for row in range(N))


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(str(FONT_PATH), size=size)


def _valid_matrix(matrix: object) -> bool:
    return isinstance(matrix, list) and len(matrix) == N and all(isinstance(row, list) and len(row) == N and all(value in (0, 1) for value in row) for row in matrix)


def _signatures(matrix: list[list[int]], tags: list[int]) -> list[tuple[int, ...]]:
    by_tag = {int(tag): row for tag, row in zip(tags, matrix, strict=True)}
    return [tuple(by_tag[tag][col] for tag in range(N)) for col in range(N)]


def _select_donors(signature: tuple[int, ...], controls: list[int], filled: list[int], hollow: list[int]) -> tuple[int, ...]:
    selected = tuple(int(filled[target] if signature[int(controls[target])] else hollow[target]) for target in range(N))
    if sorted(selected) != list(range(N)):
        raise ValueError("condition-selected donors are not a permutation")
    return selected


def _route(signature: tuple[int, ...], controls: list[int], filled: list[int], hollow: list[int]) -> tuple[int, ...]:
    return tuple(signature[donor] for donor in _select_donors(signature, controls, filled, hollow))


def _donor_maps(signature: tuple[int, ...], rng: random.Random) -> tuple[list[int], list[int], list[int]]:
    for _ in range(1000):
        controls = list(range(N))
        selected = list(range(N))
        rng.shuffle(controls)
        rng.shuffle(selected)
        if any(controls[target] == selected[target] for target in range(N)):
            continue
        conditions = [signature[controls[target]] for target in range(N)]
        ones = [target for target, value in enumerate(conditions) if value]
        zeros = [target for target, value in enumerate(conditions) if not value]
        filled = selected.copy()
        hollow = selected.copy()
        shift_zero = 1 + rng.randrange(2)
        shift_one = 1 + rng.randrange(2)
        for offset, target in enumerate(zeros):
            filled[target] = selected[zeros[(offset + shift_zero) % 3]]
        for offset, target in enumerate(ones):
            hollow[target] = selected[ones[(offset + shift_one) % 3]]
        if any(filled[target] == hollow[target] for target in range(N)):
            continue
        return controls, filled, hollow
    raise RuntimeError("could not construct role-separated donor maps")


def _target_signatures(required: list[tuple[int, ...]], rng: random.Random) -> list[tuple[int, ...]] | None:
    required = list(dict.fromkeys(required))
    if len(required) != 4 or any(sum(bits) != 3 for bits in required):
        return None
    pool = [bits for bits in itertools.product((0, 1), repeat=N) if sum(bits) == 3 and bits not in required]
    rng.shuffle(pool)
    need = tuple(3 - sum(bits[row] for bits in required) for row in range(N))
    for extras in itertools.combinations(pool, N - len(required)):
        if all(sum(bits[row] for bits in extras) == need[row] for row in range(N)):
            return [*required, *extras]
    return None


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    for _ in range(2000):
        entity_order = list(range(N))
        source_tags = list(range(N))
        target_tags = list(range(N))
        route_order = list(range(N))
        rng.shuffle(entity_order)
        rng.shuffle(source_tags)
        rng.shuffle(target_tags)
        rng.shuffle(route_order)
        start_index = rng.randrange(N)
        source = [[_CANONICAL[tag][entity] for entity in entity_order] for tag in source_tags]
        start_signature = _signatures(source, source_tags)[start_index]
        controls, filled, hollow = _donor_maps(start_signature, rng)
        gold = _route(start_signature, controls, filled, hollow)
        check_shortcut = tuple(start_signature[controls[target]] for target in range(N))
        opposite_shortcut = tuple(start_signature[hollow[target] if start_signature[controls[target]] else filled[target]] for target in range(N))
        filled_shortcut = tuple(start_signature[filled[target]] for target in range(N))
        targets = _target_signatures([gold, check_shortcut, opposite_shortcut, filled_shortcut], rng)
        if targets is None:
            continue
        gold_index = int(seed) % N
        targets[0], targets[gold_index] = targets[gold_index], targets[0]
        target = [[targets[col][tag] for col in range(N)] for tag in target_tags]
        scene = {
            "source_matrix": source,
            "target_matrix": target,
            "source_row_tags": source_tags,
            "target_row_tags": target_tags,
            "route_order": route_order,
            "control_mapping": controls,
            "filled_donor_mapping": filled,
            "hollow_donor_mapping": hollow,
            "start_index": start_index,
        }
        if not is_quarantined(scene):
            return scene
    raise RuntimeError("could not construct control-selected donor registry")


def analytic_gold(scene: dict) -> str:
    source = _signatures(scene["source_matrix"], scene["source_row_tags"])
    target = _signatures(scene["target_matrix"], scene["target_row_tags"])
    routed = _route(source[int(scene["start_index"])], scene["control_mapping"], scene["filled_donor_mapping"], scene["hollow_donor_mapping"])
    matches = [index for index, signature in enumerate(target) if signature == routed]
    if len(matches) != 1:
        raise ValueError("routed START does not have one target")
    return DECISIONS[matches[0]]


def margin(scene: dict) -> float:
    if is_quarantined(scene):
        return 0.0
    source = _signatures(scene["source_matrix"], scene["source_row_tags"])
    target = _signatures(scene["target_matrix"], scene["target_row_tags"])
    routed = _route(source[int(scene["start_index"])], scene["control_mapping"], scene["filled_donor_mapping"], scene["hollow_donor_mapping"])
    return float(min(sum(a != b for a, b in zip(routed, other, strict=True)) for other in target if other != routed))


def is_quarantined(scene: dict) -> bool:
    try:
        source, target = scene["source_matrix"], scene["target_matrix"]
        source_tags, target_tags = scene["source_row_tags"], scene["target_row_tags"]
        controls = [int(value) for value in scene["control_mapping"]]
        filled = [int(value) for value in scene["filled_donor_mapping"]]
        hollow = [int(value) for value in scene["hollow_donor_mapping"]]
        if not _valid_matrix(source) or not _valid_matrix(target):
            return True
        if any(sorted(values) != list(range(N)) for values in (source_tags, target_tags, scene["route_order"], controls, filled, hollow)):
            return True
        if any(filled[tag] == hollow[tag] for tag in range(N)) or int(scene["start_index"]) not in range(N):
            return True
        source_sigs = _signatures(source, source_tags)
        target_sigs = _signatures(target, target_tags)
        if len(set(source_sigs)) != N or len(set(target_sigs)) != N:
            return True
        if any(sum(signature) != 3 for signature in source_sigs + target_sigs):
            return True
        if any(sum(row) != 3 for row in source + target):
            return True
        signature = source_sigs[int(scene["start_index"])]
        selected = _select_donors(signature, controls, filled, hollow)
        if any(selected[tag] == controls[tag] for tag in range(N)):
            return True
        routed = tuple(signature[donor] for donor in selected)
        check_shortcut = tuple(signature[controls[tag]] for tag in range(N))
        opposite_shortcut = tuple(signature[hollow[tag] if signature[controls[tag]] else filled[tag]] for tag in range(N))
        filled_shortcut = tuple(signature[filled[tag]] for tag in range(N))
        required = {routed, check_shortcut, opposite_shortcut, filled_shortcut}
        return len(required) != 4 or not required.issubset(set(target_sigs)) or target_sigs.count(routed) != 1
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _centered_text(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str, size: int, fill: tuple[int, int, int]) -> None:
    font = _font(size)
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2 - 2), text, font=font, fill=fill)


def _draw_badge(draw: ImageDraw.ImageDraw, cx: int, cy: int, tag: int, side: str) -> None:
    color = (SOURCE_COLORS if side == "source" else TARGET_COLORS)[tag]
    if tag == 0:
        draw.polygon(((cx, cy - 18), (cx - 20, cy + 17), (cx + 20, cy + 17)), fill=color, outline=INK)
    elif tag == 1:
        draw.rectangle((cx - 17, cy - 17, cx + 17, cy + 17), fill=color, outline=INK, width=2)
    elif tag == 2:
        draw.ellipse((cx - 18, cy - 18, cx + 18, cy + 18), fill=color, outline=INK, width=2)
    elif tag == 3:
        draw.polygon(((cx, cy - 19), (cx - 19, cy), (cx, cy + 19), (cx + 19, cy)), fill=color, outline=INK)
    elif tag == 4:
        draw.polygon(((cx - 17, cy - 9), (cx, cy - 19), (cx + 17, cy - 9), (cx + 17, cy + 9), (cx, cy + 19), (cx - 17, cy + 9)), fill=color, outline=INK)
    else:
        draw.rectangle((cx - 6, cy - 19, cx + 6, cy + 19), fill=color, outline=INK, width=2)
        draw.rectangle((cx - 19, cy - 6, cx + 19, cy + 6), fill=color, outline=INK, width=2)


def _draw_board(draw: ImageDraw.ImageDraw, x0: int, matrix: list[list[int]], tags: list[int], labels: tuple[str, ...], caption: str, start: int | None, side: str) -> None:
    _centered_text(draw, (x0 + N * CELL // 2, 142), caption, 28, INK)
    for col, label in enumerate(labels):
        cx = x0 + col * CELL + CELL // 2
        if col == start:
            draw.rounded_rectangle((cx - 25, 190, cx + 25, 236), radius=11, fill=START)
            _centered_text(draw, (cx, 213), label, 22, WHITE)
            _centered_text(draw, (cx, 176), "START", 14, START)
        else:
            draw.ellipse((cx - 20, 193, cx + 20, 233), fill=WHITE, outline=INK, width=3)
            _centered_text(draw, (cx, 213), label, 21, INK)
    for index in range(N + 1):
        draw.line((x0, GRID_Y + index * CELL, x0 + N * CELL, GRID_Y + index * CELL), fill=GRID, width=2)
        draw.line((x0 + index * CELL, GRID_Y, x0 + index * CELL, GRID_Y + N * CELL), fill=GRID, width=2)
    for row in range(N):
        cy = GRID_Y + row * CELL + CELL // 2
        _draw_badge(draw, x0 - 45, cy, int(tags[row]), side)
        for col in range(N):
            cx = x0 + col * CELL + CELL // 2
            draw.ellipse((cx - 15, cy - 15, cx + 15, cy + 15), fill=INK if matrix[row][col] else WHITE, outline=INK, width=4)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), PAPER)
    draw = ImageDraw.Draw(image)
    _centered_text(draw, (WIDTH // 2, 43), "CONTROL-SELECTED DONOR REGISTRY", 34, INK)
    _draw_board(draw, SOURCE_X, scene["source_matrix"], scene["source_row_tags"], tuple(str(i + 1) for i in range(N)), "SOURCE", int(scene["start_index"]), "source")
    _draw_board(draw, TARGET_X, scene["target_matrix"], scene["target_row_tags"], DECISIONS, "TARGET", None, "target")
    _centered_text(draw, (716, 113), "DONOR TABLE", 27, INK)
    draw.rounded_rectangle((463, 145, 970, 653), radius=16, fill=WHITE, outline=GRID, width=3)
    _centered_text(draw, (CHECK_X, 192), "CHECK", 15, INK)
    _centered_text(draw, (FILLED_DONOR_X, 177), "IF CHECK FILLED", 13, FILLED_HEAD)
    _centered_text(draw, (FILLED_DONOR_X, 202), "TAKE", 16, FILLED_HEAD)
    _centered_text(draw, (HOLLOW_DONOR_X, 177), "IF CHECK HOLLOW", 13, HOLLOW_HEAD)
    _centered_text(draw, (HOLLOW_DONOR_X, 202), "TAKE", 16, HOLLOW_HEAD)
    _centered_text(draw, (TARGET_BADGE_X, 192), "PUT", 16, INK)
    draw.line((553, 218, 553, 635), fill=GRID, width=2)
    draw.line((711, 218, 711, 635), fill=GRID, width=2)
    draw.line((865, 218, 865, 635), fill=GRID, width=2)
    for route_row, target_tag in enumerate(scene["route_order"]):
        cy = ROUTE_Y + route_row * ROUTE_STEP
        _draw_badge(draw, CHECK_X, cy, int(scene["control_mapping"][target_tag]), "source")
        _draw_badge(draw, FILLED_DONOR_X, cy, int(scene["filled_donor_mapping"][target_tag]), "source")
        _draw_badge(draw, HOLLOW_DONOR_X, cy, int(scene["hollow_donor_mapping"][target_tag]), "source")
        draw.line((827, cy, 887, cy), fill=INK, width=4)
        draw.polygon(((887, cy), (875, cy - 8), (875, cy + 8)), fill=INK)
        _draw_badge(draw, TARGET_BADGE_X, cy, int(target_tag), "target")
    _centered_text(draw, (WIDTH // 2, 704), "CHECK chooses which TAKE badge supplies the state.", 20, INK)
    _centered_text(draw, (WIDTH // 2, 747), "Copy the chosen badge's START state to PUT; then match TARGET.", 20, INK)
    return image
