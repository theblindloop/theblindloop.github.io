"""Independent image-only inverse for Control-Selected Donor Registry."""

from __future__ import annotations

import numpy as np
from PIL import Image

N = 6
DECISIONS = tuple("ABCDEF")
INK = np.array((35, 39, 48), dtype=np.int16)
START = np.array((198, 38, 139), dtype=np.int16)
SOURCE_COLORS = tuple(np.array(color, dtype=np.int16) for color in ((220, 55, 62), (35, 105, 210), (20, 153, 100), (235, 137, 25), (126, 70, 190), (10, 157, 177)))
TARGET_COLORS = tuple(np.array(color, dtype=np.int16) for color in ((119, 63, 45), (224, 91, 168), (88, 126, 35), (34, 65, 139), (191, 103, 26), (72, 72, 78)))
CELL = 58
GRID_Y = 284
SOURCE_X = 76
TARGET_X = 1036
CHECK_X = 500
FILLED_DONOR_X = 632
HOLLOW_DONOR_X = 790
TARGET_BADGE_X = 929
ROUTE_Y = 302
ROUTE_STEP = 55


def _mask(arr: np.ndarray, color: np.ndarray, tolerance: int = 10) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - color), axis=2) <= tolerance


def _tag_in_crop(crop: np.ndarray, colors: tuple[np.ndarray, ...]) -> int | None:
    scores = [int(_mask(crop, color).sum()) for color in colors]
    winners = [index for index, score in enumerate(scores) if score > 170]
    return winners[0] if len(winners) == 1 else None


def _read_board(arr: np.ndarray, x0: int, colors: tuple[np.ndarray, ...]) -> tuple[list[list[int]], list[int]] | None:
    matrix, tags = [], []
    for row in range(N):
        cy = GRID_Y + row * CELL + CELL // 2
        tag = _tag_in_crop(arr[cy - 22:cy + 23, x0 - 69:x0 - 20], colors)
        if tag is None:
            return None
        tags.append(tag)
        values = []
        for col in range(N):
            cx = x0 + col * CELL + CELL // 2
            values.append(1 if int(_mask(arr[cy - 9:cy + 10, cx - 9:cx + 10], INK).sum()) > 250 else 0)
        matrix.append(values)
    if sorted(tags) != list(range(N)) or any(sum(row) != 3 for row in matrix):
        return None
    if any(sum(matrix[row][col] for row in range(N)) != 3 for col in range(N)):
        return None
    return matrix, tags


def _read_routes(arr: np.ndarray) -> tuple[list[int], list[int], list[int]] | None:
    controls: list[int | None] = [None] * N
    filled: list[int | None] = [None] * N
    hollow: list[int | None] = [None] * N
    for row in range(N):
        cy = ROUTE_Y + row * ROUTE_STEP
        control = _tag_in_crop(arr[cy - 22:cy + 23, CHECK_X - 23:CHECK_X + 24], SOURCE_COLORS)
        f_donor = _tag_in_crop(arr[cy - 22:cy + 23, FILLED_DONOR_X - 23:FILLED_DONOR_X + 24], SOURCE_COLORS)
        h_donor = _tag_in_crop(arr[cy - 22:cy + 23, HOLLOW_DONOR_X - 23:HOLLOW_DONOR_X + 24], SOURCE_COLORS)
        target = _tag_in_crop(arr[cy - 22:cy + 23, TARGET_BADGE_X - 23:TARGET_BADGE_X + 24], TARGET_COLORS)
        if control is None or f_donor is None or h_donor is None or target is None or controls[target] is not None:
            return None
        controls[target], filled[target], hollow[target] = control, f_donor, h_donor
    for values in (controls, filled, hollow):
        if sorted(value for value in values if value is not None) != list(range(N)):
            return None
    if any(filled[tag] == hollow[tag] for tag in range(N)):
        return None
    return ([int(value) for value in controls], [int(value) for value in filled], [int(value) for value in hollow])


def _signatures(matrix: list[list[int]], tags: list[int]) -> list[tuple[int, ...]]:
    by_tag = {tag: row for tag, row in zip(tags, matrix, strict=True)}
    return [tuple(by_tag[tag][col] for tag in range(N)) for col in range(N)]


def decision_from_image(image: Image.Image) -> str:
    try:
        if image.size != (1460, 800):
            return "abstain"
        arr = np.asarray(image.convert("RGB"))
        source = _read_board(arr, SOURCE_X, SOURCE_COLORS)
        target = _read_board(arr, TARGET_X, TARGET_COLORS)
        routes = _read_routes(arr)
        if source is None or target is None or routes is None:
            return "abstain"
        starts = []
        for col in range(N):
            cx = SOURCE_X + col * CELL + CELL // 2
            starts.append(int(_mask(arr[188:238, cx - 27:cx + 28], START).sum()) > 750)
        if sum(starts) != 1:
            return "abstain"
        signature = _signatures(*source)[starts.index(True)]
        controls, filled, hollow = routes
        selected = [filled[tag] if signature[controls[tag]] else hollow[tag] for tag in range(N)]
        if sorted(selected) != list(range(N)) or any(selected[tag] == controls[tag] for tag in range(N)):
            return "abstain"
        routed = tuple(signature[donor] for donor in selected)
        target_signatures = _signatures(*target)
        matches = [index for index, candidate in enumerate(target_signatures) if candidate == routed]
        return DECISIONS[matches[0]] if len(matches) == 1 else "abstain"
    except Exception:
        return "abstain"
