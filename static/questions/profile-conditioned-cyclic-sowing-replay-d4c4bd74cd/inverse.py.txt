"""Independent image-only inverse for the cyclic sowing world."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int, float, float]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        todo = deque([(int(y), int(x))])
        seen[y, x] = True
        xs, ys = [], []
        while todo:
            yy, xx = todo.popleft()
            xs.append(xx)
            ys.append(yy)
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    todo.append((ny, nx))
        out.append((min(xs), min(ys), max(xs), max(ys), len(xs), float(sum(xs)) / len(xs), float(sum(ys)) / len(ys)))
    return out


def _frames(arr: np.ndarray, lower: bool) -> list[tuple[int, int, int, int]]:
    h, _ = arr.shape[:2]
    dark_gray = (
        (arr[:, :, 0] >= 55) & (arr[:, :, 0] <= 90)
        & (arr[:, :, 1] >= 60) & (arr[:, :, 1] <= 100)
        & (arr[:, :, 2] >= 70) & (arr[:, :, 2] <= 110)
    )
    dark_gray[: int(h * (0.58 if lower else 0.0))] = False if lower else dark_gray[: int(h * 0.0)]
    if not lower:
        dark_gray[int(h * 0.58) :] = False
    boxes = []
    for x0, y0, x1, y1, area, _, _ in _components(dark_gray):
        if x1 - x0 > 120 and y1 - y0 > 100 and area > 600:
            boxes.append((x0, y0, x1, y1))
    return sorted(boxes)


def _replay(counts: list[int], moves: list[int]) -> list[int]:
    state = list(counts)
    for source in moves:
        hand = state[source]
        state[source] = 0
        for step in range(hand):
            state[(source + 1 + step) % 4] += 1
    return state


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    h, _ = arr.shape[:2]
    bins = _frames(arr, lower=False)
    cards = _frames(arr, lower=True)
    if len(bins) != 4 or len(cards) != 5:
        raise ValueError(f"expected four bins and five cards, found {len(bins)} and {len(cards)}")

    r, g, b = arr[:, :, 0].astype(int), arr[:, :, 1].astype(int), arr[:, :, 2].astype(int)
    chroma = np.maximum.reduce([r, g, b]) - np.minimum.reduce([r, g, b])
    colored = (chroma > 55) & (np.maximum.reduce([r, g, b]) > 100)
    magenta = (r > 180) & (g < 90) & (b > 100)
    colored &= ~magenta
    colored[int(h * 0.56) :] = False

    counts = []
    colored_components = _components(colored)
    for x0, y0, x1, y1 in bins:
        n = sum(180 <= area <= 500 and x0 < cx < x1 and y0 < cy < y1 for _, _, _, _, area, cx, cy in colored_components)
        counts.append(n)
    if sum(counts) != 8:
        raise ValueError(f"counter segmentation recovered total {sum(counts)}, not 8")

    magenta_components = [c for c in _components(magenta) if c[4] > 500]
    if len(magenta_components) != 1:
        raise ValueError("query ring is missing or ambiguous")
    qx = magenta_components[0][5]
    query = min(range(4), key=lambda i: abs(qx - (bins[i][0] + bins[i][2]) / 2))

    black = (r < 45) & (g < 45) & (b < 45)
    black[: int(h * 0.58)] = False
    black_components = _components(black)
    moves = []
    for x0, y0, x1, y1 in cards:
        pips = sum(90 <= area <= 220 and x0 < cx < x1 and y0 + 35 < cy < y1 - 30 for _, _, _, _, area, cx, cy in black_components)
        if pips not in (1, 2, 3, 4):
            raise ValueError(f"card source has {pips} pips")
        moves.append(pips - 1)
    return str(_replay(counts, moves)[query])
