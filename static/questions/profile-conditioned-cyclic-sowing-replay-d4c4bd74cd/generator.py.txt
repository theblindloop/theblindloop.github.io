"""Deterministic sampler, renderer, and analytic replay."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1000, 560
TOTAL_COUNTERS = 8
COUNTER_PALETTES = (
    [45, 107, 220],
    [22, 150, 105],
    [230, 120, 35],
    [120, 80, 190],
)
FRAME = (70, 78, 90)
TEXT = (35, 43, 55)
MAGENTA = (220, 35, 150)
TEAL = (20, 155, 160)
PAPER = (250, 250, 247)
BIN_FILL = (242, 245, 247)


def _replay(counts: list[int], moves: list[int]) -> list[int]:
    state = list(counts)
    for source in moves:
        hand = state[source]
        state[source] = 0
        for step in range(hand):
            state[(source + 1 + step) % 4] += 1
    return state


def _composition(rng: random.Random, total: int) -> list[int]:
    cuts = sorted(rng.sample(range(1, total + 3), 3))
    values = [cuts[0] - 1, cuts[1] - cuts[0] - 1, cuts[2] - cuts[1] - 1, total + 3 - cuts[2]]
    return values


def _counter_points(counts: list[int], shift: int, rng: random.Random) -> list[list[list[int]]]:
    result: list[list[list[int]]] = []
    offsets = [(35, 66), (65, 66), (95, 66), (125, 66), (35, 108), (65, 108), (95, 108), (125, 108)]
    for i, count in enumerate(counts):
        left = 70 + 200 * i + shift
        points = [[left + dx, 105 + dy] for dx, dy in offsets[:count]]
        rng.shuffle(points)
        result.append(points)
    return result


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    # Rejection keeps every answer inside the public finite candidate set and
    # avoids the degenerate all-zero query result dominating small datasets.
    for _ in range(1000):
        counts = _composition(rng, TOTAL_COUNTERS)
        moves = [rng.randrange(4) for _ in range(5)]
        query = rng.randrange(4)
        final = _replay(counts, moves)
        if 0 <= final[query] <= TOTAL_COUNTERS:
            shift = rng.choice((-12, 0, 12))
            palette = list(rng.choice(COUNTER_PALETTES))
            return {
                "counts": counts,
                "moves": moves,
                "query": query,
                "counter_points": _counter_points(counts, shift, rng),
                "palette": palette,
                "x_shift": shift,
            }
    raise RuntimeError("could not sample a valid scene")


def _font(size: int = 18):
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _draw_pips(draw: ImageDraw.ImageDraw, center_x: int, center_y: int, count: int, radius: int = 6) -> None:
    offsets = {1: [(0, 0)], 2: [(-12, 0), (12, 0)], 3: [(-18, 0), (0, 0), (18, 0)], 4: [(-11, -8), (11, -8), (-11, 8), (11, 8)]}[count]
    for dx, dy in offsets:
        draw.ellipse((center_x + dx - radius, center_y + dy - radius, center_x + dx + radius, center_y + dy + radius), fill=(15, 18, 22))


def render(scene: dict) -> Image.Image:
    counts = scene["counts"]
    moves = scene["moves"]
    query = scene["query"]
    points = scene["counter_points"]
    counter_color = tuple(scene["palette"])
    shift = int(scene["x_shift"])
    image = Image.new("RGB", (WIDTH, HEIGHT), PAPER)
    draw = ImageDraw.Draw(image)
    draw.text((34, 20), "START COUNTERS", fill=TEXT, font=_font(24))
    draw.text((34, 315), "SOW CARDS", fill=TEXT, font=_font(24))
    draw.line((210, 331, 900, 331), fill=TEAL, width=4)
    draw.polygon([(900, 323), (922, 331), (900, 339)], fill=TEAL)

    for i in range(4):
        left = 70 + 200 * i + shift
        box = (left, 105, left + 160, 255)
        draw.rounded_rectangle(box, radius=10, fill=BIN_FILL, outline=FRAME, width=4)
        _draw_pips(draw, left + 80, 78, i + 1, radius=6)
        if i == query:
            draw.rounded_rectangle((left - 10, 95, left + 170, 265), radius=15, outline=MAGENTA, width=6)
        for x, y in points[i]:
            draw.ellipse((x - 11, y - 11, x + 11, y + 11), fill=counter_color, outline=(255, 255, 255), width=2)

    for j, source in enumerate(moves):
        left = 50 + 180 * j + shift
        draw.rounded_rectangle((left, 365, left + 150, 500), radius=10, fill=(255, 255, 255), outline=FRAME, width=4)
        draw.rectangle((left + 4, 369, left + 146, 394), fill=TEAL)
        _draw_pips(draw, left + 75, 439, source + 1, radius=7)
        draw.line((left + 48, 478, left + 102, 478), fill=TEAL, width=5)
        draw.polygon([(left + 102, 468), (left + 124, 478), (left + 102, 488)], fill=TEAL)

    draw.text((48, 525), "EMPTY NAMED BIN   •   DISTRIBUTE RIGHT   •   WRAP 4 → 1", fill=TEXT, font=_font(17))
    return image


def analytic_gold(scene: dict) -> str:
    counts = scene["counts"]
    moves = scene["moves"]
    query = scene["query"]
    return str(_replay(counts, moves)[query])


def margin(scene: dict) -> float:
    # Minimum fixed clearance (pixels) between colored counter components after
    # their white outlines; the runner-up component interpretation is separated
    # by this amount throughout the supported world.
    return 6.0


def is_quarantined(scene: dict) -> bool:
    return any(c < 0 or c > 8 for c in scene["counts"]) or len(scene["moves"]) != 5


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # Counter positions are an unordered latent set represented as a list. Its
    # enumeration order is a genuine render alias because disks never overlap.
    twin = copy.deepcopy(scene)
    twin["counter_points"] = [list(reversed(group)) for group in twin["counter_points"]]
    return [("reverse_unordered_counter_enumeration", twin)]
