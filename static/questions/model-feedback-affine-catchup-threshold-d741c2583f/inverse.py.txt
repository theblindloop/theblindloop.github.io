"""Independent PNG-only inverse for the affine catch-up board."""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH = 1100
HEIGHT = 700
MACHINE_COLORS = {
    "blue": np.array((43, 104, 189), dtype=np.int16),
    "orange": np.array((218, 124, 48), dtype=np.int16),
}
DISPLAY_BG = np.array((35, 43, 53), dtype=np.int16)
SEGMENT_ON = np.array((250, 239, 184), dtype=np.int16)
QUERY_COLOR = np.array((38, 139, 126), dtype=np.int16)
CAL_Y = (235, 350)
SEGMENT_CENTERS = {
    "a": (0, -18), "g": (0, 0), "d": (0, 18),
    "f": (-12, -9), "b": (12, -9), "e": (-12, 9), "c": (12, 9),
}
DIGIT_SEGMENTS = {
    0: frozenset("abcedf"), 1: frozenset("bc"), 2: frozenset("abged"),
    3: frozenset("abgcd"), 4: frozenset("fgbc"), 5: frozenset("afgcd"),
    6: frozenset("afgecd"), 7: frozenset("abc"), 8: frozenset("abcdefg"),
    9: frozenset("abfgcd"),
}
SEGMENTS_TO_DIGIT = {value: key for key, value in DIGIT_SEGMENTS.items()}


class OracleError(ValueError):
    pass


def _near(patch: np.ndarray, color: np.ndarray, tolerance: int = 22) -> np.ndarray:
    delta = patch.astype(np.int32) - color.astype(np.int32)
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _fraction(array: np.ndarray, box: tuple[int, int, int, int], color: np.ndarray, tolerance: int = 22) -> float:
    x0, y0, x1, y1 = box
    patch = array[y0:y1, x0:x1]
    if patch.size == 0:
        return 0.0
    return float(np.mean(_near(patch, color, tolerance)))


def _read_digit(array: np.ndarray, cx: int, cy: int, label: str) -> int:
    active = set()
    for name, (dx, dy) in SEGMENT_CENTERS.items():
        patch = array[cy + dy - 2:cy + dy + 3, cx + dx - 2:cx + dx + 3]
        fraction = float(np.mean(_near(patch, SEGMENT_ON, 18)))
        if fraction >= 0.78:
            active.add(name)
        elif fraction > 0.18:
            raise OracleError(f"{label}: ambiguous segment {name}")
    key = frozenset(active)
    if key not in SEGMENTS_TO_DIGIT:
        raise OracleError(f"{label}: invalid segment pattern {sorted(active)}")
    return SEGMENTS_TO_DIGIT[key]


def _read_number(array: np.ndarray, cx: int, cy: int, label: str) -> int:
    if _fraction(array, (cx - 35, cy - 27, cx + 36, cy + 28), DISPLAY_BG, 20) < 0.56:
        raise OracleError(f"{label}: display body is missing or damaged")
    tens = _read_digit(array, cx - 18, cy, f"{label} tens")
    ones = _read_digit(array, cx + 18, cy, f"{label} ones")
    return 10 * tens + ones


def _infer_rule(examples: tuple[tuple[int, int], tuple[int, int]], label: str) -> tuple[int, int]:
    (x1, y1), (x2, y2) = examples
    if x1 == x2 or (y2 - y1) % (x2 - x1):
        raise OracleError(f"{label}: examples do not determine an integer multiplier")
    multiplier = (y2 - y1) // (x2 - x1)
    offset = y1 - multiplier * x1
    if not 1 <= multiplier <= 5 or not 0 <= offset <= 35:
        raise OracleError(f"{label}: inferred rule is outside range")
    if multiplier * x2 + offset != y2:
        raise OracleError(f"{label}: examples are inconsistent")
    return multiplier, offset


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    if rgb.size != (WIDTH, HEIGHT):
        raise OracleError(f"unexpected image size {rgb.size}")
    array = np.asarray(rgb)
    if _fraction(array, (70, 116, 105, 145), MACHINE_COLORS["blue"], 25) < 0.92:
        raise OracleError("blue calibration header is missing")
    if _fraction(array, (595, 116, 630, 145), MACHINE_COLORS["orange"], 25) < 0.92:
        raise OracleError("orange calibration header is missing")
    if _fraction(array, (260, 444, 840, 450), QUERY_COLOR, 25) < 0.48:
        raise OracleError("question frame is missing")
    rules = {}
    for name, input_x, output_x in (("blue", 145, 435), ("orange", 665, 955)):
        examples = tuple(
            (
                _read_number(array, input_x, y, f"{name} input {index}"),
                _read_number(array, output_x, y, f"{name} output {index}"),
            )
            for index, y in enumerate(CAL_Y, 1)
        )
        rules[name] = _infer_rule(examples, name)
    blue_a, blue_b = rules["blue"]
    orange_a, orange_b = rules["orange"]
    slope_gain = orange_a - blue_a
    head_start = blue_b - orange_b
    if slope_gain <= 0 or head_start <= 0:
        raise OracleError("visible trends do not define an orange catch-up")
    catch = (head_start + slope_gain - 1) // slope_gain
    if not 5 <= catch <= 9:
        raise OracleError("visible first catch is outside the answer set")
    if blue_a * (catch - 1) + blue_b <= orange_a * (catch - 1) + orange_b:
        raise OracleError("visible catch is not the first whole-number input")
    if orange_a * catch + orange_b < blue_a * catch + blue_b:
        raise OracleError("visible orange output has not caught blue")
    return str(catch)
