"""Deterministic renderer for an affine catch-up threshold puzzle."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1100
HEIGHT = 700
MACHINE_NAMES = ("blue", "orange")
MACHINE_COLORS = {"blue": (43, 104, 189), "orange": (218, 124, 48)}
BACKGROUND = (238, 242, 244)
PAPER = (255, 255, 252)
INK = (35, 41, 50)
MUTED = (96, 105, 116)
DISPLAY_BG = (35, 43, 53)
SEGMENT_ON = (250, 239, 184)
QUERY_COLOR = (38, 139, 126)
CAL_Y = (235, 350)
_FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
_BOLD_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
SEGMENTS = {
    0: "abcedf", 1: "bc", 2: "abged", 3: "abgcd", 4: "fgbc",
    5: "afgcd", 6: "afgecd", 7: "abc", 8: "abcdefg", 9: "abfgcd",
}


def _int(value: object, label: str, lo: int = 0, hi: int = 99) -> int:
    if isinstance(value, bool) or not isinstance(value, int) or not lo <= value <= hi:
        raise ValueError(f"{label} must be an integer in [{lo}, {hi}]")
    return int(value)


def _examples(value: object, label: str) -> tuple[tuple[int, int], tuple[int, int]]:
    if not isinstance(value, (list, tuple)) or len(value) != 2:
        raise ValueError(f"{label} must contain two examples")
    result = []
    for index, pair in enumerate(value):
        if not isinstance(pair, (list, tuple)) or len(pair) != 2:
            raise ValueError(f"{label} example {index + 1} must be [input, output]")
        result.append((_int(pair[0], "input"), _int(pair[1], "output")))
    return result[0], result[1]


def _infer_rule(examples: tuple[tuple[int, int], tuple[int, int]]) -> tuple[int, int]:
    (x1, y1), (x2, y2) = examples
    if x1 == x2 or (y2 - y1) % (x2 - x1):
        raise ValueError("examples do not determine an integer multiplier")
    multiplier = (y2 - y1) // (x2 - x1)
    offset = y1 - multiplier * x1
    if not 1 <= multiplier <= 5 or not 0 <= offset <= 35:
        raise ValueError("affine rule is outside the declared range")
    if multiplier * x2 + offset != y2:
        raise ValueError("examples are not affine")
    return multiplier, offset


def _rules(scene: dict) -> dict[str, tuple[int, int]]:
    if set(scene) != {"machines"}:
        raise ValueError("scene must contain only machines")
    machines = scene.get("machines")
    if not isinstance(machines, dict) or set(machines) != set(MACHINE_NAMES):
        raise ValueError("machines must contain blue and orange")
    result = {}
    for name in MACHINE_NAMES:
        spec = machines[name]
        if not isinstance(spec, dict) or set(spec) != {"examples"}:
            raise ValueError(f"{name} machine must contain only examples")
        result[name] = _infer_rule(_examples(spec["examples"], name))
    return result


def _catch_input(scene: dict) -> int:
    rules = _rules(scene)
    blue_a, blue_b = rules["blue"]
    orange_a, orange_b = rules["orange"]
    slope_gain = orange_a - blue_a
    head_start = blue_b - orange_b
    if slope_gain <= 0 or head_start <= 0:
        raise ValueError("orange must rise faster from a lower intercept")
    catch = (head_start + slope_gain - 1) // slope_gain
    if not 5 <= catch <= 9:
        raise ValueError("first catch input must be one of 5 through 9")
    if blue_a * (catch - 1) + blue_b <= orange_a * (catch - 1) + orange_b:
        raise ValueError("orange catches before the declared first input")
    if orange_a * catch + orange_b < blue_a * catch + blue_b:
        raise ValueError("orange has not caught blue")
    return catch


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) * 15485863 + 32452843)
    catch = 5 + int(seed) % 5
    blue_a = rng.randint(1, 3)
    slope_gain = rng.randint(1, min(3, 5 - blue_a))
    orange_a = blue_a + slope_gain
    exact = ((int(seed) // 5) % 2) == 0 or slope_gain == 1
    if exact:
        head_start = slope_gain * catch
    else:
        head_start = slope_gain * (catch - 1) + rng.randint(1, slope_gain - 1)
    orange_b = rng.randint(1, max(1, 30 - head_start))
    blue_b = orange_b + head_start
    params = {"blue": (blue_a, blue_b), "orange": (orange_a, orange_b)}
    machines = {}
    for name in MACHINE_NAMES:
        a, b = params[name]
        valid_x = [x for x in range(1, 10) if a * x + b <= 79]
        x1, x2 = sorted(rng.sample(valid_x, 2))
        examples = [[x1, a * x1 + b], [x2, a * x2 + b]]
        if rng.randrange(2):
            examples.reverse()
        machines[name] = {"examples": examples}
    scene = {"machines": machines}
    if _catch_input(scene) != catch:
        raise RuntimeError("sampler class construction failed")
    return scene


def analytic_gold(scene: dict) -> str:
    return str(_catch_input(scene))


def margin(scene: dict) -> float:
    rules = _rules(scene)
    catch = _catch_input(scene)
    ba, bb = rules["blue"]
    oa, ob = rules["orange"]
    before = ba * (catch - 1) + bb - (oa * (catch - 1) + ob)
    at_or_after = oa * catch + ob - (ba * catch + bb)
    return float(max(1, min(before, at_or_after if at_or_after > 0 else oa - ba)))


def is_quarantined(scene: dict) -> bool:
    try:
        _catch_input(scene)
        return False
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int, *, bold: bool = False) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(_BOLD_PATH if bold else _FONT_PATH, size=size)


def _centered(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font: ImageFont.ImageFont, fill: tuple[int, int, int]) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2 - box[1]), text, font=font, fill=fill)


def _draw_arrow(draw: ImageDraw.ImageDraw, x0: int, x1: int, y: int) -> None:
    draw.line((x0, y, x1 - 10, y), fill=INK, width=5)
    draw.polygon(((x1 - 12, y - 9), (x1, y), (x1 - 12, y + 9)), fill=INK)


def _draw_digit(draw: ImageDraw.ImageDraw, cx: int, cy: int, digit: int) -> None:
    centers = {
        "a": (cx, cy - 18, False), "g": (cx, cy, False), "d": (cx, cy + 18, False),
        "f": (cx - 12, cy - 9, True), "b": (cx + 12, cy - 9, True),
        "e": (cx - 12, cy + 9, True), "c": (cx + 12, cy + 9, True),
    }
    active = set(SEGMENTS[digit])
    for name, (x, y, vertical) in centers.items():
        if name in active:
            box = (x - 3, y - 8, x + 3, y + 8) if vertical else (x - 10, y - 3, x + 10, y + 3)
            draw.rounded_rectangle(box, radius=2, fill=SEGMENT_ON)


def _draw_number(draw: ImageDraw.ImageDraw, cx: int, cy: int, value: int) -> None:
    value = _int(value, "display value")
    draw.rounded_rectangle((cx - 39, cy - 31, cx + 39, cy + 31), radius=8, fill=DISPLAY_BG, outline=(112, 121, 128), width=2)
    tens, ones = divmod(value, 10)
    _draw_digit(draw, cx - 18, cy, tens)
    _draw_digit(draw, cx + 18, cy, ones)


def _draw_machine(draw: ImageDraw.ImageDraw, cx: int, cy: int, name: str) -> None:
    draw.rounded_rectangle((cx - 56, cy - 31, cx + 56, cy + 31), radius=12, fill=MACHINE_COLORS[name], outline=INK, width=3)
    _centered(draw, (cx, cy), name.upper(), _font(16, bold=True), (255, 255, 255))


def render(scene: dict) -> Image.Image:
    _catch_input(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((18, 16, WIDTH - 18, HEIGHT - 16), radius=20, fill=PAPER, outline=(193, 201, 204), width=3)
    _centered(draw, (WIDTH / 2, 43), "LEARN BOTH OUTPUT TRENDS", _font(29, bold=True), INK)
    _centered(draw, (WIDTH / 2, 75), "each machine multiplies, then adds its fixed offset", _font(18, bold=True), MUTED)
    for name, bounds, machine_x in (
        ("blue", (40, 105, 535, 420), 290),
        ("orange", (565, 105, 1060, 420), 810),
    ):
        x0, y0, x1, y1 = bounds
        draw.rounded_rectangle(bounds, radius=16, fill=(248, 249, 247), outline=MACHINE_COLORS[name], width=4)
        draw.rounded_rectangle((x0 + 4, y0 + 4, x1 - 4, y0 + 58), radius=11, fill=MACHINE_COLORS[name])
        _centered(draw, ((x0 + x1) / 2, y0 + 31), f"{name.upper()} EXAMPLES", _font(22, bold=True), (255, 255, 255))
        examples = _examples(scene["machines"][name]["examples"], name)
        for (inp, out), y in zip(examples, CAL_Y, strict=True):
            in_x, out_x = (145, 435) if name == "blue" else (665, 955)
            _draw_number(draw, in_x, y, inp)
            _draw_arrow(draw, in_x + 48, machine_x - 65, y)
            _draw_machine(draw, machine_x, y, name)
            _draw_arrow(draw, machine_x + 65, out_x - 48, y)
            _draw_number(draw, out_x, y, out)
    draw.rounded_rectangle((38, 445, 1062, 680), radius=17, fill=(239, 249, 247), outline=QUERY_COLOR, width=4)
    _centered(draw, (WIDTH / 2, 478), "QUESTION — START WITH INPUT 0", _font(23, bold=True), QUERY_COLOR)
    _centered(draw, (WIDTH / 2, 520), "FIRST WHOLE-NUMBER INPUT WHERE", _font(20, bold=True), MUTED)
    _centered(draw, (430, 565), "ORANGE", _font(26, bold=True), MACHINE_COLORS["orange"])
    _centered(draw, (550, 565), "≥", _font(37, bold=True), INK)
    _centered(draw, (665, 565), "BLUE", _font(26, bold=True), MACHINE_COLORS["blue"])
    for value, x in zip(range(5, 10), (290, 420, 550, 680, 810), strict=True):
        draw.rounded_rectangle((x - 43, 606, x + 43, 663), radius=12, fill=(255, 255, 255), outline=(151, 168, 169), width=3)
        _centered(draw, (x, 634), str(value), _font(30, bold=True), INK)
    return image
