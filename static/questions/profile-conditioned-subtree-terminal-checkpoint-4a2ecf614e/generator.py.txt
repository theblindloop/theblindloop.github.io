"""Deterministic renderer and analytic arm for a rooted terminal-count tree."""

from __future__ import annotations

import random
from collections import deque
from PIL import Image, ImageDraw

BACKGROUND = (247, 249, 252)
DARK = (31, 36, 44)
GREEN = (32, 170, 92)
GRAY = (132, 141, 153)
COLORS = {
    "orange": (242, 139, 38),
    "purple": (142, 68, 173),
    "blue": (45, 117, 220),
}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    dx, dy = rng.randint(-8, 8), rng.randint(-6, 6)
    lanes = [64 + dy, 165 + dy, 266 + dy]
    counts = [1, 2, 3]
    rng.shuffle(counts)
    target_lane = counts.index(3)
    target_color = list(COLORS)[int(seed) % 3]
    remaining = [c for c in COLORS if c != target_color]
    rng.shuffle(remaining)
    lane_colors = remaining[:]
    lane_colors.insert(target_lane, target_color)

    positions: dict[str, list[int]] = {"root": [46 + dx, 165 + dy]}
    edges: list[list[str]] = []
    checkpoint_nodes: list[str] = []
    checkpoint_colors: dict[str, str] = {}
    terminal_nodes: list[str] = []
    offsets = {1: [0], 2: [-27, 27], 3: [-36, 0, 36]}
    for lane, (y, count, color) in enumerate(zip(lanes, counts, lane_colors)):
        cp = f"checkpoint_{lane}"
        checkpoint_nodes.append(cp)
        checkpoint_colors[cp] = color
        positions[cp] = [176 + dx, y]
        edges.append(["root", cp])
        for number, off in enumerate(offsets[count]):
            terminal = f"terminal_{lane}_{number}"
            terminal_nodes.append(terminal)
            positions[terminal] = [326 + dx, y + off]
            edges.append([cp, terminal])
    return {
        "width": 380,
        "height": 330,
        "positions": positions,
        "edges": edges,
        "root_node": "root",
        "checkpoint_nodes": checkpoint_nodes,
        "checkpoint_colors": checkpoint_colors,
        "terminal_nodes": terminal_nodes,
    }


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    positions = scene["positions"]
    edges = scene["edges"]
    root_node = scene["root_node"]
    checkpoint_nodes = scene["checkpoint_nodes"]
    checkpoint_colors = scene["checkpoint_colors"]
    terminal_nodes = scene["terminal_nodes"]
    scale = 4
    image = Image.new("RGB", (width * scale, height * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)

    def point(name: str) -> tuple[int, int]:
        x, y = positions[name]
        return int(x * scale), int(y * scale)

    for a, b in edges:
        draw.line([point(a), point(b)], fill=DARK, width=4 * scale)
    radius = 21 * scale
    rx, ry = point(root_node)
    draw.ellipse((rx - radius, ry - radius, rx + radius, ry + radius), fill=GREEN)
    for node in checkpoint_nodes:
        x, y = point(node)
        draw.ellipse(
            (x - radius, y - radius, x + radius, y + radius),
            fill=COLORS[checkpoint_colors[node]],
        )
    half = 11 * scale
    for node in terminal_nodes:
        x, y = point(node)
        draw.rectangle((x - half, y - half, x + half, y + half), fill=GRAY)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def _descendant_counts(scene: dict) -> dict[str, int]:
    edges = scene["edges"]
    root = scene["root_node"]
    checkpoints = scene["checkpoint_nodes"]
    colors = scene["checkpoint_colors"]
    terminals = set(scene["terminal_nodes"])
    graph: dict[str, set[str]] = {}
    for a, b in edges:
        graph.setdefault(a, set()).add(b)
        graph.setdefault(b, set()).add(a)
    parent = {root: None}
    queue = deque([root])
    while queue:
        node = queue.popleft()
        for nxt in graph[node]:
            if nxt not in parent:
                parent[nxt] = node
                queue.append(nxt)
    result = {}
    for cp in checkpoints:
        count = 0
        for leaf in terminals:
            node = leaf
            while node is not None and node != cp:
                node = parent.get(node)
            count += node == cp
        result[colors[cp]] = count
    return result


def analytic_gold(scene: dict) -> str:
    counts = _descendant_counts(scene)
    winners = [color for color, count in counts.items() if count == 3]
    if len(winners) != 1:
        raise ValueError(f"expected one three-terminal checkpoint, got {counts}")
    return winners[0]


def margin(scene: dict) -> float:
    counts = _descendant_counts(scene)
    distances = sorted(abs(count - 3) for count in counts.values())
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    try:
        counts = _descendant_counts(scene)
        return list(counts.values()).count(3) != 1 or margin(scene) < 1.0
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
