"""Image-only inverse arm: recover markers, links, rooted tree, and leaf counts."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

PALETTE = np.array(
    [
        (247, 249, 252),
        (31, 36, 44),
        (32, 170, 92),
        (242, 139, 38),
        (142, 68, 173),
        (45, 117, 220),
        (132, 141, 153),
    ],
    dtype=np.int16,
)
COLOR_NAMES = {3: "orange", 4: "purple", 5: "blue"}


def _components(mask: np.ndarray, minimum: int = 8) -> list[np.ndarray]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        pixels = []
        while stack:
            cy, cx = stack.pop()
            pixels.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(pixels) >= minimum:
            result.append(np.asarray(pixels, dtype=np.float64))
    return result


def _center(component: np.ndarray) -> tuple[float, float]:
    return float(component[:, 1].mean()), float(component[:, 0].mean())


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)

    def mask_for(index: int, tolerance: int = 24) -> np.ndarray:
        color = PALETTE[index].astype(np.int32)
        return ((arr - color) ** 2).sum(axis=2) <= tolerance**2

    nodes: list[tuple[str, tuple[float, float]]] = []
    green = _components(mask_for(2), 80)
    if len(green) != 1:
        raise ValueError("green root marker missing or split")
    nodes.append(("root", _center(green[0])))
    checkpoint_ids: list[str] = []
    checkpoint_color: dict[str, str] = {}
    for label, color in COLOR_NAMES.items():
        comps = _components(mask_for(label), 80)
        if len(comps) != 1:
            raise ValueError(f"expected one {color} checkpoint")
        node = f"checkpoint_{color}"
        checkpoint_ids.append(node)
        checkpoint_color[node] = color
        nodes.append((node, _center(comps[0])))
    terminals = _components(mask_for(6), 80)
    if len(terminals) != 6:
        raise ValueError(f"expected six gray terminals, found {len(terminals)}")
    terminal_ids = []
    for index, component in enumerate(terminals):
        node = f"terminal_{index}"
        terminal_ids.append(node)
        nodes.append((node, _center(component)))

    graph = {name: set() for name, _ in nodes}
    dark_components = _components(mask_for(1, 18), 10)
    if len(dark_components) != 9:
        raise ValueError(f"expected nine separated links, found {len(dark_components)}")
    for component in dark_components:
        xy = component[:, ::-1]
        ranked = []
        for name, (cx, cy) in nodes:
            distance2 = ((xy[:, 0] - cx) ** 2 + (xy[:, 1] - cy) ** 2).min()
            ranked.append((float(distance2), name))
        ranked.sort()
        (d1, a), (d2, b) = ranked[:2]
        if d2 > 30.0**2 or a == b:
            raise ValueError("link endpoints cannot be assigned to markers")
        graph[a].add(b)
        graph[b].add(a)

    parent: dict[str, str | None] = {"root": None}
    queue = deque(["root"])
    while queue:
        node = queue.popleft()
        for nxt in graph[node]:
            if nxt not in parent:
                parent[nxt] = node
                queue.append(nxt)
    if len(parent) != len(nodes):
        raise ValueError("recovered graph is disconnected")
    winners = []
    for checkpoint in checkpoint_ids:
        count = 0
        for leaf in terminal_ids:
            node: str | None = leaf
            while node is not None and node != checkpoint:
                node = parent.get(node)
            count += node == checkpoint
        if count == 3:
            winners.append(checkpoint_color[checkpoint])
    if len(winners) != 1:
        raise ValueError("pixel graph has no unique three-terminal checkpoint")
    return winners[0]
