"""Deterministic two-axis geometric-transformation renderer and analytic arm.

Each local tile exposes two differently colored, nonparallel mirror axes.  The
visible program is to reflect a complete asymmetric token across the violet
axis and then across the gold axis.  One hollow target is the exact composed
silhouette; the other targets are deliberately plausible one-step or
reverse-order near misses.
"""

from __future__ import annotations

import copy
import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 1020
HEIGHT = 740
PANEL_W = 480
PANEL_H = 340
MARGIN_X = 20
MARGIN_Y = 20
GAP_X = 20
GAP_Y = 20
PANEL_LABELS = ("A", "B", "C", "D")

BACKGROUND = (247, 249, 250)
PANEL_FILL = (255, 255, 255)
FRAME = (78, 98, 111)
LABEL = (50, 65, 78)
SOURCE = (30, 66, 105)
TARGET_FILL = (255, 244, 216)
TARGET_EDGE = (210, 94, 53)
VIOLET_AXIS = (105, 78, 151)
GOLD_AXIS = (183, 125, 36)

QUARANTINE_MARGIN = 20.0

# A large concave, handed token.  The flat collar, notch, and unequal points
# make both the reflection order and the full silhouette observable.
BASE_SHAPE = (
    (-48, -30),
    (-15, -30),
    (-1, -49),
    (18, -28),
    (47, -21),
    (29, -2),
    (52, 18),
    (17, 20),
    (8, 50),
    (-12, 24),
    (-42, 40),
    (-31, 10),
    (-53, 2),
)


def panel_origin(slot: int) -> tuple[int, int]:
    return (
        MARGIN_X + (int(slot) % 2) * (PANEL_W + GAP_X),
        MARGIN_Y + (int(slot) // 2) * (PANEL_H + GAP_Y),
    )


def _rotate_point(
    point: tuple[float, float], center: tuple[float, float], angle: float
) -> tuple[float, float]:
    radians = math.radians(angle)
    c, s = math.cos(radians), math.sin(radians)
    dx, dy = point[0] - center[0], point[1] - center[1]
    return (center[0] + c * dx - s * dy, center[1] + s * dx + c * dy)


def _rounded(points: Iterable[tuple[float, float]]) -> list[list[int]]:
    return [[int(round(x)), int(round(y))] for x, y in points]


def _centroid(points: Iterable[Iterable[float]]) -> tuple[float, float]:
    values = [(float(p[0]), float(p[1])) for p in points]
    return (
        sum(p[0] for p in values) / len(values),
        sum(p[1] for p in values) / len(values),
    )


def _source_polygon(center: tuple[float, float], angle: float) -> list[list[int]]:
    return _rounded(
        _rotate_point((center[0] + dx, center[1] + dy), center, angle)
        for dx, dy in BASE_SHAPE
    )


def _reflect(
    point: Iterable[float],
    axis_point: Iterable[float],
    axis_direction: Iterable[float],
) -> tuple[float, float]:
    px, py = float(point[0]), float(point[1])
    ax, ay = float(axis_point[0]), float(axis_point[1])
    dx, dy = float(axis_direction[0]), float(axis_direction[1])
    norm = math.hypot(dx, dy) or 1.0
    dx, dy = dx / norm, dy / norm
    vx, vy = px - ax, py - ay
    projection = vx * dx + vy * dy
    foot = (ax + projection * dx, ay + projection * dy)
    return (2.0 * foot[0] - px, 2.0 * foot[1] - py)


def _reflect_polygon(
    points: Iterable[Iterable[float]], axis_point: Iterable[float], axis_direction: Iterable[float]
) -> list[tuple[float, float]]:
    return [_reflect(point, axis_point, axis_direction) for point in points]


def _transformed_polygon(*, panel: dict) -> list[list[int]]:
    first = _reflect_polygon(
        panel["source"], panel["axis_one_point"], panel["axis_one_direction"]
    )
    second = _reflect_polygon(
        first, panel["axis_two_point"], panel["axis_two_direction"]
    )
    return _rounded(second)


def _reverse_polygon(*, panel: dict) -> list[list[int]]:
    first = _reflect_polygon(
        panel["source"], panel["axis_two_point"], panel["axis_two_direction"]
    )
    second = _reflect_polygon(
        first, panel["axis_one_point"], panel["axis_one_direction"]
    )
    return _rounded(second)


def _first_reflection_polygon(*, panel: dict) -> list[list[int]]:
    return _rounded(
        _reflect_polygon(
            panel["source"], panel["axis_one_point"], panel["axis_one_direction"]
        )
    )


def _second_reflection_polygon(*, panel: dict) -> list[list[int]]:
    return _rounded(
        _reflect_polygon(
            panel["source"], panel["axis_two_point"], panel["axis_two_direction"]
        )
    )


def _mean_vertex_error(left: list[list[int]], right: list[list[int]]) -> float:
    return sum(math.hypot(a[0] - b[0], a[1] - b[1]) for a, b in zip(left, right)) / len(left)


def _inside(points: Iterable[Iterable[int]]) -> bool:
    # The top strip is reserved for the tile label; the lower strip keeps all
    # polygon ink away from the rounded panel frame.
    return all(30 <= int(p[0]) <= 450 and 70 <= int(p[1]) <= 320 for p in points)


def _signed_line_distance(
    point: Iterable[float], axis_point: Iterable[float], axis_direction: Iterable[float]
) -> float:
    px, py = float(point[0]), float(point[1])
    ax, ay = float(axis_point[0]), float(axis_point[1])
    dx, dy = float(axis_direction[0]), float(axis_direction[1])
    norm = math.hypot(dx, dy) or 1.0
    return (dx * (py - ay) - dy * (px - ax)) / norm


def _line_clear(points: Iterable[Iterable[float]], axis_point: Iterable[float], axis_direction: Iterable[float]) -> bool:
    distances = [
        _signed_line_distance(point, axis_point, axis_direction) for point in points
    ]
    if not distances:
        return False
    return min(abs(value) for value in distances) >= 16.0 and (
        all(value > 16.0 for value in distances)
        or all(value < -16.0 for value in distances)
    )


def _panel_geometry_clear(panel: dict) -> bool:
    for points in (panel["source"], panel["target"]):
        if not _inside(points):
            return False
        if not _line_clear(points, panel["axis_one_point"], panel["axis_one_direction"]):
            return False
        if not _line_clear(points, panel["axis_two_point"], panel["axis_two_direction"]):
            return False
    source_center = _centroid(panel["source"])
    target_center = _centroid(panel["target"])
    return math.hypot(source_center[0] - target_center[0], source_center[1] - target_center[1]) >= 62.0


def _scores(scene: dict) -> list[float]:
    values = []
    for panel in sorted(scene["panels"], key=lambda item: int(item["slot"])):
        values.append(_mean_vertex_error(_transformed_polygon(panel=panel), panel["target"]))
    return values


def _direction(angle: float) -> list[float]:
    radians = math.radians(angle)
    return [round(math.cos(radians), 8), round(math.sin(radians), 8)]


def _decoy_target(rng: random.Random, panel: dict) -> list[list[int]] | None:
    expected = _transformed_polygon(panel=panel)
    source_center = _centroid(panel["source"])

    def valid(candidate: list[list[int]]) -> bool:
        if not _inside(candidate):
            return False
        if _mean_vertex_error(expected, candidate) < 24.0:
            return False
        if not _line_clear(candidate, panel["axis_one_point"], panel["axis_one_direction"]):
            return False
        if not _line_clear(candidate, panel["axis_two_point"], panel["axis_two_direction"]):
            return False
        candidate_center = _centroid(candidate)
        return math.hypot(
            source_center[0] - candidate_center[0],
            source_center[1] - candidate_center[1],
        ) >= 62.0

    # Prefer mechanism-revealing near misses, but try all variants because a
    # particular two-axis placement can put a one-step result outside the
    # local tile.  The fallback still keeps the target large, separated, and
    # close enough to be a genuine whole-shape comparison.
    for attempt in range(80):
        decoy_kind = ("reverse", "first", "second", "near")[attempt % 4]
        if decoy_kind == "reverse":
            candidate = _reverse_polygon(panel=panel)
        elif decoy_kind == "first":
            candidate = _first_reflection_polygon(panel=panel)
        elif decoy_kind == "second":
            candidate = _second_reflection_polygon(panel=panel)
        else:
            center = _centroid(expected)
            rotation = rng.choice((-1.0, 1.0)) * rng.uniform(9.0, 21.0)
            offset_angle = rng.uniform(-180.0, 180.0)
            offset_length = rng.uniform(18.0, 34.0)
            offset = (
                offset_length * math.cos(math.radians(offset_angle)),
                offset_length * math.sin(math.radians(offset_angle)),
            )
            candidate = _rounded(
                (
                    _rotate_point((float(point[0]), float(point[1])), center, rotation)[0] + offset[0],
                    _rotate_point((float(point[0]), float(point[1])), center, rotation)[1] + offset[1],
                )
                for point in expected
            )
        if valid(candidate):
            return candidate
    return None


def _make_panel(rng: random.Random, slot: int, is_winner: bool) -> dict:
    """Rejection-sample one tile with two readable, nonparallel axes."""

    for _ in range(4000):
        center = (rng.randint(82, 177), rng.randint(137, 247))
        source = _source_polygon(center, rng.uniform(-30.0, 30.0))

        hinge = [rng.randint(215, 295), rng.randint(135, 230)]
        angle_one = rng.uniform(34.0, 146.0)
        delta = rng.choice((-1.0, 1.0)) * rng.uniform(29.0, 68.0)
        angle_two = angle_one + delta
        if not (18.0 <= angle_two <= 162.0):
            continue
        axis_one_direction = _direction(angle_one)
        axis_two_direction = _direction(angle_two)
        provisional = {
            "slot": int(slot),
            "source": source,
            "axis_one_point": list(hinge),
            "axis_one_direction": axis_one_direction,
            "axis_two_point": list(hinge),
            "axis_two_direction": axis_two_direction,
        }
        expected = _transformed_polygon(panel=provisional)
        if not _inside(source) or not _inside(expected):
            continue
        if not _line_clear(source, hinge, axis_one_direction) or not _line_clear(source, hinge, axis_two_direction):
            continue
        if not _line_clear(expected, hinge, axis_one_direction) or not _line_clear(expected, hinge, axis_two_direction):
            continue
        if math.hypot(_centroid(source)[0] - _centroid(expected)[0], _centroid(source)[1] - _centroid(expected)[1]) < 62.0:
            continue

        provisional["target"] = expected if is_winner else []
        if not is_winner:
            candidate = _decoy_target(rng, provisional)
            if candidate is None:
                continue
            provisional["target"] = candidate
        if not _panel_geometry_clear(provisional):
            continue
        return provisional
    raise RuntimeError(f"could not sample tile {slot}")


def sample_scene(seed: int) -> dict:
    """Sample four tiles with one exact two-axis reflection match."""

    rng = random.Random(int(seed))
    winner = int(seed) % 4
    for _ in range(400):
        panels = [_make_panel(rng, slot, slot == winner) for slot in range(4)]
        scene = {"panels": panels}
        scores = _scores(scene)
        if scores[winner] <= 1.0e-9 and min(scores[index] for index in range(4) if index != winner) >= 24.0:
            return scene
    raise RuntimeError("could not sample a clear four-tile scene")


def _clip_line(
    point: tuple[float, float],
    direction: tuple[float, float],
    bounds: tuple[float, float, float, float],
) -> tuple[float, float] | None:
    """Return the parameter interval for a line clipped to a rectangle."""

    low, high = -1.0e9, 1.0e9
    xmin, ymin, xmax, ymax = bounds
    for coordinate, delta, lo, hi in (
        (point[0], direction[0], xmin, xmax),
        (point[1], direction[1], ymin, ymax),
    ):
        if abs(delta) < 1.0e-12:
            if coordinate < lo or coordinate > hi:
                return None
            continue
        first, second = (lo - coordinate) / delta, (hi - coordinate) / delta
        if first > second:
            first, second = second, first
        low, high = max(low, first), min(high, second)
    return (low, high) if low <= high else None


def _global(origin: tuple[int, int], point: Iterable[float]) -> tuple[float, float]:
    return origin[0] + float(point[0]), origin[1] + float(point[1])


def _draw_axis(
    draw: ImageDraw.ImageDraw,
    origin: tuple[int, int],
    point: Iterable[float],
    direction: Iterable[float],
    color: tuple[int, int, int],
) -> None:
    local_point = (float(point[0]), float(point[1]))
    local_direction = (float(direction[0]), float(direction[1]))
    clipped = _clip_line(local_point, local_direction, (18.0, 54.0, 462.0, 326.0))
    if clipped is None:
        return
    start, end = clipped
    cursor = start
    while cursor < end:
        dash_end = min(cursor + 11.0, end)
        p0 = _global(
            origin,
            (
                local_point[0] + cursor * local_direction[0],
                local_point[1] + cursor * local_direction[1],
            ),
        )
        p1 = _global(
            origin,
            (
                local_point[0] + dash_end * local_direction[0],
                local_point[1] + dash_end * local_direction[1],
            ),
        )
        draw.line((p0, p1), fill=color, width=3)
        cursor += 22.0


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for panel in sorted(scene["panels"], key=lambda item: int(item["slot"])):
        slot = int(panel["slot"])
        origin = panel_origin(slot)
        x0, y0 = origin
        x1, y1 = x0 + PANEL_W, y0 + PANEL_H
        draw.rounded_rectangle(
            (x0, y0, x1, y1), radius=12, fill=PANEL_FILL, outline=FRAME, width=3
        )
        draw.text((x0 + 16, y0 + 12), f"tile-{PANEL_LABELS[slot]}", fill=LABEL)
        _draw_axis(
            draw,
            origin,
            panel["axis_one_point"],
            panel["axis_one_direction"],
            VIOLET_AXIS,
        )
        _draw_axis(
            draw,
            origin,
            panel["axis_two_point"],
            panel["axis_two_direction"],
            GOLD_AXIS,
        )
        target = [_global(origin, point) for point in panel["target"]]
        source = [_global(origin, point) for point in panel["source"]]
        draw.polygon(target, fill=TARGET_FILL, outline=TARGET_EDGE)
        draw.line(target + [target[0]], fill=TARGET_EDGE, width=3, joint="curve")
        draw.polygon(source, fill=SOURCE, outline=SOURCE)
        draw.line(source + [source[0]], fill=SOURCE, width=2, joint="curve")
    return image


def analytic_gold(scene: dict) -> str:
    panels = sorted(scene["panels"], key=lambda item: int(item["slot"]))
    scores = [
        _mean_vertex_error(_transformed_polygon(panel=panel), panel["target"])
        for panel in panels
    ]
    winner = min(range(len(scores)), key=lambda index: scores[index])
    return f"tile-{PANEL_LABELS[winner]}"


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene))
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    scores = _scores(scene)
    best = min(scores)
    tied = len({round(value, 9) for value in scores if value <= best + 1.0e-9}) != 1
    return tied or margin(scene) < QUARANTINE_MARGIN or any(
        not _panel_geometry_clear(panel) for panel in scene["panels"]
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["panels"] = list(reversed(twin["panels"]))
    return [("panel_record_order", twin)]
