"""Scene-free pixel oracle for the even-split-shelf world.

Receives only the rendered PNG. It segments the flat car colour, splits the
canvas into shelf bands from the rows of cars, groups each band's cars into
trains by the wide background gaps between runs, and counts the cars in every
train. It then searches, for each shelf, every way of dealing that shelf's
trains into two groups and reports the unique shelf that admits a deal with
equal car totals.

It imports no renderer, prompt, generator, verifier or gold code and never
sees latent scene data.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_CAR = (46, 104, 186)
_TRAIN_GAP = 15  # background wider than this separates two trains

_NO_ANSWER = "shelf-0"


def _mask(arr: np.ndarray, colour: tuple[int, int, int]) -> np.ndarray:
    return (
        (arr[:, :, 0] == colour[0])
        & (arr[:, :, 1] == colour[1])
        & (arr[:, :, 2] == colour[2])
    )


def _runs(flags: np.ndarray) -> list[tuple[int, int]]:
    out: list[tuple[int, int]] = []
    start = None
    for i, on in enumerate(flags):
        if on and start is None:
            start = i
        elif not on and start is not None:
            out.append((start, i - 1))
            start = None
    if start is not None:
        out.append((start, len(flags) - 1))
    return out


def _splittable(values: list[int]) -> bool:
    total = sum(values)
    if total % 2:
        return False
    reachable = {0}
    for v in values:
        reachable |= {s + v for s in reachable}
    return (total // 2) in reachable


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    cars = _mask(arr, _CAR)

    bands = _runs(cars.any(axis=1))
    if len(bands) < 2:
        return _NO_ANSWER

    hits: list[int] = []
    for index, (top, bottom) in enumerate(bands):
        band = cars[top : bottom + 1]
        car_runs = _runs(band.any(axis=0))
        if not car_runs:
            return _NO_ANSWER
        trains: list[int] = [1]
        for previous, current in zip(car_runs, car_runs[1:]):
            if current[0] - previous[1] - 1 > _TRAIN_GAP:
                trains.append(1)
            else:
                trains[-1] += 1
        if _splittable(trains):
            hits.append(index)

    if len(hits) != 1:
        return _NO_ANSWER
    return f"shelf-{hits[0] + 1}"
