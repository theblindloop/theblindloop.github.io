"""Even-split-shelf question world.

Four shelves stand one above another. Every shelf carries the same number of
trains and the same total number of square cars, so ink, train count and shelf
length carry no information. Exactly one shelf's trains can be dealt into two
groups whose car totals are equal; the label is that shelf.

The executable content is an exact-partition search: three shelves must be
certified impossible and one must be certified possible, and no formula
evaluated on a shelf decides it.
"""

from __future__ import annotations

import random
from itertools import combinations

from PIL import Image, ImageDraw

WIDTH = 1160
HEIGHT = 500

N_SHELVES = 4
N_TRAINS = 6
CAR = 20
CAR_GAP = 5
PITCH = CAR + CAR_GAP
MIN_TRAIN_GAP = 30
MAX_TRAIN_GAP = 44
RAIL_H = 8
TOP_MARGIN = 40
BAND = 120
SIDE_MARGIN = 30

BG = (246, 246, 240)
RAIL = (128, 136, 148)
CARCOL = (46, 104, 186)

TOTALS = (32, 36)
VMIN = 1
VMAX = 12
QUARANTINE_MARGIN = 2.0


# ---------------------------------------------------------------- arithmetic

def _subset_sums(values: tuple[int, ...]) -> set[int]:
    sums = {0}
    for v in values:
        sums |= {s + v for s in sums}
    return sums


def _deviation(values: tuple[int, ...]) -> float:
    """Smallest |group - half| over all ways of splitting the trains."""
    half = sum(values) / 2.0
    return min(abs(s - half) for s in _subset_sums(values))


def _balanced_partitions(values: tuple[int, ...]) -> set[tuple[tuple[int, ...], tuple[int, ...]]]:
    """Distinct equal-car partitions, written as a sorted pair of multisets."""
    total = sum(values)
    if total % 2:
        return set()
    half = total // 2
    out = set()
    idx = range(len(values))
    for size in range(1, len(values)):
        for pick in combinations(idx, size):
            if sum(values[i] for i in pick) != half:
                continue
            a = tuple(sorted(values[i] for i in pick))
            b = tuple(sorted(values[i] for i in idx if i not in pick))
            out.add(tuple(sorted((a, b))))
    return out


def _balanced_sizes(values: tuple[int, ...]) -> set[int]:
    total = sum(values)
    half = total // 2
    sizes = set()
    idx = range(len(values))
    for size in range(1, len(values)):
        for pick in combinations(idx, size):
            if sum(values[i] for i in pick) == half:
                sizes.add(size)
    return sizes


def _best_half_and_half(values: tuple[int, ...]) -> float:
    """How close the three-against-three deal can come to an even load."""
    half = sum(values) / 2.0
    return min(
        abs(sum(values[i] for i in pick) - half)
        for pick in combinations(range(len(values)), len(values) // 2)
    )


def _multisets(total: int) -> list[tuple[int, ...]]:
    out: list[tuple[int, ...]] = []

    def walk(prefix: tuple[int, ...], low: int, remaining: int) -> None:
        slots = N_TRAINS - len(prefix)
        if slots == 0:
            if remaining == 0:
                out.append(prefix)
            return
        for v in range(low, VMAX + 1):
            if v * slots > remaining:
                break
            if VMAX * (slots - 1) < remaining - v:
                continue
            walk(prefix + (v,), v, remaining - v)

    walk((), VMIN, total)
    return out


_POOL_CACHE: dict[int, tuple[list, list]] = {}


def _pools(total: int) -> tuple[list, list]:
    """(answer-eligible multisets, decoy multisets keyed with their deviation)."""
    if total in _POOL_CACHE:
        return _POOL_CACHE[total]
    answers = []
    decoys = []
    for values in _multisets(total):
        if max(values) > total // 2 - 1:
            continue  # never let one dominant train decide a shelf
        parts = _balanced_partitions(values)
        if parts:
            if len(parts) != 1:
                continue  # the winning deal must be unique
            if N_TRAINS // 2 in _balanced_sizes(values):
                continue  # a three-against-three deal must never win
            answers.append(values)
        else:
            decoys.append((values, _deviation(values)))
    _POOL_CACHE[total] = (answers, decoys)
    return answers, decoys


# ------------------------------------------------------------------ sampling

def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 2654435761 % (2**63))
    for _ in range(4000):
        total = rng.choice(TOTALS)
        answers, decoys = _pools(total)
        if not answers or len(decoys) < 3:
            continue
        answer = rng.choice(answers)
        peak = max(answer)
        target = 1 if rng.random() < 0.2 else 2
        pool = [d for d in decoys if d[1] >= target and d[0] != answer]
        if len(pool) < 3:
            continue
        chosen = rng.sample(pool, 3)
        if len({d[0] for d in chosen}) != 3:
            continue
        # the closest wrong shelf must sit exactly at the sampled margin
        if min(d[1] for d in chosen) != target:
            continue
        # the longest train in the picture never marks the answering shelf,
        # and neither does the shortest
        if peak > max(max(d[0]) for d in chosen):
            continue
        if peak < min(max(d[0]) for d in chosen):
            continue
        # at least one decoy must look balanced as three trains against three
        if min(_best_half_and_half(d[0]) for d in chosen) > 2:
            continue

        shelves_multisets = [answer] + [d[0] for d in chosen]
        order = list(range(N_SHELVES))
        rng.shuffle(order)
        placed = [None] * N_SHELVES
        for slot, which in enumerate(order):
            values = list(shelves_multisets[slot])
            rng.shuffle(values)
            placed[which] = values

        gaps = []
        for _shelf in range(N_SHELVES):
            gaps.append([rng.randint(MIN_TRAIN_GAP, MAX_TRAIN_GAP) for _ in range(N_TRAINS - 1)])

        paint_order = list(range(N_SHELVES))
        rng.shuffle(paint_order)

        scene = {"shelves": placed, "gaps": gaps, "paint_order": paint_order}
        if _decide(scene) is not None:
            return scene
    raise RuntimeError("sampler failed to find a scene")


# -------------------------------------------------------------------- labels

def _decide(scene: dict) -> int | None:
    hits = [
        i
        for i, values in enumerate(scene["shelves"])
        if _deviation(tuple(values)) == 0.0
    ]
    if len(hits) != 1:
        return None
    return hits[0]


def analytic_gold(scene: dict) -> str:
    index = _decide(scene)
    if index is None:
        raise ValueError("scene has no unique splittable shelf")
    return f"shelf-{index + 1}"


def margin(scene: dict) -> float:
    index = _decide(scene)
    if index is None:
        raise ValueError("scene has no unique splittable shelf")
    return float(
        min(
            _deviation(tuple(values))
            for i, values in enumerate(scene["shelves"])
            if i != index
        )
    )


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    order = list(scene["paint_order"])
    rotated = dict(scene)
    rotated["paint_order"] = order[1:] + order[:1]
    reversed_ = dict(scene)
    reversed_["paint_order"] = list(reversed(order))
    return [("paint_order_rotate", rotated), ("paint_order_reverse", reversed_)]


# ------------------------------------------------------------------ renderer

def _shelf_width(values: list[int], gaps: list[int]) -> int:
    cars = sum(values)
    return cars * PITCH - N_TRAINS * CAR_GAP + sum(gaps)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)

    for shelf in scene["paint_order"]:
        values = scene["shelves"][shelf]
        gaps = scene["gaps"][shelf]
        rail_top = TOP_MARGIN + shelf * BAND + CAR + 12
        draw.rectangle(
            [SIDE_MARGIN, rail_top, WIDTH - SIDE_MARGIN - 1, rail_top + RAIL_H - 1],
            fill=RAIL,
        )
        width = _shelf_width(values, gaps)
        x = (WIDTH - width) // 2
        top = rail_top - CAR
        for t, count in enumerate(values):
            for c in range(count):
                left = x + c * PITCH
                draw.rectangle([left, top, left + CAR - 1, rail_top - 1], fill=CARCOL)
            x += count * PITCH - CAR_GAP
            if t < len(gaps):
                x += gaps[t]
    return image
