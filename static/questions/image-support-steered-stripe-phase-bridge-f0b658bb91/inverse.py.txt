"""Independent pixel-only inverse for the stripe-phase relation.

The inverse discovers saturated connected regions, infers their pair axis, finds
the three pale slots from transverse occupancy troughs, and compares their
translation phases. It has no access to latent scene data or renderer code.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    """The required two-window, three-slot evidence is absent or ambiguous."""


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]:
            continue
        queue = deque([(y0, x0)])
        seen[y0, x0] = True
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= 180:
            output.append(np.asarray(points, dtype=np.int32))
    return output


def _groups(indices: np.ndarray) -> list[np.ndarray]:
    if len(indices) == 0:
        return []
    breaks = np.flatnonzero(np.diff(indices) > 1) + 1
    return [chunk for chunk in np.split(indices, breaks) if len(chunk)]


def _slot_centers(mask: np.ndarray, points: np.ndarray, horizontal: bool) -> np.ndarray:
    ys, xs = points[:, 0], points[:, 1]
    y0, y1 = int(ys.min()), int(ys.max())
    x0, x1 = int(xs.min()), int(xs.max())
    if horizontal:
        if (x1 - x0) > 35 or (y1 - y0) < 45:
            raise OracleAbstention("window geometry is inconsistent with the inferred axis")
        core = mask[y0 : y1 + 1, x0 + 5 : x1 - 4]
        profile = core.mean(axis=1)
        candidates = np.flatnonzero(profile < 0.28)
        candidates = candidates[(candidates >= 6) & (candidates <= len(profile) - 7)]
        grouped = _groups(candidates)
        centers = np.asarray([y0 + float(chunk.mean()) for chunk in grouped])
    else:
        if (y1 - y0) > 35 or (x1 - x0) < 45:
            raise OracleAbstention("window geometry is inconsistent with the inferred axis")
        core = mask[y0 + 5 : y1 - 4, x0 : x1 + 1]
        profile = core.mean(axis=0)
        candidates = np.flatnonzero(profile < 0.28)
        candidates = candidates[(candidates >= 6) & (candidates <= len(profile) - 7)]
        grouped = _groups(candidates)
        centers = np.asarray([x0 + float(chunk.mean()) for chunk in grouped])
    if len(centers) != 3:
        raise OracleAbstention(f"expected three stripe slots, recovered {len(centers)}")
    spacing = np.diff(centers)
    if np.any((spacing < 13.0) | (spacing > 19.0)):
        raise OracleAbstention("stripe repetition is not a stable 16-pixel texture")
    return centers


def phase_measurement(image: Image.Image) -> float:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    chroma = array.max(axis=2) - array.min(axis=2)
    saturated = chroma >= 52
    components = _components(saturated)
    if len(components) != 2:
        raise OracleAbstention(f"expected two saturated windows, recovered {len(components)}")

    centroids = np.asarray([[points[:, 1].mean(), points[:, 0].mean()] for points in components])
    delta = centroids[1] - centroids[0]
    horizontal = abs(float(delta[0])) > abs(float(delta[1]))
    order_axis = 0 if horizontal else 1
    order = np.argsort(centroids[:, order_axis])
    first = _slot_centers(saturated, components[int(order[0])], horizontal)
    second = _slot_centers(saturated, components[int(order[1])], horizontal)
    return float(abs(np.median(second - first)))


def decision_from_image(image: Image.Image) -> str:
    displacement = phase_measurement(image)
    if displacement <= 2.0:
        return "matching"
    if displacement >= 6.0:
        return "interleaved"
    raise OracleAbstention("phase displacement lies in the quarantined midpoint band")
