"""Deterministic latent sampler and raster renderer for stripe_phase_bridge."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS = 512
SUPERSAMPLE = 4
BACKGROUND = (248, 247, 242)
PALE = BACKGROUND
PALETTES = (
    (24, 105, 166),
    (176, 55, 92),
    (45, 128, 83),
    (113, 72, 161),
)
NEUTRAL = (205, 204, 198)
PHASE_PERIOD = 16
PHASE_MIDPOINT = 4.0
INTERLEAVE_SHIFT = 8


def _distractors(rng: random.Random) -> list[list[int]]:
    """Canonical, visibly reflected remote neutral marks: x, y, kind, size."""

    anchors = ((72, 78), (438, 82), (76, 430), (432, 424))
    marks: list[list[int]] = []
    for index, (x, y) in enumerate(anchors):
        marks.append(
            [
                x + rng.randint(-13, 13),
                y + rng.randint(-13, 13),
                (index + rng.randrange(3)) % 3,
                7 + rng.randrange(5),
            ]
        )
    return marks


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample balanced classes while independently cycling nuisance factors."""

    rng = random.Random(int(seed) ^ 0x5A17C9)
    phase_class = "matching" if int(seed) % 2 == 0 else "interleaved"
    axis = "horizontal" if (int(seed) // 2) % 2 == 0 else "vertical"
    # Tile centers occupy unique regions of two grid cells separated by one
    # untouched cell. This local inset moves among central grid locations.
    if axis == "horizontal":
        cell_x = 2 + ((int(seed) // 4) % 2)
        cell_y = 2 + ((int(seed) // 8) % 4)
    else:
        cell_x = 2 + ((int(seed) // 8) % 4)
        cell_y = 2 + ((int(seed) // 4) % 2)
    phase_sign = 0 if phase_class == "matching" else (-1 if rng.randrange(2) == 0 else 1)
    # Keep all three shifted slots comfortably inside the rounded window.
    base_offset = rng.choice((-2, 0, 2)) if phase_sign == 0 else -2 * phase_sign
    return {
        "phase_class": phase_class,
        "axis": axis,
        "cell_x": cell_x,
        "cell_y": cell_y,
        "palette_index": rng.randrange(len(PALETTES)),
        "phase_sign": phase_sign,
        "base_offset": base_offset,
        "distractors": _distractors(rng),
    }


def _scale_box(box: tuple[float, float, float, float], scale: int) -> tuple[int, int, int, int]:
    return tuple(int(round(value * scale)) for value in box)  # type: ignore[return-value]


def _draw_neutral_mark(draw: ImageDraw.ImageDraw, mark: list[int], scale: int) -> None:
    x, y, kind, size = mark
    x *= scale
    y *= scale
    size *= scale
    width = max(2, 2 * scale)
    if kind == 0:
        draw.ellipse((x - size, y - size, x + size, y + size), outline=NEUTRAL, width=width)
    elif kind == 1:
        draw.line((x - size, y, x + size, y), fill=NEUTRAL, width=width)
        draw.line((x, y - size, x, y + size), fill=NEUTRAL, width=width)
    else:
        draw.line((x - size, y - size, x + size, y + size), fill=NEUTRAL, width=width)
        draw.line((x - size, y + size, x + size, y - size), fill=NEUTRAL, width=width)


def _tile_centers(scene: dict[str, Any]) -> tuple[tuple[int, int], tuple[int, int]]:
    x = 64 * int(scene["cell_x"]) + 32
    y = 64 * int(scene["cell_y"]) + 32
    if scene["axis"] == "horizontal":
        return (x, y), (x + 128, y)
    return (x, y), (x, y + 128)


def render(scene: dict[str, Any]) -> Image.Image:
    """Render two detached texture samples plus nondiagnostic remote marks."""

    scale = SUPERSAMPLE
    image = Image.new("RGB", (CANVAS * scale, CANVAS * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for mark in scene["distractors"]:
        _draw_neutral_mark(draw, mark, scale)

    color = PALETTES[int(scene["palette_index"])]
    centers = _tile_centers(scene)
    horizontal = scene["axis"] == "horizontal"
    base = int(scene["base_offset"])
    shift = (
        0
        if scene["phase_class"] == "matching"
        else int(scene["phase_sign"]) * INTERLEAVE_SHIFT
    )

    for tile_index, (cx, cy) in enumerate(centers):
        cross_shift = 0 if tile_index == 0 else shift
        if horizontal:
            outer = (cx - 14, cy - 29, cx + 14, cy + 29)
        else:
            outer = (cx - 29, cy - 14, cx + 29, cy + 14)
        draw.rounded_rectangle(_scale_box(outer, scale), radius=3 * scale, fill=color)

        for stripe_offset in (-PHASE_PERIOD, 0, PHASE_PERIOD):
            offset = base + cross_shift + stripe_offset
            if horizontal:
                slot = (cx - 9, cy + offset - 2, cx + 9, cy + offset + 2)
            else:
                slot = (cx + offset - 2, cy - 9, cx + offset + 2, cy + 9)
            draw.rounded_rectangle(_scale_box(slot, scale), radius=1 * scale, fill=PALE)

    return image.resize((CANVAS, CANVAS), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(scene["phase_class"])


def margin(scene: dict[str, Any]) -> float:
    displacement = 0.0 if analytic_gold(scene) == "matching" else float(INTERLEAVE_SHIFT)
    return abs(displacement - PHASE_MIDPOINT)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) <= 2.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
