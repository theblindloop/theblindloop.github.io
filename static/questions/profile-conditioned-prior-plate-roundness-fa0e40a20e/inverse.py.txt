"""Pixel-only inverse arm: decide circle vs oval from the raster alone.

This module receives only the rendered PNG. It imports nothing from renderer,
prompts, generate, verify, or analytic-gold code, and it accepts no latent
scene data. It classifies each pixel to a fixed world palette by nearest
Euclidean color, isolates the plate, and measures whole-object ellipticity as
the ratio of the largest to the smallest radial extent of the plate around its
centroid. That ratio is ~1 for a perfect circle and clearly above 1 for an
oval regardless of the oval's rotation, color, or position.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

# Fixed world palette (must match renderer constants).
BACKGROUND = (255, 255, 255)
PLATE_COLORS = [
    (205, 62, 62),
    (70, 122, 205),
    (212, 158, 42),
]
_PALETTE = [BACKGROUND] + PLATE_COLORS
_BACKGROUND_LABEL = 0
_N_BINS = 360
_OVAL_RATIO_THRESHOLD = 1.10


def _plate_mask(arr: np.ndarray) -> np.ndarray:
    """Nearest-palette label; return True where a plate pixel (not background)."""
    dists = np.stack(
        [np.linalg.norm(arr - np.array(c, dtype=np.float32), axis=-1) for c in _PALETTE],
        axis=-1,
    )
    labels = dists.argmin(axis=-1)
    return labels != _BACKGROUND_LABEL


def _ellipticity_ratio(mask: np.ndarray) -> float:
    ys, xs = np.nonzero(mask)
    if ys.size < 64:
        return 1.0
    cy = float(np.mean(ys))
    cx = float(np.mean(xs))
    dx = xs.astype(np.float64) - cx
    dy = ys.astype(np.float64) - cy
    radii = np.sqrt(dx * dx + dy * dy)
    angles = np.arctan2(dy, dx)
    bin_idx = np.mod(np.floor(angles / (2.0 * np.pi) * _N_BINS).astype(np.int64), _N_BINS)
    bin_max = np.zeros(_N_BINS, dtype=np.float64)
    np.maximum.at(bin_max, bin_idx, radii)
    used = bin_max[bin_max > 0]
    if used.size < 32:
        return 1.0
    r_max = float(used.max())
    r_min = float(used.min())
    if r_min <= 0:
        return 1.0
    return r_max / r_min


def decision_from_image(image: Image.Image) -> str:
    """Return 'yes' if the plate outline is a perfect circle, else 'no'."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    ratio = _ellipticity_ratio(_plate_mask(arr))
    return "yes" if ratio < _OVAL_RATIO_THRESHOLD else "no"
