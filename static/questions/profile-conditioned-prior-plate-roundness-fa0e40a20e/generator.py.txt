"""Prior-conflict plate-roundness world: report circle vs oval from pixels.

A single dinner plate is drawn as seen from directly above. A familiar
canonical prior asserts that a plate is PERFECTLY ROUND; the counterfactual
renders a visibly OVAL (elliptical) plate. The decision -- is the visible
outline a perfect circle? -- must be read from the pixels, not from the
round-plate prior. The pixel oracle measures the whole-object ellipticity
(ratio of the largest to the smallest radial extent around the plate centroid)
from the raster alone.

Latent alias mode is ``not_applicable``: the analytic gold reads only
``kind``, which the renderer also reads (to pick the ellipse semi-axes), and
every other latent field (position, size, aspect, rotation, color, canvas) is
directly reflected in the pixels, so the rendering is injective in the latent
scene and there is no render-invariant alias. The canonical-prior name "plate"
is a matched-control/counterfactual manipulation of the *question*, not of any
hidden latent field.
"""

from __future__ import annotations

import random

import numpy as np
from PIL import Image

CANVAS = 512

BACKGROUND = (255, 255, 255)
PLATE_COLORS = [
    (205, 62, 62),    # brick red
    (70, 122, 205),   # slate blue
    (212, 158, 42),   # amber
]

# Decision reference: a plate is reported "oval" when its aspect ratio
# (major/minor semi-axis) clearly departs from unity. The analytic boundary is
# asp = 1.16; circles (asp = 1.0) and ovals (asp in [1.32, 1.60]) are kept far
# from it so the binary is unambiguous at 512px resolution.
BOUNDARY_ASPECT = 1.16
OVAL_ASPECT_MIN = 1.32
OVAL_ASPECT_MAX = 1.60
QUAR_MARGIN = 0.08


def sample_scene(seed: int) -> dict:
    """Deterministic latent scene from an integer seed."""
    rng = random.Random(seed)
    kind = "oval" if rng.random() < 0.5 else "circle"
    aspect = 1.0
    if kind == "oval":
        aspect = rng.uniform(OVAL_ASPECT_MIN, OVAL_ASPECT_MAX)
    # Center, size, color, and rotation are sampled independently of kind so
    # that none of them predicts the class (cheap-cue counterbalance).
    cx = float(rng.uniform(172.0, 340.0))
    cy = float(rng.uniform(172.0, 340.0))
    room = min(cx, cy, CANVAS - cx, CANVAS - cy) - 12.0
    base_r = rng.uniform(110.0, 150.0)
    r = min(base_r, room / aspect, room)
    angle = rng.uniform(0.0, 180.0) if kind == "oval" else 0.0
    color = PLATE_COLORS[rng.randrange(len(PLATE_COLORS))]
    return {
        "kind": kind,
        "cx": round(cx, 3),
        "cy": round(cy, 3),
        "r": round(r, 3),
        "aspect": round(aspect, 3),
        "angle": round(angle, 4),
        "color": color,
        "canvas": CANVAS,
    }


def render(scene: dict) -> Image.Image:
    """Render the single plate (circle or rotated oval) to an RGB image."""
    canvas = int(scene["canvas"])
    cx = float(scene["cx"])
    cy = float(scene["cy"])
    r = float(scene["r"])
    aspect = float(scene["aspect"])
    angle = float(scene["angle"])
    kind = scene["kind"]
    color = tuple(int(v) for v in scene["color"])

    if kind == "oval":
        rx = r * aspect
        ry = r
    else:
        rx = r
        ry = r

    xs = np.arange(canvas, dtype=np.float64)
    ys = np.arange(canvas, dtype=np.float64)
    xx, yy = np.meshgrid(xs, ys)
    dx = xx - cx
    dy = yy - cy
    rad = np.radians(angle)
    cos_t = np.cos(-rad)
    sin_t = np.sin(-rad)
    # Local ellipse coordinates (rotation-invariant membership test).
    ex = dx * cos_t - dy * sin_t
    ey = dx * sin_t + dy * cos_t
    mask = (ex / rx) ** 2 + (ey / ry) ** 2 <= 1.0

    arr = np.zeros((canvas, canvas, 3), dtype=np.uint8)
    arr[:] = BACKGROUND
    arr[mask] = color
    return Image.fromarray(arr, mode="RGB")


def analytic_gold(scene: dict) -> str:
    """Decision = is the visible outline a perfect circle? reads ``kind``."""
    return "yes" if scene["kind"] == "circle" else "no"


def _aspect_of(scene: dict) -> float:
    if scene["kind"] == "circle":
        return 1.0
    return float(scene["aspect"])


def margin(scene: dict) -> float:
    """Verifiability margin = distance of the aspect ratio from the boundary."""
    return round(abs(_aspect_of(scene) - BOUNDARY_ASPECT), 6)


def is_quarantined(scene: dict) -> bool:
    """Quarantine nearly-round ovals that are ambiguous at pixel scale."""
    if scene["kind"] not in ("circle", "oval"):
        return True
    return margin(scene) < QUAR_MARGIN


def latent_symmetries(scene: dict):
    """not_applicable: empty, the latent rendering is injective."""
    return []
