"""Renderer for pivot_moment_duel.

One pale instrument sheet carries a grey horizontal beam, a single black
triangular pointer above the beam that marks the pivot, and two stacks of
identical small solid squares hanging below the beam, one navy and one crimson,
one on each side of the pivot and both pressed against the far edges of the
sheet.

The decisive quantity of a stack is the product of two different measurables:
its cardinality (how many squares) and a metric distance (how far its column
sits from the pivot). The answer is the colour whose product is larger. Neither
factor alone decides it: every emitted scene is sampled so that the winning
stack either has fewer squares than the loser or sits closer to the pivot than
the loser, never both more squares and farther away.

Everything is drawn at final resolution with axis-aligned rectangles and one
polygon, so every ink colour is exact and the pixel arm needs no tolerance.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "pivot_moment_duel-renderer-0.1.0"

WIDTH = 720
HEIGHT = 440

SHEET = (244, 242, 238)
BLACK = (16, 16, 16)
GREY = (150, 150, 150)
NAVY = (28, 58, 140)
CRIMSON = (186, 36, 48)
ORANGE = (232, 150, 44)
OLIVE = (138, 150, 64)

COLOR_RGB = {"navy": NAVY, "crimson": CRIMSON}

BEAM_TOP = 150
BEAM_HEIGHT = 9
BEAM_X0 = 24
BEAM_X1 = 695

STACK_TOP = BEAM_TOP + BEAM_HEIGHT + 3  # squares hang from just under the beam

PIVOT_HALF_WIDTH = 22
PIVOT_HEIGHT = 39

# Legibility / counterbalance floors. MARGIN is the horizontal shift, in pixels,
# that the winning stack would need towards the pivot to balance the two
# products; it is the visible slack of the decision.
MARGIN_FLOOR = 12.0
QUARANTINE_MARGIN = 20.0
MIN_DISTANCE_GAP = 30.0
MIN_ARM = 110.0


def _stack_geometry(x0: int, side: int, gap: int, count: int) -> list[tuple[int, int, int, int]]:
    """Inclusive pixel boxes of the squares of one hanging stack, top first."""
    boxes = []
    for index in range(count):
        top = STACK_TOP + index * (side + gap)
        boxes.append((x0, top, x0 + side - 1, top + side - 1))
    return boxes


def _stack_center(stack: dict) -> float:
    """Horizontal middle of a stack's column, in the same convention as pixels."""
    return stack["x0"] + (stack["side"] - 1) / 2.0


def _products(scene: dict) -> dict[str, float]:
    """Count times distance-to-the-pointer, per stack colour.

    The column middle is inlined here rather than delegated so that the gold
    path reads exactly the two declared label inputs of the scene.
    """
    pivot_x = float(scene["pivot"]["x"])
    products = {}
    for stack in scene["stacks"]:
        center = stack["x0"] + (stack["side"] - 1) / 2.0
        products[stack["color"]] = stack["count"] * abs(center - pivot_x)
    return products


def analytic_gold(scene: dict) -> str:
    """Colour of the stack whose count-times-distance product is larger."""
    products = _products(scene)
    return "navy" if products["navy"] > products["crimson"] else "crimson"


def margin(scene: dict) -> float:
    """Pixels the winning stack must slide towards the pivot to balance.

    Sliding the winner by delta reduces its product by count*delta, so the
    balance point is reached at (P_win - P_lose) / count_win. The quantity is a
    visible horizontal distance on the beam.
    """
    products = _products(scene)
    winner = analytic_gold(scene)
    loser = "crimson" if winner == "navy" else "navy"
    count = next(s["count"] for s in scene["stacks"] if s["color"] == winner)
    return round((products[winner] - products[loser]) / count, 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def cue_type(scene: dict) -> str:
    """Which cheap cue points at the wrong stack in this scene."""
    winner = analytic_gold(scene)
    pivot_x = float(scene["pivot"]["x"])
    win = next(s for s in scene["stacks"] if s["color"] == winner)
    lose = next(s for s in scene["stacks"] if s["color"] != winner)
    if win["count"] > lose["count"]:
        return "winner_has_more_squares_but_is_nearer"
    _ = pivot_x
    return "winner_has_fewer_squares_but_is_farther"


def _arms(scene: dict) -> tuple[float, float]:
    pivot_x = float(scene["pivot"]["x"])
    left = next(s for s in scene["stacks"] if _stack_center(s) < pivot_x)
    right = next(s for s in scene["stacks"] if _stack_center(s) > pivot_x)
    return pivot_x - _stack_center(left), _stack_center(right) - pivot_x


# Deterministic margin bands, so every dataset carries quarantined,
# near-boundary and comfortable scenes instead of only easy ones.
MARGIN_BANDS: dict[str, tuple[float, float]] = {
    "quarantined": (MARGIN_FLOOR, QUARANTINE_MARGIN),
    "near_boundary": (QUARANTINE_MARGIN, 34.0),
    "clear": (34.0, 1.0e9),
}
_BAND_ORDER = ("quarantined", "near_boundary", "clear", "clear")


def sample_scene(seed: int) -> dict:
    """Deterministically sample one legible, cue-conflicting scene.

    The seed also fixes which margin band the scene must land in.
    """
    rng = random.Random(seed * 7919 + 13)
    band = _BAND_ORDER[seed % len(_BAND_ORDER)]
    low, high = MARGIN_BANDS[band]
    for _ in range(20000):
        side = rng.choice((14, 16, 18))
        gap = rng.choice((4, 5, 6))
        left_x0 = rng.randint(14, 44)
        right_x0 = rng.randint(676 - side, 706 - side)
        pivot_x = rng.randint(200, 520)
        counts = rng.sample([1, 2, 3, 4, 5, 6], 2)
        navy_left = rng.random() < 0.5

        left_color = "navy" if navy_left else "crimson"
        right_color = "crimson" if navy_left else "navy"
        stacks = [
            {"color": left_color, "x0": left_x0, "side": side, "gap": gap, "count": counts[0]},
            {"color": right_color, "x0": right_x0, "side": side, "gap": gap, "count": counts[1]},
        ]
        scene = {
            "width": WIDTH,
            "height": HEIGHT,
            "beam": {"x0": BEAM_X0, "x1": BEAM_X1, "top": BEAM_TOP, "height": BEAM_HEIGHT},
            "pivot": {
                "x": pivot_x,
                "half_width": PIVOT_HALF_WIDTH,
                "height": PIVOT_HEIGHT,
                "apex_y": BEAM_TOP - 1,
            },
            "stacks": stacks,
            "decoys": [],
            "ticks": [],
        }

        left_arm, right_arm = _arms(scene)
        if min(left_arm, right_arm) < MIN_ARM:
            continue
        if abs(left_arm - right_arm) < MIN_DISTANCE_GAP:
            continue

        winner = analytic_gold(scene)
        win = next(s for s in stacks if s["color"] == winner)
        lose = next(s for s in stacks if s["color"] != winner)
        win_arm = left_arm if _stack_center(win) < pivot_x else right_arm
        lose_arm = right_arm if _stack_center(win) < pivot_x else left_arm
        # Counterbalance: never let the winner be both the bigger count and the
        # farther column, so neither cue alone predicts the answer.
        if win["count"] > lose["count"] and win_arm > lose_arm:
            continue
        if not low <= margin(scene) < high:
            continue

        scene["decoys"] = _sample_decoys(rng, scene)
        scene["ticks"] = _sample_ticks(rng)
        return scene
    raise RuntimeError(f"could not sample a legible scene for seed {seed}")


def _sample_decoys(rng: random.Random, scene: dict) -> list[dict]:
    """Decoration that the question declares irrelevant.

    Everything sits in the horizontal band 110..640 and never in the two edge
    columns that hold the coloured stacks, and never on the black pointer, so no
    decoration can occlude a decisive mark.
    """
    decoys: list[dict] = []
    pivot = scene["pivot"]
    if rng.random() < 0.55:
        side = rng.choice((12, 14))
        count = rng.randint(2, 5)
        x0 = rng.randint(210, 540)
        if abs(x0 - pivot["x"]) > 40:
            decoys.append({"kind": "olive_stack", "x0": x0, "side": side, "count": count})
    for _ in range(rng.randint(2, 4)):
        radius = rng.randint(7, 13)
        cx = rng.randint(130, 620)
        cy = rng.choice((rng.randint(34, 96), rng.randint(330, 400)))
        if abs(cx - pivot["x"]) < pivot["half_width"] + radius + 8 and cy < 120:
            continue
        decoys.append({"kind": "ring", "cx": cx, "cy": cy, "radius": radius})
    for _ in range(rng.randint(1, 3)):
        decoys.append(
            {
                "kind": "bar",
                "x0": rng.randint(140, 560),
                "y0": rng.choice((rng.randint(40, 92), rng.randint(340, 396))),
                "length": rng.randint(24, 70),
                "thickness": rng.randint(5, 9),
            }
        )
    return decoys


def _sample_ticks(rng: random.Random) -> list[dict]:
    start = rng.randint(40, 70)
    pitch = rng.randint(26, 40)
    ticks = []
    x = start
    while x < WIDTH - 40:
        ticks.append({"x": x, "height": rng.randint(10, 17)})
        x += pitch
    return ticks


def render(scene: dict) -> Image.Image:
    """Draw the sheet at final resolution: exact colours, no resampling."""
    image = Image.new("RGB", (scene["width"], scene["height"]), SHEET)
    draw = ImageDraw.Draw(image)

    for tick in scene["ticks"]:
        top = scene["height"] - 8 - tick["height"]
        draw.rectangle((tick["x"], top, tick["x"] + 2, scene["height"] - 9), fill=GREY)

    beam = scene["beam"]
    draw.rectangle(
        (beam["x0"], beam["top"], beam["x1"], beam["top"] + beam["height"] - 1), fill=GREY
    )

    for decoy in scene["decoys"]:
        if decoy["kind"] == "olive_stack":
            for box in _stack_geometry(decoy["x0"], decoy["side"], 5, decoy["count"]):
                draw.rectangle(box, fill=OLIVE)
        elif decoy["kind"] == "ring":
            radius = decoy["radius"]
            draw.ellipse(
                (
                    decoy["cx"] - radius,
                    decoy["cy"] - radius,
                    decoy["cx"] + radius,
                    decoy["cy"] + radius,
                ),
                outline=ORANGE,
                width=3,
            )
        else:
            draw.rectangle(
                (
                    decoy["x0"],
                    decoy["y0"],
                    decoy["x0"] + decoy["length"],
                    decoy["y0"] + decoy["thickness"],
                ),
                fill=ORANGE,
            )

    pivot = scene["pivot"]
    apex_y = pivot["apex_y"]
    base_y = apex_y - pivot["height"] + 1
    draw.polygon(
        [
            (pivot["x"], apex_y),
            (pivot["x"] - pivot["half_width"], base_y),
            (pivot["x"] + pivot["half_width"], base_y),
        ],
        fill=BLACK,
    )

    for stack in scene["stacks"]:
        rgb = COLOR_RGB[stack["color"]]
        for box in _stack_geometry(stack["x0"], stack["side"], stack["gap"], stack["count"]):
            draw.rectangle(box, fill=rgb)

    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent alias: every sampled field is drawn (see candidate.json)."""
    _ = scene
    return []
