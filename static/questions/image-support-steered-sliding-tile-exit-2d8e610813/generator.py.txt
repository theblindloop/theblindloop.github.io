"""Deterministic renderer and analytic arm for Sliding Tile Exit."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

QUESTION = (
    "The blue square is a sliding tile. Without rotating it, keep the whole tile "
    "on the dark floor and trace from its blue start. Which exit can it reach, "
    "red or gold? Answer red or gold."
)

BACKGROUND = "#f6f2ea"
FLOOR = "#193f4d"
BLUE = "#2867d6"
RED = "#df3f4c"
GOLD = "#e1a51b"

PIECE_NAMES = (
    "trunk",
    "spine",
    "top_left",
    "top_gate",
    "top_right",
    "bottom_left",
    "bottom_gate",
    "bottom_right",
)


def sample_scene(seed: int) -> dict[str, Any]:
    """Return a balanced randomized scene for one integer seed."""

    rng = random.Random(int(seed))
    # Consecutive seeds cycle through all branch/color bindings, giving exact
    # answer balance without coupling the answer to either position.
    wide_branch = "top" if int(seed) % 2 == 0 else "bottom"
    red_branch = "top" if (int(seed) // 2) % 2 == 0 else "bottom"
    order = list(PIECE_NAMES)
    rng.shuffle(order)
    return {
        "width": 384,
        "height": 384,
        "background": BACKGROUND,
        "floor_color": FLOOR,
        "tile_size": 42,
        "wide_branch": wide_branch,
        "red_branch": red_branch,
        "offset_x": rng.randint(-5, 5),
        "offset_y": rng.randint(-5, 5),
        "wide_clearance": rng.randint(10, 16),
        "narrow_deficit": rng.randint(9, 15),
        "piece_order": order,
    }


def _geometry(scene: dict[str, Any]) -> tuple[dict[str, tuple[int, int, int, int]], dict[str, tuple[int, int]]]:
    ox = int(scene["offset_x"])
    oy = int(scene["offset_y"])
    tile = int(scene["tile_size"])
    wide = tile + int(scene["wide_clearance"])
    narrow = tile - int(scene["narrow_deficit"])
    top_width = wide if scene["wide_branch"] == "top" else narrow
    bottom_width = wide if scene["wide_branch"] == "bottom" else narrow

    def rect(x0: int, y0: int, x1: int, y1: int) -> tuple[int, int, int, int]:
        return (x0 + ox, y0 + oy, x1 + ox, y1 + oy)

    def gate(y: int, thickness: int) -> tuple[int, int, int, int]:
        half_lo = thickness // 2
        return rect(248, y - half_lo, 306, y + (thickness - half_lo))

    pieces = {
        "trunk": rect(20, 130, 160, 254),
        "spine": rect(118, 38, 208, 346),
        "top_left": rect(166, 38, 250, 140),
        "top_gate": gate(89, top_width),
        "top_right": rect(304, 38, 364, 140),
        "bottom_left": rect(166, 244, 250, 346),
        "bottom_gate": gate(295, bottom_width),
        "bottom_right": rect(304, 244, 364, 346),
    }
    centers = {
        "start": (76 + ox, 192 + oy),
        "top": (338 + ox, 89 + oy),
        "bottom": (338 + ox, 295 + oy),
    }
    return pieces, centers


def _square(draw: ImageDraw.ImageDraw, center: tuple[int, int], size: int, color: str) -> None:
    cx, cy = center
    left = cx - size // 2
    top = cy - size // 2
    draw.rectangle((left, top, left + size - 1, top + size - 1), fill=color)


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only visible scene fields into a crisp RGB raster."""

    image = Image.new(
        "RGB",
        (int(scene["width"]), int(scene["height"])),
        str(scene["background"]),
    )
    draw = ImageDraw.Draw(image)
    pieces, centers = _geometry(scene)
    for name in scene["piece_order"]:
        # Pillow rectangles include the final coordinate. Normalizing to x1-1
        # makes each latent rectangle half-open and its width unambiguous.
        x0, y0, x1, y1 = pieces[str(name)]
        draw.rectangle((x0, y0, x1 - 1, y1 - 1), fill=str(scene["floor_color"]))

    _square(draw, centers["start"], int(scene["tile_size"]), BLUE)
    red_center = centers[str(scene["red_branch"])]
    gold_branch = "bottom" if scene["red_branch"] == "top" else "top"
    _square(draw, red_center, 24, RED)
    _square(draw, centers[gold_branch], 24, GOLD)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Select the colored exit on the clearance-passable branch."""

    tile_size = int(scene["tile_size"])
    wide_branch = str(scene["wide_branch"])
    red_branch = str(scene["red_branch"])
    # The visible wide doorway is constructed above tile_size while the other
    # is below it. Keeping this explicit makes tile extent a true label input.
    if tile_size <= 0 or wide_branch not in {"top", "bottom"}:
        raise ValueError("invalid clearance scene")
    return "red" if red_branch == wide_branch else "gold"


def margin(scene: dict[str, Any]) -> float:
    """Runner-up margin: closest doorway-to-tile boundary in rendered pixels."""

    return float(min(int(scene["wide_clearance"]), int(scene["narrow_deficit"])))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) <= 6.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """Opaque same-color floor pieces commute under paint-order reversal."""

    twin = dict(scene)
    twin["piece_order"] = list(reversed(list(scene["piece_order"])))
    return [("reverse_same_color_piece_order", twin)]
