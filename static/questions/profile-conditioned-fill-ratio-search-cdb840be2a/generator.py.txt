"""Reference fill-ratio visual search world.

A dark filled gauge is shown in a top reference panel. Four quadrants below
each contain one gauge whose dark fill rises from the bottom to some fraction
of the track height. Exactly one candidate gauge has the same fill fraction as
the reference; the decision is the quadrant containing that gauge. All geometry
and the decision are visible in the raster; rendering is deterministic.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw

SIZE = 480
BG = (255, 255, 255)
TRACK = (232, 232, 232)
FILL = (25, 25, 25)

GW = 34
GH = 90
FILL_PAD = 5

REF_BOX = (223, 43, 257, 133)  # reference gauge track (x0, y0, x1, y1)

SECTORS = ("TL", "TR", "BL", "BR")
SECTOR_BOXES = {
    "TL": (103, 255, 137, 345),
    "TR": (343, 255, 377, 345),
    "BL": (103, 370, 137, 460),
    "BR": (343, 370, 377, 460),
}

FRAC_LO = 0.15
FRAC_HI = 0.85
MIN_DIFF = 0.06  # minimum |fraction - reference| for every distractor
QUARANTINE_THRESH = 0.065


def _fraction_to_height(frac: float) -> int:
    return int(round(frac * GH))


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(seed)
    ref_frac = float(rng.uniform(FRAC_LO, FRAC_HI))

    winner_sector = SECTORS[int(rng.integers(0, len(SECTORS)))]
    fractions: dict[str, float] = {winner_sector: ref_frac}

    for sector in SECTORS:
        if sector in fractions:
            continue
        for _ in range(1000):
            candidate = float(rng.uniform(FRAC_LO, FRAC_HI))
            if abs(candidate - ref_frac) < MIN_DIFF:
                continue
            fractions[sector] = candidate
            break
        else:  # pragma: no cover - extremely unlikely path
            fractions[sector] = float((ref_frac + 0.25) if ref_frac < 0.5 else (ref_frac - 0.25))

    reference = {"fraction": ref_frac, "box": list(REF_BOX)}
    candidates = [
        {
            "fraction": float(fractions[sector]),
            "sector": sector,
            "box": list(SECTOR_BOXES[sector]),
        }
        for sector in SECTORS
    ]
    return {"reference": reference, "candidates": candidates}


def _draw_gauge(draw: ImageDraw.ImageDraw, box: tuple, fraction: float) -> None:
    x0, y0, x1, y1 = box
    draw.rectangle([x0, y0, x1, y1], fill=TRACK)
    fill_h = _fraction_to_height(fraction)
    if fill_h > 0:
        draw.rectangle(
            [x0 + FILL_PAD, y1 - fill_h, x1 - FILL_PAD, y1],
            fill=FILL,
        )


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (SIZE, SIZE), BG)
    draw = ImageDraw.Draw(img)

    ref = scene["reference"]
    _draw_gauge(draw, tuple(ref["box"]), ref["fraction"])

    for cand in scene["candidates"]:
        _draw_gauge(draw, tuple(cand["box"]), cand["fraction"])

    return img


def _decision(scene: dict) -> tuple[str, float]:
    ref_frac = scene["reference"]["fraction"]
    best_sector = None
    best_diff = float("inf")
    for cand in scene["candidates"]:
        diff = abs(cand["fraction"] - ref_frac)
        if diff < best_diff:
            best_diff = diff
            best_sector = cand["sector"]
    runner_up = min(
        abs(c["fraction"] - ref_frac)
        for c in scene["candidates"]
        if c["sector"] != best_sector
    )
    return best_sector, float(runner_up)


def analytic_gold(scene: dict) -> str:
    sector, _ = _decision(scene)
    return sector


def margin(scene: dict) -> float:
    _, m = _decision(scene)
    return m


def is_quarantined(scene: dict) -> bool:
    _, m = _decision(scene)
    return bool(m < QUARANTINE_THRESH)


def latent_symmetries(scene: dict) -> list:
    return []
