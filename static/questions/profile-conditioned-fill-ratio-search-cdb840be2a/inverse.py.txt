"""Independent pixel-only inverse arm for fill_ratio_search.

Receives only a PIL image and the public question contract. It isolates each
gauge's dark fill from its visible light track, measures the fill fraction of
the top reference gauge and of each quadrant gauge, and selects the quadrant
whose measured fill fraction is closest to the reference's.

Imports nothing from the renderer, prompts, generator, or verifier.
"""

from __future__ import annotations

import numpy as np

# Public visual layout (disclosed by the question world).
GH = 90.0  # track height in pixels
REF_BOX = (223, 43, 257, 133)

SECTOR_BOXES = {
    "TL": (103, 255, 137, 345),
    "TR": (343, 255, 377, 345),
    "BL": (103, 370, 137, 460),
    "BR": (343, 370, 377, 460),
}


def _fill_fraction(arr, box) -> float:
    x0, y0, x1, y1 = box
    dark = arr[:, :, 0] < 150
    region = dark[y0:y1, x0:x1]
    idx = np.nonzero(region)
    if idx[0].size == 0:
        return 0.0
    top = int(idx[0].min())
    bottom = int(idx[0].max())
    fill_h = bottom - top + 1
    return fill_h / GH


def decision_from_image(image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    ref_frac = _fill_fraction(arr, REF_BOX)
    best_sector = None
    best_diff = float("inf")
    for sector, box in SECTOR_BOXES.items():
        frac = _fill_fraction(arr, box)
        diff = abs(frac - ref_frac)
        if diff < best_diff:
            best_diff = diff
            best_sector = sector
    return best_sector if best_sector is not None else "TL"
