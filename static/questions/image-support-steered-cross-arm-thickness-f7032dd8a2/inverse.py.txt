"""Independent image-only inverse for the connected cross sample."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _largest_component(mask: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    best: list[tuple[int, int]] = []
    for y_value, x_value in np.argwhere(mask):
        y = int(y_value)
        x = int(x_value)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        component: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            component.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(component) > len(best):
            best = component
    output = np.zeros_like(mask, dtype=bool)
    for y, x in best:
        output[y, x] = True
    return output


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    border = np.concatenate(
        (array[:4].reshape(-1, 3), array[-4:].reshape(-1, 3),
         array[:, :4].reshape(-1, 3), array[:, -4:].reshape(-1, 3)),
        axis=0,
    )
    background = np.median(border, axis=0)
    distance = np.linalg.norm(array - background, axis=2)
    component = _largest_component(distance > 55.0)
    points = np.argwhere(component)
    if len(points) < 650:
        raise ValueError("no intact colored measurement sample")
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    height = int(y1 - y0 + 1)
    width = int(x1 - x0 + 1)
    if not (55 <= width <= 125 and 55 <= height <= 125):
        raise ValueError("colored component does not have the expected compact cross extent")
    crop = component[y0 : y1 + 1, x0 : x1 + 1]
    row_counts = crop.sum(axis=1)
    column_counts = crop.sum(axis=0)
    horizontal_thickness = int(np.count_nonzero(row_counts >= 0.80 * row_counts.max()))
    vertical_thickness = int(np.count_nonzero(column_counts >= 0.80 * column_counts.max()))
    if min(horizontal_thickness, vertical_thickness) < 12:
        raise ValueError("cross arms are incomplete")
    if abs(horizontal_thickness - vertical_thickness) < 4:
        raise ValueError("arm-thickness decision is raster-ambiguous")
    return "horizontal" if horizontal_thickness > vertical_thickness else "vertical"
