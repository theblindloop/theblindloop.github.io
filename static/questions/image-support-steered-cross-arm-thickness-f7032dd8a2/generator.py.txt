"""Deterministic renderer and analytic arm-thickness measurement."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw


PALETTE = (
    (17, 139, 126),
    (36, 105, 176),
    (196, 79, 96),
    (122, 83, 173),
    (205, 112, 29),
)
BACKGROUNDS = ((249, 248, 244), (247, 249, 250), (250, 247, 248))
SCALE = 3


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    width = 384
    height = 384
    base = rng.randint(18, 26)
    gap = rng.randint(8, 14)
    horizontal_is_thicker = int(seed) % 2 == 0
    horizontal_thickness = base + gap if horizontal_is_thicker else base
    vertical_thickness = base if horizontal_is_thicker else base + gap
    # The salient longitudinal spans are independent of the answer and each other.
    horizontal_length = rng.randint(76, 104)
    vertical_length = rng.randint(76, 104)
    half_extent = 56
    return {
        "width": width,
        "height": height,
        "center_x": rng.randint(half_extent + 24, width - half_extent - 24),
        "center_y": rng.randint(half_extent + 24, height - half_extent - 24),
        "horizontal_length": horizontal_length,
        "vertical_length": vertical_length,
        "horizontal_thickness": horizontal_thickness,
        "vertical_thickness": vertical_thickness,
        "foreground": list(PALETTE[rng.randrange(len(PALETTE))]),
        "background": list(BACKGROUNDS[rng.randrange(len(BACKGROUNDS))]),
    }


def _scaled_box(cx: int, cy: int, span_x: int, span_y: int) -> tuple[int, int, int, int]:
    left = round((cx - span_x / 2) * SCALE)
    top = round((cy - span_y / 2) * SCALE)
    right = round((cx + span_x / 2) * SCALE)
    bottom = round((cy + span_y / 2) * SCALE)
    return left, top, right, bottom


def render(scene: dict[str, Any]) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    center_x = int(scene["center_x"])
    center_y = int(scene["center_y"])
    horizontal_length = int(scene["horizontal_length"])
    vertical_length = int(scene["vertical_length"])
    horizontal_thickness = int(scene["horizontal_thickness"])
    vertical_thickness = int(scene["vertical_thickness"])
    foreground = tuple(int(value) for value in scene["foreground"])
    background = tuple(int(value) for value in scene["background"])

    image = Image.new("RGB", (width * SCALE, height * SCALE), background)
    draw = ImageDraw.Draw(image)
    draw.rectangle(
        _scaled_box(center_x, center_y, horizontal_length, horizontal_thickness),
        fill=foreground,
    )
    draw.rectangle(
        _scaled_box(center_x, center_y, vertical_thickness, vertical_length),
        fill=foreground,
    )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    return (
        "horizontal"
        if int(scene["horizontal_thickness"]) > int(scene["vertical_thickness"])
        else "vertical"
    )


def margin(scene: dict[str, Any]) -> float:
    return float(
        abs(int(scene["horizontal_thickness"]) - int(scene["vertical_thickness"]))
    )


def is_quarantined(scene: dict[str, Any]) -> bool:
    return abs(
        int(scene["horizontal_thickness"]) - int(scene["vertical_thickness"])
    ) < 6


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
