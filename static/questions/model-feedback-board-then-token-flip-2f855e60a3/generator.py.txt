"""Deterministic renderer and analytic program for nested geometric transforms."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont


WIDTH = 1100
HEIGHT = 760
N = 4
TOKEN_COLORS = {
    "red": (220, 55, 68),
    "blue": (44, 112, 210),
    "green": (38, 155, 104),
    "purple": (135, 73, 190),
}
MODES = (
    "correct",
    "local_before_board",
    "positions_only_then_local",
    "whole_mirror_after_board",
)
OPTION_LABELS = ("A", "B", "C", "D")


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(name, size)


def _flip_dir(direction: int) -> int:
    return {0: 0, 1: 3, 2: 2, 3: 1}[int(direction)]


def _rotate_token(token: dict) -> dict:
    return {
        "color": token["color"],
        "row": int(token["col"]),
        "col": N - 1 - int(token["row"]),
        "dir": (int(token["dir"]) + 1) % 4,
    }


def _local_flip(token: dict) -> dict:
    return {**token, "dir": _flip_dir(int(token["dir"]))}


def _board_for_mode(source_tokens: Iterable[dict], mode: str) -> list[dict]:
    source = [dict(token) for token in source_tokens]
    if mode == "correct":
        return [_local_flip(_rotate_token(token)) for token in source]
    if mode == "local_before_board":
        return [_rotate_token(_local_flip(token)) for token in source]
    if mode == "positions_only_then_local":
        return [
            {
                "color": token["color"],
                "row": int(token["col"]),
                "col": N - 1 - int(token["row"]),
                "dir": _flip_dir(int(token["dir"])),
            }
            for token in source
        ]
    if mode == "whole_mirror_after_board":
        rotated = [_rotate_token(token) for token in source]
        return [
            {
                "color": token["color"],
                "row": int(token["row"]),
                "col": N - 1 - int(token["col"]),
                "dir": _flip_dir(int(token["dir"])),
            }
            for token in rotated
        ]
    raise ValueError(f"unknown option mode {mode!r}")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    cells = rng.sample(range(N * N), len(TOKEN_COLORS))
    source_tokens = []
    for color, cell in zip(TOKEN_COLORS, cells):
        source_tokens.append(
            {
                "color": color,
                "row": cell // N,
                "col": cell % N,
                "dir": rng.randrange(4),
            }
        )
    shift = int(seed) % len(MODES)
    option_modes = list(MODES[shift:] + MODES[:shift])
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "source_tokens": source_tokens,
        "option_modes": option_modes,
    }


def _arrow_points(cx: float, cy: float, radius: float, direction: int) -> list[tuple[int, int]]:
    base = [
        (-0.44, -0.16),
        (0.06, -0.16),
        (0.06, -0.36),
        (0.46, 0.0),
        (0.06, 0.36),
        (0.06, 0.16),
        (-0.44, 0.16),
    ]
    points = []
    for x, y in base:
        # Logical directions are 0=up, 1=right, 2=down, 3=left.
        # The base polygon points right and image y increases downward.
        for _ in range((int(direction) + 3) % 4):
            x, y = -y, x
        points.append((round(cx + x * radius * 2), round(cy + y * radius * 2)))
    return points


def _draw_board(draw: ImageDraw.ImageDraw, x: int, y: int, size: int, tokens: Iterable[dict]) -> None:
    cell = size / N
    draw.rounded_rectangle(
        (x - 5, y - 5, x + size + 5, y + size + 5),
        radius=10,
        fill=(250, 252, 255),
        outline=(58, 67, 82),
        width=4,
    )
    for index in range(N + 1):
        px = round(x + index * cell)
        py = round(y + index * cell)
        draw.line((px, y, px, y + size), fill=(155, 165, 179), width=2)
        draw.line((x, py, x + size, py), fill=(155, 165, 179), width=2)
    for token in tokens:
        cx = x + (int(token["col"]) + 0.5) * cell
        cy = y + (int(token["row"]) + 0.5) * cell
        points = _arrow_points(cx, cy, cell * 0.39, int(token["dir"]))
        draw.polygon(points, fill=TOKEN_COLORS[str(token["color"])])
        draw.line(points + [points[0]], fill=(31, 36, 45), width=max(1, round(cell * 0.035)), joint="curve")


def _draw_turn_icon(draw: ImageDraw.ImageDraw, cx: int, cy: int) -> None:
    draw.arc((cx - 45, cy - 45, cx + 45, cy + 45), 205, 535, fill=(39, 88, 154), width=10)
    draw.polygon([(cx + 48, cy - 5), (cx + 27, cy - 13), (cx + 35, cy + 11)], fill=(39, 88, 154))
    draw.rectangle((cx - 14, cy - 14, cx + 14, cy + 14), outline=(92, 102, 118), width=3)


def _draw_flip_icon(draw: ImageDraw.ImageDraw, cx: int, cy: int) -> None:
    draw.line((cx, cy - 48, cx, cy + 48), fill=(112, 78, 151), width=4)
    draw.polygon(_arrow_points(cx - 35, cy, 25, 1), fill=(112, 78, 151))
    draw.polygon(_arrow_points(cx + 35, cy, 25, 3), fill=(112, 78, 151))


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    source_tokens = [dict(token) for token in scene["source_tokens"]]
    option_modes = list(scene["option_modes"])
    image = Image.new("RGB", (width, height), (245, 247, 251))
    draw = ImageDraw.Draw(image)

    draw.text((width // 2, 24), "TWO FRAMES, TWO MOVES", font=_font(27, True), fill=(31, 37, 48), anchor="ma")
    draw.text((50, 82), "START", font=_font(24, True), fill=(31, 37, 48))
    _draw_board(draw, 50, 120, 300, source_tokens)

    draw.rounded_rectangle((415, 132, 690, 356), radius=18, fill=(235, 244, 255), outline=(76, 116, 170), width=4)
    draw.text((552, 151), "STEP 1", font=_font(22, True), fill=(35, 73, 126), anchor="ma")
    _draw_turn_icon(draw, 552, 226)
    draw.text((552, 288), "WHOLE BOARD", font=_font(20, True), fill=(31, 37, 48), anchor="ma")
    draw.text((552, 319), "turn 90 clockwise", font=_font(18), fill=(60, 67, 78), anchor="ma")
    draw.line((706, 244, 750, 244), fill=(85, 94, 108), width=5)
    draw.polygon([(750, 244), (734, 234), (734, 254)], fill=(85, 94, 108))

    draw.rounded_rectangle((770, 132, 1045, 374), radius=18, fill=(247, 239, 255), outline=(126, 87, 160), width=4)
    draw.text((907, 151), "STEP 2", font=_font(22, True), fill=(102, 60, 137), anchor="ma")
    _draw_flip_icon(draw, 907, 226)
    draw.text((907, 282), "EACH ARROW", font=_font(20, True), fill=(31, 37, 48), anchor="ma")
    draw.text((907, 312), "mirror left-right", font=_font(18), fill=(60, 67, 78), anchor="ma")
    draw.text((907, 338), "cells stay put", font=_font(16, True), fill=(102, 60, 137), anchor="ma")

    draw.text((50, 447), "RESULT OPTIONS", font=_font(24, True), fill=(31, 37, 48))
    option_x = (40, 310, 580, 850)
    for label, mode, x in zip(OPTION_LABELS, option_modes, option_x):
        draw.text((x + 105, 462), label, font=_font(28, True), fill=(31, 37, 48), anchor="ma")
        _draw_board(draw, x, 495, 210, _board_for_mode(source_tokens, str(mode)))
    return image


def analytic_gold(scene: dict) -> str:
    option_modes = scene["option_modes"]
    matches = [index for index, mode in enumerate(option_modes) if mode == "correct"]
    if len(matches) != 1:
        raise ValueError("scene does not have exactly one correct result panel")
    return OPTION_LABELS[matches[0]]


def margin(scene: dict) -> float:
    boards = [_board_for_mode(scene["source_tokens"], str(mode)) for mode in scene["option_modes"]]
    keys = [set((item["color"], item["row"], item["col"], item["dir"]) for item in board) for board in boards]
    target_index = OPTION_LABELS.index(analytic_gold(scene))
    target = keys[target_index]
    return float(min(len(target.symmetric_difference(other)) for i, other in enumerate(keys) if i != target_index))


def is_quarantined(scene: dict) -> bool:
    boards = [_board_for_mode(scene["source_tokens"], str(mode)) for mode in scene["option_modes"]]
    keys = [tuple(sorted((item["color"], item["row"], item["col"], item["dir"]) for item in board)) for board in boards]
    return len(set(keys)) != len(keys)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
