"""Independent raster-only inverse for the nested-frame transformation task."""

from __future__ import annotations

from PIL import Image


N = 4
PALETTE = {
    "red": (220, 55, 68),
    "blue": (44, 112, 210),
    "green": (38, 155, 104),
    "purple": (135, 73, 190),
}
LABELS = ("A", "B", "C", "D")


def _read_board(image: Image.Image, x: int, y: int, size: int) -> list[dict]:
    pixels = image.load()
    cell = size / N
    tokens = []
    for color, rgb in PALETTE.items():
        points = []
        for py in range(y, y + size + 1):
            for px in range(x, x + size + 1):
                if pixels[px, py] == rgb:
                    points.append((px, py))
        if len(points) < 40:
            raise ValueError(f"missing {color} token")
        xs = [point[0] for point in points]
        ys = [point[1] for point in points]
        min_x, max_x = min(xs), max(xs)
        min_y, max_y = min(ys), max(ys)
        cx = sum(xs) / len(xs)
        cy = sum(ys) / len(ys)
        col = int((cx - x) / cell)
        row = int((cy - y) / cell)
        if not (0 <= row < N and 0 <= col < N):
            raise ValueError("token center outside board")
        if (max_x - min_x) > (max_y - min_y):
            left_edge = sum(px == min_x for px, _ in points)
            right_edge = sum(px == max_x for px, _ in points)
            if left_edge == right_edge:
                raise ValueError("ambiguous horizontal arrow")
            direction = 1 if right_edge < left_edge else 3
        else:
            top_edge = sum(py == min_y for _, py in points)
            bottom_edge = sum(py == max_y for _, py in points)
            if top_edge == bottom_edge:
                raise ValueError("ambiguous vertical arrow")
            direction = 0 if top_edge < bottom_edge else 2
        tokens.append({"color": color, "row": row, "col": col, "dir": direction})
    return tokens


def _flip(direction: int) -> int:
    return {0: 0, 1: 3, 2: 2, 3: 1}[direction]


def _expected(source: list[dict]) -> list[dict]:
    result = []
    for token in source:
        rotated_direction = (token["dir"] + 1) % 4
        result.append(
            {
                "color": token["color"],
                "row": token["col"],
                "col": N - 1 - token["row"],
                "dir": _flip(rotated_direction),
            }
        )
    return result


def _key(tokens: list[dict]) -> tuple[tuple, ...]:
    return tuple(sorted((item["color"], item["row"], item["col"], item["dir"]) for item in tokens))


def decision_from_image(image: Image.Image) -> str:
    image = image.convert("RGB")
    if image.size != (1100, 760):
        raise ValueError(f"unexpected image size {image.size}")
    source = _read_board(image, 50, 120, 300)
    expected = _key(_expected(source))
    boards = [_read_board(image, x, 495, 210) for x in (40, 310, 580, 850)]
    matches = [index for index, board in enumerate(boards) if _key(board) == expected]
    if len(matches) != 1:
        raise ValueError(f"expected one matching result panel, found {len(matches)}")
    return LABELS[matches[0]]
