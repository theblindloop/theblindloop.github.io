"""Renderer and analytic gold for the declared low-pass reveal world."""

from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image, ImageDraw

RENDERER_VERSION = "bandlimited_blur_icon-0.1.0"
CANVAS_SIZE = (640, 440)
TILE_SIZE = 256
MARK_SIZE = 12
MARK_OFFSET = 14
CARRIER_TILE = 16
BLUR_SIZE = 13
TARGET_INK = 2400
SIGNAL_BOUNDARY = 27.0
SYMBOLS = ("ring", "cross", "triangle", "zigzag")
MARKER_COLORS = (
    (24, 190, 224),
    (218, 72, 166),
    (104, 204, 76),
    (244, 170, 42),
)


def _segment_distance(
    x: np.ndarray,
    y: np.ndarray,
    a: tuple[float, float],
    b: tuple[float, float],
) -> np.ndarray:
    ax, ay = a
    bx, by = b
    vx = bx - ax
    vy = by - ay
    denominator = vx * vx + vy * vy
    t = ((x - ax) * vx + (y - ay) * vy) / denominator
    t = np.clip(t, 0.0, 1.0)
    return np.hypot(x - (ax + t * vx), y - (ay + t * vy))


def _glyph_mask(symbol: str, dx: int = 0, dy: int = 0) -> np.ndarray:
    """Create equal-ink line glyphs; translation is a nuisance parameter."""

    yy, xx = np.indices((TILE_SIZE, TILE_SIZE), dtype=np.float64)
    x = xx - (TILE_SIZE / 2.0 + int(dx))
    y = yy - (TILE_SIZE / 2.0 + int(dy))
    radius = np.hypot(x, y)
    if symbol == "ring":
        score = np.abs(radius - 42.0) + 0.001 * radius
        score = np.where(radius <= 58.0, score, score + 100.0)
    elif symbol == "cross":
        score = np.minimum(np.abs(x), np.abs(y)) + 0.018 * np.maximum(np.abs(x), np.abs(y))
        score = np.where(np.maximum(np.abs(x), np.abs(y)) <= 52.0, score, score + 100.0)
    elif symbol == "triangle":
        points = [(0.0, -52.0), (-52.0, 45.0), (52.0, 45.0), (0.0, -52.0)]
        score = np.minimum.reduce(
            [_segment_distance(x, y, a, b) for a, b in zip(points, points[1:])]
        )
        score = np.where((np.abs(x) <= 58.0) & (y >= -60.0) & (y <= 54.0), score, score + 100.0)
        score = score + 0.002 * radius
    elif symbol == "zigzag":
        points = [(-54.0, -38.0), (-18.0, 38.0), (18.0, -38.0), (54.0, 38.0)]
        score = np.minimum.reduce(
            [_segment_distance(x, y, a, b) for a, b in zip(points, points[1:])]
        )
        score = np.where((np.abs(x) <= 62.0) & (np.abs(y) <= 52.0), score, score + 100.0)
        score = score + 0.002 * radius
    else:
        raise ValueError(f"unknown symbol: {symbol}")
    order = np.lexsort((xx.ravel(), yy.ravel(), score.ravel()))
    mask = np.zeros(TILE_SIZE * TILE_SIZE, dtype=bool)
    mask[order[:TARGET_INK]] = True
    return mask.reshape((TILE_SIZE, TILE_SIZE))


def _carrier(texture_seed: int, phase: int, base_level: int, amplitude: int) -> np.ndarray:
    rng = np.random.default_rng(int(texture_seed))
    tile = rng.integers(
        -int(amplitude), int(amplitude) + 1, size=(CARRIER_TILE, CARRIER_TILE), dtype=np.int16
    )
    yy, xx = np.indices((TILE_SIZE, TILE_SIZE))
    phase = int(phase) % CARRIER_TILE
    values = int(base_level) + tile
    return values[(yy + phase) % CARRIER_TILE, (xx + 3 * phase) % CARRIER_TILE]


def _tile_pixels(scene: dict) -> np.ndarray:
    carrier = _carrier(
        scene["texture_seed"],
        scene["texture_phase"],
        scene["base_level"],
        scene["carrier_amplitude"],
    )
    mask = _glyph_mask(scene["symbol"], scene["symbol_dx"], scene["symbol_dy"])
    pixels = carrier.astype(np.int16)
    pixels[mask] += int(scene["signal_strength"])
    return np.clip(pixels, 0, 255).astype(np.uint8)


def _draw_marker(draw: ImageDraw.ImageDraw, x: int, y: int, color: tuple[int, int, int]) -> None:
    draw.rectangle((x, y, x + MARK_SIZE - 1, y + MARK_SIZE - 1), fill=color)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS_SIZE, (244, 244, 244))
    tile_x = int(scene["tile_x"])
    tile_y = int(scene["tile_y"])
    tile = Image.fromarray(_tile_pixels(scene), mode="L").convert("RGB")
    image.paste(tile, (tile_x, tile_y))
    draw = ImageDraw.Draw(image)
    draw.rectangle(
        (tile_x - 1, tile_y - 1, tile_x + TILE_SIZE, tile_y + TILE_SIZE),
        outline=(56, 56, 56),
        width=2,
    )
    color = tuple(int(value) for value in scene["marker_color"])
    positions = (
        (tile_x - MARK_OFFSET, tile_y - MARK_OFFSET),
        (tile_x + TILE_SIZE + 2, tile_y - MARK_OFFSET),
        (tile_x - MARK_OFFSET, tile_y + TILE_SIZE + 2),
        (tile_x + TILE_SIZE + 2, tile_y + TILE_SIZE + 2),
    )
    for marker_x, marker_y in positions:
        _draw_marker(draw, marker_x, marker_y, color)
    return image


def sample_scene(seed: int) -> dict:
    seed = int(seed)
    rng = random.Random(seed)
    return {
        "seed": seed,
        "symbol": SYMBOLS[rng.randrange(len(SYMBOLS))],
        "texture_seed": rng.randrange(1, 2**31 - 1),
        "texture_phase": rng.randrange(CARRIER_TILE),
        "signal_strength": (24, 32, 40, 48)[rng.randrange(4)],
        "carrier_amplitude": (44, 52, 60)[rng.randrange(3)],
        "base_level": (94, 110, 126, 142)[rng.randrange(4)],
        "marker_color": MARKER_COLORS[rng.randrange(len(MARKER_COLORS))],
        "tile_x": rng.randint(58, 326),
        "tile_y": rng.randint(72, 156),
        "symbol_dx": rng.randint(-24, 24),
        "symbol_dy": rng.randint(-22, 22),
    }


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def margin(scene: dict) -> float:
    return float(int(scene["signal_strength"]) - SIGNAL_BOUNDARY)


def is_quarantined(scene: dict) -> bool:
    return int(scene["signal_strength"]) <= int(SIGNAL_BOUNDARY)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["texture_phase"] = int(scene["texture_phase"]) + CARRIER_TILE
    return [("carrier_phase_period", twin)]
