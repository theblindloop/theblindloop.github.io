"""Deterministic renderer and analytic consequences for the hole-count world."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "enclosed_void_compare-0.1.0"
WIDTH, HEIGHT = 960, 430
PANEL_BOXES = ((40, 70, 440, 370), (520, 70, 920, 370))
OUTER_RECT = (70, 58, 330, 242)  # local x0,y0,x1-exclusive,y1-exclusive
PANEL_BG = (235, 241, 247)
CANVAS_BG = (250, 251, 253)
PANEL_OUTLINE = (205, 214, 224)
TEXT = (40, 52, 66)
PALETTE = (
    (26, 126, 154),
    (218, 104, 48),
    (113, 78, 166),
    (31, 145, 96),
)


def _hole_rects(count: int, variant: int) -> list[list[int]]:
    """Return local, half-open rectangles with equal total area for 1 or 2 holes."""

    layouts = {
        0: {
            1: [(184, 118, 216, 142)],
            2: [(140, 118, 156, 142), (244, 118, 260, 142)],
        },
        1: {
            1: [(142, 158, 174, 182)],
            2: [(116, 158, 132, 182), (220, 158, 236, 182)],
        },
        2: {
            1: [(226, 96, 258, 120)],
            2: [(152, 96, 168, 120), (256, 96, 272, 120)],
        },
        3: {
            1: [(174, 188, 206, 212)],
            2: [(128, 188, 144, 212), (232, 188, 248, 212)],
        },
    }
    if count not in (1, 2):
        raise ValueError(f"unsupported hole count: {count}")
    return [list(rect) for rect in layouts[int(variant) % 4][count]]


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; the integer seed is the only sampling input."""

    rng = random.Random(int(seed))
    relation = int(seed) % 3
    if relation == 1:
        count_a, count_b = 2, 1
    elif relation == 2:
        count_a, count_b = 1, 2
    else:
        tied_count = 1 + ((int(seed) // 3) % 2)
        count_a = count_b = tied_count
    variant_a = rng.randrange(4)
    variant_b = rng.randrange(4)
    colors = rng.sample(list(PALETTE), 2)
    return {
        "holes_a": _hole_rects(count_a, variant_a),
        "holes_b": _hole_rects(count_b, variant_b),
        "color_a": list(colors[0]),
        "color_b": list(colors[1]),
    }


def _paint_panel(
    draw: ImageDraw.ImageDraw,
    panel: tuple[int, int, int, int],
    label: str,
    holes: list[list[int]],
    color: list[int],
) -> None:
    x0, y0, x1, y1 = panel
    draw.rounded_rectangle(
        (x0, y0, x1 - 1, y1 - 1),
        radius=22,
        fill=PANEL_BG,
        outline=PANEL_OUTLINE,
        width=2,
    )
    draw.text((x0 + 18, y0 + 14), label, fill=TEXT)
    ox, oy, ox1, oy1 = OUTER_RECT
    draw.rounded_rectangle(
        (x0 + ox, y0 + oy, x0 + ox1 - 1, y0 + oy1 - 1),
        radius=28,
        fill=tuple(color),
    )
    for hx0, hy0, hx1, hy1 in holes:
        draw.rectangle(
            (x0 + hx0, y0 + hy0, x0 + hx1 - 1, y0 + hy1 - 1),
            fill=PANEL_BG,
        )


def render(scene: dict) -> Image.Image:
    """Render only visible scene fields into a fixed RGB raster."""

    image = Image.new("RGB", (WIDTH, HEIGHT), CANVAS_BG)
    draw = ImageDraw.Draw(image)
    _paint_panel(draw, PANEL_BOXES[0], "A", scene["holes_a"], scene["color_a"])
    _paint_panel(draw, PANEL_BOXES[1], "B", scene["holes_b"], scene["color_b"])
    return image


def analytic_gold(scene: dict) -> str:
    """Compare the two visible hole-list cardinalities."""

    count_a = len(scene["holes_a"])
    count_b = len(scene["holes_b"])
    if count_a > count_b:
        return "A"
    if count_b > count_a:
        return "B"
    return "tie"


def margin(scene: dict) -> float:
    """Signed count gap, positive when panel A wins."""

    return float(len(scene["holes_a"]) - len(scene["holes_b"]))


def _valid_hole_list(holes: object) -> bool:
    if not isinstance(holes, list):
        return False
    seen: list[tuple[int, int, int, int]] = []
    for rect in holes:
        if not isinstance(rect, list) or len(rect) != 4:
            return False
        try:
            x0, y0, x1, y1 = (int(value) for value in rect)
        except (TypeError, ValueError):
            return False
        if not (0 <= x0 < x1 <= 400 and 0 <= y0 < y1 <= 300):
            return False
        if x1 - x0 < 12 or y1 - y0 < 12:
            return False
        if not (OUTER_RECT[0] + 12 <= x0 and x1 <= OUTER_RECT[2] - 12):
            return False
        if not (OUTER_RECT[1] + 12 <= y0 and y1 <= OUTER_RECT[3] - 12):
            return False
        current = (x0, y0, x1, y1)
        if any(
            current[0] < other[2]
            and current[2] > other[0]
            and current[1] < other[3]
            and current[3] > other[1]
            for other in seen
        ):
            return False
        seen.append(current)
    return True


def is_quarantined(scene: dict) -> bool:
    """Exclude ties and any malformed finite-resolution geometry."""

    if not _valid_hole_list(scene.get("holes_a")) or not _valid_hole_list(scene.get("holes_b")):
        return True
    return len(scene["holes_a"]) == len(scene["holes_b"])


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No aliases are claimed: every analytic label input is visibly painted."""

    return []
