"""Independent raster-only inverse for enclosed-background hole counts."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PANEL_BOXES = ((40, 70, 440, 370), (520, 70, 920, 370))


def _colored_mask(image: Image.Image, box: tuple[int, int, int, int]) -> np.ndarray:
    """Find saturated silhouette pixels without using the renderer palette."""

    x0, y0, x1, y1 = box
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)[y0:y1, x0:x1]
    spread = arr.max(axis=2) - arr.min(axis=2)
    brightness = arr.max(axis=2)
    return (spread >= 45) & (brightness <= 245)


def _outside_reachable(background: np.ndarray) -> np.ndarray:
    """Return background pixels connected to the crop boundary by 4-neighbour paths."""

    height, width = background.shape
    reachable = np.zeros_like(background, dtype=bool)
    queue: deque[tuple[int, int]] = deque()
    for x in range(width):
        for y in (0, height - 1):
            if background[y, x] and not reachable[y, x]:
                reachable[y, x] = True
                queue.append((y, x))
    for y in range(height):
        for x in (0, width - 1):
            if background[y, x] and not reachable[y, x]:
                reachable[y, x] = True
                queue.append((y, x))
    while queue:
        y, x = queue.popleft()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < height and 0 <= nx < width and background[ny, nx] and not reachable[ny, nx]:
                reachable[ny, nx] = True
                queue.append((ny, nx))
    return reachable


def _enclosed_count(mask: np.ndarray) -> int:
    background = ~mask
    reachable = _outside_reachable(background)
    enclosed = background & ~reachable
    visited = np.zeros_like(enclosed, dtype=bool)
    count = 0
    height, width = enclosed.shape
    for y in range(height):
        for x in range(width):
            if not enclosed[y, x] or visited[y, x]:
                continue
            count += 1
            queue: deque[tuple[int, int]] = deque([(y, x)])
            visited[y, x] = True
            while queue:
                cy, cx = queue.popleft()
                for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                    if (
                        0 <= ny < height
                        and 0 <= nx < width
                        and enclosed[ny, nx]
                        and not visited[ny, nx]
                    ):
                        visited[ny, nx] = True
                        queue.append((ny, nx))
    return count


def decision_from_image(image: Image.Image) -> str:
    """Return A, B, or tie from the final PNG alone."""

    counts = [_enclosed_count(_colored_mask(image, box)) for box in PANEL_BOXES]
    if counts[0] > counts[1]:
        return "A"
    if counts[1] > counts[0]:
        return "B"
    return "tie"
