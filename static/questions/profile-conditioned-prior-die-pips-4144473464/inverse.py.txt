"""Independent pixel-only inverse arm for the die-pip world.

Receives ONLY the final PNG and returns the number of pips drawn on the die
face as a string. It never imports renderer, prompts, generate, verify, or
analytic-gold code, and accepts no latent scene data.

Method: the die-face pips are the only dark pixels in the image (the face is a
light ivory rounded square and the background is a lighter mid-blue-gray). The
oracle thresholds the dark pip colour, then counts connected components of the
resulting binary mask via union-find over the (sparse) foreground pixels. Every
pip is a separated dark disk, so each distinct component is exactly one pip.
Tiny speckle components below a size floor are discarded.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "prior_die_pips-oracle-0.1.0"

_PIP = np.array((22, 26, 52), dtype=np.float32)
_COLOR_DIST = 80.0
_MIN_COMPONENT_AREA = 40


class _UnionFind:
    def __init__(self, size: int) -> None:
        self.parent = list(range(size))

    def find(self, a: int) -> int:
        while self.parent[a] != a:
            self.parent[a] = self.parent[self.parent[a]]
            a = self.parent[a]
        return a

    def union(self, a: int, b: int) -> None:
        ra, rb = self.find(a), self.find(b)
        if ra != rb:
            self.parent[rb] = ra


def _label_count(mask: np.ndarray) -> int:
    ys, xs = np.nonzero(mask)
    points = list(zip(xs.tolist(), ys.tolist()))
    if not points:
        return 0
    index = {}
    for idx, (x, y) in enumerate(points):
        index[(x, y)] = idx
    uf = _UnionFind(len(points))
    for idx, (x, y) in enumerate(points):
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                other = index.get((x + dx, y + dy))
                if other is not None:
                    uf.union(idx, other)
    sizes: dict[int, int] = {}
    for idx in range(len(points)):
        root = uf.find(idx)
        sizes[root] = sizes.get(root, 0) + 1
    return sum(1 for area in sizes.values() if area >= _MIN_COMPONENT_AREA)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    mask = np.linalg.norm(arr - _PIP, axis=-1) < _COLOR_DIST
    return str(_label_count(mask))


def measure(image: Image.Image) -> dict:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    mask = np.linalg.norm(arr - _PIP, axis=-1) < _COLOR_DIST
    return {
        "pip_px": int(mask.sum()),
        "components": _label_count(mask),
        "size": image.size,
    }
