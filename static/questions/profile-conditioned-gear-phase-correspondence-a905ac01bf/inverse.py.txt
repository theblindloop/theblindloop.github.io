"""Independent PNG-only inverse arm for the gear phase task."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

GRAY = np.array((220, 228, 234), dtype=np.int32)
MAGENTA = np.array((216, 27, 96), dtype=np.int32)
MARKERS = {
    "red": np.array((229, 57, 53), dtype=np.int32),
    "green": np.array((35, 155, 86), dtype=np.int32),
    "blue": np.array((40, 100, 220), dtype=np.int32),
    "orange": np.array((242, 140, 40), dtype=np.int32),
}
SLOTS = 8


def _near(arr: np.ndarray, rgb: np.ndarray, tolerance: int = 30) -> np.ndarray:
    delta = arr.astype(np.int32) - rgb
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _components(mask: np.ndarray, minimum_area: int) -> list[tuple[int, float, float]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    results: list[tuple[int, float, float]] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        count = sx = sy = 0
        while queue:
            cy, cx = queue.popleft()
            count += 1
            sx += cx
            sy += cy
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if count >= minimum_area:
            results.append((count, sx / count, sy / count))
    return results


def _one_centroid(arr: np.ndarray, rgb: np.ndarray, name: str) -> tuple[float, float]:
    parts = _components(_near(arr, rgb), minimum_area=35)
    if len(parts) != 1:
        raise ValueError(f"expected one {name} marker, found {len(parts)}")
    return parts[0][1], parts[0][2]


def _slot_index(point: tuple[float, float], center: tuple[float, float]) -> int:
    dx, dy = point[0] - center[0], point[1] - center[1]
    radius = math.hypot(dx, dy)
    if not 34.0 <= radius <= 52.0:
        raise ValueError(f"marker radius {radius:.2f} is outside the visible slot ring")
    phase = math.atan2(dx, -dy) % (2 * math.pi)
    raw = phase / (2 * math.pi / SLOTS)
    slot = int(round(raw)) % SLOTS
    residual = abs(raw - round(raw))
    residual = min(residual, abs(residual - SLOTS))
    if residual > 0.18:
        raise ValueError("marker is not centered on a resolved eight-slot phase")
    return slot


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    gear_parts = _components(_near(arr, GRAY), minimum_area=1200)
    if not 2 <= len(gear_parts) <= 4:
        raise ValueError(f"expected a 2-4 gear chain, found {len(gear_parts)} gears")
    gear_centers = sorted(((part[1], part[2]) for part in gear_parts), key=lambda item: item[0])
    if any(gear_centers[i + 1][0] - gear_centers[i][0] < 110 for i in range(len(gear_centers) - 1)):
        raise ValueError("gear centers are not a resolved left-to-right chain")

    drive_point = _one_centroid(arr, MAGENTA, "magenta drive")
    drive_slot = _slot_index(drive_point, gear_centers[0])
    if drive_slot == 0:
        raise ValueError("drive marker already lies at the top pointer")

    terminal_slots = {
        color: _slot_index(_one_centroid(arr, rgb, color), gear_centers[-1])
        for color, rgb in MARKERS.items()
    }
    if len(set(terminal_slots.values())) != 4:
        raise ValueError("terminal markers do not occupy four distinct slots")

    drive_steps = (-drive_slot) % SLOTS
    contacts = len(gear_centers) - 1
    terminal_rotation = drive_steps if contacts % 2 == 0 else (-drive_steps) % SLOTS
    required_slot = (-terminal_rotation) % SLOTS
    winners = [color for color, slot in terminal_slots.items() if slot == required_slot]
    if len(winners) != 1:
        raise ValueError(f"expected one marker in transported slot {required_slot}, found {winners}")
    return winners[0]
