"""Deterministic renderer for equal-gear phase correspondence."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 800, 360
SUPERSAMPLE = 3
GEAR_Y = 184
GEAR_SPACING = 150
GEAR_RADIUS = 68
SLOT_RADIUS = 43
SLOT_COUNT = 8

WHITE = "#FFFFFF"
NAVY = "#17324D"
GRAY = "#DCE4EA"
CYAN = "#00A6C8"
MAGENTA = "#D81B60"
MARKER_COLORS = ("red", "green", "blue", "orange")
RGB = {
    "red": "#E53935",
    "green": "#239B56",
    "blue": "#2864DC",
    "orange": "#F28C28",
}


def _terminal_target_slot(gear_count: int, drive_slot: int) -> int:
    """Slot index (0=top, increasing clockwise) that will arrive at top."""
    drive_steps_clockwise = (-drive_slot) % SLOT_COUNT
    contacts = gear_count - 1
    terminal_rotation = (
        drive_steps_clockwise if contacts % 2 == 0 else -drive_steps_clockwise
    ) % SLOT_COUNT
    return (-terminal_rotation) % SLOT_COUNT


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    gear_count = 2 + ((int(seed) // 4) % 3)
    drive_slot = 1 + rng.randrange(SLOT_COUNT - 1)
    target_slot = _terminal_target_slot(gear_count, drive_slot)

    # Answer identity cycles independently of the randomized phases. Every
    # four consecutive scene seeds contain all four labels.
    answer_color = MARKER_COLORS[int(seed) % len(MARKER_COLORS)]
    marker_slots: dict[str, int] = {answer_color: target_slot}
    remaining_colors = [color for color in MARKER_COLORS if color != answer_color]
    remaining_slots = [slot for slot in range(SLOT_COUNT) if slot != target_slot]
    rng.shuffle(remaining_colors)
    rng.shuffle(remaining_slots)
    for color, slot in zip(remaining_colors, remaining_slots[:3]):
        marker_slots[color] = slot

    # Reinsert in a stable, color-name order for canonical manifests.
    marker_slots = {color: int(marker_slots[color]) for color in MARKER_COLORS}
    return {
        "gear_count": int(gear_count),
        "drive_slot": int(drive_slot),
        "marker_slots": marker_slots,
    }


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(name, size=size * SUPERSAMPLE)


def _ellipse(draw: ImageDraw.ImageDraw, box, *, fill, outline=None, width=1) -> None:
    draw.ellipse(
        tuple(int(round(value * SUPERSAMPLE)) for value in box),
        fill=fill,
        outline=outline,
        width=max(1, width * SUPERSAMPLE),
    )


def _polygon(draw: ImageDraw.ImageDraw, points, *, fill, outline=None, width=1) -> None:
    scaled = [(int(round(x * SUPERSAMPLE)), int(round(y * SUPERSAMPLE))) for x, y in points]
    draw.polygon(scaled, fill=fill)
    if outline:
        draw.line(scaled + [scaled[0]], fill=outline, width=width * SUPERSAMPLE, joint="curve")


def _text(draw: ImageDraw.ImageDraw, xy, value: str, *, size: int, fill, anchor="mm", bold=False):
    draw.text(
        (int(xy[0] * SUPERSAMPLE), int(xy[1] * SUPERSAMPLE)),
        value,
        font=_font(size, bold=bold),
        fill=fill,
        anchor=anchor,
    )


def _gear_centers(gear_count: int) -> list[tuple[float, float]]:
    start_x = (WIDTH - GEAR_SPACING * (gear_count - 1)) / 2
    return [(start_x + index * GEAR_SPACING, GEAR_Y) for index in range(gear_count)]


def _slot_center(center: tuple[float, float], slot: int) -> tuple[float, float]:
    angle = 2 * math.pi * int(slot) / SLOT_COUNT
    return (
        center[0] + SLOT_RADIUS * math.sin(angle),
        center[1] - SLOT_RADIUS * math.cos(angle),
    )


def render(scene: dict) -> Image.Image:
    gear_count = int(scene["gear_count"])
    drive_slot = int(scene["drive_slot"])
    marker_slots = {str(key): int(value) for key, value in scene["marker_slots"].items()}
    if not 2 <= gear_count <= 4:
        raise ValueError("gear_count must be in 2..4")
    if not 0 <= drive_slot < SLOT_COUNT:
        raise ValueError("drive_slot is invalid")

    image = Image.new("RGB", (WIDTH * SUPERSAMPLE, HEIGHT * SUPERSAMPLE), WHITE)
    draw = ImageDraw.Draw(image)
    centers = _gear_centers(gear_count)

    _text(draw, (WIDTH / 2, 26), "EQUAL 8-SLOT GEAR TRAIN", size=21, fill=NAVY, bold=True)
    _text(
        draw,
        (WIDTH / 2, 53),
        "each contact reverses the same number of slots",
        size=14,
        fill=NAVY,
    )

    for index, center in enumerate(centers):
        # Sixteen shallow teeth make the contact chain visible while preserving
        # a large connected gray region for independent pixel recovery.
        cog = []
        for tooth in range(32):
            angle = 2 * math.pi * tooth / 32
            radius = GEAR_RADIUS + (4 if tooth % 2 == 0 else 0)
            cog.append((center[0] + radius * math.cos(angle), center[1] + radius * math.sin(angle)))
        _polygon(draw, cog, fill=GRAY, outline=NAVY, width=3)
        _ellipse(
            draw,
            (center[0] - 12, center[1] - 12, center[0] + 12, center[1] + 12),
            fill=WHITE,
            outline=NAVY,
            width=3,
        )
        for slot in range(SLOT_COUNT):
            sx, sy = _slot_center(center, slot)
            fill = WHITE
            if index == 0 and slot == drive_slot:
                fill = MAGENTA
            if index == gear_count - 1:
                for color in MARKER_COLORS:
                    if marker_slots[color] == slot:
                        fill = RGB[color]
                        break
            _ellipse(draw, (sx - 8, sy - 8, sx + 8, sy + 8), fill=fill, outline=NAVY, width=2)

        _text(draw, (center[0], center[1] + 91), str(index + 1), size=13, fill=NAVY, bold=True)

    # Tangency diamonds explicitly mark every local gear-contact relation.
    for left, right in zip(centers, centers[1:]):
        mx = (left[0] + right[0]) / 2
        _polygon(
            draw,
            [(mx - 7, GEAR_Y), (mx, GEAR_Y - 7), (mx + 7, GEAR_Y), (mx, GEAR_Y + 7)],
            fill=NAVY,
        )

    # Both endpoints expose the same top phase with identical cyan pointers.
    for center in (centers[0], centers[-1]):
        _polygon(
            draw,
            [(center[0] - 12, GEAR_Y - 100), (center[0] + 12, GEAR_Y - 100), (center[0], GEAR_Y - 77)],
            fill=CYAN,
            outline=NAVY,
            width=2,
        )

    # Magenta is reserved for the single drive marker so the inverse arm cannot
    # confuse instructional lettering with task evidence.
    _text(draw, (centers[0][0], 315), "TURN CLOCKWISE", size=13, fill=NAVY, bold=True)
    _text(draw, (centers[-1][0], 315), "WHICH DOT ARRIVES?", size=13, fill=NAVY, bold=True)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    gear_count = int(scene["gear_count"])
    drive_slot = int(scene["drive_slot"])
    marker_slots = {str(key): int(value) for key, value in scene["marker_slots"].items()}
    target_slot = _terminal_target_slot(gear_count, drive_slot)
    matches = [color for color in MARKER_COLORS if marker_slots.get(color) == target_slot]
    if len(matches) != 1:
        raise ValueError("scene does not bind exactly one color to the arriving slot")
    return matches[0]


def margin(scene: dict) -> float:
    gear_count = int(scene["gear_count"])
    drive_slot = int(scene["drive_slot"])
    marker_slots = {str(key): int(value) for key, value in scene["marker_slots"].items()}
    target = _terminal_target_slot(gear_count, drive_slot)
    distances = []
    for slot in marker_slots.values():
        delta = abs(int(slot) - target) % SLOT_COUNT
        distances.append(min(delta, SLOT_COUNT - delta) * (360.0 / SLOT_COUNT))
    ordered = sorted(distances)
    return float(ordered[1] - ordered[0]) if len(ordered) >= 2 else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        gear_count = int(scene["gear_count"])
        drive_slot = int(scene["drive_slot"])
        marker_slots = {str(key): int(value) for key, value in scene["marker_slots"].items()}
        target = _terminal_target_slot(gear_count, drive_slot)
        return not (
            2 <= gear_count <= 4
            and 0 <= drive_slot < SLOT_COUNT
            and set(marker_slots) == set(MARKER_COLORS)
            and len(set(marker_slots.values())) == 4
            and sum(slot == target for slot in marker_slots.values()) == 1
            and margin(scene) >= 45.0
        )
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
