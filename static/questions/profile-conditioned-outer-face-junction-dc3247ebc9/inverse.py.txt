"""Independent RGB-only inverse for the enclosed-junction question."""

from __future__ import annotations

from collections import deque

from PIL import Image


WHITE = (255, 255, 255)
COLORS = {
    "red": (218, 64, 64),
    "green": (42, 154, 92),
    "blue": (55, 103, 205),
}


def _color_components(image: Image.Image, color: tuple[int, int, int]) -> list[list[tuple[int, int]]]:
    pixels = image.load()
    width, height = image.size
    mask = bytearray(width * height)
    for y in range(height):
        row = y * width
        for x in range(width):
            if pixels[x, y] == color:
                mask[row + x] = 1

    components: list[list[tuple[int, int]]] = []
    for y in range(height):
        for x in range(width):
            index = y * width + x
            if not mask[index]:
                continue
            mask[index] = 0
            queue = deque([(x, y)])
            component: list[tuple[int, int]] = []
            while queue:
                px, py = queue.popleft()
                component.append((px, py))
                for nx, ny in ((px - 1, py), (px + 1, py), (px, py - 1), (px, py + 1)):
                    if 0 <= nx < width and 0 <= ny < height:
                        ni = ny * width + nx
                        if mask[ni]:
                            mask[ni] = 0
                            queue.append((nx, ny))
            components.append(component)
    return components


def _border_reachable(image: Image.Image) -> bytearray:
    pixels = image.load()
    width, height = image.size
    reachable = bytearray(width * height)
    queue = deque()

    def add(x: int, y: int) -> None:
        index = y * width + x
        if not reachable[index] and pixels[x, y] == WHITE:
            reachable[index] = 1
            queue.append((x, y))

    for x in range(width):
        add(x, 0)
        add(x, height - 1)
    for y in range(1, height - 1):
        add(0, y)
        add(width - 1, y)

    while queue:
        x, y = queue.popleft()
        for nx, ny in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
            if 0 <= nx < width and 0 <= ny < height:
                index = ny * width + nx
                if not reachable[index] and pixels[nx, ny] == WHITE:
                    reachable[index] = 1
                    queue.append((nx, ny))
    return reachable


def _touches_outside(component: list[tuple[int, int]], reachable: bytearray, size: tuple[int, int]) -> bool:
    width, height = size
    min_x = min(x for x, _ in component)
    max_x = max(x for x, _ in component)
    min_y = min(y for _, y in component)
    max_y = max(y for _, y in component)
    center_x = sum(x for x, _ in component) / len(component)
    center_y = sum(y for _, y in component) / len(component)
    component_radius = max(max_x - min_x, max_y - min_y) / 2.0
    low = component_radius + 1.5
    high = component_radius + 14.0
    low_sq = low * low
    high_sq = high * high
    left = max(0, int(center_x - high - 1))
    right = min(width - 1, int(center_x + high + 1))
    top = max(0, int(center_y - high - 1))
    bottom = min(height - 1, int(center_y + high + 1))
    pixels_to_check = []
    for y in range(top, bottom + 1):
        for x in range(left, right + 1):
            dx = x - center_x
            dy = y - center_y
            distance_sq = dx * dx + dy * dy
            if low_sq <= distance_sq <= high_sq:
                pixels_to_check.append((x, y))
    return any(reachable[y * width + x] for x, y in pixels_to_check)


def decision_from_image(image: Image.Image) -> str:
    """Return the only colored junction whose adjacent blank faces are bounded."""

    image = image.convert("RGB")
    centers: dict[str, list[tuple[int, int]]] = {}
    for label, color in COLORS.items():
        components = _color_components(image, color)
        if len(components) != 1 or len(components[0]) < 200:
            raise ValueError(f"expected one solid {label} junction")
        centers[label] = components[0]

    reachable = _border_reachable(image)
    enclosed = [
        label
        for label, component in centers.items()
        if not _touches_outside(component, reachable, image.size)
    ]
    if len(enclosed) != 1:
        raise ValueError(f"ambiguous outside-face result: {enclosed}")
    return enclosed[0]
