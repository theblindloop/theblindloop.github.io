"""Deterministic renderer and analytic oracle for the outer-face question."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw


WIDTH = 640
HEIGHT = 440
BACKGROUND = (255, 255, 255)
LINK = (34, 45, 58)
COLORS = {
    "red": (218, 64, 64),
    "green": (42, 154, 92),
    "blue": (55, 103, 205),
}
COLOR_NAMES = tuple(COLORS)
NODE_RADIUS = 18
LINK_WIDTH = 9
NODE_COUNT_PER_RING = 6


def _point(cx: float, cy: float, radius: float, angle: float) -> tuple[int, int]:
    return (int(round(cx + radius * math.cos(angle))), int(round(cy + radius * math.sin(angle))))


def _edge_list() -> list[tuple[str, str]]:
    edges: list[tuple[str, str]] = []
    for prefix in ("o", "i"):
        for index in range(NODE_COUNT_PER_RING):
            edges.append((f"{prefix}{index}", f"{prefix}{(index + 1) % NODE_COUNT_PER_RING}"))
    for index in range(NODE_COUNT_PER_RING):
        edges.append((f"o{index}", f"i{index}"))
    return edges


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a safe annular network with one inner colored junction."""

    rng = random.Random(int(seed))
    scale = rng.choice((0.94, 0.98, 1.02, 1.06))
    center_x = rng.choice((306, 320, 334))
    center_y = rng.choice((208, 220, 232))
    phase = rng.choice((0.0, math.pi / 6.0, math.pi / 3.0, math.pi / 2.0))
    outer_radius = round(148.0 * scale, 3)
    inner_radius = round(80.0 * scale, 3)

    nodes: list[dict[str, Any]] = []
    for ring, radius in (("outer", outer_radius), ("inner", inner_radius)):
        prefix = "o" if ring == "outer" else "i"
        for index in range(NODE_COUNT_PER_RING):
            x, y = _point(center_x, center_y, radius, phase + 2.0 * math.pi * index / 6.0)
            nodes.append(
                {
                    "id": f"{prefix}{index}",
                    "ring": ring,
                    "x": x,
                    "y": y,
                    "color": "neutral",
                }
            )

    inner_index = rng.randrange(NODE_COUNT_PER_RING)
    outer_indices = rng.sample(range(NODE_COUNT_PER_RING), 2)
    slots = [f"i{inner_index}", *(f"o{index}" for index in outer_indices)]
    labels = list(COLOR_NAMES)
    rng.shuffle(labels)
    by_id = {node["id"]: node for node in nodes}
    for node_id, label in zip(slots, labels):
        by_id[node_id]["color"] = label

    return {
        "network": {
            "width": WIDTH,
            "height": HEIGHT,
            "center": [center_x, center_y],
            "outer_radius": outer_radius,
            "inner_radius": inner_radius,
            "phase": phase,
            "nodes": nodes,
            "edges": [[left, right] for left, right in _edge_list()],
        }
    }


def render(scene: dict[str, Any]) -> Image.Image:
    graph = scene["network"]
    image = Image.new("RGB", (int(graph["width"]), int(graph["height"])), BACKGROUND)
    draw = ImageDraw.Draw(image)
    by_id = {node["id"]: node for node in graph["nodes"]}

    for left_id, right_id in graph["edges"]:
        left = by_id[left_id]
        right = by_id[right_id]
        draw.line(
            (int(left["x"]), int(left["y"]), int(right["x"]), int(right["y"])),
            fill=LINK,
            width=LINK_WIDTH,
        )

    diameter = 2 * NODE_RADIUS
    for node in graph["nodes"]:
        x = int(node["x"])
        y = int(node["y"])
        fill = COLORS.get(node["color"], LINK)
        draw.ellipse((x - NODE_RADIUS, y - NODE_RADIUS, x + NODE_RADIUS, y + NODE_RADIUS), fill=fill)
        if node["color"] != "neutral":
            # A small same-color center keeps the candidate disks solid after
            # any future Pillow version changes to ellipse edge coverage.
            draw.ellipse((x - diameter // 2 + 1, y - diameter // 2 + 1,
                          x + diameter // 2 - 1, y + diameter // 2 - 1), fill=fill)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    graph = scene["network"]
    candidates = [
        node
        for node in graph["nodes"]
        if node["color"] in COLORS
    ]
    enclosed = [node["color"] for node in candidates if node["ring"] == "inner"]
    if len(enclosed) != 1:
        raise ValueError("scene does not contain exactly one inner candidate")
    return enclosed[0]


def margin(scene: dict[str, Any]) -> float:
    graph = scene["network"]
    cx, cy = (float(value) for value in graph["center"])
    outer = float(graph["outer_radius"])
    inner = float(graph["inner_radius"])
    ring_gap = outer - inner - 2.0 * NODE_RADIUS - LINK_WIDTH
    canvas_gap = min(cx - outer - NODE_RADIUS, float(graph["width"]) - cx - outer - NODE_RADIUS,
                     cy - outer - NODE_RADIUS, float(graph["height"]) - cy - outer - NODE_RADIUS)
    return float(min(ring_gap, canvas_gap))


def is_quarantined(scene: dict[str, Any]) -> bool:
    graph = scene["network"]
    candidates = [node for node in graph["nodes"] if node["color"] in COLORS]
    if len(candidates) != 3 or len({node["color"] for node in candidates}) != 3:
        return True
    if sum(node["ring"] == "inner" for node in candidates) != 1:
        return True
    if margin(scene) < 15.0:
        return True
    points = [(float(node["x"]), float(node["y"])) for node in graph["nodes"]]
    for index, left in enumerate(points):
        for right in points[index + 1:]:
            if math.dist(left, right) < 30.0:
                return True
    return False


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
