"""Independent raster-only inverse for negative-space gate connectivity."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

# Public gate names have fixed display colors. These values are independently
# specified here; no scene, renderer, or generator state is consulted.
_COLOR_ORDER = ("red", "blue", "green", "yellow")
_REFERENCE_RGB = {
    "red": (232, 68, 83),
    "blue": (43, 121, 235),
    "green": (31, 161, 104),
    "yellow": (242, 177, 52),
}


def _marker_masks(array: np.ndarray) -> dict[str, np.ndarray]:
    # int32 is intentional: squared 8-bit channel deltas overflow int16.
    work = array.astype(np.int32)
    masks = {}
    for name, rgb in _REFERENCE_RGB.items():
        delta = work - np.asarray(rgb, dtype=np.int32)
        masks[name] = np.sum(delta * delta, axis=2) <= 30 * 30
        if int(masks[name].sum()) < 20:
            raise ValueError(f"missing or damaged {name} gate")
    return masks


def _component_labels(passable: np.ndarray) -> np.ndarray:
    height, width = passable.shape
    labels = np.full((height, width), -1, dtype=np.int32)
    next_label = 0
    for y, x in np.argwhere(passable):
        y, x = int(y), int(x)
        if labels[y, x] >= 0:
            continue
        labels[y, x] = next_label
        queue = deque([(y, x)])
        while queue:
            cy, cx = queue.popleft()
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and passable[ny, nx] and labels[ny, nx] < 0:
                    labels[ny, nx] = next_label
                    queue.append((ny, nx))
        next_label += 1
    return labels


def reconstruct_gate_components(image: Image.Image) -> dict[str, int]:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    masks = _marker_masks(array)
    # Maze floor is near-white; the deliberately gray exterior and charcoal
    # walls are not passable. Colored markers are included as gate pixels.
    passable = np.all(array >= 245, axis=2)
    for mask in masks.values():
        passable |= mask
    labels = _component_labels(passable)
    result: dict[str, int] = {}
    for name, mask in masks.items():
        values, counts = np.unique(labels[mask], return_counts=True)
        usable = [(int(count), int(value)) for value, count in zip(values, counts) if value >= 0]
        if not usable:
            raise ValueError(f"{name} gate is outside the passable reconstruction")
        result[name] = max(usable)[1]
    return result


def decision_from_image(image: Image.Image) -> str:
    components = reconstruct_gate_components(image)
    pairs = []
    for i, left in enumerate(_COLOR_ORDER):
        for right in _COLOR_ORDER[i + 1 :]:
            if components[left] == components[right]:
                pairs.append(f"{left}-{right}")
    if len(pairs) != 1:
        raise ValueError(f"raster does not contain one unique connected gate pair: {pairs}")
    return pairs[0]
