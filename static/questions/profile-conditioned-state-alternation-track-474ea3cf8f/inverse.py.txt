"""Independent pixel-only inverse for the state-alternation track."""

from __future__ import annotations

from PIL import Image


LABELS = ("A", "B", "C", "D")
CANVAS = (1760, 800)
STATE_COLORS = ((45, 126, 214), (224, 104, 64))
PANEL_COUNT = 5
TILE_COUNT = 4
PANEL_X0 = 24
PANEL_TOP = 154
PANEL_WIDTH = 328
PANEL_GAP = 14
KEY_Y = 88
KEY_XS = (154, 394)
ROW_TOP = 230
ROW_STEP = 105
MARKER_X_OFFSET = 238
MARKER_Y_OFFSET = 39


def _palette(frame: Image.Image) -> dict[tuple[int, int, int], int]:
    pixels = frame.load()
    mapping: dict[tuple[int, int, int], int] = {}
    for state, x in enumerate(KEY_XS):
        color = pixels[x, KEY_Y]
        if color not in STATE_COLORS:
            raise ValueError("state key is missing or unreadable")
        if color in mapping:
            raise ValueError("state key reuses one color")
        mapping[color] = state
    return mapping


def _decode_trajectory(frame: Image.Image, palette: dict[tuple[int, int, int], int]) -> list[list[int]]:
    pixels = frame.load()
    table: list[list[int]] = []
    for panel in range(PANEL_COUNT):
        panel_x = PANEL_X0 + panel * (PANEL_WIDTH + PANEL_GAP)
        row: list[int] = []
        for tile in range(TILE_COUNT):
            x = panel_x + MARKER_X_OFFSET
            y = ROW_TOP + tile * ROW_STEP + MARKER_Y_OFFSET
            color = pixels[x, y]
            if color not in palette:
                raise ValueError(f"state marker P{panel + 1}/{LABELS[tile]} is unreadable")
            row.append(palette[color])
        table.append(row)
    return table


def decision_from_image(image: Image.Image) -> str:
    """Read only the rasterized state key and twenty state markers."""

    frame = image.convert("RGB")
    if frame.size != CANVAS:
        raise ValueError(f"unexpected raster size {frame.size}")
    palette = _palette(frame)
    states = _decode_trajectory(frame, palette)
    scores = [
        sum(states[panel][tile] != states[panel - 1][tile] for panel in range(1, PANEL_COUNT))
        for tile in range(TILE_COUNT)
    ]
    best = max(scores)
    winners = [index for index, score in enumerate(scores) if score == best]
    if len(winners) != 1:
        raise ValueError("trajectory does not expose one unique winner")
    return LABELS[winners[0]]
