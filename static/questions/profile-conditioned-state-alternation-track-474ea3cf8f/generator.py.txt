"""Deterministic renderer and analytic side for the state-alternation world."""

from __future__ import annotations

import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont


LABELS = ("A", "B", "C", "D")
PANEL_COUNT = 5
TILE_COUNT = 4
STATE_COUNT = 2
CANVAS = (1760, 800)

BACKGROUND = (248, 250, 252)
CARD = (255, 255, 255)
INK = (38, 46, 58)
MUTED = (91, 103, 116)
OUTLINE = (169, 180, 191)
STATE_COLORS = ((45, 126, 214), (224, 104, 64))
STATE_NAMES = ("REST", "ACTIVE")
PANEL_TINTS = (
    (239, 245, 251),
    (246, 241, 235),
    (240, 248, 242),
    (246, 241, 249),
)

PANEL_X0 = 24
PANEL_TOP = 154
PANEL_WIDTH = 328
PANEL_HEIGHT = 610
PANEL_GAP = 14
KEY_Y = 88
KEY_XS = (154, 394)
ROW_TOP = 230
ROW_STEP = 105
TILE_LEFT_INSET = 18
TILE_TOP_OFFSET = 0
TILE_WIDTH = PANEL_WIDTH - 36
TILE_HEIGHT = 78
MARKER_X_OFFSET = 238
MARKER_Y_OFFSET = 39

_FONT_PATHS = (
    "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
    "/usr/share/fonts/dejavu/DejaVuSans.ttf",
)


def _font(size: int):
    for candidate in _FONT_PATHS:
        if Path(candidate).is_file():
            return ImageFont.truetype(candidate, size=size)
    return ImageFont.load_default()


def _centered_text(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font, fill):
    box = draw.textbbox((0, 0), text, font=font)
    width = box[2] - box[0]
    height = box[3] - box[1]
    draw.text((xy[0] - width / 2, xy[1] - height / 2 - box[1]), text, font=font, fill=fill)


def _state_color(scene: dict, state: int) -> tuple[int, int, int]:
    order = scene["state_palette_order"]
    return STATE_COLORS[int(order[int(state)])]


def _flip_scores(scene: dict) -> list[int]:
    states = scene["states"]
    if len(states) != PANEL_COUNT or any(len(panel) != TILE_COUNT for panel in states):
        raise ValueError("states must be a five by four binary table")
    scores = [0] * TILE_COUNT
    for panel in range(1, PANEL_COUNT):
        for tile in range(TILE_COUNT):
            before = int(states[panel - 1][tile])
            after = int(states[panel][tile])
            if before not in (0, 1) or after not in (0, 1):
                raise ValueError("state values must be binary")
            scores[tile] += before != after
    return scores


def sample_scene(seed: int) -> dict:
    """Sample one balanced scene with exactly one four-flip trajectory."""

    seed = int(seed)
    logic_rng = random.Random(seed * 1000003 + 17)
    style_rng = random.Random((seed + 104729) * 1000033 + 31)
    winner = seed % TILE_COUNT
    states = [[0 for _ in range(TILE_COUNT)] for _ in range(PANEL_COUNT)]
    start = logic_rng.randrange(STATE_COUNT)
    for panel in range(PANEL_COUNT):
        states[panel][winner] = start ^ (panel & 1)

    for tile in range(TILE_COUNT):
        if tile == winner:
            continue
        while True:
            trajectory = [logic_rng.randrange(STATE_COUNT) for _ in range(PANEL_COUNT)]
            flips = sum(trajectory[index] != trajectory[index - 1] for index in range(1, PANEL_COUNT))
            if flips < PANEL_COUNT - 1:
                break
        for panel, state in enumerate(trajectory):
            states[panel][tile] = state

    state_palette_order = [0, 1]
    style_rng.shuffle(state_palette_order)
    marker_size = 7 if seed % 19 == 0 else style_rng.choice((14, 16, 18, 20))
    panel_tint = style_rng.randrange(len(PANEL_TINTS))
    scene = {
        "states": states,
        "state_palette_order": state_palette_order,
        "marker_size": marker_size,
        "panel_tint": panel_tint,
    }
    if analytic_gold(scene) != LABELS[winner]:
        raise AssertionError("sampler failed to create the requested unique winner")
    return scene


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS, BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(25)
    section_font = _font(16)
    label_font = _font(22)
    small_font = _font(13)
    state_font = _font(14)

    draw.text((24, 20), "STATE ALTERNATION TRACK", font=title_font, fill=INK)
    draw.text(
        (24, 57),
        "Compare each labeled tile across the numbered panels; the final answer is not printed.",
        font=small_font,
        fill=MUTED,
    )
    draw.text((24, 78), "VISIBLE STATES", font=section_font, fill=INK)
    for state, cx in enumerate(KEY_XS):
        draw.ellipse(
            (cx - 14, KEY_Y - 14, cx + 14, KEY_Y + 14),
            fill=_state_color(scene, state),
            outline=INK,
            width=1,
        )
        _centered_text(draw, (cx + 67, KEY_Y), f"{state}: {STATE_NAMES[state]}", state_font, INK)

    tint = PANEL_TINTS[int(scene["panel_tint"])]
    marker_size = int(scene["marker_size"])
    for panel in range(PANEL_COUNT):
        panel_x = PANEL_X0 + panel * (PANEL_WIDTH + PANEL_GAP)
        draw.rounded_rectangle(
            (panel_x, PANEL_TOP, panel_x + PANEL_WIDTH, PANEL_TOP + PANEL_HEIGHT),
            radius=12,
            fill=tint,
            outline=OUTLINE,
            width=2,
        )
        draw.text((panel_x + 16, PANEL_TOP + 14), f"P{panel + 1}", font=section_font, fill=INK)
        draw.line(
            (panel_x + 16, PANEL_TOP + 53, panel_x + PANEL_WIDTH - 16, PANEL_TOP + 53),
            fill=OUTLINE,
            width=1,
        )
        for tile, label in enumerate(LABELS):
            top = ROW_TOP + tile * ROW_STEP
            left = panel_x + TILE_LEFT_INSET
            state = int(scene["states"][panel][tile])
            draw.rounded_rectangle(
                (left, top, left + TILE_WIDTH, top + TILE_HEIGHT),
                radius=8,
                fill=CARD,
                outline=OUTLINE,
                width=1,
            )
            _centered_text(draw, (left + 35, top + TILE_HEIGHT / 2), label, label_font, INK)
            draw.line((left + 64, top + 13, left + 64, top + TILE_HEIGHT - 13), fill=OUTLINE, width=1)
            marker_x = panel_x + MARKER_X_OFFSET
            marker_y = top + MARKER_Y_OFFSET
            draw.ellipse(
                (marker_x - marker_size, marker_y - marker_size,
                 marker_x + marker_size, marker_y + marker_size),
                fill=_state_color(scene, state),
                outline=INK,
                width=1,
            )
            _centered_text(draw, (left + 269, top + TILE_HEIGHT / 2), STATE_NAMES[state], state_font, INK)
    return image


def analytic_gold(scene: dict) -> str:
    scores = _flip_scores(scene)
    best = max(scores)
    winners = [index for index, score in enumerate(scores) if score == best]
    if len(winners) != 1:
        raise ValueError("scene does not have a unique alternation winner")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    scores = _flip_scores(scene)
    ordered = sorted(scores, reverse=True)
    return float(ordered[0] - ordered[1])


def is_quarantined(scene: dict) -> bool:
    scores = _flip_scores(scene)
    return int(scene["marker_size"]) < 10 or scores.count(max(scores)) != 1 or margin(scene) <= 0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No nontrivial aliases: state trajectories determine the label directly."""

    return []
