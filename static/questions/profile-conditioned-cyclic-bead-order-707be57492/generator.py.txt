"""Deterministic renderer for clockwise cyclic bead-order correspondence."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 900, 720
COLORS = ("red", "orange", "yellow", "green", "blue", "purple")
PALETTE = {
    "red": (220, 55, 60),
    "orange": (242, 133, 34),
    "yellow": (242, 205, 48),
    "green": (45, 158, 84),
    "blue": (48, 112, 210),
    "purple": (145, 75, 190),
}
PANEL_CENTERS = ((450, 215), (170, 545), (450, 545), (730, 545))
OPTION_LABELS = ("A", "B", "C")


def _rotate(values: list[str], amount: int) -> list[str]:
    amount %= len(values)
    return values[amount:] + values[:amount]


def _canonical(order: list[str]) -> tuple[str, ...]:
    values = tuple(order)
    return min(values[index:] + values[:index] for index in range(len(values)))


def _score(reference: list[str], option: list[str]) -> int:
    n = len(reference)
    return max(
        sum(reference[index] == option[(index + shift) % n] for index in range(n))
        for shift in range(n)
    )


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    reference = list(COLORS)
    rng.shuffle(reference)

    # One near-miss has an adjacent transposition (score 4); the other is the
    # reversed orientation. Both contain exactly the same colors and ink.
    transposed = list(reference)
    swap_at = rng.randrange(6)
    other = (swap_at + 1) % 6
    transposed[swap_at], transposed[other] = transposed[other], transposed[swap_at]
    reflected = [reference[0], *reversed(reference[1:])]

    correct_index = int(seed) % 3
    distractors = [transposed, reflected]
    rng.shuffle(distractors)
    options: list[list[str]] = []
    nuisance_rotation = rng.randrange(6)
    for index in range(3):
        if index == correct_index:
            options.append(_rotate(reference, nuisance_rotation))
        else:
            options.append(distractors.pop())

    orders = [reference, *options]
    phases = [rng.randrange(6) for _ in range(4)]
    shape_choices = [(91, 70), (82, 80), (96, 63), (76, 86), (88, 74)]
    loop_shapes = []
    for _ in range(4):
        rx, ry = rng.choice(shape_choices)
        loop_shapes.append({"rx": rx, "ry": ry})
    return {"orders": orders, "phases": phases, "loop_shapes": loop_shapes}


def render(scene: dict) -> Image.Image:
    orders = scene["orders"]
    phases = scene["phases"]
    loop_shapes = scene["loop_shapes"]
    image = Image.new("RGB", (WIDTH, HEIGHT), (248, 247, 243))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    draw.text((450, 20), "CLOCKWISE BEAD-LOOP CORRESPONDENCE", fill=(25, 31, 38), font=font, anchor="ma")
    draw.text((450, 40), "Same clockwise color cycle; starting bead and loop shape do not matter. Reversal does not match.", fill=(55, 62, 70), font=font, anchor="ma")

    panel_boxes = ((322, 91, 578, 339), (42, 411, 298, 679), (322, 411, 578, 679), (602, 411, 858, 679))
    for panel_index, (center, box, order, phase, shape) in enumerate(
        zip(PANEL_CENTERS, panel_boxes, orders, phases, loop_shapes)
    ):
        draw.rounded_rectangle(box, radius=16, fill=(255, 255, 255), outline=(167, 173, 181), width=3)
        label = "REFERENCE" if panel_index == 0 else OPTION_LABELS[panel_index - 1]
        draw.text((center[0], box[1] + 20), label, fill=(25, 31, 38), font=font, anchor="mm")
        cx, cy = center
        rx, ry = int(shape["rx"]), int(shape["ry"])
        draw.ellipse((cx - rx, cy - ry, cx + rx, cy + ry), outline=(55, 61, 68), width=7)
        for index, color in enumerate(order):
            slot = (index + int(phase)) % 6
            angle = 2.0 * math.pi * slot / 6.0
            x = round(cx + rx * math.cos(angle))
            y = round(cy + ry * math.sin(angle))
            radius = 15
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=PALETTE[color], outline=(30, 34, 39), width=3)
            draw.ellipse((x - 4, y - 6, x + 3, y + 1), fill=(255, 255, 255))
    return image


def _panel_scores(scene: dict) -> list[int]:
    orders = scene["orders"]
    return [_score(orders[0], option) for option in orders[1:]]


def analytic_gold(scene: dict) -> str:
    scores = _panel_scores(scene)
    winner = max(range(3), key=scores.__getitem__)
    if scores.count(scores[winner]) != 1:
        raise ValueError("scene has no unique matching option")
    return OPTION_LABELS[winner]


def margin(scene: dict) -> float:
    scores = sorted(_panel_scores(scene), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    scores = _panel_scores(scene)
    return scores.count(max(scores)) != 1 or margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["orders"] = [_rotate(list(order), 1) for order in scene["orders"]]
    twin["phases"] = [(int(phase) + 1) % 6 for phase in scene["phases"]]
    return [("cyclic_representation_shift", twin)]
