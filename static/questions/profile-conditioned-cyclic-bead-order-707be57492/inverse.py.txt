"""Independent image-only inverse for cyclic bead order.

The implementation segments the six public bead colors, finds their connected
components, groups components into panels from the observed gaps, and compares
clockwise centroid order. It receives no scene or renderer state.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

PALETTE = {
    "red": (220, 55, 60),
    "orange": (242, 133, 34),
    "yellow": (242, 205, 48),
    "green": (45, 158, 84),
    "blue": (48, 112, 210),
    "purple": (145, 75, 190),
}
LABELS = ("A", "B", "C")


def _components(image: Image.Image) -> list[tuple[str, float, float, int]]:
    rgb = np.asarray(image.convert("RGB"))
    height, width, _ = rgb.shape
    found: list[tuple[str, float, float, int]] = []
    for name, target in PALETTE.items():
        ys, xs = np.nonzero(np.all(rgb == np.asarray(target, dtype=np.uint8), axis=2))
        remaining = {int(y) * width + int(x) for y, x in zip(ys, xs)}
        while remaining:
            stack = [remaining.pop()]
            count = sx = sy = 0
            while stack:
                pos = stack.pop()
                y, x = divmod(pos, width)
                count += 1
                sx += x
                sy += y
                for neighbor in (pos - 1, pos + 1, pos - width, pos + width):
                    if neighbor in remaining:
                        remaining.remove(neighbor)
                        stack.append(neighbor)
            if count >= 250:
                found.append((name, sx / count, sy / count, count))
    return found


def _canonical(panel: list[tuple[str, float, float, int]]) -> tuple[str, ...]:
    if len(panel) != 6 or {item[0] for item in panel} != set(PALETTE):
        raise ValueError("panel does not contain one component of every bead color")
    cx = sum(item[1] for item in panel) / 6.0
    cy = sum(item[2] for item in panel) / 6.0
    clockwise = sorted(panel, key=lambda item: math.atan2(item[2] - cy, item[1] - cx))
    values = tuple(item[0] for item in clockwise)
    return min(values[index:] + values[:index] for index in range(6))


def decision_from_image(image: Image.Image) -> str:
    components = _components(image)
    if len(components) != 24:
        raise ValueError(f"expected 24 visible beads, found {len(components)}")

    by_y = sorted(components, key=lambda item: item[2])
    # The uniquely largest vertical gap separates the reference row from options.
    gaps = [by_y[index + 1][2] - by_y[index][2] for index in range(len(by_y) - 1)]
    split = max(range(len(gaps)), key=gaps.__getitem__) + 1
    top, bottom = by_y[:split], by_y[split:]
    if len(top) != 6 or len(bottom) != 18:
        raise ValueError("could not recover the reference and option rows")

    ordered_bottom = sorted(bottom, key=lambda item: item[1])
    options = [ordered_bottom[index : index + 6] for index in range(0, 18, 6)]
    reference_signature = _canonical(top)
    matches = [_canonical(option) == reference_signature for option in options]
    if sum(matches) != 1:
        raise ValueError("image does not contain one unique cyclic-order match")
    return LABELS[matches.index(True)]
