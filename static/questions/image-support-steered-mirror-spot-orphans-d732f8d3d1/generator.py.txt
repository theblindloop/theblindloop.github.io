"""Renderer and analytic gold for ``mirror_spot_orphans``.

One moth-like figure of medium span carries several small filled spots of a
single colour.  Two small dark-grey squares mark the two ends of the figure's
mirror line.  A spot is *paired* when another spot sits at its mirrored
position across that line; the decision is how many spots are unpaired.

The gold is recomputed geometrically from the stored spot offsets, never read
off a stored answer field.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "mirror_spot_orphans-renderer-0.1.0"

W = H = 512
BG = (250, 249, 246)
WING_FILL = (233, 232, 237)
WING_EDGE = (213, 212, 219)
TICK = (92, 92, 98)

SPOT_COLOURS: dict[str, tuple[int, int, int]] = {
    "navy": (26, 54, 132),
    "crimson": (166, 28, 54),
    "forest": (24, 102, 62),
    "plum": (108, 38, 120),
    "teal": (16, 96, 110),
}

SPOT_R = 10.0
TICK_HALF = 8.0

# Declared construction rules.  A spot is paired only when its mirror position
# is occupied to within MATCH_TOL px; it is unpaired only when the mirror
# position is empty by at least CLEAR_MIN px.  Both bounds are visible sizes:
# MATCH_TOL is well under one spot radius of slack and CLEAR_MIN is more than a
# whole spot diameter of empty background.
MATCH_TOL = 12.0
CLEAR_MIN = 24.0
MIN_SPOT_GAP = 32.0
MIN_OFF_AXIS = 20.0
EDGE_KEEP = 34.0
MAX_UNPAIRED = 3

QUARANTINE_MARGIN = 6.0
TIGHT_MARGIN = 12.0

DECISIONS = ("0", "1", "2", "3")

# Wing silhouette in axis-local coordinates (u along the mirror line, positive
# towards the head; v across it, positive to the right of that direction).
_WING = (
    (74.0, 12.0),
    (60.0, 58.0),
    (22.0, 92.0),
    (-32.0, 88.0),
    (-66.0, 50.0),
    (-54.0, 14.0),
    (-10.0, 9.0),
)


# --------------------------------------------------------------------------
# geometry helpers
# --------------------------------------------------------------------------
def axis_frame(scene: dict) -> tuple[tuple[float, float], tuple[float, float]]:
    """Unit vector along the mirror line and the unit vector across it."""
    th = math.radians(float(scene["theta"]))
    along = (math.sin(th), -math.cos(th))
    across = (math.cos(th), math.sin(th))
    return along, across


def to_image(scene: dict, u: float, v: float) -> tuple[float, float]:
    along, across = axis_frame(scene)
    return (
        float(scene["ax"]) + u * along[0] + v * across[0],
        float(scene["ay"]) + u * along[1] + v * across[1],
    )


def spot_offsets(scene: dict) -> list[tuple[float, float]]:
    """Every drawn spot in axis-local coordinates, in draw order."""
    out: list[tuple[float, float]] = []
    for u, v in scene["pairs"]:
        out.append((float(u), float(v)))
        out.append((float(u), -float(v)))
    for u, v in scene["orphans"]:
        out.append((float(u), float(v)))
    return out


def _mirror_clearances(offsets: list[tuple[float, float]]) -> list[float]:
    """For each spot, the distance from its mirror position to the nearest
    *other* spot centre."""
    out = []
    for i, (u, v) in enumerate(offsets):
        mu, mv = u, -v
        best = float("inf")
        for j, (pu, pv) in enumerate(offsets):
            if i == j:
                continue
            best = min(best, math.hypot(mu - pu, mv - pv))
        out.append(best)
    return out


# --------------------------------------------------------------------------
# contract API
# --------------------------------------------------------------------------
def analytic_gold(scene: dict) -> str:
    """Number of spots whose mirror position holds no other spot."""
    offsets = spot_offsets(scene)
    unpaired = sum(1 for c in _mirror_clearances(offsets) if c > MATCH_TOL)
    return str(min(unpaired, MAX_UNPAIRED + 1))


def margin(scene: dict) -> float:
    """Minimum distance, in canvas px, to any answer- or legibility-changing
    event."""
    offsets = spot_offsets(scene)
    if not offsets:
        return 0.0
    terms: list[float] = []

    clear = _mirror_clearances(offsets)
    for c in clear:
        # every spot is either well inside the paired band or well outside it
        terms.append(abs(c - MATCH_TOL) if c <= MATCH_TOL else c - CLEAR_MIN)

    for i, (u, v) in enumerate(offsets):
        terms.append(2.0 * abs(v) - 2.0 * MIN_OFF_AXIS)  # never sits on the axis
        for j in range(i + 1, len(offsets)):
            pu, pv = offsets[j]
            terms.append(math.hypot(u - pu, v - pv) - MIN_SPOT_GAP)

    half = float(scene["half_len"])
    marks = [to_image(scene, u, v) for u, v in offsets]
    ticks = [to_image(scene, half, 0.0), to_image(scene, -half, 0.0)]
    for tx, ty in ticks:
        for mx, my in marks:
            terms.append(math.hypot(tx - mx, ty - my) - MIN_SPOT_GAP)
    for x, y in marks + ticks:
        terms.append(min(x, y, W - x, H - y) - EDGE_KEEP)

    return float(min(terms))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def margin_bin(m: float) -> str:
    if m < QUARANTINE_MARGIN:
        return "boundary"
    if m < TIGHT_MARGIN:
        return "tight"
    return "clear"


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Re-descriptions of the same picture that must not move the gold."""
    out: list[tuple[str, dict]] = []

    wrapped = dict(scene)
    wrapped["theta"] = float(scene["theta"]) + 360.0
    out.append(("theta_full_turn", wrapped))

    if scene["pairs"]:
        flipped = dict(scene)
        pairs = [list(p) for p in scene["pairs"]]
        pairs[0] = [pairs[0][0], -pairs[0][1]]
        flipped["pairs"] = pairs
        out.append(("pair_side_relabel", flipped))

    if len(scene["pairs"]) > 1 or len(scene["orphans"]) > 1:
        reordered = dict(scene)
        reordered["pairs"] = [list(p) for p in reversed(scene["pairs"])]
        reordered["orphans"] = [list(o) for o in reversed(scene["orphans"])]
        out.append(("spot_listing_order", reordered))

    return out


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------
def _try_scene(rng: random.Random) -> dict | None:
    theta = rng.uniform(-36.0, 36.0)
    ax = rng.uniform(198.0, 314.0)
    ay = rng.uniform(198.0, 314.0)
    half_len = rng.uniform(86.0, 100.0)
    colour = rng.choice(sorted(SPOT_COLOURS))
    tight = rng.random() < 0.35

    n_orph = rng.choice((0, 1, 2, 3))
    n_pairs = rng.choice((2, 3, 4)) if n_orph < 2 else rng.choice((2, 2, 3))
    split = n_orph >= 2 and rng.random() < 0.6

    pairs: list[list[float]] = []
    placed: list[tuple[float, float]] = []

    def ok(u: float, v: float, mirror_free: bool) -> bool:
        if abs(v) < 26.0 or abs(v) > 72.0 or abs(u) > 62.0:
            return False
        for pu, pv in placed:
            if math.hypot(u - pu, v - pv) < 44.0:
                return False
        if mirror_free:
            for pu, pv in placed:
                if math.hypot(u - pu, -v - pv) < 32.0:
                    return False
        return True

    for _ in range(n_pairs):
        for _ in range(90):
            u = rng.uniform(-62.0, 62.0)
            v = rng.uniform(28.0, 70.0)
            if ok(u, v, False) and ok(u, -v, False):
                pairs.append([u, v])
                placed.extend([(u, v), (u, -v)])
                break
        else:
            return None

    orphans: list[list[float]] = []
    for k in range(n_orph):
        sign = 1.0 if (not split or k == 0) else -1.0
        if not split:
            sign = 1.0 if rng.random() < 0.5 else -1.0
        want_tight = tight and k == 0
        for _ in range(260):
            u = rng.uniform(-62.0, 62.0)
            v = sign * rng.uniform(28.0, 70.0)
            if not ok(u, v, True):
                continue
            probe = placed + [(u, v)]
            clear = min(
                math.hypot(u - pu, -v - pv)
                for pu, pv in probe
                if (pu, pv) != (u, v)
            )
            if want_tight:
                if not (30.5 <= clear <= 35.8):
                    continue
            elif clear < 38.0:
                continue
            orphans.append([u, v])
            placed.append((u, v))
            break
        else:
            return None

    if not orphans and tight:
        return None

    scene = {
        "width": W,
        "height": H,
        "ax": ax,
        "ay": ay,
        "theta": theta,
        "half_len": half_len,
        "pairs": pairs,
        "orphans": orphans,
        "colour": colour,
    }
    if analytic_gold(scene) != str(n_orph):
        return None
    if margin(scene) < QUARANTINE_MARGIN:
        return None

    offsets = spot_offsets(scene)
    right = sum(1 for _, v in offsets if v > 0)
    left = len(offsets) - right
    prior = str(min(abs(right - left), MAX_UNPAIRED + 1))
    scene["canonical_prior_answer"] = prior
    scene["variant"] = "control" if prior == analytic_gold(scene) else "counterfactual"
    return scene


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) & 0x7FFFFFFF)
    for _ in range(400):
        scene = _try_scene(rng)
        if scene is not None:
            return scene
    raise RuntimeError(f"could not sample a legible scene for seed {seed}")


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BG)
    draw = ImageDraw.Draw(image)

    for side in (1.0, -1.0):
        poly = [to_image(scene, u, side * v) for u, v in _WING]
        draw.polygon(poly, fill=WING_FILL, outline=WING_EDGE)

    half = float(scene["half_len"])
    for end in (half, -half):
        tx, ty = to_image(scene, end, 0.0)
        draw.rectangle(
            (tx - TICK_HALF, ty - TICK_HALF, tx + TICK_HALF, ty + TICK_HALF),
            fill=TICK,
        )

    rgb = SPOT_COLOURS[scene["colour"]]
    for u, v in spot_offsets(scene):
        x, y = to_image(scene, u, v)
        draw.ellipse((x - SPOT_R, y - SPOT_R, x + SPOT_R, y + SPOT_R), fill=rgb)

    return image
