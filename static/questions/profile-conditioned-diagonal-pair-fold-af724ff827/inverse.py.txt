"""Independent inverse arm: consumes pixels only."""

from __future__ import annotations

from PIL import Image

PATTERNS = {
    "anchor": ("00011000", "00011000", "00111100", "00011000", "10011001", "00111100", "00011000", "00000000"),
    "boot":   ("00110000", "00110000", "00110000", "00110000", "00110000", "00111110", "00111110", "00000000"),
    "chair":  ("00110000", "00110000", "00111110", "00110010", "00110010", "00111110", "00000000", "00000000"),
    "sail":   ("00010000", "00110000", "01110000", "11110000", "01111000", "00111100", "00011000", "00000000"),
}


def _runs(values: list[int]) -> list[int]:
    groups = []
    for value in values:
        if not groups or value > groups[-1][-1] + 1:
            groups.append([value])
        else:
            groups[-1].append(value)
    return [round(sum(group) / len(group)) for group in groups]


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    cyan = []
    for y in range(rgb.height):
        for x in range(rgb.width):
            r, g, b = rgb.getpixel((x, y))
            if b > 130 and g > 110 and r < 40 and b > r + 80:
                cyan.append((x, y))
    if len(cyan) < 100:
        raise ValueError("cyan block grid not found")
    x_counts, y_counts = {}, {}
    for x, y in cyan:
        x_counts[x] = x_counts.get(x, 0) + 1
        y_counts[y] = y_counts.get(y, 0) + 1
    xs = _runs(sorted(x for x, count in x_counts.items() if count > 100))
    ys = _runs(sorted(y for y, count in y_counts.items() if count > 100))
    if len(xs) != 5 or len(ys) != 9:
        raise ValueError("expected four by eight visibly tiled block grid")
    source = [[0] * 8 for _ in range(16)]
    for br in range(8):
        for bc in range(4):
            x0, x1, y0, y1 = xs[bc], xs[bc + 1], ys[br], ys[br + 1]
            for dr in range(2):
                for dc in range(2):
                    x = round(x0 + (dc + 0.5) * (x1 - x0) / 2)
                    y = round(y0 + (dr + 0.5) * (y1 - y0) / 2)
                    r, g, b = rgb.getpixel((x, y))
                    source[2 * br + dr][2 * bc + dc] = int(r + g + b < 300)
    decoded = [[0] * 8 for _ in range(8)]
    for br in range(8):
        for bc in range(4):
            r, c = 2 * br, 2 * bc
            decoded[br][2 * bc] = source[r][c] ^ source[r + 1][c + 1]
            decoded[br][2 * bc + 1] = source[r][c + 1] ^ source[r + 1][c]
    ranked = sorted((sum(bit != int(char) for row, template in zip(decoded, rows) for bit, char in zip(row, template)), label) for label, rows in PATTERNS.items())
    if ranked[1][0] - ranked[0][0] < 4:
        raise ValueError("ambiguous decoded symbol")
    return ranked[0][1]
