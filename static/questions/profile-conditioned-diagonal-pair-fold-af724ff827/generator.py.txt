"""Deterministic renderer and analytic transform for diagonal_pair_fold."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

LABELS = ("anchor", "boot", "chair", "sail")
PATTERNS = {
    "anchor": ("00011000", "00011000", "00111100", "00011000", "10011001", "00111100", "00011000", "00000000"),
    "boot":   ("00110000", "00110000", "00110000", "00110000", "00110000", "00111110", "00111110", "00000000"),
    "chair":  ("00110000", "00110000", "00111110", "00110010", "00110010", "00111110", "00000000", "00000000"),
    "sail":   ("00010000", "00110000", "01110000", "11110000", "01111000", "00111100", "00011000", "00000000"),
}
PALETTES = ((28, 35, 48), (55, 29, 72), (19, 55, 52), (65, 36, 24))


def _bits(label: str) -> list[list[int]]:
    return [[int(value) for value in row] for row in PATTERNS[label]]


def _decode(source: list[list[int]]) -> list[list[int]]:
    result = [[0] * 8 for _ in range(8)]
    for block_row in range(8):
        for block_col in range(4):
            r, c = 2 * block_row, 2 * block_col
            result[block_row][2 * block_col] = source[r][c] ^ source[r + 1][c + 1]
            result[block_row][2 * block_col + 1] = source[r][c + 1] ^ source[r + 1][c]
    return result


def _distances(decoded: list[list[int]]) -> list[tuple[int, str]]:
    return sorted((sum(a != b for row_a, row_b in zip(decoded, _bits(label)) for a, b in zip(row_a, row_b)), label) for label in LABELS)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    label = LABELS[int(seed) % len(LABELS)]
    output = _bits(label)
    flat = [value for row in output for value in row]
    zero_indices = [index for index, value in enumerate(flat) if value == 0]
    rng.shuffle(zero_indices)
    high_zeros = set(zero_indices[: len(zero_indices) // 2])
    bases = []
    for index, value in enumerate(flat):
        bases.append(rng.randrange(2) if value else int(index in high_zeros))
    source = [[0] * 8 for _ in range(16)]
    for block_row in range(8):
        for block_col in range(4):
            for offset in range(2):
                out_index = block_row * 8 + 2 * block_col + offset
                value, base = flat[out_index], bases[out_index]
                r, c = 2 * block_row, 2 * block_col
                if offset == 0:
                    source[r][c], source[r + 1][c + 1] = base, base ^ value
                else:
                    source[r][c + 1], source[r + 1][c] = base, base ^ value
    return {"source": source, "palette": list(PALETTES[rng.randrange(len(PALETTES))])}


def render(scene: dict) -> Image.Image:
    source = scene["source"]
    dark = tuple(scene.get("palette", PALETTES[0]))
    image = Image.new("RGB", (560, 620), (247, 245, 238))
    draw = ImageDraw.Draw(image)
    left, top, cell = 168, 70, 28
    for r in range(16):
        for c in range(8):
            x0, y0 = left + c * cell, top + r * cell
            fill = dark if source[r][c] else (238, 236, 228)
            draw.rectangle((x0 + 2, y0 + 2, x0 + cell - 2, y0 + cell - 2), fill=fill)
    minor = (166, 169, 166)
    cyan = (0, 151, 184)
    for c in range(9):
        x = left + c * cell
        draw.line((x, top, x, top + 16 * cell), fill=cyan if c % 2 == 0 else minor, width=4 if c % 2 == 0 else 2)
    for r in range(17):
        y = top + r * cell
        draw.line((left, y, left + 8 * cell, y), fill=cyan if r % 2 == 0 else minor, width=4 if r % 2 == 0 else 2)
    draw.text((118, 24), "PAIR BOTH DIAGONALS IN EACH 2 x 2 BLOCK", fill=(45, 55, 63))
    draw.text((134, 548), "different = dark     same = light", fill=(45, 55, 63))
    draw.text((146, 578), "write main diagonal result first", fill=(45, 55, 63))
    return image


def analytic_gold(scene: dict) -> str:
    return _distances(_decode(scene["source"]))[0][1]


def margin(scene: dict) -> float:
    ranked = _distances(_decode(scene["source"]))
    return float(ranked[1][0] - ranked[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
