"""Latent sampler and raster renderer for cumulative switch-trace correspondence."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 768
HEIGHT = 768
RUN_LENGTH = 156
ROW_OFFSETS = (-18, -6, 6, 18)
ATTACHMENT_DISTANCES = (30, 60, 90, 120)
ANSWER_COLORS = ("red", "green", "blue")
RGB = {
    "red": (214, 60, 60),
    "green": (45, 157, 91),
    "blue": (53, 105, 210),
    "gold": (226, 174, 38),
}
PALETTES = (
    ((28, 34, 40), (246, 247, 242)),
    ((24, 47, 57), (248, 245, 239)),
    ((49, 35, 47), (244, 248, 246)),
)


def _different_code(reference: list[int], flips: Iterable[int]) -> list[int]:
    code = list(reference)
    for index in flips:
        code[index] = 1 - code[index]
    return code


def _execute(gates: list[int]) -> list[int]:
    """Return state after each gate; 0=L, 1=R and 1 means toggle."""
    state = 0
    output = []
    for gate in gates:
        if int(gate) == 1:
            state = 1 - state
        output.append(state)
    return output


def sample_scene(seed: int) -> dict:
    """Sample a balanced, unambiguous scene from one integer seed."""
    rng = random.Random(int(seed) ^ 0xC0517A7E)
    gates = [rng.randrange(2) for _ in range(4)]
    target_states = _execute(gates)
    answer = ANSWER_COLORS[int(seed) % len(ANSWER_COLORS)]
    candidate_colors = list(ANSWER_COLORS)
    rng.shuffle(candidate_colors)
    target_row = candidate_colors.index(answer)

    one_flip = rng.randrange(4)
    remaining = [i for i in range(4) if i != one_flip]
    two_flips = rng.sample(remaining, 2)
    distractors = [
        _different_code(target_states, [one_flip]),
        _different_code(target_states, two_flips),
    ]
    rng.shuffle(distractors)
    candidate_states: list[list[int]] = []
    distractor_index = 0
    for row in range(3):
        if row == target_row:
            candidate_states.append(list(target_states))
        else:
            candidate_states.append(distractors[distractor_index])
            distractor_index += 1

    # The whole connected trace remains inside three adjacent 96-pixel grid
    # columns and one 96-pixel row, while its absolute location is randomized.
    x0 = rng.choice((66, 258, 450)) + rng.randint(-3, 3)
    y_center = rng.choice((48, 240, 432, 624))
    ink, background = rng.choice(PALETTES)
    return {
        "gates": gates,
        "candidate_states": candidate_states,
        "candidate_colors": candidate_colors,
        "x0": x0,
        "y_center": y_center,
        "ink_color": list(ink),
        "background": list(background),
    }


def _distances(scene: dict) -> list[int]:
    target = _execute([int(value) for value in scene["gates"]])
    return [
        sum(int(a != b) for a, b in zip(target, code, strict=True))
        for code in scene["candidate_states"]
    ]


def analytic_gold(scene: dict) -> str:
    distances = _distances(scene)
    winner = min(range(3), key=distances.__getitem__)
    return str(scene["candidate_colors"][winner])


def margin(scene: dict) -> float:
    ordered = sorted(_distances(scene))
    return float(ordered[1] - ordered[0])


def is_quarantined(scene: dict) -> bool:
    distances = _distances(scene)
    return distances.count(0) != 1 or sorted(distances)[1] < 1


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _draw_reference_gates(
    draw: ImageDraw.ImageDraw,
    *,
    start_x: int,
    y: int,
    direction: int,
    gates: list[int],
    ink: tuple[int, int, int],
) -> None:
    # All reference teeth point traversal-left. Their width, not side, is the
    # instruction: three pixels = hold, seven pixels = toggle. Keeping their
    # vertical reach equal confines decisive ablation support to the thin rail.
    page_sign = -direction
    for distance, gate in zip(ATTACHMENT_DISTANCES, gates, strict=True):
        x = start_x + direction * distance
        width = 3 if int(gate) == 0 else 7
        draw.line((x, y, x, y + page_sign * 7), fill=ink, width=width)


def _draw_state_spurs(
    draw: ImageDraw.ImageDraw,
    *,
    start_x: int,
    y: int,
    direction: int,
    states: list[int],
    ink: tuple[int, int, int],
) -> None:
    for distance, state in zip(ATTACHMENT_DISTANCES, states, strict=True):
        x = start_x + direction * distance
        page_sign = -direction if int(state) == 0 else direction
        draw.line((x, y, x, y + page_sign * 7), fill=ink, width=3)


def render(scene: dict) -> Image.Image:
    background = tuple(int(v) for v in scene["background"])
    ink = tuple(int(v) for v in scene["ink_color"])
    image = Image.new("RGB", (WIDTH, HEIGHT), background)
    draw = ImageDraw.Draw(image)
    x0 = int(scene["x0"])
    x1 = x0 + RUN_LENGTH
    ys = [int(scene["y_center"]) + offset for offset in ROW_OFFSETS]

    for y in ys:
        draw.line((x0, y, x1, y), fill=ink, width=3)
    draw.line((x1, ys[0], x1, ys[1]), fill=ink, width=3)
    draw.line((x0, ys[1], x0, ys[2]), fill=ink, width=3)
    draw.line((x1, ys[2], x1, ys[3]), fill=ink, width=3)

    starts = [x0, x1, x0, x1]
    directions = [1, -1, 1, -1]
    _draw_reference_gates(
        draw,
        start_x=starts[0],
        y=ys[0],
        direction=directions[0],
        gates=[int(value) for value in scene["gates"]],
        ink=ink,
    )
    for index in range(3):
        _draw_state_spurs(
            draw,
            start_x=starts[index + 1],
            y=ys[index + 1],
            direction=directions[index + 1],
            states=[int(value) for value in scene["candidate_states"][index]],
            ink=ink,
        )

    bead_names = ["gold", *scene["candidate_colors"]]
    for name, x, y in zip(bead_names, starts, ys, strict=True):
        draw.ellipse((x - 5, y - 5, x + 5, y + 5), fill=RGB[name], outline=ink, width=1)
    return image
