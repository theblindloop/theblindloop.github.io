"""Deterministic renderer for row/column-address transposition."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

CANVAS = 512
BACKGROUND = (252, 250, 246)
PALE = (232, 237, 244)
GUIDE = (158, 170, 186)
INK = (24, 38, 64)
TAG_RGB = {
    "gold": (225, 172, 24),
    "red": (218, 65, 74),
    "green": (36, 160, 92),
    "blue": (48, 105, 210),
}
ANSWER_COLORS = ("red", "green", "blue")


def _transpose(bits: list[int]) -> list[int]:
    # Source addresses are 2 rows x 3 columns; target is 3 rows x 2 columns.
    return [bits[r * 3 + c] for c in range(3) for r in range(2)]


def _hamming(left: list[int], right: list[int]) -> int:
    return sum(a != b for a, b in zip(left, right, strict=True))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    reference = [0] * 6
    for index in rng.sample(range(6), 3):
        reference[index] = 1
    target = _transpose(reference)

    variants: set[tuple[int, ...]] = set()
    ones = [i for i, value in enumerate(target) if value]
    zeros = [i for i, value in enumerate(target) if not value]
    pairs = [(one, zero) for one in ones for zero in zeros]
    rng.shuffle(pairs)
    for one, zero in pairs:
        changed = target.copy()
        changed[one], changed[zero] = 0, 1
        variants.add(tuple(changed))
        if len(variants) == 2:
            break
    distractors = [list(item) for item in sorted(variants)]

    correct_slot = rng.randrange(3)
    candidates = distractors.copy()
    candidates.insert(correct_slot, target)

    answer_color = ANSWER_COLORS[int(seed) % 3]
    remaining = [color for color in ANSWER_COLORS if color != answer_color]
    rng.shuffle(remaining)
    tag_order: list[str] = []
    for slot in range(3):
        tag_order.append(answer_color if slot == correct_slot else remaining.pop())

    # Translation by whole 64-pixel cells preserves the intended two-component
    # high-resolution ablation geometry while randomizing absolute position.
    ref_cell_x = 1 + rng.randrange(4)
    center_cell_y = 2 + rng.randrange(3)
    origin = [32 + 64 * ref_cell_x, 32 + 64 * center_cell_y]
    return {
        "reference": reference,
        "candidates": candidates,
        "tag_order": tag_order,
        "origin": origin,
    }


def _draw_card(
    draw: ImageDraw.ImageDraw,
    *,
    center: tuple[int, int],
    rows: int,
    cols: int,
    bits: list[int],
    tag: str,
) -> None:
    spacing = 6
    pad = 5
    width = 2 * pad + spacing * (cols - 1)
    height = 2 * pad + spacing * (rows - 1)
    cx, cy = center
    x0 = cx - width // 2
    y0 = cy - height // 2
    x1 = x0 + width - 1
    y1 = y0 + height - 1
    draw.rounded_rectangle((x0, y0, x1, y1), radius=3, fill=PALE)

    tag_x1 = x0 - 2
    tag_x0 = tag_x1 - 4
    draw.rounded_rectangle((tag_x0, cy - 5, tag_x1, cy + 4), radius=2, fill=TAG_RGB[tag])

    for row in range(rows):
        for col in range(cols):
            px = x0 + pad + col * spacing
            py = y0 + pad + row * spacing
            draw.ellipse((px - 2, py - 2, px + 2, py + 2), outline=GUIDE, width=1)
            if bits[row * cols + col]:
                draw.ellipse((px - 2, py - 2, px + 2, py + 2), fill=INK)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (CANVAS, CANVAS), BACKGROUND)
    draw = ImageDraw.Draw(image)
    origin_x, origin_y = scene["origin"]
    _draw_card(
        draw,
        center=(origin_x, origin_y),
        rows=2,
        cols=3,
        bits=scene["reference"],
        tag="gold",
    )
    candidate_x = origin_x + 128
    for offset, bits, tag in zip(
        (-64, 0, 64), scene["candidates"], scene["tag_order"], strict=True
    ):
        _draw_card(
            draw,
            center=(candidate_x, origin_y + offset),
            rows=3,
            cols=2,
            bits=bits,
            tag=tag,
        )
    return image


def analytic_gold(scene: dict) -> str:
    target = _transpose(scene["reference"])
    distances = [_hamming(target, item) for item in scene["candidates"]]
    best = min(distances)
    winners = [index for index, value in enumerate(distances) if value == best]
    if best != 0 or len(winners) != 1:
        return "abstain"
    return scene["tag_order"][winners[0]]


def margin(scene: dict) -> float:
    target = _transpose(scene["reference"])
    distances = sorted(_hamming(target, item) for item in scene["candidates"])
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    try:
        target = _transpose(scene["reference"])
        candidates = scene["candidates"]
        tags = scene["tag_order"]
        origin = scene["origin"]
        distances = [_hamming(target, item) for item in candidates]
        return not (
            len(candidates) == 3
            and len({tuple(item) for item in candidates}) == 3
            and sorted(tags) == sorted(ANSWER_COLORS)
            and len(origin) == 2
            and distances.count(0) == 1
            and sorted(distances)[1] >= 2
        )
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    del scene
    return []
