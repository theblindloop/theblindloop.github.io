"""Slice-sort reassembly world.

A square symbol sheet has been cut into eight vertical slices and the slices
have been shuffled left to right. Under every slice a key bar is printed; the
bar heights are all different and state the slice's rank in the original sheet
(shortest bar = leftmost slice). Re-sorting the slices by bar height rebuilds an
8x8 dark-cell pattern that matches one of five named abstract symbols.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

ROWS = 8
COLS = 8
SYMBOLS = ("ex", "frame", "tee", "ell", "diamond")
QUARANTINE_MARGIN = 2.0

_BACKGROUNDS = ([255, 255, 255], [250, 249, 245], [248, 250, 252])
_LINES = ([158, 158, 158], [166, 162, 156], [148, 152, 158])
_INKS = ([28, 28, 32], [44, 44, 44], [24, 30, 40], [50, 46, 42])
_KEYS = ([206, 52, 40], [222, 126, 18], [176, 34, 128], [24, 132, 72], [36, 84, 208])
_CELLS = (26, 30, 34)
_GAPS = (8, 12)
_BAR_UNIT = 7
_BAR_BASE = 13
_BAR_W = 14
_LANE_PAD = 10
_MARGIN = 26


def _template(name: str) -> list[list[int]]:
    """Five 8x8 templates, each using exactly sixteen dark cells."""

    grid = [[0] * COLS for _ in range(ROWS)]
    for r in range(ROWS):
        for c in range(COLS):
            if name == "ex":
                on = r == c or r + c == ROWS - 1
            elif name == "frame":
                on = 1 <= r <= 6 and 2 <= c <= 5 and (r in (1, 6) or c in (2, 5))
            elif name == "tee":
                on = r == 0 or (c in (3, 4) and 1 <= r <= 4)
            elif name == "ell":
                on = (c in (2, 3) and r <= 5) or (r == 6 and 2 <= c <= 5)
            else:
                on = abs(2 * r - 7) + abs(2 * c - 7) == 8
            grid[r][c] = 1 if on else 0
    return grid


TEMPLATES = {name: _template(name) for name in SYMBOLS}
INK_COUNT = {name: sum(map(sum, TEMPLATES[name])) for name in SYMBOLS}


def _distance(pattern: list[list[int]], template: list[list[int]]) -> int:
    return sum(
        1 for r in range(ROWS) for c in range(COLS) if pattern[r][c] != template[r][c]
    )


def _scored(pattern: list[list[int]]) -> list[tuple[int, int, str]]:
    """Template distances with a fixed public tie-break by declared symbol order."""

    return sorted(
        (_distance(pattern, TEMPLATES[name]), SYMBOLS.index(name), name)
        for name in SYMBOLS
    )


def _sheet(scene: dict) -> list[list[int]]:
    """The 8x8 pattern the declared re-sorting reveals."""

    grid = [row[:] for row in TEMPLATES[scene["symbol"]]]
    for cell in scene["flips"]:
        r = int(cell[0])
        c = int(cell[1])
        grid[r][c] = 1 - grid[r][c]
    return grid


def _ranks(scene: dict) -> list[int]:
    """Displayed slice -> its rank in the rebuilt sheet (bar height order)."""

    return [int(value) % COLS for value in scene["ranks"]]


def _displayed(scene: dict) -> list[list[int]]:
    """Dark cells as printed: displayed[row][slot] for the shuffled slices."""

    sheet = _sheet(scene)
    ranks = _ranks(scene)
    shown = [[0] * COLS for _ in range(ROWS)]
    for slot in range(COLS):
        source = ranks[slot]
        for r in range(ROWS):
            shown[r][slot] = sheet[r][source]
    return shown


def _naive_readable(scene: dict) -> int:
    """Smallest template mismatch reachable WITHOUT the declared re-sorting."""

    shown = _displayed(scene)
    return min(_distance(shown, TEMPLATES[name]) for name in SYMBOLS)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    for _ in range(600):
        symbol = SYMBOLS[rng.randrange(len(SYMBOLS))]
        budget = rng.choice([0, 0, 1, 1, 2, 2, 3, 3, 4, 5, 6, 7])
        cells = [[r, c] for r in range(ROWS) for c in range(COLS)]
        if rng.random() < 0.5:
            rival = SYMBOLS[rng.randrange(len(SYMBOLS))]
            if rival == symbol:
                rival = SYMBOLS[(SYMBOLS.index(symbol) + 1) % len(SYMBOLS)]
            pool = [
                cell
                for cell in cells
                if TEMPLATES[symbol][cell[0]][cell[1]]
                != TEMPLATES[rival][cell[0]][cell[1]]
            ]
        else:
            pool = list(cells)
        rng.shuffle(pool)
        flips = sorted(pool[:budget])
        ranks = list(range(COLS))
        rng.shuffle(ranks)
        if sum(1 for slot, rank in enumerate(ranks) if slot == rank) > 2:
            continue
        scene = {
            "symbol": symbol,
            "flips": flips,
            "ranks": ranks,
            "cell": _CELLS[rng.randrange(len(_CELLS))],
            "gap": _GAPS[rng.randrange(len(_GAPS))],
            "jitter_x": rng.randint(-6, 6),
            "jitter_y": rng.randint(-5, 5),
            "bg": list(_BACKGROUNDS[rng.randrange(len(_BACKGROUNDS))]),
            "line": list(_LINES[rng.randrange(len(_LINES))]),
            "ink": list(_INKS[rng.randrange(len(_INKS))]),
            "key": list(_KEYS[rng.randrange(len(_KEYS))]),
        }
        if _naive_readable(scene) < 8:
            continue
        return scene
    raise RuntimeError("scene sampling failed to satisfy the declared constraints")


def _canvas(scene: dict) -> tuple[int, int, int, int, int, int]:
    cell = int(scene["cell"])
    gap = int(scene["gap"])
    lane = _BAR_BASE + (COLS - 1) * _BAR_UNIT + 6
    width = 2 * _MARGIN + COLS * cell + (COLS - 1) * gap
    height = 2 * _MARGIN + ROWS * cell + _LANE_PAD + lane
    ox = _MARGIN + int(scene["jitter_x"])
    oy = _MARGIN + int(scene["jitter_y"])
    return width, height, ox, oy, cell, gap


def render(scene: dict) -> Image.Image:
    shown = _displayed(scene)
    ranks = _ranks(scene)
    width, height, ox, oy, cell, gap = _canvas(scene)
    image = Image.new("RGB", (width, height), tuple(scene["bg"]))
    draw = ImageDraw.Draw(image)
    ink_rgb = tuple(scene["ink"])
    line_rgb = tuple(scene["line"])
    key_rgb = tuple(scene["key"])
    lane_top = oy + ROWS * cell + _LANE_PAD
    lane_bottom = lane_top + _BAR_BASE + (COLS - 1) * _BAR_UNIT
    for slot in range(COLS):
        x0 = ox + slot * (cell + gap)
        for r in range(ROWS):
            y0 = oy + r * cell
            if shown[r][slot]:
                draw.rectangle(
                    [x0 + 3, y0 + 3, x0 + cell - 3, y0 + cell - 3], fill=ink_rgb
                )
        draw.rectangle([x0, oy, x0 + cell, oy + ROWS * cell], outline=line_rgb, width=1)
        for r in range(1, ROWS):
            y = oy + r * cell
            draw.line([x0, y, x0 + cell, y], fill=line_rgb, width=1)
        bar_h = _BAR_BASE + ranks[slot] * _BAR_UNIT
        bx = x0 + (cell - _BAR_W) // 2
        draw.rectangle(
            [bx, lane_bottom - bar_h, bx + _BAR_W, lane_bottom], fill=key_rgb
        )
    return image


def analytic_gold(scene: dict) -> str:
    return _scored(_sheet(scene))[0][2]


def margin(scene: dict) -> float:
    ranking = _scored(_sheet(scene))
    return float(ranking[1][0] - ranking[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical relabelings that must leave the analytic gold unchanged."""

    modular = dict(scene)
    modular["ranks"] = [int(value) + COLS for value in scene["ranks"]]
    reordered = dict(scene)
    reordered["flips"] = list(reversed(list(scene["flips"])))
    return [("rank_modulus", modular), ("flip_listing_order", reordered)]
