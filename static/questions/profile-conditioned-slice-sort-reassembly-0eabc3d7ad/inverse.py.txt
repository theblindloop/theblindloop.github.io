"""Pixel-only inverse arm for the slice-sort reassembly world.

This module receives a rendered image and nothing else. It locates the printed
slices and key bars, executes the declared re-sorting on the raster evidence,
reads the rebuilt dark-cell pattern, and names the closest symbol. It imports
only PIL and shares no code, constants file, or scene data with the renderer.
"""

from __future__ import annotations

from PIL import Image

ABSTAIN = "abstain"
_SYMBOLS = ("ex", "frame", "tee", "ell", "diamond")
_N = 8
_MIN_BAR_GAP = 40


def _templates() -> dict[str, list[list[int]]]:
    """Independently written copies of the five public 8x8 symbol templates."""

    out: dict[str, list[list[int]]] = {}
    for name in _SYMBOLS:
        grid = [[0] * _N for _ in range(_N)]
        for r in range(_N):
            for c in range(_N):
                if name == "ex":
                    on = (r == c) or (r + c == _N - 1)
                elif name == "frame":
                    on = 1 <= r <= 6 and 2 <= c <= 5 and (r in (1, 6) or c in (2, 5))
                elif name == "tee":
                    on = r == 0 or (c in (3, 4) and 1 <= r <= 4)
                elif name == "ell":
                    on = (c in (2, 3) and r <= 5) or (r == 6 and 2 <= c <= 5)
                else:
                    on = abs(2 * r - 7) + abs(2 * c - 7) == 8
                grid[r][c] = 1 if on else 0
        out[name] = grid
    return out


def _runs(flags: list[bool]) -> list[tuple[int, int]]:
    runs: list[tuple[int, int]] = []
    start = None
    for index, flag in enumerate(flags):
        if flag and start is None:
            start = index
        elif not flag and start is not None:
            runs.append((start, index - 1))
            start = None
    if start is not None:
        runs.append((start, len(flags) - 1))
    return runs


def decision_from_image(image: Image.Image) -> str:
    return _decide(image)[0]


def decision_with_margin(image: Image.Image) -> tuple[str, float]:
    """Reported decision plus the runner-up gap in mismatching cells."""

    return _decide(image)


def _decide(image: Image.Image) -> tuple[str, float]:
    rgb = image.convert("RGB")
    width, height = rgb.size
    if width < 60 or height < 60:
        return ABSTAIN, 0.0
    pixels = rgb.load()
    key = [[False] * width for _ in range(height)]
    ink = [[False] * width for _ in range(height)]
    line = [[False] * width for _ in range(height)]
    for y in range(height):
        key_row = key[y]
        ink_row = ink[y]
        line_row = line[y]
        for x in range(width):
            r, g, b = pixels[x, y]
            if max(r, g, b) - min(r, g, b) > 55:
                key_row[x] = True
                continue
            lum = 0.299 * r + 0.587 * g + 0.114 * b
            if lum < 110.0:
                ink_row[x] = True
            elif lum < 215.0:
                line_row[x] = True

    marked_cols = [
        any(key[y][x] or ink[y][x] or line[y][x] for y in range(height))
        for x in range(width)
    ]
    slices = _runs(marked_cols)
    if len(slices) != _N:
        return ABSTAIN, 0.0

    line_rows = [any(line[y]) for y in range(height)]
    lattice = _runs(line_rows)
    if not lattice:
        return ABSTAIN, 0.0
    top, bottom = max(lattice, key=lambda run: run[1] - run[0])
    span = bottom - top
    if span < 8 * _N:
        return ABSTAIN, 0.0
    step = span / _N

    shown = [[0] * _N for _ in range(_N)]
    bar_mass: list[int] = []
    for slot, (xa, xb) in enumerate(slices):
        if xb - xa < 12:
            return ABSTAIN, 0.0
        x0 = xa + 5
        x1 = xb - 5
        mass = 0
        for y in range(height):
            row = key[y]
            mass += sum(1 for x in range(xa, xb + 1) if row[x])
        bar_mass.append(mass)
        for r in range(_N):
            ya = int(round(top + r * step)) + 5
            yb = int(round(top + (r + 1) * step)) - 5
            if yb <= ya or x1 <= x0:
                return ABSTAIN, 0.0
            hits = 0
            total = 0
            for y in range(ya, yb + 1):
                row = ink[y]
                for x in range(x0, x1 + 1):
                    total += 1
                    if row[x]:
                        hits += 1
            shown[r][slot] = 1 if total and hits / total > 0.5 else 0

    if min(bar_mass) <= 0:
        return ABSTAIN, 0.0
    ordered = sorted(bar_mass)
    if any(ordered[i + 1] - ordered[i] < _MIN_BAR_GAP for i in range(_N - 1)):
        return ABSTAIN, 0.0
    rank = {slot: ordered.index(mass) for slot, mass in enumerate(bar_mass)}
    if len(set(rank.values())) != _N:
        return ABSTAIN, 0.0

    sheet = [[0] * _N for _ in range(_N)]
    for slot in range(_N):
        for r in range(_N):
            sheet[r][rank[slot]] = shown[r][slot]

    templates = _templates()
    scores = sorted(
        (
            sum(
                1
                for r in range(_N)
                for c in range(_N)
                if sheet[r][c] != templates[name][r][c]
            ),
            _SYMBOLS.index(name),
            name,
        )
        for name in _SYMBOLS
    )
    return scores[0][2], float(scores[1][0] - scores[0][0])
