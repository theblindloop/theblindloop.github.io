"""Independent pixel-only inverse arm for the negative-space cavity world.

Reconstructs the foreground complement -- the enclosed empty (white) region
that is fully surrounded by colored tiles and does not touch the image border --
then classifies its silhouette into one of {circle, square, diamond, triangle}
from shape geometry (bounding-box fill ratio and convex-hull corner count).

This module imports no renderer / prompts / generate / verify / analytic-gold
code and never receives the latent scene.
"""

from __future__ import annotations

from PIL import Image
import numpy as np

_SQUARE_MIN = 0.90
_CIRCLE_MIN = 0.68
_OFFSET_MIN = 0.08


def _cavity_mask(arr: np.ndarray) -> np.ndarray:
    """White cavity = white pixels that do not touch the image border."""
    white = (arr[..., 0] > 245) & (arr[..., 1] > 245) & (arr[..., 2] > 245)
    h, w = white.shape
    border = white.copy()
    border[0, :] = False
    border[-1, :] = False
    border[:, 0] = False
    border[:, -1] = False
    return border


def decision_from_image(image: Image.Image) -> str:
    """Return the cavity shape recovered from the final raster alone."""
    arr = np.asarray(image.convert("RGB")).astype(int)
    mask = _cavity_mask(arr)
    ys, xs = np.where(mask)
    if xs.size == 0:
        return "circle"
    h = int(ys.max()) - int(ys.min()) + 1
    w = int(xs.max()) - int(xs.min()) + 1
    fill = float(mask.sum()) / float(h * w)
    if fill >= _SQUARE_MIN:
        return "square"
    if fill >= _CIRCLE_MIN:
        return "circle"
    centre_y = (int(ys.min()) + int(ys.max())) / 2.0
    offset = abs(float(ys.mean()) - centre_y) / h
    return "triangle" if offset > _OFFSET_MIN else "diamond"
