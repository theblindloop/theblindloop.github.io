"""Negative-space cavity world: the shape formed by the empty region.

A mosaic field of colored square tiles covers the canvas except for one central
empty (white) cavity. The cavity is pure negative space -- the silhouette seen
through the gap in the surrounding tiles. The decision is: which of four shapes
(circle, square, diamond, triangle) does that empty region form?

The renderer owns scene sampling + rasterization + the structural analytic
gold. The independent pixel-only inverse arm (``oracle.py``) reconstructs the
foreground complement (the enclosed empty region) from the final PNG alone and
classifies its silhouette.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
CENTER = WIDTH // 2
RADIUS = 125
TILE = 18
BACKGROUND = (255, 255, 255)
PALETTE = [
    (215, 40, 40),
    (40, 110, 220),
    (40, 170, 80),
    (230, 150, 20),
    (150, 60, 180),
    (40, 190, 190),
]
SHAPES = ["circle", "square", "diamond", "triangle"]

# Classification thresholds: fill ratio of the cavity within its bounding box.
_SQUARE_R = 1.0        # axis-aligned filled square
_CIRCLE_R = np.pi / 4  # filled circle
_HALF_R = 0.5          # both diamond and triangle fill half their bounding box
_R_SQUARE_MIN = 0.90
_R_CIRCLE_MIN = 0.68
# Triangle centroid sits about 1/6 of the bounding-box height off-centre; a
# diamond's centroid sits at the bounding-box centre. Discretization-stable.
_OFFSET_TRIANGLE = 1.0 / 6.0
_OFFSET_DIAMOND = 0.0
_OFFSET_MIN = 0.08
_MARGIN_QUARANTINE = 0.05


def _color(cidx: int) -> tuple[int, int, int]:
    return PALETTE[cidx % len(PALETTE)]


def sample_scene(seed: int) -> dict:
    """Sample one latent scene deterministically from ``seed``."""
    rng = np.random.default_rng(seed)
    shape = SHAPES[int(rng.integers(0, len(SHAPES)))]
    tiles: list[list[int]] = []
    for ty in range(0, HEIGHT, TILE):
        for tx in range(0, WIDTH, TILE):
            tiles.append([tx, ty, int(rng.integers(0, len(PALETTE)))])
    return {"cavity_shape": shape, "tiles": tiles}


def _carve(draw: ImageDraw.ImageDraw, shape: str) -> None:
    """Draw the empty (white) cavity silhouette over the tile field."""
    c = CENTER
    r = RADIUS
    if shape == "square":
        draw.rectangle([c - r, c - r, c + r, c + r], fill=BACKGROUND)
    elif shape == "circle":
        draw.ellipse([c - r, c - r, c + r, c + r], fill=BACKGROUND)
    elif shape == "diamond":
        draw.polygon(
            [(c, c - r), (c + r, c), (c, c + r), (c - r, c)], fill=BACKGROUND
        )
    elif shape == "triangle":
        draw.polygon(
            [
                (c, c - r),
                (c - int(0.9 * r), c + int(0.7 * r)),
                (c + int(0.9 * r), c + int(0.7 * r)),
            ],
            fill=BACKGROUND,
        )
    else:
        raise ValueError(f"unknown cavity shape {shape!r}")


def _draw_scene(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    d = ImageDraw.Draw(img)
    for (tx, ty, cidx) in scene["tiles"]:
        d.rectangle([tx, ty, tx + TILE, ty + TILE], fill=_color(cidx))
    _carve(d, scene["cavity_shape"])
    return img


def render(scene: dict) -> Image.Image:
    """Rasterize the latent scene (a deterministic PNGA)."""
    return _draw_scene(scene)


def analytic_gold(scene: dict) -> str:
    """Structural gold: the cavity shape is the declared latent shape."""
    return scene["cavity_shape"]


def _cavity_mask(scene: dict) -> np.ndarray:
    arr = np.asarray(_draw_scene(scene).convert("RGB")).astype(int)
    return (arr[..., 0] > 245) & (arr[..., 1] > 245) & (arr[..., 2] > 245)


def _cavity_features(mask: np.ndarray) -> tuple[float, float]:
    """Return (fill-ratio within bounding box, vertical centroid offset frac)."""
    ys, xs = np.where(mask)
    if xs.size == 0:
        return 0.0, 0.0
    h = int(ys.max()) - int(ys.min()) + 1
    w = int(xs.max()) - int(xs.min()) + 1
    fill = float(mask.sum()) / float(h * w)
    centre_y = (int(ys.min()) + int(ys.max())) / 2.0
    offset = abs(float(ys.mean()) - centre_y) / h
    return fill, offset


def classify_shape(fill: float, offset: float) -> str:
    if fill >= _R_SQUARE_MIN:
        return "square"
    if fill >= _R_CIRCLE_MIN:
        return "circle"
    return "triangle" if offset > _OFFSET_MIN else "diamond"


def _shape_scores(fill: float, offset: float) -> dict[str, float]:
    return {
        "square": 1.0 / (1.0 + 3.0 * abs(fill - _SQUARE_R)),
        "circle": 1.0 / (1.0 + 3.0 * abs(fill - _CIRCLE_R)),
        "diamond": 1.0 / (1.0 + 2.0 * abs(fill - _HALF_R) + 6.0 * abs(offset - _OFFSET_DIAMOND)),
        "triangle": 1.0 / (1.0 + 2.0 * abs(fill - _HALF_R) + 6.0 * abs(offset - _OFFSET_TRIANGLE)),
    }


def margin(scene: dict) -> float:
    """Commitment: best-minus-second cavity-shape fit score."""
    fill, offset = _cavity_features(_cavity_mask(scene))
    scores = sorted(_shape_scores(fill, offset).values(), reverse=True)
    return float(scores[0] - scores[1] if len(scores) > 1 else scores[0])


def is_quarantined(scene: dict) -> bool:
    """Quarantine near-boundary cavities whose shape fit is ambiguous."""
    return margin(scene) < _MARGIN_QUARANTINE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The scene is injectively rendered; no latent aliases are declared."""
    return []
