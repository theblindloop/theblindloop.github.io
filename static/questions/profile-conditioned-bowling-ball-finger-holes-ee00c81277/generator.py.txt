"""Deterministic bowling-ball scene sampler and Pillow renderer."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

SCALE = 4
PALETTE = [
    [42, 118, 214],
    [178, 66, 184],
    [24, 156, 126],
    [225, 112, 42],
    [206, 52, 82],
    [92, 92, 210],
]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    hole_count = 2 + (int(seed) % 4)
    canvas_size = [384, 384]
    ball_radius = rng.randint(126, 142)
    ball_center = [192 + rng.randint(-18, 18), 194 + rng.randint(-15, 15)]
    ball_color = PALETTE[rng.randrange(len(PALETTE))]
    hole_color = [18 + rng.randint(0, 8), 22 + rng.randint(0, 8), 30 + rng.randint(0, 8)]

    # Radius scales inversely with sqrt(count), approximately balancing total
    # dark-hole ink across all four answer identities.
    hole_radius = math.sqrt(5200.0 / (math.pi * hole_count))
    layout_radius = 54.0 if hole_count <= 3 else 61.0
    rotation = rng.uniform(0.0, 2.0 * math.pi)
    hole_specs = []
    for index in range(hole_count):
        angle = rotation + (2.0 * math.pi * index / hole_count)
        radial = layout_radius + rng.uniform(-3.0, 3.0)
        x = ball_center[0] + radial * math.cos(angle) + rng.uniform(-2.0, 2.0)
        y = ball_center[1] + radial * math.sin(angle) + rng.uniform(-2.0, 2.0)
        radius = hole_radius * rng.uniform(0.94, 1.06)
        hole_specs.append([round(x, 3), round(y, 3), round(radius, 3)])

    glint_angle = rotation + 0.71
    glint_offset = [
        round(0.48 * ball_radius * math.cos(glint_angle), 3),
        round(0.48 * ball_radius * math.sin(glint_angle), 3),
    ]
    return {
        "hole_count": hole_count,
        "canvas_size": canvas_size,
        "ball_center": ball_center,
        "ball_radius": ball_radius,
        "ball_color": ball_color,
        "hole_color": hole_color,
        "hole_specs": hole_specs,
        "glint_offset": glint_offset,
    }


def render(scene: dict) -> Image.Image:
    width, height = [int(v) for v in scene["canvas_size"]]
    cx, cy = [float(v) for v in scene["ball_center"]]
    ball_radius = float(scene["ball_radius"])
    ball_color = tuple(int(v) for v in scene["ball_color"])
    hole_color = tuple(int(v) for v in scene["hole_color"])
    hole_specs = scene["hole_specs"]
    glint_offset = [float(v) for v in scene["glint_offset"]]
    hole_count = int(scene["hole_count"])
    if hole_count != len(hole_specs):
        raise ValueError("hole_count must match the visible hole specification")

    image = Image.new("RGB", (width * SCALE, height * SCALE), (248, 247, 243))
    draw = ImageDraw.Draw(image)

    def box(x0: float, y0: float, x1: float, y1: float) -> tuple[int, int, int, int]:
        return tuple(round(value * SCALE) for value in (x0, y0, x1, y1))

    # A soft offset shadow supplies objecthood without adding dark components:
    # its luminance remains far above the oracle's hole threshold.
    draw.ellipse(
        box(cx - ball_radius + 7, cy - ball_radius + 11, cx + ball_radius + 7, cy + ball_radius + 11),
        fill=(208, 205, 198),
    )
    draw.ellipse(
        box(cx - ball_radius, cy - ball_radius, cx + ball_radius, cy + ball_radius),
        fill=ball_color,
    )
    for hx, hy, radius in hole_specs:
        draw.ellipse(box(hx - radius, hy - radius, hx + radius, hy + radius), fill=hole_color)

    gx = cx + glint_offset[0]
    gy = cy + glint_offset[1]
    gr = max(7.0, ball_radius * 0.065)
    draw.ellipse(box(gx - gr, gy - gr, gx + gr, gy + gr), fill=(238, 240, 244))
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(int(scene["hole_count"]))


def margin(scene: dict) -> float:
    cx, cy = [float(v) for v in scene["ball_center"]]
    ball_radius = float(scene["ball_radius"])
    specs = [[float(v) for v in spec] for spec in scene["hole_specs"]]
    clearances = []
    for hx, hy, radius in specs:
        clearances.append(ball_radius - math.hypot(hx - cx, hy - cy) - radius)
        clearances.append(radius)
    for i, (ax, ay, ar) in enumerate(specs):
        for bx, by, br in specs[i + 1 :]:
            clearances.append(math.hypot(ax - bx, ay - by) - ar - br)
    return round(min(clearances), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
