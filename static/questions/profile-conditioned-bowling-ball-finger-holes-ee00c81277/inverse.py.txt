"""Independent PNG-only finger-hole counter.

The oracle knows only the public visual rule: counted holes are separate very
dark circular regions. It receives no latent data and imports no world module.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y, x in zip(*np.nonzero(mask), strict=True):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        area = 0
        min_x = max_x = int(x)
        min_y = max_y = int(y)
        while queue:
            py, px = queue.popleft()
            area += 1
            min_x, max_x = min(min_x, px), max(max_x, px)
            min_y, max_y = min(min_y, py), max(max_y, py)
            for ny, nx in ((py - 1, px), (py + 1, px), (py, px - 1), (py, px + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        found.append((area, min_x, min_y, max_x, max_y))
    return found


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    # All valid palettes keep the ball and shadow well above this threshold;
    # only the deliberately near-black depressions remain.
    dark = np.max(arr, axis=2) < 72
    valid = []
    for area, min_x, min_y, max_x, max_y in _components(dark):
        width = max_x - min_x + 1
        height = max_y - min_y + 1
        if area < 650 or width < 24 or height < 24:
            continue
        aspect = width / height
        fill = area / (width * height)
        if 0.80 <= aspect <= 1.25 and 0.66 <= fill <= 0.86:
            valid.append((area, min_x, min_y, max_x, max_y))
    if not 2 <= len(valid) <= 5:
        raise ValueError(f"ambiguous finger-hole count: {len(valid)}")
    return str(len(valid))
