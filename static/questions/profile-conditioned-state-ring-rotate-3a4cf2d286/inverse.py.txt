"""Independent pixel-only inverse arm for the cyclic rotation register.

Receives only the final PNG. From the raster alone it:
  1. locates the four start bit-disks in the left panel by dark-colour
     segmentation and reads whether each is FILLED or HOLLOW from its centre;
  2. finds the orange ring and maps it to the ringed (target) cell index;
  3. reads the ordered operator glyphs from the operator boxes by colour and
     left-to-right position;
  4. replays the rotations on the four-bit start word and reports the final
     fill state of the ringed cell's bit.

No latent scene, renderer, prompt, or gold code is accessed.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

DARK = (80, 80, 80)
BG = (255, 255, 255)
RING_COLOR = (255, 150, 30)
OP_R_COLOR = (40, 180, 60)
OP_L_COLOR = (220, 60, 40)
NCELL = 4
NOPS = 4
WINDOW = 4


class UnreadableImageError(ValueError):
    """The raster does not expose the expected register-and-operators structure."""


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    visited = np.zeros_like(mask, dtype=bool)
    ys, xs = np.nonzero(mask)
    comps = []
    for y, x in zip(ys.tolist(), xs.tolist()):
        if visited[y, x]:
            continue
        points = []
        queue = deque([(y, x)])
        visited[y, x] = True
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dy == 0 and dx == 0:
                        continue
                    ny, nx = cy + dy, cx + dx
                    if (
                        0 <= ny < mask.shape[0]
                        and 0 <= nx < mask.shape[1]
                        and mask[ny, nx]
                        and not visited[ny, nx]
                    ):
                        visited[ny, nx] = True
                        queue.append((ny, nx))
        comps.append(points)
    return comps


def _mask_equal(arr: np.ndarray, color: tuple[int, int, int]) -> np.ndarray:
    r, g, b = color
    return (arr[:, :, 0] == r) & (arr[:, :, 1] == g) & (arr[:, :, 2] == b)


def _centroid(points) -> tuple[int, int]:
    ys = [p[0] for p in points]
    xs = [p[1] for p in points]
    return int(round(np.mean(xs))), int(round(np.mean(ys)))


def _fill_state(arr: np.ndarray, center: tuple[int, int]) -> int:
    cx, cy = center
    y0 = max(0, cy - WINDOW)
    y1 = min(arr.shape[0], cy + WINDOW + 1)
    x0 = max(0, cx - WINDOW)
    x1 = min(arr.shape[1], cx + WINDOW + 1)
    win = arr[y0:y1, x0:x1]
    dark_count = int(_mask_equal(win, DARK).sum())
    bg_count = int(_mask_equal(win, BG).sum())
    if dark_count > bg_count:
        return 1
    if bg_count > dark_count:
        return 0
    raise UnreadableImageError("fill state is ambiguous at disk centre")


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))

    disk_comps = _components(_mask_equal(arr, DARK))
    big = [c for c in disk_comps if len(c) > 40]
    if len(big) != NCELL:
        raise UnreadableImageError(
            f"expected {NCELL} start bit-disks; found {len(big)} dark components"
        )
    centers = sorted(_centroid(c) for c in big)
    centers.sort(key=lambda c: (c[0], c[1]))

    ring_comps = _components(_mask_equal(arr, RING_COLOR))
    rings = [c for c in ring_comps if len(c) > 40]
    if len(rings) != 1:
        raise UnreadableImageError(f"expected one target ring; found {len(rings)}")
    rx, ry = _centroid(rings[0])
    target = min(range(NCELL), key=lambda i: (centers[i][0] - rx) ** 2 + (centers[i][1] - ry) ** 2)

    words = [_fill_state(arr, centers[i]) for i in range(NCELL)]

    op_hits = []
    for comp in _components(_mask_equal(arr, OP_R_COLOR)):
        if len(comp) > 20:
            op_hits.append(("R", _centroid(comp)[0]))
    for comp in _components(_mask_equal(arr, OP_L_COLOR)):
        if len(comp) > 20:
            op_hits.append(("L", _centroid(comp)[0]))
    op_hits.sort(key=lambda item: item[1])
    if len(op_hits) != NOPS:
        raise UnreadableImageError(f"expected {NOPS} operators; found {len(op_hits)}")
    ops = [op for op, _ in op_hits]

    word = list(words)
    for op in ops:
        if op == "R":
            word = [word[-1]] + word[:-1]
        else:
            word = word[1:] + [word[0]]
    return "filled" if word[target] else "outline"
