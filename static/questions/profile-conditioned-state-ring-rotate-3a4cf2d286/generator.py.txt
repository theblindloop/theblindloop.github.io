"""Cyclic binary-word rotation register (state tracking).

A single horizontal register of four bit cells (slot 0 to slot 3, left to
right) is shown once at its START state on the left. Each cell is a disk that
is either FILLED (solid, bit 1) or HOLLOW (outline ring, bit 0). An orange ring
surrounds exactly one cell and marks the TARGET cell whose state must be
reported. To the right, an ordered sequence of OPERATOR glyphs is read left to
right: a green right-pointing triangle means ROTATE the whole four-bit word one
step to the right (the rightmost bit wraps around to the leftmost), and a red
left-pointing triangle means ROTATE it one step to the left (the leftmost bit
wraps around to the rightmost). The register is never drawn again, so the final
state of the ringed cell is recovered only by replaying the rotations from the
marked start.

Latent scene: ``word`` (list of four bit values for the start state),
``target`` (index 0-3 of the ringed cell), ``ops`` (list of rotation steps,
each 'R' for a right rotation or 'L' for a left rotation). The answer is the
fill state (filled/outline) of the bit that ends at the ringed cell's index
after applying every rotation in order.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 900
HEIGHT = 220

DISK_R = 18
RING_GAP = 8
RING_W = 5
LIGHT_W = 5

BG = (255, 255, 255)
DARK = (80, 80, 80)
RING_COLOR = (255, 150, 30)
OP_R_COLOR = (40, 180, 60)
OP_L_COLOR = (220, 60, 40)
BORDER = (210, 210, 210)

NCELL = 4
NOPS = 4

PANEL_LEFT = 40
PANEL_TOP = 70
CELL_DX = 82
CELL_Y = 128

OP_LEFT = 460
OP_TOP = 90
OP_W = 66
OP_H = 66
OP_DX = 74


def _cell_x(i: int) -> int:
    return PANEL_LEFT + 46 + i * CELL_DX


def _op_box_left(i: int) -> int:
    return OP_LEFT + i * OP_DX


def _draw_disk(draw: ImageDraw.ImageDraw, cx: int, cy: int, bit: int) -> None:
    box = [cx - DISK_R, cy - DISK_R, cx + DISK_R, cy + DISK_R]
    if bit:
        draw.ellipse(box, fill=DARK)
    else:
        draw.ellipse(box, fill=BG, outline=DARK, width=LIGHT_W)


def _draw_ring(draw: ImageDraw.ImageDraw, cx: int, cy: int) -> None:
    r = DISK_R + RING_GAP
    draw.ellipse([cx - r, cy - r, cx + r, cy + r], outline=RING_COLOR, width=RING_W)


def _triangle(draw: ImageDraw.ImageDraw, bx: int, by: int, direction: str, color) -> None:
    cx = bx + OP_W / 2
    cy = by + OP_H / 2
    pad = 12
    half = 14
    if direction == "R":
        pts = [(cx - pad, cy - half), (cx - pad, cy + half), (cx + pad, cy)]
    else:
        pts = [(cx + pad, cy - half), (cx + pad, cy + half), (cx - pad, cy)]
    draw.polygon(pts, fill=color)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    word = [rng.randint(0, 1) for _ in range(NCELL)]
    target = rng.randrange(NCELL)
    ops = [rng.choice(("R", "L")) for _ in range(NOPS)]
    return {"word": word, "target": target, "ops": ops}


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)
    word = scene["word"]
    target = scene["target"]
    ops = scene["ops"]

    draw.rectangle(
        [PANEL_LEFT, PANEL_TOP - 40, PANEL_LEFT + 4 * CELL_DX + 40, PANEL_TOP + 70],
        outline=BORDER,
        width=2,
    )
    for i in range(NCELL):
        cx = _cell_x(i)
        _draw_disk(draw, cx, CELL_Y, int(word[i]))
    _draw_ring(draw, _cell_x(target), CELL_Y)

    for j, op in enumerate(ops):
        bx = _op_box_left(j)
        draw.rectangle([bx, OP_TOP, bx + OP_W, OP_TOP + OP_H], outline=BORDER, width=2)
        color = OP_R_COLOR if op == "R" else OP_L_COLOR
        _triangle(draw, bx, OP_TOP, op, color)
    return img


def analytic_gold(scene: dict) -> str:
    word = list(scene["word"])
    target = int(scene["target"])
    for op in scene["ops"]:
        if op == "R":
            word = [word[-1]] + word[:-1]
        else:
            word = word[1:] + [word[0]]
    return "filled" if word[target] else "outline"


def margin(scene: dict) -> float:
    """Robust discrete margin: the number of dark pixels in one filled disk,
    i.e. the strength of the single bit whose state decides the answer."""
    cx, cy = _cell_x(0), CELL_Y
    r = DISK_R
    return float(sum(
        1 for dy in range(-r, r + 1) for dx in range(-r, r + 1)
        if dx * dx + dy * dy <= r * r
    ))


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    """No nontrivial render-invariant latent symmetry: the drawn start word,
    the ringed target cell, and every operator glyph directly fix the pixels,
    and analytic gold reads exactly those same fields. Any reparametrization
    that changed the bits, the ringed index, or the operator order would change
    the raster."""
    return []
