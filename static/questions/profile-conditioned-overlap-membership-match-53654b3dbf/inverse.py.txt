"""Independent pixel-only inverse arm."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

DOT_RGB = {0: (235, 111, 36), 1: (16, 151, 136), 2: (105, 166, 43), 3: (211, 62, 126)}
CIRCLE_RGB = (35, 126, 196)
RECT_RGB = (137, 78, 180)
BORDER_RGB = {
    "reference": (100, 108, 118), "red": (205, 52, 55),
    "blue": (47, 87, 184), "gold": (216, 157, 28),
}


class OracleError(ValueError):
    pass


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        xs, ys = [], []
        while queue:
            cy, cx = queue.popleft()
            xs.append(cx)
            ys.append(cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        found.append((min(xs), min(ys), max(xs), max(ys), len(xs)))
    return found


def _mask(arr: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    return np.all(arr == np.asarray(rgb, dtype=np.uint8), axis=2)


def _border_box(arr: np.ndarray, rgb: tuple[int, int, int]) -> tuple[int, int, int, int]:
    candidates = [component for component in _components(_mask(arr, rgb))
                  if component[2] - component[0] > 250 and component[3] - component[1] > 220]
    if len(candidates) != 1:
        raise OracleError("missing or ambiguous panel border")
    return candidates[0][:4]


def _centroid(arr: np.ndarray, rgb: tuple[int, int, int], box: tuple[int, int, int, int]) -> tuple[float, float]:
    x0, y0, x1, y1 = box
    crop = arr[y0:y1 + 1, x0:x1 + 1]
    ys, xs = np.nonzero(_mask(crop, rgb))
    if len(xs) < 250:
        raise OracleError("colored dot missing")
    return float(xs.mean() + x0), float(ys.mean() + y0)


def _geometry(arr: np.ndarray, box: tuple[int, int, int, int]) -> tuple[tuple[float, float, float], tuple[float, float, float, float]]:
    x0, y0, x1, y1 = box
    crop = arr[y0:y1 + 1, x0:x1 + 1]
    circle_parts = [part for part in _components(_mask(crop, CIRCLE_RGB)) if part[4] > 300]
    rect_parts = [part for part in _components(_mask(crop, RECT_RGB)) if part[4] > 500]
    if not circle_parts or len(rect_parts) != 1:
        raise OracleError("region outlines missing")
    # The rectangle can split the circle at two crossings; pool every substantial blue arc.
    circle_pixels = _mask(crop, CIRCLE_RGB)
    ys, xs = np.nonzero(circle_pixels)
    if len(xs) < 500:
        raise OracleError("circle outline missing")
    left, right, top, bottom = float(xs.min() + x0), float(xs.max() + x0), float(ys.min() + y0), float(ys.max() + y0)
    cx, cy = (left + right) / 2.0, (top + bottom) / 2.0
    radius = ((right - left) + (bottom - top)) / 4.0
    rx0, ry0, rx1, ry1, _ = rect_parts[0]
    return (cx, cy, radius), (rx0 + x0, ry0 + y0, rx1 + x0, ry1 + y0)


def _signature(arr: np.ndarray, box: tuple[int, int, int, int]) -> tuple[tuple[bool, bool], ...]:
    circle, rect = _geometry(arr, box)
    cx, cy, radius = circle
    x0, y0, x1, y1 = rect
    result = []
    for color_id in range(4):
        x, y = _centroid(arr, DOT_RGB[color_id], box)
        result.append((math.hypot(x - cx, y - cy) < radius, x0 < x < x1 and y0 < y < y1))
    return tuple(result)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    boxes = {label: _border_box(arr, rgb) for label, rgb in BORDER_RGB.items()}
    reference = _signature(arr, boxes["reference"])
    matches = [label for label in ("red", "blue", "gold") if _signature(arr, boxes[label]) == reference]
    if len(matches) != 1:
        raise OracleError("raster does not contain one unique matching membership map")
    return matches[0]
