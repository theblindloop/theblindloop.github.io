"""Renderer and analytic contract for the complementary stencil candidate.

Two same-sized grayscale plates contain a shared dark glyph. At every other
pixel, one plate is dark and the other is light according to a visible raster
carrier. The question's pixelwise maximum therefore removes the carrier while
leaving the common glyph.
"""

from __future__ import annotations

import random
from typing import Any

import numpy as np
from PIL import Image, ImageDraw

SYMBOLS = ("ring", "cross", "diamond", "zigzag")
ROLE_COLORS = (
    (220, 45, 45),
    (35, 105, 220),
    (30, 160, 92),
    (184, 56, 184),
)
MARKER_SIZE = 16
MARKER_GAP = 8
CANVAS_SIZE = (768, 640)
_RESAMPLE = getattr(Image, "Resampling", Image).LANCZOS


def _clamp(value: int, low: int, high: int) -> int:
    return max(low, min(high, value))


def _glyph_alpha(
    symbol: str,
    side: int,
    dx: int,
    dy: int,
    stroke_width: int,
) -> np.ndarray:
    """Render a soft glyph mask in a private supersampled buffer."""

    supersample = 4
    work = Image.new("L", (side * supersample, side * supersample), 0)
    draw = ImageDraw.Draw(work)
    multipliers = {"ring": 0.78, "cross": 1.02, "diamond": 0.78, "zigzag": 0.94}
    width = max(2, int(round(stroke_width * multipliers[symbol] * supersample)))

    def point(u: float, v: float) -> tuple[int, int]:
        return (
            int(round((u * side + dx) * supersample)),
            int(round((v * side + dy) * supersample)),
        )

    if symbol == "ring":
        left, top = point(0.20, 0.20)
        right, bottom = point(0.80, 0.80)
        draw.ellipse((left, top, right, bottom), outline=255, width=width)
    elif symbol == "cross":
        draw.line([point(0.18, 0.50), point(0.82, 0.50)], fill=255, width=width)
        draw.line([point(0.50, 0.18), point(0.50, 0.82)], fill=255, width=width)
    elif symbol == "diamond":
        pts = [point(0.50, 0.16), point(0.84, 0.50), point(0.50, 0.84), point(0.16, 0.50)]
        draw.line(pts + [pts[0]], fill=255, width=width, joint="curve")
    elif symbol == "zigzag":
        pts = [
            point(0.16, 0.68),
            point(0.33, 0.32),
            point(0.50, 0.68),
            point(0.67, 0.32),
            point(0.84, 0.68),
        ]
        draw.line(pts, fill=255, width=width, joint="curve")
    else:
        raise ValueError(f"unknown symbol {symbol!r}")

    reduced = work.resize((side, side), _RESAMPLE)
    return np.asarray(reduced, dtype=np.float32) / 255.0


def _carrier_mask(side: int, variant: int, phase: int) -> np.ndarray:
    yy, xx = np.indices((side, side), dtype=np.int32)
    if variant == 0:
        return ((xx + yy + phase) % 4) < 2
    if variant == 1:
        return ((2 * xx + 3 * yy + phase) % 6) < 3
    return ((xx // 2 + yy + phase) % 4) < 2


def _plate(
    scene: dict[str, Any],
    *,
    plate_number: int,
) -> np.ndarray:
    side = int(scene["tile_size"])
    alpha = _glyph_alpha(
        str(scene["symbol"]),
        side,
        int(scene["glyph_dx"]),
        int(scene["glyph_dy"]),
        int(scene["stroke_width"]),
    )
    base = float(scene["base_level"])
    symbol = float(scene["symbol_level"])
    dark = float(scene["carrier_dark"])
    carrier = _carrier_mask(side, int(scene["carrier_variant"]), int(scene["carrier_phase"]))
    common = base - (base - symbol) * alpha
    carrier_values = np.where(carrier, dark, base)
    if plate_number == 0:
        values = np.where(alpha > 0.001, common, carrier_values)
    else:
        values = np.where(alpha > 0.001, common, np.where(carrier, base, dark))
    return np.uint8(np.rint(np.clip(values, 0.0, 255.0)))


def _draw_markers(image: Image.Image, x: int, y: int, side: int, color: tuple[int, int, int]) -> None:
    draw = ImageDraw.Draw(image)
    boxes = (
        (x - MARKER_GAP - MARKER_SIZE, y - MARKER_GAP - MARKER_SIZE),
        (x + side + MARKER_GAP, y - MARKER_GAP - MARKER_SIZE),
        (x - MARKER_GAP - MARKER_SIZE, y + side + MARKER_GAP),
        (x + side + MARKER_GAP, y + side + MARKER_GAP),
    )
    for left, top in boxes:
        draw.rectangle(
            (left, top, left + MARKER_SIZE - 1, top + MARKER_SIZE - 1),
            fill=color,
        )


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one visible scene; the generator balances symbols explicitly."""

    seed = int(seed)
    rng = random.Random(seed * 1_000_003 + 97)
    width, height = CANVAS_SIZE
    side = rng.randint(176, 208)
    layout = "horizontal" if rng.randrange(2) == 0 else "vertical"
    x_max = width - side - 24
    y_max = height - side - 24
    if layout == "horizontal":
        plate_a_x = rng.randint(48, 96)
        plate_b_x = rng.randint(width - side - 96, width - side - 48)
        base_y = rng.randint(150, 218)
        plate_a_y = base_y
        plate_b_y = _clamp(base_y + rng.randint(-22, 22), 48, y_max)
    else:
        plate_a_y = rng.randint(48, 86)
        plate_b_y = rng.randint(height - side - 86, height - side - 48)
        base_x = rng.randint(170, 230)
        plate_a_x = base_x
        plate_b_x = _clamp(base_x + rng.randint(-22, 22), 48, x_max)

    colors = list(ROLE_COLORS)
    rng.shuffle(colors)
    boundary_case = (seed % 13) == 0
    base_level = rng.randint(178, 210)
    contrast = rng.randint(64, 78) if boundary_case else rng.randint(118, 158)
    symbol_level = base_level - contrast
    return {
        "symbol": SYMBOLS[seed % len(SYMBOLS)],
        "canvas_width": width,
        "canvas_height": height,
        "tile_size": side,
        "plate_a_x": plate_a_x,
        "plate_a_y": plate_a_y,
        "plate_b_x": plate_b_x,
        "plate_b_y": plate_b_y,
        "plate_a_color": list(colors[0]),
        "plate_b_color": list(colors[1]),
        "carrier_variant": rng.randrange(3),
        "carrier_phase": rng.randrange(12),
        "base_level": base_level,
        "symbol_level": symbol_level,
        "carrier_dark": rng.randint(34, 72),
        "glyph_dx": rng.randint(-12, 12),
        "glyph_dy": rng.randint(-12, 12),
        "stroke_width": rng.randint(12, 18),
        "layout": layout,
        "boundary_case": boundary_case,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    width = int(scene["canvas_width"])
    height = int(scene["canvas_height"])
    side = int(scene["tile_size"])
    image = Image.new("RGB", (width, height), (255, 255, 255))
    for number, prefix in ((0, "a"), (1, "b")):
        x = int(scene[f"plate_{prefix}_x"])
        y = int(scene[f"plate_{prefix}_y"])
        gray = _plate(scene, plate_number=number)
        rgb = np.repeat(gray[:, :, None], 3, axis=2)
        image.paste(Image.fromarray(rgb, mode="RGB"), (x, y))
    _draw_markers(image, int(scene["plate_a_x"]), int(scene["plate_a_y"]), side, tuple(scene["plate_a_color"]))
    _draw_markers(image, int(scene["plate_b_x"]), int(scene["plate_b_y"]), side, tuple(scene["plate_b_color"]))
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(scene["symbol"])


def margin(scene: dict[str, Any]) -> float:
    """Signal contrast above the explicit 90-level finite-resolution boundary."""

    return round((float(scene["base_level"]) - float(scene["symbol_level"]) - 90.0) / 255.0, 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No latent aliases are claimed for this fully rendered parameterization."""

    return []
