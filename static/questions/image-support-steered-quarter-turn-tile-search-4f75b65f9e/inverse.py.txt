"""Independent PNG-only inverse arm for quarter-turn tile search."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_INK = np.array((18, 155, 180), dtype=np.int16)
_LABELS = ("A", "B", "C", "D")


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for start_y, start_x in np.argwhere(mask):
        if seen[start_y, start_x]:
            continue
        queue = deque([(int(start_y), int(start_x))])
        seen[start_y, start_x] = True
        points = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= 350:
            output.append(np.asarray(points, dtype=np.int32))
    return output


def _occupancy(component: np.ndarray, color_mask: np.ndarray) -> tuple[np.ndarray, float]:
    y0, x0 = component.min(axis=0)
    y1, x1 = component.max(axis=0) + 1
    height, width = int(y1 - y0), int(x1 - x0)
    if not (65 <= height <= 80 and 65 <= width <= 80):
        raise ValueError("candidate component has an incompatible extent")
    occupancy = np.zeros((7, 7), dtype=bool)
    strengths = []
    for row in range(7):
        ay = int(round(y0 + row * height / 7.0))
        by = int(round(y0 + (row + 1) * height / 7.0))
        for col in range(7):
            ax = int(round(x0 + col * width / 7.0))
            bx = int(round(x0 + (col + 1) * width / 7.0))
            fraction = float(color_mask[ay:by, ax:bx].mean())
            occupancy[row, col] = fraction >= 0.55
            strengths.append(abs(fraction - 0.5))
    return occupancy, min(strengths)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    distance = np.linalg.norm(array - _INK, axis=2)
    color_mask = distance < 28.0
    candidates = []
    for component in _components(color_mask):
        try:
            occupancy, cell_confidence = _occupancy(component, color_mask)
        except ValueError:
            continue
        if int(occupancy.sum()) < 18 or cell_confidence < 0.18:
            continue
        mismatch = int(np.count_nonzero(occupancy != np.rot90(occupancy)))
        center_x = float(component[:, 1].mean())
        candidates.append((mismatch, center_x))
    if not candidates:
        raise ValueError("no measurable cyan tiles")
    ranked = sorted(candidates)
    if ranked[0][0] > 2:
        raise ValueError("no quarter-turn invariant tile")
    if len(ranked) > 1 and ranked[1][0] - ranked[0][0] < 4:
        raise ValueError("quarter-turn winner is ambiguous")
    width = array.shape[1]
    slot = min(3, max(0, int(4.0 * ranked[0][1] / width)))
    return _LABELS[slot]
