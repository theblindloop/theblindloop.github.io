"""Renderer for stencil_cell_color_binding.

A 4x4 stencil card marks four of its sixteen cells. Three 4x4 colour grids are
drawn below it at a different cell size. Registration is by grid position, not
by pixel location: cell (row, column) of the stencil corresponds to cell
(row, column) of every grid. Exactly one grid carries a single colour in all
four marked positions; the answer is that grid's letter and that colour.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 660
HEIGHT = 560
BACKGROUND = (255, 255, 255)
BORDER = (0, 0, 0)
BORDER_WIDTH = 3
GRIDLINE = (215, 215, 215)
MARK = (60, 60, 60)
LABEL = (110, 110, 110)

GRID_N = 4
N_CELLS = GRID_N * GRID_N
N_MARKS = 4
LETTERS = ("A", "B", "C")

COLOR_NAMES = ("red", "blue", "green", "orange")
COLOR_RGB = {
    "red": (200, 40, 40),
    "blue": (45, 80, 200),
    "green": (40, 150, 60),
    "orange": (235, 150, 20),
}

MIN_STRUCTURAL_GAP = 2
MIN_PIXEL_SLACK = 6.0
GAP_SCALE = 6.0


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------
def _grid_from_multiset(rng: np.random.Generator) -> list[str]:
    """A row-major grid using every colour exactly four times."""
    pool = [name for name in COLOR_NAMES for _ in range(N_MARKS)]
    order = list(map(int, rng.permutation(N_CELLS)))
    return [pool[index] for index in order]


def _defect(flat: list[str], marked: list[tuple[int, int]]) -> int:
    """Smallest number of marked cells that spoil a single-colour reading."""
    best = N_MARKS
    for name in COLOR_NAMES:
        wrong = sum(1 for r, c in marked if flat[r * GRID_N + c] != name)
        best = min(best, wrong)
    return int(best)


def sample_scene(seed: int) -> dict:
    """Deterministically sample one balanced scene."""
    rng = np.random.default_rng(int(seed) & 0x7FFFFFFF)

    key_cell = int(rng.integers(25, 35))
    while True:
        grid_cell = int(rng.integers(20, 31))
        if abs(grid_cell - key_cell) >= 4:
            break

    marked_flat = sorted(map(int, rng.choice(N_CELLS, size=N_MARKS, replace=False)))
    marked = [[index // GRID_N, index % GRID_N] for index in marked_flat]
    marked_pairs = [(int(r), int(c)) for r, c in marked]

    answer_color = COLOR_NAMES[int(rng.integers(0, len(COLOR_NAMES)))]
    answer_index = int(rng.integers(0, len(LETTERS)))

    others = [name for name in COLOR_NAMES if name != answer_color]
    filler = [name for name in others for _ in range(N_MARKS)]
    filler = [filler[i] for i in map(int, rng.permutation(len(filler)))]
    winner = [None] * N_CELLS
    for r, c in marked_pairs:
        winner[r * GRID_N + c] = answer_color
    free = [i for i in range(N_CELLS) if winner[i] is None]
    for slot, name in zip(free, filler):
        winner[slot] = name

    flats: list[list[str]] = []
    for position in range(len(LETTERS)):
        if position == answer_index:
            flats.append(list(winner))
            continue
        while True:
            trial = _grid_from_multiset(rng)
            if _defect(trial, marked_pairs) >= MIN_STRUCTURAL_GAP and trial not in flats:
                flats.append(trial)
                break

    key_width = GRID_N * key_cell + 2 * BORDER_WIDTH
    grid_width = GRID_N * grid_cell + 2 * BORDER_WIDTH
    key_x = (WIDTH - key_width) // 2 + int(rng.integers(-30, 31))
    key_y = 36 + int(rng.integers(0, 13))

    spacing = int(rng.integers(28, 45))
    row_width = 3 * grid_width + 2 * spacing
    row_x = (WIDTH - row_width) // 2 + int(rng.integers(-14, 15))
    row_y = 300 + int(rng.integers(0, 25))
    grid_origins = [[row_x + i * (grid_width + spacing), row_y] for i in range(3)]

    return {
        "seed": int(seed),
        "key_cell": key_cell,
        "grid_cell": grid_cell,
        "key_origin": [int(key_x), int(key_y)],
        "grid_origins": [[int(x), int(y)] for x, y in grid_origins],
        "marked": [[int(r), int(c)] for r, c in marked],
        "grids": [
            [[index // GRID_N, index % GRID_N, flat[index]] for index in range(N_CELLS)]
            for flat in flats
        ],
        "answer_letter": LETTERS[answer_index],
    }


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def _flat_grid(triples: list) -> list[str]:
    flat: list[str] = [""] * N_CELLS
    for row, col, name in triples:
        flat[int(row) * GRID_N + int(col)] = str(name)
    if any(not name for name in flat):
        raise ValueError("grid does not cover every cell exactly once")
    return flat


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    origin: tuple[int, int],
    cell: int,
    fills: list,
    label: str,
) -> None:
    x0, y0 = origin
    span = GRID_N * cell
    x1 = x0 + span + 2 * BORDER_WIDTH - 1
    y1 = y0 + span + 2 * BORDER_WIDTH - 1
    inner = (x0 + BORDER_WIDTH, y0 + BORDER_WIDTH)
    for row in range(GRID_N):
        for col in range(GRID_N):
            fill = fills[row * GRID_N + col]
            if fill is None:
                continue
            cx = inner[0] + col * cell
            cy = inner[1] + row * cell
            draw.rectangle([cx, cy, cx + cell - 1, cy + cell - 1], fill=fill)
    for step in range(1, GRID_N):
        offset = step * cell
        draw.line(
            [inner[0] + offset, inner[1], inner[0] + offset, inner[1] + span - 1],
            fill=GRIDLINE,
        )
        draw.line(
            [inner[0], inner[1] + offset, inner[0] + span - 1, inner[1] + offset],
            fill=GRIDLINE,
        )
    draw.rectangle([x0, y0, x1, y1], outline=BORDER, width=BORDER_WIDTH)
    draw.text((x0, y0 - 14), label, fill=LABEL)


def render(scene: dict) -> Image.Image:
    """Rasterise a scene; deterministic and free of antialiasing."""
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    marked = {(int(r), int(c)) for r, c in scene["marked"]}
    key_fills = [
        MARK if (index // GRID_N, index % GRID_N) in marked else None
        for index in range(N_CELLS)
    ]
    _draw_panel(
        draw,
        (int(scene["key_origin"][0]), int(scene["key_origin"][1])),
        int(scene["key_cell"]),
        key_fills,
        "KEY",
    )

    for position, triples in enumerate(scene["grids"]):
        flat = _flat_grid(triples)
        fills = [COLOR_RGB[name] for name in flat]
        origin = scene["grid_origins"][position]
        _draw_panel(
            draw,
            (int(origin[0]), int(origin[1])),
            int(scene["grid_cell"]),
            fills,
            LETTERS[position],
        )
    return image


# --------------------------------------------------------------------------
# analytic consequences of the scene
# --------------------------------------------------------------------------
def _solve(scene: dict) -> tuple[list[int], list[str]]:
    """Per-grid defect and the single colour each grid would read out."""
    marked = [(int(r), int(c)) for r, c in scene["marked"]]
    defects: list[int] = []
    readouts: list[str] = []
    for triples in scene["grids"]:
        flat = _flat_grid(triples)
        values = {flat[r * GRID_N + c] for r, c in marked}
        defects.append(_defect(flat, marked))
        readouts.append(next(iter(values)) if len(values) == 1 else "")
    return defects, readouts


def analytic_gold(scene: dict) -> str:
    """Decision string `LETTER|colour` for the uniquely single-coloured grid."""
    defects, readouts = _solve(scene)
    winners = [index for index, defect in enumerate(defects) if defect == 0]
    if len(winners) != 1:
        raise ValueError("scene does not have a unique single-coloured grid")
    index = winners[0]
    return f"{LETTERS[index]}|{readouts[index]}"


def _pixel_slack(scene: dict) -> float:
    smallest = min(int(scene["key_cell"]), int(scene["grid_cell"]))
    return float(smallest) / 2.0 - 3.0


def _structural_gap(scene: dict) -> int:
    defects, _ = _solve(scene)
    return int(min(defect for defect in defects if defect > 0))


def margin(scene: dict) -> float:
    """Decision margin: runner-up structural defect against sampling slack."""
    return float(min(GAP_SCALE * _structural_gap(scene), _pixel_slack(scene)))


def is_quarantined(scene: dict) -> bool:
    """Near-boundary scenes: weak runner-up separation or tight cell sampling."""
    return bool(
        _structural_gap(scene) < MIN_STRUCTURAL_GAP
        or _pixel_slack(scene) < MIN_PIXEL_SLACK
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Label-preserving latent aliases that render byte-identically."""
    rng = np.random.default_rng(int(scene["seed"]) + 4231)

    mark_order = list(map(int, rng.permutation(len(scene["marked"]))))
    reordered_marks = dict(scene)
    reordered_marks["marked"] = [list(scene["marked"][i]) for i in mark_order]

    reordered_cells = dict(scene)
    reordered_cells["grids"] = [
        [list(triples[i]) for i in map(int, rng.permutation(len(triples)))]
        for triples in scene["grids"]
    ]
    return [
        ("marked_cell_order_permutation", reordered_marks),
        ("grid_cell_list_reordering", reordered_cells),
    ]
