"""Independent pixel-only inverse arm for the trace-amplitude world.

This module receives ONLY the rasterized PNG and re-measures which color trace
has the greater vertical amplitude. It shares no geometry with renderer.render()
and never imports the renderer, prompts, generator, verifier, or analytic gold.

Pipeline: (1) label every pixel to its nearest fixed color (red, blue, or light
background); (2) for each color, recover its lowest and highest inked row; (3)
take that color's vertical extent (max y - min y + 1) as its amplitude; (4)
return the color whose amplitude is larger.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "trace_amplitude_compare-oracle-0.1.0"

RED_RGB = (205, 28, 38)
BLUE_RGB = (20, 60, 200)
BACKGROUND = (252, 252, 252)

_REF = np.array(
    [
        BACKGROUND,
        RED_RGB,
        BLUE_RGB,
    ],
    dtype=np.float32,
)


def _label(arr: np.ndarray) -> np.ndarray:
    """Label every pixel 0=background, 1=red, 2=blue (nearest fixed color)."""
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - _REF[np.newaxis, np.newaxis, :, :]
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def _amplitude(labels: np.ndarray, color_label: int) -> float:
    """Vertical extent (max row - min row) of a color's inked pixels."""
    ys, _xs = np.nonzero(labels == color_label)
    if len(ys) == 0:
        return 0.0
    return float(int(np.max(ys)) - int(np.min(ys)) + 1)


def decision_from_image(image: Image.Image) -> str:
    """Return the color whose trace has the greater vertical amplitude."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label(arr)
    amp_red = _amplitude(labels, 1)
    amp_blue = _amplitude(labels, 2)
    if amp_red > amp_blue:
        return "red"
    return "blue"
