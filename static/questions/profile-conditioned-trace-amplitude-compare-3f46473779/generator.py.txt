"""Trace-amplitude comparison world: which colored trace has the greater vertical range.

A light panel shows two squiggly colored traces, one red and one blue, each a
continuous polyline that rises and falls across the panel. The question asks
which color's trace has the GREATER vertical amplitude, i.e. the larger total
range from its lowest point to its highest point. This is a *measurement* of a
plotted signal: the pixel arm must locate each color's lowest and highest ink
and compare the two vertical extents.

Because each trace wobbles around its own independently sampled vertical center,
the color that reaches the highest peak is NOT necessarily the one with the
larger amplitude (range), and the color that dips lowest is not either. Only the
max-minus-min vertical extent separates the two traces.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "trace_amplitude_compare-0.1.0"

RED_RGB = (205, 28, 38)
BLUE_RGB = (20, 60, 200)
BACKGROUND = (252, 252, 252)

WIDTH = 560
HEIGHT = 320

X0 = 24.0
X1 = 536.0
CENTER_LO = 105.0
CENTER_HI = 215.0
AMP_LO = 46.0
AMP_HI = 104.0
LINE_WIDTH = 2

MIN_AMP_MARGIN = 8.0
QUARANTINE_AMP_MARGIN = 22.0

# Normalized vertical offsets (relative to center, units of half-amplitude).
# The pattern always reaches +1 (top, center - amp/2) and -1 (bottom, center +
# amp/2) within each cycle, so the rendered vertical extent equals the declared
# amplitude exactly. Slight irregularity keeps the trace looking like a plot
# signal rather than a square wave.
PROFILE = (1.0, 0.3, -0.3, -1.0, -0.3, 0.3, 1.0, 0.4, -0.2, -1.0, -0.6, 0.2, 1.0, -0.1, -0.5, 1.0)


def sample_scene(seed: int) -> dict:
    """Sample a deterministic scene with a clearly separated amplitude decision."""
    rng = random.Random(seed * 313 + 41)
    for _attempt in range(20000):
        amp_red = rng.uniform(AMP_LO, AMP_HI)
        amp_blue = rng.uniform(AMP_LO, AMP_HI)
        if abs(amp_red - amp_blue) < MIN_AMP_MARGIN:
            continue
        center_red = rng.uniform(CENTER_LO, CENTER_HI)
        center_blue = rng.uniform(CENTER_LO, CENTER_HI)

        traces = [
            _build_trace(rng, "red", center_red, amp_red),
            _build_trace(rng, "blue", center_blue, amp_blue),
        ]
        scene = {
            "traces": traces,
            "width": WIDTH,
            "height": HEIGHT,
        }
        return scene
    raise RuntimeError("could not sample a separated amplitude scene")


def _build_trace(rng: random.Random, color: str, center: float, amplitude: float) -> dict:
    half = amplitude / 2.0
    points: list[list[float]] = []
    i = 0
    x = X0
    while x <= X1:
        offset = PROFILE[i % len(PROFILE)]
        y = center - half * offset
        points.append([round(x, 3), round(y, 3)])
        x += 26.0 + rng.uniform(0.0, 14.0)
        i += 1
    points[-1][0] = X1
    return {
        "color": color,
        "center": float(center),
        "amplitude": float(amplitude),
        "points": points,
    }


def render(scene: dict) -> Image.Image:
    """Render I = R(z): two colored polylines on a light panel."""
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    for trace in scene["traces"]:
        color = trace["color"]
        rgb = RED_RGB if color == "red" else BLUE_RGB
        coords = [(px, py) for px, py in trace["points"]]
        draw.line(coords, fill=rgb, width=LINE_WIDTH, joint="curve")
    return img


def _amplitudes(scene: dict) -> dict:
    return {trace["color"]: float(trace["amplitude"]) for trace in scene["traces"]}


def analytic_gold(scene: dict) -> str:
    """Color of the trace with the greater vertical amplitude."""
    amps = _amplitudes(scene)
    return "red" if amps["red"] > amps["blue"] else "blue"


def margin(scene: dict) -> float:
    """Absolute difference between the two declared vertical amplitudes."""
    amps = _amplitudes(scene)
    return abs(amps["red"] - amps["blue"])


def is_quarantined(scene: dict) -> bool:
    """Near-boundary: amplitude difference is uncomfortably small."""
    return margin(scene) < QUARANTINE_AMP_MARGIN


def latent_symmetries(scene: dict) -> list:
    """not_applicable: every latent field is drawn verbatim as visible ink.

    Each trace's color, vertical center, declared amplitude, and the points
    through which the trace is drawn all appear as visible colored ink, and the
    analytic decision reads only the directly rendered trace amplitudes. There
    is no hidden scene field that could change the amplitude comparison without
    changing the raster, so no nontrivial pixel-identical, gold-preserving
    latent transformation exists.
    """
    return []
