"""Deterministic sampler, raster renderer and analytic gold for weave_cycle_panel.

Each scene draws five square framed panels in a row.  Every panel carries three
straight opaque bands of equal thickness, each in its own colour, each running
all the way across the panel.  The three bands meet in three separate crossings
and at every crossing exactly one band is painted in front of the other.  Four
panels use a tournament of in-front-of relations that admits a single global
stacking order; exactly one panel uses a cyclic tournament, which no stack of
flat bands can realise.  The analytic gold names that woven panel.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

CANVAS_MARGIN = 18
PANEL_SIZE = 168
PANEL_GAP = 14
N_PANELS = 5
FRAME_WIDTH = 3
BAND_WIDTH = 17
BAND_HALF = BAND_WIDTH / 2.0
BAND_LENGTH = 460.0

MAX_OFFSET = 34.0
ANGLE_JITTER = 12.0
MIN_CROSS_BORDER = 30.0
MIN_CROSS_SEPARATION = 36.0
QUARANTINE_MARGIN = 18.0

WIDTH = 2 * CANVAS_MARGIN + N_PANELS * PANEL_SIZE + (N_PANELS - 1) * PANEL_GAP
HEIGHT = 2 * CANVAS_MARGIN + PANEL_SIZE

PAIRS = ((0, 1), (0, 2), (1, 2))

PALETTES = [
    {
        "page": (250, 249, 244),
        "frame": (120, 122, 126),
        "bands": [(196, 74, 62), (58, 104, 166), (74, 140, 92), (176, 132, 44)],
    },
    {
        "page": (246, 246, 250),
        "frame": (110, 112, 118),
        "bands": [(158, 76, 150), (52, 128, 140), (188, 96, 54), (86, 108, 178)],
    },
    {
        "page": (252, 248, 240),
        "frame": (128, 124, 118),
        "bands": [(64, 118, 108), (170, 66, 88), (104, 96, 168), (154, 138, 52)],
    },
]

AMBIGUOUS = "ambiguous"


def _tournaments():
    """All eight in-front-of assignments for the three crossings, split by type."""
    cyclic, acyclic = [], []
    for a in PAIRS[0]:
        for b in PAIRS[1]:
            for c in PAIRS[2]:
                order = (a, b, c)
                wins = [0, 0, 0]
                for winner in order:
                    wins[winner] += 1
                (cyclic if sorted(wins) == [1, 1, 1] else acyclic).append(order)
    return tuple(cyclic), tuple(acyclic)


CYCLIC_ORDERS, ACYCLIC_ORDERS = _tournaments()


def _axes(angle_deg: float):
    """Unit direction and unit normal, folded so that a 180 degree turn is exact."""
    folded = round(float(angle_deg) % 360.0, 4)
    sign = 1.0
    if folded >= 180.0:
        folded = round(folded - 180.0, 4)
        sign = -1.0
    radians = math.radians(folded)
    cos, sin = math.cos(radians), math.sin(radians)
    return (sign * cos, sign * sin), (-sign * sin, sign * cos)


def _crossing(angles, offsets, i: int, j: int):
    """Panel-local intersection of two band centre lines, or None if degenerate."""
    _, n_i = _axes(angles[i])
    _, n_j = _axes(angles[j])
    det = n_i[0] * n_j[1] - n_j[0] * n_i[1]
    if abs(det) < 1e-9:
        return None
    d_i, d_j = float(offsets[i]), float(offsets[j])
    x = (d_i * n_j[1] - d_j * n_i[1]) / det
    y = (n_i[0] * d_j - n_j[0] * d_i) / det
    return (x, y)


def _crossing_clarities(angles, offsets):
    """Per-crossing readability in pixels, or None when a crossing is degenerate."""
    points = []
    for i, j in PAIRS:
        point = _crossing(angles, offsets, i, j)
        if point is None:
            return None
        points.append(point)
    half = PANEL_SIZE / 2.0 - FRAME_WIDTH
    clarities = []
    for index, (x, y) in enumerate(points):
        border = min(half - x, half + x, half - y, half + y)
        separation = min(
            math.hypot(x - other[0], y - other[1])
            for position, other in enumerate(points)
            if position != index
        )
        clarities.append(min(border, separation))
    return clarities


def _panel_clarities(scene: dict, index: int):
    """Crossing readabilities of one panel, addressed through the scene."""
    panel = scene["panels"][index]
    return _crossing_clarities(panel["angles"], panel["offsets"])


def _is_cyclic(order) -> bool:
    wins = [0, 0, 0]
    for winner in order:
        wins[int(winner)] += 1
    return sorted(wins) == [1, 1, 1]


def _cyclic_panels(scene: dict):
    return [
        index
        for index, panel in enumerate(scene["panels"])
        if _is_cyclic(panel["order"])
    ]


def _sample_panel(rng: random.Random, cyclic: bool) -> dict:
    while True:
        base = rng.uniform(0.0, 180.0)
        angles = [
            base,
            base + 60.0 + rng.uniform(-ANGLE_JITTER, ANGLE_JITTER),
            base + 120.0 + rng.uniform(-ANGLE_JITTER, ANGLE_JITTER),
        ]
        angles = [round(angle % 360.0, 4) for angle in angles]
        for _ in range(160):
            offsets = [
                round(rng.uniform(-MAX_OFFSET, MAX_OFFSET), 3) for _ in range(3)
            ]
            if _crossing_clarities(angles, offsets) is None:
                continue
            points = [_crossing(angles, offsets, i, j) for i, j in PAIRS]
            half = PANEL_SIZE / 2.0 - FRAME_WIDTH
            borders = [
                min(half - x, half + x, half - y, half + y) for x, y in points
            ]
            separations = [
                math.hypot(points[a][0] - points[b][0], points[a][1] - points[b][1])
                for a, b in ((0, 1), (0, 2), (1, 2))
            ]
            if min(borders) < MIN_CROSS_BORDER:
                continue
            if min(separations) < MIN_CROSS_SEPARATION:
                continue
            pool = CYCLIC_ORDERS if cyclic else ACYCLIC_ORDERS
            colours = rng.sample(range(4), 3)
            return {
                "angles": angles,
                "offsets": offsets,
                "colours": colours,
                "order": list(pool[rng.randrange(len(pool))]),
            }


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    palette = rng.randrange(len(PALETTES))
    woven = rng.randrange(N_PANELS)
    panels = [_sample_panel(rng, index == woven) for index in range(N_PANELS)]
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "palette": palette,
        "panels": panels,
    }


def _band_polygon(centre, angle: float, offset: float):
    unit, normal = _axes(angle)
    px = centre[0] + offset * normal[0]
    py = centre[1] + offset * normal[1]
    corners = []
    for along in (BAND_LENGTH, -BAND_LENGTH):
        for across in (BAND_HALF, -BAND_HALF):
            corners.append(
                (
                    px + along * unit[0] + across * normal[0],
                    py + along * unit[1] + across * normal[1],
                )
            )
    return [corners[0], corners[1], corners[3], corners[2]]


def _oriented(polygon):
    area = 0.0
    for index, (x, y) in enumerate(polygon):
        nx, ny = polygon[(index + 1) % len(polygon)]
        area += x * ny - nx * y
    return polygon if area >= 0 else list(reversed(polygon))


def _clip(subject, clipper):
    """Sutherland-Hodgman clip of a polygon against a convex clipper."""
    output = list(subject)
    clipper = _oriented(list(clipper))
    for index in range(len(clipper)):
        ax, ay = clipper[index]
        bx, by = clipper[(index + 1) % len(clipper)]

        def inside(point, ax=ax, ay=ay, bx=bx, by=by):
            return (bx - ax) * (point[1] - ay) - (by - ay) * (point[0] - ax) >= 0.0

        def cut(p, q, ax=ax, ay=ay, bx=bx, by=by):
            rx, ry = bx - ax, by - ay
            sx, sy = q[0] - p[0], q[1] - p[1]
            denominator = rx * sy - ry * sx
            if abs(denominator) < 1e-12:
                return p
            t = ((p[0] - ax) * ry - (p[1] - ay) * rx) / denominator
            return (p[0] + t * sx, p[1] + t * sy)

        current, output = output, []
        for position, point in enumerate(current):
            previous = current[position - 1]
            if inside(point):
                if not inside(previous):
                    output.append(cut(previous, point))
                output.append(point)
            elif inside(previous):
                output.append(cut(previous, point))
        if not output:
            return []
    return output


def _panel_box(index: int):
    x0 = CANVAS_MARGIN + index * (PANEL_SIZE + PANEL_GAP)
    y0 = CANVAS_MARGIN
    return x0, y0, x0 + PANEL_SIZE - 1, y0 + PANEL_SIZE - 1


def render(scene: dict) -> Image.Image:
    palette = PALETTES[int(scene["palette"]) % len(PALETTES)]
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), palette["page"])
    draw = ImageDraw.Draw(image)
    for index, panel in enumerate(scene["panels"]):
        x0, y0, x1, y1 = _panel_box(index)
        draw.rectangle([x0, y0, x1, y1], outline=palette["frame"], width=FRAME_WIDTH)
        interior = [
            (x0 + FRAME_WIDTH, y0 + FRAME_WIDTH),
            (x1 - FRAME_WIDTH, y0 + FRAME_WIDTH),
            (x1 - FRAME_WIDTH, y1 - FRAME_WIDTH),
            (x0 + FRAME_WIDTH, y1 - FRAME_WIDTH),
        ]
        centre = (x0 + (PANEL_SIZE - 1) / 2.0, y0 + (PANEL_SIZE - 1) / 2.0)
        polygons = [
            _clip(
                _band_polygon(centre, panel["angles"][band], panel["offsets"][band]),
                interior,
            )
            for band in range(3)
        ]
        colours = [palette["bands"][slot] for slot in panel["colours"]]
        for band in range(3):
            if polygons[band]:
                draw.polygon(polygons[band], fill=colours[band])
        for position, (i, j) in enumerate(PAIRS):
            overlap = _clip(polygons[i], polygons[j]) if polygons[i] and polygons[j] else []
            if overlap:
                draw.polygon(overlap, fill=colours[int(panel["order"][position])])
    return image


def margin(scene: dict) -> float:
    """Smallest visible evidence, in pixels, separating this scene from another answer."""
    if len(_cyclic_panels(scene)) != 1:
        return 0.0
    worst = None
    for index in range(len(scene["panels"])):
        clarities = _panel_clarities(scene, index)
        if clarities is None:
            return 0.0
        panel_worst = min(clarities)
        worst = panel_worst if worst is None else min(worst, panel_worst)
    return float(worst)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def analytic_gold(scene: dict) -> str:
    if is_quarantined(scene):
        return AMBIGUOUS
    return f"panel-{_cyclic_panels(scene)[0] + 1}"


def _flip_band(panel: dict, band: int) -> dict:
    copy = {
        "angles": list(panel["angles"]),
        "offsets": list(panel["offsets"]),
        "colours": list(panel["colours"]),
        "order": list(panel["order"]),
    }
    copy["angles"][band] = round((float(panel["angles"][band]) + 180.0) % 360.0, 4)
    copy["offsets"][band] = -float(panel["offsets"][band])
    return copy


def latent_symmetries(scene: dict) -> list:
    """Declared latent aliases: pixel-identical, gold-preserving scene rewrites."""
    period = dict(scene)
    period["palette"] = int(scene["palette"]) + len(PALETTES)

    flipped_all = dict(scene)
    flipped_all["panels"] = [
        _flip_band(_flip_band(_flip_band(panel, 0), 1), 2) for panel in scene["panels"]
    ]

    flipped_first = dict(scene)
    flipped_first["panels"] = [
        _flip_band(panel, 0) if index == 0 else panel
        for index, panel in enumerate(scene["panels"])
    ]

    return [
        ("palette_period", period),
        ("band_direction_reversal_all", flipped_all),
        ("band_direction_reversal_first_panel", flipped_first),
    ]
