"""Deterministic renderer and analytic reference for the concyclic search world."""

from __future__ import annotations

import copy
import itertools
import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1000
HEIGHT = 760
PANEL_RECTS = (
    (70, 80, 470, 340),
    (530, 80, 930, 340),
    (70, 420, 470, 680),
    (530, 420, 930, 680),
)
LABELS = ("A", "B", "C", "D")
MARKER_RADIUS = 12
MARKER_COLOR = (31, 47, 66)
TEXT_COLOR = (78, 92, 108)
PANEL_FILL = (250, 251, 253)
PANEL_OUTLINE = (155, 166, 178)
QUARANTINE_GAP = 8.0


def _circumcircle(a: list[float], b: list[float], c: list[float]):
    ax, ay = a
    bx, by = b
    cx, cy = c
    determinant = 2.0 * (ax * (by - cy) + bx * (cy - ay) + cx * (ay - by))
    if abs(determinant) < 1e-9:
        return None
    aa = ax * ax + ay * ay
    bb = bx * bx + by * by
    cc = cx * cx + cy * cy
    ux = (aa * (by - cy) + bb * (cy - ay) + cc * (ay - by)) / determinant
    uy = (aa * (cx - bx) + bb * (ax - cx) + cc * (bx - ax)) / determinant
    return ux, uy, math.hypot(ux - ax, uy - ay)


def circle_residual(points: list[list[float]]) -> float:
    """Minimum fourth-point radial residual over the four triple fits."""

    values = []
    for triple in itertools.combinations(range(4), 3):
        remaining = next(index for index in range(4) if index not in triple)
        circle = _circumcircle(*(points[index] for index in triple))
        if circle is None:
            continue
        cx, cy, radius = circle
        px, py = points[remaining]
        values.append(abs(math.hypot(px - cx, py - cy) - radius))
    return min(values) if values else 1_000_000.0


def _panel_scores(scene: dict) -> list[float]:
    return [circle_residual(panel["points"]) for panel in scene["panels"]]


def analytic_gold(scene: dict) -> str:
    scores = _panel_scores(scene)
    best = min(range(len(scores)), key=lambda index: scores[index])
    return scene["panels"][best]["label"]


def margin(scene: dict) -> float:
    scores = sorted(_panel_scores(scene))
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def _minimum_pairwise_distance(points: list[list[int]]) -> float:
    return min(
        math.hypot(points[i][0] - points[j][0], points[i][1] - points[j][1])
        for i, j in itertools.combinations(range(len(points)), 2)
    )


def _target_offsets(rng: random.Random) -> list[tuple[int, int]]:
    circle_points = [
        (x, y)
        for x in range(-65, 66)
        for y in range(-65, 66)
        if x * x + y * y == 65 * 65
    ]
    for _ in range(200):
        chosen = rng.sample(circle_points, 4)
        if _minimum_pairwise_distance([[x, y] for x, y in chosen]) >= 34.0:
            return chosen
    return [(0, -65), (16, 63), (63, 16), (-63, -16)]


def _target_points(rng: random.Random, origin: tuple[int, int]) -> list[list[int]]:
    center_x = 200 + rng.randint(-10, 10)
    center_y = 135 + rng.randint(-8, 8)
    offsets = _target_offsets(rng)
    points = [[origin[0] + center_x + dx, origin[1] + center_y + dy] for dx, dy in offsets]
    rng.shuffle(points)
    return points


def _distractor_points(rng: random.Random, origin: tuple[int, int]) -> list[list[int]]:
    for _ in range(2000):
        center_x = 200 + rng.randint(-10, 10)
        center_y = 135 + rng.randint(-8, 8)
        angles = rng.sample(range(0, 360), 4)
        radii = [rng.choice((44, 50, 56, 74, 82, 90)) for _ in range(4)]
        local = [
            [
                center_x + int(round(radius * math.cos(math.radians(angle)))),
                center_y + int(round(radius * math.sin(math.radians(angle)))),
            ]
            for angle, radius in zip(angles, radii)
        ]
        if any(x < 42 or x > 358 or y < 36 or y > 234 for x, y in local):
            continue
        if _minimum_pairwise_distance(local) < 34.0:
            continue
        if circle_residual(local) < 20.0:
            continue
        points = [[origin[0] + x, origin[1] + y] for x, y in local]
        rng.shuffle(points)
        return points
    # This deterministic fallback is intentionally far from circular.
    fallback = [[82, 54], [314, 74], [110, 206], [304, 220]]
    points = [[origin[0] + x, origin[1] + y] for x, y in fallback]
    rng.shuffle(points)
    return points


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target_index = int(seed) % 4
    panels = []
    for index, rect in enumerate(PANEL_RECTS):
        origin = (rect[0], rect[1])
        points = (
            _target_points(rng, origin)
            if index == target_index
            else _distractor_points(rng, origin)
        )
        panels.append({"label": LABELS[index], "points": points})
    scene = {"panels": panels}
    if is_quarantined(scene):
        # The construction loops above make this unreachable for normal seeds;
        # retaining the explicit check documents the public boundary contract.
        raise RuntimeError(f"seed {seed} produced a quarantined scene")
    return scene


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), "white")
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    for index, panel in enumerate(scene["panels"]):
        x0, y0, x1, y1 = PANEL_RECTS[index]
        draw.rounded_rectangle(
            (x0, y0, x1, y1),
            radius=12,
            fill=PANEL_FILL,
            outline=PANEL_OUTLINE,
            width=2,
        )
        draw.text((x0 + 14, y0 + 10), panel["label"], fill=TEXT_COLOR, font=font)
        for x, y in panel["points"]:
            draw.ellipse(
                (x - MARKER_RADIUS, y - MARKER_RADIUS, x + MARKER_RADIUS, y + MARKER_RADIUS),
                fill=MARKER_COLOR,
            )
    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    for panel in twin["panels"]:
        panel["points"] = list(reversed(panel["points"]))
    return [("point_order_reversal", twin)]
