"""Independent raster-only inverse for the concyclic search world."""

from __future__ import annotations

import itertools
import math

from PIL import Image

PANEL_RECTS = (
    (70, 80, 470, 340),
    (530, 80, 930, 340),
    (70, 420, 470, 680),
    (530, 420, 930, 680),
)
MARKER_RADIUS = 12
LABELS = ("A", "B", "C", "D")


def _circumcircle(a: tuple[float, float], b: tuple[float, float], c: tuple[float, float]):
    ax, ay = a
    bx, by = b
    cx, cy = c
    determinant = 2.0 * (ax * (by - cy) + bx * (cy - ay) + cx * (ay - by))
    if abs(determinant) < 1e-9:
        return None
    aa = ax * ax + ay * ay
    bb = bx * bx + by * by
    cc = cx * cx + cy * cy
    ux = (aa * (by - cy) + bb * (cy - ay) + cc * (ay - by)) / determinant
    uy = (aa * (cx - bx) + bb * (ax - cx) + cc * (bx - ax)) / determinant
    return ux, uy, math.hypot(ux - ax, uy - ay)


def _circle_residual(points: list[tuple[float, float]]) -> float:
    values = []
    for triple in itertools.combinations(range(4), 3):
        remaining = next(index for index in range(4) if index not in triple)
        circle = _circumcircle(*(points[index] for index in triple))
        if circle is None:
            continue
        cx, cy, radius = circle
        px, py = points[remaining]
        values.append(abs(math.hypot(px - cx, py - cy) - radius))
    return min(values) if values else 1_000_000.0


def _marker_centers(image: Image.Image, rect: tuple[int, int, int, int]):
    x0, y0, x1, y1 = rect
    pixels = image.load()
    visited: set[tuple[int, int]] = set()
    centers = []
    for y in range(y0, y1 + 1):
        for x in range(x0, x1 + 1):
            if (x, y) in visited:
                continue
            red, green, blue = pixels[x, y]
            if not (red < 60 and green < 75 and blue < 95):
                continue
            stack = [(x, y)]
            visited.add((x, y))
            component = []
            while stack:
                px, py = stack.pop()
                component.append((px, py))
                for nx, ny in ((px - 1, py), (px + 1, py), (px, py - 1), (px, py + 1)):
                    if nx < x0 or nx > x1 or ny < y0 or ny > y1 or (nx, ny) in visited:
                        continue
                    nr, ng, nb = pixels[nx, ny]
                    if nr < 60 and ng < 75 and nb < 95:
                        visited.add((nx, ny))
                        stack.append((nx, ny))
            if 350 <= len(component) <= 1000:
                centers.append(
                    (
                        sum(px for px, _ in component) / len(component),
                        sum(py for _, py in component) / len(component),
                    )
                )
    return centers


def decision_from_image(image: Image.Image) -> str:
    """Return a panel label using only pixels from the supplied PNG image."""

    try:
        scores = []
        for rect in PANEL_RECTS:
            centers = _marker_centers(image.convert("RGB"), rect)
            if len(centers) != 4:
                return "ABSTAIN"
            scores.append(_circle_residual(centers))
        order = sorted(range(4), key=lambda index: scores[index])
        if scores[order[1]] - scores[order[0]] < 8.0:
            return "ABSTAIN"
        return LABELS[order[0]]
    except Exception:
        return "ABSTAIN"
