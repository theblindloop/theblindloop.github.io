"""Independent pixel-only inverse for the paired-silhouette world.

This module deliberately duplicates the public raster contract instead of
importing the renderer. It samples the ten-by-ten field in each fixed card,
checks the two recovered connected components, applies its own quarter-turn,
and reads the badge color only after correspondence is solved.
"""

from __future__ import annotations

from collections import deque

from PIL import Image

WIDTH, HEIGHT = 512, 512
GRID = 10
CELL = 10
PANEL_X = (196, 64, 328)
PANEL_Y = (100, 300, 300)
CONTENT_OFFSET = (10, 44)
AMBER = (231, 157, 43)
VIOLET = (155, 94, 194)
INK_SUM_LIMIT = 410


def _ink(pixel: tuple[int, int, int]) -> bool:
    return sum(int(value) for value in pixel) < INK_SUM_LIMIT


def _mask(image: Image.Image, panel_x: int, panel_y: int) -> list[list[int]]:
    pixels = image.load()
    origin_x = panel_x + CONTENT_OFFSET[0]
    origin_y = panel_y + CONTENT_OFFSET[1]
    result: list[list[int]] = []
    for row in range(GRID):
        values = []
        for col in range(GRID):
            x = origin_x + col * CELL + CELL // 2
            y = origin_y + row * CELL + CELL // 2
            values.append(int(_ink(pixels[x, y])))
        result.append(values)
    return result


def _components(mask: list[list[int]]) -> list[set[tuple[int, int]]]:
    remaining = {(y, x) for y in range(GRID) for x in range(GRID) if mask[y][x]}
    found: list[set[tuple[int, int]]] = []
    while remaining:
        start = min(remaining)
        queue = deque([start])
        component = {start}
        remaining.remove(start)
        while queue:
            y, x = queue.popleft()
            for neighbor in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if neighbor in remaining:
                    remaining.remove(neighbor)
                    component.add(neighbor)
                    queue.append(neighbor)
        found.append(component)
    return found


def _check_mask(mask: list[list[int]]) -> None:
    components = _components(mask)
    if sorted(len(component) for component in components) != [5, 5]:
        raise ValueError("card does not expose two separated five-cell islands")


def _rotate_cw(mask: list[list[int]]) -> list[list[int]]:
    return [list(row) for row in zip(*mask[::-1])]


def _error(left: list[list[int]], right: list[list[int]]) -> int:
    return sum(int(a != b) for row_a, row_b in zip(left, right) for a, b in zip(row_a, row_b))


def _badge(image: Image.Image, panel_x: int, panel_y: int) -> str:
    pixels = image.load()
    x = panel_x + 100
    y = panel_y + 30
    observed = tuple(int(value) for value in pixels[x, y])
    distances = {
        "amber": sum((observed[index] - AMBER[index]) ** 2 for index in range(3)),
        "violet": sum((observed[index] - VIOLET[index]) ** 2 for index in range(3)),
    }
    best = min(distances, key=distances.get)
    if distances[best] > 9000:
        raise ValueError("candidate badge is unreadable")
    return best


def decision_from_image(image: Image.Image) -> str:
    """Return the matching candidate's badge color using only the final PNG."""

    if image.size != (WIDTH, HEIGHT):
        raise ValueError(f"expected {WIDTH}x{HEIGHT} RGB image")
    reference = _mask(image, PANEL_X[0], PANEL_Y[0])
    candidate_a = _mask(image, PANEL_X[1], PANEL_Y[1])
    candidate_b = _mask(image, PANEL_X[2], PANEL_Y[2])
    _check_mask(reference)
    _check_mask(candidate_a)
    _check_mask(candidate_b)
    errors = {
        "A": _error(_rotate_cw(candidate_a), reference),
        "B": _error(_rotate_cw(candidate_b), reference),
    }
    if errors["A"] == errors["B"]:
        raise ValueError("candidate silhouette errors are tied")
    winner = 1 if errors["A"] < errors["B"] else 2
    return _badge(image, PANEL_X[winner], PANEL_Y[winner])
