"""Independent pixel-only inverse arm for the scatter-scope correspondence.

Receives ONLY the final rasterized PNG and the public question contract (this
file's own visual spec). It never imports the renderer, prompts, generator,
verifier, or analytic-gold code, and never reads scene data.

Public visual spec:

  * the top panel holds one neutral gray "constellation": a scatter of filled
    dots equally spaced on a ring of a canonical radius (the scatter scope) from
    the set {30,56,82,108,134} px;
  * a horizontal dark divider separates it from a column of three colored
    constellations, one red, one green, one blue, each a scatter of the same
    dot count and dot size on a ring of a distinct canonical scope;
  * the task: decide which constellation color has exactly the same scatter
    scope as the gray reference.

The oracle, from pixels alone: finds the divider, segments the gray dots of the
reference and the colored dots of each candidate, and for each cluster recovers
the centroid and the mean distance of its dots from that centroid (the scatter
scope), binned to the nearest canonical scope. It returns the color of the
candidate whose binned scope equals the reference's.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "correspondence_scatter_scope-oracle-0.1.0"

PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (235, 40, 40),
    "green": (0, 170, 70),
    "blue": (40, 60, 235),
}
GRAY = (150, 150, 150)
CANONICAL_SCOPES = [30, 45, 60, 75, 90]
CAND_Y = [480, 680, 880]
_BAND = 95
_N_DARK = 90.0
_COLOR_DIST = 70.0


def _classify(arr: np.ndarray, color: tuple[int, int, int]) -> np.ndarray:
    c = np.asarray(color, dtype=np.float32)
    d = np.linalg.norm(arr - c[None, None, :], axis=-1)
    return d < _COLOR_DIST


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    ys, xs = np.nonzero(mask)
    if ys.size == 0:
        return []
    remaining = set(zip(ys.tolist(), xs.tolist()))
    comps = []
    while remaining:
        start = remaining.pop()
        comp = []
        stack = [start]
        while stack:
            r, c = stack.pop()
            comp.append((r, c))
            for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                nb = (r + dr, c + dc)
                if nb in remaining:
                    remaining.remove(nb)
                    stack.append(nb)
        if len(comp) >= 20:
            comps.append(comp)
    return comps


def _dot_centers(mask: np.ndarray) -> list[tuple[float, float]]:
    centers = []
    for comp in _components(mask):
        cs = np.mean([p[1] for p in comp])
        rs = np.mean([p[0] for p in comp])
        centers.append((cs, rs))
    return centers


def _scope(centers: list[tuple[float, float]]) -> float:
    cx = np.mean([c[0] for c in centers])
    cy = np.mean([c[1] for c in centers])
    dists = [math.hypot(c[0] - cx, c[1] - cy) for c in centers]
    mean = float(np.mean(dists)) if dists else 0.0
    return float(min(CANONICAL_SCOPES, key=lambda value: abs(value - mean)))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.float32)

    dark = arr.max(axis=2) < _N_DARK
    line_counts = dark.sum(axis=1)
    grid = line_counts > 0.45 * dark.shape[1]
    rows = np.where(grid)[0]
    if rows.size == 0:
        raise ValueError("no divider line found")
    divider_mid = float(rows.mean())

    gray_mask = _classify(arr, GRAY)
    ref_centers = [c for c in _dot_centers(gray_mask) if c[1] < divider_mid]
    if not ref_centers:
        raise ValueError("no reference dots found above divider")
    ref_scope = _scope(ref_centers)

    color_masks = {name: _classify(arr, rgb) for name, rgb in PALETTE.items()}

    matched_color = None
    for cy in CAND_Y:
        band_centers = {}
        for name, mask in color_masks.items():
            centers = [c for c in _dot_centers(mask) if abs(c[1] - cy) < _BAND]
            band_centers[name] = centers
        dominant = max(band_centers, key=lambda name: len(band_centers[name]))
        centers = band_centers[dominant]
        if not centers:
            continue
        scope = _scope(centers)
        if scope == ref_scope:
            matched_color = dominant
            break
    if matched_color is None:
        raise ValueError(f"no candidate matched reference scope {ref_scope}")
    return matched_color


if __name__ == "__main__":
    import sys
    img = Image.open(sys.argv[1]).convert("RGB")
    print(decision_from_image(img))
