"""Point-set scatter-scope correspondence renderer (paper-semantic family).

A cross-panel dispersion correspondence task. A neutral gray "constellation"
(a small scatter of filled dots placed on a ring) is shown at the top: the
reference. Its spatial scope (the radius of the scatter, i.e. how far the
points spread from the cluster center) is drawn from a finite canonical set.
Below a divider sit three colored constellations (red / green / blue), each a
scatter of dots sharing a fixed dot count and dot size but with a distinct
canonical scope. Exactly one candidate has the same scatter scope as the
reference. The question asks for that candidate's color.

The pixel-only inverse arm must, for every panel, segment the dots, recover
the cluster centroid and the mean distance of the dots from it (the scatter
scope), solve the cross-panel assignment by scope equality, and only then read
out the bound color of the single matched constellation.

Latent state z (every field that influences the gold is rendered -- nothing
hidden):

    ref_scope  : scatter radius (px) of the gray reference constellation, drawn
                 from CANONICAL_SCOPES.
    cand_scopes: length-3 list of candidate scatter scopes; exactly one equals
                 ``ref_scope``, the other two are distinct and different.
    cand_colors: the fixed 3-color palette bound one-to-one to the candidates.
                 Each candidate constellation's dots are drawn in its bound color.
    dot_angles : a per-constellation random starting angle for the dot placement
                 on the ring, so the absolute orientation of the scatter is
                 counterbalanced and only its scope (not direction) is meaningful.
    n_dots, dot_r, centers, divider_y, width, height : fixed geometry.

The analytic gold is the color of the candidate whose scatter scope equals the
reference's. Because the reference scope, each candidate scope, and the bound
colors are all rendered verbatim, the gold is recoverable from the pixels alone.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "correspondence_scatter_scope-0.1.0"

BACKGROUND = "white"
DIVIDER = (70, 70, 70)
GRAY = (150, 150, 150)

PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (235, 40, 40),
    "green": (0, 170, 70),
    "blue": (40, 60, 235),
}
CAND_COLOR_ORDER = ["red", "green", "blue"]

# Canonical scatter scopes (px). Adjacent values are separated by at least 15
# px, well above the finite-resolution oracle tolerance.
CANONICAL_SCOPES = [30, 45, 60, 75, 90]

WIDTH = 900
HEIGHT = 960
DIVIDER_Y = 230
DIVIDER_THICK = 3
N_DOTS = 8
DOT_R = 7
REF_CENTER = (450, 120)
CAND_CENTERS = [(450, 480), (450, 680), (450, 880)]

_QUARANTINE_FLOOR = 6.0


def sample_scene(seed: int) -> dict:
    """Sample a balanced, deterministic latent scene (dict)."""
    rng = random.Random(seed)
    ref_scope = rng.choice(CANONICAL_SCOPES)
    pool = [s for s in CANONICAL_SCOPES if s != ref_scope]
    rng.shuffle(pool)
    distractors = pool[:2]
    cand_scopes = [ref_scope] + distractors
    rng.shuffle(cand_scopes)

    starts = [rng.uniform(0.0, 2.0 * math.pi) for _ in range(4)]

    return {
        "ref_scope": ref_scope,
        "cand_scopes": cand_scopes,
        "cand_colors": list(PALETTE.keys()),
        "dot_angles": starts,
        "n_dots": N_DOTS,
        "dot_r": DOT_R,
        "ref_center": list(REF_CENTER),
        "cand_centers": [[c[0], c[1]] for c in CAND_CENTERS],
        "divider_y": DIVIDER_Y,
        "width": WIDTH,
        "height": HEIGHT,
    }


def _draw_constellation(draw, center, scope, start_angle, n, color):
    cx, cy = center
    for k in range(n):
        theta = start_angle + 2.0 * math.pi * k / n
        x = cx + scope * math.cos(theta)
        y = cy + scope * math.sin(theta)
        draw.ellipse([x - DOT_R, y - DOT_R, x + DOT_R, y + DOT_R], fill=color)


def render(scene: dict) -> Image.Image:
    """Render I = R(z) deterministically at pixel resolution."""
    w = int(scene["width"])
    h = int(scene["height"])
    img = Image.new("RGB", (w, h), BACKGROUND)
    draw = ImageDraw.Draw(img)
    _draw_constellation(
        draw,
        tuple(scene["ref_center"]),
        scene["ref_scope"],
        scene["dot_angles"][0],
        scene["n_dots"],
        GRAY,
    )
    for i, center in enumerate(scene["cand_centers"]):
        color = PALETTE[scene["cand_colors"][i]]
        _draw_constellation(
            draw,
            tuple(center),
            scene["cand_scopes"][i],
            scene["dot_angles"][1 + i],
            scene["n_dots"],
            color,
        )
    draw.rectangle([0, scene["divider_y"], w, scene["divider_y"] + DIVIDER_THICK], fill=DIVIDER)
    return img


def analytic_gold(scene: dict) -> str:
    """Decision: color (red/green/blue) of the constellation with the ref scope."""
    idx = scene["cand_scopes"].index(scene["ref_scope"])
    return scene["cand_colors"][idx]


def margin(scene: dict) -> float:
    """Runner-up margin: minimum absolute scope difference (px) between the
    reference constellation and any distractor candidate."""
    ref = scene["ref_scope"]
    diffs = [abs(ref - s) for s in scene["cand_scopes"] if s != ref]
    return float(min(diffs)) if diffs else 0.0


def is_quarantined(scene: dict) -> bool:
    """Quarantine only an ambiguous scene where a distractor comes within
    _QUARANTINE_FLOOR px of the reference scope (never produced because
    canonical scopes are separated by at least 24 px)."""
    return margin(scene) < _QUARANTINE_FLOOR


def latent_symmetries(scene: dict) -> list:
    """not_applicable: every label-influencing field (reference scope, each
    candidate scope, and the bound colors) is rendered verbatim, so the raster
    uniquely determines the gold; no pixel-identical transform exists."""
    return []
