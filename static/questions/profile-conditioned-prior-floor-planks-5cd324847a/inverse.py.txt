"""Pixel-only inverse arm: decide plank-width uniformity from the raster alone.

This module receives only the rendered PNG and never imports renderer,
prompts, generate, verify, or analytic-gold code, and it accepts no latent
scene data. It classifies each pixel to a fixed world palette by nearest
Euclidean color, reads the run lengths of plank-colored pixels along a
horizontal scanline through the planks, and reports whether every plank run is
the same width.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

# Fixed world palette (must match renderer constants).
PLANK_COLOR = (196, 160, 120)
GROUT_COLOR = (66, 54, 46)
BACKGROUND = (252, 252, 250)

_PALETTE = [PLANK_COLOR, GROUT_COLOR, BACKGROUND]
_PLANK_LABEL = 0


def _label_pixels(arr: np.ndarray) -> np.ndarray:
    dists = np.stack(
        [np.linalg.norm(arr - np.array(c, dtype=np.float32), axis=-1) for c in _PALETTE],
        axis=-1,
    )
    return dists.argmin(axis=-1)


def _plank_run_lengths(labels: np.ndarray):
    """Return the pixel widths of plank-colored runs on a middle scanline."""
    h, w = labels.shape
    y = h // 2
    row = labels[y]
    lengths: list[int] = []
    cur = 0
    for value in row:
        if value == _PLANK_LABEL:
            cur += 1
        else:
            if cur > 0:
                lengths.append(cur)
                cur = 0
    if cur > 0:
        lengths.append(cur)
    return lengths


def decision_from_image(image: Image.Image) -> str:
    """Return 'uniform' when all plank runs share one width, else 'nonuniform'."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label_pixels(arr)
    lengths = _plank_run_lengths(labels)
    if not lengths:
        return "uniform"
    widths = set(lengths)
    return "uniform" if len(widths) == 1 else "nonuniform"
