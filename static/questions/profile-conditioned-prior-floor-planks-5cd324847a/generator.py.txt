"""Prior-conflict plank-floor world: test whether all visible planks share one width.

The latent scene lays a set of vertical wooden floorboards (planks) across the
canvas. A familiar canonical prior asserts that a well-laid floor has planks of
uniform (equal) width. The counterfactual renders one plank whose width differs
from the rest, and the decision must be read from the measured widths of the
planks in the pixels, never from the prior. The pixel oracle compares the run
widths of plank-colored regions along a horizontal scanline.

Latent alias mode is ``not_applicable``: every latent field (the list of plank
widths and the canvas size) is directly reflected in the raster, so the
rendering is injective in the latent scene and there is no render-invariant
alias. The canonical-prior name "floor" is a matched-control/counterfactual
manipulation of the *question*, not of any latent field.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = (540, 360)
GR = 2  # grout line width in px between planks
PLANK_COLOR = (196, 160, 120)  # warm wood
GROUT_COLOR = (66, 54, 46)     # dark seam
BACKGROUND = (252, 252, 250)

# Off-quarantine width-control ranges. The count of planks, their base width,
# and the outlier position all vary independently of the decision so that no
# cheap positional/size/count cue predicts the answer.
N_RANGE = (5, 7)
BASE_RANGE = (38, 48)
DELTAS = (-12, 12)

MIN_PLANE_PX = 26.0   # a plank narrower than this is indistinguishable as a plank
MIN_DELTA_PX = 8.0    # a width gap smaller than this is ambiguous at pixel scale


def _fit_widths(n: int, base: int, uniform: bool, rng: random.Random):
    widths = [base] * n
    if not uniform:
        j = rng.randrange(n)
        while True:
            delta = DELTAS[rng.randrange(len(DELTAS))]
            candidate = base + delta
            if candidate >= MIN_PLANE_PX:
                widths[j] = int(candidate)
                break
    return [int(w) for w in widths]


def sample_scene(seed: int) -> dict:
    """Deterministic scene dict from an integer seed."""
    rng = random.Random(seed)
    n = rng.randint(*N_RANGE)
    base = rng.randint(*BASE_RANGE)
    uniform = rng.random() < 0.5
    widths = _fit_widths(n, base, uniform, rng)
    return {
        "widths": widths,
        "canvas": list(CANVAS),
    }


def render(scene: dict) -> Image.Image:
    """Render the plank floor to an RGB image."""
    w, h = scene["canvas"]
    img = Image.new("RGB", (int(w), int(h)), BACKGROUND)
    draw = ImageDraw.Draw(img)
    widths = [int(x) for x in scene["widths"]]
    x = 0
    for idx, pw in enumerate(widths):
        draw.rectangle([x, 0, x + pw - 1, h - 1], fill=PLANK_COLOR)
        x += pw
        if idx < len(widths) - 1:
            draw.rectangle([x, 0, x + GR - 1, h - 1], fill=GROUT_COLOR)
            x += GR
    return img


def analytic_gold(scene: dict) -> str:
    """Decision = whether all planks share one width."""
    widths = {int(x) for x in scene["widths"]}
    return "uniform" if len(widths) == 1 else "nonuniform"


def margin(scene: dict) -> float:
    """Verifiability margin = width spread, or the common width when uniform."""
    widths = [int(x) for x in scene["widths"]]
    spread = float(max(widths) - min(widths))
    if spread == 0:
        return float(widths[0])
    return spread


def is_quarantined(scene: dict) -> bool:
    """Quarantine ambiguous resolutions (too-thin planks or tiny width gaps)."""
    widths = [int(x) for x in scene["widths"]]
    if any(pw < MIN_PLANE_PX for pw in widths):
        return True
    spread = float(max(widths) - min(widths))
    if spread != 0 and spread < MIN_DELTA_PX:
        return True
    return False


def latent_symmetries(scene: dict):
    """not_applicable: empty, the latent rendering is injective."""
    return []
