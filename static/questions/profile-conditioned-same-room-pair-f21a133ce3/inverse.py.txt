"""Pixel-only inverse arm for the same-room world.

Receives one rendered PNG and nothing else: no scene, no renderer, no gold. It
segments the painted fences and the three coloured dots out of the raster,
labels the white complement with a run-based union-find (4-connectivity), and
reports which colour pair falls in one and the same white component.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

INK_LEVEL = 100          # every channel darker than this counts as painted fence
CHANNEL_LEAD = 55        # how far a dot's own channel must lead the others
MIN_DOT_PIXELS = 40


class _Union:
    def __init__(self) -> None:
        self.parent: list[int] = []

    def make(self) -> int:
        self.parent.append(len(self.parent))
        return len(self.parent) - 1

    def find(self, node: int) -> int:
        root = node
        while self.parent[root] != root:
            root = self.parent[root]
        while self.parent[node] != root:
            self.parent[node], node = root, self.parent[node]
        return root

    def join(self, a: int, b: int) -> None:
        ra, rb = self.find(a), self.find(b)
        if ra != rb:
            self.parent[max(ra, rb)] = min(ra, rb)


def _runs(row: np.ndarray) -> list[tuple[int, int]]:
    """Maximal [start, end) runs of True in a boolean row."""
    padded = np.concatenate(([False], row, [False]))
    edges = np.flatnonzero(padded[1:] != padded[:-1])
    return list(zip(edges[0::2].tolist(), edges[1::2].tolist()))


def _label_free_space(free: np.ndarray) -> tuple[np.ndarray, list]:
    """Label 4-connected components of the free (non-ink) mask, run by run."""
    height, width = free.shape
    labels = np.full((height, width), -1, dtype=np.int32)
    union = _Union()
    previous: list[tuple[int, int, int]] = []
    for y in range(height):
        current = []
        for start, end in _runs(free[y]):
            node = union.make()
            for pstart, pend, pnode in previous:
                if pstart < end and start < pend:
                    union.join(node, pnode)
            labels[y, start:end] = node
            current.append((start, end, node))
        previous = current
    return labels, union


def _dot_masks(array: np.ndarray) -> dict[str, np.ndarray]:
    red = array[:, :, 0].astype(np.int16)
    green = array[:, :, 1].astype(np.int16)
    blue = array[:, :, 2].astype(np.int16)
    return {
        "red": (red - green > CHANNEL_LEAD) & (red - blue > CHANNEL_LEAD),
        "green": (green - red > CHANNEL_LEAD) & (green - blue > CHANNEL_LEAD),
        "blue": (blue - red > CHANNEL_LEAD) & (blue - green > CHANNEL_LEAD),
    }


def decision_from_image(image: Image.Image) -> str:
    """Return 'red_green', 'red_blue', 'green_blue' or 'abstain'."""
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    ink = np.all(array < INK_LEVEL, axis=2)
    free = ~ink

    centres: dict[str, tuple[int, int]] = {}
    for color, mask in _dot_masks(array).items():
        ys, xs = np.nonzero(mask)
        if ys.size < MIN_DOT_PIXELS:
            return "abstain"
        centres[color] = (int(round(ys.mean())), int(round(xs.mean())))

    labels, union = _label_free_space(free)
    roots: dict[str, int] = {}
    for color, (y, x) in centres.items():
        if labels[y, x] < 0:
            return "abstain"  # a dot centroid landed on painted ink
        roots[color] = union.find(int(labels[y, x]))

    together = [
        f"{a}_{b}"
        for a, b in (("red", "green"), ("red", "blue"), ("green", "blue"))
        if roots[a] == roots[b]
    ]
    if len(together) != 1:
        return "abstain"
    return together[0]
