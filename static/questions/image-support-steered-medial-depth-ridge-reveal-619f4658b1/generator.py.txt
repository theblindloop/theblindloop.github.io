"""Deterministic latent sampler and Pillow renderer for the depth-ridge world."""

from __future__ import annotations

from collections import deque
import random

from PIL import Image, ImageDraw


N = 13
CANVAS = 512
ANSWERS = ("fork", "arch", "bolt")
TOTAL_INK = 84

_POINTS = {
    "fork": ((10, 6), (9, 6), (8, 6), (7, 6), (6, 6), (5, 6),
             (4, 5), (3, 4), (2, 3), (4, 7), (3, 8), (2, 9)),
    "arch": ((9, 3), (8, 3), (7, 3), (6, 4), (5, 5), (4, 6),
             (5, 7), (6, 8), (7, 9), (8, 9), (9, 9)),
    "bolt": ((2, 4), (3, 4), (4, 5), (5, 6), (6, 7), (7, 8),
             (8, 8), (9, 7), (10, 7)),
}

_INKS = ((38, 48, 62), (66, 43, 73), (38, 67, 55), (70, 48, 38))
_FRAME = (18, 157, 174)
_BACKGROUND = (247, 245, 238)
_PALE = (210, 207, 197)


def _dilate(points: tuple[tuple[int, int], ...], radius: int = 2) -> list[list[bool]]:
    return [
        [min(abs(y - py) + abs(x - px) for py, px in points) <= radius for x in range(N)]
        for y in range(N)
    ]


def _depth(cells: list[list[bool]]) -> list[list[int]]:
    """Breadth-first distance from visible light cells (the declared transform)."""
    distance = [[-1 if cells[y][x] else 0 for x in range(N)] for y in range(N)]
    queue = deque((y, x) for y in range(N) for x in range(N) if not cells[y][x])
    while queue:
        y, x = queue.popleft()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < N and 0 <= nx < N and distance[ny][nx] < 0:
                distance[ny][nx] = distance[y][x] + 1
                queue.append((ny, nx))
    return distance


def _ridge(cells: list[list[bool]]) -> list[list[bool]]:
    distance = _depth(cells)
    result = [[False] * N for _ in range(N)]
    for y in range(N):
        for x in range(N):
            if not cells[y][x]:
                continue
            neighborhood = (
                distance[ny][nx]
                for ny in range(max(0, y - 1), min(N, y + 2))
                for nx in range(max(0, x - 1), min(N, x + 2))
            )
            result[y][x] = distance[y][x] >= max(neighborhood)
    return result


_BASES = {name: _dilate(points) for name, points in _POINTS.items()}
_TEMPLATES = {name: _ridge(cells) for name, cells in _BASES.items()}


def _measure(cells: list[list[bool]]) -> tuple[str, int, dict[str, int]]:
    ridge = _ridge(cells)
    distances = {
        name: sum(
            ridge[y][x] != template[y][x]
            for y in range(N)
            for x in range(N)
        )
        for name, template in _TEMPLATES.items()
    }
    ordered = sorted(distances, key=lambda name: (distances[name], name))
    return ordered[0], distances[ordered[1]] - distances[ordered[0]], distances


def _connected(cells: list[list[bool]]) -> bool:
    points = {(y, x) for y in range(N) for x in range(N) if cells[y][x]}
    if not points:
        return False
    seen = {next(iter(points))}
    queue = list(seen)
    while queue:
        y, x = queue.pop()
        for point in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if point in points and point not in seen:
                seen.add(point)
                queue.append(point)
    return seen == points


def _camouflaged_cells(name: str, rng: random.Random) -> list[list[bool]]:
    cells = [row[:] for row in _BASES[name]]
    attempts = 0
    while sum(map(sum, cells)) < TOTAL_INK and attempts < 5000:
        attempts += 1
        candidates = []
        for y in range(N):
            for x in range(N):
                if cells[y][x]:
                    continue
                if any(
                    0 <= ny < N and 0 <= nx < N and cells[ny][nx]
                    for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1))
                ):
                    candidates.append((y, x))
        y, x = candidates[rng.randrange(len(candidates))]
        cells[y][x] = True
        decision, gap, _ = _measure(cells)
        if decision != name or gap < 3:
            cells[y][x] = False
    if sum(map(sum, cells)) != TOTAL_INK or not _connected(cells):
        raise RuntimeError("could not form a balanced connected island")
    return cells


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    name = ANSWERS[int(seed) % len(ANSWERS)]
    cells = _camouflaged_cells(name, rng)
    cell_size = rng.choice((5, 6))
    extent = N * cell_size
    outer = extent + 6
    frame_x = rng.randint(42, CANVAS - outer - 42) + 3
    frame_y = rng.randint(42, CANVAS - outer - 42) + 3
    distractors = []
    while len(distractors) < 8:
        x, y = rng.randint(24, CANVAS - 25), rng.randint(24, CANVAS - 25)
        if not (frame_x - 28 <= x <= frame_x + extent + 28 and
                frame_y - 28 <= y <= frame_y + extent + 28):
            distractors.append([x, y, rng.choice((2, 3, 4))])
    return {
        "cells": ["".join("1" if value else "0" for value in row) for row in cells],
        "frame_x": frame_x,
        "frame_y": frame_y,
        "cell_size": cell_size,
        "dark_rgb": list(rng.choice(_INKS)),
        "frame_rgb": list(_FRAME),
        "background_rgb": list(_BACKGROUND),
        "distractors": distractors,
    }


def _scene_cells(scene: dict) -> list[list[bool]]:
    rows = scene["cells"]
    if len(rows) != N or any(len(row) != N or set(row) - {"0", "1"} for row in rows):
        raise ValueError("cells must be a 13 by 13 binary mosaic")
    return [[value == "1" for value in row] for row in rows]


def render(scene: dict) -> Image.Image:
    cells = _scene_cells(scene)
    frame_x = int(scene["frame_x"])
    frame_y = int(scene["frame_y"])
    cell_size = int(scene["cell_size"])
    dark_rgb = tuple(int(value) for value in scene["dark_rgb"])
    frame_rgb = tuple(int(value) for value in scene["frame_rgb"])
    background_rgb = tuple(int(value) for value in scene["background_rgb"])
    distractors = scene["distractors"]
    if cell_size not in (5, 6):
        raise ValueError("unsupported cell size")
    image = Image.new("RGB", (CANVAS, CANVAS), background_rgb)
    draw = ImageDraw.Draw(image)
    for dx, dy, radius in distractors:
        draw.ellipse((dx - radius, dy - radius, dx + radius, dy + radius), outline=_PALE, width=1)
    extent = N * cell_size
    for y in range(N):
        for x in range(N):
            if cells[y][x]:
                x0, y0 = frame_x + x * cell_size, frame_y + y * cell_size
                draw.rectangle((x0, y0, x0 + cell_size - 1, y0 + cell_size - 1), fill=dark_rgb)
    draw.rectangle(
        (frame_x - 3, frame_y - 3, frame_x + extent + 2, frame_y + extent + 2),
        outline=frame_rgb,
        width=3,
    )
    # One-pixel lattice points expose the cell scale without severing the ink island.
    for y in range(1, N):
        for x in range(1, N):
            draw.point((frame_x + x * cell_size, frame_y + y * cell_size), fill=_PALE)
    return image


def analytic_gold(scene: dict) -> str:
    decision, _, _ = _measure(_scene_cells({"cells": scene["cells"]}))
    return decision


def margin(scene: dict) -> float:
    _, gap, _ = _measure(_scene_cells({"cells": scene["cells"]}))
    return float(gap)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
