"""Deterministic scene sampler, analytic replay, and Pillow renderer."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 640
STATES = ("circle", "square", "triangle")
PALETTES = (
    ((44, 125, 210), (235, 145, 38)),
    ((28, 160, 140), (155, 95, 190)),
)
DARK = (33, 43, 54)
GRAY = (118, 128, 138)
LIGHT = (225, 230, 235)
MAGENTA = (205, 55, 145)
WHITE = (255, 255, 255)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    transitions = [[rng.randrange(3), rng.randrange(3)] for _ in STATES]
    # Require a nonconstant table and make both inputs visibly necessary.
    while len({value for row in transitions for value in row}) < 2:
        transitions = [[rng.randrange(3), rng.randrange(3)] for _ in STATES]
    inputs = [0, 1] + [rng.randrange(2) for _ in range(4)]
    rng.shuffle(inputs)
    row_order = list(range(3))
    rng.shuffle(row_order)
    return {
        "start_state": rng.randrange(3),
        "transitions": transitions,
        "inputs": inputs,
        "row_order": row_order,
        "palette_variant": rng.randrange(len(PALETTES)),
        "layout_dx": rng.randint(-8, 8),
        "layout_dy": rng.randint(-6, 6),
        "nonce": rng.randrange(1_000_000_000),
    }


def _replay(start: int, transitions: list[list[int]], inputs: list[int]) -> int:
    state = int(start)
    for value in inputs:
        state = int(transitions[state][int(value)])
    return state


def analytic_gold(scene: dict) -> str:
    return STATES[_replay(scene["start_state"], scene["transitions"], scene["inputs"])]


def margin(scene: dict) -> float:
    # The complete table defines one legal successor at each step and no runner-up.
    # 1.0 is the discrete one-versus-zero consistency margin.
    _ = (scene["start_state"], scene["transitions"], scene["inputs"])
    return 1.0


def is_quarantined(scene: dict) -> bool:
    valid_states = set(range(3))
    try:
        return not (
            scene["start_state"] in valid_states
            and len(scene["transitions"]) == 3
            and all(len(row) == 2 and set(row) <= valid_states for row in scene["transitions"])
            and len(scene["inputs"]) == 6
            and set(scene["inputs"]) == {0, 1}
        )
    except (KeyError, TypeError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["nonce"] = (int(twin["nonce"]) + 1) % 1_000_000_000
    return [("unused_nonce_change", twin)]


def _shape(draw: ImageDraw.ImageDraw, state: int, cx: int, cy: int, radius: int = 24) -> None:
    if state == 0:
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=DARK)
    elif state == 1:
        draw.rectangle((cx - radius, cy - radius, cx + radius, cy + radius), fill=DARK)
    else:
        draw.polygon(
            [(cx, cy - radius - 2), (cx - radius - 2, cy + radius), (cx + radius + 2, cy + radius)],
            fill=DARK,
        )


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), WHITE)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    dx, dy = int(scene["layout_dx"]), int(scene["layout_dy"])
    palette = PALETTES[int(scene["palette_variant"])]
    row_y = [190 + dy, 310 + dy, 430 + dy]
    start_x, row_x, col_x = 100 + dx, 285 + dx, [515 + dx, 720 + dx]

    draw.text((54 + dx, 112 + dy), "START", fill=GRAY, font=font)
    draw.ellipse((start_x - 40, row_y[1] - 40, start_x + 40, row_y[1] + 40), outline=MAGENTA, width=7)
    _shape(draw, int(scene["start_state"]), start_x, row_y[1])

    # Visible table frame, headers, and row-key/output icons.
    draw.rounded_rectangle((215 + dx, 92 + dy, 805 + dx, 485 + dy), radius=12, outline=LIGHT, width=5)
    draw.text((242 + dx, 110 + dy), "CURRENT", fill=GRAY, font=font)
    for col, cx in enumerate(col_x):
        draw.rounded_rectangle((cx - 34, 108 + dy, cx + 34, 150 + dy), radius=8, fill=palette[col])
    for index in range(1, 3):
        y = (row_y[index - 1] + row_y[index]) // 2
        draw.line((220 + dx, y, 800 + dx, y), fill=LIGHT, width=4)
    draw.line((392 + dx, 100 + dy, 392 + dx, 478 + dy), fill=LIGHT, width=4)
    draw.line((617 + dx, 100 + dy, 617 + dx, 478 + dy), fill=LIGHT, width=4)
    for visual_row, state in enumerate(scene["row_order"]):
        y = row_y[visual_row]
        _shape(draw, int(state), row_x, y)
        _shape(draw, int(scene["transitions"][state][0]), col_x[0], y)
        _shape(draw, int(scene["transitions"][state][1]), col_x[1], y)

    draw.text((58 + dx, 515 + dy), "INPUTS", fill=GRAY, font=font)
    card_x = [245, 345, 445, 545, 645, 745]
    for cx, value in zip(card_x, scene["inputs"]):
        cx += dx
        draw.rounded_rectangle((cx - 34, 525 + dy, cx + 34, 575 + dy), radius=9, fill=palette[int(value)])
    # Small neutral arrows make order explicit without entering oracle masks.
    for left, right in zip(card_x, card_x[1:]):
        x1, x2, y = left + 38 + dx, right - 38 + dx, 550 + dy
        draw.line((x1, y, x2, y), fill=GRAY, width=3)
        draw.polygon([(x2, y), (x2 - 7, y - 5), (x2 - 7, y + 5)], fill=GRAY)
    return image
