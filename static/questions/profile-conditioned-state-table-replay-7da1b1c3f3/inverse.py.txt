"""Independent inverse arm: accepts only final raster pixels."""

from __future__ import annotations

import numpy as np
from PIL import Image

DARK = np.array((33, 43, 54), dtype=np.uint8)
STATES = ("circle", "square", "triangle")


def _runs(values: np.ndarray, minimum: int = 1) -> list[tuple[int, int]]:
    indices = np.flatnonzero(values)
    if len(indices) == 0:
        return []
    out = []
    start = prior = int(indices[0])
    for raw in indices[1:]:
        value = int(raw)
        if value > prior + 1:
            if prior - start + 1 >= minimum:
                out.append((start, prior))
            start = value
        prior = value
    if prior - start + 1 >= minimum:
        out.append((start, prior))
    return out


def _classify(mask: np.ndarray) -> int:
    ys, xs = np.nonzero(mask)
    if len(xs) < 500:
        raise ValueError("shape ink missing")
    crop = mask[ys.min() : ys.max() + 1, xs.min() : xs.max() + 1]
    fill = float(crop.mean())
    if fill > 0.92:
        return 1
    if fill > 0.68:
        return 0
    if fill > 0.42:
        return 2
    raise ValueError("unrecognized shape")


def _shapes(arr: np.ndarray) -> list[list[int]]:
    mask = np.all(arr == DARK, axis=2)
    x_runs = _runs(mask.any(axis=0), minimum=35)
    if len(x_runs) != 4:
        raise ValueError(f"expected four discovered shape columns, got {len(x_runs)}")
    columns: list[list[int]] = []
    for x0, x1 in x_runs:
        strip = mask[:, x0 : x1 + 1]
        y_runs = _runs(strip.any(axis=1), minimum=35)
        columns.append([_classify(mask[y0 : y1 + 1, x0 : x1 + 1]) for y0, y1 in y_runs])
    if [len(column) for column in columns] != [1, 3, 3, 3]:
        raise ValueError("shape table is incomplete or ambiguous")
    return columns


def _input_sequence(arr: np.ndarray) -> list[int]:
    # Saturated large color regions are precisely the two headers and six cards.
    pixels = arr.reshape(-1, 3)
    colors, counts = np.unique(pixels, axis=0, return_counts=True)
    usable = []
    for color, count in zip(colors, counts):
        spread = int(color.max()) - int(color.min())
        if count > 700 and spread > 55 and not (int(color[0]) > 170 and int(color[2]) > 100):
            usable.append(tuple(int(v) for v in color))
    if len(usable) != 2:
        raise ValueError(f"expected two input colors, got {len(usable)}")

    headers = []
    cards = []
    height = arr.shape[0]
    for color in usable:
        mask = np.all(arr == np.array(color, dtype=np.uint8), axis=2)
        top = mask.copy()
        top[height // 2 :, :] = False
        ys, xs = np.nonzero(top)
        if len(xs) < 500:
            raise ValueError("input header missing")
        headers.append((float(xs.mean()), color))
        bottom = mask.copy()
        bottom[: height // 2, :] = False
        for x0, x1 in _runs(bottom.any(axis=0), minimum=50):
            yy, xx = np.nonzero(bottom[:, x0 : x1 + 1])
            if len(xx) > 700:
                cards.append((float(x0 + xx.mean()), color))
    if len(cards) != 6:
        raise ValueError(f"expected six input cards, got {len(cards)}")
    color_to_column = {color: index for index, (_, color) in enumerate(sorted(headers))}
    return [color_to_column[color] for _, color in sorted(cards)]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    columns = _shapes(arr)
    start = columns[0][0]
    row_keys, left_outputs, right_outputs = columns[1:]
    transitions: dict[tuple[int, int], int] = {}
    for row_index, key in enumerate(row_keys):
        if key in [known[0] for known in transitions]:
            raise ValueError("duplicate row identity")
        transitions[(key, 0)] = left_outputs[row_index]
        transitions[(key, 1)] = right_outputs[row_index]
    if set(row_keys) != {0, 1, 2} or len(transitions) != 6:
        raise ValueError("transition table does not define every state")
    state = start
    for value in _input_sequence(arr):
        state = transitions[(state, value)]
    return STATES[state]
