"""Independent pixel-only inverse arm for loop_enclosure.

Recovers the closed-loop / open decision from the raster alone: it counts the
enclosed white regions (holes) of the single connected black loop by 4-connected
region labelling of the background complement of the black ink.

The black loop is the only drawn object; the number of black connected
components is 1, and the decision is simply whether that loop encloses exactly
one interior white region. This module imports neither the renderer nor any
scene data; it reads only the pixels.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _mask_white(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    return (arr[:, :, 0] > 128) & (arr[:, :, 1] > 128) & (arr[:, :, 2] > 128)


def _flood(size: tuple[int, int], starts, allow) -> tuple[np.ndarray, int]:
    """4-connected flood fill from a list of (r, c) start pixels.

    Expands only through pixels where ``allow`` (a flat boolean mask) is True.
    Returns (visited_mask, visited_count). Pixels are enumerated in row-major
    order as integer indices so the deque stores plain ints (fast).
    """
    h, w = size
    visited = np.zeros(h * w, dtype=bool)
    total = 0
    frontier = deque()
    for r, c in starts:
        idx = r * w + c
        if allow[idx] and not visited[idx]:
            visited[idx] = True
            total += 1
            frontier.append(idx)
    while frontier:
        idx = frontier.popleft()
        r, c = divmod(idx, w)
        if r > 0:
            ni = idx - w
            if allow[ni] and not visited[ni]:
                visited[ni] = True
                total += 1
                frontier.append(ni)
        if r < h - 1:
            ni = idx + w
            if allow[ni] and not visited[ni]:
                visited[ni] = True
                total += 1
                frontier.append(ni)
        if c > 0:
            ni = idx - 1
            if allow[ni] and not visited[ni]:
                visited[ni] = True
                total += 1
                frontier.append(ni)
        if c < w - 1:
            ni = idx + 1
            if allow[ni] and not visited[ni]:
                visited[ni] = True
                total += 1
                frontier.append(ni)
    return visited, total


def _count_components(size: tuple[int, int], mask_flat: np.ndarray) -> int:
    """Count 4-connected components among the True pixels of mask_flat."""
    h, w = size
    unseen = np.nonzero(mask_flat)[0]
    labels = np.zeros(h * w, dtype=bool)  # component seen
    count = 0
    for start_idx in unseen:
        if labels[start_idx]:
            continue
        count += 1
        labels[start_idx] = True
        frontier = deque([start_idx])
        while frontier:
            idx = frontier.popleft()
            r, c = divmod(idx, w)
            if r > 0:
                ni = idx - w
                if mask_flat[ni] and not labels[ni]:
                    labels[ni] = True
                    frontier.append(ni)
            if r < h - 1:
                ni = idx + w
                if mask_flat[ni] and not labels[ni]:
                    labels[ni] = True
                    frontier.append(ni)
            if c > 0:
                ni = idx - 1
                if mask_flat[ni] and not labels[ni]:
                    labels[ni] = True
                    frontier.append(ni)
            if c < w - 1:
                ni = idx + 1
                if mask_flat[ni] and not labels[ni]:
                    labels[ni] = True
                    frontier.append(ni)
    return count


def decision_from_image(image: Image.Image) -> str:
    """Return 'closed' if the black loop encloses exactly one white region."""
    white = _mask_white(image)
    h, w = white.shape
    white_flat = white.reshape(-1)

    border_starts = []
    for c in range(w):
        if white[0, c]:
            border_starts.append((0, c))
        if white[h - 1, c]:
            border_starts.append((h - 1, c))
    for r in range(1, h - 1):
        if white[r, 0]:
            border_starts.append((r, 0))
        if white[r, w - 1]:
            border_starts.append((r, w - 1))

    exterior, _ = _flood((h, w), border_starts, white_flat)
    interior_white_flat = white_flat & ~exterior

    n_holes = _count_components((h, w), interior_white_flat)
    return "closed" if n_holes == 1 else "open"
