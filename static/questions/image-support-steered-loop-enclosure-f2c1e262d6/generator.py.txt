"""Single thick connected black ring, closed-loop hole-enclosure world.

Latent scene z describes one thick black band drawn as an annulus. When the band
is a full closed loop it encloses exactly one connected white region (a hole);
the interior and exterior white are disconnected. When the band carries a single
radial gap that fully breaks the thickness, the interior fuses with the exterior
(no hole) while the black band remains one connected area-like component.

The decision is recovered purely from the raster by counting enclosed white
regions (holes) of the single connected black component.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS = (512, 512)
W, H = CANVAS
PAD = 18.0

# Geometry constraints guaranteeing clean, unambiguously resolvable rasters.
OUTER_MIN, OUTER_MAX = 150.0, 188.0
THICK_MIN, THICK_MAX = 34.0, 56.0
GAP_WIDTH_DEG_MIN, GAP_WIDTH_DEG_MAX = 10.0, 26.0
MIN_GAP_LINEAR_PX = 16.0  # linear width of the gap at the band midline


def sample_scene(seed: int) -> dict:
    """Deterministically sample one scene dict from a seed (no external state)."""
    rng = random.Random(seed)
    sealed = rng.random() < 0.5
    # Center chosen so the whole outer circle (and its interior hole) fits on the
    # canvas with clearance, keeping the decisive support spread across the image.
    max_center = min(W / 2, (W - PAD - OUTER_MAX))
    min_center = max(W / 2, (PAD + OUTER_MAX))
    cx = rng.uniform(min_center, max_center)
    cy = rng.uniform(min_center, max_center)
    center = (cx, cy)
    outer = rng.uniform(OUTER_MIN, OUTER_MAX)
    thickness = rng.uniform(THICK_MIN, THICK_MAX)
    # For open scenes choose a gap wide enough to break the band at pixels.
    gap_width_deg = rng.uniform(GAP_WIDTH_DEG_MIN, GAP_WIDTH_DEG_MAX)
    gap_dir = rng.uniform(0.0, 360.0)
    return {
        "center": center,
        "outer": outer,
        "thickness": thickness,
        "sealed": sealed,
        "gap_width_deg": gap_width_deg,
        "gap_dir": gap_dir,
    }


def _gap_linear_width(scene: dict) -> float:
    """Arc width (px) of the gap measured at the band midline radius."""
    mid_radius = scene["outer"] - scene["thickness"] / 2.0
    return mid_radius * math.radians(scene["gap_width_deg"])


def render(scene: dict) -> Image.Image:
    """Render the raster I = R(z). Deterministic, no noise, no text."""
    img = Image.new("RGB", CANVAS, "white")
    d = ImageDraw.Draw(img)
    cx, cy = scene["center"]
    outer = scene["outer"]
    thickness = scene["thickness"]
    r = outer - thickness

    box = [cx - outer, cy - outer, cx + outer, cy + outer]
    d.ellipse(box, fill="black")
    ibox = [cx - r, cy - r, cx + r, cy + r]
    d.ellipse(ibox, fill="white")

    if not scene["sealed"]:
        gap_rad = math.radians(scene["gap_width_deg"])
        ga = math.radians(scene["gap_dir"])
        far = outer + 40.0
        a0 = ga - gap_rad / 2.0
        a1 = ga + gap_rad / 2.0
        pts = [
            (cx, cy),
            (cx + far * math.cos(a0), cy + far * math.sin(a0)),
            (cx + far * math.cos(a1), cy + far * math.sin(a1)),
        ]
        d.polygon(pts, fill="white")
    return img


def analytic_gold(scene: dict) -> str:
    """The scene's decision: 'closed' iff the ring is a sealed complete loop."""
    return "closed" if scene["sealed"] else "open"


def margin(scene: dict) -> float:
    """Robustness of the decision: a positive clearance in pixels.

    For a sealed ring the hole is robust to a thickness-scale perturbation; for
    an open ring the gap must stay wide enough to remain clearly open.
    """
    if scene["sealed"]:
        return float(scene["thickness"])
    return _gap_linear_width(scene)


def is_quarantined(scene: dict) -> bool:
    """Exclude pixel-ambiguous boundary cases before they reach headlines."""
    if margin(scene) < MIN_GAP_LINEAR_PX:
        return True
    if scene["thickness"] < THICK_MIN - 1e-9:
        return True
    return False


def latent_symmetries(scene: dict) -> list:
    """Declared empty: the raster fully determines the gold (not_applicable)."""
    return []
