"""Independent image-only inverse arm for the six-lobe waveform."""
from __future__ import annotations

import numpy as np
from PIL import Image


def _largest_component(mask: np.ndarray) -> tuple[np.ndarray, int]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    best: list[tuple[int, int]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        component = []
        while stack:
            cy, cx = stack.pop()
            component.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(component) > len(best):
            best = component
    output = np.zeros_like(mask, dtype=bool)
    if best:
        yy, xx = zip(*best)
        output[np.array(yy), np.array(xx)] = True
    return output, len(best)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    r, g, b = arr[..., 0], arr[..., 1], arr[..., 2]
    blue = (b - r > 65) & (b - g > 20) & (b > 125)
    component, area = _largest_component(blue)
    if area < 1800:
        raise ValueError("wave trace missing or too small")
    ys, xs = np.nonzero(component)
    xmin, xmax = int(xs.min()), int(xs.max())
    if xmax - xmin < int(image.width * 0.79):
        raise ValueError("wave trace lacks required distributed span")
    covered = np.unique(xs)
    if len(covered) < int(0.94 * (xmax - xmin + 1)):
        raise ValueError("wave trace is horizontally broken")

    neutral = (np.max(arr, axis=2) - np.min(arr, axis=2) <= 16) & (r >= 105) & (r <= 165)
    gy, gx = np.nonzero(neutral)
    if len(gy) < 150:
        raise ValueError("gray centerline missing")
    # Text is dark and excluded; the dashed line dominates neutral pixels and is horizontal.
    counts = np.bincount(gy, minlength=image.height)
    baseline = int(np.argmax(counts))
    if counts[baseline] < 80:
        raise ValueError("gray centerline is not recoverable")

    span = (xmax - xmin) / 6.0
    signs = []
    for index in range(6):
        left = xmin + span * (index + 0.20)
        right = xmin + span * (index + 0.80)
        local_y = ys[(xs >= left) & (xs <= right)]
        if len(local_y) < 130:
            raise ValueError("one of six lobes is incomplete")
        offset = baseline - float(np.median(local_y))
        if abs(offset) < 18:
            raise ValueError("lobe side is ambiguous")
        signs.append(1 if offset > 0 else -1)
    return "yes" if all(a != b for a, b in zip(signs, signs[1:])) else "no"
