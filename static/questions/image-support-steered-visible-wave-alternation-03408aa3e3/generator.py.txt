"""Deterministic renderer and analytic arm for visible wave alternation."""
from __future__ import annotations

import math
import random
from PIL import Image, ImageDraw, ImageFont

PALETTES = [(24, 96, 184), (28, 112, 169), (45, 88, 190)]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    alternating = (int(seed) % 2 == 0)
    start = rng.choice((-1, 1))
    if alternating:
        signs = [start * (-1 if i % 2 else 1) for i in range(6)]
    else:
        balanced = []
        for bits in range(64):
            seq = [1 if bits & (1 << i) else -1 for i in range(6)]
            if sum(seq) == 0 and any(seq[i] == seq[i + 1] for i in range(5)):
                balanced.append(seq)
        signs = list(rng.choice(balanced))
    return {
        "width": 640,
        "height": 360,
        "x0": rng.randint(42, 58),
        "x1": rng.randint(582, 598),
        "baseline": rng.randint(177, 193),
        "amplitude": rng.randint(50, 72),
        "stroke": rng.choice((6, 7, 8)),
        "signs": signs,
        "palette_index": rng.randrange(len(PALETTES)),
    }


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    x0, x1 = float(scene["x0"]), float(scene["x1"])
    baseline, amplitude = float(scene["baseline"]), float(scene["amplitude"])
    stroke, signs = int(scene["stroke"]), list(scene["signs"])
    blue = PALETTES[int(scene["palette_index"])]
    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), (250, 248, 241))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    draw.text((25 * scale, 20 * scale), "SINE WAVE STUDY", fill=(55, 59, 65), font=font, stroke_width=0)
    gray = (132, 136, 142)
    for left in range(int(x0), int(x1), 22):
        draw.line(((left * scale, baseline * scale), (min(left + 13, int(x1)) * scale, baseline * scale)), fill=gray, width=2 * scale)
    points = []
    span = (x1 - x0) / 6.0
    for index, sign in enumerate(signs):
        for step in range(41):
            if index and step == 0:
                continue
            t = step / 40.0
            x = x0 + span * (index + t)
            y = baseline - float(sign) * amplitude * math.sin(math.pi * t)
            points.append((round(x * scale), round(y * scale)))
    draw.line(points, fill=blue, width=stroke * scale, joint="curve")
    for point in (points[0], points[-1]):
        r = stroke * scale / 2
        draw.ellipse((point[0] - r, point[1] - r, point[0] + r, point[1] + r), fill=blue)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    signs = list(scene["signs"])
    return "yes" if all(a != b for a, b in zip(signs, signs[1:])) else "no"


def margin(scene: dict) -> float:
    return float(scene["amplitude"]) - 38.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
