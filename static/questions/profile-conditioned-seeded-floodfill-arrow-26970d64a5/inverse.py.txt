"""Independent image-only inverse for the seeded flood-fill world."""

from __future__ import annotations

from collections import deque
from dataclasses import dataclass

from PIL import Image

GRID_N = 17
FRAME_INSET = 4
MIN_GAP = 2.0 / 49.0

_TEMPLATE_ROWS = {
    "up": (
        "...#...", "..###..", ".#####.", "#######", "..###..", "..###..", "..###..",
    ),
    "right": (
        "...#...", "...##..", "######.", "#######", "######.", "...##..", "...#...",
    ),
    "down": (
        "..###..", "..###..", "..###..", "#######", ".#####.", "..###..", "...#...",
    ),
    "left": (
        "...#...", "..##...", ".######", "#######", ".######", "..##...", "...#...",
    ),
}
_TEMPLATES = {
    name: {(r, c) for r, row in enumerate(rows) for c, value in enumerate(row) if value == "#"}
    for name, rows in _TEMPLATE_ROWS.items()
}


class PixelAbstention(ValueError):
    """Raised when visible fiducials do not support an unambiguous decision."""


@dataclass(frozen=True)
class PixelDecision:
    decision: str
    margin: float
    component_size: int
    start_cell: tuple[int, int]


def _is_cyan(pixel: tuple[int, int, int]) -> bool:
    r, g, b = pixel
    return r < 70 and g > 120 and b > 145 and b > g


def _is_gold(pixel: tuple[int, int, int]) -> bool:
    r, g, b = pixel
    return r > 205 and 120 < g < 215 and b < 105


def _is_open(pixel: tuple[int, int, int]) -> bool:
    r, g, b = pixel
    return (r + g + b) > 420


def _frame_geometry(image: Image.Image) -> tuple[int, int, int]:
    pixels = image.load()
    cyan = [
        (x, y)
        for y in range(image.height)
        for x in range(image.width)
        if _is_cyan(pixels[x, y])
    ]
    if not cyan:
        raise PixelAbstention("missing cyan frame")
    xs = [point[0] for point in cyan]
    ys = [point[1] for point in cyan]
    left, right = min(xs), max(xs)
    top, bottom = min(ys), max(ys)
    inner_w = right - left + 1 - 2 * FRAME_INSET
    inner_h = bottom - top + 1 - 2 * FRAME_INSET
    if inner_w != inner_h or inner_w <= 0 or inner_w % GRID_N:
        raise PixelAbstention("cyan frame does not define a square 17-cell lattice")
    cell = inner_w // GRID_N
    if not 20 <= cell <= 30:
        raise PixelAbstention("unsupported visible cell scale")
    return left + FRAME_INSET, top + FRAME_INSET, cell


def _gold_centroid(image: Image.Image) -> tuple[float, float]:
    pixels = image.load()
    points = [
        (x, y)
        for y in range(image.height)
        for x in range(image.width)
        if _is_gold(pixels[x, y])
    ]
    if len(points) < 25:
        raise PixelAbstention("missing gold start marker")
    return (
        sum(x for x, _ in points) / len(points),
        sum(y for _, y in points) / len(points),
    )


def _flood(open_cells: set[tuple[int, int]], start: tuple[int, int]) -> set[tuple[int, int]]:
    if start not in open_cells:
        raise PixelAbstention("gold marker is not in a pale cell")
    reached = {start}
    queue = deque([start])
    while queue:
        r, c = queue.popleft()
        for point in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
            if point in open_cells and point not in reached:
                reached.add(point)
                queue.append(point)
    return reached


def decision_with_margin(image: Image.Image) -> PixelDecision:
    raster = image.convert("RGB")
    x0, y0, cell = _frame_geometry(raster)
    cx, cy = _gold_centroid(raster)
    start_col = int((cx - x0) // cell)
    start_row = int((cy - y0) // cell)
    if not (0 <= start_row < GRID_N and 0 <= start_col < GRID_N):
        raise PixelAbstention("gold marker lies outside the framed grid")

    pixels = raster.load()
    open_cells: set[tuple[int, int]] = set()
    for row in range(GRID_N):
        for col in range(GRID_N):
            x = x0 + col * cell + cell // 2
            y = y0 + row * cell + cell // 2
            if _is_open(pixels[x, y]) or (row, col) == (start_row, start_col):
                open_cells.add((row, col))
    reached = _flood(open_cells, (start_row, start_col))
    if not 23 <= len(reached) <= 27:
        raise PixelAbstention(f"unexpected filled-component size {len(reached)}")
    min_r = min(r for r, _ in reached)
    max_r = max(r for r, _ in reached)
    min_c = min(c for _, c in reached)
    max_c = max(c for _, c in reached)
    if (max_r - min_r, max_c - min_c) != (6, 6):
        raise PixelAbstention("filled component lacks a visible 7-by-7 arrow extent")
    normalized = {(r - min_r, c - min_c) for r, c in reached}
    scores = {
        name: 1.0 - len(normalized.symmetric_difference(template)) / 49.0
        for name, template in _TEMPLATES.items()
    }
    ranked = sorted(scores.items(), key=lambda item: (-item[1], item[0]))
    gap = ranked[0][1] - ranked[1][1]
    if gap <= MIN_GAP:
        raise PixelAbstention(f"arrow decision is ambiguous; top gap={gap:.6f}")
    return PixelDecision(ranked[0][0], gap, len(reached), (start_row, start_col))


def decision_from_image(image: Image.Image) -> str:
    """Return the arrow direction using only raster pixels."""
    return decision_with_margin(image).decision
