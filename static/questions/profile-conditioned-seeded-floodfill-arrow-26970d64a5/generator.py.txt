"""Deterministic raster world for a visibly declared four-neighbor flood fill."""

from __future__ import annotations

import random
from collections import deque

from PIL import Image, ImageDraw

CANVAS = 560
GRID_N = 17
DIRECTIONS = ("up", "right", "down", "left")
SLOT_ORIGINS = ((1, 1), (1, 9), (9, 1), (9, 9))
QUARANTINE_GAP = 2.0 / 49.0

_UP_ROWS = (
    "...#...",
    "..###..",
    ".#####.",
    "#######",
    "..###..",
    "..###..",
    "..###..",
)


def _cells(rows: tuple[str, ...]) -> set[tuple[int, int]]:
    return {(r, c) for r, row in enumerate(rows) for c, value in enumerate(row) if value == "#"}


def _rotate_clockwise(cells: set[tuple[int, int]]) -> set[tuple[int, int]]:
    return {(c, 6 - r) for r, c in cells}


_up = _cells(_UP_ROWS)
_right = _rotate_clockwise(_up)
_down = _rotate_clockwise(_right)
_left = _rotate_clockwise(_down)
TEMPLATES = dict(zip(DIRECTIONS, (_up, _right, _down, _left)))


def _connected(cells: set[tuple[int, int]]) -> bool:
    if not cells:
        return False
    seen = {next(iter(cells))}
    queue = deque(seen)
    while queue:
        r, c = queue.popleft()
        for point in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
            if point in cells and point not in seen:
                seen.add(point)
                queue.append(point)
    return seen == cells


def _scores(cells: set[tuple[int, int]]) -> dict[str, float]:
    return {
        direction: 1.0 - len(cells.symmetric_difference(template)) / 49.0
        for direction, template in TEMPLATES.items()
    }


def _decision_and_gap(cells: set[tuple[int, int]]) -> tuple[str, float]:
    ranked = sorted(_scores(cells).items(), key=lambda item: (-item[1], item[0]))
    if ranked[0][1] == ranked[1][1]:
        raise ValueError("arrow template tie")
    return ranked[0][0], ranked[0][1] - ranked[1][1]


def _minor_equal_ink_variant(
    direction: str, rng: random.Random, enabled: bool
) -> set[tuple[int, int]]:
    base = set(TEMPLATES[direction])
    if not enabled:
        return base
    removals = list(base)
    rng.shuffle(removals)
    for removed in removals:
        remainder = base - {removed}
        if not _connected(remainder):
            continue
        additions = {
            (nr, nc)
            for r, c in remainder
            for nr, nc in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1))
            if 0 <= nr < 7 and 0 <= nc < 7 and (nr, nc) not in base
        }
        additions = list(additions)
        rng.shuffle(additions)
        for added in additions:
            trial = remainder | {added}
            rows = [r for r, _ in trial]
            cols = [c for _, c in trial]
            if (min(rows), max(rows), min(cols), max(cols)) != (0, 6, 0, 6):
                continue
            winner, gap = _decision_and_gap(trial)
            if winner == direction and gap > QUARANTINE_GAP:
                return trial
    raise RuntimeError("could not construct a safe equal-ink arrow variant")


def sample_scene(seed: int) -> dict:
    """Sample one balanced scene from an integer seed."""
    seed = int(seed)
    answer_index = seed % 4
    block = seed // 4
    desired = DIRECTIONS[answer_index]
    target_slot = block % 4
    order_rng = random.Random(f"component-order:{seed}")
    remaining = [value for value in DIRECTIONS if value != desired]
    order_rng.shuffle(remaining)
    order: list[str] = []
    for slot in range(4):
        order.append(desired if slot == target_slot else remaining.pop())

    # Every consecutive group of four scenes contains all four answers under
    # exactly the same nuisance settings.  The variant is built once for "up"
    # and rotated, so all four visible arrows have equal ink and equal margins.
    layout_rng = random.Random(f"layout-block:{block}")
    use_variant = (block % 2) == 1
    base_variant = _minor_equal_ink_variant("up", layout_rng, use_variant)
    rotated_variants: dict[str, set[tuple[int, int]]] = {}
    current = base_variant
    for direction in DIRECTIONS:
        rotated_variants[direction] = current
        current = _rotate_clockwise(current)
    components: list[list[list[int]]] = []
    for direction in order:
        component = rotated_variants[direction]
        components.append([[r, c] for r, c in sorted(component)])

    target_cells = components[target_slot]
    marker_local = list(target_cells[layout_rng.randrange(len(target_cells))])
    wall_palette = (
        "#20324A",
        "#49304F",
        "#24483E",
        "#543A2D",
    )
    cell_size = (23, 24, 25)[block % 3]
    grid_pixels = GRID_N * cell_size
    origin_x = (CANVAS - grid_pixels) // 2 + layout_rng.randrange(-9, 10)
    origin_y = (CANVAS - grid_pixels) // 2 + layout_rng.randrange(-9, 10)
    return {
        "width": CANVAS,
        "height": CANVAS,
        "grid_n": GRID_N,
        "cell_size": cell_size,
        "origin_x": origin_x,
        "origin_y": origin_y,
        "background": "#F2EEE5",
        "wall_color": wall_palette[block % len(wall_palette)],
        "floor_color": "#FBF8EE",
        "grid_color": "#159FB3",
        "marker_color": "#F2B134",
        "component_cells": components,
        "target_slot": target_slot,
        "marker_local": marker_local,
    }


def _global_open_cells(component_cells: list[list[list[int]]]) -> set[tuple[int, int]]:
    output: set[tuple[int, int]] = set()
    for (slot_r, slot_c), component in zip(SLOT_ORIGINS, component_cells):
        output.update((slot_r + int(r), slot_c + int(c)) for r, c in component)
    return output


def render(scene: dict) -> Image.Image:
    """Render only the public visual state contained in ``scene``."""
    width = int(scene["width"])
    height = int(scene["height"])
    grid_n = int(scene["grid_n"])
    cell = int(scene["cell_size"])
    x0 = int(scene["origin_x"])
    y0 = int(scene["origin_y"])
    background = scene["background"]
    wall_color = scene["wall_color"]
    floor_color = scene["floor_color"]
    grid_color = scene["grid_color"]
    marker_color = scene["marker_color"]
    components = scene["component_cells"]
    target_slot = int(scene["target_slot"])
    marker_local = scene["marker_local"]

    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)
    total = grid_n * cell
    draw.rectangle((x0, y0, x0 + total - 1, y0 + total - 1), fill=wall_color)
    for r, c in _global_open_cells(components):
        draw.rectangle(
            (x0 + c * cell, y0 + r * cell, x0 + (c + 1) * cell - 1, y0 + (r + 1) * cell - 1),
            fill=floor_color,
        )
    for index in range(grid_n + 1):
        x = x0 + index * cell
        y = y0 + index * cell
        draw.line((x, y0, x, y0 + total), fill=grid_color, width=1)
        draw.line((x0, y, x0 + total, y), fill=grid_color, width=1)
    draw.rectangle(
        (x0 - 4, y0 - 4, x0 + total + 3, y0 + total + 3),
        outline=grid_color,
        width=4,
    )

    slot_r, slot_c = SLOT_ORIGINS[target_slot]
    mr = slot_r + int(marker_local[0])
    mc = slot_c + int(marker_local[1])
    cx = x0 + mc * cell + cell // 2
    cy = y0 + mr * cell + cell // 2
    radius = max(4, cell // 5)
    draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=marker_color)
    return image


def analytic_gold(scene: dict) -> str:
    cells = {(int(r), int(c)) for r, c in scene["component_cells"][int(scene["target_slot"]) ]}
    return _decision_and_gap(cells)[0]


def margin(scene: dict) -> float:
    cells = {(int(r), int(c)) for r, c in scene["component_cells"][int(scene["target_slot"]) ]}
    return _decision_and_gap(cells)[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
