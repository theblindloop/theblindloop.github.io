"""Deterministic latent sampler and renderer for Adjacent-Change Mosaic Reveal."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
SOURCE_ROWS = 10
SOURCE_COLS = 9
OUTPUT_ROWS = 10
OUTPUT_COLS = 8
LOGICAL_ROWS = 5
PAD = 14
BACKGROUND = (248, 247, 242)
FRAME = (72, 78, 90)
PANEL = (166, 172, 174)
LIGHT = (218, 222, 214)
CYAN = (25, 190, 205)
GOLD = (235, 165, 35)
DARK_PALETTES = ((32, 52, 75), (92, 38, 60), (34, 82, 58))
LABELS = ("diamond", "fork", "stair")


def _logical_template(name: str) -> tuple[int, ...]:
    rows = {
        "diamond": (
            "...##...",
            "..#..#..",
            ".#....#.",
            "..#..#..",
            "...##...",
        ),
        "fork": (
            "#..##..#",
            "....#.#.",
            ".#..#...",
            "....#...",
            "....#...",
        ),
        "stair": (
            "##......",
            ".##.....",
            "...##...",
            "...##...",
            "....##..",
        ),
    }
    try:
        selected = rows[name]
    except KeyError:
        raise KeyError(name) from None
    bits = tuple(int(value == "#") for row in selected for value in row)
    if len(bits) != LOGICAL_ROWS * OUTPUT_COLS or sum(bits) != 10:
        raise AssertionError("literal template must have shape 5 by 8 and weight 10")
    return bits


LOGICAL_TEMPLATES = {name: _logical_template(name) for name in LABELS}


def _expand(logical: list[int] | tuple[int, ...]) -> tuple[int, ...]:
    values = tuple(int(value) for value in logical)
    if len(values) != LOGICAL_ROWS * OUTPUT_COLS:
        raise ValueError("logical derivative must contain 40 values")
    expanded: list[int] = []
    for row in range(LOGICAL_ROWS):
        segment = values[row * OUTPUT_COLS : (row + 1) * OUTPUT_COLS]
        expanded.extend(segment)
        expanded.extend(segment)
    return tuple(expanded)


TEMPLATES = {name: _expand(bits) for name, bits in LOGICAL_TEMPLATES.items()}


def _validated_bits(bits: list[int] | tuple[int, ...]) -> tuple[int, ...]:
    values = tuple(int(value) for value in bits)
    if len(values) != OUTPUT_ROWS * OUTPUT_COLS or any(value not in (0, 1) for value in values):
        raise ValueError("derivative_bits must contain 80 binary values")
    if sum(values) != 20:
        raise ValueError("derivative_bits must have fixed weight 20")
    for pair in range(LOGICAL_ROWS):
        first = values[(2 * pair) * OUTPUT_COLS : (2 * pair + 1) * OUTPUT_COLS]
        second = values[(2 * pair + 1) * OUTPUT_COLS : (2 * pair + 2) * OUTPUT_COLS]
        if first != second:
            raise ValueError("each logical derivative row must be visibly doubled")
    return values


def _distances(bits: list[int] | tuple[int, ...]) -> dict[str, int]:
    values = _validated_bits(bits)
    return {
        name: sum(left != right for left, right in zip(values, template, strict=True))
        for name, template in TEMPLATES.items()
    }


def _classify(bits: list[int] | tuple[int, ...]) -> tuple[str, int]:
    distances = _distances(bits)
    ordered = sorted((distance, name) for name, distance in distances.items())
    gap = ordered[1][0] - ordered[0][0]
    return ("ambiguous" if gap == 0 else ordered[0][1]), gap


def _noisy_pattern(target: str, swaps: int, rng: random.Random) -> list[int]:
    original = list(LOGICAL_TEMPLATES[target])
    for _ in range(4000):
        logical = original.copy()
        ones = [index for index, value in enumerate(logical) if value]
        zeros = [index for index, value in enumerate(logical) if not value]
        for index in rng.sample(ones, swaps):
            logical[index] = 0
        for index in rng.sample(zeros, swaps):
            logical[index] = 1
        expanded = list(_expand(logical))
        if _classify(expanded)[0] == target:
            return expanded
    raise RuntimeError("could not sample a uniquely classified fixed-weight derivative")


def _tie_pattern(left: str, right: str, rng: random.Random) -> list[int]:
    a = set(index for index, value in enumerate(LOGICAL_TEMPLATES[left]) if value)
    b = set(index for index, value in enumerate(LOGICAL_TEMPLATES[right]) if value)
    a_only = sorted(a - b)
    b_only = sorted(b - a)
    if len(a_only) != len(b_only) or len(a_only) % 2:
        raise AssertionError("template pair must have an equal-weight discrete midpoint")
    half = len(a_only) // 2
    for _ in range(4000):
        rng.shuffle(a_only)
        rng.shuffle(b_only)
        logical_set = (a & b) | set(a_only[:half]) | set(b_only[:half])
        logical = [int(index in logical_set) for index in range(LOGICAL_ROWS * OUTPUT_COLS)]
        expanded = list(_expand(logical))
        if _classify(expanded)[0] == "ambiguous":
            return expanded
    raise RuntimeError("could not construct a fixed-weight visible template tie")


def sample_scene(seed: int) -> dict:
    """Sample balanced labels, margins, and answer-independent visual nuisances."""
    seed = int(seed)
    rng = random.Random(seed)
    target = LABELS[seed % len(LABELS)]
    difficulty = (seed // len(LABELS)) % 6
    if difficulty == 0:
        other = LABELS[(LABELS.index(target) + 1) % len(LABELS)]
        derivative_bits = _tie_pattern(target, other, rng)
    else:
        derivative_bits = _noisy_pattern(target, min(4, difficulty - 1), rng)

    row_phases: list[int] = []
    for _ in range(LOGICAL_ROWS):
        phase = rng.randrange(2)
        row_phases.extend((phase, 1 - phase))

    orientation = rng.randrange(4)
    cell_size = (18, 19, 20)[rng.randrange(3)]
    base_width = SOURCE_COLS * cell_size + 2 * PAD
    base_height = SOURCE_ROWS * cell_size + 2 * PAD
    patch_width, patch_height = (
        (base_width, base_height) if orientation % 2 == 0 else (base_height, base_width)
    )
    origin_x = (WIDTH - patch_width) // 2 + rng.randint(-16, 16)
    origin_y = (HEIGHT - patch_height) // 2 + rng.randint(-16, 16)
    return {
        "derivative_bits": derivative_bits,
        "row_phases": row_phases,
        "orientation": orientation,
        "cell_size": cell_size,
        "origin_x": origin_x,
        "origin_y": origin_y,
        "palette": rng.randrange(len(DARK_PALETTES)),
        "distractor_seed": rng.randrange(1_000_000_000),
    }


def _encoded_cells(scene: dict) -> list[list[int]]:
    derivative = _validated_bits(scene["derivative_bits"])
    phases = [int(value) for value in scene["row_phases"]]
    if len(phases) != SOURCE_ROWS or any(value not in (0, 1) for value in phases):
        raise ValueError("row_phases must contain ten binary values")
    cells: list[list[int]] = []
    for row in range(SOURCE_ROWS):
        current = phases[row]
        encoded = [current]
        for col in range(OUTPUT_COLS):
            current ^= derivative[row * OUTPUT_COLS + col]
            encoded.append(current)
        cells.append(encoded)
    for pair in range(LOGICAL_ROWS):
        if any(
            left == right
            for left, right in zip(cells[2 * pair], cells[2 * pair + 1], strict=True)
        ):
            raise ValueError("paired carrier rows must be complementary")
    if sum(map(sum, cells)) != 45:
        raise AssertionError("the complementary carrier must contain exactly 45 dark cells")
    return cells


def _draw_patch(scene: dict) -> Image.Image:
    cell = int(scene["cell_size"])
    width = SOURCE_COLS * cell + 2 * PAD
    height = SOURCE_ROWS * cell + 2 * PAD
    patch = Image.new("RGB", (width, height), PANEL)
    draw = ImageDraw.Draw(patch)
    draw.rectangle((0, 0, width - 1, height - 1), outline=FRAME, width=4)
    draw.rectangle((5, 5, 10, 10), fill=CYAN)
    draw.rectangle((width - 11, 5, width - 6, 10), fill=GOLD)
    dark = DARK_PALETTES[int(scene["palette"])]
    cells = _encoded_cells(scene)
    for row in range(SOURCE_ROWS):
        for col in range(SOURCE_COLS):
            x0 = PAD + col * cell + 1
            y0 = PAD + row * cell + 1
            x1 = PAD + (col + 1) * cell - 2
            y1 = PAD + (row + 1) * cell - 2
            draw.rectangle((x0, y0, x1, y1), fill=dark if cells[row][col] else LIGHT)
    for _ in range(int(scene["orientation"]) % 4):
        patch = patch.transpose(Image.Transpose.ROTATE_90)
    return patch


def render(scene: dict) -> Image.Image:
    """Render from declared fields only; no labels, text, or metadata are embedded."""
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    patch = _draw_patch(scene)
    origin = (int(scene["origin_x"]), int(scene["origin_y"]))
    image.paste(patch, origin)
    draw = ImageDraw.Draw(image)
    rng = random.Random(int(scene["distractor_seed"]))
    x0, y0 = origin
    x1, y1 = x0 + patch.width, y0 + patch.height
    colors = ((180, 145, 155), (145, 165, 190), (175, 170, 130))
    placed = 0
    while placed < 7:
        x = rng.randint(28, WIDTH - 28)
        y = rng.randint(28, HEIGHT - 28)
        if x0 - 24 <= x <= x1 + 24 and y0 - 24 <= y <= y1 + 24:
            continue
        radius = rng.randint(3, 6)
        color = colors[rng.randrange(len(colors))]
        if rng.randrange(2):
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)
        else:
            draw.rectangle((x - radius, y - radius, x + radius, y + radius), fill=color)
        placed += 1
    return image


def analytic_gold(scene: dict) -> str:
    return _classify(scene["derivative_bits"])[0]


def margin(scene: dict) -> float:
    return float(_classify(scene["derivative_bits"])[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
