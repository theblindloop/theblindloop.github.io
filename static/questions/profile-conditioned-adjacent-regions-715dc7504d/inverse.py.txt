"""Independent pixel-only inverse arm for the adjacent-regions world.

receives ONLY the final PNG. It never imports or reads the renderer, the
analytic scene, prompts, or renderer state. It recovers the decision by:

1. segmenting the three colored filled regions by color;
2. for each colour, dilating its region mask by a small radius, so that a tiny
   shared boundary band becomes a broad overlap;
3. a pair of regions is ADJACENT exactly when their dilated masks intersect
   (their boundaries meet / their filled areas overlap), while separated pairs
   stay far enough apart that their dilated masks remain disjoint.

The answer is the unique pair of coloured regions that shares a boundary.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "adjacent_regions-oracle-0.4.0"

_WHITE_THRESHOLD = 240
_COLOR_TOLERANCE = 95.0
_DILATE_RADIUS = 8
_PAIRS = [("red", "green"), ("red", "blue"), ("green", "blue")]
_RANK = {"red": 0, "green": 1, "blue": 2}
_COLORS = {
    "red": np.array([230.0, 45.0, 45.0]),
    "green": np.array([50.0, 185.0, 60.0]),
    "blue": np.array([50.0, 90.0, 230.0]),
}


def _is_white(arr: np.ndarray, threshold: int = _WHITE_THRESHOLD) -> np.ndarray:
    return arr.min(axis=-1) > threshold


def _color_mask(arr: np.ndarray, target: np.ndarray, tolerance: float) -> np.ndarray:
    diff = arr.astype(np.float32) - target
    return np.sqrt((diff * diff).sum(axis=-1)) < tolerance


def _dilate(mask: np.ndarray, radius: int) -> np.ndarray:
    out = np.zeros_like(mask)
    h, w = mask.shape
    for dy in range(-radius, radius + 1):
        for dx in range(-radius, radius + 1):
            if dx * dx + dy * dy > radius * radius:
                continue
            shifted = np.zeros_like(mask)
            src_y0 = max(0, dy)
            src_y1 = min(h, h + dy)
            src_x0 = max(0, dx)
            src_x1 = min(w, w + dx)
            dst_y0 = max(0, -dy)
            dst_y1 = dst_y0 + (src_y1 - src_y0)
            dst_x0 = max(0, -dx)
            dst_x1 = dst_x0 + (src_x1 - src_x0)
            patch = mask[src_y0:src_y1, src_x0:src_x1]
            out[dst_y0:dst_y1, dst_x0:dst_x1] |= patch
    return out


def _adjacent(arr: np.ndarray, c1: str, c2: str) -> bool:
    m1 = _color_mask(arr, _COLORS[c1], _COLOR_TOLERANCE)
    m2 = _color_mask(arr, _COLORS[c2], _COLOR_TOLERANCE)
    if not m1.any() or not m2.any():
        return False
    d1 = _dilate(m1, _DILATE_RADIUS)
    d2 = _dilate(m2, _DILATE_RADIUS)
    return bool((d1 & d2).any())


def decision_from_image(image: Image.Image) -> str:
    """The pair of coloured regions that share a boundary, from pixels alone."""
    arr = np.asarray(image.convert("RGB")).astype(np.float32)
    for c1, c2 in _PAIRS:
        if _adjacent(arr, c1, c2):
            return "-".join(sorted((c1, c2), key=lambda c: _RANK[c]))
    return "red-green"
