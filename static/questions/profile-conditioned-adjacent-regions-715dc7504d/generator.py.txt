"""Adjacent-regions world renderer.

Three colored filled regions (red, green, blue) are drawn on a white canvas, all
of the same fixed size. Exactly one pair of regions shares a boundary (their
filled areas overlap by a small band, so they are adjacent in the planar
arrangement), while the other two pairs are separated by a clear white gap.

Decision : which pair of colored regions (of red/green/blue) are adjacent
(their boundaries meet) in the picture.

The scene is a plain dict so the verifier's AST reachability audit is free of
side effects.

    z = {
        "width": 500,
        "height": 500,
        "regions": [
            {"x0": int, "y0": int, "x1": int, "y1": int, "color": str},
            ...
        ],  # exactly three regions, all same size, exactly one pair overlapping
    }
"""

from __future__ import annotations

import math
import random
from collections.abc import Iterator

from PIL import Image, ImageDraw

RENDERER_VERSION = "adjacent_regions-0.4.0"

WHITE = (255, 255, 255)
_COLOR_RGB = {"red": (230, 45, 45), "green": (50, 185, 60), "blue": (50, 90, 230)}

PAIRS = [("red", "green"), ("red", "blue"), ("green", "blue")]
_RANK = {"red": 0, "green": 1, "blue": 2}


def _pair_label(c1: str, c2: str) -> str:
    return "-".join(sorted((c1, c2), key=lambda c: _RANK[c]))


# base centre slots for each which-pair-is-adjacent case.
# The overlapping pair sits horizontally side by side on the left; the lone
# region is placed far up-right so it is cleanly separated from both members.
_BASE = {
    frozenset(("red", "green")): ["red", "green", "blue"],
    frozenset(("red", "blue")): ["red", "blue", "green"],
    frozenset(("green", "blue")): ["green", "blue", "red"],
}
_PAIR_SLOT = (140.0, 210.0)
_SECOND_SLOT = (202.0, 210.0)
_FAR_SLOT = (360.0, 350.0)


def _rect(cx: float, cy: float, half: float) -> tuple[float, float, float, float]:
    return (cx - half, cy - half, cx + half, cy + half)


def sample_scene(
    rng: random.Random,
    width: int = 500,
    height: int = 500,
) -> dict:
    """Sample a scene with exactly one adjacent pair of colored regions."""
    size = 92.0
    half = size / 2.0
    adjacent = rng.choice(PAIRS)
    order = _BASE[frozenset(adjacent)]
    jx = rng.uniform(-7.0, 7.0)
    jy = rng.uniform(-7.0, 7.0)
    pair_centers = {
        order[0]: (_PAIR_SLOT[0] + jx, _PAIR_SLOT[1] + jy),
        order[1]: (_SECOND_SLOT[0] + jx, _SECOND_SLOT[1] + jy),
        order[2]: (_FAR_SLOT[0] - jx, _FAR_SLOT[1] + jy),
    }
    regions: list[dict] = []
    for color in ("red", "green", "blue"):
        cx, cy = pair_centers[color]
        x0, y0, x1, y1 = _rect(cx, cy, half)
        regions.append(
            {
                "x0": int(round(x0)),
                "y0": int(round(y0)),
                "x1": int(round(x1)),
                "y1": int(round(y1)),
                "color": color,
            }
        )
    return {"width": width, "height": height, "regions": regions}


def _intersect_area(a: dict, b: dict) -> float:
    ow = min(a["x1"], b["x1"]) - max(a["x0"], b["x0"])
    oh = min(a["y1"], b["y1"]) - max(a["y0"], b["y0"])
    if ow <= 0 or oh <= 0:
        return 0.0
    return ow * oh


def analytic_gold(scene: dict) -> str:
    """The pair of colored regions that share a boundary."""
    regions = scene["regions"]
    best_area = 0.0
    best = _pair_label(*PAIRS[0])
    for i in range(len(regions)):
        for j in range(i + 1, len(regions)):
            a = regions[i]
            b = regions[j]
            ow = min(a["x1"], b["x1"]) - max(a["x0"], b["x0"])
            oh = min(a["y1"], b["y1"]) - max(a["y0"], b["y0"])
            area = ow * oh if (ow > 0 and oh > 0) else 0.0
            if area > best_area:
                best_area = area
                best = _pair_label(a["color"], b["color"])
    return best


def _gap(a: dict, b: dict) -> float:
    dx = max(a["x0"] - b["x1"], b["x0"] - a["x1"], 0)
    dy = max(a["y0"] - b["y1"], b["y0"] - a["y1"], 0)
    return math.hypot(dx, dy)


def margin(scene: dict) -> float:
    """Robustness margin: the minimum separation between non-adjacent pairs.

    Positive and large when the two non-touching pairs are cleanly separated by
    white background, guarding the image-only proximity test against raster
    perturbation.
    """
    regions = scene["regions"]
    adjacent = analytic_gold(scene)
    best: float = float("inf")
    for i in range(len(regions)):
        for j in range(i + 1, len(regions)):
            if _pair_label(regions[i]["color"], regions[j]["color"]) == adjacent:
                continue
            best = min(best, _gap(regions[i], regions[j]))
    return float(best)


def _adjacent_overlap(scene: dict) -> float:
    regions = scene["regions"]
    adjacent = analytic_gold(scene)
    for i in range(len(regions)):
        for j in range(i + 1, len(regions)):
            if _pair_label(regions[i]["color"], regions[j]["color"]) == adjacent:
                return _intersect_area(regions[i], regions[j])
    return 0.0


def is_quarantined(scene: dict) -> bool:
    """Quarantine scenes where non-adjacent regions get too close or where the
    adjacent pair is only a near-tangent touch."""
    if margin(scene) < 30.0:
        return True
    if _adjacent_overlap(scene) < 1400.0:
        return True
    return False


def render(scene: dict, supersample: int = 4) -> Image.Image:
    """Render R(z) deterministically. Pure function of the scene dict."""
    width = int(scene["width"])
    height = int(scene["height"])
    regions = scene["regions"]
    ss = supersample
    img = Image.new("RGB", (width * ss, height * ss), WHITE)
    draw = ImageDraw.Draw(img)
    for region in regions:
        x0 = int(round(region["x0"] * ss))
        y0 = int(round(region["y0"] * ss))
        x1 = int(round(region["x1"] * ss))
        y1 = int(round(region["y1"] * ss))
        draw.rounded_rectangle(
            [x0, y0, x1, y1],
            radius=int(14 * ss),
            fill=_COLOR_RGB[region["color"]],
        )
    if ss > 1:
        img = img.resize((width, height), Image.LANCZOS)
    return img


def latent_symmetries(scene: dict) -> Iterator[dict]:
    """No nontrivial pixel-identical transform: every field directly redraws.

    Which color sits in the overlapping pair is a label-transforming relabel
    (it changes the image) that the decision tracks exactly; declared
    not_applicable.
    """
    return iter(())
