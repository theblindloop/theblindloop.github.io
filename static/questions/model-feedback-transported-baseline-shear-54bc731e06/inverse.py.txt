"""Independent image-only inverse for transported baseline shear."""

from __future__ import annotations

import numpy as np
from PIL import Image

CANVAS = (1200, 850)
N = 9
KEY_CELL, KEY_Y = 24, 58
KEY_X = (20, 268, 684, 932)
STAMP_X, STAMP_Y, STAMP_CELL = 492, 330, 24
CANDIDATE_X = (24, 492, 960)
CANDIDATE_Y, CANDIDATE_CELL = 612, 24
INK = (29, 34, 40)
PAPER = (247, 249, 251)
GOLD = (238, 178, 24)
LANDMARK_RGB = {"red": (214, 65, 70), "blue": (35, 113, 190), "green": (39, 151, 94)}
TAG_RGB = {"amber": (226, 151, 28), "cyan": (20, 157, 181), "violet": (137, 78, 191)}
ALLOWED = {(255, 255, 255), PAPER, (132, 141, 150), INK, (71, 79, 91), GOLD,
           *LANDMARK_RGB.values(), *TAG_RGB.values()}
SIGNS = (-1, 1)
FRAMES = ("identity", "rot90", "rot180", "rot270", "flip_lr", "flip_ud", "transpose", "anti")


class OracleError(ValueError):
    pass


def _frame(point: tuple[int, int], name: str) -> tuple[int, int]:
    r, c = point
    if name == "identity": return r, c
    if name == "rot90": return c, N - 1 - r
    if name == "rot180": return N - 1 - r, N - 1 - c
    if name == "rot270": return N - 1 - c, r
    if name == "flip_lr": return r, N - 1 - c
    if name == "flip_ud": return N - 1 - r, c
    if name == "transpose": return c, r
    if name == "anti": return N - 1 - c, N - 1 - r
    raise OracleError("unknown frame")


def _inverse(name: str) -> str:
    try:
        return {"identity": "identity", "rot90": "rot270", "rot180": "rot180",
                "rot270": "rot90", "flip_lr": "flip_lr", "flip_ud": "flip_ud",
                "transpose": "transpose", "anti": "anti"}[name]
    except KeyError as exc:
        raise OracleError("unknown inverse") from exc


def _shear(point: tuple[int, int], baseline: int, sign: int) -> tuple[int, int]:
    r, c = point
    if sign not in SIGNS:
        raise OracleError("invalid shear sign")
    return r, c + sign * (r - baseline)


def _read_key(arr: np.ndarray, x: int) -> dict[str, tuple[int, int]]:
    found: dict[str, tuple[int, int]] = {}
    for r in range(N):
        for c in range(N):
            cx, cy = x + c * KEY_CELL + KEY_CELL // 2, KEY_Y + r * KEY_CELL + KEY_CELL // 2
            pixel = tuple(int(v) for v in arr[cy, cx])
            names = [name for name, rgb in LANDMARK_RGB.items() if pixel == rgb]
            if names:
                if names[0] in found:
                    raise OracleError("duplicate landmark color")
                found[names[0]] = (r, c)
            elif pixel not in (PAPER, GOLD):
                raise OracleError("damaged key cell center")
    if set(found) != set(LANDMARK_RGB):
        raise OracleError("missing colored landmark")
    return found


def _read_baseline(arr: np.ndarray, x: int) -> int:
    scores = []
    for r in range(N):
        cy = KEY_Y + r * KEY_CELL + KEY_CELL // 2
        line = arr[cy, x + 5:x + N * KEY_CELL - 4]
        scores.append(int(np.all(line == GOLD, axis=1).sum()))
    matches = [r for r, score in enumerate(scores) if score >= 120]
    if len(matches) != 1 or any(score > 0 for r, score in enumerate(scores) if r != matches[0]):
        raise OracleError("missing or damaged gold baseline")
    return matches[0]


def _read_marks(arr: np.ndarray, x: int, y: int, cell: int) -> set[tuple[int, int]]:
    found = set()
    for r in range(N):
        for c in range(N):
            cx, cy = x + c * cell + cell // 2, y + r * cell + cell // 2
            patch = arr[cy - 4:cy + 5, cx - 4:cx + 5]
            count = int(np.all(patch == INK, axis=2).sum())
            if count >= 60:
                found.add((r, c))
            elif count != 0:
                raise OracleError("damaged black cell mark")
    return found


def _infer_shear(source: dict[str, tuple[int, int]], target: dict[str, tuple[int, int]], baseline: int) -> int:
    matches = [sign for sign in SIGNS if all(
        _shear(source[color], baseline, sign) == target[color] for color in LANDMARK_RGB)]
    if len(matches) != 1:
        raise OracleError("ambiguous shear")
    return matches[0]


def _infer_frame(source: dict[str, tuple[int, int]], target: dict[str, tuple[int, int]]) -> str:
    matches = [name for name in FRAMES if all(
        _frame(source[color], name) == target[color] for color in LANDMARK_RGB)]
    if len(matches) != 1:
        raise OracleError("ambiguous frame")
    return matches[0]


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    if rgb.size != CANVAS:
        raise OracleError("wrong canvas size")
    colors = rgb.getcolors(maxcolors=64)
    if colors is None or not {tuple(color) for _, color in colors} <= ALLOWED:
        raise OracleError("unknown raster colors")
    arr = np.asarray(rgb)
    move_source = _read_key(arr, KEY_X[0])
    move_target = _read_key(arr, KEY_X[1])
    baseline_a = _read_baseline(arr, KEY_X[0])
    baseline_b = _read_baseline(arr, KEY_X[1])
    if baseline_a != baseline_b:
        raise OracleError("move baselines disagree")
    frame_source = _read_key(arr, KEY_X[2])
    frame_target = _read_key(arr, KEY_X[3])
    sign = _infer_shear(move_source, move_target, baseline_a)
    frame = _infer_frame(frame_source, frame_target)
    inverse = _inverse(frame)
    stamp = _read_marks(arr, STAMP_X, STAMP_Y, STAMP_CELL)
    if len(stamp) != 6:
        raise OracleError("stamp must contain six marks")
    target = {_frame(_shear(_frame(point, inverse), baseline_a, sign), frame) for point in stamp}
    if len(target) != 6 or any(not (0 <= r < N and 0 <= c < N) for r, c in target):
        raise OracleError("transported result is invalid")

    matches, seen_colors = [], set()
    for x in CANDIDATE_X:
        border = tuple(int(v) for v in arr[CANDIDATE_Y + 2, x + 2])
        names = [name for name, value in TAG_RGB.items() if border == value]
        if len(names) != 1 or names[0] in seen_colors:
            raise OracleError("candidate border colors invalid")
        seen_colors.add(names[0])
        pattern = _read_marks(arr, x, CANDIDATE_Y, CANDIDATE_CELL)
        if len(pattern) != 6:
            raise OracleError("candidate cardinality differs from stamp")
        if pattern == target:
            matches.append(names[0])
    if seen_colors != set(TAG_RGB) or len(matches) != 1:
        raise OracleError("no unique matching candidate")
    return matches[0]
