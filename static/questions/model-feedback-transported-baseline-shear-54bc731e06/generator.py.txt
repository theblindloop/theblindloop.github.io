"""Deterministic renderer for a baseline shear transported across frames."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = (1200, 850)
N = 9
KEY_CELL, KEY_Y = 24, 58
KEY_X = (20, 268, 684, 932)
STAMP_X, STAMP_Y, STAMP_CELL = 492, 330, 24
CANDIDATE_X = (24, 492, 960)
CANDIDATE_Y, CANDIDATE_CELL = 612, 24
ANSWERS = ("amber", "cyan", "violet")
LANDMARKS = ("red", "blue", "green")
FRAMES = ("rot90", "rot270", "transpose", "anti")
ALL_FRAMES = ("identity", "rot90", "rot180", "rot270", "flip_lr", "flip_ud", "transpose", "anti")
SIGNS = (-1, 1)

RGB = {
    "background": (255, 255, 255), "paper": (247, 249, 251),
    "grid": (132, 141, 150), "ink": (29, 34, 40), "arrow": (71, 79, 91),
    "red": (214, 65, 70), "blue": (35, 113, 190), "green": (39, 151, 94),
    "gold": (238, 178, 24), "amber": (226, 151, 28),
    "cyan": (20, 157, 181), "violet": (137, 78, 191),
}


def _inside(point: tuple[int, int]) -> bool:
    return 0 <= point[0] < N and 0 <= point[1] < N


def _frame(point: tuple[int, int], name: str) -> tuple[int, int]:
    r, c = point
    if name == "identity": return r, c
    if name == "rot90": return c, N - 1 - r
    if name == "rot180": return N - 1 - r, N - 1 - c
    if name == "rot270": return N - 1 - c, r
    if name == "flip_lr": return r, N - 1 - c
    if name == "flip_ud": return N - 1 - r, c
    if name == "transpose": return c, r
    if name == "anti": return N - 1 - c, N - 1 - r
    raise ValueError(f"unknown frame {name}")


def _inverse_frame(name: str) -> str:
    return {"identity": "identity", "rot90": "rot270", "rot180": "rot180",
            "rot270": "rot90", "flip_lr": "flip_lr", "flip_ud": "flip_ud",
            "transpose": "transpose", "anti": "anti"}[name]


def _shear(point: tuple[int, int], baseline: int, sign: int) -> tuple[int, int]:
    r, c = point
    if sign not in SIGNS:
        raise ValueError("invalid shear sign")
    return r, c + sign * (r - baseline)


def _infer_shear(source: dict[str, list[int]], target: dict[str, list[int]], baseline: int) -> int:
    matches = [sign for sign in SIGNS if all(
        _shear(tuple(source[color]), baseline, sign) == tuple(target[color]) for color in LANDMARKS)]
    if len(matches) != 1:
        raise ValueError("shear demonstration is ambiguous")
    return matches[0]


def _infer_frame(source: dict[str, list[int]], target: dict[str, list[int]]) -> str:
    matches = [name for name in ALL_FRAMES if all(
        _frame(tuple(source[color]), name) == tuple(target[color]) for color in LANDMARKS)]
    if len(matches) != 1:
        raise ValueError("frame demonstration is ambiguous")
    return matches[0]


def _map_set(points: set[tuple[int, int]], fn) -> set[tuple[int, int]]:
    return {fn(point) for point in points}


def _sets(scene: dict) -> tuple[set[tuple[int, int]], set[tuple[int, int]], set[tuple[int, int]]]:
    baseline = int(scene["baseline"])
    sign = _infer_shear(scene["move_source"], scene["move_target"], baseline)
    frame = _infer_frame(scene["frame_source"], scene["frame_target"])
    inverse = _inverse_frame(frame)
    stamp = {tuple(point) for point in scene["stamp"]}
    correct = _map_set(stamp, lambda point: _frame(_shear(_frame(point, inverse), baseline, sign), frame))
    untransported = _map_set(stamp, lambda point: _shear(point, baseline, sign))
    reversed_sign = _map_set(stamp, lambda point: _frame(_shear(_frame(point, inverse), baseline, -sign), frame))
    return correct, untransported, reversed_sign


def _random_frame_key(rng: random.Random, name: str) -> tuple[dict[str, list[int]], dict[str, list[int]]]:
    while True:
        cells = rng.sample([(r, c) for r in range(N) for c in range(N)], 3)
        source = {color: list(cell) for color, cell in zip(LANDMARKS, cells)}
        target = {color: list(_frame(cell, name)) for color, cell in zip(LANDMARKS, cells)}
        try:
            if _infer_frame(source, target) == name:
                return source, target
        except ValueError:
            pass


def sample_scene(seed: int) -> dict:
    rng = random.Random((int(seed) ^ 0x5EAF19D3) & 0xFFFFFFFFFFFFFFFF)
    while True:
        sign = rng.choice(SIGNS)
        frame = rng.choice(FRAMES)
        baseline = rng.choice((3, 4, 5))
        rows = (baseline - 2, baseline, baseline + 2)
        move_cells = []
        for r in rows:
            choices = [(r, c) for c in range(1, N - 1) if _inside(_shear((r, c), baseline, sign))]
            move_cells.append(rng.choice(choices))
        rng.shuffle(move_cells)
        move_source = {color: list(cell) for color, cell in zip(LANDMARKS, move_cells)}
        move_target = {color: list(_shear(cell, baseline, sign)) for color, cell in zip(LANDMARKS, move_cells)}
        try:
            if _infer_shear(move_source, move_target, baseline) != sign:
                continue
        except ValueError:
            continue
        frame_source, frame_target = _random_frame_key(rng, frame)
        inverse = _inverse_frame(frame)
        valid_stamp = []
        for point in ((r, c) for r in range(N) for c in range(N)):
            outcomes = (
                _frame(_shear(_frame(point, inverse), baseline, sign), frame),
                _shear(point, baseline, sign),
                _frame(_shear(_frame(point, inverse), baseline, -sign), frame),
            )
            if all(_inside(outcome) for outcome in outcomes):
                valid_stamp.append(point)
        if len(valid_stamp) < 6:
            continue
        stamp = set(rng.sample(valid_stamp, 6))
        provisional = {
            "baseline": baseline, "shear_sign": sign,
            "move_source": move_source, "move_target": move_target,
            "frame_source": frame_source, "frame_target": frame_target,
            "stamp": sorted(map(list, stamp)),
        }
        correct, untransported, reversed_sign = _sets(provisional)
        patterns = (correct, untransported, reversed_sign)
        if (len({frozenset(pattern) for pattern in patterns}) == 3
                and all(len(pattern) == 6 for pattern in patterns)
                and min(len(correct.symmetric_difference(pattern)) for pattern in patterns[1:]) >= 4):
            break

    target_slot = int(seed) % 3
    distractors = [untransported, reversed_sign]
    rng.shuffle(distractors)
    arranged = list(distractors)
    arranged.insert(target_slot, correct)
    answer_color = ANSWERS[(int(seed) // 3) % 3]
    other_colors = [color for color in ANSWERS if color != answer_color]
    rng.shuffle(other_colors)
    tag_colors: list[str | None] = [None, None, None]
    tag_colors[target_slot] = answer_color
    fill = iter(other_colors)
    for index in range(3):
        if tag_colors[index] is None:
            tag_colors[index] = next(fill)
    provisional["candidates"] = [sorted(map(list, pattern)) for pattern in arranged]
    provisional["tag_colors"] = [str(color) for color in tag_colors]
    return provisional


def _grid(draw: ImageDraw.ImageDraw, x: int, y: int, cell: int,
          border: tuple[int, int, int], width: int = 4) -> None:
    size = N * cell
    draw.rectangle((x, y, x + size, y + size), fill=RGB["paper"], outline=border, width=width)
    for k in range(1, N):
        draw.line((x + k * cell, y, x + k * cell, y + size), fill=RGB["grid"], width=2)
        draw.line((x, y + k * cell, x + size, y + k * cell), fill=RGB["grid"], width=2)


def _baseline(draw: ImageDraw.ImageDraw, x: int, y: int, cell: int, row: int) -> None:
    cy = y + row * cell + cell // 2
    draw.line((x + 5, cy, x + N * cell - 5, cy), fill=RGB["gold"], width=7)


def _dot(draw: ImageDraw.ImageDraw, x: int, y: int, cell: int, color: tuple[int, int, int]) -> None:
    cx, cy = x + cell // 2, y + cell // 2
    draw.ellipse((cx - 8, cy - 8, cx + 8, cy + 8), fill=color, outline=RGB["ink"], width=2)


def _cell_mark(draw: ImageDraw.ImageDraw, x: int, y: int, cell: int) -> None:
    draw.rounded_rectangle((x + 6, y + 6, x + cell - 6, y + cell - 6), radius=3, fill=RGB["ink"])


def _arrow(draw: ImageDraw.ImageDraw, x1: int, x2: int, y: int) -> None:
    draw.line((x1, y, x2, y), fill=RGB["arrow"], width=6)
    direction = 1 if x2 > x1 else -1
    draw.polygon(((x2, y), (x2 - direction * 13, y - 10), (x2 - direction * 13, y + 10)), fill=RGB["arrow"])


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS, RGB["background"])
    draw = ImageDraw.Draw(image)
    baseline = int(scene["baseline"])
    keys = (scene["move_source"], scene["move_target"], scene["frame_source"], scene["frame_target"])
    # One black bar labels SLANT; two bars label FRAME without relying on font rendering.
    draw.rounded_rectangle((103, 20, 165, 30), radius=3, fill=RGB["ink"])
    draw.rounded_rectangle((787, 15, 839, 24), radius=3, fill=RGB["ink"])
    draw.rounded_rectangle((798, 31, 850, 40), radius=3, fill=RGB["ink"])
    for x in KEY_X:
        _grid(draw, x, KEY_Y, KEY_CELL, RGB["ink"])
    for x in KEY_X[:2]:
        _baseline(draw, x, KEY_Y, KEY_CELL, baseline)
    for key_index, key in enumerate(keys):
        x = KEY_X[key_index]
        for color, (r, c) in key.items():
            _dot(draw, x + int(c) * KEY_CELL, KEY_Y + int(r) * KEY_CELL, KEY_CELL, RGB[color])
    _arrow(draw, KEY_X[0] + N * KEY_CELL + 9, KEY_X[1] - 9, KEY_Y + N * KEY_CELL // 2)
    _arrow(draw, KEY_X[2] + N * KEY_CELL + 9, KEY_X[3] - 9, KEY_Y + N * KEY_CELL // 2)

    _grid(draw, STAMP_X, STAMP_Y, STAMP_CELL, RGB["ink"])
    for r, c in scene["stamp"]:
        _cell_mark(draw, STAMP_X + int(c) * STAMP_CELL, STAMP_Y + int(r) * STAMP_CELL, STAMP_CELL)
    # Composition reminder beneath the stamp: FRAME inverse, SLANT, FRAME.
    draw.rounded_rectangle((415, 563, 451, 570), radius=2, fill=RGB["ink"])
    draw.rounded_rectangle((415, 575, 451, 582), radius=2, fill=RGB["ink"])
    _arrow(draw, 405, 364, 573)
    draw.rounded_rectangle((516, 569, 570, 577), radius=2, fill=RGB["ink"])
    _arrow(draw, 584, 625, 573)
    draw.rounded_rectangle((637, 563, 673, 570), radius=2, fill=RGB["ink"])
    draw.rounded_rectangle((637, 575, 673, 582), radius=2, fill=RGB["ink"])

    for slot, x in enumerate(CANDIDATE_X):
        color = str(scene["tag_colors"][slot])
        _grid(draw, x, CANDIDATE_Y, CANDIDATE_CELL, RGB[color], width=8)
        for r, c in scene["candidates"][slot]:
            _cell_mark(draw, x + int(c) * CANDIDATE_CELL, CANDIDATE_Y + int(r) * CANDIDATE_CELL, CANDIDATE_CELL)
    return image


def analytic_gold(scene: dict) -> str:
    target = _sets(scene)[0]
    matches = [index for index, pattern in enumerate(scene["candidates"])
               if {tuple(point) for point in pattern} == target]
    if len(matches) != 1:
        raise ValueError("no unique transported shear result")
    return str(scene["tag_colors"][matches[0]])


def margin(scene: dict) -> float:
    target = _sets(scene)[0]
    distances = sorted(len(target.symmetric_difference({tuple(point) for point in pattern}))
                       for pattern in scene["candidates"])
    return float(distances[1]) if len(distances) == 3 and distances[0] == 0 else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        baseline = int(scene["baseline"])
        sign = _infer_shear(scene["move_source"], scene["move_target"], baseline)
        frame = _infer_frame(scene["frame_source"], scene["frame_target"])
        correct, untransported, reversed_sign = _sets(scene)
        patterns = [{tuple(point) for point in pattern} for pattern in scene["candidates"]]
        return (baseline not in (3, 4, 5) or sign not in SIGNS or frame not in FRAMES
                or len({frozenset(correct), frozenset(untransported), frozenset(reversed_sign)}) != 3
                or any(len(pattern) != 6 for pattern in patterns)
                or any(not _inside(point) for pattern in patterns for point in pattern)
                or sum(pattern == correct for pattern in patterns) != 1
                or len(set(scene["tag_colors"])) != 3 or margin(scene) < 4)
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
