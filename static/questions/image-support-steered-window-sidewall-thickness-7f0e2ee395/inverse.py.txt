"""Independent inverse arm using RGB pixels only."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    seen = np.zeros(mask.shape, dtype=bool)
    parts: list[list[tuple[int, int]]] = []
    height, width = mask.shape
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        part: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            part.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        parts.append(part)
    return parts


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    # Broad chromatic threshold tolerates PNG conversions but rejects gray background.
    teal = (rgb[:, :, 1] > 105) & (rgb[:, :, 1] - rgb[:, :, 0] > 55) & (
        rgb[:, :, 1] - rgb[:, :, 2] > -20
    ) & (rgb[:, :, 2] > 105)
    parts = [part for part in _components(teal) if len(part) >= 300]
    if len(parts) != 1:
        return "abstain"
    ys = np.array([p[0] for p in parts[0]])
    xs = np.array([p[1] for p in parts[0]])
    y0, y1 = int(ys.min()), int(ys.max())
    x0, x1 = int(xs.min()), int(xs.max())
    height, width = y1 - y0 + 1, x1 - x0 + 1
    if not (38 <= height <= 70 and 40 <= width <= 70):
        return "abstain"
    crop = teal[y0 : y1 + 1, x0 : x1 + 1]
    # A valid rectangular frame has full outer borders and a single rectangular void.
    if not (crop[0].all() and crop[-1].all() and crop[:, 0].all() and crop[:, -1].all()):
        return "abstain"
    interior_void = ~crop
    void_parts = [part for part in _components(interior_void) if len(part) >= 100]
    if len(void_parts) != 1:
        return "abstain"
    vys = np.array([p[0] for p in void_parts[0]])
    vxs = np.array([p[1] for p in void_parts[0]])
    vy0, vy1 = int(vys.min()), int(vys.max())
    vx0, vx1 = int(vxs.min()), int(vxs.max())
    expected_area = (vy1 - vy0 + 1) * (vx1 - vx0 + 1)
    if len(void_parts[0]) != expected_area or vy0 < 7 or vy1 > height - 8:
        return "abstain"
    left = vx0
    right = width - vx1 - 1
    if min(left, right) < 7 or abs(left - right) < 4:
        return "abstain"
    return "left" if left < right else "right"
