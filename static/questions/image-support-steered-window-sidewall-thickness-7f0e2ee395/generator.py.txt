"""Deterministic compact window-frame world."""

from __future__ import annotations

import random

import numpy as np
from PIL import Image


TEAL = (25, 145, 150)
BACKGROUND = (247, 248, 244)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    canvas_w = 256
    canvas_h = 256
    outer_w = rng.randint(46, 54)
    outer_h = rng.randint(46, 55)
    window_w = rng.randint(20, 24)
    window_h = rng.randint(18, 24)

    # Seed parity counterbalances answer identity independently of position and size.
    thin = rng.randint(8, 10)
    gap = rng.randint(8, 10)
    thick = thin + gap
    if seed % 2 == 0:
        left_wall, right_wall = thin, thick
    else:
        left_wall, right_wall = thick, thin
    # Make the outer width exactly consistent while retaining randomized window width.
    outer_w = left_wall + window_w + right_wall
    top_wall = rng.randint(8, 12)
    bottom_wall = outer_h - window_h - top_wall
    if bottom_wall < 8:
        outer_h = top_wall + window_h + rng.randint(8, 12)

    outer_x = rng.randint(18, canvas_w - outer_w - 18)
    outer_y = rng.randint(18, canvas_h - outer_h - 18)
    window_x = outer_x + left_wall
    window_y = outer_y + top_wall
    return {
        "canvas_w": canvas_w,
        "canvas_h": canvas_h,
        "outer_x": outer_x,
        "outer_y": outer_y,
        "outer_w": outer_w,
        "outer_h": outer_h,
        "window_x": window_x,
        "window_y": window_y,
        "window_w": window_w,
        "window_h": window_h,
    }


def render(scene: dict) -> Image.Image:
    canvas_w = int(scene["canvas_w"])
    canvas_h = int(scene["canvas_h"])
    outer_x = int(scene["outer_x"])
    outer_y = int(scene["outer_y"])
    outer_w = int(scene["outer_w"])
    outer_h = int(scene["outer_h"])
    window_x = int(scene["window_x"])
    window_y = int(scene["window_y"])
    window_w = int(scene["window_w"])
    window_h = int(scene["window_h"])
    pixels = np.empty((canvas_h, canvas_w, 3), dtype=np.uint8)
    pixels[:, :] = BACKGROUND
    pixels[outer_y : outer_y + outer_h, outer_x : outer_x + outer_w] = TEAL
    pixels[window_y : window_y + window_h, window_x : window_x + window_w] = BACKGROUND
    return Image.fromarray(pixels, mode="RGB")


def analytic_gold(scene: dict) -> str:
    left = int(scene["window_x"]) - int(scene["outer_x"])
    right = (int(scene["outer_x"]) + int(scene["outer_w"])) - (
        int(scene["window_x"]) + int(scene["window_w"])
    )
    return "left" if left < right else "right"


def margin(scene: dict) -> float:
    left = int(scene["window_x"]) - int(scene["outer_x"])
    right = (int(scene["outer_x"]) + int(scene["outer_w"])) - (
        int(scene["window_x"]) + int(scene["window_w"])
    )
    return float(abs(left - right))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
