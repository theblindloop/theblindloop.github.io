"""Independent pixel-only inverse arm for Bulk Color Rewrite Replay.

This module deliberately duplicates only the public visual vocabulary.  It does
not import or inspect the renderer, scene data, manifest, prompts, filenames, or
metadata.  Colored components are recovered by relative row bands and x-runs,
not by renderer coordinates or OCR.
"""

from __future__ import annotations

from collections import Counter

import numpy as np
from PIL import Image

_TARGET = "blue"
_PROTOTYPES: dict[tuple[int, int, int], str] = {
    (220, 66, 70): "red",
    (48, 105, 205): "blue",
    (45, 154, 92): "green",
    (238, 170, 35): "gold",
    (194, 52, 66): "red",
    (42, 91, 181): "blue",
    (36, 135, 80): "green",
    (218, 145, 24): "gold",
    (235, 83, 82): "red",
    (62, 123, 222): "blue",
    (56, 171, 105): "green",
    (246, 185, 50): "gold",
}


class PixelParseError(ValueError):
    pass


def _runs(active: np.ndarray, *, minimum: int) -> list[tuple[int, int]]:
    padded = np.pad(active.astype(np.int8), (1, 1))
    edges = np.diff(padded)
    starts = np.flatnonzero(edges == 1)
    ends = np.flatnonzero(edges == -1)
    return [(int(a), int(b)) for a, b in zip(starts, ends, strict=True) if b - a >= minimum]


def _symbol_for_blob(arr: np.ndarray, known: np.ndarray, yspan: tuple[int, int], xspan: tuple[int, int]) -> str:
    y0, y1 = yspan
    x0, x1 = xspan
    pixels = arr[y0:y1, x0:x1][known[y0:y1, x0:x1]]
    if len(pixels) < 450:
        raise PixelParseError("colored endpoint is too small or erased")
    counts = Counter(map(tuple, pixels.tolist()))
    symbolic = Counter()
    for rgb, count in counts.items():
        label = _PROTOTYPES.get(rgb)
        if label is not None:
            symbolic[label] += count
    if not symbolic:
        raise PixelParseError("colored component has no recognized solid interior")
    best, best_count = symbolic.most_common(1)[0]
    if best_count < 0.96 * sum(symbolic.values()):
        raise PixelParseError("colored component has an ambiguous label")
    return best


def _decode(image: Image.Image) -> tuple[list[str], list[tuple[str, str]]]:
    arr = np.asarray(image.convert("RGB"))
    if arr.ndim != 3 or arr.shape[0] < 200 or arr.shape[1] < 500:
        raise PixelParseError("unexpected raster dimensions")

    known = np.zeros(arr.shape[:2], dtype=bool)
    for rgb in _PROTOTYPES:
        known |= np.all(arr == np.asarray(rgb, dtype=np.uint8), axis=2)

    row_bands = _runs(known.sum(axis=1) >= 20, minimum=20)
    if len(row_bands) != 2:
        raise PixelParseError(f"expected two colored row bands, found {len(row_bands)}")

    decoded_rows: list[list[str]] = []
    expected_components = (6, 10)
    for yspan, expected in zip(row_bands, expected_components, strict=True):
        y0, y1 = yspan
        xspans = _runs(known[y0:y1].sum(axis=0) >= 10, minimum=15)
        if len(xspans) != expected:
            raise PixelParseError(f"expected {expected} colored components, found {len(xspans)}")
        decoded_rows.append([_symbol_for_blob(arr, known, yspan, xspan) for xspan in xspans])

    initial = decoded_rows[0]
    endpoints = decoded_rows[1]
    cards = list(zip(endpoints[::2], endpoints[1::2], strict=True))
    if any(source == target for source, target in cards):
        raise PixelParseError("a card has identical source and target colors")
    return initial, cards


def decision_from_image(image: Image.Image) -> str:
    initial, cards = _decode(image)
    state = list(initial)
    for source, target in cards:
        state = [target if color == source else color for color in state]
    return str(state.count(_TARGET))
