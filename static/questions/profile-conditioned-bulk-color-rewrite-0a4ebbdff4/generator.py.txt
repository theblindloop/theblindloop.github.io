"""Renderer and analytic arm for Bulk Color Rewrite Replay.

The scene contains six colored beads and five ordered rewrite cards.  A card
``source -> target`` changes every bead currently in ``source`` to ``target``
simultaneously.  The analytic arm uses only the two logical fields named in the
candidate's label-input declaration.  Rendering nuisances are visibly encoded.
"""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1000
HEIGHT = 470
N_BEADS = 6
N_CARDS = 5
COLOR_NAMES = ("red", "blue", "green", "gold")
TARGET_COLOR = "blue"

PALETTES: tuple[dict[str, tuple[int, int, int]], ...] = (
    {
        "red": (220, 66, 70),
        "blue": (48, 105, 205),
        "green": (45, 154, 92),
        "gold": (238, 170, 35),
    },
    {
        "red": (194, 52, 66),
        "blue": (42, 91, 181),
        "green": (36, 135, 80),
        "gold": (218, 145, 24),
    },
    {
        "red": (235, 83, 82),
        "blue": (62, 123, 222),
        "green": (56, 171, 105),
        "gold": (246, 185, 50),
    },
)


def _font(size: int, *, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _replay(initial: Iterable[str], cards: Iterable[Iterable[str]]) -> list[str]:
    state = list(initial)
    for source, target in cards:
        state = [target if color == source else color for color in state]
    return state


def _program_stats(initial: list[str], cards: list[list[str]]) -> tuple[int, bool]:
    state = list(initial)
    effective = 0
    merged = False
    for source, target in cards:
        if source in state:
            effective += 1
            if target in state:
                merged = True
        state = [target if color == source else color for color in state]
    return effective, merged


def sample_scene(seed: int) -> dict:
    """Sample a balanced, order-sensitive scene deterministically.

    Consecutive seeds rotate through all seven possible final blue counts.  The
    construction rejects programs for which the initial blue count gives away
    the answer, reversing the cards preserves the answer, fewer than three
    cards take effect, or no occupied color classes merge.
    """

    rng = random.Random(int(seed))
    desired = int(seed) % (N_BEADS + 1)
    for _ in range(200_000):
        # Keep every symbolic color visible while varying the two repeated colors.
        initial = list(COLOR_NAMES)
        initial.extend(rng.choice(COLOR_NAMES) for _ in range(N_BEADS - len(COLOR_NAMES)))
        rng.shuffle(initial)
        cards: list[list[str]] = []
        for _card in range(N_CARDS):
            source = rng.choice(COLOR_NAMES)
            target = rng.choice(tuple(color for color in COLOR_NAMES if color != source))
            cards.append([source, target])

        final = _replay(initial, cards)
        if final.count(TARGET_COLOR) != desired:
            continue
        if initial.count(TARGET_COLOR) == desired:
            continue
        if _replay(initial, reversed(cards)).count(TARGET_COLOR) == desired:
            continue
        effective, merged = _program_stats(initial, cards)
        if effective < 3 or not merged:
            continue

        bead_order = list(range(N_BEADS))
        rng.shuffle(bead_order)
        return {
            "initial_colors": initial,
            "rewrite_cards": cards,
            "bead_order": bead_order,
            "palette": rng.randrange(len(PALETTES)),
            "layout_variant": rng.randrange(4),
        }
    raise RuntimeError(f"could not construct balanced rewrite scene for seed {seed}")


def analytic_gold(scene: dict) -> str:
    final = _replay(scene["initial_colors"], scene["rewrite_cards"])
    return str(final.count(TARGET_COLOR))


def margin(scene: dict) -> float:
    """Nearest runner-up color margin in RGB units for the visible palette."""

    palette = PALETTES[int(scene["palette"])]
    values = list(palette.values())
    return min(math.dist(left, right) for i, left in enumerate(values) for right in values[i + 1 :])


def is_quarantined(scene: dict) -> bool:
    try:
        initial = scene["initial_colors"]
        cards = scene["rewrite_cards"]
        order = scene["bead_order"]
        palette = int(scene["palette"])
        layout = int(scene["layout_variant"])
        structurally_valid = (
            len(initial) == N_BEADS
            and all(value in COLOR_NAMES for value in initial)
            and len(cards) == N_CARDS
            and all(
                len(card) == 2
                and card[0] in COLOR_NAMES
                and card[1] in COLOR_NAMES
                and card[0] != card[1]
                for card in cards
            )
            and sorted(order) == list(range(N_BEADS))
            and 0 <= palette < len(PALETTES)
            and 0 <= layout < 4
        )
        return not structurally_valid or margin(scene) < 85.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _centered(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font, fill) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text(
        (xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2 - box[1]),
        text,
        font=font,
        fill=fill,
    )


def render(scene: dict) -> Image.Image:
    initial = scene["initial_colors"]
    cards = scene["rewrite_cards"]
    order = scene["bead_order"]
    palette_index = int(scene["palette"])
    layout = int(scene["layout_variant"])
    palette = PALETTES[palette_index]

    image = Image.new("RGB", (WIDTH, HEIGHT), (250, 249, 246))
    draw = ImageDraw.Draw(image)
    ink = (31, 38, 46)
    muted = (92, 101, 112)
    panel = (255, 255, 255)
    title_font = _font(25, bold=True)
    label_font = _font(17, bold=True)
    small_font = _font(14, bold=True)

    _centered(draw, (WIDTH / 2, 27), "BULK COLOR REWRITE", title_font, ink)
    _centered(draw, (WIDTH / 2, 55), "Every matching bead changes together", _font(15), muted)

    dx = (-9, 8, -3, 10)[layout]
    top_y = (137, 131, 142, 135)[layout]
    draw.rounded_rectangle((42, 77, 958, 199), radius=18, fill=panel, outline=(160, 168, 177), width=3)
    draw.text((62, 88), "START", font=label_font, fill=ink)
    displayed = [initial[index] for index in order]
    for index, color_name in enumerate(displayed):
        x = 126 + index * 150 + dx
        r = 28
        draw.ellipse((x - r, top_y - r, x + r, top_y + r), fill=palette[color_name], outline=ink, width=4)

    draw.text((42, 222), "READ LEFT TO RIGHT", font=label_font, fill=ink)
    card_y0 = (260, 255, 264, 258)[layout]
    card_y1 = card_y0 + 158
    dot_y = card_y0 + 83
    for index, (source, target) in enumerate(cards):
        left = 25 + index * 192
        right = left + 174
        draw.rounded_rectangle((left, card_y0, right, card_y1), radius=15, fill=panel, outline=(151, 160, 170), width=3)
        _centered(draw, ((left + right) / 2, card_y0 + 22), str(index + 1), small_font, muted)
        source_x = left + 48
        target_x = left + 126
        draw.line((source_x + 24, dot_y, target_x - 26, dot_y), fill=ink, width=5)
        draw.polygon(
            ((target_x - 26, dot_y - 9), (target_x - 26, dot_y + 9), (target_x - 13, dot_y)),
            fill=ink,
        )
        r = 20
        draw.ellipse((source_x - r, dot_y - r, source_x + r, dot_y + r), fill=palette[source], outline=ink, width=3)
        draw.ellipse((target_x - r, dot_y - r, target_x + r, dot_y + r), fill=palette[target], outline=ink, width=3)
        _centered(draw, ((left + right) / 2, card_y0 + 130), "ALL", small_font, muted)

    return image
