"""Pixel-only inverse arm for the fold-count tracing world.

Reads only the final RGB PNG. Segments the blue stroke by nearest-colour
labelling, then counts the number of horizontal lane bands of blue ink across
the image rows (a row belongs to a lane when it carries a substantial stretch of
blue). The single connected winding path is made of horizontal lanes joined by
short vertical U-turn connectors, so the number of U-turn folds equals the number
of horizontal lane bands minus one. Only the final pixels are consumed.

No renderer, prompt, generator, verifier or scene data is imported or read.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_PALETTE = {
    "blue": (40, 90, 210),
    "white": (255, 255, 255),
}
_NAMES = ["blue", "white"]
_LANE_ROW_THRESH = 60


def _label_pixels(arr: np.ndarray) -> np.ndarray:
    palette = np.array([*_PALETTE.values()], dtype=np.float32)
    dists = np.stack([np.linalg.norm(arr - c, axis=-1) for c in palette], axis=-1)
    return dists.argmin(axis=-1)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label_pixels(arr)
    idx = {name: i for i, name in enumerate(_NAMES)}
    blue = labels == idx["blue"]
    if not blue.any():
        raise ValueError("no blue path found")

    row_counts = blue.sum(axis=1).astype(np.int64)
    lane_rows = row_counts > _LANE_ROW_THRESH

    n_lanes = 0
    in_band = False
    for value in lane_rows.tolist():
        if value and not in_band:
            n_lanes += 1
        in_band = bool(value)
    if n_lanes < 1:
        raise ValueError("no horizontal lane bands detected")
    folds = n_lanes - 1
    return str(folds)
