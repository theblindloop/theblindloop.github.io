"""Tracing fold-count world: how many U-turn folds does the single blue path make.

A single connected winding blue polyline is drawn as a serpentine: several
horizontal lanes joined at their ends by short vertical connectors. Each
vertical connector is a sharp hairpin where the stroke doubles back on itself
(a U-turn fold). The answer is the number of such U-turn folds in the path.

Because the path folds back and forth, counting the folds is a tracing /
topological fact about the drawn stroke (its sequence of direction reversals),
not a geodesic distance, an identity selection, or a count of disconnected
components.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "path_fold_count-0.1.0"

CANVAS = (480, 480)
BG = (255, 255, 255)
COLOR_BLUE = (40, 90, 210)

STROKE = 20          # blue stroke thickness (px)
MARGIN_PAD = 46      # keep geometry away from image borders
LANE_GAP = 52.0      # center-to-center vertical gap between adjacent lanes
MIN_CLEARANCE = 24.0  # quarantine below this off-quarantine margin threshold


def _pts(points: list) -> list[tuple[float, float]]:
    return [(float(p[0]), float(p[1])) for p in points]


def _rejection_loop(rng: random.Random) -> dict:
    w, h = CANVAS
    lo, hi = MARGIN_PAD, w - MARGIN_PAD
    for _ in range(20000):
        n_lanes = rng.randint(4, 6)          # folds in {3, 4, 5}
        x_start = rng.uniform(lo + 15, lo + 60)
        x_end = rng.uniform(hi - 60, hi - 15)
        total_h = (n_lanes - 1) * LANE_GAP
        y_lo = (h - total_h) / 2.0
        lane_ys = [y_lo + lane * LANE_GAP for lane in range(n_lanes)]
        path = []
        for lane, y in enumerate(lane_ys):
            if lane % 2 == 0:
                path.append((x_start, y))
                path.append((x_end, y))
            else:
                path.append((x_end, y))
                path.append((x_start, y))
        path = _pts(path)
        if not (20 < x_start < x_end < w - 20):
            continue
        if not (20 < lane_ys[0] and lane_ys[-1] < h - 20):
            continue
        return {"path": path}
    raise RuntimeError("could not satisfy path constraints")


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    return _rejection_loop(rng)


def _draw_polyline(draw: ImageDraw.ImageDraw, pts, color, width):
    for a, b in zip(pts, pts[1:]):
        draw.line([a, b], fill=color, width=width, joint="curve")


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", CANVAS, BG)
    draw = ImageDraw.Draw(img)
    path = _pts(scene["path"])
    _draw_polyline(draw, path, COLOR_BLUE, STROKE)
    return img


def _fold_count(scene: dict) -> int:
    path = _pts(scene["path"])
    folds = 0
    for a, b in zip(path, path[1:]):
        if abs(a[0] - b[0]) < 1e-6 and abs(a[1] - b[1]) > 1e-6:
            folds += 1
    return folds


def analytic_gold(scene: dict) -> str:
    """Number of U-turn folds in the single blue path, as a string."""
    return str(_fold_count(scene))


def margin(scene: dict) -> float:
    """Clearance between adjacent lane strokes (px), the counting ambiguity margin."""
    return float(LANE_GAP - STROKE)


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < MIN_CLEARANCE)


def latent_symmetries(scene: dict) -> list:
    return []
