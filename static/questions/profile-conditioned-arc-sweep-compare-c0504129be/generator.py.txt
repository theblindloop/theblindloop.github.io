"""Deterministic renderer and analytic side for the open-arc comparison world."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "arc_sweep_compare-renderer-0.1.0"
WIDTH = 760
HEIGHT = 420
SUPERSAMPLE = 4
CENTER_A = (190.0, 220.0)
CENTER_B = (570.0, 220.0)
MIN_GAP_DEG = 14.0
PALETTE = ("#2767b1", "#d1495b", "#269c82", "#8755ad", "#d88427")
LABEL_FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"


def _arc_points(
    center: tuple[float, float],
    radius: float,
    start_deg: float,
    sweep_deg: float,
) -> list[tuple[float, float]]:
    """Sample a mathematical counter-clockwise arc in image coordinates."""

    count = max(48, int(abs(sweep_deg) * radius / 1.25))
    cx, cy = center
    points: list[tuple[float, float]] = []
    for index in range(count + 1):
        angle = math.radians(start_deg + sweep_deg * index / count)
        points.append((cx + radius * math.cos(angle), cy - radius * math.sin(angle)))
    return points


def _scaled_points(points: Iterable[tuple[float, float]], scale: int) -> list[tuple[float, float]]:
    return [(round(x * scale), round(y * scale)) for x, y in points]


def sample_scene(seed: int) -> dict:
    """Sample one balanced, off-boundary scene from an integer seed."""

    rng = random.Random(int(seed))
    center_sweep = rng.uniform(72.0, 116.0)
    gap = rng.uniform(24.0, 54.0)
    if rng.random() < 0.5:
        sweep_a, sweep_b = center_sweep + gap / 2.0, center_sweep - gap / 2.0
    else:
        sweep_a, sweep_b = center_sweep - gap / 2.0, center_sweep + gap / 2.0

    colors = rng.sample(PALETTE, 2)
    return {
        "sweep_a": round(sweep_a, 6),
        "sweep_b": round(sweep_b, 6),
        "start_a": round(rng.uniform(0.0, 360.0), 6),
        "start_b": round(rng.uniform(0.0, 360.0), 6),
        "radius_a": round(rng.uniform(88.0, 126.0), 6),
        "radius_b": round(rng.uniform(88.0, 126.0), 6),
        "stroke_a": round(rng.uniform(7.0, 12.0), 6),
        "stroke_b": round(rng.uniform(7.0, 12.0), 6),
        "color_a": colors[0],
        "color_b": colors[1],
    }


def render(scene: dict) -> Image.Image:
    """Render both arcs; all scene fields are visibly used here."""

    # Keep these direct accesses in this function so the latent-field audit can
    # see that every analytic label input is also rendered.
    sweep_a = float(scene["sweep_a"])
    sweep_b = float(scene["sweep_b"])
    start_a = float(scene["start_a"])
    start_b = float(scene["start_b"])
    radius_a = float(scene["radius_a"])
    radius_b = float(scene["radius_b"])
    stroke_a = float(scene["stroke_a"])
    stroke_b = float(scene["stroke_b"])
    color_a = str(scene["color_a"])
    color_b = str(scene["color_b"])

    scale = SUPERSAMPLE
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), "white")
    draw = ImageDraw.Draw(image)
    label_font = ImageFont.truetype(LABEL_FONT_PATH, 24 * scale)

    for center, label, start, sweep, radius, stroke, color in (
        (CENTER_A, "A", start_a, sweep_a, radius_a, stroke_a, color_a),
        (CENTER_B, "B", start_b, sweep_b, radius_b, stroke_b, color_b),
    ):
        points = _scaled_points(_arc_points(center, radius, start, sweep), scale)
        draw.line(
            points,
            fill=color,
            width=max(1, round(stroke * scale)),
            joint="curve",
        )
        cx, cy = round(center[0] * scale), round(center[1] * scale)
        center_radius = 5 * scale
        draw.ellipse(
            (cx - center_radius, cy - center_radius, cx + center_radius, cy + center_radius),
            fill=(100, 100, 100),
        )
        # Labels are outside the annulus used by the inverse and identify the
        # public panel order without carrying answer-specific styling.
        bounds = draw.textbbox((0, 0), label, font=label_font)
        label_width = bounds[2] - bounds[0]
        draw.text(
            (cx - label_width // 2, 24 * scale),
            label,
            fill=(35, 35, 35),
            font=label_font,
        )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """Return the panel with the larger visible angular sweep."""

    return "A" if float(scene["sweep_a"]) > float(scene["sweep_b"]) else "B"


def margin(scene: dict) -> float:
    """Positive when A's sweep is larger, in degrees."""

    return float(scene["sweep_a"]) - float(scene["sweep_b"])


def is_quarantined(scene: dict) -> bool:
    """Near ties are explicit even though the standard sampler avoids them."""

    return abs(margin(scene)) < MIN_GAP_DEG


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No hidden alias is claimed: every label-bearing field is rendered."""

    return []
