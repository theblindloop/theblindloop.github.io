"""Independent pixel-only inverse for the open-arc comparison."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


# These are part of the public fixed page geometry, not imported renderer state.
_CENTERS = ((190.0, 220.0), (570.0, 220.0))
_WINDOW_HALF_WIDTH = 150.0
_MIN_RADIUS = 70.0
_MAX_RADIUS = 145.0


def _saturated_mask(rgb: np.ndarray) -> np.ndarray:
    values = rgb.astype(np.float32) / 255.0
    high = values.max(axis=2)
    low = values.min(axis=2)
    saturation = (high - low) / np.maximum(high, 1e-6)
    # Arc inks are saturated; gray center dots, black labels, and white paper
    # are rejected. The generous high cutoff keeps antialiased colored edges.
    return (saturation > 0.24) & (high < 0.99)


def _polar_span(rgb: np.ndarray, center: tuple[float, float]) -> float:
    height, width = rgb.shape[:2]
    cx, cy = center
    y0 = max(0, int(round(cy - _WINDOW_HALF_WIDTH)))
    y1 = min(height, int(round(cy + _WINDOW_HALF_WIDTH + 1)))
    x0 = max(0, int(round(cx - _WINDOW_HALF_WIDTH)))
    x1 = min(width, int(round(cx + _WINDOW_HALF_WIDTH + 1)))
    patch = rgb[y0:y1, x0:x1]
    mask = _saturated_mask(patch)
    yy, xx = np.nonzero(mask)
    if len(xx) < 100:
        raise ValueError("arc ink is not sufficiently visible")
    xx = xx.astype(np.float64) + x0 + 0.5
    yy = yy.astype(np.float64) + y0 + 0.5
    radii = np.hypot(xx - cx, yy - cy)
    keep = (radii >= _MIN_RADIUS) & (radii <= _MAX_RADIUS)
    if int(keep.sum()) < 100:
        raise ValueError("arc pixels do not occupy the expected annulus")
    angles = np.degrees(np.arctan2(-(yy[keep] - cy), xx[keep] - cx)) % 360.0
    angles.sort()
    gaps = np.diff(np.concatenate((angles, angles[:1] + 360.0)))
    largest_gap_index = int(np.argmax(gaps))
    start = float(angles[(largest_gap_index + 1) % len(angles)])
    unwrapped = np.where(angles < start, angles + 360.0, angles)
    unwrapped.sort()
    # Trim only the antialiased endpoint tails. The wide complement gap makes
    # this unwrap stable even when the arc crosses the 0-degree direction.
    lo = float(np.quantile(unwrapped, 0.015))
    hi = float(np.quantile(unwrapped, 0.985))
    span = hi - lo
    if not (5.0 < span < 250.0):
        raise ValueError(f"unreliable polar span {span:.3f}")
    return span


def decision_from_image(image: Image.Image) -> str:
    """Measure both visible polar spans and choose the larger one."""

    rgb = np.asarray(image.convert("RGB"))
    span_a = _polar_span(rgb, _CENTERS[0])
    span_b = _polar_span(rgb, _CENTERS[1])
    if abs(span_a - span_b) < 0.75:
        raise ValueError("pixel spans are too close to classify")
    return "A" if span_a > span_b else "B"
