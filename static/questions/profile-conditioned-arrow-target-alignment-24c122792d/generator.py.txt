"""Deterministic raster world for visible arrow-to-bullseye alignment."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "arrow_target_alignment-renderer-0.1.0"
WIDTH = 600
HEIGHT = 520
SUPERSAMPLE = 4

BACKGROUND_PALETTE = (
    (248, 250, 247),
    (247, 248, 252),
    (250, 247, 250),
    (247, 250, 250),
)
SHAFT_PALETTE = (
    (108, 119, 132),
    (124, 105, 120),
    (104, 128, 119),
    (128, 112, 96),
)
HEAD_PALETTE = (
    (31, 42, 53),
    (45, 35, 55),
    (33, 61, 48),
    (55, 42, 31),
)
TARGET_PALETTE = (
    (215, 67, 71),
    (49, 137, 211),
    (225, 161, 39),
    (179, 73, 177),
    (46, 162, 119),
)


def _rgb(value: Any) -> tuple[int, int, int]:
    return tuple(int(channel) for channel in value)  # type: ignore[return-value]


def _point(cx: float, cy: float, angle: float, distance: float) -> tuple[float, float]:
    return (cx + distance * math.cos(angle), cy + distance * math.sin(angle))


def _round_point(point: tuple[float, float], scale: int) -> tuple[int, int]:
    return (int(round(point[0] * scale)), int(round(point[1] * scale)))


def _angular_separation(left: float, right: float) -> float:
    return abs((left - right + math.pi) % (2.0 * math.pi) - math.pi)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced arrow and bullseye scene.

    Even scene seeds are clear inside-cone cases and odd seeds are clear
    outside-cone cases. The seed parity is used only to balance the data; it is
    never stored as an answer field and the rendered geometry determines gold.
    A small set of deterministic near-edge scenes makes quarantine observable.
    """

    seed = int(seed)
    rng = random.Random(seed * 1_000_003 + 73_991)
    arrow_angle = rng.uniform(-math.pi, math.pi)
    cone_half_angle = math.radians(rng.uniform(17.0, 23.0))
    side = -1.0 if rng.random() < 0.5 else 1.0

    if seed % 19 == 0:
        # These are intentionally close to the visible side edge but remain
        # readable enough for the pixel oracle to classify.
        edge_clearance = math.radians(rng.uniform(4.5, 6.0))
        offset = math.degrees(cone_half_angle - edge_clearance) if seed % 2 == 0 else math.degrees(cone_half_angle + edge_clearance)
    elif seed % 2 == 0:
        offset = rng.uniform(0.0, max(1.0, math.degrees(cone_half_angle) - 12.0))
    else:
        offset = rng.uniform(math.degrees(cone_half_angle) + 12.0, 55.0)
    target_angle = arrow_angle + side * math.radians(offset)

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "center_x": 300.0,
        "center_y": 260.0,
        "arrow_angle": round(arrow_angle, 8),
        "cone_half_angle": round(cone_half_angle, 8),
        "head_length": round(rng.uniform(45.0, 52.0), 4),
        "shaft_length": round(rng.uniform(136.0, 158.0), 4),
        "target_angle": round(target_angle, 8),
        "target_distance": round(rng.uniform(104.0, 124.0), 4),
        "target_radius": round(rng.uniform(26.0, 34.0), 4),
        "shaft_width": rng.choice((13, 15, 17)),
        "shaft_color": list(rng.choice(SHAFT_PALETTE)),
        "head_color": list(rng.choice(HEAD_PALETTE)),
        "target_color": list(rng.choice(TARGET_PALETTE)),
        "background_color": list(rng.choice(BACKGROUND_PALETTE)),
        "antialias": True,
    }


def _geometry(scene: dict[str, Any]) -> dict[str, tuple[float, float]]:
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    arrow_angle = float(scene["arrow_angle"])
    half = float(scene["cone_half_angle"])
    head_length = float(scene["head_length"])
    shaft_length = float(scene["shaft_length"])
    direction = (math.cos(arrow_angle), math.sin(arrow_angle))
    perpendicular = (-direction[1], direction[0])
    center = (cx, cy)
    shaft_start = (
        cx - direction[0] * shaft_length / 2.0,
        cy - direction[1] * shaft_length / 2.0,
    )
    base = (
        cx + direction[0] * shaft_length / 2.0,
        cy + direction[1] * shaft_length / 2.0,
    )
    tip = (
        base[0] + direction[0] * head_length,
        base[1] + direction[1] * head_length,
    )
    half_width = math.tan(half) * head_length
    base_left = (
        base[0] + perpendicular[0] * half_width,
        base[1] + perpendicular[1] * half_width,
    )
    base_right = (
        base[0] - perpendicular[0] * half_width,
        base[1] - perpendicular[1] * half_width,
    )
    target = _point(
        tip[0],
        tip[1],
        float(scene["target_angle"]),
        float(scene["target_distance"]),
    )
    return {
        "center": center,
        "shaft_start": shaft_start,
        "base": base,
        "tip": tip,
        "base_left": base_left,
        "base_right": base_right,
        "target": target,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render a gray-shaft arrow, a dark triangular head, and a marked target."""

    width = int(scene["width"])
    height = int(scene["height"])
    scale = SUPERSAMPLE if bool(scene["antialias"]) else 1
    background = _rgb(scene["background_color"])
    shaft = _rgb(scene["shaft_color"])
    head = _rgb(scene["head_color"])
    target_color = _rgb(scene["target_color"])
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    geometry = _geometry(scene)

    draw.line(
        [_round_point(geometry["shaft_start"], scale), _round_point(geometry["base"], scale)],
        fill=shaft,
        width=max(1, int(round(float(scene["shaft_width"]) * scale))),
    )
    draw.polygon(
        [
            _round_point(geometry["tip"], scale),
            _round_point(geometry["base_left"], scale),
            _round_point(geometry["base_right"], scale),
        ],
        fill=head,
    )

    tx, ty = geometry["target"]
    radius = float(scene["target_radius"])
    draw.ellipse(
        _round_point((tx - radius, ty - radius), scale)
        + _round_point((tx + radius, ty + radius), scale),
        fill=target_color,
    )
    center_mark = max(7.0, radius * 0.28)
    draw.ellipse(
        _round_point((tx - center_mark, ty - center_mark), scale)
        + _round_point((tx + center_mark, ty + center_mark), scale),
        fill=background,
    )

    if scale > 1:
        image = image.resize((width, height), Image.Resampling.LANCZOS)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return whether the target angle is inside the visible head aperture."""

    delta = _angular_separation(float(scene["target_angle"]), float(scene["arrow_angle"]))
    return "yes" if delta <= float(scene["cone_half_angle"]) else "no"


def margin(scene: dict[str, Any]) -> float:
    """Signed angular clearance in degrees from the nearer head side edge."""

    delta = _angular_separation(float(scene["target_angle"]), float(scene["arrow_angle"]))
    return round(math.degrees(float(scene["cone_half_angle"]) - delta), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Exclude centers too close to either visible head side edge."""

    return abs(margin(scene)) < 8.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No latent aliases are declared; every label-bearing angle is visible."""

    return []
