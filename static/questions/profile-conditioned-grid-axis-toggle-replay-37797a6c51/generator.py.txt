"""Deterministic Pillow world for ordered row/column lamp toggles."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 900, 500
QUESTION = (
    "Start with the marked lamp's shown state. Read the five cards from left to "
    "right. Each card's colored bar marks one whole row or column to flip; every "
    "lamp on that bar changes ON to OFF or OFF to ON. Is the marked lamp ON or "
    "OFF after all five flips?"
)

PALETTES = (
    {"on": (46, 134, 222), "off": (226, 231, 238), "command": (239, 144, 46)},
    {"on": (131, 79, 195), "off": (232, 228, 219), "command": (37, 158, 127)},
)
DARK = (35, 43, 54)
RED = (206, 48, 62)
WHITE = (255, 255, 255)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size=size)
    except OSError:
        return ImageFont.load_default()


def _affected(commands: list[dict], row: int, col: int) -> int:
    return sum(
        (item["axis"] == "row" and item["index"] == row)
        or (item["axis"] == "col" and item["index"] == col)
        for item in commands
    )


def sample_scene(seed: int) -> dict:
    """Sample a balanced, uniquely replayable scene from one integer seed."""
    rng = random.Random(int(seed) ^ 0x5A17C3)
    target_row = rng.randrange(3)
    target_col = rng.randrange(3)

    # Both axes appear, and the marked lamp is affected at least once but not by
    # every card. This keeps all five events meaningful and avoids degenerate cues.
    while True:
        commands = [
            {"axis": "row" if rng.randrange(2) == 0 else "col", "index": rng.randrange(3)}
            for _ in range(5)
        ]
        hits = _affected(commands, target_row, target_col)
        if {item["axis"] for item in commands} == {"row", "col"} and 1 <= hits <= 4:
            break

    initial_states = [[rng.randrange(2) for _ in range(3)] for _ in range(3)]
    desired_final = 1 if int(seed) % 2 == 0 else 0
    initial_states[target_row][target_col] = desired_final ^ (hits % 2)
    return {
        "initial_states": initial_states,
        "target_row": target_row,
        "target_col": target_col,
        "commands": commands,
        "palette_mode": (int(seed) // 2) % 2,
        "x_offset": (-6, 0, 6)[(int(seed) // 4) % 3],
        "y_offset": (-4, 0, 4)[(int(seed) // 12) % 3],
    }


def analytic_gold(scene: dict) -> str:
    states = scene["initial_states"]
    row = int(scene["target_row"])
    col = int(scene["target_col"])
    commands = scene["commands"]
    value = int(states[row][col])
    for item in commands:
        if (item["axis"] == "row" and int(item["index"]) == row) or (
            item["axis"] == "col" and int(item["index"]) == col
        ):
            value ^= 1
    return "ON" if value else "OFF"


def render(scene: dict) -> Image.Image:
    states = scene["initial_states"]
    target_row = int(scene["target_row"])
    target_col = int(scene["target_col"])
    commands = scene["commands"]
    palette = PALETTES[int(scene["palette_mode"])]
    x_offset = int(scene["x_offset"])
    y_offset = int(scene["y_offset"])

    image = Image.new("RGB", (WIDTH, HEIGHT), WHITE)
    draw = ImageDraw.Draw(image)
    draw.text((34, 24), "GRID AXIS TOGGLE", fill=DARK, font=_font(27, True))
    draw.text((34, 62), "START GRID", fill=DARK, font=_font(19, True))
    draw.text((335 + x_offset, 62), "FLIP CARDS  —  READ LEFT TO RIGHT", fill=DARK, font=_font(18, True))
    draw.text((35, 436), "Colored bar = the whole row or column that flips", fill=DARK, font=_font(16))

    gx, gy = 66 + x_offset, 130 + y_offset
    cell, pitch = 58, 72
    for row in range(3):
        for col in range(3):
            x = gx + col * pitch
            y = gy + row * pitch
            draw.rectangle((x, y, x + cell, y + cell), fill=DARK)
            fill = palette["on"] if int(states[row][col]) else palette["off"]
            draw.rectangle((x + 4, y + 4, x + cell - 4, y + cell - 4), fill=fill)
            if row == target_row and col == target_col:
                draw.ellipse((x - 6, y - 6, x + cell + 6, y + cell + 6), outline=RED, width=6)

    card_x0, card_y = 330 + x_offset, 126 + y_offset
    card_w, card_h, card_step = 82, 128, 100
    for number, item in enumerate(commands, start=1):
        left = card_x0 + (number - 1) * card_step
        draw.rectangle((left, card_y, left + card_w, card_y + card_h), outline=DARK, width=3)
        draw.text((left + card_w / 2, card_y + 16), str(number), anchor="mm", fill=DARK, font=_font(15, True))

        mx, my, mini = left + 17, card_y + 39, 48
        draw.rectangle((mx, my, mx + mini, my + mini), fill=(247, 248, 250), outline=DARK, width=2)
        for k in (1, 2):
            draw.line((mx + k * 16, my, mx + k * 16, my + mini), fill=DARK, width=2)
            draw.line((mx, my + k * 16, mx + mini, my + k * 16), fill=DARK, width=2)
        index = int(item["index"])
        if item["axis"] == "row":
            draw.rectangle((mx + 3, my + index * 16 + 3, mx + 45, my + index * 16 + 13), fill=palette["command"])
            label = f"ROW {index + 1}"
        else:
            draw.rectangle((mx + index * 16 + 3, my + 3, mx + index * 16 + 13, my + 45), fill=palette["command"])
            label = f"COL {index + 1}"
        draw.text((left + card_w / 2, card_y + 108), label, anchor="mm", fill=DARK, font=_font(12, True))

    # A legend makes both binary states explicit without encoding the answer.
    ly = 385
    draw.rectangle((608, ly, 628, ly + 20), fill=palette["on"], outline=DARK, width=2)
    draw.text((636, ly + 10), "ON", anchor="lm", fill=DARK, font=_font(14, True))
    draw.rectangle((708, ly, 728, ly + 20), fill=palette["off"], outline=DARK, width=2)
    draw.text((736, ly + 10), "OFF", anchor="lm", fill=DARK, font=_font(14, True))
    return image


def margin(scene: dict) -> float:
    # One card can move parity by exactly one; visual primitives are separated by
    # at least ten solid pixels, so this discrete replay margin is conservative.
    return 10.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def distractor_permutation(scene: dict) -> dict:
    """Test helper: reverse only non-target lamp states, preserving the label."""
    twin = deepcopy(scene)
    row, col = int(scene["target_row"]), int(scene["target_col"])
    values = [
        int(scene["initial_states"][r][c])
        for r in range(3)
        for c in range(3)
        if (r, c) != (row, col)
    ][::-1]
    cursor = 0
    for r in range(3):
        for c in range(3):
            if (r, c) != (row, col):
                twin["initial_states"][r][c] = values[cursor]
                cursor += 1
    return twin
