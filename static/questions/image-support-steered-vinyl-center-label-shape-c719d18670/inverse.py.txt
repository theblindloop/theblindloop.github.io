"""Independent pixel-only inverse for a record center-label boundary shape."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_MIN_CHROMA = 70
_MIN_LABEL_AREA = 18000
_MAX_LABEL_AREA = 34000


class OracleAbstention(ValueError):
    pass


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dy == 0 and dx == 0:
                        continue
                    ny, nx = cy + dy, cx + dx
                    if (
                        0 <= ny < height
                        and 0 <= nx < width
                        and mask[ny, nx]
                        and not seen[ny, nx]
                    ):
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        output.append(np.asarray(points, dtype=np.int32))
    return output


def _label_component(array: np.ndarray) -> np.ndarray:
    high = array.max(axis=2).astype(np.int16)
    low = array.min(axis=2).astype(np.int16)
    mask = ((high - low) >= _MIN_CHROMA) & (high >= 120)
    large = []
    for points in _components(mask):
        if _MIN_LABEL_AREA <= len(points) <= _MAX_LABEL_AREA:
            large.append(points)
    if len(large) != 1:
        raise OracleAbstention(f"expected one broad chromatic center label, found {len(large)}")
    return large[0]


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 256 or array.shape[1] < 256:
        raise OracleAbstention("raster is too small")
    points = _label_component(array)
    ys, xs = points[:, 0], points[:, 1]
    y0, y1 = int(ys.min()), int(ys.max())
    x0, x1 = int(xs.min()), int(xs.max())
    height, width = y1 - y0 + 1, x1 - x0 + 1
    if not (140 <= width <= 205 and 140 <= height <= 205):
        raise OracleAbstention("label extent is outside the supported range")
    aspect = width / float(height)
    if not 0.95 <= aspect <= 1.05:
        raise OracleAbstention("label boundary is not balanced in width and height")
    if x0 < 120 or y0 < 120 or x1 >= array.shape[1] - 120 or y1 >= array.shape[0] - 120:
        raise OracleAbstention("label is not in the record's broad center region")

    area = float(len(points))
    fill = area / float(width * height)
    center_x, center_y = float(xs.mean()), float(ys.mean())
    quadrants = np.asarray(
        [
            np.count_nonzero((xs <= center_x) & (ys <= center_y)),
            np.count_nonzero((xs > center_x) & (ys <= center_y)),
            np.count_nonzero((xs <= center_x) & (ys > center_y)),
            np.count_nonzero((xs > center_x) & (ys > center_y)),
        ],
        dtype=np.float64,
    )
    if float(quadrants.max() - quadrants.min()) / area > 0.018:
        raise OracleAbstention("label has a missing or asymmetric boundary sector")

    # The two area-matched, unit-aspect classes separate sharply by bbox fill:
    # pi/4 for a circle and nearly one for a square (minus the tiny spindle hole).
    if 0.745 <= fill <= 0.825:
        return "circle"
    if 0.972 <= fill <= 1.001:
        return "square"
    raise OracleAbstention(f"boundary model is ambiguous or corrupted (fill={fill:.4f})")
