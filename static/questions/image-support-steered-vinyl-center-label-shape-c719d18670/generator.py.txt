"""Deterministic renderer for visible vinyl center-label boundary shape."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

CANVAS_SIZE = 512
SUPERSAMPLE = 4
MODEL_THRESHOLD = 0.885

LABEL_PALETTE = (
    (224, 63, 72),
    (25, 156, 143),
    (56, 103, 214),
    (229, 154, 34),
    (143, 80, 190),
)
BACKGROUNDS = (
    (244, 241, 233),
    (235, 242, 244),
    (242, 237, 244),
    (239, 240, 234),
)
RECORD_TONES = (
    (29, 32, 35),
    (35, 34, 39),
    (27, 36, 38),
)


def sample_scene(seed: int) -> dict:
    """Return an adjacent-seed matched circular control/square counterfactual."""

    seed = int(seed)
    pair = seed // 2
    rng = random.Random(pair ^ 0x51A9E17)
    return {
        "label_shape": "circle" if seed % 2 == 0 else "square",
        "canvas_size": CANVAS_SIZE,
        "center_x": 256 + rng.randrange(-18, 19),
        "center_y": 256 + rng.randrange(-18, 19),
        "label_scale": (82, 86, 90, 94, 98)[pair % 5],
        "label_color_code": (pair * 3 + pair // 5) % len(LABEL_PALETTE),
        "record_tone": (pair // 2) % len(RECORD_TONES),
        "background_code": (pair // 3) % len(BACKGROUNDS),
        "groove_variant": rng.randrange(12),
        "spindle_radius": 5 + (pair % 4),
    }


def _box(values: tuple[float, float, float, float], scale: int) -> tuple[int, int, int, int]:
    return tuple(round(value * scale) for value in values)


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas_size"])
    cx = int(scene["center_x"])
    cy = int(scene["center_y"])
    label_scale = int(scene["label_scale"])
    shape = str(scene["label_shape"])
    color_code = int(scene["label_color_code"])
    record_code = int(scene["record_tone"])
    background_code = int(scene["background_code"])
    groove_variant = int(scene["groove_variant"])
    spindle_radius = int(scene["spindle_radius"])

    ss = SUPERSAMPLE
    background = BACKGROUNDS[background_code % len(BACKGROUNDS)]
    record = RECORD_TONES[record_code % len(RECORD_TONES)]
    label = LABEL_PALETTE[color_code % len(LABEL_PALETTE)]
    image = Image.new("RGB", (size * ss, size * ss), background)
    draw = ImageDraw.Draw(image)

    # Answer-independent sleeve seams and small chromatic price stickers.
    seam = tuple(max(0, value - 18) for value in background)
    for index in range(3):
        x = 34 + ((groove_variant * 47 + index * 181) % 444)
        y = 30 if (groove_variant + index) % 2 == 0 else size - 31
        draw.rounded_rectangle(
            _box((x - 20, y - 3, x + 20, y + 3), ss),
            radius=3 * ss,
            fill=seam,
        )
    for index in range(2):
        marker_color = LABEL_PALETTE[(color_code + 2 + index) % len(LABEL_PALETTE)]
        x = 38 if (groove_variant + index) % 2 == 0 else size - 38
        y = 92 + ((groove_variant * 37 + index * 241) % 324)
        draw.ellipse(_box((x - 5, y - 5, x + 5, y + 5), ss), fill=marker_color)

    record_radius = 188
    shadow = tuple(max(0, value - 26) for value in background)
    draw.ellipse(
        _box((cx - record_radius + 6, cy - record_radius + 8,
              cx + record_radius + 6, cy + record_radius + 8), ss),
        fill=shadow,
    )
    draw.ellipse(
        _box((cx - record_radius, cy - record_radius,
              cx + record_radius, cy + record_radius), ss),
        fill=record,
    )

    groove = tuple(min(255, value + 18) for value in record)
    groove_shift = groove_variant % 3
    for radius in range(112 + groove_shift * 3, 177, 13):
        draw.ellipse(
            _box((cx - radius, cy - radius, cx + radius, cy + radius), ss),
            outline=groove,
            width=1 * ss,
        )
    # Short subtle radial scuffs make styling vary without touching the label.
    for index in range(4):
        angle = math.radians((groove_variant * 29 + index * 83) % 360)
        r0, r1 = 124, 169
        draw.line(
            [
                (round((cx + r0 * math.cos(angle)) * ss), round((cy + r0 * math.sin(angle)) * ss)),
                (round((cx + r1 * math.cos(angle)) * ss), round((cy + r1 * math.sin(angle)) * ss)),
            ],
            fill=groove,
            width=1 * ss,
        )

    # Circle radius r and square side sqrt(pi)*r have the same analytic area.
    if shape == "circle":
        draw.ellipse(
            _box((cx - label_scale, cy - label_scale,
                  cx + label_scale, cy + label_scale), ss),
            fill=label,
        )
    elif shape == "square":
        half_side = math.sqrt(math.pi) * label_scale / 2.0
        draw.rectangle(
            _box((cx - half_side, cy - half_side,
                  cx + half_side, cy + half_side), ss),
            fill=label,
        )
    else:
        raise ValueError(f"unsupported label_shape {shape!r}")

    hole = tuple(min(255, value + 92) for value in record)
    draw.ellipse(
        _box((cx - spindle_radius, cy - spindle_radius,
              cx + spindle_radius, cy + spindle_radius), ss),
        fill=hole,
    )
    draw.ellipse(
        _box((cx - 2, cy - 2, cx + 2, cy + 2), ss),
        fill=record,
    )

    return image.resize((size, size), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    value = str(scene["label_shape"])
    if value not in {"circle", "square"}:
        raise ValueError(value)
    return value


def margin(scene: dict) -> float:
    radius = float(scene["label_scale"])
    hole = float(scene["spindle_radius"])
    if str(scene["label_shape"]) == "circle":
        fill = (math.pi * radius * radius - math.pi * hole * hole) / (4.0 * radius * radius)
    else:
        fill = 1.0 - (hole * hole) / (radius * radius)
    return abs(fill - MODEL_THRESHOLD)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.09


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["label_color_code"] = int(twin["label_color_code"]) + len(LABEL_PALETTE)
    return [("one_full_label_palette_cycle", twin)]
