"""Independent image-only route-geodesic search oracle."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

COLOR_RGB = {
    "red": (207, 66, 62),
    "blue": (45, 111, 181),
    "green": (49, 143, 91),
    "purple": (132, 76, 173),
}
DARK_RGB = (35, 40, 47)
GOLD_RGB = (221, 169, 43)
NEIGHBORS = ((-1, 0), (1, 0), (0, -1), (0, 1))


class OracleAbstention(ValueError):
    """The raster does not expose one certifiable route-mileage winner."""


def _components(mask: np.ndarray, minimum: int = 1) -> list[np.ndarray]:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    found: list[np.ndarray] = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if visited[sy, sx]:
            continue
        visited[sy, sx] = True
        queue = deque([(sy, sx)])
        pixels = []
        while queue:
            y, x = queue.popleft()
            pixels.append((y, x))
            for dy, dx in NEIGHBORS:
                ny, nx = y + dy, x + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    queue.append((ny, nx))
        if len(pixels) >= minimum:
            found.append(np.asarray(pixels, dtype=np.int32))
    return found


def _bfs(mask: np.ndarray, start: tuple[int, int]) -> np.ndarray:
    distances = np.full(mask.shape, -1, dtype=np.int32)
    distances[start] = 0
    queue = deque([start])
    while queue:
        y, x = queue.popleft()
        next_distance = int(distances[y, x]) + 1
        for dy, dx in NEIGHBORS:
            ny, nx = y + dy, x + dx
            if 0 <= ny < mask.shape[0] and 0 <= nx < mask.shape[1] and mask[ny, nx] and distances[ny, nx] < 0:
                distances[ny, nx] = next_distance
                queue.append((ny, nx))
    return distances


def _farthest(distances: np.ndarray) -> tuple[int, int]:
    maximum = int(distances.max())
    if maximum < 0:
        raise OracleAbstention("dark route is absent")
    y, x = np.argwhere(distances == maximum)[0]
    return int(y), int(x)


def _mask_center(mask: np.ndarray, label: str) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 300:
        raise OracleAbstention(f"{label}: checkpoint ink is missing")
    width = int(xs.max() - xs.min() + 1)
    height = int(ys.max() - ys.min() + 1)
    if not (28 <= width <= 36 and 28 <= height <= 36):
        raise OracleAbstention(f"{label}: checkpoint extent is malformed")
    return ((float(xs.min()) + float(xs.max())) / 2.0, (float(ys.min()) + float(ys.max())) / 2.0)


def _route_distances_at(
    mask: np.ndarray,
    from_a: np.ndarray,
    from_b: np.ndarray,
    center: tuple[float, float],
) -> tuple[float, float]:
    cx, cy = center
    y0, x0 = int(round(cy)), int(round(cx))
    choices = []
    for radius in range(0, 10):
        for y in range(max(0, y0 - radius), min(mask.shape[0], y0 + radius + 1)):
            for x in range(max(0, x0 - radius), min(mask.shape[1], x0 + radius + 1)):
                if mask[y, x] and from_a[y, x] >= 0 and from_b[y, x] >= 0:
                    choices.append(((x - cx) ** 2 + (y - cy) ** 2, int(from_a[y, x]), int(from_b[y, x])))
        if choices:
            break
    if not choices:
        raise OracleAbstention("a colored checkpoint is detached from the dark route")
    nearest = min(item[0] for item in choices)
    local = [item for item in choices if item[0] <= nearest + 1.0]
    return (float(np.median([item[1] for item in local])), float(np.median([item[2] for item in local])))


def decision_from_image(image: Image.Image) -> str:
    """Enumerate colored candidates and return the unique pixel-geodesic midpoint."""

    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if rgb.shape[:2] != (512, 768):
        raise OracleAbstention(f"unexpected raster size {image.size}")
    route = np.all(rgb == np.asarray(DARK_RGB, dtype=np.uint8), axis=2)
    components = _components(route, minimum=1)
    if len(components) != 1 or len(components[0]) < 8000:
        raise OracleAbstention(f"expected one long unbroken route, found {len(components)} components")
    pixels = components[0]
    if int(pixels[:, 1].max() - pixels[:, 1].min()) < 640 or int(pixels[:, 0].max() - pixels[:, 0].min()) < 300:
        raise OracleAbstention("dark route does not span the visual field")

    arbitrary = tuple(map(int, pixels[0]))
    endpoint_a = _farthest(_bfs(route, arbitrary))
    from_a = _bfs(route, endpoint_a)
    endpoint_b = _farthest(from_a)
    from_b = _bfs(route, endpoint_b)
    if int(from_a[endpoint_b]) < 1400:
        raise OracleAbstention("dark route is too short")

    gold = np.all(rgb == np.asarray(GOLD_RGB, dtype=np.uint8), axis=2)
    gold_components = _components(gold, minimum=80)
    if len(gold_components) != 2:
        raise OracleAbstention(f"expected two gold endpoint rings, found {len(gold_components)}")
    for component in gold_components:
        cy, cx = component.mean(axis=0)
        if min((cx - endpoint_a[1]) ** 2 + (cy - endpoint_a[0]) ** 2, (cx - endpoint_b[1]) ** 2 + (cy - endpoint_b[0]) ** 2) > 28.0**2:
            raise OracleAbstention("a gold ring does not mark a route endpoint")

    imbalances: dict[str, float] = {}
    for color, value in COLOR_RGB.items():
        mask = np.all(rgb == np.asarray(value, dtype=np.uint8), axis=2)
        center = _mask_center(mask, color)
        left, right = _route_distances_at(route, from_a, from_b, center)
        imbalances[color] = abs(left - right)
    ranked = sorted(imbalances.items(), key=lambda item: item[1])
    if ranked[0][1] > 24.0 or ranked[1][1] - ranked[0][1] < 100.0:
        raise OracleAbstention(f"route midpoint winner is ambiguous: {ranked}")
    return ranked[0][0]
