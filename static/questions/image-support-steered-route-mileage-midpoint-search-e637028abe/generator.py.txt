"""Deterministic latent world for global route-mileage midpoint search."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

WIDTH = 768
HEIGHT = 512
COLORS = ("red", "blue", "green", "purple")
PALETTE = {
    "red": (207, 66, 62),
    "blue": (45, 111, 181),
    "green": (49, 143, 91),
    "purple": (132, 76, 173),
}
DARK = (35, 40, 47)
GOLD = (221, 169, 43)
BACKGROUNDS = ((248, 247, 242), (244, 248, 247), (248, 245, 249))
ROUTE_WIDTH = 7
RING_OUTER = 17
RING_INNER = 9

_BASE_X = (48, 148, 276, 404, 532, 660, 724)
_BASE_Y = (84, 230, 108, 364, 188, 426)


def _route_length(points: list[list[int]]) -> int:
    total = 0
    for left, right in zip(points, points[1:]):
        dx = abs(int(right[0]) - int(left[0]))
        dy = abs(int(right[1]) - int(left[1]))
        if (dx == 0) == (dy == 0):
            raise ValueError("route sections must be nonzero and axis-aligned")
        total += dx + dy
    return total


def _point_at(points: list[list[int]], offset: int) -> tuple[int, int]:
    remaining = int(offset)
    total = _route_length(points)
    if not 0 <= remaining <= total:
        raise ValueError(f"checkpoint offset {remaining} outside route length {total}")
    for left, right in zip(points, points[1:]):
        x1, y1 = map(int, left)
        x2, y2 = map(int, right)
        length = abs(x2 - x1) + abs(y2 - y1)
        if remaining <= length:
            if x1 != x2:
                return (x1 + (remaining if x2 > x1 else -remaining), y1)
            return (x1, y1 + (remaining if y2 > y1 else -remaining))
        remaining -= length
    return tuple(map(int, points[-1]))


def _make_route(rng: random.Random, transform: int) -> list[list[int]]:
    xs = [base + 2 * rng.randint(-5, 5) for base in _BASE_X]
    ys = [base + 2 * rng.randint(-6, 6) for base in _BASE_Y]
    points = [
        [xs[0], ys[0]], [xs[1], ys[0]],
        [xs[1], ys[1]], [xs[2], ys[1]],
        [xs[2], ys[2]], [xs[3], ys[2]],
        [xs[3], ys[3]], [xs[4], ys[3]],
        [xs[4], ys[4]], [xs[5], ys[4]],
        [xs[5], ys[5]], [xs[6], ys[5]],
    ]
    if transform & 1:
        points = [[WIDTH - x, y] for x, y in points]
    if transform & 2:
        points = [[x, HEIGHT - y] for x, y in points]
    return points


def _offsets(total: int, target_rank: int, rng: random.Random) -> list[int]:
    bands = {
        0: ((0.61, 0.66), (0.74, 0.80), (0.88, 0.93)),
        1: ((0.12, 0.25), (0.68, 0.76), (0.85, 0.93)),
        2: ((0.08, 0.17), (0.28, 0.39), (0.77, 0.91)),
        3: ((0.07, 0.14), (0.21, 0.30), (0.37, 0.44)),
    }[target_rank]
    values = [int(round(rng.uniform(lo, hi) * total)) for lo, hi in bands]
    values.append(total // 2)
    if len(set(values)) != 4 or min(abs(a - b) for i, a in enumerate(values) for b in values[i + 1 :]) < 75:
        raise RuntimeError("checkpoint offsets lost their separation")
    return sorted(values)


def sample_scene(seed: int) -> dict:
    """Sample balanced answer identity, route rank, geometry, and reflection."""

    seed = int(seed)
    rng = random.Random(seed)
    route = _make_route(rng, (seed // 16) % 4)
    total = _route_length(route)
    if total % 2:
        raise AssertionError("even-coordinate route must have even total length")
    target_color = COLORS[seed % len(COLORS)]
    target_rank = (seed // 4) % 4
    offsets = _offsets(total, target_rank, rng)
    other_colors = [color for color in COLORS if color != target_color]
    rng.shuffle(other_colors)
    checkpoints = []
    other_index = 0
    for offset in offsets:
        if offset == total // 2:
            color = target_color
        else:
            color = other_colors[other_index]
            other_index += 1
        checkpoints.append({"color": color, "offset": offset})
    rng.shuffle(checkpoints)
    return {
        "route_points": route,
        "checkpoints": checkpoints,
        "background": list(BACKGROUNDS[(seed // 64) % len(BACKGROUNDS)]),
    }


def _disk(draw: ImageDraw.ImageDraw, center: tuple[int, int], radius: int, fill: tuple[int, int, int]) -> None:
    x, y = center
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=fill)


def render(scene: dict) -> Image.Image:
    """Render equal-style candidate rings behind one uninterrupted dark route."""

    points = [list(map(int, point)) for point in scene["route_points"]]
    checkpoints = list(scene["checkpoints"])
    background = tuple(map(int, scene["background"]))
    total = _route_length(points)
    image = Image.new("RGB", (WIDTH, HEIGHT), background)
    draw = ImageDraw.Draw(image)

    for endpoint in (tuple(points[0]), tuple(points[-1])):
        _disk(draw, endpoint, 20, GOLD)
        _disk(draw, endpoint, 12, background)
    for checkpoint in checkpoints:
        color = str(checkpoint["color"])
        center = _point_at(points, int(checkpoint["offset"]))
        _disk(draw, center, RING_OUTER, PALETTE[color])
        _disk(draw, center, RING_INNER, background)

    draw.line([tuple(point) for point in points], fill=DARK, width=ROUTE_WIDTH)
    for point in points:
        _disk(draw, tuple(point), ROUTE_WIDTH // 2, DARK)
    if total < 1400:
        raise ValueError("route is too short")
    return image


def _scores(scene: dict) -> dict[str, float]:
    total = _route_length(scene["route_points"])
    scores = {color: -1e9 for color in COLORS}
    for checkpoint in scene["checkpoints"]:
        color = str(checkpoint["color"])
        offset = int(checkpoint["offset"])
        scores[color] = -float(abs(offset - (total - offset)))
    return scores


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    ranked = sorted(scores.items(), key=lambda item: item[1], reverse=True)
    if ranked[0][1] != 0.0 or ranked[0][1] == ranked[1][1]:
        raise ValueError("scene has no unique exact route-mileage midpoint")
    return ranked[0][0]


def margin(scene: dict) -> float:
    values = sorted(_scores(scene).values(), reverse=True)
    return float(values[0] - values[1])


def is_quarantined(scene: dict) -> bool:
    try:
        colors = [str(item["color"]) for item in scene["checkpoints"]]
        return sorted(colors) != sorted(COLORS) or margin(scene) < 150.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse route parameterization and complement every route offset."""

    twin = copy.deepcopy(scene)
    total = _route_length(scene["route_points"])
    twin["route_points"] = list(reversed(scene["route_points"]))
    for checkpoint in twin["checkpoints"]:
        checkpoint["offset"] = total - int(checkpoint["offset"])
    return [("reverse_route_parameterization", twin)]
