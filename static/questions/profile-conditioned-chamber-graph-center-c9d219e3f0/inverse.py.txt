"""Independent image-only inverse arm for chamber-graph eccentricity."""

from __future__ import annotations

from collections import Counter, deque

import numpy as np
from PIL import Image

COLORS = {
    "orange": np.array((232, 93, 4)),
    "green": np.array((22, 139, 63)),
    "purple": np.array((119, 73, 194)),
}


def _components(passable: np.ndarray) -> tuple[np.ndarray, int]:
    height, width = passable.shape
    labels = np.full((height, width), -1, dtype=np.int32)
    next_label = 0
    for y in range(height):
        for x in range(width):
            if not passable[y, x] or labels[y, x] >= 0:
                continue
            labels[y, x] = next_label
            queue = deque([(y, x)])
            while queue:
                yy, xx = queue.popleft()
                for y2, x2 in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                    if 0 <= y2 < height and 0 <= x2 < width and passable[y2, x2] and labels[y2, x2] < 0:
                        labels[y2, x2] = next_label
                        queue.append((y2, x2))
            next_label += 1
    return labels, next_label


def _eccentricity(graph: dict[int, set[int]], start: int) -> int | None:
    distances = {start: 0}
    queue = deque([start])
    while queue:
        here = queue.popleft()
        for neighbor in graph.get(here, ()):
            if neighbor not in distances:
                distances[neighbor] = distances[here] + 1
                queue.append(neighbor)
    if len(distances) != len(graph):
        return None
    return max(distances.values())


def decision_from_image(image: Image.Image) -> str:
    # Downsampling is an independent robustness choice: 9-pixel walls remain
    # at least two samples thick while the graph extraction stays inexpensive.
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)[::4, ::4]
    black = np.max(arr, axis=2) < 70
    labels, count = _components(~black)
    if count < 4:
        return "abstain"
    outside = int(labels[0, 0])

    marker_labels: dict[str, int] = {}
    for name, rgb in COLORS.items():
        distance = np.sqrt(np.sum((arr - rgb) ** 2, axis=2))
        ys, xs = np.where(distance < 35)
        if len(xs) < 10:
            return "abstain"
        cy, cx = int(round(float(ys.mean()))), int(round(float(xs.mean())))
        label = int(labels[cy, cx])
        if label < 0 or label == outside:
            return "abstain"
        marker_labels[name] = label

    # A true wall side supplies many repeated opposite-component observations.
    # Junction/corner coincidences supply only a few and are filtered out.
    evidence: Counter[tuple[int, int]] = Counter()
    gap = 2
    height, width = black.shape
    for y, x in np.argwhere(black):
        if gap <= x < width - gap:
            a, b = int(labels[y, x - gap]), int(labels[y, x + gap])
            if a >= 0 and b >= 0 and a != b and outside not in (a, b):
                evidence[tuple(sorted((a, b)))] += 1
        if gap <= y < height - gap:
            a, b = int(labels[y - gap, x]), int(labels[y + gap, x])
            if a >= 0 and b >= 0 and a != b and outside not in (a, b):
                evidence[tuple(sorted((a, b)))] += 1
    graph: dict[int, set[int]] = {label: set() for label in range(count) if label != outside}
    for (a, b), votes in evidence.items():
        if votes >= 8:
            graph.setdefault(a, set()).add(b)
            graph.setdefault(b, set()).add(a)
    values = {name: _eccentricity(graph, label) for name, label in marker_labels.items()}
    if any(value is None for value in values.values()):
        return "abstain"
    best = min(values.values())
    matches = [name for name, value in values.items() if value == best]
    return matches[0] if len(matches) == 1 else "abstain"
