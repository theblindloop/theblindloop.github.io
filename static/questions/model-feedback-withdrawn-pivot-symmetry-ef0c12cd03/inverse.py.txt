"""Independent PNG-only inverse arm for the withdraw-pivot-symmetry world.

Recovers the ten square-cell ON/OFF states and the column the up-pointing arrow
marks directly from the final raster (structure-mask column bands for the cells
and the dark arrowhead's centroid below the row), removes the marked cell, and
reports whether the surviving nine-cell sequence is a left-right palindrome.
Uses no renderer geometry, scene parameters, prompt, or gold code; it operates
only on the pixels and structural/colour segmentation.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

N_CELLS = 10


def _structure_mask(arr: np.ndarray) -> np.ndarray:
    chroma = arr.max(axis=2).astype(np.int16) - arr.min(axis=2).astype(np.int16)
    grays = chroma <= 16
    notwhite = (255 - arr.mean(axis=2)) >= 10
    return grays & notwhite


def _columns_at(band: np.ndarray) -> list[tuple[int, int]]:
    cols = band.any(axis=0)
    trans = np.diff(cols.astype(np.int8))
    starts = np.flatnonzero(trans == 1)
    ends = np.flatnonzero(trans == -1) + 1
    if cols[0]:
        starts = np.concatenate(([0], starts))
    if cols[-1]:
        ends = np.concatenate((ends, [len(cols)]))
    pairs = [(int(s), int(e)) for s, e in zip(starts, ends) if int(e) - int(s) >= 20]
    pairs.sort()
    return pairs


def _dark_mask(arr: np.ndarray) -> np.ndarray:
    return arr.mean(axis=2) < 120


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    struct = _structure_mask(arr)
    walls = np.where(struct.any(axis=1))[0]
    if len(walls) == 0:
        raise ValueError("no cell structure found in image")
    top = int(walls.min())
    y = top
    while y < arr.shape[0] and struct[y].any():
        y += 1
    cell_center = (top + y) // 2
    band = struct[cell_center - 2 : cell_center + 3, :]
    pairs = _columns_at(band)
    if len(pairs) != N_CELLS:
        raise ValueError(f"expected {N_CELLS} cells, found {len(pairs)}")
    bits: list[int] = []
    centers: list[int] = []
    for start, end in pairs:
        x = (start + end) // 2
        centers.append(x)
        px = arr[cell_center, x].astype(np.int16)
        bits.append(1 if float(px.mean()) < 120 else 0)

    dark = _dark_mask(arr)
    below = np.zeros_like(dark)
    below[y + 8 :, :] = dark[y + 8 :, :]
    pixels = set(zip(*np.nonzero(below)))
    arrow_xs: list[int] = []
    arrow_ys: list[int] = []
    mark = None
    best = None
    if pixels:
        start = pixels.pop()
        stack = [start]
        cells = [start]
        while stack:
            cy, cx = stack.pop()
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if (ny, nx) in pixels:
                    pixels.remove((ny, nx))
                    cells.append((ny, nx))
                    stack.append((ny, nx))
        if len(cells) >= 40:
            acx = sum(x for _, x in cells) / len(cells)
            acy = sum(y for y, _ in cells) / len(cells)
            arrow_xs.append(acx)
            arrow_ys.append(acy)
        if arrow_xs:
            ax = arrow_xs[0]
            best = min(range(N_CELLS), key=lambda i: abs(centers[i] - ax))
            mark = best
    if mark is None:
        raise ValueError("no arrowhead found below the row")
    reduced = bits[:mark] + bits[mark + 1 :]
    return "yes" if reduced == reduced[::-1] else "no"
