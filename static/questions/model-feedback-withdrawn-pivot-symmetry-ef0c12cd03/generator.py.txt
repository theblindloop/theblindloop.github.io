"""Deterministic renderer and analytic arm for the withdraw-pivot-symmetry world.

A single horizontal row of ten square cells is drawn, each cell filled either
dark or pale.  A small solid arrowhead points up at exactly one chosen cell,
marking it as the pivot.

The reasoning program is a counterfactual intervention.  The ten-cell row is the
actually drawn world.  The question asks about a state that is NEVER drawn: the
shorter nine-cell row that WOULD result if the arrow-marked cell were withdrawn
from the row and the remaining cells slid together to close the gap, preserving
their left-to-right order.  Concretely the solver must remove the marked cell
and test whether the surviving nine-cell sequence reads the same left-to-right
as right-to-left (a palindrome).  The answer is carried only by that composed
mirrored comparison of the post-withdrawal row; no crop, no single cell, and no
count of the visible ten-cell row decides it.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 760
HEIGHT = 560
N_CELLS = 10
CELL = 54
GAP = 12
ROW_Y = 210
ARROW_TOP = 292
ARROW_BASE = 342
ARROW_HALFW = 17
BACKGROUND = (252, 252, 252)
OUTLINE = (120, 120, 120)
ON_FILL = (40, 40, 40)
OFF_FILL = (236, 236, 236)
ARROW_FILL = (40, 40, 40)


def _make_reduced(rng: random.Random, target: str) -> list[int]:
    """A 9-bit post-withdrawal row with the requested palindrome class."""
    if target == "yes":
        half = [rng.randint(0, 1) for _ in range(5)]
        reduced = half + half[-2::-1]
    else:
        reduced = [rng.randint(0, 1) for _ in range(N_CELLS - 1)]
        if reduced == reduced[::-1]:
            reduced[0] = 1 - reduced[0]
    return reduced


def is_palindrome(seq: list[int]) -> bool:
    return seq == seq[::-1]


def sample_scene(seed: int, target: str | None = None) -> dict:
    rng = random.Random(int(seed))
    if target is None:
        target = "yes" if rng.randint(0, 1) == 0 else "no"
    mark = rng.randint(0, N_CELLS - 1)
    reduced = _make_reduced(rng, target)
    marked_color = rng.randint(0, 1)
    cells = reduced[:mark] + [marked_color] + reduced[mark:]
    return {
        "canvas_size": [WIDTH, HEIGHT],
        "cells": cells,
        "mark": mark,
        "cell_px": CELL,
        "gap_px": GAP,
        "row_y": ROW_Y,
        "arrow_top": ARROW_TOP,
        "arrow_base": ARROW_BASE,
        "arrow_halfw": ARROW_HALFW,
    }


def _row_left() -> int:
    total = N_CELLS * CELL + (N_CELLS - 1) * GAP
    return (WIDTH - total) // 2


def render(scene: dict) -> Image.Image:
    width, height = (int(value) for value in scene["canvas_size"])
    cells = [int(value) for value in scene["cells"]]
    mark = int(scene["mark"])
    cell = int(scene["cell_px"])
    gap = int(scene["gap_px"])
    row_y = int(scene["row_y"])
    arrow_top = int(scene["arrow_top"])
    arrow_base = int(scene["arrow_base"])
    arrow_halfw = int(scene["arrow_halfw"])
    left = _row_left()
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    centers: list[int] = []
    for index, value in enumerate(cells):
        x = left + index * (cell + gap)
        cx = x + cell // 2
        centers.append(cx)
        draw.rectangle(
            (x, row_y, x + cell, row_y + cell),
            fill=ON_FILL if value == 1 else OFF_FILL,
            outline=OUTLINE,
            width=3,
        )
    pivot_cx = centers[mark]
    draw.polygon(
        [
            (pivot_cx, arrow_top),
            (pivot_cx - arrow_halfw, arrow_base),
            (pivot_cx + arrow_halfw, arrow_base),
        ],
        fill=ARROW_FILL,
    )
    return image


def _surviving_after_withdrawal(cells: list[int], mark: int) -> list[int]:
    return cells[:mark] + cells[mark + 1 :]


def analytic_gold(scene: dict) -> str:
    cells = [int(value) for value in scene["cells"]]
    mark = int(scene["mark"])
    surviving = _surviving_after_withdrawal(cells, mark)
    return "yes" if is_palindrome(surviving) else "no"


def margin(scene: dict) -> float:
    return 1.0


def is_quarantined(scene: dict) -> bool:
    cells = scene["cells"]
    mark = int(scene["mark"])
    if len(cells) != N_CELLS:
        return True
    if any(value not in (0, 1) for value in cells):
        return True
    if mark < 0 or mark >= N_CELLS:
        return True
    return False


def latent_symmetries(scene: dict) -> list:
    return []
