"""Independent pixel-only inverse for the visible shield area balance."""

from __future__ import annotations

import numpy as np
from PIL import Image

_CENTROID_THRESHOLD = 0.015
_MIN_FOREGROUND = 2500
_BACKGROUND_DELTA = 28.0
_MIN_FILL_RATIO = 0.68


class OracleAbstain(ValueError):
    """The PNG does not expose a sufficiently large foreground region."""


def _background(array: np.ndarray) -> np.ndarray:
    border = np.concatenate(
        (
            array[:3].reshape(-1, 3),
            array[-3:].reshape(-1, 3),
            array[:, :3].reshape(-1, 3),
            array[:, -3:].reshape(-1, 3),
        ),
        axis=0,
    )
    return np.median(border, axis=0).astype(np.float32)


def _foreground(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    background = _background(array)
    distance = np.linalg.norm(array - background, axis=2)
    return distance > _BACKGROUND_DELTA


def _centroid_gap(mask: np.ndarray) -> float:
    points = np.argwhere(mask)
    if len(points) < _MIN_FOREGROUND:
        raise OracleAbstain("no large connected shield foreground")
    ys = points[:, 0]
    xs = points[:, 1]
    span = float(xs.max() - xs.min() + 1)
    if span < 40.0:
        raise OracleAbstain("foreground span is below the visible scale floor")
    bbox_area = float((ys.max() - ys.min() + 1) * (xs.max() - xs.min() + 1))
    if float(len(points)) / bbox_area < _MIN_FILL_RATIO:
        raise OracleAbstain("foreground body is materially incomplete")
    centroid_x = float(xs.mean())
    bbox_midpoint = (float(xs.min()) + float(xs.max())) / 2.0
    return abs(centroid_x - bbox_midpoint) / span


def decision_from_image(image: Image.Image) -> str:
    """Measure the visible foreground centroid from the final PNG only."""

    gap = _centroid_gap(_foreground(image))
    return "yes" if gap <= _CENTROID_THRESHOLD else "no"
