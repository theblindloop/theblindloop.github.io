"""Deterministic raster world for a visible shield-balance question."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "shield-balance-centroid-renderer-0.1.0"
WIDTH = 640
HEIGHT = 640
BACKGROUND = "#ffffff"
CENTROID_THRESHOLD = 0.015
QUARANTINE_GAP = 0.0008
BASE_OFFSET = 0.0165

FILL_COLORS = (
    "#d94747",
    "#347bb7",
    "#3d9663",
    "#c58a22",
    "#7955a4",
    "#df783f",
)
BORDER_COLORS = ("#20242b", "#34495e", "#4a382d")


def _local_vertices(balance_class: int, bias_strength: float) -> list[tuple[float, float]]:
    """Return a convex shield outline in a fixed clockwise order.

    The centered contour is bilaterally mirrored. Counterfactuals keep the
    extreme left/right points fixed and narrow the opposite interior flank,
    so the visible bounding-box midpoint stays a stable reference while the
    connected region's centroid moves left or right.
    """

    points = [
        (0.0, -190.0),
        (88.0, -166.0),
        (135.0, -98.0),
        (126.0, -5.0),
        (111.0, 82.0),
        (61.0, 154.0),
        (0.0, 200.0),
        (-61.0, 154.0),
        (-111.0, 82.0),
        (-126.0, -5.0),
        (-135.0, -98.0),
        (-88.0, -166.0),
    ]
    strength = max(0.70, min(1.20, float(bias_strength)))
    if balance_class < 0:
        # Keep the right extreme fixed and make the right interior narrower.
        points[3] = (points[3][0] - 15.0 * strength, points[3][1])
        points[4] = (points[4][0] - 18.0 * strength, points[4][1])
        points[5] = (points[5][0] - 12.0 * strength, points[5][1])
    elif balance_class > 0:
        # Mirror operation: keep the left extreme fixed and narrow its flank.
        points[9] = (points[9][0] + 15.0 * strength, points[9][1])
        points[8] = (points[8][0] + 18.0 * strength, points[8][1])
        points[7] = (points[7][0] + 12.0 * strength, points[7][1])
    return points


def _scene_vertices(scene: dict) -> list[tuple[float, float]]:
    balance_class = int(scene["balance_class"])
    if balance_class not in (-1, 0, 1):
        raise ValueError("balance_class must be -1, 0, or 1")
    local = _local_vertices(balance_class, float(scene["bias_strength"]))
    center_x = float(scene["center_x"])
    center_y = float(scene["center_y"])
    scale = float(scene["scale"])
    points = [(center_x + x * scale, center_y + y * scale) for x, y in local]
    order = int(scene["vertex_order"]) % len(points)
    return points[order:] + points[:order]


def sample_scene(seed: int) -> dict:
    """Sample balanced controls and mirrored counterfactuals."""

    rng = random.Random(int(seed))
    balance_class = (-1, 0, 1)[int(seed) % 3]
    scale = rng.uniform(0.88, 1.08)
    center_x = WIDTH / 2.0 + rng.uniform(-38.0, 38.0)
    center_y = HEIGHT / 2.0 + rng.uniform(-26.0, 26.0)
    bias_strength = rng.uniform(0.99, 1.01)
    fill_color = FILL_COLORS[(int(seed) * 37 + 5) % len(FILL_COLORS)]
    border_color = BORDER_COLORS[(int(seed) * 19 + 1) % len(BORDER_COLORS)]
    border_width = 7 + (int(seed) % 3)
    return {
        "balance_class": balance_class,
        "center_x": round(center_x, 6),
        "center_y": round(center_y, 6),
        "scale": round(scale, 6),
        "bias_strength": round(bias_strength, 6),
        "vertex_order": 0,
        "fill_color": fill_color,
        "border_color": border_color,
        "border_width": border_width,
        "width": WIDTH,
        "height": HEIGHT,
        "background": BACKGROUND,
        "antialias": True,
    }


def render(scene: dict) -> Image.Image:
    """Render one filled shield from serialized visible scene fields."""

    # Reading balance_class here is substantive: it selects the visible
    # centered/control or mirrored counterfactual contour.
    if int(scene["balance_class"]) not in (-1, 0, 1):
        raise ValueError("unsupported balance class")
    width = int(scene["width"])
    height = int(scene["height"])
    supersample = 4 if bool(scene["antialias"]) else 1
    image = Image.new(
        "RGB", (width * supersample, height * supersample), scene["background"]
    )
    draw = ImageDraw.Draw(image)
    vertices = _scene_vertices(scene)
    outer = [(x * supersample, y * supersample) for x, y in vertices]
    draw.polygon(outer, fill=scene["border_color"])

    # A scaled inner copy leaves a continuous, high-contrast outline while
    # keeping the foreground one connected area-like component.
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    border = max(1.0, float(scene["border_width"]))
    inset = max(0.82, 1.0 - border / (235.0 * float(scene["scale"])))
    inner = [
        ((cx + (x - cx) * inset) * supersample,
         (cy + (y - cy) * inset) * supersample)
        for x, y in vertices
    ]
    draw.polygon(inner, fill=scene["fill_color"])
    if supersample > 1:
        image = image.resize((width, height), Image.Resampling.LANCZOS)
    return image


def analytic_gold(scene: dict) -> str:
    return "yes" if int(scene["balance_class"]) == 0 else "no"


def margin(scene: dict) -> float:
    """One-sided distance from the expected centroid gap to the visible threshold."""

    class_value = abs(int(scene["balance_class"]))
    expected_gap = BASE_OFFSET * float(scene["bias_strength"]) * class_value
    return float(abs(expected_gap - CENTROID_THRESHOLD))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Rotate the starting vertex of the same closed polygon representation."""

    twin = dict(scene)
    twin["vertex_order"] = (int(scene["vertex_order"]) + 1) % 12
    return [("cyclic_vertex_order", twin)]
