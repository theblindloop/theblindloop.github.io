"""Independent pixel-only oracle for the solid-badge sector world.

Receives only the final PNG. Never imports renderer/prompts/generate/verify and
never accepts latent scene data.

The oracle detects the dark disc strokes as connected components, measures each
disc's solidity (fill) and size from its bounding box, and identifies the unique
component that is both solid (high fill) and of standard size. It then reports
which quadrant of the fixed center cross contains that disc's centroid. If the
unique target is missing or ambiguous it abstains with "none".
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ACC_R_LO = 9.0
ACC_R_HI = 15.0
FILL_TH = 0.60


def _quadrant_of(x: float, y: float) -> str:
    if y < 256.0:
        return "top_left" if x < 256.0 else "top_right"
    return "bottom_left" if x < 256.0 else "bottom_right"


def _connected_components(mask: np.ndarray, min_area: int = 4):
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components = []
    for start_y, start_x in np.argwhere(mask):
        sy, sx = int(start_y), int(start_x)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        points = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if (
                    0 <= ny < height and 0 <= nx < width
                    and mask[ny, nx] and not seen[ny, nx]
                ):
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= min_area:
            points = np.asarray(points)
            ys = points[:, 0]
            xs = points[:, 1]
            y0, y1 = int(ys.min()), int(ys.max())
            x0, x1 = int(xs.min()), int(xs.max())
            bw = x1 - x0 + 1
            bh = y1 - y0 + 1
            area = float(len(points))
            fill = area / (bw * bh) if bw and bh else 0.0
            radius = min(bw, bh) / 2.0
            aspect = bh / bw if bw else 0.0
            components.append(
                {
                    "cy": float(np.mean(ys)),
                    "cx": float(np.mean(xs)),
                    "fill": fill,
                    "radius": radius,
                    "aspect": aspect,
                }
            )
    return components


def decision_from_image(image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    if arr.ndim != 3:
        return "none"
    r = arr[..., 0]
    g = arr[..., 1]
    b = arr[..., 2]
    dark = (r < 90) & (g < 90) & (b < 90)
    comps = _connected_components(dark)
    valid = []
    for c in comps:
        if c["aspect"] < 0.70 or c["aspect"] > 1.42:
            continue
        if c["fill"] < FILL_TH:
            continue
        if not (ACC_R_LO <= c["radius"] <= ACC_R_HI):
            continue
        valid.append(c)
    if len(valid) != 1:
        return "none"
    target = valid[0]
    return _quadrant_of(target["cx"], target["cy"])
