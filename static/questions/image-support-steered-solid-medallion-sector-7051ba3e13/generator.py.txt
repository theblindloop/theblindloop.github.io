"""Solid-badge sector search renderer.

Latent scene: a set of dark discs scattered on a white canvas, divided into four
quadrants by a light-gray cross at the center. Exactly one disc is the "solid
badge": a fully filled disc of standard size. Every other disc is a distractor
that fails at least one conjunct: a hollow outline ring (fails solidity), a
solid disc that is too small, or a solid disc that is too large. The question
asks which quadrant contains the unique solid badge.

The decisive evidence is exactly the one compact solid-badge disc: ablating that
small local region makes the pixel oracle lose the unique target and abstain,
while removing any distractor or the gray cross leaves the answer unchanged.
"""

from __future__ import annotations

import dataclasses
import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "solid_medallion_sector-0.1.0"

WIDTH = 512
HEIGHT = 512
BG = (255, 255, 255)
DISK_INK = (45, 45, 45)
DIVIDER = (150, 150, 150)
CROSS_W = 3

CX = WIDTH // 2
CY = HEIGHT // 2

# Target (solid badge) size band and oracle acceptance radius band.
TARGET_R_LO = 9
TARGET_R_HI = 11
ACC_R_LO = 8.0
ACC_R_HI = 15.0
FILL_TH = 0.60
OUTLINE_T = 5

GRID_N = 8
CELL = WIDTH // GRID_N  # 64
GRID_MARGIN = 28
CROSS_MARGIN = 15
_SMALL_R = (4.0, 6.0)
_LARGE_R = (22.0, 34.0)
_HOLLOW_R = (10.0, 30.0)


@dataclasses.dataclass(frozen=True)
class Disc:
    cx: float
    cy: float
    r: float
    kind: str  # "target", "hollow", "small", "large"


def _quadrant_of(x: float, y: float) -> str:
    if y < CY:
        return "top_left" if x < CX else "top_right"
    return "bottom_left" if x < CX else "bottom_right"


def _sample_target_center(rng: random.Random):
    for _ in range(400):
        gi = rng.randrange(GRID_N)
        gj = rng.randrange(GRID_N)
        x = gj * CELL + rng.uniform(GRID_MARGIN, CELL - GRID_MARGIN)
        y = gi * CELL + rng.uniform(GRID_MARGIN, CELL - GRID_MARGIN)
        if abs(x - CX) >= CROSS_MARGIN and abs(y - CY) >= CROSS_MARGIN:
            return x, y
    raise RuntimeError("could not place target")


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    discs = []

    tr = rng.uniform(TARGET_R_LO, TARGET_R_HI)
    tx, ty = _sample_target_center(rng)
    discs.append({"cx": tx, "cy": ty, "r": tr, "kind": "target"})

    # Ensure the two single-conjunct distractor families are always present.
    kinds = ["hollow", "large"]
    extra = rng.randrange(0, 3)
    for _ in range(extra):
        kinds.append(rng.choice(["hollow", "small", "large"]))
    rng.shuffle(kinds)

    placed = list(discs)
    for kind in kinds:
        if kind == "hollow":
            r = rng.uniform(*_HOLLOW_R)
        elif kind == "small":
            r = rng.uniform(*_SMALL_R)
        else:  # large
            r = rng.uniform(*_LARGE_R)
        cx = cy = None
        for _ in range(600):
            px = rng.uniform(r + OUTLINE_T + 4, WIDTH - r - OUTLINE_T - 4)
            py = rng.uniform(r + OUTLINE_T + 4, HEIGHT - r - OUTLINE_T - 4)
            if all(
                math.hypot(px - d["cx"], py - d["cy"]) >= (r + d["r"] + 9.0)
                for d in placed
            ):
                cx, cy = px, py
                break
        if cx is None:
            continue
        discs.append({"cx": cx, "cy": cy, "r": r, "kind": kind})
        placed.append(discs[-1])

    rng.shuffle(discs)
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "discs": [dict(d, cx=round(d["cx"], 3), cy=round(d["cy"], 3)) for d in discs],
    }


def render(scene) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)

    # light-gray quadrant cross (nondiagnostic for the oracle)
    draw.line([(CX, 0), (CX, HEIGHT)], fill=DIVIDER, width=CROSS_W)
    draw.line([(0, CY), (WIDTH, CY)], fill=DIVIDER, width=CROSS_W)

    for d in scene["discs"]:
        x0, y0 = d["cx"] - d["r"], d["cy"] - d["r"]
        x1, y1 = d["cx"] + d["r"], d["cy"] + d["r"]
        if d["kind"] == "hollow":
            draw.ellipse([x0, y0, x1, y1], outline=DISK_INK, width=OUTLINE_T)
        else:
            draw.ellipse([x0, y0, x1, y1], fill=DISK_INK)
    return img


def _target_disc(scene):
    hits = [d for d in scene["discs"] if d["kind"] == "target"]
    return hits[0] if len(hits) == 1 else None


def analytic_gold(scene) -> str:
    t = _target_disc(scene)
    if t is None:
        return "none"
    return _quadrant_of(t["cx"], t["cy"])


def margin(scene) -> float:
    t = _target_disc(scene)
    if t is None:
        return 0.0
    r = t["r"]
    fill_gap = 0.785 - FILL_TH  # solid disc fill vs threshold
    size_gap = min(r - ACC_R_LO, ACC_R_HI - r)
    cross_gap = min(abs(t["cx"] - CX), abs(t["cy"] - CY))
    return round(min(fill_gap, size_gap, cross_gap / 10.0), 6)


def is_quarantined(scene) -> bool:
    t = _target_disc(scene)
    if t is None:
        return True
    r = t["r"]
    if not (ACC_R_LO <= r <= ACC_R_HI):
        return True
    if margin(scene) <= 0.05:
        return True
    return abs(t["cx"] - CX) < CROSS_MARGIN or abs(t["cy"] - CY) < CROSS_MARGIN


def latent_symmetries(scene):
    return iter(())
