"""Renderer and analytic world for threshold-qualified bubble markers.

The image contains four same-colour filled bubbles and a visible dashed cross.
The small white geometric marker inside each bubble is its identity.  The gold
first applies two strict, visible half-plane tests to bubble centres and then
selects the largest qualifying radius.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "bubble_threshold_marker-0.1.0"
WIDTH = 640
HEIGHT = 480
BACKGROUND = (250, 252, 255)
BUBBLE_FILL = (39, 147, 167)
THRESHOLD_INK = (70, 83, 105)
MARKER_INK = BACKGROUND
MARKERS = ("circle", "square", "triangle", "diamond")
SCALE = 4
MARKER_AREA = 180.0
MIN_DECISION_GAP = 5.0
MIN_SEPARATION_GAP = 12.0


def _marker_parameters() -> dict[str, float]:
    circle_radius = math.sqrt(MARKER_AREA / math.pi)
    square_half = math.sqrt(MARKER_AREA) / 2.0
    triangle_side = math.sqrt(4.0 * MARKER_AREA / math.sqrt(3.0))
    diamond_half_diagonal = math.sqrt(MARKER_AREA / 2.0)
    return {
        "circle_radius": circle_radius,
        "square_half": square_half,
        "triangle_side": triangle_side,
        "diamond_half_diagonal": diamond_half_diagonal,
    }


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced, well-separated scene from one integer seed."""

    rng = random.Random(int(seed))
    threshold_x = round(rng.uniform(235.0, 295.0), 3)
    threshold_y = round(rng.uniform(255.0, 300.0), 3)
    target = MARKERS[int(seed) % len(MARKERS)]
    outside = MARKERS[(int(seed) + 1) % len(MARKERS)]
    qualifying = [shape for shape in MARKERS if shape != outside]

    radii: dict[str, float] = {}
    for shape in qualifying:
        if shape == target:
            radii[shape] = round(rng.uniform(32.0, 36.0), 3)
        else:
            radii[shape] = round(rng.uniform(20.0, 25.5), 3)
    radii[outside] = round(rng.uniform(32.0, 38.0), 3)

    # Three generous slots in the upper-right quadrant keep every qualifying
    # centre well away from both threshold lines while allowing marker identity
    # and position to vary independently.
    slots = [
        (threshold_x + 92.0, 82.0),
        (threshold_x + 215.0, 88.0),
        (threshold_x + 145.0, threshold_y - 93.0),
    ]
    rng.shuffle(qualifying)
    centers: dict[str, tuple[float, float]] = {}
    for shape, (slot_x, slot_y) in zip(qualifying, slots, strict=True):
        centers[shape] = (
            round(slot_x + rng.uniform(-10.0, 10.0), 3),
            round(slot_y + rng.uniform(-9.0, 9.0), 3),
        )
    centers[outside] = (
        round(threshold_x - 115.0 + rng.uniform(-10.0, 10.0), 3),
        round(threshold_y + 105.0 + rng.uniform(-10.0, 10.0), 3),
    )

    return {
        "threshold_x": threshold_x,
        "threshold_y": threshold_y,
        "bubbles": {
            shape: {
                "cx": centers[shape][0],
                "cy": centers[shape][1],
                "radius": radii[shape],
            }
            for shape in MARKERS
        },
    }


def _scaled_box(cx: float, cy: float, radius: float) -> tuple[int, int, int, int]:
    return (
        round((cx - radius) * SCALE),
        round((cy - radius) * SCALE),
        round((cx + radius) * SCALE),
        round((cy + radius) * SCALE),
    )


def _draw_marker(draw: ImageDraw.ImageDraw, shape: str, cx: float, cy: float) -> None:
    p = _marker_parameters()
    x = cx * SCALE
    y = cy * SCALE
    if shape == "circle":
        r = p["circle_radius"] * SCALE
        draw.ellipse((round(x - r), round(y - r), round(x + r), round(y + r)), fill=MARKER_INK)
    elif shape == "square":
        h = p["square_half"] * SCALE
        draw.rectangle((round(x - h), round(y - h), round(x + h), round(y + h)), fill=MARKER_INK)
    elif shape == "triangle":
        side = p["triangle_side"] * SCALE
        height = side * math.sqrt(3.0) / 2.0
        points = (
            (round(x), round(y - 2.0 * height / 3.0)),
            (round(x - side / 2.0), round(y + height / 3.0)),
            (round(x + side / 2.0), round(y + height / 3.0)),
        )
        draw.polygon(points, fill=MARKER_INK)
    elif shape == "diamond":
        d = p["diamond_half_diagonal"] * SCALE
        draw.polygon(
            ((round(x), round(y - d)), (round(x + d), round(y)),
             (round(x), round(y + d)), (round(x - d), round(y))),
            fill=MARKER_INK,
        )
    else:
        raise ValueError(f"unknown marker shape {shape!r}")


def _draw_dashed_cross(draw: ImageDraw.ImageDraw, threshold_x: float, threshold_y: float) -> None:
    x = round(threshold_x * SCALE)
    y = round(threshold_y * SCALE)
    width = 3 * SCALE
    dash = 13 * SCALE
    gap = 9 * SCALE
    for start in range(24 * SCALE, (HEIGHT - 24) * SCALE, dash + gap):
        draw.line((x, start, x, min(start + dash, (HEIGHT - 24) * SCALE)), fill=THRESHOLD_INK, width=width)
    for start in range(24 * SCALE, (WIDTH - 24) * SCALE, dash + gap):
        draw.line((start, y, min(start + dash, (WIDTH - 24) * SCALE), y), fill=THRESHOLD_INK, width=width)


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_dashed_cross(draw, float(scene["threshold_x"]), float(scene["threshold_y"]))
    bubbles = scene["bubbles"]
    for shape in MARKERS:
        spec = bubbles[shape]
        draw.ellipse(_scaled_box(float(spec["cx"]), float(spec["cy"]), float(spec["radius"])), fill=BUBBLE_FILL)
        _draw_marker(draw, shape, float(spec["cx"]), float(spec["cy"]))
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def _qualifying(scene: dict[str, Any]) -> list[tuple[str, dict[str, float]]]:
    tx = float(scene["threshold_x"])
    ty = float(scene["threshold_y"])
    return [
        (shape, scene["bubbles"][shape])
        for shape in MARKERS
        if float(scene["bubbles"][shape]["cx"]) > tx
        and float(scene["bubbles"][shape]["cy"]) < ty
    ]


def analytic_gold(scene: dict[str, Any]) -> str:
    qualifying = _qualifying(scene)
    if len(qualifying) < 2:
        return "quarantine"
    ordered = sorted(qualifying, key=lambda item: float(item[1]["radius"]), reverse=True)
    return ordered[0][0]


def margin(scene: dict[str, Any]) -> float:
    qualifying = _qualifying(scene)
    if len(qualifying) < 2:
        return -1.0
    radii = sorted((float(spec["radius"]) for _, spec in qualifying), reverse=True)
    return radii[0] - radii[1]


def is_quarantined(scene: dict[str, Any]) -> bool:
    qualifying = _qualifying(scene)
    if len(qualifying) != 3 or margin(scene) < MIN_DECISION_GAP:
        return True
    tx = float(scene["threshold_x"])
    ty = float(scene["threshold_y"])
    bubbles = scene["bubbles"]
    for shape in MARKERS:
        spec = bubbles[shape]
        if min(abs(float(spec["cx"]) - tx), abs(float(spec["cy"]) - ty)) < 45.0:
            return True
    for index, left_shape in enumerate(MARKERS):
        left = bubbles[left_shape]
        for right_shape in MARKERS[index + 1:]:
            right = bubbles[right_shape]
            distance = math.hypot(float(left["cx"]) - float(right["cx"]), float(left["cy"]) - float(right["cy"]))
            clearance = distance - float(left["radius"]) - float(right["radius"])
            if clearance < MIN_SEPARATION_GAP:
                return True
    return False


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
