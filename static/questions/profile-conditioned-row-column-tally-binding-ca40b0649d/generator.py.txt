"""Row/column tally correspondence world.

An empty 5x5 grid is shown with a red dot tally beside every row and a blue dot
tally under every column.  Three filled 5x5 candidate grids are drawn below.
Exactly one candidate reproduces both tallies; each other candidate reproduces
only one of the two marginals, so neither tally alone decides the answer.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WORLD_VERSION = "row_column_tally_binding-0.1.0"

SIZE = 5
CELL = 40
GRID_PX = SIZE * CELL

CANVAS = (980, 760)
TARGET_XY = (60, 60)
ROW_DOTS_X = 300
COL_DOTS_Y = 300
DOT_PITCH = 26
DOT_R = 7
PANEL_Y = 500
PANEL_X = (60, 380, 700)
LETTER_Y = 466

WHITE = (255, 255, 255)
BLACK = (0, 0, 0)
GRIDLINE = (205, 205, 205)
GREEN = (34, 139, 34)
RED = (220, 30, 30)
BLUE = (30, 60, 200)
GRAY = (90, 90, 90)

SLOTS = ("a", "b", "c")


# ---------------------------------------------------------------- scene


def _tallies(cells):
    rows = [0] * SIZE
    cols = [0] * SIZE
    for r, c in cells:
        rows[r] += 1
        cols[c] += 1
    return rows, cols


def _l1(cells, rows0, cols0):
    rows, cols = _tallies(cells)
    return sum(abs(a - b) for a, b in zip(rows, rows0)) + sum(
        abs(a - b) for a, b in zip(cols, cols0)
    )


def _perturb(cells, axis, swaps, rng):
    """Move filled cells inside lines of `axis`, preserving that axis' tally."""
    cur = {tuple(cell) for cell in cells}
    for _ in range(swaps):
        options = []
        for line in range(SIZE):
            filled = [i for i in range(SIZE) if _at(axis, line, i) in cur]
            empty = [i for i in range(SIZE) if _at(axis, line, i) not in cur]
            if filled and empty:
                options.append((line, filled, empty))
        if not options:
            return None
        line, filled, empty = options[rng.randrange(len(options))]
        src = filled[rng.randrange(len(filled))]
        dst = empty[rng.randrange(len(empty))]
        cur.discard(_at(axis, line, src))
        cur.add(_at(axis, line, dst))
    return sorted(cur)


def _perturb_to(cells, axis, want, rows0, cols0, rng):
    """Perturbation preserving one marginal and missing the other by `want` dots."""
    for _ in range(80):
        moved = _perturb(cells, axis, 1 if want == 2 else 2, rng)
        if moved is not None and _l1(moved, rows0, cols0) == want:
            return moved
    return None


def _at(axis, line, index):
    return (line, index) if axis == "row" else (index, line)


def sample_scene(seed: int) -> dict:
    """Sample one balanced scene; the answer slot is drawn independently."""
    rng = random.Random(seed)
    while True:
        total = rng.randint(9, 13)
        target = sorted(divmod(k, SIZE) for k in rng.sample(range(SIZE * SIZE), total))
        rows0, cols0 = _tallies(target)
        if min(rows0) < 1 or min(cols0) < 1:
            continue
        want_rows = 2 if rng.random() < 0.3 else 4
        want_cols = 2 if rng.random() < 0.3 else 4
        keep_rows = _perturb_to(target, "row", want_rows, rows0, cols0, rng)
        keep_cols = _perturb_to(target, "col", want_cols, rows0, cols0, rng)
        if keep_rows is None or keep_cols is None:
            continue
        if keep_rows == target or keep_cols == target or keep_rows == keep_cols:
            continue
        answer_slot = rng.randrange(3)
        others = [i for i in range(3) if i != answer_slot]
        rng.shuffle(others)
        panels = [None, None, None]
        panels[answer_slot] = [list(cell) for cell in target]
        panels[others[0]] = [list(cell) for cell in keep_rows]
        panels[others[1]] = [list(cell) for cell in keep_cols]
        return {
            "target": [list(cell) for cell in target],
            "panels": panels,
            "answer_slot": answer_slot,
        }


# ---------------------------------------------------------------- gold


def _panel_matches(panel, rows0, cols0):
    rows, cols = _tallies([tuple(cell) for cell in panel])
    return rows == rows0 and cols == cols0


def analytic_gold(scene: dict) -> str:
    """Slot letter of the unique candidate reproducing both drawn tallies."""
    rows0, cols0 = _tallies([tuple(cell) for cell in scene["target"]])
    hits = [i for i, panel in enumerate(scene["panels"]) if _panel_matches(panel, rows0, cols0)]
    if len(hits) != 1:
        raise ValueError("scene does not have a unique agreeing candidate")
    return SLOTS[hits[0]]


def margin(scene: dict) -> float:
    """Runner-up tally discrepancy, in pixels of drawn dot pitch."""
    rows0, cols0 = _tallies([tuple(cell) for cell in scene["target"]])
    gaps = []
    for i, panel in enumerate(scene["panels"]):
        if _panel_matches(panel, rows0, cols0):
            continue
        gaps.append(_l1([tuple(cell) for cell in panel], rows0, cols0))
    return float(DOT_PITCH * min(gaps))


def is_quarantined(scene: dict) -> bool:
    """Minimal discrete evidence: runner-up off by one dot in only two tallies."""
    return margin(scene) < 2.5 * DOT_PITCH


def latent_symmetries(scene: dict) -> list:
    """Label-preserving aliases: the cell lists carry an arbitrary order."""
    reversed_target = dict(scene)
    reversed_target = {
        "target": list(reversed(scene["target"])),
        "panels": [list(p) for p in scene["panels"]],
        "answer_slot": scene["answer_slot"],
    }
    reversed_panels = {
        "target": [list(cell) for cell in scene["target"]],
        "panels": [list(reversed(p)) for p in scene["panels"]],
        "answer_slot": scene["answer_slot"],
    }
    return [
        ("target_cell_reordering", reversed_target),
        ("panel_cell_reordering", reversed_panels),
    ]


# ---------------------------------------------------------------- render


def _grid(draw, x0, y0, cells=None):
    for k in range(1, SIZE):
        draw.line([(x0 + k * CELL, y0), (x0 + k * CELL, y0 + GRID_PX)], fill=GRIDLINE, width=1)
        draw.line([(x0, y0 + k * CELL), (x0 + GRID_PX, y0 + k * CELL)], fill=GRIDLINE, width=1)
    if cells:
        for r, c in sorted(tuple(cell) for cell in cells):
            draw.rectangle(
                [
                    x0 + c * CELL + 6,
                    y0 + r * CELL + 6,
                    x0 + (c + 1) * CELL - 7,
                    y0 + (r + 1) * CELL - 7,
                ],
                fill=GREEN,
            )
    draw.rectangle([x0, y0, x0 + GRID_PX, y0 + GRID_PX], outline=BLACK, width=3)


def _dot(draw, cx, cy, color):
    draw.ellipse([cx - DOT_R, cy - DOT_R, cx + DOT_R, cy + DOT_R], fill=color)


def render(scene: dict) -> Image.Image:
    """Deterministic raster; drawing order is canonical in the cell lists."""
    image = Image.new("RGB", CANVAS, WHITE)
    draw = ImageDraw.Draw(image)
    tx, ty = TARGET_XY
    _grid(draw, tx, ty, None)
    rows0, cols0 = _tallies([tuple(cell) for cell in scene["target"]])
    for r, count in enumerate(rows0):
        cy = ty + r * CELL + CELL // 2
        for i in range(count):
            _dot(draw, ROW_DOTS_X + i * DOT_PITCH + DOT_PITCH // 2, cy, RED)
    for c, count in enumerate(cols0):
        cx = tx + c * CELL + CELL // 2
        for i in range(count):
            _dot(draw, cx, COL_DOTS_Y + i * DOT_PITCH + DOT_PITCH // 2, BLUE)
    for slot, px in enumerate(PANEL_X):
        _grid(draw, px, PANEL_Y, scene["panels"][slot])
        draw.text((px + GRID_PX // 2 - 3, LETTER_Y), "ABC"[slot], fill=GRAY)
    return image
