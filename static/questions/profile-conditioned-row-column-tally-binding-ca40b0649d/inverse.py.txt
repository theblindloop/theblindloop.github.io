"""Pixel-only decision arm.

Receives one rendered PNG and nothing else.  It locates the black panel frames
by ink projection, reads each candidate grid's filled cells, counts the red and
blue tally dots, and returns the slot whose row and column counts both agree.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ABSTAIN = "abstain"
SLOTS = ("a", "b", "c")
GRID_N = 5


def _mask(arr, target, tol=60):
    diff = np.abs(arr.astype(np.int16) - np.array(target, dtype=np.int16))
    return (diff.max(axis=2) <= tol)


def _runs(flags, min_len=1):
    runs = []
    start = None
    for i, on in enumerate(flags):
        if on and start is None:
            start = i
        elif not on and start is not None:
            if i - start >= min_len:
                runs.append((start, i - 1))
            start = None
    if start is not None and len(flags) - start >= min_len:
        runs.append((start, len(flags) - 1))
    return runs


def _dot_counts(mask, axis):
    """Counts of dot blobs per band; bands are separated along `axis`."""
    if axis == "row":
        bands = _runs(mask.any(axis=1), min_len=4)
        if len(bands) != GRID_N:
            return None
        return [len(_runs(mask[a : b + 1, :].any(axis=0), min_len=3)) for a, b in bands]
    bands = _runs(mask.any(axis=0), min_len=4)
    if len(bands) != GRID_N:
        return None
    return [len(_runs(mask[:, a : b + 1].any(axis=1), min_len=3)) for a, b in bands]


def _frames(black):
    """Black frame boxes: horizontal ink bands, split by their top edges."""
    boxes = []
    for top, bottom in _runs(black.any(axis=1), min_len=10):
        for left, right in _runs(black[top + 1, :], min_len=10):
            boxes.append((left, top, right, bottom))
    return boxes


def _cells(green, box):
    left, top, right, bottom = box
    inner = green[top + 4 : bottom - 3, left + 4 : right - 3]
    height, width = inner.shape
    if height < GRID_N * 6 or width < GRID_N * 6:
        return None
    cells = set()
    for r in range(GRID_N):
        for c in range(GRID_N):
            block = inner[
                (r * height) // GRID_N : ((r + 1) * height) // GRID_N,
                (c * width) // GRID_N : ((c + 1) * width) // GRID_N,
            ]
            if block.sum() > 0.25 * block.size:
                cells.add((r, c))
    return cells


def decision_from_image(image: "Image.Image") -> str:
    """Return 'a', 'b', 'c', or 'abstain' when the evidence is not readable."""
    arr = np.asarray(image.convert("RGB"))
    black = _mask(arr, (0, 0, 0), tol=70)
    red = _mask(arr, (220, 30, 30), tol=60)
    blue = _mask(arr, (30, 60, 200), tol=60)
    green = _mask(arr, (34, 139, 34), tol=60)

    row_counts = _dot_counts(red, "row")
    col_counts = _dot_counts(blue, "col")
    if row_counts is None or col_counts is None:
        return ABSTAIN
    if sum(row_counts) != sum(col_counts):
        return ABSTAIN

    boxes = _frames(black)
    if len(boxes) != 4:
        return ABSTAIN
    boxes.sort(key=lambda b: b[1])
    panels = sorted(boxes[1:], key=lambda b: b[0])
    if len({(b[3] - b[1]) // 20 for b in panels}) != 1:
        return ABSTAIN

    hits = []
    for slot, box in enumerate(panels):
        cells = _cells(green, box)
        if cells is None:
            return ABSTAIN
        rows = [0] * GRID_N
        cols = [0] * GRID_N
        for r, c in cells:
            rows[r] += 1
            cols[c] += 1
        if rows == row_counts and cols == col_counts:
            hits.append(slot)
    if len(hits) != 1:
        return ABSTAIN
    return SLOTS[hits[0]]
