"""Deterministic port-level socket-weave connectivity world."""

from __future__ import annotations

import random
from collections import Counter, deque
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT, SCALE = 1200, 820, 2
BACKGROUND = (238, 241, 245)
PAPER = (255, 255, 255)
INK = (38, 44, 52)
MUTED = (166, 174, 184)
START_COLOR = (30, 143, 86)
TERMINAL_COLOR = (224, 146, 28)
BADGE_COLORS = {
    "red": (211, 61, 67),
    "blue": (45, 100, 205),
    "orange": (231, 116, 31),
    "violet": (137, 75, 186),
    "teal": (25, 151, 157),
}
BADGE_ORDER = tuple(f"{name}_{shape}" for name in BADGE_COLORS for shape in ("circle", "square", "diamond"))
CARD_CENTERS = tuple((200 + 400 * column, 225 + 240 * row) for row in range(3) for column in range(3))
PORT_OFFSETS = ((-145, -38), (-145, 38), (145, -38), (145, 38))
STRAIGHT = ((0, 2), (1, 3))
CROSS = ((0, 3), (1, 2))
MARKER_SLOTS = {
    "START": (0, 0), "A": (1, 1), "B": (2, 2),
    "C": (3, 3), "D": (4, 0), "E": (5, 2),
}
ANSWERS = ("A", "B", "C", "D", "E")


class _Union:
    def __init__(self, n: int) -> None:
        self.parent = list(range(n))

    def find(self, value: int) -> int:
        while self.parent[value] != value:
            self.parent[value] = self.parent[self.parent[value]]
            value = self.parent[value]
        return value

    def join(self, left: int, right: int) -> None:
        a, b = self.find(left), self.find(right)
        if a != b:
            self.parent[max(a, b)] = min(a, b)


def _wire_pairs(mode: str) -> tuple[tuple[int, int], tuple[int, int]]:
    return STRAIGHT if mode == "straight" else CROSS


def _coarse_connected(remote_edges: list[tuple[tuple[int, int], tuple[int, int]]]) -> bool:
    union = _Union(9)
    for (left_plate, _), (right_plate, _) in remote_edges:
        union.join(left_plate, right_plate)
    return len({union.find(plate) for plate in range(9)}) == 1


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = ANSWERS[int(seed) % len(ANSWERS)]
    modes = [rng.choice(("straight", "cross")) for _ in range(9)]
    if len(set(modes)) == 1:
        modes[int(seed) % 9] = "cross" if modes[0] == "straight" else "straight"

    segments: list[list[tuple[int, int]]] = []
    slot_to_segment: dict[tuple[int, int], int] = {}
    for plate, mode in enumerate(modes):
        for pair in _wire_pairs(mode):
            index = len(segments)
            segment = [(plate, pair[0]), (plate, pair[1])]
            segments.append(segment)
            for slot in segment:
                slot_to_segment[slot] = index

    special_segment: dict[str, int] = {
        label: slot_to_segment[slot] for label, slot in MARKER_SLOTS.items()
    }
    if len(set(special_segment.values())) != 6:
        raise RuntimeError("endpoint markers must occupy distinct cord segments")
    ordinary_segments = [index for index in range(18) if index not in special_segment.values()]
    remaining = [label for label in ANSWERS if label != target]

    for _ in range(20000):
        rng.shuffle(remaining)
        endpoint_pairs = [("START", target), (remaining[0], remaining[1]), (remaining[2], remaining[3])]
        interiors = ordinary_segments[:]
        rng.shuffle(interiors)
        remote_edges: list[tuple[tuple[int, int], tuple[int, int]]] = []
        valid = True
        for path_index, (left_label, right_label) in enumerate(endpoint_pairs):
            nodes = [special_segment[left_label], *interiors[path_index * 4:(path_index + 1) * 4],
                     special_segment[right_label]]
            oriented: list[tuple[tuple[int, int], tuple[int, int]]] = []
            for node_index, node in enumerate(nodes):
                slots = segments[node][:]
                marker_slot = None
                for label in (left_label, right_label):
                    if MARKER_SLOTS[label] in slots:
                        marker_slot = MARKER_SLOTS[label]
                if marker_slot is not None:
                    free = slots[0] if slots[1] == marker_slot else slots[1]
                    oriented.append((free, free))
                else:
                    if rng.randrange(2):
                        slots.reverse()
                    oriented.append((slots[0], slots[1]))
            for index in range(len(oriented) - 1):
                edge = (oriented[index][1], oriented[index + 1][0])
                if edge[0][0] == edge[1][0]:
                    valid = False
                    break
                remote_edges.append(edge)
            if not valid:
                break
        if valid and len(remote_edges) == 15 and _coarse_connected(remote_edges):
            break
    else:
        raise RuntimeError("could not sample a balanced connected socket weave")

    identities = list(BADGE_ORDER)
    rng.shuffle(identities)
    plates = [{"ports": [None, None, None, None],
               "wires": [list(pair) for pair in _wire_pairs(mode)]} for mode in modes]
    for label, (plate, port) in MARKER_SLOTS.items():
        plates[plate]["ports"][port] = f"marker:{label}"
    for identity, (left, right) in zip(identities, remote_edges, strict=True):
        plates[left[0]]["ports"][left[1]] = identity
        plates[right[0]]["ports"][right[1]] = identity
    if any(value is None for plate in plates for value in plate["ports"]):
        raise RuntimeError("unassigned port")
    return {"plates": plates}


def _adjacency(scene: dict) -> dict[tuple[int, int], set[tuple[int, int]]]:
    adjacency = {(plate, port): set() for plate in range(9) for port in range(4)}
    locations: dict[str, list[tuple[int, int]]] = {}
    for plate_index, plate in enumerate(scene["plates"]):
        for left, right in plate["wires"]:
            a, b = (plate_index, int(left)), (plate_index, int(right))
            adjacency[a].add(b)
            adjacency[b].add(a)
        for port, value in enumerate(plate["ports"]):
            if not str(value).startswith("marker:"):
                locations.setdefault(str(value), []).append((plate_index, port))
    for identity in BADGE_ORDER:
        pair = locations.get(identity, [])
        if len(pair) != 2:
            raise ValueError(f"socket {identity} occurs {len(pair)} times")
        adjacency[pair[0]].add(pair[1])
        adjacency[pair[1]].add(pair[0])
    return adjacency


def _components(scene: dict) -> list[set[tuple[int, int]]]:
    adjacency = _adjacency(scene)
    unseen = set(adjacency)
    groups: list[set[tuple[int, int]]] = []
    while unseen:
        root = min(unseen)
        group = {root}
        queue = deque([root])
        while queue:
            node = queue.popleft()
            for neighbor in adjacency[node]:
                if neighbor not in group:
                    group.add(neighbor)
                    queue.append(neighbor)
        unseen -= group
        groups.append(group)
    return groups


def analytic_gold(scene: dict) -> str:
    start = MARKER_SLOTS["START"]
    group = next(group for group in _components(scene) if start in group)
    reached = [label for label in ANSWERS if MARKER_SLOTS[label] in group]
    if len(reached) != 1:
        raise ValueError(f"expected one START terminal, got {reached}")
    return reached[0]


def margin(scene: dict) -> float:
    _ = analytic_gold(scene)
    return 18.0


def is_quarantined(scene: dict) -> bool:
    try:
        if set(scene) != {"plates"} or not isinstance(scene["plates"], list) or len(scene["plates"]) != 9:
            return True
        markers: dict[str, tuple[int, int]] = {}
        counts: Counter[str] = Counter()
        remote_edges: list[tuple[tuple[int, int], tuple[int, int]]] = []
        locations: dict[str, list[tuple[int, int]]] = {}
        for plate_index, plate in enumerate(scene["plates"]):
            if set(plate) != {"ports", "wires"} or len(plate["ports"]) != 4:
                return True
            normalized = {tuple(sorted(map(int, pair))) for pair in plate["wires"]}
            if normalized not in ({tuple(sorted(pair)) for pair in STRAIGHT},
                                  {tuple(sorted(pair)) for pair in CROSS}):
                return True
            if len(plate["wires"]) != 2:
                return True
            for port, value in enumerate(plate["ports"]):
                value = str(value)
                if value.startswith("marker:"):
                    markers[value[7:]] = (plate_index, port)
                else:
                    counts[value] += 1
                    locations.setdefault(value, []).append((plate_index, port))
        if markers != MARKER_SLOTS or counts != Counter({identity: 2 for identity in BADGE_ORDER}):
            return True
        for identity in BADGE_ORDER:
            pair = locations[identity]
            if pair[0][0] == pair[1][0]:
                return True
            remote_edges.append((pair[0], pair[1]))
        groups = _components(scene)
        if sorted(len(group) for group in groups) != [12, 12, 12]:
            return True
        if any(sum(node in MARKER_SLOTS.values() for node in group) != 2 for group in groups):
            return True
        if not _coarse_connected(remote_edges):
            return True
        return analytic_gold(scene) not in ANSWERS
    except (KeyError, TypeError, ValueError, IndexError, StopIteration):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    for plate in twin["plates"]:
        plate["wires"].reverse()
    return [("reverse_unordered_cord_serialization", twin)]


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype("DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf", size * SCALE)


def _p(point: tuple[int, int]) -> tuple[int, int]:
    return point[0] * SCALE, point[1] * SCALE


def _b(box: tuple[int, int, int, int]) -> tuple[int, int, int, int]:
    return tuple(value * SCALE for value in box)  # type: ignore[return-value]


def _draw_badge(draw: ImageDraw.ImageDraw, center: tuple[int, int], identity: str) -> None:
    color_name, shape = identity.split("_", 1)
    color = BADGE_COLORS[color_name]
    x, y = center
    if shape == "circle":
        draw.ellipse(_b((x - 23, y - 23, x + 23, y + 23)), fill=color, outline=INK, width=4 * SCALE)
    elif shape == "square":
        draw.rounded_rectangle(_b((x - 23, y - 23, x + 23, y + 23)), radius=3 * SCALE,
                               fill=color, outline=INK, width=4 * SCALE)
    else:
        points = [_p((x, y - 27)), _p((x + 27, y)), _p((x, y + 27)), _p((x - 27, y))]
        draw.polygon(points, fill=color, outline=INK, width=4 * SCALE)


def _draw_marker(draw: ImageDraw.ImageDraw, center: tuple[int, int], label: str) -> None:
    x, y = center
    if label == "START":
        draw.rounded_rectangle(_b((x - 42, y - 20, x + 42, y + 20)), radius=15 * SCALE,
                               fill=START_COLOR, outline=INK, width=3 * SCALE)
        draw.text(_p((x, y - 1)), "START", anchor="mm", fill=PAPER, font=_font(12, True))
    else:
        draw.ellipse(_b((x - 23, y - 23, x + 23, y + 23)), fill=TERMINAL_COLOR,
                     outline=INK, width=4 * SCALE)
        draw.text(_p((x, y - 1)), label, anchor="mm", fill=PAPER, font=_font(21, True))


def render(scene: dict) -> Image.Image:
    if is_quarantined(scene):
        raise ValueError("cannot render quarantined scene")
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle(_b((20, 14, 1180, 108)), radius=15 * SCALE, fill=PAPER,
                           outline=MUTED, width=2 * SCALE)
    draw.text(_p((42, 29)), "SOCKET WEAVE", fill=INK, font=_font(25, True))
    draw.text(_p((42, 67)), "same color + shape = jump", fill=INK, font=_font(16))
    draw.text(_p((455, 38)), "black cords pair ports", fill=INK, font=_font(20, True))
    draw.text(_p((455, 71)), "white bridge = cross, not join", fill=INK, font=_font(16))
    draw.text(_p((913, 51)), "START  →  ?", fill=START_COLOR, font=_font(24, True))

    reverse_markers = {slot: label for label, slot in MARKER_SLOTS.items()}
    for plate_index, (cx, cy) in enumerate(CARD_CENTERS):
        draw.rounded_rectangle(_b((cx - 180, cy - 100, cx + 180, cy + 100)), radius=15 * SCALE,
                               fill=PAPER, outline=MUTED, width=3 * SCALE)
        draw.text(_p((cx, cy - 82)), str(plate_index + 1), anchor="mm", fill=MUTED, font=_font(12, True))
        centers = [(cx + dx, cy + dy) for dx, dy in PORT_OFFSETS]
        normalized = {tuple(sorted(map(int, pair))) for pair in scene["plates"][plate_index]["wires"]}
        if normalized == {tuple(sorted(pair)) for pair in STRAIGHT}:
            draw.line([_p(centers[0]), _p(centers[2])], fill=INK, width=11 * SCALE)
            draw.line([_p(centers[1]), _p(centers[3])], fill=INK, width=11 * SCALE)
        else:
            draw.line([_p(centers[0]), _p(centers[3])], fill=INK, width=11 * SCALE)
            draw.line([_p(centers[1]), _p(centers[2])], fill=PAPER, width=25 * SCALE)
            draw.line([_p(centers[1]), _p(centers[2])], fill=INK, width=11 * SCALE)
        for port, center in enumerate(centers):
            marker = reverse_markers.get((plate_index, port))
            if marker is not None:
                _draw_marker(draw, center, marker)
            else:
                _draw_badge(draw, center, str(scene["plates"][plate_index]["ports"][port]))
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
