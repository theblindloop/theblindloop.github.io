"""Dispatch-head-shuffle world.

Latent scene z = {arrangement, head, key, arrow_size, paint_order}.

Seven discs of seven flat colours sit in seven slots numbered 1..7 from the
left; the strip is a loop. A black arrow marks one slot (the head). A key of
seven rows gives every colour a direction (a solid triangle) and a distance
(that many grey squares, 1 to 3). The machine runs twelve steps:

    c    = strip[head]                   # read the disc under the head
    d    = key[c]                        # the read colour selects the step
    t    = (head + d) mod 7              # the target slot
    swap strip[head] and strip[t]
    head = (head + 1) mod 7

so the *instruction* executed at every step is chosen by the current state,
while the *operands* are addressed by position. Nothing on the canvas is an
ordered program: the step sequence is generated by the machine itself, so it
cannot be transcribed before execution, and a single wrong read forks the whole
remaining trajectory. The sampler forces at least three steps in which the red
disc moves while the head is reading some other colour, at least two steps
whose count wraps around an end of the loop, and it rejects any scene whose
answer coincides with reading the drawn strip as if it never changed, with a
frozen head, with a leftward head, with mirrored directions, with a clamped
(non-looping) count, or with an eleven-step or six-step prefix.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "dispatch_head_shuffle-0.1.0"

N_SLOTS = 7
RED = 0
STEPS = 12

# palette: index 0 is the queried red disc
COLORS = (
    (214, 39, 40),    # red
    (31, 119, 180),   # blue
    (44, 160, 44),    # green
    (255, 140, 0),    # orange
    (148, 60, 190),   # purple
    (23, 190, 190),   # teal
    (240, 110, 180),  # pink
)
TRI = (45, 45, 55)      # key direction triangles
GREY = (120, 120, 130)  # key distance squares
BLACK = (0, 0, 0)       # the head arrow
FRAME = (170, 172, 178)
BG = (255, 255, 255)

WIDTH = 1100
HEIGHT = 980

SLOT_X0 = 60
SLOT_PITCH = 145
SLOT_BOX = 132
SLOT_Y0 = 50
DISC_TOP = 104
FRAME_W = 3

HEAD_TIP_Y = 210
HEAD_BASE_Y = 272
HEAD_HALF = 38

RULE_Y = 300
RULE_H = 6

ROW_Y0 = 340
ROW_PITCH = 88
ROW_HALF = 36
KEY_DISC = 58
KEY_DISC_X = 330
TRI_X = 440
SQ_X0 = 620
SQ_W = 36
SQ_H = 36
SQ_GAP = 22

ARROW_MIN = 42
ARROW_MAX = 96
QUARANTINE_MARGIN = 8.0

_SIDES = ("L", "R")


def slot_center_x(i: int) -> int:
    return SLOT_X0 + i * SLOT_PITCH + SLOT_BOX // 2


# --------------------------------------------------------------------------
# dynamics
# --------------------------------------------------------------------------
def _run(arrangement, head, key, mode: str = "std", steps: int = STEPS):
    """Execute the machine and return (final 1-based slot of red, stats).

    mode 'std'       -- the rule;
    mode 'mirror'    -- read every key triangle the other way round;
    mode 'fixedhead' -- the head never advances;
    mode 'leftadv'   -- the head advances leftwards;
    mode 'noloop'    -- a count that runs off an end does nothing;
    mode 'stale'     -- read the colour from the strip as drawn, ignoring the
                        swaps already made (binding to the initial picture).
    """
    strip = list(arrangement)
    initial = list(arrangement)
    h = int(head)
    wraps = 0
    indirect_red = 0
    red_moves = 0
    for _ in range(steps):
        c = initial[h] if mode == "stale" else strip[h]
        side, dist = key[c]
        if mode == "mirror":
            side = "L" if side == "R" else "R"
        delta = dist if side == "R" else -dist
        raw = h + delta
        if raw < 0 or raw >= N_SLOTS:
            wraps += 1
            if mode == "noloop":
                h = (h - 1) % N_SLOTS if mode == "leftadv" else (h + 1) % N_SLOTS
                continue
        t = raw % N_SLOTS
        before = strip.index(RED)
        strip[h], strip[t] = strip[t], strip[h]
        after = strip.index(RED)
        if after != before:
            red_moves += 1
            if c != RED:
                indirect_red += 1
        if mode == "fixedhead":
            pass
        elif mode == "leftadv":
            h = (h - 1) % N_SLOTS
        else:
            h = (h + 1) % N_SLOTS
    stats = {
        "wraps": wraps,
        "indirect_red": indirect_red,
        "red_moves": red_moves,
        "final_strip": strip,
    }
    return strip.index(RED) + 1, stats


def _key_of(scene: dict) -> list:
    return [[str(s), int(d)] for s, d in scene["key"]]


def _variant(scene: dict, mode: str, steps: int = STEPS) -> int:
    return _run(list(scene["arrangement"]), int(scene["head"]), _key_of(scene), mode, steps)[0]


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------
def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 2654435761 % (2 ** 61 - 1))
    for _ in range(20000):
        arrangement = list(range(N_SLOTS))
        rng.shuffle(arrangement)
        head = rng.randrange(N_SLOTS)
        key = [[rng.choice(_SIDES), rng.randint(1, 3)] for _ in range(N_SLOTS)]

        sides = [s for s, _ in key]
        if sides.count("L") < 2 or sides.count("R") < 2:
            continue
        if len({d for _, d in key}) < 2:
            continue

        gold, stats = _run(arrangement, head, key, "std")
        if stats["red_moves"] < 4:
            continue
        if stats["indirect_red"] < 3:
            continue
        if stats["wraps"] < 2:
            continue
        if gold == arrangement.index(RED) + 1:
            continue
        variants = ("mirror", "fixedhead", "leftadv", "noloop", "stale")
        if any(gold == _run(arrangement, head, key, m)[0] for m in variants):
            continue
        if any(gold == _run(arrangement, head, key, "std", k)[0] for k in (6, 11)):
            continue

        arrow_size = rng.randint(ARROW_MIN, ARROW_MAX)
        paint_order = list(range(N_SLOTS))
        rng.shuffle(paint_order)
        return {
            "arrangement": arrangement,
            "head": head,
            "key": key,
            "arrow_size": arrow_size,
            "paint_order": paint_order,
        }
    raise RuntimeError(f"sampling failed for seed {seed}")


def analytic_gold(scene: dict) -> str:
    slot, _ = _run(list(scene["arrangement"]), int(scene["head"]), _key_of(scene), "std")
    return str(slot)


def margin(scene: dict) -> float:
    """Direction margin: a key triangle's centroid-to-bbox-centre offset."""
    return round(int(scene["arrow_size"]) / 6.0, 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list:
    """Key-row paint order is sampled but photometrically invisible."""
    order = list(scene["paint_order"])
    out = []
    if len(order) > 1:
        rotated = dict(scene)
        rotated["paint_order"] = order[1:] + order[:1]
        out.append(("rotated_paint_order", rotated))
        reversed_ = dict(scene)
        reversed_["paint_order"] = order[::-1]
        out.append(("reversed_paint_order", reversed_))
    return out


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def _tri_height(arrow_size: int) -> int:
    return 2 * max(14, min(32, arrow_size // 3))


def render(scene: dict) -> Image.Image:
    arrangement = [int(c) for c in scene["arrangement"]]
    head = int(scene["head"])
    key = [[str(k[0]), int(k[1])] for k in scene["key"]]
    arrow_size = int(scene["arrow_size"])
    order = [int(i) for i in scene["paint_order"]]

    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)

    for i, color in enumerate(arrangement):
        bx = SLOT_X0 + i * SLOT_PITCH
        draw.rectangle(
            [bx, SLOT_Y0, bx + SLOT_BOX, SLOT_Y0 + SLOT_BOX],
            outline=FRAME,
            width=FRAME_W,
        )
        inset = (SLOT_BOX - DISC_TOP) // 2
        dx = bx + inset
        dy = SLOT_Y0 + inset
        draw.ellipse([dx, dy, dx + DISC_TOP, dy + DISC_TOP], fill=COLORS[color])

    cx = slot_center_x(head)
    draw.polygon(
        [(cx, HEAD_TIP_Y), (cx - HEAD_HALF, HEAD_BASE_Y), (cx + HEAD_HALF, HEAD_BASE_Y)],
        fill=BLACK,
    )

    draw.rectangle([SLOT_X0, RULE_Y, WIDTH - SLOT_X0, RULE_Y + RULE_H], fill=FRAME)

    h = _tri_height(arrow_size)
    for j in order:
        side, dist = key[j]
        yc = ROW_Y0 + j * ROW_PITCH + ROW_HALF
        r = KEY_DISC // 2
        draw.ellipse(
            [KEY_DISC_X, yc - r, KEY_DISC_X + KEY_DISC, yc + r],
            fill=COLORS[j],
        )
        if side == "R":
            pts = [
                (TRI_X, yc - h // 2),
                (TRI_X, yc + h // 2),
                (TRI_X + arrow_size, yc),
            ]
        else:
            pts = [
                (TRI_X + arrow_size, yc - h // 2),
                (TRI_X + arrow_size, yc + h // 2),
                (TRI_X, yc),
            ]
        draw.polygon(pts, fill=TRI)
        for s in range(dist):
            sx = SQ_X0 + s * (SQ_W + SQ_GAP)
            draw.rectangle([sx, yc - SQ_H // 2, sx + SQ_W, yc + SQ_H // 2], fill=GREY)

    return image
