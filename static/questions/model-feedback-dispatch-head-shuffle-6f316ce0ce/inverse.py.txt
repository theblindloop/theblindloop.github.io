"""Independent pixel-only inverse arm for the dispatch-head-shuffle world.

From the final PNG alone it

1. masks the seven flat disc hues, the flat key-triangle ink, the flat grey
   distance squares and the pure black head arrow by exact colour;
2. splits the combined hue/triangle/square ink into horizontal bands; the top
   band is the strip and each later band is one key row;
3. reads the strip band: each hue's mean x fixes its slot, and sorting the
   hues by x recovers the arrangement together with the seven slot centres;
4. reads the head slot as the slot centre nearest the black arrow's centroid;
5. reads each key row: the hue present names the colour, the triangle's
   centroid-to-bbox-centre offset gives the direction (a solid triangle's mass
   sits behind its apex), and the number of separated grey column runs gives
   the distance;
6. runs the twelve-step machine on the recovered state and reports the 1-based
   slot holding the red disc.

It imports no renderer / prompts / generate / verify / analytic-gold code and
never receives the latent scene.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_PALETTE = (
    (214, 39, 40),
    (31, 119, 180),
    (44, 160, 44),
    (255, 140, 0),
    (148, 60, 190),
    (23, 190, 190),
    (240, 110, 180),
)
_TRI = (45, 45, 55)
_GREY = (120, 120, 130)
_BLACK = (0, 0, 0)
_RED = 0
_N_SLOTS = 7
_STEPS = 12


def _near(arr: np.ndarray, color, tol: int) -> np.ndarray:
    diff = np.abs(arr.astype(int) - np.array(color, dtype=int))
    return (diff <= tol).all(axis=-1)


def _runs(flags, min_len: int = 6):
    out = []
    start = None
    for i, f in enumerate(flags):
        if f and start is None:
            start = i
        elif not f and start is not None:
            if i - start >= min_len:
                out.append((start, i - 1))
            start = None
    if start is not None and len(flags) - start >= min_len:
        out.append((start, len(flags) - 1))
    return out


def _side(sub: np.ndarray) -> str:
    ys, xs = np.nonzero(sub)
    if xs.size == 0:
        return "R"
    ox = xs.mean() - (xs.min() + xs.max()) / 2.0
    return "R" if ox < 0 else "L"


def decision_from_image(image: Image.Image) -> str:
    """Return '1'..'7': the slot holding the red disc after twelve steps."""
    arr = np.asarray(image.convert("RGB"))
    hue_masks = [_near(arr, c, 20) for c in _PALETTE]
    tri = _near(arr, _TRI, 20)
    grey = _near(arr, _GREY, 20)
    black = _near(arr, _BLACK, 20)

    ink = tri | grey
    for m in hue_masks:
        ink |= m
    bands = _runs(ink.any(axis=1), min_len=8)
    if len(bands) < 1 + _N_SLOTS:
        return "1"

    top = bands[0]
    present = []
    for idx, m in enumerate(hue_masks):
        sub = m[top[0] : top[1] + 1]
        ys, xs = np.nonzero(sub)
        if xs.size:
            present.append((float(xs.mean()), idx))
    if len(present) != _N_SLOTS:
        return "1"
    present.sort()
    strip = [idx for _, idx in present]
    centers = [cx for cx, _ in present]

    ys, xs = np.nonzero(black)
    if xs.size == 0:
        return "1"
    ax = float(xs.mean())
    head = min(range(_N_SLOTS), key=lambda i: abs(centers[i] - ax))

    key = {}
    for lo, hi in bands[1 : 1 + _N_SLOTS]:
        color, best = None, 0
        for idx, m in enumerate(hue_masks):
            count = int(m[lo : hi + 1].sum())
            if count > best:
                best, color = count, idx
        if color is None:
            continue
        side = _side(tri[lo : hi + 1])
        dist = len(_runs(grey[lo : hi + 1].any(axis=0), min_len=6))
        if dist < 1:
            continue
        key[color] = (side, dist)
    if len(key) != _N_SLOTS:
        return "1"

    h = head
    for _ in range(_STEPS):
        side, dist = key[strip[h]]
        delta = dist if side == "R" else -dist
        t = (h + delta) % _N_SLOTS
        strip[h], strip[t] = strip[t], strip[h]
        h = (h + 1) % _N_SLOTS

    return str(strip.index(_RED) + 1)
