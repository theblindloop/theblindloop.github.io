"""Deterministic renderer for a quarter-turn self-XOR reveal."""

from __future__ import annotations

import random
import numpy as np
from PIL import Image, ImageDraw

WIDTH = HEIGHT = 512
GRID = 20
SYMBOLS = ("circle", "diamond", "plus", "x")
FRAME_RGB = (0, 174, 198)
MAGENTA = (220, 32, 132)
YELLOW = (245, 190, 20)
QUARANTINE_MARGIN = 0.12


def _orbits() -> list[list[tuple[int, int]]]:
    unseen = {(r, c) for r in range(GRID) for c in range(GRID)}
    groups = []
    while unseen:
        r, c = min(unseen)
        orbit = []
        for _ in range(4):
            orbit.append((r, c))
            r, c = c, GRID - 1 - r
        # Preserve clockwise cycle order; the minimum coordinate is the stable start.
        start = orbit.index(min(orbit))
        orbit = orbit[start:] + orbit[:start]
        groups.append(orbit)
        unseen.difference_update(orbit)
    return groups


def _templates() -> dict[str, np.ndarray]:
    functions = {
        "circle": lambda x, y: abs((x*x + y*y) ** 0.5 - 0.70),
        "diamond": lambda x, y: abs(abs(x) + abs(y) - 0.82),
        "plus": lambda x, y: min(abs(x), abs(y)) + 0.16 * max(abs(x), abs(y)),
        "x": lambda x, y: abs(abs(x) - abs(y)) + 0.16 * max(abs(x), abs(y)),
    }
    result = {}
    for name, fn in functions.items():
        ranked = []
        for orbit in _orbits():
            scores = []
            for r, c in orbit:
                x = (c + 0.5 - GRID / 2) / (GRID / 2)
                y = (r + 0.5 - GRID / 2) / (GRID / 2)
                scores.append(fn(x, y))
            ranked.append((sum(scores) / len(scores), orbit))
        ranked.sort(key=lambda item: (item[0], item[1]))
        mask = np.zeros((GRID, GRID), dtype=np.uint8)
        for _, orbit in ranked[:20]:
            for r, c in orbit:
                mask[r, c] = 1
        result[name] = mask
    return result


TEMPLATES = _templates()


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    symbol = SYMBOLS[int(seed) % len(SYMBOLS)]
    target = TEMPLATES[symbol]
    base = np.zeros((GRID, GRID), dtype=np.uint8)
    inactive = []
    for orbit in _orbits():
        r, c = orbit[0]
        if target[r, c]:
            phase = rng.randrange(2)
            # Around each four-cycle, 0101 XOR its clockwise shift is 1111.
            for index, (rr, cc) in enumerate(orbit):
                base[rr, cc] = (index + phase) & 1
        else:
            inactive.append(orbit)
    # Exactly 40 inactive orbits are all black: every source has 200 black cells.
    for orbit in rng.sample(inactive, 40):
        for r, c in orbit:
            base[r, c] = 1
    cell_size = rng.choice((16, 17, 18))
    field = GRID * cell_size
    ox = (WIDTH - field) // 2 + rng.randint(-14, 14)
    oy = (HEIGHT - field) // 2 + rng.randint(-14, 14)
    return {
        "symbol": symbol,
        "grid": base.tolist(),
        "offset": [ox, oy],
        "cell_size": cell_size,
        "frame_rgb": list(FRAME_RGB),
        "marker_rgbs": [list(MAGENTA), list(YELLOW)],
    }


def _grid(scene: dict) -> np.ndarray:
    base = np.asarray(scene["grid"], dtype=np.uint8)
    symbol = str(scene["symbol"])
    if base.shape != (GRID, GRID) or symbol not in TEMPLATES:
        raise ValueError("invalid scene grid")
    if not np.array_equal(np.bitwise_xor(base, np.rot90(base, -1)), TEMPLATES[symbol]):
        raise ValueError("scene grid does not encode declared symbol")
    return base


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def margin(scene: dict) -> float:
    transformed = np.bitwise_xor(_grid(scene), np.rot90(_grid(scene), -1))
    scores = sorted((1.0 - float(np.mean(transformed != mask)) for mask in TEMPLATES.values()), reverse=True)
    return round(scores[0] - scores[1], 12)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    base = _grid(scene)
    ox, oy = (int(v) for v in scene["offset"])
    cell = int(scene["cell_size"])
    frame_rgb = tuple(int(v) for v in scene["frame_rgb"])
    marker_rgbs = [tuple(int(v) for v in rgb) for rgb in scene["marker_rgbs"]]
    field = GRID * cell
    image = Image.new("RGB", (WIDTH, HEIGHT), (248, 247, 242))
    draw = ImageDraw.Draw(image)
    draw.rectangle((ox, oy, ox + field - 1, oy + field - 1), fill=(255, 255, 255))
    for r in range(GRID):
        for c in range(GRID):
            x0, y0 = ox + c * cell, oy + r * cell
            fill = (24, 25, 29) if base[r, c] else (255, 255, 255)
            draw.rectangle((x0, y0, x0 + cell - 1, y0 + cell - 1), fill=fill)
    for k in range(GRID + 1):
        x, y = ox + k * cell, oy + k * cell
        draw.line((x, oy, x, oy + field), fill=(205, 207, 210), width=1)
        draw.line((ox, y, ox + field, y), fill=(205, 207, 210), width=1)
    draw.rectangle((ox - 5, oy - 5, ox + field + 4, oy + field + 4), outline=frame_rgb, width=5)
    # Directional corner fiducials: magenta top-left moves to yellow top-right.
    draw.rectangle((ox - 12, oy - 12, ox + 2, oy + 2), fill=marker_rgbs[0])
    draw.rectangle((ox + field - 3, oy - 12, ox + field + 11, oy + 2), fill=marker_rgbs[1])
    return image
