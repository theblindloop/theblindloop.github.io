"""Independent image-only inverse for the quarter-turn self-XOR task."""

from __future__ import annotations

import numpy as np
from PIL import Image

GRID = 20
FRAME_RGB = (0, 174, 198)
MAGENTA = (220, 32, 132)
YELLOW = (245, 190, 20)
QUARANTINE_MARGIN = 0.12


def _orbits():
    unseen = {(r, c) for r in range(GRID) for c in range(GRID)}
    groups = []
    while unseen:
        r, c = min(unseen)
        orbit = []
        for _ in range(4):
            orbit.append((r, c))
            r, c = c, GRID - 1 - r
        start = orbit.index(min(orbit))
        orbit = orbit[start:] + orbit[:start]
        groups.append(orbit)
        unseen.difference_update(orbit)
    return groups


def _reference_glyphs():
    curves = {
        "circle": lambda x, y: abs((x*x + y*y) ** 0.5 - 0.70),
        "diamond": lambda x, y: abs(abs(x) + abs(y) - 0.82),
        "plus": lambda x, y: min(abs(x), abs(y)) + 0.16 * max(abs(x), abs(y)),
        "x": lambda x, y: abs(abs(x) - abs(y)) + 0.16 * max(abs(x), abs(y)),
    }
    refs = {}
    for label, curve in curves.items():
        candidates = []
        for orbit in _orbits():
            value = 0.0
            for r, c in orbit:
                x = (c + 0.5 - GRID / 2) / (GRID / 2)
                y = (r + 0.5 - GRID / 2) / (GRID / 2)
                value += curve(x, y) / len(orbit)
            candidates.append((value, orbit))
        candidates.sort(key=lambda item: (item[0], item[1]))
        ref = np.zeros((GRID, GRID), dtype=bool)
        for _, orbit in candidates[:20]:
            for r, c in orbit:
                ref[r, c] = True
        refs[label] = ref
    return refs


REFERENCES = _reference_glyphs()


def _near(arr: np.ndarray, rgb: tuple[int, int, int], radius: int = 45) -> np.ndarray:
    delta = arr.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2) <= radius * radius


def _measure_grid(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"))
    cyan = _near(arr, FRAME_RGB)
    ys, xs = np.nonzero(cyan)
    if len(xs) < 100:
        raise ValueError("cyan frame missing")
    x0, x1, y0, y1 = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    if abs((x1 - x0) - (y1 - y0)) > 3:
        raise ValueError("frame is not square")
    magenta = _near(arr, MAGENTA)
    yellow = _near(arr, YELLOW)
    if magenta[max(0, y0-10):y0+10, max(0, x0-10):x0+10].sum() < 25:
        raise ValueError("top-left direction marker missing")
    if yellow[max(0, y0-10):y0+10, x1-10:min(arr.shape[1], x1+11)].sum() < 25:
        raise ValueError("top-right direction marker missing")
    inner_x0, inner_x1 = x0 + 5, x1 - 4
    inner_y0, inner_y1 = y0 + 5, y1 - 4
    width = inner_x1 - inner_x0 + 1
    height = inner_y1 - inner_y0 + 1
    if abs(width - height) > 2 or not (300 <= width <= 380):
        raise ValueError("implausible framed grid")
    cell = width / GRID
    base = np.zeros((GRID, GRID), dtype=bool)
    for r in range(GRID):
        for c in range(GRID):
            cx = int(round(inner_x0 + (c + 0.5) * cell))
            cy = int(round(inner_y0 + (r + 0.5) * cell))
            patch = arr[cy-2:cy+3, cx-2:cx+3]
            base[r, c] = float(patch.mean()) < 110.0
    return base


def decision_from_image(image: Image.Image) -> str:
    base = _measure_grid(image)
    transformed = np.logical_xor(base, np.rot90(base, -1))
    ranked = sorted(
        ((1.0 - float(np.mean(transformed != ref)), label) for label, ref in REFERENCES.items()),
        reverse=True,
    )
    if ranked[0][0] - ranked[1][0] < QUARANTINE_MARGIN:
        raise ValueError("ambiguous transformed glyph")
    return ranked[0][1]
