"""Slanting-rain world under a contradictory verticality prior.

Profile: paper-semantic@0.3.0:prior_conflict. Report visible truth (whether the
visible rain is actually falling straight down) when the familiar canonical
prior ("rain falls straight down / vertically") is contradicted by the actual
image, which in the counterfactual scenes shows the rain streaks visibly slanted
by the wind.

Latent z = {tilt, length, thickness, streaks, starts, width, height}.
``tilt`` is the drawn angular deviation from vertical in degrees (0 for vertical
rain, in [20, 42] for slanting rain) and is the single closed-form label input.
``length`` and ``thickness`` are the streak geometry, ``streaks`` is the number
of streaks, ``starts`` the per-streak start positions, and ``width``/``height``
the canvas size. The closed-form gold is

    decision = "yes" if scene["tilt"] <= TOL else "no"

read through the single label input ``tilt``.

The renderer draws ``streaks`` short, parallel, blue rain streaks on a plain
white background. In control scenes the streaks are exactly vertical; in
counterfactual scenes every streak is slanted by the same wind angle ``tilt``
from vertical. The streaks are the only non-white pixels on the canvas. The
pixel-only inverse arm therefore recovers the common streak orientation from the
blue-pixel point cloud (principal axis of the pixel covariance) and decides
whether that orientation is within ``TOL`` degrees of the vertical.

The continuous verifiability margin is the magnitude of the signed clearance
between the drawn tilt and the verticality decision threshold ``TOL``. Scenes in
which the tilt is too close to the threshold, or in which the streaks are too
short or too few to resolve an orientation, are quarantined.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "prior_slanting_rain-0.1.0"

CANVAS = (432, 432)
BG = (255, 255, 255)          # white paper background (no other object drawn)
RAIN = (78, 132, 204)         # blue rain streak colour (the only non-white pixels)
TOL = 8.0                     # decision threshold: tilt within this many degrees = vertical
MIN_TILT = 20.0               # minimum tilt of a clear slanting counterfactual
MAX_TILT = 42.0
MIN_LEN = 70
MIN_STREAKS = 8
MAX_STREAKS = 12


def margin(scene: dict) -> float:
    """Magnitude of the signed clearance between the drawn tilt and the threshold."""
    tilt = float(scene["tilt"])
    return round(abs(tilt - TOL), 6)


def is_quarantined(scene: dict) -> bool:
    tilt = float(scene["tilt"])
    if abs(abs(tilt) - TOL) <= 3.0:
        return True
    if int(scene["length"]) < MIN_LEN:
        return True
    if int(scene["streaks"]) < MIN_STREAKS:
        return True
    w, h = int(scene["width"]), int(scene["height"])
    length = int(scene["length"])
    shift = length * math.sin(math.radians(tilt))
    for x0, y0 in scene["starts"]:
        x1 = x0 + shift
        y1 = y0 + length * math.cos(math.radians(tilt))
        if x1 < 12 or x1 > w - 12:
            return True
        if y0 < 12 or y1 > h - 12:
            return True
    return False


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    w, h = CANVAS
    while True:
        if seed % 2 == 0:
            tilt = 0.0
        else:
            tilt = rng.uniform(MIN_TILT, MAX_TILT)
        length = rng.randint(MIN_LEN, 108)
        thickness = rng.randint(3, 5)
        streaks = rng.randint(MIN_STREAKS, MAX_STREAKS)
        shift = length * math.sin(math.radians(tilt))
        starts = []
        for _ in range(streaks):
            x0 = rng.uniform(50, w - 50 - 90)
            y0 = rng.uniform(40, h - 150)
            starts.append((x0, y0))
        scene = {
            "tilt": tilt,
            "length": length,
            "thickness": thickness,
            "streaks": streaks,
            "starts": starts,
            "width": w,
            "height": h,
        }
        if not is_quarantined(scene):
            return scene


def _segment_start(scene: dict, index: int):
    x0, y0 = scene["starts"][index]
    return x0, y0


def render(scene: dict) -> Image.Image:
    w, h = int(scene["width"]), int(scene["height"])
    length = int(scene["length"])
    thickness = int(scene["thickness"])
    tilt = float(scene["tilt"])
    rad = math.radians(tilt)
    sin_a = math.sin(rad)
    cos_a = math.cos(rad)

    img = Image.new("RGB", (w, h), BG)
    draw = ImageDraw.Draw(img)
    for index in range(int(scene["streaks"])):
        x0, y0 = _segment_start(scene, index)
        x1 = x0 + length * sin_a
        y1 = y0 + length * cos_a
        draw.line([x0, y0, x1, y1], fill=RAIN, width=thickness)
    return img


def analytic_gold(scene: dict) -> str:
    return "yes" if float(scene["tilt"]) <= TOL else "no"


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No pixel-identical latent alias exists: the scene maps injectively.

    The verticality label is directly visible in the raster as the orientation
    of the streak point cloud: vertical control scenes produce axis-aligned
    streaks while slanting counterfactual scenes produce tilted streaks, so the
    label cannot be recovered from any other field and changing it changes the
    pixels. This is a `not_applicable` latent-alias mode, so no transforms are
    declared.
    """
    return []
