"""Independent pixel-only inverse arm for the slanting-rain world.

Receives ONLY the final PNG and returns "yes" (rain falling straight down /
vertical) or "no" (rain slanted by the wind) as a string. It shares no scene
constants with the renderer (only the generic facts, visible in the image, that
the rain streaks are the only non-white, coloured pixels on a plain white
background). It never imports renderer, prompts, generate, verify, or
analytic-gold code, and accepts no latent scene data.

Method: the blue rain streaks are the only non-white pixels on the white canvas.
The oracle masks those pixels, builds the point cloud, and computes the central
moment of inertia of the pixels to recover the principal axis -- the direction
along which the streaks are elongated. That is the common streak orientation.
The decision is "yes" when this orientation lies within TOL degrees of the
vertical, and "no" when the streaks are visibly slanted by the wind.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "prior_slanting_rain-oracle-0.1.0"

TOL = 8.0


def _streak_mask(arr: np.ndarray) -> np.ndarray:
    r = arr[..., 0].astype(np.float32)
    g = arr[..., 1].astype(np.float32)
    b = arr[..., 2].astype(np.float32)
    mx = np.maximum(np.maximum(r, g), b)
    mn = np.minimum(np.minimum(r, g), b)
    return (mx <= 240.0) & ((mx - mn) >= 40.0)


def _components(mask: np.ndarray, min_px: int = 40) -> list[np.ndarray]:
    """Return pixel masks of 8-connected components with at least min_px pixels."""
    h, w = mask.shape
    visited = np.zeros((h, w), dtype=bool)
    comps = []
    for i in range(h):
        for j in np.nonzero(mask[i])[0]:
            if visited[i, j]:
                continue
            stack = [(i, j)]
            visited[i, j] = True
            idx = []
            while stack:
                x, y = stack.pop()
                idx.append((x, y))
                x0 = max(0, x - 1)
                x1 = min(h, x + 2)
                y0 = max(0, y - 1)
                y1 = min(w, y + 2)
                region = mask[x0:x1, y0:y1] & ~visited[x0:x1, y0:y1]
                ys, xs = np.nonzero(region)
                for xx, yy in zip(ys + x0, xs + y0):
                    visited[xx, yy] = True
                    stack.append((xx, yy))
            if len(idx) >= min_px:
                comps.append(np.array(idx))
    return comps


def _signed_lean(pts: np.ndarray) -> float:
    """Signed angle (deg) of a blob's elongation off the vertical, in [-90, 90]."""
    coords = pts.astype(np.float64)
    center = coords.mean(axis=0)
    centered = coords - center
    cov = centered.T @ centered / len(coords)
    vals, vecs = np.linalg.eigh(cov)
    vrow, vcol = vecs[:, int(np.argmax(vals))]
    lean = math.degrees(math.atan2(vcol, vrow))
    if lean > 90.0:
        lean -= 180.0
    elif lean < -90.0:
        lean += 180.0
    return lean


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    mask = _streak_mask(arr)
    comps = _components(mask)
    if not comps:
        return "yes"
    dev = float(np.mean([abs(_signed_lean(c)) for c in comps]))
    return "yes" if dev <= TOL else "no"


def measure(image: Image.Image) -> dict:
    """Diagnostic measurement (not used for the decision)."""
    arr = np.asarray(image.convert("RGB"))
    mask = _streak_mask(arr)
    comps = _components(mask)
    dev = float(np.mean([abs(_signed_lean(c)) for c in comps])) if comps else 0.0
    return {
        "streak_px": int(mask.sum()),
        "components": len(comps),
        "mean_deviation_deg": round(dev, 3),
        "size": image.size,
    }
