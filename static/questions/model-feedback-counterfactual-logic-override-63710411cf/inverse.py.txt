"""Independent image-only inverse for the visible Boolean override grammar."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

WIDTH = 640
HEIGHT = 720
SOURCE_CENTERS = {
    "W": (80, 78),
    "X": (240, 78),
    "Y": (400, 78),
    "Z": (560, 78),
}
GATE_CENTERS = {
    "A": (150, 210),
    "B": (320, 210),
    "C": (490, 210),
    "D": (230, 350),
    "E": (410, 350),
    "F": (320, 485),
    "P": (230, 620),
    "Q": (410, 620),
}
EDGES = (
    ("W", "A"), ("X", "A"), ("X", "B"), ("Y", "B"),
    ("Y", "C"), ("Z", "C"), ("A", "D"), ("B", "D"),
    ("B", "E"), ("C", "E"), ("D", "F"), ("E", "F"),
    ("D", "P"), ("F", "P"), ("F", "Q"), ("E", "Q"),
)
GATE_RGB = {
    "XOR": (36, 116, 216),
    "AND": (242, 169, 0),
    "OR": (135, 84, 200),
}
INK_RGB = (23, 33, 43)
RED_RGB = (213, 38, 47)
WHITE_RGB = (255, 255, 255)


class OracleError(ValueError):
    pass


def _near(arr: np.ndarray, rgb: tuple[int, int, int], radius: float) -> np.ndarray:
    delta = arr.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2, dtype=np.int32) <= radius * radius


def _recover_bit(arr: np.ndarray, center: tuple[int, int], name: str) -> int:
    cx, cy = center
    patch = arr[cy - 13 : cy + 14, cx - 13 : cx + 14]
    dark = int(_near(patch, INK_RGB, 48.0).sum())
    light = int(_near(patch, WHITE_RGB, 24.0).sum())
    if dark >= 590 and light <= 60:
        return 1
    if light >= 590 and dark <= 60:
        return 0
    raise OracleError(f"{name} bit cell is missing or ambiguous")


def _recover_gate(arr: np.ndarray, center: tuple[int, int], name: str) -> str:
    cx, cy = center
    radius = 31
    patch = arr[cy - radius : cy + radius + 1, cx - radius : cx + radius + 1]
    yy, xx = np.ogrid[-radius : radius + 1, -radius : radius + 1]
    annulus = ((xx * xx + yy * yy) >= 18 * 18) & ((xx * xx + yy * yy) <= 29 * 29)
    scores = {
        kind: int((_near(patch, rgb, 58.0) & annulus).sum())
        for kind, rgb in GATE_RGB.items()
    }
    ranked = sorted(scores.items(), key=lambda item: item[1], reverse=True)
    if ranked[0][1] < 1250 or ranked[1][1] > 140:
        raise OracleError(f"gate {name} is missing or has ambiguous type: {scores}")
    return ranked[0][0]


def _validate_intervention(arr: np.ndarray) -> None:
    red = _near(arr, RED_RGB, 52.0)
    box_count = int(red[154:266, 270:370].sum())
    cell_count = int(red[181:239, 381:439].sum())
    bridge_count = int(red[202:219, 360:388].sum())
    if box_count < 1450:
        raise OracleError("red intervention box around B is missing")
    if cell_count < 650:
        raise OracleError("red-outlined forced bit is missing")
    if bridge_count < 100:
        raise OracleError("forced bit is not visibly linked to B")


def _validate_wires(arr: np.ndarray) -> None:
    ink = _near(arr, INK_RGB, 68.0)
    centers = {**SOURCE_CENTERS, **GATE_CENTERS}
    for parent, child in EDGES:
        x0, y0 = centers[parent]
        x1, y1 = centers[child]
        # The midpoint lies outside all nodes and away from arrowheads.
        mx = int(round((x0 + x1) / 2))
        my = int(round((y0 + y1) / 2))
        patch = ink[my - 6 : my + 7, mx - 6 : mx + 7]
        if int(patch.sum()) < 24:
            raise OracleError(f"wire {parent}->{child} is missing")


def _gate(kind: str, left: int, right: int) -> int:
    if kind == "XOR":
        return int(left != right)
    if kind == "AND":
        return int(left == 1 and right == 1)
    if kind == "OR":
        return int(left == 1 or right == 1)
    raise OracleError(f"invalid recovered gate {kind!r}")


def decision_from_image(image: Image.Image) -> str:
    if image.size != (WIDTH, HEIGHT):
        raise OracleError(f"expected {(WIDTH, HEIGHT)} image, got {image.size}")
    arr = np.asarray(image.convert("RGB"))
    _validate_intervention(arr)
    _validate_wires(arr)
    source = {name: _recover_bit(arr, center, name) for name, center in SOURCE_CENTERS.items()}
    kinds = {name: _recover_gate(arr, center, name) for name, center in GATE_CENTERS.items()}
    forced = _recover_bit(arr, (410, 210), "forced B")

    values = dict(source)
    values["A"] = _gate(kinds["A"], values["W"], values["X"])
    values["B"] = forced
    values["C"] = _gate(kinds["C"], values["Y"], values["Z"])
    values["D"] = _gate(kinds["D"], values["A"], values["B"])
    values["E"] = _gate(kinds["E"], values["B"], values["C"])
    values["F"] = _gate(kinds["F"], values["D"], values["E"])
    values["P"] = _gate(kinds["P"], values["D"], values["F"])
    values["Q"] = _gate(kinds["Q"], values["F"], values["E"])
    return f'{values["P"]}{values["Q"]}'
