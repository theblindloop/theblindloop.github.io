"""Deterministic Pillow renderer for a surgically overridden Boolean DAG."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 640
HEIGHT = 720
SCALE = 3

SOURCE_CENTERS = {
    "W": (80, 78),
    "X": (240, 78),
    "Y": (400, 78),
    "Z": (560, 78),
}
GATE_CENTERS = {
    "A": (150, 210),
    "B": (320, 210),
    "C": (490, 210),
    "D": (230, 350),
    "E": (410, 350),
    "F": (320, 485),
    "P": (230, 620),
    "Q": (410, 620),
}
EDGES = (
    ("W", "A"), ("X", "A"), ("X", "B"), ("Y", "B"),
    ("Y", "C"), ("Z", "C"), ("A", "D"), ("B", "D"),
    ("B", "E"), ("C", "E"), ("D", "F"), ("E", "F"),
    ("D", "P"), ("F", "P"), ("F", "Q"), ("E", "Q"),
)
GATE_IDS = tuple(GATE_CENTERS)
GATE_TYPES = ("XOR", "AND", "OR")
GATE_COLORS = {
    "XOR": "#2474D8",
    "AND": "#F2A900",
    "OR": "#8754C8",
}
INK = "#17212B"
RED = "#D5262F"
GRAY = "#6F7A85"
PALE = "#F4F6F8"


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size * SCALE
    )


def _gate(kind: str, left: int, right: int) -> int:
    if kind == "XOR":
        return int(left != right)
    if kind == "AND":
        return int(left == 1 and right == 1)
    if kind == "OR":
        return int(left == 1 or right == 1)
    raise ValueError(f"unknown gate type {kind!r}")


def _evaluate(scene: dict[str, Any], intervene: bool) -> dict[str, int]:
    source = {name: int(value) for name, value in zip("WXYZ", scene["source_bits"])}
    kinds = {str(name): str(value) for name, value in scene["gate_types"].items()}
    values = dict(source)
    values["A"] = _gate(kinds["A"], values["W"], values["X"])
    factual_b = _gate(kinds["B"], values["X"], values["Y"])
    values["B"] = int(scene["forced_bit"]) if intervene else factual_b
    values["C"] = _gate(kinds["C"], values["Y"], values["Z"])
    values["D"] = _gate(kinds["D"], values["A"], values["B"])
    values["E"] = _gate(kinds["E"], values["B"], values["C"])
    values["F"] = _gate(kinds["F"], values["D"], values["E"])
    values["P"] = _gate(kinds["P"], values["D"], values["F"])
    values["Q"] = _gate(kinds["Q"], values["F"], values["E"])
    return values


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    for _ in range(20000):
        source_bits = [rng.randrange(2) for _ in SOURCE_CENTERS]
        gate_types = {name: rng.choice(GATE_TYPES) for name in GATE_IDS}
        factual_b = _gate(gate_types["B"], source_bits[1], source_bits[2])
        scene: dict[str, Any] = {
            "source_bits": source_bits,
            "gate_types": gate_types,
            "forced_bit": 1 - factual_b,
        }
        factual = _evaluate(scene, intervene=False)
        counterfactual = _evaluate(scene, intervene=True)
        if (
            set(source_bits) == {0, 1}
            and set(gate_types.values()) == set(GATE_TYPES)
            and factual["P"] != counterfactual["P"]
            and factual["Q"] != counterfactual["Q"]
        ):
            return scene
    raise RuntimeError("could not sample a causally relevant scene")


def analytic_gold(scene: dict[str, Any]) -> str:
    values = _evaluate(scene, intervene=True)
    return f'{values["P"]}{values["Q"]}'


def margin(scene: dict[str, Any]) -> float:
    return 14.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        source = list(scene["source_bits"])
        kinds = dict(scene["gate_types"])
        forced = int(scene["forced_bit"])
        if (
            len(source) != 4
            or any(type(value) is not int or value not in (0, 1) for value in source)
            or set(source) != {0, 1}
            or set(kinds) != set(GATE_IDS)
            or any(value not in GATE_TYPES for value in kinds.values())
            or set(kinds.values()) != set(GATE_TYPES)
            or type(scene["forced_bit"]) is not int
            or forced not in (0, 1)
        ):
            return True
        factual_b = _gate(str(kinds["B"]), int(source[1]), int(source[2]))
        if forced == factual_b:
            return True
        factual = _evaluate(scene, intervene=False)
        counterfactual = _evaluate(scene, intervene=True)
        return not (
            factual["P"] != counterfactual["P"]
            and factual["Q"] != counterfactual["Q"]
        )
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict]]:
    return []


def _center_text(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    text: str,
    font: ImageFont.FreeTypeFont,
    fill: str,
) -> None:
    cx, cy = center
    box = draw.textbbox((0, 0), text, font=font)
    width = box[2] - box[0]
    height = box[3] - box[1]
    draw.text(
        (int(round(cx * SCALE - width / 2)), int(round(cy * SCALE - height / 2 - SCALE))),
        text,
        fill=fill,
        font=font,
    )


def _arrow(
    draw: ImageDraw.ImageDraw,
    start: tuple[int, int],
    end: tuple[int, int],
    start_pad: float,
    end_pad: float,
) -> None:
    dx = float(end[0] - start[0])
    dy = float(end[1] - start[1])
    length = math.hypot(dx, dy)
    ux, uy = dx / length, dy / length
    p0 = (start[0] + ux * start_pad, start[1] + uy * start_pad)
    tip = (end[0] - ux * end_pad, end[1] - uy * end_pad)
    base = (tip[0] - ux * 11, tip[1] - uy * 11)
    normal = (-uy, ux)
    left = (base[0] + normal[0] * 6, base[1] + normal[1] * 6)
    right = (base[0] - normal[0] * 6, base[1] - normal[1] * 6)
    scale_point = lambda p: (int(round(p[0] * SCALE)), int(round(p[1] * SCALE)))
    draw.line((*scale_point(p0), *scale_point(base)), fill=INK, width=4 * SCALE)
    draw.polygon([scale_point(tip), scale_point(left), scale_point(right)], fill=INK)


def render(scene: dict[str, Any]) -> Image.Image:
    if is_quarantined(scene):
        raise ValueError("cannot render quarantined scene")
    source_bits = [int(value) for value in scene["source_bits"]]
    gate_types = {str(name): str(value) for name, value in scene["gate_types"].items()}
    forced_bit = int(scene["forced_bit"])

    canvas = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), "white")
    draw = ImageDraw.Draw(canvas)
    sx = lambda value: int(round(value * SCALE))

    draw.text((sx(20), sx(13)), "START VALUES", fill=GRAY, font=_font(16))
    draw.text((sx(382), sx(13)), "DARK = 1    LIGHT = 0", fill=GRAY, font=_font(14))

    centers = {**SOURCE_CENTERS, **GATE_CENTERS}
    for parent, child in EDGES:
        _arrow(
            draw,
            centers[parent],
            centers[child],
            29.0 if parent in SOURCE_CENTERS else 36.0,
            38.0,
        )

    source_font = _font(17)
    for (name, (cx, cy)), bit in zip(SOURCE_CENTERS.items(), source_bits):
        half = 27
        draw.rounded_rectangle(
            (sx(cx - half), sx(cy - half), sx(cx + half), sx(cy + half)),
            radius=sx(7),
            fill=INK if bit else "white",
            outline=INK,
            width=sx(4),
        )
        _center_text(draw, (cx, cy - 44), name, source_font, INK)

    gate_id_font = _font(14)
    gate_type_font = _font(11)
    for name, (cx, cy) in GATE_CENTERS.items():
        kind = gate_types[name]
        fill = GATE_COLORS[kind]
        text_fill = INK if kind == "AND" else "white"
        radius = 34
        draw.ellipse(
            (sx(cx - radius), sx(cy - radius), sx(cx + radius), sx(cy + radius)),
            fill=fill,
            outline=INK,
            width=sx(3),
        )
        _center_text(draw, (cx, cy - 13), name, gate_id_font, text_fill)
        _center_text(draw, (cx, cy + 10), kind, gate_type_font, text_fill)

    # The red box identifies the intervened variable. Its incoming arrows are
    # still visible because a surgical override cuts their causal effect, not
    # the diagram itself. The adjacent broad cell visibly supplies do(B=bit).
    draw.rounded_rectangle(
        (sx(274), sx(158), sx(366), sx(262)),
        radius=sx(11),
        outline=RED,
        width=sx(6),
    )
    draw.line((sx(366), sx(210), sx(382), sx(210)), fill=RED, width=sx(5))
    _center_text(draw, (410, 169), "FORCE B", _font(14), RED)
    half = 24
    draw.rounded_rectangle(
        (sx(386), sx(186), sx(434), sx(234)),
        radius=sx(7),
        fill=INK if forced_bit else "white",
        outline=RED,
        width=sx(6),
    )

    _center_text(draw, (320, 665), "OUTPUTS:  P  then  Q", _font(15), INK)

    legend_y = 699
    legend = (
        (52, "XOR", "different"),
        (245, "AND", "both"),
        (425, "OR", "either"),
    )
    for x, kind, meaning in legend:
        draw.rounded_rectangle(
            (sx(x), sx(legend_y - 13), sx(x + 40), sx(legend_y + 13)),
            radius=sx(5),
            fill=GATE_COLORS[kind],
            outline=INK,
            width=sx(2),
        )
        text_fill = INK if kind == "AND" else "white"
        _center_text(draw, (x + 20, legend_y), kind, _font(8), text_fill)
        draw.text((sx(x + 49), sx(legend_y - 10)), meaning, fill=GRAY, font=_font(12))

    return canvas.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
