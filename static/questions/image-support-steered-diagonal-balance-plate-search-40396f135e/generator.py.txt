"""Renderer and analytic label for connected diagonal-balance occupancy plates."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

RENDERER_VERSION = "diagonal-balance-plate-0.1.0"
WIDTH = 512
HEIGHT = 512
GRID_N = 11
OCCUPIED_CELLS = 61
QUADRANTS = ("NW", "NE", "SW", "SE")
_CENTER = (GRID_N - 1) / 2.0

# A connected diamond with full-span midpoints. The target is D4-balanced but
# not a single marked cell or isolated symbol.
TARGET_GRID = [
    ".....#.....",
    "....###....",
    "...#####...",
    "..#######..",
    ".#########.",
    "###########",
    ".#########.",
    "..#######..",
    "...#####...",
    "....###....",
    ".....#.....",
]
_BASE_COORDS = {
    (r, c)
    for r, row in enumerate(TARGET_GRID)
    for c, value in enumerate(row)
    if value == "#"
}


def _moved_grid(source: tuple[int, int], destination: tuple[int, int]) -> list[str]:
    coords = set(_BASE_COORDS)
    if source not in coords or destination in coords:
        raise AssertionError("invalid template move")
    coords.remove(source)
    coords.add(destination)
    return ["".join("#" if (r, c) in coords else "." for c in range(GRID_N)) for r in range(GRID_N)]


DISTRACTOR_GRIDS = [
    _moved_grid((6, 5), (0, 4)),
    _moved_grid((5, 4), (4, 10)),
    _moved_grid((4, 5), (10, 4)),
]
TEMPLATES = [TARGET_GRID, *DISTRACTOR_GRIDS]

PALETTES = (
    ((248, 250, 247), (26, 53, 48)),
    ((248, 247, 251), (53, 36, 74)),
    ((252, 248, 242), (77, 45, 25)),
    ((245, 249, 252), (25, 52, 83)),
)


def _rotate_grid(grid: list[str], quarter_turns: int) -> list[str]:
    rows = [list(row) for row in grid]
    for _ in range((quarter_turns // 90) % 4):
        rows = [list(row) for row in zip(*rows[::-1])]
    return ["".join(row) for row in rows]


def _grid_coords(grid: list[str]) -> Iterable[tuple[int, int]]:
    for r, row in enumerate(grid):
        for c, value in enumerate(row):
            if value == "#":
                yield r, c


def _score_grid(grid: list[str]) -> tuple[int, int, int]:
    main = 0
    anti = 0
    count = 0
    for r, c in _grid_coords(grid):
        count += 1
        if r < c:
            main -= 1
        elif r > c:
            main += 1
        if r + c < GRID_N - 1:
            anti -= 1
        elif r + c > GRID_N - 1:
            anti += 1
    return main, anti, count


def _scores(panels: list[dict]) -> list[tuple[int, str, int, int]]:
    scored = []
    for panel in panels:
        main, anti, count = _score_grid(panel["grid"])
        scored.append((abs(main) + abs(anti), str(panel["quadrant"]), main, anti))
        if count != OCCUPIED_CELLS:
            return scored
    return scored


def sample_scene(seed: int) -> dict:
    """Sample one balanced four-quadrant plate gallery from a stable integer seed."""
    rng = random.Random(int(seed))
    target_index = rng.randrange(4)
    shuffled_templates = [None] * 4
    distractors = iter(DISTRACTOR_GRIDS)
    for index in range(4):
        shuffled_templates[index] = TARGET_GRID if index == target_index else next(distractors)
    rng.shuffle(shuffled_templates)
    background, ink = PALETTES[rng.randrange(len(PALETTES))]
    centers = {
        "NW": (128, 128),
        "NE": (384, 128),
        "SW": (128, 384),
        "SE": (384, 384),
    }
    panels = []
    for quadrant, grid in zip(QUADRANTS, shuffled_templates, strict=True):
        cell = rng.randint(12, 15)
        rotation = rng.choice((0, 90, 180, 270))
        size = GRID_N * cell
        cx, cy = centers[quadrant]
        jitter_x = rng.randint(-14, 14)
        jitter_y = rng.randint(-14, 14)
        x = int(cx + jitter_x - size / 2)
        y = int(cy + jitter_y - size / 2)
        panels.append(
            {
                "quadrant": quadrant,
                "x": x,
                "y": y,
                "cell": cell,
                "rotation": rotation,
                "grid": list(grid),
                "background": list(background),
                "ink": list(ink),
            }
        )
    return {"panels": panels}


def render(scene: dict) -> Image.Image:
    panels = scene["panels"]
    if not panels:
        raise ValueError("scene has no panels")
    image = Image.new("RGB", (WIDTH, HEIGHT), tuple(panels[0]["background"]))
    draw = ImageDraw.Draw(image)
    for panel in panels:
        cell = int(panel["cell"])
        grid = _rotate_grid(list(panel["grid"]), int(panel["rotation"]))
        x0, y0 = int(panel["x"]), int(panel["y"])
        ink = tuple(panel["ink"])
        for r, c in _grid_coords(grid):
            left = x0 + c * cell
            top = y0 + r * cell
            draw.rectangle((left, top, left + cell - 1, top + cell - 1), fill=ink)
    return image


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene["panels"])
    if len(scores) != 4:
        raise ValueError(f"scene has no unique balanced plate: {scores}")
    winners = [item[1] for item in scores if item[0] == 0]
    if len(winners) != 1:
        raise ValueError(f"balanced plate tie: {scores}")
    return winners[0]


def margin(scene: dict) -> float:
    scores = sorted(item[0] for item in _scores(scene["panels"]))
    if len(scores) < 2:
        return 0.0
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    try:
        scores = _scores(scene["panels"])
        if len(scores) != 4:
            return True
        if len({item[1] for item in scores if item[0] == 0}) != 1:
            return True
        return margin(scene) < 2.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse the unordered visible panel-record list without changing pixels."""
    twin = {"panels": list(reversed(scene["panels"]))}
    return [("reverse_panel_records", twin)]
