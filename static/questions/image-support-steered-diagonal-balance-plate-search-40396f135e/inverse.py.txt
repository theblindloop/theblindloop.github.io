"""Independent pixel-only inverse for diagonal-balance occupancy plates."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

GRID_N = 11
OCCUPIED_CELLS = 61
WIDTH = 512
HEIGHT = 512
QUADRANTS = ("NW", "NE", "SW", "SE")


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    found: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if visited[y, x]:
            continue
        queue = deque([(y, x)])
        visited[y, x] = True
        pixels: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            pixels.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    queue.append((ny, nx))
        found.append(np.asarray(pixels, dtype=np.int32))
    return found


def _background(array: np.ndarray) -> np.ndarray:
    border = np.concatenate(
        (
            array[:3].reshape(-1, 3),
            array[-3:].reshape(-1, 3),
            array[:, :3].reshape(-1, 3),
            array[:, -3:].reshape(-1, 3),
        ),
        axis=0,
    )
    return np.median(border, axis=0).astype(np.uint8)


def _score(occupancy: np.ndarray) -> tuple[int, int, int]:
    main = 0
    anti = 0
    count = 0
    for r, c in zip(*np.where(occupancy), strict=True):
        count += 1
        if r < c:
            main -= 1
        elif r > c:
            main += 1
        if r + c < GRID_N - 1:
            anti -= 1
        elif r + c > GRID_N - 1:
            anti += 1
    return main, anti, count


def _component_record(component: np.ndarray, mask: np.ndarray) -> tuple[str, tuple[int, int, int]] | None:
    y_min, x_min = component.min(axis=0)
    y_max, x_max = component.max(axis=0)
    box_h = int(y_max - y_min + 1)
    box_w = int(x_max - x_min + 1)
    if box_h != box_w or box_w % GRID_N != 0:
        return None
    cell = box_w // GRID_N
    if not (8 <= cell <= 20):
        return None
    crop = mask[int(y_min) : int(y_max) + 1, int(x_min) : int(x_max) + 1]
    occupancy = np.zeros((GRID_N, GRID_N), dtype=bool)
    for r in range(GRID_N):
        for c in range(GRID_N):
            block = crop[r * cell : (r + 1) * cell, c * cell : (c + 1) * cell]
            pixels = int(block.sum())
            if pixels == cell * cell:
                occupancy[r, c] = True
            elif pixels != 0:
                return None
    if int(crop.sum()) != int(occupancy.sum()) * cell * cell:
        return None
    main, anti, count = _score(occupancy)
    if count != OCCUPIED_CELLS:
        return None
    center_x = (int(x_min) + int(x_max)) / 2.0
    center_y = (int(y_min) + int(y_max)) / 2.0
    if center_x < WIDTH / 2 and center_y < HEIGHT / 2:
        quadrant = "NW"
    elif center_x >= WIDTH / 2 and center_y < HEIGHT / 2:
        quadrant = "NE"
    elif center_x < WIDTH / 2 and center_y >= HEIGHT / 2:
        quadrant = "SW"
    else:
        quadrant = "SE"
    return quadrant, (main, anti, count)


def decision_from_image(image: Image.Image) -> str:
    """Recover the unique zero/zero diagonal-mass plate from final RGB pixels."""
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] != HEIGHT or array.shape[1] != WIDTH:
        raise ValueError("unexpected raster dimensions")
    background = _background(array)
    distance = np.linalg.norm(array.astype(np.int16) - background.astype(np.int16), axis=2)
    mask = distance > 18.0
    records = []
    for component in _components(mask):
        if len(component) < 100:
            continue
        record = _component_record(component, mask)
        if record is not None:
            records.append(record)
    if not records or len(records) > 4:
        raise ValueError(f"expected at least one and at most four complete plate components, found {len(records)}")
    by_quadrant = {quadrant: score for quadrant, score in records}
    if len(by_quadrant) != len(records):
        raise ValueError("plate quadrants are duplicated")
    winners = [quadrant for quadrant, (main, anti, _count) in by_quadrant.items() if main == 0 and anti == 0]
    if len(winners) != 1:
        raise ValueError(f"no unique balanced plate: {by_quadrant}")
    ranked = sorted(abs(score[0]) + abs(score[1]) for score in by_quadrant.values())
    if len(ranked) < 2 or ranked[1] - ranked[0] < 2:
        raise ValueError("balanced plate runner-up margin is too small")
    return winners[0]
