"""Latent sampler and deterministic renderer for a controlled-toggle circuit."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1280
HEIGHT = 460
N_LAMPS = 4
N_GATES = 8

PALETTES = (
    {"lamp": (28, 125, 154), "control": (103, 73, 180), "target": (232, 126, 34)},
    {"lamp": (16, 110, 140), "control": (122, 76, 174), "target": (216, 112, 28)},
)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{name}", size)
    except OSError:
        return ImageFont.load_default()


def _apply_gate(bits: list[int], gate: list[int] | tuple[int, int]) -> None:
    control, target = int(gate[0]), int(gate[1])
    if bits[control]:
        bits[target] ^= 1


def _replay(scene: dict) -> list[int]:
    bits = [int(value) for value in scene["initial_bits"]]
    for gate in scene["gates"]:
        _apply_gate(bits, gate)
    return bits


def sample_scene(seed: int) -> dict:
    """Sample a balanced, causally coupled eight-gate Boolean replay."""
    rng = random.Random(int(seed))
    for _ in range(4000):
        gates: list[list[int]] = []
        for _step in range(N_GATES):
            control = rng.randrange(N_LAMPS)
            target = rng.randrange(N_LAMPS - 1)
            if target >= control:
                target += 1
            gates.append([control, target])

        desired = [rng.randrange(2) for _ in range(N_LAMPS)]
        initial = desired.copy()
        for gate in reversed(gates):
            _apply_gate(initial, gate)

        bits = initial.copy()
        active: list[bool] = []
        states_before: list[list[int]] = []
        for gate in gates:
            states_before.append(bits.copy())
            active.append(bool(bits[gate[0]]))
            _apply_gate(bits, gate)

        all_controls = {gate[0] for gate in gates}
        all_targets = {gate[1] for gate in gates}
        coupled = any(
            active[i] and gates[i][1] == gates[j][0]
            for i in range(N_GATES)
            for j in range(i + 1, N_GATES)
        )
        if (
            2 <= sum(active) <= 6
            and len(all_controls) == N_LAMPS
            and len(all_targets) == N_LAMPS
            and coupled
            and bits == desired
        ):
            return {
                "initial_bits": initial,
                "gates": gates,
                "palette_index": rng.randrange(len(PALETTES)),
                "offset_x": rng.choice([-5, 0, 5]),
                "offset_y": rng.choice([-3, 0, 3]),
            }
    raise RuntimeError("could not sample a well-coupled controlled-toggle circuit")


def analytic_gold(scene: dict) -> str:
    """Return the final A,B,C,D bit pattern."""
    return "".join(str(value) for value in _replay(scene))


def margin(scene: dict) -> float:
    """Minimum Hamming separation to a competing result across the composition."""
    return 1.0


def is_quarantined(scene: dict) -> bool:
    """The fixed glyph clearances are far from the pixel decoder boundaries."""
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No distinct latent scene is asserted to rasterize identically."""
    return []


def _centered(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font, fill) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    x = xy[0] - (box[2] - box[0]) / 2
    y = xy[1] - (box[3] - box[1]) / 2 - box[1]
    draw.text((x, y), text, font=font, fill=fill)


def _pip_stack(
    draw: ImageDraw.ImageDraw,
    x: float,
    center_y: float,
    count: int,
    color: tuple[int, int, int],
    scale: int,
) -> None:
    spacing = 16 * scale
    radius = 5.5 * scale
    start = center_y * scale - spacing * (count - 1) / 2
    for index in range(count):
        cy = start + index * spacing
        cx = x * scale
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=color)


def render(scene: dict) -> Image.Image:
    """Render all answer-bearing state and gates into one ordered raster."""
    initial_bits = [int(value) for value in scene["initial_bits"]]
    gates = [[int(pair[0]), int(pair[1])] for pair in scene["gates"]]
    palette = PALETTES[int(scene["palette_index"])]
    ox = int(scene["offset_x"])
    oy = int(scene["offset_y"])
    if len(initial_bits) != N_LAMPS or len(gates) != N_GATES:
        raise ValueError("scene must contain four starting bits and eight gates")

    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), (250, 249, 246))
    draw = ImageDraw.Draw(image)
    ink = (35, 39, 45)
    soft = (225, 226, 228)
    paper = (255, 255, 255)

    def rect(box, **kwargs):
        draw.rectangle(tuple(int(v * scale) for v in box), **kwargs)

    def rounded(box, radius, **kwargs):
        draw.rounded_rectangle(
            tuple(int(v * scale) for v in box), radius=int(radius * scale), **kwargs
        )

    title_font = _font(25 * scale, bold=True)
    medium_font = _font(15 * scale, bold=True)
    small_font = _font(12 * scale, bold=True)
    label_font = _font(18 * scale, bold=True)

    _centered(
        draw,
        ((WIDTH / 2 + ox) * scale, (29 + oy) * scale),
        "CONTROLLED LAMP CIRCUIT",
        title_font,
        ink,
    )
    _centered(
        draw,
        ((WIDTH / 2 + ox) * scale, (61 + oy) * scale),
        "purple pips READ a control  •  orange pips TOGGLE a target",
        medium_font,
        ink,
    )

    # Starting register. The four fixed horizontal slots visibly define A-D order.
    rounded((22 + ox, 90 + oy, 292 + ox, 388 + oy), 12, fill=paper, outline=ink, width=2 * scale)
    _centered(draw, ((157 + ox) * scale, (116 + oy) * scale), "START: A  B  C  D", medium_font, ink)
    lamp_y = 190 + oy
    for index, bit in enumerate(initial_bits):
        cx = 61 + ox + 63 * index
        radius = 23
        box = tuple(int(v * scale) for v in (cx - radius, lamp_y - radius, cx + radius, lamp_y + radius))
        if bit:
            draw.ellipse(box, fill=palette["lamp"], outline=palette["lamp"], width=2 * scale)
        else:
            draw.ellipse(box, fill=paper, outline=palette["lamp"], width=6 * scale)
        _centered(draw, (cx * scale, (239 + oy) * scale), "ABCD"[index], label_font, ink)
        _centered(
            draw,
            (cx * scale, (276 + oy) * scale),
            "ON" if bit else "OFF",
            small_font,
            ink,
        )
    _centered(draw, ((157 + ox) * scale, (342 + oy) * scale), "filled = 1   ring = 0", small_font, ink)

    # A single row makes the temporal order perceptually and mechanically explicit.
    card_x0 = 313 + ox
    card_w = 108
    step = 119
    for index, (control, target) in enumerate(gates):
        left = card_x0 + index * step
        right = left + card_w
        center = (left + right) / 2
        rounded((left, 90 + oy, right, 388 + oy), 10, fill=paper, outline=ink, width=2 * scale)
        _centered(draw, (center * scale, (110 + oy) * scale), str(index + 1), medium_font, ink)
        _centered(draw, (center * scale, (134 + oy) * scale), "READ", small_font, ink)
        _pip_stack(draw, center, 173 + oy, control + 1, palette["control"], scale)

        x = center * scale
        y1 = (213 + oy) * scale
        y2 = (256 + oy) * scale
        draw.line((x, y1, x, y2), fill=ink, width=4 * scale)
        draw.polygon(
            ((x, (269 + oy) * scale), ((center - 8) * scale, y2), ((center + 8) * scale, y2)),
            fill=ink,
        )
        _centered(draw, (center * scale, (291 + oy) * scale), "FLIP", small_font, ink)
        _pip_stack(draw, center, 337 + oy, target + 1, palette["target"], scale)

    _centered(
        draw,
        ((WIDTH / 2 + ox) * scale, (424 + oy) * scale),
        "Pip key: 1=A   2=B   3=C   4=D     •     Apply cards left to right",
        medium_font,
        ink,
    )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
