"""Independent pixel-only inverse arm for ensemble_average_reveal.

Receives ONLY the final PNG. It de-interleaves the K column-phase exposures
(column j belongs to frame p = j mod K), averages them element-wise, and
correlates the reconstructed field with the two declared symbol templates
(circle = A, square = B). It reads no scene data and no analytic gold.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "ensemble_average_reveal-oracle-0.1.0"

IMG_W = 256
IMG_H = 128
KFRAMES = 4
EW = IMG_W // KFRAMES
EH = IMG_H
SYM = 28
BASE = 112.0

_CACHE: dict[str, np.ndarray] = {}


def _symbol_pos(kind: str) -> np.ndarray:
    ys, xs = np.indices((EH, EW))
    cy, cx = EH // 2, EW // 2
    if kind == "A":
        r2 = (ys - cy) ** 2 + (xs - cx) ** 2
        return r2 <= SYM * SYM
    return (np.abs(ys - cy) <= SYM) & (np.abs(xs - cx) <= SYM)


def _two_tone(kind: str) -> np.ndarray:
    if kind not in _CACHE:
        _CACHE[kind] = np.where(_symbol_pos(kind) > 0, 1.0, -1.0)
    return _CACHE[kind]


def _demux_average(arr: np.ndarray) -> np.ndarray:
    """Mean over the K column-phase frames of a raw (EH, IMG_W) luminance field."""
    acc = np.zeros((EH, EW), dtype=float)
    for c in range(EW):
        cols = arr[:, c * KFRAMES : (c + 1) * KFRAMES]
        acc[:, c] = cols.mean(axis=1)
    return acc - BASE


def _score(field: np.ndarray, ref: np.ndarray) -> float:
    a = field - field.mean()
    b = ref - ref.mean()
    denom = float(np.linalg.norm(a) * np.linalg.norm(b))
    return float(np.dot(a.ravel(), b.ravel()) / denom) if denom > 1e-12 else 0.0


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("L"), dtype=float)
    field = _demux_average(arr)
    sa = _score(field, _two_tone("A"))
    sb = _score(field, _two_tone("B"))
    return "A" if sa >= sb else "B"


def reveal_margin(image: Image.Image) -> float:
    """A/B template-correlation margin recovered from pixels alone."""
    arr = np.asarray(image.convert("L"), dtype=float)
    field = _demux_average(arr)
    return abs(_score(field, _two_tone("A")) - _score(field, _two_tone("B")))
