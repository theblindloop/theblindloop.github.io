"""ensemble_average_reveal: column-multiplexed multi-copy averaging glyph reveal.

Declared transform: the image shows K independent, independently-noised
exposures of the same faint two-tone symbol, multiplexed at the column level
(column j belongs to exposure p = j mod K), so no single exposure is ever
contiguous and the symbol is invisible in direct view. De-interleave the K
exposures by their column phase, then average them element-wise (the ensemble
mean across frames). The shared symbol bias accumulates coherently across the K
frames while the independent per-frame noise averages toward zero, revealing a
coarse symbol. A = filled circle, B = filled square.

Latent scene z = {noise_seed, glyph, bias}. label = glyph.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RENDERER_VERSION = "ensemble_average_reveal-0.1.0"

IMG_W = 256
IMG_H = 128
KFRAMES = 4                 # number of column-phase exposures
EW = IMG_W // KFRAMES       # exposure width per frame (64)
EH = IMG_H
SYM = 28                    # symbol radius (circle) / half-side (square), px
BASE = 112.0                # luminance offset so field values map into [0, 255]
BIAS = 40                   # two-tone symbol bias magnitude (levels)
NOISE_FACTOR = 1.8          # per-exposure noise amplitude relative to bias
Q_TOL = 0.12                # quarantine threshold on the A/B template margin

_TEMPLATE_CACHE: dict[str, np.ndarray] = {}


def _symbol_pos(shape: tuple[int, int], kind: str) -> np.ndarray:
    """(EH, EW) boolean mask: True inside the glyph symbol for the given kind."""
    eh, ew = shape
    ys, xs = np.indices(shape)
    cy, cx = eh // 2, ew // 2
    if kind == "A":
        r2 = (ys - cy) ** 2 + (xs - cx) ** 2
        inside = r2 <= SYM * SYM
    else:  # "B"
        inside = (np.abs(ys - cy) <= SYM) & (np.abs(xs - cx) <= SYM)
    return inside


def two_tone_mask(kind: str) -> np.ndarray:
    """(EH, EW) zero-mean two-tone bias field: +1 inside symbol, -1 outside."""
    if kind not in _TEMPLATE_CACHE:
        _TEMPLATE_CACHE[kind] = np.where(_symbol_pos((EH, EW), kind) > 0, 1.0, -1.0)
    return _TEMPLATE_CACHE[kind]


def build_scene(noise_seed: int, glyph: str, bias: int = BIAS) -> dict:
    return {"noise_seed": int(noise_seed), "glyph": glyph, "bias": int(bias)}


def sample_scene(seed: int) -> dict:
    glyph = "A" if int(seed) % 2 == 0 else "B"
    return build_scene(noise_seed=int(seed), glyph=glyph, bias=BIAS)


def exposure_fields(scene: dict) -> list[np.ndarray]:
    """K (EH, EW) float exposure fields for a scene, deterministic from the seed."""
    rng = np.random.default_rng(int(scene["noise_seed"]) & 0xFFFFFFFF)
    bias = float(scene["bias"])
    tmpl = two_tone_mask(scene["glyph"])
    amp = bias * NOISE_FACTOR
    fields = []
    for _ in range(KFRAMES):
        fields.append(bias * tmpl + rng.uniform(-1.0, 1.0, size=(EH, EW)) * amp)
    return fields


def average_field(scene: dict) -> np.ndarray:
    """(EH, EW) mean over the K de-interleaved exposures (the revealed field)."""
    fields = exposure_fields(scene)
    acc = fields[0].copy()
    for field in fields[1:]:
        acc = acc + field
    return acc / KFRAMES


def render(scene: dict) -> Image.Image:
    """Interleave the K exposures at the column level into a single raster."""
    fields = exposure_fields(scene)
    arr = np.zeros((EH, IMG_W), dtype=float)
    for p, field in enumerate(fields):
        for c in range(EW):
            arr[:, p + c * KFRAMES] = field[:, c]
    arr = np.clip(arr + BASE, 0.0, 255.0).round().astype(np.uint8)
    return Image.fromarray(arr, "L").convert("RGB")


def _score(field: np.ndarray, ref: np.ndarray) -> float:
    """Normalized correlation between a field and a zero-mean symbol template."""
    a = field - field.mean()
    b = ref - ref.mean()
    denom = float(np.linalg.norm(a) * np.linalg.norm(b))
    return float(np.dot(a.ravel(), b.ravel()) / denom) if denom > 1e-12 else 0.0


def margin(scene: dict) -> float:
    """A/B template-correlation margin computed on the reconstructed field."""
    field = average_field(scene)
    return round(_score(field, two_tone_mask("A")) - _score(field, two_tone_mask("B")), 6)


def analytic_gold(scene: dict) -> str:
    return scene["glyph"]


def is_quarantined(scene: dict) -> bool:
    return abs(margin(scene)) < Q_TOL


def latent_symmetries(scene: dict):
    """not_applicable: every latent field directly sets the raster and the
    independent per-frame noise decorrelates the field, so no distinct latent
    vector reproduces the same image and the label is not hidden from rendering.
    Declared empty."""
    return iter(())
