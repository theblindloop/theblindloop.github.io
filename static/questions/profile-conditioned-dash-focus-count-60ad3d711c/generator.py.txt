"""Deterministic latent sampler, raster renderer, and analytic arm."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

SIZE = 512
N_DASHES = 16


def _dash_endpoints(fx: float, fy: float, angle: float, radius: float, length: int):
    ux, uy = math.cos(angle), math.sin(angle)
    cx, cy = fx + radius * ux, fy + radius * uy
    hx, hy = 0.5 * length * ux, 0.5 * length * uy
    return [int(round(cx - hx)), int(round(cy - hy)), int(round(cx + hx)), int(round(cy + hy))]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    focus_count = 1 + (int(seed) & 1)
    center_x = rng.randint(244, 268)
    center_y = rng.randint(244, 268)
    rotation = rng.uniform(0.0, math.pi)
    if focus_count == 1:
        foci = [[center_x, center_y]]
    else:
        separation = rng.randint(104, 128)
        dx = 0.5 * separation * math.cos(rotation)
        dy = 0.5 * separation * math.sin(rotation)
        foci = [
            [round(center_x - dx, 6), round(center_y - dy, 6)],
            [round(center_x + dx, 6), round(center_y + dy, 6)],
        ]

    # Every class has the same count and length/width distributions.  Angles are
    # globally rotated and jittered; only their line-consensus structure differs.
    dashes = []
    centers = []
    for i in range(N_DASHES):
        focus_index = 0 if focus_count == 1 else i % 2
        local_rank = i if focus_count == 1 else i // 2
        divisor = N_DASHES if focus_count == 1 else N_DASHES // 2
        base_angle = rotation + math.pi * local_rank / divisor
        if focus_count == 2 and focus_index == 1:
            base_angle += 0.17
        sign = 1 if ((local_rank + focus_index) & 1) == 0 else -1
        length = rng.choice([26, 28, 30, 32])
        for attempt in range(80):
            spread = 0.06 if attempt < 32 else 0.45
            angle = base_angle + rng.uniform(-spread, spread)
            trial_sign = sign if attempt < 32 else rng.choice([-1, 1])
            radius = trial_sign * rng.randint(124, 158)
            x1, y1, x2, y2 = _dash_endpoints(*foci[focus_index], angle, radius, length)
            center = ((x1 + x2) / 2, (y1 + y2) / 2)
            if all(math.hypot(center[0] - px, center[1] - py) >= 38 for px, py in centers):
                break
        centers.append(center)
        dashes.append([focus_index, x1, y1, x2, y2, rng.choice([5, 6])])

    shade = rng.choice([18, 26, 34, 42, 50])
    return {
        "focus_count": focus_count,
        "foci": foci,
        "dashes": dashes,
        "ink": [shade, shade, shade],
        "background": [248, 248, 246],
    }


def _polygon_for_dash(dash):
    _, x1, y1, x2, y2, width = dash
    vx, vy = x2 - x1, y2 - y1
    norm = max(1.0, math.hypot(vx, vy))
    px, py = -vy * width / (2 * norm), vx * width / (2 * norm)
    return [
        (round(x1 + px), round(y1 + py)),
        (round(x2 + px), round(y2 + py)),
        (round(x2 - px), round(y2 - py)),
        (round(x1 - px), round(y1 - py)),
    ]


def render(scene: dict) -> Image.Image:
    # Explicitly access all declared visual/label fields.  Foci are latent
    # construction variables; only the dashes themselves are painted.
    _ = scene["focus_count"]
    _ = scene["foci"]
    image = Image.new("RGB", (SIZE, SIZE), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    for dash in scene["dashes"]:
        draw.polygon(_polygon_for_dash(dash), fill=tuple(scene["ink"]))
    return image


def analytic_gold(scene: dict) -> str:
    return "one" if int(scene["focus_count"]) == 1 else "two"


def margin(scene: dict) -> float:
    # All sampled scenes have 16 lines at one focus or balanced 8/8 support.
    # The classification boundary is 12 supporting lines at one focus.
    support = N_DASHES if int(scene["focus_count"]) == 1 else N_DASHES // 2
    return float(abs(support - 12))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    for dash in twin["dashes"]:
        dash[1], dash[3] = dash[3], dash[1]
        dash[2], dash[4] = dash[4], dash[2]
    return [("reverse_all_endpoint_pairs", twin)]
