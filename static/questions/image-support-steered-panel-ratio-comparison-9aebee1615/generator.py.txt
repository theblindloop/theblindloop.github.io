"""Deterministic renderer and analytic arm for panel_ratio_comparison."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH = 900
HEIGHT = 480
BAR_WIDTH = 58
PANEL_TOP = 64
PANEL_BOTTOM = 420
LEFT_PANEL = (16, 346)
RIGHT_PANEL = (554, 884)
LEFT_BAR_X = (42, 242)
RIGHT_BAR_X = (600, 800)
BACKGROUND = (248, 248, 245)
INK = (56, 61, 67)
FRAME = (112, 118, 124)
BASELINE = (76, 82, 88)
PALETTE = (
    (24, 116, 205),
    (224, 82, 55),
    (23, 157, 105),
    (146, 78, 188),
    (204, 54, 124),
    (13, 151, 171),
    (224, 145, 18),
    (76, 101, 210),
)
QUARANTINE_MARGIN = 0.12


def _ratio(heights: Iterable[int]) -> float:
    a, b = (int(value) for value in heights)
    return max(a, b) / min(a, b)


def _cue_condition(
    winner: tuple[int, int], loser: tuple[int, int], align: bool
) -> bool:
    winner_max, loser_max = max(winner), max(loser)
    winner_sum, loser_sum = sum(winner), sum(loser)
    if align:
        return winner_max > loser_max + 8 and winner_sum > loser_sum + 12
    return winner_max + 8 < loser_max and winner_sum + 12 < loser_sum


def sample_scene(seed: int) -> dict:
    """Sample balanced answers and counterbalance raw height/ink cues."""

    rng = random.Random(int(seed))
    winner_is_left = int(seed) % 2 == 0
    cue_aligns = (int(seed) // 2) % 2 == 0
    target_gaps = (0.08, 0.14, 0.20, 0.32, 0.50, 0.74)
    target_gap = target_gaps[int(seed) % len(target_gaps)]

    for _ in range(20_000):
        lower_ratio = rng.uniform(1.25, 2.08)
        higher_ratio = lower_ratio + target_gap
        winner_short = rng.randint(54, 112)
        loser_short = rng.randint(54, 112)
        winner = (winner_short, round(winner_short * higher_ratio))
        loser = (loser_short, round(loser_short * lower_ratio))
        if max(*winner, *loser) > 250:
            continue
        actual_gap = _ratio(winner) - _ratio(loser)
        if actual_gap <= 0.035 or abs(actual_gap - target_gap) > 0.035:
            continue
        if not _cue_condition(winner, loser, cue_aligns):
            continue
        winner_slots = list(winner)
        loser_slots = list(loser)
        rng.shuffle(winner_slots)
        rng.shuffle(loser_slots)
        if winner_is_left:
            left_heights, right_heights = winner_slots, loser_slots
        else:
            left_heights, right_heights = loser_slots, winner_slots
        break
    else:
        raise RuntimeError(f"could not sample a counterbalanced scene for seed {seed}")

    colors = list(PALETTE)
    rng.shuffle(colors)
    # Small independent translations exercise position variation while retaining
    # the large blank gap required by the support-behavior target.
    panel_offsets = [
        [rng.randint(-4, 4), rng.randint(-5, 5)],
        [rng.randint(-4, 4), rng.randint(-5, 5)],
    ]
    return {
        "left_heights": left_heights,
        "right_heights": right_heights,
        "left_colors": [list(colors[0]), list(colors[1])],
        "right_colors": [list(colors[2]), list(colors[3])],
        "panel_offsets": panel_offsets,
    }


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    panel: tuple[int, int],
    bar_xs: tuple[int, int],
    heights: list[int],
    colors: list[list[int]],
    offset: list[int],
    label: str,
) -> None:
    dx, dy = (int(offset[0]), int(offset[1]))
    x0, x1 = panel[0] + dx, panel[1] + dx
    top, bottom = PANEL_TOP + dy, PANEL_BOTTOM + dy
    draw.rounded_rectangle((x0, top, x1, bottom), radius=8, outline=FRAME, width=3)
    baseline_y = bottom - 29
    draw.line((x0 + 14, baseline_y, x1 - 14, baseline_y), fill=BASELINE, width=4)
    try:
        font = ImageFont.load_default(size=24)
    except TypeError:  # Pillow compatibility fallback
        font = ImageFont.load_default()
    draw.text(((x0 + x1) // 2, top + 18), label, fill=INK, anchor="mm", font=font)
    for slot, base_x in enumerate(bar_xs):
        bar_x = base_x + dx
        height = int(heights[slot])
        # Saturated rectangles stop four pixels above the neutral baseline.
        bar_bottom = baseline_y - 4
        draw.rectangle(
            (bar_x, bar_bottom - height + 1, bar_x + BAR_WIDTH - 1, bar_bottom),
            fill=tuple(int(value) for value in colors[slot]),
        )


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_panel(
        draw,
        LEFT_PANEL,
        LEFT_BAR_X,
        scene["left_heights"],
        scene["left_colors"],
        scene["panel_offsets"][0],
        "LEFT",
    )
    _draw_panel(
        draw,
        RIGHT_PANEL,
        RIGHT_BAR_X,
        scene["right_heights"],
        scene["right_colors"],
        scene["panel_offsets"][1],
        "RIGHT",
    )
    return image


def analytic_gold(scene: dict) -> str:
    left_ratio = _ratio(scene["left_heights"])
    right_ratio = _ratio(scene["right_heights"])
    return "LEFT" if left_ratio > right_ratio else "RIGHT"


def margin(scene: dict) -> float:
    return abs(_ratio(scene["left_heights"]) - _ratio(scene["right_heights"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
