"""Independent pixel-only inverse arm for the panel ratio comparison."""

from __future__ import annotations

from collections import deque
from dataclasses import dataclass

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    """Raised when four intact colored bar components cannot be recovered."""


@dataclass(frozen=True)
class Component:
    x0: int
    y0: int
    x1: int
    y1: int
    area: int

    @property
    def width(self) -> int:
        return self.x1 - self.x0 + 1

    @property
    def height(self) -> int:
        return self.y1 - self.y0 + 1

    @property
    def cx(self) -> float:
        return (self.x0 + self.x1) / 2.0


def _saturated_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    spread = array.max(axis=2) - array.min(axis=2)
    # Neutral frames, labels, baseline, and background have near-zero spread.
    return (spread >= 55) & (array.max(axis=2) <= 245) & (array.min(axis=2) <= 205)


def _components(mask: np.ndarray) -> list[Component]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[Component] = []
    for start_y, start_x in np.argwhere(mask):
        sy, sx = int(start_y), int(start_x)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        area = len(xs)
        if area >= 500:
            output.append(Component(min(xs), min(ys), max(xs), max(ys), area))
    return output


def _measure_heights(image: Image.Image) -> tuple[tuple[int, int], tuple[int, int]]:
    components = sorted(_components(_saturated_mask(image)), key=lambda item: item.cx)
    if len(components) != 4:
        raise OracleAbstention(f"expected four intact colored bars, found {len(components)}")
    widths = [item.width for item in components]
    if min(widths) < 40 or max(widths) - min(widths) > 3:
        raise OracleAbstention(f"bar widths are incomplete or inconsistent: {widths}")
    if any(item.area < item.width * item.height * 0.98 for item in components):
        raise OracleAbstention("a colored bar is not a filled rectangle")
    local_left_gap = components[1].cx - components[0].cx
    panel_gap = components[2].cx - components[1].cx
    local_right_gap = components[3].cx - components[2].cx
    if panel_gap <= 1.6 * max(local_left_gap, local_right_gap):
        raise OracleAbstention("the two separated bar panels cannot be bound reliably")
    left = (components[0].height, components[1].height)
    right = (components[2].height, components[3].height)
    if min(*left, *right) < 45:
        raise OracleAbstention("a recovered bar is too short")
    return left, right


def _ratio(pair: tuple[int, int]) -> float:
    return max(pair) / min(pair)


def decision_from_image(image: Image.Image) -> str:
    """Return LEFT or RIGHT using only segmented rectangle heights."""

    left, right = _measure_heights(image)
    difference = _ratio(left) - _ratio(right)
    if abs(difference) < 0.035:
        raise OracleAbstention("the recovered panel ratios are indistinguishable")
    return "LEFT" if difference > 0 else "RIGHT"
