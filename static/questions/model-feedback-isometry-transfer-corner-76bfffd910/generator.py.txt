"""Isometry-transfer world.

The raster shows three squares. The left square holds a solid black pentomino;
the middle square holds the same pentomino after one unknown element of the
square's symmetry group (a quarter turn or a mirror) has been applied. A thick
black arrow points from the left square to the middle square. To the right of a
light separator a third square holds four differently colored disks, one near
each of its corners.

The decision is the color that occupies the top-left corner of the third square
once the same unknown motion has been applied to its contents. That configuration
is drawn nowhere: the solver must recover the motion from the example pair and
transfer it to a different object.

Public interface required by question-world@0.4.0:
``sample_scene``, ``render``, ``analytic_gold``, ``margin``, ``is_quarantined``
and ``latent_symmetries``.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 1200
HEIGHT = 440

GRID = 4
CELL = 46
BORDER = 5
INNER = GRID * CELL          # 184
OUTER = INNER + 2 * BORDER   # 194

DOT_R = 22
JITTER = 8
QUARANTINE_PX = 11.0

WHITE = (255, 255, 255)
BORDER_GRAY = (150, 150, 150)
SEPARATOR_GRAY = (205, 205, 205)
INK = (0, 0, 0)

# Corner order is TL, TR, BR, BL throughout.
CORNER_NAMES = ("top-left", "top-right", "bottom-right", "bottom-left")
CORNER_RC = ((0, 0), (0, 1), (1, 1), (1, 0))

COLOR_NAMES = ("blue", "green", "red", "yellow")
PALETTE = {
    "red": (220, 40, 40),
    "blue": (40, 80, 220),
    "green": (30, 150, 60),
    "yellow": (240, 190, 20),
}

# Pentominoes whose symmetry group is trivial, so the motion that maps the left
# glyph onto the middle glyph is unique.
SHAPES = {
    "f": ((0, 1), (0, 2), (1, 0), (1, 1), (2, 1)),
    "l": ((0, 0), (1, 0), (2, 0), (3, 0), (3, 1)),
    "n": ((0, 1), (1, 1), (2, 0), (2, 1), (3, 0)),
    "p": ((0, 0), (0, 1), (1, 0), (1, 1), (2, 0)),
    "y": ((0, 1), (1, 0), (1, 1), (2, 1), (3, 1)),
}
SHAPE_NAMES = tuple(sorted(SHAPES))

ELEMENTS = (
    "identity",
    "turn_right",       # 90 degrees clockwise
    "half_turn",        # 180 degrees
    "turn_left",        # 90 degrees counter-clockwise
    "mirror_vertical",  # left-right mirror (vertical axis)
    "mirror_horizontal",  # top-bottom mirror (horizontal axis)
    "mirror_main",      # mirror across the top-left/bottom-right diagonal
    "mirror_anti",      # mirror across the top-right/bottom-left diagonal
)

# ``mirror_main`` and ``identity`` both keep the top-left corner where it is, so
# they are never sampled as the hidden motion: reporting the drawn top-left
# color is then always a wrong policy.
HIDDEN_ELEMENTS = (
    "turn_right",
    "half_turn",
    "turn_left",
    "mirror_vertical",
    "mirror_horizontal",
    "mirror_anti",
)

TILE_BASE = ((60, 120), (410, 120), (900, 120))
SEPARATOR_X = 770
ARROW_X0 = 290
ARROW_X1 = 368


def _apply(cells, element, rows, cols):
    """Map ``cells`` through ``element`` of the square's symmetry group."""

    out = []
    for r, c in cells:
        if element == "identity":
            out.append((r, c))
        elif element == "turn_right":
            out.append((c, rows - 1 - r))
        elif element == "half_turn":
            out.append((rows - 1 - r, cols - 1 - c))
        elif element == "turn_left":
            out.append((cols - 1 - c, r))
        elif element == "mirror_vertical":
            out.append((r, cols - 1 - c))
        elif element == "mirror_horizontal":
            out.append((rows - 1 - r, c))
        elif element == "mirror_main":
            out.append((c, r))
        elif element == "mirror_anti":
            out.append((cols - 1 - c, rows - 1 - r))
        else:
            raise ValueError(f"unknown element {element!r}")
    if element in ("turn_right", "turn_left", "mirror_main", "mirror_anti"):
        return sorted(out), cols, rows
    return sorted(out), rows, cols


def _normalize(cells):
    """Crop a cell set to its bounding box; returns (cells, rows, cols)."""

    rs = [r for r, _ in cells]
    cs = [c for _, c in cells]
    r0, c0 = min(rs), min(cs)
    return (
        sorted((r - r0, c - c0) for r, c in cells),
        max(rs) - r0 + 1,
        max(cs) - c0 + 1,
    )


def _glyph_masks(scene):
    """Bounding-box-normalized cell sets drawn in the left and middle tiles."""

    base, rows, cols = _apply(SHAPES[scene["shape"]], scene["base"], 4, 4)
    left = _normalize(base)
    moved, rows2, cols2 = _apply(left[0], scene["element"], left[1], left[2])
    return left, _normalize(moved)


def _element_between(left, right):
    """The unique element mapping normalized ``left`` onto normalized ``right``."""

    hits = []
    for element in ELEMENTS:
        cells, rows, cols = _apply(left[0], element, left[1], left[2])
        if _normalize(cells) == right:
            hits.append(element)
    return hits


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    corners = list(COLOR_NAMES)
    rng.shuffle(corners)
    scene = {
        "shape": rng.choice(SHAPE_NAMES),
        "base": rng.choice(ELEMENTS),
        "element": rng.choice(HIDDEN_ELEMENTS),
        "corners": corners,
        "jitter": [rng.randint(-JITTER, JITTER) for _ in range(6)],
    }
    return scene


def _tiles(scene):
    jitter = scene["jitter"]
    return [
        (TILE_BASE[i][0] + jitter[2 * i], TILE_BASE[i][1] + jitter[2 * i + 1])
        for i in range(3)
    ]


def _draw_tile(draw, x, y):
    draw.rectangle(
        [x, y, x + OUTER - 1, y + OUTER - 1], outline=BORDER_GRAY, width=BORDER
    )


def _draw_glyph(draw, x, y, mask):
    cells, rows, cols = mask
    off_r = (GRID - rows) // 2
    off_c = (GRID - cols) // 2
    for r, c in cells:
        px = x + BORDER + (c + off_c) * CELL
        py = y + BORDER + (r + off_r) * CELL
        draw.rectangle([px, py, px + CELL - 1, py + CELL - 1], fill=INK)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), WHITE)
    draw = ImageDraw.Draw(image)

    left, right = _glyph_masks(scene)
    tiles = _tiles(scene)

    _draw_tile(draw, *tiles[0])
    _draw_glyph(draw, tiles[0][0], tiles[0][1], left)
    _draw_tile(draw, *tiles[1])
    _draw_glyph(draw, tiles[1][0], tiles[1][1], right)

    # Arrow from the left tile to the middle tile.
    mid_y = tiles[0][1] + OUTER // 2
    draw.rectangle([ARROW_X0, mid_y - 9, ARROW_X1 - 30, mid_y + 9], fill=INK)
    draw.polygon(
        [(ARROW_X1 - 34, mid_y - 26), (ARROW_X1, mid_y), (ARROW_X1 - 34, mid_y + 26)],
        fill=INK,
    )

    draw.rectangle(
        [SEPARATOR_X, 54, SEPARATOR_X + 3, HEIGHT - 55], fill=SEPARATOR_GRAY
    )

    _draw_tile(draw, *tiles[2])
    cx, cy = tiles[2]
    for name, (r, c) in zip(scene["corners"], CORNER_RC):
        # Corner cell centers of the 4x4 lattice: cells (0,0), (0,3), (3,3), (3,0).
        cell_r = 0 if r == 0 else GRID - 1
        cell_c = 0 if c == 0 else GRID - 1
        px = cx + BORDER + cell_r * 0 + (cell_c + 0.5) * CELL
        py = cy + BORDER + (cell_r + 0.5) * CELL
        draw.ellipse(
            [px - DOT_R, py - DOT_R, px + DOT_R, py + DOT_R], fill=PALETTE[name]
        )
    return image


def analytic_gold(scene: dict) -> str:
    """Color standing in the top-left corner after the hidden motion."""

    element = scene["element"]
    for index, (r, c) in enumerate(CORNER_RC):
        moved, _, _ = _apply([(r, c)], element, 2, 2)
        if moved[0] == (0, 0):
            return scene["corners"][index]
    raise AssertionError("no corner maps to the top-left corner")


def margin(scene: dict) -> float:
    left, right = _glyph_masks(scene)

    # Shape decisiveness: how many cells separate the true motion's image from
    # the closest competing element's image of the left glyph.
    best = None
    for element in ELEMENTS:
        cells, rows, cols = _apply(left[0], element, left[1], left[2])
        candidate = _normalize(cells)
        if candidate == right:
            continue
        diff = len(set(candidate[0]) ^ set(right[0]))
        if candidate[1] != right[1] or candidate[2] != right[2]:
            diff = max(diff, 2)
        best = diff if best is None else min(best, diff)
    shape_px = float(CELL * (best if best is not None else 0))

    # Disk clearance inside the probe tile.
    dot_gap = float((GRID - 1) * CELL - 2 * DOT_R)

    # Layout clearances between neighbouring marks and the canvas edge.
    tiles = _tiles(scene)
    gaps = [
        float(tiles[1][0] - (tiles[0][0] + OUTER)),
        float(SEPARATOR_X - (tiles[1][0] + OUTER)),
        float(tiles[2][0] - (SEPARATOR_X + 4)),
        float(WIDTH - (tiles[2][0] + OUTER)),
        float(tiles[0][0]),
        float(min(t[1] for t in tiles)),
        float(HEIGHT - max(t[1] for t in tiles) - OUTER),
        float(ARROW_X0 - (tiles[0][0] + OUTER)),
        float(tiles[1][0] - ARROW_X1),
    ]
    layout = min(gaps)

    return float(min(shape_px, dot_gap, layout, float(CELL)))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list:
    """No latent aliases exist, so no scene transform can preserve the raster.

    Rendering is injective over sampled scenes: the pentomino identity and its
    starting orientation fix the left glyph's cells, the hidden motion fixes the
    middle glyph's cells, each disk color is drawn at the corner it is bound to,
    and the jitter shifts tile pixels directly. Distinct scenes therefore give
    distinct rasters and every raster carries exactly one decision, so the
    symmetry gate is reported as ``not_applicable`` rather than passed.
    """

    return []
