"""Independent image-only inverse arm for trace chord lobe balance."""

from __future__ import annotations

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    """The raster does not contain one intact, measurable trace."""


def _trace_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red = array[:, :, 0]
    green = array[:, :, 1]
    blue = array[:, :, 2]
    return (
        (red < 92)
        & (green > 82)
        & (blue > 92)
        & ((green - red) > 58)
        & ((blue - red) > 62)
    )


def _longest_false_run(values: np.ndarray) -> int:
    best = current = 0
    for value in values.tolist():
        if value:
            current = 0
        else:
            current += 1
            best = max(best, current)
    return best


def measure_signed_mean_gap(image: Image.Image) -> float:
    """Recover the trace and independently integrate chord-minus-trace height."""
    mask = _trace_mask(image)
    height, width = mask.shape
    ys, xs = np.nonzero(mask)
    if len(xs) < 1400:
        raise OracleAbstention("too little teal ink for a complete trace")
    x0 = int(xs.min())
    x1 = int(xs.max())
    if x0 > int(0.11 * width) or x1 < int(0.89 * width):
        raise OracleAbstention("trace endpoints do not span the canvas")
    counts = mask[:, x0 : x1 + 1].sum(axis=0)
    active = counts >= 2
    if float(active.mean()) < 0.985 or _longest_false_run(active) > 2:
        raise OracleAbstention("trace is cropped or broken across columns")

    x_values = np.arange(x0, x1 + 1, dtype=np.float64)
    centers = np.full(len(x_values), np.nan, dtype=np.float64)
    for offset, x in enumerate(range(x0, x1 + 1)):
        column_y = np.flatnonzero(mask[:, x])
        if len(column_y) >= 2:
            if int(column_y.max() - column_y.min()) > 30:
                raise OracleAbstention("a column contains multiple trace branches")
            centers[offset] = float(np.median(column_y))
    valid = np.isfinite(centers)
    if valid.sum() < 0.985 * len(centers):
        raise OracleAbstention("insufficient centerline coverage")
    centers = np.interp(x_values, x_values[valid], centers[valid])

    fit_count = min(22, max(10, len(centers) // 12))
    left_fit = np.polyfit(x_values[:fit_count], centers[:fit_count], 1)
    right_fit = np.polyfit(x_values[-fit_count:], centers[-fit_count:], 1)
    left_y = float(np.polyval(left_fit, x_values[0]))
    right_y = float(np.polyval(right_fit, x_values[-1]))
    chord = left_y + (right_y - left_y) * (x_values - x_values[0]) / (
        x_values[-1] - x_values[0]
    )
    signed_mean = float(np.mean(chord - centers))
    if not np.isfinite(signed_mean) or abs(signed_mean) < 0.75:
        raise OracleAbstention("gap balance is below raster resolution")
    return signed_mean


def decision_from_image(image: Image.Image) -> str:
    value = measure_signed_mean_gap(image)
    return "ABOVE" if value > 0.0 else "BELOW"
