"""Latent sampler, raster renderer, and analytic arm for trace chord balance."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 640
HEIGHT = 400
STRESS_SAFE_MARGIN = 4.0
TRACE_PALETTE = (
    (4, 126, 154),
    (8, 142, 132),
    (14, 116, 174),
    (0, 137, 161),
)
BACKGROUNDS = ((250, 249, 246), (248, 250, 249), (251, 250, 247))


def _signed_mean_gap(points: Iterable[Iterable[float]]) -> float:
    """Mean signed vertical gap, positive when the trace lies above its chord."""
    pts = [(float(p[0]), float(p[1])) for p in points]
    x0, y0 = pts[0]
    x1, y1 = pts[-1]
    span = x1 - x0
    if span <= 0:
        raise ValueError("trace endpoints must be ordered left to right")
    area = 0.0
    for (xa, ya), (xb, yb) in zip(pts, pts[1:]):
        if xb <= xa:
            raise ValueError("trace must be strictly x-monotone")
        ca = y0 + (y1 - y0) * (xa - x0) / span
        cb = y0 + (y1 - y0) * (xb - x0) / span
        ga = ca - ya
        gb = cb - yb
        area += 0.5 * (ga + gb) * (xb - xa)
    return area / span


def _make_points(rng: random.Random, desired_gap: float) -> list[list[float]]:
    """Make a varied two-sided trace with a prescribed signed mean gap."""
    count = rng.choice((11, 12, 13))
    x0 = rng.uniform(30.0, 45.0)
    x1 = rng.uniform(595.0, 610.0)
    span = x1 - x0
    raw_xs = [x0]
    for i in range(1, count - 1):
        nominal = x0 + span * i / (count - 1)
        raw_xs.append(nominal + rng.uniform(-8.0, 8.0))
    raw_xs.append(x1)
    xs = sorted(raw_xs)

    y0 = rng.uniform(155.0, 235.0)
    y1 = y0 + rng.uniform(-42.0, 42.0)
    phase1 = rng.uniform(0.0, 2.0 * math.pi)
    phase2 = rng.uniform(0.0, 2.0 * math.pi)
    amp1 = rng.uniform(38.0, 64.0)
    amp2 = rng.uniform(18.0, 34.0)
    deviations = []
    weights = []
    for x in xs:
        t = (x - x0) / span
        envelope = math.sin(math.pi * t)
        value = envelope * (
            amp1 * math.sin(3.0 * math.pi * t + phase1)
            + amp2 * math.sin(5.0 * math.pi * t + phase2)
            + rng.uniform(-8.0, 8.0)
        )
        deviations.append(value)
        weights.append(envelope)
    deviations[0] = deviations[-1] = 0.0
    weights[0] = weights[-1] = 0.0

    def points_for(adjustment: float) -> list[list[float]]:
        result = []
        for x, deviation, weight in zip(xs, deviations, weights):
            chord_y = y0 + (y1 - y0) * (x - x0) / span
            result.append([x, chord_y + deviation + adjustment * weight])
        return result

    base = points_for(0.0)
    unit = points_for(1.0)
    base_gap = _signed_mean_gap(base)
    unit_effect = _signed_mean_gap(unit) - base_gap
    if abs(unit_effect) < 0.1:
        raise RuntimeError("degenerate adjustment basis")
    adjustment = (desired_gap - base_gap) / unit_effect
    points = points_for(adjustment)
    if min(y for _, y in points) < 38.0 or max(y for _, y in points) > HEIGHT - 38.0:
        raise RuntimeError("sampled trace exceeded vertical safety bounds")
    return [[round(x, 3), round(y, 3)] for x, y in points]


def sample_scene(seed: int) -> dict:
    """Sample a deterministic, class-balanced, raster-safe trace."""
    rng = random.Random(int(seed) ^ 0x5A17C0DE)
    near_boundary = int(seed) % 12 in (0, 1)
    magnitude = rng.uniform(2.35, 3.25) if near_boundary else rng.uniform(15.0, 27.0)
    desired = magnitude if int(seed) % 2 == 0 else -magnitude
    for _ in range(120):
        try:
            points = _make_points(rng, desired)
        except RuntimeError:
            continue
        actual = _signed_mean_gap(points)
        if (actual > 0) == (desired > 0) and abs(actual) > 2.0:
            return {
                "points": points,
                "width": WIDTH,
                "height": HEIGHT,
                "stroke_width": rng.choice((6, 7, 8)),
                "trace_color": list(rng.choice(TRACE_PALETTE)),
                "background": list(rng.choice(BACKGROUNDS)),
            }
    raise RuntimeError(f"could not sample safe trace for seed {seed}")


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    stroke_width = int(scene["stroke_width"])
    trace_color = tuple(int(v) for v in scene["trace_color"])
    background = tuple(int(v) for v in scene["background"])
    points = [(float(p[0]), float(p[1])) for p in scene["points"]]
    scale = 4
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    scaled = [(round(x * scale), round(y * scale)) for x, y in points]
    draw.line(scaled, fill=trace_color, width=stroke_width * scale, joint="curve")
    radius = stroke_width * scale / 2.0
    for x, y in scaled[1:-1]:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=trace_color)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    value = _signed_mean_gap(scene["points"])
    return "ABOVE" if value > 0.0 else "BELOW"


def margin(scene: dict) -> float:
    return abs(_signed_mean_gap(scene["points"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= STRESS_SAFE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
