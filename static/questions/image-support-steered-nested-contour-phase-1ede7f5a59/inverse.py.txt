"""Independent image-only inverse arm for nested contour phase."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


def _foreground(rgb: np.ndarray) -> np.ndarray:
    # The canvas is a low-saturation warm neutral; every allowed ring is saturated.
    spread = rgb.max(axis=2).astype(np.int16) - rgb.min(axis=2).astype(np.int16)
    return spread > 35


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    mask = _foreground(rgb)
    ys, xs = np.nonzero(mask)
    if len(xs) < 900:
        raise ValueError("no complete cog-ring found")
    cx, cy = float(xs.mean()), float(ys.mean())
    angles = np.linspace(0.0, 2.0 * math.pi, 720, endpoint=False)
    inner_values = []
    outer_values = []
    usable_angles = []
    max_radius = min(rgb.shape[0], rgb.shape[1]) // 3
    radii = np.arange(1, max_radius, dtype=float)
    for theta in angles:
        xx = np.rint(cx + radii * math.cos(theta)).astype(int)
        yy = np.rint(cy + radii * math.sin(theta)).astype(int)
        inside = (xx >= 0) & (xx < mask.shape[1]) & (yy >= 0) & (yy < mask.shape[0])
        hits = radii[inside][mask[yy[inside], xx[inside]]]
        # The staggered contours intentionally make the band narrow at a few
        # angles; three radial pixels still identify both boundaries reliably.
        if len(hits) < 3:
            continue
        inner_values.append(float(hits.min()))
        outer_values.append(float(hits.max()))
        usable_angles.append(theta)
    if len(usable_angles) < 600:
        raise ValueError("ring contour is incomplete")
    a = np.asarray(usable_angles)
    harmonic_cos = np.cos(8.0 * a)
    harmonic_sin = np.sin(8.0 * a)
    outer = np.asarray(outer_values)
    inner = np.asarray(inner_values)
    outer -= outer.mean()
    inner -= inner.mean()
    oc, os = float(np.dot(outer, harmonic_cos)), float(np.dot(outer, harmonic_sin))
    ic, ins = float(np.dot(inner, harmonic_cos)), float(np.dot(inner, harmonic_sin))
    strength = math.hypot(oc, os) * math.hypot(ic, ins)
    if strength < 10000.0:
        raise ValueError("periodic contour energy is too weak")
    phase_agreement = (oc * ic + os * ins) / strength
    if abs(phase_agreement) < 0.03:
        raise ValueError("phase is too close to the visible quarter-step boundary")
    return "aligned" if phase_agreement > 0.0 else "staggered"
