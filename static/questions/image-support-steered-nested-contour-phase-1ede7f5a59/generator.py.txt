"""Deterministic sampler and renderer for one compact nested-contour phase cue."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

SIZE = 384
N_TEETH = 8
TOOTH_STEP = 2.0 * math.pi / N_TEETH
BACKGROUND = (248, 246, 240)
PALETTE = ((32, 102, 154), (35, 126, 105), (126, 76, 150), (184, 86, 48))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    phase_class = "aligned" if rng.randrange(2) == 0 else "staggered"
    base = 0.0 if phase_class == "aligned" else TOOTH_STEP / 2.0
    # Roughly one scene in eight approaches the visible quarter-step boundary.
    if seed % 8 == 0:
        side = -1.0 if phase_class == "aligned" else 1.0
        phase_offset = TOOTH_STEP / 4.0 + side * rng.uniform(0.010, 0.016)
    else:
        phase_offset = base + rng.uniform(-0.055, 0.055)
    return {
        "phase_class": phase_class,
        "phase_offset": phase_offset,
        "center_x": rng.randint(105, 279),
        "center_y": rng.randint(105, 279),
        "rotation": rng.uniform(0.0, TOOTH_STEP),
        "outer_radius": rng.randint(52, 59),
        "inner_radius": rng.randint(27, 32),
        "outer_amplitude": rng.randint(8, 11),
        "inner_amplitude": rng.randint(6, 9),
        "color_index": rng.randrange(len(PALETTE)),
    }


def _radial_polygon(cx: float, cy: float, base: float, amplitude: float, phase: float) -> list[tuple[int, int]]:
    points = []
    for index in range(720):
        theta = 2.0 * math.pi * index / 720.0
        radius = base + amplitude * math.cos(N_TEETH * (theta - phase))
        points.append((round(cx + radius * math.cos(theta)), round(cy + radius * math.sin(theta))))
    return points


def render(scene: dict) -> Image.Image:
    phase_class = scene["phase_class"]
    if phase_class not in {"aligned", "staggered"}:
        raise ValueError("invalid phase class")
    image = Image.new("RGB", (SIZE, SIZE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    cx, cy = float(scene["center_x"]), float(scene["center_y"])
    rotation = float(scene["rotation"])
    outer = _radial_polygon(
        cx, cy, float(scene["outer_radius"]), float(scene["outer_amplitude"]), rotation
    )
    inner = _radial_polygon(
        cx,
        cy,
        float(scene["inner_radius"]),
        float(scene["inner_amplitude"]),
        rotation + float(scene["phase_offset"]),
    )
    draw.polygon(outer, fill=PALETTE[int(scene["color_index"])])
    draw.polygon(inner, fill=BACKGROUND)
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["phase_class"])


def margin(scene: dict) -> float:
    phase = float(scene["phase_offset"]) % TOOTH_STEP
    boundaries = (TOOTH_STEP / 4.0, 3.0 * TOOTH_STEP / 4.0)
    return min(abs(phase - boundary) for boundary in boundaries) * 180.0 / math.pi


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["rotation"] = float(scene["rotation"]) + TOOTH_STEP
    return [("rotate_latent_by_one_tooth", twin)]
