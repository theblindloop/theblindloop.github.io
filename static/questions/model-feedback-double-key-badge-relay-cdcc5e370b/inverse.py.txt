"""Independent raster-only inverse for the double-key relay board."""

from __future__ import annotations

import math

from PIL import Image


STATIONS = ("red", "blue", "green", "orange", "violet")
STATION_RGB = (
    (215, 56, 69),
    (45, 118, 190),
    (42, 157, 91),
    (239, 143, 42),
    (128, 83, 184),
)
BADGES = (
    ("0111110", "1100011", "1000001", "1000001", "1000001", "1100011", "0111110"),
    ("0001000", "0001000", "0001000", "1111111", "0001000", "0001000", "0001000"),
    ("1111111", "0111110", "0011100", "0001000", "0001000", "0001000", "0001000"),
    ("1000001", "0100010", "0010100", "0001000", "0010100", "0100010", "1000001"),
    ("1010101", "0000000", "1111111", "0000000", "1010101", "0000000", "1111111"),
)
SIGILS = (
    ("0011100", "0100010", "1000001", "1000001", "1000001", "0100010", "0011100"),
    ("0001000", "0011100", "0101010", "1000001", "1111111", "0001000", "0001000"),
    ("1111111", "1000001", "1010101", "1000001", "1010101", "1000001", "1111111"),
    ("1000001", "0100010", "0010100", "0001000", "0010100", "0100010", "1000001"),
    ("1111111", "0000001", "0000010", "0000100", "0001000", "0010000", "1111111"),
)

ROW_TOP = 188
ROW_H = 112
ICON_SIZE = 98
QUERY_ICON_X, QUERY_ICON_Y = 131, 268
A_BADGE_X, A_SIGIL_X = 382, 716
B_SIGIL_X, B_SWATCH_X = 930, 1260


def _dark(pixel: tuple[int, int, int]) -> bool:
    return sum(pixel) < 360


def _read_bitmap(image: Image.Image, left: int, top: int, templates: tuple[tuple[str, ...], ...]) -> int:
    # The renderer places a 7x7 bitmap at a 14-pixel inset with 10-pixel cells.
    observed = []
    for row in range(7):
        values = []
        for col in range(7):
            pixel = image.getpixel((left + 14 + col * 10 + 5, top + 14 + row * 10 + 5))
            values.append("1" if _dark(pixel) else "0")
        observed.append("".join(values))
    scores = []
    for template in templates:
        scores.append(sum(a != b for a, b in zip(observed, template)))
    best = min(range(len(scores)), key=scores.__getitem__)
    if scores[best] > 0:
        raise ValueError("icon bitmap is not legible")
    return best


def _read_station(image: Image.Image, left: int, top: int) -> int:
    pixel = image.getpixel((left + ICON_SIZE // 2, top + ICON_SIZE // 2))
    distances = [
        math.sqrt(sum((float(a) - float(b)) ** 2 for a, b in zip(pixel, rgb)))
        for rgb in STATION_RGB
    ]
    best = min(range(len(distances)), key=distances.__getitem__)
    if distances[best] > 20.0:
        raise ValueError("station swatch is not legible")
    return best


def decision_from_image(image: Image.Image) -> str:
    """Recover badge -> sigil -> station using pixels and no latent scene."""

    image = image.convert("RGB")
    query = _read_bitmap(image, QUERY_ICON_X, QUERY_ICON_Y, BADGES)
    badge_to_sigil = {}
    for row in range(5):
        top = ROW_TOP + row * ROW_H
        badge = _read_bitmap(image, A_BADGE_X, top, BADGES)
        sigil = _read_bitmap(image, A_SIGIL_X, top, SIGILS)
        if badge in badge_to_sigil:
            raise ValueError("duplicate badge in KEY A")
        badge_to_sigil[badge] = sigil

    sigil_to_station = {}
    for row in range(5):
        top = ROW_TOP + row * ROW_H
        sigil = _read_bitmap(image, B_SIGIL_X, top, SIGILS)
        station = _read_station(image, B_SWATCH_X, top)
        if sigil in sigil_to_station:
            raise ValueError("duplicate sigil in KEY B")
        sigil_to_station[sigil] = station

    if set(badge_to_sigil) != set(range(5)) or set(sigil_to_station) != set(range(5)):
        raise ValueError("incomplete visible key")
    station = sigil_to_station[badge_to_sigil[query]]
    return STATIONS[station]
