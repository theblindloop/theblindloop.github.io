def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    candidates = (
        Path("/usr/share/fonts/truetype/dejavu") / name,
        Path("/usr/share/fonts/dejavu") / name,
    )
    for path in candidates:
        if path.is_file():
            return ImageFont.truetype(str(path), size)
    return ImageFont.load_default()


def _draw_bitmap(
    draw: ImageDraw.ImageDraw,
    left: int,
    top: int,
    bitmap: tuple[str, ...],
    *,
    cell: int = 10,
    fill: tuple[int, int, int] = INK,
) -> None:
    for row_index, row in enumerate(bitmap):
        for col_index, value in enumerate(row):
            if value == "1":
                x0 = left + col_index * cell
                y0 = top + row_index * cell
                draw.rectangle((x0, y0, x0 + cell - 1, y0 + cell - 1), fill=fill)


def _draw_icon_box(
    draw: ImageDraw.ImageDraw,
    left: int,
    top: int,
    bitmap: tuple[str, ...],
    *,
    outline: tuple[int, int, int] = INK,
    fill: tuple[int, int, int] = (242, 246, 249),
    width: int = 3,
) -> None:
    draw.rounded_rectangle(
        (left, top, left + ICON_SIZE - 1, top + ICON_SIZE - 1),
        radius=13,
        fill=fill,
        outline=outline,
        width=width,
    )
    _draw_bitmap(draw, left + 14, top + 14, bitmap)


def _draw_arrow(draw: ImageDraw.ImageDraw, x0: int, y: int, x1: int) -> None:
    draw.line((x0, y, x1 - 13, y), fill=MUTED, width=4)
    draw.polygon(((x1 - 13, y - 9), (x1, y), (x1 - 13, y + 9)), fill=MUTED)


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    box: tuple[int, int, int, int],
    title: str,
    subtitle: str,
) -> None:
    x, y, w, h = box
    draw.rounded_rectangle((x, y, x + w, y + h), radius=18, fill=PANEL, outline=LINE, width=3)
    draw.text((x + 22, y + 18), title, font=_font(25, True), fill=INK)
    draw.text((x + 22, y + 54), subtitle, font=_font(15), fill=MUTED)


def _valid_pair_rows(rows: object) -> bool:
    if not isinstance(rows, list) or len(rows) != 5:
        return False
    for row in rows:
        if not isinstance(row, list) or len(row) != 2:
            return False
        if any(not isinstance(value, int) or isinstance(value, bool) for value in row):
            return False
    return True


def _legal_scene_values(query_badge: object, key_one: object, key_two: object) -> bool:
    if not isinstance(query_badge, int) or isinstance(query_badge, bool) or not 0 <= query_badge < 5:
        return False
    if not _valid_pair_rows(key_one) or not _valid_pair_rows(key_two):
        return False
    if sorted(row[0] for row in key_one) != list(range(5)):
        return False
    if sorted(row[1] for row in key_one) != list(range(5)):
        return False
    if sorted(row[0] for row in key_two) != list(range(5)):
        return False
    if sorted(row[1] for row in key_two) != list(range(5)):
        return False
    return True


def sample_scene(seed: int) -> dict:
    """Sample two independent visible permutations and one visible query."""

    rng = random.Random(int(seed))
    query_badge = rng.randrange(5)
    badge_order = list(range(5))
    rng.shuffle(badge_order)
    badge_to_sigil = list(range(5))
    rng.shuffle(badge_to_sigil)
    key_one = [[badge, badge_to_sigil[badge]] for badge in badge_order]

    sigil_order = list(range(5))
    rng.shuffle(sigil_order)
    sigil_to_station = list(range(5))
    rng.shuffle(sigil_to_station)
    key_two = [[sigil, sigil_to_station[sigil]] for sigil in sigil_order]
    return {
        "query_badge": query_badge,
        "key_one": key_one,
        "key_two": key_two,
    }


def analytic_gold(scene: dict) -> str:
    """Follow badge -> sigil -> station using only the declared scene fields."""

    query_badge = scene["query_badge"]
    key_one = scene["key_one"]
    key_two = scene["key_two"]
    if not _legal_scene_values(query_badge, key_one, key_two):
        raise ValueError("scene is not a pair of five-way bijective keys")
    badge_to_sigil = {badge: sigil for badge, sigil in key_one}
    sigil_to_station = {sigil: station for sigil, station in key_two}
    return STATION_NAMES[sigil_to_station[badge_to_sigil[query_badge]]]


def margin(scene: dict) -> float:
    query_badge = scene["query_badge"]
    key_one = scene["key_one"]
    key_two = scene["key_two"]
    return 1.0 if _legal_scene_values(query_badge, key_one, key_two) else 0.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    """Render a single board with both key rows in their sampled order."""

    query_badge = scene["query_badge"]
    key_one = scene["key_one"]
    key_two = scene["key_two"]
    if not _legal_scene_values(query_badge, key_one, key_two):
        raise ValueError("cannot render an invalid double-key scene")

    image = Image.new("RGB", (WIDTH, HEIGHT), PAPER)
    draw = ImageDraw.Draw(image)
    draw.text((34, 25), "DOUBLE-KEY RELAY", font=_font(31, True), fill=INK)
    draw.text(
        (34, 66),
        "Carry one visible identity through two shuffled keys.",
        font=_font(16),
        fill=MUTED,
    )

    _draw_panel(draw, (30, PANEL_Y, 290, PANEL_H), "QUERY", "the gold outline marks the badge")
    _draw_panel(draw, (PANEL_A_X, PANEL_Y, PANEL_A_W, PANEL_H), "KEY A", "badge  →  sigil")
    _draw_panel(draw, (PANEL_B_X, PANEL_Y, PANEL_B_W, PANEL_H), "KEY B", "sigil  →  station")

    draw.rounded_rectangle((112, 228, 248, 374), radius=18, fill=(255, 252, 242), outline=GOLD, width=5)
    _draw_icon_box(
        draw,
        QUERY_ICON_X,
        QUERY_ICON_Y,
        BADGE_BITMAPS[query_badge],
        outline=GOLD,
        fill=(255, 255, 255),
        width=4,
    )
    draw.text((78, 421), "use this badge", font=_font(19, True), fill=INK)
    draw.text((75, 452), "as the query", font=_font(19, True), fill=INK)
    draw.line((74, 507, 274, 507), fill=LINE, width=2)
    draw.text((73, 535), "Read its matching", font=_font(16), fill=MUTED)
    draw.text((73, 560), "sigil in KEY A.", font=_font(16), fill=MUTED)
    draw.text((73, 608), "Then read that sigil", font=_font(16), fill=MUTED)
    draw.text((73, 633), "again in KEY B.", font=_font(16), fill=MUTED)

    for row_index, (badge, sigil) in enumerate(key_one):
        top = ROW_TOP + row_index * ROW_H
        draw.rounded_rectangle(
            (PANEL_A_X + 17, top - 8, PANEL_A_X + PANEL_A_W - 17, top + 101),
            radius=13,
            fill=ROW,
            outline=(225, 231, 236),
            width=2,
        )
        _draw_icon_box(draw, A_BADGE_X, top, BADGE_BITMAPS[badge])
        _draw_arrow(draw, A_BADGE_X + ICON_SIZE + 14, top + 49, A_SIGIL_X - 14)
        _draw_icon_box(draw, A_SIGIL_X, top, SIGIL_BITMAPS[sigil], fill=(245, 243, 250))

    for row_index, (sigil, station) in enumerate(key_two):
        top = ROW_TOP + row_index * ROW_H
        draw.rounded_rectangle(
            (PANEL_B_X + 17, top - 8, PANEL_B_X + PANEL_B_W - 17, top + 101),
            radius=13,
            fill=ROW,
            outline=(225, 231, 236),
            width=2,
        )
        _draw_icon_box(draw, B_SIGIL_X, top, SIGIL_BITMAPS[sigil], fill=(245, 243, 250))
        _draw_arrow(draw, B_SIGIL_X + ICON_SIZE + 14, top + 49, B_SWATCH_X - 14)
        draw.rounded_rectangle(
            (B_SWATCH_X, top, B_SWATCH_X + ICON_SIZE - 1, top + ICON_SIZE - 1),
            radius=13,
            fill=STATION_COLORS[station],
            outline=INK,
            width=3,
        )
        draw.text(
            (B_SWATCH_X + ICON_SIZE + 16, top + 37),
            STATION_NAMES[station].upper(),
            font=_font(16, True),
            fill=INK,
        )

    draw.text((PANEL_A_X + 22, 785), "Every row is a shuffled pair.", font=_font(14), fill=MUTED)
    draw.text((PANEL_B_X + 22, 785), "The station is the final answer.", font=_font(14), fill=MUTED)
    return image
