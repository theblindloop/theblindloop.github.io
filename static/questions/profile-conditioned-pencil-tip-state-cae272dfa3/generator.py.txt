"""Deterministic raster world for the visible pencil-tip state question."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 512
HEIGHT = 512
SUPERSAMPLE = 4

BACKGROUND_PALETTE = (
    (248, 250, 247),
    (255, 252, 244),
    (245, 249, 252),
    (250, 247, 251),
)
BODY_PALETTE = (
    (38, 126, 205),
    (224, 83, 70),
    (44, 166, 111),
    (229, 159, 42),
    (135, 83, 194),
    (24, 155, 166),
)
WOOD_PALETTE = (
    (206, 147, 72),
    (224, 169, 89),
    (190, 126, 62),
    (232, 184, 110),
)
FERRULE_PALETTE = ((183, 190, 194), (211, 216, 213), (153, 171, 178))
ERASER_PALETTE = ((224, 103, 116), (214, 128, 143), (198, 91, 108))


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one state-balanced scene with independent visual nuisances."""

    rng = random.Random(int(seed))
    tip_state = rng.choice(("pointed", "flat"))
    tip_side = rng.choice(("left", "right"))
    near_boundary = rng.random() < 0.22
    body_length = rng.uniform(270.0, 330.0)
    body_height = rng.uniform(72.0, 100.0)
    tip_depth = rng.uniform(32.0, 39.5) if near_boundary else rng.uniform(52.0, 72.0)
    total_length = 52.0 + body_length + tip_depth
    center_x = rng.uniform(total_length / 2.0 + 18.0, WIDTH - total_length / 2.0 - 18.0)
    center_y = rng.uniform(body_height / 2.0 + 48.0, HEIGHT - body_height / 2.0 - 62.0)
    return {
        "tip_state": tip_state,
        "tip_side": tip_side,
        "center_x": round(center_x, 3),
        "center_y": round(center_y, 3),
        "body_length": round(body_length, 3),
        "body_height": round(body_height, 3),
        "tip_depth": round(tip_depth, 3),
        "background": list(rng.choice(BACKGROUND_PALETTE)),
        "body_color": list(rng.choice(BODY_PALETTE)),
        "wood_color": list(rng.choice(WOOD_PALETTE)),
        "ferrule_color": list(rng.choice(FERRULE_PALETTE)),
        "eraser_color": list(rng.choice(ERASER_PALETTE)),
        "outline_color": [35, 44, 49],
        "label_color": [92, 101, 105],
        "label_text": "PENCIL",
        "antialias": True,
    }


def _font(size: int):
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _box(box: tuple[float, float, float, float], scale: int) -> tuple[float, ...]:
    return tuple(value * scale for value in box)


def _canonical_geometry(scene: dict[str, Any]) -> tuple[float, float, float, float, float, float]:
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    body_length = float(scene["body_length"])
    body_height = float(scene["body_height"])
    tip_depth = float(scene["tip_depth"])
    total = 52.0 + body_length + tip_depth
    x0 = cx - total / 2.0
    eraser_end = x0 + 24.0
    ferrule_end = x0 + 52.0
    body_end = ferrule_end + body_length
    tip_x = body_end + tip_depth
    top = cy - body_height / 2.0
    bottom = cy + body_height / 2.0
    return x0, eraser_end, ferrule_end, body_end, tip_x, top, bottom


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the pencil; a mirror only changes the visible pose, not the state."""

    ss = SUPERSAMPLE if bool(scene["antialias"]) else 1
    image = Image.new("RGB", (WIDTH * ss, HEIGHT * ss), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    x0, eraser_end, ferrule_end, body_end, tip_x, top, bottom = _canonical_geometry(scene)
    tip_state = str(scene["tip_state"])
    mirror = str(scene["tip_side"]) == "left"
    outline = tuple(scene["outline_color"])

    def tx(x: float) -> float:
        return WIDTH - x if mirror else x

    def rect(box: tuple[float, float, float, float], **kwargs: Any) -> None:
        left, right = sorted((tx(box[0]), tx(box[2])))
        draw.rectangle(_box((left, box[1], right, box[3]), ss), **kwargs)

    def rounded(box: tuple[float, float, float, float], radius: float, **kwargs: Any) -> None:
        left, right = sorted((tx(box[0]), tx(box[2])))
        draw.rounded_rectangle(_box((left, box[1], right, box[3]), ss), radius=radius * ss, **kwargs)

    def polygon(points: list[tuple[float, float]], **kwargs: Any) -> None:
        draw.polygon([(tx(x) * ss, y * ss) for x, y in points], **kwargs)

    # The label is intentionally separate from the large object component.
    draw.text(
        (float(scene["center_x"]) * ss, (bottom + 36.0) * ss),
        str(scene["label_text"]),
        fill=tuple(scene["label_color"]),
        anchor="mm",
        font=_font(15 * ss),
    )

    # Tail, ferrule, body, and wood overlap by a pixel so the whole pencil is
    # one visible object even after antialiasing and mirroring.
    rounded(
        (x0, top + 2.0, eraser_end + 2.0, bottom - 2.0),
        10.0,
        fill=tuple(scene["eraser_color"]),
        outline=outline,
        width=4 * ss,
    )
    rect(
        (eraser_end - 2.0, top, ferrule_end + 2.0, bottom),
        fill=tuple(scene["ferrule_color"]),
        outline=outline,
        width=4 * ss,
    )
    # Two shallow ferrule bands are visual detail, not answer-bearing marks.
    for band_x in (eraser_end + 6.0, eraser_end + 14.0):
        rect((band_x, top + 3.0, band_x + 3.0, bottom - 3.0), fill=(112, 128, 134))
    rect(
        (ferrule_end - 1.0, top, body_end + 1.0, bottom),
        fill=tuple(scene["body_color"]),
        outline=outline,
        width=4 * ss,
    )

    if tip_state == "pointed":
        wood = [(body_end - 1.0, top), (body_end - 1.0, bottom), (tip_x, (top + bottom) / 2.0)]
    else:
        wood = [
            (body_end - 1.0, top),
            (tip_x, top),
            (tip_x, bottom),
            (body_end - 1.0, bottom),
        ]
    polygon(wood, fill=tuple(scene["wood_color"]), outline=outline, width=4 * ss)

    # A short graphite line makes the apex/face cue explicit without adding a
    # second disconnected component.
    graphite = (48, 52, 49)
    if tip_state == "pointed":
        polygon(
            [(body_end + 2.0, top + (bottom - top) * 0.39),
             (tip_x - 1.0, (top + bottom) / 2.0),
             (body_end + 2.0, top + (bottom - top) * 0.61)],
            fill=graphite,
        )
    else:
        rect((tip_x - 5.0, top + 4.0, tip_x - 1.0, bottom - 4.0), fill=graphite)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS) if ss > 1 else image


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(scene["tip_state"])


def margin(scene: dict[str, Any]) -> float:
    """One-sided terminal-depth clearance in pixels before the shallow-tip boundary."""

    return float(scene["tip_depth"]) - 24.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    return float(scene["tip_depth"]) < 42.0 or float(scene["body_height"]) < 64.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No latent aliases are declared; mirror changes pixels and is not an alias."""

    return []
