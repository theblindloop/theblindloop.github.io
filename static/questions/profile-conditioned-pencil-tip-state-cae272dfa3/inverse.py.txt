"""Independent image-only inverse for the pencil terminal silhouette."""

from __future__ import annotations

import numpy as np


def decision_from_image(image) -> str:
    """Classify the visible wooden end using only foreground pixel geometry."""

    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    background = arr[0, 0]
    difference = np.max(np.abs(arr - background), axis=2)
    mask = difference >= 12
    # The pencil is the only broad horizontal foreground object.  Selecting
    # its row band by raster occupancy is independent of renderer state and
    # avoids relying on any component labels or metadata.
    row_counts = np.count_nonzero(mask, axis=1)
    broad_rows = np.flatnonzero(row_counts >= max(30, int(row_counts.max() * 0.18)))
    if len(broad_rows) < 30:
        raise ValueError("no broad pencil row band found")
    y0, y1 = int(broad_rows.min()), int(broad_rows.max())
    component = mask[y0 : y1 + 1, :]
    ys, xs = np.where(component)
    if len(xs) < 5000:
        raise ValueError("foreground component is too small")
    min_x, max_x = int(xs.min()), int(xs.max())
    min_y, max_y = int(ys.min()), int(ys.max())
    width = max_x - min_x + 1
    height = max_y - min_y + 1
    if width < 240 or height < 45 or width / float(height) < 3.0:
        raise ValueError("pencil silhouette is too small or malformed")

    spans = np.zeros(width, dtype=np.int32)
    for offset, x in enumerate(range(min_x, max_x + 1)):
        column = np.flatnonzero(component[:, x])
        if len(column):
            spans[offset] = int(column[-1] - column[0] + 1)
    body_lo = max(0, int(width * 0.22))
    body_hi = min(width, int(width * 0.72))
    body_span = float(np.median(spans[body_lo:body_hi]))
    if body_span < 35:
        raise ValueError("body profile is not resolvable")
    window = max(8, min(14, width // 18))
    left_end = float(np.max(spans[:window]))
    right_end = float(np.max(spans[-window:]))
    terminal_ratio = min(left_end, right_end) / body_span
    if terminal_ratio < 0.68:
        return "pointed"
    if terminal_ratio > 0.78:
        return "flat"
    raise ValueError(f"ambiguous terminal profile ratio {terminal_ratio:.3f}")
