"""Pixel-only inverse arm for the plate-cut area balance world.

Input is one rendered PNG and the public question contract: a grey plate is cut
by one dark curve with exactly one height per horizontal position; report the
side of the curve that holds more grey plate area. Nothing here reads the latent
scene, the renderer, the prompts, the generator or the analytic gold.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "plate_cut_balance-oracle-0.1.0"

ABSTAIN_FRACTION = 0.012


def _masks(image: Image.Image) -> tuple[np.ndarray, np.ndarray]:
    """Segment the neutral plate fill and the dark cutting curve."""
    rgb = np.asarray(image.convert("RGB")).astype(np.int16)
    hi = rgb.max(axis=2)
    lo = rgb.min(axis=2)
    chroma = hi - lo
    value = rgb.mean(axis=2)
    plate = (chroma <= 26) & (value >= 120.0) & (value <= 220.0)
    curve = (chroma <= 40) & (value <= 95.0)
    return plate, curve


def _curve_band(curve: np.ndarray) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Per-column first and last curve row, plus a column validity flag."""
    height, width = curve.shape
    rows = np.arange(height, dtype=np.int32)[:, None]
    present = curve.any(axis=0)
    big = np.iinfo(np.int32).max
    top = np.where(curve, rows, big).min(axis=0).astype(np.float64)
    bottom = np.where(curve, rows, -1).max(axis=0).astype(np.float64)
    columns = np.arange(width)
    if present.any():
        top = np.interp(columns, columns[present], top[present])
        bottom = np.interp(columns, columns[present], bottom[present])
    return top, bottom, present


def side_areas_from_image(image: Image.Image) -> tuple[float, float]:
    """Grey pixel counts strictly above and strictly below the curve band."""
    plate, curve = _masks(image)
    plate = plate & ~curve
    top, bottom, present = _curve_band(curve)
    if not present.any():
        return 0.0, 0.0
    height, width = plate.shape
    rows = np.arange(height, dtype=np.float64)[:, None]
    above = plate & (rows < top[None, :])
    below = plate & (rows > bottom[None, :])
    return float(above.sum()), float(below.sum())


def decision_from_image(image: Image.Image) -> str:
    """`above`, `below`, or `abstain` when the two sides are indistinguishable."""
    above, below = side_areas_from_image(image)
    total = above + below
    if total < 400.0:
        return "abstain"
    if abs(above - below) / total < ABSTAIN_FRACTION:
        return "abstain"
    return "above" if above > below else "below"
