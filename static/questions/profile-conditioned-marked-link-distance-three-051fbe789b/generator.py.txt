"""Renderer and analytic arm for the three-link shortest-route world."""

from __future__ import annotations

import random
from collections import deque
from copy import deepcopy

from PIL import Image, ImageDraw


ROWS = 3
COLS = 4
NODE_COUNT = ROWS * COLS
EDGE_COUNT = 12


def _potential_edges() -> list[tuple[int, int]]:
    edges: list[tuple[int, int]] = []
    for row in range(ROWS):
        for col in range(COLS):
            here = row * COLS + col
            if col + 1 < COLS:
                edges.append((here, here + 1))
            if row + 1 < ROWS:
                edges.append((here, here + COLS))
    return edges


def _distance_from_parts(edges: list[list[int]] | list[tuple[int, int]], start: int, target: int) -> int:
    adjacency = [set() for _ in range(NODE_COUNT)]
    for left, right in edges:
        adjacency[left].add(right)
        adjacency[right].add(left)
    queue = deque([(start, 0)])
    seen = {start}
    while queue:
        node, distance = queue.popleft()
        if node == target:
            return distance
        for neighbor in sorted(adjacency[node]):
            if neighbor not in seen:
                seen.add(neighbor)
                queue.append((neighbor, distance + 1))
    return 99


def sample_scene(seed: int) -> dict:
    """Sample a connected sparse grid graph with counterbalanced marked distance.

    Both classes use exactly twelve junctions, twelve links, and terminal pairs
    whose Manhattan separation in the layout is three.  The negative class is
    created only by the global link arrangement, never by marker geometry.
    """

    rng = random.Random(int(seed))
    want_yes = int(seed) % 2 == 0
    possible = _potential_edges()
    pairs = [
        (a, b)
        for a in range(NODE_COUNT)
        for b in range(a + 1, NODE_COUNT)
        if abs(a // COLS - b // COLS) + abs(a % COLS - b % COLS) == 3
    ]

    selected_edges: list[tuple[int, int]] | None = None
    selected_pair: tuple[int, int] | None = None
    for _ in range(20_000):
        edges = sorted(rng.sample(possible, EDGE_COUNT))
        if _distance_from_parts(edges, 0, NODE_COUNT - 1) == 99:
            continue
        choices = []
        for pair in pairs:
            distance = _distance_from_parts(edges, pair[0], pair[1])
            if (want_yes and distance == 3) or (not want_yes and distance in (4, 5, 6)):
                choices.append(pair)
        if choices:
            selected_edges = edges
            selected_pair = rng.choice(choices)
            break
    if selected_edges is None or selected_pair is None:
        raise RuntimeError(f"could not sample a balanced graph for seed {seed}")

    start, target = selected_pair
    if rng.random() < 0.5:
        start, target = target, start

    nodes = []
    for row in range(ROWS):
        for col in range(COLS):
            nodes.append([68 + 114 * col + rng.randint(-7, 7), 70 + 124 * row + rng.randint(-7, 7)])

    return {
        "nodes": nodes,
        "edges": [[a, b] for a, b in selected_edges],
        "start": start,
        "target": target,
        "node_colors": {
            "ordinary": "#38BDF8",
            "start": "#22C55E",
            "target": "#A855F7",
            "link": "#263244",
            "background": "#F8FAFC",
        },
        "offset": [rng.randint(-4, 4), rng.randint(-4, 4)],
        "canvas": [480, 392],
    }


def _shortest_distance(scene: dict) -> int:
    return _distance_from_parts(scene["edges"], scene["start"], scene["target"])


def analytic_gold(scene: dict) -> str:
    return "yes" if _shortest_distance(scene) == 3 else "no"


def margin(scene: dict) -> float:
    """Discrete runner-up margin, measured in one complete link edit."""

    _shortest_distance(scene)
    return 1.0


def is_quarantined(scene: dict) -> bool:
    _shortest_distance(scene)
    return False


def render(scene: dict) -> Image.Image:
    width, height = scene["canvas"]
    colors = scene["node_colors"]
    dx, dy = scene["offset"]
    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), colors["background"])
    draw = ImageDraw.Draw(image)

    points = [((x + dx) * scale, (y + dy) * scale) for x, y in scene["nodes"]]
    for left, right in scene["edges"]:
        draw.line([points[left], points[right]], fill=colors["link"], width=12 * scale)

    radius = 17 * scale
    for index, (x, y) in enumerate(points):
        fill = colors["ordinary"]
        if index == scene["start"]:
            fill = colors["start"]
        elif index == scene["target"]:
            fill = colors["target"]
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=fill)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["edges"] = list(reversed(twin["edges"]))
    return [("reverse_edge_enumeration", twin)]
