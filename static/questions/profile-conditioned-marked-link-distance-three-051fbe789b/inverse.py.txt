"""Independent pixel-only inverse arm for a visible junction-link graph."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


_PALETTE = np.array(
    [
        (248, 250, 252),  # background
        (38, 50, 68),     # dark links
        (56, 189, 248),    # ordinary junctions
        (34, 197, 94),     # green terminal
        (168, 85, 247),    # purple terminal
    ],
    dtype=np.int32,
)


def _components(mask: np.ndarray, *, eight: bool) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    neighbors = [(-1, 0), (1, 0), (0, -1), (0, 1)]
    if eight:
        neighbors += [(-1, -1), (-1, 1), (1, -1), (1, 1)]
    result: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = deque([(int(y), int(x))])
        pixels: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            pixels.append((cy, cx))
            for oy, ox in neighbors:
                ny, nx = cy + oy, cx + ox
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        result.append(np.asarray(pixels, dtype=np.int32))
    return result


def _recover_graph(image: Image.Image) -> tuple[list[set[int]], int, int]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    distances = ((arr[:, :, None, :] - _PALETTE[None, None, :, :]) ** 2).sum(axis=3)
    labels = distances.argmin(axis=2)

    node_components = [part for part in _components(labels >= 2, eight=True) if len(part) >= 450]
    if len(node_components) != 12:
        raise ValueError(f"expected 12 visible junctions, recovered {len(node_components)}")

    centers: list[tuple[float, float]] = []
    kinds: list[int] = []
    for part in node_components:
        centers.append((float(part[:, 1].mean()), float(part[:, 0].mean())))
        values = labels[part[:, 0], part[:, 1]]
        counts = np.bincount(values, minlength=5)
        kinds.append(int(np.argmax(counts[2:]) + 2))
    if kinds.count(3) != 1 or kinds.count(4) != 1:
        raise ValueError("green and purple terminal junctions are not uniquely visible")
    start = kinds.index(3)
    target = kinds.index(4)

    link_components = [part for part in _components(labels == 1, eight=True) if len(part) >= 250]
    if len(link_components) != 12:
        raise ValueError(f"expected 12 complete dark links, recovered {len(link_components)}")

    adjacency = [set() for _ in centers]
    for part in link_components:
        xs = part[:, 1].astype(np.float64)
        ys = part[:, 0].astype(np.float64)
        endpoint_scores = []
        for index, (cx, cy) in enumerate(centers):
            minimum = float(np.min((xs - cx) ** 2 + (ys - cy) ** 2))
            endpoint_scores.append((minimum, index))
        endpoint_scores.sort()
        if endpoint_scores[1][0] > 26.0**2:
            raise ValueError("a dark link does not visibly terminate at two junctions")
        left, right = endpoint_scores[0][1], endpoint_scores[1][1]
        if left == right or right in adjacency[left]:
            raise ValueError("ambiguous or duplicate dark link recovered")
        adjacency[left].add(right)
        adjacency[right].add(left)
    return adjacency, start, target


def decision_from_image(image: Image.Image) -> str:
    adjacency, start, target = _recover_graph(image)
    queue = deque([(start, 0)])
    seen = {start}
    while queue:
        node, distance = queue.popleft()
        if node == target:
            return "yes" if distance == 3 else "no"
        for neighbor in sorted(adjacency[node]):
            if neighbor not in seen:
                seen.add(neighbor)
                queue.append((neighbor, distance + 1))
    raise ValueError("the marked junctions are visibly disconnected")
