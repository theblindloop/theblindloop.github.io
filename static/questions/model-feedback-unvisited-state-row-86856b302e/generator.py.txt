"""Unvisited-state-row world.

Latent scene z = {start, rows, answer_index, disc, paint_order}.

Six discs of six flat colours sit in six places numbered 1..6 from the left.
The row above the grey rule is the starting configuration of a deterministic
machine whose step is state-dependent:

    step(s):  swap the red disc with the disc in the next place to its right,
              wrapping past place 6 back to place 1;
              then move the disc in place 6 to place 1 and slide the other five
              discs one place to the right.

Running the step twelve times gives thirteen configurations in all (the start
plus twelve more).  Four candidate rows are drawn below the rule.  Exactly
three of them occur among those thirteen configurations and exactly one never
does; the decision is the 1-based top-to-bottom index of the row that never
occurs.

The answer is therefore a property of the whole orbit, not of any endpoint: the
three occurring rows are forced to occur only at late steps (7..12), so no
prefix of the simulation can certify the negative, and every candidate row is
forced to have the same permutation parity as the start, so the machine's
even-permutation invariant cannot separate them either.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "unvisited_state_row-0.1.0"

N_PLACES = 6
RED = 0
N_STEPS = 12
N_ROWS = 4
EARLY_STEPS = 6
LATE_MIN = 7

# palette: index 0 is the red disc named by the rule
COLORS = (
    (214, 39, 40),    # red
    (31, 119, 180),   # blue
    (44, 160, 44),    # green
    (255, 140, 0),    # orange
    (148, 60, 190),   # purple
    (23, 190, 190),   # teal
)
RULE_COLOR = (170, 172, 178)
BG = (255, 255, 255)

WIDTH = 1000
HEIGHT = 980
PLACE_X0 = 125
PLACE_PITCH = 150

START_Y = 120
RULE_Y = 232
RULE_H = 6
ROW_Y0 = 340
ROW_PITCH = 180

DISC_MIN = 104
DISC_MAX = 142
QUARANTINE_MARGIN = 16.0


# --------------------------------------------------------------------------
# dynamics
# --------------------------------------------------------------------------
def step(state: list[int]) -> list[int]:
    """One machine step: content-addressed swap of red, then a fixed shift."""
    s = list(state)
    p = s.index(RED)
    q = (p + 1) % N_PLACES
    s[p], s[q] = s[q], s[p]
    return [s[-1]] + s[:-1]


def trajectory(start: list[int], steps: int = N_STEPS) -> list[list[int]]:
    out = [list(start)]
    for _ in range(steps):
        out.append(step(out[-1]))
    return out


def _parity(perm: list[int]) -> int:
    seen = [False] * len(perm)
    par = 0
    for i in range(len(perm)):
        if seen[i]:
            continue
        j, size = i, 0
        while not seen[j]:
            seen[j] = True
            j = perm[j]
            size += 1
        par ^= (size - 1) & 1
    return par


def _hamming(a: list[int], b: list[int]) -> int:
    return sum(1 for x, y in zip(a, b) if x != y)


def _rotate3(state: list[int], places: tuple[int, int, int]) -> list[int]:
    """Apply a 3-cycle to three places: an even permutation, so parity holds."""
    i, j, k = places
    s = list(state)
    s[i], s[j], s[k] = state[k], state[i], state[j]
    return s


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------
def sample_scene(seed: int) -> dict:
    """Deterministically sample a scene; retries until every guard holds."""
    for attempt in range(4000):
        rng = random.Random((seed * 1000003 + attempt * 7919 + 11) & 0xFFFFFFFF)
        start = list(range(N_PLACES))
        rng.shuffle(start)
        traj = trajectory(start)
        keys = [tuple(s) for s in traj]
        if len(set(keys)) != len(keys):
            continue
        visited = set(keys)
        red_places = {s.index(RED) for s in traj}

        late = list(range(LATE_MIN, N_STEPS + 1))
        rng.shuffle(late)
        shown = [traj[t] for t in sorted(late[:3])]
        if any(tuple(r) in {tuple(x) for x in traj[: EARLY_STEPS + 1]} for r in shown):
            continue

        # the never-occurring row: a 3-cycle off some occurring row, so it has
        # the same parity as the start and differs from it in only 3 places.
        base = traj[rng.randrange(LATE_MIN, N_STEPS + 1)]
        triples = [
            (i, j, k)
            for i in range(N_PLACES)
            for j in range(i + 1, N_PLACES)
            for k in range(j + 1, N_PLACES)
        ]
        rng.shuffle(triples)
        missing = None
        for triple in triples:
            cand = _rotate3(base, triple)
            if tuple(cand) in visited:
                continue
            if cand.index(RED) not in red_places:
                continue
            if any(tuple(cand) == tuple(r) for r in shown):
                continue
            missing = cand
            break
        if missing is None:
            continue
        if _parity(missing) != _parity(start):
            continue

        rows = list(shown) + [missing]
        # the never-occurring row must not be the unique nearest or unique
        # farthest row from the drawn start row
        dists = [_hamming(r, start) for r in rows]
        if dists[3] == min(dists) and dists.count(min(dists)) == 1:
            continue
        if dists[3] == max(dists) and dists.count(max(dists)) == 1:
            continue

        order = list(range(N_ROWS))
        rng.shuffle(order)
        shuffled = [rows[i] for i in order]
        answer_index = order.index(3) + 1

        disc = rng.randrange(DISC_MIN, DISC_MAX + 1)
        paint_order = list(range(N_ROWS))
        rng.shuffle(paint_order)
        return {
            "start": start,
            "rows": shuffled,
            "answer_index": answer_index,
            "disc": disc,
            "paint_order": paint_order,
        }
    raise RuntimeError(f"sampler exhausted for seed {seed}")


def analytic_gold(scene: dict) -> str:
    """'1'..'4': the top-to-bottom index of the row the machine never reaches."""
    visited = {tuple(s) for s in trajectory(list(scene["start"]))}
    missing = [i + 1 for i, row in enumerate(scene["rows"]) if tuple(row) not in visited]
    if len(missing) != 1:
        raise ValueError("scene does not have exactly one never-occurring row")
    return str(missing[0])


def margin(scene: dict) -> float:
    """Clear background gap in px between discs in neighbouring places."""
    return float(PLACE_PITCH - int(scene["disc"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Row paint order is sampled but photometrically invisible."""
    order = list(scene["paint_order"])
    out: list[tuple[str, dict]] = []
    if len(order) > 1:
        rotated = dict(scene)
        rotated["paint_order"] = order[1:] + order[:1]
        out.append(("rotated_paint_order", rotated))
        reversed_ = dict(scene)
        reversed_["paint_order"] = order[::-1]
        out.append(("reversed_paint_order", reversed_))
    return out


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def _disc(draw: ImageDraw.ImageDraw, cx: int, cy: int, d: int, color) -> None:
    r = d // 2
    draw.ellipse([cx - r, cy - r, cx - r + d - 1, cy - r + d - 1], fill=color)


def _row(draw: ImageDraw.ImageDraw, arrangement, cy: int, d: int) -> None:
    for place, hue in enumerate(arrangement):
        _disc(draw, PLACE_X0 + place * PLACE_PITCH, cy, d, COLORS[hue])


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    d = int(scene["disc"])

    _row(draw, list(scene["start"]), START_Y, d)
    draw.rectangle([60, RULE_Y, WIDTH - 61, RULE_Y + RULE_H - 1], fill=RULE_COLOR)

    for idx in scene["paint_order"]:
        cy = ROW_Y0 + idx * ROW_PITCH
        _row(draw, list(scene["rows"][idx]), cy, d)
    return image


# --------------------------------------------------------------------------
# shortcut audit helpers (used by the tests)
# --------------------------------------------------------------------------
def _variant_step(state: list[int], mode: str) -> list[int]:
    s = list(state)
    p = s.index(RED)
    if mode == "left":
        q = (p - 1) % N_PLACES
    else:
        q = (p + 1) % N_PLACES
    s[p], s[q] = s[q], s[p]
    if mode == "noshift":
        return s
    if mode == "shiftleft":
        return s[1:] + [s[0]]
    return [s[-1]] + s[:-1]


def _variant(scene: dict, mode: str) -> str:
    """Answer produced by a mis-read machine: first row it never reaches."""
    steps = EARLY_STEPS if mode == "short" else N_STEPS
    state = list(scene["start"])
    seen = {tuple(state)}
    for _ in range(steps):
        state = _variant_step(state, mode)
        seen.add(tuple(state))
    missing = [i + 1 for i, row in enumerate(scene["rows"]) if tuple(row) not in seen]
    return str(missing[0]) if len(missing) == 1 else "0"


def _unvisited_count_after(scene: dict, steps: int) -> int:
    traj = trajectory(list(scene["start"]), steps)
    seen = {tuple(s) for s in traj}
    return sum(1 for row in scene["rows"] if tuple(row) not in seen)
