"""Independent pixel-only inverse arm for the unvisited-state-row world.

From the final PNG alone it

1. masks the six flat disc hues by exact colour;
2. splits the union of those masks into horizontal bands (each drawn row of six
   discs occupies its own band, so the y-projection has one clean run per row);
3. reads the topmost band as the starting configuration and each later band as
   one numbered candidate row: within a band, each hue's mean x fixes its place
   and sorting the six hues by x recovers the arrangement;
4. runs the machine twelve times from the recovered starting configuration,
   collecting the thirteen configurations it passes through;
5. reports the 1-based top-to-bottom index of the candidate row that never
   occurs among them.

It imports no renderer / prompts / generate / verify / analytic-gold code and
never receives the latent scene.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_PALETTE = (
    (214, 39, 40),
    (31, 119, 180),
    (44, 160, 44),
    (255, 140, 0),
    (148, 60, 190),
    (23, 190, 190),
)
_RED = 0
_N_PLACES = 6
_N_ROWS = 4
_N_STEPS = 12


def _near(arr: np.ndarray, color, tol: int) -> np.ndarray:
    diff = np.abs(arr.astype(int) - np.array(color, dtype=int))
    return (diff <= tol).all(axis=-1)


def _bands(flags: np.ndarray, min_len: int = 12) -> list[tuple[int, int]]:
    out: list[tuple[int, int]] = []
    start = None
    for i, f in enumerate(flags):
        if f and start is None:
            start = i
        elif not f and start is not None:
            if i - start >= min_len:
                out.append((start, i - 1))
            start = None
    if start is not None and len(flags) - start >= min_len:
        out.append((start, len(flags) - 1))
    return out


def _step(state: list[int]) -> list[int]:
    s = list(state)
    p = s.index(_RED)
    q = (p + 1) % _N_PLACES
    s[p], s[q] = s[q], s[p]
    return [s[-1]] + s[:-1]


def _read_row(hue_masks, lo: int, hi: int):
    present = []
    for idx, mask in enumerate(hue_masks):
        sub = mask[lo : hi + 1]
        _, xs = np.nonzero(sub)
        if xs.size:
            present.append((float(xs.mean()), idx))
    if len(present) != _N_PLACES:
        return None
    return [idx for _, idx in sorted(present)]


def decision_from_image(image: Image.Image) -> str:
    """Return '1'..'4': the numbered row the machine never reaches."""
    arr = np.asarray(image.convert("RGB"))
    hue_masks = [_near(arr, c, 30) for c in _PALETTE]

    ink = np.zeros(arr.shape[:2], dtype=bool)
    for mask in hue_masks:
        ink |= mask
    bands = _bands(ink.any(axis=1))
    if len(bands) != _N_ROWS + 1:
        return "1"

    rows = []
    for lo, hi in bands:
        arrangement = _read_row(hue_masks, lo, hi)
        if arrangement is None:
            return "1"
        rows.append(arrangement)

    state = rows[0]
    visited = {tuple(state)}
    for _ in range(_N_STEPS):
        state = _step(state)
        visited.add(tuple(state))

    missing = [i for i, row in enumerate(rows[1:], start=1) if tuple(row) not in visited]
    if len(missing) != 1:
        return "1"
    return str(missing[0])
