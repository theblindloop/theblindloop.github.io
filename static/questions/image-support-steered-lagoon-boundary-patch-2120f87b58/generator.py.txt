"""Deterministic scene sampler, renderer, and analytic topology arm."""

from __future__ import annotations

import random
from collections import deque

from PIL import Image, ImageDraw

CANVAS = 512
N = 12
TEAL_PALETTES = ("#178C86", "#247FA3", "#398B6B", "#536FA8")
YELLOW = "#F2B705"


def _rotate(grid: list[list[int]]) -> list[list[int]]:
    return [list(row) for row in zip(*grid[::-1])]


def _build_grid(answer_yes: bool, negative_outer: bool, rotation: int) -> list[list[int]]:
    # 0 = lagoon/background, 1 = teal island, 2 = yellow patch.
    grid = [[1 for _ in range(N)] for _ in range(N)]
    for y in range(4, 8):
        for x in range(4, 8):
            grid[y][x] = 0
    if answer_yes:
        cells = {(y, x) for y in range(4) for x in (5, 6)}
    elif negative_outer:
        cells = {(y, x) for y in (0, 1) for x in range(4, 8)}
    else:
        cells = {(y, x) for y in (2, 3) for x in range(4, 8)}
    for y, x in cells:
        grid[y][x] = 2
    for _ in range(rotation % 4):
        grid = _rotate(grid)
    return grid


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    return {
        "grid": _build_grid((int(seed) % 2) == 0, bool((int(seed) // 2) % 2), rng.randrange(4)),
        "palette": TEAL_PALETTES[rng.randrange(len(TEAL_PALETTES))],
        "offset": [rng.randint(-24, 24), rng.randint(-24, 24)],
        "cell_size": rng.choice((20, 22, 24)),
    }


def render(scene: dict) -> Image.Image:
    grid = scene["grid"]
    cell = int(scene["cell_size"])
    dx, dy = scene["offset"]
    teal = scene["palette"]
    side = len(grid) * cell
    x0 = (CANVAS - side) // 2 + int(dx)
    y0 = (CANVAS - side) // 2 + int(dy)
    image = Image.new("RGB", (CANVAS, CANVAS), "white")
    draw = ImageDraw.Draw(image)
    for y, row in enumerate(grid):
        for x, value in enumerate(row):
            if value:
                color = teal if value == 1 else YELLOW
                draw.rectangle((x0 + x * cell, y0 + y * cell,
                                x0 + (x + 1) * cell - 1, y0 + (y + 1) * cell - 1), fill=color)
    return image


def _component_count(points: set[tuple[int, int]]) -> int:
    unseen = set(points)
    count = 0
    while unseen:
        count += 1
        queue = deque([unseen.pop()])
        while queue:
            y, x = queue.popleft()
            for p in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if p in unseen:
                    unseen.remove(p)
                    queue.append(p)
    return count


def _incidence(grid: list[list[int]]) -> tuple[bool, bool]:
    h, w = len(grid), len(grid[0])
    outside = {(y, x) for y in range(h) for x in range(w) if grid[y][x] == 0 and
               (y in {0, h - 1} or x in {0, w - 1})}
    lagoon = {(y, x) for y in range(h) for x in range(w) if grid[y][x] == 0} - outside
    yellow = {(y, x) for y in range(h) for x in range(w) if grid[y][x] == 2}
    # A patch cell on the grid rim shares an edge with the exterior sea.
    touches_sea = any(y in {0, h - 1} or x in {0, w - 1} for y, x in yellow)
    touches_lagoon = any((ny, nx) in lagoon for y, x in yellow
                         for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)))
    return touches_sea, touches_lagoon


def analytic_gold(scene: dict) -> str:
    sea, lagoon = _incidence(scene["grid"])
    return "yes" if sea and lagoon else "no"


def margin(scene: dict) -> float:
    grid = scene["grid"]
    yellow = {(y, x) for y, row in enumerate(grid) for x, value in enumerate(row) if value == 2}
    sea_edges = sum(y in {0, len(grid) - 1} or x in {0, len(grid[0]) - 1} for y, x in yellow)
    lagoon_edges = 0
    for y, x in yellow:
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < len(grid) and 0 <= nx < len(grid[0]) and grid[ny][nx] == 0:
                lagoon_edges += 1
    return float(max(1, min(sea_edges if sea_edges else 2, lagoon_edges if lagoon_edges else 2)))


def is_quarantined(scene: dict) -> bool:
    grid = scene["grid"]
    if len(grid) != N or any(len(row) != N for row in grid):
        return True
    island = {(y, x) for y in range(N) for x in range(N) if grid[y][x] in {1, 2}}
    lagoon = {(y, x) for y in range(N) for x in range(N) if grid[y][x] == 0}
    yellow = {(y, x) for y in range(N) for x in range(N) if grid[y][x] == 2}
    return (_component_count(island) != 1 or _component_count(lagoon) != 1 or
            _component_count(yellow) != 1 or len(lagoon) < 12 or len(yellow) < 6)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
