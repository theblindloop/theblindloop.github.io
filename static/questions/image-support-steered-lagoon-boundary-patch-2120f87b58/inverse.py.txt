"""Independent inverse arm; receives only final pixels."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    seen = np.zeros(mask.shape, dtype=bool)
    output: list[np.ndarray] = []
    h, w = mask.shape
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]:
            continue
        seen[y0, x0] = True
        queue = deque([(y0, x0)])
        points = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        component = np.zeros(mask.shape, dtype=bool)
        yy, xx = zip(*points)
        component[np.asarray(yy), np.asarray(xx)] = True
        output.append(component)
    return sorted(output, key=lambda item: int(item.sum()), reverse=True)


def _adjacent(left: np.ndarray, right: np.ndarray) -> int:
    near = np.zeros_like(right)
    near[1:] |= right[:-1]
    near[:-1] |= right[1:]
    near[:, 1:] |= right[:, :-1]
    near[:, :-1] |= right[:, 1:]
    return int((left & near).sum())


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or min(array.shape[:2]) < 100:
        raise ValueError("invalid raster")
    high = array.max(axis=2).astype(np.int16)
    low = array.min(axis=2).astype(np.int16)
    colored = ((high - low) > 45) & (high < 250)
    points = np.argwhere(colored)
    if len(points) < 20000:
        raise ValueError("colored island missing or too small")
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    height, width = int(y1 - y0 + 1), int(x1 - x0 + 1)
    if not (235 <= height <= 300 and 235 <= width <= 300 and abs(height - width) <= 3):
        raise ValueError("island extent outside contract")
    pad = 3
    ay0, ax0 = max(0, int(y0) - pad), max(0, int(x0) - pad)
    ay1, ax1 = min(array.shape[0], int(y1) + pad + 1), min(array.shape[1], int(x1) + pad + 1)
    crop = array[ay0:ay1, ax0:ax1]
    hi = crop.max(axis=2).astype(np.int16)
    lo = crop.min(axis=2).astype(np.int16)
    land = ((hi - lo) > 45) & (hi < 250)
    yellow = (crop[:, :, 0] > 190) & (crop[:, :, 1] > 120) & (crop[:, :, 2] < 90)
    # The visible world is a thick, area-like annulus: substantial erasure
    # anywhere across its medium central footprint must invalidate the inverse,
    # while a small edge nick is tolerated.  This is also a corruption check,
    # not a coordinate lookup: it is normalized by the observed island bounds.
    island_area_fraction = float(colored.sum()) / float(height * width)
    if not 0.80 <= island_area_fraction <= 0.91:
        raise ValueError("island area fraction outside contract")
    if int(yellow.sum()) < 2500 or int(yellow.sum()) > 6500:
        raise ValueError("yellow patch outside contract")
    land_components = _components(land)
    yellow_components = _components(yellow)
    if len(land_components) != 1 or len(yellow_components) != 1:
        raise ValueError("island or patch disconnected")
    water_components = _components(~land)
    if len(water_components) != 2:
        raise ValueError("expected exactly sea and one lagoon")
    sea = next((part for part in water_components if part[0].any() or part[-1].any() or
                part[:, 0].any() or part[:, -1].any()), None)
    if sea is None:
        raise ValueError("outer sea not identifiable")
    lagoon = next(part for part in water_components if part is not sea)
    if int(lagoon.sum()) < 5000:
        raise ValueError("lagoon too small")
    sea_contact = _adjacent(yellow, sea)
    lagoon_contact = _adjacent(yellow, lagoon)
    if max(sea_contact, lagoon_contact) < 20:
        raise ValueError("no stable boundary contact")
    return "yes" if sea_contact >= 20 and lagoon_contact >= 20 else "no"
