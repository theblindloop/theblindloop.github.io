"""Deterministic table-driven local rewrite machine and raster renderer."""

from __future__ import annotations

import random
from typing import NamedTuple

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1050, 720
SYMBOLS, TAPE_SIZE, TURNS = 4, 6, 7
TABLE_X, TABLE_Y, ENTRY_W, ENTRY_H = 130, 120, 105, 86
TAPE_X, CHIP, TAPE_GAP = 650, 52, 8
START_TOP = 110
OPTION_TOPS = (285, 367, 449, 531, 613)
BACKGROUND = "#ffffff"
NAVY = "#263f52"
GOLD = "#d79a19"
GRID = "#aebbc4"
COLORS = ("#cf4557", "#e28a25", "#28966d", "#287eb8")
LETTERS = "ROGB"
FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"


class State(NamedTuple):
    tape: tuple[int, ...]
    cursor: int


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(FONT_PATH, size=size)


def _instruction(value: object) -> tuple[int, int, int]:
    if not isinstance(value, (list, tuple)) or len(value) != 3:
        raise ValueError("instruction must be [left output, right output, move]")
    a, b, move = map(int, value)
    if a not in range(SYMBOLS) or b not in range(SYMBOLS) or move not in (-1, 1):
        raise ValueError("invalid instruction")
    return a, b, move


def _table(value: object) -> tuple[tuple[tuple[int, int, int], ...], ...]:
    if not isinstance(value, list) or len(value) != SYMBOLS:
        raise ValueError("table must have four rows")
    result = tuple(tuple(_instruction(cell) for cell in row) for row in value)
    if any(len(row) != SYMBOLS for row in result):
        raise ValueError("table must be four by four")
    return result


def _state(tape: object, cursor: object) -> State:
    if not isinstance(tape, (list, tuple)) or len(tape) != TAPE_SIZE:
        raise ValueError("tape must have six symbols")
    checked = tuple(int(x) for x in tape)
    position = int(cursor)
    if any(x not in range(SYMBOLS) for x in checked) or position not in range(TAPE_SIZE - 1):
        raise ValueError("invalid tape or cursor")
    return State(checked, position)


def _scene(scene: dict) -> tuple[tuple, State, list[State]]:
    if not isinstance(scene, dict) or set(scene) != {"table", "start", "cursor", "options"}:
        raise ValueError("scene fields do not match label inputs")
    table = _table(scene["table"])
    start = _state(scene["start"], scene["cursor"])
    if not isinstance(scene["options"], list) or len(scene["options"]) != 5:
        raise ValueError("five options required")
    options = []
    for option in scene["options"]:
        if not isinstance(option, dict) or set(option) != {"tape", "cursor"}:
            raise ValueError("malformed option")
        options.append(_state(option["tape"], option["cursor"]))
    if len(set(options)) != 5:
        raise ValueError("options must be distinct")
    return table, start, options


def _advance(table: tuple, state: State, mode: str = "normal") -> State:
    tape = list(state.tape)
    p = state.cursor
    left, right = tape[p], tape[p + 1]
    if mode == "reverse_key":
        out_left, out_right, move = table[right][left]
    else:
        out_left, out_right, move = table[left][right]
    if mode == "swap_outputs":
        out_left, out_right = out_right, out_left
    if mode == "single_write":
        out_right = right
    tape[p], tape[p + 1] = out_left, out_right
    if mode == "opposite_move":
        move = -move
    return State(tuple(tape), (p + move) % (TAPE_SIZE - 1))


def _evolve(table: tuple, start: State, turns: int = TURNS, mode: str = "normal") -> State:
    current = start
    for _ in range(turns):
        current = _advance(table, current, mode)
    return current


def _trace(table: tuple, start: State) -> tuple[list[State], list[tuple[int, int, int, int]]]:
    states = [start]
    reads = []
    for _ in range(TURNS):
        state = states[-1]
        a, b = state.tape[state.cursor : state.cursor + 2]
        reads.append((state.cursor, a, b, table[a][b][2]))
        states.append(_advance(table, state))
    return states, reads


def _json_state(state: State) -> dict:
    return {"tape": list(state.tape), "cursor": state.cursor}


def _distance(first: State, second: State) -> int:
    return sum(a != b for a, b in zip(first.tape, second.tape)) + int(first.cursor != second.cursor)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    for _ in range(6000):
        table = tuple(
            tuple((rng.randrange(SYMBOLS), rng.randrange(SYMBOLS), rng.choice((-1, 1))) for _ in range(SYMBOLS))
            for _ in range(SYMBOLS)
        )
        start = State(tuple(rng.randrange(SYMBOLS) for _ in range(TAPE_SIZE)), rng.randrange(TAPE_SIZE - 1))
        states, reads = _trace(table, start)
        target = states[-1]
        revisits: dict[int, set[tuple[int, int]]] = {}
        for p, a, b, _ in reads:
            revisits.setdefault(p, set()).add((a, b))
        if not (
            set(start.tape) == set(range(SYMBOLS))
            and len({(a, b) for _, a, b, _ in reads}) >= 5
            and {move for *_, move in reads} == {-1, 1}
            and len({p for p, *_ in reads}) >= 4
            and any(len(values) >= 2 for values in revisits.values())
            and target != start
        ):
            continue
        preferred = [
            states[-2],
            _evolve(table, start, TURNS, "reverse_key"),
            _evolve(table, start, TURNS, "swap_outputs"),
            _evolve(table, start, TURNS, "opposite_move"),
            _evolve(table, start, TURNS, "single_write"),
            _evolve(table, start, TURNS + 1),
        ]
        distractors: list[State] = []
        for item in preferred:
            if item != target and item not in distractors:
                distractors.append(item)
        while len(distractors) < 4:
            tape = list(target.tape)
            if rng.randrange(2):
                tape[rng.randrange(TAPE_SIZE)] = rng.randrange(SYMBOLS)
                variant = State(tuple(tape), target.cursor)
            else:
                variant = State(tuple(tape), (target.cursor + rng.choice((-1, 1))) % (TAPE_SIZE - 1))
            if variant != target and variant not in distractors:
                distractors.append(variant)
        options = distractors[:4]
        options.insert(int(seed) % 5, target)
        return {
            "table": [[[a, b, m] for a, b, m in row] for row in table],
            "start": list(start.tape),
            "cursor": start.cursor,
            "options": [_json_state(option) for option in options],
        }
    raise RuntimeError("could not sample a feedback-rich pair rewrite machine")


def analytic_gold(scene: dict) -> str:
    table, start, options = _scene(scene)
    target = _evolve(table, start)
    matches = [i for i, option in enumerate(options) if option == target]
    if len(matches) != 1:
        raise ValueError("final state must match exactly one option")
    return "ABCDE"[matches[0]]


def margin(scene: dict) -> float:
    table, start, options = _scene(scene)
    target = _evolve(table, start)
    return float(min(_distance(target, option) for option in options if option != target))


def is_quarantined(scene: dict) -> bool:
    try:
        table, start, options = _scene(scene)
        target = _evolve(table, start)
        return sum(option == target for option in options) != 1 or margin(scene) < 1.0
    except (TypeError, ValueError, KeyError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    _scene(scene)
    return []


def _chip(draw: ImageDraw.ImageDraw, symbol: int, left: int, top: int, size: int) -> None:
    draw.rounded_rectangle((left, top, left + size, top + size), radius=max(4, size // 7), fill=COLORS[symbol], outline=NAVY, width=2)
    draw.text((left + size / 2, top + size / 2), LETTERS[symbol], fill="white", font=_font(15 if size < 40 else 23), anchor="mm")


def _arrow(draw: ImageDraw.ImageDraw, cx: int, cy: int, move: int) -> None:
    tip = cx + move * 29
    base = cx + move * 16
    draw.line((cx, cy, tip, cy), fill=NAVY, width=5)
    draw.polygon(((tip, cy), (base, cy - 9), (base, cy + 9)), fill=NAVY)


def _tape(draw: ImageDraw.ImageDraw, state: State, top: int) -> None:
    for index, symbol in enumerate(state.tape):
        _chip(draw, symbol, TAPE_X + index * (CHIP + TAPE_GAP), top, CHIP)
    gap_x = TAPE_X + (state.cursor + 1) * CHIP + state.cursor * TAPE_GAP + TAPE_GAP // 2
    draw.polygon(((gap_x, top + CHIP + 4), (gap_x - 11, top + CHIP + 23), (gap_x + 11, top + CHIP + 23)), fill=GOLD)


def render(scene: dict) -> Image.Image:
    table, start, options = _scene(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((WIDTH / 2, 20), "PAIR-REWRITE CURSOR MACHINE", fill=NAVY, font=_font(24), anchor="mm")
    draw.text((TABLE_X + 2 * ENTRY_W, 52), "RIGHT CHIP", fill=NAVY, font=_font(17), anchor="mm")
    draw.text((24, TABLE_Y + 2 * ENTRY_H), "LEFT\nCHIP", fill=NAVY, font=_font(17), anchor="lm", spacing=4)
    for symbol in range(SYMBOLS):
        _chip(draw, symbol, TABLE_X + symbol * ENTRY_W + (ENTRY_W - 40) // 2, 68, 40)
        _chip(draw, symbol, 70, TABLE_Y + symbol * ENTRY_H + (ENTRY_H - 40) // 2, 40)
    for row in range(SYMBOLS):
        for column in range(SYMBOLS):
            left = TABLE_X + column * ENTRY_W
            top = TABLE_Y + row * ENTRY_H
            draw.rounded_rectangle((left, top, left + ENTRY_W - 4, top + ENTRY_H - 4), radius=7, fill="#f5f7f8", outline=GRID, width=2)
            out_a, out_b, move = table[row][column]
            _chip(draw, out_a, left + 16, top + 9, 32)
            _chip(draw, out_b, left + 55, top + 9, 32)
            _arrow(draw, left + ENTRY_W // 2, top + 67, move)
    draw.text((610, START_TOP + CHIP / 2), "START", fill=GOLD, font=_font(21), anchor="rm")
    _tape(draw, start, START_TOP)
    draw.text((625, 204), "READ ORDER: LEFT, RIGHT", fill=NAVY, font=_font(17), anchor="lm")
    draw.text((625, 232), "ENTRY: NEW PAIR + NEXT CURSOR MOVE", fill=NAVY, font=_font(17), anchor="lm")
    draw.text((625, 260), "MOVE ONE GAP; ENDS WRAP", fill=NAVY, font=_font(17), anchor="lm")
    draw.line((600, 276, 1022, 276), fill=GRID, width=2)
    for label, option, top in zip("ABCDE", options, OPTION_TOPS):
        cy = top + CHIP // 2
        draw.ellipse((576, cy - 22, 620, cy + 22), fill=NAVY)
        draw.text((598, cy), label, fill="white", font=_font(24), anchor="mm")
        _tape(draw, option, top)
    return image
