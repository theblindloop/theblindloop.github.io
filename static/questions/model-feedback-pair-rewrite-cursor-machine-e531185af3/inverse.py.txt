"""Independent PNG-only inverse for the pair-rewrite cursor machine."""

from __future__ import annotations

import numpy as np
from PIL import Image

EXPECTED_SIZE = (1050, 720)
SYMBOLS, TAPE_SIZE, TURNS = 4, 6, 7
TABLE_X, TABLE_Y, ENTRY_W, ENTRY_H = 130, 120, 105, 86
TAPE_X, CHIP, TAPE_GAP = 650, 52, 8
START_TOP = 110
OPTION_TOPS = (285, 367, 449, 531, 613)
COLORS = np.array(((207, 69, 87), (226, 138, 37), (40, 150, 109), (40, 126, 184)), dtype=np.float32)
NAVY = np.array((38, 63, 82), dtype=np.float32)
GOLD = np.array((215, 154, 25), dtype=np.float32)


class OracleError(ValueError):
    pass


def _symbol(array: np.ndarray, left: int, top: int, size: int) -> int:
    pad, width = (4, 6) if size <= 32 else (5, 8)
    patches = (
        array[top + pad : top + pad + width, left + pad : left + pad + width],
        array[top + pad : top + pad + width, left + size - pad - width : left + size - pad],
        array[top + size - pad - width : top + size - pad, left + pad : left + pad + width],
        array[top + size - pad - width : top + size - pad, left + size - pad - width : left + size - pad],
    )
    mean = np.concatenate([patch.reshape(-1, 3) for patch in patches]).astype(np.float32).mean(axis=0)
    distances = np.linalg.norm(COLORS - mean, axis=1)
    order = np.argsort(distances)
    if float(distances[order[0]]) > 12 or float(distances[order[1]] - distances[order[0]]) < 35:
        raise OracleError("missing or ambiguous symbol chip")
    return int(order[0])


def _direction(array: np.ndarray, left: int, top: int) -> int:
    cx, cy = left + ENTRY_W // 2, top + 67
    patch = array[cy - 12 : cy + 13, cx - 34 : cx + 35].astype(np.float32)
    mask = np.linalg.norm(patch - NAVY, axis=2) < 20
    ys, xs = np.nonzero(mask)
    if len(xs) < 120:
        raise OracleError("instruction arrow missing")
    delta = float(xs.mean()) - 34.0
    if abs(delta) < 5.0:
        raise OracleError("instruction arrow direction ambiguous")
    return 1 if delta > 0 else -1


def _read_table(array: np.ndarray) -> list[list[tuple[int, int, int]]]:
    for symbol in range(SYMBOLS):
        column_key = _symbol(array, TABLE_X + symbol * ENTRY_W + (ENTRY_W - 40) // 2, 68, 40)
        row_key = _symbol(array, 70, TABLE_Y + symbol * ENTRY_H + (ENTRY_H - 40) // 2, 40)
        if column_key != symbol or row_key != symbol:
            raise OracleError("table axis key corrupted")
    table = []
    for row in range(SYMBOLS):
        output_row = []
        for column in range(SYMBOLS):
            left = TABLE_X + column * ENTRY_W
            top = TABLE_Y + row * ENTRY_H
            output_row.append((_symbol(array, left + 16, top + 9, 32), _symbol(array, left + 55, top + 9, 32), _direction(array, left, top)))
        table.append(output_row)
    return table


def _cursor(array: np.ndarray, top: int) -> int:
    scores = []
    for p in range(TAPE_SIZE - 1):
        cx = TAPE_X + (p + 1) * CHIP + p * TAPE_GAP + TAPE_GAP // 2
        patch = array[top + CHIP + 2 : top + CHIP + 25, cx - 13 : cx + 14].astype(np.float32)
        scores.append(int((np.linalg.norm(patch - GOLD, axis=2) < 20).sum()))
    order = np.argsort(scores)[::-1]
    if scores[int(order[0])] < 130 or scores[int(order[1])] > 15:
        raise OracleError("cursor missing or ambiguous")
    return int(order[0])


def _state(array: np.ndarray, top: int) -> tuple[list[int], int]:
    tape = [_symbol(array, TAPE_X + i * (CHIP + TAPE_GAP), top, CHIP) for i in range(TAPE_SIZE)]
    return tape, _cursor(array, top)


def _validate_option_labels(array: np.ndarray) -> None:
    for top in OPTION_TOPS:
        cy = top + CHIP // 2
        patch = array[cy - 18 : cy + 19, 580:617].astype(np.float32)
        if int((np.linalg.norm(patch - NAVY, axis=2) < 20).sum()) < 700:
            raise OracleError("option label disk missing")


def decision_from_image(image: Image.Image) -> str:
    if image.size != EXPECTED_SIZE:
        raise OracleError(f"unexpected raster size {image.size}")
    array = np.asarray(image.convert("RGB"))
    table = _read_table(array)
    tape, cursor = _state(array, START_TOP)
    _validate_option_labels(array)
    options = [_state(array, top) for top in OPTION_TOPS]
    signatures = {(tuple(tape_values), p) for tape_values, p in options}
    if len(signatures) != 5:
        raise OracleError("options are not distinct")
    for _ in range(TURNS):
        left, right = tape[cursor], tape[cursor + 1]
        out_left, out_right, move = table[left][right]
        tape[cursor], tape[cursor + 1] = out_left, out_right
        cursor = (cursor + move) % (TAPE_SIZE - 1)
    matches = [i for i, (candidate, p) in enumerate(options) if candidate == tape and p == cursor]
    if len(matches) != 1:
        raise OracleError("recovered final state does not match exactly one option")
    return "ABCDE"[matches[0]]
