"""Independent pixel-only inverse arm.

Receives only the final PNG. It segments the gray reference cluster on top and
each colored candidate cluster below by color, counts the connected disk
components per cluster, finds the candidate whose component count equals the
reference count, and returns that candidate's color name.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

GRAY = (150, 150, 150)
PALETTE = {
    "red": (215, 60, 60),
    "green": (55, 180, 75),
    "blue": (65, 100, 215),
}
TOL = 70


def _close(image: np.ndarray, color: tuple) -> np.ndarray:
    return np.abs(image.astype(int) - np.array(color)).sum(axis=2) < TOL


def _count_components(mask: np.ndarray) -> int:
    h, w = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    ys, xs = np.nonzero(mask)
    count = 0
    for y, x in zip(ys.tolist(), xs.tolist()):
        if visited[y, x]:
            continue
        count += 1
        stack = [(y, x)]
        visited[y, x] = True
        while stack:
            cy, cx = stack.pop()
            y0 = max(0, cy - 1)
            y1 = min(h, cy + 2)
            x0 = max(0, cx - 1)
            x1 = min(w, cx + 2)
            for ny in range(y0, y1):
                row = mask[ny]
                for nx in range(x0, x1):
                    if row[nx] and not visited[ny, nx]:
                        visited[ny, nx] = True
                        stack.append((ny, nx))
    return count


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"))
    ref = _count_components(_close(rgb, GRAY))
    if ref == 0:
        return "unknown"
    for name, color in PALETTE.items():
        candidate = _count_components(_close(rgb, color))
        if candidate == ref:
            return name
    return "unknown"
