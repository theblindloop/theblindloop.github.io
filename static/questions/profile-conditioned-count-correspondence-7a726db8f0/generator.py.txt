"""Cross-panel count-correspondence world: match a reference cluster's disk
count to a candidate, then report the bound color of the matched cluster."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 620, 380
REF_CENTER = (310, 130)
CAND_CENTERS = [(160, 290), (310, 290), (460, 290)]

DISK_R = 16
SPACING = 46

GRAY = (150, 150, 150)
PALETTE = {
    "red": (215, 60, 60),
    "green": (55, 180, 75),
    "blue": (65, 100, 215),
}
COLOR_NAMES = ["red", "green", "blue"]

COUNTS = [2, 3, 4, 5]
MIN_MARGIN = 1.0


def _row_centers(center, count: int) -> list[tuple[int, int]]:
    cx, cy = center
    start_x = cx - (count - 1) * SPACING // 2
    return [(start_x + i * SPACING, cy) for i in range(count)]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    colors = list(COLOR_NAMES)
    rng.shuffle(colors)
    ref_count = rng.choice(COUNTS)
    remainder = [c for c in COUNTS if c != ref_count]
    other_a, other_b = rng.sample(remainder, 2)
    counts_for_colors = [ref_count, other_a, other_b]
    rng.shuffle(counts_for_colors)
    slots = [(color, count) for color, count in zip(colors, counts_for_colors)]
    positions = list(CAND_CENTERS)
    rng.shuffle(positions)
    return {
        "ref_count": ref_count,
        "candidate_colors": colors,
        "candidate_counts": counts_for_colors,
        "candidate_positions": positions,
    }


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), (252, 252, 252))
    draw = ImageDraw.Draw(img)
    for cx, cy in _row_centers(REF_CENTER, scene["ref_count"]):
        draw.ellipse([cx - DISK_R, cy - DISK_R, cx + DISK_R, cy + DISK_R], fill=GRAY)
    for color_name, count, (cx, cy) in zip(
        scene["candidate_colors"],
        scene["candidate_counts"],
        scene["candidate_positions"],
    ):
        for x, y in _row_centers((cx, cy), count):
            draw.ellipse([x - DISK_R, y - DISK_R, x + DISK_R, y + DISK_R], fill=PALETTE[color_name])
    return img


def analytic_gold(scene: dict) -> str:
    ref = scene["ref_count"]
    index = scene["candidate_counts"].index(ref)
    return scene["candidate_colors"][index]


def margin(scene: dict) -> float:
    ref = scene["ref_count"]
    others = [c for c in scene["candidate_counts"] if c != ref]
    if not others:
        return float("inf")
    return float(min(abs(ref - c) for c in others))


def is_quarantined(scene: dict) -> bool:
    counts = scene["candidate_counts"]
    if len(set(counts)) != len(counts):
        return True
    return margin(scene) < MIN_MARGIN


def latent_symmetries(scene: dict) -> list:
    return []
