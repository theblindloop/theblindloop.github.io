"""Renderer and analytic arm for the Antipodal Midpoint Fold world.

The label is carried only by the midpoint of bead pairs equidistant from the
two ends of one long thread.  A large antisymmetric carrier scaffold cancels
under the declared midpoint operation and visually masks the small glyph.
"""

from __future__ import annotations

import math
import random
from typing import Iterable

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 640
HEIGHT = 640
BEAD_COUNT = 18
PAIR_COUNT = 9
LABELS = ("chevron", "zigzag", "hook")

_RAW_TEMPLATES = {
    # Four equal diagonals to a tip, then four equal diagonals away.
    "chevron": [
        (-4, 4), (-3, 3), (-2, 2), (-1, 1), (0, 0),
        (1, 1), (2, 2), (3, 3), (4, 4),
    ],
    # Eight equal horizontal advances, alternating across the center line.
    "zigzag": [
        (-4, 0), (-3, 2), (-2, -2), (-1, 2), (0, -2),
        (1, 2), (2, -2), (3, 2), (4, 0),
    ],
    # Six equal steps along a stem, then two equal right-angle steps.
    "hook": [
        (-4, 0), (-3, 0), (-2, 0), (-1, 0), (0, 0),
        (1, 0), (2, 0), (2, 1), (2, 2),
    ],
}

_PALETTES = (
    ((37, 176, 190), (34, 39, 49)),
    ((28, 162, 181), (39, 42, 52)),
    ((47, 184, 177), (31, 39, 48)),
)
_BACKGROUND = (248, 246, 238)
_RED = (218, 66, 70)
_GOLD = (235, 174, 46)
_PALE = (214, 214, 205)


def _template(label: str) -> np.ndarray:
    points = np.asarray(_RAW_TEMPLATES[label], dtype=np.float64)
    points -= points.mean(axis=0, keepdims=True)
    points /= math.sqrt(float(np.mean(np.sum(points * points, axis=1))))
    return points


def _fit_error(points: np.ndarray, label: str) -> float:
    observed = np.asarray(points, dtype=np.float64)
    observed = observed - observed.mean(axis=0, keepdims=True)
    norm = math.sqrt(float(np.sum(observed * observed)))
    if norm < 1e-9:
        return float("inf")
    observed /= norm
    target = _template(label)
    target /= math.sqrt(float(np.sum(target * target)))
    u, _, vt = np.linalg.svd(observed.T @ target)
    rotation = u @ vt
    if np.linalg.det(rotation) < 0:
        u[:, -1] *= -1
        rotation = u @ vt
    delta = observed @ rotation - target
    return float(np.sum(delta * delta))


def _classification(points: np.ndarray) -> tuple[str, float, dict[str, float]]:
    errors = {label: _fit_error(points, label) for label in LABELS}
    ranked = sorted(errors, key=lambda label: (errors[label], label))
    return ranked[0], errors[ranked[1]] - errors[ranked[0]], errors


def _shape_coordinates(scene: dict) -> np.ndarray:
    primary = _template(scene["label"])
    competitor = _template(scene["mix_label"])
    alpha = float(scene["mix_alpha"])
    jitter = np.asarray(scene["shape_jitter"], dtype=np.float64)
    return (1.0 - alpha) * primary + alpha * competitor + jitter


def _local_beads(scene: dict) -> np.ndarray:
    midpoint_shape = _shape_coordinates(scene) * float(scene["target_scale"])
    scaffold = np.asarray(scene["pair_offsets"], dtype=np.float64)
    first = scaffold + midpoint_shape
    second = -scaffold[::-1] + midpoint_shape[::-1]
    return np.vstack((first, second))


def _scene_beads(scene: dict) -> np.ndarray:
    points = _local_beads(scene) * float(scene["scale"])
    angle = math.radians(float(scene["rotation"]))
    rotation = np.asarray(
        [[math.cos(angle), -math.sin(angle)], [math.sin(angle), math.cos(angle)]],
        dtype=np.float64,
    )
    points = points @ rotation.T
    points[:, 0] += float(scene["center_x"])
    points[:, 1] += float(scene["center_y"])
    return points


def _endpoint_centers(beads: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    start_direction = beads[0] - beads[1]
    start_direction /= np.linalg.norm(start_direction)
    end_direction = beads[-1] - beads[-2]
    end_direction /= np.linalg.norm(end_direction)
    return beads[0] + 23.0 * start_direction, beads[-1] + 23.0 * end_direction


def sample_scene(seed: int) -> dict:
    """Sample a deterministic balanced scene from one integer seed."""

    rng = random.Random(int(seed))
    label_index = int(seed) % len(LABELS)
    label = LABELS[label_index]
    mix_label = LABELS[(label_index + 1 + (int(seed) // 3) % 2) % len(LABELS)]
    # Every eighth seed approaches the visible template boundary without entering it.
    mix_alpha = rng.uniform(0.25, 0.31) if int(seed) % 8 == 0 else rng.uniform(0.03, 0.13)
    shape_jitter = [[rng.uniform(-0.018, 0.018), rng.uniform(-0.018, 0.018)] for _ in range(9)]

    amplitude = rng.uniform(88.0, 108.0)
    cycles = rng.choice((1.0, 2.0))
    pair_offsets = []
    for index in range(PAIR_COUNT):
        x = -230.0 + index * (216.0 / 8.0) + rng.uniform(-1.5, 1.5)
        y = amplitude * math.sin(cycles * math.pi * x / 230.0) + rng.uniform(-4.0, 4.0)
        pair_offsets.append([x, y])

    scene = {
        "label": label,
        "mix_label": mix_label,
        "mix_alpha": mix_alpha,
        "shape_jitter": shape_jitter,
        "target_scale": rng.uniform(25.0, 30.0),
        "rotation": rng.uniform(0.0, 360.0),
        "scale": rng.uniform(0.90, 0.98),
        "center_x": WIDTH / 2 + rng.uniform(-7.0, 7.0),
        "center_y": HEIGHT / 2 + rng.uniform(-7.0, 7.0),
        "pair_offsets": pair_offsets,
        "palette_index": rng.randrange(len(_PALETTES)),
        "distractor_seed": rng.randrange(1 << 30),
    }
    decision, decision_margin, _ = _classification(_shape_coordinates(scene))
    if decision != label or decision_margin <= 0.045:
        # The deterministic fallback remains randomized in pose and scaffold.
        scene["mix_alpha"] = 0.08
        scene["shape_jitter"] = [[0.0, 0.0] for _ in range(9)]
    return scene


def analytic_gold(scene: dict) -> str:
    return _classification(_shape_coordinates(scene))[0]


def margin(scene: dict) -> float:
    """Runner-up minus winning normalized Procrustes error."""

    return _classification(_shape_coordinates(scene))[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.04


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _scaled(points: Iterable[tuple[float, float]], factor: int) -> list[tuple[float, float]]:
    return [(float(x) * factor, float(y) * factor) for x, y in points]


def render(scene: dict) -> Image.Image:
    """Render the final raster.  No textual or metadata payload is embedded."""

    supersample = 3
    bead_color, dark = _PALETTES[int(scene["palette_index"])]
    image = Image.new("RGB", (WIDTH * supersample, HEIGHT * supersample), _BACKGROUND)
    draw = ImageDraw.Draw(image)

    # Pale, label-independent distractors occupy off-thread regions.
    nuisance = random.Random(int(scene["distractor_seed"]))
    for _ in range(28):
        x = nuisance.uniform(24, WIDTH - 24) * supersample
        y = nuisance.uniform(24, HEIGHT - 24) * supersample
        r = nuisance.uniform(1.2, 2.5) * supersample
        draw.ellipse((x - r, y - r, x + r, y + r), fill=_PALE)

    beads = _scene_beads(scene)
    start, end = _endpoint_centers(beads)
    carrier = np.vstack((start, beads, end))
    draw.line(_scaled(carrier, supersample), fill=dark, width=5 * supersample, joint="curve")

    # Dark bead rims keep the carrier topologically connected after cyan fill covers its core.
    for x, y in beads:
        cx, cy = x * supersample, y * supersample
        outer = 9.0 * supersample
        inner = 6.1 * supersample
        draw.ellipse((cx - outer, cy - outer, cx + outer, cy + outer), fill=dark)
        draw.ellipse((cx - inner, cy - inner, cx + inner, cy + inner), fill=bead_color)

    sx, sy = start * supersample
    draw.ellipse((sx - 13 * supersample, sy - 13 * supersample,
                  sx + 13 * supersample, sy + 13 * supersample), fill=dark)
    draw.ellipse((sx - 10 * supersample, sy - 10 * supersample,
                  sx + 10 * supersample, sy + 10 * supersample), fill=_RED)
    draw.ellipse((sx - 5 * supersample, sy - 5 * supersample,
                  sx + 5 * supersample, sy + 5 * supersample), fill=_BACKGROUND)

    angle = math.atan2(end[1] - beads[-1, 1], end[0] - beads[-1, 0])
    directions = (angle, angle + 2.45, angle - 2.45)
    outer_triangle = [
        (end[0] + 15 * math.cos(value), end[1] + 15 * math.sin(value))
        for value in directions
    ]
    inner_triangle = [
        (end[0] + 10.5 * math.cos(value), end[1] + 10.5 * math.sin(value))
        for value in directions
    ]
    draw.polygon(_scaled(outer_triangle, supersample), fill=dark)
    draw.polygon(_scaled(inner_triangle, supersample), fill=_GOLD)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
