"""Independent raster-only inverse arm for the antipodal midpoint fold.

This module intentionally imports no candidate renderer, prompt, generator,
verifier, latent scene, or analytic-gold code.  It finds color fiducials in the
PNG, proves that one dark component links them, orders beads by geodesic distance
from the red endpoint, performs the declared end-pair midpoint construction, and
fits the three publicly defined templates with a direct-similarity Procrustes fit.
"""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

_LABELS = ("chevron", "zigzag", "hook")
_RAW = {
    "chevron": [
        (-4, 4), (-3, 3), (-2, 2), (-1, 1), (0, 0),
        (1, 1), (2, 2), (3, 3), (4, 4),
    ],
    "zigzag": [
        (-4, 0), (-3, 2), (-2, -2), (-1, 2), (0, -2),
        (1, 2), (2, -2), (3, 2), (4, 0),
    ],
    "hook": [
        (-4, 0), (-3, 0), (-2, 0), (-1, 0), (0, 0),
        (1, 0), (2, 0), (2, 1), (2, 2),
    ],
}
_CYANS = np.asarray(((37, 176, 190), (28, 162, 181), (47, 184, 177)), dtype=np.float32)
_DARKS = np.asarray(((34, 39, 49), (39, 42, 52), (31, 39, 48)), dtype=np.float32)
_RED = np.asarray((218, 66, 70), dtype=np.float32)
_GOLD = np.asarray((235, 174, 46), dtype=np.float32)


class OracleAbstention(ValueError):
    """Raised when visible fiducials or the connected carrier are damaged."""


def _near_palette(array: np.ndarray, palette: np.ndarray, tolerance: float) -> np.ndarray:
    colors = palette.reshape((-1, 1, 1, 3))
    delta = array.astype(np.float32)[None, ...] - colors
    return np.min(np.sum(delta * delta, axis=3), axis=0) <= tolerance * tolerance


def _components(mask: np.ndarray, *, minimum: int, maximum: int) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = [(y, x)]
        points = []
        cursor = 0
        while cursor < len(queue):
            cy, cx = queue[cursor]
            cursor += 1
            points.append((cy, cx))
            for ny in range(max(0, cy - 1), min(height, cy + 2)):
                for nx in range(max(0, cx - 1), min(width, cx + 2)):
                    if mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        if minimum <= len(points) <= maximum:
            output.append(np.asarray(points, dtype=np.int32))
    return output


def _single_center(mask: np.ndarray, name: str, limits: tuple[int, int]) -> np.ndarray:
    components = _components(mask, minimum=limits[0], maximum=limits[1])
    if len(components) != 1:
        raise OracleAbstention(f"expected one intact {name} fiducial, found {len(components)}")
    yx = components[0].mean(axis=0)
    return np.asarray((yx[1], yx[0]), dtype=np.float64)


def _bead_centers(mask: np.ndarray) -> list[np.ndarray]:
    components = _components(mask, minimum=70, maximum=180)
    if len(components) != 18:
        raise OracleAbstention(f"expected eighteen intact cyan beads, found {len(components)}")
    centers = []
    for points in components:
        yx = points.mean(axis=0)
        # Large partial circles are rejected by their radial second moment.
        spread = float(np.mean(np.sum((points - yx) ** 2, axis=1)))
        if not 12.0 <= spread <= 27.0:
            raise OracleAbstention("a cyan bead is clipped or non-circular")
        centers.append(np.asarray((yx[1], yx[0]), dtype=np.float64))
    return centers


def _carrier_distances(
    dark: np.ndarray,
    red: np.ndarray,
    gold: np.ndarray,
    beads: list[np.ndarray],
) -> tuple[list[float], float]:
    height, width = dark.shape
    yy, xx = np.indices(dark.shape)
    red_d2 = (xx - red[0]) ** 2 + (yy - red[1]) ** 2
    starts = np.argwhere(dark & (red_d2 <= 16.0**2))
    if len(starts) < 18:
        raise OracleAbstention("red ring is not attached to the dark carrier")

    distance = np.full((height, width), -1, dtype=np.int32)
    queue: deque[tuple[int, int]] = deque()
    for raw_y, raw_x in starts:
        y, x = int(raw_y), int(raw_x)
        distance[y, x] = 0
        queue.append((y, x))
    while queue:
        y, x = queue.popleft()
        next_distance = int(distance[y, x]) + 1
        for ny in range(max(0, y - 1), min(height, y + 2)):
            for nx in range(max(0, x - 1), min(width, x + 2)):
                if dark[ny, nx] and distance[ny, nx] < 0:
                    distance[ny, nx] = next_distance
                    queue.append((ny, nx))

    reachable = distance >= 0
    reachable_points = np.argwhere(reachable)
    if len(reachable_points) < 1800:
        raise OracleAbstention("the red-connected carrier is too short")
    span = np.hypot(
        np.ptp(reachable_points[:, 0]),
        np.ptp(reachable_points[:, 1]),
    ) / np.hypot(height, width)
    if span < 0.40:
        raise OracleAbstention("the carrier no longer spans the image")

    def geodesic_near(center: np.ndarray, radius: float) -> float:
        d2 = (xx - center[0]) ** 2 + (yy - center[1]) ** 2
        values = distance[(d2 <= radius * radius) & reachable]
        if len(values) < 8:
            raise OracleAbstention("a fiducial is disconnected from the carrier")
        return float(values.min())

    bead_distances = [geodesic_near(center, 11.5) for center in beads]
    gold_distance = geodesic_near(gold, 17.0)
    ordered_distances = sorted(bead_distances)
    if any(b - a < 1.0 for a, b in zip(ordered_distances, ordered_distances[1:])):
        raise OracleAbstention("bead travel order is not uniquely separated")
    if gold_distance <= ordered_distances[-1] + 3.0:
        raise OracleAbstention("gold endpoint does not terminate the ordered carrier")
    return bead_distances, gold_distance


def _template(label: str) -> np.ndarray:
    points = np.asarray(_RAW[label], dtype=np.float64)
    points -= points.mean(axis=0, keepdims=True)
    points /= math.sqrt(float(np.sum(points * points)))
    return points


def _fit_error(points: np.ndarray, label: str) -> float:
    observed = np.asarray(points, dtype=np.float64).copy()
    observed -= observed.mean(axis=0, keepdims=True)
    norm = math.sqrt(float(np.sum(observed * observed)))
    if norm < 4.0:
        raise OracleAbstention("the folded midpoint path has negligible extent")
    observed /= norm
    target = _template(label)
    u, _, vt = np.linalg.svd(observed.T @ target)
    rotation = u @ vt
    if np.linalg.det(rotation) < 0:
        u[:, -1] *= -1
        rotation = u @ vt
    difference = observed @ rotation - target
    return float(np.sum(difference * difference))


def decision_from_image(image: Image.Image) -> str:
    """Recover one answer from final RGB pixels alone."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 300 or array.shape[1] < 300:
        raise OracleAbstention("unsupported or undersized raster")

    cyan_mask = _near_palette(array, _CYANS, 54.0)
    red_mask = _near_palette(array, _RED[None, :], 54.0)
    gold_mask = _near_palette(array, _GOLD[None, :], 54.0)
    dark_mask = _near_palette(array, _DARKS, 52.0)

    red = _single_center(red_mask, "red ring", (150, 390))
    gold = _single_center(gold_mask, "gold triangle", (90, 320))
    beads = _bead_centers(cyan_mask)
    distances, _ = _carrier_distances(dark_mask, red, gold, beads)
    ordered = [center for _, center in sorted(zip(distances, beads), key=lambda item: item[0])]

    midpoints = np.asarray(
        [(ordered[index] + ordered[17 - index]) / 2.0 for index in range(9)],
        dtype=np.float64,
    )
    errors = {label: _fit_error(midpoints, label) for label in _LABELS}
    ranked = sorted(errors, key=lambda label: (errors[label], label))
    if errors[ranked[1]] - errors[ranked[0]] <= 0.012:
        return "ambiguous"
    return ranked[0]
