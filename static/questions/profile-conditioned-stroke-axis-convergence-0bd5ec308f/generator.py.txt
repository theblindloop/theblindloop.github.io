"""Stroke-axis convergence: a grey stroke field whose axes meet at one coloured dot.

The scene is a plain canvas covered with short grey strokes of identical length
and width.  Every stroke is aimed at one hidden spot, up to a small independent
wobble, so an individual stroke constrains the spot only to a fat band while the
whole field pins it down.  Four coloured dots sit on a common circle; exactly one
of them is the spot the field converges on.

The analytic gold reads only the two rendered fields ``strokes`` and ``markers``:
for every dot it averages the perpendicular distance from that dot to the infinite
line carried by each stroke, and names the dot with the smallest average.  No
sampling scaffolding reaches the label, so an edited stroke field is scored on the
edited geometry.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 720
HEIGHT = 560
BG_COLOR = (247, 246, 242)
STROKE_COLOR = (105, 105, 105)
STROKE_LEN = 34
STROKE_WIDTH = 3
MARKER_RADIUS = 11
MARKER_RING_RANGE = (95.0, 205.0)
STROKE_COUNT = 44
JITTER_RANGE_DEG = (8.0, 24.0)

CLASS_NAMES = ("red", "green", "blue", "orange")
PALETTE = {
    "red": (214, 40, 40),
    "green": (30, 148, 64),
    "blue": (40, 82, 208),
    "orange": (235, 140, 18),
}

QUARANTINE_MARGIN = 14.0
SAMPLING_FLOOR = 22.0


def _point_line_distance(px: float, py: float, sx: float, sy: float, angle_deg: float) -> float:
    """Perpendicular distance from (px, py) to the infinite line through (sx, sy)."""
    theta = math.radians(angle_deg)
    return abs(-math.sin(theta) * (px - sx) + math.cos(theta) * (py - sy))


def _mean_axis_distance(px: float, py: float, strokes: list[dict]) -> float:
    """Mean perpendicular distance from one dot centre to every stroke axis."""
    total = 0.0
    for stroke in strokes:
        total += _point_line_distance(
            px, py, float(stroke["x"]), float(stroke["y"]), float(stroke["angle_deg"])
        )
    return total / float(len(strokes))


def _segment_endpoints(stroke: dict) -> tuple[float, float, float, float]:
    theta = math.radians(float(stroke["angle_deg"]))
    half = STROKE_LEN / 2.0
    dx, dy = math.cos(theta) * half, math.sin(theta) * half
    return (
        float(stroke["x"]) - dx, float(stroke["y"]) - dy,
        float(stroke["x"]) + dx, float(stroke["y"]) + dy,
    )


def _segment_distance(a: dict, b: dict) -> float:
    ax0, ay0, ax1, ay1 = _segment_endpoints(a)
    bx0, by0, bx1, by1 = _segment_endpoints(b)
    best = float("inf")
    for steps_a in range(9):
        ta = steps_a / 8.0
        pax, pay = ax0 + (ax1 - ax0) * ta, ay0 + (ay1 - ay0) * ta
        for steps_b in range(9):
            tb = steps_b / 8.0
            pbx, pby = bx0 + (bx1 - bx0) * tb, by0 + (by1 - by0) * tb
            best = min(best, math.hypot(pax - pbx, pay - pby))
    return best


def _segment_point_distance(stroke: dict, px: float, py: float) -> float:
    x0, y0, x1, y1 = _segment_endpoints(stroke)
    vx, vy = x1 - x0, y1 - y0
    length_sq = vx * vx + vy * vy
    t = 0.0 if length_sq == 0 else ((px - x0) * vx + (py - y0) * vy) / length_sq
    t = max(0.0, min(1.0, t))
    return math.hypot(px - (x0 + vx * t), py - (y0 + vy * t))


def sample_scene(seed: int) -> dict:
    """Sample one balanced scene; rejection sampling keeps the field legible."""
    rng = random.Random(seed)
    for _ in range(4000):
        rotation = rng.uniform(0.0, 360.0)
        marker_ring = rng.uniform(*MARKER_RING_RANGE)
        jitter_deg = rng.uniform(*JITTER_RANGE_DEG)
        centre_x = WIDTH / 2.0 + rng.uniform(-16.0, 16.0)
        centre_y = HEIGHT / 2.0 + rng.uniform(-16.0, 16.0)
        colours = list(CLASS_NAMES)
        rng.shuffle(colours)
        markers = []
        for slot, colour in enumerate(colours):
            phi = math.radians(rotation + 90.0 * slot)
            markers.append(
                {
                    "color": colour,
                    "x": round(centre_x + marker_ring * math.cos(phi), 3),
                    "y": round(centre_y + marker_ring * math.sin(phi), 3),
                }
            )
        target = markers[rng.randrange(4)]

        strokes: list[dict] = []
        attempts = 0
        while len(strokes) < STROKE_COUNT and attempts < 900:
            attempts += 1
            sx = rng.uniform(40.0, WIDTH - 40.0)
            sy = rng.uniform(40.0, HEIGHT - 40.0)
            # Stroke placement is uniform over the canvas and independent of which
            # dot the field is aimed at, so the ink centroid carries no answer cue.
            # Only a small exclusion disc keeps every aim direction well defined.
            if math.hypot(sx - float(target["x"]), sy - float(target["y"])) < 60.0:
                continue
            aim = math.degrees(math.atan2(float(target["y"]) - sy, float(target["x"]) - sx))
            candidate = {
                "x": round(sx, 3),
                "y": round(sy, 3),
                "angle_deg": round(aim + rng.uniform(-jitter_deg, jitter_deg), 3),
            }
            if any(
                _segment_point_distance(candidate, float(m["x"]), float(m["y"]))
                < MARKER_RADIUS + 13.0
                for m in markers
            ):
                continue
            if any(_segment_distance(candidate, other) < 13.0 for other in strokes):
                continue
            strokes.append(candidate)
        if len(strokes) < STROKE_COUNT:
            continue

        scene = {
            "width": WIDTH,
            "height": HEIGHT,
            "bg_color": list(BG_COLOR),
            "stroke_color": list(STROKE_COLOR),
            "stroke_len": STROKE_LEN,
            "stroke_width": STROKE_WIDTH,
            "marker_radius": MARKER_RADIUS,
            "strokes": strokes,
            "markers": markers,
        }
        if margin(scene) < SAMPLING_FLOOR:
            continue
        return scene
    raise RuntimeError(f"no legible scene for seed {seed}")


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), tuple(scene["bg_color"]))
    draw = ImageDraw.Draw(image)
    half = float(scene["stroke_len"]) / 2.0
    for stroke in scene["strokes"]:
        theta = math.radians(float(stroke["angle_deg"]))
        dx, dy = math.cos(theta) * half, math.sin(theta) * half
        draw.line(
            [
                (float(stroke["x"]) - dx, float(stroke["y"]) - dy),
                (float(stroke["x"]) + dx, float(stroke["y"]) + dy),
            ],
            fill=tuple(scene["stroke_color"]),
            width=int(scene["stroke_width"]),
        )
    radius = float(scene["marker_radius"])
    for marker in scene["markers"]:
        mx, my = float(marker["x"]), float(marker["y"])
        draw.ellipse(
            [mx - radius, my - radius, mx + radius, my + radius],
            fill=PALETTE[marker["color"]],
        )
    return image


def _ranked_distances(scene: dict) -> list[tuple[float, str]]:
    ranked = [
        (
            _mean_axis_distance(float(marker["x"]), float(marker["y"]), scene["strokes"]),
            str(marker["color"]),
        )
        for marker in scene["markers"]
    ]
    ranked.sort(key=lambda item: (item[0], item[1]))
    return ranked


def analytic_gold(scene: dict) -> str:
    return _ranked_distances(scene)[0][1]


def margin(scene: dict) -> float:
    ranked = _ranked_distances(scene)
    return float(ranked[1][0] - ranked[0][0])


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_MARGIN)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical relabellings of the two rendered lists."""
    reversed_strokes = dict(scene)
    reversed_strokes["strokes"] = list(reversed(scene["strokes"]))
    shifted_markers = dict(scene)
    shifted_markers["markers"] = list(scene["markers"][1:]) + list(scene["markers"][:1])
    return [
        ("stroke_order_reversal", reversed_strokes),
        ("marker_order_cyclic_shift", shifted_markers),
    ]
