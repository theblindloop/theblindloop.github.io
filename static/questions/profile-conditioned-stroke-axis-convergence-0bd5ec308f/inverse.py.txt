"""Pixel-only oracle: rebuild the stroke-axis field and score the coloured dots.

Receives a PNG raster and nothing else.  It segments the coloured dots and the
grey stroke texture straight out of the pixels, fits one axis per stroke by
principal-axis analysis of that stroke's own pixels, and returns the dot with the
smallest mean perpendicular distance to those axes.  It abstains when the two best
dots are not separated.
"""

from __future__ import annotations

import math

from PIL import Image

_CLASS_NAMES = ("red", "green", "blue", "orange")
_REFERENCE = {
    "red": (214, 40, 40),
    "green": (30, 148, 64),
    "blue": (40, 82, 208),
    "orange": (235, 140, 18),
}
_SATURATION_MIN = 60
_STROKE_MAX_LEVEL = 190
_MIN_STROKE_PIXELS = 24
_ABSTAIN_MARGIN = 6.0
ABSTAIN = "ambiguous"


def _classify(pixel: tuple[int, int, int]) -> str | None:
    red, green, blue = pixel
    if max(pixel) - min(pixel) < _SATURATION_MIN:
        return None
    best_name, best_cost = None, None
    for name, reference in _REFERENCE.items():
        cost = sum((a - b) ** 2 for a, b in zip(pixel, reference))
        if best_cost is None or cost < best_cost:
            best_name, best_cost = name, cost
    return best_name


def _components(mask: set[tuple[int, int]]) -> list[list[tuple[int, int]]]:
    """Eight-connected components of a pixel set, found with an explicit stack."""
    remaining = set(mask)
    groups: list[list[tuple[int, int]]] = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        group = [start]
        while stack:
            cx, cy = stack.pop()
            for dx in (-1, 0, 1):
                for dy in (-1, 0, 1):
                    neighbour = (cx + dx, cy + dy)
                    if neighbour in remaining:
                        remaining.discard(neighbour)
                        stack.append(neighbour)
                        group.append(neighbour)
        groups.append(group)
    return groups


def _principal_axis(group: list[tuple[int, int]]) -> tuple[float, float, float]:
    count = float(len(group))
    mean_x = sum(point[0] for point in group) / count
    mean_y = sum(point[1] for point in group) / count
    sxx = syy = sxy = 0.0
    for px, py in group:
        dx, dy = px - mean_x, py - mean_y
        sxx += dx * dx
        syy += dy * dy
        sxy += dx * dy
    angle = 0.5 * math.atan2(2.0 * sxy, sxx - syy)
    return mean_x, mean_y, angle


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    width, height = rgb.size
    pixels = rgb.load()

    coloured: dict[str, list[tuple[int, int]]] = {}
    dark: set[tuple[int, int]] = set()
    for y in range(height):
        for x in range(width):
            pixel = pixels[x, y]
            name = _classify(pixel)
            if name is not None:
                coloured.setdefault(name, []).append((x, y))
            elif max(pixel) <= _STROKE_MAX_LEVEL:
                dark.add((x, y))

    markers: dict[str, tuple[float, float]] = {}
    for name in _CLASS_NAMES:
        blob = coloured.get(name)
        if not blob or len(blob) < _MIN_STROKE_PIXELS:
            continue
        markers[name] = (
            sum(point[0] for point in blob) / float(len(blob)),
            sum(point[1] for point in blob) / float(len(blob)),
        )
    axes = [
        _principal_axis(group)
        for group in _components(dark)
        if len(group) >= _MIN_STROKE_PIXELS
    ]
    if len(markers) < 2 or len(axes) < 8:
        return ABSTAIN

    scores: list[tuple[float, str]] = []
    for name, (mx, my) in markers.items():
        total = 0.0
        for ax, ay, angle in axes:
            total += abs(-math.sin(angle) * (mx - ax) + math.cos(angle) * (my - ay))
        scores.append((total / float(len(axes)), name))
    scores.sort(key=lambda item: (item[0], item[1]))
    if scores[1][0] - scores[0][0] < _ABSTAIN_MARGIN:
        return ABSTAIN
    return scores[0][1]
