from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
COLORS = ("blue", "orange", "green")
RGB = {
    "blue": (31, 119, 180),
    "orange": (245, 125, 32),
    "green": (44, 160, 70),
    "magenta": (190, 45, 145),
    "black": (34, 36, 42),
    "guide": (220, 224, 230),
    "background": (250, 250, 248),
}
DOT_RADIUS = 12.0
ORIGIN_RADIUS = 9.0
TARGET_RADIUS = 15.0
QUARANTINE_GAP = 12.0


def _projection_values(scene: dict) -> dict[str, float]:
    ox, oy = scene["origin"]
    tx, ty = scene["target"]
    dx, dy = tx - ox, ty - oy
    norm = math.hypot(dx, dy)
    ux, uy = dx / norm, dy / norm
    return {
        color: (xy[0] - ox) * ux + (xy[1] - oy) * uy
        for color, xy in scene["points"].items()
    }


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    winner = COLORS[int(seed) % len(COLORS)]

    for _ in range(500):
        angle = rng.uniform(0.0, 2.0 * math.pi)
        ux, uy = math.cos(angle), math.sin(angle)
        vx, vy = -uy, ux
        origin = [rng.uniform(238.0, 274.0), rng.uniform(238.0, 274.0)]
        target_distance = rng.uniform(126.0, 148.0)
        target = [origin[0] + target_distance * ux, origin[1] + target_distance * uy]

        near_boundary = int(seed) % 7 == 0
        gap = rng.uniform(15.0, 18.0) if near_boundary else rng.uniform(30.0, 52.0)
        runner_q = rng.uniform(72.0, 108.0)
        third_q = runner_q - rng.uniform(24.0, 48.0)
        q_values = [runner_q + gap, runner_q, third_q]
        rng.shuffle(q_values[1:])

        losers = [color for color in COLORS if color != winner]
        rng.shuffle(losers)
        q_by_color = {winner: q_values[0], losers[0]: q_values[1], losers[1]: q_values[2]}
        p_by_color = {color: rng.uniform(-128.0, 128.0) for color in COLORS}
        points = {
            color: [
                origin[0] + q_by_color[color] * ux + p_by_color[color] * vx,
                origin[1] + q_by_color[color] * uy + p_by_color[color] * vy,
            ]
            for color in COLORS
        }

        landmarks = [origin, target, *points.values()]
        if any(not (24.0 <= x <= WIDTH - 24.0 and 24.0 <= y <= HEIGHT - 24.0) for x, y in landmarks):
            continue
        if any(math.dist(points[a], points[b]) < 42.0 for i, a in enumerate(COLORS) for b in COLORS[i + 1 :]):
            continue
        if any(math.dist(points[color], target) < 40.0 for color in COLORS):
            continue
        if any(math.dist(points[color], origin) < 54.0 for color in COLORS):
            continue

        draw_order = list(COLORS)
        rng.shuffle(draw_order)
        scene = {
            "width": WIDTH,
            "height": HEIGHT,
            "origin": [round(value, 4) for value in origin],
            "target": [round(value, 4) for value in target],
            "points": {
                color: [round(value, 4) for value in points[color]] for color in COLORS
            },
            "draw_order": draw_order,
        }
        if analytic_gold(scene) == winner and not is_quarantined(scene):
            return scene
    raise RuntimeError(f"could not sample a separated projection scene for seed {seed}")


def _scaled(point: list[float], scale: int) -> tuple[float, float]:
    return point[0] * scale, point[1] * scale


def render(scene: dict) -> Image.Image:
    scale = 4
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), RGB["background"])
    draw = ImageDraw.Draw(image)
    origin = scene["origin"]
    target = scene["target"]

    # A faint full line makes the one-dimensional projection axis visible.
    dx, dy = target[0] - origin[0], target[1] - origin[1]
    norm = math.hypot(dx, dy)
    ux, uy = dx / norm, dy / norm
    a = [origin[0] - 188.0 * ux, origin[1] - 188.0 * uy]
    b = [origin[0] + 188.0 * ux, origin[1] + 188.0 * uy]
    draw.line([_scaled(a, scale), _scaled(b, scale)], fill=RGB["guide"], width=2 * scale)

    # The magenta dashed ray and terminal diamond define positive direction.
    for start in range(14, int(norm) - 12, 18):
        end = min(start + 10, norm - 13)
        p0 = [origin[0] + start * ux, origin[1] + start * uy]
        p1 = [origin[0] + end * ux, origin[1] + end * uy]
        draw.line([_scaled(p0, scale), _scaled(p1, scale)], fill=RGB["magenta"], width=4 * scale)

    tx, ty = _scaled(target, scale)
    rr = TARGET_RADIUS * scale
    draw.polygon([(tx, ty - rr), (tx + rr, ty), (tx, ty + rr), (tx - rr, ty)], fill=RGB["magenta"])

    for color in scene["draw_order"]:
        x, y = _scaled(scene["points"][color], scale)
        r = DOT_RADIUS * scale
        draw.ellipse([x - r, y - r, x + r, y + r], fill=RGB[color])

    ox, oy = _scaled(origin, scale)
    r0 = ORIGIN_RADIUS * scale
    draw.ellipse([ox - r0, oy - r0, ox + r0, oy + r0], fill=RGB["black"])
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    values = _projection_values(scene)
    return max(values, key=values.get)


def margin(scene: dict) -> float:
    ranked = sorted(_projection_values(scene).values(), reverse=True)
    return float(ranked[0] - ranked[1])


def is_quarantined(scene: dict) -> bool:
    if margin(scene) < QUARANTINE_GAP:
        return True
    points = list(scene["points"].values())
    if min(math.dist(points[i], points[j]) for i in range(3) for j in range(i + 1, 3)) < 34.0:
        return True
    if min(math.dist(point, scene["origin"]) for point in points) < 48.0:
        return True
    if min(math.dist(point, scene["target"]) for point in points) < 34.0:
        return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["draw_order"] = list(reversed(twin["draw_order"]))
    if twin["draw_order"] == scene["draw_order"]:
        twin["draw_order"] = twin["draw_order"][1:] + twin["draw_order"][:1]
    return [("permute_disjoint_dot_draw_order", twin)]
