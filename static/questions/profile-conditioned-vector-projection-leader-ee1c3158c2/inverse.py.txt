from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

COLORS = ("blue", "orange", "green")
RGB = {
    "blue": (31, 119, 180),
    "orange": (245, 125, 32),
    "green": (44, 160, 70),
    "magenta": (190, 45, 145),
    "black": (34, 36, 42),
}


def _mask(array: np.ndarray, color: str, tolerance: float = 52.0) -> np.ndarray:
    target = np.asarray(RGB[color], dtype=np.float32)
    return np.linalg.norm(array.astype(np.float32) - target, axis=2) <= tolerance


def _centroid(mask: np.ndarray, minimum: int) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if len(xs) < minimum:
        raise ValueError("required colored landmark is absent")
    return float(xs.mean()), float(ys.mean())


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components: list[np.ndarray] = []
    for y, x in zip(*np.nonzero(mask), strict=True):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        pixels: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            pixels.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(pixels) >= 8:
            components.append(np.asarray(pixels, dtype=np.float64))
    return components


def _magenta_target(array: np.ndarray) -> tuple[float, float]:
    components = _components(_mask(array, "magenta"))
    if not components:
        raise ValueError("magenta direction landmark is absent")
    # The terminal diamond is substantially larger than every separated dash.
    diamond = max(components, key=len)
    if len(diamond) < 180:
        raise ValueError("magenta terminal diamond is corrupted")
    return float(diamond[:, 1].mean()), float(diamond[:, 0].mean())


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    origin = _centroid(_mask(array, "black"), 120)
    target = _magenta_target(array)
    dx, dy = target[0] - origin[0], target[1] - origin[1]
    norm = math.hypot(dx, dy)
    if norm < 70.0:
        raise ValueError("reference direction is unresolved")
    ux, uy = dx / norm, dy / norm

    values: dict[str, float] = {}
    for color in COLORS:
        px, py = _centroid(_mask(array, color), 180)
        values[color] = (px - origin[0]) * ux + (py - origin[1]) * uy
    ranked = sorted(values.items(), key=lambda item: item[1], reverse=True)
    if ranked[0][1] - ranked[1][1] < 6.0:
        raise ValueError("projection ranking is raster-ambiguous")
    return ranked[0][0]
