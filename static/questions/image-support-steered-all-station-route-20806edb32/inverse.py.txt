"""Independent image-only inverse arm for the station graph."""

from __future__ import annotations

from collections import deque
import math
import numpy as np
from PIL import Image

START = np.array((232, 91, 42), dtype=np.int16)
END = np.array((40, 111, 205), dtype=np.int16)
STATION = np.array((244, 190, 44), dtype=np.int16)
TRACK = np.array((74, 82, 91), dtype=np.int16)


def _mask(array: np.ndarray, color: np.ndarray, tolerance: int = 18) -> np.ndarray:
    # int32 prevents squared RGB distances from wrapping at 32767.
    delta = array.astype(np.int32) - color.astype(np.int32)
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = deque([(int(y), int(x))])
        group = []
        while queue:
            cy, cx = queue.popleft()
            group.append((cy, cx))
            for ny, nx in ((cy-1,cx),(cy+1,cx),(cy,cx-1),(cy,cx+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(group) >= 40:
            groups.append(group)
    return groups


def _centroids(mask: np.ndarray) -> list[tuple[float, float]]:
    result = []
    for group in _components(mask):
        ys = [p[0] for p in group]
        xs = [p[1] for p in group]
        result.append((sum(xs) / len(xs), sum(ys) / len(ys)))
    return result


def _near_segment(point, a, b) -> tuple[float, float]:
    px, py = point
    ax, ay = a
    bx, by = b
    dx, dy = bx-ax, by-ay
    denom = dx*dx + dy*dy
    if denom == 0:
        return math.dist(point, a), 0.0
    t = ((px-ax)*dx + (py-ay)*dy) / denom
    t_clip = max(0.0, min(1.0, t))
    q = (ax + t_clip*dx, ay + t_clip*dy)
    return math.dist(point, q), t


def _edge_visible(track: np.ndarray, a, b, all_centers) -> bool:
    distance = math.dist(a, b)
    if distance < 55 or distance > 135:
        return False
    for c in all_centers:
        if c == a or c == b:
            continue
        gap, t = _near_segment(c, a, b)
        if 0.12 < t < 0.88 and gap < 25:
            return False
    samples = 41
    hits = 0
    for k in range(samples):
        t = 0.20 + 0.60 * k / (samples - 1)
        x = int(round(a[0] + t * (b[0] - a[0])))
        y = int(round(a[1] + t * (b[1] - a[1])))
        y0, y1 = max(0, y-4), min(track.shape[0], y+5)
        x0, x1 = max(0, x-4), min(track.shape[1], x+5)
        hits += bool(track[y0:y1, x0:x1].any())
    return hits / samples >= 0.88


def _hamilton(adj: list[set[int]], start: int, end: int) -> bool:
    n = len(adj)
    def visit(v: int, used: int) -> bool:
        if v == end:
            return used.bit_count() == n
        for w in adj[v]:
            bit = 1 << w
            if not used & bit and visit(w, used | bit):
                return True
        return False
    return visit(start, 1 << start)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    starts = _centroids(_mask(array, START))
    ends = _centroids(_mask(array, END))
    ordinary = _centroids(_mask(array, STATION))
    if len(starts) != 1 or len(ends) != 1 or len(ordinary) != 5:
        raise ValueError("expected one orange, one blue, and five yellow stations")
    centers = starts + ends + ordinary
    track = _mask(array, TRACK, tolerance=20)
    adj = [set() for _ in centers]
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            if _edge_visible(track, centers[i], centers[j], centers):
                adj[i].add(j)
                adj[j].add(i)
    if sum(len(v) for v in adj) < 12 or any(not v for v in adj):
        raise ValueError("visible tracks do not form a robust connected station graph")
    return "yes" if _hamilton(adj, 0, 1) else "no"
