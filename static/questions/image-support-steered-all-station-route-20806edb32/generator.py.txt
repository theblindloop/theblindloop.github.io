"""Deterministic station-graph sampler and renderer."""

from __future__ import annotations

import itertools
import math
import random
from PIL import Image, ImageDraw

BACKGROUND = (250, 248, 242)
TRACK = (74, 82, 91)
START = (232, 91, 42)
END = (40, 111, 205)
STATION = (244, 190, 44)
OUTLINE = (32, 36, 40)


def _connected(n: int, edges: list[list[int]]) -> bool:
    adj = [set() for _ in range(n)]
    for a, b in edges:
        adj[a].add(b)
        adj[b].add(a)
    # The compact explicit traversal avoids reliance on any graph package.
    seen = {0}
    stack = [0]
    while stack:
        v = stack.pop()
        for w in adj[v]:
            if w not in seen:
                seen.add(w)
                stack.append(w)
    return len(seen) == n


def _has_spanning_path(n: int, edges: list[list[int]], start: int, end: int) -> bool:
    adj = [set() for _ in range(n)]
    for a, b in edges:
        adj[a].add(b)
        adj[b].add(a)

    def visit(v: int, used: int) -> bool:
        if v == end:
            return used.bit_count() == n
        for w in adj[v]:
            bit = 1 << w
            if not used & bit and visit(w, used | bit):
                return True
        return False

    return visit(start, 1 << start)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = "yes" if int(seed) % 2 == 0 else "no"
    # A shallow regional corridor makes the connected railway visibly elongated
    # while leaving generous margins to every image edge.
    lattice = [(0, 0), (1, 0), (2, 0), (3, 0), (0, 1), (1, 1), (2, 1), (3, 1)]
    for _ in range(20000):
        occupied = sorted(rng.sample(range(8), 7))
        index = {cell: i for i, cell in enumerate(occupied)}
        pool = []
        for cell, i in index.items():
            x, y = lattice[cell]
            for neighbor in ((x + 1, y), (x, y + 1)):
                if neighbor in lattice:
                    key = lattice.index(neighbor)
                    if key in index:
                        pool.append([i, index[key]])
        if len(pool) < 7:
            continue
        edges = sorted(rng.sample(pool, 7))
        if not _connected(7, edges):
            continue
        roles = [0, 1]
        rng.shuffle(roles)
        # Fully randomize terminal identities over station positions.
        start, end = rng.sample(range(7), 2)
        roles = [start, end]
        decision = "yes" if _has_spanning_path(7, edges, start, end) else "no"
        if decision != target:
            continue
        jitter = [(rng.randint(-7, 7), rng.randint(-7, 7)) for _ in occupied]
        vertices = []
        for cell, (jx, jy) in zip(occupied, jitter):
            x, y = lattice[cell]
            vertices.append([115 + x * 94 + jx, 196 + y * 106 + jy])
        return {
            "vertices": vertices,
            "edges": edges,
            "roles": roles,
            "canvas": [512, 500],
            "station_radius": 15,
            "track_width": 10,
        }
    raise RuntimeError("could not sample a balanced robust graph")


def render(scene: dict) -> Image.Image:
    width, height = scene["canvas"]
    radius = int(scene["station_radius"])
    track_width = int(scene["track_width"])
    vertices = [tuple(v) for v in scene["vertices"]]
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for a, b in scene["edges"]:
        draw.line([vertices[a], vertices[b]], fill=TRACK, width=track_width)
    start, end = scene["roles"]
    for i, (x, y) in enumerate(vertices):
        color = START if i == start else END if i == end else STATION
        draw.ellipse((x-radius, y-radius, x+radius, y+radius), fill=color, outline=OUTLINE, width=3)
    return image


def analytic_gold(scene: dict) -> str:
    edges = scene["edges"]
    start, end = scene["roles"]
    n = 1 + max(max(edge) for edge in edges)
    return "yes" if _has_spanning_path(n, edges, start, end) else "no"


def margin(scene: dict) -> float:
    vertices = scene["vertices"]
    radius = float(scene["station_radius"])
    width = float(scene["track_width"])
    lengths = [math.dist(vertices[a], vertices[b]) for a, b in scene["edges"]]
    return min(min(lengths) - 2 * radius, 2 * radius - width)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 18.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
