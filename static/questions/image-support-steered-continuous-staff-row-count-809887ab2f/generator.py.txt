"""Deterministic scene sampler and raster renderer for a continuous music staff."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 768, 512
SCALE = 3
STAFF_COLORS = ("#173a63", "#185f69", "#5b2d64", "#303642")
BACKGROUNDS = ("#fffdf7", "#f7fbff", "#fff8fb")
NOTE_COLORS = ("#d88a18", "#d05a4e", "#3f83c5")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    # Cycling makes every 24-scene stress set balanced while style remains seed-random.
    row_count = 3 + (int(seed) % 5)
    stroke_width = rng.choice((6, 7, 8, 9))
    tight = int(seed) % 11 == 0
    row_gap = (stroke_width + rng.choice((4, 5, 6))) if tight else rng.choice((30, 32, 34))
    total_h = (row_count - 1) * row_gap
    top_y = rng.randint(112, max(112, HEIGHT - 86 - total_h))
    x0 = rng.randint(46, 86)
    x1 = rng.randint(682, 722)
    return {
        "row_count": row_count,
        "x0": x0,
        "x1": x1,
        "top_y": top_y,
        "row_gap": row_gap,
        "stroke_width": stroke_width,
        "staff_color": rng.choice(STAFF_COLORS),
        "background_color": rng.choice(BACKGROUNDS),
        "note_phase": rng.randrange(4),
    }


def _route_points(scene: dict) -> list[tuple[int, int]]:
    n = int(scene["row_count"])
    x0, x1 = int(scene["x0"]), int(scene["x1"])
    top, gap = int(scene["top_y"]), int(scene["row_gap"])
    points: list[tuple[int, int]] = []
    for i in range(n):
        y = top + i * gap
        if i % 2 == 0:
            a, b = (x0, y), (x1, y)
        else:
            a, b = (x1, y), (x0, y)
        if not points:
            points.append(a)
        points.append(b)
        if i + 1 < n:
            points.append((b[0], y + gap))
    return points


def render(scene: dict) -> Image.Image:
    # Explicit access to every declared field keeps the rendered-field audit honest.
    n = int(scene["row_count"])
    x0, x1 = int(scene["x0"]), int(scene["x1"])
    top, gap = int(scene["top_y"]), int(scene["row_gap"])
    sw = int(scene["stroke_width"])
    staff_color = str(scene["staff_color"])
    background = str(scene["background_color"])
    phase = int(scene["note_phase"])

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), background)
    draw = ImageDraw.Draw(image)
    r = max(7, sw)
    # Colored noteheads establish the musical prior but are separated from and
    # ignored by the dark-stroke oracle. Their count and placement do not encode n.
    for j in range(3):
        frac = (j + 1) / 4
        cx = int(x0 + frac * (x1 - x0) + ((phase + j) % 3 - 1) * 17)
        band = (phase + 2 * j) % max(1, n - 1)
        cy = top + band * gap + gap // 2
        box = ((cx-r)*SCALE, (cy-r)*SCALE, (cx+r)*SCALE, (cy+r)*SCALE)
        draw.ellipse(box, fill=NOTE_COLORS[j])
        draw.line(((cx+r)*SCALE, cy*SCALE, (cx+r)*SCALE, (cy-24)*SCALE),
                  fill=NOTE_COLORS[j], width=3*SCALE)
    # Draw the decisive route last so decorative noteheads can never occlude or
    # split it; crossings retain the staff ink and its image-only connectivity.
    pts = [(x * SCALE, y * SCALE) for x, y in _route_points(scene)]
    draw.line(pts, fill=staff_color, width=sw * SCALE, joint="curve")
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(int(scene["row_count"]))


def margin(scene: dict) -> float:
    return float(int(scene["row_gap"]) - int(scene["stroke_width"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
