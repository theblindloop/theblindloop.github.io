"""Image-only inverse arm for the continuous staff.

It knows only the public raster convention: the staff is the large dark,
low-luminance connected trace. It receives no latent scene or gold state.
"""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[tuple[int, np.ndarray, np.ndarray]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        q = deque([(int(y), int(x))])
        seen[y, x] = True
        ys, xs = [], []
        while q:
            yy, xx = q.popleft()
            ys.append(yy); xs.append(xx)
            for ny, nx in ((yy-1,xx),(yy+1,xx),(yy,xx-1),(yy,xx+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    q.append((ny, nx))
        out.append((len(ys), np.asarray(ys), np.asarray(xs)))
    return out


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    # Dark staff inks have low maximum channel; saturated note colors do not.
    mask = arr.max(axis=2) < 135
    comps = [c for c in _components(mask) if c[0] >= 300]
    if len(comps) != 1:
        return "abstain"
    area, ys, xs = comps[0]
    span_x = int(xs.max() - xs.min() + 1)
    span_y = int(ys.max() - ys.min() + 1)
    if span_x < int(image.width * 0.70) or span_y < 20 or area < 1500:
        return "abstain"
    sub = mask[int(ys.min()):int(ys.max())+1, int(xs.min()):int(xs.max())+1]
    # A counted traversal occupies most of the component's horizontal span.
    long_y = sub.sum(axis=1) >= 0.72 * span_x
    groups = []
    start = None
    for idx, value in enumerate(np.r_[long_y, False]):
        if value and start is None:
            start = idx
        elif not value and start is not None:
            groups.append((start, idx - 1))
            start = None
    if not 3 <= len(groups) <= 7 or any(b - a + 1 < 3 for a, b in groups):
        return "abstain"
    gaps = [groups[i+1][0] - groups[i][1] - 1 for i in range(len(groups)-1)]
    if gaps and min(gaps) < 3:
        return "abstain"
    return str(len(groups))
