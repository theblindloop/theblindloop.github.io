"""Renderer for chain_free_ring.

Five coloured rings sit on a closed necklace layout, each ring overlapping
exactly two neighbours in two points. At every crossing one ring is drawn with
a short break, meaning it passes behind the other. A pair of rings interlocks
when the break alternates (each ring is in front at one of the two crossings).
Exactly one ring never interlocks: at both crossings with each of its two
neighbours the same ring is in front, so that ring is only lying on top of or
underneath the chain and could be lifted away. The decision is the colour of
that ring.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "chain_free_ring-renderer-0.1.0"

CANVAS = 450
SS = 3
STROKE = 7
BACKGROUND = (248, 247, 243)
N_RINGS = 5

PALETTE: dict[str, tuple[int, int, int]] = {
    "blue": (33, 82, 196),
    "brown": (121, 76, 40),
    "green": (22, 128, 60),
    "orange": (230, 126, 20),
    "purple": (124, 52, 172),
    "red": (198, 40, 44),
}
COLOR_NAMES = tuple(sorted(PALETTE))
WARM_COLORS = ("brown", "orange", "red")

QUARANTINE_PX = 5.6


# --------------------------------------------------------------------------
# geometry helpers
# --------------------------------------------------------------------------
def _circle_intersections(
    c1: tuple[float, float], r1: float, c2: tuple[float, float], r2: float
) -> list[tuple[float, float]] | None:
    dx, dy = c2[0] - c1[0], c2[1] - c1[1]
    d = math.hypot(dx, dy)
    if d >= r1 + r2 or d <= abs(r1 - r2) or d == 0.0:
        return None
    a = (r1 * r1 - r2 * r2 + d * d) / (2.0 * d)
    h2 = r1 * r1 - a * a
    if h2 <= 0.0:
        return None
    h = math.sqrt(h2)
    bx, by = c1[0] + a * dx / d, c1[1] + a * dy / d
    ux, uy = -dy / d, dx / d
    return [(bx + h * ux, by + h * uy), (bx - h * ux, by - h * uy)]


def _cross_angle(c1: tuple[float, float], r1: float, c2: tuple[float, float], r2: float) -> float:
    """Angle between the two circles' tangents at their intersection, degrees."""
    d = math.hypot(c2[0] - c1[0], c2[1] - c1[1])
    cosg = (r1 * r1 + r2 * r2 - d * d) / (2.0 * r1 * r2)
    cosg = max(-1.0, min(1.0, cosg))
    gamma = math.degrees(math.acos(cosg))
    return 180.0 - gamma


def _ring_dist(point: tuple[float, float], ring: dict) -> float:
    """Distance from a point to a ring's centre line."""
    return abs(math.hypot(point[0] - ring["cx"], point[1] - ring["cy"]) - ring["r"])


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------
def _attempt(rng: random.Random) -> dict | None:
    free_index = rng.randrange(N_RINGS)
    colors = rng.sample(COLOR_NAMES, N_RINGS)
    gap_half = rng.choice([rng.uniform(20.0, 24.0)] * 8 + [rng.uniform(15.5, 17.5)])
    spin = rng.uniform(0.0, 360.0)

    rings: list[dict] = []
    for i in range(N_RINGS):
        ang = math.radians(spin + 72.0 * i + rng.uniform(-6.0, 6.0))
        rad = rng.uniform(108.0, 116.0)
        rings.append(
            {
                "cx": CANVAS / 2.0 + rad * math.cos(ang),
                "cy": CANVAS / 2.0 + rad * math.sin(ang),
                "r": rad * rng.uniform(0.665, 0.700),
                "color": colors[i],
            }
        )

    for ring in rings:
        reach = ring["r"] + STROKE
        if not (
            reach + 6 < ring["cx"] < CANVAS - reach - 6
            and reach + 6 < ring["cy"] < CANVAS - reach - 6
        ):
            return None

    pairs: list[dict] = []
    for i in range(N_RINGS):
        j = (i + 1) % N_RINGS
        a, b = rings[i], rings[j]
        pts = _circle_intersections((a["cx"], a["cy"]), a["r"], (b["cx"], b["cy"]), b["r"])
        if pts is None:
            return None
        d = math.hypot(b["cx"] - a["cx"], b["cy"] - a["cy"])
        if a["r"] + b["r"] - d < 15.0 or d - abs(a["r"] - b["r"]) < 24.0:
            return None
        if _cross_angle((a["cx"], a["cy"]), a["r"], (b["cx"], b["cy"]), b["r"]) < 50.0:
            return None
        interlocked = i != free_index and j != free_index
        if interlocked:
            first = i if rng.random() < 0.5 else j
            under = [first, j if first == i else i]
        else:
            top = i if rng.random() < 0.5 else j
            other = j if top == i else i
            under = [other, other]
        pairs.append(
            {
                "a": i,
                "b": j,
                "points": [[pts[0][0], pts[0][1]], [pts[1][0], pts[1][1]]],
                "under": under,
                "interlocked": interlocked,
            }
        )

    # non-adjacent rings must stay clear of each other
    for i in range(N_RINGS):
        for j in range(i + 1, N_RINGS):
            if j == (i + 1) % N_RINGS or i == (j + 1) % N_RINGS:
                continue
            d = math.hypot(rings[j]["cx"] - rings[i]["cx"], rings[j]["cy"] - rings[i]["cy"])
            if d - (rings[i]["r"] + rings[j]["r"]) < 12.0:
                return None

    # every crossing must stand well apart from every other crossing
    all_points = [tuple(p) for pair in pairs for p in pair["points"]]
    for u in range(len(all_points)):
        for v in range(u + 1, len(all_points)):
            if math.dist(all_points[u], all_points[v]) < 2.0 * gap_half + 8.0:
                return None

    scene = {
        "canvas": CANVAS,
        "background": list(BACKGROUND),
        "stroke": STROKE,
        "gap_half_len": gap_half,
        "rings": rings,
        "pairs": pairs,
        "free_index": free_index,
    }
    if _clearances(scene) < 4.6:
        return None
    return scene


def sample_scene(seed: int) -> dict:
    rng = random.Random((int(seed) * 7919 + 104729) & 0xFFFFFFFF)
    for _ in range(6000):
        scene = _attempt(rng)
        if scene is not None:
            scene["seed"] = int(seed)
            return scene
    raise RuntimeError(f"could not sample a chain_free_ring scene for seed {seed}")


# --------------------------------------------------------------------------
# gold, margin, quarantine
# --------------------------------------------------------------------------
def analytic_gold(scene: dict) -> str:
    """Colour of the single ring that interlocks with neither neighbour."""
    linked = {i: False for i in range(len(scene["rings"]))}
    for pair in scene["pairs"]:
        if pair["under"][0] != pair["under"][1]:
            linked[pair["a"]] = True
            linked[pair["b"]] = True
    free = [i for i, is_linked in linked.items() if not is_linked]
    if len(free) != 1:
        raise ValueError("scene does not contain exactly one free ring")
    return str(scene["rings"][free[0]]["color"])


def _clearances(scene: dict) -> float:
    """Smallest decision-critical background clearance in pixels."""
    rings = scene["rings"]
    gap_half = float(scene["gap_half_len"])
    stroke = float(scene["stroke"])
    worst = 1e9
    for pair in scene["pairs"]:
        for t, point in enumerate(pair["points"]):
            under = rings[pair["under"][t]]
            over = rings[pair["b"] if pair["under"][t] == pair["a"] else pair["a"]]
            phi = math.atan2(point[1] - under["cy"], point[0] - under["cx"])
            alpha = gap_half / under["r"]
            for sign in (-1.0, 1.0):
                ang = phi + sign * alpha
                end = (
                    under["cx"] + under["r"] * math.cos(ang),
                    under["cy"] + under["r"] * math.sin(ang),
                )
                worst = min(worst, _ring_dist(end, over) - stroke)
            for k, other in enumerate(rings):
                if k in (pair["a"], pair["b"]):
                    continue
                worst = min(worst, _ring_dist(tuple(point), other) - stroke)
    return worst


def margin(scene: dict) -> float:
    return round(float(_clearances(scene)), 6)


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_PX)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases: every label input is drawn, so this list is empty."""
    return []


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def _omit_windows(scene: dict, index: int) -> list[tuple[float, float]]:
    ring = scene["rings"][index]
    gap_half = float(scene["gap_half_len"])
    half_deg = math.degrees(gap_half / ring["r"])
    windows = []
    for pair in scene["pairs"]:
        if index not in (pair["a"], pair["b"]):
            continue
        for t, point in enumerate(pair["points"]):
            if pair["under"][t] != index:
                continue
            phi = math.degrees(math.atan2(point[1] - ring["cy"], point[0] - ring["cx"])) % 360.0
            windows.append((phi - half_deg, phi + half_deg))
    return windows


def _allowed_arcs(windows: list[tuple[float, float]]) -> list[tuple[float, float]]:
    if not windows:
        return [(0.0, 360.0)]
    ordered = sorted(windows, key=lambda w: w[0] % 360.0)
    arcs = []
    for idx, (_, end) in enumerate(ordered):
        nxt_start = ordered[(idx + 1) % len(ordered)][0]
        start = end % 360.0
        stop = nxt_start % 360.0
        if stop <= start:
            stop += 360.0
        if stop - start > 0.4:
            arcs.append((start, stop))
    return arcs


def render(scene: dict) -> Image.Image:
    size = CANVAS * SS
    img = Image.new("RGB", (size, size), tuple(scene["background"]))
    draw = ImageDraw.Draw(img)
    width = int(round(scene["stroke"] * SS))
    for index, ring in enumerate(scene["rings"]):
        color = PALETTE[ring["color"]]
        cx, cy, r = ring["cx"] * SS, ring["cy"] * SS, ring["r"] * SS
        for start, stop in _allowed_arcs(_omit_windows(scene, index)):
            steps = max(3, int(math.ceil((stop - start) / 1.2)))
            pts = []
            for s in range(steps + 1):
                ang = math.radians(start + (stop - start) * s / steps)
                pts.append((cx + r * math.cos(ang), cy + r * math.sin(ang)))
            draw.line(pts, fill=color, width=width, joint="curve")
    return img.resize((CANVAS, CANVAS), Image.BOX)
