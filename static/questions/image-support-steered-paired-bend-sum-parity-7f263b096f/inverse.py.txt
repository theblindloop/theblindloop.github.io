"""Independent image-only inverse arm."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

_NAVY = np.array((27, 54, 93), dtype=np.int16)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    seen = np.zeros(mask.shape, dtype=bool)
    out = []
    h, w = mask.shape
    for yy, xx in np.argwhere(mask):
        y, x = int(yy), int(xx)
        if seen[y, x]:
            continue
        q = deque([(y, x)])
        seen[y, x] = True
        pts = []
        while q:
            cy, cx = q.popleft()
            pts.append((cy, cx))
            for ny, nx in ((cy-1,cx),(cy+1,cx),(cy,cx-1),(cy,cx+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    q.append((ny, nx))
        if len(pts) > 150:
            out.append(np.asarray(pts, dtype=int))
    return out


def _groups(values: np.ndarray) -> int:
    indices = np.flatnonzero(values)
    if len(indices) == 0:
        return 0
    return 1 + int(np.sum(np.diff(indices) > 1))


def _has_long_run(lines: np.ndarray, threshold: int = 32) -> np.ndarray:
    """For each row, report whether it contains a long contiguous True run."""
    result = []
    for line in lines:
        padded = np.pad(line.astype(np.int8), (1, 1))
        edges = np.diff(padded)
        starts = np.flatnonzero(edges == 1)
        ends = np.flatnonzero(edges == -1)
        result.append(bool(len(starts) and np.max(ends - starts) >= threshold))
    return np.asarray(result, dtype=bool)


def _long_run_mask(lines: np.ndarray, threshold: int = 32) -> np.ndarray:
    """Keep pixels belonging to long contiguous runs within each row."""
    output = np.zeros_like(lines, dtype=bool)
    for index, line in enumerate(lines):
        padded = np.pad(line.astype(np.int8), (1, 1))
        edges = np.diff(padded)
        for start, end in zip(np.flatnonzero(edges == 1), np.flatnonzero(edges == -1)):
            if end - start >= threshold:
                output[index, start:end] = True
    return output


def _bend_count(points: np.ndarray) -> int:
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    local = np.zeros((y1-y0+1, x1-x0+1), dtype=bool)
    local[points[:,0]-y0, points[:,1]-x0] = True
    # Long row/column bands are the visible horizontal/vertical directional runs.
    horizontal = len(_components(_long_run_mask(local)))
    vertical = len(_components(_long_run_mask(local.T)))
    runs = horizontal + vertical
    if runs < 3 or runs > 7:
        raise ValueError("route directional runs are ambiguous")
    return runs - 1


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    mask = np.max(np.abs(arr - _NAVY), axis=2) <= 3
    components = _components(mask)
    if len(components) != 2:
        raise ValueError("expected two disconnected navy routes")
    bends = [_bend_count(points) for points in components]
    return "odd" if sum(bends) % 2 else "even"
