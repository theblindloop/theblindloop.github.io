"""Deterministic latent sampler and renderer for paired bend-sum parity."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 1000, 360
NAVY = (27, 54, 93)
GREEN = (27, 158, 119)
RED = (216, 71, 67)
PALE = (221, 226, 232)
INK = (43, 48, 55)
BG = (250, 248, 242)
STROKE = 13


def _route_points(rng: random.Random, panel_x: int, bends: int) -> list[list[int]]:
    """Make an alternating-axis, non-self-touching route in one panel."""
    left, right = panel_x + 48, panel_x + 312
    top = 60 + rng.randrange(-8, 9)
    bottom = 292 + rng.randrange(-8, 9)
    ys = [round(top + i * (bottom - top) / ((bends + 2) // 2)) for i in range((bends + 2) // 2 + 1)]
    x_lo = left + rng.randrange(0, 25)
    x_hi = right - rng.randrange(0, 25)
    pts = [[x_lo, ys[0]]]
    horizontal_right = bool(rng.randrange(2))
    y_index = 0
    axis_horizontal = True
    for _ in range(bends + 1):
        x, y = pts[-1]
        if axis_horizontal:
            nx = x_hi if horizontal_right else x_lo
            if nx == x:
                nx = x_lo if horizontal_right else x_hi
            pts.append([nx, y])
            horizontal_right = not horizontal_right
        else:
            y_index += 1
            pts.append([x, ys[y_index]])
        axis_horizontal = not axis_horizontal
    if rng.randrange(2):
        # Reflect vertically inside the visible panel; bend count and clearances stay fixed.
        pts = [[x, HEIGHT - y] for x, y in pts]
    return pts


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    # Cycling the total across consecutive seeds guarantees both parity classes.
    options = ((2, 3), (2, 4), (3, 4), (3, 5), (4, 5), (4, 4))
    b1, b2 = options[int(seed) % len(options)]
    if rng.randrange(2):
        b1, b2 = b2, b1
    routes = [
        {"points": _route_points(rng, 15, b1), "bends": b1},
        {"points": _route_points(rng, 615, b2), "bends": b2},
    ]
    decorations = []
    for panel_x in (15, 615):
        for _ in range(3):
            decorations.append([
                panel_x + rng.randrange(28, 335),
                rng.randrange(25, 335),
                rng.choice([4, 5, 6]),
            ])
    return {"routes": routes, "decorations": decorations}


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((14, 18, 384, 342), radius=18, outline=PALE, width=3)
    draw.rounded_rectangle((614, 18, 984, 342), radius=18, outline=PALE, width=3)
    for x, y, r in scene["decorations"]:
        draw.ellipse((x-r, y-r, x+r, y+r), outline=PALE, width=2)
    for route in scene["routes"]:
        points = [tuple(p) for p in route["points"]]
        draw.line(points, fill=NAVY, width=STROKE, joint="curve")
        # Endpoint rings expose the two terminals but keep the navy centerline visible.
        for color, (x, y) in ((GREEN, points[0]), (RED, points[-1])):
            draw.ellipse((x-11, y-11, x+11, y+11), outline=color, width=4)
    draw.text((426, 8), "BOTH ROUTES", fill=INK)
    return image


def analytic_gold(scene: dict) -> str:
    total = sum(int(route["bends"]) for route in scene["routes"])
    return "odd" if total % 2 else "even"


def margin(scene: dict) -> float:
    values = []
    for route in scene["routes"]:
        pts = route["points"]
        values.extend(abs(a[i] - b[i]) for a, b in zip(pts, pts[1:]) for i in (0, 1) if a[i] != b[i])
    return float(min(values) - 2 * STROKE)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 18.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
