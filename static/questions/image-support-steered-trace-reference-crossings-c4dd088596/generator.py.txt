from __future__ import annotations

import random
from PIL import Image, ImageDraw

SCALE = 3


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    width, height = 512, 384
    x0 = rng.randint(88, 108)
    step = rng.randint(48, 53)
    xs = [x0 + i * step for i in range(7)]
    baseline_y = rng.randint(184, 200)
    crossings = 2 + (int(seed) % 4)
    flip_slots = set(rng.sample(range(6), crossings))
    sign = -1 if rng.random() < 0.5 else 1
    offsets = []
    for i in range(7):
        magnitude = rng.randint(30, 50)
        offsets.append(sign * magnitude)
        if i < 6 and i in flip_slots:
            sign *= -1
    return {
        "width": width,
        "height": height,
        "xs": xs,
        "baseline_y": baseline_y,
        "offsets": offsets,
        "route_color": [12, 151, 181],
        "axis_color": [65, 72, 78],
        "background": [248, 247, 242],
    }


def _crossing_positions(scene: dict) -> list[float]:
    xs = scene["xs"]
    offsets = scene["offsets"]
    out = []
    for x1, x2, a, b in zip(xs, xs[1:], offsets, offsets[1:]):
        if (a < 0 < b) or (b < 0 < a):
            out.append(x1 + (x2 - x1) * abs(a) / (abs(a) + abs(b)))
    return out


def analytic_gold(scene: dict) -> str:
    offsets = scene["offsets"]
    return str(sum((a < 0 < b) or (b < 0 < a) for a, b in zip(offsets, offsets[1:])))


def margin(scene: dict) -> float:
    crossings = _crossing_positions(scene)
    node_clearance = min(abs(v) for v in scene["offsets"])
    separation = min((b - a for a, b in zip(crossings, crossings[1:])), default=999.0)
    return float(min(node_clearance, separation / 2.0))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 18.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    w, h = int(scene["width"]), int(scene["height"])
    bg = tuple(scene["background"])
    image = Image.new("RGB", (w * SCALE, h * SCALE), bg)
    draw = ImageDraw.Draw(image)
    axis = tuple(scene["axis_color"])
    cyan = tuple(scene["route_color"])
    y0 = int(scene["baseline_y"] * SCALE)
    left, right = 72 * SCALE, 440 * SCALE
    dash, gap = 16 * SCALE, 10 * SCALE
    x = left
    while x < right:
        draw.line([(x, y0), (min(x + dash, right), y0)], fill=axis, width=3 * SCALE)
        x += dash + gap
    # Sparse pale guide lines establish a measurement-plot context without
    # becoming decision evidence.
    guide = (221, 222, 217)
    for gx in (128, 192, 256, 320, 384):
        draw.line([(gx * SCALE, 68 * SCALE), (gx * SCALE, 316 * SCALE)], fill=guide, width=SCALE)
    points = [
        (int(x * SCALE), int((scene["baseline_y"] + off) * SCALE))
        for x, off in zip(scene["xs"], scene["offsets"])
    ]
    draw.line(points, fill=cyan, width=8 * SCALE, joint="curve")
    for p in points:
        r = 4 * SCALE
        draw.ellipse([p[0] - r, p[1] - r, p[0] + r, p[1] + r], fill=cyan)
    return image.resize((w, h), Image.Resampling.LANCZOS).convert("RGB")
