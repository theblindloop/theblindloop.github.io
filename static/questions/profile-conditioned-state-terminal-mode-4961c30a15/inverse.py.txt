"""Independent image-only inverse for terminal state recurrence."""

from __future__ import annotations

from PIL import Image


LABELS = ("A", "B", "C", "D")
STATE_COUNT = 3
PANEL_COUNT = 6
TILE_COUNT = 4
CANVAS = (1560, 900)
PANEL_LEFTS = tuple(20 + index * 256 for index in range(PANEL_COUNT))
PANEL_WIDTH = 240
ROW_Y = (320, 440, 560, 680)
KEY_Y = 108
LABEL_KEY_X = (58, 178, 298, 418)
STATE_KEY_X = (760, 880, 1000)
TILE_LEFT_INSET = 10
LABEL_CENTER_OFFSET = 37
MARKER_CENTER_OFFSET = 156


def _read_keyed_palettes(frame: Image.Image) -> tuple[dict[tuple[int, int, int], int], dict[tuple[int, int, int], int]]:
    pixels = frame.load()
    label_palette: dict[tuple[int, int, int], int] = {}
    for identity, x in enumerate(LABEL_KEY_X):
        color = pixels[x, KEY_Y]
        if color in label_palette:
            raise ValueError("identity key reuses one color")
        label_palette[color] = identity
    state_palette: dict[tuple[int, int, int], int] = {}
    for state, x in enumerate(STATE_KEY_X):
        color = pixels[x, KEY_Y]
        if color in state_palette or color in label_palette:
            raise ValueError("state key has an unreadable or reused color")
        state_palette[color] = state
    return label_palette, state_palette


def _decode_states(frame: Image.Image) -> list[list[int]]:
    pixels = frame.load()
    label_palette, state_palette = _read_keyed_palettes(frame)
    trajectories: list[list[int] | None] = [None] * TILE_COUNT
    for panel, panel_left in enumerate(PANEL_LEFTS):
        seen: set[int] = set()
        for row, y in enumerate(ROW_Y):
            tile_left = panel_left + TILE_LEFT_INSET
            identity_color = pixels[tile_left + LABEL_CENTER_OFFSET, y]
            state_color = pixels[tile_left + MARKER_CENTER_OFFSET, y]
            if identity_color not in label_palette:
                raise ValueError(f"identity tile P{panel + 1}, row {row + 1} is unreadable")
            if state_color not in state_palette:
                raise ValueError(f"state tile P{panel + 1}, row {row + 1} is unreadable")
            identity = label_palette[identity_color]
            state = state_palette[state_color]
            if identity in seen or trajectories[identity] is not None and len(trajectories[identity]) == panel + 1:
                raise ValueError(f"duplicate identity in panel P{panel + 1}")
            seen.add(identity)
            if trajectories[identity] is None:
                trajectories[identity] = [state]
            else:
                trajectories[identity].append(state)
        if seen != set(range(TILE_COUNT)):
            raise ValueError(f"panel P{panel + 1} does not expose all identities")
    if any(trajectory is None or len(trajectory) != PANEL_COUNT for trajectory in trajectories):
        raise ValueError("identity trajectories are incomplete")
    return [list(trajectory) for trajectory in trajectories if trajectory is not None]


def decision_from_image(image: Image.Image) -> str:
    """Decode only keyed pixels in the final PNG and compare terminal recurrence counts."""

    frame = image.convert("RGB")
    if frame.size != CANVAS:
        raise ValueError(f"unexpected raster size {frame.size}")
    trajectories = _decode_states(frame)
    scores = [sum(int(value == trajectory[-1]) for value in trajectory) for trajectory in trajectories]
    best = max(scores)
    winners = [index for index, score in enumerate(scores) if score == best]
    if len(winners) != 1:
        raise ValueError("terminal recurrence has no unique winner")
    return LABELS[winners[0]]
