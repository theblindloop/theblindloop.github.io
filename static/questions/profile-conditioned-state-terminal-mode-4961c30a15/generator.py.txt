"""Deterministic raster renderer and analytic answer for terminal recurrence."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont


WIDTH = 1560
HEIGHT = 900
CANVAS = (WIDTH, HEIGHT)
LABELS = ("A", "B", "C", "D")
STATE_COUNT = 3
PANEL_COUNT = 6
TILE_COUNT = 4

BACKGROUND = (255, 255, 255)
INK = (35, 43, 55)
MUTED = (103, 113, 127)
OUTLINE = (168, 178, 192)
CARD = (255, 255, 255)
HEADER = (30, 42, 62)

LABEL_COLORS = (
    (214, 70, 70),
    (52, 112, 218),
    (55, 166, 100),
    (155, 78, 188),
)
STATE_COLORS = (
    (242, 166, 48),
    (30, 165, 177),
    (220, 67, 145),
)
STATE_SHAPE_NAMES = ("circle", "square", "triangle")
PANEL_TINTS = (
    (247, 250, 255),
    (250, 248, 244),
    (247, 252, 248),
    (250, 247, 252),
    (248, 250, 250),
    (252, 249, 246),
)

PANEL_LEFTS = tuple(20 + index * 256 for index in range(PANEL_COUNT))
PANEL_WIDTH = 240
PANEL_TOP = 190
PANEL_BOTTOM = 820
ROW_Y = (320, 440, 560, 680)
TILE_LEFT_INSET = 10
TILE_RIGHT_INSET = 10
TILE_HALF_HEIGHT = 39
LABEL_CENTER_OFFSET = 37
MARKER_CENTER_OFFSET = 156
KEY_Y = 108
LABEL_KEY_X = (58, 178, 298, 418)
STATE_KEY_X = (760, 880, 1000)

FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(FONT_PATH, size)


def _rgb(value: Iterable[int]) -> tuple[int, int, int]:
    return tuple(int(channel) for channel in value)  # type: ignore[return-value]


def _text_center(
    draw: ImageDraw.ImageDraw,
    xy: tuple[int, int],
    text: str,
    font: ImageFont.ImageFont,
    fill: tuple[int, int, int],
) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text(
        (xy[0] - (box[2] - box[0]) // 2, xy[1] - (box[3] - box[1]) // 2 - 1),
        text,
        font=font,
        fill=fill,
    )


def _draw_state(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    radius: int,
    shape: str,
    color: tuple[int, int, int],
) -> None:
    cx, cy = center
    if shape == "circle":
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=color)
    elif shape == "square":
        draw.rounded_rectangle(
            (cx - radius, cy - radius, cx + radius, cy + radius),
            radius=max(2, radius // 5),
            fill=color,
        )
    elif shape == "triangle":
        draw.polygon(
            ((cx, cy - radius), (cx - radius, cy + radius), (cx + radius, cy + radius)),
            fill=color,
        )
    else:
        raise ValueError(f"unknown state shape {shape!r}")


def _terminal_scores(states: list[list[int]]) -> list[int]:
    if len(states) != TILE_COUNT:
        raise ValueError("expected four identity trajectories")
    scores: list[int] = []
    for trajectory in states:
        if len(trajectory) != PANEL_COUNT or any(value not in range(STATE_COUNT) for value in trajectory):
            raise ValueError("expected six-panel trajectories over three states")
        terminal = trajectory[-1]
        scores.append(sum(int(value == terminal) for value in trajectory))
    return scores


def _trajectory_score(trajectory: list[int]) -> int:
    if not trajectory:
        raise ValueError("trajectory cannot be empty")
    return sum(int(value == trajectory[-1]) for value in trajectory)


def _sample_trajectory(rng: random.Random, maximum_terminal_count: int) -> list[int]:
    for _ in range(10000):
        candidate = [rng.randrange(STATE_COUNT) for _ in range(PANEL_COUNT)]
        score = _trajectory_score(candidate)
        if score <= maximum_terminal_count:
            return candidate
    raise RuntimeError("could not sample a bounded terminal recurrence trajectory")


def _sample_exact_trajectory(rng: random.Random, terminal_count: int) -> list[int]:
    for _ in range(10000):
        candidate = [rng.randrange(STATE_COUNT) for _ in range(PANEL_COUNT)]
        if _trajectory_score(candidate) == terminal_count:
            return candidate
    raise RuntimeError("could not sample the requested terminal recurrence boundary")


def sample_scene(seed: int) -> dict:
    """Sample a balanced, uniquely answerable six-panel scene."""

    seed = int(seed)
    rng = random.Random(seed)
    answer_index = seed % TILE_COUNT
    winner_state = (seed // TILE_COUNT) % STATE_COUNT

    winner = [winner_state] * PANEL_COUNT
    winner[rng.randrange(PANEL_COUNT - 1)] = (winner_state + rng.randrange(1, STATE_COUNT)) % STATE_COUNT
    winner[-1] = winner_state
    states: list[list[int]] = [list(winner) for _ in range(TILE_COUNT)]

    near_boundary = seed % 13 == 0
    distractor_limit = 4 if near_boundary else 3
    near_assigned = False
    for identity in range(TILE_COUNT):
        if identity == answer_index:
            continue
        if near_boundary and not near_assigned:
            trajectory = _sample_exact_trajectory(rng, 4)
            near_assigned = True
        else:
            trajectory = _sample_trajectory(rng, distractor_limit)
        while _trajectory_score(trajectory) >= _trajectory_score(winner):
            if near_boundary and not near_assigned:
                trajectory = _sample_exact_trajectory(rng, 4)
                near_assigned = True
            else:
                trajectory = _sample_trajectory(rng, distractor_limit)
        states[identity] = trajectory

    row_orders = [rng.sample(range(TILE_COUNT), TILE_COUNT) for _ in range(PANEL_COUNT)]
    label_palette = [list(color) for color in LABEL_COLORS]
    state_palette = [list(color) for color in STATE_COLORS]
    rng.shuffle(label_palette)
    rng.shuffle(state_palette)
    panel_tints = [list(PANEL_TINTS[index]) for index in rng.sample(range(len(PANEL_TINTS)), len(PANEL_TINTS))]
    state_shapes = list(STATE_SHAPE_NAMES)
    rng.shuffle(state_shapes)
    storage_order = list(range(TILE_COUNT))
    rng.shuffle(storage_order)
    marker_radius = 7 if seed % 19 == 0 else rng.choice((14, 15, 16, 17, 18))

    scene = {
        "states": states,
        "row_orders": row_orders,
        "label_palette": label_palette,
        "state_palette": state_palette,
        "panel_tints": panel_tints,
        "state_shapes": state_shapes,
        "marker_radius": marker_radius,
        "tile_outline_width": rng.choice((1, 2, 3)),
        "storage_order": storage_order,
    }
    scores = _terminal_scores(states)
    if scores.count(max(scores)) != 1 or scores.index(max(scores)) != answer_index:
        raise AssertionError(f"sampler failed to create the requested winner: {scores}")
    return scene


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS, BACKGROUND)
    draw = ImageDraw.Draw(image)
    label_palette = [_rgb(color) for color in scene["label_palette"]]
    state_palette = [_rgb(color) for color in scene["state_palette"]]
    panel_tints = [_rgb(color) for color in scene["panel_tints"]]
    states = scene["states"]
    row_orders = scene["row_orders"]
    state_shapes = scene["state_shapes"]
    marker_radius = int(scene["marker_radius"])
    outline_width = int(scene["tile_outline_width"])

    title_font = _font(24)
    section_font = _font(16)
    label_font = _font(21)
    state_font = _font(15)
    small_font = _font(13)

    draw.rectangle((0, 0, WIDTH - 1, 52), fill=HEADER)
    draw.text((25, 14), "TERMINAL STATE RECURRENCE", font=title_font, fill=(255, 255, 255))
    draw.text(
        (25, 61),
        "Track each identity through P1-P6; the question compares each label's final-state history.",
        font=small_font,
        fill=MUTED,
    )

    draw.text((25, 82), "IDENTITY KEY", font=section_font, fill=INK)
    for identity, x in enumerate(LABEL_KEY_X):
        draw.rectangle((x - 20, KEY_Y - 18, x + 20, KEY_Y + 18), fill=label_palette[identity])
        _text_center(draw, (x, KEY_Y), LABELS[identity], label_font, (255, 255, 255))
    draw.text((650, 82), "STATE KEY", font=section_font, fill=INK)
    for state, x in enumerate(STATE_KEY_X):
        _draw_state(draw, (x, KEY_Y), 16, state_shapes[state], state_palette[state])
        _text_center(draw, (x, KEY_Y + 31), str(state), state_font, INK)

    for panel, panel_left in enumerate(PANEL_LEFTS):
        panel_right = panel_left + PANEL_WIDTH
        draw.rounded_rectangle(
            (panel_left, PANEL_TOP, panel_right, PANEL_BOTTOM),
            radius=12,
            fill=panel_tints[panel],
            outline=OUTLINE,
            width=2,
        )
        _text_center(draw, (panel_left + 28, PANEL_TOP + 25), f"P{panel + 1}", section_font, INK)
        draw.line(
            (panel_left + 15, PANEL_TOP + 52, panel_right - 15, PANEL_TOP + 52),
            fill=OUTLINE,
            width=1,
        )
        if panel < PANEL_COUNT - 1:
            arrow_start = panel_right + 2
            arrow_end = PANEL_LEFTS[panel + 1] - 2
            arrow_y = PANEL_TOP + 25
            draw.line((arrow_start, arrow_y, arrow_end, arrow_y), fill=OUTLINE, width=2)
            draw.polygon(
                ((arrow_end - 7, arrow_y - 5), (arrow_end, arrow_y), (arrow_end - 7, arrow_y + 5)),
                fill=OUTLINE,
            )

        tile_left = panel_left + TILE_LEFT_INSET
        tile_right = panel_right - TILE_RIGHT_INSET
        for row, identity in enumerate(row_orders[panel]):
            y = ROW_Y[row]
            draw.rounded_rectangle(
                (tile_left, y - TILE_HALF_HEIGHT, tile_right, y + TILE_HALF_HEIGHT),
                radius=9,
                fill=CARD,
                outline=(133, 145, 162),
                width=outline_width,
            )
            label_x = tile_left + LABEL_CENTER_OFFSET
            draw.rectangle((label_x - 20, y - 19, label_x + 20, y + 19), fill=label_palette[identity])
            _text_center(draw, (label_x, y), LABELS[identity], label_font, (255, 255, 255))
            state = int(states[identity][panel])
            _draw_state(
                draw,
                (tile_left + MARKER_CENTER_OFFSET, y),
                marker_radius,
                state_shapes[state],
                state_palette[state],
            )
            _text_center(draw, (tile_left + 207, y), str(state), state_font, INK)

    draw.text(
        (25, 850),
        "Each row is independently rearranged. Use the identity key to follow A, B, C, and D.",
        font=small_font,
        fill=MUTED,
    )
    return image


def analytic_gold(scene: dict) -> str:
    scores = _terminal_scores(scene["states"])
    best = max(scores)
    winners = [index for index, score in enumerate(scores) if score == best]
    if len(winners) != 1:
        raise ValueError(f"scene has no unique terminal recurrence leader: {scores}")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    scores = sorted(_terminal_scores(scene["states"]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    scores = _terminal_scores(scene["states"])
    best = max(scores)
    return (
        scores.count(best) != 1
        or margin(scene) < 2.0
        or int(scene["marker_radius"]) < 10
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["storage_order"] = list(reversed(scene["storage_order"]))
    return [("reverse_storage_order", twin)]
