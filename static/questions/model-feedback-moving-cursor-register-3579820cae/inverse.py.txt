"""Independent pixel-only inverse for the cursor-addressed tape."""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH = 1860
HEIGHT = 560
CELL_CENTERS = ((80, 286), (145, 286), (210, 286), (275, 286))
CURSOR_SAMPLE_Y = 216
CARD_X = tuple(360 + 144 * index for index in range(10))
CARD_Y0, CARD_Y1 = 128, 440
LIT = np.array([34, 44, 58], dtype=np.uint8)
PALE = np.array([225, 233, 240], dtype=np.uint8)
CURSOR = np.array([210, 137, 39], dtype=np.uint8)
TOGGLE = np.array([215, 73, 65], dtype=np.uint8)
ADVANCE = np.array([48, 111, 201], dtype=np.uint8)
COPY = np.array([122, 71, 171], dtype=np.uint8)


def _step(cells: list[int], cursor: int, command: int) -> int:
    if command == 0:
        cells[cursor] = 1 - cells[cursor]
    elif command == 1:
        cursor = (cursor + 1) % 4
    elif command == 2:
        cells[(cursor + 1) % 4] = cells[cursor]
    else:
        raise ValueError("unknown command")
    return cursor


def _pixel(arr: np.ndarray, center: tuple[int, int]) -> np.ndarray:
    x, y = center
    return arr[y, x]


def _read_bit(arr: np.ndarray, center: tuple[int, int]) -> int:
    pixel = _pixel(arr, center)
    if np.array_equal(pixel, LIT):
        return 1
    if np.array_equal(pixel, PALE):
        return 0
    raise ValueError("tape cell is unreadable")


def decision_from_image(image: Image.Image) -> str:
    """Read visible cells, cursor, and command badge colors, then simulate."""
    rgb = image.convert("RGB")
    if rgb.size != (WIDTH, HEIGHT):
        raise ValueError("unexpected image size")
    arr = np.asarray(rgb)
    cells = [_read_bit(arr, center) for center in CELL_CENTERS]

    cursor_candidates = []
    for x, _ in CELL_CENTERS:
        cursor_candidates.append(np.array_equal(arr[CURSOR_SAMPLE_Y, x], CURSOR))
    if sum(cursor_candidates) != 1:
        raise ValueError("cursor is unreadable")
    cursor = cursor_candidates.index(True)

    colors = (TOGGLE, ADVANCE, COPY)
    commands = []
    for x0 in CARD_X:
        crop = (slice(CARD_Y0, CARD_Y1 + 1), slice(x0, x0 + 133))
        counts = [int(np.all(arr[crop] == color, axis=2).sum()) for color in colors]
        command = int(np.argmax(counts))
        if counts[command] < 400 or sum(value >= 400 for value in counts) != 1:
            raise ValueError("command badge is unreadable")
        commands.append(command)

    for command in commands:
        cursor = _step(cells, cursor, command)
    labels = {
        (0, 0, 0, 0): "clear",
        (1, 1, 0, 0): "front-pair",
        (0, 0, 1, 1): "split-pair",
        (1, 0, 1, 0): "alternating",
    }
    final = tuple(cells)
    if final not in labels:
        raise ValueError("final tape pattern is outside declared answer set")
    return labels[final]
