"""Renderer and analytic arm for a cursor-addressed binary tape process."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1860
HEIGHT = 560
START_RECT = (24, 128, 330, 440)
CARD_RECTS = tuple((360 + 144 * index, 128, 492 + 144 * index, 440) for index in range(10))
CELL_CENTERS = ((80, 286), (145, 286), (210, 286), (275, 286))
CURSOR_SAMPLE_Y = 216
COMMAND_BADGE_Y = 280

BACKGROUND = (244, 247, 250)
PANEL = (255, 255, 255)
FRAME = (116, 130, 145)
INK = (34, 44, 58)
LIT = (34, 44, 58)
PALE = (225, 233, 240)
CURSOR = (210, 137, 39)
TOGGLE = (215, 73, 65)
ADVANCE = (48, 111, 201)
COPY = (122, 71, 171)
CONNECTOR = (157, 169, 181)

COMMAND_NAMES = ("TOGGLE", "NEXT", "COPY")
COMMAND_COLORS = (TOGGLE, ADVANCE, COPY)
PATTERN_LABELS = {
    (0, 0, 0, 0): "clear",
    (1, 1, 0, 0): "front-pair",
    (0, 0, 1, 1): "split-pair",
    (1, 0, 1, 0): "alternating",
}
TARGET_PATTERNS = tuple(PATTERN_LABELS)


def _font(size: int = 18, bold: bool = False) -> ImageFont.ImageFont:
    filename = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{filename}", size)
    except OSError:
        return ImageFont.load_default()


def _step(cells: list[int], cursor: int, command: int) -> int:
    if command == 0:
        cells[cursor] = 1 - int(cells[cursor])
    elif command == 1:
        cursor = (cursor + 1) % 4
    elif command == 2:
        cells[(cursor + 1) % 4] = int(cells[cursor])
    else:
        raise ValueError(f"bad command {command}")
    return cursor


def _simulate(initial_cells: list[int] | tuple[int, ...], initial_cursor: int, commands: list[int] | tuple[int, ...]) -> tuple[int, ...]:
    cells = [int(value) for value in initial_cells]
    cursor = int(initial_cursor)
    for command in commands:
        cursor = _step(cells, cursor, int(command))
    return tuple(cells)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    desired = TARGET_PATTERNS[int(seed) % len(TARGET_PATTERNS)]
    for _attempt in range(10000):
        initial_cells = [rng.randrange(2) for _ in range(4)]
        initial_cursor = rng.randrange(4)
        commands = [rng.randrange(3) for _ in range(10)]
        if set(commands) != {0, 1, 2}:
            continue
        if commands.count(0) < 2 or commands.count(1) < 2 or commands.count(2) < 2:
            continue
        if _simulate(initial_cells, initial_cursor, commands) == desired:
            return {
                "initial_cells": initial_cells,
                "initial_cursor": initial_cursor,
                "commands": commands,
            }
    raise AssertionError("unable to sample a valid cursor-tape scene")


def _arrowhead(draw: ImageDraw.ImageDraw, tip: tuple[float, float], color: tuple[int, int, int]) -> None:
    x, y = tip
    draw.polygon(((x, y), (x - 12, y - 8), (x - 12, y + 8)), fill=color)


def _draw_cell(draw: ImageDraw.ImageDraw, center: tuple[int, int], value: int) -> None:
    cx, cy = center
    fill = LIT if int(value) else PALE
    draw.rounded_rectangle((cx - 23, cy - 23, cx + 23, cy + 23), radius=7, fill=fill, outline=INK, width=3)


def _draw_cursor(draw: ImageDraw.ImageDraw, center: tuple[int, int]) -> None:
    cx, _ = center
    draw.polygon(((cx, 238), (cx - 17, 198), (cx + 17, 198)), fill=CURSOR, outline=INK)


def _draw_badge(draw: ImageDraw.ImageDraw, center: tuple[int, int], command: int, font: ImageFont.ImageFont) -> None:
    cx, cy = center
    color = COMMAND_COLORS[command]
    draw.rounded_rectangle((cx - 39, cy - 39, cx + 39, cy + 39), radius=12, fill=color, outline=INK, width=2)
    if command == 0:
        draw.text((cx - 15, cy - 22), "T", fill=PANEL, font=_font(42, bold=True))
    elif command == 1:
        draw.line((cx - 24, cy, cx + 18, cy), fill=PANEL, width=8)
        draw.polygon(((cx + 25, cy), (cx + 8, cy - 13), (cx + 8, cy + 13)), fill=PANEL)
    else:
        draw.rectangle((cx - 23, cy - 20, cx - 2, cy + 1), outline=PANEL, width=5)
        draw.rectangle((cx + 2, cy - 1, cx + 23, cy + 20), outline=PANEL, width=5)
    label = COMMAND_NAMES[command]
    bbox = draw.textbbox((0, 0), label, font=font)
    draw.text((cx - (bbox[2] - bbox[0]) / 2, cy + 57), label, fill=INK, font=font)


def _draw_connector(draw: ImageDraw.ImageDraw, start_x: int, end_x: int) -> None:
    y = COMMAND_BADGE_Y
    draw.line((start_x, y, end_x, y), fill=CONNECTOR, width=4)
    _arrowhead(draw, (end_x, y), CONNECTOR)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(20, bold=True)
    small_font = _font(16)
    card_font = _font(15, bold=True)

    draw.text((28, 48), "CURSOR TAPE — APPLY COMMANDS LEFT TO RIGHT", fill=INK, font=title_font)
    draw.line((28, 84, 1830, 84), fill=(210, 218, 226), width=4)

    draw.rounded_rectangle(START_RECT, radius=18, fill=PANEL, outline=FRAME, width=3)
    draw.text((48, 154), "START", fill=INK, font=_font(20, bold=True))
    draw.text((48, 187), "gold arrow = cursor", fill=FRAME, font=small_font)
    draw.text((48, 360), "dark = 1   pale = 0", fill=INK, font=small_font)
    draw.text((48, 391), "cells 1 → 4", fill=FRAME, font=small_font)
    for index, center in enumerate(CELL_CENTERS):
        _draw_cell(draw, center, int(scene["initial_cells"][index]))
        label = str(index + 1)
        bbox = draw.textbbox((0, 0), label, font=small_font)
        draw.text((center[0] - (bbox[2] - bbox[0]) / 2, 320), label, fill=INK, font=small_font)
    _draw_cursor(draw, CELL_CENTERS[int(scene["initial_cursor"])])

    first_x = CARD_RECTS[0][0]
    _draw_connector(draw, START_RECT[2] + 6, first_x - 6)
    for index, rect in enumerate(CARD_RECTS):
        x0, y0, x1, y1 = rect
        draw.rounded_rectangle(rect, radius=16, fill=PANEL, outline=FRAME, width=3)
        draw.text((x0 + 12, y0 + 17), str(index + 1), fill=FRAME, font=small_font)
        _draw_badge(draw, ((x0 + x1) // 2, COMMAND_BADGE_Y), int(scene["commands"][index]), card_font)
        if index < len(CARD_RECTS) - 1:
            _draw_connector(draw, x1 + 4, CARD_RECTS[index + 1][0] - 4)
    _draw_connector(draw, CARD_RECTS[-1][2] + 4, 1820)
    draw.text((1782, 318), "END", fill=FRAME, font=small_font)
    return image


def _final_state(scene: dict) -> tuple[int, ...]:
    return _simulate(scene["initial_cells"], scene["initial_cursor"], scene["commands"])


def analytic_gold(scene: dict) -> str:
    return PATTERN_LABELS[_final_state(scene)]


def margin(scene: dict) -> float:
    return 32.0


def is_quarantined(scene: dict) -> bool:
    cells = scene.get("initial_cells")
    cursor = scene.get("initial_cursor")
    commands = scene.get("commands")
    if not isinstance(cells, list) or len(cells) != 4 or any(value not in (0, 1) for value in cells):
        return True
    if not isinstance(cursor, int) or not 0 <= cursor < 4:
        return True
    if not isinstance(commands, list) or len(commands) != 10 or any(value not in (0, 1, 2) for value in commands):
        return True
    return _final_state(scene) not in PATTERN_LABELS


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
