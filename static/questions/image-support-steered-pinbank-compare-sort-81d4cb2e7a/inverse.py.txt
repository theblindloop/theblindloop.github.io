"""Independent pixel-only inverse arm for the pin-bank comparator-sort world.

Input is one rendered RGB image and nothing else: no latent scene, no renderer
geometry, no analytic gold.  The arm locates the framed window as the largest
frame-grey component, clips every later measurement to that window's interior
(so all marks outside it are inert by construction), segments the four coloured
pin stacks, measures each stack height from its own pixel extent, reads the
black brackets top to bottom and the black read marker, replays the declared
compare-exchange rule, and reports the colour left in the marked column.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "pinbank_compare_sort-oracle-0.1.0"

UNKNOWN = "unknown"

_PIN_RGB = {
    "red": (206, 48, 44),
    "blue": (44, 84, 196),
    "green": (30, 140, 66),
    "purple": (128, 52, 168),
}
_FRAME_RGB = (70, 70, 76)
_TOL = 34

_MIN_FRAME_AREA = 2000
_MIN_PIN_AREA = 250
_MIN_BLACK_AREA = 60


def _near(rgb: np.ndarray, target: tuple[int, int, int], tol: int = _TOL) -> np.ndarray:
    diff = np.abs(rgb.astype(np.int16) - np.array(target, dtype=np.int16))
    return diff.max(axis=2) <= tol


def _components(mask: np.ndarray, min_area: int) -> list[tuple[int, int, int, int, int]]:
    """4-connected components as (area, ymin, ymax, xmin, xmax) tuples."""
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    out = []
    ys, xs = np.nonzero(mask)
    for sy, sx in zip(ys, xs):
        if seen[sy, sx]:
            continue
        stack = [(int(sy), int(sx))]
        seen[sy, sx] = True
        area = 0
        y0 = y1 = int(sy)
        x0 = x1 = int(sx)
        while stack:
            y, x = stack.pop()
            area += 1
            y0, y1 = min(y0, y), max(y1, y)
            x0, x1 = min(x0, x), max(x1, x)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if area >= min_area:
            out.append((area, y0, y1, x0, x1))
    return out


def _black_mask(rgb: np.ndarray) -> np.ndarray:
    return rgb.astype(np.int16).max(axis=2) <= 60


def decision_from_image(image: Image.Image) -> str:
    """Return the final colour in the marked column, or ``unknown``."""
    rgb = np.asarray(image.convert("RGB"))

    frames = _components(_near(rgb, _FRAME_RGB), _MIN_FRAME_AREA)
    if not frames:
        return UNKNOWN
    frames.sort(key=lambda item: -item[0])
    _, fy0, fy1, fx0, fx1 = frames[0]
    if fy1 - fy0 < 80 or fx1 - fx0 < 80:
        return UNKNOWN

    iy0, iy1 = fy0 + 4, fy1 - 3
    ix0, ix1 = fx0 + 4, fx1 - 3
    inner = rgb[iy0:iy1, ix0:ix1]
    if inner.shape[0] < 60 or inner.shape[1] < 60:
        return UNKNOWN

    # ---- pin stacks: one connected coloured block per column
    pins = []
    for name, target in _PIN_RGB.items():
        found = _components(_near(inner, target), _MIN_PIN_AREA)
        if len(found) != 1:
            return UNKNOWN
        area, y0, y1, x0, x1 = found[0]
        span_x = x1 - x0 + 1
        span_y = y1 - y0 + 1
        if span_x < 18 or span_y < 14:
            return UNKNOWN
        if area < 0.80 * span_x * span_y:
            return UNKNOWN
        cells = span_y / 26.0
        height = int(round(cells))
        if height < 1 or height > 5 or abs(cells - height) > 0.2:
            return UNKNOWN
        pins.append(((x0 + x1) / 2.0, name, height))
    pins.sort(key=lambda item: item[0])
    centres = [item[0] for item in pins]
    if min(centres[i + 1] - centres[i] for i in range(3)) < 12.0:
        return UNKNOWN
    colours = [item[1] for item in pins]
    heights = [item[2] for item in pins]

    def slot_of(x: float) -> int:
        return min(range(4), key=lambda i: abs(centres[i] - x))

    # ---- black marks: wide bars are brackets, the compact square is the marker
    bars = []
    markers = []
    for area, y0, y1, x0, x1 in _components(_black_mask(inner), _MIN_BLACK_AREA):
        span_x = x1 - x0 + 1
        span_y = y1 - y0 + 1
        if span_x >= 3 * span_y:
            bars.append(((y0 + y1) / 2.0, x0, x1))
        elif span_x <= 2 * span_y and span_y <= 2 * span_x:
            markers.append(((x0 + x1) / 2.0, y0))
        else:
            return UNKNOWN
    if len(markers) != 1 or not bars:
        return UNKNOWN

    marked = slot_of(markers[0][0])
    bars.sort(key=lambda item: item[0])
    if len(bars) < 2:
        return UNKNOWN

    for _, x0, x1 in bars:
        left = slot_of(float(x0))
        right = slot_of(float(x1))
        if right != left + 1:
            return UNKNOWN
        if heights[right] > heights[left]:
            colours[left], colours[right] = colours[right], colours[left]
            heights[left], heights[right] = heights[right], heights[left]

    return colours[marked]
