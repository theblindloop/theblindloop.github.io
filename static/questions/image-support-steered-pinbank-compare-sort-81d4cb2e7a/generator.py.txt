"""Renderer and analytic gold for the pin-bank comparator-sort world.

One small framed window holds a bank of four pin columns.  Each column is a
solid stack of coloured cells growing from the floor, so the pin's height is an
area, not a stroke.  Beneath the bank a short stack of black brackets is drawn,
each bracket joining one adjacent pair of columns.  Read top to bottom, every
bracket is a *data-dependent* compare-exchange: the taller of the two joined
pins is moved into the left column of that pair, the shorter into the right.
Heights travel with their pins, so later comparisons see the rearranged bank.
A black square above one column marks the read slot; the answer is the colour
of the pin that finishes there.

Everything decisive lives inside the window, whose absolute position is
randomised.  Marks outside the window (grey decoration, decoy coloured squares
and decoy black bars) are inert.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "pinbank_compare_sort-renderer-0.1.0"

CANVAS_W = 1300
CANVAS_H = 980

WIN_W = 196
WIN_H = 236
FRAME = 5

SLOT_X0 = 26          # window-relative x of slot 0
SLOT_W = 30
SLOT_PITCH = 38
N_SLOTS = 4

BANK_TOP = 34         # window-relative y of the top of every column
CELL_PITCH = 26       # pixel height of one filled unit
N_ROWS = 5

MARK_Y = 12
MARK_S = 14

STEP_Y0 = 170
STEP_PITCH = 13
BAR_H = 8
BAR_INSET = 6
N_STEPS = 4

BG = (250, 250, 246)
WIN_BG = (244, 244, 238)
FRAME_RGB = (70, 70, 76)
EMPTY_RGB = (206, 206, 200)
BLACK = (12, 12, 14)
DECOR = (170, 170, 166)

COLOUR_NAMES = ("red", "blue", "green", "purple")
COLOUR_RGB = {
    "red": (206, 48, 44),
    "blue": (44, 84, 196),
    "green": (30, 140, 66),
    "purple": (128, 52, 168),
}

PAIRS = ((0, 1), (1, 2), (2, 3))


# --------------------------------------------------------------- replay logic
def _replay(colours: list[str], heights: list[int], steps: list[int]) -> tuple[list[str], list[int], list[int]]:
    """Apply the ordered compare-exchange steps; return the final bank and gaps."""
    cols = list(colours)
    hs = list(heights)
    gaps: list[int] = []
    for step in steps:
        a, b = PAIRS[step]
        gaps.append(abs(hs[a] - hs[b]))
        if hs[b] > hs[a]:
            cols[a], cols[b] = cols[b], cols[a]
            hs[a], hs[b] = hs[b], hs[a]
    return cols, hs, gaps


def _steps_ok(steps: list[int]) -> bool:
    return all(steps[i] != steps[i + 1] for i in range(len(steps) - 1))


def sample_scene(seed: int) -> dict:
    """Sample one latent scene.  Every field below is painted into the image."""
    rng = random.Random(seed)

    win_x = rng.randint(18, CANVAS_W - WIN_W - 18)
    win_y = rng.randint(16, CANVAS_H - WIN_H - 16)

    colours = list(COLOUR_NAMES)
    rng.shuffle(colours)

    tie_scene = rng.random() < 0.16
    marked = rng.randrange(N_SLOTS)

    heights: list[int] = []
    steps: list[int] = []
    # Counterbalance the "ignore every bracket" shortcut down to chance level:
    # the marked slot's initial occupant is the answer about a quarter of the
    # time, exactly as often as a uniform guess over the four columns would be.
    want_shortcut = rng.random() < 0.25
    for _ in range(240):
        if tie_scene:
            base = rng.sample(range(1, N_ROWS + 1), 3)
            heights = base + [rng.choice(base)]
            rng.shuffle(heights)
        else:
            heights = rng.sample(range(1, N_ROWS + 1), N_SLOTS)
        steps = [rng.randrange(len(PAIRS)) for _ in range(N_STEPS)]
        if not _steps_ok(steps):
            continue
        final, _, gaps = _replay(colours, heights, steps)
        if tie_scene != (min(gaps) == 0):
            continue
        if (final[marked] == colours[marked]) != want_shortcut:
            continue
        break

    distractors = []
    for _ in range(rng.randint(7, 11)):
        kind = rng.choice(("rect", "ellipse", "bar", "chip"))
        w = rng.randint(16, 54)
        h = rng.randint(12, 40)
        if kind == "bar":
            w, h = rng.randint(40, 70), rng.randint(6, 9)
        if kind == "chip":
            w = h = rng.randint(14, 22)
        for _ in range(80):
            x = rng.randint(6, CANVAS_W - w - 6)
            y = rng.randint(6, CANVAS_H - h - 6)
            if (
                x + w + 16 < win_x
                or x > win_x + WIN_W + 16
                or y + h + 16 < win_y
                or y > win_y + WIN_H + 16
            ):
                distractors.append(
                    {
                        "kind": kind,
                        "x": x,
                        "y": y,
                        "w": w,
                        "h": h,
                        "colour": rng.choice(COLOUR_NAMES) if kind == "chip" else None,
                    }
                )
                break

    return {
        "win_x": win_x,
        "win_y": win_y,
        "colours": colours,
        "heights": heights,
        "steps": steps,
        "marked_slot": marked,
        "distractors": distractors,
    }


# ------------------------------------------------------------------- geometry
def _slot_box(scene: dict, slot: int) -> tuple[int, int]:
    x = scene["win_x"] + SLOT_X0 + slot * SLOT_PITCH
    return x, x + SLOT_W - 1


def _bank_top(scene: dict) -> int:
    return scene["win_y"] + BANK_TOP


def render(scene: dict) -> Image.Image:
    """Deterministic PIL render: axis-aligned filled boxes on integer pixels."""
    image = Image.new("RGB", (CANVAS_W, CANVAS_H), BG)
    draw = ImageDraw.Draw(image)

    for item in scene["distractors"]:
        x, y, w, h = item["x"], item["y"], item["w"], item["h"]
        if item["kind"] == "ellipse":
            draw.ellipse([x, y, x + w - 1, y + h - 1], fill=DECOR)
        elif item["kind"] == "bar":
            draw.rectangle([x, y, x + w - 1, y + h - 1], fill=BLACK)
        elif item["kind"] == "chip":
            draw.rectangle([x, y, x + w - 1, y + h - 1], fill=COLOUR_RGB[item["colour"]])
        else:
            draw.rectangle([x, y, x + w - 1, y + h - 1], fill=DECOR)

    wx, wy = scene["win_x"], scene["win_y"]
    draw.rectangle([wx, wy, wx + WIN_W - 1, wy + WIN_H - 1], fill=FRAME_RGB)
    draw.rectangle(
        [wx + FRAME, wy + FRAME, wx + WIN_W - 1 - FRAME, wy + WIN_H - 1 - FRAME],
        fill=WIN_BG,
    )

    # read marker above the marked column
    mx0, mx1 = _slot_box(scene, scene["marked_slot"])
    cx = (mx0 + mx1) // 2
    draw.rectangle(
        [cx - MARK_S // 2, wy + MARK_Y, cx - MARK_S // 2 + MARK_S - 1, wy + MARK_Y + MARK_S - 1],
        fill=BLACK,
    )

    # pin bank: one solid column per slot, filled from the floor upward
    top = _bank_top(scene)
    floor = top + N_ROWS * CELL_PITCH - 1
    for slot in range(N_SLOTS):
        x0, x1 = _slot_box(scene, slot)
        fill_top = floor - scene["heights"][slot] * CELL_PITCH + 1
        rgb = COLOUR_RGB[scene["colours"][slot]]
        if fill_top > top:
            draw.rectangle([x0, top, x1, fill_top - 1], fill=EMPTY_RGB)
        draw.rectangle([x0, fill_top, x1, floor], fill=rgb)

    # comparator brackets, read top to bottom
    for index, step in enumerate(scene["steps"]):
        a, b = PAIRS[step]
        ax0, _ = _slot_box(scene, a)
        _, bx1 = _slot_box(scene, b)
        left = ax0 + BAR_INSET
        right = bx1 - BAR_INSET
        y0 = wy + STEP_Y0 + index * STEP_PITCH
        draw.rectangle([left, y0, right, y0 + BAR_H - 1], fill=BLACK)

    return image


def analytic_gold(scene: dict) -> str:
    """Colour of the pin that finishes in the marked column."""
    final, _, _ = _replay(scene["colours"], scene["heights"], scene["steps"])
    return final[scene["marked_slot"]]


def margin(scene: dict) -> float:
    """Smallest compared height gap, in pixels of stack height."""
    _, _, gaps = _replay(scene["colours"], scene["heights"], scene["steps"])
    return float(min(gaps) * CELL_PITCH)


def is_quarantined(scene: dict) -> bool:
    """Ties are the decision boundary of a compare-exchange and are excluded."""
    return margin(scene) < 13.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist: every sampled field is painted (see candidate.json)."""
    return []
