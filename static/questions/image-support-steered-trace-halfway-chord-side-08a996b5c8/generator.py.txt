"""Deterministic scene sampler and raster renderer for Trace Halfway Chord Side."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 640
HEIGHT = 360
TRACE_SAMPLES = 545
CONTROL_POINTS = 17
CONTROL_STEP = (TRACE_SAMPLES - 1) // (CONTROL_POINTS - 1)
QUARANTINE_PX = 7.5

PALETTES = (
    (10, 111, 124),
    (33, 92, 171),
    (154, 57, 112),
    (175, 82, 24),
)
BACKGROUNDS = (
    (250, 249, 246),
    (247, 250, 250),
    (251, 248, 251),
)


def _dense_trace(controls: list[int]) -> list[int]:
    """Linearly interpolate integer controls into one y value per x column."""
    values: list[int] = []
    for index in range(len(controls) - 1):
        y0, y1 = controls[index], controls[index + 1]
        for offset in range(CONTROL_STEP):
            fraction = offset / CONTROL_STEP
            values.append(int(round(y0 + fraction * (y1 - y0))))
    values.append(int(controls[-1]))
    if len(values) != TRACE_SAMPLES:
        raise AssertionError("dense trace length changed")
    return values


def _halfway_measure(y_values: Iterable[float]) -> tuple[float, float, float, float]:
    """Return (signed gap, half x-index, half y, total arc length).

    Positive signed gap means that the trace midpoint is ABOVE its chord in
    image coordinates (smaller y). Every adjacent sample is one pixel apart in x.
    """
    ys = [float(value) for value in y_values]
    if len(ys) < 2:
        raise ValueError("trace needs at least two samples")
    lengths = [math.hypot(1.0, ys[i + 1] - ys[i]) for i in range(len(ys) - 1)]
    total = sum(lengths)
    target = total / 2.0
    traveled = 0.0
    half_x = 0.0
    half_y = ys[0]
    for index, segment_length in enumerate(lengths):
        if traveled + segment_length >= target:
            fraction = (target - traveled) / segment_length
            half_x = index + fraction
            half_y = ys[index] + fraction * (ys[index + 1] - ys[index])
            break
        traveled += segment_length
    chord_y = ys[0] + (ys[-1] - ys[0]) * half_x / (len(ys) - 1)
    return chord_y - half_y, half_x, half_y, total


def _candidate_controls(rng: random.Random) -> list[int]:
    # A bounded random walk gives a legible scientific trace with distant bends.
    values = [rng.randint(118, 242)]
    for _ in range(CONTROL_POINTS - 1):
        for _attempt in range(100):
            proposal = values[-1] + rng.randint(-92, 92)
            if 54 <= proposal <= 306 and abs(proposal - values[-1]) >= 18:
                values.append(proposal)
                break
        else:
            values.append(180)
    return values


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x5A17C0DE)
    near_boundary = ((abs(int(seed)) // 2) % 7) == 0
    lower, upper = (10.0, 15.5) if near_boundary else (18.0, 62.0)
    values: list[int] | None = None
    signed = 0.0
    for _ in range(20_000):
        candidate = _dense_trace(_candidate_controls(rng))
        signed, _, _, _ = _halfway_measure(candidate)
        if lower <= abs(signed) <= upper:
            values = candidate
            break
    if values is None:
        raise RuntimeError("could not sample a well-separated trace")

    target_above = (int(seed) & 1) == 0
    if (signed > 0) != target_above:
        values = [HEIGHT - 1 - value for value in values]

    # Quotient-based styling ensures adjacent opposite-label seeds share style.
    style_key = abs(int(seed)) // 2
    palette = PALETTES[style_key % len(PALETTES)]
    background = BACKGROUNDS[(style_key // len(PALETTES)) % len(BACKGROUNDS)]
    x_start = 42 + ((style_key // 3) % 13)
    stroke_width = 5 + ((style_key // 5) % 2)
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "x_start": x_start,
        "y_values": values,
        "stroke_width": stroke_width,
        "trace_rgb": list(palette),
        "background_rgb": list(background),
    }


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    scale = 3
    background = tuple(int(v) for v in scene["background_rgb"])
    trace_rgb = tuple(int(v) for v in scene["trace_rgb"])
    x_start = int(scene["x_start"])
    y_values = [int(value) for value in scene["y_values"]]
    stroke_width = int(scene["stroke_width"])

    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    points = [((x_start + index) * scale, y * scale) for index, y in enumerate(y_values)]
    draw.line(points, fill=trace_rgb, width=stroke_width * scale, joint="curve")
    radius = stroke_width * scale / 2.0
    for x, y in (points[0], points[-1]):
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=trace_rgb)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    signed, _, _, _ = _halfway_measure(scene["y_values"])
    return "ABOVE" if signed > 0 else "BELOW"


def margin(scene: dict) -> float:
    signed, _, _, _ = _halfway_measure(scene["y_values"])
    return abs(signed)


def is_quarantined(scene: dict) -> bool:
    signed, _, _, _ = _halfway_measure(scene["y_values"])
    return abs(signed) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # All stored fields are visibly rendered; no pixel-identical latent alias.
    return []
