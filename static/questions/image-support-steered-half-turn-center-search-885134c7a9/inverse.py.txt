"""Independent inverse arm. It receives only final RGB pixels."""

from __future__ import annotations

import numpy as np
from PIL import Image

CYAN = np.array((22, 157, 187), dtype=np.uint8)
PIN_COLORS = {
    "red": np.array((224, 67, 54), dtype=np.uint8),
    "gold": np.array((235, 170, 35), dtype=np.uint8),
    "violet": np.array((130, 82, 185), dtype=np.uint8),
    "green": np.array((45, 158, 92), dtype=np.uint8),
}


def _exact_color(rgb: np.ndarray, color: np.ndarray) -> np.ndarray:
    return np.all(rgb == color, axis=2)


def _component_center(mask: np.ndarray) -> tuple[int, int] | None:
    ys, xs = np.nonzero(mask)
    if len(xs) < 64:
        return None
    if (int(xs.max()) - int(xs.min()) + 1) * (int(ys.max()) - int(ys.min()) + 1) != len(xs):
        return None
    # Return twice the half-pixel center, so rotation remains integer exact.
    return int(xs.min() + xs.max()), int(ys.min() + ys.max())


def _rotation_mismatch(mask: np.ndarray, center2: tuple[int, int]) -> int:
    height, width = mask.shape
    ys, xs = np.nonzero(mask)
    rx = center2[0] - xs
    ry = center2[1] - ys
    valid = (rx >= 0) & (rx < width) & (ry >= 0) & (ry < height)
    overlap = int(np.count_nonzero(valid))
    if overlap:
        overlap = int(np.count_nonzero(mask[ry[valid], rx[valid]]))
    # Both masks have equal area. Symmetric difference is 2*(area-overlap).
    return 2 * (len(xs) - overlap)


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    cyan = _exact_color(rgb, CYAN)
    plate = cyan.copy()
    centers: dict[str, tuple[int, int]] = {}
    for name, color in PIN_COLORS.items():
        marker = _exact_color(rgb, color)
        center = _component_center(marker)
        if center is None:
            return "abstain"
        centers[name] = center
        plate |= marker

    if int(np.count_nonzero(plate)) < 80000:
        return "abstain"
    scores = sorted((_rotation_mismatch(plate, center), name) for name, center in centers.items())
    if scores[0][0] != 0 or scores[1][0] - scores[0][0] < 4000:
        return "abstain"
    return scores[0][1]
