"""Latent sampler, Pillow renderer, and analytic arm."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
N = 17
CELL = 22
BACKGROUND = (246, 245, 239)
CYAN = (22, 157, 187)
PIN_COLORS = {
    "red": (224, 67, 54),
    "gold": (235, 170, 35),
    "violet": (130, 82, 185),
    "green": (45, 158, 92),
}
COLOR_NAMES = tuple(PIN_COLORS)
CENTER = (N // 2, N // 2)


def _rotate_cell(cell: tuple[int, int], pivot: tuple[int, int]) -> tuple[int, int]:
    return (2 * pivot[0] - cell[0], 2 * pivot[1] - cell[1])


def _mismatch_cells(cells: set[tuple[int, int]], pivot: tuple[int, int]) -> int:
    rotated = {_rotate_cell(cell, pivot) for cell in cells}
    return len(cells.symmetric_difference(rotated))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    all_cells = {(row, col) for row in range(N) for col in range(N)}

    # Paired interior courtyards make the global correspondence nontrivial while
    # the intact border guarantees that the remaining cyan region is connected.
    candidate_offsets = [
        (-5, -4), (-5, 3), (-3, 5), (3, 5), (5, 3),
        (5, -4), (3, -5), (-4, -5), (-2, 4), (4, 2),
    ]
    chosen_offsets = rng.sample(candidate_offsets, 3)
    pin_cells = [CENTER] + [(CENTER[0] + dr, CENTER[1] + dc) for dr, dc in chosen_offsets]
    protected = set(pin_cells)
    protected |= {_rotate_cell(cell, CENTER) for cell in pin_cells}
    protected |= {
        (CENTER[0] + dr, CENTER[1] + dc)
        for dr in range(-1, 2)
        for dc in range(-1, 2)
    }

    holes: set[tuple[int, int]] = set()
    options = [
        (row, col)
        for row in range(2, N - 2)
        for col in range(2, N - 2)
        if (row, col) != CENTER
    ]
    rng.shuffle(options)
    for cell in options:
        mate = _rotate_cell(cell, CENTER)
        if cell in protected or mate in protected or cell in holes or mate in holes:
            continue
        # Keep courtyards visually separated, avoiding accidental large blank glyphs.
        neighborhood = {
            (cell[0] + dr, cell[1] + dc)
            for dr in range(-1, 2)
            for dc in range(-1, 2)
        }
        mate_neighborhood = {
            (mate[0] + dr, mate[1] + dc)
            for dr in range(-1, 2)
            for dc in range(-1, 2)
        }
        if holes & (neighborhood | mate_neighborhood):
            continue
        holes.update((cell, mate))
        if len(holes) >= 28:
            break

    cells = sorted(all_cells - holes)
    colors = list(COLOR_NAMES)
    rng.shuffle(colors)
    shuffled_positions = pin_cells[:]
    rng.shuffle(shuffled_positions)
    pins = [
        {"color": color, "row": position[0], "col": position[1]}
        for color, position in zip(colors, shuffled_positions)
    ]
    answer_color = next(pin["color"] for pin in pins if (pin["row"], pin["col"]) == CENTER)
    return {
        "cells": [[row, col] for row, col in cells],
        "pins": pins,
        "answer_color": answer_color,
        "offset_x": rng.choice((58, 66, 74, 82)),
        "offset_y": rng.choice((58, 66, 74, 82)),
    }


def render(scene: dict) -> Image.Image:
    cells = scene["cells"]
    pins = scene["pins"]
    answer_color = scene["answer_color"]
    x0 = int(scene["offset_x"])
    y0 = int(scene["offset_y"])
    if answer_color not in PIN_COLORS:
        raise ValueError("unknown answer color")

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for row, col in cells:
        left = x0 + int(col) * CELL
        top = y0 + int(row) * CELL
        draw.rectangle((left, top, left + CELL - 1, top + CELL - 1), fill=CYAN)

    # Even-sized pin squares have half-pixel centers, matching cell centers exactly.
    pin_size = 12
    inset = (CELL - pin_size) // 2
    for pin in pins:
        left = x0 + int(pin["col"]) * CELL + inset
        top = y0 + int(pin["row"]) * CELL + inset
        draw.rectangle(
            (left, top, left + pin_size - 1, top + pin_size - 1),
            fill=PIN_COLORS[pin["color"]],
        )
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["answer_color"])


def _scores(scene: dict) -> list[tuple[int, str]]:
    cells = {(int(row), int(col)) for row, col in scene["cells"]}
    return sorted(
        (_mismatch_cells(cells, (int(pin["row"]), int(pin["col"]))), pin["color"])
        for pin in scene["pins"]
    )


def margin(scene: dict) -> float:
    scores = _scores(scene)
    return float(scores[1][0] - scores[0][0])


def is_quarantined(scene: dict) -> bool:
    scores = _scores(scene)
    return not (
        scores[0][0] == 0
        and scores[0][1] == scene["answer_color"]
        and scores[1][0] - scores[0][0] >= 10
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # List order is a latent serialization alias: drawing and gold are unchanged.
    twin = copy.deepcopy(scene)
    twin["cells"] = list(reversed(twin["cells"]))
    twin["pins"] = list(reversed(twin["pins"]))
    return [("reverse_serialization_order", twin)]
