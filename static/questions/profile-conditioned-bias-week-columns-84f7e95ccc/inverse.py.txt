"""Independent PNG-only inverse arm for counting WEEK-grid body columns."""

from __future__ import annotations

import numpy as np
from PIL import Image

# Public visual palette, repeated here rather than imported from rendering code.
_GRID_COLORS = np.array(((38, 116, 140), (102, 88, 145), (166, 91, 71)), dtype=np.int32)


def _runs(mask: np.ndarray) -> list[tuple[int, int]]:
    padded = np.pad(mask.astype(np.int8), (1, 1))
    changes = np.diff(padded)
    starts = np.flatnonzero(changes == 1)
    ends = np.flatnonzero(changes == -1) - 1
    return [(int(start), int(end)) for start, end in zip(starts, ends)]


def decision_from_image(image: Image.Image) -> str:
    if image.width < 224 or image.height < 224:
        raise ValueError("image resolution is too low")
    # int32 prevents RGB-square overflow while remaining tiny at this canvas size.
    pixels = np.asarray(image.convert("RGB"), dtype=np.int32)
    distances = ((pixels[:, :, None, :] - _GRID_COLORS[None, None, :, :]) ** 2).sum(axis=3)
    grid_ink = distances.min(axis=2) <= 42 * 42

    vertical_ink = grid_ink.sum(axis=0)
    peak = int(vertical_ink.max(initial=0))
    if peak < image.height * 0.42:
        raise ValueError("full-height WEEK-grid dividers are absent")
    divider_columns = vertical_ink >= peak * 0.68
    runs = _runs(divider_columns)
    if len(runs) not in (7, 8, 9):
        raise ValueError("visible divider count is outside the WEEK-grid contract")

    centers = np.array([(start + end) / 2.0 for start, end in runs], dtype=float)
    widths = np.array([end - start + 1 for start, end in runs], dtype=float)
    spacing = np.diff(centers)
    if spacing.min(initial=999.0) < 28.0:
        raise ValueError("planner intervals are below the resolution boundary")
    if spacing.max() / spacing.min() > 1.08:
        raise ValueError("dividers do not partition a regular WEEK grid")
    if widths.max(initial=0.0) > 10.0 or widths.max() - widths.min() > 4.0:
        raise ValueError("divider evidence is inconsistent")
    return str(len(runs) - 1)
