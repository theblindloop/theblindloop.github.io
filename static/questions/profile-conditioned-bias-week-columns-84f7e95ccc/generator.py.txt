"""Matched WEEK-planner renderer with canonical and prior-conflicting columns."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

SIZE = 448
SCALE = 3
COUNTS = (6, 7, 8)
GRID_PALETTE = (
    (38, 116, 140),
    (102, 88, 145),
    (166, 91, 71),
)
HEADER_PALETTE = (
    (31, 49, 69),
    (53, 44, 75),
    (76, 48, 43),
)
MARKER_COLORS = (
    (238, 184, 74),
    (104, 170, 143),
    (208, 112, 126),
    (111, 145, 204),
)
FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed ^ 0x57EE6C1)
    columns = COUNTS[seed % len(COUNTS)]
    body_rows = 4 + rng.randrange(2)
    total_cells = columns * body_rows
    marker_cells = rng.sample(range(total_cells), 12)
    return {
        "column_count": columns,
        "body_rows": body_rows,
        "card_x": rng.randint(-10, 10),
        "card_y": rng.randint(-7, 7),
        "card_width": rng.randint(344, 372),
        "body_height": rng.randint(232, 264),
        "grid_color": list(GRID_PALETTE[rng.randrange(len(GRID_PALETTE))]),
        "header_color": list(HEADER_PALETTE[rng.randrange(len(HEADER_PALETTE))]),
        "background": [rng.randint(231, 241), rng.randint(235, 244), rng.randint(239, 248)],
        "line_width": 3 + rng.randrange(2),
        "marker_cells": marker_cells,
        "marker_style": rng.randrange(len(MARKER_COLORS)),
    }


def _color(value: str | list[int] | tuple[int, int, int]) -> tuple[int, int, int]:
    if isinstance(value, str):
        raw = value.removeprefix("#")
        if len(raw) != 6:
            raise ValueError("hex colors must have six digits")
        return tuple(int(raw[index:index + 2], 16) for index in (0, 2, 4))
    return tuple(int(channel) for channel in value)


def _geometry(scene: dict) -> tuple[float, float, float, float, float, int]:
    width = float(scene["card_width"])
    body_height = float(scene["body_height"])
    left = (SIZE - width) / 2.0 + float(scene["card_x"])
    top = 28.0 + float(scene["card_y"])
    header_height = 76.0
    grid_top = top + header_height
    rows = int(scene["body_rows"])
    return left, top, width, body_height, grid_top, rows


def render(scene: dict) -> Image.Image:
    columns = int(scene["column_count"])
    if columns not in COUNTS:
        raise ValueError("column_count must be 6, 7, or 8")
    left, top, width, body_height, grid_top, rows = _geometry(scene)
    if rows not in (4, 5):
        raise ValueError("body_rows must be 4 or 5")
    right = left + width
    bottom = grid_top + body_height
    grid = _color(scene["grid_color"])
    header = _color(scene["header_color"])
    background = _color(scene["background"])
    line_width = int(scene["line_width"])
    marker_cells = [int(value) for value in scene["marker_cells"]]
    marker_style = int(scene["marker_style"])
    if len(marker_cells) != 12 or len(set(marker_cells)) != 12:
        raise ValueError("marker_cells must contain twelve unique cells")
    if any(value < 0 or value >= columns * rows for value in marker_cells):
        raise ValueError("marker cell is outside the planner body")

    image = Image.new("RGB", (SIZE * SCALE, SIZE * SCALE), background)
    draw = ImageDraw.Draw(image)
    box = lambda values: tuple(int(round(value * SCALE)) for value in values)

    # A neutral shadow and fixed card silhouette establish a familiar planner object.
    draw.rounded_rectangle(box((left + 5, top + 7, right + 5, bottom + 7)), radius=14 * SCALE,
                           fill=(198, 204, 210))
    draw.rounded_rectangle(box((left, top, right, bottom)), radius=14 * SCALE,
                           fill=(250, 249, 245), outline=(84, 91, 99), width=2 * SCALE)
    draw.rounded_rectangle(box((left, top, right, grid_top + 8)), radius=14 * SCALE, fill=header)
    draw.rectangle(box((left, grid_top - 8, right, grid_top + 8)), fill=header)

    font = ImageFont.truetype(FONT_PATH, 34 * SCALE)
    label = "WEEK"
    bounds = draw.textbbox((0, 0), label, font=font)
    text_width = bounds[2] - bounds[0]
    text_height = bounds[3] - bounds[1]
    draw.text(((left + width / 2) * SCALE - text_width / 2,
               (top + 38) * SCALE - text_height / 2 - bounds[1]),
              label, font=font, fill=(250, 250, 247))

    # Same outer body size distribution for all classes; only interval count changes.
    draw.rectangle(box((left, grid_top, right, bottom)), fill=(250, 249, 245))
    x_lines = [int(round((left + width * index / columns) * SCALE)) for index in range(columns + 1)]
    y_lines = [int(round((grid_top + body_height * index / rows) * SCALE)) for index in range(rows + 1)]
    scaled_line = line_width * SCALE
    for x in x_lines:
        draw.line((x, y_lines[0], x, y_lines[-1]), fill=grid, width=scaled_line)
    for y in y_lines:
        draw.line((x_lines[0], y, x_lines[-1], y), fill=grid, width=scaled_line)

    # Exactly twelve small appointment marks in every scene counterbalance object count/ink.
    marker_color = MARKER_COLORS[marker_style % len(MARKER_COLORS)]
    for cell_index in marker_cells:
        row, column = divmod(cell_index, columns)
        cx = left + width * (column + 0.5) / columns
        cy = grid_top + body_height * (row + 0.5) / rows
        radius = 5.0
        draw.ellipse(box((cx - radius, cy - radius, cx + radius, cy + radius)), fill=marker_color)

    return image.resize((SIZE, SIZE), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(int(scene["column_count"]))


def margin(scene: dict) -> float:
    interval = float(scene["card_width"]) / int(scene["column_count"])
    return round(interval - float(scene["line_width"]), 4)


def is_quarantined(scene: dict) -> bool:
    left, top, width, body_height, grid_top, _ = _geometry(scene)
    right = left + width
    bottom = grid_top + body_height
    return (
        margin(scene) < 28.0
        or left < 12.0
        or top < 12.0
        or right > SIZE - 12.0
        or bottom > SIZE - 12.0
        or body_height < 180.0
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    value = _color(scene["grid_color"])
    twin["grid_color"] = "#" + "".join(f"{channel:02x}" for channel in value) \
        if not isinstance(scene["grid_color"], str) else list(value)
    return [("equivalent_grid_color_encoding", twin)]
