"""Independent raster-only inverse for orthogonal route detour ratios."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_COLORS = {
    "blue": (35, 100, 205),
    "green": (20, 145, 85),
    "purple": (157, 67, 181),
}
_WIDTH = 9


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    work = mask.copy()
    height, width = work.shape
    output: list[tuple[int, int, int, int, int]] = []
    for y0, x0 in zip(*np.nonzero(work), strict=True):
        if not work[y0, x0]:
            continue
        queue = deque([(int(y0), int(x0))])
        work[y0, x0] = False
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = y + dy, x + dx
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        output.append((len(xs), min(xs), max(xs), min(ys), max(ys)))
    return output


def _ratio(arr: np.ndarray, rgb: tuple[int, int, int]) -> float:
    mask = np.all(arr == np.asarray(rgb, dtype=np.uint8), axis=2)
    components = [item for item in _components(mask) if item[0] > 100]
    if len(components) != 1:
        raise ValueError(f"expected one intact colored route, found {len(components)}")
    area, x0, x1, y0, y1 = components[0]
    chord = x1 - x0
    if not 220 <= chord <= 340 or y1 - y0 < 25:
        raise ValueError("route extent is outside the public geometry")
    # Five inclusive axis-aligned rectangles. At each of four one-sided
    # orthogonal joints, only a (half-width + 1) square is double-painted.
    half_joint = _WIDTH // 2 + 1
    numerator = area - 5 * _WIDTH + 4 * half_joint * half_joint
    if numerator <= 0 or numerator % _WIDTH:
        raise ValueError("route ink is not a valid fixed-width orthogonal centerline")
    length = numerator / _WIDTH
    ratio = length / chord
    if not 1.15 <= ratio <= 1.70:
        raise ValueError("recovered detour ratio is implausible")
    return float(ratio)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    scores = {color: _ratio(arr, rgb) for color, rgb in _COLORS.items()}
    ordered = sorted(scores.items(), key=lambda item: item[1], reverse=True)
    if ordered[0][1] - ordered[1][1] < 0.05:
        raise ValueError("top route ratios are unresolved")
    return ordered[0][0]
