"""Deterministic Pillow world for normalized orthogonal-route length."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

COLOR_RGB = {
    "blue": (35, 100, 205),
    "green": (20, 145, 85),
    "purple": (157, 67, 181),
}
BACKGROUND = (247, 249, 250)
PANEL = (255, 255, 255)
GRID = (218, 223, 229)
INK = (38, 43, 51)

_ROLE_OPTIONS = {
    "high": ((240, 72), (280, 75), (320, 78)),
    "mid": ((240, 48), (280, 55), (320, 62)),
    "low": ((240, 28), (280, 34), (320, 40)),
}


def _route_ratio(route: dict[str, Any]) -> float:
    points = [(int(x), int(y)) for x, y in route["points"]]
    length = sum(math.hypot(x2 - x1, y2 - y1) for (x1, y1), (x2, y2) in zip(points, points[1:]))
    chord = math.hypot(points[-1][0] - points[0][0], points[-1][1] - points[0][1])
    return length / chord


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    colors = list(COLOR_RGB)
    winner = colors[int(seed) % len(colors)]
    others = [color for color in colors if color != winner]
    rng.shuffle(others)
    roles = {winner: "high", others[0]: "mid", others[1]: "low"}
    panel_colors = colors.copy()
    rng.shuffle(panel_colors)
    panel_tops = (66, 238, 410)
    routes: list[dict[str, Any]] = []
    for panel_top, color in zip(panel_tops, panel_colors, strict=True):
        role = roles[color]
        chord, amplitude = rng.choice(_ROLE_OPTIONS[role])
        x0 = rng.randint(118, 158)
        x3 = x0 + chord
        x1 = x0 + rng.randint(64, chord // 2 - 24)
        x2 = x1 + rng.randint(62, x3 - x1 - 58)
        base_y = panel_top + 85
        middle_y = base_y + rng.choice((-1, 1)) * amplitude
        routes.append(
            {
                "color": color,
                "panel_top": panel_top,
                "points": [[x0, base_y], [x1, base_y], [x1, middle_y], [x2, middle_y], [x2, base_y], [x3, base_y]],
            }
        )
    return {"width": 700, "height": 590, "line_width": 9, "routes": routes}


def analytic_gold(scene: dict[str, Any]) -> str:
    scores: dict[str, float] = {}
    for route in scene["routes"]:
        points = [(int(x), int(y)) for x, y in route["points"]]
        length = sum(math.hypot(x2 - x1, y2 - y1) for (x1, y1), (x2, y2) in zip(points, points[1:]))
        chord = math.hypot(points[-1][0] - points[0][0], points[-1][1] - points[0][1])
        scores[str(route["color"])] = length / chord
    best = max(scores.values())
    winners = [color for color, score in scores.items() if abs(score - best) < 1e-12]
    return winners[0] if len(winners) == 1 else "abstain"


def margin(scene: dict[str, Any]) -> float:
    scores = sorted((_route_ratio(route) for route in scene["routes"]), reverse=True)
    return 100.0 * (scores[0] - scores[1])


def is_quarantined(scene: dict[str, Any]) -> bool:
    if analytic_gold(scene) == "abstain" or margin(scene) < 5.0:
        return True
    for route in scene["routes"]:
        points = [(int(x), int(y)) for x, y in route["points"]]
        if len(points) != 6 or any(math.hypot(b[0] - a[0], b[1] - a[1]) < 24 for a, b in zip(points, points[1:])):
            return True
    return False


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []


def _rect_route(draw: ImageDraw.ImageDraw, points: list[list[int]], fill: tuple[int, int, int], width: int) -> None:
    half = width // 2
    for (x1, y1), (x2, y2) in zip(points, points[1:]):
        if y1 == y2:
            draw.rectangle((min(x1, x2), y1 - half, max(x1, x2), y1 + half), fill=fill)
        elif x1 == x2:
            draw.rectangle((x1 - half, min(y1, y2), x1 + half, max(y1, y2)), fill=fill)
        else:
            raise ValueError("routes must be orthogonal")


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BACKGROUND)
    draw = ImageDraw.Draw(image)
    try:
        title_font = ImageFont.truetype("DejaVuSans-Bold.ttf", 17)
        label_font = ImageFont.truetype("DejaVuSans-Bold.ttf", 14)
        note_font = ImageFont.truetype("DejaVuSans.ttf", 12)
    except OSError:
        title_font = label_font = note_font = ImageFont.load_default()
    draw.text((52, 18), "Route length compared with endpoint distance", fill=INK, font=title_font)
    draw.text((505, 24), "START  →  END", fill=INK, font=note_font)
    width = int(scene["line_width"])
    for route in scene["routes"]:
        top = int(route["panel_top"])
        color = str(route["color"])
        points = [[int(x), int(y)] for x, y in route["points"]]
        draw.rectangle((50, top, 650, top + 154), fill=PANEL, outline=GRID, width=2)
        draw.text((64, top + 12), color, fill=INK, font=label_font)
        _rect_route(draw, points, COLOR_RGB[color], width)
    return image
