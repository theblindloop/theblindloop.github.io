"""Image-only inverse arm for panel_scale_units.

Receives a rendered PNG and nothing else: no scene, no renderer constants, no
gold. It re-discovers the page structure from ink alone —

1. dark ink columns with tall vertical runs are the two panel axes;
2. the dark ink immediately to the left of an axis is that panel's tick ladder;
   clustering it into rows gives the unit spacing and the zero level;
3. the saturated (non-gray, non-black) pixels standing on a panel's zero level
   are its bar, whose height is walked upward column by column;

then reports which panel's bar/spacing ratio is larger. Ambiguous or damaged
structure returns "abstain" instead of a guess.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "panel_scale_units-oracle-1.0.0"

DARK_MAX = 110          # every channel below this counts as black ink
SAT_SPREAD = 70         # channel spread above this counts as a colored mark
SAT_MAX_MIN = 90        # ... and the mark must not be near-black
AXIS_MIN_RUN = 200      # dark pixels in a column before it can be an axis
TICK_OUTER = 20         # tick search window to the left of an axis
TICK_INNER = 3
MIN_TICKS = 3
ABSTAIN = "abstain"


def _runs(flags: np.ndarray) -> list[tuple[int, int]]:
    """Inclusive [start, end] index runs of True in a 1-D boolean array."""
    out: list[tuple[int, int]] = []
    start = None
    for index, value in enumerate(flags):
        if value and start is None:
            start = index
        elif not value and start is not None:
            out.append((start, index - 1))
            start = None
    if start is not None:
        out.append((start, len(flags) - 1))
    return out


def _panel_value(dark: np.ndarray, sat: np.ndarray, axis: tuple[int, int],
                 x_hi: int) -> float | None:
    """Units-tall of one panel's bar, or None when the panel is unreadable."""
    axis_start, axis_end = axis
    strip = dark[:, max(0, axis_start - TICK_OUTER): max(0, axis_start - TICK_INNER)]
    if strip.size == 0:
        return None
    tick_rows = strip.any(axis=1)
    clusters = _runs(tick_rows)
    if len(clusters) < MIN_TICKS:
        return None
    centers = sorted((lo + hi) / 2.0 for lo, hi in clusters)
    gaps = np.diff(np.array(centers))
    if gaps.size == 0 or gaps.min() <= 1.0:
        return None
    spacing = float(np.median(gaps))
    if spacing <= 4.0 or float(gaps.max() - gaps.min()) > 0.75 * spacing:
        return None
    zero = centers[-1]

    zero_row = int(round(zero))
    bottom = zero_row - 1
    if bottom < 1 or bottom >= sat.shape[0]:
        return None
    lo_x, hi_x = axis_end + 1, min(x_hi, sat.shape[1] - 1)
    if hi_x <= lo_x:
        return None
    columns = np.flatnonzero(sat[bottom, lo_x: hi_x + 1]) + lo_x
    if columns.size < 8:
        return None

    # Contiguous upward run of colored ink from the row just above zero.
    stack = sat[: bottom + 1, columns][::-1]
    heights_array = np.cumprod(stack, axis=0).sum(axis=0).astype(float)
    if float(heights_array.max() - heights_array.min()) > 2.0:
        return None
    return float(np.median(heights_array)) / spacing


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.int16)
    high = arr.max(axis=2)
    low = arr.min(axis=2)
    dark = high < DARK_MAX
    sat = ((high - low) > SAT_SPREAD) & (high > SAT_MAX_MIN)

    axis_columns = dark.sum(axis=0) >= AXIS_MIN_RUN
    axes = [run for run in _runs(axis_columns) if run[1] - run[0] <= 8]
    if len(axes) != 2:
        return ABSTAIN

    values = []
    for index, axis in enumerate(axes):
        x_hi = axes[index + 1][0] - 20 if index + 1 < len(axes) else arr.shape[1] - 1
        value = _panel_value(dark, sat, axis, x_hi)
        if value is None:
            return ABSTAIN
        values.append(value)

    if values[0] == values[1]:
        return ABSTAIN
    return "left" if values[0] > values[1] else "right"
