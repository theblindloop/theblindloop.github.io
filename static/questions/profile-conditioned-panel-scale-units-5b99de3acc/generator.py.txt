"""panel_scale_units — two panels, two independent visible unit scales.

Each panel draws a ladder of tick marks beside its vertical axis; consecutive
ticks are exactly one unit apart *in that panel*, and the lowest tick is zero.
A colored bar rises from the zero level. The decision compares the two
dimensionless, scale-normalized magnitudes

    u_i = bar_height_px_i / px_per_unit_i

so the raw rendered extent is deliberately not sufficient: the sampler
counterbalances the pixel-taller bar against the answer.

Quantization note (this world's only latent alias): a bar is drawn with an
integer pixel height h = round(value * px_per_unit), so every value inside one
rounding bucket produces a byte-identical raster. Gold is therefore defined on
the *effective* value h / px_per_unit, and ``latent_symmetries`` returns explicit
sub-pixel value perturbations that leave both the raster and the answer alone.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "panel_scale_units-1.0.0"

# --- fixed page geometry (integer coordinates only; no antialiasing anywhere) --
WIDTH = 680
HEIGHT = 440
Y_BASE = 372          # zero level of both panels, before the global offset
Y_TOP = 60            # highest row a tick may occupy
AXIS_X = (96, 396)    # left column of each panel's 3px vertical axis
PANEL_RIGHT = (326, 626)
BAR_INSET = 60        # bar's left edge, measured from the axis
TICK_OUTER = 16       # tick spans x in [axis - TICK_OUTER, axis - TICK_INNER]
TICK_INNER = 5
INK = (20, 20, 20)
GRID = (205, 205, 205)
BACKGROUND = (255, 255, 255)
MAX_SPAN = Y_BASE - Y_TOP     # 312 px of usable vertical room
MIN_BAR_PX = 20

# Saturated, pairwise well separated bar/swatch colors.
PALETTE = (
    "#d62728",  # red
    "#1f77b4",  # blue
    "#2ca02c",  # green
    "#e88a00",  # orange
    "#8b3fd0",  # purple
    "#0f9b9b",  # teal
)

QUARANTINE_UNITS = 0.30       # |u_left - u_right| below this is a boundary case
BANDS = {
    "wide": (0.90, 3.20),
    "near": (0.42, 0.90),
    "ambiguous": (0.02, 0.26),
}


# --- label-relevant quantities -------------------------------------------------
# Only ``values_units`` and ``px_per_unit`` may reach the analytic gold; both are
# drawn (bar extent and tick spacing), which is what makes the boundary visible.


def _px_heights(scene: dict) -> list[int]:
    """Integer pixel height of each bar above its zero tick."""
    return [
        int(round(value * span))
        for value, span in zip(scene["values_units"], scene["px_per_unit"])
    ]


def _effective_values(scene: dict) -> list[float]:
    """Post-quantization unit values, i.e. exactly what the pixels encode."""
    return [
        height / span for height, span in zip(_px_heights(scene), scene["px_per_unit"])
    ]


def analytic_gold(scene: dict) -> str:
    left, right = _effective_values(scene)
    return "left" if left > right else "right"


def margin(scene: dict) -> float:
    left, right = _effective_values(scene)
    return abs(left - right)


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_UNITS)


# --- sampling ------------------------------------------------------------------


def _tick_count(span: int) -> int:
    return MAX_SPAN // span


def sample_scene(seed: int) -> dict:
    rng = random.Random((int(seed) * 7919 + 104729) & 0xFFFFFFFF)

    band = rng.choices(("wide", "near", "ambiguous"), weights=(0.50, 0.32, 0.18))[0]
    lo, hi = BANDS[band]
    want_left = rng.random() < 0.5
    want_congruent = rng.random() < 0.5

    sampled = None
    for _ in range(20000):
        small = rng.randint(24, 38)
        big = min(60, int(round(small * rng.uniform(1.30, 1.75))))
        if big <= small:
            continue
        gap = rng.uniform(lo, hi)
        # A conflict scene needs the unit winner drawn on the *finer* scale, so
        # its bar can still be the shorter one in raw pixels.
        if band != "ambiguous" and not want_congruent:
            span_win, span_lose = small, big
            floor = gap / (1.0 - small / big) * 1.02
        else:
            span_win, span_lose = (
                (big, small)
                if band != "ambiguous"
                else ((small, big) if rng.random() < 0.5 else (big, small))
            )
            floor = 0.85 + gap
        top = _tick_count(span_win) - 0.25
        floor = max(floor, 0.85 + gap)
        if floor >= top:
            continue
        win = rng.uniform(floor, top)
        lose = win - gap
        if not 0.85 <= lose <= _tick_count(span_lose) - 0.25:
            continue

        spans = [span_win, span_lose] if want_left else [span_lose, span_win]
        values = [win, lose] if want_left else [lose, win]
        trial = {"values_units": values, "px_per_unit": spans}
        heights = _px_heights(trial)
        if min(heights) < MIN_BAR_PX:
            continue
        units = _effective_values(trial)
        effective = abs(units[0] - units[1])
        if band == "ambiguous":
            if not 0.02 <= effective <= 0.26:
                continue
        else:
            if effective < 0.42 or effective > hi + 0.05:
                continue
            if (units[0] > units[1]) is not want_left:
                continue
            if abs(heights[0] - heights[1]) < 10:
                continue
            if ((heights[0] > heights[1]) is want_left) is not want_congruent:
                continue
        sampled = (values, spans)
        break
    if sampled is None:  # pragma: no cover - the bands are always satisfiable
        raise RuntimeError(f"could not sample a {band} scene for seed {seed}")
    values, spans = sampled
    counts = [_tick_count(spans[0]), _tick_count(spans[1])]

    colors = rng.sample(PALETTE, 4)
    return {
        "values_units": values,
        "px_per_unit": spans,
        "n_ticks": counts,
        "bar_colors": colors[:2],
        "swatch_colors": colors[2:],
        "bar_widths": [rng.randint(46, 84), rng.randint(46, 84)],
        "swatch_slots": [rng.randint(0, 1), rng.randint(0, 1)],
        "offset": [rng.randint(0, 10), rng.randint(-6, 6)],
    }


# --- rendering -----------------------------------------------------------------


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    heights = _px_heights(scene)
    dx, dy = scene["offset"]
    base = Y_BASE + dy

    for panel in (0, 1):
        span = scene["px_per_unit"][panel]
        ticks = scene["n_ticks"][panel]
        axis = AXIS_X[panel] + dx
        right_edge = PANEL_RIGHT[panel] + dx

        for step in range(1, ticks + 1):
            row = base - step * span
            draw.rectangle([axis + 3, row, right_edge, row], fill=GRID)
        draw.rectangle([axis, base - ticks * span - 8, axis + 2, base + 2], fill=INK)
        draw.rectangle([axis, base, right_edge, base + 2], fill=INK)
        for step in range(0, ticks + 1):
            row = base - step * span
            draw.rectangle(
                [axis - TICK_OUTER, row - 1, axis - TICK_INNER, row + 1], fill=INK
            )

        swatch_top = base - 282 + 40 * scene["swatch_slots"][panel]
        draw.rectangle(
            [right_edge - 40, swatch_top, right_edge - 16, swatch_top + 24],
            fill=scene["swatch_colors"][panel],
        )

        bar_left = axis + BAR_INSET
        bar_right = bar_left + scene["bar_widths"][panel] - 1
        draw.rectangle(
            [bar_left, base - heights[panel], bar_right, base - 1],
            fill=scene["bar_colors"][panel],
        )
    return image


# --- declared latent alias ------------------------------------------------------


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Sub-pixel value perturbations: same raster, same gold, different latent.

    ``render`` quantizes each bar to h = round(value * px_per_unit); any value in
    the same rounding bucket is indistinguishable in the image. These are the
    world's only aliases and they are exercised, not merely asserted.
    """
    transforms: list[tuple[str, dict]] = []
    heights = _px_heights(scene)
    for panel, name in ((0, "left"), (1, "right")):
        for label, nudge in (("up", 0.25), ("down", -0.25)):
            span = scene["px_per_unit"][panel]
            twin = dict(scene)
            values = list(scene["values_units"])
            values[panel] = (heights[panel] + nudge) / span
            twin["values_units"] = values
            transforms.append((f"subpixel_{name}_{label}", twin))
    return transforms


# --- helpers used by tests and the generator -----------------------------------


def congruent(scene: dict) -> bool:
    """True when the pixel-taller bar is also the unit-taller bar."""
    heights = _px_heights(scene)
    units = _effective_values(scene)
    return (heights[0] > heights[1]) == (units[0] > units[1])


def effective_values(scene: dict) -> list[float]:
    return _effective_values(scene)


def pixel_heights(scene: dict) -> list[int]:
    return _px_heights(scene)
