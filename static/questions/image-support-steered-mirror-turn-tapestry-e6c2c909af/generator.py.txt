"""Deterministic Pillow world for the mirror-turn tapestry."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

SIZE = 720
BOARD0 = 36
CELL = 216
BG = (246, 243, 235)
INK = (30, 42, 56)
GOLD = (226, 177, 55)
CYAN = (41, 174, 190)
CORAL = (224, 91, 83)
WHITE = (255, 255, 255)

ORDER = ((0, 0), (0, 1), (0, 2), (1, 2), (1, 1), (1, 0), (2, 0), (2, 1), (2, 2))
NAMES = ("up", "right", "down", "left")


def _step(heading: int, operation: str) -> int:
    if operation == "turn":
        return (heading + 1) % 4
    if operation == "mirror":
        return (-heading) % 4
    raise ValueError("unknown operation")


def _replay(scene: dict) -> int:
    heading = int(scene["initial_heading"])
    for operation in scene["operations"]:
        heading = _step(heading, operation)
    return heading


def sample_scene(seed: int) -> dict:
    """Sample both operation kinds, then backsolve a balanced initial heading."""
    rng = random.Random(int(seed))
    while True:
        operations = [rng.choice(("turn", "mirror")) for _ in range(8)]
        if 2 <= operations.count("turn") <= 6:
            break
    target = int(seed) % 4
    initial = target
    for operation in reversed(operations):
        initial = (initial - 1) % 4 if operation == "turn" else (-initial) % 4
    return {"initial_heading": initial, "operations": operations}


def _arrow(draw: ImageDraw.ImageDraw, cx: int, cy: int, heading: int) -> None:
    vectors = ((0, -1), (1, 0), (0, 1), (-1, 0))
    dx, dy = vectors[heading]
    px, py = -dy, dx
    tail = (cx - 54 * dx, cy - 54 * dy)
    neck = (cx + 24 * dx, cy + 24 * dy)
    tip = (cx + 72 * dx, cy + 72 * dy)
    draw.line((tail, neck), fill=WHITE, width=20)
    draw.polygon(
        [tip, (cx + 18 * dx + 34 * px, cy + 18 * dy + 34 * py),
         (cx + 18 * dx - 34 * px, cy + 18 * dy - 34 * py)],
        fill=WHITE,
    )


def _turn_icon(draw: ImageDraw.ImageDraw, cx: int, cy: int) -> None:
    draw.arc((cx - 55, cy - 55, cx + 55, cy + 55), 205, 520, fill=WHITE, width=18)
    draw.polygon([(cx + 62, cy - 13), (cx + 25, cy - 32), (cx + 31, cy + 10)], fill=WHITE)


def _mirror_icon(draw: ImageDraw.ImageDraw, cx: int, cy: int) -> None:
    draw.rounded_rectangle((cx - 8, cy - 66, cx + 8, cy + 66), radius=7, fill=WHITE)
    draw.polygon([(cx - 25, cy), (cx - 65, cy - 28), (cx - 65, cy + 28)], fill=WHITE)
    draw.polygon([(cx + 25, cy), (cx + 65, cy - 28), (cx + 65, cy + 28)], outline=WHITE, width=10)


def render(scene: dict) -> Image.Image:
    initial_heading = int(scene["initial_heading"])
    operations = list(scene["operations"])
    if initial_heading not in range(4) or len(operations) != 8:
        raise ValueError("invalid scene")
    image = Image.new("RGB", (SIZE, SIZE), BG)
    draw = ImageDraw.Draw(image)
    for index, (row, col) in enumerate(ORDER):
        x0, y0 = BOARD0 + col * CELL, BOARD0 + row * CELL
        x1, y1 = x0 + CELL, y0 + CELL
        fill = GOLD if index == 0 else (CYAN if operations[index - 1] == "turn" else CORAL)
        draw.rectangle((x0, y0, x1 - 1, y1 - 1), fill=fill)
        cx, cy = x0 + CELL // 2, y0 + CELL // 2
        if index == 0:
            _arrow(draw, cx, cy, initial_heading)
        elif operations[index - 1] == "turn":
            _turn_icon(draw, cx, cy)
        else:
            _mirror_icon(draw, cx, cy)
    # Dark seams and a common thick perimeter make the tapestry one filled component.
    for k in range(4):
        p = BOARD0 + k * CELL
        draw.line((BOARD0, p, BOARD0 + 3 * CELL - 1, p), fill=INK, width=8)
        draw.line((p, BOARD0, p, BOARD0 + 3 * CELL - 1), fill=INK, width=8)
    draw.rectangle((BOARD0, BOARD0, BOARD0 + 3 * CELL - 1, BOARD0 + 3 * CELL - 1), outline=INK, width=12)
    # Snake-order chevrons sit inside the common perimeter at the two row turns.
    draw.polygon([(672, 246), (680, 270), (672, 294)], fill=INK)
    draw.polygon([(48, 462), (40, 486), (48, 510)], fill=INK)
    return image


def analytic_gold(scene: dict) -> str:
    return NAMES[_replay(scene)]


def margin(scene: dict) -> float:
    return 1.0 if not is_quarantined(scene) else 0.0


def is_quarantined(scene: dict) -> bool:
    operations = scene.get("operations", [])
    return (
        scene.get("initial_heading") not in range(4)
        or len(operations) != 8
        or any(operation not in {"turn", "mirror"} for operation in operations)
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
