"""Independent image-only inverse arm for the tapestry raster."""

from __future__ import annotations

import numpy as np
from PIL import Image

BG = np.array((246, 243, 235), dtype=np.float32)
GOLD = np.array((226, 177, 55), dtype=np.float32)
CYAN = np.array((41, 174, 190), dtype=np.float32)
CORAL = np.array((224, 91, 83), dtype=np.float32)
WHITE = np.array((255, 255, 255), dtype=np.float32)
NAMES = ("up", "right", "down", "left")
ORDER = ((0, 0), (0, 1), (0, 2), (1, 2), (1, 1), (1, 0), (2, 0), (2, 1), (2, 2))


def _nearest(mean: np.ndarray) -> int:
    palette = np.stack((GOLD, CYAN, CORAL))
    distances = np.linalg.norm(palette - mean.astype(np.float32), axis=1)
    winner = int(np.argmin(distances))
    if float(distances[winner]) > 45.0:
        raise ValueError("unrecognized tile paint")
    return winner


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or min(array.shape[:2]) < 400:
        raise ValueError("invalid raster")
    delta = np.linalg.norm(array.astype(np.float32) - BG, axis=2)
    ys, xs = np.nonzero(delta > 40.0)
    if len(xs) < 250000:
        raise ValueError("tapestry is incomplete")
    x0, x1, y0, y1 = int(xs.min()), int(xs.max()) + 1, int(ys.min()), int(ys.max()) + 1
    width, height = x1 - x0, y1 - y0
    if width < 620 or height < 620 or abs(width - height) > 25:
        raise ValueError("tapestry boundary is incomplete")
    # Destructive holes anywhere in the connected painted rectangle invalidate replay.
    interior = array[y0 + 8:y1 - 8, x0 + 8:x1 - 8]
    holes = np.linalg.norm(interior.astype(np.float32) - BG, axis=2) < 22.0
    if int(holes.sum()) > 80:
        raise ValueError("painted operation support has been erased")
    cell_w, cell_h = width / 3.0, height / 3.0
    kinds = []
    for row, col in ORDER:
        cx = x0 + (col + 0.5) * cell_w
        cy = y0 + (row + 0.5) * cell_h
        # Four corner patches avoid the white pictogram and dark seams.
        patches = []
        for oy in (-0.30, 0.30):
            for ox in (-0.30, 0.30):
                px, py = int(cx + ox * cell_w), int(cy + oy * cell_h)
                patches.append(array[py - 10:py + 11, px - 10:px + 11].reshape(-1, 3))
        kinds.append(_nearest(np.median(np.concatenate(patches), axis=0)))
    if kinds[0] != 0 or any(kind == 0 for kind in kinds[1:]):
        raise ValueError("start tile or operation paint is ambiguous")
    # Recover initial arrow direction from the centroid of white ink in the gold tile.
    cx, cy = x0 + cell_w / 2.0, y0 + cell_h / 2.0
    tile = array[int(y0):int(y0 + cell_h), int(x0):int(x0 + cell_w)]
    white = np.linalg.norm(tile.astype(np.float32) - WHITE, axis=2) < 24.0
    wy, wx = np.nonzero(white)
    if len(wx) < 1200:
        raise ValueError("initial arrow is missing")
    dx, dy = float(wx.mean() - cell_w / 2.0), float(wy.mean() - cell_h / 2.0)
    if max(abs(dx), abs(dy)) < 7 or abs(abs(dx) - abs(dy)) < 4:
        raise ValueError("initial heading is ambiguous")
    if abs(dx) > abs(dy):
        heading = 1 if dx > 0 else 3
    else:
        heading = 2 if dy > 0 else 0
    for kind in kinds[1:]:
        heading = (heading + 1) % 4 if kind == 1 else (-heading) % 4
    return NAMES[heading]
