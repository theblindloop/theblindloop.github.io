"""Independent pixel-only inverse for route homotopy around islands."""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw


def _near(array: np.ndarray, color: tuple[int, int, int], tolerance: int = 48) -> np.ndarray:
    delta = array.astype(np.int32) - np.asarray(color, dtype=np.int32)
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"))
    blue = _near(rgb, (50, 102, 190))
    orange = _near(rgb, (230, 126, 46))
    green = _near(rgb, (50, 168, 105))
    dark = _near(rgb, (35, 39, 47))

    if blue.sum() < 700 or orange.sum() < 700 or green.sum() < 900 or dark.sum() < 2500:
        return "abstain"

    barrier = blue | orange | green
    barrier_small = barrier[::2, ::2]
    dark_small = dark[::2, ::2]
    # Copy detaches Pillow from NumPy's read-only shared buffer so floodfill mutates it.
    flood = Image.fromarray(
        np.where(barrier_small, 255, 0).astype(np.uint8), mode="L"
    ).copy()
    ImageDraw.floodfill(flood, (0, 0), 128, thresh=0)
    flooded = np.asarray(flood)
    exterior = flooded == 128
    bounded = (~barrier_small) & (~exterior)
    if int(bounded.sum()) < 500:
        return "abstain"
    enclosed_island_pixels = int((bounded & dark_small).sum())
    return "no" if enclosed_island_pixels >= 80 else "yes"
