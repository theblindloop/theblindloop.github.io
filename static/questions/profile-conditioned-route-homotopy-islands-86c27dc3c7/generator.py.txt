"""Deterministic renderer and analytic arm for route homotopy around islands."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

BACKGROUND = (248, 246, 240)
BLUE = (50, 102, 190)
ORANGE = (230, 126, 46)
GREEN = (50, 168, 105)
DARK = (35, 39, 47)


def _transform_point(point: tuple[int, int], mode: int, dx: int, dy: int) -> list[int]:
    x = point[0] - 256
    y = point[1] - 256
    if mode >= 4:
        x = -x
    for _ in range(mode % 4):
        x, y = -y, x
    return [int(x + 256 + dx), int(y + 256 + dy)]


def sample_scene(seed: int) -> dict:
    """Sample paired nuisance settings: consecutive seeds differ chiefly in class."""

    rng = random.Random(seed // 2)
    center_y = 256 + rng.choice((-10, -4, 4, 10))
    left_x = 88 + rng.choice((-4, 0, 4))
    right_x = 424 + rng.choice((-4, 0, 4))
    bend_left = 154 + rng.choice((-8, -4, 0, 4, 8))
    bend_right = 358 + rng.choice((-8, -4, 0, 4, 8))
    top_y = 154 + rng.choice((-8, -4, 0, 4, 8))
    bottom_y = 358 + rng.choice((-8, -4, 0, 4, 8))

    blue = [
        (left_x, center_y),
        (bend_left, top_y),
        (bend_right, top_y + rng.choice((-4, 0, 4))),
        (right_x, center_y),
    ]
    orange = [
        (left_x, center_y),
        (bend_left, bottom_y),
        (bend_right, bottom_y + rng.choice((-4, 0, 4))),
        (right_x, center_y),
    ]

    outside_slots = [
        (66, 68),
        (256, 64),
        (446, 70),
        (68, 438),
        (256, 448),
        (444, 440),
    ]
    # Keep every island well inside the central rectangle; D4 transforms then
    # preserve a generous clearance from both routes.
    inside_slots = [(232, 240), (248, 280), (280, 240), (280, 280)]
    rng.shuffle(outside_slots)
    rng.shuffle(inside_slots)
    islands = outside_slots[:4] if seed % 2 == 0 else outside_slots[:3] + inside_slots[:1]

    mode = rng.randrange(8)
    dx = rng.choice((-12, -6, 0, 6, 12))
    dy = rng.choice((-12, -6, 0, 6, 12))
    blue_out = [_transform_point(point, mode, dx, dy) for point in blue]
    orange_out = [_transform_point(point, mode, dx, dy) for point in orange]
    islands_out = [_transform_point(point, mode, dx, dy) for point in islands]

    return {
        "blue_path": blue_out,
        "orange_path": orange_out,
        "islands": islands_out,
        "canvas_size": 512,
        "route_width": 16,
        "island_radius": 20 + 2 * (rng.randrange(2)),
    }


def _closed_polygon(scene: dict) -> list[list[int]]:
    return list(scene["blue_path"]) + list(reversed(scene["orange_path"]))


def _point_in_polygon(point: list[int], polygon: list[list[int]]) -> bool:
    x, y = point
    inside = False
    j = len(polygon) - 1
    for i in range(len(polygon)):
        xi, yi = polygon[i]
        xj, yj = polygon[j]
        if (yi > y) != (yj > y):
            crossing_x = (xj - xi) * (y - yi) / (yj - yi) + xi
            if x < crossing_x:
                inside = not inside
        j = i
    return inside


def analytic_gold(scene: dict) -> str:
    polygon = _closed_polygon(scene)
    return "no" if any(_point_in_polygon(point, polygon) for point in scene["islands"]) else "yes"


def _point_segment_distance(point: list[int], start: list[int], end: list[int]) -> float:
    px, py = point
    ax, ay = start
    bx, by = end
    vx, vy = bx - ax, by - ay
    length_sq = vx * vx + vy * vy
    if length_sq == 0:
        return math.hypot(px - ax, py - ay)
    t = max(0.0, min(1.0, ((px - ax) * vx + (py - ay) * vy) / length_sq))
    return math.hypot(px - (ax + t * vx), py - (ay + t * vy))


def margin(scene: dict) -> float:
    segments = []
    for path in (scene["blue_path"], scene["orange_path"]):
        segments.extend(zip(path, path[1:]))
    clearance = min(
        _point_segment_distance(island, start, end)
        - float(scene["island_radius"])
        - float(scene["route_width"]) / 2.0
        for island in scene["islands"]
        for start, end in segments
    )
    return round(clearance, 6)


def is_quarantined(scene: dict) -> bool:
    if len(scene["blue_path"]) < 2 or len(scene["orange_path"]) < 2:
        return True
    if scene["blue_path"][0] != scene["orange_path"][0]:
        return True
    if scene["blue_path"][-1] != scene["orange_path"][-1]:
        return True
    return margin(scene) < 24.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["blue_path"] = list(reversed(twin["blue_path"]))
    twin["orange_path"] = list(reversed(twin["orange_path"]))
    twin["islands"] = list(reversed(twin["islands"]))
    return [("reverse_route_and_island_order", twin)]


def _draw_route(draw: ImageDraw.ImageDraw, points: list[list[int]], color: tuple[int, int, int], width: int) -> None:
    normalized = []
    for first, second in zip(points, points[1:]):
        pair = tuple(sorted((tuple(first), tuple(second))))
        normalized.append(pair)
    radius = width // 2
    for first, second in sorted(normalized):
        draw.line((first, second), fill=color, width=width)
    for x, y in sorted(map(tuple, points)):
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas_size"])
    width = int(scene["route_width"])
    radius = int(scene["island_radius"])
    image = Image.new("RGB", (size, size), BACKGROUND)
    draw = ImageDraw.Draw(image)

    for x, y in sorted(map(tuple, scene["islands"])):
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=DARK)
    _draw_route(draw, scene["blue_path"], BLUE, width)
    _draw_route(draw, scene["orange_path"], ORANGE, width)

    endpoint_radius = width
    endpoints = sorted((tuple(scene["blue_path"][0]), tuple(scene["blue_path"][-1])))
    for x, y in endpoints:
        draw.ellipse(
            (x - endpoint_radius, y - endpoint_radius, x + endpoint_radius, y + endpoint_radius),
            fill=GREEN,
        )
    return image
