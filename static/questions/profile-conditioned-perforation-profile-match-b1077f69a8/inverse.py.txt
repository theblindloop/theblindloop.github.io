"""Independent image-only inverse for the perforation-profile world."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

PALETTE = {
    "cyan": (25, 174, 194), "magenta": (205, 63, 139),
    "green": (65, 154, 82), "orange": (230, 132, 38),
    "red": (214, 54, 57), "blue": (48, 102, 190),
    "gold": (212, 158, 25), "gray": (91, 97, 105),
}
TILE_COLORS = ("cyan", "magenta", "green", "orange")
BORDERS = ("red", "blue", "gold")


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    h, w = mask.shape
    seen = np.zeros((h, w), dtype=bool)
    found = []
    for yy, xx in zip(*np.nonzero(mask & ~seen)):
        if seen[yy, xx]:
            continue
        stack = [(int(yy), int(xx))]
        seen[yy, xx] = True
        area = 0
        xs, ys = [], []
        while stack:
            y, x = stack.pop()
            area += 1; xs.append(x); ys.append(y)
            for ny, nx in ((y-1,x),(y+1,x),(y,x-1),(y,x+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True; stack.append((ny, nx))
        found.append((area, min(xs), min(ys), max(xs), max(ys)))
    return found


def _largest_bbox(rgb: np.ndarray, color: str) -> tuple[int, int, int, int]:
    mask = np.all(rgb == np.array(PALETTE[color], dtype=np.uint8), axis=2)
    comps = [c for c in _components(mask) if c[0] > 900]
    if not comps:
        raise ValueError(f"missing {color} card border")
    _, x0, y0, x1, y1 = max(comps)
    if x1 - x0 < 200 or y1 - y0 < 200:
        raise ValueError(f"invalid {color} card border")
    return x0, y0, x1, y1


def _hole_count(mask: np.ndarray, bbox: tuple[int, int, int, int]) -> int:
    x0, y0, x1, y1 = bbox
    local = ~mask[y0:y1+1, x0:x1+1]
    # The tile's solid rectangular rim fills every local boundary pixel, so each
    # non-mask component is a fully enclosed opening.
    comps = [c for c in _components(local) if c[0] > 120]
    return len(comps)


def _profile(rgb: np.ndarray, panel: tuple[int, int, int, int]) -> dict[str, int]:
    px0, py0, px1, py1 = panel
    profile = {}
    for color in TILE_COLORS:
        full = np.all(rgb == np.array(PALETTE[color], dtype=np.uint8), axis=2)
        window = np.zeros_like(full)
        window[py0+7:py1-6, px0+7:px1-6] = True
        comps = [c for c in _components(full & window) if 4000 < c[0] < 6000]
        if len(comps) != 1:
            raise ValueError(f"panel has {len(comps)} {color} tiles")
        profile[color] = _hole_count(full, comps[0][1:])
    if sorted(profile.values()) != [0, 1, 2, 3]:
        raise ValueError("panel does not contain the declared hole-count multiset")
    return profile


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"))
    panels = {color: _largest_bbox(rgb, color) for color in ("gray", *BORDERS)}
    reference = _profile(rgb, panels["gray"])
    matches = [color for color in BORDERS if _profile(rgb, panels[color]) == reference]
    if len(matches) != 1:
        raise ValueError("pixels do not identify exactly one matching option")
    return matches[0]
