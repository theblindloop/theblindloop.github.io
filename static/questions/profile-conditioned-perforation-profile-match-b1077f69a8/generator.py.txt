"""Deterministic renderer for colored perforation-profile correspondence."""

from __future__ import annotations

import copy
import random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 900, 720
BACKGROUND = (250, 249, 246)
INK = (35, 38, 43)
PALETTE = {
    "cyan": (25, 174, 194),
    "magenta": (205, 63, 139),
    "green": (65, 154, 82),
    "orange": (230, 132, 38),
    "red": (214, 54, 57),
    "blue": (48, 102, 190),
    "gold": (212, 158, 25),
    "gray": (91, 97, 105),
}
TILE_COLORS = ("cyan", "magenta", "green", "orange")
BORDERS = ("red", "blue", "gold")
CARD_LAYOUTS = (
    ("reference", "gray", 250, 42, 400, 278),
    ("option", None, 24, 395, 270, 278),
    ("option", None, 315, 395, 270, 278),
    ("option", None, 606, 395, 270, 278),
)
TILE_W, TILE_H = 86, 66
HOLE_R = 9


def _hole_centers(count: int) -> list[list[int]]:
    patterns = {
        0: [],
        1: [[0, 0]],
        2: [[-20, 0], [20, 0]],
        3: [[-23, 0], [0, 0], [23, 0]],
    }
    return copy.deepcopy(patterns[count])


def _profile(cards: list[dict], role: str, border: str | None = None) -> dict[str, int]:
    matches = [c for c in cards if c["role"] == role and (border is None or c["border"] == border)]
    if len(matches) != 1:
        return {}
    return {tile["color"]: len(tile["holes"]) for tile in matches[0]["tiles"]}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    counts = [0, 1, 2, 3]
    rng.shuffle(counts)
    reference = dict(zip(TILE_COLORS, counts))

    # Both distractors preserve the complete hole-count multiset and change two bindings.
    pairs = [(0, 1), (0, 2), (0, 3), (1, 2), (1, 3), (2, 3)]
    rng.shuffle(pairs)
    distractors = []
    for a, b in pairs:
        candidate = dict(reference)
        ca, cb = TILE_COLORS[a], TILE_COLORS[b]
        candidate[ca], candidate[cb] = candidate[cb], candidate[ca]
        if candidate not in distractors:
            distractors.append(candidate)
        if len(distractors) == 2:
            break

    correct_border = BORDERS[int(seed) % len(BORDERS)]
    border_order = list(BORDERS)
    rng.shuffle(border_order)
    wrong_iter = iter(distractors)
    profiles = {
        border: (dict(reference) if border == correct_border else next(wrong_iter))
        for border in border_order
    }

    cards = []
    layouts = [CARD_LAYOUTS[0], *[CARD_LAYOUTS[i + 1] for i in range(3)]]
    assigned_borders = ["gray", *border_order]
    assigned_profiles = [reference, *[profiles[b] for b in border_order]]
    for layout, border, profile in zip(layouts, assigned_borders, assigned_profiles):
        role, _, x, y, w, h = layout
        if role == "reference":
            slots = [(55, 63), (259, 63), (55, 171), (259, 171)]
        else:
            slots = [(30, 62), (154, 62), (30, 170), (154, 170)]
        rng.shuffle(slots)
        colors = list(TILE_COLORS)
        rng.shuffle(colors)
        tiles = []
        for color, (tx, ty) in zip(colors, slots):
            holes = _hole_centers(profile[color])
            rng.shuffle(holes)
            tiles.append({"color": color, "x": tx, "y": ty, "holes": holes})
        cards.append({"role": role, "border": border, "x": x, "y": y, "w": w, "h": h, "tiles": tiles})
    return {"cards": cards}


def render(scene: dict) -> Image.Image:
    cards = scene["cards"]
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    draw.text((330, 13), "REFERENCE", fill=INK, font=font)
    draw.text((409, 365), "OPTIONS", fill=INK, font=font)
    for card in cards:
        x, y, w, h = card["x"], card["y"], card["w"], card["h"]
        border_rgb = PALETTE[card["border"]]
        draw.rectangle((x, y, x + w - 1, y + h - 1), outline=border_rgb, width=7)
        if card["role"] == "option":
            draw.rectangle((x + 84, y + h - 25, x + w - 84, y + h - 8), fill=border_rgb)
        for tile in card["tiles"]:
            tx, ty = x + tile["x"], y + tile["y"]
            draw.rectangle((tx, ty, tx + TILE_W - 1, ty + TILE_H - 1), fill=PALETTE[tile["color"]])
            cx, cy = tx + TILE_W // 2, ty + TILE_H // 2
            for dx, dy in tile["holes"]:
                draw.ellipse((cx + dx - HOLE_R, cy + dy - HOLE_R, cx + dx + HOLE_R, cy + dy + HOLE_R), fill=BACKGROUND)
    return image


def analytic_gold(scene: dict) -> str:
    cards = scene["cards"]
    reference = _profile(cards, "reference")
    matches = [border for border in BORDERS if _profile(cards, "option", border) == reference]
    if len(matches) != 1:
        raise ValueError("scene does not have exactly one matching option")
    return matches[0]


def margin(scene: dict) -> float:
    cards = scene["cards"]
    reference = _profile(cards, "reference")
    distances = []
    for border in BORDERS:
        profile = _profile(cards, "option", border)
        distances.append(sum(profile.get(c) != reference.get(c) for c in TILE_COLORS))
    ordered = sorted(distances)
    return float(ordered[1] - ordered[0])


def is_quarantined(scene: dict) -> bool:
    try:
        cards = scene["cards"]
        if len(cards) != 4 or analytic_gold(scene) not in BORDERS:
            return True
        for card in cards:
            if {t["color"] for t in card["tiles"]} != set(TILE_COLORS):
                return True
            if sorted(len(t["holes"]) for t in card["tiles"]) != [0, 1, 2, 3]:
                return True
        return margin(scene) < 2.0
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    for card in twin["cards"]:
        for tile in card["tiles"]:
            tile["holes"].reverse()
    return [("reverse_unordered_hole_center_lists", twin)]
