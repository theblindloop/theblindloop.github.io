"""Deterministic raster world with two disconnected butterfly-wing masks."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "butterfly-wing-mirror-match-renderer-0.1.0"
WIDTH = 640
HEIGHT = 640
BACKGROUND = "#f7f5ef"
THRESHOLD = 0.92
MIN_GAP = 18.0
MIN_SPAN = 150.0
MIN_AREA = 7000.0

def _scaled_left_points(base_variant: int, scale: float) -> list[list[float]]:
    return [
        [round(float(x) * scale, 6), round(float(y) * scale, 6)]
        for x, y in BASE_WINGS[base_variant]
    ]


def _mirror_points(
    left_points: list[list[float]], center_x: float
) -> list[list[float]]:
    return [
        [
            round(2.0 * center_x - float(x), 6),
            round(float(y), 6),
        ]
        for x, y in left_points
    ]


def _deform_right(
    mirrored: list[list[float]],
    center_x: float,
    center_y: float,
    mode: int,
    strength: float,
) -> list[list[float]]:
    output: list[list[float]] = []
    if mode == 0:
        factor = 1.0 + strength
        for x, y in mirrored:
            output.append([round(x, 6), round(center_y + (y - center_y) * factor, 6)])
    elif mode == 1:
        factor = 1.0 - 0.68 * strength
        for x, y in mirrored:
            output.append(
                [round(x, 6), round(center_y + (y - center_y) * factor + 16.0, 6)]
            )
    else:
        # A visible shear changes the correspondence across most of the wing,
        # while preserving one connected filled component and similar ink.
        shear = 0.55 * strength
        for x, y in mirrored:
            output.append(
                [
                    round(x + shear * (y - center_y), 6),
                    round(y, 6),
                ]
            )
    return output


def sample_scene(seed: int) -> dict:
    """Sample a balanced, fully visible pair of separated wing polygons."""
    rng = random.Random(int(seed))
    # Consecutive scene seeds in every public run alternate the two classes.
    symmetry_class = "yes" if int(seed) % 2 == 0 else "no"
    base_variant = (int(seed) // 2) % len(BASE_WINGS)
    asymmetry_mode = (int(seed) // 3) % 3
    center_x = WIDTH / 2.0 + rng.uniform(-26.0, 26.0)
    center_y = HEIGHT / 2.0 + rng.uniform(-24.0, 24.0)
    scale = rng.uniform(0.88, 0.99)
    wing_gap = rng.uniform(172.0, 188.0)
    # The gap is a nuisance field recorded in the scene. It changes both inner
    # edges symmetrically without carrying the answer.
    left_relative = _scaled_left_points(base_variant, scale)
    inner_target = wing_gap / 2.0
    for point in left_relative:
        if point[0] > -inner_target:
            point[0] = round(-inner_target, 6)
    left_points = [
        [round(center_x + point[0], 6), round(center_y + point[1], 6)]
        for point in left_relative
    ]
    mirrored = _mirror_points(left_points, center_x)
    deformation_strength = rng.uniform(0.24, 0.34)
    if symmetry_class == "yes":
        right_points = [point[:] for point in mirrored]
    else:
        right_points = _deform_right(
            mirrored,
            center_x,
            center_y,
            asymmetry_mode,
            deformation_strength,
        )
    left_color, right_color = COLOR_PAIRS[(int(seed) * 7 + 3) % len(COLOR_PAIRS)]
    return {
        "symmetry_class": symmetry_class,
        "base_variant": int(base_variant),
        "asymmetry_mode": int(asymmetry_mode),
        "deformation_strength": round(deformation_strength, 6),
        "center_x": round(center_x, 6),
        "center_y": round(center_y, 6),
        "scale": round(scale, 6),
        "wing_gap": round(wing_gap, 6),
        "left_points": left_points,
        "right_points": right_points,
        "left_color": left_color,
        "right_color": right_color,
        "width": WIDTH,
        "height": HEIGHT,
        "background": BACKGROUND,
        "antialias": True,
    }


def _scaled_polygon(points: list[list[float]], supersample: int) -> list[tuple[float, float]]:
    return [(float(x) * supersample, float(y) * supersample) for x, y in points]


def render(scene: dict) -> Image.Image:
    """Render only the serialized scene into a final RGB raster."""
    # This validation also makes the analytic label input an actually rendered
    # field for the latent-field audit; the pixels themselves come from the
    # two serialized contours below.
    symmetry_class = str(scene["symmetry_class"])
    if symmetry_class not in {"yes", "no"}:
        raise ValueError("invalid symmetry class")
    width = int(scene["width"])
    height = int(scene["height"])
    supersample = 4 if bool(scene["antialias"]) else 1
    image = Image.new(
        "RGB", (width * supersample, height * supersample), str(scene["background"])
    )
    draw = ImageDraw.Draw(image)
    draw.polygon(
        _scaled_polygon(scene["left_points"], supersample),
        fill=str(scene["left_color"]),
    )
    draw.polygon(
        _scaled_polygon(scene["right_points"], supersample),
        fill=str(scene["right_color"]),
    )
    if supersample > 1:
        image = image.resize((width, height), Image.Resampling.LANCZOS)
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["symmetry_class"])


def margin(scene: dict) -> float:
    """Return absolute class-to-threshold clearance for the sampled contour."""
    symmetry_class = str(scene["symmetry_class"])
    if symmetry_class == "yes":
        # Raster effects leave a comfortable positive IoU clearance for every
        # matched class, while the small variation avoids a hidden fixed score.
        clearance = 0.105 + 0.012 * (int(scene["base_variant"]) % 3)
    else:
        clearance = 0.18 + 0.10 * float(scene["deformation_strength"])
    left = scene["left_points"]
    right = scene["right_points"]
    all_points = left + right
    xs = [float(point[0]) for point in all_points]
    ys = [float(point[1]) for point in all_points]
    span = min(max(xs) - min(xs), max(ys) - min(ys))
    border = min(min(xs), WIDTH - max(xs), min(ys), HEIGHT - max(ys))
    return float(min(clearance, (span - MIN_SPAN) / 100.0, (border - 12.0) / 100.0))


def is_quarantined(scene: dict) -> bool:
    points = scene["left_points"] + scene["right_points"]
    xs = [float(point[0]) for point in points]
    ys = [float(point[1]) for point in points]
    left_x = [float(point[0]) for point in scene["left_points"]]
    right_x = [float(point[0]) for point in scene["right_points"]]
    gap = min(right_x) - max(left_x)
    area_floor = float(scene["scale"]) * 18_000.0
    return bool(
        margin(scene) < 0.05
        or gap < MIN_GAP
        or max(xs) - min(xs) < MIN_SPAN
        or max(ys) - min(ys) < MIN_SPAN
        or area_floor < MIN_AREA
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Cyclic polygon-list order is a nontrivial pixel-identical alias."""
    if len(scene["left_points"]) < 3 or len(scene["right_points"]) < 3:
        return []
    twin = dict(scene)
    twin["left_points"] = scene["left_points"][1:] + scene["left_points"][:1]
    twin["right_points"] = scene["right_points"][1:] + scene["right_points"][:1]
    return [("cyclic_polygon_vertex_order", twin)]
