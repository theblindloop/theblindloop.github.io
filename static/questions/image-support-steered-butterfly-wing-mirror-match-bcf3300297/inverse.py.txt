"""Independent image-only inverse for the visible wing mirror comparison."""

from __future__ import annotations

import numpy as np
from PIL import Image

_BACKGROUND_BORDER = 6
_FOREGROUND_DISTANCE = 18.0
_MIN_COMPONENT_AREA = 7000
_MIN_COMPONENT_FILL = 0.34
_THRESHOLD = 0.92


class OracleAbstain(ValueError):
    """The raster does not expose two complete, separated wing components."""


def _foreground(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    border = np.concatenate(
        (
            array[:_BACKGROUND_BORDER].reshape(-1, 3),
            array[-_BACKGROUND_BORDER:].reshape(-1, 3),
            array[:, :_BACKGROUND_BORDER].reshape(-1, 3),
            array[:, -_BACKGROUND_BORDER:].reshape(-1, 3),
        ),
        axis=0,
    )
    background = np.median(border, axis=0).astype(np.float32)
    distance = np.linalg.norm(array.astype(np.float32) - background, axis=2)
    return distance > _FOREGROUND_DISTANCE


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    components: list[np.ndarray] = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if visited[y0, x0]:
            continue
        stack = [(y0, x0)]
        visited[y0, x0] = True
        pixels: list[tuple[int, int]] = []
        while stack:
            y, x = stack.pop()
            pixels.append((y, x))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dy == 0 and dx == 0:
                        continue
                    ny, nx = y + dy, x + dx
                    if (
                        0 <= ny < height
                        and 0 <= nx < width
                        and mask[ny, nx]
                        and not visited[ny, nx]
                    ):
                        visited[ny, nx] = True
                        stack.append((ny, nx))
        if len(pixels) >= _MIN_COMPONENT_AREA:
            component = np.zeros_like(mask, dtype=bool)
            ys, xs = zip(*pixels)
            component[np.asarray(ys), np.asarray(xs)] = True
            components.append(component)
    components.sort(key=lambda item: int(item.sum()), reverse=True)
    return components


def _component_fill(component: np.ndarray) -> float:
    ys, xs = np.nonzero(component)
    if len(xs) == 0:
        return 0.0
    box_area = (int(xs.max()) - int(xs.min()) + 1) * (int(ys.max()) - int(ys.min()) + 1)
    return float(component.sum()) / float(box_area)


def _measure(image: Image.Image) -> float:
    components = _components(_foreground(image))
    if len(components) != 2:
        raise OracleAbstain(f"expected two wing components, found {len(components)}")
    if any(_component_fill(component) < _MIN_COMPONENT_FILL for component in components):
        raise OracleAbstain("one wing is too incomplete for a stable mask")
    left, right = sorted(
        components[:2],
        key=lambda component: float(np.nonzero(component)[1].mean()),
    )
    left_y, left_x = np.nonzero(left)
    right_y, right_x = np.nonzero(right)
    if float(right_x.mean() - left_x.mean()) < 35.0:
        raise OracleAbstain("wing centroids are not visibly separated")
    axis = 0.5 * (float(left_x.mean()) + float(right_x.mean()))
    reflected = np.zeros_like(left, dtype=bool)
    reflected_x = np.rint(2.0 * axis - left_x).astype(int)
    valid = (reflected_x >= 0) & (reflected_x < reflected.shape[1])
    reflected[left_y[valid], reflected_x[valid]] = True
    intersection = int(np.logical_and(reflected, right).sum())
    union = int(np.logical_or(reflected, right).sum())
    if union <= 0:
        raise OracleAbstain("empty reflected union")
    return float(intersection) / float(union)


def decision_from_image(image: Image.Image) -> str:
    """Return yes/no using only the final rasterized PNG."""
    return "yes" if _measure(image) >= _THRESHOLD else "no"
