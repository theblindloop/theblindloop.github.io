"""Independent pixel-only inverse arm for checker_phase_reveal.

Receives ONLY the final PNG. It re-discovers the revealed glyph by:
  1. reading the mean luminance of every even-parity checkerboard cell;
  2. correlating that even-phase signal against the two declared diagonal-band
     templates A (rising) and B (falling);
  3. reporting the template with the larger signed inside-vs-outside contrast.

No scene data, no analytic gold, no renderer imports.
The extraction is polarity-invariant in that it compares relative inside/outside
contrast, so a global brightness inversion cannot flip the outcome.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "checker_phase_reveal-oracle-0.1.0"

W = 512
NGRID = 32
CELL = W // NGRID
BAND_WIDTH = 5


def _band_masks() -> dict[str, np.ndarray]:
    r, c = np.indices((NGRID, NGRID))
    mask_a = np.abs((r + c) - (NGRID - 1)) <= BAND_WIDTH
    mask_b = np.abs(r - c) <= BAND_WIDTH
    return {"A": mask_a, "B": mask_b}


def _even_signal(image: Image.Image) -> np.ndarray:
    """(NGRID, NGRID) mean-luminance of each even-parity cell, 0 elsewhere."""
    arr = np.asarray(image.convert("L"), dtype=np.float64)
    n = NGRID
    r, c = np.indices((n, n))
    even = ((r + c) % 2) == 0
    sig = np.zeros((n, n), dtype=np.float64)
    big = arr
    for i in range(n):
        for j in range(n):
            if even[i, j]:
                cell = big[i * CELL:(i + 1) * CELL, j * CELL:(j + 1) * CELL]
                sig[i, j] = cell.mean()
    return sig


def _contrast(sig: np.ndarray, mask: np.ndarray) -> float:
    inside = sig[mask].mean()
    outside = sig[~mask].mean()
    return float(inside - outside)


def decision_from_image(image: Image.Image) -> str:
    sig = _even_signal(image)
    masks = _band_masks()
    sa = _contrast(sig, masks["A"])
    sb = _contrast(sig, masks["B"])
    return "A" if abs(sa) >= abs(sb) else "B"


def reveal_margin(image: Image.Image) -> float:
    sig = _even_signal(image)
    masks = _band_masks()
    sa = _contrast(sig, masks["A"])
    sb = _contrast(sig, masks["B"])
    return abs(abs(sa) - abs(sb))
