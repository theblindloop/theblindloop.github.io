"""checker_phase_reveal: checkerboard-phase deinterlace glyph reveal world.

Declared transform: deinterlace the checkerboard lattice by keeping only the
even-parity cells (cells whose (row + column) index is even) and downsampling
each kept cell to its mean luminance. The raw image is a statistically flat
field of grey noise because every adjacent even/odd cell pair carries opposite
polarity and averages to a uniform grey; only the even-phase subsample collapses
the masked bias to a single coarse diagonal band. The band is the glyph: rising
(A) or falling (B). The question asks which glyph the deinterlaced phase reveals.

Latent scene z = {noise, glyph, cell}. label = glyph.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RENDERER_VERSION = "checker_phase_reveal-0.1.0"

W = 512
H = 512
NGRID = 32          # lattice cells per side
CELL = W // NGRID   # 16
BASE = 128.0
BIAS = 62.0
NOISE_SIGMA = 46.0
BAND_WIDTH = 5      # half-width of the diagonal band in grid units
Q_TOL = 0.15        # quarantine threshold on the A/B correlation margin


def band_masks() -> dict[str, np.ndarray]:
    """(NGRID, NGRID) bool masks: A rising '/', B falling '\\'."""
    r, c = np.indices((NGRID, NGRID))
    # A: rising band (down-left), center line r + c = NGRID-1.
    mask_a = np.abs((r + c) - (NGRID - 1)) <= BAND_WIDTH
    # B: falling band (down-right), center line r - c = 0.
    mask_b = np.abs(r - c) <= BAND_WIDTH
    return {"A": mask_a, "B": mask_b}


def build_scene(noise: int, glyph: str, cell: int = CELL) -> dict:
    return {
        "noise": int(noise),
        "glyph": glyph,
        "cell": int(cell),
    }


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(int(seed) & 0xFFFFFFFF)
    glyph = "A" if rng.integers(0, 2) else "B"
    return build_scene(noise=int(seed), glyph=glyph)


def _glyph_mask(glyph: str) -> np.ndarray:
    return band_masks()[glyph]


def render_matrix(scene: dict) -> np.ndarray:
    """(NGRID, NGRID) mean-luminance of each lattice cell."""
    cell = int(scene.get("cell", CELL))
    n = W // cell
    r, c = np.indices((n, n))
    even = ((r + c) % 2) == 0
    inside = _glyph_mask(scene["glyph"])[:n, :n]
    polarity = np.where(inside, 1.0, -1.0)
    sign = np.where(even, polarity, -polarity)
    val = BASE + BIAS * sign
    rng = np.random.default_rng(int(scene["noise"]) & 0xFFFFFFFF)
    val = val + rng.normal(0.0, NOISE_SIGMA, size=(n, n))
    return val


def render(scene: dict) -> Image.Image:
    cell = int(scene.get("cell", CELL))
    n = W // cell
    mat = render_matrix(scene)
    arr = np.repeat(np.repeat(mat, cell, axis=0), cell, axis=1)
    arr = np.clip(arr, 0, 255)
    canvas = np.repeat(arr[:, :, None], 3, axis=2).astype(np.uint8)
    canvas = canvas[:W, :H]
    return Image.fromarray(canvas, "RGB")


def margin(scene: dict) -> float:
    chosen = _glyph_mask(scene["glyph"])
    other = _glyph_mask("B" if scene["glyph"] == "A" else "A")
    c_filled = int(chosen.sum())
    o_filled = int(other.sum())
    corr_other = float((chosen & other).sum()) / max(1, o_filled)
    return round(1.0 - corr_other, 6)


def analytic_gold(scene: dict) -> str:
    return scene["glyph"]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < Q_TOL


def latent_symmetries(scene: dict):
    """not_applicable: every latent field (the per-cell noise seed, the answer
    glyph, the cell pitch) directly changes the rendered pixels, so no distinct
    latent vector reproduces the same image. Declared empty."""
    return iter(())
