"""Deterministic cube-net opposite-face question world."""

from __future__ import annotations

import random
from collections import deque
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 980, 620
BACKGROUND = (247, 249, 252)
PANEL = (255, 255, 255)
FACE = (232, 237, 244)
INK = (30, 35, 44)
MUTED = (91, 101, 116)
MAGENTA = (205, 45, 146)
COLOR_RGB = {
    "red": (214, 56, 68),
    "green": (38, 157, 90),
    "blue": (54, 105, 210),
}
ANSWER_COLORS = ("red", "green", "blue")

Vec = tuple[int, int, int]
Frame = tuple[Vec, Vec, Vec]  # local east, local south, outward normal


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    path = (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
        if bold
        else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
    )
    try:
        return ImageFont.truetype(path, size)
    except OSError:
        return ImageFont.load_default()


def _neg(vector: Vec) -> Vec:
    return (-vector[0], -vector[1], -vector[2])


def _step_frame(frame: Frame, dx: int, dy: int) -> Frame:
    """Fold one edge from a face to its planar grid neighbor."""

    east, south, normal = frame
    if (dx, dy) == (1, 0):
        return (_neg(normal), south, east)
    if (dx, dy) == (-1, 0):
        return (normal, south, _neg(east))
    if (dx, dy) == (0, 1):
        return (east, _neg(normal), south)
    if (dx, dy) == (0, -1):
        return (east, normal, _neg(south))
    raise ValueError("grid steps must cross one shared edge")


def _fold_frames(cells: Iterable[Iterable[int]]) -> list[Frame]:
    points = [tuple(map(int, point)) for point in cells]
    if len(points) != 6 or len(set(points)) != 6:
        raise ValueError("a cube net needs six distinct squares")
    lookup = {point: index for index, point in enumerate(points)}
    frames: dict[int, Frame] = {0: ((1, 0, 0), (0, 1, 0), (0, 0, 1))}
    queue = deque([0])
    while queue:
        index = queue.popleft()
        x, y = points[index]
        for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            neighbor = lookup.get((x + dx, y + dy))
            if neighbor is None:
                continue
            proposed = _step_frame(frames[index], dx, dy)
            if neighbor in frames:
                if frames[neighbor] != proposed:
                    raise ValueError("net has inconsistent fold orientation")
            else:
                frames[neighbor] = proposed
                queue.append(neighbor)
    if len(frames) != 6:
        raise ValueError("net is disconnected")
    ordered = [frames[index] for index in range(6)]
    normals = [frame[2] for frame in ordered]
    if len(set(normals)) != 6:
        raise ValueError("net overlaps cube faces when folded")
    return ordered


def _normalize(points: Iterable[tuple[int, int]]) -> tuple[tuple[int, int], ...]:
    values = list(points)
    min_x = min(x for x, _ in values)
    min_y = min(y for _, y in values)
    return tuple(sorted((x - min_x, y - min_y) for x, y in values))


def _dihedral_variants(
    shape: Iterable[tuple[int, int]],
) -> tuple[tuple[tuple[int, int], ...], ...]:
    variants = set()
    for reflected in (False, True):
        points = [(-x if reflected else x, y) for x, y in shape]
        for _ in range(4):
            variants.add(_normalize(points))
            points = [(-y, x) for x, y in points]
    return tuple(sorted(variants))


def _enumerate_cube_nets() -> tuple[tuple[tuple[int, int], ...], ...]:
    """Enumerate the eleven free cube nets from the fold rule itself."""

    fixed: set[tuple[tuple[int, int], ...]] = {((0, 0),)}
    for _ in range(5):
        grown: set[tuple[tuple[int, int], ...]] = set()
        for shape in fixed:
            occupied = set(shape)
            for x, y in shape:
                for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    point = (x + dx, y + dy)
                    if point not in occupied:
                        grown.add(_normalize((*shape, point)))
        fixed = grown
    free: dict[tuple[tuple[int, int], ...], tuple[tuple[int, int], ...]] = {}
    for shape in fixed:
        canonical = min(_dihedral_variants(shape))
        if canonical in free:
            continue
        try:
            _fold_frames(canonical)
        except ValueError:
            continue
        free[canonical] = canonical
    nets = tuple(sorted(free))
    if len(nets) != 11:
        raise RuntimeError(f"expected 11 cube nets, found {len(nets)}")
    return nets


NETS = _enumerate_cube_nets()


def _validate_scene(scene: dict) -> None:
    cells = scene["cells"]
    marker_index = int(scene["marker_index"])
    face_colors = scene["face_colors"]
    origin = scene["origin"]
    cell_size = int(scene["cell_size"])
    frames = _fold_frames(cells)
    if marker_index not in range(6):
        raise ValueError("marker index outside the six faces")
    if not isinstance(face_colors, list) or len(face_colors) != 6:
        raise ValueError("face_colors must align with the six cells")
    if face_colors[marker_index] is not None:
        raise ValueError("the marked face must be neutral")
    if sorted(color for color in face_colors if color is not None) != sorted(ANSWER_COLORS):
        raise ValueError("each answer color must appear exactly once")
    if not isinstance(origin, list) or len(origin) != 2:
        raise ValueError("origin must be a visible two-coordinate placement")
    if not 62 <= cell_size <= 78:
        raise ValueError("cell_size outside the robust rendering range")
    opposite_normal = _neg(frames[marker_index][2])
    opposite = [index for index, frame in enumerate(frames) if frame[2] == opposite_normal]
    if len(opposite) != 1 or face_colors[opposite[0]] not in ANSWER_COLORS:
        raise ValueError("the opposite face must carry one answer color")


def sample_scene(seed: int) -> dict:
    """Sample a balanced valid cube net from one integer seed."""

    rng = random.Random(int(seed) ^ 0xC0B3_5EED)
    template = NETS[rng.randrange(len(NETS))]
    variant = _dihedral_variants(template)[rng.randrange(len(_dihedral_variants(template)))]
    cells = [list(point) for point in variant]
    frames = _fold_frames(cells)
    marker_index = rng.randrange(6)
    marker_normal = frames[marker_index][2]
    opposite_index = next(
        index for index, frame in enumerate(frames) if frame[2] == _neg(marker_normal)
    )
    answer = ANSWER_COLORS[int(seed) % len(ANSWER_COLORS)]
    face_colors: list[str | None] = [None] * 6
    face_colors[opposite_index] = answer
    available = [index for index in range(6) if index not in {marker_index, opposite_index}]
    rng.shuffle(available)
    distractor_colors = [color for color in ANSWER_COLORS if color != answer]
    rng.shuffle(distractor_colors)
    for index, color in zip(available[:2], distractor_colors):
        face_colors[index] = color

    cell_size = (64, 70, 76)[rng.randrange(3)]
    grid_width = (max(x for x, _ in variant) + 1) * cell_size
    grid_height = (max(y for _, y in variant) + 1) * cell_size
    origin = [
        345 - grid_width // 2 + rng.randrange(-18, 19),
        338 - grid_height // 2 + rng.randrange(-14, 15),
    ]
    scene = {
        "cells": cells,
        "marker_index": marker_index,
        "face_colors": face_colors,
        "origin": origin,
        "cell_size": cell_size,
    }
    _validate_scene(scene)
    return scene


def analytic_gold(scene: dict) -> str:
    """Fold the visible adjacency graph and read the antipodal face color."""

    cells = scene["cells"]
    marker_index = int(scene["marker_index"])
    face_colors = scene["face_colors"]
    frames = _fold_frames(cells)
    target = _neg(frames[marker_index][2])
    opposite_index = next(index for index, frame in enumerate(frames) if frame[2] == target)
    answer = face_colors[opposite_index]
    if answer not in ANSWER_COLORS:
        raise ValueError("opposite face has no answer color")
    return str(answer)


def margin(scene: dict) -> float:
    """Runner-up minus best antipodal-normal residual among colored faces."""

    _validate_scene(scene)
    frames = _fold_frames(scene["cells"])
    target = _neg(frames[int(scene["marker_index"])][2])
    residuals = []
    for index, color in enumerate(scene["face_colors"]):
        if color in ANSWER_COLORS:
            normal = frames[index][2]
            residuals.append(sum((normal[axis] - target[axis]) ** 2 for axis in range(3)))
    ranked = sorted(residuals)
    return float(ranked[1] - ranked[0])


def is_quarantined(scene: dict) -> bool:
    """Malformed or sub-two-unit orientation gaps are explicitly excluded."""

    try:
        return margin(scene) < 2.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Exercise the local-grid-origin gauge without changing any pixel."""

    shift_x, shift_y = 2, -1
    cell_size = int(scene["cell_size"])
    twin = {
        "cells": [
            [int(x) + shift_x, int(y) + shift_y] for x, y in scene["cells"]
        ],
        "marker_index": int(scene["marker_index"]),
        "face_colors": list(scene["face_colors"]),
        "origin": [
            int(scene["origin"][0]) - shift_x * cell_size,
            int(scene["origin"][1]) - shift_y * cell_size,
        ],
        "cell_size": cell_size,
    }
    return [("local_grid_origin_gauge", twin)]


def _round_rectangle(
    draw: ImageDraw.ImageDraw,
    box: tuple[int, int, int, int],
    *,
    radius: int,
    fill: tuple[int, int, int],
    outline: tuple[int, int, int] | None = None,
    width: int = 1,
) -> None:
    draw.rounded_rectangle(box, radius=radius, fill=fill, outline=outline, width=width)


def render(scene: dict) -> Image.Image:
    """Render the complete question world as one deterministic RGB raster."""

    _validate_scene(scene)
    cells = [tuple(point) for point in scene["cells"]]
    marker_index = int(scene["marker_index"])
    face_colors = scene["face_colors"]
    origin_x, origin_y = map(int, scene["origin"])
    cell_size = int(scene["cell_size"])

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((40, 24), "CUBE-NET OPPOSITE FACE", fill=INK, font=_font(25, True))
    draw.text(
        (40, 60),
        "Fold the six edge-joined squares into a cube.",
        fill=MUTED,
        font=_font(17),
    )
    draw.text(
        (40, 84),
        "Which colored face is opposite the magenta-ringed face?",
        fill=MUTED,
        font=_font(17),
    )

    _round_rectangle(draw, (34, 126, 700, 574), radius=18, fill=PANEL, outline=(215, 221, 230), width=2)
    _round_rectangle(draw, (722, 126, 946, 574), radius=18, fill=PANEL, outline=(215, 221, 230), width=2)
    draw.text((58, 146), "NET", fill=INK, font=_font(19, True))
    draw.text((746, 146), "ANSWER COLORS", fill=INK, font=_font(17, True))

    for index, (grid_x, grid_y) in enumerate(cells):
        x0 = origin_x + grid_x * cell_size
        y0 = origin_y + grid_y * cell_size
        x1 = x0 + cell_size
        y1 = y0 + cell_size
        color_name = face_colors[index]
        fill = FACE if color_name is None else COLOR_RGB[color_name]
        draw.rectangle((x0, y0, x1, y1), fill=fill, outline=INK, width=4)
        if index == marker_index:
            center_x = (x0 + x1) // 2
            center_y = (y0 + y1) // 2
            radius = max(14, cell_size // 4)
            draw.ellipse(
                (center_x - radius, center_y - radius, center_x + radius, center_y + radius),
                outline=MAGENTA,
                width=7,
            )
            draw.ellipse(
                (center_x - 3, center_y - 3, center_x + 3, center_y + 3),
                fill=MAGENTA,
            )

    legend_y = 210
    for offset, color_name in enumerate(ANSWER_COLORS):
        y0 = legend_y + offset * 92
        _round_rectangle(
            draw,
            (758, y0, 824, y0 + 54),
            radius=9,
            fill=COLOR_RGB[color_name],
            outline=INK,
            width=2,
        )
        draw.text((842, y0 + 13), color_name.upper(), fill=INK, font=_font(17, True))

    draw.text((746, 500), "Edge joins fold;", fill=MUTED, font=_font(15))
    draw.text((746, 522), "marks stay on faces.", fill=MUTED, font=_font(15))
    return image
