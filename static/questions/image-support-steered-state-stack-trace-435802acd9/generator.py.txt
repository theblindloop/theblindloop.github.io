"""Renderer and analytic arm for the connected LIFO stack-trace world."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "state-stack-trace-0.1.0"
CANVAS = (640, 480)
STATION_COUNT = 8
TOKEN_COUNT = 3
TOKEN_NAMES = ("A", "B", "C")

BACKGROUND = (249, 251, 253)
ROUTE = (37, 57, 76)
OPERATION = (214, 92, 58)
START = (37, 145, 94)
FINISH = (127, 84, 159)
DISTRACTOR = (163, 171, 181)
TOKEN_COLORS = ((218, 74, 86), (47, 125, 183), (232, 166, 56))


def _token_palette(scene: dict[str, Any]) -> dict[int, tuple[int, int, int]]:
    values = scene["token_palette"]
    return {int(key): tuple(int(channel) for channel in value) for key, value in values.items()}


def _replay(initial_stack: list[int], operations: list[dict[str, Any]]) -> int:
    """Replay a stack without consulting any geometric scene fields."""

    stack = [int(value) for value in initial_stack]
    if not stack:
        raise ValueError("empty initial stack")
    for operation in operations:
        kind = str(operation["kind"])
        if kind == "push":
            stack.append(int(operation["token"]))
        elif kind == "pop":
            if len(stack) < 2:
                raise ValueError("pop would empty the stack")
            stack.pop()
        elif kind == "swap":
            if len(stack) < 2:
                raise ValueError("swap needs two badges")
            stack[-1], stack[-2] = stack[-2], stack[-1]
        else:
            raise ValueError(f"unknown stack operation: {kind}")
    return int(stack[-1])


def _sample_route(rng: random.Random) -> list[list[int]]:
    """Return an x-ordered serpentine route contained in the central field."""

    x0 = 126 + rng.randint(-7, 7)
    y0 = 240 + rng.randint(-7, 7)
    offsets = [-72, 70, -68, 73, -70, 68, -58, 0]
    points = [[x0, y0]]
    for index, offset in enumerate(offsets):
        x = x0 + 44 * (index + 1)
        y = y0 + offset + rng.randint(-4, 4)
        points.append([x, y])
    points.append([x0 + 44 * (STATION_COUNT + 1), y0])
    return points


def _apply_operation(stack: list[int], operation: dict[str, Any]) -> None:
    kind = str(operation["kind"])
    if kind == "push":
        stack.append(int(operation["token"]))
    elif kind == "pop":
        stack.pop()
    else:
        stack[-1], stack[-2] = stack[-2], stack[-1]


def _sample_operations(rng: random.Random, initial: list[int], target: int) -> list[dict[str, Any]]:
    """Sample valid, mixed stack programs with all three operation kinds."""

    for _ in range(4000):
        stack = list(initial)
        operations: list[dict[str, Any]] = []
        counts = {"push": 0, "pop": 0, "swap": 0}
        for index in range(STATION_COUNT):
            choices = ["push"]
            if len(stack) >= 2:
                choices.extend(("pop", "swap"))
            if index >= STATION_COUNT - 2 and counts["pop"] < 2:
                choices = ["pop"] if len(stack) >= 2 else ["push"]
            if index == STATION_COUNT - 1 and counts["swap"] == 0 and len(stack) >= 2:
                choices = ["swap"]
            kind = rng.choice(choices)
            if kind == "push":
                operation = {"kind": "push", "token": rng.randrange(TOKEN_COUNT)}
            else:
                operation = {"kind": kind}
            operations.append(operation)
            _apply_operation(stack, operation)
            counts[kind] += 1
        if (
            counts["push"] >= 2
            and counts["pop"] >= 2
            and counts["swap"] >= 1
            and stack
            and int(stack[-1]) == target
        ):
            return operations
    raise RuntimeError("could not sample a balanced stack program")


def _sample_distractors(rng: random.Random) -> list[dict[str, int]]:
    anchors = [(44, 48), (596, 48), (44, 432), (596, 432)]
    rng.shuffle(anchors)
    return [
        {
            "kind": index % 2,
            "x": x + rng.randint(-7, 7),
            "y": y + rng.randint(-7, 7),
            "size": rng.randint(7, 11),
        }
        for index, (x, y) in enumerate(anchors)
    ]


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced and raster-unambiguous stack trace."""

    rng = random.Random(int(seed))
    route_points = _sample_route(rng)
    target = int(seed) % TOKEN_COUNT
    for _ in range(4000):
        initial_stack = [rng.randrange(TOKEN_COUNT) for _ in range(rng.randint(3, 5))]
        try:
            operations = _sample_operations(rng, initial_stack, target)
        except RuntimeError:
            continue
        break
    else:
        raise RuntimeError("could not sample an initial stack")

    color_values = list(TOKEN_COLORS)
    rng.shuffle(color_values)
    token_palette = {str(token): list(color_values[token]) for token in range(TOKEN_COUNT)}
    return {
        "initial_stack": initial_stack,
        "operations": operations,
        "token_palette": token_palette,
        "route_points": route_points,
        "distractors": _sample_distractors(rng),
        "route_width": 6,
        "marker_size": 13,
        "canvas": list(CANVAS),
    }


def _draw_token(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    token: int,
    color: tuple[int, int, int],
    size: int = 8,
) -> None:
    x, y = center
    if int(token) == 0:
        draw.ellipse((x - size, y - size, x + size, y + size), fill=color)
    elif int(token) == 1:
        draw.rectangle((x - size, y - size, x + size, y + size), fill=color)
    else:
        draw.polygon([(x, y - size), (x + size, y + size), (x - size, y + size)], fill=color)


def _token_position(route_points: list[list[int]], index: int, token: int) -> tuple[int, int]:
    x, y = route_points[index + 1]
    # The small payload sits above and to the right of its station, without
    # touching the orange marker or the neighboring station.
    return int(x + 20), int(y - 23)


def _draw_station(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    kind: str,
    marker_size: int,
) -> None:
    x, y = center
    if kind == "push":
        draw.ellipse(
            (x - marker_size, y - marker_size, x + marker_size, y + marker_size),
            fill=OPERATION,
        )
    elif kind == "pop":
        draw.rectangle(
            (x - marker_size, y - marker_size, x + marker_size, y + marker_size),
            outline=OPERATION,
            width=3,
        )
    else:
        corners = [(x, y - marker_size), (x + marker_size, y), (x, y + marker_size), (x - marker_size, y)]
        draw.line(corners + [corners[0]], fill=OPERATION, width=3, joint="curve")
        draw.line((x - 8, y - 8, x + 8, y + 8), fill=OPERATION, width=3)
        draw.line((x - 8, y + 8, x + 8, y - 8), fill=OPERATION, width=3)


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the stack, one connected trace, and its visible operations."""

    width, height = (int(value) for value in scene["canvas"])
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    route_points = [[int(point[0]), int(point[1])] for point in scene["route_points"]]
    route_width = int(scene["route_width"])
    marker_size = int(scene["marker_size"])
    token_palette = _token_palette(scene)

    for distractor in scene["distractors"]:
        x, y, size = (int(distractor[key]) for key in ("x", "y", "size"))
        if int(distractor["kind"]) == 0:
            draw.ellipse((x - size, y - size, x + size, y + size), outline=DISTRACTOR, width=3)
        else:
            draw.polygon(
                [(x, y - size), (x + size, y), (x, y + size), (x - size, y)],
                outline=DISTRACTOR,
                width=3,
            )

    draw.line([tuple(point) for point in route_points], fill=ROUTE, width=route_width, joint="curve")

    start_x, start_y = route_points[0]
    stack_x = start_x - 27
    stack_values = [int(value) for value in scene["initial_stack"]]
    stack_center_y = start_y
    half_height = max(30, 11 * (len(stack_values) - 1) + 14)
    draw.ellipse(
        (stack_x - 17, stack_center_y - half_height, stack_x + 17, stack_center_y + half_height),
        outline=ROUTE,
        width=3,
    )
    draw.line((stack_x + 17, start_y, start_x, start_y), fill=ROUTE, width=route_width)
    for index, token in enumerate(stack_values):
        y = start_y + 11 * (len(stack_values) - 1) - 22 * index
        _draw_token(draw, (stack_x, int(y)), token, token_palette[token])

    for index, operation in enumerate(scene["operations"]):
        center = tuple(int(value) for value in route_points[index + 1])
        kind = str(operation["kind"])
        _draw_station(draw, center, kind, marker_size)
        if kind == "push":
            payload = _token_position(route_points, index, int(operation["token"]))
            _draw_token(draw, payload, int(operation["token"]), token_palette[int(operation["token"])])

    # Generic endpoints expose traversal direction but never encode the answer.
    draw.ellipse(
        (start_x - marker_size, start_y - marker_size, start_x + marker_size, start_y + marker_size),
        fill=START,
    )
    finish_x, finish_y = route_points[-1]
    draw.rectangle(
        (finish_x - marker_size, finish_y - marker_size, finish_x + marker_size, finish_y + marker_size),
        fill=FINISH,
    )
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the final top label using only rendered label-bearing fields."""

    token = _replay(scene["initial_stack"], scene["operations"])
    return TOKEN_NAMES[token]


def margin(scene: dict[str, Any]) -> float:
    """Continuous clearance between route stations and stacked badges."""

    points = [tuple(float(value) for value in point) for point in scene["route_points"]]
    marker_size = float(scene["marker_size"])
    clearances = [
        math.dist(points[index], points[index + 1]) - 2.0 * marker_size
        for index in range(1, len(points) - 2)
    ]
    clearances.append(22.0 - 16.0)
    return round(min(clearances), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 4.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict]]:
    # No nontrivial label-preserving pixel-identical transform is claimed: the
    # stack order and station order are both visible and answer-bearing.
    return []
