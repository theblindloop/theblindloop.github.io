"""Independent image-only inverse arm based on raster components."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _x_separated_components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    """Recover components separated by at least one blank raster column."""
    occupied = np.flatnonzero(mask.any(axis=0))
    if occupied.size == 0:
        return []
    breaks = np.flatnonzero(np.diff(occupied) > 1) + 1
    groups = np.split(occupied, breaks)
    found = []
    for group in groups:
        x0, x1 = int(group[0]), int(group[-1])
        ys, xs = np.where(mask[:, x0 : x1 + 1])
        if ys.size:
            found.append((x0 + int(xs.min()), int(ys.min()), x0 + int(xs.max()), int(ys.max()), int(ys.size)))
    return found


def _near(array: np.ndarray, colors: tuple[tuple[int, int, int], ...], tolerance: int = 18) -> np.ndarray:
    signed = array.astype(np.int16)
    masks = [np.max(np.abs(signed - np.asarray(color, dtype=np.int16)), axis=2) <= tolerance for color in colors]
    return np.logical_or.reduce(masks)


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    width, height = rgb.size
    if width < 200 or height < 150:
        raise ValueError("raster is too small")

    fill_colors = ((24, 112, 205), (18, 151, 105))
    operation_color = ((190, 45, 116),)
    array = np.asarray(rgb)
    fill_mask = _near(array, fill_colors)
    op_mask = _near(array, operation_color)

    fills = [box for box in _x_separated_components(fill_mask) if box[4] > 1800 and box[2] - box[0] > 45 and box[3] - box[1] > 35]
    operations = [box for box in _x_separated_components(op_mask) if box[4] > 400 and box[2] - box[0] > 35]
    if len(fills) > 5:
        raise ValueError("too many meter fills")
    if len(operations) != 6:
        raise ValueError(f"expected six operation glyphs, found {len(operations)}")

    operations.sort(key=lambda box: (box[0] + box[2]) / 2)
    value = len(fills)
    for x0, y0, x1, y1, area in operations:
        glyph_width = x1 - x0 + 1
        glyph_height = y1 - y0 + 1
        if glyph_height >= 0.72 * glyph_width:
            value = min(5, value + 1)
        elif glyph_height <= 0.40 * glyph_width:
            value = max(0, value - 1)
        else:
            raise ValueError("ambiguous operation glyph")
    return str(value)
