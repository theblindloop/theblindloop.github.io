"""Deterministic latent sampler and Pillow renderer."""

from __future__ import annotations

import itertools
import random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 900, 400
CAPACITY = 5
FILL_COLORS = ((24, 112, 205), (18, 151, 105))
INK = (34, 39, 48)
OPERATION = (190, 45, 116)
CARD = (244, 238, 246)
BACKGROUND = (252, 251, 248)


def _replay(initial: int, operations: list[str]) -> int:
    value = initial
    for operation in operations:
        if operation == "+":
            value = min(CAPACITY, value + 1)
        elif operation == "-":
            value = max(0, value - 1)
        else:
            raise ValueError(f"invalid operation {operation!r}")
    return value


def _has_saturation(initial: int, operations: list[str]) -> bool:
    value = initial
    saturated = False
    for operation in operations:
        before = value
        value = min(CAPACITY, value + 1) if operation == "+" else max(0, value - 1)
        saturated |= value == before
    return saturated


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    target = seed % (CAPACITY + 1)
    choices = []
    for initial in range(CAPACITY + 1):
        for bits in itertools.product(("-", "+"), repeat=6):
            operations = list(bits)
            if (
                _replay(initial, operations) == target
                and "+" in operations
                and "-" in operations
                and _has_saturation(initial, operations)
                and _replay(initial, list(reversed(operations))) != target
            ):
                choices.append((initial, operations))
    initial, operations = choices[rng.randrange(len(choices))]
    return {
        "initial": initial,
        "operations": operations,
        "meter_palette": rng.randrange(len(FILL_COLORS)),
        "card_spacing": 18 + 4 * rng.randrange(5),
    }


def render(scene: dict) -> Image.Image:
    initial = int(scene["initial"])
    operations = list(scene["operations"])
    fill_color = FILL_COLORS[int(scene["meter_palette"])]
    card_spacing = int(scene["card_spacing"])
    if not 0 <= initial <= CAPACITY or len(operations) != 6:
        raise ValueError("invalid scene")

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    draw.text((58, 38), "STARTING METER", fill=INK, font=font)

    cell_w, cell_h, gap = 82, 74, 12
    meter_x, meter_y = 58, 72
    for index in range(CAPACITY):
        x0 = meter_x + index * (cell_w + gap)
        box = (x0, meter_y, x0 + cell_w, meter_y + cell_h)
        draw.rounded_rectangle(box, radius=8, fill=(255, 255, 255), outline=INK, width=5)
        if index < initial:
            draw.rounded_rectangle(
                (x0 + 10, meter_y + 10, x0 + cell_w - 10, meter_y + cell_h - 10),
                radius=4,
                fill=fill_color,
            )

    draw.text((58, 201), "OPERATIONS - READ LEFT TO RIGHT", fill=INK, font=font)
    card_w, card_h = 102, 108
    x = 58
    y = 228
    for index, operation in enumerate(operations):
        draw.rounded_rectangle((x, y, x + card_w, y + card_h), radius=10, fill=CARD, outline=INK, width=4)
        draw.text((x + 9, y + 8), str(index + 1), fill=(92, 96, 106), font=font)
        cx, cy = x + card_w // 2, y + 62
        draw.rectangle((cx - 24, cy - 6, cx + 24, cy + 6), fill=OPERATION)
        if operation == "+":
            draw.rectangle((cx - 6, cy - 24, cx + 6, cy + 24), fill=OPERATION)
        x += card_w + card_spacing
    return image


def analytic_gold(scene: dict) -> str:
    initial = int(scene["initial"])
    operations = list(scene["operations"])
    return str(_replay(initial, operations))


def margin(scene: dict) -> float:
    return float(min(24, int(scene["card_spacing"])))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
