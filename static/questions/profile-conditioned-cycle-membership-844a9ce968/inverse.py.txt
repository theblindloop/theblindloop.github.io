"""Independent pixel-only inverse arm for the cycle-membership world.

receives ONLY the final PNG. It never imports or reads the renderer, the
analytic scene, prompts, or renderer state. It recovers the decision by:

1. segmenting "ink" pixels (anything not white) and, within them, the three
   colored links by color;
2. for each colored link, temporarily erasing its pixels and dilating the
   remaining part of that link to reach its two endpoint nodes, then counting
   how many distinct 4-connected ink components of the residual network touch
   the dilated region;
3. a colored link lies ON A LOOP exactly when deleting it leaves its two
   endpoint nodes joined through the rest of the network, i.e. when only a
   single residual component touches the link's neighbourhood (a bridge, by
   contrast, leaves two separate components).

The answer is the unique colored link that lies on a loop.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "cycle_membership-oracle-0.4.0"

_WHITE_THRESHOLD = 240
_COLOR_TOLERANCE = 95.0
_ERASE_RADIUS = 3
_DILATE_RADIUS = 15
_AREA_MIN = 60
_COLORS = {
    "red": np.array([230.0, 45.0, 45.0]),
    "green": np.array([50.0, 185.0, 60.0]),
    "blue": np.array([50.0, 90.0, 230.0]),
}


def _is_white(arr: np.ndarray, threshold: int = _WHITE_THRESHOLD) -> np.ndarray:
    return arr.min(axis=-1) > threshold


def _color_mask(arr: np.ndarray, target: np.ndarray, tolerance: float) -> np.ndarray:
    diff = arr.astype(np.float32) - target
    return np.sqrt((diff * diff).sum(axis=-1)) < tolerance


def _dilate(mask: np.ndarray, radius: int) -> np.ndarray:
    out = np.zeros_like(mask)
    h, w = mask.shape
    for dy in range(-radius, radius + 1):
        for dx in range(-radius, radius + 1):
            if dx * dx + dy * dy > radius * radius:
                continue
            shifted = np.zeros_like(mask)
            src_y0 = max(0, dy)
            src_y1 = min(h, h + dy)
            src_x0 = max(0, dx)
            src_x1 = min(w, w + dx)
            dst_y0 = max(0, -dy)
            dst_y1 = dst_y0 + (src_y1 - src_y0)
            dst_x0 = max(0, -dx)
            dst_x1 = dst_x0 + (src_x1 - src_x0)
            patch = mask[src_y0:src_y1, src_x0:src_x1]
            out[dst_y0:dst_y1, dst_x0:dst_x1] |= patch
    return out


def _components(ink: np.ndarray) -> np.ndarray:
    """Label 4-connected components of an ink mask. Returns int label array."""
    h, w = ink.shape
    labels = np.zeros((h, w), dtype=np.int32)
    current = 1
    stack: list[tuple[int, int]] = []
    for y in range(h):
        for x in range(w):
            if ink[y, x] and labels[y, x] == 0:
                labels[y, x] = current
                stack.append((x, y))
                while stack:
                    px, py = stack.pop()
                    for nx, ny in ((px + 1, py), (px - 1, py), (px, py + 1), (px, py - 1)):
                        if 0 <= nx < w and 0 <= ny < h and ink[ny, nx] and labels[ny, nx] == 0:
                            labels[ny, nx] = current
                            stack.append((nx, ny))
                current += 1
    return labels


def _link_on_loop(arr: np.ndarray, ink: np.ndarray, color: str) -> bool:
    target = _COLORS[color]
    cmask = _color_mask(arr, target, _COLOR_TOLERANCE)
    if not cmask.any():
        return False
    removed = ink & ~_dilate(cmask, _ERASE_RADIUS)
    if not removed.any():
        return False
    labels = _components(removed)
    sizes = np.bincount(labels.ravel(), minlength=labels.max() + 1)
    region = _dilate(cmask, _DILATE_RADIUS) & removed
    ids = set()
    for value in np.unique(labels[region]):
        value = int(value)
        if value == 0:
            continue
        if sizes[value] >= _AREA_MIN:
            ids.add(value)
    return len(ids) == 1


def decision_from_image(image: Image.Image) -> str:
    """Color of the unique colored link that lies on a loop, from pixels alone."""
    arr = np.asarray(image.convert("RGB")).astype(np.float32)
    ink = ~_is_white(arr)
    for color in ("red", "green", "blue"):
        if _link_on_loop(arr, ink, color):
            return color
    return "red"
