"""Cycle-membership link world renderer.

A small network of thick links connects circular nodes on a white background.
The network contains exactly one closed cycle (a ring) together with a few
"leaf" chains hanging off it; every non-ring link is a bridge. Three links of
the network are drawn in three distinct colors (red, green, blue); the rest are
black. Exactly one colored link lies ON A LOOP, meaning it is part of the
single closed cycle of the network: an edge lies on a loop exactly when, after
deleting that edge, its two endpoint nodes are still connected to each other
through the remaining links (i.e. the edge is a non-bridge).

Decision : which of the three colored links (red/green/blue) lies on a loop.

The scene is a plain dict so the verifier's AST reachability audit is free of
side effects.

    z = {
        "width": 500,
        "height": 500,
        "node_r": float,          # node circle radius (pixels)
        "link_w": float,          # link stroke width (pixels)
        "nodes": [[x, y], ...],   # absolute node centre coordinates
        "edges": [[i, j], ...],   # undirected node-index pairs (incl. colored)
        "color_edges": {"red": e, "green": e, "blue": e},  # edge index per color
    }
"""

from __future__ import annotations

import math
import random
from collections.abc import Iterator

from PIL import Image, ImageDraw

RENDERER_VERSION = "cycle_membership-0.4.0"

WHITE = (255, 255, 255)
BLACK = (25, 25, 25)
_COLOR_RGB = {"red": (230, 45, 45), "green": (50, 185, 60), "blue": (50, 90, 230)}


def _vertex(center: tuple[float, float], angle: float, radius: float) -> list[float]:
    cx, cy = center
    return [cx + radius * math.cos(angle), cy + radius * math.sin(angle)]


def _pt_dist(a, b) -> float:
    return math.hypot(a[0] - b[0], a[1] - b[1])


def _pick_leaf_verts(k: int, edge_ends: tuple[int, int], angles: list[float]) -> list[int]:
    """Choose two ring vertices for leaves that keep the printed leaves spread."""
    cands = [i for i in range(k) if i not in edge_ends]

    def angular(i: int, j: int) -> float:
        d = abs(angles[i] - angles[j])
        d = min(d, 2.0 * math.pi - d)
        return d

    if len(cands) == 2:
        return cands
    best = None
    for ia in range(len(cands)):
        for ib in range(ia + 1, len(cands)):
            d = angular(cands[ia], cands[ib])
            if best is None or d > best[0]:
                best = (d, cands[ia], cands[ib])
    return [best[1], best[2]]


def sample_scene(
    rng: random.Random,
    width: int = 500,
    height: int = 500,
) -> dict:
    """Sample a network with exactly one closed cycle and three colored links."""
    cx = width / 2.0
    cy = height / 2.0
    node_r = 11.0
    link_w = 13.0
    ring_r = rng.uniform(118.0, 142.0)
    k = rng.choice([4, 5])
    rot = rng.uniform(0.0, 2.0 * math.pi / k)
    angles = [rot + 2.0 * math.pi * i / k for i in range(k)]

    ring = [_vertex((cx, cy), a, ring_r) for a in angles]
    jc = rng.randrange(k)
    edge_ends = (jc, (jc + 1) % k)
    leaf_verts = _pick_leaf_verts(k, edge_ends, angles)

    nodes = [list(p) for p in ring]
    edges: list[list[int]] = []
    for i in range(k):
        edges.append([i, (i + 1) % k])

    leaf_r = ring_r + rng.uniform(85.0, 102.0)
    leaf_edge_idx: list[int] = []
    for vi in leaf_verts:
        lidx = len(nodes)
        nodes.append(_vertex((cx, cy), angles[vi], leaf_r))
        edges.append([vi, lidx])
        leaf_edge_idx.append(len(edges) - 1)

    roles = [("cyclic", jc)] + [(f"bridge{i + 1}", ei) for i, ei in enumerate(leaf_edge_idx)]
    rng.shuffle(roles)
    color_edges: dict[str, int] = {}
    for color, eidx in zip(("red", "green", "blue"), roles):
        color_edges[color] = eidx[1]

    return {
        "width": width,
        "height": height,
        "node_r": node_r,
        "link_w": link_w,
        "nodes": nodes,
        "edges": edges,
        "color_edges": color_edges,
    }


def _is_bridge(nodes: list, edges: list, skip: int) -> bool:
    """Whether deleting edge ``skip`` separates its two endpoint nodes."""
    a, b = edges[skip]
    adj: list[list[int]] = [[] for _ in nodes]
    for idx, (u, v) in enumerate(edges):
        if idx == skip:
            continue
        adj[u].append(v)
        adj[v].append(u)
    seen = {a}
    stack = [a]
    while stack:
        cur = stack.pop()
        if cur == b:
            return False
        for nxt in adj[cur]:
            if nxt not in seen:
                seen.add(nxt)
                stack.append(nxt)
    return True


def render(scene: dict, supersample: int = 4) -> Image.Image:
    """Render R(z) deterministically. Pure function of the scene dict."""
    width = int(scene["width"])
    height = int(scene["height"])
    node_r = float(scene["node_r"])
    link_w = float(scene["link_w"])
    nodes = scene["nodes"]
    edges = scene["edges"]
    color_edges = scene["color_edges"]

    ss = supersample
    img = Image.new("RGB", (width * ss, height * ss), WHITE)
    draw = ImageDraw.Draw(img)
    for color, eidx in color_edges.items():
        a, b = edges[eidx]
        p1 = (int(nodes[a][0] * ss), int(nodes[a][1] * ss))
        p2 = (int(nodes[b][0] * ss), int(nodes[b][1] * ss))
        draw.line([p1, p2], fill=_COLOR_RGB[color], width=max(1, int(link_w * ss)))
    for idx, (a, b) in enumerate(edges):
        if idx in set(color_edges.values()):
            continue
        p1 = (int(nodes[a][0] * ss), int(nodes[a][1] * ss))
        p2 = (int(nodes[b][0] * ss), int(nodes[b][1] * ss))
        draw.line([p1, p2], fill=BLACK, width=max(1, int(link_w * ss)))
    for x, y in nodes:
        r = int(node_r * ss)
        draw.ellipse(
            [int(x * ss) - r, int(y * ss) - r, int(x * ss) + r, int(y * ss) + r],
            fill=BLACK,
        )
    if ss > 1:
        img = img.resize((width, height), Image.LANCZOS)
    return img


def analytic_gold(scene: dict) -> str:
    """Color of the unique colored link that lies on a loop (non-bridge)."""
    nodes = scene["nodes"]
    edges = scene["edges"]
    color_edges = scene["color_edges"]
    for color, eidx in color_edges.items():
        if not _is_bridge(nodes, edges, eidx):
            return color
    return sorted(color_edges)[0]


def _pt_seg_dist(p, a, b) -> float:
    ax, ay = a
    bx, by = b
    px, py = p
    dx = bx - ax
    dy = by - ay
    if dx == 0 and dy == 0:
        return math.hypot(px - ax, py - ay)
    t = ((px - ax) * dx + (py - ay) * dy) / (dx * dx + dy * dy)
    t = max(0.0, min(1.0, t))
    return math.hypot(px - (ax + t * dx), py - (ay + t * dy))


def _seg_seg_dist(a, b, c, d) -> float:
    return min(_pt_seg_dist(c, a, b), _pt_seg_dist(d, a, b), _pt_seg_dist(a, c, d), _pt_seg_dist(b, c, d))


def margin(scene: dict) -> float:
    """Robustness margin: min clearance between non-incident graph features.

    Guards the image-only arm's per-link region test: the smallest gap between
    any colored band and the nearest ink it is not supposed to touch. Large
    positive values mean the decision survives ordinary raster perturbation.
    """
    nodes = scene["nodes"]
    edges = scene["edges"]
    color_edges = scene["color_edges"]
    node_r = float(scene["node_r"])
    best: float = float("inf")
    for i in range(len(nodes)):
        for j in range(i + 1, len(nodes)):
            best = min(best, _pt_dist(nodes[i], nodes[j]) - 2.0 * node_r)
    for eidx in color_edges.values():
        a, b = edges[eidx]
        pa, pb = nodes[a], nodes[b]
        for j, f in enumerate(edges):
            if j == eidx:
                continue
            if f[0] in (a, b) or f[1] in (a, b):
                continue
            best = min(best, _seg_seg_dist(pa, pb, nodes[f[0]], nodes[f[1]]))
        for ni, p in enumerate(nodes):
            if ni in (a, b):
                continue
            best = min(best, _pt_seg_dist(p, pa, pb) - node_r)
    return float(best)


def is_quarantined(scene: dict) -> bool:
    """Quarantine scenes whose inter-link/feature clearance is tiny."""
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> Iterator[dict]:
    """No nontrivial pixel-identical transform: every field directly redraws.

    Permuting which color labels which candidate edge is a label-transforming
    relabel (it changes the image) that the decision tracks exactly; declared
    not_applicable.
    """
    return iter(())
