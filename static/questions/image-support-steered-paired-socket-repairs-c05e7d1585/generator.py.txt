"""Deterministic scene sampler and raster renderer for paired socket repairs."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 960, 900
BACKGROUND = [249, 247, 241]
INK = [31, 49, 68]
PANEL = (228, 229, 225)
HOLE = (255, 255, 252)
TAB_RGB = [
    (224, 76, 67),
    (42, 157, 105),
    (54, 111, 202),
    (153, 83, 184),
]
TAG_RGB = {"red": (191, 43, 50), "green": (24, 128, 71), "blue": (35, 82, 166)}


def _swapped(code: list[int], a: int, b: int) -> list[int]:
    out = list(code)
    out[a], out[b] = out[b], out[a]
    return out


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    left = list(range(4))
    right = list(range(4))
    rng.shuffle(left)
    rng.shuffle(right)
    match_row = int(seed) % 3
    other_rows = [row for row in range(3) if row != match_row]
    # Each wrong row preserves exactly one side. Consequently both distant
    # sockets are necessary, and every runner-up differs on two edge positions.
    wrong_left = _swapped(left, int(seed) % 4, (int(seed) + 1) % 4)
    wrong_right = _swapped(right, (int(seed) + 2) % 4, (int(seed) + 3) % 4)
    pairs: list[list[list[int]]] = [[[], []] for _ in range(3)]
    pairs[match_row] = [list(left), list(right)]
    pairs[other_rows[0]] = [list(left), wrong_right]
    pairs[other_rows[1]] = [wrong_left, list(right)]

    tags = ["red", "green", "blue"]
    rng.shuffle(tags)
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
        "ink": list(INK),
        "socket_codes": [left, right],
        "candidate_pairs": pairs,
        "tag_colors": tags,
        "socket_offsets": [[rng.randint(-4, 4), rng.randint(-3, 3)] for _ in range(2)],
        "row_offsets": [rng.randint(-3, 3) for _ in range(3)],
    }


def _draw_code(draw: ImageDraw.ImageDraw, cx: int, cy: int, code: list[int], *, socket: bool) -> None:
    half = 48
    if socket:
        draw.rounded_rectangle((cx - 92, cy - 92, cx + 92, cy + 92), radius=18, fill=PANEL, outline=tuple(INK), width=4)
        draw.rectangle((cx - half, cy - half, cx + half, cy + half), fill=HOLE, outline=tuple(INK), width=4)
    else:
        draw.rounded_rectangle((cx - half, cy - half, cx + half, cy + half), radius=9, fill=(242, 241, 236), outline=tuple(INK), width=4)

    # Code order is top, right, bottom, left. Tabs are deliberately detached
    # from the navy boundary so the pixel inverse can recover each color part.
    color = [TAB_RGB[int(value)] for value in code]
    draw.rounded_rectangle((cx - 17, cy - 68, cx + 17, cy - 51), radius=4, fill=color[0])
    draw.rounded_rectangle((cx + 51, cy - 17, cx + 68, cy + 17), radius=4, fill=color[1])
    draw.rounded_rectangle((cx - 17, cy + 51, cx + 17, cy + 68), radius=4, fill=color[2])
    draw.rounded_rectangle((cx - 68, cy - 17, cx - 51, cy + 17), radius=4, fill=color[3])


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    # Distant source sockets. Their wide separation is intentional support geometry.
    for index, base_x in enumerate((132, 828)):
        dx, dy = scene["socket_offsets"][index]
        _draw_code(draw, base_x + int(dx), 126 + int(dy), scene["socket_codes"][index], socket=True)

    # Neutral rails bind each ordered pair but never connect it to either socket.
    row_centers = []
    for row, base_y in enumerate((310, 560, 810)):
        cy = base_y + int(scene["row_offsets"][row])
        row_centers.append(cy)
        draw.rounded_rectangle((264, cy - 61, 748, cy + 61), radius=20, fill=PANEL)
    # Draw all foreground parts after all rails so no later background card can
    # occlude a tab near the space between rows.
    for row, cy in enumerate(row_centers):
        _draw_code(draw, 382, cy, scene["candidate_pairs"][row][0], socket=False)
        _draw_code(draw, 606, cy, scene["candidate_pairs"][row][1], socket=False)
        tag = TAG_RGB[scene["tag_colors"][row]]
        draw.ellipse((700, cy - 21, 742, cy + 21), fill=tag, outline=tuple(scene["ink"]), width=3)
    return image


def _edge_mismatches(first: list[int], second: list[int]) -> int:
    return sum(int(a) != int(b) for a, b in zip(first, second, strict=True))


def _scores(scene: dict) -> list[int]:
    left, right = scene["socket_codes"]
    return [
        8 - _edge_mismatches(left, pair[0]) - _edge_mismatches(right, pair[1])
        for pair in scene["candidate_pairs"]
    ]


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    best = max(scores)
    matches = [index for index, score in enumerate(scores) if score == best and score == 8]
    if len(matches) != 1:
        raise ValueError("scene must contain exactly one two-socket repair")
    return str(scene["tag_colors"][matches[0]])


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
