"""Deterministic renderer for a landmark-induced conjugated frame move."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = (1000, 700)
N = 5
KEY_CELL = 28
KEY_Y = 66
KEY_X = (42, 238, 548, 744)
STAMP_X, STAMP_Y, STAMP_CELL = 420, 260, 32
CANDIDATE_X = (54, 425, 796)
CANDIDATE_Y, CANDIDATE_CELL = 490, 30
ANSWERS = ("amber", "cyan", "violet")
LANDMARKS = ("red", "blue", "green")

RGB = {
    "background": (255, 255, 255),
    "paper": (247, 249, 251),
    "grid": (132, 141, 150),
    "ink": (29, 34, 40),
    "arrow": (71, 79, 91),
    "red": (214, 65, 70),
    "blue": (35, 113, 190),
    "green": (39, 151, 94),
    "amber": (226, 151, 28),
    "cyan": (20, 157, 181),
    "violet": (137, 78, 191),
}

TRANSFORMS = ("rot90", "rot180", "rot270", "flip_lr", "flip_ud", "transpose", "anti")


def _transform(point: tuple[int, int], name: str) -> tuple[int, int]:
    r, c = point
    if name == "identity":
        return r, c
    if name == "rot90":
        return c, N - 1 - r
    if name == "rot180":
        return N - 1 - r, N - 1 - c
    if name == "rot270":
        return N - 1 - c, r
    if name == "flip_lr":
        return r, N - 1 - c
    if name == "flip_ud":
        return N - 1 - r, c
    if name == "transpose":
        return c, r
    if name == "anti":
        return N - 1 - c, N - 1 - r
    raise ValueError(f"unknown transform {name}")


def _apply(points: set[tuple[int, int]], name: str) -> set[tuple[int, int]]:
    return {_transform(point, name) for point in points}


def _infer(source: dict[str, list[int]], target: dict[str, list[int]]) -> str:
    matches = []
    for name in ("identity",) + TRANSFORMS:
        if all(_transform(tuple(source[color]), name) == tuple(target[color]) for color in LANDMARKS):
            matches.append(name)
    if len(matches) != 1:
        raise ValueError("landmarks do not identify exactly one transform")
    return matches[0]


def _inverse(name: str) -> str:
    return {"identity": "identity", "rot90": "rot270", "rot180": "rot180",
            "rot270": "rot90", "flip_lr": "flip_lr", "flip_ud": "flip_ud",
            "transpose": "transpose", "anti": "anti"}[name]


def _sets(scene: dict) -> tuple[set[tuple[int, int]], set[tuple[int, int]], set[tuple[int, int]]]:
    move = _infer(scene["move_source"], scene["move_target"])
    frame = _infer(scene["frame_source"], scene["frame_target"])
    stamp = {tuple(p) for p in scene["stamp"]}
    # The stamp is already expressed in right-frame coordinates. Convert it
    # back, perform MOVE, then return it to the right frame: F M F^-1.
    correct = _apply(_apply(_apply(stamp, _inverse(frame)), move), frame)
    direct = _apply(stamp, move)
    missing_return = _apply(_apply(stamp, _inverse(frame)), move)
    return correct, direct, missing_return


def _random_key(rng: random.Random, transform: str) -> tuple[dict[str, list[int]], dict[str, list[int]]]:
    while True:
        cells = rng.sample([(r, c) for r in range(N) for c in range(N)], len(LANDMARKS))
        source = {color: [*cell] for color, cell in zip(LANDMARKS, cells)}
        target = {color: [*_transform(cell, transform)] for color, cell in zip(LANDMARKS, cells)}
        try:
            if _infer(source, target) == transform:
                return source, target
        except ValueError:
            pass


def sample_scene(seed: int) -> dict:
    rng = random.Random((int(seed) ^ 0xC04F4A6E) & 0xFFFFFFFFFFFFFFFF)
    while True:
        move, frame = rng.sample(TRANSFORMS, 2)
        source1, target1 = _random_key(rng, move)
        source2, target2 = _random_key(rng, frame)
        stamp = set(rng.sample([(r, c) for r in range(N) for c in range(N)], 5))
        provisional = {
            "move_source": source1, "move_target": target1,
            "frame_source": source2, "frame_target": target2,
            "stamp": sorted([list(p) for p in stamp]),
        }
        correct, direct, missing_return = _sets(provisional)
        patterns = (correct, direct, missing_return)
        if len({frozenset(p) for p in patterns}) == 3:
            break

    target_slot = int(seed) % 3
    distractors = [direct, missing_return]
    rng.shuffle(distractors)
    arranged = distractors
    arranged.insert(target_slot, correct)

    answer_color = ANSWERS[(int(seed) // 3) % 3]
    other_colors = [color for color in ANSWERS if color != answer_color]
    rng.shuffle(other_colors)
    tag_colors: list[str | None] = [None, None, None]
    tag_colors[target_slot] = answer_color
    fill = iter(other_colors)
    for index in range(3):
        if tag_colors[index] is None:
            tag_colors[index] = next(fill)

    provisional["candidates"] = [sorted([list(p) for p in pattern]) for pattern in arranged]
    provisional["tag_colors"] = [str(color) for color in tag_colors]
    return provisional


def _grid(draw: ImageDraw.ImageDraw, x: int, y: int, cell: int, border: tuple[int, int, int], width: int = 4) -> None:
    size = N * cell
    draw.rectangle((x, y, x + size, y + size), fill=RGB["paper"], outline=border, width=width)
    for k in range(1, N):
        draw.line((x + k * cell, y, x + k * cell, y + size), fill=RGB["grid"], width=2)
        draw.line((x, y + k * cell, x + size, y + k * cell), fill=RGB["grid"], width=2)


def _dot(draw: ImageDraw.ImageDraw, x: int, y: int, cell: int, color: tuple[int, int, int]) -> None:
    cx, cy = x + cell // 2, y + cell // 2
    radius = max(7, cell // 3)
    draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=color)


def _cell_mark(draw: ImageDraw.ImageDraw, x: int, y: int, cell: int) -> None:
    pad = max(7, cell // 4)
    draw.rounded_rectangle((x + pad, y + pad, x + cell - pad, y + cell - pad), radius=3, fill=RGB["ink"])


def _arrow(draw: ImageDraw.ImageDraw, x1: int, x2: int, y: int) -> None:
    draw.line((x1, y, x2, y), fill=RGB["arrow"], width=6)
    draw.polygon([(x2, y), (x2 - 14, y - 10), (x2 - 14, y + 10)], fill=RGB["arrow"])


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS, RGB["background"])
    draw = ImageDraw.Draw(image)
    # Explicit accesses keep the protected rendered-field audit aligned with the
    # four nested visual key inputs (dynamic scene[field] access is not inferred).
    move_source = scene["move_source"]
    move_target = scene["move_target"]
    frame_source = scene["frame_source"]
    frame_target = scene["frame_target"]
    # One bar names MOVE; two bars name FRAME without font rasterization.
    draw.rounded_rectangle((118, 28, 158, 38), radius=3, fill=RGB["ink"])
    draw.rounded_rectangle((613, 22, 653, 32), radius=3, fill=RGB["ink"])
    draw.rounded_rectangle((639, 37, 679, 47), radius=3, fill=RGB["ink"])
    for x in KEY_X:
        _grid(draw, x, KEY_Y, KEY_CELL, RGB["ink"])
    for key_index, key in enumerate((move_source, move_target, frame_source, frame_target)):
        x = KEY_X[key_index]
        for color, (r, c) in key.items():
            _dot(draw, x + int(c) * KEY_CELL, KEY_Y + int(r) * KEY_CELL, KEY_CELL, RGB[color])
    _arrow(draw, KEY_X[0] + N * KEY_CELL + 13, KEY_X[1] - 13, KEY_Y + N * KEY_CELL // 2)
    _arrow(draw, KEY_X[2] + N * KEY_CELL + 13, KEY_X[3] - 13, KEY_Y + N * KEY_CELL // 2)

    _grid(draw, STAMP_X, STAMP_Y, STAMP_CELL, RGB["ink"])
    for r, c in scene["stamp"]:
        _cell_mark(draw, STAMP_X + int(c) * STAMP_CELL, STAMP_Y + int(r) * STAMP_CELL, STAMP_CELL)
    # Beneath the right-frame stamp: FRAME points back, then MOVE, then FRAME
    # points forward. This redundantly exposes the inverse/forward sandwich.
    draw.rounded_rectangle((364, 434, 396, 442), radius=2, fill=RGB["ink"])
    draw.rounded_rectangle((364, 447, 396, 455), radius=2, fill=RGB["ink"])
    _arrow(draw, 350, 300, 444)
    draw.rounded_rectangle((425, 440, 457, 448), radius=2, fill=RGB["ink"])
    _arrow(draw, 470, 520, 444)
    draw.rounded_rectangle((538, 434, 570, 442), radius=2, fill=RGB["ink"])
    draw.rounded_rectangle((538, 447, 570, 455), radius=2, fill=RGB["ink"])
    for slot, x in enumerate(CANDIDATE_X):
        color = str(scene["tag_colors"][slot])
        _grid(draw, x, CANDIDATE_Y, CANDIDATE_CELL, RGB[color], width=8)
        for r, c in scene["candidates"][slot]:
            _cell_mark(draw, x + int(c) * CANDIDATE_CELL, CANDIDATE_Y + int(r) * CANDIDATE_CELL, CANDIDATE_CELL)
    return image


def analytic_gold(scene: dict) -> str:
    target = _sets(scene)[0]
    matches = [index for index, pattern in enumerate(scene["candidates"]) if {tuple(p) for p in pattern} == target]
    if len(matches) != 1:
        raise ValueError("no unique composed result")
    return str(scene["tag_colors"][matches[0]])


def margin(scene: dict) -> float:
    target = _sets(scene)[0]
    distances = sorted(len(target.symmetric_difference({tuple(p) for p in pattern})) for pattern in scene["candidates"])
    return float(distances[1]) if len(distances) == 3 and distances[0] == 0 else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        correct, direct, missing_return = _sets(scene)
        patterns = [{tuple(p) for p in pattern} for pattern in scene["candidates"]]
        return (len({frozenset(correct), frozenset(direct), frozenset(missing_return)}) != 3
                or len(set(map(frozenset, patterns))) != 3
                or sum(pattern == correct for pattern in patterns) != 1
                or len(set(scene["tag_colors"])) != 3 or margin(scene) < 2)
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
