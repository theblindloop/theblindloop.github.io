"""Deterministic renderer for the neighbor-difference emblem world.

The tile is deliberately a fine horizontal carrier.  Its amplitude is slightly
higher over the hidden broad mask, so the public operation (absolute horizontal
neighbor difference followed by a small mean) converts the carrier into a
readable emblem.  The inverse implementation in ``oracle.py`` is written
independently and receives only the final PNG.
"""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageColor, ImageDraw


CANVAS_SIZE = 512
MARK_SIZE = 16
MARK_GAP = 8
FILTER_SIZE = 5
TEMPLATE_SIZE = 160
BASE_TONE = 128.0
LOW_DIFF = 38.0
HIGH_DIFF = 132.0
SYMBOLS = ("wave", "ring", "hook", "crown")
MARKER_PALETTE = (
    "#d1495b",
    "#247ba0",
    "#2a9d8f",
    "#8e5bb7",
    "#e08e0b",
)


def _stroke(draw: ImageDraw.ImageDraw, points: list[tuple[int, int]], width: int) -> None:
    draw.line(points, fill=255, width=width, joint="curve")


def _symbol_mask(symbol: str, size: int) -> np.ndarray:
    """Return one broad normalized emblem mask."""

    if symbol not in SYMBOLS:
        raise ValueError(f"unknown symbol {symbol!r}")
    image = Image.new("L", (size, size), 0)
    draw = ImageDraw.Draw(image)
    width = max(7, int(round(size * 0.062)))
    p = lambda value: int(round(value * (size - 1)))

    if symbol == "wave":
        _stroke(
            draw,
            [
                (p(0.15), p(0.56)),
                (p(0.31), p(0.30)),
                (p(0.48), p(0.56)),
                (p(0.65), p(0.80)),
                (p(0.85), p(0.48)),
            ],
            width,
        )
    elif symbol == "ring":
        draw.ellipse(
            [p(0.22), p(0.18), p(0.78), p(0.76)],
            outline=255,
            width=width,
        )
        _stroke(draw, [(p(0.50), p(0.76)), (p(0.50), p(0.88))], width)
        _stroke(draw, [(p(0.38), p(0.88)), (p(0.62), p(0.88))], width)
    elif symbol == "hook":
        _stroke(
            draw,
            [
                (p(0.34), p(0.17)),
                (p(0.34), p(0.66)),
                (p(0.45), p(0.80)),
                (p(0.68), p(0.84)),
                (p(0.82), p(0.68)),
            ],
            width,
        )
        _stroke(draw, [(p(0.23), p(0.17)), (p(0.45), p(0.17))], width)
    elif symbol == "crown":
        _stroke(
            draw,
            [
                (p(0.16), p(0.73)),
                (p(0.24), p(0.28)),
                (p(0.50), p(0.56)),
                (p(0.76), p(0.28)),
                (p(0.84), p(0.73)),
                (p(0.16), p(0.73)),
            ],
            width,
        )
        _stroke(draw, [(p(0.24), p(0.84)), (p(0.76), p(0.84))], width)
    return np.asarray(image, dtype=np.uint8) > 127


def _mean_filter(values: np.ndarray) -> np.ndarray:
    radius = FILTER_SIZE // 2
    padded = np.pad(values, ((radius, radius), (radius, radius)), mode="edge")
    windows = np.lib.stride_tricks.sliding_window_view(
        padded, (FILTER_SIZE, FILTER_SIZE)
    )
    return windows.mean(axis=(-1, -2))


def _neighbor_difference(tile: np.ndarray) -> np.ndarray:
    values = tile.astype(np.float32)
    right = np.concatenate([values[:, 1:], values[:, -1:]], axis=1)
    return np.abs(values - right)


def _encoded_tile(scene: dict) -> np.ndarray:
    side = int(scene["side"])
    mask = _symbol_mask(str(scene["symbol"]), side)
    yy, xx = np.indices((side, side))
    phase = int(scene["carrier_phase"]) & 1
    carrier_sign = np.where(((xx + phase) & 1) == 0, 1.0, -1.0)
    strength = np.where(mask, 1.0, 0.35)
    amplitude = float(scene["carrier_amplitude"])
    values = BASE_TONE + amplitude * strength * carrier_sign
    return np.clip(np.rint(values), 0, 255).astype(np.uint8)


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene with nuisance fields independent of the label."""

    rng = random.Random(int(seed))
    side = (208, 224, 240)[int(seed) % 3]
    symbol = SYMBOLS[(int(seed) * 7 + 3) % len(SYMBOLS)]
    marker_color = MARKER_PALETTE[(int(seed) * 11 + 1) % len(MARKER_PALETTE)]
    margin = MARK_SIZE + MARK_GAP + 22
    x = rng.randint(margin, CANVAS_SIZE - side - margin - 1)
    y = rng.randint(margin, CANVAS_SIZE - side - margin - 1)
    return {
        "canvas_size": CANVAS_SIZE,
        "side": side,
        "x": x,
        "y": y,
        "symbol": symbol,
        "marker_color": marker_color,
        "carrier_phase": rng.randrange(2),
        "carrier_amplitude": (58.0, 64.0, 70.0)[int(seed) % 3],
    }


def render(scene: dict) -> Image.Image:
    canvas_size = int(scene["canvas_size"])
    side = int(scene["side"])
    x = int(scene["x"])
    y = int(scene["y"])
    array = np.full((canvas_size, canvas_size, 3), 255, dtype=np.uint8)
    tile = _encoded_tile(scene)
    array[y : y + side, x : x + side, :] = tile[:, :, None]

    positions = (
        (x - MARK_GAP - MARK_SIZE, y - MARK_GAP - MARK_SIZE),
        (x + side + MARK_GAP, y - MARK_GAP - MARK_SIZE),
        (x - MARK_GAP - MARK_SIZE, y + side + MARK_GAP),
        (x + side + MARK_GAP, y + side + MARK_GAP),
    )
    rgb = np.asarray(ImageColor.getrgb(str(scene["marker_color"])), dtype=np.uint8)
    for px, py in positions:
        array[py : py + MARK_SIZE, px : px + MARK_SIZE, :] = rgb
    return Image.fromarray(array, mode="RGB")


def _template_response(symbol: str) -> np.ndarray:
    mask = _symbol_mask(symbol, TEMPLATE_SIZE).astype(np.float32)
    return _mean_filter(mask)


def _classification_scores(tile: np.ndarray) -> dict[str, float]:
    transformed = _mean_filter(_neighbor_difference(tile))
    normalized = Image.fromarray(transformed.astype(np.float32), mode="F").resize(
        (TEMPLATE_SIZE, TEMPLATE_SIZE), Image.Resampling.BILINEAR
    )
    values = np.asarray(normalized, dtype=np.float32)
    strength = np.clip((values - LOW_DIFF) / (HIGH_DIFF - LOW_DIFF), 0.0, 1.0)
    return {
        symbol: float(1.0 - np.mean(np.abs(strength - _template_response(symbol))))
        for symbol in SYMBOLS
    }


def _top_margin(scores: dict[str, float]) -> float:
    values = sorted(scores.values(), reverse=True)
    return float(values[0] - values[1])


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def margin(scene: dict) -> float:
    return _top_margin(_classification_scores(_encoded_tile(scene)))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.025


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases are claimed; every generated field is rendered."""

    return []
