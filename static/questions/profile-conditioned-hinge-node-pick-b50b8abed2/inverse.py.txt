"""Pixel-only inverse arm for the hinge-node world.

Receives the final PNG and nothing else: no scene, no renderer, no gold. It
segments the grey wiring and the three coloured patches from colour alone,
reconstructs each patch as a round region from its own pixels, and then, for
each patch in turn, deletes that region from the wire mask and counts the
connected pieces of grey ink that survive. The patch whose deletion leaves more
than one piece is the hinge.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "hinge_node_pick-oracle-0.1.0"

MIN_PATCH_PIXELS = 250
MIN_COMPONENT_PIXELS = 60
ERASE_PAD_PX = 6.0
AMBIGUOUS = "ambiguous"


def _wire_mask(rgb: np.ndarray) -> np.ndarray:
    """Dark neutral ink: the grey wiring, excluding paper and coloured patches."""
    high = rgb.max(axis=2)
    low = rgb.min(axis=2)
    return (high < 150) & ((high - low) < 40)


def _patch_masks(rgb: np.ndarray) -> dict[str, np.ndarray]:
    red, green, blue = rgb[:, :, 0], rgb[:, :, 1], rgb[:, :, 2]
    return {
        "red": (red - np.maximum(green, blue) > 45) & (red > 110),
        "green": (green - np.maximum(red, blue) > 40) & (green > 90),
        "blue": (blue - np.maximum(red, green) > 45) & (blue > 120),
    }


def _patch_disc(mask: np.ndarray) -> tuple[float, float, float]:
    """Centre and radius of the round patch, recovered from its visible pixels.

    The wiring is painted across each patch, so the coloured pixels form a disc
    minus a few strips. The bounding box of that set is still the bounding box
    of the disc, which fixes the centre; the radius is taken from the furthest
    coloured pixel.
    """
    ys, xs = np.nonzero(mask)
    cx = (float(xs.min()) + float(xs.max())) / 2.0
    cy = (float(ys.min()) + float(ys.max())) / 2.0
    radius = float(np.sqrt((xs - cx) ** 2 + (ys - cy) ** 2).max())
    return cx, cy, radius


def _count_components(mask: np.ndarray) -> int:
    """Row-run union-find with 8-connectivity; tiny specks are discarded."""
    height, width = mask.shape
    parent: list[int] = []
    area: list[int] = []

    def make(size: int) -> int:
        parent.append(len(parent))
        area.append(size)
        return len(parent) - 1

    def find(a: int) -> int:
        while parent[a] != a:
            parent[a] = parent[parent[a]]
            a = parent[a]
        return a

    def union(a: int, b: int) -> None:
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[ra] = rb
            area[rb] += area[ra]

    previous: list[tuple[int, int, int]] = []
    for y in range(height):
        row = mask[y]
        if not row.any():
            previous = []
            continue
        edges = np.flatnonzero(np.diff(np.concatenate(([0], row.view(np.int8), [0]))))
        runs = list(zip(edges[0::2], edges[1::2]))
        current: list[tuple[int, int, int]] = []
        for start, stop in runs:
            label = make(int(stop - start))
            for pstart, pstop, plabel in previous:
                if pstart - 1 < stop and start - 1 < pstop:
                    union(label, plabel)
            current.append((int(start), int(stop), label))
        previous = current

    totals: dict[int, int] = {}
    for index in range(len(parent)):
        if parent[index] == index:
            totals[index] = area[index]
    return sum(1 for size in totals.values() if size >= MIN_COMPONENT_PIXELS)


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    wires = _wire_mask(rgb)
    height, width = wires.shape
    yy, xx = np.mgrid[0:height, 0:width]

    hinges: list[str] = []
    for colour, mask in sorted(_patch_masks(rgb).items()):
        if int(mask.sum()) < MIN_PATCH_PIXELS:
            return AMBIGUOUS
        cx, cy, radius = _patch_disc(mask)
        reach = radius + ERASE_PAD_PX
        keep = ((xx - cx) ** 2 + (yy - cy) ** 2) > reach * reach
        if _count_components(wires & keep) > 1:
            hinges.append(colour)
    if len(hinges) != 1:
        return AMBIGUOUS
    return hinges[0]
