"""Pair-separation selection world: which colored dot pair is farthest apart.

Latent scene z carries a set of colors, each drawn as exactly two same-colored
dot markers on a shared light panel. The task is a pure measurement: find the
color whose two dots are separated by the largest straight-line distance.

Because each color appears as exactly two disconnected disk blobs and the disks
are kept well apart, the independent inverse arm can locate each dot centroid
and measure the centre-to-centre distance, then select the color with the
largest separation. This is a measurement-driven set-argmax selection,
counterbalancing which color holds the extremum and the arbitrary spatial
layout of every pair.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "widest_pair-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
BACKGROUND = (250, 250, 250)

WIDTH = 480
HEIGHT = 320
DOT_RADIUS = 6

MAX_SEP = 150.0
MIN_SEP = 48.0
MIN_MARGIN_PX = 16.0
QUARANTINE_PX = 20.0


def sample_scene(seed: int) -> dict:
    """Sample a deterministic scene with a guaranteed, counterbalanced extremum.

    ``dots`` is a list of six {"color", "x", "y"} entries, exactly two per color,
    whose pairwise centre-to-centre separations are all distinct.
    """
    rng = random.Random(seed * 1009 + 7)
    ncols = len(PALETTE)
    while True:
        seps = [rng.uniform(MIN_SEP, MAX_SEP) for _ in range(ncols)]
        seps_sorted = sorted(seps, reverse=True)
        if seps_sorted[0] - seps_sorted[1] >= MIN_MARGIN_PX:
            break
    colors = list(PALETTE)
    rng.shuffle(colors)
    mid_margin = MAX_SEP / 2.0 + DOT_RADIUS + 8
    x0, x1 = mid_margin, WIDTH - mid_margin
    y0, y1 = mid_margin, HEIGHT - mid_margin
    dots = []
    for color, sep in zip(colors, seps):
        mx = rng.uniform(x0, x1)
        my = rng.uniform(y0, y1)
        angle = rng.uniform(0.0, 2.0 * math.pi)
        dx = (sep / 2.0) * math.cos(angle)
        dy = (sep / 2.0) * math.sin(angle)
        dots.append({"color": color, "x": round(mx - dx, 3), "y": round(my - dy, 3)})
        dots.append({"color": color, "x": round(mx + dx, 3), "y": round(my + dy, 3)})
    return {
        "dots": dots,
        "width": WIDTH,
        "height": HEIGHT,
        "background": "light",
    }


def render(scene: dict) -> Image.Image:
    """Render I = R(z): six filled dot markers, two per color, on a light panel."""
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    for dot in scene["dots"]:
        r = DOT_RADIUS
        draw.ellipse(
            [(dot["x"] - r, dot["y"] - r), (dot["x"] + r, dot["y"] + r)],
            fill=COLOR_RGB[dot["color"]],
        )
    return img


def _color_separations(dots: list[dict]) -> "dict[str, float] | None":
    """Pair up the two dots of each color and measure their centre distance."""
    by_color: dict[str, list[tuple[float, float]]] = {}
    for dot in dots:
        by_color.setdefault(dot["color"], []).append((float(dot["x"]), float(dot["y"])))
    if len(PALETTE) != len(by_color) or any(len(v) != 2 for v in by_color.values()):
        return None
    out = {}
    for color, pair in by_color.items():
        dx = pair[0][0] - pair[1][0]
        dy = pair[0][1] - pair[1][1]
        out[color] = math.hypot(dx, dy)
    return out


def analytic_gold(scene: dict) -> str:
    """The color whose two dots are separated by the largest distance."""
    seps = _color_separations(scene["dots"])
    best = max(seps.items(), key=lambda item: item[1])
    return best[0]


def margin(scene: dict) -> float:
    """Verifiability margin: the top-two separation gap in pixels."""
    seps = sorted(_color_separations(scene["dots"]).values(), reverse=True)
    return float(seps[0] - seps[1])


def is_quarantined(scene: dict) -> bool:
    """Near-boundary: the top-two separation gap is finite-resolution small."""
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list:
    """not_applicable: no nontrivial pixel-identical latent symmetry.

    Swapping which color owns a pair changes the correct answer, and any global
    reflection/rotation turns the raster into a different image. Every latent
    field (each dot's color and coordinates) is directly rendered as visible ink.
    """
    return []
