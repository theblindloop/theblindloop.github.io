"""Independent pixel-only inverse arm for the widest_pair world.

This module receives ONLY the rasterized PNG (plus the world's fixed color
naming convention) and re-measures which color's two dots are farthest apart.
It shares no geometry with renderer.render() and never imports the renderer,
prompts, generator, verifier, or any analytical gold code.
"""

from __future__ import annotations

from typing import Any

import numpy as np
from PIL import Image

ORACLE_VERSION = "widest_pair-oracle-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
BACKGROUND = (250, 250, 250)


def _nearest_color(arr: np.ndarray) -> np.ndarray:
    """Label every pixel to its nearest palette entry (dot colors + background)."""
    refs = np.stack(
        [np.array(COLOR_RGB[c], dtype=np.float32) for c in PALETTE]
        + [np.array(BACKGROUND, dtype=np.float32)],
        axis=0,
    )
    diffs = (
        arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    )
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def _components(mask: np.ndarray) -> list[tuple[float, float]]:
    """Return the centroid (y, x) of each 4-connected foreground blob.

    Processed once per colour; blobs are small, well-separated disks so a simple
    flood fill is both exact and fast.
    """
    ys, xs = np.nonzero(mask)
    if len(ys) < 4:
        return []
    visited = np.zeros(mask.shape, dtype=bool)
    centroids = []
    for start_y, start_x in zip(ys, xs):
        if visited[start_y, start_x]:
            continue
        queue = [(int(start_y), int(start_x))]
        visited[start_y, start_x] = True
        sy = 0.0
        sx = 0.0
        count = 0
        while queue:
            cy, cx = queue.pop()
            sy += cy
            sx += cx
            count += 1
            for dy, dx in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                ny, nx = cy + dy, cx + dx
                if (
                    0 <= ny < mask.shape[0]
                    and 0 <= nx < mask.shape[1]
                    and mask[ny, nx]
                    and not visited[ny, nx]
                ):
                    visited[ny, nx] = True
                    queue.append((ny, nx))
        centroids.append((sy / count, sx / count))
    return centroids


def decision_from_image(image: Image.Image) -> str:
    """Return the color whose two dots are farthest apart, from pixels alone."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _nearest_color(arr)
    best_color = PALETTE[0]
    best_dist = -1.0
    for idx, color in enumerate(PALETTE):
        mask = labels == idx
        centroids = _components(mask)
        if len(centroids) < 2:
            continue
        # Take the two most separated blobs (there are exactly two disks).
        a, b = centroids[0], centroids[1]
        for i in range(len(centroids)):
            for j in range(i + 1, len(centroids)):
                dy = centroids[i][0] - centroids[j][0]
                dx = centroids[i][1] - centroids[j][1]
                d = dy * dy + dx * dx
                if d > (b[0] - a[0]) ** 2 + (b[1] - a[1]) ** 2:
                    a, b = centroids[i], centroids[j]
        dist = ((a[0] - b[0]) ** 2 + (a[1] - b[1]) ** 2) ** 0.5
        if dist > best_dist:
            best_dist = dist
            best_color = color
    return best_color
