"""Independent image-only inverse arm for teapot appendage opposition."""

from __future__ import annotations

import numpy as np
from PIL import Image

# Public named-color appearance ranges, duplicated rather than imported.
_BLUE = ((42, 105, 160), (51, 116, 173), (35, 91, 146))
_ORANGE = ((230, 122, 44), (218, 106, 35), (241, 138, 51))
_GREEN = ((47, 145, 91), (40, 135, 84), (56, 155, 101))


def _color_mask(array: np.ndarray, prototypes: tuple[tuple[int, int, int], ...]) -> np.ndarray:
    # int32 is deliberate: squared RGB deltas overflow int16.
    pixels = array.astype(np.int32)
    distances = [np.sum((pixels - np.asarray(color, dtype=np.int32)) ** 2, axis=2) for color in prototypes]
    return np.min(np.stack(distances, axis=0), axis=0) <= 32**2


def _horizontal_center(mask: np.ndarray, label: str, minimum: int) -> float:
    ys, xs = np.nonzero(mask)
    if len(xs) < minimum:
        raise ValueError(f"abstain: {label} evidence is missing or too small")
    # A trimmed horizontal midpoint is stable to antialiasing and shape taper.
    lo, hi = np.percentile(xs, (2, 98))
    if hi - lo < 10:
        raise ValueError(f"abstain: {label} evidence is fragmented")
    return float((lo + hi) / 2.0)


def decision_from_image(image: Image.Image) -> str:
    """Recover yes/no solely from the rendered RGB pixels."""

    array = np.asarray(image.convert("RGB"))
    if array.ndim != 3 or array.shape[0] < 64 or array.shape[1] < 64:
        raise ValueError("abstain: unsupported raster")
    blue = _color_mask(array, _BLUE)
    orange = _color_mask(array, _ORANGE)
    green = _color_mask(array, _GREEN)
    body_x = _horizontal_center(blue, "blue body", 2500)
    spout_x = _horizontal_center(orange, "orange spout", 500)
    handle_x = _horizontal_center(green, "green handle", 450)
    spout_delta = spout_x - body_x
    handle_delta = handle_x - body_x
    if abs(spout_delta) < 35 or abs(handle_delta) < 35:
        raise ValueError("abstain: an appendage side is ambiguous")
    return "yes" if (spout_delta < 0) != (handle_delta < 0) else "no"
