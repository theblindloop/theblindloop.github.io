"""Deterministic renderer for the teapot appendage-opposition world."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
AA = 3

# Named-color shades are varied without changing their ordinary color labels.
PALETTES = (
    {"blue": (42, 105, 160), "orange": (230, 122, 44), "green": (47, 145, 91)},
    {"blue": (51, 116, 173), "orange": (218, 106, 35), "green": (40, 135, 84)},
    {"blue": (35, 91, 146), "orange": (241, 138, 51), "green": (56, 155, 101)},
)
BACKGROUND = (249, 247, 241)
OUTLINE = (41, 45, 53)
MARKER = (132, 91, 169)


def sample_scene(seed: int) -> dict:
    """Return a matched, balanced scene.

    Consecutive seeds form a control/counterfactual pair with identical nuisance
    parameters and spout side. Even scene seeds use opposite appendage sides;
    odd scene seeds put both appendages on the same side.
    """

    rng = random.Random((int(seed) // 2) ^ 0x54EA90)
    spout_side = rng.choice(("left", "right"))
    opposite = int(seed) % 2 == 0
    handle_side = (
        ("right" if spout_side == "left" else "left") if opposite else spout_side
    )
    return {
        "spout_side": spout_side,
        "handle_side": handle_side,
        "body_center_x": rng.randint(246, 266),
        "body_center_y": rng.randint(253, 269),
        "body_width": rng.randrange(178, 207, 2),
        "body_height": rng.randrange(138, 163, 2),
        "spout_y_offset": rng.randint(-43, -31),
        "handle_y_offset": rng.randint(19, 31),
        "palette_variant": rng.randrange(len(PALETTES)),
        "marker_variant": rng.randrange(4),
        "lid_width": rng.randrange(92, 125, 2),
        "spout_length": rng.randrange(82, 105, 2),
        "handle_extent": rng.randrange(91, 112, 2),
    }


def _sc(value: float) -> int:
    return int(round(value * AA))


def _box(box: tuple[float, float, float, float]) -> tuple[int, int, int, int]:
    return tuple(_sc(v) for v in box)  # type: ignore[return-value]


def _points(points: list[tuple[float, float]]) -> list[tuple[int, int]]:
    return [(_sc(x), _sc(y)) for x, y in points]


def render(scene: dict) -> Image.Image:
    """Render only from public scene fields; no analytic answer is embedded."""

    spout_side = scene["spout_side"]
    handle_side = scene["handle_side"]
    cx = int(scene["body_center_x"])
    cy = int(scene["body_center_y"])
    body_w = int(scene["body_width"])
    body_h = int(scene["body_height"])
    spout_y_offset = int(scene["spout_y_offset"])
    handle_y_offset = int(scene["handle_y_offset"])
    palette_variant = int(scene["palette_variant"])
    marker_variant = int(scene["marker_variant"])
    lid_width = int(scene["lid_width"])
    spout_length = int(scene["spout_length"])
    handle_extent = int(scene["handle_extent"])

    if spout_side not in {"left", "right"} or handle_side not in {"left", "right"}:
        raise ValueError("appendage sides must be left or right")
    palette = PALETTES[palette_variant]
    image = Image.new("RGB", (WIDTH * AA, HEIGHT * AA), BACKGROUND)
    draw = ImageDraw.Draw(image)
    half_w = body_w / 2
    half_h = body_h / 2

    # Small purple diamonds are balanced nuisance marks and never share a target color.
    corners = ((35, 38), (477, 38), (35, 474), (477, 474))
    for index in (marker_variant, (marker_variant + 2) % 4):
        mx, my = corners[index]
        r = 8 + (marker_variant % 2)
        draw.polygon(_points([(mx, my - r), (mx + r, my), (mx, my + r), (mx - r, my)]), fill=MARKER)

    # The loop handle is painted behind the body, leaving a broad, visibly attached arc.
    hs = -1 if handle_side == "left" else 1
    body_edge_h = cx + hs * half_w
    handle_center_x = body_edge_h + hs * (handle_extent * 0.48)
    handle_center_y = cy + handle_y_offset
    outer = (
        handle_center_x - handle_extent / 2,
        handle_center_y - 57,
        handle_center_x + handle_extent / 2,
        handle_center_y + 57,
    )
    inner = (outer[0] + 17, outer[1] + 17, outer[2] - 17, outer[3] - 17)
    draw.ellipse(_box(outer), fill=palette["green"], outline=OUTLINE, width=_sc(3))
    draw.ellipse(_box(inner), fill=BACKGROUND, outline=OUTLINE, width=_sc(3))

    # A filled triangular spout, also behind the body, with a dark visible mouth.
    ss = -1 if spout_side == "left" else 1
    body_edge_s = cx + ss * half_w
    sy = cy + spout_y_offset
    spout = [
        (body_edge_s - ss * 12, sy + 28),
        (body_edge_s - ss * 9, sy - 22),
        (body_edge_s + ss * spout_length, sy - 36),
        (body_edge_s + ss * (spout_length - 7), sy - 15),
    ]
    draw.polygon(_points(spout), fill=palette["orange"], outline=OUTLINE, width=_sc(3))
    mouth_x = body_edge_s + ss * (spout_length - 3)
    draw.line(
        _points([(mouth_x, sy - 38), (mouth_x - ss * 5, sy - 14)]),
        fill=OUTLINE,
        width=_sc(4),
    )

    # Main blue vessel body, flat base, lid, and knob make the object unmistakable.
    body_box = (cx - half_w, cy - half_h, cx + half_w, cy + half_h)
    draw.ellipse(_box(body_box), fill=palette["blue"], outline=OUTLINE, width=_sc(5))
    base_y = cy + half_h - 9
    draw.line(
        _points([(cx - half_w * 0.62, base_y), (cx + half_w * 0.62, base_y)]),
        fill=OUTLINE,
        width=_sc(5),
    )
    lid_y = cy - half_h + 5
    draw.rounded_rectangle(
        _box((cx - lid_width / 2, lid_y - 13, cx + lid_width / 2, lid_y + 8)),
        radius=_sc(8),
        fill=palette["blue"],
        outline=OUTLINE,
        width=_sc(4),
    )
    draw.ellipse(
        _box((cx - 15, lid_y - 31, cx + 15, lid_y - 8)),
        fill=palette["blue"],
        outline=OUTLINE,
        width=_sc(4),
    )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "yes" if scene["spout_side"] != scene["handle_side"] else "no"


def margin(scene: dict) -> float:
    """Runner-up gap for the two left/right side classifications, in pixels."""

    half_w = float(scene["body_width"]) / 2.0
    spout_gap = half_w + float(scene["spout_length"]) * 0.45
    handle_gap = half_w + float(scene["handle_extent"]) * 0.32
    return min(spout_gap, handle_gap)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 70.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # No pixel-identical latent alias is claimed. Geometric equivariance is tested separately.
    return []


def translated(scene: dict, dx: int, dy: int) -> dict:
    twin = deepcopy(scene)
    twin["body_center_x"] += int(dx)
    twin["body_center_y"] += int(dy)
    return twin


def reflected(scene: dict) -> dict:
    twin = deepcopy(scene)
    twin["body_center_x"] = WIDTH - int(scene["body_center_x"])
    twin["spout_side"] = "right" if scene["spout_side"] == "left" else "left"
    twin["handle_side"] = "right" if scene["handle_side"] == "left" else "left"
    return twin
