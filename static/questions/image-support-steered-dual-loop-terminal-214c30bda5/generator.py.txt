"""Deterministic renderer and analytic arm for the dual-loop route question."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
SCALE = 4
BACKGROUND = (255, 255, 255)
ROUTE_RGB = (35, 52, 70)
GREEN_RGB = (35, 169, 101)
AMBER_RGB = (229, 150, 32)
VIOLET_RGB = (146, 76, 190)
ROUTE_WIDTH = 10
MARKER_RADIUS = 11
QUARANTINE_MARGIN = 20.0


def _distance(first: tuple[float, float], second: tuple[float, float]) -> float:
    return math.hypot(first[0] - second[0], first[1] - second[1])


def _base_points(topology: str, variant: int) -> list[tuple[float, float]]:
    """Return a visibly open path or a path that traverses one rectangular cycle."""

    if topology == "loop":
        if variant == 0:
            return [
                (0.0, 55.0),
                (24.0, 55.0),
                (43.0, 55.0),
                (43.0, 16.0),
                (116.0, 16.0),
                (116.0, 96.0),
                (43.0, 96.0),
                (43.0, 55.0),
                (23.0, 101.0),
                (0.0, 110.0),
            ]
        return [
            (0.0, 60.0),
            (23.0, 60.0),
            (42.0, 60.0),
            (42.0, 20.0),
            (112.0, 20.0),
            (112.0, 91.0),
            (42.0, 91.0),
            (42.0, 60.0),
            (22.0, 105.0),
            (0.0, 111.0),
        ]
    if variant == 0:
        return [
            (0.0, 58.0),
            (22.0, 25.0),
            (49.0, 91.0),
            (76.0, 25.0),
            (103.0, 88.0),
            (128.0, 36.0),
            (145.0, 68.0),
        ]
    return [
        (0.0, 64.0),
        (22.0, 35.0),
        (51.0, 32.0),
        (77.0, 77.0),
        (104.0, 98.0),
        (128.0, 59.0),
        (145.0, 48.0),
    ]


def _transform_points(
    points: list[tuple[float, float]],
    *,
    slot: tuple[float, float],
    mirror_x: bool,
    flip_y: bool,
    scale_x: float,
    scale_y: float,
) -> list[list[float]]:
    x0, y0 = slot
    transformed: list[list[float]] = []
    for x, y in points:
        sx = float(x) * scale_x
        sy = float(y) * scale_y
        if mirror_x:
            sx = 150.0 * scale_x - sx
        if flip_y:
            sy = 122.0 * scale_y - sy
        transformed.append([round(x0 + sx, 4), round(y0 + sy, 4)])
    return transformed


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample two far-separated networks with balanced terminal identities."""

    rng = random.Random(int(seed))
    # The answer bit and the route-slot bit come from different deterministic
    # functions so terminal identity is not tied to the loop's location.
    loop_index = int(seed) % 2
    loop_terminal = "amber" if ((int(seed) // 2) % 2 == 0) else "violet"
    terminal_assignment = {
        str(loop_index): loop_terminal,
        str(1 - loop_index): "violet" if loop_terminal == "amber" else "amber",
    }
    diagonal = rng.randrange(2)
    slots = (
        [(20.0 + rng.uniform(-4.0, 4.0), 55.0 + rng.uniform(-5.0, 5.0)),
         (345.0 + rng.uniform(-4.0, 4.0), 320.0 + rng.uniform(-5.0, 5.0))]
        if diagonal == 0
        else [(20.0 + rng.uniform(-4.0, 4.0), 320.0 + rng.uniform(-5.0, 5.0)),
              (345.0 + rng.uniform(-4.0, 4.0), 55.0 + rng.uniform(-5.0, 5.0))]
    )

    networks: list[dict[str, Any]] = []
    for index, slot in enumerate(slots):
        topology = "loop" if index == loop_index else "open"
        variant = rng.randrange(2)
        scale_x = rng.uniform(0.94, 1.02)
        scale_y = rng.uniform(0.94, 1.03)
        points = _transform_points(
            _base_points(topology, variant),
            slot=slot,
            mirror_x=bool(rng.randrange(2)),
            flip_y=bool(rng.randrange(2)),
            scale_x=scale_x,
            scale_y=scale_y,
        )
        networks.append(
            {
                "network_id": f"network_{index}",
                "topology": topology,
                "points": points,
                "terminal": terminal_assignment[str(index)],
            }
        )
    return {"networks": networks}


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only visible scene fields with deterministic supersampling."""

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    route_width = ROUTE_WIDTH * SCALE
    route_radius = route_width // 2
    for network in scene["networks"]:
        points = [(round(float(x) * SCALE), round(float(y) * SCALE)) for x, y in network["points"]]
        draw.line(points, fill=ROUTE_RGB, width=route_width, joint="curve")
        for x, y in points:
            draw.ellipse((x - route_radius, y - route_radius, x + route_radius, y + route_radius), fill=ROUTE_RGB)

    terminal_rgb = {"amber": AMBER_RGB, "violet": VIOLET_RGB}
    marker_radius = MARKER_RADIUS * SCALE
    for network in scene["networks"]:
        points = network["points"]
        start = (round(float(points[0][0]) * SCALE), round(float(points[0][1]) * SCALE))
        terminal = (round(float(points[-1][0]) * SCALE), round(float(points[-1][1]) * SCALE))
        for center, color in ((start, GREEN_RGB), (terminal, terminal_rgb[str(network["terminal"])])):
            x, y = center
            draw.ellipse((x - marker_radius, y - marker_radius, x + marker_radius, y + marker_radius), fill=color)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def _points(network: dict[str, Any]) -> list[tuple[float, float]]:
    return [(float(point[0]), float(point[1])) for point in network["points"]]


def _bbox(points: list[tuple[float, float]]) -> tuple[float, float, float, float]:
    xs = [point[0] for point in points]
    ys = [point[1] for point in points]
    return min(xs), min(ys), max(xs), max(ys)


def analytic_gold(scene: dict[str, Any]) -> str:
    networks = scene["networks"]
    if len(networks) != 2 or {str(item["topology"]) for item in networks} != {"loop", "open"}:
        raise ValueError("scene must contain exactly one loop and one open network")
    loop = next(item for item in networks if item["topology"] == "loop")
    return str(loop["terminal"])


def margin(scene: dict[str, Any]) -> float:
    """Return a visible clearance margin for the topology decision."""

    loop = next(item for item in scene["networks"] if item["topology"] == "loop")
    points = _points(loop)
    x0, y0, x1, y1 = _bbox(points)
    cycle_clearance = min(x1 - x0, y1 - y0) - 2.0 * ROUTE_WIDTH
    first, second = (_bbox(_points(item)) for item in scene["networks"])
    horizontal_gap = max(first[0] - second[2], second[0] - first[2], 0.0)
    vertical_gap = max(first[1] - second[3], second[1] - first[3], 0.0)
    separation = math.hypot(horizontal_gap, vertical_gap)
    return float(min(cycle_clearance, separation if separation > 0 else 0.0))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # The scene exposes every label-bearing geometry and identity; no hidden
    # pixel-identical transform is declared.
    return []
