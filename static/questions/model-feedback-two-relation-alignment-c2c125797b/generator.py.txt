"""Deterministic renderer for two-relation identity alignment."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw, ImageFont

N = 7
WIDTH = 980
HEIGHT = 510
CELL = 44
GRID_Y = 110
LEFT_GRID_X = 105
RIGHT_GRID_X = 570
TOP_HEADER_Y = 84
LEFT_HEADER_X = 79

COLORS = ("red", "navy", "green", "purple", "gold", "teal", "pink")
LETTERS = ("A", "B", "C", "D", "E", "F", "G")

RGB = {
    "background": (248, 247, 242),
    "panel": (235, 239, 243),
    "white": (255, 255, 255),
    "ink": (27, 35, 48),
    "guide": (112, 124, 138),
    "blue_relation": (38, 108, 210),
    "orange_relation": (232, 128, 32),
    "red": (211, 61, 62),
    "navy": (47, 83, 166),
    "green": (45, 148, 84),
    "purple": (126, 74, 183),
    "gold": (222, 171, 35),
    "teal": (23, 151, 157),
    "pink": (219, 79, 142),
}

GLYPHS = {
    "A": ("010", "101", "111", "101", "101"),
    "B": ("110", "101", "110", "101", "110"),
    "C": ("011", "100", "100", "100", "011"),
    "D": ("110", "101", "101", "101", "110"),
    "E": ("111", "100", "110", "100", "111"),
    "F": ("111", "100", "110", "100", "100"),
    "G": ("011", "100", "101", "101", "011"),
}


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size=size)
    except OSError:
        return ImageFont.load_default()


def _cycle_from_order(order: list[int]) -> list[int]:
    mapping = [0] * N
    for index, source in enumerate(order):
        mapping[source] = order[(index + 1) % N]
    return mapping


def _is_permutation(values: object) -> bool:
    return (
        isinstance(values, list)
        and len(values) == N
        and all(isinstance(value, int) and not isinstance(value, bool) for value in values)
        and set(values) == set(range(N))
    )


def _is_full_cycle(mapping: list[int]) -> bool:
    seen: set[int] = set()
    node = 0
    for _ in range(N):
        if node in seen:
            return False
        seen.add(node)
        node = mapping[node]
    return node == 0 and len(seen) == N


def _preserves(blue: list[int], orange: list[int], permutation: tuple[int, ...]) -> bool:
    return all(
        permutation[blue[node]] == blue[permutation[node]]
        and permutation[orange[node]] == orange[permutation[node]]
        for node in range(N)
    )


def _automorphism_count(blue: list[int], orange: list[int], stop_after: int = 2) -> int:
    count = 0
    for permutation in itertools.permutations(range(N)):
        if _preserves(blue, orange, permutation):
            count += 1
            if count >= stop_after:
                break
    return count


def sample_scene(seed: int) -> dict:
    """Sample two regular asymmetric relations and a balanced hidden alignment."""

    seed = int(seed)
    rng = random.Random((seed ^ 0x2A71E1A7) & 0xFFFFFFFFFFFFFFFF)
    for _ in range(10000):
        blue_order = list(range(N))
        orange_order = list(range(N))
        rng.shuffle(blue_order)
        rng.shuffle(orange_order)
        blue = _cycle_from_order(blue_order)
        orange = _cycle_from_order(orange_order)
        if any(blue[node] == orange[node] for node in range(N)):
            continue
        if _automorphism_count(blue, orange) == 1:
            break
    else:
        raise RuntimeError("could not sample an asymmetric relation pair")

    query_index = (3 * seed + seed // N) % N
    desired_letter = seed % N
    remaining_colors = [value for value in range(N) if value != query_index]
    remaining_letters = [value for value in range(N) if value != desired_letter]
    rng.shuffle(remaining_letters)
    alignment = [-1] * N
    alignment[query_index] = desired_letter
    for color_index, letter_index in zip(remaining_colors, remaining_letters, strict=True):
        alignment[color_index] = letter_index

    return {
        "blue_map": blue,
        "orange_map": orange,
        "color_to_letter": alignment,
        "query_color": COLORS[query_index],
    }


def _validate_scene(scene: dict) -> None:
    if not isinstance(scene, dict) or set(scene) != {
        "blue_map",
        "orange_map",
        "color_to_letter",
        "query_color",
    }:
        raise ValueError("scene fields do not match the world schema")
    blue = scene["blue_map"]
    orange = scene["orange_map"]
    alignment = scene["color_to_letter"]
    query_color = scene["query_color"]
    if not _is_permutation(blue) or not _is_full_cycle(blue):
        raise ValueError("blue relation must be one seven-token cycle")
    if not _is_permutation(orange) or not _is_full_cycle(orange):
        raise ValueError("orange relation must be one seven-token cycle")
    if any(blue[node] == orange[node] for node in range(N)):
        raise ValueError("relation marks may not overlap")
    if _automorphism_count(blue, orange) != 1:
        raise ValueError("relation pair must have a unique identity alignment")
    if not _is_permutation(alignment):
        raise ValueError("color-to-letter alignment must be bijective")
    if query_color not in COLORS:
        raise ValueError("query color is outside the visible identity palette")


def _letter_maps(scene: dict) -> tuple[list[int], list[int]]:
    blue = scene["blue_map"]
    orange = scene["orange_map"]
    alignment = scene["color_to_letter"]
    blue_letters = [0] * N
    orange_letters = [0] * N
    for color_index in range(N):
        letter_index = alignment[color_index]
        blue_letters[letter_index] = alignment[blue[color_index]]
        orange_letters[letter_index] = alignment[orange[color_index]]
    return blue_letters, orange_letters


def _draw_color_token(draw: ImageDraw.ImageDraw, center_x: int, center_y: int, name: str) -> None:
    draw.ellipse(
        (center_x - 12, center_y - 12, center_x + 12, center_y + 12),
        fill=RGB[name],
        outline=RGB["ink"],
        width=2,
    )


def _draw_letter_token(draw: ImageDraw.ImageDraw, center_x: int, center_y: int, letter: str) -> None:
    draw.rounded_rectangle(
        (center_x - 15, center_y - 16, center_x + 15, center_y + 16),
        radius=5,
        fill=RGB["white"],
        outline=RGB["ink"],
        width=2,
    )
    scale = 4
    left = center_x - 6
    top = center_y - 10
    for row, bits in enumerate(GLYPHS[letter]):
        for column, bit in enumerate(bits):
            if bit == "1":
                x = left + column * scale
                y = top + row * scale
                draw.rectangle((x, y, x + scale - 1, y + scale - 1), fill=RGB["ink"])


def _draw_board(
    draw: ImageDraw.ImageDraw,
    grid_x: int,
    blue: list[int],
    orange: list[int],
    *,
    use_colors: bool,
) -> None:
    for index in range(N):
        center_x = grid_x + index * CELL + CELL // 2
        center_y = GRID_Y + index * CELL + CELL // 2
        if use_colors:
            _draw_color_token(draw, center_x, TOP_HEADER_Y, COLORS[index])
            _draw_color_token(draw, LEFT_HEADER_X if grid_x == LEFT_GRID_X else grid_x - 26, center_y, COLORS[index])
        else:
            _draw_letter_token(draw, center_x, TOP_HEADER_Y, LETTERS[index])
            _draw_letter_token(draw, grid_x - 26, center_y, LETTERS[index])

    for row in range(N):
        for column in range(N):
            left = grid_x + column * CELL
            top = GRID_Y + row * CELL
            draw.rectangle(
                (left, top, left + CELL - 1, top + CELL - 1),
                fill=RGB["white"],
                outline=RGB["guide"],
                width=2,
            )
        blue_x = grid_x + blue[row] * CELL + CELL // 2
        orange_x = grid_x + orange[row] * CELL + CELL // 2
        center_y = GRID_Y + row * CELL + CELL // 2
        draw.ellipse(
            (blue_x - 9, center_y - 9, blue_x + 9, center_y + 9),
            fill=RGB["blue_relation"],
            outline=RGB["ink"],
            width=1,
        )
        draw.rectangle(
            (orange_x - 8, center_y - 8, orange_x + 8, center_y + 8),
            fill=RGB["orange_relation"],
            outline=RGB["ink"],
            width=1,
        )


def render(scene: dict) -> Image.Image:
    """Render two legible relation matrices and one visible query token."""

    _validate_scene(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), RGB["background"])
    draw = ImageDraw.Draw(image)
    title = _font(19, bold=True)
    small = _font(14, bold=True)
    tiny = _font(12, bold=True)

    draw.rounded_rectangle((18, 15, 450, 428), radius=12, fill=RGB["panel"], outline=RGB["guide"], width=2)
    draw.rounded_rectangle((483, 15, 950, 428), radius=12, fill=RGB["panel"], outline=RGB["guide"], width=2)
    draw.text((34, 27), "COLOR MAP", fill=RGB["ink"], font=title)
    draw.text((499, 27), "LETTER MAP", fill=RGB["ink"], font=title)

    for x in (244, 709):
        draw.ellipse((x, 31, x + 15, 46), fill=RGB["blue_relation"], outline=RGB["ink"], width=1)
        draw.text((x + 21, 30), "BLUE", fill=RGB["ink"], font=tiny)
        draw.rectangle((x + 76, 32, x + 90, 46), fill=RGB["orange_relation"], outline=RGB["ink"], width=1)
        draw.text((x + 97, 30), "ORANGE", fill=RGB["ink"], font=tiny)

    draw.text((61, 91), "FROM", fill=RGB["ink"], font=tiny)
    draw.text((249, 57), "TO", fill=RGB["ink"], font=tiny)
    draw.text((526, 91), "FROM", fill=RGB["ink"], font=tiny)
    draw.text((714, 57), "TO", fill=RGB["ink"], font=tiny)

    _draw_board(draw, LEFT_GRID_X, scene["blue_map"], scene["orange_map"], use_colors=True)
    letter_blue, letter_orange = _letter_maps(scene)
    _draw_board(draw, RIGHT_GRID_X, letter_blue, letter_orange, use_colors=False)

    draw.rounded_rectangle((270, 443, 710, 495), radius=10, fill=RGB["panel"], outline=RGB["guide"], width=2)
    draw.text((304, 457), "QUERY COLOR", fill=RGB["ink"], font=small)
    _draw_color_token(draw, 468, 469, scene["query_color"])
    draw.text((500, 457), "= WHICH LETTER?", fill=RGB["ink"], font=small)
    return image


def analytic_gold(scene: dict) -> str:
    _validate_scene(scene)
    query_index = COLORS.index(scene["query_color"])
    return LETTERS[scene["color_to_letter"][query_index]]


def margin(scene: dict) -> float:
    """Minimum number of visibly wrong relation cells for any wrong alignment."""

    _validate_scene(scene)
    blue_letters, orange_letters = _letter_maps(scene)
    true_alignment = tuple(scene["color_to_letter"])
    best = 2 * N
    for proposed in itertools.permutations(range(N)):
        if proposed == true_alignment:
            continue
        mismatches = 0
        for node in range(N):
            mismatches += blue_letters[proposed[node]] != proposed[scene["blue_map"][node]]
            mismatches += orange_letters[proposed[node]] != proposed[scene["orange_map"][node]]
        best = min(best, mismatches)
    return float(best)


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 1.0
    except (TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    _validate_scene(scene)
    return []
