"""Independent PNG-only inverse for two-relation identity alignment."""

from __future__ import annotations

import itertools

import numpy as np
from PIL import Image

N = 7
WIDTH = 980
HEIGHT = 510
CELL = 44
GRID_Y = 110
LEFT_GRID_X = 105
RIGHT_GRID_X = 570
TOP_HEADER_Y = 84
LEFT_HEADER_X = 79

BACKGROUND = (248, 247, 242)
WHITE = (255, 255, 255)
INK = (27, 35, 48)
GUIDE = (112, 124, 138)
BLUE_RELATION = (38, 108, 210)
ORANGE_RELATION = (232, 128, 32)
COLORS = ("red", "navy", "green", "purple", "gold", "teal", "pink")
LETTERS = ("A", "B", "C", "D", "E", "F", "G")
COLOR_RGB = {
    "red": (211, 61, 62),
    "navy": (47, 83, 166),
    "green": (45, 148, 84),
    "purple": (126, 74, 183),
    "gold": (222, 171, 35),
    "teal": (23, 151, 157),
    "pink": (219, 79, 142),
}
GLYPHS = {
    "A": ("010", "101", "111", "101", "101"),
    "B": ("110", "101", "110", "101", "110"),
    "C": ("011", "100", "100", "100", "011"),
    "D": ("110", "101", "101", "101", "110"),
    "E": ("111", "100", "110", "100", "111"),
    "F": ("111", "100", "110", "100", "100"),
    "G": ("011", "100", "101", "101", "011"),
}


class OracleError(ValueError):
    """Raised when the raster is not one intact alignment worksheet."""


def _uniform(array: np.ndarray, box: tuple[int, int, int, int], color: tuple[int, int, int], what: str) -> None:
    left, top, right, bottom = box
    patch = array[top:bottom, left:right]
    if patch.size == 0 or not np.all(patch == np.asarray(color, dtype=np.uint8)):
        raise OracleError(f"{what} is missing or damaged")


def _read_color(array: np.ndarray, center_x: int, center_y: int, what: str) -> str:
    patch = array[center_y - 5 : center_y + 6, center_x - 5 : center_x + 6]
    matches = [
        name
        for name, rgb in COLOR_RGB.items()
        if np.all(patch == np.asarray(rgb, dtype=np.uint8))
    ]
    if len(matches) != 1:
        raise OracleError(f"{what} has no unique visible identity color")
    return matches[0]


def _read_letter(array: np.ndarray, center_x: int, center_y: int, what: str) -> str:
    scale = 4
    left = center_x - 6
    top = center_y - 10
    bits: list[str] = []
    for row in range(5):
        line: list[str] = []
        for column in range(3):
            x = left + column * scale + 1
            y = top + row * scale + 1
            patch = array[y : y + 2, x : x + 2]
            if np.all(patch == np.asarray(INK, dtype=np.uint8)):
                line.append("1")
            elif np.all(patch == np.asarray(WHITE, dtype=np.uint8)):
                line.append("0")
            else:
                raise OracleError(f"{what} contains an ambiguous block-letter cell")
        bits.append("".join(line))
    matches = [letter for letter, pattern in GLYPHS.items() if tuple(bits) == pattern]
    if len(matches) != 1:
        raise OracleError(f"{what} is not a declared letter token")
    return matches[0]


def _read_board(array: np.ndarray, grid_x: int, what: str) -> tuple[list[int], list[int]]:
    blue = [-1] * N
    orange = [-1] * N
    allowed = {
        WHITE,
        INK,
        BLUE_RELATION,
        ORANGE_RELATION,
    }
    for row in range(N):
        for column in range(N):
            left = grid_x + column * CELL
            top = GRID_Y + row * CELL
            _uniform(array, (left + 4, top, left + CELL - 4, top + 2), GUIDE, f"{what} top grid edge")
            _uniform(array, (left, top + 4, left + 2, top + CELL - 4), GUIDE, f"{what} left grid edge")
            patch = array[top + 3 : top + CELL - 3, left + 3 : left + CELL - 3]
            colors = {tuple(int(channel) for channel in pixel) for pixel in patch.reshape(-1, 3)}
            if not colors <= allowed:
                raise OracleError(f"{what} cell {row},{column} contains undeclared pixels")
            blue_count = int(np.sum(np.all(patch == np.asarray(BLUE_RELATION, dtype=np.uint8), axis=2)))
            orange_count = int(np.sum(np.all(patch == np.asarray(ORANGE_RELATION, dtype=np.uint8), axis=2)))
            if blue_count and orange_count:
                raise OracleError(f"{what} cell {row},{column} overlaps relation marks")
            if blue_count:
                if not 220 <= blue_count <= 300 or blue[row] != -1:
                    raise OracleError(f"{what} has a malformed blue relation")
                blue[row] = column
            if orange_count:
                if not 220 <= orange_count <= 300 or orange[row] != -1:
                    raise OracleError(f"{what} has a malformed orange relation")
                orange[row] = column
    if set(blue) != set(range(N)) or set(orange) != set(range(N)):
        raise OracleError(f"{what} relations are not complete permutations")
    if any(blue[row] == orange[row] for row in range(N)):
        raise OracleError(f"{what} relation marks overlap")
    return blue, orange


def decision_from_image(image: Image.Image) -> str:
    """Recover the unique relation-preserving color-to-letter alignment."""

    if image.size != (WIDTH, HEIGHT):
        raise OracleError(f"expected a {WIDTH} by {HEIGHT} raster")
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.shape != (HEIGHT, WIDTH, 3):
        raise OracleError("expected one RGB worksheet")
    for x, y in ((0, 0), (WIDTH - 1, 0), (0, HEIGHT - 1), (WIDTH - 1, HEIGHT - 1)):
        if tuple(int(value) for value in array[y, x]) != BACKGROUND:
            raise OracleError("canvas corners do not contain the declared background")

    for index, expected in enumerate(COLORS):
        center_x = LEFT_GRID_X + index * CELL + CELL // 2
        center_y = GRID_Y + index * CELL + CELL // 2
        if _read_color(array, center_x, TOP_HEADER_Y, f"color column {index}") != expected:
            raise OracleError("color columns are not in their visible declared order")
        if _read_color(array, LEFT_HEADER_X, center_y, f"color row {index}") != expected:
            raise OracleError("color rows are not in their visible declared order")

    for index, expected in enumerate(LETTERS):
        center_x = RIGHT_GRID_X + index * CELL + CELL // 2
        center_y = GRID_Y + index * CELL + CELL // 2
        if _read_letter(array, center_x, TOP_HEADER_Y, f"letter column {index}") != expected:
            raise OracleError("letter columns are not in their visible declared order")
        if _read_letter(array, RIGHT_GRID_X - 26, center_y, f"letter row {index}") != expected:
            raise OracleError("letter rows are not in their visible declared order")

    color_blue, color_orange = _read_board(array, LEFT_GRID_X, "color map")
    letter_blue, letter_orange = _read_board(array, RIGHT_GRID_X, "letter map")
    query_color = _read_color(array, 468, 469, "query token")
    query_index = COLORS.index(query_color)

    matches: list[tuple[int, ...]] = []
    for proposed in itertools.permutations(range(N)):
        if all(
            letter_blue[proposed[node]] == proposed[color_blue[node]]
            and letter_orange[proposed[node]] == proposed[color_orange[node]]
            for node in range(N)
        ):
            matches.append(proposed)
            if len(matches) > 1:
                break
    if len(matches) != 1:
        raise OracleError("the two maps do not determine one unique identity alignment")
    return LETTERS[matches[0][query_index]]
