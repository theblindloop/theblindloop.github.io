"""Scene sampling, rasterisation and analytic gold for the column-settle world.

A framed lattice of 9 columns and 8 rows holds dark square cells scattered at
arbitrary heights inside their columns.  The declared transform lets every dark
cell fall straight down inside its own column until it rests on the floor or on
another dark cell.  The settled stacks form an outline that is matched against
five named height profiles.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

COLS = 9
ROWS = 8
TOTAL_INK = 36

PROFILES: dict[str, tuple[int, ...]] = {
    "ramp": (0, 1, 2, 3, 4, 5, 6, 7, 8),
    "peak": (1, 2, 4, 7, 8, 7, 4, 2, 1),
    "valley": (8, 6, 3, 1, 0, 1, 3, 6, 8),
    "table": (1, 5, 5, 5, 4, 5, 5, 5, 1),
    "comb": (7, 1, 7, 1, 4, 1, 7, 1, 7),
}
PROFILE_NAMES: tuple[str, ...] = tuple(sorted(PROFILES))

_BACKGROUNDS = ((246, 246, 243), (250, 248, 240), (240, 244, 248))
_INKS = ((30, 30, 34), (22, 26, 22), (44, 34, 30))
_FRAMES = ((30, 90, 200), (198, 60, 40), (20, 140, 92), (120, 60, 190))


def _heights_from_cells(dark_cells) -> tuple[int, ...]:
    """Column occupancy counts: the settled stack height of every column."""

    columns = [set() for _ in range(COLS)]
    for cell in dark_cells:
        col = int(cell[0])
        row = int(cell[1])
        if 0 <= col < COLS and 0 <= row < ROWS:
            columns[col].add(row)
    return tuple(len(column) for column in columns)


def _distances(heights) -> list[tuple[int, str]]:
    scored = [
        (sum(abs(a - b) for a, b in zip(heights, PROFILES[name])), name) for name in PROFILE_NAMES
    ]
    return sorted(scored)


def analytic_gold(scene) -> str:
    """Name the settled outline; reads only the printed dark cells."""

    scored = _distances(_heights_from_cells(scene["dark_cells"]))
    return scored[0][1]


def margin(scene) -> float:
    """Runner-up profile distance minus the best profile distance."""

    scored = _distances(_heights_from_cells(scene["dark_cells"]))
    return float(scored[1][0] - scored[0][0])


def is_quarantined(scene) -> bool:
    return margin(scene) < 3.0


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    name = PROFILE_NAMES[rng.randrange(len(PROFILE_NAMES))]
    heights = list(PROFILES[name])
    for _ in range(rng.randrange(0, 9)):
        donors = [index for index in range(COLS) if heights[index] > 0]
        takers = [index for index in range(COLS) if heights[index] < ROWS]
        if not donors or not takers:
            break
        donor = rng.choice(donors)
        taker = rng.choice([index for index in takers if index != donor] or takers)
        heights[donor] -= 1
        heights[taker] += 1
    dark_cells: list[list[int]] = []
    for col, height in enumerate(heights):
        rows = rng.sample(range(ROWS), height)
        rng.shuffle(rows)
        dark_cells.extend([col, row] for row in rows)
    rng.shuffle(dark_cells)
    return {
        "dark_cells": dark_cells,
        "pitch": rng.choice((26, 30, 34)),
        "gap": rng.choice((4, 6)),
        "pad": rng.choice((8, 12, 16)),
        "frame_w": rng.choice((5, 7)),
        "seed_tag": int(seed),
        "bg": list(_BACKGROUNDS[rng.randrange(len(_BACKGROUNDS))]),
        "ink": list(_INKS[rng.randrange(len(_INKS))]),
        "frame": list(_FRAMES[rng.randrange(len(_FRAMES))]),
    }


def render(scene) -> Image.Image:
    pitch = int(scene["pitch"])
    gap = int(scene["gap"])
    pad = int(scene["pad"])
    frame_w = int(scene["frame_w"])
    bg = tuple(scene["bg"])
    ink = tuple(scene["ink"])
    frame = tuple(scene["frame"])
    interior_w = COLS * pitch
    interior_h = ROWS * pitch
    width = 2 * pad + 2 * frame_w + interior_w
    height = 2 * pad + 2 * frame_w + interior_h
    image = Image.new("RGB", (width, height), bg)
    draw = ImageDraw.Draw(image)
    draw.rectangle(
        [pad, pad, width - 1 - pad, height - 1 - pad],
        outline=frame,
        width=frame_w,
    )
    origin_x = pad + frame_w
    origin_y = pad + frame_w
    inset = gap // 2
    side = pitch - gap
    for cell in scene["dark_cells"]:
        col = int(cell[0])
        row = int(cell[1])
        if not (0 <= col < COLS and 0 <= row < ROWS):
            continue
        x0 = origin_x + col * pitch + inset
        y0 = origin_y + row * pitch + inset
        draw.rectangle([x0, y0, x0 + side - 1, y0 + side - 1], fill=ink)
    return image


def latent_symmetries(scene) -> list[tuple[str, dict]]:
    """Render-invariant relabelings of the latent scene."""

    twins: list[tuple[str, dict]] = []
    reversed_scene = dict(scene)
    reversed_scene["dark_cells"] = list(reversed(scene["dark_cells"]))
    twins.append(("reversed_paint_order", reversed_scene))
    if scene["dark_cells"]:
        repeated = dict(scene)
        repeated["dark_cells"] = list(scene["dark_cells"]) + [list(scene["dark_cells"][0])]
        twins.append(("repeated_cell", repeated))
    tagged = dict(scene)
    tagged["seed_tag"] = int(scene["seed_tag"]) + 977
    twins.append(("unread_tag_shift", tagged))
    return twins


def naive_read_distance(scene) -> float:
    """How badly the printed picture itself matches the answer profile.

    The printed cells sit at scattered heights, so treating the topmost printed
    cell of every column as a stack height names the answer only by accident.
    """

    columns = [[] for _ in range(COLS)]
    for cell in scene["dark_cells"]:
        columns[int(cell[0])].append(int(cell[1]))
    naive = tuple(ROWS - min(column) if column else 0 for column in columns)
    scored = _distances(naive)
    return float(scored[0][0])
