"""Pixel-only inverse arm for the column-settle world.

The arm receives one rasterised picture and the public question contract only.
It locates the coloured frame, recovers the lattice interior, reads which cells
are printed dark, executes the declared settle transform by counting the dark
cells of every column, and names the nearest of the five declared profiles with
a reported runner-up margin.
"""

from __future__ import annotations

import numpy as np

COLS = 9
ROWS = 8

PROFILES: dict[str, tuple[int, ...]] = {
    "comb": (7, 1, 7, 1, 4, 1, 7, 1, 7),
    "peak": (1, 2, 4, 7, 8, 7, 4, 2, 1),
    "ramp": (0, 1, 2, 3, 4, 5, 6, 7, 8),
    "table": (1, 5, 5, 5, 4, 5, 5, 5, 1),
    "valley": (8, 6, 3, 1, 0, 1, 3, 6, 8),
}
PROFILE_NAMES: tuple[str, ...] = tuple(sorted(PROFILES))

_CHROMA = 45
_DARK = 140
_ABSTAIN = "abstain"


def _interior_box(chroma_mask: np.ndarray) -> tuple[int, int, int, int] | None:
    rows = np.flatnonzero(chroma_mask.any(axis=1))
    cols = np.flatnonzero(chroma_mask.any(axis=0))
    if rows.size < 8 or cols.size < 8:
        return None
    top, bottom = int(rows[0]), int(rows[-1])
    left, right = int(cols[0]), int(cols[-1])
    window = chroma_mask[top : bottom + 1, left : right + 1]
    height, width = window.shape
    col_frac = window.sum(axis=0) / float(height)
    row_frac = window.sum(axis=1) / float(width)
    solid_cols = np.flatnonzero(col_frac > 0.5)
    solid_rows = np.flatnonzero(row_frac > 0.5)
    if solid_cols.size < 2 or solid_rows.size < 2:
        return None
    x0 = left + int(solid_cols[solid_cols < width // 2].max()) + 1
    x1 = left + int(solid_cols[solid_cols > width // 2].min()) - 1
    y0 = top + int(solid_rows[solid_rows < height // 2].max()) + 1
    y1 = top + int(solid_rows[solid_rows > height // 2].min()) - 1
    if x1 - x0 < COLS * 4 or y1 - y0 < ROWS * 4:
        return None
    return x0, y0, x1, y1


def _column_heights(image) -> tuple[int, ...] | None:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    high = array.max(axis=2)
    low = array.min(axis=2)
    chroma = (high - low) > _CHROMA
    box = _interior_box(chroma)
    if box is None:
        return None
    x0, y0, x1, y1 = box
    luminance = array.mean(axis=2)
    ink = (luminance < _DARK) & (~chroma)
    pitch_x = (x1 - x0 + 1) / float(COLS)
    pitch_y = (y1 - y0 + 1) / float(ROWS)
    heights = []
    for col in range(COLS):
        filled = 0
        for row in range(ROWS):
            cx = x0 + (col + 0.5) * pitch_x
            cy = y0 + (row + 0.5) * pitch_y
            half_x = max(1.0, pitch_x * 0.2)
            half_y = max(1.0, pitch_y * 0.2)
            patch = ink[
                int(round(cy - half_y)) : int(round(cy + half_y)) + 1,
                int(round(cx - half_x)) : int(round(cx + half_x)) + 1,
            ]
            if patch.size and float(patch.mean()) > 0.5:
                filled += 1
        heights.append(filled)
    return tuple(heights)


def scored_profiles(image) -> list[tuple[int, str]]:
    heights = _column_heights(image)
    if heights is None:
        return []
    scored = [
        (int(sum(abs(a - b) for a, b in zip(heights, PROFILES[name]))), name)
        for name in PROFILE_NAMES
    ]
    return sorted(scored)


def decision_with_margin(image) -> tuple[str, float]:
    scored = scored_profiles(image)
    if len(scored) < 2:
        return _ABSTAIN, 0.0
    return scored[0][1], float(scored[1][0] - scored[0][0])


def decision_from_image(image):
    """Public inverse entry point: name the settled profile from pixels alone."""

    return decision_with_margin(image)[0]
