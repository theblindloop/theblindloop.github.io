"""Independent image-only inverse for Shuffled Record Swap Audit."""
from __future__ import annotations
from PIL import Image, ImageDraw
CANVAS=(1000,800); INK=(35,40,48); MUTED=(222,226,229); SHAPE=(25,126,111); BADGE=(58,48,92); PANEL=(255,255,255)
COLORS={"amber":(219,139,22),"cyan":(18,153,177),"violet":(133,76,181)}; SHAPES=("circle","square","triangle","diamond"); ROW_Y=(205,315,425,535)
class OracleError(ValueError): pass
def _shape_template(name):
    im=Image.new("RGB",(80,80),PANEL); d=ImageDraw.Draw(im); cx=cy=40
    if name=="circle": d.ellipse((cx-25,cy-25,cx+25,cy+25),fill=SHAPE)
    elif name=="square": d.rectangle((cx-23,cy-23,cx+23,cy+23),fill=SHAPE)
    elif name=="triangle": d.polygon(((cx,cy-29),(cx-29,cy+23),(cx+29,cy+23)),fill=SHAPE)
    elif name=="diamond": d.polygon(((cx,cy-30),(cx-30,cy),(cx,cy+30),(cx+30,cy)),fill=SHAPE)
    return tuple(p==SHAPE for p in im.getdata())
def _read_shape(im,cx,cy):
    obs=tuple(p==SHAPE for p in im.crop((cx-40,cy-40,cx+40,cy+40)).getdata()); scores={n:sum(a!=b for a,b in zip(obs,_shape_template(n))) for n in SHAPES}; best=min(scores,key=scores.get)
    if scores[best]!=0 or sum(obs)<1500: raise OracleError("damaged shape")
    return best
def _read_badge(im,cx,cy):
    px=im.load()
    if px[cx-48,cy]!=INK or px[cx+48,cy]!=INK: raise OracleError("badge border missing")
    bits=[]
    for i in range(6):
        col,row=i%3,i//3; x0,y0=cx-34+col*24,cy-23+row*28; samples=[px[x,y] for x in range(x0+3,x0+16,4) for y in range(y0+3,y0+16,4)]
        if all(p==BADGE for p in samples): bits.append(1)
        elif all(p==MUTED for p in samples): bits.append(0)
        else: raise OracleError("badge cell damaged")
    if sum(bits)!=3: raise OracleError("invalid badge")
    return tuple(bits)
def _read_table(im,x0):
    px=im.load(); out={}
    if px[x0+225,65]!=INK: raise OracleError("panel border missing")
    for y in ROW_Y:
        key=_read_shape(im,x0+75,y)
        if key in out: raise OracleError("duplicate key")
        out[key]=(_read_badge(im,x0+235,y),_read_badge(im,x0+365,y))
    if set(out)!=set(SHAPES): raise OracleError("missing key")
    return out
def decision_from_image(image):
    image=image.convert("RGB")
    if image.size!=CANVAS: raise OracleError("wrong canvas")
    before,after=_read_table(image,35),_read_table(image,515)
    if any(before[k][0]!=after[k][0] for k in SHAPES): raise OracleError("left field changed")
    rights0=[before[k][1] for k in SHAPES]; rights1=[after[k][1] for k in SHAPES]
    if len(set([before[k][0] for k in SHAPES]+rights0))!=8 or sorted(rights0)!=sorted(rights1): raise OracleError("badge inventory damaged")
    changed=[k for k in SHAPES if before[k][1]!=after[k][1]]
    if len(changed)!=2 or before[changed[0]][1]!=after[changed[1]][1] or before[changed[1]][1]!=after[changed[0]][1]: raise OracleError("not one swap")
    target=set(changed); matches=[]; seen=set(); px=image.load()
    for cx in (175,500,825):
        names=[n for n,c in COLORS.items() if px[cx,635]==c]
        if len(names)!=1: raise OracleError("answer border damaged")
        pair=frozenset((_read_shape(image,cx-62,712),_read_shape(image,cx+62,712)))
        if len(pair)!=2 or pair in seen: raise OracleError("invalid answer pair")
        seen.add(pair)
        if set(pair)==target: matches.append(names[0])
    if len(matches)!=1: raise OracleError("non-unique answer")
    return matches[0]
