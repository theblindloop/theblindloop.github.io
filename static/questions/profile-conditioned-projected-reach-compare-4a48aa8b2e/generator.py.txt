"""Projected-reach pair selection world (measurement).

A gray vertical reference line marks a fixed x-column ``X_REF``. Two thick
diagonal strokes, one red and one blue, are drawn on opposite sides of that
line, each with its own randomly sampled slope and its own ``reach`` value (the
row coordinate at which that stroke's best-fit line meets the gray line when
extrapolated straight along its tilt). The decision is which color's fitted
line reaches HIGHER on the panel at the reference column: the color with the
smaller reach (smaller row index, i.e. closer to the top).

The measured quantity is a projected reading that depends on BOTH the stroke's
slope and its position/intercept, so no single cheap cue (higher stroke, longer
stroke, steeper stroke, or relative position alone) determines the winner.
The winner color and the two slopes / reaches / positions are all randomized.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "projected_reach_compare-0.1.0"

PALETTE = ["red", "blue"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
}
REFERENCE = (120, 120, 120)
BACKGROUND = (250, 250, 250)

WIDTH = 560
HEIGHT = 420

# The gray vertical reference line's fixed x-column (public world contract).
X_REF = 300

STROKE_WIDTH = 16
STROKE_LEN = 150

# Per-side x-bands so the two colored strokes never overlap one another.
LEFT_BAND = (50, X_REF - 40)
RIGHT_BAND = (X_REF + 40, WIDTH - 50)

# Reach values (the line's y at X_REF) kept within the canvas.
REACH_MIN = 120.0
REACH_MAX = 300.0

# Slope magnitudes kept small so the extrapolation stays well-conditioned.
SLOPE_MIN = 0.12
SLOPE_MAX = 0.55

# Minimum reach separation for a cleanly decidable scene.
MIN_REACH_MARGIN = 18.0
# Below this reach separation the scene is a finite-resolution boundary case.
QUARANTINE_REACH = 10.0


def _line_y(slope: float, reach: float, x: float) -> float:
    """y (row) on the stroke's line at column x, given reach at X_REF."""
    return reach + slope * (x - X_REF)


def sample_scene(seed: int) -> dict:
    """Deterministic scene with a guaranteed, counterbalanced winner."""
    rng = random.Random(seed * 1049 + 11)
    winner = PALETTE[seed % 2]
    loser = PALETTE[1 - seed % 2]
    while True:
        slope = {c: rng.uniform(SLOPE_MIN, SLOPE_MAX) * rng.choice([-1.0, 1.0]) for c in PALETTE}
        reach = {c: rng.uniform(REACH_MIN, REACH_MAX) for c in PALETTE}
        # Force the winner's reach to be clearly smaller (higher) than the loser's.
        if abs(reach[winner] - reach[loser]) < MIN_REACH_MARGIN:
            reach[winner] = min(reach[winner], reach[loser]) - MIN_REACH_MARGIN
        if not (REACH_MIN <= reach[winner] <= REACH_MAX):
            continue
        gap = reach[loser] - reach[winner]
        if gap < MIN_REACH_MARGIN:
            continue

        # Left stroke x-range (red) and right stroke x-range (blue).
        xr_lo = rng.uniform(LEFT_BAND[0], LEFT_BAND[1] - STROKE_LEN)
        blue_x_lo = rng.uniform(RIGHT_BAND[0], RIGHT_BAND[1] - STROKE_LEN)

        # Check every stroke stays within the vertical canvas over its x-range.
        strokes = {}
        ok = True
        candidates = [("red", xr_lo, xr_lo + STROKE_LEN, slope["red"]),
                      ("blue", blue_x_lo, blue_x_lo + STROKE_LEN, slope["blue"])]
        for color, x_a, x_b, s in candidates:
            y_a = _line_y(s, reach[color], x_a)
            y_b = _line_y(s, reach[color], x_b)
            lo, hi = min(y_a, y_b), max(y_a, y_b)
            if lo < 24 or hi > HEIGHT - 24:
                ok = False
                break
            strokes[color] = {"x0": x_a, "x1": x_b, "y0": y_a, "y1": y_b,
                              "slope": s, "reach": reach[color]}
        if not ok:
            continue
        real_winner = min(strokes, key=lambda c: strokes[c]["reach"])
        if real_winner != winner:
            continue
        break

    return {
        "strokes": [
            {"color": c,
             "x0": float(strokes[c]["x0"]), "x1": float(strokes[c]["x1"]),
             "y0": float(strokes[c]["y0"]), "y1": float(strokes[c]["y1"]),
             "slope": float(strokes[c]["slope"]), "reach": float(strokes[c]["reach"])}
            for c in PALETTE
        ],
        "width": WIDTH,
        "height": HEIGHT,
        "background": "light",
    }


def render(scene: dict) -> Image.Image:
    """Render I = R(z): gray reference line plus two thick colored strokes."""
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    draw.line([X_REF, 12, X_REF, scene["height"] - 12], fill=REFERENCE, width=4)
    for stroke in scene["strokes"]:
        draw.line(
            [(stroke["x0"], stroke["y0"]), (stroke["x1"], stroke["y1"])],
            fill=COLOR_RGB[stroke["color"]],
            width=STROKE_WIDTH,
        )
    return img


def analytic_gold(scene: dict) -> str:
    """Color of the stroke whose fitted line reaches higher (smaller reach)."""
    strokes = scene["strokes"]
    best = min(strokes, key=lambda s: s["reach"])
    return best["color"]


def margin(scene: dict) -> float:
    """Verifiability margin: the reach gap (loser minus winner), in pixels."""
    reaches = sorted(s["reach"] for s in scene["strokes"])
    return float(reaches[1] - reaches[0])


def is_quarantined(scene: dict) -> bool:
    """Near-boundary: the reach gap is finite-resolution small."""
    return margin(scene) < QUARANTINE_REACH


def latent_symmetries(scene: dict) -> list:
    """not_applicable: no nontrivial pixel-identical latent symmetry.

    Every latent field carried by a stroke (its color label, its slope, and its
    reach -- which together determine its two endpoints and the measured
    reading) is directly rendered as visible ink: the color is the stroke's
    colour, and the slope and reach are encoded in the drawn diagonal's tilt
    and location. Swapping the two labels changes which color is correct, and
    any global reflection/rotation turns the raster into a distinct image, so
    no latent transform is both pixel-identical and gold-preserving.
    """
    return []
