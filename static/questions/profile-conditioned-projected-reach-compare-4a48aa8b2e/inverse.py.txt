"""Independent pixel-only inverse arm for the projected_reach_compare world.

Receives ONLY the rasterized PNG (plus the world's fixed public geometry and
palette, which are the naming convention) and, for each colored thick stroke,
fits a best-fit line through that colour's ink and reads the row at which that
line meets the fixed gray reference column. It shares no geometry with
renderer.render() and never imports the renderer, prompts, generator, verifier,
or any analytical gold code.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "projected_reach_compare-oracle-0.1.0"

PALETTE = ["red", "blue"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
}
REFERENCE = (120, 120, 120)
BACKGROUND = (250, 250, 250)

# Fixed public geometry: the gray vertical reference column.
X_REF = 300


def _nearest(arr: np.ndarray, refs: np.ndarray) -> np.ndarray:
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def _fit_reach(cols: np.ndarray, rows: np.ndarray, x_ref: float) -> float:
    """Least-squares row = a*col + b and evaluate it at x_ref."""
    if cols.shape[0] < 8:
        raise ValueError("insufficient ink for a line fit")
    a, b = np.polyfit(cols, rows, 1)
    return float(a * x_ref + b)


def decision_from_image(image: Image.Image) -> str:
    """Color of the stroke whose fitted line reaches higher (smaller row)."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    refs = np.stack(
        [np.array(COLOR_RGB[c], dtype=np.float32) for c in PALETTE]
        + [np.array(REFERENCE, dtype=np.float32)]
        + [np.array(BACKGROUND, dtype=np.float32)],
        axis=0,
    )
    labels = _nearest(arr, refs)

    best_color = PALETTE[0]
    best_reach = float("inf")
    for idx, color in enumerate(PALETTE):
        color_mask = labels == idx
        rows, cols = np.nonzero(color_mask)
        if rows.size < 8:
            continue
        reach = _fit_reach(cols.astype(np.float32), rows.astype(np.float32), float(X_REF))
        if reach < best_reach:
            best_reach = reach
            best_color = color
    return best_color
