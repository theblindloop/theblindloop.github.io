"""Renderer and analytic arm for the region-balance measurement world."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS = (400, 320)
PALETTE = (
    (22, 148, 166),
    (215, 91, 77),
    (103, 91, 190),
    (35, 158, 103),
)
BACKGROUND = (248, 248, 245)
REFERENCE = (82, 91, 101)
QUARANTINE_PX = 8.0


def _polygon_centroid_y(points: list[list[int]]) -> float:
    twice_area = 0.0
    moment = 0.0
    for index, (x0, y0) in enumerate(points):
        x1, y1 = points[(index + 1) % len(points)]
        cross = x0 * y1 - x1 * y0
        twice_area += cross
        moment += (y0 + y1) * cross
    if abs(twice_area) < 1.0:
        raise ValueError("degenerate polygon")
    return moment / (3.0 * twice_area)


def _polygon_area(points: list[list[int]]) -> float:
    twice_area = 0.0
    for index, (x0, y0) in enumerate(points):
        x1, y1 = points[(index + 1) % len(points)]
        twice_area += x0 * y1 - x1 * y0
    return abs(twice_area) / 2.0


def _absolute_polygon(scene: dict[str, Any]) -> list[list[int]]:
    dx, dy = scene["translation"]
    return [[int(x + dx), int(y + dy)] for x, y in scene["polygon"]]


def sample_scene(seed: int) -> dict:
    """Sample a balanced, well-separated connected footprint."""
    rng = random.Random(int(seed))
    desired = "above" if rng.randrange(2) == 0 else "below"
    xs = [-82, -48, -12, 22, 55, 82]
    for _ in range(200):
        # A broad central slab guarantees a compact, connected filled region.
        base_top = rng.randint(38, 48)
        base_bottom = rng.randint(38, 48)
        extra = rng.randint(24, 42)
        top = [-(base_top + rng.randint(-5, 6)) for _ in xs]
        bottom = [(base_bottom + rng.randint(-5, 6)) for _ in xs]
        if desired == "above":
            for index in rng.sample(range(len(xs)), k=3):
                top[index] -= extra + rng.randint(-4, 8)
        else:
            for index in rng.sample(range(len(xs)), k=3):
                bottom[index] += extra + rng.randint(-4, 8)
        # Top boundary left-to-right, bottom boundary right-to-left.
        polygon = [[x, y] for x, y in zip(xs, top, strict=True)]
        polygon += [[x, y] for x, y in reversed(list(zip(xs, bottom, strict=True)))]
        reference_y = rng.randint(-5, 5)
        cy = _polygon_centroid_y(polygon)
        gap = abs(cy - reference_y)
        area = _polygon_area(polygon)
        if (
            16400.0 <= area <= 17400.0
            and 11.0 <= gap <= 27.0
            and ((cy < reference_y) == (desired == "above"))
        ):
            break
    else:
        raise RuntimeError("could not sample separated centroid scene")
    return {
        "canvas_size": list(CANVAS),
        "polygon": polygon,
        "reference_y": reference_y,
        "region_color": list(PALETTE[rng.randrange(len(PALETTE))]),
        "translation": [rng.randint(180, 220), rng.randint(145, 175)],
    }


def render(scene: dict) -> Image.Image:
    width, height = (int(v) for v in scene["canvas_size"])
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    dx, dy = (int(v) for v in scene["translation"])
    ref_y = int(scene["reference_y"]) + dy
    # Local dashed reference: broad enough to measure, but not image-spanning.
    for x0 in range(dx - 112, dx + 112, 18):
        draw.line((x0, ref_y, min(x0 + 10, dx + 112), ref_y), fill=REFERENCE, width=4)
    points = _absolute_polygon(scene)
    draw.polygon([tuple(point) for point in points], fill=tuple(scene["region_color"]))
    # A thin neutral frame locates the regional measurement without encoding class.
    draw.rounded_rectangle((dx - 116, dy - 100, dx + 116, dy + 100), radius=8, outline=(205, 208, 207), width=2)
    return image


def _signed_gap(scene: dict) -> float:
    # Explicit accesses are audited against latent_alias.label_inputs.
    local_polygon = scene["polygon"]
    local_reference = float(scene["reference_y"])
    translation_y = float(scene["translation"][1])
    centroid_y = _polygon_centroid_y(local_polygon) + translation_y
    reference_y = local_reference + translation_y
    return reference_y - centroid_y


def analytic_gold(scene: dict) -> str:
    return "above" if _signed_gap(scene) > 0.0 else "below"


def margin(scene: dict) -> float:
    return abs(_signed_gap(scene))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
