"""Independent pixel-only inverse for area-centroid balance."""

from __future__ import annotations

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    pass


def _largest_component(mask: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    best: list[tuple[int, int]] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) > len(best):
            best = points
    output = np.zeros_like(mask, dtype=bool)
    if best:
        ys, xs = zip(*best, strict=True)
        output[np.asarray(ys), np.asarray(xs)] = True
    return output


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    spread = array.max(axis=2) - array.min(axis=2)
    # The footprint is the only saturated component; palette identity is irrelevant.
    saturated = (spread > 48) & (array.min(axis=2) < 205)
    region = _largest_component(saturated)
    ys, xs = np.nonzero(region)
    # Clean scenes occupy a deliberately narrow 16.4k--17.4k pixel-area band.
    # The lower completeness bound makes erased interior patches explicit rather
    # than silently treating a damaged silhouette as a different valid scene.
    if len(ys) < 15800:
        raise OracleAbstention("filled region missing or materially erased")
    span_x = int(xs.max() - xs.min() + 1)
    span_y = int(ys.max() - ys.min() + 1)
    if span_x < 145 or span_y < 92:
        raise OracleAbstention("footprint extent is not recoverable")
    fill = len(ys) / float(span_x * span_y)
    if fill < 0.56:
        raise OracleAbstention("footprint has a destructive interior hole")

    # Neutral dark pixels are the dashed reference. The pale frame/background fail
    # the luminance test and the saturated footprint fails the chroma test.
    mean = array.mean(axis=2)
    neutral = (spread < 32) & (mean > 45) & (mean < 150)
    line_ys, line_xs = np.nonzero(neutral)
    if len(line_ys) < 55:
        raise OracleAbstention("reference line is not recoverable")
    reference_y = float(np.median(line_ys))
    centroid_y = float(np.mean(ys))
    gap = reference_y - centroid_y
    if abs(gap) < 5.0:
        raise OracleAbstention("pixel centroid is too close to the reference")
    return "above" if gap > 0.0 else "below"
