"""Procedural renderer for intrinsic color order along one visible thread."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
SUPERSAMPLE = 4

_PALETTES = (
    {"background": (248, 247, 242), "thread": (35, 43, 57)},
    {"background": (244, 249, 250), "thread": (48, 39, 54)},
    {"background": (250, 246, 250), "thread": (31, 50, 49)},
)
_MARKERS = {
    "red": (218, 53, 61),
    "blue": (43, 103, 207),
    "gold": (229, 165, 32),
    "start": (29, 158, 121),
    "finish": (143, 67, 179),
}
_DECISIONS = ("red-blue-gold", "gold-blue-red")


def _transform_point(x: float, y: float, quarter_turns: int, reflect: bool) -> tuple[float, float]:
    if reflect:
        x = -x
    for _ in range(quarter_turns):
        x, y = -y, x
    return x, y


def sample_scene(seed: int) -> dict:
    """Sample a folded, crossing-free route with answer-neutral geometry."""

    # Adjacent seeds form an exact nuisance-matched answer pair.
    rng = random.Random((int(seed) // 2) ^ 0x5EEDB0A7)
    top = -rng.uniform(42.0, 50.0)
    middle = rng.uniform(-3.0, 3.0)
    bottom = rng.uniform(42.0, 50.0)
    left_top = -rng.uniform(145.0, 169.0)
    right_turn = rng.uniform(116.0, 148.0)
    inner_left = -rng.uniform(62.0, 94.0)
    right_finish = rng.uniform(145.0, 169.0)
    route = [
        (left_top, top),
        (right_turn, top),
        (right_turn, middle),
        (inner_left, middle),
        (inner_left, bottom),
        (right_finish, bottom),
    ]
    quarter_turns = rng.randrange(4)
    reflect = bool(rng.randrange(2))
    dx = rng.uniform(-17.0, 17.0)
    dy = rng.uniform(-17.0, 17.0)
    transformed = []
    for x, y in route:
        tx, ty = _transform_point(x, y, quarter_turns, reflect)
        transformed.append([round(WIDTH / 2 + dx + tx, 3), round(HEIGHT / 2 + dy + ty, 3)])

    fractions = [
        rng.uniform(0.175, 0.245),
        rng.uniform(0.455, 0.545),
        rng.uniform(0.755, 0.825),
    ]
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "route_points": transformed,
        "bead_fractions": [round(value, 6) for value in fractions],
        "bead_order": _DECISIONS[int(seed) & 1],
        "stroke": rng.choice([10, 11, 12]),
        "bead_radius": rng.choice([11, 12, 13]),
        "cap_radius": rng.choice([16, 17, 18]),
        "palette_index": rng.randrange(len(_PALETTES)),
    }


def _lengths(points: Iterable[Iterable[float]]) -> tuple[list[tuple[float, float]], list[float]]:
    pts = [(float(item[0]), float(item[1])) for item in points]
    cumulative = [0.0]
    for first, second in zip(pts, pts[1:]):
        cumulative.append(cumulative[-1] + math.hypot(second[0] - first[0], second[1] - first[1]))
    return pts, cumulative


def _point_at(points: Iterable[Iterable[float]], fraction: float) -> tuple[float, float]:
    pts, cumulative = _lengths(points)
    target = float(fraction) * cumulative[-1]
    for index in range(len(pts) - 1):
        if target <= cumulative[index + 1]:
            span = cumulative[index + 1] - cumulative[index]
            alpha = (target - cumulative[index]) / span
            x = pts[index][0] + alpha * (pts[index + 1][0] - pts[index][0])
            y = pts[index][1] + alpha * (pts[index + 1][1] - pts[index][1])
            return x, y
    return pts[-1]


def render(scene: dict) -> Image.Image:
    """Render only declared scene fields; the label is visible as bead placement."""

    width = int(scene["width"])
    height = int(scene["height"])
    points = [[float(v) * SUPERSAMPLE for v in item] for item in scene["route_points"]]
    fractions = [float(value) for value in scene["bead_fractions"]]
    bead_order = str(scene["bead_order"])
    stroke = int(scene["stroke"]) * SUPERSAMPLE
    bead_radius = int(scene["bead_radius"]) * SUPERSAMPLE
    cap_radius = int(scene["cap_radius"]) * SUPERSAMPLE
    palette = _PALETTES[int(scene["palette_index"]) % len(_PALETTES)]

    image = Image.new("RGB", (width * SUPERSAMPLE, height * SUPERSAMPLE), palette["background"])
    draw = ImageDraw.Draw(image)
    draw.line([tuple(item) for item in points], fill=palette["thread"], width=stroke, joint="curve")

    # Dark halos keep every marker visibly attached to the same physical thread.
    start = tuple(points[0])
    finish = tuple(points[-1])
    halo = 3 * SUPERSAMPLE
    draw.ellipse(
        [start[0] - cap_radius - halo, start[1] - cap_radius - halo,
         start[0] + cap_radius + halo, start[1] + cap_radius + halo],
        fill=palette["thread"],
    )
    draw.ellipse(
        [start[0] - cap_radius, start[1] - cap_radius,
         start[0] + cap_radius, start[1] + cap_radius],
        fill=_MARKERS["start"],
    )
    draw.rectangle(
        [finish[0] - cap_radius - halo, finish[1] - cap_radius - halo,
         finish[0] + cap_radius + halo, finish[1] + cap_radius + halo],
        fill=palette["thread"],
    )
    draw.rectangle(
        [finish[0] - cap_radius, finish[1] - cap_radius,
         finish[0] + cap_radius, finish[1] + cap_radius],
        fill=_MARKERS["finish"],
    )

    names = bead_order.split("-")
    for fraction, name in zip(fractions, names, strict=True):
        x, y = _point_at(scene["route_points"], fraction)
        x *= SUPERSAMPLE
        y *= SUPERSAMPLE
        draw.ellipse(
            [x - bead_radius - halo, y - bead_radius - halo,
             x + bead_radius + halo, y + bead_radius + halo],
            fill=palette["thread"],
        )
        draw.ellipse(
            [x - bead_radius, y - bead_radius, x + bead_radius, y + bead_radius],
            fill=_MARKERS[name],
        )

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    decision = str(scene["bead_order"])
    if decision not in _DECISIONS:
        raise ValueError(f"unsupported bead order: {decision}")
    return decision


def margin(scene: dict) -> float:
    """Minimum clear arc-length between consecutive required landmarks."""

    _, cumulative = _lengths(scene["route_points"])
    total = cumulative[-1]
    stations = [0.0, *[float(value) * total for value in scene["bead_fractions"]], total]
    clearance = min(b - a for a, b in zip(stations, stations[1:]))
    return float(clearance - 2.0 * float(scene["bead_radius"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 55.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
