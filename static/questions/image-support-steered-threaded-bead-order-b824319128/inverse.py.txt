"""Independent pixel-only route follower for the threaded bead world."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_VALID = {"red-blue-gold", "gold-blue-red"}


def _masks(image: Image.Image) -> tuple[np.ndarray, dict[str, np.ndarray]]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    r, g, b = arr[..., 0], arr[..., 1], arr[..., 2]
    corners = np.concatenate(
        [arr[:8, :8].reshape(-1, 3), arr[:8, -8:].reshape(-1, 3),
         arr[-8:, :8].reshape(-1, 3), arr[-8:, -8:].reshape(-1, 3)]
    )
    background = np.median(corners, axis=0)
    foreground = np.sum((arr - background) ** 2, axis=2) > 45 ** 2
    markers = {
        "red": (r > 155) & (g < 115) & (b < 125) & (r - g > 65) & (r - b > 55),
        "blue": (b > 140) & (b - r > 65) & (b - g > 30),
        "gold": (r > 170) & (g > 105) & (r - g > 25) & (g - b > 55),
        "start": (g > 120) & (g - r > 65) & (g - b > 18),
        "finish": (r > 105) & (b > 125) & (r - g > 35) & (b - g > 45),
    }
    return foreground, markers


def _route_distances(foreground: np.ndarray, starts: np.ndarray) -> np.ndarray:
    height, width = foreground.shape
    distance = np.full((height, width), -1, dtype=np.int32)
    ys, xs = np.nonzero(starts & foreground)
    queue: deque[tuple[int, int]] = deque()
    for y, x in zip(ys.tolist(), xs.tolist(), strict=True):
        distance[y, x] = 0
        queue.append((y, x))
    while queue:
        y, x = queue.popleft()
        next_distance = int(distance[y, x]) + 1
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < height and 0 <= nx < width and foreground[ny, nx] and distance[ny, nx] < 0:
                distance[ny, nx] = next_distance
                queue.append((ny, nx))
    return distance


def _nearest(distance: np.ndarray, mask: np.ndarray) -> int | None:
    values = distance[mask & (distance >= 0)]
    return int(values.min()) if values.size else None


def decision_from_image(image: Image.Image) -> str:
    """Return bead order, or ``abstain`` when the visible route is incomplete."""

    foreground, markers = _masks(image)
    if any(int(mask.sum()) < 80 for mask in markers.values()):
        return "abstain"
    distance = _route_distances(foreground, markers["start"])
    finish_distance = _nearest(distance, markers["finish"])
    bead_distances = {name: _nearest(distance, markers[name]) for name in ("red", "blue", "gold")}
    if finish_distance is None or any(value is None for value in bead_distances.values()):
        return "abstain"
    ordered = sorted(bead_distances, key=lambda name: int(bead_distances[name]))
    # Raster-safe scenes place stations far apart; a near-tie indicates damage.
    values = sorted(int(value) for value in bead_distances.values())
    if min(b - a for a, b in zip(values, values[1:])) < 30:
        return "abstain"
    decision = "-".join(ordered)
    return decision if decision in _VALID else "abstain"
