"""Deterministic paired route-tree sampler and raster renderer."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "paired-tree-trip-length-renderer-0.1.0"
WIDTH = 512
HEIGHT = 512
BACKGROUND = (248, 250, 252)
TOP_Y = 128
BOTTOM_Y = 384

PALETTES = {
    "harbor": {
        "route": (31, 41, 55),
        "center": (45, 194, 167),
        "start": (37, 99, 235),
        "finish": (234, 88, 12),
        "faint": (211, 219, 229),
        "panel": (255, 255, 255),
    },
    "ocean": {
        "route": (22, 45, 70),
        "center": (55, 205, 232),
        "start": (42, 98, 190),
        "finish": (226, 116, 43),
        "faint": (218, 224, 230),
        "panel": (255, 255, 255),
    },
}

_DETOUR = {"short": 16, "long": 40}
_X0 = 164
_X1 = 348
_J1 = 220
_J2 = 292
_TOTAL_NETWORK_LENGTH = 352


def _route_length(kind: str) -> int:
    if kind not in _DETOUR:
        raise ValueError(f"unknown route kind {kind!r}")
    return (_X1 - _X0) + 2 * _DETOUR[kind]


def _make_decorations(rng: random.Random) -> list[list[int]]:
    decorations: list[list[int]] = []
    zones = ((72, 106, 205, 307), (406, 440, 205, 307))
    for x0, x1, y0, y1 in zones:
        for _ in range(3):
            x = rng.randint(x0, x1)
            y = rng.randint(y0, y1)
            length = rng.choice((10, 13, 16))
            decorations.append([x, y, x + rng.choice((-length, length)), y + rng.choice((-3, 0, 3))])
    return decorations


def sample_scene(seed: int) -> dict:
    """Sample balanced answers and label-preserving pose/palette nuisances."""

    rng = random.Random(int(seed))
    top_long = int(seed) % 2 == 0
    return {
        "top_route_kind": "long" if top_long else "short",
        "bottom_route_kind": "short" if top_long else "long",
        "top_direction": rng.choice((-1, 1)),
        "bottom_direction": rng.choice((-1, 1)),
        "top_mirror": bool(rng.randrange(2)),
        "bottom_mirror": bool(rng.randrange(2)),
        "top_endpoint_swap": bool(rng.randrange(2)),
        "bottom_endpoint_swap": bool(rng.randrange(2)),
        "palette_id": rng.choice(sorted(PALETTES)),
        "dx": rng.randint(-4, 4),
        "dy": rng.randint(-3, 3),
        "decorations": _make_decorations(rng),
        "draw_order": rng.randrange(2),
        "width": WIDTH,
        "height": HEIGHT,
    }


def _transform(
    points: list[tuple[int, int]], *, center_y: int, direction: int, mirror: bool, dx: int
) -> list[tuple[int, int]]:
    transformed = []
    mirror_axis = 256 + dx
    for x, local_y in points:
        px = x + dx
        if mirror:
            px = 2 * mirror_axis - px
        transformed.append((px, center_y + direction * local_y))
    return transformed


def _network(
    kind: str, *, center_y: int, direction: int, mirror: bool, dx: int
) -> tuple[list[list[tuple[int, int]]], tuple[int, int], tuple[int, int]]:
    """Return one equal-total-ink tree and its two main-route endpoints."""

    detour = _DETOUR[kind]
    main = [(_X0, 0), (_J1, 0), (_J1, detour), (_J2, detour), (_J2, 0), (_X1, 0)]
    branch_total = _TOTAL_NETWORK_LENGTH - _route_length(kind)
    branch_length = branch_total // 2
    horizontal_hook = branch_length - 40
    branch_a = [(_J1, 0), (_J1, -40), (_J1 - horizontal_hook, -40)]
    branch_b = [(_J2, 0), (_J2, -40), (_J2 + horizontal_hook, -40)]
    paths = [main, branch_a, branch_b]
    placed = [
        _transform(path, center_y=center_y, direction=direction, mirror=mirror, dx=dx)
        for path in paths
    ]
    return placed, placed[0][0], placed[0][-1]


def _draw_ring(
    draw: ImageDraw.ImageDraw, center: tuple[int, int], color: tuple[int, int, int]
) -> None:
    x, y = center
    draw.ellipse((x - 12, y - 12, x + 12, y + 12), fill="white", outline=color, width=5)


def _draw_map(
    draw: ImageDraw.ImageDraw,
    paths: list[list[tuple[int, int]]],
    start: tuple[int, int],
    finish: tuple[int, int],
    *,
    endpoint_swap: bool,
    palette: dict[str, tuple[int, int, int]],
    draw_order: int,
) -> None:
    if endpoint_swap:
        start, finish = finish, start
    _draw_ring(draw, start, palette["start"])
    _draw_ring(draw, finish, palette["finish"])
    ordered = paths if draw_order == 0 else list(reversed(paths))
    for path in ordered:
        draw.line(path, fill=palette["route"], width=11, joint="curve")
    for path in ordered:
        draw.line(path, fill=palette["center"], width=3, joint="curve")


def render(scene: dict) -> Image.Image:
    width = int(scene.get("width", WIDTH))
    height = int(scene.get("height", HEIGHT))
    palette = PALETTES[str(scene["palette_id"])]
    dx, dy = int(scene["dx"]), int(scene["dy"])
    top, top_start, top_finish = _network(
        str(scene["top_route_kind"]),
        center_y=TOP_Y + dy,
        direction=int(scene["top_direction"]),
        mirror=bool(scene["top_mirror"]),
        dx=dx,
    )
    bottom, bottom_start, bottom_finish = _network(
        str(scene["bottom_route_kind"]),
        center_y=BOTTOM_Y + dy,
        direction=int(scene["bottom_direction"]),
        mirror=bool(scene["bottom_mirror"]),
        dx=dx,
    )

    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for center_y in (TOP_Y + dy, BOTTOM_Y + dy):
        draw.rounded_rectangle(
            (136 + dx, center_y - 54, 376 + dx, center_y + 54),
            radius=14,
            fill=palette["panel"],
            outline=palette["faint"],
            width=2,
        )
    for x1, y1, x2, y2 in scene["decorations"]:
        draw.line((x1, y1, x2, y2), fill=palette["faint"], width=3)
    _draw_map(
        draw,
        top,
        top_start,
        top_finish,
        endpoint_swap=bool(scene["top_endpoint_swap"]),
        palette=palette,
        draw_order=int(scene["draw_order"]),
    )
    _draw_map(
        draw,
        bottom,
        bottom_start,
        bottom_finish,
        endpoint_swap=bool(scene["bottom_endpoint_swap"]),
        palette=palette,
        draw_order=int(scene["draw_order"]),
    )
    return image


def analytic_gold(scene: dict) -> str:
    top_kind = str(scene["top_route_kind"])
    bottom_kind = str(scene["bottom_route_kind"])
    top_length = _route_length(top_kind)
    bottom_length = _route_length(bottom_kind)
    if top_length == bottom_length:
        raise ValueError("equal endpoint-conditioned route lengths are quarantined")
    return "top" if top_length > bottom_length else "bottom"


def margin(scene: dict) -> float:
    top_length = _route_length(str(scene["top_route_kind"]))
    bottom_length = _route_length(str(scene["bottom_route_kind"]))
    return float(abs(top_length - bottom_length))


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 20.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Same-color tree strokes commute within both raster drawing passes."""

    twin = copy.deepcopy(scene)
    twin["draw_order"] = 1 - int(twin["draw_order"])
    return [("reverse_same_color_branch_draw_order", twin)]
