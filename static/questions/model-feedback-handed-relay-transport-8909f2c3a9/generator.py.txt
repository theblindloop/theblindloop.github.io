"""Handed relay transport: two unknown plane motions, composed in a stated order.

Two congruent tile pairs each demonstrate one plane motion of the square lattice.
At least one of the two pairs is a mirror pair, so the *type* of each motion has
to be read off the handedness of the drawn tiles before either can be applied.
The purple dot is carried first by the green-to-orange motion and then by the
blue-to-cyan motion; the four black squares are the four competing composites.

All coordinates live in half-cell units so that lattice-vertex pivots and
cell-centre pivots are both integers.  Cell (c, r) has half-unit centre
(2c + 1, 2r + 1).
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CELLS = 11
CELL_PX = 84
BORDER = 42
CANVAS = CELLS * CELL_PX + 2 * BORDER

BACKGROUND = (255, 255, 255)
RULE = (200, 200, 204)
BLACK = (0, 0, 0)
DOT = (122, 47, 191)
TILE_COLORS = {
    "green": (33, 160, 74),
    "orange": (239, 122, 16),
    "blue": (34, 85, 204),
    "cyan": (18, 183, 200),
}

BASE_TILE = ((0, 0), (1, 0), (2, 0), (2, 1))


# --------------------------------------------------------------------------
# lattice geometry in half-cell units
# --------------------------------------------------------------------------


def _centre(cell) -> tuple[int, int]:
    return (2 * int(cell[0]) + 1, 2 * int(cell[1]) + 1)


def _cell_of(point) -> tuple[int, int]:
    u, v = point
    return ((u - 1) // 2, (v - 1) // 2)


def apply_motion(motion, point) -> tuple[int, int]:
    """Apply a lattice motion, given in half-cell units, to a half-unit point."""
    u, v = point
    kind = motion[0]
    if kind == "rot":
        k, pu, pv = motion[1], motion[2], motion[3]
        for _ in range(k % 4):
            u, v = pu - (v - pv), pv + (u - pu)
        return (u, v)
    if kind == "mir":
        axis, m = motion[1], motion[2]
        if axis == "v":
            return (2 * m - u, v)
        return (u, 2 * m - v)
    if kind == "trans":
        return (u + motion[1], v + motion[2])
    raise ValueError(f"unknown motion {motion!r}")


def _reverses(motion) -> bool:
    return motion[0] == "mir"


def _motion_cells(motion, cells):
    return [list(_cell_of(apply_motion(motion, _centre(cell)))) for cell in cells]


def all_motions(cells: int = CELLS):
    """Every lattice motion the solver may consider, in half-cell units."""
    span = 2 * cells
    out = []
    for pu in range(0, span + 1):
        for pv in range(0, span + 1):
            if (pu + pv) % 2 == 0:
                out.append(("rot", 1, pu, pv))
                out.append(("rot", 3, pu, pv))
            out.append(("rot", 2, pu, pv))
    for m in range(0, span + 1):
        out.append(("mir", "v", m))
        out.append(("mir", "h", m))
    for du in range(-span, span + 1, 2):
        for dv in range(-span, span + 1, 2):
            if du or dv:
                out.append(("trans", du, dv))
    return out


_MOTION_TABLE = None


def solve_motion(source, image_cells, cells: int = CELLS):
    """The unique lattice motion carrying ``source`` onto ``image_cells``."""
    global _MOTION_TABLE
    if _MOTION_TABLE is None:
        _MOTION_TABLE = all_motions(cells)
    want = {tuple(cell) for cell in image_cells}
    src = [tuple(cell) for cell in source]
    if len(want) != len(src) or len(set(src)) != len(src):
        return None
    found = []
    for motion in _MOTION_TABLE:
        if {tuple(c) for c in _motion_cells(motion, src)} == want:
            found.append(motion)
            if len(found) > 1:
                return None
    return found[0] if found else None


# --------------------------------------------------------------------------
# tile poses
# --------------------------------------------------------------------------


def _poses():
    out = []
    for flip in (False, True):
        cells = [(-c, r) if flip else (c, r) for c, r in BASE_TILE]
        for _ in range(4):
            cells = [(-r, c) for c, r in cells]
            lo_c = min(c for c, _ in cells)
            lo_r = min(r for _, r in cells)
            out.append(tuple(sorted((c - lo_c, r - lo_r) for c, r in cells)))
    return sorted(set(out))


POSES = _poses()


def _place(rng, cells: int):
    pose = list(POSES[rng.randrange(len(POSES))])
    width = max(c for c, _ in pose) + 1
    height = max(r for _, r in pose) + 1
    oc = rng.randrange(0, cells - width + 1)
    orr = rng.randrange(0, cells - height + 1)
    return [[c + oc, r + orr] for c, r in pose]


def _inside(cells_list, cells: int) -> bool:
    return all(0 <= c < cells and 0 <= r < cells for c, r in cells_list)


def _sample_motion(rng, cells: int):
    span = 2 * cells
    kind = rng.choice(("rot", "rot", "mir", "mir", "trans"))
    if kind == "rot":
        while True:
            pu = rng.randrange(0, span + 1)
            pv = rng.randrange(0, span + 1)
            k = rng.choice((1, 2, 3))
            if k == 2 or (pu + pv) % 2 == 0:
                return ("rot", k, pu, pv)
    if kind == "mir":
        return ("mir", rng.choice(("v", "h")), rng.randrange(2, span - 1))
    while True:
        du = 2 * rng.randrange(-cells + 1, cells)
        dv = 2 * rng.randrange(-cells + 1, cells)
        if du or dv:
            return ("trans", du, dv)


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) * 7919 + 104729)
    n = CELLS
    for _ in range(200000):
        motion_a = _sample_motion(rng, n)
        motion_b = _sample_motion(rng, n)
        if not (_reverses(motion_a) or _reverses(motion_b)):
            continue
        green = _place(rng, n)
        orange = _motion_cells(motion_a, green)
        if not _inside(orange, n):
            continue
        blue = _place(rng, n)
        cyan = _motion_cells(motion_b, blue)
        if not _inside(cyan, n):
            continue
        dot = [rng.randrange(n), rng.randrange(n)]
        p = _centre(dot)
        after_a = apply_motion(motion_a, p)
        after_b = apply_motion(motion_b, p)
        composite = apply_motion(motion_b, after_a)
        reversed_order = apply_motion(motion_a, after_b)
        targets = [_cell_of(q) for q in (composite, reversed_order, after_a, after_b)]
        if not _inside(targets, n):
            continue
        if len(set(targets)) != 4:
            continue
        if len({c for c, _ in targets}) != 4:
            continue
        tiles = [tuple(c) for c in green + orange + blue + cyan]
        if len(set(tiles)) != 16:
            continue
        busy = set(tiles) | {tuple(dot)}
        if any(t in busy for t in targets):
            continue
        # the demonstrations must not sit on the dot or on each other
        if tuple(dot) in set(tiles):
            continue
        # solvability: each pair must pin down exactly one lattice motion
        if solve_motion(green, orange, n) is None or solve_motion(blue, cyan, n) is None:
            continue
        order = list(range(4))
        rng.shuffle(order)
        scene = {
            "canvas": CANVAS,
            "cells": n,
            "cell_px": CELL_PX,
            "green_cells": [list(c) for c in green],
            "orange_cells": [list(c) for c in orange],
            "blue_cells": [list(c) for c in blue],
            "cyan_cells": [list(c) for c in cyan],
            "dot_cell": [int(dot[0]), int(dot[1])],
            "target_cells": [list(targets[i]) for i in order],
        }
        return scene
    raise RuntimeError("sampler exhausted")


# --------------------------------------------------------------------------
# analytic gold
# --------------------------------------------------------------------------


def winning_cell(scene: dict) -> tuple[int, int]:
    n = int(scene["cells"])
    motion_a = solve_motion(scene["green_cells"], scene["orange_cells"], n)
    motion_b = solve_motion(scene["blue_cells"], scene["cyan_cells"], n)
    if motion_a is None or motion_b is None:
        raise ValueError("scene does not determine both motions")
    point = _centre(scene["dot_cell"])
    return _cell_of(apply_motion(motion_b, apply_motion(motion_a, point)))


def _ordered_targets(scene: dict):
    return sorted((tuple(cell) for cell in scene["target_cells"]), key=lambda cell: cell[0])


def analytic_gold(scene: dict) -> str:
    winner = winning_cell(scene)
    ordered = _ordered_targets(scene)
    if winner not in ordered:
        raise ValueError("the composed image is not a drawn target square")
    return f"square-{ordered.index(winner) + 1}"


def margin(scene: dict) -> float:
    winner = winning_cell(scene)
    others = [cell for cell in _ordered_targets(scene) if cell != winner]
    gap = min(
        max(abs(cell[0] - winner[0]), abs(cell[1] - winner[1])) for cell in others
    )
    return float(gap * int(scene["cell_px"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2 * int(scene["cell_px"]) - 1e-9


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical latent rewrites that must preserve the gold answer."""
    transforms: list[tuple[str, dict]] = []

    rotated = dict(scene)
    rotated["target_cells"] = list(scene["target_cells"][1:]) + [scene["target_cells"][0]]
    transforms.append(("rotate_target_serialization", rotated))

    reversed_tiles = dict(scene)
    for key in ("green_cells", "orange_cells", "blue_cells", "cyan_cells"):
        reversed_tiles[key] = list(reversed(list(scene[key])))
    transforms.append(("reverse_tile_cell_serialization", reversed_tiles))
    return transforms


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------


def _origin(scene: dict) -> int:
    return (int(scene["canvas"]) - int(scene["cells"]) * int(scene["cell_px"])) // 2


def _cell_centre_px(scene: dict, cell) -> tuple[int, int]:
    step = int(scene["cell_px"])
    o = _origin(scene)
    return (o + int(cell[0]) * step + step // 2, o + int(cell[1]) * step + step // 2)


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas"])
    n = int(scene["cells"])
    step = int(scene["cell_px"])
    o = _origin(scene)
    image = Image.new("RGB", (size, size), BACKGROUND)
    draw = ImageDraw.Draw(image)

    lo, hi = o, o + n * step - 1
    for k in range(n + 1):
        x = o + k * step
        draw.rectangle((x - 1, lo, x + 1, hi), fill=RULE)
        draw.rectangle((lo, x - 1, hi, x + 1), fill=RULE)

    for cell in scene["target_cells"]:
        cx, cy = _cell_centre_px(scene, cell)
        draw.rectangle((cx - 40, cy - 40, cx + 39, cy + 39), outline=BLACK, width=9)

    for colour, tile in (
        (TILE_COLORS["green"], scene["green_cells"]),
        (TILE_COLORS["orange"], scene["orange_cells"]),
        (TILE_COLORS["blue"], scene["blue_cells"]),
        (TILE_COLORS["cyan"], scene["cyan_cells"]),
    ):
        for cell in tile:
            cx, cy = _cell_centre_px(scene, cell)
            draw.rectangle((cx - 36, cy - 36, cx + 35, cy + 35), fill=colour)

    cx, cy = _cell_centre_px(scene, scene["dot_cell"])
    draw.ellipse((cx - 28, cy - 28, cx + 27, cy + 27), fill=DOT)
    return image
