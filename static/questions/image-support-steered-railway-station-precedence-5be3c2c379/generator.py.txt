"""Deterministic scene sampler and renderer for station order on one path."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

BACKGROUND = (248, 247, 242)
RAIL = (30, 34, 39)
ORANGE = (235, 107, 30)
BLUE = (42, 104, 196)
CYAN = (0, 164, 184)
YELLOW = (246, 194, 48)
RAIL_WIDTH = 9
MARKER_RADIUS = 12


def _segments(path: list[list[int]]) -> Iterable[tuple[tuple[int, int], tuple[int, int], float]]:
    for a, b in zip(path, path[1:]):
        length = math.hypot(b[0] - a[0], b[1] - a[1])
        yield (a[0], a[1]), (b[0], b[1]), length


def _path_length(path: list[list[int]]) -> float:
    return sum(length for _, _, length in _segments(path))


def _point_at(path: list[list[int]], fraction: float) -> list[int]:
    target = fraction * _path_length(path)
    walked = 0.0
    for a, b, length in _segments(path):
        if walked + length >= target:
            t = (target - walked) / length
            return [round(a[0] + t * (b[0] - a[0])), round(a[1] + t * (b[1] - a[1]))]
        walked += length
    return list(path[-1])


def _distance_at(path: list[list[int]], point: list[int]) -> float:
    """Arc distance from path[0] to a raster-rounded marker center."""
    walked = 0.0
    px, py = point
    best: tuple[float, float] | None = None
    for a, b, length in _segments(path):
        ax, ay = a
        bx, by = b
        dot = (px - ax) * (bx - ax) + (py - ay) * (by - ay)
        t = min(1.0, max(0.0, dot / (length * length)))
        qx, qy = ax + t * (bx - ax), ay + t * (by - ay)
        offset = math.hypot(px - qx, py - qy)
        candidate = (offset, walked + t * length)
        if best is None or candidate < best:
            best = candidate
        walked += length
    if best is None or best[0] > 1.1:
        raise ValueError(f"marker {point!r} is not on the railway")
    return best[1]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    width = height = 416
    xs = [rng.randint(25, 40), rng.randint(194, 220), rng.randint(376, 391)]
    upper = rng.randint(35, 52)
    lower = rng.randint(364, 381)
    if rng.randrange(2):
        path = [[xs[0], lower], [xs[1], upper], [xs[2], lower]]
    else:
        path = [[xs[0], upper], [xs[1], lower], [xs[2], upper]]
    if rng.randrange(2):
        path.reverse()

    # Marker fractions vary independently while retaining a large route-order margin.
    f_early = rng.uniform(0.20, 0.31)
    f_late = rng.uniform(0.69, 0.80)
    early = _point_at(path, f_early)
    late = _point_at(path, f_late)
    orange_first = bool(rng.randrange(2))
    orange = list(path[0] if orange_first else path[-1])
    blue = list(path[-1] if orange_first else path[0])

    desired_yes = int(seed) % 2 == 0
    cyan_is_raw_early = desired_yes if orange_first else not desired_yes
    cyan, yellow = (early, late) if cyan_is_raw_early else (late, early)
    return {
        "path": path,
        "orange": orange,
        "blue": blue,
        "cyan": cyan,
        "yellow": yellow,
        "width": width,
        "height": height,
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BACKGROUND)
    draw = ImageDraw.Draw(image)
    path = [tuple(point) for point in scene["path"]]
    draw.line(path, fill=RAIL, width=RAIL_WIDTH, joint="curve")
    # Small black joint disks prevent raster pinholes at right-angle turns.
    joint_radius = RAIL_WIDTH // 2
    for x, y in path[1:-1]:
        draw.ellipse((x-joint_radius, y-joint_radius, x+joint_radius, y+joint_radius), fill=RAIL)
    markers = ((scene["cyan"], CYAN), (scene["yellow"], YELLOW),
               (scene["orange"], ORANGE), (scene["blue"], BLUE))
    for center, color in markers:
        x, y = center
        r = MARKER_RADIUS
        draw.ellipse((x-r, y-r, x+r, y+r), fill=color)
        # A black center pin makes it explicit that the station is centered on the route.
        draw.ellipse((x-2, y-2, x+2, y+2), fill=RAIL)
    return image


def _route_distance_from_orange(scene: dict, point: list[int]) -> float:
    path = scene["path"]
    raw = _distance_at(path, point)
    if list(scene["orange"]) == list(path[0]):
        return raw
    if list(scene["orange"]) == list(path[-1]):
        return _path_length(path) - raw
    raise ValueError("orange marker must be at a railway endpoint")


def analytic_gold(scene: dict) -> str:
    cyan_distance = _route_distance_from_orange(scene, scene["cyan"])
    yellow_distance = _route_distance_from_orange(scene, scene["yellow"])
    return "yes" if cyan_distance < yellow_distance else "no"


def margin(scene: dict) -> float:
    cyan_distance = _route_distance_from_orange(scene, scene["cyan"])
    yellow_distance = _route_distance_from_orange(scene, scene["yellow"])
    return abs(cyan_distance - yellow_distance)


def is_quarantined(scene: dict) -> bool:
    try:
        path = scene["path"]
        if len(path) < 2 or list(scene["orange"]) not in (list(path[0]), list(path[-1])):
            return True
        return margin(scene) < 48.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
