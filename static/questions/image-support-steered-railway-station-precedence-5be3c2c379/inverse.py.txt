"""Independent pixel-only geodesic-order oracle."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE = np.asarray([
    (30, 34, 39),       # black railway
    (235, 107, 30),     # orange start
    (42, 104, 196),     # blue finish
    (0, 164, 184),      # cyan station
    (246, 194, 48),     # yellow station
], dtype=np.int32)
COLOR_TOLERANCE_SQ = 42 * 42


def _masks(image: Image.Image) -> list[np.ndarray]:
    # int32 avoids overflow when squaring 8-bit channel differences.
    array = np.asarray(image.convert("RGB"), dtype=np.int32)
    distances = ((array[:, :, None, :] - PALETTE[None, None, :, :]) ** 2).sum(axis=3)
    return [distances[:, :, index] <= COLOR_TOLERANCE_SQ for index in range(len(PALETTE))]


def _centroid(mask: np.ndarray, name: str) -> tuple[int, int]:
    points = np.argwhere(mask)
    if len(points) < 80:
        raise ValueError(f"missing or damaged {name} marker")
    y, x = np.rint(points.mean(axis=0)).astype(int)
    return int(y), int(x)


def _distances(mask: np.ndarray, start: tuple[int, int]) -> np.ndarray:
    height, width = mask.shape
    distance = np.full((height, width), -1, dtype=np.int32)
    sy, sx = start
    if not (0 <= sy < height and 0 <= sx < width and mask[sy, sx]):
        raise ValueError("orange center is not on the visible railway")
    queue = deque([(sy, sx)])
    distance[sy, sx] = 0
    while queue:
        y, x = queue.popleft()
        next_distance = int(distance[y, x]) + 1
        for ny, nx in ((y-1, x), (y+1, x), (y, x-1), (y, x+1)):
            if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and distance[ny, nx] < 0:
                distance[ny, nx] = next_distance
                queue.append((ny, nx))
    return distance


def decision_from_image(image: Image.Image) -> str:
    black, orange_mask, blue_mask, cyan_mask, yellow_mask = _masks(image)
    orange = _centroid(orange_mask, "orange")
    blue = _centroid(blue_mask, "blue")
    cyan = _centroid(cyan_mask, "cyan")
    yellow = _centroid(yellow_mask, "yellow")
    route = black | orange_mask | blue_mask | cyan_mask | yellow_mask
    distances = _distances(route, orange)
    targets = {"blue": blue, "cyan": cyan, "yellow": yellow}
    measured: dict[str, int] = {}
    for name, point in targets.items():
        value = int(distances[point])
        if value < 0:
            raise ValueError(f"{name} is disconnected from the orange terminal")
        measured[name] = value
    if abs(measured["cyan"] - measured["yellow"]) < 32:
        raise ValueError("station precedence is ambiguous at raster resolution")
    return "yes" if measured["cyan"] < measured["yellow"] else "no"
