"""Renderer and analytic scorer for the cyclic gap alternation search.

Each image has four equal panels. Every panel contains four identical disks
with the same inventory of two short and two long cyclic center gaps. Only one
panel puts those two classes in strict alternation around the wraparound.
"""

from __future__ import annotations

import copy
import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

RENDERER_VERSION = "cyclic_gap_alternation_search-0.1.0"
CANVAS_SIZE = (960, 700)
LABELS = ("A", "B", "C", "D")
PANEL_BOXES = (
    (40, 70, 440, 330),
    (520, 70, 920, 330),
    (40, 370, 440, 630),
    (520, 370, 920, 630),
)
BACKGROUND = (255, 255, 255)
PANEL_FILL = (249, 249, 247)
PANEL_BORDER = (151, 158, 166)
LABEL_INK = (36, 42, 48)
SUPER_SAMPLE = 4

# The two angle classes create center gaps separated by roughly 75 pixels on
# the chosen radius. The exact numbers are visible consequences of the
# rendered point locations, not a hidden label field.
TARGET_GAPS = (50.0, 130.0, 50.0, 130.0)
DISTRACTOR_GAPS = (
    (50.0, 50.0, 130.0, 130.0),
    (50.0, 130.0, 130.0, 50.0),
    (130.0, 50.0, 50.0, 130.0),
)
INK_PALETTE = (
    (28, 38, 48),
    (42, 51, 59),
    (55, 61, 68),
)


def _panel_center(box: Iterable[int]) -> tuple[float, float]:
    left, top, right, bottom = (float(value) for value in box)
    return ((left + right) / 2.0, (top + bottom) / 2.0 + 8.0)


def _raw_points(
    center: tuple[float, float], radius: float, rotation: float, gaps: Iterable[float]
) -> list[list[float]]:
    """Place four points on a circle using the supplied cyclic angle gaps."""

    cx, cy = center
    points: list[list[float]] = []
    angle = rotation
    for gap in gaps:
        points.append([cx + radius * math.cos(angle), cy + radius * math.sin(angle)])
        angle += math.radians(float(gap))
    if abs(angle - rotation - 2.0 * math.pi) > 1e-8:
        raise ValueError("cyclic gap pattern must sum to 360 degrees")
    return points


def _center_points(points: list[list[float]], center: tuple[float, float]) -> list[list[float]]:
    """Translate points so every panel has the same visible centroid location."""

    mean_x = sum(float(point[0]) for point in points) / len(points)
    mean_y = sum(float(point[1]) for point in points) / len(points)
    dx = center[0] - mean_x
    dy = center[1] - mean_y
    return [[round(float(x) + dx, 4), round(float(y) + dy, 4)] for x, y in points]


def sample_scene(seed: int) -> dict:
    """Sample a deterministic, balanced four-panel scene from an integer seed."""

    nuisance = random.Random(int(seed) ^ 0x9E3779B9)
    # A cyclic seed mapping gives every contiguous stress batch all four
    # answer positions while keeping the visible geometry independent of the
    # label choice. All rendered nuisances below come from a separate stream.
    target_index = (int(seed) + 1) % 4
    # The smaller ring keeps every disk well inside the panel even for the
    # nonalternating patterns whose centroid translation is not zero.
    ring_radius = float(nuisance.choice((60, 64, 68)))
    disk_radius = int(nuisance.choice((14, 15, 16)))
    ink = list(nuisance.choice(INK_PALETTE))

    panels: list[dict] = []
    for panel_index, (label, box) in enumerate(zip(LABELS, PANEL_BOXES, strict=True)):
        nominal_center = _panel_center(box)
        center = (
            nominal_center[0] + nuisance.uniform(-8.0, 8.0),
            nominal_center[1] + nuisance.uniform(-7.0, 7.0),
        )
        rotation = nuisance.uniform(0.0, 2.0 * math.pi)
        if panel_index == target_index:
            gaps = TARGET_GAPS
        else:
            gaps = DISTRACTOR_GAPS[nuisance.randrange(len(DISTRACTOR_GAPS))]
        points = _raw_points(center, ring_radius, rotation, gaps)
        points = _center_points(points, center)
        # Rendering is independent of record order. This also supplies a
        # naturally exercised list-order alias for the symmetry gate.
        nuisance.shuffle(points)
        panels.append({"label": label, "points": points})

    return {
        "panels": panels,
        "style": {
            "disk_radius": disk_radius,
            "ink": ink,
            "background": list(BACKGROUND),
        },
    }


def render(scene: dict) -> Image.Image:
    """Render the public RGB raster from the visible scene fields."""

    style = scene["style"]
    disk_radius = int(style["disk_radius"])
    ink = tuple(int(channel) for channel in style["ink"])
    background = tuple(int(channel) for channel in style["background"])
    scale = SUPER_SAMPLE
    image = Image.new(
        "RGB", (CANVAS_SIZE[0] * scale, CANVAS_SIZE[1] * scale), background
    )
    draw = ImageDraw.Draw(image)
    for panel, box in zip(scene["panels"], PANEL_BOXES, strict=True):
        left, top, right, bottom = (int(value) for value in box)
        draw.rounded_rectangle(
            (left * scale, top * scale, right * scale, bottom * scale),
            radius=14 * scale,
            fill=PANEL_FILL,
            outline=PANEL_BORDER,
            width=2 * scale,
        )
        for point in panel["points"]:
            x, y = float(point[0]) * scale, float(point[1]) * scale
            radius = disk_radius * scale
            draw.ellipse(
                (
                    round(x - radius),
                    round(y - radius),
                    round(x + radius),
                    round(y + radius),
                ),
                fill=ink,
            )
    image = image.resize(CANVAS_SIZE, Image.Resampling.LANCZOS)
    label_draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    for label, (left, top, _right, _bottom) in zip(LABELS, PANEL_BOXES, strict=True):
        label_draw.text((left + 14, top + 11), label, fill=LABEL_INK, font=font)
    return image


def _ordered_points(points: Iterable[Iterable[float]]) -> list[tuple[float, float]]:
    values = [(float(point[0]), float(point[1])) for point in points]
    if len(values) != 4:
        raise ValueError("each panel must have exactly four points")
    cx = sum(point[0] for point in values) / 4.0
    cy = sum(point[1] for point in values) / 4.0
    # In image coordinates increasing atan2 angle follows clockwise motion on
    # the page. Alternation is invariant to choosing the opposite direction.
    return sorted(values, key=lambda point: math.atan2(point[1] - cy, point[0] - cx))


def _cyclic_gaps(points: Iterable[Iterable[float]]) -> list[float]:
    ordered = _ordered_points(points)
    return [math.dist(ordered[index], ordered[(index + 1) % 4]) for index in range(4)]


def _alternation_score(points: Iterable[Iterable[float]]) -> int:
    gaps = _cyclic_gaps(points)
    ranked = sorted(gaps)
    # The two classes must be separated. With valid generated scenes the gap
    # between ranks 2 and 3 is large; a tie is a quarantine condition.
    if ranked[2] - ranked[1] <= 8.0:
        raise ValueError("short and long gap classes are unresolved")
    cut = (ranked[1] + ranked[2]) / 2.0
    short = [gap < cut for gap in gaps]
    if sum(short) != 2:
        raise ValueError("cyclic gap classes are not two short and two long")
    return sum(short[index] != short[(index + 1) % 4] for index in range(4))


def _scores(scene: dict) -> list[int]:
    panels = scene["panels"]
    if not isinstance(panels, list) or len(panels) != 4:
        raise ValueError("scene must contain four panels")
    return [_alternation_score(panel["points"]) for panel in panels]


def analytic_gold(scene: dict) -> str:
    """Select the unique panel with four alternating class transitions."""

    scores = _scores(scene)
    best = max(scores)
    if best != 4 or scores.count(best) != 1:
        raise ValueError(f"scene has no unique alternating winner: scores={scores}")
    return LABELS[scores.index(best)]


def margin(scene: dict) -> float:
    """Return the top alternation-score gap over the runner-up."""

    scores = sorted(_scores(scene), reverse=True)
    if len(scores) != 4:
        raise ValueError("need four panel scores")
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    """Reject malformed, unresolved, or non-unique finite-resolution scenes."""

    try:
        panels = scene["panels"]
        style = scene["style"]
        disk_radius = float(style["disk_radius"])
        if disk_radius <= 0.0 or not isinstance(panels, list) or len(panels) != 4:
            return True
        for panel in panels:
            points = [(float(point[0]), float(point[1])) for point in panel["points"]]
            if len(points) != 4:
                return True
            if min(
                math.dist(points[first], points[second])
                for first in range(4)
                for second in range(first)
            ) <= 2.0 * disk_radius + 6.0:
                return True
        scores = _scores(scene)
        return max(scores) != 4 or scores.count(max(scores)) != 1 or margin(scene) <= 0.0
    except (KeyError, IndexError, TypeError, ValueError, ZeroDivisionError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse same-colored point record order without changing any pixels."""

    twin = copy.deepcopy(scene)
    for panel in twin["panels"]:
        panel["points"] = list(reversed(panel["points"]))
    return [("reverse_point_record_order", twin)]
