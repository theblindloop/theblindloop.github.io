"""Independent image-only inverse for distinct nearest-square assignment.

The oracle segments visible objects, obtains their centroids from raster pixels,
groups the four visible cards, enumerates all dot-to-square distances, and then
checks global nearest-square coverage. It imports no candidate-side module.
"""

from __future__ import annotations

from collections import deque
import math

import numpy as np
from PIL import Image


LABELS = ("A", "B", "C", "D")


def _components(mask: np.ndarray) -> list[tuple[float, float, int, int, int]]:
    work = mask.copy()
    height, width = work.shape
    found: list[tuple[float, float, int, int, int]] = []
    while True:
        pixels = np.argwhere(work)
        if len(pixels) == 0:
            break
        sy, sx = (int(value) for value in pixels[0])
        queue = deque([(sy, sx)])
        work[sy, sx] = False
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        found.append((float(np.mean(xs)), float(np.mean(ys)),
                      max(xs) - min(xs) + 1, max(ys) - min(ys) + 1, len(xs)))
    return found


def _objects(image: Image.Image) -> tuple[list[tuple[float, float]], list[tuple[float, float]]]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    dark = arr.max(axis=2) < 150
    warm = (
        (arr[:, :, 0] > 180) & (arr[:, :, 1] > 50) & (arr[:, :, 1] < 165)
        & (arr[:, :, 2] < 105) & ((arr[:, :, 0] - arr[:, :, 1]) > 55)
    )
    squares: list[tuple[float, float]] = []
    for cx, cy, width, height, area in _components(dark):
        if 24 <= width <= 34 and 24 <= height <= 34 and area >= 500 and 0.82 <= width / height <= 1.22:
            squares.append((cx, cy))
    dots: list[tuple[float, float]] = []
    for cx, cy, width, height, area in _components(warm):
        if 15 <= width <= 22 and 15 <= height <= 22 and area >= 150 and 0.78 <= width / height <= 1.28:
            dots.append((cx, cy))
    if len(squares) != 12 or len(dots) != 12:
        raise ValueError(f"expected 12 squares and 12 dots, found {len(squares)} and {len(dots)}")
    return squares, dots


def _group_by_card(
    image: Image.Image, squares: list[tuple[float, float]], dots: list[tuple[float, float]]
) -> list[tuple[list[tuple[float, float]], list[tuple[float, float]]]]:
    width, height = image.size
    groups: list[tuple[list[tuple[float, float]], list[tuple[float, float]]]] = []
    for row in range(2):
        for column in range(2):
            def belongs(point: tuple[float, float]) -> bool:
                return (point[0] < width / 2) == (column == 0) and (point[1] < height / 2) == (row == 0)
            sq = [point for point in squares if belongs(point)]
            dt = [point for point in dots if belongs(point)]
            if len(sq) != 3 or len(dt) != 3:
                raise ValueError(f"card must contain three squares and dots, found {len(sq)} and {len(dt)}")
            groups.append((sq, dt))
    return groups


def _coverage_and_gaps(
    squares: list[tuple[float, float]], dots: list[tuple[float, float]]
) -> tuple[int, list[float]]:
    assignments: list[int] = []
    gaps: list[float] = []
    for dot_x, dot_y in dots:
        distances = sorted(
            (math.hypot(dot_x - square_x, dot_y - square_y), index)
            for index, (square_x, square_y) in enumerate(squares)
        )
        assignments.append(distances[0][1])
        gaps.append(distances[1][0] - distances[0][0])
    return len(set(assignments)), gaps


def decision_from_image(image: Image.Image) -> str:
    squares, dots = _objects(image)
    measurements = [_coverage_and_gaps(*group) for group in _group_by_card(image, squares, dots)]
    coverages = [value[0] for value in measurements]
    winners = [index for index, coverage in enumerate(coverages) if coverage == 3]
    if len(winners) != 1 or sorted(coverages, reverse=True) != [3, 2, 2, 2]:
        raise ValueError(f"image does not show one distinct-nearest panel: {coverages}")
    nearest_gap = min(gap for _, gaps in measurements for gap in gaps)
    if nearest_gap < 16.0:
        raise ValueError(f"image-only nearest assignment is ambiguous: {nearest_gap:.2f}px")
    return LABELS[winners[0]]
