from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT, SCALE = 768, 512, 3
BG = (248, 247, 242)
INK = (31, 38, 44)
CYAN = (25, 158, 186)
PANEL = (139, 145, 149)
LABELS = ("A", "B", "C", "D")

# Six-segment orthogonal glyphs. Every member has the same ordered segment
# lengths (2,1,2,1,2,1 units), total ink, and stroke width.
SHAPES = (
    ((0, 0), (2, 0), (2, 1), (0, 1), (0, 2), (2, 2), (2, 3)),
    ((0, 0), (2, 0), (2, 1), (4, 1), (4, 2), (2, 2), (2, 3)),
    ((0, 0), (2, 0), (2, -1), (4, -1), (4, -2), (2, -2), (2, -3)),
    ((0, 0), (0, 2), (1, 2), (1, 4), (2, 4), (2, 2), (3, 2)),
    ((0, 0), (0, 2), (-1, 2), (-1, 0), (-2, 0), (-2, 2), (-3, 2)),
    ((0, 0), (0, 2), (1, 2), (1, 0), (2, 0), (2, -2), (3, -2)),
    ((0, 0), (-2, 0), (-2, 1), (0, 1), (0, 2), (-2, 2), (-2, 3)),
    ((0, 0), (0, -2), (1, -2), (1, 0), (2, 0), (2, -2), (3, -2)),
)


def _edges(points: list[list[int]] | tuple[tuple[int, int], ...]) -> set[tuple[tuple[int, int], tuple[int, int]]]:
    out = set()
    for a, b in zip(points, points[1:]):
        a, b = tuple(a), tuple(b)
        out.add(tuple(sorted((a, b))))
    return out


def _dist(a, b) -> int:
    return len(_edges(a) ^ _edges(b))


def _scores(scene: dict) -> list[int]:
    return [_dist(scene["reference_points"], points) for points in scene["candidate_points"]]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    ref_index = rng.randrange(len(SHAPES))
    target = int(seed) % 4
    others = [i for i in range(len(SHAPES)) if i != ref_index]
    rng.shuffle(others)
    chosen = others[:3]
    candidates = []
    j = 0
    for slot in range(4):
        candidates.append(SHAPES[ref_index] if slot == target else SHAPES[chosen[j]])
        j += slot != target
    centers = [(334, 218), (602, 218), (334, 408), (602, 408)]
    placements = []
    clutter = []
    for cx, cy in centers:
        placements.append([cx + rng.randint(-5, 5), cy + rng.randint(-4, 4)])
        # Short disconnected strokes occupy different corners of each field.
        strokes = []
        for dx, dy, horizontal in ((-88, -55, True), (87, -52, False), (-84, 56, False), (86, 54, True)):
            jitter = rng.randint(-5, 5)
            x, y = cx + dx, cy + dy
            if horizontal:
                strokes.append([[x - 9 + jitter, y], [x + 9 + jitter, y]])
            else:
                strokes.append([[x, y - 9 + jitter], [x, y + 9 + jitter]])
        clutter.append(strokes)
    return {
        "reference_points": [list(p) for p in SHAPES[ref_index]],
        "candidate_points": [[list(p) for p in shape] for shape in candidates],
        "clutter_strokes": clutter,
        "placements": placements,
        "canvas": [WIDTH, HEIGHT],
        "palette": {"background": list(BG), "ink": list(INK), "cyan": list(CYAN), "panel": list(PANEL)},
    }


def _font(size: int, bold: bool = False):
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    path = "/usr/share/fonts/truetype/dejavu/" + name
    try:
        return ImageFont.truetype(path, size * SCALE)
    except OSError:
        return ImageFont.load_default()


def _polyline(draw: ImageDraw.ImageDraw, points, center, color, unit=18, width=8):
    # Pillow's antialiased stroke raster can depend on traversal direction.
    # Canonicalize the undirected path so the declared reversal alias is exact.
    points = [tuple(p) for p in points]
    if tuple(reversed(points)) < tuple(points):
        points = list(reversed(points))
    xs = [p[0] for p in points]
    ys = [p[1] for p in points]
    mx = (min(xs) + max(xs)) / 2.0
    my = (min(ys) + max(ys)) / 2.0
    cx, cy = center
    scaled = [((cx + (p[0] - mx) * unit) * SCALE, (cy + (p[1] - my) * unit) * SCALE) for p in points]
    draw.line(scaled, fill=color, width=width * SCALE, joint="curve")
    radius = width * SCALE // 2
    for x, y in scaled:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)


def render(scene: dict) -> Image.Image:
    w, h = map(int, scene["canvas"])
    palette = scene["palette"]
    bg, ink = tuple(palette["background"]), tuple(palette["ink"])
    cyan, panel = tuple(palette["cyan"]), tuple(palette["panel"])
    image = Image.new("RGB", (w * SCALE, h * SCALE), bg)
    draw = ImageDraw.Draw(image)
    draw.text((w * SCALE // 2, 18 * SCALE), "FIND THE EXACT GLYPH IN THE CLUTTER", font=_font(19, True), fill=ink, anchor="ma")
    draw.rounded_rectangle((22 * SCALE, 126 * SCALE, 194 * SCALE, 316 * SCALE), radius=15 * SCALE, outline=cyan, width=5 * SCALE)
    draw.text((108 * SCALE, 108 * SCALE), "REFERENCE", font=_font(14, True), fill=cyan, anchor="mm")
    _polyline(draw, scene["reference_points"], (108, 220), ink)
    boxes = ((218, 126, 458, 308), (486, 126, 726, 308), (218, 318, 458, 500), (486, 318, 726, 500))
    for idx, (box, points, center, strokes) in enumerate(zip(boxes, scene["candidate_points"], scene["placements"], scene["clutter_strokes"], strict=True)):
        draw.rounded_rectangle(tuple(v * SCALE for v in box), radius=13 * SCALE, outline=panel, width=3 * SCALE)
        for stroke in strokes:
            coords = [(p[0] * SCALE, p[1] * SCALE) for p in stroke]
            draw.line(coords, fill=ink, width=5 * SCALE)
        _polyline(draw, points, center, ink)
        draw.ellipse(((box[0] + 10) * SCALE, (box[1] + 9) * SCALE, (box[0] + 44) * SCALE, (box[1] + 43) * SCALE), fill=bg, outline=panel, width=2 * SCALE)
        draw.text(((box[0] + 27) * SCALE, (box[1] + 26) * SCALE), LABELS[idx], font=_font(18, True), fill=ink, anchor="mm")
    return image.resize((w, h), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    winners = [i for i, score in enumerate(scores) if score == min(scores)]
    if len(winners) != 1 or scores[winners[0]] != 0:
        raise ValueError("no unique exact glyph")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene))
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    scores = sorted(_scores(scene))
    return scores[0] != 0 or scores[1] <= scores[0]


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["reference_points"] = list(reversed(twin["reference_points"]))
    twin["candidate_points"] = [list(reversed(points)) for points in twin["candidate_points"]]
    return [("reverse_undirected_traversals", twin)]
