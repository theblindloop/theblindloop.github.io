from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

INK = np.array((31, 38, 44), dtype=np.int16)
LABELS = ("A", "B", "C", "D")


def _components(mask: np.ndarray) -> list[np.ndarray]:
    work = mask.copy()
    found = []
    height, width = work.shape
    for sy, sx in np.argwhere(work):
        sy, sx = int(sy), int(sx)
        if not work[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        work[sy, sx] = False
        points = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        found.append(np.asarray(points, dtype=np.int32))
    return found


def _glyph_mask(array: np.ndarray, bounds: tuple[int, int, int, int], *, reference: bool = False) -> np.ndarray:
    x0, y0, x1, y1 = bounds
    crop = array[y0:y1, x0:x1]
    distance = np.linalg.norm(crop.astype(np.int16) - INK, axis=2)
    pieces = []
    for points in _components(distance < 68.0):
        ys, xs = points[:, 0], points[:, 1]
        area = len(points)
        bw, bh = int(xs.max() - xs.min() + 1), int(ys.max() - ys.min() + 1)
        if area >= 520 and bw >= 40 and bh >= 40:
            pieces.append(points)
    if len(pieces) != 1:
        raise ValueError("expected exactly one complete glyph in region")
    points = pieces[0]
    ys, xs = points[:, 0], points[:, 1]
    mask = np.zeros((int(ys.max() - ys.min() + 1), int(xs.max() - xs.min() + 1)), dtype=bool)
    mask[ys - ys.min(), xs - xs.min()] = True
    return mask


def _distance(a: np.ndarray, b: np.ndarray) -> float:
    h, w = max(a.shape[0], b.shape[0]), max(a.shape[1], b.shape[1])
    aa, bb = np.zeros((h, w), bool), np.zeros((h, w), bool)
    aa[:a.shape[0], :a.shape[1]] = a
    bb[:b.shape[0], :b.shape[1]] = b
    union = np.count_nonzero(aa | bb)
    return float(np.count_nonzero(aa ^ bb) / max(1, union))


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    height, width = array.shape[:2]
    if width < 600 or height < 400:
        raise ValueError("raster is too small")
    ref = _glyph_mask(array, (20, 120, 200, 320), reference=True)
    regions = ((218, 126, 458, 308), (486, 126, 726, 308), (218, 318, 458, 500), (486, 318, 726, 500))
    candidates = [_glyph_mask(array, region) for region in regions]
    scores = sorted((_distance(ref, candidate), idx) for idx, candidate in enumerate(candidates))
    if scores[0][0] > 0.012 or scores[1][0] - scores[0][0] < 0.035:
        raise ValueError("no unique exact raster glyph match")
    return LABELS[scores[0][1]]
