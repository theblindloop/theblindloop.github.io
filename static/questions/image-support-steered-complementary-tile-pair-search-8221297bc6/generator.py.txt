"""Direct RGB renderer and analytic world for complementary tile pairs."""

from __future__ import annotations

import copy
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "complementary-tile-pair-search-0.1.0"
QUADRANTS = ("NW", "NE", "SW", "SE")
CANVAS = (640, 640)
GUIDE = (207, 218, 229)
PAIR_OFFSET = 90
MIN_RUNNER_UP_MARGIN = 3

_CENTERS = {
    "NW": (160, 160),
    "NE": (480, 160),
    "SW": (160, 480),
    "SE": (480, 480),
}

# Each mask has twelve occupied cells.  Every complement has thirteen cells;
# both sides are connected polyominoes, so a pair is still two detached tiles.
# The deliberately varied silhouettes prevent one local contour from naming the
# answer.  The light frame exposes the shared five-by-five coordinate system.
MASKS = (
    "0000100001010110111100111",
    "0001100111001110011100001",
    "0000000001011110111100111",
    "1111101011010000111000000",
    "0000100001000011011111111",
    "0011100111000110001100011",
    "0000000001011110011101111",
    "1000011000110101111011000",
)

BACKGROUND_PALETTE = (
    (248, 250, 247),
    (243, 248, 251),
    (250, 247, 241),
    (245, 249, 245),
)
INK_PALETTE = (
    (28, 43, 63),
    (55, 38, 72),
    (37, 67, 51),
    (77, 42, 39),
)


def _mask_rows(mask: str) -> tuple[str, ...]:
    if len(mask) != 25 or set(mask) - {"0", "1"}:
        raise ValueError("a tile mask must contain exactly 25 binary cells")
    return tuple(mask[row * 5 : row * 5 + 5] for row in range(5))


def _complement(mask: str) -> str:
    return "".join("0" if cell == "1" else "1" for cell in mask)


def _hamming(left: str, right: str) -> int:
    return sum(a != b for a, b in zip(left, right, strict=True))


def _connected(mask: str, value: str) -> bool:
    cells = {
        (row, col)
        for row in range(5)
        for col in range(5)
        if mask[row * 5 + col] == value
    }
    if not cells:
        return False
    pending = [next(iter(cells))]
    seen = set(pending)
    while pending:
        row, col = pending.pop()
        for drow, dcol in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            neighbor = (row + drow, col + dcol)
            if neighbor in cells and neighbor not in seen:
                seen.add(neighbor)
                pending.append(neighbor)
    return seen == cells


def _validate_masks() -> None:
    if len(MASKS) < 4 or len(set(MASKS)) != len(MASKS):
        raise AssertionError("mask library must contain distinct masks")
    for mask in MASKS:
        _mask_rows(mask)
        if mask.count("1") != 12:
            raise AssertionError("base masks must contain twelve cells")
        if not _connected(mask, "1") or not _connected(_complement(mask), "1"):
            raise AssertionError("both a tile and its complement must be connected")


_validate_masks()


def _choose_base_indices(rng: random.Random) -> list[int]:
    """Choose four local contours with a visible runner-up margin."""

    for _ in range(200):
        choices = rng.sample(range(len(MASKS)), 4)
        if min(
            _hamming(MASKS[left], MASKS[right])
            for position, left in enumerate(choices)
            for right in choices[position + 1 :]
        ) >= MIN_RUNNER_UP_MARGIN:
            return choices
    # The library is curated so this fallback is deterministic and separated.
    return [0, 1, 3, 7]


def _distractor_index(rng: random.Random, base_index: int) -> int:
    options = [
        index
        for index, mask in enumerate(MASKS)
        if index != base_index and _hamming(MASKS[base_index], mask) >= MIN_RUNNER_UP_MARGIN
    ]
    if not options:
        raise AssertionError("mask library has no separated distractor")
    return rng.choice(options)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced scene; the seed modulo four counterbalances answer position."""

    rng = random.Random(int(seed))
    target_index = int(seed) % len(QUADRANTS)
    base_indices = _choose_base_indices(rng)
    pair_masks: list[list[str]] = []
    for quadrant_index, base_index in enumerate(base_indices):
        partner_index = (
            base_index
            if quadrant_index == target_index
            else _distractor_index(rng, base_index)
        )
        pair_masks.append([MASKS[base_index], _complement(MASKS[partner_index])])
    draw_order = list(range(8))
    rng.shuffle(draw_order)
    return {
        "pair_masks": pair_masks,
        "background": list(rng.choice(BACKGROUND_PALETTE)),
        "ink": list(rng.choice(INK_PALETTE)),
        "cell_size": rng.choice((11, 12, 13)),
        "draw_order": draw_order,
    }


def _tile_origin(quadrant: str, side: int, cell_size: int) -> tuple[int, int]:
    cx, cy = _CENTERS[quadrant]
    tile_size = 5 * cell_size
    center_x = cx + (-PAIR_OFFSET if side == 0 else PAIR_OFFSET)
    center_y = cy + (-PAIR_OFFSET if side == 0 else PAIR_OFFSET)
    return (round(center_x - tile_size / 2), round(center_y - tile_size / 2))


def _draw_tile(
    draw: ImageDraw.ImageDraw,
    mask: str,
    origin: tuple[int, int],
    cell_size: int,
    ink: tuple[int, int, int],
) -> None:
    x0, y0 = origin
    tile_size = 5 * cell_size
    # A separated, neutral frame makes the finite 5-by-5 coordinate system
    # observable without joining the two decisive dark components.
    draw.rectangle(
        [x0 - 2, y0 - 2, x0 + tile_size + 1, y0 + tile_size + 1],
        outline=GUIDE,
        width=2,
    )
    for row in range(5):
        for col in range(5):
            if mask[row * 5 + col] != "1":
                continue
            left = x0 + col * cell_size
            top = y0 + row * cell_size
            draw.rectangle(
                [left, top, left + cell_size - 1, top + cell_size - 1],
                fill=ink,
            )


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only the visible fields of a sampled scene."""

    cell_size = int(scene["cell_size"])
    if cell_size < 8 or cell_size > 20:
        raise ValueError("cell size is outside the visible raster range")
    image = Image.new("RGB", CANVAS, tuple(int(value) for value in scene["background"]))
    draw = ImageDraw.Draw(image)
    pair_masks = scene["pair_masks"]
    ink = tuple(int(value) for value in scene["ink"])
    for flat_index in scene["draw_order"]:
        quadrant_index, side = divmod(int(flat_index), 2)
        quadrant = QUADRANTS[quadrant_index]
        mask = str(pair_masks[quadrant_index][side])
        _draw_tile(draw, mask, _tile_origin(quadrant, side, cell_size), cell_size, ink)
    return image


def _pair_scores(pair_masks: list[list[str]]) -> list[int]:
    scores = []
    for pair in pair_masks:
        if not isinstance(pair, list) or len(pair) != 2:
            raise ValueError("each quadrant must contain two tile masks")
        first, second = str(pair[0]), str(pair[1])
        _mask_rows(first)
        _mask_rows(second)
        scores.append(_hamming(second, _complement(first)))
    if len(scores) != 4:
        raise ValueError("four quadrant pair scores are required")
    return scores


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the unique quadrant with zero cellwise complement error."""

    scores = _pair_scores(scene["pair_masks"])
    best = min(scores)
    winners = [index for index, score in enumerate(scores) if score == best]
    return QUADRANTS[winners[0]]


def margin(scene: dict[str, Any]) -> float:
    """Return the gap between the best complement score and its runner-up."""

    scores = sorted(_pair_scores(scene["pair_masks"]))
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Quarantine ties, weak raster margins, and non-complement tile counts."""

    try:
        scores = _pair_scores(scene["pair_masks"])
    except (KeyError, TypeError, ValueError):
        return True
    ranked = sorted(scores)
    best = ranked[0]
    return scores.count(best) != 1 or (ranked[1] - ranked[0]) < MIN_RUNNER_UP_MARGIN


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """Reversing non-overlapping draw records is a nontrivial pixel alias."""

    twin = copy.deepcopy(scene)
    twin["draw_order"] = list(reversed(twin["draw_order"]))
    return [("reverse_nonoverlapping_tile_draw_order", twin)]
