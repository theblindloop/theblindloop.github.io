"""Deterministic renderer and analytic arm for modular color unmasking."""

from __future__ import annotations

import copy
import random
from typing import Iterable

from PIL import Image, ImageDraw

N = 7
ANSWERS = ("up", "right", "down", "left")
PHYSICAL_COLORS = (
    (245, 183, 54),   # gold
    (232, 99, 111),   # coral
    (48, 191, 201),   # cyan
    (139, 105, 204),  # violet
)
GRID_RGB = (38, 47, 56)
BG_RGB = (250, 248, 242)
NEUTRAL_RGB = (174, 180, 184)


def _base_up() -> list[list[int]]:
    grid = [[0 for _ in range(N)] for _ in range(N)]
    for r, c in ((0, 3), (1, 2), (1, 3), (1, 4),
                 (2, 1), (2, 2), (2, 3), (2, 4), (2, 5),
                 (3, 3), (4, 3), (5, 3), (6, 3)):
        grid[r][c] = 1
    return grid


def _rotate_clockwise(grid: list[list[int]]) -> list[list[int]]:
    return [[grid[N - 1 - r][c] for r in range(N)] for c in range(N)]


def _templates() -> dict[str, list[list[int]]]:
    out: dict[str, list[list[int]]] = {}
    grid = _base_up()
    for name in ANSWERS:
        out[name] = grid
        grid = _rotate_clockwise(grid)
    return out


TEMPLATES = _templates()


def _decode(scene: dict) -> list[list[int]]:
    encoded = scene["encoded_cells"]
    row_keys = scene["row_keys"]
    col_keys = scene["col_keys"]
    return [
        [(int(encoded[r][c]) - int(row_keys[r]) - int(col_keys[c])) % 4 for c in range(N)]
        for r in range(N)
    ]


def _ink(decoded: list[list[int]]) -> list[list[int]]:
    return [[1 if value == 0 else 0 for value in row] for row in decoded]


def _distances(bits: list[list[int]]) -> list[tuple[int, str]]:
    scores = []
    for name, template in TEMPLATES.items():
        distance = sum(bits[r][c] != template[r][c] for r in range(N) for c in range(N))
        scores.append((int(distance), name))
    return sorted(scores)


def _classify_bits(bits: list[list[int]]) -> tuple[str, int]:
    scores = _distances(bits)
    return scores[0][1], scores[1][0] - scores[0][0]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = ANSWERS[int(seed) % len(ANSWERS)]
    bits = copy.deepcopy(TEMPLATES[target])

    # Add zero to four label-independent flips toward the opposite-direction
    # runner-up. This deterministically covers Hamming gaps 12, 10, 8, 6, and 4.
    opposite = ANSWERS[(ANSWERS.index(target) + 2) % len(ANSWERS)]
    positions = [
        (r, c) for r in range(N) for c in range(N)
        if TEMPLATES[target][r][c] != TEMPLATES[opposite][r][c]
    ]
    rng.shuffle(positions)
    wanted = (int(seed) // 4) % 5
    accepted = 0
    for r, c in positions:
        if accepted >= wanted:
            break
        bits[r][c] ^= 1
        winner, gap = _classify_bits(bits)
        if winner == target and gap >= 4:
            accepted += 1
        else:
            bits[r][c] ^= 1

    # Decoded 0 is ink and decoded 2 is blank, so raw color adjacency is unhelpful.
    decoded = [[0 if bits[r][c] else 2 for c in range(N)] for r in range(N)]
    row_keys = [0, 0, 1, 1, 2, 2, 3]
    col_keys = [0, 0, 1, 1, 2, 2, 3]
    rng.shuffle(row_keys)
    rng.shuffle(col_keys)
    encoded = [
        [(decoded[r][c] + row_keys[r] + col_keys[c]) % 4 for c in range(N)]
        for r in range(N)
    ]

    palette = list(PHYSICAL_COLORS)
    rng.shuffle(palette)
    cell_px = rng.choice((34, 36, 38))
    origin_x = rng.randint(164, 188)
    origin_y = rng.randint(228, 248)

    distractors: list[list[int]] = []
    for i in range(6):
        side = i % 2
        x = rng.randint(42, 105) if side == 0 else rng.randint(510, 578)
        y = rng.randint(170, 548)
        radius = rng.randint(3, 6)
        distractors.append([x, y, radius])

    return {
        "encoded_cells": encoded,
        "row_keys": row_keys,
        "col_keys": col_keys,
        "palette_rgb": [list(rgb) for rgb in palette],
        "cell_px": cell_px,
        "origin_x": origin_x,
        "origin_y": origin_y,
        "distractors": distractors,
        "canvas_width": 620,
        "canvas_height": 620,
    }


def render(scene: dict) -> Image.Image:
    encoded = scene["encoded_cells"]
    row_keys = scene["row_keys"]
    col_keys = scene["col_keys"]
    palette = [tuple(int(v) for v in rgb) for rgb in scene["palette_rgb"]]
    cell = int(scene["cell_px"])
    x0, y0 = int(scene["origin_x"]), int(scene["origin_y"])
    width, height = int(scene["canvas_width"]), int(scene["canvas_height"])
    image = Image.new("RGB", (width, height), BG_RGB)
    draw = ImageDraw.Draw(image)

    for x, y, radius in scene["distractors"]:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=NEUTRAL_RGB)

    # The left-to-right rail visibly binds arbitrary physical colors to residues 0..3.
    swatch_w, swatch_h, gap = 44, 28, 18
    rail_w = 4 * swatch_w + 3 * gap
    rail_x = x0 + (N * cell - rail_w) // 2
    rail_y = 58
    for index, color in enumerate(palette):
        sx = rail_x + index * (swatch_w + gap)
        draw.rectangle((sx, rail_y, sx + swatch_w, rail_y + swatch_h), fill=color,
                       outline=GRID_RGB, width=3)
        draw.text((sx + swatch_w // 2 - 3, rail_y - 17), str(index), fill=GRID_RGB)
        if index < 3:
            ax = sx + swatch_w + 4
            ay = rail_y + swatch_h // 2
            draw.line((ax, ay, ax + gap - 8, ay), fill=GRID_RGB, width=2)
            draw.polygon(((ax + gap - 8, ay), (ax + gap - 13, ay - 4),
                          (ax + gap - 13, ay + 4)), fill=GRID_RGB)

    bead_radius = max(7, int(round(cell * 0.22)))
    bead_offset = int(round(cell * 0.70))
    for r, key in enumerate(row_keys):
        cx, cy = x0 - bead_offset, y0 + int((r + 0.5) * cell)
        draw.ellipse((cx - bead_radius, cy - bead_radius, cx + bead_radius, cy + bead_radius),
                     fill=palette[int(key)], outline=GRID_RGB, width=2)
    for c, key in enumerate(col_keys):
        cx, cy = x0 + int((c + 0.5) * cell), y0 - bead_offset
        draw.ellipse((cx - bead_radius, cy - bead_radius, cx + bead_radius, cy + bead_radius),
                     fill=palette[int(key)], outline=GRID_RGB, width=2)

    for r in range(N):
        for c in range(N):
            xa, ya = x0 + c * cell, y0 + r * cell
            draw.rectangle((xa, ya, xa + cell, ya + cell), fill=palette[int(encoded[r][c])])
    for i in range(N + 1):
        draw.line((x0 + i * cell, y0, x0 + i * cell, y0 + N * cell),
                  fill=GRID_RGB, width=3)
        draw.line((x0, y0 + i * cell, x0 + N * cell, y0 + i * cell),
                  fill=GRID_RGB, width=3)
    return image


def analytic_gold(scene: dict) -> str:
    bits = _ink(_decode(scene))
    return _classify_bits(bits)[0]


def margin(scene: dict) -> float:
    bits = _ink(_decode(scene))
    return float(_classify_bits(bits)[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["distractors"] = list(reversed(twin["distractors"]))
    return [("reverse_nonoverlapping_distractor_order", twin)]
