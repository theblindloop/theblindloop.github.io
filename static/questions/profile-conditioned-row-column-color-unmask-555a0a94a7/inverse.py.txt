"""Independent PNG-only inverse for the modular color mosaic.

The inverse finds the largest connected charcoal component (the grid), learns
the arbitrary color-to-residue binding from the ordered top rail, samples the
row and column beads relative to the recovered grid, performs modular
subtraction, and reports the winner/runner-up Hamming gap.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

GRID_RGB = (38, 47, 56)
PHYSICAL_COLORS = (
    (245, 183, 54),
    (232, 99, 111),
    (48, 191, 201),
    (139, 105, 204),
)
N = 7
ANSWER_ORDER = ("up", "right", "down", "left")


class PixelDecodeError(ValueError):
    pass


def _largest_component_bbox(mask: np.ndarray) -> tuple[int, int, int, int]:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    best: list[tuple[int, int]] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if visited[y, x]:
            continue
        queue = deque([(y, x)])
        visited[y, x] = True
        component: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            component.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    queue.append((ny, nx))
        if len(component) > len(best):
            best = component
    if len(best) < 1500:
        raise PixelDecodeError("main grid component is missing")
    ys = [p[0] for p in best]
    xs = [p[1] for p in best]
    return min(xs), min(ys), max(xs), max(ys)


def _rail_colors(arr: np.ndarray, grid_top: int, cell: float) -> list[tuple[int, int, int]]:
    cutoff = max(1, int(grid_top - 2.0 * cell))
    located: list[tuple[float, tuple[int, int, int]]] = []
    for rgb in PHYSICAL_COLORS:
        mask = np.all(arr[:cutoff] == np.array(rgb, dtype=np.uint8), axis=2)
        ys, xs = np.nonzero(mask)
        if len(xs) < 600:
            raise PixelDecodeError("a palette rail swatch is missing")
        located.append((float(np.median(xs)), rgb))
    located.sort()
    if len({round(x) for x, _ in located}) != 4:
        raise PixelDecodeError("palette rail order is ambiguous")
    return [rgb for _, rgb in located]


def _nearest_index(rgb: np.ndarray, palette: list[tuple[int, int, int]]) -> int:
    values = np.asarray(palette, dtype=np.int32)
    delta = values - rgb.astype(np.int32)
    distances = np.sum(delta * delta, axis=1)
    winner = int(np.argmin(distances))
    if int(distances[winner]) > 900:
        raise PixelDecodeError("sampled mark is not a rail color")
    return winner


def _dominant_palette_index(region: np.ndarray, palette: list[tuple[int, int, int]]) -> int:
    counts = [int(np.all(region == np.array(rgb, dtype=np.uint8), axis=2).sum()) for rgb in palette]
    order = sorted(range(4), key=lambda i: counts[i], reverse=True)
    if counts[order[0]] < 60 or counts[order[0]] <= counts[order[1]]:
        raise PixelDecodeError("key bead is missing or ambiguous")
    return order[0]


def _up_template() -> np.ndarray:
    out = np.zeros((N, N), dtype=np.uint8)
    for r, c in ((0, 3), (1, 2), (1, 3), (1, 4),
                 (2, 1), (2, 2), (2, 3), (2, 4), (2, 5),
                 (3, 3), (4, 3), (5, 3), (6, 3)):
        out[r, c] = 1
    return out


def _classify(bits: np.ndarray) -> tuple[str, int, list[int]]:
    template = _up_template()
    distances: list[int] = []
    for _ in ANSWER_ORDER:
        distances.append(int(np.count_nonzero(bits != template)))
        template = np.rot90(template, -1)
    ranked = sorted((distance, index) for index, distance in enumerate(distances))
    gap = ranked[1][0] - ranked[0][0]
    if gap < 1:
        raise PixelDecodeError("decoded arrow has a tied best template")
    return ANSWER_ORDER[ranked[0][1]], int(gap), distances


def measurement_from_image(image: Image.Image) -> dict:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    grid_mask = np.all(arr == np.array(GRID_RGB, dtype=np.uint8), axis=2)
    x0, y0, x1, y1 = _largest_component_bbox(grid_mask)
    grid_w, grid_h = float(x1 - x0), float(y1 - y0)
    if min(grid_w, grid_h) < 200 or abs(grid_w - grid_h) > 4:
        raise PixelDecodeError("recovered grid geometry is invalid")
    cell = (grid_w + grid_h) / (2.0 * N)
    palette = _rail_colors(arr, y0, cell)

    row_keys: list[int] = []
    col_keys: list[int] = []
    for r in range(N):
        cy = y0 + (r + 0.5) * cell
        region = arr[max(0, int(cy - 0.34 * cell)):min(arr.shape[0], int(cy + 0.34 * cell) + 1),
                     max(0, int(x0 - 1.25 * cell)):x0]
        row_keys.append(_dominant_palette_index(region, palette))
    for c in range(N):
        cx = x0 + (c + 0.5) * cell
        region = arr[max(0, int(y0 - 1.25 * cell)):y0,
                     max(0, int(cx - 0.34 * cell)):min(arr.shape[1], int(cx + 0.34 * cell) + 1)]
        col_keys.append(_dominant_palette_index(region, palette))

    encoded = np.zeros((N, N), dtype=np.int16)
    decoded = np.zeros((N, N), dtype=np.int16)
    for r in range(N):
        for c in range(N):
            px = int(round(x0 + (c + 0.5) * cell))
            py = int(round(y0 + (r + 0.5) * cell))
            encoded[r, c] = _nearest_index(arr[py, px], palette)
            decoded[r, c] = (encoded[r, c] - row_keys[r] - col_keys[c]) % 4
    bits = (decoded == 0).astype(np.uint8)
    decision, gap, distances = _classify(bits)
    return {
        "decision": decision,
        "margin": gap,
        "distances": distances,
        "decoded_cells": decoded.tolist(),
        "grid_bbox": [x0, y0, x1, y1],
    }


def decision_from_image(image: Image.Image) -> str:
    return str(measurement_from_image(image)["decision"])
