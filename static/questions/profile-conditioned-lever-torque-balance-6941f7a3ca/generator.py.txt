"""Measurement world: which side of a lever tips down under two torques.

A horizontal gray beam is drawn across a light canvas, resting on a small
black fulcrum triangle at a visible x-position. A red disk sits on the beam to
the left of the fulcrum and a blue disk sits on the beam to the right of the
fulcrum. The scene declares that each disk's weight is proportional to its area
(pi * radius^2) and its torque is weight times the horizontal distance from its
center to the fulcrum. The decision measures which color's torque is larger:
that is the color of the side that tips down.

Latent scene z carries the three directly-visible marks: ``red`` (left disk
center-x and radius), ``blue`` (right disk center-x and radius), and
``fulcrum`` (the x-position of the fulcrum triangle's apex on the beam). All
three fields are rendered as visible ink (the two solid colored disks and the
black fulcrum triangle), so the latent representation is injective and no
nontrivial pixel-identical gold-preserving latent symmetry exists.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "lever_torque_balance-0.1.0"

PALETTE = ["red", "blue"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
}
BEAM_RGB = (120, 120, 120)
FULCRUM_RGB = (30, 30, 30)
BACKGROUND = (250, 250, 250)

WIDTH = 560
HEIGHT = 360
BEAM_Y = 210

MIN_RADIUS = 20.0
MAX_RADIUS = 38.0
MIN_DIST = 55.0
MAX_DIST = 165.0
MIN_FULCRUM = 170.0
MAX_FULCRUM = WIDTH - 170.0
DISK_CLEARANCE = 6.0

# Torque is measured in pixel^3 (area pixels x distance pixels); these are the
# verifiability thresholds on the absolute torque gap.
QUARANTINE = 18000.0
MIN_MARGIN = 26000.0
MIN_RELATIVE_GAP = 0.12


def _area(radius: float) -> float:
    return math.pi * radius * radius


def _torque(radius: float, dist: float) -> float:
    return _area(radius) * dist


def _sample_disk(rng, radius: float, allowed_range: tuple[float, float]) -> float:
    lo, hi = allowed_range
    cx = rng.uniform(lo, hi)
    return cx


def sample_scene(seed: int) -> dict:
    """Deterministic, counterbalanced scene with a guaranteed larger torque."""
    rng = random.Random(seed * 7919 + 3)
    winner = PALETTE[seed % 2]

    while True:
        fulcrum = rng.uniform(MIN_FULCRUM, MAX_FULCRUM)
        r_red = rng.uniform(MIN_RADIUS, MAX_RADIUS)
        r_blue = rng.uniform(MIN_RADIUS, MAX_RADIUS)
        red_cx = _sample_disk(rng, r_red, (DISK_CLEARANCE + MIN_RADIUS, fulcrum - MIN_DIST))
        blue_cx = _sample_disk(rng, r_blue, (fulcrum + MIN_DIST, WIDTH - DISK_CLEARANCE - MIN_RADIUS))

        d_red = fulcrum - red_cx
        d_blue = blue_cx - fulcrum
        if d_red < MIN_DIST or d_blue < MIN_DIST:
            continue
        if d_red > MAX_DIST or d_blue > MAX_DIST:
            continue

        t_red = _torque(r_red, d_red)
        t_blue = _torque(r_blue, d_blue)
        gap = abs(t_red - t_blue)
        if gap < MIN_MARGIN:
            continue
        if gap < MIN_RELATIVE_GAP * max(t_red, t_blue):
            continue
        analytic_winner = "red" if t_red > t_blue else "blue"
        if analytic_winner != winner:
            continue
        break

    return {
        "red": {"cx": round(red_cx, 2), "radius": round(r_red, 2)},
        "blue": {"cx": round(blue_cx, 2), "radius": round(r_blue, 2)},
        "fulcrum": round(fulcrum, 2),
    }


def render(scene: dict) -> Image.Image:
    """Render I = R(z): gray beam, black fulcrum, and two colored disks."""
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(img)
    draw.line([(30, BEAM_Y), (WIDTH - 30, BEAM_Y)], fill=BEAM_RGB, width=5)

    fx = scene["fulcrum"]
    tri = [(fx, BEAM_Y), (fx - 15, BEAM_Y + 28), (fx + 15, BEAM_Y + 28)]
    draw.polygon(tri, fill=FULCRUM_RGB)

    red_cx = scene["red"]["cx"]
    red_r = scene["red"]["radius"]
    draw.ellipse(
        [red_cx - red_r, BEAM_Y - red_r, red_cx + red_r, BEAM_Y + red_r],
        fill=COLOR_RGB["red"],
    )
    blue_cx = scene["blue"]["cx"]
    blue_r = scene["blue"]["radius"]
    draw.ellipse(
        [blue_cx - blue_r, BEAM_Y - blue_r, blue_cx + blue_r, BEAM_Y + blue_r],
        fill=COLOR_RGB["blue"],
    )
    return img


def analytic_gold(scene: dict) -> str:
    """Color of the side with the larger torque (the side that tips down)."""
    fulcrum = scene["fulcrum"]
    t_red = _torque(scene["red"]["radius"], fulcrum - scene["red"]["cx"])
    t_blue = _torque(scene["blue"]["radius"], scene["blue"]["cx"] - fulcrum)
    return "red" if t_red > t_blue else "blue"


def margin(scene: dict) -> float:
    """Verifiability margin: the absolute torque gap (pixel^3)."""
    fulcrum = scene["fulcrum"]
    t_red = _torque(scene["red"]["radius"], fulcrum - scene["red"]["cx"])
    t_blue = _torque(scene["blue"]["radius"], scene["blue"]["cx"] - fulcrum)
    return float(abs(t_red - t_blue))


def is_quarantined(scene: dict) -> bool:
    """Near-boundary: the torque gap is small relative to the torques."""
    fulcrum = scene["fulcrum"]
    t_red = _torque(scene["red"]["radius"], fulcrum - scene["red"]["cx"])
    t_blue = _torque(scene["blue"]["radius"], scene["blue"]["cx"] - fulcrum)
    gap = abs(t_red - t_blue)
    if gap < QUARANTINE:
        return True
    if gap < MIN_RELATIVE_GAP * max(t_red, t_blue):
        return True
    return False


def latent_symmetries(scene: dict) -> list:
    """not_applicable: no nontrivial pixel-identical gold-preserving symmetry.

    Every latent field (the red disk's center and radius, the blue disk's
    center and radius, and the fulcrum x-position) is directly rendered as
    visible ink: the disk centers and radii are the drawn disks and the
    fulcrum is the drawn black triangle. Swapping the two color labels swaps
    which color is correct, and any global reflection changes the relative
    distances to the fulcrum while turning the raster into a distinct image,
    so no latent transform is both pixel-identical and gold-preserving.
    """
    return []
