"""Independent pixel-only inverse arm for lever_torque_balance.

Receives ONLY the rasterized PNG (plus the world's public color naming
convention) and re-measures, from pixels alone, the area and horizontal
distance-from-fulcrum of each colored disk, computes each side's torque
(area x distance), and answers which color has the larger torque. It shares no
geometry or code with renderer.render() and never imports the renderer,
prompts, generator, verifier, or any analytic-gold code.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "lever_torque_balance-oracle-0.1.0"

PALETTE = ["red", "blue"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
}
BEAM_RGB = (120, 120, 120)
FULCRUM_RGB = (30, 30, 30)
BACKGROUND = (250, 250, 250)


def decision_from_image(image: Image.Image) -> str:
    """Color of the side with the larger torque (the side that tips down)."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    refs = np.stack(
        [
            np.array(COLOR_RGB[c], dtype=np.float32) for c in PALETTE
        ]
        + [
            np.array(BEAM_RGB, dtype=np.float32),
            np.array(FULCRUM_RGB, dtype=np.float32),
            np.array(BACKGROUND, dtype=np.float32),
        ],
        axis=0,
    )
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    norms = np.linalg.norm(diffs, axis=-1)
    labels = np.argmin(norms, axis=-1)

    ys, xs = np.nonzero(labels == 3)  # fulcrum black triangle index
    if ys.size < 20:
        return "none"
    fulcrum_x = float(xs.mean())

    torques = {}
    for idx, color in enumerate(PALETTE):
        ys_c, xs_c = np.nonzero(labels == idx)
        if ys_c.size < 40:
            return "none"
        cx = float(xs_c.mean())
        area = float(ys_c.size)
        dist = abs(cx - fulcrum_x)
        torques[color] = area * dist

    t_red = torques["red"]
    t_blue = torques["blue"]
    gap = abs(t_red - t_blue)
    if gap < 20000.0 or gap < 0.08 * max(t_red, t_blue):
        return "none"
    return "red" if t_red > t_blue else "blue"
