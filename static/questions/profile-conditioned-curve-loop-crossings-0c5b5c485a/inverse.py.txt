"""Independent pixel-only inverse arm for the curve-loop crossing-count world.

Receives ONLY the final PNG. It never imports or reads the renderer, the analytic
scene, prompts, or renderer state. It recovers the decision by:

1. masking the red thread pixels and the black loop-ring pixels;
2. recovering the loop center (mass centroid of the ring pixels) and radius (mean
   pixel distance from that center);
3. forming the thin annular "band" of pixels whose distance from the recovered
   center lies near the recovered radius (the ring-rim neighbourhood);
4. grouping the red pixels that fall inside that band into 8-connected
   components -- each component is one transverse crossing of the thread through
   the loop boundary;
5. returning how many such crossing components exist (0..3).
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "curve_loop_crossings-oracle-0.4.0"

_RED = (220, 30, 40)
_BLACK_MAX = 60
_BAND = 13.0
_MIN_PIXELS = 12


def _components8(mask: np.ndarray):
    h, w = mask.shape
    labels = np.zeros_like(mask, dtype=np.int32)
    current = 0
    comps = []
    for y in range(h):
        for x in range(w):
            if not mask[y, x] or labels[y, x]:
                continue
            current += 1
            labels[y, x] = current
            stack = [(x, y)]
            pts = []
            while stack:
                cx, cy = stack.pop()
                pts.append((cx, cy))
                for nx in (cx - 1, cx, cx + 1):
                    for ny in (cy - 1, cy, cy + 1):
                        if (
                            0 <= nx < w
                            and 0 <= ny < h
                            and mask[ny, nx]
                            and labels[ny, nx] == 0
                        ):
                            labels[ny, nx] = current
                            stack.append((nx, ny))
            comps.append(pts)
    return comps


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.int16)
    r = arr[..., 0]
    g = arr[..., 1]
    b = arr[..., 2]

    # tolerant reddish-thread mask (captures anti-aliased red pixels)
    red = (r >= 160) & (g <= 110) & (b <= 110) & (r >= g + 60) & (r >= b + 60)
    black = (r <= _BLACK_MAX) & (g <= _BLACK_MAX) & (b <= _BLACK_MAX)

    ys, xs = np.nonzero(black)
    if len(xs) < 40:
        return "0"
    cx = float(xs.mean())
    cy = float(ys.mean())
    dists = np.hypot(xs.astype(float) - cx, ys.astype(float) - cy)
    radius = float(dists.mean())

    red_ys, red_xs = np.nonzero(red)
    if len(red_xs) < 4:
        return "0"
    d_red = np.hypot(red_xs.astype(float) - cx, red_ys.astype(float) - cy)
    in_band = np.abs(d_red - radius) <= _BAND
    band_mask = np.zeros_like(red)
    band_mask[red_ys[in_band], red_xs[in_band]] = True

    count = 0
    for comp in _components8(band_mask):
        if len(comp) >= _MIN_PIXELS:
            count += 1
    return str(min(count, 3))
