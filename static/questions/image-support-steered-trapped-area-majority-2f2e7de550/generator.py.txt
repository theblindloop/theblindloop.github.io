"""Deterministic latent sampler, raster renderer, and analytic arm."""

from __future__ import annotations

import random
from collections import deque

from PIL import Image, ImageDraw

CANVAS = 512
N = 10
FRAME = 7
PALETTES = ("#168C86", "#237CA3", "#3B8E67", "#506FA7")


def _rotate(grid: list[list[int]]) -> list[list[int]]:
    return [list(row) for row in zip(*grid[::-1])]


def _build_grid(answer_yes: bool, variant: int, rotation: int) -> list[list[int]]:
    grid = [[1 for _ in range(N)] for _ in range(N)]
    large, small = ((3, 6), (2, 5)) if variant == 0 else ((4, 5), (2, 5))
    enclosed = large if answer_yes else small
    opened = small if answer_yes else large
    oh, ow = opened
    ox = (N - ow) // 2
    for y in range(oh):
        for x in range(ox, ox + ow):
            grid[y][x] = 0
    eh, ew = enclosed
    ex = (N - ew) // 2
    ey = N - eh - 2
    for y in range(ey, ey + eh):
        for x in range(ex, ex + ew):
            grid[y][x] = 0
    for _ in range(rotation % 4):
        grid = _rotate(grid)
    return grid


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    return {
        "grid": _build_grid((int(seed) % 2) == 0, rng.randrange(2), rng.randrange(4)),
        "palette": PALETTES[rng.randrange(len(PALETTES))],
        "offset": [rng.randint(-26, 26), rng.randint(-26, 26)],
        "cell_size": rng.choice((22, 24, 26)),
    }


def render(scene: dict) -> Image.Image:
    grid = scene["grid"]
    color = scene["palette"]
    dx, dy = scene["offset"]
    cell = int(scene["cell_size"])
    side = N * cell
    x0 = (CANVAS - side) // 2 + int(dx)
    y0 = (CANVAS - side) // 2 + int(dy)
    image = Image.new("RGB", (CANVAS, CANVAS), "white")
    draw = ImageDraw.Draw(image)
    for y, row in enumerate(grid):
        for x, value in enumerate(row):
            if value:
                draw.rectangle((x0 + x * cell, y0 + y * cell,
                                x0 + (x + 1) * cell - 1, y0 + (y + 1) * cell - 1),
                               fill=color)
    draw.rectangle((x0 - FRAME, y0 - FRAME, x0 + side - 1 + FRAME, y0 + side - 1 + FRAME),
                   outline="#161B22", width=FRAME)
    return image


def _counts(grid: list[list[int]]) -> tuple[int, int]:
    h, w = len(grid), len(grid[0])
    white = {(y, x) for y in range(h) for x in range(w) if grid[y][x] == 0}
    reachable = {p for p in white if p[0] in {0, h - 1} or p[1] in {0, w - 1}}
    queue = deque(reachable)
    while queue:
        y, x = queue.popleft()
        for p in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if p in white and p not in reachable:
                reachable.add(p)
                queue.append(p)
    return len(white - reachable), len(reachable)


def _component_count(points: set[tuple[int, int]]) -> int:
    unseen = set(points)
    count = 0
    while unseen:
        count += 1
        queue = deque([unseen.pop()])
        while queue:
            y, x = queue.popleft()
            for p in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if p in unseen:
                    unseen.remove(p)
                    queue.append(p)
    return count


def analytic_gold(scene: dict) -> str:
    enclosed, opened = _counts(scene["grid"])
    return "yes" if enclosed > opened else "no"


def margin(scene: dict) -> float:
    enclosed, opened = _counts(scene["grid"])
    return float(abs(enclosed - opened))


def is_quarantined(scene: dict) -> bool:
    grid = scene["grid"]
    if len(grid) != N or any(len(row) != N for row in grid):
        return True
    enclosed, opened = _counts(grid)
    teal = {(y, x) for y in range(N) for x in range(N) if grid[y][x] == 1}
    white = {(y, x) for y in range(N) for x in range(N) if grid[y][x] == 0}
    edge_white = {p for p in white if p[0] in {0, N - 1} or p[1] in {0, N - 1}}
    return (enclosed == opened or min(enclosed, opened) < 8 or
            _component_count(teal) != 1 or _component_count(edge_white) != 1 or
            _component_count(white) != 2)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
