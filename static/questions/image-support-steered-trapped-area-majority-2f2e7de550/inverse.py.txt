"""Independent inverse arm: final RGB pixels are the only input."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _component_sizes(mask: np.ndarray) -> list[int]:
    unseen = {(int(y), int(x)) for y, x in np.argwhere(mask)}
    sizes: list[int] = []
    while unseen:
        queue = deque([unseen.pop()])
        size = 1
        while queue:
            y, x = queue.popleft()
            for point in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if point in unseen:
                    unseen.remove(point)
                    queue.append(point)
                    size += 1
        sizes.append(size)
    return sorted(sizes, reverse=True)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or min(array.shape[:2]) < 100:
        raise ValueError("invalid raster")
    black = np.max(array, axis=2) < 55
    points = np.argwhere(black)
    if len(points) < 500:
        raise ValueError("black frame missing")
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    height, width = int(y1 - y0 + 1), int(x1 - x0 + 1)
    if not (220 <= height <= 290 and 220 <= width <= 290 and abs(height - width) <= 2):
        raise ValueError("frame geometry invalid")
    band = 8
    sides = (black[y0:y0 + band, x0:x1 + 1], black[y1 - band + 1:y1 + 1, x0:x1 + 1],
             black[y0:y1 + 1, x0:x0 + band], black[y0:y1 + 1, x1 - band + 1:x1 + 1])
    if min(float(side.mean()) for side in sides) < 0.68:
        raise ValueError("frame is broken")
    inner = array[y0 + 7:y1 - 6, x0 + 7:x1 - 6]
    mx = inner.max(axis=2).astype(np.int16)
    mn = inner.min(axis=2).astype(np.int16)
    teal = ((mx - mn) > 45) & (mx < 210)
    fraction = float(teal.mean())
    if not 0.55 <= fraction <= 0.78:
        raise ValueError("field fill outside contract")
    teal_sizes = _component_sizes(teal)
    if len(teal_sizes) != 1 or teal_sizes[0] < 0.50 * teal.size:
        raise ValueError("teal field is not one connected region")
    white = ~teal
    h, w = white.shape
    reachable = np.zeros_like(white)
    queue: deque[tuple[int, int]] = deque()
    for x in range(w):
        for y in (0, h - 1):
            if white[y, x] and not reachable[y, x]:
                reachable[y, x] = True
                queue.append((y, x))
    for y in range(h):
        for x in (0, w - 1):
            if white[y, x] and not reachable[y, x]:
                reachable[y, x] = True
                queue.append((y, x))
    while queue:
        y, x = queue.popleft()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < h and 0 <= nx < w and white[ny, nx] and not reachable[ny, nx]:
                reachable[ny, nx] = True
                queue.append((ny, nx))
    enclosed = white & ~reachable
    if len(_component_sizes(reachable)) != 1 or len(_component_sizes(enclosed)) != 1:
        raise ValueError("expected one open and one enclosed white component")
    open_area, enclosed_area = int(reachable.sum()), int(enclosed.sum())
    if min(open_area, enclosed_area) < 0.07 * white.size:
        raise ValueError("white component too small")
    if abs(enclosed_area - open_area) < 0.04 * white.size:
        raise ValueError("area comparison too close")
    return "yes" if enclosed_area > open_area else "no"
