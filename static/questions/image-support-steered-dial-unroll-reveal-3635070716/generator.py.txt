"""Renderer for the dial-unroll reveal world.

One compact dial carries a ring of polar cells.  The abstract symbol exists only
after the ring is unrolled into a band-by-sector rectangle starting at the black
notch, so all decisive evidence lives inside one small connected disk.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "dial_unroll_reveal-renderer-0.1.0"

CANVAS = 800
BANDS = 5
SECTORS = 8
R_OUT = 72
R_IN = 27
BAND_W = (R_OUT - R_IN) / BANDS
HUB_WIDTH = 2
NOTCH_OUT = 15
NOTCH_HALF_DEG = 7.0
INK_CELLS = 16
MAX_FLIP_PAIRS = 3
QUARANTINE_MARGIN = 10

WHITE = (255, 255, 255)
BODY = (222, 226, 232)
INK = (38, 54, 86)
NOTCH = (0, 0, 0)
DECOY_BODY = (240, 242, 245)
DECOY_INK = (206, 212, 220)

GLYPHS: dict[str, tuple[str, ...]] = {
    "kite": ("00000000", "01111000", "11100010", "00111110", "00010110"),
    "vane": ("01001100", "01111000", "00001001", "00001001", "00011111"),
    "crest": ("11010000", "11010100", "01111110", "00100010", "00100000"),
    "spur": ("00111111", "01101100", "00100000", "01110000", "00110000"),
}
GLYPH_NAMES = tuple(sorted(GLYPHS))
OUTER_HEAVY = tuple(
    sorted(name for name, rows in GLYPHS.items() if rows[BANDS - 1].count("1") >= 3)
)


def _grid(rows: tuple[str, ...]) -> list[list[int]]:
    return [[1 if ch == "1" else 0 for ch in row] for row in rows]


def _hamming(a: list[list[int]], b: list[list[int]]) -> int:
    return sum(
        a[r][c] != b[r][c] for r in range(BANDS) for c in range(SECTORS)
    )


def _ranked(bitmap: list[list[int]]) -> list[tuple[int, str]]:
    return sorted((_hamming(bitmap, _grid(GLYPHS[name])), name) for name in GLYPH_NAMES)


def _bitmap(scene: dict) -> list[list[int]]:
    return _grid(tuple(scene["bitmap"]))


def _random_pattern(rng: random.Random) -> tuple[str, ...]:
    cells = rng.sample(range(BANDS * SECTORS), INK_CELLS)
    return tuple(
        "".join("1" if r * SECTORS + c in cells else "0" for c in range(SECTORS))
        for r in range(BANDS)
    )


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    while True:
        glyph = rng.choice(GLYPH_NAMES)
        grid = _grid(GLYPHS[glyph])
        on = [(r, c) for r in range(BANDS) for c in range(SECTORS) if grid[r][c]]
        off = [(r, c) for r in range(BANDS) for c in range(SECTORS) if not grid[r][c]]
        pairs = rng.randint(0, MAX_FLIP_PAIRS)
        drop = rng.sample(on, pairs)
        add = rng.sample(off, pairs)
        for r, c in drop:
            grid[r][c] = 0
        for r, c in add:
            grid[r][c] = 1
        ranked = _ranked(grid)
        if ranked[0][1] != glyph:
            continue
        bitmap = ["".join(str(v) for v in row) for row in grid]

        # The declared operation must be load bearing: no other start sector,
        # no reversed sense and no inside-out reading may land near a symbol.
        def near(rows: list[list[int]]) -> bool:
            return _ranked(rows)[0][0] <= 6

        bad = False
        for k in range(SECTORS):
            for flip_r in (False, True):
                for mirror in (False, True):
                    if k == 0 and not flip_r and not mirror:
                        continue
                    view = [row[::-1] if mirror else list(row) for row in grid]
                    if flip_r:
                        view = view[::-1]
                    view = [[row[(c - k) % SECTORS] for c in range(SECTORS)] for row in view]
                    if near(view):
                        bad = True
        if bad:
            continue

        start_deg = float(rng.randrange(0, 360, 3))
        cx = rng.randint(150, CANVAS - 150)
        cy = rng.randint(150, CANVAS - 150)
        decoys = []
        guard = 0
        while len(decoys) < 2 and guard < 400:
            guard += 1
            radius = rng.choice((40, 48, 56))
            dx = rng.randint(radius + 24, CANVAS - radius - 24)
            dy = rng.randint(radius + 24, CANVAS - radius - 24)
            if math.hypot(dx - cx, dy - cy) < R_OUT + radius + NOTCH_OUT + 40:
                continue
            if any(
                math.hypot(dx - d["cx"], dy - d["cy"]) < radius + d["r_out"] + 30
                for d in decoys
            ):
                continue
            decoys.append(
                {
                    "cx": dx,
                    "cy": dy,
                    "r_out": radius,
                    "r_in": int(round(radius * R_IN / R_OUT)),
                    "start_deg": float(rng.randrange(0, 360, 5)),
                    "pattern": list(_random_pattern(rng)),
                }
            )
        if len(decoys) < 2:
            continue
        return {
            "canvas": CANVAS,
            "bands": BANDS,
            "sectors": SECTORS,
            "r_out": R_OUT,
            "r_in": R_IN,
            "cx": cx,
            "cy": cy,
            "start_deg": start_deg,
            "bitmap": bitmap,
            "flip_pairs": pairs,
            "decoys": decoys,
        }


def _draw_dial(
    draw: ImageDraw.ImageDraw,
    cx: float,
    cy: float,
    r_out: int,
    r_in: int,
    start_deg: float,
    pattern: list[str],
    body: tuple[int, int, int],
    ink: tuple[int, int, int],
) -> None:
    band_w = (r_out - r_in) / BANDS
    draw.ellipse([cx - r_out, cy - r_out, cx + r_out, cy + r_out], fill=body)
    for b in range(BANDS - 1, -1, -1):
        r_hi = r_in + band_w * (b + 1)
        for s in range(SECTORS):
            if pattern[b][s] != "1":
                continue
            a0 = start_deg + s * (360.0 / SECTORS)
            draw.pieslice(
                [cx - r_hi, cy - r_hi, cx + r_hi, cy + r_hi],
                a0,
                a0 + 360.0 / SECTORS,
                fill=ink,
            )
        r_lo = r_in + band_w * b
        draw.ellipse([cx - r_lo, cy - r_lo, cx + r_lo, cy + r_lo], fill=body)
    draw.ellipse(
        [cx - r_in, cy - r_in, cx + r_in, cy + r_in], outline=ink, width=HUB_WIDTH
    )


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas"])
    image = Image.new("RGB", (size, size), WHITE)
    draw = ImageDraw.Draw(image)
    for decoy in scene["decoys"]:
        _draw_dial(
            draw,
            decoy["cx"],
            decoy["cy"],
            decoy["r_out"],
            decoy["r_in"],
            decoy["start_deg"],
            list(decoy["pattern"]),
            DECOY_BODY,
            DECOY_INK,
        )
    cx, cy = float(scene["cx"]), float(scene["cy"])
    _draw_dial(
        draw,
        cx,
        cy,
        int(scene["r_out"]),
        int(scene["r_in"]),
        float(scene["start_deg"]),
        list(scene["bitmap"]),
        BODY,
        INK,
    )
    axis = math.radians(float(scene["start_deg"]) + 180.0 / SECTORS)
    r_out = int(scene["r_out"])
    apex = (cx + math.cos(axis) * (r_out + NOTCH_OUT), cy + math.sin(axis) * (r_out + NOTCH_OUT))
    left = math.radians(float(scene["start_deg"]) + 180.0 / SECTORS - NOTCH_HALF_DEG)
    right = math.radians(float(scene["start_deg"]) + 180.0 / SECTORS + NOTCH_HALF_DEG)
    base_r = r_out - 3
    draw.polygon(
        [
            apex,
            (cx + math.cos(left) * base_r, cy + math.sin(left) * base_r),
            (cx + math.cos(right) * base_r, cy + math.sin(right) * base_r),
        ],
        fill=NOTCH,
    )
    return image


def analytic_gold(scene: dict) -> str:
    return _ranked(_bitmap(scene))[0][1]


def margin(scene: dict) -> float:
    ranked = _ranked(_bitmap(scene))
    return float(ranked[1][0] - ranked[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
