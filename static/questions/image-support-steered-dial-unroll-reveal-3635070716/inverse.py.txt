"""Pixel-only inverse arm for the dial-unroll reveal world.

Input is the final PNG and the public question contract only: no scene, no
renderer, no analytic gold.  The arm locates the visible fiducials (the marked
dial's body, its hub circle and its black notch), executes the declared unroll
of the ring into a band-by-sector rectangle, and classifies the symbol.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "dial_unroll_reveal-oracle-0.1.0"

BANDS = 5
SECTORS = 8
ABSTAIN = "abstain"

_BODY = (222, 226, 232)
_INK = (38, 54, 86)
_NOTCH_MAX = 40

GLYPHS: dict[str, tuple[str, ...]] = {
    "kite": ("00000000", "01111000", "11100010", "00111110", "00010110"),
    "vane": ("01001100", "01111000", "00001001", "00001001", "00011111"),
    "crest": ("11010000", "11010100", "01111110", "00100010", "00100000"),
    "spur": ("00111111", "01101100", "00100000", "01110000", "00110000"),
}


def _mask(arr: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    return np.all(arr == np.array(rgb, dtype=np.int16), axis=-1)


def _largest_component(mask: np.ndarray) -> np.ndarray:
    labels = np.zeros(mask.shape, dtype=np.int32)
    current = 0
    best: np.ndarray | None = None
    best_size = 0
    ys, xs = np.nonzero(mask)
    for y0, x0 in zip(ys.tolist(), xs.tolist()):
        if labels[y0, x0]:
            continue
        current += 1
        stack = [(y0, x0)]
        labels[y0, x0] = current
        size = 0
        while stack:
            y, x = stack.pop()
            size += 1
            for dy, dx in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                ny, nx = y + dy, x + dx
                if 0 <= ny < mask.shape[0] and 0 <= nx < mask.shape[1]:
                    if mask[ny, nx] and not labels[ny, nx]:
                        labels[ny, nx] = current
                        stack.append((ny, nx))
        if size > best_size:
            best_size = size
            best = labels == current
    return best if best is not None else np.zeros(mask.shape, dtype=bool)


def _hamming(a: list[list[int]], b: tuple[str, ...]) -> int:
    return sum(
        a[r][c] != (1 if b[r][c] == "1" else 0)
        for r in range(BANDS)
        for c in range(SECTORS)
    )


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    body = _mask(arr, _BODY)
    ink = _mask(arr, _INK)
    dial = _largest_component(body | ink)
    if dial.sum() < 400:
        return ABSTAIN
    ys, xs = np.nonzero(dial)
    cy = float(ys.mean())
    cx = float(xs.mean())
    radial = np.hypot(ys - cy, xs - cx)
    r_out = float(radial.max())

    ink_dial = ink & dial
    iy, ix = np.nonzero(ink_dial)
    if iy.size == 0:
        return ABSTAIN
    ink_radial = np.hypot(iy - cy, ix - cx)
    r_in = float(ink_radial.min()) + 1.0
    if not 0.15 * r_out < r_in < 0.6 * r_out:
        return ABSTAIN

    dark = np.all(arr <= _NOTCH_MAX, axis=-1)
    ny, nx = np.nonzero(dark)
    if ny.size == 0:
        return ABSTAIN
    keep = np.hypot(ny - cy, nx - cx) < r_out * 2.2
    if not keep.any():
        return ABSTAIN
    notch_angle = math.degrees(
        math.atan2(float(ny[keep].mean()) - cy, float(nx[keep].mean()) - cx)
    )
    start = notch_angle - 180.0 / SECTORS

    band_w = (r_out - r_in) / BANDS
    bitmap: list[list[int]] = []
    for b in range(BANDS):
        radius = r_in + band_w * (b + 0.5)
        row: list[int] = []
        for s in range(SECTORS):
            angle = math.radians(start + (s + 0.5) * (360.0 / SECTORS))
            px = cx + math.cos(angle) * radius
            py = cy + math.sin(angle) * radius
            votes = 0
            total = 0
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    yy = int(round(py)) + dy
                    xx = int(round(px)) + dx
                    if 0 <= yy < arr.shape[0] and 0 <= xx < arr.shape[1]:
                        total += 1
                        if ink_dial[yy, xx]:
                            votes += 1
            if total == 0:
                return ABSTAIN
            row.append(1 if votes * 2 > total else 0)
        bitmap.append(row)

    ranked = sorted((_hamming(bitmap, GLYPHS[name]), name) for name in sorted(GLYPHS))
    if ranked[1][0] - ranked[0][0] <= 0:
        return ABSTAIN
    return ranked[0][1]
