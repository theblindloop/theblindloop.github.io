"""Deterministic latent sampler, analytic arm, and Pillow renderer."""

from __future__ import annotations

import copy
import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 768
HEIGHT = 320
BACKGROUND = (248, 246, 239)
CORD = (33, 43, 65)
LABEL = (92, 98, 112)
BASE_PALETTE = [
    [218, 62, 72],
    [241, 171, 45],
    [27, 154, 151],
    [126, 82, 194],
]


def _rng(seed: int) -> random.Random:
    return random.Random(int(seed) ^ 0xB4A6E17)


def _make_codes(rng: random.Random, positive: bool) -> list[int]:
    if positive:
        a = rng.randrange(4)
        b = rng.randrange(4)
        while b == a:
            b = rng.randrange(4)
        c = rng.randrange(4)
        return [a, b, c, b, a]

    # Draw every position from the same marginal distribution as the controls,
    # then reject only palindromes. No position or individual color predicts class.
    while True:
        values = [rng.randrange(4) for _ in range(5)]
        if values[0] != values[4] or values[1] != values[3]:
            return values


def _curve_y(x: float, phase: float, amp1: float, amp2: float) -> float:
    t = (x - 48.0) / (WIDTH - 96.0)
    return (
        HEIGHT / 2
        + amp1 * math.sin(2.0 * math.pi * t + phase)
        + amp2 * math.sin(4.0 * math.pi * t - 0.55 * phase)
    )


def sample_scene(seed: int) -> dict[str, Any]:
    rng = _rng(seed)
    positive = int(seed) % 2 == 0
    phase = rng.uniform(-0.55, 0.55)
    amp1 = rng.uniform(35.0, 48.0)
    amp2 = rng.uniform(10.0, 18.0)
    xs = [48 + i * 8 for i in range(85)]
    cord_points = [[int(x), int(round(_curve_y(x, phase, amp1, amp2)))] for x in xs]
    bead_xs = [128, 256, 384, 512, 640]
    bead_centers = [
        [x, int(round(_curve_y(x, phase, amp1, amp2)))] for x in bead_xs
    ]
    rotation = rng.randrange(4)
    palette = BASE_PALETTE[rotation:] + BASE_PALETTE[:rotation]
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "color_codes": _make_codes(rng, positive),
        "palette": copy.deepcopy(palette),
        "cord_points": cord_points,
        "bead_centers": bead_centers,
        "bead_radius": rng.randint(23, 27),
        "cord_width": rng.randint(8, 11),
    }


def analytic_gold(scene: dict[str, Any]) -> str:
    codes = scene["color_codes"]
    return "yes" if codes[0] == codes[4] and codes[1] == codes[3] else "no"


def margin(scene: dict[str, Any]) -> float:
    codes = scene["color_codes"]
    if codes[0] == codes[4] and codes[1] == codes[3]:
        return 1.0
    return float((codes[0] != codes[4]) + (codes[1] != codes[3]))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return bool(scene["bead_radius"] < 20 or scene["cord_width"] < 7)


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # Increment each abstract code, and rotate the lookup in the opposite
    # direction: new_palette[new_code] == old_palette[old_code].
    twin = copy.deepcopy(scene)
    twin["color_codes"] = [(value + 1) % 4 for value in scene["color_codes"]]
    palette = scene["palette"]
    twin["palette"] = [copy.deepcopy(palette[-1]), *copy.deepcopy(palette[:-1])]
    return [("cyclic_code_palette_relabel", twin)]


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict[str, Any]) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    codes = list(scene["color_codes"])
    palette = [tuple(value) for value in scene["palette"]]
    points = [tuple(value) for value in scene["cord_points"]]
    centers = [tuple(value) for value in scene["bead_centers"]]
    radius = int(scene["bead_radius"])
    cord_width = int(scene["cord_width"])

    scale = 2
    image = Image.new("RGB", (width * scale, height * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)

    # Quiet registration marks make the object read as a crafted design sheet;
    # they are deliberately outside every oracle color mask.
    for x in range(96, width - 64, 96):
        draw.line(
            [(x * scale, 48 * scale), (x * scale, (height - 48) * scale)],
            fill=(229, 226, 219),
            width=1 * scale,
        )

    scaled_points = [(x * scale, y * scale) for x, y in points]
    draw.line(scaled_points, fill=CORD, width=cord_width * scale, joint="curve")

    # Beads cover the cord locally, but their filled disks touch its two sides;
    # the union remains a single visible threaded component.
    for code, (cx, cy) in zip(codes, centers, strict=True):
        box = [
            (cx - radius) * scale,
            (cy - radius) * scale,
            (cx + radius) * scale,
            (cy + radius) * scale,
        ]
        draw.ellipse(box, fill=palette[code], outline=CORD, width=3 * scale)
        hole = max(3, cord_width // 2)
        draw.ellipse(
            [
                (cx - hole) * scale,
                (cy - hole) * scale,
                (cx + hole) * scale,
                (cy + hole) * scale,
            ],
            fill=CORD,
        )

    # Reinforce the exposed tails, keeping the decisive support continuous from
    # the first canvas column group to the last.
    draw.ellipse(
        [(39 * scale, (points[0][1] - 9) * scale), (57 * scale, (points[0][1] + 9) * scale)],
        fill=CORD,
    )
    draw.ellipse(
        [((width - 57) * scale, (points[-1][1] - 9) * scale), ((width - 39) * scale, (points[-1][1] + 9) * scale)],
        fill=CORD,
    )

    image = image.resize((width, height), Image.Resampling.LANCZOS)
    draw = ImageDraw.Draw(image)
    draw.text((25, 24), "START", fill=LABEL, font=_font(17, bold=True))
    end_text = "END"
    bbox = draw.textbbox((0, 0), end_text, font=_font(17, bold=True))
    draw.text((width - (bbox[2] - bbox[0]) - 25, 24), end_text, fill=LABEL, font=_font(17, bold=True))
    draw.text(
        (width // 2 - 126, height - 37),
        "VISIBLE BEAD PATTERN",
        fill=(146, 145, 148),
        font=_font(14),
    )
    return image
