"""Independent pixel-only inverse arm for bracelet_color_palindrome."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

BACKGROUND = np.array((248, 246, 239), dtype=np.int16)
CORD = np.array((33, 43, 65), dtype=np.int16)
COLORS = np.array(
    [(218, 62, 72), (241, 171, 45), (27, 154, 151), (126, 82, 194)],
    dtype=np.int16,
)


class OracleAbstention(ValueError):
    """The raster no longer contains one unambiguous threaded five-bead path."""


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y = int(y)
        x = int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny in range(max(0, cy - 1), min(height, cy + 2)):
                for nx in range(max(0, cx - 1), min(width, cx + 2)):
                    if mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        output.append(np.asarray(points, dtype=np.int32))
    return output


def _measure(image: Image.Image) -> list[int]:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    height, width = array.shape[:2]
    if width < 500 or height < 180:
        raise OracleAbstention("canvas too small for the declared bracelet")

    cord_distance = np.linalg.norm(array - CORD, axis=2)
    color_distance = np.linalg.norm(array[:, :, None, :] - COLORS[None, None, :, :], axis=3)
    nearest_color = color_distance.argmin(axis=2)
    best_color_distance = color_distance.min(axis=2)

    # The threaded object is the union of cord ink and bead fill. Requiring one
    # spanning component makes every distant stretch of the visible cord causal.
    object_mask = (cord_distance < 58.0) | (best_color_distance < 58.0)
    object_components = [component for component in _components(object_mask) if len(component) >= 90]
    if len(object_components) != 1:
        raise OracleAbstention(f"expected one threaded component, found {len(object_components)}")
    object_points = object_components[0]
    xs = object_points[:, 1]
    if int(xs.min()) > int(width * 0.075) or int(xs.max()) < int(width * 0.925):
        raise OracleAbstention("threaded component does not visibly span START to END")

    beads: list[tuple[float, float, int, int]] = []
    for color_index in range(len(COLORS)):
        mask = (nearest_color == color_index) & (best_color_distance < 55.0)
        for component in _components(mask):
            area = len(component)
            if area < 650:
                continue
            ys_comp = component[:, 0]
            xs_comp = component[:, 1]
            box_w = int(xs_comp.max() - xs_comp.min() + 1)
            box_h = int(ys_comp.max() - ys_comp.min() + 1)
            if not (36 <= box_w <= 58 and 36 <= box_h <= 58):
                raise OracleAbstention("large color component is not bead-sized")
            beads.append((float(xs_comp.mean()), float(ys_comp.mean()), color_index, area))

    if len(beads) != 5:
        raise OracleAbstention(f"expected five large colored beads, found {len(beads)}")
    beads.sort(key=lambda item: item[0])
    centers = [item[0] for item in beads]
    gaps = np.diff(np.asarray(centers))
    if np.any(gaps < 85.0) or np.any(gaps > 170.0):
        raise OracleAbstention("bead order is crowded or has a missing interval")
    if centers[0] > width * 0.22 or centers[-1] < width * 0.78:
        raise OracleAbstention("beads do not cover the visible cord")
    return [item[2] for item in beads]


def decision_from_image(image: Image.Image) -> str:
    codes = _measure(image)
    return "yes" if codes[0] == codes[4] and codes[1] == codes[3] else "no"
