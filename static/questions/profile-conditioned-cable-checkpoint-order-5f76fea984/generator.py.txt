"""Deterministic scene sampler and Pillow renderer for a path-order task."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

SIZE = 512
BACKGROUND = (250, 249, 245)
CABLE = (24, 50, 74)
GREEN = (31, 157, 105)
ORANGE = (235, 126, 34)
PURPLE = (132, 80, 190)
PALE = (189, 195, 199)

# Consecutive orthogonal segments form one non-self-intersecting cable.  The
# checkpoint at index 4 is physically nearer the start than index 2, despite
# being much farther along the cable: this blocks straight-line proximity.
BASE_ROUTE = [
    [70, 80], [430, 80], [430, 180], [120, 180],
    [120, 280], [400, 280], [400, 390], [70, 390],
]
BASE_DISTRACTORS = [
    [[235, 125], [335, 125]],
    [[205, 230], [315, 230]],
    [[185, 340], [285, 340]],
]


def _pose(point: list[int], code: int) -> list[int]:
    """Apply one of eight rigid square symmetries around the canvas center."""
    x, y = point
    if code >= 4:
        x = SIZE - x
    for _ in range(code % 4):
        x, y = SIZE - y, x
    return [int(x), int(y)]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    pose = rng.randrange(8)
    route = [_pose(p, pose) for p in BASE_ROUTE]
    distractors = [[_pose(p, pose) for p in line] for line in BASE_DISTRACTORS]
    rng.shuffle(distractors)
    # The answer identity is independent of pose and alternates for any run of
    # consecutive seeds, including every public 24-scene stress batch.
    if int(seed) % 2 == 0:
        orange_index, purple_index = 2, 4
    else:
        orange_index, purple_index = 4, 2
    return {
        "route_points": route,
        "start_index": 0,
        "orange_index": orange_index,
        "purple_index": purple_index,
        "transform": pose,
        "distractors": distractors,
        "stroke_width": 10 + 2 * rng.randrange(2),
    }


def _arc_distance(points: list[list[int]], start: int, target: int) -> float:
    lo, hi = sorted((int(start), int(target)))
    return sum(
        math.hypot(points[i + 1][0] - points[i][0], points[i + 1][1] - points[i][1])
        for i in range(lo, hi)
    )


def analytic_gold(scene: dict) -> str:
    points = scene["route_points"]
    start = scene["start_index"]
    orange = _arc_distance(points, start, scene["orange_index"])
    purple = _arc_distance(points, start, scene["purple_index"])
    return "orange" if orange < purple else "purple"


def margin(scene: dict) -> float:
    points = scene["route_points"]
    start = scene["start_index"]
    orange = _arc_distance(points, start, scene["orange_index"])
    purple = _arc_distance(points, start, scene["purple_index"])
    return float(abs(orange - purple))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 35.0


def render(scene: dict) -> Image.Image:
    scale = 3
    image = Image.new("RGB", (SIZE * scale, SIZE * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)

    def pts(values: list[list[int]]) -> list[tuple[int, int]]:
        return [(int(x * scale), int(y * scale)) for x, y in values]

    # Pale fragments are deliberately disconnected and cannot answer the rule.
    for fragment in scene["distractors"]:
        draw.line(pts(fragment), fill=PALE, width=6 * scale)

    route = scene["route_points"]
    width = int(scene["stroke_width"]) * scale
    segments = sorted(
        (list(a), list(b)) if tuple(a) <= tuple(b) else (list(b), list(a))
        for a, b in zip(route, route[1:])
    )
    for first, second in segments:
        draw.line(pts([first, second]), fill=CABLE, width=width)
    radius = width // 2
    for x, y in pts(sorted(route)):
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=CABLE)

    marker_radius = 15 * scale
    for index, color in (
        (scene["orange_index"], ORANGE),
        (scene["purple_index"], PURPLE),
    ):
        x, y = pts([route[index]])[0]
        draw.ellipse(
            (x - marker_radius, y - marker_radius, x + marker_radius, y + marker_radius),
            fill=color,
        )

    # A green ring marks the endpoint from which traversal begins. Its dark
    # center keeps the ring visibly joined to the cable.
    x, y = pts([route[scene["start_index"]]])[0]
    outer, inner = 18 * scale, 9 * scale
    draw.ellipse((x - outer, y - outer, x + outer, y + outer), fill=GREEN)
    draw.ellipse((x - inner, y - inner, x + inner, y + inner), fill=CABLE)

    image = image.resize((SIZE, SIZE), Image.Resampling.LANCZOS)
    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse the latent point list while preserving every rendered role."""
    twin = deepcopy(scene)
    count = len(scene["route_points"])
    twin["route_points"] = list(reversed(scene["route_points"]))
    twin["start_index"] = count - 1 - int(scene["start_index"])
    twin["orange_index"] = count - 1 - int(scene["orange_index"])
    twin["purple_index"] = count - 1 - int(scene["purple_index"])
    return [("reverse_route_parameterization", twin)]
