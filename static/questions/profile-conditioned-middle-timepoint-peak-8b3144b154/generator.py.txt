"""Deterministic renderer and analytic arm for ``middle_timepoint_peak``.

The latent series encode four equal-ink three-sample trends.  Exactly one has a
strict local maximum at the middle sample.  Rendering positions the trends in
separate, randomly assigned vertical bands so draw order is a genuine latent
alias while color, band, translation, and palette are nuisance variables.
"""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 640
HEIGHT = 480
LABELS = ("red", "blue", "green", "orange")
PALETTES: dict[str, dict[str, tuple[int, int, int]]] = {
    "standard": {
        "red": (202, 55, 62),
        "blue": (45, 105, 190),
        "green": (28, 143, 85),
        "orange": (230, 126, 34),
    },
    "deep": {
        "red": (176, 39, 55),
        "blue": (30, 88, 165),
        "green": (18, 121, 73),
        "orange": (205, 103, 21),
    },
}
BACKGROUND = (255, 255, 255)
PANEL = (249, 251, 253)
GUIDE = (207, 214, 224)
INK = (53, 61, 72)
FAINT = (235, 239, 244)


def sample_scene(seed: int) -> dict:
    """Return a balanced randomized scene for one integer seed."""

    rng = random.Random(int(seed))
    target = LABELS[int(seed) % len(LABELS)]
    other_labels = [label for label in LABELS if label != target]
    rng.shuffle(other_labels)

    shapes = [
        (target, [0.0, 1.0, 0.0]),       # unique local peak
        (other_labels[0], [0.0, -1.0, 0.0]),
        (other_labels[1], [-1.0, 0.0, 1.0]),
        (other_labels[2], [1.0, 0.0, -1.0]),
    ]
    bands = list(range(4))
    rng.shuffle(bands)
    series = [
        {"label": label, "band": band, "values": values}
        for (label, values), band in zip(shapes, bands, strict=True)
    ]
    rng.shuffle(series)

    amplitudes = (10, 14, 18, 22, 26)
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "chart_left": 100 + rng.randint(-14, 14),
        "chart_top": 73 + rng.randint(-8, 8),
        "chart_width": 440,
        "chart_height": 330,
        "amplitude": amplitudes[int(seed) % len(amplitudes)],
        "palette_variant": "standard" if (int(seed) // 4) % 2 == 0 else "deep",
        "series": series,
    }


def _font(size: int, supersample: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("DejaVuSans.ttf", size * supersample)
    except OSError:
        return ImageFont.load_default()


def _layout(scene: dict) -> tuple[list[float], list[float]]:
    left = float(scene["chart_left"])
    top = float(scene["chart_top"])
    xs = [left + 50.0, left + 220.0, left + 390.0]
    ys = [top + 48.0 + 78.0 * band for band in range(4)]
    return xs, ys


def render(scene: dict) -> Image.Image:
    """Render a scene to one deterministic RGB raster."""

    ss = 2
    width = int(scene["width"])
    height = int(scene["height"])
    left = float(scene["chart_left"])
    top = float(scene["chart_top"])
    chart_width = float(scene["chart_width"])
    chart_height = float(scene["chart_height"])
    amplitude = float(scene["amplitude"])
    palette = PALETTES[str(scene["palette_variant"])]
    series = scene["series"]
    xs, band_centers = _layout(scene)

    image = Image.new("RGB", (width * ss, height * ss), BACKGROUND)
    draw = ImageDraw.Draw(image)

    def xy(values: tuple[float, float, float, float]) -> tuple[int, int, int, int]:
        return tuple(int(round(value * ss)) for value in values)  # type: ignore[return-value]

    draw.rectangle(xy((left, top, left + chart_width, top + chart_height)), fill=PANEL)
    for band in range(1, 4):
        y = top + 9.0 + 78.0 * band
        draw.line(xy((left + 1, y, left + chart_width - 1, y)), fill=FAINT, width=ss)
    for x in xs:
        draw.line(
            xy((x, top + 18, x, top + chart_height - 18)),
            fill=GUIDE,
            width=2 * ss,
        )
    draw.rectangle(
        xy((left, top, left + chart_width, top + chart_height)),
        outline=INK,
        width=2 * ss,
    )

    for item in series:
        label = str(item.get("label"))
        band = int(item.get("band"))
        values = [float(value) for value in item.get("values", [])]
        if label not in palette or band not in range(4) or len(values) != 3:
            raise ValueError("invalid series entry")
        points = [
            (int(round(x * ss)), int(round((band_centers[band] - amplitude * value) * ss)))
            for x, value in zip(xs, values, strict=True)
        ]
        draw.line(points, fill=palette[label], width=7 * ss, joint="curve")
        radius = 8 * ss
        for x, y in points:
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=palette[label])

    label_font = _font(14, ss)
    title_font = _font(17, ss)
    draw.text(xy((left, top - 29, 0, 0))[:2], "THREE-TIMEPOINT TRENDS", fill=INK, font=title_font)
    for index, x in enumerate(xs, start=1):
        text = str(index)
        draw.text(
            (int(round((x - 4) * ss)), int(round((top + chart_height + 6) * ss))),
            text,
            fill=INK,
            font=label_font,
        )
    draw.text(
        (int(round((left + chart_width - 40) * ss)), int(round((top + chart_height + 6) * ss))),
        "TIME",
        fill=INK,
        font=label_font,
    )

    legend_y = height - 25
    legend_xs = [90, 225, 360, 495]
    for label, x in zip(LABELS, legend_xs, strict=True):
        draw.ellipse(xy((x, legend_y - 5, x + 10, legend_y + 5)), fill=palette[label])
        draw.text(
            (int(round((x + 16) * ss)), int(round((legend_y - 9) * ss))),
            label,
            fill=INK,
            font=label_font,
        )

    resampling = getattr(Image, "Resampling", Image).LANCZOS
    return image.resize((width, height), resampling).convert("RGB")


def analytic_gold(scene: dict) -> str:
    """Derive the unique local-peak label from latent plotted values."""

    ranked: list[tuple[float, str]] = []
    for item in scene["series"]:
        values = [float(value) for value in item.get("values", [])]
        if len(values) != 3:
            raise ValueError("each series needs three values")
        score = min(values[1] - values[0], values[1] - values[2])
        ranked.append((score, str(item.get("label"))))
    ranked.sort(reverse=True)
    if len(ranked) != 4 or ranked[0][0] <= 0 or ranked[0][0] == ranked[1][0]:
        raise ValueError("scene has no unique strict local peak")
    return ranked[0][1]


def margin(scene: dict) -> float:
    """Winner-to-runner-up local-peak score gap in rendered pixels."""

    scores: list[float] = []
    for item in scene["series"]:
        values = [float(value) for value in item.get("values", [])]
        scores.append(min(values[1] - values[0], values[1] - values[2]))
    scores.sort(reverse=True)
    if len(scores) < 2:
        return 0.0
    return float(scene["amplitude"]) * (scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    """Flag ties, non-peaks, or a runner-up separation below 16 pixels."""

    try:
        scores = []
        for item in scene["series"]:
            values = [float(value) for value in item.get("values", [])]
            scores.append(min(values[1] - values[0], values[1] - values[2]))
        scores.sort(reverse=True)
        return len(scores) != 4 or scores[0] <= 0 or margin(scene) < 16.0
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Permuting disjoint series draw order is pixel-identical."""

    twin = copy.deepcopy(scene)
    twin["series"] = list(reversed(twin["series"]))
    return [("reverse_disjoint_series_order", twin)]
