"""Independent raster-only inverse arm for the middle-timepoint peak task."""

from __future__ import annotations

import numpy as np
from PIL import Image

LABELS = ("red", "blue", "green", "orange")
_PALETTE_CORES = (
    ("red", (202, 55, 62)),
    ("blue", (45, 105, 190)),
    ("green", (28, 143, 85)),
    ("orange", (230, 126, 34)),
    ("red", (176, 39, 55)),
    ("blue", (30, 88, 165)),
    ("green", (18, 121, 73)),
    ("orange", (205, 103, 21)),
)
_GUIDE = np.array((207, 214, 224), dtype=np.int32)


def _runs(columns: np.ndarray) -> list[np.ndarray]:
    if len(columns) == 0:
        return []
    cuts = np.flatnonzero(np.diff(columns) > 1) + 1
    return [chunk for chunk in np.split(columns, cuts) if len(chunk)]


def _recover_guides(arr: np.ndarray) -> tuple[list[float], tuple[int, int]]:
    delta = arr.astype(np.int32) - _GUIDE
    guide_mask = np.sum(delta * delta, axis=2) <= 18 * 18
    columns = np.flatnonzero(guide_mask.sum(axis=0) >= 120)
    groups = _runs(columns)
    if len(groups) != 3:
        raise ValueError(f"abstain: expected three visible time guides, found {len(groups)}")
    centers = [float(group.mean()) for group in groups]
    selected = np.zeros(guide_mask.shape[1], dtype=bool)
    for group in groups:
        selected[group] = True
    ys = np.flatnonzero((guide_mask[:, selected]).any(axis=1))
    if len(ys) < 180:
        raise ValueError("abstain: time guides are too damaged")
    return centers, (int(ys.min()), int(ys.max()))


def _color_masks(arr: np.ndarray) -> dict[str, np.ndarray]:
    pixels = arr.astype(np.int32)
    distances = np.stack(
        [np.sum((pixels - np.array(rgb, dtype=np.int32)) ** 2, axis=2) for _, rgb in _PALETTE_CORES],
        axis=2,
    )
    nearest = distances.argmin(axis=2)
    best = distances.min(axis=2)
    masks: dict[str, np.ndarray] = {}
    for label in LABELS:
        indices = [index for index, (name, _) in enumerate(_PALETTE_CORES) if name == label]
        masks[label] = np.isin(nearest, indices) & (best <= 58 * 58)
    return masks


def decision_from_image(image: Image.Image) -> str:
    """Recover guide positions, sample each colored trace there, and select its peak."""

    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.ndim != 3 or arr.shape[0] < 200 or arr.shape[1] < 300:
        raise ValueError("abstain: raster is too small")
    guides, (top, bottom) = _recover_guides(arr)
    masks = _color_masks(arr)

    scores: dict[str, float] = {}
    for label in LABELS:
        observed: list[float] = []
        mask = masks[label]
        for guide in guides:
            x0 = max(0, int(round(guide)) - 5)
            x1 = min(mask.shape[1], int(round(guide)) + 6)
            local = mask[top : bottom + 1, x0:x1]
            ys = np.nonzero(local)[0]
            if len(ys) < 25:
                raise ValueError(f"abstain: {label} trace is missing at a time guide")
            observed.append(float(np.median(ys) + top))
        scores[label] = min(observed[0] - observed[1], observed[2] - observed[1])

    ranked = sorted(((score, label) for label, score in scores.items()), reverse=True)
    if ranked[0][0] < 7.0:
        raise ValueError("abstain: no strict middle-timepoint peak is resolved")
    if ranked[0][0] - ranked[1][0] < 15.0:
        raise ValueError("abstain: winner-to-runner-up pixel margin is too small")
    return ranked[0][1]
