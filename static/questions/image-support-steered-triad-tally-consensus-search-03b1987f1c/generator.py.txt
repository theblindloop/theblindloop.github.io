"""Renderer for the triad tally consensus search world.

Three two-by-two blocks of dot tallies sit near the top-left corner, the
top-right corner and the bottom centre of an otherwise empty page. A tally is a
horizontal row of equal small squares, so a tally carries exactly one number:
how many squares it holds. Each block has the same four positions -- upper-left,
upper-right, lower-left, lower-right -- and exactly one position holds the same
number of squares in all three blocks. That position is the answer.

The three blocks are jointly necessary by construction. Each of the three
non-answer positions is planted so that it agrees in exactly one *pair* of
blocks and disagrees in the third, and the three non-answer positions are
assigned to the three distinct pairs. Drop any single block and the two blocks
that remain show two positions with equal counts, so the answer stops being
unique. The decisive support is therefore three mutually disconnected marks
spread to three far apart parts of the page, and no pair of them can answer.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "triad_tally_consensus_search-renderer-0.1.0"

CANVAS_W = 900
CANVAS_H = 640

DOT = 9                   # side of every square, in pixels
DOT_PITCH = 18            # centre-to-centre spacing inside one tally
CELL_PITCH_X = 161        # horizontal spacing between the two columns of a block
CELL_PITCH_Y = 54         # vertical spacing between the two rows of a block
MIN_COUNT = 2
MAX_COUNT = 8
JITTER = 8                # independent whole-block placement jitter

ANCHORS = ((30, 40), (574, 40), (302, 470))

WORDS = ("upper-left", "upper-right", "lower-left", "lower-right")
QUARANTINE_MARGIN = 2.0

PALETTES = (
    ((246, 244, 238), (38, 42, 50)),
    ((238, 242, 246), (34, 46, 62)),
    ((243, 240, 232), (46, 40, 34)),
    ((240, 240, 244), (30, 52, 48)),
)

_PAIRS = ((0, 1), (0, 2), (1, 2))


def _counts_by_slot(scene: dict) -> list:
    """counts[block_anchor][cell_slot], indexed by stored slot, not list order."""

    table = [[0] * 4 for _ in range(3)]
    for block in scene["blocks"]:
        for cell in block["cells"]:
            table[int(block["anchor"])][int(cell["slot"])] = int(cell["count"])
    return table


def _spreads(table: list) -> list:
    return [
        max(table[b][i] for b in range(3)) - min(table[b][i] for b in range(3))
        for i in range(4)
    ]


def _sample_counts(rng: random.Random, winner: int, rank_target: int):
    """A count table with one three-way consensus and three one-pair decoys."""

    for _ in range(8000):
        shared = rng.randint(MIN_COUNT, MAX_COUNT)
        others = [i for i in range(4) if i != winner]
        pairs = list(_PAIRS)
        rng.shuffle(pairs)
        floor = 1 if rng.random() < 0.18 else rng.choice((2, 3))
        gaps = [floor] + [rng.randint(floor, 5) for _ in range(2)]
        rng.shuffle(gaps)

        table = [[0] * 4 for _ in range(3)]
        for block in range(3):
            table[block][winner] = shared
        feasible = True
        for position, index in enumerate(others):
            near, also = pairs[position]
            far = ({0, 1, 2} - {near, also}).pop()
            gap = gaps[position]
            agreed = rng.randint(MIN_COUNT, MAX_COUNT)
            odd = agreed + rng.choice((-1, 1)) * gap
            if not MIN_COUNT <= odd <= MAX_COUNT:
                odd = 2 * agreed - odd
            if not MIN_COUNT <= odd <= MAX_COUNT:
                feasible = False
                break
            table[near][index] = agreed
            table[also][index] = agreed
            table[far][index] = odd
        if not feasible:
            continue

        spreads = _spreads(table)
        if spreads[winner] != 0 or min(spreads[i] for i in others) != floor:
            continue
        # No single block may rank the answer first or last on its own.
        if any(not min(row) < row[winner] < max(row) for row in table):
            continue
        totals = [sum(table[b][i] for b in range(3)) for i in range(4)]
        if len(set(totals)) != 4:
            continue
        if sorted(totals).index(totals[winner]) != rank_target:
            continue
        return table
    raise RuntimeError("count sampling failed to satisfy the declared constraints")


def sample_scene(seed: int) -> dict:
    """Sample one scene. Answer slot and total-ink rank are both counterbalanced."""

    index = int(seed) % 1000
    winner = index % 4
    rank_target = ((index // 4) + (index % 4)) % 4
    rng = random.Random(int(seed) * 7919 + 13)
    table = _sample_counts(rng, winner, rank_target)

    blocks = []
    for anchor in range(3):
        cells = [{"slot": slot, "count": int(table[anchor][slot])} for slot in range(4)]
        rng.shuffle(cells)
        blocks.append(
            {
                "anchor": anchor,
                "shift_x": rng.randint(-JITTER, JITTER),
                "shift_y": rng.randint(-JITTER, JITTER),
                "cells": cells,
            }
        )
    rng.shuffle(blocks)
    return {
        "width": CANVAS_W,
        "height": CANVAS_H,
        "palette": rng.randrange(len(PALETTES)),
        "blocks": blocks,
    }


def cell_origin(block: dict, slot: int) -> tuple:
    """Top-left pixel of the first square of one tally."""

    base_x, base_y = ANCHORS[int(block["anchor"])]
    return (
        base_x + int(block["shift_x"]) + (int(slot) % 2) * CELL_PITCH_X,
        base_y + int(block["shift_y"]) + (int(slot) // 2) * CELL_PITCH_Y,
    )


def cell_boxes(block: dict, cell: dict) -> list:
    origin_x, origin_y = cell_origin(block, int(cell["slot"]))
    return [
        (origin_x + step * DOT_PITCH, origin_y,
         origin_x + step * DOT_PITCH + DOT - 1, origin_y + DOT - 1)
        for step in range(int(cell["count"]))
    ]


def cell_bbox(block: dict, cell: dict) -> tuple:
    boxes = cell_boxes(block, cell)
    return (boxes[0][0], boxes[0][1], boxes[-1][2], boxes[-1][3])


def block_bbox(block: dict) -> tuple:
    boxes = [cell_bbox(block, cell) for cell in block["cells"]]
    return (
        min(box[0] for box in boxes),
        min(box[1] for box in boxes),
        max(box[2] for box in boxes),
        max(box[3] for box in boxes),
    )


def dot_boxes(scene: dict) -> list:
    out = []
    for block in scene["blocks"]:
        for cell in block["cells"]:
            out.extend(cell_boxes(block, cell))
    return out


def render(scene: dict) -> Image.Image:
    """Rasterise the scene. Squares never overlap, so draw order is irrelevant."""

    background, ink = PALETTES[int(scene["palette"]) % len(PALETTES)]
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), background)
    draw = ImageDraw.Draw(image)
    for box in dot_boxes(scene):
        draw.rectangle(box, fill=ink)
    return image


def analytic_gold(scene: dict) -> str:
    """The unique block position whose count is shared by all three blocks."""

    spreads = _spreads(_counts_by_slot(scene))
    winners = [index for index, value in enumerate(spreads) if value == 0]
    if len(winners) != 1:
        raise ValueError("scene has no unique three-way consensus position")
    return WORDS[winners[0]]


def margin(scene: dict) -> float:
    """Smallest count spread among the non-answer positions, in squares."""

    spreads = _spreads(_counts_by_slot(scene))
    return float(min(value for value in spreads if value > 0))


def is_quarantined(scene: dict) -> bool:
    """A one-square runner-up spread is a boundary case and is excluded."""

    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list:
    """Pixel-identical, answer-preserving rewrites of the latent scene."""

    out = []
    if len(scene["blocks"]) > 1:
        rotated = dict(scene)
        rotated["blocks"] = scene["blocks"][1:] + scene["blocks"][:1]
        out.append(("block_record_order_rotated", rotated))
        flipped = dict(scene)
        flipped["blocks"] = list(reversed(scene["blocks"]))
        out.append(("block_record_order_reversed", flipped))
    permuted = dict(scene)
    permuted["blocks"] = [
        {**block, "cells": block["cells"][1:] + block["cells"][:1]}
        for block in scene["blocks"]
    ]
    out.append(("cell_record_order_rotated", permuted))
    return out
