"""Renderer and analytic arm for the three-cell component sieve."""

from __future__ import annotations

import random
from typing import Iterable

import numpy as np
from PIL import Image, ImageDraw

SYMBOLS = ("flag", "hook", "bolt", "cup")
GRID = 7
MICRO = 5
FINE = GRID * MICRO
CELL = 6
PANEL_PX = FINE * CELL
CANVAS = 700

_ROWS = {
    "flag": (
        ".####..", ".##....", ".##....", ".#.....", ".#.....", ".#.....", ".#.....",
    ),
    "hook": (
        ".###...", ".#.....", ".#.....", ".#.....", ".#..#..", ".####..", ".......",
    ),
    "bolt": (
        "...##..", "..##...", "..#....", ".###...", "..#....", ".##....", ".#.....",
    ),
    "cup": (
        ".#...#.", ".#...#.", ".#...#.", ".#####.", "...#...", ".......", ".......",
    ),
}
TEMPLATES = {
    name: np.array([[ch == "#" for ch in row] for row in rows], dtype=bool)
    for name, rows in _ROWS.items()
}
assert all(int(mask.sum()) == 12 for mask in TEMPLATES.values())

_MOTIFS = {
    1: ((2, 2),),
    2: ((2, 2), (2, 3)),
    3: ((2, 1), (2, 2), (2, 3)),
    4: ((1, 1), (1, 2), (2, 1), (2, 2)),
}
_PALETTES = (
    {"background": (250, 249, 246), "grid": (210, 214, 216), "ink": (24, 27, 31), "frame": (0, 157, 181)},
    {"background": (247, 250, 248), "grid": (205, 211, 207), "ink": (38, 25, 45), "frame": (0, 142, 174)},
    {"background": (251, 248, 250), "grid": (214, 207, 213), "ink": (18, 35, 38), "frame": (0, 168, 164)},
)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x5A17C0DE)
    target = SYMBOLS[int(seed) % len(SYMBOLS)]
    other_sizes = [1, 2, 4]
    rng.shuffle(other_sizes)
    symbol_sizes = {target: 3}
    for symbol, size in zip((s for s in SYMBOLS if s != target), other_sizes):
        symbol_sizes[symbol] = size
    panel_symbols = list(SYMBOLS)
    rng.shuffle(panel_symbols)
    return {
        "panel_symbols": panel_symbols,
        "symbol_sizes": symbol_sizes,
        "frame_dx": rng.randint(-10, 10),
        "frame_dy": rng.randint(-10, 10),
        "palette_variant": rng.randrange(len(_PALETTES)),
    }


def _fine_occupancy(symbol: str, component_size: int) -> np.ndarray:
    result = np.zeros((FINE, FINE), dtype=bool)
    for gy, gx in np.argwhere(TEMPLATES[symbol]):
        for my, mx in _MOTIFS[int(component_size)]:
            result[gy * MICRO + my, gx * MICRO + mx] = True
    return result


def _panel_origins(dx: int, dy: int) -> Iterable[tuple[int, int]]:
    for y in (92 + dy, 388 + dy):
        for x in (92 + dx, 388 + dx):
            yield x, y


def render(scene: dict) -> Image.Image:
    panel_symbols = list(scene["panel_symbols"])
    symbol_sizes = dict(scene["symbol_sizes"])
    dx = int(scene["frame_dx"])
    dy = int(scene["frame_dy"])
    palette = _PALETTES[int(scene["palette_variant"])]
    image = Image.new("RGB", (CANVAS, CANVAS), palette["background"])
    draw = ImageDraw.Draw(image)
    for (x0, y0), symbol in zip(_panel_origins(dx, dy), panel_symbols):
        draw.rectangle(
            (x0 - 4, y0 - 4, x0 + PANEL_PX + 3, y0 + PANEL_PX + 3),
            outline=palette["frame"], width=4,
        )
        for i in range(FINE + 1):
            p = i * CELL
            draw.line((x0 + p, y0, x0 + p, y0 + PANEL_PX), fill=palette["grid"], width=1)
            draw.line((x0, y0 + p, x0 + PANEL_PX, y0 + p), fill=palette["grid"], width=1)
        occupied = _fine_occupancy(symbol, int(symbol_sizes[symbol]))
        for row, col in np.argwhere(occupied):
            left = x0 + int(col) * CELL + 1
            top = y0 + int(row) * CELL + 1
            draw.rectangle((left, top, left + CELL - 2, top + CELL - 2), fill=palette["ink"])
    return image


def _target(scene: dict) -> str:
    matches = [name for name, size in scene["symbol_sizes"].items() if int(size) == 3]
    if len(matches) != 1:
        raise ValueError("scene must map exactly one symbol to component size three")
    return matches[0]


def analytic_gold(scene: dict) -> str:
    return _target(scene)


def margin(scene: dict) -> float:
    target = TEMPLATES[_target(scene)]
    distances = sorted(int(np.count_nonzero(target ^ TEMPLATES[name])) for name in SYMBOLS if name != _target(scene))
    return float(distances[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
