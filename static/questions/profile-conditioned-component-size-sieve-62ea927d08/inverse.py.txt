"""Independent PNG-only inverse for the three-cell component sieve."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

GRID = 7
MICRO = 5
FINE = 35
CELL = 6
PANEL_PX = 210

_ROWS = {
    "flag": (".####..", ".##....", ".##....", ".#.....", ".#.....", ".#.....", ".#....."),
    "hook": (".###...", ".#.....", ".#.....", ".#.....", ".#..#..", ".####..", "......."),
    "bolt": ("...##..", "..##...", "..#....", ".###...", "..#....", ".##....", ".#....."),
    "cup": (".#...#.", ".#...#.", ".#...#.", ".#####.", "...#...", ".......", "......."),
}
_TEMPLATES = {
    name: np.array([[ch == "#" for ch in row] for row in rows], dtype=bool)
    for name, rows in _ROWS.items()
}


def _pixel_components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[list[tuple[int, int]]] = []
    height, width = mask.shape
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = deque([(y, x)])
        group: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            group.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        groups.append(group)
    return groups


def _frame_boxes(arr: np.ndarray) -> list[tuple[int, int, int, int]]:
    cyan = (arr[:, :, 0] < 70) & (arr[:, :, 1] > 110) & (arr[:, :, 2] > 120)
    boxes = []
    for group in _pixel_components(cyan):
        if len(group) < 500:
            continue
        ys = [p[0] for p in group]
        xs = [p[1] for p in group]
        box = (min(xs), min(ys), max(xs), max(ys))
        if 214 <= box[2] - box[0] + 1 <= 222 and 214 <= box[3] - box[1] + 1 <= 222:
            boxes.append(box)
    boxes.sort(key=lambda box: (box[1], box[0]))
    if len(boxes) != 4:
        raise ValueError(f"expected four visible cyan frames, found {len(boxes)}")
    return boxes


def _logical_black(arr: np.ndarray, box: tuple[int, int, int, int]) -> np.ndarray:
    x0, y0 = box[0] + 4, box[1] + 4
    logical = np.zeros((FINE, FINE), dtype=bool)
    for row in range(FINE):
        for col in range(FINE):
            cx = x0 + col * CELL + CELL // 2
            cy = y0 + row * CELL + CELL // 2
            patch = arr[cy - 1 : cy + 2, cx - 1 : cx + 2]
            logical[row, col] = float(patch.mean()) < 105.0
    return logical


def _sieved_macro(logical: np.ndarray) -> np.ndarray:
    kept = np.zeros_like(logical, dtype=bool)
    for group in _pixel_components(logical):
        if len(group) == 3:
            for y, x in group:
                kept[y, x] = True
    macro = np.zeros((GRID, GRID), dtype=bool)
    for gy in range(GRID):
        for gx in range(GRID):
            macro[gy, gx] = bool(kept[gy * MICRO : (gy + 1) * MICRO, gx * MICRO : (gx + 1) * MICRO].any())
    return macro


def decision_with_margin(image: Image.Image) -> tuple[str, float]:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    panels = [_sieved_macro(_logical_black(arr, box)) for box in _frame_boxes(arr)]
    ranked: list[tuple[int, str, int]] = []
    for panel_index, panel in enumerate(panels):
        for name, template in _TEMPLATES.items():
            ranked.append((int(np.count_nonzero(panel ^ template)), name, panel_index))
    ranked.sort()
    best = ranked[0]
    runner_distance = next(distance for distance, name, panel in ranked[1:] if name != best[1] or panel != best[2])
    if best[0] > 0 or runner_distance - best[0] < 4:
        raise ValueError(f"ambiguous transformed glyph: best={best}, runner_distance={runner_distance}")
    return best[1], float(runner_distance - best[0])


def decision_from_image(image: Image.Image) -> str:
    return decision_with_margin(image)[0]
