"""Independent pixel-only inverse arm for Trace Band Span.

Only the final RGB raster is accepted. The oracle segments the cool-colored
connected trace, recovers the two warm-colored dashed guide rows, validates
that the trace remains one unbroken x-progressing component, and sums columns
belonging to horizontal runs in each visible band. It imports no renderer,
prompt, generator, verifier, scene, metadata, or filename state.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_BANDS = ("top", "middle", "bottom")


def _masks(image: Image.Image) -> tuple[np.ndarray, np.ndarray]:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[:, :, 0], array[:, :, 1], array[:, :, 2]
    chroma = array.max(axis=2) - array.min(axis=2)
    route = (
        (chroma >= 58)
        & ((green - red) >= 48)
        & ((blue - red) >= 48)
        & (green >= 95)
    )
    guides = (
        (chroma >= 70)
        & ((red - green) >= 55)
        & ((red - blue) >= 55)
        & (red >= 155)
    )
    return route, guides


def _component_size(mask: np.ndarray, start: tuple[int, int]) -> int:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    stack = [start]
    seen[start] = True
    count = 0
    while stack:
        y, x = stack.pop()
        count += 1
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                seen[ny, nx] = True
                stack.append((ny, nx))
    return count


def _guide_rows(mask: np.ndarray, image_width: int) -> tuple[float, float]:
    row_counts = mask.sum(axis=1)
    candidate_rows = np.flatnonzero(row_counts >= max(120, int(image_width * 0.17)))
    groups: list[list[int]] = []
    for value in candidate_rows.tolist():
        if not groups or value > groups[-1][-1] + 1:
            groups.append([value])
        else:
            groups[-1].append(value)
    if len(groups) != 2 or any(len(group) > 5 for group in groups):
        raise ValueError("two dashed guide rows could not be recovered")
    centers = tuple(float(np.median(group)) for group in groups)
    if not image_width * 0.08 <= centers[1] - centers[0] <= image_width * 0.20:
        raise ValueError("guide separation is implausible")
    return centers


def _measure(image: Image.Image) -> tuple[list[int], tuple[float, float]]:
    route, guide_mask = _masks(image)
    ys, xs = np.nonzero(route)
    if len(xs) < 1800:
        raise ValueError("teal trace is missing or materially erased")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    route_width = x1 - x0 + 1
    route_height = y1 - y0 + 1
    if not image.width * 0.56 <= route_width <= image.width * 0.68:
        raise ValueError("trace horizontal extent is implausible")
    if not image.height * 0.22 <= route_height <= image.height * 0.28:
        raise ValueError("trace vertical extent is implausible")

    first = tuple(int(value) for value in np.argwhere(route)[0])
    if _component_size(route, first) != int(route.sum()):
        raise ValueError("teal trace is severed")

    guides = _guide_rows(guide_mask, image.width)
    totals = [0, 0, 0]
    missing = 0
    horizontal_columns = 0
    for x in range(x0, x1 + 1):
        column_y = np.flatnonzero(route[:, x])
        if not len(column_y):
            missing += 1
            continue
        # Columns through a vertical connector have a tall colored span and
        # correctly contribute zero horizontal length.
        if int(column_y[-1] - column_y[0]) > 16:
            continue
        center = float(np.median(column_y))
        if min(abs(center - guides[0]), abs(center - guides[1])) <= 5.0:
            continue
        band = 0 if center < guides[0] else (1 if center < guides[1] else 2)
        totals[band] += 1
        horizontal_columns += 1
    if missing > 3:
        raise ValueError("trace has missing interior columns")
    if horizontal_columns < image.width * 0.50 or min(totals) < 38:
        raise ValueError("horizontal trace runs are incomplete")
    return totals, guides


def decision_from_image(image: Image.Image) -> str:
    totals, _ = _measure(image)
    order = sorted(range(3), key=lambda index: totals[index], reverse=True)
    if totals[order[0]] - totals[order[1]] <= 2:
        raise ValueError("pixel-measured band spans are tied")
    return _BANDS[order[0]]
