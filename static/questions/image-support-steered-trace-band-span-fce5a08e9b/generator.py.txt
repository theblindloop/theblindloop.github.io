"""Deterministic renderer and analytic arm for Trace Band Span."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 800
HEIGHT = 600
BANDS = ("top", "middle", "bottom")
QUARANTINE_PX = 25.0
BACKGROUND_PALETTE = ((248, 247, 243), (246, 248, 247), (249, 247, 244))
INK = (39, 45, 52)
FRAME = (190, 193, 191)
ROUTE_PALETTE = (
    (14, 132, 154),
    (35, 116, 174),
    (24, 145, 115),
)
GUIDE_PALETTE = (
    (202, 100, 55),
    (188, 83, 66),
    (201, 117, 38),
)


def _font(size: int, *, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _split_two(total: int, offset: int) -> tuple[int, int]:
    first = total // 2 + offset
    second = total - first
    if min(first, second) < 42:
        raise ValueError("run split is too short")
    return first, second


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample cue-balanced seed triplets with all three answer identities.

    Seeds ``3k``, ``3k+1``, and ``3k+2`` use the same chart furniture,
    palette, band-visit order, total route span, connector geometry, and run
    length multiset.  Only the assignment of the three aggregate spans to the
    visible bands rotates, so the winner is TOP, MIDDLE, then BOTTOM.
    """

    seed = int(seed)
    group = seed // 3
    answer_index = seed % 3
    rng = random.Random(0x7A6E_2026 + group * 1_000_003)

    # Every seventh triplet lies near the equality boundary but remains far
    # enough from a pixel tie for the image-only oracle to agree reliably.
    rank_totals = (180, 158, 152) if group % 7 == 0 else (220, 155, 115)
    rank_splits = [
        _split_two(total, rng.randint(-12, 12)) for total in rank_totals
    ]

    # The same six-visit path is used across a triplet. Reflections/reversal
    # diversify position without changing the number or lengths of connectors.
    sequences = (
        (0, 1, 2, 0, 1, 2),
        (2, 1, 0, 2, 1, 0),
        (2, 1, 0, 2, 1, 0),
        (0, 1, 2, 0, 1, 2),
    )
    sequence = sequences[group % len(sequences)]

    # Rank 0 (the largest aggregate) is assigned to answer_index. The two
    # runner-up ranks rotate with it, preserving the global width multiset.
    rank_for_band = {
        answer_index: 0,
        (answer_index + 1) % 3: 1,
        (answer_index + 2) % 3: 2,
    }
    pending = {
        band: list(rank_splits[rank_for_band[band]]) for band in range(3)
    }

    chart_left = 130 + (group % 5 - 2) * 4
    chart_top = 160 + (group % 3 - 1) * 5
    chart_width = 540
    chart_height = 210
    guides = [chart_top + 70, chart_top + 140]
    band_y = [chart_top + 35, chart_top + 105, chart_top + 175]

    x = chart_left + 25
    vertices: list[list[int]] = []
    for band in sequence:
        y = band_y[band]
        if not vertices:
            vertices.append([x, y])
        elif vertices[-1][1] != y:
            vertices.append([x, y])
        width = pending[band].pop(0)
        x += width
        vertices.append([x, y])
    if x != chart_left + 515:
        raise AssertionError("route span must remain exactly 490 pixels")
    if any(pending[band] for band in pending):
        raise AssertionError("every band must be visited exactly twice")

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "chart_left": chart_left,
        "chart_top": chart_top,
        "chart_width": chart_width,
        "chart_height": chart_height,
        "guides": guides,
        "vertices": vertices,
        "route_rgb": list(ROUTE_PALETTE[group % len(ROUTE_PALETTE)]),
        "guide_rgb": list(GUIDE_PALETTE[group % len(GUIDE_PALETTE)]),
        "background_rgb": list(BACKGROUND_PALETTE[group % len(BACKGROUND_PALETTE)]),
    }


def _horizontal_band_spans(scene: dict[str, Any]) -> list[int]:
    guides = sorted(int(value) for value in scene["guides"])
    if len(guides) != 2 or guides[0] >= guides[1]:
        raise ValueError("scene must contain two ordered guides")
    vertices = [[int(axis) for axis in point] for point in scene["vertices"]]
    if len(vertices) < 4:
        raise ValueError("trace has too few vertices")
    totals = [0, 0, 0]
    for first, second in zip(vertices, vertices[1:], strict=False):
        x0, y0 = first
        x1, y1 = second
        if y0 == y1:
            if x0 == x1:
                continue
            center_y = y0
            band = 0 if center_y < guides[0] else (1 if center_y < guides[1] else 2)
            totals[band] += abs(x1 - x0)
        elif x0 != x1:
            raise ValueError("trace segments must be horizontal or vertical")
    if min(totals) <= 0:
        raise ValueError("each band must contain horizontal trace support")
    return totals


def analytic_gold(scene: dict[str, Any]) -> str:
    totals = _horizontal_band_spans(scene)
    order = sorted(range(3), key=lambda index: totals[index], reverse=True)
    if totals[order[0]] == totals[order[1]]:
        raise ValueError("largest horizontal band span is tied")
    return BANDS[order[0]]


def margin(scene: dict[str, Any]) -> float:
    totals = sorted(_horizontal_band_spans(scene), reverse=True)
    return float(totals[0] - totals[1])


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) <= QUARANTINE_PX


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # The trace is undirected. Reversing its stored traversal order changes the
    # latent representation but preserves every rendered pixel and all spans.
    twin = dict(scene)
    twin["vertices"] = list(reversed([list(point) for point in scene["vertices"]]))
    return [("reverse_vertex_order", twin)]


def render(scene: dict[str, Any]) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    left = int(scene["chart_left"])
    top = int(scene["chart_top"])
    chart_width = int(scene["chart_width"])
    chart_height = int(scene["chart_height"])
    guides = [int(value) for value in scene["guides"]]
    vertices = [tuple(int(axis) for axis in point) for point in scene["vertices"]]
    route = tuple(int(value) for value in scene["route_rgb"])
    guide = tuple(int(value) for value in scene["guide_rgb"])
    background = tuple(int(value) for value in scene["background_rgb"])
    if (width, height) != (WIDTH, HEIGHT):
        raise ValueError("unexpected canvas size")
    if len(guides) != 2 or len(vertices) < 4:
        raise ValueError("malformed chart geometry")
    if not all(0 <= channel <= 255 for color in (route, guide, background) for channel in color):
        raise ValueError("RGB channel outside range")
    right = left + chart_width
    bottom = top + chart_height
    if left < 90 or right > width - 90 or top < 100 or bottom > height - 90:
        raise ValueError("chart does not fit the canvas")
    if not all(left <= x <= right and top <= y <= bottom for x, y in vertices):
        raise ValueError("trace leaves chart frame")

    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)
    title_font = _font(24, bold=True)
    label_font = _font(17, bold=True)
    small_font = _font(14)
    draw.text((left, 31), "HORIZONTAL SPAN BY BAND", fill=INK, font=title_font)
    draw.text((left, 67), "Add teal horizontal runs; vertical connectors count as zero.", fill=INK, font=small_font)

    draw.rectangle((left, top, right, bottom), outline=FRAME, width=2)
    dash = 18
    gap = 12
    for y in guides:
        x = left + 8
        while x < right - 8:
            draw.line((x, y, min(x + dash, right - 8), y), fill=guide, width=3)
            x += dash + gap

    band_centers = (top + 35, top + 105, top + 175)
    for name, y in zip(("TOP", "MIDDLE", "BOTTOM"), band_centers, strict=True):
        draw.text((left - 18, y), name, fill=INK, font=label_font, anchor="rm")

    draw.line(vertices, fill=route, width=7, joint="curve")
    draw.text((left, bottom + 25), "dashed guides define the band boundaries", fill=INK, font=small_font)
    return image
