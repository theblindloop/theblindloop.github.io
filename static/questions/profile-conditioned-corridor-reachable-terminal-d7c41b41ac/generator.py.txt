"""Deterministic raster world and analytic arm for corridor reachability."""

from __future__ import annotations

import copy
import random
from collections import deque

from PIL import Image, ImageDraw


COLORS = ("red", "blue", "orange")
PALETTE = {
    "background": (255, 255, 255),
    "wall": (27, 30, 35),
    "outline": (125, 130, 136),
    "green": (35, 170, 92),
    "red": (220, 55, 65),
    "blue": (47, 105, 210),
    "orange": (240, 145, 35),
}
ROWS = 7
COLS = 9


def _neighbors(cell: tuple[int, int]) -> list[tuple[int, int]]:
    row, col = cell
    return [
        (row - 1, col),
        (row + 1, col),
        (row, col - 1),
        (row, col + 1),
    ]


def _inside(cell: tuple[int, int], rows: int, cols: int) -> bool:
    return 0 <= cell[0] < rows and 0 <= cell[1] < cols


def _normal_edge(
    left: tuple[int, int], right: tuple[int, int]
) -> tuple[tuple[int, int], tuple[int, int]]:
    return tuple(sorted((left, right)))  # type: ignore[return-value]


def _grow_equal_regions(seed: int) -> tuple[dict[tuple[int, int], int], list]:
    """Grow three equal connected regions and retain each growth-tree edge."""

    target = ROWS * COLS // 3
    base_seeds = [(1, 1), (1, COLS - 2), (ROWS - 2, COLS // 2)]
    for attempt in range(256):
        rng = random.Random(seed * 1009 + attempt * 9176 + 43)
        seeds = list(base_seeds)
        if (seed // 9) % 2:
            seeds = [(row, COLS - 1 - col) for row, col in seeds]
        if (seed // 18) % 2:
            seeds = [(ROWS - 1 - row, col) for row, col in seeds]
        assignment = {cell: index for index, cell in enumerate(seeds)}
        sizes = [1, 1, 1]
        tree_edges: list[tuple[tuple[int, int], tuple[int, int]]] = []
        while len(assignment) < ROWS * COLS:
            frontier: dict[int, list[tuple[tuple[int, int], tuple[int, int]]]] = {
                0: [], 1: [], 2: []
            }
            for parent, component in assignment.items():
                if sizes[component] >= target:
                    continue
                for cell in _neighbors(parent):
                    if _inside(cell, ROWS, COLS) and cell not in assignment:
                        frontier[component].append((cell, parent))
            eligible = [
                component
                for component in range(3)
                if sizes[component] < target and frontier[component]
            ]
            if not eligible:
                break
            smallest = min(sizes[component] for component in eligible)
            balanced = [component for component in eligible if sizes[component] == smallest]
            component = rng.choice(balanced)
            cell = rng.choice(sorted({pair[0] for pair in frontier[component]}))
            parents = [parent for candidate, parent in frontier[component] if candidate == cell]
            parent = rng.choice(sorted(set(parents)))
            assignment[cell] = component
            sizes[component] += 1
            tree_edges.append(_normal_edge(cell, parent))
        if len(assignment) == ROWS * COLS and sizes == [target, target, target]:
            return assignment, tree_edges
    raise RuntimeError("could not construct three balanced connected maze regions")


def _adjacency(
    rows: int, cols: int, open_edges: list[list[list[int]]]
) -> dict[tuple[int, int], set[tuple[int, int]]]:
    adjacency = {
        (row, col): set() for row in range(rows) for col in range(cols)
    }
    seen = set()
    for raw_left, raw_right in open_edges:
        left, right = tuple(raw_left), tuple(raw_right)
        edge = _normal_edge(left, right)
        if (
            not _inside(left, rows, cols)
            or not _inside(right, rows, cols)
            or abs(left[0] - right[0]) + abs(left[1] - right[1]) != 1
            or edge in seen
        ):
            raise ValueError(f"invalid open doorway {raw_left!r}-{raw_right!r}")
        seen.add(edge)
        adjacency[left].add(right)
        adjacency[right].add(left)
    return adjacency


def _distances(
    adjacency: dict[tuple[int, int], set[tuple[int, int]]], start: tuple[int, int]
) -> dict[tuple[int, int], int]:
    distance = {start: 0}
    queue = deque([start])
    while queue:
        cell = queue.popleft()
        for neighbor in adjacency[cell]:
            if neighbor not in distance:
                distance[neighbor] = distance[cell] + 1
                queue.append(neighbor)
    return distance


def sample_scene(seed: int) -> dict:
    """Sample a balanced three-component wall maze.

    Component area and doorway count are exactly matched. Answer color cycles
    every seed, while START's component cycles independently every three seeds.
    """

    seed = int(seed)
    rng = random.Random(seed)
    assignment, tree_edges = _grow_equal_regions(seed)
    edge_set = set(tree_edges)

    # Add the same number of loop-producing doorways to every component.
    for component in range(3):
        extras = []
        for row in range(ROWS):
            for col in range(COLS):
                cell = (row, col)
                if assignment[cell] != component:
                    continue
                for neighbor in ((row + 1, col), (row, col + 1)):
                    edge = _normal_edge(cell, neighbor)
                    if (
                        _inside(neighbor, ROWS, COLS)
                        and assignment[neighbor] == component
                        and edge not in edge_set
                    ):
                        extras.append(edge)
        rng.shuffle(extras)
        if len(extras) < 3:
            raise RuntimeError("balanced maze region lacks optional loop doorways")
        edge_set.update(extras[:3])

    open_edges = [
        [[left[0], left[1]], [right[0], right[1]]]
        for left, right in sorted(edge_set)
    ]
    rng.shuffle(open_edges)
    adjacency = _adjacency(ROWS, COLS, open_edges)

    endpoints: dict[int, tuple[tuple[int, int], tuple[int, int]]] = {}
    for component in range(3):
        cells = sorted(cell for cell, value in assignment.items() if value == component)
        first = max(_distances(adjacency, cells[0]), key=lambda cell: (_distances(adjacency, cells[0])[cell], cell))
        from_first = _distances(adjacency, first)
        second = max(from_first, key=lambda cell: (from_first[cell], cell))
        endpoints[component] = (first, second)

    start_component = (seed // 3) % 3
    answer_color = COLORS[seed % 3]
    other_colors = [color for color in COLORS if color != answer_color]
    rng.shuffle(other_colors)
    markers = [
        {"role": "start", "color": "green", "cell": list(endpoints[start_component][0])},
        {
            "role": "terminal",
            "color": answer_color,
            "cell": list(endpoints[start_component][1]),
        },
    ]
    for component, color in zip(
        [value for value in range(3) if value != start_component], other_colors
    ):
        marker_cell = endpoints[component][(seed + component) % 2]
        markers.append({"role": "terminal", "color": color, "cell": list(marker_cell)})
    rng.shuffle(markers)

    return {
        "canvas": {"width": 780, "height": 640, "background": "white"},
        "grid": {
            "rows": ROWS,
            "cols": COLS,
            "x0": 75 + rng.choice((-6, -3, 0, 3, 6)),
            "y0": 75 + rng.choice((-6, -3, 0, 3, 6)),
            "cell": 70,
            "wall_width": 8,
            "door_width": 30,
            "marker_radius": 14,
        },
        "open_edges": open_edges,
        "markers": markers,
    }


def _wall_box(center: int, low: int, high: int, width: int) -> tuple[int, int, int, int]:
    half = width // 2
    return (center - half, low - half, center + width - half - 1, high + half - 1)


def render(scene: dict) -> Image.Image:
    canvas = scene["canvas"]
    grid = scene["grid"]
    open_edges = scene["open_edges"]
    markers = scene["markers"]
    width, height = int(canvas["width"]), int(canvas["height"])
    image = Image.new("RGB", (width, height), PALETTE["background"])
    draw = ImageDraw.Draw(image)
    rows, cols = int(grid["rows"]), int(grid["cols"])
    x0, y0, cell = int(grid["x0"]), int(grid["y0"]), int(grid["cell"])
    wall_width, door_width = int(grid["wall_width"]), int(grid["door_width"])

    for col in range(cols + 1):
        x = x0 + col * cell
        draw.rectangle(
            _wall_box(x, y0, y0 + rows * cell, wall_width), fill=PALETTE["wall"]
        )
    for row in range(rows + 1):
        y = y0 + row * cell
        box = _wall_box(y, x0, x0 + cols * cell, wall_width)
        draw.rectangle((box[1], box[0], box[3], box[2]), fill=PALETTE["wall"])

    door_half = door_width // 2
    wall_half = wall_width // 2
    for raw_left, raw_right in open_edges:
        left, right = tuple(raw_left), tuple(raw_right)
        if left[0] == right[0]:
            boundary_x = x0 + max(left[1], right[1]) * cell
            center_y = y0 + left[0] * cell + cell // 2
            draw.rectangle(
                (
                    boundary_x - wall_half,
                    center_y - door_half,
                    boundary_x + wall_width - wall_half - 1,
                    center_y + door_half - 1,
                ),
                fill=PALETTE["background"],
            )
        else:
            boundary_y = y0 + max(left[0], right[0]) * cell
            center_x = x0 + left[1] * cell + cell // 2
            draw.rectangle(
                (
                    center_x - door_half,
                    boundary_y - wall_half,
                    center_x + door_half - 1,
                    boundary_y + wall_width - wall_half - 1,
                ),
                fill=PALETTE["background"],
            )

    radius = int(grid["marker_radius"])
    for marker in markers:
        row, col = marker["cell"]
        x = x0 + int(col) * cell + cell // 2
        y = y0 + int(row) * cell + cell // 2
        draw.ellipse(
            (x - radius, y - radius, x + radius, y + radius),
            fill=PALETTE[marker["color"]],
            outline=PALETTE["outline"],
            width=2,
        )
    return image


def analytic_gold(scene: dict) -> str:
    grid = scene["grid"]
    open_edges = scene["open_edges"]
    markers = scene["markers"]
    rows, cols = int(grid["rows"]), int(grid["cols"])
    adjacency = _adjacency(rows, cols, open_edges)
    starts = [tuple(marker["cell"]) for marker in markers if marker["role"] == "start" and marker["color"] == "green"]
    terminals = [marker for marker in markers if marker["role"] == "terminal" and marker["color"] in COLORS]
    if len(starts) != 1 or sorted(marker["color"] for marker in terminals) != sorted(COLORS):
        raise ValueError("scene must contain one green START and one terminal of each answer color")
    reachable = _distances(adjacency, starts[0])
    winners = [marker["color"] for marker in terminals if tuple(marker["cell"]) in reachable]
    if len(winners) != 1:
        raise ValueError(f"expected exactly one reachable colored terminal, found {winners}")
    return winners[0]


def margin(scene: dict) -> float:
    try:
        analytic_gold(scene)
    except ValueError:
        return 0.0
    return float(scene["grid"]["wall_width"])


def is_quarantined(scene: dict) -> bool:
    grid = scene["grid"]
    if int(grid["wall_width"]) < 6 or int(grid["door_width"]) <= 2 * int(grid["wall_width"]):
        return True
    try:
        analytic_gold(scene)
    except ValueError:
        return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["open_edges"] = list(reversed(twin["open_edges"]))
    twin["markers"] = list(reversed(twin["markers"]))
    return [("reverse_doorway_and_marker_storage", twin)]
