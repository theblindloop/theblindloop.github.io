"""Independent PNG-only inverse for free-space corridor connectivity.

Only final RGB pixels enter this module. It segments the four marker colors,
segments dark wall ink, and performs a four-neighbor flood fill from green.
No grid dimensions, wall coordinates, latent scene, or renderer state are used.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


REFERENCE = {
    "green": np.array((35, 170, 92), dtype=np.int32),
    "red": np.array((220, 55, 65), dtype=np.int32),
    "blue": np.array((47, 105, 210), dtype=np.int32),
    "orange": np.array((240, 145, 35), dtype=np.int32),
}
CANDIDATES = ("red", "blue", "orange")


def _components(mask: np.ndarray) -> list[tuple[int, int, int]]:
    work = mask.copy()
    height, width = work.shape
    found: list[tuple[int, int, int]] = []
    for start_y, start_x in zip(*np.nonzero(work)):
        if not work[start_y, start_x]:
            continue
        queue = [(int(start_y), int(start_x))]
        work[start_y, start_x] = False
        total_x = total_y = area = 0
        while queue:
            y, x = queue.pop()
            total_x += x
            total_y += y
            area += 1
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        if area >= 250:
            found.append((total_y // area, total_x // area, area))
    return found


def _marker_masks(pixels: np.ndarray) -> dict[str, np.ndarray]:
    rgb = pixels.astype(np.int32)
    masks = {}
    for label, reference in REFERENCE.items():
        distance2 = np.sum((rgb - reference) ** 2, axis=2)
        mask = distance2 < 36 * 36
        components = _components(mask)
        if len(components) != 1 or not 450 <= components[0][2] <= 850:
            raise ValueError(f"expected one intact {label} marker, found {components}")
        masks[label] = mask
    return masks


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"))
    if pixels.ndim != 3 or pixels.shape[0] < 300 or pixels.shape[1] < 300:
        raise ValueError("raster is too small for the declared corridor world")
    masks = _marker_masks(pixels)
    dark_wall = np.max(pixels.astype(np.int16), axis=2) < 70
    passable = ~dark_wall
    start_points = np.argwhere(masks["green"])
    if not len(start_points):
        raise ValueError("green START has no raster support")
    start_y, start_x = (int(value) for value in start_points[len(start_points) // 2])
    height, width = passable.shape
    visited = np.zeros_like(passable)
    visited[start_y, start_x] = True
    queue = deque([(start_y, start_x)])
    while queue:
        y, x = queue.popleft()
        if y and passable[y - 1, x] and not visited[y - 1, x]:
            visited[y - 1, x] = True
            queue.append((y - 1, x))
        if y + 1 < height and passable[y + 1, x] and not visited[y + 1, x]:
            visited[y + 1, x] = True
            queue.append((y + 1, x))
        if x and passable[y, x - 1] and not visited[y, x - 1]:
            visited[y, x - 1] = True
            queue.append((y, x - 1))
        if x + 1 < width and passable[y, x + 1] and not visited[y, x + 1]:
            visited[y, x + 1] = True
            queue.append((y, x + 1))
    winners = [color for color in CANDIDATES if np.any(visited & masks[color])]
    if len(winners) != 1:
        raise ValueError(f"ambiguous free-space connectivity: reachable terminals={winners}")
    return winners[0]
