"""Maximum inter-token separation across ordered panels (state tracking).

The image shows four panels side by side, read left to right. Each panel shows
a horizontal track divided into six slots, slot 0 at the left up to slot 5 at
the right. Two tokens sit on the track: a red token and a green token. The red
and green tokens keep their identity from panel to panel and may move left or
right between consecutive panels (their positions are fully drawn in every
panel, so the movement is visible). In each panel the horizontal separation
between the two tokens is the absolute difference of their slot indices.

Replay the four ordered panels and, for each panel, compute the slot separation
between the red token and the green token. Report the LARGEST separation (in
slots) that ever occurs across the whole sequence. The answer is one of the
integers 0, 1, 2, 3, 4, or 5.

Because the reported value is a temporal extremum over all four panels of a
two-identity distance, it cannot be read from any single panel, from either
token's own colour, from the sequence of one token alone, or from a plain sum
of their movements: it requires tracking both identities across the ordered
sequence and scanning the evolving inter-token gap.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 1180
HEIGHT = 300
PANEL_TOP = 60
PANEL_H = 150
PANEL_W = 260
START_X = 30
OP_STEP = 275
N_PANELS = 4
N_SLOTS = 6

BG = (255, 255, 255)
BORDER = (200, 200, 200)
TRACK = (120, 120, 120)
RED = (215, 60, 60)
GREEN = (40, 180, 90)
TOKEN_R = 16
RAIL_PAD = 38
SLOT_W = (PANEL_W - 2 * RAIL_PAD) / N_SLOTS
PANEL_MID = PANEL_TOP + PANEL_H // 2


def _panel_left(panel_index: int) -> int:
    return START_X + panel_index * OP_STEP


def _slot_center(panel_left: int, slot: int) -> tuple[int, int]:
    rail_left = panel_left + RAIL_PAD
    x = int(rail_left + SLOT_W * (slot + 0.5))
    return (x, PANEL_MID)


def _draw_track(draw: ImageDraw.ImageDraw, panel_left: int) -> None:
    x0 = panel_left + RAIL_PAD - 6
    x1 = panel_left + PANEL_W - RAIL_PAD + 6
    y = PANEL_MID
    draw.line([x0, y, x1, y], fill=TRACK, width=2)
    for k in range(N_SLOTS):
        cx = _slot_center(panel_left, k)[0]
        draw.line([cx, y - 6, cx, y + 6], fill=TRACK, width=2)


def _draw_token(draw: ImageDraw.ImageDraw, center: tuple[int, int], color) -> None:
    cx, cy = center
    draw.ellipse(
        [cx - TOKEN_R, cy - TOKEN_R, cx + TOKEN_R, cy + TOKEN_R],
        fill=color,
        outline=(0, 0, 0),
        width=2,
    )


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    posns = []
    for _ in range(N_PANELS):
        red = rng.randrange(N_SLOTS)
        green = rng.randrange(N_SLOTS)
        while green == red:
            green = rng.randrange(N_SLOTS)
        posns.append([red, green])
    return {"posns": posns}


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)
    posns = scene["posns"]
    for p in range(N_PANELS):
        left = _panel_left(p)
        draw.rectangle([left, PANEL_TOP, left + PANEL_W, PANEL_TOP + PANEL_H], outline=BORDER, width=3)
        _draw_track(draw, left)
        red_slot, green_slot = posns[p]
        _draw_token(draw, _slot_center(left, red_slot), RED)
        _draw_token(draw, _slot_center(left, green_slot), GREEN)
    return img


def analytic_gold(scene: dict) -> str:
    posns = scene["posns"]
    best = 0
    for red, green in posns:
        sep = abs(red - green)
        if sep > best:
            best = sep
    return str(best)


def margin(scene: dict) -> float:
    """Discrete-boundary margin: the reported max-separation is an exact integer
    class whose gap to every other integer class is at least one slot, so every
    non-degenerate scene carries a fixed positive boundary separation."""
    return 2.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    return []
