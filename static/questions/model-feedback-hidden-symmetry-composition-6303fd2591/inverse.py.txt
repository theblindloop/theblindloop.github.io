"""Independent image-only inverse arm for demonstrated square symmetries."""

from __future__ import annotations

import numpy as np
from PIL import Image


N = 5
GRID0 = 20
STEP = 34
BOARD = 210
PANELS = {
    "demo_a_in": (45, 95),
    "demo_a_out": (315, 95),
    "demo_b_in": (45, 410),
    "demo_b_out": (315, 410),
    "start": (650, 250),
    "A": (940, 95),
    "B": (1195, 95),
    "C": (940, 410),
    "D": (1195, 410),
}
RGB = {
    "red": np.array((223, 75, 63), dtype=np.int16),
    "blue": np.array((40, 103, 178), dtype=np.int16),
    "green": np.array((43, 155, 104), dtype=np.int16),
    "purple": np.array((139, 76, 165), dtype=np.int16),
}
TRANSFORMS = ("id", "r90", "r180", "r270", "flip_lr", "flip_ud", "diag", "anti")


def _transform(cell: tuple[int, int], name: str) -> tuple[int, int]:
    r, c = cell
    last = N - 1
    if name == "id": return r, c
    if name == "r90": return c, last - r
    if name == "r180": return last - r, last - c
    if name == "r270": return last - c, r
    if name == "flip_lr": return r, last - c
    if name == "flip_ud": return last - r, c
    if name == "diag": return c, r
    if name == "anti": return last - c, last - r
    raise ValueError(name)


def _read_board(arr: np.ndarray, key: str, colors: tuple[str, ...]) -> dict[str, tuple[int, int]]:
    x0, y0 = PANELS[key]
    crop = arr[y0:y0 + BOARD, x0:x0 + BOARD].astype(np.int32)
    found = {}
    for color in colors:
        delta = crop - RGB[color].astype(np.int32)
        mask = np.sum(delta * delta, axis=2) < 35 * 35
        ys, xs = np.nonzero(mask)
        if len(xs) < 40:
            raise ValueError(f"{key}: missing {color} landmark")
        col = int(round((float(xs.mean()) - GRID0 - STEP / 2) / STEP))
        row = int(round((float(ys.mean()) - GRID0 - STEP / 2) / STEP))
        if not (0 <= row < N and 0 <= col < N):
            raise ValueError(f"{key}: {color} outside grid")
        found[color] = (row, col)
    if len(set(found.values())) != len(found):
        raise ValueError(f"{key}: coincident landmarks")
    return found


def _infer(before: dict[str, tuple[int, int]], after: dict[str, tuple[int, int]]) -> str:
    matches = [
        name for name in TRANSFORMS
        if all(_transform(before[color], name) == after[color] for color in before)
    ]
    if len(matches) != 1:
        raise ValueError(f"demonstration identifies {len(matches)} transforms")
    return matches[0]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    demo_colors = ("red", "blue", "green")
    all_colors = ("red", "blue", "green", "purple")
    a = _infer(_read_board(arr, "demo_a_in", demo_colors), _read_board(arr, "demo_a_out", demo_colors))
    b = _infer(_read_board(arr, "demo_b_in", demo_colors), _read_board(arr, "demo_b_out", demo_colors))
    start = _read_board(arr, "start", all_colors)
    expected = {color: _transform(_transform(cell, a), b) for color, cell in start.items()}
    matches = [label for label in ("A", "B", "C", "D") if _read_board(arr, label, all_colors) == expected]
    if len(matches) != 1:
        raise ValueError(f"found {len(matches)} matching result panels")
    return matches[0]
