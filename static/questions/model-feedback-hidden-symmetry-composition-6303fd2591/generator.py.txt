"""Deterministic renderer for two demonstrated, unnamed square symmetries."""

from __future__ import annotations

import copy
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont


WIDTH, HEIGHT = 1450, 730
N = 5
BOARD = 210
GRID0 = 20
STEP = 34
PANEL_POSITIONS = {
    "demo_a_in": (45, 95),
    "demo_a_out": (315, 95),
    "demo_b_in": (45, 410),
    "demo_b_out": (315, 410),
    "start": (650, 250),
    "A": (940, 95),
    "B": (1195, 95),
    "C": (940, 410),
    "D": (1195, 410),
}
COLORS = {
    "red": "#df4b3f",
    "blue": "#2867b2",
    "green": "#2b9b68",
    "purple": "#8b4ca5",
}
TRANSFORMS = ("r90", "r180", "r270", "flip_lr", "flip_ud", "diag", "anti")
ALL_TRANSFORMS = ("id",) + TRANSFORMS
OPTION_LABELS = ("A", "B", "C", "D")


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    suffix = "Bold" if bold else ""
    path = f"/usr/share/fonts/truetype/dejavu/DejaVuSans{('-' + suffix) if suffix else ''}.ttf"
    return ImageFont.truetype(path, size=size)


def transform_cell(cell: tuple[int, int], name: str) -> tuple[int, int]:
    r, c = cell
    last = N - 1
    table = {
        "id": (r, c),
        "r90": (c, last - r),
        "r180": (last - r, last - c),
        "r270": (last - c, r),
        "flip_lr": (r, last - c),
        "flip_ud": (last - r, c),
        "diag": (c, r),
        "anti": (last - c, last - r),
    }
    return table[name]


def _apply(markers: Iterable[dict], name: str) -> list[dict]:
    result = []
    for marker in markers:
        row, col = transform_cell((int(marker["row"]), int(marker["col"])), name)
        result.append({"color": marker["color"], "row": row, "col": col})
    return result


def _state(markers: Iterable[dict]) -> tuple[tuple[str, int, int], ...]:
    return tuple(sorted((m["color"], int(m["row"]), int(m["col"])) for m in markers))


def _unique_demo(rng: random.Random) -> list[dict]:
    cells = [(r, c) for r in range(N) for c in range(N)]
    while True:
        chosen = rng.sample(cells, 3)
        markers = [
            {"color": color, "row": cell[0], "col": cell[1]}
            for color, cell in zip(("red", "blue", "green"), chosen)
        ]
        if len({_state(_apply(markers, name)) for name in ALL_TRANSFORMS}) == 8:
            return markers


def _unique_start(rng: random.Random) -> list[dict]:
    cells = [(r, c) for r in range(N) for c in range(N)]
    while True:
        chosen = rng.sample(cells, 4)
        markers = [
            {"color": color, "row": cell[0], "col": cell[1]}
            for color, cell in zip(("red", "blue", "green", "purple"), chosen)
        ]
        if len({_state(_apply(markers, name)) for name in ALL_TRANSFORMS}) == 8:
            return markers


def _mode_state(start: list[dict], a: str, b: str, mode: str) -> tuple[tuple[str, int, int], ...]:
    if mode == "AB":
        return _state(_apply(_apply(start, a), b))
    if mode == "BA":
        return _state(_apply(_apply(start, b), a))
    if mode == "A":
        return _state(_apply(start, a))
    if mode == "B":
        return _state(_apply(start, b))
    raise ValueError(mode)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    demo_a = _unique_demo(rng)
    demo_b = _unique_demo(rng)
    start = _unique_start(rng)
    while True:
        transform_a, transform_b = rng.sample(TRANSFORMS, 2)
        states = {
            mode: _mode_state(start, transform_a, transform_b, mode)
            for mode in ("AB", "BA", "A", "B")
        }
        if len(set(states.values())) == 4:
            break
    correct_slot = int(seed) % 4
    distractors = ["BA", "A", "B"]
    rng.shuffle(distractors)
    option_modes = distractors[:]
    option_modes.insert(correct_slot, "AB")
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "transform_a": transform_a,
        "transform_b": transform_b,
        "demo_a": demo_a,
        "demo_b": demo_b,
        "start": start,
        "option_modes": option_modes,
    }


def _markers_from_state(state: tuple[tuple[str, int, int], ...]) -> list[dict]:
    return [{"color": color, "row": row, "col": col} for color, row, col in state]


def _draw_marker(draw: ImageDraw.ImageDraw, marker: dict, x0: int, y0: int) -> None:
    cx = x0 + GRID0 + STEP // 2 + int(marker["col"]) * STEP
    cy = y0 + GRID0 + STEP // 2 + int(marker["row"]) * STEP
    color = COLORS[marker["color"]]
    radius = 12
    if marker["color"] == "red":
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=color, outline="#182333", width=2)
    elif marker["color"] == "blue":
        draw.rounded_rectangle((cx - radius, cy - radius, cx + radius, cy + radius), radius=3, fill=color, outline="#182333", width=2)
    elif marker["color"] == "green":
        draw.polygon([(cx, cy - 14), (cx - 14, cy + 12), (cx + 14, cy + 12)], fill=color, outline="#182333")
        draw.line([(cx, cy - 14), (cx - 14, cy + 12), (cx + 14, cy + 12), (cx, cy - 14)], fill="#182333", width=2)
    else:
        draw.polygon([(cx, cy - 14), (cx - 14, cy), (cx, cy + 14), (cx + 14, cy)], fill=color, outline="#182333")
        draw.line([(cx, cy - 14), (cx - 14, cy), (cx, cy + 14), (cx + 14, cy), (cx, cy - 14)], fill="#182333", width=2)


def _draw_board(draw: ImageDraw.ImageDraw, x0: int, y0: int, markers: list[dict], label: str) -> None:
    draw.rounded_rectangle((x0, y0, x0 + BOARD, y0 + BOARD), radius=10, fill="#fbfcfe", outline="#34445a", width=3)
    left, top = x0 + GRID0, y0 + GRID0
    right, bottom = left + N * STEP, top + N * STEP
    draw.rectangle((left, top, right, bottom), fill="#f5f7fa", outline="#7b8798", width=2)
    for k in range(1, N):
        xx, yy = left + k * STEP, top + k * STEP
        draw.line((xx, top, xx, bottom), fill="#c3cad4", width=1)
        draw.line((left, yy, right, yy), fill="#c3cad4", width=1)
    for marker in markers:
        _draw_marker(draw, marker, x0, y0)
    draw.text((x0 + BOARD // 2, y0 - 10), label, font=_font(19, True), fill="#182333", anchor="ms")


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), "#f2f4f7")
    draw = ImageDraw.Draw(image)
    draw.text((45, 30), "INFER EACH WHOLE-BOARD MOVE", font=_font(25, True), fill="#182333")
    draw.text((650, 30), "APPLY A, THEN B", font=_font(25, True), fill="#182333")

    a_in = list(scene["demo_a"])
    b_in = list(scene["demo_b"])
    _draw_board(draw, *PANEL_POSITIONS["demo_a_in"], a_in, "MACHINE A — IN")
    _draw_board(draw, *PANEL_POSITIONS["demo_a_out"], _apply(a_in, scene["transform_a"]), "OUT")
    _draw_board(draw, *PANEL_POSITIONS["demo_b_in"], b_in, "MACHINE B — IN")
    _draw_board(draw, *PANEL_POSITIONS["demo_b_out"], _apply(b_in, scene["transform_b"]), "OUT")
    for y in (200, 515):
        draw.line((265, y, 302, y), fill="#182333", width=5)
        draw.polygon([(302, y), (286, y - 10), (286, y + 10)], fill="#182333")

    _draw_board(draw, *PANEL_POSITIONS["start"], list(scene["start"]), "START")
    draw.text((1045, 44), "RESULTS", font=_font(23, True), fill="#182333", anchor="ms")
    for label, mode in zip(OPTION_LABELS, scene["option_modes"]):
        state = _mode_state(list(scene["start"]), scene["transform_a"], scene["transform_b"], mode)
        _draw_board(draw, *PANEL_POSITIONS[label], _markers_from_state(state), f"RESULT {label}")
    return image


def analytic_gold(scene: dict) -> str:
    return OPTION_LABELS[list(scene["option_modes"]).index("AB")]


def margin(scene: dict) -> float:
    correct = _mode_state(list(scene["start"]), scene["transform_a"], scene["transform_b"], "AB")
    other = [
        _mode_state(list(scene["start"]), scene["transform_a"], scene["transform_b"], mode)
        for mode in scene["option_modes"] if mode != "AB"
    ]
    correct_map = {color: (r, c) for color, r, c in correct}
    return float(min(sum(correct_map[color] != (r, c) for color, r, c in state) for state in other))


def _matching_transforms(before: list[dict], after: list[dict]) -> list[str]:
    target = _state(after)
    return [name for name in ALL_TRANSFORMS if _state(_apply(before, name)) == target]


def is_quarantined(scene: dict) -> bool:
    a_after = _apply(list(scene["demo_a"]), scene["transform_a"])
    b_after = _apply(list(scene["demo_b"]), scene["transform_b"])
    if len(_matching_transforms(list(scene["demo_a"]), a_after)) != 1:
        return True
    if len(_matching_transforms(list(scene["demo_b"]), b_after)) != 1:
        return True
    states = [
        _mode_state(list(scene["start"]), scene["transform_a"], scene["transform_b"], mode)
        for mode in scene["option_modes"]
    ]
    return len(set(states)) != 4


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    altered = copy.deepcopy(scene)
    for field in ("demo_a", "demo_b", "start"):
        altered[field] = list(reversed(altered[field]))
    return [("reverse_unordered_landmark_serialization", altered)]
