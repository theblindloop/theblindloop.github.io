"""Independent pixel-only inverse for balanced perforated plates.

The inverse knows only the public palette and the visible 2-by-2 panel layout.
It enumerates all four candidate regions, measures each blue mask's area and
centroid, locates the gold pin from pixels, and requires a unique inside-pin
winner with a runner-up gap. It imports no forward-world modules.
"""

from __future__ import annotations

from collections import deque
import math

import numpy as np
from PIL import Image

PLATE_RGB = (35, 105, 153)
PIN_RGB = (242, 178, 52)
LABELS = ("A", "B", "C", "D")
COLOR_TOLERANCE = 28
PIN_RADIUS_PX = 7.0
MIN_PLATE_AREA = 20_000
INTACT_AREA_RATIO = 0.999
MIN_RUNNER_GAP_PX = 9.5


def _color_mask(array: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    delta = array.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2) <= COLOR_TOLERANCE * COLOR_TOLERANCE


def _component_count(mask: np.ndarray) -> int:
    pending = mask.copy()
    count = 0
    height, width = pending.shape
    while pending.any():
        y, x = (int(value) for value in np.argwhere(pending)[0])
        count += 1
        queue = deque([(y, x)])
        pending[y, x] = False
        while queue:
            cy, cx = queue.popleft()
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and pending[ny, nx]:
                    pending[ny, nx] = False
                    queue.append((ny, nx))
    return count


def _centroid(mask: np.ndarray, x_offset: int, y_offset: int) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if not len(xs):
        raise ValueError("candidate region contains no plate-colored pixels")
    return float(xs.mean() + x_offset), float(ys.mean() + y_offset)


def analyze_image(image: Image.Image) -> dict:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    height, width = array.shape[:2]
    if width < 300 or height < 300:
        raise ValueError("image is too small for four candidate panels")
    plate = _color_mask(array, PLATE_RGB)
    pin = _color_mask(array, PIN_RGB)
    x_cuts = (0, width // 2, width)
    y_cuts = (0, height // 2, height)
    records = []
    index = 0
    for row in range(2):
        for column in range(2):
            x0, x1 = x_cuts[column], x_cuts[column + 1]
            y0, y1 = y_cuts[row], y_cuts[row + 1]
            local_plate = plate[y0:y1, x0:x1]
            local_pin = pin[y0:y1, x0:x1]
            area = int(local_plate.sum())
            plate_center = _centroid(local_plate, x0, y0)
            if int(local_pin.sum()) >= 20:
                pin_center = _centroid(local_pin, x0, y0)
            else:
                # The clean image always supplies the visible pin. This fallback
                # makes destructive ablations of a losing panel nondiagnostic.
                pin_center = ((x0 + x1 - 1) / 2.0, (y0 + y1 - 1) / 2.0)
            distance = math.hypot(
                plate_center[0] - pin_center[0], plate_center[1] - pin_center[1]
            )
            records.append(
                {
                    "label": LABELS[index],
                    "area": area,
                    "distance_px": distance,
                    "components": _component_count(local_plate),
                }
            )
            index += 1

    maximum_area = max(record["area"] for record in records)
    if maximum_area < MIN_PLATE_AREA:
        raise ValueError("no intact plate-sized candidate was recovered")
    eligible = [
        record
        for record in records
        if record["area"] >= maximum_area * INTACT_AREA_RATIO
    ]
    if len(eligible) < 2:
        raise ValueError("fewer than two intact candidates remain")
    ranked = sorted(eligible, key=lambda record: (record["distance_px"], record["label"]))
    inside = [record for record in ranked if record["distance_px"] <= PIN_RADIUS_PX]
    if len(inside) != 1 or inside[0] is not ranked[0]:
        raise ValueError(f"expected one inside-pin winner, found {len(inside)}")
    if ranked[0]["components"] != 1:
        raise ValueError("winning plate is not one connected visible region")
    gap = float(ranked[1]["distance_px"] - ranked[0]["distance_px"])
    if gap < MIN_RUNNER_GAP_PX:
        raise ValueError(f"centroid search is ambiguous: runner-up gap {gap:.2f}px")
    return {
        "decision": str(ranked[0]["label"]),
        "margin_px": gap,
        "measurements": records,
    }


def decision_from_image(image: Image.Image) -> str:
    return str(analyze_image(image)["decision"])
