"""Renderer for the summed-reach ruler search world.

Two reference bars sit in opposite far corners of an otherwise empty page: one
near the top-left corner, one near the bottom-right corner. Four candidate bars
are stacked in one column near the middle of the page. Exactly one candidate bar
is as long as the two reference bars laid end to end, that is, as long as their
two lengths added together.

The decisive evidence is therefore deliberately split across three widely
separated, mutually disconnected marks: reference bar A, reference bar B and the
winning candidate bar. Neither reference alone determines the answer, because one
distractor is built to match a single reference bar or a doubled reference bar.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "sum_reach_ruler_search-renderer-0.1.0"

CANVAS_W = 768
CANVAS_H = 576

BAR_T = 12                # every bar has this thickness, in pixels
REF_A_X = 40              # left end of the top-left reference bar
REF_B_RIGHT = 728         # right end of the bottom-right reference bar
STACK_PITCH = 62          # vertical spacing between candidate bars
SPECK = 3
CLEARANCE = 18            # specks stay this far away from every bar
MIN_REF = 50
MAX_REF = 100

WORDS = ("first", "second", "third", "fourth")
QUARANTINE_MARGIN = 16.0

PALETTES = (
    ((246, 244, 238), (38, 42, 50), (188, 72, 56)),
    ((238, 242, 246), (44, 38, 46), (36, 108, 172)),
    ((243, 240, 232), (34, 48, 42), (152, 96, 24)),
    ((240, 240, 244), (48, 40, 40), (28, 118, 108)),
)


def _offsets(rng: random.Random, count: int, forced: int | None) -> list | None:
    """``count`` distinct positive length offsets, pairwise at least 12 apart."""

    if count == 0:
        return []
    values: list = []
    if forced is not None:
        values.append(forced)
    guard = 0
    while len(values) < count:
        guard += 1
        if guard > 400:
            return None
        pick = rng.randrange(10, 46, 2)
        if all(abs(pick - other) >= 12 for other in values):
            values.append(pick)
    rng.shuffle(values)
    return values


def _sample_bars(rng: random.Random, winner: int, rank: int, ref_a: int, ref_b: int):
    """Four candidate lengths whose unique sum-match sits at length rank ``rank``."""

    total = ref_a + ref_b
    above_count = rank - 1          # distractors strictly longer than the sum
    below_count = 4 - rank          # distractors strictly shorter than the sum
    big_ref = max(ref_a, ref_b)
    small_ref = min(ref_a, ref_b)

    # One distractor always reads as "a single reference bar" (length small_ref,
    # reachable only from below) or as "one reference counted twice" (length
    # total + big_ref, reachable only from above), so a solver that looks at one
    # corner alone is actively misled.
    if below_count > 0:
        below = _offsets(rng, below_count, big_ref)
        above = _offsets(rng, above_count, None)
    else:
        below = []
        above = _offsets(rng, above_count, big_ref)
    if below is None or above is None:
        return None

    lengths = [total + off for off in above] + [total] + [total - off for off in below]
    if len(set(lengths)) != 4:
        return None
    if min(lengths) < 40 or max(lengths) > 300:
        return None
    if sorted(lengths, reverse=True).index(total) != rank - 1:
        return None
    if any(abs(a - b) < 12 for i, a in enumerate(lengths) for b in lengths[i + 1:]):
        return None
    if small_ref not in lengths and (total + big_ref) not in lengths:
        return None

    distractors = [value for value in lengths if value != total]
    rng.shuffle(distractors)
    slots = [slot for slot in range(4) if slot != winner]
    bars = [{"slot": winner, "length": total}]
    for slot, value in zip(slots, distractors):
        bars.append({"slot": slot, "length": int(value)})
    rng.shuffle(bars)
    return bars


def sample_scene(seed: int) -> dict:
    """Deterministic scene: two corner reference bars plus a candidate column."""

    rng = random.Random(seed * 7919 + 29)
    winner = seed % 4
    rank = (seed // 4) % 4 + 1

    for _ in range(500):
        ref_a = rng.randrange(MIN_REF, MAX_REF + 1, 2)
        ref_b = rng.randrange(MIN_REF, MAX_REF + 1, 2)
        if abs(ref_a - ref_b) < 14:
            continue
        bars = _sample_bars(rng, winner, rank, ref_a, ref_b)
        if bars is not None:
            break
    else:  # pragma: no cover - the loop succeeds well inside 500 attempts
        raise RuntimeError("failed to sample a counterbalanced scene")

    scene = {
        "width": CANVAS_W,
        "height": CANVAS_H,
        "palette": rng.randrange(len(PALETTES)),
        "ref_a": {"x": REF_A_X, "y": rng.randrange(34, 71, 2), "length": int(ref_a)},
        "ref_b": {
            "x": REF_B_RIGHT - int(ref_b),
            "y": rng.randrange(498, 535, 2),
            "length": int(ref_b),
        },
        "stack_x": rng.randrange(250, 301, 2),
        "stack_y": rng.randrange(186, 217, 2),
        "bars": bars,
        "specks": [],
    }

    boxes = [bar_box(scene, bar) for bar in bars]
    boxes.append(ref_box(scene["ref_a"]))
    boxes.append(ref_box(scene["ref_b"]))
    specks: list = []
    wanted = rng.randint(6, 14)
    guard = 0
    while len(specks) < wanted and guard < 4000:
        guard += 1
        sx = rng.randint(8, CANVAS_W - 8 - SPECK)
        sy = rng.randint(8, CANVAS_H - 8 - SPECK)
        if any(
            x0 - CLEARANCE <= sx + SPECK and sx <= x1 + CLEARANCE
            and y0 - CLEARANCE <= sy + SPECK and sy <= y1 + CLEARANCE
            for x0, y0, x1, y1 in boxes
        ):
            continue
        if any(abs(sx - ox) < 4 * SPECK and abs(sy - oy) < 4 * SPECK for ox, oy in specks):
            continue
        specks.append([sx, sy])
    scene["specks"] = specks
    return scene


def ref_box(ref: dict) -> tuple:
    """Pixel box of one reference bar."""

    x0 = int(ref["x"])
    y0 = int(ref["y"])
    return (x0, y0, x0 + int(ref["length"]) - 1, y0 + BAR_T - 1)


def bar_box(scene: dict, bar: dict) -> tuple:
    """Pixel box of one candidate bar, placed by its stack slot."""

    x0 = int(scene["stack_x"])
    y0 = int(scene["stack_y"]) + int(bar["slot"]) * STACK_PITCH
    return (x0, y0, x0 + int(bar["length"]) - 1, y0 + BAR_T - 1)


def analytic_gold(scene: dict) -> str:
    """Stack position of the only candidate as long as both references together."""

    total = int(scene["ref_a"]["length"]) + int(scene["ref_b"]["length"])
    hits = [bar for bar in scene["bars"] if int(bar["length"]) == total]
    if len(hits) != 1:
        raise ValueError("scene does not have a unique summed-reach candidate")
    return WORDS[int(hits[0]["slot"])]


def margin(scene: dict) -> float:
    """Smallest competing length error against the combined reference length."""

    total = int(scene["ref_a"]["length"]) + int(scene["ref_b"]["length"])
    errors = [abs(int(bar["length"]) - total) for bar in scene["bars"]]
    competing = [value for value in errors if value > 0]
    if len(competing) != len(errors) - 1:
        raise ValueError("scene does not have exactly one summed-reach candidate")
    return float(min(competing))


def is_quarantined(scene: dict) -> bool:
    """Near-boundary scenes where a distractor is nearly the combined length."""

    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list:
    """Pixel-identical, answer-preserving rewrites of the latent scene."""

    out = []
    if len(scene["specks"]) > 1:
        flipped = dict(scene)
        flipped["specks"] = list(reversed(scene["specks"]))
        out.append(("speck_order_reversed", flipped))
    if len(scene["bars"]) > 1:
        rotated = dict(scene)
        rotated["bars"] = scene["bars"][1:] + scene["bars"][:1]
        out.append(("bar_record_order_rotated", rotated))
    return out


def render(scene: dict) -> Image.Image:
    """Rasterise the scene. Marks never overlap, so drawing order is irrelevant."""

    background, ink, reference = PALETTES[int(scene["palette"]) % len(PALETTES)]
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), background)
    draw = ImageDraw.Draw(image)
    for sx, sy in scene["specks"]:
        draw.rectangle((sx, sy, sx + SPECK - 1, sy + SPECK - 1), fill=ink)
    for ref in (scene["ref_a"], scene["ref_b"]):
        draw.rectangle(ref_box(ref), fill=reference)
    for bar in scene["bars"]:
        draw.rectangle(bar_box(scene, bar), fill=ink)
    return image
