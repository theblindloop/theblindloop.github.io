"""Independent image-only inverse for cumulative state coverage.

This module repeats the public keyed-tile geometry instead of importing the
renderer or any latent-scene code.  It reads the visible state key from the
PNG, recognizes identity glyph silhouettes, reconstructs the ordered table,
and then performs the cumulative-set computation.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw

LABELS = ("A", "B", "C", "D")
STATE_NAMES = ("LOW", "MID", "HIGH")
WIDTH = 1500
HEIGHT = 590
PANEL_COUNT = 7
PANEL_X0 = 18
PANEL_STEP = 210
ROW_Y0 = 177
ROW_STEP = 86
GLYPH_DX = 34
SWATCH_CENTER_DX = 123
KEY_X0 = 420
KEY_STEP = 95
KEY_SWATCH_CENTER_DX = 27
KEY_Y = 52


def _templates() -> dict[str, np.ndarray]:
    templates: dict[str, np.ndarray] = {}
    for label in LABELS:
        mask = Image.new("L", (37, 37), 0)
        draw = ImageDraw.Draw(mask)
        cx = cy = 18
        if label == "A":
            draw.ellipse((cx - 12, cy - 12, cx + 12, cy + 12), fill=255)
        elif label == "B":
            draw.polygon(((cx, cy - 14), (cx - 13, cy + 12), (cx + 13, cy + 12)), fill=255)
        elif label == "C":
            draw.rectangle((cx - 11, cy - 11, cx + 11, cy + 11), fill=255)
        else:
            draw.polygon(((cx, cy - 14), (cx + 14, cy), (cx, cy + 14), (cx - 14, cy)), fill=255)
        templates[label] = np.asarray(mask, dtype=np.uint8) > 0
    return templates


_GLYPH_TEMPLATES = _templates()


def _read_label(arr: np.ndarray, cx: int, cy: int) -> str:
    patch = arr[cy - 18 : cy + 19, cx - 18 : cx + 19]
    if patch.shape != (37, 37, 3):
        raise ValueError("identity glyph is clipped")
    dark = np.max(patch, axis=2) < 150
    scores = {
        label: int(np.logical_xor(dark, template).sum())
        for label, template in _GLYPH_TEMPLATES.items()
    }
    ranked = sorted(LABELS, key=lambda label: (scores[label], label))
    if scores[ranked[0]] > 42 or scores[ranked[1]] - scores[ranked[0]] < 4:
        raise ValueError(f"unresolved identity glyph: {scores}")
    return ranked[0]


def _read_palette(arr: np.ndarray) -> np.ndarray:
    colors = []
    for state in range(len(STATE_NAMES)):
        x = KEY_X0 + state * KEY_STEP + KEY_SWATCH_CENTER_DX
        colors.append(arr[KEY_Y, x].astype(np.int16))
    palette = np.asarray(colors, dtype=np.int16)
    if min(float(np.linalg.norm(palette[i] - palette[j])) for i in range(3) for j in range(i)) < 80:
        raise ValueError("state key colors are not separated")
    return palette


def _read_state(arr: np.ndarray, x: int, y: int, palette: np.ndarray) -> int:
    pixel = arr[y, x].astype(np.int16)
    distances = np.linalg.norm(palette - pixel, axis=1)
    state = int(np.argmin(distances))
    if float(distances[state]) > 45.0:
        raise ValueError("state swatch is missing or corrupted")
    return state


def _coverage_times(table: dict[str, list[int]]) -> list[int | None]:
    seen = [set() for _ in STATE_NAMES]
    result: list[int | None] = [None] * len(STATE_NAMES)
    for panel in range(PANEL_COUNT):
        for label in LABELS:
            state = int(table[label][panel])
            if state not in range(len(STATE_NAMES)):
                raise ValueError("decoded state index out of range")
            seen[state].add(label)
        for state in range(len(STATE_NAMES)):
            if result[state] is None and len(seen[state]) == len(LABELS):
                result[state] = panel
    return result


def decision_from_image(image: Image.Image) -> str:
    """Recover the winning state from the final PNG alone."""
    rgb = image.convert("RGB")
    if rgb.size != (WIDTH, HEIGHT):
        raise ValueError(f"unexpected board size {rgb.size}")
    arr = np.asarray(rgb)
    palette = _read_palette(arr)
    table = {label: [] for label in LABELS}
    for panel in range(PANEL_COUNT):
        x0 = PANEL_X0 + panel * PANEL_STEP
        seen: set[str] = set()
        for row in range(len(LABELS)):
            cy = ROW_Y0 + row * ROW_STEP
            label = _read_label(arr, x0 + GLYPH_DX, cy)
            if label in seen:
                raise ValueError("duplicate identity in a panel")
            seen.add(label)
            table[label].append(_read_state(arr, x0 + SWATCH_CENTER_DX, cy, palette))
        if seen != set(LABELS):
            raise ValueError("panel does not contain all four identities")

    times = _coverage_times(table)
    if any(time is None for time in times):
        raise ValueError("a state never reaches full identity coverage")
    ordered = sorted(range(len(STATE_NAMES)), key=lambda state: (int(times[state]), state))
    if int(times[ordered[0]]) == int(times[ordered[1]]):
        raise ValueError(f"ambiguous recovered coverage times: {times}")
    return STATE_NAMES[ordered[0]]
