"""Deterministic renderer and analytic gold for cumulative state coverage."""

from __future__ import annotations

import copy
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

LABELS = ("A", "B", "C", "D")
STATE_NAMES = ("LOW", "MID", "HIGH")
BASE_STATE_COLORS = (
    (227, 89, 89),
    (79, 169, 112),
    (79, 125, 214),
)
BACKGROUND = (255, 255, 255)
PANEL_TINTS = (
    (248, 250, 253),
    (250, 249, 246),
    (247, 251, 248),
    (250, 248, 252),
)
INK = (35, 39, 48)
MUTED = (133, 143, 155)
RULE = (221, 226, 232)

WIDTH = 1500
HEIGHT = 590
PANEL_COUNT = 7
PANEL_X0 = 18
PANEL_STEP = 210
PANEL_W = 190
PANEL_Y = 108
PANEL_H = 445
ROW_Y0 = 177
ROW_STEP = 86
GLYPH_DX = 34
SWATCH_DX0 = 74
SWATCH_DX1 = 172
SWATCH_CENTER_DX = 123
SWATCH_HALF_H = 15
KEY_X0 = 420
KEY_STEP = 95
KEY_SWATCH_CENTER_DX = 27
KEY_Y = 52


def _coverage_times(states: dict[str, list[int]]) -> list[int | None]:
    """Return the first panel at which each state has seen every label."""
    seen = [set() for _ in STATE_NAMES]
    result: list[int | None] = [None] * len(STATE_NAMES)
    for panel in range(PANEL_COUNT):
        for label in LABELS:
            state = int(states[label][panel])
            if state not in range(len(STATE_NAMES)):
                raise ValueError("state index out of range")
            seen[state].add(label)
        for state in range(len(STATE_NAMES)):
            if result[state] is None and len(seen[state]) == len(LABELS):
                result[state] = panel
    return result


def _winner_index(states: dict[str, list[int]]) -> int:
    times = _coverage_times(states)
    if any(time is None for time in times):
        raise ValueError("not every state reaches full identity coverage")
    ordered = sorted(range(len(STATE_NAMES)), key=lambda state: (int(times[state]), state))
    if int(times[ordered[0]]) == int(times[ordered[1]]):
        raise ValueError("state coverage completion is tied")
    return ordered[0]


def _top_gap(states: dict[str, list[int]]) -> float:
    times = _coverage_times(states)
    if any(time is None for time in times):
        return 0.0
    ordered = sorted(int(time) for time in times if time is not None)
    return float(ordered[1] - ordered[0])


def _shape_points(label: str, cx: int, cy: int) -> tuple[str, object]:
    if label == "A":
        return "ellipse", (cx - 12, cy - 12, cx + 12, cy + 12)
    if label == "B":
        return "polygon", ((cx, cy - 14), (cx - 13, cy + 12), (cx + 13, cy + 12))
    if label == "C":
        return "rectangle", (cx - 11, cy - 11, cx + 11, cy + 11)
    if label == "D":
        return "polygon", ((cx, cy - 14), (cx + 14, cy), (cx, cy + 14), (cx - 14, cy))
    raise ValueError(f"unknown identity label {label!r}")


def _draw_identity_glyph(draw: ImageDraw.ImageDraw, label: str, cx: int, cy: int) -> None:
    kind, payload = _shape_points(label, cx, cy)
    if kind == "ellipse":
        draw.ellipse(payload, fill=INK)
    elif kind == "rectangle":
        draw.rectangle(payload, fill=INK)
    else:
        draw.polygon(payload, fill=INK)


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    panel_index: int,
    row_order: Iterable[str],
    states: dict[str, list[int]],
    state_palette: list[list[int]],
    panel_tint: list[int],
    font: ImageFont.ImageFont,
) -> None:
    x = PANEL_X0 + panel_index * PANEL_STEP
    draw.rectangle(
        (x, PANEL_Y, x + PANEL_W, PANEL_Y + PANEL_H),
        fill=tuple(panel_tint),
        outline=INK,
        width=2,
    )
    title = f"P{panel_index + 1}"
    draw.text((x + 84, PANEL_Y + 9), title, fill=INK, font=font)
    for row_index, label in enumerate(row_order):
        cy = ROW_Y0 + row_index * ROW_STEP
        if row_index:
            draw.line(
                (x + 12, cy - 43, x + PANEL_W - 12, cy - 43),
                fill=RULE,
                width=1,
            )
        _draw_identity_glyph(draw, label, x + GLYPH_DX, cy)
        state = int(states[label][panel_index])
        draw.rectangle(
            (x + SWATCH_DX0, cy - SWATCH_HALF_H, x + SWATCH_DX1, cy + SWATCH_HALF_H),
            fill=tuple(state_palette[state]),
            outline=INK,
            width=1,
        )


def render(scene: dict) -> Image.Image:
    """Render one scene to a deterministic RGB PNG."""
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    draw.text((24, 20), "IDENTITY KEY", fill=INK, font=font)
    for index, label in enumerate(LABELS):
        cx = 116 + index * 52
        _draw_identity_glyph(draw, label, cx, 52)
        draw.text((cx - 4, 70), label, fill=INK, font=font)

    draw.text((340, 20), "STATE KEY", fill=INK, font=font)
    for state, name in enumerate(STATE_NAMES):
        x = KEY_X0 + state * KEY_STEP
        draw.rectangle(
            (x, KEY_Y - 13, x + 54, KEY_Y + 13),
            fill=tuple(scene["state_palette"][state]),
            outline=INK,
            width=1,
        )
        draw.text((x + 61, KEY_Y - 5), name, fill=INK, font=font)
    draw.text((790, 38), "Read numbered panels left to right", fill=MUTED, font=font)

    # Panels are disjoint.  draw_order is deliberately a render-only nuisance
    # so reversing it is a pixel-identical latent alias.
    for panel_index in scene["draw_order"]:
        panel_index = int(panel_index)
        _draw_panel(
            draw,
            panel_index,
            scene["row_orders"][panel_index],
            scene["states"],
            scene["state_palette"],
            scene["panel_tints"][panel_index],
            font,
        )
    return image


def analytic_gold(scene: dict) -> str:
    """Return the state whose cumulative identity set completes first."""
    return STATE_NAMES[_winner_index(scene["states"])]


def margin(scene: dict) -> float:
    """Return the runner-up completion-panel gap."""
    return _top_gap(scene["states"])


def is_quarantined(scene: dict) -> bool:
    """Quarantine ties, incomplete coverage, and one-panel leads."""
    times = _coverage_times(scene["states"])
    return (
        any(time is None for time in times)
        or len({time for time in times if time is not None}) != len(STATE_NAMES)
        or margin(scene) < 2.0
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Return a nontrivial panel draw-order transform with identical pixels."""
    twin = copy.deepcopy(scene)
    twin["draw_order"] = list(reversed(twin["draw_order"]))
    return [("reverse_disjoint_panel_draw_order", twin)]


def _sample_states(rng: random.Random, target: int, minimum_gap: int) -> dict[str, list[int]]:
    for _ in range(50000):
        states = {
            label: [rng.randrange(len(STATE_NAMES)) for _ in range(PANEL_COUNT)]
            for label in LABELS
        }
        if any(set(states[label]) != set(range(len(STATE_NAMES))) for label in LABELS):
            continue
        times = _coverage_times(states)
        if any(time is None for time in times):
            continue
        ordered = sorted(range(len(STATE_NAMES)), key=lambda state: (int(times[state]), state))
        if ordered[0] != target:
            continue
        if int(times[ordered[1]]) - int(times[ordered[0]]) < minimum_gap:
            continue
        if len({int(time) for time in times if time is not None}) != len(STATE_NAMES):
            continue
        return states
    raise RuntimeError(f"unable to sample coverage scene for state {target}")


def sample_scene(seed: int) -> dict:
    """Sample a deterministic, answer-balanced scene from one integer seed."""
    seed = int(seed)
    rng = random.Random(seed)
    target = seed % len(STATE_NAMES)
    minimum_gap = 1 if seed % 5 == 0 else 2
    states = _sample_states(rng, target, minimum_gap)

    row_orders: list[list[str]] = []
    for _ in range(PANEL_COUNT):
        order = list(LABELS)
        rng.shuffle(order)
        row_orders.append(order)

    palette = [list(color) for color in BASE_STATE_COLORS]
    rng.shuffle(palette)
    panel_tints = [list(PANEL_TINTS[rng.randrange(len(PANEL_TINTS))]) for _ in range(PANEL_COUNT)]
    draw_order = list(range(PANEL_COUNT))
    rng.shuffle(draw_order)
    return {
        "states": states,
        "row_orders": row_orders,
        "state_palette": palette,
        "panel_tints": panel_tints,
        "draw_order": draw_order,
    }
