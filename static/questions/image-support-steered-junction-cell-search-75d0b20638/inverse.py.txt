"""Independent pixel-only inverse arm for junction-cell search."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

_PALETTE = {
    "red": (220, 62, 66),
    "blue": (48, 112, 205),
    "green": (42, 153, 91),
    "purple": (145, 78, 190),
}


def _occupancy(mask: np.ndarray) -> set[tuple[int, int]] | None:
    ys, xs = np.nonzero(mask)
    area = int(len(xs))
    if area < 9 * 12 * 12:
        return None
    cell = int(round(math.sqrt(area / 9.0)))
    if cell < 12 or abs(area - 9 * cell * cell) > cell * cell // 8:
        return None
    x0, y0 = int(xs.min()), int(ys.min())
    width = int(xs.max()) - x0 + 1
    height = int(ys.max()) - y0 + 1
    if width % cell or height % cell or width // cell > 5 or height // cell > 5:
        return None
    cells: set[tuple[int, int]] = set()
    for gy in range(height // cell):
        for gx in range(width // cell):
            block = mask[y0 + gy * cell : y0 + (gy + 1) * cell, x0 + gx * cell : x0 + (gx + 1) * cell]
            fraction = float(block.mean())
            if fraction > 0.96:
                cells.add((gx, gy))
            elif fraction > 0.04:
                return None
    return cells if len(cells) == 9 else None


def _has_four_way(cells: set[tuple[int, int]]) -> bool:
    for x, y in cells:
        if all(point in cells for point in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1))):
            return True
    return False


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    winners = []
    for name, rgb in _PALETTE.items():
        delta = np.linalg.norm(array - np.asarray(rgb, dtype=np.int16), axis=2)
        cells = _occupancy(delta < 8.0)
        if cells is not None and _has_four_way(cells):
            winners.append(name)
    if len(winners) != 1:
        raise ValueError("image does not contain one unambiguous four-way patch")
    return winners[0]
