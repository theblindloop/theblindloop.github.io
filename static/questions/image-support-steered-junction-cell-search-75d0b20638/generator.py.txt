"""Deterministic latent sampler and raster renderer for local junction search."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

PALETTE = {
    "red": (220, 62, 66),
    "blue": (48, 112, 205),
    "green": (42, 153, 91),
    "purple": (145, 78, 190),
}
COLOR_NAMES = tuple(PALETTE)
BACKGROUND = (250, 249, 246)


def _normalize(cells: Iterable[tuple[int, int]]) -> list[list[int]]:
    points = set(cells)
    x0 = min(x for x, _ in points)
    y0 = min(y for _, y in points)
    return [[x - x0, y - y0] for x, y in sorted(points, key=lambda p: (p[1], p[0]))]


def _degree(cells: Iterable[Iterable[int]], point: tuple[int, int]) -> int:
    occupied = {tuple(p) for p in cells}
    x, y = point
    return sum((x + dx, y + dy) in occupied for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)))


def _max_degree(cells: Iterable[Iterable[int]]) -> int:
    occupied = {tuple(p) for p in cells}
    return max(_degree(occupied, p) for p in occupied)


def _grow_polyomino(rng: random.Random, *, require_four: bool) -> list[list[int]]:
    """Grow a compact nine-cell polyomino with or without one four-way cell."""
    for _ in range(2000):
        if require_four:
            cells = {(1, 1), (0, 1), (2, 1), (1, 0), (1, 2)}
        else:
            cells = {(0, 0)}
        while len(cells) < 9:
            frontier = {
                (x + dx, y + dy)
                for x, y in cells
                for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1))
                if (x + dx, y + dy) not in cells
            }
            choices = list(frontier)
            rng.shuffle(choices)
            added = False
            for point in choices:
                trial = cells | {point}
                width = max(x for x, _ in trial) - min(x for x, _ in trial) + 1
                height = max(y for _, y in trial) - min(y for _, y in trial) + 1
                degrees = [_degree(trial, p) for p in trial]
                if width > 5 or height > 5:
                    continue
                if require_four and degrees.count(4) != 1:
                    continue
                if not require_four and max(degrees) >= 4:
                    continue
                cells = trial
                added = True
                break
            if not added:
                break
        if len(cells) != 9:
            continue
        normalized = _normalize(cells)
        xs = [p[0] for p in normalized]
        ys = [p[1] for p in normalized]
        # Every candidate has exactly the same 4x4 extent and 9/16 fill ratio.
        # Thus neither total ink nor bounding-box compactness can reveal the winner.
        if max(xs) + 1 != 4 or max(ys) + 1 != 4:
            continue
        degree4 = sum(_degree(normalized, tuple(p)) == 4 for p in normalized)
        if degree4 == (1 if require_four else 0):
            return normalized
    raise RuntimeError("could not sample a suitable polyomino")


def _rotate(cells: list[list[int]], turns: int) -> list[list[int]]:
    points = [tuple(p) for p in cells]
    for _ in range(turns % 4):
        points = [(-y, x) for x, y in points]
    return _normalize(points)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target_color = COLOR_NAMES[int(seed) % 4]
    target_slot = (int(seed) // 4) % 4
    colors = list(COLOR_NAMES)
    rng.shuffle(colors)
    colors.remove(target_color)
    colors.insert(target_slot, target_color)

    patches = []
    for slot in range(4):
        shape = _grow_polyomino(rng, require_four=(slot == target_slot))
        patches.append(_rotate(shape, rng.randrange(4)))

    base = [(132, 132), (380, 132), (132, 380), (380, 380)]
    anchors = [[x + rng.randint(-18, 18), y + rng.randint(-18, 18)] for x, y in base]
    return {
        "patches": patches,
        "colors": colors,
        "anchors": anchors,
        "cell_size": 18 + 2 * rng.randrange(2),
        "canvas_size": 512,
    }


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas_size"])
    image = Image.new("RGB", (size, size), BACKGROUND)
    draw = ImageDraw.Draw(image)
    cell_size = int(scene["cell_size"])
    for cells, color_name, anchor in zip(scene["patches"], scene["colors"], scene["anchors"], strict=True):
        width = (max(int(p[0]) for p in cells) + 1) * cell_size
        height = (max(int(p[1]) for p in cells) + 1) * cell_size
        x0 = int(anchor[0]) - width // 2
        y0 = int(anchor[1]) - height // 2
        color = PALETTE[str(color_name)]
        for px, py in cells:
            left = x0 + int(px) * cell_size
            top = y0 + int(py) * cell_size
            draw.rectangle((left, top, left + cell_size - 1, top + cell_size - 1), fill=color)
    return image


def analytic_gold(scene: dict) -> str:
    winners = [str(color) for cells, color in zip(scene["patches"], scene["colors"], strict=True) if _max_degree(cells) == 4]
    if len(winners) != 1:
        raise ValueError("scene does not have a unique degree-four patch")
    return winners[0]


def margin(scene: dict) -> float:
    scores = sorted((_max_degree(cells) for cells in scene["patches"]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    try:
        scores = [_max_degree(cells) for cells in scene["patches"]]
        return scores.count(4) != 1 or sorted(scores, reverse=True)[0] - sorted(scores, reverse=True)[1] < 1
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["patches"] = list(reversed(scene["patches"]))
    twin["colors"] = list(reversed(scene["colors"]))
    twin["anchors"] = list(reversed(scene["anchors"]))
    return [("reverse_parallel_enumeration", twin)]
