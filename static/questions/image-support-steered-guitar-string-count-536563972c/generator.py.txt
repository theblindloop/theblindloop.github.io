"""Deterministic renderer for a paired-endpoint guitar setup card."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

SIZE = 512
SCALE = 3
SLOT_COUNT = 12
SLOT_X0 = 125
SLOT_STEP = 24
TOP_Y = 168
BOTTOM_Y = 344
BLUE = (25, 92, 178)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    count = 4 + (int(seed) % 5)
    slots = list(range(SLOT_COUNT))
    rng.shuffle(slots)
    complete = sorted(slots[:count])
    remaining = slots[count:]
    unpaired_count = 16 - 2 * count
    unpaired = remaining[:unpaired_count]
    rng.shuffle(unpaired)
    split = rng.randrange(unpaired_count + 1) if unpaired_count else 0
    top_only = sorted(unpaired[:split])
    bottom_only = sorted(unpaired[split:])
    return {
        "complete_slots": complete,
        "top_only_slots": top_only,
        "bottom_only_slots": bottom_only,
        "x_shift": rng.randint(-5, 5),
        "y_shift": rng.randint(-5, 5),
        "neck_style": rng.choice(["maple", "rosewood"]),
        "fret_offset": rng.randint(-3, 3),
    }


def _xy(scene: dict, slot: int, y: int) -> tuple[int, int]:
    return SLOT_X0 + SLOT_STEP * int(slot) + int(scene["x_shift"]), y + int(scene["y_shift"])


def render(scene: dict) -> Image.Image:
    complete = {int(v) for v in scene["complete_slots"]}
    top_only = {int(v) for v in scene["top_only_slots"]}
    bottom_only = {int(v) for v in scene["bottom_only_slots"]}
    x_shift = int(scene["x_shift"])
    y_shift = int(scene["y_shift"])
    neck_style = str(scene["neck_style"])
    fret_offset = int(scene["fret_offset"])

    image = Image.new("RGB", (SIZE * SCALE, SIZE * SCALE), (248, 247, 242))
    draw = ImageDraw.Draw(image)

    def box(coords, **kwargs):
        draw.rectangle(tuple(int(v * SCALE) for v in coords), **kwargs)

    def ellipse(coords, **kwargs):
        draw.ellipse(tuple(int(v * SCALE) for v in coords), **kwargs)

    def line(points, **kwargs):
        draw.line([(int(x * SCALE), int(y * SCALE)) for x, y in points], **kwargs)

    # Decorative instrument silhouette, deliberately neutral to the answer.
    body = (205, 145, 92) if neck_style == "maple" else (128, 82, 59)
    body_edge = (91, 62, 45)
    ellipse((92, 326, 420, 566), fill=body, outline=body_edge, width=4 * SCALE)
    box((104, 88, 408, 386), fill=(229, 205, 158), outline=body_edge, width=4 * SCALE)
    box((112, 54, 400, 112), fill=body, outline=body_edge, width=4 * SCALE)

    # Faint candidate courses and horizontal frets are context, not counted ink.
    for slot in range(SLOT_COUNT):
        x = SLOT_X0 + SLOT_STEP * slot + x_shift
        line([(x, 136 + y_shift), (x, 376 + y_shift)], fill=(190, 190, 184), width=2 * SCALE)
    for y in range(202 + fret_offset + y_shift, 326 + y_shift, 31):
        line([(110, y), (406, y)], fill=(139, 132, 120), width=3 * SCALE)

    # Visible endpoint-bank rails.
    line([(108, TOP_Y + y_shift), (404, TOP_Y + y_shift)], fill=(70, 70, 68), width=7 * SCALE)
    line([(108, BOTTOM_Y + y_shift), (404, BOTTOM_Y + y_shift)], fill=(70, 70, 68), width=7 * SCALE)

    # Gold tuning pegs outside the decisive banks.
    gold = (220, 161, 42)
    for slot in (0, 2, 4, 7, 9, 11):
        x, _ = _xy(scene, slot, TOP_Y)
        side_y = 76 + (slot % 3) * 13
        ellipse((x - 7, side_y - 7, x + 7, side_y + 7), fill=gold, outline=(116, 82, 24), width=2 * SCALE)

    # Blue target markers are drawn last and remain disconnected circular components.
    radius = 10
    top_blue = complete | top_only
    bottom_blue = complete | bottom_only
    for slot in sorted(top_blue):
        x, y = _xy(scene, slot, TOP_Y)
        ellipse((x - radius, y - radius, x + radius, y + radius), fill=BLUE, outline=(12, 48, 102), width=2 * SCALE)
    for slot in sorted(bottom_blue):
        x, y = _xy(scene, slot, BOTTOM_Y)
        ellipse((x - radius, y - radius, x + radius, y + radius), fill=BLUE, outline=(12, 48, 102), width=2 * SCALE)

    return image.resize((SIZE, SIZE), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(len(scene["complete_slots"]))


def margin(scene: dict) -> float:
    # Paired centroids differ by 0 px; nearest distinct slot differs by 24 px.
    # Against an 8 px pairing tolerance, the runner-up gap is at least 8 px.
    _ = scene["complete_slots"]
    return 8.0


def is_quarantined(scene: dict) -> bool:
    complete = set(scene["complete_slots"])
    top = set(scene["top_only_slots"])
    bottom = set(scene["bottom_only_slots"])
    valid_slots = set(range(SLOT_COUNT))
    return bool(
        not complete <= valid_slots
        or not top <= valid_slots
        or not bottom <= valid_slots
        or complete & top
        or complete & bottom
        or top & bottom
        or 2 * len(complete) + len(top) + len(bottom) != 16
        or len(complete) not in {4, 5, 6, 7, 8}
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    for key in ("complete_slots", "top_only_slots", "bottom_only_slots"):
        twin[key] = list(reversed(twin[key]))
    return [("reverse_slot_storage_order", twin)]
