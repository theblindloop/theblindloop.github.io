"""Pixel-only inverse arm for aligned blue endpoint pairs."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int, float, float]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        ys = [p[0] for p in points]
        xs = [p[1] for p in points]
        found.append((len(points), min(xs), min(ys), max(xs), max(ys), float(np.mean(xs)), float(np.mean(ys))))
    return found


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = rgb[..., 0], rgb[..., 1], rgb[..., 2]
    mask = (blue >= 112) & (blue - red >= 45) & (blue - green >= 35) & (red <= 90)
    markers = []
    for area, x0, y0, x1, y1, cx, cy in _components(mask):
        width = x1 - x0 + 1
        height = y1 - y0 + 1
        if 175 <= area <= 450 and 15 <= width <= 25 and 15 <= height <= 25:
            markers.append((cx, cy))
    if not 8 <= len(markers) <= 20:
        return "abstain"
    ys = sorted(cy for _, cy in markers)
    gaps = [(ys[index + 1] - ys[index], index) for index in range(len(ys) - 1)]
    largest_gap, gap_index = max(gaps)
    if largest_gap < 80:
        return "abstain"
    split = (ys[gap_index] + ys[gap_index + 1]) / 2.0
    top = sorted(cx for cx, cy in markers if cy < split)
    bottom = sorted(cx for cx, cy in markers if cy > split)
    if not top or not bottom or len(top) + len(bottom) != len(markers):
        return "abstain"
    used = set()
    pairs = 0
    for tx in top:
        choices = [(abs(tx - bx), index) for index, bx in enumerate(bottom) if index not in used]
        if choices:
            distance, index = min(choices)
            if distance <= 8.0:
                used.add(index)
                pairs += 1
    return str(pairs) if 0 <= pairs <= 12 else "abstain"
