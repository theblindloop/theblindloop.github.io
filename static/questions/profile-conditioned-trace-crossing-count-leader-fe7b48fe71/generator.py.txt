"""Deterministic Pillow world for adjacent baseline-crossing aggregation."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

COLOR_RGB = {
    "blue": (42, 103, 210),
    "green": (18, 150, 91),
    "purple": (168, 64, 188),
}
ORANGE = (235, 126, 34)
BACKGROUND = (250, 250, 248)
PLOT = (255, 255, 255)
GRID = (222, 225, 229)
INK = (42, 47, 54)
N_CHECKPOINTS = 7


def _sign_sequence(rng: random.Random, transitions: int) -> list[int]:
    change_after = set(rng.sample(range(N_CHECKPOINTS - 1), transitions))
    sign = rng.choice((-1, 1))
    result = [sign]
    for index in range(N_CHECKPOINTS - 1):
        if index in change_after:
            sign *= -1
        result.append(sign)
    return result


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    width, height = 640, 420
    baseline_y = rng.choice((182, 196, 210, 224))
    checkpoint_xs = [94 + 76 * index for index in range(N_CHECKPOINTS)]
    colors = list(COLOR_RGB)

    # Winner identity cycles independently of randomized geometry. Counts are
    # separated by one discrete crossing, the smallest non-tied runner-up gap.
    winner = int(seed) % len(colors)
    other = [index for index in range(len(colors)) if index != winner]
    rng.shuffle(other)
    transition_counts = [0, 0, 0]
    transition_counts[winner] = rng.choice((5, 6))
    transition_counts[other[0]] = transition_counts[winner] - 1
    transition_counts[other[1]] = rng.choice((1, 2))

    max_offset = min(baseline_y - 68, 326 - baseline_y)
    signs_by_trace = [_sign_sequence(rng, transitions) for transitions in transition_counts]
    values = [[0] * N_CHECKPOINTS for _ in colors]
    # Keep differently colored markers visibly separate at every checkpoint so
    # draw order cannot erase the evidence used by the inverse arm. Slot order
    # is randomized independently at each checkpoint to avoid a height proxy.
    slot_sets = {1: [rng.randint(42, int(max_offset) - 4)], 2: [40, 88], 3: [36, 68, 100]}
    for checkpoint in range(N_CHECKPOINTS):
        for sign in (-1, 1):
            members = [index for index in range(3) if signs_by_trace[index][checkpoint] == sign]
            if not members:
                continue
            slots = slot_sets[len(members)].copy()
            rng.shuffle(slots)
            for index, offset in zip(members, slots, strict=True):
                values[index][checkpoint] = sign * offset

    return {
        "width": width,
        "height": height,
        "baseline_y": baseline_y,
        "checkpoint_xs": checkpoint_xs,
        "colors": colors,
        "values": values,
    }


def _crossing_count(values: list[int]) -> int:
    return sum((a < 0 < b) or (b < 0 < a) for a, b in zip(values, values[1:]))


def analytic_gold(scene: dict[str, Any]) -> str:
    colors = scene["colors"]
    values = scene["values"]
    counts = [_crossing_count(row) for row in values]
    best = max(counts)
    if counts.count(best) != 1:
        return "abstain"
    return str(colors[counts.index(best)])


def margin(scene: dict[str, Any]) -> float:
    counts = sorted((_crossing_count(row) for row in scene["values"]), reverse=True)
    return float(counts[0] - counts[1])


def is_quarantined(scene: dict[str, Any]) -> bool:
    values = scene["values"]
    counts = [_crossing_count(row) for row in values]
    return counts.count(max(counts)) != 1 or min(abs(value) for row in values for value in row) < 18


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []


def render(scene: dict[str, Any]) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    baseline_y = int(scene["baseline_y"])
    checkpoint_xs = [int(value) for value in scene["checkpoint_xs"]]
    colors = [str(value) for value in scene["colors"]]
    values = [[int(value) for value in row] for row in scene["values"]]

    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    try:
        title_font = ImageFont.truetype("DejaVuSans-Bold.ttf", 15 * scale)
        label_font = ImageFont.truetype("DejaVuSans.ttf", 12 * scale)
    except OSError:
        title_font = label_font = ImageFont.load_default()
    plot_left, plot_top, plot_right, plot_bottom = 58, 48, 574, 338
    box = tuple(value * scale for value in (plot_left, plot_top, plot_right, plot_bottom))
    draw.rectangle(box, fill=PLOT, outline=GRID, width=2 * scale)

    for index, x in enumerate(checkpoint_xs, start=1):
        draw.line((x * scale, plot_top * scale, x * scale, plot_bottom * scale), fill=GRID, width=1 * scale)
        draw.text((x * scale - 4 * scale, 348 * scale), str(index), fill=INK, font=label_font)

    # The boundary is fully observable and drawn before traces, so colored trace
    # pixels remain measurable even at crossings.
    dash = 15
    x = plot_left + 2
    while x < plot_right - 2:
        draw.line(
            (x * scale, baseline_y * scale, min(x + dash, plot_right - 2) * scale, baseline_y * scale),
            fill=ORANGE,
            width=4 * scale,
        )
        x += dash + 10

    for color, row in zip(colors, values, strict=True):
        rgb = COLOR_RGB[color]
        points = [(x * scale, (baseline_y - value) * scale) for x, value in zip(checkpoint_xs, row, strict=True)]
        draw.line(points, fill=rgb, width=5 * scale, joint="curve")
        radius = 9 * scale
        for px, py in points:
            draw.ellipse((px - radius, py - radius, px + radius, py + radius), fill=rgb, outline=PLOT, width=1 * scale)

    draw.text((58 * scale, 16 * scale), "Baseline crossings at adjacent checkpoints", fill=INK, font=title_font)
    legend_x = 78
    for color in colors:
        rgb = COLOR_RGB[color]
        draw.line((legend_x * scale, 382 * scale, (legend_x + 24) * scale, 382 * scale), fill=rgb, width=5 * scale)
        draw.text(((legend_x + 29) * scale, 373 * scale), color, fill=INK, font=label_font)
        legend_x += 150

    return image.resize((width, height), Image.Resampling.LANCZOS)
