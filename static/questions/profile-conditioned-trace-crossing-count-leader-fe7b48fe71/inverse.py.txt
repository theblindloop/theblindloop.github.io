"""Independent inverse arm: only the rasterized PNG enters this module."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_COLORS = {
    "blue": (42, 103, 210),
    "green": (18, 150, 91),
    "purple": (168, 64, 188),
}
_ORANGE = (235, 126, 34)


def _near(arr: np.ndarray, rgb: tuple[int, int, int], tolerance: float = 48.0) -> np.ndarray:
    target = np.asarray(rgb, dtype=np.int32)
    delta = arr.astype(np.int32) - target
    return np.sum(delta * delta, axis=2, dtype=np.int32) <= tolerance * tolerance


def _erode(mask: np.ndarray, iterations: int = 3) -> np.ndarray:
    current = mask.copy()
    for _ in range(iterations):
        padded = np.pad(current, 1, constant_values=False)
        current = np.logical_and.reduce(
            [padded[dy : dy + mask.shape[0], dx : dx + mask.shape[1]] for dy in range(3) for dx in range(3)]
        )
    return current


def _components(mask: np.ndarray) -> list[tuple[int, float, float]]:
    work = mask.copy()
    height, width = work.shape
    output: list[tuple[int, float, float]] = []
    for y0, x0 in zip(*np.nonzero(work), strict=True):
        if not work[y0, x0]:
            continue
        queue = deque([(int(y0), int(x0))])
        work[y0, x0] = False
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                        work[ny, nx] = False
                        queue.append((ny, nx))
        if len(xs) >= 18:
            output.append((len(xs), float(np.mean(xs)), float(np.mean(ys))))
    return output


def _baseline_y(arr: np.ndarray) -> float:
    mask = _near(arr, _ORANGE, 45.0)
    ys, xs = np.nonzero(mask)
    if len(xs) < 120:
        raise ValueError("orange dashed baseline is missing")
    return float(np.median(ys))


def _marker_centers(arr: np.ndarray, rgb: tuple[int, int, int]) -> list[tuple[float, float]]:
    mask = _near(arr, rgb, 48.0)
    # Three erosions remove the five-pixel connecting stroke while retaining the
    # interiors of nine-pixel circular markers. Legend remnants are filtered by
    # requiring the seven horizontally ordered marker components in the plot.
    components = _components(_erode(mask, 3))
    plot = [(x, y) for area, x, y in components if area >= 20 and 45 < x < 590 and 45 < y < 340]
    plot.sort()
    if len(plot) != 7:
        raise ValueError(f"expected seven circular checkpoints, recovered {len(plot)}")
    if min(b[0] - a[0] for a, b in zip(plot, plot[1:])) < 35:
        raise ValueError("checkpoint order is not resolvable")
    return plot


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    baseline = _baseline_y(arr)
    counts: dict[str, int] = {}
    for name, rgb in _COLORS.items():
        centers = _marker_centers(arr, rgb)
        signed = [baseline - y for _, y in centers]
        if min(abs(value) for value in signed) < 12:
            raise ValueError("checkpoint too close to baseline")
        counts[name] = sum((a < 0 < b) or (b < 0 < a) for a, b in zip(signed, signed[1:]))
    best = max(counts.values())
    winners = [name for name, value in counts.items() if value == best]
    if len(winners) != 1:
        raise ValueError("top crossing count is tied")
    return winners[0]
