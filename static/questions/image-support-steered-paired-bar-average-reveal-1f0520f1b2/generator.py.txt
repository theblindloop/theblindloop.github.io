"""Deterministic sampler and renderer for Paired Bar Average Reveal."""

from __future__ import annotations

import copy
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 768
HEIGHT = 512
BACKGROUND = (250, 248, 242)
DARK = (31, 38, 51)
GRID = (205, 209, 207)
DECOR = (177, 178, 173)

PALETTES = (
    ((20, 170, 190), (230, 165, 30)),
    ((126, 82, 188), (42, 157, 91)),
    ((210, 72, 82), (54, 108, 210)),
)

SYMBOLS = ("arrow", "diamond", "funnel")
TEMPLATES: dict[str, tuple[int, ...]] = {
    "arrow": (2, 4, 6, 8, 4, 4, 4),
    "diamond": (2, 4, 6, 8, 6, 4, 2),
    "funnel": (8, 6, 4, 2, 2, 4, 6),
}
NEAR_VARIANTS: dict[str, tuple[int, ...]] = {
    "arrow": (2, 4, 6, 8, 5, 4, 4),
    "diamond": (2, 4, 6, 8, 5, 4, 2),
    "funnel": TEMPLATES["funnel"],
}
AMBIGUOUS_PROFILE = (2, 4, 6, 8, 5, 4, 3)


def _centered_row(length: int) -> tuple[int, ...]:
    length = int(length)
    if not 1 <= length <= 9:
        raise ValueError("bar length must be between one and nine cells")
    start = (9 - length) // 2
    return tuple(int(start <= column < start + length) for column in range(9))


def _distances(profile: Iterable[int]) -> dict[str, int]:
    observed = tuple(_centered_row(int(length)) for length in profile)
    if len(observed) != 7:
        raise ValueError("exactly seven averaged rows are required")
    output: dict[str, int] = {}
    for name, reference_profile in TEMPLATES.items():
        reference = tuple(_centered_row(length) for length in reference_profile)
        output[name] = sum(
            int(a != b)
            for observed_row, reference_row in zip(observed, reference, strict=True)
            for a, b in zip(observed_row, reference_row, strict=True)
        )
    return output


def _classification(lengths_a: Iterable[int], lengths_b: Iterable[int]) -> tuple[str, float]:
    left = tuple(int(value) for value in lengths_a)
    right = tuple(int(value) for value in lengths_b)
    if len(left) != 7 or len(right) != 7:
        raise ValueError("both racks must contain exactly seven rows")
    profile = []
    for first, second in zip(left, right, strict=True):
        if (first + second) % 2:
            raise ValueError("the declared bar average is not a whole cell")
        profile.append((first + second) // 2)
    ranked = sorted((distance, name) for name, distance in _distances(profile).items())
    if ranked[0][0] == ranked[1][0]:
        return "ambiguous", 0.0
    return ranked[0][1], float(ranked[1][0] - ranked[0][0])


def sample_scene(seed: int) -> dict:
    """Sample balanced labels with answer-independent rendering nuisances."""

    seed = int(seed)
    rng = random.Random(seed)
    target = SYMBOLS[seed % len(SYMBOLS)]
    if seed % 11 == 0:
        profile = AMBIGUOUS_PROFILE
    elif seed % 7 == 0:
        profile = NEAR_VARIANTS[target]
    else:
        profile = TEMPLATES[target]

    signs = [1, -1, 1, -1, 1, -1, rng.choice((-1, 1))]
    rng.shuffle(signs)
    lengths_a = [int(length + sign) for length, sign in zip(profile, signs, strict=True)]
    lengths_b = [int(length - sign) for length, sign in zip(profile, signs, strict=True)]
    panel_order = ["a", "b"]
    rng.shuffle(panel_order)
    anchors = ((82, 92), (686, 86), (89, 419), (681, 427), (385, 95), (384, 421))
    decorations = [
        [x + rng.randint(-5, 5), y + rng.randint(-5, 5), rng.choice((3, 4))]
        for x, y in anchors
    ]
    return {
        "bar_lengths_a": lengths_a,
        "bar_lengths_b": lengths_b,
        "flip_a": bool(rng.getrandbits(1)),
        "flip_b": bool(rng.getrandbits(1)),
        "panel_order": panel_order,
        "palette": rng.randrange(len(PALETTES)),
        "scale": rng.choice((9, 10)),
        "x_shift": rng.randint(-4, 4),
        "y_shift": rng.randint(-6, 6),
        "decorations": decorations,
    }


def _draw_card(
    draw: ImageDraw.ImageDraw,
    *,
    center_x: int,
    y0: int,
    scale: int,
    color: tuple[int, int, int],
    lengths: list[int],
    flipped: bool,
) -> None:
    width = 11 * scale
    height = 9 * scale
    x0 = int(center_x - width // 2)
    x1 = x0 + width
    y1 = y0 + height
    stroke = 3
    draw.rectangle((x0, y0, x1, y1), outline=color, width=stroke)
    notch_x = x0 + width // 2
    if flipped:
        draw.polygon(
            ((notch_x, y1 + 8), (notch_x - 7, y1 - 2), (notch_x + 7, y1 - 2)),
            fill=color,
        )
    else:
        draw.polygon(
            ((notch_x, y0 - 8), (notch_x - 7, y0 + 2), (notch_x + 7, y0 + 2)),
            fill=color,
        )

    rack_left = x0 + scale // 2
    rack_right = x0 + 19 * scale // 2
    rack_top = y0 + scale // 2
    rack_bottom = y0 + 15 * scale // 2
    for column in range(10):
        x = rack_left + column * scale
        draw.line((x, rack_top, x, rack_bottom), fill=GRID, width=1)
    for row in range(8):
        y = rack_top + row * scale
        draw.line((rack_left, y, rack_right, y), fill=GRID, width=1)

    radius = 3
    for logical_row, length in enumerate(lengths):
        physical_row = 6 - logical_row if flipped else logical_row
        y = y0 + (1 + physical_row) * scale
        selected = range(9 - int(length), 9) if flipped else range(int(length))
        for column in selected:
            x = x0 + (1 + int(column)) * scale
            draw.rectangle((x - radius, y - radius, x + radius, y + radius), fill=DARK)


def render(scene: dict) -> Image.Image:
    """Render two detached racks; the averaged output is never drawn."""

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x, y, radius in scene["decorations"]:
        radius = int(radius)
        draw.ellipse((int(x) - radius, int(y) - radius, int(x) + radius, int(y) + radius), fill=DECOR)

    scale = int(scene["scale"])
    y0 = 215 + int(scene["y_shift"])
    centers = (205 + int(scene["x_shift"]), 560 + int(scene["x_shift"]))
    colors = PALETTES[int(scene["palette"])]
    payload = {
        "a": (scene["bar_lengths_a"], bool(scene["flip_a"]), colors[0]),
        "b": (scene["bar_lengths_b"], bool(scene["flip_b"]), colors[1]),
    }
    for center_x, name in zip(centers, scene["panel_order"], strict=True):
        lengths, flipped, color = payload[str(name)]
        _draw_card(
            draw,
            center_x=int(center_x),
            y0=y0,
            scale=scale,
            color=color,
            lengths=[int(value) for value in lengths],
            flipped=bool(flipped),
        )
    return image


def analytic_gold(scene: dict) -> str:
    return _classification(scene["bar_lengths_a"], scene["bar_lengths_b"])[0]


def margin(scene: dict) -> float:
    return _classification(scene["bar_lengths_a"], scene["bar_lengths_b"])[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["decorations"] = list(reversed(twin["decorations"]))
    return [("reverse_nonoverlapping_decoration_draw_order", twin)]
