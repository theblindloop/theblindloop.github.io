"""Independent pixel-only inverse arm for the state half-count world.

Receives only the final PNG. From the raster alone it:
  1. labels pixels against the fixed palette and recovers the 4 slot columns
     from the panel-box border edges;
  2. reads the ring in the first panel, samples the tracked disk's color at the
     ring center, and identifies that token by its unique color;
  3. for each ordered panel, locates which column holds the tracked color and
     whether that column is a left-half slot (slot 1 or 2);
  4. returns the total number of panels (0..4) in which the tracked token sits
     in the left half.

No latent scene, renderer, prompt, or gold code is accessed.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

BG = (255, 255, 255)
BORDER = (95, 95, 95)
TOKEN_COLORS = [
    (210, 40, 40),
    (40, 170, 40),
    (50, 80, 220),
    (240, 160, 30),
]
RING = (210, 0, 210)

PALETTE = [BG, BORDER, *TOKEN_COLORS, RING]
_BG, _BRD, _TC0, _TC1, _TC2, _TC3, _RING = range(7)
TOKEN_INDICES = [_TC0, _TC1, _TC2, _TC3]

EXPECTED_COLUMNS = 4
SLOT_H = 90
PANEL_GAP = 40
COUNT_LABELS = {0: "0", 1: "1", 2: "2", 3: "3", 4: "4"}


class UnreadableImageError(ValueError):
    """The raster does not expose the expected panel structure."""


def _label(arr: np.ndarray) -> np.ndarray:
    dists = np.stack(
        [np.linalg.norm(arr - np.array(c, dtype=np.float32), axis=-1) for c in PALETTE],
        axis=-1,
    )
    return dists.argmin(axis=-1)


def _cluster(sorted_vals: list[int], tol: int) -> list[int]:
    if not sorted_vals:
        return []
    out = [[sorted_vals[0]]]
    for v in sorted_vals[1:]:
        if v - out[-1][-1] > tol:
            out.append([v])
        else:
            out[-1].append(v)
    return [r[len(r) // 2] for r in out]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label(arr)
    h, w = labels.shape

    border_xs = np.nonzero(labels == _BRD)[1]
    if border_xs.size == 0:
        raise UnreadableImageError("no panel borders found")

    counts = np.bincount(border_xs, minlength=w)
    edge_cols = [int(x) for x in range(w) if counts[x] > 40]
    edges = _cluster(sorted(set(edge_cols)), 6)
    if len(edges) < 2 * EXPECTED_COLUMNS:
        raise UnreadableImageError(
            f"expected {2 * EXPECTED_COLUMNS} box side edges, found {len(edges)}"
        )
    edge = edges[: 2 * EXPECTED_COLUMNS]
    col_centers = [(edge[2 * i] + edge[2 * i + 1]) // 2 for i in range(EXPECTED_COLUMNS)]

    border_ys = np.nonzero(labels == _BRD)[0]
    top_edge = int(border_ys.min())

    ring_pts = np.nonzero(labels == _RING)
    if ring_pts[0].size == 0:
        raise UnreadableImageError("no magenta ring found")
    ring_cx = sum(int(x) for x in ring_pts[1]) // len(ring_pts[1])
    ring_cy = sum(int(y) for y in ring_pts[0]) // len(ring_pts[0])
    tracked_label = int(labels[ring_cy, ring_cx])
    if tracked_label not in TOKEN_INDICES:
        raise UnreadableImageError("ring does not enclose a detectable token color")

    count = 0
    for k in range(4):
        panel_cy = top_edge + k * (SLOT_H + PANEL_GAP) + SLOT_H // 2
        token_col = None
        for i in range(EXPECTED_COLUMNS):
            sample = int(labels[panel_cy, col_centers[i]])
            if sample == tracked_label:
                token_col = i
                break
        if token_col is None:
            raise UnreadableImageError(f"panel {k}: tracked token color not found")
        if token_col in (0, 1):
            count += 1
    return COUNT_LABELS[count]
