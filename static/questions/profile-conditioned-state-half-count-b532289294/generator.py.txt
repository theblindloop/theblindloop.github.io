"""State half-count: tally the left-half panels of a color-tracked token.

The image is a vertical timeline of PANELS=4 panels. Each panel shows one row of
4 slot boxes; every slot holds one token disk. The four token disks use a fixed
four-color palette as a permutation in each panel, so token identity is carried
uniquely by color across the whole image. A magenta ring in the first (top)
panel marks the token to track. Reading panels top to bottom and following that
color, the decision is the integer count of panels in which the tracked token
occupies a LEFT-half slot (slot 1 or slot 2).
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

N = 4
PANELS = 4

LEFT_M = 60
TOP_M = 70
SLOT_W = 110
SLOT_H = 90
CELL_GAP = 30
PANEL_GAP = 40
TOKEN_R = 20
RING_R = TOKEN_R + 8

WIDTH = LEFT_M + N * SLOT_W + (N - 1) * CELL_GAP + LEFT_M
HEIGHT = TOP_M + PANELS * SLOT_H + (PANELS - 1) * PANEL_GAP + 40

BG = (255, 255, 255)
BORDER = (95, 95, 95)
TOKEN_COLORS = [
    (210, 40, 40),
    (40, 170, 40),
    (50, 80, 220),
    (240, 160, 30),
]
RING = (210, 0, 210)

COUNT_LABELS = {0: "0", 1: "1", 2: "2", 3: "3", 4: "4"}
LEFT_HALF_SLOTS = {0, 1}


def _col_center(i: int) -> int:
    return LEFT_M + SLOT_W // 2 + i * (SLOT_W + CELL_GAP)


def _col_x0(i: int) -> int:
    return _col_center(i) - SLOT_W // 2


def _panel_top(k: int) -> int:
    return TOP_M + k * (SLOT_H + PANEL_GAP)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    start = rng.randint(0, N - 1)
    panels = []
    for _ in range(PANELS):
        perm = [0, 1, 2, 3]
        rng.shuffle(perm)
        panels.append(perm)
    return {"panels": panels, "start": start}


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)
    panels = scene["panels"]
    start = scene["start"]

    for k in range(PANELS):
        top = _panel_top(k)
        for i in range(N):
            x0 = _col_x0(i)
            draw.rectangle([x0, top, x0 + SLOT_W, top + SLOT_H], outline=BORDER, width=3)
            cx = _col_center(i)
            cy = top + SLOT_H // 2
            color = TOKEN_COLORS[panels[k][i]]
            draw.ellipse([cx - TOKEN_R, cy - TOKEN_R, cx + TOKEN_R, cy + TOKEN_R], fill=color)
        if k == 0:
            cx = _col_center(start)
            cy = top + SLOT_H // 2
            draw.ellipse(
                [cx - RING_R, cy - RING_R, cx + RING_R, cy + RING_R], outline=RING, width=5
            )
    return img


def analytic_gold(scene: dict) -> str:
    panels = scene["panels"]
    tracked = panels[0][scene["start"]]
    count = 0
    for k in range(PANELS):
        slot = panels[k].index(tracked)
        if slot in LEFT_HALF_SLOTS:
            count += 1
    return COUNT_LABELS[count]


def margin(scene: dict) -> float:
    return 10.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    """No nontrivial render-invariant latent symmetry: ``panels`` fixes every
    drawn disk color and ``start`` fixes which first-panel disk is ringed, so any
    permutation of colors or panels changes the raster; no alias exists."""
    return []
