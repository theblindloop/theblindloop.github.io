"""Independent fixed-layout pixel inverse for the register-exchange walk."""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH, HEIGHT = 1040, 800
N, CELL = 5, 90
GRID_X, GRID_Y = 295, 198
QUEUE_XS, QUEUE_Y = (383, 463, 543, 623), 105
BACKGROUND = (247, 246, 241)
GOLD = (244, 190, 51)
BLUE = (54, 128, 194)
RED = (211, 46, 57)
VIOLET = (124, 58, 188)
DIRS = ((-1, 0), (0, 1), (1, 0), (0, -1))


def _mask(arr: np.ndarray, color: tuple[int, int, int], tolerance: int = 5) -> np.ndarray:
    return np.all(np.abs(arr.astype(np.int16) - np.array(color, dtype=np.int16)) <= tolerance, axis=2)


def _near(pixel: np.ndarray, color: tuple[int, int, int], tolerance: int = 5) -> bool:
    return bool(np.max(np.abs(pixel.astype(np.int16) - np.array(color, dtype=np.int16))) <= tolerance)


def _token(arr: np.ndarray, color: tuple[int, int, int]) -> tuple[list[int], int] | None:
    crop = arr[GRID_Y:GRID_Y + N * CELL, GRID_X:GRID_X + N * CELL]
    mask = _mask(crop, color)
    hits = []
    for row in range(N):
        for col in range(N):
            patch = mask[row * CELL + 8:row * CELL + 70, col * CELL + 8:col * CELL + 70].copy()
            patch[47:, 47:] = False  # exclude the fixed bottom-right register badge
            if int(patch.sum()) > 300:
                hits.append((row, col))
    if len(hits) != 1:
        return None
    row, col = hits[0]
    patch = mask[row * CELL + 5:row * CELL + 73, col * CELL + 5:col * CELL + 73].copy()
    patch[50:, 50:] = False
    ys, xs = np.nonzero(patch)
    if len(xs) < 300:
        return None
    cx, cy = 34, 34
    extents = (cy - int(ys.min()), int(xs.max()) - cx, int(ys.max()) - cy, cx - int(xs.min()))
    best = max(extents)
    if extents.count(best) != 1 or best < 20:
        return None
    return [row, col], extents.index(best)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.shape != (HEIGHT, WIDTH, 3) or not _near(arr[785, 1025], BACKGROUND):
        return "abstain"
    turns, registers = [], []
    for row in range(N):
        turn_row, reg_row = [], []
        for col in range(N):
            turn_pixel = arr[GRID_Y + row * CELL + 12, GRID_X + col * CELL + 12]
            # Sample the left interior of the badge, away from its white letter and rim.
            reg_pixel = arr[GRID_Y + row * CELL + 72, GRID_X + col * CELL + 63]
            if _near(turn_pixel, GOLD):
                turn_row.append(0)
            elif _near(turn_pixel, BLUE):
                turn_row.append(1)
            else:
                return "abstain"
            if _near(reg_pixel, RED):
                reg_row.append(0)
            elif _near(reg_pixel, VIOLET):
                reg_row.append(1)
            else:
                return "abstain"
        turns.append(turn_row)
        registers.append(reg_row)
    queue = []
    for cx in QUEUE_XS:
        pixel = arr[QUEUE_Y - 15, cx]
        if _near(pixel, RED):
            queue.append(0)
        elif _near(pixel, VIOLET):
            queue.append(1)
        else:
            return "abstain"
    red, violet = _token(arr, RED), _token(arr, VIOLET)
    if red is None or violet is None or red[0] == violet[0]:
        return "abstain"
    positions = [red[0], violet[0]]
    headings = [red[1], violet[1]]
    for _ in range(80):
        actor = queue.pop(0)
        row, col = positions[actor]
        old = turns[row][col]
        headings[actor] = (headings[actor] + (1 if old == 0 else -1)) % 4
        turns[row][col] ^= 1
        stored = registers[row][col]
        registers[row][col] = actor
        queue.append(stored)
        dr, dc = DIRS[headings[actor]]
        positions[actor] = [row + dr, col + dc]
        nr, nc = positions[actor]
        if nr < 0 or nr >= N or nc < 0 or nc >= N:
            return ("token-red", "token-violet")[actor]
    return "abstain"
