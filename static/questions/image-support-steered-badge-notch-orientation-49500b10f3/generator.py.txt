"""Renderer and analytic arm for the compact notched-badge world.

The label is the edge on which a filled badge has a single V-shaped bite.  The
badge is deliberately the largest connected foreground component, while four
small remote marks provide nuisance clutter and an order-permutation symmetry.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw


RENDERER_VERSION = "badge_notch_orientation-0.1.0"
WIDTH = 480
HEIGHT = 360
SCALE = 4
BACKGROUND = (255, 255, 255)
BADGE_COLORS = ((20, 126, 170), (190, 78, 84), (92, 74, 166))
REMOTE_COLORS = ((82, 92, 108), (214, 150, 55), (55, 132, 100), (126, 88, 140))


def _badge_points(scene: dict[str, Any]) -> list[tuple[float, float]]:
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    half_w = float(scene["half_width"])
    half_h = float(scene["half_height"])
    notch_w = float(scene["notch_width"])
    notch_d = float(scene["notch_depth"])
    left = cx - half_w
    right = cx + half_w
    top = cy - half_h
    bottom = cy + half_h
    corner = min(12.0, half_w * 0.20, half_h * 0.20)
    points = [
        (left + corner, top),
        (cx - notch_w / 2.0, top),
        (cx, top + notch_d),
        (cx + notch_w / 2.0, top),
        (right - corner, top),
        (right, top + corner),
        (right, bottom - corner),
        (right - corner, bottom),
        (left + corner, bottom),
        (left, bottom - corner),
        (left, top + corner),
    ]
    if scene["notch_side"] == "lower":
        points = [(x, top + bottom - y) for x, y in points]
    return points


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample independent label, nuisance geometry, color, and decorations."""

    side_rng = random.Random(seed * 1009 + 17)
    geometry_rng = random.Random(seed * 9176 + 31)
    color_rng = random.Random(seed * 6133 + 47)
    remote_rng = random.Random(seed * 7919 + 61)

    side = "upper" if side_rng.randrange(2) == 0 else "lower"
    cx = geometry_rng.randint(125, 355)
    cy = geometry_rng.randint(115, 245)
    half_w = geometry_rng.randint(52, 68)
    half_h = geometry_rng.randint(45, 58)
    notch_width = geometry_rng.randint(32, 46)
    notch_depth = geometry_rng.randint(22, 31)
    badge_rgb = list(BADGE_COLORS[color_rng.randrange(len(BADGE_COLORS))])

    slots = ((42, 42), (438, 42), (42, 318), (438, 318))
    remote_marks: list[dict[str, Any]] = []
    for index, (x, y) in enumerate(slots):
        jitter_x = remote_rng.randint(-5, 5)
        jitter_y = remote_rng.randint(-5, 5)
        radius = remote_rng.randint(7, 10)
        remote_marks.append(
            {
                "kind": "ring",
                "x": x + jitter_x,
                "y": y + jitter_y,
                "radius": radius,
                "stroke": remote_rng.randint(3, 5),
                "rgb": list(REMOTE_COLORS[(index + remote_rng.randrange(3)) % len(REMOTE_COLORS)]),
            }
        )

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "center_x": cx,
        "center_y": cy,
        "half_width": half_w,
        "half_height": half_h,
        "notch_width": notch_width,
        "notch_depth": notch_depth,
        "notch_side": side,
        "badge_rgb": badge_rgb,
        "remote_marks": remote_marks,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render one deterministic RGB PNG-sized image."""

    width = int(scene["width"])
    height = int(scene["height"])
    image = Image.new("RGB", (width * SCALE, height * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)

    def scale_point(x: float, y: float) -> tuple[int, int]:
        return (int(round(x * SCALE)), int(round(y * SCALE)))

    for mark in scene["remote_marks"]:
        x = float(mark["x"])
        y = float(mark["y"])
        radius = float(mark["radius"])
        box = (
            int(round((x - radius) * SCALE)),
            int(round((y - radius) * SCALE)),
            int(round((x + radius) * SCALE)),
            int(round((y + radius) * SCALE)),
        )
        rgb = tuple(int(value) for value in mark["rgb"])
        stroke = max(1, int(round(float(mark["stroke"]) * SCALE)))
        draw.ellipse(box, outline=rgb, width=stroke)

    badge = [scale_point(x, y) for x, y in _badge_points(scene)]
    draw.polygon(badge, fill=tuple(int(value) for value in scene["badge_rgb"]))
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    """Closed-form label, keyed only by the rendered notch side."""

    return str(scene["notch_side"])


def margin(scene: dict[str, Any]) -> float:
    """Visible edge-location clearance, measured in pixels of notch depth."""

    return round(min(float(scene["notch_depth"]) - 10.0, float(scene["notch_width"]) - 16.0), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Reject shallow or narrow bites whose raster occupancy is ambiguous."""

    return margin(scene) < 7.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """Swap nonoverlapping remote marks; pixels and the label are invariant."""

    twin = dict(scene)
    twin["remote_marks"] = list(reversed(scene["remote_marks"]))
    return [("permute_remote_mark_order", twin)]
