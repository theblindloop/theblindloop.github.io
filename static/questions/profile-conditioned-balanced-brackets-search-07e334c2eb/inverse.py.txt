"""Independent image-only inverse arm for bracket-direction sequences."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_LABELS = ("A", "B", "C", "D")


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Return coordinate arrays for 4-connected colored components."""
    work = mask.copy()
    found: list[np.ndarray] = []
    height, width = work.shape
    for sy, sx in zip(*np.nonzero(work), strict=True):
        if not work[sy, sx]:
            continue
        work[sy, sx] = False
        queue: deque[tuple[int, int]] = deque([(int(sy), int(sx))])
        pixels: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            pixels.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        if len(pixels) >= 300:
            found.append(np.asarray(pixels, dtype=np.int32))
    return found


def _kind(component: np.ndarray) -> tuple[str, float]:
    ys, xs = component[:, 0], component[:, 1]
    x0, x1 = int(xs.min()), int(xs.max())
    width = x1 - x0 + 1
    if width < 20:
        raise ValueError("bracket component is too narrow")
    band = max(5, width // 4)
    left = int(np.count_nonzero(xs < x0 + band))
    right = int(np.count_nonzero(xs > x1 - band))
    confidence = abs(left - right) / max(left + right, 1)
    if confidence < 0.20:
        raise ValueError("bracket direction is ambiguous")
    return ("open" if left > right else "close", confidence)


def _read_rows(image: Image.Image) -> tuple[list[list[str]], float]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    chroma = arr.max(axis=2) - arr.min(axis=2)
    brightness = arr.mean(axis=2)
    mask = (chroma > 38) & (brightness < 225)
    comps = _components(mask)
    if len(comps) != 24:
        raise ValueError(f"expected 24 bracket components, found {len(comps)}")
    features = []
    confidences = []
    for comp in comps:
        kind, confidence = _kind(comp)
        features.append((float(comp[:, 0].mean()), float(comp[:, 1].mean()), kind))
        confidences.append(confidence)
    features.sort(key=lambda item: (item[0], item[1]))
    rows: list[list[str]] = []
    for start in range(0, 24, 6):
        group = sorted(features[start : start + 6], key=lambda item: item[1])
        if max(item[0] for item in group) - min(item[0] for item in group) > 12:
            raise ValueError("colored components do not form clean six-symbol rows")
        rows.append([item[2] for item in group])
    row_centers = [sum(features[i + j][0] for j in range(6)) / 6 for i in range(0, 24, 6)]
    if min(b - a for a, b in zip(row_centers, row_centers[1:])) < 70:
        raise ValueError("row separation is ambiguous")
    return rows, min(confidences)


def _balanced(kinds: list[str]) -> bool:
    running = 0
    for kind in kinds:
        running += 1 if kind == "open" else -1
        if running < 0:
            return False
    return running == 0


def decision_from_image(image: Image.Image) -> str:
    """Enumerate all visible rows and require one well-formed bracket sequence."""
    rows, _ = _read_rows(image)
    winners = [_LABELS[index] for index, row in enumerate(rows) if _balanced(row)]
    if len(winners) != 1:
        raise ValueError(f"expected one pixel winner, found {winners}")
    return winners[0]


def diagnostic_from_image(image: Image.Image) -> dict:
    rows, confidence = _read_rows(image)
    scores = []
    for row in rows:
        running = 0
        low = 0
        for kind in row:
            running += 1 if kind == "open" else -1
            low = min(low, running)
        scores.append(low if running == 0 else -10 - abs(running))
    ranked = sorted(scores, reverse=True)
    return {
        "rows": rows,
        "classification_confidence": float(confidence),
        "runner_up_margin": float(ranked[0] - ranked[1]),
    }
