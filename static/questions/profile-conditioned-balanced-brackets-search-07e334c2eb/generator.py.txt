"""Deterministic latent sampler and renderer for balanced-bracket search."""

from __future__ import annotations

import copy
import itertools
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 900, 590
LABELS = ("A", "B", "C", "D")
PALETTES = ("#1f5f99", "#6c438d", "#147d74", "#a24d2f")
_FONT = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"


def _directions(symbols: list[dict]) -> tuple[str, ...]:
    ordered = sorted(symbols, key=lambda item: int(item["slot"]))
    return tuple(str(item["kind"]) for item in ordered)


def _prefix_score(symbols: list[dict]) -> int:
    running = 0
    lowest = 0
    for kind in _directions(symbols):
        running += 1 if kind == "open" else -1
        lowest = min(lowest, running)
    return lowest if running == 0 else -10 - abs(running)


def _balanced(symbols: list[dict]) -> bool:
    return _prefix_score(symbols) >= 0


_ALL = list(itertools.combinations(range(6), 3))
_VALID: list[tuple[str, ...]] = []
_INVALID: list[tuple[str, ...]] = []
for opens in _ALL:
    seq = tuple("open" if index in opens else "close" for index in range(6))
    records = [{"slot": i, "kind": kind} for i, kind in enumerate(seq)]
    (_VALID if _balanced(records) else _INVALID).append(seq)


def sample_scene(seed: int) -> dict:
    """Return a balanced, deterministic scene; the answer cycles evenly by seed."""
    rng = random.Random(int(seed))
    winner = int(seed) % 4
    valid = _VALID[rng.randrange(len(_VALID))]
    distractors = rng.sample(_INVALID, 3)
    seqs: list[tuple[str, ...]] = []
    distractor_index = 0
    for row_index in range(4):
        if row_index == winner:
            seqs.append(valid)
        else:
            seqs.append(distractors[distractor_index])
            distractor_index += 1
    rows = [
        {
            "label": LABELS[row_index],
            "symbols": [
                {"slot": slot, "kind": kind} for slot, kind in enumerate(seqs[row_index])
            ],
        }
        for row_index in range(4)
    ]
    return {
        "rows": rows,
        "palette": PALETTES[rng.randrange(len(PALETTES))],
        "offset_x": rng.randint(-8, 8),
        "offset_y": rng.randint(-5, 5),
    }


def analytic_gold(scene: dict) -> str:
    winners = [
        str(row["label"])
        for row in scene["rows"]
        if _balanced(list(row["symbols"]))
    ]
    if len(winners) != 1:
        raise ValueError(f"expected one balanced row, found {winners}")
    return winners[0]


def margin(scene: dict) -> float:
    scores = sorted(
        (_prefix_score(list(row["symbols"])) for row in scene["rows"]), reverse=True
    )
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 1.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    for row in twin["rows"]:
        row["symbols"] = list(reversed(row["symbols"]))
    return [("reverse_serialized_symbol_records", twin)]


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype(_FONT, size=size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    """Render four visible rows; symbol placement uses explicit slots, not list order."""
    ss = 3
    image = Image.new("RGB", (WIDTH * ss, HEIGHT * ss), "#ffffff")
    draw = ImageDraw.Draw(image)
    dx = int(scene["offset_x"])
    dy = int(scene["offset_y"])
    ink = str(scene["palette"])
    label_font = _font(28 * ss)
    hint_font = _font(18 * ss)

    draw.text((22 * ss, 10 * ss), "opening  [      closing  ]", fill="#4b4b4b", font=hint_font)
    for row_index, row in enumerate(scene["rows"]):
        top = 52 + dy + row_index * 132
        draw.rounded_rectangle(
            ((70 + dx) * ss, top * ss, (850 + dx) * ss, (top + 104) * ss),
            radius=12 * ss,
            outline="#b8bcc2",
            width=2 * ss,
            fill="#fafbfc",
        )
        draw.ellipse(
            ((84 + dx) * ss, (top + 28) * ss, (132 + dx) * ss, (top + 76) * ss),
            fill="#eceff2",
            outline="#737a82",
            width=2 * ss,
        )
        label = str(row["label"])
        box = draw.textbbox((0, 0), label, font=label_font)
        tw, th = box[2] - box[0], box[3] - box[1]
        draw.text(
            ((108 + dx) * ss - tw / 2, (top + 52) * ss - th / 2 - 3 * ss),
            label,
            fill="#30343a",
            font=label_font,
        )
        for symbol in sorted(row["symbols"], key=lambda item: int(item["slot"])):
            slot = int(symbol["slot"])
            kind = str(symbol["kind"])
            x = 190 + dx + slot * 103
            y = top + 20
            w, h, stroke = 44, 64, 8
            if kind == "open":
                draw.rectangle((x * ss, y * ss, (x + stroke) * ss, (y + h) * ss), fill=ink)
                draw.rectangle((x * ss, y * ss, (x + w) * ss, (y + stroke) * ss), fill=ink)
                draw.rectangle(
                    (x * ss, (y + h - stroke) * ss, (x + w) * ss, (y + h) * ss),
                    fill=ink,
                )
            elif kind == "close":
                draw.rectangle(
                    ((x + w - stroke) * ss, y * ss, (x + w) * ss, (y + h) * ss),
                    fill=ink,
                )
                draw.rectangle((x * ss, y * ss, (x + w) * ss, (y + stroke) * ss), fill=ink)
                draw.rectangle(
                    (x * ss, (y + h - stroke) * ss, (x + w) * ss, (y + h) * ss),
                    fill=ink,
                )
            else:
                raise ValueError(f"unknown bracket kind {kind!r}")
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
