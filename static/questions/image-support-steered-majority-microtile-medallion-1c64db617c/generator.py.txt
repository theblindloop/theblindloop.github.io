"""Analytic sampler and raster renderer for Majority Microtile Medallion."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS = 512
FRAME = (92, 48, 132)
GOLD = (224, 167, 54)
BACKGROUND = (247, 245, 238)
DARK_PALETTES = ((25, 43, 73), (47, 35, 69), (22, 62, 62))
LIGHT_PALETTES = ((234, 221, 184), (226, 231, 213), (236, 216, 203))
LABELS = ("X", "plus", "T", "L")

TEMPLATES = {
    "X": ("10001", "01010", "00100", "01010", "10001"),
    "plus": ("00100", "00100", "11111", "00100", "00100"),
    "T": ("11111", "00100", "00100", "00100", "00100"),
    "L": ("10000", "10000", "10000", "10000", "11111"),
}


def _target_bits(label: str) -> list[list[int]]:
    return [[int(value) for value in row] for row in TEMPLATES[label]]


def _pool_board(board: list[list[int]]) -> list[list[int]]:
    return [
        [
            int(sum(board[by * 3 + dy][bx * 3 + dx] for dy in range(3) for dx in range(3)) >= 5)
            for bx in range(5)
        ]
        for by in range(5)
    ]


def _classify(bits: list[list[int]]) -> tuple[str, int, int]:
    distances = sorted(
        (sum(bits[y][x] != int(TEMPLATES[label][y][x]) for y in range(5) for x in range(5)), label)
        for label in LABELS
    )
    return distances[0][1], distances[0][0], distances[1][0]


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    label = LABELS[int(seed) % len(LABELS)]
    target = _target_bits(label)
    board = [[0 for _ in range(15)] for _ in range(15)]
    for by in range(5):
        for bx in range(5):
            count = 5 if target[by][bx] else 4
            positions = list(range(9))
            rng.shuffle(positions)
            for index in positions[:count]:
                board[by * 3 + index // 3][bx * 3 + index % 3] = 1
    cell_px = rng.choice((4, 5))
    outer_side = 15 * cell_px + 12
    origin_x = rng.randint(28, CANVAS - outer_side - 28)
    origin_y = rng.randint(28, CANVAS - outer_side - 28)
    palette_index = rng.randrange(len(DARK_PALETTES))
    return {
        "board": board,
        "origin_x": origin_x,
        "origin_y": origin_y,
        "cell_px": cell_px,
        "dark_rgb": list(DARK_PALETTES[palette_index]),
        "light_rgb": list(LIGHT_PALETTES[palette_index]),
        "frame_rgb": list(FRAME),
    }


def render(scene: dict[str, Any]) -> Image.Image:
    board = scene["board"]
    origin_x = int(scene["origin_x"])
    origin_y = int(scene["origin_y"])
    cell_px = int(scene["cell_px"])
    dark_rgb = tuple(scene["dark_rgb"])
    light_rgb = tuple(scene["light_rgb"])
    frame_rgb = tuple(scene["frame_rgb"])
    board_side = 15 * cell_px
    image = Image.new("RGB", (CANVAS, CANVAS), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rectangle(
        (origin_x, origin_y, origin_x + board_side + 11, origin_y + board_side + 11),
        fill=frame_rgb,
    )
    inner_x, inner_y = origin_x + 6, origin_y + 6
    for row in range(15):
        for col in range(15):
            x0 = inner_x + col * cell_px
            y0 = inner_y + row * cell_px
            color = dark_rgb if board[row][col] else light_rgb
            draw.rectangle((x0, y0, x0 + cell_px - 1, y0 + cell_px - 1), fill=color)
    for boundary in range(0, 15, 3):
        coordinate_x = inner_x + boundary * cell_px
        coordinate_y = inner_y + boundary * cell_px
        draw.line((coordinate_x, inner_y, coordinate_x, inner_y + board_side - 1), fill=GOLD, width=1)
        draw.line((inner_x, coordinate_y, inner_x + board_side - 1, coordinate_y), fill=GOLD, width=1)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    label, _, _ = _classify(_pool_board(scene["board"]))
    return label


def margin(scene: dict[str, Any]) -> float:
    _, best, runner_up = _classify(_pool_board(scene["board"]))
    return float(runner_up - best)


def is_quarantined(scene: dict[str, Any]) -> bool:
    board = scene["board"]
    if int(scene["cell_px"]) < 4 or len(board) != 15 or any(len(row) != 15 for row in board):
        return True
    for by in range(5):
        for bx in range(5):
            votes = sum(board[by * 3 + dy][bx * 3 + dx] for dy in range(3) for dx in range(3))
            if votes not in (4, 5):
                return True
    return margin(scene) <= 0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
