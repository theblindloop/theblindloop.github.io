"""Independent pixel-only inverse for the majority microtile transform."""

from __future__ import annotations

import numpy as np
from PIL import Image

FRAME = np.array((92, 48, 132), dtype=np.int16)
TEMPLATES = {
    "X": ("10001", "01010", "00100", "01010", "10001"),
    "plus": ("00100", "00100", "11111", "00100", "00100"),
    "T": ("11111", "00100", "00100", "00100", "00100"),
    "L": ("10000", "10000", "10000", "10000", "11111"),
}


def _recover(image: Image.Image) -> tuple[list[list[int]], float]:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    distance = np.linalg.norm(pixels - FRAME[None, None, :], axis=2)
    points = np.argwhere(distance < 12.0)
    if len(points) < 1200:
        raise ValueError("violet frame missing or damaged")
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    if x1 - x0 != y1 - y0:
        raise ValueError("frame is not square")
    board_side = int(x1 - x0 + 1 - 12)
    if board_side not in (60, 75) or board_side % 15:
        raise ValueError("frame has unsupported scale")
    cell = board_side // 15
    inner_x, inner_y = int(x0 + 6), int(y0 + 6)
    strips = (
        distance[y0 : y0 + 5, x0 : x1 + 1],
        distance[y1 - 4 : y1 + 1, x0 : x1 + 1],
        distance[y0 : y1 + 1, x0 : x0 + 5],
        distance[y0 : y1 + 1, x1 - 4 : x1 + 1],
    )
    if any(float(np.mean(strip < 12.0)) < 0.96 for strip in strips):
        raise ValueError("frame continuity check failed")
    board: list[list[int]] = []
    confidences: list[float] = []
    for row in range(15):
        output_row = []
        for col in range(15):
            cx = inner_x + col * cell + cell // 2
            cy = inner_y + row * cell + cell // 2
            patch = pixels[cy - 1 : cy + 2, cx - 1 : cx + 2].astype(np.float64)
            luminance = float(np.median(0.2126 * patch[:, :, 0] + 0.7152 * patch[:, :, 1] + 0.0722 * patch[:, :, 2]))
            if 120.0 <= luminance <= 165.0:
                raise ValueError("ambiguous microcell luminance")
            output_row.append(int(luminance < 142.5))
            confidences.append(abs(luminance - 142.5))
        board.append(output_row)
    pooled = []
    for by in range(5):
        row = []
        for bx in range(5):
            votes = sum(board[by * 3 + dy][bx * 3 + dx] for dy in range(3) for dx in range(3))
            if votes not in (4, 5):
                raise ValueError("block vote is outside the declared one-vote boundary")
            row.append(int(votes >= 5))
        pooled.append(row)
    return pooled, min(confidences)


def _rank(bits: list[list[int]]) -> list[tuple[int, str]]:
    return sorted(
        (sum(bits[y][x] != int(rows[y][x]) for y in range(5) for x in range(5)), label)
        for label, rows in TEMPLATES.items()
    )


def decision_with_margin(image: Image.Image) -> tuple[str, float]:
    bits, pixel_margin = _recover(image)
    ranked = _rank(bits)
    if ranked[0][0] != 0 or ranked[1][0] <= ranked[0][0]:
        raise ValueError("pooled board does not uniquely match a declared symbol")
    return ranked[0][1], float(min(ranked[1][0] - ranked[0][0], pixel_margin))


def decision_from_image(image: Image.Image) -> str:
    return decision_with_margin(image)[0]
