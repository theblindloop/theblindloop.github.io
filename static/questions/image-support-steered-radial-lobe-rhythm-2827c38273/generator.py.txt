"""Renderer for a single closed contour with four or five global radial lobes."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS = 512
PALETTES = [
    ("#1769aa", "#ffffff"),
    ("#2453a6", "#fffdf8"),
    ("#087f8c", "#fffefe"),
    ("#5b4bb7", "#fffefb"),
]


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    lobes = 4 if rng.randrange(2) == 0 else 5
    ink, background = PALETTES[rng.randrange(len(PALETTES))]
    # A long, compact oval makes the decisive support regional and pathlike.
    major = rng.uniform(125.0, 137.0)
    minor = rng.uniform(34.0, 40.0)
    vertical = bool(rng.randrange(2))
    rotation = (math.pi / 2 if vertical else 0.0) + rng.uniform(-0.055, 0.055)
    return {
        "lobe_count": lobes,
        "cx": rng.uniform(252.0, 260.0),
        "cy": rng.uniform(252.0, 260.0),
        "major_radius": major,
        "minor_radius": minor,
        "wave_strength": rng.uniform(0.155, 0.205),
        "rotation": rotation,
        "nuisance_strength": rng.uniform(0.018, 0.038),
        "nuisance_phase": rng.uniform(0.0, 2.0 * math.pi),
        "stroke": rng.randrange(7, 10),
        "ink": ink,
        "background": background,
        "width": CANVAS,
        "height": CANVAS,
    }


def _points(scene: dict[str, Any], scale: int) -> list[tuple[float, float]]:
    lobes = int(scene["lobe_count"])
    cx, cy = float(scene["cx"]), float(scene["cy"])
    major, minor = float(scene["major_radius"]), float(scene["minor_radius"])
    wave = float(scene["wave_strength"])
    rotation = float(scene["rotation"])
    nuisance = float(scene["nuisance_strength"])
    nuisance_phase = float(scene["nuisance_phase"])
    # Reading style fields here keeps the latent/render boundary explicit.
    _ = (scene["stroke"], scene["ink"], scene["background"], scene["width"], scene["height"])
    cr, sr = math.cos(rotation), math.sin(rotation)
    points = []
    for index in range(960):
        theta = 2.0 * math.pi * index / 960.0
        radial = 1.0 + wave * math.cos(lobes * theta) + nuisance * math.cos(
            2.0 * theta + nuisance_phase
        )
        ux = major * radial * math.cos(theta)
        uy = minor * radial * math.sin(theta)
        x = cx + cr * ux - sr * uy
        y = cy + sr * ux + cr * uy
        points.append((x * scale, y * scale))
    return points


def render(scene: dict[str, Any]) -> Image.Image:
    scale = 4
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), scene["background"])
    draw = ImageDraw.Draw(image)
    pts = _points(scene, scale)
    draw.line(
        pts + [pts[0]],
        fill=scene["ink"],
        width=int(scene["stroke"]) * scale,
        joint="curve",
    )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    return "four" if int(scene["lobe_count"]) == 4 else "five"


def margin(scene: dict[str, Any]) -> float:
    """Designed dominant-to-runner-up normalized radial amplitude gap."""
    return float(scene["wave_strength"]) - float(scene["nuisance_strength"])


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 0.08


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
