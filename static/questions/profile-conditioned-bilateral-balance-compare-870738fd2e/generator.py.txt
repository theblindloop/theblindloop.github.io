"""Renderer and analytic reference for the bilateral-balance question world."""

from __future__ import annotations

import math
import random
from typing import Iterable, Sequence

from PIL import Image, ImageDraw, ImageFont


WIDTH = 1000
HEIGHT = 560
BACKGROUND = (248, 249, 252)
PANEL_FILL = (255, 255, 255)
PANEL_EDGE = (177, 184, 196)
TEXT = (35, 43, 56)
GUIDE = (112, 121, 136)
GUIDE_DASH = 10
GUIDES = (272, 728)
PANEL_BOXES = ((55, 105, 490, 515), (510, 105, 945, 515))
SHAPE_Y = (185, 218, 258, 300, 342, 384, 424, 456)
QUARANTINE_MARGIN = 0.08
PALETTE = (
    (42, 112, 208),
    (224, 104, 54),
    (24, 155, 122),
    (168, 81, 188),
    (207, 145, 32),
)


def _font() -> ImageFont.ImageFont:
    return ImageFont.load_default()


def _polygon_area(points: Sequence[Sequence[float]]) -> float:
    total = 0.0
    for first, second in zip(points, points[1:] + points[:1]):
        total += float(first[0]) * float(second[1])
        total -= float(second[0]) * float(first[1])
    return abs(total) * 0.5


def _clip_x(
    points: Sequence[Sequence[float]],
    boundary: float,
    keep_left: bool,
) -> list[list[float]]:
    """Clip a polygon against one vertical half-plane."""

    if not points:
        return []
    output: list[list[float]] = []

    def inside(point: Sequence[float]) -> bool:
        return float(point[0]) <= boundary if keep_left else float(point[0]) >= boundary

    def intersection(start: Sequence[float], end: Sequence[float]) -> list[float]:
        dx = float(end[0]) - float(start[0])
        if abs(dx) < 1e-12:
            return [boundary, float(start[1])]
        ratio = (boundary - float(start[0])) / dx
        return [boundary, float(start[1]) + ratio * (float(end[1]) - float(start[1]))]

    previous = points[-1]
    previous_inside = inside(previous)
    for current in points:
        current_inside = inside(current)
        if current_inside:
            if not previous_inside:
                output.append(intersection(previous, current))
            output.append([float(current[0]), float(current[1])])
        elif previous_inside:
            output.append(intersection(previous, current))
        previous = current
        previous_inside = current_inside
    return output


def _score_points(points: Sequence[Sequence[float]], guide_x: float) -> float:
    total = _polygon_area(points)
    left = _polygon_area(_clip_x(points, guide_x, True))
    right = _polygon_area(_clip_x(points, guide_x, False))
    if total <= 1e-9:
        raise ValueError("degenerate silhouette")
    return abs(left - right) / total


def _shape_points(
    rng: random.Random,
    guide_x: int,
    target: float,
    *,
    y_shift: int,
) -> list[list[int]]:
    """Build a simple, connected silhouette with a controlled area skew."""

    profile = (0.68, 0.88, 1.02, 1.15, 1.18, 1.08, 0.90, 0.66)
    base = rng.randint(68, 86)
    sign = -1 if rng.randrange(2) else 1
    left_factor = 1.0 - target if sign > 0 else 1.0 + target
    right_factor = 1.0 + target if sign > 0 else 1.0 - target
    jitter = [rng.uniform(0.94, 1.06) for _ in profile]
    y_jitter = [rng.randint(-3, 3) for _ in profile]
    right_side: list[list[int]] = []
    left_side: list[list[int]] = []
    for index, shape_y in enumerate(SHAPE_Y):
        width = base * profile[index] * jitter[index]
        left_width = max(18.0, width * left_factor)
        right_width = max(18.0, width * right_factor)
        y = shape_y + y_shift + y_jitter[index]
        right_side.append([int(round(guide_x + right_width)), int(y)])
        left_side.append([int(round(guide_x - left_width)), int(y)])
    return right_side + list(reversed(left_side))


def sample_scene(seed: int) -> dict:
    """Sample one balanced, visible scene from an integer seed."""

    rng = random.Random(int(seed))
    answer_a = int(seed) % 2 == 0
    near_boundary = int(seed) % 13 == 0
    if near_boundary:
        high = 0.245 + rng.uniform(-0.004, 0.004)
        low = high - rng.uniform(0.012, 0.024)
    else:
        high = rng.uniform(0.36, 0.54)
        low = rng.uniform(0.10, 0.22)
    target_a, target_b = (high, low) if answer_a else (low, high)
    color_a = list(PALETTE[rng.randrange(len(PALETTE))])
    color_b = list(PALETTE[rng.randrange(len(PALETTE))])
    if color_b == color_a:
        color_b = list(PALETTE[(PALETTE.index(tuple(color_a)) + 1) % len(PALETTE)])
    shape_a = _shape_points(
        rng,
        GUIDES[0],
        target_a,
        y_shift=rng.randint(-8, 8),
    )
    shape_b = _shape_points(
        rng,
        GUIDES[1],
        target_b,
        y_shift=rng.randint(-8, 8),
    )
    return {
        "shape_a": {"points": shape_a},
        "shape_b": {"points": shape_b},
        "color_a": color_a,
        "color_b": color_b,
    }


def _draw_dashed_guide(draw: ImageDraw.ImageDraw, x: int) -> None:
    for y in range(160, 478, GUIDE_DASH * 2):
        draw.line((x, y, x, min(y + GUIDE_DASH, 478)), fill=GUIDE, width=2)


def _draw_shape(
    draw: ImageDraw.ImageDraw,
    points: Sequence[Sequence[int]],
    color: Sequence[int],
) -> None:
    draw.polygon([tuple(point) for point in points], fill=tuple(color))


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = _font()
    draw.text((55, 30), "BILATERAL AREA BALANCE", fill=TEXT, font=font)
    draw.text((55, 53), "Compare normalized colored-area imbalance about each guide", fill=TEXT, font=font)
    for label, box in zip(("A", "B"), PANEL_BOXES):
        draw.rounded_rectangle(box, radius=8, fill=PANEL_FILL, outline=PANEL_EDGE, width=2)
        draw.text((box[0] + 16, box[1] + 14), label, fill=TEXT, font=font)

    _draw_shape(draw, scene["shape_a"]["points"], scene["color_a"])
    _draw_shape(draw, scene["shape_b"]["points"], scene["color_b"])
    _draw_dashed_guide(draw, GUIDES[0])
    _draw_dashed_guide(draw, GUIDES[1])
    draw.text((GUIDES[0] - 36, 488), "guide", fill=GUIDE, font=font)
    draw.text((GUIDES[1] - 36, 488), "guide", fill=GUIDE, font=font)
    return image


def analytic_gold(scene: dict) -> str:
    score_a = _score_points(scene["shape_a"]["points"], GUIDES[0])
    score_b = _score_points(scene["shape_b"]["points"], GUIDES[1])
    return "A" if score_a > score_b else "B"


def margin(scene: dict) -> float:
    score_a = _score_points(scene["shape_a"]["points"], GUIDES[0])
    score_b = _score_points(scene["shape_b"]["points"], GUIDES[1])
    return abs(score_a - score_b)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
