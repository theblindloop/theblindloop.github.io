"""Independent raster-only inverse for bilateral silhouette imbalance.

This module intentionally duplicates only the public canvas geometry.  It does
not import the renderer, scene generator, prompts, or any analytic reference.
"""

from __future__ import annotations

from PIL import Image


GUIDES = (272, 728)
PANEL_X_RANGES = ((70, 475), (525, 930))
Y_RANGE = range(140, 486)


def _is_colored(pixel: tuple[int, int, int]) -> bool:
    red, green, blue = pixel
    high = max(pixel)
    low = min(pixel)
    return high - low >= 55 and high < 245 and low < 220


def _score_panel(image: Image.Image, x_range: tuple[int, int], guide: int) -> float:
    pixels = image.load()
    left = 0
    right = 0
    colored = 0
    for y in Y_RANGE:
        for x in range(x_range[0], x_range[1]):
            if not _is_colored(pixels[x, y]):
                continue
            colored += 1
            if x < guide:
                left += 1
            elif x > guide:
                right += 1
    if colored < 1000 or left == 0 or right == 0:
        raise ValueError("colored silhouette is missing or not straddling its guide")
    return abs(left - right) / float(colored)


def decision_from_image(image: Image.Image) -> str:
    """Return the panel with the larger normalized colored-area imbalance."""

    rgb = image.convert("RGB")
    score_a = _score_panel(rgb, PANEL_X_RANGES[0], GUIDES[0])
    score_b = _score_panel(rgb, PANEL_X_RANGES[1], GUIDES[1])
    return "A" if score_a > score_b else "B"
