"""Analytic sampler and renderer for a visible checkerboard column count."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 480
BACKGROUND = (250, 248, 242)
COLUMN_COUNTS = (7, 8, 9)
ROW_COUNTS = (7, 8, 9)
PALETTES = (
    ((32, 92, 151), (241, 181, 54)),
    ((157, 55, 96), (73, 177, 157)),
    ((96, 67, 153), (232, 145, 67)),
    ((42, 120, 92), (225, 111, 98)),
)
MIN_CELL_WIDTH = 24.0


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0xC4EC4E2)
    column_count = COLUMN_COUNTS[int(seed) % len(COLUMN_COUNTS)]
    row_count = ROW_COUNTS[(int(seed) // len(COLUMN_COUNTS)) % len(ROW_COUNTS)]
    width = rng.randrange(392, 433)
    height = rng.randrange(300, 341)
    x = rng.randrange(92, WIDTH - width - 92 + 1)
    y = rng.randrange(62, HEIGHT - height - 62 + 1)
    pair = PALETTES[rng.randrange(len(PALETTES))]
    if rng.randrange(2):
        pair = (pair[1], pair[0])
    return {
        "column_count": column_count,
        "row_count": row_count,
        "x": x,
        "y": y,
        "width": width,
        "height": height,
        "color_a": list(pair[0]),
        "color_b": list(pair[1]),
        "start_phase": rng.randrange(2),
        "border_phase": rng.randrange(8),
    }


def render(scene: dict) -> Image.Image:
    columns = int(scene["column_count"])
    rows = int(scene["row_count"])
    x, y = int(scene["x"]), int(scene["y"])
    width, height = int(scene["width"]), int(scene["height"])
    color_a = tuple(int(v) for v in scene["color_a"])
    color_b = tuple(int(v) for v in scene["color_b"])
    phase = int(scene["start_phase"]) % 2
    border_phase = int(scene["border_phase"]) % 2

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    border = (28, 31, 38) if border_phase == 0 else (50, 42, 34)
    draw.rectangle((x, y, x + width - 1, y + height - 1), fill=border)
    inset = 8
    ix, iy = x + inset, y + inset
    iw, ih = width - 2 * inset, height - 2 * inset
    for row in range(rows):
        top = iy + round(row * ih / rows)
        bottom = iy + round((row + 1) * ih / rows)
        for column in range(columns):
            left = ix + round(column * iw / columns)
            right = ix + round((column + 1) * iw / columns)
            fill = color_a if (row + column + phase) % 2 == 0 else color_b
            draw.rectangle((left, top, right - 1, bottom - 1), fill=fill)
    return image


def analytic_gold(scene: dict) -> str:
    return str(int(scene["column_count"]))


def margin(scene: dict) -> float:
    return float((int(scene["width"]) - 16) / int(scene["column_count"]) - MIN_CELL_WIDTH)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["border_phase"] = int(scene["border_phase"]) + 2
    return [("border_phase_plus_two", twin)]
