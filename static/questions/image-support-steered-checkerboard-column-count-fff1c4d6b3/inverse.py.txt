"""Independent image-only inverse for a filled checkerboard's column count.

The inverse receives only the raster and the public contract: one fully filled,
outlined rectangular checkerboard with broad alternating cells. It infers the
board, interior, and repeated color-transition runs directly from pixels.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


def _groups(indices: np.ndarray) -> list[np.ndarray]:
    if len(indices) == 0:
        return []
    cuts = np.where(np.diff(indices) > 1)[0] + 1
    return [part for part in np.split(indices, cuts) if len(part)]


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    if array.ndim != 3 or array.shape[0] < 300 or array.shape[1] < 400:
        raise ValueError("canvas is too small")

    # Estimate the canvas color from the corners, then recover the one filled board.
    corners = np.stack((array[0, 0], array[0, -1], array[-1, 0], array[-1, -1]))
    background = np.median(corners, axis=0)
    foreground = np.linalg.norm(array - background, axis=2) > 28.0
    ys, xs = np.nonzero(foreground)
    if len(xs) < 90000:
        raise ValueError("filled checkerboard region is missing")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    width, height = x1 - x0 + 1, y1 - y0 + 1
    if not (380 <= width <= 445 and 285 <= height <= 355):
        raise ValueError("foreground is not the expected broad board rectangle")

    board_mask = foreground[y0 : y1 + 1, x0 : x1 + 1]
    # This strict regional completeness check makes damage anywhere in the broad
    # connected evidence area causally relevant to the pixel-only decision.
    if float(board_mask.mean()) < 0.995 or np.any(board_mask.mean(axis=0) < 0.98) or np.any(board_mask.mean(axis=1) < 0.98):
        raise ValueError("checkerboard area is incomplete")

    board = array[y0 : y1 + 1, x0 : x1 + 1]
    # Border columns are vertically uniform; interior cell columns alternate over
    # rows and therefore have high across-row color variance. Infer the interior.
    vertical_variation = board.std(axis=0).mean(axis=1)
    active_x = np.flatnonzero(vertical_variation > 22.0)
    groups_x = _groups(active_x)
    if len(groups_x) != 1 or len(groups_x[0]) < width - 28:
        raise ValueError("could not recover a single alternating board interior")
    ix0, ix1 = int(groups_x[0][0]), int(groups_x[0][-1])

    horizontal_variation = board.std(axis=1).mean(axis=1)
    active_y = np.flatnonzero(horizontal_variation > 22.0)
    groups_y = _groups(active_y)
    if len(groups_y) != 1 or len(groups_y[0]) < height - 28:
        raise ValueError("could not recover alternating checker rows")
    iy0, iy1 = int(groups_y[0][0]), int(groups_y[0][-1])
    interior = board[iy0 : iy1 + 1, ix0 : ix1 + 1]

    # A true vertical cell boundary changes color across nearly every row.
    x_delta = np.linalg.norm(interior[:, 1:] - interior[:, :-1], axis=2).mean(axis=0)
    edge_groups = _groups(np.flatnonzero(x_delta > 70.0))
    edges = np.array([float(group.mean()) for group in edge_groups])
    count = len(edges) + 1
    if count not in {7, 8, 9}:
        raise ValueError("column count is outside the public answer set")
    if len(edges) > 1:
        gaps = np.diff(edges)
        expected = interior.shape[1] / count
        if float(np.max(np.abs(gaps - expected))) > 3.0:
            raise ValueError("column transitions are not regularly spaced")

    # Independently require the visible horizontal alternation to be a plausible
    # checkerboard, preventing a striped raster from satisfying the contract.
    y_delta = np.linalg.norm(interior[1:] - interior[:-1], axis=2).mean(axis=1)
    row_edges = _groups(np.flatnonzero(y_delta > 70.0))
    if len(row_edges) + 1 not in {7, 8, 9}:
        raise ValueError("visible checker rows are malformed")
    return str(count)
