"""Independent image-only inverse arm for the implicit nearest-dot partition."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

GRID_STEP = 2


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        xs, ys = [], []
        while queue:
            py, px = queue.popleft()
            xs.append(px)
            ys.append(py)
            for ny, nx in ((py - 1, px), (py + 1, px), (py, px - 1), (py, px + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= 20:
            found.append((float(np.mean(xs)), float(np.mean(ys)), len(xs)))
    return found


def _recover(image: Image.Image) -> list[tuple[float, float, str]]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    red = (arr[..., 0] > 150) & (arr[..., 0] > arr[..., 1] + 70) & (arr[..., 0] > arr[..., 2] + 50)
    blue = (arr[..., 2] > 130) & (arr[..., 2] > arr[..., 0] + 70) & (arr[..., 2] > arr[..., 1] + 50)
    dark = (arr.max(axis=2) < 90) & ~red & ~blue
    groups = []
    for role, mask in (("red", red), ("blue", blue), ("black", dark)):
        components = _components(mask)
        if role != "black" and len(components) != 1:
            raise ValueError(f"expected one {role} dot, found {len(components)}")
        groups.extend((x, y, role) for x, y, _ in components)
    if len(groups) < 6:
        raise ValueError("too few visible dot centers")
    return groups


def _area_counts(width: int, height: int, centers: list[tuple[float, float, str]]) -> np.ndarray:
    xs = np.arange(0.5, width, GRID_STEP, dtype=np.float64)
    ys = np.arange(0.5, height, GRID_STEP, dtype=np.float64)
    xx, yy = np.meshgrid(xs, ys)
    best = np.full(xx.shape, np.inf)
    owner = np.zeros(xx.shape, dtype=np.int16)
    for index, (cx, cy, _) in enumerate(centers):
        d2 = (xx - cx) ** 2 + (yy - cy) ** 2
        take = d2 < best
        owner[take] = index
        best[take] = d2[take]
    return np.bincount(owner.ravel(), minlength=len(centers))


def decision_from_image(image: Image.Image) -> str:
    centers = _recover(image)
    areas = _area_counts(image.width, image.height, centers)
    roles = [item[2] for item in centers]
    red_area = int(areas[roles.index("red")])
    blue_area = int(areas[roles.index("blue")])
    if red_area == blue_area:
        raise ValueError("territory tie is ambiguous")
    return "red" if red_area > blue_area else "blue"
