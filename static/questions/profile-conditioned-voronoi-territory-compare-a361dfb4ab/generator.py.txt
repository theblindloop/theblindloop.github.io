"""Deterministic dot field whose implicit Voronoi cells answer the question."""

from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 384
HEIGHT = 384
RADIUS = 6
WHITE = (250, 249, 246)
BLACK = (35, 38, 43)
RED = (218, 45, 58)
BLUE = (35, 99, 220)
GRID_STEP = 2
QUARANTINE_GAP = 180.0


def _territories(width: int, height: int, points: list[dict]) -> np.ndarray:
    """Sample the full bounded plane on a fixed dense grid and count ownership."""
    xs = np.arange(0.5, width, GRID_STEP, dtype=np.float64)
    ys = np.arange(0.5, height, GRID_STEP, dtype=np.float64)
    xx, yy = np.meshgrid(xs, ys)
    centers = np.array([(p["x"], p["y"]) for p in points], dtype=np.float64)
    best = np.full(xx.shape, np.inf)
    owners = np.zeros(xx.shape, dtype=np.int16)
    for index, (cx, cy) in enumerate(centers):
        dist = (xx - cx) ** 2 + (yy - cy) ** 2
        take = dist < best
        owners[take] = index
        best[take] = dist[take]
    return np.bincount(owners.ravel(), minlength=len(points)).astype(np.float64)


def _rotate(point: tuple[float, float], angle: float) -> tuple[float, float]:
    x, y = point
    c, s = math.cos(angle), math.sin(angle)
    return c * x - s * y, s * x + c * y


def _build_geometry(rng: random.Random) -> tuple[list[tuple[int, int]], tuple[int, int], tuple[int, int]]:
    """Return black dots plus a compact-cell and an open-cell candidate center."""
    phase = rng.uniform(0, 2 * math.pi)
    separation = rng.uniform(62.0, 69.0)
    compact = (separation, 0.0)
    open_cell = (-separation, 0.0)

    # Both candidates have five competitors and the same closest-neighbor scale.
    # Around compact, directions cover 360 degrees. Around open_cell, they occupy
    # one broad side, leaving territory to be limited by the global field/border.
    base_distance = rng.uniform(47.0, 50.0)
    compact_neighbors = []
    for k in range(5):
        a = 2 * math.pi * k / 5 + rng.uniform(-0.055, 0.055)
        d = base_distance if k == 0 else rng.uniform(50.0, 59.0)
        compact_neighbors.append((compact[0] + d * math.cos(a), compact[1] + d * math.sin(a)))
    open_neighbors = []
    for k, degrees in enumerate((-78, -39, 0, 39, 78)):
        a = math.pi + math.radians(degrees) + rng.uniform(-0.045, 0.045)
        d = base_distance if k == 2 else rng.uniform(50.0, 59.0)
        open_neighbors.append((open_cell[0] + d * math.cos(a), open_cell[1] + d * math.sin(a)))

    # Balanced remote distractors prevent the open cell from simply reaching every border.
    remote = [(0.0, -146.0), (0.0, 146.0), (146.0, 0.0), (-146.0, 0.0)]
    raw_black = compact_neighbors + open_neighbors + remote
    cx = WIDTH / 2 + rng.uniform(-5.0, 5.0)
    cy = HEIGHT / 2 + rng.uniform(-5.0, 5.0)

    def place(p: tuple[float, float]) -> tuple[int, int]:
        rx, ry = _rotate(p, phase)
        return int(round(cx + rx)), int(round(cy + ry))

    black = [place(p) for p in raw_black]
    return black, place(compact), place(open_cell)


def _valid_centers(centers: list[tuple[int, int]]) -> bool:
    pad = RADIUS + 4
    if any(not (pad <= x < WIDTH - pad and pad <= y < HEIGHT - pad) for x, y in centers):
        return False
    return all(
        math.hypot(x1 - x2, y1 - y2) >= 2 * RADIUS + 5
        for i, (x1, y1) in enumerate(centers)
        for x2, y2 in centers[i + 1 :]
    )


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    desired = "red" if int(seed) % 2 == 0 else "blue"
    for _ in range(500):
        black, compact, open_cell = _build_geometry(rng)
        centers = black + [compact, open_cell]
        if not _valid_centers(centers):
            continue
        # Assign color independently of which spatial candidate is the open cell.
        if desired == "red":
            red, blue = open_cell, compact
        else:
            red, blue = compact, open_cell
        points = [
            *({"x": x, "y": y, "role": "black"} for x, y in black),
            {"x": red[0], "y": red[1], "role": "red"},
            {"x": blue[0], "y": blue[1], "role": "blue"},
        ]
        # Serialization order is a nuisance and is deliberately randomized.
        rng.shuffle(points)
        scene = {"width": WIDTH, "height": HEIGHT, "radius": RADIUS, "points": points}
        areas = _territories(WIDTH, HEIGHT, points)
        roles = [p["role"] for p in points]
        gap = abs(float(areas[roles.index("red")] - areas[roles.index("blue")]))
        nearest = {}
        coordinates = [(p["x"], p["y"]) for p in points]
        for role in ("red", "blue"):
            idx = roles.index(role)
            x, y = coordinates[idx]
            nearest[role] = min(
                math.hypot(x - ox, y - oy)
                for j, (ox, oy) in enumerate(coordinates)
                if j != idx
            )
        if gap > 260 and abs(nearest["red"] - nearest["blue"]) <= 2.5:
            if analytic_gold(scene) == desired:
                return scene
    raise RuntimeError("could not sample a robust balanced territory scene")


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    radius = int(scene["radius"])
    image = Image.new("RGB", (width, height), WHITE)
    draw = ImageDraw.Draw(image)
    palette = {"black": BLACK, "red": RED, "blue": BLUE}
    for point in scene["points"]:
        x, y = int(point["x"]), int(point["y"])
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=palette[point["role"]])
    return image


def _signed_gap(scene: dict) -> float:
    points = scene["points"]
    areas = _territories(int(scene["width"]), int(scene["height"]), points)
    roles = [p["role"] for p in points]
    return float(areas[roles.index("red")] - areas[roles.index("blue")])


def analytic_gold(scene: dict) -> str:
    return "red" if _signed_gap(scene) > 0 else "blue"


def margin(scene: dict) -> float:
    return abs(_signed_gap(scene))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["points"] = list(reversed(scene["points"]))
    return [("reverse_serialized_dot_order", twin)]
