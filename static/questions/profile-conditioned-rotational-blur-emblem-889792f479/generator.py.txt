"""Renderer for the rotational-blur emblem question world."""

from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image, ImageDraw


SYMBOLS = ("ring", "bullseye", "halo", "target")
CANVAS = (320, 300)
MARKER_PALETTE = (
    (220, 45, 65),
    (35, 150, 75),
    (45, 105, 215),
    (190, 70, 175),
)
CARRIER_FREQUENCY = 7
PHASE_STATES = 16
ROTATION_SAMPLES = 32


def _symbol_mask(symbol: str, rho: np.ndarray) -> np.ndarray:
    if symbol == "ring":
        return ((rho >= 0.235) & (rho < 0.315)).astype(np.float32)
    if symbol == "bullseye":
        return (rho < 0.155).astype(np.float32)
    if symbol == "halo":
        return ((rho >= 0.405) & (rho < 0.475)).astype(np.float32)
    if symbol == "target":
        return ((rho < 0.115) | ((rho >= 0.300) & (rho < 0.390))).astype(np.float32)
    raise ValueError(f"unknown symbol {symbol!r}")


def sample_scene(seed: int) -> dict:
    """Return one deterministic scene; the seed is not itself a rendered field."""
    rng = random.Random(int(seed))
    tile_size = rng.choice((144, 152, 160, 168))
    tile_x = rng.randint(38, CANVAS[0] - tile_size - 38)
    tile_y = rng.randint(34, CANVAS[1] - tile_size - 34)
    signal_strength = 48 + 5 * rng.randrange(5)
    return {
        "symbol": SYMBOLS[int(seed) % len(SYMBOLS)],
        "tile_x": tile_x,
        "tile_y": tile_y,
        "tile_size": tile_size,
        "marker_color": list(rng.choice(MARKER_PALETTE)),
        "marker_size": rng.choice((9, 11, 13)),
        "phase_index": rng.randrange(PHASE_STATES),
        "carrier_amplitude": 78 + 2 * rng.randrange(6),
        "signal_strength": signal_strength,
        "base_level": rng.choice((142, 148, 154, 160)),
        "background": rng.choice((242, 246, 250)),
    }


def render(scene: dict) -> Image.Image:
    """Render a gray square carrying a radial emblem plus an angular carrier."""
    width, height = CANVAS
    background = int(scene["background"])
    image = np.full((height, width, 3), background, dtype=np.uint8)
    size = int(scene["tile_size"])
    yy, xx = np.indices((size, size), dtype=np.float32)
    center = (size - 1) / 2.0
    dx = xx - center
    dy = yy - center
    rho = np.hypot(dx, dy) / (size / 2.0)
    theta = np.arctan2(dy, dx)
    phase = 2.0 * math.pi * (int(scene["phase_index"]) % PHASE_STATES) / PHASE_STATES
    carrier = float(scene["carrier_amplitude"]) * np.sin(CARRIER_FREQUENCY * theta + phase)
    carrier *= (rho < 0.49).astype(np.float32)
    mask = _symbol_mask(str(scene["symbol"]), rho)
    tile = float(scene["base_level"]) + carrier - float(scene["signal_strength"]) * mask
    tile = np.clip(tile, 0, 255).astype(np.uint8)
    rgb_tile = np.repeat(tile[:, :, None], 3, axis=2)
    tx, ty = int(scene["tile_x"]), int(scene["tile_y"])
    image[ty : ty + size, tx : tx + size] = rgb_tile

    # The marks sit just outside the four tile corners, so the inverse can crop
    # the complete raster without having to paint over answer-bearing pixels.
    draw = ImageDraw.Draw(Image.fromarray(image, mode="RGB"))
    # Re-create the draw target after the NumPy assignment without any implicit
    # resampling; this keeps all output pixels integer and deterministic.
    output = Image.fromarray(image, mode="RGB")
    draw = ImageDraw.Draw(output)
    half = int(scene["marker_size"]) // 2
    color = tuple(int(v) for v in scene["marker_color"])
    centers = (
        (tx - 8, ty - 8),
        (tx + size + 7, ty - 8),
        (tx - 8, ty + size + 7),
        (tx + size + 7, ty + size + 7),
    )
    for cx, cy in centers:
        draw.rectangle((cx - half, cy - half, cx + half, cy + half), fill=color)
    return output


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def margin(scene: dict) -> float:
    """One-sided finite-resolution margin for the transformed evidence."""
    return float(scene["signal_strength"]) - 48.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["phase_index"] = int(scene["phase_index"]) + PHASE_STATES
    return [("carrier_phase_period", twin)]
