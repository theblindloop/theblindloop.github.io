"""Independent pixel-only inverse for the rotational-blur emblem."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


SYMBOLS = ("ring", "bullseye", "halo", "target")
ROTATION_SAMPLES = 32


def _marker_centers(saturated: np.ndarray) -> list[tuple[float, float]]:
    """Find disconnected colored fiducials without using renderer geometry."""
    height, width = saturated.shape
    seen = np.zeros_like(saturated, dtype=bool)
    centers: list[tuple[float, float]] = []
    for y, x in np.argwhere(saturated):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and saturated[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= 20:
            values = np.asarray(points, dtype=np.float64)
            centers.append((float(values[:, 1].mean()), float(values[:, 0].mean())))
    if len(centers) != 4:
        raise ValueError(f"expected four colored fiducials, found {len(centers)}")
    return centers


def _crop_square(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    saturation = arr.max(axis=2).astype(np.int16) - arr.min(axis=2).astype(np.int16)
    centers = _marker_centers(saturation > 55)
    xs = [value[0] for value in centers]
    ys = [value[1] for value in centers]
    # Marker centers are eight pixels before the left/top tile edge and seven
    # pixels after the right/bottom edge. This is the public corner-mark frame,
    # inferred from pixels rather than passed as scene data.
    x0 = int(round(min(xs) + 8.0))
    y0 = int(round(min(ys) + 8.0))
    x1 = int(round(max(xs) - 7.0))
    y1 = int(round(max(ys) - 7.0))
    if x1 - x0 != y1 - y0 or not (80 <= x1 - x0 <= 220):
        raise ValueError(f"fiducials do not bracket one square: {(x0, y0, x1, y1)}")
    return arr[y0:y1, x0:x1, 0].astype(np.float32)


def _rotational_average(tile: np.ndarray) -> np.ndarray:
    size = tile.shape[0]
    center = (size - 1) / 2.0
    yy, xx = np.indices(tile.shape, dtype=np.float32)
    dx, dy = xx - center, yy - center
    result = np.zeros_like(tile, dtype=np.float32)
    for index in range(ROTATION_SAMPLES):
        angle = 2.0 * math.pi * index / ROTATION_SAMPLES
        source_x = np.rint(dx * math.cos(angle) - dy * math.sin(angle) + center).astype(np.int32)
        source_y = np.rint(dx * math.sin(angle) + dy * math.cos(angle) + center).astype(np.int32)
        source_x = np.clip(source_x, 0, size - 1)
        source_y = np.clip(source_y, 0, size - 1)
        result += tile[source_y, source_x]
    return result / float(ROTATION_SAMPLES)


def _radial_profile(tile: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    size = tile.shape[0]
    center = (size - 1) / 2.0
    yy, xx = np.indices(tile.shape, dtype=np.float32)
    rho = np.hypot(xx - center, yy - center) / (size / 2.0)
    edges = np.linspace(0.03, 0.49, 48)
    values: list[float] = []
    for radius in edges:
        band = (rho >= radius - 0.0048) & (rho < radius + 0.0048)
        values.append(float(tile[band].mean()))
    return edges, np.asarray(values, dtype=np.float32)


def _mask(symbol: str, radius: np.ndarray) -> np.ndarray:
    if symbol == "ring":
        return ((radius >= 0.235) & (radius < 0.315)).astype(np.float32)
    if symbol == "bullseye":
        return (radius < 0.155).astype(np.float32)
    if symbol == "halo":
        return ((radius >= 0.405) & (radius < 0.475)).astype(np.float32)
    if symbol == "target":
        return ((radius < 0.115) | ((radius >= 0.300) & (radius < 0.390))).astype(np.float32)
    raise ValueError(symbol)


def _score(observed: np.ndarray, template: np.ndarray) -> float:
    design = np.stack((np.ones_like(template), template), axis=1)
    coefficients, _, _, _ = np.linalg.lstsq(design, observed, rcond=None)
    residual = observed - design @ coefficients
    return -float(np.mean(residual * residual))


def decision_from_image(image: Image.Image) -> str:
    """Recover only the transformed radial symbol from a final PNG."""
    tile = _crop_square(image)
    blurred = _rotational_average(tile)
    radii, observed = _radial_profile(blurred)
    scores = {symbol: _score(observed, _mask(symbol, radii)) for symbol in SYMBOLS}
    return max(scores, key=scores.get)
