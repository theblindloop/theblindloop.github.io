"""Independent final-PNG-only inverse arm for the tile intervention.

This module intentionally does not import renderer, prompts, generation,
verification, or any analytic-gold implementation.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int, bool]]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    output = []
    for y0, x0 in zip(*np.nonzero(mask)):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]:
            continue
        seen[y0, x0] = True
        queue = deque([(y0, x0)])
        area = 0
        min_x = max_x = x0
        min_y = max_y = y0
        touches = False
        while queue:
            y, x = queue.popleft()
            area += 1
            min_x, max_x = min(min_x, x), max(max_x, x)
            min_y, max_y = min(min_y, y), max(max_y, y)
            if x == 0 or y == 0 or x == width - 1 or y == height - 1:
                touches = True
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        output.append((min_x, min_y, max_x, max_y, area, touches))
    return output


def _topology(mask: np.ndarray) -> tuple[int, int]:
    ys, xs = np.nonzero(mask)
    if not len(xs):
        raise ValueError("no black figure recovered")
    y0, y1 = max(0, int(ys.min()) - 1), min(mask.shape[0], int(ys.max()) + 2)
    x0, x1 = max(0, int(xs.min()) - 1), min(mask.shape[1], int(xs.max()) + 2)
    crop = mask[y0:y1, x0:x1]
    foreground = [item for item in _components(crop) if item[4] >= 250]
    background = [item for item in _components(~crop) if item[4] >= 250]
    holes = sum(1 for item in background if not item[5])
    return len(foreground), holes


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    channel_max = np.max(rgb, axis=2)
    black = channel_max < 70
    masks = {
        "red": (rgb[:, :, 0] > 150) & (rgb[:, :, 1] < 100) & (rgb[:, :, 2] < 100),
        "green": (rgb[:, :, 0] < 80) & (rgb[:, :, 1] > 100) & (rgb[:, :, 2] < 100),
        "blue": (rgb[:, :, 0] < 80) & (rgb[:, :, 1] < 130) & (rgb[:, :, 2] > 120),
    }
    for color, mask in masks.items():
        components = [item for item in _components(mask) if item[4] >= 250]
        if len(components) != 1 or not 650 <= components[0][4] <= 900:
            raise ValueError(f"could not recover one complete {color} tile")
    signatures = {color: _topology(black | mask) for color, mask in masks.items()}
    matches = [color for color, signature in signatures.items() if signature == (1, 2)]
    if len(matches) != 1:
        raise ValueError(f"ambiguous counterfactual topology {signatures}")
    return matches[0]
