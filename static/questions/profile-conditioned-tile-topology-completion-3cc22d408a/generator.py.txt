"""Deterministic renderer and latent analytic arm for a topology intervention."""

from __future__ import annotations

from collections import deque
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 520
CELL = 28
COLORS = {
    "red": "#d32f2f",
    "green": "#168a3a",
    "blue": "#1565c0",
}
ROLES = ("closure", "spur", "island")


def _canonical_cells() -> tuple[set[tuple[int, int]], dict[str, tuple[int, int]]]:
    """Return a joined closed-left/open-right double-loop block figure."""
    black: set[tuple[int, int]] = set()
    for x in range(1, 8):
        black.add((x, 1))
        black.add((x, 8))
    for y in range(1, 9):
        black.add((1, y))
        black.add((7, y))
    for x in range(7, 14):
        if x != 10:
            black.add((x, 1))
        black.add((x, 8))
    for y in range(1, 9):
        black.add((7, y))
        black.add((13, y))
    tiles = {
        "closure": (10, 1),  # closes the open right lobe
        "spur": (14, 4),     # extends the exterior wall without a new hole
        "island": (4, 4),    # becomes a disconnected piece inside the left hole
    }
    return black, tiles


def _apply_transform(point: tuple[int, int], transform: int) -> tuple[int, int]:
    x, y = point
    if transform >= 4:
        x = -x
    for _ in range(transform % 4):
        x, y = -y, x
    return x, y


def _transformed_cells(scene: dict) -> tuple[set[tuple[int, int]], dict[str, tuple[int, int]]]:
    black, tiles = _canonical_cells()
    transform = int(scene["transform"])
    all_points = [_apply_transform(p, transform) for p in [*black, *tiles.values()]]
    min_x = min(x for x, _ in all_points)
    min_y = min(y for _, y in all_points)

    def normalized(point: tuple[int, int]) -> tuple[int, int]:
        x, y = _apply_transform(point, transform)
        return x - min_x, y - min_y

    return {normalized(p) for p in black}, {role: normalized(p) for role, p in tiles.items()}


def _dimensions(transform: int) -> tuple[int, int]:
    dummy = {"transform": transform}
    black, tiles = _transformed_cells(dummy)
    points = [*black, *tiles.values()]
    return max(x for x, _ in points) + 1, max(y for _, y in points) + 1


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    transform = rng.randrange(8)
    grid_w, grid_h = _dimensions(transform)
    # A cyclic assignment guarantees exact answer balance over consecutive seeds;
    # remaining role colors and geometry vary independently.
    closure_color = ("red", "green", "blue")[int(seed) % 3]
    remaining = [color for color in COLORS if color != closure_color]
    rng.shuffle(remaining)
    color_by_role = {
        "closure": closure_color,
        "spur": remaining[0],
        "island": remaining[1],
    }
    centered_x = (WIDTH - grid_w * CELL) // 2
    centered_y = (HEIGHT - grid_h * CELL) // 2
    offset = [centered_x + rng.randint(-22, 22), centered_y + rng.randint(-18, 18)]
    return {"transform": transform, "offset": offset, "color_by_role": color_by_role}


def _cell_box(point: tuple[int, int], offset: list[int]) -> tuple[int, int, int, int]:
    x, y = point
    x0 = int(offset[0]) + x * CELL
    y0 = int(offset[1]) + y * CELL
    return x0, y0, x0 + CELL - 1, y0 + CELL - 1


def render(scene: dict) -> Image.Image:
    if set(scene) != {"transform", "offset", "color_by_role"}:
        raise ValueError("scene fields do not match the public latent contract")
    if sorted(scene["color_by_role"].values()) != sorted(COLORS):
        raise ValueError("color_by_role must be a color bijection")
    black, tiles = _transformed_cells(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), "white")
    draw = ImageDraw.Draw(image)
    for point in sorted(black):
        draw.rectangle(_cell_box(point, scene["offset"]), fill="#111111")
    for role in ROLES:
        draw.rectangle(
            _cell_box(tiles[role], scene["offset"]),
            fill=COLORS[scene["color_by_role"][role]],
        )
    return image


def _component_count(cells: set[tuple[int, int]]) -> int:
    unseen = set(cells)
    count = 0
    while unseen:
        count += 1
        start = unseen.pop()
        queue = deque([start])
        while queue:
            x, y = queue.popleft()
            for neighbor in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
                if neighbor in unseen:
                    unseen.remove(neighbor)
                    queue.append(neighbor)
    return count


def _hole_count(cells: set[tuple[int, int]]) -> int:
    min_x = min(x for x, _ in cells) - 1
    max_x = max(x for x, _ in cells) + 1
    min_y = min(y for _, y in cells) - 1
    max_y = max(y for _, y in cells) + 1
    background = {
        (x, y)
        for y in range(min_y, max_y + 1)
        for x in range(min_x, max_x + 1)
        if (x, y) not in cells
    }
    holes = 0
    while background:
        start = background.pop()
        queue = deque([start])
        touches_border = False
        while queue:
            x, y = queue.popleft()
            if x in (min_x, max_x) or y in (min_y, max_y):
                touches_border = True
            for neighbor in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
                if neighbor in background:
                    background.remove(neighbor)
                    queue.append(neighbor)
        if not touches_border:
            holes += 1
    return holes


def intervention_signatures(scene: dict) -> dict[str, tuple[int, int]]:
    black, tiles = _transformed_cells(scene)
    return {
        role: (_component_count(black | {point}), _hole_count(black | {point}))
        for role, point in tiles.items()
    }


def analytic_gold(scene: dict) -> str:
    signatures = intervention_signatures(scene)
    matches = [role for role, signature in signatures.items() if signature == (1, 2)]
    if matches != ["closure"]:
        raise ValueError(f"expected unique closure intervention, got {signatures}")
    return scene["color_by_role"][matches[0]]


def margin(scene: dict) -> float:
    # Every topology-changing wall and tile is a complete CELL-wide block.
    return float(CELL)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 20.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
