"""Tracing world: how many colored terminals are reached before a checkpoint.

A single winding blue path (one connected, non-self-crossing serpentine
polyline) is drawn. An orange port ring sits near the start. Two coloured
terminal dots (red and green) rest on the path at two separated on-path
positions, and one magenta checkpoint dot rests at a third on-path position.
Walking only along the blue path from the orange ring, the answer is how many of
the two coloured terminals (red, green) are reached strictly before the magenta
checkpoint: 0, 1, or 2. Because the path winds back and forth in a serpentine,
the on-path order of the markers need not follow their straight-line proximity,
so the decision cannot be solved by Euclidean projection alone; it requires
tracing the drawn stroke geodesically.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "terminals_before_checkpoint-0.1.0"

CANVAS = (480, 480)
BG = (255, 255, 255)
COLOR_BLUE = (40, 90, 210)
COLOR_RED = (225, 40, 40)
COLOR_GREEN = (40, 170, 70)
COLOR_MAGENTA = (200, 40, 200)
COLOR_ORANGE = (245, 150, 20)

STROKE = 20          # blue stroke thickness (px)
PORT_R = 5           # orange port radius
MARK_R = 4           # terminal / checkpoint marker radius
MARGIN_PAD = 46      # keep geometry away from image borders
LANE_GAP = 30.0      # min gap between serpentine lanes
ARC_SEP = 30.0       # min arc separation between markers and port
END_MARGIN = 40.0    # min distance from path endpoints for markers

TERMINAL_COLORS = ["red", "green"]
CHECKPOINT_COLOR = "magenta"
CLASSES = ["0", "1", "2"]


def _pts(points: list) -> list[tuple[float, float]]:
    return [(float(p[0]), float(p[1])) for p in points]


def _path_length(points: list[tuple[float, float]]) -> float:
    return sum(math.dist(a, b) for a, b in zip(points, points[1:]))


def _point_at(points: list[tuple[float, float]], s: float) -> tuple[float, float]:
    remaining = s
    for a, b in zip(points, points[1:]):
        seg = math.dist(a, b)
        if remaining <= seg or seg == 0:
            t = 0.0 if seg == 0 else remaining / seg
            return (a[0] + (b[0] - a[0]) * t, a[1] + (b[1] - a[1]) * t)
        remaining -= seg
    return points[-1]


def _rejection_loop(rng: random.Random) -> dict:
    w, h = CANVAS
    lo, hi = MARGIN_PAD, w - MARGIN_PAD
    for _ in range(20000):
        lane_vert_gap = LANE_GAP
        n_lanes = rng.randint(4, 7)
        y_lo = lo + 10
        y_hi = hi - 10
        usable_h = y_hi - y_lo
        span_h = (n_lanes - 1) * lane_vert_gap
        span_h = min(span_h, usable_h)
        step_h = span_h / (n_lanes - 1) if n_lanes > 1 else 0.0
        x_start = rng.uniform(lo + 15, lo + 50)
        x_end = rng.uniform(hi - 50, hi - 15)
        path = []
        for lane in range(n_lanes):
            y = y_lo + lane * step_h
            if lane % 2 == 0:
                path.append((x_start, y))
                path.append((x_end, y))
            else:
                path.append((x_end, y))
                path.append((x_start, y))
        path = _pts(path)
        total = _path_length(path)
        if total < END_MARGIN * 5:
            continue

        s_port = END_MARGIN * rng.uniform(1.0, 1.5)
        lo_s = s_port + 2 * ARC_SEP
        hi_s = total - END_MARGIN - 20
        if hi_s - lo_s < 6 * ARC_SEP:
            continue

        # Two terminal positions, well separated.
        s_a = rng.uniform(lo_s, hi_s)
        s_b = rng.uniform(lo_s, hi_s)
        if abs(s_a - s_b) < 2 * ARC_SEP:
            continue
        lo_term, hi_term = min(s_a, s_b), max(s_a, s_b)
        hi_term = min(hi_term, hi_s)

        # Random distinct-colour assignment for the two terminals.
        if rng.random() < 0.5:
            term_colors = {"red": s_a, "green": s_b}
        else:
            term_colors = {"red": s_b, "green": s_a}

        # Choose the count class: how many terminals precede the checkpoint.
        cls = rng.choice(CLASSES)
        if cls == "0":
            s_cp = rng.uniform(lo_s, lo_term - ARC_SEP)
        elif cls == "1":
            s_cp = rng.uniform(lo_term + ARC_SEP, hi_term - ARC_SEP)
        else:  # "2"
            s_cp = rng.uniform(hi_term + ARC_SEP, hi_s)
        if not (lo_s <= s_cp <= hi_s):
            continue

        marks = [
            {"x": float(_point_at(path, s) [0]), "y": float(_point_at(path, s) [1]), "color": color}
            for color, s in (("red", term_colors["red"]), ("green", term_colors["green"]))
        ]
        cp_pos = _point_at(path, s_cp)
        marks.append({"x": float(cp_pos[0]), "y": float(cp_pos[1]), "color": CHECKPOINT_COLOR})

        port = _point_at(path, s_port)
        allgood = True
        for m in marks + [{"x": port[0], "y": port[1]}]:
            if not (20 < m["x"] < w - 20 and 20 < m["y"] < h - 20):
                allgood = False
                break
        if not allgood:
            continue

        scene = {
            "path": path,
            "port": [float(port[0]), float(port[1])],
            "marks": marks,
            "checkpoint": [float(cp_pos[0]), float(cp_pos[1])],
            "cp_arc": float(s_cp),
            "class": cls,
        }
        return scene
    raise RuntimeError("could not satisfy path constraints")


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    return _rejection_loop(rng)


def _draw_polyline(draw: ImageDraw.ImageDraw, pts, color, width):
    for a, b in zip(pts, pts[1:]):
        draw.line([a, b], fill=color, width=width, joint="curve")


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", CANVAS, BG)
    draw = ImageDraw.Draw(img)
    path = _pts(scene["path"])
    _draw_polyline(draw, path, COLOR_BLUE, STROKE)
    colors = {
        "red": COLOR_RED,
        "green": COLOR_GREEN,
        "magenta": COLOR_MAGENTA,
    }
    for mark in scene["marks"]:
        r = MARK_R
        px, py = float(mark["x"]), float(mark["y"])
        draw.ellipse([px - r, py - r, px + r, py + r], fill=colors[mark["color"]])
    px, py = float(scene["port"][0]), float(scene["port"][1])
    draw.ellipse([px - PORT_R, py - PORT_R, px + PORT_R, py + PORT_R], fill=COLOR_ORANGE)
    return img


def _arc_position(scene: dict, pt: tuple[float, float]) -> float:
    best = None
    best_d = float("inf")
    cum = 0.0
    for a, b in zip(_pts(scene["path"]), _pts(scene["path"])[1:]):
        abx, aby = b[0] - a[0], b[1] - a[1]
        seg2 = abx * abx + aby * aby
        t = 0.0 if seg2 == 0 else ((pt[0] - a[0]) * abx + (pt[1] - a[1]) * aby) / seg2
        t = max(0.0, min(1.0, t))
        proj = (a[0] + abx * t, a[1] + aby * t)
        d = math.dist(pt, proj)
        if d < best_d:
            best_d = d
            best = cum + math.dist(a, proj)
        cum += math.dist(a, b)
    return best


def _magenta_position(scene: dict) -> tuple[float, float]:
    for m in scene["marks"]:
        if m["color"] == CHECKPOINT_COLOR:
            return (float(m["x"]), float(m["y"]))
    raise KeyError("magenta checkpoint missing")


def analytic_gold(scene: dict) -> str:
    """Count of the two terminals (red, green) reached strictly before magenta."""
    cp = _arc_position(scene, _magenta_position(scene))
    count = 0
    for m in scene["marks"]:
        if m["color"] in TERMINAL_COLORS:
            if _arc_position(scene, (m["x"], m["y"])) < cp:
                count += 1
    return str(count)


def margin(scene: dict) -> float:
    """Smallest on-path separation between the checkpoint and any terminal."""
    cp = _arc_position(scene, _magenta_position(scene))
    best = float("inf")
    for m in scene["marks"]:
        if m["color"] in TERMINAL_COLORS:
            d = abs(_arc_position(scene, (m["x"], m["y"])) - cp)
            best = min(best, d)
    return float(best)


def is_quarantined(scene: dict) -> bool:
    w, h = CANVAS
    if margin(scene) < ARC_SEP * 1.5:
        return True
    port = tuple(scene["port"])
    if abs(_arc_position(scene, port) - _arc_position(scene, _magenta_position(scene))) < ARC_SEP:
        return True
    for m in scene["marks"] + [{"x": port[0], "y": port[1]}]:
        if not (20 <= m["x"] <= w - 20 and 20 <= m["y"] <= h - 20):
            return True
    return False


def latent_symmetries(scene: dict) -> list:
    return []
