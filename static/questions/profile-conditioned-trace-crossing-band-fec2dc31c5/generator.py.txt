"""trace_crossing_band — where do the two named traces exchange order?

A plot frame is cut into four vertical bands by three dashed black partition
lines placed at irregular positions. A red and a blue polyline share the same
vertex abscissas and cross each other exactly once; a green polyline is drawn
underneath as a distractor. The answer is the *band number* that contains the
red/blue crossing abscissa.

Nothing is inked at the answer-bearing location: the crossing abscissa has to be
reconstructed from the reversal of the vertical order of the two named traces,
and only then read against the visible partition.

Quantization note (this world's only latent alias): every ordinate is painted at
the integer row ``round(y)``, so all ordinate lists inside one rounding bucket
render byte-identically. Gold is therefore solved from the *quantized* vertical
difference sequence, and ``latent_symmetries`` returns explicit sub-pixel
ordinate perturbations that stay inside those buckets.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "trace_crossing_band-1.0.0"

# --- fixed page geometry (integer raster, no antialiasing anywhere) ------------
WIDTH = 760
HEIGHT = 470
FRAME_X0 = 68
FRAME_X1 = 724
FRAME_Y0 = 44
FRAME_Y1 = 414
VERTEX_COUNT = 7
TRACE_X0 = FRAME_X0 + 10
TRACE_X1 = FRAME_X1 - 10
TRACE_Y0 = FRAME_Y0 + 24
TRACE_Y1 = FRAME_Y1 - 24
BAND_COUNT = 4
MIN_BAND_WIDTH = 118
EDGE_CLEAR = 26.0        # off-quarantine crossings keep at least this clearance
QUARANTINE_PX = 14.0     # crossings nearer than this to a partition are boundary cases
NEAR_LO = 6.5            # sampled boundary cases live in [NEAR_LO, NEAR_HI)
NEAR_HI = 13.0
MIN_GAP = 20             # smallest vertical separation at any shared abscissa
MAX_GAP = 160
T_LO = 0.20              # crossing stays inside the middle of a segment
T_HI = 0.80

INK = (25, 25, 25)
GRID = (205, 205, 205)
BACKGROUND = (255, 255, 255)
RED = (214, 39, 40)
BLUE = (31, 119, 180)
GREEN = (44, 160, 44)

DASH_ON = 10
DASH_OFF = 8
TRACE_WIDTH = 3


def _xs() -> list[int]:
    span = TRACE_X1 - TRACE_X0
    return [TRACE_X0 + round(index * span / (VERTEX_COUNT - 1)) for index in range(VERTEX_COUNT)]


# --- label-relevant quantities -------------------------------------------------
# Only ``xs``, ``red_y``, ``blue_y`` and ``band_edges`` may reach the analytic
# gold. All four are drawn: the two traces carry the order reversal and the
# partition lines carry the band addresses.


def _quantized_gaps(scene: dict) -> list[int]:
    """Vertical difference red-minus-blue at each shared abscissa, in painted rows."""
    reds = [int(round(float(value))) for value in scene["red_y"]]
    blues = [int(round(float(value))) for value in scene["blue_y"]]
    return [red - blue for red, blue in zip(reds, blues)]


def _crossing_abscissa(scene: dict) -> float:
    """Abscissa where the painted vertical order of the two named traces reverses."""
    gaps = _quantized_gaps(scene)
    xs = [float(value) for value in scene["xs"]]
    for index in range(len(gaps) - 1):
        left, right = gaps[index], gaps[index + 1]
        if left == 0 or right == 0 or (left < 0) == (right < 0):
            continue
        step = xs[index + 1] - xs[index]
        return xs[index] + step * (-left) / (right - left)
    return 0.5 * (xs[0] + xs[-1])


def _sign_changes(scene: dict) -> int:
    gaps = _quantized_gaps(scene)
    return sum(
        1
        for index in range(len(gaps) - 1)
        if gaps[index] != 0 and gaps[index + 1] != 0 and (gaps[index] < 0) != (gaps[index + 1] < 0)
    )


def _band_index(scene: dict) -> int:
    crossing = _crossing_abscissa(scene)
    return 1 + sum(1 for edge in scene["band_edges"] if float(edge) < crossing)


def _edge_clearance(scene: dict) -> float:
    crossing = _crossing_abscissa(scene)
    return min(abs(crossing - float(edge)) for edge in scene["band_edges"])


def analytic_gold(scene: dict) -> str:
    return str(_band_index(scene))


def margin(scene: dict) -> float:
    """Pixel clearance between the crossing abscissa and the nearest partition."""
    return float(_edge_clearance(scene))


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_PX)


# --- diagnostics (never consulted by the analytic gold) ------------------------


def crossing_abscissa(scene: dict) -> float:
    """Public read-out of the reconstructed crossing abscissa, for diagnostics."""
    return float(_crossing_abscissa(scene))


def in_widest_band(scene: dict) -> bool:
    """Whether the answer band is also the widest band, a cheap-cue diagnostic."""
    bounds = [float(FRAME_X0), *(float(edge) for edge in scene["band_edges"]), float(FRAME_X1)]
    widths = [bounds[i + 1] - bounds[i] for i in range(BAND_COUNT)]
    return bool(_band_index(scene) - 1 == widths.index(max(widths)))


# --- sampling ------------------------------------------------------------------


def _sample_edges(rng: random.Random) -> list[int]:
    while True:
        edges = sorted(rng.randint(FRAME_X0 + 4, FRAME_X1 - 4) for _ in range(BAND_COUNT - 1))
        bounds = [FRAME_X0, *edges, FRAME_X1]
        if all(bounds[i + 1] - bounds[i] >= MIN_BAND_WIDTH for i in range(BAND_COUNT)):
            return edges


def _segment_of(xs: list[int], value: float) -> tuple[int, float] | None:
    for index in range(len(xs) - 1):
        if xs[index] <= value <= xs[index + 1]:
            step = xs[index + 1] - xs[index]
            return index, (value - xs[index]) / step
    return None


def _sample_crossing(
    rng: random.Random, edges: list[int], xs: list[int], band: int, near: bool
) -> tuple[float, int, float] | None:
    bounds = [float(FRAME_X0), *(float(edge) for edge in edges), float(FRAME_X1)]
    low, high = bounds[band - 1], bounds[band]
    low = max(low + (0.0 if band == 1 else EDGE_CLEAR), xs[0] + 16.0)
    high = min(high - (0.0 if band == BAND_COUNT else EDGE_CLEAR), xs[-1] - 16.0)
    if near:
        # A boundary case hugs one partition; the band it lands in follows from
        # the side, so ``band`` only selects which partition is used.
        edge = float(edges[min(band, BAND_COUNT - 1) - 1])
        side = 1.0 if rng.random() < 0.5 else -1.0
        target = edge + side * rng.uniform(NEAR_LO, NEAR_HI)
        if not (xs[0] + 16.0 <= target <= xs[-1] - 16.0):
            return None
        found = _segment_of(xs, target)
        if found is None or not (T_LO <= found[1] <= T_HI):
            return None
        return target, found[0], found[1]
    if high - low < 8.0:
        return None
    for _ in range(200):
        target = rng.uniform(low, high)
        found = _segment_of(xs, target)
        if found is None or not (T_LO <= found[1] <= T_HI):
            continue
        return target, found[0], found[1]
    return None


def _gap_magnitudes(rng: random.Random, share: float) -> tuple[float, float] | None:
    ratio = share / (1.0 - share)
    low = max(float(MIN_GAP), MIN_GAP / ratio)
    high = min(float(MAX_GAP), MAX_GAP / ratio)
    if high <= low:
        return None
    left = rng.uniform(low, high)
    return left, left * ratio


def sample_scene(seed: int) -> dict:
    rng = random.Random((int(seed) * 7919 + 15485863) & 0xFFFFFFFF)
    xs = _xs()
    near = rng.random() < 0.18

    for _ in range(4000):
        edges = _sample_edges(rng)
        band = rng.randint(1, BAND_COUNT)
        picked = _sample_crossing(rng, edges, xs, band, near)
        if picked is None:
            continue
        target, index, share = picked
        magnitudes = _gap_magnitudes(rng, share)
        if magnitudes is None:
            continue
        left_magnitude, right_magnitude = magnitudes
        red_starts_above = rng.random() < 0.5
        left_sign = -1.0 if red_starts_above else 1.0

        gaps: list[float] = []
        for position in range(VERTEX_COUNT):
            sign = left_sign if position <= index else -left_sign
            if position == index:
                size = left_magnitude
            elif position == index + 1:
                size = right_magnitude
            else:
                size = rng.uniform(MIN_GAP + 6, MAX_GAP)
            gaps.append(sign * size)

        red_y: list[float] = []
        blue_y: list[float] = []
        for gap in gaps:
            half = abs(gap) / 2.0
            middle = rng.uniform(TRACE_Y0 + half, TRACE_Y1 - half)
            red_y.append(middle + gap / 2.0)
            blue_y.append(middle - gap / 2.0)

        green_y = [rng.uniform(TRACE_Y0, TRACE_Y1) for _ in range(VERTEX_COUNT)]
        scene = {
            "xs": xs,
            "red_y": [round(value, 4) for value in red_y],
            "blue_y": [round(value, 4) for value in blue_y],
            "green_y": [round(value, 4) for value in green_y],
            "band_edges": edges,
            "marker_size": rng.choice([2, 3]),
            "grid_rows": sorted(
                rng.sample(range(FRAME_Y0 + 30, FRAME_Y1 - 30, 14), rng.randint(3, 4))
            ),
            "red_last": rng.random() < 0.5,
        }
        if _sign_changes(scene) != 1:
            continue
        clearance = _edge_clearance(scene)
        if near:
            if not (NEAR_LO - 0.5 <= clearance < QUARANTINE_PX):
                continue
        elif clearance < EDGE_CLEAR - 2.0:
            continue
        if abs(_crossing_abscissa(scene) - target) > 6.0:
            continue
        return scene
    raise RuntimeError("scene sampler failed to converge")


# --- rendering -----------------------------------------------------------------


def _draw_trace(
    draw: ImageDraw.ImageDraw, xs: list[int], ys: list[float], color: tuple, size: int
) -> None:
    points = [(int(x), int(round(float(y)))) for x, y in zip(xs, ys)]
    draw.line(points, fill=color, width=TRACE_WIDTH, joint="curve")
    for x, y in points:
        draw.rectangle([x - size, y - size, x + size, y + size], fill=color)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    for row in scene["grid_rows"]:
        draw.line([(FRAME_X0 + 2, int(row)), (FRAME_X1 - 2, int(row))], fill=GRID, width=1)

    for edge in scene["band_edges"]:
        y = FRAME_Y0 + 3
        while y < FRAME_Y1 - 3:
            bottom = min(y + DASH_ON, FRAME_Y1 - 3)
            draw.line([(int(edge), y), (int(edge), bottom)], fill=INK, width=3)
            y = bottom + DASH_OFF

    draw.rectangle([FRAME_X0, FRAME_Y0, FRAME_X1, FRAME_Y1], outline=INK, width=2)

    xs = [int(value) for value in scene["xs"]]
    size = int(scene["marker_size"])
    _draw_trace(draw, xs, scene["green_y"], GREEN, size)
    order = [(scene["blue_y"], BLUE), (scene["red_y"], RED)]
    if not scene["red_last"]:
        order.reverse()
    for ys, color in order:
        _draw_trace(draw, xs, ys, color, size)
    return image


# --- declared latent aliases ---------------------------------------------------


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Sub-pixel ordinate jitter that provably stays inside the painted buckets."""
    out: list[tuple[str, dict]] = []
    for step, shift in enumerate((0.31, -0.29, 0.44, -0.41), start=1):
        twin = dict(scene)
        for key in ("red_y", "blue_y", "green_y"):
            twin[key] = [
                round(float(int(round(float(value)))) + shift * (1 if (i + step) % 2 else -1), 4)
                for i, value in enumerate(scene[key])
            ]
        out.append((f"subpixel_jitter_{step}", twin))
    return out
