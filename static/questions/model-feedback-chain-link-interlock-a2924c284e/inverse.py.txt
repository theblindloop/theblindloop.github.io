"""Scene-free pixel oracle: which colored loop is chained through the black loop.

Reads only the rasterized image.  Segments the exact ink hues, fits a circle to
each loop's surviving pixels, computes where each colored loop meets the black
loop, and at each meeting point decides which strand is unbroken.  A loop is
chained when the unbroken strand is not the same at its two meeting points.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

INK = {
    "black": (0, 0, 0),
    "green": (0, 150, 60),
    "orange": (240, 130, 0),
    "violet": (140, 60, 210),
}
COLORS = ("green", "orange", "violet")
CHECK_R = 14.0


def _pixels(arr: np.ndarray, rgb) -> np.ndarray:
    hit = (arr[:, :, 0] == rgb[0]) & (arr[:, :, 1] == rgb[1]) & (arr[:, :, 2] == rgb[2])
    ys, xs = np.nonzero(hit)
    return np.stack([xs + 0.5, ys + 0.5], axis=1)


def _fit_circle(points: np.ndarray):
    """Geometric circle fit; stable on an annulus arc with pieces missing."""
    x = points[:, 0]
    y = points[:, 1]
    cx = float(x.mean())
    cy = float(y.mean())
    radius = float(np.hypot(x - cx, y - cy).mean())
    for _ in range(200):
        dx = cx - x
        dy = cy - y
        dist = np.hypot(dx, dy)
        dist = np.where(dist < 1e-9, 1e-9, dist)
        radius = float(dist.mean())
        nx = float((dx / dist).mean())
        ny = float((dy / dist).mean())
        new_cx = float(x.mean()) + radius * nx
        new_cy = float(y.mean()) + radius * ny
        if abs(new_cx - cx) < 1e-7 and abs(new_cy - cy) < 1e-7:
            cx, cy = new_cx, new_cy
            break
        cx, cy = new_cx, new_cy
    return (cx, cy), radius


def _intersections(c0, r0, c1, r1):
    dx = c1[0] - c0[0]
    dy = c1[1] - c0[1]
    dist = math.hypot(dx, dy)
    if dist <= 1e-9 or dist >= r0 + r1 or dist <= abs(r0 - r1):
        return None
    a = (dist * dist + r0 * r0 - r1 * r1) / (2.0 * dist)
    h_sq = r0 * r0 - a * a
    if h_sq <= 0.0:
        return None
    h = math.sqrt(h_sq)
    mx = c0[0] + a * dx / dist
    my = c0[1] + a * dy / dist
    ux, uy = -dy / dist, dx / dist
    return ((mx + h * ux, my + h * uy), (mx - h * ux, my - h * uy))


def _present(points: np.ndarray, spot) -> bool:
    if points.size == 0:
        return False
    near = np.hypot(points[:, 0] - spot[0], points[:, 1] - spot[1]) <= CHECK_R
    return bool(near.any())


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    black_pts = _pixels(arr, INK["black"])
    if black_pts.shape[0] < 200:
        return ""
    black_c, black_r = _fit_circle(black_pts)

    chained = []
    for color in COLORS:
        pts = _pixels(arr, INK[color])
        if pts.shape[0] < 200:
            continue
        centre, radius = _fit_circle(pts)
        meet = _intersections(black_c, black_r, centre, radius)
        if meet is None:
            continue
        over = []
        for spot in meet:
            colored_here = _present(pts, spot)
            black_here = _present(black_pts, spot)
            if colored_here == black_here:
                over.append(None)
            else:
                over.append(colored_here)
        if None in over:
            continue
        if over[0] != over[1]:
            chained.append(color)

    if len(chained) != 1:
        return ""
    return chained[0]
