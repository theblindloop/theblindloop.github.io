"""Deterministic latent scene, raster, analytic gold and margin.

Four wire loops are drawn on white paper: one black loop and three colored
loops, each colored loop crossing the black loop exactly twice.  At every
crossing the strand that runs behind is broken, so the picture records an
over/under decision at each of the six crossings.  A colored loop is chained
through the black loop when it runs over at one of its two crossings and under
at the other; a loop that runs over at both crossings lies loose on top of the
black loop and a loop that runs under at both lies loose beneath it, and both
of those come apart without cutting.  The label is therefore a linking
invariant of a pair of closed curves, not a property of any drawn region.
"""

from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image

CANVAS = 860
CX = CY = CANVAS / 2.0
STROKE = 16.0          # wire width in pixels
HALF = STROKE / 2.0
GAP_R = 30.0           # radius of the break cut into the under-running strand
MIN_MARGIN = 11.0

RING_COLORS = ("green", "orange", "violet")

PALETTE = {
    "background": (255, 255, 255),
    "black": (0, 0, 0),
    "green": (0, 150, 60),
    "orange": (240, 130, 0),
    "violet": (140, 60, 210),
}


# --------------------------------------------------------------------------
# geometry helpers
# --------------------------------------------------------------------------
def _intersections(c0, r0, c1, r1):
    """Two intersection points of circles (c0, r0) and (c1, r1), or None."""
    dx = c1[0] - c0[0]
    dy = c1[1] - c0[1]
    dist = math.hypot(dx, dy)
    if dist <= 1e-9 or dist >= r0 + r1 or dist <= abs(r0 - r1):
        return None
    a = (dist * dist + r0 * r0 - r1 * r1) / (2.0 * dist)
    h_sq = r0 * r0 - a * a
    if h_sq <= 0.0:
        return None
    h = math.sqrt(h_sq)
    mx = c0[0] + a * dx / dist
    my = c0[1] + a * dy / dist
    ux, uy = -dy / dist, dx / dist
    return ((mx + h * ux, my + h * uy), (mx - h * ux, my - h * uy))


def _cross_angle(r0, r1, dist):
    """Angle between the two circles at an intersection point, in radians."""
    cos_a = (r0 * r0 + r1 * r1 - dist * dist) / (2.0 * r0 * r1)
    cos_a = max(-1.0, min(1.0, cos_a))
    return math.acos(cos_a)


def _lens_area(r0, r1, d):
    """Area shared by the two filled disks; a proxy for how deep the overlap looks."""
    if d >= r0 + r1:
        return 0.0
    if d <= abs(r0 - r1):
        return math.pi * min(r0, r1) ** 2
    a0 = math.acos((d * d + r0 * r0 - r1 * r1) / (2 * d * r0))
    a1 = math.acos((d * d + r1 * r1 - r0 * r0) / (2 * d * r1))
    return (
        r0 * r0 * (a0 - math.sin(2 * a0) / 2.0)
        + r1 * r1 * (a1 - math.sin(2 * a1) / 2.0)
    )


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------
def _try_scene(rng: random.Random):
    big_r = rng.uniform(200.0, 218.0)
    base = rng.uniform(0.0, 2 * math.pi)
    offsets = [0.0, 2 * math.pi / 3, 4 * math.pi / 3]
    rings = []
    for k in range(3):
        ang = base + offsets[k] + math.radians(rng.uniform(-11.0, 11.0))
        dist = rng.uniform(190.0, 240.0)
        rad = rng.uniform(130.0, 142.0)
        rings.append(
            {
                "color": RING_COLORS[k],
                "cx": CX + dist * math.cos(ang),
                "cy": CY + dist * math.sin(ang),
                "r": rad,
                "d": dist,
            }
        )

    # every colored loop must cross the black loop transversally, twice
    for ring in rings:
        pts = _intersections((CX, CY), big_r, (ring["cx"], ring["cy"]), ring["r"])
        if pts is None:
            return None
        alpha = _cross_angle(big_r, ring["r"], ring["d"])
        if not (math.radians(45.0) <= alpha <= math.radians(135.0)):
            return None
        if math.hypot(pts[0][0] - pts[1][0], pts[0][1] - pts[1][1]) < 130.0:
            return None
        ring["pts"] = [list(pts[0]), list(pts[1])]
        ring["alpha"] = alpha
        # stay well inside the sheet
        if ring["d"] + ring["r"] + HALF > CX - 20.0:
            return None

    # colored loops never touch one another
    for i in range(3):
        for j in range(i + 1, 3):
            a, b = rings[i], rings[j]
            sep = math.hypot(a["cx"] - b["cx"], a["cy"] - b["cy"])
            if sep < a["r"] + b["r"] + 40.0:
                return None

    # every break must sit clear of the third loop and of every other break
    for i, ring in enumerate(rings):
        for p in ring["pts"]:
            for j, other in enumerate(rings):
                if i == j:
                    continue
                gap = abs(math.hypot(p[0] - other["cx"], p[1] - other["cy"]) - other["r"])
                if gap < GAP_R + HALF + 14.0:
                    return None
                for q in other["pts"]:
                    if math.hypot(p[0] - q[0], p[1] - q[1]) < 100.0:
                        return None

    # one loop alternates (chained); one runs over both crossings, one under both
    answer = rng.randrange(3)
    others = [k for k in range(3) if k != answer]
    rng.shuffle(others)
    flags = {answer: [True, False], others[0]: [True, True], others[1]: [False, False]}
    if rng.random() < 0.5:
        flags[answer] = [False, True]
    for k, ring in enumerate(rings):
        ring["over"] = list(flags[k])

    # the chained loop must not be the deepest overlap, the nearest loop, or the
    # topmost loop on the page, so no salience rule selects it
    areas = [_lens_area(big_r, r["r"], r["d"]) for r in rings]
    if areas[answer] == max(areas):
        return None
    if rings[answer]["d"] == min(r["d"] for r in rings):
        return None
    if rings[answer]["cy"] == min(r["cy"] for r in rings):
        return None

    scene = {
        "canvas": CANVAS,
        "black": {"cx": CX, "cy": CY, "r": big_r},
        "rings": [
            {
                "color": r["color"],
                "cx": r["cx"],
                "cy": r["cy"],
                "r": r["r"],
                "crossings": [
                    {"x": r["pts"][0][0], "y": r["pts"][0][1], "over": bool(r["over"][0])},
                    {"x": r["pts"][1][0], "y": r["pts"][1][1], "over": bool(r["over"][1])},
                ],
            }
            for r in rings
        ],
    }
    if is_quarantined(scene):
        return None
    return scene


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    for _ in range(4000):
        scene = _try_scene(rng)
        if scene is not None:
            return scene
    raise RuntimeError("scene sampling failed")


# --------------------------------------------------------------------------
# raster
# --------------------------------------------------------------------------
def _band(dist: np.ndarray, radius: float) -> np.ndarray:
    return np.abs(dist - radius) <= HALF


def _dist_field(grid_x, grid_y, cx, cy):
    return np.hypot(grid_x - cx, grid_y - cy)


def render(scene: dict) -> Image.Image:
    n = int(scene["canvas"])
    ys, xs = np.mgrid[0:n, 0:n]
    grid_x = xs.astype(np.float64) + 0.5
    grid_y = ys.astype(np.float64) + 0.5

    black = scene["black"]
    black_mask = _band(_dist_field(grid_x, grid_y, black["cx"], black["cy"]), black["r"])

    ring_masks = []
    for ring in scene["rings"]:
        ring_masks.append(
            _band(_dist_field(grid_x, grid_y, ring["cx"], ring["cy"]), ring["r"])
        )

    for idx, ring in enumerate(scene["rings"]):
        for crossing in ring["crossings"]:
            cut = _dist_field(grid_x, grid_y, crossing["x"], crossing["y"]) <= GAP_R
            if crossing["over"]:
                black_mask &= ~cut
            else:
                ring_masks[idx] &= ~cut

    canvas = np.full((n, n, 3), PALETTE["background"], dtype=np.uint8)
    canvas[black_mask] = PALETTE["black"]
    for idx, ring in enumerate(scene["rings"]):
        canvas[ring_masks[idx]] = PALETTE[ring["color"]]
    return Image.fromarray(canvas, mode="RGB")


# --------------------------------------------------------------------------
# label, margin, quarantine, aliases
# --------------------------------------------------------------------------
def _chained(scene: dict) -> list:
    return [
        ring["color"]
        for ring in scene["rings"]
        if ring["crossings"][0]["over"] != ring["crossings"][1]["over"]
    ]


def analytic_gold(scene: dict) -> str:
    chained = _chained(scene)
    if len(chained) != 1:
        raise ValueError("scene has no unique chained loop")
    return chained[0]


def margin(scene: dict) -> float:
    black = scene["black"]
    values = []
    rings = scene["rings"]
    for i, ring in enumerate(rings):
        dist = math.hypot(ring["cx"] - black["cx"], ring["cy"] - black["cy"])
        alpha = _cross_angle(black["r"], ring["r"], dist)
        # length of clear break visible on either side of the crossing strand
        values.append(GAP_R - HALF / math.sin(alpha))
        p, q = ring["crossings"]
        values.append(math.hypot(p["x"] - q["x"], p["y"] - q["y"]) - 2.0 * GAP_R)
        for crossing in ring["crossings"]:
            for j, other in enumerate(rings):
                if i == j:
                    continue
                clear = abs(
                    math.hypot(crossing["x"] - other["cx"], crossing["y"] - other["cy"])
                    - other["r"]
                )
                values.append(clear - HALF - GAP_R)
                for far in other["crossings"]:
                    values.append(
                        math.hypot(crossing["x"] - far["x"], crossing["y"] - far["y"])
                        - 2.0 * GAP_R
                    )
        for j in range(i + 1, len(rings)):
            other = rings[j]
            sep = math.hypot(ring["cx"] - other["cx"], ring["cy"] - other["cy"])
            values.append(sep - ring["r"] - other["r"] - STROKE)
        values.append(
            scene["canvas"] / 2.0
            - 20.0
            - (dist + ring["r"] + HALF)
        )
    return float(round(min(values), 6))


def is_quarantined(scene: dict) -> bool:
    if len(_chained(scene)) != 1:
        return True
    return margin(scene) < MIN_MARGIN


def latent_symmetries(scene: dict) -> list:
    cycled = {**scene, "rings": scene["rings"][1:] + scene["rings"][:1]}
    swapped = {
        **scene,
        "rings": [
            {**ring, "crossings": list(reversed(ring["crossings"]))}
            for ring in scene["rings"]
        ],
    }
    return [
        ("ring_record_order", cycled),
        ("crossing_record_order_within_each_ring", swapped),
    ]


def close_boundary(scene: dict) -> dict:
    """Flip the over/under decision at the first crossing of the chained loop.

    The chained loop then runs over (or under) at both crossings and comes free,
    no loop is chained any more, and the question loses its unique answer.
    """
    gold = analytic_gold(scene)
    rings = []
    for ring in scene["rings"]:
        if ring["color"] == gold:
            first = {**ring["crossings"][0], "over": not ring["crossings"][0]["over"]}
            ring = {**ring, "crossings": [first, ring["crossings"][1]]}
        rings.append(ring)
    return {**scene, "rings": rings}
