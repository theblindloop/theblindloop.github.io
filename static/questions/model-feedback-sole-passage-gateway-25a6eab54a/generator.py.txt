"""Sole passage gateway: which coloured room lies on every route between two dots.

The scene is a 6 by 6 lattice of square rooms.  Every room boundary carries a
thick wall; a boundary that is a *doorway* is drawn with a central gap instead of
a solid run.  The doorway graph is one single connected free space, so region
size and marker load carry no information.  Two rooms hold a pure black terminal
dot.  Four rooms have a flat coloured floor.  Exactly one of the four coloured
rooms is an obligatory gateway: every walk between the two terminals passes
through it.  Each of the other three coloured rooms lies on some shortest route
between the terminals and yet can be avoided by another route, so exhibiting one
route can never settle the question.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = 1000
MARGIN_PX = 50
CELLS = 6
CELL_PX = 150
WALL_PX = 12
DOT_R = 30

BACKGROUND = (247, 246, 242)
WALL = (28, 30, 34)
TERMINAL = (0, 0, 0)

COLOURS: dict[str, tuple[int, int, int]] = {
    "orange": (235, 130, 25),
    "green": (26, 150, 60),
    "blue": (32, 82, 214),
    "pink": (240, 105, 185),
}
COLOUR_ORDER = ("orange", "green", "blue", "pink")

GAP_MIN = 48
GAP_MAX = 78
QUARANTINE_GAP = 56


# --------------------------------------------------------------------------- #
# lattice helpers
# --------------------------------------------------------------------------- #
def _neighbours(cell: tuple[int, int]) -> list[tuple[int, int]]:
    row, col = cell
    out = []
    for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
        nr, nc = row + dr, col + dc
        if 0 <= nr < CELLS and 0 <= nc < CELLS:
            out.append((nr, nc))
    return out


def _edge_key(a: tuple[int, int], b: tuple[int, int]) -> tuple[int, int, int, int]:
    first, second = sorted((a, b))
    return (first[0], first[1], second[0], second[1])


def _connected(cells: set[tuple[int, int]]) -> bool:
    """Is this set of rooms connected through lattice adjacency?"""
    if not cells:
        return False
    start = min(cells)
    seen = {start}
    stack = [start]
    while stack:
        current = stack.pop()
        for n in _neighbours(current):
            if n in cells and n not in seen:
                seen.add(n)
                stack.append(n)
    return len(seen) == len(cells)


def _graph(doorways: set[tuple[int, int, int, int]]) -> dict:
    adjacency: dict[tuple[int, int], list[tuple[int, int]]] = {}
    for row in range(CELLS):
        for col in range(CELLS):
            cell = (row, col)
            adjacency[cell] = [
                n for n in _neighbours(cell) if _edge_key(cell, n) in doorways
            ]
    return adjacency


def _reach(
    adjacency: dict, source: tuple[int, int], blocked: set[tuple[int, int]] = frozenset()
) -> dict[tuple[int, int], int]:
    """Breadth-first distances from source with a set of rooms sealed off."""
    if source in blocked:
        return {}
    seen = {source: 0}
    queue = [source]
    head = 0
    while head < len(queue):
        current = queue[head]
        head += 1
        for n in adjacency[current]:
            if n in seen or n in blocked:
                continue
            seen[n] = seen[current] + 1
            queue.append(n)
    return seen


def _spanning_edges(
    rng: random.Random, cells: list[tuple[int, int]], extra: int
) -> list[tuple[int, int, int, int]] | None:
    """A random spanning tree over these rooms plus `extra` cycle-making edges."""
    pool = set(cells)
    start = cells[rng.randrange(len(cells))]
    visited = {start}
    frontier = [start]
    edges: list[tuple[int, int, int, int]] = []
    while frontier:
        index = rng.randrange(len(frontier)) if rng.random() < 0.4 else len(frontier) - 1
        current = frontier[index]
        options = [n for n in _neighbours(current) if n in pool and n not in visited]
        if not options:
            frontier.pop(index)
            continue
        nxt = rng.choice(options)
        visited.add(nxt)
        edges.append(_edge_key(current, nxt))
        frontier.append(nxt)
    if len(visited) != len(pool):
        return None
    chosen = set(edges)
    spare = [
        _edge_key(a, b)
        for a in cells
        for b in _neighbours(a)
        if b in pool and a < b and _edge_key(a, b) not in chosen
    ]
    rng.shuffle(spare)
    edges.extend(spare[:extra])
    return sorted(set(edges))


def _manhattan(a: tuple[int, int], b: tuple[int, int]) -> int:
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


def _halves(rng: random.Random) -> tuple[set, set] | None:
    """Grow the lattice into two connected interlocking halves.

    The two halves are grown from random seeds, so the boundary between them is
    a jagged interlocking front rather than a straight line: the run of solid
    wall that only the gateway room bridges is never a single salient stroke.
    """
    all_cells = [(r, c) for r in range(CELLS) for c in range(CELLS)]
    seeds = rng.sample(all_cells, 2)
    owner = {cell: index for index, cell in enumerate(seeds)}
    quota = [rng.randrange(15, 22)]
    quota.append(len(all_cells) - quota[0])
    sizes = [1, 1]
    for _ in range(len(all_cells) * 12):
        if sum(sizes) == len(all_cells):
            break
        hungry = [i for i in (0, 1) if sizes[i] < quota[i]]
        if not hungry:
            return None
        index = rng.choice(hungry)
        options = sorted(
            {
                n
                for cell, who in owner.items()
                if who == index
                for n in _neighbours(cell)
                if n not in owner
            }
        )
        if not options:
            quota[index] = sizes[index]
            continue
        pick = options[rng.randrange(len(options))]
        owner[pick] = index
        sizes[index] += 1
    if len(owner) != len(all_cells):
        return None
    first = {cell for cell, who in owner.items() if who == 0}
    second = {cell for cell, who in owner.items() if who == 1}
    if min(len(first), len(second)) < 14:
        return None
    if not _connected(first) or not _connected(second):
        return None
    return first, second


# --------------------------------------------------------------------------- #
# sampling
# --------------------------------------------------------------------------- #
def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) * 6151 + 29)
    for _ in range(4000):
        split = _halves(rng)
        if split is None:
            continue
        first, second = split
        # the gateway room sits on the frontier and touches both halves
        frontier = [
            cell
            for cell in sorted(first)
            if any(n in second for n in _neighbours(cell))
        ] + [
            cell
            for cell in sorted(second)
            if any(n in first for n in _neighbours(cell))
        ]
        if not frontier:
            continue
        gateway = frontier[rng.randrange(len(frontier))]
        own, other = (first, second) if gateway in first else (second, first)
        side_a = set(own) - {gateway}
        side_b = set(other)
        if not _connected(side_a) or not _connected(side_b):
            continue
        if len(side_a) < 8 or len(side_b) < 8:
            continue
        edges_a = _spanning_edges(rng, sorted(side_a), rng.randrange(4, 8))
        edges_b = _spanning_edges(rng, sorted(side_b), rng.randrange(4, 8))
        if edges_a is None or edges_b is None:
            continue
        links_a = [n for n in _neighbours(gateway) if n in side_a]
        links_b = [n for n in _neighbours(gateway) if n in side_b]
        if not links_a or not links_b:
            continue
        rng.shuffle(links_a)
        rng.shuffle(links_b)
        gate_edges = [
            _edge_key(gateway, n) for n in links_a[: rng.randrange(1, min(2, len(links_a)) + 1)]
        ] + [
            _edge_key(gateway, n) for n in links_b[: rng.randrange(1, min(2, len(links_b)) + 1)]
        ]
        doorways = sorted(set(edges_a) | set(edges_b) | set(gate_edges))
        door_set = set(doorways)
        adjacency = _graph(door_set)
        # every route between the halves must use the gateway, by construction
        from_gate = _reach(adjacency, gateway)
        if len(from_gate) != CELLS * CELLS:
            continue
        dist_a = _reach(adjacency, gateway, blocked=side_b)
        dist_b = _reach(adjacency, gateway, blocked=side_a)
        far_a = [c for c in sorted(side_a) if dist_a.get(c, 0) >= 3]
        far_b = [c for c in sorted(side_b) if dist_b.get(c, 0) >= 3]
        if not far_a or not far_b:
            continue
        start = far_a[rng.randrange(len(far_a))]
        target = far_b[rng.randrange(len(far_b))]
        if _manhattan(start, target) < 4:
            continue
        # candidate distractors: on a shortest route to the gateway, yet avoidable
        pool: list[tuple[int, int]] = []
        for side, anchor, dist, blocked in (
            (side_a, start, dist_a, side_b),
            (side_b, target, dist_b, side_a),
        ):
            from_anchor = _reach(adjacency, anchor, blocked=blocked)
            span = from_anchor.get(gateway)
            if span is None:
                continue
            for cell in sorted(side):
                if cell in (start, target, gateway):
                    continue
                if from_anchor.get(cell) is None or dist.get(cell) is None:
                    continue
                if from_anchor[cell] + dist[cell] != span:
                    continue
                if _reach(adjacency, anchor, blocked={cell}).get(target) is None:
                    continue
                pool.append(cell)
        if len(pool) < 3:
            continue
        rng.shuffle(pool)
        picks = pool[:3]
        if len({(c in side_a) for c in picks}) < 2:
            continue
        rooms = [gateway] + picks
        if len(set(rooms)) != 4:
            continue
        degree = {c: len(adjacency[c]) for c in rooms}
        # block the "most between", "nearest the midpoint" and "fewest doors" reads
        between = {c: _manhattan(c, start) + _manhattan(c, target) for c in rooms}
        if min(between.values()) == between[gateway]:
            continue
        mid = ((start[0] + target[0]) / 2.0, (start[1] + target[1]) / 2.0)
        centre = {
            c: (c[0] - mid[0]) ** 2 + (c[1] - mid[1]) ** 2 for c in rooms
        }
        if min(centre.values()) == centre[gateway]:
            continue
        if min(degree[c] for c in picks) > degree[gateway]:
            continue
        colours = list(COLOUR_ORDER)
        rng.shuffle(colours)
        assignment = {colour: room for colour, room in zip(colours, rooms)}
        gap = rng.randrange(GAP_MIN, GAP_MAX + 1, 2)
        scene = {
            "canvas": CANVAS,
            "cells": CELLS,
            "cell_px": CELL_PX,
            "wall_px": WALL_PX,
            "gap_px": int(gap),
            "doorways": [list(edge) for edge in doorways],
            "terminals": [list(start), list(target)],
            "rooms": {
                colour: [int(assignment[colour][0]), int(assignment[colour][1])]
                for colour in COLOUR_ORDER
            },
            "draw_order": [COLOUR_ORDER.index(c) for c in colours],
        }
        try:
            analytic_gold(scene)
        except ValueError:
            continue
        return scene
    raise RuntimeError(f"failed to sample a scene for seed {seed}")


# --------------------------------------------------------------------------- #
# rendering
# --------------------------------------------------------------------------- #
def _origin(scene: dict) -> int:
    return (int(scene["canvas"]) - int(scene["cells"]) * int(scene["cell_px"])) // 2


def render(scene: dict) -> Image.Image:
    canvas = int(scene["canvas"])
    cells = int(scene["cells"])
    step = int(scene["cell_px"])
    half = int(scene["wall_px"]) // 2
    gap = int(scene["gap_px"])
    off = _origin(scene)
    doorways = {tuple(int(v) for v in edge) for edge in scene["doorways"]}

    image = Image.new("RGB", (canvas, canvas), BACKGROUND)
    draw = ImageDraw.Draw(image)

    for index in scene["draw_order"]:
        colour = COLOUR_ORDER[int(index)]
        row, col = (int(v) for v in scene["rooms"][colour])
        x0 = off + col * step + half
        y0 = off + row * step + half
        draw.rectangle([x0, y0, x0 + step - 2 * half - 1, y0 + step - 2 * half - 1],
                       fill=COLOURS[colour])

    def horizontal(row: int, col: int, open_gap: bool) -> None:
        y = off + row * step
        x0 = off + col * step
        x1 = x0 + step
        if open_gap:
            mid = (x0 + x1) // 2
            draw.rectangle([x0, y - half, mid - gap // 2, y + half - 1], fill=WALL)
            draw.rectangle([mid + gap // 2, y - half, x1 - 1, y + half - 1], fill=WALL)
        else:
            draw.rectangle([x0, y - half, x1 - 1, y + half - 1], fill=WALL)

    def vertical(row: int, col: int, open_gap: bool) -> None:
        x = off + col * step
        y0 = off + row * step
        y1 = y0 + step
        if open_gap:
            mid = (y0 + y1) // 2
            draw.rectangle([x - half, y0, x + half - 1, mid - gap // 2], fill=WALL)
            draw.rectangle([x - half, mid + gap // 2, x + half - 1, y1 - 1], fill=WALL)
        else:
            draw.rectangle([x - half, y0, x + half - 1, y1 - 1], fill=WALL)

    for row in range(cells):
        for col in range(cells + 1):
            if col == 0 or col == cells:
                vertical(row, col, False)
            else:
                vertical(row, col, _edge_key((row, col - 1), (row, col)) in doorways)
    for col in range(cells):
        for row in range(cells + 1):
            if row == 0 or row == cells:
                horizontal(row, col, False)
            else:
                horizontal(row, col, _edge_key((row - 1, col), (row, col)) in doorways)
    # square junction blocks so no doorway can leak around a wall end
    for row in range(cells + 1):
        for col in range(cells + 1):
            x = off + col * step
            y = off + row * step
            draw.rectangle([x - half, y - half, x + half - 1, y + half - 1], fill=WALL)

    for cell in scene["terminals"]:
        row, col = int(cell[0]), int(cell[1])
        cx = off + col * step + step // 2
        cy = off + row * step + step // 2
        draw.ellipse([cx - DOT_R, cy - DOT_R, cx + DOT_R, cy + DOT_R], fill=TERMINAL)
    return image


# --------------------------------------------------------------------------- #
# analytic gold
# --------------------------------------------------------------------------- #
def _separators(scene: dict) -> list[str]:
    door_set = {tuple(int(v) for v in edge) for edge in scene["doorways"]}
    adjacency = _graph(door_set)
    start = (int(scene["terminals"][0][0]), int(scene["terminals"][0][1]))
    target = (int(scene["terminals"][1][0]), int(scene["terminals"][1][1]))
    if _reach(adjacency, start).get(target) is None:
        raise ValueError("the two terminals are not connected at all")
    out: list[str] = []
    for colour in COLOUR_ORDER:
        room = (int(scene["rooms"][colour][0]), int(scene["rooms"][colour][1]))
        if room in (start, target):
            raise ValueError("a coloured room holds a terminal dot")
        if _reach(adjacency, start, blocked={room}).get(target) is None:
            out.append(colour)
    return out


def analytic_gold(scene: dict) -> str:
    winners = _separators(scene)
    if len(winners) != 1:
        raise ValueError(f"scene does not have exactly one gateway room: {winners}")
    return winners[0]


def margin(scene: dict) -> float:
    """Smallest pixel dimension that separates a doorway from a solid wall."""
    gap = int(scene["gap_px"])
    stub = (int(scene["cell_px"]) - gap) // 2
    return float(min(gap, stub * 2))


def is_quarantined(scene: dict) -> bool:
    return int(scene["gap_px"]) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    reversed_doors = dict(scene)
    reversed_doors["doorways"] = list(reversed(scene["doorways"]))
    swapped = dict(scene)
    swapped["terminals"] = [list(scene["terminals"][1]), list(scene["terminals"][0])]
    reordered = dict(scene)
    reordered["draw_order"] = list(reversed(scene["draw_order"]))
    return [
        ("reverse_doorway_serialisation", reversed_doors),
        ("swap_terminal_endpoints", swapped),
        ("reverse_floor_draw_order", reordered),
    ]
