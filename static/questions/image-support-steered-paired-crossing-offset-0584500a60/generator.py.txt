"""Renderer and analytic contract for the paired baseline-crossing world.

The visible question compares two independent calibration cards.  In each card a
single saturated trace crosses a dashed horizontal reference line exactly once.
The label is the order of the two crossing abscissae, normalized to each card's
own plotting width.  The cards are deliberately far apart and neither card is a
path connecting the other, so the comparison requires both disconnected pieces.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "paired_crossing_offset-renderer-0.1.0"
WIDTH = 768
HEIGHT = 420
PANEL_Y0 = 62
PANEL_Y1 = 372
LEFT_X0, LEFT_X1 = 42, 300
RIGHT_X0, RIGHT_X1 = 468, 726
PLOT_PAD_X = 24
PLOT_Y0, PLOT_Y1 = 112, 350

BACKGROUND = (250, 251, 248)
PANEL_EDGE = (92, 104, 112)
BASELINE = (62, 70, 78)
PANEL_FILL = (255, 255, 253)
TRACE_PALETTE = (
    (22, 119, 164),
    (183, 70, 62),
    (111, 78, 164),
    (22, 142, 105),
)

# The named bins make the evidence dense around the visible crossing-order
# boundary while keeping the actual raster separation well above one pixel.
MARGIN_BINS = {
    "far_yes": (0.18, 0.34),
    "near_yes": (0.075, 0.14),
    "boundary_yes": (0.032, 0.055),
    "boundary_no": (-0.055, -0.032),
    "near_no": (-0.14, -0.075),
    "far_no": (-0.34, -0.18),
}


def _bucket_for_seed(seed: int) -> str:
    return (
        "far_yes",
        "far_no",
        "near_yes",
        "near_no",
        "boundary_yes",
        "boundary_no",
    )[seed % 6]


def sample_scene(seed: int) -> dict:
    """Sample one deterministic scene from a seed, with answer-balanced margins."""
    rng = random.Random(int(seed))
    bucket = _bucket_for_seed(int(seed))
    lo, hi = MARGIN_BINS[bucket]
    delta = rng.uniform(lo, hi)
    anchor = rng.uniform(0.38, 0.62)
    left_cross = anchor - delta / 2.0
    right_cross = anchor + delta / 2.0
    trace_color = list(TRACE_PALETTE[rng.randrange(len(TRACE_PALETTE))])
    return {
        "left_cross": round(left_cross, 6),
        "right_cross": round(right_cross, 6),
        "left_baseline": round(rng.uniform(0.43, 0.62), 6),
        "right_baseline": round(rng.uniform(0.43, 0.62), 6),
        "left_amplitude": round(rng.uniform(58.0, 90.0), 6),
        "right_amplitude": round(rng.uniform(58.0, 90.0), 6),
        "left_curve": round(rng.uniform(0.05, 0.34), 6),
        "right_curve": round(rng.uniform(0.05, 0.34), 6),
        "trace_color": trace_color,
    }


def _panel_geometry(side: str) -> tuple[int, int]:
    if side == "left":
        return LEFT_X0, LEFT_X1
    if side == "right":
        return RIGHT_X0, RIGHT_X1
    raise ValueError(f"unknown panel {side!r}")


def _trace_points(scene: dict, side: str, crossing: float | None = None) -> list[tuple[int, int]]:
    x0, x1 = _panel_geometry(side)
    cross = scene[f"{side}_cross"] if crossing is None else crossing
    baseline_fraction = scene[f"{side}_baseline"]
    amplitude = scene[f"{side}_amplitude"]
    curve = scene[f"{side}_curve"]
    start = x0 + PLOT_PAD_X
    end = x1 - PLOT_PAD_X
    span = float(end - start)
    base_y = PLOT_Y0 + baseline_fraction * (PLOT_Y1 - PLOT_Y0)
    points: list[tuple[int, int]] = []
    # A monotone cubic displacement has a single, visible zero crossing while
    # retaining enough local shape to look like a measurement trace rather than
    # an answer-specific tick.
    for i in range(13):
        u = i / 12.0
        dx = u - cross
        displacement = amplitude * dx * (1.0 + curve * dx * dx)
        x = int(round(start + span * u))
        y = int(round(base_y + displacement))
        points.append((x, y))
    return points


def _draw_dashed_baseline(draw: ImageDraw.ImageDraw, x0: int, x1: int, y: int) -> None:
    start = x0 + PLOT_PAD_X
    end = x1 - PLOT_PAD_X
    dash = 14
    gap = 9
    x = start
    while x < end:
        draw.line((x, y, min(x + dash, end), y), fill=BASELINE, width=3)
        x += dash + gap


def _draw_panel(
    draw: ImageDraw.ImageDraw, scene: dict, side: str, crossing: float
) -> None:
    x0, x1 = _panel_geometry(side)
    draw.rectangle((x0, PANEL_Y0, x1, PANEL_Y1), fill=PANEL_FILL, outline=PANEL_EDGE, width=3)
    # Small, position-neutral card key: the oracle does not need to read text;
    # left/right identity is supplied by the public question itself.
    marker_x = x0 + 15
    marker_y = PANEL_Y0 + 15
    draw.ellipse((marker_x - 4, marker_y - 4, marker_x + 4, marker_y + 4), fill=PANEL_EDGE)
    baseline_fraction = scene[f"{side}_baseline"]
    base_y = int(round(PLOT_Y0 + baseline_fraction * (PLOT_Y1 - PLOT_Y0)))
    _draw_dashed_baseline(draw, x0, x1, base_y)
    points = _trace_points(scene, side, crossing)
    draw.line(points, fill=tuple(scene["trace_color"]), width=8, joint="curve")
    # The crossing itself is not marked; it must be measured from the trace and
    # reference line rather than read from an answer-colored token.


def render(scene: dict) -> Image.Image:
    """Render only from the public scene fields into a deterministic RGB PNG."""
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    # Keep the two decision-bearing fields explicit in this entry point.  This
    # is both readable and auditable: the visible trace positions are passed to
    # the two independent panel renderings, not hidden behind metadata.
    left_cross = scene["left_cross"]
    right_cross = scene["right_cross"]
    _draw_panel(draw, scene, "left", left_cross)
    _draw_panel(draw, scene, "right", right_cross)
    return image


def analytic_gold(scene: dict) -> str:
    """Return whether the left crossing precedes the right crossing."""
    left = scene["left_cross"]
    right = scene["right_cross"]
    return "yes" if left < right else "no"


def margin(scene: dict) -> float:
    """Positive margin means the left crossing is earlier."""
    return float(scene["right_cross"] - scene["left_cross"])


def is_quarantined(scene: dict) -> bool:
    """Quarantine near-coincident crossings whose raster order is fragile."""
    return abs(margin(scene)) <= 0.060


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """There are no unobserved render-invariant aliases in this parameterization."""
    return []
