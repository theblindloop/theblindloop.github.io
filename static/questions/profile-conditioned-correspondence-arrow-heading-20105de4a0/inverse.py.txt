"""Independent pixel-only inverse arm for the directed arrow-heading world.

Receives ONLY the final rasterized PNG and the public question contract (the
visual spec). It never imports the renderer, prompt, generator, verifier, or
analytic-gold code, and never reads scene data.

Public visual spec (this file's own specification):

  * the image contains four arrows stacked top to bottom in horizontal bands;
    the top one is the ``reference`` arrow (dark gray); the next three are
    candidate arrows A, B, C (each a distinct saturated color);
  * every arrow is a single solid stroke: a straight shaft ending in a pointed
    triangular head; it points in one of eight compass classes (the eight
    45-degree directions, e.g. east, northeast, north, ...);
  * the task: decide which candidate arrow (A, B, or C) points in the same
    compass direction as the reference arrow.

The oracle recovers, from pixels alone: each arrow as one connected region, its
shaft axis and the direction from the tail butt toward the pointed tip, the
pointing angle quantized to the nearest of eight octants, and finally which
candidate shares the reference's octant.
"""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "correspondence_arrow_heading-oracle-0.1.0"

_BACKGROUND_DIST = 30.0
_MIN_PX = 250
_GROUP_TOL = 100.0
_OCTANTS = 8


def _is_background(rgb: np.ndarray) -> np.ndarray:
    return np.linalg.norm(rgb - np.array([255.0, 255.0, 255.0]), axis=-1) < _BACKGROUND_DIST


def _connected_components(mask: np.ndarray) -> list[np.ndarray]:
    h, w = mask.shape
    visited = np.zeros((h, w), dtype=bool)
    comps = []
    for y in range(h):
        for x in range(w):
            if mask[y, x] and not visited[y, x]:
                queue = deque([(x, y)])
                visited[y, x] = True
                cur = []
                while queue:
                    cx, cy = queue.popleft()
                    cur.append((cx, cy))
                    for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                        nx_, ny_ = cx + dx, cy + dy
                        if 0 <= nx_ < w and 0 <= ny_ < h and mask[ny_, nx_] and not visited[ny_, nx_]:
                            visited[ny_, nx_] = True
                            queue.append((nx_, ny_))
                comps.append(np.array(cur, dtype=np.float32))
    return comps


def _arrow_octant(points: np.ndarray) -> int:
    """Quantize the pointing direction of one arrow to one of eight octants.

    The arrow is a straight shaft ending in a wide pointed head and a narrow
    flat tail butt, so its principal axis (PCA of the region) runs tail-to-tip.
    The two axial ends are the two extremes of the projection onto that axis;
    the end with the larger transverse width is the wide head, and the arrow
    points from the narrow tail end toward the wide head end. The pointing
    angle is that tail-to-head direction quantized to the nearest 45-degree
    compass octant.
    """
    mean = points.mean(axis=0)
    cov = np.cov(points.T)
    eigvals, eigvecs = np.linalg.eigh(cov)
    axis = eigvecs[:, int(np.argmax(eigvals))]
    proj = (points - mean) @ axis
    pmin = float(proj.min())
    pmax = float(proj.max())
    band = 10.0
    end_lo = points[proj <= pmin + band]
    end_hi = points[proj >= pmax - band]

    def width(end):
        perp = np.array([-axis[1], axis[0]])
        vals = end @ perp
        return float(vals.max() - vals.min())

    if width(end_hi) >= width(end_lo):
        tail, tip = end_lo, end_hi
    else:
        tail, tip = end_hi, end_lo
    tail_c = tail.mean(axis=0)
    tip_c = tip.mean(axis=0)
    deg = math.degrees(math.atan2(tip_c[1] - tail_c[1], tip_c[0] - tail_c[0])) % 360.0
    return int(round(deg / (360.0 / _OCTANTS))) % _OCTANTS


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.float32)
    mask = ~_is_background(arr)
    comps = [c for c in _connected_components(mask) if len(c) >= _MIN_PX]
    if len(comps) != 4:
        raise ValueError(f"expected 4 arrows, got {len(comps)}")

    arrows = []
    for comp in comps:
        center = comp.mean(axis=0)
        arrows.append({"y": float(center[1]), "points": comp})
    arrows.sort(key=lambda a: a["y"])

    octants = [_arrow_octant(a["points"]) for a in arrows]
    ref_octant = octants[0]
    for i in range(1, 4):
        if octants[i] == ref_octant:
            return "ABC"[i - 1]
    raise ValueError("no candidate reproduced the reference pointing direction")


if __name__ == "__main__":
    import sys

    img = Image.open(sys.argv[1]).convert("RGB")
    print(decision_from_image(img))
