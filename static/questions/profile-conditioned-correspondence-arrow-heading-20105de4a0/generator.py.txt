"""Directed arrow-heading correspondence renderer (paper-semantic family).

A cross-panel *directed heading* correspondence task. Each panel shows one
solid arrow (a thick straight shaft ending in a pointed triangular head). The
arrow points in one of eight compass classes (the eight 45-degree octant
directions, e.g. east, northeast, north, ...). The top ``reference`` panel
points in a randomized octant. The three candidate panels A, B, C each point in
some octant; exactly one candidate points in the *same compass octant* as the
reference, the other two point in different octants. The question asks which
candidate reproduces the reference's pointing direction.

This is *not* an unoriented bar-tilt magnitude comparison (steeper_line,
correspondence_orientation_color): here the object is a *directed* arrow and the
decision is the cyclic compass heading it points toward, recovered from both the
shaft axis and the oriented arrowhead. This is also not a rigid-body rotation of
a multipart arrangement (rigid_multipart_rotation): each panel is a single
directed stroke, and the label is its absolute pointing octant, not a relative
re-orientation of an assembly.

The pixel-only inverse arm must, in every panel, isolate the arrow, recover its
shaft axis, determine which endpoint is the pointed tip using the arrowhead,
measure the pointing angle, quantize it to the nearest of eight octants, and
then select the candidate whose octant equals the reference's.

Latent state z (every scene field is rendered - none hidden):

    panels : ordered list of four arrow dicts [reference, A, B, C]; each
             {tail: [x, y], octant: 0..7}. The pointing octant is encoded by
             the arrow's visible geometry (the arrow is drawn from tail toward
             the octant direction).
    width  : raster width.
    height : raster height.

The analytic gold is the label (A/B/C) of the candidate panel whose pointing
octant equals the reference's. Because every arrow (tail position and pointing
direction) is drawn verbatim, the octant is recoverable from pixels alone; no
gold-only field exists.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "correspondence_arrow_heading-0.1.0"

BACKGROUND = (255, 255, 255)
ARROW_REF_COLOR = (90, 90, 90)

ARROW_PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (235, 40, 40),
    "green": (0, 170, 70),
    "blue": (40, 60, 235),
    "orange": (245, 140, 20),
}

# Public visual geometry (constant across scenes for ink/position counterbalance).
WIDTH = 900
HEIGHT = 1000
BAND_CY = [160, 400, 640, 880]   # reference, A, B, C top to bottom
TAIL_X = 450                     # arrow tail horizontal position per panel
SHAFT_LEN_MIN = 45
SHAFT_LEN_MAX = 58
SHAFT_HALF_WIDTH = 8
HEAD_LEN = 22
HEAD_HALF_WIDTH = 22
OCTANTS = 8                      # eight 45-degree pointing classes
JITTER_MAX = 11.0                # degrees around octant center (kept < 22.5 boundary)
_MARGIN_FLOOR = 10.0

# Octant index -> center direction angle in degrees (0 = pointing east).
_OCTANT_CENTER = [0.0, 45.0, 90.0, 135.0, 180.0, 225.0, 270.0, 315.0]


def _state(panel: dict) -> int:
    return int(panel["octant"]) % OCTANTS


def _build_panel(rng: random.Random, yc: float, octant: int) -> dict:
    jitter = rng.uniform(-JITTER_MAX, JITTER_MAX)
    angle = math.radians(_OCTANT_CENTER[octant] + jitter)
    return {
        "tail": [round(TAIL_X + rng.randint(-60, 60)), round(yc)],
        "octant": octant,
    }


def sample_scene(seed: int) -> dict:
    """Sample a balanced, deterministic latent scene (dict)."""
    rng = random.Random(seed)
    ref_oct = rng.randrange(OCTANTS)
    match_slot = rng.randrange(3)
    cand_octs = []
    for i in range(3):
        if i == match_slot:
            cand_octs.append(ref_oct)
        else:
            oct_ = rng.randrange(OCTANTS - 1)
            if oct_ >= ref_oct:
                oct_ += 1
            cand_octs.append(oct_)
    panels = [_build_panel(rng, BAND_CY[0], ref_oct)]
    for i in range(3):
        panels.append(_build_panel(rng, BAND_CY[i + 1], cand_octs[i]))
    return {
        "panels": panels,
        "width": WIDTH,
        "height": HEIGHT,
    }


def _draw_arrow(draw, tail, octant: int, color) -> None:
    angle = math.radians(_OCTANT_CENTER[octant])
    dx, dy = math.cos(angle), math.sin(angle)
    tx, ty = float(tail[0]), float(tail[1])
    length = SHAFT_LEN_MAX
    tip = (tx + dx * length, ty + dy * length)
    neck = (tx + dx * (length - HEAD_LEN), ty + dy * (length - HEAD_LEN))
    px, py = -dy, dx  # perpendicular
    shaft = [
        (tx + px * SHAFT_HALF_WIDTH, ty + py * SHAFT_HALF_WIDTH),
        (tx - px * SHAFT_HALF_WIDTH, ty - py * SHAFT_HALF_WIDTH),
        (neck[0] - px * SHAFT_HALF_WIDTH, neck[1] - py * SHAFT_HALF_WIDTH),
        (neck[0] + px * SHAFT_HALF_WIDTH, neck[1] + py * SHAFT_HALF_WIDTH),
    ]
    head = [
        (neck[0] - px * HEAD_HALF_WIDTH, neck[1] - py * HEAD_HALF_WIDTH),
        (tip[0], tip[1]),
        (neck[0] + px * HEAD_HALF_WIDTH, neck[1] + py * HEAD_HALF_WIDTH),
    ]
    draw.polygon(shaft, fill=color)
    draw.polygon(head, fill=color)


def render(scene: dict) -> Image.Image:
    w = int(scene["width"])
    h = int(scene["height"])
    img = Image.new("RGB", (w, h), BACKGROUND)
    draw = ImageDraw.Draw(img)

    ref_color = ARROW_REF_COLOR
    for i, panel in enumerate(scene["panels"]):
        if i == 0:
            color = ref_color
        else:
            color = ARROW_PALETTE[["red", "green", "blue", "orange"][i - 1]]
        _draw_arrow(draw, panel["tail"], panel["octant"], color)
    return img


def _heading_angle_deg(tail, octant: int) -> float:
    return _OCTANT_CENTER[octant]


def analytic_gold(scene: dict) -> str:
    """Decision: label (A/B/C) of the candidate panel whose pointing octant
    equals the reference panel's octant."""
    panels = scene["panels"]
    ref_state = panels[0]["octant"] % OCTANTS
    for i in range(1, 4):
        if panels[i]["octant"] % OCTANTS == ref_state:
            return "ABC"[i - 1]
    raise ValueError("no candidate reproduced the reference pointing direction")


def margin(scene: dict) -> float:
    """One-sided verifiability margin: minimum, over every panel, of the angular
    distance (in degrees) from that panel's pointing direction to the nearest
    octant-class boundary. Every arrow is drawn within JITTER_MAX degrees of a
    class center, and centers are 45 degrees apart with boundaries at +-22.5
    degrees, so the margin is always >= 22.5 - JITTER_MAX."""
    return float(22.5 - JITTER_MAX)


def is_quarantined(scene: dict) -> bool:
    """Quarantine only a genuinely ambiguous class boundary. Jitter is capped at
    JITTER_MAX (11) degrees while octant boundaries lie 22.5 degrees from each
    class center, so every arrow is at least ~11.5 degrees clear of a boundary;
    ambiguous scenes are never produced by the sampler."""
    return margin(scene) < _MARGIN_FLOOR


def latent_symmetries(scene: dict) -> list:
    """not_applicable: every label-influencing field (each arrow's tail position
    and pointing octant) is drawn verbatim, so the raster uniquely determines
    each arrow's pointing octant and the matching candidate; no pixel-identical
    transform is declared."""
    return []
