"""Constellation polygon-sides world renderer.

Latent scene z = {sides, center, circumradius, dot_radius, canvas, background,
dot_color, dots}. A constellation of small filled disks is placed exactly on the
perimeter of a regular convex polygon (sides in {3,4,5,6}) together with a few
interior noise disks. The answer is the number of sides the polygon has, read
globally from the arrangement of the dots (a Gestalt / global-structure reveal):
the convex hull of the dot centroids is the polygon itself, and its vertex count
is the number of sides.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "constellation_polygon_sides-0.1.0"

SIDES_OPTIONS = (3, 4, 5, 6)
CIRCUMRADIUS_RANGE = (95.0, 150.0)
DOT_RADIUS = 6.0
DOTS_PER_EDGE = 5
CANVAS = 512
INTERIOR_NOISE_MAX = 3

# Repeated dots below this circumradius make the corners ambiguous at pixel
# scale (adjacent corner clusters merge). Sample strictly above the floor.
QUARANTINE_CIRCUMRADIUS = 60.0


def _regular_vertices(sides: int, radius: float) -> list[tuple[float, float]]:
    """The `sides` corner coordinates of a regular polygon on the circumcircle."""
    return [
        (radius * math.cos(2 * math.pi * k / sides), radius * math.sin(2 * math.pi * k / sides))
        for k in range(sides)
    ]


def sample_scene(seed: int) -> dict:
    """Deterministically sample a full latent scene from an integer seed."""
    rng = random.Random(seed)
    sides = rng.choice(SIDES_OPTIONS)
    circumradius = rng.uniform(*CIRCUMRADIUS_RANGE)
    cx = CANVAS / 2
    cy = CANVAS / 2
    verts = _regular_vertices(sides, circumradius)

    dots: list[list[float]] = []
    for k in range(sides):
        ax, ay = verts[k]
        bx, by = verts[(k + 1) % sides]
        for t in range(DOTS_PER_EDGE):
            f = t / (DOTS_PER_EDGE - 1)
            dots.append((ax + f * (bx - ax), ay + f * (by - ay)))

    inradius = abs(circumradius * math.cos(math.pi / sides))
    for _ in range(rng.randint(0, INTERIOR_NOISE_MAX)):
        ang = rng.uniform(0, 2 * math.pi)
        rad = rng.uniform(0.0, inradius * 0.7)
        dots.append((rad * math.cos(ang), rad * math.sin(ang)))

    return {
        "sides": sides,
        "center": [cx, cy],
        "circumradius": round(circumradius, 3),
        "dot_radius": DOT_RADIUS,
        "canvas": CANVAS,
        "background": "white",
        "dot_color": "black",
        "dots": [list(round(v, 3) for v in dot) for dot in dots],
    }


def render(scene: dict, supersample: int = 4) -> Image.Image:
    """Render the constellation using supersampled antialiased disk plotting."""
    sides = int(scene["sides"])
    if sides not in SIDES_OPTIONS:
        raise ValueError(f"unsupported polygon side count {sides}")
    width = int(scene["canvas"])
    ss = supersample
    img = Image.new("RGB", (width * ss, width * ss), scene["background"])
    draw = ImageDraw.Draw(img)
    radius = float(scene["dot_radius"]) * ss
    cx, cy = float(scene["center"][0]), float(scene["center"][1])
    for dot in scene["dots"]:
        dx = cx + float(dot[0])
        dy = cy + float(dot[1])
        draw.ellipse(
            [dx * ss - radius, dy * ss - radius, dx * ss + radius, dy * ss + radius],
            fill=scene["dot_color"],
        )
    return img.resize((width, width), Image.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """The answer is the polygon's side count as a string."""
    return str(scene["sides"])


def margin(scene: dict) -> float:
    """One-sided verifiability margin: polygon circumradius in pixels.

    Larger circumradii make the corner clusters more clearly resolvable; scenes
    whose circumradius drops near the resolution floor are quarantined.
    """
    return float(scene["circumradius"])


def is_quarantined(scene: dict) -> bool:
    """Exclude very small polygons where corners merge at pixel scale."""
    return bool(float(scene["circumradius"]) < QUARANTINE_CIRCUMRADIUS)


def latent_symmetries(scene: dict) -> list:
    """No declared latent symmetry.

    The scene is fully described by its (already visible) constellation; the
    side count is an injective, pixel-revealed property, so no render-invariant
    relabeling leaves both pixels and gold fixed. Declared empty.
    """
    return []
