"""Independent pixel-only inverse arm for the constellation polygon-sides world.

Receives only the rendered PNG. Segments the dark disks from the light
background by colour, takes each connected disk's centroid, computes the convex
hull of the centroids, and counts the "true" corner vertices (hull vertices
whose turn angle is far from a straight line). That count is the number of sides
of the outlined polygon.

This module imports neither renderer, prompts, generate, verify, nor any
analytic-gold or scene code, and never receives latent scene data.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

# Disks darker than this (sum of RGB) are treated as ink.
_INK_LUMINANCE_SUM = 400
# A disk must occupy at least this many pixels to count as a mark.
_MIN_COMPONENT_PIXELS = 8
# A hull vertex is a "real" corner when its turn angle (the change of direction
# at that vertex) is well below a straight line. Near-collinear edge disks turn
# by ~180 degrees and are excluded.
_STRAIGHT_LIMIT_DEG = 150.0


def _connected_centroids(mask: np.ndarray) -> list[tuple[float, float]]:
    """Return the (x, y) centroid of each 4-connected dark mark."""
    coords = set(map(tuple, np.argwhere(mask)))
    centroids: list[tuple[float, float]] = []
    visited: set[tuple[int, int]] = set()
    for start in coords:
        if start in visited:
            continue
        stack = [start]
        component: list[tuple[int, int]] = []
        while stack:
            cell = stack.pop()
            if cell in visited:
                continue
            visited.add(cell)
            component.append(cell)
            r, c = cell
            for nr, nc in ((r + 1, c), (r - 1, c), (r, c + 1), (r, c - 1)):
                if (nr, nc) in coords and (nr, nc) not in visited:
                    stack.append((nr, nc))
        if len(component) >= _MIN_COMPONENT_PIXELS:
            xs = [p[1] for p in component]
            ys = [p[0] for p in component]
            centroids.append((float(sum(xs)) / len(xs), float(sum(ys)) / len(ys)))
    return centroids


def _convex_hull(points: list[tuple[float, float]]) -> list[tuple[float, float]]:
    """Andrew's monotone chain; removes collinear boundary points."""
    pts = sorted(set((float(x), float(y)) for x, y in points))
    if len(pts) < 3:
        return pts

    def cross(o, a, b):
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

    lower: list[tuple[float, float]] = []
    for p in pts:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    upper: list[tuple[float, float]] = []
    for p in reversed(pts):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    return lower[:-1] + upper[:-1]


def _count_corners(hull: list[tuple[float, float]]) -> int:
    n = len(hull)
    count = 0
    for i in range(n):
        ax, ay = hull[(i - 1) % n]
        bx, by = hull[i]
        cx, cy = hull[(i + 1) % n]
        v1 = (ax - bx, ay - by)
        v2 = (cx - bx, cy - by)
        cross = v1[0] * v2[1] - v1[1] * v2[0]
        dot = v1[0] * v2[0] + v1[1] * v2[1]
        turn = math.degrees(math.atan2(abs(cross), dot))
        if turn < _STRAIGHT_LIMIT_DEG:
            count += 1
    return count


def decision_from_image(image: Image.Image) -> str:
    """Recover the polygon's side count from the rasterized PNG alone."""
    arr = np.asarray(image.convert("RGB")).astype(np.int32)
    mask = arr.sum(axis=2) < _INK_LUMINANCE_SUM
    centroids = _connected_centroids(mask)
    hull = _convex_hull(centroids)
    return str(_count_corners(hull))
