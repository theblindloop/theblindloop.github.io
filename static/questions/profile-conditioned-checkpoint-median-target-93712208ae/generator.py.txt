from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

COLORS = ("blue", "orange", "green")
DEFAULT_PALETTE = {
    "blue": [35, 105, 210],
    "orange": [230, 115, 25],
    "green": [25, 155, 85],
    "target": [185, 45, 155],
}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    canvas = [512, 360]
    target_y = rng.randint(145, 205)
    winner = int(seed) % 3
    other = [i for i in range(3) if i != winner]
    rng.shuffle(other)
    first = rng.randint(12, 24)
    runner_gap = rng.randint(9, 12) if int(seed) % 8 == 0 else rng.randint(20, 30)
    distances = [0, 0, 0]
    distances[winner] = first
    distances[other[0]] = first + runner_gap
    distances[other[1]] = first + runner_gap + rng.randint(15, 25)
    x_base = [100, 180, 260, 340, 420]
    points: dict[str, list[list[int]]] = {}
    for index, name in enumerate(COLORS):
        sign = rng.choice((-1, 1))
        median_y = target_y + sign * distances[index]
        offsets = [-32, -16, 0, 16, 32]
        rng.shuffle(offsets)
        points[name] = [
            [x + (index - 1) * 8, median_y + offset]
            for x, offset in zip(x_base, offsets, strict=True)
        ]
    return {
        "canvas": canvas,
        "points": points,
        "target_y": target_y,
        "palette": {key: list(value) for key, value in DEFAULT_PALETTE.items()},
    }


def render(scene: dict) -> Image.Image:
    width, height = map(int, scene["canvas"])
    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), (250, 250, 248))
    draw = ImageDraw.Draw(image)
    plot = (55, 28, width - 28, height - 38)
    draw.rectangle(tuple(v * scale for v in plot), fill=(255, 255, 255), outline=(45, 50, 58), width=2 * scale)
    for x in (100, 180, 260, 340, 420):
        draw.line((x * scale, plot[1] * scale, x * scale, plot[3] * scale), fill=(226, 228, 232), width=1 * scale)
    target_y = int(scene["target_y"])
    target = tuple(int(v) for v in scene["palette"]["target"])
    dash = 14
    for x in range(plot[0] + 4, plot[2] - 3, dash * 2):
        draw.line((x * scale, target_y * scale, min(x + dash, plot[2] - 4) * scale, target_y * scale), fill=target, width=3 * scale)
    radius = 7
    for name in COLORS:
        color = tuple(int(v) for v in scene["palette"][name])
        for x, y in scene["points"][name]:
            box = ((x - radius) * scale, (y - radius) * scale, (x + radius) * scale, (y + radius) * scale)
            draw.ellipse(box, fill=color, outline=(255, 255, 255), width=1 * scale)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def _distances(scene: dict) -> dict[str, float]:
    target_y = float(scene["target_y"])
    palette = scene["palette"]
    points = scene["points"]
    result: dict[str, float] = {}
    for name in COLORS:
        if name not in palette:
            raise ValueError("missing series palette entry")
        ys = sorted(float(point[1]) for point in points[name])
        if len(ys) != 5:
            raise ValueError("each color must contain five checkpoints")
        result[name] = abs(ys[2] - target_y)
    return result


def analytic_gold(scene: dict) -> str:
    distances = _distances(scene)
    return min(COLORS, key=lambda name: distances[name])


def margin(scene: dict) -> float:
    values = sorted(_distances(scene).values())
    return float(values[1] - values[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
