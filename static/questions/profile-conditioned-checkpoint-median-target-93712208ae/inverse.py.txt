from __future__ import annotations

import numpy as np
from PIL import Image

SERIES = {
    "blue": np.array([35, 105, 210], dtype=np.int32),
    "orange": np.array([230, 115, 25], dtype=np.int32),
    "green": np.array([25, 155, 85], dtype=np.int32),
}
TARGET = np.array([185, 45, 155], dtype=np.int32)


def _mask(arr: np.ndarray, rgb: np.ndarray, tolerance: int = 52) -> np.ndarray:
    delta = arr.astype(np.int32) - rgb
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _five_centers(mask: np.ndarray) -> list[tuple[float, float]]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 180:
        raise ValueError("series ink is missing")
    order = np.argsort(xs)
    xs = xs[order]
    ys = ys[order]
    breaks = np.where(np.diff(xs) > 24)[0] + 1
    x_groups = np.split(np.arange(len(xs)), breaks)
    groups = [group for group in x_groups if len(group) >= 25]
    if len(groups) != 5:
        raise ValueError(f"expected five separated checkpoint components, found {len(groups)}")
    return [(float(xs[g].mean()), float(ys[g].mean())) for g in groups]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    target_mask = _mask(arr, TARGET)
    ty, tx = np.nonzero(target_mask)
    if len(ty) < 120 or np.ptp(tx) < 250:
        raise ValueError("visible target line is missing")
    target_y = float(np.median(ty))
    distances: dict[str, float] = {}
    for name, rgb in SERIES.items():
        centers = _five_centers(_mask(arr, rgb))
        median_y = float(np.median([center[1] for center in centers]))
        distances[name] = abs(median_y - target_y)
    ranked = sorted(distances, key=distances.get)
    if distances[ranked[1]] - distances[ranked[0]] <= 3.0:
        raise ValueError("pixel decision is ambiguous")
    return ranked[0]
