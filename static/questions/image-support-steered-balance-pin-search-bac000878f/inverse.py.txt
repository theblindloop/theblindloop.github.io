"""Independent image-only inverse arm for bilateral cyan-area balance."""

from __future__ import annotations

import numpy as np
from PIL import Image

LABELS = ("upper-left", "upper-right", "lower-left", "lower-right")


def _masks(image: Image.Image) -> tuple[np.ndarray, np.ndarray]:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    r, g, b = array[:, :, 0], array[:, :, 1], array[:, :, 2]
    cyan = (g > 115) & (b > 115) & (g - r > 55) & (b - r > 55)
    red = (r > 150) & (r - g > 80) & (r - b > 70)
    return cyan, red


def decision_from_image(image: Image.Image) -> str:
    cyan, red = _masks(image)
    height, width = cyan.shape
    if width < 100 or height < 100:
        return "abstain"
    scores: list[int] = []
    for panel in range(4):
        x0, x1 = (0, width // 2) if panel % 2 == 0 else (width // 2, width)
        y0, y1 = (0, height // 2) if panel < 2 else (height // 2, height)
        panel_red = red[y0:y1, x0:x1]
        ys, xs = np.nonzero(panel_red)
        if len(xs) < 40:
            return "abstain"
        pin_x = x0 + int(np.rint(np.median(xs)))
        panel_cyan = cyan[y0:y1, x0:x1]
        if int(panel_cyan.sum()) < 12000:
            return "abstain"
        global_x = np.arange(x0, x1)[None, :]
        left = int((panel_cyan & (global_x < pin_x)).sum())
        right = int((panel_cyan & (global_x > pin_x)).sum())
        scores.append(min(left, right))
    ranked = sorted(range(4), key=lambda index: scores[index], reverse=True)
    if scores[ranked[0]] - scores[ranked[1]] < 900:
        return "abstain"
    return LABELS[ranked[0]]
