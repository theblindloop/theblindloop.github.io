"""Deterministic latent scenes and rasterization for balance-pin search."""

from __future__ import annotations

import copy
import random
from PIL import Image, ImageDraw

SIZE = 640
GRID = 15
CELL = 14
PATCH = GRID * CELL
CENTERS = ((160, 160), (480, 160), (160, 480), (480, 480))
LABELS = ("upper-left", "upper-right", "lower-left", "lower-right")
PALETTES = (
    {"background": [248, 246, 239], "cyan": [20, 166, 191], "red": [221, 55, 48], "ink": [45, 49, 53]},
    {"background": [245, 248, 246], "cyan": [22, 151, 183], "red": [210, 47, 62], "ink": [42, 47, 52]},
    {"background": [250, 247, 242], "cyan": [13, 174, 178], "red": [226, 62, 45], "ink": [48, 46, 49]},
)


def _pattern(left_removed: int, right_removed: int, rng: random.Random) -> list[list[int]]:
    """Return an equal-ink connected patch with controlled side deficits."""
    cells = [[1 for _ in range(GRID)] for _ in range(GRID)]
    # Removals are shallow boundary notches. Keeping row 0 and row 14 intact
    # guarantees one thick connected component and fixed extents.
    left_candidates = [(r, c) for c in (0, 1, 2) for r in range(1, GRID - 1)]
    right_candidates = [(r, c) for c in (GRID - 1, GRID - 2, GRID - 3) for r in range(1, GRID - 1)]
    rng.shuffle(left_candidates)
    rng.shuffle(right_candidates)
    for r, c in left_candidates[:left_removed]:
        cells[r][c] = 0
    for r, c in right_candidates[:right_removed]:
        cells[r][c] = 0
    return cells


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    # Every patch removes 30 cells, so total cyan ink cannot identify the winner.
    deficits = [(15, 15), (8, 22), (5, 25), (2, 28)]
    patterns = [_pattern(left, right, rng) for left, right in deficits]
    # Deterministically balance the answer position while randomizing distractors.
    winner_panel = int(seed) % 4
    others = [1, 2, 3]
    rng.shuffle(others)
    panel_order = [None, None, None, None]
    panel_order[winner_panel] = 0
    for panel, pattern_index in zip((p for p in range(4) if p != winner_panel), others):
        panel_order[panel] = pattern_index
    return {
        "patterns": patterns,
        "panel_order": panel_order,
        "palette": copy.deepcopy(PALETTES[int(seed) % len(PALETTES)]),
        "alias_tag": 0,
    }


def _side_counts(pattern: list[list[int]]) -> tuple[int, int]:
    mid = GRID // 2
    left = sum(row[c] for row in pattern for c in range(mid))
    right = sum(row[c] for row in pattern for c in range(mid + 1, GRID))
    return left, right


def analytic_gold(scene: dict) -> str:
    patterns = scene["patterns"]
    order = scene["panel_order"]
    scores = [min(_side_counts(patterns[index])) for index in order]
    return LABELS[max(range(4), key=lambda index: scores[index])]


def margin(scene: dict) -> float:
    patterns = scene["patterns"]
    order = scene["panel_order"]
    scores = sorted((min(_side_counts(patterns[index])) for index in order), reverse=True)
    return float((scores[0] - scores[1]) * CELL * CELL)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0 * CELL * CELL


def render(scene: dict) -> Image.Image:
    palette = scene["palette"]
    # Read the alias field as declared; it intentionally has no visual effect.
    _ = int(scene.get("alias_tag", 0))
    image = Image.new("RGB", (SIZE, SIZE), tuple(palette["background"]))
    draw = ImageDraw.Draw(image)
    ink = tuple(palette["ink"])
    cyan = tuple(palette["cyan"])
    red = tuple(palette["red"])
    # Visible panel boundaries make the four candidate regions explicit.
    draw.line((320, 18, 320, 622), fill=ink, width=3)
    draw.line((18, 320, 622, 320), fill=ink, width=3)
    for panel, (cx, cy) in enumerate(CENTERS):
        pattern = scene["patterns"][scene["panel_order"][panel]]
        x0, y0 = cx - PATCH // 2, cy - PATCH // 2
        for row in range(GRID):
            for col in range(GRID):
                if pattern[row][col]:
                    draw.rectangle(
                        (x0 + col * CELL, y0 + row * CELL,
                         x0 + (col + 1) * CELL - 1, y0 + (row + 1) * CELL - 1),
                        fill=cyan,
                    )
        # The visible pin overwrites the neutral center column equally in all panels.
        draw.rectangle((cx - 3, y0 - 10, cx + 3, y0 + PATCH + 9), fill=red)
        draw.ellipse((cx - 8, y0 - 18, cx + 8, y0 - 2), fill=red)
    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["alias_tag"] = int(scene.get("alias_tag", 0)) + 1
    return [("increment_render_inert_alias_tag", twin)]
