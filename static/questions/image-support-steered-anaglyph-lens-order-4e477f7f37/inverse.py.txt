"""Independent PNG-only inverse arm for visible anaglyph lens order.

This module deliberately duplicates the public palette and uses no candidate
renderer, latent state, coordinates, filenames, or metadata.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_RED = np.array((220, 45, 55), dtype=np.int16)
_CYAN = np.array((0, 176, 205), dtype=np.int16)
_MAX_COLOR_DISTANCE = 52.0
_MIN_AREA = 420
_MAX_AREA = 900


def _components(mask: np.ndarray) -> list[tuple[int, float, float, int, int, int, int]]:
    """Return area, centroid, and bbox for 8-connected components."""
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[int, float, float, int, int, int, int]] = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]:
            continue
        q = deque([(y0, x0)])
        seen[y0, x0] = True
        pixels: list[tuple[int, int]] = []
        while q:
            y, x = q.popleft()
            pixels.append((y, x))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dy == dx == 0:
                        continue
                    yy, xx = y + dy, x + dx
                    if 0 <= yy < h and 0 <= xx < w and mask[yy, xx] and not seen[yy, xx]:
                        seen[yy, xx] = True
                        q.append((yy, xx))
        if len(pixels) >= 12:
            ys = np.fromiter((p[0] for p in pixels), dtype=np.int32)
            xs = np.fromiter((p[1] for p in pixels), dtype=np.int32)
            found.append((len(pixels), float(xs.mean()), float(ys.mean()), int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())))
    return found


def _one_lens(arr: np.ndarray, reference: np.ndarray):
    distance = np.linalg.norm(arr.astype(np.int16) - reference, axis=2)
    candidates = [c for c in _components(distance <= _MAX_COLOR_DISTANCE) if _MIN_AREA <= c[0] <= _MAX_AREA]
    if len(candidates) != 1:
        return None
    comp = candidates[0]
    area, _, _, x0, y0, x1, y1 = comp
    bw, bh = x1 - x0 + 1, y1 - y0 + 1
    if min(bw, bh) < 24 or max(bw, bh) > 34 or abs(bw - bh) > 4:
        return None
    fill = area / float(bw * bh)
    if not (0.68 <= fill <= 0.86):
        return None
    return comp


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    red = _one_lens(arr, _RED)
    cyan = _one_lens(arr, _CYAN)
    if red is None or cyan is None:
        return "abstain"
    red_x, cyan_x = red[1], cyan[1]
    separation = abs(red_x - cyan_x)
    if separation < 110.0:
        return "abstain"
    return "yes" if red_x < cyan_x else "no"
