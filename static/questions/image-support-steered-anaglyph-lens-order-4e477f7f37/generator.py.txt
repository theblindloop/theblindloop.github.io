"""Deterministic renderer for a compact anaglyph-glasses order card."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

RED = (220, 45, 55)
CYAN = (0, 176, 205)
SCALE = 4


def sample_scene(seed: int) -> dict:
    """Return paired nuisances: seeds 2k/2k+1 differ only in lens order."""
    rng = random.Random(int(seed) // 2)
    radius = rng.randint(13, 15)
    # Lens centers occupy two nonadjacent 8x8 probe cells. Their joint span is
    # still local, while the intervening background makes the support multipart.
    cx = rng.choice((160, 224, 288, 352))
    cy = rng.choice((96, 160, 224, 288, 352, 416))
    gap = 128 - 2 * radius
    frame_tones = [(42, 48, 58), (55, 48, 42), (40, 54, 51)]
    backgrounds = [(247, 244, 237), (241, 246, 248), (248, 242, 247)]
    screw_dx = 64 + radius + 8
    screws = [[cx - screw_dx, cy], [cx + screw_dx, cy]]
    return {
        "width": 512,
        "height": 512,
        "center": [cx, cy],
        "lens_radius": radius,
        "lens_gap": gap,
        "order": "red_cyan" if int(seed) % 2 == 0 else "cyan_red",
        "frame_tone": list(rng.choice(frame_tones)),
        "background_tone": list(rng.choice(backgrounds)),
        "screws": screws,
    }


def _ellipse(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], **kwargs) -> None:
    draw.ellipse(tuple(v * SCALE for v in box), **kwargs)


def _line(draw: ImageDraw.ImageDraw, points: list[tuple[int, int]], **kwargs) -> None:
    draw.line([(x * SCALE, y * SCALE) for x, y in points], **kwargs)


def render(scene: dict) -> Image.Image:
    w, h = int(scene["width"]), int(scene["height"])
    bg = tuple(scene["background_tone"])
    frame = tuple(scene["frame_tone"])
    image = Image.new("RGB", (w * SCALE, h * SCALE), bg)
    draw = ImageDraw.Draw(image)
    cx, cy = map(int, scene["center"])
    r = int(scene["lens_radius"])
    gap = int(scene["lens_gap"])
    offset = r + gap // 2
    left = (cx - offset, cy)
    right = (cx + offset, cy)
    colors = (RED, CYAN) if scene["order"] == "red_cyan" else (CYAN, RED)

    # Neutral arms, bridge, and rims establish the glasses icon but never encode order.
    _line(draw, [(left[0] - r - 20, cy - 3), (left[0] - r + 1, cy)], fill=frame, width=5 * SCALE)
    _line(draw, [(right[0] + r - 1, cy), (right[0] + r + 20, cy - 3)], fill=frame, width=5 * SCALE)
    _line(draw, [(left[0] + r - 1, cy), (right[0] - r + 1, cy)], fill=frame, width=5 * SCALE)

    for (lx, ly), color in zip((left, right), colors, strict=True):
        _ellipse(draw, (lx - r - 4, ly - r - 4, lx + r + 4, ly + r + 4), fill=frame)
        _ellipse(draw, (lx - r, ly - r, lx + r, ly + r), fill=color)

    # Two identical neutral screws: their stored ordering is a pixel-identical alias.
    for sx, sy in scene["screws"]:
        _ellipse(draw, (int(sx) - 2, int(sy) - 2, int(sx) + 2, int(sy) + 2), fill=(150, 153, 154))

    return image.resize((w, h), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "yes" if scene["order"] == "red_cyan" else "no"


def margin(scene: dict) -> float:
    # Center-to-center lens separation minus two radii: visible background gap.
    return float(int(scene["lens_gap"]))


def is_quarantined(scene: dict) -> bool:
    return int(scene["lens_radius"]) < 12 or margin(scene) < 90.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    transformed = copy.deepcopy(scene)
    transformed["screws"] = list(reversed(transformed["screws"]))
    return [("reverse_screw_order", transformed)]
