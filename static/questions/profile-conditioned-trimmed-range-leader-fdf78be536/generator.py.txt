"""Deterministic renderer and analytic arm for one-point-trimmed marker ranges."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

WIDTH = 600
HEIGHT = 480
PANEL_RECTS = (
    (25, 40, 275, 220),
    (325, 40, 575, 220),
    (25, 270, 275, 450),
    (325, 270, 575, 450),
)
BACKGROUND = (255, 255, 255)
PANEL_FILL = (248, 249, 250)
GRID = (218, 223, 228)
INK = (32, 38, 45)
MARKER_PALETTE = (
    (35, 103, 180),
    (195, 62, 105),
    (24, 145, 125),
    (155, 92, 200),
)
DOT_RADIUS = 5
QUARANTINE_GAP = 10.0
LABELS = "ABCD"
_X_LANES = (28, 60, 92, 125, 158, 190, 222)

_GLYPHS = {
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "B": ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    "C": ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    "D": ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
}


def _panel_points(rng: random.Random, trimmed_span: int, full_span: int) -> list[list[int]]:
    """Construct seven separated points with independently controlled full/trimmed spans."""
    if trimmed_span >= full_span - 15:
        raise ValueError("full span must leave visible outlier gaps")
    outer_center = 90 + rng.randint(-3, 3)
    slack = (full_span - trimmed_span) // 2
    shift_limit = max(0, slack - 8)
    inner_center = outer_center + rng.randint(-shift_limit, shift_limit)
    ys = [
        outer_center - full_span // 2,
        inner_center - trimmed_span // 2,
        inner_center - trimmed_span // 4 + rng.randint(-2, 2),
        inner_center + rng.randint(-2, 2),
        inner_center + trimmed_span // 4 + rng.randint(-2, 2),
        inner_center + trimmed_span // 2,
        outer_center + full_span // 2,
    ]
    ys.sort()
    if min(b - a for a, b in zip(ys, ys[1:])) < 6:
        raise RuntimeError("sampled marker heights are not visibly distinct")
    rng.shuffle(ys)
    return [[x, y] for x, y in zip(_X_LANES, ys, strict=True)]


def sample_scene(seed: int) -> dict:
    """Sample four panels; seed modulo four exactly counterbalances the winner."""
    rng = random.Random(int(seed))
    winner = int(seed) % 4
    runner_gap = (12, 16, 20, 24)[(int(seed) // 4) % 4]
    smaller = [96 - runner_gap, 62, 46]
    rng.shuffle(smaller)
    trimmed_spans = smaller[:]
    trimmed_spans.insert(winner, 96)

    # Full bounding-box ranges are shuffled independently of the trimmed statistic,
    # so neither the widest nor narrowest overall panel is a stable answer cue.
    full_spans = [120, 128, 136, 144]
    rng.shuffle(full_spans)
    panels = [
        {"points": _panel_points(rng, trimmed, full)}
        for trimmed, full in zip(trimmed_spans, full_spans, strict=True)
    ]
    marker_color = list(MARKER_PALETTE[rng.randrange(len(MARKER_PALETTE))])
    return {"panels": panels, "marker_color": marker_color}


def _trimmed_ranges(panels: list[dict]) -> list[float]:
    values: list[float] = []
    for panel in panels:
        ys = sorted(float(point[1]) for point in panel["points"])
        if len(ys) != 7:
            raise ValueError("every panel must contain seven markers")
        values.append(ys[-2] - ys[1])
    return values


def analytic_gold(scene: dict) -> str:
    values = _trimmed_ranges(scene["panels"])
    return LABELS[max(range(4), key=values.__getitem__)]


def margin(scene: dict) -> float:
    values = sorted(_trimmed_ranges(scene["panels"]), reverse=True)
    return values[0] - values[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Same-color draw order is latent because the seven disks are disjoint."""
    twin = copy.deepcopy(scene)
    for panel in twin["panels"]:
        panel["points"].reverse()
    return [("reverse_same_color_marker_order", twin)]


def recolor_markers(scene: dict, color: tuple[int, int, int]) -> dict:
    """Label-preserving palette nuisance transform used by tests."""
    if color not in MARKER_PALETTE:
        raise ValueError("unsupported test palette")
    twin = copy.deepcopy(scene)
    twin["marker_color"] = list(color)
    return twin


def translate_markers(scene: dict, dx: int, dy: int) -> dict:
    """Translate every panel's complete marker set; vertical ranges are invariant."""
    twin = copy.deepcopy(scene)
    for panel in twin["panels"]:
        panel["points"] = [[x + dx, y + dy] for x, y in panel["points"]]
    return twin


def mirror_abscissae(scene: dict) -> dict:
    """Permute distractor x positions without changing any measured height."""
    twin = copy.deepcopy(scene)
    for panel in twin["panels"]:
        panel["points"] = [[250 - x, y] for x, y in panel["points"]]
    return twin


def _draw_glyph(draw: ImageDraw.ImageDraw, label: str, x: int, y: int, scale: int = 3) -> None:
    for row, bits in enumerate(_GLYPHS[label]):
        for col, bit in enumerate(bits):
            if bit == "1":
                draw.rectangle(
                    (
                        x + col * scale,
                        y + row * scale,
                        x + (col + 1) * scale - 1,
                        y + (row + 1) * scale - 1,
                    ),
                    fill=INK,
                )


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    panels = scene["panels"]
    marker_color = tuple(int(channel) for channel in scene["marker_color"])
    if len(panels) != 4 or len(marker_color) != 3:
        raise ValueError("scene must contain four panels and one RGB marker color")
    for index, (panel, rect) in enumerate(zip(panels, PANEL_RECTS, strict=True)):
        x0, y0, x1, y1 = rect
        draw.rectangle(rect, fill=PANEL_FILL, outline=INK, width=3)
        for gy in (45, 90, 135):
            draw.line((x0 + 3, y0 + gy, x1 - 3, y0 + gy), fill=GRID, width=1)
        _draw_glyph(draw, LABELS[index], x0 + 5, y0 - 27)
        points = panel["points"]
        if len(points) != 7:
            raise ValueError("every panel must contain seven markers")
        for px, py in points:
            cx, cy = x0 + int(px), y0 + int(py)
            draw.ellipse(
                (cx - DOT_RADIUS, cy - DOT_RADIUS, cx + DOT_RADIUS, cy + DOT_RADIUS),
                fill=marker_color,
            )
    return image
