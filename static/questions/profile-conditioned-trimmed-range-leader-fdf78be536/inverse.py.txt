"""Independent inverse arm for one-point-trimmed ranges, using final PNG pixels only."""

from __future__ import annotations

import numpy as np
from PIL import Image

_LABELS = "ABCD"


def _runs(values: np.ndarray) -> list[int]:
    """Reduce consecutive frame coordinates to integer run centers."""
    if len(values) == 0:
        return []
    groups: list[list[int]] = [[int(values[0])]]
    for value in values[1:]:
        value = int(value)
        if value == groups[-1][-1] + 1:
            groups[-1].append(value)
        else:
            groups.append([value])
    return [int(round(sum(group) / len(group))) for group in groups]


def _panel_rectangles(arr: np.ndarray) -> list[tuple[int, int, int, int]]:
    dark = np.all(arr < 90, axis=2)
    xs = _runs(np.flatnonzero(dark.sum(axis=0) > arr.shape[0] * 0.20))
    ys = _runs(np.flatnonzero(dark.sum(axis=1) > arr.shape[1] * 0.20))
    if len(xs) != 4 or len(ys) != 4:
        raise ValueError("abstain: four panel frames could not be recovered")
    if min(xs[1] - xs[0], xs[3] - xs[2], ys[1] - ys[0], ys[3] - ys[2]) < 40:
        raise ValueError("abstain: recovered panel frames are degenerate")
    return [
        (xs[0], ys[0], xs[1], ys[1]),
        (xs[2], ys[0], xs[3], ys[1]),
        (xs[0], ys[2], xs[1], ys[3]),
        (xs[2], ys[2], xs[3], ys[3]),
    ]


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Return substantial 8-connected components as arrays of (y, x) pixels."""
    work = mask.copy()
    components: list[np.ndarray] = []
    height, width = work.shape
    while bool(work.any()):
        y0, x0 = np.argwhere(work)[0]
        stack = [(int(y0), int(x0))]
        work[y0, x0] = False
        pixels: list[tuple[int, int]] = []
        while stack:
            y, x = stack.pop()
            pixels.append((y, x))
            for ny in range(max(0, y - 1), min(height, y + 2)):
                for nx in range(max(0, x - 1), min(width, x + 2)):
                    if work[ny, nx]:
                        work[ny, nx] = False
                        stack.append((ny, nx))
        if len(pixels) >= 30:
            components.append(np.asarray(pixels, dtype=np.float64))
    return components


def _trimmed_range(crop: np.ndarray) -> float:
    # Scientific marks are saturated; frames, labels, grid, and panel fill are gray.
    spread = crop.max(axis=2).astype(np.int32) - crop.min(axis=2).astype(np.int32)
    components = _components(spread > 55)
    if len(components) != 7:
        raise ValueError(f"abstain: expected seven colored markers, recovered {len(components)}")
    sizes = [len(component) for component in components]
    if min(sizes) < 70 or max(sizes) - min(sizes) > 8:
        raise ValueError("abstain: colored marker areas are inconsistent")
    ys = sorted(float(component[:, 0].mean()) for component in components)
    if min(b - a for a, b in zip(ys, ys[1:])) < 4.0:
        raise ValueError("abstain: marker height order is raster-ambiguous")
    return ys[-2] - ys[1]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    if arr.ndim != 3 or arr.shape[0] < 100 or arr.shape[1] < 100:
        raise ValueError("abstain: raster is too small")
    values: list[float] = []
    for x0, y0, x1, y1 in _panel_rectangles(arr):
        crop = arr[y0 + 3 : y1 - 2, x0 + 3 : x1 - 2]
        values.append(_trimmed_range(crop))
    order = sorted(range(4), key=values.__getitem__, reverse=True)
    if values[order[0]] - values[order[1]] < 9.0:
        raise ValueError("abstain: top two trimmed ranges are raster-ambiguous")
    return _LABELS[order[0]]
