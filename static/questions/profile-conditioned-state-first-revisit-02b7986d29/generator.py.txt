"""Earliest state repeat (first revisit) in an ordered token walk.

Visual world: five stacked panels P1, P2, P3, P4 and P5 read top to bottom in
order. Every panel shows the same horizontal track divided into NSLOT=5 equal
slots (slot 1 leftmost to slot 5 rightmost) by thin vertical dividers, and one
red token sits in exactly one slot of each panel. The red token is the same
token in every panel: its colour is fixed, so identity is carried by the single
token, and between any two consecutive panels it always moves to a DIFFERENT
slot.

A REVISIT occurs at panel k (k = 2..4, zero-based) when the token's slot in
that panel equals a slot it occupied in any EARLIER (higher) panel. The
decision is the earliest panel at which the first such repeated state occurs,
or the label 'never' if the whole ordered walk revisits no prior slot.

The latent scene stores only ``positions``: the list (length KPANEL) of slot
indices (0..NSLOT-1) the red token occupies in P1..P5. The gold is recovered by
replaying that walk and reporting the earliest index at which the running
position joins the set of previously visited positions.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

NSLOT = 5
KPANEL = 5

LEFT_M = 60
TOP_M = 70
SLOT_W = 72
SLOT_H = 60
GAP = 14
PANEL_GAP = 40
TOKEN_R = 17

BG = (255, 255, 255)
BORDER = (95, 95, 95)
DIVIDER = (205, 205, 205)
TOKEN = (205, 45, 45)

WIDTH = LEFT_M + NSLOT * SLOT_W + (NSLOT - 1) * GAP + LEFT_M
HEIGHT = TOP_M + KPANEL * SLOT_H + (KPANEL - 1) * PANEL_GAP + 44


def col_center(i: int) -> int:
    return LEFT_M + SLOT_W // 2 + i * (SLOT_W + GAP)


def col_x0(i: int) -> int:
    return col_center(i) - SLOT_W // 2


def panel_top(k: int) -> int:
    return TOP_M + k * (SLOT_H + PANEL_GAP)


def _first_revisit_panel(positions: list[int]) -> str | None:
    """Return the zero-based panel index of the first revisit, else None."""
    seen = {positions[0]}
    for k in range(1, len(positions)):
        if positions[k] in seen:
            return k
        seen.add(positions[k])
    return None


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    positions = []
    cur = rng.randrange(NSLOT)
    positions.append(cur)
    for _ in range(1, KPANEL):
        options = [s for s in range(NSLOT) if s != cur]
        cur = rng.choice(options)
        positions.append(cur)
    return {"positions": positions}


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)
    positions = scene["positions"]

    # slot-number caption row below the stack
    for i in range(NSLOT):
        label = str(i + 1)
        x = col_center(i)
        draw.text((x - 4, HEIGHT - 38), label, fill=(60, 60, 60))

    for k in range(KPANEL):
        top = panel_top(k)
        left = col_x0(0)
        right = col_x0(NSLOT - 1) + SLOT_W
        draw.rectangle([left, top, right, top + SLOT_H], outline=BORDER, width=3)
        for i in range(1, NSLOT):
            vx = col_x0(i)
            draw.line([vx, top, vx, top + SLOT_H], fill=DIVIDER, width=2)
        slot = positions[k]
        cx = col_center(slot)
        cy = top + SLOT_H // 2
        draw.ellipse(
            [cx - TOKEN_R, cy - TOKEN_R, cx + TOKEN_R, cy + TOKEN_R],
            fill=TOKEN,
        )
    return img


def analytic_gold(scene: dict) -> str:
    positions = scene["positions"]
    panel = _first_revisit_panel(positions)
    if panel is None:
        return "never"
    return f"panel {panel + 1}"


def margin(scene: dict) -> float:
    """Discrete decision margin: distance (in panel steps) between the first
    revisit and the next revisit, or the space to the end of the walk when
    there is no later revisit. Always strictly positive for an on-track scene,
    so no scene is quarantined."""
    positions = scene["positions"]
    events = [k for k in range(1, len(positions)) if positions[k] in positions[:k]]
    if not events:
        return float(KPANEL)
    earliest = events[0]
    if len(events) >= 2:
        return float(events[1] - events[0])
    return float(KPANEL - 1 - earliest + 1)


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    """No nontrivial render-invariant latent symmetry exists.

    The only latent field, ``positions``, directly fixes the drawn slot of the
    red token in every panel, and analytic gold reads exactly that field. Any
    perturbation of a slot index changes the raster; permuting panels would
    change the image; the single fixed token colour leaves no identity
    re-labelling. There is therefore no reparametrization yielding
    pixel-identical renders with a different gold, so no alias is declared.
    """
    return []
