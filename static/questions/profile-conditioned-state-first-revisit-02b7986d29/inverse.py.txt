"""Independent pixel-only inverse arm for the earliest-state-repeat world.

Receives only the final PNG. From the raster alone it:
  1. labels red pixels into connected components, one per panel token;
  2. orders the token centroids top to bottom to recover the panel order
     P1..P5 (an ordered walk);
  3. maps each token's horizontal position to a slot index (1..5);
  4. replays the walk and reports the earliest panel at which the token first
     revisits a slot it occupied earlier, or 'never'.

No latent scene, renderer, prompt, or analytic-gold code is accessed.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

NSLOT = 5
KPANEL = 5
LEFT_M = 60
SLOT_W = 72
GAP = 14
TOKEN = (205, 45, 45)
PITCH = SLOT_W + GAP
SLOT_MID_OFFSET = SLOT_W // 2


class UnreadableImageError(ValueError):
    """The raster does not expose the expected token-walk structure."""


def _mask(arr: np.ndarray, color: tuple[int, int, int]) -> np.ndarray:
    r, g, b = color
    return (arr[:, :, 0] == r) & (arr[:, :, 1] == g) & (arr[:, :, 2] == b)


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    visited = np.zeros_like(mask, dtype=bool)
    ys, xs = np.nonzero(mask)
    comps = []
    for y, x in zip(ys.tolist(), xs.tolist()):
        if visited[y, x]:
            continue
        points = []
        queue = deque([(y, x)])
        visited[y, x] = True
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            y0 = max(0, cy - 1)
            y1 = min(mask.shape[0], cy + 2)
            x0 = max(0, cx - 1)
            x1 = min(mask.shape[1], cx + 2)
            for ny in range(y0, y1):
                for nx in range(x0, x1):
                    if mask[ny, nx] and not visited[ny, nx]:
                        visited[ny, nx] = True
                        queue.append((ny, nx))
        comps.append(points)
    return comps


def _slot_from_x(cx: float) -> int:
    slot = int(round((cx - (LEFT_M + SLOT_MID_OFFSET)) / PITCH))
    slot = max(0, min(NSLOT - 1, slot))
    return slot


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    red = _mask(arr, TOKEN)

    # empty red tokens (e.g. corrupted raster) must be rejected
    if int(red.sum()) < 50:
        raise UnreadableImageError("no red token pixels found")

    comps = _components(red)
    # keep only substantial disk components
    comps = [c for c in comps if len(c) >= 50]
    if len(comps) != KPANEL:
        raise UnreadableImageError(f"expected {KPANEL} tokens; found {len(comps)}")

    centroids = []
    for comp in comps:
        ys = [p[0] for p in comp]
        xs = [p[1] for p in comp]
        centroids.append((np.mean(xs), np.mean(ys)))
    centroids.sort(key=lambda c: c[1])

    positions = [_slot_from_x(c) for c, _ in centroids]

    seen = {positions[0]}
    for k in range(1, len(positions)):
        if positions[k] in seen:
            return f"panel {k + 1}"
        seen.add(positions[k])
    return "never"
