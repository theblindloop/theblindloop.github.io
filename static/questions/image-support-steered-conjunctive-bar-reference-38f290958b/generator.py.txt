"""Conjunctive-bar reference world renderer.

Latent scene: a single coloured bar (the reference) sits in the left band and a
2-by-2 grid of four coloured bars sits in the right band. Each bar has a colour
(red/blue/green) and a tilt (vertical / horizontal / right-slant / left-slant).
Exactly one candidate bar matches the reference bar in BOTH colour AND tilt; the
answer is that bar's grid position (top-left / top-right / bottom-left /
bottom-right).

The decisive support is deliberately *regional* and *multipart*: it consists of
the reference bar (one disconnected component, far left) and the matching
candidate bar (a second disconnected component, far right), which jointly must
be bound together to answer. Neither a tiny crop of the reference nor a tiny
crop of only the winner is sufficient, and the two decisive components never
touch; they span a substantial middle region without crossing the image edges.

```text
I = R(z)
winner = the candidate bar whose (colour, tilt) == reference (colour, tilt)
answer = grid position of the winner
```
"""

from __future__ import annotations

import dataclasses
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "conjunctive_bar_reference-0.1.0"

CANVAS = (512, 512)
BG = (255, 255, 255)

COLORS = {
    "red": (214, 48, 49),
    "blue": (41, 82, 214),
    "green": (38, 166, 91),
}
COLOR_NAMES = ("red", "blue", "green")

# Tilt orientations in degrees measured from the +x axis (mod 180).
ORIENTATIONS = ("horizontal", "slant_r", "vertical", "slant_l")
ORIENT_DEG = {
    "horizontal": 0.0,
    "slant_r": 45.0,
    "vertical": 90.0,
    "slant_l": 135.0,
}

BAR_LENGTH = 46.0
BAR_WIDTH = 18.0
_DIAG = BAR_LENGTH / (2.0 * 2.0**0.5)  # half-length projected on each axis

# Reference bar location (left band).
REF_POS = (78.0, 266.0)

# Candidate 2x2 grid centres (right band). Kept inside grid cells 4..6 so the
# support stays within a medium region without touching the image edges.
GRID_X = (363.0, 429.0)
GRID_Y = (184.0, 340.0)
LABELS = (("tl", "tr"), ("bl", "br"))


@dataclasses.dataclass(frozen=True)
class BarScene:
    reference: dict
    candidates: list
    width: int = 512
    height: int = 512


def _angular_distance(a, b):
    d = abs(ORIENT_DEG[a] - ORIENT_DEG[b]) % 180.0
    return min(d, 180.0 - d)


def _pick_distractor_orientation(rng, o_ref):
    """A tilt different from the reference tilt (all differ by 45 degrees)."""
    return rng.choice([o for o in ORIENTATIONS if o != o_ref])


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    colors = list(COLOR_NAMES)
    rng.shuffle(colors)
    c_ref = colors[0]
    other_colors = colors[1:]

    o_ref = rng.choice(ORIENTATIONS)

    winner = {"pos": None, "color": c_ref, "orientation": o_ref, "label": None}

    # A distractor with the SAME colour (so colour alone cannot isolate).
    d1 = {"color": c_ref, "orientation": _pick_distractor_orientation(rng, o_ref)}
    # A distractor with the SAME tilt (so tilt alone cannot isolate).
    d2 = {"color": rng.choice(other_colors), "orientation": o_ref}
    # A distractor with a different colour AND a different tilt.
    d3_color = rng.choice(
        [c for c in other_colors if c != d2["color"]] or other_colors
    )
    d3 = {
        "color": d3_color,
        "orientation": _pick_distractor_orientation(rng, o_ref),
    }

    slots = [("tl", GRID_X[0], GRID_Y[0]),
             ("tr", GRID_X[1], GRID_Y[0]),
             ("bl", GRID_X[0], GRID_Y[1]),
             ("br", GRID_X[1], GRID_Y[1])]
    rng.shuffle(slots)

    def jitter(pos):
        x, y = pos
        return (x + rng.uniform(-4.0, 4.0), y + rng.uniform(-4.0, 4.0))

    winner_label = slots[0]
    candidate_defs = [
        (winner, winner_label),
        (d1, slots[1]),
        (d2, slots[2]),
        (d3, slots[3]),
    ]
    candidates = []
    for definition, (label, sx, sy) in candidate_defs:
        px, py = jitter((sx, sy))
        candidates.append(
            {
                "pos": (px, py),
                "color": definition["color"],
                "orientation": definition["orientation"],
                "label": label,
            }
        )

    return {
        "width": 512,
        "height": 512,
        "reference": {
            "pos": REF_POS,
            "color": c_ref,
            "orientation": o_ref,
        },
        "candidates": candidates,
    }


def _draw_bar(draw, center, orientation, color):
    """Draw a thick bar (an elongated rectangle) at the given tilt."""
    x, y = center
    (R, G, B) = color
    fill = (R, G, B)
    if orientation == "vertical":
        a = (x, y - BAR_LENGTH / 2.0)
        b = (x, y + BAR_LENGTH / 2.0)
    elif orientation == "horizontal":
        a = (x - BAR_LENGTH / 2.0, y)
        b = (x + BAR_LENGTH / 2.0, y)
    elif orientation == "slant_r":
        a = (x - _DIAG, y - _DIAG)
        b = (x + _DIAG, y + _DIAG)
    else:  # slant_l
        a = (x - _DIAG, y + _DIAG)
        b = (x + _DIAG, y - _DIAG)
    draw.line([a, b], fill=fill, width=int(round(BAR_WIDTH)))


def render(scene) -> Image.Image:
    w = int(scene["width"])
    h = int(scene["height"])
    ss = 2
    img = Image.new("RGB", (w * ss, h * ss), BG)
    draw = ImageDraw.Draw(img)

    def S(p):
        return (p[0] * ss, p[1] * ss)

    ref = scene["reference"]
    _draw_bar(draw, S(ref["pos"]), ref["orientation"], COLORS[ref["color"]])

    for candidate in scene["candidates"]:
        _draw_bar(
            draw,
            S(candidate["pos"]),
            candidate["orientation"],
            COLORS[candidate["color"]],
        )

    img = img.resize((w, h), Image.LANCZOS)
    return img


def _winner_label(scene) -> str:
    ref = scene["reference"]
    for candidate in scene["candidates"]:
        if (
            candidate["color"] == ref["color"]
            and candidate["orientation"] == ref["orientation"]
        ):
            return candidate["label"]
    raise ValueError("no candidate matches the reference conjunctively")


def analytic_gold(scene) -> str:
    return _winner_label(scene)


def margin(scene) -> float:
    """Runner-up margin: distance from the best non-winner to a full match.

    A candidate's distance to a full match combines a colour gap (0 if it shares
    the reference colour else 1.0) with a normalised tilt gap. The closest
    non-winner is the same-colour, different-tilt distractor, giving a stable
    margin of ~0.25; the decision stays comfortably above the quarantine floor.
    """
    ref = scene["reference"]
    distances = []
    for candidate in scene["candidates"]:
        if (
            candidate["color"] == ref["color"]
            and candidate["orientation"] == ref["orientation"]
        ):
            continue
        color_gap = 0.0 if candidate["color"] == ref["color"] else 1.0
        tilt_gap = _angular_distance(ref["orientation"], candidate["orientation"]) / 180.0
        distances.append(color_gap + tilt_gap)
    if not distances:
        return 0.0
    return float(min(distances))


def is_quarantined(scene) -> bool:
    return margin(scene) < 0.15


def latent_symmetries(scene):
    """not_applicable: minimal, injective, fully-rendered latent parametrization.

    Every label-input field (reference, candidates) is directly and visibly
    realised in the raster as coloured/tilted bars; there is no latent answer
    field hidden from rendering, no pixel-identical alternative scene with a
    different decision, and therefore no declared transform to exercise.
    """
    return iter(())
