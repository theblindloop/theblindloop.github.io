"""separable_backplane_reveal: separable row/column backplane-removal reveal.

Declared transform: remove the separable brightness backplane that runs across
the field by subtracting, per row, that row's mean brightness and, per column,
that column's mean brightness (a non-linear, rank-one separable flatten). What
remains is the non-separable residual that still carries a coarse diagonal-band
glyph. In the raw raster the glyph is masked by a strong row/column backplane
(independent, high-amplitude, non-linear 1-D brightness profiles on both axes)
plus high-amplitude per-cell noise, so no local patch and no single row or
column shows it; only removing the separable backplane exposes the band.

The glyph is a diagonal band: A = rising '/' (anti-diagonal, r+c ~ const),
B = falling '\\' (main diagonal, r ~ c).

Latent scene z = {noise, glyph, seedb, maskamp}. label = glyph.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RENDERER_VERSION = "separable_backplane_reveal-0.1.0"

W = 512
H = 512
NGRID = 32          # cells per side
CELL = W // NGRID   # 16
BASE = 128.0
BIAS = 46.0
MASKAMP = 130.0     # half-amplitude of each 1-D row/column backplane profile
NOISE_SIGMA = 36.0
BAND_WIDTH = 5      # half-width of the diagonal band in cell units
Q_TOL = 0.15


def band_masks() -> dict[str, np.ndarray]:
    """(NGRID, NGRID) bool masks: A rising '/', B falling '\\'."""
    r, c = np.indices((NGRID, NGRID))
    mask_a = np.abs((r + c) - (NGRID - 1)) <= BAND_WIDTH
    mask_b = np.abs(r - c) <= BAND_WIDTH
    return {"A": mask_a, "B": mask_b}


def build_scene(noise: int, glyph: str, seedb: int, maskamp: float = MASKAMP) -> dict:
    return {
        "noise": int(noise),
        "glyph": glyph,
        "seedb": int(seedb),
        "maskamp": float(maskamp),
    }


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(int(seed) & 0xFFFFFFFF)
    glyph = "A" if rng.integers(0, 2) else "B"
    return build_scene(noise=int(seed), glyph=glyph, seedb=int(seed) + 77)


def _glyph_mask(glyph: str) -> np.ndarray:
    return band_masks()[glyph]


def _row_col_profiles(scene: dict) -> tuple[np.ndarray, np.ndarray]:
    """(n,) row and column 1-D brightness profiles, mutually independent and
    intentionally non-linear so a planar (linear in row and column) fit could
    never remove them; only removing each row's and each column's own mean."""
    rng = np.random.default_rng(int(scene["seedb"]) & 0xFFFFFFFF)
    n = NGRID
    amp = float(scene["maskamp"])
    rowp = rng.uniform(-amp, amp, size=(n,))
    colp = rng.uniform(-amp, amp, size=(n,))
    rowf = np.convolve(rowp, np.ones(3) / 3.0, mode="same")
    colf = np.convolve(colp, np.ones(3) / 3.0, mode="same")
    return rowf, colf


def render_matrix(scene: dict) -> np.ndarray:
    """(NGRID, NGRID) mean-luminance of each lattice cell before rasterizing."""
    n = NGRID
    inside = _glyph_mask(scene["glyph"])
    sign = np.where(inside, 1.0, -1.0)
    bias = BIAS * sign
    rowf, colf = _row_col_profiles(scene)
    rng = np.random.default_rng(int(scene["noise"]) & 0xFFFFFFFF)
    noise = rng.normal(0.0, NOISE_SIGMA, size=(n, n))
    return BASE + rowf[:, None] + colf[None, :] + bias + noise


def render(scene: dict) -> Image.Image:
    cell = CELL
    mat = render_matrix(scene)
    arr = np.repeat(np.repeat(mat, cell, axis=0), cell, axis=1)
    arr = np.clip(arr, 0, 255)
    canvas = np.repeat(arr[:, :, None], 3, axis=2).astype(np.uint8)
    canvas = canvas[:W, :H]
    return Image.fromarray(canvas, "RGB")


def margin(scene: dict) -> float:
    chosen = _glyph_mask(scene["glyph"])
    other = _glyph_mask("B" if scene["glyph"] == "A" else "A")
    o_filled = int(other.sum())
    corr_other = float((chosen & other).sum()) / max(1, o_filled)
    return round(1.0 - corr_other, 6)


def analytic_gold(scene: dict) -> str:
    return scene["glyph"]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < Q_TOL


def latent_symmetries(scene: dict):
    """not_applicable: every latent field (the cell-noise seed, the answer
    glyph, the backplane seed, the backplane amplitude) directly changes the
    rendered pixels, so no distinct latent vector reproduces the same image.
    Declared empty."""
    return iter(())
