"""Independent pixel-only inverse arm for separable_backplane_reveal.

Receives ONLY the final PNG. It:
  1. reads the mean luminance of every lattice cell;
  2. removes the separable backplane by subtracting, per row, that row's mean
     and, per column, that column's mean (a rank-one separable flatten);
  3. correlates the residual against the two declared diagonal-band templates
     A (rising) and B (falling);
  4. reports the template with the larger signed inside-vs-outside contrast.

No scene data, no analytic gold, no renderer imports. The residual is
polarity-invariant (relative inside/outside contrast), so a global brightness
inversion cannot flip the outcome.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "separable_backplane_reveal-oracle-0.1.0"

W = 512
NGRID = 32
CELL = W // NGRID
BAND_WIDTH = 5


def _band_masks() -> dict[str, np.ndarray]:
    r, c = np.indices((NGRID, NGRID))
    mask_a = np.abs((r + c) - (NGRID - 1)) <= BAND_WIDTH
    mask_b = np.abs(r - c) <= BAND_WIDTH
    return {"A": mask_a, "B": mask_b}


def _cell_means(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("L"), dtype=np.float64)
    g = np.zeros((NGRID, NGRID), dtype=np.float64)
    for i in range(NGRID):
        for j in range(NGRID):
            cell = arr[i * CELL:(i + 1) * CELL, j * CELL:(j + 1) * CELL]
            g[i, j] = cell.mean()
    return g


def _remove_separable(g: np.ndarray) -> np.ndarray:
    """Subtract, per row, the row mean and, per column, the column mean. This
    exactly removes any separable (rank-one) backplane of the form R(row)+C(col)
    while leaving the non-separable diagonal-band residual intact."""
    g = g - np.mean(g, axis=1, keepdims=True)
    return g - np.mean(g, axis=0, keepdims=True)


def _contrast(res: np.ndarray, mask: np.ndarray) -> float:
    inside = res[mask].mean()
    outside = res[~mask].mean()
    return float(inside - outside)


def decision_from_image(image: Image.Image) -> str:
    g = _cell_means(image)
    res = _remove_separable(g)
    masks = _band_masks()
    sa = _contrast(res, masks["A"])
    sb = _contrast(res, masks["B"])
    return "A" if abs(sa) >= abs(sb) else "B"


def reveal_margin(image: Image.Image) -> float:
    g = _cell_means(image)
    res = _remove_separable(g)
    masks = _band_masks()
    sa = _contrast(res, masks["A"])
    sb = _contrast(res, masks["B"])
    return abs(abs(sa) - abs(sb))
