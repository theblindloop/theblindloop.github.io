"""Pixel-only decision arm.

Receives a rendered PNG and nothing else.  It segments the picture by generic
colour predicates, treats the cable together with the dots sitting on it as the
walkable set, measures walk distance from the ringed end with a breadth-first
front over that set, locates the first wall crossing along the walk, and reports
the colour of the first dot whose walk distance exceeds it.
"""

from __future__ import annotations

from collections import deque

from PIL import Image

DOT_NAMES = ("red", "green", "blue")
ABSTAIN = "abstain"


def _masks(image):
    width, height = image.size
    pixels = image.load()
    cable = set()
    wall = set()
    ring = []
    dots = {name: [] for name in DOT_NAMES}
    for y in range(height):
        for x in range(width):
            r, g, b = pixels[x, y][:3]
            hi = max(r, g, b)
            lo = min(r, g, b)
            if hi < 70:
                wall.add((x, y))
            elif hi - lo < 45 and 60 < hi < 210:
                cable.add((x, y))
            elif r > 190 and 90 < g < 195 and b < 95:
                ring.append((x, y))
            elif r > 150 and g < 115 and b < 115:
                dots["red"].append((x, y))
            elif g > 115 and r < 135 and b < 135:
                dots["green"].append((x, y))
            elif b > 150 and r < 135 and g < 135:
                dots["blue"].append((x, y))
    return cable, wall, ring, dots


def _walk_distance(walkable, start):
    dist = {start: 0}
    queue = deque([start])
    while queue:
        x, y = queue.popleft()
        base = dist[(x, y)]
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                nxt = (x + dx, y + dy)
                if nxt in walkable and nxt not in dist:
                    dist[nxt] = base + 1
                    queue.append(nxt)
    return dist


def decision_from_image(image: Image.Image) -> str:
    image = image.convert("RGB")
    cable, wall, ring, dots = _masks(image)
    if not cable or not wall or not ring:
        return ABSTAIN
    if any(len(points) < 40 for points in dots.values()):
        return ABSTAIN

    walkable = set(cable)
    for points in dots.values():
        walkable.update(points)

    ring_x = sum(p[0] for p in ring) / len(ring)
    ring_y = sum(p[1] for p in ring) / len(ring)
    start = min(walkable, key=lambda p: (p[0] - ring_x) ** 2 + (p[1] - ring_y) ** 2)
    dist = _walk_distance(walkable, start)
    if len(dist) < 0.9 * len(walkable):
        return ABSTAIN

    wx0 = min(p[0] for p in wall)
    wx1 = max(p[0] for p in wall)
    wy0 = min(p[1] for p in wall)
    wy1 = max(p[1] for p in wall)
    over_wall = [
        value
        for (x, y), value in dist.items()
        if wx0 - 3 <= x <= wx1 + 3 and wy0 - 3 <= y <= wy1 + 3
    ]
    if not over_wall:
        return ABSTAIN
    first_cross = min(over_wall)
    return _pick(dots, dist, first_cross)


def _pick(dots, dist, first_cross):
    ranked = []
    for name in DOT_NAMES:
        values = [dist[p] for p in dots[name] if p in dist]
        if not values:
            return ABSTAIN
        ranked.append((min(values), name))
    ranked.sort()
    after = [item for item in ranked if item[0] > first_cross]
    if not after:
        return ABSTAIN
    return after[0][1]
