"""Cable-over-wall world.

One grey cable is drawn as a self-avoiding rectilinear route on a coarse
lattice.  A short black wall bar lies between two lattice rows; wherever the
cable runs over the bar the bar is interrupted, which is the visible crossing
event.  Three coloured dots sit on the cable.  The decision is the colour of the
first dot met, walking from the ringed end, *after* the first wall crossing.

Everything the analytic gold uses is drawn: the cable polyline, the wall bar and
the dots.  No hidden label field exists, so no latent alias is declared.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 720
HEIGHT = 520
CELL = 80
COL0 = 70
ROW0 = 60
NCOL = 8
NROW = 6

STROKE = 7
DOT_R = 9
RING_R = 18
RING_W = 4
WALL_T = 9

BG = (255, 255, 255)
CABLE_RGB = (110, 110, 110)
WALL_RGB = (25, 25, 25)
RING_RGB = (242, 140, 20)
DOT_RGB = {"red": (220, 30, 30), "green": (20, 160, 60), "blue": (30, 80, 220)}
COLOR_ORDER = ("red", "green", "blue")

QUARANTINE_MARGIN = 70.0


# --------------------------------------------------------------------------
# geometry helpers (scene-free)
# --------------------------------------------------------------------------
def _cumulative(points):
    out = [0.0]
    for (x0, y0), (x1, y1) in zip(points, points[1:]):
        out.append(out[-1] + abs(x1 - x0) + abs(y1 - y0))
    return out


def _crossing_positions(points, wall):
    """Arc-length positions where the polyline runs over the wall bar."""
    cum = _cumulative(points)
    y_w = wall["y"]
    x0, x1 = wall["x0"], wall["x1"]
    hits = []
    for index, ((ax, ay), (bx, by)) in enumerate(zip(points, points[1:])):
        if ax != bx:
            continue
        if not (x0 < ax < x1):
            continue
        if (ay - y_w) * (by - y_w) < 0:
            hits.append(cum[index] + abs(y_w - ay))
    return sorted(hits)


def _dot_positions(dots):
    return [(float(dot["s"]), str(dot["color"])) for dot in dots]


def _decide(points, wall, dots):
    """Return (colour, margin) for the first dot after the first crossing."""
    crossings = _crossing_positions(points, wall)
    marks = sorted(_dot_positions(dots))
    if not crossings or not marks:
        return None, 0.0
    first = crossings[0]
    after = [item for item in marks if item[0] > first]
    if not after:
        return None, 0.0
    winner = after[0]
    gaps = [abs(s - c) for s, _ in marks for c in crossings]
    gaps += [abs(marks[i][0] - marks[i - 1][0]) for i in range(1, len(marks))]
    return winner[1], float(min(gaps))


# --------------------------------------------------------------------------
# required interface
# --------------------------------------------------------------------------
def _lattice_path(rng, length):
    for _ in range(300):
        cell = (rng.randrange(NCOL), rng.randrange(NROW))
        path = [cell]
        used = {cell}
        while len(path) < length:
            i, j = path[-1]
            options = [
                (i + 1, j),
                (i - 1, j),
                (i, j + 1),
                (i, j - 1),
            ]
            options = [
                c
                for c in options
                if 0 <= c[0] < NCOL and 0 <= c[1] < NROW and c not in used
            ]
            if not options:
                break
            step = options[rng.randrange(len(options))]
            path.append(step)
            used.add(step)
        if len(path) >= length:
            return path
    return None


def _pixels(cells):
    return [[COL0 + CELL * i, ROW0 + CELL * j] for i, j in cells]


def _candidate_slots(points, wall, crossings, spacing_to_event):
    cum = _cumulative(points)
    total = cum[-1]
    y_w = wall["y"]
    x0, x1 = wall["x0"], wall["x1"]
    slots = []
    s = 40.0
    while s <= total - 40.0:
        near_vertex = min(abs(s - value) for value in cum)
        if near_vertex < 26.0:
            s += 4.0
            continue
        if crossings and min(abs(s - c) for c in crossings) < spacing_to_event:
            s += 4.0
            continue
        point = _point_at(points, cum, s)
        if abs(point[1] - y_w) < 26.0 and x0 - 26.0 <= point[0] <= x1 + 26.0:
            s += 4.0
            continue
        slots.append((s, point))
        s += 4.0
    return slots


def _point_at(points, cum, s):
    for index in range(len(points) - 1):
        if cum[index] <= s <= cum[index + 1]:
            ax, ay = points[index]
            bx, by = points[index + 1]
            span = cum[index + 1] - cum[index]
            t = 0.0 if span == 0 else (s - cum[index]) / span
            return [ax + (bx - ax) * t, ay + (by - ay) * t]
    return list(points[-1])


def sample_scene(seed: int) -> dict:
    rng = random.Random(0x5EED ^ (int(seed) * 2654435761 % (1 << 32)))
    for _ in range(4000):
        cells = _lattice_path(rng, rng.randint(11, 16))
        if cells is None:
            continue
        points = _pixels(cells)
        if rng.random() < 0.5:
            points = points[::-1]
        row_gap = rng.randrange(1, NROW)
        col_a = rng.randrange(0, NCOL - 2)
        col_b = rng.randrange(col_a + 1, NCOL)
        wall = {
            "y": ROW0 + CELL * row_gap - CELL // 2,
            "x0": COL0 + CELL * col_a - CELL // 2,
            "x1": COL0 + CELL * col_b + CELL // 2,
            "thickness": WALL_T,
        }
        crossings = _crossing_positions(points, wall)
        if not 1 <= len(crossings) <= 3:
            continue
        ring = {"x": points[0][0], "y": points[0][1], "r": RING_R}
        if abs(ring["y"] - wall["y"]) < 34:
            continue
        roll = rng.random()
        if roll < 0.22:
            separation = 38.0
        elif roll < 0.45:
            separation = 50.0
        else:
            separation = 72.0
        slots = _candidate_slots(points, wall, crossings, separation)
        first = crossings[0]
        before = [item for item in slots if item[0] < first - separation]
        after = [item for item in slots if item[0] > first + separation]
        want_before = 1 if rng.random() < 0.5 else 2
        if len(before) < want_before or len(after) < 3 - want_before:
            continue
        head = _pick_spread(rng, before, want_before, separation)
        tail = _pick_spread(rng, after, 3 - want_before, separation)
        if len(head) + len(tail) != 3:
            continue
        if separation < 60.0:
            # keep genuinely near-boundary scenes in the sampled population
            head[-1] = before[-1]
            tail[0] = after[0]
            head.sort()
            tail.sort()
        chosen = head + tail
        chosen.sort()
        if min(
            chosen[i][0] - chosen[i - 1][0] for i in range(1, 3)
        ) < separation:
            continue
        if _too_close_to_cable(points, [item[1] for item in chosen]):
            continue
        colors = list(COLOR_ORDER)
        rng.shuffle(colors)
        dots = [
            {"color": color, "s": round(s, 3), "x": round(pt[0], 3), "y": round(pt[1], 3)}
            for (s, pt), color in zip(chosen, colors)
        ]
        decision, _ = _decide(points, wall, dots)
        if decision is None:
            continue
        return {
            "size": [WIDTH, HEIGHT],
            "cable": [[float(x), float(y)] for x, y in points],
            "stroke": STROKE,
            "wall": wall,
            "ring": ring,
            "dots": dots,
            "seed": int(seed),
        }
    raise RuntimeError(f"could not sample a scene for seed {seed}")


def _pick_spread(rng, slots, count, separation):
    if count <= 0:
        return []
    for _ in range(240):
        picks = []
        pool = list(slots)
        rng.shuffle(pool)
        for item in pool:
            if all(abs(item[0] - other[0]) >= separation for other in picks):
                picks.append(item)
            if len(picks) == count:
                return sorted(picks)
    return []


def _too_close_to_cable(points, dot_points):
    """Reject dots that sit close to a *different* part of the cable."""
    for dx, dy in dot_points:
        near = 0
        for (ax, ay), (bx, by) in zip(points, points[1:]):
            if _segment_distance(dx, dy, ax, ay, bx, by) <= DOT_R + STROKE:
                near += 1
        if near > 1:
            return True
    return False


def _segment_distance(px, py, ax, ay, bx, by):
    vx, vy = bx - ax, by - ay
    length = vx * vx + vy * vy
    if length == 0:
        return ((px - ax) ** 2 + (py - ay) ** 2) ** 0.5
    t = max(0.0, min(1.0, ((px - ax) * vx + (py - ay) * vy) / length))
    qx, qy = ax + t * vx, ay + t * vy
    return ((px - qx) ** 2 + (py - qy) ** 2) ** 0.5


def render(scene: dict) -> Image.Image:
    width, height = scene["size"]
    image = Image.new("RGB", (int(width), int(height)), BG)
    draw = ImageDraw.Draw(image)

    wall = scene["wall"]
    half = wall["thickness"] // 2
    draw.rectangle(
        [wall["x0"], wall["y"] - half, wall["x1"], wall["y"] + half],
        fill=WALL_RGB,
    )

    ring = scene["ring"]
    draw.ellipse(
        [
            ring["x"] - ring["r"],
            ring["y"] - ring["r"],
            ring["x"] + ring["r"],
            ring["y"] + ring["r"],
        ],
        outline=RING_RGB,
        width=RING_W,
    )

    points = [(float(x), float(y)) for x, y in scene["cable"]]
    stroke = int(scene["stroke"])
    for (ax, ay), (bx, by) in zip(points, points[1:]):
        draw.line([ax, ay, bx, by], fill=CABLE_RGB, width=stroke)
    for x, y in points[1:-1]:
        draw.rectangle(
            [x - stroke // 2, y - stroke // 2, x + stroke // 2, y + stroke // 2],
            fill=CABLE_RGB,
        )

    for dot in scene["dots"]:
        draw.ellipse(
            [
                dot["x"] - DOT_R,
                dot["y"] - DOT_R,
                dot["x"] + DOT_R,
                dot["y"] + DOT_R,
            ],
            fill=DOT_RGB[dot["color"]],
        )
    return image


def analytic_gold(scene: dict) -> str:
    decision, _ = _decide(scene["cable"], scene["wall"], scene["dots"])
    return str(decision)


def margin(scene: dict) -> float:
    _, value = _decide(scene["cable"], scene["wall"], scene["dots"])
    return float(round(value, 3))


def is_quarantined(scene: dict) -> bool:
    decision, value = _decide(scene["cable"], scene["wall"], scene["dots"])
    return decision is None or value < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list:
    """No latent alias exists: every sampled field is drawn."""
    return []
