"""Deterministic latent sampler and Pillow renderer."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

WIDTH = 640
HEIGHT = 640
BG = (249, 248, 244)
DARK = (45, 49, 56)
CYAN = (24, 174, 196)
GREEN = (42, 166, 82)
COLORS = {"orange": (237, 132, 38), "purple": (137, 82, 190), "blue": (54, 112, 201)}
PAL_PATTERNS = ("UUDD", "UDUD", "DUDU", "DDUU")
NONPAL_PATTERNS = ("UDDU", "DUUD")


def _turns(points: list[list[int]]) -> str:
    out = []
    for a, b, c in zip(points, points[1:], points[2:]):
        ux, uy = b[0] - a[0], b[1] - a[1]
        vx, vy = c[0] - b[0], c[1] - b[1]
        cross = ux * vy - uy * vx
        out.append("R" if cross > 0 else "L")  # image y increases downward
    return "".join(out)


def _points(pattern: str, base_y: int, vertical_lengths: list[int], x_offset: int) -> list[list[int]]:
    xs = [70 + x_offset, 160 + x_offset, 250 + x_offset, 340 + x_offset, 430 + x_offset, 520 + x_offset]
    pts = [[xs[0], base_y], [xs[1], base_y]]
    y = base_y
    for i, direction in enumerate(pattern):
        y += vertical_lengths[i] * (1 if direction == "D" else -1)
        pts.append([xs[i + 1], y])
        pts.append([xs[i + 2], y])
    return pts


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    answer_color = COLORS.keys()
    terminal_colors = list(answer_color)
    rng.shuffle(terminal_colors)
    pal = rng.choice(PAL_PATTERNS)
    patterns = [pal, *NONPAL_PATTERNS]
    rng.shuffle(patterns)
    rows = [112, 320, 528]
    lengths_base = [24, 30, 34, 28]
    routes = []
    for row, pattern, color in zip(rows, patterns, terminal_colors):
        lengths = lengths_base[:]
        rng.shuffle(lengths)
        x_offset = rng.choice([-10, 0, 10])
        pts = _points(pattern, row, lengths, x_offset)
        routes.append({
            "points": pts,
            "terminal_color": color,
            "pattern": pattern,
            "turns": _turns(pts),
        })
    return {"width": WIDTH, "height": HEIGHT, "routes": routes}


def analytic_gold(scene: dict) -> str:
    winners = [r["terminal_color"] for r in scene["routes"] if r["turns"] == r["turns"][::-1]]
    return winners[0] if len(winners) == 1 else "abstain"


def margin(scene: dict) -> float:
    distances = [sum(a != b for a, b in zip(r["turns"], r["turns"][::-1])) for r in scene["routes"]]
    ordered = sorted(distances)
    return float(ordered[1] - ordered[0]) if len(ordered) == 3 else 0.0


def is_quarantined(scene: dict) -> bool:
    routes = scene.get("routes", [])
    return len(routes) != 3 or analytic_gold(scene) == "abstain" or margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BG)
    draw = ImageDraw.Draw(image)
    for route in scene["routes"]:
        pts = [tuple(p) for p in route["points"]]
        draw.line(pts, fill=DARK, width=7, joint="curve")
        for x, y in pts[1:-1]:
            draw.ellipse((x - 6, y - 6, x + 6, y + 6), fill=CYAN, outline=DARK, width=2)
        sx, sy = pts[0]
        draw.ellipse((sx - 10, sy - 10, sx + 10, sy + 10), fill=GREEN, outline=DARK, width=3)
        tx, ty = pts[-1]
        rgb = COLORS[route["terminal_color"]]
        draw.rectangle((tx - 10, ty - 10, tx + 10, ty + 10), fill=rgb, outline=DARK, width=3)
    return image
