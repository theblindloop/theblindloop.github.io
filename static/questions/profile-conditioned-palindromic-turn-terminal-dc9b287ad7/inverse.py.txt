"""Pixel-only inverse arm; deliberately independent of latent-world modules."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

GREEN = (42, 166, 82)
CYAN = (24, 174, 196)
DARK = (45, 49, 56)
TERMINALS = {
    "orange": (237, 132, 38),
    "purple": (137, 82, 190),
    "blue": (54, 112, 201),
}


def _components(mask: np.ndarray, minimum: int = 12) -> list[tuple[float, float, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        sx = sy = n = 0
        while stack:
            yy, xx = stack.pop()
            sx += xx
            sy += yy
            n += 1
            for dy, dx in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                ny, nx = yy + dy, xx + dx
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if n >= minimum:
            out.append((sx / n, sy / n, n))
    return out


def _mask(arr: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    return np.all(arr == np.asarray(rgb, dtype=np.uint8), axis=2)


def _line_present(dark: np.ndarray, a: tuple[float, float], b: tuple[float, float]) -> bool:
    ax, ay = (int(round(v)) for v in a)
    bx, by = (int(round(v)) for v in b)
    if abs(ay - by) <= 2:
        lo, hi = sorted((ax, bx))
        if hi - lo < 25:
            return False
        y = (ay + by) // 2
        strip = dark[max(0, y - 2): y + 3, lo + 9: hi - 8]
    elif abs(ax - bx) <= 2:
        lo, hi = sorted((ay, by))
        if hi - lo < 18:
            return False
        x = (ax + bx) // 2
        strip = dark[lo + 9: hi - 8, max(0, x - 2): x + 3]
    else:
        return False
    return bool(strip.size and np.mean(np.any(strip, axis=0 if strip.shape[0] == 5 else 1)) > 0.8)


def _turn_string(points: list[tuple[float, float]]) -> str:
    turns = []
    for a, b, c in zip(points, points[1:], points[2:]):
        ux, uy = b[0] - a[0], b[1] - a[1]
        vx, vy = c[0] - b[0], c[1] - b[1]
        cross = ux * vy - uy * vx
        if abs(cross) < 100:
            raise ValueError("a marked bend is not visibly orthogonal")
        turns.append("R" if cross > 0 else "L")
    return "".join(turns)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    dark = _mask(arr, DARK)
    nodes: list[dict] = []
    for x, y, _ in _components(_mask(arr, GREEN)):
        nodes.append({"xy": (x, y), "kind": "start", "color": None})
    for x, y, _ in _components(_mask(arr, CYAN)):
        nodes.append({"xy": (x, y), "kind": "bend", "color": None})
    for name, rgb in TERMINALS.items():
        for x, y, _ in _components(_mask(arr, rgb)):
            nodes.append({"xy": (x, y), "kind": "terminal", "color": name})
    if sum(n["kind"] == "start" for n in nodes) != 3 or sum(n["kind"] == "bend" for n in nodes) != 24 or sum(n["kind"] == "terminal" for n in nodes) != 3:
        raise ValueError("required starts, bend dots, or terminals are missing")

    graph = {i: [] for i in range(len(nodes))}
    for i in range(len(nodes)):
        for j in range(i + 1, len(nodes)):
            if _line_present(dark, nodes[i]["xy"], nodes[j]["xy"]):
                # Only nearest collinear neighbors have an uninterrupted segment.
                graph[i].append(j)
                graph[j].append(i)
    winners = []
    for start in (i for i, n in enumerate(nodes) if n["kind"] == "start"):
        path = [start]
        previous = None
        current = start
        while nodes[current]["kind"] != "terminal":
            choices = [j for j in graph[current] if j != previous]
            if len(choices) != 1:
                raise ValueError("dark cable is broken, branching, or ambiguous")
            previous, current = current, choices[0]
            if current in path or len(path) > 10:
                raise ValueError("dark cable contains a cycle")
            path.append(current)
        if len(path) != 10 or any(nodes[i]["kind"] != "bend" for i in path[1:-1]):
            raise ValueError("cable does not contain exactly eight ordered bend dots")
        turns = _turn_string([nodes[i]["xy"] for i in path])
        if turns == turns[::-1]:
            winners.append(nodes[current]["color"])
    if len(winners) != 1:
        raise ValueError("image does not expose one unique palindromic route")
    return winners[0]
