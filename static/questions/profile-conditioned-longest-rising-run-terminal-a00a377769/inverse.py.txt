"""Pixel-only inverse arm for the rising-run chain diagram.

The arm segments marker ink, reconstructs each row's ordered chain, checks that
black stroke pixels connect adjacent markers, measures center-to-center link
lengths, and computes the longest strictly increasing contiguous run.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

BLACK = (31, 36, 42)
MARKERS = {
    "green": (31, 166, 91),
    "purple": (126, 82, 190),
    "red": (224, 67, 76),
    "blue": (43, 111, 211),
    "orange": (238, 137, 39),
}
TERMINALS = {"red", "blue", "orange"}


def _mask_near(array: np.ndarray, rgb: tuple[int, int, int], tolerance: int = 18) -> np.ndarray:
    target = np.asarray(rgb, dtype=np.int16)
    return np.max(np.abs(array.astype(np.int16) - target), axis=2) <= tolerance


def _components(mask: np.ndarray) -> list[dict]:
    """Return substantial four-connected components without external CV state."""
    ys, xs = np.nonzero(mask)
    remaining = {(int(x), int(y)) for y, x in zip(ys, xs)}
    found = []
    while remaining:
        seed = remaining.pop()
        stack = [seed]
        points = [seed]
        while stack:
            x, y = stack.pop()
            for neighbor in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
                if neighbor in remaining:
                    remaining.remove(neighbor)
                    stack.append(neighbor)
                    points.append(neighbor)
        if len(points) < 80:
            continue
        px = np.asarray([point[0] for point in points], dtype=np.float64)
        py = np.asarray([point[1] for point in points], dtype=np.float64)
        found.append(
            {
                "x": float(px.mean()),
                "y": float(py.mean()),
                "area": len(points),
                "diameter": float(max(px.max() - px.min() + 1, py.max() - py.min() + 1)),
            }
        )
    return found


def _black_link_present(
    black_mask: np.ndarray, left: dict, right: dict, marker_diameter: float
) -> bool:
    x1, y1 = float(left["x"]), float(left["y"])
    x2, y2 = float(right["x"]), float(right["y"])
    distance = math.hypot(x2 - x1, y2 - y1)
    if distance <= marker_diameter + 10:
        return False
    skip = marker_diameter * 0.75
    sample_count = max(12, int(distance - 2 * skip))
    hits = 0
    height, width = black_mask.shape
    for index in range(sample_count):
        along = skip + (distance - 2 * skip) * (index + 0.5) / sample_count
        fraction = along / distance
        x = int(round(x1 + (x2 - x1) * fraction))
        y = int(round(y1 + (y2 - y1) * fraction))
        xlo, xhi = max(0, x - 1), min(width, x + 2)
        ylo, yhi = max(0, y - 1), min(height, y + 2)
        hits += bool(black_mask[ylo:yhi, xlo:xhi].any())
    return hits / sample_count >= 0.9


def _longest_rising_run(values: list[float]) -> int:
    best = current = 1
    for previous, following in zip(values, values[1:]):
        if following > previous + 5.0:
            current += 1
            best = max(best, current)
        elif previous > following + 5.0:
            current = 1
        else:
            raise ValueError("adjacent link lengths are pixel-ambiguous")
    return best


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    nodes = []
    for kind, rgb in MARKERS.items():
        for component in _components(_mask_near(array, rgb)):
            component["kind"] = kind
            nodes.append(component)
    if len(nodes) != 21:
        raise ValueError(f"expected 21 visible node markers, found {len(nodes)}")

    marker_diameter = float(np.median([node["diameter"] for node in nodes]))
    row_tolerance = max(10.0, marker_diameter * 0.8)
    rows: list[list[dict]] = []
    for node in sorted(nodes, key=lambda item: item["y"]):
        if not rows or abs(node["y"] - np.mean([item["y"] for item in rows[-1]])) > row_tolerance:
            rows.append([node])
        else:
            rows[-1].append(node)
    if len(rows) != 3 or any(len(row) != 7 for row in rows):
        raise ValueError("could not reconstruct three seven-node chains")

    black_mask = _mask_near(array, BLACK, tolerance=24)
    scored: list[tuple[int, str]] = []
    for row in rows:
        starts = [node for node in row if node["kind"] == "green"]
        terminals = [node for node in row if node["kind"] in TERMINALS]
        purples = [node for node in row if node["kind"] == "purple"]
        if len(starts) != 1 or len(terminals) != 1 or len(purples) != 5:
            raise ValueError("row marker roles are not a single START, five waypoints, and terminal")
        start, terminal = starts[0], terminals[0]
        reverse = terminal["x"] < start["x"]
        ordered = sorted(row, key=lambda item: item["x"], reverse=reverse)
        if ordered[0] is not start or ordered[-1] is not terminal:
            raise ValueError("START and terminal do not bound an ordered chain")
        for left, right in zip(ordered, ordered[1:]):
            if not _black_link_present(black_mask, left, right, marker_diameter):
                raise ValueError("a required black link is missing or broken")
        lengths = [
            math.hypot(right["x"] - left["x"], right["y"] - left["y"])
            for left, right in zip(ordered, ordered[1:])
        ]
        scored.append((_longest_rising_run(lengths), str(terminal["kind"])))

    scored.sort(reverse=True)
    if scored[0][0] == scored[1][0]:
        raise ValueError("top two rising-run scores are tied")
    return scored[0][1]
