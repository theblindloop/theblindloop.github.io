"""Deterministic renderer and analytic arm for the rising-run world."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 860, 520
BACKGROUND = (248, 247, 242)
PANEL_FILL = (255, 255, 255)
PANEL_EDGE = (210, 214, 218)
BLACK = (31, 36, 42)
GREEN = (31, 166, 91)
PURPLE = (126, 82, 190)
TERMINAL_RGB = {
    "red": (224, 67, 76),
    "blue": (43, 111, 211),
    "orange": (238, 137, 39),
}
TERMINAL_COLORS = ("red", "blue", "orange")
NODE_RADIUS = 11
START_X = 80
ROW_YS = (105, 260, 415)

# Every route has the same multiset and total length (700 px). Only order differs.
# Their longest strictly increasing runs contain 6, 4, and 3 links respectively.
LENGTH_PATTERNS = (
    (75, 90, 105, 120, 135, 175),
    (105, 120, 135, 175, 75, 90),
    (120, 135, 175, 90, 105, 75),
)


def _longest_rising_run(values: list[float] | tuple[float, ...]) -> int:
    if not values:
        return 0
    best = current = 1
    for previous, following in zip(values, values[1:]):
        if following > previous:
            current += 1
            best = max(best, current)
        else:
            current = 1
    return best


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; consecutive seeds cycle the winning color."""
    rng = random.Random(int(seed))
    winner_color = TERMINAL_COLORS[int(seed) % len(TERMINAL_COLORS)]
    other_colors = [color for color in TERMINAL_COLORS if color != winner_color]
    rng.shuffle(other_colors)
    color_for_pattern = {0: winner_color, 1: other_colors[0], 2: other_colors[1]}
    pattern_order = [0, 1, 2]
    rng.shuffle(pattern_order)
    chains = []
    for y, pattern_index in zip(ROW_YS, pattern_order):
        chains.append(
            {
                "y": y,
                "terminal": color_for_pattern[pattern_index],
                "lengths": list(LENGTH_PATTERNS[pattern_index]),
            }
        )
    return {"chains": chains}


def render(scene: dict) -> Image.Image:
    """Render three isolated straight chains with links behind circular nodes."""
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for chain in scene["chains"]:
        y = int(chain["y"])
        lengths = [int(value) for value in chain["lengths"]]
        terminal = str(chain["terminal"])
        draw.rounded_rectangle(
            (28, y - 60, WIDTH - 28, y + 60),
            radius=16,
            fill=PANEL_FILL,
            outline=PANEL_EDGE,
            width=2,
        )
        xs = [START_X]
        for length in lengths:
            xs.append(xs[-1] + length)
        for left, right in zip(xs, xs[1:]):
            draw.line((left, y, right, y), fill=BLACK, width=7)
        fills = [GREEN] + [PURPLE] * (len(xs) - 2) + [TERMINAL_RGB[terminal]]
        for x, fill in zip(xs, fills):
            draw.ellipse(
                (x - NODE_RADIUS, y - NODE_RADIUS, x + NODE_RADIUS, y + NODE_RADIUS),
                fill=fill,
                outline=BLACK,
                width=3,
            )
    return image


def analytic_gold(scene: dict) -> str:
    scored = []
    for chain in scene["chains"]:
        score = _longest_rising_run(chain["lengths"])
        scored.append((score, chain["terminal"]))
    scored.sort(reverse=True)
    if len(scored) < 2 or scored[0][0] == scored[1][0]:
        raise ValueError("scene has no unique longest rising run")
    return str(scored[0][1])


def margin(scene: dict) -> float:
    scores = sorted(
        (_longest_rising_run(chain["lengths"]) for chain in scene["chains"]),
        reverse=True,
    )
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    scores = []
    minimum_comparison_gap = float("inf")
    for chain in scene["chains"]:
        values = [float(value) for value in chain["lengths"]]
        scores.append(_longest_rising_run(values))
        differences = [abs(right - left) for left, right in zip(values, values[1:])]
        minimum_comparison_gap = min(minimum_comparison_gap, *differences)
    scores.sort(reverse=True)
    return minimum_comparison_gap < 8.0 or len(scores) < 2 or scores[0] == scores[1]


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """There are no hidden render-invariant scene aliases."""
    return []
