"""Deterministic world renderer for a visible sequence on one connected ribbon."""

from __future__ import annotations

import hashlib
import math
import random

from PIL import Image, ImageDraw

CANONICAL_ORDER = ("red", "orange", "yellow", "green", "blue", "violet")
COLORS = {
    "red": (211, 45, 55),
    "orange": (238, 126, 34),
    "yellow": (236, 190, 36),
    "green": (39, 151, 88),
    "blue": (43, 101, 194),
    "violet": (126, 72, 169),
}
BACKGROUND = (248, 247, 242)


def _rng(seed: int, namespace: str) -> random.Random:
    digest = hashlib.sha256(f"{namespace}:{int(seed)}".encode()).digest()
    return random.Random(int.from_bytes(digest[:8], "big"))


def sample_scene(seed: int) -> dict:
    """Sample balanced control/counterfactual order and answer-neutral geometry."""
    seed = int(seed)
    geom = _rng(seed, "rainbow-ribbon-geometry-v1")
    order_rng = _rng(seed, "rainbow-ribbon-order-v1")
    if seed % 2:
        color_order = list(CANONICAL_ORDER)
    else:
        # A varied two-position swap retains four canonical positions, preventing
        # any fixed location or color from being a sufficient counterfactual cue.
        i = order_rng.randrange(6)
        j = (i + order_rng.choice((1, 2, 3, 4, 5))) % 6
        color_order = list(CANONICAL_ORDER)
        color_order[i], color_order[j] = color_order[j], color_order[i]
    return {
        "width": 512,
        "height": 320,
        "color_order": color_order,
        "x_left": geom.randint(27, 39),
        "x_right": geom.randint(473, 485),
        "center_y": geom.randint(148, 172),
        "amplitude": geom.randint(42, 64),
        "phase": geom.uniform(-0.45, 0.45),
        "thickness": geom.randint(17, 23),
        "route_style": geom.randrange(3),
        "alias_cycle": geom.randrange(6),
    }


def _center_y(scene: dict, x: float) -> float:
    left, right = float(scene["x_left"]), float(scene["x_right"])
    t = (x - left) / (right - left)
    phase = float(scene["phase"])
    style = int(scene["route_style"])
    cycle_shift = (int(scene["alias_cycle"]) % 6 - 2.5) * 0.7
    if style == 0:
        wave = math.sin(2.0 * math.pi * (1.25 * t + phase))
    elif style == 1:
        wave = 0.72 * math.sin(2.0 * math.pi * (1.55 * t + phase)) + 0.28 * math.sin(
            2.0 * math.pi * (3.1 * t + 0.4)
        )
    else:
        wave = 0.78 * math.sin(2.0 * math.pi * (1.4 * t + phase)) + 0.22 * math.cos(
            2.0 * math.pi * (2.8 * t - 0.2)
        )
    return float(scene["center_y"]) + float(scene["amplitude"]) * wave + cycle_shift


def render(scene: dict) -> Image.Image:
    """Draw adjacent color strokes whose union is one thin x-monotone path."""
    scale = 4
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    left, right = int(scene["x_left"]), int(scene["x_right"])
    span = right - left
    order = tuple(scene["color_order"])
    if sorted(order) != sorted(CANONICAL_ORDER) or len(order) != 6:
        raise ValueError("color_order must contain each declared ribbon color exactly once")
    points = [(x * scale, round(_center_y(scene, x) * scale)) for x in range(left, right + 1)]
    line_width = int(scene["thickness"]) * scale
    for section, name in enumerate(order):
        start = round(section * span / 6)
        stop = round((section + 1) * span / 6)
        # One-point overlap makes differently colored sections physically touch.
        segment = points[max(0, start - 1) : min(len(points), stop + 2)]
        draw.line(segment, fill=COLORS[name], width=line_width, joint="curve")
    # Small same-color endpoint disks make both ends visually deliberate without
    # introducing a separate pointer or answer-correlated decoration.
    radius = line_width // 2
    for point, name in ((points[0], order[0]), (points[-1], order[-1])):
        x, y = point
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=COLORS[name])
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "yes" if tuple(scene["color_order"]) == CANONICAL_ORDER else "no"


def margin(scene: dict) -> float:
    section_span = (float(scene["x_right"]) - float(scene["x_left"])) / 6.0
    return round(min(section_span - 42.0, float(scene["thickness"]) - 9.0), 3)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 7.5


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["color_order"] = list(scene["color_order"])
    twin["alias_cycle"] = int(scene["alias_cycle"]) + 6
    return [("alias_cycle_plus_six", twin)]
