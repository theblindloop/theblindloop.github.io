"""Independent pixel-only inverse for the single colored ribbon."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

CANONICAL = ("red", "orange", "yellow", "green", "blue", "violet")
PALETTE = np.array(
    [(211, 45, 55), (238, 126, 34), (236, 190, 36), (39, 151, 88), (43, 101, 194), (126, 72, 169)],
    dtype=np.int16,
)


def _largest_component_fraction(mask: np.ndarray) -> float:
    """Return the fraction of foreground in its largest 8-connected component."""
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    best = 0
    total = int(mask.sum())
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = deque([(int(y), int(x))])
        size = 0
        while queue:
            cy, cx = queue.popleft()
            size += 1
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dy == dx == 0:
                        continue
                    ny, nx = cy + dy, cx + dx
                    if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        best = max(best, size)
    return best / total if total else 0.0


def _mode_filter(values: list[int], radius: int = 2) -> list[int]:
    out = []
    for i in range(len(values)):
        window = values[max(0, i - radius) : min(len(values), i + radius + 1)]
        counts = np.bincount(np.asarray(window), minlength=6)
        out.append(int(counts.argmax()))
    return out


def _recover_sequence(image: Image.Image) -> tuple[str, ...]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    palette = PALETTE.astype(np.int32)
    distances = ((arr[:, :, None, :] - palette[None, None, :, :]) ** 2).sum(axis=3)
    labels = distances.argmin(axis=2)
    foreground = distances.min(axis=2) < 58**2
    if int(foreground.sum()) < 4000:
        raise ValueError("too little ribbon ink")
    if _largest_component_fraction(foreground) < 0.965:
        raise ValueError("ribbon is not one intact connected component")

    counts = np.stack([((labels == i) & foreground).sum(axis=0) for i in range(6)], axis=1)
    strength = counts.max(axis=1)
    active = strength >= 5
    xs = np.flatnonzero(active)
    if len(xs) < int(image.width * 0.78):
        raise ValueError("ribbon does not span enough of the image")
    left, right = int(xs[0]), int(xs[-1])
    gaps = np.flatnonzero(~active[left : right + 1])
    if len(gaps):
        runs = np.split(gaps, np.where(np.diff(gaps) != 1)[0] + 1)
        if max(map(len, runs)) > 5:
            raise ValueError("ribbon has an interior break")

    column_labels = counts[left : right + 1].argmax(axis=1).tolist()
    column_labels = _mode_filter(column_labels)
    runs: list[tuple[int, int]] = []
    for label in column_labels:
        if runs and runs[-1][0] == label:
            runs[-1] = (label, runs[-1][1] + 1)
        else:
            runs.append((label, 1))
    # Suppress only tiny antialias transition glitches; real sections are >65 px.
    cleaned: list[tuple[int, int]] = []
    for label, length in runs:
        if length <= 4 and cleaned:
            cleaned[-1] = (cleaned[-1][0], cleaned[-1][1] + length)
        elif cleaned and cleaned[-1][0] == label:
            cleaned[-1] = (label, cleaned[-1][1] + length)
        else:
            cleaned.append((label, length))
    if len(cleaned) != 6:
        raise ValueError(f"expected six visible color sections, recovered {len(cleaned)}")
    if min(length for _, length in cleaned) < 45:
        raise ValueError("a visible color section is too short")
    recovered = tuple(CANONICAL[label] for label, _ in cleaned)
    if len(set(recovered)) != 6:
        raise ValueError("each declared color must appear exactly once")
    return recovered


def decision_from_image(image: Image.Image) -> str:
    """Return yes iff the complete pixel-recovered sequence is canonical."""
    return "yes" if _recover_sequence(image) == CANONICAL else "no"
