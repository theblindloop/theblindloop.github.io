"""Scene-free pixel inverse arm for lru_slot_replay.

Receives only a rendered RGB image.  It re-finds the rack discs, the black query
pointer and the ordered row of request cards from the raster alone, then replays
the declared least-recently-used loading rule and returns the colour standing in
the pointed-at slot.

This module imports no renderer, prompt, generator, verifier or gold code and
never sees latent scene data.  Its geometry knowledge is limited to what the
public question text states: the rack is the upper band of discs in one row, the
request cards are the lower band read left to right, and the pointer is the only
black triangle.
"""

from __future__ import annotations

from PIL import Image

ORACLE_VERSION = "lru_slot_replay-oracle-0.1.0"

_PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (206, 48, 42),
    "orange": (238, 150, 26),
    "blue": (46, 86, 196),
    "green": (36, 142, 72),
    "purple": (128, 60, 168),
}
_INK = (15, 15, 15)
_MIN_BLOB = 200


def _components(pixels: set[tuple[int, int]]) -> list[set[tuple[int, int]]]:
    """Four-connected components of a pixel set."""
    seen: set[tuple[int, int]] = set()
    out: list[set[tuple[int, int]]] = []
    for start in pixels:
        if start in seen:
            continue
        stack = [start]
        seen.add(start)
        blob = {start}
        while stack:
            x, y = stack.pop()
            for nx, ny in ((x + 1, y), (x - 1, y), (x, y + 1), (x, y - 1)):
                p = (nx, ny)
                if p in pixels and p not in seen:
                    seen.add(p)
                    blob.add(p)
                    stack.append(p)
        out.append(blob)
    return out


def _centroid(blob: set[tuple[int, int]]) -> tuple[float, float]:
    n = float(len(blob))
    return sum(p[0] for p in blob) / n, sum(p[1] for p in blob) / n


def _blobs(image: Image.Image) -> tuple[list[tuple[float, float, str]], list[tuple[float, float]]]:
    rgb = image.convert("RGB")
    w, h = rgb.size
    px = rgb.load()
    by_colour: dict[str, set[tuple[int, int]]] = {name: set() for name in _PALETTE}
    ink: set[tuple[int, int]] = set()
    lookup = {v: k for k, v in _PALETTE.items()}
    for y in range(h):
        for x in range(w):
            p = px[x, y]
            name = lookup.get(p)
            if name is not None:
                by_colour[name].add((x, y))
            elif p == _INK:
                ink.add((x, y))

    discs: list[tuple[float, float, str]] = []
    for name, pixels in by_colour.items():
        for blob in _components(pixels):
            if len(blob) < _MIN_BLOB:
                continue
            cx, cy = _centroid(blob)
            discs.append((cx, cy, name))
    pointers = [_centroid(b) for b in _components(ink) if len(b) >= 60]
    return discs, pointers


def _replay(initial: list[str], requests: list[str]) -> list[str]:
    state = list(initial)
    recency = list(initial)
    for colour in requests:
        if colour not in state:
            victim = next(c for c in recency if c in state)
            state[state.index(victim)] = colour
        recency = [c for c in recency if c != colour] + [colour]
    return state


def decision_from_image(image: Image.Image) -> str:
    discs, pointers = _blobs(image)
    if not discs or not pointers:
        return "unreadable"

    ys = sorted(cy for _, cy, _ in discs)
    split = (ys[0] + ys[-1]) / 2.0
    rack = sorted([d for d in discs if d[1] < split], key=lambda d: d[0])
    cards = sorted([d for d in discs if d[1] >= split], key=lambda d: d[0])
    if len(rack) < 2 or not cards:
        return "unreadable"

    rack_y = sum(d[1] for d in rack) / float(len(rack))
    pointer = min(pointers, key=lambda p: abs(p[1] - rack_y))
    slot = min(range(len(rack)), key=lambda i: abs(rack[i][0] - pointer[0]))

    final = _replay([d[2] for d in rack], [d[2] for d in cards])
    return final[slot]
