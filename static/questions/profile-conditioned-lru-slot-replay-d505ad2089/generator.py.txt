"""Renderer, analytic gold and margin for lru_slot_replay.

A rack of four slots holds four distinct coloured discs.  A black triangle drawn
under the rack points at one slot: that is the queried place.  Below the rack an
ordered row of request cards is applied left to right.  A card asks for its
drawn colour.  If that colour is already standing on the rack nothing on the
rack moves; the request only refreshes how recently that colour was used.  If it
is not on the rack it must be loaded, and it takes the slot of the colour whose
own most recent use is the oldest among the four colours currently standing on
the rack.  A colour's most recent use is the last card that asked for it, and a
colour that no card has yet asked for counts as used before every card, in the
drawn left-to-right order of the starting rack.

The decision is the colour standing in the pointed-at slot after every card has
acted.  Only the starting rack is drawn.

The scene is a plain JSON-able dict.  Cards live in an unordered bag: each card
carries its own drawn position, so the bag may be permuted without changing the
raster or the answer.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "lru_slot_replay-renderer-0.1.0"

# ---------------------------------------------------------------- palette ---
COLOURS: dict[str, tuple[int, int, int]] = {
    "red": (206, 48, 42),
    "orange": (238, 150, 26),
    "blue": (46, 86, 196),
    "green": (36, 142, 72),
    "purple": (128, 60, 168),
}
COLOUR_NAMES = ("red", "orange", "blue", "green", "purple")

BG = (245, 247, 250)
CELL_FILL = (228, 228, 232)
CELL_EDGE = (160, 160, 168)
CARD_FILL = (250, 238, 205)
CARD_EDGE = (176, 162, 128)
INK = (15, 15, 15)
ARROW_GREY = (150, 150, 150)

# ----------------------------------------------------------------- layout ---
W, H = 760, 300
N_SLOTS = 4
CELL, CELL_GAP = 64, 18
PITCH = CELL + CELL_GAP
RACK_X0 = (W - (N_SLOTS * CELL + (N_SLOTS - 1) * CELL_GAP)) // 2
RACK_Y0 = 28
DISC_R = 22
POINTER_DY = 20                     # below the rack cells
HALF_PITCH = PITCH / 2.0
JITTER_MAX = 26.0
QUARANTINE_MARGIN = 8.0

CARD_W, CARD_H, CARD_GAP = 84, 84, 18
CARD_Y0 = 160
CARD_DISC_R = 20


def _cell_box(i: int) -> tuple[int, int, int, int]:
    x0 = RACK_X0 + i * PITCH
    return x0, RACK_Y0, x0 + CELL, RACK_Y0 + CELL


def _cell_centre(i: int) -> float:
    x0, _, x1, _ = _cell_box(i)
    return (x0 + x1) / 2.0


def _cards_x0(n_cards: int) -> int:
    total = n_cards * CARD_W + (n_cards - 1) * CARD_GAP
    return (W - total) // 2


def _card_x0(n_cards: int, pos: int) -> int:
    return _cards_x0(n_cards) + pos * (CARD_W + CARD_GAP)


# --------------------------------------------------------------- readings ---
def _ordered_cards(scene: dict[str, Any]) -> list[dict[str, Any]]:
    return sorted(scene["cards"], key=lambda c: int(c["pos"]))


def _pointer_x(scene: dict[str, Any]) -> float:
    return _cell_centre(int(scene["query_slot"])) + float(scene["jitter"])


def _read_query_slot(scene: dict[str, Any]) -> int:
    """The declared reading: the rack slot whose centre is nearest the pointer."""
    px = _pointer_x(scene)
    best, best_d = 0, None
    for i in range(N_SLOTS):
        d = abs(px - _cell_centre(i))
        if best_d is None or d < best_d:
            best, best_d = i, d
    return best


def _replay(initial: list[str], requests: list[str]) -> list[str]:
    """Least-recently-used replacement over a full rack of distinct colours."""
    state = list(initial)
    recency = list(initial)          # oldest use first; leftmost drawn is oldest
    for colour in requests:
        if colour not in state:
            victim = next(c for c in recency if c in state)
            state[state.index(victim)] = colour
        recency = [c for c in recency if c != colour] + [colour]
    return state


def _final_state(scene: dict[str, Any]) -> list[str]:
    return _replay(
        list(scene["initial"]),
        [str(c["colour"]) for c in _ordered_cards(scene)],
    )


def _unique_mode(values: list[str]) -> str | None:
    counts: dict[str, int] = {}
    for v in values:
        counts[v] = counts.get(v, 0) + 1
    if not counts:
        return None
    top = max(counts.values())
    winners = [k for k, v in counts.items() if v == top]
    return winners[0] if len(winners) == 1 else None


# --------------------------------------------------------------- sampling ---
def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(seed)
    target_answer = COLOUR_NAMES[seed % len(COLOUR_NAMES)]
    near_boundary = (seed % 7) == 3

    relax_after = 3000
    for attempt in range(6000):
        n_cards = rng.choice([4, 5, 6])
        initial = rng.sample(list(COLOUR_NAMES), N_SLOTS)
        requests = [rng.choice(COLOUR_NAMES) for _ in range(n_cards)]
        query_slot = rng.randrange(N_SLOTS)
        if near_boundary:
            mag = rng.uniform(HALF_PITCH - 7.0, HALF_PITCH - 1.5)
            jitter = round(mag * rng.choice([-1.0, 1.0]), 4)
        else:
            jitter = round(rng.uniform(-JITTER_MAX, JITTER_MAX), 4)

        cards = [{"pos": i, "colour": requests[i]} for i in range(n_cards)]
        rng.shuffle(cards)           # the bag order carries nothing
        scene = {
            "canvas": [W, H],
            "n_slots": N_SLOTS,
            "n_cards": n_cards,
            "query_slot": query_slot,
            "jitter": jitter,
            "initial": initial,
            "cards": cards,
        }
        read_slot = _read_query_slot(scene)
        final = _final_state(scene)
        answer = final[read_slot]
        if answer != target_answer:
            continue
        if attempt < relax_after:
            ordered = [str(c["colour"]) for c in _ordered_cards(scene)]
            if initial[read_slot] == answer:
                continue                       # "read the pointed slot as drawn"
            if ordered[-1] == answer:
                continue                       # "the last card's colour"
            if ordered[0] == answer:
                continue                       # "the first card's colour"
            if _unique_mode(ordered) == answer:
                continue                       # "the commonest request"
            if answer not in ordered:
                continue                       # keep every answer a loaded colour
            if len({c for c in ordered if c not in initial}) < 1:
                continue                       # at least one eviction happens
        return scene
    raise RuntimeError(f"sampling failed for seed {seed}")


# ------------------------------------------------------------------ gold ----
def analytic_gold(scene: dict[str, Any]) -> str:
    return _final_state(scene)[_read_query_slot(scene)]


def margin(scene: dict[str, Any]) -> float:
    return round(float(HALF_PITCH - abs(float(scene["jitter"]))), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """Permutations of the card bag: pixel-identical and answer preserving."""
    bag = list(scene["cards"])
    out: list[tuple[str, dict[str, Any]]] = []
    for name, perm in (
        ("card_bag_reversed", list(reversed(bag))),
        ("card_bag_rotated", bag[1:] + bag[:1]),
        ("card_bag_pos_sorted", sorted(bag, key=lambda c: int(c["pos"]))),
    ):
        alt = dict(scene)
        alt["cards"] = [dict(c) for c in perm]
        out.append((name, alt))
    return out


# ---------------------------------------------------------------- render ----
def render(scene: dict[str, Any]) -> Image.Image:
    img = Image.new("RGB", (W, H), BG)
    d = ImageDraw.Draw(img)

    for i in range(N_SLOTS):
        x0, y0, x1, y1 = _cell_box(i)
        d.rectangle([x0, y0, x1, y1], fill=CELL_FILL, outline=CELL_EDGE, width=2)
        cx, cy = (x0 + x1) / 2.0, (y0 + y1) / 2.0
        d.ellipse(
            [cx - DISC_R, cy - DISC_R, cx + DISC_R, cy + DISC_R],
            fill=COLOURS[scene["initial"][i]],
        )

    px = int(round(_pointer_x(scene)))
    py = RACK_Y0 + CELL + POINTER_DY
    d.polygon([(px, py - 16), (px - 11, py), (px + 11, py)], fill=INK)

    n = int(scene["n_cards"])
    for card in _ordered_cards(scene):
        cx0 = _card_x0(n, int(card["pos"]))
        d.rectangle(
            [cx0, CARD_Y0, cx0 + CARD_W, CARD_Y0 + CARD_H],
            fill=CARD_FILL,
            outline=CARD_EDGE,
            width=2,
        )
        gx, gy = cx0 + CARD_W / 2.0, CARD_Y0 + CARD_H / 2.0
        d.ellipse(
            [gx - CARD_DISC_R, gy - CARD_DISC_R, gx + CARD_DISC_R, gy + CARD_DISC_R],
            fill=COLOURS[card["colour"]],
        )

    ax0 = _cards_x0(n)
    ax1 = ax0 + n * CARD_W + (n - 1) * CARD_GAP
    ay = CARD_Y0 + CARD_H + 22
    d.line([ax0, ay, ax1, ay], fill=ARROW_GREY, width=4)
    d.polygon([(ax1 + 12, ay), (ax1 - 6, ay - 9), (ax1 - 6, ay + 9)], fill=ARROW_GREY)
    return img
