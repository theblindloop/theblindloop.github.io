"""Independent pixel-only inverse for the hinged-grid reveal.

It locates the gold hinge, segments circular dot components, reflects every
right-hand centroid across the hinge, retains geometric coincidences, recovers
the lattice spacing from those points, and compares the result with four public
arrow templates.  No latent data or generation module is available here.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_LEFT_COLORS = ((34, 55, 95), (22, 112, 103), (91, 58, 137))
_RIGHT_COLORS = ((225, 92, 38), (190, 61, 112), (211, 126, 30))
_GOLD = (211, 166, 48)
_ARROW_NAMES = ("up", "right", "down", "left")


def _rotated(points: set[tuple[int, int]], turns: int) -> set[tuple[int, int]]:
    out = set(points)
    for _ in range(turns % 4):
        out = {(4 - row, col) for col, row in out}
    return out


_UP = {
    (2, 0),
    (1, 1), (2, 1), (3, 1),
    (0, 2), (1, 2), (2, 2), (3, 2), (4, 2),
    (2, 3), (2, 4),
}
_TEMPLATES = {
    name: _rotated(_UP, turns) for turns, name in enumerate(_ARROW_NAMES)
}


def _near_any(array: np.ndarray, colors: tuple[tuple[int, int, int], ...], tolerance: float = 12.0) -> np.ndarray:
    work = array.astype(np.int32)
    best = np.full(array.shape[:2], 10**9, dtype=np.int32)
    for color in colors:
        delta = work - np.asarray(color, dtype=np.int32)
        best = np.minimum(best, np.sum(delta * delta, axis=2, dtype=np.int32))
    return best <= int(tolerance * tolerance)


def _component_centroids(mask: np.ndarray) -> list[tuple[float, float, int]]:
    """Return centroids of dot-sized 8-connected components."""
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components: list[tuple[float, float, int]] = []
    for start_y, start_x in np.argwhere(mask):
        sy, sx = int(start_y), int(start_x)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for oy in (-1, 0, 1):
                for ox in (-1, 0, 1):
                    if ox == 0 and oy == 0:
                        continue
                    ny, nx = y + oy, x + ox
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        area = len(xs)
        if 55 <= area <= 240:
            components.append((float(np.mean(xs)), float(np.mean(ys)), area))
    return components


def _hinge_x(array: np.ndarray) -> float:
    gold = _near_any(array, (_GOLD,), tolerance=10.0)
    counts = gold.sum(axis=0)
    peak = int(counts.max()) if counts.size else 0
    if peak < 80:
        raise ValueError("abstain: visible gold hinge is missing")
    columns = np.flatnonzero(counts >= 0.72 * peak)
    if len(columns) < 2:
        raise ValueError("abstain: gold hinge cannot be localized")
    # Restrict to the consecutive peak group containing the tallest column; this
    # prevents the curved instruction arrow from biasing the hinge coordinate.
    peak_col = int(np.argmax(counts))
    group = [peak_col]
    x = peak_col - 1
    while x >= 0 and counts[x] >= 0.72 * peak:
        group.append(x)
        x -= 1
    x = peak_col + 1
    while x < len(counts) and counts[x] >= 0.72 * peak:
        group.append(x)
        x += 1
    return float(np.average(group, weights=[max(1, int(counts[item])) for item in group]))


def _coincidences(
    left: list[tuple[float, float, int]],
    right: list[tuple[float, float, int]],
    hinge: float,
) -> list[tuple[float, float]]:
    reflected = [(2.0 * hinge - x, y) for x, y, _ in right]
    available = set(range(len(reflected)))
    kept: list[tuple[float, float]] = []
    for lx, ly, _ in left:
        if not available:
            break
        nearest = min(available, key=lambda index: (lx - reflected[index][0]) ** 2 + (ly - reflected[index][1]) ** 2)
        rx, ry = reflected[nearest]
        if (lx - rx) ** 2 + (ly - ry) ** 2 <= 8.0**2:
            kept.append(((lx + rx) / 2.0, (ly + ry) / 2.0))
            available.remove(nearest)
    return kept


def _lattice_cells(points: list[tuple[float, float]]) -> set[tuple[int, int]]:
    if len(points) < 7:
        raise ValueError("abstain: too few post-fold coincidences")
    unique_x = sorted({round(x, 3) for x, _ in points})
    unique_y = sorted({round(y, 3) for _, y in points})
    differences = [
        later - earlier
        for values in (unique_x, unique_y)
        for earlier, later in zip(values, values[1:])
        if later - earlier > 10.0
    ]
    if not differences:
        raise ValueError("abstain: coincidence lattice spacing is unrecoverable")
    step = min(differences)
    if not 20.0 <= step <= 40.0:
        raise ValueError("abstain: recovered lattice spacing is implausible")
    x0 = min(x for x, _ in points)
    y0 = min(y for _, y in points)
    cells = {(int(round((x - x0) / step)), int(round((y - y0) / step))) for x, y in points}
    if len(cells) != len(points):
        raise ValueError("abstain: multiple coincidences collapse into one lattice cell")
    return cells


def _classify(cells: set[tuple[int, int]]) -> tuple[str, float, dict[str, int]]:
    distances: dict[str, int] = {}
    for name, template in _TEMPLATES.items():
        distances[name] = min(
            len(cells.symmetric_difference({(col + dx, row + dy) for col, row in template}))
            for dx in range(-7, 12)
            for dy in range(-7, 12)
        )
    ranked = sorted(_ARROW_NAMES, key=lambda name: (distances[name], _ARROW_NAMES.index(name)))
    gap = float(distances[ranked[1]] - distances[ranked[0]])
    if gap < 2.0:
        raise ValueError(f"abstain: runner-up arrow margin is only {gap:.1f} cells")
    return ranked[0], gap, distances


def measure_image(image: Image.Image) -> dict:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    hinge = _hinge_x(array)
    left = _component_centroids(_near_any(array, _LEFT_COLORS))
    right = _component_centroids(_near_any(array, _RIGHT_COLORS))
    if not 20 <= len(left) <= 40 or not 20 <= len(right) <= 40:
        raise ValueError(f"abstain: expected two populated dot grids, found {len(left)} and {len(right)} dots")
    if any(x >= hinge for x, _, _ in left) or any(x <= hinge for x, _, _ in right):
        raise ValueError("abstain: dot colors do not remain on their visible grid halves")
    kept = _coincidences(left, right, hinge)
    cells = _lattice_cells(kept)
    decision, score_gap, distances = _classify(cells)
    return {
        "decision": decision,
        "margin": score_gap,
        "distances": distances,
        "left_dot_count": len(left),
        "right_dot_count": len(right),
        "coincidence_count": len(kept),
        "hinge_x": hinge,
    }


def decision_from_image(image: Image.Image) -> str:
    """Return the arrow recovered solely from final RGB pixels."""
    return str(measure_image(image)["decision"])
