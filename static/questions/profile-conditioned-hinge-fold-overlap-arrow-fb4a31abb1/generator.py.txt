"""Deterministic renderer for a mirror-fold and overlap reveal task.

The analytic decision is derived from the two rendered occupancy lists, not
from an answer tag.  Folding maps right column c to left column 8-c.  The kept
cells are the set intersection after that reflection.
"""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

GRID_SIZE = 9
WIDTH = 720
HEIGHT = 430
ARROW_NAMES = ("up", "right", "down", "left")
PALETTES = (
    ((34, 55, 95), (225, 92, 38)),
    ((22, 112, 103), (190, 61, 112)),
    ((91, 58, 137), (211, 126, 30)),
)
BACKGROUNDS = ((250, 250, 247), (246, 249, 252), (255, 249, 242))
GRID_COLOR = (190, 199, 207)
GOLD = (211, 166, 48)


def _rotated(points: set[tuple[int, int]], turns: int) -> set[tuple[int, int]]:
    out = set(points)
    for _ in range(turns % 4):
        out = {(4 - row, col) for col, row in out}
    return out


_UP = {
    (2, 0),
    (1, 1), (2, 1), (3, 1),
    (0, 2), (1, 2), (2, 2), (3, 2), (4, 2),
    (2, 3), (2, 4),
}
TEMPLATES: dict[str, set[tuple[int, int]]] = {
    name: _rotated(_UP, turns) for turns, name in enumerate(ARROW_NAMES)
}


def _cell_set(values: list[list[int]] | list[tuple[int, int]]) -> set[tuple[int, int]]:
    return {(int(value[0]), int(value[1])) for value in values}


def _folded_overlap(scene: dict) -> set[tuple[int, int]]:
    left = _cell_set(scene["left_cells"])
    right_folded = {(GRID_SIZE - 1 - col, row) for col, row in _cell_set(scene["right_cells"])}
    return left & right_folded


def _template_distances(cells: set[tuple[int, int]]) -> dict[str, int]:
    """Best symmetric-difference distance for every arrow over all translations."""
    distances: dict[str, int] = {}
    for name, template in TEMPLATES.items():
        best = min(
            len(cells.symmetric_difference({(col + dx, row + dy) for col, row in template}))
            for dx in range(-4, GRID_SIZE + 1)
            for dy in range(-4, GRID_SIZE + 1)
        )
        distances[name] = best
    return distances


def _classification(cells: set[tuple[int, int]]) -> tuple[str, float, dict[str, int]]:
    distances = _template_distances(cells)
    ranked = sorted(ARROW_NAMES, key=lambda name: (distances[name], ARROW_NAMES.index(name)))
    gap = float(distances[ranked[1]] - distances[ranked[0]])
    return ranked[0], gap, distances


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; answer class cycles exactly with the integer seed."""
    seed = int(seed)
    rng = random.Random(seed ^ 0x5A17C3)
    target_name = ARROW_NAMES[seed % len(ARROW_NAMES)]

    # Blocks of four consecutive seeds share nuisance settings, so each answer
    # receives every palette, radius, background, and scale at the same rate.
    block = seed // 4
    palette_index = block % len(PALETTES)
    background_variant = (block // len(PALETTES)) % len(BACKGROUNDS)
    dot_radius = 5 + ((block // 3) % 3)
    cell_size = (28, 30, 32)[(block // 5) % 3]
    crease_x = 354 + 6 * ((block // 7) % 3)
    top = 82 + 4 * ((block // 11) % 3)

    dx = rng.randrange(0, GRID_SIZE - 4)
    dy = rng.randrange(0, GRID_SIZE - 4)
    signal = {(col + dx, row + dy) for col, row in TEMPLATES[target_name]}

    # Preserve a constant 11 coincidences while varying reveal clarity.  Removed
    # arrow cells are replaced by common noise cells.  Only cases with an explicit
    # >=2-cell runner-up gap are admitted.
    noise_count = block % 3
    all_cells = {(col, row) for col in range(GRID_SIZE) for row in range(GRID_SIZE)}
    common: set[tuple[int, int]] | None = None
    for _ in range(800):
        removed = set(rng.sample(sorted(signal), noise_count)) if noise_count else set()
        candidates = sorted(all_cells - signal)
        added = set(rng.sample(candidates, noise_count)) if noise_count else set()
        trial = (signal - removed) | added
        predicted, score_gap, _ = _classification(trial)
        if predicted == target_name and score_gap >= 2.0:
            common = trial
            break
    if common is None:
        raise RuntimeError("could not sample a sufficiently separated reveal")

    # Both raw panels contain exactly 30 dots.  Their 19 distractors are disjoint
    # after folding, so the post-fold intersection is exactly `common`.
    pool = sorted(all_cells - common)
    rng.shuffle(pool)
    extra_count = 19
    left_fold_cells = set(common) | set(pool[:extra_count])
    right_fold_cells = set(common) | set(pool[extra_count : 2 * extra_count])
    right_actual = {(GRID_SIZE - 1 - col, row) for col, row in right_fold_cells}

    left_list = [list(cell) for cell in left_fold_cells]
    right_list = [list(cell) for cell in right_actual]
    rng.shuffle(left_list)
    rng.shuffle(right_list)
    return {
        "left_cells": left_list,
        "right_cells": right_list,
        "cell_size": cell_size,
        "top": top,
        "crease_x": crease_x,
        "dot_radius": dot_radius,
        "background_variant": background_variant,
        "palette_index": palette_index,
    }


def _layout(scene: dict) -> tuple[int, int, int, int, int]:
    cell = int(scene["cell_size"])
    top = int(scene["top"])
    crease = int(scene["crease_x"])
    left_x = crease - GRID_SIZE * cell
    right_x = crease
    return cell, top, crease, left_x, right_x


def _center(origin_x: int, top: int, cell: int, col: int, row: int) -> tuple[int, int]:
    return (origin_x + col * cell + cell // 2, top + row * cell + cell // 2)


def render(scene: dict) -> Image.Image:
    """Render two visible grids, a dashed hinge, a fold arrow, and their dots."""
    cell, top, crease, left_x, right_x = _layout(scene)
    bottom = top + GRID_SIZE * cell
    background = BACKGROUNDS[int(scene["background_variant"]) % len(BACKGROUNDS)]
    left_color, right_color = PALETTES[int(scene["palette_index"]) % len(PALETTES)]
    radius = int(scene["dot_radius"])
    image = Image.new("RGB", (WIDTH, HEIGHT), background)
    draw = ImageDraw.Draw(image)

    # Visible cell boundaries are the fold fiducials; both panels share the hinge.
    for k in range(GRID_SIZE + 1):
        y = top + k * cell
        draw.line((left_x, y, crease + GRID_SIZE * cell, y), fill=GRID_COLOR, width=2)
        xl = left_x + k * cell
        xr = right_x + k * cell
        draw.line((xl, top, xl, bottom), fill=GRID_COLOR, width=2)
        draw.line((xr, top, xr, bottom), fill=GRID_COLOR, width=2)

    # A visible curved leftward arrow declares which half moves.
    arrow_y = top - 38
    draw.arc((crease - 8, arrow_y - 20, crease + 76, arrow_y + 26), 202, 337, fill=GOLD, width=5)
    draw.polygon(
        [(crease - 5, arrow_y + 1), (crease + 11, arrow_y - 8), (crease + 10, arrow_y + 10)],
        fill=GOLD,
    )
    for y in range(top - 10, bottom + 11, 15):
        draw.line((crease, y, crease, min(y + 8, bottom + 10)), fill=GOLD, width=5)

    for col, row in _cell_set(scene["left_cells"]):
        x, y = _center(left_x, top, cell, col, row)
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=left_color)
    for col, row in _cell_set(scene["right_cells"]):
        x, y = _center(right_x, top, cell, col, row)
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=right_color)
    return image


def analytic_gold(scene: dict) -> str:
    """Derive the answer only from the two rendered occupancy fields."""
    decision, _, _ = _classification(_folded_overlap(scene))
    return decision


def margin(scene: dict) -> float:
    """Top-versus-runner-up template distance gap, measured in grid cells."""
    _, score_gap, _ = _classification(_folded_overlap(scene))
    return score_gap


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """List order has no visual or semantic meaning for these nonoverlapping dots."""
    twin = deepcopy(scene)
    twin["left_cells"] = list(reversed(twin["left_cells"]))
    twin["right_cells"] = list(reversed(twin["right_cells"]))
    return [("reverse_unordered_dot_lists", twin)]
