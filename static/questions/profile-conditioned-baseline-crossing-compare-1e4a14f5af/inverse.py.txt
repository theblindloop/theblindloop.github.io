"""Independent inverse for paired baseline-crossing images.

This module intentionally contains its own raster constants and geometry.  It
accepts only a PIL image, segments the blue trace, estimates each baseline from
the dark horizontal run, samples nine visible marker centres, and compares the
resulting side-transition counts.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_CHECKPOINTS = 9
_PANEL_RANGES = ((40, 380), (380, 720))
_Y_RANGE = (80, 335)


def _trace_mask(image: Image.Image) -> np.ndarray:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = pixels[..., 0], pixels[..., 1], pixels[..., 2]
    return (red < 105) & (green > 60) & (blue > 125) & (blue > green + 35)


def _baseline_y(image: Image.Image, x_left: int, x_right: int) -> float:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    crop = pixels[_Y_RANGE[0] : _Y_RANGE[1], x_left:x_right]
    dark = np.max(crop, axis=2) < 130
    row_counts = dark.sum(axis=1).astype(float)
    if float(row_counts.max(initial=0.0)) < 100.0:
        raise ValueError("no sufficiently long visible baseline")
    peak = row_counts.max()
    rows = np.flatnonzero(row_counts >= peak - 1.0)
    weights = row_counts[rows]
    return float(_Y_RANGE[0] + np.average(rows, weights=weights))


def _checkpoint_heights(mask: np.ndarray, x_left: int, x_right: int) -> list[float]:
    coordinates = np.argwhere(mask[:, x_left:x_right])
    if len(coordinates) < 120:
        raise ValueError("trace color is not measurable")
    global_x = coordinates[:, 1] + x_left
    x_min = int(global_x.min()) + 6
    x_max = int(global_x.max()) - 6
    if x_max <= x_min:
        raise ValueError("trace has no usable horizontal extent")
    centres = np.linspace(x_min, x_max, _CHECKPOINTS)
    heights = []
    for centre in centres:
        lo = max(x_left, int(round(centre)) - 5)
        hi = min(x_right, int(round(centre)) + 6)
        ys = np.argwhere(mask[:, lo:hi])[:, 0]
        ys = ys[(ys >= _Y_RANGE[0]) & (ys < _Y_RANGE[1])]
        if len(ys) < 4:
            raise ValueError("a checkpoint marker is not visible")
        heights.append(float(np.median(ys)))
    return heights


def _count_side_changes(heights: list[float], baseline: float) -> int:
    signs = [1 if value < baseline else -1 if value > baseline else 0 for value in heights]
    if any(sign == 0 for sign in signs):
        raise ValueError("checkpoint is indistinguishable from baseline")
    return sum(left * right < 0 for left, right in zip(signs, signs[1:]))


def decision_from_image(image: Image.Image) -> str:
    """Return A or B using only pixels in the supplied raster."""

    if image.width < 500 or image.height < 300:
        raise ValueError("unexpected raster dimensions")
    mask = _trace_mask(image)
    counts = []
    for x_left, x_right in _PANEL_RANGES:
        baseline = _baseline_y(image, x_left, x_right)
        heights = _checkpoint_heights(mask, x_left, x_right)
        counts.append(_count_side_changes(heights, baseline))
    return "A" if counts[0] >= counts[1] else "B"
