"""Deterministic paired trace renderer for the baseline-crossing question."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH = 760
HEIGHT = 420
BASELINE_Y = 238.0
PANEL_TOP = 64
PANEL_BOTTOM = 348
PANEL_X0 = (54, 410)
PANEL_X1 = (350, 706)
CHECKPOINT_X0 = (86, 442)
CHECKPOINT_X1 = (318, 674)
CHECKPOINTS = 9

PANEL_FILL = (250, 252, 255)
PANEL_EDGE = (214, 221, 229)
BASELINE_COLOR = (70, 70, 70)
TRACE_COLOR = (31, 114, 186)
TEXT_COLOR = (24, 35, 48)


def _crossing_count(values: Iterable[float]) -> int:
    """Count strict sign changes in the ordered checkpoint offsets."""

    signs = [1 if float(value) > 0 else -1 if float(value) < 0 else 0 for value in values]
    return sum(left * right < 0 for left, right in zip(signs, signs[1:]))


def _sign_pattern(rng: random.Random, crossings: int) -> list[int]:
    positions = set(rng.sample(range(1, CHECKPOINTS), crossings))
    sign = rng.choice((-1, 1))
    result = []
    for index in range(CHECKPOINTS):
        if index in positions:
            sign *= -1
        result.append(sign)
    return result


def _trace_values(rng: random.Random, crossings: int) -> list[float]:
    signs = _sign_pattern(rng, crossings)
    # Every marker has a generous, randomized clearance from the baseline.
    return [
        round(sign * (38.0 + 40.0 * rng.random() + 3.0 * (rng.random() - 0.5)), 3)
        for sign in signs
    ]


def sample_scene(seed: int) -> dict:
    """Sample a balanced, non-boundary scene from one integer seed."""

    rng = random.Random(int(seed))
    a_wins = int(seed) % 2 == 0
    pairs_a_wins = ((6, 2), (5, 1), (7, 3), (4, 2))
    pairs_b_wins = tuple((right, left) for left, right in pairs_a_wins)
    crossings_a, crossings_b = rng.choice(pairs_a_wins if a_wins else pairs_b_wins)
    trace_a = _trace_values(rng, crossings_a)
    trace_b = _trace_values(rng, crossings_b)
    return {"trace_a": trace_a, "trace_b": trace_b}


def _checkpoint_points(trace: list[float], panel: int) -> list[tuple[float, float]]:
    x0 = CHECKPOINT_X0[panel]
    x1 = CHECKPOINT_X1[panel]
    step = (x1 - x0) / (CHECKPOINTS - 1)
    return [(x0 + step * index, BASELINE_Y - float(value)) for index, value in enumerate(trace)]


def render(scene: dict) -> Image.Image:
    """Render only the visible fields of a scene into an RGB PNG image."""

    image = Image.new("RGB", (WIDTH, HEIGHT), "white")
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    traces = (scene["trace_a"], scene["trace_b"])
    labels = ("A", "B")

    for panel in range(2):
        draw.rectangle(
            (PANEL_X0[panel], PANEL_TOP, PANEL_X1[panel], PANEL_BOTTOM),
            fill=PANEL_FILL,
            outline=PANEL_EDGE,
            width=1,
        )
        label_x = PANEL_X0[panel] + 7
        draw.text((label_x, 42), labels[panel], fill=TEXT_COLOR, font=font)
        draw.line(
            (PANEL_X0[panel] + 16, BASELINE_Y, PANEL_X1[panel] - 16, BASELINE_Y),
            fill=BASELINE_COLOR,
            width=3,
        )
        points = _checkpoint_points(traces[panel], panel)
        draw.line(points, fill=TRACE_COLOR, width=5, joint="curve")
        for x, y in points:
            radius = 7
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=TRACE_COLOR)

    return image


def analytic_gold(scene: dict) -> str:
    """Select the panel with the greater strict checkpoint-side-change count."""

    count_a = _crossing_count(scene["trace_a"])
    count_b = _crossing_count(scene["trace_b"])
    return "A" if count_a >= count_b else "B"


def margin(scene: dict) -> float:
    """Return the top-two decision gap in crossing counts."""

    return float(abs(_crossing_count(scene["trace_a"]) - _crossing_count(scene["trace_b"])))


def is_quarantined(scene: dict) -> bool:
    """Reject baseline contacts and count gaps too small for a stable raster answer."""

    values = [abs(float(value)) for value in scene["trace_a"] + scene["trace_b"]]
    return min(values, default=0.0) < 24.0 or margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """There are no unused latent aliases: both sequences are label inputs."""

    return []
