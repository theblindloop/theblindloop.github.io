"""Renderer and analytic arm for an arrow-directed node-link network."""

from __future__ import annotations

import math
import random
from collections import defaultdict, deque

from PIL import Image, ImageDraw

PALETTE = {
    "background": (248, 249, 251),
    "dark": (40, 48, 60),
    "gray": (155, 163, 175),
    "green": (45, 168, 82),
    "cyan": (15, 180, 205),
    "red": (220, 65, 70),
    "orange": (240, 145, 35),
    "purple": (145, 75, 190),
    "white": (255, 255, 255),
}
TERMINALS = ("t0", "t1", "t2")
TERMINAL_PARENTS = ("c", "d", "e")
QUESTION = (
    "Starting at the green node, follow dark links only in the direction of their "
    "cyan arrowheads. Which colored square can you reach: red, orange, or purple?"
)


def _reachable(scene: dict) -> set[str]:
    nodes = scene["nodes"]
    edges = scene["edges"]
    directions = scene["directions"]
    terminal_colors = scene["terminal_colors"]
    # Touch every declared label input. Node/color content determines role binding;
    # edges and directions determine the directed connectivity computation.
    if "s" not in nodes or set(terminal_colors) != set(TERMINALS):
        return set()
    allowed = {tuple(edge) for edge in edges}
    graph: dict[str, list[str]] = defaultdict(list)
    for source, target in directions:
        if tuple(sorted((source, target))) in allowed:
            graph[source].append(target)
    seen = {"s"}
    queue = deque(["s"])
    while queue:
        for nxt in graph[queue.popleft()]:
            if nxt not in seen:
                seen.add(nxt)
                queue.append(nxt)
    return seen


def analytic_gold(scene: dict) -> str:
    nodes = scene["nodes"]
    edges = scene["edges"]
    directions = scene["directions"]
    terminal_colors = scene["terminal_colors"]
    local = {
        "nodes": nodes,
        "edges": edges,
        "directions": directions,
        "terminal_colors": terminal_colors,
    }
    reached = _reachable(local)
    answers = [terminal_colors[name] for name in TERMINALS if name in reached]
    if len(answers) != 1:
        raise ValueError(f"scene has {len(answers)} reachable terminals")
    return answers[0]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target_color = ("red", "orange", "purple")[int(seed) % 3]
    target_terminal = rng.choice(TERMINALS)
    remaining_colors = [c for c in ("red", "orange", "purple") if c != target_color]
    rng.shuffle(remaining_colors)
    terminal_colors = {target_terminal: target_color}
    for name, color in zip((t for t in TERMINALS if t != target_terminal), remaining_colors):
        terminal_colors[name] = color

    base = {
        "s": (72, 220),
        "a": (190, 112),
        "b": (190, 328),
        "c": (340, 76),
        "d": (340, 220),
        "e": (340, 364),
        "t0": (566, 76),
        "t1": (566, 220),
        "t2": (566, 364),
    }
    nodes: dict[str, list[int]] = {}
    for name, (x, y) in base.items():
        # Small visible layout variation prevents coordinate memorization while
        # retaining broad clearance for link and arrow segmentation.
        jx = rng.randint(-7, 7) if name not in {"s", "t0", "t1", "t2"} else rng.randint(-3, 3)
        jy = rng.randint(-7, 7) if name not in {"s", "t0", "t1", "t2"} else rng.randint(-3, 3)
        nodes[name] = [x + jx, y + jy]

    edges = [
        ["a", "s"], ["b", "s"],
        ["a", "c"], ["a", "d"], ["b", "d"], ["b", "e"],
        ["c", "d"], ["d", "e"],
        ["c", "t0"], ["d", "t1"], ["e", "t2"],
    ]
    edges = [sorted(edge) for edge in edges]
    fixed = [("s", "a"), ("s", "b"), ("c", "t0"), ("d", "t1"), ("e", "t2")]
    internal = [("a", "c"), ("a", "d"), ("b", "d"), ("b", "e"), ("c", "d"), ("d", "e")]

    directions: list[list[str]] | None = None
    for _ in range(4096):
        trial = list(fixed)
        for left, right in internal:
            trial.append((left, right) if rng.getrandbits(1) else (right, left))
        candidate = {
            "nodes": nodes,
            "edges": edges,
            "directions": [list(item) for item in trial],
            "terminal_colors": terminal_colors,
        }
        reached = _reachable(candidate)
        reached_terminals = [name for name in TERMINALS if name in reached]
        if reached_terminals == [target_terminal]:
            directions = [list(item) for item in trial]
            break
    if directions is None:
        raise RuntimeError("could not sample a uniquely reachable terminal")

    return {
        "nodes": nodes,
        "edges": edges,
        "directions": directions,
        "terminal_colors": terminal_colors,
        "style": {
            "width": 640,
            "height": 440,
            "line_width": 8,
            "node_radius": 13,
            "terminal_half": 15,
            "arrow_length": 27,
            "arrow_half_width": 12,
            "supersample": 3,
        },
    }


def _triangle(mid: tuple[float, float], unit: tuple[float, float], length: float, half_width: float):
    ux, uy = unit
    px, py = -uy, ux
    tip = (mid[0] + ux * length * 0.58, mid[1] + uy * length * 0.58)
    base_x = mid[0] - ux * length * 0.42
    base_y = mid[1] - uy * length * 0.42
    return [tip, (base_x + px * half_width, base_y + py * half_width), (base_x - px * half_width, base_y - py * half_width)]


def render(scene: dict) -> Image.Image:
    nodes = scene["nodes"]
    edges = scene["edges"]
    directions = scene["directions"]
    terminal_colors = scene["terminal_colors"]
    style = scene["style"]
    ss = int(style["supersample"])
    width, height = int(style["width"]), int(style["height"])
    image = Image.new("RGB", (width * ss, height * ss), PALETTE["background"])
    draw = ImageDraw.Draw(image)

    def p(point):
        return tuple(int(round(value * ss)) for value in point)

    for left, right in edges:
        draw.line([p(nodes[left]), p(nodes[right])], fill=PALETTE["dark"], width=int(style["line_width"] * ss))

    for source, target in directions:
        x0, y0 = nodes[source]
        x1, y1 = nodes[target]
        dx, dy = x1 - x0, y1 - y0
        norm = math.hypot(dx, dy)
        unit = (dx / norm, dy / norm)
        mid = ((x0 + x1) / 2, (y0 + y1) / 2)
        tri = _triangle(mid, unit, float(style["arrow_length"]), float(style["arrow_half_width"]))
        draw.polygon([p(point) for point in tri], fill=PALETTE["cyan"])

    radius = int(style["node_radius"] * ss)
    for name in ("a", "b", "c", "d", "e"):
        x, y = p(nodes[name])
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=PALETTE["gray"], outline=PALETTE["white"], width=2 * ss)
    x, y = p(nodes["s"])
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=PALETTE["green"], outline=PALETTE["white"], width=2 * ss)
    half = int(style["terminal_half"] * ss)
    for name in TERMINALS:
        x, y = p(nodes[name])
        draw.rectangle((x - half, y - half, x + half, y + half), fill=PALETTE[terminal_colors[name]], outline=PALETTE["white"], width=2 * ss)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def margin(scene: dict) -> float:
    style = scene["style"]
    # Runner-up margin: minimum of link half-width, arrow half-width, and
    # arrow-to-node clearance after accounting for marker radius.
    nodes = scene["nodes"]
    shortest = min(math.dist(nodes[a], nodes[b]) for a, b in scene["edges"])
    return float(min(style["line_width"] / 2, style["arrow_half_width"], shortest / 2 - style["node_radius"] - style["arrow_length"]))


def is_quarantined(scene: dict) -> bool:
    try:
        answer = analytic_gold(scene)
    except (KeyError, TypeError, ValueError):
        return True
    return answer not in {"red", "orange", "purple"} or margin(scene) < 3.5


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
