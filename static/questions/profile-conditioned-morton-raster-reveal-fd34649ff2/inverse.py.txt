"""Independent image-only inverse for the Morton raster reveal.

This module intentionally duplicates the small amount of geometry needed for
the inverse instead of importing renderer or any other candidate-world code.
It sees only the final RGB image and the public transform contract embodied by
the fixed templates below.
"""

from __future__ import annotations

import math
from functools import lru_cache

import numpy as np
from PIL import Image, ImageDraw

ORACLE_VERSION = "morton-raster-reveal-oracle-0.1.0"
SYMBOLS = ("beacon", "cusp", "ladder", "wave")
QUARANTINE_MARGIN = 0.08
_MIN_COMPONENT = 24


def _glyph_mask(size: int, symbol: str) -> np.ndarray:
    mask = Image.new("L", (size, size), 0)
    draw = ImageDraw.Draw(mask)
    width = max(5, int(round(size * 0.055)))
    left = int(round(size * 0.23))
    right = int(round(size * 0.77))
    top = int(round(size * 0.22))
    bottom = int(round(size * 0.78))
    mid = int(round(size * 0.50))
    if symbol == "beacon":
        radius = int(round(size * 0.22))
        draw.ellipse((mid - radius, mid - radius - 8, mid + radius, mid + radius - 8), outline=255, width=width)
        draw.line((mid, int(round(size * 0.27)), mid, int(round(size * 0.75))), fill=255, width=width)
        draw.line((int(round(size * 0.39)), int(round(size * 0.75)), int(round(size * 0.61)), int(round(size * 0.75))), fill=255, width=width)
    elif symbol == "cusp":
        draw.line((left, top + 5, mid, bottom, right, top + 5), fill=255, width=width, joint="curve")
        draw.line((int(round(size * 0.35)), int(round(size * 0.57)), mid, int(round(size * 0.43))), fill=255, width=width)
    elif symbol == "ladder":
        draw.line((left, top, left, bottom), fill=255, width=width)
        draw.line((right, top, right, bottom), fill=255, width=width)
        for fraction in (0.31, 0.45, 0.59, 0.73):
            y = int(round(size * fraction))
            draw.line((left, y, right, y), fill=255, width=width)
    elif symbol == "wave":
        points = []
        for i in range(61):
            x = left + (right - left) * i / 60.0
            y = mid + size * 0.15 * math.sin(2.0 * math.pi * (i / 60.0) * 1.55)
            points.append((int(round(x)), int(round(y))))
        draw.line(points, fill=255, width=width, joint="curve")
        draw.line((left, int(round(size * 0.66)), right, int(round(size * 0.66))), fill=255, width=max(3, width // 2))
    else:
        raise ValueError(f"unknown symbol {symbol!r}")
    return np.asarray(mask, dtype=np.float32) / 255.0


@lru_cache(maxsize=None)
def _morton_order(size: int) -> tuple[tuple[int, int], ...]:
    if size < 1 or size & (size - 1):
        raise ValueError("the marked raster side is not a power of two")
    order: list[tuple[int, int]] = []

    def visit(row: int, col: int, side: int) -> None:
        if side == 1:
            order.append((row, col))
            return
        half = side // 2
        visit(row, col, half)
        visit(row, col + half, half)
        visit(row + half, col, half)
        visit(row + half, col + half, half)

    visit(0, 0, size)
    return tuple(order)


def _components(chromatic: np.ndarray) -> list[dict[str, float]]:
    """Find four connected chromatic marker components without scene knowledge."""

    h, w = chromatic.shape
    visited = np.zeros_like(chromatic, dtype=bool)
    components: list[dict[str, float]] = []
    for row, col in zip(*np.nonzero(chromatic)):
        if visited[row, col]:
            continue
        stack = [(int(row), int(col))]
        visited[row, col] = True
        pixels: list[tuple[int, int]] = []
        while stack:
            r, c = stack.pop()
            pixels.append((r, c))
            for nr, nc in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
                if 0 <= nr < h and 0 <= nc < w and chromatic[nr, nc] and not visited[nr, nc]:
                    visited[nr, nc] = True
                    stack.append((nr, nc))
        if len(pixels) < _MIN_COMPONENT:
            continue
        rows = [p[0] for p in pixels]
        cols = [p[1] for p in pixels]
        components.append(
            {
                "x0": float(min(cols)),
                "x1": float(max(cols)),
                "y0": float(min(rows)),
                "y1": float(max(rows)),
                "cx": float(sum(cols) / len(cols)),
                "cy": float(sum(rows) / len(rows)),
                "area": float(len(pixels)),
            }
        )
    return components


def _find_tile(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    chroma = (arr.max(axis=2).astype(np.int16) - arr.min(axis=2).astype(np.int16)) > 35
    components = _components(chroma)
    if len(components) != 4:
        raise ValueError(f"expected four visible chromatic fiducials, found {len(components)}")
    top = sorted(components, key=lambda item: item["cy"])[:2]
    bottom = sorted(components, key=lambda item: item["cy"])[2:]
    if abs(top[0]["cy"] - top[1]["cy"]) > 4.0 or abs(bottom[0]["cy"] - bottom[1]["cy"]) > 4.0:
        raise ValueError("fiducials do not form two horizontal rows")
    tl, tr = sorted(top, key=lambda item: item["cx"])
    bl, br = sorted(bottom, key=lambda item: item["cx"])
    if not (tl["cx"] < tr["cx"] and bl["cx"] < br["cx"]):
        raise ValueError("fiducial columns are ambiguous")
    x0 = int(round(tl["x1"] + 1))
    x1 = int(round(tr["x0"] - 1))
    y0 = int(round(tl["y1"] + 1))
    y1 = int(round(bl["y0"] - 1))
    size_x = x1 - x0 + 1
    size_y = y1 - y0 + 1
    if size_x != size_y or size_x not in (128, 256):
        raise ValueError(f"fiducials imply invalid square raster {size_x}x{size_y}")
    if x0 < 0 or y0 < 0 or x1 >= arr.shape[1] or y1 >= arr.shape[0]:
        raise ValueError("inferred raster lies outside image")
    # The encoded tile is grayscale; use its first channel only after the
    # chromatic registration has isolated the exact crop.
    return arr[y0 : y1 + 1, x0 : x1 + 1, 0].astype(np.float32)


def _restore(tile: np.ndarray) -> np.ndarray:
    size = tile.shape[0]
    flat = tile.reshape(-1)
    restored = np.empty_like(tile, dtype=np.float32)
    for index, (row, col) in enumerate(_morton_order(size)):
        restored[row, col] = flat[index]
    return restored


def _box5(values: np.ndarray) -> np.ndarray:
    padded = np.pad(values, 2, mode="edge")
    result = np.zeros_like(values, dtype=np.float32)
    for dy in range(5):
        for dx in range(5):
            result += padded[dy : dy + values.shape[0], dx : dx + values.shape[1]]
    return result / 25.0


def _scores(tile: np.ndarray) -> list[float]:
    restored = _restore(tile)
    filtered = _box5(restored)
    observed_dark = 236.0 - filtered
    scores: list[float] = []
    for symbol in SYMBOLS:
        target = _box5(190.0 * _glyph_mask(tile.shape[0], symbol))
        error = float(np.mean(np.abs(observed_dark - target)))
        scores.append(1.0 - error / 190.0)
    return scores


def confidence_margin(image: Image.Image) -> float:
    scores = sorted(_scores(_find_tile(image)), reverse=True)
    return float(scores[0] - scores[1])


def decision_from_image(image: Image.Image) -> str:
    """Return the glyph selected from the final raster, or abstain if ambiguous."""

    scores = sorted(enumerate(_scores(_find_tile(image))), key=lambda item: item[1], reverse=True)
    margin = float(scores[0][1] - scores[1][1])
    if margin < QUARANTINE_MARGIN:
        raise ValueError(f"template decision margin {margin:.4f} is below quarantine")
    return SYMBOLS[scores[0][0]]
