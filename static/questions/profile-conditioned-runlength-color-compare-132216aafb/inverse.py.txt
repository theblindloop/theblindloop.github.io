"""Independent pixel-only inverse arm for the runlength_color_compare world.

This module receives ONLY the rasterized PNG and re-measures which color covers
the greater total horizontal length. Because the bar height is constant and
shared by both colors, each color's total covered width is proportional to its
total on-screen pixel ink, so counting the pixels of each color and comparing
equivocally recovers the analytic comparison. It never imports the renderer,
prompts, generator, verifier, or any analytical gold code.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "runlength_color_compare-oracle-0.1.0"

COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
}


def _ink_count(arr: np.ndarray, rgb: tuple) -> int:
    """Number of pixels whose RGB exactly equals the target color."""
    return int(np.sum(np.all(arr == np.array(rgb), axis=-1)))


def decision_from_image(image: Image.Image) -> str:
    """Return the color covering the greater total horizontal length in the raster."""
    arr = np.asarray(image.convert("RGB"), dtype=np.int64)
    red = _ink_count(arr, COLOR_RGB["red"])
    blue = _ink_count(arr, COLOR_RGB["blue"])
    return "red" if red > blue else "blue"
