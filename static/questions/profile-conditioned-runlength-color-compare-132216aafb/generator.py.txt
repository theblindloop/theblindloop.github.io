"""Interleaved colored-track measurement world: which color covers more total length.

Latent scene z is a single field ``tiles``: an ordered left-to-right tiling of a
fixed horizontal bar by red and blue filled rectangles of arbitrary widths, with
no gaps and no overlaps. The image is I = R(z). The task is a pure measurement:
total the horizontal length covered by each color's tiles and decide which color
covers more.

Because the bar height is constant and shared, a color's total covered width is
proportional to its on-screen pixel area, so the independent inverse arm can
count each color's ink and compare. Color identity and the block arrangement are
fully visible, so there is no hidden scene field and no pixel-identical latent
transform.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "runlength_color_compare-0.1.0"

COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
}
BACKGROUND = (250, 250, 250)
BAR_OUTLINE = (80, 80, 80)

WIDTH = 640
HEIGHT = 420
BAR_LEFT = 60.0
BAR_RIGHT = 580.0
BAR_TOP = 160.0
BAR_BOTTOM = 260.0
BAR_W = BAR_RIGHT - BAR_LEFT
BAR_H = BAR_BOTTOM - BAR_TOP

N_TILES = 8
MIN_MARGIN_PX = 16.0
QUARANTINE_PX = 14.0


def _total_length(tiles: list[dict], color: str) -> float:
    """Sum of the widths of every tile of the given color."""
    return float(sum(t["width"] for t in tiles if t["color"] == color))


def sample_scene(seed: int) -> dict:
    """Sample a deterministic tiling whose per-color totals differ by a clear margin.

    Widths are drawn at random and normalized to tile the full bar; each tile's
    color is a fair coin flip. Rejection-samples until the two color totals are
    separated by at least MIN_MARGIN_PX, guaranteeing an off-quarantine answer.
    """
    rng = random.Random(seed * 1009 + 7)
    while True:
        weights = [rng.uniform(0.03, 0.35) for _ in range(N_TILES)]
        total = sum(weights)
        widths = [BAR_W * w / total for w in weights]
        colors = [rng.choice(("red", "blue")) for _ in range(N_TILES)]
        tiles = []
        left = BAR_LEFT
        for width, color in zip(widths, colors):
            tiles.append(
                {"color": color, "left": round(left, 3), "width": round(width, 3)}
            )
            left += width
        red = _total_length(tiles, "red")
        blue = _total_length(tiles, "blue")
        if abs(red - blue) >= MIN_MARGIN_PX:
            return {"tiles": tiles}


def render(scene: dict) -> Image.Image:
    """Render I = R(z): the tiled horizontal bar on a light panel."""
    tiles = scene["tiles"]
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(img)
    for tile in tiles:
        color = tile["color"]
        left = tile["left"]
        width = tile["width"]
        draw.rectangle(
            [left, BAR_TOP, left + width, BAR_BOTTOM], fill=COLOR_RGB[color]
        )
    draw.rectangle(
        [BAR_LEFT, BAR_TOP, BAR_RIGHT, BAR_BOTTOM], outline=BAR_OUTLINE, width=2
    )
    return img


def analytic_gold(scene: dict) -> str:
    """The color whose tiles sum to the greater total horizontal length."""
    tiles = scene["tiles"]
    red = _total_length(tiles, "red")
    blue = _total_length(tiles, "blue")
    return "red" if red > blue else "blue"


def margin(scene: dict) -> float:
    """Verifiability margin: the difference of the two per-color totals, in px."""
    tiles = scene["tiles"]
    return float(abs(_total_length(tiles, "red") - _total_length(tiles, "blue")))


def is_quarantined(scene: dict) -> bool:
    """Near-boundary: the per-color total difference is finite-resolution small."""
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list:
    """not_applicable: no nontrivial pixel-identical latent symmetry.

    Every latent field is carried inside the single fully visible ``tiles``
    field: each tile's color and width are drawn as visible ink, and flipping
    which color covers more changes the correct answer. Reversing the tile order
    or reflecting the bar turns the raster into a different image, so no latent
    transformation is both pixel-identical and gold-preserving.
    """
    return []
