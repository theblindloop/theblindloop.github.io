"""Deterministic box-plot sampler and Pillow renderer."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 440
COLOR_RGB = {
    "red": (208, 50, 62),
    "blue": (39, 105, 204),
    "green": (31, 146, 88),
    "purple": (139, 65, 172),
}
COLORS = tuple(COLOR_RGB)
X_SLOTS = (125, 255, 385, 515)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    answer = COLORS[int(seed) % len(COLORS)]
    others = [color for color in COLORS if color != answer]
    rng.shuffle(others)

    # The first distractor always owns the longest raw upper whisker, while the
    # answer owns the largest normalized upper-whisker / box-height ratio.
    roles = {
        answer: (rng.randint(52, 58), rng.uniform(1.58, 1.64)),
        others[0]: (rng.randint(92, 100), rng.uniform(1.12, 1.18)),
        others[1]: (rng.randint(70, 82), rng.uniform(0.76, 0.84)),
        others[2]: (rng.randint(48, 62), rng.uniform(0.48, 0.59)),
    }
    slots = list(X_SLOTS)
    rng.shuffle(slots)
    boxplots = []
    for color, x in zip(COLORS, slots):
        box_height, target_ratio = roles[color]
        upper_length = int(round(box_height * target_ratio))
        box_top = rng.randint(210, 229)
        lower_length = rng.randint(34, 63)
        boxplots.append(
            {
                "color": color,
                "x": x,
                "box_top": box_top,
                "box_bottom": box_top + box_height,
                "upper_cap": box_top - upper_length,
                "lower_cap": box_top + box_height + lower_length,
                "median_y": box_top + rng.randint(15, box_height - 15),
            }
        )
    return {"boxplots": boxplots}


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (247, 248, 246))
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle(
        (58, 42, 600, 390), radius=8, fill=(255, 255, 255), outline=(94, 100, 106), width=2
    )
    for y in (90, 150, 210, 270, 330):
        draw.line((78, y, 580, y), fill=(226, 229, 232), width=2)
    draw.line((78, 366, 580, 366), fill=(85, 91, 97), width=3)
    for x in X_SLOTS:
        draw.line((x, 366, x, 373), fill=(85, 91, 97), width=3)

    for item in scene["boxplots"]:
        color = COLOR_RGB[item["color"]]
        x = int(item["x"])
        top = int(item["box_top"])
        bottom = int(item["box_bottom"])
        upper = int(item["upper_cap"])
        lower = int(item["lower_cap"])
        median = int(item["median_y"])
        draw.line((x, upper, x, top), fill=color, width=5)
        draw.line((x - 12, upper, x + 12, upper), fill=color, width=5)
        draw.line((x, bottom, x, lower), fill=color, width=5)
        draw.line((x - 12, lower, x + 12, lower), fill=color, width=5)
        draw.rectangle((x - 24, top, x + 24, bottom), fill=color)
        draw.line((x - 7, median, x + 7, median), fill=(255, 255, 255), width=3)
    return image


def _ratios(scene: dict) -> dict[str, float]:
    return {
        item["color"]: (float(item["box_top"]) - float(item["upper_cap"]))
        / (float(item["box_bottom"]) - float(item["box_top"]))
        for item in scene["boxplots"]
    }


def analytic_gold(scene: dict) -> str:
    ratios = _ratios(scene)
    return max(ratios, key=ratios.get)


def margin(scene: dict) -> float:
    ordered = sorted(_ratios(scene).values(), reverse=True)
    return 100.0 * (ordered[0] - ordered[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 18.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def shifted(scene: dict, dy: int) -> dict:
    """Public helper used only by tests to exercise label-preserving translation."""
    moved = copy.deepcopy(scene)
    for item in moved["boxplots"]:
        for field in ("box_top", "box_bottom", "upper_cap", "lower_cap", "median_y"):
            item[field] += int(dy)
    return moved
