"""Independent inverse arm: accepts only a final PNG."""

from __future__ import annotations

import numpy as np
from PIL import Image

COLOR_RGB = {
    "red": (208, 50, 62),
    "blue": (39, 105, 204),
    "green": (31, 146, 88),
    "purple": (139, 65, 172),
}


def _near(array: np.ndarray, rgb: tuple[int, int, int], radius: float = 15.0) -> np.ndarray:
    delta = array.astype(np.float32) - np.asarray(rgb, dtype=np.float32)
    return np.sum(delta * delta, axis=2) <= radius * radius


def _measure_ratio(mask: np.ndarray, name: str) -> float:
    ys, xs = np.nonzero(mask)
    if len(xs) < 1_500:
        raise ValueError(f"{name} box plot is absent or too damaged")
    row_counts = np.bincount(ys, minlength=mask.shape[0])
    # Filled box rows are at least 34 px wide even where the white median mark
    # interrupts them; 25-px caps and 5-px whiskers remain below this threshold.
    box_rows = np.flatnonzero(row_counts >= 30)
    if len(box_rows) < 35:
        raise ValueError(f"{name} box body is malformed")
    top, bottom = int(box_rows.min()), int(box_rows.max())
    if bottom - top < 35 or bottom - top > 110:
        raise ValueError(f"{name} box height is invalid")
    above = np.flatnonzero(row_counts[:top] > 0)
    if len(above) < 20:
        raise ValueError(f"{name} upper whisker is missing")
    cap_rows = above[row_counts[above] >= 20]
    if len(cap_rows) < 3:
        raise ValueError(f"{name} upper cap is missing")
    cap = int(cap_rows.min())
    if top - cap < 20:
        raise ValueError(f"{name} upper whisker is too short")

    # Require a narrow, nearly vertical bridge between cap and box. This makes
    # evidence erasure or unrelated same-color blobs cause abstention.
    bridge_counts = row_counts[cap + 3 : top]
    if len(bridge_counts) < 10 or np.mean((bridge_counts >= 3) & (bridge_counts <= 8)) < 0.9:
        raise ValueError(f"{name} upper whisker bridge is malformed")
    width = float(bottom - top + 1)
    return float(top - cap) / width


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    if array.shape[0] < 300 or array.shape[1] < 500:
        raise ValueError("raster is too small")
    ratios = {name: _measure_ratio(_near(array, rgb), name) for name, rgb in COLOR_RGB.items()}
    ordered = sorted(ratios, key=ratios.get, reverse=True)
    if ratios[ordered[0]] - ratios[ordered[1]] < 0.18:
        raise ValueError("top ratios are pixel-ambiguous")
    return ordered[0]
