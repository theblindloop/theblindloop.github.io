"""Independent image-only inverse for the signed-prefix route world."""

from __future__ import annotations

from collections import deque

import numpy as np

NAVY = np.array((31, 48, 71), dtype=np.int16)
COLORS = {
    "cyan": np.array((20, 183, 205), dtype=np.int16),
    "orange": np.array((238, 132, 35), dtype=np.int16),
    "green": np.array((49, 166, 82), dtype=np.int16),
    "red": np.array((211, 61, 64), dtype=np.int16),
}


def _mask_near(array: np.ndarray, color: np.ndarray, tolerance: int = 12) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _components(mask: np.ndarray, minimum: int = 25) -> list[tuple[float, float]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    centers: list[tuple[float, float]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        pixels: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            pixels.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(pixels) >= minimum:
            centers.append((sum(x for _, x in pixels) / len(pixels), sum(y for y, _ in pixels) / len(pixels)))
    return centers


def _dilate(mask: np.ndarray, rounds: int) -> np.ndarray:
    expanded = mask.copy()
    for _ in range(rounds):
        prior = expanded
        expanded = prior.copy()
        expanded[1:, :] |= prior[:-1, :]
        expanded[:-1, :] |= prior[1:, :]
        expanded[:, 1:] |= prior[:, :-1]
        expanded[:, :-1] |= prior[:, 1:]
    return expanded


def _route_distances(route: np.ndarray, start: tuple[float, float]) -> np.ndarray:
    height, width = route.shape
    sx, sy = map(lambda value: int(round(value)), start)
    candidates = np.argwhere(route)
    if not len(candidates):
        raise ValueError("navy route is absent")
    near = candidates[(candidates[:, 0] - sy) ** 2 + (candidates[:, 1] - sx) ** 2 <= 22 ** 2]
    if not len(near):
        raise ValueError("green start is detached from route")
    origin = near[np.argmin((near[:, 0] - sy) ** 2 + (near[:, 1] - sx) ** 2)]
    distance = np.full((height, width), -1, dtype=np.int32)
    oy, ox = map(int, origin)
    distance[oy, ox] = 0
    queue = deque([(oy, ox)])
    while queue:
        y, x = queue.popleft()
        next_distance = int(distance[y, x]) + 1
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < height and 0 <= nx < width and route[ny, nx] and distance[ny, nx] < 0:
                distance[ny, nx] = next_distance
                queue.append((ny, nx))
    return distance


def _distance_at(distance: np.ndarray, center: tuple[float, float]) -> int:
    x, y = center
    yy, xx = np.ogrid[: distance.shape[0], : distance.shape[1]]
    neighborhood = (xx - x) ** 2 + (yy - y) ** 2 <= 18 ** 2
    values = distance[(distance >= 0) & neighborhood]
    if not len(values):
        raise ValueError("checkpoint is detached from recovered route")
    return int(values.min())


def decision_from_image(image) -> str:
    array = np.asarray(image.convert("RGB"))
    green = _components(_mask_near(array, COLORS["green"]), minimum=45)
    red = _components(_mask_near(array, COLORS["red"]), minimum=45)
    cyan = _components(_mask_near(array, COLORS["cyan"]), minimum=40)
    orange = _components(_mask_near(array, COLORS["orange"]), minimum=40)
    if len(green) != 1 or len(red) != 1 or len(cyan) != 4 or len(orange) != 4:
        raise ValueError("expected one start, one finish, and four rings of each checkpoint color")

    route = _dilate(_mask_near(array, NAVY), 6)
    distance = _route_distances(route, green[0])
    if _distance_at(distance, red[0]) < 100:
        raise ValueError("finish is not connected by a sufficiently long route")
    ordered = sorted(
        [(_distance_at(distance, center), "cyan") for center in cyan]
        + [(_distance_at(distance, center), "orange") for center in orange]
    )
    if any(b[0] - a[0] < 35 for a, b in zip(ordered, ordered[1:])):
        raise ValueError("checkpoint order lacks a stable pixel margin")
    total = 0
    for _, token in ordered:
        total += 1 if token == "cyan" else -1
        if total < 0:
            return "yes"
    return "no"
