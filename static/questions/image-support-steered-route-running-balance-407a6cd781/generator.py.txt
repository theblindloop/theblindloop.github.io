"""Deterministic Pillow world for a signed-prefix computation on one route."""

from __future__ import annotations

import itertools
import math
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 420
BACKGROUND = (250, 248, 242)
NAVY = (31, 48, 71)
CYAN = (20, 183, 205)
ORANGE = (238, 132, 35)
GREEN = (49, 166, 82)
RED = (211, 61, 64)


def _prefix_min(tokens: list[str]) -> int:
    total = 0
    low = 0
    for token in tokens:
        total += 1 if token == "cyan" else -1
        low = min(low, total)
    return low


def _sequence_pools() -> tuple[list[list[str]], list[list[str]]]:
    safe, violating = [], []
    for cyan_positions in itertools.combinations(range(8), 4):
        tokens = ["orange"] * 8
        for index in cyan_positions:
            tokens[index] = "cyan"
        # Hold the two endpoint-adjacent tokens constant across both answers.
        if tokens[0] != "cyan" or tokens[-1] != "orange":
            continue
        (safe if _prefix_min(tokens) == 0 else violating).append(tokens)
    return safe, violating


SAFE_SEQUENCES, VIOLATING_SEQUENCES = _sequence_pools()


def sample_scene(seed: int) -> dict:
    """Sample balanced classes with identical token counts and endpoint colors."""
    rng = random.Random(int(seed))
    pool = VIOLATING_SEQUENCES if seed % 2 == 0 else SAFE_SEQUENCES
    tokens = list(pool[rng.randrange(len(pool))])
    top = 88 + rng.randint(-5, 5)
    bottom = 332 + rng.randint(-5, 5)
    left = 52 + rng.randint(0, 12)
    right = 588 - rng.randint(0, 12)
    # An open U is long-range but deliberately non-space-filling at the public
    # ablation scale: two horizontal arms joined by one thin vertical arm.
    points = [[left, top], [right, top], [right, bottom], [left, bottom]]
    return {"points": points, "tokens": tokens}


def _point_at_fraction(points: list[list[int]], fraction: float) -> tuple[float, float]:
    lengths = [
        math.hypot(b[0] - a[0], b[1] - a[1])
        for a, b in zip(points, points[1:])
    ]
    target = fraction * sum(lengths)
    for (a, b), length in zip(zip(points, points[1:]), lengths):
        if target <= length:
            ratio = target / length if length else 0.0
            return (a[0] + ratio * (b[0] - a[0]), a[1] + ratio * (b[1] - a[1]))
        target -= length
    return tuple(map(float, points[-1]))


def _ring(draw: ImageDraw.ImageDraw, center: tuple[float, float], color: tuple[int, int, int], radius: int, width: int) -> None:
    x, y = center
    box = (round(x - radius), round(y - radius), round(x + radius), round(y + radius))
    # Keep the navy centerline visible through the ring interior. Only the
    # colored annulus crosses it, so the inverse can bridge a short visible gap.
    draw.ellipse(box, outline=color, width=width)


def render(scene: dict) -> Image.Image:
    points = scene["points"]
    tokens = scene["tokens"]
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((10, 10, WIDTH - 11, HEIGHT - 11), radius=12, outline=(218, 215, 206), width=2)
    draw.line([tuple(point) for point in points], fill=NAVY, width=7, joint="curve")

    for index, token in enumerate(tokens):
        center = _point_at_fraction(points, (index + 1) / 9.0)
        _ring(draw, center, CYAN if token == "cyan" else ORANGE, radius=10, width=5)

    _ring(draw, tuple(points[0]), GREEN, radius=13, width=6)
    _ring(draw, tuple(points[-1]), RED, radius=13, width=6)
    return image


def analytic_gold(scene: dict) -> str:
    tokens = scene["tokens"]
    return "yes" if _prefix_min(tokens) < 0 else "no"


def margin(scene: dict) -> float:
    tokens = scene["tokens"]
    # Integer distance to the nearest opposite-class prefix boundary.
    low = _prefix_min(tokens)
    return float(-low if low < 0 else 1)


def is_quarantined(scene: dict) -> bool:
    points = scene["points"]
    tokens = scene["tokens"]
    return len(points) != 4 or len(tokens) != 8 or tokens.count("cyan") != 4 or tokens.count("orange") != 4


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
