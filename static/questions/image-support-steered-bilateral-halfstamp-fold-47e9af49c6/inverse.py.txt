"""Independent image-only inverse for the bilateral half-stamp transform."""

from __future__ import annotations

import numpy as np
from PIL import Image

_GRID = 4
_NAMES = ("X", "ring", "H", "U")
_PATTERNS = {
    "X": ("#..#", ".##.", ".##.", "#..#"),
    "ring": ("####", "#..#", "#..#", "####"),
    "H": ("#..#", "####", "#..#", "#..#"),
    "U": ("#..#", "#..#", "#..#", ".##."),
}


def _mask(rows: tuple[str, ...]) -> np.ndarray:
    return np.asarray([[char == "#" for char in row] for row in rows], dtype=bool)


def _extract(array: np.ndarray, family: str) -> np.ndarray:
    rgb = array.astype(np.int16)
    r, g, b = rgb[..., 0], rgb[..., 1], rgb[..., 2]
    if family == "cyan":
        colored = (g > r + 38) & (b > r + 45) & (b >= g - 12)
    elif family == "coral":
        colored = (r > g + 38) & (r > b + 35)
    else:
        raise ValueError("unknown stamp family")
    points = np.argwhere(colored)
    if len(points) < 240:
        raise ValueError(f"missing or erased {family} stamp")
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    height, width = int(y1 - y0 + 1), int(x1 - x0 + 1)
    if width != height or width not in (16, 20) or width % _GRID:
        raise ValueError(f"ambiguous {family} stamp geometry")
    crop_mask = colored[y0 : y1 + 1, x0 : x1 + 1]
    if not bool(crop_mask.all()):
        raise ValueError(f"{family} stamp has missing raster evidence")
    crop = rgb[y0 : y1 + 1, x0 : x1 + 1]
    cell = width // _GRID
    luminance = crop @ np.asarray([0.2126, 0.7152, 0.0722])
    means = luminance.reshape(_GRID, cell, _GRID, cell).mean(axis=(1, 3))
    ordered = np.sort(means.ravel())
    if float(ordered[8] - ordered[7]) < 24.0:
        raise ValueError(f"{family} dark/light boundary is ambiguous")
    threshold = float((ordered[7] + ordered[8]) / 2.0)
    bits = means < threshold
    if int(bits.sum()) != 8:
        raise ValueError(f"{family} stamp does not have the declared equal ink")
    return bits


def decision_with_margin(image: Image.Image) -> tuple[str, float]:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 128 or array.shape[1] < 128:
        raise ValueError("image is too small")
    raw_a = _extract(array, "cyan")
    raw_b = _extract(array, "coral")
    revealed = np.fliplr(raw_a) ^ np.flipud(raw_b)
    distances = {
        name: int(np.count_nonzero(revealed ^ _mask(rows)))
        for name, rows in _PATTERNS.items()
    }
    ordered = sorted(distances.items(), key=lambda item: (item[1], item[0]))
    gap = float(ordered[1][1] - ordered[0][1])
    if ordered[0][1] > 1 or gap < 3.0:
        raise ValueError("revealed symbol is outside the finite-resolution margin")
    return ordered[0][0], gap


def decision_from_image(image: Image.Image) -> str:
    return decision_with_margin(image)[0]
