"""Render two tiny secret-shared stamps for a declared reflection/XOR reveal."""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw

CANVAS = 512
GRID = 4
SYMBOLS = ("X", "ring", "H", "U")
BACKGROUND = (248, 246, 240)
PALETTES = (
    ((19, 91, 108), (128, 215, 224), (132, 44, 62), (245, 169, 176)),
    ((22, 105, 122), (151, 226, 231), (145, 55, 72), (249, 184, 184)),
    ((25, 82, 101), (112, 201, 215), (119, 48, 70), (235, 156, 169)),
)


def _template(name: str) -> np.ndarray:
    rows = {
        "X": ("#..#", ".##.", ".##.", "#..#"),
        "ring": ("####", "#..#", "#..#", "####"),
        "H": ("#..#", "####", "#..#", "#..#"),
        "U": ("#..#", "#..#", "#..#", ".##."),
    }[name]
    return np.asarray([[char == "#" for char in row] for row in rows], dtype=bool)


def _encode(mask: np.ndarray) -> list[str]:
    return ["".join("1" if value else "0" for value in row) for row in mask]


def _decode(rows: list[str]) -> np.ndarray:
    if len(rows) != GRID or any(len(row) != GRID or set(row) - {"0", "1"} for row in rows):
        raise ValueError("stamp rows must encode a 4x4 binary raster")
    return np.asarray([[char == "1" for char in row] for row in rows], dtype=bool)


def _decision_and_margin(scene: dict) -> tuple[str, float]:
    raw_a = _decode(scene["stamp_a_rows"])
    raw_b = _decode(scene["stamp_b_rows"])
    revealed = np.fliplr(raw_a) ^ np.flipud(raw_b)
    distances = {name: int(np.count_nonzero(revealed ^ _template(name))) for name in SYMBOLS}
    ordered = sorted(distances.items(), key=lambda item: (item[1], item[0]))
    return ordered[0][0], float(ordered[1][1] - ordered[0][1])


def sample_scene(seed: int) -> dict:
    """Balance labels while making each displayed stamp an equal-ink random share."""
    seed = int(seed)
    rng = random.Random((seed ^ 0xB17A7E5) & 0xFFFFFFFF)
    symbol = SYMBOLS[seed % len(SYMBOLS)]
    target = _template(symbol)
    target_positions = list(zip(*np.nonzero(target)))
    blank_positions = list(zip(*np.nonzero(~target)))
    rng.shuffle(target_positions)
    rng.shuffle(blank_positions)
    # Canonical A has exactly eight dark cells and overlaps half of the target.
    canonical_a = np.zeros((GRID, GRID), dtype=bool)
    chosen = target_positions[: len(target_positions) // 2]
    chosen += blank_positions[: 8 - len(chosen)]
    for y, x in chosen:
        canonical_a[int(y), int(x)] = True
    canonical_b = canonical_a ^ target
    if int(canonical_a.sum()) != 8 or int(canonical_b.sum()) != 8:
        raise AssertionError("both displayed shares must have equal ink")
    raw_a = np.fliplr(canonical_a)
    raw_b = np.flipud(canonical_b)
    return {
        "stamp_a_rows": _encode(raw_a),
        "stamp_b_rows": _encode(raw_b),
        "palette_index": (seed // 8) % len(PALETTES),
        "cell_px": 4 + ((seed // 4) % 2),
        "layout_flip": bool((seed // 12) % 2),
        "offset_x": rng.randrange(-3, 4),
        "offset_y": rng.randrange(-3, 4),
        "decoration_variant": (seed // 16) % 3,
    }


def _paint_stamp(
    draw: ImageDraw.ImageDraw,
    rows: list[str],
    center: tuple[int, int],
    cell: int,
    dark: tuple[int, int, int],
    light: tuple[int, int, int],
) -> None:
    size = GRID * cell
    left = int(center[0] - size // 2)
    top = int(center[1] - size // 2)
    draw.rectangle((left - 2, top - 2, left + size + 1, top + size + 1), outline=(82, 83, 88), width=1)
    for gy, row in enumerate(rows):
        for gx, bit in enumerate(row):
            x0, y0 = left + gx * cell, top + gy * cell
            draw.rectangle((x0, y0, x0 + cell - 1, y0 + cell - 1), fill=dark if bit == "1" else light)


def render(scene: dict) -> Image.Image:
    rows_a = list(scene["stamp_a_rows"])
    rows_b = list(scene["stamp_b_rows"])
    _decode(rows_a)
    _decode(rows_b)
    palette = PALETTES[int(scene["palette_index"]) % len(PALETTES)]
    cell = int(scene["cell_px"])
    if cell not in (4, 5):
        raise ValueError("cell_px must be 4 or 5")
    flip = bool(scene["layout_flip"])
    ox, oy = int(scene["offset_x"]), int(scene["offset_y"])
    decoration = int(scene["decoration_variant"])
    if not (-3 <= ox <= 3 and -3 <= oy <= 3 and 0 <= decoration <= 2):
        raise ValueError("scene nuisance is outside the declared range")
    image = Image.new("RGB", (CANVAS, CANVAS), BACKGROUND)
    draw = ImageDraw.Draw(image)
    shade = (155 + 10 * decoration,) * 3
    # Neutral badge and fold guides are instructional context, never label evidence.
    draw.rounded_rectangle((174 + ox, 174 + oy, 338 + ox, 338 + oy), radius=20, outline=shade, width=2)
    for step in range(7):
        p = 244 + step * 4
        draw.line((p + ox, p + oy, p + 1 + ox, p + 1 + oy), fill=shade, width=1)
    centers = ((224 + ox, 224 + oy), (288 + ox, 288 + oy))
    if flip:
        centers = centers[::-1]
    _paint_stamp(draw, rows_a, centers[0], cell, palette[0], palette[1])
    _paint_stamp(draw, rows_b, centers[1], cell, palette[2], palette[3])
    return image


def analytic_gold(scene: dict) -> str:
    return _decision_and_margin(scene)[0]


def margin(scene: dict) -> float:
    return _decision_and_margin(scene)[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 3.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
