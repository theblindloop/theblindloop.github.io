"""Deterministic renderer and analytic arm for balanced_link_cut."""

from __future__ import annotations

import math
import random
from collections import deque
from copy import deepcopy

from PIL import Image, ImageDraw

CANVAS = 448
PALETTES = (
    {"background": [248, 246, 239], "edge": [43, 49, 59], "node": [35, 150, 175], "mark": [235, 120, 38]},
    {"background": [250, 247, 241], "edge": [50, 47, 58], "node": [64, 132, 190], "mark": [224, 112, 36]},
    {"background": [247, 245, 238], "edge": [44, 51, 58], "node": [55, 158, 115], "mark": [232, 126, 42]},
)


def _cycle(start: int, count: int) -> list[list[int]]:
    return [[start + i, start + ((i + 1) % count)] for i in range(count)]


def _polygon(cx: float, cy: float, radius: float, count: int, phase: float) -> list[list[int]]:
    return [
        [round(cx + radius * math.cos(phase + 2 * math.pi * i / count)),
         round(cy + radius * math.sin(phase + 2 * math.pi * i / count))]
        for i in range(count)
    ]


def _dihedral(point: list[int], code: int, dx: int, dy: int) -> list[int]:
    x, y = point[0] - CANVAS / 2, point[1] - CANVAS / 2
    if code >= 4:
        x = -x
    for _ in range(code % 4):
        x, y = -y, x
    return [round(x + CANVAS / 2 + dx), round(y + CANVAS / 2 + dy)]


def sample_scene(seed: int) -> dict:
    """Sample balanced classes with matched V/E counts and degree sequence."""
    rng = random.Random(int(seed))
    positive = (seed % 2 == 0)
    radius = rng.randint(47, 53)

    if positive or seed % 4 == 1:
        # Positive: 5+5. Negative bridge control: 4+6. Both are two cycles
        # joined by the marked edge, hence V=10, E=11 and degrees 3,3,2x8.
        left_n = 5 if positive else rng.choice((4, 6))
        right_n = 10 - left_n
        nodes = _polygon(130, 224, radius, left_n, 0.0)
        nodes += _polygon(318, 224, radius, right_n, math.pi)
        edges = _cycle(0, left_n) + _cycle(left_n, right_n)
        marked = [0, left_n]
        edges.append(marked[:])
    else:
        # Negative connectivity control: a decagon plus one marked chord. Its
        # removal leaves the perimeter connected, with the same V/E/degrees.
        nodes = _polygon(224, 224, rng.randint(94, 101), 10, -math.pi / 2)
        edges = _cycle(0, 10)
        marked = [0, 2]
        edges.append(marked[:])

    code = rng.randrange(8)
    dx, dy = rng.randint(-7, 7), rng.randint(-7, 7)
    nodes = [_dihedral(p, code, dx, dy) for p in nodes]

    # Randomize latent identities without affecting the visible topology.
    order = list(range(10))
    rng.shuffle(order)
    inverse = {old: new for new, old in enumerate(order)}
    nodes = [nodes[old] for old in order]
    edges = [[inverse[a], inverse[b]] for a, b in edges]
    marked = [inverse[marked[0]], inverse[marked[1]]]
    rng.shuffle(edges)

    return {
        "nodes": nodes,
        "edges": edges,
        "marked_edge": marked,
        "palette": deepcopy(PALETTES[rng.randrange(len(PALETTES))]),
    }


def _edge_key(edge: list[int]) -> tuple[int, int]:
    a, b = int(edge[0]), int(edge[1])
    return (a, b) if a < b else (b, a)


def render(scene: dict) -> Image.Image:
    """Draw edges behind solid colored junction disks at 3x antialiasing."""
    scale = 3
    palette = scene["palette"]
    image = Image.new("RGB", (CANVAS * scale, CANVAS * scale), tuple(palette["background"]))
    draw = ImageDraw.Draw(image)
    nodes = scene["nodes"]
    marked = _edge_key(scene["marked_edge"])
    for edge in scene["edges"]:
        a, b = int(edge[0]), int(edge[1])
        xy = [tuple(v * scale for v in nodes[a]), tuple(v * scale for v in nodes[b])]
        color = palette["mark"] if _edge_key(edge) == marked else palette["edge"]
        draw.line(xy, fill=tuple(color), width=8 * scale)
    radius = 12 * scale
    for x, y in nodes:
        cx, cy = x * scale, y * scale
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=tuple(palette["node"]))
        # A small light center makes every junction visibly discrete while the
        # saturated annulus remains one oracle component.
        dot = 3 * scale
        draw.ellipse((cx - dot, cy - dot, cx + dot, cy + dot), fill=tuple(palette["background"]))
    return image.resize((CANVAS, CANVAS), Image.Resampling.LANCZOS)


def _components_after_cut(edges: list[list[int]], marked_edge: list[int]) -> list[set[int]]:
    vertices = {int(v) for edge in edges for v in edge}
    cut = _edge_key(marked_edge)
    adjacency = {v: set() for v in vertices}
    removed = False
    for edge in edges:
        a, b = int(edge[0]), int(edge[1])
        if _edge_key(edge) == cut and not removed:
            removed = True
            continue
        adjacency[a].add(b)
        adjacency[b].add(a)
    groups: list[set[int]] = []
    unseen = set(vertices)
    while unseen:
        root = min(unseen)
        group = {root}
        queue = deque([root])
        unseen.remove(root)
        while queue:
            for nxt in adjacency[queue.popleft()]:
                if nxt in unseen:
                    unseen.remove(nxt)
                    group.add(nxt)
                    queue.append(nxt)
        groups.append(group)
    return groups


def analytic_gold(scene: dict) -> str:
    groups = _components_after_cut(scene["edges"], scene["marked_edge"])
    return "yes" if len(groups) == 2 and len(groups[0]) == len(groups[1]) else "no"


def margin(scene: dict) -> float:
    """Discrete runner-up gap: component-count or half-size mismatch units."""
    groups = _components_after_cut(scene["edges"], scene["marked_edge"])
    if len(groups) == 2:
        return float(max(1, abs(len(groups[0]) - len(groups[1])) // 2))
    return float(max(1, abs(len(groups) - 2)))


def is_quarantined(scene: dict) -> bool:
    nodes = scene["nodes"]
    if len(nodes) != 10 or len(scene["edges"]) != 11:
        return True
    # Generated centers stay far enough apart to expose every incident link.
    return min(math.dist(a, b) for i, a in enumerate(nodes) for b in nodes[i + 1:]) < 38


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["marked_edge"] = list(reversed(scene["marked_edge"]))
    return [("reverse_marked_undirected_encoding", twin)]
