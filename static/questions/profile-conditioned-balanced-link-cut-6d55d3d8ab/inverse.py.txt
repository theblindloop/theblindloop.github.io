"""Independent image-only inverse arm for balanced_link_cut.

It segments junction annuli and individual link strokes, binds each stroke to
its two nearest junction centers, removes the uniquely orange stroke, and then
computes component cardinalities. It intentionally imports no candidate module.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = deque([(int(y), int(x))])
        pts = []
        while queue:
            cy, cx = queue.popleft()
            pts.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(pts) >= minimum:
            output.append(np.asarray(pts, dtype=np.float64))
    return output


def _bind_link(points: np.ndarray, centers: np.ndarray) -> tuple[int, int]:
    xy = points[:, ::-1]
    # Minimum ink-to-center distance is insensitive to stroke length/direction.
    distances = []
    for center in centers:
        delta = xy - center
        distances.append(float(np.min(np.sum(delta * delta, axis=1))))
    order = np.argsort(distances)
    if len(order) < 2 or distances[int(order[1])] > 25.0 ** 2:
        raise ValueError("link endpoints are not recoverable")
    return tuple(sorted((int(order[0]), int(order[1]))))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = arr[..., 0], arr[..., 1], arr[..., 2]
    orange = (red > 175) & (red > green + 60) & (green > 65) & (green < 180) & (blue < 115)
    # All node palettes have a strong green-over-red chroma. This excludes
    # gray edge antialias pixels even when their absolute green value is high.
    node = (red < 150) & (green > 105) & ((green - red) > 40) & (blue > 70) & ~orange
    dark = (red < 105) & (green < 105) & (blue < 120) & ~node

    node_parts = _components(node, 120)
    if len(node_parts) != 10:
        raise ValueError(f"expected 10 visible junctions, found {len(node_parts)}")
    centers = np.asarray([[part[:, 1].mean(), part[:, 0].mean()] for part in node_parts])

    dark_parts = _components(dark, 45)
    orange_parts = _components(orange, 45)
    if len(orange_parts) != 1:
        raise ValueError(f"expected one orange link, found {len(orange_parts)}")

    edges = [_bind_link(part, centers) for part in dark_parts]
    marked = _bind_link(orange_parts[0], centers)
    if len(edges) != 10 or marked in edges or len(set(edges)) != 10:
        raise ValueError("visible link graph is incomplete or ambiguous")

    adjacency = {i: set() for i in range(10)}
    for a, b in edges:  # orange edge is deliberately omitted
        adjacency[a].add(b)
        adjacency[b].add(a)
    unseen = set(adjacency)
    sizes = []
    while unseen:
        root = min(unseen)
        unseen.remove(root)
        queue = deque([root])
        size = 0
        while queue:
            size += 1
            for nxt in adjacency[queue.popleft()]:
                if nxt in unseen:
                    unseen.remove(nxt)
                    queue.append(nxt)
        sizes.append(size)
    return "yes" if sorted(sizes) == [5, 5] else "no"
