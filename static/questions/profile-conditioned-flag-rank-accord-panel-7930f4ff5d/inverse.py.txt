"""Image-only inverse arm for the flag-rank-accord panel world.

It receives a PNG and nothing else: no scene, no renderer, no analytic gold. It
rules the image into panels with the grey frame ink, splits each panel interior
into flag columns, measures every flag's pole height and disk area from pixels
alone, and reports the unique panel whose disk areas rise with pole height.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

AMBIGUOUS = "ambiguous"

EXPECTED_PANELS = 5
EXPECTED_FLAGS = 4


def _masks(image: Image.Image):
    a = np.asarray(image.convert("RGB"), dtype=np.int16)
    hi = a.max(axis=2)
    lo = a.min(axis=2)
    chroma = hi - lo
    dark = (hi < 100) & (chroma < 40)
    accent = (chroma >= 60) & (hi >= 60)
    grey = (chroma < 30) & (lo >= 100) & (hi <= 200)
    return dark, accent, grey


def _runs(flags: np.ndarray) -> list:
    """Contiguous True runs as (start, stop) half-open column intervals."""
    out = []
    start = None
    for index, value in enumerate(flags):
        if value and start is None:
            start = index
        elif not value and start is not None:
            out.append((start, index))
            start = None
    if start is not None:
        out.append((start, len(flags)))
    return out


def _panel_bounds(grey: np.ndarray) -> list:
    height = grey.shape[0]
    columns = grey.sum(axis=0) > 0.6 * height
    lines = _runs(columns)
    if len(lines) != 2 * EXPECTED_PANELS:
        return []
    bounds = []
    for index in range(EXPECTED_PANELS):
        left = lines[2 * index][1]
        right = lines[2 * index + 1][0]
        if right - left < 40:
            return []
        bounds.append((left, right))
    return bounds


def _panel_flags(dark: np.ndarray, accent: np.ndarray, left: int, right: int):
    dark_columns = dark[:, left:right].sum(axis=0)
    accent_columns = accent[:, left:right].sum(axis=0)
    ink = (dark_columns + accent_columns) > 0
    runs = _runs(ink)
    if len(runs) != EXPECTED_FLAGS:
        return None
    measured = []
    for start, stop in runs:
        pole = int(dark_columns[start:stop].max())
        disk = int(accent_columns[start:stop].sum())
        if pole <= 0 or disk <= 0:
            return None
        measured.append((pole, disk))
    return measured


def _is_accordant(measured) -> bool:
    ordered = sorted(measured)
    for i in range(len(ordered) - 1):
        if ordered[i][0] == ordered[i + 1][0]:
            return False
        if ordered[i][1] >= ordered[i + 1][1]:
            return False
    return True


def decision_from_image(image: Image.Image) -> str:
    dark, accent, grey = _masks(image)
    bounds = _panel_bounds(grey)
    if not bounds:
        return AMBIGUOUS
    winners = []
    for index, (left, right) in enumerate(bounds):
        measured = _panel_flags(dark, accent, left, right)
        if measured is None:
            return AMBIGUOUS
        if _is_accordant(measured):
            winners.append(index)
    if len(winners) != 1:
        return AMBIGUOUS
    return f"panel-{winners[0] + 1}"
