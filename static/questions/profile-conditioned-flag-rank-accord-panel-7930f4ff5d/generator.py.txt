"""Renderer for the flag-rank-accord panel world.

Five framed panels stand side by side. Every panel holds four flags: a dark
vertical pole with a solid disk resting on its top. Within a scene all five
panels reuse the very same four pole heights and the very same four disk sizes,
so panel ink, panel count, size range and colour are identical across panels and
only the pairing of a height with a disk size differs. Exactly one panel is
rank-accordant: sorting its four flags by pole height also sorts them by disk
size, so a taller pole always carries a larger disk.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WORLD_VERSION = "flag_rank_accord_panel-0.1.0"

CANVAS_W = 1340
CANVAS_H = 250
PANELS = 5
PANEL_W = 260
PANEL_GAP = 8
PANEL_TOP = 8
PANEL_BOTTOM = 242
MARGIN_X = (CANVAS_W - (PANELS * PANEL_W + (PANELS - 1) * PANEL_GAP)) // 2
BASELINE = 224
POLE_W = 14
SLOT_DX = (38, 100, 162, 224)
FLAGS = 4
QUARANTINE_MARGIN = 4.0

BG = (247, 247, 244)
FRAME = (128, 128, 128)
POLE = (32, 32, 36)
ACCENTS = (
    (198, 52, 44),
    (38, 86, 176),
    (28, 132, 84),
    (150, 60, 160),
    (196, 120, 20),
)

_IDENTITY = (0, 1, 2, 3)
_REVERSAL = (3, 2, 1, 0)
_NEAR_MISS = (0, 2, 1, 3)


def _panel_left(index: int) -> int:
    return MARGIN_X + index * (PANEL_W + PANEL_GAP)


def _all_permutations() -> list:
    perms = []

    def walk(prefix, rest):
        if not rest:
            perms.append(tuple(prefix))
            return
        for item in rest:
            walk(prefix + [item], [r for r in rest if r != item])

    walk([], [0, 1, 2, 3])
    return perms


def _is_monotone(values) -> bool:
    ascending = all(values[i] < values[i + 1] for i in range(len(values) - 1))
    descending = all(values[i] > values[i + 1] for i in range(len(values) - 1))
    return ascending or descending


def sample_scene(seed: int) -> dict:
    """Sample one balanced five-panel scene."""
    rng = random.Random(seed)
    for _ in range(4000):
        heights = [rng.randint(40, 50)]
        for _ in range(FLAGS - 1):
            heights.append(heights[-1] + rng.randint(16, 22))
        radii = [rng.randint(8, 9)]
        for _ in range(FLAGS - 1):
            radii.append(radii[-1] + rng.randint(5, 6))

        winner = rng.randrange(PANELS)
        pool = [p for p in _all_permutations() if p not in (_IDENTITY, _REVERSAL, _NEAR_MISS)]
        rng.shuffle(pool)
        losers = [_REVERSAL, _NEAR_MISS, pool[0], pool[1]]
        rng.shuffle(losers)
        assignments = list(losers)
        assignments.insert(winner, _IDENTITY)

        panels = []
        spatially_monotone = []
        for index, assignment in enumerate(assignments):
            order = list(range(FLAGS))
            rng.shuffle(order)
            left = _panel_left(index)
            flags = []
            for slot, rank in enumerate(order):
                flags.append(
                    {
                        "x": left + SLOT_DX[slot],
                        "h": heights[rank],
                        "r": radii[assignment[rank]],
                    }
                )
            panels.append({"flags": flags})
            spatially_monotone.append(_is_monotone([f["h"] for f in flags]))

        # The winner must not also be the panel whose poles rise or fall across
        # the row, and some loser must be spatially monotone, so "sorted from
        # left to right" never picks the answer.
        if spatially_monotone[winner]:
            continue
        if not any(spatially_monotone[i] for i in range(PANELS) if i != winner):
            continue

        scene = {
            "canvas": [CANVAS_W, CANVAS_H],
            "palette": rng.randrange(len(ACCENTS)) + len(ACCENTS) * rng.randint(0, 2),
            "panels": panels,
        }
        if margin(scene) >= QUARANTINE_MARGIN and _accordant(scene).count(True) == 1:
            return scene
    raise RuntimeError("failed to sample a balanced scene")


def _sorted_by_height(flags: list) -> list:
    return sorted(flags, key=lambda flag: flag["h"])


def _accordant(scene: dict) -> list:
    out = []
    for panel in scene["panels"]:
        flags = _sorted_by_height(panel["flags"])
        out.append(all(flags[i]["r"] < flags[i + 1]["r"] for i in range(len(flags) - 1)))
    return out


def _evidence(flags: list) -> float:
    """Pixel evidence certifying one panel's accordance verdict."""
    flags = _sorted_by_height(flags)
    inversions = [
        min(flags[j]["h"] - flags[i]["h"], flags[i]["r"] - flags[j]["r"])
        for i in range(len(flags))
        for j in range(i + 1, len(flags))
        if flags[i]["r"] > flags[j]["r"]
    ]
    if inversions:
        return float(max(inversions))
    return float(
        min(
            min(flags[i + 1]["h"] - flags[i]["h"], flags[i + 1]["r"] - flags[i]["r"])
            for i in range(len(flags) - 1)
        )
    )


def margin(scene: dict) -> float:
    """Smallest pixel evidence over the five per-panel verdicts."""
    return float(min(_evidence(panel["flags"]) for panel in scene["panels"]))


def is_quarantined(scene: dict) -> bool:
    return _accordant(scene).count(True) != 1 or margin(scene) < QUARANTINE_MARGIN


def analytic_gold(scene: dict) -> str:
    verdicts = _accordant(scene)
    if verdicts.count(True) != 1:
        raise ValueError("scene does not hold exactly one rank-accordant panel")
    return f"panel-{verdicts.index(True) + 1}"


def render(scene: dict) -> Image.Image:
    width, height = scene["canvas"]
    image = Image.new("RGB", (width, height), BG)
    draw = ImageDraw.Draw(image)
    accent = ACCENTS[scene["palette"] % len(ACCENTS)]
    for index, panel in enumerate(scene["panels"]):
        left = _panel_left(index)
        draw.rectangle(
            [left, PANEL_TOP, left + PANEL_W - 1, PANEL_BOTTOM], outline=FRAME, width=2
        )
        draw.text((left + 10, PANEL_BOTTOM - 15), str(index + 1), fill=FRAME)
        for flag in sorted(panel["flags"], key=lambda item: item["x"]):
            x = flag["x"]
            top = BASELINE - flag["h"]
            draw.rectangle([x - POLE_W // 2, top, x + POLE_W // 2 - 1, BASELINE], fill=POLE)
            r = flag["r"]
            draw.ellipse([x - r, top - 2 * r, x + r, top], fill=accent)
    return image


def latent_symmetries(scene: dict) -> list:
    """Pixel-identical, answer-preserving relabelings of the latent scene."""
    reversed_slots = {
        "canvas": list(scene["canvas"]),
        "palette": scene["palette"],
        "panels": [{"flags": list(reversed(panel["flags"]))} for panel in scene["panels"]],
    }
    shifted_palette = {
        "canvas": list(scene["canvas"]),
        "palette": scene["palette"] + len(ACCENTS),
        "panels": [{"flags": list(panel["flags"])} for panel in scene["panels"]],
    }
    return [
        ("flag_slot_reverse", reversed_slots),
        ("palette_period", shifted_palette),
    ]
