def _label_points(label_index: int) -> frozenset[tuple[int, int]]:
    pattern = LABEL_PATTERNS[label_index]
    return frozenset(
        (x * LABEL_SCALE + dx, y * LABEL_SCALE + dy)
        for y, row in enumerate(pattern)
        for x, value in enumerate(row)
        if value == "1"
        for dx in range(LABEL_SCALE)
        for dy in range(LABEL_SCALE)
    )

def _state_code_points(state: int) -> frozenset[tuple[int, int]]:
    pattern = STATE_CODE_PATTERNS[state]
    return frozenset(
        (x * STATE_CODE_SCALE + dx, y * STATE_CODE_SCALE + dy)

def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    names = (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf",
        "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
    ) if bold else (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf",
    )
    for name in names:
        if Path(name).is_file():
            return ImageFont.truetype(name, size)
    return ImageFont.load_default()


def _centered_text(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    text: str,
    font: ImageFont.FreeTypeFont | ImageFont.ImageFont,
    fill: tuple[int, int, int],
) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text(
        (center[0] - (box[2] - box[0]) / 2, center[1] - (box[3] - box[1]) / 2 - box[1]),
        text,
        font=font,
        fill=fill,
    )


def _draw_points(
    draw: ImageDraw.ImageDraw,
    origin: tuple[int, int],
    points: frozenset[tuple[int, int]],
    fill: tuple[int, int, int],
) -> None:
    ox, oy = origin
    for x, y in points:
        draw.point((ox + x, oy + y), fill=fill)


def _core_tables(scene: dict) -> tuple[list[list[int]], list[list[int]]]:
    states_raw = scene["states"]
    positions_raw = scene["positions"]
    if not isinstance(states_raw, list) or len(states_raw) != PANEL_COUNT:
        raise ValueError("states must contain seven panels")
    if not isinstance(positions_raw, list) or len(positions_raw) != PANEL_COUNT:
        raise ValueError("positions must contain seven panels")
    states: list[list[int]] = []
    positions: list[list[int]] = []
    for state_panel, position_panel in zip(states_raw, positions_raw):
        if not isinstance(state_panel, list) or len(state_panel) != IDENTITY_COUNT:
            raise ValueError("each state panel must contain four identities")
        if not isinstance(position_panel, list) or len(position_panel) != IDENTITY_COUNT:
            raise ValueError("each position panel must contain four identities")
        state_values = [int(value) for value in state_panel]
        position_values = [int(value) for value in position_panel]
        if any(value not in range(STATE_COUNT) for value in state_values):
            raise ValueError("states must use LOW=0, MID=1, HIGH=2")
        if sorted(position_values) != list(range(SLOT_COUNT)):
            raise ValueError("each position panel must be a slot permutation")
        states.append(state_values)
        positions.append(position_values)
    return states, positions


def _echo_scores(scene: dict) -> list[int]:
    states, positions = _core_tables(scene)
    return [
        sum(
            states[panel][label] != states[panel + 1][label]
            and positions[panel + 1][label] != positions[panel + 2][label]
            for panel in range(PANEL_COUNT - 2)
        )
        for label in range(IDENTITY_COUNT)
    ]


def _swap_labels(position_panel: list[int], pair: tuple[int, int] | None) -> list[int]:
    result = list(position_panel)
    if pair is not None:
        first, second = pair
        result[first], result[second] = result[second], result[first]
    return result


def _sample_positions(rng: random.Random) -> list[list[int]]:
    start = list(range(SLOT_COUNT))
    rng.shuffle(start)
    panels = [start]
    first_operation = rng.choice((*PAIR_OPERATIONS, None))
    panels.append(_swap_labels(panels[-1], first_operation))
    operations: list[tuple[int, int] | None] = [*PAIR_OPERATIONS, None]
    rng.shuffle(operations)
    for operation in operations:
        panels.append(_swap_labels(panels[-1], operation))
    if len(panels) != PANEL_COUNT:
        raise AssertionError("position sampler produced the wrong panel count")
    return panels


def _sample_states(
    rng: random.Random,
    positions: list[list[int]],
    winner: int,
) -> list[list[int]]:
    movement_window = [
        [positions[panel + 1][label] != positions[panel + 2][label] for panel in range(5)]
        for label in range(IDENTITY_COUNT)
    ]
    if any(sum(flags) != 2 for flags in movement_window):
        raise AssertionError("scored movement schedule must move every label twice")
    change_indices: list[set[int]] = []
    for label, flags in enumerate(movement_window):
        moving = [index for index, flag in enumerate(flags) if flag]
        still = [index for index, flag in enumerate(flags) if not flag]
        if label == winner:
            selected = moving
        else:
            selected = rng.sample(still, 2)
        change_indices.append(set(selected))

    states_by_panel = [[0 for _ in range(IDENTITY_COUNT)] for _ in range(PANEL_COUNT)]
    for label in range(IDENTITY_COUNT):
        states_by_panel[0][label] = rng.randrange(STATE_COUNT)
    for edge in range(PANEL_COUNT - 1):
        for label in range(IDENTITY_COUNT):
            previous = states_by_panel[edge][label]
            should_change = edge in change_indices[label]
            if edge == PANEL_COUNT - 2 and rng.random() < 0.5:
                should_change = True
            if should_change:
                choices = [state for state in range(STATE_COUNT) if state != previous]
                states_by_panel[edge + 1][label] = rng.choice(choices)
            else:
                states_by_panel[edge + 1][label] = previous
    return states_by_panel


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene whose requested label has a unique echo winner."""
    seed = int(seed)
    logic_rng = random.Random(seed * 1_000_003 + 17)
    style_rng = random.Random((seed + 104_729) * 1_000_033 + 31)
    winner = seed % IDENTITY_COUNT
    positions = _sample_positions(logic_rng)
    states = _sample_states(logic_rng, positions, winner)
    scene = {
        "states": states,
        "positions": positions,
        "state_palette": [
            list(color) for color in style_rng.choice(STATE_PALETTES)
        ],
        "state_key_order": list(range(STATE_COUNT)),
        "panel_tints": [],
    }
    style_rng.shuffle(scene["state_key_order"])
    scene["panel_tints"] = [
        list(style_rng.choice(PANEL_TINTS)) for _ in range(PANEL_COUNT)
    ]
    if analytic_gold(scene) != LABELS[winner]:
        raise AssertionError("sampler failed to create the requested echo winner")
    return scene


def render(scene: dict) -> Image.Image:
    states, positions = _core_tables(scene)
    palette_raw = scene["state_palette"]
    key_order_raw = scene["state_key_order"]
    tints_raw = scene["panel_tints"]
    if not isinstance(palette_raw, list) or len(palette_raw) != STATE_COUNT:
        raise ValueError("state_palette must contain three colors")
    palette = [tuple(int(channel) for channel in color) for color in palette_raw]
    if any(len(color) != 3 or any(channel not in range(256) for channel in color) for color in palette):
        raise ValueError("state_palette contains an invalid RGB color")
    key_order = [int(value) for value in key_order_raw]
    if sorted(key_order) != list(range(STATE_COUNT)):
        raise ValueError("state_key_order must be a permutation")
    if not isinstance(tints_raw, list) or len(tints_raw) != PANEL_COUNT:
        raise ValueError("panel_tints must contain seven RGB colors")
    tints = [tuple(int(channel) for channel in color) for color in tints_raw]
    if any(len(color) != 3 or any(channel not in range(256) for channel in color) for color in tints):
        raise ValueError("panel_tints contains an invalid RGB color")

    image = Image.new("RGB", CANVAS, BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(27, bold=True)
    section_font = _font(16, bold=True)
    body_font = _font(14)
    small_font = _font(12)
    label_font = _font(17, bold=True)
    tiny_font = _font(11)

    draw.text((28, 18), "STATE → MOTION ECHO", font=title_font, fill=INK)
    draw.text(
        (28, 56),
        "Align each identity's incoming state change with its outgoing slot change.",
        font=small_font,
        fill=MUTED,
    )
    draw.text((28, 84), "VISIBLE STATE KEY", font=section_font, fill=INK)
    for key_slot, state in enumerate(key_order):
        key_x = KEY_XS[key_slot]
        draw.rectangle(
            (key_x, KEY_Y, key_x + STATE_BOX_SIZE - 1, KEY_Y + STATE_BOX_SIZE - 1),
            fill=palette[state],
            outline=STATE_OUTLINE,
            width=2,
        )
        _draw_points(
            draw,
            (key_x + STATE_CODE_X_OFFSET, KEY_Y + STATE_CODE_Y_OFFSET),
            STATE_CODE_POINTS[state],
            INK,
        )
        _centered_text(
            draw,
            (key_x + 85, KEY_Y + STATE_BOX_SIZE / 2),
            STATE_NAMES[state],
            body_font,
            INK,
        )

    for panel in range(PANEL_COUNT - 1):
        left = PANEL_X0 + panel * (PANEL_WIDTH + PANEL_GAP) + PANEL_WIDTH + 2
        right = PANEL_X0 + (panel + 1) * (PANEL_WIDTH + PANEL_GAP) - 2
        y = PANEL_TOP - 25
        draw.line((left, y, right - 8, y), fill=OUTLINE, width=2)
        draw.polygon(((right - 8, y - 5), (right, y), (right - 8, y + 5)), fill=OUTLINE)

    for panel in range(PANEL_COUNT):
        panel_x = PANEL_X0 + panel * (PANEL_WIDTH + PANEL_GAP)
        draw.rounded_rectangle(
            (panel_x, PANEL_TOP, panel_x + PANEL_WIDTH, PANEL_TOP + PANEL_HEIGHT),
            radius=12,
            fill=tints[panel],
            outline=OUTLINE,
            width=2,
        )
        draw.text((panel_x + 16, PANEL_TOP + 14), f"P{panel + 1}", font=section_font, fill=INK)
        draw.text((panel_x + 112, PANEL_TOP + 17), "TOKEN / SLOT", font=small_font, fill=MUTED)
        draw.line(
            (panel_x + 16, PANEL_TOP + 55, panel_x + PANEL_WIDTH - 16, PANEL_TOP + 55),
            fill=OUTLINE,
            width=1,
        )
        label_at_slot = [0] * IDENTITY_COUNT
        for label, slot in enumerate(positions[panel]):
            label_at_slot[slot] = label
        for slot, (slot_x, slot_y) in enumerate(SLOT_ORIGINS):
            card_x = panel_x + slot_x
            card_y = PANEL_TOP + slot_y
            draw.rounded_rectangle(
                (card_x, card_y, card_x + CARD_WIDTH, card_y + CARD_HEIGHT),
                radius=9,
                fill=CARD,
                outline=OUTLINE,
                width=1,
            )
            draw.text((card_x + 10, card_y + 9), SLOT_NAMES[slot], font=tiny_font, fill=MUTED)
            label = label_at_slot[slot]
            _draw_points(
                draw,
                (card_x + LABEL_X_OFFSET, card_y + LABEL_Y_OFFSET),
                LABEL_POINTS[label],
                INK,
            )
            draw.text((card_x + 43, card_y + 44), LABELS[label], font=label_font, fill=INK)
            state = states[panel][label]
            state_x = card_x + STATE_X_OFFSET
            state_y = card_y + STATE_Y_OFFSET
            draw.rectangle(
                (state_x, state_y, state_x + STATE_BOX_SIZE - 1, state_y + STATE_BOX_SIZE - 1),
                fill=palette[state],
                outline=STATE_OUTLINE,
                width=2,
            )
            draw.text((card_x + 39, card_y + 100), STATE_NAMES[state], font=tiny_font, fill=MUTED)
            draw.line(
                (card_x + 12, card_y + 137, card_x + CARD_WIDTH - 12, card_y + 137),
                fill=(218, 226, 231),
                width=1,
            )
            draw.text((card_x + 12, card_y + 150), "incoming state", font=tiny_font, fill=MUTED)
            draw.text((card_x + 12, card_y + 166), "outgoing slot", font=tiny_font, fill=MUTED)
    return image


def analytic_gold(scene: dict) -> str:
    scores = _echo_scores(scene)
    best = max(scores)
    winners = [label for label, score in enumerate(scores) if score == best]
    ordered = sorted(scores, reverse=True)
    if len(winners) != 1 or ordered[0] - ordered[1] < 2:
        raise ValueError("echo scores are tied or below the two-count decision margin")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    ordered = sorted(_echo_scores(scene), reverse=True)
    return float(ordered[0] - ordered[1])


def is_quarantined(scene: dict) -> bool:
    try:
        scores = _echo_scores(scene)
        ordered = sorted(scores, reverse=True)
        return len([score for score in scores if score == ordered[0]]) != 1 or ordered[0] - ordered[1] < 2
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No render-invariant aliases are claimed for this fully keyed scene."""
    return []
