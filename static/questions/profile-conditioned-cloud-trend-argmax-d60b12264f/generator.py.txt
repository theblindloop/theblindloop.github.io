"""Steeper-trend comparison world: which colored dot cloud has the greater |slope| trend.

Latent scene z carries two colored scatter clouds (red, blue), each confined to
its own horizontal band of the panel. Every cloud consists of a fixed number of
dot markers scattered with small perpendicular jitter around a straight trend
line of a given slope. The task is a pure measurement: recover each cloud's
best-fit trend slope from the raster ink and select the color whose cloud is
drawn with the steeper trend (larger absolute slope). The answer is bound to a
color label, counterbalanced over which color owns the extremum and over the
arbitrary spatial layout of both clouds.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "cloud_trend_argmax-0.1.0"

PALETTE = ["red", "blue"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
}
BACKGROUND = (250, 250, 250)

WIDTH = 480
HEIGHT = 400
DOT_RADIUS = 3
POINTS_PER_CLOUD = 15
X0 = 60
X1 = 420
X_MID = (X0 + X1) / 2.0

# Two dedicated bands, each tall enough for clearly visible trend slopes.
BAND_TOPS = [45.0, 215.0]
BAND_H = 150.0
MAX_EFF_SLOPE = 0.5 * BAND_H / (X1 - X0)  # keep half the band height for the rise
JITTER = 3.5

# Minimum top-two |slope| gap required for a decidable comparison.
MIN_MARGIN = 0.012
# Below this |slope| gap the scene is quarantined as near-boundary yet still
# decidable (well above estimated fit noise).
QUARANTINE = 0.022


def _band_geometries(slopes: list[float], band_indexes: list[int]) -> "list[dict]":
    """Return per-band trend {slope, intercept} constrained to the band height."""
    out = []
    for slope, band in zip(slopes, band_indexes):
        band_center = BAND_TOPS[band] + BAND_H / 2.0
        intercept = band_center - slope * X_MID
        out.append({"slope": slope, "intercept": intercept})
    return out


def sample_scene(seed: int) -> dict:
    """Sample a deterministic scene with a guaranteed, counterbalanced extremum.

    Effective trend slopes are sampled directly in a range that always fits the
    band, and the top-two |slope| gap is enforced, so both clouds are decidable.
    Returns a plain dict (JSON-serialisable) that both render() and the manifest
    understand. ``clouds`` is a list of {"color", "x", "y"} for every dot.
    """
    rng = random.Random(seed * 1009 + 7)
    ncols = len(PALETTE)
    while True:
        raw_mags = [rng.uniform(0.03, MAX_EFF_SLOPE) for _ in range(ncols)]
        signs = [1 if rng.random() < 0.5 else -1 for _ in range(ncols)]
        mags_sorted = sorted(raw_mags, reverse=True)
        if mags_sorted[0] - mags_sorted[1] >= MIN_MARGIN:
            break
    colors = list(PALETTE)
    rng.shuffle(colors)
    bands = rng.sample(range(len(BAND_TOPS)), ncols)
    desired = [signs[i] * raw_mags[i] for i in range(ncols)]
    geos = _band_geometries(desired, bands)
    clouds = []
    for color, band, geo in zip(colors, bands, geos):
        xs = sorted(rng.uniform(X0, X1) for _ in range(POINTS_PER_CLOUD))
        for x in xs:
            ideal_y = geo["slope"] * x + geo["intercept"]
            y = ideal_y + rng.uniform(-JITTER, JITTER)
            clouds.append({"color": color, "x": x, "y": y})
    return {
        "clouds": clouds,
        "width": WIDTH,
        "height": HEIGHT,
        "background": "light",
    }


def render(scene: dict) -> Image.Image:
    """Render I = R(z): two colored dot clouds, each in its own band."""
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    for dot in scene["clouds"]:
        x, y, color = dot["x"], dot["y"], dot["color"]
        draw.ellipse(
            [x - DOT_RADIUS, y - DOT_RADIUS, x + DOT_RADIUS, y + DOT_RADIUS],
            fill=COLOR_RGB[color],
        )
    return img


def _cloud_slope(scene: dict, color: str) -> float:
    """Analytic best-fit trend slope of one cloud from its latent coordinates."""
    xs, ys = [], []
    for dot in scene["clouds"]:
        if dot["color"] == color:
            xs.append(dot["x"])
            ys.append(dot["y"])
    n = len(xs)
    xm = sum(xs) / n
    ym = sum(ys) / n
    num = sum((x - xm) * (y - ym) for x, y in zip(xs, ys))
    den = sum((x - xm) ** 2 for x in xs)
    return num / den if den != 0 else 0.0


def _steeper(scene: dict) -> "tuple[str, float, float]":
    """Return (winning color, its |slope|, decision margin)."""
    keyed = [(color, abs(_cloud_slope(scene, color))) for color in PALETTE]
    keyed.sort(key=lambda item: item[1], reverse=True)
    top, second = keyed[0], keyed[1]
    return top[0], top[1], top[1] - second[1]


def analytic_gold(scene: dict) -> str:
    """The color of the cloud with the larger absolute trend slope."""
    top, _mag, _margin = _steeper(scene)
    return top


def margin(scene: dict) -> float:
    """Verifiability margin: the top-two |slope| gap."""
    _top, _mag, m = _steeper(scene)
    return float(m)


def is_quarantined(scene: dict) -> bool:
    """Near-boundary: the |slope| gap is finite-resolution small."""
    return margin(scene) < QUARANTINE


def latent_symmetries(scene: dict) -> list:
    """not_applicable: no nontrivial pixel-identical latent symmetry.

    Every latent field (each dot's color label and its position) is directly
    rendered as visible ink, so there is no hidden scene field that could change
    the gold without changing the raster. Swapping the two clouds' colors changes
    which color is correct, and any global reflection or rotation turns the
    raster into a distinct image.
    """
    return []
