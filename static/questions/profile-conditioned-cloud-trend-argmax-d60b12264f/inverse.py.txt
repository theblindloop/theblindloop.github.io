"""Independent pixel-only inverse arm for the cloud-trend comparison world.

This module receives ONLY the rasterized PNG (plus the public palette, which is
the world's fixed naming convention) and re-measures, for each colored dot cloud,
the slope of the best-fit trend line through its ink, then selects the color
with the larger absolute slope. It shares no geometry with renderer.render() and
never imports the renderer, prompts, generator, verifier, or any analytical gold
code.
"""

from __future__ import annotations

from typing import Any

import numpy as np
from PIL import Image

ORACLE_VERSION = "cloud_trend_argmax-oracle-0.1.0"

PALETTE = ["red", "blue"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
}
BACKGROUND = (250, 250, 250)


def _nearest_color(arr: np.ndarray) -> np.ndarray:
    """Label every pixel to its nearest palette entry (segment colors + background)."""
    refs = np.stack(
        [np.array(COLOR_RGB[c], dtype=np.float32) for c in PALETTE]
        + [np.array(BACKGROUND, dtype=np.float32)],
        axis=0,
    )
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def _trend_slope(mask: np.ndarray) -> float:
    """Absolute least-squares trend slope of a cloud's ink.

    x is the column index and y is the row index of every inked pixel of that
    color. Because all dots of a cloud share one radius and none are clipped, the
    per-dot pixel clusters have equal weight, so the regression slope through the
    filled pixels approximates the trend slope of the cloud's scatter.
    """
    ys, xs = np.nonzero(mask)
    if len(xs) < 12:
        return -1.0
    xf = xs.astype(np.float64)
    yf = ys.astype(np.float64)
    xm = xf.mean()
    ym = yf.mean()
    den = float(((xf - xm) ** 2).sum())
    if den == 0.0:
        return 0.0
    slope = float(((xf - xm) * (yf - ym)).sum() / den)
    return abs(slope)


def decision_from_image(image: Image.Image) -> str:
    """Return the color of the cloud with the steeper trend from pixels alone."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _nearest_color(arr)
    best_color = PALETTE[0]
    best_slope = -1.0
    for idx, color in enumerate(PALETTE):
        mask = labels == idx
        slope = _trend_slope(mask)
        if slope > best_slope:
            best_slope = slope
            best_color = color
    return best_color
