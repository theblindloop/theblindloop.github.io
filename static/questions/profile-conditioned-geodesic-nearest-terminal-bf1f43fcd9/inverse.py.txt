"""Pixel-only inverse arm for the geodesic-nearest-terminal world.

Reads only the final RGB PNG. Segments colours by nearest-palette labelling,
labels the single 4-connected blue path, then computes a geodesic (BFS) distance
field over the blue stroke seeded from the pixels adjacent to the orange port.
Each coloured terminal marker is mapped to the BFS distance of its nearest blue
pixel (its position along the path). The decision is the colour of the terminal
with the smallest on-path distance from the port.

No renderer, prompt, generator, verifier or scene data is imported or read.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_PALETTE = {
    "blue": (40, 90, 210),
    "red": (225, 40, 40),
    "green": (40, 170, 70),
    "blue_mark": (30, 80, 180),
    "orange": (245, 150, 20),
    "white": (255, 255, 255),
}
_NAMES = ["blue", "red", "green", "blue_mark", "orange", "white"]
_NEAR_R = 8


def _label_pixels(arr):
    palette = np.array([*_PALETTE.values()], dtype=np.float32)
    dists = np.stack([np.linalg.norm(arr - c, axis=-1) for c in palette], axis=-1)
    return dists.argmin(axis=-1)


def _marker_centroid(mask):
    ys, xs = np.nonzero(mask)
    if len(ys) == 0:
        return None
    return int(round(float(ys.mean()))), int(round(float(xs.mean())))


def _nearest_blue_dist(blue_mask, dist, cy, cx):
    """Return BFS distance at the blue pixel nearest to marker centroid (cy,cx)."""
    h, w = blue_mask.shape
    best_d = float("inf")
    best = None
    for yy in range(max(0, cy - _NEAR_R), min(h, cy + _NEAR_R)):
        for xx in range(max(0, cx - _NEAR_R), min(w, cx + _NEAR_R)):
            if blue_mask[yy, xx]:
                dd = (yy - cy) ** 2 + (xx - cx) ** 2
                if dd < best_d:
                    best_d = dd
                    best = (yy, xx)
    if best is None:
        return None
    yy, xx = best
    return dist[yy, xx]


def _bfs_distances(blue_mask, seed_pts):
    h, w = blue_mask.shape
    dist = np.full((h, w), -1, dtype=np.int64)
    dq = deque()
    for (y, x) in seed_pts:
        if 0 <= y < h and 0 <= x < w and blue_mask[y, x]:
            dist[y, x] = 0
            dq.append((y, x))
    while dq:
        y, x = dq.popleft()
        nd = dist[y, x] + 1
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < h and 0 <= nx < w and blue_mask[ny, nx] and dist[ny, nx] == -1:
                dist[ny, nx] = nd
                dq.append((ny, nx))
    return dist


def _port_seed_blue(labels, idx):
    orange = labels == idx["orange"]
    h, w = orange.shape
    seeds = []
    for y in range(h):
        for x in range(w):
            if orange[y, x]:
                for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                    if 0 <= ny < h and 0 <= nx < w and labels[ny, nx] == idx["blue"]:
                        seeds.append((ny, nx))
    return seeds


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label_pixels(arr)
    idx = {name: i for i, name in enumerate(_NAMES)}
    blue = labels == idx["blue"]

    seeds = _port_seed_blue(labels, idx)
    if not seeds:
        raise ValueError("port does not touch the blue path")
    dist = _bfs_distances(blue, seeds)
    if not (dist >= 0).any():
        raise ValueError("no blue reachable from port")

    best_colour = None
    best_d = None
    for name in ("red", "green", "blue_mark"):
        centroid = _marker_centroid(labels == idx[name])
        if centroid is None:
            raise ValueError(f"{name} marker not found")
        cy, cx = centroid
        d = _nearest_blue_dist(blue, dist, cy, cx)
        if d is None or d < 0:
            raise ValueError(f"{name} marker not on reachable path")
        if best_d is None or d < best_d:
            best_d = d
            best_colour = "blue" if name == "blue_mark" else name
    if best_colour is None:
        raise ValueError("no terminal found")
    return best_colour
