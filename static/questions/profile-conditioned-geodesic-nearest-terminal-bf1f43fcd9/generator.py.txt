"""Tracing world: which coloured terminal is nearest to the ring along the path.

A single winding blue path (one connected, non-self-crossing serpentine
polyline) is drawn. An orange port ring sits near the start of the path. Three
distinctly coloured terminal dots (red, green, blue) sit on the path at
separated on-path positions. Walking from the orange ring along the blue path,
the answer is the colour of the terminal whose on-path (geodesic) distance from
the ring is smallest. Because the path winds back on itself in a serpentine,
on-path order need not equal straight-line (Euclidean) proximity, so the
decision requires tracing the drawn stroke and is not recoverable by Euclidean
distance.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "geodesic_nearest_terminal-0.1.0"

CANVAS = (480, 480)
BG = (255, 255, 255)
COLOR_BLUE = (40, 90, 210)
COLOR_RED = (225, 40, 40)
COLOR_GREEN = (40, 170, 70)
COLOR_BLUE_MARK = (30, 80, 180)
COLOR_ORANGE = (245, 150, 20)

STROKE = 20
PORT_R = 5
MARK_R = 4
MARGIN_PAD = 46
LANE_GAP = 30.0
ARC_SEP = 36.0
END_MARGIN = 40.0

TERMINAL_COLORS = ["red", "green", "blue"]
MARK_COLORS = {"red": COLOR_RED, "green": COLOR_GREEN, "blue": COLOR_BLUE_MARK}


def _pts(points):
    return [(float(p[0]), float(p[1])) for p in points]


def _path_length(points):
    return sum(math.dist(a, b) for a, b in zip(points, points[1:]))


def _point_at(points, s):
    remaining = s
    for a, b in zip(points, points[1:]):
        seg = math.dist(a, b)
        if remaining <= seg or seg == 0:
            t = 0.0 if seg == 0 else remaining / seg
            return (a[0] + (b[0] - a[0]) * t, a[1] + (b[1] - a[1]) * t)
        remaining -= seg
    return points[-1]


def _rejection_loop(rng):
    w, h = CANVAS
    lo, hi = MARGIN_PAD, w - MARGIN_PAD
    for _ in range(20000):
        lane_vert_gap = LANE_GAP
        n_lanes = rng.randint(4, 7)
        y_lo = lo + 10
        y_hi = hi - 10
        usable_h = y_hi - y_lo
        span_h = (n_lanes - 1) * lane_vert_gap
        span_h = min(span_h, usable_h)
        step_h = span_h / (n_lanes - 1) if n_lanes > 1 else 0.0
        x_start = rng.uniform(lo + 15, lo + 50)
        x_end = rng.uniform(hi - 50, hi - 15)
        path = []
        for lane in range(n_lanes):
            y = y_lo + lane * step_h
            if lane % 2 == 0:
                path.append((x_start, y))
                path.append((x_end, y))
            else:
                path.append((x_end, y))
                path.append((x_start, y))
        path = _pts(path)
        total = _path_length(path)
        if total < END_MARGIN * 5:
            continue

        s_port = END_MARGIN * rng.uniform(1.0, 1.4)
        lo_s = s_port + ARC_SEP * 2
        hi_s = total - END_MARGIN
        if hi_s - lo_s < 4 * ARC_SEP:
            continue
        # three on-path positions, increasing arc distance from the ring.
        t1 = rng.uniform(lo_s, lo_s + (hi_s - lo_s) * 0.25)
        t2 = t1 + rng.uniform(ARC_SEP, ARC_SEP + (hi_s - t1) * 0.45)
        t3 = t2 + rng.uniform(ARC_SEP, max(ARC_SEP, hi_s - t2))
        if t3 > hi_s:
            continue
        arc_positions = [s_port, t1, t2, t3]
        arc_positions.sort()
        ok = all(b - a >= ARC_SEP for a, b in zip(arc_positions, arc_positions[1:]))
        if not ok:
            continue

        # Assign the three colours to the three on-path positions at random, so
        # any colour can be the nearest across scenes.
        colors = TERMINAL_COLORS[:]
        rng.shuffle(colors)
        order = sorted([t1, t2, t3])
        marks = [
            {"x": _point_at(path, order[0])[0], "y": _point_at(path, order[0])[1], "color": colors[0]},
            {"x": _point_at(path, order[1])[0], "y": _point_at(path, order[1])[1], "color": colors[1]},
            {"x": _point_at(path, order[2])[0], "y": _point_at(path, order[2])[1], "color": colors[2]},
        ]
        rng.shuffle(marks)
        port = _point_at(path, s_port)

        allgood = True
        for m in marks + [{"x": port[0], "y": port[1]}]:
            if not (20 < m["x"] < w - 20 and 20 < m["y"] < h - 20):
                allgood = False
                break
        if not allgood:
            continue

        return {
            "path": path,
            "port": [float(port[0]), float(port[1])],
            "marks": marks,
        }
    raise RuntimeError("could not satisfy path constraints")


def sample_scene(seed):
    rng = random.Random(seed)
    return _rejection_loop(rng)


def _draw_polyline(draw, pts, color, width):
    for a, b in zip(pts, pts[1:]):
        draw.line([a, b], fill=color, width=width, joint="curve")


def render(scene):
    img = Image.new("RGB", CANVAS, BG)
    draw = ImageDraw.Draw(img)
    _draw_polyline(draw, _pts(scene["path"]), COLOR_BLUE, STROKE)
    for mark in scene["marks"]:
        r = MARK_R
        px, py = float(mark["x"]), float(mark["y"])
        draw.ellipse([px - r, py - r, px + r, py + r], fill=MARK_COLORS[mark["color"]])
    px, py = float(scene["port"][0]), float(scene["port"][1])
    draw.ellipse([px - PORT_R, py - PORT_R, px + PORT_R, py + PORT_R], fill=COLOR_ORANGE)
    return img


def _arc_position(scene, pt):
    best = None
    best_d = float("inf")
    cum = 0.0
    for a, b in zip(_pts(scene["path"]), _pts(scene["path"])[1:]):
        abx, aby = b[0] - a[0], b[1] - a[1]
        seg2 = abx * abx + aby * aby
        t = 0.0 if seg2 == 0 else ((pt[0] - a[0]) * abx + (pt[1] - a[1]) * aby) / seg2
        t = max(0.0, min(1.0, t))
        proj = (a[0] + abx * t, a[1] + aby * t)
        d = math.dist(pt, proj)
        if d < best_d:
            best_d = d
            best = cum + math.dist(a, proj)
        cum += math.dist(a, b)
    return best


def analytic_gold(scene):
    """Colour of the terminal with smallest on-path distance from the ring."""
    s_port = _arc_position(scene, (scene["port"][0], scene["port"][1]))
    best = None
    best_d = None
    for m in scene["marks"]:
        d = _arc_position(scene, (m["x"], m["y"])) - s_port
        if best_d is None or d < best_d:
            best_d = d
            best = m["color"]
    return best


def margin(scene):
    """Minimum on-path arc gap between consecutive markers (px)."""
    pos = [scene["port"]] + [[m["x"], m["y"]] for m in scene["marks"]]
    arc = sorted(_arc_position(scene, (p[0], p[1])) for p in pos)
    return min(b - a for a, b in zip(arc, arc[1:]))


def is_quarantined(scene):
    w, h = CANVAS
    if margin(scene) < ARC_SEP * 0.8:
        return True
    for m in scene["marks"] + [{"x": scene["port"][0], "y": scene["port"][1]}]:
        if not (20 <= m["x"] <= w - 20 and 20 <= m["y"] <= h - 20):
            return True
    return False


def latent_symmetries(scene):
    return []
