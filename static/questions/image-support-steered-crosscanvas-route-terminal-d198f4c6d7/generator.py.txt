"""Cross-canvas route-terminal correspondence world.

One thin, dark, CONNECTED trace begins at a black source dot on the far LEFT of
the canvas, winds continuously across almost the whole image, and ends at
exactly one of three coloured terminal dots on the far RIGHT. The question asks
which colour that single trace reaches; answering requires FOLLOWING the one
connected thin route from its left end all the way to its right end, because the
identity of the reached terminal is not legible from the source side alone nor
from the terminal side alone.

The decisive pixel support is the single thin connected trace spanning widely
separated parts of the canvas (far left source -> far right terminals): a crop
holding only the left half cannot see which terminal is reached, and a crop
holding only the right half cannot know that the reached terminal belongs to the
same continuous route as the source. The pixel oracle follows the connected
component from the source and abstains whenever the thin route is severed, so
every distinguishing region is jointly necessary.

Latent state z includes label `target_color` (the colour of the reached
terminal), determined only by the label field; position/slot assignment and the
winding are painted directly into the raster. Mode `not_applicable`: no
nontrivial pixel-identical transform, injective over its label input.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "crosscanvas_route_terminal-0.1.0"

CANVAS = 512
TRACE_WIDTH = 6
SOURCE_RADIUS = 14
TERMINAL_RADIUS = 17

WHITE = (255, 255, 255)
GRAPHITE = (72, 72, 72)
SOURCE_COLOR = (20, 20, 20)

PALETTE: dict[str, tuple[int, int, int]] = {
    "coral": (232, 93, 77),
    "teal": (31, 158, 141),
    "violet": (106, 81, 176),
}
PALETTE_NAMES = tuple(PALETTE)

SLOT_NAMES = ("top", "middle", "bottom")
SLOT_Y: dict[str, float] = {"top": 110.0, "middle": 256.0, "bottom": 402.0}
SOURCE_X = 48.0
TERMINAL_X = 462.0
SLOT_GAP = 146.0  # vertical separation between adjacent terminal slots


def _target_slot(scene: dict) -> str:
    target = scene["target_color"]
    for slot in SLOT_NAMES:
        if scene[f"slot_{slot}"] == target:
            return slot
    raise ValueError("target_color not present in any slot")


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    # Deterministically rotate the target colour across the scene stream so every
    # dataset unquestionably carries all three decision classes.
    target_color = PALETTE_NAMES[seed % len(PALETTE_NAMES)]
    target_slot = rng.choice(SLOT_NAMES)
    others = [name for name in PALETTE_NAMES if name != target_color]
    rng.shuffle(others)
    slot_colors = {}
    others_iter = iter(others)
    for slot in SLOT_NAMES:
        slot_colors[slot] = target_color if slot == target_slot else next(others_iter)
    return {
        "target_color": target_color,
        "slot_top": slot_colors["top"],
        "slot_middle": slot_colors["middle"],
        "slot_bottom": slot_colors["bottom"],
        "source_y": float(rng.uniform(140.0, 372.0)),
    }


def _waypoints(scene: dict) -> list[tuple[float, float]]:
    target_y = SLOT_Y[_target_slot(scene)]
    sy = scene["source_y"]
    n = 9
    xs = [SOURCE_X + (TERMINAL_X - SOURCE_X) * k / (n - 1) for k in range(n)]
    amp = 14.0
    pts = []
    for k in range(n):
        t = k / (n - 1)
        base = sy + (target_y - sy) * t
        if 0 < k < n - 1:
            base += amp * math.sin(1.7 * k)
        pts.append((xs[k], base))
    pts[-1] = (TERMINAL_X, target_y)
    return pts


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (CANVAS, CANVAS), WHITE)
    draw = ImageDraw.Draw(img)
    pts = _waypoints(scene)
    # Draw the whole connected thin trace as one continuous polyline (its ends
    # will be covered by the source/terminal discs so it stays visually joined).
    for a, b in zip(pts, pts[1:]):
        draw.line([a, b], fill=GRAPHITE, width=TRACE_WIDTH)
    # Source disc on the far left.
    draw.ellipse(
        [
            SOURCE_X - SOURCE_RADIUS,
            scene["source_y"] - SOURCE_RADIUS,
            SOURCE_X + SOURCE_RADIUS,
            scene["source_y"] + SOURCE_RADIUS,
        ],
        fill=SOURCE_COLOR,
    )
    # Terminal discs on the far right.
    for slot in SLOT_NAMES:
        cy = SLOT_Y[slot]
        draw.ellipse(
            [
                TERMINAL_X - TERMINAL_RADIUS,
                cy - TERMINAL_RADIUS,
                TERMINAL_X + TERMINAL_RADIUS,
                cy + TERMINAL_RADIUS,
            ],
            fill=PALETTE[scene[f"slot_{slot}"]],
        )
    return img


def analytic_gold(scene: dict) -> str:
    return scene["target_color"]


def margin(scene: dict) -> float:
    # The selection margin is the clear vertical separation between the reached
    # terminal slot and its nearest decoy slot, which stays crisp.
    return SLOT_GAP


def is_quarantined(scene: dict) -> bool:
    # Terminal slots are fixed and evenly spaced by 146px and the thin trace
    # converges onto the reached slot, never grazing a decoy; every scene is a
    # clean off-quarantine class.
    return False


def latent_symmetries(scene: dict) -> list:
    return []


def scene_targets(scene: dict) -> dict:
    return {
        "target_color": scene["target_color"],
        "target_slot": _target_slot(scene),
        "slot_top": scene["slot_top"],
        "slot_middle": scene["slot_middle"],
        "slot_bottom": scene["slot_bottom"],
        "source_y": scene["source_y"],
    }
