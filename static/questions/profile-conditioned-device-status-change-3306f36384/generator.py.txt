"""Deterministic two-panel device-state world."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont


WIDTH = 960
HEIGHT = 560
PIP_COLOR = (36, 156, 86)
INK = (31, 39, 52)
PANEL_FILL = (244, 247, 250)
CARD_FILL = (255, 255, 255)
PALETTES = (
    ((42, 111, 219), (236, 126, 46)),
    ((113, 78, 190), (228, 177, 38)),
    ((25, 145, 157), (214, 72, 118)),
)


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed) * 7919 + 104729)
    changed = int(seed) % 4 + 1
    flip_direction = (int(seed) // 4) % 2

    others = [index for index in range(4) if index != changed - 1]
    rng.shuffle(others)
    before_states = [0, 0, 0, 0]
    before_states[changed - 1] = flip_direction
    # Exactly two lamps begin in each state, while the changed lamp direction
    # alternates independently of its identity.
    before_states[others[0]] = flip_direction
    before_states[others[1]] = 1 - flip_direction
    before_states[others[2]] = 1 - flip_direction
    after_states = list(before_states)
    after_states[changed - 1] = 1 - after_states[changed - 1]

    before_order = [1, 2, 3, 4]
    after_order = [1, 2, 3, 4]
    rng.shuffle(before_order)
    rng.shuffle(after_order)
    if after_order == before_order:
        after_order = after_order[1:] + after_order[:1]

    return {
        "before_order": before_order,
        "after_order": after_order,
        "before_states": before_states,
        "after_states": after_states,
        "palette_mode": int(seed) % len(PALETTES),
        "x_offset": (-12, 0, 12)[(int(seed) // 3) % 3],
        "y_offset": (-5, 0, 5)[(int(seed) // 7) % 3],
        "lamp_radius": (18, 22, 26)[(int(seed) // 5) % 3],
    }


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    title: str,
    order: list[int],
    states: list[int],
    palette: tuple[tuple[int, int, int], tuple[int, int, int]],
    panel_top: int,
    x_offset: int,
    lamp_radius: int,
) -> None:
    left, right = 42 + x_offset, 918 + x_offset
    draw.rounded_rectangle(
        (left, panel_top, right, panel_top + 208),
        radius=18,
        fill=PANEL_FILL,
        outline=(173, 183, 196),
        width=3,
    )
    draw.text((left + 18, panel_top + 12), title, fill=INK, font=_font(24, True))
    centers = [151 + x_offset, 371 + x_offset, 591 + x_offset, 811 + x_offset]
    for slot, identity in enumerate(order):
        cx = centers[slot]
        top = panel_top + 50
        draw.rounded_rectangle(
            (cx - 82, top, cx + 82, top + 142),
            radius=14,
            fill=CARD_FILL,
            outline=(91, 103, 119),
            width=3,
        )
        # The visible identity is only the count of green pips.
        gap = 21
        start_x = cx - ((identity - 1) * gap) // 2
        for pip in range(identity):
            px = start_x + pip * gap
            py = top + 35
            draw.ellipse((px - 6, py - 6, px + 6, py + 6), fill=PIP_COLOR)
        draw.line((cx - 52, top + 60, cx + 52, top + 60), fill=(205, 211, 219), width=2)
        cy = top + 101
        draw.ellipse(
            (cx - lamp_radius - 3, cy - lamp_radius - 3,
             cx + lamp_radius + 3, cy + lamp_radius + 3),
            fill=INK,
        )
        draw.ellipse(
            (cx - lamp_radius, cy - lamp_radius,
             cx + lamp_radius, cy + lamp_radius),
            fill=palette[states[identity - 1]],
        )


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (255, 255, 255))
    draw = ImageDraw.Draw(image)
    palette = PALETTES[int(scene["palette_mode"])]
    x_offset = int(scene["x_offset"])
    y_offset = int(scene["y_offset"])
    lamp_radius = int(scene["lamp_radius"])
    _draw_panel(
        draw,
        "1  BEFORE",
        list(scene["before_order"]),
        list(scene["before_states"]),
        palette,
        42 + y_offset,
        x_offset,
        lamp_radius,
    )
    _draw_panel(
        draw,
        "2  AFTER",
        list(scene["after_order"]),
        list(scene["after_states"]),
        palette,
        306 + y_offset,
        x_offset,
        lamp_radius,
    )
    draw.text(
        (48 + x_offset, 268 + y_offset),
        "Identity = green-dot count     |     Find the one lamp that switched color",
        fill=INK,
        font=_font(18, True),
    )
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    before = list(scene["before_states"])
    after = list(scene["after_states"])
    changed = [index + 1 for index, (left, right) in enumerate(zip(before, after)) if left != right]
    return str(changed[0]) if len(changed) == 1 else "abstain"


def margin(scene: dict[str, Any]) -> float:
    # Positive clearance above the minimum lamp radius accepted by the inverse arm.
    return float(int(scene["lamp_radius"]) - 14)


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        before_order = list(scene["before_order"])
        after_order = list(scene["after_order"])
        before = list(scene["before_states"])
        after = list(scene["after_states"])
        valid_orders = sorted(before_order) == [1, 2, 3, 4] and sorted(after_order) == [1, 2, 3, 4]
        valid_states = len(before) == len(after) == 4 and set(before + after) <= {0, 1}
        one_change = sum(left != right for left, right in zip(before, after)) == 1
        return not (valid_orders and valid_states and one_change and margin(scene) > 0)
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
