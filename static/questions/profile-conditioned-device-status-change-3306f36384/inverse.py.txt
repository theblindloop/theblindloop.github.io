"""Independent inverse arm: reads only final RGB pixels."""

from __future__ import annotations

from collections import deque

from PIL import Image


PIP_COLOR = (36, 156, 86)
LAMP_COLORS = {
    (42, 111, 219),
    (236, 126, 46),
    (113, 78, 190),
    (228, 177, 38),
    (25, 145, 157),
    (214, 72, 118),
}


def _components(image: Image.Image, accepted: set[tuple[int, int, int]], minimum: int) -> list[dict]:
    rgb = image.convert("RGB")
    width, height = rgb.size
    pixels = rgb.load()
    selected = bytearray(width * height)
    for y in range(height):
        base = y * width
        for x in range(width):
            if pixels[x, y] in accepted:
                selected[base + x] = 1
    seen = bytearray(width * height)
    found: list[dict] = []
    for y0 in range(height):
        for x0 in range(width):
            start = y0 * width + x0
            if not selected[start] or seen[start]:
                continue
            queue = deque([(x0, y0)])
            seen[start] = 1
            count = sx = sy = 0
            colors: dict[tuple[int, int, int], int] = {}
            while queue:
                x, y = queue.popleft()
                color = pixels[x, y]
                colors[color] = colors.get(color, 0) + 1
                count += 1
                sx += x
                sy += y
                for nx, ny in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
                    if 0 <= nx < width and 0 <= ny < height:
                        index = ny * width + nx
                        if selected[index] and not seen[index]:
                            seen[index] = 1
                            queue.append((nx, ny))
            if count >= minimum:
                found.append(
                    {
                        "x": sx / count,
                        "y": sy / count,
                        "count": count,
                        "color": max(colors, key=colors.get),
                    }
                )
    return found


def decision_from_image(image: Image.Image) -> str:
    lamps = _components(image, LAMP_COLORS, 700)
    pips = _components(image, {PIP_COLOR}, 60)
    if len(lamps) != 8 or len(pips) != 20:
        return "abstain"

    lamps.sort(key=lambda item: item["y"])
    before_lamps = sorted(lamps[:4], key=lambda item: item["x"])
    after_lamps = sorted(lamps[4:], key=lambda item: item["x"])
    if before_lamps[-1]["y"] + 100 >= after_lamps[0]["y"]:
        return "abstain"

    def panel_map(group: list[dict]) -> dict[int, tuple[int, int, int]] | None:
        mapping: dict[int, tuple[int, int, int]] = {}
        for lamp in group:
            identity = sum(
                abs(pip["x"] - lamp["x"]) < 58
                and 55 < lamp["y"] - pip["y"] < 95
                for pip in pips
            )
            if identity not in (1, 2, 3, 4) or identity in mapping:
                return None
            mapping[identity] = lamp["color"]
        return mapping if len(mapping) == 4 else None

    before = panel_map(before_lamps)
    after = panel_map(after_lamps)
    if before is None or after is None:
        return "abstain"
    changed = [identity for identity in range(1, 5) if before[identity] != after[identity]]
    return str(changed[0]) if len(changed) == 1 else "abstain"
