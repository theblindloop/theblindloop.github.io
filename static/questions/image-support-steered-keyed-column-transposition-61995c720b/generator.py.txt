"""Deterministic renderer for the keyed column-transposition reveal world."""

from __future__ import annotations

import copy
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 768
HEIGHT = 512
BACKGROUND = (246, 243, 235)
FRAME = (91, 101, 117)
DARK = (30, 38, 52)
LIGHT = (220, 225, 219)
DECOR = (203, 204, 198)
TAG_COLORS: tuple[tuple[int, int, int], ...] = (
    (210, 58, 63),
    (232, 128, 40),
    (221, 184, 45),
    (54, 153, 91),
    (43, 158, 177),
    (54, 104, 196),
    (143, 82, 177),
)
SYMBOLS = ("arch", "fork", "zigzag")

_TEMPLATE_ROWS = {
    "arch": (
        "0111110",
        "1000001",
        "1000001",
        "1000001",
        "1000001",
        "1000001",
        "1000001",
    ),
    "fork": (
        "1001001",
        "1001001",
        "1001001",
        "1111111",
        "0001000",
        "0000000",
        "0000000",
    ),
    "zigzag": (
        "1111110",
        "0000010",
        "0000100",
        "0001000",
        "0010000",
        "0100000",
        "0111111",
    ),
}
TEMPLATES = {
    name: tuple(tuple(int(value) for value in row) for row in rows)
    for name, rows in _TEMPLATE_ROWS.items()
}


def _hamming(left: Iterable[Iterable[int]], right: Iterable[Iterable[int]]) -> int:
    return sum(a != b for row_a, row_b in zip(left, right) for a, b in zip(row_a, row_b))


def _variant_grid(symbol: str, difficulty: int) -> list[list[int]]:
    """Return an equal-ink exact, medium, or near-boundary glyph variant."""

    grid = [list(row) for row in TEMPLATES[symbol]]
    if difficulty <= 0:
        return grid
    competitor = {"arch": "fork", "fork": "arch", "zigzag": "arch"}[symbol]
    target = TEMPLATES[symbol]
    other = TEMPLATES[competitor]
    down = [
        (y, x)
        for y in range(7)
        for x in range(7)
        if target[y][x] == 1 and other[y][x] == 0
    ]
    up = [
        (y, x)
        for y in range(7)
        for x in range(7)
        if target[y][x] == 0 and other[y][x] == 1
    ]
    swap_count = 2 if difficulty == 1 else (5 if symbol == "zigzag" else 4)
    for y, x in down[:swap_count]:
        grid[y][x] = 0
    for y, x in up[:swap_count]:
        grid[y][x] = 1
    return grid


def _ordered_grid(scene: dict) -> tuple[tuple[int, ...], ...]:
    key_order = [int(value) for value in scene["key_order"]]
    payload_order = [int(value) for value in scene["payload_order"]]
    columns = [[int(bit) for bit in column] for column in scene["columns"]]
    if sorted(key_order) != list(range(7)) or sorted(payload_order) != list(range(7)):
        raise ValueError("key and payload orders must each contain the seven tags")
    if len(columns) != 7 or any(len(column) != 7 for column in columns):
        raise ValueError("payload must contain seven columns of seven cells")
    by_tag = {tag: column for tag, column in zip(payload_order, columns)}
    ordered_columns = [by_tag[tag] for tag in key_order]
    return tuple(tuple(ordered_columns[x][y] for x in range(7)) for y in range(7))


def _classification(scene: dict) -> tuple[str, int, int]:
    grid = _ordered_grid(scene)
    ranked = sorted((_hamming(grid, template), name) for name, template in TEMPLATES.items())
    best_distance, best_name = ranked[0]
    runner_distance = ranked[1][0]
    if runner_distance == best_distance:
        return "ambiguous", best_distance, 0
    return best_name, best_distance, runner_distance - best_distance


def sample_scene(seed: int) -> dict:
    seed = int(seed)
    rng = random.Random(seed ^ 0x5A17C0DE)
    symbol = SYMBOLS[seed % len(SYMBOLS)]
    difficulty = (seed // len(SYMBOLS)) % 3
    grid = _variant_grid(symbol, difficulty)

    key_order = list(range(7))
    payload_order = list(range(7))
    rng.shuffle(key_order)
    rng.shuffle(payload_order)
    target_columns = [[grid[y][x] for y in range(7)] for x in range(7)]
    keyed_columns = {tag: target_columns[position] for position, tag in enumerate(key_order)}
    columns = [keyed_columns[tag] for tag in payload_order]

    decorations = []
    for index in range(3):
        decorations.append(
            [
                300 + index * 76 + rng.randint(-8, 8),
                92 + (index % 2) * 318 + rng.randint(-6, 6),
                4 + (index % 2),
            ]
        )
    return {
        "key_order": key_order,
        "payload_order": payload_order,
        "columns": columns,
        "scale": 11 + rng.randrange(2),
        "x_shift": rng.randint(-3, 3),
        "y_shift": rng.randint(-4, 4),
        "decorations": decorations,
    }


def analytic_gold(scene: dict) -> str:
    return _classification(scene)[0]


def margin(scene: dict) -> float:
    return float(_classification(scene)[2])


def is_quarantined(scene: dict) -> bool:
    return analytic_gold(scene) == "ambiguous" or margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["decorations"] = list(reversed(twin["decorations"]))
    return [("reverse_nonoverlapping_decorations", twin)]


def _rect(draw: ImageDraw.ImageDraw, x: int, y: int, width: int, height: int, fill) -> None:
    draw.rectangle((x, y, x + width - 1, y + height - 1), fill=fill)


def render(scene: dict) -> Image.Image:
    key_order = [int(value) for value in scene["key_order"]]
    payload_order = [int(value) for value in scene["payload_order"]]
    columns = [[int(bit) for bit in column] for column in scene["columns"]]
    scale = int(scene["scale"])
    x_shift = int(scene["x_shift"])
    y_shift = int(scene["y_shift"])
    decorations = scene["decorations"]
    if scale not in (11, 12):
        raise ValueError("scale must be 11 or 12")

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x, y, radius in decorations:
        draw.ellipse(
            (int(x) - int(radius), int(y) - int(radius), int(x) + int(radius), int(y) + int(radius)),
            fill=DECOR,
        )

    chip_x = 64 + x_shift
    chip_y = 166 + y_shift
    chip_width = 44 + 2 * (scale - 11)
    chip_height = 14
    chip_pitch = 25
    key_left = chip_x - 10
    key_top = chip_y - 18
    key_right = chip_x + chip_width + 9
    key_bottom = chip_y + 6 * chip_pitch + chip_height + 9
    draw.rectangle((key_left, key_top, key_right, key_bottom), outline=FRAME, width=3)
    notch_x = (key_left + key_right) // 2
    draw.polygon(
        ((notch_x - 7, key_top), (notch_x + 7, key_top), (notch_x, key_top + 9)),
        fill=FRAME,
    )
    for position, tag in enumerate(key_order):
        _rect(draw, chip_x, chip_y + position * chip_pitch, chip_width, chip_height, TAG_COLORS[tag])

    side = scale
    gap = 3
    pitch = side + gap
    grid_x = 616 + x_shift
    header_y = 174 + y_shift
    grid_y = header_y + side + 14
    grid_width = 7 * side + 6 * gap
    panel_left = grid_x - 10
    panel_top = header_y - 10
    panel_right = grid_x + grid_width + 9
    panel_bottom = grid_y + grid_width + 9
    draw.rectangle((panel_left, panel_top, panel_right, panel_bottom), outline=FRAME, width=3)
    for x_index, tag in enumerate(payload_order):
        x = grid_x + x_index * pitch
        _rect(draw, x, header_y, side, side, TAG_COLORS[tag])
        column = columns[x_index]
        if len(column) != 7:
            raise ValueError("each payload column must contain seven cells")
        for y_index, bit in enumerate(column):
            _rect(draw, x, grid_y + y_index * pitch, side, side, DARK if bit else LIGHT)
    return image
