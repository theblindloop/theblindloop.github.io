"""Renderer and analytic answer for pairwise state-change synchrony."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont


CANVAS = (1470, 560)
PANEL_COUNT = 6
PANEL_X0 = 18
PANEL_Y = 112
PANEL_W = 228
PANEL_H = 410
PANEL_GAP = 10
CARD_X_OFFSET = 9
CARD_Y_OFFSET = 55
CARD_W = 210
ROW_GAP = 8
TILE_HEIGHT = 62
BADGE_SIZE = 40
BADGE_X_OFFSET = 10
GLYPH_SCALE = 5
GLYPH_X_OFFSET = 12
GLYPH_Y_OFFSET = 10
STATE_CENTER_X_OFFSET = 164
STATE_TEXT_X_OFFSET = 181
KEY_X = 520
KEY_Y = 21
KEY_SWATCH_OFFSET = 82
KEY_SWATCH_SIDE = 26
LABELS = ("A", "B", "C", "D")
PAIRS = ("AB", "AC", "AD", "BC", "BD", "CD")
STATE_COUNT = 3
STATE_NAMES = ("LOW", "MID", "HIGH")

BACKGROUND = (247, 245, 239)
PANEL_FILL = (255, 253, 248)
CARD_FILL = (255, 255, 255)
CARD_BORDER = (201, 205, 211)
TEXT_FILL = (39, 45, 54)
MUTED_FILL = (92, 100, 111)
BADGE_FILL = (43, 52, 67)
GLYPH_FILL = (245, 247, 250)

PALETTE_OPTIONS = (
    ((45, 117, 190), (218, 140, 43), (190, 70, 91)),
    ((47, 145, 107), (178, 80, 153), (211, 113, 42)),
    ((44, 125, 140), (198, 83, 71), (119, 91, 171)),
)
PANEL_TINTS = ((84, 102, 122), (112, 91, 128), (73, 116, 102), (139, 103, 62))

GLYPHS = {
    "A": ("010", "101", "111", "101", "101"),
    "B": ("110", "101", "110", "101", "110"),
    "C": ("011", "100", "100", "100", "011"),
    "D": ("110", "101", "101", "101", "110"),
}


def _font(size: int):
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _centered_text(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font, fill) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text(
        (xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2),
        text,
        font=font,
        fill=fill,
    )


def _pair_scores(trajectories: dict[str, list[int]]) -> dict[str, int]:
    scores: dict[str, int] = {}
    for pair in PAIRS:
        left, right = pair
        scores[pair] = sum(
            trajectories[left][step] != trajectories[left][step - 1]
            and trajectories[right][step] != trajectories[right][step - 1]
            for step in range(1, PANEL_COUNT)
        )
    return scores


def _sample_trajectories(rng: random.Random, desired: str, gap: int) -> dict[str, list[int]]:
    for _ in range(120_000):
        trajectories = {
            label: [rng.randrange(STATE_COUNT) for _ in range(PANEL_COUNT)] for label in LABELS
        }
        scores = _pair_scores(trajectories)
        ranking = sorted(scores.items(), key=lambda item: (-item[1], item[0]))
        if ranking[0][0] == desired and ranking[0][1] - ranking[1][1] == gap:
            return trajectories
    raise RuntimeError(f"could not sample pair winner {desired!r} with gap {gap}")


def sample_scene(seed: int) -> dict:
    """Sample a balanced six-panel scene with one pairwise co-change winner."""
    rng = random.Random(int(seed))
    desired = PAIRS[int(seed) % len(PAIRS)]
    gap = 1 if int(seed) % 5 == 0 or int(seed) % 7 == 0 else rng.choice((2, 3))
    trajectories = _sample_trajectories(rng, desired, gap)
    rows: list[list[str]] = []
    for _ in range(PANEL_COUNT):
        row = list(LABELS)
        rng.shuffle(row)
        rows.append(row)

    base_palette = PALETTE_OPTIONS[rng.randrange(len(PALETTE_OPTIONS))]
    order = list(range(STATE_COUNT))
    rng.shuffle(order)
    state_palette = [list(base_palette[index]) for index in order]
    state_names = [STATE_NAMES[index] for index in order]
    return {
        "width": CANVAS[0],
        "height": CANVAS[1],
        "panel_count": PANEL_COUNT,
        "panel_order": [f"P{index + 1}" for index in range(PANEL_COUNT)],
        "panel_rows": rows,
        "trajectories": trajectories,
        "state_palette": state_palette,
        "state_names": state_names,
        "swatch_side": rng.choice((24, 26, 28, 30)),
        "tile_height": TILE_HEIGHT,
        "panel_tints": [list(PANEL_TINTS[rng.randrange(len(PANEL_TINTS))]) for _ in range(PANEL_COUNT)],
    }


def _draw_identity_badge(draw: ImageDraw.ImageDraw, left: int, top: int, label: str) -> None:
    draw.rectangle([left, top, left + BADGE_SIZE - 1, top + BADGE_SIZE - 1], fill=BADGE_FILL)
    for row_index, row in enumerate(GLYPHS[label]):
        for col_index, bit in enumerate(row):
            if bit == "1":
                x0 = left + GLYPH_X_OFFSET + col_index * GLYPH_SCALE
                y0 = top + GLYPH_Y_OFFSET + row_index * GLYPH_SCALE
                draw.rectangle([x0, y0, x0 + GLYPH_SCALE - 1, y0 + GLYPH_SCALE - 1], fill=GLYPH_FILL)


def render(scene: dict) -> Image.Image:
    """Render the visible identity/state board."""
    width = int(scene["width"])
    height = int(scene["height"])
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(21)
    panel_font = _font(15)
    small_font = _font(10)

    draw.text((18, 16), "STATE SYNCHRONY BOARD", font=title_font, fill=TEXT_FILL)
    draw.text((18, 49), "Count adjacent transitions where both labels change", font=small_font, fill=MUTED_FILL)
    draw.text((KEY_X, 4), "STATE KEY", font=small_font, fill=MUTED_FILL)
    for index, (color, name) in enumerate(zip(scene["state_palette"], scene["state_names"])):
        sx = KEY_X + KEY_SWATCH_OFFSET + index * 150
        draw.rectangle(
            [sx, KEY_Y, sx + KEY_SWATCH_SIDE - 1, KEY_Y + KEY_SWATCH_SIDE - 1],
            fill=tuple(color),
            outline=(70, 79, 92),
        )
        draw.text((sx + 34, KEY_Y + 5), str(name), font=small_font, fill=TEXT_FILL)

    for panel_index in range(int(scene["panel_count"])):
        panel_x = PANEL_X0 + panel_index * (PANEL_W + PANEL_GAP)
        panel_tint = tuple(scene["panel_tints"][panel_index])
        draw.rectangle(
            [panel_x, PANEL_Y, panel_x + PANEL_W, PANEL_Y + PANEL_H],
            fill=PANEL_FILL,
            outline=panel_tint,
            width=2,
        )
        _centered_text(
            draw,
            (panel_x + PANEL_W / 2, PANEL_Y + 24),
            str(scene["panel_order"][panel_index]),
            panel_font,
            TEXT_FILL,
        )
        draw.line(
            [panel_x + 12, PANEL_Y + 43, panel_x + PANEL_W - 12, PANEL_Y + 43],
            fill=(221, 222, 220),
            width=1,
        )
        for row_index, label in enumerate(scene["panel_rows"][panel_index]):
            card_x = panel_x + CARD_X_OFFSET
            card_y = PANEL_Y + CARD_Y_OFFSET + row_index * (int(scene["tile_height"]) + ROW_GAP)
            draw.rectangle(
                [card_x, card_y, card_x + CARD_W - 1, card_y + int(scene["tile_height"]) - 1],
                fill=CARD_FILL,
                outline=CARD_BORDER,
                width=1,
            )
            badge_top = card_y + (int(scene["tile_height"]) - BADGE_SIZE) // 2
            _draw_identity_badge(draw, card_x + BADGE_X_OFFSET, badge_top, label)
            state_index = int(scene["trajectories"][label][panel_index])
            center_x = panel_x + STATE_CENTER_X_OFFSET
            center_y = card_y + int(scene["tile_height"]) // 2
            half = int(scene["swatch_side"]) // 2
            draw.rectangle(
                [center_x - half, center_y - half, center_x + half - 1, center_y + half - 1],
                fill=tuple(scene["state_palette"][state_index]),
                outline=panel_tint,
            )
            draw.text(
                (panel_x + STATE_TEXT_X_OFFSET, center_y - 6),
                str(scene["state_names"][state_index]),
                font=small_font,
                fill=MUTED_FILL,
            )

        if panel_index < int(scene["panel_count"]) - 1:
            arrow_y = PANEL_Y + PANEL_H // 2
            start_x = panel_x + PANEL_W + 2
            end_x = panel_x + PANEL_W + PANEL_GAP - 2
            draw.line([start_x, arrow_y, end_x, arrow_y], fill=MUTED_FILL, width=2)
            draw.polygon(
                [(end_x, arrow_y), (end_x - 6, arrow_y - 4), (end_x - 6, arrow_y + 4)],
                fill=MUTED_FILL,
            )
    return image


def analytic_gold(scene: dict) -> str:
    """Select the unique label pair with the most simultaneous changes."""
    trajectories = scene["trajectories"]
    scores = _pair_scores(trajectories)
    ranking = sorted(scores.items(), key=lambda item: (-item[1], item[0]))
    if ranking[0][1] == ranking[1][1]:
        raise ValueError("ambiguous pairwise synchrony")
    return ranking[0][0]


def margin(scene: dict) -> float:
    scores = sorted(_pair_scores(scene["trajectories"]).values(), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent answer-bearing aliases are claimed for this labelled board."""
    return []
