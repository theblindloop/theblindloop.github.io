"""Independent pixel-only inverse for the pairwise synchrony board."""

from __future__ import annotations

from PIL import Image


PANEL_COUNT = 6
PANEL_X0 = 18
PANEL_Y = 112
PANEL_W = 228
PANEL_GAP = 10
CARD_X_OFFSET = 9
CARD_Y_OFFSET = 55
ROW_GAP = 8
TILE_HEIGHT = 62
BADGE_SIZE = 40
BADGE_X_OFFSET = 10
GLYPH_SCALE = 5
GLYPH_X_OFFSET = 12
GLYPH_Y_OFFSET = 10
STATE_CENTER_X_OFFSET = 164
KEY_X = 520
KEY_Y = 21
KEY_SWATCH_OFFSET = 82
KEY_SWATCH_SIDE = 26
LABELS = ("A", "B", "C", "D")
PAIRS = ("AB", "AC", "AD", "BC", "BD", "CD")

GLYPHS = {
    "A": ("010", "101", "111", "101", "101"),
    "B": ("110", "101", "110", "101", "110"),
    "C": ("011", "100", "100", "100", "011"),
    "D": ("110", "101", "101", "101", "110"),
}
PATTERN_TO_LABEL = {tuple(rows): label for label, rows in GLYPHS.items()}


def _glyph_from_pixels(pixels, left: int, top: int) -> str:
    pattern = []
    for row_index in range(5):
        bits = []
        for col_index in range(3):
            x = left + GLYPH_X_OFFSET + col_index * GLYPH_SCALE + GLYPH_SCALE // 2
            y = top + GLYPH_Y_OFFSET + row_index * GLYPH_SCALE + GLYPH_SCALE // 2
            bits.append("1" if sum(pixels[x, y]) > 500 else "0")
        pattern.append("".join(bits))
    key = tuple(pattern)
    if key not in PATTERN_TO_LABEL:
        raise ValueError(f"unresolved identity glyph {key!r}")
    return PATTERN_TO_LABEL[key]


def _nearest(rgb: tuple[int, int, int], palette: list[tuple[int, int, int]]) -> int:
    distances = [sum((int(a) - int(b)) ** 2 for a, b in zip(rgb, color)) for color in palette]
    best = min(range(len(distances)), key=distances.__getitem__)
    if sorted(distances)[1] - distances[best] < 25:
        raise ValueError("state swatch is too close to another keyed color")
    return best


def _pair_scores(trajectories: dict[str, list[int]]) -> dict[str, int]:
    scores = {}
    for pair in PAIRS:
        left, right = pair
        scores[pair] = sum(
            trajectories[left][step] != trajectories[left][step - 1]
            and trajectories[right][step] != trajectories[right][step - 1]
            for step in range(1, PANEL_COUNT)
        )
    return scores


def decision_from_image(image: Image.Image) -> str:
    """Recover all identity/state trajectories and select the pair from pixels alone."""
    raster = image.convert("RGB")
    if raster.width < PANEL_X0 + PANEL_COUNT * PANEL_W:
        raise ValueError(f"unexpected board width {raster.size}")
    pixels = raster.load()
    state_palette = []
    for index in range(3):
        x = KEY_X + KEY_SWATCH_OFFSET + index * 150 + KEY_SWATCH_SIDE // 2
        y = KEY_Y + KEY_SWATCH_SIDE // 2
        state_palette.append(tuple(pixels[x, y]))

    trajectories: dict[str, list[int]] = {label: [] for label in LABELS}
    for panel_index in range(PANEL_COUNT):
        panel_x = PANEL_X0 + panel_index * (PANEL_W + PANEL_GAP)
        seen: set[str] = set()
        for row_index in range(4):
            card_x = panel_x + CARD_X_OFFSET
            card_y = PANEL_Y + CARD_Y_OFFSET + row_index * (TILE_HEIGHT + ROW_GAP)
            badge_left = card_x + BADGE_X_OFFSET
            badge_top = card_y + (TILE_HEIGHT - BADGE_SIZE) // 2
            label = _glyph_from_pixels(pixels, badge_left, badge_top)
            if label in seen:
                raise ValueError(f"duplicate identity {label} in panel {panel_index + 1}")
            seen.add(label)
            state_x = panel_x + STATE_CENTER_X_OFFSET
            state_y = card_y + TILE_HEIGHT // 2
            trajectories[label].append(_nearest(tuple(pixels[state_x, state_y]), state_palette))
        if seen != set(LABELS):
            raise ValueError(f"panel {panel_index + 1} does not contain A-D")

    scores = _pair_scores(trajectories)
    ranking = sorted(scores.items(), key=lambda item: (-item[1], item[0]))
    if ranking[0][1] == ranking[1][1]:
        raise ValueError("ambiguous pixel synchrony")
    return ranking[0][0]
