"""Independent pixel-only inverse for the state-recurrence world.

Only the public raster geometry and its visible legends are used here. This
module intentionally does not import the renderer, prompt, generator, verifier,
or any analytic-gold implementation.
"""

from __future__ import annotations

from PIL import Image


LABELS = ("A", "B", "C", "D")
STATE_COUNT = 3
PANEL_COUNT = 6
CANVAS = (1600, 700)
PANEL_X0 = 20
PANEL_STEP = 260
PANEL_TOP = 120
PANEL_W = 240
PANEL_H = 500
SLOT_CENTERS = ((64, 150), (176, 150), (64, 340), (176, 340))
ID_KEY_X = 310
ID_KEY_Y = 90
ID_KEY_STEP = 96
STATE_KEY_X = 790
STATE_KEY_Y = 90
STATE_KEY_STEP = 112


def _visible_key(frame: Image.Image) -> tuple[dict[tuple[int, int, int], str], dict[tuple[int, int, int], int]]:
    """Read the identity and state color maps from the visible legends."""

    pixels = frame.load()
    identity: dict[tuple[int, int, int], str] = {}
    states: dict[tuple[int, int, int], int] = {}
    for index, label in enumerate(LABELS):
        x = ID_KEY_X + index * ID_KEY_STEP
        color = pixels[x, ID_KEY_Y]
        if color in identity:
            raise ValueError("identity key reuses a color")
        identity[color] = label
    for state in range(STATE_COUNT):
        x = STATE_KEY_X + state * STATE_KEY_STEP
        color = pixels[x, STATE_KEY_Y]
        if color in states:
            raise ValueError("state key reuses a color")
        states[color] = state
    if len(identity) != 4 or len(states) != 3:
        raise ValueError("visible keys are incomplete")
    return identity, states


def _panel_trajectory(
    frame: Image.Image,
    identity_colors: dict[tuple[int, int, int], str],
    state_colors: dict[tuple[int, int, int], int],
    panel_index: int,
) -> dict[str, int]:
    """Read one public slot row using its stable outer and inner pixels."""

    pixels = frame.load()
    left = PANEL_X0 + panel_index * PANEL_STEP
    result: dict[str, int] = {}
    for slot_x, slot_y in SLOT_CENTERS:
        # The tile's left-middle outer band is identity color; its centered
        # inner marker is state color. Both locations are public layout rules,
        # not latent scene coordinates.
        identity_pixel = pixels[left + slot_x - 30, PANEL_TOP + slot_y]
        state_pixel = pixels[left + slot_x, PANEL_TOP + slot_y + 5]
        label = identity_colors.get(identity_pixel)
        state = state_colors.get(state_pixel)
        if label is None:
            raise ValueError(f"panel {panel_index + 1} has an unreadable identity tile")
        if state is None:
            raise ValueError(f"panel {panel_index + 1} has an unreadable state marker")
        if label in result:
            raise ValueError("duplicate identity in panel")
        result[label] = state
    if set(result) != set(LABELS):
        raise ValueError("panel identity set is incomplete")
    return result


def decision_from_image(image: Image.Image) -> str:
    """Decode all six visible panels and return the earliest valid reentry label."""

    frame = image.convert("RGB")
    if frame.size != CANVAS:
        raise ValueError(f"unexpected raster size {frame.size}")
    identity_colors, state_colors = _visible_key(frame)
    trajectories = {label: [] for label in LABELS}
    for panel_index in range(PANEL_COUNT):
        panel = _panel_trajectory(frame, identity_colors, state_colors, panel_index)
        for label in LABELS:
            trajectories[label].append(panel[label])

    returns: list[tuple[int, str]] = []
    for label in LABELS:
        initial = trajectories[label][0]
        first = None
        for index in range(2, PANEL_COUNT):
            if trajectories[label][index - 1] != initial and trajectories[label][index] == initial:
                first = index + 1
                break
        if first is None:
            raise ValueError(f"tile {label} has no visible return after departure")
        returns.append((first, label))
    earliest = min(value for value, _ in returns)
    winners = [label for value, label in returns if value == earliest]
    if len(winners) != 1:
        raise ValueError("first return is ambiguous")
    return winners[0]
