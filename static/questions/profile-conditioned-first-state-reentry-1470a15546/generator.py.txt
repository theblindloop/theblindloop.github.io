"""Deterministic renderer and analytic side of the state-recurrence world."""

from __future__ import annotations

import math
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont


LABELS = ("A", "B", "C", "D")
STATE_LABELS = ("S0", "S1", "S2")
PANEL_COUNT = 6
CANVAS = (1600, 700)
PANEL_X0 = 20
PANEL_STEP = 260
PANEL_TOP = 120
PANEL_W = 240
PANEL_H = 500
SLOT_CENTERS = ((64, 150), (176, 150), (64, 340), (176, 340))
TILE_W = 86
TILE_H = 118

BACKGROUND = (247, 249, 252)
PANEL_FILL = (255, 255, 255)
INK = (36, 45, 57)
MUTED = (93, 105, 119)
GRID = (199, 207, 216)
ARROW = (112, 124, 138)
ID_BASE_COLORS = (
    (211, 70, 83),
    (46, 113, 190),
    (45, 150, 100),
    (151, 85, 181),
)
STATE_BASE_COLORS = (
    (237, 151, 45),
    (30, 155, 169),
    (116, 88, 188),
)

ID_KEY_X = 310
ID_KEY_Y = 90
ID_KEY_STEP = 96
STATE_KEY_X = 790
STATE_KEY_Y = 90
STATE_KEY_STEP = 112

_FONT_PATHS = (
    "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
    "/usr/share/fonts/dejavu/DejaVuSans.ttf",
)


def _font(size: int):
    for candidate in _FONT_PATHS:
        if Path(candidate).is_file():
            return ImageFont.truetype(candidate, size=size)
    return ImageFont.load_default()


def _panel_left(index: int) -> int:
    return PANEL_X0 + index * PANEL_STEP


def _centered_text(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font, fill):
    box = draw.textbbox((0, 0), text, font=font)
    width = box[2] - box[0]
    height = box[3] - box[1]
    draw.text(
        (xy[0] - width / 2, xy[1] - height / 2 - box[1]),
        text,
        font=font,
        fill=fill,
    )


def _identity_color(scene: dict, label: str) -> tuple[int, int, int]:
    return ID_BASE_COLORS[int(scene["label_palette_order"][LABELS.index(label)])]


def _state_color(scene: dict, state: int) -> tuple[int, int, int]:
    return STATE_BASE_COLORS[int(scene["state_palette_order"][int(state)])]


def _draw_state_marker(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    state: int,
    color: tuple[int, int, int],
    radius: int,
) -> None:
    cx, cy = center
    if state == 0:
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=color, outline=INK, width=2)
    elif state == 1:
        draw.polygon(
            ((cx, cy - radius), (cx + radius, cy + radius), (cx - radius, cy + radius)),
            fill=color,
            outline=INK,
        )
    elif state == 2:
        draw.polygon(
            ((cx, cy - radius), (cx + radius, cy), (cx, cy + radius), (cx - radius, cy)),
            fill=color,
            outline=INK,
        )
    else:
        raise ValueError("state must be S0, S1, or S2")


def _draw_identity_key(draw: ImageDraw.ImageDraw, scene: dict, label_font, small_font) -> None:
    draw.text((220, 68), "IDENTITY", font=small_font, fill=MUTED)
    for index, label in enumerate(LABELS):
        cx = ID_KEY_X + index * ID_KEY_STEP
        draw.rounded_rectangle(
            (cx - 16, ID_KEY_Y - 12, cx + 16, ID_KEY_Y + 12),
            radius=5,
            fill=_identity_color(scene, label),
            outline=INK,
            width=1,
        )
        _centered_text(draw, (cx + 34, ID_KEY_Y), label, label_font, INK)


def _draw_state_key(draw: ImageDraw.ImageDraw, scene: dict, label_font, small_font) -> None:
    draw.text((710, 68), "STATE KEY", font=small_font, fill=MUTED)
    for state, label in enumerate(STATE_LABELS):
        cx = STATE_KEY_X + state * STATE_KEY_STEP
        _draw_state_marker(draw, (cx, STATE_KEY_Y), state, _state_color(scene, state), 13)
        _centered_text(draw, (cx + 35, STATE_KEY_Y), label, label_font, INK)


def sample_scene(seed: int) -> dict:
    """Sample a balanced six-panel scene with four unique return times."""

    logic_rng = random.Random(int(seed) * 1000003 + 17)
    style_rng = random.Random(int(seed) * 1000033 + 31)
    layout_rng = random.Random(int(seed) * 1000211 + 47)

    # The four return panels are P3, P4, P5, and P6. Their label order is
    # independently shuffled, so the winner is not a fixed panel position.
    return_order = logic_rng.sample(list(LABELS), len(LABELS))
    return_panel = {label: 3 + index for index, label in enumerate(return_order)}
    states: dict[str, list[int]] = {}
    for label in LABELS:
        initial = logic_rng.randrange(3)
        target_panel = return_panel[label]
        sequence = [initial]
        for panel_number in range(2, target_panel):
            choices = [state for state in range(3) if state != initial]
            sequence.append(logic_rng.choice(choices))
        sequence.append(initial)
        while len(sequence) < PANEL_COUNT:
            sequence.append(logic_rng.randrange(3))
        states[label] = sequence

    label_palette_order = style_rng.sample(list(range(4)), 4)
    state_palette_order = style_rng.sample(list(range(3)), 3)
    tile_size = style_rng.choice((88, 92, 96))
    corner_radius = style_rng.choice((8, 10, 12))
    # Small glyphs remain decodable by the pixel arm but are explicitly
    # quarantined as finite-resolution boundary examples.
    state_radius = 5 if int(seed) % 19 == 0 else style_rng.choice((13, 14, 15))
    panel_tile_order = [layout_rng.sample(list(LABELS), len(LABELS)) for _ in range(PANEL_COUNT)]

    return_times = []
    for label in LABELS:
        initial = states[label][0]
        for index in range(2, PANEL_COUNT):
            if states[label][index - 1] != initial and states[label][index] == initial:
                return_times.append(index + 1)
                break
    if len(return_times) != len(LABELS) or len(set(return_times)) != len(return_times):
        raise AssertionError("sampler did not create four unique return events")

    scene = {
        "canvas": list(CANVAS),
        "panel_count": PANEL_COUNT,
        "panel_order": list(range(1, PANEL_COUNT + 1)),
        "slot_centers": [list(center) for center in SLOT_CENTERS],
        "label_palette_order": label_palette_order,
        "state_palette_order": state_palette_order,
        "states": states,
        "panel_tile_order": panel_tile_order,
        "tile_size": tile_size,
        "corner_radius": corner_radius,
        "state_radius": state_radius,
        "runner_up_return_gap": float(max(return_times) - min(return_times)),
        "state_clearance": float(state_radius - 2),
    }
    if scene["runner_up_return_gap"] <= 0:
        raise AssertionError("return events must be separated")
    return scene


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", tuple(scene["canvas"]), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(23)
    panel_font = _font(16)
    label_font = _font(17)
    small_font = _font(12)
    tile_label_font = _font(16)

    draw.text((20, 18), "STATE REENTRY", font=title_font, fill=INK)
    draw.text(
        (20, 54),
        "A return is a state matching that tile's P1 state after a different state",
        font=small_font,
        fill=MUTED,
    )
    _draw_identity_key(draw, scene, label_font, small_font)
    _draw_state_key(draw, scene, label_font, small_font)

    for panel_index, panel_number in enumerate(scene["panel_order"]):
        left = _panel_left(panel_index)
        top = PANEL_TOP
        right = left + PANEL_W
        bottom = top + PANEL_H
        draw.rounded_rectangle(
            (left, top, right, bottom),
            radius=9,
            fill=PANEL_FILL,
            outline=INK,
            width=2,
        )
        draw.text((left + 10, top + 10), f"P{panel_number}", font=panel_font, fill=INK)
        if panel_index < PANEL_COUNT - 1:
            next_left = _panel_left(panel_index + 1)
            y = top - 15
            draw.line((right + 4, y, next_left - 7, y), fill=ARROW, width=2)
            draw.polygon(
                ((next_left - 7, y), (next_left - 14, y - 4), (next_left - 14, y + 4)),
                fill=ARROW,
            )

        for slot_index, label in enumerate(scene["panel_tile_order"][panel_index]):
            cx, cy = scene["slot_centers"][slot_index]
            cx += left
            cy += top
            half_w = scene["tile_size"] / 2
            half_h = TILE_H / 2
            outer = _identity_color(scene, label)
            draw.rounded_rectangle(
                (cx - half_w, cy - half_h, cx + half_w, cy + half_h),
                radius=scene["corner_radius"],
                fill=outer,
                outline=INK,
                width=2,
            )
            _centered_text(draw, (cx, cy - 38), label, tile_label_font, (255, 255, 255))
            state = scene["states"][label][panel_index]
            _draw_state_marker(
                draw,
                (cx, cy + 5),
                state,
                _state_color(scene, state),
                int(scene["state_radius"]),
            )
            _centered_text(draw, (cx, cy + 39), STATE_LABELS[state], small_font, (255, 255, 255))

    return image


def _first_return(sequence: list[int]) -> int | None:
    initial = int(sequence[0])
    for index in range(2, len(sequence)):
        if int(sequence[index - 1]) != initial and int(sequence[index]) == initial:
            return index + 1
    return None


def analytic_gold(scene: dict) -> str:
    returns = []
    for label in LABELS:
        first = _first_return(scene["states"][label])
        if first is None:
            raise ValueError(f"tile {label} never returns after a departure")
        returns.append((first, label))
    earliest = min(first for first, _ in returns)
    winners = [label for first, label in returns if first == earliest]
    if len(winners) != 1:
        raise ValueError("first return is tied")
    return winners[0]


def margin(scene: dict) -> float:
    """One-sided recoverability margin combining time and glyph clearance."""

    return float(min(5.0 * scene["runner_up_return_gap"], scene["state_clearance"]))


def is_quarantined(scene: dict) -> bool:
    return bool(scene["state_radius"] < 8 or scene["runner_up_return_gap"] <= 0)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # No unrendered record-order or gold-only field is used: all state-bearing
    # inputs and style nuisances are visible in the final raster.
    return []
