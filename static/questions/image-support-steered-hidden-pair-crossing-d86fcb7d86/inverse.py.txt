"""Independent inverse arm using only saturated components in the final PNG."""

from __future__ import annotations

import itertools

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[np.ndarray] = []
    for y0, x0 in np.argwhere(mask):
        if seen[y0, x0]:
            continue
        stack = [(int(y0), int(x0))]
        seen[y0, x0] = True
        points: list[tuple[int, int]] = []
        while stack:
            y, x = stack.pop()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= 120:
            groups.append(np.asarray(points, dtype=np.int32))
    return groups


def _color_distance(a: np.ndarray, b: np.ndarray) -> float:
    return float(np.linalg.norm(a.astype(float) - b.astype(float)))


def decision_from_image(image: Image.Image) -> str:
    """Recover color correspondences and compare their vertical orders."""
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    high = array.max(axis=2).astype(np.int16)
    low = array.min(axis=2).astype(np.int16)
    saturated = (high - low > 62) & (high > 115) & (low < 190)
    groups = _components(saturated)
    if len(groups) != 4:
        raise ValueError(f"expected four colored ports, found {len(groups)}")
    areas = np.asarray([len(group) for group in groups], dtype=float)
    if areas.max() / areas.min() > 1.30:
        raise ValueError("a colored port is missing or materially clipped")

    centers = np.asarray([[group[:, 1].mean(), group[:, 0].mean()] for group in groups])
    colors = np.asarray([np.median(array[group[:, 0], group[:, 1]], axis=0) for group in groups])
    x_order = np.argsort(centers[:, 0])
    left = list(map(int, x_order[:2]))
    right = list(map(int, x_order[2:]))
    if centers[right, 0].min() - centers[left, 0].max() < image.width * 0.20:
        raise ValueError("left and right port groups are not visibly separated")

    pairings = []
    for perm in itertools.permutations(right):
        score = sum(_color_distance(colors[i], colors[j]) for i, j in zip(left, perm))
        pairings.append((score, perm))
    pairings.sort(key=lambda item: item[0])
    if pairings[1][0] - pairings[0][0] < 45.0:
        raise ValueError("color correspondence is ambiguous")
    match = {left[i]: int(pairings[0][1][i]) for i in range(2)}
    left_top, left_bottom = sorted(left, key=lambda i: centers[i, 1])
    if centers[left_bottom, 1] - centers[left_top, 1] < image.height * 0.13:
        raise ValueError("left-side vertical order is ambiguous")
    right_top, right_bottom = sorted(right, key=lambda i: centers[i, 1])
    if centers[right_bottom, 1] - centers[right_top, 1] < image.height * 0.13:
        raise ValueError("right-side vertical order is ambiguous")
    return "uncrossed" if match[left_top] == right_top else "cross"
