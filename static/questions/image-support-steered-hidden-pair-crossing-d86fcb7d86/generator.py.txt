"""Deterministic visual world for completing two hidden matched connections."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS = 384
PALETTES = (
    ((218, 68, 83), (35, 123, 196)),
    ((239, 136, 38), (47, 142, 90)),
    ((157, 75, 190), (25, 153, 171)),
    ((208, 72, 145), (66, 119, 190)),
)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced scene without storing its answer as a latent field."""
    rng = random.Random(int(seed))
    width = height = CANVAS
    tx = rng.randint(-14, 14)
    ty = rng.randint(-12, 12)
    card_w = rng.randint(112, 132)
    card_h = rng.randint(190, 218)
    cx, cy = width // 2 + tx, height // 2 + ty
    card = [cx - card_w // 2, cy - card_h // 2, cx + card_w // 2, cy + card_h // 2]

    gap_left = rng.randint(72, 116)
    gap_right = rng.randint(72, 116)
    left_ys = [cy - gap_left // 2, cy + gap_left // 2]
    right_ys = [cy - gap_right // 2, cy + gap_right // 2]
    left_x = card[0] - rng.randint(25, 35)
    right_x = card[2] + rng.randint(25, 35)
    radius = rng.randint(13, 16)

    colors = list(PALETTES[rng.randrange(len(PALETTES))])
    if rng.randrange(2):
        colors.reverse()
    crossing = bool(int(seed) & 1)
    right_colors = list(reversed(colors)) if crossing else list(colors)
    ports = [
        {"side": "left", "center": [left_x, left_ys[i]], "radius": radius, "color": list(colors[i])}
        for i in range(2)
    ] + [
        {"side": "right", "center": [right_x, right_ys[i]], "radius": radius, "color": list(right_colors[i])}
        for i in range(2)
    ]
    # Draw order has no semantic role and is independently randomized.
    rng.shuffle(ports)
    return {
        "width": width,
        "height": height,
        "background": [249, 247, 242],
        "card": card,
        "ports": ports,
    }


def _pair_order(ports: list[dict[str, Any]]) -> tuple[list[tuple[int, int, int]], list[tuple[int, int, int]]]:
    left = sorted((p for p in ports if p["side"] == "left"), key=lambda p: p["center"][1])
    right = sorted((p for p in ports if p["side"] == "right"), key=lambda p: p["center"][1])
    if len(left) != 2 or len(right) != 2:
        raise ValueError("scene must contain two ports on each side")
    return [tuple(p["color"]) for p in left], [tuple(p["color"]) for p in right]


def analytic_gold(scene: dict[str, Any]) -> str:
    left, right = _pair_order(scene["ports"])
    if set(left) != set(right) or len(set(left)) != 2:
        raise ValueError("each of two colors must occur once on each side")
    return "uncrossed" if left == right else "cross"


def margin(scene: dict[str, Any]) -> float:
    ports = scene["ports"]
    left = sorted((p for p in ports if p["side"] == "left"), key=lambda p: p["center"][1])
    right = sorted((p for p in ports if p["side"] == "right"), key=lambda p: p["center"][1])
    dy_left = abs(left[0]["center"][1] - left[1]["center"][1])
    dy_right = abs(right[0]["center"][1] - right[1]["center"][1])
    colors = {tuple(p["color"]) for p in ports}
    if len(colors) != 2:
        return 0.0
    c1, c2 = colors
    color_clearance = sum((a - b) ** 2 for a, b in zip(c1, c2)) ** 0.5
    return float(min(dy_left, dy_right, color_clearance))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 52.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []


def render(scene: dict[str, Any]) -> Image.Image:
    scale = 4
    width, height = int(scene["width"]), int(scene["height"])
    background = tuple(scene["background"])
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    card = [int(v * scale) for v in scene["card"]]
    shadow = [card[0] + 5 * scale, card[1] + 6 * scale, card[2] + 5 * scale, card[3] + 6 * scale]
    draw.rounded_rectangle(shadow, radius=16 * scale, fill=(222, 218, 210))
    draw.rounded_rectangle(
        card,
        radius=16 * scale,
        fill=(238, 239, 237),
        outline=(186, 188, 185),
        width=2 * scale,
    )
    # A small neutral center mark reinforces that the card is an occluder, not a panel to read.
    cx = (card[0] + card[2]) // 2
    cy = (card[1] + card[3]) // 2
    draw.ellipse(
        [cx - 4 * scale, cy - 4 * scale, cx + 4 * scale, cy + 4 * scale],
        fill=(202, 203, 199),
    )
    for port in scene["ports"]:
        x, y = (int(v * scale) for v in port["center"])
        radius = int(port["radius"] * scale)
        halo = radius + 4 * scale
        draw.ellipse([x - halo, y - halo, x + halo, y + halo], fill=(82, 84, 88))
        draw.ellipse([x - radius, y - radius, x + radius, y + radius], fill=tuple(port["color"]))
        dot = 3 * scale
        draw.ellipse([x - dot, y - dot, x + dot, y + dot], fill=(249, 247, 242))
    return image.resize((width, height), Image.Resampling.LANCZOS)
