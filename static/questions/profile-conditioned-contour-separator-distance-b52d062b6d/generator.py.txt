def _inside(point: list[int], contour: list[int]) -> bool:
    x, y = point
    x0, y0, x1, y1 = contour
    return x0 < x < x1 and y0 < y < y1


def _separator_count(scene: dict) -> int:
    green = scene["green_dot"]
    purple = scene["purple_dot"]
    return sum(
        _inside(green, contour) != _inside(purple, contour)
        for contour in scene["contours"]
    )


def _mirror_point(point: list[int], mirror_x: bool, mirror_y: bool) -> list[int]:
    x, y = point
    return [WIDTH - 1 - x if mirror_x else x, HEIGHT - 1 - y if mirror_y else y]


def _mirror_rect(rect: list[int], mirror_x: bool, mirror_y: bool) -> list[int]:
    x0, y0, x1, y1 = rect
    if mirror_x:
        x0, x1 = WIDTH - 1 - x1, WIDTH - 1 - x0
    if mirror_y:
        y0, y1 = HEIGHT - 1 - y1, HEIGHT - 1 - y0
    return [x0, y0, x1, y1]


def _anchors(rectangles: list[list[int]]) -> dict[str, list[list[int]]]:
    left = rectangles[:3]
    right = rectangles[3:]
    ys = [170, 210, 250, 290]
    result: dict[str, list[list[int]]] = {
        "O": [[224, y] for y in ys],
    }
    for prefix, group in (("L", left), ("R", right)):
        outer, middle, inner = group
        # Pillow draws rectangle outlines inward.  Use the centres of the
        # *visible white bands*, with nine pixels reserved for the largest
        # sampled stroke, rather than the centres between latent edge lines.
        inset = 9
        result[f"{prefix}1"] = [
            [(outer[0] + inset + middle[0]) // 2, y] for y in ys
        ] + [[(middle[2] + outer[2] - inset) // 2, y] for y in ys]
        result[f"{prefix}2"] = [
            [(middle[0] + inset + inner[0]) // 2, y] for y in ys
        ] + [[(inner[2] + middle[2] - inset) // 2, y] for y in ys]
        center_x = (inner[0] + inner[2]) // 2
        result[f"{prefix}3"] = [[center_x, y] for y in ys]
    return result


def _signature(zone: str) -> frozenset[str]:
    if zone == "O":
        return frozenset()
    cluster, depth_text = zone[0], zone[1:]
    depth = int(depth_text)
    return frozenset(f"{cluster}{index}" for index in range(depth))


def sample_scene(seed: int) -> dict:
    """Sample a balanced, robust scene from an integer seed."""

    rng = random.Random(int(seed))
    left_dx = rng.randint(-3, 3)
    right_dx = rng.randint(-3, 3)
    left_dy = rng.randint(-4, 4)
    right_dy = rng.randint(-4, 4)
    contours = [
        [14 + left_dx, 30 + left_dy, 202 + left_dx, 418 + left_dy],
        [50 + left_dx, 70 + left_dy, 166 + left_dx, 378 + left_dy],
        [84 + left_dx, 112 + left_dy, 132 + left_dx, 336 + left_dy],
        [246 + right_dx, 30 + right_dy, 434 + right_dx, 418 + right_dy],
        [282 + right_dx, 70 + right_dy, 398 + right_dx, 378 + right_dy],
        [316 + right_dx, 112 + right_dy, 364 + right_dx, 336 + right_dy],
    ]

    # Alternating residues give 12 yes and 12 no scenes in every consecutive
    # block of 24, while the no class spans distances zero, one, and three.
    desired = (2, 0, 2, 1, 2, 3)[int(seed) % 6]
    anchor_map = _anchors(contours)
    zones = sorted(anchor_map)
    pairs = [
        (first, second)
        for first in zones
        for second in zones
        if len(_signature(first) ^ _signature(second)) == desired
    ]
    first_zone, second_zone = rng.choice(pairs)
    candidates = [
        (first, second)
        for first in anchor_map[first_zone]
        for second in anchor_map[second_zone]
        if math.dist(first, second) >= 36
    ]
    green, purple = rng.choice(candidates)
    if rng.random() < 0.5:
        green, purple = purple, green

    decorations = [[224, 22, 5], [224, 426, 5]]
    if rng.random() < 0.5:
        decorations.append([10, 224, 4])
    if rng.random() < 0.5:
        decorations.append([438, 224, 4])

    mirror_x = bool(rng.getrandbits(1))
    mirror_y = bool(rng.getrandbits(1))
    contours = [_mirror_rect(item, mirror_x, mirror_y) for item in contours]
    green = _mirror_point(green, mirror_x, mirror_y)
    purple = _mirror_point(purple, mirror_x, mirror_y)
    decorations = [
        [*_mirror_point(item[:2], mirror_x, mirror_y), item[2]]
        for item in decorations
    ]
    rng.shuffle(contours)
    rng.shuffle(decorations)

    scene = {
        "contours": contours,
        "green_dot": green,
        "purple_dot": purple,
        "width": WIDTH,
        "height": HEIGHT,
        "stroke_width": 8 + (int(seed) % 2),
        "palette": copy.deepcopy(PALETTES[int(seed) % len(PALETTES)]),
        "decorations": decorations,
    }
    assert _separator_count(scene) == desired
    assert not is_quarantined(scene)
    return scene


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    palette = scene["palette"]
    image = Image.new("RGB", (width, height), tuple(palette["background"]))
    draw = ImageDraw.Draw(image)
    for x, y, radius in scene["decorations"]:
        draw.ellipse(
            (x - radius, y - radius, x + radius, y + radius),
            fill=tuple(palette["decoration"]),
        )
    stroke_width = int(scene["stroke_width"])
    for contour in scene["contours"]:
        draw.rectangle(tuple(contour), outline=tuple(palette["ink"]), width=stroke_width)
    green_x, green_y = scene["green_dot"]
    draw.ellipse(
        (
            green_x - DOT_RADIUS,
            green_y - DOT_RADIUS,
            green_x + DOT_RADIUS,
            green_y + DOT_RADIUS,
        ),
        fill=tuple(palette["green"]),
    )
    purple_x, purple_y = scene["purple_dot"]
    draw.ellipse(
        (
            purple_x - DOT_RADIUS,
            purple_y - DOT_RADIUS,
            purple_x + DOT_RADIUS,
            purple_y + DOT_RADIUS,
        ),
        fill=tuple(palette["purple"]),
    )
    return image


def analytic_gold(scene: dict) -> str:
    return "yes" if _separator_count(scene) == 2 else "no"


def margin(scene: dict) -> float:
    count = _separator_count(scene)
    return float(1 if count == 2 else abs(count - 2))


def is_quarantined(scene: dict) -> bool:
    if int(scene["width"]) < 420 or int(scene["height"]) < 420:
        return True
    stroke = int(scene["stroke_width"])
    if stroke < 6:
        return True
    if len(scene["contours"]) < 4:
        return True
    for contour in scene["contours"]:
        x0, y0, x1, y1 = contour
        if min(x0, y0) < 4 or x1 >= scene["width"] - 4 or y1 >= scene["height"] - 4:
            return True
        if x1 - x0 < 5 * stroke or y1 - y0 < 5 * stroke:
            return True
    for point in (scene["green_dot"], scene["purple_dot"]):
        for x0, y0, x1, y1 in scene["contours"]:
            if min(abs(point[0] - x0), abs(point[0] - x1)) <= DOT_RADIUS + 2 and y0 <= point[1] <= y1:
                return True
            if min(abs(point[1] - y0), abs(point[1] - y1)) <= DOT_RADIUS + 2 and x0 <= point[0] <= x1:
                return True
    background = scene["palette"]["background"]
    ink = scene["palette"]["ink"]
    contrast = math.sqrt(sum((int(a) - int(b)) ** 2 for a, b in zip(background, ink)))
    return contrast < 180


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["contours"] = list(reversed(twin["contours"]))
    twin["decorations"] = list(reversed(twin["decorations"]))
    return [("reverse_unordered_contours_and_decorations", twin)]
