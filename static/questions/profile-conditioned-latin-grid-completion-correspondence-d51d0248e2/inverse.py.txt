"""Independent image-only inverse for Latin-grid completion."""

from __future__ import annotations

import numpy as np
from PIL import Image

PALETTE = {
    "ink": (15, 23, 42),
    "grid": (184, 196, 211),
    "magenta": (217, 70, 239),
    "red": (220, 38, 38),
    "green": (22, 163, 74),
    "blue": (37, 99, 235),
    "orange": (234, 88, 12),
}


def _mask(arr: np.ndarray, rgb: tuple[int, int, int], tolerance: int = 3) -> np.ndarray:
    delta = arr.astype(np.int16) - np.asarray(rgb, dtype=np.int16)
    return np.max(np.abs(delta), axis=2) <= tolerance


def _bbox(mask: np.ndarray) -> tuple[int, int, int, int]:
    ys, xs = np.nonzero(mask)
    if len(xs) == 0:
        raise ValueError("required visual marker is missing")
    return int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())


def _runs(values: np.ndarray) -> list[float]:
    indices = np.flatnonzero(values)
    if not len(indices):
        return []
    groups = [[int(indices[0])]]
    for value in indices[1:]:
        value = int(value)
        if value > groups[-1][-1] + 1:
            groups.append([value])
        else:
            groups[-1].append(value)
    return [sum(group) / len(group) for group in groups]


def _main_grid_lines(grid_mask: np.ndarray) -> tuple[list[float], list[float]]:
    # The only strokes spanning most of a 4x4 panel are its five guide lines.
    x_runs = _runs(grid_mask.sum(axis=0) > 380)
    if len(x_runs) != 5:
        raise ValueError(f"expected five long vertical grid lines, got {len(x_runs)}")
    x0, x1 = int(x_runs[0]), int(x_runs[-1])
    span = max(1, x1 - x0)
    y_runs = _runs(grid_mask[:, x0 : x1 + 1].sum(axis=1) > 0.72 * span)
    if len(y_runs) != 5:
        raise ValueError(f"expected five long horizontal grid lines, got {len(y_runs)}")
    return x_runs, y_runs


def _decode_symbol(arr: np.ndarray, box: tuple[float, float, float, float]) -> tuple[int, ...]:
    x0, y0, x1, y1 = box
    cx, cy = (x0 + x1) / 2.0, (y0 + y1) / 2.0
    pitch = max(12, int(min(x1 - x0, y1 - y0) * 0.21))
    ink = _mask(arr, PALETTE["ink"])
    bits = []
    radius = max(2, pitch // 5)
    for row in range(3):
        for col in range(3):
            px = int(round(cx + (col - 1) * pitch))
            py = int(round(cy + (row - 1) * pitch))
            sample = ink[py - radius : py + radius + 1, px - radius : px + radius + 1]
            bits.append(int(sample.size > 0 and float(sample.mean()) > 0.55))
    if sum(bits) != 4:
        raise ValueError(f"glyph has {sum(bits)} filled microcells instead of four")
    return tuple(bits)


def _candidate_boxes(arr: np.ndarray) -> list[tuple[str, tuple[int, int, int, int]]]:
    boxes = []
    for name in ("red", "green", "blue", "orange"):
        box = _bbox(_mask(arr, PALETTE[name]))
        if box[2] - box[0] < 140 or box[3] - box[1] < 110:
            raise ValueError(f"{name} candidate frame is malformed")
        boxes.append((name, box))
    return boxes


def decision_from_image(image: Image.Image) -> str:
    """Infer the missing glyph from pixels, then read its candidate border color."""
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    xs, ys = _main_grid_lines(_mask(arr, PALETTE["grid"]))
    magenta = _bbox(_mask(arr, PALETTE["magenta"]))
    mx, my = (magenta[0] + magenta[2]) / 2.0, (magenta[1] + magenta[3]) / 2.0
    col = next((index for index in range(4) if xs[index] < mx < xs[index + 1]), None)
    row = next((index for index in range(4) if ys[index] < my < ys[index + 1]), None)
    if row is None or col is None:
        raise ValueError("magenta target is outside the symbol grid")

    row_patterns = {
        _decode_symbol(arr, (xs[c], ys[row], xs[c + 1], ys[row + 1]))
        for c in range(4)
        if c != col
    }
    col_patterns = {
        _decode_symbol(arr, (xs[col], ys[r], xs[col + 1], ys[r + 1]))
        for r in range(4)
        if r != row
    }
    if len(row_patterns) != 3 or len(col_patterns) != 3 or row_patterns != col_patterns:
        raise ValueError("row and column exclusions are not uniquely readable")

    candidates = {
        _decode_symbol(arr, tuple(float(value) for value in box)): color
        for color, box in _candidate_boxes(arr)
    }
    if len(candidates) != 4:
        raise ValueError("candidate glyphs are not one-to-one")
    missing = set(candidates) - row_patterns
    if len(missing) != 1:
        raise ValueError("pixel evidence does not yield one missing glyph")
    return candidates[next(iter(missing))]
