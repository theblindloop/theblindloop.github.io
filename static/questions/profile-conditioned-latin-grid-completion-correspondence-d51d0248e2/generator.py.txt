"""Deterministic scene model and rasterizer for Latin-grid completion."""

from __future__ import annotations

import itertools
import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 980, 720
BACKGROUND = (248, 250, 252)
CARD = (255, 255, 255)
INK = (15, 23, 42)
GRID = (184, 196, 211)
BLOCK = (226, 232, 240)
BLOCK_X = (100, 116, 139)
MAGENTA = (217, 70, 239)
BORDER_RGB = {
    "red": (220, 38, 38),
    "green": (22, 163, 74),
    "blue": (37, 99, 235),
    "orange": (234, 88, 12),
}
BORDER_NAMES = ("red", "green", "blue", "orange")


def _patterns(rng: random.Random) -> list[list[int]]:
    """Choose four equal-ink 3x3 glyphs with a visible Hamming separation."""
    pool = []
    for cells in itertools.combinations(range(9), 4):
        # Avoid nearly featureless full rows/columns and keep all glyphs substantial.
        rows = {cell // 3 for cell in cells}
        cols = {cell % 3 for cell in cells}
        if len(rows) >= 2 and len(cols) >= 2:
            pattern = [int(index in cells) for index in range(9)]
            pool.append(pattern)
    rng.shuffle(pool)
    chosen: list[list[int]] = []
    for pattern in pool:
        if all(sum(a != b for a, b in zip(pattern, other)) >= 4 for other in chosen):
            chosen.append(pattern)
            if len(chosen) == 4:
                return chosen
    raise RuntimeError("could not choose separated symbol patterns")


def _missing_symbol(grid: list[list[int]]) -> int | None:
    target = [(r, c) for r in range(4) for c in range(4) if grid[r][c] == -1]
    if len(target) != 1:
        return None
    row, col = target[0]
    row_missing = set(range(4)) - {value for value in grid[row] if value >= 0}
    col_missing = set(range(4)) - {grid[r][col] for r in range(4) if grid[r][col] >= 0}
    if len(row_missing) != 1 or row_missing != col_missing:
        return None
    return next(iter(row_missing))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    symbol_patterns = _patterns(rng)

    row_order = list(range(4))
    col_order = list(range(4))
    symbol_order = list(range(4))
    rng.shuffle(row_order)
    rng.shuffle(col_order)
    rng.shuffle(symbol_order)
    full = [
        [symbol_order[row_order[r] ^ col_order[c]] for c in range(4)]
        for r in range(4)
    ]

    removals = None
    permutations = list(itertools.permutations(range(4)))
    rng.shuffle(permutations)
    for columns in permutations:
        cells = [(r, columns[r]) for r in range(4)]
        if len({full[r][c] for r, c in cells}) == 4:
            removals = cells
            break
    if removals is None:
        raise RuntimeError("could not balance removed symbols")

    requested_symbol = (int(seed) // 4) % 4
    target_cell = next((r, c) for r, c in removals if full[r][c] == requested_symbol)
    grid_symbols = deepcopy(full)
    for r, c in removals:
        grid_symbols[r][c] = -1 if (r, c) == target_cell else -2

    answer_color = BORDER_NAMES[int(seed) % 4]
    remaining_colors = [color for color in BORDER_NAMES if color != answer_color]
    rng.shuffle(remaining_colors)
    color_for_symbol = {requested_symbol: answer_color}
    for symbol, color in zip(
        [value for value in range(4) if value != requested_symbol], remaining_colors
    ):
        color_for_symbol[symbol] = color
    candidate_symbols = list(range(4))
    rng.shuffle(candidate_symbols)

    return {
        "symbol_patterns": symbol_patterns,
        "grid_symbols": grid_symbols,
        "candidate_symbols": candidate_symbols,
        "border_colors": [color_for_symbol[symbol] for symbol in candidate_symbols],
        "layout_offset": [rng.randint(-10, 10), rng.randint(-6, 6)],
    }


def _outline(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], color, width: int) -> None:
    x0, y0, x1, y1 = box
    draw.rectangle((x0, y0, x1, y0 + width - 1), fill=color)
    draw.rectangle((x0, y1 - width + 1, x1, y1), fill=color)
    draw.rectangle((x0, y0, x0 + width - 1, y1), fill=color)
    draw.rectangle((x1 - width + 1, y0, x1, y1), fill=color)


def _draw_symbol(
    draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], pattern: list[int]
) -> None:
    x0, y0, x1, y1 = box
    cx, cy = (x0 + x1) // 2, (y0 + y1) // 2
    pitch = max(12, int(min(x1 - x0, y1 - y0) * 0.21))
    left, top = cx - 3 * pitch // 2, cy - 3 * pitch // 2
    draw.rectangle((left, top, left + 3 * pitch, top + 3 * pitch), outline=GRID, width=2)
    for k in (1, 2):
        draw.line((left + k * pitch, top, left + k * pitch, top + 3 * pitch), fill=GRID, width=2)
        draw.line((left, top + k * pitch, left + 3 * pitch, top + k * pitch), fill=GRID, width=2)
    inset = max(3, pitch // 6)
    for index, filled in enumerate(pattern):
        if not filled:
            continue
        row, col = divmod(index, 3)
        bx0 = left + col * pitch + inset
        by0 = top + row * pitch + inset
        bx1 = left + (col + 1) * pitch - inset
        by1 = top + (row + 1) * pitch - inset
        draw.rectangle((bx0, by0, bx1, by1), fill=INK)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    dx, dy = (int(value) for value in scene["layout_offset"])
    patterns = scene["symbol_patterns"]
    grid = scene["grid_symbols"]
    candidates = scene["candidate_symbols"]
    colors = scene["border_colors"]

    gx, gy, cell = 280 + dx, 35 + dy, 105
    draw.rounded_rectangle((gx - 22, gy - 18, gx + 4 * cell + 22, gy + 4 * cell + 18), radius=14, fill=CARD)
    for r in range(4):
        for c in range(4):
            x0, y0 = gx + c * cell, gy + r * cell
            box = (x0, y0, x0 + cell, y0 + cell)
            symbol = int(grid[r][c])
            if symbol >= 0:
                _draw_symbol(draw, box, patterns[symbol])
            elif symbol == -1:
                _outline(draw, (x0 + 9, y0 + 9, x0 + cell - 9, y0 + cell - 9), MAGENTA, 6)
            else:
                draw.rounded_rectangle((x0 + 13, y0 + 13, x0 + cell - 13, y0 + cell - 13), radius=8, fill=BLOCK)
                draw.line((x0 + 34, y0 + 34, x0 + cell - 34, y0 + cell - 34), fill=BLOCK_X, width=7)
                draw.line((x0 + cell - 34, y0 + 34, x0 + 34, y0 + cell - 34), fill=BLOCK_X, width=7)
    # Draw the shared grid last, keeping its five long guide lines connected and readable.
    for k in range(5):
        draw.line((gx + k * cell, gy, gx + k * cell, gy + 4 * cell), fill=GRID, width=3)
        draw.line((gx, gy + k * cell, gx + 4 * cell, gy + k * cell), fill=GRID, width=3)

    card_y = 535 + dy
    for index, (symbol, color_name) in enumerate(zip(candidates, colors)):
        x0 = 25 + dx + index * 237
        y0, x1, y1 = card_y, x0 + 215, card_y + 160
        draw.rectangle((x0, y0, x1, y1), fill=CARD)
        _outline(draw, (x0, y0, x1, y1), BORDER_RGB[color_name], 7)
        _draw_symbol(draw, (x0, y0, x1, y1), patterns[int(symbol)])
    return image


def analytic_gold(scene: dict) -> str:
    missing = _missing_symbol(scene["grid_symbols"])
    if missing is None:
        return "abstain"
    candidates = scene["candidate_symbols"]
    colors = scene["border_colors"]
    winners = [index for index, symbol in enumerate(candidates) if int(symbol) == missing]
    return colors[winners[0]] if len(winners) == 1 else "abstain"


def margin(scene: dict) -> float:
    missing = _missing_symbol(scene["grid_symbols"])
    if missing is None:
        return 0.0
    patterns = scene["symbol_patterns"]
    target = patterns[missing]
    distances = sorted(
        sum(a != b for a, b in zip(target, patterns[int(symbol)]))
        for symbol in scene["candidate_symbols"]
    )
    return float(distances[1] - distances[0]) if len(distances) >= 2 else 0.0


def is_quarantined(scene: dict) -> bool:
    missing = _missing_symbol(scene["grid_symbols"])
    candidates = [int(value) for value in scene["candidate_symbols"]]
    colors = list(scene["border_colors"])
    patterns = scene["symbol_patterns"]
    return (
        missing is None
        or candidates.count(missing) != 1
        or sorted(candidates) != list(range(4))
        or sorted(colors) != sorted(BORDER_NAMES)
        or len({tuple(pattern) for pattern in patterns}) != 4
        or margin(scene) < 2.0
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Consistently relabel internal indices without changing a rendered pixel."""
    mapping = {0: 1, 1: 2, 2: 3, 3: 0}
    twin = deepcopy(scene)
    relabeled_patterns: list[list[int] | None] = [None, None, None, None]
    for old, new in mapping.items():
        relabeled_patterns[new] = deepcopy(scene["symbol_patterns"][old])
    twin["symbol_patterns"] = relabeled_patterns
    twin["grid_symbols"] = [
        [mapping[value] if value >= 0 else value for value in row]
        for row in scene["grid_symbols"]
    ]
    twin["candidate_symbols"] = [mapping[int(value)] for value in scene["candidate_symbols"]]
    return [("cyclic_internal_symbol_relabel", twin)]
