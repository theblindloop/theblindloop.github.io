"""A band of repeating marks whose spatial period must be matched to one tick.

The latent scene stores a canvas, a background and an ink colour, one integer
period, a motif of small rectangles that lives inside a single period cell, the
geometry of the repeated band, the x of a black reference tick and an unordered
list of four coloured ticks, each with a horizontal offset from the reference.

Nothing in the scene stores the answer.  The analytic gold reads the period and
the tick offsets and names the colour of the single tick whose offset equals one
period; the pixel arm never sees the period and recovers it by scoring how well
the drawn band maps onto itself under each drawn tick's shift.
"""

from __future__ import annotations

import random

import numpy as np
from PIL import Image

ABSTAIN = "abstain"

WIDTH = 520
HEIGHT = 300
BG_COLOR = (243, 241, 234)
INK_COLOR = (58, 60, 68)
ANCHOR_COLOR = (12, 12, 16)

STRIP_TOP = 44
STRIP_BOTTOM = 152
STRIP_X0 = 48
TICK_TOP = 208
TICK_BOTTOM = 258
TICK_HALFWIDTH = 4

PALETTE = {
    "red": (206, 42, 40),
    "green": (28, 142, 66),
    "blue": (40, 84, 206),
    "orange": (228, 146, 18),
}
CLASS_NAMES = ("red", "green", "blue", "orange")

QUARANTINE_PX = 12.0
SAMPLING_FLOOR_PX = 18.0
MIN_TICK_GAP_PX = 16
MIN_OFFSET_PX = 26
MAX_SELF_MATCH = 0.45
MIN_MATCH_GAP = 0.25


def _multiple_distance(offset: int, period: int) -> float:
    """Distance in pixels from an offset to the nearest whole period multiple."""
    best = float(offset)
    m = 1
    while m * period <= offset + period:
        best = min(best, abs(offset - m * period))
        m += 1
    return best


def _strip_mask(scene: dict) -> np.ndarray:
    """Boolean ink mask of the repeated band, rebuilt from scene geometry."""
    period = int(scene["period"])
    repeats = int(scene["repeats"])
    height = STRIP_BOTTOM - STRIP_TOP
    mask = np.zeros((height, repeats * period), dtype=bool)
    for cell in range(repeats):
        base = cell * period
        for mark in scene["motif"]:
            x = base + int(mark["x"])
            y = int(mark["y"])
            mask[y : y + int(mark["h"]), x : x + int(mark["w"])] = True
    return mask


def _self_match(mask: np.ndarray, shift: int) -> float:
    """Intersection over union of the band with a copy shifted right by `shift`."""
    width = mask.shape[1]
    if shift <= 0 or shift >= width:
        return 0.0
    left = mask[:, : width - shift]
    right = mask[:, shift:]
    union = int(np.count_nonzero(left | right))
    if union == 0:
        return 0.0
    return float(np.count_nonzero(left & right)) / float(union)


def sample_scene(seed: int) -> dict:
    """Sample one balanced scene; the answer colour is drawn uniformly."""
    rng = random.Random(seed)
    for _ in range(4000):
        period = rng.randint(42, 74)
        repeats = (WIDTH - 2 * STRIP_X0) // period
        if repeats < 5:
            continue
        span = repeats * period
        motif = []
        for _mark in range(rng.randint(4, 7)):
            w = rng.randint(5, 11)
            h = rng.randint(9, 26)
            motif.append(
                {
                    "x": rng.randint(0, period - w),
                    "y": rng.randint(0, (STRIP_BOTTOM - STRIP_TOP) - h),
                    "w": w,
                    "h": h,
                }
            )
        anchor_x = STRIP_X0 + rng.randint(-6, 6)
        max_offset = min(span // 2, WIDTH - 24 - anchor_x)
        if max_offset < period + 40:
            continue
        offsets = [period]
        ok = True
        for _distractor in range(3):
            for _attempt in range(300):
                cand = rng.randint(MIN_OFFSET_PX, max_offset)
                if _multiple_distance(cand, period) < SAMPLING_FLOOR_PX:
                    continue
                if any(abs(cand - other) < MIN_TICK_GAP_PX for other in offsets):
                    continue
                offsets.append(cand)
                break
            else:
                ok = False
                break
        if not ok:
            continue

        mask = _strip_mask(scene={"period": period, "repeats": repeats, "motif": motif})
        if not 0.14 <= mask.mean() <= 0.36:
            continue
        scores = {off: _self_match(mask, off) for off in offsets}
        if scores[period] < 0.98:
            continue
        worst = max(scores[off] for off in offsets[1:])
        if worst > MAX_SELF_MATCH or scores[period] - worst < MIN_MATCH_GAP:
            continue

        colours = list(CLASS_NAMES)
        rng.shuffle(colours)
        ticks = [
            {"colour": colour, "offset": int(off)}
            for colour, off in zip(colours, offsets)
        ]
        rng.shuffle(ticks)
        return {
            "width": WIDTH,
            "height": HEIGHT,
            "bg_color": list(BG_COLOR),
            "ink_color": list(INK_COLOR),
            "anchor_color": list(ANCHOR_COLOR),
            "anchor_x": int(anchor_x),
            "period": int(period),
            "repeats": int(repeats),
            "motif": motif,
            "ticks": ticks,
        }
    raise RuntimeError(f"scene sampling failed for seed {seed}")


def render(scene: dict) -> Image.Image:
    """Paint the repeated band, the black reference tick and the coloured ticks."""
    width = int(scene["width"])
    height = int(scene["height"])
    canvas = np.zeros((height, width, 3), dtype=np.uint8)
    canvas[:, :] = np.array(scene["bg_color"], dtype=np.uint8)

    mask = _strip_mask(scene)
    band = canvas[STRIP_TOP:STRIP_BOTTOM, STRIP_X0 : STRIP_X0 + mask.shape[1]]
    band[mask] = np.array(scene["ink_color"], dtype=np.uint8)

    def bar(x_center: int, colour: tuple[int, int, int]) -> None:
        x0 = max(0, x_center - TICK_HALFWIDTH)
        x1 = min(width, x_center + TICK_HALFWIDTH + 1)
        canvas[TICK_TOP:TICK_BOTTOM, x0:x1] = np.array(colour, dtype=np.uint8)

    bar(int(scene["anchor_x"]), tuple(scene["anchor_color"]))
    for tick in scene["ticks"]:
        bar(int(scene["anchor_x"]) + int(tick["offset"]), PALETTE[tick["colour"]])
    return Image.fromarray(canvas, mode="RGB")


def analytic_gold(scene: dict) -> str:
    """Colour of the single tick whose offset is exactly one period."""
    period = int(scene["period"])
    hits = [t["colour"] for t in scene["ticks"] if int(t["offset"]) == period]
    if len(hits) != 1:
        return ABSTAIN
    return hits[0]


def margin(scene: dict) -> float:
    """Pixel shift mismatch of the best wrong tick from any whole period."""
    period = int(scene["period"])
    others = [int(t["offset"]) for t in scene["ticks"] if int(t["offset"]) != period]
    if not others:
        return 0.0
    return float(min(_multiple_distance(off, period) for off in others))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The tick list and the motif list are unordered: reordering is pixel identical."""
    aliases: list[tuple[str, dict]] = []
    ticks = list(scene["ticks"])
    if len(ticks) > 1:
        rotated = dict(scene)
        rotated["ticks"] = [dict(t) for t in ticks[1:]] + [dict(ticks[0])]
        aliases.append(("tick_list_rotation", rotated))
        reversed_scene = dict(scene)
        reversed_scene["ticks"] = [dict(t) for t in reversed(ticks)]
        aliases.append(("tick_list_reversal", reversed_scene))
    motif = list(scene["motif"])
    if len(motif) > 1:
        shuffled = dict(scene)
        shuffled["motif"] = [dict(m) for m in reversed(motif)]
        aliases.append(("motif_paint_order", shuffled))
    return aliases
