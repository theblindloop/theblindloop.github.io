"""Independent pixel-only inverse arm for white-space connectivity."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

BORDER = np.array((90, 98, 110), dtype=np.int16)
WALL = np.array((30, 34, 40), dtype=np.int16)
MAGENTA = np.array((210, 44, 132), dtype=np.int16)
AMBER = np.array((238, 158, 28), dtype=np.int16)
LABELS = ("A", "B", "C", "D")


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(x), int(y))])
        seen[y, x] = True
        xs, ys = [], []
        while queue:
            px, py = queue.popleft()
            xs.append(px); ys.append(py)
            for nx, ny in ((px-1, py), (px+1, py), (px, py-1), (px, py+1)):
                if 0 <= nx < width and 0 <= ny < height and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((nx, ny))
        found.append((min(xs), min(ys), max(xs), max(ys), len(xs)))
    return found


def _centroid(mask: np.ndarray) -> tuple[int, int]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 40:
        raise ValueError("terminal not found")
    return int(round(float(xs.mean()))), int(round(float(ys.mean())))


def _reachable(blocked: np.ndarray, start: tuple[int, int], goal: tuple[int, int]) -> bool:
    height, width = blocked.shape
    queue = deque([start])
    seen = np.zeros_like(blocked, dtype=bool)
    seen[start[1], start[0]] = True
    while queue:
        x, y = queue.popleft()
        if abs(x-goal[0]) <= 1 and abs(y-goal[1]) <= 1:
            return True
        for nx, ny in ((x-1, y), (x+1, y), (x, y-1), (x, y+1)):
            if 0 <= nx < width and 0 <= ny < height and not blocked[ny, nx] and not seen[ny, nx]:
                seen[ny, nx] = True
                queue.append((nx, ny))
    return False


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    border_mask = np.max(np.abs(pixels - BORDER), axis=2) <= 1
    boxes = [item for item in _components(border_mask) if item[2]-item[0] > 250 and item[3]-item[1] > 180]
    if len(boxes) != 4:
        raise ValueError(f"expected four visible panels, found {len(boxes)}")
    boxes.sort(key=lambda box: (box[1] // 100, box[0]))
    separated = []
    for index, (x0, y0, x1, y1, _) in enumerate(boxes):
        crop = pixels[y0+3:y1-2, x0+3:x1-2]
        magenta = np.max(np.abs(crop - MAGENTA), axis=2) <= 2
        amber = np.max(np.abs(crop - AMBER), axis=2) <= 2
        m = _centroid(magenta)
        a = _centroid(amber)
        blocked = np.max(np.abs(crop - WALL), axis=2) <= 2
        blocked[magenta | amber] = False
        if not _reachable(blocked, m, a):
            separated.append(index)
    if len(separated) != 1:
        raise ValueError(f"expected one separated panel, found {len(separated)}")
    return LABELS[separated[0]]
