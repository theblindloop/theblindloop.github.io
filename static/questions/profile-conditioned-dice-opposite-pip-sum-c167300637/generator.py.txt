"""Deterministic matched die-face records with visible opposite-face sums."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT, SCALE = 512, 384, 4
ANSWERS = (5, 6, 7, 8, 9)
PALETTES = (
    {"gold": (224, 165, 35), "neutral": (71, 79, 91), "pip": (31, 35, 42), "face": (255, 253, 245)},
    {"gold": (205, 145, 25), "neutral": (58, 70, 82), "pip": (25, 32, 38), "face": (250, 253, 249)},
    {"gold": (235, 178, 42), "neutral": (76, 72, 82), "pip": (36, 32, 39), "face": (255, 250, 241)},
)

PIP_GRID = {
    1: ((0.50, 0.50),),
    2: ((0.30, 0.30), (0.70, 0.70)),
    3: ((0.28, 0.28), (0.50, 0.50), (0.72, 0.72)),
    4: ((0.29, 0.29), (0.71, 0.29), (0.29, 0.71), (0.71, 0.71)),
    5: ((0.27, 0.27), (0.73, 0.27), (0.50, 0.50), (0.27, 0.73), (0.73, 0.73)),
    6: ((0.28, 0.24), (0.72, 0.24), (0.28, 0.50), (0.72, 0.50), (0.28, 0.76), (0.72, 0.76)),
}


def _pair_for_sum(total: int, rng: random.Random) -> list[int]:
    pairs = [(a, total - a) for a in range(1, 7) if 1 <= total - a <= 6]
    return list(rng.choice(pairs))


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed ^ 0xD1CE5A7)
    answer = ANSWERS[seed % len(ANSWERS)]
    selected_slots = sorted(rng.sample(range(4), 2))
    selected_values = _pair_for_sum(answer, rng)
    distractor_values = _pair_for_sum(14 - answer, rng)
    rng.shuffle(selected_values)
    rng.shuffle(distractor_values)
    values = [0, 0, 0, 0]
    si = di = 0
    for slot in range(4):
        if slot in selected_slots:
            values[slot] = selected_values[si]
            si += 1
        else:
            values[slot] = distractor_values[di]
            di += 1
    order = list(range(4))
    rng.shuffle(order)
    return {
        "face_values": values,
        "selected_slots": selected_slots,
        "layout": "row" if rng.randrange(2) == 0 else "grid",
        "palette_index": rng.randrange(len(PALETTES)),
        "panel_draw_order": order,
    }


def _positions(scene: dict) -> list[tuple[int, int, int]]:
    if scene["layout"] == "row":
        return [(25, 153, 102), (145, 153, 102), (265, 153, 102), (385, 153, 102)]
    return [(135, 103, 104), (273, 103, 104), (135, 237, 104), (273, 237, 104)]


def _draw_panel(draw: ImageDraw.ImageDraw, box: tuple[int, int, int], value: int, selected: bool, palette: dict) -> None:
    x, y, size = (v * SCALE for v in box)
    radius = 11 * SCALE
    outline = palette["gold"] if selected else palette["neutral"]
    width = (8 if selected else 5) * SCALE
    draw.rounded_rectangle((x, y, x + size, y + size), radius=radius, fill=palette["face"], outline=outline, width=width)
    pip_r = 8 * SCALE
    inset = 8 * SCALE
    inner = size - 2 * inset
    for px, py in PIP_GRID[int(value)]:
        cx = x + inset + px * inner
        cy = y + inset + py * inner
        draw.ellipse((cx - pip_r, cy - pip_r, cx + pip_r, cy + pip_r), fill=palette["pip"])


def render(scene: dict) -> Image.Image:
    palette = PALETTES[int(scene["palette_index"])]
    positions = _positions(scene)
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), (239, 243, 247))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default(size=24 * SCALE)
    small = ImageFont.load_default(size=16 * SCALE)
    draw.text((WIDTH * SCALE / 2, 28 * SCALE), "DIE FACE RECORD", font=font, fill=(38, 45, 55), anchor="mm")
    draw.text((WIDTH * SCALE / 2, 61 * SCALE), "Gold frames mark the declared opposite pair", font=small, fill=(71, 78, 88), anchor="mm")
    for slot in scene["panel_draw_order"]:
        _draw_panel(
            draw,
            positions[int(slot)],
            int(scene["face_values"][int(slot)]),
            int(slot) in scene["selected_slots"],
            palette,
        )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(sum(int(scene["face_values"][int(slot)]) for slot in scene["selected_slots"]))


def margin(scene: dict) -> float:
    # Minimum nominal edge gap between 16 px pips in the densest six-pip layout.
    return 9.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["panel_draw_order"] = list(reversed(scene["panel_draw_order"]))
    return [("reverse_disjoint_panel_draw_order", twin)]
