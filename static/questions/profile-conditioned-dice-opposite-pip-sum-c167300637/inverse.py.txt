"""Independent PNG-only selector and pip counter for the marked die faces."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int) -> list[tuple[int, int, int, int, int]]:
    """Return (area, x0, y0, x1, y1) for 4-connected components."""
    mask = mask.astype(bool, copy=True)
    out: list[tuple[int, int, int, int, int]] = []
    height, width = mask.shape
    for y0, x0 in zip(*np.nonzero(mask), strict=True):
        if not mask[y0, x0]:
            continue
        queue = deque([(int(y0), int(x0))])
        mask[y0, x0] = False
        area = 0
        min_x = max_x = int(x0)
        min_y = max_y = int(y0)
        while queue:
            y, x = queue.popleft()
            area += 1
            min_x, max_x = min(min_x, x), max(max_x, x)
            min_y, max_y = min(min_y, y), max(max_y, y)
            for yy, xx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= yy < height and 0 <= xx < width and mask[yy, xx]:
                    mask[yy, xx] = False
                    queue.append((yy, xx))
        if area >= minimum:
            out.append((area, min_x, min_y, max_x, max_y))
    return out


def _gold_frames(pixels: np.ndarray) -> list[tuple[int, int, int, int]]:
    r, g, b = pixels[..., 0], pixels[..., 1], pixels[..., 2]
    mask = (r > 155) & (g > 95) & (b < 105) & (r > g * 1.08) & (g > b * 1.45)
    frames = []
    for area, x0, y0, x1, y1 in _components(mask, 700):
        width, height = x1 - x0 + 1, y1 - y0 + 1
        if 85 <= width <= 120 and 85 <= height <= 120 and area >= 1200:
            frames.append((x0, y0, x1, y1))
    if len(frames) != 2:
        raise ValueError(f"expected exactly two recoverable gold frames, found {len(frames)}")
    return frames


def _count_pips(pixels: np.ndarray, frame: tuple[int, int, int, int]) -> int:
    x0, y0, x1, y1 = frame
    inset = 14
    crop = pixels[y0 + inset:y1 - inset + 1, x0 + inset:x1 - inset + 1]
    dark = (crop[..., 0] < 105) & (crop[..., 1] < 105) & (crop[..., 2] < 110)
    components = _components(dark, 120)
    pips = [item for item in components if 150 <= item[0] <= 320 and 12 <= item[3] - item[1] + 1 <= 22 and 12 <= item[4] - item[2] + 1 <= 22]
    if not 1 <= len(pips) <= 6:
        raise ValueError(f"selected face has ambiguous pip count {len(pips)}")
    return len(pips)


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    frames = _gold_frames(pixels)
    return str(sum(_count_pips(pixels, frame) for frame in frames))
