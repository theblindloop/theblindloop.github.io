"""Deterministic renderer for an eight-like glyph with 1, 2, or 3 holes."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS_SIZE = 512
SUPERSAMPLE = 4
INK_PALETTES = (
    ((25, 31, 50), (249, 247, 240)),
    ((43, 36, 49), (244, 248, 250)),
    ((24, 55, 48), (250, 246, 250)),
    ((57, 31, 45), (247, 249, 242)),
)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one member of a matched canonical/counterfactual pair.

    Consecutive integer seeds share all style fields.  The even member is the
    canonical two-hole control; the odd member is a one- or three-hole
    counterfactual, alternating by pair.
    """

    pair_seed = int(seed) // 2
    rng = random.Random(pair_seed)
    control = int(seed) % 2 == 0
    hole_count = 2 if control else (1 if pair_seed % 2 == 0 else 3)
    ink_rgb, background_rgb = INK_PALETTES[rng.randrange(len(INK_PALETTES))]
    return {
        "hole_count": hole_count,
        "filled_lobe": -1 if rng.random() < 0.5 else 1,
        "center_x": rng.uniform(205.0, 307.0),
        "center_y": rng.uniform(250.0, 262.0),
        "scale": rng.uniform(0.88, 1.03),
        "rotation_deg": rng.uniform(-10.0, 10.0),
        "ink_rgb": list(ink_rgb),
        "background_rgb": list(background_rgb),
        "outer_rx": rng.uniform(105.0, 115.0),
        "outer_ry": rng.uniform(122.0, 128.0),
        "lobe_offset": rng.uniform(80.0, 85.0),
    }


def _ellipse(draw: ImageDraw.ImageDraw, cx: float, cy: float, rx: float, ry: float, fill: int) -> None:
    draw.ellipse((cx - rx, cy - ry, cx + rx, cy + ry), fill=fill)


def _glyph_mask(scene: dict[str, Any]) -> Image.Image:
    """Draw a high-resolution local mask; all dimensions are latent-visible."""

    ss = SUPERSAMPLE
    local_w, local_h = 350 * ss, 480 * ss
    mask = Image.new("L", (local_w, local_h), 0)
    draw = ImageDraw.Draw(mask)
    cx, cy = local_w / 2.0, local_h / 2.0
    scale = float(scene["scale"])
    rx = float(scene["outer_rx"]) * scale * ss
    ry = float(scene["outer_ry"]) * scale * ss
    offset = float(scene["lobe_offset"]) * scale * ss

    # Two overlapping lobes form one connected, recognizably eight-like body.
    _ellipse(draw, cx, cy - offset, rx, ry, 255)
    _ellipse(draw, cx, cy + offset, rx, ry, 255)

    count = int(scene["hole_count"])
    bias = int(scene["filled_lobe"])
    if count == 1:
        # One enlarged counter has nearly the same total removed area as the
        # two- and three-hole variants.  Its lobe alternates independently.
        _ellipse(
            draw,
            cx,
            cy + bias * offset,
            58.0 * scale * ss,
            78.0 * scale * ss,
            0,
        )
    elif count == 2:
        # A slight mirrored asymmetry makes the bias field visible without
        # changing total light area or count.
        top_rx = (42.0 + 1.5 * bias) * scale * ss
        bottom_rx = (42.0 - 1.5 * bias) * scale * ss
        _ellipse(draw, cx, cy - offset, top_rx, 55.0 * scale * ss, 0)
        _ellipse(draw, cx, cy + offset, bottom_rx, 55.0 * scale * ss, 0)
    elif count == 3:
        top_rx = (40.0 + 1.5 * bias) * scale * ss
        bottom_rx = (40.0 - 1.5 * bias) * scale * ss
        hole_offset = 92.0 * scale * ss
        _ellipse(draw, cx, cy - hole_offset, top_rx, 48.0 * scale * ss, 0)
        _ellipse(draw, cx, cy + hole_offset, bottom_rx, 48.0 * scale * ss, 0)
        _ellipse(draw, cx, cy, 28.0 * scale * ss, 28.0 * scale * ss, 0)
    else:
        raise ValueError("hole_count must be 1, 2, or 3")

    return mask.rotate(
        float(scene["rotation_deg"]),
        resample=Image.Resampling.BICUBIC,
        expand=True,
        fillcolor=0,
    )


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only the visible figure; no answer or metadata is embedded."""

    ss = SUPERSAMPLE
    background = tuple(int(v) for v in scene["background_rgb"])
    ink = tuple(int(v) for v in scene["ink_rgb"])
    canvas = Image.new("RGB", (CANVAS_SIZE * ss, CANVAS_SIZE * ss), background)
    mask = _glyph_mask(scene)
    x = int(round(float(scene["center_x"]) * ss - mask.width / 2.0))
    y = int(round(float(scene["center_y"]) * ss - mask.height / 2.0))
    layer = Image.new("RGB", mask.size, ink)
    canvas.paste(layer, (x, y), mask)
    return canvas.resize((CANVAS_SIZE, CANVAS_SIZE), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(int(scene["hole_count"]))


def margin(scene: dict[str, Any]) -> float:
    """Conservative minimum ink-bridge clearance in final-resolution pixels."""

    # The tightest bridge is around the added central hole in the 3-hole case;
    # 9 px at unit scale is a deliberately conservative geometric lower bound.
    return float(scene["scale"]) * 9.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # This is a minimal, pixel-injective parameterization rather than a slot list.
    return []
