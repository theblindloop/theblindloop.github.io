"""Deterministic latent sampler, renderer, and analytic AUC decision arm."""

from __future__ import annotations

import itertools
import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH = 760
HEIGHT = 650
CHECKPOINT_DX = 48
BASELINE_OFFSET = 180
QUARANTINE_GAP = 8.0

PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (214, 54, 64),
    "blue": (40, 112, 184),
    "green": (35, 142, 82),
    "orange": (226, 126, 34),
}
COLORS = tuple(PALETTE)
AXIS_RGB = (103, 112, 126)
PANEL_RGB = (255, 255, 255)

_PANEL_LEFT = (42, 398)
_PANEL_TOP = (96, 352)
_BASE_HEIGHT = 30
_LARGE_CHANGE = 48


def panel_geometry(panel: int) -> tuple[int, int, int, list[int]]:
    """Return panel left/top, visible zero-row, and six checkpoint columns."""
    if panel not in range(4):
        raise ValueError(f"invalid panel {panel}")
    left = _PANEL_LEFT[panel % 2]
    top = _PANEL_TOP[panel // 2]
    baseline = top + BASELINE_OFFSET
    x0 = left + 52
    return left, top, baseline, [x0 + index * CHECKPOINT_DX for index in range(6)]


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size=size)
    except OSError:
        return ImageFont.load_default()


def _pattern_pool(small_change: int) -> dict[int, list[list[int]]]:
    changes = [_LARGE_CHANGE, small_change, 0, -_LARGE_CHANGE, -small_change]
    pools: dict[int, list[list[int]]] = {}
    for order in sorted(set(itertools.permutations(changes))):
        values = [_BASE_HEIGHT]
        for delta in order:
            values.append(values[-1] + int(delta))
        if min(values) != _BASE_HEIGHT or max(values) != _BASE_HEIGHT + _LARGE_CHANGE + small_change:
            continue
        score2 = sum(values[index] + values[index + 1] for index in range(5))
        pools.setdefault(score2, []).append(values)
    return pools


def sample_scene(seed: int) -> dict:
    """Sample one balanced four-curve scene from an integer seed."""
    seed = int(seed)
    rng = random.Random(seed ^ 0xA0C5EED)
    winner = COLORS[seed % 4]
    winner_panel = (seed // 4) % 4
    near_boundary = seed % 11 == 0

    if near_boundary:
        small_change = 4
        target_scores2 = [420, 508, 692, 700]
    else:
        small_change = 24
        target_scores2 = [540, 636, 732, 780]
    pools = _pattern_pool(small_change)
    if any(score not in pools for score in target_scores2):
        raise RuntimeError("configured trapezoid pattern is unavailable")

    nonwinners = [color for color in COLORS if color != winner]
    rng.shuffle(nonwinners)
    nonwinner_scores = list(target_scores2[:-1])
    rng.shuffle(nonwinner_scores)
    score_by_color = {winner: target_scores2[-1]}
    score_by_color.update(dict(zip(nonwinners, nonwinner_scores, strict=True)))

    other_colors = list(nonwinners)
    rng.shuffle(other_colors)
    other_panels = [panel for panel in range(4) if panel != winner_panel]
    rng.shuffle(other_panels)
    panel_by_color = {winner: winner_panel}
    panel_by_color.update(dict(zip(other_colors, other_panels, strict=True)))

    curves = []
    for color in COLORS:
        score2 = score_by_color[color]
        variants = pools[score2]
        values = list(variants[rng.randrange(len(variants))])
        curves.append({"color": color, "panel": panel_by_color[color], "values": values})
    rng.shuffle(curves)
    return {"curves": curves}


def analytic_gold(scene: dict) -> str:
    ranked: list[tuple[float, str]] = []
    for curve in scene["curves"]:
        values = [float(value) for value in curve["values"]]
        if len(values) != 6:
            raise ValueError("each curve must contain six heights")
        area_without_common_spacing = sum(
            (values[index] + values[index + 1]) / 2.0 for index in range(5)
        )
        ranked.append((area_without_common_spacing, str(curve["color"])))
    ranked.sort(reverse=True)
    if len(ranked) != 4 or ranked[0][0] == ranked[1][0]:
        raise ValueError("scene does not have a unique AUC leader")
    return ranked[0][1]


def margin(scene: dict) -> float:
    scores = []
    for curve in scene["curves"]:
        values = [float(value) for value in curve["values"]]
        scores.append(
            sum((values[index] + values[index + 1]) / 2.0 for index in range(5))
        )
    scores.sort(reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["curves"] = list(reversed(twin["curves"]))
    return [("reverse_curve_record_order", twin)]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (242, 245, 249))
    draw = ImageDraw.Draw(image)
    draw.text((42, 20), "AREA UNDER CHECKPOINT CURVES", fill=(40, 49, 63), font=_font(22, True))
    draw.text(
        (42, 52),
        "Same scale  •  equal spacing  •  straight segments  •  compare all five trapezoids",
        fill=(84, 94, 109),
        font=_font(14),
    )

    for panel in range(4):
        left, top, baseline, xs = panel_geometry(panel)
        draw.rounded_rectangle(
            (left, top, left + 320, top + 222),
            radius=8,
            fill=PANEL_RGB,
            outline=(210, 217, 227),
            width=1,
        )
        draw.text((left + 12, top + 11), f"P{panel + 1}", fill=(78, 88, 103), font=_font(13, True))
        for level in (50, 100):
            y = baseline - level
            draw.line((xs[0], y, xs[-1], y), fill=(231, 235, 241), width=1)
            draw.text((left + 21, y - 7), str(level), fill=(137, 146, 159), font=_font(10))
        draw.line((xs[0], baseline - 112, xs[0], baseline), fill=AXIS_RGB, width=2)
        draw.line((xs[0], baseline, xs[-1], baseline), fill=AXIS_RGB, width=2)
        draw.text((left + 31, baseline - 7), "0", fill=(103, 112, 126), font=_font(10))
        for index, x in enumerate(xs, start=1):
            draw.line((x, baseline, x, baseline + 4), fill=AXIS_RGB, width=1)
            draw.text((x - 3, baseline + 7), str(index), fill=(126, 136, 150), font=_font(9))

    seen_panels: set[int] = set()
    seen_colors: set[str] = set()
    for curve in scene["curves"]:
        color = str(curve["color"])
        panel = int(curve["panel"])
        values = [int(value) for value in curve["values"]]
        if color not in PALETTE or panel not in range(4) or len(values) != 6:
            raise ValueError("invalid curve record")
        if color in seen_colors or panel in seen_panels:
            raise ValueError("colors and panels must each be unique")
        if min(values) <= 0 or max(values) > 112:
            raise ValueError("curve heights exceed the visible scale")
        seen_colors.add(color)
        seen_panels.add(panel)
        _left, _top, baseline, xs = panel_geometry(panel)
        points = [(x, baseline - value) for x, value in zip(xs, values, strict=True)]
        draw.line(points, fill=PALETTE[color], width=4, joint="curve")
        for x, y in points:
            draw.ellipse((x - 8, y - 8, x + 8, y + 8), fill=PANEL_RGB)
            draw.ellipse((x - 5, y - 5, x + 5, y + 5), fill=PALETTE[color])

    if seen_colors != set(COLORS) or seen_panels != set(range(4)):
        raise ValueError("scene must contain every color and panel exactly once")
    draw.text((560, 622), "checkpoint  →", fill=(84, 94, 109), font=_font(12, True))
    return image
