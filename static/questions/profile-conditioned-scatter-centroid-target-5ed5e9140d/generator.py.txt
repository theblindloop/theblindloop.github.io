"""Deterministic renderer and analytic arm for scatter_centroid_target."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw


WIDTH, HEIGHT = 640, 480
COLORS = {
    "blue": (43, 108, 176),
    "orange": (230, 126, 34),
    "green": (39, 155, 88),
}
NAMES = tuple(COLORS)
QUARANTINE_PX = 12.0


def _point(cx: float, cy: float, radial: tuple[float, float], tangential: tuple[float, float], r: float, t: float) -> list[int]:
    return [round(cx + r * radial[0] + t * tangential[0]), round(cy + r * radial[1] + t * tangential[1])]


def _ordinary_cloud(cx: int, cy: int, distance: float, angle: float, phase: int) -> list[list[int]]:
    radial = (math.cos(angle), math.sin(angle))
    tangential = (-radial[1], radial[0])
    # Seven equally weighted points whose tangential offsets sum to zero.
    offsets = [-54, -36, -18, 0, 18, 36, 54]
    radial_jitter = [-4, 3, 1, 0, -1, -3, 4]
    if phase % 2:
        radial_jitter = list(reversed(radial_jitter))
    return [
        _point(cx, cy, radial, tangential, distance + radial_jitter[i], value)
        for i, value in enumerate(offsets)
    ]


def _decoy_cloud(cx: int, cy: int, distance: float, angle: float) -> list[list[int]]:
    radial = (math.cos(angle), math.sin(angle))
    tangential = (-radial[1], radial[0])
    # One salient dot is only 24 px from the target, but the other six make the
    # full cloud centroid land at the requested (more distant) location.
    points = [_point(cx, cy, radial, tangential, 24.0, 0.0)]
    remaining_radial = (7.0 * distance - 24.0) / 6.0
    for value in (-50, -30, -10, 10, 30, 50):
        points.append(_point(cx, cy, radial, tangential, remaining_radial, value))
    return points


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = [rng.randint(285, 355), rng.randint(205, 275)]
    winner = int(seed) % 3
    runner = (winner + 1 + (int(seed) // 3) % 2) % 3
    third = 3 - winner - runner
    d1 = 34.0 + (int(seed) % 5)
    gap = 14.0 + 10.0 * (int(seed) % 4)
    distances = [0.0, 0.0, 0.0]
    distances[winner] = d1
    distances[runner] = d1 + gap
    distances[third] = d1 + gap + 54.0
    base = rng.uniform(0.0, 2.0 * math.pi)
    angles = [base + 2.0 * math.pi * i / 3.0 for i in range(3)]
    points: dict[str, list[list[int]]] = {}
    for i, name in enumerate(NAMES):
        if i == runner:
            points[name] = _decoy_cloud(target[0], target[1], distances[i], angles[i])
        else:
            points[name] = _ordinary_cloud(target[0], target[1], distances[i], angles[i], seed + i)
    return {"points": points, "target": target}


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (255, 255, 255))
    draw = ImageDraw.Draw(image)
    draw.rectangle((34, 32, WIDTH - 35, HEIGHT - 33), outline=(205, 211, 218), width=2)
    tx, ty = scene["target"]
    draw.line((tx - 13, ty, tx + 13, ty), fill=(20, 20, 20), width=3)
    draw.line((tx, ty - 13, tx, ty + 13), fill=(20, 20, 20), width=3)
    for name in NAMES:
        fill = COLORS[name]
        for x, y in scene["points"][name]:
            draw.ellipse((x - 7, y - 7, x + 7, y + 7), fill=fill, outline=(255, 255, 255), width=1)
    return image


def _centroid(points: list[list[int]]) -> tuple[float, float]:
    return (
        sum(point[0] for point in points) / len(points),
        sum(point[1] for point in points) / len(points),
    )


def _distances(scene: dict) -> dict[str, float]:
    tx, ty = scene["target"]
    values = {}
    for name, points in scene["points"].items():
        cx, cy = _centroid(points)
        values[name] = math.hypot(cx - tx, cy - ty)
    return values


def analytic_gold(scene: dict) -> str:
    values = _distances(scene)
    return min(values, key=values.get)


def margin(scene: dict) -> float:
    ordered = sorted(_distances(scene).values())
    return ordered[1] - ordered[0]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    del scene
    return []
