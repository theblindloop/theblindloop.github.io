"""Scene-free inverse arm: recover the answer from RGB pixels only."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


PALETTE = {
    "blue": np.array((43, 108, 176), dtype=np.int16),
    "orange": np.array((230, 126, 34), dtype=np.int16),
    "green": np.array((39, 155, 88), dtype=np.int16),
}


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[float, float, int]] = []
    for y0, x0 in zip(*np.nonzero(mask)):
        if seen[y0, x0]:
            continue
        stack = [(int(y0), int(x0))]
        seen[y0, x0] = True
        xs: list[int] = []
        ys: list[int] = []
        while stack:
            y, x = stack.pop()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(xs) >= 25:
            found.append((sum(xs) / len(xs), sum(ys) / len(ys), len(xs)))
    return found


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    black = np.all(pixels < 60, axis=2)
    ys, xs = np.nonzero(black)
    if len(xs) < 30:
        raise ValueError("black crosshair not recoverable")
    target = (float(xs.mean()), float(ys.mean()))
    distances: dict[str, float] = {}
    for name, rgb in PALETTE.items():
        mask = np.max(np.abs(pixels - rgb), axis=2) <= 2
        components = _components(mask)
        if len(components) != 7:
            raise ValueError(f"expected seven {name} dots, found {len(components)}")
        cx = sum(item[0] for item in components) / 7.0
        cy = sum(item[1] for item in components) / 7.0
        distances[name] = math.hypot(cx - target[0], cy - target[1])
    ordered = sorted(distances, key=distances.get)
    if distances[ordered[1]] - distances[ordered[0]] < 8.0:
        raise ValueError("pixel decision is ambiguous")
    return ordered[0]
