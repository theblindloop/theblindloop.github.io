"""Independent pixel-only inverse arm for the dead-end wall world.

From one rasterised PNG this module thresholds the paper against the ink,
labels the white pixels with a run-length union-find under 4-connectivity,
discards the component that reaches the image corner (the paper outside the
plate) and reports how many enclosed white components remain.  A wall with a
loose end leaves a white channel that keeps two sides of a room in one
component; a wall that runs the whole way across cuts the component in two.

It shares no code and no model with the renderer: there is no plate rectangle,
no wall list, no guillotine tree, no cell decomposition and no gap bookkeeping
here - only a luminance threshold, scanline runs and union-find.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "dead_end_wall_rooms-oracle-0.1.0"

WHITE_LEVEL = 170
MIN_ROOM_AREA = 400
MAX_ROOMS = 12


def _white_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    luminance = (array[..., 0] + array[..., 1] + array[..., 2]) // 3
    return luminance >= WHITE_LEVEL


def _runs(row: np.ndarray):
    """Half-open [start, end) runs of True in one scanline."""
    padded = np.concatenate(([0], row.astype(np.int8), [0]))
    edges = np.flatnonzero(np.diff(padded))
    return list(zip(edges[0::2].tolist(), edges[1::2].tolist()))


def _label(mask: np.ndarray):
    """Run-length connected components under 4-connectivity."""
    parent: list[int] = []

    def find(node: int) -> int:
        while parent[node] != node:
            parent[node] = parent[parent[node]]
            node = parent[node]
        return node

    def union(left: int, right: int) -> None:
        left, right = find(left), find(right)
        if left != right:
            parent[max(left, right)] = min(left, right)

    previous: list[tuple[int, int, int]] = []
    every: list[tuple[int, int, int, int]] = []
    for y in range(mask.shape[0]):
        current: list[tuple[int, int, int]] = []
        for start, end in _runs(mask[y]):
            parent.append(len(parent))
            index = len(parent) - 1
            current.append((start, end, index))
            every.append((y, start, end, index))
            for other_start, other_end, other_index in previous:
                if other_start < end and start < other_end:
                    union(index, other_index)
        previous = current
    return find, every


def decision_from_image(image: Image.Image) -> str:
    mask = _white_mask(image)
    if not mask.any():
        return "abstain"
    find, every = _label(mask)
    areas: dict[int, int] = {}
    outside: int | None = None
    for y, start, end, index in every:
        root = find(index)
        areas[root] = areas.get(root, 0) + (end - start)
        if y == 0 and start == 0:
            outside = root
    if outside is None:
        return "abstain"
    rooms = [root for root, area in areas.items() if root != outside and area >= MIN_ROOM_AREA]
    if not rooms or len(rooms) > MAX_ROOMS:
        return "abstain"
    return str(len(rooms))
