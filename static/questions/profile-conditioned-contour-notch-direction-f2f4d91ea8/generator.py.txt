"""Contour-notch-direction world: global dotted-ring concavity reveal.

A small number of dark dots is arranged around a shared center so that, by
proximity closure, they read as one closed ring-like outline (a gestalt contour
grouping). Seven dots sit on a circle of radius ``R`` around the center; exactly
one dot is tucked inward closer to the center, so the otherwise round closed
outline carries a single concave notch. The emergent global structure is the
one side of that closed contour which is dented inward. The answer is which of
the four cardinal sides the notch opens toward.

The renderer owns latent sampling + rasterization + the structural analytic
gold. The independent pixel-only inverse arm (``oracle.py``) recovers the same
global notch direction from the final PNG alone by finding which dot is closest
to the group center of the clustered dots.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (255, 255, 255)
FIGURE = (40, 80, 200)
# Radius of each discrete dot marker (px).
_DOT_RADIUS = 7
# Octagon radius of the ring contour (px): the seven undented dots sit here.
_R_LO = 150
_R_HI = 185
# How far the single dented dot is pulled inward toward the center (px).
_DENT_LO = 40
_DENT_HI = 90
# Independent-dot spacing check floor so contour dots never overlap each other.
_MIN_DOT_GAP = 3
# Below this margin a scene is quarantined (dent too shallow to resolve).
MIN_MARGIN = 16

_SIDES = ("top", "bottom", "left", "right")
# Cardinal angle (degrees from +x axis, screen y-down) for each dent side.
_SIDE_ANGLES = {"right": 0.0, "left": 180.0, "bottom": 90.0, "top": 270.0}


def _direction_of(dx: float, dy: float) -> str:
    """Cardinal direction of a vector (from the group center)."""
    if abs(dx) > abs(dy):
        return "right" if dx > 0 else "left"
    return "top" if dy < 0 else "bottom"


def _vertices_from(dent_side: str, r: float, dent: float, cx: float, cy: float) -> list:
    """Eight dot centers: a regular octagon with the cardinal vertex pulled in."""
    vertices = []
    for k in range(8):
        angle = math.radians(k * 45.0)
        radius = r
        if k * 45.0 == _SIDE_ANGLES[dent_side]:
            radius = r - dent
        x = cx + radius * math.cos(angle)
        y = cy + radius * math.sin(angle)
        vertices.append([round(x, 6), round(y, 6)])
    return vertices


def sample_scene(seed: int, dent_side: str | None = None) -> dict:
    """Sample one latent scene deterministically from ``seed``."""
    rng = np.random.default_rng(seed)
    if dent_side is None:
        dent_side = _SIDES[int(rng.integers(0, 4))]
    r = float(rng.integers(_R_LO, _R_HI + 1))
    dent = float(rng.integers(_DENT_LO, _DENT_HI + 1))
    cx = WIDTH / 2.0
    cy = HEIGHT / 2.0
    vertices = _vertices_from(dent_side, r, dent, cx, cy)
    while _min_dot_gap(vertices) < _MIN_DOT_GAP:
        rng = np.random.default_rng(rng.integers(0, 2**32))
        r = float(rng.integers(_R_LO, _R_HI + 1))
        vertices = _vertices_from(dent_side, r, dent, cx, cy)
    return {"vertices": vertices}


def _min_dot_gap(vertices: list) -> float:
    """Minimum pairwise centre distance among the vertices (px)."""
    vals = np.asarray(vertices, dtype=float)
    best = math.inf
    for i in range(len(vals)):
        for j in range(i + 1, len(vals)):
            best = min(best, float(np.hypot(*(vals[i] - vals[j]))))
    return best


def _draw_scene(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    d = ImageDraw.Draw(img)
    for x, y in scene["vertices"]:
        d.ellipse([x - _DOT_RADIUS, y - _DOT_RADIUS, x + _DOT_RADIUS, y + _DOT_RADIUS], fill=FIGURE)
    return img


def render(scene: dict) -> Image.Image:
    """Rasterize the latent scene (deterministic, boundary-free PNG)."""
    return _draw_scene(scene)


def _dent_direction(vertices: list) -> str:
    """Global decision: cardinal side of the single inward-tucked dot."""
    arr = np.asarray(vertices, dtype=float)
    center = arr.mean(axis=0)
    dists = np.hypot(arr[:, 0] - center[0], arr[:, 1] - center[1])
    vx, vy = arr[int(dists.argmin())]
    return _direction_of(vx - center[0], vy - center[1])


def analytic_gold(scene: dict) -> str:
    """Structural gold: the cardinal side the closed contour is notched into."""
    return _dent_direction(scene["vertices"])


def margin(scene: dict) -> float:
    """Commitment magnitude: radial gap between the dented dot and the ring."""
    arr = np.asarray(scene["vertices"], dtype=float)
    center = arr.mean(axis=0)
    dists = np.sort(np.hypot(arr[:, 0] - center[0], arr[:, 1] - center[1]))
    return float(dists[1] - dists[0])


def is_quarantined(scene: dict) -> bool:
    """Quarantine when the dent is too shallow for a clean cardinal read."""
    return margin(scene) < MIN_MARGIN


def latent_symmetries(scene: dict) -> list:
    """The scene is injectively rendered; no latent aliases are declared."""
    return []
