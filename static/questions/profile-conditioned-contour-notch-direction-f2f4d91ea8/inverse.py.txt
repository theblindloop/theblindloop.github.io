"""Independent pixel-only inverse arm for the contour-notch-direction world.

Recovers the global notch direction of the dotted ring from the final PNG alone:
it thresholds out the eight dark contour dots, labels the connected components,
and reports the cardinal side of the dot whose centroid lies closest to the
centroid of the whole cluster. It never imports renderer / prompts / generate /
verify / analytic-gold code and never receives the latent scene.
"""

from __future__ import annotations

from functools import lru_cache

import numpy as np
from PIL import Image

_DARK = 120
_MIN_COMPONENT = 25
_SIDES = ("top", "bottom", "left", "right")


def _connected_centroids(mask: np.ndarray) -> list[list[float]]:
    """Centroid (x, y) of every dark connected component via 8-connectivity BFS."""
    h, w = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    centroids = []
    ys, xs = np.nonzero(mask)
    for sy, sx in zip(ys.tolist(), xs.tolist()):
        if visited[sy, sx]:
            continue
        stack = [(sy, sx)]
        visited[sy, sx] = True
        count = 0
        sx_sum = 0
        sy_sum = 0
        while stack:
            y, x = stack.pop()
            count += 1
            sx_sum += x
            sy_sum += y
            for ny in (y - 1, y + 1):
                if 0 <= ny < h and mask[ny, x] and not visited[ny, x]:
                    visited[ny, x] = True
                    stack.append((ny, x))
            for nx in (x - 1, x + 1):
                if 0 <= nx < w and mask[y, nx] and not visited[y, nx]:
                    visited[y, nx] = True
                    stack.append((y, nx))
            for ny, nx in ((y - 1, x - 1), (y - 1, x + 1), (y + 1, x - 1), (y + 1, x + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    stack.append((ny, nx))
        if count >= _MIN_COMPONENT:
            centroids.append([sx_sum / count, sy_sum / count])
    return centroids


@lru_cache(maxsize=None)
def _decide(arr_bytes: bytes, shape: tuple[int, int]) -> str:
    a = np.frombuffer(arr_bytes, dtype=np.uint8).reshape(shape)
    mask = a < _DARK
    centroids = _connected_centroids(mask)
    if len(centroids) < 4:
        raise ValueError(f"expected a closed dotted ring, found {len(centroids)} components")
    pts = np.asarray(centroids, dtype=float)
    center = pts.mean(axis=0)
    dists = np.hypot(pts[:, 0] - center[0], pts[:, 1] - center[1])
    vx, vy = pts[int(dists.argmin())]
    dx, dy = vx - center[0], vy - center[1]
    if abs(dx) > abs(dy):
        return "right" if dx > 0 else "left"
    return "top" if dy < 0 else "bottom"


def decision_from_image(image: Image.Image) -> str:
    """Return the notched side ('top'/'bottom'/'left'/'right') from the PNG."""
    arr = np.asarray(image.convert("RGB"))
    gray = np.asarray(arr, dtype=np.uint8).min(axis=2)
    return _decide(gray.tobytes(), gray.shape)
