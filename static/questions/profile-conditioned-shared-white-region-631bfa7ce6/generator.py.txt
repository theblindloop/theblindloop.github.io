"""Shared-white-region renderer (region-connectivity of three coloured dots).

Several closed black rings are drawn on a white canvas. Each ring's interior
is a sealed white pocket, and the background outside all rings is one further
(connected) white region. Three small coloured dots (red, green, blue) are
placed so that exactly two of them lie inside the same sealed ring pocket and
the third lies in a different white region (either a different ring pocket or
the outside). The dots that are in the SAME connected white region form the
answer pair.

Decision: which two of the three coloured dots are in the same connected
white region (the same pocket of white cut off by the black rings).

The scene is a plain dict so the verifier's AST reachability audit is free of
side effects.

    z = {
        "width": int,
        "height": int,
        "rings": [ {"cx": int, "cy": int, "radius": int}, ... ],  # closed black rings
        "dots":  [ {"x": int, "y": int, "color": str}, ... ],     # three coloured dots
    }
"""

from __future__ import annotations

import math
import random
from collections import defaultdict
from collections.abc import Iterator

from PIL import Image, ImageDraw

RENDERER_VERSION = "shared_white_region-0.4.0"

WHITE = (255, 255, 255)
_RING = (0, 0, 0)
_COLOR_RGB = {"red": (230, 45, 45), "green": (50, 185, 60), "blue": (50, 90, 230)}

PAIRS = [("red", "green"), ("red", "blue"), ("green", "blue")]
_RANK = {"red": 0, "green": 1, "blue": 2}
_COLORS = ("red", "green", "blue")

# spread-out base slots (disjoint, well clear of the border)
_RING_SLOTS = [(135.0, 150.0), (365.0, 150.0), (250.0, 370.0)]
_RING_RADIUS = 62.0
_BAND = 16.0
_DOT_R = 12.0

_QUARANTINE_MARGIN = 17.0


def _pair_label(c1: str, c2: str) -> str:
    return "-".join(sorted((c1, c2), key=lambda c: _RANK[c]))


def _dist(ax: float, ay: float, bx: float, by: float) -> float:
    return math.hypot(ax - bx, ay - by)


def _inside_pocket(px: float, py: float, ring: dict) -> bool:
    """True when the point is strictly inside the ring's sealed white pocket."""
    return _dist(px, py, ring["cx"], ring["cy"]) < ring["radius"] - _BAND


def _clear_from_any_ring(px: float, py: float, rings: list[dict], clear: float) -> bool:
    for ring in rings:
        d = _dist(px, py, ring["cx"], ring["cy"])
        # not inside this ring's pocket and clear of its band
        if d < ring["radius"] - _BAND + clear:
            return False
        if abs(d - ring["radius"]) < clear:
            return False
    return True


def _dot_in_ring(px: float, py: float, ring: dict, clear: float) -> bool:
    d = _dist(px, py, ring["cx"], ring["cy"])
    return d < ring["radius"] - _BAND - _DOT_R - clear


def sample_scene(
    rng: random.Random,
    width: int = 500,
    height: int = 500,
) -> dict:
    """Sample a scene with exactly one pair of dots in the same white region."""
    radii = [_RING_RADIUS + rng.uniform(-8.0, 8.0) for _ in range(3)]
    centers = [
        (slot[0] + rng.uniform(-16.0, 16.0), slot[1] + rng.uniform(-16.0, 16.0))
        for slot in _RING_SLOTS
    ]
    rings = [
        {"cx": int(round(centers[i][0])), "cy": int(round(centers[i][1])),
         "radius": int(round(radii[i]))}
        for i in range(3)
    ]

    host = rng.randrange(3)
    share = rng.choice(PAIRS)
    share_colors = list(share)
    outside_color = next(c for c in _COLORS if c not in share_colors)

    # place the two sharing dots inside the host ring pocket, at opposite sides
    host_center = (float(centers[host][0]), float(centers[host][1]))
    host_radius = float(radii[host])
    maxd = host_radius - _BAND - _DOT_R - 9.0
    d = rng.uniform(_DOT_R + 6.0, max( _DOT_R + 6.0, maxd))
    t0 = rng.uniform(0.0, 2.0 * math.pi)
    dots: list[dict] = []
    for k, color in enumerate(share_colors):
        ang = t0 + k * math.pi
        px = host_center[0] + d * math.cos(ang)
        py = host_center[1] + d * math.sin(ang)
        dots.append({"x": int(round(px)), "y": int(round(py)), "color": color})

    # place the odd dot in a *different* white region
    odd_mode = rng.choice(["outside", "ring"])
    ox = oy = 0.0
    if odd_mode == "ring":
        odd_ring = rng.choice([i for i in range(3) if i != host])
        rc = (float(centers[odd_ring][0]), float(centers[odd_ring][1]))
        rr = float(radii[odd_ring])
        od = rng.uniform(_DOT_R + 6.0, max(_DOT_R + 6.0, rr - _BAND - _DOT_R - 9.0))
        ta = rng.uniform(0.0, 2.0 * math.pi)
        ox = rc[0] + od * math.cos(ta)
        oy = rc[1] + od * math.sin(ta)
    else:
        # place outside all rings, in the border-connected corner region
        for _ in range(200):
            ox = rng.uniform(18.0, width - 18.0)
            oy = rng.uniform(18.0, height - 18.0)
            if _clear_from_any_ring(ox, oy, rings, _DOT_R + 14.0):
                break
    dots.append({"x": int(round(ox)), "y": int(round(oy)), "color": outside_color})

    return {"width": width, "height": height, "rings": rings, "dots": dots}


def analytic_gold(scene: dict) -> str:
    """The pair of coloured dots lying in the same connected white region."""
    rings = scene["rings"]
    dots = scene["dots"]
    groups: dict[object, list[str]] = defaultdict(list)
    for dot in dots:
        label: object = -1
        for i, ring in enumerate(rings):
            if _inside_pocket(dot["x"], dot["y"], ring):
                label = i
                break
        groups[label].append(dot["color"])
    for members in groups.values():
        if len(members) == 2:
            return _pair_label(*members)
    return _pair_label(*PAIRS[0])


def margin(scene: dict) -> float:
    """Robustness margin: the minimum dot-to-ring-band clearance.

    Every dot must end up clearly inside or clearly outside every ring pocket,
    so the connected-white-region membership is stable under small raster
    perturbation. The margin is the minimum, over dots and rings, of the
    distance between a dot and the midpoint of a ring's black band.
    """
    rings = scene["rings"]
    dots = scene["dots"]
    best: float = float("inf")
    for dot in dots:
        for ring in rings:
            d = _dist(dot["x"], dot["y"], ring["cx"], ring["cy"])
            best = min(best, abs(d - ring["radius"]))
    return float(best)


def is_quarantined(scene: dict) -> bool:
    """Quarantine marginal scenes: a dot too close to a ring band, two rings
    whose bands nearly touch, or a ring too close to the picture border (which
    would break the sealing of its pocket)."""
    if margin(scene) < _QUARANTINE_MARGIN:
        return True
    rings = scene["rings"]
    for i in range(len(rings)):
        for j in range(i + 1, len(rings)):
            d = _dist(
                rings[i]["cx"], rings[i]["cy"],
                rings[j]["cx"], rings[j]["cy"],
            )
            if d < rings[i]["radius"] + rings[j]["radius"] + _BAND + 10.0:
                return True
    pad = 16.0
    for ring in rings:
        if (
            ring["cx"] - ring["radius"] - pad < 0
            or ring["cy"] - ring["radius"] - pad < 0
            or ring["cx"] + ring["radius"] + pad > scene["width"]
            or ring["cy"] + ring["radius"] + pad > scene["height"]
        ):
            return True
    return False


def render(scene: dict, supersample: int = 4) -> Image.Image:
    """Render R(z) deterministically. Pure function of the scene dict."""
    width = int(scene["width"])
    height = int(scene["height"])
    rings = scene["rings"]
    dots = scene["dots"]
    ss = supersample
    img = Image.new("RGB", (width * ss, height * ss), WHITE)
    draw = ImageDraw.Draw(img)
    for ring in rings:
        rx = ring["cx"] * ss
        ry = ring["cy"] * ss
        rr = ring["radius"] * ss
        rb = int(round(_BAND * ss))
        draw.ellipse([rx - rr, ry - rr, rx + rr, ry + rr], fill=_RING)
        draw.ellipse([rx - rr + rb, ry - rr + rb, rx + rr - rb, ry + rr - rb], fill=WHITE)
    for dot in dots:
        dx = dot["x"] * ss
        dy = dot["y"] * ss
        dr = int(round(_DOT_R * ss))
        draw.ellipse([dx - dr, dy - dr, dx + dr, dy + dr], fill=_COLOR_RGB[dot["color"]])
    if ss > 1:
        img = img.resize((width, height), Image.LANCZOS)
    return img


def latent_symmetries(scene: dict) -> Iterator[dict]:
    """No nontrivial pixel-identical transform: every stored field redraws.

    Which colour sits in the sharing pair is a label-transforming relabel (it
    changes the image) that the decision tracks exactly; declared not_applicable.
    """
    return iter(())
