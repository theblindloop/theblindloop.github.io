"""Scene-free pixel arm: recover the symbol from the rendered PNG alone.

This module receives only the final image and the public question contract.
It locates the two printed frames, reads the legend's colour-to-stroke pairing,
transcribes the plan, performs the declared substitution and matches the
resulting board against the five symbol templates transcribed from the
published question text.
"""

from __future__ import annotations

from collections import Counter

import numpy as np
from PIL import Image

ORACLE_VERSION = "legend_expand_reveal-oracle-0.1.0"
ABSTAIN = "abstain"

_LEGEND_FRAME = (30, 60, 220)
_PLAN_FRAME = (20, 140, 60)

# Symbol templates transcribed from the five verbal descriptions in the
# published question. b = bar, p = post, k = back-slash, f = fore-slash.
_ROWS = {
    "chevron": ("bbbb", "kbbf", "bkfb", "bbbb"),
    "cross": ("kbbf", "bkfb", "bfkb", "fbbk"),
    "ladder": ("pbbp", "pbbp", "pbbp", "pbbp"),
    "pillar": ("bppb", "bppb", "bppb", "bppb"),
    "zigzag": ("kkbb", "bbff", "kkbb", "bbff"),
}
_LONG = {"b": "bar", "p": "post", "k": "back", "f": "fore"}
_TEMPLATES = {
    name: [_LONG[ch] for row in rows for ch in row] for name, rows in _ROWS.items()
}


def _frame_box(arr: np.ndarray, colour: tuple[int, int, int]):
    mask = (
        (arr[:, :, 0] == colour[0])
        & (arr[:, :, 1] == colour[1])
        & (arr[:, :, 2] == colour[2])
    )
    if int(mask.sum()) < 400:
        return None
    ys, xs = np.nonzero(mask)
    y0, y1 = int(ys.min()), int(ys.max())
    x0, x1 = int(xs.min()), int(xs.max())
    mid = (y0 + y1) // 2
    thickness = 0
    while thickness < 12 and mask[mid, x0 + thickness]:
        thickness += 1
    if thickness < 1:
        return None
    inner = (x0 + thickness, y0 + thickness, x1 - thickness, y1 - thickness)
    if inner[2] - inner[0] < 40 or inner[3] - inner[1] < 40:
        return None
    return inner


def _dominant_colour(patch: np.ndarray):
    flat = patch.reshape(-1, 3)
    keep = flat[(flat.max(axis=1).astype(int) - flat.min(axis=1).astype(int)) > 45]
    if keep.shape[0] < 50:
        return None
    counts = Counter(map(tuple, keep.tolist()))
    return counts.most_common(1)[0][0]


def _stroke_kind(patch: np.ndarray):
    dark = patch.max(axis=2) < 90
    ys, xs = np.nonzero(dark)
    if xs.size < 40:
        return None
    width = float(xs.max() - xs.min() + 1)
    height = float(ys.max() - ys.min() + 1)
    if width >= 2.0 * height:
        return "bar"
    if height >= 2.0 * width:
        return "post"
    cov = float(np.mean((xs - xs.mean()) * (ys - ys.mean())))
    if abs(cov) < 1e-6:
        return None
    return "back" if cov > 0 else "fore"


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    legend = _frame_box(arr, _LEGEND_FRAME)
    plan = _frame_box(arr, _PLAN_FRAME)
    if legend is None or plan is None or legend[0] >= plan[0]:
        return ABSTAIN

    lx0, ly0, lx1, ly1 = legend
    height = ly1 - ly0 + 1
    split = lx0 + (lx1 - lx0 + 1) // 2
    pairs = []
    for band in range(4):
        top = ly0 + int(round(height * band / 4.0))
        bottom = ly0 + int(round(height * (band + 1) / 4.0))
        if bottom - top < 20:
            return ABSTAIN
        colour = _dominant_colour(arr[top:bottom, lx0:split])
        kind = _stroke_kind(arr[top:bottom, split:lx1 + 1])
        if colour is None or kind is None:
            return ABSTAIN
        pairs.append((np.array(colour, dtype=float), kind))
    if len({kind for _, kind in pairs}) != 4:
        return ABSTAIN

    px0, py0, px1, py1 = plan
    cell_w = (px1 - px0 + 1) / 4.0
    cell_h = (py1 - py0 + 1) / 4.0
    if cell_w < 20 or cell_h < 20:
        return ABSTAIN
    board = []
    for row in range(4):
        for col in range(4):
            cx = px0 + int(cell_w * (col + 0.5))
            cy = py0 + int(cell_h * (row + 0.5))
            patch = arr[cy - 10:cy + 11, cx - 10:cx + 11]
            colour = _dominant_colour(patch)
            if colour is None:
                return ABSTAIN
            here = np.array(colour, dtype=float)
            ranked = sorted(
                (float(np.linalg.norm(here - ref)), kind) for ref, kind in pairs
            )
            if ranked[0][0] > 60.0 or ranked[1][0] - ranked[0][0] < 40.0:
                return ABSTAIN
            board.append(ranked[0][1])

    scores = sorted(
        (sum(1 for a, b in zip(board, tiles) if a != b), name)
        for name, tiles in _TEMPLATES.items()
    )
    return scores[0][1]
