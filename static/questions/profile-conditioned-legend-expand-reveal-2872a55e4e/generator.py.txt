"""Legend-keyed tile expansion world.

A four-by-four plan of coloured cells and a printed legend that pairs each
colour with one stroke tile.  Substituting every plan cell by its legend tile
builds a four-by-four board of strokes that matches one of five named symbols.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "legend_expand_reveal-renderer-0.1.0"

TILE_TYPES = ("bar", "post", "back", "fore")

# Stroke masks on the three-by-three cell block of one tile.
TILE_MASKS: dict[str, tuple[tuple[int, int], ...]] = {
    "bar": ((1, 0), (1, 1), (1, 2)),
    "post": ((0, 1), (1, 1), (2, 1)),
    "back": ((0, 0), (1, 1), (2, 2)),
    "fore": ((0, 2), (1, 1), (2, 0)),
}

_CHAR = {"b": "bar", "p": "post", "k": "back", "f": "fore"}

_TEMPLATE_ROWS: dict[str, tuple[str, ...]] = {
    "chevron": ("bbbb", "kbbf", "bkfb", "bbbb"),
    "cross": ("kbbf", "bkfb", "bfkb", "fbbk"),
    "ladder": ("pbbp", "pbbp", "pbbp", "pbbp"),
    "pillar": ("bppb", "bppb", "bppb", "bppb"),
    "zigzag": ("kkbb", "bbff", "kkbb", "bbff"),
}

SYMBOLS = tuple(sorted(_TEMPLATE_ROWS))

TEMPLATES: dict[str, tuple[str, ...]] = {
    name: tuple(_CHAR[ch] for row in rows for ch in row)
    for name, rows in _TEMPLATE_ROWS.items()
}

# Swatch colours; the plan cell colour is a pure relabelling of the tile type.
POOL = ((214, 40, 40), (240, 160, 20), (140, 60, 200), (0, 150, 160))
FRAME_LEGEND = (30, 60, 220)
FRAME_PLAN = (20, 140, 60)
INK = (15, 15, 15)
PAPER = (255, 255, 255)

CANVAS = (720, 420)
FRAME_W = 3
LEGEND_BOX = (30, 30, 270, 364)  # outer rectangle, inclusive corners
PLAN_BOX = (326, 30, 668, 372)
BAND_H = 82  # legend interior height 328 / 4
PLAN_CELL = 84  # plan interior 336 / 4


def _distance(left: tuple[str, ...], right: tuple[str, ...]) -> int:
    return sum(1 for a, b in zip(left, right) if a != b)


def _perturb(base: str, other: str, steps: int) -> list[str]:
    """Move `steps` tiles of the base template towards another template."""
    tiles = list(TEMPLATES[base])
    target = TEMPLATES[other]
    differing = [i for i in range(16) if tiles[i] != target[i]]
    for index in differing[:steps]:
        tiles[index] = target[index]
    return tiles


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    base = SYMBOLS[seed % len(SYMBOLS)]
    other = rng.choice([name for name in SYMBOLS if name != base])
    gap = _distance(TEMPLATES[base], TEMPLATES[other])
    mode = ("clean", "clean", "near", "boundary")[seed % 4]
    if mode == "clean":
        steps = 0
    elif mode == "near":
        steps = max(1, gap // 4)
    else:
        steps = (gap - 2) // 2
    tiles = _perturb(base, other, steps)

    colour_perm = list(range(len(POOL)))
    rng.shuffle(colour_perm)
    legend_codes = list(range(len(TILE_TYPES)))
    rng.shuffle(legend_codes)
    plan_codes = [TILE_TYPES.index(tile) for tile in tiles]
    return {
        "plan_codes": plan_codes,
        "legend_codes": legend_codes,
        "colour_perm": colour_perm,
        "jitter": [rng.randint(-8, 8), rng.randint(-8, 8)],
    }


def _tiles(scene: dict) -> tuple[str, ...]:
    return tuple(TILE_TYPES[code % len(TILE_TYPES)] for code in scene["plan_codes"])


def _ranked(scene: dict) -> list[tuple[int, str]]:
    tiles = _tiles(scene)
    return sorted((_distance(tiles, TEMPLATES[name]), name) for name in SYMBOLS)


def analytic_gold(scene: dict) -> str:
    return _ranked(scene)[0][1]


def margin(scene: dict) -> float:
    ranked = _ranked(scene)
    return float(ranked[1][0] - ranked[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical, gold-preserving relabellings of the latent scene."""
    plan_shift = dict(scene)
    plan_shift["plan_codes"] = [code + len(TILE_TYPES) for code in scene["plan_codes"]]

    legend_shift = dict(scene)
    legend_shift["legend_codes"] = [
        code + len(TILE_TYPES) for code in scene["legend_codes"]
    ]
    return [("plan_code_shift", plan_shift), ("legend_code_shift", legend_shift)]


def _draw_tile(draw: ImageDraw.ImageDraw, x: int, y: int, cell: int, tile: str) -> None:
    for row, col in TILE_MASKS[tile]:
        left = x + col * cell
        top = y + row * cell
        draw.rectangle([left, top, left + cell - 1, top + cell - 1], fill=INK)


def render(scene: dict) -> Image.Image:
    dx, dy = scene["jitter"]
    image = Image.new("RGB", CANVAS, PAPER)
    draw = ImageDraw.Draw(image)

    colours = [POOL[index] for index in scene["colour_perm"]]

    lx0, ly0, lx1, ly1 = (LEGEND_BOX[0] + dx, LEGEND_BOX[1] + dy,
                          LEGEND_BOX[2] + dx, LEGEND_BOX[3] + dy)
    draw.rectangle([lx0, ly0, lx1, ly1], outline=FRAME_LEGEND, width=FRAME_W)
    for band in range(len(TILE_TYPES)):
        tile = TILE_TYPES[scene["legend_codes"][band] % len(TILE_TYPES)]
        top = ly0 + FRAME_W + band * BAND_H
        swatch_x = lx0 + 18
        swatch_y = top + 11
        draw.rectangle(
            [swatch_x, swatch_y, swatch_x + 59, swatch_y + 59],
            fill=colours[TILE_TYPES.index(tile)],
        )
        _draw_tile(draw, lx0 + 140, top + 14, 18, tile)

    px0, py0, px1, py1 = (PLAN_BOX[0] + dx, PLAN_BOX[1] + dy,
                          PLAN_BOX[2] + dx, PLAN_BOX[3] + dy)
    draw.rectangle([px0, py0, px1, py1], outline=FRAME_PLAN, width=FRAME_W)
    tiles = _tiles(scene)
    for index, tile in enumerate(tiles):
        row, col = divmod(index, 4)
        left = px0 + FRAME_W + col * PLAN_CELL
        top = py0 + FRAME_W + row * PLAN_CELL
        draw.rectangle(
            [left + 6, top + 6, left + PLAN_CELL - 7, top + PLAN_CELL - 7],
            fill=colours[TILE_TYPES.index(tile)],
        )
    return image
