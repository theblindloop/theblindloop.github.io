"""Independent inverse arm: reconstruct the minimum circle from RGB pixels only."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        pixels = []
        while stack:
            cy, cx = stack.pop()
            pixels.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        groups.append(np.asarray(pixels, dtype=float))
    return groups


def _centers(image: Image.Image) -> list[tuple[float, float]]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    mask = (arr.max(axis=2) < 105) & ((arr.max(axis=2) - arr.min(axis=2)) < 30)
    groups = [group for group in _components(mask) if len(group) > 70]
    if len(groups) != 14:
        raise ValueError(f"expected 14 visible dot components, found {len(groups)}")
    return [(float(group[:, 1].mean()), float(group[:, 0].mean())) for group in groups]


def _pair_circle(a, b):
    x = (a[0] + b[0]) / 2.0
    y = (a[1] + b[1]) / 2.0
    return (x, y, math.hypot(a[0] - b[0], a[1] - b[1]) / 2.0)


def _triple_circle(a, b, c):
    cross = 2.0 * (a[0] * (b[1] - c[1]) + b[0] * (c[1] - a[1]) + c[0] * (a[1] - b[1]))
    if abs(cross) < 1e-7:
        return None
    a2 = a[0] ** 2 + a[1] ** 2
    b2 = b[0] ** 2 + b[1] ** 2
    c2 = c[0] ** 2 + c[1] ** 2
    x = (a2 * (b[1] - c[1]) + b2 * (c[1] - a[1]) + c2 * (a[1] - b[1])) / cross
    y = (a2 * (c[0] - b[0]) + b2 * (a[0] - c[0]) + c2 * (b[0] - a[0])) / cross
    return (x, y, math.hypot(x - a[0], y - a[1]))


def _solve(points: list[tuple[float, float]]) -> tuple[float, float, float]:
    legal = []
    n = len(points)
    for i in range(n):
        for j in range(i + 1, n):
            circle = _pair_circle(points[i], points[j])
            if all(math.hypot(px - circle[0], py - circle[1]) <= circle[2] + 0.35 for px, py in points):
                legal.append(circle)
    for i in range(n):
        for j in range(i + 1, n):
            for k in range(j + 1, n):
                circle = _triple_circle(points[i], points[j], points[k])
                if circle is not None and all(math.hypot(px - circle[0], py - circle[1]) <= circle[2] + 0.35 for px, py in points):
                    legal.append(circle)
    if not legal:
        raise ValueError("could not reconstruct an enclosing circle")
    return min(legal, key=lambda item: item[2])


def decision_from_image(image: Image.Image) -> str:
    points = _centers(image)
    cx, cy, radius = _solve(points)
    deficits = [radius - math.hypot(x - cx, y - cy) for x, y in points]
    support = sum(value <= 1.2 for value in deficits)
    interior = [value for value in deficits if value > 1.2]
    if support not in (2, 3) or not interior or min(interior) < 15.0:
        raise ValueError("ambiguous pixel support set")
    return "two" if support == 2 else "three"
