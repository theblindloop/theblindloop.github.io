"""Deterministic point-field renderer for a minimum-enclosing-circle task."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
DOT_RADIUS = 7
DOT_COUNT = 14
INK = (27, 31, 35)
BACKGROUND = (248, 247, 243)


def _circle_from_pair(a: tuple[float, float], b: tuple[float, float]):
    cx = (a[0] + b[0]) / 2.0
    cy = (a[1] + b[1]) / 2.0
    return (cx, cy, math.hypot(a[0] - b[0], a[1] - b[1]) / 2.0)


def _circle_from_triple(a, b, c):
    ax, ay = a
    bx, by = b
    cx, cy = c
    d = 2.0 * (ax * (by - cy) + bx * (cy - ay) + cx * (ay - by))
    if abs(d) < 1e-8:
        return None
    aa = ax * ax + ay * ay
    bb = bx * bx + by * by
    cc = cx * cx + cy * cy
    ux = (aa * (by - cy) + bb * (cy - ay) + cc * (ay - by)) / d
    uy = (aa * (cx - bx) + bb * (ax - cx) + cc * (bx - ax)) / d
    return (ux, uy, math.hypot(ux - ax, uy - ay))


def _minimum_circle(points: list[list[float]]) -> tuple[float, float, float]:
    pts = [(float(x), float(y)) for x, y in points]
    candidates = []
    for i in range(len(pts)):
        for j in range(i + 1, len(pts)):
            candidates.append(_circle_from_pair(pts[i], pts[j]))
    for i in range(len(pts)):
        for j in range(i + 1, len(pts)):
            for k in range(j + 1, len(pts)):
                circle = _circle_from_triple(pts[i], pts[j], pts[k])
                if circle is not None:
                    candidates.append(circle)
    valid = [
        circle for circle in candidates
        if all(math.hypot(x - circle[0], y - circle[1]) <= circle[2] + 1e-5 for x, y in pts)
    ]
    if not valid:
        raise ValueError("point set has no enclosing-circle candidate")
    return min(valid, key=lambda item: item[2])


def _measure(points: list[list[float]]) -> tuple[int, float, tuple[float, float, float]]:
    circle = _minimum_circle(points)
    cx, cy, radius = circle
    deficits = sorted(radius - math.hypot(float(x) - cx, float(y) - cy) for x, y in points)
    support = sum(value <= 0.05 for value in deficits)
    if support not in (2, 3):
        raise ValueError(f"expected two or three active supports, found {support}")
    interior = [value for value in deficits if value > 0.05]
    clearance = min(interior) if interior else 0.0
    return support, clearance, circle


def _point(cx: float, cy: float, radius: float, angle: float) -> list[float]:
    return [cx + radius * math.cos(angle), cy + radius * math.sin(angle)]


def sample_scene(seed: int) -> dict:
    """Balance labels by seed parity; all nuisance geometry is sampled independently."""
    rng = random.Random(int(seed))
    support_count = 2 if int(seed) % 2 == 0 else 3
    radius = rng.uniform(132.0, 166.0)
    cx = rng.uniform(radius + 24.0, WIDTH - radius - 24.0)
    cy = rng.uniform(radius + 24.0, HEIGHT - radius - 24.0)
    phase = rng.uniform(0.0, 2.0 * math.pi)
    if support_count == 2:
        dots = [_point(cx, cy, radius, phase), _point(cx, cy, radius, phase + math.pi)]
    else:
        offsets = [rng.uniform(-0.12, 0.12) for _ in range(3)]
        dots = [
            _point(cx, cy, radius, phase + index * 2.0 * math.pi / 3.0 + offsets[index])
            for index in range(3)
        ]

    while len(dots) < DOT_COUNT:
        radial = radius * math.sqrt(rng.uniform(0.025, 0.44))
        angle = rng.uniform(0.0, 2.0 * math.pi)
        candidate = _point(cx, cy, radial, angle)
        if all(math.hypot(candidate[0] - x, candidate[1] - y) >= 25.0 for x, y in dots):
            dots.append(candidate)
    rng.shuffle(dots)
    scene = {"width": WIDTH, "height": HEIGHT, "dots": dots}
    measured, clearance, _ = _measure(dots)
    if measured != support_count or clearance < 38.0:
        raise RuntimeError("constructed point set lost its intended robust support")
    return scene


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    dots = scene["dots"]
    scale = 4
    image = Image.new("RGB", (width * scale, height * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    rr = DOT_RADIUS * scale
    for x, y in dots:
        xx, yy = float(x) * scale, float(y) * scale
        draw.ellipse((xx - rr, yy - rr, xx + rr, yy + rr), fill=INK)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    support, _, _ = _measure(scene["dots"])
    return "two" if support == 2 else "three"


def margin(scene: dict) -> float:
    _, clearance, _ = _measure(scene["dots"])
    return float(clearance)


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 20.0
    except ValueError:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["dots"] = list(reversed(twin["dots"]))
    return [("reverse_unordered_dots", twin)]
