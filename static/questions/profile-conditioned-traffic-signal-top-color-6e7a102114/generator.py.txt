"""Deterministic traffic-signal renderer and analytic arm."""

from __future__ import annotations

import itertools
import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 480
HEIGHT = 480
AA = 3
COLORS = {
    "red": (220, 45, 55),
    "yellow": (245, 190, 35),
    "green": (36, 170, 92),
}
PERMUTATIONS = tuple(itertools.permutations(("red", "yellow", "green")))


def sample_scene(seed: int) -> dict:
    """Sample matched styling; the seed's residue cycles through all six orders."""
    rng = random.Random(int(seed))
    radius = rng.randint(38, 48)
    center_gap = rng.randint(104, 122)
    center_x = rng.randint(214, 266)
    total_height = 2 * radius + 2 * center_gap
    top_y = rng.randint(44 + radius, HEIGHT - 44 - total_height + radius)
    return {
        "lamp_colors": list(PERMUTATIONS[int(seed) % len(PERMUTATIONS)]),
        "housing_tone": rng.choice([28, 34, 40, 46]),
        "background_tone": rng.choice([238, 244, 250]),
        "center_x": center_x,
        "top_y": top_y,
        "radius": radius,
        "center_gap": center_gap,
        "render_tag": rng.choice(["control", "replicate", "audit"]),
    }


def _scaled_box(cx: int, cy: int, half_w: int, half_h: int) -> tuple[int, int, int, int]:
    return tuple(v * AA for v in (cx - half_w, cy - half_h, cx + half_w, cy + half_h))


def render(scene: dict) -> Image.Image:
    """Render a familiar vertical signal; render_tag is intentionally pixel-inert."""
    bg = int(scene["background_tone"])
    image = Image.new("RGB", (WIDTH * AA, HEIGHT * AA), (bg, bg, bg))
    draw = ImageDraw.Draw(image)
    cx = int(scene["center_x"])
    top = int(scene["top_y"])
    radius = int(scene["radius"])
    gap = int(scene["center_gap"])
    ys = [top + i * gap for i in range(3)]
    housing_half_w = radius + 24
    housing_half_h = gap + radius + 24
    tone = int(scene["housing_tone"])
    outline = (12, 16, 20)
    draw.rounded_rectangle(
        _scaled_box(cx, ys[1], housing_half_w, housing_half_h),
        radius=22 * AA,
        fill=(tone, tone + 4, tone + 8),
        outline=outline,
        width=5 * AA,
    )
    # A short neutral post and base make the icon unambiguously signal-like.
    base_y = ys[2] + radius + 24
    draw.rectangle(
        tuple(v * AA for v in (cx - 9, base_y, cx + 9, min(base_y + 30, HEIGHT - 20))),
        fill=outline,
    )
    for name, cy in zip(scene["lamp_colors"], ys, strict=True):
        # Neutral bezel is identical in all slots and all variants.
        draw.ellipse(_scaled_box(cx, cy, radius + 7, radius + 7), fill=(10, 13, 16))
        draw.ellipse(_scaled_box(cx, cy, radius, radius), fill=COLORS[name])
        # Fixed white glint prevents total-ink equality from being the only cue.
        glint = max(5, radius // 6)
        gx, gy = cx - radius // 3, cy - radius // 3
        draw.ellipse(_scaled_box(gx, gy, glint, glint), fill=(250, 250, 250))
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["lamp_colors"][0])


def margin(scene: dict) -> float:
    """Runner-up decision margin: vertical centroid separation in final pixels."""
    return float(scene["center_gap"])


def is_quarantined(scene: dict) -> bool:
    return bool(scene["center_gap"] < 10 or scene["radius"] < 12)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["render_tag"] = {
        "control": "replicate",
        "replicate": "audit",
        "audit": "control",
    }.get(str(scene["render_tag"]), "control")
    return [("pixel_inert_render_tag_cycle", twin)]
