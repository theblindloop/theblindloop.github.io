"""Independent PNG-only inverse arm for the traffic-signal question."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_PROTOTYPES = {
    "red": np.array((220, 45, 55), dtype=np.int32),
    "yellow": np.array((245, 190, 35), dtype=np.int32),
    "green": np.array((36, 170, 92), dtype=np.int32),
}
_MAX_COLOR_DISTANCE = 62.0
_MIN_AREA = 1800


def _largest_component(mask: np.ndarray) -> tuple[int, float, float]:
    """Return area and centroid of the largest 4-connected true region."""
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    best = (0, 0.0, 0.0)
    for y0, x0 in zip(*np.nonzero(mask & ~seen), strict=True):
        if seen[y0, x0]:
            continue
        queue = deque([(int(y0), int(x0))])
        seen[y0, x0] = True
        area = 0
        sx = 0.0
        sy = 0.0
        while queue:
            y, x = queue.popleft()
            area += 1
            sx += x
            sy += y
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if area > best[0]:
            best = (area, sx / area, sy / area)
    return best


def decision_from_image(image: Image.Image) -> str:
    """Measure all colored lamp regions and report the smallest y centroid."""
    # int32 avoids overflow when three squared 8-bit channel deltas are summed.
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    recovered: list[tuple[float, str, int, float]] = []
    for name, prototype in _PROTOTYPES.items():
        distance = np.sqrt(np.sum((arr - prototype) ** 2, axis=2))
        area, cx, cy = _largest_component(distance <= _MAX_COLOR_DISTANCE)
        if area < _MIN_AREA:
            raise ValueError(f"missing or too-small {name} lamp region")
        recovered.append((cy, name, area, cx))
    recovered.sort()
    if recovered[1][0] - recovered[0][0] < 10.0:
        raise ValueError("top two colored regions have ambiguous vertical order")
    xs = [item[3] for item in recovered]
    if max(xs) - min(xs) > 18.0:
        raise ValueError("colored regions do not form one vertical signal")
    return recovered[0][1]
