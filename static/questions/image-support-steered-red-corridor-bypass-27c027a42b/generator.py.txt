"""Deterministic latent scene and raster renderer for Red Corridor Bypass."""

from __future__ import annotations

import random
from collections import deque
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
N = 5
BASE = 64
STEP = 96

PALETTES = (
    {
        "background": (248, 246, 240),
        "corridor": (35, 70, 83),
        "neutral": (181, 215, 210),
        "start": (41, 168, 92),
        "goal": (238, 173, 43),
        "red": (205, 67, 70),
    },
    {
        "background": (244, 246, 249),
        "corridor": (54, 61, 87),
        "neutral": (188, 213, 228),
        "start": (42, 165, 105),
        "goal": (235, 166, 52),
        "red": (205, 70, 76),
    },
)


def _edge(a: tuple[int, int], b: tuple[int, int]) -> tuple[tuple[int, int], tuple[int, int]]:
    return tuple(sorted((a, b)))  # type: ignore[return-value]


def _transform(point: tuple[int, int], mode: int) -> tuple[int, int]:
    r, c = point
    if mode >= 4:
        c = N - 1 - c
    for _ in range(mode % 4):
        r, c = c, N - 1 - r
    return r, c


def _transform_edge(
    edge: tuple[tuple[int, int], tuple[int, int]], mode: int
) -> tuple[tuple[int, int], tuple[int, int]]:
    return _edge(_transform(edge[0], mode), _transform(edge[1], mode))


def sample_scene(seed: int) -> dict:
    """Build one dense network that contains both bridge and bypass query edges."""
    rng = random.Random(int(seed))
    edges: set[tuple[tuple[int, int], tuple[int, int]]] = set()

    # Five tall, internally connected strips.
    for r in range(N - 1):
        for c in range(N):
            edges.add(_edge((r, c), (r + 1, c)))
    # Dense two-column lobes at the left and right.
    for r in range(N):
        edges.add(_edge((r, 0), (r, 1)))
        edges.add(_edge((r, 3), (r, 4)))

    single_left = bool(rng.getrandbits(1))
    extreme_rows = [0, N - 1]
    single_row = rng.choice(extreme_rows)
    dual_rows = extreme_rows
    single_cut = 1 if single_left else 2
    dual_cut = 2 if single_left else 1
    single_edge = _edge((single_row, single_cut), (single_row, single_cut + 1))
    dual_edges = [
        _edge((row, dual_cut), (row, dual_cut + 1)) for row in dual_rows
    ]
    edges.add(single_edge)
    edges.update(dual_edges)

    # Consecutive integer seeds are exactly class-balanced.
    wants_bypass = int(seed) % 2 == 0
    red_edge = rng.choice(dual_edges) if wants_bypass else single_edge
    start, goal = red_edge
    if rng.getrandbits(1):
        start, goal = goal, start

    mode = rng.randrange(8)
    transformed_edges = sorted(
        ([list(a), list(b)] for a, b in (_transform_edge(e, mode) for e in edges)),
        key=lambda item: (item[0], item[1]),
    )
    transformed_red = _transform_edge(red_edge, mode)
    start_t = _transform(start, mode)
    goal_t = _transform(goal, mode)
    return {
        "edges": transformed_edges,
        "red_edge": [list(transformed_red[0]), list(transformed_red[1])],
        "start": list(start_t),
        "goal": list(goal_t),
        "offset_x": rng.randint(-7, 7),
        "offset_y": rng.randint(-7, 7),
        "palette_id": rng.randrange(len(PALETTES)),
    }


def _point(node: Iterable[int], offset_x: int, offset_y: int) -> tuple[int, int]:
    r, c = (int(value) for value in node)
    return BASE + c * STEP + offset_x, BASE + r * STEP + offset_y


def _normalized_edges(scene: dict) -> set[tuple[tuple[int, int], tuple[int, int]]]:
    return {
        _edge(tuple(int(v) for v in pair[0]), tuple(int(v) for v in pair[1]))
        for pair in scene["edges"]
    }


def _red(scene: dict) -> tuple[tuple[int, int], tuple[int, int]]:
    return _edge(
        tuple(int(v) for v in scene["red_edge"][0]),
        tuple(int(v) for v in scene["red_edge"][1]),
    )


def render(scene: dict) -> Image.Image:
    """Render a broad, connected corridor component with visible node studs."""
    edges = scene["edges"]
    red_edge = scene["red_edge"]
    start = scene["start"]
    goal = scene["goal"]
    ox = int(scene["offset_x"])
    oy = int(scene["offset_y"])
    palette = PALETTES[int(scene["palette_id"]) % len(PALETTES)]
    ss = 3
    image = Image.new("RGB", (WIDTH * ss, HEIGHT * ss), palette["background"])
    draw = ImageDraw.Draw(image)

    def p(node: Iterable[int]) -> tuple[int, int]:
        x, y = _point(node, ox, oy)
        return x * ss, y * ss

    red_norm = _edge(tuple(red_edge[0]), tuple(red_edge[1]))
    for pair in edges:
        draw.line([p(pair[0]), p(pair[1])], fill=palette["corridor"], width=42 * ss)
    for r in range(N):
        for c in range(N):
            x, y = p((r, c))
            radius = 27 * ss
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=palette["corridor"])

    draw.line([p(red_norm[0]), p(red_norm[1])], fill=palette["red"], width=30 * ss)
    start_t = tuple(start)
    goal_t = tuple(goal)
    for r in range(N):
        for c in range(N):
            node = (r, c)
            color = palette["start"] if node == start_t else palette["goal"] if node == goal_t else palette["neutral"]
            radius = (10 if node in (start_t, goal_t) else 7) * ss
            x, y = p(node)
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def _reachable(scene: dict) -> bool:
    edges = _normalized_edges(scene)
    edges.discard(_red(scene))
    start = tuple(int(v) for v in scene["start"])
    goal = tuple(int(v) for v in scene["goal"])
    graph: dict[tuple[int, int], set[tuple[int, int]]] = {}
    for a, b in edges:
        graph.setdefault(a, set()).add(b)
        graph.setdefault(b, set()).add(a)
    queue = deque([start])
    seen = {start}
    while queue:
        node = queue.popleft()
        if node == goal:
            return True
        for nxt in graph.get(node, ()):
            if nxt not in seen:
                seen.add(nxt)
                queue.append(nxt)
    return False


def analytic_gold(scene: dict) -> str:
    # The four accessed fields are the complete declared label input.
    _ = scene["edges"], scene["red_edge"], scene["start"], scene["goal"]
    return "yes" if _reachable(scene) else "no"


def margin(scene: dict) -> float:
    """Discrete runner-up margin: one graph-edge event is needed to flip reachability."""
    return 1.0


def is_quarantined(scene: dict) -> bool:
    edges = _normalized_edges(scene)
    red = _red(scene)
    start = tuple(int(v) for v in scene["start"])
    goal = tuple(int(v) for v in scene["goal"])
    return red not in edges or set(red) != {start, goal} or len(edges) != 33


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
