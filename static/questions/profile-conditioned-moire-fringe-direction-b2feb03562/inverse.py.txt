"""Image-only inverse arm: recover the broad interference bands from pixels.

Input is one RGB picture plus the public question contract.  The arm converts
the picture to a luminance field, removes its mean, apodises it with a raised
cosine window, and low-pass filters it with a Gaussian wide enough to erase any
drawn line family (their spacing is a handful of pixels) while keeping the broad
bands (tens of pixels wide).  The orientation of what survives is read from the
structure tensor of that smoothed field, and the surviving direction is named
with the four compass classes of the public contract.  If the smoothed field
carries almost no energy, or its orientation sits between two classes, the arm
abstains.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ABSTAIN = "abstain"
CLASS_ANGLES = {"horizontal": 0.0, "falling": 45.0, "vertical": 90.0, "rising": 135.0}
SIGMA_PX = 11.0
MIN_BAND_ENERGY = 0.010
MIN_ANGULAR_MARGIN_DEG = 8.0


def _angdist(a: float, b: float) -> float:
    d = (a - b) % 180.0
    return min(d, 180.0 - d)


def _lowpass(field: np.ndarray, sigma: float) -> np.ndarray:
    rows, cols = field.shape
    fy = np.fft.fftfreq(rows)[:, None]
    fx = np.fft.fftfreq(cols)[None, :]
    kernel = np.exp(-2.0 * (math.pi ** 2) * (sigma ** 2) * (fx * fx + fy * fy))
    return np.real(np.fft.ifft2(np.fft.fft2(field) * kernel))


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.float64) / 255.0
    lum = array.mean(axis=2)
    rows, cols = lum.shape
    if rows < 64 or cols < 64:
        return ABSTAIN

    wy = np.hanning(rows)[:, None]
    wx = np.hanning(cols)[None, :]
    window = wy * wx
    centred = (lum - float(lum.mean())) * window
    smooth = _lowpass(centred, SIGMA_PX)

    inner = np.zeros_like(window)
    my, mx = int(rows * 0.16), int(cols * 0.16)
    inner[my : rows - my, mx : cols - mx] = 1.0
    weight = inner * window
    total = float(weight.sum())
    if total <= 0.0:
        return ABSTAIN

    energy = math.sqrt(float((weight * smooth * smooth).sum()) / total)
    if energy < MIN_BAND_ENERGY:
        return ABSTAIN

    gy, gx = np.gradient(smooth)
    sxx = float((weight * gx * gx).sum())
    syy = float((weight * gy * gy).sum())
    sxy = float((weight * gx * gy).sum())
    if sxx + syy <= 0.0:
        return ABSTAIN

    # Dominant gradient direction; the bands run perpendicular to it.
    grad_deg = 0.5 * math.degrees(math.atan2(2.0 * sxy, sxx - syy))
    band_deg = (grad_deg + 90.0) % 180.0

    ranked = sorted(CLASS_ANGLES, key=lambda name: _angdist(band_deg, CLASS_ANGLES[name]))
    gap = _angdist(band_deg, CLASS_ANGLES[ranked[1]]) - _angdist(band_deg, CLASS_ANGLES[ranked[0]])
    if gap < MIN_ANGULAR_MARGIN_DEG:
        return ABSTAIN
    return ranked[0]
