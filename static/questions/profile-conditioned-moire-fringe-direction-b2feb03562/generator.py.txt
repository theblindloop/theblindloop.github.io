"""Two overlapping line gratings whose interference paints broad moire bands.

The latent scene stores a canvas, a background and an ink colour, and an
unordered pair of straight-line gratings, each with a tilt, a spacing, a duty
cycle and a phase.  Nothing in the scene stores the answer: the analytic gold
recomputes the beat wave vector k1 - k2 from the two stored gratings, and the
answer is the compass class of the direction along which the broad bands run.
That direction is carried by no drawn line, because both gratings are kept at
least 22 degrees away from it.
"""

from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image

ABSTAIN = "abstain"
CLASS_NAMES = ("horizontal", "falling", "vertical", "rising")
# Band direction in array coordinates (x to the right, y downward), degrees.
CLASS_ANGLES = {"horizontal": 0.0, "falling": 45.0, "vertical": 90.0, "rising": 135.0}

QUARANTINE_DEG = 20.0
SAMPLING_FLOOR_DEG = 30.0
MAX_TILT_JITTER_DEG = 7.0
MIN_GRATING_SEPARATION_DEG = 22.0

WIDTH = 384
HEIGHT = 384
BG_COLOR = (243, 241, 234)
INK_COLOR = (28, 30, 38)


def _angdist(a: float, b: float) -> float:
    """Distance between two undirected directions in degrees, in [0, 90]."""
    d = (a - b) % 180.0
    return min(d, 180.0 - d)


def _beat_vector(scene: dict) -> tuple[float, float] | None:
    """Wave vector of the broad bands, recomputed from the stored gratings."""
    gratings = scene["gratings"]
    if len(gratings) != 2:
        return None
    vecs = []
    for g in gratings:
        a = math.radians(float(g["tilt_deg"]))
        k = 2.0 * math.pi / float(g["pitch"])
        vecs.append((k * math.cos(a), k * math.sin(a)))
    return (vecs[0][0] - vecs[1][0], vecs[0][1] - vecs[1][1])


def band_direction_deg(scene: dict) -> float | None:
    """Direction the broad bands run, perpendicular to the beat wave vector."""
    beat = _beat_vector(scene)
    if beat is None:
        return None
    kx, ky = beat
    if math.hypot(kx, ky) < 2.0 * math.pi / 400.0:
        return None
    return (math.degrees(math.atan2(ky, kx)) + 90.0) % 180.0


def analytic_gold(scene: dict) -> str:
    theta = band_direction_deg(scene)
    if theta is None:
        return ABSTAIN
    ranked = sorted(CLASS_NAMES, key=lambda n: _angdist(theta, CLASS_ANGLES[n]))
    if _angdist(theta, CLASS_ANGLES[ranked[1]]) - _angdist(theta, CLASS_ANGLES[ranked[0]]) <= 0.0:
        return ABSTAIN
    return ranked[0]


def margin(scene: dict) -> float:
    """Angular decision margin in degrees: runner-up gap of the band direction."""
    theta = band_direction_deg(scene)
    if theta is None:
        return 0.0
    dists = sorted(_angdist(theta, CLASS_ANGLES[n]) for n in CLASS_NAMES)
    return float(dists[1] - dists[0])


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_DEG)


def _mixing_is_clean(k1: tuple[float, float], k2: tuple[float, float]) -> bool:
    """No low-frequency square-wave mixing term other than multiples of k1 - k2."""
    limit = 2.0 * math.pi / 26.0
    for n in range(-3, 4):
        for m in range(-3, 4):
            if n == 0 and m == 0:
                continue
            if n == -m:  # multiples of the declared beat term
                continue
            vx = n * k1[0] + m * k2[0]
            vy = n * k1[1] + m * k2[1]
            if math.hypot(vx, vy) < limit:
                return False
    return True


def sample_scene(seed: int) -> dict:
    """Balanced rejection sampling of one scene; the class is stratified on seed."""
    rng = random.Random(seed * 7919 + 13)
    target = CLASS_NAMES[seed % len(CLASS_NAMES)]
    base = CLASS_ANGLES[target]
    for _ in range(4000):
        jitter = rng.uniform(-MAX_TILT_JITTER_DEG, MAX_TILT_JITTER_DEG)
        band_deg = (base + jitter) % 180.0
        period = rng.uniform(50.0, 84.0)
        beat_deg = (band_deg + 90.0) % 180.0
        kb = (
            (2.0 * math.pi / period) * math.cos(math.radians(beat_deg)),
            (2.0 * math.pi / period) * math.sin(math.radians(beat_deg)),
        )
        pitch1 = rng.uniform(7.0, 11.5)
        tilt1 = rng.uniform(0.0, 180.0)
        k1 = (
            (2.0 * math.pi / pitch1) * math.cos(math.radians(tilt1)),
            (2.0 * math.pi / pitch1) * math.sin(math.radians(tilt1)),
        )
        k2 = (k1[0] - kb[0], k1[1] - kb[1])
        mag2 = math.hypot(k2[0], k2[1])
        pitch2 = 2.0 * math.pi / mag2
        if not 6.5 <= pitch2 <= 13.0:
            continue
        tilt2 = math.degrees(math.atan2(k2[1], k2[0])) % 180.0
        # Lines run perpendicular to their own wave vector.
        line1 = (tilt1 + 90.0) % 180.0
        line2 = (tilt2 + 90.0) % 180.0
        if min(_angdist(line1, band_deg), _angdist(line2, band_deg)) < MIN_GRATING_SEPARATION_DEG:
            continue
        if _angdist(line1, line2) < 6.0:
            continue
        if not _mixing_is_clean(k1, k2):
            continue
        # Counterbalance: the answer must not be the unique compass direction
        # that is farthest away from both drawn line families.
        clearance = {
            name: min(_angdist(CLASS_ANGLES[name], line1), _angdist(CLASS_ANGLES[name], line2))
            for name in CLASS_NAMES
        }
        if all(clearance[name] < clearance[target] for name in CLASS_NAMES if name != target):
            continue
        scene = {
            "width": WIDTH,
            "height": HEIGHT,
            "bg_color": list(BG_COLOR),
            "ink_color": list(INK_COLOR),
            "gratings": [
                {
                    "tilt_deg": round(tilt1, 6),
                    "pitch": round(pitch1, 6),
                    "duty": round(rng.uniform(0.34, 0.42), 6),
                    "phase": round(rng.uniform(0.0, 1.0), 6),
                },
                {
                    "tilt_deg": round(tilt2, 6),
                    "pitch": round(pitch2, 6),
                    "duty": round(rng.uniform(0.34, 0.42), 6),
                    "phase": round(rng.uniform(0.0, 1.0), 6),
                },
            ],
        }
        if analytic_gold(scene) != target or margin(scene) < SAMPLING_FLOOR_DEG:
            continue
        return scene
    raise RuntimeError(f"scene sampling failed for seed {seed}")


def _coverage(grid_x: np.ndarray, grid_y: np.ndarray, grating: dict) -> np.ndarray:
    """Antialiased ink coverage of one grating of straight parallel lines."""
    a = math.radians(float(grating["tilt_deg"]))
    pitch = float(grating["pitch"])
    u = (grid_x * math.cos(a) + grid_y * math.sin(a)) / pitch + float(grating["phase"])
    t = np.mod(u + 0.5, 1.0) - 0.5
    dist = np.abs(t) * pitch
    half = 0.5 * float(grating["duty"]) * pitch
    return np.clip(half - dist + 0.5, 0.0, 1.0)


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    grid_y, grid_x = np.mgrid[0:height, 0:width]
    grid_x = grid_x + 0.5
    grid_y = grid_y + 0.5
    ink = np.zeros((height, width), dtype=np.float64)
    for grating in scene["gratings"]:
        ink = np.maximum(ink, _coverage(grid_x, grid_y, grating))
    bg = np.array(scene["bg_color"], dtype=np.float64)
    fg = np.array(scene["ink_color"], dtype=np.float64)
    canvas = bg[None, None, :] * (1.0 - ink)[:, :, None] + fg[None, None, :] * ink[:, :, None]
    return Image.fromarray(np.rint(canvas).astype(np.uint8), mode="RGB")


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The grating pair is unordered: swapping it is pixel identical."""
    gratings = scene["gratings"]
    if len(gratings) != 2:
        return []
    alias = dict(scene)
    alias["gratings"] = [dict(gratings[1]), dict(gratings[0])]
    return [("grating_pair_swap", alias)]
