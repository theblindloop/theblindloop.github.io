"""Independent pixel-only inverse arm for Addressed Lamp Ribbon."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

CYAN = np.array((36, 184, 201), dtype=np.int16)
CORAL = np.array((235, 105, 86), dtype=np.int16)
SPAN_COUNT = 8


def _components(mask: np.ndarray, kind: str) -> list[tuple[float, float, int, int, int, str]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[tuple[float, float, int, int, int, str]] = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) < 400:
            continue
        ys = np.fromiter((p[0] for p in points), dtype=np.int32)
        xs = np.fromiter((p[1] for p in points), dtype=np.int32)
        box_w = int(xs.max() - xs.min() + 1)
        box_h = int(ys.max() - ys.min() + 1)
        output.append((float(xs.mean()), float(ys.mean()), box_w, box_h, len(points), kind))
    return output


def _read_operations(image: Image.Image) -> list[str]:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    cyan_mask = np.linalg.norm(array - CYAN, axis=2) < 18.0
    coral_mask = np.linalg.norm(array - CORAL, axis=2) < 18.0
    components = _components(cyan_mask, "F") + _components(coral_mask, "S")
    if len(components) != SPAN_COUNT:
        raise ValueError(f"expected eight intact operation spans, found {len(components)}")
    components.sort(key=lambda item: item[0])

    widths = np.array([item[2] for item in components], dtype=float)
    heights = np.array([item[3] for item in components], dtype=float)
    areas = np.array([item[4] for item in components], dtype=float)
    centers_x = np.array([item[0] for item in components], dtype=float)
    centers_y = np.array([item[1] for item in components], dtype=float)
    gaps = np.diff(centers_x)

    if widths.min() < 50 or widths.max() - widths.min() > 3:
        raise ValueError("operation span widths are damaged or inconsistent")
    if heights.min() < 20 or heights.max() - heights.min() > 3:
        raise ValueError("operation span heights are damaged or inconsistent")
    if np.max(np.abs(centers_y - np.median(centers_y))) > 2.0:
        raise ValueError("operation spans do not form one ordered ribbon")
    if gaps.min() < 55 or gaps.max() > 72:
        raise ValueError("operation span spacing is ambiguous")
    if np.any(areas < 0.96 * widths * heights):
        raise ValueError("an operation span is visibly incomplete")
    return [item[5] for item in components]


def decision_from_image(image: Image.Image) -> str:
    operations = _read_operations(image)
    lamps = [0, 0]
    active = 0
    for operation in operations:
        if operation == "S":
            active = 1 - active
        else:
            lamps[active] ^= 1
    return f"{lamps[0]}{lamps[1]}"
