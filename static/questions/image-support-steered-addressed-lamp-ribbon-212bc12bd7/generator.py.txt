"""Deterministic renderer and analytic arm for Addressed Lamp Ribbon."""

from __future__ import annotations

import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 900, 600
BACKGROUND = (247, 245, 239)
INK = (42, 47, 57)
CYAN = (36, 184, 201)
CORAL = (235, 105, 86)
GOLD = (238, 184, 48)
PALE = (231, 228, 219)
SPAN_WIDTH = 64
SPAN_COUNT = 8
ROUTE_LEFT = 194
ROUTE_CENTER_Y = 337


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    path = Path("/usr/share/fonts/truetype/dejavu") / name
    return ImageFont.truetype(str(path), size=size)


def _replay_operations(operations: list[str]) -> str:
    lamps = [0, 0]
    active = 0
    for operation in operations:
        if operation == "S":
            active = 1 - active
        elif operation == "F":
            lamps[active] ^= 1
        else:
            raise ValueError(f"unknown operation {operation!r}")
    return f"{lamps[0]}{lamps[1]}"


def sample_scene(seed: int) -> dict:
    """Sample a balanced valid sequence; consecutive seeds rotate target answers."""
    rng = random.Random(int(seed))
    target = ("00", "01", "10", "11")[int(seed) % 4]
    for _ in range(5000):
        operations = [rng.choice(("F", "S")) for _ in range(SPAN_COUNT)]
        if 2 <= operations.count("F") <= 6 and _replay_operations(operations) == target:
            return {
                "operations": operations,
                "thickness": 22 + 3 * (int(seed) % 5),
            }
    raise RuntimeError("could not sample a balanced operation sequence")


def analytic_gold(scene: dict) -> str:
    operations = list(scene["operations"])
    return _replay_operations(operations)


def margin(scene: dict) -> float:
    return float(scene["thickness"])


def is_quarantined(scene: dict) -> bool:
    return len(scene["operations"]) != SPAN_COUNT or int(scene["thickness"]) < 20


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    operations = list(scene["operations"])
    thickness = int(scene["thickness"])
    if len(operations) != SPAN_COUNT:
        raise ValueError("the public world requires exactly eight spans")

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(31, True)
    body_font = _font(19)
    small_bold = _font(17, True)
    label_font = _font(18, True)

    draw.text((70, 54), "ADDRESS THE LAMPS", fill=INK, font=title_font)
    draw.text(
        (70, 101),
        "Both lamps begin OFF  •  A begins active",
        fill=INK,
        font=body_font,
    )

    # A compact visible initial-state key. It is redundant with the public
    # question; the pixel oracle intentionally depends only on the route.
    box = (300, 154, 600, 246)
    draw.rounded_rectangle(box, radius=14, fill=(252, 251, 247), outline=INK, width=3)
    draw.text((326, 171), "START", fill=INK, font=small_bold)
    for index, label in enumerate(("A", "B")):
        cx = 458 + index * 77
        draw.ellipse((cx - 15, 169, cx + 15, 199), fill=PALE, outline=INK, width=3)
        draw.text((cx - 7, 207), label, fill=INK, font=label_font)
    draw.line((445, 235, 471, 235), fill=GOLD, width=6)

    y0 = ROUTE_CENTER_Y - thickness // 2
    y1 = y0 + thickness - 1
    route_right = ROUTE_LEFT + SPAN_COUNT * SPAN_WIDTH

    # The black backing and touching endpoint symbols make one connected trace.
    draw.rounded_rectangle(
        (ROUTE_LEFT - 5, y0 - 5, route_right + 5, y1 + 5),
        radius=(thickness + 10) // 2,
        fill=INK,
    )
    draw.ellipse(
        (ROUTE_LEFT - 45, ROUTE_CENTER_Y - 25, ROUTE_LEFT + 5, ROUTE_CENTER_Y + 25),
        fill=GOLD,
        outline=INK,
        width=4,
    )
    for index, operation in enumerate(operations):
        x0 = ROUTE_LEFT + index * SPAN_WIDTH + 2
        x1 = ROUTE_LEFT + (index + 1) * SPAN_WIDTH - 3
        color = CYAN if operation == "F" else CORAL
        draw.rectangle((x0, y0, x1, y1), fill=color)

    tip_x = route_right + 42
    draw.polygon(
        [
            (route_right + 1, ROUTE_CENTER_Y - 26),
            (tip_x, ROUTE_CENTER_Y),
            (route_right + 1, ROUTE_CENTER_Y + 26),
        ],
        fill=INK,
    )
    draw.text((ROUTE_LEFT - 55, ROUTE_CENTER_Y + 48), "START", fill=INK, font=small_bold)
    draw.text((ROUTE_LEFT + 94, 431), "CYAN = FLIP ACTIVE LAMP", fill=INK, font=small_bold)
    draw.text((ROUTE_LEFT + 94, 463), "CORAL = SWITCH ACTIVE LAMP", fill=INK, font=small_bold)
    draw.text((273, 527), "Read every span from the round end to the arrow.", fill=INK, font=body_font)
    return image
