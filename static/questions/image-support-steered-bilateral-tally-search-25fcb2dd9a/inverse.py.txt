"""Independent image-only inverse arm for bilateral tally search."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_CYAN = np.array((20, 170, 200), dtype=np.int16)
_AMBER = np.array((235, 155, 35), dtype=np.int16)
_LABELS = ("A", "B", "C", "D")


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y0, x0 in np.argwhere(mask):
        if seen[y0, x0]:
            continue
        queue = deque([(int(y0), int(x0))])
        seen[y0, x0] = True
        xs, ys = [], []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= 180:
            output.append((float(np.mean(xs)), float(np.mean(ys)), len(xs)))
    return output


def _lane_centers(points: list[tuple[float, float, int]]) -> list[float]:
    ys = sorted(point[1] for point in points)
    if not ys:
        raise ValueError("no tally components")
    groups = [[ys[0]]]
    for value in ys[1:]:
        if value - groups[-1][-1] > 66:
            groups.append([value])
        else:
            groups[-1].append(value)
    if len(groups) != 4:
        raise ValueError(f"expected four visual lanes, found {len(groups)}")
    return [float(np.mean(group)) for group in groups]


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    cyan_mask = np.linalg.norm(array - _CYAN, axis=2) < 18
    amber_mask = np.linalg.norm(array - _AMBER, axis=2) < 18
    cyan = _components(cyan_mask)
    amber = _components(amber_mask)
    if len(cyan) < 8 or len(amber) < 8:
        raise ValueError("insufficient visible tally evidence")
    centers = _lane_centers(cyan + amber)
    counts = []
    for center in centers:
        left = sum(abs(point[1] - center) < 55 for point in cyan)
        right = sum(abs(point[1] - center) < 55 for point in amber)
        counts.append((left, right))
    winners = [index for index, pair in enumerate(counts) if pair[0] == pair[1]]
    mismatches = sorted(abs(left - right) for left, right in counts)
    if len(winners) != 1 or len(mismatches) < 2 or mismatches[1] <= 0:
        raise ValueError("no unique equality winner with runner-up margin")
    return _LABELS[winners[0]]
