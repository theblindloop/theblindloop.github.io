"""Deterministic latent renderer for bilateral tally equality search."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw, ImageFont

CYAN = (20, 170, 200)
AMBER = (235, 155, 35)
INK = (38, 46, 54)
PALE = (226, 232, 236)
BACKGROUND = (250, 249, 246)
LABELS = ("A", "B", "C", "D")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    a, b, c = rng.sample(range(2, 8), 3)
    winner = int(seed) % 4
    distractors = [(a, c), (b, a), (b, c)]
    rng.shuffle(distractors)
    pairs = distractors[:]
    pairs.insert(winner, (a, a))
    lanes = [
        {"left_count": left, "right_count": right}
        for left, right in pairs
    ]
    return {"lanes": lanes, "labels": list(LABELS), "width": 960, "height": 640}


def _dot_centers(count: int, side: str, cy: int) -> list[tuple[int, int]]:
    columns = 4
    spacing_x, spacing_y = 42, 36
    base_x = 92 if side == "left" else 742
    points = []
    rows = (count + columns - 1) // columns
    for index in range(count):
        row, col = divmod(index, columns)
        x = base_x + col * spacing_x
        y = cy + (row - (rows - 1) / 2.0) * spacing_y
        points.append((int(round(x)), int(round(y))))
    return points


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    lanes = scene["lanes"]
    labels = scene["labels"]
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    draw.text((72, 43), "CYAN TALLY", fill=INK, font=font)
    draw.text((772, 43), "AMBER TALLY", fill=INK, font=font)
    for index, lane in enumerate(lanes):
        cy = 135 + index * 132
        draw.line((58, cy + 57, 304, cy + 57), fill=PALE, width=2)
        draw.line((708, cy + 57, 902, cy + 57), fill=PALE, width=2)
        label = str(labels[index])
        draw.rounded_rectangle((454, cy - 20, 506, cy + 20), radius=8, fill=INK)
        bbox = draw.textbbox((0, 0), label, font=font)
        tw, th = bbox[2] - bbox[0], bbox[3] - bbox[1]
        draw.text((480 - tw / 2, cy - th / 2), label, fill=BACKGROUND, font=font)
        for x, y in _dot_centers(int(lane["left_count"]), "left", cy):
            draw.ellipse((x - 11, y - 11, x + 11, y + 11), fill=CYAN)
        for x, y in _dot_centers(int(lane["right_count"]), "right", cy):
            draw.ellipse((x - 11, y - 11, x + 11, y + 11), fill=AMBER)
    return image


def analytic_gold(scene: dict) -> str:
    lanes = scene["lanes"]
    labels = scene["labels"]
    winners = [i for i, lane in enumerate(lanes) if lane["left_count"] == lane["right_count"]]
    if len(winners) != 1:
        raise ValueError("scene must have exactly one equality lane")
    return str(labels[winners[0]])


def margin(scene: dict) -> float:
    mismatches = sorted(
        abs(int(lane["left_count"]) - int(lane["right_count"])) for lane in scene["lanes"]
    )
    if not mismatches or mismatches[0] != 0:
        return 0.0
    return float(mismatches[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    del scene
    return []
