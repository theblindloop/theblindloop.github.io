"""Treemap quarter-share world: renderer, scene sampler and analytic gold.

One rectangular panel is tiled by exactly four flat-coloured blocks (a guillotine
treemap).  The decision is the colour of the block whose area is closest to one
quarter of the panel's total coloured area.  The reference quantity (a quarter of
the total) is supplied by the panel itself, so the answer is a proportion read
across the whole tiled region rather than an extremum of any single block.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "treemap_quarter_share-0.1.0"

CANVAS_W = 640
CANVAS_H = 480
SHEET = (244, 242, 236)
FRAME = (58, 58, 62)
DECO = (168, 168, 172)

# Exact flat fills; rectangles are drawn without antialiasing so every block's
# rendered pixel count equals its analytic area.
COLOR_RGB: dict[str, tuple[int, int, int]] = {
    "red": (198, 42, 42),
    "blue": (38, 78, 188),
    "green": (28, 138, 70),
    "orange": (232, 148, 18),
}
COLOR_ORDER = ("red", "blue", "green", "orange")

GAP = 3               # sheet-coloured gutter inset applied to every block
QUARANTINE_MARGIN = 0.02
MIN_BLOCK_W = 62
MIN_BLOCK_H = 46


# --------------------------------------------------------------------------
# analytic gold (reads scene["blocks"] only)
# --------------------------------------------------------------------------
def _block_areas(scene: dict) -> list[tuple[str, int]]:
    """(colour, rendered pixel area) for each drawn block, in scene order."""
    out = []
    for block in scene["blocks"]:
        x0, y0, x1, y1 = block["rect"]
        out.append((block["color"], (x1 - x0 + 1) * (y1 - y0 + 1)))
    return out


def _deviations(scene: dict) -> list[tuple[float, str]]:
    """|share - 1/4| for every block, paired with its colour."""
    areas = _block_areas(scene)
    total = float(sum(area for _, area in areas))
    return sorted((abs(area / total - 0.25), color) for color, area in areas)


def analytic_gold(scene: dict) -> str:
    """Colour of the block whose area share is nearest to one quarter."""
    return _deviations(scene)[0][1]


def margin(scene: dict) -> float:
    """Runner-up decision margin in share units (second-best minus best |dev|)."""
    ranked = _deviations(scene)
    return round(ranked[1][0] - ranked[0][0], 12)


def is_quarantined(scene: dict) -> bool:
    """Near-tie between the best and the runner-up share deviation."""
    return bool(margin(scene) < QUARANTINE_MARGIN)


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (scene["canvas_w"], scene["canvas_h"]), SHEET)
    draw = ImageDraw.Draw(image)

    fx0, fy0, fx1, fy1 = scene["frame"]
    draw.rectangle([fx0, fy0, fx1, fy1], outline=FRAME, width=2)

    for tick in scene["deco"]["ticks"]:
        draw.line(tick, fill=DECO, width=2)
    for ring in scene["deco"]["rings"]:
        draw.ellipse(ring, outline=DECO, width=2)

    for block in scene["blocks"]:
        draw.rectangle(block["rect"], fill=COLOR_RGB[block["color"]])

    # Legend swatches: same four colours, far smaller than any block, printed
    # outside the panel.  They are decoration and are never measured.
    for swatch in scene["deco"]["swatches"]:
        draw.rectangle(swatch["rect"], fill=COLOR_RGB[swatch["color"]])

    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical latent aliases: the block list order is not observable."""
    twins: list[tuple[str, dict]] = []
    blocks = list(scene["blocks"])
    if len(blocks) > 1:
        rotated = dict(scene)
        rotated["blocks"] = blocks[1:] + blocks[:1]
        twins.append(("block_order_rotation", rotated))
        reversed_scene = dict(scene)
        reversed_scene["blocks"] = list(reversed(blocks))
        twins.append(("block_order_reversal", reversed_scene))
    swatches = list(scene["deco"]["swatches"])
    if swatches:
        shuffled = dict(scene)
        deco = dict(scene["deco"])
        deco["swatches"] = swatches[1:] + swatches[:1]
        shuffled["deco"] = deco
        twins.append(("swatch_order_rotation", shuffled))
    return twins


# --------------------------------------------------------------------------
# scene sampling
# --------------------------------------------------------------------------
def _guillotine(rng: random.Random) -> list[list[int]]:
    """Four inset rectangles tiling one panel, or [] when the draw is degenerate."""
    w = rng.randint(276, 314)
    h = rng.randint(212, 236)
    x0 = rng.randint(70, CANVAS_W - 70 - w)
    y0 = rng.randint(84, CANVAS_H - 80 - h)
    x1, y1 = x0 + w, y0 + h

    cx = x0 + int(round(w * rng.uniform(0.31, 0.69)))
    ly = y0 + int(round(h * rng.uniform(0.28, 0.72)))
    ry = y0 + int(round(h * rng.uniform(0.28, 0.72)))

    cells = [
        [x0, y0, cx, ly],
        [x0, ly, cx, y1],
        [cx, y0, x1, ry],
        [cx, ry, x1, y1],
    ]
    rects = []
    for ax0, ay0, ax1, ay1 in cells:
        r = [ax0 + GAP, ay0 + GAP, ax1 - GAP, ay1 - GAP]
        if r[2] - r[0] + 1 < MIN_BLOCK_W or r[3] - r[1] + 1 < MIN_BLOCK_H:
            return []
        rects.append(r)
    return rects


def _decoration(rng: random.Random, frame: list[int]) -> dict:
    ticks = []
    for i in range(9):
        tx = 40 + i * 70
        ticks.append([tx, CANVAS_H - 26, tx, CANVAS_H - 14])
    rings = []
    for _ in range(rng.randint(2, 4)):
        r = rng.randint(7, 13)
        cx = rng.choice([rng.randint(20, 46), rng.randint(CANVAS_W - 46, CANVAS_W - 20)])
        cy = rng.randint(30, CANVAS_H - 60)
        rings.append([cx - r, cy - r, cx + r, cy + r])
    swatches = []
    order = list(COLOR_ORDER)
    rng.shuffle(order)
    for i, color in enumerate(order):
        sx = 44 + i * 34
        swatches.append({"color": color, "rect": [sx, 28, sx + 13, 41]})
    return {"ticks": ticks, "rings": rings, "swatches": swatches}


def _assign_colors(rects: list[list[int]], rng: random.Random, forced: str) -> list[dict]:
    """Give the winning cell colour ``forced``; deal the rest at random."""
    areas = [(r[2] - r[0] + 1) * (r[3] - r[1] + 1) for r in rects]
    total = float(sum(areas))
    devs = [abs(a / total - 0.25) for a in areas]
    winner = min(range(4), key=lambda i: (devs[i], i))
    rest = [c for c in COLOR_ORDER if c != forced]
    rng.shuffle(rest)
    colors: list[str] = []
    it = iter(rest)
    for i in range(4):
        colors.append(forced if i == winner else next(it))
    return [{"color": colors[i], "rect": rects[i]} for i in range(4)]


def _shape_ok(rects: list[list[int]], want_quarantine: bool) -> bool:
    areas = [(r[2] - r[0] + 1) * (r[3] - r[1] + 1) for r in rects]
    total = float(sum(areas))
    devs = sorted((abs(a / total - 0.25), i) for i, a in enumerate(areas))
    gap = devs[1][0] - devs[0][0]
    if want_quarantine:
        if not 0.004 <= gap < QUARANTINE_MARGIN:
            return False
    elif gap < 0.035:
        return False
    winner = devs[0][1]
    by_area = sorted(range(4), key=lambda i: areas[i])
    # Anti-cue: the answer is never simply the biggest or the smallest block.
    if winner in (by_area[0], by_area[-1]):
        return False
    if len(set(areas)) != 4:
        return False
    return True


def sample_scene(seed: int) -> dict:
    """Deterministic scene for ``seed``; every fourth seed targets a near-tie."""
    rng = random.Random(0x51D5 ^ (int(seed) * 2654435761 % (2**32)))
    want_quarantine = int(seed) % 8 == 3
    forced = COLOR_ORDER[int(seed) % 4]

    rects: list[list[int]] = []
    for _ in range(4000):
        candidate = _guillotine(rng)
        if candidate and _shape_ok(candidate, want_quarantine):
            rects = candidate
            break
    if not rects:
        for _ in range(20000):
            candidate = _guillotine(rng)
            if candidate and _shape_ok(candidate, False):
                rects = candidate
                break
    if not rects:
        raise RuntimeError(f"scene sampling failed for seed {seed}")

    xs = [r[0] for r in rects] + [r[2] for r in rects]
    ys = [r[1] for r in rects] + [r[3] for r in rects]
    frame = [min(xs) - GAP - 4, min(ys) - GAP - 4, max(xs) + GAP + 4, max(ys) + GAP + 4]

    return {
        "seed": int(seed),
        "canvas_w": CANVAS_W,
        "canvas_h": CANVAS_H,
        "frame": frame,
        "blocks": _assign_colors(rects, rng, forced),
        "deco": _decoration(rng, frame),
    }
