"""Pixel-only inverse arm for the treemap quarter-share world.

Input is the final PNG and nothing else: no latent scene, no renderer import, no
analytic gold.  The inverse arm segments the four flat block colours, keeps the
largest solid rectangular component of each colour (legend swatches and any
speckle are discarded), measures each block's area in pixels, forms the panel
total, and reports the colour whose measured share is closest to one quarter.
Structurally invalid panels abstain.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

TARGET_RGB: dict[str, tuple[int, int, int]] = {
    "red": (198, 42, 42),
    "blue": (38, 78, 188),
    "green": (28, 138, 70),
    "orange": (232, 148, 18),
}

MIN_BLOCK_AREA = 1500      # legend swatches are two orders of magnitude smaller
MIN_SOLIDITY = 0.9995      # a block must completely fill its own bounding box
ABSTAIN = "abstain"


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Run-based 4-connected labelling; returns one boolean mask per component."""
    height, width = mask.shape
    parent: list[int] = []

    def find(a: int) -> int:
        while parent[a] != a:
            parent[a] = parent[parent[a]]
            a = parent[a]
        return a

    def union(a: int, b: int) -> None:
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    runs: list[list[tuple[int, int, int]]] = []
    for y in range(height):
        row = mask[y]
        if not row.any():
            runs.append([])
            continue
        edges = np.flatnonzero(np.diff(np.concatenate(([0], row.view(np.int8), [0]))))
        row_runs = []
        for start, stop in zip(edges[0::2], edges[1::2]):
            label = len(parent)
            parent.append(label)
            row_runs.append((int(start), int(stop), label))
        runs.append(row_runs)

    for y in range(1, height):
        for start, stop, label in runs[y]:
            for pstart, pstop, plabel in runs[y - 1]:
                if pstart < stop and start < pstop:
                    union(label, plabel)

    groups: dict[int, list[tuple[int, int, int]]] = {}
    for y, row_runs in enumerate(runs):
        for start, stop, label in row_runs:
            groups.setdefault(find(label), []).append((y, start, stop))

    out = []
    for spans in groups.values():
        comp = np.zeros_like(mask)
        for y, start, stop in spans:
            comp[y, start:stop] = True
        out.append(comp)
    return out


def _largest_solid_block(pixels: np.ndarray, rgb: tuple[int, int, int]) -> int | None:
    mask = np.all(pixels == np.array(rgb, dtype=np.uint8), axis=-1)
    if not mask.any():
        return None
    best = max(_components(mask), key=lambda comp: int(comp.sum()))
    area = int(best.sum())
    if area < MIN_BLOCK_AREA:
        return None
    ys, xs = np.nonzero(best)
    box = (ys.max() - ys.min() + 1) * (xs.max() - xs.min() + 1)
    if area / float(box) < MIN_SOLIDITY:
        return None
    return area


def decision_from_image(image):
    """Colour whose measured block area is closest to a quarter of the total."""
    pixels = np.asarray(image.convert("RGB"), dtype=np.uint8)

    areas: dict[str, int] = {}
    for color, rgb in TARGET_RGB.items():
        area = _largest_solid_block(pixels, rgb)
        if area is None:
            return ABSTAIN
        areas[color] = area

    total = float(sum(areas.values()))
    if total <= 0:
        return ABSTAIN
    ranked = sorted((abs(area / total - 0.25), color) for color, area in areas.items())
    if ranked[0][0] == ranked[1][0]:
        return ABSTAIN
    return ranked[0][1]
