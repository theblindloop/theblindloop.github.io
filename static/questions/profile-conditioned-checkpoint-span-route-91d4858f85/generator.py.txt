"""Renderer and analytic arm for interior checkpoint-span comparison."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw


COLORS = ("orange", "purple", "blue")
PALETTES = (
    {"background": (249, 247, 241), "dark": (32, 38, 47),
     "green": (34, 174, 91), "cyan": (28, 181, 204),
     "orange": (235, 126, 28), "purple": (142, 73, 200), "blue": (50, 103, 213)},
    {"background": (253, 250, 245), "dark": (25, 34, 44),
     "green": (29, 161, 83), "cyan": (24, 169, 198),
     "orange": (225, 113, 22), "purple": (153, 75, 207), "blue": (43, 93, 201)},
    {"background": (246, 248, 244), "dark": (40, 42, 51),
     "green": (41, 183, 96), "cyan": (31, 187, 210),
     "orange": (242, 139, 35), "purple": (132, 67, 190), "blue": (57, 111, 221)},
)


def _route(kind: str, y: int) -> tuple[list[list[int]], list[int]]:
    """Each complete route is exactly 590 Manhattan pixels long."""
    if kind == "long":
        points = [[30, y], [180, y], [180, y - 35], [235, y - 35],
                  [235, y + 35], [300, y + 35], [300, y], [480, y]]
        checkpoints = [1, 6]  # interior span 260
    elif kind == "medium":
        points = [[30, y], [30, y + 22], [180, y + 22], [180, y],
                  [180, y - 26], [300, y - 26], [300, y],
                  [300, y + 22], [480, y + 22], [480, y]]
        checkpoints = [3, 6]  # interior span 172
    elif kind == "short":
        points = [[30, y], [30, y + 35], [180, y + 35], [180, y],
                  [300, y], [300, y - 35], [480, y - 35], [480, y]]
        checkpoints = [3, 4]  # interior span 120
    else:
        raise ValueError(f"unknown route kind: {kind}")
    assert sum(math.dist(a, b) for a, b in zip(points, points[1:])) == 590
    return points, checkpoints


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    kinds = ["long", "medium", "short"]
    rng.shuffle(kinds)
    ys = [70, 192, 314]
    y_shift = rng.randint(-2, 2)
    x_shift = rng.randint(-3, 3)
    routes, checkpoint_indices = [], []
    for kind, y in zip(kinds, ys):
        points, indices = _route(kind, y + y_shift)
        routes.append([[x + x_shift, py] for x, py in points])
        checkpoint_indices.append(indices)

    long_index = kinds.index("long")
    winning = COLORS[int(seed) % 3]
    remaining = [c for c in COLORS if c != winning]
    rng.shuffle(remaining)
    terminal_colors = [None, None, None]
    terminal_colors[long_index] = winning
    for index, color in zip((i for i in range(3) if i != long_index), remaining):
        terminal_colors[index] = color

    reflected = bool(rng.randrange(2))
    if reflected:
        routes = [[[511 - x, y] for x, y in points] for points in routes]
    return {
        "width": 512, "height": 384,
        "routes": routes,
        "checkpoint_indices": checkpoint_indices,
        "terminal_colors": terminal_colors,
        "palette": deepcopy(PALETTES[rng.randrange(len(PALETTES))]),
    }


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    palette = {key: tuple(value) for key, value in scene["palette"].items()}
    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), palette["background"])
    draw = ImageDraw.Draw(image)

    def disk(x: int, y: int, radius: int, color: tuple[int, int, int]) -> None:
        draw.ellipse(((x - radius) * scale, (y - radius) * scale,
                      (x + radius) * scale, (y + radius) * scale), fill=color)

    for points, cp_indices, terminal_color in zip(
            scene["routes"], scene["checkpoint_indices"], scene["terminal_colors"]):
        pts = [tuple(map(int, p)) for p in points]
        sx, sy = pts[0]
        ex, ey = pts[-1]
        # Markers first, so the subsequently painted centerline remains connected.
        disk(sx, sy, 15, palette["green"])
        disk(sx, sy, 9, palette["background"])
        draw.rectangle(((ex - 15) * scale, (ey - 15) * scale,
                        (ex + 15) * scale, (ey + 15) * scale), fill=palette[terminal_color])
        draw.rectangle(((ex - 9) * scale, (ey - 9) * scale,
                        (ex + 9) * scale, (ey + 9) * scale), fill=palette["background"])
        for cp_index in cp_indices:
            cx, cy = pts[int(cp_index)]
            diamond = [(cx * scale, (cy - 14) * scale), ((cx + 14) * scale, cy * scale),
                       (cx * scale, (cy + 14) * scale), ((cx - 14) * scale, cy * scale)]
            inner = [(cx * scale, (cy - 8) * scale), ((cx + 8) * scale, cy * scale),
                     (cx * scale, (cy + 8) * scale), ((cx - 8) * scale, cy * scale)]
            draw.polygon(diamond, fill=palette["cyan"])
            draw.polygon(inner, fill=palette["background"])
        draw.line([(x * scale, y * scale) for x, y in pts], fill=palette["dark"],
                  width=7 * scale, joint="curve")

    return image.resize((width, height), Image.Resampling.LANCZOS)


def _span(points: list[list[int]], pair: list[int]) -> float:
    start, stop = map(int, pair)
    return sum(math.dist(a, b) for a, b in zip(points[start:stop], points[start + 1:stop + 1]))


def _spans(scene: dict) -> list[float]:
    return [_span(points, pair) for points, pair in
            zip(scene["routes"], scene["checkpoint_indices"])]


def analytic_gold(scene: dict) -> str:
    spans = _spans(scene)
    return str(scene["terminal_colors"][max(range(3), key=spans.__getitem__)])


def margin(scene: dict) -> float:
    ranked = sorted(_spans(scene), reverse=True)
    return round(ranked[0] - ranked[1], 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 40.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
