"""Deterministic renderer and analytic reference for the singleton-state world."""

from __future__ import annotations

import random
from collections import Counter
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont


WIDTH = 1460
HEIGHT = 660
PANEL_LEFT = 18
PANEL_TOP = 126
PANEL_WIDTH = 230
PANEL_HEIGHT = 498
PANEL_GAP = 10
ROW_TOP = 50
ROW_HEIGHT = 82
ROW_BOX_X = 8
ROW_BOX_Y = 4
ROW_BOX_W = 214
ROW_BOX_H = 72
LABEL_X = 25
LABEL_Y = 25
SWATCH_X = 94
SWATCH_Y = 22
SWATCH_SIZE = 38
STATE_KEY_X = 388
STATE_KEY_Y = 47
STATE_KEY_STEP = 176
STATE_KEY_SIZE = 38
BITMAP_SCALE = 4

LABELS = ("A", "B", "C", "D", "E")
STATE_NAMES = ("LOW", "MID", "HIGH")

# A tiny bitmap alphabet makes the identity channel explicit and lets the
# inverse recognize it without OCR, fonts, or renderer imports.
LABEL_BITMAPS = {
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "B": ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    "C": ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    "D": ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
    "E": ("11111", "10000", "10000", "11110", "10000", "10000", "11111"),
}

# Each palette is legible on the pale canvas and sufficiently separated for a
# pixel inverse. The scene samples one palette and independently binds its
# entries to LOW/MID/HIGH through the visible key.
PALETTE_LIBRARY = (
    ((226, 72, 72), (54, 116, 218), (44, 164, 102)),
    ((237, 137, 40), (117, 76, 190), (25, 151, 178)),
    ((202, 73, 145), (49, 151, 81), (235, 169, 42)),
    ((66, 120, 204), (215, 88, 74), (121, 75, 177)),
    ((45, 158, 139), (222, 110, 43), (105, 79, 181)),
)


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    candidates = (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
        "/usr/share/fonts/dejavu/DejaVuSans.ttf",
    )
    for name in candidates:
        if Path(name).is_file():
            return ImageFont.truetype(name, size=size)
    return ImageFont.load_default()


def _draw_bitmap(draw: ImageDraw.ImageDraw, label: str, x: int, y: int) -> None:
    pattern = LABEL_BITMAPS[label]
    for row, bits in enumerate(pattern):
        for col, bit in enumerate(bits):
            if bit == "1":
                left = x + col * BITMAP_SCALE
                top = y + row * BITMAP_SCALE
                draw.rectangle(
                    [left, top, left + BITMAP_SCALE - 1, top + BITMAP_SCALE - 1],
                    fill=(23, 29, 38),
                )


def _draw_swatch(
    draw: ImageDraw.ImageDraw,
    x: int,
    y: int,
    color: tuple[int, int, int],
    size: int,
) -> None:
    draw.rounded_rectangle(
        [x, y, x + size - 1, y + size - 1],
        radius=7,
        fill=color,
        outline=(31, 37, 47),
        width=2,
    )


def _panel_x(panel: int) -> int:
    return PANEL_LEFT + panel * (PANEL_WIDTH + PANEL_GAP)


def _score_table(trajectories: dict[str, list[int]]) -> dict[str, int]:
    counts_by_panel = []
    for panel in range(6):
        counts_by_panel.append(Counter(trajectories[label][panel] for label in LABELS))
    scores = {}
    for label in LABELS:
        scores[label] = sum(
            counts_by_panel[panel][trajectories[label][panel]] == 1 for panel in range(6)
        )
    return scores


def _panel_is_valid(trajectories: dict[str, list[int]], panel: int) -> bool:
    counts = Counter(trajectories[label][panel] for label in LABELS)
    return sorted(counts.values()) == [1, 2, 2] and len(counts) == 3


def sample_scene(seed: int) -> dict:
    """Sample a balanced, non-quarantined scene deterministically."""

    rng = random.Random(int(seed))
    for _attempt in range(10000):
        trajectories = {label: [0] * 6 for label in LABELS}
        for panel in range(6):
            values = [0, 0, 1, 1, 2]
            rng.shuffle(values)
            for index, label in enumerate(LABELS):
                trajectories[label][panel] = values[index]
        scores = _score_table(trajectories)
        ordered = sorted(scores.values(), reverse=True)
        if all(_panel_is_valid(trajectories, panel) for panel in range(6)) and (
            ordered[0] - ordered[1] >= 1
        ):
            orders = []
            for _panel in range(6):
                row = list(LABELS)
                rng.shuffle(row)
                orders.append(row)
            palette = [list(color) for color in PALETTE_LIBRARY[rng.randrange(len(PALETTE_LIBRARY))]]
            rng.shuffle(palette)
            return {
                "trajectories": trajectories,
                "panel_orders": orders,
                "state_palette": palette,
            }
    raise RuntimeError(f"could not sample a non-quarantined scene for seed {seed}")


def render(scene: dict) -> Image.Image:
    """Render only from visible scene fields; the PNG is the oracle's input."""

    trajectories = scene["trajectories"]
    panel_orders = scene["panel_orders"]
    state_palette = scene["state_palette"]

    image = Image.new("RGB", (WIDTH, HEIGHT), (247, 249, 252))
    draw = ImageDraw.Draw(image)
    title_font = _font(24)
    small_font = _font(15)
    panel_font = _font(18)
    draw.text((22, 13), "Singleton-state sequence", fill=(28, 35, 47), font=title_font)
    draw.text(
        (22, 47),
        "STATE KEY",
        fill=(65, 74, 88),
        font=small_font,
    )
    for state, name in enumerate(STATE_NAMES):
        key_x = STATE_KEY_X + state * STATE_KEY_STEP
        _draw_swatch(draw, key_x, STATE_KEY_Y, tuple(state_palette[state]), STATE_KEY_SIZE)
        draw.text(
            (key_x + 53, STATE_KEY_Y + 9),
            name,
            fill=(58, 67, 80),
            font=small_font,
        )

    for panel in range(6):
        px = _panel_x(panel)
        draw.rounded_rectangle(
            [px, PANEL_TOP, px + PANEL_WIDTH - 1, PANEL_TOP + PANEL_HEIGHT - 1],
            radius=10,
            fill=(255, 255, 255),
            outline=(176, 185, 198),
            width=2,
        )
        draw.text(
            (px + 11, PANEL_TOP + 12),
            f"P{panel + 1}",
            fill=(49, 61, 78),
            font=panel_font,
        )
        draw.line(
            [px + 9, PANEL_TOP + 42, px + PANEL_WIDTH - 10, PANEL_TOP + 42],
            fill=(220, 225, 232),
            width=1,
        )
        for row, label in enumerate(panel_orders[panel]):
            ry = PANEL_TOP + ROW_TOP + row * ROW_HEIGHT
            draw.rounded_rectangle(
                [
                    px + ROW_BOX_X,
                    ry + ROW_BOX_Y,
                    px + ROW_BOX_X + ROW_BOX_W - 1,
                    ry + ROW_BOX_Y + ROW_BOX_H - 1,
                ],
                radius=7,
                fill=(252, 253, 255),
                outline=(224, 228, 235),
                width=1,
            )
            _draw_bitmap(draw, label, px + LABEL_X, ry + LABEL_Y)
            state = int(trajectories[label][panel])
            _draw_swatch(
                draw,
                px + SWATCH_X,
                ry + SWATCH_Y,
                tuple(state_palette[state]),
                SWATCH_SIZE,
            )
            draw.text(
                (px + 151, ry + 30),
                "state",
                fill=(142, 149, 160),
                font=small_font,
            )
    return image


def analytic_gold(scene: dict) -> str:
    trajectories = scene["trajectories"]
    scores = _score_table(trajectories)
    return max(LABELS, key=lambda label: (scores[label], -LABELS.index(label)))


def margin(scene: dict) -> float:
    trajectories = scene["trajectories"]
    scores = sorted(_score_table(trajectories).values(), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    trajectories = scene["trajectories"]
    return any(not _panel_is_valid(trajectories, panel) for panel in range(6)) or margin(scene) < 1


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # No hidden aliases are claimed: the identity-keyed states are the only
    # gold inputs, and all of them are painted as keyed swatches.
    return []
