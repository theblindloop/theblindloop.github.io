"""Independent image-only inverse for the singleton-state PNGs.

This module intentionally duplicates only the public raster layout and a tiny
bitmap alphabet. It has no access to renderer scene data or analytic gold.
"""

from __future__ import annotations

from collections import Counter
import math

from PIL import Image


WIDTH = 1460
HEIGHT = 660
PANEL_LEFT = 18
PANEL_TOP = 126
PANEL_WIDTH = 230
PANEL_GAP = 10
ROW_TOP = 50
ROW_HEIGHT = 82
LABEL_X = 25
LABEL_Y = 25
SWATCH_X = 94
SWATCH_Y = 22
SWATCH_SIZE = 38
STATE_KEY_X = 388
STATE_KEY_Y = 47
STATE_KEY_STEP = 176
STATE_KEY_SIZE = 38
BITMAP_SCALE = 4

LABELS = ("A", "B", "C", "D", "E")
LABEL_BITMAPS = {
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "B": ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    "C": ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    "D": ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
    "E": ("11111", "10000", "10000", "11110", "10000", "10000", "11111"),
}


def _panel_x(panel: int) -> int:
    return PANEL_LEFT + panel * (PANEL_WIDTH + PANEL_GAP)


def _dark(pixel: tuple[int, int, int]) -> bool:
    return sum(pixel) < 250


def _label_match(image: Image.Image, label: str, x: int, y: int) -> float:
    pattern = LABEL_BITMAPS[label]
    correct = 0
    total = 0
    for row, bits in enumerate(pattern):
        for col, bit in enumerate(bits):
            cx = x + col * BITMAP_SCALE + BITMAP_SCALE // 2
            cy = y + row * BITMAP_SCALE + BITMAP_SCALE // 2
            observed = _dark(image.getpixel((cx, cy)))
            correct += observed == (bit == "1")
            total += 1
    return correct / total


def _decode_label(image: Image.Image, x: int, y: int) -> str:
    scores = sorted(
        ((_label_match(image, label, x, y), label) for label in LABELS),
        reverse=True,
    )
    if scores[0][0] < 0.98 or scores[0][0] - scores[1][0] < 0.01:
        raise ValueError("identity glyph is missing or ambiguous")
    return scores[0][1]


def _color_distance(left: tuple[int, int, int], right: tuple[int, int, int]) -> float:
    return math.sqrt(sum((a - b) ** 2 for a, b in zip(left, right)))


def _decode_palette(image: Image.Image) -> list[tuple[int, int, int]]:
    palette = []
    for state in range(3):
        x = STATE_KEY_X + state * STATE_KEY_STEP + STATE_KEY_SIZE // 2
        y = STATE_KEY_Y + STATE_KEY_SIZE // 2
        palette.append(tuple(image.getpixel((x, y))))
    if len(set(palette)) != 3:
        raise ValueError("state key colors are not distinct")
    return palette


def _decode_state(
    image: Image.Image,
    x: int,
    y: int,
    palette: list[tuple[int, int, int]],
) -> int:
    # The swatch center is a solid fill, away from its outline and rounded
    # corners. A generous bound detects erased/corrupted evidence.
    pixel = tuple(image.getpixel((x + SWATCH_SIZE // 2, y + SWATCH_SIZE // 2)))
    distances = sorted((_color_distance(pixel, color), state) for state, color in enumerate(palette))
    if distances[0][0] > 40 or distances[0][0] + 3 > distances[1][0]:
        raise ValueError("state swatch is missing or ambiguous")
    return distances[0][1]


def decision_from_image(image: Image.Image) -> str:
    """Return the winning label from the PNG alone, or raise on ambiguity."""

    if image.size != (WIDTH, HEIGHT):
        raise ValueError(f"unexpected raster size {image.size}")
    image = image.convert("RGB")
    palette = _decode_palette(image)
    singleton_memberships = Counter()
    for panel in range(6):
        px = _panel_x(panel)
        decoded = []
        for row in range(5):
            ry = PANEL_TOP + ROW_TOP + row * ROW_HEIGHT
            label = _decode_label(image, px + LABEL_X, ry + LABEL_Y)
            state = _decode_state(image, px + SWATCH_X, ry + SWATCH_Y, palette)
            decoded.append((label, state))
        labels = [label for label, _state in decoded]
        if len(set(labels)) != 5:
            raise ValueError("panel contains duplicate or missing identities")
        counts = Counter(state for _label, state in decoded)
        if sorted(counts.values()) != [1, 2, 2] or len(counts) != 3:
            raise ValueError("panel does not expose one unique singleton state")
        singleton_state = next(state for state, count in counts.items() if count == 1)
        for label, state in decoded:
            if state == singleton_state:
                singleton_memberships[label] += 1
    scores = sorted(
        ((singleton_memberships.get(label, 0), label) for label in LABELS),
        reverse=True,
    )
    if len(scores) != 5 or scores[0][0] - scores[1][0] < 1:
        raise ValueError("singleton occupancy leader is tied or incomplete")
    return scores[0][1]
