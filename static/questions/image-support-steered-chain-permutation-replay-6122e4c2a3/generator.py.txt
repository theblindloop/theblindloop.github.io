"""Renderer and analytic arm for a local connected bead-permutation world."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "chain-permutation-replay-0.1.0"
CANVAS = (512, 512)
OPERATION_COUNT = 6
POINT_COUNT = OPERATION_COUNT + 2
BEAD_COLORS = ("red", "green", "blue")

PALETTE = {
    "background": (250, 247, 240),
    "route": (30, 67, 102),
    "station": (222, 126, 24),
    "finish": (125, 78, 158),
    "red": (201, 65, 69),
    "green": (25, 132, 82),
    "blue": (48, 102, 177),
    "distractor": (155, 164, 172),
}


def _palette(scene: dict[str, Any]) -> dict[str, tuple[int, int, int]]:
    return {
        name: tuple(int(channel) for channel in value)
        for name, value in scene["palette"].items()
    }


def _replay(initial_order: list[str], operations: list[int]) -> list[str]:
    order = list(initial_order)
    for operation in operations:
        if int(operation) == 0:
            order[0], order[1] = order[1], order[0]
        elif int(operation) == 1:
            order[1], order[2] = order[2], order[1]
        else:
            raise ValueError(f"unknown station orientation {operation}")
    return order


def _route_for(rng: random.Random) -> list[list[int]]:
    """Make a short, x-monotone trace whose absolute position is randomized."""
    x0 = rng.randint(120, 245)
    y0 = rng.randint(140, 300)
    # Seven nearly equal spans keep all six stations distinct while the entire
    # evidence-bearing world remains a small local inset of the canvas.
    offsets = (0, 6, 12, 18, 24, 30, 36, 42)
    points: list[list[int]] = []
    for index, offset in enumerate(offsets):
        jitter = 0 if index in (0, len(offsets) - 1) else rng.randint(-1, 1)
        points.append([x0 + 11 * index, y0 + offset + jitter])
    return points


def _distractors(rng: random.Random) -> list[dict[str, int]]:
    anchors = [(58, 68), (454, 72), (61, 446), (451, 448)]
    rng.shuffle(anchors)
    result: list[dict[str, int]] = []
    for index, (x, y) in enumerate(anchors):
        result.append(
            {
                "kind": index % 2,
                "x": x + rng.randint(-10, 10),
                "y": y + rng.randint(-10, 10),
                "size": rng.randint(5, 9),
            }
        )
    return result


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced, unambiguous six-station permutation chain."""
    rng = random.Random(int(seed))
    initial_order = list(BEAD_COLORS)
    rng.shuffle(initial_order)
    desired_index = int(seed) % len(BEAD_COLORS)
    desired = BEAD_COLORS[desired_index]
    operations: list[int] = []
    for _ in range(10000):
        trial = [rng.randrange(2) for _ in range(OPERATION_COUNT)]
        if set(trial) != {0, 1}:
            continue
        if _replay(initial_order, trial)[-1] == desired:
            operations = trial
            break
    if not operations:
        # Adjacent transpositions generate all three positions. This fallback
        # is deterministic and is reachable only if the rejection loop is
        # interrupted by an unexpected random implementation.
        operations = [0, 1, 0, 1, 0, 1]
        if _replay(initial_order, operations)[-1] != desired:
            operations[-1] ^= 1

    return {
        "operations": operations,
        "initial_order": initial_order,
        "route_points": _route_for(rng),
        "distractors": _distractors(rng),
        "route_width": rng.choice([3, 4]),
        "station_radius": 4,
        "bead_radius": 5,
        "palette": {name: list(value) for name, value in PALETTE.items()},
        "canvas": list(CANVAS),
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the trace, oriented stations, initial beads, and nuisance marks."""
    width, height = (int(value) for value in scene["canvas"])
    colors = _palette(scene)
    image = Image.new("RGB", (width, height), colors["background"])
    draw = ImageDraw.Draw(image)

    for distractor in scene["distractors"]:
        x, y, size = (int(distractor[key]) for key in ("x", "y", "size"))
        if int(distractor["kind"]) == 0:
            draw.ellipse(
                (x - size, y - size, x + size, y + size),
                outline=colors["distractor"],
                width=3,
            )
        else:
            draw.rectangle(
                (x - size, y - size, x + size, y + size),
                outline=colors["distractor"],
                width=3,
            )

    points = [(int(point[0]), int(point[1])) for point in scene["route_points"]]
    route_color = colors["route"]
    # A small rack and connector bind the three initial beads to the same
    # visible trace without making their colors carry operation semantics.
    bead_x = points[0][0] - 18
    bead_y = points[0][1]
    draw.line(
        [(bead_x + 5, bead_y - 12), (bead_x + 5, bead_y + 12)],
        fill=route_color,
        width=int(scene["route_width"]),
    )
    draw.line(
        [(bead_x + 5, bead_y), points[0]],
        fill=route_color,
        width=int(scene["route_width"]),
    )
    draw.line(
        points,
        fill=route_color,
        width=int(scene["route_width"]),
        joint="curve",
    )

    radius = int(scene["station_radius"])
    for point, operation in zip(points[1:-1], scene["operations"]):
        x, y = point
        if int(operation) == 0:
            # Left-pointing triangle: its broad vertical base is on the right.
            vertices = [(x - radius, y), (x + radius, y - radius), (x + radius, y + radius)]
        else:
            # Right-pointing triangle: its broad vertical base is on the left.
            vertices = [(x + radius, y), (x - radius, y - radius), (x - radius, y + radius)]
        draw.polygon(vertices, fill=colors["station"])

    finish_x, finish_y = points[-1]
    finish_radius = int(scene["station_radius"]) + 1
    draw.ellipse(
        (
            finish_x - finish_radius,
            finish_y - finish_radius,
            finish_x + finish_radius,
            finish_y + finish_radius,
        ),
        fill=colors["finish"],
    )

    bead_radius = int(scene["bead_radius"])
    for index, color_name in enumerate(scene["initial_order"]):
        center_y = bead_y + (index - 1) * 12
        draw.ellipse(
            (bead_x - bead_radius, center_y - bead_radius, bead_x + bead_radius, center_y + bead_radius),
            fill=colors[color_name],
            outline=route_color,
            width=2,
        )
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Replay only the visibly answer-bearing order and station fields."""
    return _replay(scene["initial_order"], scene["operations"])[-1]


def margin(scene: dict[str, Any]) -> float:
    """Return the smallest geometric separation available to the inverse arm."""
    points = [tuple(float(value) for value in point) for point in scene["route_points"]]
    radius = float(scene["station_radius"])
    station_spacing = min(math.dist(points[index], points[index + 1]) for index in range(1, 6))
    bead_spacing = 12.0 - 2.0 * float(scene["bead_radius"])
    return round(min(station_spacing - 2.0 * radius, bead_spacing, 2.0), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 1.5


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict]]:
    # The labeled order and oriented operations are answer-bearing and are not
    # render-invariant. No unrendered alias is declared.
    return []
