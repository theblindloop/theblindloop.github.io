"""Independent image-only oracle for terminal route redundancy.

It recovers the visible straight link set from dark corridor pixels, then
enumerates simple paths between the two colored terminal positions.  The code
contains no candidate-module imports and receives only a raster image.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_BOXES = (
    (40, 100, 680, 390),
    (720, 100, 1360, 390),
    (40, 430, 680, 720),
    (720, 430, 1360, 720),
)
_INK = np.array([28, 38, 52], dtype=np.float32)
_GREEN = np.array([25, 160, 90], dtype=np.float32)
_AMBER = np.array([224, 145, 27], dtype=np.float32)
_LABELS = ("panel-A", "panel-B", "panel-C", "panel-D")
_NODE_LAYOUTS = (
    (
        (-270, 0), (-145, -96), (0, -112), (155, -78),
        (155, 78), (0, 112), (-145, 96), (270, 0),
    ),
    (
        (-270, 0), (-145, 96), (0, 112), (155, 78),
        (155, -78), (0, -112), (-145, -96), (270, 0),
    ),
)
_MESH_EDGES = (
    (0, 1), (1, 2), (2, 3), (3, 7), (7, 4),
    (4, 5), (5, 6), (6, 0), (1, 6), (2, 5), (2, 4),
)
_PINNED_EDGES = (
    (0, 1), (1, 2), (2, 5), (5, 6), (6, 0),
    (2, 3), (3, 7), (7, 4), (2, 4), (1, 6), (3, 4),
)
_ALL_EDGES = tuple(dict.fromkeys(_MESH_EDGES + _PINNED_EDGES))


def _mask(arr: np.ndarray, color: np.ndarray, tolerance: float) -> np.ndarray:
    return np.linalg.norm(arr - color.reshape(1, 1, 3), axis=2) <= tolerance


def _points(
    layout: tuple[tuple[float, float], ...],
    box: tuple[int, int, int, int],
) -> list[tuple[float, float]]:
    x0, y0, x1, y1 = box
    cx = (x0 + x1) / 2.0
    cy = (y0 + y1) / 2.0
    return [(cx + x, cy + y) for x, y in layout]


def _support(dark: np.ndarray, first: tuple[float, float], second: tuple[float, float]) -> float:
    values = []
    height, width = dark.shape
    for fraction in np.linspace(0.22, 0.78, 25):
        x = first[0] * (1.0 - fraction) + second[0] * fraction
        y = first[1] * (1.0 - fraction) + second[1] * fraction
        ix, iy = int(round(x)), int(round(y))
        x0, x1 = max(0, ix - 2), min(width, ix + 3)
        y0, y1 = max(0, iy - 2), min(height, iy + 3)
        values.append(bool(dark[y0:y1, x0:x1].any()))
    return float(np.mean(values))


def _recover_edges(arr: np.ndarray, box: tuple[int, int, int, int]) -> list[tuple[int, int]]:
    x0, y0, x1, y1 = box
    sub = arr[y0 + 4 : y1 - 4, x0 + 4 : x1 - 4]
    dark = _mask(sub, _INK, 58.0)
    if int(_mask(sub, _GREEN, 70.0).sum()) < 100 or int(_mask(sub, _AMBER, 70.0).sum()) < 100:
        return []

    best_edges: list[tuple[int, int]] = []
    best_score = -1.0
    for layout in _NODE_LAYOUTS:
        full_points = _points(layout, box)
        points = [(x - (x0 + 4), y - (y0 + 4)) for x, y in full_points]
        supports = [(_support(dark, points[a], points[b]), (a, b)) for a, b in _ALL_EDGES]
        score = sum(value for value, _ in supports)
        edges = [edge for value, edge in supports if value >= 0.72]
        if score > best_score:
            best_score = score
            best_edges = edges
    return best_edges


def _has_two_internal_disjoint_routes(edges: list[tuple[int, int]]) -> bool:
    adjacency = [[] for _ in range(8)]
    for first, second in edges:
        adjacency[first].append(second)
        adjacency[second].append(first)
    for neighbors in adjacency:
        neighbors.sort()
    internal_sets: list[frozenset[int]] = []

    def visit(node: int, path: list[int], used: set[int]) -> None:
        if node == 7:
            internal_sets.append(frozenset(path[1:-1]))
            return
        for neighbor in adjacency[node]:
            if neighbor not in used:
                used.add(neighbor)
                path.append(neighbor)
                visit(neighbor, path, used)
                path.pop()
                used.remove(neighbor)

    visit(0, [0], {0})
    return any(
        left.isdisjoint(right)
        for index, left in enumerate(internal_sets)
        for right in internal_sets[index + 1 :]
    )


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    winners = [
        index
        for index, box in enumerate(_BOXES)
        if _has_two_internal_disjoint_routes(_recover_edges(arr, box))
    ]
    if len(winners) != 1:
        return "ambiguous"
    return _LABELS[winners[0]]
