"""Renderer and analytic arm for a two-vertex-disjoint-routes question.

Every panel contains the same eight visible junction positions and eleven dark
links.  The mesh kind has an outer terminal cycle.  The pinned kind swaps one
local link so that two cyclic lobes meet at one ordinary junction.  Thus the
answer is determined by terminal-route redundancy, not by the number of links
or by the degree multiset.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1400
HEIGHT = 820
PANEL_BOXES = (
    (40, 100, 680, 390),
    (720, 100, 1360, 390),
    (40, 430, 680, 720),
    (720, 430, 1360, 720),
)
PANEL_LABELS = ("panel-A", "panel-B", "panel-C", "panel-D")

BACKGROUND = (247, 249, 252)
PANEL_FILL = (255, 255, 255)
FRAME = (105, 119, 137)
INK = (28, 38, 52)
GREEN = (25, 160, 90)
AMBER = (224, 145, 27)

EDGE_WIDTH = 16
NODE_RADIUS = 12
TERMINAL_RADIUS = 17
TERMINAL_HALO = 24

# The two layouts are y-reflections.  The terminal vertices are 0 and 7;
# all other indices are ordinary junctions.
NODE_LAYOUTS: tuple[tuple[tuple[float, float], ...], ...] = (
    (
        (-270, 0), (-145, -96), (0, -112), (155, -78),
        (155, 78), (0, 112), (-145, 96), (270, 0),
    ),
    (
        (-270, 0), (-145, 96), (0, 112), (155, 78),
        (155, -78), (0, -112), (-145, -96), (270, 0),
    ),
)

# Both graph kinds have eleven edges and the same degree multiset.  They
# differ only by edge (4, 5) versus edge (3, 4).
MESH_EDGES = (
    (0, 1), (1, 2), (2, 3), (3, 7), (7, 4),
    (4, 5), (5, 6), (6, 0), (1, 6), (2, 5), (2, 4),
)
PINNED_EDGES = (
    (0, 1), (1, 2), (2, 5), (5, 6), (6, 0),
    (2, 3), (3, 7), (7, 4), (2, 4), (1, 6), (3, 4),
)
GRAPH_EDGES = {"mesh": MESH_EDGES, "pinned": PINNED_EDGES}
ALL_EDGES = tuple(dict.fromkeys(MESH_EDGES + PINNED_EDGES))


def _font(size: int) -> ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", size)
    except OSError:
        return ImageFont.load_default()


def graph_edges(panel: list) -> tuple[tuple[int, int], ...]:
    kind = str(panel[0])
    if kind not in GRAPH_EDGES:
        raise ValueError(f"unknown graph kind: {kind}")
    return GRAPH_EDGES[kind]


def panel_points(panel: list, box: tuple[int, int, int, int]) -> tuple[tuple[int, int], ...]:
    layout = NODE_LAYOUTS[int(panel[1])]
    x0, y0, x1, y1 = box
    cx = (x0 + x1) / 2.0
    cy = (y0 + y1) / 2.0
    return tuple((round(cx + x), round(cy + y)) for x, y in layout)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) * 9176 + 31)
    winner = int(seed) % len(PANEL_BOXES)
    panels = []
    for index in range(len(PANEL_BOXES)):
        kind = "mesh" if index == winner else "pinned"
        panels.append([kind, rng.randrange(len(NODE_LAYOUTS))])
    return {"panels": panels}


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((42, 24), "Two-route network", fill=INK, font=_font(29))
    draw.text(
        (735, 31),
        "routes may meet only at green and amber",
        fill=FRAME,
        font=_font(18),
    )

    for index, (box, panel) in enumerate(zip(PANEL_BOXES, scene["panels"])):
        x0, y0, x1, y1 = box
        draw.rounded_rectangle(box, radius=16, fill=PANEL_FILL, outline=FRAME, width=3)
        draw.text((x0 + 20, y0 + 14), PANEL_LABELS[index][-1], fill=INK, font=_font(25))
        points = panel_points(panel, box)
        for first, second in graph_edges(panel):
            draw.line((points[first], points[second]), fill=INK, width=EDGE_WIDTH, joint="curve")
        for point in points:
            px, py = point
            draw.ellipse(
                (px - NODE_RADIUS, py - NODE_RADIUS, px + NODE_RADIUS, py + NODE_RADIUS),
                fill=INK,
            )

        for node, color in ((0, GREEN), (7, AMBER)):
            px, py = points[node]
            draw.ellipse(
                (
                    px - TERMINAL_HALO,
                    py - TERMINAL_HALO,
                    px + TERMINAL_HALO,
                    py + TERMINAL_HALO,
                ),
                fill=PANEL_FILL,
            )
            draw.ellipse(
                (
                    px - TERMINAL_RADIUS,
                    py - TERMINAL_RADIUS,
                    px + TERMINAL_RADIUS,
                    py + TERMINAL_RADIUS,
                ),
                fill=color,
                outline=PANEL_FILL,
                width=3,
            )
    return image


def _adjacency(edges: tuple[tuple[int, int], ...] | list[tuple[int, int]]) -> list[list[int]]:
    adjacency = [[] for _ in range(8)]
    for first, second in edges:
        adjacency[first].append(second)
        adjacency[second].append(first)
    for neighbors in adjacency:
        neighbors.sort()
    return adjacency


def has_two_internal_disjoint_routes(
    edges: tuple[tuple[int, int], ...] | list[tuple[int, int]],
) -> bool:
    """Return whether terminals 0 and 7 have two internally disjoint simple paths."""
    adjacency = _adjacency(edges)
    internal_sets: list[frozenset[int]] = []

    def visit(node: int, path: list[int], used: set[int]) -> None:
        if node == 7:
            internal_sets.append(frozenset(path[1:-1]))
            return
        for neighbor in adjacency[node]:
            if neighbor not in used:
                used.add(neighbor)
                path.append(neighbor)
                visit(neighbor, path, used)
                path.pop()
                used.remove(neighbor)

    visit(0, [0], {0})
    return any(
        left.isdisjoint(right)
        for index, left in enumerate(internal_sets)
        for right in internal_sets[index + 1 :]
    )


def analytic_gold(scene: dict) -> str:
    winners = [
        index
        for index, panel in enumerate(scene["panels"])
        if has_two_internal_disjoint_routes(graph_edges(panel))
    ]
    if len(winners) != 1:
        raise ValueError(f"expected one route-redundant panel, got {winners}")
    return PANEL_LABELS[winners[0]]


def margin(scene: dict) -> float:
    """Smallest node clearance after reserving thick-link raster width."""
    clearances = []
    for panel, box in zip(scene["panels"], PANEL_BOXES):
        points = panel_points(panel, box)
        for first in range(len(points)):
            for second in range(first):
                distance = math.hypot(
                    points[first][0] - points[second][0],
                    points[first][1] - points[second][1],
                )
                clearances.append(distance - 2.0 * NODE_RADIUS - EDGE_WIDTH / 2.0)
    return float(min(clearances))


def is_quarantined(scene: dict) -> bool:
    try:
        if len(scene["panels"]) != len(PANEL_BOXES):
            return True
        if len(
            [
                panel
                for panel in scene["panels"]
                if has_two_internal_disjoint_routes(graph_edges(panel))
            ]
        ) != 1:
            return True
        return margin(scene) < 18.0
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """All graph and layout fields are rendered; no latent aliases are claimed."""
    return []
