"""Independent PNG-only inverse for the tip-to-tail resultant world."""

from __future__ import annotations

import numpy as np
from PIL import Image

_INKS = np.asarray([(225, 150, 25), (13, 139, 148)], dtype=np.float32)


def _largest_component(mask: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    seen: set[tuple[int, int]] = set()
    best: list[tuple[int, int]] = []
    for yy, xx in np.argwhere(mask):
        start = (int(yy), int(xx))
        if start in seen:
            continue
        stack = [start]
        seen.add(start)
        component = []
        while stack:
            y, x = stack.pop()
            component.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                p = (ny, nx)
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and p not in seen:
                    seen.add(p)
                    stack.append(p)
        if len(component) > len(best):
            best = component
    if len(best) < 150:
        raise ValueError("colored arrow missing or fragmented")
    return np.asarray(best, dtype=np.float64)[:, ::-1]  # x,y


def _direction(points: np.ndarray) -> tuple[int, int]:
    centered = points - points.mean(axis=0)
    values, vectors = np.linalg.eigh(centered.T @ centered)
    axis = vectors[:, int(np.argmax(values))]
    projections = centered @ axis
    lo, hi = float(projections.min()), float(projections.max())
    span = hi - lo
    if span < 24.0:
        raise ValueError("arrow axis too short")
    # The broad head shifts the colored-pixel centroid toward the tip relative
    # to the silhouette bounding-box center. This supplies the sign that PCA
    # alone lacks, without any location or scene-state assumption.
    box_center = (points.min(axis=0) + points.max(axis=0)) / 2.0
    signed = points.mean(axis=0) - box_center
    if min(abs(float(signed[0])), abs(float(signed[1]))) < 0.35:
        raise ValueError("arrow head/tail orientation is ambiguous")
    ax, ay = float(axis[0]), float(axis[1])
    if min(abs(ax), abs(ay)) < 0.48 or max(abs(ax), abs(ay)) > 0.88:
        raise ValueError("recovered arrow is not decisively diagonal")
    return (1 if signed[0] > 0 else -1, 1 if signed[1] > 0 else -1)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    directions = []
    for ink in _INKS:
        distance = np.linalg.norm(array - ink, axis=2)
        points = _largest_component(distance < 70.0)
        directions.append(_direction(points))
    sx = directions[0][0] + directions[1][0]
    sy = directions[0][1] + directions[1][1]
    if sx == 0 and sy == -2:
        return "up"
    if sx == 2 and sy == 0:
        return "right"
    if sx == 0 and sy == 2:
        return "down"
    if sx == -2 and sy == 0:
        return "left"
    raise ValueError("the two recovered arrows do not have a cardinal resultant")
