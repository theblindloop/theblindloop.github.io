"""Render two detached directed displacements for tip-to-tail composition."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

WIDTH = 640
HEIGHT = 640
SCALE = 3
BACKGROUND = (248, 247, 242)
PALETTE = [(225, 150, 25), (13, 139, 148)]  # amber, teal

_PAIRS = {
    "up": [(-1, -1), (1, -1)],
    "right": [(1, -1), (1, 1)],
    "down": [(-1, 1), (1, 1)],
    "left": [(-1, -1), (-1, 1)],
}
_DIRECTIONS = ("up", "right", "down", "left")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = _DIRECTIONS[int(seed) % len(_DIRECTIONS)]
    vectors = [list(v) for v in _PAIRS[target]]
    rng.shuffle(vectors)

    # The badge moves across a coarse lattice. This keeps its absolute position
    # genuinely variable while leaving each detached operand well inside its
    # own local sampling neighborhood under deterministic raster ablations.
    badge_center = [rng.choice((280, 360)), rng.choice((200, 280, 360, 440))]
    separation = 160
    slope = 0
    centers = [
        [badge_center[0] - separation // 2, badge_center[1] - slope],
        [badge_center[0] + separation // 2, badge_center[1] + slope],
    ]
    # Remote low-contrast rings make the absolute badge location nontrivial but
    # never enter either the analytic or pixel decision.
    decorations = []
    for _ in range(7):
        for _attempt in range(100):
            x, y = rng.randint(35, 605), rng.randint(35, 605)
            if math.hypot(x - badge_center[0], y - badge_center[1]) > 165:
                decorations.append([x, y, rng.randint(5, 10)])
                break
    return {
        "vectors": vectors,
        "centers": centers,
        "colors": [list(PALETTE[0]), list(PALETTE[1])],
        "badge_center": badge_center,
        "decorations": decorations,
    }


def _arrow_polygon(center: list[int], vector: list[int], scale: int) -> list[tuple[float, float]]:
    cx, cy = center
    dx, dy = vector
    norm = math.hypot(dx, dy)
    ux, uy = dx / norm, dy / norm
    px, py = -uy, ux
    # A flat narrow tail and broad triangular head expose the sign of the axis.
    local = [
        (-18, 5), (3, 5), (2, 10), (19, 0), (2, -10), (3, -5), (-18, -5)
    ]
    return [
        ((cx + along * ux + across * px) * scale,
         (cy + along * uy + across * py) * scale)
        for along, across in local
    ]


def render(scene: dict) -> Image.Image:
    vectors = scene["vectors"]
    centers = scene["centers"]
    colors = scene["colors"]
    badge_center = scene["badge_center"]
    decorations = scene["decorations"]

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)

    for x, y, radius in decorations:
        box = [(x - radius) * SCALE, (y - radius) * SCALE,
               (x + radius) * SCALE, (y + radius) * SCALE]
        draw.ellipse(box, outline=(222, 220, 213), width=2 * SCALE)

    bx, by = badge_center
    panel = [(bx - 132) * SCALE, (by - 58) * SCALE,
             (bx + 132) * SCALE, (by + 58) * SCALE]
    draw.rounded_rectangle(panel, radius=18 * SCALE, fill=(255, 255, 252),
                           outline=(105, 91, 125), width=3 * SCALE)
    # Small neutral tail-to-tip pictogram; it is constant across all labels.
    draw.line([(bx - 10) * SCALE, (by + 40) * SCALE,
               (bx + 10) * SCALE, (by + 40) * SCALE],
              fill=(190, 185, 192), width=2 * SCALE)

    for center, vector, color in zip(centers, vectors, colors, strict=True):
        polygon = _arrow_polygon(center, vector, SCALE)
        draw.polygon(polygon, fill=tuple(color), outline=(44, 50, 57), width=2 * SCALE)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    vectors = scene["vectors"]
    sx = sum(int(v[0]) for v in vectors)
    sy = sum(int(v[1]) for v in vectors)
    if sx == 0 and sy < 0:
        return "up"
    if sx > 0 and sy == 0:
        return "right"
    if sx == 0 and sy > 0:
        return "down"
    if sx < 0 and sy == 0:
        return "left"
    return "abstain"


def margin(scene: dict) -> float:
    vectors = scene["vectors"]
    sx = abs(sum(int(v[0]) for v in vectors))
    sy = abs(sum(int(v[1]) for v in vectors))
    return float(max(sx, sy) - min(sx, sy))


def is_quarantined(scene: dict) -> bool:
    vectors = scene["vectors"]
    return len(vectors) != 2 or analytic_gold(scene) == "abstain" or margin(scene) < 1.5


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    for key in ("vectors", "centers", "colors"):
        twin[key] = list(reversed(twin[key]))
    return [("swap_operand_records", twin)]
