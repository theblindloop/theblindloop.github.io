from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 800, 640
BACKGROUND = (247, 244, 236)
INK = (43, 48, 56)
MUTED = (106, 111, 118)
GOLD = (202, 157, 42)
PALETTES = (
    ((20, 157, 190), (222, 92, 72)),
    ((28, 177, 204), (238, 112, 84)),
)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(name, size=size)


def _simulate(history: list[int], events: list[int]) -> bool:
    rolling = list(history)
    lamp = False
    for value in events:
        rolling.append(value)
        a, b, c = rolling[-3:]
        if a == c and a != b:
            lamp = not lamp
    return lamp


def _balanced_sequences(target: int) -> list[tuple[int, ...]]:
    values = []
    for sequence in itertools.product((0, 1), repeat=6):
        if sum(sequence) != 3:
            continue
        if int(_simulate(list(sequence[:2]), list(sequence[2:]))) == target:
            values.append(sequence)
    return values


OFF_SEQUENCES = _balanced_sequences(0)
ON_SEQUENCES = _balanced_sequences(1)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = int(seed) & 1
    pool = ON_SEQUENCES if target else OFF_SEQUENCES
    sequence = pool[rng.randrange(len(pool))]
    # Centers sit in two nonadjacent high-resolution ablation cells, while the
    # entire decisive inset remains local. Absolute placement varies by scene.
    origins = ((250, 280), (350, 280), (250, 360), (350, 360))
    return {
        "history": list(sequence[:2]),
        "events": list(sequence[2:]),
        "origin": list(origins[rng.randrange(len(origins))]),
        "palette": rng.randrange(len(PALETTES)),
    }


def _centered(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str, font, fill) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2), text, font=font, fill=fill)


def _bead(draw: ImageDraw.ImageDraw, x: int, y: int, color: tuple[int, int, int]) -> None:
    radius = 7
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), outline=INK, width=2)


def render(scene: dict) -> Image.Image:
    history = list(scene["history"])
    events = list(scene["events"])
    left_x, center_y = (int(v) for v in scene["origin"])
    palette = PALETTES[int(scene["palette"])]
    right_x = left_x + 200

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    _centered(draw, (WIDTH // 2, 56), "ALTERNATING TRIPLET PATCH", _font(27, True), INK)
    _centered(draw, (WIDTH // 2, 94), "MEMORY top→bottom; then NEW in row order.", _font(18), MUTED)
    _centered(draw, (WIDTH // 2, 122), "Newest three alternate?  Flip the lamp.   Lamp starts OFF.", _font(18), MUTED)

    # Neutral remote marks are randomized-looking decoration but carry no state.
    for x in range(92, 709, 88):
        draw.ellipse((x - 3, 174, x + 3, 180), fill=(199, 197, 190))
    draw.line((82, 488, 718, 488), fill=(215, 211, 201), width=2)

    # Two detached decisive chips, with visible background between them.
    draw.rounded_rectangle((left_x - 35, center_y - 42, left_x + 35, center_y + 42), radius=12, fill=(255, 253, 248), outline=GOLD, width=4)
    _centered(draw, (left_x, center_y - 58), "MEMORY", _font(14, True), INK)
    for index, value in enumerate(history):
        _bead(draw, left_x, center_y - 11 + index * 22, palette[value])

    draw.rounded_rectangle((right_x - 38, center_y - 42, right_x + 38, center_y + 42), radius=12, fill=(255, 253, 248), outline=INK, width=2)
    _centered(draw, (right_x, center_y - 58), "NEW", _font(14, True), INK)
    positions = (
        (right_x - 13, center_y - 11),
        (right_x + 13, center_y - 11),
        (right_x - 13, center_y + 11),
        (right_x + 13, center_y + 11),
    )
    for value, (x, y) in zip(events, positions):
        _bead(draw, x, y, palette[value])

    _centered(draw, (WIDTH // 2, 527), "alternating = cyan–coral–cyan  or  coral–cyan–coral", _font(17), INK)
    _centered(draw, (WIDTH // 2, 568), "Is the lamp ON after all 4 NEW beads?   YES / NO", _font(21, True), INK)
    return image


def analytic_gold(scene: dict) -> str:
    return "yes" if _simulate(list(scene["history"]), list(scene["events"])) else "no"


def margin(scene: dict) -> float:
    return 64.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
