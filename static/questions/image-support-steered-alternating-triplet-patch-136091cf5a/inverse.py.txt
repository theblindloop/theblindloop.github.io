from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[tuple[int, float, float]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y0, x0 in np.argwhere(mask):
        if seen[y0, x0]:
            continue
        queue = deque([(int(y0), int(x0))])
        seen[y0, x0] = True
        xs = []
        ys = []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= 70:
            output.append((len(xs), float(np.mean(xs)), float(np.mean(ys))))
    return output


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[:, :, 0], array[:, :, 1], array[:, :, 2]
    masks = (
        (blue > red + 55) & (green > red + 45) & (blue > 145),
        (red > green + 70) & (red > blue + 85) & (red > 190),
    )
    beads = []
    for value, mask in enumerate(masks):
        for area, x, y in _components(mask):
            if 70 <= area <= 260:
                beads.append((x, y, value, area))
    if len(beads) != 6:
        raise ValueError(f"expected six separated colored beads, found {len(beads)}")

    ordered_x = sorted(beads, key=lambda item: item[0])
    gaps = [ordered_x[i + 1][0] - ordered_x[i][0] for i in range(5)]
    split = int(np.argmax(gaps)) + 1
    left, right = ordered_x[:split], ordered_x[split:]
    if len(left) != 2 or len(right) != 4 or max(gaps) < 100:
        raise ValueError("could not resolve the detached MEMORY and NEW chips")

    history = [item[2] for item in sorted(left, key=lambda item: item[1])]
    right = sorted(right, key=lambda item: (item[1], item[0]))
    events = [item[2] for item in right]
    rolling = list(history)
    lamp = False
    for value in events:
        rolling.append(value)
        a, b, c = rolling[-3:]
        if a == c and a != b:
            lamp = not lamp
    return "yes" if lamp else "no"
