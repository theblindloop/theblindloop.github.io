"""Deterministic latent sampler and renderer for nested_detour_order."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

BG = (247, 245, 239)
PANEL = (232, 226, 212)
NAVY = (28, 49, 68)
GREEN = (39, 166, 88)
ORANGE = (239, 130, 34)
PALE = (202, 206, 201)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    return {
        "width": 512,
        "height": 512,
        "cx": rng.choice((148, 212, 276, 340, 404)),
        "cy": rng.choice((108, 172, 236, 300, 364)),
        "rotation": rng.choice((45, 225)),
        "topology": "nested" if int(seed) % 2 == 0 else "sequential",
        "line_width": 6 if int(seed) % 13 == 0 else rng.choice((7, 8, 9)),
        "decorations": [
            [rng.randint(25, 487), rng.randint(25, 487), rng.choice((3, 4, 5))]
            for _ in range(6)
        ],
    }


def _turn(point: tuple[float, float], rotation: int) -> tuple[float, float]:
    x, y = point
    angle = math.radians(rotation)
    return x * math.cos(angle) - y * math.sin(angle), x * math.sin(angle) + y * math.cos(angle)


def _points(scene: dict, points: Iterable[tuple[float, float]]) -> list[tuple[int, int]]:
    cx, cy = int(scene["cx"]), int(scene["cy"])
    rotation = int(scene["rotation"])
    return [
        (round(cx + rx), round(cy + ry))
        for rx, ry in (_turn(point, rotation) for point in points)
    ]


def _paths(topology: str) -> list[list[tuple[float, float]]]:
    if topology == "sequential":
        return [
            [(-52, 0), (-38, 0)],
            [(-38, 0), (-22, -12), (-6, 0)],
            [(-38, 0), (-22, 12), (-6, 0)],
            [(-6, 0), (6, 0)],
            [(6, 0), (22, -12), (38, 0)],
            [(6, 0), (22, 12), (38, 0)],
            [(38, 0), (52, 0)],
        ]
    if topology == "nested":
        return [
            [(-52, 0), (-38, 0)],
            [(-38, 0), (0, 16), (38, 0)],
            [(-38, 0), (-16, -18)],
            [(-16, -18), (0, -34), (16, -18)],
            [(-16, -18), (0, -5), (16, -18)],
            [(16, -18), (38, 0)],
            [(38, 0), (52, 0)],
        ]
    raise ValueError(f"unknown topology {topology!r}")


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    cx, cy = int(scene["cx"]), int(scene["cy"])
    topology = str(scene["topology"])
    line_width = int(scene["line_width"])
    image = Image.new("RGB", (width, height), BG)
    draw = ImageDraw.Draw(image)
    for x, y, radius in scene["decorations"]:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=PALE)

    # The entire decisive graph stays inside one small, translated route stamp.
    draw.rounded_rectangle((cx - 58, cy - 58, cx + 58, cy + 58), radius=17, fill=PANEL)
    for path in _paths(topology):
        draw.line(_points(scene, path), fill=NAVY, width=line_width, joint="curve")

    # Explicit junction disks make raster topology stable across stroke widths.
    junctions = (
        [(-38, 0), (-6, 0), (6, 0), (38, 0)]
        if topology == "sequential"
        else [(-38, 0), (-16, -18), (16, -18), (38, 0)]
    )
    node_radius = max(4, line_width // 2)
    for x, y in _points(scene, junctions):
        draw.ellipse((x - node_radius, y - node_radius, x + node_radius, y + node_radius), fill=NAVY)

    start, finish = _points(scene, [(-52, 0), (52, 0)])
    marker_radius = 7
    draw.ellipse(
        (start[0] - marker_radius, start[1] - marker_radius,
         start[0] + marker_radius, start[1] + marker_radius),
        fill=GREEN, outline=(255, 255, 255), width=2,
    )
    draw.ellipse(
        (finish[0] - marker_radius, finish[1] - marker_radius,
         finish[0] + marker_radius, finish[1] + marker_radius),
        fill=ORANGE, outline=(255, 255, 255), width=2,
    )
    return image


def analytic_gold(scene: dict) -> str:
    topology = str(scene["topology"])
    if topology not in {"nested", "sequential"}:
        raise ValueError(topology)
    return topology


def margin(scene: dict) -> float:
    # Both templates have at least this much projected interval overlap/gap;
    # stroke width is the limiting raster margin near the finite-resolution edge.
    return float(min(18, int(scene["line_width"])))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
