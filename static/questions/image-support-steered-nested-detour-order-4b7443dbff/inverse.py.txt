"""Independent PNG-only inverse arm for split/rejoin nesting order."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_NAVY = np.array((28, 49, 68), dtype=np.int16)
_GREEN = np.array((39, 166, 88), dtype=np.int16)
_ORANGE = np.array((239, 130, 34), dtype=np.int16)


def _mask(array: np.ndarray, color: np.ndarray, radius: float = 24.0) -> np.ndarray:
    delta = array.astype(np.int32) - color.astype(np.int32)
    return np.sum(delta * delta, axis=2) <= radius * radius


def _components(mask: np.ndarray) -> list[np.ndarray]:
    seen = np.zeros(mask.shape, dtype=bool)
    height, width = mask.shape
    output: list[np.ndarray] = []
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = [(y, x)]
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.pop()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        output.append(np.asarray(points, dtype=float))
    return output


def _marker(mask: np.ndarray, name: str) -> np.ndarray:
    components = [component for component in _components(mask) if len(component) >= 45]
    if len(components) != 1:
        raise ValueError(f"expected one intact {name} endpoint marker")
    return components[0].mean(axis=0)


def _enclosed_faces(route: np.ndarray) -> list[np.ndarray]:
    ys, xs = np.nonzero(route)
    if len(ys) < 300:
        raise ValueError("route ink is missing or materially erased")
    y0, y1 = max(0, int(ys.min()) - 3), min(route.shape[0], int(ys.max()) + 4)
    x0, x1 = max(0, int(xs.min()) - 3), min(route.shape[1], int(xs.max()) + 4)
    crop = route[y0:y1, x0:x1]
    open_space = ~crop
    exterior = np.zeros_like(open_space)
    queue: deque[tuple[int, int]] = deque()
    height, width = crop.shape
    for x in range(width):
        for y in (0, height - 1):
            if open_space[y, x] and not exterior[y, x]:
                exterior[y, x] = True
                queue.append((y, x))
    for y in range(height):
        for x in (0, width - 1):
            if open_space[y, x] and not exterior[y, x]:
                exterior[y, x] = True
                queue.append((y, x))
    while queue:
        cy, cx = queue.popleft()
        for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
            if 0 <= ny < height and 0 <= nx < width and open_space[ny, nx] and not exterior[ny, nx]:
                exterior[ny, nx] = True
                queue.append((ny, nx))
    faces = []
    for component in _components(open_space & ~exterior):
        if len(component) >= 20:
            shifted = component.copy()
            shifted[:, 0] += y0
            shifted[:, 1] += x0
            faces.append(shifted)
    return faces


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    route = _mask(array, _NAVY)
    route_components = [component for component in _components(route) if len(component) >= 100]
    if len(route_components) != 1:
        raise ValueError("expected one intact connected dark route")
    route_points = route_components[0]
    span_y = float(np.ptp(route_points[:, 0]) + 1)
    span_x = float(np.ptp(route_points[:, 1]) + 1)
    if min(span_x, span_y) < 25 or max(span_x, span_y) > 122:
        raise ValueError("route dimensions fall outside the compact-stamp contract")

    start = _marker(_mask(array, _GREEN), "green")
    finish = _marker(_mask(array, _ORANGE), "orange")
    direction = finish - start
    distance = float(np.linalg.norm(direction))
    if distance < 88.0:
        raise ValueError("endpoint markers are missing or too close")
    direction /= distance

    faces = _enclosed_faces(route)
    if len(faces) != 2:
        raise ValueError(f"expected exactly two complete detours, found {len(faces)}")
    intervals = []
    for face in faces:
        projections = face @ direction
        intervals.append((float(projections.min()), float(projections.max())))
    first, second = intervals
    overlap = min(first[1], second[1]) - max(first[0], second[0])
    if abs(overlap) < 4.0:
        raise ValueError("detour order is too close to the raster decision boundary")
    return "nested" if overlap > 0 else "sequential"
