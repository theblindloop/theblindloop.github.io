"""Independent pixel-only decision arm for ``hinge_chain_joint_binding``.

This module receives a rasterized PNG and nothing else.  It imports no renderer,
prompt, generator, verifier or analytic-gold code and never sees a latent scene.
It works from generic color anchors and the visible panel frames:

1. locate the two framed panels;
2. segment the joint blobs in each panel and their centroids;
3. rebuild each chain's rod incidence by probing the ink along every chord, and
   order the six joints into a simple path;
4. find the black ring in REF and attach it to its joint;
5. align the two scale-normalized cumulative arclength profiles, choosing the
   traversal direction with the smaller deviation, and transfer the marked index;
6. read the color painted on the corresponding COPY joint.

It returns ``"abstain"`` whenever the evidence it needs is absent or ambiguous.
"""

from collections import deque

import numpy as np

BORDER = (140, 140, 150)
STROKE = (70, 70, 78)
GRAY_JOINT = (150, 150, 158)
BLACK = (0, 0, 0)

COLOR_RGB = {
    "red": (214, 42, 48),
    "orange": (232, 124, 20),
    "yellow": (226, 198, 32),
    "green": (36, 152, 66),
    "blue": (44, 86, 208),
    "purple": (146, 62, 186),
}

TOL = 18
N_JOINT = 6
N_SEG = 5
ADJ_RATIO = 0.92
ABSTAIN_MARGIN_PX = 8.0
ABSTAIN = "abstain"


def _mask(arr, rgb, tol=TOL):
    diff = np.abs(arr.astype(np.int16) - np.array(rgb, dtype=np.int16)).sum(axis=2)
    return diff <= tol


def _components(mask, min_size):
    h, w = mask.shape
    seen = np.zeros((h, w), dtype=bool)
    out = []
    ys, xs = np.nonzero(mask)
    for sy, sx in zip(ys.tolist(), xs.tolist()):
        if seen[sy, sx]:
            continue
        q = deque([(sy, sx)])
        seen[sy, sx] = True
        cells = []
        while q:
            y, x = q.popleft()
            cells.append((y, x))
            for dy, dx in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                ny, nx = y + dy, x + dx
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    q.append((ny, nx))
        if len(cells) >= min_size:
            cy = sum(c[0] for c in cells) / float(len(cells))
            cx = sum(c[1] for c in cells) / float(len(cells))
            y0 = min(c[0] for c in cells)
            y1 = max(c[0] for c in cells)
            x0 = min(c[1] for c in cells)
            x1 = max(c[1] for c in cells)
            out.append({"cx": cx, "cy": cy, "size": len(cells), "box": (x0, y0, x1, y1)})
    return out


def _panels(arr):
    comps = _components(_mask(arr, BORDER), min_size=600)
    frames = [c for c in comps
              if (c["box"][2] - c["box"][0]) > 200 and (c["box"][3] - c["box"][1]) > 100]
    frames.sort(key=lambda c: c["cy"])
    return [c["box"] for c in frames]


def _ink_mask(arr):
    m = _mask(arr, STROKE) | _mask(arr, GRAY_JOINT) | _mask(arr, BLACK)
    for rgb in COLOR_RGB.values():
        m = m | _mask(arr, rgb)
    return m


def _chord_ratio(ink, a, b):
    hit = 0
    n = 25
    for k in range(n):
        t = 0.2 + 0.6 * k / (n - 1.0)
        x = int(round(a[0] + t * (b[0] - a[0])))
        y = int(round(a[1] + t * (b[1] - a[1])))
        if 0 <= y < ink.shape[0] and 0 <= x < ink.shape[1] and ink[y, x]:
            hit += 1
    return hit / float(n)


def _order_path(nodes, ink):
    edges = []
    for i in range(len(nodes)):
        for j in range(i + 1, len(nodes)):
            if _chord_ratio(ink, nodes[i], nodes[j]) >= ADJ_RATIO:
                edges.append((i, j))
    if len(edges) != N_SEG:
        return None
    adj = {i: [] for i in range(len(nodes))}
    for i, j in edges:
        adj[i].append(j)
        adj[j].append(i)
    ends = [i for i in adj if len(adj[i]) == 1]
    if len(ends) != 2 or any(len(v) > 2 for v in adj.values()):
        return None
    order = [ends[0]]
    prev = None
    while True:
        cur = order[-1]
        nxt = [k for k in adj[cur] if k != prev]
        if not nxt:
            break
        prev, cur = cur, nxt[0]
        order.append(cur)
    if len(order) != len(nodes) or len(set(order)) != len(nodes):
        return None
    return order


def _cum_norm(points):
    segs = [((points[i + 1][0] - points[i][0]) ** 2
             + (points[i + 1][1] - points[i][1]) ** 2) ** 0.5
            for i in range(len(points) - 1)]
    total = sum(segs)
    if total <= 0:
        return None, 0.0
    out, acc = [0.0], 0.0
    for s in segs:
        acc += s
        out.append(acc / total)
    return out, total


def decision_from_image(image) -> str:
    arr = np.asarray(image.convert("RGB"))

    boxes = _panels(arr)
    if len(boxes) != 2:
        return ABSTAIN
    (rx0, ry0, rx1, ry1), (cx0, cy0, cx1, cy1) = boxes
    ref = arr[ry0:ry1 + 1, rx0:rx1 + 1]
    cop = arr[cy0:cy1 + 1, cx0:cx1 + 1]

    # --- REF panel: six gray joints, one circled in black -------------------
    ref_blobs = _components(_mask(ref, GRAY_JOINT), min_size=120)
    if len(ref_blobs) != N_JOINT:
        return ABSTAIN
    ref_pts = [(b["cx"], b["cy"]) for b in ref_blobs]
    ref_order = _order_path(ref_pts, _ink_mask(ref))
    if ref_order is None:
        return ABSTAIN
    ref_chain = [ref_pts[i] for i in ref_order]

    rings = _components(_mask(ref, BLACK), min_size=120)
    if len(rings) != 1:
        return ABSTAIN
    ring = (rings[0]["cx"], rings[0]["cy"])
    dists = [((p[0] - ring[0]) ** 2 + (p[1] - ring[1]) ** 2) ** 0.5 for p in ref_chain]
    marked = min(range(N_JOINT), key=lambda i: dists[i])
    if dists[marked] > 10.0 or sorted(dists)[1] < 30.0:
        return ABSTAIN

    # --- COPY panel: six painted joints -------------------------------------
    cop_pts, cop_names = [], []
    for name, rgb in sorted(COLOR_RGB.items()):
        blobs = _components(_mask(cop, rgb), min_size=120)
        if len(blobs) != 1:
            return ABSTAIN
        cop_pts.append((blobs[0]["cx"], blobs[0]["cy"]))
        cop_names.append(name)
    cop_order = _order_path(cop_pts, _ink_mask(cop))
    if cop_order is None:
        return ABSTAIN
    cop_chain = [cop_pts[i] for i in cop_order]
    cop_chain_names = [cop_names[i] for i in cop_order]

    # --- scale-normalized profile alignment ---------------------------------
    a, _ = _cum_norm(ref_chain)
    b, total_copy = _cum_norm(cop_chain)
    if a is None or b is None:
        return ABSTAIN
    b_rev = [1.0 - b[N_JOINT - 1 - i] for i in range(N_JOINT)]
    d_fwd = max(abs(a[i] - b[i]) for i in range(N_JOINT))
    d_rev = max(abs(a[i] - b_rev[i]) for i in range(N_JOINT))
    if abs(d_rev - d_fwd) * total_copy < ABSTAIN_MARGIN_PX:
        return ABSTAIN

    slot = marked if d_fwd < d_rev else (N_JOINT - 1 - marked)
    return cop_chain_names[slot]
