"""Independent pixel-only inverse for Rank Relay Correspondence."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

INK = np.array((31, 33, 36), dtype=np.uint8)
BAR = np.array((43, 84, 113), dtype=np.uint8)
TAG_COLORS = {
    "amber": np.array((230, 151, 38), dtype=np.uint8),
    "cyan": np.array((31, 169, 184), dtype=np.uint8),
    "violet": np.array((143, 94, 190), dtype=np.uint8),
}


def _components(mask: np.ndarray, minimum: int) -> list[dict]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]:
            continue
        queue = deque([(y0, x0)])
        seen[y0, x0] = True
        points = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) < minimum:
            continue
        ys = [point[0] for point in points]
        xs = [point[1] for point in points]
        output.append(
            {
                "area": len(points),
                "x0": min(xs), "x1": max(xs),
                "y0": min(ys), "y1": max(ys),
                "cx": float(sum(xs)) / len(xs),
                "cy": float(sum(ys)) / len(ys),
            }
        )
    return output


def _split_cards(items: list[dict]) -> tuple[list[dict], list[dict]]:
    if len(items) != 6:
        raise ValueError(f"expected six components, found {len(items)}")
    ordered = sorted(items, key=lambda item: item["cx"])
    gaps = [ordered[index + 1]["cx"] - ordered[index]["cx"] for index in range(5)]
    split = int(np.argmax(gaps)) + 1
    left, right = ordered[:split], ordered[split:]
    if len(left) != 3 or len(right) != 3 or max(gaps) < 200:
        raise ValueError("could not separate two three-item cards")
    return left, right


def _shape_name(component: dict) -> str:
    width = component["x1"] - component["x0"] + 1
    height = component["y1"] - component["y0"] + 1
    if not 29 <= width <= 43 or not 29 <= height <= 43:
        raise ValueError("malformed identity symbol")
    fill = component["area"] / float(width * height)
    if fill > 0.92:
        return "square"
    if fill > 0.68:
        return "circle"
    if fill > 0.42:
        return "triangle"
    raise ValueError("unrecognized identity symbol")


def _card_shapes(components: list[dict]) -> dict[str, dict]:
    result = {_shape_name(component): component for component in components}
    if set(result) != {"circle", "square", "triangle"}:
        raise ValueError("each card must contain one of every shape")
    return result


def _bind_bars(shapes: dict[str, dict], bars: list[dict]) -> dict[str, int]:
    if len(bars) != 3:
        raise ValueError("each card must contain three bars")
    remaining = list(bars)
    lengths = {}
    for name, shape in sorted(shapes.items(), key=lambda item: item[1]["cy"]):
        bar = min(remaining, key=lambda item: abs(item["cy"] - shape["cy"]))
        if abs(bar["cy"] - shape["cy"]) > 12:
            raise ValueError("bar is not aligned with its identity symbol")
        height = bar["y1"] - bar["y0"] + 1
        if not 20 <= height <= 27:
            raise ValueError("malformed bar")
        lengths[name] = int(bar["x1"] - bar["x0"] + 1)
        remaining.remove(bar)
    if remaining or len(set(lengths.values())) != 3:
        raise ValueError("bar ranking is ambiguous")
    return lengths


def _ranks(lengths: dict[str, int]) -> dict[str, int]:
    return {
        name: rank
        for rank, (name, _) in enumerate(sorted(lengths.items(), key=lambda item: item[1]))
    }


def decision_from_image(image: Image.Image) -> str:
    """Recover the answer using segmentation, component geometry, and binding only."""
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 200 or array.shape[1] < 400:
        raise ValueError("invalid raster")

    shape_components = _components(np.all(array == INK, axis=2), minimum=400)
    left_shape_list, right_shape_list = _split_cards(shape_components)
    left_shapes = _card_shapes(left_shape_list)
    right_shapes = _card_shapes(right_shape_list)

    bar_components = _components(np.all(array == BAR, axis=2), minimum=700)
    left_bars, right_bars = _split_cards(bar_components)
    left_rank = _ranks(_bind_bars(left_shapes, left_bars))
    right_rank = _ranks(_bind_bars(right_shapes, right_bars))
    matches = [name for name in left_rank if left_rank[name] == right_rank[name]]
    if len(matches) != 1:
        raise ValueError("rank-preserving shape is not unique")
    target = right_shapes[matches[0]]

    tags = []
    for name, color in TAG_COLORS.items():
        components = _components(np.all(array == color, axis=2), minimum=400)
        if len(components) != 1:
            raise ValueError(f"expected one {name} tag")
        component = components[0]
        width = component["x1"] - component["x0"] + 1
        height = component["y1"] - component["y0"] + 1
        if width < 22 or height < 22:
            raise ValueError("damaged answer tag")
        tags.append((name, component))
    answer, tag = min(tags, key=lambda item: abs(item[1]["cy"] - target["cy"]))
    if abs(tag["cy"] - target["cy"]) > 18:
        raise ValueError("answer tag is not bound to the selected shape")
    return answer
