"""Deterministic renderer for Rank Relay Correspondence."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1200, 600
BACKGROUND = (250, 249, 246)
PANEL = (112, 119, 126)
INK = (31, 33, 36)
BAR = (43, 84, 113)
TAG_COLORS = {
    "amber": (230, 151, 38),
    "cyan": (31, 169, 184),
    "violet": (143, 94, 190),
}
SHAPES = ("circle", "square", "triangle")
ROWS = (175, 300, 425)


def _rank_indices(lengths: Iterable[int]) -> list[int]:
    values = list(lengths)
    order = sorted(range(3), key=lambda index: values[index])
    ranks = [0, 0, 0]
    for rank, shape_index in enumerate(order):
        ranks[shape_index] = rank
    return ranks


def sample_scene(seed: int) -> dict:
    """Sample balanced scenes with exactly one cross-card rank fixed point."""
    rng = random.Random(int(seed))
    left_order = [0, 1, 2]
    rng.shuffle(left_order)
    left_rank = [0, 0, 0]
    for rank, shape_index in enumerate(left_order):
        left_rank[shape_index] = rank

    target = rng.randrange(3)
    right_rank = list(left_rank)
    other = [index for index in range(3) if index != target]
    right_rank[other[0]], right_rank[other[1]] = right_rank[other[1]], right_rank[other[0]]

    def lengths_for(ranks: list[int], near: bool) -> list[int]:
        gap1 = rng.randint(38, 55)
        gap2 = rng.randint(38, 55)
        if near:
            if rng.randrange(2):
                gap1 = rng.randint(14, 19)
            else:
                gap2 = rng.randint(14, 19)
        ordered = [rng.randint(58, 68)]
        ordered.append(ordered[0] + gap1)
        ordered.append(ordered[1] + gap2)
        return [ordered[ranks[index]] for index in range(3)]

    # Periodic finite-resolution stress cases exercise the declared quarantine.
    near_left = int(seed) % 17 == 0
    near_right = int(seed) % 19 == 0
    left_lengths = lengths_for(left_rank, near_left)
    right_lengths = lengths_for(right_rank, near_right)

    left_rows = [0, 1, 2]
    right_rows = [0, 1, 2]
    rng.shuffle(left_rows)
    rng.shuffle(right_rows)

    answer_tag = int(seed) % 3
    tag_names = list(TAG_COLORS)
    tag_by_shape: list[str | None] = [None, None, None]
    tag_by_shape[target] = tag_names[answer_tag]
    remaining_shapes = [index for index in range(3) if index != target]
    remaining_tags = [name for index, name in enumerate(tag_names) if index != answer_tag]
    rng.shuffle(remaining_tags)
    for shape_index, tag in zip(remaining_shapes, remaining_tags, strict=True):
        tag_by_shape[shape_index] = tag

    return {
        "left_rows": left_rows,
        "right_rows": right_rows,
        "tag_by_shape": tag_by_shape,
        "left_lengths": left_lengths,
        "right_lengths": right_lengths,
    }


def _draw_shape(draw: ImageDraw.ImageDraw, name: str, cx: int, cy: int) -> None:
    if name == "circle":
        draw.ellipse((cx - 18, cy - 18, cx + 18, cy + 18), fill=INK)
    elif name == "square":
        draw.rectangle((cx - 16, cy - 16, cx + 16, cy + 16), fill=INK)
    elif name == "triangle":
        draw.polygon(((cx, cy - 19), (cx - 19, cy + 17), (cx + 19, cy + 17)), fill=INK)
    else:
        raise ValueError(f"unknown shape {name!r}")


def _draw_card(
    draw: ImageDraw.ImageDraw,
    *,
    x0: int,
    rows: list[int],
    lengths: list[int],
    tags: list[str | None] | None,
) -> None:
    draw.rounded_rectangle((x0, 45, x0 + 330, 555), radius=18, outline=PANEL, width=5)
    for row_index, shape_index in enumerate(rows):
        cy = ROWS[row_index]
        _draw_shape(draw, SHAPES[shape_index], x0 + 50, cy)
        length = int(lengths[shape_index])
        draw.rectangle((x0 + 100, cy - 12, x0 + 100 + length - 1, cy + 11), fill=BAR)
        if tags is not None:
            tag = tags[shape_index]
            if tag not in TAG_COLORS:
                raise ValueError("invalid tag mapping")
            draw.rectangle((x0 + 283, cy - 14, x0 + 310, cy + 13), fill=TAG_COLORS[tag])


def render(scene: dict) -> Image.Image:
    """Render two deliberately separated evidence cards."""
    left_rows = list(scene["left_rows"])
    right_rows = list(scene["right_rows"])
    tags = list(scene["tag_by_shape"])
    left_lengths = [int(value) for value in scene["left_lengths"]]
    right_lengths = [int(value) for value in scene["right_lengths"]]
    if sorted(left_rows) != [0, 1, 2] or sorted(right_rows) != [0, 1, 2]:
        raise ValueError("row lists must be permutations")
    if sorted(tags) != sorted(TAG_COLORS):
        raise ValueError("tags must be a permutation")
    if len(set(left_lengths)) != 3 or len(set(right_lengths)) != 3:
        raise ValueError("bar lengths must be distinct")

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    draw.text((47, 76), "LEFT RANKS", fill=PANEL, font=font)
    draw.text((867, 76), "RIGHT RANKS + TAGS", fill=PANEL, font=font)
    draw.text((489, 286), "shortest = 1", fill=PANEL, font=font)
    draw.text((491, 306), "longest  = 3", fill=PANEL, font=font)
    _draw_card(draw, x0=25, rows=left_rows, lengths=left_lengths, tags=None)
    _draw_card(draw, x0=845, rows=right_rows, lengths=right_lengths, tags=tags)
    return image


def analytic_gold(scene: dict) -> str:
    left = _rank_indices(scene["left_lengths"])
    right = _rank_indices(scene["right_lengths"])
    matches = [index for index in range(3) if left[index] == right[index]]
    if len(matches) != 1:
        raise ValueError("rank correspondence is not unique")
    return str(scene["tag_by_shape"][matches[0]])


def margin(scene: dict) -> float:
    gaps = []
    for key in ("left_lengths", "right_lengths"):
        ordered = sorted(int(value) for value in scene[key])
        gaps.extend((ordered[1] - ordered[0], ordered[2] - ordered[1]))
    return float(min(gaps))


def is_quarantined(scene: dict) -> bool:
    try:
        analytic_gold(scene)
    except ValueError:
        return True
    return margin(scene) < 20.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
