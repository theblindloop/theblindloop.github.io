"""Independent PNG-only inverse for the row-seam composite world."""

from __future__ import annotations

import numpy as np
from PIL import Image

GRID_N = 5
EXPECTED_PANEL_DARK = 12
EXPECTED_REFERENCE_DARK = 9
MIN_MARGIN = 6


class OracleAbstention(ValueError):
    """Raised when visible fiducials or transformed evidence are ambiguous."""


def _luminance(pixels: np.ndarray) -> np.ndarray:
    values = pixels.astype(np.float32)
    return 0.2126 * values[..., 0] + 0.7152 * values[..., 1] + 0.0722 * values[..., 2]


def _runs(values: np.ndarray) -> list[tuple[int, int]]:
    if len(values) == 0:
        return []
    groups: list[tuple[int, int]] = []
    start = previous = int(values[0])
    for raw in values[1:]:
        value = int(raw)
        if value != previous + 1:
            groups.append((start, previous))
            start = value
        previous = value
    groups.append((start, previous))
    return groups


def _sample_patch(luma: np.ndarray, x: float, y: float, radius: int = 2) -> float:
    ix, iy = int(round(x)), int(round(y))
    if ix - radius < 0 or iy - radius < 0 or ix + radius >= luma.shape[1] or iy + radius >= luma.shape[0]:
        raise OracleAbstention("cell sample lies outside the raster")
    return float(np.median(luma[iy - radius : iy + radius + 1, ix - radius : ix + radius + 1]))


def _sample_grid(
    luma: np.ndarray,
    box: tuple[int, int, int, int],
    size: int,
    border: int,
) -> np.ndarray:
    xmin, ymin, xmax, ymax = box
    width = xmax - xmin + 1 - 2 * border
    height = ymax - ymin + 1 - 2 * border
    if width < size * 14 or height < size * 14 or abs(width - height) > 3:
        raise OracleAbstention("framed grid geometry is malformed")
    mask = np.zeros((size, size), dtype=bool)
    for row in range(size):
        for column in range(size):
            x = xmin + border + (column + 0.5) * width / size
            y = ymin + border + (row + 0.5) * height / size
            mask[row, column] = _sample_patch(luma, x, y) < 145.0
    return mask


def _source_masks(arr: np.ndarray, luma: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    cyan = (arr[..., 0] < 70) & (arr[..., 1] > 130) & (arr[..., 2] > 160)
    x_runs = [(a, b) for a, b in _runs(np.flatnonzero(cyan.any(axis=0))) if b - a > 145]
    y_runs = [(a, b) for a, b in _runs(np.flatnonzero(cyan.any(axis=1))) if b - a > 145]
    if len(x_runs) != 1 or len(y_runs) != 2:
        raise OracleAbstention(
            f"expected two vertically aligned cyan panels, found {len(x_runs)} by {len(y_runs)}"
        )
    xmin, xmax = x_runs[0]
    panels = [
        _sample_grid(luma, (xmin, ymin, xmax, ymax), GRID_N, 6)
        for ymin, ymax in sorted(y_runs)
    ]
    if any(int(panel.sum()) != EXPECTED_PANEL_DARK for panel in panels):
        raise OracleAbstention("a source panel violates the equal-ink construction")
    return panels[0], panels[1]


def _seam_cuts(arr: np.ndarray) -> tuple[int, ...]:
    gold = (
        (arr[..., 0] > 205)
        & (arr[..., 1] > 135)
        & (arr[..., 1] < 205)
        & (arr[..., 2] < 95)
    )
    x_runs = [(a, b) for a, b in _runs(np.flatnonzero(gold.any(axis=0))) if b - a > 145]
    y_runs = [(a, b) for a, b in _runs(np.flatnonzero(gold.any(axis=1))) if b - a > 145]
    if len(x_runs) != 1 or len(y_runs) != 1:
        raise OracleAbstention("gold seam-key frame is missing or duplicated")
    xmin, xmax = x_runs[0]
    ymin, ymax = y_runs[0]
    width = xmax - xmin + 1 - 12
    height = ymax - ymin + 1 - 12
    if width < GRID_N * 14 or abs(width - height) > 3:
        raise OracleAbstention("gold seam-key geometry is malformed")
    cuts: list[int] = []
    for row in range(GRID_N):
        hits: list[int] = []
        for column in range(GRID_N):
            x = int(round(xmin + 6 + (column + 0.5) * width / GRID_N))
            y = int(round(ymin + 6 + (row + 0.5) * height / GRID_N))
            patch = gold[y - 2 : y + 3, x - 2 : x + 3]
            if patch.shape == (5, 5) and float(patch.mean()) > 0.8:
                hits.append(column)
        if len(hits) != 1 or hits[0] >= GRID_N - 1:
            raise OracleAbstention("each key row must mark exactly one nonfinal splice column")
        cuts.append(hits[0] + 1)
    return tuple(cuts)


def _reference_masks(arr: np.ndarray, luma: np.ndarray) -> list[np.ndarray]:
    coral = (
        (arr[..., 0] > 190)
        & (arr[..., 1] < 110)
        & (arr[..., 2] > 90)
        & (arr[..., 2] < 165)
    )
    x_runs = [(a, b) for a, b in _runs(np.flatnonzero(coral.any(axis=0))) if b - a > 80]
    y_runs = [(a, b) for a, b in _runs(np.flatnonzero(coral.any(axis=1))) if b - a > 80]
    if len(x_runs) != 2 or len(y_runs) != 2:
        raise OracleAbstention(
            f"expected a 2-by-2 array of reference frames, found {len(x_runs)} by {len(y_runs)}"
        )
    references: list[np.ndarray] = []
    for ymin, ymax in sorted(y_runs):
        for xmin, xmax in sorted(x_runs):
            mask = _sample_grid(luma, (xmin, ymin, xmax, ymax), GRID_N, 5)
            if int(mask.sum()) != EXPECTED_REFERENCE_DARK:
                raise OracleAbstention("a reference glyph violates the equal-ink construction")
            references.append(mask)
    if len({mask.tobytes() for mask in references}) != 4:
        raise OracleAbstention("reference glyphs are not distinct")
    return references


def _composite(upper: np.ndarray, lower: np.ndarray, cuts: tuple[int, ...]) -> np.ndarray:
    result = np.zeros_like(upper)
    for row, cut in enumerate(cuts):
        result[row, :cut] = upper[row, :cut]
        result[row, cut:] = lower[row, cut:]
    return result


def analyze_image(image: Image.Image) -> tuple[str, float, tuple[int, ...]]:
    """Return (A-D decision, runner-up margin, four visible Hamming scores)."""
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.ndim != 3 or min(arr.shape[:2]) < 128:
        raise OracleAbstention("unsupported raster")
    luma = _luminance(arr)
    upper, lower = _source_masks(arr, luma)
    cuts = _seam_cuts(arr)
    references = _reference_masks(arr, luma)
    composite = _composite(upper, lower, cuts)
    if int(composite.sum()) != EXPECTED_REFERENCE_DARK:
        raise OracleAbstention("spliced result violates the balanced construction")
    scores = tuple(int(np.count_nonzero(composite != reference)) for reference in references)
    order = sorted(range(4), key=lambda index: (scores[index], index))
    decision_margin = float(scores[order[1]] - scores[order[0]])
    if scores[order[0]] > 2 or decision_margin < MIN_MARGIN:
        raise OracleAbstention(
            f"ambiguous seam composite: best={scores[order[0]]}, margin={decision_margin:g}"
        )
    return "ABCD"[order[0]], decision_margin, scores


def decision_from_image(image: Image.Image) -> str:
    """Recover the answer using only raster pixels and the public transform rule."""
    return analyze_image(image)[0]
