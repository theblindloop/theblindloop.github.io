"""Renderer and analytic label for colored convex-hull correspondence."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "convex_hull_membership_correspondence-0.1.0"
CANVAS_SIZE = (1200, 600)
SUPERSAMPLE = 3
N_POINTS = 5
PANEL_BOXES = {
    "reference": (18, 46, 382, 554),
    "A": (418, 46, 782, 554),
    "B": (818, 46, 1182, 554),
}
PANEL_CENTERS = {
    "reference": (200.0, 300.0),
    "A": (600.0, 300.0),
    "B": (1000.0, 300.0),
}
PALETTE = {
    "red": (214, 55, 56),
    "blue": (42, 101, 190),
    "green": (35, 149, 83),
    "amber": (229, 151, 25),
    "purple": (132, 71, 172),
}
COLORS = tuple(PALETTE)
BASE_SHAPES = (
    (
        (-126.0, -104.0),
        (132.0, -78.0),
        (146.0, 100.0),
        (-92.0, 122.0),
        (3.0, 7.0),
    ),
    (
        (-118.0, -96.0),
        (126.0, -101.0),
        (139.0, 108.0),
        (-112.0, 112.0),
        (-8.0, 4.0),
    ),
    (
        (-136.0, -76.0),
        (108.0, -119.0),
        (146.0, 86.0),
        (-82.0, 127.0),
        (12.0, -5.0),
    ),
)
EDGE_CLEARANCE = 16.0
TURN_CLEARANCE = 10.0


def _cross(a: tuple[float, float], b: tuple[float, float], c: tuple[float, float]) -> float:
    return (b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0])


def _convex_hull_indices(parts: Iterable[dict]) -> list[int]:
    """Return the indices of hull vertices in counter-clockwise order."""

    indexed = []
    for index, part in enumerate(parts):
        indexed.append((float(part["x"]), float(part["y"]), index))
    if len({(x, y) for x, y, _ in indexed}) != len(indexed):
        return []
    ordered = sorted(indexed, key=lambda item: (item[0], item[1], item[2]))

    lower: list[tuple[float, float, int]] = []
    for item in ordered:
        while len(lower) >= 2 and _cross(lower[-2][:2], lower[-1][:2], item[:2]) <= 1e-8:
            lower.pop()
        lower.append(item)
    upper: list[tuple[float, float, int]] = []
    for item in reversed(ordered):
        while len(upper) >= 2 and _cross(upper[-2][:2], upper[-1][:2], item[:2]) <= 1e-8:
            upper.pop()
        upper.append(item)
    cycle = lower[:-1] + upper[:-1]
    return [item[2] for item in cycle]


def _point_segment_distance(
    point: tuple[float, float], start: tuple[float, float], end: tuple[float, float]
) -> float:
    dx, dy = end[0] - start[0], end[1] - start[1]
    length_sq = dx * dx + dy * dy
    if length_sq <= 1e-12:
        return math.dist(point, start)
    t = max(0.0, min(1.0, ((point[0] - start[0]) * dx + (point[1] - start[1]) * dy) / length_sq))
    projection = (start[0] + t * dx, start[1] + t * dy)
    return math.dist(point, projection)


def _valid_panel(parts: object) -> bool:
    if not isinstance(parts, list) or len(parts) != N_POINTS:
        return False
    seen_colors: set[str] = set()
    seen_coords: set[tuple[float, float]] = set()
    for part in parts:
        if not isinstance(part, dict) or set(part) != {"x", "y", "color"}:
            return False
        try:
            x, y, color = float(part["x"]), float(part["y"]), str(part["color"])
        except (TypeError, ValueError):
            return False
        if not math.isfinite(x) or not math.isfinite(y) or color not in PALETTE:
            return False
        if color in seen_colors or (x, y) in seen_coords:
            return False
        seen_colors.add(color)
        seen_coords.add((x, y))
    hull = _convex_hull_indices(parts)
    if len(hull) < 3 or len(hull) >= N_POINTS:
        return False
    coords = [(float(part["x"]), float(part["y"])) for part in parts]
    interior = [index for index in range(N_POINTS) if index not in hull]
    for index in interior:
        point = coords[index]
        clearance = min(
            _point_segment_distance(point, coords[a], coords[b])
            for a, b in zip(hull, hull[1:] + hull[:1])
        )
        if clearance < EDGE_CLEARANCE:
            return False
    for index, vertex in enumerate(hull):
        before = coords[hull[index - 1]]
        current = coords[vertex]
        after = coords[hull[(index + 1) % len(hull)]]
        turn_height = abs(_cross(before, current, after)) / max(math.dist(before, after), 1e-6)
        if turn_height < TURN_CLEARANCE:
            return False
    return True


def _hull_signature(parts: list[dict]) -> frozenset[str]:
    hull = _convex_hull_indices(parts)
    return frozenset(str(parts[index]["color"]) for index in hull)


def _mismatch(reference: frozenset[str], candidate: frozenset[str]) -> int:
    return len(reference.symmetric_difference(candidate))


def _transform(
    base: tuple[tuple[float, float], ...],
    center: tuple[float, float],
    angle: float,
    scale: float,
    colors: list[str],
) -> list[dict]:
    cosine, sine = math.cos(angle), math.sin(angle)
    parts = []
    for (x, y), color in zip(base, colors):
        tx = scale * (cosine * x - sine * y) + center[0]
        ty = scale * (sine * x + cosine * y) + center[1]
        parts.append({"x": round(tx, 4), "y": round(ty, 4), "color": str(color)})
    return parts


def sample_scene(seed: int) -> dict:
    """Sample one balanced reference/candidate hull-signature scene."""

    rng = random.Random(int(seed))
    base = rng.choice(BASE_SHAPES)
    colors = list(COLORS)
    rng.shuffle(colors)
    reference = _transform(
        base,
        (200.0 + rng.uniform(-8.0, 8.0), 300.0 + rng.uniform(-8.0, 8.0)),
        rng.uniform(-math.pi, math.pi),
        rng.uniform(0.86, 1.03),
        colors,
    )
    candidate_center = (600.0 + rng.uniform(-8.0, 8.0), 300.0 + rng.uniform(-8.0, 8.0))
    candidate_positions_a = _transform(
        base,
        candidate_center,
        rng.uniform(-math.pi, math.pi),
        rng.uniform(0.86, 1.03),
        ["red"] * N_POINTS,
    )
    candidate_positions_a = [
        {"x": part["x"], "y": part["y"], "color": color}
        for part, color in zip(candidate_positions_a, colors)
    ]
    candidate_positions_b = [
        {"x": round(float(part["x"]) + 400.0, 4), "y": part["y"], "color": part["color"]}
        for part in candidate_positions_a
    ]
    hull = _convex_hull_indices(reference)
    interior = [index for index in range(N_POINTS) if index not in hull]
    if len(interior) != 1 or len(hull) != 4:
        raise RuntimeError("base shape did not produce a four-vertex hull")
    wrong_colors = list(colors)
    swap_hull = rng.choice(hull)
    wrong_colors[interior[0]], wrong_colors[swap_hull] = (
        wrong_colors[swap_hull],
        wrong_colors[interior[0]],
    )
    correct = [
        {"x": part["x"], "y": part["y"], "color": color}
        for part, color in zip(candidate_positions_a, colors)
    ]
    wrong = [
        {"x": part["x"], "y": part["y"], "color": color}
        for part, color in zip(candidate_positions_a, wrong_colors)
    ]
    correct_b = [
        {"x": part["x"], "y": part["y"], "color": color}
        for part, color in zip(candidate_positions_b, colors)
    ]
    wrong_b = [
        {"x": part["x"], "y": part["y"], "color": color}
        for part, color in zip(candidate_positions_b, wrong_colors)
    ]
    if int(seed) % 2 == 0:
        candidate_a, candidate_b = correct, wrong_b
    else:
        candidate_a, candidate_b = wrong, correct_b
    return {
        "reference": reference,
        "candidate_a": candidate_a,
        "candidate_b": candidate_b,
        "marker_radius": round(rng.uniform(16.0, 20.0), 3),
        "hull_width": int(rng.choice((3, 4, 5))),
    }


def render(scene: dict) -> Image.Image:
    """Render only the visible panels, points, and convex-hull outlines."""

    width, height = CANVAS_SIZE
    work = Image.new("RGB", (width * SUPERSAMPLE, height * SUPERSAMPLE), (255, 255, 255))
    draw = ImageDraw.Draw(work)
    for box in PANEL_BOXES.values():
        scaled = tuple(int(value * SUPERSAMPLE) for value in box)
        draw.rounded_rectangle(
            scaled,
            radius=14 * SUPERSAMPLE,
            fill=(249, 250, 252),
            outline=(76, 83, 96),
            width=2 * SUPERSAMPLE,
        )
    radius = float(scene.get("marker_radius", 18.0))
    hull_width = max(1, int(scene.get("hull_width", 4)))
    panels = (
        ("reference", scene["reference"], "reference"),
        ("candidate_a", scene["candidate_a"], "A"),
        ("candidate_b", scene["candidate_b"], "B"),
    )
    for name, parts, panel_name in panels:
        hull = _convex_hull_indices(parts)
        points = [(float(part["x"]), float(part["y"])) for part in parts]
        if len(hull) >= 3:
            outline = [points[index] for index in hull] + [points[hull[0]]]
            draw.line(
                [(int(x * SUPERSAMPLE), int(y * SUPERSAMPLE)) for x, y in outline],
                fill=(79, 88, 105),
                width=hull_width * SUPERSAMPLE,
                joint="curve",
            )
        for part in parts:
            x, y = float(part["x"]), float(part["y"])
            box = (
                int((x - radius) * SUPERSAMPLE),
                int((y - radius) * SUPERSAMPLE),
                int((x + radius) * SUPERSAMPLE),
                int((y + radius) * SUPERSAMPLE),
            )
            draw.ellipse(
                box,
                fill=PALETTE[str(part["color"])],
                outline=(31, 37, 49),
                width=2 * SUPERSAMPLE,
            )
    image = work.resize(CANVAS_SIZE, Image.Resampling.LANCZOS)
    final_draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    labels = {"reference": "REFERENCE", "A": "A", "B": "B"}
    for name, box in PANEL_BOXES.items():
        final_draw.text((box[0] + 14, box[1] + 13), labels[name], fill=(25, 29, 38), font=font)
    return image


def analytic_gold(scene: dict) -> str:
    """Select the unique candidate with the reference hull-color signature."""

    if not all(_valid_panel(scene[name]) for name in ("reference", "candidate_a", "candidate_b")):
        raise ValueError("malformed hull scene")
    reference = _hull_signature(scene["reference"])
    scores = (
        _mismatch(reference, _hull_signature(scene["candidate_a"])),
        _mismatch(reference, _hull_signature(scene["candidate_b"])),
    )
    if scores[0] == scores[1]:
        raise ValueError("candidate hull signatures are tied")
    return "A" if scores[0] < scores[1] else "B"


def margin(scene: dict) -> float:
    """Return the gap between the two signature-mismatch scores."""

    reference = _hull_signature(scene["reference"])
    score_a = _mismatch(reference, _hull_signature(scene["candidate_a"]))
    score_b = _mismatch(reference, _hull_signature(scene["candidate_b"]))
    return float(abs(score_a - score_b))


def is_quarantined(scene: dict) -> bool:
    """Exclude malformed, near-edge, or non-unique finite-resolution scenes."""

    try:
        if not all(_valid_panel(scene[name]) for name in ("reference", "candidate_a", "candidate_b")):
            return True
        reference = _hull_signature(scene["reference"])
        scores = [
            _mismatch(reference, _hull_signature(scene["candidate_a"])),
            _mismatch(reference, _hull_signature(scene["candidate_b"])),
        ]
        return scores[0] == scores[1] or min(scores) > 0 and min(scores) >= max(scores)
    except (KeyError, TypeError, ValueError, ZeroDivisionError):
        return True
