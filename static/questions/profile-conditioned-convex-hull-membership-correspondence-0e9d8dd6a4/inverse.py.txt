"""Independent pixel-only inverse for the convex-hull question."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


PALETTE = {
    "red": (214, 55, 56),
    "blue": (42, 101, 190),
    "green": (35, 149, 83),
    "amber": (229, 151, 25),
    "purple": (132, 71, 172),
}
PANEL_RANGES = {
    "reference": (18, 382),
    "A": (418, 782),
    "B": (818, 1182),
}
Y_RANGE = (46, 554)


def _cross(a: tuple[float, float], b: tuple[float, float], c: tuple[float, float]) -> float:
    return (b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0])


def _hull_indices(points: list[tuple[float, float, str]]) -> list[int]:
    indexed = sorted(
        ((x, y, index) for index, (x, y, _) in enumerate(points)),
        key=lambda item: (item[0], item[1], item[2]),
    )
    if len({(x, y) for x, y, _ in indexed}) != len(indexed):
        raise ValueError("coincident colored centroids")
    lower: list[tuple[float, float, int]] = []
    for item in indexed:
        while len(lower) >= 2 and _cross(lower[-2][:2], lower[-1][:2], item[:2]) <= 1e-6:
            lower.pop()
        lower.append(item)
    upper: list[tuple[float, float, int]] = []
    for item in reversed(indexed):
        while len(upper) >= 2 and _cross(upper[-2][:2], upper[-1][:2], item[:2]) <= 1e-6:
            upper.pop()
        upper.append(item)
    cycle = lower[:-1] + upper[:-1]
    if len(cycle) < 3 or len(cycle) == len(points):
        raise ValueError("invalid pixel hull")
    return [item[2] for item in cycle]


def _centroids(image: Image.Image, x_range: tuple[int, int]) -> list[tuple[float, float, str]]:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    x0, x1 = x_range
    y0, y1 = Y_RANGE
    local = array[y0:y1, x0:x1]
    parts = []
    for color_name, rgb in PALETTE.items():
        mask = np.all(local == np.asarray(rgb, dtype=np.uint8), axis=2)
        pixels = np.argwhere(mask)
        if len(pixels) < 80:
            raise ValueError(f"missing or tiny {color_name} component")
        # A panel contains one marker per palette color. The solid color core
        # gives a stable centroid even though the visible boundary is antialiased.
        y_mean, x_mean = pixels.mean(axis=0)
        parts.append((float(x_mean + x0), float(y_mean + y0), color_name))
    return parts


def _signature(points: list[tuple[float, float, str]]) -> frozenset[str]:
    hull = _hull_indices(points)
    return frozenset(points[index][2] for index in hull)


def _mismatch(reference: frozenset[str], candidate: frozenset[str]) -> int:
    return len(reference.symmetric_difference(candidate))


def decision_from_image(image: Image.Image) -> str:
    """Recover A/B using only RGB pixels and the public panel contract."""

    if image.size[0] < 1183 or image.size[1] < 555:
        raise ValueError(f"unexpected raster size {image.size}")
    reference = _signature(_centroids(image, PANEL_RANGES["reference"]))
    candidate_a = _signature(_centroids(image, PANEL_RANGES["A"]))
    candidate_b = _signature(_centroids(image, PANEL_RANGES["B"]))
    score_a = _mismatch(reference, candidate_a)
    score_b = _mismatch(reference, candidate_b)
    if score_a == score_b:
        raise ValueError("pixel hull signatures are tied")
    return "A" if score_a < score_b else "B"
