"""Independent pixel-only inverse arm for the grid-cell correspondence world.

Receives ONLY the final rasterized PNG and the public question contract (the
visual spec). It never imports the renderer, prompt, generator, verifier, or
analytic-gold code, and never reads scene data.

The public visual spec (duplicated here as the oracle's own specification):

  * the top panel shows one neutral 3x3 grid (gray lines) with a dark marker dot
    in one cell;
  * a dark horizontal divider separates the panels;
  * the bottom panel shows three candidate objects, each a 3x3 grid with a dark
    marker dot in some cell and a solid colored border;
  * the task: find the candidate whose marker is in the same (row, column) cell
    as the reference marker, then report that candidate's border color.

The oracle recovers, from pixels alone: the divider, the reference marker cell,
each candidate's marker cell and border color, then the answer color after the
cross-panel cell correspondence.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "grid_cell_correspondence-oracle-0.1.0"

COLOR_PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (220, 40, 40),
    "green": (30, 170, 40),
    "blue": (40, 60, 230),
    "magenta": (230, 0, 230),
}

REF_LEFT = 320
REF_TOP = 50
REF_CELL = 40
CAND_CELL = 32
CAND_TOP = 304
CAND_LEFTS = [132, 332, 532]


def _nearest_names(pixels: np.ndarray, palette: dict[str, tuple[int, int, int]]) -> np.ndarray:
    names = list(palette)
    rgbs = np.array([palette[n] for n in names], dtype=np.float32)
    d = np.linalg.norm(pixels[..., None, :] - rgbs[None, :, :], axis=-1)
    return np.array(names)[d.argmin(axis=-1)]


def _marker_cell(black_ps: np.ndarray, left: int, top: int, cell: int) -> tuple[int, int]:
    cx = float(black_ps[:, 1].mean())
    cy = float(black_ps[:, 0].mean())
    row = int((cy - top) // cell)
    col = int((cx - left) // cell)
    row = max(0, min(2, row))
    col = max(0, min(2, col))
    return row, col


def _border_color(frame_arr: np.ndarray) -> str:
    r = frame_arr[..., 0]
    g = frame_arr[..., 1]
    b = frame_arr[..., 2]
    mx = frame_arr.max(axis=-1)
    mn = frame_arr.min(axis=-1)
    white = (r > 240) & (g > 240) & (b > 240)
    gray = (mx - mn) < 25
    black = mx < 60
    colored = ~(white | gray | black)
    if not colored.any():
        return "red"
    names = _nearest_names(frame_arr[colored][None, :, :], COLOR_PALETTE)[0]
    vals, counts = np.unique(names, return_counts=True)
    return str(vals[int(np.argmax(counts))])


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.float32)
    h, w = arr.shape[:2]
    r, g, b = arr[..., 0], arr[..., 1], arr[..., 2]
    mx = arr.max(axis=-1)
    line = (r < 110) & (g < 110) & (b < 110)
    line_counts = line.sum(axis=1)
    divider_rows = np.where(line_counts > 0.5 * w)[0]
    divider_y = int(round(float(divider_rows.mean()))) if divider_rows.size else h // 2

    black = mx < 60

    top_black = black[:divider_y, :]
    ref_ps = np.argwhere(top_black)
    if ref_ps.shape[0] < 5:
        raise ValueError("could not find the reference marker")
    ref_row, ref_col = _marker_cell(ref_ps, REF_LEFT, REF_TOP, REF_CELL)

    answer = None
    for left in CAND_LEFTS:
        x0 = left
        x1 = left + 3 * CAND_CELL
        y_gl0 = CAND_TOP
        y_gl1 = CAND_TOP + 3 * CAND_CELL
        cell_mask = black[y_gl0:y_gl1, x0:x1]
        cand_ps = np.argwhere(cell_mask)
        if cand_ps.shape[0] < 3:
            raise ValueError("could not find a candidate marker")
        cand_ps[:, 1] += x0
        cand_ps[:, 0] += y_gl0
        row, col = _marker_cell(cand_ps, left, CAND_TOP, CAND_CELL)
        color = _border_color(arr[y_gl0:y_gl1, x0:x1])
        if (row, col) == (ref_row, ref_col):
            answer = color

    if answer is None:
        raise ValueError("could not establish cross-panel cell correspondence")
    return answer
