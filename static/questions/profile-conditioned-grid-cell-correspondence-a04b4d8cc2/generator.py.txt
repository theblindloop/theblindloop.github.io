"""Grid-cell correspondence world renderer (paper-semantic correspondence family).

A cross-panel lattice-position binding task. The top panel shows a neutral
3x3 grid (a square divided into nine equal cells) with a single dark marker dot
occupying exactly one cell. The bottom panel shows three candidate objects, each
its own 3x3 grid of the same shape with a dark marker dot in some cell, and each
outlined by a distinct colored border. Exactly one candidate has its marker dot
in the *same (row, column) cell* as the reference. The question asks: after
matching that object across the panels by its exact grid cell, what is the color
of the border bound to that candidate?

The reference is neutral (only grid lines and a marker); it carries no color, so
the answer color can only be read from the matching candidate in the lower panel.
The pixel-only inverse arm must first recover each object's marker cell (a 2D
lattice coordinate), establish the cross-panel cell correspondence, then read
the bound border color.

Latent state z (all exposed scene fields are rendered -- nothing hidden):

    ref_row, ref_col : the reference marker's grid cell (row 0-2, col 0-2)
    cand_rows        : marker row of each candidate slot (rows 0-2)
    cand_cols        : marker col of each candidate slot (cols 0-2)
    cand_colors      : border color of each candidate slot (palette member)

The analytic gold is derived only from these rendered fields: locate the unique
candidate slot whose (row, col) equals (ref_row, ref_col), then report that
slot's border color. Every such field is drawn into the raster.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "grid_cell_correspondence-0.1.0"

BACKGROUND = "white"
GRID_LINE = (150, 150, 150)
MARKER = (20, 20, 20)

COLOR_PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (220, 40, 40),
    "green": (30, 170, 40),
    "blue": (40, 60, 230),
    "magenta": (230, 0, 230),
}

# Geometry (public visual spec; constant across scenes for counterbalance).
WIDTH = 760
HEIGHT = 440
DIVIDER_Y = 230
DIVIDER_THICK = 3
DIVIDER = (70, 70, 70)

# Reference 3x3 grid.
REF_CELL = 40            # grid cell side in px
REF_LEFT = 320
REF_TOP = 50
REF_MARKER_R = 9

# Candidate 3x3 grids.
CAND_CELL = 32
CAND_TOP = 304
CAND_LEFTS = [132, 332, 532]
_BORDER_W = 5
CAND_MARKER_R = 7

_MARGIN_FLOOR = 180.0


def _color_dist(a: tuple[int, int, int], b: tuple[int, int, int]) -> float:
    return math.dist(a, b)


def _draw_grid(draw, left: int, top: int, cell: int, border_rgb, marker_rgb, mrow: int, mcol: int):
    side = 3 * cell
    right = left + side
    bottom = top + side
    # Outer border in the candidate's color (or neutral gray for the reference).
    draw.rectangle([left, top, right, bottom], outline=border_rgb, width=_BORDER_W)
    # Inner grid lines.
    for k in (1, 2):
        x = left + k * cell
        y = top + k * cell
        draw.line([x, top + 2, x, bottom - 2], fill=GRID_LINE, width=2)
        draw.line([left + 2, y, right - 2, y], fill=GRID_LINE, width=2)
    # Marker dot at the center of cell (mrow, mcol).
    mx = left + mcol * cell + cell // 2
    my = top + mrow * cell + cell // 2
    r = REF_MARKER_R if marker_rgb == MARKER and cell == REF_CELL else CAND_MARKER_R
    draw.ellipse([mx - r, my - r, mx + r, my + r], fill=marker_rgb)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    ref_row = rng.randrange(3)
    ref_col = rng.randrange(3)
    n = 3
    # Exactly one candidate shares the reference cell; the other two do not.
    match_idx = rng.randrange(n)
    cand_rows = []
    cand_cols = []
    for i in range(n):
        if i == match_idx:
            row, col = ref_row, ref_col
        else:
            while True:
                row = rng.randrange(3)
                col = rng.randrange(3)
                if (row, col) != (ref_row, ref_col):
                    break
        cand_rows.append(row)
        cand_cols.append(col)
    colors = rng.sample(list(COLOR_PALETTE), n)
    return {
        "ref_row": ref_row,
        "ref_col": ref_col,
        "cand_rows": cand_rows,
        "cand_cols": cand_cols,
        "cand_colors": colors,
        "width": WIDTH,
        "height": HEIGHT,
    }


def render(scene: dict) -> Image.Image:
    w = int(scene["width"])
    h = int(scene["height"])
    img = Image.new("RGB", (w, h), BACKGROUND)
    draw = ImageDraw.Draw(img)
    _draw_grid(
        draw,
        REF_LEFT,
        REF_TOP,
        REF_CELL,
        GRID_LINE,
        MARKER,
        int(scene["ref_row"]),
        int(scene["ref_col"]),
    )
    for left, row, col, color_name in zip(
        CAND_LEFTS,
        scene["cand_rows"],
        scene["cand_cols"],
        scene["cand_colors"],
        strict=True,
    ):
        _draw_grid(
            draw,
            int(left),
            CAND_TOP,
            CAND_CELL,
            COLOR_PALETTE[color_name],
            MARKER,
            int(row),
            int(col),
        )
    draw.rectangle([0, DIVIDER_Y, w, DIVIDER_Y + DIVIDER_THICK], fill=DIVIDER)
    return img


def _matched_index(scene: dict) -> int:
    for i in range(3):
        if scene["cand_rows"][i] == scene["ref_row"] and scene["cand_cols"][i] == scene["ref_col"]:
            return i
    raise ValueError("no candidate matches the reference cell")


def analytic_gold(scene: dict) -> str:
    """Decision: the border color bound to the reference-matched candidate."""
    return scene["cand_colors"][_matched_index(scene)]


def margin(scene: dict) -> float:
    """Runner-up margin: distance from the answer color to the nearest other
    candidate color. Large because candidate border colors are always distinct."""
    idx = _matched_index(scene)
    answer = scene["cand_colors"][idx]
    d = [
        _color_dist(COLOR_PALETTE[answer], COLOR_PALETTE[scene["cand_colors"][i]])
        for i in range(3)
        if i != idx
    ]
    return float(min(d))


def is_quarantined(scene: dict) -> bool:
    """Quarantine only a genuinely ambiguous scene (never produced by the sampler,
    which always gives candidates distinct border colors and a unique cell match)."""
    return margin(scene) < _MARGIN_FLOOR


def latent_symmetries(scene: dict) -> list:
    """not_applicable: every label-influencing field is rendered, so the raster
    uniquely determines the gold; no pixel-identical transform is declared."""
    return []
