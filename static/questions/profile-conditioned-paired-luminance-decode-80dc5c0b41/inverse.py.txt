"""Independent pixel-only inverse for the paired luminance world."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _find_inner_frame(array: np.ndarray) -> tuple[int, int, int, int]:
    spread = array.max(axis=2).astype(np.int16) - array.min(axis=2).astype(np.int16)
    saturated = spread > 60
    ys, xs = np.nonzero(saturated)
    if len(xs) < 100:
        raise ValueError("visible saturated frame is missing")
    left, right = int(xs.min()), int(xs.max())
    top, bottom = int(ys.min()), int(ys.max())
    cy, cx = (top + bottom) // 2, (left + right) // 2

    horizontal = saturated[cy]
    x = left
    while x <= right and horizontal[x]:
        x += 1
    inner_left = x
    x = right
    while x >= left and horizontal[x]:
        x -= 1
    inner_right = x

    vertical = saturated[:, cx]
    y = top
    while y <= bottom and vertical[y]:
        y += 1
    inner_top = y
    y = bottom
    while y >= top and vertical[y]:
        y -= 1
    inner_bottom = y
    if inner_right - inner_left < 250 or inner_bottom - inner_top < 250:
        raise ValueError("visible frame has no usable interior")
    return inner_left, inner_top, inner_right + 1, inner_bottom + 1


def _public_templates() -> dict[str, np.ndarray]:
    plus = np.zeros((9, 9), dtype=np.uint8)
    plus[4, :] = 1
    plus[:, 4] = 1
    cross = np.zeros((9, 9), dtype=np.uint8)
    indices = np.arange(9)
    cross[indices, indices] = 1
    cross[indices, 8 - indices] = 1
    return {"plus": plus, "X": cross}


def _decode(image: Image.Image) -> tuple[np.ndarray, float]:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    left, top, right, bottom = _find_inner_frame(array)
    crop = array[top:bottom, left:right].astype(np.float32).mean(axis=2)
    height, width = crop.shape
    if abs(width - height) > 3:
        raise ValueError("framed matrix is not square")
    bits = np.zeros((9, 9), dtype=np.uint8)
    contrasts = []
    for row in range(9):
        ya, yb = round(row * height / 9), round((row + 1) * height / 9)
        for col in range(9):
            xa, xb = round(col * width / 9), round((col + 1) * width / 9)
            cell = crop[ya:yb, xa:xb]
            h, w = cell.shape
            band = cell[round(0.22 * h) : round(0.78 * h)]
            left_tone = float(np.median(band[:, round(0.12 * w) : round(0.40 * w)]))
            right_tone = float(np.median(band[:, round(0.60 * w) : round(0.88 * w)]))
            difference = right_tone - left_tone
            contrasts.append(abs(difference))
            bits[row, col] = int(difference > 0)
    return bits, min(contrasts)


def decision_from_image(image: Image.Image) -> str:
    bits, minimum_contrast = _decode(image)
    if minimum_contrast < 8:
        raise ValueError("split-tone evidence is erased or below raster resolution")
    ranked = sorted(
        (int(np.count_nonzero(bits != template)), label)
        for label, template in _public_templates().items()
    )
    if ranked[0][0] > 4 or ranked[1][0] - ranked[0][0] < 6:
        raise ValueError("decoded grid is not an unambiguous plus or X")
    return ranked[0][1]
