"""Deterministic latent sampler and raster renderer for paired luminance decoding."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

GRID = 9
CANVAS = 520
LABELS = ("plus", "X")


def _template(name: str) -> list[list[int]]:
    bits = [[0 for _ in range(GRID)] for _ in range(GRID)]
    if name == "plus":
        for i in range(GRID):
            bits[4][i] = 1
            bits[i][4] = 1
    elif name == "X":
        for i in range(GRID):
            bits[i][i] = 1
            bits[i][GRID - 1 - i] = 1
    else:
        raise ValueError(f"unknown symbol {name!r}")
    return bits


def _flat(values: Iterable[Iterable[int]]) -> list[int]:
    return [int(value) for row in values for value in row]


def _classify(bits: list[list[int]]) -> tuple[str, int, int]:
    observed = _flat(bits)
    scores = []
    for label in LABELS:
        expected = _flat(_template(label))
        scores.append((sum(a != b for a, b in zip(observed, expected)), label))
    scores.sort()
    return scores[0][1], scores[0][0], scores[1][0] - scores[0][0]


def sample_scene(seed: int) -> dict:
    """Pair consecutive labels with identical nuisance variables for balance."""
    label = LABELS[seed % 2]
    nuisance = random.Random(seed // 2)
    bits = _template(label)
    cell_size = nuisance.choice((38, 40, 42, 44))
    gap = nuisance.choice((1, 2))
    grid_size = GRID * cell_size
    centered = (CANVAS - grid_size) // 2
    offset_x = centered + nuisance.randint(-18, 18)
    offset_y = centered + nuisance.randint(-18, 18)
    palettes = (
        ((24, 126, 206), (245, 178, 38)),
        ((18, 151, 102), (214, 68, 116)),
        ((139, 77, 191), (236, 153, 34)),
        ((210, 92, 30), (31, 143, 186)),
    )
    frame_color, marker_color = palettes[nuisance.randrange(len(palettes))]
    low_contrast = (seed // 2) % 11 == 0
    choices = (16, 18) if low_contrast else (48, 56, 64, 72, 80)
    pair_values = []
    for row in bits:
        out_row = []
        for bit in row:
            delta = nuisance.choice(choices)
            darker, lighter = 150 - delta // 2, 150 + delta // 2
            out_row.append([darker, lighter] if bit else [lighter, darker])
        pair_values.append(out_row)
    return {
        "bits": bits,
        "pair_values": pair_values,
        "cell_size": cell_size,
        "gap": gap,
        "offset_x": offset_x,
        "offset_y": offset_y,
        "frame_color": list(frame_color),
        "marker_color": list(marker_color),
    }


def render(scene: dict) -> Image.Image:
    bits = scene["bits"]
    pairs = scene["pair_values"]
    cell = int(scene["cell_size"])
    line = int(scene["gap"])
    x0 = int(scene["offset_x"])
    y0 = int(scene["offset_y"])
    frame = tuple(scene["frame_color"])
    marker = tuple(scene["marker_color"])
    size = GRID * cell
    image = Image.new("RGB", (CANVAS, CANVAS), (250, 250, 248))
    draw = ImageDraw.Draw(image)

    # The saturated frame is also the measurable crop fiducial.
    draw.rectangle((x0 - 5, y0 - 5, x0 + size + 4, y0 + size + 4), outline=frame, width=5)
    draw.polygon(
        ((x0 - 5, y0 - 5), (x0 + 15, y0 - 5), (x0 - 5, y0 + 15)), fill=marker
    )
    for r in range(GRID):
        for c in range(GRID):
            # Access bits during rendering so the label input is visibly committed.
            bit = int(bits[r][c])
            left, right = (int(v) for v in pairs[r][c])
            if (left < right) != bool(bit):
                raise ValueError("pair_values disagree with rendered comparison bits")
            xa, ya = x0 + c * cell, y0 + r * cell
            xb, yb = xa + cell - 1, ya + cell - 1
            mid = xa + cell // 2
            draw.rectangle((xa, ya, mid - 1, yb), fill=(left, left, left))
            draw.rectangle((mid, ya, xb, yb), fill=(right, right, right))
            draw.rectangle((xa, ya, xb, yb), outline=(226, 226, 223), width=line)
            draw.line((mid, ya, mid, yb), fill=(226, 226, 223), width=line)
    return image


def analytic_gold(scene: dict) -> str:
    label, best, gap = _classify(scene["bits"])
    if best != 0 or gap < 6:
        raise ValueError("latent bits do not encode one unambiguous public template")
    return label


def margin(scene: dict) -> float:
    _, _, runner_gap = _classify(scene["bits"])
    contrast = min(abs(int(a) - int(b)) for row in scene["pair_values"] for a, b in row)
    return float(min(runner_gap, contrast / 8.0))


def is_quarantined(scene: dict) -> bool:
    _, _, runner_gap = _classify(scene["bits"])
    contrast = min(abs(int(a) - int(b)) for row in scene["pair_values"] for a, b in row)
    return contrast < 20 or runner_gap < 6


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
