"""Deterministic raster world for a courtyard with an island or peninsula."""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (248, 247, 242)
PALE = (211, 212, 209)
PALETTES = ((24, 126, 137), (38, 103, 157), (112, 78, 151))


def sample_scene(seed: int) -> dict:
    """Sample balanced, comfortably separated topology from one integer seed."""
    rng = random.Random(int(seed))
    decorations = []
    corner_boxes = ((40, 52), (438, 48), (48, 438), (442, 442))
    for x, y in rng.sample(list(corner_boxes), 2):
        decorations.append([x + rng.randint(-5, 5), y + rng.randint(-5, 5), rng.randint(7, 11)])
    return {
        "detached": bool(int(seed) % 2 == 0),
        "origin_x": 256 + rng.randint(-4, 4),
        "origin_y": 256 + rng.randint(-4, 4),
        "rotation": rng.randrange(4),
        "reflection": bool(rng.randrange(2)),
        "palette": rng.randrange(len(PALETTES)),
        "decorations": decorations,
        "margin_px": 32,
    }


def _emblem(detached: bool, rotation: int, reflection: bool) -> np.ndarray:
    """Build equal-ink masks; only component attachment differs."""
    mask = np.ones((224, 224), dtype=np.uint8)
    mask[32:192, 32:192] = 0                         # broad courtyard
    mask[80:144, 96:160] = 1                         # inner region
    if detached:
        mask[96:128, 32:64] = 1                      # equal-ink outer tab
    else:
        mask[96:128, 160:192] = 1                    # broad joining causeway
    if reflection:
        mask = np.fliplr(mask)
    return np.rot90(mask, int(rotation) % 4).copy()


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x, y, radius in scene["decorations"]:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=PALE)
        draw.line((x - radius - 4, y, x + radius + 4, y), fill=PALE, width=3)

    mask = _emblem(
        bool(scene["detached"]),
        int(scene["rotation"]),
        bool(scene["reflection"]),
    )
    color = PALETTES[int(scene["palette"]) % len(PALETTES)]
    patch = np.empty((224, 224, 3), dtype=np.uint8)
    patch[:] = np.asarray(BACKGROUND, dtype=np.uint8)
    patch[mask.astype(bool)] = np.asarray(color, dtype=np.uint8)
    left = int(scene["origin_x"]) - 112
    top = int(scene["origin_y"]) - 112
    image.paste(Image.fromarray(patch, mode="RGB"), (left, top))
    return image


def analytic_gold(scene: dict) -> str:
    return "yes" if bool(scene["detached"]) else "no"


def margin(scene: dict) -> float:
    return float(scene["margin_px"])


def is_quarantined(scene: dict) -> bool:
    return float(scene["margin_px"]) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
