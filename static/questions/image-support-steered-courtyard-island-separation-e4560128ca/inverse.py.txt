"""Independent image-only inverse for courtyard island topology."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[int]:
    """Return four-connected component sizes within a modest crop."""
    work = mask.copy()
    height, width = work.shape
    sizes: list[int] = []
    for y, x in np.argwhere(work):
        y = int(y)
        x = int(x)
        if not work[y, x]:
            continue
        work[y, x] = False
        queue = deque([(y, x)])
        size = 0
        while queue:
            cy, cx = queue.popleft()
            size += 1
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        sizes.append(size)
    return sorted(sizes, reverse=True)


def _bounded_white_components(foreground: np.ndarray) -> list[int]:
    """Count complement components not connected to the crop border."""
    background = ~foreground
    height, width = background.shape
    queue: deque[tuple[int, int]] = deque()
    for x in range(width):
        for y in (0, height - 1):
            if background[y, x]:
                background[y, x] = False
                queue.append((y, x))
    for y in range(height):
        for x in (0, width - 1):
            if background[y, x]:
                background[y, x] = False
                queue.append((y, x))
    while queue:
        cy, cx = queue.popleft()
        for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
            if 0 <= ny < height and 0 <= nx < width and background[ny, nx]:
                background[ny, nx] = False
                queue.append((ny, nx))
    return _components(background)


def decision_from_image(image: Image.Image) -> str:
    """Infer only from saturated solid pixels in the final RGB raster."""
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    channel_max = array.max(axis=2)
    channel_min = array.min(axis=2)
    foreground = ((channel_max - channel_min) >= 45) & (channel_max <= 190)
    points = np.argwhere(foreground)
    if len(points) == 0:
        raise ValueError("no colored sheet")

    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    box_height = int(y1 - y0 + 1)
    box_width = int(x1 - x0 + 1)
    if not (220 <= box_height <= 228 and 220 <= box_width <= 228):
        raise ValueError("colored sheet has an invalid extent")
    pixel_count = int(foreground.sum())
    if abs(pixel_count - 29696) > 500:
        raise ValueError("colored sheet has damaged regional support")

    pad_y0 = max(0, int(y0) - 1)
    pad_x0 = max(0, int(x0) - 1)
    pad_y1 = min(foreground.shape[0], int(y1) + 2)
    pad_x1 = min(foreground.shape[1], int(x1) + 2)
    crop = foreground[pad_y0:pad_y1, pad_x0:pad_x1]
    solid_sizes = _components(crop)
    if len(solid_sizes) not in (1, 2):
        raise ValueError("unexpected colored component count")
    if solid_sizes[0] < 25000:
        raise ValueError("outer sheet is fragmented")
    if len(solid_sizes) == 2 and not (3500 <= solid_sizes[1] <= 4500):
        raise ValueError("detached region has an invalid area")

    pockets = _bounded_white_components(crop)
    if len(pockets) != 1 or not (14000 <= pockets[0] <= 22000):
        raise ValueError("the broad enclosed courtyard is not recoverable")
    return "yes" if len(solid_sizes) == 2 else "no"
