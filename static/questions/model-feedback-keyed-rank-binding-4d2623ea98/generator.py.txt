"""Keyed rank-binding correspondence world.

A KEY row shows N coloured tokens, each with a rank ruler beneath it: one
small filled cell per unit of that token's rank, so the ruler's filled-cell
count is the token's rank (1..N). The KEY therefore records a colour-identity
to-rank binding. Two candidate rows A and B below show the same N colours
without rulers. Exactly one candidate row, when each token is translated to its
KEY rank and the translated ranks are read left-to-right, is strictly ascending
1..N; the other is not. The decision is which single row, A or B, lists the
tokens in increasing rank order -- a correspondence-binding program that binds
each token identity to its keyed rank and then carries that binding across
panels to recover order.

The renderer owns scene sampling, rasterization and the analytic gold. The
independent pixel-only inverse arm (``oracle.py``) recovers the same decision
from the final PNG alone: it detects each coloured token, reads the ruler
filled-cell count under each KEY token to build the colour->rank map, then for
each candidate row translates its left-to-right colour sequence to ranks and
checks which one is ascending.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw, ImageFont

WIDTH = 720
HEIGHT = 660
BACKGROUND = (255, 255, 255)
LABEL = (35, 35, 35)
PIP = (45, 45, 45)

N = 6
COLOR_NAMES = ["red", "green", "blue", "orange", "purple", "brown"]
COLOR_RGB = {
    "red": (220, 40, 40),
    "green": (40, 170, 40),
    "blue": (40, 80, 220),
    "orange": (255, 150, 20),
    "purple": (150, 60, 190),
    "brown": (120, 80, 40),
}

TOKEN = 46
PITCH = 96
FIRST_CX = 150
KEY_TOP = 96
A_TOP = 408
B_TOP = 546
LABEL_Y = 52

CELL = 9
CELL_GAP = 2
RULER_GAP = 12
RULER_W = N * CELL + (N - 1) * CELL_GAP

RANK_MAX = N
QUAR_DECOY_MISPLACE = 2

_FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"

_CX = [FIRST_CX + i * PITCH for i in range(N)]


def _perm(rng: np.random.Generator, n: int) -> list[int]:
    return list(rng.permutation(n))


def sample_scene(seed: int) -> dict:
    """Sample one latent scene deterministically from ``seed``."""
    rng = np.random.default_rng(seed)

    key_colors = [COLOR_NAMES[i] for i in _perm(rng, N)]
    key_ranks = [int(r) + 1 for r in _perm(rng, N)]
    color_to_rank = dict(zip(key_colors, key_ranks))

    sorted_seq = sorted(COLOR_NAMES, key=lambda name: color_to_rank[name])

    used_positions = set()
    while True:
        decoy = [COLOR_NAMES[i] for i in _perm(rng, N)]
        misplaced = sum(color_to_rank[name] != (pos + 1) for pos, name in enumerate(decoy))
        if misplaced >= QUAR_DECOY_MISPLACE:
            break

    correct_index = int(rng.integers(0, 2))
    if correct_index == 0:
        a_order = list(sorted_seq)
        b_order = decoy
    else:
        a_order = decoy
        b_order = list(sorted_seq)

    return {
        "key_colors": key_colors,
        "key_ranks": key_ranks,
        "a_order": a_order,
        "b_order": b_order,
        "correct_index": correct_index,
    }


def _draw_token(draw: ImageDraw.ImageDraw, cx: int, top: int, color: str) -> None:
    x0 = cx - TOKEN // 2
    draw.rectangle([x0, top, x0 + TOKEN, top + TOKEN], fill=COLOR_RGB[color], outline=(255, 255, 255), width=1)


def _draw_ruler(draw: ImageDraw.ImageDraw, cx: int, top: int, rank: int) -> None:
    x0 = cx - RULER_W // 2
    y0 = top
    for i in range(rank):
        x = x0 + i * (CELL + CELL_GAP)
        draw.rectangle([x, y0, x + CELL, y0 + CELL], fill=PIP)


def render(scene: dict) -> Image.Image:
    """Rasterize the latent scene (a deterministic PNG)."""
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(img)
    try:
        font = ImageFont.truetype(_FONT_PATH, 26)
    except OSError:
        font = ImageFont.load_default()

    draw.text((24, LABEL_Y), "KEY", font=font, fill=LABEL)
    for i, color in enumerate(scene["key_colors"]):
        _draw_token(draw, _CX[i], KEY_TOP, color)
        _draw_ruler(draw, _CX[i], KEY_TOP + TOKEN + RULER_GAP, scene["key_ranks"][i])

    draw.text((24, A_TOP + 8), "A", font=font, fill=LABEL)
    for i, color in enumerate(scene["a_order"]):
        _draw_token(draw, _CX[i], A_TOP, color)

    draw.text((24, B_TOP + 8), "B", font=font, fill=LABEL)
    for i, color in enumerate(scene["b_order"]):
        _draw_token(draw, _CX[i], B_TOP, color)

    return img


def _color_to_rank(scene: dict) -> dict:
    return dict(zip(scene["key_colors"], scene["key_ranks"]))


def _is_ascending(seq: list, color_to_rank: dict) -> bool:
    ranks = [color_to_rank[name] for name in seq]
    return ranks == list(range(1, N + 1))


def analytic_gold(scene: dict) -> str:
    """Structural gold: which candidate row is bound to ascending rank order."""
    color_to_rank = _color_to_rank(scene)
    a_asc = _is_ascending(scene["a_order"], color_to_rank)
    b_asc = _is_ascending(scene["b_order"], color_to_rank)
    if a_asc:
        return "A"
    if b_asc:
        return "B"
    raise ValueError("neither candidate row is bound to ascending rank order")


def margin(scene: dict) -> float:
    """Commitment magnitude: how disordered the decoy row is (misplaced tokens)."""
    color_to_rank = _color_to_rank(scene)
    a_asc = _is_ascending(scene["a_order"], color_to_rank)
    b_asc = _is_ascending(scene["b_order"], color_to_rank)
    decoy = scene["b_order"] if a_asc else scene["a_order"]
    misplaced = sum(color_to_rank[name] != (pos + 1) for pos, name in enumerate(decoy))
    return float(misplaced)


def is_quarantined(scene: dict) -> bool:
    """Quarantine scenes where the decoy is barely disordered (ambiguous)."""
    return margin(scene) < QUAR_DECOY_MISPLACE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The scene is injectively rendered; no latent aliases are declared."""
    return []
