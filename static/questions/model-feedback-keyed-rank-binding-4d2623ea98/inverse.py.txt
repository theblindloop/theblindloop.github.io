"""Independent pixel-only inverse arm for the keyed rank-binding world.

Recovers the decision (which candidate row is bound to ascending KEY rank
order) from the final rasterized PNG alone. It locates each coloured token in
the three rows by exact colour, reads the filled-cell rank ruler beneath each
KEY token to build the colour->rank map, then for each candidate row translates
its left-to-right colour sequence to ranks and returns the label of the row
whose ranks ascend 1..N. It never receives the latent scene or analytic gold and
imports only numpy/PIL.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

COLOR_RGB = {
    "red": (220, 40, 40),
    "green": (40, 170, 40),
    "blue": (40, 80, 220),
    "orange": (255, 150, 20),
    "purple": (150, 60, 190),
    "brown": (120, 80, 40),
}
PIP = (45, 45, 45)

TOKEN = 46
KEY_TOP = 96
A_TOP = 408
B_TOP = 546
CELL = 9
CELL_GAP = 2
RULER_GAP = 12
RULER_W = 6 * CELL + 5 * CELL_GAP
RULER_TOP = KEY_TOP + TOKEN + RULER_GAP


def _tokens_in_band(arr: np.ndarray, top: int) -> list[tuple[str, int]]:
    """Return (color_name, center_x) for each token in a row band, left-to-right."""
    band = arr[top : top + TOKEN]
    found = []
    for name, rgb in COLOR_RGB.items():
        mask = (band == np.array(rgb)).all(axis=-1)
        cols = np.nonzero(np.any(mask, axis=0))[0]
        if cols.size == 0:
            continue
        cx = int(cols.min() + cols.max()) // 2
        found.append((name, cx))
    found.sort(key=lambda item: item[1])
    return found


def _ruler_rank(arr: np.ndarray, cx: int) -> int:
    """Read the filled-cell count (rank) under a KEY token at center x."""
    band = arr[RULER_TOP : RULER_TOP + CELL]
    pip = (band == np.array(PIP)).all(axis=-1)
    x0 = cx - RULER_W // 2
    runs = 0
    prev_on = False
    for dx in range(RULER_W):
        x = x0 + dx
        if x < 0 or x >= band.shape[1]:
            continue
        on = bool(np.any(pip[:, x]))
        if on and not prev_on:
            runs += 1
        prev_on = on
    return runs


def _row_ranks(arr: np.ndarray, top: int, color_to_rank: dict[str, int]) -> list[int]:
    tokens = _tokens_in_band(arr, top)
    return [color_to_rank[name] for name, _ in tokens]


def decision_from_image(image: Image.Image) -> str:
    """Return A or B: which candidate row is bound to ascending KEY rank order."""
    arr = np.asarray(image.convert("RGB"))

    key_tokens = _tokens_in_band(arr, KEY_TOP)
    if len(key_tokens) != 6:
        return "A"
    color_to_rank = {}
    for name, cx in key_tokens:
        r = _ruler_rank(arr, cx)
        color_to_rank[name] = r if r >= 1 else 1
    if len(color_to_rank) != 6:
        return "A"

    a_ranks = _row_ranks(arr, A_TOP, color_to_rank)
    b_ranks = _row_ranks(arr, B_TOP, color_to_rank)
    target = list(range(1, 7))
    return "A" if a_ranks == target else "B"
