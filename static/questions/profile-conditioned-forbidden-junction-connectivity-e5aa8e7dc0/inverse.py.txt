"""Independent inverse arm: only final RGB pixels enter this module."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE = np.array(
    [
        [248, 247, 243],
        [39, 45, 54],
        [60, 126, 201],
        [35, 166, 92],
        [235, 126, 38],
    ],
    dtype=np.int32,
)


def _nearest_labels(image: Image.Image) -> np.ndarray:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int32)
    labels = np.zeros(pixels.shape[:2], dtype=np.uint8)
    best = np.full(pixels.shape[:2], 2**31 - 1, dtype=np.int64)
    for index, color in enumerate(PALETTE):
        delta = pixels - color
        distance = (delta * delta).sum(axis=2)
        take = distance < best
        labels[take] = index
        best[take] = distance[take]
    return labels


def _green_components(green: np.ndarray) -> list[tuple[int, int]]:
    remaining = green.copy()
    representatives = []
    height, width = green.shape
    while remaining.any():
        y, x = np.argwhere(remaining)[0]
        representatives.append((int(y), int(x)))
        remaining[y, x] = False
        queue = deque([(int(y), int(x))])
        while queue:
            cy, cx = queue.popleft()
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and remaining[ny, nx]:
                    remaining[ny, nx] = False
                    queue.append((ny, nx))
    return representatives


def decision_from_image(image: Image.Image) -> str:
    labels = _nearest_labels(image)
    pixels = np.asarray(image.convert("RGB"), dtype=np.int32)
    green_delta = pixels - PALETTE[3]
    green_core = (green_delta * green_delta).sum(axis=2) < 900
    markers = _green_components(green_core)
    if len(markers) != 3:
        raise ValueError(f"expected three visible green junctions, found {len(markers)}")
    traversable = (labels == 1) | (labels == 2) | (labels == 3)
    height, width = traversable.shape
    visited = np.zeros_like(traversable)
    start = markers[0]
    visited[start] = True
    queue = deque([start])
    while queue:
        cy, cx = queue.popleft()
        for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
            if (0 <= ny < height and 0 <= nx < width and traversable[ny, nx]
                    and not visited[ny, nx]):
                visited[ny, nx] = True
                queue.append((ny, nx))
    return "yes" if all(visited[marker] for marker in markers) else "no"
