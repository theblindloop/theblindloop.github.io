"""Latent sampler and raster renderer for forbidden-junction connectivity."""

from __future__ import annotations

import itertools
import random
from collections import deque

from PIL import Image, ImageDraw

NODES = tuple(range(16))
GRID_EDGES = tuple(
    [(4 * r + c, 4 * r + c + 1) for r in range(4) for c in range(3)]
    + [(4 * r + c, 4 * (r + 1) + c) for r in range(3) for c in range(4)]
)
BASE_PALETTE = {
    "background": [248, 247, 243],
    "link": [39, 45, 54],
    "neutral": [60, 126, 201],
    "terminal": [35, 166, 92],
    "forbidden": [235, 126, 38],
}


def _reachable(edges, starts, removed):
    allowed = set(NODES) - set(removed)
    if not starts or starts[0] not in allowed:
        return set()
    adjacency = {node: [] for node in allowed}
    for a, b in edges:
        if a in allowed and b in allowed:
            adjacency[a].append(b)
            adjacency[b].append(a)
    seen = {starts[0]}
    queue = deque([starts[0]])
    while queue:
        node = queue.popleft()
        for other in adjacency[node]:
            if other not in seen:
                seen.add(other)
                queue.append(other)
    return seen


def _decision(edges, green_nodes, orange_nodes):
    return all(node in _reachable(edges, green_nodes, orange_nodes) for node in green_nodes)


def _whole_graph_connected(edges):
    return len(_reachable(edges, [0], [])) == len(NODES)


def _positions(rng):
    transform = rng.randrange(8)
    points = []
    for node in NODES:
        r, c = divmod(node, 4)
        if transform & 1:
            r, c = c, r
        if transform & 2:
            r = 3 - r
        if transform & 4:
            c = 3 - c
        x = 91 + 110 * c + rng.randint(-7, 7)
        y = 91 + 110 * r + rng.randint(-7, 7)
        points.append([x, y])
    return points


def sample_scene(seed: int) -> dict:
    """Return a balanced deterministic scene; seed parity fixes only the label."""
    rng = random.Random(int(seed))
    wanted = (int(seed) % 2 == 0)
    for _ in range(20000):
        green = sorted(rng.sample(NODES, 3))
        orange = sorted(rng.sample([n for n in NODES if n not in green], 2))
        edges = sorted(rng.sample(GRID_EDGES, 17))
        if not _whole_graph_connected(edges):
            continue
        if any(sum(node in edge for edge in edges) < 2 for node in green + orange):
            continue
        if _decision(edges, green, orange) != wanted:
            continue
        return {
            "edges": [list(edge) for edge in edges],
            "green_nodes": green,
            "orange_nodes": orange,
            "positions": _positions(rng),
            "palette": {key: list(value) for key, value in BASE_PALETTE.items()},
        }
    raise RuntimeError(f"could not sample requested class for seed {seed}")


def render(scene: dict) -> Image.Image:
    edges = scene["edges"]
    green = set(scene["green_nodes"])
    orange = set(scene["orange_nodes"])
    positions = scene["positions"]
    palette = scene["palette"]
    scale = 3
    image = Image.new("RGB", (512 * scale, 512 * scale), tuple(palette["background"]))
    draw = ImageDraw.Draw(image)
    for a, b in edges:
        pa = tuple(value * scale for value in positions[a])
        pb = tuple(value * scale for value in positions[b])
        draw.line([pa, pb], fill=tuple(palette["link"]), width=9 * scale)
    radius = 16 * scale
    for node, (x, y) in enumerate(positions):
        role = "terminal" if node in green else "forbidden" if node in orange else "neutral"
        box = ((x * scale - radius), (y * scale - radius),
               (x * scale + radius), (y * scale + radius))
        outline_role = "background" if role == "forbidden" else "link"
        outline_width = 5 * scale if role == "forbidden" else 2 * scale
        draw.ellipse(box, fill=tuple(palette[role]), outline=tuple(palette[outline_role]),
                     width=outline_width)
    return image.resize((512, 512), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    edges = scene["edges"]
    green_nodes = scene["green_nodes"]
    orange_nodes = scene["orange_nodes"]
    return "yes" if _decision(edges, green_nodes, orange_nodes) else "no"


def margin(scene: dict) -> float:
    edges = scene["edges"]
    green = scene["green_nodes"]
    orange = scene["orange_nodes"]
    if not _decision(edges, green, orange):
        for count in (1, 2):
            for restored in itertools.combinations(orange, count):
                removed = [node for node in orange if node not in restored]
                if _decision(edges, green, removed):
                    return float(count)
        return 3.0
    blue = [node for node in NODES if node not in green and node not in orange]
    for count in range(1, len(blue) + 1):
        for added in itertools.combinations(blue, count):
            if not _decision(edges, green, list(orange) + list(added)):
                return float(count)
    return float(len(blue) + 1)


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["edges"] = list(reversed(scene["edges"]))
    return [("reverse_same_ink_edge_draw_order", twin)]
