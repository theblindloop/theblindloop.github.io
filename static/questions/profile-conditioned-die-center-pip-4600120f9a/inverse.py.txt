"""Independent pixel-only inverse arm for die_center_pip.

Receives the rasterised PNG and nothing else: no latent scene, no renderer,
prompt, generator, verifier or analytic-gold import. It locates the die face as
the dominant non-background colour, takes the exact centre of that face's
bounding box, and reports whether that location is painted in the pip colour.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "die_center_pip-oracle-0.1.0"
ABSTAIN = "abstain"

VOTE_RADIUS = 2  # pixels sampled around the recovered face centre


def _modal_color(pixels: np.ndarray) -> tuple[int, int, int] | None:
    if pixels.size == 0:
        return None
    flat = pixels.reshape(-1, 3).astype(np.int64)
    keys = flat[:, 0] * 65536 + flat[:, 1] * 256 + flat[:, 2]
    vals, counts = np.unique(keys, return_counts=True)
    k = int(vals[int(np.argmax(counts))])
    return (k >> 16 & 255, k >> 8 & 255, k & 255)


def decision_from_image(image: Image.Image) -> str:
    try:
        arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
        if arr.ndim != 3 or min(arr.shape[:2]) < 32:
            return ABSTAIN

        border = np.concatenate([
            arr[:2].reshape(-1, 3), arr[-2:].reshape(-1, 3),
            arr[:, :2].reshape(-1, 3), arr[:, -2:].reshape(-1, 3),
        ])
        bg = _modal_color(border)
        if bg is None:
            return ABSTAIN
        bg_mask = np.all(arr == np.array(bg, dtype=np.uint8), axis=-1)
        if bg_mask.mean() < 0.05 or bg_mask.all():
            return ABSTAIN

        face = _modal_color(arr[~bg_mask])
        if face is None or tuple(face) == tuple(bg):
            return ABSTAIN
        face_mask = np.all(arr == np.array(face, dtype=np.uint8), axis=-1)
        if face_mask.sum() < 400:
            return ABSTAIN

        ys, xs = np.nonzero(face_mask)
        y0, y1, x0, x1 = int(ys.min()), int(ys.max()), int(xs.min()), int(xs.max())
        h, w = y1 - y0 + 1, x1 - x0 + 1
        if min(h, w) < 40 or not 0.8 <= h / w <= 1.25:
            return ABSTAIN
        # The face is a solid rounded square: its own colour must fill most of
        # that bounding box, otherwise this is not a die face.
        if face_mask[y0:y1 + 1, x0:x1 + 1].mean() < 0.45:
            return ABSTAIN

        sub = arr[y0:y1 + 1, x0:x1 + 1]
        sub_other = sub[~(np.all(sub == np.array(face, dtype=np.uint8), axis=-1)
                          | np.all(sub == np.array(bg, dtype=np.uint8), axis=-1))]
        pip = _modal_color(sub_other)
        if pip is None or sub_other.shape[0] < 200:
            return ABSTAIN

        cy, cx = (y0 + y1) // 2, (x0 + x1) // 2
        pip_arr = np.array(pip, dtype=np.uint8)
        votes = []
        for dy in range(-VOTE_RADIUS, VOTE_RADIUS + 1):
            for dx in range(-VOTE_RADIUS, VOTE_RADIUS + 1):
                if dx * dx + dy * dy > VOTE_RADIUS * VOTE_RADIUS:
                    continue
                yy, xx = cy + dy, cx + dx
                if not (0 <= yy < arr.shape[0] and 0 <= xx < arr.shape[1]):
                    return ABSTAIN
                votes.append(bool(np.all(arr[yy, xx] == pip_arr)))
        frac = sum(votes) / len(votes)
        if frac >= 0.8:
            return "covered"
        if frac <= 0.2:
            return "empty"
        return ABSTAIN
    except Exception:
        return ABSTAIN
