"""Renderer for the unit-transfer count world.

A measurement sheet carries exactly three pure-black marks that live inside one
small inset placed at a random location on the sheet:

* a left black tick (a short vertical bar),
* a right black tick, and
* a black horizontal measuring bar, offset vertically so that white paper is
  always visible between it and the ticks.

The decision is how many whole copies of the measuring bar fit inside the span
between the two tick centres, i.e. ``span // unit``.  Every other mark on the
sheet (the light grey ruler, coloured ticks and coloured bars) is drawn with a
maximum RGB channel far above the black threshold and carries no information.

All lengths are integers in raster units and both arms measure exactly the same
integers, so the analytic decision and the pixel-only decision are the same
function of the same two visible quantities.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "unit_transfer_count-renderer-0.1.0"

# --- fixed drawing constants (not latent fields: they never vary per scene) ---
CAP_W = 3           # black tick width in px (odd, so the centre is an integer)
CAP_H = 22          # black tick height in px
BAR_H = 7           # measuring-bar thickness in px
GRID = 8            # ablation lattice used to place the two decisive groups
BAND_PAD = 4        # keep each group strictly inside its own lattice band
INSET_H = CAP_H

BLACK = (0, 0, 0)
PAPER = (255, 255, 255)
RULER = (198, 198, 198)
DECOR_COLORS = (
    (214, 62, 62),
    (46, 148, 78),
    (58, 96, 208),
    (226, 142, 34),
    (146, 78, 186),
    (32, 158, 162),
)

SPAN_MIN, SPAN_MAX = 34, 100
UNIT_MIN, UNIT_MAX = 13, 42
CLASSES = (2, 3, 4)
QUARANTINE_MARGIN_PX = 6.5
CLEARANCE = 24      # keep decoration this far from either decisive group


def _decision_parts(scene: dict) -> tuple[int, int, int]:
    """Return ``(span, unit, count)`` in exact raster units."""
    span = int(scene["span"])
    unit = int(scene["unit"])
    return span, unit, span // unit


def safe_band(length: int, index: int) -> tuple[int, int]:
    """Interior band of lattice cell ``index`` that no neighbouring cell reaches.

    The support probe ablates each cell of a fixed ``GRID`` lattice widened by a
    quarter-cell halo on every side.  A mark drawn strictly inside this band is
    therefore erased by its own cell and by no other, which is what keeps the two
    decisive groups separate components of the support.
    """
    start = length * index // GRID
    end = length * (index + 1) // GRID
    prev_end = length * index // GRID
    prev_start = length * (index - 1) // GRID
    next_start = length * (index + 1) // GRID
    next_end = length * (index + 2) // GRID
    left_halo = -(-(prev_end - prev_start) // 4)
    right_halo = -(-(next_end - next_start) // 4)
    return start + left_halo + BAND_PAD, end - right_halo - BAND_PAD


def _group_origin(rng: random.Random, width: int, height: int, cell: tuple[int, int],
                  glyph_w: int, glyph_h: int) -> tuple[int, int]:
    """Random top-left corner for a group of size ``glyph_w`` x ``glyph_h``."""
    x0, x1 = safe_band(width, cell[1])
    y0, y1 = safe_band(height, cell[0])
    return rng.randint(x0, max(x0, x1 - glyph_w)), rng.randint(y0, max(y0, y1 - glyph_h))


def sample_scene(seed: int) -> dict:
    """Sample one answerable scene deterministically from ``seed``."""
    rng = random.Random(seed)
    width, height = 1800, 900

    target = rng.choice(CLASSES)
    span = unit = 0
    for _ in range(600):
        cand_unit = rng.randint(UNIT_MIN, UNIT_MAX)
        frac = rng.uniform(0.10, 0.90)
        cand_span = int(round(cand_unit * (target + frac)))
        if not SPAN_MIN <= cand_span <= SPAN_MAX:
            continue
        if cand_span // cand_unit != target:
            continue
        low = cand_span - target * cand_unit
        high = (target + 1) * cand_unit - cand_span
        if min(low, high) < 4:
            continue
        span, unit = cand_span, cand_unit
        break
    if span == 0:
        # Deterministic fallback that always satisfies every constraint.
        unit = 18
        span = 18 * target + 9

    # The span group and the unit group live in two diagonally adjacent cells of
    # the fixed lattice, so the decisive support is two disconnected parts held
    # inside one small neighbourhood of the sheet.
    base_row = rng.randint(1, GRID - 2)
    base_col = rng.randint(1, GRID - 2)
    d_row = rng.choice((-1, 1))
    d_col = rng.choice((-1, 1))
    span_first = rng.random() < 0.5
    span_cell = (base_row, base_col) if span_first else (base_row + d_row, base_col + d_col)
    unit_cell = (base_row + d_row, base_col + d_col) if span_first else (base_row, base_col)

    span_x, span_y = _group_origin(rng, width, height, span_cell, span + CAP_W, CAP_H)
    unit_x, unit_y = _group_origin(rng, width, height, unit_cell, unit, BAR_H)

    ruler_y = rng.randint(26, 44)
    ruler_step = rng.randint(30, 46)
    ruler_phase = rng.randint(0, ruler_step - 1)

    reserved = [
        (span_x - CLEARANCE, span_y - CLEARANCE,
         span_x + span + CAP_W + CLEARANCE, span_y + CAP_H + CLEARANCE),
        (unit_x - CLEARANCE, unit_y - CLEARANCE,
         unit_x + unit + CLEARANCE, unit_y + BAR_H + CLEARANCE),
    ]
    distractors: list[dict] = []
    tries = 0
    while len(distractors) < 16 and tries < 700:
        tries += 1
        kind = rng.choice(("tickpair", "bar", "box"))
        color = rng.randrange(len(DECOR_COLORS))
        if kind == "tickpair":
            d_span = rng.randint(30, 110)
            d_unit = rng.randint(13, 44)
            w, h = d_span + CAP_W + 1, CAP_H
        elif kind == "bar":
            d_span, d_unit = 0, rng.randint(16, 70)
            w, h = d_unit, BAR_H
        else:
            d_span, d_unit = 0, rng.randint(13, 28)
            w = h = d_unit
        x = rng.randint(24, width - 24 - w)
        y = rng.randint(ruler_y + 26, height - 26 - h)
        if any(
            x + w >= box[0] and x <= box[2] and y + h >= box[1] and y <= box[3]
            for box in reserved
        ):
            continue
        distractors.append(
            {
                "kind": kind,
                "x": x,
                "y": y,
                "span": d_span,
                "unit": d_unit,
                "color": color,
                "below": bool(rng.random() < 0.5),
            }
        )

    return {
        "width": width,
        "height": height,
        "span": span,
        "unit": unit,
        "span_x": span_x,
        "span_y": span_y,
        "unit_x": unit_x,
        "unit_y": unit_y,
        "ruler_y": ruler_y,
        "ruler_step": ruler_step,
        "ruler_phase": ruler_phase,
        "distractors": distractors,
    }


def black_glyph_boxes(scene: dict) -> list[tuple[int, int, int, int]]:
    """Inclusive pixel boxes of the three decisive black marks."""
    span, unit, _ = _decision_parts(scene)
    tick_top = int(scene["span_y"])
    bar_top = int(scene["unit_y"])
    left_center = int(scene["span_x"]) + CAP_W // 2
    right_center = left_center + span
    half = CAP_W // 2
    bar_x0 = int(scene["unit_x"])
    return [
        (left_center - half, tick_top, left_center + half, tick_top + CAP_H - 1),
        (right_center - half, tick_top, right_center + half, tick_top + CAP_H - 1),
        (bar_x0, bar_top, bar_x0 + unit - 1, bar_top + BAR_H - 1),
    ]


def render(scene: dict) -> Image.Image:
    """Rasterise the scene.  No antialiasing is used anywhere."""
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width, height), PAPER)
    draw = ImageDraw.Draw(image)

    # Decorative grey ruler across the top of the sheet.
    ruler_y = int(scene["ruler_y"])
    draw.rectangle([24, ruler_y, width - 25, ruler_y + 1], fill=RULER)
    x = 24 + int(scene["ruler_phase"])
    while x < width - 24:
        draw.rectangle([x, ruler_y - 7, x + 1, ruler_y - 1], fill=RULER)
        x += int(scene["ruler_step"])

    # Coloured decoration: never black, never inside the reserved inset region.
    for item in scene["distractors"]:
        color = DECOR_COLORS[int(item["color"]) % len(DECOR_COLORS)]
        ix, iy = int(item["x"]), int(item["y"])
        if item["kind"] == "tickpair":
            d_span, d_unit = int(item["span"]), int(item["unit"])
            bar_top = iy + CAP_H - BAR_H if item["below"] else iy
            draw.rectangle([ix, iy, ix + CAP_W - 1, iy + CAP_H - 1], fill=color)
            draw.rectangle(
                [ix + d_span, iy, ix + d_span + CAP_W - 1, iy + CAP_H - 1], fill=color
            )
            draw.rectangle([ix + 6, bar_top, ix + 6 + d_unit - 1, bar_top + BAR_H - 1], fill=color)
        elif item["kind"] == "bar":
            draw.rectangle([ix, iy, ix + int(item["unit"]) - 1, iy + BAR_H - 1], fill=color)
        else:
            side = int(item["unit"])
            draw.rectangle([ix, iy, ix + side - 1, iy + side - 1], outline=color, width=2)

    # The three decisive black marks.
    for box in black_glyph_boxes(scene):
        draw.rectangle(list(box), fill=BLACK)
    return image


def analytic_gold(scene: dict) -> str:
    """Whole copies of the measuring bar that fit inside the span."""
    _, _, count = _decision_parts(scene)
    return str(count)


def margin(scene: dict) -> float:
    """Pixels to the nearest span length that would change the count."""
    span, unit, count = _decision_parts(scene)
    return float(min(span - count * unit, (count + 1) * unit - span))


def ratio_margin(scene: dict) -> float:
    """Same margin expressed in measuring-bar lengths."""
    _, unit, _ = _decision_parts(scene)
    return float(margin(scene) / unit)


def is_quarantined(scene: dict) -> bool:
    """Flag scenes whose span sits too close to an exact multiple of the unit."""
    return bool(margin(scene) < QUARANTINE_MARGIN_PX)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist: every sampled field is drawn (see candidate.json)."""
    return []


# --- helpers used only by the candidate tests -------------------------------

def translated(scene: dict, dx: int, dy: int) -> dict:
    """Label-preserving translation of both decisive groups."""
    moved = {key: value for key, value in scene.items()}
    moved["span_x"] = int(scene["span_x"]) + dx
    moved["span_y"] = int(scene["span_y"]) + dy
    moved["unit_x"] = int(scene["unit_x"]) + dx
    moved["unit_y"] = int(scene["unit_y"]) + dy
    return moved


def repalette(scene: dict, shift: int) -> dict:
    """Label-preserving permutation of the decoration palette."""
    moved = {key: value for key, value in scene.items()}
    moved["distractors"] = [
        {**item, "color": (int(item["color"]) + shift) % len(DECOR_COLORS)}
        for item in scene["distractors"]
    ]
    return moved


def permuted_distractors(scene: dict) -> dict:
    """Label-preserving reversal of the decoration draw order."""
    moved = {key: value for key, value in scene.items()}
    moved["distractors"] = list(reversed(scene["distractors"]))
    return moved
