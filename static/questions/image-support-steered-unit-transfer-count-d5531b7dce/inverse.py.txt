"""Pixel-only inverse arm for the unit-transfer count world.

Input is a single PIL image.  Nothing else is read: no latent scene, no
renderer, no prompts, no manifest, no file name.  The arm segments the pure
black ink, splits it into connected components, identifies the two tall narrow
ticks and the one wide flat measuring bar, and reports how many whole bar
lengths fit between the tick centres.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "unit_transfer_count-oracle-0.1.0"

BLACK_MAX_CHANNEL = 60      # a pixel is ink only if every channel is this dark
UNREADABLE = "unreadable"
ALLOWED = ("2", "3", "4")

TICK_MAX_W, TICK_MIN_H = 6, 15
BAR_MIN_W, BAR_MAX_H = 10, 12
MIN_COMPONENT_PX = 12
MAX_INK_PX = 4000          # far above the ~470 px of ink a valid sheet carries


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    """Return ``(x0, y0, x1, y1, area)`` for each 8-connected ink component."""
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out: list[tuple[int, int, int, int, int]] = []
    ys, xs = np.nonzero(mask)
    for sy, sx in zip(ys.tolist(), xs.tolist()):
        if seen[sy, sx]:
            continue
        stack = [(sy, sx)]
        seen[sy, sx] = True
        x0 = x1 = sx
        y0 = y1 = sy
        area = 0
        while stack:
            cy, cx = stack.pop()
            area += 1
            if cx < x0:
                x0 = cx
            if cx > x1:
                x1 = cx
            if cy < y0:
                y0 = cy
            if cy > y1:
                y1 = cy
            for ny in (cy - 1, cy, cy + 1):
                if ny < 0 or ny >= height:
                    continue
                for nx in (cx - 1, cx, cx + 1):
                    if nx < 0 or nx >= width:
                        continue
                    if mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        out.append((x0, y0, x1, y1, area))
        if len(out) > 24:
            break
    return out


def decision_from_image(image: Image.Image) -> str:
    """Recover the whole-copy count from the raster alone."""
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    mask = arr.max(axis=2) < BLACK_MAX_CHANNEL
    ink = int(mask.sum())
    if ink == 0 or ink > MAX_INK_PX:
        return UNREADABLE

    comps = [c for c in _components(mask) if c[4] >= MIN_COMPONENT_PX]
    if len(comps) != 3:
        return UNREADABLE

    ticks: list[tuple[int, int, int, int, int]] = []
    bars: list[tuple[int, int, int, int, int]] = []
    for comp in comps:
        x0, y0, x1, y1, _ = comp
        w, h = x1 - x0 + 1, y1 - y0 + 1
        if w <= TICK_MAX_W and h >= TICK_MIN_H:
            ticks.append(comp)
        elif w >= BAR_MIN_W and h <= BAR_MAX_H:
            bars.append(comp)
        else:
            return UNREADABLE
    if len(ticks) != 2 or len(bars) != 1:
        return UNREADABLE

    if abs((ticks[0][3] - ticks[0][1]) - (ticks[1][3] - ticks[1][1])) > 2:
        return UNREADABLE

    left, right = sorted(ticks, key=lambda c: c[0] + c[2])
    span = ((right[0] + right[2]) - (left[0] + left[2])) // 2
    unit = bars[0][2] - bars[0][0] + 1
    if unit <= 0 or span <= 0:
        return UNREADABLE

    count = span // unit
    answer = str(count)
    return answer if answer in ALLOWED else UNREADABLE
