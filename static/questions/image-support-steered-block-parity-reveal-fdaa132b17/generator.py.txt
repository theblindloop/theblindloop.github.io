"""Deterministic latent sampler and renderer for Block Parity Reveal."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
SOURCE_N = 14
OUTPUT_N = 7
PAD = 14
BACKGROUND = (248, 247, 242)
FRAME = (72, 78, 90)
PANEL = (166, 172, 174)
LIGHT = (218, 222, 214)
CYAN = (25, 190, 205)
GOLD = (235, 165, 35)
DARK_PALETTES = ((32, 52, 75), (92, 38, 60), (34, 82, 58))
LABELS = ("plus", "X", "H")


def _template(name: str) -> tuple[int, ...]:
    bits = []
    for row in range(OUTPUT_N):
        for col in range(OUTPUT_N):
            if name == "plus":
                value = row == 3 or col == 3
            elif name == "X":
                value = row == col or row + col == 6
            elif name == "H":
                value = ((col in (1, 5)) and 1 <= row <= 5) or (
                    row == 3 and 2 <= col <= 4
                )
            else:
                raise KeyError(name)
            bits.append(int(value))
    return tuple(bits)


TEMPLATES = {name: _template(name) for name in LABELS}


def _distances(bits: list[int] | tuple[int, ...]) -> dict[str, int]:
    values = tuple(int(value) for value in bits)
    if len(values) != OUTPUT_N * OUTPUT_N or sum(values) != 13:
        raise ValueError("parity_bits must contain 49 binary values with weight 13")
    return {
        name: sum(left != right for left, right in zip(values, template, strict=True))
        for name, template in TEMPLATES.items()
    }


def _classify(bits: list[int] | tuple[int, ...]) -> tuple[str, int]:
    distances = _distances(bits)
    ordered = sorted((distance, name) for name, distance in distances.items())
    margin_value = ordered[1][0] - ordered[0][0]
    decision = "ambiguous" if margin_value == 0 else ordered[0][1]
    return decision, margin_value


def _noisy_pattern(target: str, swaps: int, rng: random.Random) -> list[int]:
    original = list(TEMPLATES[target])
    for _ in range(2000):
        bits = original.copy()
        ones = [index for index, value in enumerate(bits) if value]
        zeros = [index for index, value in enumerate(bits) if not value]
        for index in rng.sample(ones, swaps):
            bits[index] = 0
        for index in rng.sample(zeros, swaps):
            bits[index] = 1
        if _classify(bits)[0] == target:
            return bits
    raise RuntimeError("could not sample a uniquely classified fixed-weight pattern")


def _tie_pattern(left: str, right: str, rng: random.Random) -> list[int]:
    a = set(index for index, value in enumerate(TEMPLATES[left]) if value)
    b = set(index for index, value in enumerate(TEMPLATES[right]) if value)
    a_only = sorted(a - b)
    b_only = sorted(b - a)
    half = len(a_only) // 2
    for _ in range(2000):
        rng.shuffle(a_only)
        rng.shuffle(b_only)
        chosen = (a & b) | set(a_only[:half]) | set(b_only[:half])
        bits = [int(index in chosen) for index in range(OUTPUT_N * OUTPUT_N)]
        if _classify(bits)[0] == "ambiguous":
            return bits
    raise RuntimeError("could not construct a fixed-weight template tie")


def sample_scene(seed: int) -> dict:
    """Sample balanced classes and nuisance variables from one integer seed."""
    rng = random.Random(int(seed))
    target = LABELS[int(seed) % len(LABELS)]
    difficulty = (int(seed) // len(LABELS)) % 6
    if difficulty == 0:
        other = LABELS[(LABELS.index(target) + 1) % len(LABELS)]
        parity_bits = _tie_pattern(target, other, rng)
    else:
        parity_bits = _noisy_pattern(target, min(4, difficulty - 1), rng)
    cell_size = (12, 13, 14)[rng.randrange(3)]
    side = SOURCE_N * cell_size + 2 * PAD
    origin_x = (WIDTH - side) // 2 + rng.randint(-18, 18)
    origin_y = (HEIGHT - side) // 2 + rng.randint(-18, 18)
    return {
        "parity_bits": parity_bits,
        "carrier_seed": rng.randrange(1_000_000_000),
        "orientation": rng.randrange(4),
        "cell_size": cell_size,
        "origin_x": origin_x,
        "origin_y": origin_y,
        "palette": rng.randrange(len(DARK_PALETTES)),
        "distractor_seed": rng.randrange(1_000_000_000),
    }


def _carrier_cells(scene: dict) -> list[list[int]]:
    bits = [int(value) for value in scene["parity_bits"]]
    rng = random.Random(int(scene["carrier_seed"]))
    odd_indices = [index for index, value in enumerate(bits) if value]
    if len(odd_indices) != 13:
        raise ValueError("the parity field must have fixed weight 13")
    rng.shuffle(odd_indices)
    use_three = set(odd_indices[:7])
    cells = [[0 for _ in range(SOURCE_N)] for _ in range(SOURCE_N)]
    for out_row in range(OUTPUT_N):
        for out_col in range(OUTPUT_N):
            index = out_row * OUTPUT_N + out_col
            count = 3 if index in use_three else (1 if bits[index] else 2)
            positions = rng.sample(range(4), count)
            for position in positions:
                row = out_row * 2 + position // 2
                col = out_col * 2 + position % 2
                cells[row][col] = 1
    if sum(map(sum, cells)) != 99:
        raise AssertionError("carrier ink count must be exactly 99")
    return cells


def _draw_patch(scene: dict) -> Image.Image:
    cell = int(scene["cell_size"])
    side = SOURCE_N * cell + 2 * PAD
    patch = Image.new("RGB", (side, side), PANEL)
    draw = ImageDraw.Draw(patch)
    draw.rectangle((0, 0, side - 1, side - 1), outline=FRAME, width=4)
    draw.rectangle((5, 5, 10, 10), fill=CYAN)
    draw.rectangle((side - 11, 5, side - 6, 10), fill=GOLD)
    dark = DARK_PALETTES[int(scene["palette"])]
    cells = _carrier_cells(scene)
    for row in range(SOURCE_N):
        for col in range(SOURCE_N):
            x0 = PAD + col * cell + 1
            y0 = PAD + row * cell + 1
            x1 = PAD + (col + 1) * cell - 2
            y1 = PAD + (row + 1) * cell - 2
            draw.rectangle((x0, y0, x1, y1), fill=dark if cells[row][col] else LIGHT)
    turns = int(scene["orientation"]) % 4
    for _ in range(turns):
        patch = patch.transpose(Image.Transpose.ROTATE_90)
    return patch


def render(scene: dict) -> Image.Image:
    """Render only from declared scene fields; no metadata channel is written."""
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    patch = _draw_patch(scene)
    origin = (int(scene["origin_x"]), int(scene["origin_y"]))
    image.paste(patch, origin)
    draw = ImageDraw.Draw(image)
    rng = random.Random(int(scene["distractor_seed"]))
    x0, y0 = origin
    x1, y1 = x0 + patch.width, y0 + patch.height
    colors = ((180, 145, 155), (145, 165, 190), (175, 170, 130))
    placed = 0
    while placed < 7:
        x = rng.randint(28, WIDTH - 28)
        y = rng.randint(28, HEIGHT - 28)
        if x0 - 24 <= x <= x1 + 24 and y0 - 24 <= y <= y1 + 24:
            continue
        radius = rng.randint(3, 6)
        color = colors[rng.randrange(len(colors))]
        if rng.randrange(2):
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)
        else:
            draw.rectangle((x - radius, y - radius, x + radius, y + radius), fill=color)
        placed += 1
    return image


def analytic_gold(scene: dict) -> str:
    return _classify(scene["parity_bits"])[0]


def margin(scene: dict) -> float:
    return float(_classify(scene["parity_bits"])[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
