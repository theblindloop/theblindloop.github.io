"""Independent pixel-only inverse for Block Parity Reveal.

The implementation locates the fixed-color outer frame, uses the cyan-to-gold
corner ordering to rectify the patch, reads all 196 mini-squares, performs the
declared local parity pooling, and classifies the resulting 7 by 7 bitmap.  It
does not import or receive renderer state.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

FRAME_RGB = np.array((72, 78, 90), dtype=np.int16)
SOURCE_N = 14
OUTPUT_N = 7
PAD = 14


class OracleError(ValueError):
    pass


def _literal_templates() -> dict[str, tuple[int, ...]]:
    plus = tuple(int(row == 3 or col == 3) for row in range(7) for col in range(7))
    cross = tuple(int(row == col or row + col == 6) for row in range(7) for col in range(7))
    letter_h = tuple(
        int(((col in (1, 5)) and 1 <= row <= 5) or (row == 3 and 2 <= col <= 4))
        for row in range(7)
        for col in range(7)
    )
    return {"plus": plus, "X": cross, "H": letter_h}


TEMPLATES = _literal_templates()


def _rectified_patch(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    signed = array.astype(np.int16)
    frame = np.max(np.abs(signed - FRAME_RGB), axis=2) <= 1
    ys, xs = np.nonzero(frame)
    if len(xs) < 500:
        raise OracleError("complete marked frame not found")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    side_x, side_y = x1 - x0 + 1, y1 - y0 + 1
    if side_x != side_y or not 190 <= side_x <= 225:
        raise OracleError("marked frame is not a supported square")
    crop = array[y0 : y1 + 1, x0 : x1 + 1]
    candidates = []
    for turns in range(4):
        trial = np.rot90(crop, turns)
        cyan = (trial[:, :, 0] < 60) & (trial[:, :, 1] > 150) & (trial[:, :, 2] > 160)
        gold = (trial[:, :, 0] > 190) & (trial[:, :, 1] > 120) & (trial[:, :, 2] < 80)
        cy, cx = np.nonzero(cyan)
        gy, gx = np.nonzero(gold)
        if len(cx) < 20 or len(gx) < 20:
            continue
        side = trial.shape[0]
        if cx.mean() < side / 2 and cy.mean() < side / 2 and gx.mean() > side / 2 and gy.mean() < side / 2:
            candidates.append(trial.copy())
    if len(candidates) != 1:
        raise OracleError("cyan-to-gold orientation fiducials are missing or ambiguous")
    return candidates[0]


def _read_source(patch: np.ndarray) -> np.ndarray:
    side = patch.shape[0]
    usable = side - 2 * PAD
    if usable <= 0 or usable % SOURCE_N:
        raise OracleError("frame scale is incompatible with the declared 14 by 14 grid")
    cell = usable // SOURCE_N
    if cell not in (12, 13, 14):
        raise OracleError("unsupported mini-square scale")
    source = np.zeros((SOURCE_N, SOURCE_N), dtype=np.uint8)
    for row in range(SOURCE_N):
        for col in range(SOURCE_N):
            cx = PAD + col * cell + cell // 2
            cy = PAD + row * cell + cell // 2
            sample = patch[cy - 2 : cy + 3, cx - 2 : cx + 3].astype(np.float32)
            if sample.shape != (5, 5, 3):
                raise OracleError("mini-square sampling escaped the frame")
            channel_std = float(sample.reshape(-1, 3).std(axis=0).max())
            mean = sample.mean(axis=(0, 1))
            luminance = float(0.2126 * mean[0] + 0.7152 * mean[1] + 0.0722 * mean[2])
            if channel_std > 3.0 or luminance > 235.0 or 120.0 < luminance < 185.0:
                raise OracleError("a required mini-square is erased or corrupted")
            source[row, col] = int(luminance < 120.0)
    if int(source.sum()) != 99:
        raise OracleError("mosaic does not have the declared fixed-ink carrier")
    return source


def _pool(source: np.ndarray) -> tuple[int, ...]:
    values = []
    for row in range(OUTPUT_N):
        for col in range(OUTPUT_N):
            block = source[2 * row : 2 * row + 2, 2 * col : 2 * col + 2]
            values.append(int(int(block.sum()) % 2 == 1))
    if sum(values) != 13:
        raise OracleError("parity reduction violates the declared equal-ink output family")
    return tuple(values)


def _classify(bits: tuple[int, ...]) -> tuple[str, float, dict[str, int]]:
    distances = {
        name: sum(left != right for left, right in zip(bits, template, strict=True))
        for name, template in TEMPLATES.items()
    }
    ordered = sorted((distance, name) for name, distance in distances.items())
    gap = float(ordered[1][0] - ordered[0][0])
    return ("ambiguous" if gap == 0 else ordered[0][1]), gap, distances


def measure(image: Image.Image) -> dict:
    patch = _rectified_patch(image)
    source = _read_source(patch)
    bits = _pool(source)
    decision, gap, distances = _classify(bits)
    return {
        "decision": decision,
        "margin": gap,
        "parity_bits": list(bits),
        "distances": distances,
    }


def decision_from_image(image: Image.Image) -> str:
    return str(measure(image)["decision"])
