"""Pixel-only inverse arm for the clasp-cut pocket-survival world.

Receives one RGB image and nothing else: no scene, no palette table, no band
geometry, no lattice pitch, no analytic gold.  It segments the thread by ink
class, checks that the thread it sees is one intact trace, deletes the red
stretch from the barrier set, and reports whether any background pocket is
still enclosed by the remaining black thread.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

UNRESOLVED = "unresolved"
MIN_POCKET_AREA = 12


def _masks(array: np.ndarray):
    red = array[:, :, 0].astype(np.int16)
    green = array[:, :, 1].astype(np.int16)
    blue = array[:, :, 2].astype(np.int16)
    dark = (red < 70) & (green < 70) & (blue < 70)
    crimson = (red > 150) & (green < 90) & (blue < 90)
    return dark, crimson


def _label_count(mask: np.ndarray) -> int:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    count = 0
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        count += 1
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        while queue:
            cy, cx = queue.popleft()
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
    return count


def _crop_to(mask: np.ndarray, reference: np.ndarray, pad: int = 3):
    """Crop a barrier mask to the padded bounding box of the reference ink."""
    points = np.argwhere(reference)
    y0 = max(0, int(points[:, 0].min()) - pad)
    y1 = min(mask.shape[0], int(points[:, 0].max()) + pad + 1)
    x0 = max(0, int(points[:, 1].min()) - pad)
    x1 = min(mask.shape[1], int(points[:, 1].max()) + pad + 1)
    return mask[y0:y1, x0:x1]


def _enclosed_pockets(barrier: np.ndarray) -> int:
    """Count background regions that the barrier separates from the border."""
    height, width = barrier.shape
    free = ~barrier
    reached = np.zeros_like(free, dtype=bool)
    queue: deque = deque()
    for x in range(width):
        for y in (0, height - 1):
            if free[y, x] and not reached[y, x]:
                reached[y, x] = True
                queue.append((y, x))
    for y in range(height):
        for x in (0, width - 1):
            if free[y, x] and not reached[y, x]:
                reached[y, x] = True
                queue.append((y, x))
    while queue:
        cy, cx = queue.popleft()
        for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
            if 0 <= ny < height and 0 <= nx < width and free[ny, nx] and not reached[ny, nx]:
                reached[ny, nx] = True
                queue.append((ny, nx))
    trapped = free & ~reached
    if not trapped.any():
        return 0
    pockets = 0
    seen = np.zeros_like(trapped, dtype=bool)
    for y, x in np.argwhere(trapped):
        if seen[y, x]:
            continue
        area = 0
        stack = [(int(y), int(x))]
        seen[y, x] = True
        while stack:
            cy, cx = stack.pop()
            area += 1
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if (
                    0 <= ny < trapped.shape[0]
                    and 0 <= nx < trapped.shape[1]
                    and trapped[ny, nx]
                    and not seen[ny, nx]
                ):
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if area >= MIN_POCKET_AREA:
            pockets += 1
    return pockets


def decision_from_image(image: "Image.Image") -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    dark, crimson = _masks(array)
    thread = dark | crimson
    if not thread.any() or not crimson.any():
        return UNRESOLVED
    if (
        thread[0, :].any()
        or thread[-1, :].any()
        or thread[:, 0].any()
        or thread[:, -1].any()
    ):
        return UNRESOLVED
    if _label_count(thread) != 1 or _label_count(crimson) != 1:
        return UNRESOLVED
    if not dark.any():
        return UNRESOLVED
    local = _crop_to(dark, thread)
    return "survives" if _enclosed_pockets(local) >= 1 else "vanishes"
