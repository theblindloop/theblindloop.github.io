"""Renderer for the clasp-cut pocket-survival question world.

A single thin black thread is drawn as a rectilinear trace on a small lattice
confined to one narrow border band of the page.  The trace always contains
exactly one closed loop, so exactly one white pocket is enclosed before any
cut.  One short stretch of the thread is repainted red: the clasp.  The
decision is whether an enclosed pocket still exists once the red stretch is
taken out of the thread, i.e. whether the red stretch lies on the loop itself
or on one of the dangling branches.

Only geometry that is drawn into the raster is sampled; nothing is gold-only.
"""

from __future__ import annotations

import random

import numpy as np
from PIL import Image

CANVAS = 1152
THREAD_INK = (16, 16, 16)
CLASP_INK = (210, 31, 31)
BACKGROUND = (255, 255, 255)
DECOR_INKS = (
    (47, 95, 191),
    (46, 139, 87),
    (154, 154, 154),
    (123, 79, 168),
    (222, 140, 30),
)

COLS = 3          # lattice node columns across the band
ROWS = 7          # lattice node rows along the band
SIDES = ("left", "right", "top", "bottom")
STROKES = (4, 5, 6)
PITCHES = (34, 36, 38)
BAND_INSET = 8    # local offset of the first lattice node from the page border
QUARANTINE_MARGIN = 5.0
DECISIONS = ("survives", "vanishes")


# --------------------------------------------------------------------------
# lattice helpers
# --------------------------------------------------------------------------
def _edge_nodes(edge):
    kind, i, j = edge
    if kind == "h":
        return ((i, j), (i + 1, j))
    return ((i, j), (i, j + 1))


def _incident(node):
    i, j = node
    out = []
    if i + 1 < COLS:
        out.append(("h", i, j))
    if i - 1 >= 0:
        out.append(("h", i - 1, j))
    if j + 1 < ROWS:
        out.append(("v", i, j))
    if j - 1 >= 0:
        out.append(("v", i, j - 1))
    return out


def _other(edge, node):
    first, second = _edge_nodes(edge)
    return second if first == node else first


def _loop_edges(block):
    i0, j0, width, height = block
    edges = []
    for i in range(i0, i0 + width):
        edges.append(("h", i, j0))
        edges.append(("h", i, j0 + height))
    for j in range(j0, j0 + height):
        edges.append(("v", i0, j))
        edges.append(("v", i0 + width, j))
    return edges


def _betti_one(edges):
    """Independent-cycle count of an edge set on the lattice."""
    edges = [tuple(edge) for edge in edges]
    nodes = set()
    for edge in edges:
        nodes.update(_edge_nodes(edge))
    parent = {node: node for node in nodes}

    def find(node):
        while parent[node] != node:
            parent[node] = parent[parent[node]]
            node = parent[node]
        return node

    for edge in edges:
        first, second = _edge_nodes(edge)
        ra, rb = find(first), find(second)
        if ra != rb:
            parent[ra] = rb
    components = len({find(node) for node in nodes})
    return len(edges) - len(nodes) + components


def _components(edges):
    edges = [tuple(edge) for edge in edges]
    if not edges:
        return 0
    nodes = set()
    for edge in edges:
        nodes.update(_edge_nodes(edge))
    parent = {node: node for node in nodes}

    def find(node):
        while parent[node] != node:
            parent[node] = parent[parent[node]]
            node = parent[node]
        return node

    for edge in edges:
        first, second = _edge_nodes(edge)
        ra, rb = find(first), find(second)
        if ra != rb:
            parent[ra] = rb
    return len({find(node) for node in nodes})


def _inside_block(edge, block):
    """True when both endpoints of the edge lie in the closed face block."""
    i0, j0, width, height = block
    for i, j in _edge_nodes(edge):
        if not (i0 <= i <= i0 + width and j0 <= j <= j0 + height):
            return False
    return True


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------
def sample_scene(seed: int) -> dict:
    rng = random.Random(7919 * int(seed) + 13)
    wanted = DECISIONS[int(seed) % 2]
    stroke = STROKES[(int(seed) // 2) % len(STROKES)]

    for _ in range(400):
        pitch = rng.choice(PITCHES)
        width = rng.choice((1, 2))
        height = rng.choice((1, 2, 3))
        i0 = rng.randrange(0, COLS - width)
        j0 = rng.randrange(0, ROWS - height)
        block = (i0, j0, width, height)
        loop = _loop_edges(block)

        edge_set = list(loop)
        seen = set(edge_set)
        used = set()
        for edge in loop:
            used.update(_edge_nodes(edge))
        tail = []
        for _ in range(rng.randint(3, 5)):
            current = rng.choice(sorted(used))
            for _ in range(rng.randint(1, 3)):
                options = [
                    candidate
                    for candidate in _incident(current)
                    if candidate not in seen and _other(candidate, current) not in used
                ]
                if not options:
                    break
                chosen = rng.choice(options)
                seen.add(chosen)
                edge_set.append(chosen)
                tail.append(chosen)
                current = _other(chosen, current)
                used.add(current)
        if not tail:
            continue
        rows = [node[1] for edge in edge_set for node in _edge_nodes(tuple(edge))]
        if max(rows) - min(rows) < 5:
            continue
        if _betti_one(edge_set) != 1 or _components(edge_set) != 1:
            continue

        if wanted == "vanishes":
            pool = list(loop)
        else:
            interior = [edge for edge in tail if _inside_block(edge, block)]
            pool = interior if (interior and rng.random() < 0.6) else tail
        clasp = rng.choice(sorted(pool))

        surviving = [edge for edge in edge_set if tuple(edge) != tuple(clasp)]
        if (_betti_one(surviving) >= 1) != (wanted == "survives"):
            continue

        side = rng.choice(SIDES)
        span = pitch * (ROWS - 1) + stroke
        offset = rng.randrange(130, CANVAS - 170 - span)
        scene = {
            "canvas": CANVAS,
            "side": side,
            "offset": int(offset),
            "pitch": int(pitch),
            "stroke": int(stroke),
            "block": [int(value) for value in block],
            "edges": sorted([list(edge) for edge in edge_set]),
            "clasp": list(clasp),
            "decor": _sample_decor(rng, side, offset, span, pitch),
        }
        return scene
    raise RuntimeError("thread sampling failed")


def _band_bbox(side, offset, span, pitch):
    """Global bounding box of the whole decisive band, generously padded."""
    depth = BAND_INSET + pitch * (COLS - 1) + 8
    pad = 44
    if side == "left":
        box = (0, offset, depth, offset + span)
    elif side == "right":
        box = (CANVAS - depth, offset, CANVAS, offset + span)
    elif side == "top":
        box = (offset, 0, offset + span, depth)
    else:
        box = (offset, CANVAS - depth, offset + span, CANVAS)
    return (box[0] - pad, box[1] - pad, box[2] + pad, box[3] + pad)


def _sample_decor(rng, side, offset, span, pitch):
    forbidden = _band_bbox(side, offset, span, pitch)
    placed = []
    items = []
    attempts = 0
    wanted = rng.randint(5, 8)
    while len(items) < wanted and attempts < 400:
        attempts += 1
        kind = rng.choice(("ring", "ring", "bar", "chain"))
        thickness = rng.choice((4, 5, 6, 7))
        w = rng.randrange(60, 210)
        h = rng.randrange(60, 210)
        x = rng.randrange(20, CANVAS - w - 20)
        y = rng.randrange(20, CANVAS - h - 20)
        box = (x, y, x + w, y + h)
        if _overlaps(box, forbidden, 0):
            continue
        if any(_overlaps(box, other, 26) for other in placed):
            continue
        placed.append(box)
        items.append(
            {
                "kind": kind,
                "box": [int(value) for value in box],
                "thickness": int(thickness),
                "ink": int(rng.randrange(len(DECOR_INKS))),
            }
        )
    return items


def _overlaps(first, second, pad):
    return not (
        first[2] + pad <= second[0]
        or second[2] + pad <= first[0]
        or first[3] + pad <= second[1]
        or second[3] + pad <= first[1]
    )


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def _to_global(side, offset, u0, v0, u1, v1):
    """Map a half-open local band rectangle to global half-open pixel bounds."""
    if side == "left":
        return (u0, offset + v0, u1, offset + v1)
    if side == "right":
        return (CANVAS - u1, offset + v0, CANVAS - u0, offset + v1)
    if side == "top":
        return (offset + v0, u0, offset + v1, u1)
    return (offset + v0, CANVAS - u1, offset + v1, CANVAS - u0)


def _edge_rects(edge, pitch, stroke):
    """Local half-open (u0, v0, u1, v1) rectangles: whole edge, red stretch."""
    kind, i, j = edge
    ui = BAND_INSET + pitch * i
    vj = BAND_INSET + pitch * j
    if kind == "h":
        whole = (ui, vj, ui + pitch + stroke, vj + stroke)
        inner = (ui + stroke, vj, ui + pitch, vj + stroke)
    else:
        whole = (ui, vj, ui + stroke, vj + pitch + stroke)
        inner = (ui, vj + stroke, ui + stroke, vj + pitch)
    return whole, inner


def _fill(array, box, colour):
    x0, y0, x1, y1 = (int(value) for value in box)
    x0 = max(0, x0)
    y0 = max(0, y0)
    x1 = min(array.shape[1], x1)
    y1 = min(array.shape[0], y1)
    if x1 > x0 and y1 > y0:
        array[y0:y1, x0:x1] = colour


def render(scene: dict) -> Image.Image:
    canvas = int(scene["canvas"])
    array = np.zeros((canvas, canvas, 3), dtype=np.uint8)
    array[:, :] = BACKGROUND
    side = scene["side"]
    offset = int(scene["offset"])
    pitch = int(scene["pitch"])
    stroke = int(scene["stroke"])

    for item in sorted(scene["decor"], key=lambda entry: tuple(entry["box"])):
        x0, y0, x1, y1 = (int(value) for value in item["box"])
        thickness = int(item["thickness"])
        colour = DECOR_INKS[int(item["ink"]) % len(DECOR_INKS)]
        if item["kind"] == "ring":
            _fill(array, (x0, y0, x1, y0 + thickness), colour)
            _fill(array, (x0, y1 - thickness, x1, y1), colour)
            _fill(array, (x0, y0, x0 + thickness, y1), colour)
            _fill(array, (x1 - thickness, y0, x1, y1), colour)
        elif item["kind"] == "bar":
            middle = (y0 + y1) // 2
            _fill(array, (x0, middle, x1, middle + thickness), colour)
        else:
            middle = (x0 + x1) // 2
            _fill(array, (x0, y0, x1, y0 + thickness), colour)
            _fill(array, (middle, y0, middle + thickness, y1), colour)
            _fill(array, (x0, y1 - thickness, middle + thickness, y1), colour)

    for edge in sorted(tuple(item) for item in scene["edges"]):
        whole, _ = _edge_rects(edge, pitch, stroke)
        _fill(array, _to_global(side, offset, *whole), THREAD_INK)

    _, inner = _edge_rects(tuple(scene["clasp"]), pitch, stroke)
    _fill(array, _to_global(side, offset, *inner), CLASP_INK)
    return Image.fromarray(array, mode="RGB")


# --------------------------------------------------------------------------
# analytic consequences of the scene
# --------------------------------------------------------------------------
def analytic_gold(scene: dict) -> str:
    clasp = tuple(scene["clasp"])
    surviving = [tuple(edge) for edge in scene["edges"] if tuple(edge) != clasp]
    return "survives" if _betti_one(surviving) >= 1 else "vanishes"


def margin(scene: dict) -> float:
    """Weakest decisive width in pixels."""
    stroke = int(scene["stroke"])
    pitch = int(scene["pitch"])
    return float(min(stroke, pitch - stroke))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list:
    """Declared pixel-identical, gold-preserving latent relabelings."""
    reversed_edges = dict(scene)
    reversed_edges["edges"] = list(reversed(list(scene["edges"])))
    rotated_decor = dict(scene)
    decor = list(scene["decor"])
    rotated_decor["decor"] = decor[1:] + decor[:1]
    return [
        ("thread_edge_order_reversal", reversed_edges),
        ("decor_order_rotation", rotated_decor),
    ]
