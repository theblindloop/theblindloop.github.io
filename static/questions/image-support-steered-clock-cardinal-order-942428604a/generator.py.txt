"""Deterministic renderer and analytic arm for clock_cardinal_order."""

from __future__ import annotations

import copy
import math
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

CANONICAL = ("12", "3", "6", "9")
COUNTERFACTUALS = (
    ("3", "12", "6", "9"),
    ("12", "6", "3", "9"),
    ("9", "3", "6", "12"),
    ("12", "3", "9", "6"),
    ("3", "6", "9", "12"),
    ("9", "6", "3", "12"),
)

_FONT_PATHS = (
    Path("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"),
    Path("/usr/share/fonts/truetype/liberation2/LiberationSans-Bold.ttf"),
)


def _font(size: int) -> ImageFont.FreeTypeFont:
    for path in _FONT_PATHS:
        if path.is_file():
            return ImageFont.truetype(str(path), size=size)
    raise RuntimeError("a supported bundled bold sans font is required")


def sample_scene(seed: int) -> dict:
    """Sample matched conventional/counterfactual scenes with label-neutral style."""

    rng = random.Random(int(seed) ^ 0xC10C_FACE)
    is_control = int(seed) % 2 == 0
    labels = CANONICAL if is_control else COUNTERFACTUALS[(int(seed) // 2) % len(COUNTERFACTUALS)]
    face_palette = ((250, 248, 240), (247, 250, 252), (252, 247, 244))
    rim_palette = ((48, 120, 176), (53, 139, 113), (129, 86, 170), (197, 92, 78))
    tick_palette = ((139, 145, 151), (145, 139, 151), (134, 148, 145))
    tick_order = list(range(12))
    rng.shuffle(tick_order)
    return {
        "canvas_size": [512, 512],
        "center": [256 + rng.randint(-10, 10), 256 + rng.randint(-10, 10)],
        "radius": rng.randint(158, 168),
        "face_color": list(rng.choice(face_palette)),
        "rim_color": list(rng.choice(rim_palette)),
        "tick_color": list(rng.choice(tick_palette)),
        "label_color": [24, 27, 31],
        "labels_clockwise": list(labels),
        "numeral_font_size": rng.randint(44, 48),
        "tick_draw_order": tick_order,
    }


def _text_center(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font) -> None:
    box = draw.textbbox((0, 0), text, font=font, stroke_width=0)
    width = box[2] - box[0]
    height = box[3] - box[1]
    draw.text(
        (xy[0] - width / 2.0 - box[0], xy[1] - height / 2.0 - box[1]),
        text,
        fill=(24, 27, 31),
        font=font,
    )


def render(scene: dict) -> Image.Image:
    """Render one antialiased RGB clock face from public scene fields."""

    scale = 3
    width, height = (int(value) for value in scene["canvas_size"])
    cx, cy = (float(value) for value in scene["center"])
    radius = float(scene["radius"])
    face_color = tuple(int(value) for value in scene["face_color"])
    rim_color = tuple(int(value) for value in scene["rim_color"])
    tick_color = tuple(int(value) for value in scene["tick_color"])
    label_color = tuple(int(value) for value in scene["label_color"])
    labels = tuple(str(value) for value in scene["labels_clockwise"])
    font_size = int(scene["numeral_font_size"])
    tick_order = [int(value) for value in scene["tick_draw_order"]]

    image = Image.new("RGB", (width * scale, height * scale), (232, 238, 242))
    draw = ImageDraw.Draw(image)

    def point(value: float) -> int:
        return int(round(value * scale))

    shadow = [point(cx - radius + 5), point(cy - radius + 7), point(cx + radius + 5), point(cy + radius + 7)]
    draw.ellipse(shadow, fill=(196, 204, 210))
    outer = [point(cx - radius), point(cy - radius), point(cx + radius), point(cy + radius)]
    draw.ellipse(outer, fill=rim_color)
    inner_r = radius - 7
    inner = [point(cx - inner_r), point(cy - inner_r), point(cx + inner_r), point(cy + inner_r)]
    draw.ellipse(inner, fill=face_color)

    # Twelve pairwise-disjoint ticks; their explicit draw order is a real latent alias.
    for index in tick_order:
        angle = math.radians(index * 30.0 - 90.0)
        length = 13 if index % 3 == 0 else 8
        outer_tick = radius - 14
        inner_tick = outer_tick - length
        x0 = point(cx + inner_tick * math.cos(angle))
        y0 = point(cy + inner_tick * math.sin(angle))
        x1 = point(cx + outer_tick * math.cos(angle))
        y1 = point(cy + outer_tick * math.sin(angle))
        draw.line((x0, y0, x1, y1), fill=tick_color, width=point(3), joint="curve")

    font = _font(font_size * scale)
    # Keep the four decisive glyph groups disconnected even under the public
    # overlapping 8x8 ablation grid, while their union remains a central region.
    label_radius = radius * 0.72
    anchors = (
        (cx, cy - label_radius),
        (cx + label_radius, cy),
        (cx, cy + label_radius),
        (cx - label_radius, cy),
    )
    for anchor, label in zip(anchors, labels, strict=True):
        box = draw.textbbox((0, 0), label, font=font)
        text_w = box[2] - box[0]
        text_h = box[3] - box[1]
        draw.text(
            (point(anchor[0]) - text_w / 2.0 - box[0], point(anchor[1]) - text_h / 2.0 - box[1]),
            label,
            fill=label_color,
            font=font,
        )

    # A colored center cap makes the object read as a clock without adding dark oracle ink.
    hub_r = 5 * scale
    draw.ellipse(
        (point(cx) - hub_r, point(cy) - hub_r, point(cx) + hub_r, point(cy) + hub_r),
        fill=rim_color,
    )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "yes" if tuple(scene["labels_clockwise"]) == CANONICAL else "no"


def margin(scene: dict) -> float:
    # Legibility and regional separation headroom above explicit quarantine floors.
    return float(min(int(scene["numeral_font_size"]) - 36, int(scene["radius"]) - 118))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["tick_draw_order"] = list(reversed(twin["tick_draw_order"]))
    return [("reverse_disjoint_tick_draw_order", twin)]
