"""Independent image-only inverse for the spiral-track raster."""

from __future__ import annotations

import numpy as np
from PIL import Image

N = 7
GUIDE = np.array((16, 126, 143), dtype=np.int32)

_UP = (
    "0001000",
    "0011100",
    "0111110",
    "1111111",
    "0001000",
    "0001000",
    "0001000",
)


def _rotate_clockwise(rows: tuple[str, ...]) -> tuple[str, ...]:
    return tuple("".join(rows[N - 1 - r][c] for r in range(N)) for c in range(N))


def _templates() -> dict[str, tuple[str, ...]]:
    names = ("up", "right", "down", "left")
    rows = _UP
    out = {}
    for name in names:
        out[name] = rows
        rows = _rotate_clockwise(rows)
    return out


TEMPLATES = _templates()


def _spiral_positions() -> tuple[tuple[int, int], ...]:
    positions = []
    top, bottom, left, right = 0, N - 1, 0, N - 1
    while top <= bottom and left <= right:
        positions.extend((top, col) for col in range(left, right + 1))
        top += 1
        positions.extend((row, right) for row in range(top, bottom + 1))
        right -= 1
        if top <= bottom:
            positions.extend((bottom, col) for col in range(right, left - 1, -1))
            bottom -= 1
        if left <= right:
            positions.extend((row, left) for row in range(bottom, top - 1, -1))
            left += 1
    return tuple(positions)


SPIRAL = _spiral_positions()


def _groups(indices: np.ndarray) -> list[float]:
    if len(indices) == 0:
        return []
    chunks = np.split(indices, np.where(np.diff(indices) > 1)[0] + 1)
    return [float(chunk.mean()) for chunk in chunks if len(chunk)]


def _grid_lines(mask: np.ndarray) -> tuple[list[float], list[float]]:
    ys, xs = np.where(mask)
    if len(xs) < 500:
        raise ValueError("cyan traversal and frame are missing")
    xmin, xmax, ymin, ymax = xs.min(), xs.max(), ys.min(), ys.max()
    crop = mask[ymin : ymax + 1, xmin : xmax + 1]
    x_candidates = _groups(np.where(crop.sum(axis=0) >= 0.96 * crop.shape[0])[0])
    y_candidates = _groups(np.where(crop.sum(axis=1) >= 0.96 * crop.shape[1])[0])
    if len(x_candidates) != N + 1 or len(y_candidates) != N + 1:
        raise ValueError(f"could not recover eight grid rails: {len(x_candidates)}/{len(y_candidates)}")
    return [xmin + value for value in x_candidates], [ymin + value for value in y_candidates]


def _recover_cells(image: Image.Image) -> tuple[str, ...]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    cyan_distance = np.sqrt(((arr - GUIDE) ** 2).sum(axis=2))
    xs, ys = _grid_lines(cyan_distance < 42.0)
    values: list[float] = []
    for row in range(N):
        for col in range(N):
            x0, x1 = xs[col], xs[col + 1]
            y0, y1 = ys[row], ys[row + 1]
            pad_x, pad_y = 0.16 * (x1 - x0), 0.16 * (y1 - y0)
            patch = arr[
                round(y0 + pad_y) : round(y1 - pad_y),
                round(x0 + pad_x) : round(x1 - pad_x),
            ]
            if patch.size == 0:
                raise ValueError("empty recovered tile")
            # The path occupies only a thin stripe; the median remains the tile fill.
            luma = 0.2126 * patch[:, :, 0] + 0.7152 * patch[:, :, 1] + 0.0722 * patch[:, :, 2]
            values.append(float(np.median(luma)))
    low, high = min(values), max(values)
    if high - low < 55.0:
        raise ValueError("two tile tones are not recoverable")
    threshold = (low + high) / 2.0
    bits = ["1" if value < threshold else "0" for value in values]
    return tuple("".join(bits[row * N : (row + 1) * N]) for row in range(N))


def _unspiral(source: tuple[str, ...]) -> tuple[str, ...]:
    stream = "".join(source[row][col] for row, col in SPIRAL)
    return tuple(stream[row * N : (row + 1) * N] for row in range(N))


def _hamming(a: tuple[str, ...], b: tuple[str, ...]) -> int:
    return sum(x != y for row_a, row_b in zip(a, b) for x, y in zip(row_a, row_b))


def decision_from_image(image: Image.Image) -> str:
    output = _unspiral(_recover_cells(image))
    scores = sorted((_hamming(output, template), name) for name, template in TEMPLATES.items())
    return scores[0][1]
