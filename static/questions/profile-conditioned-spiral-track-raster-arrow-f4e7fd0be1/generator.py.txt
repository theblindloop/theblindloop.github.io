"""Deterministic latent sampler and renderer for the spiral-track world."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

N = 7
DIRECTIONS = ("up", "right", "down", "left")

_UP = (
    "0001000",
    "0011100",
    "0111110",
    "1111111",
    "0001000",
    "0001000",
    "0001000",
)

_PALETTES = (
    ((38, 50, 56), (239, 243, 244), (250, 249, 246)),
    ((32, 50, 79), (244, 238, 218), (250, 250, 248)),
    ((72, 38, 61), (229, 239, 229), (248, 249, 247)),
)
GUIDE = (16, 126, 143)
MARKER = (236, 171, 38)


def _rotate_clockwise(rows: Iterable[str]) -> tuple[str, ...]:
    matrix = tuple(rows)
    return tuple("".join(matrix[N - 1 - r][c] for r in range(N)) for c in range(N))


def _templates() -> dict[str, tuple[str, ...]]:
    out: dict[str, tuple[str, ...]] = {}
    rows = _UP
    for name in DIRECTIONS:
        out[name] = rows
        rows = _rotate_clockwise(rows)
    return out


TEMPLATES = _templates()


def spiral_positions() -> tuple[tuple[int, int], ...]:
    positions: list[tuple[int, int]] = []
    top, bottom, left, right = 0, N - 1, 0, N - 1
    while top <= bottom and left <= right:
        positions.extend((top, col) for col in range(left, right + 1))
        top += 1
        positions.extend((row, right) for row in range(top, bottom + 1))
        right -= 1
        if top <= bottom:
            positions.extend((bottom, col) for col in range(right, left - 1, -1))
            bottom -= 1
        if left <= right:
            positions.extend((row, left) for row in range(bottom, top - 1, -1))
            left += 1
    return tuple(positions)


SPIRAL = spiral_positions()


def _hamming(rows: tuple[str, ...], template: tuple[str, ...]) -> int:
    return sum(a != b for row_a, row_b in zip(rows, template) for a, b in zip(row_a, row_b))


def _classification(rows: tuple[str, ...]) -> tuple[str, int, tuple[tuple[int, str], ...]]:
    scores = tuple(sorted((_hamming(rows, TEMPLATES[name]), name) for name in DIRECTIONS))
    return scores[0][1], scores[1][0] - scores[0][0], scores


def _source_from_output(rows: tuple[str, ...]) -> tuple[str, ...]:
    flat = "".join(rows)
    source = [["0"] * N for _ in range(N)]
    for bit, (row, col) in zip(flat, SPIRAL):
        source[row][col] = bit
    return tuple("".join(row) for row in source)


def _output_from_source(rows: tuple[str, ...]) -> tuple[str, ...]:
    stream = "".join(rows[row][col] for row, col in SPIRAL)
    return tuple(stream[row * N : (row + 1) * N] for row in range(N))


def sample_scene(seed: int) -> dict:
    target_index = seed % len(DIRECTIONS)
    target = DIRECTIONS[target_index]
    neighbor = DIRECTIONS[(target_index + 1) % len(DIRECTIONS)]
    rows = [list(row) for row in TEMPLATES[target]]
    differing = [
        (r, c)
        for r in range(N)
        for c in range(N)
        if TEMPLATES[target][r][c] != TEMPLATES[neighbor][r][c]
    ]
    style_index = seed // len(DIRECTIONS)
    rng = random.Random(0x5A17C3 ^ style_index)
    rng.shuffle(differing)
    # 0, 1, and 2 flips give margins 6, 4, and 2. Three creates a declared tie.
    flip_count = style_index % 4
    for r, c in differing[:flip_count]:
        rows[r][c] = TEMPLATES[neighbor][r][c]
    output = tuple("".join(row) for row in rows)
    dark, light, background = _PALETTES[style_index % len(_PALETTES)]
    cell_px = (48, 52, 56)[(style_index // 3) % 3]
    return {
        "cells": list(_source_from_output(output)),
        "cell_px": cell_px,
        "origin_x": 42 + rng.randrange(0, 19),
        "origin_y": 54 + rng.randrange(0, 23),
        "dark_rgb": list(dark),
        "light_rgb": list(light),
        "background_rgb": list(background),
        "guide_rgb": list(GUIDE),
        "marker_rgb": list(MARKER),
    }


def _rows(scene: dict) -> tuple[str, ...]:
    rows = tuple(str(row) for row in scene["cells"])
    if len(rows) != N or any(len(row) != N or set(row) - {"0", "1"} for row in rows):
        raise ValueError("cells must be a 7-by-7 binary matrix")
    return rows


def analytic_gold(scene: dict) -> str:
    output = _output_from_source(_rows(scene))
    return _classification(output)[0]


def margin(scene: dict) -> float:
    output = _output_from_source(_rows(scene))
    return float(_classification(output)[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _rgb(value: object) -> tuple[int, int, int]:
    seq = tuple(int(v) for v in value)  # type: ignore[arg-type]
    if len(seq) != 3:
        raise ValueError("RGB value must have three channels")
    return seq


def render(scene: dict) -> Image.Image:
    rows = _rows(scene)
    scale = 3
    width, height = 820, 520
    cell = int(scene["cell_px"])
    ox, oy = int(scene["origin_x"]), int(scene["origin_y"])
    dark, light = _rgb(scene["dark_rgb"]), _rgb(scene["light_rgb"])
    background = _rgb(scene["background_rgb"])
    guide, marker = _rgb(scene["guide_rgb"]), _rgb(scene["marker_rgb"])
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)

    def box(coords: tuple[float, float, float, float]) -> tuple[int, int, int, int]:
        return tuple(round(v * scale) for v in coords)  # type: ignore[return-value]

    for row in range(N):
        for col in range(N):
            x0, y0 = ox + col * cell, oy + row * cell
            fill = dark if rows[row][col] == "1" else light
            draw.rectangle(box((x0, y0, x0 + cell, y0 + cell)), fill=fill)

    grid_width = 3 * scale
    for index in range(N + 1):
        x = (ox + index * cell) * scale
        y = (oy + index * cell) * scale
        draw.line((x, oy * scale, x, (oy + N * cell) * scale), fill=guide, width=grid_width)
        draw.line((ox * scale, y, (ox + N * cell) * scale, y), fill=guide, width=grid_width)

    centers = [((ox + (c + 0.5) * cell) * scale, (oy + (r + 0.5) * cell) * scale) for r, c in SPIRAL]
    draw.line(centers, fill=guide, width=4 * scale, joint="curve")
    # Gold start fiducial and a small arrowhead on the first segment.
    sx, sy = centers[0]
    radius = 7 * scale
    draw.ellipse((sx - radius, sy - radius, sx + radius, sy + radius), fill=marker, outline=(255, 255, 255), width=scale)
    ax = centers[1][0] - 10 * scale
    ay = centers[1][1]
    draw.polygon(((ax, ay), (ax - 9 * scale, ay - 6 * scale), (ax - 9 * scale, ay + 6 * scale)), fill=marker)

    # A glyphic blank receiving raster, visibly linked without natural-language payload.
    rx, ry, rc = 520, 150, 28
    muted = (148, 156, 160)
    draw.line(((ox + N * cell + 22) * scale, (oy + 3.5 * cell) * scale, (rx - 24) * scale, (ry + 3.5 * rc) * scale), fill=marker, width=5 * scale)
    draw.polygon((((rx - 18) * scale, (ry + 3.5 * rc) * scale), ((rx - 32) * scale, (ry + 3.5 * rc - 9) * scale), ((rx - 32) * scale, (ry + 3.5 * rc + 9) * scale)), fill=marker)
    for index in range(N + 1):
        x = (rx + index * rc) * scale
        y = (ry + index * rc) * scale
        draw.line((x, ry * scale, x, (ry + N * rc) * scale), fill=muted, width=2 * scale)
        draw.line((rx * scale, y, (rx + N * rc) * scale, y), fill=muted, width=2 * scale)

    return image.resize((width, height), Image.Resampling.LANCZOS)
