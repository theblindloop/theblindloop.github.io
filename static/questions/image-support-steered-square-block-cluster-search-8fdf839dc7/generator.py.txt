"""Deterministic latent sampler and renderer for the block-cluster search world."""

from __future__ import annotations

import copy
import random
from typing import Iterable

from PIL import Image, ImageDraw


COLOR_ORDER = ("red", "blue", "green", "purple")
PALETTE = {
    "red": (205, 58, 68),
    "blue": (45, 105, 194),
    "green": (38, 145, 91),
    "purple": (132, 76, 174),
}
BACKGROUND = (247, 244, 237)
CANVAS_SIZE = 512

# A 3x2 rectangle with one cell removed.  The target loses a corner and
# therefore retains one 2x2 block; a distractor loses a middle cell and has no
# 2x2 block.  Both use exactly five equal cells and the same bounding extent.
_TARGET = ((0, 0), (1, 0), (2, 0), (0, 1), (1, 1))
_DISTRACTOR = ((0, 0), (2, 0), (0, 1), (1, 1), (2, 1))


def _normalize(cells: Iterable[tuple[int, int]]) -> tuple[tuple[int, int], ...]:
    points = list(cells)
    min_x = min(x for x, _ in points)
    min_y = min(y for _, y in points)
    return tuple(sorted((x - min_x, y - min_y) for x, y in points))


def _transform(
    cells: Iterable[tuple[int, int]], rotation: int, mirrored: bool
) -> tuple[tuple[int, int], ...]:
    points = list(cells)
    if mirrored:
        points = [(-x, y) for x, y in points]
    for _ in range(rotation % 4):
        points = [(-y, x) for x, y in points]
    return _normalize(points)


def _block_count(cells: Iterable[Iterable[int]]) -> int:
    occupied = {(int(x), int(y)) for x, y in cells}
    if not occupied:
        return 0
    count = 0
    for x, y in occupied:
        if {(x, y), (x + 1, y), (x, y + 1), (x + 1, y + 1)} <= occupied:
            count += 1
    return count


def sample_scene(seed: int) -> dict:
    """Sample one balanced scene; consecutive seeds cycle the answer color."""

    rng = random.Random(int(seed))
    target_color = COLOR_ORDER[int(seed) % len(COLOR_ORDER)]
    slot_colors = list(COLOR_ORDER)
    rng.shuffle(slot_colors)
    x_centers = [64, 192, 320, 448]
    row_center = rng.choice([192, 256, 320])
    cell_size = rng.choice([14, 15, 16])

    pieces = []
    for x_center, color in zip(x_centers, slot_colors, strict=True):
        base = _TARGET if color == target_color else _DISTRACTOR
        rotation = rng.randrange(4)
        mirrored = bool(rng.randrange(2))
        cells = _transform(base, rotation, mirrored)
        columns = max(x for x, _ in cells) + 1
        rows = max(y for _, y in cells) + 1
        if color == target_color:
            occupied = set(cells)
            block_x, block_y = next(
                (x, y)
                for x, y in cells
                if {(x, y), (x + 1, y), (x, y + 1), (x + 1, y + 1)} <= occupied
            )
            # Center the decisive 2x2 area on a randomized coarse-grid
            # intersection.  Rotation still varies, but the compact evidence
            # never becomes an accidental one-dimensional ablation trace.
            left = x_center - (block_x + 1) * cell_size
            top = row_center - (block_y + 1) * cell_size
        else:
            left = x_center - (columns * cell_size) // 2
            top = row_center - (rows * cell_size) // 2
        pieces.append(
            {
                "color": color,
                "origin": [left, top],
                "cells": [list(point) for point in cells],
            }
        )
    rng.shuffle(pieces)
    return {
        "canvas_size": CANVAS_SIZE,
        "cell_size": cell_size,
        "background": list(BACKGROUND),
        "pieces": pieces,
    }


def _highlight(color: tuple[int, int, int]) -> tuple[int, int, int]:
    return tuple(round(channel * 0.82 + 255 * 0.18) for channel in color)


def render(scene: dict) -> Image.Image:
    """Render only visible colored clusters on an otherwise neutral canvas."""

    size = int(scene["canvas_size"])
    background = tuple(int(value) for value in scene["background"])
    cell_size = int(scene["cell_size"])
    pieces = scene["pieces"]
    image = Image.new("RGB", (size, size), background)
    draw = ImageDraw.Draw(image)

    for piece in pieces:
        color_name = str(piece["color"])
        color = PALETTE[color_name]
        accent = _highlight(color)
        cells = [(int(x), int(y)) for x, y in piece["cells"]]
        left, top = (int(value) for value in piece["origin"])
        for cell_x, cell_y in cells:
            x0 = left + cell_x * cell_size
            y0 = top + cell_y * cell_size
            x1 = x0 + cell_size - 1
            y1 = y0 + cell_size - 1
            draw.rectangle((x0, y0, x1, y1), fill=color)
            # Same-hue one-pixel highlights make the equal-cell construction
            # visible without disconnecting the filled area.
            draw.line((x0, y0, x1, y0), fill=accent, width=1)
            draw.line((x0, y0, x0, y1), fill=accent, width=1)
    return image


def _scores(pieces: list[dict]) -> list[tuple[int, str]]:
    return sorted(
        [(_block_count(piece["cells"]), str(piece["color"])) for piece in pieces],
        reverse=True,
    )


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene["pieces"])
    top_score, top_color = scores[0]
    runner_up = scores[1][0]
    if top_score < 1 or top_score == runner_up:
        raise ValueError("scene has no unique block-bearing color")
    return top_color


def margin(scene: dict) -> float:
    scores = _scores(scene["pieces"])
    return float(scores[0][0] - scores[1][0])


def is_quarantined(scene: dict) -> bool:
    scores = _scores(scene["pieces"])
    return scores[0][0] < 1 or scores[0][0] - scores[1][0] < 1


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["pieces"] = list(reversed(twin["pieces"]))
    return [("reverse_nonoverlapping_piece_records", twin)]
