"""Independent image-only inverse arm for the block-cluster search world."""

from __future__ import annotations

import numpy as np
from PIL import Image


_PALETTE = {
    "red": (205, 58, 68),
    "blue": (45, 105, 194),
    "green": (38, 145, 91),
    "purple": (132, 76, 174),
}


def _occupancy_score(mask: np.ndarray) -> int:
    points = np.argwhere(mask)
    if len(points) < 80:
        return 0
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    width = int(x1 - x0 + 1)
    height = int(y1 - y0 + 1)
    aspect = max(width, height) / max(1, min(width, height))
    if not 1.35 <= aspect <= 1.65:
        return 0
    if width > height:
        columns, rows = 3, 2
    else:
        columns, rows = 2, 3
    if abs(width / columns - height / rows) > 1.5:
        return 0

    occupied = np.zeros((rows, columns), dtype=bool)
    for row in range(rows):
        sy0 = int(y0 + round(row * height / rows))
        sy1 = int(y0 + round((row + 1) * height / rows))
        for column in range(columns):
            sx0 = int(x0 + round(column * width / columns))
            sx1 = int(x0 + round((column + 1) * width / columns))
            tile = mask[sy0:sy1, sx0:sx1]
            occupied[row, column] = bool(tile.size and float(tile.mean()) >= 0.55)

    if int(occupied.sum()) != 5:
        return 0

    score = 0
    for row in range(rows - 1):
        for column in range(columns - 1):
            if occupied[row : row + 2, column : column + 2].all():
                score += 1
    return score


def decision_from_image(image: Image.Image) -> str:
    """Enumerate color masks, recover their cell grids, and require one winner."""

    array = np.asarray(image.convert("RGB"), dtype=np.int32)
    scores: list[tuple[int, str]] = []
    for name, rgb in _PALETTE.items():
        delta = array - np.asarray(rgb, dtype=np.int32)
        distance_squared = np.sum(delta * delta, axis=2, dtype=np.int32)
        mask = distance_squared <= 65 * 65
        scores.append((_occupancy_score(mask), name))
    scores.sort(reverse=True)
    top_score, answer = scores[0]
    runner_up = scores[1][0]
    if top_score < 1 or top_score - runner_up < 1:
        raise ValueError("PNG has no unique filled-2x2 winner with a one-block margin")
    return answer
