"""Independent image-only inverse arm for the odd-corner operation."""

from __future__ import annotations

from dataclasses import dataclass

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    pass


@dataclass(frozen=True)
class Measurement:
    decision: str
    margin: float
    ring_side: str
    selected_y: tuple[float, ...]


def _largest_component(mask: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    best: list[tuple[int, int]] = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        stack = [(sy, sx)]
        seen[sy, sx] = True
        points: list[tuple[int, int]] = []
        while stack:
            y, x = stack.pop()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) > len(best):
            best = points
    output = np.zeros_like(mask, dtype=bool)
    for y, x in best:
        output[y, x] = True
    return output


def measure(image: Image.Image) -> Measurement:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    high = array.max(axis=2)
    low = array.min(axis=2)
    # The path is the only saturated chromatic component. This criterion is
    # palette-agnostic and does not encode the answer.
    chroma = high - low
    trace = (chroma > 58) & (high > 115)
    trace = _largest_component(trace)
    ys, xs = np.nonzero(trace)
    if len(xs) < 120:
        raise OracleAbstention("connected trace is missing or too short")
    xmin, xmax = int(xs.min()), int(xs.max())
    ymin, ymax = int(ys.min()), int(ys.max())
    if xmax - xmin < 42 or xmax - xmin > 75 or ymax - ymin < 10 or ymax - ymin > 42:
        raise OracleAbstention("trace dimensions are inconsistent with the public construction")

    # Locate the dark ring independently of the path colour.
    gray_spread = array.max(axis=2) - array.min(axis=2)
    dark = (array.mean(axis=2) < 105) & (gray_spread < 32)
    dy, dx = np.nonzero(dark)
    if len(dx) < 20:
        raise OracleAbstention("ringed traversal start is missing")
    ring_x = float(np.median(dx))
    left_distance = abs(ring_x - xmin)
    right_distance = abs(ring_x - xmax)
    if min(left_distance, right_distance) > 12 or abs(left_distance - right_distance) < 15:
        raise OracleAbstention("ring does not identify exactly one trace end")
    ring_left = left_distance < right_distance

    centers: list[float] = []
    for index in range(10):
        x = xmin + (xmax - xmin) * index / 9.0
        lo = max(xmin, int(round(x)) - 1)
        hi = min(xmax, int(round(x)) + 1)
        local_y, _ = np.nonzero(trace[:, lo : hi + 1])
        if len(local_y) < 2:
            raise OracleAbstention("a required corner sample is missing")
        centers.append(float(np.median(local_y)))
    if not ring_left:
        centers.reverse()
    selected = np.asarray(centers[0:9:2], dtype=np.float64)
    selected -= selected.mean()
    template = np.asarray((1.0, -1.0, 0.0, -1.0, 1.0), dtype=np.float64)
    template -= template.mean()
    score = float(np.dot(selected, template) / np.dot(template, template))
    # The runner-up is the reflected template, so the score gap is twice the
    # absolute signed projection. Very small gaps are an honest abstention.
    gap = 2.0 * abs(score)
    if gap < 1.0:
        raise OracleAbstention("M/W template score gap is below one pixel")
    return Measurement("M" if score > 0 else "W", gap, "left" if ring_left else "right", tuple(selected))


def decision_from_image(image: Image.Image) -> str:
    return measure(image).decision
