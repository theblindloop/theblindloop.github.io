"""Deterministic latent scene and raster renderer for odd-vertex reassembly."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 384
PALETTES = (
    {"background": (248, 246, 240), "trace": (15, 163, 205), "marker": (35, 38, 43)},
    {"background": (245, 248, 247), "trace": (18, 142, 224), "marker": (42, 37, 48)},
    {"background": (250, 247, 243), "trace": (12, 177, 181), "marker": (38, 42, 40)},
)
TEMPLATE = (1.0, -1.0, 0.0, -1.0, 1.0)
DECOY_PATTERNS = (
    (-7.0, 8.0, -8.0, 7.0, -7.5),
    (8.0, -7.0, 7.5, -8.0, 7.0),
    (-8.0, 7.0, 8.0, -7.5, 7.0),
    (7.0, 8.0, -7.5, 7.0, -8.0),
)


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed) ^ 0x5A17C3)
    # Every sixth scene probes the visible low-amplitude boundary; all others
    # have large score gaps. Sign alternates deterministically for balance.
    magnitude = rng.uniform(3.0, 3.7) if seed % 6 == 0 else rng.uniform(7.0, 8.0)
    amplitude = magnitude if seed % 2 == 0 else -magnitude
    step_x = rng.choice((6.0, 6.25, 6.5))
    span = 9.0 * step_x
    x_cell = rng.randrange(1, 7)
    origin_x = 48.0 * x_cell + 24.0 - span / 2.0
    origin_y = 48.0 * rng.randrange(1, 7) + 24.0
    return {
        "amplitude": round(amplitude, 4),
        "origin_x": round(origin_x, 4),
        "origin_y": round(origin_y, 4),
        "step_x": step_x,
        "decoy_offsets": list(rng.choice(DECOY_PATTERNS)),
        "palette_id": rng.randrange(len(PALETTES)),
        "stroke_width": rng.choice((3.5, 4.0, 4.5)),
        "ring_left": bool(rng.randrange(2)),
        "canvas_width": WIDTH,
        "canvas_height": HEIGHT,
    }


def _vertices(scene: dict[str, Any]) -> list[tuple[float, float]]:
    amp = float(scene["amplitude"])
    x0 = float(scene["origin_x"])
    y0 = float(scene["origin_y"])
    step = float(scene["step_x"])
    decoys = [float(value) for value in scene["decoy_offsets"]]
    ring_left = bool(scene["ring_left"])
    target_offsets = [amp * value for value in TEMPLATE]
    points: list[tuple[float, float]] = []
    for index in range(10):
        target_slot = (index % 2 == 0) if ring_left else (index % 2 == 1)
        rank = index // 2
        offset = target_offsets[rank] if target_slot else decoys[rank]
        points.append((x0 + index * step, y0 + offset))
    return points


def render(scene: dict[str, Any]) -> Image.Image:
    palette = PALETTES[int(scene["palette_id"])]
    scale = 4
    width = int(scene.get("canvas_width", WIDTH))
    height = int(scene.get("canvas_height", HEIGHT))
    image = Image.new("RGB", (width * scale, height * scale), palette["background"])
    draw = ImageDraw.Draw(image)
    points = _vertices(scene)
    start = points[0] if bool(scene["ring_left"]) else points[-1]
    radius = 6.0
    box = tuple((value - radius) * scale for value in start) + tuple(
        (value + radius) * scale for value in start
    )
    draw.ellipse(box, fill=palette["background"], outline=palette["marker"], width=2 * scale)
    draw.line(
        [(round(x * scale), round(y * scale)) for x, y in points],
        fill=palette["trace"],
        width=round(float(scene["stroke_width"]) * scale),
        joint="curve",
    )
    # Small circular joins make all ten corners unambiguous while preserving a
    # single connected thin component.
    join_r = float(scene["stroke_width"]) * 0.52
    for x, y in points:
        draw.ellipse(
            ((x - join_r) * scale, (y - join_r) * scale,
             (x + join_r) * scale, (y + join_r) * scale),
            fill=palette["trace"],
        )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    return "M" if float(scene["amplitude"]) > 0.0 else "W"


def margin(scene: dict[str, Any]) -> float:
    return abs(float(scene["amplitude"]))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return abs(float(scene["amplitude"])) < 4.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
