"""Independent pixel-only inverse arm for the column-cover clip search world.

The oracle receives one rasterized PNG and nothing else. It segments ink from
the background, groups ink into clip blobs, recovers each clip's two rows and
its eight column positions from that blob's own ink extent, counts how many
squares occupy every column, and selects the clip whose eight columns are each
covered exactly once. It abstains whenever the declared structure is absent or
the winner is not unique.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "column_cover_clip_search-oracle-0.1.0"

COLUMNS = 8
BLOCK = 8
DILATE = 3
ABSTAIN = "abstain"
SLOTS = ("upper-left", "upper-right", "lower-left", "lower-right")

MIN_CLIP_W, MAX_CLIP_W = 70, 104
MIN_CLIP_H, MAX_CLIP_H = 52, 78
MIN_CLIP_INK, MAX_CLIP_INK = 1300, 2200
MIN_BAND_H, MAX_BAND_H = 16, 28


def _mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    border = np.concatenate(
        (
            array[:3].reshape(-1, 3),
            array[-3:].reshape(-1, 3),
            array[:, :3].reshape(-1, 3),
            array[:, -3:].reshape(-1, 3),
        ),
        axis=0,
    )
    background = np.median(border, axis=0)
    delta = np.abs(array - background[None, None, :]).sum(axis=2)
    return delta > 90


def _block_max(mask: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    ph = (-height) % BLOCK
    pw = (-width) % BLOCK
    padded = np.pad(mask, ((0, ph), (0, pw)), constant_values=False)
    small = padded.reshape(padded.shape[0] // BLOCK, BLOCK, padded.shape[1] // BLOCK, BLOCK)
    return small.max(axis=(1, 3))


def _dilate(small: np.ndarray, radius: int) -> np.ndarray:
    out = small.copy()
    for _ in range(radius):
        shifted = out.copy()
        shifted[1:, :] |= out[:-1, :]
        shifted[:-1, :] |= out[1:, :]
        shifted[:, 1:] |= out[:, :-1]
        shifted[:, :-1] |= out[:, 1:]
        out = shifted
    return out


def _components(grid: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = grid.shape
    seen = np.zeros_like(grid, dtype=bool)
    blobs: list[list[tuple[int, int]]] = []
    for start in np.argwhere(grid):
        sy, sx = int(start[0]), int(start[1])
        if seen[sy, sx]:
            continue
        stack = [(sy, sx)]
        seen[sy, sx] = True
        cells: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            cells.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and grid[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        blobs.append(cells)
    return blobs


def _row_bands(rows: np.ndarray) -> list[tuple[int, int]]:
    bands: list[tuple[int, int]] = []
    start = None
    for index, value in enumerate(rows):
        if value and start is None:
            start = index
        elif not value and start is not None:
            bands.append((start, index - 1))
            start = None
    if start is not None:
        bands.append((start, len(rows) - 1))
    return bands


def _read_clip(mask: np.ndarray, y0: int, y1: int, x0: int, x1: int):
    window = mask[y0 : y1 + 1, x0 : x1 + 1]
    bands = _row_bands(window.any(axis=1))
    if len(bands) != 2:
        return None
    counts = [0] * COLUMNS
    cell = (x1 - x0 + 1) / COLUMNS
    for top, bottom in bands:
        if not (MIN_BAND_H <= bottom - top + 1 <= MAX_BAND_H):
            return None
        centre = (top + bottom) // 2
        for column in range(COLUMNS):
            cx = int(round((column + 0.5) * cell))
            patch = window[
                max(0, centre - 3) : centre + 4,
                max(0, cx - 2) : cx + 3,
            ]
            if patch.size == 0:
                return None
            if patch.mean() > 0.5:
                counts[column] += 1
    if max(counts) > 2:
        return None
    if counts[0] != 1 or counts[COLUMNS - 1] != 1:
        return None
    if sum(counts) != COLUMNS:
        return None
    score = sum(1 for value in counts if value != 1)
    return counts, score


def measure_from_image(image: Image.Image):
    """Recover every clip's column coverage and the runner-up margin, or None."""

    mask = _mask(image)
    if not mask.any():
        return None
    grid = _dilate(_block_max(mask), DILATE)
    clips = []
    for cells in _components(grid):
        ys = [cell[0] for cell in cells]
        xs = [cell[1] for cell in cells]
        wy0 = max(0, min(ys) * BLOCK)
        wy1 = min(mask.shape[0] - 1, (max(ys) + 1) * BLOCK - 1)
        wx0 = max(0, min(xs) * BLOCK)
        wx1 = min(mask.shape[1] - 1, (max(xs) + 1) * BLOCK - 1)
        window = mask[wy0 : wy1 + 1, wx0 : wx1 + 1]
        if not window.any():
            continue
        rows = np.flatnonzero(window.any(axis=1))
        cols = np.flatnonzero(window.any(axis=0))
        y0, y1 = wy0 + int(rows[0]), wy0 + int(rows[-1])
        x0, x1 = wx0 + int(cols[0]), wx0 + int(cols[-1])
        height = y1 - y0 + 1
        width = x1 - x0 + 1
        ink = int(window.sum())
        if not (MIN_CLIP_W <= width <= MAX_CLIP_W and MIN_CLIP_H <= height <= MAX_CLIP_H):
            continue
        if not (MIN_CLIP_INK <= ink <= MAX_CLIP_INK):
            continue
        reading = _read_clip(mask, y0, y1, x0, x1)
        if reading is None:
            return None
        counts, score = reading
        clips.append(
            {
                "centre": ((y0 + y1) / 2.0, (x0 + x1) / 2.0),
                "counts": counts,
                "score": score,
            }
        )
    if len(clips) != 4:
        return None

    clips.sort(key=lambda item: item["centre"][0])
    top = sorted(clips[:2], key=lambda item: item["centre"][1])
    bottom = sorted(clips[2:], key=lambda item: item["centre"][1])
    if abs(top[0]["centre"][0] - top[1]["centre"][0]) > 60:
        return None
    if abs(bottom[0]["centre"][0] - bottom[1]["centre"][0]) > 60:
        return None
    if bottom[0]["centre"][0] - top[0]["centre"][0] < 200:
        return None
    if abs(top[0]["centre"][1] - bottom[0]["centre"][1]) > 60:
        return None
    if abs(top[1]["centre"][1] - bottom[1]["centre"][1]) > 60:
        return None
    if top[1]["centre"][1] - top[0]["centre"][1] < 200:
        return None

    ordered = [top[0], top[1], bottom[0], bottom[1]]
    scores = [item["score"] for item in ordered]
    best = min(scores)
    if best != 0 or scores.count(0) != 1:
        return None
    runner_up = min(score for score in scores if score != 0)
    if runner_up < 2:
        return None
    return {
        "decision": SLOTS[scores.index(0)],
        "pixel_margin": float(runner_up),
        "scores": scores,
        "counts": [item["counts"] for item in ordered],
    }


def decision_from_image(image: Image.Image) -> str:
    """Pixel-only decision, or `abstain` when the declared structure is not readable."""

    report = measure_from_image(image)
    if report is None:
        return ABSTAIN
    return str(report["decision"])
