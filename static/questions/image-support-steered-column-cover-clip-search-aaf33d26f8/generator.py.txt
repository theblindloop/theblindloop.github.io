"""Renderer and analytic gold for the column-cover clip search world.

Four clips sit in one tight two-by-two cluster on a plain page. Each clip is two
separate rows of small solid squares standing on the same eight column
positions. A column of a clip is covered once, twice, or left bare. Exactly one
clip covers every one of its eight columns exactly once.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "column_cover_clip_search-renderer-0.1.0"

WIDTH = 1536
HEIGHT = 1152
BACKGROUND = (250, 249, 246)

COLUMNS = 8
TILE_W = 10
TILE_H = 22
PITCH = 11
ROW_GAP = 24
CLIP_W = PITCH * (COLUMNS - 1) + TILE_W          # 87
CLIP_H = 2 * TILE_H + ROW_GAP                    # 68

# Slot centres are the centres of the 8x8 ablation cells (2,2), (2,5), (5,2)
# and (5,5); the jitter keeps every clip box inside the safe core of its cell.
SLOT_CENTRES = {
    "upper-left": (480, 360),
    "upper-right": (1056, 360),
    "lower-left": (480, 792),
    "lower-right": (1056, 792),
}
SLOTS = ("upper-left", "upper-right", "lower-left", "lower-right")
JITTER_X = 3
JITTER_Y = 2

INKS = (
    (32, 38, 54),
    (122, 26, 34),
    (24, 74, 58),
    (66, 34, 96),
)

INTERIOR = tuple(range(1, COLUMNS - 1))
ROW_KEYS = ("upper", "lower")


def _column_counts(clip: dict) -> list[int]:
    counts = [0] * COLUMNS
    for row_key in ROW_KEYS:
        for column in clip[row_key]:
            counts[int(column)] += 1
    return counts


def _clip_score(clip: dict) -> int:
    """Number of columns of this clip that are not covered exactly once."""

    return sum(1 for count in _column_counts(clip) if count != 1)


def _sample_partition(rng: random.Random, size: int) -> tuple[list[int], list[int]]:
    upper = sorted(rng.sample(range(COLUMNS), size))
    lower = [column for column in range(COLUMNS) if column not in upper]
    return upper, lower


def _perturb(rng: random.Random, upper: list[int], lower: list[int], moves: int):
    """Move `moves` squares sideways inside one row of a complementary clip.

    Each move empties one interior column and stacks a second square onto
    another interior column, so the clip keeps exactly eight squares and both
    row counts, but gains `moves` bare and `moves` doubly covered columns.
    """

    if rng.random() < 0.5:
        source, target = list(upper), list(lower)
        flip = False
    else:
        source, target = list(lower), list(upper)
        flip = True
    donors = [column for column in source if column in INTERIOR]
    receivers = [column for column in target if column in INTERIOR]
    if len(donors) < moves or len(receivers) < moves:
        return None
    picked_from = rng.sample(donors, moves)
    picked_to = rng.sample(receivers, moves)
    for old, new in zip(picked_from, picked_to):
        source.remove(old)
        source.append(new)
    source.sort()
    if flip:
        return sorted(target), source
    return source, sorted(target)


def _sample_loser(rng: random.Random, moves: int, size: int | None):
    for _ in range(200):
        chosen = size if size is not None else rng.choice((3, 4, 5))
        upper, lower = _sample_partition(rng, chosen)
        result = _perturb(rng, upper, lower, moves)
        if result is None:
            continue
        new_upper, new_lower = result
        clip = {"upper": new_upper, "lower": new_lower}
        if _clip_score(clip) != 2 * moves:
            continue
        counts = _column_counts(clip)
        if counts[0] != 1 or counts[COLUMNS - 1] != 1:
            continue
        if max(counts) > 2:
            continue
        return new_upper, new_lower
    return None


def sample_scene(seed: int) -> dict[str, Any]:
    """Deterministically sample one latent scene."""

    rng = random.Random(seed * 7919 + 13)
    winner_slot = SLOTS[seed % 4]
    min_moves = 1 + (seed // 4) % 3
    winner_size = 3 + (seed // 12) % 3

    winner_upper, winner_lower = _sample_partition(rng, winner_size)

    loser_slots = [slot for slot in SLOTS if slot != winner_slot]
    move_plan = [min_moves]
    while len(move_plan) < 3:
        move_plan.append(min(3, min_moves + rng.choice((0, 1))))
    rng.shuffle(move_plan)

    inks = list(INKS)
    rng.shuffle(inks)

    clips: list[dict[str, Any]] = []
    for index, slot in enumerate(SLOTS):
        centre_x, centre_y = SLOT_CENTRES[slot]
        origin_x = centre_x - CLIP_W // 2 + rng.randint(-JITTER_X, JITTER_X)
        origin_y = centre_y - CLIP_H // 2 + rng.randint(-JITTER_Y, JITTER_Y)
        if slot == winner_slot:
            upper, lower = list(winner_upper), list(winner_lower)
        else:
            position = loser_slots.index(slot)
            forced = winner_size if position == 0 else None
            sampled = _sample_loser(rng, move_plan[position], forced)
            if sampled is None:
                sampled = _sample_loser(rng, move_plan[position], None)
            if sampled is None:  # pragma: no cover - exhaustive retry never fails
                raise RuntimeError("could not sample a losing clip")
            upper, lower = list(sampled[0]), list(sampled[1])
        clips.append(
            {
                "slot": slot,
                "ink": list(inks[index]),
                "origin_x": int(origin_x),
                "origin_y": int(origin_y),
                "upper": [int(value) for value in upper],
                "lower": [int(value) for value in lower],
            }
        )

    boxes = [
        (
            clip["origin_x"] - 110,
            clip["origin_y"] - 110,
            clip["origin_x"] + CLIP_W + 110,
            clip["origin_y"] + CLIP_H + 110,
        )
        for clip in clips
    ]
    specks: list[dict[str, Any]] = []
    attempts = 0
    while len(specks) < 14 and attempts < 800:
        attempts += 1
        x = rng.randint(12, WIDTH - 16)
        y = rng.randint(12, HEIGHT - 16)
        if any(x0 <= x <= x1 and y0 <= y <= y1 for x0, y0, x1, y1 in boxes):
            continue
        if any(abs(x - item["x"]) < 24 and abs(y - item["y"]) < 24 for item in specks):
            continue
        specks.append({"x": int(x), "y": int(y), "ink": list(rng.choice(INKS))})

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
        "columns": COLUMNS,
        "tile_w": TILE_W,
        "tile_h": TILE_H,
        "pitch": PITCH,
        "row_gap": ROW_GAP,
        "speck_size": 3,
        "clips": clips,
        "specks": specks,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Rasterize a latent scene. Drawing order is irrelevant: all marks are disjoint."""

    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    tile_w = int(scene["tile_w"])
    tile_h = int(scene["tile_h"])
    pitch = int(scene["pitch"])
    gap = int(scene["row_gap"])
    for clip in scene["clips"]:
        ink = tuple(clip["ink"])
        ox = int(clip["origin_x"])
        oy = int(clip["origin_y"])
        for band, columns in ((0, clip["upper"]), (1, clip["lower"])):
            top = oy + band * (tile_h + gap)
            for column in columns:
                left = ox + int(column) * pitch
                draw.rectangle(
                    [left, top, left + tile_w - 1, top + tile_h - 1], fill=ink
                )
    size = int(scene["speck_size"])
    for speck in scene["specks"]:
        x = int(speck["x"])
        y = int(speck["y"])
        draw.rectangle([x, y, x + size - 1, y + size - 1], fill=tuple(speck["ink"]))
    return image


def _scores(scene: dict[str, Any]) -> dict[str, int]:
    return {str(clip["slot"]): _clip_score(clip) for clip in scene["clips"]}


def analytic_gold(scene: dict[str, Any]) -> str:
    """The slot of the unique clip whose eight columns are each covered exactly once."""

    scores = _scores(scene)
    winners = [slot for slot, score in scores.items() if score == 0]
    if len(winners) != 1:
        raise ValueError("scene does not contain a unique fully covered clip")
    return winners[0]


def margin(scene: dict[str, Any]) -> float:
    """Runner-up decision margin in columns: the smallest losing clip score."""

    scores = _scores(scene)
    winner = analytic_gold(scene)
    return float(min(score for slot, score in scores.items() if slot != winner))


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Flag scenes whose runner-up is only two columns away from the winner."""

    return margin(scene) < 4.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """Declared label-preserving, pixel-identical rewrites of the latent scene."""

    rotated = dict(scene)
    rotated["clips"] = list(scene["clips"][1:]) + list(scene["clips"][:1])
    reversed_specks = dict(scene)
    reversed_specks["specks"] = list(reversed(scene["specks"]))
    return [
        ("clip_record_rotation", rotated),
        ("speck_list_reversal", reversed_specks),
    ]
