"""Deterministic renderer and analytic topology label for loop multiplicity.

The public scene contains only visible node coordinates and visible links.  The
three modules have the same nine-node, ten-link budget and equal link lengths;
their highlighted links participate in zero, one, and two simple loops.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw


RENDERER_VERSION = "loop_multiplicity_link-renderer-0.1.0"
WIDTH, HEIGHT = 1500, 600
LINK_LENGTH = 68.0
NODE_RADIUS = 12
LINK_WIDTH = 8
BACKGROUND = (248, 247, 242)
NODE = (32, 36, 44)
LINK = (117, 123, 135)
HIGHLIGHT_COLORS: dict[str, tuple[int, int, int]] = {
    "red": (218, 60, 65),
    "green": (43, 155, 91),
    "blue": (59, 103, 199),
}
ALL_EDGE_COLORS = {"gray": LINK, **HIGHLIGHT_COLORS}
SLOT_CENTERS = (250.0, 750.0, 1250.0)


def _base_specs() -> list[tuple[list[tuple[float, float]], list[tuple[int, int]], int]]:
    """Return three equal-budget graph embeddings and their marked edge index."""

    h = math.sqrt(3.0) / 2.0
    # Two cyclic pieces joined by the highlighted bridge.  The triangle and
    # square are deliberately separate blocks, so the bridge is in no loop.
    bridge_nodes = [
        (0.0, 0.0),
        (1.0, 0.0),
        (0.5, h),
        (2.0, 0.0),
        (3.0, 0.0),
        (3.0, 1.0),
        (2.0, 1.0),
        (-1.0, 0.0),
        (3.0, 2.0),
    ]
    bridge_edges = [
        (0, 1), (1, 2), (2, 0),
        (3, 4), (4, 5), (5, 6), (6, 3),
        (1, 3),
        (0, 7), (5, 8),
    ]

    # A square and a triangle sharing only one node.  The marked square edge
    # has exactly one alternate simple path between its endpoints.
    single_nodes = [
        (0.0, 0.0),
        (1.0, 0.0),
        (1.0, 1.0),
        (0.0, 1.0),
        (-h, -0.5),
        (-h, 0.5),
        (2.0, 0.0),
        (1.0, 2.0),
        (-1.0, 1.0),
    ]
    single_edges = [
        (0, 1), (1, 2), (2, 3), (3, 0),
        (0, 4), (4, 5), (5, 0),
        (1, 6), (2, 7), (3, 8),
    ]

    # Two equilateral triangles share the marked top-to-bottom diagonal.  The
    # two alternate paths are the left and right sides, so the marked link is
    # in exactly two simple loops.  The remaining edges are trees.
    double_nodes = [
        (0.0, 0.0),
        (0.0, 1.0),
        (-h, 0.5),
        (h, 0.5),
        (0.0, -1.0),
        (0.0, -2.0),
        (0.0, 2.0),
        (-h - 1.0, 0.5),
        (h + 1.0, 0.5),
    ]
    double_edges = [
        (0, 2), (2, 1), (1, 3), (3, 0),
        (0, 1),
        (0, 4), (4, 5), (1, 6), (2, 7), (3, 8),
    ]
    return [
        (bridge_nodes, bridge_edges, 7),
        (single_nodes, single_edges, 0),
        (double_nodes, double_edges, 4),
    ]


def _transform(
    points: list[tuple[float, float]],
    slot: int,
    quarter_turns: int,
    mirror: bool,
) -> list[list[int]]:
    min_x = min(point[0] for point in points)
    max_x = max(point[0] for point in points)
    min_y = min(point[1] for point in points)
    max_y = max(point[1] for point in points)
    center_x = (min_x + max_x) / 2.0
    center_y = (min_y + max_y) / 2.0
    angle = quarter_turns * math.pi / 2.0
    cosine, sine = math.cos(angle), math.sin(angle)
    result: list[list[int]] = []
    for x, y in points:
        x = (x - center_x) * LINK_LENGTH
        y = (y - center_y) * LINK_LENGTH
        if mirror:
            x = -x
        rotated_x = cosine * x - sine * y
        rotated_y = sine * x + cosine * y
        result.append(
            [
                int(round(SLOT_CENTERS[slot] + rotated_x)),
                int(round(HEIGHT / 2.0 + rotated_y)),
            ]
        )
    return result


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced three-module graph scene from an integer seed."""

    rng = random.Random(int(seed))
    specs = _base_specs()
    order = rng.sample(range(3), 3)
    color_order = rng.sample(tuple(HIGHLIGHT_COLORS), 3)
    color_for_spec = {spec_index: color_order[spec_index] for spec_index in range(3)}
    all_nodes: list[list[list[int]]] = []
    all_edges: list[list[list[Any]]] = []
    for slot, spec_index in enumerate(order):
        raw_nodes, raw_edges, marked_index = specs[spec_index]
        nodes = _transform(
            raw_nodes,
            slot,
            rng.randrange(4),
            bool(rng.randrange(2)),
        )
        edges: list[list[Any]] = []
        for edge_index, (first, second) in enumerate(raw_edges):
            edges.append([
                int(first),
                int(second),
                color_for_spec[spec_index] if edge_index == marked_index else "gray",
            ])
        all_nodes.append(nodes)
        all_edges.append(edges)
    # Keep the answer-bearing fields flat and explicit.  Each value is visible
    # in the raster: coordinates become node disks and each edge triple becomes
    # one drawn link.
    return {"nodes": all_nodes, "edges": all_edges}


def _adjacency(node_count: int, edges: list[list[Any]], skip_edge: int | None = None) -> list[list[int]]:
    adjacency = [[] for _ in range(node_count)]
    for index, edge in enumerate(edges):
        if index == skip_edge:
            continue
        first, second = int(edge[0]), int(edge[1])
        if not (0 <= first < node_count and 0 <= second < node_count):
            raise ValueError("edge endpoint is outside the node list")
        adjacency[first].append(second)
        adjacency[second].append(first)
    return adjacency


def _alternate_path_count(node_count: int, edges: list[list[Any]], skip_edge: int) -> int:
    """Count simple paths between a skipped edge's endpoints."""

    edge = edges[skip_edge]
    start, target = int(edge[0]), int(edge[1])
    adjacency = _adjacency(node_count, edges, skip_edge=skip_edge)
    visited = {start}
    count = 0

    def visit(current: int) -> None:
        nonlocal count
        if current == target:
            count += 1
            return
        for neighbor in adjacency[current]:
            if neighbor not in visited:
                visited.add(neighbor)
                visit(neighbor)
                visited.remove(neighbor)

    visit(start)
    return count


def _marked_counts(scene: dict[str, Any]) -> dict[str, int]:
    counts: dict[str, int] = {}
    for nodes, edges in zip(scene["nodes"], scene["edges"]):
        marked = [
            index
            for index, edge in enumerate(edges)
            if edge[2] in HIGHLIGHT_COLORS
        ]
        if len(marked) != 1:
            raise ValueError("each module must have exactly one highlighted link")
        color = str(edges[marked[0]][2])
        if color in counts:
            raise ValueError("highlight colors must be unique")
        counts[color] = _alternate_path_count(len(nodes), edges, marked[0])
    if set(counts) != set(HIGHLIGHT_COLORS) or sorted(counts.values()) != [0, 1, 2]:
        raise ValueError(f"expected loop counts 0, 1, 2, got {counts}")
    return counts


def analytic_gold(scene: dict[str, Any]) -> str:
    counts = _marked_counts(scene)
    return max(counts, key=counts.get)


def margin(scene: dict[str, Any]) -> float:
    counts = _marked_counts(scene)
    ordered = sorted(counts.values(), reverse=True)
    return float(ordered[0] - ordered[1])


def _orientation(first: tuple[float, float], second: tuple[float, float], third: tuple[float, float]) -> float:
    return (second[0] - first[0]) * (third[1] - first[1]) - (second[1] - first[1]) * (third[0] - first[0])


def _segments_intersect(
    first: tuple[float, float],
    second: tuple[float, float],
    third: tuple[float, float],
    fourth: tuple[float, float],
) -> bool:
    eps = 1e-6
    a = _orientation(first, second, third)
    b = _orientation(first, second, fourth)
    c = _orientation(third, fourth, first)
    d = _orientation(third, fourth, second)
    if (a > eps and b < -eps or a < -eps and b > eps) and (
        c > eps and d < -eps or c < -eps and d > eps
    ):
        return True
    return False


def _geometry_is_clear(scene: dict[str, Any]) -> bool:
    if len(scene["nodes"]) != 3 or len(scene["edges"]) != 3:
        return False
    all_nodes: list[tuple[float, float]] = []
    all_segments: list[tuple[int, int, tuple[float, float], tuple[float, float]]] = []
    for nodes_raw, edges in zip(scene["nodes"], scene["edges"]):
        nodes = [tuple(float(value) for value in node) for node in nodes_raw]
        if len(nodes) != 9 or len(edges) != 10:
            return False
        offset = len(all_nodes)
        all_nodes.extend(nodes)
        for edge in edges:
            first, second = int(edge[0]), int(edge[1])
            if not (0 <= first < len(nodes) and 0 <= second < len(nodes)):
                return False
            if edge[2] not in ALL_EDGE_COLORS:
                return False
            length = math.dist(nodes[first], nodes[second])
            if not 0.82 * LINK_LENGTH <= length <= 1.18 * LINK_LENGTH:
                return False
            all_segments.append((offset + first, offset + second, nodes[first], nodes[second]))
    if any(
        point[0] < 30.0 or point[0] > WIDTH - 30.0 or point[1] < 30.0 or point[1] > HEIGHT - 30.0
        for point in all_nodes
    ):
        return False
    for first in range(len(all_nodes)):
        for second in range(first):
            if math.dist(all_nodes[first], all_nodes[second]) < 2.0 * NODE_RADIUS + 5.0:
                return False
    for index, (first, second, start, end) in enumerate(all_segments):
        for other_first, other_second, other_start, other_end in all_segments[:index]:
            if {first, second} & {other_first, other_second}:
                continue
            if _segments_intersect(start, end, other_start, other_end):
                return False
    return True


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        return not _geometry_is_clear(scene) or margin(scene) < 0.5
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the visible node/link diagram as an RGB PNG source image."""

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for nodes, edges in zip(scene["nodes"], scene["edges"]):
        for edge in edges:
            first = nodes[int(edge[0])]
            second = nodes[int(edge[1])]
            dx = float(second[0]) - float(first[0])
            dy = float(second[1]) - float(first[1])
            length = math.hypot(dx, dy)
            ux, uy = dx / length, dy / length
            start = (
                int(round(float(first[0]) + ux * (NODE_RADIUS + 3))),
                int(round(float(first[1]) + uy * (NODE_RADIUS + 3))),
            )
            end = (
                int(round(float(second[0]) - ux * (NODE_RADIUS + 3))),
                int(round(float(second[1]) - uy * (NODE_RADIUS + 3))),
            )
            draw.line((start, end), fill=ALL_EDGE_COLORS[str(edge[2])], width=LINK_WIDTH)
    for nodes in scene["nodes"]:
        for x, y in nodes:
            draw.ellipse(
                (int(x) - NODE_RADIUS, int(y) - NODE_RADIUS,
                 int(x) + NODE_RADIUS, int(y) + NODE_RADIUS),
                fill=NODE,
            )
    return image


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No hidden, render-invariant scene fields are declared."""

    return []
