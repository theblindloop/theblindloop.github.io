"""Renderer and analytic replay for stable color-to-edge partitions."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 560
BG = (250, 249, 246)
DARK = (35, 40, 48)
PALE = (222, 226, 232)
MAGENTA = (194, 42, 127)
PALETTE = {
    "red": (220, 64, 62),
    "green": (43, 157, 92),
    "blue": (54, 112, 203),
}


def _replay(tokens: list[str], marked_index: int, commands: list[list[str]]) -> int:
    row = [(color, index) for index, color in enumerate(tokens)]
    for color, edge in commands:
        selected = [token for token in row if token[0] == color]
        others = [token for token in row if token[0] != color]
        row = selected + others if edge == "L" else others + selected
    return next(index for index, token in enumerate(row) if token[1] == marked_index)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) * 65537 + 1709)
    target = int(seed) % 6
    for _ in range(20000):
        tokens = ["red", "red", "green", "green", "blue", "blue"]
        rng.shuffle(tokens)
        marked_index = rng.randrange(6)
        commands = [[rng.choice(tuple(PALETTE)), rng.choice(("L", "R"))] for _ in range(5)]
        if _replay(tokens, marked_index, commands) == target:
            return {"tokens": tokens, "marked_index": marked_index, "commands": commands}
    raise RuntimeError("failed to sample balanced stable-partition scene")


def _font(size: int, bold: bool = False):
    try:
        name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _center(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str, size: int, fill=DARK, bold=False):
    font = _font(size, bold)
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2), text, font=font, fill=fill)


def _arrow(draw: ImageDraw.ImageDraw, cx: int, cy: int, direction: str, color: tuple[int, int, int]):
    # A broad asymmetric polygon: the larger triangular half points to the named edge.
    if direction == "R":
        pts = [(cx - 45, cy - 13), (cx + 4, cy - 13), (cx + 4, cy - 31),
               (cx + 47, cy), (cx + 4, cy + 31), (cx + 4, cy + 13), (cx - 45, cy + 13)]
    else:
        pts = [(cx + 45, cy - 13), (cx - 4, cy - 13), (cx - 4, cy - 31),
               (cx - 47, cy), (cx - 4, cy + 31), (cx - 4, cy + 13), (cx + 45, cy + 13)]
    draw.polygon(pts, fill=color)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    _center(draw, (WIDTH // 2, 36), "STABLE COLOR-TO-EDGE REPLAY", 27, bold=True)
    _center(draw, (WIDTH // 2, 72), "Move every matching color; preserve order within both groups", 18)

    centers = [155 + 130 * index for index in range(6)]
    y = 188
    for index, (x, color) in enumerate(zip(centers, scene["tokens"], strict=True)):
        draw.ellipse((x - 37, y - 37, x + 37, y + 37), fill=PALETTE[color], outline=DARK, width=3)
        _center(draw, (x, 264), str(index + 1), 20, bold=True)
    marked_x = centers[int(scene["marked_index"])]
    draw.ellipse((marked_x - 48, y - 48, marked_x + 48, y + 48), outline=MAGENTA, width=8)
    _center(draw, (89, y), "START", 17, bold=True)

    _center(draw, (WIDTH // 2, 320), "COMMANDS  —  READ LEFT TO RIGHT", 18, bold=True)
    card_xs = [124 + 178 * index for index in range(5)]
    card_y = 430
    for index, (x, command) in enumerate(zip(card_xs, scene["commands"], strict=True)):
        color, edge = command
        draw.rounded_rectangle((x - 70, card_y - 70, x + 70, card_y + 70), radius=14,
                               fill=(255, 255, 255), outline=PALE, width=4)
        _arrow(draw, x, card_y, edge, PALETTE[color])
        _center(draw, (x, card_y + 94), str(index + 1), 17, bold=True)
    return image


def analytic_gold(scene: dict) -> str:
    return str(_replay(list(scene["tokens"]), int(scene["marked_index"]), [list(x) for x in scene["commands"]]) + 1)


def margin(scene: dict) -> float:
    # Nearest palette-channel separation from the inverse parser thresholds.
    return 42.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 20.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
