"""Prior-conflict piano white-key count world renderer.

Latent state z = {n_keys, body_x, body_y, key_w, gap_width, key_h}.

A single dark keyboard body rectangle is drawn, with `n_keys` white key
rectangles placed on it in a row, each separated from its neighbours by a dark
`gap_width`-pixel strip. The closed-form gold is the scene's key count:

    key_count = n_keys   (a scene parameter, not a pixel measurement)

The canonical prior — that a piano octave contains seven white keys — is
recorded only in the question narrative; it is never used as ground truth.
The visible truth is whatever number of white keys is actually drawn.

Continuous verifiability margin (px): the gap between two adjacent white keys.
When the gap stays above a small finite-resolution floor, the white keys remain
separable at pixel scale and the count is unambiguous from the raster.

Latent-alias mode: not_applicable. The key count is pinned by n_keys, which is
rendered as a row of distinct white rectangles; different n_keys always produce
a different (readable) raster, so there is no render-invariant latent alias and
sample_scene returns no pixel-identical twins.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS_W = 384
CANVAS_H = 384
# Reordered palette shades so that neither the wording nor the layout predicts
# the class. Canvas is a light neutral field; the keyboard body is dark; the
# counted keys are pure white.
CANVAS_BG = (238, 238, 238)
BODY_COLOR = (60, 60, 66)
KEY_COLOR = (255, 255, 255)

N_KEYS_RANGE = (5, 9)      # inclusive white-key count
KEY_H = 180.0              # px, final scale
KEY_W_RANGE = (26.0, 34.0)
GAP_RANGE = (4.0, 8.0)     # px, final scale; comfortably above the 2px floor
BODY_MARGIN = 120.0        # keep the body fully inside the canvas

RENDERER_VERSION = "prior_piano_keys-0.1.0"


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(seed)
    n_keys = rng.randint(*N_KEYS_RANGE)
    key_w = rng.uniform(*KEY_W_RANGE)
    gap = rng.uniform(*GAP_RANGE)
    # Total body width must fit fully inside the canvas so every key is visible.
    body_w = n_keys * key_w + (n_keys - 1) * gap
    pad = 14.0
    bx = rng.uniform(body_w / 2.0 + pad, CANVAS_W - body_w / 2.0 - pad)
    by = rng.uniform(KEY_H / 2.0 + pad, CANVAS_H - KEY_H / 2.0 - pad)
    return {
        "n_keys": n_keys,
        "body_x": round(bx, 3),
        "body_y": round(by, 3),
        "key_w": round(key_w, 3),
        "gap_width": round(gap, 3),
        "key_h": KEY_H,
    }


def _lefts(scene: dict[str, Any]) -> list[float]:
    n = int(scene["n_keys"])
    left = float(scene["body_x"]) - n * float(scene["key_w"]) / 2.0 - (n - 1) * float(scene["gap_width"]) / 2.0
    return [left + k * (float(scene["key_w"]) + float(scene["gap_width"])) for k in range(n)]


def render(scene: dict[str, Any]) -> Image.Image:
    n = int(scene["n_keys"])
    if n < 1:
        raise ValueError("a keyboard needs at least one white key")
    scale = 2
    big_w = CANVAS_W * scale
    big_h = CANVAS_H * scale
    big = Image.new("RGB", (big_w, big_h), CANVAS_BG)
    draw = ImageDraw.Draw(big)

    x0 = float(scene["body_x"]) - n * float(scene["key_w"]) / 2.0 - (n - 1) * float(scene["gap_width"]) / 2.0
    y0 = float(scene["body_y"]) - KEY_H / 2.0
    body_right = x0 + n * float(scene["key_w"]) + (n - 1) * float(scene["gap_width"])
    body_bottom = y0 + KEY_H
    draw.rectangle([x0 * scale, y0 * scale, body_right * scale, body_bottom * scale], fill=BODY_COLOR)

    for left in _lefts(scene):
        right = left + float(scene["key_w"])
        draw.rectangle([left * scale, y0 * scale, right * scale, body_bottom * scale], fill=KEY_COLOR)
    return big.resize((CANVAS_W, CANVAS_H), Image.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(int(scene["n_keys"]))


def margin(scene: dict[str, Any]) -> float:
    # The decision-relevant separation is the smallest gap between two adjacent
    # white keys; it is uniform across a scene.
    return float(scene["gap_width"])


def is_quarantined(scene: dict[str, Any]) -> bool:
    # Adjacent white keys become pixel-ambiguous when the gap between them drops
    # below a small finite-resolution floor.
    return margin(scene) < 2.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # not_applicable: the gold is pinned by the rendered n_keys; there is no
    # declared pixel-identical alias to exercise.
    return []
