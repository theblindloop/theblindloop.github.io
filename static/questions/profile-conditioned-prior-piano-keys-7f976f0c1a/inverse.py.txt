"""Independent pixel-only inverse oracle for the piano white-key count world.

Receives only the final PNG and re-counts the white keys directly from the
raster. It imports nothing from the renderer, prompts, generator, verifier, or
analytic-gold code, and accepts no latent scene data.

Method: the counted entities are pure-white key rectangles set on a dark
keyboard body over a light neutral field. A horizontal scanline through the
vertical centre of the white keys therefore crosses every key once: each key
appears as one contiguous white run, and the dark gap strips separate adjacent
keys. The number of white runs along that scanline equals the number of keys.
The colour threshold below is duplicated deliberately rather than imported, so
a renderer/oracle disagreement is exposed by the certification path instead of
becoming a shared-code blind spot.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_WHITE = 248  # only near-pure-white pixels count as key ink


def _white_mask(arr: np.ndarray) -> np.ndarray:
    r = arr[..., 0].astype(np.int16)
    g = arr[..., 1].astype(np.int16)
    b = arr[..., 2].astype(np.int16)
    return (r >= _WHITE) & (g >= _WHITE) & (b >= _WHITE)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.int16)
    mask = _white_mask(arr)
    ys, xs = np.nonzero(mask)
    if ys.size == 0:
        return "0"
    mid = int(round((ys.min() + ys.max()) / 2.0))
    row = mask[mid]
    runs = 0
    previous = False
    for value in row:
        if value and not previous:
            runs += 1
        previous = bool(value)
    return str(runs)
