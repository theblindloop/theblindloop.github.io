"""Deterministic latent sampler, renderer, and analytic topology arm."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 500, 380
COLORS = ("red", "green", "blue")
RGB = {
    "red": (220, 55, 70),
    "green": (20, 150, 90),
    "blue": (40, 100, 210),
}

# The realizable signatures form a forest: O={}, A={0}, Aa={0,1},
# Aaa={0,1,2}, B={3}, C={4}.  These three perfect matchings all have
# separation counts {1,2,3}, but place the count-two relation differently.
TEMPLATES = (
    {1: ("A", "Aa"), 2: ("B", "C"), 3: ("O", "Aaa")},
    {1: ("O", "B"), 2: ("A", "Aaa"), 3: ("Aa", "C")},
    {1: ("O", "C"), 2: ("A", "Aaa"), 3: ("Aa", "B")},
)


def _transform_point(point: tuple[int, int], transform: int) -> list[int]:
    x, y = point
    if transform & 1:
        x = WIDTH - 1 - x
    if transform & 2:
        y = HEIGHT - 1 - y
    return [x, y]


def _transform_rect(rect: tuple[int, int, int, int], transform: int) -> list[int]:
    x0, y0, x1, y1 = rect
    points = [_transform_point((x0, y0), transform), _transform_point((x1, y1), transform)]
    xs = [p[0] for p in points]
    ys = [p[1] for p in points]
    return [min(xs), min(ys), max(xs), max(ys)]


def sample_scene(seed: int) -> dict:
    """Sample a balanced presentation without changing the abstract topology."""
    rng = random.Random(int(seed))
    loops0 = (
        (35, 35, 245, 345),
        (75, 75, 205, 305),
        (115, 115, 165, 265),
        (290, 45, 465, 160),
        (290, 220, 465, 345),
    )
    slots0 = {
        "O": (377, 190),
        "A": (55, 190),
        "Aa": (95, 190),
        "Aaa": (140, 190),
        "B": (377, 102),
        "C": (377, 282),
    }
    transform = rng.randrange(4)
    loops = [_transform_rect(rect, transform) for rect in loops0]
    slots = {}
    for name, point in slots0.items():
        jittered = (point[0] + rng.randint(-1, 1), point[1] + rng.randint(-1, 1))
        slots[name] = _transform_point(jittered, transform)

    template_id = rng.randrange(len(TEMPLATES))
    palette = list(COLORS)
    rng.shuffle(palette)
    markers = []
    for color, count in zip(palette, (1, 2, 3), strict=True):
        left, right = TEMPLATES[template_id][count]
        markers.append({"color": color, "points": [slots[left], slots[right]]})
    rng.shuffle(markers)  # list order is nonsemantic and not a color-position cue
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": [250, 249, 246],
        "wall_color": [25, 27, 31],
        "wall_width": 9,
        "marker_radius": 9,
        "loops": loops,
        "markers": markers,
        "template_id": template_id,
        "transform": transform,
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    for rect in scene["loops"]:
        draw.rectangle(tuple(rect), outline=tuple(scene["wall_color"]), width=int(scene["wall_width"]))
    radius = int(scene["marker_radius"])
    for group in scene["markers"]:
        color = RGB[group["color"]]
        for x, y in group["points"]:
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)
    return image


def _inside_rect(point: list[int], rect: list[int]) -> bool:
    x, y = point
    x0, y0, x1, y1 = rect
    return x0 < x < x1 and y0 < y < y1


def _separation_count(scene: dict, points: list[list[int]]) -> int:
    return sum(_inside_rect(points[0], rect) != _inside_rect(points[1], rect) for rect in scene["loops"])


def analytic_gold(scene: dict) -> str:
    winners = [group["color"] for group in scene["markers"] if _separation_count(scene, group["points"]) == 2]
    if len(winners) != 1:
        raise ValueError(f"expected one count-two pair, found {winners}")
    return winners[0]


def _point_edge_clearance(point: list[int], rect: list[int]) -> float:
    x, y = point
    x0, y0, x1, y1 = rect
    if x0 < x < x1 and y0 < y < y1:
        return min(x - x0, x1 - x, y - y0, y1 - y)
    dx = max(x0 - x, 0, x - x1)
    dy = max(y0 - y, 0, y - y1)
    return math.hypot(dx, dy)


def pixel_clearance(scene: dict) -> float:
    radius = float(scene["marker_radius"])
    half_wall = float(scene["wall_width"]) / 2.0
    return min(
        _point_edge_clearance(point, rect) - radius - half_wall
        for group in scene["markers"]
        for point in group["points"]
        for rect in scene["loops"]
    )


def runner_up_margin(scene: dict) -> float:
    scores = sorted(abs(_separation_count(scene, g["points"]) - 2) for g in scene["markers"])
    return float(scores[1] - scores[0])


def margin(scene: dict) -> float:
    # Composition margin: a one-loop runner-up gap, capped by normalized raster clearance.
    return float(min(runner_up_margin(scene), pixel_clearance(scene) / 5.0))


def is_quarantined(scene: dict) -> bool:
    counts = sorted(_separation_count(scene, group["points"]) for group in scene["markers"])
    return counts != [1, 2, 3] or pixel_clearance(scene) < 5.0 or margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["loops"] = list(reversed(twin["loops"]))
    return [("reverse_non_touching_loop_order", twin)]
