"""Independent pixel-only inverse arm for loop enclosure signatures."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE = {
    "red": np.array((220, 55, 70), dtype=np.int16),
    "green": np.array((20, 150, 90), dtype=np.int16),
    "blue": np.array((40, 100, 210), dtype=np.int16),
}


class PixelTopologyError(ValueError):
    pass


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Return 4-connected components as arrays of (y,x) pixel coordinates."""
    work = mask.copy()
    h, w = work.shape
    components = []
    for sy, sx in np.argwhere(work):
        if not work[sy, sx]:
            continue
        q = deque([(int(sy), int(sx))])
        work[sy, sx] = False
        pixels = []
        while q:
            y, x = q.popleft()
            pixels.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < h and 0 <= nx < w and work[ny, nx]:
                    work[ny, nx] = False
                    q.append((ny, nx))
        components.append(np.asarray(pixels, dtype=np.int32))
    return components


def _marker_centers(arr: np.ndarray) -> dict[str, list[tuple[int, int]]]:
    centers = {}
    for name, rgb in PALETTE.items():
        distance2 = ((arr.astype(np.int32) - rgb.astype(np.int32)) ** 2).sum(axis=2)
        comps = [c for c in _components(distance2 < 35 * 35) if len(c) >= 250]
        if len(comps) != 2:
            raise PixelTopologyError(f"{name}: expected two dot components, found {len(comps)}")
        centers[name] = [
            (int(round(float(comp[:, 1].mean()))), int(round(float(comp[:, 0].mean()))))
            for comp in comps
        ]
    return centers


def _enclosed_by_component(component: np.ndarray, point: tuple[int, int]) -> bool:
    """Flood-fill the complement around one wall component in a local crop."""
    y0 = max(0, int(component[:, 0].min()) - 1)
    y1 = int(component[:, 0].max()) + 2
    x0 = max(0, int(component[:, 1].min()) - 1)
    x1 = int(component[:, 1].max()) + 2
    x, y = point
    if not (x0 <= x < x1 and y0 <= y < y1):
        return False
    barrier = np.zeros((y1 - y0, x1 - x0), dtype=bool)
    barrier[component[:, 0] - y0, component[:, 1] - x0] = True
    open_mask = ~barrier
    reachable = np.zeros_like(open_mask)
    q: deque[tuple[int, int]] = deque()
    h, w = open_mask.shape
    for xx in range(w):
        for yy in (0, h - 1):
            if open_mask[yy, xx] and not reachable[yy, xx]:
                reachable[yy, xx] = True
                q.append((yy, xx))
    for yy in range(h):
        for xx in (0, w - 1):
            if open_mask[yy, xx] and not reachable[yy, xx]:
                reachable[yy, xx] = True
                q.append((yy, xx))
    while q:
        yy, xx = q.popleft()
        for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
            if 0 <= ny < h and 0 <= nx < w and open_mask[ny, nx] and not reachable[ny, nx]:
                reachable[ny, nx] = True
                q.append((ny, nx))
    py, px = y - y0, x - x0
    if barrier[py, px]:
        raise PixelTopologyError("a marker center contacts a wall")
    return not bool(reachable[py, px])


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    dark = (arr[:, :, 0] < 70) & (arr[:, :, 1] < 70) & (arr[:, :, 2] < 70)
    loops = [component for component in _components(dark) if len(component) >= 300]
    if len(loops) != 5:
        raise PixelTopologyError(f"expected five closed wall components, found {len(loops)}")
    centers = _marker_centers(arr)
    counts = {}
    for color, pair in centers.items():
        counts[color] = sum(
            _enclosed_by_component(loop, pair[0]) != _enclosed_by_component(loop, pair[1])
            for loop in loops
        )
    winners = [color for color, count in counts.items() if count == 2]
    if len(winners) != 1 or sorted(counts.values()) != [1, 2, 3]:
        raise PixelTopologyError(f"ambiguous separation counts: {counts}")
    return winners[0]
