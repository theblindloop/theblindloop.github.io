"""Independent pixel-only inverse arm.

Receives only the final PNG. It segments the gray probe circle on top and the
colored candidate circles below by color, measures each radius from its bounding
box, finds the candidate whose radius best matches the probe, and returns that
candidate's color name.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

GRAY = (150, 150, 150)
PALETTE = {
    "red": (215, 60, 60),
    "green": (55, 180, 75),
    "blue": (65, 100, 215),
}
TOL = 70


def _radius_of_mask(mask: np.ndarray) -> float | None:
    ys, xs = np.nonzero(mask)
    if len(xs) == 0:
        return None
    return (float(xs.max() - xs.min()) + 1.0) / 2.0


def _close(image: np.ndarray, color: tuple) -> np.ndarray:
    return np.abs(image.astype(int) - np.array(color)).sum(axis=2) < TOL


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"))
    probe_r = _radius_of_mask(_close(rgb, GRAY))
    if probe_r is None:
        return "unknown"
    best_color = "unknown"
    best_delta = float("inf")
    for name, color in PALETTE.items():
        radius = _radius_of_mask(_close(rgb, color))
        if radius is None:
            continue
        delta = abs(radius - probe_r)
        if delta < best_delta:
            best_delta = delta
            best_color = name
    return best_color
