"""Cross-panel size-correspondence world: match a probe's size to a candidate,
then report the bound color of the matched object."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 620, 380
PROBE_CENTER = (310, 130)
CAND_CENTERS = [(160, 290), (310, 290), (460, 290)]

GRAY = (150, 150, 150)
PALETTE = {
    "red": (215, 60, 60),
    "green": (55, 180, 75),
    "blue": (65, 100, 215),
}
COLOR_NAMES = ["red", "green", "blue"]

RADIUS_POOL = [30, 44, 58, 72, 86, 100]
MIN_GAP = 10.0


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    colors = list(COLOR_NAMES)
    rng.shuffle(colors)
    radii = rng.sample(RADIUS_POOL, 3)
    probe_radius = rng.choice(radii)
    positions = list(CAND_CENTERS)
    rng.shuffle(positions)
    return {
        "probe_radius": probe_radius,
        "candidate_colors": colors,
        "candidate_radii": radii,
        "candidate_positions": positions,
    }


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), (252, 252, 252))
    draw = ImageDraw.Draw(img)
    px, py = PROBE_CENTER
    pr = scene["probe_radius"]
    draw.ellipse([px - pr, py - pr, px + pr, py + pr], fill=GRAY)
    for color_name, radius, (cx, cy) in zip(
        scene["candidate_colors"],
        scene["candidate_radii"],
        scene["candidate_positions"],
    ):
        draw.ellipse([cx - radius, cy - radius, cx + radius, cy + radius], fill=PALETTE[color_name])
    return img


def analytic_gold(scene: dict) -> str:
    pr = scene["probe_radius"]
    radii = scene["candidate_radii"]
    colors = scene["candidate_colors"]
    return colors[radii.index(pr)]


def margin(scene: dict) -> float:
    pr = scene["probe_radius"]
    distances = sorted(abs(pr - radius) for radius in scene["candidate_radii"] if radius != pr)
    return float(distances[0]) if distances else float("inf")


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < MIN_GAP


def latent_symmetries(scene: dict) -> list:
    return []
