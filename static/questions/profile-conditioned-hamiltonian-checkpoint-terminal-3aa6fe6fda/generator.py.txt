"""Deterministic latent sampling, rendering, and analytic graph solution."""

from __future__ import annotations

import random
from collections import defaultdict

from PIL import Image, ImageDraw

SIZE = 420
DARK = (43, 50, 62)
PALETTE = {
    "green": (47, 171, 91),
    "cyan": (48, 190, 211),
    "orange": (238, 137, 43),
    "purple": (151, 85, 205),
    "blue": (54, 113, 211),
}
TERMINAL_COLORS = ("orange", "purple", "blue")

_BASE_POSITIONS = {
    "S": (48, 130),
    "A": (120, 130),
    "B": (220, 130),
    "C": (220, 230),
    "D": (120, 230),
    "E": (320, 280),
    "F": (320, 130),
    "T0": (220, 52),
    "T1": (120, 308),
    "T2": (392, 130),
}
_EDGES = (
    ("S", "A"),
    ("A", "B"),
    ("B", "C"),
    ("C", "D"),
    ("D", "E"),
    ("E", "F"),
    ("A", "D"),
    ("C", "E"),
    ("B", "T0"),
    ("D", "T1"),
    ("F", "T2"),
)


def _transform(point: tuple[int, int], variant: int) -> tuple[int, int]:
    x, y = point
    x -= SIZE // 2
    y -= SIZE // 2
    if variant >= 4:
        x = -x
    for _ in range(variant % 4):
        x, y = -y, x
    return x + SIZE // 2, y + SIZE // 2


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x6A09E667)
    answer_color = TERMINAL_COLORS[int(seed) % 3]
    remaining = [color for color in TERMINAL_COLORS if color != answer_color]
    rng.shuffle(remaining)
    terminal_map = {"T2": answer_color, "T0": remaining[0], "T1": remaining[1]}
    variant = (int(seed) // 3) % 8
    nodes = []
    for node_id, point in _BASE_POSITIONS.items():
        x, y = _transform(point, variant)
        if node_id == "S":
            kind, color = "start", "green"
        elif node_id.startswith("T"):
            kind, color = "terminal", terminal_map[node_id]
        else:
            kind, color = "checkpoint", "cyan"
        nodes.append({"id": node_id, "x": x, "y": y, "kind": kind, "color": color})
    return {"nodes": nodes, "edges": [list(edge) for edge in _EDGES]}


def _coverage_scores(nodes: list[dict], edges: list[list[str]]) -> dict[str, int]:
    by_id = {node["id"]: node for node in nodes}
    adjacency: dict[str, list[str]] = defaultdict(list)
    for left, right in edges:
        adjacency[left].append(right)
        adjacency[right].append(left)
    start = next(node["id"] for node in nodes if node["kind"] == "start")
    checkpoint_ids = {node["id"] for node in nodes if node["kind"] == "checkpoint"}
    terminals = {node["id"]: node["color"] for node in nodes if node["kind"] == "terminal"}
    scores = {color: 0 for color in terminals.values()}

    def walk(current: str, visited: set[str], covered: int) -> None:
        if current in terminals:
            color = terminals[current]
            scores[color] = max(scores[color], covered)
            return
        for neighbor in adjacency[current]:
            if neighbor not in visited:
                walk(
                    neighbor,
                    visited | {neighbor},
                    covered + int(neighbor in checkpoint_ids),
                )

    walk(start, {start}, 0)
    if set(adjacency) != set(by_id):
        return {color: 0 for color in scores}
    return scores


def analytic_gold(scene: dict) -> str:
    scores = _coverage_scores(scene["nodes"], scene["edges"])
    best = max(scores.values())
    winners = sorted(color for color, score in scores.items() if score == best)
    if len(winners) != 1:
        raise ValueError("scene has no unique coverage winner")
    return winners[0]


def margin(scene: dict) -> float:
    scores = sorted(_coverage_scores(scene["nodes"], scene["edges"]).values(), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    scores = _coverage_scores(scene["nodes"], scene["edges"])
    checkpoint_count = sum(node["kind"] == "checkpoint" for node in scene["nodes"])
    ordered = sorted(scores.values(), reverse=True)
    return ordered[0] != checkpoint_count or ordered[0] - ordered[1] < 1


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    nodes = scene["nodes"]
    edges = scene["edges"]
    image = Image.new("RGB", (SIZE, SIZE), (250, 249, 246))
    draw = ImageDraw.Draw(image)
    positions = {node["id"]: (int(node["x"]), int(node["y"])) for node in nodes}
    for left, right in edges:
        draw.line((positions[left], positions[right]), fill=(185, 190, 198), width=13)
        draw.line((positions[left], positions[right]), fill=DARK, width=7)
    for node in nodes:
        x, y = positions[node["id"]]
        fill = PALETTE[node["color"]]
        if node["kind"] == "terminal":
            box = (x - 13, y - 13, x + 13, y + 13)
            draw.rectangle(box, fill=fill, outline=DARK, width=3)
        else:
            radius = 13 if node["kind"] == "start" else 12
            box = (x - radius, y - radius, x + radius, y + radius)
            draw.ellipse(box, fill=fill, outline=DARK, width=3)
            if node["kind"] == "start":
                draw.ellipse((x - 5, y - 5, x + 5, y + 5), outline=(246, 250, 247), width=2)
    return image
