"""Independent PNG-only node/link reconstruction and coverage solver."""

from __future__ import annotations

from collections import defaultdict, deque

import numpy as np
from PIL import Image

COLORS = {
    "green": np.array((47, 171, 91), dtype=np.int16),
    "cyan": np.array((48, 190, 211), dtype=np.int16),
    "orange": np.array((238, 137, 43), dtype=np.int16),
    "purple": np.array((151, 85, 205), dtype=np.int16),
    "blue": np.array((54, 113, 211), dtype=np.int16),
}
DARK = np.array((43, 50, 62), dtype=np.int16)


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        xs, ys = [], []
        while queue:
            yy, xx = queue.popleft()
            xs.append(xx)
            ys.append(yy)
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= 60:
            output.append((float(sum(xs)) / len(xs), float(sum(ys)) / len(ys), len(xs)))
    return output


def _has_link(dark: np.ndarray, first: tuple[float, float], second: tuple[float, float]) -> bool:
    x0, y0 = first
    x1, y1 = second
    length = float(np.hypot(x1 - x0, y1 - y0))
    if length < 45:
        return False
    steps = max(2, int(length) - 28)
    hits = []
    for index in range(steps + 1):
        ratio = (index + 14) / length
        if ratio >= 1.0 - 14.0 / length:
            break
        x = int(round(x0 + ratio * (x1 - x0)))
        y = int(round(y0 + ratio * (y1 - y0)))
        ylo, yhi = max(0, y - 2), min(dark.shape[0], y + 3)
        xlo, xhi = max(0, x - 2), min(dark.shape[1], x + 3)
        hits.append(bool(dark[ylo:yhi, xlo:xhi].any()))
    if not hits:
        return False
    longest_gap = gap = 0
    for hit in hits:
        gap = 0 if hit else gap + 1
        longest_gap = max(longest_gap, gap)
    return sum(hits) / len(hits) > 0.90 and longest_gap <= 6


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    detected = []
    for label, reference in COLORS.items():
        distance = np.max(np.abs(pixels - reference), axis=2)
        for x, y, _area in _components(distance <= 8):
            kind = "start" if label == "green" else ("checkpoint" if label == "cyan" else "terminal")
            detected.append({"key": len(detected), "x": x, "y": y, "kind": kind, "color": label})
    starts = [node for node in detected if node["kind"] == "start"]
    checkpoints = [node for node in detected if node["kind"] == "checkpoint"]
    terminals = [node for node in detected if node["kind"] == "terminal"]
    if len(starts) != 1 or len(checkpoints) != 6 or len(terminals) != 3:
        raise ValueError("could not recover the expected visible node set")

    dark = np.max(np.abs(pixels - DARK), axis=2) <= 10
    adjacency: dict[int, list[int]] = defaultdict(list)
    for left_index, left in enumerate(detected):
        for right in detected[left_index + 1 :]:
            if _has_link(dark, (left["x"], left["y"]), (right["x"], right["y"])):
                adjacency[left["key"]].append(right["key"])
                adjacency[right["key"]].append(left["key"])

    terminal_by_key = {node["key"]: node["color"] for node in terminals}
    checkpoint_keys = {node["key"] for node in checkpoints}
    scores = {node["color"]: 0 for node in terminals}

    def walk(current: int, visited: set[int], covered: int) -> None:
        if current in terminal_by_key:
            color = terminal_by_key[current]
            scores[color] = max(scores[color], covered)
            return
        for neighbor in adjacency[current]:
            if neighbor not in visited:
                walk(neighbor, visited | {neighbor}, covered + int(neighbor in checkpoint_keys))

    start_key = starts[0]["key"]
    walk(start_key, {start_key}, 0)
    best = max(scores.values())
    winners = [color for color, score in scores.items() if score == best]
    if best != len(checkpoint_keys) or len(winners) != 1:
        raise ValueError(f"ambiguous pixel graph coverage: {scores}")
    return winners[0]
