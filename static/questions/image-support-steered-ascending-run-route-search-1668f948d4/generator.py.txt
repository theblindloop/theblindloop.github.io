"""Deterministic latent sampler and renderer for ascending-run route search."""

from __future__ import annotations

import itertools
import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 640
BACKGROUND = [248, 247, 243]
BLACK = (34, 38, 43)
CYAN = (0, 170, 190)
TERMINALS = {
    "red": (214, 58, 63),
    "gold": (222, 159, 24),
    "violet": (132, 82, 190),
    "green": (42, 151, 91),
}
BASE_LENGTHS = (7, 13, 19, 25, 31)


def _score(route: dict[str, Any]) -> float:
    lengths = [float(v) for v in route["lengths"]]
    return min(b - a for a, b in zip(lengths, lengths[1:]))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target_slot = rng.randrange(4)
    target_color = tuple(TERMINALS)[int(seed) % 4]
    remaining_colors = [c for c in TERMINALS if c != target_color]
    rng.shuffle(remaining_colors)
    slot_colors = remaining_colors[:]
    slot_colors.insert(target_slot, target_color)

    distractor_perms = [p for p in itertools.permutations(BASE_LENGTHS) if p != BASE_LENGTHS]
    rng.shuffle(distractor_perms)
    chosen: list[tuple[int, ...]] = []
    for perm in distractor_perms:
        if min(b - a for a, b in zip(perm, perm[1:])) <= -6:
            chosen.append(perm)
        if len(chosen) == 3:
            break

    offset = [rng.randint(178, 244), rng.randint(238, 304)]
    routes = []
    di = 0
    for slot in range(4):
        lengths = BASE_LENGTHS if slot == target_slot else chosen[di]
        di += slot != target_slot
        col, row = slot % 2, slot // 2
        direction = 1 if rng.random() < 0.5 else -1
        vertical_sign = rng.choice((-1, 1))
        # Five shallow runs remain one compact elongated trace. Alternating
        # slopes make every run boundary visible without spreading the support.
        angles = (0.0, vertical_sign * 25.0, 0.0, -vertical_sign * 25.0, 0.0)
        vectors = [
            (direction * length * math.cos(math.radians(angle)), length * math.sin(math.radians(angle)))
            for length, angle in zip(lengths, angles)
        ]
        rel = [(0.0, 0.0)]
        for dx, dy in vectors:
            rel.append((rel[-1][0] + dx, rel[-1][1] + dy))
        center_x = offset[0] + col * 116 + 52
        center_y = offset[1] + row * 56 + 25
        x0 = center_x - (min(x for x, _ in rel) + max(x for x, _ in rel)) / 2
        y0 = center_y - (min(y for _, y in rel) + max(y for _, y in rel)) / 2
        points = [[x0, y0]]
        x, y = x0, y0
        for dx, dy in vectors:
            x += dx
            y += dy
            points.append([x, y])
        routes.append({
            "terminal": slot_colors[slot],
            "lengths": list(lengths),
            "points": points,
            "slot": slot,
        })
    return {
        "routes": routes,
        "offset": offset,
        "canvas": [WIDTH, HEIGHT],
        "background": list(BACKGROUND),
    }


def render(scene: dict) -> Image.Image:
    width, height = [int(v) for v in scene["canvas"]]
    image = Image.new("RGB", (width, height), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    _ = scene["offset"]  # audited rendered nuisance field; positions already include it
    for route in scene["routes"]:
        pts = [(round(x), round(y)) for x, y in route["points"]]
        draw.line(pts, fill=BLACK, width=1, joint="curve")
        sx, sy = pts[0]
        ex, ey = pts[-1]
        r = 4
        draw.ellipse((sx-r, sy-r, sx+r, sy+r), fill=CYAN, outline=(255, 255, 255), width=1)
        color = TERMINALS[route["terminal"]]
        draw.ellipse((ex-r, ey-r, ex+r, ey+r), fill=color, outline=(255, 255, 255), width=1)
    return image


def analytic_gold(scene: dict) -> str:
    scored = []
    for route in scene["routes"]:
        lengths = [float(v) for v in route["lengths"]]
        score = min(b-a for a, b in zip(lengths, lengths[1:]))
        scored.append((score, route["terminal"]))
    scored.sort(reverse=True)
    return scored[0][1]


def margin(scene: dict) -> float:
    scores = sorted((_score(r) for r in scene["routes"]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    scores = sorted((_score(r) for r in scene["routes"]), reverse=True)
    return scores[0] <= 4.0 or scores[0] - scores[1] <= 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
