"""Scene sampler, raster renderer and analytic gold for running_total_crossing.

The picture is a five-bar column chart on a common baseline. Every bar height
is an exact integer multiple of the gridline spacing, so a value read from
pixels equals the value the analytic arm uses. The decision is an order
statistic along the left-to-right reading order: the first bar at which the
running total of the bar values reaches at least half of the grand total.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "running_total_crossing-1.0.0"

WIDTH = 520
HEIGHT = 320
BASE = 280          # baseline row; bars occupy rows BASE - h*ppu .. BASE - 1
TOP = 30            # highest row a gridline may occupy
AXIS_X = 44
SLOT_X = 60
SLOT_STEP = 80
N_BARS = 5

QUARANTINE_UNITS = 2.0

COLOR_NAMES = ("red", "orange", "green", "blue", "purple")
COLOR_RGB = {
    "red": (214, 38, 38),
    "orange": (232, 138, 18),
    "green": (28, 150, 62),
    "blue": (38, 78, 214),
    "purple": (140, 48, 180),
}
GRID_RGB = (172, 172, 172)
INK = (20, 20, 20)


# --- sampling -------------------------------------------------------------


def _heights(rng: random.Random) -> list[int]:
    values = [rng.randint(1, 12) for _ in range(N_BARS)]
    if rng.random() < 0.55:
        boosted = rng.randrange(N_BARS)
        values[boosted] = min(14, values[boosted] * rng.choice((2, 3)))
    return values


def _crossing_index(heights: list[int]) -> int:
    total = sum(heights)
    running = 0
    for index, value in enumerate(heights):
        running += value
        if 2 * running >= total:
            return index
    return len(heights) - 1


def _margin_units(heights: list[int]) -> float:
    """Slack, in gridline units, on both sides of the crossing event.

    ``total - 2*cum_before`` is how much the running total still misses half of
    the grand total just before the crossing bar; ``2*cum_at - total`` is how
    far past half it lands. The smaller of the two is how much a single misread
    unit could move before a different bar becomes the crossing bar.
    """
    total = sum(heights)
    index = _crossing_index(heights)
    cum_at = sum(heights[: index + 1])
    cum_before = cum_at - heights[index]
    return float(min(total - 2 * cum_before, 2 * cum_at - total))


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    target = seed % N_BARS
    want_quarantine = (seed // N_BARS) % 9 == 0

    heights = _heights(rng)
    for _ in range(600):
        candidate = _heights(rng)
        hit_index = _crossing_index(candidate) == target
        quarantined = _margin_units(candidate) < QUARANTINE_UNITS
        if hit_index and quarantined == want_quarantine:
            heights = candidate
            break
        if hit_index and not want_quarantine and not quarantined:
            heights = candidate
            break
    else:
        heights = heights

    order = list(COLOR_NAMES)
    rng.shuffle(order)
    return {
        "heights_units": heights,
        "slot_colors": order,
        "widths_px": [rng.randrange(22, 42, 2) for _ in range(N_BARS)],
        "px_per_unit": rng.choice((13, 15, 17)),
        "x_jitter": [rng.randint(0, 10) for _ in range(N_BARS)],
        "offset": [rng.randint(-6, 6), rng.randint(-5, 5)],
    }


# --- gold -----------------------------------------------------------------


def analytic_gold(scene: dict) -> str:
    return scene["slot_colors"][_crossing_index(list(scene["heights_units"]))]


def margin(scene: dict) -> float:
    return _margin_units(list(scene["heights_units"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_UNITS


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist: see candidate.json latent_alias.mode."""
    return []


def cue_congruence(scene: dict) -> dict:
    """Do the cheap local shortcuts point at the answer?"""
    heights = list(scene["heights_units"])
    widths = list(scene["widths_px"])
    gold = analytic_gold(scene)
    tallest = scene["slot_colors"][max(range(N_BARS), key=lambda i: (heights[i], -i))]
    areas = [heights[i] * widths[i] for i in range(N_BARS)]
    biggest_area = scene["slot_colors"][max(range(N_BARS), key=lambda i: (areas[i], -i))]
    middle = scene["slot_colors"][N_BARS // 2]
    return {
        "tallest_bar": tallest == gold,
        "largest_area": biggest_area == gold,
        "middle_slot": middle == gold,
    }


def totals_units(scene: dict) -> dict:
    heights = list(scene["heights_units"])
    return {
        "grand_total": sum(heights),
        "crossing_index": _crossing_index(heights),
        "running": [sum(heights[: i + 1]) for i in range(N_BARS)],
    }


# --- raster ---------------------------------------------------------------


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (255, 255, 255))
    draw = ImageDraw.Draw(image)
    dx, dy = scene["offset"]
    ppu = int(scene["px_per_unit"])
    base = BASE + dy
    axis_x = AXIS_X + dx

    level = 1
    while base - level * ppu >= TOP + dy:
        y = base - level * ppu
        draw.rectangle([axis_x + 1, y, WIDTH - 12 + dx, y], fill=GRID_RGB)
        level += 1

    draw.rectangle([axis_x, TOP + dy - 6, axis_x, base], fill=INK)
    draw.rectangle([axis_x, base, WIDTH - 12 + dx, base], fill=INK)

    for index in range(N_BARS):
        height = int(scene["heights_units"][index]) * ppu
        left = SLOT_X + dx + index * SLOT_STEP + int(scene["x_jitter"][index])
        right = left + int(scene["widths_px"][index]) - 1
        top = base - height
        draw.rectangle([left, top, right, base - 1], fill=COLOR_RGB[scene["slot_colors"][index]])
    return image
