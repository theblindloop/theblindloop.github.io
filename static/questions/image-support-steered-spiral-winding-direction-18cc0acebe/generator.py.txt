"""Deterministic Pillow world for directed winding along a visible spiral."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 800, 400
BACKGROUND = (250, 248, 242)
NAVY = (31, 48, 71)
GREEN = (49, 166, 82)
RED = (211, 61, 64)
YELLOW = (236, 184, 39)


def sample_scene(seed: int) -> dict:
    """Pair consecutive seeds: identical anchors, opposite winding signs."""
    seed = int(seed)
    rng = random.Random(seed // 2)
    center = [400 + rng.randint(-4, 4), 200 + rng.randint(-4, 4)]
    # Horizontal anchor rays keep the ablation support elongated; the paired
    # answer classes still share the exact same randomly selected side.
    start_angle = 0.0 if rng.randrange(2) == 0 else math.pi
    outer_radius = rng.randint(365, 370)
    inner_radius = rng.randint(270, 280)
    return {
        "center": center,
        "start_angle": start_angle,
        "direction": "clockwise" if seed % 2 == 0 else "counterclockwise",
        "outer_radius": outer_radius,
        "inner_radius": inner_radius,
        "vertical_scale": rng.uniform(0.20, 0.215),
        "samples": 241,
    }


def _route_points(scene: dict) -> list[tuple[float, float]]:
    cx, cy = scene["center"]
    sign = 1.0 if scene["direction"] == "clockwise" else -1.0
    count = int(scene["samples"])
    points = []
    for index in range(count):
        fraction = index / (count - 1)
        radius = scene["outer_radius"] + fraction * (scene["inner_radius"] - scene["outer_radius"])
        # Image coordinates have y downward, so increasing polar angle is clockwise.
        angle = scene["start_angle"] + sign * 2.0 * math.pi * fraction
        points.append((cx + radius * math.cos(angle), cy + scene["vertical_scale"] * radius * math.sin(angle)))
    return points


def _ring(draw: ImageDraw.ImageDraw, center: tuple[float, float], color: tuple[int, int, int], radius: int, width: int) -> None:
    x, y = center
    draw.ellipse((round(x-radius), round(y-radius), round(x+radius), round(y+radius)), outline=color, width=width)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((10, 10, WIDTH-11, HEIGHT-11), radius=12, outline=(218, 215, 206), width=2)
    points = _route_points(scene)
    draw.line(points, fill=NAVY, width=7, joint="curve")
    cx, cy = scene["center"]
    draw.ellipse((cx-17, cy-17, cx+17, cy+17), fill=YELLOW, outline=NAVY, width=2)
    _ring(draw, points[0], GREEN, 13, 6)
    _ring(draw, points[-1], RED, 13, 6)
    return image


def analytic_gold(scene: dict) -> str:
    return scene["direction"]


def margin(scene: dict) -> float:
    # Degrees beyond the zero-winding boundary; all scenes make one full turn.
    return 360.0


def is_quarantined(scene: dict) -> bool:
    return not (
        scene.get("direction") in {"clockwise", "counterclockwise"}
        and int(scene.get("samples", 0)) >= 121
        and 360 <= float(scene.get("outer_radius", 0)) <= 375
        and 260 <= float(scene.get("inner_radius", 0)) <= 285
        and 0.19 <= float(scene.get("vertical_scale", 0)) <= 0.225
        and float(scene["outer_radius"]) - float(scene["inner_radius"]) >= 80
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
