"""Independent image-only inverse for directed winding around a visible post."""

from __future__ import annotations

from collections import deque

import numpy as np

NAVY = np.array((31, 48, 71), dtype=np.int16)
GREEN = np.array((49, 166, 82), dtype=np.int16)
RED = np.array((211, 61, 64), dtype=np.int16)
YELLOW = np.array((236, 184, 39), dtype=np.int16)


def _mask_near(array: np.ndarray, color: np.ndarray, tolerance: int = 12) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _centroid(mask: np.ndarray, minimum: int) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if len(xs) < minimum:
        raise ValueError("required colored marker is absent")
    return float(xs.mean()), float(ys.mean())


def _dilate(mask: np.ndarray, rounds: int) -> np.ndarray:
    expanded = mask.copy()
    for _ in range(rounds):
        prior = expanded
        expanded = prior.copy()
        expanded[1:] |= prior[:-1]
        expanded[:-1] |= prior[1:]
        expanded[:, 1:] |= prior[:, :-1]
        expanded[:, :-1] |= prior[:, 1:]
    return expanded


def _distance_map(route: np.ndarray, start: tuple[float, float]) -> np.ndarray:
    sy, sx = int(round(start[1])), int(round(start[0]))
    pixels = np.argwhere(route)
    nearby = pixels[(pixels[:, 0]-sy)**2 + (pixels[:, 1]-sx)**2 <= 24**2]
    if not len(nearby):
        raise ValueError("start is detached from navy route")
    # The colored endpoint ring occludes a short terminal segment. Start from
    # the far side of that ring, which belongs to the long main route rather
    # than the tiny navy stub left inside the annulus.
    origin = nearby[np.argmax((nearby[:, 0]-sy)**2 + (nearby[:, 1]-sx)**2)]
    distances = np.full(route.shape, -1, dtype=np.int32)
    oy, ox = map(int, origin)
    distances[oy, ox] = 0
    queue = deque([(oy, ox)])
    while queue:
        y, x = queue.popleft()
        nd = int(distances[y, x]) + 1
        for ny, nx in ((y-1,x),(y+1,x),(y,x-1),(y,x+1)):
            if 0 <= ny < route.shape[0] and 0 <= nx < route.shape[1] and route[ny,nx] and distances[ny,nx] < 0:
                distances[ny,nx] = nd
                queue.append((ny,nx))
    return distances


def _near_distance(distances: np.ndarray, point: tuple[float, float], radius: int = 22) -> int:
    yy, xx = np.ogrid[:distances.shape[0], :distances.shape[1]]
    region = (xx-point[0])**2 + (yy-point[1])**2 <= radius**2
    values = distances[(distances >= 0) & region]
    if not len(values):
        raise ValueError("finish is detached from recovered route")
    return int(values.max())


def decision_from_image(image) -> str:
    array = np.asarray(image.convert("RGB"))
    start = _centroid(_mask_near(array, GREEN), 60)
    finish = _centroid(_mask_near(array, RED), 60)
    post = _centroid(_mask_near(array, YELLOW), 300)
    navy = _mask_near(array, NAVY)
    route = _dilate(navy, 2)
    distances = _distance_map(route, start)
    finish_distance = _near_distance(distances, finish)
    if finish_distance < 600:
        raise ValueError("recovered route is too short for a winding decision")

    # Use only true navy pixels, ordered by scene-free geodesic distance.
    ys, xs = np.nonzero(navy & (distances >= 0) & (distances <= finish_distance + 20))
    ds = distances[ys, xs]
    observations: list[tuple[float, float]] = []
    for lo in range(0, finish_distance + 1, 18):
        select = (ds >= lo) & (ds < lo + 18)
        if int(select.sum()) < 5:
            continue
        x = float(np.median(xs[select]))
        y = float(np.median(ys[select]))
        radius = float(np.hypot(x-post[0], y-post[1]))
        if radius > 45:
            observations.append((lo + 9.0, float(np.arctan2(y-post[1], x-post[0]))))
    if len(observations) < 28:
        raise ValueError("too little ordered route support")
    angles = np.unwrap(np.array([angle for _, angle in observations]))
    total = float(angles[-1] - angles[0])
    if abs(total) < 4.5:
        raise ValueError("angular winding margin is ambiguous")
    # In image coordinates positive angular change is clockwise.
    return "clockwise" if total > 0 else "counterclockwise"
