"""Render a dense, broad, single-component route network."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 720
COLS, ROWS, CELL = 11, 17, 36
X0, Y0 = 122, 54
BACKGROUND = (248, 246, 238)
NAVY = (31, 55, 82)
NAVY_EDGE = (18, 35, 55)
GREEN = (34, 166, 103)
RED = (220, 74, 64)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    branch_count = 2 + (int(seed) & 1)
    # Equalize added route area: both classes add exactly four grid cells.
    if branch_count == 2:
        lengths = [2, 2]
        gaps = [0, 3]
    else:
        lengths = [2, 1, 1]
        gaps = [0, 2, 3]
        rng.shuffle(lengths)
    columns = [2, 5, 8]
    rng.shuffle(columns)
    specs = [
        {"lane": int(gap), "column": int(columns[i]), "length": int(length)}
        for i, (gap, length) in enumerate(zip(gaps, lengths))
    ]
    rng.shuffle(specs)
    return {
        "seed": int(seed),
        "branch_count": branch_count,
        "branch_specs": specs,
        "mirror": bool(rng.randrange(2)),
    }


def _occupied(scene: dict) -> set[tuple[int, int]]:
    expected = int(scene["branch_count"])
    specs = list(scene["branch_specs"])
    if len(specs) != expected:
        raise ValueError("branch specification count disagrees with branch_count")
    cells: set[tuple[int, int]] = set()
    for row in (0, 4, 8, 12, 16):
        for col in range(COLS):
            cells.add((row, col))
    for gap in range(4):
        col = 10 if gap % 2 == 0 else 0
        for row in range(gap * 4 + 1, gap * 4 + 4):
            cells.add((row, col))
    for spec in specs:
        lane = int(spec["lane"])
        col = int(spec["column"])
        length = int(spec["length"])
        if lane not in range(4) or col not in (2, 5, 8) or length not in (1, 2):
            raise ValueError("invalid branch specification")
        for step in range(1, length + 1):
            cells.add((lane * 4 + step, col))
    if scene["mirror"]:
        cells = {(row, COLS - 1 - col) for row, col in cells}
    return cells


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    cells = _occupied(scene)
    for row, col in cells:
        x0, y0 = X0 + col * CELL, Y0 + row * CELL
        draw.rectangle((x0, y0, x0 + CELL - 1, y0 + CELL - 1), fill=NAVY)
    # A subtle outline is entirely within the route and preserves connectivity.
    for row, col in cells:
        x0, y0 = X0 + col * CELL, Y0 + row * CELL
        if (row - 1, col) not in cells:
            draw.line((x0, y0, x0 + CELL - 1, y0), fill=NAVY_EDGE, width=2)
        if (row + 1, col) not in cells:
            draw.line((x0, y0 + CELL - 1, x0 + CELL - 1, y0 + CELL - 1), fill=NAVY_EDGE, width=2)
        if (row, col - 1) not in cells:
            draw.line((x0, y0, x0, y0 + CELL - 1), fill=NAVY_EDGE, width=2)
        if (row, col + 1) not in cells:
            draw.line((x0 + CELL - 1, y0, x0 + CELL - 1, y0 + CELL - 1), fill=NAVY_EDGE, width=2)
    mirrored = bool(scene["mirror"])
    start_col = 10 if mirrored else 0
    finish_col = 10 if mirrored else 0
    for row, col, color in ((0, start_col, GREEN), (16, finish_col, RED)):
        cx = X0 + col * CELL + CELL // 2
        cy = Y0 + row * CELL + CELL // 2
        draw.ellipse((cx - 13, cy - 13, cx + 13, cy + 13), fill=color, outline=BACKGROUND, width=3)
        draw.ellipse((cx - 4, cy - 4, cx + 4, cy + 4), fill=BACKGROUND)
    return image


def analytic_gold(scene: dict) -> str:
    return str(int(scene["branch_count"]))


def margin(scene: dict) -> float:
    # The nearest alternative is one topological branch away.
    return 1.0


def is_quarantined(scene: dict) -> bool:
    try:
        cells = _occupied(scene)
    except (KeyError, TypeError, ValueError):
        return True
    return len(cells) != 71


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["branch_specs"] = list(reversed(twin["branch_specs"]))
    return [("reverse_branch_storage_order", twin)]
