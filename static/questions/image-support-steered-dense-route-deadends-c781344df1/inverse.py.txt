"""Independent image-only inverse for the dense route topology."""

from __future__ import annotations

import numpy as np


def _component_count(cells: set[tuple[int, int]]) -> int:
    remaining = set(cells)
    count = 0
    while remaining:
        count += 1
        stack = [remaining.pop()]
        while stack:
            row, col = stack.pop()
            for point in ((row-1, col), (row+1, col), (row, col-1), (row, col+1)):
                if point in remaining:
                    remaining.remove(point)
                    stack.append(point)
    return count


def decision_from_image(image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    if rgb.ndim != 3 or rgb.shape[2] != 3:
        raise ValueError("RGB image required")
    border = np.concatenate((rgb[:4].reshape(-1, 3), rgb[-4:].reshape(-1, 3),
                             rgb[:, :4].reshape(-1, 3), rgb[:, -4:].reshape(-1, 3)))
    background = np.median(border, axis=0)
    delta = np.linalg.norm(rgb - background, axis=2)
    foreground = delta > 45.0
    points = np.argwhere(foreground)
    if len(points) < 40000:
        raise ValueError("route region is missing or damaged")
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0) + 1
    height, width = int(y1-y0), int(x1-x0)
    if not (590 <= height <= 620 and 380 <= width <= 405):
        raise ValueError("route bounding box is invalid")
    occupied: set[tuple[int, int]] = set()
    for row in range(17):
        ya = int(round(y0 + row * height / 17))
        yb = int(round(y0 + (row + 1) * height / 17))
        for col in range(11):
            xa = int(round(x0 + col * width / 11))
            xb = int(round(x0 + (col + 1) * width / 11))
            if float(foreground[ya:yb, xa:xb].mean()) > 0.42:
                occupied.add((row, col))
    # Recover the five complete transverse lanes without assuming reflection.
    for row in (0, 4, 8, 12, 16):
        if any((row, col) not in occupied for col in range(11)):
            raise ValueError("a broad route lane is incomplete")
    left_connectors = sum(all((row, 0) in occupied for row in range(g*4+1, g*4+4)) for g in range(4))
    right_connectors = sum(all((row, 10) in occupied for row in range(g*4+1, g*4+4)) for g in range(4))
    if sorted((left_connectors, right_connectors)) != [2, 2]:
        raise ValueError("serpentine connectors are invalid")
    if len(occupied) != 71 or _component_count(occupied) != 1:
        raise ValueError("route topology or equal-ink invariant is invalid")
    endpoints = 0
    branch_junctions = 0
    for row, col in occupied:
        degree = sum(point in occupied for point in ((row-1,col),(row+1,col),(row,col-1),(row,col+1)))
        if degree == 1:
            endpoints += 1
        elif degree == 3:
            branch_junctions += 1
        elif degree > 3:
            raise ValueError("ambiguous high-degree junction")
    branches = endpoints - 2
    if branches not in (2, 3) or branch_junctions != branches:
        raise ValueError("dead-end branch count is ambiguous")
    return str(branches)
