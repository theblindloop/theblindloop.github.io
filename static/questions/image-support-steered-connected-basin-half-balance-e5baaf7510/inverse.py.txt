"""Independent image-only inverse for connected-basin half-area balance."""

from __future__ import annotations

import numpy as np
from PIL import Image

MID_X = 256
MIN_FOREGROUND = 500
MAX_CERTIFIED_MARGIN = 2600


def _border_background(array: np.ndarray) -> np.ndarray:
    border = np.concatenate(
        (
            array[:3].reshape(-1, 3),
            array[-3:].reshape(-1, 3),
            array[:, :3].reshape(-1, 3),
            array[:, -3:].reshape(-1, 3),
        ),
        axis=0,
    )
    return np.median(border, axis=0).astype(np.float32)


def decision_from_image(image: Image.Image) -> str:
    """Segment foreground from the PNG and compare its two half-plane areas."""

    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    background = _border_background(array)
    foreground = np.linalg.norm(array - background, axis=2) > 24.0
    if int(foreground.sum()) < MIN_FOREGROUND:
        raise ValueError("foreground basin is absent or too small")
    left = int(foreground[:, :MID_X].sum())
    right = int(foreground[:, MID_X:].sum())
    if left == right:
        raise ValueError("foreground areas form an unresolved tie")
    if abs(right - left) > MAX_CERTIFIED_MARGIN:
        raise ValueError("foreground balance is outside the certified near-balance band")
    return "right" if right > left else "left"
