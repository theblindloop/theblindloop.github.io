"""Renderer and analytic arm for a connected filled-basin measurement world.

The only label-bearing latent fields are four integer shoulder dimensions.  A
thick bridge makes the entire foreground one connected component.  Rendering
uses half-open integer rectangles, so the analytic area arithmetic and the
pixel oracle have the same discrete support.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
MID_X = 256
# Reach into the outer ablation cells: distant left and right support must both
# be capable of changing the near-balanced area decision.
BASE_X0 = 16
BASE_X1 = 496
BASE_Y0 = 330
BASE_Y1 = 480
SHOULDER_Y1 = 350
MIN_MARGIN = 300
MAX_CERTIFIED_MARGIN = 2600
MIN_HEIGHT = 260
MAX_HEIGHT = 320

INKS = ("#17324d", "#1e5661", "#3f2f6c", "#28553d")
BACKGROUNDS = ("#fbfcf8", "#f7f9fc", "#fffaf2")


def _rect(draw: ImageDraw.ImageDraw, x0: int, y0: int, x1: int, y1: int, fill: str) -> None:
    """Draw a half-open integer rectangle with exact pixel area."""

    draw.rectangle((x0, y0, x1 - 1, y1 - 1), fill=fill)


def _side_areas(scene: dict[str, Any]) -> tuple[int, int]:
    """Return exact discrete foreground pixel counts on the two page halves."""

    base_side = (MID_X - BASE_X0) * (BASE_Y1 - BASE_Y0)
    left = base_side + scene["left_width"] * (scene["left_height"] - (SHOULDER_Y1 - BASE_Y0))
    right = base_side + scene["right_width"] * (scene["right_height"] - (SHOULDER_Y1 - BASE_Y0))
    return int(left), int(right)


def _candidate_geometry(rng: random.Random, desired_sign: int) -> tuple[int, int, int, int]:
    """Sample varied widths/heights while keeping the balance close and nonzero."""

    for _ in range(20_000):
        left_width = rng.randint(176, 212)
        right_width = rng.randint(176, 212)
        left_height = rng.randint(MIN_HEIGHT, MAX_HEIGHT)
        overlap = SHOULDER_Y1 - BASE_Y0
        left_extra = left_width * (left_height - overlap)
        requested = rng.randint(450, 1600)
        desired_right_extra = left_extra + desired_sign * requested
        right_height = round(desired_right_extra / right_width) + overlap
        if not MIN_HEIGHT <= right_height <= MAX_HEIGHT:
            continue
        right_extra = right_width * (right_height - overlap)
        difference = right_extra - left_extra
        if 380 <= abs(difference) <= 2100 and (1 if difference > 0 else -1) == desired_sign:
            return left_width, right_width, left_height, right_height

    # This branch is deterministic insurance for an unusually unlucky seed.
    for left_width in range(176, 213):
        for right_width in range(176, 213):
            for left_height in range(MIN_HEIGHT, MAX_HEIGHT + 1):
                left_extra = left_width * (left_height - (SHOULDER_Y1 - BASE_Y0))
                for right_height in range(MIN_HEIGHT, MAX_HEIGHT + 1):
                    difference = right_width * (right_height - (SHOULDER_Y1 - BASE_Y0)) - left_extra
                    if 380 <= abs(difference) <= 2100 and (1 if difference > 0 else -1) == desired_sign:
                        return left_width, right_width, left_height, right_height
    raise RuntimeError("could not sample a balanced connected basin")


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced scene with randomized presentation colors."""

    rng = random.Random(int(seed))
    desired_sign = 1 if rng.randrange(2) else -1
    left_width, right_width, left_height, right_height = _candidate_geometry(rng, desired_sign)
    return {
        "left_width": left_width,
        "right_width": right_width,
        "left_height": left_height,
        "right_height": right_height,
        "ink": rng.choice(INKS),
        "background": rng.choice(BACKGROUNDS),
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render one connected, high-fill basin as a PNG-ready RGB image."""

    image = Image.new("RGB", (WIDTH, HEIGHT), scene["background"])
    draw = ImageDraw.Draw(image)
    ink = scene["ink"]

    # The bridge is deliberately broad.  Both shoulders overlap it by 20 px,
    # making one connected foreground component without a thin route cue.
    _rect(draw, BASE_X0, BASE_Y0, BASE_X1, BASE_Y1, ink)
    _rect(
        draw,
        BASE_X0,
        SHOULDER_Y1 - scene["left_height"],
        BASE_X0 + scene["left_width"],
        SHOULDER_Y1,
        ink,
    )
    _rect(
        draw,
        BASE_X1 - scene["right_width"],
        SHOULDER_Y1 - scene["right_height"],
        BASE_X1,
        SHOULDER_Y1,
        ink,
    )
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the side with greater visible foreground area."""

    left, right = _side_areas(scene)
    return "right" if right > left else "left"


def margin(scene: dict[str, Any]) -> float:
    """Continuous signed margin: right foreground pixels minus left pixels."""

    left, right = _side_areas(scene)
    return float(right - left)


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Mark ties and out-of-band balances as outside the certified oracle band."""

    value = abs(margin(scene))
    return value <= MIN_MARGIN or value > MAX_CERTIFIED_MARGIN


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """There are no nontrivial latent aliases in this injective geometry."""

    return []
