from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image


WIDTH = 256
HEIGHT = 256
SUPERSAMPLE = 4
PALETTES = (
    ("#f6f1e7", "#176b87"),
    ("#f3f0f7", "#70428f"),
    ("#f8f1e8", "#a34d2d"),
    ("#edf5ef", "#267259"),
    ("#f4f3ed", "#405b79"),
)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    label = "aligned" if int(seed) % 2 == 0 else "alternating"
    outer_radius = rng.uniform(77.0, 84.0)
    inner_radius = rng.uniform(34.0, 39.0)
    outer_amp = rng.uniform(10.0, 13.0)
    inner_amp = rng.uniform(7.5, 10.0)
    rotation = rng.uniform(0.0, math.pi / 2.0)
    clearance = (outer_radius - outer_amp) - (inner_radius + inner_amp)
    if clearance < 15.0:
        raise RuntimeError("sampled annulus clearance is too small")
    background, ink = PALETTES[rng.randrange(len(PALETTES))]
    return {
        "label": label,
        "cx": rng.uniform(119.0, 137.0),
        "cy": rng.uniform(119.0, 137.0),
        "outer_radius": outer_radius,
        "inner_radius": inner_radius,
        "outer_amp": outer_amp,
        "inner_amp": inner_amp,
        "rotation": rotation,
        "ink": ink,
        "background": background,
    }


def render(scene: dict) -> Image.Image:
    scale = SUPERSAMPLE
    height, width = HEIGHT * scale, WIDTH * scale
    yy, xx = np.indices((height, width), dtype=np.float64)
    x = (xx + 0.5) / scale
    y = (yy + 0.5) / scale
    dx = x - float(scene["cx"])
    dy = y - float(scene["cy"])
    radius = np.hypot(dx, dy)
    theta = np.arctan2(dy, dx)
    rotation = float(scene["rotation"])
    outer = float(scene["outer_radius"]) + float(scene["outer_amp"]) * np.cos(
        4.0 * (theta - rotation)
    )
    phase_offset = 0.0 if scene["label"] == "aligned" else math.pi / 4.0
    inner = float(scene["inner_radius"]) + float(scene["inner_amp"]) * np.cos(
        4.0 * (theta - rotation - phase_offset)
    )
    mask = (radius <= outer) & (radius >= inner)
    background = np.asarray(Image.new("RGB", (1, 1), scene["background"]))[0, 0]
    ink = np.asarray(Image.new("RGB", (1, 1), scene["ink"]))[0, 0]
    array = np.empty((height, width, 3), dtype=np.uint8)
    array[:] = background
    array[mask] = ink
    image = Image.fromarray(array, mode="RGB")
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    label = scene["label"]
    if label not in ("aligned", "alternating"):
        raise ValueError("unsupported contour phase label")
    return label


def margin(scene: dict) -> float:
    clearance = (
        float(scene["outer_radius"])
        - float(scene["outer_amp"])
        - float(scene["inner_radius"])
        - float(scene["inner_amp"])
    )
    return min(float(scene["outer_amp"]), float(scene["inner_amp"]), clearance)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
