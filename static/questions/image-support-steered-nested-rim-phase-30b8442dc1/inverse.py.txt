from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _largest_component(mask: np.ndarray) -> tuple[np.ndarray, int]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    largest: list[tuple[int, int]] = []
    count = 0
    for y, x in np.argwhere(mask):
        yi, xi = int(y), int(x)
        if seen[yi, xi]:
            continue
        count += 1
        seen[yi, xi] = True
        queue = deque([(yi, xi)])
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) > len(largest):
            largest = points
    output = np.zeros_like(mask, dtype=bool)
    if largest:
        yy, xx = zip(*largest)
        output[np.asarray(yy), np.asarray(xx)] = True
    return output, count


def _radial_profiles(mask: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 9000:
        raise ValueError("the connected annular evidence is incomplete")
    cx = float(xs.mean())
    cy = float(ys.mean())
    width = float(xs.max() - xs.min() + 1)
    height = float(ys.max() - ys.min() + 1)
    if not (142.0 <= width <= 205.0 and 142.0 <= height <= 205.0):
        raise ValueError("annulus extent is outside the declared world")
    dx = xs.astype(np.float64) - cx
    dy = ys.astype(np.float64) - cy
    radii = np.hypot(dx, dy)
    angles = np.mod(np.arctan2(dy, dx), 2.0 * np.pi)
    bins = 180
    indices = np.floor(angles * bins / (2.0 * np.pi)).astype(int) % bins
    inner = np.full(bins, np.inf, dtype=np.float64)
    outer = np.full(bins, -np.inf, dtype=np.float64)
    np.minimum.at(inner, indices, radii)
    np.maximum.at(outer, indices, radii)
    if not (np.isfinite(inner).all() and np.isfinite(outer).all()):
        raise ValueError("annulus does not cover every direction")
    if float(np.min(outer - inner)) < 8.0:
        raise ValueError("annular band is broken or too thin")
    return inner, outer


def _fourfold_fit(profile: np.ndarray) -> tuple[np.ndarray, float, float]:
    count = len(profile)
    theta = (np.arange(count, dtype=np.float64) + 0.5) * (2.0 * np.pi / count)
    design = np.column_stack(
        [np.ones(count), np.cos(4.0 * theta), np.sin(4.0 * theta)]
    )
    coefficients, *_ = np.linalg.lstsq(design, profile, rcond=None)
    fitted = design @ coefficients
    amplitude = float(np.hypot(coefficients[1], coefficients[2]))
    residual = float(np.sqrt(np.mean((profile - fitted) ** 2)))
    return coefficients[1:3], amplitude, residual


def _measure(image: Image.Image) -> float:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    border = np.concatenate(
        [
            array[:3].reshape(-1, 3),
            array[-3:].reshape(-1, 3),
            array[:, :3].reshape(-1, 3),
            array[:, -3:].reshape(-1, 3),
        ],
        axis=0,
    )
    background = np.median(border, axis=0)
    raw = np.linalg.norm(array - background, axis=2) > 42.0
    mask, component_count = _largest_component(raw)
    if component_count != 1:
        raise ValueError("expected exactly one colored component")
    inner, outer = _radial_profiles(mask)
    inner_vector, inner_amp, inner_residual = _fourfold_fit(inner)
    outer_vector, outer_amp, outer_residual = _fourfold_fit(outer)
    if min(inner_amp, outer_amp) < 5.5:
        raise ValueError("one rim lacks a recoverable fourfold contour")
    if max(inner_residual, outer_residual) > 3.0:
        raise ValueError("a rim is not a complete smooth fourfold contour")
    score = float(np.dot(inner_vector, outer_vector) / (inner_amp * outer_amp))
    if abs(score) < 0.55:
        raise ValueError("nested contour phase is ambiguous")
    return score


def decision_from_image(image: Image.Image) -> str:
    return "aligned" if _measure(image) > 0.0 else "alternating"
