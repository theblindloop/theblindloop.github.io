"""Independent image-only inverse arm for vector-tab scatter."""

from __future__ import annotations

import numpy as np
from PIL import Image

GRID = 12
SHIFT = 4


def _templates() -> dict[str, set[tuple[int, int]]]:
    a = {(r, c) for r in range(4, 10) for c in (3, 8)} | {(3, c) for c in range(4, 8)} | {(4, c) for c in (4, 7)} | {(10, c) for c in (2, 3, 4, 7, 8, 9)}
    u = {(r, c) for r in range(3, 9) for c in (3, 8)} | {(9, c) for c in range(4, 8)} | {(2, c) for c in (2, 3, 8, 9)} | {(10, c) for c in range(4, 8)}
    f = {(r, c) for r in range(5, 11) for c in (5, 6)} | {(r, c) for r in range(2, 5) for c in (2, 3, 8, 9)} | {(4, c) for c in (4, 7)}
    f -= {(2, 3), (2, 8)}
    b = {(2, c) for c in range(4, 9)} | {(3, c) for c in range(3, 8)} | {(4, c) for c in range(4, 7)} | {(5, c) for c in range(3, 6)} | {(6, c) for c in range(2, 5)} | {(7, c) for c in range(3, 6)} | {(8, c) for c in (2, 3)}
    return {"arch": a, "cup": u, "fork": f, "bolt": b}


REFERENCE = _templates()


def _measure(image: Image.Image) -> tuple[set[tuple[int, int]], float]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    cyan = (arr[:, :, 0] < 90) & (arr[:, :, 1] > 105) & (arr[:, :, 2] > 120)
    ys, xs = np.where(cyan)
    if len(xs) < 500:
        raise ValueError("cyan frame missing")
    x0, x1, y0, y1 = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    span_x, span_y = x1 - x0, y1 - y0
    if abs(span_x - span_y) > 2 or span_x % GRID != 0:
        raise ValueError("ambiguous grid frame")
    cell = span_x // GRID
    if cell < 20:
        raise ValueError("grid cells too small")
    dark = (arr[:, :, 0] < 80) & (arr[:, :, 1] < 80) & (arr[:, :, 2] < 80)
    magenta = (arr[:, :, 0] > 150) & (arr[:, :, 1] < 100) & (arr[:, :, 2] > 70)
    moved: list[tuple[int, int]] = []
    for r in range(GRID):
        for c in range(GRID):
            xa, xb = x0 + c * cell, x0 + (c + 1) * cell
            ya, yb = y0 + r * cell, y0 + (r + 1) * cell
            q = max(3, cell // 5)
            center = dark[ya + q : yb - q + 1, xa + q : xb - q + 1]
            if int(center.sum()) < 30:
                continue
            yy, xx = np.where(magenta[ya + 2 : yb - 1, xa + 2 : xb - 1])
            if len(xx) < 12:
                raise ValueError("token direction tab missing")
            dx = float(xx.mean() + 2) - cell / 2.0
            dy = float(yy.mean() + 2) - cell / 2.0
            if abs(dx) > abs(dy):
                dr, dc = (0, SHIFT) if dx > 0 else (0, -SHIFT)
            else:
                dr, dc = (SHIFT, 0) if dy > 0 else (-SHIFT, 0)
            moved.append(((r + dr) % GRID, (c + dc) % GRID))
    if len(moved) != 24 or len(set(moved)) != len(moved):
        raise ValueError("token count or destination collision is invalid")
    output = set(moved)
    scores = {name: -len(output.symmetric_difference(mask)) for name, mask in REFERENCE.items()}
    ranked = sorted(scores.items(), key=lambda item: item[1], reverse=True)
    gap = float(ranked[0][1] - ranked[1][1])
    if gap < 4.0:
        raise ValueError("ambiguous reconstructed symbol")
    return output, gap


def decision_from_image(image: Image.Image) -> str:
    output, _ = _measure(image)
    return min(REFERENCE, key=lambda name: len(output.symmetric_difference(REFERENCE[name])))


def decision_with_margin(image: Image.Image) -> tuple[str, float]:
    output, gap = _measure(image)
    decision = min(REFERENCE, key=lambda name: len(output.symmetric_difference(REFERENCE[name])))
    return decision, gap
