"""Deterministic renderer and analytic arm for vector-tab scatter."""

from __future__ import annotations

import copy
import random
from typing import Iterable

from PIL import Image, ImageDraw

GRID = 12
SHIFT = 4
DIRECTIONS = ("up", "right", "down", "left")
VECTORS = {"up": (-SHIFT, 0), "right": (0, SHIFT), "down": (SHIFT, 0), "left": (0, -SHIFT)}


def _templates() -> dict[str, set[tuple[int, int]]]:
    arch = (
        {(r, c) for r in range(4, 10) for c in (3, 8)}
        | {(3, c) for c in range(4, 8)}
        | {(4, c) for c in (4, 7)}
        | {(10, c) for c in (2, 3, 4, 7, 8, 9)}
    )
    cup = (
        {(r, c) for r in range(3, 9) for c in (3, 8)}
        | {(9, c) for c in range(4, 8)}
        | {(2, c) for c in (2, 3, 8, 9)}
        | {(10, c) for c in range(4, 8)}
    )
    fork = (
        {(r, c) for r in range(5, 11) for c in (5, 6)}
        | {(r, c) for r in range(2, 5) for c in (2, 3, 8, 9)}
        | {(4, c) for c in (4, 7)}
    )
    fork.remove((2, 3))
    fork.remove((2, 8))
    bolt = (
        {(2, c) for c in range(4, 9)}
        | {(3, c) for c in range(3, 8)}
        | {(4, c) for c in range(4, 7)}
        | {(5, c) for c in range(3, 6)}
        | {(6, c) for c in range(2, 5)}
        | {(7, c) for c in range(3, 6)}
        | {(8, c) for c in (2, 3)}
    )
    result = {"arch": arch, "cup": cup, "fork": fork, "bolt": bolt}
    assert {len(value) for value in result.values()} == {24}
    return result


TEMPLATES = _templates()
SYMBOLS = tuple(TEMPLATES)
PALETTES = (
    {"background": [249, 248, 244], "frame": [16, 154, 184], "grid": [207, 214, 217], "token": [27, 31, 39], "tab": [220, 35, 116], "distractor": [168, 175, 180]},
    {"background": [246, 249, 247], "frame": [22, 142, 176], "grid": [202, 212, 214], "token": [35, 32, 42], "tab": [207, 29, 128], "distractor": [174, 177, 172]},
    {"background": [250, 247, 246], "frame": [12, 166, 171], "grid": [210, 211, 220], "token": [30, 36, 37], "tab": [228, 42, 105], "distractor": [163, 172, 182]},
)


def _move(row: int, col: int, direction: str) -> tuple[int, int]:
    dr, dc = VECTORS[direction]
    return ((int(row) + dr) % GRID, (int(col) + dc) % GRID)


def _reconstructed(tokens: Iterable[dict]) -> set[tuple[int, int]]:
    return {_move(token["row"], token["col"], token["direction"]) for token in tokens}


def _score_map(tokens: Iterable[dict]) -> dict[str, int]:
    output = _reconstructed(tokens)
    return {name: -len(output.symmetric_difference(mask)) for name, mask in TEMPLATES.items()}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    symbol = SYMBOLS[int(seed) % len(SYMBOLS)]
    targets = sorted(TEMPLATES[symbol])
    base_directions = [direction for direction in DIRECTIONS for _ in range(6)]
    tokens: list[dict] | None = None
    for _ in range(20000):
        directions = base_directions[:]
        rng.shuffle(directions)
        proposed = []
        occupied = set()
        for (target_r, target_c), direction in zip(targets, directions):
            dr, dc = VECTORS[direction]
            source = ((target_r - dr) % GRID, (target_c - dc) % GRID)
            if source in occupied:
                break
            occupied.add(source)
            proposed.append({"row": source[0], "col": source[1], "direction": direction})
        if len(proposed) == len(targets):
            tokens = proposed
            break
    if tokens is None:
        raise RuntimeError("could not sample a collision-free scatter")
    free = [(r, c) for r in range(GRID) for c in range(GRID) if (r, c) not in occupied]
    distractors = [list(cell) for cell in rng.sample(free, 18)]
    cell_px = rng.choice((28, 30, 32))
    span = GRID * cell_px
    frame_x = (512 - span) // 2 + rng.choice((-6, -3, 0, 3, 6))
    frame_y = (512 - span) // 2 + rng.choice((-6, -3, 0, 3, 6))
    return {
        "tokens": tokens,
        "distractors": distractors,
        "frame_x": frame_x,
        "frame_y": frame_y,
        "cell_px": cell_px,
        "palette": copy.deepcopy(rng.choice(PALETTES)),
    }


def render(scene: dict) -> Image.Image:
    tokens = scene["tokens"]
    distractors = scene["distractors"]
    x0, y0 = int(scene["frame_x"]), int(scene["frame_y"])
    cell = int(scene["cell_px"])
    palette = scene["palette"]
    image = Image.new("RGB", (512, 512), tuple(palette["background"]))
    draw = ImageDraw.Draw(image)
    span = GRID * cell
    draw.rectangle((x0, y0, x0 + span, y0 + span), fill=tuple(palette["background"]), outline=tuple(palette["frame"]), width=4)
    for index in range(1, GRID):
        x = x0 + index * cell
        y = y0 + index * cell
        draw.line((x, y0 + 4, x, y0 + span - 4), fill=tuple(palette["grid"]), width=1)
        draw.line((x0 + 4, y, x0 + span - 4, y), fill=tuple(palette["grid"]), width=1)
    pad = max(5, cell // 5)
    for r, c in distractors:
        xa, ya = x0 + c * cell + pad, y0 + r * cell + pad
        xb, yb = x0 + (c + 1) * cell - pad, y0 + (r + 1) * cell - pad
        draw.rectangle((xa, ya, xb, yb), fill=tuple(palette["distractor"]))
    for token in tokens:
        r, c, direction = int(token["row"]), int(token["col"]), token["direction"]
        cx = x0 + c * cell + cell // 2
        cy = y0 + r * cell + cell // 2
        half = max(4, cell // 7)
        draw.rectangle((cx - half, cy - half, cx + half, cy + half), fill=tuple(palette["token"]))
        length = max(8, cell // 3)
        thick = max(3, cell // 9)
        inset = max(4, cell // 7)
        if direction == "up":
            box = (cx - length // 2, y0 + r * cell + inset, cx + length // 2, y0 + r * cell + inset + thick)
        elif direction == "down":
            yy = y0 + (r + 1) * cell - inset
            box = (cx - length // 2, yy - thick, cx + length // 2, yy)
        elif direction == "left":
            box = (x0 + c * cell + inset, cy - length // 2, x0 + c * cell + inset + thick, cy + length // 2)
        else:
            xx = x0 + (c + 1) * cell - inset
            box = (xx - thick, cy - length // 2, xx, cy + length // 2)
        draw.rectangle(box, fill=tuple(palette["tab"]))
    return image


def analytic_gold(scene: dict) -> str:
    scores = _score_map(scene["tokens"])
    return max(scores, key=scores.get)


def margin(scene: dict) -> float:
    values = sorted(_score_map(scene["tokens"]).values(), reverse=True)
    return float(values[0] - values[1])


def is_quarantined(scene: dict) -> bool:
    moved = [_move(token["row"], token["col"], token["direction"]) for token in scene["tokens"]]
    return len(moved) != len(set(moved)) or margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["tokens"] = list(reversed(twin["tokens"]))
    twin["distractors"] = list(reversed(twin["distractors"]))
    return [("reverse_latent_list_order", twin)]
