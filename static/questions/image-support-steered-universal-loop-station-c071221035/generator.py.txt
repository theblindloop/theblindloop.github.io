"""Deterministic scene sampler and renderer for the universal-loop station task."""

from __future__ import annotations

import random
from collections import defaultdict, deque

from PIL import Image, ImageDraw


BACKGROUND = (250, 249, 246)
TRACK = (38, 43, 51)
PALETTE = {"cyan": (0, 174, 239), "orange": (244, 119, 48)}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    width, height = 512, 384
    x_left = rng.randint(58, 82)
    x_right = rng.randint(430, 454)
    y_mid = rng.randint(181, 203)
    half_height = rng.randint(67, 79)
    y_top, y_bottom = y_mid - half_height, y_mid + half_height
    x_spur = rng.randint(205, 307)
    spur_sign = -1 if rng.randrange(2) == 0 else 1
    spur_len = rng.randint(35, 47)

    nodes = {
        "u": [x_left, y_mid],
        "v": [x_right, y_mid],
        "ul": [x_left, y_top],
        "ur": [x_right, y_top],
        "ll": [x_left, y_bottom],
        "lr": [x_right, y_bottom],
        "d": [x_spur, y_mid],
        "s": [x_spur, y_mid + spur_sign * spur_len],
    }
    edges = [
        ["u", "ul"], ["ul", "ur"], ["ur", "v"],
        ["u", "d"], ["d", "v"], ["d", "s"],
        ["u", "ll"], ["ll", "lr"], ["lr", "v"],
    ]
    target_node = "u" if rng.randrange(2) == 0 else "v"
    target_color = "cyan" if int(seed) % 2 == 0 else "orange"
    distractor_color = "orange" if target_color == "cyan" else "cyan"
    return {
        "width": width,
        "height": height,
        "nodes": nodes,
        "edges": edges,
        "target_node": target_node,
        "distractor_node": "d",
        "colors": {"target": target_color, "distractor": distractor_color},
        "track_width": 9,
        "station_radius": 14,
    }


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    nodes = scene["nodes"]
    edges = scene["edges"]
    target_node = scene["target_node"]
    distractor_node = scene["distractor_node"]
    colors = scene["colors"]
    track_width = int(scene["track_width"])
    station_radius = int(scene["station_radius"])

    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for a, b in edges:
        draw.line([tuple(nodes[a]), tuple(nodes[b])], fill=TRACK, width=track_width)

    degree = defaultdict(int)
    for a, b in edges:
        degree[a] += 1
        degree[b] += 1
    junction_radius = max(5, track_width // 2 + 1)
    for name, point in nodes.items():
        if degree[name] != 2:
            x, y = point
            draw.ellipse(
                [x - junction_radius, y - junction_radius,
                 x + junction_radius, y + junction_radius],
                fill=TRACK,
            )

    for role, node_name in (("target", target_node), ("distractor", distractor_node)):
        x, y = nodes[node_name]
        color = PALETTE[colors[role]]
        draw.ellipse(
            [x - station_radius, y - station_radius,
             x + station_radius, y + station_radius],
            fill=color,
        )
    return image


def _cycle_rank_after_deleting(scene: dict, deleted: str) -> int:
    nodes = scene["nodes"]
    edges = scene["edges"]
    remaining = set(nodes) - {deleted}
    adjacency = {name: set() for name in remaining}
    edge_count = 0
    for a, b in edges:
        if a in remaining and b in remaining:
            adjacency[a].add(b)
            adjacency[b].add(a)
            edge_count += 1
    components = 0
    unseen = set(remaining)
    while unseen:
        components += 1
        start = unseen.pop()
        queue = deque([start])
        while queue:
            here = queue.popleft()
            for neighbor in adjacency[here]:
                if neighbor in unseen:
                    unseen.remove(neighbor)
                    queue.append(neighbor)
    return edge_count - len(remaining) + components


def analytic_gold(scene: dict) -> str:
    target_score = _cycle_rank_after_deleting(scene, scene["target_node"])
    distractor_score = _cycle_rank_after_deleting(scene, scene["distractor_node"])
    colors = scene["colors"]
    if target_score >= distractor_score:
        raise ValueError("scene does not have a unique universal-loop station")
    return str(colors["target"])


def margin(scene: dict) -> float:
    target_score = _cycle_rank_after_deleting(scene, scene["target_node"])
    distractor_score = _cycle_rank_after_deleting(scene, scene["distractor_node"])
    return float(distractor_score - target_score)


def is_quarantined(scene: dict) -> bool:
    if margin(scene) < 1.0:
        return True
    nodes = scene["nodes"]
    radius = int(scene["station_radius"])
    target = nodes[scene["target_node"]]
    distractor = nodes[scene["distractor_node"]]
    return abs(target[0] - distractor[0]) < 4 * radius


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
