"""Independent pixel-only inverse for the universal-loop station task."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


CYAN = np.asarray((0, 174, 239), dtype=np.int16)
ORANGE = np.asarray((244, 119, 48), dtype=np.int16)


def _components(mask: np.ndarray, *, include_border: bool = True) -> list[tuple[int, bool]]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=np.uint8)
    output: list[tuple[int, bool]] = []
    for sy, sx in np.argwhere(mask & (seen == 0)):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        seen[sy, sx] = 1
        queue = deque([(sy, sx)])
        area = 0
        border = False
        while queue:
            y, x = queue.popleft()
            area += 1
            border = border or y == 0 or x == 0 or y == height - 1 or x == width - 1
            if y and mask[y - 1, x] and not seen[y - 1, x]:
                seen[y - 1, x] = 1
                queue.append((y - 1, x))
            if y + 1 < height and mask[y + 1, x] and not seen[y + 1, x]:
                seen[y + 1, x] = 1
                queue.append((y + 1, x))
            if x and mask[y, x - 1] and not seen[y, x - 1]:
                seen[y, x - 1] = 1
                queue.append((y, x - 1))
            if x + 1 < width and mask[y, x + 1] and not seen[y, x + 1]:
                seen[y, x + 1] = 1
                queue.append((y, x + 1))
        if include_border or not border:
            output.append((area, border))
    return output


def _shrink_any(mask: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    height -= height % 2
    width -= width % 2
    return mask[:height, :width].reshape(height // 2, 2, width // 2, 2).any(axis=(1, 3))


def _hole_count(barrier: np.ndarray) -> int:
    small = _shrink_any(barrier)
    complement = ~small
    return sum(area >= 180 and not border for area, border in _components(complement))


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    if array.ndim != 3 or array.shape[2] != 3:
        return "abstain"
    border = np.concatenate(
        (array[:3].reshape(-1, 3), array[-3:].reshape(-1, 3),
         array[:, :3].reshape(-1, 3), array[:, -3:].reshape(-1, 3)),
        axis=0,
    )
    background = np.median(border, axis=0)
    cyan = np.linalg.norm(array - CYAN, axis=2) < 18
    orange = np.linalg.norm(array - ORANGE, axis=2) < 18
    if not (420 <= int(cyan.sum()) <= 850 and 420 <= int(orange.sum()) <= 850):
        return "abstain"
    if len(_components(cyan)) != 1 or len(_components(orange)) != 1:
        return "abstain"

    barrier = np.linalg.norm(array - background, axis=2) > 35
    if _hole_count(barrier) != 2:
        return "abstain"
    scores = {}
    for name, station in (("cyan", cyan), ("orange", orange)):
        remaining = barrier & ~station
        scores[name] = _hole_count(remaining)
    ordered = sorted(scores, key=lambda name: scores[name])
    if scores[ordered[0]] != 0 or scores[ordered[1]] < 1:
        return "abstain"
    return ordered[0]
