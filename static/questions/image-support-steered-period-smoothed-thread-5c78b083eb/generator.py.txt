"""Deterministic latent sampler and raster renderer for Period-Smoothed Thread."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
TRACE_LENGTH = 96
TOOTH_COUNT = 12
LABELS = ("arch", "cup", "wave")


def _curve_value(label: str, t: float) -> float:
    angle = 2.0 * math.pi * t
    if label == "arch":
        return math.cos(angle)
    if label == "cup":
        return -math.cos(angle)
    if label == "wave":
        return math.sin(angle)
    raise ValueError(f"unknown label {label!r}")


def sample_scene(seed: int) -> dict:
    """Sample one balanced label with independently randomized visible nuisances."""
    rng = random.Random(int(seed))
    label = LABELS[int(seed) % len(LABELS)]
    x0 = 137 + 64 * rng.randrange(4)
    y0 = 96 + 64 * rng.randrange(6)
    signal_amplitude = 5.05 if int(seed) % 17 == 0 else rng.uniform(5.9, 7.2)
    tooth_amplitude = rng.uniform(3.1, 3.8)
    tooth_phase = rng.uniform(0.0, 2.0 * math.pi)
    style = rng.randrange(3)
    decorations = []
    for _ in range(9):
        for _attempt in range(100):
            x = rng.randint(30, WIDTH - 31)
            y = rng.randint(30, HEIGHT - 31)
            if not (x0 - 25 <= x <= x0 + TRACE_LENGTH + 25 and y0 - 35 <= y <= y0 + 35):
                break
        decorations.append(
            {
                "kind": rng.choice(("dash", "circle", "corner")),
                "x": x,
                "y": y,
                "size": rng.randint(7, 15),
                "tone": rng.randrange(3),
            }
        )
    return {
        "label": label,
        "x0": x0,
        "y0": y0,
        "signal_amplitude": round(signal_amplitude, 6),
        "tooth_amplitude": round(tooth_amplitude, 6),
        "tooth_phase": round(tooth_phase, 6),
        "style": style,
        "decorations": decorations,
    }


def _scaled_point(x: float, y: float, scale: int) -> tuple[int, int]:
    return (round(x * scale), round(y * scale))


def render(scene: dict) -> Image.Image:
    """Render only scene fields; the low-frequency label is carried by one thread."""
    label = scene["label"]
    x0 = float(scene["x0"])
    y0 = float(scene["y0"])
    signal = float(scene["signal_amplitude"])
    tooth = float(scene["tooth_amplitude"])
    tooth_phase = float(scene["tooth_phase"])
    style = int(scene["style"])
    decorations = scene["decorations"]

    scale = 4
    backgrounds = ((250, 249, 246), (248, 250, 249), (251, 249, 250))
    thread_colors = ((34, 42, 56), (29, 48, 59), (44, 39, 55))
    pale = ((213, 222, 226), (225, 216, 222), (218, 224, 208))
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), backgrounds[style])
    draw = ImageDraw.Draw(image)

    for mark in decorations:
        x = int(mark["x"]) * scale
        y = int(mark["y"]) * scale
        size = int(mark["size"]) * scale
        color = pale[int(mark["tone"])]
        width = 2 * scale
        if mark["kind"] == "dash":
            draw.line((x - size, y, x + size, y), fill=color, width=width)
        elif mark["kind"] == "circle":
            draw.ellipse((x - size, y - size, x + size, y + size), outline=color, width=width)
        else:
            draw.line((x - size, y + size, x - size, y - size, x + size, y - size), fill=color, width=width)

    points = []
    for index in range(TRACE_LENGTH * 3 + 1):
        t = index / (TRACE_LENGTH * 3)
        x = x0 + TRACE_LENGTH * t
        low = signal * _curve_value(label, t)
        fine = tooth * math.sin(2.0 * math.pi * TOOTH_COUNT * t + tooth_phase)
        points.append(_scaled_point(x, y0 + low + fine, scale))
    draw.line(points, fill=thread_colors[style], width=3 * scale, joint="curve")

    endpoint_radius = 4.2 * scale
    for point, color in ((points[0], (220, 55, 65)), (points[-1], (235, 165, 28))):
        cx, cy = point
        draw.ellipse(
            (cx - endpoint_radius, cy - endpoint_radius, cx + endpoint_radius, cy + endpoint_radius),
            fill=color,
            outline=(250, 249, 246),
            width=scale,
        )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["label"])


def margin(scene: dict) -> float:
    """One-sided smoothing headroom above the calibrated raster tie band."""
    return round(float(scene["signal_amplitude"]) - 4.45, 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.75


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
