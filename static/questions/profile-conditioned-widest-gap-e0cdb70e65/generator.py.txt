"""Widest-gap selection world: which colored bar pair has the largest empty interval.

Latent scene z carries, for each of three colors, two aligned horizontal filled
bars that are vertically stacked and separated by an empty horizontal band. The
task is a pure interval measurement: find the color whose two bars are separated
by the tallest empty vertical gap (the adjacent-edge distance between the lower
edge of the upper bar and the upper edge of the lower bar).

All bars share the same fixed width and each color is confined to its own
column, so the empty gap is read cleanly from the raster by segmenting each
color and locating the two filled bands of that color.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "widest_gap-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
BACKGROUND = (250, 250, 250)

WIDTH = 480
HEIGHT = 340
BAR_WIDTH_PX = 64
COLUMN_TOP = 70
COLUMN_BOTTOM = 300

MIN_MARGIN_PX = 14.0
QUARANTINE_PX = 22.0


def sample_scene(seed: int) -> dict:
    """Sample a deterministic scene with a guaranteed, counterbalanced widest gap.

    Returns a plain dict whose ``bars`` is a list of six filled rectangular bars
    (exactly two per color) with color labels and bounding boxes.
    """
    rng = random.Random(seed * 1009 + 7)
    ncols = len(PALETTE)
    col_width = WIDTH // ncols
    # Three distinct gaps, strongest at least MIN_MARGIN_PX above the runner-up.
    while True:
        gaps = [rng.uniform(40.0, 120.0) for _ in range(ncols)]
        gaps_sorted = sorted(gaps, reverse=True)
        if gaps_sorted[0] - gaps_sorted[1] >= MIN_MARGIN_PX:
            break
    colors = list(PALETTE)
    rng.shuffle(colors)
    columns = list(range(ncols))
    rng.shuffle(columns)
    bars = []
    for color, col, gap in zip(colors, columns, gaps):
        thickness = rng.uniform(8.0, 14.0)
        available = (COLUMN_BOTTOM - COLUMN_TOP) - (2 * thickness + gap)
        start_y = COLUMN_TOP + (rng.uniform(0.0, max(available, 0.0)) if available > 0 else 0.0)
        x0 = col * col_width + (col_width - BAR_WIDTH_PX) / 2.0
        x1 = x0 + BAR_WIDTH_PX
        bars.append({"color": color, "x0": x0, "y0": start_y, "x1": x1, "y1": start_y + thickness})
        bars.append(
            {
                "color": color,
                "x0": x0,
                "y0": start_y + thickness + gap,
                "x1": x1,
                "y1": start_y + thickness + gap + thickness,
            }
        )
    return {"bars": bars, "width": WIDTH, "height": HEIGHT, "background": "light"}


def render(scene: dict) -> Image.Image:
    """Render I = R(z): for each color, two horizontal filled bars with a gap."""
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(img)
    for bar in scene["bars"]:
        draw.rectangle(
            [bar["x0"], bar["y0"], bar["x1"], bar["y1"]],
            fill=COLOR_RGB[bar["color"]],
        )
    return img


def _gap_by_color(bars: list[dict]) -> dict[str, float]:
    """Return the measured empty vertical interval height for each color."""
    per_color: dict[str, list[tuple[float, float]]] = {}
    for bar in bars:
        per_color.setdefault(bar["color"], []).append((bar["y0"], bar["y1"]))
    gaps: dict[str, float] = {}
    for color, ys in per_color.items():
        ys.sort()
        (upper_y0, upper_y1), (lower_y0, lower_y1) = ys[0], ys[1]
        gaps[color] = round(lower_y0 - upper_y1, 6)
    return gaps


def _widest_gap(bars: list[dict]) -> tuple[str, float, float]:
    """Return (widest color, its gap, top-two gap margin) from the analytic scene."""
    gaps = _gap_by_color(bars)
    ranked = sorted(gaps.items(), key=lambda item: item[1], reverse=True)
    top, second = ranked[0], ranked[1]
    margin = top[1] - second[1]
    return top[0], top[1], margin


def analytic_gold(scene: dict) -> str:
    """The color whose two bars are separated by the largest empty interval."""
    color, _gap, _margin = _widest_gap(scene["bars"])
    return color


def margin(scene: dict) -> float:
    """Verifiability margin: the top-two gap difference in pixels."""
    _color, _gap, margin_px = _widest_gap(scene["bars"])
    return float(margin_px)


def is_quarantined(scene: dict) -> bool:
    """Near-boundary: the top-two gap difference is finite-resolution small."""
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list:
    """not_applicable: no nontrivial pixel-identical latent symmetry.

    Swapping two colors' bar pairs would change which color is correct, and any
    global flip/rotation turns the raster into a different image, so the
    parametrisation has no hidden label alias and every latent field is visible.
    """
    return []
