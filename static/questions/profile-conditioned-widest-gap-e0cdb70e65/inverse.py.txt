"""Independent pixel-only inverse arm for the widest-gap world.

This module receives ONLY the rasterized PNG (plus the public palette, which is
the world's fixed naming convention) and re-measures which color's two stacked
bars are separated by the tallest empty vertical interval. It shares no geometry
with renderer.render() and never imports the renderer, prompts, generator,
verifier, or any analytical gold code.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "widest_gap-oracle-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
BACKGROUND = (250, 250, 250)


def _nearest_color(arr: np.ndarray) -> np.ndarray:
    """Label every pixel to its nearest palette entry (segment colors + background).

    Edge blends split at their ~50% point, which is dominated by the background,
    so the labeled band masks stay clean regardless of any anti-aliasing.
    """
    refs = np.stack(
        [np.array(COLOR_RGB[c], dtype=np.float32) for c in PALETTE]
        + [np.array(BACKGROUND, dtype=np.float32)],
        axis=0,
    )
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def _gap_from_mask(mask: np.ndarray) -> float:
    """Height of the empty vertical interval between a color's two filled bands.

    The colored rows of one color form two solid horizontal bands (the upper and
    lower bar). The empty gap is the distance from the last row of the upper
    band to the first row of the lower band.
    """
    rows = np.where(mask.any(axis=1))[0]
    if len(rows) < 2:
        return -1.0
    bands: list[tuple[int, int]] = []
    start = prev = int(rows[0])
    for row in rows[1:]:
        if row - prev > 1:
            bands.append((start, prev))
            start = int(row)
        prev = int(row)
    bands.append((start, prev))
    bands.sort()
    if len(bands) < 2:
        return -1.0
    (_u0, upper_last), (lower_first, _l1) = bands[0], bands[1]
    return float(lower_first - upper_last)


def decision_from_image(image: Image.Image) -> str:
    """Return the color whose two stacked bars have the largest empty gap."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _nearest_color(arr)
    best_color = PALETTE[0]
    best_gap = -1.0
    for idx, color in enumerate(PALETTE):
        mask = labels == idx
        gap = _gap_from_mask(mask)
        if gap > best_gap:
            best_gap = gap
            best_color = color
    return best_color
