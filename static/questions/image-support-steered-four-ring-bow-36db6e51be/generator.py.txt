from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

PALETTES = [
    ("#176B87", "#FFF9EC"),
    ("#7A3E9D", "#F7FAFC"),
    ("#B6452C", "#FFFDF5"),
    ("#176F4D", "#FFF8F0"),
]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    foreground, background = PALETTES[rng.randrange(len(PALETTES))]
    radius = rng.randint(20, 25)
    stroke = rng.randint(7, 10)
    xs = [rng.randint(48, 66), rng.randint(166, 194), rng.randint(318, 346), rng.randint(446, 464)]
    intercept = rng.uniform(238.0, 274.0)
    slope = rng.uniform(-0.28, 0.28)
    sign = -1.0 if int(seed) % 2 == 0 else 1.0
    amplitude = rng.uniform(70.0, 105.0)
    centers = []
    for index, x in enumerate(xs):
        baseline = intercept + slope * (x - 256.0)
        offset = 0.0 if index in (0, 3) else sign * (amplitude + rng.uniform(-8.0, 8.0))
        centers.append([float(x), round(baseline + offset, 4)])
    return {
        "centers": centers,
        "radius": radius,
        "stroke": stroke,
        "foreground": foreground,
        "background": background,
    }


def render(scene: dict) -> Image.Image:
    scale = 4
    background = scene["background"]
    foreground = scene["foreground"]
    radius = float(scene["radius"])
    stroke = int(scene["stroke"])
    image = Image.new("RGB", (512 * scale, 512 * scale), background)
    draw = ImageDraw.Draw(image)
    for cx, cy in scene["centers"]:
        box = ((cx - radius) * scale, (cy - radius) * scale,
               (cx + radius) * scale, (cy + radius) * scale)
        draw.ellipse(box, outline=foreground, width=stroke * scale)
    return image.resize((512, 512), Image.Resampling.LANCZOS)


def _ordered_centers(scene: dict) -> list[tuple[float, float]]:
    return sorted((float(x), float(y)) for x, y in scene["centers"])


def _signed_distances(scene: dict) -> list[float]:
    points = _ordered_centers(scene)
    x0, y0 = points[0]
    x3, y3 = points[-1]
    dx, dy = x3 - x0, y3 - y0
    length = math.hypot(dx, dy)
    return [(dx * (y - y0) - dy * (x - x0)) / length for x, y in points[1:3]]


def analytic_gold(scene: dict) -> str:
    distances = _signed_distances(scene)
    if all(value < 0 for value in distances):
        return "arch"
    if all(value > 0 for value in distances):
        return "bowl"
    raise ValueError("middle rings do not occupy a common chord side")


def margin(scene: dict) -> float:
    return min(abs(value) for value in _signed_distances(scene))


def is_quarantined(scene: dict) -> bool:
    distances = _signed_distances(scene)
    return min(abs(value) for value in distances) < 28.0 or distances[0] * distances[1] <= 0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["centers"] = list(reversed(twin["centers"]))
    return [("reverse_identical_ring_storage", twin)]
