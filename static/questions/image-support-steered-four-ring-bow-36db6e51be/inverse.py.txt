from __future__ import annotations

import math

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        stack = [(sy, sx)]
        seen[sy, sx] = True
        points = []
        while stack:
            y, x = stack.pop()
            points.append((y, x))
            for ny, nx in ((y-1,x),(y+1,x),(y,x-1),(y,x+1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= 80:
            output.append(np.asarray(points, dtype=np.float64))
    return output


def _ring_center(points: np.ndarray) -> tuple[float, float]:
    ys, xs = points[:, 0], points[:, 1]
    width = xs.max() - xs.min() + 1.0
    height = ys.max() - ys.min() + 1.0
    fill = len(points) / (width * height)
    if not (31.0 <= width <= 55.0 and 31.0 <= height <= 55.0):
        raise ValueError("foreground component does not have intact ring extent")
    if not (0.78 <= width / height <= 1.28 and 0.22 <= fill <= 0.68):
        raise ValueError("foreground component does not have intact ring geometry")
    return ((xs.min() + xs.max()) / 2.0, (ys.min() + ys.max()) / 2.0)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    border = np.concatenate((array[:4].reshape(-1,3), array[-4:].reshape(-1,3),
                             array[:,:4].reshape(-1,3), array[:,-4:].reshape(-1,3)))
    background = np.median(border, axis=0)
    mask = np.linalg.norm(array - background, axis=2) > 48.0
    components = _components(mask)
    if len(components) != 4:
        raise ValueError(f"expected four disconnected rings, found {len(components)}")
    points = sorted(_ring_center(component) for component in components)
    x0, y0 = points[0]
    x3, y3 = points[-1]
    dx, dy = x3 - x0, y3 - y0
    length = math.hypot(dx, dy)
    if length < 300.0:
        raise ValueError("endpoint rings are not widely separated")
    distances = [(dx * (y-y0) - dy * (x-x0)) / length for x, y in points[1:3]]
    if min(abs(value) for value in distances) < 12.0 or distances[0] * distances[1] <= 0:
        raise ValueError("ambiguous endpoint-chord side")
    return "arch" if distances[0] < 0 else "bowl"
