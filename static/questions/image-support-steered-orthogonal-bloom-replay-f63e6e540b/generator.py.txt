"""Deterministic renderer and analytic arm for Orthogonal Bloom Replay."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont

CANVAS = 512
SIZE = 7
ROUNDS = 6
CELL = 28
HEADER = 26
PAD = 4
N_ON = 24

BACKGROUND = (246, 243, 235)
GRID = (42, 47, 55)
OFF = (251, 249, 241)
MARKER = (238, 147, 42)
ON_COLORS = (
    (48, 112, 169),
    (49, 137, 105),
    (127, 87, 166),
    (187, 79, 67),
)


def _step(grid: list[list[bool]]) -> list[list[bool]]:
    n = len(grid)
    result = [[False] * n for _ in range(n)]
    for row in range(n):
        for col in range(n):
            neighbors = 0
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                yy, xx = row + dy, col + dx
                if 0 <= yy < n and 0 <= xx < n and grid[yy][xx]:
                    neighbors += 1
            result[row][col] = neighbors == 1
    return result


def _evolve(scene: dict) -> list[list[bool]]:
    grid = [[bool(value) for value in row] for row in scene["initial_on"]]
    for _ in range(int(scene["rounds"])):
        grid = _step(grid)
    return grid


def _final_neighbor_count(scene: dict) -> int:
    grid = [[bool(value) for value in row] for row in scene["initial_on"]]
    rounds = int(scene["rounds"])
    for _ in range(rounds - 1):
        grid = _step(grid)
    center = int(scene["size"]) // 2
    return sum(
        grid[center + dy][center + dx]
        for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1))
    )


def sample_scene(seed: int) -> dict:
    """Sample a balanced fixed-ink field with a requested final class.

    Even/odd seeds alternate OFF/ON.  Seed pairs share palette, translation,
    initial-center state, and total ink.  Rejection sampling selects only the
    initial arrangement needed to attain the requested class with a one-event
    runner-up margin.
    """

    desired = "on" if seed % 2 else "off"
    desired_neighbor_count = 1 if desired == "on" else (0 if (seed // 2) % 2 else 2)
    desired_initial_center = bool((seed // 2) % 2)
    rng = random.Random((int(seed) ^ 0x6B10_0D5E) & ((1 << 64) - 1))
    indexes = list(range(SIZE * SIZE))
    center_index = (SIZE * SIZE) // 2

    chosen: list[list[int]] | None = None
    for _ in range(60000):
        selected = set(rng.sample(indexes, N_ON))
        if (center_index in selected) != desired_initial_center:
            continue
        grid = [
            [1 if row * SIZE + col in selected else 0 for col in range(SIZE)]
            for row in range(SIZE)
        ]
        probe = {
            "initial_on": grid,
            "rounds": ROUNDS,
            "size": SIZE,
        }
        if _final_neighbor_count(probe) != desired_neighbor_count:
            continue
        if analytic_gold(probe) != desired:
            continue
        chosen = grid
        break
    if chosen is None:
        raise RuntimeError(f"could not sample balanced bloom scene for seed {seed}")

    pair = seed // 2
    return {
        "size": SIZE,
        "rounds": ROUNDS,
        "initial_on": chosen,
        "board_x": 158 + (pair % 5) - 2,
        "board_y": 171,
        "cell_size": CELL,
        "on_color": list(ON_COLORS[pair % len(ON_COLORS)]),
        "background": list(BACKGROUND),
        "grid_color": list(GRID),
        "off_color": list(OFF),
        "marker_color": list(MARKER),
    }


def analytic_gold(scene: dict) -> str:
    final = _evolve(scene)
    center = int(scene["size"]) // 2
    return "on" if final[center][center] else "off"


def margin(scene: dict) -> float:
    """Minimum neighbor-status changes needed to cross the exactly-one rule."""

    count = _final_neighbor_count(scene)
    return float(1 if count <= 2 else count - 1)


def is_quarantined(scene: dict) -> bool:
    on = tuple(int(value) for value in scene["on_color"])
    off = tuple(int(value) for value in scene["off_color"])
    clearance_sq = sum((left - right) ** 2 for left, right in zip(on, off, strict=True))
    return clearance_sq < 70 * 70


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int, *, bold: bool = False) -> ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    size = int(scene["size"])
    rounds = int(scene["rounds"])
    cell = int(scene["cell_size"])
    x0 = int(scene["board_x"])
    y0 = int(scene["board_y"])
    background = tuple(scene["background"])
    grid_color = tuple(scene["grid_color"])
    off_color = tuple(scene["off_color"])
    on_color = tuple(scene["on_color"])
    marker_color = tuple(scene["marker_color"])
    initial = scene["initial_on"]

    image = Image.new("RGB", (CANVAS, CANVAS), background)
    draw = ImageDraw.Draw(image)
    board_px = size * cell
    header_top = y0 - HEADER

    # Soft offset card shadow, followed by one connected operation patch.
    draw.rounded_rectangle(
        (x0 + 5, header_top + 7, x0 + board_px + 5, y0 + board_px + 7),
        radius=9,
        fill=(218, 213, 202),
    )
    draw.rounded_rectangle(
        (x0, header_top, x0 + board_px - 1, y0 + board_px - 1),
        radius=8,
        fill=grid_color,
    )
    # Square the join so the header and grid are a single contiguous field.
    draw.rectangle((x0, y0 - 3, x0 + board_px - 1, y0 + board_px - 1), fill=grid_color)

    # Six visible beads are the ordered repeated-round declaration.
    bead_gap = board_px / (rounds + 1)
    for index in range(rounds):
        cx = round(x0 + bead_gap * (index + 1))
        cy = header_top + HEADER // 2
        draw.ellipse((cx - 4, cy - 4, cx + 4, cy + 4), fill=marker_color)

    for row in range(size):
        for col in range(size):
            left = x0 + col * cell + PAD
            top = y0 + row * cell + PAD
            right = x0 + (col + 1) * cell - PAD - 1
            bottom = y0 + (row + 1) * cell - PAD - 1
            fill = on_color if bool(initial[row][col]) else off_color
            draw.rounded_rectangle((left, top, right, bottom), radius=3, fill=fill)

    center = size // 2
    left = x0 + center * cell + PAD + 1
    top = y0 + center * cell + PAD + 1
    right = x0 + (center + 1) * cell - PAD - 2
    bottom = y0 + (center + 1) * cell - PAD - 2
    draw.rounded_rectangle((left, top, right, bottom), radius=4, outline=marker_color, width=3)

    # A small visual legend is intentionally outside the operation patch.
    legend_y = 404
    draw.rounded_rectangle((142, legend_y, 370, legend_y + 42), radius=20, fill=(255, 254, 249))
    draw.rectangle((166, legend_y + 11, 184, legend_y + 29), fill=on_color)
    draw.text((192, legend_y + 11), "ON", font=_font(15, bold=True), fill=grid_color)
    draw.rectangle((258, legend_y + 11, 276, legend_y + 29), fill=off_color, outline=grid_color)
    draw.text((284, legend_y + 11), "OFF", font=_font(15, bold=True), fill=grid_color)
    draw.text(
        (137, 55),
        "ORTHOGONAL BLOOM",
        font=_font(23, bold=True),
        fill=grid_color,
    )
    draw.text(
        (164, 88),
        "six simultaneous rounds",
        font=_font(15),
        fill=(92, 96, 102),
    )
    return image
