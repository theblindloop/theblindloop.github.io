"""Independent image-only oracle for the gateway-ladder reach world.

The oracle works only from the raster.  It recovers the pale background, the
blue source dot, and the four coloured target dots by simple colour masks, then
flood-fills the pale field starting from the blue source.  A target dot is
counted exactly when the flood reaches any of its pixels; the answer is the
number of reachable targets.  No latent scene state is used.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_TARGET = {
    "cyan": (0, 160, 190),
    "magenta": (200, 50, 170),
    "lime": (140, 190, 40),
    "gold": (230, 170, 40),
}
_LABELS = ("zero", "one", "two", "three", "four")


def _mask(arr, centre, tol):
    return (
        (np.abs(arr[:, :, 0].astype(np.int16) - centre[0]) < tol)
        & (np.abs(arr[:, :, 1].astype(np.int16) - centre[1]) < tol)
        & (np.abs(arr[:, :, 2].astype(np.int16) - centre[2]) < tol)
    )


def _flood(shape, seeds, passable):
    h, w = shape
    seen = np.zeros(shape, dtype=bool)
    stack = list(seeds)
    for (y, x) in stack:
        seen[y, x] = True
    while stack:
        y, x = stack.pop()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < h and 0 <= nx < w and passable[ny, nx] and not seen[ny, nx]:
                seen[ny, nx] = True
                stack.append((ny, nx))
    return seen


def decision_from_image(image: Image.Image) -> str:
    if image.mode != "RGB":
        image = image.convert("RGB")
    arr = np.asarray(image, dtype=np.int16)
    bg = _mask(arr, (250, 250, 246), 40)
    blue = _mask(arr, (34, 66, 214), 70)
    if not blue.any():
        raise ValueError("source colour not found")
    passable = bg | blue
    for centre in _TARGET.values():
        passable = passable | _mask(arr, centre, 70)
    seeds = [(int(y), int(x)) for y, x in zip(*np.where(blue))]
    reached = _flood(passable.shape, seeds, passable)
    count = 0
    for centre in _TARGET.values():
        portion = _mask(arr, centre, 70)
        if reached[portion].any():
            count += 1
    return _LABELS[count]
