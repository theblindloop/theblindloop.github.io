"""Deterministic renderer and analytic gold for the gateway-ladder reach world.

A pale sheet is split by four thick vertical walls into five vertical strips.
The far-left strip holds a blue source dot; the other four strips hold one
coloured target dot each (cyan, magenta, lime, gold), in strip order.  Each
wall is either open (it leaves a tall pale gap at its top so a route can pass
over it) or sealed (it runs top to bottom and blocks everything to its right).
A target dot is reachable from the blue source exactly when every wall strictly
to its left is open, because the strips are separated by full-height walls and
the only passages are the top gaps.  The answer is the number of the four
coloured targets that lie in the source's connected pale component.
A decorative key rectangle in one corner is not part of the field.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

W = 800
H = 560
BG = (250, 250, 246)
INK = (42, 46, 54)
BLUE = (34, 66, 214)
TARGETS = {
    "cyan": (0, 160, 190),
    "magenta": (200, 50, 170),
    "lime": (140, 190, 40),
    "gold": (230, 170, 40),
}
WALL_W = 26
OPEN_CEIL = 210

WALL_XS = (220, 350, 480, 610)
SOURCE_POS = (103, 300)
TARGET_XS = (285, 415, 545, 711)
TARGET_YS = 300
PORT_RAD = 20

_LABELS = ("zero", "one", "two", "three", "four")

PROMPT_FAMILIES = (
    "pf1_gateway_reach_count",
    "pf2_serial_rooms_reach",
    "pf3_zone_reach_tally",
)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    walls = [bool(rng.randrange(2)) for _ in range(4)]
    variant = rng.randrange(4)
    return {"walls": walls, "variant": variant}


def _vwall(draw, x: float, y0: float, y1: float) -> None:
    half = WALL_W / 2
    draw.rectangle([x - half, y0, x + half, y1], fill=INK)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (W, H), BG)
    draw = ImageDraw.Draw(image)
    walls = scene["walls"]
    for x, opened in zip(WALL_XS, walls):
        if opened:
            _vwall(draw, x, OPEN_CEIL, H)
        else:
            _vwall(draw, x, 0, H)
    draw.ellipse(
        [
            SOURCE_POS[0] - PORT_RAD,
            SOURCE_POS[1] - PORT_RAD,
            SOURCE_POS[0] + PORT_RAD,
            SOURCE_POS[1] + PORT_RAD,
        ],
        fill=BLUE,
    )
    for x, colour in zip(TARGET_XS, TARGETS.values()):
        draw.ellipse(
            [x - PORT_RAD, TARGET_YS - PORT_RAD, x + PORT_RAD, TARGET_YS + PORT_RAD],
            fill=colour,
        )
    kx = 30 + (scene["variant"] % 3) * 16
    draw.rectangle([kx, 12, kx + 22, 34], outline=INK, width=4)
    return image


def analytic_gold(scene: dict) -> str:
    count = 0
    for opened in scene["walls"]:
        if not opened:
            break
        count += 1
    return _LABELS[count]


def margin(scene: dict) -> float:
    clearances = []
    for x, opened in zip(WALL_XS, scene["walls"]):
        if opened:
            clearances.append(OPEN_CEIL)
    for x in (SOURCE_POS[0], *TARGET_XS):
        clearances.append(min(abs(x - (wx - WALL_W / 2)) for wx in WALL_XS))
    return float(min(clearances))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list:
    return []
