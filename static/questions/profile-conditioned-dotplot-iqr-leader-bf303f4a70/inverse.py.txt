"""Pixel-only inverse arm: segment dots, recover centers, and compare IQRs."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

COLORS = {
    "blue": np.array([35, 112, 180], dtype=np.int16),
    "orange": np.array([230, 126, 34], dtype=np.int16),
}


def _component_centers(mask: np.ndarray) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    centers = []
    for y0, x0 in zip(*np.nonzero(mask)):
        if seen[y0, x0]:
            continue
        queue = deque([(int(y0), int(x0))])
        seen[y0, x0] = True
        xs = []
        ys = []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= 80:
            centers.append((float(np.mean(xs)), float(np.mean(ys)), len(xs)))
    return centers


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    iqrs = {}
    for name, target in COLORS.items():
        distance = np.max(np.abs(pixels - target), axis=2)
        components = _component_centers(distance <= 5)
        if len(components) != 5:
            raise ValueError(f"expected five {name} dot components, found {len(components)}")
        xs = sorted(center[0] for center in components)
        iqrs[name] = xs[3] - xs[1]
    gap = abs(iqrs["blue"] - iqrs["orange"])
    if gap < 8.0:
        raise ValueError("pixel IQR comparison is ambiguous")
    return "blue" if iqrs["blue"] > iqrs["orange"] else "orange"
