"""Deterministic renderer and analytic arm for the dot-plot IQR world."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 640, 360
BLUE = (35, 112, 180)
ORANGE = (230, 126, 34)
PALETTE = {"blue": BLUE, "orange": ORANGE}


def _five_positions(rng: random.Random, iqr: int) -> list[int]:
    second = rng.randint(175, 230)
    fourth = second + iqr
    first = second - rng.randint(35, 55)
    middle = rng.randint(second + 28, fourth - 28)
    fifth = fourth + rng.randint(35, 55)
    return [first, second, middle, fourth, fifth]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    winner = "blue" if int(seed) % 2 == 0 else "orange"
    loser_iqr = rng.randint(120, 164)
    gap = 20 + 14 * (abs(int(seed)) % 6)
    iqrs = {
        winner: loser_iqr + gap,
        ("orange" if winner == "blue" else "blue"): loser_iqr,
    }
    lane_names = ["blue", "orange"]
    rng.shuffle(lane_names)
    lanes = {lane_names[0]: 118, lane_names[1]: 220}
    series = [
        {
            "name": name,
            "rgb": list(PALETTE[name]),
            "lane_y": lanes[name],
            "xs": _five_positions(rng, iqrs[name]),
        }
        for name in ("blue", "orange")
    ]
    rng.shuffle(series)
    return {"series": series}


def _iqrs(scene: dict) -> dict[str, float]:
    values = {}
    for item in scene["series"]:
        xs = sorted(float(value) for value in item["xs"])
        values[item["name"]] = xs[3] - xs[1]
    return values


def analytic_gold(scene: dict) -> str:
    values = _iqrs(scene)
    return max(values, key=values.get)


def margin(scene: dict) -> float:
    values = sorted(_iqrs(scene).values(), reverse=True)
    return float(values[0] - values[1])


def is_quarantined(scene: dict) -> bool:
    if len(scene["series"]) != 2 or margin(scene) < 14.0:
        return True
    for item in scene["series"]:
        xs = sorted(item["xs"])
        if len(xs) != 5 or any(b - a < 20 for a, b in zip(xs, xs[1:])):
            return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["series"] = list(reversed(twin["series"]))
    return [("reverse_disjoint_draw_order", twin)]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (250, 250, 248))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    draw.rounded_rectangle((52, 58, 588, 270), radius=12, fill=(255, 255, 255), outline=(80, 86, 92), width=2)
    draw.text((62, 24), "FIVE MEASUREMENTS PER COLOR", fill=(36, 40, 44), font=font)
    for y in (118, 220):
        draw.line((82, y, 558, y), fill=(214, 218, 221), width=2)

    axis_y = 304
    draw.line((82, axis_y, 558, axis_y), fill=(45, 49, 53), width=2)
    for x, label in zip((82, 201, 320, 439, 558), ("0", "25", "50", "75", "100")):
        draw.line((x, axis_y - 5, x, axis_y + 6), fill=(45, 49, 53), width=2)
        draw.text((x - 7, axis_y + 11), label, fill=(45, 49, 53), font=font)
    draw.text((255, 340), "SHARED HORIZONTAL SCALE", fill=(45, 49, 53), font=font)

    for item in scene["series"]:
        y = int(item["lane_y"])
        color = tuple(int(channel) for channel in item["rgb"])
        for x_value in item["xs"]:
            x = int(x_value)
            draw.ellipse((x - 11, y - 11, x + 11, y + 11), fill=color, outline=(25, 28, 31), width=2)
    return image
