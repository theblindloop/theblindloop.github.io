"""Deterministic renderer for the next-track triangle/bar prior conflict."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

WIDTH = 384
HEIGHT = 384
SUPERSAMPLE = 4
QUARANTINE_MARGIN = 18.0

PALETTES = (
    ((36, 202, 190), (249, 104, 82)),
    ((250, 186, 55), (70, 132, 238)),
    ((236, 92, 156), (77, 192, 115)),
    ((142, 104, 235), (245, 139, 54)),
)
MARK_COLORS = ((102, 210, 199), (246, 126, 106), (250, 199, 84), (132, 112, 224))


def sample_scene(seed: int) -> dict:
    """Return one member of an exactly style-matched even/odd treatment pair."""
    seed = int(seed)
    relation = "toward" if seed % 2 == 0 else "away"
    rng = random.Random(seed // 2)
    palette = list(PALETTES[rng.randrange(len(PALETTES))])
    if rng.randrange(2):
        palette.reverse()
    marker_colors = list(MARK_COLORS)
    rng.shuffle(marker_colors)
    panel_size = rng.randrange(272, 301, 4)
    inset = panel_size // 2 - 26
    center_x = WIDTH // 2 + rng.randint(-13, 13)
    center_y = HEIGHT // 2 + rng.randint(-13, 13)
    distractors = []
    for (sx, sy), color in zip(((-1, -1), (1, -1), (1, 1), (-1, 1)), marker_colors):
        distractors.append(
            {
                "x": WIDTH // 2 + sx * inset,
                "y": HEIGHT // 2 + sy * inset,
                "size": rng.randint(6, 9),
                "color": list(color),
            }
        )
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "center_x": center_x,
        "center_y": center_y,
        "quarter_turn": rng.randrange(4),
        "bar_side": -1 if rng.randrange(2) == 0 else 1,
        "relation": relation,
        "scale": rng.randint(80, 96),
        "gap": rng.randint(32, 46),
        "panel_size": panel_size,
        "triangle_color": list(palette[0]),
        "bar_color": list(palette[1]),
        "panel_color": [31, 40, 55],
        "background": [247, 245, 239],
        "distractors": distractors,
    }


def _rotate(x: float, y: float, quarter_turn: int) -> tuple[float, float]:
    q = int(quarter_turn) % 4
    if q == 0:
        return x, y
    if q == 1:
        return -y, x
    if q == 2:
        return -x, -y
    return y, -x


def _geometry(scene: dict) -> tuple[list[tuple[float, float]], list[tuple[float, float]]]:
    scale = float(scene["scale"])
    gap = float(scene["gap"])
    side = int(scene["bar_side"])
    relation = str(scene["relation"])
    q = int(scene["quarter_turn"])
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])

    half_axis = 0.62 * scale
    half_base = 0.62 * scale
    bar_thickness = 0.20 * scale
    bar_half_length = 0.59 * scale
    distance = half_axis + gap + bar_thickness / 2.0
    triangle_center_x = -side * distance * 0.32
    bar_center_x = triangle_center_x + side * distance
    point_sign = side if relation == "toward" else -side

    triangle_local = [
        (triangle_center_x + point_sign * half_axis, 0.0),
        (triangle_center_x - point_sign * 0.85 * half_axis, -half_base),
        (triangle_center_x - point_sign * 0.85 * half_axis, half_base),
    ]
    bar_local = [
        (bar_center_x - bar_thickness / 2.0, -bar_half_length),
        (bar_center_x + bar_thickness / 2.0, -bar_half_length),
        (bar_center_x + bar_thickness / 2.0, bar_half_length),
        (bar_center_x - bar_thickness / 2.0, bar_half_length),
    ]

    def place(points: list[tuple[float, float]]) -> list[tuple[float, float]]:
        return [(cx + _rotate(x, y, q)[0], cy + _rotate(x, y, q)[1]) for x, y in points]

    return place(triangle_local), place(bar_local)


def _scaled_polygon(points: list[tuple[float, float]]) -> list[tuple[int, int]]:
    return [(round(x * SUPERSAMPLE), round(y * SUPERSAMPLE)) for x, y in points]


def render(scene: dict) -> Image.Image:
    """Render solely from the public scene dictionary to one RGB raster."""
    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(int(v) for v in scene["background"])
    image = Image.new("RGB", (width * SUPERSAMPLE, height * SUPERSAMPLE), background)
    draw = ImageDraw.Draw(image)

    panel_size = int(scene["panel_size"])
    px = (width - panel_size) / 2.0
    py = (height - panel_size) / 2.0
    panel_box = tuple(round(v * SUPERSAMPLE) for v in (px, py, px + panel_size, py + panel_size))
    draw.rounded_rectangle(
        panel_box,
        radius=round(panel_size * 0.19 * SUPERSAMPLE),
        fill=tuple(int(v) for v in scene["panel_color"]),
    )

    for mark in scene["distractors"]:
        mx = int(mark["x"]) * SUPERSAMPLE
        my = int(mark["y"]) * SUPERSAMPLE
        ms = int(mark["size"]) * SUPERSAMPLE
        draw.polygon(
            [(mx, my - ms), (mx + ms, my), (mx, my + ms), (mx - ms, my)],
            fill=tuple(int(v) for v in mark["color"]),
        )

    triangle, bar = _geometry(scene)
    draw.polygon(_scaled_polygon(triangle), fill=tuple(int(v) for v in scene["triangle_color"]))
    draw.polygon(_scaled_polygon(bar), fill=tuple(int(v) for v in scene["bar_color"]))
    return image.resize((width, height), Image.Resampling.LANCZOS).convert("RGB")


def analytic_gold(scene: dict) -> str:
    relation = str(scene["relation"])
    if relation not in {"toward", "away"}:
        raise ValueError(f"invalid relation {relation!r}")
    return relation


def margin(scene: dict) -> float:
    """One-sided visible clearance/taper margin in nominal output pixels."""
    return float(min(float(scene["gap"]), 0.40 * float(scene["scale"])))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN or float(scene["scale"]) < 60.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    full_turn = copy.deepcopy(scene)
    full_turn["quarter_turn"] = int(full_turn["quarter_turn"]) + 4
    reversed_marks = copy.deepcopy(scene)
    reversed_marks["distractors"] = list(reversed(reversed_marks["distractors"]))
    return [
        ("quarter_turn_plus_full_revolution", full_turn),
        ("reverse_nonoverlapping_diamond_order", reversed_marks),
    ]
