from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw


PALETTES = (
    {"background": "#fbfaf5", "channel": "#183653", "start": "#16a35b", "cyan": "#18b7c9", "orange": "#ed8628"},
    {"background": "#f8fbf7", "channel": "#24334f", "start": "#249b55", "cyan": "#20aebe", "orange": "#e8792b"},
    {"background": "#fffaf2", "channel": "#123b5c", "start": "#0f9f68", "cyan": "#0eb4c7", "orange": "#ef8f22"},
)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    # Answer is exactly counterbalanced over adjacent seeds; every other visual
    # variable is sampled independently of that parity.
    cyan_wide = int(seed) % 2 == 0
    narrow = rng.randint(60, 68)
    gap = rng.randint(24, 34)
    center = [rng.randint(242, 270), rng.randint(242, 270)]
    rotation = rng.uniform(0.0, 360.0)
    swap_terminals = bool(rng.getrandbits(1))
    cyan_angle_offset = 120.0 if not swap_terminals else -120.0
    orange_angle_offset = -120.0 if not swap_terminals else 120.0
    return {
        "width": 512,
        "height": 512,
        "center": center,
        "rotation_deg": rotation,
        "radius": rng.randint(126, 138),
        "start_width": rng.randint(72, 84),
        "cyan_width": narrow + gap if cyan_wide else narrow,
        "orange_width": narrow if cyan_wide else narrow + gap,
        "marker_radius": rng.randint(17, 20),
        "cyan_angle_offset": cyan_angle_offset,
        "orange_angle_offset": orange_angle_offset,
        "palette_variant": rng.randrange(len(PALETTES)),
    }


def _endpoint(scene: dict, offset_deg: float) -> tuple[float, float]:
    angle = math.radians(float(scene["rotation_deg"]) + offset_deg)
    return (
        float(scene["center"][0]) + float(scene["radius"]) * math.cos(angle),
        float(scene["center"][1]) + float(scene["radius"]) * math.sin(angle),
    )


def render(scene: dict) -> Image.Image:
    scale = 4
    width, height = int(scene["width"]), int(scene["height"])
    palette = PALETTES[int(scene["palette_variant"])]
    image = Image.new("RGB", (width * scale, height * scale), palette["background"])
    draw = ImageDraw.Draw(image)
    center = (float(scene["center"][0]), float(scene["center"][1]))
    endpoints = {
        "start": _endpoint(scene, 0.0),
        "cyan": _endpoint(scene, float(scene["cyan_angle_offset"])),
        "orange": _endpoint(scene, float(scene["orange_angle_offset"])),
    }
    arm_widths = {
        "start": int(scene["start_width"]),
        "cyan": int(scene["cyan_width"]),
        "orange": int(scene["orange_width"]),
    }
    c = tuple(round(value * scale) for value in center)
    # Draw the narrower arms first, then seal their shared interior with a disk.
    for name in sorted(endpoints, key=lambda key: arm_widths[key]):
        p = tuple(round(value * scale) for value in endpoints[name])
        draw.line([c, p], fill=palette["channel"], width=arm_widths[name] * scale)
    seal_r = max(arm_widths.values()) * scale // 2
    draw.ellipse([c[0] - seal_r, c[1] - seal_r, c[0] + seal_r, c[1] + seal_r], fill=palette["channel"])
    marker_r = int(scene["marker_radius"]) * scale
    for name, point in endpoints.items():
        p = tuple(round(value * scale) for value in point)
        draw.ellipse([p[0] - marker_r, p[1] - marker_r, p[0] + marker_r, p[1] + marker_r], fill=palette[name])
        # A white center dot makes each endpoint unmistakably a port while keeping
        # all three endpoint sizes identical.
        dot = max(3 * scale, marker_r // 5)
        draw.ellipse([p[0] - dot, p[1] - dot, p[0] + dot, p[1] + dot], fill=palette["background"])
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    cyan_width = float(scene["cyan_width"])
    orange_width = float(scene["orange_width"])
    return "cyan" if cyan_width > orange_width else "orange"


def margin(scene: dict) -> float:
    return abs(float(scene["cyan_width"]) - float(scene["orange_width"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 16.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
