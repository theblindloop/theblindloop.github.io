"""Independent pixel-only inverse arm for paired shared-span fractions."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_COLORS = {
    "coral": np.array((226, 82, 70), dtype=np.int16),
    "blue": np.array((39, 105, 181), dtype=np.int16),
}


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]:
            continue
        queue = deque([(y0, x0)])
        seen[y0, x0] = True
        xs, ys = [], []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= 300:
            found.append((min(xs), min(ys), max(xs), max(ys), len(xs)))
    return found


def _intervals(array: np.ndarray, name: str) -> list[tuple[int, int]]:
    target = _COLORS[name]
    distance = np.max(np.abs(array.astype(np.int16) - target), axis=2)
    comps = _components(distance <= 8)
    if len(comps) != 2:
        raise ValueError(f"expected two {name} bars, found {len(comps)}")
    result = []
    for x0, y0, x1, y1, area in comps:
        width, height = x1 - x0 + 1, y1 - y0 + 1
        if not (45 <= width <= 120 and 9 <= height <= 13):
            raise ValueError(f"malformed {name} bar")
        if area < 0.96 * width * height:
            raise ValueError(f"damaged {name} bar")
        result.append((x0, x1))
    return result


def _score(pair: list[tuple[int, int]]) -> tuple[int, int]:
    (a0, a1), (b0, b1) = pair
    shared = min(a1, b1) - max(a0, b0) + 1
    span = max(a1, b1) - min(a0, b0) + 1
    if shared <= 0:
        raise ValueError("bar pair has no shared horizontal length")
    return shared, span


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    c_num, c_den = _score(_intervals(array, "coral"))
    b_num, b_den = _score(_intervals(array, "blue"))
    left, right = c_num * b_den, b_num * c_den
    if left == right:
        raise ValueError("pixel fraction tie")
    return "coral" if left > right else "blue"
