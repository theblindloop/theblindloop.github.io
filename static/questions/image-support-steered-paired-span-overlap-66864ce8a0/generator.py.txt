"""Deterministic sampler and renderer for paired shared-span fractions."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

CORAL = (226, 82, 70)
BLUE = (39, 105, 181)
GRAY = (202, 204, 201)
BACKGROUND = (250, 249, 245)


def _fraction(pair: list[list[int]]) -> float:
    (a0, a1), (b0, b1) = pair
    shared = min(a1, b1) - max(a0, b0) + 1
    span = max(a1, b1) - min(a0, b0) + 1
    if shared <= 0 or span <= 0:
        raise ValueError("bars must have positive horizontal overlap")
    return shared / span


def _random_pair(rng: random.Random) -> list[list[int]]:
    width_a = rng.randint(58, 108)
    width_b = rng.randint(58, 108)
    offset = rng.randint(-58, 58)
    a0 = -(width_a // 2)
    b0 = offset - (width_b // 2)
    pair = [[a0, a0 + width_a - 1], [b0, b0 + width_b - 1]]
    return pair


def _pair_in_band(rng: random.Random, low: float, high: float) -> list[list[int]]:
    for _ in range(5000):
        pair = _random_pair(rng)
        score = _fraction(pair) if min(pair[0][1], pair[1][1]) >= max(pair[0][0], pair[1][0]) else -1.0
        if low <= score <= high:
            if rng.random() < 0.5:
                pair.reverse()
            return pair
    raise RuntimeError("could not sample requested fraction band")


def _display_pairs(scene: dict) -> tuple[list[list[int]], list[list[int]]]:
    coral = [[int(x) for x in row] for row in scene["coral_intervals"]]
    blue = [[int(x) for x in row] for row in scene["blue_intervals"]]
    return (blue, coral) if bool(scene["palette_swap"]) else (coral, blue)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    desired = "coral" if int(seed) % 2 == 0 else "blue"

    # One sixth of scenes sits just outside quarantine for boundary coverage.
    if int(seed) % 6 == 0:
        high = _pair_in_band(rng, 0.58, 0.73)
        high_score = _fraction(high)
        low = _pair_in_band(rng, max(0.28, high_score - 0.115), high_score - 0.082)
    else:
        high = _pair_in_band(rng, 0.64, 0.86)
        low = _pair_in_band(rng, 0.25, 0.49)

    display_coral, display_blue = (high, low) if desired == "coral" else (low, high)
    palette_swap = bool(rng.getrandbits(1))
    latent_coral, latent_blue = (
        (display_blue, display_coral) if palette_swap else (display_coral, display_blue)
    )

    # Four disconnected decisive bars occupy a centered regional footprint:
    # two colors separated horizontally and each pair separated vertically.
    dx, dy = rng.randint(-18, 18), rng.randint(-18, 18)
    left_center = [180 + dx, 256 + dy]
    right_center = [332 + dx, 256 + dy]
    side_swap = bool(rng.getrandbits(1))
    centers = [left_center, right_center]

    context_marks = []
    for _ in range(8):
        for _attempt in range(200):
            x, y = rng.randint(26, 486), rng.randint(26, 486)
            if all((x - cx) ** 2 + (y - cy) ** 2 > 125 ** 2 for cx, cy in centers):
                context_marks.append([x, y, rng.choice((5, 7, 9))])
                break

    return {
        "coral_intervals": latent_coral,
        "blue_intervals": latent_blue,
        "bar_height": 11,
        "row_offset": 100,
        "centers": centers,
        "palette_swap": palette_swap,
        "side_swap": side_swap,
        "context_marks": context_marks,
        "width": 512,
        "height": 512,
        "background": list(BACKGROUND),
    }


def _draw_pair(draw: ImageDraw.ImageDraw, center: list[int], pair: list[list[int]], color: tuple[int, int, int], height: int, row_offset: int) -> None:
    cx, cy = center
    for y, (left, right) in zip((cy - row_offset, cy + row_offset), pair, strict=True):
        draw.rectangle((cx + left, y - height // 2, cx + right, y + height // 2), fill=color)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    for x, y, radius in scene["context_marks"]:
        r = int(radius)
        draw.ellipse((int(x) - r, int(y) - r, int(x) + r, int(y) + r), outline=GRAY, width=2)
    coral, blue = _display_pairs(scene)
    coral_center, blue_center = scene["centers"]
    if bool(scene["side_swap"]):
        coral_center, blue_center = blue_center, coral_center
    _draw_pair(draw, coral_center, coral, CORAL, int(scene["bar_height"]), int(scene["row_offset"]))
    _draw_pair(draw, blue_center, blue, BLUE, int(scene["bar_height"]), int(scene["row_offset"]))
    return image


def analytic_gold(scene: dict) -> str:
    coral, blue = _display_pairs(scene)
    a, b = _fraction(coral), _fraction(blue)
    if a == b:
        raise ValueError("tied shared-span fractions")
    return "coral" if a > b else "blue"


def margin(scene: dict) -> float:
    coral, blue = _display_pairs(scene)
    return 100.0 * abs(_fraction(coral) - _fraction(blue))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["context_marks"] = list(reversed(twin["context_marks"]))
    return [("reverse_context_order", twin)]
