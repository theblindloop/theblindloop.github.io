"""Independent image-only inverse arm for paired mask intersection.

The inverse locates the two visible square frames without fixed coordinates,
uses the small corner blocks to undo each quarter-turn, intersects the recovered
cell rasters, and reports the nearest-template runner-up margin. It imports no
renderer, scene, prompt, generator, verifier, or analytic-gold code.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

GRID_SIZE = 7
SYMBOLS = ("diamond", "plus", "X")


def _independent_templates() -> dict[str, np.ndarray]:
    result: dict[str, np.ndarray] = {}
    diamond = np.zeros((7, 7), dtype=bool)
    for row, col in (
        (0, 3),
        (1, 2),
        (1, 4),
        (2, 1),
        (2, 5),
        (3, 0),
        (3, 3),
        (3, 6),
        (4, 1),
        (4, 5),
        (5, 2),
        (5, 4),
        (6, 3),
    ):
        diamond[row, col] = True
    result["diamond"] = diamond
    plus = np.zeros((7, 7), dtype=bool)
    for index in range(7):
        plus[3, index] = True
        plus[index, 3] = True
    result["plus"] = plus
    cross = np.zeros((7, 7), dtype=bool)
    for index in range(7):
        cross[index, index] = True
        cross[index, 6 - index] = True
    result["X"] = cross
    return result


TEMPLATES = _independent_templates()


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for start_y, start_x in np.argwhere(mask):
        y0, x0 = int(start_y), int(start_x)
        if seen[y0, x0]:
            continue
        queue = deque([(y0, x0)])
        seen[y0, x0] = True
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        output.append(np.asarray(points, dtype=int))
    return output


def _box(points: np.ndarray) -> tuple[int, int, int, int]:
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    return int(x0), int(y0), int(x1), int(y1)


def _frame_score(points: np.ndarray) -> float:
    x0, y0, x1, y1 = _box(points)
    width, height = x1 - x0 + 1, y1 - y0 + 1
    if not (28 <= width <= 40 and 28 <= height <= 40 and abs(width - height) <= 2):
        return 0.0
    xy = {(int(x), int(y)) for y, x in points}
    sides = (
        sum((x, y0) in xy for x in range(x0, x1 + 1)) / width,
        sum((x, y1) in xy for x in range(x0, x1 + 1)) / width,
        sum((x0, y) in xy for y in range(y0, y1 + 1)) / height,
        sum((x1, y) in xy for y in range(y0, y1 + 1)) / height,
    )
    return float(min(sides))


def _locate_frames(array: np.ndarray, background: tuple[int, int, int]) -> tuple[tuple[int, int, int], list[tuple[int, int, int, int]]]:
    colors = [tuple(int(v) for v in row) for row in np.unique(array.reshape(-1, 3), axis=0)]
    choices: list[tuple[float, tuple[int, int, int], list[tuple[int, int, int, int]]]] = []
    for color in colors:
        if color == background:
            continue
        mask = np.all(array == np.asarray(color, dtype=np.uint8), axis=2)
        components = _components(mask)
        framed = [(float(_frame_score(points)), _box(points)) for points in components]
        framed = [item for item in framed if item[0] >= 0.75]
        if len(framed) == 2:
            choices.append((min(item[0] for item in framed), color, [item[1] for item in framed]))
    if len(choices) != 1:
        raise ValueError(f"expected one two-frame color, found {len(choices)}")
    _, frame_color, boxes = choices[0]
    boxes.sort(key=lambda item: (item[0], item[1]))
    if boxes[1][0] - boxes[0][2] < 20:
        raise ValueError("the two frame components are not visibly separated")
    return frame_color, boxes


def _recover_panel(
    array: np.ndarray,
    box: tuple[int, int, int, int],
    frame_color: tuple[int, int, int],
    background: tuple[int, int, int],
) -> np.ndarray:
    x0, y0, x1, y1 = box
    width, height = x1 - x0 + 1, y1 - y0 + 1
    if width != 32 or height != 32:
        raise ValueError(f"unsupported frame dimensions {width}x{height}")
    frame_rgb = np.asarray(frame_color, dtype=np.uint8)
    bg_rgb = np.asarray(background, dtype=np.uint8)
    corner_origins = ((x0, y0), (x0, y1 - 1), (x1 - 1, y1 - 1), (x1 - 1, y0))
    marker_hits = []
    for index, (cx, cy) in enumerate(corner_origins):
        patch = array[cy : cy + 2, cx : cx + 2]
        special = np.logical_and(
            np.any(patch != frame_rgb, axis=2),
            np.any(patch != bg_rgb, axis=2),
        )
        if int(special.sum()) == 4:
            marker_hits.append(index)
    if len(marker_hits) != 1:
        raise ValueError(f"expected one complete corner marker, found {len(marker_hits)}")
    rotation = marker_hits[0]

    shown = np.zeros((GRID_SIZE, GRID_SIZE), dtype=bool)
    for row in range(GRID_SIZE):
        for col in range(GRID_SIZE):
            px = x0 + 2 + col * 4
            py = y0 + 2 + row * 4
            patch = array[py : py + 4, px : px + 4]
            background_pixels = np.all(patch == bg_rgb, axis=2)
            shown[row, col] = int((~background_pixels).sum()) >= 12
    return np.rot90(shown, -rotation)


def _measure(image: Image.Image) -> tuple[np.ndarray, dict[str, int]]:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[2] != 3 or min(array.shape[:2]) < 64:
        raise ValueError("image is not a usable RGB raster")
    border = np.concatenate((array[0], array[-1], array[:, 0], array[:, -1]), axis=0)
    colors, counts = np.unique(border, axis=0, return_counts=True)
    background = tuple(int(v) for v in colors[int(np.argmax(counts))])
    frame_color, boxes = _locate_frames(array, background)
    left = _recover_panel(array, boxes[0], frame_color, background)
    right = _recover_panel(array, boxes[1], frame_color, background)
    combined = left & right
    distances = {
        name: int(np.count_nonzero(combined != template))
        for name, template in TEMPLATES.items()
    }
    return combined, distances


def decision_with_margin(image: Image.Image) -> tuple[str, float]:
    """Return nearest glyph and runner-up Hamming gap from final pixels alone."""

    _, distances = _measure(image)
    ranked = sorted(distances.items(), key=lambda item: (item[1], SYMBOLS.index(item[0])))
    gap = float(ranked[1][1] - ranked[0][1])
    if gap <= 0:
        raise ValueError("intersection is tied between glyph templates")
    return ranked[0][0], gap


def decision_from_image(image: Image.Image) -> str:
    return decision_with_margin(image)[0]
