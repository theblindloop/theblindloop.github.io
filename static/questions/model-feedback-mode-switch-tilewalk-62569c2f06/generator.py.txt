"""Deterministic renderer for a two-mode, landing-switched tile automaton."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

CANVAS = (660, 660)
N, CELL, X0, Y0, STEPS = 5, 100, 80, 80, 10
DIRS = ((0, -1), (1, 0), (0, 1), (-1, 0))
MODE_COLORS = ((25, 105, 220), (190, 45, 170))
GOAL_COLORS = {"amber": (235, 150, 20), "cyan": (0, 165, 185), "violet": (125, 70, 195)}


def _inside(cell):
    return 0 <= cell[0] < N and 0 <= cell[1] < N


def _step(cell, direction):
    dx, dy = DIRS[direction]
    return cell[0] + dx, cell[1] + dy


def _trace(instructions, switches, start, start_mode, *, rule="correct"):
    cell, mode = start, start_mode
    cells, states = [cell], [(cell, mode)]
    for _ in range(STEPS):
        departed = cell
        use_mode = start_mode if rule == "never_switch" else mode
        cell = _step(cell, instructions[cell[1]][cell[0]][use_mode])
        if not _inside(cell):
            raise ValueError("walk leaves grid")
        if rule == "correct" and switches[cell[1]][cell[0]]:
            mode ^= 1
        elif rule == "depart_after" and switches[departed[1]][departed[0]]:
            mode ^= 1
        cells.append(cell)
        states.append((cell, mode))
    return cell, mode, cells, states


def _construct(rng, start, start_mode):
    """Find a path that revisits one tile in the opposite mode."""
    for _attempt in range(4000):
        switches = [[rng.random() < 0.28 for _ in range(N)] for _ in range(N)]
        switches[start[1]][start[0]] = False
        instructions = [[[rng.randrange(4), rng.randrange(4)] for _x in range(N)] for _y in range(N)]
        trail, states = [start], [(start, start_mode)]
        assignments = {}

        def visit(cell, mode):
            if len(trail) == STEPS + 1:
                repeated = any(trail[i] == trail[j] and states[i][1] != states[j][1]
                               for i in range(len(trail)) for j in range(i))
                used_switches = sum(switches[y][x] for x, y in trail[1:])
                return repeated and used_switches >= 2
            key = (cell, mode)
            choices = list(range(4))
            rng.shuffle(choices)
            for direction in choices:
                if key in assignments and assignments[key] != direction:
                    continue
                nxt = _step(cell, direction)
                if not _inside(nxt):
                    continue
                nmode = mode ^ int(switches[nxt[1]][nxt[0]])
                if (nxt, nmode) in states:
                    continue
                prior = assignments.get(key)
                assignments[key] = direction
                trail.append(nxt)
                states.append((nxt, nmode))
                if visit(nxt, nmode):
                    return True
                trail.pop(); states.pop()
                if prior is None:
                    assignments.pop(key, None)
            return False

        if not visit(start, start_mode):
            continue
        for (tile, active), direction in assignments.items():
            instructions[tile[1]][tile[0]][active] = direction
        try:
            correct = _trace(instructions, switches, start, start_mode)
            no_switch = _trace(instructions, switches, start, start_mode, rule="never_switch")
            current_first = _trace(instructions, switches, start, start_mode, rule="depart_after")
        except ValueError:
            continue
        decoys = [no_switch[0], current_first[0]]
        if len(set(decoys + [correct[0]])) != 3:
            continue
        if correct[0] in correct[2][:-1]:
            continue
        if any(decoy in correct[2][:-1] for decoy in decoys):
            continue
        return instructions, switches
    raise RuntimeError("could not construct a mode-switch walk")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0xA17E5B19)
    side = int(seed) % 4
    offset = rng.randrange(1, N - 1)
    start = ((offset, 0), (N - 1, offset), (offset, N - 1), (0, offset))[side]
    start_mode = (int(seed) // 3) % 2
    instructions, switches = _construct(rng, start, start_mode)
    correct = _trace(instructions, switches, start, start_mode)
    wrong_a = _trace(instructions, switches, start, start_mode, rule="never_switch")
    wrong_b = _trace(instructions, switches, start, start_mode, rule="depart_after")
    answer = ("amber", "cyan", "violet")[int(seed) % 3]
    others = [name for name in GOAL_COLORS if name != answer]
    rng.shuffle(others)
    return {"instructions": instructions, "switches": switches, "start": list(start),
            "start_mode": start_mode,
            "goals": {answer: list(correct[0]), others[0]: list(wrong_a[0]), others[1]: list(wrong_b[0])}}


def _simulate(scene, rule="correct"):
    return _trace(scene["instructions"], scene["switches"], tuple(scene["start"]),
                  int(scene["start_mode"]), rule=rule)


def analytic_gold(scene):
    endpoint = _simulate(scene)[0]
    matches = [name for name, cell in scene["goals"].items() if tuple(cell) == endpoint]
    if len(matches) != 1:
        raise ValueError("endpoint does not have exactly one goal")
    return matches[0]


def margin(scene):
    return 1.0 if not is_quarantined(scene) else 0.0


def is_quarantined(scene):
    try:
        ins, switches = scene["instructions"], scene["switches"]
        if len(ins) != N or any(len(row) != N for row in ins): return True
        if any(len(pair) != 2 or any(d not in range(4) for d in pair) for row in ins for pair in row): return True
        if len(switches) != N or any(len(row) != N or any(type(v) is not bool for v in row) for row in switches): return True
        start, mode = tuple(scene["start"]), int(scene["start_mode"])
        if not _inside(start) or mode not in (0, 1) or switches[start[1]][start[0]]: return True
        endpoint, _, trail, states = _simulate(scene)
        if len(states) != len(set(states)): return True
        if not any(trail[i] == trail[j] and states[i][1] != states[j][1]
                   for i in range(len(trail)) for j in range(i)): return True
        goal_cells = [tuple(cell) for cell in scene["goals"].values()]
        if set(scene["goals"]) != set(GOAL_COLORS) or len(set(goal_cells)) != 3: return True
        if endpoint not in goal_cells or any(cell in goal_cells for cell in trail[:-1]): return True
        wrong = {_simulate(scene, "never_switch")[0], _simulate(scene, "depart_after")[0]}
        if wrong != set(goal_cells) - {endpoint}: return True
        return not (start[0] in (0, N - 1) or start[1] in (0, N - 1))
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene):
    return []


def _arrow(draw, center, direction, color):
    cx, cy = center
    if direction == 0:
        draw.line((cx, cy + 12, cx, cy - 5), fill=color, width=8)
        draw.polygon(((cx, cy - 17), (cx - 10, cy - 3), (cx + 10, cy - 3)), fill=color)
    elif direction == 1:
        draw.line((cx - 12, cy, cx + 5, cy), fill=color, width=8)
        draw.polygon(((cx + 17, cy), (cx + 3, cy - 10), (cx + 3, cy + 10)), fill=color)
    elif direction == 2:
        draw.line((cx, cy - 12, cx, cy + 5), fill=color, width=8)
        draw.polygon(((cx, cy + 17), (cx - 10, cy + 3), (cx + 10, cy + 3)), fill=color)
    else:
        draw.line((cx + 12, cy, cx - 5, cy), fill=color, width=8)
        draw.polygon(((cx - 17, cy), (cx - 3, cy - 10), (cx - 3, cy + 10)), fill=color)


def _draw_start(draw, start, mode):
    x, y = start
    cx, cy = X0 + x * CELL + CELL // 2, Y0 + y * CELL + CELL // 2
    badge = MODE_COLORS[mode]
    if y == 0:
        draw.line((cx, 18, cx, 58), fill="black", width=10); draw.polygon(((cx-17,52),(cx+17,52),(cx,72)), fill="black")
        draw.ellipse((cx+23,25,cx+49,51), fill=badge, outline="black", width=3)
    elif x == N - 1:
        draw.line((642,cy,602,cy), fill="black", width=10); draw.polygon(((608,cy-17),(608,cy+17),(588,cy)), fill="black")
        draw.ellipse((609,cy+23,635,cy+49), fill=badge, outline="black", width=3)
    elif y == N - 1:
        draw.line((cx,642,cx,602), fill="black", width=10); draw.polygon(((cx-17,608),(cx+17,608),(cx,588)), fill="black")
        draw.ellipse((cx+23,609,cx+49,635), fill=badge, outline="black", width=3)
    else:
        draw.line((18,cy,58,cy), fill="black", width=10); draw.polygon(((52,cy-17),(52,cy+17),(72,cy)), fill="black")
        draw.ellipse((25,cy+23,51,cy+49), fill=badge, outline="black", width=3)


def render(scene):
    image = Image.new("RGB", CANVAS, (248, 248, 246)); draw = ImageDraw.Draw(image)
    for y in range(N):
        for x in range(N):
            fill = (255,255,255) if (x+y)%2 == 0 else (243,246,248)
            draw.rectangle((X0+x*CELL,Y0+y*CELL,X0+(x+1)*CELL,Y0+(y+1)*CELL), fill=fill)
    for name, cell in scene["goals"].items():
        x, y = cell
        draw.rectangle((X0+x*CELL+6,Y0+y*CELL+6,X0+(x+1)*CELL-6,Y0+(y+1)*CELL-6),
                       outline=GOAL_COLORS[name], width=8)
    for i in range(N+1):
        draw.line((X0+i*CELL,Y0,X0+i*CELL,Y0+N*CELL), fill=(45,50,55), width=3)
        draw.line((X0,Y0+i*CELL,X0+N*CELL,Y0+i*CELL), fill=(45,50,55), width=3)
    for y in range(N):
        for x in range(N):
            cx, cy = X0+x*CELL+CELL//2-8, Y0+y*CELL+CELL//2
            _arrow(draw,(cx,cy-23),scene["instructions"][y][x][0],MODE_COLORS[0])
            _arrow(draw,(cx,cy+23),scene["instructions"][y][x][1],MODE_COLORS[1])
            if scene["switches"][y][x]:
                sx = X0+x*CELL+82
                draw.ellipse((sx-9,cy-9,sx+9,cy+9), outline=(10,12,15), width=5)
    _draw_start(draw, tuple(scene["start"]), int(scene["start_mode"]))
    return image
