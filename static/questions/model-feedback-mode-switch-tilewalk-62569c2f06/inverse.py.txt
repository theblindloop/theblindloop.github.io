"""Independent image-only decoder for the mode-switch tile walk."""

from __future__ import annotations

import numpy as np
from PIL import Image

CANVAS = (660, 660)
N, CELL, X0, Y0, STEPS = 5, 100, 80, 80, 10
DIRS = ((0, -1), (1, 0), (0, 1), (-1, 0))
MODE_COLORS = (np.array((25, 105, 220)), np.array((190, 45, 170)))
GOAL_COLORS = {"amber": np.array((235,150,20)), "cyan": np.array((0,165,185)),
               "violet": np.array((125,70,195))}


class OracleError(ValueError):
    pass


def _near(crop, rgb, distance=25):
    delta = crop.astype(np.int32) - rgb.astype(np.int32)
    return np.sum(delta * delta, axis=2) < distance * distance


def _dark(crop):
    return np.all(crop < 70, axis=2)


def _decode_start(arr):
    regions = {
        0: (slice(5, 76), slice(X0, X0 + N*CELL)),
        1: (slice(Y0, Y0 + N*CELL), slice(584, 655)),
        2: (slice(584, 655), slice(X0, X0 + N*CELL)),
        3: (slice(Y0, Y0 + N*CELL), slice(5, 76)),
    }
    hits = []
    for side, (ys, xs) in regions.items():
        crop = arr[ys, xs]
        yy, xx = np.nonzero(_dark(crop))
        if len(xx) > 500:
            hits.append((side, float(np.median(xx + xs.start)), float(np.median(yy + ys.start)), crop))
    if len(hits) != 1:
        raise OracleError("start arrow missing or duplicated")
    side, cx, cy, crop = hits[0]
    counts = [int(_near(crop, rgb).sum()) for rgb in MODE_COLORS]
    modes = [i for i, count in enumerate(counts) if count > 250]
    if len(modes) != 1:
        raise OracleError("start mode badge missing or duplicated")
    if side in (0, 2):
        start = (int((cx-X0)//CELL), 0 if side == 0 else N-1)
    else:
        start = (N-1 if side == 1 else 0, int((cy-Y0)//CELL))
    if not (0 <= start[0] < N and 0 <= start[1] < N):
        raise OracleError("start arrow not aligned")
    return start, modes[0]


def _arrow_direction(arr, cx, cy, rgb):
    crop = arr[cy-20:cy+21, cx-20:cx+21]
    yy, xx = np.nonzero(_near(crop, rgb))
    if not 210 <= len(xx) <= 360:
        raise OracleError(f"unreadable arrow ink count {len(xx)}")
    width, height = int(xx.max()-xx.min()), int(yy.max()-yy.min())
    mean_x, mean_y = float(xx.mean()-20), float(yy.mean()-20)
    if height > width + 5:
        return 0 if mean_y < -0.8 else 2
    if width > height + 5:
        return 3 if mean_x < -0.8 else 1
    raise OracleError("arrow orientation ambiguous")


def _decode_grid(arr):
    instructions, switches = [], []
    for y in range(N):
        irow, srow = [], []
        for x in range(N):
            cx, cy = X0+x*CELL+CELL//2-8, Y0+y*CELL+CELL//2
            irow.append([_arrow_direction(arr,cx,cy-23,MODE_COLORS[0]),
                         _arrow_direction(arr,cx,cy+23,MODE_COLORS[1])])
            sx = X0+x*CELL+82
            count = int(_dark(arr[cy-12:cy+13, sx-12:sx+13]).sum())
            if count < 30: srow.append(False)
            elif 120 <= count <= 300: srow.append(True)
            else: raise OracleError(f"unreadable switch ring at {(x,y)}: {count}")
        instructions.append(irow); switches.append(srow)
    return instructions, switches


def _decode_goals(arr):
    found = {}
    for y in range(N):
        for x in range(N):
            crop = arr[Y0+y*CELL+3:Y0+(y+1)*CELL-3, X0+x*CELL+3:X0+(x+1)*CELL-3]
            matches = [name for name, rgb in GOAL_COLORS.items() if int(_near(crop,rgb).sum()) > 1800]
            if len(matches) > 1: raise OracleError("multiple goal colors on tile")
            if matches: found[(x,y)] = matches[0]
    if len(found) != 3 or set(found.values()) != set(GOAL_COLORS):
        raise OracleError("expected one goal of each color")
    return found


def decision_from_image(image: Image.Image) -> str:
    if image.size != CANVAS: raise OracleError(f"wrong image size {image.size}")
    arr = np.asarray(image.convert("RGB"))
    cell, mode = _decode_start(arr)
    instructions, switches = _decode_grid(arr)
    goals = _decode_goals(arr)
    trail, states = [cell], [(cell,mode)]
    for _ in range(STEPS):
        direction = instructions[cell[1]][cell[0]][mode]
        dx, dy = DIRS[direction]
        cell = cell[0]+dx, cell[1]+dy
        if not (0 <= cell[0] < N and 0 <= cell[1] < N): raise OracleError("walk leaves grid")
        if switches[cell[1]][cell[0]]: mode ^= 1
        if (cell,mode) in states: raise OracleError("walk repeats full state")
        trail.append(cell); states.append((cell,mode))
    if not any(trail[i] == trail[j] and states[i][1] != states[j][1]
               for i in range(len(trail)) for j in range(i)):
        raise OracleError("walk lacks opposite-mode revisit")
    if cell not in goals: raise OracleError("final tile is not a goal")
    if any(tile in goals for tile in trail[:-1]): raise OracleError("goal visited early")
    return goals[cell]
