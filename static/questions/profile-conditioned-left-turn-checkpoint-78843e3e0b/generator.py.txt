"""Deterministic scene model and rasterizer for ``left_turn_checkpoint``.

The route is an undirected black stroke in the raster.  Green and purple endpoint
disks visibly establish its traversal direction.  Exactly three colored disks sit
on bends; exactly one of those bends is leftward under green-to-purple travel.
"""

from __future__ import annotations

import copy
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
LINE_WIDTH = 11
MARKER_RADIUS = 14

PALETTE = {
    "background": (250, 249, 246),
    "black": (28, 31, 36),
    "green": (28, 154, 91),
    "purple": (132, 74, 175),
    "red": (216, 55, 56),
    "blue": (48, 105, 190),
    "orange": (238, 137, 35),
}
ANSWER_COLORS = ("red", "blue", "orange")


def _transform(point: list[int], code: int) -> list[int]:
    """Apply one of the eight square symmetries around the 256px center."""
    x, y = point
    if code == 0:
        return [x, y]
    if code == 1:
        return [WIDTH - x, y]
    if code == 2:
        return [x, HEIGHT - y]
    if code == 3:
        return [WIDTH - x, HEIGHT - y]
    if code == 4:
        return [y, x]
    if code == 5:
        return [WIDTH - y, x]
    if code == 6:
        return [y, HEIGHT - x]
    return [WIDTH - y, HEIGHT - x]


def _oriented_path(scene: dict) -> list[list[int]]:
    path = [list(point) for point in scene["path"]]
    start = list(scene["start"])
    finish = list(scene["finish"])
    if path[0] == start and path[-1] == finish:
        return path
    if path[-1] == start and path[0] == finish:
        return list(reversed(path))
    raise ValueError("explicit START and FINISH must be the two route endpoints")


def _turn_score(path: list[list[int]], center: list[int]) -> float:
    try:
        index = path.index(list(center))
    except ValueError as exc:
        raise ValueError("checkpoint is not a route vertex") from exc
    if not 0 < index < len(path) - 1:
        raise ValueError("checkpoint cannot be a route endpoint")
    ax, ay = path[index - 1]
    bx, by = path[index]
    cx, cy = path[index + 1]
    incoming = (bx - ax, by - ay)
    outgoing = (cx - bx, cy - by)
    if not ((incoming[0] == 0) ^ (incoming[1] == 0)):
        raise ValueError("incoming arm is not axis aligned")
    if not ((outgoing[0] == 0) ^ (outgoing[1] == 0)):
        raise ValueError("outgoing arm is not axis aligned")
    in_len = abs(incoming[0]) + abs(incoming[1])
    out_len = abs(outgoing[0]) + abs(outgoing[1])
    # In screen coordinates y grows downward, so negative cross product is left.
    cross = incoming[0] * outgoing[1] - incoming[1] * outgoing[0]
    return -cross / float(in_len * out_len)


def _segment_clearance(path: list[list[int]]) -> float:
    """Minimum L-infinity separation between nonincident axis-aligned segments."""
    segments = list(zip(path, path[1:]))
    best = 10_000.0
    for i, (a, b) in enumerate(segments):
        ax0, ax1 = sorted((a[0], b[0]))
        ay0, ay1 = sorted((a[1], b[1]))
        for j, (c, d) in enumerate(segments):
            if j <= i + 1:
                continue
            cx0, cx1 = sorted((c[0], d[0]))
            cy0, cy1 = sorted((c[1], d[1]))
            dx = max(ax0 - cx1, cx0 - ax1, 0)
            dy = max(ay0 - cy1, cy0 - ay1, 0)
            best = min(best, max(dx, dy))
    return best


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    # A monotone six-link staircase is self-avoiding by construction. Independent
    # jitter changes every link length while retaining generous raster clearance.
    xs = [58 + rng.randint(0, 16), 158 + rng.randint(-14, 14),
          278 + rng.randint(-14, 14), 430 + rng.randint(-14, 14)]
    ys = [438 + rng.randint(-14, 0), 326 + rng.randint(-14, 14),
          202 + rng.randint(-14, 14), 66 + rng.randint(0, 14)]
    # The single transform is shared by all vertices.
    rng2 = random.Random(int(seed) ^ 0x5A17)
    transform_code = rng2.randrange(8)
    canonical = [
        [xs[0], ys[0]], [xs[1], ys[0]], [xs[1], ys[1]],
        [xs[2], ys[1]], [xs[2], ys[2]], [xs[3], ys[2]],
        [xs[3], ys[3]],
    ]
    path = [_transform(point, transform_code) for point in canonical]
    if rng2.randrange(2):
        start, finish = list(path[-1]), list(path[0])
    else:
        start, finish = list(path[0]), list(path[-1])

    oriented = path if path[0] == start else list(reversed(path))
    left_vertices = [point for point in oriented[1:-1] if _turn_score(oriented, point) > 0.5]
    right_vertices = [point for point in oriented[1:-1] if _turn_score(oriented, point) < -0.5]
    chosen_left = list(rng2.choice(left_vertices))
    chosen_rights = [list(point) for point in rng2.sample(right_vertices, 2)]

    answer_color = ANSWER_COLORS[int(seed) % len(ANSWER_COLORS)]
    other_colors = [color for color in ANSWER_COLORS if color != answer_color]
    rng2.shuffle(other_colors)
    checkpoints = [{"center": chosen_left, "color": answer_color}]
    checkpoints.extend(
        {"center": center, "color": color}
        for center, color in zip(chosen_rights, other_colors)
    )
    rng2.shuffle(checkpoints)

    # Randomize only the stored polyline orientation. This is the exercised
    # representation alias and does not alter visible endpoint roles.
    stored_path = list(reversed(path)) if rng2.randrange(2) else list(path)
    return {
        "path": stored_path,
        "start": start,
        "finish": finish,
        "checkpoints": checkpoints,
        "width": WIDTH,
        "height": HEIGHT,
        "line_width": LINE_WIDTH,
        "marker_radius": MARKER_RADIUS,
    }


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    line_width = int(scene["line_width"])
    marker_radius = int(scene["marker_radius"])
    path = [list(point) for point in scene["path"]]
    start = list(scene["start"])
    finish = list(scene["finish"])
    checkpoints = list(scene["checkpoints"])

    image = Image.new("RGB", (width, height), PALETTE["background"])
    draw = ImageDraw.Draw(image)
    radius = line_width // 2
    # Segment-by-segment plus explicit joints makes reversal byte-identical.
    for a, b in zip(path, path[1:]):
        draw.line([tuple(a), tuple(b)], fill=PALETTE["black"], width=line_width)
    for x, y in path:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=PALETTE["black"])

    def disk(center: Iterable[int], color: str) -> None:
        x, y = center
        r = marker_radius
        draw.ellipse((x - r, y - r, x + r, y + r), fill=PALETTE[color])

    disk(start, "green")
    disk(finish, "purple")
    for checkpoint in checkpoints:
        disk(checkpoint["center"], checkpoint["color"])
    return image


def analytic_gold(scene: dict) -> str:
    path = _oriented_path(scene)
    scored = [
        (_turn_score(path, list(checkpoint["center"])), str(checkpoint["color"]))
        for checkpoint in scene["checkpoints"]
    ]
    winners = [color for score, color in scored if score > 0.5]
    if len(winners) != 1:
        raise ValueError("scene must have exactly one left-turn colored checkpoint")
    return winners[0]


def margin(scene: dict) -> float:
    path = _oriented_path(scene)
    scores = sorted(
        (_turn_score(path, list(checkpoint["center"])) for checkpoint in scene["checkpoints"]),
        reverse=True,
    )
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    try:
        path = _oriented_path(scene)
        if len(path) < 4 or len(scene["checkpoints"]) != 3:
            return True
        if len({item["color"] for item in scene["checkpoints"]}) != 3:
            return True
        if _segment_clearance(path) < 42:
            return True
        for a, b in zip(path, path[1:]):
            if not ((a[0] == b[0]) ^ (a[1] == b[1])):
                return True
            if abs(a[0] - b[0]) + abs(a[1] - b[1]) < 60:
                return True
        return analytic_gold(scene) not in ANSWER_COLORS or margin(scene) < 1.5
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["path"] = list(reversed(twin["path"]))
    return [("reverse_stored_polyline_order", twin)]
