"""Sampler and renderer for two alternating walkers on shared mutable tiles."""

from __future__ import annotations

import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 920, 760
N, CELL = 5, 90
GRID_X, GRID_Y = 235, 164
BACKGROUND = (247, 246, 241)
INK = (28, 36, 48)
GOLD = (244, 190, 51)
BLUE = (54, 128, 194)
RED = (211, 46, 57)
VIOLET = (124, 58, 188)
WHITE = (255, 255, 255)
DECISIONS = ("token-red", "token-violet")
DIRS = ((-1, 0), (0, 1), (1, 0), (0, -1))


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    filename = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    path = Path("/usr/share/fonts/truetype/dejavu") / filename
    return ImageFont.truetype(str(path), size) if path.exists() else ImageFont.load_default()


def evolve(tiles: list[list[int]], red_start: list[int], red_direction: int,
           violet_start: list[int], violet_direction: int, limit: int = 80) -> dict:
    board = [[int(v) for v in row] for row in tiles]
    if len(board) != N or any(len(row) != N for row in board):
        raise ValueError("five by five board required")
    if any(v not in (0, 1) for row in board for v in row):
        raise ValueError("binary tiles required")
    positions = [list(map(int, red_start)), list(map(int, violet_start))]
    headings = [int(red_direction), int(violet_direction)]
    if positions[0] == positions[1]:
        raise ValueError("token starts must differ")
    if any(not (0 <= p[0] < N and 0 <= p[1] < N) for p in positions) or any(h not in range(4) for h in headings):
        raise ValueError("invalid token state")
    visits = [0, 0]
    last_writer: dict[tuple[int, int], int] = {}
    cross_reads = 0
    trace: list[list[int]] = []
    for action in range(int(limit)):
        actor = action % 2
        row, col = positions[actor]
        if last_writer.get((row, col)) == 1 - actor:
            cross_reads += 1
        old = board[row][col]
        headings[actor] = (headings[actor] + (1 if old == 0 else -1)) % 4
        board[row][col] ^= 1
        last_writer[(row, col)] = actor
        visits[actor] += 1
        trace.append([actor, row, col, old, headings[actor]])
        dr, dc = DIRS[headings[actor]]
        positions[actor] = [row + dr, col + dc]
        nr, nc = positions[actor]
        if nr < 0 or nr >= N or nc < 0 or nc >= N:
            return {
                "decision": DECISIONS[actor], "actions": action + 1,
                "cross_reads": cross_reads, "visits": visits, "trace": trace,
            }
    return {"decision": "abstain", "actions": int(limit), "cross_reads": cross_reads, "visits": visits, "trace": trace}


def private_evolve(tiles: list[list[int]], red_start: list[int], red_direction: int,
                   violet_start: list[int], violet_direction: int, limit: int = 80) -> str:
    boards = [[[int(v) for v in row] for row in tiles] for _ in range(2)]
    positions = [list(map(int, red_start)), list(map(int, violet_start))]
    headings = [int(red_direction), int(violet_direction)]
    for action in range(int(limit)):
        actor = action % 2
        row, col = positions[actor]
        old = boards[actor][row][col]
        headings[actor] = (headings[actor] + (1 if old == 0 else -1)) % 4
        boards[actor][row][col] ^= 1
        dr, dc = DIRS[headings[actor]]
        positions[actor] = [row + dr, col + dc]
        nr, nc = positions[actor]
        if nr < 0 or nr >= N or nc < 0 or nc >= N:
            return DECISIONS[actor]
    return "abstain"


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x53484152)
    target = DECISIONS[int(seed) % 2]
    for _ in range(300000):
        tiles = [[rng.randrange(2) for _ in range(N)] for _ in range(N)]
        starts = [[rng.randrange(1, N - 1), rng.randrange(1, N - 1)] for _ in range(2)]
        if starts[0] == starts[1]:
            continue
        headings = [rng.randrange(4), rng.randrange(4)]
        outcome = evolve(tiles, starts[0], headings[0], starts[1], headings[1])
        if outcome["decision"] != target or not (18 <= outcome["actions"] <= 34):
            continue
        if outcome["cross_reads"] < 2 or min(outcome["visits"]) < 7:
            continue
        if private_evolve(tiles, starts[0], headings[0], starts[1], headings[1]) == target:
            continue
        return {
            "tiles": tiles, "red_start": starts[0], "red_direction": headings[0],
            "violet_start": starts[1], "violet_direction": headings[1],
        }
    raise RuntimeError("could not sample consequential shared-board scene")


def _validate(scene: dict) -> tuple[list[list[int]], list[int], int, list[int], int]:
    tiles = [[int(v) for v in row] for row in scene["tiles"]]
    rs, rd = [int(v) for v in scene["red_start"]], int(scene["red_direction"])
    vs, vd = [int(v) for v in scene["violet_start"]], int(scene["violet_direction"])
    evolve(tiles, rs, rd, vs, vd)
    return tiles, rs, rd, vs, vd


def analytic_gold(scene: dict) -> str:
    try:
        return str(evolve(*_validate(scene))["decision"])
    except (KeyError, TypeError, ValueError, IndexError):
        return "abstain"


def margin(scene: dict) -> float:
    return 1.0 if analytic_gold(scene) in DECISIONS else 0.0


def is_quarantined(scene: dict) -> bool:
    return analytic_gold(scene) == "abstain"


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _arrow_points(cx: int, cy: int, direction: int, scale: int = 36) -> list[tuple[int, int]]:
    points = [(-.48, -.24), (.04, -.24), (.04, -.50), (.80, 0), (.04, .50), (.04, .24), (-.48, .24)]
    result = []
    for x, y in points:
        for _ in range((int(direction) + 3) % 4):
            x, y = -y, x
        result.append((round(cx + x * scale), round(cy + y * scale)))
    return result


def render(scene: dict) -> Image.Image:
    tiles, rs, rd, vs, vd = _validate(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((WIDTH // 2, 22), "SHARED TILES · ALTERNATING TURNS", font=_font(28, True), fill=INK, anchor="ma")
    draw.text((WIDTH // 2, 62), "RED moves, then VIOLET; repeat. Stop when either token exits.", font=_font(19, True), fill=INK, anchor="ma")

    draw.rounded_rectangle((34, 145, 205, 310), radius=14, fill=WHITE, outline=INK, width=3)
    draw.text((119, 160), "EACH MOVE", font=_font(18, True), fill=INK, anchor="ma")
    draw.rectangle((53, 194, 87, 228), fill=GOLD, outline=INK, width=2)
    draw.text((99, 211), "turn RIGHT", font=_font(16, True), fill=INK, anchor="lm")
    draw.rectangle((53, 240, 87, 274), fill=BLUE, outline=INK, width=2)
    draw.text((99, 257), "turn LEFT", font=_font(16, True), fill=INK, anchor="lm")
    draw.text((119, 292), "then FLIP + MOVE", font=_font(15, True), fill=INK, anchor="mm")

    draw.rounded_rectangle((712, 145, 886, 310), radius=14, fill=WHITE, outline=INK, width=3)
    draw.text((799, 160), "TURN ORDER", font=_font(18, True), fill=INK, anchor="ma")
    draw.polygon(_arrow_points(755, 211, 1, 29), fill=RED)
    draw.text((785, 211), "1st RED", font=_font(17, True), fill=INK, anchor="lm")
    draw.polygon(_arrow_points(755, 257, 1, 29), fill=VIOLET)
    draw.text((785, 257), "2nd VIOLET", font=_font(17, True), fill=INK, anchor="lm")
    draw.text((799, 293), "same board", font=_font(16, True), fill=INK, anchor="mm")

    for row in range(N):
        for col in range(N):
            x0, y0 = GRID_X + col * CELL, GRID_Y + row * CELL
            draw.rectangle((x0, y0, x0 + CELL, y0 + CELL), fill=GOLD if tiles[row][col] == 0 else BLUE, outline=INK, width=4)

    for label, (cx, cy) in (
        ("N", (GRID_X + N * CELL // 2, GRID_Y - 38)),
        ("E", (GRID_X + N * CELL + 38, GRID_Y + N * CELL // 2)),
        ("S", (GRID_X + N * CELL // 2, GRID_Y + N * CELL + 38)),
        ("W", (GRID_X - 38, GRID_Y + N * CELL // 2)),
    ):
        draw.rounded_rectangle((cx - 25, cy - 23, cx + 25, cy + 23), radius=9, fill=WHITE, outline=INK, width=3)
        draw.text((cx, cy), label, font=_font(23, True), fill=INK, anchor="mm")

    for start, direction, color in ((rs, rd, RED), (vs, vd, VIOLET)):
        row, col = start
        cx, cy = GRID_X + col * CELL + CELL // 2, GRID_Y + row * CELL + CELL // 2
        arrow = _arrow_points(cx, cy, direction)
        draw.polygon(arrow, fill=color)
        draw.line(arrow + [arrow[0]], fill=WHITE, width=4, joint="curve")

    draw.text((WIDTH // 2, 678), "A flip changes the tile for BOTH tokens.", font=_font(20, True), fill=INK, anchor="ma")
    draw.text((WIDTH // 2, 720), "Which token exits first: RED or VIOLET?", font=_font(25, True), fill=INK, anchor="ma")
    return image
