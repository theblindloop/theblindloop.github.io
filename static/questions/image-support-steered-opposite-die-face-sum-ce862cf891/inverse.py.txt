"""Independent image-only inverse for the outlined die-face pair."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int, float, float]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out = []
    for y, x in zip(*np.nonzero(mask & ~seen), strict=True):
        if seen[y, x]:
            continue
        q = deque([(int(y), int(x))])
        seen[y, x] = True
        xs, ys = [], []
        while q:
            yy, xx = q.popleft()
            xs.append(xx); ys.append(yy)
            for ny, nx in ((yy-1, xx), (yy+1, xx), (yy, xx-1), (yy, xx+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    q.append((ny, nx))
        area = len(xs)
        out.append((min(xs), min(ys), max(xs)+1, max(ys)+1, area, sum(xs)/area, sum(ys)/area))
    return out


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    # All palette variants use dark chromatic ink; gray labels are excluded by
    # requiring both low brightness and a modest channel spread.
    mx = arr.max(axis=2)
    mn = arr.min(axis=2)
    ink = (mx < 135) & ((mx - mn) > 14)
    comps = _components(ink)
    borders = []
    for x0, y0, x1, y1, area, cx, cy in comps:
        bw, bh = x1-x0, y1-y0
        # Rounded face outline is the only near-square component this large.
        # The lower area bound also checks that the enclosing frame remains
        # materially complete; destructive erasure of either distant face must
        # not be mistaken for an intact counting region.
        if 130 <= bw <= 158 and 130 <= bh <= 158 and 2850 <= area <= 4500 and 0.82 <= bw/bh <= 1.22:
            borders.append((x0, y0, x1, y1, cx))
    if len(borders) != 2:
        return "abstain"
    borders.sort(key=lambda v: v[4])
    if borders[1][4] - borders[0][4] < 250:
        return "abstain"

    counts = []
    for bx0, by0, bx1, by1, _ in borders:
        n = 0
        for x0, y0, x1, y1, area, cx, cy in comps:
            bw, bh = x1-x0, y1-y0
            if bx0 + 14 < cx < bx1 - 14 and by0 + 14 < cy < by1 - 14:
                fill = area / max(1, bw * bh)
                if 16 <= bw <= 26 and 16 <= bh <= 26 and 250 <= area <= 520 and fill > 0.62:
                    n += 1
        if not (1 <= n <= 6):
            return "abstain"
        counts.append(n)
    return "yes" if sum(counts) == 7 else "no"
