"""Deterministic renderer for two visibly separated die faces."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 640, 384
FACE_SIZE = 144
PIP_R = 11
TOTAL_DARK_CIRCLES = 14
PALETTES = (
    ((25, 55, 88), (238, 244, 247), (238, 178, 67)),
    ((85, 35, 62), (249, 242, 232), (73, 152, 133)),
    ((34, 72, 55), (244, 241, 249), (119, 103, 171)),
)

PIP_POSITIONS = {
    1: ((0.50, 0.50),),
    2: ((0.29, 0.29), (0.71, 0.71)),
    3: ((0.29, 0.29), (0.50, 0.50), (0.71, 0.71)),
    4: ((0.29, 0.29), (0.71, 0.29), (0.29, 0.71), (0.71, 0.71)),
    5: ((0.29, 0.29), (0.71, 0.29), (0.50, 0.50), (0.29, 0.71), (0.71, 0.71)),
    6: ((0.29, 0.25), (0.29, 0.50), (0.29, 0.75), (0.71, 0.25), (0.71, 0.50), (0.71, 0.75)),
}


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    # Alternate class by seed while sampling diverse matched face values.
    is_control = (seed % 2 == 0)
    if is_control:
        left = 1 + rng.randrange(6)
        right = 7 - left
    else:
        choices = [(a, b) for a in range(1, 7) for b in range(1, 7) if a + b in (5, 6, 8, 9)]
        left, right = choices[rng.randrange(len(choices))]
    shift_x = rng.randrange(-10, 11)
    shift_y = rng.randrange(-8, 9)
    palette = rng.randrange(len(PALETTES))
    nuisance_count = TOTAL_DARK_CIRCLES - left - right
    slots = list(range(16))
    rng.shuffle(slots)
    return {
        "left_count": left,
        "right_count": right,
        "x_shift": shift_x,
        "y_shift": shift_y,
        "palette_index": palette,
        "distractor_slots": slots[:nuisance_count],
    }


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(name, size)


def _centered(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str, font, fill) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2), text, font=font, fill=fill)


def _face(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], count: int, ink, paper) -> None:
    x0, y0, x1, y1 = box
    factor = (x1 - x0) / FACE_SIZE
    draw.rounded_rectangle(box, radius=round(16 * factor), fill=paper, outline=ink, width=round(6 * factor))
    for fx, fy in PIP_POSITIONS[count]:
        cx = round(x0 + fx * (x1 - x0))
        cy = round(y0 + fy * (y1 - y0))
        radius = round(PIP_R * factor)
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=ink)


def render(scene: dict) -> Image.Image:
    left_count = int(scene["left_count"])
    right_count = int(scene["right_count"])
    sx = int(scene["x_shift"])
    sy = int(scene["y_shift"])
    palette_index = int(scene["palette_index"])
    distractor_slots = list(scene["distractor_slots"])
    ink, paper, accent = PALETTES[palette_index]

    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), (252, 251, 248))
    draw = ImageDraw.Draw(image)
    # Draw in logical coordinates on a high-resolution canvas.
    def S(box):
        return tuple(int(round(v * scale)) for v in box)
    ink3, paper3, accent3 = ink, paper, accent
    title_font = _font(24 * scale, True)
    label_font = _font(16 * scale, True)
    small_font = _font(13 * scale, False)

    _centered(draw, (WIDTH * scale // 2, 34 * scale), "OPPOSITE DIE FACES", title_font, ink3)
    _centered(draw, (WIDTH * scale // 2, 66 * scale), "COUNT WHAT IS VISIBLE", small_font, (92, 96, 100))

    y0 = 120 + sy
    left_box = (54 + sx, y0, 54 + sx + FACE_SIZE, y0 + FACE_SIZE)
    right_box = (442 + sx, y0, 442 + sx + FACE_SIZE, y0 + FACE_SIZE)
    _face(draw, S(left_box), left_count, ink3, paper3)
    _face(draw, S(right_box), right_count, ink3, paper3)

    # Decorative arrows visually bind the label to each face but do not join them.
    for cx in (left_box[0] + FACE_SIZE // 2, right_box[0] + FACE_SIZE // 2):
        draw.line(S((cx, 103 + sy, cx, 114 + sy)), fill=accent3, width=4 * scale)
        draw.polygon([S((cx, 118 + sy))[0:2], S((cx - 6, 109 + sy))[0:2], S((cx + 6, 109 + sy))[0:2]], fill=accent3)

    _centered(draw, ((left_box[0] + left_box[2]) * scale // 2, 288 * scale), "LEFT FACE", label_font, ink3)
    _centered(draw, ((right_box[0] + right_box[2]) * scale // 2, 288 * scale), "RIGHT FACE", label_font, ink3)

    # Constant-total nuisance circles outside the outlined faces. Their selected
    # slots and the two face counts always total fourteen dark circular marks.
    slot_xy = []
    for row_y in (322, 352):
        for col in range(8):
            slot_xy.append((215 + 30 * col, row_y))
    draw.rounded_rectangle(S((202, 306, 438, 370)), radius=10 * scale, outline=(178, 181, 181), width=2 * scale)
    _centered(draw, (320 * scale, 300 * scale), "SPARE MARKS — DO NOT COUNT", small_font, (105, 108, 109))
    for slot in distractor_slots:
        cx, cy = slot_xy[int(slot)]
        r = PIP_R
        draw.ellipse(S((cx - r, cy - r, cx + r, cy + r)), fill=ink3)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "yes" if int(scene["left_count"]) + int(scene["right_count"]) == 7 else "no"


def margin(scene: dict) -> float:
    total = int(scene["left_count"]) + int(scene["right_count"])
    return float(max(1, abs(total - 7)))


def is_quarantined(scene: dict) -> bool:
    left = int(scene["left_count"])
    right = int(scene["right_count"])
    return not (1 <= left <= 6 and 1 <= right <= 6)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
