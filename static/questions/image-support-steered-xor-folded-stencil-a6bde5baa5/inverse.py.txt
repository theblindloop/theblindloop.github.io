"""Independent image-only inverse arm for the folded-stencil world."""

from __future__ import annotations

import numpy as np
from PIL import Image

ROWS = 4
PALETTE = {
    "plaque": np.array((31, 45, 67), dtype=np.int16),
    "ink": np.array((53, 181, 176), dtype=np.int16),
    "paper": np.array((244, 235, 204), dtype=np.int16),
    "reference": np.array((196, 207, 220), dtype=np.int16),
    "ruby": np.array((211, 61, 78), dtype=np.int16),
    "green": np.array((42, 169, 103), dtype=np.int16),
    "blue": np.array((53, 116, 218), dtype=np.int16),
    "gold": np.array((235, 168, 42), dtype=np.int16),
}


def _mask(array: np.ndarray, color: np.ndarray, tolerance: int = 8) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _bbox(mask: np.ndarray, minimum: int = 100) -> tuple[int, int, int, int]:
    points = np.argwhere(mask)
    if len(points) < minimum:
        raise ValueError("required visible region is missing")
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    return int(x0), int(y0), int(x1), int(y1)


def _validate_plaque(array: np.ndarray) -> None:
    mask = _mask(array, PALETTE["plaque"])
    x0, y0, x1, y1 = _bbox(mask, 30000)
    width, height = x1 - x0 + 1, y1 - y0 + 1
    if not (330 <= width <= 370 and 330 <= height <= 370):
        raise ValueError("central plaque geometry is corrupted")
    # The question world is deliberately one thick regional plaque. Its four by
    # four interior sectors must remain substantially present before parsing.
    for gy in range(4):
        sy0 = round(y0 + height * gy / 4)
        sy1 = round(y0 + height * (gy + 1) / 4)
        for gx in range(4):
            sx0 = round(x0 + width * gx / 4)
            sx1 = round(x0 + width * (gx + 1) / 4)
            if float(mask[sy0:sy1, sx0:sx1].mean()) < 0.38:
                raise ValueError("a decisive plaque sector is erased")


def _read_grid(array: np.ndarray, rim_color: np.ndarray, cols: int) -> tuple[tuple[int, ...], ...]:
    x0, y0, x1, y1 = _bbox(_mask(array, rim_color))
    width, height = x1 - x0 + 1, y1 - y0 + 1
    expected_ratio = cols / ROWS
    if not (88 <= height <= 125 and abs(width / height - expected_ratio) < 0.25):
        raise ValueError("rim geometry is corrupted")
    border = 10
    ix0, iy0, ix1, iy1 = x0 + border, y0 + border, x1 - border, y1 - border
    cell_w = (ix1 - ix0 + 1) / cols
    cell_h = (iy1 - iy0 + 1) / ROWS
    matrix = []
    for row in range(ROWS):
        values = []
        for col in range(cols):
            cx = round(ix0 + (col + 0.5) * cell_w)
            cy = round(iy0 + (row + 0.5) * cell_h)
            patch = array[cy - 2:cy + 3, cx - 2:cx + 3].astype(np.float32)
            mean = patch.reshape(-1, 3).mean(axis=0)
            ink_distance = float(np.linalg.norm(mean - PALETTE["ink"]))
            paper_distance = float(np.linalg.norm(mean - PALETTE["paper"]))
            if min(ink_distance, paper_distance) > 30:
                raise ValueError("cell interior is corrupted")
            values.append(1 if ink_distance < paper_distance else 0)
        matrix.append(tuple(values))
    return tuple(matrix)


def _fold(source) -> tuple[tuple[int, ...], ...]:
    return tuple(
        tuple(source[row][3 - col] ^ source[row][4 + col] for col in range(4))
        for row in range(4)
    )


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 512 or array.shape[1] < 512:
        raise ValueError("unsupported raster")
    _validate_plaque(array)
    source = _read_grid(array, PALETTE["reference"], 8)
    target = _fold(source)
    matches = []
    occupancies = []
    for label in ("ruby", "green", "blue", "gold"):
        candidate = _read_grid(array, PALETTE[label], 4)
        occupancies.append(sum(map(sum, candidate)))
        if candidate == target:
            matches.append(label)
    if len(set(occupancies)) != 1:
        raise ValueError("candidate ink totals leak the answer")
    if len(matches) != 1:
        raise ValueError("image lacks one unique folded correspondence")
    return matches[0]
