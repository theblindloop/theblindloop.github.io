"""Renderer and analytic arm for exclusive-overlap stencil folding."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = HEIGHT = 768
ROWS = 4
SOURCE_COLS = 8
RESULT_COLS = 4
RIM_COLORS = ("ruby", "green", "blue", "gold")
RGB = {
    "plaque": (31, 45, 67),
    "ink": (53, 181, 176),
    "paper": (244, 235, 204),
    "reference": (196, 207, 220),
    "crease": (238, 174, 53),
    "ruby": (211, 61, 78),
    "green": (42, 169, 103),
    "blue": (53, 116, 218),
    "gold": (235, 168, 42),
}
PLAQUE = (210, 210, 558, 558)
SOURCE_BOX = (270, 238, 498, 350)
SLOTS = {
    "nw": (240, 365, 330, 455),
    "ne": (438, 365, 528, 455),
    "sw": (240, 465, 330, 555),
    "se": (438, 465, 528, 555),
}


def _as_matrix(values) -> tuple[tuple[int, ...], ...]:
    return tuple(tuple(int(value) for value in row) for row in values)


def _fold_xor(source) -> tuple[tuple[int, ...], ...]:
    matrix = _as_matrix(source)
    return tuple(
        tuple(matrix[row][3 - col] ^ matrix[row][4 + col] for col in range(RESULT_COLS))
        for row in range(ROWS)
    )


def _hamming(left, right) -> int:
    a, b = _as_matrix(left), _as_matrix(right)
    return sum(a[r][c] != b[r][c] for r in range(ROWS) for c in range(RESULT_COLS))


def _random_source(rng: random.Random) -> tuple[tuple[int, ...], ...]:
    while True:
        values = [0] * 16 + [1] * 16
        rng.shuffle(values)
        matrix = tuple(tuple(values[r * SOURCE_COLS + c] for c in range(SOURCE_COLS)) for r in range(ROWS))
        occupancy = sum(map(sum, _fold_xor(matrix)))
        if 6 <= occupancy <= 10:
            return matrix


def _same_occupancy_distractor(rng: random.Random, target, seen) -> tuple[tuple[int, ...], ...]:
    count = sum(map(sum, target))
    for _ in range(2000):
        values = [1] * count + [0] * (ROWS * RESULT_COLS - count)
        rng.shuffle(values)
        matrix = tuple(tuple(values[r * RESULT_COLS + c] for c in range(RESULT_COLS)) for r in range(ROWS))
        if matrix not in seen and _hamming(matrix, target) >= 4:
            return matrix
    raise RuntimeError("could not sample a separated distractor")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    source = _random_source(rng)
    target = _fold_xor(source)
    seen = {target}
    distractors = []
    while len(distractors) < 3:
        item = _same_occupancy_distractor(rng, target, seen)
        seen.add(item)
        distractors.append(item)

    correct_index = int(seed) % 4
    candidates = list(distractors)
    candidates.insert(correct_index, target)

    desired_answer = RIM_COLORS[(int(seed) // 4) % 4]
    rim_colors = list(RIM_COLORS)
    rng.shuffle(rim_colors)
    answer_index = rim_colors.index(desired_answer)
    rim_colors[correct_index], rim_colors[answer_index] = rim_colors[answer_index], rim_colors[correct_index]
    candidate_slots = list(SLOTS)
    rng.shuffle(candidate_slots)
    return {
        "source": [list(row) for row in source],
        "candidates": [[list(row) for row in matrix] for matrix in candidates],
        "rim_colors": rim_colors,
        "candidate_slots": candidate_slots,
        "width": WIDTH,
        "height": HEIGHT,
    }


def _draw_grid(draw: ImageDraw.ImageDraw, matrix, box, rim, border: int) -> None:
    x0, y0, x1, y1 = box
    rows = len(matrix)
    cols = len(matrix[0])
    draw.rounded_rectangle(box, radius=7, fill=rim)
    ix0, iy0, ix1, iy1 = x0 + border, y0 + border, x1 - border, y1 - border
    cell_w = (ix1 - ix0 + 1) / cols
    cell_h = (iy1 - iy0 + 1) / rows
    for row in range(rows):
        for col in range(cols):
            left = round(ix0 + col * cell_w)
            top = round(iy0 + row * cell_h)
            right = round(ix0 + (col + 1) * cell_w) - 1
            bottom = round(iy0 + (row + 1) * cell_h) - 1
            fill = RGB["ink"] if int(matrix[row][col]) else RGB["paper"]
            draw.rectangle((left, top, right, bottom), fill=fill, outline=RGB["plaque"], width=2)


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    source = scene["source"]
    candidates = scene["candidates"]
    rims = scene["rim_colors"]
    slots = scene["candidate_slots"]
    image = Image.new("RGB", (width, height), "white")
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle(PLAQUE, radius=18, fill=RGB["plaque"])
    font = ImageFont.load_default()
    draw.text((350, 218), "FOLD", fill=RGB["paper"], font=font)
    _draw_grid(draw, source, SOURCE_BOX, RGB["reference"], 10)
    crease_x = (SOURCE_BOX[0] + SOURCE_BOX[2]) // 2
    for y in range(SOURCE_BOX[1] + 10, SOURCE_BOX[3] - 8, 10):
        draw.line((crease_x, y, crease_x, min(y + 5, SOURCE_BOX[3] - 10)), fill=RGB["crease"], width=3)
    for matrix, rim, slot in zip(candidates, rims, slots, strict=True):
        _draw_grid(draw, matrix, SLOTS[slot], RGB[rim], 10)
    return image


def analytic_gold(scene: dict) -> str:
    target = _fold_xor(scene["source"])
    matches = [i for i, item in enumerate(scene["candidates"]) if _as_matrix(item) == target]
    if len(matches) != 1:
        raise ValueError("scene lacks one exact folded result")
    return str(scene["rim_colors"][matches[0]])


def margin(scene: dict) -> float:
    target = _fold_xor(scene["source"])
    distances = sorted(_hamming(target, item) for item in scene["candidates"])
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    try:
        target = _fold_xor(scene["source"])
        distances = [_hamming(target, item) for item in scene["candidates"]]
    except Exception:
        return True
    return distances.count(0) != 1 or sorted(distances)[1] < 3


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # Candidate-list order is an intentional latent alias: render bindings are
    # triples (pattern, rim, slot), so rotating all three lists together leaves
    # every pixel and the gold unchanged.
    twin = copy.deepcopy(scene)
    for key in ("candidates", "rim_colors", "candidate_slots"):
        twin[key] = twin[key][1:] + twin[key][:1]
    return [("rotate_candidate_record_order", twin)]
