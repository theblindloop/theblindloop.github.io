"""Deterministic renderer and analytic arm for a topological wall puzzle."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

SIZE = 384
BG = (255, 255, 255)
BLACK = (31, 38, 45)
CYAN = (0, 166, 190)
ORANGE = (239, 125, 33)
COLORS = {"cyan": CYAN, "orange": ORANGE}
THICK = 14


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed) ^ 0xD0C0A17)
    return {
        "orientation": rng.randrange(4),
        "offset_x": rng.choice((-10, -5, 0, 5, 10)),
        "offset_y": rng.choice((-10, -5, 0, 5, 10)),
        "opening_width": rng.choice((30, 42, 54)),
        "target_color": "cyan" if int(seed) % 2 == 0 else "orange",
    }


def _rect(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], color: tuple[int, int, int]) -> None:
    draw.rectangle(box, fill=color)


def render(scene: dict[str, Any]) -> Image.Image:
    orientation = int(scene["orientation"])
    ox = int(scene["offset_x"])
    oy = int(scene["offset_y"])
    opening = int(scene["opening_width"])
    target_color = str(scene["target_color"])
    if orientation not in range(4) or target_color not in COLORS:
        raise ValueError("invalid scene")

    # In the canonical orientation, the first two cells are sealed courtyards;
    # the right cell is a visually similar bay connected to the exterior by a
    # broad gap in the top boundary. All wall strokes are one component.
    canvas = Image.new("RGB", (SIZE, SIZE), BG)
    draw = ImageDraw.Draw(canvas)
    left, top, right, bottom = 62 + ox, 160 + oy, 322 + ox, 224 + oy
    seam1, seam2 = 148 + ox, 235 + ox
    h = THICK // 2

    _rect(draw, (left, top - h, seam2, top + h), BLACK)
    # Two shoulders leave a centered opening above the right-hand bay.
    gap_center = (seam2 + right) // 2
    _rect(draw, (seam2, top - h, gap_center - opening // 2, top + h), BLACK)
    _rect(draw, (gap_center + opening // 2, top - h, right, top + h), BLACK)
    _rect(draw, (left, bottom - h, right, bottom + h), BLACK)
    _rect(draw, (left - h, top - h, left + h, bottom + h), BLACK)
    _rect(draw, (right - h, top - h, right + h, bottom + h), BLACK)

    target = COLORS[target_color]
    other_name = "orange" if target_color == "cyan" else "cyan"
    distractor = COLORS[other_name]
    # Black caps at both ends keep the colored seams equally sized and prevent
    # junction shape or segment length from revealing the answer.
    _rect(draw, (seam1 - h, top - h, seam1 + h, top + 11), BLACK)
    _rect(draw, (seam1 - h, top + 11, seam1 + h, bottom - 11), target)
    _rect(draw, (seam1 - h, bottom - 11, seam1 + h, bottom + h), BLACK)
    _rect(draw, (seam2 - h, top - h, seam2 + h, top + 11), BLACK)
    _rect(draw, (seam2 - h, top + 11, seam2 + h, bottom - 11), distractor)
    _rect(draw, (seam2 - h, bottom - 11, seam2 + h, bottom + h), BLACK)

    if orientation:
        canvas = canvas.rotate(90 * orientation, resample=Image.Resampling.NEAREST)
    return canvas


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(scene["target_color"])


def margin(scene: dict[str, Any]) -> float:
    # Pixel width of the only exterior channel; closing it is the nearest
    # topological boundary event that makes the distractor border two holes.
    return float(scene["opening_width"])


def is_quarantined(scene: dict[str, Any]) -> bool:
    return int(scene["opening_width"]) < 24


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
