"""Independent final-PNG inverse arm.

It segments wall colors, labels the four-connected white complement, marks the
components that reach the image border, and counts distinct enclosed labels
edge-adjacent to each colored wall segment. No scene data or renderer module is
available here.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE = {
    "black": np.array((31, 38, 45), dtype=np.int16),
    "cyan": np.array((0, 166, 190), dtype=np.int16),
    "orange": np.array((239, 125, 33), dtype=np.int16),
}


def _near(arr: np.ndarray, rgb: np.ndarray, tolerance: int = 18) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - rgb), axis=2) <= tolerance


def _label_open(open_mask: np.ndarray) -> tuple[np.ndarray, set[int]]:
    height, width = open_mask.shape
    labels = np.zeros((height, width), dtype=np.int32)
    border_labels: set[int] = set()
    label = 0
    for sy, sx in np.argwhere(open_mask):
        sy, sx = int(sy), int(sx)
        if labels[sy, sx]:
            continue
        label += 1
        labels[sy, sx] = label
        queue = deque([(sy, sx)])
        reaches_border = False
        while queue:
            y, x = queue.popleft()
            if y == 0 or x == 0 or y == height - 1 or x == width - 1:
                reaches_border = True
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and open_mask[ny, nx] and labels[ny, nx] == 0:
                    labels[ny, nx] = label
                    queue.append((ny, nx))
        if reaches_border:
            border_labels.add(label)
    return labels, border_labels


def _adjacent_enclosed(labels: np.ndarray, color_mask: np.ndarray, border: set[int]) -> set[int]:
    neighbor_labels: set[int] = set()
    for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
        shifted = np.roll(labels, (dy, dx), axis=(0, 1))
        vals = shifted[color_mask]
        neighbor_labels.update(int(v) for v in np.unique(vals) if int(v) > 0)
    return neighbor_labels - border


def decision_from_image(image: Image.Image) -> str:
    # Nearest-neighbor downsampling preserves the wide walls while making the
    # component traversal inexpensive and robust to very large source images.
    work = image.convert("RGB")
    if work.width > 256 or work.height > 256:
        work.thumbnail((192, 192), Image.Resampling.NEAREST)
    arr = np.asarray(work)
    masks = {name: _near(arr, rgb) for name, rgb in PALETTE.items()}
    if int(masks["cyan"].sum()) < 40 or int(masks["orange"].sum()) < 40:
        raise ValueError("both colored wall segments are required")
    wall = masks["black"] | masks["cyan"] | masks["orange"]
    labels, border = _label_open(~wall)
    counts = {
        color: len(_adjacent_enclosed(labels, masks[color], border))
        for color in ("cyan", "orange")
    }
    ranked = sorted(counts, key=lambda color: counts[color], reverse=True)
    if counts[ranked[0]] < 2 or counts[ranked[0]] == counts[ranked[1]]:
        raise ValueError(f"ambiguous enclosed-boundary multiplicity: {counts}")
    return ranked[0]
