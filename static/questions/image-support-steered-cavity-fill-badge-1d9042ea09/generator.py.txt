"""Deterministic scene sampler and renderer for the cavity-fill badge."""

from __future__ import annotations

from collections import deque
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (247, 243, 232)
NAVY = (24, 48, 67)
CYAN = (69, 191, 199)
CREAM = (244, 224, 174)
GOLD = (232, 177, 54)
TAG_COLORS = {
    "red": (211, 73, 75),
    "green": (65, 157, 102),
    "blue": (68, 111, 190),
}
DECOR = (218, 215, 205)


def _filled(source: list[list[int]]) -> list[list[int]]:
    rows, cols = len(source), len(source[0])
    reachable: set[tuple[int, int]] = set()
    queue: deque[tuple[int, int]] = deque()
    for row in range(rows):
        for col in range(cols):
            if row in (0, rows - 1) or col in (0, cols - 1):
                if source[row][col] == 0 and (row, col) not in reachable:
                    reachable.add((row, col))
                    queue.append((row, col))
    while queue:
        row, col = queue.popleft()
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nr, nc = row + dr, col + dc
            if 0 <= nr < rows and 0 <= nc < cols:
                if source[nr][nc] == 0 and (nr, nc) not in reachable:
                    reachable.add((nr, nc))
                    queue.append((nr, nc))
    return [
        [1 if source[row][col] or (row, col) not in reachable else 0 for col in range(cols)]
        for row in range(rows)
    ]


def _source_grid(rng: random.Random) -> list[list[int]]:
    size = 6
    grid = [[0 for _ in range(size)] for _ in range(size)]
    center_row = rng.choice((2, 3))
    center_col = rng.choice((2, 3))
    for dr in (-1, 0, 1):
        for dc in (-1, 0, 1):
            if dr or dc:
                grid[center_row + dr][center_col + dc] = 1
    protected = {
        (center_row + dr, center_col + dc)
        for dr in (-1, 0, 1)
        for dc in (-1, 0, 1)
    }
    for row in range(size):
        for col in range(size):
            if (row, col) not in protected and rng.random() < 0.22:
                grid[row][col] = 1
    grid[center_row][center_col] = 0
    # Keep visible exterior entry points on every scene.
    grid[0][0] = grid[0][5] = grid[5][0] = grid[5][5] = 0
    return grid


def _wrong_results(source: list[list[int]], expected: list[list[int]], rng: random.Random):
    filled_cells = [
        (r, c)
        for r in range(6)
        for c in range(6)
        if expected[r][c] == 1 and source[r][c] == 0
    ]
    exterior_cream = [
        (r, c)
        for r in range(6)
        for c in range(6)
        if expected[r][c] == 0
    ]
    rng.shuffle(filled_cells)
    rng.shuffle(exterior_cream)
    results = []
    for index in range(2):
        wrong = [row[:] for row in expected]
        remove = filled_cells[index % len(filled_cells)]
        add = exterior_cream[index % len(exterior_cream)]
        wrong[remove[0]][remove[1]] = 0
        wrong[add[0]][add[1]] = 1
        if wrong in results:
            add = exterior_cream[(index + 2) % len(exterior_cream)]
            wrong = [row[:] for row in expected]
            wrong[remove[0]][remove[1]] = 0
            wrong[add[0]][add[1]] = 1
        results.append(wrong)
    return results


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    source = _source_grid(rng)
    expected = _filled(source)
    wrong_a, wrong_b = _wrong_results(source, expected, rng)
    tags = ["red", "green", "blue"]
    correct_index = rng.randrange(3)
    candidates = [wrong_a, wrong_b]
    candidates.insert(correct_index, expected)
    layout = ["source", "red", "green", "blue"]
    rng.shuffle(layout)
    origin = [rng.randint(70, 320), rng.randint(70, 320)]
    decorations = []
    ox, oy = origin
    for _ in range(18):
        for _attempt in range(80):
            x, y = rng.randint(22, 490), rng.randint(22, 490)
            if not (ox - 28 <= x <= ox + 150 and oy - 28 <= y <= oy + 150):
                decorations.append([x, y, rng.randint(3, 8), rng.randrange(2)])
                break
    return {
        "source": source,
        "candidates": candidates,
        "tags": tags,
        "layout": layout,
        "origin": origin,
        "decorations": decorations,
    }


def _draw_grid(draw: ImageDraw.ImageDraw, x: int, y: int, frame, grid) -> None:
    draw.rectangle((x, y, x + 47, y + 47), fill=frame)
    draw.rectangle((x + 3, y + 3, x + 44, y + 44), fill=NAVY)
    start_x, start_y = x + 6, y + 6
    for row in range(6):
        for col in range(6):
            left = start_x + 6 * col
            top = start_y + 6 * row
            fill = CYAN if grid[row][col] else CREAM
            draw.rectangle((left, top, left + 5, top + 5), fill=fill, outline=NAVY, width=1)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x, y, radius, kind in scene["decorations"]:
        if kind:
            draw.rectangle((x - radius, y - radius, x + radius, y + radius), outline=DECOR, width=2)
        else:
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), outline=DECOR, width=2)
    ox, oy = scene["origin"]
    draw.rounded_rectangle((ox, oy, ox + 121, oy + 121), radius=12, fill=NAVY)
    slots = [(8, 8), (66, 8), (8, 66), (66, 66)]
    by_name = {"source": (GOLD, scene["source"])}
    for tag, grid in zip(scene["tags"], scene["candidates"]):
        by_name[tag] = (TAG_COLORS[tag], grid)
    for name, (dx, dy) in zip(scene["layout"], slots):
        frame, grid = by_name[name]
        _draw_grid(draw, ox + dx, oy + dy, frame, grid)
    return image


def analytic_gold(scene: dict) -> str:
    expected = _filled(scene["source"])
    matches = [tag for tag, grid in zip(scene["tags"], scene["candidates"]) if grid == expected]
    if len(matches) != 1:
        raise ValueError("scene has no unique exact completion")
    return matches[0]


def margin(scene: dict) -> float:
    expected = _filled(scene["source"])
    distances = [
        sum(a != b for row_a, row_b in zip(grid, expected) for a, b in zip(row_a, row_b))
        for grid in scene["candidates"]
    ]
    nonzero = [value for value in distances if value > 0]
    return float(min(nonzero)) if distances.count(0) == 1 and nonzero else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        expected = _filled(scene["source"])
        matches = sum(grid == expected for grid in scene["candidates"])
        added = sum(expected[r][c] > scene["source"][r][c] for r in range(6) for c in range(6))
        inks = [sum(map(sum, grid)) for grid in scene["candidates"]]
        return matches != 1 or added < 1 or len(set(inks)) != 1 or margin(scene) < 2.0
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
