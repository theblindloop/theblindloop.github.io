"""Independent pixel-only inverse arm for the cavity-fill badge."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

GOLD = np.array((232, 177, 54), dtype=np.int16)
CYAN = np.array((69, 191, 199), dtype=np.int16)
FRAMES = {
    "red": np.array((211, 73, 75), dtype=np.int16),
    "green": np.array((65, 157, 102), dtype=np.int16),
    "blue": np.array((68, 111, 190), dtype=np.int16),
}


def _mask(array: np.ndarray, color: np.ndarray, tolerance: float = 18.0) -> np.ndarray:
    return np.linalg.norm(array.astype(np.int16) - color, axis=2) <= tolerance


def _frame_box(array: np.ndarray, color: np.ndarray) -> tuple[int, int, int, int]:
    ys, xs = np.where(_mask(array, color))
    if len(xs) < 250:
        raise ValueError("required colored frame is missing")
    x0, x1, y0, y1 = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    if not (45 <= x1 - x0 + 1 <= 51 and 45 <= y1 - y0 + 1 <= 51):
        raise ValueError("colored frame geometry is damaged")
    return x0, y0, x1, y1


def _decode(array: np.ndarray, box: tuple[int, int, int, int]) -> list[list[int]]:
    x0, y0, _, _ = box
    grid = []
    for row in range(6):
        values = []
        for col in range(6):
            x, y = x0 + 9 + 6 * col, y0 + 9 + 6 * row
            patch = array[y - 1 : y + 2, x - 1 : x + 2]
            if patch.shape != (3, 3, 3):
                raise ValueError("grid sample lies outside image")
            distance = np.linalg.norm(patch.astype(np.int16) - CYAN, axis=2)
            values.append(1 if float(np.mean(distance <= 24.0)) >= 0.55 else 0)
        grid.append(values)
    return grid


def _fill(source: list[list[int]]) -> list[list[int]]:
    reachable = set()
    queue = deque()
    for row in range(6):
        for col in range(6):
            if (row in (0, 5) or col in (0, 5)) and source[row][col] == 0:
                reachable.add((row, col))
                queue.append((row, col))
    while queue:
        row, col = queue.popleft()
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            point = row + dr, col + dc
            if 0 <= point[0] < 6 and 0 <= point[1] < 6:
                if source[point[0]][point[1]] == 0 and point not in reachable:
                    reachable.add(point)
                    queue.append(point)
    return [[1 if source[r][c] or (r, c) not in reachable else 0 for c in range(6)] for r in range(6)]


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    source_box = _frame_box(array, GOLD)
    candidate_boxes = {name: _frame_box(array, color) for name, color in FRAMES.items()}
    centers = [
        ((box[0] + box[2]) / 2.0, (box[1] + box[3]) / 2.0)
        for box in [source_box, *candidate_boxes.values()]
    ]
    if max(x for x, _ in centers) - min(x for x, _ in centers) > 72:
        raise ValueError("frames do not form one compact badge")
    if max(y for _, y in centers) - min(y for _, y in centers) > 72:
        raise ValueError("frames do not form one compact badge")
    expected = _fill(_decode(array, source_box))
    matches = [name for name, box in candidate_boxes.items() if _decode(array, box) == expected]
    if len(matches) != 1:
        raise ValueError("pixel evidence has no unique matching completion")
    return matches[0]
