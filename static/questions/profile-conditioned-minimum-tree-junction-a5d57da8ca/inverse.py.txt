"""Independent image-only inverse arm for the implicit minimum tree.

This module intentionally duplicates palette measurement, component labeling,
centroid recovery, and Kruskal's algorithm.  It accepts only a final Pillow
image and has no access to latent state or analytic-gold code.
"""

from __future__ import annotations

import math
from dataclasses import dataclass

import numpy as np
from PIL import Image

_BACKGROUND = np.array((248, 248, 244), dtype=np.float64)
_COLORS: dict[str, np.ndarray] = {
    "dark": np.array((38, 43, 50), dtype=np.float64),
    "red": np.array((214, 58, 67), dtype=np.float64),
    "blue": np.array((48, 105, 205), dtype=np.float64),
    "green": np.array((34, 151, 91), dtype=np.float64),
    "orange": np.array((229, 133, 35), dtype=np.float64),
}
_ANSWER_COLORS = ("red", "blue", "green", "orange")
_MIN_COMPONENTS = 8
_MAX_COMPONENTS = 30
_MIN_COMPONENT_PIXELS = 180
_MAX_COMPONENT_PIXELS = 400


@dataclass(frozen=True)
class PixelMeasurement:
    points: tuple[tuple[float, float], ...]
    colors: tuple[str, ...]
    degrees: tuple[int, ...]
    exchange_gap: float


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    visited = np.zeros(mask.shape, dtype=bool)
    output: list[np.ndarray] = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if visited[y0, x0]:
            continue
        stack = [(y0, x0)]
        visited[y0, x0] = True
        pixels: list[tuple[int, int]] = []
        while stack:
            y, x = stack.pop()
            pixels.append((y, x))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = y + dy, x + dx
                if (
                    0 <= ny < height
                    and 0 <= nx < width
                    and mask[ny, nx]
                    and not visited[ny, nx]
                ):
                    visited[ny, nx] = True
                    stack.append((ny, nx))
        output.append(np.asarray(pixels, dtype=np.int32))
    return output


def _classify_center(arr: np.ndarray, x: float, y: float) -> str:
    xx, yy = int(round(x)), int(round(y))
    patch = arr[max(0, yy - 3) : yy + 4, max(0, xx - 3) : xx + 4].astype(np.float64)
    if not patch.size:
        raise ValueError("empty center patch")
    rgb = np.median(patch.reshape(-1, 3), axis=0)
    distances = {name: float(np.linalg.norm(rgb - value)) for name, value in _COLORS.items()}
    name = min(distances, key=distances.get)
    if distances[name] > 35.0:
        raise ValueError(f"unrecognized center color near {(x, y)}")
    return name


def _distance(a: tuple[float, float], b: tuple[float, float]) -> float:
    return math.hypot(a[0] - b[0], a[1] - b[1])


def _minimum_tree(
    points: tuple[tuple[float, float], ...],
) -> list[tuple[int, int, float]]:
    n = len(points)
    edges = sorted(
        (_distance(points[i], points[j]), i, j)
        for i in range(n)
        for j in range(i + 1, n)
    )
    parent = list(range(n))
    rank = [0] * n

    def find(value: int) -> int:
        while parent[value] != value:
            parent[value] = parent[parent[value]]
            value = parent[value]
        return value

    tree: list[tuple[int, int, float]] = []
    for weight, left, right in edges:
        a, b = find(left), find(right)
        if a == b:
            continue
        if rank[a] < rank[b]:
            a, b = b, a
        parent[b] = a
        if rank[a] == rank[b]:
            rank[a] += 1
        tree.append((left, right, weight))
        if len(tree) == n - 1:
            break
    if len(tree) != n - 1:
        raise ValueError("recovered centers do not form a connected complete graph")
    return tree


def _tree_degrees(count: int, tree: list[tuple[int, int, float]]) -> tuple[int, ...]:
    degrees = [0] * count
    for left, right, _ in tree:
        degrees[left] += 1
        degrees[right] += 1
    return tuple(degrees)


def _exchange_gap(
    points: tuple[tuple[float, float], ...], tree: list[tuple[int, int, float]]
) -> float:
    adjacency: list[list[tuple[int, float]]] = [[] for _ in points]
    pairs: set[tuple[int, int]] = set()
    for left, right, weight in tree:
        adjacency[left].append((right, weight))
        adjacency[right].append((left, weight))
        pairs.add((min(left, right), max(left, right)))

    def maximum_on_path(start: int, goal: int) -> float:
        pending = [(start, -1, 0.0)]
        while pending:
            node, previous, maximum = pending.pop()
            if node == goal:
                return maximum
            for neighbor, weight in adjacency[node]:
                if neighbor != previous:
                    pending.append((neighbor, node, max(maximum, weight)))
        raise ValueError("no recovered tree path")

    gap = math.inf
    for left in range(len(points)):
        for right in range(left + 1, len(points)):
            if (left, right) not in pairs:
                gap = min(
                    gap,
                    _distance(points[left], points[right]) - maximum_on_path(left, right),
                )
    return gap


def measure_image(image: Image.Image) -> PixelMeasurement:
    arr = np.asarray(image.convert("RGB"))
    if arr.ndim != 3 or arr.shape[2] != 3:
        raise ValueError("expected an RGB raster")
    distance_from_background = np.linalg.norm(arr.astype(np.float64) - _BACKGROUND, axis=2)
    foreground = distance_from_background > 45.0
    components = _components(foreground)
    components = [part for part in components if len(part) >= _MIN_COMPONENT_PIXELS]
    if not _MIN_COMPONENTS <= len(components) <= _MAX_COMPONENTS:
        raise ValueError(f"implausible mark count {len(components)}")
    if any(len(part) > _MAX_COMPONENT_PIXELS for part in components):
        raise ValueError("a foreground component is too large to be one mark")

    measured: list[tuple[float, float, str]] = []
    for component in components:
        y = float(component[:, 0].mean())
        x = float(component[:, 1].mean())
        measured.append((x, y, _classify_center(arr, x, y)))
    measured.sort(key=lambda item: (item[1], item[0]))
    points = tuple((x, y) for x, y, _ in measured)
    colors = tuple(color for _, _, color in measured)
    if any(colors.count(color) != 1 for color in _ANSWER_COLORS):
        raise ValueError("expected one mark of every answer color")
    tree = _minimum_tree(points)
    return PixelMeasurement(
        points=points,
        colors=colors,
        degrees=_tree_degrees(len(points), tree),
        exchange_gap=_exchange_gap(points, tree),
    )


def decision_from_image(image: Image.Image) -> str:
    measured = measure_image(image)
    matches = [
        color
        for color, degree in zip(measured.colors, measured.degrees, strict=True)
        if color != "dark" and degree == 3
    ]
    if len(matches) != 1:
        raise ValueError(f"expected one colored degree-three vertex, found {matches}")
    return matches[0]
