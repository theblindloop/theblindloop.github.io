"""Deterministic renderer and analytic arm for twin-column rank-echo search."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

WIDTH = 800
HEIGHT = 800
BACKGROUND = [250, 248, 244]
COLORS = ("red", "blue", "green", "purple")
PALETTE = {
    "red": [210, 48, 64],
    "blue": [40, 92, 205],
    "green": [20, 142, 84],
    "purple": [128, 55, 176],
}
ROW_CENTERS = (150, 350, 550, 750)
COLUMN_X = {"left": 150, "right": 650}
MIN_ORDER_CENTER_GAP_PX = 100.0


def _rankings(scene: dict) -> dict[str, list[str]]:
    """Recover the two latent-visible color orders without trusting list order."""
    markers = scene["markers"]
    if len(markers) != 2 * len(COLORS):
        raise ValueError("scene must have exactly eight markers")
    rankings: dict[str, list[str]] = {}
    for side in ("left", "right"):
        side_markers = [marker for marker in markers if marker["side"] == side]
        if len(side_markers) != len(COLORS):
            raise ValueError(f"{side} column must contain four markers")
        ordered = sorted(side_markers, key=lambda marker: float(marker["y"]))
        colors = [str(marker["color"]) for marker in ordered]
        if set(colors) != set(COLORS) or len(colors) != len(set(colors)):
            raise ValueError(f"{side} column must contain every color exactly once")
        rankings[side] = colors
    return rankings


def _rank_displacements(scene: dict) -> dict[str, int]:
    rankings = _rankings(scene)
    left_rank = {color: rank for rank, color in enumerate(rankings["left"])}
    right_rank = {color: rank for rank, color in enumerate(rankings["right"])}
    return {color: abs(left_rank[color] - right_rank[color]) for color in COLORS}


def order_clearance(scene: dict) -> float:
    """Smallest visible center-to-center separation of adjacent markers."""
    clearances = []
    for side in ("left", "right"):
        ys = sorted(float(marker["y"]) for marker in scene["markers"] if marker["side"] == side)
        if len(ys) != len(COLORS):
            return 0.0
        clearances.extend(b - a for a, b in zip(ys, ys[1:]))
    return float(min(clearances)) if clearances else 0.0


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target_color = COLORS[int(seed) % len(COLORS)]

    left_order = list(COLORS)
    rng.shuffle(left_order)
    fixed_rank = left_order.index(target_color)
    moving_ranks = [rank for rank in range(len(COLORS)) if rank != fixed_rank]
    moving_colors = [left_order[rank] for rank in moving_ranks]
    shift = 1 if rng.random() < 0.5 else 2
    right_order = list(left_order)
    for index, rank in enumerate(moving_ranks):
        right_order[rank] = moving_colors[(index + shift) % len(moving_colors)]

    marker_radius = rng.choice((30, 32, 34, 36))
    markers = []
    for side, order in (("left", left_order), ("right", right_order)):
        for rank, color in enumerate(order):
            markers.append(
                {
                    "side": side,
                    "color": color,
                    "x": float(COLUMN_X[side] + rng.randint(-7, 7)),
                    "y": float(ROW_CENTERS[rank] + rng.randint(-9, 9)),
                }
            )
    rng.shuffle(markers)
    return {
        "markers": markers,
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
        "marker_radius": marker_radius,
        "column_guides": [[72, 48, 228, 798], [572, 48, 728, 798]],
    }


def render(scene: dict) -> Image.Image:
    scale = 4
    width = int(scene["width"])
    height = int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)

    guide_fill = (238, 239, 242)
    guide_outline = (190, 194, 202)
    for x0, y0, x1, y1 in scene["column_guides"]:
        box = tuple(int(value * scale) for value in (x0, y0, x1, y1))
        draw.rounded_rectangle(
            box,
            radius=26 * scale,
            fill=guide_fill,
            outline=guide_outline,
            width=3 * scale,
        )

    radius = float(scene["marker_radius"]) * scale
    for marker in scene["markers"]:
        x = float(marker["x"]) * scale
        y = float(marker["y"]) * scale
        rgb = tuple(PALETTE[str(marker["color"])])
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=rgb)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    displacements = _rank_displacements(scene)
    winners = [color for color, displacement in displacements.items() if displacement == 0]
    if len(winners) != 1:
        raise ValueError("scene must have exactly one fixed-rank color")
    return winners[0]


def margin(scene: dict) -> float:
    """Runner-up gap in ordinal-displacement units (zero is the winning score)."""
    ranked = sorted(_rank_displacements(scene).values())
    if len(ranked) < 2:
        return 0.0
    return float(ranked[1] - ranked[0])


def is_quarantined(scene: dict) -> bool:
    return order_clearance(scene) < MIN_ORDER_CENTER_GAP_PX or margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["markers"] = list(reversed(twin["markers"]))
    return [("reverse_unordered_nonoverlapping_marker_records", twin)]
