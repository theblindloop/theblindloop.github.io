"""Independent final-PNG inverse arm for twin-column rank-echo search.

The oracle knows only the public palette and question rule. It segments every
color, requires one compact marker component in each distant column, recovers
the two top-to-bottom color permutations from component centroids, and returns
the unique color whose ordinal displacement is zero with a runner-up gap.
"""

from __future__ import annotations

from dataclasses import dataclass

import numpy as np
from PIL import Image

COLORS = ("red", "blue", "green", "purple")
PALETTE = np.asarray(
    ([210, 48, 64], [40, 92, 205], [20, 142, 84], [128, 55, 176], [250, 248, 244]),
    dtype=np.float32,
)
MIN_COMPONENT_AREA = 1800
MAX_COMPONENT_AREA = 4800
MIN_FILL_RATIO = 0.66
MAX_FILL_RATIO = 0.86
MIN_VERTICAL_GAP_PX = 90.0
MIN_RUNNER_UP_RANK_GAP = 1


@dataclass(frozen=True)
class Marker:
    color: str
    cx: float
    cy: float
    area: int


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]:
            continue
        stack = [(y0, x0)]
        seen[y0, x0] = True
        points = []
        while stack:
            y, x = stack.pop()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= MIN_COMPONENT_AREA:
            output.append(np.asarray(points, dtype=np.float64))
    return output


def _fit_marker(points_yx: np.ndarray, color: str, shape: tuple[int, int]) -> Marker:
    area = len(points_yx)
    if not MIN_COMPONENT_AREA <= area <= MAX_COMPONENT_AREA:
        raise ValueError(f"{color} marker has implausible area {area}")
    ys = points_yx[:, 0]
    xs = points_yx[:, 1]
    box_h = float(ys.max() - ys.min() + 1.0)
    box_w = float(xs.max() - xs.min() + 1.0)
    aspect = box_w / box_h
    fill = area / (box_w * box_h)
    if not 0.86 <= aspect <= 1.16:
        raise ValueError(f"{color} component is not a compact round marker")
    if not MIN_FILL_RATIO <= fill <= MAX_FILL_RATIO:
        raise ValueError(f"{color} component fill ratio is not marker-like")
    height, width = shape
    if xs.min() <= 0 or ys.min() <= 0 or xs.max() >= width - 1 or ys.max() >= height - 1:
        raise ValueError(f"{color} marker touches the raster boundary")
    return Marker(color, float(xs.mean()), float(ys.mean()), int(area))


def _measure_markers(image: Image.Image) -> dict[str, list[Marker]]:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    if array.ndim != 3 or array.shape[2] != 3 or min(array.shape[:2]) < 64:
        raise ValueError("unsupported raster")
    distances = ((array[:, :, None, :] - PALETTE[None, None, :, :]) ** 2).sum(axis=3)
    labels = distances.argmin(axis=2)
    by_side: dict[str, list[Marker]] = {"left": [], "right": []}
    height, width = labels.shape
    for index, color in enumerate(COLORS):
        components = _components(labels == index)
        if len(components) != 2:
            raise ValueError(f"{color} must have exactly two visible marker components")
        fitted = [_fit_marker(component, color, labels.shape) for component in components]
        left = [marker for marker in fitted if marker.cx < 0.35 * width]
        right = [marker for marker in fitted if marker.cx > 0.65 * width]
        if len(left) != 1 or len(right) != 1:
            raise ValueError(f"{color} must occur once in each separated column")
        by_side["left"].append(left[0])
        by_side["right"].append(right[0])

    for side in ("left", "right"):
        ordered = sorted(by_side[side], key=lambda marker: marker.cy)
        gaps = [b.cy - a.cy for a, b in zip(ordered, ordered[1:])]
        if len(ordered) != len(COLORS) or min(gaps) < MIN_VERTICAL_GAP_PX:
            raise ValueError(f"{side} marker order is missing or vertically unresolved")
        by_side[side] = ordered
    return by_side


def decision_from_image(image: Image.Image) -> str:
    by_side = _measure_markers(image)
    left_rank = {marker.color: rank for rank, marker in enumerate(by_side["left"])}
    right_rank = {marker.color: rank for rank, marker in enumerate(by_side["right"])}
    scored = sorted(
        ((abs(left_rank[color] - right_rank[color]), color) for color in COLORS),
        key=lambda item: (item[0], item[1]),
    )
    if scored[0][0] != 0 or scored[1][0] - scored[0][0] < MIN_RUNNER_UP_RANK_GAP:
        raise ValueError("no unique fixed-rank color with a runner-up gap")
    return scored[0][1]
