"""Deterministic renderer and analytic arm for the neighbourhood-parity cascade.

A 5-by-5 square grid of cells is drawn; each cell is filled either dark (an ON
cell) or pale (an OFF cell).  A small teal dot-key below the grid shows the
number of transition rounds S to apply (one, two, or three dots).

The transition is a whole-grid cellular rule that depends on the *current* two
dimensional state: for every cell, count how many of its eight surrounding
neighbours (the eight cells sharing an edge or a corner with it) are dark.  In
the next round the cell is dark if and only if that count is exactly two, and
pale otherwise.  The rule is non-monotone: a dark cell may turn pale and a pale
cell may turn dark, so every cell is recomputed from its full 3-by-3
neighbourhood in every round.  The rule is applied S times, converting each
round's grid into the next round's grid, and the question asks whether the final
number of dark cells in the 5-by-5 grid is odd.  The intermediate and terminal
grids are never drawn; they must be recovered by iterating the visible initial
grid through S rounds of the rule.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = (780, 880)
N = 5
CELL = 76
GAP = 16
GRID_LEFT = (CANVAS[0] - (N * CELL + (N - 1) * GAP)) // 2
GRID_TOP = 150
DOT_Y = 700
DOT_R = 22
DOT_SP = 56
BACKGROUND = (250, 250, 250)
OUTLINE = (128, 128, 128)
ON_FILL = (35, 35, 35)
OFF_FILL = (235, 235, 235)
DOT_COLOR = (0, 150, 150)
MAX_STEP = 3
DUTY = 0.45


def step(grid: list[list[int]]) -> list[list[int]]:
    """One round of the whole-grid neighbourhood-parity cascade on a bit grid."""

    n = len(grid)
    m = len(grid[0])
    out = [[0] * m for _ in range(n)]
    for i in range(n):
        for j in range(m):
            dark = 0
            for di in (-1, 0, 1):
                for dj in (-1, 0, 1):
                    if di == 0 and dj == 0:
                        continue
                    ni, nj = i + di, j + dj
                    if 0 <= ni < n and 0 <= nj < m:
                        dark += grid[ni][nj]
            out[i][j] = 1 if dark == 2 else 0
    return out


def _simulate(grid: list[list[int]], steps: int) -> list[list[int]]:
    current = [list(row) for row in grid]
    for _ in range(steps):
        current = step(current)
    return current


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    grid = [[1 if rng.random() < DUTY else 0 for _ in range(N)] for _ in range(N)]
    steps = rng.randint(1, MAX_STEP)
    return {
        "canvas_size": list(CANVAS),
        "grid": grid,
        "steps": steps,
        "cell_px": CELL,
        "gap_px": GAP,
        "grid_left": GRID_LEFT,
        "grid_top": GRID_TOP,
        "dot_y": DOT_Y,
        "dot_r": DOT_R,
        "dot_spacing": DOT_SP,
    }


def render(scene: dict) -> Image.Image:
    width, height = (int(v) for v in scene["canvas_size"])
    grid = [[int(v) for v in row] for row in scene["grid"]]
    steps = int(scene["steps"])
    cell = int(scene["cell_px"])
    gap = int(scene["gap_px"])
    gl = int(scene["grid_left"])
    gt = int(scene["grid_top"])
    dot_y = int(scene["dot_y"])
    dot_r = int(scene["dot_r"])
    dot_spacing = int(scene["dot_spacing"])
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for i, row in enumerate(grid):
        for j, value in enumerate(row):
            x = gl + j * (cell + gap)
            y = gt + i * (cell + gap)
            draw.rectangle(
                (x, y, x + cell, y + cell),
                fill=ON_FILL if value == 1 else OFF_FILL,
                outline=OUTLINE,
                width=3,
            )
    total_dots = steps * dot_spacing
    x0 = (width - total_dots) // 2 + dot_spacing // 2
    for k in range(steps):
        cx = x0 + k * dot_spacing
        draw.ellipse(
            (cx - dot_r, dot_y - dot_r, cx + dot_r, dot_y + dot_r),
            fill=DOT_COLOR,
        )
    return image


def analytic_gold(scene: dict) -> str:
    grid = [[int(v) for v in row] for row in scene["grid"]]
    steps = int(scene["steps"])
    final = _simulate(grid, steps)
    return "yes" if sum(sum(row) for row in final) % 2 == 1 else "no"


def margin(scene: dict) -> float:
    return 1.0


def is_quarantined(scene: dict) -> bool:
    grid = scene["grid"]
    steps = int(scene["steps"])
    if len(grid) != N or any(len(row) != N for row in grid):
        return True
    if steps < 1 or steps > MAX_STEP:
        return True
    if any(value not in (0, 1) for row in grid for value in row):
        return True
    return False


def latent_symmetries(scene: dict) -> list:
    return []
