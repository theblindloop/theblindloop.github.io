"""Region-contact-degree world renderer.

The image shows several filled colored rectangular regions on a white
background. Two regions are *neighbors* exactly when their outlines touch
along a stretch of boundary of positive length (not merely at a single corner
point). The decision is: which colored region has the MOST neighbors in this
contact graph.

The scene is a plain dict so the verifier's AST reachability audit stays free
of side effects:

    z = {
        "regions": [ {"color": str, "x": .., "y": .., "w": .., "h": ..}, ... ],
        "width": 400,
        "height": 400,
        "background": "white",
    }

All coordinates are integers so region edges land exactly on pixel grid lines:
filled regions are drawn at 1x with hard (non-antialiased) boundaries, so the
only colours present are the four region colours and white, and the contact
graph recovered from pixels is exact on robust scenes.

Two layout families are produced:

  * hub  : one central rectangle touches all three others (degree 3); each leaf
           touches only the hub (degree 1). Unique maximum -> answer is the hub.
  * grid : a 2x2 grid where every region touches its row and column neighbour
           (each degree 2), so the maximum is a tie -> margin 0 -> quarantined.
"""

from __future__ import annotations

import random
from collections.abc import Iterator

from PIL import Image, ImageDraw

RENDERER_VERSION = "region_contact_degree-0.4.0"

# Fixed order for deterministic tie-breaking among equal maximum degrees.
PALETTE = ["red", "green", "blue", "magenta"]

W = 400
H = 400

# Robustness margins (pixels).
MIN_SHARE_PX = 24.0   # a genuine boundary contact must overlap at least this much
QUARANTINE_MIN = 1.0  # margin (top_degree - second_degree) below this is quarantined
PAD = 6.0             # clearance to the picture border
GRID_PAD = 10.0       # clearance between the 2x2 grid block and the border


def _rect_contact(a: dict, b: dict, eps: float = 0.0) -> bool:
    """True iff two axis-aligned rectangles share a positive-length edge."""
    ax1, ay1, ax2, ay2 = a["x"], a["y"], a["x"] + a["w"], a["y"] + a["h"]
    bx1, by1, bx2, by2 = b["x"], b["y"], b["x"] + b["w"], b["y"] + b["h"]
    x_overlap = min(ax2, bx2) - max(ax1, bx1)
    y_overlap = min(ay2, by2) - max(ay1, by1)
    if ax2 == bx1 and y_overlap >= MIN_SHARE_PX:
        return True
    if bx2 == ax1 and y_overlap >= MIN_SHARE_PX:
        return True
    if ay2 == by1 and x_overlap >= MIN_SHARE_PX:
        return True
    if by2 == ay1 and x_overlap >= MIN_SHARE_PX:
        return True
    return False


def _degrees(scene: dict) -> dict[str, int]:
    """Color -> number of neighbouring (contacting) regions, from geometry."""
    regions = scene["regions"]
    neighbors: dict[str, set[str]] = {r["color"]: set() for r in regions}
    for i in range(len(regions)):
        for j in range(i + 1, len(regions)):
            if _rect_contact(regions[i], regions[j]):
                neighbors[regions[i]["color"]].add(regions[j]["color"])
                neighbors[regions[j]["color"]].add(regions[i]["color"])
    return {color: len(s) for color, s in neighbors.items()}


def _sample_hub(rng: random.Random) -> list[dict]:
    """Unique-maximum hub configuration: one region touches all three others."""
    hub_color = rng.choice(PALETTE)
    leaves = [c for c in PALETTE if c != hub_color]
    for _ in range(20_000):
        hw = rng.randint(90, 150)
        hh = rng.randint(90, 150)
        lw = rng.randint(40, 80)
        rw = rng.randint(40, 80)
        bw = rng.randint(40, 80)
        lh = rng.randint(max(int(MIN_SHARE_PX) + 4, hh // 2), int(0.92 * hh))
        rh = rng.randint(max(int(MIN_SHARE_PX) + 4, hh // 2), int(0.92 * hh))
        bh = rng.randint(40, 80)

        hx = rng.randint(int(PAD) + lw, W - int(PAD) - hw - rw)
        hy = rng.randint(int(PAD) + (hh - lh) // 2, H - int(PAD) - hh - bh)

        ly = hy + (hh - lh) // 2
        ry = hy + (hh - rh) // 2
        bx = hx + (hw - bw) // 2

        hub = {"color": hub_color, "x": hx, "y": hy, "w": hw, "h": hh}
        regions = [
            hub,
            {"color": leaves[0], "x": hx - lw, "y": ly, "w": lw, "h": lh},
            {"color": leaves[1], "x": hx + hw, "y": ry, "w": rw, "h": rh},
            {"color": leaves[2], "x": bx, "y": hy + hh, "w": bw, "h": bh},
        ]
        if all(r["x"] >= 0 and r["y"] >= 0 and r["x"] + r["w"] <= W and r["y"] + r["h"] <= H
               for r in regions):
            return regions
    raise RuntimeError("could not place hub configuration")


def _sample_grid(rng: random.Random) -> list[dict]:
    """Tie (quarantined) configuration: a 2x2 grid, every region degree 2."""
    colors = PALETTE[:]
    rng.shuffle(colors)
    gw = rng.randint(60, 130)
    gh = rng.randint(60, 130)
    gx = rng.randint(int(GRID_PAD), W - int(GRID_PAD) - 2 * gw)
    gy = rng.randint(int(GRID_PAD), H - int(GRID_PAD) - 2 * gh)
    regions = [
        {"color": colors[0], "x": gx, "y": gy, "w": gw, "h": gh},
        {"color": colors[1], "x": gx + gw, "y": gy, "w": gw, "h": gh},
        {"color": colors[2], "x": gx, "y": gy + gh, "w": gw, "h": gh},
        {"color": colors[3], "x": gx + gw, "y": gy + gh, "w": gw, "h": gh},
    ]
    return regions


def sample_scene(rng: random.Random, width: int = W, height: int = H) -> dict:
    """Sample a scene. ~15% are tie (2x2 grid) scenes; the rest are hub scenes."""
    if rng.random() < 0.15:
        regions = _sample_grid(rng)
    else:
        regions = _sample_hub(rng)
    return {
        "regions": regions,
        "width": width,
        "height": height,
        "background": "white",
    }


def render(scene: dict, supersample: int = 1) -> Image.Image:
    """Render I = R(z). Deterministic, hard-edged, exact task geometry."""
    width = int(scene["width"])
    height = int(scene["height"])
    background = scene["background"]
    img = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(img)
    for region in scene["regions"]:
        x = int(region["x"])
        y = int(region["y"])
        draw.rectangle([x, y, x + int(region["w"]), y + int(region["h"])], fill=region["color"])
    return img


def _winner(scene: dict) -> str:
    """Colour with the most neighbours; ties broken by PALETTE order."""
    degrees = _degrees(scene)
    top = max(degrees.values())
    candidates = [c for c in PALETTE if degrees.get(c, -1) == top]
    return candidates[0]


def analytic_gold(scene: dict) -> str:
    """Winner colour recovered from the analytic contact graph.

    Self-contained (no calls into module-level helpers) so the PIXEL-oracle
    independence / latent-alias AST audit sees it reading only scene["regions"].
    """
    regions = scene["regions"]
    n = len(regions)
    neighbors: dict[str, set[str]] = {r["color"]: set() for r in regions}
    for i in range(n):
        ax1, ay1 = regions[i]["x"], regions[i]["y"]
        ax2 = ax1 + regions[i]["w"]
        ay2 = ay1 + regions[i]["h"]
        for j in range(i + 1, n):
            bx1, by1 = regions[j]["x"], regions[j]["y"]
            bx2 = bx1 + regions[j]["w"]
            by2 = by1 + regions[j]["h"]
            x_overlap = min(ax2, bx2) - max(ax1, bx1)
            y_overlap = min(ay2, by2) - max(ay1, by1)
            touch = False
            if ax2 == bx1 and y_overlap >= MIN_SHARE_PX:
                touch = True
            elif bx2 == ax1 and y_overlap >= MIN_SHARE_PX:
                touch = True
            elif ay2 == by1 and x_overlap >= MIN_SHARE_PX:
                touch = True
            elif by2 == ay1 and x_overlap >= MIN_SHARE_PX:
                touch = True
            if touch:
                neighbors[regions[i]["color"]].add(regions[j]["color"])
                neighbors[regions[j]["color"]].add(regions[i]["color"])
    degrees = {c: len(s) for c, s in neighbors.items()}
    top = max(degrees.values())
    candidates = [c for c in PALETTE if degrees.get(c, -1) == top]
    return candidates[0]


def margin(scene: dict) -> float:
    """Decision margin: count_difference between the top two contact degrees."""
    degrees = sorted(_degrees(scene).values(), reverse=True)
    if len(degrees) < 2:
        return float(degrees[0]) if degrees else 0.0
    return float(degrees[0] - degrees[1])


def is_quarantined(scene: dict) -> bool:
    """Quarantine scenes whose maximum is ambiguous (a tie or near-tie)."""
    return margin(scene) < QUARANTINE_MIN


def latent_symmetries(scene: dict) -> Iterator[dict]:
    """No nontrivial render symmetry: every colour is unique and every field
    directly changes the picture; the contact-graph decision is recovered purely
    from pixels. Declared not_applicable."""
    return iter(())
