"""Independent pixel-only inverse arm for the region-contact-degree world.

receives ONLY the final PNG. It never imports or reads the renderer, the
analytic scene, prompts, or any renderer state. It recovers the decision by:

1. labelling every pixel to its nearest palette colour among the four region
   colours plus white (a 50%-split nearest-colour assignment, so antialiased
   edge pixels land on the true boundary and put no third colour in the way);
2. for each pair of distinct region colours, checking whether any pixel of one
   colour is 4-adjacent to a pixel of the other colour (a genuine shared edge,
   not a diagonal corner contact);
3. counting that per-colour contact degree and returning the colour with the
   maximum degree, breaking ties deterministically by PALETTE order.

This reproduces the analytic contact graph exactly on robust scenes.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "region_contact_degree-oracle-0.4.0"

# Must mirror renderer.PALETTE order so tie-breaking is identical, but this
# module shares no code with the renderer (it only records the colour names).
PALETTE = ["red", "green", "blue", "magenta"]
_COLOR_RGBS = {
    "red": (255, 0, 0),
    "green": (0, 255, 0),
    "blue": (0, 0, 255),
    "magenta": (255, 0, 255),
}
_BACKGROUND = "white"
_WHITE_RGB = (255, 255, 255)


def _pairwise_contacts(masks: dict[str, np.ndarray]) -> set[tuple[str, str]]:
    """Colour pairs that are 4-adjacent somewhere (shared edge, not a corner)."""
    h, w = next(iter(masks.values())).shape
    # Combined integer label image: -1 background, else palette index.
    labels = np.full((h, w), -1, dtype=np.int64)
    for idx, color in enumerate(PALETTE):
        labels[masks[color]] = idx
    contacts: set[tuple[str, str]] = set()
    for y in range(h):
        for x in range(w):
            c = labels[y, x]
            if c < 0:
                continue
            for nx, ny in ((x + 1, y), (x, y + 1)):
                if 0 <= nx < w and 0 <= ny < h:
                    d = labels[ny, nx]
                    if d >= 0 and d != c:
                        contacts.add((PALETTE[c], PALETTE[d]))
    return contacts


def decision_from_image(image: Image.Image) -> str:
    """Winner colour (max contact degree) recovered from the PNG alone."""
    arr = np.asarray(image.convert("RGB")).astype(np.float32)
    # Nearest-colour labelling over the four region colours plus white.
    rgb_list = [_WHITE_RGB] + [_COLOR_RGBS[c] for c in PALETTE]
    dists = np.stack(
        [np.linalg.norm(arr - np.array(rgb, dtype=np.float32), axis=-1) for rgb in rgb_list],
        axis=-1,
    )
    labels = dists.argmin(axis=-1)  # 0 = white, 1..4 = region colours in order

    masks = {color: labels == (1 + idx) for idx, color in enumerate(PALETTE)}
    contacts = _pairwise_contacts(masks)
    degree = {color: 0 for color in PALETTE}
    for a, b in contacts:
        degree[a] += 1
        degree[b] += 1
    top = max(degree.values())
    winners = [c for c in PALETTE if degree[c] == top]
    return winners[0]
