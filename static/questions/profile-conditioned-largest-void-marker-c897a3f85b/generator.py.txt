"""Largest-void marker selection world.

Global figure-ground / texture-energy reveal: a scattered field of small dark
dots covers a white canvas, leaving one large empty void that is clearly larger
than every other opening in the field. Four coloured marker dots are placed;
exactly one sits inside that largest void (far from every dot), while the other
three sit right among the dots. The answer is the colour of the marker that
occupies the largest empty void, which is a property of the whole dot field's
negative space rather than of any single mark.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "largest_void_marker-0.1.0"

CANVAS = 512
BACKGROUND = "white"
INK_COLOR = (25, 25, 25)

MARKER_COLORS = {
    "red": (220, 40, 40),
    "green": (40, 175, 70),
    "blue": (45, 85, 225),
    "orange": (240, 145, 35),
}
MARKER_RADIUS = 8.0
DOT_RADIUS = 4.5
DOT_COUNT = 140

VOID_R_MIN = 120.0
VOID_R_MAX = 150.0
VOID_CENTER_RANGE = (135.0, 377.0)

# A marker that is this close to the dot cloud's nearest ink is ambiguous.
QUARANTINE_MARGIN = 30.0


def sample_scene(seed: int) -> dict:
    """Deterministically sample a full latent scene from an integer seed."""
    rng = random.Random(seed)

    answer_color = rng.choice(sorted(MARKER_COLORS))

    cx = rng.uniform(*VOID_CENTER_RANGE)
    cy = rng.uniform(*VOID_CENTER_RANGE)
    void_r = rng.uniform(VOID_R_MIN, VOID_R_MAX)

    def outside_void(x: float, y: float, pad: float = 0.0) -> bool:
        return math.hypot(x - cx, y - cy) > (void_r + pad)

    # Scatter the dark dots uniformly, keeping every dot clear of the void.
    dots = []
    attempts = 0
    while len(dots) < DOT_COUNT and attempts < DOT_COUNT * 60:
        attempts += 1
        x = rng.uniform(10.0, CANVAS - 10.0)
        y = rng.uniform(10.0, CANVAS - 10.0)
        if outside_void(x, y, pad=DOT_RADIUS + 4.0):
            if all(math.hypot(x - dx, y - dy) > 2.2 * DOT_RADIUS for dx, dy in dots):
                dots.append((x, y))
    if len(dots) < DOT_COUNT:  # pragmatic guarantee: top up uniformly outside void
        for _ in range(DOT_COUNT):
            x = rng.uniform(20.0, CANVAS - 20.0)
            y = rng.uniform(20.0, CANVAS - 20.0)
            if outside_void(x, y, pad=DOT_RADIUS):
                dots.append((x, y))
            if len(dots) >= DOT_COUNT:
                break

    # Target marker sits safely inside the large void.
    angle = rng.uniform(0.0, 2.0 * math.pi)
    off = rng.uniform(0.0, 0.4 * void_r)
    tx = cx + off * math.cos(angle)
    ty = cy + off * math.sin(angle)

    def nearest_dot_dist(px, py, exclude=None):
        best = float("inf")
        for idx, (dx, dy) in enumerate(dots):
            if exclude is not None and idx == exclude:
                continue
            best = min(best, math.hypot(px - dx, py - dy))
        return best

    target = {"color": answer_color, "x": tx, "y": ty, "radius": MARKER_RADIUS}
    target_clearance = nearest_dot_dist(tx, ty)

    # Distractors: three markers dropped in the dot field, each near a dot but
    # never inside the void. Their nearest-dot clearance stays small.
    distractors = []
    distractor_colors = [c for c in sorted(MARKER_COLORS) if c != answer_color]
    markers_in_scene = [target]
    used_dots = set()
    for k, color in enumerate(distractor_colors):
        placed = False
        for _attempt in range(200):
            cand = rng.randrange(len(dots))
            if cand in used_dots:
                continue
            dx, dy = dots[cand]
            a2 = rng.uniform(0.0, 2.0 * math.pi)
            r2 = rng.uniform(20.0, 32.0)
            px = dx + r2 * math.cos(a2)
            py = dy + r2 * math.sin(a2)
            if not (10.0 < px < CANVAS - 10.0 and 10.0 < py < CANVAS - 10.0):
                continue
            if not outside_void(px, py, pad=5.0):
                continue
            if any(math.hypot(px - m["x"], py - m["y"]) < 26.0 for m in markers_in_scene):
                continue
            used_dots.add(cand)
            distractors.append({"color": color, "x": px, "y": py, "radius": MARKER_RADIUS})
            markers_in_scene.append({"x": px, "y": py})
            placed = True
            break
        if not placed:  # pragmatic fallback: put the marker beside the first dot
            dx, dy = dots[0]
            px, py = dx + 30.0, dy + 0.0
            distractors.append({"color": color, "x": px, "y": py, "radius": MARKER_RADIUS})

    markers = [target] + distractors

    return {
        "canvas": CANVAS,
        "background": BACKGROUND,
        "ink_color": list(INK_COLOR),
        "void": {"cx": round(cx, 3), "cy": round(cy, 3), "r": round(void_r, 3)},
        "dots": [[round(x, 3), round(y, 3)] for x, y in dots],
        "markers": markers,
        "answer_color": answer_color,
        "target_clearance": round(target_clearance, 3),
        "void_radius": round(void_r, 3),
    }


def render(scene: dict, supersample: int = 4) -> Image.Image:
    """Render the dot field with the four coloured markers."""
    ans = scene["answer_color"]
    if ans not in MARKER_COLORS:
        raise ValueError(f"unknown answer colour {ans!r}")
    width = int(scene["canvas"])
    ss = supersample
    img = Image.new("RGB", (width * ss, width * ss), scene["background"])
    draw = ImageDraw.Draw(img)

    for x, y in scene["dots"]:
        r = DOT_RADIUS * ss
        draw.ellipse([x * ss - r, y * ss - r, x * ss + r, y * ss + r], fill=INK_COLOR)

    for marker in scene["markers"]:
        color = MARKER_COLORS[marker["color"]]
        r = float(marker["radius"]) * ss
        mx = float(marker["x"]) * ss
        my = float(marker["y"]) * ss
        draw.ellipse([mx - r, my - r, mx + r, my + r], fill=color)

    return img.resize((width, width), Image.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """The answer is the colour of the marker sitting in the largest void."""
    return scene["answer_color"]


def _clearances(scene: dict) -> dict:
    dots = [(float(x), float(y)) for x, y in scene["dots"]]
    clearances = {}
    for marker in scene["markers"]:
        mx, my = float(marker["x"]), float(marker["y"])
        best = min(math.hypot(mx - dx, my - dy) for dx, dy in dots)
        clearances[marker["color"]] = best
    return clearances


def margin(scene: dict) -> float:
    """Two-sided verifiability margin: the target marker's nearest-dot clearance
    minus the largest distractor clearance. A large positive gap means the void
    marker is unambiguous at pixel scale; tiny gaps are quarantined."""
    cls = _clearances(scene)
    answer = scene["answer_color"]
    target_c = cls[answer]
    near = max(v for c, v in cls.items() if c != answer)
    return round(target_c - near, 3)


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_MARGIN)


def latent_symmetries(scene: dict) -> list:
    """No declared latent symmetry.

    The void geometry, the dot field, and the marker colours are all directly
    visible; the void marker colour is an injective function of the rendered
    scene, and no render-invariant relabelling leaves both pixels and gold fixed.
    """
    return []
