"""Independent pixel-only inverse arm for the largest-void marker world.

Receives only the rendered PNG. It segments the saturated coloured markers from
the dark-dot field, takes each marker's centroid, measures the distance from
every marker to the nearest dark-ink pixel (the surrounding dot cloud), and
returns the colour of the single marker whose clearance is largest -- the one
lying in the largest empty void.

This module imports neither renderer, prompts, generate, verify, nor any
analytic-gold or scene code, and never receives latent scene data.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_WHITE_MIN = 200
_INK_MAX = 70
_MIN_MARKER_PIXELS = 4

_PALETTE = {
    "red": np.array([220, 40, 40]),
    "green": np.array([40, 175, 70]),
    "blue": np.array([45, 85, 225]),
    "orange": np.array([240, 145, 35]),
}


def _marker_components(arr: np.ndarray):
    """Find 4-connected saturated coloured marker blobs and name their colour."""
    h, w, _ = arr.shape
    r = arr[:, :, 0].astype(int)
    g = arr[:, :, 1].astype(int)
    b = arr[:, :, 2].astype(int)
    mx = np.maximum(np.maximum(r, g), b)
    mn = np.minimum(np.minimum(r, g), b)
    marker_mask = (mx - mn) >= 60

    visited = np.zeros((h, w), dtype=bool)
    coords = set(map(tuple, np.argwhere(marker_mask)))
    components = []
    for start in coords:
        if visited[start]:
            continue
        stack = {start}
        blob = []
        while stack:
            cell = stack.pop()
            if visited[cell]:
                continue
            visited[cell] = True
            blob.append(cell)
            rr, cc = cell
            for nr, nc in ((rr + 1, cc), (rr - 1, cc), (rr, cc + 1), (rr, cc - 1)):
                if 0 <= nr < h and 0 <= nc < w and marker_mask[nr, nc] and not visited[nr, nc]:
                    stack.add((nr, nc))
        if len(blob) >= _MIN_MARKER_PIXELS:
            xs = sum(p[1] for p in blob) / len(blob)
            ys = sum(p[0] for p in blob) / len(blob)
            colour = _classify(arr[int(round(ys)), int(round(xs))])
            components.append((xs, ys, colour))
    return components


def _classify(rgb) -> str:
    rgb = np.asarray(rgb, dtype=float)
    best, best_name = None, None
    for name, pcolour in _PALETTE.items():
        dist = float(np.linalg.norm(rgb - pcolour))
        if best is None or dist < best:
            best, best_name = dist, name
    return best_name


def _nearest_ink_distance(arr: np.ndarray, start: tuple[int, int]) -> float:
    """BFS distance from a marker centroid to the nearest dark-ink pixel.

    The distance is measured in pixel steps over the region that is neither the
    marker's own colour nor already 'visited'. Because the dot field is small
    isolated ink while the void is empty, the number of steps until the first
    dark ink pixel is a robust proxy for the geometric void clearance.
    """
    h, w, _ = arr.shape
    max_channel = np.max(arr, axis=2)
    is_ink = max_channel < _INK_MAX
    if is_ink[start]:
        return 0.0
    visited = np.zeros((h, w), dtype=bool)
    frontier = {start}
    visited[start] = True
    steps = 0
    while frontier:
        steps += 1
        nxt = set()
        for rr, cc in frontier:
            for nr, nc in ((rr + 1, cc), (rr - 1, cc), (rr, cc + 1), (rr, cc - 1)):
                if 0 <= nr < h and 0 <= nc < w and not visited[nr, nc]:
                    if is_ink[nr, nc]:
                        return float(steps)
                    visited[nr, nc] = True
                    nxt.add((nr, nc))
        frontier = nxt
    return float(max(h, w))


def decision_from_image(image: Image.Image) -> str:
    """Recover the void marker colour from the rasterized PNG alone."""
    arr = np.asarray(image.convert("RGB")).astype(np.int32)
    markers = _marker_components(arr)
    if not markers:
        return None
    best_colour, best_dist = None, -1.0
    for (x, y, colour) in markers:
        start = (int(round(y)), int(round(x)))
        dist = _nearest_ink_distance(arr, start)
        if dist > best_dist:
            best_dist, best_colour = dist, colour
    return best_colour
