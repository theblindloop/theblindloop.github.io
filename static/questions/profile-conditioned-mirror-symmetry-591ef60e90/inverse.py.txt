"""Independent pixel-only inverse arm for the mirror-symmetry world.

Recovers the decision (mirror-symmetric across the dashed vertical axis?)
from the final rasterized PNG alone. It shares no geometry, palette, sizes or
thresholds with ``renderer`` beyond the physical fact that a gray dashed
vertical line lies near the canvas centre; it detects that axis itself.

This module imports no renderer / prompts / generate / verify / analytic-gold
code and never receives the latent scene.
"""

from __future__ import annotations

from PIL import Image
import numpy as np

AXIS = None  # detected at runtime, never assumed
BAND = 6
_DIFF = 25
_DECIDE_YES = 60


def _find_axis(arr: np.ndarray) -> int:
    """Locate the vertical gray dashed axis by its gray-pixel column density."""
    r, g, b = arr[..., 0].astype(int), arr[..., 1].astype(int), arr[..., 2].astype(int)
    grayish = (np.abs(r - g) < 15) & (np.abs(g - b) < 15) & (r > 80) & (r < 230)
    col_count = grayish.sum(axis=0)
    peak = int(col_count.max())
    if peak <= 0:
        return 0
    cols = np.where(col_count >= 0.9 * peak)[0]
    return int(round((float(cols.min()) + float(cols.max())) / 2.0))


def decision_from_image(image: Image.Image) -> str:
    """Return 'yes' if the figure is mirror-symmetric about the dashed axis."""
    arr = np.asarray(image.convert("RGB")).astype(int)
    x0 = _find_axis(arr)
    w = arr.shape[1]
    xs = np.arange(1, x0 - BAND)
    xr = 2 * x0 - xs
    valid = (xr > 0) & (xr < w)
    xs, xr = xs[valid], xr[valid]
    if xs.size == 0:
        return "no"
    diff = np.abs(arr[:, xs] - arr[:, xr]).sum(axis=2)
    mismatch = int((diff > _DIFF).sum())
    return "yes" if mismatch <= _DECIDE_YES else "no"
