"""Mirror-symmetry world: detect a global reflection-symmetry relation.

A dashed vertical axis splits the canvas into left and right halves, each
carrying colored disks. The decision is the global Gestalt property of
mirror symmetry: whether the right half is an exact horizontal reflection
(mirror) of the left half across the dashed axis.

The renderer owns scene sampling + rasterization + the structural analytic
gold. The independent pixel-only inverse arm (``oracle.py``) recovers the
same decision from the final PNG alone.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
AXIS = WIDTH // 2  # 256
BACKGROUND = (255, 255, 255)
DASH_GRAY = (140, 140, 140)
DASH_LEN = 12
DASH_GAP = 10
DASH_WIDTH = 3
# Saturated, mutually distinct ink colors (none is gray, so the oracle can
# unambiguously find the gray dashed axis).
PALETTE = [
    (215, 40, 40),
    (40, 110, 220),
    (40, 170, 80),
    (230, 150, 20),
    (150, 60, 180),
    (40, 190, 190),
]

# Extent inside which left-half element centers are placed so that no element
# touches the axis band or the image border.
_LEFT_LO = 40
_LEFT_HI = AXIS - 40
_TOP = 30
_BOT = HEIGHT - 30
_MIN_SIZE = 10
_MAX_SIZE = 15
# Axis exclusion band used by the mirror-mismatch measurement (px each side).
_BAND = 6
# Pixel-difference threshold: pairs whose per-channel sums exceed this are
# counted as mismatched.
_DIFF = 25
# Decision / quarantine thresholds on the mirror-mismatch count.
YES_MAX = 60
NO_MIN = 280


def _color(cidx: int) -> tuple[int, int, int]:
    return PALETTE[cidx % len(PALETTE)]


def sample_scene(seed: int) -> dict:
    """Sample one latent scene deterministically from ``seed``."""
    rng = np.random.default_rng(seed)
    n = int(rng.integers(3, 6))
    left: list[list[int]] = []
    for _ in range(n):
        x = int(rng.integers(_LEFT_LO, _LEFT_HI))
        y = int(rng.integers(_TOP, _BOT))
        cidx = int(rng.integers(0, len(PALETTE)))
        rsize = int(rng.integers(_MIN_SIZE, _MAX_SIZE + 1))
        left.append([x, y, cidx, rsize])

    right: list[list[int]] = []
    for (x, y, cidx, rsize) in left:
        right.append([AXIS + (AXIS - x), y, cidx, rsize])

    symmetric = bool(rng.integers(0, 2))
    if not symmetric:
        k = int(rng.integers(0, len(right)))
        if int(rng.integers(0, 2)) == 0:
            # Color swap: break symmetry without moving the ink footprint.
            new_c = (right[k][2] + 1 + int(rng.integers(0, len(PALETTE) - 1))) % len(PALETTE)
            right[k][2] = new_c
        else:
            # Displacement: shove the element further right (kept on canvas).
            dx = int(rng.integers(24, 56))
            right[k][0] = min(WIDTH - 15, right[k][0] + dx)

    return {"left_elements": left, "right_elements": right}


def _draw_scene(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    d = ImageDraw.Draw(img)
    y = 0
    while y < HEIGHT:
        d.rectangle(
            [AXIS - DASH_WIDTH // 2, y, AXIS - DASH_WIDTH // 2 + DASH_WIDTH, y + DASH_LEN],
            fill=DASH_GRAY,
        )
        y += DASH_LEN + DASH_GAP
    for (x, yc, cidx, rsize) in scene["left_elements"]:
        d.ellipse([x - rsize, yc - rsize, x + rsize, yc + rsize], fill=_color(cidx))
    for (x, yc, cidx, rsize) in scene["right_elements"]:
        d.ellipse([x - rsize, yc - rsize, x + rsize, yc + rsize], fill=_color(cidx))
    return img


def render(scene: dict) -> Image.Image:
    """Rasterize the latent scene (a deterministic anti-aliased-free PNG)."""
    return _draw_scene(scene)


def _mirror_of_left(left: list[list[int]]) -> list[tuple[int, int, int, int]]:
    return [(AXIS + (AXIS - x), y, cidx, rsize) for (x, y, cidx, rsize) in left]


def analytic_gold(scene: dict) -> str:
    """Structural gold: right half is a mirror of the left half?"""
    left = scene["left_elements"]
    right = scene["right_elements"]
    if len(right) != len(left):
        return "no"
    refl = sorted(map(tuple, _mirror_of_left(left)))
    actual = sorted(map(tuple, right))
    return "yes" if refl == actual else "no"


def mirror_mismatch(scene: dict) -> int:
    """Count pixels where reflecting the left half disagrees with the right."""
    arr = np.asarray(_draw_scene(scene).convert("RGB")).astype(int)
    x0 = AXIS
    xs = np.arange(1, x0 - _BAND)
    xr = 2 * x0 - xs
    valid = xr < WIDTH
    xs, xr = xs[valid], xr[valid]
    diff = np.abs(arr[:, xs] - arr[:, xr]).sum(axis=2)
    return int((diff > _DIFF).sum())


def margin(scene: dict) -> float:
    """Commitment magnitude: the mirror-mismatch pixel count."""
    return float(mirror_mismatch(scene))


def is_quarantined(scene: dict) -> bool:
    """Quarantine the ambiguous mismatch band between YES_MAX and NO_MIN."""
    m = mirror_mismatch(scene)
    return YES_MAX < m < NO_MIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The scene is injectively rendered; no latent aliases are declared."""
    return []
