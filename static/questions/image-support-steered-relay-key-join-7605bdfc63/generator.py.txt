"""Deterministic latent sampler and raster renderer."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 960, 640
BG = (247, 242, 226)
PANEL = (255, 253, 246)
NAVY = (24, 48, 72)
GOLD = (211, 164, 45)
GRAY = (155, 151, 139)
RELAY = [(201, 72, 88), (35, 154, 143), (235, 145, 52), (126, 91, 166)]
TAGS = [(221, 53, 69), (45, 157, 76), (42, 111, 196)]
TAG_NAMES = ("red", "green", "blue")


def _wrong_permutations(true_relation: list[int], rng: random.Random) -> list[list[int]]:
    pool = []
    for i in range(4):
        for j in range(i + 1, 4):
            item = list(true_relation)
            item[i], item[j] = item[j], item[i]
            pool.append(item)
    rng.shuffle(pool)
    return pool[:2]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    shape_to_relay = list(range(4))
    relay_to_tile = list(range(4))
    rng.shuffle(shape_to_relay)
    rng.shuffle(relay_to_tile)
    true_relation = [relay_to_tile[shape_to_relay[s]] for s in range(4)]
    wrong = _wrong_permutations(true_relation, rng)
    true_position = rng.randrange(3)
    candidate_relations = []
    wi = 0
    for position in range(3):
        if position == true_position:
            candidate_relations.append(true_relation)
        else:
            candidate_relations.append(wrong[wi])
            wi += 1
    target_tag = seed % 3
    other_tags = [value for value in range(3) if value != target_tag]
    rng.shuffle(other_tags)
    tag_order = [None, None, None]
    tag_order[true_position] = target_tag
    oi = 0
    for position in range(3):
        if tag_order[position] is None:
            tag_order[position] = other_tags[oi]
            oi += 1
    left_order = list(range(4))
    right_order = list(range(4))
    rng.shuffle(left_order)
    rng.shuffle(right_order)
    candidate_orders = []
    for _ in range(3):
        order = list(range(4))
        rng.shuffle(order)
        candidate_orders.append(order)
    return {
        "shape_to_relay": shape_to_relay,
        "relay_to_tile": relay_to_tile,
        "left_order": left_order,
        "right_order": right_order,
        "candidate_relations": candidate_relations,
        "candidate_orders": candidate_orders,
        "tag_order": tag_order,
    }


def _beads(draw: ImageDraw.ImageDraw, cx: int, cy: int, count: int, scale: int) -> None:
    r = 7 * scale
    gap = 18 * scale
    start = cx * scale - (count - 1) * gap // 2
    for i in range(count):
        x = start + i * gap
        y = cy * scale
        draw.ellipse((x - r, y - r, x + r, y + r), fill=NAVY)


def _tile(draw: ImageDraw.ImageDraw, cx: int, cy: int, count: int, scale: int) -> None:
    w, h = 64 * scale, 30 * scale
    x0, y0 = cx * scale - w // 2, cy * scale - h // 2
    draw.rounded_rectangle((x0, y0, x0 + w, y0 + h), radius=5 * scale, fill=(232, 228, 214), outline=GRAY, width=2 * scale)
    gap = 12 * scale
    start = cx * scale - (count - 1) * gap // 2
    for i in range(count):
        x = start + i * gap
        draw.rounded_rectangle((x - 3 * scale, y0 + 4 * scale, x + 3 * scale, y0 + h - 4 * scale), radius=2 * scale, fill=NAVY)


def _relay_dot(draw: ImageDraw.ImageDraw, cx: int, cy: int, relay: int, scale: int) -> None:
    r = 12 * scale
    x, y = cx * scale, cy * scale
    draw.ellipse((x - r, y - r, x + r, y + r), fill=RELAY[relay], outline=NAVY, width=2 * scale)


def _arrow(draw: ImageDraw.ImageDraw, x0: int, x1: int, y: int, scale: int) -> None:
    draw.line((x0 * scale, y * scale, x1 * scale, y * scale), fill=GRAY, width=2 * scale)
    draw.polygon(((x1 * scale, y * scale), ((x1 - 7) * scale, (y - 5) * scale), ((x1 - 7) * scale, (y + 5) * scale)), fill=GRAY)


def render(scene: dict) -> Image.Image:
    shape_to_relay = scene["shape_to_relay"]
    relay_to_tile = scene["relay_to_tile"]
    left_order = scene["left_order"]
    right_order = scene["right_order"]
    candidate_relations = scene["candidate_relations"]
    candidate_orders = scene["candidate_orders"]
    tag_order = scene["tag_order"]
    scale = 2
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BG)
    draw = ImageDraw.Draw(image)

    for box in ((35, 20, 415, 234), (545, 20, 925, 234)):
        draw.rounded_rectangle(tuple(value * scale for value in box), radius=16 * scale, fill=PANEL, outline=GOLD, width=7 * scale)
    ys = [48, 98, 148, 198]
    for row, shape in enumerate(left_order):
        y = ys[row]
        _beads(draw, 125, y, shape + 1, scale)
        _arrow(draw, 205, 285, y, scale)
        _relay_dot(draw, 330, y, shape_to_relay[shape], scale)
    relay_to_shape = {relay: shape for shape, relay in enumerate(shape_to_relay)}
    for row, relay in enumerate(right_order):
        y = ys[row]
        _relay_dot(draw, 625, y, relay, scale)
        _arrow(draw, 670, 750, y, scale)
        _tile(draw, 830, y, relay_to_tile[relay] + 1, scale)

    centers = [165, 480, 795]
    for position, cx in enumerate(centers):
        draw.rounded_rectangle(((cx - 132) * scale, 380 * scale, (cx + 132) * scale, 625 * scale), radius=14 * scale, fill=PANEL, outline=GRAY, width=3 * scale)
        tag = tag_order[position]
        draw.ellipse(((cx - 18) * scale, 350 * scale, (cx + 18) * scale, 386 * scale), fill=TAGS[tag], outline=NAVY, width=3 * scale)
        relation = candidate_relations[position]
        for row, shape in enumerate(candidate_orders[position]):
            y = 414 + row * 52
            _beads(draw, cx - 65, y, shape + 1, scale)
            _arrow(draw, cx - 20, cx + 17, y, scale)
            _tile(draw, cx + 75, y, relation[shape] + 1, scale)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def _true_relation(scene: dict) -> list[int]:
    shape_to_relay = scene["shape_to_relay"]
    relay_to_tile = scene["relay_to_tile"]
    return [relay_to_tile[shape_to_relay[shape]] for shape in range(4)]


def _scores(scene: dict) -> list[int]:
    true_relation = _true_relation(scene)
    candidate_relations = scene["candidate_relations"]
    return [sum(a != b for a, b in zip(candidate, true_relation)) for candidate in candidate_relations]


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    best = scores.index(min(scores))
    tag_order = scene["tag_order"]
    return TAG_NAMES[tag_order[best]]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene))
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    scores = _scores(scene)
    if len(scene["left_order"]) != 4 or len(scene["right_order"]) != 4 or len(scene["candidate_orders"]) != 3:
        return True
    return scores.count(0) != 1 or margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
