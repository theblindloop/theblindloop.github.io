"""Independent pixel-only inverse for the two-key relay world."""

from __future__ import annotations

import numpy as np
from PIL import Image

NAVY = np.array((24, 48, 72), dtype=np.int16)
RELAY = [np.array(c, dtype=np.int16) for c in ((201, 72, 88), (35, 154, 143), (235, 145, 52), (126, 91, 166))]
TAGS = [np.array(c, dtype=np.int16) for c in ((221, 53, 69), (45, 157, 76), (42, 111, 196))]
TAG_NAMES = ("red", "green", "blue")


def _mask(arr: np.ndarray, color: np.ndarray, radius: float = 46.0) -> np.ndarray:
    return np.linalg.norm(arr.astype(np.int16) - color, axis=2) < radius


def _components(mask: np.ndarray, minimum: int = 8) -> list[tuple[int, int, int, int, int, float, float]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for sy, sx in np.argwhere(mask):
        if seen[sy, sx]:
            continue
        stack = [(int(sy), int(sx))]
        seen[sy, sx] = True
        xs, ys = [], []
        while stack:
            y, x = stack.pop()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(xs) >= minimum:
            output.append((min(xs), min(ys), max(xs), max(ys), len(xs), float(np.mean(xs)), float(np.mean(ys))))
    return output


def _color_markers(arr: np.ndarray, palette: list[np.ndarray], y_limit: tuple[float, float]) -> list[tuple[int, float, float, int]]:
    found = []
    for index, color in enumerate(palette):
        for component in _components(_mask(arr, color), 40):
            x0, y0, x1, y1, area, cx, cy = component
            if y_limit[0] < cy < y_limit[1] and 80 <= area <= 1800 and x1 - x0 >= 8 and y1 - y0 >= 8:
                found.append((index, cx, cy, area))
    return found


def _validated_navy_components(arr: np.ndarray) -> list[tuple[int, int, int, int, int, float, float]]:
    components = _components(_mask(arr, NAVY), 18)
    clean = []
    for component in components:
        x0, y0, x1, y1, area, cx, cy = component
        w, h = x1 - x0 + 1, y1 - y0 + 1
        # Only the compact bead disks and narrow tile stripes are data marks.
        if 4 <= w <= 18 and 7 <= h <= 26 and 28 <= area <= 330:
            clean.append(component)
    return clean


def _cluster_rows(items: list[tuple], expected: int = 4) -> list[list[tuple]]:
    if not items:
        raise ValueError("missing row marks")
    groups = []
    for item in sorted(items, key=lambda value: value[-1]):
        cy = item[-1]
        if not groups or cy - np.mean([entry[-1] for entry in groups[-1]]) > 24:
            groups.append([item])
        else:
            groups[-1].append(item)
    if len(groups) != expected:
        raise ValueError(f"expected {expected} visible rows, found {len(groups)}")
    return groups


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    height, width = arr.shape[:2]
    if width < 600 or height < 400:
        raise ValueError("canvas too small")
    top_cut = height * 0.43
    relay_marks = _color_markers(arr, RELAY, (0, top_cut))
    if len(relay_marks) != 8:
        raise ValueError(f"expected eight intact relay dots, found {len(relay_marks)}")
    left_relays = sorted([item for item in relay_marks if item[1] < width / 2], key=lambda item: item[2])
    right_relays = sorted([item for item in relay_marks if item[1] > width / 2], key=lambda item: item[2])
    if len(left_relays) != 4 or len(right_relays) != 4 or {x[0] for x in left_relays} != set(range(4)) or {x[0] for x in right_relays} != set(range(4)):
        raise ValueError("relay keys are not two complete visible bijections")

    navy = _validated_navy_components(arr)
    left_beads = [item for item in navy if item[5] < width * 0.25 and item[6] < top_cut]
    right_stripes = [item for item in navy if item[5] > width * 0.75 and item[6] < top_cut]
    left_rows = _cluster_rows(left_beads)
    right_rows = _cluster_rows(right_stripes)
    left_counts = [len(row) for row in left_rows]
    right_counts = [len(row) for row in right_rows]
    if sorted(left_counts) != [1, 2, 3, 4] or sorted(right_counts) != [1, 2, 3, 4]:
        raise ValueError("key bead or stripe counts are incomplete")
    shape_to_relay = {count - 1: relay[0] for count, relay in zip(left_counts, left_relays)}
    relay_to_tile = {relay[0]: count - 1 for count, relay in zip(right_counts, right_relays)}
    true_relation = tuple(relay_to_tile[shape_to_relay[shape]] for shape in range(4))

    tags = _color_markers(arr, TAGS, (top_cut, height))
    if not tags or len({item[0] for item in tags}) != len(tags):
        raise ValueError("no usable candidate tag")
    tags.sort(key=lambda item: item[1])
    candidate_relations = []
    for tag in tags:
        cx = tag[1]
        left_bound = cx - width * 0.155
        right_bound = cx + width * 0.155
        marks = [item for item in navy if left_bound < item[5] < right_bound and item[6] > top_cut + 20]
        bead_marks = [item for item in marks if item[5] < cx]
        stripe_marks = [item for item in marks if item[5] > cx]
        try:
            bead_rows = _cluster_rows(bead_marks)
            stripe_rows = _cluster_rows(stripe_marks)
            bead_counts = [len(row) for row in bead_rows]
            stripe_counts = [len(row) for row in stripe_rows]
            if sorted(bead_counts) != [1, 2, 3, 4] or sorted(stripe_counts) != [1, 2, 3, 4]:
                raise ValueError("candidate card has incomplete row marks")
            relation = [None] * 4
            for beads, stripes in zip(bead_counts, stripe_counts):
                relation[beads - 1] = stripes - 1
            candidate_relations.append((tag[0], tuple(relation)))
        except ValueError:
            continue
    exact = [tag_index for tag_index, candidate in candidate_relations if candidate == true_relation]
    if len(exact) != 1:
        raise ValueError("candidate assignment is not uniquely exact")
    return TAG_NAMES[exact[0]]
