"""Cyclic phase-orientation register (state tracking).

The image shows a single arrow token carried through K=4 ordered panels (left
to right). Each panel draws the token's current orientation as an arrow whose
colored head points in one of four directions: EAST, SOUTH, WEST, or NORTH.
A red marker ring around the leftmost arrow marks the start of the tracked
chain. Between each consecutive panel sits an operator glyph: a RED circle
means "rotate the token one quarter turn clockwise" and a BLUE square means
"rotate the token one quarter turn counterclockwise".

The latent scene stores only ``initial`` (the starting direction index), the
``ops`` (the operators drawn between the panels), and ``orientations`` (each
panel's arrow direction index, already consistent with the operators). The
decision is the final orientation, which is a cumulative function of the whole
sequence: final index = (initial + sum(ops)) mod 4 over a cyclic register, so
the state wraps around the closed 4-direction loop rather than ending at a
bounded boundary.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 820
HEIGHT = 200
PANEL_W = 140
PANEL_H = 150
GAP = 60
MARGIN_LEFT = 30
PITCH = PANEL_W + GAP
HALF_H = 75
CENTER_Y = HEIGHT // 2

TOKEN_R = 26
HEAD_RADIUS = 46
STEM_LEN = 34
HEAD_W = 11
STEM_W = 7

BG = (255, 255, 255)
BORDER = (200, 200, 200)
INK = (0, 0, 0)
RING_MARK = (240, 150, 20)
CW_COLOR = (210, 40, 40)
CCW_COLOR = (40, 60, 210)
HEAD_COLOR = (170, 40, 170)

GLYPH_R = 14

# Direction index encoding: 0=E, 1=S, 2=W, 3=N. Under screen coords (+y down)
# advancing the index +1 moves the arrow clockwise (E->S->W->N->E).
DIR_VECS = [(1, 0), (0, 1), (-1, 0), (0, -1)]
DIR_LABELS = {0: "east", 1: "south", 2: "west", 3: "north"}
PANEL_COUNT = 4


def _panel_center(index: int) -> tuple[int, int]:
    return (MARGIN_LEFT + PITCH * index + PANEL_W // 2, CENTER_Y)


def _glyph_center(index: int) -> tuple[int, int]:
    left = MARGIN_LEFT + PITCH * index + PANEL_W
    return (left + GAP // 2, CENTER_Y)


def sample_scene(seed: int) -> dict:
    """Deterministic latent scene from an integer seed."""
    rng = random.Random(seed)
    initial = rng.randint(0, 3)
    ops = [rng.choice([1, -1]) for _ in range(PANEL_COUNT - 1)]
    orientations = [initial]
    current = initial
    for op in ops:
        current = (current + op) % 4
        orientations.append(current)
    return {"initial": initial, "ops": ops, "orientations": orientations}


def _draw_arrow(draw: ImageDraw.ImageDraw, cx: int, cy: int, direction: int) -> None:
    dx, dy = DIR_VECS[direction]
    ax, ay = cx + dx * STEM_LEN, cy + dy * STEM_LEN
    draw.line([cx, cy, ax, ay], fill=INK, width=STEM_W)

    apex_x, apex_y = cx + dx * HEAD_RADIUS, cy + dy * HEAD_RADIUS
    base_mx, base_my = cx + dx * (HEAD_RADIUS - HEAD_W), cy + dy * (HEAD_RADIUS - HEAD_W)
    px, py = -dy, dx
    b1 = (base_mx + px * HEAD_W, base_my + py * HEAD_W)
    b2 = (base_mx - px * HEAD_W, base_my - py * HEAD_W)
    draw.polygon([(apex_x, apex_y), b1, b2], fill=HEAD_COLOR)


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)
    orientations = scene["orientations"]

    for i in range(PANEL_COUNT):
        left = MARGIN_LEFT + PITCH * i
        top = CENTER_Y - HALF_H
        draw.rectangle([left, top, left + PANEL_W, top + PANEL_H], outline=BORDER, width=3)
        cx, cy = _panel_center(i)
        _draw_arrow(draw, cx, cy, orientations[i])

    # Red marker ring around the starting (leftmost) arrow.
    cx, cy = _panel_center(0)
    draw.ellipse(
        [cx - TOKEN_R - 7, cy - TOKEN_R - 7, cx + TOKEN_R + 7, cy + TOKEN_R + 7],
        outline=RING_MARK,
        width=4,
    )

    for i, op in enumerate(scene["ops"]):
        gx, gy = _glyph_center(i)
        if op == 1:
            draw.ellipse(
                [gx - GLYPH_R, gy - GLYPH_R, gx + GLYPH_R, gy + GLYPH_R], fill=CW_COLOR
            )
        else:
            draw.rectangle(
                [gx - GLYPH_R, gy - GLYPH_R, gx + GLYPH_R, gy + GLYPH_R], fill=CCW_COLOR
            )
    return img


def analytic_gold(scene: dict) -> str:
    final = scene["orientations"][-1]
    return DIR_LABELS[final]


def margin(scene: dict) -> float:
    return 10.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    """No nontrivial render-invariant latent symmetry: every latent field
    (``initial``/``orientations``) fixes the drawn arrow in each panel and
    ``ops`` fixes the drawn operator glyphs. Reordering slots would reorder the
    panels and change the pixels, so no reparametrization yields pixel-identical
    renders with a different gold."""
    return []
