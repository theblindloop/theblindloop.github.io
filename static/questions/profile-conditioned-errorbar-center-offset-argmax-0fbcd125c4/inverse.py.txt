"""Independent pixel-only inverse arm for the error-bar center-offset world.

This module receives ONLY the rasterized PNG (plus the world's fixed palette)
and re-measures, for each colored error bar, the absolute vertical offset of the
middle tick from the midpoint of the two outer ticks. It shares no geometry with
renderer.render() and never imports the renderer, prompts, generator, verifier,
or any analytical gold code.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "errorbar_center_offset_argmax-oracle-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
BACKGROUND = (250, 250, 250)
MIN_ROWS = 3


def _nearest_color(arr: np.ndarray) -> np.ndarray:
    """Label every pixel to its nearest palette entry (segment colors + background)."""
    refs = np.stack(
        [np.array(COLOR_RGB[c], dtype=np.float32) for c in PALETTE]
        + [np.array(BACKGROUND, dtype=np.float32)],
        axis=0,
    )
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def _outer_abs_offset(rows: np.ndarray) -> float:
    """Absolute offset of the middle row from the midpoint of the two outer rows.

    A bar contributes exactly three colored tick rows (low, middle, high). After
    sorting them by row, rows[0] and rows[2] are the two outer ticks and rows[1]
    is the middle tick. The measured offset is the middle row's deviation from
    the midpoint of the two outer rows.
    """
    if rows.shape[0] != MIN_ROWS:
        return -1.0
    s = np.sort(rows)
    mid = (s[0] + s[2]) / 2.0
    return float(abs(s[1] - mid))


def decision_from_image(image: Image.Image) -> str:
    """Return the color of the most asymmetric error bar from pixels alone."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _nearest_color(arr)
    best_color = PALETTE[0]
    best_offset = -1.0
    for idx, color in enumerate(PALETTE):
        rows = np.unique(np.nonzero(labels == idx)[0])
        offset = _outer_abs_offset(rows)
        if offset > best_offset:
            best_offset = offset
            best_color = color
    return best_color
