"""Error-bar center-offset selection world: which colored bar is most asymmetric.

Latent scene z carries three colored vertical error bars (one red, one blue,
one green), each drawn as three horizontally-aligned short ticks at three
distinct heights: a LOW tick, a HIGH tick (the two outermost ticks), and a
MIDDLE tick that sits between them. The middle tick of a bar may be displaced
by a signed offset ``s`` from the vertical midpoint of that bar's own low and
high ticks. The decision is a measurement: recover each bar's middle tick row
and the vertical midpoint row of its two outer ticks, take the absolute
deviation between them, and select the color of the most asymmetric bar
(largest absolute center offset). This world counterbalances which color owns
the maximum offset, the absolute vertical position and half-range of every bar,
the displacement direction (up/down), and the horizontal column of every bar.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "errorbar_center_offset_argmax-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
BACKGROUND = (250, 250, 250)

WIDTH = 480
HEIGHT = 340
COL_CENTERS = (80.0, 240.0, 400.0)
TICK_HALF = 11              # half-length (pixels) of each horizontal tick
OFFSET_RANGE = (0, 13)      # integer center-offsets sampled within this range

# Minimum top-two center-offset gap (pixels) required for a decidable selection.
MIN_MARGIN_PX = 4.0
# Below this top-two offset gap the scene is quarantined as a near-boundary case.
QUARANTINE_PX = 2.0


def sample_scene(seed: int) -> dict:
    """Sample a deterministic scene with a guaranteed, counterbalanced extremum.

    ``bars`` is a list of {"color", "col", "y_low", "y_center", "y_high", "offset"}.
    ``y_low < y_center < y_high`` always; ``offset`` is the signed displacement of
    the center tick from the symmetric midpoint ``(y_low+y_high)/2``.
    """
    rng = random.Random(seed * 1009 + 7)
    while True:
        offsets = sorted(
            rng.sample(range(OFFSET_RANGE[0], OFFSET_RANGE[1] + 1), len(PALETTE))
        )
        if offsets[-1] - offsets[-2] >= MIN_MARGIN_PX:
            break
    colors = list(PALETTE)
    rng.shuffle(colors)
    columns = list(range(len(PALETTE)))
    rng.shuffle(columns)
    bars = []
    for color, column, offset in zip(colors, columns, offsets):
        half_range = rng.randint(offset + 6, 24)
        yb = rng.randint(120, 205)                 # symmetric midpoint (integer)
        sign = rng.choice((-1, 1))
        y_low = yb - half_range
        y_high = yb + half_range
        y_center = yb + sign * offset
        bars.append(
            {
                "color": color,
                "col": int(column),
                "y_low": int(y_low),
                "y_center": int(y_center),
                "y_high": int(y_high),
                "offset": int(sign * offset),
            }
        )
    return {"bars": bars, "width": WIDTH, "height": HEIGHT, "background": "light"}


def render(scene: dict) -> Image.Image:
    """Render I = R(z): three colored error bars whose center ticks differ in offset."""
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    for bar in scene["bars"]:
        cx = COL_CENTERS[bar["col"]]
        color = COLOR_RGB[bar["color"]]
        for row in (bar["y_low"], bar["y_center"], bar["y_high"]):
            draw.line(
                [(cx - TICK_HALF, row), (cx + TICK_HALF, row)],
                fill=color,
                width=1,
            )
    return img


def margin(scene: dict) -> float:
    """Verifiability margin: the top-two absolute center-offset gap in pixels."""
    deviations = sorted(abs(bar["y_center"] - (bar["y_low"] + bar["y_high"]) / 2.0)
                        for bar in scene["bars"])
    return float(deviations[-1] - deviations[-2])


def is_quarantined(scene: dict) -> bool:
    """Near-boundary: the top-two center-offset gap is finite-resolution small."""
    return margin(scene) < QUARANTINE_PX


def analytic_gold(scene: dict) -> str:
    """The color of the most asymmetric error bar (largest absolute center offset)."""
    best_color = PALETTE[0]
    best_dev = -1.0
    for bar in scene["bars"]:
        mid = (bar["y_low"] + bar["y_high"]) / 2.0
        dev = abs(bar["y_center"] - mid)
        if dev > best_dev:
            best_dev = dev
            best_color = bar["color"]
    return best_color


def latent_symmetries(scene: dict) -> list:
    """not_applicable: no nontrivial pixel-identical latent symmetry.

    Every latent field of a bar (color, column, and the three tick rows plus the
    offset) is directly rendered as visible ink. Swapping two bars' offsets
    changes which color is correct, and any global flip/reflection turns the
    raster into a distinctly shifted image. There is no hidden field that could
    change the gold without changing the raster.
    """
    return []
