"""Independent pixel-only inverse for normalized interval offsets.

The inverse sees only the PNG. It segments saturated marker ink, locates the
dark horizontal interval and its endpoint caps, and compares the normalized
marker displacement from each recovered interval midpoint. It does not import
renderer, prompt, generator, verifier, or latent-scene code.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


PANELS = ((38, 426), (474, 862))
SCAN_Y = (120, 320)
MIN_MARKER_AREA = 120
MIN_DECISION_MARGIN = 0.02


def _color_mask(rgb: np.ndarray) -> np.ndarray:
    values = rgb.astype(np.float32)
    spread = values.max(axis=2) - values.min(axis=2)
    return spread > 45.0


def _components(mask: np.ndarray) -> list[tuple[int, float, float, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = deque([(int(y), int(x))])
        area = 0
        sum_x = 0.0
        sum_y = 0.0
        min_x = max_x = int(x)
        min_y = max_y = int(y)
        while queue:
            cy, cx = queue.popleft()
            area += 1
            sum_x += cx + 0.5
            sum_y += cy + 0.5
            min_x = min(min_x, cx)
            max_x = max(max_x, cx)
            min_y = min(min_y, cy)
            max_y = max(max_y, cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if area >= MIN_MARKER_AREA:
            found.append((area, sum_x / area, sum_y / area, min_x, min_y, max_x, max_y))
    return found


def _marker_center(rgb: np.ndarray, panel: tuple[int, int]) -> tuple[float, float]:
    left, right = panel
    y0, y1 = SCAN_Y
    components = _components(_color_mask(rgb[y0:y1, left:right]))
    if len(components) != 1:
        raise ValueError(f"expected one saturated marker, found {len(components)}")
    area, cx, cy, _, _, _, _ = components[0]
    if area < MIN_MARKER_AREA:
        raise ValueError("marker component is too small")
    return (left + cx, y0 + cy)


def _interval_endpoints(rgb: np.ndarray, panel: tuple[int, int]) -> tuple[float, float, float]:
    left, right = panel
    y0, y1 = SCAN_Y
    dark = rgb[:, :, :].max(axis=2) < 150.0
    inner = dark[y0:y1, left + 18:right - 18]
    row_counts = inner.sum(axis=1)
    row = int(np.argmax(row_counts)) + y0
    if row_counts.max() < 80:
        raise ValueError("interval line is not visible")
    band = dark[max(y0, row - 3):min(y1, row + 4), left + 18:right - 18]
    ys, xs = np.nonzero(band)
    if len(xs) < 80:
        raise ValueError("interval endpoints are not visible")
    x_values = xs.astype(np.float64) + (left + 18)
    lo = float(x_values.min())
    hi = float(x_values.max())
    if hi - lo < 100.0:
        raise ValueError("recovered interval is too short")
    return lo, hi, float(row)


def _panel_offset(rgb: np.ndarray, panel: tuple[int, int]) -> float:
    marker_x, marker_y = _marker_center(rgb, panel)
    interval_x0, interval_x1, line_y = _interval_endpoints(rgb, panel)
    if abs(marker_y - line_y) > 28.0:
        raise ValueError("marker is not on the interval")
    return abs(marker_x - (interval_x0 + interval_x1) / 2.0) / (interval_x1 - interval_x0)


def decision_from_image(image: Image.Image) -> str:
    """Return A or B from the two recovered normalized offsets."""

    rgb = np.asarray(image.convert("RGB"))
    offset_a = _panel_offset(rgb, PANELS[0])
    offset_b = _panel_offset(rgb, PANELS[1])
    if abs(offset_a - offset_b) < MIN_DECISION_MARGIN:
        raise ValueError("normalized offsets are too close to classify")
    return "A" if offset_a > offset_b else "B"
