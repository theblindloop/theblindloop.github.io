"""Deterministic renderer for normalized interval midpoint offsets."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont


WIDTH = 900
HEIGHT = 440
SUPERSAMPLE = 4
PANEL_BOUNDS = ((38, 426), (474, 862))
MARKER_PALETTE = ("#d1495b", "#0077b6", "#2a9d8f", "#e09f3e", "#7b2cbf", "#f15bb5")
LINE_COLOR = "#394452"
FRAME_COLOR = "#c7cdd4"
PANEL_FILL = "#fbfcfe"


def _font(size: int):
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size * SUPERSAMPLE)
    except OSError:
        return ImageFont.load_default()


def _offset(x0: float, x1: float, marker: float) -> float:
    return abs(marker - (x0 + x1) / 2.0) / (x1 - x0)


def _sample_panel(rng: random.Random, bounds: tuple[int, int], offset: float, side: int) -> dict:
    left, right = bounds
    width = rng.uniform(210.0, 278.0)
    x0 = rng.uniform(left + 48.0, right - 48.0 - width)
    x1 = x0 + width
    y = rng.uniform(175.0, 255.0)
    radius = rng.uniform(12.0, 19.0)
    marker = (x0 + x1) / 2.0 + side * offset * width
    return {
        "interval_x0": round(x0, 6),
        "interval_x1": round(x1, 6),
        "marker_x": round(marker, 6),
        "interval_y": round(y, 6),
        "marker_radius": round(radius, 6),
        "marker_color": rng.choice(MARKER_PALETTE),
    }


def sample_scene(seed: int) -> dict:
    """Sample balanced scenes, including a visible near-equality band."""

    rng = random.Random(int(seed))
    low = rng.uniform(0.10, 0.24)
    if rng.random() < 0.24:
        difference = rng.uniform(0.045, 0.070)
    else:
        difference = rng.uniform(0.10, 0.34)
    high = low + difference
    smaller_panel = rng.randrange(2)
    offsets = (low, high) if smaller_panel == 0 else (high, low)
    panel_a = _sample_panel(rng, PANEL_BOUNDS[0], offsets[0], rng.choice((-1, 1)))
    panel_b = _sample_panel(rng, PANEL_BOUNDS[1], offsets[1], rng.choice((-1, 1)))
    return {
        "a_interval_x0": panel_a["interval_x0"],
        "a_interval_x1": panel_a["interval_x1"],
        "a_marker_x": panel_a["marker_x"],
        "a_interval_y": panel_a["interval_y"],
        "a_marker_radius": panel_a["marker_radius"],
        "a_marker_color": panel_a["marker_color"],
        "b_interval_x0": panel_b["interval_x0"],
        "b_interval_x1": panel_b["interval_x1"],
        "b_marker_x": panel_b["marker_x"],
        "b_interval_y": panel_b["interval_y"],
        "b_marker_radius": panel_b["marker_radius"],
        "b_marker_color": panel_b["marker_color"],
    }


def render(scene: dict) -> Image.Image:
    """Render two labeled panels with a line interval and an interior marker."""

    a_x0 = float(scene["a_interval_x0"])
    a_x1 = float(scene["a_interval_x1"])
    a_marker = float(scene["a_marker_x"])
    a_y = float(scene["a_interval_y"])
    a_radius = float(scene["a_marker_radius"])
    a_color = str(scene["a_marker_color"])
    b_x0 = float(scene["b_interval_x0"])
    b_x1 = float(scene["b_interval_x1"])
    b_marker = float(scene["b_marker_x"])
    b_y = float(scene["b_interval_y"])
    b_radius = float(scene["b_marker_radius"])
    b_color = str(scene["b_marker_color"])

    scale = SUPERSAMPLE
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), "white")
    draw = ImageDraw.Draw(image)
    label_font = _font(25)
    for (left, right), label in zip(PANEL_BOUNDS, ("A", "B")):
        draw.rounded_rectangle(
            (left * scale, 34 * scale, right * scale, 406 * scale),
            radius=12 * scale,
            fill=PANEL_FILL,
            outline=FRAME_COLOR,
            width=2 * scale,
        )
        box = draw.textbbox((0, 0), label, font=label_font)
        text_width = box[2] - box[0]
        draw.text((((left + right) * scale - text_width) / 2.0, 52 * scale), label, fill="#27313b", font=label_font)

    line_width = 5 * scale
    cap_width = 4 * scale
    for x0, x1, y, marker, radius, color in (
        (a_x0, a_x1, a_y, a_marker, a_radius, a_color),
        (b_x0, b_x1, b_y, b_marker, b_radius, b_color),
    ):
        draw.line((round(x0 * scale), round(y * scale), round(x1 * scale), round(y * scale)), fill=LINE_COLOR, width=line_width)
        draw.line((round(x0 * scale), round((y - 18.0) * scale), round(x0 * scale), round((y + 18.0) * scale)), fill=LINE_COLOR, width=cap_width)
        draw.line((round(x1 * scale), round((y - 18.0) * scale), round(x1 * scale), round((y + 18.0) * scale)), fill=LINE_COLOR, width=cap_width)
        draw.ellipse(
            (round((marker - radius) * scale), round((y - radius) * scale),
             round((marker + radius) * scale), round((y + radius) * scale)),
            fill=color,
        )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """Select the panel with the larger normalized midpoint offset."""

    a_x0 = float(scene["a_interval_x0"])
    a_x1 = float(scene["a_interval_x1"])
    a_marker = float(scene["a_marker_x"])
    b_x0 = float(scene["b_interval_x0"])
    b_x1 = float(scene["b_interval_x1"])
    b_marker = float(scene["b_marker_x"])
    offset_a = _offset(a_x0, a_x1, a_marker)
    offset_b = _offset(b_x0, b_x1, b_marker)
    return "A" if offset_a > offset_b else "B"


def margin(scene: dict) -> float:
    """Positive margin means panel A has the larger normalized offset."""

    a_x0 = float(scene["a_interval_x0"])
    a_x1 = float(scene["a_interval_x1"])
    a_marker = float(scene["a_marker_x"])
    b_x0 = float(scene["b_interval_x0"])
    b_x1 = float(scene["b_interval_x1"])
    b_marker = float(scene["b_marker_x"])
    return _offset(a_x0, a_x1, a_marker) - _offset(b_x0, b_x1, b_marker)


def is_quarantined(scene: dict) -> bool:
    return abs(margin(scene)) < 0.08


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No render-invariant latent alias is claimed for this visible encoding."""

    return []
