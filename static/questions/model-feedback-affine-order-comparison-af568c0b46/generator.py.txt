"""Deterministic renderer for inferred affine-machine order comparison."""

MACHINE_COLORS = {

    "orange": ((665, CAL_Y[0]), (955, CAL_Y[0]), (665, CAL_Y[1]), (955, CAL_Y[1])),
}

SEGMENTS = {
    0: "abcedf",
    1: "bc",
    2: "abged",
    3: "abgcd",
    4: "fgbc",
    5: "afgcd",
    6: "afgecd",
    7: "abc",
    8: "abcdefg",
    9: "abfgcd",
}


def _int(value: object, label: str, lo: int = 0, hi: int = 99) -> int:
    if isinstance(value, bool) or not isinstance(value, int) or not lo <= value <= hi:
        raise ValueError(f"{label} must be an integer in [{lo}, {hi}]")
    return int(value)


def _examples(value: object, label: str) -> tuple[tuple[int, int], tuple[int, int]]:
    if not isinstance(value, (list, tuple)) or len(value) != 2:
        raise ValueError(f"{label} must contain two examples")
    result = []
    for index, pair in enumerate(value):
        if not isinstance(pair, (list, tuple)) or len(pair) != 2:
            raise ValueError(f"{label} example {index + 1} must be [input, output]")
        result.append((_int(pair[0], "input"), _int(pair[1], "output")))
    return result[0], result[1]


def _infer_rule(examples: tuple[tuple[int, int], tuple[int, int]]) -> tuple[int, int]:
    (x1, y1), (x2, y2) = examples
    if x1 == x2 or (y2 - y1) % (x2 - x1):
        raise ValueError("examples do not determine an integer multiplier")
    multiplier = (y2 - y1) // (x2 - x1)
    offset = y1 - multiplier * x1
    if not 2 <= multiplier <= 4 or not -6 <= offset <= 6 or offset == 0:
        raise ValueError("affine rule is outside the declared range")
    if multiplier * x2 + offset != y2:
        raise ValueError("examples are not affine")
    return multiplier, offset


def _rules(scene: dict) -> dict[str, tuple[int, int]]:
    machines = scene.get("machines")
    if not isinstance(machines, dict) or set(machines) != set(MACHINE_NAMES):
        raise ValueError("machines must contain blue and orange")
    result = {}
    for name in MACHINE_NAMES:
        spec = machines[name]
        if not isinstance(spec, dict) or set(spec) != {"examples"}:
            raise ValueError(f"{name} machine must contain only examples")
        result[name] = _infer_rule(_examples(spec["examples"], name))
    return result


def _routes(scene: dict) -> tuple[tuple[str, str], tuple[str, str]]:
    routes = []
    for key, value in (
        ("left_route", scene["left_route"]),
        ("right_route", scene["right_route"]),
    ):
        if not isinstance(value, (list, tuple)) or len(value) != 2 or set(value) != set(MACHINE_NAMES):
            raise ValueError(f"{key} must use each machine exactly once")
        routes.append((str(value[0]), str(value[1])))
    if routes[1] != tuple(reversed(routes[0])):
        raise ValueError("routes must use opposite orders")
    return routes[0], routes[1]


def _run(start: int, route: tuple[str, str], rules: dict[str, tuple[int, int]]) -> tuple[int, int]:
    first_a, first_b = rules[route[0]]
    middle = first_a * start + first_b
    second_a, second_b = rules[route[1]]
    final = second_a * middle + second_b
    if not 1 <= middle <= 99 or not 1 <= final <= 99:
        raise ValueError("route leaves the supported positive range")
    return middle, final


def _gap(scene: dict) -> int:
    rules = _rules(scene)
    left, right = _routes(scene)
    start = _int(scene["start"], "start", 1, 99)
    return _run(start, left, rules)[1] - _run(start, right, rules)[1]


@lru_cache(maxsize=12)
def _parameter_pool(target_gap: int, left_first: str, start: int) -> tuple[tuple[int, int, int, int], ...]:
    left = (left_first, MACHINE_NAMES[1 - MACHINE_NAMES.index(left_first)])
    right = tuple(reversed(left))
    pool = []
    for blue_a in range(2, 5):
        for blue_b in range(-6, 7):
            if blue_b == 0:
                continue
            for orange_a in range(2, 5):
                for orange_b in range(-6, 7):
                    if orange_b == 0 or (blue_a, blue_b) == (orange_a, orange_b):
                        continue
                    rules = {"blue": (blue_a, blue_b), "orange": (orange_a, orange_b)}
                    try:
                        gap = _run(start, left, rules)[1] - _run(start, right, rules)[1]
                    except ValueError:
                        continue
                    if gap == target_gap:
                        pool.append((blue_a, blue_b, orange_a, orange_b))
    return tuple(pool)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) * 15485863 + 32452843)
    decision = ("left", "right", "same")[int(seed) % 3]
    target_gap = {"left": 1, "right": -1, "same": 0}[decision]
    start = rng.randint(4, 8)
    left_first = MACHINE_NAMES[rng.randrange(2)]
    pool = _parameter_pool(target_gap, left_first, start)
    if not pool:
        raise RuntimeError("no affine parameter tuple for requested class")
    blue_a, blue_b, orange_a, orange_b = pool[rng.randrange(len(pool))]
    params = {"blue": (blue_a, blue_b), "orange": (orange_a, orange_b)}
    machines = {}
    for name in MACHINE_NAMES:
        a, b = params[name]
        valid_x = [x for x in range(3, 10) if 1 <= a * x + b <= 39]
        x1, x2 = sorted(rng.sample(valid_x, 2))
        examples = [[x1, a * x1 + b], [x2, a * x2 + b]]
        if rng.randrange(2):
            examples.reverse()
        machines[name] = {"examples": examples}
    other = MACHINE_NAMES[1 - MACHINE_NAMES.index(left_first)]
    scene = {
        "machines": machines,
        "start": start,
        "left_route": [left_first, other],
        "right_route": [other, left_first],
    }
    if _gap(scene) != target_gap:
        raise RuntimeError("sampler class construction failed")
    return scene


def analytic_gold(scene: dict) -> str:
    gap = _gap(scene)
    return "left" if gap > 0 else "right" if gap < 0 else "same"


def margin(scene: dict) -> float:
    return float(max(1, abs(_gap(scene))))


def is_quarantined(scene: dict) -> bool:
    try:
        return abs(_gap(scene)) > 1
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int, *, bold: bool = False) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(_BOLD_PATH if bold else _FONT_PATH, size=size)


def _centered(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font: ImageFont.ImageFont, fill: tuple[int, int, int]) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2 - box[1]), text, font=font, fill=fill)


def _draw_arrow(draw: ImageDraw.ImageDraw, x0: int, x1: int, y: int) -> None:
    draw.line((x0, y, x1 - 10, y), fill=INK, width=5)
    draw.polygon(((x1 - 12, y - 9), (x1, y), (x1 - 12, y + 9)), fill=INK)


def _draw_digit(draw: ImageDraw.ImageDraw, cx: int, cy: int, digit: int) -> None:
    centers = {
        "a": (cx, cy - 18, False), "g": (cx, cy, False), "d": (cx, cy + 18, False),
        "f": (cx - 12, cy - 9, True), "b": (cx + 12, cy - 9, True),
        "e": (cx - 12, cy + 9, True), "c": (cx + 12, cy + 9, True),
    }
    active = set(SEGMENTS[digit])
    for name, (x, y, vertical) in centers.items():
        if name not in active:
            continue
        box = (x - 3, y - 8, x + 3, y + 8) if vertical else (x - 10, y - 3, x + 10, y + 3)
        draw.rounded_rectangle(box, radius=2, fill=SEGMENT_ON)


def _draw_number(draw: ImageDraw.ImageDraw, cx: int, cy: int, value: int) -> None:
    value = _int(value, "display value")
    draw.rounded_rectangle((cx - 39, cy - 31, cx + 39, cy + 31), radius=8, fill=DISPLAY_BG, outline=(112, 121, 128), width=2)
    tens, ones = divmod(value, 10)
    _draw_digit(draw, cx - 18, cy, tens)
    _draw_digit(draw, cx + 18, cy, ones)


def _draw_machine(draw: ImageDraw.ImageDraw, cx: int, cy: int, name: str, compact: bool = False) -> None:
    width = 126 if compact else 112
    height = 56 if compact else 62
    draw.rounded_rectangle((cx - width // 2, cy - height // 2, cx + width // 2, cy + height // 2), radius=12, fill=MACHINE_COLORS[name], outline=INK, width=3)
    _centered(draw, (cx, cy), name.upper(), _font(17 if compact else 16, bold=True), (255, 255, 255))


def render(scene: dict) -> Image.Image:
    rules = _rules(scene)
    left_route, right_route = _routes(scene)
    start = _int(scene["start"], "start", 1, 99)
    _run(start, left_route, rules)
    _run(start, right_route, rules)

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((18, 16, WIDTH - 18, HEIGHT - 16), radius=20, fill=PAPER, outline=(193, 201, 204), width=3)
    _centered(draw, (WIDTH / 2, 43), "LEARN EACH MACHINE", _font(29, bold=True), INK)
    _centered(draw, (WIDTH / 2, 75), "multiply, then add the fixed offset", _font(19, bold=True), MUTED)

    for name, bounds, machine_x in (
        ("blue", (40, 105, 535, 405), 290),
        ("orange", (565, 105, 1060, 405), 810),
    ):
        x0, y0, x1, y1 = bounds
        draw.rounded_rectangle(bounds, radius=16, fill=(248, 249, 247), outline=MACHINE_COLORS[name], width=4)
        draw.rounded_rectangle((x0 + 4, y0 + 4, x1 - 4, y0 + 58), radius=11, fill=MACHINE_COLORS[name])
        _centered(draw, ((x0 + x1) / 2, y0 + 31), f"{name.upper()} EXAMPLES", _font(22, bold=True), (255, 255, 255))
        examples = _examples(scene["machines"][name]["examples"], name)
        for row_index, ((inp, out), y) in enumerate(zip(examples, CAL_Y, strict=True)):
            in_x, out_x = (145, 435) if name == "blue" else (665, 955)
            _draw_number(draw, in_x, y, inp)
            _draw_arrow(draw, in_x + 48, machine_x - 65, y)
            _draw_machine(draw, machine_x, y, name)
            _draw_arrow(draw, machine_x + 65, out_x - 48, y)
            _draw_number(draw, out_x, y, out)

    draw.rounded_rectangle((38, 430, 1062, 700), radius=17, fill=(239, 249, 247), outline=QUERY_COLOR, width=4)
    _centered(draw, (WIDTH / 2, 457), "QUESTION — RUN LEFT TO RIGHT", _font(23, bold=True), QUERY_COLOR)
    for label, route, y in (("LEFT ROUTE", left_route, QUERY_Y[0]), ("RIGHT ROUTE", right_route, QUERY_Y[1])):
        draw.rounded_rectangle((65, y - 42, 1035, y + 42), radius=12, fill=(255, 255, 255), outline=(176, 190, 189), width=2)
        _centered(draw, (103, y), label, _font(16, bold=True), MUTED)
        _draw_number(draw, QUERY_START_X, y, start)
        _draw_arrow(draw, 265, 360, y)
        _draw_machine(draw, QUERY_BADGE_X[0], y, route[0], compact=True)
        _draw_arrow(draw, 505, 560, y)
        _draw_machine(draw, QUERY_BADGE_X[1], y, route[1], compact=True)
        _draw_arrow(draw, 705, 820, y)
        _centered(draw, (875, y), "?", _font(48, bold=True), INK)
        _centered(draw, (955, y), "FINAL", _font(17, bold=True), MUTED)
    return image
