"""Deterministic renderer for a compact connected-trace stencil world."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 256
HEIGHT = 256
BACKGROUND = (248, 247, 241)
NAVY = (28, 57, 126)
RED = (211, 58, 57)
CYAN_STYLES = ((47, 173, 184), (40, 158, 194))
FRAME_STYLES = ((196, 201, 207), (177, 188, 199))

WAVE = (1, 1, 0, 0, 1, 1, 0, 0)
FORK = (1, 0, 0, 1, 1, 0, 0, 0)
TEMPLATES = {"A": WAVE, "B": FORK}
DIFF_POSITIONS = tuple(i for i, (a, b) in enumerate(zip(WAVE, FORK)) if a != b)

# Every route is a single non-self-intersecting polyline.  Its x ordering is
# deliberately only an implementation nuisance: the red cap and the trace
# remain the visible source of traversal order.
ROUTES = (
    ((6, 16), (14, 15), (22, 15), (30, 16), (38, 16), (46, 16), (54, 15), (62, 14)),
    ((6, 14), (14, 14), (22, 15), (30, 16), (38, 16), (46, 15), (54, 14), (62, 15)),
    ((6, 16), (14, 16), (22, 15), (30, 15), (38, 16), (46, 16), (54, 15), (62, 14)),
    ((6, 14), (14, 15), (22, 16), (30, 15), (38, 14), (46, 14), (54, 15), (62, 16)),
)
LOCAL_WIDTH = 68
LOCAL_HEIGHT = 34


def _hamming(code: Iterable[int], template: Iterable[int]) -> int:
    return sum(int(a) != int(b) for a, b in zip(code, template))


def _classify_code(code: Iterable[int]) -> str:
    values = tuple(int(value) for value in code)
    distances = {label: _hamming(values, template) for label, template in TEMPLATES.items()}
    return "A" if distances["A"] < distances["B"] else "B"


def sample_scene(seed: int) -> dict:
    """Sample one balanced, locally framed scene from an integer seed."""

    rng = random.Random(int(seed))
    label = "A" if int(seed) % 2 == 0 else "B"
    code = list(TEMPLATES[label])
    # A small, visible near-boundary subpopulation mutates one disagreement
    # position.  The odd template distance keeps the analytic decision unique;
    # its margin is one and is explicitly quarantined.
    near_boundary = int(seed) % 7 == 0
    if near_boundary:
        index = DIFF_POSITIONS[int(seed) % len(DIFF_POSITIONS)]
        code[index] = 1 - code[index]
    return {
        "code": code,
        "route_variant": rng.randrange(len(ROUTES)),
        # Keep the small inset randomized in absolute position while giving
        # the ablation grid a stable two-row/three-column local footprint.
        "offset_x": rng.choice((42, 74, 106, 138, 170)),
        "offset_y": rng.choice((64, 96, 128, 160, 192)),
        "mirror": bool(rng.randrange(2)),
        "frame_variant": rng.randrange(len(FRAME_STYLES)),
        "station_style": rng.randrange(len(CYAN_STYLES)),
    }


def _points(scene: dict) -> list[tuple[float, float]]:
    base = ROUTES[int(scene["route_variant"]) % len(ROUTES)]
    mirror = bool(scene["mirror"])
    ox = float(scene["offset_x"])
    oy = float(scene["offset_y"])
    return [
        (ox + (LOCAL_WIDTH - x if mirror else x), oy + y)
        for x, y in base
    ]


def _normal(points: list[tuple[float, float]], index: int) -> tuple[float, float]:
    if index == 0:
        dx = points[1][0] - points[0][0]
        dy = points[1][1] - points[0][1]
    elif index == len(points) - 1:
        dx = points[-1][0] - points[-2][0]
        dy = points[-1][1] - points[-2][1]
    else:
        dx = points[index + 1][0] - points[index - 1][0]
        dy = points[index + 1][1] - points[index - 1][1]
    length = math.hypot(dx, dy)
    return (-dy / length, dx / length)


def render(scene: dict) -> Image.Image:
    """Render one RGB image; the trace and all code marks are local."""

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    points = _points(scene)
    ox, oy = int(scene["offset_x"]), int(scene["offset_y"])
    frame = FRAME_STYLES[int(scene["frame_variant"]) % len(FRAME_STYLES)]
    cyan = CYAN_STYLES[int(scene["station_style"]) % len(CYAN_STYLES)]
    draw.rounded_rectangle(
        (ox, oy, ox + LOCAL_WIDTH, oy + LOCAL_HEIGHT),
        radius=7,
        outline=frame,
        width=2,
        fill=(242, 242, 239),
    )
    draw.line(points, fill=NAVY, width=3, joint="curve")
    code = tuple(int(value) for value in scene["code"])
    for index, (point, bit) in enumerate(zip(points, code)):
        nx, ny = _normal(points, index)
        sign = 1.0 if bit else -1.0
        end = (point[0] + sign * 5.0 * nx, point[1] + sign * 5.0 * ny)
        draw.line((point, end), fill=NAVY, width=3)
    for index, (x, y) in enumerate(points):
        radius = 3.5 if index == 0 else 2.2
        color = RED if index == 0 else cyan
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)
    return image


def analytic_gold(scene: dict) -> str:
    code = tuple(int(value) for value in scene["code"])
    return _classify_code(code)


def margin(scene: dict) -> float:
    code = tuple(int(value) for value in scene["code"])
    distance_a = _hamming(code, WAVE)
    distance_b = _hamming(code, FORK)
    return float(abs(distance_a - distance_b))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
