"""Deterministic renderer for visible-legend sequence decoding."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 720, 560
SHAPES = ("circle", "square", "triangle", "diamond")
PAYLOADS = ("cyan", "orange", "purple", "green")
TAGS = ("red", "blue", "gold")
RGB = {
    "cyan": (25, 166, 190),
    "orange": (232, 126, 35),
    "purple": (133, 76, 180),
    "green": (51, 153, 85),
    "red": (210, 54, 62),
    "blue": (48, 100, 205),
    "gold": (218, 166, 32),
}
LEGEND_BG = (255, 244, 218)
MESSAGE_BG = (224, 244, 255)
OPTION_BG = (242, 242, 242)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    colors = list(PAYLOADS)
    rng.shuffle(colors)
    legend = {shape: color for shape, color in zip(SHAPES, colors, strict=True)}
    message = list(SHAPES)
    message.append(rng.choice(SHAPES))
    rng.shuffle(message)
    decoded = [legend[s] for s in message]

    distractors: list[list[str]] = []
    pairs = [(i, j) for i in range(5) for j in range(i + 1, 5) if decoded[i] != decoded[j]]
    rng.shuffle(pairs)
    for i, j in pairs:
        candidate = decoded.copy()
        candidate[i], candidate[j] = candidate[j], candidate[i]
        if candidate not in distractors:
            distractors.append(candidate)
        if len(distractors) == 2:
            break
    options = [decoded.copy(), *distractors]
    rng.shuffle(options)
    tags = list(TAGS)
    # Rotation guarantees all three answer identities over any three consecutive seeds.
    rotation = seed % 3
    tags = tags[rotation:] + tags[:rotation]
    if seed % 2:
        tags[0], tags[2] = tags[2], tags[0]
    roles = {"legend_side": "left" if seed % 2 == 0 else "right"}
    return {"legend": legend, "message": message, "options": options, "tags": tags, "roles": roles}


def _shape(draw: ImageDraw.ImageDraw, kind: str, cx: int, cy: int, size: int = 16) -> None:
    fill = (28, 31, 36)
    if kind == "circle":
        draw.ellipse((cx - size, cy - size, cx + size, cy + size), fill=fill)
    elif kind == "square":
        draw.rectangle((cx - size + 1, cy - size + 1, cx + size - 1, cy + size - 1), fill=fill)
    elif kind == "triangle":
        draw.polygon([(cx, cy - size), (cx - size, cy + size), (cx + size, cy + size)], fill=fill)
    elif kind == "diamond":
        draw.polygon([(cx, cy - size), (cx + size, cy), (cx, cy + size), (cx - size, cy)], fill=fill)
    else:
        raise ValueError(kind)


def _disk(draw: ImageDraw.ImageDraw, color: str, cx: int, cy: int, r: int = 14) -> None:
    draw.ellipse((cx - r, cy - r, cx + r, cy + r), fill=RGB[color], outline=(255, 255, 255), width=2)


def render(scene: dict) -> Image.Image:
    legend = scene["legend"]
    message = scene["message"]
    options = scene["options"]
    tags = scene["tags"]
    roles = scene["roles"]
    image = Image.new("RGB", (WIDTH, HEIGHT), (255, 255, 255))
    draw = ImageDraw.Draw(image)
    left = (35, 30, 335, 285)
    right = (385, 30, 685, 285)
    legend_box, message_box = (left, right) if roles["legend_side"] == "left" else (right, left)
    draw.rounded_rectangle(legend_box, radius=18, fill=LEGEND_BG, outline=(85, 85, 85), width=4)
    draw.rounded_rectangle(message_box, radius=18, fill=MESSAGE_BG, outline=(85, 85, 85), width=4)

    lx0, ly0, lx1, ly1 = legend_box
    for row, shape in enumerate(SHAPES):
        y = ly0 + 42 + row * 55
        _shape(draw, shape, lx0 + 95, y)
        draw.line((lx0 + 135, y, lx0 + 185, y), fill=(100, 100, 100), width=4)
        _disk(draw, legend[shape], lx0 + 220, y)

    mx0, my0, mx1, my1 = message_box
    for col, shape in enumerate(message):
        _shape(draw, shape, mx0 + 45 + col * 52, my0 + 128, 15)

    option_order = [(25, 325, 230, 525), (257, 325, 462, 525), (489, 325, 694, 525)]
    for box, sequence, tag in zip(option_order, options, tags, strict=True):
        draw.rounded_rectangle(box, radius=16, fill=OPTION_BG, outline=(105, 105, 105), width=4)
        x0, y0, x1, y1 = box
        for col, color in enumerate(sequence):
            _disk(draw, color, x0 + 31 + col * 36, y0 + 68, 12)
        # A detached tag is visibly bound to its option by the short stem.
        draw.line(((x0 + x1) // 2, y0 + 112, (x0 + x1) // 2, y0 + 145), fill=(90, 90, 90), width=5)
        _disk(draw, tag, (x0 + x1) // 2, y0 + 163, 17)
    return image


def _decoded(scene: dict) -> list[str]:
    return [scene["legend"][shape] for shape in scene["message"]]


def _distances(scene: dict) -> list[int]:
    target = _decoded(scene)
    return [sum(a != b for a, b in zip(target, option, strict=True)) for option in scene["options"]]


def analytic_gold(scene: dict) -> str:
    distances = _distances(scene)
    winners = [i for i, value in enumerate(distances) if value == 0]
    if len(winners) != 1:
        raise ValueError("scene does not have a unique exact match")
    return scene["tags"][winners[0]]


def margin(scene: dict) -> float:
    distances = sorted(_distances(scene))
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    distances = _distances(scene)
    return distances.count(0) != 1 or sorted(distances)[1] < 2


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
