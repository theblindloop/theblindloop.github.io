"""Renderer, sampler and analytic gold for egocentric_turn_walk.

A small robot stands on a 6 by 6 board of cells.  It is drawn exactly once, as
a black triangle whose apex points the way the robot faces.  Under the board is
an ordered row of cream instruction cards.  A card carrying a grey up-arrow and
k black dots drives the robot k cells straight ahead in the direction it is
facing at that moment; a card carrying a green bent arrow turns it a quarter
turn to its own left and a card carrying an orange bent arrow turns it a
quarter turn to its own right, both without moving it.

The turns are egocentric: a turn card changes the meaning of every later drive
card, so the effect of a card cannot be read off the card alone.  No later
position or facing of the robot is drawn anywhere.  Five coloured pads sit on
five cells; the robot always finishes on a pad, and the answer is that pad's
colour together with the direction the robot then faces.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "egocentric_turn_walk-renderer-0.1.0"

# ---------------------------------------------------------------- geometry
CANVAS = (700, 620)
BOARD_X0, BOARD_Y0 = 40, 40
CELL = 62
GRID_N = 6
CARD_Y0, CARD_Y1 = 462, 546
CARD_W = 84
CARD_GAP = 18
CARD_X0 = 40

BG = (255, 255, 255)
BOARD_BG = (232, 232, 232)
LINE = (120, 120, 120)
ROBOT = (18, 18, 18)
CARD_BG = (246, 238, 213)
DRIVE = (90, 90, 90)
DOT = (18, 18, 18)
LEFT_C = (26, 140, 60)
RIGHT_C = (232, 124, 20)

PAD_COLOURS = {
    "red": (206, 44, 52),
    "blue": (30, 86, 200),
    "purple": (128, 52, 168),
    "teal": (0, 140, 140),
    "yellow": (226, 178, 20),
}
PAD_NAMES = ("blue", "purple", "red", "teal", "yellow")
HEADINGS = ("up", "right", "down", "left")
_STEP = {"up": (0, -1), "right": (1, 0), "down": (0, 1), "left": (-1, 0)}


# ---------------------------------------------------------------- replay
def _turn(heading: str, side: str) -> str:
    i = HEADINGS.index(heading)
    return HEADINGS[(i + (1 if side == "R" else -1)) % 4]


def _ordered_cards(scene: dict) -> list[dict]:
    return sorted(scene["cards"], key=lambda c: c["slot"])


def _ordered_pads(scene: dict) -> list[dict]:
    return sorted(scene["pads"], key=lambda p: (p["cell"][1], p["cell"][0]))


def _replay(scene: dict, swap: bool = False, ignore_turns: bool = False):
    """Run the drawn program.  Returns (cell, heading, stayed_on_board)."""
    cx, cy = scene["robot"]["cell"]
    heading = scene["robot"]["heading"]
    for card in _ordered_cards(scene):
        kind = card["kind"]
        if kind in ("L", "R"):
            if ignore_turns:
                continue
            side = kind
            if swap:
                side = "L" if kind == "R" else "R"
            heading = _turn(heading, side)
        else:
            dx, dy = _STEP[heading]
            for _ in range(card["steps"]):
                cx, cy = cx + dx, cy + dy
                if not (0 <= cx < GRID_N and 0 <= cy < GRID_N):
                    return (cx, cy), heading, False
    return (cx, cy), heading, True


def _pad_at(scene: dict, cell) -> str | None:
    for pad in scene["pads"]:
        if tuple(pad["cell"]) == tuple(cell):
            return pad["colour"]
    return None


def analytic_gold(scene: dict) -> str:
    cell, heading, ok = _replay(scene)
    if not ok:
        return "abstain"
    colour = _pad_at(scene, cell)
    if colour is None:
        return "abstain"
    return f"{colour}:{heading}"


# ---------------------------------------------------------------- margin
def margin(scene: dict) -> float:
    """Least drawn reading clearance, in pixels.

    Three drawn readings decide the answer: which cell the robot's triangle
    sits in, which way that triangle points, and how many dots a drive card
    carries.  Each has its own drawn clearance to the boundary at which the
    reading would flip, and the margin is the smallest of them.
    """
    r = scene["robot"]
    cell_clear = CELL / 2.0 - 22.0 - max(abs(r["jx"]), abs(r["jy"]))
    head_clear = (45.0 - abs(r["jr"])) / 45.0 * 20.0
    dot_clear = 40.0
    for card in scene["cards"]:
        if card["kind"] == "F" and card["steps"] > 1:
            dot_clear = min(dot_clear, float(card["gap"]) - 12.0)
    return round(min(cell_clear, head_clear, dot_clear), 6)


QUARANTINE_THRESHOLD = 4.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_THRESHOLD or analytic_gold(scene) == "abstain"


# ---------------------------------------------------------------- sampling
def _cells():
    return [(x, y) for y in range(GRID_N) for x in range(GRID_N)]


def _draw_program(rng: random.Random) -> list[dict]:
    n = rng.randint(4, 6)
    out: list[dict] = []
    for slot in range(n):
        if rng.random() < 0.45:
            out.append({"slot": slot, "kind": rng.choice(("L", "R")), "steps": 0, "gap": 20})
        else:
            out.append({"slot": slot, "kind": "F", "steps": rng.randint(1, 3),
                        "gap": rng.choice((20, 21, 22))})
    return out


def _candidate_scene(rng: random.Random, tight: bool) -> dict:
    cells = _cells()
    rng.shuffle(cells)
    jr_max = 34.0 if tight else 16.0
    jxy_max = 8.0 if tight else 5.0
    return {
        "canvas": list(CANVAS),
        "grid_n": GRID_N,
        "robot": {
            "cell": list(cells[5]),
            "heading": rng.choice(HEADINGS),
            "jx": round(rng.uniform(-jxy_max, jxy_max), 3),
            "jy": round(rng.uniform(-jxy_max, jxy_max), 3),
            "jr": round(rng.uniform(-jr_max, jr_max), 3),
        },
        "pads": [
            {"colour": name, "cell": list(cell)}
            for name, cell in zip(rng.sample(PAD_NAMES, 5), cells[:5])
        ],
        "cards": _draw_program(rng),
        "tight": tight,
    }


def _controls_pass(scene: dict) -> bool:
    """Reject scenes whose answer is reachable by a cheap wrong rule."""
    cell, heading, ok = _replay(scene)
    if not ok:
        return False
    colour = _pad_at(scene, cell)
    if colour is None:
        return False
    if tuple(cell) == tuple(scene["robot"]["cell"]):
        return False
    if not any(c["kind"] in ("L", "R") for c in scene["cards"]):
        return False
    if not any(c["kind"] == "F" for c in scene["cards"]):
        return False
    icell, _ih, iok = _replay(scene, ignore_turns=True)
    if iok and _pad_at(scene, icell) == colour:
        return False
    scell, sheading, sok = _replay(scene, swap=True)
    if sok and _pad_at(scene, scell) == colour and sheading == heading:
        return False
    sx, sy = scene["robot"]["cell"]
    nearest = min(scene["pads"],
                  key=lambda p: (p["cell"][0] - sx) ** 2 + (p["cell"][1] - sy) ** 2)
    if nearest["colour"] == colour:
        return False
    if heading == scene["robot"]["heading"]:
        return False
    return True


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    # cycle colour with stride 1 and facing with stride 3 so that consecutive
    # scene seeds balance both halves of the answer quickly
    combos = [(PAD_NAMES[i % 5], HEADINGS[(i * 3) % 4]) for i in range(20)]
    want = combos[seed % len(combos)]
    tight = (seed % 8) == 3
    fallback = None
    for _ in range(8000):
        scene = _candidate_scene(rng, tight)
        if not _controls_pass(scene):
            continue
        if fallback is None:
            fallback = scene
        colour, heading = analytic_gold(scene).split(":")
        if (colour, heading) == want:
            return scene
    if fallback is not None:
        return fallback
    raise RuntimeError("sampling failed")


def latent_symmetries(scene: dict) -> list:
    """Pixel-identical relabelings of the latent bags."""
    out = []
    cards = list(scene["cards"])
    rev = dict(scene)
    rev["cards"] = list(reversed(cards))
    out.append(("cards_bag_reversed", rev))
    rot = dict(scene)
    rot["cards"] = cards[1:] + cards[:1]
    out.append(("cards_bag_rotated", rot))
    pads = dict(scene)
    pads["pads"] = list(reversed(scene["pads"]))
    out.append(("pads_bag_reversed", pads))
    return out


# ---------------------------------------------------------------- drawing
def _cell_centre(cx: int, cy: int):
    return (BOARD_X0 + (cx + 0.5) * CELL, BOARD_Y0 + (cy + 0.5) * CELL)


def _rot(px: float, py: float, ox: float, oy: float, deg: float):
    a = math.radians(deg)
    dx, dy = px - ox, py - oy
    return (ox + dx * math.cos(a) - dy * math.sin(a), oy + dx * math.sin(a) + dy * math.cos(a))


def _arrowhead(d: ImageDraw.ImageDraw, tip, deg: float, colour, size: float = 15.0) -> None:
    pts = [(tip[0], tip[1]), (tip[0] - size * 0.7, tip[1] + size),
           (tip[0] + size * 0.7, tip[1] + size)]
    pts = [_rot(px, py, tip[0], tip[1], deg) for px, py in pts]
    d.polygon(pts, fill=colour)


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", tuple(scene["canvas"]), BG)
    d = ImageDraw.Draw(img)

    x1 = BOARD_X0 + GRID_N * CELL
    y1 = BOARD_Y0 + GRID_N * CELL
    d.rectangle([BOARD_X0, BOARD_Y0, x1, y1], fill=BOARD_BG)
    for pad in _ordered_pads(scene):
        px, py = pad["cell"]
        cx0 = BOARD_X0 + px * CELL + 9
        cy0 = BOARD_Y0 + py * CELL + 9
        d.rectangle([cx0, cy0, cx0 + CELL - 18, cy0 + CELL - 18],
                    fill=PAD_COLOURS[pad["colour"]])
    for i in range(GRID_N + 1):
        d.line([BOARD_X0 + i * CELL, BOARD_Y0, BOARD_X0 + i * CELL, y1], fill=LINE, width=2)
        d.line([BOARD_X0, BOARD_Y0 + i * CELL, x1, BOARD_Y0 + i * CELL], fill=LINE, width=2)

    r = scene["robot"]
    ox, oy = _cell_centre(*r["cell"])
    ox += r["jx"]
    oy += r["jy"]
    rad = 22.0
    base = [(ox, oy - rad), (ox - 0.80 * rad, oy + 0.62 * rad), (ox + 0.80 * rad, oy + 0.62 * rad)]
    deg = {"up": 0.0, "right": 90.0, "down": 180.0, "left": 270.0}[r["heading"]] + r["jr"]
    d.polygon([_rot(px, py, ox, oy, deg) for px, py in base], fill=ROBOT)

    for i, card in enumerate(_ordered_cards(scene)):
        cx0 = CARD_X0 + i * (CARD_W + CARD_GAP)
        cx1 = cx0 + CARD_W
        d.rectangle([cx0, CARD_Y0, cx1, CARD_Y1], fill=CARD_BG, outline=LINE, width=2)
        mx = (cx0 + cx1) / 2.0
        if card["kind"] == "F":
            d.line([mx, CARD_Y0 + 48, mx, CARD_Y0 + 22], fill=DRIVE, width=8)
            _arrowhead(d, (mx, CARD_Y0 + 12), 0.0, DRIVE, 14.0)
            k = card["steps"]
            gap = card["gap"]
            for j in range(k):
                dx = mx + (j - (k - 1) / 2.0) * gap
                d.ellipse([dx - 6, CARD_Y0 + 60, dx + 6, CARD_Y0 + 72], fill=DOT)
        else:
            colour = LEFT_C if card["kind"] == "L" else RIGHT_C
            sign = -1 if card["kind"] == "L" else 1
            my = CARD_Y0 + CARD_W / 2
            d.line([mx - sign * 20, my + 22, mx + sign * 18, my + 22], fill=colour, width=7)
            d.line([mx + sign * 18, my + 22, mx + sign * 18, my - 8], fill=colour, width=7)
            _arrowhead(d, (mx + sign * 18, my - 22), 0.0, colour, 15.0)

    n = len(scene["cards"])
    ax1 = CARD_X0 + (n - 1) * (CARD_W + CARD_GAP) + CARD_W
    d.line([CARD_X0, 578, ax1 - 14, 578], fill=LINE, width=5)
    _arrowhead(d, (ax1, 578), 90.0, LINE, 14.0)
    return img
