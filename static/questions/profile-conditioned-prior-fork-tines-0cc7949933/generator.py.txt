"""Fork-tine count world under a contradictory canonical prior.

Profile: paper-semantic@0.3.0:prior_conflict. Report visible truth (the actual
number of tines) when a familiar canonical prior ("a table fork typically has
four tines") is contradicted by the actual image, which in the counterfactual
scenes shows a fork with a clearly different number of tines drawn.

Latent z = {n, tine_w, tine_gap, tine_h, neck_h, handle_w, handle_h, cx, cy,
width, height}. ``n`` is the number of tines actually drawn on the fork head;
it is the only field that influences the analytic gold, and it is directly and
uniquely readable in the raster as the number of separated vertical prongs, so
two different values of ``n`` can never produce the same image. The scene is
the fork rendered on a plain white canvas; there is no pixel-identical latent
symmetry to exploit (a fork has no nontrivial rotational/mirror symmetry that
leaves the raster unchanged while hiding a label input), so the latent-alias
mode is ``not_applicable`` with a declared injectivity justification.

The decision boundary is the instant at which one tine (a vertical prong) is
added or removed from the fork head, directly visible as a discrete change in
the number of separated prongs. The closed-form gold is

    decision = str(n)

The pixel-only inverse arm segments the fork-prong colour and counts, over the
most fragmented horizontal band of the fork (the tine zone), the number of
contiguous prong runs in each row and takes the maximum, which equals n.

The continuous verifiability margin is the minimum of the prong width and the
slot gap between adjacent prongs in pixels. When that clearance is too small
the prongs visually merge into a solid mass at pixel scale, so such scenes are
quarantined.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "prior_fork_tines-0.1.0"

WIDTH = 512
HEIGHT = 512
BG = (255, 255, 255)
FORK = (66, 70, 82)      # the only dark, counted pixels (prongs + neck + handle)

CX = WIDTH // 2

TINE_W = 16
TINE_GAP = 14
TINE_H = 100
NECK_H = 16
HANDLE_W = 46
HANDLE_H = 140
Y0 = 140

SUPPORTED_COUNTS = (3, 4, 5, 6, 7, 8, 9)
QUAR_MIN_CLEARANCE = 8.0


def head_width(n: int) -> int:
    return n * TINE_W + (n - 1) * TINE_GAP


def margin(scene: dict) -> float:
    """Minimum prong width / slot gap (px); the count robustness."""
    return float(min(scene["tine_w"], scene["tine_gap"]))


def is_quarantined(scene: dict) -> bool:
    n = int(scene["n"])
    if n not in SUPPORTED_COUNTS:
        return True
    if margin(scene) < QUAR_MIN_CLEARANCE:
        return True
    if head_width(n) > WIDTH - 64:
        return True
    return False


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    while True:
        n = rng.choice(SUPPORTED_COUNTS)
        scene = {
            "n": n,
            "tine_w": TINE_W,
            "tine_gap": TINE_GAP,
            "tine_h": TINE_H,
            "neck_h": NECK_H,
            "handle_w": HANDLE_W,
            "handle_h": HANDLE_H,
            "cx": CX,
            "cy": HEIGHT // 2,
            "width": WIDTH,
            "height": HEIGHT,
        }
        if not is_quarantined(scene):
            return scene


def render(scene: dict) -> Image.Image:
    w, h = int(scene["width"]), int(scene["height"])
    n = int(scene["n"])
    cx = int(scene["cx"])
    tine_w = int(scene["tine_w"])
    gap = int(scene["tine_gap"])
    tine_h = int(scene["tine_h"])
    neck_h = int(scene["neck_h"])
    handle_w = int(scene["handle_w"])
    handle_h = int(scene["handle_h"])

    img = Image.new("RGB", (w, h), BG)
    draw = ImageDraw.Draw(img)

    top = Y0
    tine_bottom = top + tine_h
    neck_y1 = tine_bottom + neck_h

    hw = head_width(n)
    left = cx - hw // 2

    # Handle (lightest region, drawn first).
    hx0 = cx - handle_w // 2
    draw.rectangle([hx0, neck_y1, hx0 + handle_w, neck_y1 + handle_h], fill=FORK)

    # Neck (solid prong base).
    draw.rectangle([left, tine_bottom - 2, left + hw, neck_y1], fill=FORK)

    # Tines (n separated vertical prongs) on top.
    for k in range(n):
        x0 = left + k * (tine_w + gap)
        draw.rectangle([x0, top, x0 + tine_w, tine_bottom], fill=FORK)

    return img


def analytic_gold(scene: dict) -> str:
    return str(scene["n"])


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No nontrivial pixel-identical latent symmetry exists (see candidate.json)."""
    return []
