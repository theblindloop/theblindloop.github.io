"""Independent pixel-only inverse arm for the fork-tine world.

Receives ONLY the final PNG and returns the number of fork tines drawn as a
string. It never imports renderer, prompts, generate, verify, or analytic-gold
code, and accepts no latent scene data.

Method: the fork is the only dark object on the white canvas. The oracle
segments the fork-prong colour into a boolean mask. The fork head has ``n``
separated vertical prongs (tines) above a solid neck, so the most fragmented
horizontal band of the mask (the tine zone) contains exactly ``n`` contiguous
prong runs per row; the neck and handle are each a single contiguous run. Over
all rows, the maximum number of contiguous runs in the mask equals the number
of tines ``n``. This maximum is reported.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "prior_fork_tines-oracle-0.1.0"

_FORK = np.array((66, 70, 82), dtype=np.float32)
_COLOR_DIST = 40.0


def _row_runs(row: np.ndarray) -> int:
    """Count contiguous True-groups in a 1D boolean row (assumes >= 1 True)."""
    padded = np.concatenate([[False], row, [False]])
    return int(np.sum(padded[1:] & ~padded[:-1]))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    mask = np.linalg.norm(arr - _FORK, axis=-1) < _COLOR_DIST
    best = 0
    for row in mask:
        if not row.any():
            continue
        runs = _row_runs(row)
        if runs > best:
            best = runs
    return str(best)


def measure(image: Image.Image) -> dict:
    """Diagnostic measurement (not used for the decision)."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    mask = np.linalg.norm(arr - _FORK, axis=-1) < _COLOR_DIST
    ys, xs = np.nonzero(mask)
    cy = float(np.mean(ys)) if len(ys) else 0.0
    cx = float(np.mean(xs)) if len(xs) else 0.0
    return {
        "fork_px": int(mask.sum()),
        "centre": (round(cx, 1), round(cy, 1)),
        "size": image.size,
    }
