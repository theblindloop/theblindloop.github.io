"""Renderer and analytic arm for the three-lobe phase world."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

CANVAS = 512
BACKGROUND = (248, 246, 240)
PALETTE = ((28, 105, 196), (0, 139, 125), (177, 48, 139), (218, 96, 18))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    alternating = bool(rng.randrange(2))
    first = rng.choice((-1, 1))
    if alternating:
        signs = [first, -first, first]
    else:
        signs = rng.choice(([first, first, -first], [first, -first, -first]))
    # Cell-centred positions make the short support move while keeping each
    # trace well inside the canvas and away from accidental ablation seams.
    cx = rng.choice((96, 160, 224, 288, 352, 416)) + rng.randint(-3, 3)
    cy = rng.choice((96, 160, 224, 288, 352, 416)) + rng.randint(-3, 3)
    decorations = []
    for _ in range(rng.randint(5, 8)):
        while True:
            x, y = rng.randint(30, 482), rng.randint(30, 482)
            if abs(x - cx) > 95 or abs(y - cy) > 75:
                break
        decorations.append([x, y, rng.randint(5, 12), rng.choice((0, 1))])
    return {
        "lobe_signs": list(signs),
        "cx": cx,
        "cy": cy,
        "orientation": rng.choice((0, 90)),
        "color": list(rng.choice(PALETTE)),
        "lobe_height": rng.randint(17, 21),
        "half_width": rng.randint(44, 48),
        "stroke_width": rng.randint(6, 8),
        "decorations": decorations,
        "nuisance_token": rng.randrange(1_000_000),
    }


def _path_points(scene: dict, scale: int = 1) -> list[tuple[int, int]]:
    cx, cy = float(scene["cx"]), float(scene["cy"])
    half = float(scene["half_width"])
    height = float(scene["lobe_height"])
    signs = [int(value) for value in scene["lobe_signs"]]
    xs = (-half, -2 * half / 3, -half / 3, 0.0, half / 3, 2 * half / 3, half)
    ys = (0.0, signs[0] * height, 0.0, signs[1] * height, 0.0, signs[2] * height, 0.0)
    angle = math.radians(float(scene["orientation"]))
    cosine, sine = math.cos(angle), math.sin(angle)
    points = []
    for x, y in zip(xs, ys, strict=True):
        rx = x * cosine - y * sine
        ry = x * sine + y * cosine
        points.append((round((cx + rx) * scale), round((cy + ry) * scale)))
    return points


def render(scene: dict) -> Image.Image:
    scale = 3
    image = Image.new("RGB", (CANVAS * scale, CANVAS * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x, y, radius, kind in scene["decorations"]:
        box = tuple(int((value) * scale) for value in (x - radius, y - radius, x + radius, y + radius))
        shade = (225, 223, 216)
        if kind:
            draw.ellipse(box, outline=shade, width=2 * scale)
        else:
            draw.line(((x - radius) * scale, y * scale, (x + radius) * scale, y * scale), fill=shade, width=2 * scale)
    points = _path_points(scene, scale)
    color = tuple(int(value) for value in scene["color"])
    width = int(scene["stroke_width"]) * scale
    draw.line(points, fill=color, width=width, joint="curve")
    radius = width // 2
    for x, y in points:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)
    return image.resize((CANVAS, CANVAS), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    signs = [int(value) for value in scene["lobe_signs"]]
    return "alternating" if signs[0] == signs[2] and signs[0] != signs[1] else "repeat"


def margin(scene: dict) -> float:
    # Pixel clearance of every peak from the visible end-to-end baseline.
    return float(scene["lobe_height"] - scene["stroke_width"] / 2.0)


def is_quarantined(scene: dict) -> bool:
    return bool(
        scene["lobe_height"] < 15
        or scene["half_width"] < 40
        or scene["stroke_width"] < 5
        or len(scene["lobe_signs"]) != 3
        or any(value not in (-1, 1) for value in scene["lobe_signs"])
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["nuisance_token"] = (int(twin["nuisance_token"]) + 1) % 1_000_000
    return [("nuisance_token_shift", twin)]
