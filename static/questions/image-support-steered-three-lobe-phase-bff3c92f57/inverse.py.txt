"""Independent pixel-only inverse for the three-lobe trail."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups = []
    for sy, sx in np.argwhere(mask):
        if seen[sy, sx]:
            continue
        queue = deque([(int(sy), int(sx))])
        seen[sy, sx] = True
        points = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    ny, nx = y + dy, x + dx
                    if (
                        (dy or dx)
                        and 0 <= ny < height
                        and 0 <= nx < width
                        and mask[ny, nx]
                        and not seen[ny, nx]
                    ):
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        groups.append(np.asarray(points, dtype=np.int32))
    return sorted(groups, key=len, reverse=True)


def _window_median(u: np.ndarray, v: np.ndarray, target: float, span: float) -> float:
    selected = np.abs(u - target) <= span
    if int(selected.sum()) < 18:
        raise ValueError("trail window is missing")
    return float(np.median(v[selected]))


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    spread = array.max(axis=2) - array.min(axis=2)
    mask = (spread > 62) & (array.min(axis=2) < 205) & (array.max(axis=2) > 85)
    groups = _components(mask)
    if not groups or len(groups[0]) < 420:
        raise ValueError("colored trail absent or too incomplete")
    if len(groups) > 1 and len(groups[1]) > 35:
        raise ValueError("colored trail is disconnected")
    points = groups[0]
    ys, xs = points[:, 0].astype(float), points[:, 1].astype(float)
    xspan, yspan = float(xs.max() - xs.min()), float(ys.max() - ys.min())
    if max(xspan, yspan) < 72 or min(xspan, yspan) < 24:
        raise ValueError("trail extent is incomplete")
    if max(xspan, yspan) / max(1.0, min(xspan, yspan)) < 1.75:
        raise ValueError("trail axis is ambiguous")
    if xspan >= yspan:
        u, v = xs, ys
    else:
        u, v = ys, xs
    lo, hi = float(u.min()), float(u.max())
    length = hi - lo
    if length <= 0:
        raise ValueError("degenerate trail")
    un = (u - lo) / length
    left = _window_median(un, v, 0.035, 0.04)
    right = _window_median(un, v, 0.965, 0.04)
    values = []
    for fraction in (1 / 6, 1 / 2, 5 / 6):
        observed = _window_median(un, v, fraction, 0.055)
        baseline = left + fraction * (right - left)
        offset = observed - baseline
        if abs(offset) < 7.0:
            raise ValueError("lobe side is ambiguous")
        values.append(1 if offset > 0 else -1)
    if values[0] == values[2] and values[0] != values[1]:
        return "alternating"
    if values[0] != values[2] and (values[0] == values[1] or values[1] == values[2]):
        return "repeat"
    raise ValueError("invalid three-lobe phase")
