"""Independent image-only inverse for the dual-key cross-index worksheet."""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH = 660
HEIGHT = 500
GRID_X = 335
GRID_Y = 100
CELL = 70
ROW_KEY_Y = (82, 123, 164, 205)
COLUMN_KEY_Y = (300, 341, 382, 423)
BOARD_CENTERS_X = (370, 440, 510, 580)
BOARD_CENTERS_Y = (135, 205, 275, 345)

ROW_SYMBOLS = {
    "111100100",
    "111010010",
    "101010101",
    "110010011",
}
COLUMN_STAMPS = {
    "111101111",
    "010111010",
    "100010001",
    "001011111",
}

BACKGROUND = (248, 247, 242)
STAMP_BG = (255, 255, 255)
INK = (29, 37, 51)
STAMP_INK = (27, 74, 105)
GUIDE = (89, 101, 116)
ROW_RGB = {
    "red": (214, 68, 62),
    "blue": (48, 105, 196),
    "green": (42, 149, 91),
    "orange": (226, 126, 31),
}
DISK_RGB = {
    "magenta": (187, 52, 139),
    "teal": (22, 151, 158),
    "gold": (220, 174, 34),
    "purple": (118, 72, 184),
}
ANSWER_RGB = {
    "amber": (243, 164, 33),
    "cyan": (36, 181, 202),
    "violet": (143, 88, 203),
    "lime": (119, 184, 59),
}


class OracleError(ValueError):
    """Raised when the raster does not encode one intact cross-index worksheet."""


def _rgb(array: np.ndarray, color: tuple[int, int, int]) -> np.ndarray:
    return np.all(array == np.asarray(color, dtype=np.uint8), axis=2)


def _require_uniform(
    array: np.ndarray,
    left: int,
    top: int,
    right: int,
    bottom: int,
    color: tuple[int, int, int],
    what: str,
) -> None:
    patch = array[top:bottom, left:right]
    if patch.size == 0 or not np.all(patch == np.asarray(color, dtype=np.uint8)):
        raise OracleError(f"{what} is missing or damaged")


def _identify_uniform(
    array: np.ndarray,
    box: tuple[int, int, int, int],
    palette: dict[str, tuple[int, int, int]],
    what: str,
) -> str:
    left, top, right, bottom = box
    patch = array[top:bottom, left:right]
    matches = [
        name
        for name, color in palette.items()
        if np.all(patch == np.asarray(color, dtype=np.uint8))
    ]
    if len(matches) != 1:
        raise OracleError(f"{what} has no unique declared color")
    return matches[0]


def _read_stamp(
    array: np.ndarray,
    center_x: int,
    center_y: int,
    ink: tuple[int, int, int],
    alphabet: set[str],
    what: str,
) -> str:
    # The visible frame is checked independently from the nine content cells.
    _require_uniform(array, center_x - 14, center_y - 18, center_x + 15, center_y - 16, GUIDE, f"{what} top frame")
    _require_uniform(array, center_x - 14, center_y + 17, center_x + 15, center_y + 19, GUIDE, f"{what} bottom frame")
    values: list[str] = []
    start_x = center_x - 12
    start_y = center_y - 12
    ink_array = np.asarray(ink, dtype=np.uint8)
    white_array = np.asarray(STAMP_BG, dtype=np.uint8)
    for row in range(3):
        for column in range(3):
            left = start_x + column * 9
            top = start_y + row * 9
            patch = array[top : top + 7, left : left + 7]
            if np.all(patch == ink_array):
                values.append("1")
            elif np.all(patch == white_array):
                values.append("0")
            else:
                raise OracleError(f"{what} contains an ambiguous stamp cell")
    bits = "".join(values)
    if bits not in alphabet:
        raise OracleError(f"{what} is not a member of its visible token alphabet")
    return bits


def _validate_structure(array: np.ndarray) -> None:
    for x, y in ((0, 0), (WIDTH - 1, 0), (0, HEIGHT - 1), (WIDTH - 1, HEIGHT - 1)):
        if tuple(int(value) for value in array[y, x]) != BACKGROUND:
            raise OracleError("canvas corners do not contain the declared background")
    # Every grid cell has a four-pixel dark frame. Sampling the top and left
    # edges catches crops, erased divisions, and shifted boards.
    for row in range(4):
        for column in range(4):
            left = GRID_X + column * CELL
            top = GRID_Y + row * CELL
            _require_uniform(array, left, top, left + CELL, top + 4, INK, "grid horizontal frame")
            _require_uniform(array, left, top, left + 4, top + CELL, INK, "grid vertical frame")


def decision_from_image(image: Image.Image) -> str:
    """Recover both bindings and the intersected cell from final pixels only."""

    if image.size != (WIDTH, HEIGHT):
        raise OracleError(f"expected a {WIDTH} by {HEIGHT} raster")
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.shape != (HEIGHT, WIDTH, 3):
        raise OracleError("expected one RGB worksheet")
    _validate_structure(array)

    row_pairs: list[tuple[str, str]] = []
    for index, center_y in enumerate(ROW_KEY_Y):
        symbol = _read_stamp(array, 58, center_y, INK, ROW_SYMBOLS, f"row-key symbol {index}")
        tag = _identify_uniform(array, (174, center_y - 8, 209, center_y + 9), ROW_RGB, f"row-key tag {index}")
        row_pairs.append((symbol, tag))
    if {symbol for symbol, _ in row_pairs} != ROW_SYMBOLS or {tag for _, tag in row_pairs} != set(ROW_RGB):
        raise OracleError("row key is not a complete bijection")

    column_pairs: list[tuple[str, str]] = []
    for index, center_y in enumerate(COLUMN_KEY_Y):
        disk = _identify_uniform(array, (54, center_y - 5, 63, center_y + 5), DISK_RGB, f"column-key disk {index}")
        stamp = _read_stamp(array, 190, center_y, STAMP_INK, COLUMN_STAMPS, f"column-key stamp {index}")
        column_pairs.append((disk, stamp))
    if {disk for disk, _ in column_pairs} != set(DISK_RGB) or {stamp for _, stamp in column_pairs} != COLUMN_STAMPS:
        raise OracleError("column key is not a complete bijection")

    board_rows = [
        _identify_uniform(array, (298, center_y - 12, 317, center_y + 13), ROW_RGB, f"board row {index}")
        for index, center_y in enumerate(BOARD_CENTERS_Y)
    ]
    if set(board_rows) != set(ROW_RGB):
        raise OracleError("board row headers are not a permutation")
    board_columns = [
        _read_stamp(array, center_x, 70, STAMP_INK, COLUMN_STAMPS, f"board column {index}")
        for index, center_x in enumerate(BOARD_CENTERS_X)
    ]
    if set(board_columns) != COLUMN_STAMPS:
        raise OracleError("board column headers are not a permutation")

    cells: list[list[str]] = []
    for row, center_y in enumerate(BOARD_CENTERS_Y):
        cells.append(
            [
                _identify_uniform(
                    array,
                    (center_x - 18, center_y - 18, center_x + 19, center_y + 19),
                    ANSWER_RGB,
                    f"grid cell {row},{column}",
                )
                for column, center_x in enumerate(BOARD_CENTERS_X)
            ]
        )
    if any(set(row) != set(ANSWER_RGB) for row in cells):
        raise OracleError("each board row must visibly contain all four answer colors")

    query_symbol = _read_stamp(array, 410, 447, INK, ROW_SYMBOLS, "row query")
    query_disk = _identify_uniform(array, (568, 441, 581, 454), DISK_RGB, "column query")

    row_lookup = {symbol: tag for symbol, tag in row_pairs}
    column_lookup = {disk: stamp for disk, stamp in column_pairs}
    row_tag = row_lookup.get(query_symbol)
    column_stamp = column_lookup.get(query_disk)
    if row_tag is None or column_stamp is None:
        raise OracleError("a query token has no unique key partner")
    row = board_rows.index(row_tag)
    column = board_columns.index(column_stamp)
    return cells[row][column]
