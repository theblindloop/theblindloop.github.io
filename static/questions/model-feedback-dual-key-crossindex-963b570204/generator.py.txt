"""Deterministic renderer for the dual-key cross-index world."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 660
HEIGHT = 500
GRID_X = 335
GRID_Y = 100
CELL = 70

ROW_KEY_Y = (82, 123, 164, 205)
COLUMN_KEY_Y = (300, 341, 382, 423)
BOARD_CENTERS_X = tuple(GRID_X + CELL * column + CELL // 2 for column in range(4))
BOARD_CENTERS_Y = tuple(GRID_Y + CELL * row + CELL // 2 for row in range(4))

ROW_SYMBOLS = (
    "111100100",  # corner
    "111010010",  # tee
    "101010101",  # cross
    "110010011",  # zig
)
COLUMN_STAMPS = (
    "111101111",  # ring
    "010111010",  # plus
    "100010001",  # falling diagonal
    "001011111",  # wedge
)
ROW_TAGS = ("red", "blue", "green", "orange")
DISK_TAGS = ("magenta", "teal", "gold", "purple")
ANSWERS = ("amber", "cyan", "violet", "lime")

RGB = {
    "background": (248, 247, 242),
    "panel": (232, 236, 240),
    "panel_edge": (124, 137, 151),
    "stamp_bg": (255, 255, 255),
    "ink": (29, 37, 51),
    "stamp_ink": (27, 74, 105),
    "guide": (89, 101, 116),
    "red": (214, 68, 62),
    "blue": (48, 105, 196),
    "green": (42, 149, 91),
    "orange": (226, 126, 31),
    "magenta": (187, 52, 139),
    "teal": (22, 151, 158),
    "gold": (220, 174, 34),
    "purple": (118, 72, 184),
    "amber": (243, 164, 33),
    "cyan": (36, 181, 202),
    "violet": (143, 88, 203),
    "lime": (119, 184, 59),
}


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size=size)
    except OSError:
        return ImageFont.load_default()


def _shuffled(rng: random.Random, values: tuple[str, ...]) -> list[str]:
    result = list(values)
    rng.shuffle(result)
    return result


def sample_scene(seed: int) -> dict:
    """Sample balanced mappings, a balanced board, and two visible queries."""

    seed = int(seed)
    rng = random.Random((seed ^ 0xD041C0DE) & 0xFFFFFFFFFFFFFFFF)

    row_partner_tags = _shuffled(rng, ROW_TAGS)
    row_pairs = [[ROW_SYMBOLS[index], row_partner_tags[index]] for index in range(4)]
    rng.shuffle(row_pairs)

    column_partner_stamps = _shuffled(rng, COLUMN_STAMPS)
    column_pairs = [[DISK_TAGS[index], column_partner_stamps[index]] for index in range(4)]
    rng.shuffle(column_pairs)

    board_rows = _shuffled(rng, ROW_TAGS)
    board_columns = _shuffled(rng, COLUMN_STAMPS)
    target_row = (seed // 4) % 4
    target_column = (seed // 16 + seed // 3) % 4

    row_lookup = {tag: symbol for symbol, tag in row_pairs}
    column_lookup = {stamp: disk for disk, stamp in column_pairs}
    query_symbol = row_lookup[board_rows[target_row]]
    query_disk = column_lookup[board_columns[target_column]]

    palette = _shuffled(rng, ANSWERS)
    desired = ANSWERS[seed % 4]
    desired_index = palette.index(desired)
    offset = (desired_index - target_row - target_column) % 4
    cells = [
        [palette[(row + column + offset) % 4] for column in range(4)]
        for row in range(4)
    ]

    return {
        "row_pairs": row_pairs,
        "column_pairs": column_pairs,
        "board_rows": board_rows,
        "board_columns": board_columns,
        "cells": cells,
        "query_symbol": query_symbol,
        "query_disk": query_disk,
    }


def _draw_stamp(
    draw: ImageDraw.ImageDraw,
    center_x: int,
    center_y: int,
    bits: str,
    ink: tuple[int, int, int],
) -> None:
    draw.rectangle(
        (center_x - 18, center_y - 18, center_x + 18, center_y + 18),
        fill=RGB["stamp_bg"],
        outline=RGB["guide"],
        width=2,
    )
    start_x = center_x - 12
    start_y = center_y - 12
    for row in range(3):
        for column in range(3):
            if bits[row * 3 + column] == "1":
                left = start_x + column * 9
                top = start_y + row * 9
                draw.rectangle((left, top, left + 6, top + 6), fill=ink)


def _draw_arrow(draw: ImageDraw.ImageDraw, x1: int, y: int, x2: int) -> None:
    draw.line((x1, y, x2, y), fill=RGB["guide"], width=3)
    draw.polygon(((x2, y), (x2 - 8, y - 5), (x2 - 8, y + 5)), fill=RGB["guide"])


def _validate_scene(scene: dict) -> None:
    if set(scene) != {
        "row_pairs",
        "column_pairs",
        "board_rows",
        "board_columns",
        "cells",
        "query_symbol",
        "query_disk",
    }:
        raise ValueError("scene fields do not match the world schema")
    row_pairs = scene["row_pairs"]
    column_pairs = scene["column_pairs"]
    if not isinstance(row_pairs, list) or len(row_pairs) != 4:
        raise ValueError("row key must contain four pairs")
    if not isinstance(column_pairs, list) or len(column_pairs) != 4:
        raise ValueError("column key must contain four pairs")
    if {str(pair[0]) for pair in row_pairs} != set(ROW_SYMBOLS):
        raise ValueError("row-key symbols must form a bijection")
    if {str(pair[1]) for pair in row_pairs} != set(ROW_TAGS):
        raise ValueError("row-key tags must form a bijection")
    if {str(pair[0]) for pair in column_pairs} != set(DISK_TAGS):
        raise ValueError("column-key disks must form a bijection")
    if {str(pair[1]) for pair in column_pairs} != set(COLUMN_STAMPS):
        raise ValueError("column-key stamps must form a bijection")
    if list(sorted(str(value) for value in scene["board_rows"])) != sorted(ROW_TAGS):
        raise ValueError("board row tags must be a permutation")
    if list(sorted(str(value) for value in scene["board_columns"])) != sorted(COLUMN_STAMPS):
        raise ValueError("board column stamps must be a permutation")
    cells = scene["cells"]
    if not isinstance(cells, list) or len(cells) != 4:
        raise ValueError("board must have four rows")
    if any(not isinstance(row, list) or len(row) != 4 for row in cells):
        raise ValueError("board rows must have four cells")
    if any(str(value) not in ANSWERS for row in cells for value in row):
        raise ValueError("board contains an unknown answer color")
    if str(scene["query_symbol"]) not in ROW_SYMBOLS:
        raise ValueError("unknown row query symbol")
    if str(scene["query_disk"]) not in DISK_TAGS:
        raise ValueError("unknown column query disk")


def render(scene: dict) -> Image.Image:
    """Render a large, fixed-layout worksheet with no answer-only metadata."""

    _validate_scene(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), RGB["background"])
    draw = ImageDraw.Draw(image)
    font_small = _font(14, bold=True)
    font_title = _font(18, bold=True)

    draw.rounded_rectangle((16, 28, 250, 232), radius=10, fill=RGB["panel"], outline=RGB["panel_edge"], width=2)
    draw.text((30, 37), "ROW KEY", fill=RGB["ink"], font=font_title)
    for center_y, pair in zip(ROW_KEY_Y, scene["row_pairs"], strict=True):
        symbol, tag = str(pair[0]), str(pair[1])
        _draw_stamp(draw, 58, center_y, symbol, RGB["ink"])
        _draw_arrow(draw, 83, center_y, 155)
        draw.rectangle((170, center_y - 12, 212, center_y + 12), fill=RGB[tag], outline=RGB["ink"], width=2)

    draw.rounded_rectangle((16, 246, 250, 450), radius=10, fill=RGB["panel"], outline=RGB["panel_edge"], width=2)
    draw.text((30, 255), "COLUMN KEY", fill=RGB["ink"], font=font_title)
    for center_y, pair in zip(COLUMN_KEY_Y, scene["column_pairs"], strict=True):
        disk, stamp = str(pair[0]), str(pair[1])
        draw.ellipse((46, center_y - 12, 70, center_y + 12), fill=RGB[disk], outline=RGB["ink"], width=2)
        _draw_arrow(draw, 83, center_y, 155)
        _draw_stamp(draw, 190, center_y, stamp, RGB["stamp_ink"])

    draw.text((335, 22), "CROSS-INDEX GRID", fill=RGB["ink"], font=font_title)
    for column, stamp in enumerate(scene["board_columns"]):
        _draw_stamp(draw, BOARD_CENTERS_X[column], 70, str(stamp), RGB["stamp_ink"])
    for row, tag in enumerate(scene["board_rows"]):
        center_y = BOARD_CENTERS_Y[row]
        draw.rectangle((292, center_y - 18, 322, center_y + 18), fill=RGB[str(tag)], outline=RGB["ink"], width=2)

    for row in range(4):
        for column in range(4):
            left = GRID_X + column * CELL
            top = GRID_Y + row * CELL
            draw.rectangle((left, top, left + CELL - 1, top + CELL - 1), fill=RGB["ink"])
            draw.rectangle((left + 4, top + 4, left + CELL - 5, top + CELL - 5), fill=RGB[str(scene["cells"][row][column])])

    draw.rounded_rectangle((270, 408, 640, 482), radius=10, fill=RGB["panel"], outline=RGB["panel_edge"], width=2)
    draw.text((283, 418), "QUERY", fill=RGB["ink"], font=font_title)
    draw.text((337, 455), "row", fill=RGB["ink"], font=font_small)
    _draw_stamp(draw, 410, 447, str(scene["query_symbol"]), RGB["ink"])
    draw.text((480, 455), "column", fill=RGB["ink"], font=font_small)
    draw.ellipse((558, 431, 590, 463), fill=RGB[str(scene["query_disk"])], outline=RGB["ink"], width=2)
    return image


def analytic_gold(scene: dict) -> str:
    """Resolve both keys, bind their partners to axes, then read the cell."""

    _validate_scene(scene)
    row_map = {str(symbol): str(tag) for symbol, tag in scene["row_pairs"]}
    column_map = {str(disk): str(stamp) for disk, stamp in scene["column_pairs"]}
    row_tag = row_map[str(scene["query_symbol"])]
    column_stamp = column_map[str(scene["query_disk"])]
    row = [str(value) for value in scene["board_rows"]].index(row_tag)
    column = [str(value) for value in scene["board_columns"]].index(column_stamp)
    return str(scene["cells"][row][column])


def _hamming(left: str, right: str) -> int:
    return sum(a != b for a, b in zip(left, right, strict=True))


def margin(scene: dict) -> float:
    """Smallest visible separation between any two members of either query alphabet."""

    _validate_scene(scene)
    symbol_gap = min(_hamming(a, b) for index, a in enumerate(ROW_SYMBOLS) for b in ROW_SYMBOLS[index + 1 :])
    stamp_gap = min(_hamming(a, b) for index, a in enumerate(COLUMN_STAMPS) for b in COLUMN_STAMPS[index + 1 :])
    disk_gap = min(
        math.dist(RGB[a], RGB[b]) / 32.0
        for index, a in enumerate(DISK_TAGS)
        for b in DISK_TAGS[index + 1 :]
    )
    return float(min(symbol_gap, stamp_gap, disk_gap))


def is_quarantined(scene: dict) -> bool:
    try:
        _validate_scene(scene)
        analytic_gold(scene)
        return margin(scene) < 2.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The explicit ordered worksheet has no intended pixel-identical aliases."""

    return []
