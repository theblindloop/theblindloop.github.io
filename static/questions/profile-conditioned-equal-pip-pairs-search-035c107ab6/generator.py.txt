"""Deterministic renderer for equal-pip paired-compartment search.

The analytic arm sees only the latent scene.  The separate oracle recovers the
answer from the final PNG and shares no code with this module.
"""

from __future__ import annotations

import copy
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 760, 560
LABELS = ("A", "B", "C", "D")
GRID_CELLS = 12

_PALETTES = (
    {
        "background": [246, 248, 244],
        "panel": [255, 255, 252],
        "outer": [139, 148, 155],
        "outline": [35, 137, 151],
        "pip": [24, 48, 82],
        "accent": [225, 112, 54],
        "label_text": [255, 255, 255],
    },
    {
        "background": [250, 247, 242],
        "panel": [255, 254, 250],
        "outer": [151, 143, 137],
        "outline": [35, 137, 151],
        "pip": [24, 48, 82],
        "accent": [205, 94, 65],
        "label_text": [255, 255, 255],
    },
)


def sample_scene(seed: int) -> dict:
    """Sample four equal-inventory panels with one and only one equal pair."""
    rng = random.Random(int(seed))
    target_index = int(seed) % 4
    total = rng.choice((8, 10, 12))
    half = total // 2

    deltas = [1, 2, min(3, half - 1)]
    rng.shuffle(deltas)
    signs = [1, -1, rng.choice((-1, 1))]
    rng.shuffle(signs)
    distractor_pairs = []
    for delta, sign in zip(deltas, signs, strict=True):
        low, high = half - delta, half + delta
        distractor_pairs.append((low, high) if sign > 0 else (high, low))
    rng.shuffle(distractor_pairs)

    panels: list[dict[str, Any]] = []
    distractor_cursor = 0
    for index, label in enumerate(LABELS):
        if index == target_index:
            left_count, right_count = half, half
        else:
            left_count, right_count = distractor_pairs[distractor_cursor]
            distractor_cursor += 1
        panels.append(
            {
                "label": label,
                "left_cells": rng.sample(range(GRID_CELLS), left_count),
                "right_cells": rng.sample(range(GRID_CELLS), right_count),
            }
        )
    return {
        "panels": panels,
        "palette": copy.deepcopy(rng.choice(_PALETTES)),
        "origin": [rng.randint(-7, 7), rng.randint(-5, 5)],
    }


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _scaled_box(box: tuple[int, int, int, int], scale: int) -> tuple[int, int, int, int]:
    return tuple(int(value * scale) for value in box)  # type: ignore[return-value]


def _draw_centered(
    draw: ImageDraw.ImageDraw,
    xy: tuple[int, int],
    text: str,
    font: ImageFont.ImageFont,
    fill: tuple[int, int, int],
) -> None:
    bbox = draw.textbbox((0, 0), text, font=font)
    width, height = bbox[2] - bbox[0], bbox[3] - bbox[1]
    draw.text((xy[0] - width / 2, xy[1] - height / 2 - bbox[1]), text, font=font, fill=fill)


def render(scene: dict) -> Image.Image:
    """Render one antialiased PNG-ready image from the full latent scene."""
    scale = 3
    palette = {key: tuple(value) for key, value in scene["palette"].items()}
    ox, oy = (int(value) for value in scene["origin"])
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), palette["background"])
    draw = ImageDraw.Draw(image)
    label_font = _font(25 * scale)
    small_font = _font(14 * scale)

    panel_origins = ((42, 38), (398, 38), (42, 300), (398, 300))
    for panel, (base_x, base_y) in zip(scene["panels"], panel_origins, strict=True):
        x, y = base_x + ox, base_y + oy
        outer = _scaled_box((x, y, x + 320, y + 222), scale)
        draw.rounded_rectangle(
            outer,
            radius=16 * scale,
            fill=palette["panel"],
            outline=palette["outer"],
            width=3 * scale,
        )

        badge = _scaled_box((x + 13, y + 13, x + 51, y + 51), scale)
        draw.ellipse(badge, fill=palette["accent"])
        _draw_centered(
            draw,
            ((x + 32) * scale, (y + 32) * scale),
            str(panel["label"]),
            label_font,
            palette["label_text"],
        )

        boxes = ((x + 58, y + 48, x + 166, y + 202), (x + 194, y + 48, x + 302, y + 202))
        for side, (box, cells) in enumerate(zip(boxes, (panel["left_cells"], panel["right_cells"]), strict=True)):
            draw.rectangle(
                _scaled_box(box, scale),
                fill=palette["panel"],
                outline=palette["outline"],
                width=4 * scale,
            )
            tag_x = (box[0] + box[2]) // 2
            _draw_centered(
                draw,
                (tag_x * scale, (y + 33) * scale),
                "LEFT" if side == 0 else "RIGHT",
                small_font,
                palette["outer"],
            )
            grid_x = (box[0] + 24, box[0] + 54, box[0] + 84)
            grid_y = (box[1] + 27, box[1] + 61, box[1] + 95, box[1] + 129)
            for cell in cells:
                cx = grid_x[int(cell) % 3]
                cy = grid_y[int(cell) // 3]
                radius = 8
                draw.ellipse(
                    _scaled_box((cx - radius, cy - radius, cx + radius, cy + radius), scale),
                    fill=palette["pip"],
                )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def _differences(scene: dict) -> list[int]:
    panels = scene["panels"]
    return [abs(len(panel["left_cells"]) - len(panel["right_cells"])) for panel in panels]


def analytic_gold(scene: dict) -> str:
    """Select the visible label of the sole panel with equal compartment counts."""
    panels = scene["panels"]
    winners = [index for index, panel in enumerate(panels) if len(panel["left_cells"]) == len(panel["right_cells"])]
    if len(winners) != 1:
        raise ValueError(f"expected one equal-count panel, found {len(winners)}")
    return str(panels[winners[0]]["label"])


def margin(scene: dict) -> float:
    """Runner-up gap in absolute left/right count difference (larger is safer)."""
    ordered = sorted(_differences(scene))
    return float(ordered[1] - ordered[0]) if len(ordered) >= 2 else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        panels = scene["panels"]
        if len(panels) != 4 or [panel.get("label") for panel in panels] != list(LABELS):
            return True
        for panel in panels:
            left = panel["left_cells"]
            right = panel["right_cells"]
            if not left or not right or len(left) + len(right) not in (8, 10, 12):
                return True
            if len(left) != len(set(left)) or len(right) != len(set(right)):
                return True
            if any(not isinstance(cell, int) or not 0 <= cell < GRID_CELLS for cell in [*left, *right]):
                return True
        return len([difference for difference in _differences(scene) if difference == 0]) != 1 or margin(scene) < 2.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """List-order is latent: reversing pip draw lists leaves all pixels unchanged."""
    twin = copy.deepcopy(scene)
    for panel in twin["panels"]:
        panel["left_cells"] = list(reversed(panel["left_cells"]))
        panel["right_cells"] = list(reversed(panel["right_cells"]))
    return [("reverse_all_pip_lists", twin)]
