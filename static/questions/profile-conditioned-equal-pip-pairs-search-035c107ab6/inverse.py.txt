"""Independent image-only inverse for equal-pip paired compartments."""

from __future__ import annotations

from dataclasses import dataclass

import numpy as np
from PIL import Image

_PIP = np.array([24, 48, 82], dtype=np.int16)
_OUTLINE = np.array([35, 137, 151], dtype=np.int16)


@dataclass(frozen=True)
class Component:
    area: int
    cx: float
    cy: float
    x0: int
    y0: int
    x1: int
    y1: int


def _near(arr: np.ndarray, color: np.ndarray, radius: float) -> np.ndarray:
    delta = arr.astype(np.int32) - color.astype(np.int32)
    return np.sum(delta * delta, axis=2, dtype=np.int32) <= radius * radius


def _components(mask: np.ndarray) -> list[Component]:
    """Return 8-connected components using only the supplied raster mask."""
    ys, xs = np.nonzero(mask)
    remaining = set(zip(ys.tolist(), xs.tolist(), strict=True))
    found: list[Component] = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        pixels = [start]
        while stack:
            y, x = stack.pop()
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    neighbor = (y + dy, x + dx)
                    if neighbor in remaining:
                        remaining.remove(neighbor)
                        stack.append(neighbor)
                        pixels.append(neighbor)
        py = np.fromiter((item[0] for item in pixels), dtype=np.float64)
        px = np.fromiter((item[1] for item in pixels), dtype=np.float64)
        found.append(
            Component(
                area=len(pixels),
                cx=float(px.mean()),
                cy=float(py.mean()),
                x0=int(px.min()),
                y0=int(py.min()),
                x1=int(px.max()),
                y1=int(py.max()),
            )
        )
    return found


def _measure(image: Image.Image) -> tuple[str, tuple[tuple[int, int], ...], float]:
    arr = np.asarray(image.convert("RGB"))
    if arr.ndim != 3 or arr.shape[0] < 200 or arr.shape[1] < 300:
        raise ValueError("image is too small for four paired panels")

    boxes = [
        comp
        for comp in _components(_near(arr, _OUTLINE, 48.0))
        if comp.area >= 700 and comp.x1 - comp.x0 >= 90 and comp.y1 - comp.y0 >= 125
    ]
    if len(boxes) != 8:
        raise ValueError(f"expected eight visible compartment outlines, found {len(boxes)}")

    pips = [
        comp
        for comp in _components(_near(arr, _PIP, 48.0))
        if 110 <= comp.area <= 260 and 11 <= comp.x1 - comp.x0 <= 20 and 11 <= comp.y1 - comp.y0 <= 20
    ]
    if not 32 <= len(pips) <= 48:
        raise ValueError(f"implausible pip inventory: {len(pips)}")

    boxes.sort(key=lambda comp: (comp.cy, comp.cx))
    top = sorted(boxes[:4], key=lambda comp: comp.cx)
    bottom = sorted(boxes[4:], key=lambda comp: comp.cx)
    ordered = [*top, *bottom]
    if max(comp.cy for comp in top) >= min(comp.cy for comp in bottom):
        raise ValueError("compartment rows are not separable")

    counts = []
    assigned = 0
    for box in ordered:
        count = sum(box.x0 < pip.cx < box.x1 and box.y0 < pip.cy < box.y1 for pip in pips)
        counts.append(int(count))
        assigned += int(count)
    if assigned != len(pips):
        raise ValueError("at least one pip is outside every detected compartment")

    pairs = tuple((counts[index], counts[index + 1]) for index in range(0, 8, 2))
    differences = [abs(left - right) for left, right in pairs]
    winners = [index for index, difference in enumerate(differences) if difference == 0]
    if len(winners) != 1:
        raise ValueError(f"expected one equal-count panel, found {len(winners)}")
    ordered_differences = sorted(differences)
    measured_margin = float(ordered_differences[1] - ordered_differences[0])
    if measured_margin <= 0:
        raise ValueError("equality winner has no runner-up margin")
    return ("ABCD"[winners[0]], pairs, measured_margin)


def decision_from_image(image: Image.Image) -> str:
    """Recover the answer solely by enumerating visible boxes and pip components."""
    decision, _, _ = _measure(image)
    return decision


def diagnostics_from_image(image: Image.Image) -> dict:
    decision, pairs, measured_margin = _measure(image)
    return {"decision": decision, "pairs": [list(pair) for pair in pairs], "margin": measured_margin}
