"""Deterministic Pillow world for paired 100% composition strips."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

COLOR_RGB = {
    "blue": (42, 105, 214),
    "orange": (232, 119, 38),
    "green": (47, 158, 91),
}
BACKGROUND = (248, 249, 247)
PLOT = (255, 255, 255)
INK = (39, 44, 51)
MUTED = (101, 108, 117)
STRIP_LEFT = 134
STRIP_WIDTH = 500
STRIP_HEIGHT = 56
STRIP_Y = {"A": 112, "B": 236}


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    colors = ["blue", "orange", "green"]
    winner_index = int(seed) % len(colors)

    # All category totals across A+B are 64,68,68 percentage points (randomly
    # assigned independently of the winner), so largest total color area cannot
    # reveal the largest signed gain. Every gain profile sums to zero.
    totals = [64, 68, 68]
    rng.shuffle(totals)
    profiles = [(16, -4, -12), (14, -2, -12), (18, -8, -10)]
    winner_gain, loss_one, loss_two = profiles[(int(seed) // 3) % len(profiles)]
    losers = [index for index in range(3) if index != winner_index]
    rng.shuffle(losers)
    gains = [0, 0, 0]
    gains[winner_index] = winner_gain
    gains[losers[0]] = loss_one
    gains[losers[1]] = loss_two
    shares_a = [(total - gain) // 2 for total, gain in zip(totals, gains, strict=True)]
    shares_b = [(total + gain) // 2 for total, gain in zip(totals, gains, strict=True)]
    assert sum(shares_a) == sum(shares_b) == 100

    order_a = colors.copy()
    order_b = colors.copy()
    rng.shuffle(order_a)
    rng.shuffle(order_b)
    return {
        "width": 720,
        "height": 380,
        "colors": colors,
        "shares_a": shares_a,
        "shares_b": shares_b,
        "orders": {"A": order_a, "B": order_b},
    }


def _gains(scene: dict[str, Any]) -> list[int]:
    return [
        int(b) - int(a)
        for a, b in zip(scene["shares_a"], scene["shares_b"], strict=True)
    ]


def analytic_gold(scene: dict[str, Any]) -> str:
    gains = _gains(scene)
    ordered = sorted(gains, reverse=True)
    if len(ordered) < 2 or ordered[0] == ordered[1]:
        return "abstain"
    return str(scene["colors"][gains.index(ordered[0])])


def margin(scene: dict[str, Any]) -> float:
    gains = sorted(_gains(scene), reverse=True)
    return float(gains[0] - gains[1])


def is_quarantined(scene: dict[str, Any]) -> bool:
    rows = [[int(value) for value in scene[key]] for key in ("shares_a", "shares_b")]
    return any(sum(row) != 100 or min(row) < 18 for row in rows) or margin(scene) < 8.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []


def render(scene: dict[str, Any]) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    colors = [str(value) for value in scene["colors"]]
    by_panel = {
        "A": dict(zip(colors, [int(v) for v in scene["shares_a"]], strict=True)),
        "B": dict(zip(colors, [int(v) for v in scene["shares_b"]], strict=True)),
    }
    orders = {panel: [str(v) for v in scene["orders"][panel]] for panel in ("A", "B")}

    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    try:
        title_font = ImageFont.truetype("DejaVuSans-Bold.ttf", 18 * scale)
        subtitle_font = ImageFont.truetype("DejaVuSans.ttf", 12 * scale)
        panel_font = ImageFont.truetype("DejaVuSans-Bold.ttf", 18 * scale)
        percent_font = ImageFont.truetype("DejaVuSans-Bold.ttf", 11 * scale)
    except OSError:
        title_font = subtitle_font = panel_font = percent_font = ImageFont.load_default()

    draw.text((42 * scale, 20 * scale), "Category shares in two compositions", fill=INK, font=title_font)
    draw.text((42 * scale, 51 * scale), "Each full strip represents 100%", fill=MUTED, font=subtitle_font)

    for panel in ("A", "B"):
        y = STRIP_Y[panel]
        draw.text((72 * scale, (y + 15) * scale), panel, fill=INK, font=panel_font)
        x = STRIP_LEFT
        for color in orders[panel]:
            share = by_panel[panel][color]
            segment_width = share * 5
            draw.rectangle(
                (x * scale, y * scale, (x + segment_width) * scale, (y + STRIP_HEIGHT) * scale),
                fill=COLOR_RGB[color],
            )
            label = f"{share}%"
            box = draw.textbbox((0, 0), label, font=percent_font)
            label_width = box[2] - box[0]
            draw.text(
                ((x + (segment_width - label_width / scale) / 2) * scale, (y + 19) * scale),
                label,
                fill=PLOT,
                font=percent_font,
                stroke_width=1 * scale,
                stroke_fill=(35, 39, 45),
            )
            x += segment_width
        draw.rectangle(
            (STRIP_LEFT * scale, y * scale, (STRIP_LEFT + STRIP_WIDTH) * scale, (y + STRIP_HEIGHT) * scale),
            outline=INK,
            width=2 * scale,
        )

    draw.text((190 * scale, 326 * scale), "Compare B share minus A share for each color", fill=MUTED, font=subtitle_font)
    return image.resize((width, height), Image.Resampling.LANCZOS)
