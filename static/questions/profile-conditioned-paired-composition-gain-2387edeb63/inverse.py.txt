"""Independent pixel-only inverse for paired composition share changes."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_COLORS = {
    "blue": (42, 105, 214),
    "orange": (232, 119, 38),
    "green": (47, 158, 91),
}


def _near(arr: np.ndarray, rgb: tuple[int, int, int], tolerance: float = 48.0) -> np.ndarray:
    delta = arr.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2, dtype=np.int32) <= tolerance * tolerance


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    work = mask.copy()
    height, width = work.shape
    output: list[tuple[int, int, int, int, int]] = []
    for y0, x0 in zip(*np.nonzero(work), strict=True):
        if not work[y0, x0]:
            continue
        queue = deque([(int(y0), int(x0))])
        work[y0, x0] = False
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = y + dy, x + dx
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        output.append((len(xs), min(xs), max(xs), min(ys), max(ys)))
    return output


def _segment_widths(arr: np.ndarray, rgb: tuple[int, int, int]) -> list[tuple[float, float]]:
    segments: list[tuple[float, float]] = []
    for area, x0, x1, y0, y1 in _components(_near(arr, rgb)):
        width, height = x1 - x0 + 1, y1 - y0 + 1
        if area > 3500 and width >= 80 and height >= 45:
            segments.append(((y0 + y1) / 2.0, float(width)))
    segments.sort()
    if len(segments) != 2:
        raise ValueError(f"expected two large segments, recovered {len(segments)}")
    if segments[1][0] - segments[0][0] < 70:
        raise ValueError("composition rows are not resolvable")
    return segments


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    widths = {name: _segment_widths(arr, rgb) for name, rgb in _COLORS.items()}
    gains = {name: parts[1][1] - parts[0][1] for name, parts in widths.items()}
    ranking = sorted(gains.items(), key=lambda item: item[1], reverse=True)
    if ranking[0][1] - ranking[1][1] < 25.0:
        raise ValueError("largest share gain is not raster-separated")
    return ranking[0][0]
