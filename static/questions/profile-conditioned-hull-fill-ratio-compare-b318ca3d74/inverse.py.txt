"""Independent pixel-only inverse for convex-hull fill ratio comparison."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


PANEL_RECTS = ((60, 120, 480, 550), (520, 120, 940, 550))


def _colored_mask(image: Image.Image, rect: tuple[int, int, int, int]) -> np.ndarray:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    x0, y0, x1, y1 = rect
    # Keep well inside the panel: the crop excludes the panel rule, label, and
    # caption while leaving a generous margin around the silhouette.
    crop = pixels[y0 + 24 : y1 - 24, x0 + 24 : x1 - 24]
    spread = crop.max(axis=2) - crop.min(axis=2)
    brightness = crop.max(axis=2)
    darkness = crop.min(axis=2)
    return (spread >= 42) & (brightness <= 242) & (darkness <= 225)


def _cross(origin: tuple[float, float], first: tuple[float, float], second: tuple[float, float]) -> float:
    return (first[0] - origin[0]) * (second[1] - origin[1]) - (
        first[1] - origin[1]
    ) * (second[0] - origin[0])


def _convex_hull(points: list[tuple[float, float]]) -> list[tuple[float, float]]:
    ordered = sorted(set(points))
    if len(ordered) <= 2:
        return ordered
    lower: list[tuple[float, float]] = []
    for point in ordered:
        while len(lower) >= 2 and _cross(lower[-2], lower[-1], point) <= 0.0:
            lower.pop()
        lower.append(point)
    upper: list[tuple[float, float]] = []
    for point in reversed(ordered):
        while len(upper) >= 2 and _cross(upper[-2], upper[-1], point) <= 0.0:
            upper.pop()
        upper.append(point)
    return lower[:-1] + upper[:-1]


def _polygon_area(points: list[tuple[float, float]]) -> float:
    return 0.5 * abs(
        sum(
            points[index][0] * points[(index + 1) % len(points)][1]
            - points[(index + 1) % len(points)][0] * points[index][1]
            for index in range(len(points))
        )
    )


def _fill_deficit(image: Image.Image, rect: tuple[int, int, int, int]) -> float:
    mask = _colored_mask(image, rect)
    area = int(mask.sum())
    if area < 5000:
        raise ValueError(f"colored component is too small: {area} pixels")
    rows, columns = np.nonzero(mask)
    points = [(float(x), float(y)) for y, x in zip(rows, columns)]
    hull = _convex_hull(points)
    if len(hull) < 3:
        raise ValueError("colored component has no measurable convex hull")
    hull_area = _polygon_area(hull)
    if hull_area <= float(area):
        # Rasterization can make a nearly convex hull fraction equal to one;
        # clamp only this numerical edge rather than manufacturing a decision.
        return 0.0
    return 1.0 - float(area) / hull_area


def decision_from_image(image: Image.Image) -> str:
    """Return the panel with the larger hull deficit using only final RGB pixels."""

    score_a = _fill_deficit(image, PANEL_RECTS[0])
    score_b = _fill_deficit(image, PANEL_RECTS[1])
    if math.isclose(score_a, score_b, rel_tol=0.0, abs_tol=1e-9):
        raise ValueError("pixel scores are exactly tied")
    return "A" if score_a > score_b else "B"
