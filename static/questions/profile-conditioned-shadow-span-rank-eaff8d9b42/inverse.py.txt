"""Image-only decision arm for shadow_span_rank.

This module receives a rasterized PNG and nothing else. It segments the four rod
colours, recovers the horizontal extent each coloured mark occupies, and reports
the colour with the widest extent, abstaining when the evidence is damaged or
the runner-up gap is too small to resolve from pixels.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

TARGETS = {
    "red": (214, 30, 34),
    "green": (23, 138, 62),
    "blue": (36, 66, 204),
    "orange": (240, 140, 12),
}
TOLERANCE = 60
MIN_PIXELS = 250
MIN_GAP_PX = 7.0
MAX_LANE_HEIGHT = 130
ABSTAIN = "abstain"


def _mask(array: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    diff = np.abs(array.astype(np.int16) - np.array(rgb, dtype=np.int16))
    return diff.max(axis=2) <= TOLERANCE


def decision_from_image(image: Image.Image) -> str:
    """Return the colour with the widest shadow, or 'abstain'."""

    array = np.asarray(image.convert("RGB"))
    extents: dict[str, float] = {}
    for name, rgb in TARGETS.items():
        mask = _mask(array, rgb)
        count = int(mask.sum())
        if count < MIN_PIXELS:
            return ABSTAIN
        rows = np.flatnonzero(mask.any(axis=1))
        cols = np.flatnonzero(mask.any(axis=0))
        if rows.size == 0 or cols.size == 0:
            return ABSTAIN
        if int(rows[-1] - rows[0]) > MAX_LANE_HEIGHT:
            return ABSTAIN
        extents[name] = float(cols[-1] - cols[0])
    ranked = sorted(extents.items(), key=lambda item: item[1], reverse=True)
    if ranked[0][1] - ranked[1][1] < MIN_GAP_PX:
        return ABSTAIN
    return ranked[0][0]


def decision_margin_from_image(image: Image.Image) -> float:
    """Recovered runner-up gap in pixels; -1.0 when the arm abstains."""

    array = np.asarray(image.convert("RGB"))
    extents = []
    for rgb in TARGETS.values():
        mask = _mask(array, rgb)
        if int(mask.sum()) < MIN_PIXELS:
            return -1.0
        cols = np.flatnonzero(mask.any(axis=0))
        extents.append(float(cols[-1] - cols[0]))
    extents.sort(reverse=True)
    return extents[0] - extents[1]
