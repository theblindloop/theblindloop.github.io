"""Renderer and analytic arm for the detached grid-ledger state world."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont


WIDTH = 640
HEIGHT = 640
BACKGROUND = (250, 251, 247)
RING_COLOR = (214, 70, 66)

# The palette changes the visual ink without changing the state code.  The
# oracle identifies the filled/outlined geometry rather than relying on one
# answer-bearing color.
PALETTES = (
    {
        "ink": (34, 43, 53),
        "panel": (35, 91, 108),
        "grid": (132, 155, 165),
        "panel_fill": (239, 246, 246),
    },
    {
        "ink": (54, 38, 62),
        "panel": (113, 63, 105),
        "grid": (164, 139, 161),
        "panel_fill": (247, 241, 247),
    },
    {
        "ink": (45, 57, 38),
        "panel": (77, 107, 64),
        "grid": (147, 164, 137),
        "panel_fill": (243, 248, 239),
    },
)

BOARD_X0 = 112
BOARD_Y0 = 204
CELL = 34
BOARD_CENTERS_X = tuple(128 + CELL * index for index in range(4))
BOARD_CENTERS_Y = tuple(222 + CELL * index for index in range(4))

ROW_PANEL = (284, 110, 416, 260)
ROW_MARKER_X = 350
ROW_MARKER_Y0 = 143
ROW_MARKER_STEP = 27

COL_PANEL = (284, 310, 416, 460)
COL_MARKER_X0 = 309
COL_MARKER_STEP = 27
COL_MARKER_Y = 385


def _font(size: int) -> ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _translate(box: tuple[float, ...], dx: int, dy: int) -> tuple[int, ...]:
    values = list(box)
    for index in range(0, len(values), 2):
        values[index] = int(round(values[index])) + dx
        values[index + 1] = int(round(values[index + 1])) + dy
    return tuple(values)


def _point(point: tuple[float, float], dx: int, dy: int) -> tuple[int, int]:
    return (int(round(point[0])) + dx, int(round(point[1])) + dy)


def _draw_diamond(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    radius: int,
    filled: bool,
    color: tuple[int, int, int],
    width: int,
) -> None:
    cx, cy = center
    points = [(cx, cy - radius), (cx + radius, cy), (cx, cy + radius), (cx - radius, cy)]
    if filled:
        draw.polygon(points, fill=color)
    else:
        draw.line(points + [points[0]], fill=color, width=width, joint="curve")


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    box: tuple[int, int, int, int],
    title: str,
    palette: dict[str, tuple[int, int, int]],
    dx: int,
    dy: int,
    width: int,
) -> None:
    shifted = _translate(box, dx, dy)
    draw.rounded_rectangle(
        shifted,
        radius=8,
        fill=palette["panel_fill"],
        outline=palette["panel"],
        width=max(2, width),
    )
    x0, y0, _, _ = shifted
    draw.text((x0 + 9, y0 + 7), title, fill=palette["panel"], font=_font(15))


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced, deterministic scene from one integer seed."""

    rng = random.Random(int(seed))
    target_row = rng.randrange(4)
    target_col = rng.randrange(4)
    row_ops = [rng.randrange(2) for _ in range(4)]
    col_ops = [rng.randrange(2) for _ in range(4)]

    # Alternate the terminal class by seed parity.  The initial state is then
    # solved from the visible operations, so all nuisance choices are sampled
    # before the requested answer is balanced.
    requested = int(seed) & 1
    initial_state = requested ^ row_ops[target_row] ^ col_ops[target_col]

    boundary = int(seed) % 11 == 0
    return {
        "initial_state": int(initial_state),
        "target_row": int(target_row),
        "target_col": int(target_col),
        "row_ops": [int(value) for value in row_ops],
        "col_ops": [int(value) for value in col_ops],
        "translation": [rng.randint(-9, 9), rng.randint(-9, 9)],
        "palette_id": rng.randrange(len(PALETTES)),
        "stroke": 1.8 if boundary else float(rng.choice((3.0, 3.5, 4.0))),
        "ledger_gap": int(rng.choice((15, 17, 19))) if not boundary else 13,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the initial grid and two physically separated operation ledgers."""

    palette = PALETTES[int(scene["palette_id"]) % len(PALETTES)]
    dx, dy = (int(scene["translation"][0]), int(scene["translation"][1]))
    stroke = max(1, int(round(float(scene["stroke"]))))
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    # The three bordered panels never share a decisive connecting stroke.
    board_box = (BOARD_X0 - 10, BOARD_Y0 - 8, BOARD_X0 + 4 * CELL + 10, BOARD_Y0 + 4 * CELL + 8)
    _draw_panel(draw, board_box, "INITIAL GRID", palette, dx, dy, stroke)
    board_left = BOARD_X0 + dx
    board_top = BOARD_Y0 + dy
    board_right = board_left + 4 * CELL
    board_bottom = board_top + 4 * CELL
    for index in range(5):
        x = board_left + index * CELL
        y = board_top + index * CELL
        draw.line((x, board_top, x, board_bottom), fill=palette["grid"], width=1)
        draw.line((board_left, y, board_right, y), fill=palette["grid"], width=1)

    font = _font(16)
    for index, label in enumerate(("A", "B", "C", "D")):
        cx = BOARD_CENTERS_X[index] + dx
        cy = BOARD_CENTERS_Y[index] + dy
        draw.text((board_left - 25, cy - 9), label, fill=palette["panel"], font=font)
        draw.text((cx - 5, board_top - 25), str(index + 1), fill=palette["panel"], font=font)

    for row in range(4):
        for col in range(4):
            cx = BOARD_CENTERS_X[col] + dx
            cy = BOARD_CENTERS_Y[row] + dy
            filled = row == int(scene["target_row"]) and col == int(scene["target_col"])
            state = int(scene["initial_state"]) if filled else (row + 2 * col + int(scene["palette_id"])) % 2
            radius = 8
            if state:
                draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=palette["ink"])
            else:
                draw.ellipse(
                    (cx - radius, cy - radius, cx + radius, cy + radius),
                    fill=palette["panel_fill"],
                    outline=palette["ink"],
                    width=max(2, stroke),
                )

    target = (
        BOARD_CENTERS_X[int(scene["target_col"])] + dx,
        BOARD_CENTERS_Y[int(scene["target_row"])] + dy,
    )
    draw.ellipse(
        (target[0] - 14, target[1] - 14, target[0] + 14, target[1] + 14),
        outline=RING_COLOR,
        width=max(2, stroke),
    )

    _draw_panel(draw, ROW_PANEL, "ROW LEDGER", palette, dx, dy, stroke)
    row_x0, row_y0, row_x1, row_y1 = _translate(ROW_PANEL, dx, dy)
    for index in range(1, 4):
        y = row_y0 + 31 + index * ROW_MARKER_STEP
        draw.line((row_x0 + 8, y, row_x1 - 8, y), fill=palette["grid"], width=1)
    for index, label in enumerate(("A", "B", "C", "D")):
        cy = ROW_MARKER_Y0 + index * ROW_MARKER_STEP + dy
        draw.text((row_x0 + 11, cy - 8), label, fill=palette["panel"], font=font)
        _draw_diamond(
            draw,
            (ROW_MARKER_X + dx, cy),
            9,
            bool(scene["row_ops"][index]),
            palette["ink"],
            max(2, stroke),
        )

    gap_shift = int(scene["ledger_gap"]) - 17
    col_box = (COL_PANEL[0], COL_PANEL[1] + gap_shift, COL_PANEL[2], COL_PANEL[3] + gap_shift)
    _draw_panel(draw, col_box, "COLUMN LEDGER", palette, dx, dy, stroke)
    col_x0, col_y0, col_x1, col_y1 = _translate(col_box, dx, dy)
    marker_y = COL_MARKER_Y + gap_shift + dy
    for index in range(1, 4):
        x = col_x0 + 10 + index * COL_MARKER_STEP
        draw.line((x, col_y0 + 31, x, col_y1 - 8), fill=palette["grid"], width=1)
    for index in range(4):
        cx = COL_MARKER_X0 + index * COL_MARKER_STEP + dx
        draw.text((cx - 5, col_y0 + 12), str(index + 1), fill=palette["panel"], font=font)
        _draw_diamond(
            draw,
            (cx, marker_y),
            9,
            bool(scene["col_ops"][index]),
            palette["ink"],
            max(2, stroke),
        )

    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Replay only the five label-bearing scene fields."""

    state = int(scene["initial_state"])
    state ^= int(scene["row_ops"][int(scene["target_row"])] )
    state ^= int(scene["col_ops"][int(scene["target_col"])] )
    return "ON" if state else "OFF"


def margin(scene: dict[str, Any]) -> float:
    """Return a one-sided raster clarity margin for the replay inputs."""

    return round(min(float(scene["stroke"]) - 1.0, float(scene["ledger_gap"]) - 10.0), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return float(scene["stroke"]) <= 2.0 or float(scene["ledger_gap"]) <= 13.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # The public latent declaration is not_applicable: no pixel-identical
    # aliases are claimed, so this arm intentionally returns no transforms.
    return []
