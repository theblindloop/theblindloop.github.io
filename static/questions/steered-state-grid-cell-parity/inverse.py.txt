"""Independent pixel-only inverse for the grid-ledger raster."""

from __future__ import annotations

from math import hypot

import numpy as np
from PIL import Image


BASE_GRID_X = (128, 162, 196, 230)
BASE_GRID_Y = (222, 256, 290, 324)
ROW_X = 350
ROW_Y0 = 143
ROW_STEP = 27
COL_X0 = 309
COL_STEP = 27
COL_Y = 385


def _luma(pixel: np.ndarray) -> float:
    return float(0.2126 * pixel[0] + 0.7152 * pixel[1] + 0.0722 * pixel[2])


def _dark(array: np.ndarray, x: int, y: int) -> bool:
    height, width = array.shape[:2]
    if x < 0 or y < 0 or x >= width or y >= height:
        return False
    return _luma(array[y, x]) < 185.0


def _mean_luma(array: np.ndarray, cx: int, cy: int, radius: int = 1) -> float:
    values = []
    height, width = array.shape[:2]
    for y in range(cy - radius, cy + radius + 1):
        for x in range(cx - radius, cx + radius + 1):
            if 0 <= x < width and 0 <= y < height:
                values.append(_luma(array[y, x]))
    if not values:
        raise ValueError("empty pixel sample")
    return sum(values) / len(values)


def _accent_center(array: np.ndarray) -> tuple[float, float]:
    red = array[:, :, 0].astype(np.int16)
    green = array[:, :, 1].astype(np.int16)
    blue = array[:, :, 2].astype(np.int16)
    mask = (red > 145) & (red > green + 55) & (red > blue + 45)
    ys, xs = np.nonzero(mask)
    if len(xs) < 30:
        raise ValueError("target ring is not visible")
    return (
        float(xs.mean()),
        float(ys.mean()),
    )


def _nearest_index(value: float, choices: tuple[int, ...]) -> int:
    return min(range(len(choices)), key=lambda index: abs(value - choices[index]))


def _diamond_score(array: np.ndarray, cx: int, cy: int) -> int:
    points = ((cx, cy - 8), (cx + 8, cy), (cx, cy + 8), (cx - 8, cy))
    score = 0
    for px, py in points:
        for dy in (-1, 0, 1):
            for dx in (-1, 0, 1):
                score += int(_dark(array, px + dx, py + dy))
    return score


def _marker_state(array: np.ndarray, cx: int, cy: int) -> int:
    # Both marker encodings have dark corners.  This check makes a missing
    # marker an abstention rather than silently treating it as an outlined hold.
    if _diamond_score(array, cx, cy) < 8:
        raise ValueError("ledger marker is absent or ambiguous")
    return int(_mean_luma(array, cx, cy, 1) < 175.0)


def _column_marker_y(array: np.ndarray, cx: int, expected: int) -> int:
    candidates = range(expected - 7, expected + 8)
    best = max(candidates, key=lambda value: _diamond_score(array, cx, value))
    if _diamond_score(array, cx, best) < 8:
        raise ValueError("column ledger marker row is absent")
    return int(best)


def decision_from_image(image: Image.Image) -> str:
    """Recover the terminal state using only RGB pixels in ``image``."""

    image = image.convert("RGB")
    array = np.asarray(image, dtype=np.uint8)
    ring_x, ring_y = _accent_center(array)
    target_col = _nearest_index(ring_x, BASE_GRID_X)
    target_row = _nearest_index(ring_y, BASE_GRID_Y)
    dx = int(round(ring_x - BASE_GRID_X[target_col]))
    dy = int(round(ring_y - BASE_GRID_Y[target_row]))
    if abs(dx) > 18 or abs(dy) > 18:
        raise ValueError("grid translation is outside the observable layout")

    state_x = BASE_GRID_X[target_col] + dx
    state_y = BASE_GRID_Y[target_row] + dy
    # The state disk must remain visible inside the target ring.  A center-only
    # answer would mistake a deleted filled glyph for an outlined glyph.
    state_ink = 0
    for py in range(state_y - 7, state_y + 8):
        for px in range(state_x - 7, state_x + 8):
            if hypot(px - state_x, py - state_y) <= 8.5:
                state_ink += int(_dark(array, px, py))
    if state_ink < 20:
        raise ValueError("initial state glyph is absent")
    state = int(_mean_luma(array, state_x, state_y, 1) < 175.0)

    row_x = ROW_X + dx
    row_y = ROW_Y0 + ROW_STEP * target_row + dy
    row_update = _marker_state(array, row_x, row_y)

    col_x = COL_X0 + COL_STEP * target_col + dx
    col_y = _column_marker_y(array, col_x, COL_Y + dy)
    col_update = _marker_state(array, col_x, col_y)

    state ^= row_update
    state ^= col_update
    return "ON" if state else "OFF"
