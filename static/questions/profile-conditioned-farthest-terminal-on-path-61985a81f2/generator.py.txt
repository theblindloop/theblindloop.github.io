"""Tracing world: which coloured terminal is farther from the ring along the path.

A single winding blue path (one connected, non-self-crossing serpentine
polyline) is drawn. An orange port ring sits near the start of the path. Two
coloured terminal dots (red and green) sit at two separated on-path positions.
Walking only along the blue path from the orange ring, the solution is the colour
of the terminal whose on-path (geodesic) distance from the ring is larger. Because
the path winds back and forth in a serpentine, the on-path order of the two
terminals need not follow their straight-line proximity to the ring, so the
decision cannot be solved by Euclidean projection alone; it requires tracing the
drawn stroke.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "farthest_terminal_on_path-0.1.0"

CANVAS = (480, 480)
BG = (255, 255, 255)
COLOR_BLUE = (40, 90, 210)
COLOR_RED = (225, 40, 40)
COLOR_GREEN = (40, 170, 70)
COLOR_ORANGE = (245, 150, 20)

STROKE = 20          # blue stroke thickness (px)
PORT_R = 5           # orange port radius
MARK_R = 4           # terminal marker radius
MARGIN_PAD = 46      # keep geometry away from image borders
LANE_GAP = 30.0      # min gap between serpentine lanes so markers map to their own stroke
ARC_SEP = 34.0       # min arc distance between the two terminals and from the port
END_MARGIN = 40.0    # min distance from path endpoints for the terminals

TERMINAL_COLORS = ["red", "green"]


def _pts(points: list) -> list[tuple[float, float]]:
    return [(float(p[0]), float(p[1])) for p in points]


def _path_length(points: list[tuple[float, float]]) -> float:
    return sum(math.dist(a, b) for a, b in zip(points, points[1:]))


def _point_at(points: list[tuple[float, float]], s: float) -> tuple[float, float]:
    remaining = s
    for a, b in zip(points, points[1:]):
        seg = math.dist(a, b)
        if remaining <= seg or seg == 0:
            t = 0.0 if seg == 0 else remaining / seg
            return (a[0] + (b[0] - a[0]) * t, a[1] + (b[1] - a[1]) * t)
        remaining -= seg
    return points[-1]


def _rejection_loop(rng: random.Random) -> dict:
    w, h = CANVAS
    lo, hi = MARGIN_PAD, w - MARGIN_PAD
    for _ in range(20000):
        lane_vert_gap = LANE_GAP
        n_lanes = rng.randint(4, 7)
        y_lo = lo + 10
        y_hi = hi - 10
        usable_h = y_hi - y_lo
        span_h = (n_lanes - 1) * lane_vert_gap
        span_h = min(span_h, usable_h)
        step_h = span_h / (n_lanes - 1) if n_lanes > 1 else 0.0
        x_start = rng.uniform(lo + 15, lo + 50)
        x_end = rng.uniform(hi - 50, hi - 15)
        path = []
        for lane in range(n_lanes):
            y = y_lo + lane * step_h
            if lane % 2 == 0:
                path.append((x_start, y))
                path.append((x_end, y))
            else:
                path.append((x_end, y))
                path.append((x_start, y))
        path = _pts(path)
        total = _path_length(path)
        if total < END_MARGIN * 4:
            continue

        s_port = END_MARGIN * rng.uniform(1.0, 1.5)

        # Two well-separated terminal positions in the later part of the path.
        lo_s = s_port + 3 * ARC_SEP
        hi_s = total - END_MARGIN - 20
        if hi_s - lo_s < 4 * ARC_SEP:
            continue
        far_s = rng.uniform(lo_s + 2 * ARC_SEP, hi_s)
        max_near = far_s - 2 * ARC_SEP
        if max_near <= lo_s:
            continue
        near_s = rng.uniform(lo_s, max_near)

        far_color = rng.choice(TERMINAL_COLORS)
        near_color = "green" if far_color == "red" else "red"

        far_pos = _point_at(path, far_s)
        near_pos = _point_at(path, near_s)

        owner = {far_color: far_pos, near_color: near_pos}
        marks = [
            {"x": float(owner["red"][0]), "y": float(owner["red"][1]), "color": "red"},
            {"x": float(owner["green"][0]), "y": float(owner["green"][1]), "color": "green"},
        ]

        allgood = True
        port = _point_at(path, s_port)
        for m in marks + [{"x": port[0], "y": port[1]}]:
            if not (20 < m["x"] < w - 20 and 20 < m["y"] < h - 20):
                allgood = False
                break
        if not allgood:
            continue

        scene = {
            "path": path,
            "port": [float(port[0]), float(port[1])],
            "marks": marks,
            "answer": far_color,
        }
        return scene
    raise RuntimeError("could not satisfy path constraints")


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    return _rejection_loop(rng)


def _draw_polyline(draw: ImageDraw.ImageDraw, pts, color, width):
    for a, b in zip(pts, pts[1:]):
        draw.line([a, b], fill=color, width=width, joint="curve")


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", CANVAS, BG)
    draw = ImageDraw.Draw(img)
    path = _pts(scene["path"])
    _draw_polyline(draw, path, COLOR_BLUE, STROKE)
    colors = {"red": COLOR_RED, "green": COLOR_GREEN}
    for mark in scene["marks"]:
        r = MARK_R
        px, py = float(mark["x"]), float(mark["y"])
        draw.ellipse([px - r, py - r, px + r, py + r], fill=colors[mark["color"]])
    px, py = float(scene["port"][0]), float(scene["port"][1])
    draw.ellipse([px - PORT_R, py - PORT_R, px + PORT_R, py + PORT_R], fill=COLOR_ORANGE)
    return img


def _arc_position(scene: dict, pt: tuple[float, float]) -> float:
    best = None
    best_d = float("inf")
    cum = 0.0
    for a, b in zip(_pts(scene["path"]), _pts(scene["path"])[1:]):
        abx, aby = b[0] - a[0], b[1] - a[1]
        seg2 = abx * abx + aby * aby
        t = 0.0 if seg2 == 0 else ((pt[0] - a[0]) * abx + (pt[1] - a[1]) * aby) / seg2
        t = max(0.0, min(1.0, t))
        proj = (a[0] + abx * t, a[1] + aby * t)
        d = math.dist(pt, proj)
        if d < best_d:
            best_d = d
            best = cum + math.dist(a, proj)
        cum += math.dist(a, b)
    return best


def _dist_from_ring(scene: dict, pt: tuple[float, float]) -> float:
    return abs(_arc_position(scene, pt) - _arc_position(scene, tuple(scene["port"])))


def analytic_gold(scene: dict) -> str:
    """Colour of the terminal whose on-path distance from the ring is larger."""
    red = [m for m in scene["marks"] if m["color"] == "red"][0]
    green = [m for m in scene["marks"] if m["color"] == "green"][0]
    d_red = _dist_from_ring(scene, (red["x"], red["y"]))
    d_green = _dist_from_ring(scene, (green["x"], green["y"]))
    return "red" if d_red > d_green else "green"


def margin(scene: dict) -> float:
    """On-path separation between the two terminals (px)."""
    red = [m for m in scene["marks"] if m["color"] == "red"][0]
    green = [m for m in scene["marks"] if m["color"] == "green"][0]
    return abs(_arc_position(scene, (red["x"], red["y"])) - _arc_position(scene, (green["x"], green["y"])))


def is_quarantined(scene: dict) -> bool:
    w, h = CANVAS
    if margin(scene) < ARC_SEP * 1.5:
        return True
    for m in scene["marks"]:
        if not (20 <= m["x"] <= w - 20 and 20 <= m["y"] <= h - 20):
            return True
    return False


def latent_symmetries(scene: dict) -> list:
    return []
