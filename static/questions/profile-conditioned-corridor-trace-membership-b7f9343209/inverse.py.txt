"""Independent pixel-only inverse for the two-island corridor question."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image


# These are duplicated visual constants, rather than imported from the renderer.
NAVY = np.array((27, 50, 92), dtype=np.float32)
ISLAND = np.array((151, 157, 166), dtype=np.float32)
GREEN = np.array((44, 160, 94), dtype=np.float32)
RED = np.array((211, 73, 69), dtype=np.float32)


def _mask(arr: np.ndarray, color: np.ndarray, tolerance: float) -> np.ndarray:
    return np.linalg.norm(arr - color, axis=2) <= tolerance


def _largest_component(mask: np.ndarray) -> np.ndarray:
    """Recover the largest 8-connected stroke component from a binary mask."""

    ys, xs = np.where(mask)
    if len(xs) == 0:
        raise ValueError("no trace pixels")
    remaining = set(zip(ys.tolist(), xs.tolist()))
    best: list[tuple[int, int]] = []
    while remaining:
        start = remaining.pop()
        queue = deque([start])
        current = [start]
        while queue:
            y, x = queue.popleft()
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    neighbor = (y + dy, x + dx)
                    if neighbor in remaining:
                        remaining.remove(neighbor)
                        queue.append(neighbor)
                        current.append(neighbor)
        if len(current) > len(best):
            best = current
    if len(best) < 100:
        raise ValueError("trace component is too small")
    # Convert image indexing (row, column) to geometric (x, y) coordinates.
    return np.asarray(best, dtype=np.float32)[:, ::-1]


def _centroid(mask: np.ndarray, name: str) -> np.ndarray:
    ys, xs = np.where(mask)
    if len(xs) < 20:
        raise ValueError(f"missing {name} marker")
    return np.array((float(xs.mean()), float(ys.mean())), dtype=np.float32)


def _island_band(mask: np.ndarray) -> tuple[float, float, float, float]:
    ys, xs = np.where(mask)
    if len(xs) < 1000:
        raise ValueError("two island regions were not recovered")
    occupied_rows = np.where(mask.any(axis=1))[0]
    row_gaps = np.diff(occupied_rows)
    if len(row_gaps) == 0 or int(row_gaps.max()) < 8:
        raise ValueError("island regions do not have a visible separating gap")
    gap_index = int(np.argmax(row_gaps))
    split = float(occupied_rows[gap_index] + occupied_rows[gap_index + 1]) / 2.0
    top = np.column_stack((xs[ys <= split], ys[ys <= split])).astype(np.float32)
    bottom = np.column_stack((xs[ys > split], ys[ys > split])).astype(np.float32)
    if len(top) < 500 or len(bottom) < 500:
        raise ValueError("island masks did not split into two regions")
    top_radius = math.sqrt(len(top) / math.pi)
    bottom_radius = math.sqrt(len(bottom) / math.pi)
    center_x = float(np.concatenate((top[:, 0], bottom[:, 0])).mean())
    facing_top = float(top[:, 1].max())
    facing_bottom = float(bottom[:, 1].min())
    return center_x, facing_top, facing_bottom, max(top_radius, bottom_radius)


def decision_from_image(image: Image.Image) -> str:
    """Return yes iff the recovered ordered navy trace occupies the island gap."""

    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    navy = _mask(arr, NAVY, 6.0)
    islands = _mask(arr, ISLAND, 6.0)
    component = _largest_component(navy)
    green = _centroid(_mask(arr, GREEN, 8.0), "green")
    red = _centroid(_mask(arr, RED, 8.0), "red")
    ordered = component[np.argsort(component[:, 0])]
    if float(np.linalg.norm(component - green, axis=1).min()) > 24.0:
        raise ValueError("trace does not meet green source")
    if float(np.linalg.norm(component - red, axis=1).min()) > 24.0:
        raise ValueError("trace does not meet red terminal")
    center_x, facing_top, facing_bottom, radius = _island_band(islands)
    if facing_bottom - facing_top < 8.0:
        raise ValueError("island gap is unresolved")
    # The ordered centerline must traverse the central island-width window and
    # have pixels well inside the visible facing-edge band.
    x_window = (ordered[:, 0] >= center_x - radius) & (ordered[:, 0] <= center_x + radius)
    y_window = (ordered[:, 1] >= facing_top + 2.0) & (ordered[:, 1] <= facing_bottom - 2.0)
    through_gap = int(np.count_nonzero(x_window & y_window)) >= 12
    left_reached = float(component[:, 0].min()) < center_x - radius - 35.0
    right_reached = float(component[:, 0].max()) > center_x + radius + 35.0
    if not (left_reached and right_reached):
        raise ValueError("trace does not span both sides of the islands")
    return "yes" if through_gap else "no"
