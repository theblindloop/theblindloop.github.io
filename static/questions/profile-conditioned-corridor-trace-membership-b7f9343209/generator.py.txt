"""Deterministic renderer and analytic labels for the corridor trace world."""

from __future__ import annotations

import math
import random
from pathlib import Path

from PIL import Image, ImageDraw


WIDTH = 512
HEIGHT = 360
BACKGROUND = (248, 247, 243)
NAVY = (27, 50, 92)
ISLAND = (151, 157, 166)
PALE = (207, 218, 225)
GREEN = (44, 160, 94)
RED = (211, 73, 69)


def sample_scene(seed: int) -> dict:
    """Sample one balanced, finite-resolution-safe scene from an integer seed."""

    rng = random.Random(int(seed))
    route_mode = "gap" if int(seed) % 2 == 0 else "outside"
    mid_y = rng.randint(166, 194)
    center_x = rng.randint(238, 274)
    separation = rng.randint(110, 122)
    radius_top = rng.randint(35, 39)
    radius_bottom = rng.randint(35, 39)
    source_y = mid_y + rng.randint(-10, 10)
    target_y = mid_y + rng.randint(-10, 10)
    outer_side = "top" if rng.randrange(2) == 0 else "bottom"
    stroke = rng.choice([7, 8, 9])
    gap_wiggle = rng.choice([3, 4, 5, 6])
    distractor_variant = rng.randrange(3)
    return {
        "route_mode": route_mode,
        "outer_side": outer_side,
        "center_x": center_x,
        "mid_y": mid_y,
        "separation": separation,
        "radius_top": radius_top,
        "radius_bottom": radius_bottom,
        "source_y": source_y,
        "target_y": target_y,
        "source_x": 58,
        "target_x": 454,
        "stroke": stroke,
        "gap_wiggle": gap_wiggle,
        "distractor_variant": distractor_variant,
        "width": WIDTH,
        "height": HEIGHT,
    }


def _centers(scene: dict) -> tuple[tuple[float, float], tuple[float, float]]:
    cx = float(scene["center_x"])
    mid = float(scene["mid_y"])
    half = float(scene["separation"]) / 2.0
    return (cx, mid - half), (cx, mid + half)


def _outer_y(scene: dict) -> float:
    top, bottom = _centers(scene)
    if scene["outer_side"] == "top":
        return top[1] - float(scene["radius_top"]) - 19.0
    return bottom[1] + float(scene["radius_bottom"]) + 19.0


def _route_points(scene: dict) -> list[tuple[float, float]]:
    """Return a centerline with a long central run or a clear outer detour."""

    cx = float(scene["center_x"])
    mid = float(scene["mid_y"])
    sx, sy = float(scene["source_x"]), float(scene["source_y"])
    tx, ty = float(scene["target_x"]), float(scene["target_y"])
    left = cx - 122.0
    right = cx + 122.0
    if scene["route_mode"] == "gap":
        points: list[tuple[float, float]] = [(sx, sy), (left, mid)]
        wiggle = float(scene["gap_wiggle"])
        # The alternating centerline stays inside the visible facing-edge band.
        for index in range(0, 11):
            x = left + (right - left) * index / 10.0
            y = mid + (wiggle if index % 2 else -wiggle)
            points.append((x, y))
        points.extend([(right, mid), (tx, ty)])
        return points
    outer = _outer_y(scene)
    return [(sx, sy), (left, outer), (right, outer), (tx, ty)]


def _draw_distractor(draw: ImageDraw.ImageDraw, scene: dict) -> None:
    variant = int(scene["distractor_variant"])
    if variant == 0:
        pts = [(54, 327), (145, 305), (242, 328), (350, 304), (458, 326)]
        draw.line(pts, fill=PALE, width=4, joint="curve")
    elif variant == 1:
        pts = [(52, 38), (150, 24), (257, 41), (362, 22), (460, 37)]
        draw.line(pts, fill=PALE, width=4, joint="curve")
    else:
        draw.line([(48, 316), (174, 294), (270, 314)], fill=PALE, width=4, joint="curve")
        draw.line([(338, 48), (460, 28)], fill=PALE, width=4, joint="curve")


def _draw_circle(draw: ImageDraw.ImageDraw, center: tuple[float, float], radius: float) -> None:
    x, y = center
    box = (round(x - radius), round(y - radius), round(x + radius), round(y + radius))
    draw.ellipse(box, fill=ISLAND)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_distractor(draw, scene)
    top, bottom = _centers(scene)
    _draw_circle(draw, top, float(scene["radius_top"]))
    _draw_circle(draw, bottom, float(scene["radius_bottom"]))
    points = _route_points(scene)
    draw.line(
        [(round(x), round(y)) for x, y in points],
        fill=NAVY,
        width=int(scene["stroke"]),
        joint="curve",
    )
    source = (round(float(scene["source_x"])), round(float(scene["source_y"])))
    target = (round(float(scene["target_x"])), round(float(scene["target_y"])))
    draw.ellipse((source[0] - 13, source[1] - 13, source[0] + 13, source[1] + 13), fill=GREEN)
    draw.ellipse((target[0] - 13, target[1] - 13, target[0] + 13, target[1] + 13), fill=RED)
    return image


def analytic_gold(scene: dict) -> str:
    return "yes" if scene["route_mode"] == "gap" else "no"


def margin(scene: dict) -> float:
    """Signed centerline clearance from the facing-edge boundary in pixels."""

    top, bottom = _centers(scene)
    gap_top = top[1] + float(scene["radius_top"])
    gap_bottom = bottom[1] - float(scene["radius_bottom"])
    gap_half = (gap_bottom - gap_top) / 2.0
    half_stroke = float(scene["stroke"]) / 2.0
    if scene["route_mode"] == "gap":
        return gap_half - float(scene["gap_wiggle"]) - half_stroke
    return -(abs(_outer_y(scene) - float(scene["mid_y"])) - gap_half - half_stroke)


def is_quarantined(scene: dict) -> bool:
    return abs(float(margin(scene))) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # No interchangeable latent slots are present in this minimal scene model.
    return []


if __name__ == "__main__":
    sample = sample_scene(101)
    render(sample).save(Path("/tmp/corridor_trace_membership.png"))
    print(analytic_gold(sample), margin(sample))
