from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    pass


def _largest_component(mask: np.ndarray) -> np.ndarray:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    best: list[tuple[int, int]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            yy, xx = queue.popleft()
            points.append((yy, xx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = yy + dy, xx + dx
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) > len(best):
            best = points
    out = np.zeros_like(mask, dtype=bool)
    if best:
        yy, xx = zip(*best)
        out[np.asarray(yy), np.asarray(xx)] = True
    return out


def _profile_score(image: Image.Image) -> tuple[float, int]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    chroma = arr.max(axis=2) - arr.min(axis=2)
    # All context is neutral. This threshold admits antialiased emblem edges but
    # rejects the laptop, circular badge, outline, and background.
    mask = chroma >= 34
    component = _largest_component(mask)
    ys, xs = np.nonzero(component)
    # A valid answer requires substantial interior support from the whole filled
    # silhouette. Large erasures therefore cause abstention even if one surviving
    # horizontal slice still hints at a class.
    if len(xs) < 3000:
        raise OracleAbstention("no substantial connected chromatic emblem")
    y0, y1 = int(ys.min()), int(ys.max())
    x0, x1 = int(xs.min()), int(xs.max())
    if y1 - y0 < 55 or x1 - x0 < 34:
        raise OracleAbstention("emblem is too small")

    # Deskew by comparing widths over broad vertical bands. The renderer's small
    # rotation nuisance is far below the separation between the two profiles.
    height = y1 - y0 + 1
    widths = np.zeros(height, dtype=np.float64)
    for i, y in enumerate(range(y0, y1 + 1)):
        row = np.nonzero(component[y])[0]
        widths[i] = float(row.max() - row.min() + 1) if len(row) else 0.0
    maximum = max(1.0, float(np.percentile(widths, 95)))
    lo_a, hi_a = int(0.24 * height), int(0.42 * height)
    lo_b, hi_b = int(0.63 * height), int(0.82 * height)
    upper = float(np.median(widths[lo_a:hi_a])) / maximum
    lower = float(np.median(widths[lo_b:hi_b])) / maximum
    return lower - upper, len(xs)


def decision_from_image(image: Image.Image) -> str:
    score, _ = _profile_score(image)
    if abs(score) < 0.06:
        raise OracleAbstention(f"ambiguous width profile: {score:.3f}")
    return "pear" if score > 0 else "apple"
