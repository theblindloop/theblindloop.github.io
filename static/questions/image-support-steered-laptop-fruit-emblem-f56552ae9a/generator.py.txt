from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image, ImageDraw


WIDTH = 512
HEIGHT = 512
FRUIT_COLORS = [
    (216, 50, 62),
    (32, 116, 210),
    (30, 154, 92),
    (142, 62, 190),
    (230, 106, 28),
]
LAPTOP_COLORS = [(76, 82, 92), (104, 110, 120), (133, 139, 148), (58, 64, 73)]
BACKGROUNDS = [(242, 238, 228), (232, 239, 242), (244, 234, 240), (236, 242, 232)]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    fruit = "apple" if int(seed) % 2 else "pear"
    return {
        "fruit": fruit,
        "cx": rng.randint(205, 307),
        "cy": rng.randint(190, 280),
        "scale": rng.randint(88, 106),
        "emblem_color": list(FRUIT_COLORS[rng.randrange(len(FRUIT_COLORS))]),
        "laptop_color": list(LAPTOP_COLORS[rng.randrange(len(LAPTOP_COLORS))]),
        "background_color": list(BACKGROUNDS[rng.randrange(len(BACKGROUNDS))]),
        "tilt": rng.choice([-8, -5, -2, 2, 5, 8]),
        "width": WIDTH,
        "height": HEIGHT,
    }


def _profile_half_width(fruit: str, t: float) -> float:
    """Normalized half-width of a single connected, filled fruit silhouette."""
    if fruit == "apple":
        knots = [
            (0.00, 0.38),
            (0.12, 0.48),
            (0.26, 0.51),
            (0.48, 0.50),
            (0.68, 0.43),
            (0.86, 0.28),
            (1.00, 0.07),
        ]
    elif fruit == "pear":
        knots = [
            (0.00, 0.08),
            (0.12, 0.18),
            (0.30, 0.24),
            (0.48, 0.34),
            (0.66, 0.48),
            (0.82, 0.50),
            (0.94, 0.34),
            (1.00, 0.10),
        ]
    else:
        raise ValueError(f"unknown fruit {fruit!r}")
    for (ta, wa), (tb, wb) in zip(knots, knots[1:]):
        if ta <= t <= tb:
            u = (t - ta) / (tb - ta)
            return wa + u * (wb - wa)
    return knots[-1][1]


def _fruit_mask(fruit: str, size: int = 144) -> Image.Image:
    mask = Image.new("L", (size, size), 0)
    draw = ImageDraw.Draw(mask)
    top = int(size * (0.17 if fruit == "apple" else 0.08))
    bottom = int(size * 0.93)
    center = (size - 1) / 2.0
    for y in range(top, bottom + 1):
        t = (y - top) / (bottom - top)
        half = _profile_half_width(fruit, t) * size * 0.74
        draw.line((center - half, y, center + half, y), fill=255, width=2)
    if fruit == "apple":
        # A shallow central notch separates two visible upper lobes. The angled
        # stem touches the right lobe, so the full silhouette is still connected.
        draw.polygon(
            [
                (center - size * 0.105, top - 1),
                (center + size * 0.105, top - 1),
                (center, top + size * 0.105),
            ],
            fill=0,
        )
        draw.line(
            (center + size * 0.06, top + size * 0.025, center + size * 0.105, 2),
            fill=255,
            width=max(6, size // 20),
        )
        return mask
    # A thick pear stem remains joined to the body.
    stem_w = max(5, size // 22)
    draw.rounded_rectangle(
        (center - stem_w / 2, 1, center + stem_w / 2, top + size * 0.04),
        radius=stem_w // 2,
        fill=255,
    )
    return mask


def render(scene: dict) -> Image.Image:
    fruit = scene["fruit"]
    cx = int(scene["cx"])
    cy = int(scene["cy"])
    scale = int(scene["scale"])
    emblem_color = tuple(int(v) for v in scene["emblem_color"])
    laptop_color = tuple(int(v) for v in scene["laptop_color"])
    background_color = tuple(int(v) for v in scene["background_color"])
    tilt = float(scene["tilt"])
    width = int(scene["width"])
    height = int(scene["height"])

    ss = 3
    image = Image.new("RGB", (width * ss, height * ss), background_color)
    draw = ImageDraw.Draw(image)

    # Remote context is deliberately neutral and identical across answer classes.
    lid = [(100, 92), (412, 92), (438, 368), (74, 368)]
    lid = [(x * ss, y * ss) for x, y in lid]
    outline = tuple(max(0, c - 34) for c in laptop_color)
    draw.polygon(lid, fill=laptop_color, outline=outline, width=5 * ss)
    draw.rounded_rectangle(
        (48 * ss, 367 * ss, 464 * ss, 410 * ss),
        radius=12 * ss,
        fill=tuple(min(220, c + 35) for c in laptop_color),
        outline=outline,
        width=4 * ss,
    )
    draw.polygon(
        [(48 * ss, 402 * ss), (464 * ss, 402 * ss), (430 * ss, 427 * ss), (82 * ss, 427 * ss)],
        fill=outline,
    )

    # The badge is neutral; only the one filled fruit component is chromatic.
    badge_r = int(scale * 0.58)
    draw.ellipse(
        ((cx - badge_r) * ss, (cy - badge_r) * ss, (cx + badge_r) * ss, (cy + badge_r) * ss),
        fill=tuple(min(232, c + 95) for c in laptop_color),
        outline=outline,
        width=3 * ss,
    )
    mask = _fruit_mask(fruit, 144 * ss)
    target = int(scale * ss)
    mask = mask.resize((target, target), Image.Resampling.LANCZOS)
    mask = mask.rotate(tilt, resample=Image.Resampling.BICUBIC, expand=True)
    colored = Image.new("RGB", mask.size, emblem_color)
    px = int(cx * ss - mask.width / 2)
    py = int(cy * ss - mask.height / 2)
    image.paste(colored, (px, py), mask)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    fruit = scene["fruit"]
    if fruit not in {"apple", "pear"}:
        raise ValueError("fruit must be apple or pear")
    return fruit


def margin(scene: dict) -> float:
    fruit = scene["fruit"]
    return 0.09 if fruit == "apple" else 0.48


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.06


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
