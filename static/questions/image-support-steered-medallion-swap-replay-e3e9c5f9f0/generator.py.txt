"""Renderer for the medallion swap-replay world.

One small medallion sits somewhere on a plain canvas: a solid central disk
surrounded by four concentric rings, each ring split down its vertical middle
into a left half and a right half of two different colours. The central disk
holds the colour carried at the start; every ring is a swap rule between its
own two half colours, and the rings act innermost first, outwards. All decisive
evidence therefore lives inside one thick, connected, area-like blob whose
absolute position is randomised; every other mark on the canvas is a
low-saturation grey decoration that changes nothing.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "medallion_swap_replay-renderer-0.1.0"

WIDTH = 900
HEIGHT = 900

COLOUR_NAMES = ("red", "green", "blue", "purple", "orange")
BASE_RGB = {
    "red": (214, 42, 44),
    "green": (40, 150, 62),
    "blue": (44, 84, 206),
    "purple": (140, 52, 180),
    "orange": (235, 140, 25),
}

RINGS = 4
_JITTER = 6
_MIN_CENTRE = 200
_MAX_CENTRE = 700
_QUARANTINE_MARGIN = 6.4


def _replay(core: str, ops: list[tuple[str, str]]) -> str:
    """Apply each ring's swap rule in order to the carried colour."""
    carried = core
    for left, right in ops:
        if carried == left:
            carried = right
        elif carried == right:
            carried = left
    return carried


def _active_steps(core: str, ops: list[tuple[str, str]]) -> int:
    carried = core
    active = 0
    for left, right in ops:
        if carried in (left, right):
            active += 1
            carried = right if carried == left else left
    return active


def _sample_candidate(rng: random.Random) -> dict:
    thickness = rng.choice([6, 7, 7, 8, 8, 9])
    core_radius = rng.choice([12, 13, 14])
    core = rng.choice(COLOUR_NAMES)

    ops: list[tuple[str, str]] = []
    left_prev = core
    right_prev = core
    for _ in range(RINGS):
        left = rng.choice([c for c in COLOUR_NAMES if c != left_prev])
        right = rng.choice([c for c in COLOUR_NAMES if c not in (left, right_prev)])
        ops.append((left, right))
        left_prev, right_prev = left, right

    outer = core_radius + RINGS * thickness
    scene = {
        "core": core,
        "ops": [list(pair) for pair in ops],
        "core_radius": core_radius,
        "thickness": thickness,
        "centre_x": rng.randint(_MIN_CENTRE, _MAX_CENTRE),
        "centre_y": rng.randint(_MIN_CENTRE, _MAX_CENTRE),
        "bg_rgb": [rng.randint(246, 252), rng.randint(246, 252), rng.randint(244, 250)],
        "core_jitter": _jitter(rng),
        "ring_jitter": [[_jitter(rng), _jitter(rng)] for _ in range(RINGS)],
        "outer_radius": outer,
        "distractors": [],
    }
    scene["distractors"] = _sample_distractors(rng, scene)
    return scene


def _jitter(rng: random.Random) -> list[int]:
    return [rng.randint(-_JITTER, _JITTER) for _ in range(3)]


def paint(name: str, jitter: list[int]) -> tuple[int, int, int]:
    """Painted RGB for a named colour: the shared base tone plus scene jitter.

    The colour name is what the raster shows, so the gold-bearing fields
    ``core`` and ``ops`` are painted fields rather than hidden labels.
    """
    base = BASE_RGB[name]
    return tuple(max(0, min(255, base[i] + jitter[i])) for i in range(3))


def core_rgb(scene: dict) -> tuple[int, int, int]:
    return paint(scene["core"], scene["core_jitter"])


def ring_rgb(scene: dict, index: int) -> tuple[tuple[int, int, int], tuple[int, int, int]]:
    left, right = scene["ops"][index]
    jitter = scene["ring_jitter"][index]
    return paint(left, jitter[0]), paint(right, jitter[1])


def _sample_distractors(rng: random.Random, scene: dict) -> list[dict]:
    """Inert grey marks. Their saturation is far below any painted colour."""
    marks: list[dict] = []
    keep_out = scene["outer_radius"] + 26
    for _ in range(rng.randint(7, 12)):
        for _attempt in range(60):
            x = rng.randint(40, WIDTH - 40)
            y = rng.randint(40, HEIGHT - 40)
            size = rng.randint(14, 62)
            if math.hypot(x - scene["centre_x"], y - scene["centre_y"]) > keep_out + size:
                grey = rng.randint(150, 205)
                tint = rng.randint(-5, 5)
                marks.append(
                    {
                        "kind": rng.choice(["rect", "ellipse", "bar", "ring"]),
                        "x": x,
                        "y": y,
                        "w": size,
                        "h": rng.randint(14, 62),
                        "rgb": [grey, max(0, min(255, grey + tint)), max(0, min(255, grey - tint))],
                        "width": rng.randint(3, 9),
                    }
                )
                break
    return marks


def sample_scene(seed: int) -> dict:
    """Sample a scene whose answer is a deterministic function of the seed.

    The target answer is cycled with the seed so that answer identity is
    counterbalanced against the carried start colour, the ring colours, the
    medallion position, its size and the inert decoration.
    """
    rng = random.Random(seed * 7919 + 13)
    target = COLOUR_NAMES[(seed * 7 + 3) % len(COLOUR_NAMES)]
    want_minimal = (seed % 3) == 0
    fallback = None
    for _attempt in range(20000):
        scene = _sample_candidate(rng)
        ops = [tuple(pair) for pair in scene["ops"]]
        gold = _replay(scene["core"], ops)
        active = _active_steps(scene["core"], ops)
        if gold == scene["core"] or active < 2:
            continue
        if _replay(scene["core"], ops[::-1]) == gold:
            continue
        if _replay(scene["core"], [ops[-1]]) == gold:
            continue
        if gold != target:
            continue
        if fallback is None:
            fallback = scene
        if want_minimal and active != 2:
            continue
        return scene
    if fallback is not None:
        return fallback
    raise RuntimeError(f"no admissible scene for seed {seed}")


def _ring_bbox(scene: dict, radius: int) -> tuple[int, int, int, int]:
    cx, cy = scene["centre_x"], scene["centre_y"]
    return (cx - radius, cy - radius, cx + radius, cy + radius)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), tuple(scene["bg_rgb"]))
    draw = ImageDraw.Draw(image)

    for mark in scene["distractors"]:
        rgb = tuple(mark["rgb"])
        x, y, w, h = mark["x"], mark["y"], mark["w"], mark["h"]
        box = (x - w // 2, y - h // 2, x + w // 2, y + h // 2)
        if mark["kind"] == "rect":
            draw.rectangle(box, fill=rgb)
        elif mark["kind"] == "ellipse":
            draw.ellipse(box, fill=rgb)
        elif mark["kind"] == "bar":
            draw.rectangle((x - w // 2, y - mark["width"], x + w // 2, y + mark["width"]), fill=rgb)
        else:
            draw.ellipse(box, outline=rgb, width=mark["width"])

    # Rings, outermost first: each ring is a left half and a right half.
    for index in range(RINGS - 1, -1, -1):
        radius = scene["core_radius"] + (index + 1) * scene["thickness"]
        box = _ring_bbox(scene, radius)
        left_rgb, right_rgb = ring_rgb(scene, index)
        draw.pieslice(box, 90, 270, fill=left_rgb)
        draw.pieslice(box, -90, 90, fill=right_rgb)

    draw.ellipse(_ring_bbox(scene, scene["core_radius"]), fill=core_rgb(scene))
    return image


def analytic_gold(scene: dict) -> str:
    return _replay(scene["core"], [tuple(pair) for pair in scene["ops"]])


def _adjacent_rgb_pairs(scene: dict) -> list[tuple[list[int], list[int]]]:
    rings = [ring_rgb(scene, index) for index in range(RINGS)]
    core = core_rgb(scene)
    pairs = [(core, rings[0][0]), (core, rings[0][1])]
    for index in range(RINGS - 1):
        for side in (0, 1):
            pairs.append((rings[index][side], rings[index + 1][side]))
    for index in range(RINGS):
        pairs.append((rings[index][0], rings[index][1]))
    return pairs


def margin(scene: dict) -> float:
    """Legibility margin in pixels-equivalent units.

    It is the tightest of the ring thickness, the core radius, the weakest
    painted colour separation between two touching regions, the clearance from
    the canvas edge and the clearance from the nearest inert grey mark.
    """
    separation = min(
        math.dist(a, b) for a, b in _adjacent_rgb_pairs(scene)
    )
    outer = scene["outer_radius"]
    edge = min(
        scene["centre_x"] - outer,
        scene["centre_y"] - outer,
        WIDTH - outer - scene["centre_x"],
        HEIGHT - outer - scene["centre_y"],
    )
    clearance = min(
        (
            math.hypot(m["x"] - scene["centre_x"], m["y"] - scene["centre_y"])
            - outer
            - max(m["w"], m["h"]) / 2.0
            for m in scene["distractors"]
        ),
        default=100.0,
    )
    return round(
        min(
            float(scene["thickness"]),
            scene["core_radius"] * 0.8,
            separation / 12.0,
            edge / 12.0,
            clearance / 4.0,
        ),
        6,
    )


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < _QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist: every sampled field is painted into the raster."""
    return []
