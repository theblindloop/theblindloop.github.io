"""Deterministic raster world for state-level identity-cohort persistence."""

from __future__ import annotations

import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont


LABELS = ("A", "B", "C", "D")
STATE_NAMES = ("LOW", "MID", "HIGH")
STATE_SHAPES = ("circle", "square", "triangle")
PANEL_COUNT = 7
IDENTITY_COUNT = 4
WIDTH = 1800
HEIGHT = 900
CANVAS = (WIDTH, HEIGHT)

BACKGROUND = (250, 251, 253)
INK = (31, 42, 56)
MUTED = (92, 104, 119)
GRID = (177, 188, 202)
CARD = (255, 255, 255)
HEADER = (34, 48, 71)

LABEL_COLORS = (
    (207, 73, 88),
    (38, 126, 180),
    (111, 82, 169),
    (43, 148, 105),
)
STATE_COLORS = (
    (226, 124, 54),
    (40, 145, 185),
    (157, 89, 163),
)
PANEL_TINTS = (
    (248, 251, 255),
    (255, 250, 246),
    (248, 253, 249),
    (252, 249, 255),
    (250, 252, 252),
)

PANEL_LEFTS = tuple(20 + index * 255 for index in range(PANEL_COUNT))
PANEL_TOP = 190
PANEL_WIDTH = 235
PANEL_BOTTOM = 850
ROW_Y = (325, 450, 575, 700)
TILE_LEFT_INSET = 9
TILE_WIDTH = 217
TILE_HEIGHT = 74
IDENTITY_STRIP_WIDTH = 56
IDENTITY_SAMPLE_OFFSET = 14
MARKER_CENTER_OFFSET = 150
MARKER_RADIUS_DEFAULT = 16
KEY_Y = 104
LABEL_KEY_X = (58, 178, 298, 418)
STATE_KEY_X = (760, 890, 1020)

FONT_PATHS = (
    "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
    "/usr/share/fonts/dejavu/DejaVuSans.ttf",
)


def _font(size: int):
    for path in FONT_PATHS:
        if Path(path).is_file():
            return ImageFont.truetype(path, size=size)
    return ImageFont.load_default()


def _text_center(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str, font, fill) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    width = box[2] - box[0]
    height = box[3] - box[1]
    draw.text(
        (int(xy[0] - width / 2), int(xy[1] - height / 2 - box[1])),
        text,
        font=font,
        fill=fill,
    )


def _rgb(value) -> tuple[int, int, int]:
    return tuple(int(channel) for channel in value)


def _label_color(scene: dict, label: int) -> tuple[int, int, int]:
    return _rgb(scene["label_palette"][int(label)])


def _state_color(scene: dict, state: int) -> tuple[int, int, int]:
    return _rgb(scene["state_palette"][int(state)])


def _draw_state_marker(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    state: int,
    shape: str,
    color: tuple[int, int, int],
    radius: int,
) -> None:
    cx, cy = center
    radius = max(3, int(radius))
    if shape == "circle":
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=color, outline=INK, width=2)
    elif shape == "square":
        draw.rounded_rectangle(
            (cx - radius, cy - radius, cx + radius, cy + radius),
            radius=max(2, radius // 5),
            fill=color,
            outline=INK,
            width=2,
        )
    elif shape == "triangle":
        draw.polygon(
            ((cx, cy - radius), (cx - radius, cy + radius), (cx + radius, cy + radius)),
            fill=color,
            outline=INK,
        )
    else:
        raise ValueError(f"unknown state shape {shape!r}")


def _draw_keys(draw: ImageDraw.ImageDraw, scene: dict, label_font, state_font, small_font) -> None:
    draw.text((25, 70), "IDENTITY KEY", font=small_font, fill=MUTED)
    for identity, x in enumerate(LABEL_KEY_X):
        draw.rectangle((x - 20, KEY_Y - 18, x + 20, KEY_Y + 18), fill=_label_color(scene, identity), outline=INK, width=1)
        _text_center(draw, (x + 52, KEY_Y), LABELS[identity], label_font, INK)

    draw.text((700, 70), "STATE KEY", font=small_font, fill=MUTED)
    for state, x in enumerate(STATE_KEY_X):
        _draw_state_marker(
            draw,
            (x, KEY_Y),
            state,
            str(scene["state_shapes"][state]),
            _state_color(scene, state),
            17,
        )
        _text_center(draw, (x + 53, KEY_Y), STATE_NAMES[state], state_font, INK)


def _draw_tile(
    draw: ImageDraw.ImageDraw,
    scene: dict,
    panel_left: int,
    row: int,
    label: int,
    state: int,
    label_font,
    state_font,
) -> None:
    left = panel_left + TILE_LEFT_INSET
    center_y = ROW_Y[row]
    top = center_y - TILE_HEIGHT // 2
    draw.rounded_rectangle(
        (left, top, left + TILE_WIDTH, top + TILE_HEIGHT),
        radius=10,
        fill=CARD,
        outline=GRID,
        width=int(scene["tile_outline_width"]),
    )
    draw.rectangle(
        (left, top, left + IDENTITY_STRIP_WIDTH, top + TILE_HEIGHT),
        fill=_label_color(scene, label),
    )
    _text_center(draw, (left + IDENTITY_STRIP_WIDTH // 2, center_y), LABELS[label], label_font, (255, 255, 255))
    _draw_state_marker(
        draw,
        (left + MARKER_CENTER_OFFSET, center_y),
        state,
        str(scene["state_shapes"][state]),
        _state_color(scene, state),
        int(scene["marker_radius"]),
    )
    _text_center(draw, (left + 194, center_y), STATE_NAMES[state], state_font, MUTED)


def _draw_arrow(draw: ImageDraw.ImageDraw, left: int, right: int) -> None:
    y = PANEL_TOP + 35
    start_x = left + PANEL_WIDTH + 4
    end_x = right - 4
    draw.line((start_x, y, end_x, y), fill=MUTED, width=3)
    draw.polygon(((end_x, y), (end_x - 10, y - 6), (end_x - 10, y + 6)), fill=MUTED)


def _cohort_scores(trajectories: list[list[int]]) -> list[int]:
    if len(trajectories) != IDENTITY_COUNT or any(len(row) != PANEL_COUNT for row in trajectories):
        raise ValueError("expected four identity trajectories across seven panels")
    if any(value not in range(len(STATE_NAMES)) for row in trajectories for value in row):
        raise ValueError("state values are outside the keyed LOW/MID/HIGH range")
    cohorts: list[list[frozenset[int]]] = []
    for state in range(len(STATE_NAMES)):
        cohorts.append(
            [
                frozenset(identity for identity in range(IDENTITY_COUNT) if trajectories[identity][panel] == state)
                for panel in range(PANEL_COUNT)
            ]
        )
    return [
        sum(int(cohorts[state][panel] == cohorts[state][panel + 1]) for panel in range(PANEL_COUNT - 1))
        for state in range(len(STATE_NAMES))
    ]


def _sample_trajectories(seed: int) -> list[list[int]]:
    logic_rng = random.Random(int(seed) * 1000003 + 17)
    target = int(seed) % len(STATE_NAMES)
    for _ in range(120000):
        candidate = [
            [logic_rng.randrange(len(STATE_NAMES)) for _ in range(PANEL_COUNT)]
            for _ in range(IDENTITY_COUNT)
        ]
        scores = _cohort_scores(candidate)
        ordered = sorted(scores, reverse=True)
        if scores[target] == ordered[0] and ordered[0] - ordered[1] >= 2:
            return candidate
    raise RuntimeError("could not sample a unique state-cohort winner")


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene with independent visible nuisance variation."""

    seed = int(seed)
    style_rng = random.Random(seed * 1000033 + 31)
    trajectories = _sample_trajectories(seed)
    row_orders = [style_rng.sample(range(IDENTITY_COUNT), IDENTITY_COUNT) for _ in range(PANEL_COUNT)]
    label_palette = list(LABEL_COLORS)
    state_palette = list(STATE_COLORS)
    style_rng.shuffle(label_palette)
    style_rng.shuffle(state_palette)
    panel_tints = [PANEL_TINTS[style_rng.randrange(len(PANEL_TINTS))] for _ in range(PANEL_COUNT)]
    state_shapes = list(STATE_SHAPES)
    style_rng.shuffle(state_shapes)
    marker_radius = 7 if seed % 19 == 0 else style_rng.choice((14, 15, 16, 17, 18))
    outline_width = style_rng.choice((1, 2, 3))
    return {
        "trajectories": trajectories,
        "row_orders": row_orders,
        "label_palette": label_palette,
        "state_palette": state_palette,
        "panel_tints": panel_tints,
        "state_shapes": state_shapes,
        "marker_radius": marker_radius,
        "tile_outline_width": outline_width,
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS, BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(24)
    section_font = _font(17)
    label_font = _font(20)
    state_font = _font(14)
    small_font = _font(13)

    draw.rectangle((0, 0, WIDTH - 1, 48), fill=HEADER)
    draw.text((25, 12), "STATE COHORT PERSISTENCE", font=title_font, fill=(255, 255, 255))
    draw.text(
        (620, 16),
        "Compare exact identity sets for LOW, MID, and HIGH across arrows",
        font=small_font,
        fill=(224, 232, 243),
    )
    _draw_keys(draw, scene, label_font, state_font, small_font)
    draw.text((1310, 70), "PANEL ORDER", font=small_font, fill=MUTED)
    draw.text((1310, 96), "P1  →  P2  →  P3  →  P4  →  P5  →  P6  →  P7", font=section_font, fill=INK)

    for panel, panel_left in enumerate(PANEL_LEFTS):
        draw.rounded_rectangle(
            (panel_left, PANEL_TOP, panel_left + PANEL_WIDTH, PANEL_BOTTOM),
            radius=12,
            fill=_rgb(scene["panel_tints"][panel]),
            outline=INK,
            width=2,
        )
        _text_center(draw, (panel_left + PANEL_WIDTH // 2, PANEL_TOP + 18), f"P{panel + 1}", section_font, INK)
        draw.line((panel_left + 15, PANEL_TOP + 45, panel_left + PANEL_WIDTH - 15, PANEL_TOP + 45), fill=GRID, width=1)
        for row, label in enumerate(scene["row_orders"][panel]):
            state = int(scene["trajectories"][int(label)][panel])
            _draw_tile(draw, scene, panel_left, row, int(label), state, label_font, state_font)
        if panel < PANEL_COUNT - 1:
            _draw_arrow(draw, panel_left, PANEL_LEFTS[panel + 1])
    return image


def analytic_gold(scene: dict) -> str:
    """Return the state whose exact identity cohort persists most often."""

    scores = _cohort_scores(scene["trajectories"])
    best = max(scores)
    winners = [state for state, score in enumerate(scores) if score == best]
    if len(winners) != 1:
        raise ValueError("state-cohort persistence has no unique winner")
    return STATE_NAMES[winners[0]]


def margin(scene: dict) -> float:
    """Return the top-score gap; gaps below two are the declared boundary."""

    scores = _cohort_scores(scene["trajectories"])
    ordered = sorted(scores, reverse=True)
    return float(ordered[0] - ordered[1])


def runner_up_score(scene: dict) -> float:
    scores = _cohort_scores(scene["trajectories"])
    return float(sorted(scores, reverse=True)[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0 or int(scene["marker_radius"]) < 11


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent alias is claimed; all answer-bearing entries are rendered."""

    return []
