"""Independent image-only inverse for state-cohort persistence.

Only fixed public raster coordinates and colors are duplicated here.  The
module has no import path to renderer, prompts, generator, verifier, or gold
code, and receives only the final PNG.
"""

from __future__ import annotations

from PIL import Image


LABELS = ("A", "B", "C", "D")
STATE_NAMES = ("LOW", "MID", "HIGH")
PANEL_COUNT = 7
IDENTITY_COUNT = 4
CANVAS = (1800, 900)
PANEL_LEFTS = tuple(20 + index * 255 for index in range(PANEL_COUNT))
ROW_Y = (325, 450, 575, 700)
TILE_LEFT_INSET = 9
IDENTITY_SAMPLE_OFFSET = 14
MARKER_CENTER_OFFSET = 150
KEY_Y = 104
LABEL_KEY_X = (58, 178, 298, 418)
STATE_KEY_X = (760, 890, 1020)
LABEL_COLORS = (
    (207, 73, 88),
    (38, 126, 180),
    (111, 82, 169),
    (43, 148, 105),
)
STATE_COLORS = (
    (226, 124, 54),
    (40, 145, 185),
    (157, 89, 163),
)


def _sample(frame: Image.Image, x: int, y: int) -> tuple[int, int, int]:
    value = frame.getpixel((int(x), int(y)))
    if not isinstance(value, tuple) or len(value) != 3:
        raise ValueError("raster is not RGB")
    return tuple(int(channel) for channel in value)


def _read_palettes(frame: Image.Image) -> tuple[dict[tuple[int, int, int], int], dict[tuple[int, int, int], int]]:
    labels: dict[tuple[int, int, int], int] = {}
    for identity, x in enumerate(LABEL_KEY_X):
        color = _sample(frame, x, KEY_Y)
        if color not in LABEL_COLORS or color in labels:
            raise ValueError("identity key is unreadable or reuses a color")
        labels[color] = identity

    states: dict[tuple[int, int, int], int] = {}
    for state, x in enumerate(STATE_KEY_X):
        color = _sample(frame, x, KEY_Y)
        if color not in STATE_COLORS or color in states or color in labels:
            raise ValueError("state key is unreadable or reuses a color")
        states[color] = state
    return labels, states


def _decode_trajectories(frame: Image.Image) -> list[list[int]]:
    labels, states = _read_palettes(frame)
    trajectories: list[list[int] | None] = [None] * IDENTITY_COUNT
    pixels = frame.load()
    for panel, panel_left in enumerate(PANEL_LEFTS):
        seen: set[int] = set()
        for row, y in enumerate(ROW_Y):
            tile_left = panel_left + TILE_LEFT_INSET
            label_color = pixels[tile_left + IDENTITY_SAMPLE_OFFSET, y]
            state_color = pixels[tile_left + MARKER_CENTER_OFFSET, y]
            if label_color not in labels:
                raise ValueError(f"identity evidence missing in P{panel + 1}, row {row + 1}")
            if state_color not in states:
                raise ValueError(f"state evidence missing in P{panel + 1}, row {row + 1}")
            identity = labels[label_color]
            if identity in seen:
                raise ValueError(f"identity repeats in P{panel + 1}")
            seen.add(identity)
            if trajectories[identity] is None:
                trajectories[identity] = [states[state_color]]
            else:
                trajectories[identity].append(states[state_color])
        if seen != set(range(IDENTITY_COUNT)):
            raise ValueError(f"P{panel + 1} does not expose all identities")
    if any(row is None or len(row) != PANEL_COUNT for row in trajectories):
        raise ValueError("identity trajectory is incomplete")
    return [list(row) for row in trajectories if row is not None]


def _scores(trajectories: list[list[int]]) -> list[int]:
    cohorts: list[list[frozenset[int]]] = []
    for state in range(len(STATE_NAMES)):
        cohorts.append(
            [
                frozenset(identity for identity in range(IDENTITY_COUNT) if trajectories[identity][panel] == state)
                for panel in range(PANEL_COUNT)
            ]
        )
    return [
        sum(int(cohorts[state][panel] == cohorts[state][panel + 1]) for panel in range(PANEL_COUNT - 1))
        for state in range(len(STATE_NAMES))
    ]


def decision_from_image(image: Image.Image) -> str:
    """Decode keyed tiles, replay cohort membership, and choose the unique top state."""

    frame = image.convert("RGB")
    if frame.size != CANVAS:
        raise ValueError(f"unexpected raster size {frame.size}")
    scores = _scores(_decode_trajectories(frame))
    best = max(scores)
    winners = [state for state, score in enumerate(scores) if score == best]
    if len(winners) != 1:
        raise ValueError("cohort-persistence decision is tied")
    return STATE_NAMES[winners[0]]
