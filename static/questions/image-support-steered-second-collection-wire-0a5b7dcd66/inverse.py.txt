"""Independent pixel-only inverse for Second Collection Wire.

Only the final raster and public visual grammar are used. The parser locates all
twelve bead interiors, orders them from the gold start toward the arrow, decodes
their seven-segment letters, validates the connected wire, and replays set resets.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int = 1) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for yy, xx in np.argwhere(mask):
        y, x = int(yy), int(xx)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            output.append(np.asarray(points, dtype=np.int32))
    return output


def _decode(ink: np.ndarray, cx: float, cy: float) -> str:
    x, y = int(round(cx)), int(round(cy))
    probes = {
        "a": (x, y - 8), "g": (x, y), "d": (x, y + 8),
        "f": (x - 8, y - 5), "b": (x + 8, y - 5),
        "e": (x - 8, y + 5), "c": (x + 8, y + 5),
    }
    active = frozenset(name for name, (px, py) in probes.items() if np.any(ink[py-1:py+2, px-1:px+2]))
    patterns = {
        frozenset(("a", "b", "c", "e", "f", "g")): "A",
        frozenset(("c", "d", "e", "f", "g")): "B",
        frozenset(("a", "d", "e", "f")): "C",
    }
    if active not in patterns:
        raise ValueError("a bead letter is missing or ambiguous")
    return patterns[active]


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    if array.ndim != 3 or array.shape[2] != 3:
        raise ValueError("RGB image required")
    red, green, blue = array[:, :, 0], array[:, :, 1], array[:, :, 2]
    violet = (red >= 75) & (red <= 140) & (green >= 45) & (green <= 110) & (blue >= 135) & (blue <= 205)
    bead = (red >= 225) & (red <= 242) & (green >= 230) & (green <= 245) & (blue >= 225) & (blue <= 242)
    gold = (red >= 205) & (red <= 255) & (green >= 135) & (green <= 205) & (blue <= 75)
    ink = (red >= 20) & (red <= 80) & (green >= 20) & (green <= 85) & (blue >= 25) & (blue <= 100)

    route_components = _components(violet, 100)
    if len(route_components) != 1 or len(route_components[0]) < 5000:
        raise ValueError("the connected wire is missing or broken")
    interiors = []
    for component in _components(bead, 350):
        ys, xs = component[:, 0], component[:, 1]
        bw, bh = int(xs.max() - xs.min() + 1), int(ys.max() - ys.min() + 1)
        if 31 <= bw <= 38 and 31 <= bh <= 38 and len(component) >= 400:
            interiors.append((float(xs.mean()), float(ys.mean())))
    if len(interiors) != 12:
        raise ValueError(f"expected twelve intact bead interiors, found {len(interiors)}")
    interiors.sort(key=lambda point: point[0])
    if min(interiors[i + 1][0] - interiors[i][0] for i in range(11)) < 42:
        raise ValueError("bead order is ambiguous")
    if max(y for _, y in interiors) - min(y for _, y in interiors) > 65:
        raise ValueError("beads do not form the declared route")

    gold_components = []
    for component in _components(gold, 80):
        ys, xs = component[:, 0], component[:, 1]
        bw, bh = int(xs.max() - xs.min() + 1), int(ys.max() - ys.min() + 1)
        if len(component) >= 250 and 15 <= bw <= 30 and 15 <= bh <= 30:
            gold_components.append(component)
    if len(gold_components) != 1:
        raise ValueError("round START is missing or ambiguous")
    gy, gx = gold_components[0][:, 0].mean(), gold_components[0][:, 1].mean()
    if not (gx < interiors[0][0] - 25 and abs(gy - interiors[0][1]) < 10):
        raise ValueError("START does not identify the route direction")

    letters = [_decode(ink, cx, cy) for cx, cy in interiors]
    if sorted(letters) != list("AAAABBBBCCCC"):
        raise ValueError("letter multiplicities are invalid")
    seen: set[str] = set()
    completions: list[str] = []
    for letter in letters:
        seen.add(letter)
        if seen == {"A", "B", "C"}:
            completions.append(letter)
            seen.clear()
    if len(completions) < 2:
        raise ValueError("fewer than two complete collections")
    return completions[1]
