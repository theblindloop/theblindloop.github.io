"""Deterministic renderer and independent analytic arm for Second Collection Wire."""

from __future__ import annotations

import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 900, 600
BACKGROUND = (248, 247, 242)
INK = (43, 48, 57)
VIOLET = (105, 76, 169)
GOLD = (235, 174, 40)
BEAD = (235, 239, 235)
PALE = (222, 219, 211)
COUNT = 12
BASE_X = 164
STEP_X = 50
BASE_Y = 337
Y_OFFSETS = (0, -15, -24, -17, -2, 15, 24, 16, 1, -15, -23, -9)

SEGMENTS = {
    "A": ("a", "b", "c", "e", "f", "g"),
    "B": ("c", "d", "e", "f", "g"),
    "C": ("a", "d", "e", "f"),
}


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(str(Path("/usr/share/fonts/truetype/dejavu") / name), size)


def _second_completer(letters: list[str]) -> str:
    seen: set[str] = set()
    completions: list[str] = []
    for letter in letters:
        seen.add(letter)
        if seen == {"A", "B", "C"}:
            completions.append(letter)
            seen.clear()
    if len(completions) < 2:
        raise ValueError("sequence does not contain two complete collections")
    return completions[1]


def _second_completion_index(letters: list[str]) -> int:
    seen: set[str] = set()
    count = 0
    for index, letter in enumerate(letters):
        seen.add(letter)
        if seen == {"A", "B", "C"}:
            count += 1
            seen.clear()
            if count == 2:
                return index
    return -1


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = ("A", "B", "C")[int(seed) % 3]
    for _ in range(10000):
        letters = list("AAAABBBBCCCC")
        rng.shuffle(letters)
        index = _second_completion_index(letters)
        if index in (7, 8, 9) and _second_completer(letters) == target:
            return {
                "letters": letters,
                "offset_x": (-8, 0, 8)[rng.randrange(3)],
                "offset_y": (-10, 0, 10)[rng.randrange(3)],
                "wire_width": (7, 9, 11)[rng.randrange(3)],
            }
    raise RuntimeError("could not construct balanced collection sequence")


def _points(scene: dict) -> list[tuple[int, int]]:
    dx, dy = int(scene["offset_x"]), int(scene["offset_y"])
    return [(BASE_X + STEP_X * i + dx, BASE_Y + Y_OFFSETS[i] + dy) for i in range(COUNT)]


def _glyph(draw: ImageDraw.ImageDraw, cx: int, cy: int, letter: str) -> None:
    segments = {
        "a": (cx - 7, cy - 10, cx + 7, cy - 7),
        "g": (cx - 7, cy - 2, cx + 7, cy + 1),
        "d": (cx - 7, cy + 7, cx + 7, cy + 10),
        "f": (cx - 10, cy - 9, cx - 7, cy),
        "b": (cx + 7, cy - 9, cx + 10, cy),
        "e": (cx - 10, cy, cx - 7, cy + 9),
        "c": (cx + 7, cy, cx + 10, cy + 9),
    }
    for name in SEGMENTS[letter]:
        draw.rectangle(segments[name], fill=INK)


def render(scene: dict) -> Image.Image:
    letters = list(scene["letters"])
    if len(letters) != COUNT:
        raise ValueError("exactly twelve letters are required")
    points = _points(scene)
    wire_width = int(scene["wire_width"])
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    draw.text((70, 52), "COMPLETE TWO COLLECTIONS", fill=INK, font=_font(30, True))
    draw.text((70, 97), "Keep one A, one B, and one C. Clear after each full set.", fill=INK, font=_font(19))
    draw.rounded_rectangle((315, 154, 585, 226), radius=14, fill=(253, 252, 248), outline=PALE, width=3)
    draw.text((344, 172), "COLLECTION STARTS EMPTY", fill=INK, font=_font(18, True))
    draw.line((372, 211, 528, 211), fill=GOLD, width=5)

    # The route is a single connected component behind all touching node rims.
    draw.line(points, fill=VIOLET, width=wire_width, joint="curve")
    sx, sy = points[0]
    draw.line((sx - 44, sy, sx - 20, sy), fill=VIOLET, width=wire_width)
    draw.ellipse((sx - 58, sy - 14, sx - 30, sy + 14), fill=GOLD, outline=VIOLET, width=4)
    ex, ey = points[-1]
    draw.line((ex + 20, ey, ex + 34, ey), fill=VIOLET, width=wire_width)
    draw.polygon(((ex + 55, ey), (ex + 31, ey - 17), (ex + 31, ey + 17)), fill=VIOLET)

    for (cx, cy), letter in zip(points, letters, strict=True):
        draw.ellipse((cx - 20, cy - 20, cx + 20, cy + 20), fill=BEAD, outline=VIOLET, width=5)
        _glyph(draw, cx, cy, letter)

    draw.text((sx - 83, sy + 36), "START", fill=INK, font=_font(16, True))
    draw.text((195, 414), "REPEATS DO NOTHING  •  FULL A+B+C SETS CLEAR IMMEDIATELY", fill=INK, font=_font(17, True))
    draw.text((292, 466), "Which bead completes set 2?", fill=INK, font=_font(22, True))
    for i, letter in enumerate("ABC"):
        x = 343 + 83 * i
        draw.ellipse((x - 22, 516, x + 22, 560), fill=BEAD, outline=INK, width=3)
        draw.text((x - 8, 522), letter, fill=INK, font=_font(23, True))
    return image


def analytic_gold(scene: dict) -> str:
    return _second_completer(list(scene["letters"]))


def is_quarantined(scene: dict) -> bool:
    try:
        letters = list(scene["letters"])
        dx, dy, width = int(scene["offset_x"]), int(scene["offset_y"]), int(scene["wire_width"])
        answer = _second_completer(letters)
    except (KeyError, TypeError, ValueError):
        return True
    return not (
        len(letters) == COUNT
        and sorted(letters) == list("AAAABBBBCCCC")
        and answer in "ABC"
        and _second_completion_index(letters) in (7, 8, 9)
        and dx in (-8, 0, 8)
        and dy in (-10, 0, 10)
        and width in (7, 9, 11)
    )


def margin(scene: dict) -> float:
    return 0.0 if is_quarantined(scene) else float(int(scene["wire_width"]) + 5)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
