"""Deterministic raster world and analytic arm for start_loop_checkpoint."""

from __future__ import annotations

import math
import random
from collections import deque

from PIL import Image, ImageDraw


CANVAS = (860, 860)
SCALE = 3
NODE_RADIUS = 14
COLORS = ("red", "blue", "orange")
PALETTE = {
    "background": (250, 249, 246),
    "ink": (29, 32, 38),
    "generic": (211, 217, 224),
    "green": (38, 174, 91),
    "red": (221, 61, 61),
    "blue": (48, 107, 215),
    "orange": (239, 143, 33),
}


def _node(node_id: str, role: str, fill: str, x: int, y: int) -> dict:
    return {
        "id": node_id,
        "role": role,
        "fill": fill,
        "x": int(x),
        "y": int(y),
        "r": NODE_RADIUS,
    }


def _base_point(radius: float, slot: int) -> tuple[float, float]:
    angle = math.pi + slot * math.pi / 4.0
    return radius * math.cos(angle), radius * math.sin(angle)


def sample_scene(seed: int) -> dict:
    """Sample a crossing-free unicyclic graph with degree-matched checkpoints.

    Every block of three consecutive seeds holds geometry fixed while cycling
    the answer color. Across 24 consecutive seeds all eight rotations occur.
    """

    seed = int(seed)
    rng = random.Random(seed)
    answer_color = COLORS[seed % 3]
    distractor_colors = [color for color in COLORS if color != answer_color]
    rng.shuffle(distractor_colors)

    rotation = (seed // 3) % 8
    reflected = bool((seed // 24) % 2)
    scale_x = rng.choice((0.97, 1.00, 1.03))
    scale_y = rng.choice((0.97, 1.00, 1.03))
    translate_x = rng.choice((-6, -3, 0, 3, 6))
    translate_y = rng.choice((-6, -3, 0, 3, 6))
    angle = rotation * math.pi / 4.0
    cosine, sine = math.cos(angle), math.sin(angle)

    def transform(point: tuple[float, float]) -> tuple[int, int]:
        x, y = point
        if reflected:
            y = -y
        x *= scale_x
        y *= scale_y
        rotated_x = x * cosine - y * sine
        rotated_y = x * sine + y * cosine
        return (
            int(round(CANVAS[0] / 2 + translate_x + rotated_x)),
            int(round(CANVAS[1] / 2 + translate_y + rotated_y)),
        )

    cycle_ids = (
        "start",
        "ring_1",
        "branch_a_attach",
        "ring_3",
        "cycle_checkpoint",
        "ring_5",
        "branch_b_attach",
        "ring_7",
    )
    nodes: list[dict] = []
    for slot, node_id in enumerate(cycle_ids):
        x, y = transform(_base_point(170.0, slot))
        if node_id == "start":
            nodes.append(_node(node_id, "start", "green", x, y))
        elif node_id == "cycle_checkpoint":
            nodes.append(_node(node_id, "checkpoint", answer_color, x, y))
        else:
            nodes.append(_node(node_id, "junction", "generic", x, y))

    branch_specs = (
        ("a", 2, distractor_colors[0]),
        ("b", 6, distractor_colors[1]),
    )
    for branch, slot, color in branch_specs:
        entries = (
            (f"branch_{branch}_inner", "junction", "generic", 220.0),
            (f"branch_{branch}_checkpoint", "checkpoint", color, 270.0),
            (f"branch_{branch}_outer", "junction", "generic", 320.0),
            (f"branch_{branch}_leaf", "leaf", "generic", 370.0),
        )
        for node_id, role, fill, radius in entries:
            x, y = transform(_base_point(radius, slot))
            nodes.append(_node(node_id, role, fill, x, y))

    edges: list[list[str]] = []
    for index, node_id in enumerate(cycle_ids):
        edges.append([node_id, cycle_ids[(index + 1) % len(cycle_ids)]])
    for branch, attach in (("a", "branch_a_attach"), ("b", "branch_b_attach")):
        chain = (
            attach,
            f"branch_{branch}_inner",
            f"branch_{branch}_checkpoint",
            f"branch_{branch}_outer",
            f"branch_{branch}_leaf",
        )
        edges.extend([[left, right] for left, right in zip(chain, chain[1:])])
    return {"canvas": list(CANVAS), "nodes": nodes, "edges": edges}


def render(scene: dict) -> Image.Image:
    width, height = scene["canvas"]
    image = Image.new("RGB", (width * SCALE, height * SCALE), PALETTE["background"])
    draw = ImageDraw.Draw(image)
    by_id = {node["id"]: node for node in scene["nodes"]}
    for left_id, right_id in scene["edges"]:
        left, right = by_id[left_id], by_id[right_id]
        draw.line(
            (
                left["x"] * SCALE,
                left["y"] * SCALE,
                right["x"] * SCALE,
                right["y"] * SCALE,
            ),
            fill=PALETTE["ink"],
            width=7 * SCALE,
        )
    for node in scene["nodes"]:
        x, y, radius = node["x"] * SCALE, node["y"] * SCALE, node["r"] * SCALE
        draw.ellipse(
            (x - radius, y - radius, x + radius, y + radius),
            fill=PALETTE[node["fill"]],
            outline=PALETTE["ink"],
            width=3 * SCALE,
        )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def _adjacency(nodes: list[dict], edges: list[list[str]]) -> dict[str, set[str]]:
    ids = {node["id"] for node in nodes}
    if len(ids) != len(nodes):
        raise ValueError("node identifiers are not unique")
    adjacency = {node_id: set() for node_id in ids}
    for left, right in edges:
        if left not in adjacency or right not in adjacency or left == right:
            raise ValueError(f"invalid edge {left!r}-{right!r}")
        adjacency[left].add(right)
        adjacency[right].add(left)
    return adjacency


def _cycle_nodes(nodes: list[dict], edges: list[list[str]]) -> set[str]:
    """Return the graph's 2-core, which equals its unique cycle by construction."""

    adjacency = _adjacency(nodes, edges)
    remaining = set(adjacency)
    degrees = {node_id: len(neighbors) for node_id, neighbors in adjacency.items()}
    queue = deque(node_id for node_id, degree in degrees.items() if degree < 2)
    while queue:
        node_id = queue.popleft()
        if node_id not in remaining:
            continue
        remaining.remove(node_id)
        for neighbor in adjacency[node_id]:
            if neighbor in remaining:
                degrees[neighbor] -= 1
                if degrees[neighbor] == 1:
                    queue.append(neighbor)
    if "start" not in remaining:
        raise ValueError("START is not on a closed loop")
    return remaining


def analytic_gold(scene: dict) -> str:
    nodes = scene["nodes"]
    edges = scene["edges"]
    cycle = _cycle_nodes(nodes, edges)
    winners = [
        node["fill"]
        for node in nodes
        if node["role"] == "checkpoint" and node["id"] in cycle
    ]
    if len(winners) != 1 or winners[0] not in COLORS:
        raise ValueError(f"expected one colored checkpoint on START's loop, found {winners}")
    return winners[0]


def margin(scene: dict) -> float:
    nodes = scene["nodes"]
    edges = scene["edges"]
    cycle = _cycle_nodes(nodes, edges)
    scores = sorted(
        (
            1.0 if node["id"] in cycle else 0.0
            for node in nodes
            if node["role"] == "checkpoint"
        ),
        reverse=True,
    )
    if len(scores) != 3:
        return 0.0
    return scores[0] - scores[1]


def _point_segment_distance(point: dict, left: dict, right: dict) -> float:
    vx, vy = right["x"] - left["x"], right["y"] - left["y"]
    wx, wy = point["x"] - left["x"], point["y"] - left["y"]
    length2 = vx * vx + vy * vy
    position = 0.0 if not length2 else max(0.0, min(1.0, (wx * vx + wy * vy) / length2))
    closest_x = left["x"] + position * vx
    closest_y = left["y"] + position * vy
    return math.hypot(point["x"] - closest_x, point["y"] - closest_y)


def is_quarantined(scene: dict) -> bool:
    nodes = scene["nodes"]
    edges = scene["edges"]
    try:
        adjacency = _adjacency(nodes, edges)
        if margin({"nodes": nodes, "edges": edges}) < 1.0:
            return True
        checkpoints = [node for node in nodes if node["role"] == "checkpoint"]
        if len(checkpoints) != 3 or any(len(adjacency[node["id"]]) != 2 for node in checkpoints):
            return True
        if len(adjacency.get("start", ())) != 2:
            return True
    except (KeyError, TypeError, ValueError):
        return True

    for index, left in enumerate(nodes):
        for right in nodes[index + 1 :]:
            clearance = math.hypot(left["x"] - right["x"], left["y"] - right["y"])
            if clearance - left["r"] - right["r"] < 12.0:
                return True
    by_id = {node["id"]: node for node in nodes}
    for left_id, right_id in edges:
        left, right = by_id[left_id], by_id[right_id]
        length = math.hypot(left["x"] - right["x"], left["y"] - right["y"])
        if not 42.0 <= length <= 145.0:
            return True
        for node in nodes:
            if node["id"] not in (left_id, right_id):
                if _point_segment_distance(node, left, right) < node["r"] + 10.0:
                    return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    del scene
    return []
