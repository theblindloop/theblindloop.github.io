"""Deterministic latent sampler, analytic arm, and Pillow renderer."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

IDENTITY_NAMES = ("cyan", "magenta", "green", "orange")
IDENTITY_RGB = (
    (16, 166, 190),
    (210, 54, 150),
    (45, 157, 82),
    (236, 133, 38),
)
ANSWER_RGB = {
    "red": (211, 57, 64),
    "blue": (54, 101, 190),
    "gold": (205, 156, 28),
}
CANVAS = (720, 680)


def _swap_outputs(mapping: list[int], a: int, b: int) -> list[int]:
    result = list(mapping)
    result[a], result[b] = result[b], result[a]
    return result


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    reference = list(range(4))
    rng.shuffle(reference)
    # Each distractor changes exactly two bindings, leaving a visible score gap of 2.
    offset = seed % 4
    d1 = _swap_outputs(reference, offset, (offset + 1) % 4)
    d2 = _swap_outputs(reference, (offset + 2) % 4, (offset + 3) % 4)
    correct_index = seed % 3
    options: list[list[int] | None] = [None, None, None]
    options[correct_index] = list(reference)
    slots = [i for i in range(3) if i != correct_index]
    options[slots[0]], options[slots[1]] = d1, d2

    # A 3x3 cycle counterbalances correct position against answer-border identity.
    answer_names = ["red", "blue", "gold"]
    shift = (seed // 3) % 3
    answer_names = answer_names[shift:] + answer_names[:shift]

    row_orders = []
    for _ in range(4):
        order = list(range(4))
        rng.shuffle(order)
        row_orders.append(order)
    return {
        "seed": int(seed),
        "reference_mapping": reference,
        "option_mappings": options,
        "option_border_colors": answer_names,
        "row_orders": row_orders,
    }


def _score(reference: list[int], option: list[int]) -> int:
    return sum(int(a == b) for a, b in zip(reference, option))


def analytic_gold(scene: dict) -> str:
    reference = scene["reference_mapping"]
    options = scene["option_mappings"]
    border_colors = scene["option_border_colors"]
    scores = [_score(reference, option) for option in options]
    best = max(scores)
    if scores.count(best) != 1:
        raise ValueError("non-unique best option")
    return border_colors[scores.index(best)]


def margin(scene: dict) -> float:
    scores = sorted(
        (_score(scene["reference_mapping"], option) for option in scene["option_mappings"]),
        reverse=True,
    )
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    reference = scene["reference_mapping"]
    options = scene["option_mappings"]
    if sorted(reference) != list(range(4)):
        return True
    if any(sorted(option) != list(range(4)) for option in options):
        return True
    scores = [_score(reference, option) for option in options]
    return scores.count(max(scores)) != 1 or margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int) -> ImageFont.ImageFont:
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf", size=size)
    except OSError:
        return ImageFont.load_default()


def _draw_card(
    draw: ImageDraw.ImageDraw,
    box: tuple[int, int, int, int],
    mapping: list[int],
    row_order: list[int],
    border_rgb: tuple[int, int, int],
    label: str,
    border_width: int,
    scale: int,
) -> None:
    x0, y0, x1, y1 = box
    draw.rounded_rectangle(box, radius=16, fill=(250, 250, 247), outline=border_rgb, width=border_width)
    draw.text(((x0 + x1) // 2, y0 + 17*scale), label, fill=(43, 46, 52), anchor="mm", font=_font(16*scale))
    rail_left = int(x0 + 0.27 * (x1 - x0))
    rail_right = int(x0 + 0.73 * (x1 - x0))
    top = y0 + 62*scale
    bottom = y1 - 28*scale
    ys = [round(top + i * (bottom - top) / 3) for i in range(4)]
    draw.line((rail_left, top - 11*scale, rail_left, bottom + 11*scale), fill=(73, 78, 86), width=5*scale)
    draw.line((rail_right, top - 11*scale, rail_right, bottom + 11*scale), fill=(73, 78, 86), width=5*scale)
    for y, left_identity in zip(ys, row_order):
        right_identity = mapping[left_identity]
        draw.line((rail_left, y, rail_right, y), fill=(194, 200, 204), width=3*scale)
        r = 13*scale
        draw.ellipse((rail_left-r, y-r, rail_left+r, y+r), fill=IDENTITY_RGB[left_identity], outline=(255,255,255), width=2*scale)
        draw.ellipse((rail_right-r, y-r, rail_right+r, y+r), fill=IDENTITY_RGB[right_identity], outline=(255,255,255), width=2*scale)


def render(scene: dict) -> Image.Image:
    # Every gold-bearing field is accessed here and is encoded in the raster.
    reference = scene["reference_mapping"]
    options = scene["option_mappings"]
    border_colors = scene["option_border_colors"]
    row_orders = scene["row_orders"]
    scale = 3
    image = Image.new("RGB", (CANVAS[0] * scale, CANVAS[1] * scale), (238, 241, 244))
    draw = ImageDraw.Draw(image)
    draw.text((CANVAS[0]*scale//2, 17*scale), "MATCH THE COMPLETE PAIRING MAP", fill=(43,46,52), anchor="mm", font=_font(20*scale))

    def scaled(box):
        return tuple(v * scale for v in box)

    # Draw using scaled local coordinates so the downsampled scene has clean antialiasing.
    _draw_card(draw, scaled((190, 42, 530, 292)), reference, row_orders[0], (60,65,73), "REFERENCE", 6*scale, scale)
    option_boxes = ((15, 345, 235, 655), (250, 345, 470, 655), (485, 345, 705, 655))
    for index, box in enumerate(option_boxes):
        _draw_card(
            draw,
            scaled(box),
            options[index],
            row_orders[index + 1],
            ANSWER_RGB[border_colors[index]],
            f"OPTION {index + 1}",
            7*scale,
            scale,
        )
    image = image.resize(CANVAS, Image.Resampling.LANCZOS)
    return image
