"""Independent image-only inverse for the aligned-row pairing world."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

IDENTITY_RGB = (
    (16, 166, 190),
    (210, 54, 150),
    (45, 157, 82),
    (236, 133, 38),
)
ANSWER_RGB = {
    "red": (211, 57, 64),
    "blue": (54, 101, 190),
    "gold": (205, 156, 28),
}


def _mask_near(arr: np.ndarray, rgb: tuple[int, int, int], tolerance: float = 34.0) -> np.ndarray:
    delta = arr.astype(np.int32) - np.array(rgb, dtype=np.int32)
    return (delta * delta).sum(axis=2) <= tolerance * tolerance


def _components(mask: np.ndarray, minimum: int) -> list[tuple[int, int, int, int, float, float, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y0, x0 in np.argwhere(mask):
        if seen[y0, x0]:
            continue
        queue = deque([(int(y0), int(x0))])
        seen[y0, x0] = True
        xs, ys = [], []
        while queue:
            y, x = queue.popleft()
            xs.append(x); ys.append(y)
            for yy, xx in ((y-1,x),(y+1,x),(y,x-1),(y,x+1)):
                if 0 <= yy < h and 0 <= xx < w and mask[yy,xx] and not seen[yy,xx]:
                    seen[yy,xx] = True
                    queue.append((yy,xx))
        if len(xs) >= minimum:
            found.append((min(xs), min(ys), max(xs), max(ys), sum(xs)/len(xs), sum(ys)/len(ys), len(xs)))
    return found


def _panel_mapping(points: list[tuple[int, float, float]]) -> tuple[int, ...]:
    if len(points) != 8:
        raise ValueError(f"panel has {len(points)} beads instead of 8")
    x_values = sorted(p[1] for p in points)
    split = (x_values[3] + x_values[4]) / 2.0
    left = sorted((p for p in points if p[1] < split), key=lambda p: p[2])
    right = sorted((p for p in points if p[1] >= split), key=lambda p: p[2])
    if len(left) != 4 or len(right) != 4:
        raise ValueError("could not split rail beads")
    row_gap = max(abs(a[2] - b[2]) for a, b in zip(left, right))
    if row_gap > 3.0:
        raise ValueError("rail endpoints do not share guide rows")
    result = [-1] * 4
    for lpoint, rpoint in zip(left, right):
        result[lpoint[0]] = rpoint[0]
    if sorted(result) != list(range(4)):
        raise ValueError("panel is not a color bijection")
    return tuple(result)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    border_boxes: list[tuple[str, tuple[int,int,int,int]]] = []
    for name, rgb in ANSWER_RGB.items():
        comps = _components(_mask_near(arr, rgb), minimum=600)
        for comp in comps:
            x0, y0, x1, y1, _cx, _cy, _n = comp
            if x1 - x0 > 150 and y1 - y0 > 220:
                border_boxes.append((name, (x0, y0, x1, y1)))
    if len(border_boxes) != 3:
        raise ValueError(f"expected three answer borders, found {len(border_boxes)}")

    beads: list[tuple[int, float, float]] = []
    for identity, rgb in enumerate(IDENTITY_RGB):
        comps = _components(_mask_near(arr, rgb), minimum=180)
        for comp in comps:
            x0, y0, x1, y1, cx, cy, n = comp
            if 15 <= x1-x0 <= 35 and 15 <= y1-y0 <= 35 and n < 1000:
                beads.append((identity, cx, cy))
    if len(beads) != 32:
        raise ValueError(f"expected 32 beads, found {len(beads)}")

    option_maps: list[tuple[str, tuple[int, ...]]] = []
    assigned: set[int] = set()
    for name, (x0,y0,x1,y1) in border_boxes:
        selected = []
        for index, point in enumerate(beads):
            if x0 < point[1] < x1 and y0 < point[2] < y1:
                selected.append(point); assigned.add(index)
        option_maps.append((name, _panel_mapping(selected)))
    reference_points = [point for index, point in enumerate(beads) if index not in assigned]
    reference = _panel_mapping(reference_points)
    matches = [name for name, mapping in option_maps if mapping == reference]
    if len(matches) != 1:
        raise ValueError(f"expected one matching map, found {matches}")
    return matches[0]
