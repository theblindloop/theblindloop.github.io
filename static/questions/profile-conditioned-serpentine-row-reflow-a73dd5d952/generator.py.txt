"""Analytic scene sampler and Pillow renderer.

The source grid stores an output bitmap in boustrophedon order.  A solver must
use the visible magenta start marker and the declared alternating row scan,
then reflow the recovered stream row-major.
"""

from __future__ import annotations

import random
from typing import Any

import numpy as np
from PIL import Image, ImageDraw

SYMBOLS = ("anchor", "boot", "chair", "sail")
TEMPLATES = {
    "anchor": (
        "...##...", "...##...", ".######.", "...##...",
        "...##...", ".#.##.#.", "..####..", "...##...",
    ),
    "boot": (
        ".##.....", ".##.....", ".##.....", ".##.....",
        ".##.....", ".##.##..", ".#####..", ".#####..",
    ),
    "chair": (
        ".##..##.", ".##..##.", ".##..##.", ".######.",
        "....##..", "....##..", "..##....", "........",
    ),
    "sail": (
        "...#....", "..##....", ".###....", "####....",
        "..##....", "..##....", ".####...", "######..",
    ),
}

INKS = ((22, 28, 38), (44, 24, 68), (18, 52, 66))
CYAN = (0, 174, 205)
MAGENTA = (224, 32, 126)
YELLOW = (246, 185, 30)
DISTRACTOR_COLORS = ((238, 105, 70), (68, 160, 92), (72, 105, 210))


def _template(name: str) -> np.ndarray:
    return np.array([[ch == "#" for ch in row] for row in TEMPLATES[name]], dtype=bool)


def _classification(mask: np.ndarray) -> tuple[str, float]:
    scored = sorted(
        ((64 - int(np.count_nonzero(mask != _template(name))), name) for name in SYMBOLS),
        reverse=True,
    )
    return scored[0][1], float(scored[0][0] - scored[1][0])


def _visible_output(scene: dict[str, Any]) -> np.ndarray:
    mask = _template(scene["target"]).copy()
    for index in scene["error_cells"]:
        row, col = divmod(int(index), 8)
        mask[row, col] = ~mask[row, col]
    return mask


def _source_grid(scene: dict[str, Any]) -> np.ndarray:
    # Column-major writing is inverted here: the first eight stream values are
    # the first output column, not the first output row.
    stream = _visible_output(scene).T.reshape(-1)
    source = stream.reshape(8, 8).copy()
    source[1::2] = source[1::2, ::-1]
    return source


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = SYMBOLS[int(seed) % len(SYMBOLS)]
    error_count = (int(seed) // 4) % 5
    candidates = list(range(64))
    rng.shuffle(candidates)
    error_cells: list[int] = []
    for index in candidates:
        trial = _template(target).copy()
        for prior in [*error_cells, index]:
            trial.reshape(-1)[prior] = ~trial.reshape(-1)[prior]
        winner, gap = _classification(trial)
        if winner == target and gap >= 4:
            error_cells.append(index)
        if len(error_cells) == error_count:
            break

    cell_size = (22, 24, 26)[(int(seed) // 12) % 3]
    offset_x = 86 + rng.randrange(0, 49)
    offset_y = 74 + rng.randrange(0, 49)
    distractors = []
    for i in range(2 + (int(seed) // 4) % 3):
        side_x = 34 + 330 * (i % 2)
        side_y = 80 + ((int(seed) * 37 + i * 83) % 270)
        distractors.append([side_x, side_y, 8 + (i + int(seed)) % 5, i % 3])
    return {
        "target": target,
        "cell_size": cell_size,
        "offset_x": offset_x,
        "offset_y": offset_y,
        "ink_index": (int(seed) // 4) % len(INKS),
        "distractors": distractors,
        "error_cells": sorted(error_cells),
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (420, 420), (249, 248, 244))
    draw = ImageDraw.Draw(image)
    for x, y, radius, color_index in sorted(scene["distractors"]):
        color = DISTRACTOR_COLORS[int(color_index) % len(DISTRACTOR_COLORS)]
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)

    grid = _source_grid(scene)
    cell = int(scene["cell_size"])
    x0, y0 = int(scene["offset_x"]), int(scene["offset_y"])
    border = 5
    draw.rectangle(
        (x0 - border, y0 - border, x0 + 8 * cell + border - 1, y0 + 8 * cell + border - 1),
        fill=CYAN,
    )
    draw.rectangle((x0, y0, x0 + 8 * cell - 1, y0 + 8 * cell - 1), fill=(242, 242, 238))
    ink = INKS[int(scene["ink_index"]) % len(INKS)]
    for row in range(8):
        for col in range(8):
            left, top = x0 + col * cell, y0 + row * cell
            draw.rectangle((left, top, left + cell - 1, top + cell - 1), outline=(190, 194, 194), width=1)
            if bool(grid[row, col]):
                draw.rectangle((left + 3, top + 3, left + cell - 4, top + cell - 4), fill=ink)

    marker_x = x0 - 19
    draw.ellipse((marker_x - 7, y0 + cell // 2 - 7, marker_x + 7, y0 + cell // 2 + 7), fill=MAGENTA)
    end_y = y0 + 7 * cell + cell // 2
    draw.ellipse((marker_x - 7, end_y - 7, marker_x + 7, end_y + 7), fill=YELLOW)
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["target"])


def margin(scene: dict) -> float:
    winner, gap = _classification(_visible_output(scene))
    return gap if winner == scene["target"] else -gap


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["distractors"] = list(reversed(scene["distractors"]))
    return [("reverse_distractor_storage", twin)]
