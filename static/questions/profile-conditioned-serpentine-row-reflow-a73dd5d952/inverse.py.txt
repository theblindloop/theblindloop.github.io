"""Independent image-only inverse for the serpentine scan world.

It locates the cyan frame and both endpoint fiducials from their pixels, samples
the framed logical cells, executes the declared alternating scan, and compares
the reflowed raster with independently duplicated public glyph definitions.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_NAMES = ("anchor", "boot", "chair", "sail")
_ROWS = {
    "anchor": ("...##...", "...##...", ".######.", "...##...", "...##...", ".#.##.#.", "..####..", "...##..."),
    "boot": (".##.....", ".##.....", ".##.....", ".##.....", ".##.....", ".##.##..", ".#####..", ".#####.."),
    "chair": (".##..##.", ".##..##.", ".##..##.", ".######.", "....##..", "....##..", "..##....", "........"),
    "sail": ("...#....", "..##....", ".###....", "####....", "..##....", "..##....", ".####...", "######.."),
}
_GLYPHS = {
    name: np.array([[value == "#" for value in row] for row in rows], dtype=bool)
    for name, rows in _ROWS.items()
}


def _color_mask(arr: np.ndarray, color: tuple[int, int, int], tolerance: int = 18) -> np.ndarray:
    delta = np.abs(arr.astype(np.int16) - np.array(color, dtype=np.int16))
    return np.max(delta, axis=2) <= tolerance


def _decode(image: Image.Image) -> tuple[np.ndarray, str, float]:
    arr = np.asarray(image.convert("RGB"))
    cyan = _color_mask(arr, (0, 174, 205))
    ys, xs = np.nonzero(cyan)
    if len(xs) < 100:
        raise ValueError("cyan frame is missing")
    left, right = int(xs.min()), int(xs.max())
    top, bottom = int(ys.min()), int(ys.max())
    outer_w, outer_h = right - left + 1, bottom - top + 1
    if abs(outer_w - outer_h) > 2 or not 170 <= outer_w <= 230:
        raise ValueError("cyan frame is ambiguous")

    middle_x = (left + right) // 2
    border = 0
    while top + border <= bottom and cyan[top + border, middle_x]:
        border += 1
    if not 3 <= border <= 8:
        raise ValueError("frame border cannot be measured")
    interior = outer_w - 2 * border
    if interior % 8 != 0:
        raise ValueError("frame does not contain eight equal columns")
    cell = interior // 8
    x0, y0 = left + border, top + border

    magenta = _color_mask(arr, (224, 32, 126), 24)
    yellow = _color_mask(arr, (246, 185, 30), 24)
    region_x0, region_x1 = max(0, left - 32), left
    start_band = magenta[y0 : y0 + cell, region_x0:region_x1]
    end_band = yellow[y0 + 7 * cell : y0 + 8 * cell, region_x0:region_x1]
    if int(start_band.sum()) < 80 or int(end_band.sum()) < 80:
        raise ValueError("start or end fiducial is missing")

    source = np.zeros((8, 8), dtype=bool)
    for row in range(8):
        for col in range(8):
            cx = x0 + col * cell + cell // 2
            cy = y0 + row * cell + cell // 2
            patch = arr[cy - 3 : cy + 4, cx - 3 : cx + 4].astype(np.float32)
            luminance = 0.2126 * patch[:, :, 0] + 0.7152 * patch[:, :, 1] + 0.0722 * patch[:, :, 2]
            source[row, col] = float(luminance.mean()) < 125.0

    stream: list[bool] = []
    for row in range(8):
        values = source[row] if row % 2 == 0 else source[row, ::-1]
        stream.extend(bool(value) for value in values)
    # The public rule fills one top-to-bottom output column at a time.
    output = np.array(stream, dtype=bool).reshape(8, 8).T
    scores = sorted(
        ((64 - int(np.count_nonzero(output != glyph)), name) for name, glyph in _GLYPHS.items()),
        reverse=True,
    )
    gap = float(scores[0][0] - scores[1][0])
    if scores[0][0] == scores[1][0] or gap < 4.0:
        raise ValueError(f"ambiguous glyph; runner-up gap={gap}")
    return output, scores[0][1], gap


def decision_with_margin(image: Image.Image) -> tuple[str, float]:
    _, decision, gap = _decode(image)
    return decision, gap


def decision_from_image(image: Image.Image) -> str:
    return _decode(image)[1]
