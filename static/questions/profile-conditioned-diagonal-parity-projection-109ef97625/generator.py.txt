"""Deterministic renderer for a declared diagonal-parity projection."""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw, ImageFont

SOURCE_N = 13
RESULT_N = 5
SOURCE_DARK_COUNT = 57
IMAGE_SIZE = (920, 720)


def _pattern(rows: tuple[str, ...]) -> np.ndarray:
    mask = np.array([[value == "1" for value in row] for row in rows], dtype=bool)
    assert mask.shape == (RESULT_N, RESULT_N)
    assert int(mask.sum()) == 9
    return mask


# Four equally inked benign abstract glyphs. Pairwise Hamming distance is at
# least eight cells, so the reference-matching boundary is visibly separated.
PATTERNS = (
    _pattern(("00100", "00100", "11111", "00100", "00100")),  # plus
    _pattern(("10001", "01010", "00100", "01010", "10001")),  # X
    _pattern(("00100", "01010", "10101", "01010", "00100")),  # diamond + center
    _pattern(("11111", "00100", "00100", "00100", "00100")),  # tee
)

PALETTES = (
    ((31, 48, 73), (239, 242, 237)),
    ((62, 43, 92), (243, 239, 231)),
    ((23, 74, 67), (236, 243, 239)),
    ((83, 42, 49), (244, 239, 234)),
)
BACKGROUNDS = ((250, 248, 242), (244, 249, 250), (249, 246, 251))
GRID_COLOR = (184, 193, 198)
CYAN = (27, 169, 194)
CORAL = (220, 72, 126)
GOLD = (232, 171, 45)
LABEL = (43, 48, 54)
CARD_FILL = (253, 252, 248)


def sample_scene(seed: int) -> dict:
    """Sample one JSON-serializable scene from an integer seed."""
    rng = random.Random((int(seed) << 1) ^ 0xD1A60A1)
    reference_order = list(range(4))
    rng.shuffle(reference_order)
    return {
        "pattern_id": int(seed) % 4,
        "reference_order": reference_order,
        "fill_seed": rng.randrange(1, 2**31 - 1),
        "source_x": rng.randint(44, 68),
        "source_y": rng.randint(64, 88),
        "cell_size": rng.randint(28, 31),
        "reference_x": rng.randint(566, 584),
        "reference_y": rng.randint(112, 132),
        "palette_index": rng.randrange(len(PALETTES)),
        "canvas_variant": rng.randrange(len(BACKGROUNDS)),
    }


def _font(size: int) -> ImageFont.ImageFont:
    try:
        return ImageFont.load_default(size=size)
    except TypeError:
        return ImageFont.load_default()


def diagonal_cells(index: int) -> list[tuple[int, int]]:
    """Cells in detector bin order: bottom-left diagonal through top-right."""
    if not 0 <= index < 2 * SOURCE_N - 1:
        raise IndexError(index)
    difference = (SOURCE_N - 1) - index  # row - column
    return [
        (row, column)
        for row in range(SOURCE_N)
        for column in range(SOURCE_N)
        if row - column == difference
    ]


def source_mask(scene: dict) -> np.ndarray:
    """Make a noisy, fixed-ink source whose full-diagonal parities form the glyph."""
    pattern = PATTERNS[int(scene["pattern_id"]) % len(PATTERNS)].reshape(-1)
    if int(pattern.sum()) != 9:
        raise ValueError("detector glyph must contain exactly nine dark bits")
    rng = random.Random(int(scene["fill_seed"]) ^ 0x51D3A6)
    cells_by_diagonal = [diagonal_cells(index) for index in range(25)]
    counts = [int(bit) for bit in pattern]

    # Adding pairs cannot alter a diagonal's parity. Selecting 24 pair slots
    # gives 9 + 2*24 = 57 dark tiles for every class and nuisance sample.
    pair_slots: list[int] = []
    for index, cells in enumerate(cells_by_diagonal):
        pair_slots.extend([index] * ((len(cells) - counts[index]) // 2))
    rng.shuffle(pair_slots)
    if len(pair_slots) < (SOURCE_DARK_COUNT - 9) // 2:
        raise ValueError("insufficient parity-preserving fill capacity")
    for index in pair_slots[: (SOURCE_DARK_COUNT - 9) // 2]:
        counts[index] += 2

    mask = np.zeros((SOURCE_N, SOURCE_N), dtype=bool)
    for index, cells in enumerate(cells_by_diagonal):
        shuffled = list(cells)
        rng.shuffle(shuffled)
        for row, column in shuffled[: counts[index]]:
            mask[row, column] = True
    assert int(mask.sum()) == SOURCE_DARK_COUNT
    recovered = np.array(
        [int(mask[tuple(zip(*cells))].sum()) % 2 for cells in cells_by_diagonal],
        dtype=bool,
    )
    assert np.array_equal(recovered, pattern)
    return mask


def _draw_source(
    scene: dict,
    image: Image.Image,
    foreground: tuple[int, int, int],
    empty: tuple[int, int, int],
) -> None:
    draw = ImageDraw.Draw(image)
    x = int(scene["source_x"])
    y = int(scene["source_y"])
    cell = int(scene["cell_size"])
    border = 6
    side = SOURCE_N * cell
    draw.rectangle((x, y, x + side + 2 * border - 1, y + side + 2 * border - 1), fill=CYAN)
    mask = source_mask(scene)
    for row in range(SOURCE_N):
        for column in range(SOURCE_N):
            x0 = x + border + column * cell
            y0 = y + border + row * cell
            color = foreground if mask[row, column] else empty
            draw.rectangle(
                (x0, y0, x0 + cell - 1, y0 + cell - 1),
                fill=color,
                outline=GRID_COLOR,
                width=1,
            )

    # Visible gold guides indicate both the NW-SE diagonal direction and the
    # bottom-left -> top-right enumeration without touching cell centers.
    arrow_x = x + side + 2 * border + 18
    arrow_y = y + side // 2 - 34
    draw.line((arrow_x, arrow_y, arrow_x + 46, arrow_y + 46), fill=GOLD, width=7)
    draw.polygon(
        ((arrow_x + 46, arrow_y + 46), (arrow_x + 27, arrow_y + 41), (arrow_x + 41, arrow_y + 27)),
        fill=GOLD,
    )
    start = (x - 17, y + side + border + 17)
    finish = (x + side + border + 17, y - 17)
    draw.ellipse((start[0] - 7, start[1] - 7, start[0] + 7, start[1] + 7), fill=GOLD)
    draw.ellipse((finish[0] - 7, finish[1] - 7, finish[0] + 7, finish[1] + 7), outline=GOLD, width=4)


def _draw_references(
    scene: dict,
    image: Image.Image,
    foreground: tuple[int, int, int],
    empty: tuple[int, int, int],
) -> None:
    draw = ImageDraw.Draw(image)
    origin_x = int(scene["reference_x"])
    origin_y = int(scene["reference_y"])
    order = list(scene["reference_order"])
    cell = 18
    border = 5
    side = RESULT_N * cell
    x_gap = 136
    y_gap = 188
    font = _font(31)
    for card_index in range(4):
        card_row, card_column = divmod(card_index, 2)
        x = origin_x + card_column * x_gap
        y = origin_y + card_row * y_gap
        draw.rounded_rectangle(
            (x - 16, y - 47, x + side + 2 * border + 16, y + side + 2 * border + 16),
            radius=9,
            fill=CARD_FILL,
            outline=(207, 211, 210),
            width=2,
        )
        draw.text(
            (x + (side + 2 * border) / 2, y - 26),
            "ABCD"[card_index],
            fill=LABEL,
            anchor="mm",
            font=font,
        )
        draw.rectangle((x, y, x + side + 2 * border - 1, y + side + 2 * border - 1), fill=CORAL)
        pattern = PATTERNS[int(order[card_index]) % len(PATTERNS)]
        for row in range(RESULT_N):
            for column in range(RESULT_N):
                x0 = x + border + column * cell
                y0 = y + border + row * cell
                color = foreground if pattern[row, column] else empty
                draw.rectangle(
                    (x0, y0, x0 + cell - 1, y0 + cell - 1),
                    fill=color,
                    outline=GRID_COLOR,
                    width=1,
                )


def render(scene: dict) -> Image.Image:
    """Render one RGB scene deterministically."""
    foreground, empty = PALETTES[int(scene["palette_index"]) % len(PALETTES)]
    background = BACKGROUNDS[int(scene["canvas_variant"]) % len(BACKGROUNDS)]
    image = Image.new("RGB", IMAGE_SIZE, background)
    _draw_source(scene, image, foreground, empty)
    _draw_references(scene, image, foreground, empty)
    return image


def analytic_gold(scene: dict) -> str:
    pattern_id = int(scene["pattern_id"])
    reference_order = list(scene["reference_order"])
    return "ABCD"[reference_order.index(pattern_id)]


def margin(scene: dict) -> float:
    chosen = PATTERNS[int(scene["pattern_id"]) % len(PATTERNS)]
    distances = sorted(int(np.count_nonzero(chosen != pattern)) for pattern in PATTERNS)
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["palette_index"] = int(scene["palette_index"]) + len(PALETTES)
    return [("palette_period", twin)]
