"""Independent PNG-only inverse arm for the compass pointer decision."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _masks(image: Image.Image) -> tuple[np.ndarray, np.ndarray]:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    red = (pixels[..., 0] > 130) & (pixels[..., 0] > pixels[..., 1] * 1.55) & (pixels[..., 0] > pixels[..., 2] * 1.45)
    blue = (pixels[..., 2] > 80) & (pixels[..., 2] > pixels[..., 0] * 1.35) & (pixels[..., 2] > pixels[..., 1] * 1.12)
    return red, blue


def decision_from_image(image: Image.Image) -> str:
    red, blue = _masks(image)
    red_y, red_x = np.nonzero(red)
    blue_y, blue_x = np.nonzero(blue)
    if red_x.size < 450 or blue_x.size < 180:
        raise ValueError("required red pointer or blue hub evidence is absent")
    hub_x = float(np.median(blue_x))
    hub_y = float(np.median(blue_y))
    scores = {
        "top": hub_y - float(red_y.min()),
        "right": float(red_x.max()) - hub_x,
        "bottom": float(red_y.max()) - hub_y,
        "left": hub_x - float(red_x.min()),
    }
    ranked = sorted(scores, key=scores.get, reverse=True)
    if scores[ranked[0]] - scores[ranked[1]] < 25.0:
        raise ValueError("pointer direction is ambiguous at pixel resolution")
    return ranked[0]
