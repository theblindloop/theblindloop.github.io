"""Deterministic compass renderer with balanced prior-consistent/counterfactual scenes."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

SIZE = 384
SCALE = 3
EDGES = ("top", "right", "bottom", "left")
BASE_ANGLE = {"right": 0.0, "bottom": 90.0, "left": 180.0, "top": 270.0}


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed ^ 0xC04A55)
    return {
        "pointer_edge": EDGES[seed % 4],
        "angle_jitter_deg": round(rng.uniform(-7.5, 7.5), 4),
        "center_dx": rng.randint(-15, 15),
        "center_dy": rng.randint(-15, 15),
        "radius": rng.randint(126, 146),
        "rose_rotation_deg": round(rng.uniform(-10.0, 10.0), 4),
        "frame_gray": "#3c4650" if rng.randrange(2) else [60, 70, 80],
    }


def _point(cx: float, cy: float, distance: float, degrees: float) -> tuple[float, float]:
    radians = math.radians(degrees)
    return cx + distance * math.cos(radians), cy + distance * math.sin(radians)


def _polygon_arrow(cx: float, cy: float, radius: float, degrees: float) -> list[tuple[float, float]]:
    tip = _point(cx, cy, radius * 0.86, degrees)
    shoulder_a = _point(*_point(cx, cy, radius * 0.57, degrees), radius * 0.13, degrees - 90)
    neck_a = _point(*_point(cx, cy, radius * 0.49, degrees), radius * 0.065, degrees - 90)
    base_a = _point(cx, cy, radius * 0.10, degrees - 90)
    base_b = _point(cx, cy, radius * 0.10, degrees + 90)
    neck_b = _point(*_point(cx, cy, radius * 0.49, degrees), radius * 0.065, degrees + 90)
    shoulder_b = _point(*_point(cx, cy, radius * 0.57, degrees), radius * 0.13, degrees + 90)
    return [base_a, neck_a, shoulder_a, tip, shoulder_b, neck_b, base_b]


def render(scene: dict) -> Image.Image:
    edge = scene["pointer_edge"]
    jitter = float(scene["angle_jitter_deg"])
    cx = (SIZE / 2 + float(scene["center_dx"])) * SCALE
    cy = (SIZE / 2 + float(scene["center_dy"])) * SCALE
    radius = float(scene["radius"]) * SCALE
    rose_rotation = float(scene["rose_rotation_deg"])
    frame_gray = scene["frame_gray"]
    if isinstance(frame_gray, list):
        frame_gray = tuple(frame_gray)

    image = Image.new("RGB", (SIZE * SCALE, SIZE * SCALE), (248, 247, 242))
    draw = ImageDraw.Draw(image)
    pale = (226, 238, 244)
    gray = frame_gray
    red = (202, 48, 52)
    blue = (32, 105, 158)

    box = (cx - radius, cy - radius, cx + radius, cy + radius)
    draw.ellipse(box, fill=pale, outline=gray, width=6 * SCALE)
    inner = radius * 0.88
    draw.ellipse((cx - inner, cy - inner, cx + inner, cy + inner), fill=(252, 252, 249), outline=gray, width=2 * SCALE)

    # Neutral rose and ticks: their orientation and count never encode the answer.
    for index in range(16):
        angle = rose_rotation + index * 22.5
        outer = _point(cx, cy, inner * 0.96, angle)
        tick_len = 10 * SCALE if index % 4 else 18 * SCALE
        inside = _point(cx, cy, inner * 0.96 - tick_len, angle)
        draw.line((inside, outer), fill=gray, width=(3 if index % 4 else 5) * SCALE)
    for angle in (rose_rotation, rose_rotation + 90, rose_rotation + 180, rose_rotation + 270):
        tip = _point(cx, cy, radius * 0.55, angle)
        left = _point(cx, cy, radius * 0.16, angle - 90)
        right = _point(cx, cy, radius * 0.16, angle + 90)
        draw.polygon((tip, left, (cx, cy), right), fill=(210, 214, 214), outline=gray)

    pointer_angle = BASE_ANGLE[edge] + jitter
    draw.polygon(_polygon_arrow(cx, cy, radius, pointer_angle), fill=red, outline=(138, 25, 31), width=2 * SCALE)
    hub_r = 14 * SCALE
    draw.ellipse((cx - hub_r, cy - hub_r, cx + hub_r, cy + hub_r), fill=blue, outline=(18, 61, 94), width=2 * SCALE)

    # A small upright N on the hub identifies the red shape semantically without
    # adding red pixels that could leak its direction to the inverse arm.
    font = ImageFont.load_default(size=12 * SCALE)
    label = "N"
    bounds = draw.textbbox((0, 0), label, font=font)
    draw.text((cx - (bounds[2] - bounds[0]) / 2, cy - (bounds[3] - bounds[1]) / 2 - SCALE), label, fill="white", font=font)
    return image.resize((SIZE, SIZE), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["pointer_edge"])


def margin(scene: dict) -> float:
    return 45.0 - abs(float(scene["angle_jitter_deg"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    value = scene["frame_gray"]
    twin["frame_gray"] = [60, 70, 80] if isinstance(value, str) else "#3c4650"
    return [("equivalent_gray_encoding", twin)]
