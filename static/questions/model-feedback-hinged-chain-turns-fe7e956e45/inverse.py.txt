"""Independent image-only inverse for the moving-hinge chain folds."""

from __future__ import annotations

import numpy as np
from PIL import Image

EXPECTED_SIZE = (1100, 650)
GRID = 11
JOINTS = 7
INITIAL_X, INITIAL_Y, INITIAL_CELL = 38, 78, 24
CARD_XS, CARD_Y = (350, 590, 830), 78
OPTION_XS, OPTION_Y, OPTION_CELL = (10, 228, 446, 664, 882), 444, 15
COLORS = np.array(
    [(220, 62, 67), (231, 126, 34), (213, 170, 0), (39, 156, 90), (0, 144, 177), (62, 105, 210), (142, 72, 190)],
    dtype=np.float32,
)
INK = np.array((32, 54, 74), dtype=np.float32)
BLUE = np.array((21, 101, 192), dtype=np.float32)


class OracleError(ValueError):
    pass


def _valid(points: list[tuple[int, int]]) -> bool:
    return (
        len(points) == JOINTS
        and len(set(points)) == JOINTS
        and all(0 <= row < GRID and 0 <= col < GRID for row, col in points)
        and all(abs(a[0] - b[0]) + abs(a[1] - b[1]) == 1 for a, b in zip(points, points[1:]))
    )


def _chain(array: np.ndarray, x: int, y: int, cell: int) -> list[tuple[int, int]]:
    span = GRID * cell
    crop = array[y : y + span, x : x + span].astype(np.float32)
    result: list[tuple[int, int]] = []
    minimum = 120 if cell >= 20 else 35
    for color in COLORS:
        mask = np.linalg.norm(crop - color, axis=2) < 8.0
        ys, xs = np.nonzero(mask)
        if len(xs) < minimum:
            raise OracleError("a colored joint is missing or damaged")
        cx, cy = float(xs.mean()), float(ys.mean())
        col = int(round((cx - cell / 2) / cell))
        row = int(round((cy - cell / 2) / cell))
        expected_x, expected_y = col * cell + cell / 2, row * cell + cell / 2
        if abs(cx - expected_x) > 1.6 or abs(cy - expected_y) > 1.6:
            raise OracleError("a joint is not centered on a grid site")
        result.append((row, col))
    if not _valid(result):
        raise OracleError("recovered joints do not form one ordered unit-step chain")
    return result


def _card(array: np.ndarray, x: int) -> dict[str, object]:
    cx, cy = x + 110, CARD_Y + 127
    disk = array[cy - 19 : cy + 20, cx - 19 : cx + 20].astype(np.float32)
    counts = [int((np.linalg.norm(disk - color, axis=2) < 8.0).sum()) for color in COLORS]
    pivot = int(np.argmax(counts))
    if counts[pivot] < 500 or sorted(counts)[-2] > 20 or pivot not in range(1, JOINTS - 1):
        raise OracleError("a turn card has no unique internal colored pivot")
    crop = array[cy - 61 : cy + 62, cx - 61 : cx + 62].astype(np.float32)
    mask = np.linalg.norm(crop - BLUE, axis=2) < 8.0
    ys, xs = np.nonzero(mask)
    if len(xs) < 750 or np.ptp(xs) < 6 or np.ptp(ys) < 6:
        raise OracleError("a fold card has no full blue axis")
    x0 = xs.astype(np.float64) - xs.mean()
    y0 = ys.astype(np.float64) - ys.mean()
    var_x, var_y = float(np.mean(x0 * x0)), float(np.mean(y0 * y0))
    covariance = float(np.mean(x0 * y0))
    if var_x < 0.06 * var_y:
        axis = "vertical"
    elif var_y < 0.06 * var_x:
        axis = "horizontal"
    else:
        correlation = covariance / max((var_x * var_y) ** 0.5, 1e-9)
        if correlation > 0.93:
            axis = "falling"
        elif correlation < -0.93:
            axis = "rising"
        else:
            raise OracleError("a blue fold axis has no supported orientation")
    return {"pivot": pivot, "axis": axis}


def _turn(points: list[tuple[int, int]], instruction: dict[str, object]) -> list[tuple[int, int]]:
    result = list(points)
    pivot_index = int(instruction["pivot"])
    pr, pc = result[pivot_index]
    for index in range(pivot_index + 1, JOINTS):
        row, col = result[index]
        dr, dc = row - pr, col - pc
        if instruction["axis"] == "vertical":
            ndr, ndc = dr, -dc
        elif instruction["axis"] == "horizontal":
            ndr, ndc = -dr, dc
        elif instruction["axis"] == "falling":
            ndr, ndc = dc, dr
        else:
            ndr, ndc = -dc, -dr
        result[index] = (pr + ndr, pc + ndc)
    return result


def _validate_labels(array: np.ndarray) -> None:
    for x in OPTION_XS:
        cx, cy = x + GRID * OPTION_CELL // 2, 416
        patch = array[cy - 18 : cy + 19, cx - 18 : cx + 19].astype(np.float32)
        ink = int((np.linalg.norm(patch - INK, axis=2) < 8.0).sum())
        white = int((patch.mean(axis=2) > 245).sum())
        if ink < 700 or white < 20:
            raise OracleError("an option label is missing")


def decision_from_image(image: Image.Image) -> str:
    if image.size != EXPECTED_SIZE:
        raise OracleError(f"unexpected raster size {image.size}")
    array = np.asarray(image.convert("RGB"))
    _validate_labels(array)
    state = _chain(array, INITIAL_X, INITIAL_Y, INITIAL_CELL)
    turns = [_card(array, x) for x in CARD_XS]
    for instruction in turns:
        state = _turn(state, instruction)
    if not _valid(state):
        raise OracleError("visible turn program leaves the declared board or overlaps the chain")
    options = [_chain(array, x, OPTION_Y, OPTION_CELL) for x in OPTION_XS]
    if len({tuple(value) for value in options}) != 5:
        raise OracleError("answer options are not distinct")
    matches = [index for index, value in enumerate(options) if value == state]
    if len(matches) != 1:
        raise OracleError("the recovered final chain does not match exactly one option")
    return "ABCDE"[matches[0]]
