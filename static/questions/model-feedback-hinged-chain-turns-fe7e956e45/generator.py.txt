"""Deterministic articulated-chain fold sampler, renderer, and analytic program."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1100, 650
GRID = 11
JOINTS = 7
INITIAL_X, INITIAL_Y, INITIAL_CELL = 38, 78, 24
CARD_XS, CARD_Y = (350, 590, 830), 78
OPTION_XS, OPTION_Y, OPTION_CELL = (10, 228, 446, 664, 882), 444, 15
BACKGROUND = "#ffffff"
INK = "#20364a"
PAPER = "#faf8f1"
GRID_INK = "#d7e0e5"
BORDER = "#758794"
COLORS = ("#dc3e43", "#e77e22", "#d5aa00", "#279c5a", "#0090b1", "#3e69d2", "#8e48be")
BLUE = "#1565c0"
FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
AXES = ("vertical", "horizontal", "rising", "falling")


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(FONT_PATH, size=size)


def _valid_positions(positions: Iterable[tuple[int, int]]) -> bool:
    points = list(positions)
    return (
        len(points) == JOINTS
        and len(set(points)) == JOINTS
        and all(0 <= row < GRID and 0 <= col < GRID for row, col in points)
        and all(abs(a[0] - b[0]) + abs(a[1] - b[1]) == 1 for a, b in zip(points, points[1:]))
    )


def _positions(raw: object) -> list[tuple[int, int]]:
    if not isinstance(raw, list) or len(raw) != JOINTS:
        raise ValueError("a chain must contain exactly seven joints")
    points: list[tuple[int, int]] = []
    for pair in raw:
        if not isinstance(pair, list) or len(pair) != 2:
            raise ValueError("joint positions must be coordinate pairs")
        points.append((int(pair[0]), int(pair[1])))
    if not _valid_positions(points):
        raise ValueError("chain must be a self-avoiding unit-step path inside the board")
    return points


def _turns(raw: object) -> list[dict[str, object]]:
    if not isinstance(raw, list) or len(raw) != 3:
        raise ValueError("exactly three turns are required")
    result: list[dict[str, object]] = []
    for item in raw:
        if not isinstance(item, dict) or set(item) != {"pivot", "axis"}:
            raise ValueError("each fold needs exactly a pivot and axis")
        pivot = int(item["pivot"])
        axis = str(item["axis"])
        if pivot not in range(1, JOINTS - 1) or axis not in AXES:
            raise ValueError("pivot or axis is outside the supported set")
        result.append({"pivot": pivot, "axis": axis})
    return result


def _parse_scene(scene: dict) -> tuple[list[tuple[int, int]], list[dict[str, object]], list[list[tuple[int, int]]]]:
    if set(scene) != {"initial_positions", "turns", "options"}:
        raise ValueError("scene fields must be exactly initial_positions, turns, and options")
    initial = _positions(scene["initial_positions"])
    turns = _turns(scene["turns"])
    raw_options = scene["options"]
    if not isinstance(raw_options, list) or len(raw_options) != 5:
        raise ValueError("exactly five options are required")
    options = [_positions(value) for value in raw_options]
    if len({tuple(value) for value in options}) != 5:
        raise ValueError("answer options must be distinct")
    return initial, turns, options


def _apply_turn(
    positions: Iterable[tuple[int, int]], turn: dict[str, object], *, suffix: bool = True
) -> list[tuple[int, int]]:
    state = list(positions)
    pivot_index = int(turn["pivot"])
    pivot_row, pivot_col = state[pivot_index]
    indices = range(pivot_index + 1, JOINTS) if suffix else range(pivot_index)
    for index in indices:
        row, col = state[index]
        dr, dc = row - pivot_row, col - pivot_col
        if turn["axis"] == "vertical":
            ndr, ndc = dr, -dc
        elif turn["axis"] == "horizontal":
            ndr, ndc = -dr, dc
        elif turn["axis"] == "falling":
            ndr, ndc = dc, dr
        else:
            ndr, ndc = -dc, -dr
        state[index] = (pivot_row + ndr, pivot_col + ndc)
    return state


def _run(
    initial: Iterable[tuple[int, int]], turns: Iterable[dict[str, object]], *, suffix: bool = True
) -> list[tuple[int, int]]:
    state = list(initial)
    for turn in turns:
        state = _apply_turn(state, turn, suffix=suffix)
    return state


def _serialise(positions: Iterable[tuple[int, int]]) -> list[list[int]]:
    return [[int(row), int(col)] for row, col in positions]


def _alternate_states(
    initial: list[tuple[int, int]], turns: list[dict[str, object]]
) -> list[list[tuple[int, int]]]:
    candidates: list[list[tuple[int, int]]] = []
    state = list(initial)
    stages = [state]
    for turn in turns:
        state = _apply_turn(state, turn)
        stages.append(state)
    candidates.append(stages[-2])
    variants: list[list[dict[str, object]]] = [list(reversed(turns))]
    variants.extend(turns[index + 1 :] + turns[: index + 1] for index in range(2))
    for index in range(3):
        changed = [dict(item) for item in turns]
        pairs = {"vertical": "horizontal", "horizontal": "vertical", "rising": "falling", "falling": "rising"}
        changed[index]["axis"] = pairs[str(changed[index]["axis"])]
        variants.append(changed)
    for variant in variants:
        candidates.append(_run(initial, variant))
    candidates.append(_run(initial, turns, suffix=False))
    for index in range(3):
        state = list(initial)
        for turn_index, turn in enumerate(turns):
            state = _apply_turn(state, turn, suffix=turn_index != index)
        candidates.append(state)
    return candidates


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    moves = ((0, 1), (1, 0), (0, -1), (-1, 0))
    for _ in range(4000):
        initial = [(rng.randint(3, 7), rng.randint(3, 7))]
        while len(initial) < JOINTS:
            choices = list(moves)
            rng.shuffle(choices)
            for dr, dc in choices:
                point = (initial[-1][0] + dr, initial[-1][1] + dc)
                if 1 <= point[0] < GRID - 1 and 1 <= point[1] < GRID - 1 and point not in initial:
                    initial.append(point)
                    break
            else:
                break
        if len(initial) != JOINTS:
            continue
        first_pivot = rng.randint(1, 3)
        pivots = [first_pivot, rng.randint(first_pivot + 1, 5), rng.randint(1, 5)]
        axes = rng.sample(AXES, 3)
        turns = [
            {"pivot": pivot, "axis": axis}
            for pivot, axis in zip(pivots, axes)
        ]
        state = list(initial)
        stages = [state]
        for turn in turns:
            state = _apply_turn(state, turn)
            stages.append(state)
        target = stages[-1]
        if not all(_valid_positions(value) for value in stages):
            continue
        if len({tuple(value) for value in stages}) != 4:
            continue
        if stages[1][pivots[1]] == initial[pivots[1]]:
            continue
        reverse_result = _run(initial, reversed(turns))
        if not _valid_positions(reverse_result) or reverse_result == target:
            continue
        distractors: list[list[tuple[int, int]]] = []
        for value in _alternate_states(initial, turns):
            if not _valid_positions(value) or value == target or value in distractors:
                continue
            if sum(left != right for left, right in zip(value, target)) < 2:
                continue
            distractors.append(value)
        if len(distractors) < 4:
            continue
        rng.shuffle(distractors)
        options = distractors[:4]
        rng.shuffle(options)
        options.insert(int(seed) % 5, target)
        return {
            "initial_positions": _serialise(initial),
            "turns": turns,
            "options": [_serialise(value) for value in options],
        }
    raise RuntimeError("could not sample a nondegenerate moving-hinge program")


def analytic_gold(scene: dict) -> str:
    initial, turns, options = _parse_scene(scene)
    target = _run(initial, turns)
    matches = [index for index, value in enumerate(options) if value == target]
    if len(matches) != 1:
        raise ValueError("final chain must match exactly one option")
    return "ABCDE"[matches[0]]


def margin(scene: dict) -> float:
    initial, turns, options = _parse_scene(scene)
    target = _run(initial, turns)
    distances = [
        sum(left != right for left, right in zip(target, value))
        for value in options
        if value != target
    ]
    return float(min(distances)) if distances else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        initial, turns, options = _parse_scene(scene)
        target = _run(initial, turns)
        return sum(value == target for value in options) != 1 or margin(scene) < 2.0
    except (TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    _parse_scene(scene)
    return []


def _centre(x: int, y: int, cell: int, point: tuple[int, int]) -> tuple[int, int]:
    row, col = point
    return x + col * cell + cell // 2, y + row * cell + cell // 2


def _draw_grid(draw: ImageDraw.ImageDraw, x: int, y: int, cell: int) -> None:
    span = GRID * cell
    draw.rectangle((x, y, x + span, y + span), fill=PAPER, outline=BORDER, width=2)
    for step in range(1, GRID):
        draw.line((x + step * cell, y, x + step * cell, y + span), fill=GRID_INK, width=1)
        draw.line((x, y + step * cell, x + span, y + step * cell), fill=GRID_INK, width=1)


def _draw_chain(
    draw: ImageDraw.ImageDraw,
    positions: list[tuple[int, int]],
    x: int,
    y: int,
    cell: int,
    radius: int,
    digit_size: int,
) -> None:
    _draw_grid(draw, x, y, cell)
    centres = [_centre(x, y, cell, point) for point in positions]
    draw.line(centres, fill=INK, width=max(4, radius // 2 + 2), joint="curve")
    for index, ((cx, cy), color) in enumerate(zip(centres, COLORS), 1):
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=color, outline="white", width=2)
        digit_color = INK if index == 3 else "white"
        draw.text((cx, cy), str(index), fill=digit_color, font=_font(digit_size), anchor="mm")


def _draw_card(draw: ImageDraw.ImageDraw, x: int, number: int, turn: dict[str, object]) -> None:
    draw.rounded_rectangle((x, CARD_Y, x + 220, CARD_Y + 260), radius=13, fill="#f8fafb", outline=GRID_INK, width=3)
    draw.text((x + 110, CARD_Y + 25), f"FOLD {number}", fill=BORDER, font=_font(19), anchor="mm")
    cx, cy = x + 110, CARD_Y + 127
    axis = str(turn["axis"])
    if axis == "vertical":
        endpoints = (cx, cy - 58, cx, cy + 58)
        axis_text = "VERTICAL"
    elif axis == "horizontal":
        endpoints = (cx - 58, cy, cx + 58, cy)
        axis_text = "HORIZONTAL"
    elif axis == "falling":
        endpoints = (cx - 48, cy - 48, cx + 48, cy + 48)
        axis_text = "FALLING  ↘"
    else:
        endpoints = (cx - 48, cy + 48, cx + 48, cy - 48)
        axis_text = "RISING  ↗"
    draw.line(endpoints, fill=BLUE, width=10)
    pivot = int(turn["pivot"])
    draw.ellipse((cx - 18, cy - 18, cx + 18, cy + 18), fill=COLORS[pivot], outline="white", width=3)
    draw.text((cx, cy), str(pivot + 1), fill=INK if pivot == 2 else "white", font=_font(17), anchor="mm")
    draw.text((cx, CARD_Y + 215), axis_text, fill=INK, font=_font(15), anchor="mm")
    draw.text((cx, CARD_Y + 239), "FLIP END SIDE", fill=BORDER, font=_font(13), anchor="mm")


def render(scene: dict) -> Image.Image:
    initial, turns, options = _parse_scene(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((550, 20), "HOLD START SIDE  •  FLIP END SIDE  •  LEFT TO RIGHT", fill=INK, font=_font(22), anchor="mm")
    draw.text((170, 52), "1 START  →  7 END", fill=BORDER, font=_font(19), anchor="mm")
    _draw_chain(draw, initial, INITIAL_X, INITIAL_Y, INITIAL_CELL, 10, 12)
    for number, (x, turn) in enumerate(zip(CARD_XS, turns), 1):
        _draw_card(draw, x, number, turn)
    draw.line((26, 366, 1074, 366), fill=GRID_INK, width=2)
    draw.text((550, 389), "FINAL CHAIN", fill=BORDER, font=_font(20), anchor="mm")
    for label, positions, x in zip("ABCDE", options, OPTION_XS):
        centre = (x + GRID * OPTION_CELL // 2, 416)
        draw.ellipse((centre[0] - 20, centre[1] - 20, centre[0] + 20, centre[1] + 20), fill=INK)
        draw.text(centre, label, fill="white", font=_font(23), anchor="mm")
        _draw_chain(draw, positions, x, OPTION_Y, OPTION_CELL, 6, 8)
    return image
