"""Renderer and analytic arm for the isolated-cell reveal world."""

from __future__ import annotations

import random
from copy import deepcopy
from PIL import Image, ImageDraw

N = 12
SYMBOLS = ("anchor", "boot", "chair", "sail")

# Equal-ink (nine-dot) 5x5 glyphs, embedded with one blank cell between dots.
GLYPHS = {
    "anchor": ("00100", "00100", "10101", "01110", "00100"),
    "boot":   ("10000", "10000", "10000", "11110", "01100"),
    "chair":  ("10010", "10010", "11110", "00010", "00000"),
    "sail":   ("10000", "11000", "10100", "10010", "11000"),
}

PALETTES = (
    {"background": "#ffffff", "light": "#f3eee3", "dark": "#18212b", "frame": "#08a9c4", "grid": "#ffffff"},
    {"background": "#fbfbf8", "light": "#eee9dd", "dark": "#24303a", "frame": "#00a4c0", "grid": "#ffffff"},
    {"background": "#ffffff", "light": "#f1eadf", "dark": "#111c25", "frame": "#10aec7", "grid": "#ffffff"},
)


def _target(symbol: str) -> list[list[int]]:
    out = [[0] * N for _ in range(N)]
    for vr, row in enumerate(GLYPHS[symbol]):
        for vc, bit in enumerate(row):
            if bit == "1":
                out[1 + 2 * vr][1 + 2 * vc] = 1
    return out


def _isolated(source: list[list[int]]) -> list[list[int]]:
    out = [[0] * N for _ in range(N)]
    for r in range(N):
        for c in range(N):
            if not source[r][c]:
                continue
            neighbor = any(
                source[rr][cc]
                for rr in range(max(0, r - 1), min(N, r + 2))
                for cc in range(max(0, c - 1), min(N, c + 2))
                if (rr, cc) != (r, c)
            )
            out[r][c] = int(not neighbor)
    return out


def _distance(a: list[list[int]], b: list[list[int]]) -> int:
    return sum(a[r][c] != b[r][c] for r in range(N) for c in range(N))


def _classify(mask: list[list[int]]) -> tuple[str, float]:
    scores = sorted((_distance(mask, _target(name)), name) for name in SYMBOLS)
    return scores[0][1], float(scores[1][0] - scores[0][0])


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    symbol = SYMBOLS[int(seed) % len(SYMBOLS)]
    target = _target(symbol)
    source = deepcopy(target)

    # Add seven nuisance dominoes. Every nuisance cell has a dark neighbor, and
    # candidates are kept one king-move away from target dots.
    allowed = {
        (r, c)
        for r in range(N)
        for c in range(N)
        if not any(
            target[rr][cc]
            for rr in range(max(0, r - 1), min(N, r + 2))
            for cc in range(max(0, c - 1), min(N, c + 2))
        )
    }
    pairs = []
    for r, c in allowed:
        for dr, dc in ((0, 1), (1, 0)):
            if (r + dr, c + dc) in allowed:
                pairs.append(((r, c), (r + dr, c + dc)))
    rng.shuffle(pairs)
    used = set()
    chosen = []
    for pair in pairs:
        if not (set(pair) & used):
            chosen.append(pair)
            used.update(pair)
        if len(chosen) == 7:
            break
    if len(chosen) < 7:
        raise RuntimeError("insufficient nuisance positions")
    for pair in chosen:
        for r, c in pair:
            source[r][c] = 1

    cell_size = rng.choice((27, 28, 29, 30))
    span = N * cell_size
    frame_x = rng.randint(52, 512 - 52 - span)
    frame_y = rng.randint(52, 512 - 52 - span)
    return {
        "symbol": symbol,
        "source_grid": source,
        "frame_x": frame_x,
        "frame_y": frame_y,
        "cell_size": cell_size,
        "palette": deepcopy(rng.choice(PALETTES)),
        "alias_tag": rng.choice(("a", "b")),
    }


def render(scene: dict) -> Image.Image:
    palette = scene["palette"]
    grid = scene["source_grid"]
    x0, y0, cell = int(scene["frame_x"]), int(scene["frame_y"]), int(scene["cell_size"])
    _ = scene["symbol"]
    _ = scene["alias_tag"]  # two intentional, pixel-identical latent aliases
    image = Image.new("RGB", (512, 512), palette["background"])
    draw = ImageDraw.Draw(image)
    span = N * cell
    frame_w = 5
    draw.rectangle((x0 - frame_w, y0 - frame_w, x0 + span + frame_w - 1, y0 + span + frame_w - 1), fill=palette["frame"])
    for r in range(N):
        for c in range(N):
            left, top = x0 + c * cell, y0 + r * cell
            fill = palette["dark"] if grid[r][c] else palette["light"]
            draw.rectangle((left + 2, top + 2, left + cell - 3, top + cell - 3), fill=fill)
            # The two-pixel white gutters expose every cell boundary.
            draw.line((left, top, left + cell - 1, top), fill=palette["grid"], width=2)
            draw.line((left, top, left, top + cell - 1), fill=palette["grid"], width=2)
    return image


def analytic_gold(scene: dict) -> str:
    mask = _isolated(scene["source_grid"])
    return _classify(mask)[0]


def margin(scene: dict) -> float:
    mask = _isolated(scene["source_grid"])
    return _classify(mask)[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 3.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["alias_tag"] = "b" if scene["alias_tag"] == "a" else "a"
    return [("toggle_pixel_identical_alias", twin)]
