"""Independent PNG-only inverse for isolated-cell reveal."""

from __future__ import annotations

from PIL import Image

N = 12
NAMES = ("anchor", "boot", "chair", "sail")
PATTERNS = {
    "anchor": ("00100", "00100", "10101", "01110", "00100"),
    "boot":   ("10000", "10000", "10000", "11110", "01100"),
    "chair":  ("10010", "10010", "11110", "00010", "00000"),
    "sail":   ("10000", "11000", "10100", "10010", "11000"),
}


def _prototype(name: str) -> list[list[int]]:
    mask = [[0] * N for _ in range(N)]
    for i, row in enumerate(PATTERNS[name]):
        for j, bit in enumerate(row):
            if bit == "1":
                mask[1 + 2 * i][1 + 2 * j] = 1
    return mask


def _luma(rgb: tuple[int, int, int]) -> float:
    return 0.2126 * rgb[0] + 0.7152 * rgb[1] + 0.0722 * rgb[2]


def decision_from_image(image: Image.Image) -> str:
    im = image.convert("RGB")
    pix = im.load()
    cyan = []
    for y in range(im.height):
        for x in range(im.width):
            r, g, b = pix[x, y]
            if g > 115 and b > 125 and g > r * 2 and b > r * 1.6:
                cyan.append((x, y))
    if len(cyan) < 500:
        raise ValueError("cyan frame not found")
    minx, maxx = min(x for x, _ in cyan), max(x for x, _ in cyan)
    miny, maxy = min(y for _, y in cyan), max(y for _, y in cyan)
    outer_w, outer_h = maxx - minx + 1, maxy - miny + 1
    if abs(outer_w - outer_h) > 3:
        raise ValueError("frame is not square")
    # Renderer uses a five-pixel frame; infer cell pitch from its interior.
    x0, y0 = minx + 5, miny + 5
    cell = round((outer_w - 10) / N)
    if cell < 20 or abs(N * cell - (outer_w - 10)) > 2:
        raise ValueError("ambiguous grid pitch")

    source = [[0] * N for _ in range(N)]
    purities = []
    for row in range(N):
        for col in range(N):
            cx = x0 + col * cell + cell // 2
            cy = y0 + row * cell + cell // 2
            values = [_luma(pix[x, y]) for y in range(cy - 3, cy + 4) for x in range(cx - 3, cx + 4)]
            mean = sum(values) / len(values)
            purities.append(abs(mean - 140.0))
            source[row][col] = int(mean < 140.0)
    if min(purities) < 25:
        raise ValueError("ambiguous cell color")

    isolated = [[0] * N for _ in range(N)]
    for row in range(N):
        for col in range(N):
            if not source[row][col]:
                continue
            has_neighbor = any(
                source[rr][cc]
                for rr in range(max(0, row - 1), min(N, row + 2))
                for cc in range(max(0, col - 1), min(N, col + 2))
                if (rr, cc) != (row, col)
            )
            isolated[row][col] = int(not has_neighbor)

    ranked = []
    for name in NAMES:
        proto = _prototype(name)
        distance = sum(isolated[r][c] != proto[r][c] for r in range(N) for c in range(N))
        ranked.append((distance, name))
    ranked.sort()
    gap = ranked[1][0] - ranked[0][0]
    if ranked[0][0] > 1 or gap < 3:
        raise ValueError("glyph classification margin too small")
    return ranked[0][1]
