"""Independent inverse: final RGB pixels in, symbol decision out."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _reference_masks() -> dict[str, np.ndarray]:
    out = {name: np.zeros((10, 10), dtype=np.uint8) for name in ("ring", "plus", "X", "diamond")}
    m = out["ring"]
    m[2, 2:8] = m[7, 2:8] = 1; m[3:7, 2] = m[3:7, 7] = 1
    m = out["plus"]
    m[2:8, 4:6] = 1; m[4:6, 2:8] = 1
    m = out["X"]
    for r in range(2, 8):
        a, b = r, 9-r
        m[r, a] = m[r, b] = 1
        if r in (3, 4, 5, 6):
            m[r, a + (1 if a < 5 else -1)] = 1
            m[r, b + (1 if b < 5 else -1)] = 1
    m = out["diamond"]
    for r, cols in {2:(4,5), 3:(3,6), 4:(2,7), 5:(2,7), 6:(3,6), 7:(4,5)}.items():
        m[r, list(cols)] = 1
    return out


REFS = _reference_masks()


def _run_centers(values: np.ndarray) -> list[int]:
    indices = np.flatnonzero(values)
    if not len(indices): return []
    groups, current = [], [int(indices[0])]
    for value in indices[1:]:
        value = int(value)
        if value == current[-1] + 1: current.append(value)
        else: groups.append(current); current = [value]
    groups.append(current)
    return [int(round(sum(g) / len(g))) for g in groups]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    cyan = (arr[:,:,0] < 55) & (arr[:,:,1] > 145) & (arr[:,:,2] > 175)
    xscore = cyan.sum(axis=0)
    yscore = cyan.sum(axis=1)
    xs = _run_centers(xscore > 100)
    ys = _run_centers(yscore > 100)
    if len(xs) != 11 or len(ys) != 11:
        raise ValueError("could not recover eleven grid boundaries per axis")
    cells = np.zeros((10, 10), dtype=np.uint8)
    for r in range(10):
        for c in range(10):
            xa, xb = xs[c] + 3, xs[c+1] - 2
            ya, yb = ys[r] + 3, ys[r+1] - 2
            patch = arr[ya:yb, xa:xb]
            if patch.size == 0: raise ValueError("unstable narrow cell")
            # Ink is much darker than the near-white background for every palette.
            cells[r, c] = int(np.median(patch.mean(axis=2)) < 130)
    scored = sorted((int(np.count_nonzero(cells != ref)), name) for name, ref in REFS.items())
    gap = scored[1][0] - scored[0][0]
    if scored[0][0] > 2 or gap < 8:
        raise ValueError("ambiguous normalized mosaic")
    return scored[0][1]
