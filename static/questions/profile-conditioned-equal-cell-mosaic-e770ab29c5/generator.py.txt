"""Renderer for a nonuniform cell-mosaic rectification question."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

SIZE = 10
FRAME = (20, 185, 220)
SYMBOLS = ("ring", "plus", "X", "diamond")


def _templates() -> dict[str, list[list[int]]]:
    masks = {name: [[0] * SIZE for _ in range(SIZE)] for name in SYMBOLS}
    m = masks["ring"]
    for c in range(2, 8): m[2][c] = m[7][c] = 1
    for r in range(3, 7): m[r][2] = m[r][7] = 1
    m = masks["plus"]
    for r in range(2, 8):
        m[r][4] = m[r][5] = 1
    for c in range(2, 8):
        m[4][c] = m[5][c] = 1
    m = masks["X"]
    for r in range(2, 8):
        a = 2 + (r - 2)
        b = 7 - (r - 2)
        m[r][a] = m[r][b] = 1
        if r in (3, 4, 5, 6):
            m[r][min(9, a + (1 if a < 5 else -1))] = 1
            m[r][min(9, b + (1 if b < 5 else -1))] = 1
    m = masks["diamond"]
    for r, cols in {2:(4,5), 3:(3,6), 4:(2,7), 5:(2,7), 6:(3,6), 7:(4,5)}.items():
        for c in cols: m[r][c] = 1
    return masks


TEMPLATES = _templates()


def _parts(rng: random.Random) -> list[int]:
    values = [7, 9, 12, 16, 21, 27, 34, 42, 50, 62]
    rng.shuffle(values)
    return values


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    return {
        "symbol": SYMBOLS[seed % len(SYMBOLS)],
        "column_widths": _parts(rng),
        "row_heights": _parts(rng),
        "origin": [rng.randint(58, 84), rng.randint(56, 80)],
        "ink": rng.choice([[25, 29, 38], [54, 35, 67], [24, 55, 58], [61, 42, 29]]),
        "background": rng.choice([[248, 246, 239], [242, 247, 249], [249, 243, 247]]),
        "distractors": [[rng.randint(18, 440), rng.randint(18, 440), rng.randint(2, 5)] for _ in range(8)],
    }


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (480, 480), tuple(scene["background"]))
    draw = ImageDraw.Draw(img)
    x0, y0 = scene["origin"]
    widths = scene["column_widths"]
    heights = scene["row_heights"]
    mask = TEMPLATES[scene["symbol"]]
    xs = [x0]
    ys = [y0]
    for value in widths: xs.append(xs[-1] + value)
    for value in heights: ys.append(ys[-1] + value)
    for r in range(SIZE):
        for c in range(SIZE):
            color = tuple(scene["ink"]) if mask[r][c] else tuple(scene["background"])
            draw.rectangle((xs[c] + 2, ys[r] + 2, xs[c+1] - 2, ys[r+1] - 2), fill=color)
    for x in xs: draw.line((x, ys[0], x, ys[-1]), fill=FRAME, width=2)
    for y in ys: draw.line((xs[0], y, xs[-1], y), fill=FRAME, width=2)
    # Neutral exterior dots vary location but never enter the framed evidence.
    for x, y, radius in scene["distractors"]:
        if x0 - 10 <= x <= xs[-1] + 10 and y0 - 10 <= y <= ys[-1] + 10:
            continue
        draw.ellipse((x-radius, y-radius, x+radius, y+radius), fill=(155, 160, 164))
    return img


def analytic_gold(scene: dict) -> str:
    return scene["symbol"]


def margin(scene: dict) -> float:
    target = TEMPLATES[scene["symbol"]]
    distances = []
    for name, other in TEMPLATES.items():
        if name != scene["symbol"]:
            distances.append(sum(a != b for ra, rb in zip(target, other) for a, b in zip(ra, rb)))
    return float(min(distances))


def is_quarantined(scene: dict) -> bool:
    return min(scene["column_widths"] + scene["row_heights"]) < 6 or margin(scene) < 8


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
