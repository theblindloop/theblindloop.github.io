"""Independent pixel-only inverse for Shape–Texture Tag Transfer."""
from __future__ import annotations

import numpy as np
from PIL import Image

CARD = np.array((244, 240, 255), dtype=np.int16)
OBJECT = np.array((220, 238, 242), dtype=np.int16)
INK = np.array((35, 43, 54), dtype=np.int16)
TAG_RGB = {
    "red": np.array((218, 62, 67), dtype=np.int16),
    "blue": np.array((52, 108, 196), dtype=np.int16),
    "gold": np.array((224, 166, 45), dtype=np.int16),
}


def _runs(values: np.ndarray, minimum: int) -> list[tuple[int, int]]:
    idx = np.flatnonzero(values)
    if not len(idx):
        return []
    cuts = np.flatnonzero(np.diff(idx) > 1)
    starts = np.r_[0, cuts + 1]
    ends = np.r_[cuts, len(idx) - 1]
    return [(int(idx[a]), int(idx[b])) for a, b in zip(starts, ends) if idx[b] - idx[a] + 1 >= minimum]


def _find_cards(arr: np.ndarray) -> tuple[tuple[int, int, int, int], list[tuple[int, int, int, int]]]:
    card = np.linalg.norm(arr.astype(np.int16) - CARD, axis=2) < 4
    ybands = _runs(card.sum(axis=1) > 100, 150)
    if len(ybands) != 2:
        raise ValueError("could not detect target/reference card rows")
    boxes = []
    for ya, yb in ybands:
        xruns = _runs(card[ya:yb + 1].sum(axis=0) > 80, 180)
        boxes.append([(xa, ya, xb, yb) for xa, xb in xruns])
    if len(boxes[0]) != 1 or len(boxes[1]) != 3:
        raise ValueError("could not detect four card regions")
    return boxes[0][0], boxes[1]


def _decode_object(crop: np.ndarray) -> tuple[str, str]:
    obj = np.linalg.norm(crop.astype(np.int16) - OBJECT, axis=2) < 5
    ink = np.linalg.norm(crop.astype(np.int16) - INK, axis=2) < 5
    mask = obj | ink
    ys, xs = np.nonzero(mask)
    if len(xs) < 200:
        raise ValueError("missing shape")
    x0, x1, y0, y1 = xs.min(), xs.max(), ys.min(), ys.max()
    local = mask[y0:y1 + 1, x0:x1 + 1]
    area_ratio = float(local.mean())
    ly, lx = np.nonzero(local)
    cy = float(ly.mean()) / max(1, local.shape[0] - 1)
    if area_ratio > 0.88:
        shape = "square"
    elif area_ratio > 0.68:
        shape = "circle"
    elif cy > 0.56:
        shape = "triangle"
    else:
        shape = "diamond"

    pad = max(12, min(local.shape) // 5)
    core = ink[y0 + pad:y1 - pad + 1, x0 + pad:x1 - pad + 1]
    if core.size == 0:
        raise ValueError("empty texture core")
    row_peak = float(core.sum(axis=1).max()) / max(1, core.shape[1])
    col_peak = float(core.sum(axis=0).max()) / max(1, core.shape[0])
    if row_peak > 0.55 and col_peak > 0.55:
        pattern = "crosshatch"
    elif row_peak > 0.55:
        pattern = "horizontal"
    elif col_peak > 0.55:
        pattern = "vertical"
    else:
        pattern = "diagonal"
    return shape, pattern


def _decode_card(arr: np.ndarray, box: tuple[int, int, int, int]) -> dict[str, str]:
    x0, y0, x1, y1 = box
    w, h = x1 - x0 + 1, y1 - y0 + 1
    result = {}
    for col, row in ((0, 0), (1, 0), (0, 1), (1, 1)):
        xa = x0 + 10 + col * (w - 20) // 2
        xb = x0 + 10 + (col + 1) * (w - 20) // 2
        ya = y0 + 10 + row * (h - 20) // 2
        yb = y0 + 10 + (row + 1) * (h - 20) // 2
        shape, pattern = _decode_object(arr[ya:yb, xa:xb])
        if shape in result:
            raise ValueError("duplicate shape in card")
        result[shape] = pattern
    if len(result) != 4:
        raise ValueError("incomplete card")
    return result


def _read_tag(arr: np.ndarray, box: tuple[int, int, int, int]) -> str:
    x0, _, x1, y1 = box
    xa, xb = max(0, (x0 + x1) // 2 - 55), min(arr.shape[1], (x0 + x1) // 2 + 56)
    crop = arr[min(arr.shape[0], y1 + 8):, xa:xb].astype(np.int16)
    counts = {name: int((np.linalg.norm(crop - rgb, axis=2) < 6).sum()) for name, rgb in TAG_RGB.items()}
    name = max(counts, key=counts.get)
    if counts[name] < 300:
        raise ValueError("missing reference tag")
    return name


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    target_box, reference_boxes = _find_cards(arr)
    target = _decode_card(arr, target_box)
    references = [_decode_card(arr, box) for box in reference_boxes]
    scores = [sum(mapping.get(shape) == pattern for shape, pattern in target.items()) for mapping in references]
    ordered = sorted(scores, reverse=True)
    if ordered[0] != 4 or ordered[0] - ordered[1] < 2:
        return "abstain"
    winner = max(range(3), key=scores.__getitem__)
    return _read_tag(arr, reference_boxes[winner])
