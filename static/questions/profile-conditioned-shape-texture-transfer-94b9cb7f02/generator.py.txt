"""Renderer and analytic arm for Shape–Texture Tag Transfer."""
from __future__ import annotations

import copy
import itertools
import random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 900, 720
SHAPES = ("circle", "triangle", "square", "diamond")
PATTERNS = ("horizontal", "vertical", "diagonal", "crosshatch")
TAGS = ("red", "blue", "gold")
TAG_RGB = {"red": (218, 62, 67), "blue": (52, 108, 196), "gold": (224, 166, 45)}
BG = (255, 255, 255)
CARD = (244, 240, 255)
OBJECT = (220, 238, 242)
INK = (35, 43, 54)
TEAL = (37, 145, 142)


def _permutation_map(rng: random.Random) -> dict[str, str]:
    values = list(PATTERNS)
    rng.shuffle(values)
    return dict(zip(SHAPES, values))


def _placement(rng: random.Random) -> dict[str, int]:
    slots = list(range(4))
    rng.shuffle(slots)
    return dict(zip(SHAPES, slots))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = _permutation_map(rng)
    all_maps = [dict(zip(SHAPES, p)) for p in itertools.permutations(PATTERNS)]
    distractors = [m for m in all_maps if m != target]
    rng.shuffle(distractors)
    match_index = (int(seed) // 3) % 3
    references = [distractors[0], distractors[1]]
    references.insert(match_index, copy.deepcopy(target))

    desired_tag = TAGS[int(seed) % 3]
    remaining = [tag for tag in TAGS if tag != desired_tag]
    rng.shuffle(remaining)
    tag_order = remaining[:]
    tag_order.insert(match_index, desired_tag)
    return {
        "seed": int(seed),
        "target_map": target,
        "reference_maps": references,
        "placements": [_placement(rng) for _ in range(4)],
        "tag_order": tag_order,
    }


def _shape_mask(shape: str, size: int = 74) -> Image.Image:
    mask = Image.new("L", (size, size), 0)
    d = ImageDraw.Draw(mask)
    pad = 4
    if shape == "circle":
        d.ellipse((pad, pad, size - pad - 1, size - pad - 1), fill=255)
    elif shape == "square":
        d.rectangle((pad, pad, size - pad - 1, size - pad - 1), fill=255)
    elif shape == "triangle":
        d.polygon(((size // 2, pad), (size - pad - 1, size - pad - 1), (pad, size - pad - 1)), fill=255)
    elif shape == "diamond":
        d.polygon(((size // 2, pad), (size - pad - 1, size // 2), (size // 2, size - pad - 1), (pad, size // 2)), fill=255)
    else:
        raise ValueError(shape)
    return mask


def _draw_object(image: Image.Image, center: tuple[int, int], shape: str, pattern: str) -> None:
    size = 74
    x0, y0 = center[0] - size // 2, center[1] - size // 2
    mask = _shape_mask(shape, size)
    base = Image.new("RGB", (size, size), OBJECT)
    image.paste(base, (x0, y0), mask)

    texture = Image.new("RGB", (size, size), (0, 0, 0))
    tex_mask = Image.new("L", (size, size), 0)
    td = ImageDraw.Draw(tex_mask)
    if pattern in ("horizontal", "crosshatch"):
        for y in range(14, size - 8, 14):
            td.line((7, y, size - 8, y), fill=255, width=4)
    if pattern in ("vertical", "crosshatch"):
        for x in range(14, size - 8, 14):
            td.line((x, 7, x, size - 8), fill=255, width=4)
    if pattern == "diagonal":
        for offset in range(-size, size * 2, 15):
            td.line((offset, size - 1, offset + size, 0), fill=255, width=4)
    clipped = Image.new("L", (size, size), 0)
    import PIL.ImageChops
    clipped = PIL.ImageChops.multiply(tex_mask, mask)
    image.paste(Image.new("RGB", (size, size), INK), (x0, y0), clipped)

    outline = ImageDraw.Draw(image)
    p = 4
    box = (x0 + p, y0 + p, x0 + size - p - 1, y0 + size - p - 1)
    if shape == "circle":
        outline.ellipse(box, outline=INK, width=4)
    elif shape == "square":
        outline.rectangle(box, outline=INK, width=4)
    elif shape == "triangle":
        outline.polygon(((center[0], y0 + p), (x0 + size - p - 1, y0 + size - p - 1), (x0 + p, y0 + size - p - 1)), outline=INK, width=4)
    else:
        outline.polygon(((center[0], y0 + p), (x0 + size - p - 1, center[1]), (center[0], y0 + size - p - 1), (x0 + p, center[1])), outline=INK, width=4)


def _draw_card(image: Image.Image, box: tuple[int, int, int, int], mapping: dict[str, str], placement: dict[str, int], *, target: bool = False) -> None:
    d = ImageDraw.Draw(image)
    d.rounded_rectangle(box, radius=14, fill=CARD, outline=TEAL if target else INK, width=5)
    x0, y0, x1, y1 = box
    xs = (x0 + (x1 - x0) * 3 // 10, x0 + (x1 - x0) * 7 // 10)
    ys = (y0 + (y1 - y0) * 32 // 100, y0 + (y1 - y0) * 70 // 100)
    centers = ((xs[0], ys[0]), (xs[1], ys[0]), (xs[0], ys[1]), (xs[1], ys[1]))
    for shape, slot in placement.items():
        _draw_object(image, centers[int(slot)], shape, mapping[shape])


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    target_map = scene["target_map"]
    reference_maps = scene["reference_maps"]
    placements = scene["placements"]
    tag_order = scene["tag_order"]
    _draw_card(image, (270, 45, 630, 300), target_map, placements[0], target=True)
    ref_boxes = ((20, 365, 290, 635), (315, 365, 585, 635), (610, 365, 880, 635))
    for index, (box, mapping, tag) in enumerate(zip(ref_boxes, reference_maps, tag_order)):
        _draw_card(image, box, mapping, placements[index + 1])
        cx = (box[0] + box[2]) // 2
        ImageDraw.Draw(image).rounded_rectangle((cx - 45, 660, cx + 45, 700), radius=12, fill=TAG_RGB[tag], outline=INK, width=3)
    d = ImageDraw.Draw(image)
    font = ImageFont.load_default(size=22)
    d.text((450, 18), "TARGET", fill=INK, anchor="mm", font=font)
    d.text((450, 337), "TAGGED REFERENCES", fill=INK, anchor="mm", font=font)
    return image


def _score_references(scene: dict) -> list[int]:
    target = scene["target_map"]
    references = scene["reference_maps"]
    _ = scene["tag_order"]
    return [sum(mapping[shape] == target[shape] for shape in SHAPES) for mapping in references]


def analytic_gold(scene: dict) -> str:
    scores = _score_references(scene)
    tag_order = scene["tag_order"]
    return tag_order[max(range(3), key=scores.__getitem__)]


def margin(scene: dict) -> float:
    scores = sorted(_score_references(scene), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    scores = sorted(_score_references(scene), reverse=True)
    return scores[0] != 4 or scores[0] - scores[1] < 2


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["placements"] = [dict(reversed(list(p.items()))) for p in twin["placements"]]
    return [("reverse_nonoverlapping_draw_order", twin)]
