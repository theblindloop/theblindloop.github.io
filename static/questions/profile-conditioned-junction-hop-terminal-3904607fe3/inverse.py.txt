"""Independent image-only inverse arm for the hop-tree raster."""

from __future__ import annotations

from collections import deque
import math

import numpy as np
from PIL import Image

EDGE = np.array((38, 43, 52), dtype=np.int16)
ROOT = np.array((43, 170, 91), dtype=np.int16)
JUNCTION = np.array((76, 139, 202), dtype=np.int16)
TERMINALS = {
    "red": np.array((218, 64, 67), dtype=np.int16),
    "orange": np.array((235, 139, 45), dtype=np.int16),
    "purple": np.array((139, 82, 190), dtype=np.int16),
}


def _mask(arr: np.ndarray, color: np.ndarray, tolerance: int = 10) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - color), axis=2) <= tolerance


def _components(mask: np.ndarray, minimum: int = 40) -> list[tuple[float, float]]:
    work = mask.copy()
    centers: list[tuple[float, float]] = []
    height, width = work.shape
    while work.any():
        y0, x0 = np.argwhere(work)[0]
        queue = deque([(int(y0), int(x0))])
        work[y0, x0] = False
        pixels = []
        while queue:
            y, x = queue.popleft()
            pixels.append((y, x))
            for yy, xx in ((y-1, x), (y+1, x), (y, x-1), (y, x+1)):
                if 0 <= yy < height and 0 <= xx < width and work[yy, xx]:
                    work[yy, xx] = False
                    queue.append((yy, xx))
        if len(pixels) >= minimum:
            ys = [p[0] for p in pixels]
            xs = [p[1] for p in pixels]
            centers.append((float(sum(xs)) / len(xs), float(sum(ys)) / len(ys)))
    return centers


def _supported_line(edge_mask: np.ndarray, p: tuple[float, float], q: tuple[float, float],
                    blockers: list[tuple[float, float]]) -> bool:
    dx, dy = q[0] - p[0], q[1] - p[1]
    length = math.hypot(dx, dy)
    if length < 45:
        return False
    for r in blockers:
        rx, ry = r[0] - p[0], r[1] - p[1]
        t = (rx * dx + ry * dy) / (length * length)
        if 0.12 < t < 0.88:
            perpendicular = abs(rx * dy - ry * dx) / length
            if perpendicular < 17:
                return False
    samples = max(12, int(length) - 30)
    hits = 0
    height, width = edge_mask.shape
    for index in range(samples):
        distance = 16.0 + (length - 32.0) * index / max(1, samples - 1)
        x = p[0] + dx * distance / length
        y = p[1] + dy * distance / length
        xi, yi = int(round(x)), int(round(y))
        y0, y1 = max(0, yi-2), min(height, yi+3)
        x0, x1 = max(0, xi-2), min(width, xi+3)
        hits += bool(edge_mask[y0:y1, x0:x1].any())
    return hits / samples >= 0.92


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    edge_mask = _mask(arr, EDGE)
    roots = _components(_mask(arr, ROOT))
    junctions = _components(_mask(arr, JUNCTION))
    terminals: dict[str, tuple[float, float]] = {}
    for name, rgb in TERMINALS.items():
        parts = _components(_mask(arr, rgb))
        if len(parts) != 1:
            raise ValueError(f"expected one {name} terminal")
        terminals[name] = parts[0]
    if len(roots) != 1 or len(junctions) != 3:
        raise ValueError("expected one green start and three blue junctions")
    labels = ["root", *[f"j{i}" for i in range(3)], *terminals]
    points = [roots[0], *junctions, *[terminals[name] for name in terminals]]
    graph = {label: [] for label in labels}
    for i in range(len(points)):
        for j in range(i + 1, len(points)):
            blockers = [points[k] for k in range(len(points)) if k not in (i, j)]
            if _supported_line(edge_mask, points[i], points[j], blockers):
                graph[labels[i]].append(labels[j])
                graph[labels[j]].append(labels[i])
    distances = {"root": 0}
    queue = deque(["root"])
    while queue:
        node = queue.popleft()
        for neighbor in graph[node]:
            if neighbor not in distances:
                distances[neighbor] = distances[node] + 1
                queue.append(neighbor)
    matches = [name for name in terminals if distances.get(name) == 3]
    if len(matches) != 1:
        raise ValueError(f"could not isolate one three-hop terminal; graph={graph}")
    return matches[0]
