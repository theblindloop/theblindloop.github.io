"""Latent sampler, raster renderer, and analytic arm for the hop-tree world."""

from __future__ import annotations

import copy
import random
from collections import deque

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 512, 384
BACKGROUND = (247, 248, 250)
EDGE = (38, 43, 52)
JUNCTION = (76, 139, 202)
ROOT = (43, 170, 91)
WHITE = (255, 255, 255)
TERMINAL_RGB = {
    "red": (218, 64, 67),
    "orange": (235, 139, 45),
    "purple": (139, 82, 190),
}
TERMINAL_IDS = ("t_short", "t_mid", "t_long")


def _transform(x: int, y: int, mirror_x: bool, mirror_y: bool) -> tuple[int, int]:
    if mirror_x:
        x = WIDTH - x
    if mirror_y:
        y = HEIGHT - y
    return x, y


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    xs = [rng.randint(58, 76)]
    for low, high in ((136, 160), (244, 276), (360, 395)):
        xs.append(rng.randint(low, high))
    spine_y = rng.randint(174, 210)
    ys = [spine_y + rng.randint(-14, 14) for _ in range(4)]
    signs = [1, -1, 1]
    if rng.randrange(2):
        signs = [-v for v in signs]
    branch_nodes = []
    for index, (node_id, anchor) in enumerate(zip(TERMINAL_IDS, (1, 2, 3), strict=True)):
        ty = ys[anchor] + signs[index] * rng.randint(92, 122)
        tx = xs[anchor] + rng.randint(-22, 22)
        branch_nodes.append({"id": node_id, "x": tx, "y": ty, "kind": "terminal"})
    nodes = [
        {"id": "root", "x": xs[0], "y": ys[0], "kind": "root"},
        {"id": "a", "x": xs[1], "y": ys[1], "kind": "junction"},
        {"id": "b", "x": xs[2], "y": ys[2], "kind": "junction"},
        {"id": "c", "x": xs[3], "y": ys[3], "kind": "junction"},
        *branch_nodes,
    ]
    mirror_x = bool(rng.randrange(2))
    mirror_y = bool(rng.randrange(2))
    for node in nodes:
        node["x"], node["y"] = _transform(node["x"], node["y"], mirror_x, mirror_y)
    edges = [
        ["root", "a"], ["a", "b"], ["b", "c"],
        ["a", "t_short"], ["b", "t_mid"], ["c", "t_long"],
    ]
    permutations = (
        ("red", "orange", "purple"), ("red", "purple", "orange"),
        ("orange", "red", "purple"), ("orange", "purple", "red"),
        ("purple", "red", "orange"), ("purple", "orange", "red"),
    )
    palette_order = permutations[int(seed) % len(permutations)]
    terminal_colors = dict(zip(TERMINAL_IDS, palette_order, strict=True))
    return {"nodes": nodes, "edges": edges, "terminal_colors": terminal_colors}


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    positions = {node["id"]: (int(node["x"]), int(node["y"])) for node in scene["nodes"]}
    for left, right in scene["edges"]:
        draw.line([positions[left], positions[right]], fill=EDGE, width=7)
    terminal_colors = scene["terminal_colors"]
    for node in scene["nodes"]:
        x, y = positions[node["id"]]
        kind = node["kind"]
        radius = 15 if kind in {"root", "terminal"} else 11
        if kind == "root":
            fill = ROOT
        elif kind == "terminal":
            fill = TERMINAL_RGB[terminal_colors[node["id"]]]
        else:
            fill = JUNCTION
        draw.ellipse((x-radius-3, y-radius-3, x+radius+3, y+radius+3), fill=WHITE)
        draw.ellipse((x-radius, y-radius, x+radius, y+radius), fill=fill)
    return image


def _hop_distances(scene: dict) -> tuple[dict[str, int], dict[str, str]]:
    roots = [node["id"] for node in scene["nodes"] if node["kind"] == "root"]
    if len(roots) != 1:
        raise ValueError("scene must contain exactly one root node")
    root = roots[0]
    graph: dict[str, list[str]] = {}
    for left, right in scene["edges"]:
        graph.setdefault(left, []).append(right)
        graph.setdefault(right, []).append(left)
    distances = {root: 0}
    queue = deque([root])
    while queue:
        node = queue.popleft()
        for neighbor in graph.get(node, []):
            if neighbor not in distances:
                distances[neighbor] = distances[node] + 1
                queue.append(neighbor)
    return distances, scene["terminal_colors"]


def analytic_gold(scene: dict) -> str:
    distances, terminal_colors = _hop_distances(scene)
    matches = [color for node, color in terminal_colors.items() if distances.get(node) == 3]
    if len(matches) != 1:
        raise ValueError("scene must have exactly one terminal at three hops")
    return matches[0]


def margin(scene: dict) -> float:
    distances, terminal_colors = _hop_distances(scene)
    errors = sorted(abs(distances[node] - 3) for node in terminal_colors)
    if errors[0] != 0 or len(errors) < 2:
        return 0.0
    return float(errors[1])


def is_quarantined(scene: dict) -> bool:
    nodes = scene["nodes"]
    for index, left in enumerate(nodes):
        for right in nodes[index + 1:]:
            dx = float(left["x"] - right["x"])
            dy = float(left["y"] - right["y"])
            if (dx * dx + dy * dy) ** 0.5 < 38.0:
                return True
    try:
        return margin(scene) < 1.0
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    rename = {node["id"]: f"v_{index}" for index, node in enumerate(twin["nodes"])}
    for node in twin["nodes"]:
        node["id"] = rename[node["id"]]
    twin["edges"] = [[rename[left], rename[right]] for left, right in twin["edges"]]
    twin["terminal_colors"] = {rename[node]: color for node, color in twin["terminal_colors"].items()}
    return [("global_node_id_rename", twin)]
