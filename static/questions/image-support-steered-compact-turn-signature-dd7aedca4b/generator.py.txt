"""Renderer and analytic arm for the compact rotational-signature world.

The visible object is one filled radial contour.  A half-turn scene uses only
frequency components divisible by two; a third-turn scene uses only components
divisible by three.  A shared sixfold component gives both classes comparable
surface complexity while the class-specific component makes the two whole-shape
rotation residuals separable.
"""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw


RENDERER_VERSION = "compact_turn_signature-0.1.0"
WIDTH = 256
HEIGHT = 256
SUPERSAMPLE = 4

_SHAPE_COLORS = (
    (31, 67, 112),
    (145, 55, 35),
    (38, 105, 76),
    (104, 55, 125),
    (126, 82, 28),
)
_BACKGROUND_COLORS = (
    (250, 249, 245),
    (247, 249, 250),
    (252, 248, 242),
)


def sample_scene(seed: int) -> dict:
    """Sample one balanced, compact scene from an integer seed."""

    rng = random.Random(int(seed))
    # The parity assignment guarantees both answer classes in every stress run
    # while all nuisance fields remain independently randomized.
    symmetry_class = "half-turn" if int(seed) % 2 == 0 else "third-turn"
    scale = rng.uniform(34.0, 38.0)
    cx = rng.uniform(70.0, 186.0)
    cy = rng.uniform(70.0, 186.0)
    rotation = rng.uniform(0.0, 2.0 * math.pi)
    phase = rng.uniform(0.0, 2.0 * math.pi)
    coefficients = [
        rng.uniform(0.035, 0.060),
        rng.uniform(0.195, 0.225),
        rng.uniform(0.035, 0.060),
    ]
    return {
        "symmetry_class": symmetry_class,
        "cx": round(cx, 6),
        "cy": round(cy, 6),
        "scale": round(scale, 6),
        "rotation": round(rotation, 9),
        "radial_coefficients": [round(value, 9) for value in coefficients],
        "phase": round(phase, 9),
        "shape_color": list(rng.choice(_SHAPE_COLORS)),
        "background_color": list(rng.choice(_BACKGROUND_COLORS)),
        "width": WIDTH,
        "height": HEIGHT,
    }


def _radial_value(scene: dict, theta: float) -> float:
    coefficients = scene["radial_coefficients"]
    common6, signature, secondary = (float(value) for value in coefficients)
    u = theta - float(scene["rotation"]) + float(scene["phase"])
    if scene["symmetry_class"] == "half-turn":
        signature_frequency, secondary_frequency = 2, 4
    elif scene["symmetry_class"] == "third-turn":
        signature_frequency, secondary_frequency = 3, 9
    else:
        raise ValueError(f"unknown symmetry class: {scene['symmetry_class']!r}")
    return float(scene["scale"]) * (
        1.0
        + common6 * math.cos(6.0 * u)
        + signature * math.cos(signature_frequency * u)
        + secondary * math.sin(secondary_frequency * u + 0.37)
    )


def _contour_points(scene: dict, samples: int = 720) -> list[tuple[float, float]]:
    cx = float(scene["cx"])
    cy = float(scene["cy"])
    return [
        (
            cx + _radial_value(scene, 2.0 * math.pi * index / samples)
            * math.cos(2.0 * math.pi * index / samples),
            cy + _radial_value(scene, 2.0 * math.pi * index / samples)
            * math.sin(2.0 * math.pi * index / samples),
        )
        for index in range(samples)
    ]


def render(scene: dict) -> Image.Image:
    """Render only the compact emblem on a quiet background."""

    width = int(scene["width"])
    height = int(scene["height"])
    factor = SUPERSAMPLE
    background = tuple(int(value) for value in scene["background_color"])
    foreground = tuple(int(value) for value in scene["shape_color"])
    image = Image.new("RGB", (width * factor, height * factor), background)
    points = [(round(x * factor), round(y * factor)) for x, y in _contour_points(scene)]
    ImageDraw.Draw(image).polygon(points, fill=foreground)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def _profile(scene: dict, samples: int = 720) -> list[float]:
    return [_radial_value(scene, 2.0 * math.pi * i / samples) for i in range(samples)]


def _cyclic_rms(values: list[float], shift: int) -> float:
    size = len(values)
    return math.sqrt(
        sum((values[index] - values[(index + shift) % size]) ** 2 for index in range(size))
        / size
    )


def _analytic_residuals(scene: dict) -> tuple[float, float]:
    values = _profile(scene)
    return _cyclic_rms(values, len(values) // 2), _cyclic_rms(values, len(values) // 3)


def analytic_gold(scene: dict) -> str:
    """Return the label input; it is visibly realized by the contour renderer."""

    return str(scene["symmetry_class"])


def margin(scene: dict) -> float:
    """Return the analytic gap between the two whole-contour residuals."""

    half_residual, third_residual = _analytic_residuals(scene)
    return round(abs(half_residual - third_residual), 6)


def is_quarantined(scene: dict) -> bool:
    """Exclude finite-resolution ties from headline evidence."""

    return margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """There are no distinct pixel-identical aliases in this parametrization."""

    return []
