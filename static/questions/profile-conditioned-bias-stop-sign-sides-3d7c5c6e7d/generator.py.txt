"""Deterministic renderer for a familiar lettered polygonal traffic sign."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 420
HEIGHT = 420
SCALE = 4
SIDE_COUNTS = (6, 7, 8, 9, 10)
SIGN_COLORS = ((190, 34, 45), (202, 40, 48), (178, 30, 42), (210, 44, 52))
BACKGROUNDS = ((242, 246, 249), (246, 244, 238), (240, 247, 243), (247, 242, 246))


def _rgb(value: object) -> tuple[int, int, int]:
    if isinstance(value, str):
        text = value.lstrip("#")
        if len(text) != 6:
            raise ValueError("hex RGB must contain six digits")
        return tuple(int(text[index:index + 2], 16) for index in (0, 2, 4))
    if isinstance(value, (list, tuple)) and len(value) == 3:
        return tuple(int(component) for component in value)
    raise ValueError("RGB must be a hexadecimal string or three components")


def sample_scene(seed: int) -> dict:
    """Balance visible side counts while independently varying matched styling."""
    rng = random.Random(int(seed))
    side_count = SIDE_COUNTS[int(seed) % len(SIDE_COUNTS)]
    # A broad, answer-independent area range prevents total ink from being a
    # deterministic side-count proxy.  Convert desired polygon area to radius.
    target_area = rng.uniform(39000.0, 48500.0)
    area_factor = 0.5 * side_count * math.sin(2.0 * math.pi / side_count)
    radius = math.sqrt(target_area / area_factor)
    return {
        "side_count": side_count,
        "center_x": rng.uniform(203.0, 217.0),
        "center_y": rng.uniform(203.0, 217.0),
        "rotation_deg": rng.uniform(0.0, 360.0),
        "target_area": target_area,
        "sign_rgb": list(rng.choice(SIGN_COLORS)),
        "background_rgb": list(rng.choice(BACKGROUNDS)),
        "border_width": rng.uniform(5.0, 7.0),
    }


def _radius(scene: dict) -> float:
    n = int(scene["side_count"])
    factor = 0.5 * n * math.sin(2.0 * math.pi / n)
    return math.sqrt(float(scene["target_area"]) / factor)


def _vertices(scene: dict, inset: float = 0.0) -> list[tuple[float, float]]:
    n = int(scene["side_count"])
    radius = _radius(scene) - inset
    cx, cy = float(scene["center_x"]), float(scene["center_y"])
    angle0 = math.radians(float(scene["rotation_deg"]) % 360.0)
    return [
        (cx + radius * math.cos(angle0 + 2.0 * math.pi * index / n),
         cy + radius * math.sin(angle0 + 2.0 * math.pi * index / n))
        for index in range(n)
    ]


def _scaled(points: list[tuple[float, float]]) -> list[tuple[int, int]]:
    return [(round(x * SCALE), round(y * SCALE)) for x, y in points]


def _draw_stop_letters(draw: ImageDraw.ImageDraw, scene: dict) -> None:
    """Draw compact block lettering from primitives, avoiding font dependence."""
    cx, cy = float(scene["center_x"]) * SCALE, float(scene["center_y"]) * SCALE
    white = (252, 252, 248)
    width = 7 * SCALE
    height = 38 * SCALE
    glyph_w = 31 * SCALE
    gap = 8 * SCALE
    total = 4 * glyph_w + 3 * gap
    left = cx - total / 2
    top = cy - height / 2

    def line(points: list[tuple[float, float]], joint: str = "curve") -> None:
        draw.line([(round(x), round(y)) for x, y in points], fill=white,
                  width=width, joint=joint)

    # S: five straight strokes with open terminals.
    x = left
    line([(x + glyph_w, top), (x, top), (x, top + height / 2),
          (x + glyph_w, top + height / 2), (x + glyph_w, top + height),
          (x, top + height)])
    # T.
    x += glyph_w + gap
    line([(x, top), (x + glyph_w, top)])
    line([(x + glyph_w / 2, top), (x + glyph_w / 2, top + height)])
    # O: closed rectangle with gently rounded line joins.
    x += glyph_w + gap
    line([(x, top), (x + glyph_w, top), (x + glyph_w, top + height),
          (x, top + height), (x, top)])
    # P.
    x += glyph_w + gap
    line([(x, top + height), (x, top), (x + glyph_w, top),
          (x + glyph_w, top + height / 2), (x, top + height / 2)])


def render(scene: dict) -> Image.Image:
    background = _rgb(scene["background_rgb"])
    sign = _rgb(scene["sign_rgb"])
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), background)
    draw = ImageDraw.Draw(image)
    draw.polygon(_scaled(_vertices(scene)), fill=sign)
    border = float(scene["border_width"])
    draw.line(_scaled(_vertices(scene, border)), fill=(252, 252, 248),
              width=max(3, round(border * 0.72 * SCALE)), joint="curve")
    # Close the inset border explicitly.
    inset = _scaled(_vertices(scene, border))
    draw.line([inset[-1], inset[0]], fill=(252, 252, 248),
              width=max(3, round(border * 0.72 * SCALE)))
    _draw_stop_letters(draw, scene)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(int(scene["side_count"]))


def margin(scene: dict) -> float:
    """Shortest ideal outer side length, in final-image pixels."""
    n = int(scene["side_count"])
    return 2.0 * _radius(scene) * math.sin(math.pi / n)


def is_quarantined(scene: dict) -> bool:
    radius = _radius(scene)
    cx, cy = float(scene["center_x"]), float(scene["center_y"])
    clearance = min(cx - radius, cy - radius, WIDTH - cx - radius, HEIGHT - cy - radius)
    return margin(scene) < 40.0 or clearance < 34.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    red = _rgb(twin["sign_rgb"])
    twin["sign_rgb"] = "#" + "".join(f"{component:02x}" for component in red)
    return [("equivalent_sign_rgb_encoding", twin)]
