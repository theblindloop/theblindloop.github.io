"""Pixel-only inverse arm for the state-indexed rule table.

Receives a PIL image and nothing else.  It segments the exact mark colors,
recovers the rule table, the ring-marked starting state and the ordered strip of
black shapes by one-dimensional projections, then executes the machine itself.
It imports no renderer, prompt, generator, verifier or scene state.
"""

from __future__ import annotations

import numpy as np

_STATE_RGB = {
    "amber": (240, 170, 20),
    "teal": (20, 160, 160),
    "violet": (150, 60, 200),
    "crimson": (210, 40, 60),
}
_BLACK = (18, 18, 18)


def _mask(arr, rgb):
    return (
        (arr[:, :, 0] == rgb[0])
        & (arr[:, :, 1] == rgb[1])
        & (arr[:, :, 2] == rgb[2])
    )


def _runs(flags, min_gap=6):
    """Contiguous True runs of a 1-D boolean profile, merged across small gaps."""
    idx = np.flatnonzero(flags)
    if idx.size == 0:
        return []
    out = [[int(idx[0]), int(idx[0])]]
    for v in idx[1:]:
        v = int(v)
        if v - out[-1][1] <= min_gap:
            out[-1][1] = v
        else:
            out.append([v, v])
    return [tuple(r) for r in out]


def _classify(sub):
    """square / circle / triangle from the filled fraction of the bounding box."""
    ys, xs = np.nonzero(sub)
    h = ys.max() - ys.min() + 1
    w = xs.max() - xs.min() + 1
    ratio = float(sub.sum()) / float(h * w)
    if ratio >= 0.90:
        return "square"
    if ratio >= 0.63:
        return "circle"
    return "triangle"


def _glyphs_in_band(black, y0, y1):
    band = black[y0 : y1 + 1, :]
    out = []
    for x0, x1 in _runs(band.any(axis=0)):
        sub = band[:, x0 : x1 + 1]
        out.append(((x0 + x1) / 2.0, _classify(sub)))
    out.sort()
    return out


def decision_from_image(image):
    arr = np.asarray(image.convert("RGB"))

    color_masks = {n: _mask(arr, rgb) for n, rgb in _STATE_RGB.items()}
    any_color = np.zeros(arr.shape[:2], dtype=bool)
    for m in color_masks.values():
        any_color |= m
    black = _mask(arr, _BLACK)

    col_runs = _runs(any_color.any(axis=0))
    row_runs = _runs(any_color.any(axis=1))
    if len(col_runs) != 4 or len(row_runs) != 4:
        raise ValueError("unexpected disk layout: %d cols, %d rows"
                         % (len(col_runs), len(row_runs)))

    def disk_color(y0, y1, x0, x1):
        best, best_n = None, 0
        for name, m in color_masks.items():
            n = int(m[y0 : y1 + 1, x0 : x1 + 1].sum())
            if n > best_n:
                best, best_n = name, n
        if best is None:
            raise ValueError("no colored disk at that grid position")
        return best

    header_run = col_runs[0]
    cell_runs = col_runs[1:]
    row_states = [disk_color(y0, y1, *header_run) for y0, y1 in row_runs]

    # black y-bands: column headers on top, one ring in the table, strip below
    bands = _runs(black.any(axis=1))
    if len(bands) != 3:
        raise ValueError("expected 3 black bands, found %d" % len(bands))
    header_band, ring_band, tape_band = bands
    ring_cy = (ring_band[0] + ring_band[1]) / 2.0
    start_idx = min(
        range(4), key=lambda i: abs((row_runs[i][0] + row_runs[i][1]) / 2.0 - ring_cy)
    )
    start = row_states[start_idx]

    head = _glyphs_in_band(black, *header_band)
    if len(head) != 3:
        raise ValueError("expected 3 column headers, found %d" % len(head))
    col_shapes = []
    for x0, x1 in cell_runs:
        cx = (x0 + x1) / 2.0
        col_shapes.append(min(head, key=lambda g: abs(g[0] - cx))[1])
    if len(set(col_shapes)) != 3:
        raise ValueError("column headers are not three distinct shapes")

    tape = [s for _, s in _glyphs_in_band(black, *tape_band)]
    if not tape:
        raise ValueError("empty symbol strip")

    table = {}
    for i, (y0, y1) in enumerate(row_runs):
        table[row_states[i]] = {
            col_shapes[j]: disk_color(y0, y1, x0, x1)
            for j, (x0, x1) in enumerate(cell_runs)
        }

    cur = start
    for sym in tape:
        cur = table[cur][sym]
    return cur
