"""State-indexed rule table: a drawn transition relation plus a symbol strip.

The raster shows a rule table whose rows are labelled by state colors and whose
columns are labelled by black shapes.  A ring marks the starting state.  A strip
of black shapes at the bottom is consumed left to right; each shape moves the
machine from its current state to the color standing where that state's row
meets that shape's column.  The reported decision is the terminal state color.

Nothing about a shape's effect is fixed: the same drawn shape moves the machine
somewhere different depending on which row it is read in, so the strip cannot be
summarised, reordered or factored, and no intermediate state is drawn.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

# ---------------------------------------------------------------- constants

WIDTH = 1000
HEIGHT = 880

WHITE = (255, 255, 255)
BLACK = (18, 18, 18)

STATES = ("amber", "teal", "violet", "crimson")
STATE_RGB = {
    "amber": (240, 170, 20),
    "teal": (20, 160, 160),
    "violet": (150, 60, 200),
    "crimson": (210, 40, 60),
}
SHAPES = ("square", "circle", "triangle")

DISK_R = 38
RING_INNER_R = 52
RING_OUTER_R = 66
GLYPH_H = 60          # glyph half-extent is GLYPH_H // 2
GLYPH_HALF = 30
TRI_HALF_W = 34

HEADER_Y = 86
ROW_Y0 = 200
ROW_PITCH = 156
HEADER_COL_X = 190
CELL_X0 = 400
CELL_PITCH = 190
TAPE_Y = 796
TAPE_X0 = 130
TAPE_PITCH = 110
TAPE_LEN = 8

MIN_MARGIN = 11.0


# ---------------------------------------------------------------- sampling

def _simulate(table, start, tape):
    cur = start
    for sym in tape:
        cur = table[cur][sym]
    return cur


def _mode_cell_color(table):
    counts = {s: 0 for s in STATES}
    for row in STATES:
        for sym in SHAPES:
            counts[table[row][sym]] += 1
    best = max(counts.values())
    winners = [s for s in STATES if counts[s] == best]
    return winners[0] if len(winners) == 1 else None


def _trajectory(table, start, tape):
    cur = start
    out = [cur]
    for sym in tape:
        cur = table[cur][sym]
        out.append(cur)
    return out


def sample_scene(seed: int) -> dict:
    """Sample one latent scene.  Every field below is drawn in the raster."""
    rng = random.Random(seed * 7919 + 13)
    for _ in range(4000):
        table = {
            row: {sym: rng.choice(STATES) for sym in SHAPES} for row in STATES
        }
        # every row must actually depend on the symbol
        if any(len(set(table[row].values())) < 2 for row in STATES):
            continue
        # every symbol must actually depend on the row
        if any(len({table[row][sym] for row in STATES}) < 3 for sym in SHAPES):
            continue

        start = rng.choice(STATES)
        tape = [rng.choice(SHAPES) for _ in range(TAPE_LEN)]
        if any(tape.count(s) < 2 for s in SHAPES):
            continue

        final = _simulate(table, start, tape)
        traj = _trajectory(table, start, tape)

        # the walk must genuinely move around
        if len(set(traj)) < 3:
            continue
        if sum(1 for a, b in zip(traj, traj[1:]) if a != b) < 6:
            continue

        # every cheap shortcut must land somewhere else
        if final == start:
            continue
        if final == _simulate(table, start, list(reversed(tape))):
            continue
        if final == table[start][tape[-1]]:
            continue
        if final == table[start][tape[0]]:
            continue
        # ignoring the row dimension entirely (always reading the start row)
        if final == table[start][tape[len(tape) // 2]]:
            continue
        mode = _mode_cell_color(table)
        if mode is not None and final == mode:
            continue
        # dropping or duplicating the final step must not agree either
        if final == _simulate(table, start, tape[:-1]):
            continue

        rows = list(STATES)
        rng.shuffle(rows)
        cols = list(SHAPES)
        rng.shuffle(cols)

        jit = {
            "rows": [rng.randint(-4, 4) for _ in range(4)],
            "cols": [rng.randint(-4, 4) for _ in range(3)],
            "tape": [rng.randint(-4, 4) for _ in range(TAPE_LEN)],
            "header": rng.randint(-3, 3),
        }
        return {
            "seed": seed,
            "rows": rows,
            "cols": cols,
            "table": table,
            "start": start,
            "tape": tape,
            "jitter": jit,
        }
    raise RuntimeError("rejection sampling failed for seed %r" % (seed,))


# ---------------------------------------------------------------- geometry

def _row_y(scene, i):
    return ROW_Y0 + ROW_PITCH * i + scene["jitter"]["rows"][i]


def _col_x(scene, j):
    return CELL_X0 + CELL_PITCH * j + scene["jitter"]["cols"][j]


def _tape_x(scene, k):
    return TAPE_X0 + TAPE_PITCH * k + scene["jitter"]["tape"][k]


# ---------------------------------------------------------------- drawing

def _disk(draw, cx, cy, r, rgb):
    draw.ellipse([cx - r, cy - r, cx + r, cy + r], fill=rgb)


def _glyph(draw, cx, cy, shape):
    if shape == "square":
        draw.rectangle(
            [cx - GLYPH_HALF, cy - GLYPH_HALF, cx + GLYPH_HALF, cy + GLYPH_HALF],
            fill=BLACK,
        )
    elif shape == "circle":
        draw.ellipse(
            [cx - GLYPH_HALF, cy - GLYPH_HALF, cx + GLYPH_HALF, cy + GLYPH_HALF],
            fill=BLACK,
        )
    elif shape == "triangle":
        draw.polygon(
            [
                (cx - TRI_HALF_W, cy + GLYPH_HALF),
                (cx + TRI_HALF_W, cy + GLYPH_HALF),
                (cx, cy - GLYPH_HALF),
            ],
            fill=BLACK,
        )
    else:  # pragma: no cover - defensive
        raise ValueError(shape)


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), WHITE)
    draw = ImageDraw.Draw(img)

    hy = HEADER_Y + scene["jitter"]["header"]
    for j, shape in enumerate(scene["cols"]):
        _glyph(draw, _col_x(scene, j), hy, shape)

    for i, row_state in enumerate(scene["rows"]):
        cy = _row_y(scene, i)
        if row_state == scene["start"]:
            _disk(draw, HEADER_COL_X, cy, RING_OUTER_R, BLACK)
            _disk(draw, HEADER_COL_X, cy, RING_INNER_R, WHITE)
        _disk(draw, HEADER_COL_X, cy, DISK_R, STATE_RGB[row_state])
        for j, shape in enumerate(scene["cols"]):
            _disk(
                draw,
                _col_x(scene, j),
                cy,
                DISK_R,
                STATE_RGB[scene["table"][row_state][shape]],
            )

    for k, shape in enumerate(scene["tape"]):
        _glyph(draw, _tape_x(scene, k), TAPE_Y, shape)

    return img


# ---------------------------------------------------------------- contract

def analytic_gold(scene: dict) -> str:
    return _simulate(scene["table"], scene["start"], scene["tape"])


def margin(scene: dict) -> float:
    gaps = [
        float(RING_OUTER_R - RING_INNER_R),          # ring stroke thickness
        float(RING_INNER_R - DISK_R),                # white gap inside the ring
    ]
    rows = [_row_y(scene, i) for i in range(4)]
    for a, b in zip(rows, rows[1:]):
        gaps.append(float(b - a - 2 * DISK_R))
        gaps.append(float(b - a - RING_OUTER_R - DISK_R))
    cols = [_col_x(scene, j) for j in range(3)]
    for a, b in zip(cols, cols[1:]):
        gaps.append(float(b - a - 2 * DISK_R))
        gaps.append(float(b - a - 2 * TRI_HALF_W))
    gaps.append(float(cols[0] - HEADER_COL_X - RING_OUTER_R - DISK_R))
    taps = [_tape_x(scene, k) for k in range(TAPE_LEN)]
    for a, b in zip(taps, taps[1:]):
        gaps.append(float(b - a - 2 * TRI_HALF_W))
    hy = HEADER_Y + scene["jitter"]["header"]
    gaps.append(float(rows[0] - GLYPH_HALF - hy - GLYPH_HALF))
    gaps.append(float(TAPE_Y - GLYPH_HALF - rows[-1] - RING_OUTER_R))
    gaps.append(float(TAPE_X0 - TRI_HALF_W))
    gaps.append(float(HEIGHT - TAPE_Y - GLYPH_HALF))
    gaps.append(float(2 * GLYPH_HALF - 20))
    return float(min(gaps))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < MIN_MARGIN


def latent_symmetries(scene: dict) -> list:
    """Rendering is injective over sampled scenes; no nontrivial alias exists."""
    return []
