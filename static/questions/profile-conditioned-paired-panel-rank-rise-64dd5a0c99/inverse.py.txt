"""Independent raster-only inverse for paired-panel colored rank plots."""

from __future__ import annotations

import numpy as np
from PIL import Image

_PALETTE = {
    "red": (214, 55, 52),
    "blue": (42, 105, 190),
    "green": (35, 145, 78),
    "purple": (142, 68, 173),
}


def _color_centers(image: Image.Image) -> dict[str, tuple[float, float]]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    if arr.ndim != 3 or arr.shape[1] < 500:
        raise ValueError("image is not the declared two-panel raster")
    split = arr.shape[1] // 2
    centers: dict[str, tuple[float, float]] = {}
    for name, rgb in _PALETTE.items():
        delta = arr.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
        mask = np.sum(delta * delta, axis=2, dtype=np.int32) < 1800
        panel_y: list[float] = []
        for side in (mask[:, :split], mask[:, split:]):
            ys, xs = np.nonzero(side)
            if len(ys) < 180:
                raise ValueError(f"missing or damaged {name} point in a panel")
            panel_y.append(float(np.mean(ys)))
        centers[name] = (panel_y[0], panel_y[1])
    return centers


def decision_from_image(image: Image.Image) -> str:
    centers = _color_centers(image)
    colors = tuple(_PALETTE)
    order_a = sorted(colors, key=lambda color: centers[color][0])
    order_b = sorted(colors, key=lambda color: centers[color][1])
    rank_a = {color: index + 1 for index, color in enumerate(order_a)}
    rank_b = {color: index + 1 for index, color in enumerate(order_b)}
    scores = {color: rank_a[color] - rank_b[color] for color in colors}
    ordered = sorted(colors, key=lambda color: scores[color], reverse=True)
    if scores[ordered[0]] == scores[ordered[1]]:
        return "abstain"
    return ordered[0]
