"""Deterministic analytic scene and Pillow renderer for paired-panel ranks."""

from __future__ import annotations

import itertools
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 720, 420
COLORS = ("red", "blue", "green", "purple")
RGB = {
    "red": (214, 55, 52),
    "blue": (42, 105, 190),
    "green": (35, 145, 78),
    "purple": (142, 68, 173),
}
PANEL_CENTERS = (195, 525)
PLOT_TOP, PLOT_BOTTOM = 86, 350


def _rise_scores(scene: dict) -> dict[str, int]:
    colors = scene["colors"]
    rank_a = scene["rank_a"]
    rank_b = scene["rank_b"]
    return {color: int(a) - int(b) for color, a, b in zip(colors, rank_a, rank_b, strict=True)}


def _valid_rank_b(rank_a: list[int], winner_index: int) -> list[tuple[int, ...]]:
    valid: list[tuple[int, ...]] = []
    for rank_b in itertools.permutations(range(1, 5)):
        scores = [rank_a[i] - rank_b[i] for i in range(4)]
        ordered = sorted(range(4), key=lambda i: scores[i], reverse=True)
        if ordered[0] == winner_index and scores[ordered[0]] > scores[ordered[1]]:
            valid.append(rank_b)
    return valid


def _spaced_slots(rng: random.Random) -> list[int]:
    base = [112, 184, 256, 328]
    values = [value + rng.randint(-14, 14) for value in base]
    assert min(b - a for a, b in zip(values, values[1:])) >= 44
    return values


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    colors = list(COLORS)
    winner_index = int(seed) % len(colors)
    for _ in range(100):
        rank_a = list(range(1, 5))
        rng.shuffle(rank_a)
        choices = _valid_rank_b(rank_a, winner_index)
        if choices:
            break
    else:
        raise RuntimeError("could not construct a uniquely ranked scene")
    rank_b = list(choices[rng.randrange(len(choices))])
    return {
        "colors": colors,
        "rank_a": rank_a,
        "rank_b": rank_b,
        "y_slots_a": _spaced_slots(rng),
        "y_slots_b": _spaced_slots(rng),
        "x_jitter_a": [rng.randint(-46, 46) for _ in colors],
        "x_jitter_b": [rng.randint(-46, 46) for _ in colors],
    }


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    names = (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf" if bold
        else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
    )
    for name in names:
        try:
            return ImageFont.truetype(name, size)
        except OSError:
            pass
    return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    colors = scene["colors"]
    rank_a = scene["rank_a"]
    rank_b = scene["rank_b"]
    y_slots_a = scene["y_slots_a"]
    y_slots_b = scene["y_slots_b"]
    jitter_a = scene["x_jitter_a"]
    jitter_b = scene["x_jitter_b"]

    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), (250, 250, 248))
    draw = ImageDraw.Draw(image)
    title_font = _font(22 * scale, bold=True)
    label_font = _font(16 * scale)
    small_font = _font(13 * scale)

    def line(points: list[tuple[int, int]], fill: tuple[int, int, int], width: int) -> None:
        draw.line([(x * scale, y * scale) for x, y in points], fill=fill, width=width * scale)

    draw.text((WIDTH * scale // 2, 24 * scale), "Sample rank in two measurements", fill=(31, 35, 39), font=title_font, anchor="mm")
    for panel_index, (cx, panel_name) in enumerate(zip(PANEL_CENTERS, ("A", "B"), strict=True)):
        left, right = cx - 135, cx + 135
        line([(left, PLOT_TOP), (left, PLOT_BOTTOM)], (74, 80, 86), 2)
        line([(left, PLOT_BOTTOM), (right, PLOT_BOTTOM)], (74, 80, 86), 2)
        draw.text((cx * scale, 61 * scale), f"Panel {panel_name}", fill=(38, 42, 46), font=label_font, anchor="mm")
        for rank in range(1, 5):
            y = 98 + (rank - 1) * 78
            line([(left - 5, y), (left, y)], (74, 80, 86), 1)
            draw.text(((left - 12) * scale, y * scale), str(rank), fill=(70, 74, 79), font=small_font, anchor="rm")
        draw.text(((left + 9) * scale, 339 * scale), "rank", fill=(70, 74, 79), font=small_font, anchor="lm")

    for index, color in enumerate(colors):
        for panel_index, (ranks, slots, jitters) in enumerate(((rank_a, y_slots_a, jitter_a), (rank_b, y_slots_b, jitter_b))):
            cx = PANEL_CENTERS[panel_index] + int(jitters[index])
            cy = int(slots[int(ranks[index]) - 1])
            radius = 12
            box = ((cx - radius) * scale, (cy - radius) * scale, (cx + radius) * scale, (cy + radius) * scale)
            draw.ellipse(box, fill=RGB[color], outline=(255, 255, 255), width=2 * scale)

    line([(360, 72), (360, 365)], (205, 207, 207), 1)
    image = image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
    return image


def analytic_gold(scene: dict) -> str:
    scores = _rise_scores(scene)
    ordered = sorted(scores, key=lambda color: scores[color], reverse=True)
    if scores[ordered[0]] == scores[ordered[1]]:
        return "abstain"
    return ordered[0]


def margin(scene: dict) -> float:
    scores = sorted(_rise_scores(scene).values(), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    sep_a = min(b - a for a, b in zip(scene["y_slots_a"], scene["y_slots_a"][1:]))
    sep_b = min(b - a for a, b in zip(scene["y_slots_b"], scene["y_slots_b"][1:]))
    return margin(scene) < 1.0 or min(sep_a, sep_b) < 32


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
