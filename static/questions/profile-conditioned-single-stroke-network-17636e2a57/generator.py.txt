"""Deterministic renderer and analytic arm for a one-stroke graph question."""

from __future__ import annotations

import itertools
import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 420
HEIGHT = 420
GRID_EDGES = tuple(
    (3 * row + col, 3 * row + col + 1)
    for row in range(3)
    for col in range(2)
) + tuple(
    (3 * row + col, 3 * (row + 1) + col)
    for row in range(2)
    for col in range(3)
)

NODE_PALETTES = (
    (25, 126, 176),
    (24, 150, 112),
    (173, 75, 112),
    (128, 84, 184),
)
EDGE_PALETTES = ((38, 43, 52), (47, 43, 50), (35, 49, 48))
BACKGROUNDS = ((249, 248, 244), (247, 249, 250), (250, 247, 242))


def _degrees(edges: list[list[int]] | tuple[tuple[int, int], ...]) -> list[int]:
    degree = [0] * 9
    for left, right in edges:
        degree[int(left)] += 1
        degree[int(right)] += 1
    return degree


def _is_connected(edges: list[list[int]] | tuple[tuple[int, int], ...]) -> bool:
    adjacency = [set() for _ in range(9)]
    for left, right in edges:
        adjacency[int(left)].add(int(right))
        adjacency[int(right)].add(int(left))
    seen = {0}
    pending = [0]
    while pending:
        for neighbor in adjacency[pending.pop()]:
            if neighbor not in seen:
                seen.add(neighbor)
                pending.append(neighbor)
    return len(seen) == 9


def _odd_count_from_edges(edges: list[list[int]] | tuple[tuple[int, int], ...]) -> int:
    return sum(value % 2 for value in _degrees(edges))


def _graph_stats(scene: dict) -> tuple[bool, int]:
    edges = scene["edges"]
    return _is_connected(edges), _odd_count_from_edges(edges)


def _candidate_edge_sets(want_yes: bool) -> list[list[list[int]]]:
    choices: list[list[list[int]]] = []
    for omitted in itertools.combinations(range(len(GRID_EDGES)), 2):
        edges = [list(edge) for index, edge in enumerate(GRID_EDGES) if index not in omitted]
        if not _is_connected(edges):
            continue
        odd = _odd_count_from_edges(edges)
        if (odd in (0, 2)) == want_yes:
            choices.append(edges)
    return choices


YES_EDGE_SETS = _candidate_edge_sets(True)
NO_EDGE_SETS = _candidate_edge_sets(False)


def _transform_point(x: int, y: int, pose: int) -> tuple[int, int]:
    """Apply one of eight square symmetries around the canonical grid center."""
    x -= 210
    y -= 210
    if pose >= 4:
        x = -x
    for _ in range(pose % 4):
        x, y = -y, x
    return x + 210, y + 210


def sample_scene(seed: int) -> dict:
    """Sample a balanced, connected nine-node graph with ten equal-scale links."""
    seed = int(seed)
    graph_rng = random.Random(seed ^ 0x51A6E)
    # Consecutive even/odd seeds share nuisance styling, while their labels differ.
    # This yields explicit matched counterfactual pairs for palette, pose, and size.
    rng = random.Random((seed // 2) ^ 0xC0FFEE)
    want_yes = seed % 2 == 0
    pool = YES_EDGE_SETS if want_yes else NO_EDGE_SETS
    edges = deepcopy(pool[graph_rng.randrange(len(pool))])

    pose = rng.randrange(8)
    dx = rng.randint(-11, 11)
    dy = rng.randint(-11, 11)
    canonical = [(110 + 100 * col, 110 + 100 * row) for row in range(3) for col in range(3)]
    nodes = []
    for x, y in canonical:
        tx, ty = _transform_point(x, y, pose)
        nodes.append([tx + dx, ty + dy])

    # Four low-contrast marks are visible nuisances, never graph elements.
    decoration_slots = [(42, 48), (372, 46), (44, 370), (374, 372), (208, 43), (210, 376)]
    rng.shuffle(decoration_slots)
    decorations = [[x + rng.randint(-5, 5), y + rng.randint(-5, 5)] for x, y in decoration_slots[:4]]
    graph_rng.shuffle(edges)
    return {
        "nodes": nodes,
        "edges": edges,
        "node_radius": rng.choice((13, 14, 15)),
        "edge_width": rng.choice((9, 10, 11)),
        "node_rgb": list(rng.choice(NODE_PALETTES)),
        "edge_rgb": list(rng.choice(EDGE_PALETTES)),
        "background_rgb": list(rng.choice(BACKGROUNDS)),
        "decorations": decorations,
    }


def render(scene: dict) -> Image.Image:
    """Render only visible scene fields, with anti-aliased links and junctions."""
    scale = 3
    background = tuple(scene["background_rgb"])
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), background)
    draw = ImageDraw.Draw(image)

    for x, y in scene["decorations"]:
        radius = 5 * scale
        box = [x * scale - radius, y * scale - radius, x * scale + radius, y * scale + radius]
        pale = tuple((3 * channel + 210) // 4 for channel in background)
        draw.ellipse(box, fill=pale)

    nodes = scene["nodes"]
    edge_color = tuple(scene["edge_rgb"])
    for left, right in scene["edges"]:
        x1, y1 = nodes[int(left)]
        x2, y2 = nodes[int(right)]
        draw.line(
            (x1 * scale, y1 * scale, x2 * scale, y2 * scale),
            fill=edge_color,
            width=int(scene["edge_width"]) * scale,
        )

    radius = int(scene["node_radius"]) * scale
    node_color = tuple(scene["node_rgb"])
    for x, y in nodes:
        box = [x * scale - radius, y * scale - radius, x * scale + radius, y * scale + radius]
        draw.ellipse(box, fill=node_color)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    connected, odd = _graph_stats(scene)
    return "yes" if connected and odd in (0, 2) else "no"


def margin(scene: dict) -> float:
    """Signed distance from the even-count midpoint separating 2 from 4 odds."""
    connected, odd = _graph_stats(scene)
    if not connected:
        return -9.0
    return float(3 - odd)


def is_quarantined(scene: dict) -> bool:
    """Reject render settings that would make incidence unstable at pixel scale."""
    if int(scene["node_radius"]) < 12 or int(scene["edge_width"]) < 8:
        return True
    node = scene["node_rgb"]
    edge = scene["edge_rgb"]
    background = scene["background_rgb"]
    node_contrast = math.sqrt(sum((int(a) - int(b)) ** 2 for a, b in zip(node, background)))
    edge_contrast = math.sqrt(sum((int(a) - int(b)) ** 2 for a, b in zip(edge, background)))
    return node_contrast < 90 or edge_contrast < 150


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse the unordered edge serialization without changing pixels or gold."""
    twin = deepcopy(scene)
    twin["edges"] = list(reversed(twin["edges"]))
    return [("reverse_edge_serialization", twin)]
