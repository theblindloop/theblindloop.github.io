"""Independent pixel-only inverse arm for the visible junction-link graph."""

from __future__ import annotations

from collections import deque
import math

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Return 8-connected foreground coordinate arrays without external CV code."""
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    found: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y = int(y)
        x = int(x)
        if seen[y, x]:
            continue
        pending = deque([(y, x)])
        seen[y, x] = True
        pixels = []
        while pending:
            cy, cx = pending.popleft()
            pixels.append((cy, cx))
            for ny in range(max(0, cy - 1), min(height, cy + 2)):
                for nx in range(max(0, cx - 1), min(width, cx + 2)):
                    if mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        pending.append((ny, nx))
        found.append(np.asarray(pixels, dtype=float))
    return found


def _node_centers(rgb: np.ndarray) -> list[tuple[float, float]]:
    high = rgb.max(axis=2).astype(int)
    low = rgb.min(axis=2).astype(int)
    # Junctions are the only saturated, substantial regions in the public raster.
    saturated = (high - low > 45) & (high < 230) & (low < 190)
    components = [component for component in _components(saturated) if 350 <= len(component) <= 900]
    if len(components) < 3:
        raise ValueError(f"too few visible circular junctions: {len(components)}")
    centers = []
    for component in components:
        y, x = component.mean(axis=0)
        centers.append((float(x), float(y)))
    return centers


def _dark_fraction(rgb: np.ndarray, first: tuple[float, float], second: tuple[float, float]) -> float:
    x1, y1 = first
    x2, y2 = second
    dx = x2 - x1
    dy = y2 - y1
    length = math.hypot(dx, dy)
    px = -dy / length
    py = dx / length
    hits = 0
    total = 0
    for step in range(18, 83):
        t = step / 100.0
        bx = x1 + t * dx
        by = y1 + t * dy
        for offset in (-3, -2, -1, 0, 1, 2, 3):
            x = int(round(bx + px * offset))
            y = int(round(by + py * offset))
            if 0 <= y < rgb.shape[0] and 0 <= x < rgb.shape[1]:
                pixel = rgb[y, x].astype(int)
                total += 1
                if int(pixel.max()) < 135 and int(pixel.max() - pixel.min()) < 40:
                    hits += 1
    return hits / max(total, 1)


def _recover_graph(rgb: np.ndarray) -> tuple[list[set[int]], int]:
    centers = _node_centers(rgb)
    distances = []
    for left in range(len(centers)):
        for right in range(left + 1, len(centers)):
            distance = math.dist(centers[left], centers[right])
            distances.append((distance, left, right))
    nearest = min(distance for distance, _, _ in distances)
    adjacency = [set() for _ in centers]
    edge_count = 0
    for distance, left, right in distances:
        # Cardinal grid neighbors are ~100 px apart; diagonals are ~141 px.
        if distance > nearest * 1.22:
            continue
        if _dark_fraction(rgb, centers[left], centers[right]) > 0.72:
            adjacency[left].add(right)
            adjacency[right].add(left)
            edge_count += 1
    return adjacency, edge_count


def decision_from_image(image: Image.Image) -> str:
    """Recover incidence from RGB pixels, then apply the Euler-trail predicate."""
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if rgb.shape[0] < 100 or rgb.shape[1] < 100:
        raise ValueError("raster is too small")
    adjacency, edge_count = _recover_graph(rgb)
    if edge_count < len(adjacency) - 1:
        raise ValueError("too few visible links to form a connected network")
    seen = {0}
    pending = [0]
    while pending:
        for neighbor in adjacency[pending.pop()]:
            if neighbor not in seen:
                seen.add(neighbor)
                pending.append(neighbor)
    if len(seen) != len(adjacency):
        return "no"
    odd = sum(len(neighbors) % 2 for neighbors in adjacency)
    return "yes" if odd in (0, 2) else "no"
