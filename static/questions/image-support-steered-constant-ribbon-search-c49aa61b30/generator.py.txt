"""Deterministic latent sampler and raster renderer for constant ribbon search."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

COLORS: dict[str, tuple[int, int, int]] = {
    "red": (220, 50, 47),
    "blue": (35, 105, 190),
    "green": (38, 150, 85),
    "purple": (130, 72, 170),
}
COLOR_NAMES = tuple(COLORS)
POSITIONS = ((128, 160), (384, 160), (128, 352), (384, 352))
TARGET_WORD = (14, 14, 14, 14)
DISTRACTOR_WORDS = (
    (12, 14, 14, 16),  # range 4; equal sum 56
    (10, 12, 16, 18),  # range 8; equal sum 56
    (8, 12, 16, 20),   # range 12; equal sum 56
)
SECTION_LENGTH = 20
END_TAB_LENGTH = 4
END_TAB_WIDTH = 20


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene from one integer seed.

    The answer color cycles exactly with seed modulo four. A separately seeded
    permutation places colors independently among the four fixed anchors.
    Every ribbon has the same length and colored area (four section widths sum
    to 56, with identical 20x4 end tabs).
    """

    rng = random.Random(int(seed) ^ 0x5A17C9)
    target_color = COLOR_NAMES[int(seed) % len(COLOR_NAMES)]
    base = list(DISTRACTOR_WORDS[int(seed) % len(DISTRACTOR_WORDS)])

    patterns: dict[str, list[int]] = {}
    for color in COLOR_NAMES:
        if color == target_color:
            patterns[color] = list(TARGET_WORD)
        else:
            word = list(base)
            rng.shuffle(word)
            patterns[color] = word

    shuffled_positions = [list(p) for p in POSITIONS]
    rng.shuffle(shuffled_positions)
    positions = {
        color: shuffled_positions[index] for index, color in enumerate(COLOR_NAMES)
    }
    draw_order = list(COLOR_NAMES)
    rng.shuffle(draw_order)
    return {
        "canvas_size": [512, 512],
        "background": [255, 255, 255],
        "patterns": patterns,
        "positions": positions,
        "draw_order": draw_order,
    }


def _rect_for_width(x0: int, x1: int, cy: int, width: int) -> tuple[int, int, int, int]:
    top = cy - width // 2
    return (x0, top, x1, top + width - 1)


def render(scene: dict) -> Image.Image:
    """Render four disjoint, equal-ink connected ribbons using exact RGB fills."""

    size = tuple(int(v) for v in scene["canvas_size"])
    background = tuple(int(v) for v in scene["background"])
    image = Image.new("RGB", size, background)
    draw = ImageDraw.Draw(image)
    patterns = scene["patterns"]
    positions = scene["positions"]

    interior_length = 4 * SECTION_LENGTH
    total_length = interior_length + 2 * END_TAB_LENGTH
    for color in scene["draw_order"]:
        cx, cy = (int(v) for v in positions[color])
        x0 = cx - total_length // 2
        rgb = COLORS[color]
        draw.rectangle(
            _rect_for_width(x0, x0 + END_TAB_LENGTH - 1, cy, END_TAB_WIDTH),
            fill=rgb,
        )
        interior_x = x0 + END_TAB_LENGTH
        for index, width in enumerate(patterns[color]):
            sx0 = interior_x + index * SECTION_LENGTH
            sx1 = sx0 + SECTION_LENGTH - 1
            draw.rectangle(_rect_for_width(sx0, sx1, cy, int(width)), fill=rgb)
        right_x = interior_x + interior_length
        draw.rectangle(
            _rect_for_width(right_x, right_x + END_TAB_LENGTH - 1, cy, END_TAB_WIDTH),
            fill=rgb,
        )
    return image


def _ranges(scene: dict) -> dict[str, int]:
    return {
        color: max(int(v) for v in word) - min(int(v) for v in word)
        for color, word in scene["patterns"].items()
    }


def analytic_gold(scene: dict) -> str:
    """Return the unique color whose four interior widths are identical."""

    winners = [
        color
        for color, word in scene["patterns"].items()
        if len(set(int(v) for v in word)) == 1
    ]
    if len(winners) != 1:
        raise ValueError(f"scene has {len(winners)} constant ribbons, expected one")
    return winners[0]


def margin(scene: dict) -> float:
    values = sorted(_ranges(scene).values())
    if len(values) < 2:
        return 0.0
    return float(values[1] - values[0])


def is_quarantined(scene: dict) -> bool:
    ranges = _ranges(scene)
    return sum(value == 0 for value in ranges.values()) != 1 or margin(scene) < 3.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse disjoint-object paint order, a real pixel-identical latent alias."""

    twin = copy.deepcopy(scene)
    twin["draw_order"] = list(reversed(scene["draw_order"]))
    return [("reverse_disjoint_draw_order", twin)]
