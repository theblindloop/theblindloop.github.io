"""Independent pixel-only inverse arm for constant ribbon search.

This module deliberately contains its own public palette and geometry policy.
It receives only a final PIL image. Missing or clipped distractors are ignored;
only a full-length, internally measurable constant ribbon can win. This makes
destructive support local to the winning connected ribbon rather than to all
four candidate positions.
"""

from __future__ import annotations

from dataclasses import dataclass

import numpy as np
from PIL import Image

COLOR_NAMES = ("red", "blue", "green", "purple")
PALETTE = np.asarray(
    ((220, 50, 47), (35, 105, 190), (38, 150, 85), (130, 72, 170)),
    dtype=np.float32,
)
MAX_COLOR_DISTANCE = 36.0
MIN_FULL_LENGTH = 84
MAX_FULL_LENGTH = 92
END_TAB_COLUMNS = 4
EXPECTED_AREA = 1280
AREA_TOLERANCE = 24
CONSTANT_RANGE_TOLERANCE = 1.0


@dataclass(frozen=True)
class RibbonMeasurement:
    color: str
    bbox: tuple[int, int, int, int]
    area: int
    section_widths: tuple[float, float, float, float]
    width_range: float


class OracleAbstention(ValueError):
    """The raster does not contain exactly one complete constant ribbon."""


def _color_masks(image: Image.Image) -> dict[str, np.ndarray]:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    distances = np.sqrt(
        np.sum((array[:, :, None, :] - PALETTE[None, None, :, :]) ** 2, axis=3)
    )
    labels = np.argmin(distances, axis=2)
    nearest = np.min(distances, axis=2)
    return {
        color: (labels == index) & (nearest <= MAX_COLOR_DISTANCE)
        for index, color in enumerate(COLOR_NAMES)
    }


def _measure_one(color: str, mask: np.ndarray) -> RibbonMeasurement | None:
    ys, xs = np.nonzero(mask)
    if len(xs) == 0:
        return None
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    length = x1 - x0 + 1
    area = int(mask.sum())
    if not MIN_FULL_LENGTH <= length <= MAX_FULL_LENGTH:
        return None
    if abs(area - EXPECTED_AREA) > AREA_TOLERANCE:
        return None

    profile: list[int] = []
    previous_interval: tuple[int, int] | None = None
    for x in range(x0, x1 + 1):
        column = np.flatnonzero(mask[:, x])
        if len(column) == 0:
            return None
        lo, hi = int(column[0]), int(column[-1])
        if hi - lo + 1 != len(column):
            return None
        if previous_interval is not None:
            plo, phi = previous_interval
            if min(phi, hi) < max(plo, lo):
                return None
        previous_interval = (lo, hi)
        profile.append(len(column))

    if len(profile) <= 2 * END_TAB_COLUMNS:
        return None
    interior = np.asarray(
        profile[END_TAB_COLUMNS:-END_TAB_COLUMNS], dtype=np.float64
    )
    if len(interior) % 4:
        return None
    sections = tuple(float(np.median(chunk)) for chunk in np.split(interior, 4))
    width_range = float(max(sections) - min(sections))
    return RibbonMeasurement(
        color=color,
        bbox=(x0, y0, x1, y1),
        area=area,
        section_widths=sections,
        width_range=width_range,
    )


def measure_candidates(image: Image.Image) -> dict[str, RibbonMeasurement]:
    measured: dict[str, RibbonMeasurement] = {}
    for color, mask in _color_masks(image).items():
        item = _measure_one(color, mask)
        if item is not None:
            measured[color] = item
    return measured


def decision_from_image(image: Image.Image) -> str:
    """Select the unique full-length ribbon with constant interior thickness."""

    measured = measure_candidates(image)
    winners = [
        item.color
        for item in measured.values()
        if item.width_range <= CONSTANT_RANGE_TOLERANCE
    ]
    if len(winners) != 1:
        raise OracleAbstention(
            f"expected one complete constant ribbon, found {len(winners)}"
        )
    return winners[0]
