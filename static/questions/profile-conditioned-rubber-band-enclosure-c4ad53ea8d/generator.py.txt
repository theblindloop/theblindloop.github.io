"""Renderer for a global convex-envelope completion task."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512


def _cross(o: tuple[float, float], a: tuple[float, float], b: tuple[float, float]) -> float:
    return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])


def _hull(points: Iterable[Iterable[float]]) -> list[tuple[float, float]]:
    pts = sorted({(float(p[0]), float(p[1])) for p in points})
    if len(pts) < 3:
        return pts
    lower: list[tuple[float, float]] = []
    for p in pts:
        while len(lower) >= 2 and _cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    upper: list[tuple[float, float]] = []
    for p in reversed(pts):
        while len(upper) >= 2 and _cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    return lower[:-1] + upper[:-1]


def _inside(point: Iterable[float], polygon: list[tuple[float, float]]) -> bool:
    p = (float(point[0]), float(point[1]))
    return len(polygon) >= 3 and all(
        _cross(polygon[i], polygon[(i + 1) % len(polygon)], p) > 0
        for i in range(len(polygon))
    )


def _point_segment_distance(
    p: tuple[float, float], a: tuple[float, float], b: tuple[float, float]
) -> float:
    vx, vy = b[0] - a[0], b[1] - a[1]
    wx, wy = p[0] - a[0], p[1] - a[1]
    denom = vx * vx + vy * vy
    t = max(0.0, min(1.0, (wx * vx + wy * vy) / denom))
    return math.hypot(p[0] - (a[0] + t * vx), p[1] - (a[1] + t * vy))


def _clearance(point: Iterable[float], polygon: list[tuple[float, float]]) -> float:
    p = (float(point[0]), float(point[1]))
    return min(
        _point_segment_distance(p, polygon[i], polygon[(i + 1) % len(polygon)])
        for i in range(len(polygon))
    )


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    cx = 256 + rng.randint(-24, 24)
    cy = 256 + rng.randint(-22, 22)
    rx = rng.randint(128, 152)
    ry = rng.randint(112, 142)
    rotation = rng.uniform(0, math.tau)
    boundary_count = rng.choice((8, 9, 10))
    boundary: list[list[int]] = []
    for i in range(boundary_count):
        angle = rotation + math.tau * i / boundary_count + rng.uniform(-0.035, 0.035)
        scale = rng.uniform(0.96, 1.04)
        boundary.append([
            round(cx + rx * scale * math.cos(angle)),
            round(cy + ry * scale * math.sin(angle)),
        ])

    hull = _hull(boundary)
    inside_label = "red" if int(seed) % 2 else "blue"
    outside_label = "blue" if inside_label == "red" else "red"

    # The inside point sometimes approaches the implicit envelope, making the
    # visible boundary margin vary without becoming raster-ambiguous.
    target_clearance = 17 + (abs(int(seed)) % 39)
    edge_index = rng.randrange(len(hull))
    a, b = hull[edge_index], hull[(edge_index + 1) % len(hull)]
    t = rng.uniform(0.30, 0.70)
    edge_point = (a[0] + t * (b[0] - a[0]), a[1] + t * (b[1] - a[1]))
    inward = (cx - edge_point[0], cy - edge_point[1])
    norm = max(1.0, math.hypot(*inward))
    inside_point = [
        round(edge_point[0] + target_clearance * inward[0] / norm),
        round(edge_point[1] + target_clearance * inward[1] / norm),
    ]

    # Place the other marker beyond the opposite support edge with a generous,
    # independently varying clearance.
    out_edge = (edge_index + len(hull) // 2) % len(hull)
    a2, b2 = hull[out_edge], hull[(out_edge + 1) % len(hull)]
    t2 = rng.uniform(0.32, 0.68)
    edge2 = (a2[0] + t2 * (b2[0] - a2[0]), a2[1] + t2 * (b2[1] - a2[1]))
    outward = (edge2[0] - cx, edge2[1] - cy)
    norm2 = max(1.0, math.hypot(*outward))
    outside_clearance = rng.randint(24, 48)
    outside_point = [
        round(edge2[0] + outside_clearance * outward[0] / norm2),
        round(edge2[1] + outside_clearance * outward[1] / norm2),
    ]

    markers = {
        inside_label: inside_point,
        outside_label: outside_point,
    }
    return {
        "black_points": boundary,
        "markers": markers,
        "palette": {
            "background": [248, 247, 242],
            "black": [28, 31, 34],
            "red": [220, 54, 64],
            "blue": [43, 112, 205],
        },
        "canvas": [WIDTH, HEIGHT],
        "dot_radius": 8,
        "marker_radius": 13,
    }


def render(scene: dict) -> Image.Image:
    width, height = (int(v) for v in scene["canvas"])
    palette = scene["palette"]
    image = Image.new("RGB", (width, height), tuple(palette["background"]))
    draw = ImageDraw.Draw(image)
    dot_radius = int(scene["dot_radius"])
    for x, y in scene["black_points"]:
        draw.ellipse(
            (x - dot_radius, y - dot_radius, x + dot_radius, y + dot_radius),
            fill=tuple(palette["black"]),
        )
    marker_radius = int(scene["marker_radius"])
    for label in ("red", "blue"):
        x, y = scene["markers"][label]
        color = tuple(palette[label])
        draw.ellipse(
            (x - marker_radius, y - marker_radius, x + marker_radius, y + marker_radius),
            fill=color,
            outline=(255, 255, 255),
            width=2,
        )
        inner = marker_radius // 3
        draw.ellipse((x - inner, y - inner, x + inner, y + inner), fill=(255, 255, 255))
    return image


def analytic_gold(scene: dict) -> str:
    hull = _hull(scene["black_points"])
    markers = scene["markers"]
    enclosed = [label for label in ("red", "blue") if _inside(markers[label], hull)]
    if len(enclosed) != 1:
        raise ValueError("scene must contain exactly one enclosed colored marker")
    return enclosed[0]


def margin(scene: dict) -> float:
    hull = _hull(scene["black_points"])
    markers = scene["markers"]
    return float(min(_clearance(markers[label], hull) for label in ("red", "blue")))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
