"""Independent image-only inverse for the rubber-band enclosure world."""

from __future__ import annotations

from collections import deque
from typing import Iterable

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[float, float, int]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            py, px = queue.popleft()
            xs.append(px)
            ys.append(py)
            for ny, nx in ((py - 1, px), (py + 1, px), (py, px - 1), (py, px + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= minimum:
            found.append((sum(xs) / len(xs), sum(ys) / len(ys), len(xs)))
    return found


def _cross(o: tuple[float, float], a: tuple[float, float], b: tuple[float, float]) -> float:
    return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])


def _hull(points: Iterable[tuple[float, float]]) -> list[tuple[float, float]]:
    pts = sorted(set(points))
    lower: list[tuple[float, float]] = []
    for p in pts:
        while len(lower) >= 2 and _cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    upper: list[tuple[float, float]] = []
    for p in reversed(pts):
        while len(upper) >= 2 and _cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    return lower[:-1] + upper[:-1]


def _inside(point: tuple[float, float], polygon: list[tuple[float, float]]) -> bool:
    return len(polygon) >= 3 and all(
        _cross(polygon[i], polygon[(i + 1) % len(polygon)], point) > 0
        for i in range(len(polygon))
    )


def _single_center(mask: np.ndarray, label: str) -> tuple[float, float]:
    comps = _components(mask, minimum=40)
    if len(comps) != 1:
        raise ValueError(f"expected one {label} marker, found {len(comps)}")
    return comps[0][0], comps[0][1]


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    black = (rgb[:, :, 0] < 70) & (rgb[:, :, 1] < 70) & (rgb[:, :, 2] < 70)
    red = (rgb[:, :, 0] > 160) & (rgb[:, :, 0] > rgb[:, :, 1] + 80) & (rgb[:, :, 0] > rgb[:, :, 2] + 70)
    blue = (rgb[:, :, 2] > 140) & (rgb[:, :, 2] > rgb[:, :, 0] + 70) & (rgb[:, :, 2] > rgb[:, :, 1] + 40)
    black_components = _components(black, minimum=80)
    if len(black_components) < 6:
        raise ValueError("insufficient black-dot evidence")
    black_centers = [(item[0], item[1]) for item in black_components]
    hull = _hull(black_centers)
    marker_centers = {
        "red": _single_center(red, "red"),
        "blue": _single_center(blue, "blue"),
    }
    enclosed = [label for label, point in marker_centers.items() if _inside(point, hull)]
    if len(enclosed) != 1:
        raise ValueError("raster does not support a unique enclosed marker")
    return enclosed[0]
