"""Deterministic renderer for a compact checker-phase silhouette."""

from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image, ImageDraw

CANVAS_SIZE = 512
BACKGROUND = (246, 244, 237)
DISTRACTOR_RGB = (151, 149, 144)
PALETTES = (
    ((24, 112, 148), (144, 205, 220)),
    ((106, 61, 155), (201, 169, 226)),
    ((176, 71, 34), (237, 177, 140)),
    ((32, 128, 91), (151, 211, 183)),
)
TILE_CELLS = 16
ROUND_RADIUS = 4.75
FOUR_SIDE_HALF_EXTENT = 4.35
FOUR_SIDE_ROTATIONS = (15, 20, 30, 35, 40)
BOUNDARY_CV_TIE = 0.095


def sample_scene(seed: int) -> dict:
    value = int(seed)
    rng = random.Random(value ^ 0x5A17_2026)
    shape_class = "round" if value % 2 == 0 else "four-sided"
    center_x = rng.choice((160, 352))
    center_y = rng.choice((160, 352))
    distractors: list[dict] = []
    while len(distractors) < 7:
        x = rng.randrange(28, CANVAS_SIZE - 28)
        y = rng.randrange(28, CANVAS_SIZE - 28)
        if abs(x - center_x) < 92 and abs(y - center_y) < 92:
            continue
        distractors.append(
            {"x": x, "y": y, "kind": rng.choice(("dot", "square")), "size": rng.choice((3, 4, 5))}
        )
    return {
        "canvas_size": CANVAS_SIZE,
        "center_x": center_x,
        "center_y": center_y,
        "tile_cells": TILE_CELLS,
        "cell_size": rng.choice((6, 7)),
        "shape_class": shape_class,
        "shape_radius": ROUND_RADIUS if shape_class == "round" else FOUR_SIDE_HALF_EXTENT,
        "shape_rotation_deg": rng.choice(FOUR_SIDE_ROTATIONS),
        "phase_offset": rng.randrange(2),
        "palette": [list(color) for color in rng.choice(PALETTES)],
        "background_rgb": list(BACKGROUND),
        "distractor_marks": distractors,
    }


def _shape_cells(scene: dict) -> np.ndarray:
    count = int(scene["tile_cells"])
    coordinates = np.arange(count, dtype=np.float64) - (count - 1) / 2.0
    yy, xx = np.meshgrid(coordinates, coordinates, indexing="ij")
    if str(scene["shape_class"]) == "round":
        mask = xx * xx + yy * yy <= float(scene["shape_radius"]) ** 2
    elif str(scene["shape_class"]) == "four-sided":
        theta = math.radians(float(scene["shape_rotation_deg"]))
        u = xx * math.cos(theta) + yy * math.sin(theta)
        v = -xx * math.sin(theta) + yy * math.cos(theta)
        mask = np.maximum(np.abs(u), np.abs(v)) <= float(scene["shape_radius"])
    else:
        raise ValueError("unknown shape class")
    if int(mask.sum()) != 76:
        raise ValueError("silhouette must occupy exactly 76 checker cells")
    return mask


def _boundary_cv(mask: np.ndarray) -> float:
    interior = np.zeros_like(mask, dtype=bool)
    interior[1:-1, 1:-1] = (
        mask[1:-1, 1:-1]
        & mask[:-2, 1:-1]
        & mask[2:, 1:-1]
        & mask[1:-1, :-2]
        & mask[1:-1, 2:]
    )
    boundary = mask & ~interior
    ys, xs = np.nonzero(boundary)
    all_y, all_x = np.nonzero(mask)
    cx, cy = float(all_x.mean()), float(all_y.mean())
    radii = np.hypot(xs - cx, ys - cy)
    return float(radii.std() / radii.mean())


def render(scene: dict) -> Image.Image:
    canvas = int(scene["canvas_size"])
    background = tuple(int(v) for v in scene["background_rgb"])
    image = Image.new("RGB", (canvas, canvas), background)
    draw = ImageDraw.Draw(image)
    for mark in scene["distractor_marks"]:
        x, y, radius = int(mark["x"]), int(mark["y"]), int(mark["size"])
        bounds = (x - radius, y - radius, x + radius, y + radius)
        if str(mark["kind"]) == "dot":
            draw.ellipse(bounds, fill=DISTRACTOR_RGB)
        else:
            draw.rectangle(bounds, fill=DISTRACTOR_RGB)

    count = int(scene["tile_cells"])
    cell = int(scene["cell_size"])
    size = count * cell
    x0 = int(scene["center_x"]) - size // 2
    y0 = int(scene["center_y"]) - size // 2
    mask = _shape_cells(scene)
    palette = [tuple(int(v) for v in color) for color in scene["palette"]]
    phase = int(scene["phase_offset"])
    for row in range(count):
        for col in range(count):
            bit = (row + col + phase + int(mask[row, col])) % 2
            draw.rectangle(
                (x0 + col * cell, y0 + row * cell, x0 + (col + 1) * cell - 1, y0 + (row + 1) * cell - 1),
                fill=palette[bit],
            )
    return image


def analytic_gold(scene: dict) -> str:
    value = str(scene["shape_class"])
    if value not in ("round", "four-sided"):
        raise ValueError("invalid shape class")
    return value


def margin(scene: dict) -> float:
    return abs(_boundary_cv(_shape_cells(scene)) - BOUNDARY_CV_TIE)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.012


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["distractor_marks"] = list(reversed(scene["distractor_marks"]))
    return [("reverse_commutative_distractor_order", twin)]
