"""Independent pixel-only inverse for the checker-phase silhouette."""

from __future__ import annotations

from collections import deque
from dataclasses import dataclass

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    """The final raster does not expose the declared texture structure cleanly."""


@dataclass(frozen=True)
class Measurement:
    decision: str
    runner_up_gap: float
    boundary_radial_cv: float
    reversed_cell_count: int
    tile_bbox: tuple[int, int, int, int]


def _components(mask: np.ndarray) -> list[np.ndarray]:
    work = np.asarray(mask, dtype=bool).copy()
    height, width = work.shape
    result: list[np.ndarray] = []
    for sy, sx in np.argwhere(work):
        y0, x0 = int(sy), int(sx)
        if not work[y0, x0]:
            continue
        work[y0, x0] = False
        queue = deque([(y0, x0)])
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        result.append(np.asarray(points, dtype=np.int16))
    return sorted(result, key=len, reverse=True)


def _chromatic_tile(array: np.ndarray) -> tuple[np.ndarray, tuple[int, int, int, int]]:
    border = np.concatenate((array[:3].reshape(-1, 3), array[-3:].reshape(-1, 3), array[:, :3].reshape(-1, 3), array[:, -3:].reshape(-1, 3)))
    background = np.median(border.astype(np.float32), axis=0)
    delta = np.linalg.norm(array.astype(np.float32) - background, axis=2)
    chroma = array.max(axis=2).astype(np.int16) - array.min(axis=2).astype(np.int16)
    mask = (delta >= 48.0) & (chroma >= 38)
    components = _components(mask)
    if not components:
        raise OracleAbstention("no chromatic texture tile")
    component = components[0]
    if len(component) < 9000:
        raise OracleAbstention("too little chromatic tile area")
    ys, xs = component[:, 0], component[:, 1]
    x0, x1 = int(xs.min()), int(xs.max()) + 1
    y0, y1 = int(ys.min()), int(ys.max()) + 1
    width, height = x1 - x0, y1 - y0
    if width != height or width not in (96, 112) or len(component) != width * height:
        raise OracleAbstention("tile is not one intact compact square")
    return mask[y0:y1, x0:x1], (x0, y0, x1, y1)


def _phase_region(array: np.ndarray, bbox: tuple[int, int, int, int]) -> np.ndarray:
    x0, y0, x1, y1 = bbox
    tile = array[y0:y1, x0:x1]
    size = x1 - x0
    if size % 16:
        raise OracleAbstention("texture period is unresolved")
    cell = size // 16
    flat = tile.reshape(-1, 3)
    colors, counts = np.unique(flat, axis=0, return_counts=True)
    if len(colors) != 2 or int(counts.min()) < flat.shape[0] * 0.40:
        raise OracleAbstention("tile does not contain two balanced texture inks")
    labels = np.empty((16, 16), dtype=np.uint8)
    for row in range(16):
        for col in range(16):
            patch = tile[row * cell:(row + 1) * cell, col * cell:(col + 1) * cell]
            matches = np.all(patch == colors[0], axis=2)
            if not (matches.all() or (~matches).all()):
                raise OracleAbstention("a checker cell is damaged or mixed")
            labels[row, col] = int(not matches.all())
    parity = np.indices((16, 16)).sum(axis=0) % 2
    first = labels != parity
    second = labels == parity
    region = first if int(first.sum()) < int(second.sum()) else second
    if int(region.sum()) != 76:
        raise OracleAbstention("phase reversal has the wrong area")
    components = _components(region)
    if len(components) != 1:
        raise OracleAbstention("phase reversal is not one connected region")
    return region


def _radial_cv(mask: np.ndarray) -> float:
    padded = np.pad(mask, 1, constant_values=False)
    core = padded[1:-1, 1:-1]
    interior = core & padded[:-2, 1:-1] & padded[2:, 1:-1] & padded[1:-1, :-2] & padded[1:-1, 2:]
    boundary = core & ~interior
    ys, xs = np.nonzero(boundary)
    all_y, all_x = np.nonzero(mask)
    if len(xs) < 20:
        raise OracleAbstention("too little boundary evidence")
    center_x, center_y = float(np.median(all_x)), float(np.median(all_y))
    radii = np.hypot(xs - center_x, ys - center_y)
    return float(np.std(radii, ddof=0) / np.mean(radii))


def measure_image(image: Image.Image) -> Measurement:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.shape != (512, 512, 3):
        raise OracleAbstention("unsupported raster dimensions")
    _, bbox = _chromatic_tile(array)
    region = _phase_region(array, bbox)
    score = _radial_cv(region)
    tie = 0.095
    gap = abs(score - tie)
    if gap <= 0.012:
        raise OracleAbstention("boundary fit is too close to the round/four-sided tie")
    decision = "round" if score < tie else "four-sided"
    return Measurement(decision, gap, score, int(region.sum()), bbox)


def decision_from_image(image: Image.Image) -> str:
    return measure_image(image).decision
