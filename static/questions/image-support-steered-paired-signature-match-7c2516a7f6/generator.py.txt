"""Paired-signature match world renderer.

Latent scene: a single reference panel at the top and four candidate panels in
a 2x2 grid below. Every panel is a light-gray rectangle with a black border and
holds exactly two colored dots (a left slot and a right slot). The question asks
which candidate panel reproduces the reference signature on BOTH slots in the
same left/right order.

The decisive evidence is spread across two disconnected regions: the reference
panel (whose two slots define the target signature) and the single matching
candidate panel. Removing either region (or either slot of it) aborts the
conjunction and changes or destroys the unique answer, giving a regional,
multipart pixel-oracle support.

```text
I = R(z)
answer = label of the unique card c with c.left==ref.left and c.right==ref.right
```
"""

from __future__ import annotations

import dataclasses
import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "paired_signature_match-0.1.0"

CANVAS = (512, 512)
BG = (255, 255, 255)
PANEL_FILL = (238, 238, 238)
PANEL_BORDER = (20, 20, 20)
DOT_COLORS = {
    "red": (214, 48, 49),
    "blue": (41, 82, 214),
    "green": (38, 166, 91),
}
PANEL_W = 150
PANEL_H = 76
DOT_R = 12.0
DOT_DX = 40.0
BORDER = 2

# Reference panel and the 2x2 candidate grid (deterministic geometry so the
# oracle only needs colors; the answer position and colors are randomized).
REF_CENTER = (256, 150)
CARD_CENTERS = {
    "top_left": (140, 320),
    "top_right": (372, 320),
    "bottom_left": (140, 430),
    "bottom_right": (372, 430),
}
PALETTE = ("red", "blue", "green")


@dataclasses.dataclass(frozen=True)
class PairedScene:
    width: int = 512
    height: int = 512
    reference: dict = None
    cards: list = None


def _panel_rect(center):
    cx, cy = center
    return (
        cx - PANEL_W // 2,
        cy - PANEL_H // 2,
        cx + PANEL_W // 2,
        cy + PANEL_H // 2,
    )


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    ref_left, ref_right = rng.sample(list(PALETTE), 2)
    ref_split = ref_left + ref_right
    third = next(c for c in PALETTE if c not in (ref_left, ref_right))

    pairs = [
        (ref_left, ref_right),   # the unique correct card
        (ref_left, third),       # shares the reference left slot only
        (third, ref_right),      # shares the reference right slot only
        (third, third),          # shares neither slot
    ]
    labels = list(CARD_CENTERS)
    rng.shuffle(labels)
    cards = [{"label": label, "left": left, "right": right}
             for label, (left, right) in zip(labels, pairs)]

    return {
        "width": 512,
        "height": 512,
        "reference": {"left": ref_left, "right": ref_right},
        "cards": cards,
    }


def _draw_panel(draw, center, slots, ss):
    x0, y0, x1, y1 = _panel_rect(center)
    draw.rectangle([x0 * ss, y0 * ss, x1 * ss, y1 * ss], fill=PANEL_FILL, outline=PANEL_BORDER, width=BORDER * ss)
    cx, cy = center
    for offset, color in slots:
        dot = DOT_COLORS[color]
        dx = cx + offset
        draw.ellipse(
            [ (dx - DOT_R) * ss, (cy - DOT_R) * ss,
              (dx + DOT_R) * ss, (cy + DOT_R) * ss ],
            fill=dot,
        )


def render(scene) -> Image.Image:
    w = int(scene["width"])
    h = int(scene["height"])
    ss = 2
    img = Image.new("RGB", (w * ss, h * ss), BG)
    draw = ImageDraw.Draw(img)

    ref = scene["reference"]
    _draw_panel(draw, REF_CENTER, [(-DOT_DX, ref["left"]), (DOT_DX, ref["right"])], ss)
    for card in scene["cards"]:
        _draw_panel(
            draw,
            CARD_CENTERS[card["label"]],
            [(-DOT_DX, card["left"]), (DOT_DX, card["right"])],
            ss,
        )

    img = img.resize((w, h), Image.LANCZOS)
    return img


def _matching_label(scene):
    ref_l = scene["reference"]["left"]
    ref_r = scene["reference"]["right"]
    matches = [
        card["label"]
        for card in scene["cards"]
        if card["left"] == ref_l and card["right"] == ref_r
    ]
    if len(matches) != 1:
        return None
    return matches[0]


def analytic_gold(scene) -> str:
    return _matching_label(scene)


def margin(scene) -> float:
    """Runner-up margin: matched slots of the correct card minus the best other.

    The correct card matches both slots (score 2); by construction every other
    card matches at most one slot (conjunction guard), so the margin is 1.0.
    """
    ref_l = scene["reference"]["left"]
    ref_r = scene["reference"]["right"]
    scores = []
    for card in scene["cards"]:
        score = (card["left"] == ref_l) + (card["right"] == ref_r)
        scores.append(score)
    best = max(scores)
    scores.sort(reverse=True)
    runner_up = scores[1] if len(scores) > 1 else 0
    return float(best - runner_up)


def is_quarantined(scene) -> bool:
    # A unique match with a clear top gap is built in; a tie (runner-up also
    # matching both slots) or a missing match is ambiguous and quarantined.
    ref_l = scene["reference"]["left"]
    ref_r = scene["reference"]["right"]
    matches = [
        card for card in scene["cards"]
        if card["left"] == ref_l and card["right"] == ref_r
    ]
    return len(matches) != 1 or margin(scene) < 1.0


def latent_symmetries(scene):
    """not_applicable: minimal injective parametrization, no hidden fields."""
    return iter(())
