"""Independent pixel-only oracle for the paired-signature match world.

Receives only the final PNG. Never imports renderer/prompts/generate/verify and
never accepts latent scene data.

Algorithm: detect the light-gray panels as connected components, identify the
uppermost panel as the reference, read each panel's left and right colored
slots, and report the unique card whose two-slot signature (left color, right
color) matches the reference in the same order. A missing panel, a tie, or a
panel with no colored dot makes the oracle abstain with "none".
"""

from __future__ import annotations

from collections import deque
from dataclasses import dataclass

import numpy as np
from PIL import Image


def _connected_components(mask: np.ndarray, min_area: int = 300):
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components = []
    for start_y, start_x in np.argwhere(mask):
        sy, sx = int(start_y), int(start_x)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        points = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if (
                    0 <= ny < height and 0 <= nx < width
                    and mask[ny, nx] and not seen[ny, nx]
                ):
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= min_area:
            ys = [p[0] for p in points]
            xs = [p[1] for p in points]
            components.append(
                {
                    "y0": min(ys), "y1": max(ys),
                    "x0": min(xs), "x1": max(xs),
                    "cy": min(ys) + (max(ys) - min(ys)) / 2.0,
                    "cx": min(xs) + (max(xs) - min(xs)) / 2.0,
                }
            )
    return components


def decision_from_image(image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    if arr.ndim != 3:
        return "none"
    r = arr[..., 0]
    g = arr[..., 1]
    b = arr[..., 2]
    fill = (np.abs(r - 238) < 9) & (np.abs(g - 238) < 9) & (np.abs(b - 238) < 9)
    panels = _connected_components(fill)
    if len(panels) < 5:
        return "none"
    panels.sort(key=lambda p: p["cy"])
    reference = panels[0]
    cards = panels[1:]

    def slots(panel):
        y0, y1, x0, x1 = panel["y0"], panel["y1"], panel["x0"], panel["x1"]
        red = (r > 150) & (g < 110) & (b < 110)
        green = (g > 100) & (r < 120) & (b < 120) & (g > r) & (g > b)
        blue = (b > 150) & (r < 120) & (g < 130)
        mid = (x0 + x1) // 2

        def color_of(region):
            counts = {
                "red": int(np.count_nonzero(region & red)),
                "green": int(np.count_nonzero(region & green)),
                "blue": int(np.count_nonzero(region & blue)),
            }
            best = max(counts, key=counts.get)
            return best if counts[best] > 0 else None

        left_region = np.zeros_like(red)
        left_region[y0:y1 + 1, x0:mid + 1] = True
        right_region = np.zeros_like(red)
        right_region[y0:y1 + 1, mid + 1:x1 + 1] = True
        return color_of(left_region), color_of(right_region)

    ref_l, ref_r = slots(reference)
    if ref_l is None or ref_r is None:
        return "none"

    # assign labels by relative geometry among the four cards
    cards.sort(key=lambda p: p["cy"])
    top = cards[:2]
    bottom = cards[2:]
    top.sort(key=lambda p: p["cx"])
    bottom.sort(key=lambda p: p["cx"])
    labeled = {
        "top_left": top[0], "top_right": top[1],
        "bottom_left": bottom[0], "bottom_right": bottom[1],
    }

    matches = []
    for label, panel in labeled.items():
        cl, cr = slots(panel)
        if cl is None or cr is None:
            continue
        if cl == ref_l and cr == ref_r:
            matches.append(label)
    if len(matches) == 1:
        return matches[0]
    return "none"
