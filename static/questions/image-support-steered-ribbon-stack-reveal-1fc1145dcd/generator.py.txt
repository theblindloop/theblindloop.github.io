"""Ribbon-stack reveal world.

One orange ribbon snakes through a small inset window of the canvas as a single
unbranching chain of six equal square panels joined edge to edge.  Every panel
carries a 5x5 field of small square dots, drawn in the ribbon's own travel frame:
the panel's local rows run toward the neighbouring panel the ribbon leaves by.

The declared operation is to stack the panels in that travel frame and keep the
grid cells that carry a dot in *every* panel.  Those cells spell one abstract
glyph.  No panel shows the glyph on its own (each carries eight extra dots), and
stacking the panels in canvas orientation instead of the ribbon's travel frame
never spells a declared glyph, so the decision lives in the orientation the thin
route transports along its own interior.
"""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw

RENDERER_VERSION = "ribbon_stack_reveal-renderer-0.1.0"

CANVAS = 640
GRID = 5
PANELS = 6
LATTICE = 3
PITCH = 30
TILE = 25
NECK = 10
CELL = 5
DOT = 4
DECOY_DOTS = 8
ARROW = 6

BG = (238, 236, 230)
TAPE = (206, 122, 58)
DOT_RGB = (38, 40, 46)
ARROW_RGB = (0, 0, 0)
DECOY_TAPE = (198, 206, 196)
DECOY_DOT = (150, 158, 150)

QUARANTINE_CLEARANCE = 2.0

# Window origins keep the whole decisive ribbon, arrow included, inside one local
# neighbourhood of the canvas instead of letting it straddle the middle.
BANDS = (57, 165, 272, 378, 485)
BAND_SLACK = 7

GLYPHS = ("ell", "ex", "plus", "tee")

_DIRS = ((1, 0), (0, 1), (-1, 0), (0, -1))


def glyph_cells(name: str) -> frozenset[tuple[int, int]]:
    """The nine marked (row, col) cells of one declared 5x5 glyph."""
    if name == "plus":
        cells = {(2, c) for c in range(5)} | {(r, 2) for r in range(5)}
    elif name == "ex":
        cells = {(i, i) for i in range(5)} | {(i, 4 - i) for i in range(5)}
    elif name == "tee":
        cells = {(0, c) for c in range(5)} | {(r, 2) for r in range(1, 5)}
    elif name == "ell":
        cells = {(r, 0) for r in range(5)} | {(4, c) for c in range(1, 5)}
    else:
        raise ValueError(f"unknown glyph {name!r}")
    return frozenset(cells)


GLYPH_CELLS = {name: glyph_cells(name) for name in GLYPHS}
ALL_CELLS = frozenset((r, c) for r in range(GRID) for c in range(GRID))


def _match(cells: frozenset[tuple[int, int]]) -> str | None:
    for name in GLYPHS:
        if GLYPH_CELLS[name] == cells:
            return name
    return None


def _routes(rng: random.Random) -> list[tuple[int, int]]:
    """A random self-avoiding walk of PANELS cells inside a LATTICE window."""
    while True:
        start = (rng.randrange(LATTICE), rng.randrange(LATTICE))
        route = [start]
        seen = {start}
        ok = True
        for _ in range(PANELS - 1):
            options = []
            for dx, dy in _DIRS:
                nxt = (route[-1][0] + dx, route[-1][1] + dy)
                if 0 <= nxt[0] < LATTICE and 0 <= nxt[1] < LATTICE and nxt not in seen:
                    options.append(nxt)
            if not options:
                ok = False
                break
            nxt = rng.choice(options)
            route.append(nxt)
            seen.add(nxt)
        if not ok:
            continue
        dirs = travel_dirs(route)
        if len({tuple(d) for d in dirs}) < 3:
            continue
        behind = (route[0][0] - dirs[0][0], route[0][1] - dirs[0][1])
        if behind in seen:
            continue
        return route


def travel_dirs(route: list) -> list[tuple[int, int]]:
    """Outgoing lattice direction of every panel; the last panel reuses its own."""
    out = []
    for i in range(len(route)):
        a = route[i] if i + 1 < len(route) else route[i - 1]
        b = route[i + 1] if i + 1 < len(route) else route[i]
        out.append((b[0] - a[0], b[1] - a[1]))
    return out


def _panel_sets(rng: random.Random, target: frozenset, want: int) -> tuple[list[list[list[int]]], int]:
    """Per-panel dot fields: the glyph plus DECOY_DOTS extra cells, in travel frame.

    ``want`` is the requested decision redundancy: the number of panels that must
    independently omit a stray cell before that cell is excluded from the stack.
    """
    others = sorted(ALL_CELLS - target)
    tries = 0
    while True:
        tries += 1
        picks = [set(rng.sample(others, DECOY_DOTS)) for _ in range(PANELS)]
        omit = {cell: sum(1 for p in picks if cell not in p) for cell in others}
        clearance = min(omit.values())
        if clearance < 1:
            continue
        if clearance != want and tries < 3000:
            continue
        fields = []
        for pick in picks:
            marks = target | pick
            fields.append([[1 if (r, c) in marks else 0 for c in range(GRID)] for r in range(GRID)])
        return fields, clearance


def _decoy_tapes(rng: random.Random, window: tuple[int, int]) -> list[dict]:
    """Pale decoy ribbons placed well away from the decisive window."""
    wx, wy = window
    span = LATTICE * PITCH
    tapes = []
    tries = 0
    while len(tapes) < 2 and tries < 200:
        tries += 1
        ox = rng.randrange(30, CANVAS - 3 * PITCH - 30)
        oy = rng.randrange(30, CANVAS - 3 * PITCH - 30)
        if abs(ox - wx) < span + 40 and abs(oy - wy) < span + 40:
            continue
        route = [(0, 0)]
        for _ in range(3):
            dx, dy = rng.choice(_DIRS)
            nxt = (route[-1][0] + dx, route[-1][1] + dy)
            if nxt in route or not (0 <= nxt[0] < 2 and 0 <= nxt[1] < 2):
                nxt = (route[-1][0], route[-1][1] + 1) if (route[-1][0], route[-1][1] + 1) not in route else (route[-1][0] + 1, route[-1][1])
            route.append(nxt)
        bits = [[rng.randrange(2) for _ in range(GRID)] for _ in range(GRID)]
        tapes.append({"origin": [ox, oy], "route": [list(c) for c in route], "bits": bits})
    return tapes


def sample_scene(seed: int) -> dict:
    """Deterministically sample one scene."""
    rng = random.Random((int(seed) * 2654435761) % (2**32))
    glyph = GLYPHS[rng.randrange(len(GLYPHS))]
    route = _routes(rng)
    span = LATTICE * PITCH
    ox = rng.choice(BANDS)
    oy = rng.choice(BANDS)
    ox += rng.randrange(0, BAND_SLACK)
    oy += rng.randrange(0, BAND_SLACK)
    want = 1 if rng.random() < 0.3 else 2
    probe = {"route": [list(cell) for cell in route]}
    while True:
        fields, clearance = _panel_sets(rng, GLYPH_CELLS[glyph], want)
        probe["fields"] = fields
        if _match(canvas_frame_cells(probe)) is None:
            break
    scene = {
        "canvas": CANVAS,
        "pitch": PITCH,
        "tile_px": TILE,
        "neck_px": NECK,
        "cell_px": CELL,
        "dot_px": DOT,
        "grid": GRID,
        "origin": [ox, oy],
        "route": [list(cell) for cell in route],
        "fields": fields,
        "glyph": glyph,
        "clearance": clearance,
        "decoys": _decoy_tapes(rng, (ox, oy)),
    }
    return scene


def _center(scene: dict, cell) -> tuple[int, int]:
    ox, oy = scene["origin"]
    return (ox + cell[1] * PITCH + PITCH // 2, oy + cell[0] * PITCH + PITCH // 2)


def _frame(direction) -> tuple[tuple[int, int], tuple[int, int]]:
    """Travel unit vector u (image x, y) and its clockwise normal v."""
    dr, dc = direction
    u = (dc, dr)
    v = (-dr, dc)
    return u, v


def render(scene: dict) -> Image.Image:
    """Rasterise one scene with flat fills and no antialiasing."""
    img = Image.new("RGB", (scene["canvas"], scene["canvas"]), BG)
    draw = ImageDraw.Draw(img)
    half = TILE // 2

    for tape in scene["decoys"]:
        tx, ty = tape["origin"]
        cells = [tuple(c) for c in tape["route"]]
        for cell in cells:
            cx = tx + cell[1] * PITCH + PITCH // 2
            cy = ty + cell[0] * PITCH + PITCH // 2
            draw.rectangle([cx - half, cy - half, cx + half - 1, cy + half - 1], fill=DECOY_TAPE)
        for a, b in zip(cells, cells[1:]):
            ax = tx + a[1] * PITCH + PITCH // 2
            ay = ty + a[0] * PITCH + PITCH // 2
            bx = tx + b[1] * PITCH + PITCH // 2
            by = ty + b[0] * PITCH + PITCH // 2
            mx, my = (ax + bx) // 2, (ay + by) // 2
            if ax == bx:
                draw.rectangle([mx - NECK // 2, my - PITCH // 2, mx + NECK // 2 - 1, my + PITCH // 2 - 1], fill=DECOY_TAPE)
            else:
                draw.rectangle([mx - PITCH // 2, my - NECK // 2, mx + PITCH // 2 - 1, my + NECK // 2 - 1], fill=DECOY_TAPE)
        for cell in cells:
            cx = tx + cell[1] * PITCH + PITCH // 2
            cy = ty + cell[0] * PITCH + PITCH // 2
            for r in range(GRID):
                for c in range(GRID):
                    if tape["bits"][r][c]:
                        dx = cx + (c - 2) * CELL
                        dy = cy + (r - 2) * CELL
                        draw.rectangle([dx - DOT // 2, dy - DOT // 2, dx + DOT // 2 - 1, dy + DOT // 2 - 1], fill=DECOY_DOT)

    route = [tuple(c) for c in scene["route"]]
    for cell in route:
        cx, cy = _center(scene, cell)
        draw.rectangle([cx - half, cy - half, cx + half - 1, cy + half - 1], fill=TAPE)
    for a, b in zip(route, route[1:]):
        ax, ay = _center(scene, a)
        bx, by = _center(scene, b)
        mx, my = (ax + bx) // 2, (ay + by) // 2
        if ax == bx:
            draw.rectangle([mx - NECK // 2, my - PITCH // 2, mx + NECK // 2 - 1, my + PITCH // 2 - 1], fill=TAPE)
        else:
            draw.rectangle([mx - PITCH // 2, my - NECK // 2, mx + PITCH // 2 - 1, my + NECK // 2 - 1], fill=TAPE)

    dirs = travel_dirs(route)
    for cell, direction, field in zip(route, dirs, scene["fields"]):
        cx, cy = _center(scene, cell)
        u, v = _frame(direction)
        for r in range(GRID):
            for c in range(GRID):
                if not field[r][c]:
                    continue
                dx = cx + (c - 2) * CELL * u[0] + (r - 2) * CELL * v[0]
                dy = cy + (c - 2) * CELL * u[1] + (r - 2) * CELL * v[1]
                draw.rectangle([dx - DOT // 2, dy - DOT // 2, dx + DOT // 2 - 1, dy + DOT // 2 - 1], fill=DOT_RGB)

    sx, sy = _center(scene, route[0])
    u, _ = _frame(dirs[0])
    tipx, tipy = sx - u[0] * (half + 1), sy - u[1] * (half + 1)
    backx, backy = tipx - u[0] * ARROW, tipy - u[1] * ARROW
    px, py = -u[1], u[0]
    draw.polygon(
        [
            (tipx, tipy),
            (backx + px * ARROW // 2, backy + py * ARROW // 2),
            (backx - px * ARROW // 2, backy - py * ARROW // 2),
        ],
        fill=ARROW_RGB,
    )
    return img


def stacked_cells(scene: dict) -> frozenset[tuple[int, int]]:
    """Cells carrying a dot in every panel, read in the ribbon's travel frame."""
    keep = set(ALL_CELLS)
    for field in scene["fields"]:
        keep &= {(r, c) for r in range(GRID) for c in range(GRID) if field[r][c]}
    return frozenset(keep)


def canvas_frame_cells(scene: dict) -> frozenset[tuple[int, int]]:
    """Control reading: the same intersection with every panel left in canvas orientation."""
    route = [tuple(c) for c in scene["route"]]
    dirs = travel_dirs(route)
    keep = set(ALL_CELLS)
    for direction, field in zip(dirs, scene["fields"]):
        u, v = _frame(direction)
        marks = set()
        for r in range(GRID):
            for c in range(GRID):
                if field[r][c]:
                    x = (c - 2) * u[0] + (r - 2) * v[0]
                    y = (c - 2) * u[1] + (r - 2) * v[1]
                    marks.add((y + 2, x + 2))
        keep &= marks
    return frozenset(keep)


def analytic_gold(scene: dict) -> str:
    """The declared glyph spelled by the travel-frame panel stack."""
    name = _match(stacked_cells(scene))
    if name is None:
        raise ValueError("scene does not spell a declared glyph")
    return name


def margin(scene: dict) -> float:
    """Redundancy of the decision: panels that must agree to reject a stray cell."""
    return float(scene["clearance"])


def is_quarantined(scene: dict) -> bool:
    """True when one panel's single dot decides a grid cell."""
    return float(scene["clearance"]) < QUARANTINE_CLEARANCE


def hamming_margin(scene: dict) -> int:
    """Runner-up separation: smallest cell disagreement with a competing glyph."""
    cells = stacked_cells(scene)
    best = min(len(cells ^ GLYPH_CELLS[g]) for g in GLYPHS if GLYPH_CELLS[g] != cells)
    return int(best)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases are declared: every sampled field is rendered."""
    return []
