"""Pixel-only inverse arm for the switch-to-pad binding world.

This module receives nothing but a rendered image. It re-derives the two local
bits independently:

1. the switch bit, from the teal ink alone: the teal shape is a wide channel bar
   plus a narrow trunk stub that leaves the bar towards the middle of the tile,
   so the half of the teal bounding box that carries most of the teal mass is
   the side the trunk is attached to. No external reference point is needed, so
   the reading survives partial damage of the tile;
2. the pad bit, from the orange and purple ink alone: the pad with the smaller
   row coordinate is the upper pad.

The decision is the pad colour lying on the same side as the attached channel.
Any missing component, any tie in the mass split, and any collapsed pad
separation raise ``OracleReadError`` instead of guessing.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "switch_pad_reach-oracle-0.1.0"

_TEAL = (17, 116, 133)
_ORANGE = (214, 118, 34)
_PURPLE = (110, 58, 160)

_COLOR_TOL = 45
_MIN_INK = 12
_MASS_RATIO = 1.5
_MIN_PAD_SEPARATION = 4.0


class OracleReadError(RuntimeError):
    """Raised when the image does not resolve the decision."""


def _mask(array: np.ndarray, color: tuple[int, int, int]) -> np.ndarray:
    reference = np.asarray(color, dtype=np.int16)
    return np.abs(array - reference).sum(axis=2) <= _COLOR_TOL


def _mean_row(mask: np.ndarray) -> float:
    rows = np.nonzero(mask)[0]
    if rows.size < _MIN_INK:
        raise OracleReadError("a pad colour is missing from the image")
    return float(rows.mean())


def _attached_side(teal: np.ndarray) -> str:
    rows = np.nonzero(teal)[0]
    if rows.size < _MIN_INK:
        raise OracleReadError("no teal switch ink is visible")
    low, high = int(rows.min()), int(rows.max())
    if high - low < 4:
        raise OracleReadError("teal ink is too flat to expose an attachment side")
    middle = (low + high) / 2.0
    upper = int((rows < middle).sum())
    lower = int((rows > middle).sum())
    strong, weak = max(upper, lower), min(upper, lower)
    if weak == 0 or strong < _MASS_RATIO * weak:
        raise OracleReadError("teal mass split does not resolve an attachment side")
    return "upper" if upper > lower else "lower"


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    side = _attached_side(_mask(array, _TEAL))
    orange_row = _mean_row(_mask(array, _ORANGE))
    purple_row = _mean_row(_mask(array, _PURPLE))
    if abs(orange_row - purple_row) < _MIN_PAD_SEPARATION:
        raise OracleReadError("the two pads are not vertically separable")
    orange_side = "upper" if orange_row < purple_row else "lower"
    return "orange" if side == orange_side else "purple"
