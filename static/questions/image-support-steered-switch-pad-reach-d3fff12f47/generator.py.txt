"""Renderer for the switch-to-pad binding world.

Scene layout: a light paper canvas holds several small "stacked-bar" tiles. One
tile is the switch tile (a teal trunk stub attached to exactly one of its two
channel bars), one is the pad tile (an orange and a purple pad bar), and the
rest are all-gray decoys. The two decisive tiles sit in separate small
neighbourhoods of a coarse lattice with plain background between them, so the
answer requires binding two disconnected local orderings.

Only axis-aligned rectangles are drawn, so every rendered pixel carries an
exact palette colour and no antialiasing blur exists at any tile edge.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "switch_pad_reach-renderer-0.1.0"

CANVAS = 448
CELL = 56
JITTER = 1

BG = (246, 244, 238)
GRAY = (140, 142, 148)
TEAL = (17, 116, 133)
ORANGE = (214, 118, 34)
PURPLE = (110, 58, 160)

BAR_HALF_W = 12
BAR_INNER = 7
BAR_OUTER = 12
TAB_HALF_W = 2
TRUNK_LENGTHS = (3, 4, 5, 6, 7, 8, 9)
QUARANTINE_PX = 5.0

_SIDES = ("upper", "lower")


def _cell_center(index: int, jitter: int) -> int:
    return CELL * index + CELL // 2 + jitter


def sample_scene(seed: int) -> dict:
    """Sample one scene. Absolute tile position varies over the whole canvas."""

    rng = random.Random(1_000_003 + int(seed) * 7919)
    lattice = CANVAS // CELL
    while True:
        col = rng.randrange(0, lattice)
        row = rng.randrange(0, lattice)
        dx = rng.choice((-2, 2))
        dy = rng.choice((-1, 1))
        pad_col, pad_row = col + dx, row + dy
        if 0 <= pad_col < lattice and 0 <= pad_row < lattice:
            break
    switch_center = [
        _cell_center(col, rng.randint(-JITTER, JITTER)),
        _cell_center(row, rng.randint(-JITTER, JITTER)),
    ]
    pad_center = [
        _cell_center(pad_col, rng.randint(-JITTER, JITTER)),
        _cell_center(pad_row, rng.randint(-JITTER, JITTER)),
    ]

    used = {(col, row), (pad_col, pad_row)}
    blocked = set()
    for cc, rr in used:
        for ddx in (-1, 0, 1):
            for ddy in (-1, 0, 1):
                blocked.add((cc + ddx, rr + ddy))
    free = [
        (cc, rr)
        for cc in range(lattice)
        for rr in range(lattice)
        if (cc, rr) not in blocked
    ]
    rng.shuffle(free)
    decoys = []
    for cc, rr in free[: rng.randint(3, 6)]:
        decoys.append(
            {
                "center": [
                    _cell_center(cc, rng.randint(-JITTER, JITTER)),
                    _cell_center(rr, rng.randint(-JITTER, JITTER)),
                ],
                "tab_side": rng.choice(("upper", "lower", "none")),
                "tab_len": rng.choice(TRUNK_LENGTHS),
            }
        )

    return {
        "width": CANVAS,
        "height": CANVAS,
        "cell": CELL,
        "switch_center": switch_center,
        "pad_center": pad_center,
        "live_side": rng.choice(_SIDES),
        "orange_side": rng.choice(_SIDES),
        "trunk_len": rng.choice(TRUNK_LENGTHS),
        "decoys": decoys,
    }


def _draw_tile(
    draw: ImageDraw.ImageDraw,
    center: list[int],
    upper: tuple[int, int, int],
    lower: tuple[int, int, int],
    tab_side: str,
    tab_len: int,
    tab_color: tuple[int, int, int],
) -> None:
    cx, cy = int(center[0]), int(center[1])
    draw.rectangle(
        [cx - BAR_HALF_W, cy - BAR_OUTER, cx + BAR_HALF_W - 1, cy - BAR_INNER],
        fill=upper,
    )
    draw.rectangle(
        [cx - BAR_HALF_W, cy + BAR_INNER, cx + BAR_HALF_W - 1, cy + BAR_OUTER],
        fill=lower,
    )
    if tab_side == "upper":
        draw.rectangle(
            [cx - TAB_HALF_W, cy - BAR_INNER + 1, cx + TAB_HALF_W, cy - BAR_INNER + tab_len],
            fill=tab_color,
        )
    elif tab_side == "lower":
        draw.rectangle(
            [cx - TAB_HALF_W, cy + BAR_INNER - tab_len, cx + TAB_HALF_W, cy + BAR_INNER - 1],
            fill=tab_color,
        )


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BG)
    draw = ImageDraw.Draw(image)
    for decoy in scene["decoys"]:
        _draw_tile(
            draw,
            decoy["center"],
            GRAY,
            GRAY,
            str(decoy["tab_side"]),
            int(decoy["tab_len"]),
            GRAY,
        )
    live = str(scene["live_side"])
    _draw_tile(
        draw,
        scene["switch_center"],
        TEAL if live == "upper" else GRAY,
        TEAL if live == "lower" else GRAY,
        live,
        int(scene["trunk_len"]),
        TEAL,
    )
    orange_side = str(scene["orange_side"])
    _draw_tile(
        draw,
        scene["pad_center"],
        ORANGE if orange_side == "upper" else PURPLE,
        PURPLE if orange_side == "upper" else ORANGE,
        "none",
        3,
        GRAY,
    )
    return image


def analytic_gold(scene: dict) -> str:
    """Reached pad colour: the pad on the same side as the attached channel."""

    return "orange" if scene["live_side"] == scene["orange_side"] else "purple"


def margin(scene: dict) -> float:
    """Smallest decisive visible clearance, in pixels."""

    break_gap = float(2 * BAR_INNER - 1 - int(scene["trunk_len"]))
    pad_gap = float(2 * BAR_INNER - 1)
    clearance = float(CANVAS)
    sx, sy = scene["switch_center"]
    px, py = scene["pad_center"]
    anchors = [(int(sx), int(sy)), (int(px), int(py))]
    for index, (ax, ay) in enumerate(anchors):
        for bx, by in anchors[index + 1 :]:
            clearance = min(clearance, _tile_clearance(ax, ay, bx, by))
        for decoy in scene["decoys"]:
            dx, dy = decoy["center"]
            clearance = min(clearance, _tile_clearance(ax, ay, int(dx), int(dy)))
    return round(min(break_gap, pad_gap, clearance), 3)


def _tile_clearance(ax: int, ay: int, bx: int, by: int) -> float:
    gap_x = abs(ax - bx) - (2 * BAR_HALF_W - 1)
    gap_y = abs(ay - by) - 2 * BAR_OUTER
    if gap_x <= 0 and gap_y <= 0:
        return 0.0
    return float(max(gap_x, gap_y))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Render-invariant latent rewrites that must preserve the analytic gold."""

    twins: list[tuple[str, dict]] = []
    decoys = list(scene["decoys"])
    if len(decoys) > 1:
        reversed_scene = dict(scene)
        reversed_scene["decoys"] = list(reversed(decoys))
        twins.append(("decoy_paint_order_reversed", reversed_scene))
        rotated_scene = dict(scene)
        rotated_scene["decoys"] = decoys[1:] + decoys[:1]
        twins.append(("decoy_paint_order_rotated", rotated_scene))
    return twins


def label_transforming_symmetry(scene: dict) -> tuple[str, dict, str]:
    """Declared label-transforming rewrite: flipping one bit flips the answer."""

    twin = dict(scene)
    twin["live_side"] = "lower" if str(scene["live_side"]) == "upper" else "upper"
    expected = "orange" if twin["live_side"] == str(scene["orange_side"]) else "purple"
    return ("switch_attachment_flipped", twin, expected)


def translated_scene(scene: dict, cells_x: int, cells_y: int) -> dict:
    """Label-preserving translation of the whole evidence pair by whole cells."""

    twin = dict(scene)
    twin["switch_center"] = [
        int(scene["switch_center"][0]) + cells_x * CELL,
        int(scene["switch_center"][1]) + cells_y * CELL,
    ]
    twin["pad_center"] = [
        int(scene["pad_center"][0]) + cells_x * CELL,
        int(scene["pad_center"][1]) + cells_y * CELL,
    ]
    twin["decoys"] = []
    return twin
