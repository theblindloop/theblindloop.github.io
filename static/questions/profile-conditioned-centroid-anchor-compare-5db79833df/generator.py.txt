"""Centroid-anchor measurement world: which colored triangle's centroid is farther.

Latent scene z carries two solid filled triangles (one red, one blue) plus a
neutral gray reference dot drawn at the fixed center of a light panel. The task
is a pure geometric measurement: recover the geometric centroid (center of mass)
of each filled triangle from the raster ink, measure the Euclidean distance from
each centroid to the gray anchor dot, and compare the two distances. The answer
is the color of the triangle whose centroid is farther from the anchor.

Because each triangle is scalene, its centroid is NOT the center of its bounding
box, so the answer cannot be read from bounding-box extent, absolute position,
ink amount, or any single cheap local cue: only the true area-averaged centroid
position of each shape relative to the shared visible anchor separates the two
colors.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "centroid_anchor_compare-0.1.0"

# Fixed palette shared with oracle.py (plain list so both agree without import).
PALETTE = ["red", "blue"]
COLOR_RGB = {
    "red": (205, 45, 45),
    "blue": (35, 75, 205),
}
ANCHOR_RGB = (120, 120, 120)
BACKGROUND = (250, 250, 250)

WIDTH = 400
HEIGHT = 400
ANCHOR = (200.0, 200.0)
ANCHOR_RADIUS = 6.0

# Each triangle is confined to one half-panel so the two shapes never overlap
# each other or the central anchor dot. Boxes are interior with a canvas margin.
LEFT_BOX = (30.0, 30.0, 170.0, 370.0)      # (x0, y0, x1, y1)
RIGHT_BOX = (230.0, 30.0, 370.0, 370.0)
CANVAS_MARGIN = 4.0
MIN_TRI_AREA = 1200.0
MIN_TRI_SIDE = 18.0

# Near-boundary: top-two (here pair) centroid-distance gap below which the
# selection is a finite-resolution boundary case.
QUARANTINE_PX = 6.0
# We only ever emit scenes whose measured distance gap clears this floor.
MIN_MARGIN_PX = 8.0


def _sample_triangle(rng: random.Random, box: tuple[float, float, float, float], what: str) -> list:
    """Sample a valid, non-degenerate triangle fully inside ``box``."""
    x0, y0, x1, y1 = box
    for _ in range(2000):
        pts = [
            [rng.uniform(x0 + CANVAS_MARGIN, x1 - CANVAS_MARGIN),
             rng.uniform(y0 + CANVAS_MARGIN, y1 - CANVAS_MARGIN)]
            for _ in range(3)
        ]
        area = 0.5 * abs(
            (pts[1][0] - pts[0][0]) * (pts[2][1] - pts[0][1])
            - (pts[2][0] - pts[0][0]) * (pts[1][1] - pts[0][1])
        )
        if area < MIN_TRI_AREA:
            continue
        sides = [math.dist(pts[i], pts[(i + 1) % 3]) for i in range(3)]
        if min(sides) < MIN_TRI_SIDE:
            continue
        if any(x < 0 or x > WIDTH or y < 0 or y > HEIGHT for pt in pts for x, y in [pt]):
            continue
        return [[round(x, 2), round(y, 2)] for x, y in pts]
    raise RuntimeError(f"could not sample a valid {what} triangle")


def _centroid(pts: list) -> tuple[float, float]:
    n = len(pts)
    return (sum(p[0] for p in pts) / n, sum(p[1] for p in pts) / n)


def _dist(p: tuple[float, float], q: tuple[float, float]) -> float:
    return math.hypot(p[0] - q[0], p[1] - q[1])


def sample_scene(seed: int) -> dict:
    """Sample a deterministic, counterbalanced, decisive scene.

    Returns a plain JSON-serialisable dict that both render() and the analytic
    gold understand: ``tri_a``/``tri_b`` are the two triangles' vertex lists and
    ``anchor`` is the reference dot centre. Colors are bound to the two shapes
    after geometry so neither colour has a positional advantage.
    """
    rng = random.Random(seed * 104729 + 13)
    while True:
        side_a = rng.choice(["left", "right"])
        box_a = LEFT_BOX if side_a == "left" else RIGHT_BOX
        box_b = RIGHT_BOX if side_a == "left" else LEFT_BOX
        tri_a = _sample_triangle(rng, box_a, "A")
        tri_b = _sample_triangle(rng, box_b, "B")
        ca = _centroid(tri_a)
        cb = _centroid(tri_b)
        da = _dist(ca, ANCHOR)
        db = _dist(cb, ANCHOR)
        if abs(da - db) >= MIN_MARGIN_PX:
            break
    if rng.random() < 0.5:
        # swap which shape wears the red / blue labels
        tri_a, tri_b = tri_b, tri_a
    return {
        "tri_a": tri_a,
        "tri_b": tri_b,
        "anchor": [ANCHOR[0], ANCHOR[1]],
        "width": WIDTH,
        "height": HEIGHT,
        "background": "light",
    }


def render(scene: dict) -> Image.Image:
    """Render I = R(z): two filled triangles and the gray anchor dot."""
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    draw.polygon([tuple(p) for p in scene["tri_a"]], fill=COLOR_RGB["red"])
    draw.polygon([tuple(p) for p in scene["tri_b"]], fill=COLOR_RGB["blue"])
    ax, ay = scene["anchor"]
    draw.ellipse(
        [ax - ANCHOR_RADIUS, ay - ANCHOR_RADIUS, ax + ANCHOR_RADIUS, ay + ANCHOR_RADIUS],
        fill=ANCHOR_RGB,
    )
    return img


def _farther_color(scene: dict) -> tuple[str, float]:
    """Return (color of farther centroid, |d_red - d_blue|) from the analytic scene."""
    anchor = tuple(scene["anchor"])
    d_a = _dist(_centroid(scene["tri_a"]), anchor)
    d_b = _dist(_centroid(scene["tri_b"]), anchor)
    return ("red" if d_a > d_b else "blue"), abs(d_a - d_b)


def analytic_gold(scene: dict) -> str:
    """The colour of the triangle whose centroid is farther from the anchor."""
    color, _margin = _farther_color(scene)
    return color


def margin(scene: dict) -> float:
    """Verifiability margin: the centroid-distance gap in pixels."""
    _color, gap = _farther_color(scene)
    return float(gap)


def is_quarantined(scene: dict) -> bool:
    """Near-boundary: the centroid-distance gap is finite-resolution small."""
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list:
    """not_applicable: no nontrivial pixel-identical latent symmetry.

    Every latent field (the two triangles' vertex lists and the anchor centre)
    is directly rendered as visible ink. Swapping which triangle wears each
    colour changes which colour is correct, and any global flip/rotation turns
    the raster into a different image (and changes the measured centroid
    distances). There is no hidden field that could change the gold without
    changing the raster.
    """
    return []
