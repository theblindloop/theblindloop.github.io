"""Independent pixel-only inverse arm for the centroid-anchor world.

This module receives ONLY the rasterized PNG (plus the world's fixed public
palette / anchor colour) and re-measures which coloured triangle's geometric
centroid (centre of mass) is farther from the gray anchor dot. It shares no
geometry with renderer.render() and never imports the renderer, prompts,
generator, verifier, or any analytical gold code.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "centroid_anchor_compare-oracle-0.1.0"

PALETTE = ["red", "blue"]
COLOR_RGB = {
    "red": (205, 45, 45),
    "blue": (35, 75, 205),
}
ANCHOR_RGB = (120, 120, 120)
BACKGROUND = (250, 250, 250)

MIN_MASK_PX = 40


def _nearest_color(arr: np.ndarray) -> np.ndarray:
    """Label every pixel to its nearest palette entry (colours + anchor + bg)."""
    refs = np.stack(
        [np.array(COLOR_RGB[c], dtype=np.float32) for c in PALETTE]
        + [np.array(ANCHOR_RGB, dtype=np.float32), np.array(BACKGROUND, dtype=np.float32)],
        axis=0,
    )
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def _centroid(mask: np.ndarray) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if len(ys) < MIN_MASK_PX:
        raise ValueError("coloured region too small to measure")
    return float(xs.mean()), float(ys.mean())


def decision_from_image(image: Image.Image) -> str:
    """Return the colour whose triangle centroid is farther from the anchor.

    The centroid of a filled triangle is the pixel-mean of its mask, which for
    a uniform fill equals the geometric centroid of the shape.
    """
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _nearest_color(arr)
    try:
        anchor_c = _centroid(labels == 2)
        d_red = np.hypot(
            _centroid(labels == 0)[0] - anchor_c[0],
            _centroid(labels == 0)[1] - anchor_c[1],
        )
        d_blue = np.hypot(
            _centroid(labels == 1)[0] - anchor_c[0],
            _centroid(labels == 1)[1] - anchor_c[1],
        )
    except ValueError:
        return "invalid"
    return "red" if d_red > d_blue else "blue"
