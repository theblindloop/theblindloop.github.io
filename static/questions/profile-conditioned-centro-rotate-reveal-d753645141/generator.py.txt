"""centro_rotate_reveal: 180-degree rotational difference glyph reveal world.

Declared transform: subtract the 180-degree-rotated copy of the field from the
field itself (a centro-symmetric, point-reflection difference). The raw image
is a flat field of noise that is exactly centro-symmetric (S = S rotated 180)
so the noise cancels perfectly; only the symbol, written as an anti-symmetric
luminance bias around the point reflection (bright inside the answer band,
dark inside its 180-degree image band), survives in the difference. The
surviving field is strongly positive inside the answer band and negative
inside its reflection, giving a clean rising (A) or falling (B) diagonal band.

Latent scene z = {noise, glyph, cell}. label = glyph.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RENDERER_VERSION = "centro_rotate_reveal-0.1.0"

W = 512
H = 512
NGRID = 32          # lattice cells per side
CELL = W // NGRID   # 16
BASE = 128.0
BIAS = 40.0
NOISE_SIGMA = 20.0
TRI_H = 12         # triangle height in grid cells
TRI_W = 14         # triangle half-width at the base in grid cells
Q_TOL = 0.15       # quarantine threshold on the A/B correlation margin


def _rot180(mat: np.ndarray) -> np.ndarray:
    return np.rot90(mat, 2)


def band_masks() -> dict[str, np.ndarray]:
    """(NGRID, NGRID) bool masks: A = up triangle, B = its 180-degree image.

    A and B are exact 180-rotation images of each other, so they are disjoint
    and the centro-symmetric difference cleanly separates them.
    """
    r, c = np.indices((NGRID, NGRID))
    cm = (NGRID - 1) / 2.0
    mask_a = (r < TRI_H) & (np.abs(c - cm) <= (r * TRI_W / TRI_H) + 1e-6)
    mask_b = _rot180(mask_a)
    return {"A": mask_a, "B": mask_b}


def build_scene(noise: int, glyph: str, cell: int = CELL) -> dict:
    return {
        "noise": int(noise),
        "glyph": glyph,
        "cell": int(cell),
    }


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(int(seed) & 0xFFFFFFFF)
    glyph = "A" if rng.integers(0, 2) else "B"
    return build_scene(noise=int(seed), glyph=glyph)


def _glyph_mask(glyph: str) -> np.ndarray:
    return band_masks()[glyph]


def render_matrix(scene: dict) -> np.ndarray:
    """(NGRID, NGRID) mean-luminance of each lattice cell."""
    glyph = scene["glyph"]
    rng = np.random.default_rng(int(scene["noise"]) & 0xFFFFFFFF)
    n = NGRID
    t = rng.normal(0.0, NOISE_SIGMA, size=(n, n))
    symmetric = t + _rot180(t)
    g = _glyph_mask(glyph)
    antisym = g.astype(np.float64) - _rot180(g).astype(np.float64)
    val = BASE + symmetric + BIAS * antisym
    return val


def render(scene: dict) -> Image.Image:
    cell = int(scene.get("cell", CELL))
    n = W // cell
    mat = render_matrix(scene)[:n, :n]
    arr = np.repeat(np.repeat(mat, cell, axis=0), cell, axis=1)
    arr = np.clip(arr, 0, 255)
    canvas = np.repeat(arr[:, :, None], 3, axis=2).astype(np.uint8)
    canvas = canvas[:W, :H]
    return Image.fromarray(canvas, "RGB")


def margin(scene: dict) -> float:
    chosen = _glyph_mask(scene["glyph"])
    other = _glyph_mask("B" if scene["glyph"] == "A" else "A")
    c_filled = int(chosen.sum())
    o_filled = int(other.sum())
    corr_other = float((chosen & other).sum()) / max(1, o_filled)
    return round(1.0 - corr_other, 6)


def analytic_gold(scene: dict) -> str:
    return scene["glyph"]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < Q_TOL


def latent_symmetries(scene: dict):
    """not_applicable: every latent field (the per-cell noise seed, the answer
    glyph, the cell pitch) directly changes the rendered pixels, so no distinct
    latent vector reproduces the same image. Declared empty."""
    return iter(())
