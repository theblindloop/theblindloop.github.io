"""Rung-deal interleave reveal: one figure dealt row-wise across three far panels.

A figure lives on a 9x9 lattice. Its rows are dealt round-robin into three thin
three-row panels, so panel k receives lattice rows k, k+3 and k+6. The three
panels are then rendered far apart, in three widely separated bands of the
canvas and on alternating sides, with plenty of empty background and small solid
distractor marks between them. The declared raster operation is to order the
panels by their top edges and deal their rows back together in rotation, one row
from each panel in turn, rebuilding the 9x9 lattice.

Nothing in the picture draws the figure. Reading the panels in place, without
the deal, moves every rung to the wrong row and matches no declared figure. Each
of the four declared figures differs from another declared figure only inside the
rows of a single panel, so all three widely separated panels are jointly
necessary: the top rung is carried by the second panel, the middle rung by the
first, and the bottom rung by the third.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RENDERER_VERSION = "rung_deal_interleave_reveal-renderer-0.1.0"

CANVAS = 720
LATTICE = 9
PANEL_ROWS = 3
CELL_PX = 12
FRAME_PX = 3
PANEL_W = LATTICE * CELL_PX + 2 * FRAME_PX  # 114
PANEL_H = PANEL_ROWS * CELL_PX + 2 * FRAME_PX  # 42

BAND_Y = (70, 339, 608)
SIDE_X = {"left": 50, "right": 556}
JITTER = 10

DISTRACTOR_CLEARANCE = 25
DISTRACTOR_MIN = 3
DISTRACTOR_MAX = 6

PAPER = (250, 250, 250)
BACKGROUND = (236, 236, 238)
TINTS = ((0, 0, 0), (0, 5, 11), (11, 5, 0), (4, 11, 0))

MIN_CONTRAST = 62.0
MAX_CONTRAST = 150.0
QUARANTINE_CLEARANCE = 34.0

GLYPHS = ("ladder", "top_hook", "mid_hook", "foot_hook")
UNIFORM_GLYPHS = frozenset({"ladder"})

# Lattice rows carrying a rung, in panel order: the rung of panel 0 sits on
# lattice row 3, the rung of panel 1 on row 1 and the rung of panel 2 on row 5.
# None of 1, 3, 5 is fixed by the deal permutation, whose only fixed rows are
# 0, 4 and 8, so an un-dealt in-place reading always misplaces every rung.
RUNG_ROWS = (3, 1, 5)
LEFT_RUNG = (1, 2, 3, 4)
RIGHT_RUNG = (4, 5, 6, 7)
POSTS = (0, LATTICE - 1)

# Which lattice row turns to the right for each declared figure.
RIGHT_ROW = {"ladder": None, "top_hook": 1, "mid_hook": 3, "foot_hook": 5}


def _template(name: str) -> np.ndarray:
    grid = np.zeros((LATTICE, LATTICE), dtype=np.uint8)
    for row in range(LATTICE):
        for col in POSTS:
            grid[row, col] = 1
    for row in RUNG_ROWS:
        cols = RIGHT_RUNG if RIGHT_ROW[name] == row else LEFT_RUNG
        for col in cols:
            grid[row, col] = 1
    return grid


TEMPLATES: dict[str, np.ndarray] = {name: _template(name) for name in GLYPHS}


def panel_rows(index: int) -> tuple[int, ...]:
    """Lattice rows dealt to the panel at position ``index`` (0 is topmost)."""

    return tuple(index + 3 * step for step in range(PANEL_ROWS))


def deal(panels: list[np.ndarray]) -> np.ndarray:
    """Deal three 3x9 panel readouts back together into the 9x9 lattice."""

    grid = np.zeros((LATTICE, LATTICE), dtype=np.uint8)
    for index, panel in enumerate(panels):
        for step, row in enumerate(panel_rows(index)):
            grid[row] = panel[step]
    return grid


# Counterbalance guarantees checked at import time: every figure inks exactly 30
# lattice cells and exactly 10 of the 27 cells of every panel, so total ink,
# per-panel ink, panel size, panel order and panel position carry no information
# about the answer. Every figure also differs from some other figure only inside
# one panel's rows, which is what makes all three separated panels decisive.
for _name, _grid in TEMPLATES.items():
    assert int(_grid.sum()) == 30, _name
    for _index in range(PANEL_ROWS):
        assert int(_grid[list(panel_rows(_index))].sum()) == 10, (_name, _index)

MIN_GLYPH_SEPARATION = min(
    int(np.abs(TEMPLATES[a].astype(int) - TEMPLATES[b].astype(int)).sum())
    for i, a in enumerate(GLYPHS)
    for b in GLYPHS[i + 1:]
)
assert MIN_GLYPH_SEPARATION == 6

for _name in GLYPHS:
    _panels = [TEMPLATES[_name][list(panel_rows(i))] for i in range(PANEL_ROWS)]
    # An in-place reading stacks each panel's rows as a block instead of dealing
    # them, and must match no declared figure.
    _stacked = np.concatenate(_panels, axis=0)
    assert all(
        int(np.abs(_stacked.astype(int) - _other.astype(int)).sum()) >= 8
        for _other in TEMPLATES.values()
    ), _name
    # Each panel alone leaves at least two figures indistinguishable.
    for _index in range(PANEL_ROWS):
        _rows = list(panel_rows(_index))
        _seen = {tuple(TEMPLATES[o][_rows].ravel().tolist()) for o in GLYPHS}
        assert len(_seen) < len(GLYPHS), (_name, _index)


def _luma(color: tuple[int, int, int]) -> float:
    return 0.299 * color[0] + 0.587 * color[1] + 0.114 * color[2]


def _panel_rects(scene: dict) -> list[tuple[int, int, int, int]]:
    rects = []
    for index in range(PANEL_ROWS):
        top = BAND_Y[index] + int(scene["jitter"][index][0])
        left = SIDE_X[str(scene["sides"][index])] + int(scene["jitter"][index][1])
        rects.append((top, left, PANEL_H, PANEL_W))
    return rects


def _clear_of(rect: tuple[int, int, int, int], others: list[tuple[int, int, int, int]]) -> bool:
    top, left, height, width = rect
    for oy, ox, oh, ow in others:
        if (
            top - DISTRACTOR_CLEARANCE < oy + oh
            and oy - DISTRACTOR_CLEARANCE < top + height
            and left - DISTRACTOR_CLEARANCE < ox + ow
            and ox - DISTRACTOR_CLEARANCE < left + width
        ):
            return False
    return True


def sample_scene(seed: int) -> dict:
    """Sample one scene: figure, panel sides and jitter, tones, distractors."""

    # Two-word seeding decorrelates the first draws of consecutive scene seeds,
    # which keeps the four figure classes balanced across a contiguous dataset.
    rng = np.random.default_rng([int(seed), 0x51A7])
    glyph = GLYPHS[int(rng.integers(len(GLYPHS)))]
    first = "left" if rng.random() < 0.5 else "right"
    second = "right" if first == "left" else "left"
    third = "left" if rng.random() < 0.5 else "right"
    sides = [first, second, third]
    jitter = [
        [int(rng.integers(-JITTER, JITTER + 1)), int(rng.integers(-JITTER, JITTER + 1))]
        for _ in range(PANEL_ROWS)
    ]
    # One scene in five is drawn from the band straddling the quarantine
    # clearance so near-boundary and quarantined cases are always covered.
    if rng.random() < 0.2:
        contrast = float(np.round(rng.uniform(MIN_CONTRAST, MIN_CONTRAST + 14.0), 3))
    else:
        contrast = float(np.round(rng.uniform(MIN_CONTRAST, MAX_CONTRAST), 3))
    tint = TINTS[int(rng.integers(len(TINTS)))]

    scaffold = {"jitter": jitter, "sides": sides}
    blocked = _panel_rects(scaffold)
    distractors: list[list[int]] = []
    wanted = int(rng.integers(DISTRACTOR_MIN, DISTRACTOR_MAX + 1))
    for _ in range(240):
        if len(distractors) >= wanted:
            break
        if rng.random() < 0.5:
            height = int(rng.integers(10, 19))
            width = height
        else:
            height = int(rng.integers(8, 13))
            width = int(rng.integers(30, 61))
        top = int(rng.integers(24, CANVAS - 24 - height))
        left = int(rng.integers(24, CANVAS - 24 - width))
        rect = (top, left, height, width)
        if _clear_of(rect, blocked):
            distractors.append([top, left, height, width])
            blocked.append(rect)
    return {
        "canvas": CANVAS,
        "cell_px": CELL_PX,
        "frame_px": FRAME_PX,
        "glyph": glyph,
        "sides": sides,
        "jitter": jitter,
        "contrast": contrast,
        "tint": list(tint),
        "distractors": distractors,
    }


def _ink_color(scene: dict) -> tuple[int, int, int]:
    base = int(round(_luma(PAPER) - float(scene["contrast"])))
    tint = tuple(int(value) for value in scene["tint"])
    return tuple(max(0, min(255, base + offset)) for offset in tint)  # type: ignore[return-value]


def _clearance(scene: dict) -> float:
    """Half the rendered paper-to-ink luma gap: the per-cell decision clearance."""

    return 0.5 * (_luma(PAPER) - _luma(_ink_color(scene)))


def render(scene: dict) -> Image.Image:
    """Rasterise the scene with flat, hard-edged blocks and no antialiasing."""

    canvas = int(scene["canvas"])
    cell = int(scene["cell_px"])
    frame = int(scene["frame_px"])
    ink = np.array(_ink_color(scene), dtype=np.uint8)
    array = np.zeros((canvas, canvas, 3), dtype=np.uint8)
    array[:, :] = np.array(BACKGROUND, dtype=np.uint8)

    for top, left, height, width in scene["distractors"]:
        array[top:top + height, left:left + width] = ink

    grid = TEMPLATES[str(scene["glyph"])]
    panel_w = LATTICE * cell + 2 * frame
    panel_h = PANEL_ROWS * cell + 2 * frame
    for index, (top, left, _h, _w) in enumerate(_panel_rects(scene)):
        array[top:top + panel_h, left:left + panel_w] = ink
        array[top + frame:top + panel_h - frame, left + frame:left + panel_w - frame] = np.array(
            PAPER, dtype=np.uint8
        )
        for step, row in enumerate(panel_rows(index)):
            for col in range(LATTICE):
                if not grid[row, col]:
                    continue
                y0 = top + frame + step * cell
                x0 = left + frame + col * cell
                array[y0:y0 + cell, x0:x0 + cell] = ink
    return Image.fromarray(array, mode="RGB")


def analytic_gold(scene: dict) -> str:
    """The figure carried jointly by the three dealt panels."""

    return str(scene["glyph"])


def margin(scene: dict) -> float:
    """Per-cell luma clearance, the binding decision margin of the world."""

    return float(round(_clearance(scene), 6))


def is_quarantined(scene: dict) -> bool:
    """Flag scenes whose per-cell tone clearance is too small to read by eye."""

    return bool(_clearance(scene) < QUARANTINE_CLEARANCE)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist: every sampled field is rendered injectively."""

    return []
