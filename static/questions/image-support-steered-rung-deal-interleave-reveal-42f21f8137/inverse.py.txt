"""Pixel-only inverse arm for the rung-deal interleave reveal world.

The oracle receives one RGB image and nothing else. It thresholds at the
midpoint of its own darkest and lightest tones, finds every ink component,
keeps only the three thin bordered panels by their rendered shape, orders them
by their top edges, measures each panel's border thickness, reads its three rows
of nine cells, deals the three readouts back together in rotation into a 9x9
lattice and classifies that lattice against the declared figure alphabet. It
abstains by raising whenever the panels, their order, a cell tone, or the figure
decision is ambiguous.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "rung_deal_interleave_reveal-oracle-0.1.0"

LATTICE = 9
PANEL_ROWS = 3
MIN_PANEL_W = 104
MAX_PANEL_W = 124
MIN_PANEL_H = 34
MAX_PANEL_H = 50
MIN_ASPECT = 2.0
MAX_ASPECT = 3.4
MIN_BAND_GAP = 60
MAX_FIGURE_ERRORS = 2
MIN_FIGURE_GAP = 3

RUNG_ROWS = (3, 1, 5)
LEFT_RUNG = (1, 2, 3, 4)
RIGHT_RUNG = (4, 5, 6, 7)
POSTS = (0, LATTICE - 1)
RIGHT_ROW = {"ladder": None, "top_hook": 1, "mid_hook": 3, "foot_hook": 5}


def _templates() -> dict[str, np.ndarray]:
    grids: dict[str, np.ndarray] = {}
    for name, right_row in RIGHT_ROW.items():
        grid = np.zeros((LATTICE, LATTICE), dtype=np.uint8)
        for row in range(LATTICE):
            for col in POSTS:
                grid[row, col] = 1
        for row in RUNG_ROWS:
            for col in (RIGHT_RUNG if right_row == row else LEFT_RUNG):
                grid[row, col] = 1
        grids[name] = grid
    return grids


TEMPLATES = _templates()


def _ink_mask(array: np.ndarray) -> np.ndarray:
    luma = 0.299 * array[:, :, 0] + 0.587 * array[:, :, 1] + 0.114 * array[:, :, 2]
    darkest = float(np.percentile(luma, 0.2))
    lightest = float(np.percentile(luma, 99.8))
    if lightest - darkest < 20.0:
        raise ValueError("image has no usable ink-to-paper contrast")
    return luma < 0.5 * (darkest + lightest)


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int]]:
    """Return bounding boxes of 4-connected ink components, largest first."""

    seen = np.zeros(mask.shape, dtype=bool)
    boxes: list[tuple[int, int, int, int, int]] = []
    height, width = mask.shape
    for start_y, start_x in np.argwhere(mask):
        if seen[start_y, start_x]:
            continue
        stack = [(int(start_y), int(start_x))]
        seen[start_y, start_x] = True
        y0 = y1 = int(start_y)
        x0 = x1 = int(start_x)
        size = 0
        while stack:
            cy, cx = stack.pop()
            size += 1
            y0, y1 = min(y0, cy), max(y1, cy)
            x0, x1 = min(x0, cx), max(x1, cx)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        boxes.append((size, y0, x0, y1, x1))
    boxes.sort(reverse=True)
    return [(y0, x0, y1, x1) for _, y0, x0, y1, x1 in boxes]


def _panels(mask: np.ndarray) -> list[tuple[int, int, int, int]]:
    found = []
    for y0, x0, y1, x1 in _components(mask):
        height = y1 - y0 + 1
        width = x1 - x0 + 1
        if not (MIN_PANEL_H <= height <= MAX_PANEL_H):
            continue
        if not (MIN_PANEL_W <= width <= MAX_PANEL_W):
            continue
        if not (MIN_ASPECT <= width / height <= MAX_ASPECT):
            continue
        found.append((y0, x0, y1, x1))
    if len(found) != PANEL_ROWS:
        raise ValueError(f"expected three bordered panels, found {len(found)}")
    found.sort()
    for earlier, later in zip(found, found[1:]):
        if later[0] - earlier[0] < MIN_BAND_GAP:
            raise ValueError("panel top edges are not separated enough to order them")
    return found


def _frame_thickness(mask: np.ndarray, box: tuple[int, int, int, int]) -> int:
    y0, x0, y1, x1 = box
    window = mask[y0:y1 + 1, x0:x1 + 1]
    runs = []
    for lines in (window, window[:, ::-1], window.T, window.T[:, ::-1]):
        for line in lines:
            run = 0
            while run < line.size and line[run]:
                run += 1
            if run:
                runs.append(int(run))
    if not runs:
        raise ValueError("panel has no measurable border")
    thickness = min(runs)
    if thickness < 2 or thickness > 8:
        raise ValueError("panel border thickness is out of range")
    return thickness


def _read_rows(mask: np.ndarray, box: tuple[int, int, int, int], frame: int) -> np.ndarray:
    y0, x0, y1, x1 = box
    inner_h = (y1 + 1 - frame) - (y0 + frame)
    inner_w = (x1 + 1 - frame) - (x0 + frame)
    if inner_h < PANEL_ROWS * 6 or inner_w < LATTICE * 6:
        raise ValueError("panel interior is too small to hold a three-by-nine lattice")
    rows = np.zeros((PANEL_ROWS, LATTICE), dtype=np.uint8)
    for row in range(PANEL_ROWS):
        for col in range(LATTICE):
            cy0 = y0 + frame + round(row * inner_h / PANEL_ROWS)
            cy1 = y0 + frame + round((row + 1) * inner_h / PANEL_ROWS)
            cx0 = x0 + frame + round(col * inner_w / LATTICE)
            cx1 = x0 + frame + round((col + 1) * inner_w / LATTICE)
            patch = mask[cy0 + 2:cy1 - 2, cx0 + 2:cx1 - 2]
            if patch.size == 0:
                raise ValueError("panel cell patch is empty")
            share = float(patch.mean())
            if 0.2 < share < 0.8:
                raise ValueError("panel cell tone is ambiguous")
            rows[row, col] = 1 if share >= 0.8 else 0
    return rows


def decision_from_image(image: Image.Image) -> str:
    """Recover the dealt figure from the final pixels alone."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    mask = _ink_mask(array)
    boxes = _panels(mask)
    grid = np.zeros((LATTICE, LATTICE), dtype=np.uint8)
    for index, box in enumerate(boxes):
        frame = _frame_thickness(mask, box)
        rows = _read_rows(mask, box, frame)
        for step in range(PANEL_ROWS):
            grid[index + 3 * step] = rows[step]

    scores = sorted(
        (int(np.abs(grid.astype(int) - template.astype(int)).sum()), name)
        for name, template in TEMPLATES.items()
    )
    if scores[0][0] > MAX_FIGURE_ERRORS:
        raise ValueError("dealt lattice matches no declared figure")
    if scores[1][0] - scores[0][0] < MIN_FIGURE_GAP:
        raise ValueError("dealt lattice is ambiguous between two figures")
    return scores[0][1]
