"""Scene sampling, rendering, and analytic gold for the stripe-phase moire world.

Two square plates are printed side by side. Every cell of every plate is filled
with the same vertical ruling (same stripe width, same period); only the phase at
which a cell's ruling starts varies. A cell is *marked* when the two plates'
rulings for that cell are more out of step than in step, which is exactly the
event that superimposing the plates closes the light gaps and fills the cell
solid. The marked cells spell one of five 7x7 symbol templates.
"""

from __future__ import annotations

import random
from typing import Iterable

import numpy as np
from PIL import Image

GRID = 7
PITCH = 8          # stripe period in pixels
CELL = 32          # cell side in pixels (four full periods)
DUTY = PITCH // 2  # dark run length in pixels

# Off-quarantine phase offsets. c(d) = 1 - 4*min(d, PITCH-d)/PITCH is the
# normalised correlation of two 50% duty square waves offset by d pixels.
IN_STEP = (0, 1, 7)        # c = 1.0, 0.5, 0.5   -> unmarked
OUT_OF_STEP = (3, 4, 5)    # c = -0.5, -1.0, -0.5 -> marked
ON_BOUNDARY = (2, 6)       # c = 0.0 exactly      -> the visible decision boundary

SYMBOL_NAMES: tuple[str, ...] = ("cross", "ring", "saltire", "tee", "wedge")

_TEMPLATE_ART: dict[str, tuple[str, ...]] = {
    "cross": (
        "...#...",
        "...#...",
        "...#...",
        "#######",
        "...#...",
        "...#...",
        "...#...",
    ),
    "ring": (
        "#######",
        "#.....#",
        "#.....#",
        "#.....#",
        "#.....#",
        "#.....#",
        "#######",
    ),
    "saltire": (
        "#.....#",
        ".#...#.",
        "..#.#..",
        "...#...",
        "..#.#..",
        ".#...#.",
        "#.....#",
    ),
    "tee": (
        "#######",
        "...#...",
        "...#...",
        "...#...",
        "...#...",
        "...#...",
        "...#...",
    ),
    "wedge": (
        "#.....#",
        "#.....#",
        ".#...#.",
        ".#...#.",
        "..#.#..",
        "..#.#..",
        "...#...",
    ),
}

TEMPLATES: dict[str, tuple[tuple[int, ...], ...]] = {
    name: tuple(tuple(1 if ch == "#" else 0 for ch in row) for row in art)
    for name, art in _TEMPLATE_ART.items()
}

# Palettes: (paper, ink, frame, sheet). Paper/ink are the two ruling tones.
PALETTES: tuple[tuple[tuple[int, int, int], ...], ...] = (
    ((243, 240, 232), (32, 38, 54), (128, 96, 40), (214, 208, 196)),
    ((238, 244, 240), (22, 60, 48), (96, 40, 112), (206, 214, 210)),
    ((246, 238, 238), (58, 24, 28), (32, 80, 128), (216, 206, 206)),
    ((236, 240, 248), (26, 30, 72), (140, 60, 32), (204, 210, 222)),
)


def _correlation(offset: int) -> float:
    """Normalised correlation of two 50%-duty square waves offset by `offset`."""
    d = offset % PITCH
    return 1.0 - 4.0 * min(d, PITCH - d) / PITCH


def _marked_grid(delta: Iterable[Iterable[int]]) -> tuple[tuple[int, ...], ...]:
    """A cell is marked exactly when its two rulings correlate negatively."""
    return tuple(
        tuple(1 if _correlation(int(d)) < 0.0 else 0 for d in row) for row in delta
    )


def _hamming(a, b) -> int:
    return sum(
        1
        for i in range(GRID)
        for j in range(GRID)
        if int(a[i][j]) != int(b[i][j])
    )


def _match(grid) -> tuple[str, int, int]:
    """Nearest template by Hamming distance; returns (name, best, runner_up)."""
    scored = sorted((_hamming(grid, TEMPLATES[n]), n) for n in SYMBOL_NAMES)
    return scored[0][1], scored[0][0], scored[1][0]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    symbol = SYMBOL_NAMES[seed % len(SYMBOL_NAMES)]  # balanced across seeds
    target = [list(row) for row in TEMPLATES[symbol]]

    flips: list[list[int]] = []
    for _ in range(rng.choice((0, 0, 1, 1, 2))):
        cell = [rng.randrange(GRID), rng.randrange(GRID)]
        if cell not in flips:
            flips.append(cell)
    for r, c in flips:
        target[r][c] ^= 1

    # One cell in roughly a sixth of the scenes is printed exactly on the
    # visible decision boundary (a quarter period out of step).
    ambiguous: list[list[int]] = []
    if rng.randrange(6) == 0:
        ambiguous.append([rng.randrange(GRID), rng.randrange(GRID)])

    delta: list[list[int]] = []
    phi: list[list[int]] = []
    for r in range(GRID):
        drow: list[int] = []
        prow: list[int] = []
        for c in range(GRID):
            if [r, c] in ambiguous:
                drow.append(rng.choice(ON_BOUNDARY))
            elif target[r][c]:
                drow.append(rng.choice(OUT_OF_STEP))
            else:
                drow.append(rng.choice(IN_STEP))
            prow.append(rng.randrange(PITCH))
        delta.append(drow)
        phi.append(prow)

    paper, ink, frame, sheet = PALETTES[rng.randrange(len(PALETTES))]
    return {
        "symbol": symbol,
        "flips": [list(f) for f in flips],
        "ambiguous": [list(a) for a in ambiguous],
        "phi": phi,
        "delta": delta,
        "border": 4,
        "rule": 1,
        "pad": 20 + 2 * rng.randrange(4),
        "gap": 40 + 4 * rng.randrange(4),
        "paper": list(paper),
        "ink": list(ink),
        "frame": list(frame),
        "sheet": list(sheet),
    }


def _plate(scene: dict, which: int) -> np.ndarray:
    """Render one plate interior (GRID*CELL square) as an RGB array."""
    side = GRID * CELL
    paper = np.array(scene["paper"], dtype=np.uint8)
    ink = np.array(scene["ink"], dtype=np.uint8)
    frame = np.array(scene["frame"], dtype=np.uint8)
    out = np.empty((side, side, 3), dtype=np.uint8)
    out[:, :] = paper
    xs = np.arange(CELL)
    for r in range(GRID):
        for c in range(GRID):
            phase = int(scene["phi"][r][c])
            if which == 1:
                phase += int(scene["delta"][r][c])
            dark = ((xs + phase) % PITCH) < DUTY
            block = np.where(dark[None, :, None], ink, paper).astype(np.uint8)
            block = np.broadcast_to(block, (CELL, CELL, 3))
            out[r * CELL:(r + 1) * CELL, c * CELL:(c + 1) * CELL] = block
    # Hairline cell rules, drawn over the ruling at every interior cell border.
    rule = int(scene["rule"])
    for k in range(1, GRID):
        out[k * CELL - rule:k * CELL + rule, :] = frame
        out[:, k * CELL - rule:k * CELL + rule] = frame
    return out


def render(scene: dict) -> Image.Image:
    border = int(scene["border"])
    pad = int(scene["pad"])
    gap = int(scene["gap"])
    side = GRID * CELL
    plate = side + 2 * border
    width = 2 * pad + 2 * plate + gap
    height = 2 * pad + plate
    canvas = np.empty((height, width, 3), dtype=np.uint8)
    canvas[:, :] = np.array(scene["sheet"], dtype=np.uint8)
    for which in (0, 1):
        x0 = pad + which * (plate + gap)
        y0 = pad
        canvas[y0:y0 + plate, x0:x0 + plate] = np.array(scene["frame"], dtype=np.uint8)
        canvas[y0 + border:y0 + border + side, x0 + border:x0 + border + side] = _plate(
            scene, which
        )
    return Image.fromarray(canvas, mode="RGB")


def revealed_grid(scene: dict) -> tuple[tuple[int, ...], ...]:
    return _marked_grid(scene["delta"])


def analytic_gold(scene: dict) -> str:
    return _match(revealed_grid(scene))[0]


def margin(scene: dict) -> float:
    """Composition minimum of the phase margin and the template top gap.

    The phase term is the smallest |correlation| over the 49 cell pairs, scaled
    to cell units; the template term is the runner-up Hamming distance minus the
    best. Either term can be the binding constraint.
    """
    phase = min(
        abs(_correlation(int(d))) for row in scene["delta"] for d in row
    )
    _, best, runner_up = _match(revealed_grid(scene))
    return float(min(10.0 * phase, float(runner_up - best)))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 1.0


def naive_read_gap(scene: dict) -> float:
    """Diagnostic: how well a single plate, read without the pairing, does.

    Each plate alone yields a phase pattern (cells whose ruling starts off the
    plate's own most common phase). We report the smallest template distance
    reachable that way, minus the distance of the true reveal (always 0-2).
    A large value means the answer is not readable without the declared pairing.
    """
    best_naive = GRID * GRID
    for which in (0, 1):
        phases = [
            [
                (int(scene["phi"][r][c]) + (int(scene["delta"][r][c]) if which else 0))
                % PITCH
                for c in range(GRID)
            ]
            for r in range(GRID)
        ]
        for reference in range(PITCH):
            grid = tuple(
                tuple(1 if p != reference else 0 for p in row) for row in phases
            )
            best_naive = min(best_naive, _match(grid)[1])
    _, best_true, _ = _match(revealed_grid(scene))
    return float(best_naive - best_true)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Declared pixel-identical latent transforms.

    Cell phase and cell offset are angles of a periodic ruling: they are defined
    modulo PITCH, and both the raster and the gold depend only on their residues.
    """
    out: list[tuple[str, dict]] = []

    wrapped = {k: v for k, v in scene.items()}
    wrapped["phi"] = [
        [int(p) + PITCH * ((r + c) % 3) for c, p in enumerate(row)]
        for r, row in enumerate(scene["phi"])
    ]
    out.append(("phase_wrap_by_period", wrapped))

    shifted = {k: v for k, v in scene.items()}
    shifted["delta"] = [
        [int(d) - PITCH for d in row] for row in scene["delta"]
    ]
    out.append(("offset_wrap_negative_period", shifted))

    both = {k: v for k, v in scene.items()}
    both["phi"] = [[int(p) + 2 * PITCH for p in row] for row in scene["phi"]]
    both["delta"] = [[int(d) + 3 * PITCH for d in row] for row in scene["delta"]]
    out.append(("phase_and_offset_wrap", both))
    return out
