"""Independent pixel-only inverse arm for stroke-weight correspondence.

Receives ONLY the final rasterized PNG and the public question contract (this
file's own visual spec). It never imports the renderer, prompts, generator,
verifier, or analytic-gold code, and never reads scene data.

Public visual spec:

  * the top panel holds one neutral gray horizontal stroke (the reference) of a
    fixed length but a certain stroke weight (how tall/thick the band is);
  * a horizontal dark divider separates it from a column of three colored
    horizontal strokes, one red, one green, one blue, each of the same length and
    each with a distinct stroke weight;
  * the task: decide which stroke color has exactly the same stroke weight as
    the gray reference.

The oracle, from pixels alone: finds the divider by the long dark horizontal
band, separates the gray reference band above it from the three colored bands
below, measures each band's stroke weight as the number of consecutive filled
rows (its transverse thickness), and returns the color of the candidate band
whose measured weight equals the reference band's.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "line_weight_correspondence-oracle-0.1.0"

PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (235, 40, 40),
    "green": (0, 170, 70),
    "blue": (40, 60, 235),
}
GRAY = (150, 150, 150)
_DARK = 100.0
_COLOR_DIST = 80.0
_ROW_FRACTION = 0.4


def _classify(arr: np.ndarray, color: tuple[int, int, int]) -> np.ndarray:
    c = np.asarray(color, dtype=np.float32)
    d = np.linalg.norm(arr - c[None, None, :], axis=-1)
    return d < _COLOR_DIST


def _band_rows(mask: np.ndarray, divider_mid: float, above: bool) -> tuple[int, int] | None:
    """Return (rmin, rmax) row span of mask pixels on the given side of divider."""
    rows = np.where(mask.any(axis=1))[0]
    if rows.size == 0:
        return None
    if above:
        rows = rows[rows < divider_mid]
    else:
        rows = rows[rows > divider_mid]
    if rows.size == 0:
        return None
    return (int(rows.min()), int(rows.max()))


def _thickness(span: tuple[int, int] | None) -> int:
    if span is None:
        raise ValueError("stroke band not found")
    return span[1] - span[0] + 1


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.float32)

    dark = arr.max(axis=2) < _DARK
    line_counts = dark.sum(axis=1)
    grid = line_counts > _ROW_FRACTION * dark.shape[1]
    rows = np.where(grid)[0]
    if rows.size == 0:
        raise ValueError("no divider line found")
    divider_mid = float(rows.mean())

    gray_mask = _classify(arr, GRAY)
    ref_span = _band_rows(gray_mask, divider_mid, above=True)
    ref_thick = _thickness(ref_span)

    color_masks = {name: _classify(arr, rgb) for name, rgb in PALETTE.items()}

    matched = None
    for name, mask in color_masks.items():
        span = _band_rows(mask, divider_mid, above=False)
        thick = _thickness(span)
        if thick == ref_thick:
            if matched is not None:
                raise ValueError(f"ambiguous: more than one candidate matched weight {ref_thick}")
            matched = name
    if matched is None:
        raise ValueError(f"no candidate matched reference stroke weight {ref_thick}")
    return matched


if __name__ == "__main__":
    import sys

    img = Image.open(sys.argv[1]).convert("RGB")
    print(decision_from_image(img))
