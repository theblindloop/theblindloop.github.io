"""Cross-panel stroke-weight (line thickness) correspondence renderer.

A single neutral gray horizontal stroke at the top is the reference: it has a
specific stroke weight (perpendicular cross-section thickness) chosen from a
finite canonical set. Below a divider sit three colored horizontal strokes A, B,
C (red / green / blue), each of the identical fixed length but a distinct stroke
weight. Exactly one candidate has the same stroke weight as the reference. The
question asks for that candidate's color.

The pixel-only inverse arm must, for every panel, locate the stroke and measure
its transverse line weight (the number of filled rows of its horizontal band),
establish the cross-panel correspondence by stroke-weight equality, and only
then read out the bound color of the single matched stroke.

Latent state z (every field that influences the gold is rendered -- nothing
hidden):

    ref_thick   : stroke weight (pixels) of the gray reference stroke, from the
                  canonical set CANONICAL_THICK.
    cand_thick  : length-3 list of candidate stroke weights; exactly one equals
                  ``ref_thick``, the other two are distinct and different.
    cand_colors : the fixed 3-color palette bound one-to-one to the candidates;
                  each candidate stroke is drawn in its bound color.
    stroke_len  : the fixed, equal horizontal length shared by every stroke
                  (reference and candidates) so length and position cannot
                  identify the answer.
    ref_center, cand_centers, divider_y, width, height : fixed geometry.

The analytic gold is the color of the candidate whose stroke weight equals the
reference's. Because the reference weight, each candidate weight, and the bound
colors are all rendered verbatim as full horizontal bands, the gold is
recoverable from the pixels alone.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "line_weight_correspondence-0.1.0"

BACKGROUND = "white"
DIVIDER = (70, 70, 70)
GRAY = (150, 150, 150)

PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (235, 40, 40),
    "green": (0, 170, 70),
    "blue": (40, 60, 235),
}
CAND_COLOR_ORDER = ["red", "green", "blue"]

# Canonical stroke weights (pixels). Adjacent values are separated by 4 pixels,
# far above the finite-resolution oracle tolerance and never quarantined.
CANONICAL_THICK = [6, 10, 14, 18, 22]

WIDTH = 900
HEIGHT = 760
DIVIDER_Y = 195
DIVIDER_THICK = 3
STROKE_LEN = 300
REF_CENTER = (450, 100)
CAND_CENTERS = [(450, 330), (450, 470), (450, 610)]

_QUARANTINE_FLOOR = 3.0


def _draw_stroke(draw, center, weight: int, color) -> None:
    """Draw one horizontal stroke (a full-width band) of the given weight.

    The stroke spans ``weight`` consecutive pixel rows centered on ``center`` so
    its transverse line weight is exactly ``weight`` pixels.
    """
    cx, cy = center
    half = weight // 2
    top = cy - half
    bottom = cy + half - 1
    left = cx - STROKE_LEN // 2
    right = cx + STROKE_LEN // 2 - 1
    draw.rectangle([left, top, right, bottom], fill=color)


def sample_scene(seed: int) -> dict:
    """Sample a balanced, deterministic latent scene (dict)."""
    rng = random.Random(seed)
    ref_thick = rng.choice(CANONICAL_THICK)
    pool = [t for t in CANONICAL_THICK if t != ref_thick]
    rng.shuffle(pool)
    distractors = pool[:2]
    cand_thick = [ref_thick] + distractors
    rng.shuffle(cand_thick)

    return {
        "ref_thick": ref_thick,
        "cand_thick": cand_thick,
        "cand_colors": list(PALETTE.keys()),
        "stroke_len": STROKE_LEN,
        "ref_center": list(REF_CENTER),
        "cand_centers": [[c[0], c[1]] for c in CAND_CENTERS],
        "divider_y": DIVIDER_Y,
        "width": WIDTH,
        "height": HEIGHT,
    }


def render(scene: dict) -> Image.Image:
    """Render I = R(z) deterministically at pixel resolution."""
    w = int(scene["width"])
    h = int(scene["height"])
    img = Image.new("RGB", (w, h), BACKGROUND)
    draw = ImageDraw.Draw(img)
    _draw_stroke(draw, tuple(scene["ref_center"]), scene["ref_thick"], GRAY)
    for i, center in enumerate(scene["cand_centers"]):
        color = scene["cand_colors"][i]
        _draw_stroke(draw, tuple(center), scene["cand_thick"][i], PALETTE[color])
    draw.rectangle([0, scene["divider_y"], w, scene["divider_y"] + DIVIDER_THICK], fill=DIVIDER)
    return img


def analytic_gold(scene: dict) -> str:
    """Decision: color (red/green/blue) of the stroke with the reference weight."""
    idx = scene["cand_thick"].index(scene["ref_thick"])
    return scene["cand_colors"][idx]


def margin(scene: dict) -> float:
    """Runner-up margin: minimum absolute stroke-weight difference (pixels)
    between the reference stroke and any distractor candidate."""
    ref = scene["ref_thick"]
    diffs = [abs(ref - t) for t in scene["cand_thick"] if t != ref]
    return float(min(diffs)) if diffs else 0.0


def is_quarantined(scene: dict) -> bool:
    """Quarantine only a scene where a distractor comes within
    _QUARANTINE_FLOOR pixels of the reference weight (never produced because
    canonical weights are separated by 4 pixels)."""
    return margin(scene) < _QUARANTINE_FLOOR


def latent_symmetries(scene: dict) -> list:
    """not_applicable: every label-influencing field (reference stroke weight,
    each candidate weight, and the bound colors) is rendered verbatim as a full
    horizontal band, so the raster uniquely determines the gold; no
    pixel-identical transform exists."""
    return []
