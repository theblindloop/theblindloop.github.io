"""Deterministic visual world for signed dot-change analogy search.

The analytic arm reads the sampled scene.  The separately implemented oracle
receives only the final raster and recovers all counts from colored components.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 720
HEIGHT = 520
BACKGROUND = "#fafaf7"
PANEL_FILL = "#ffffff"
PANEL_OUTLINE = "#d1d5db"
TEXT = "#111827"
MUTED_TEXT = "#6b7280"
REFERENCE_COLOR = "#4b5563"
COLORS: dict[str, str] = {
    "red": "#d92d20",
    "blue": "#2563eb",
    "green": "#16803c",
    "purple": "#7e22ce",
}

LEFT_BOX = (38, 0, 206, 0)
RIGHT_BOX = (514, 0, 682, 0)
REFERENCE_Y = 102
ROW_YS = (204, 286, 368, 450)
BOX_HALF_HEIGHT = 30
DOT_RADIUS = 8


def _font(size: int, *, bold: bool = False) -> ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size=size)
    except OSError:
        return ImageFont.load_default()


def _pair(low: int, magnitude: int, sign: int) -> list[int]:
    high = low + magnitude
    return [low, high] if sign > 0 else [high, low]


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced, uniquely answerable count-relation analogy."""

    rng = random.Random(int(seed) ^ 0x5D07A)
    target_color = tuple(COLORS)[int(seed) % len(COLORS)]
    magnitude = 2 + ((int(seed) // 4) % 2)
    sign = 1 if ((int(seed) // 8) % 2 == 0) else -1
    valid_lows = list(range(1, 7 - magnitude))
    target_low = valid_lows[rng.randrange(len(valid_lows))]
    reference_choices = [value for value in valid_lows if value != target_low]
    reference_low = reference_choices[rng.randrange(len(reference_choices))]

    target = _pair(target_low, magnitude, sign)
    left, right = target
    # Each cheap cue is explicitly duplicated: one distractor shares the target
    # left count, one shares its right count, and the reversed pair shares its
    # total and absolute difference.  Only binding both ordered endpoints wins.
    if sign > 0:
        distractors = ([left, right - 1], [left + 1, right], [right, left])
    else:
        distractors = ([left, right + 1], [left - 1, right], [right, left])

    other_colors = [color for color in COLORS if color != target_color]
    rng.shuffle(other_colors)
    candidate_pairs: dict[str, list[int]] = {target_color: list(target)}
    for color, pair in zip(other_colors, distractors, strict=True):
        candidate_pairs[color] = list(pair)

    target_row = (int(seed) // 4) % 4
    remaining = [color for color in COLORS if color != target_color]
    rng.shuffle(remaining)
    row_order = list(remaining)
    row_order.insert(target_row, target_color)

    return {
        "reference_counts": _pair(reference_low, magnitude, sign),
        "candidate_pairs": candidate_pairs,
        "row_order": row_order,
    }


def _delta(pair: list[int]) -> int:
    return int(pair[1]) - int(pair[0])


def analytic_gold(scene: dict[str, Any]) -> str:
    reference_delta = _delta(scene["reference_counts"])
    matches = [
        color
        for color, pair in scene["candidate_pairs"].items()
        if _delta(pair) == reference_delta
    ]
    if len(matches) != 1:
        raise ValueError(f"scene must have one signed-delta winner, found {matches}")
    return matches[0]


def margin(scene: dict[str, Any]) -> float:
    reference_delta = _delta(scene["reference_counts"])
    errors = sorted(
        abs(_delta(pair) - reference_delta)
        for pair in scene["candidate_pairs"].values()
    )
    if len(errors) < 2:
        return 0.0
    return float(errors[1] - errors[0])


def is_quarantined(scene: dict[str, Any]) -> bool:
    reference_delta = _delta(scene["reference_counts"])
    winners = sum(
        _delta(pair) == reference_delta for pair in scene["candidate_pairs"].values()
    )
    return winners != 1 or margin(scene) < 1.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []


def _dot_centers(count: int, box: tuple[int, int, int, int], y: int) -> list[tuple[int, int]]:
    if not 1 <= int(count) <= 6:
        raise ValueError(f"dot count outside visible range: {count}")
    x0, _, x1, _ = box
    columns = 3
    x_values = (x0 + 45, (x0 + x1) // 2, x1 - 45)
    y_values = (y - 11, y + 11)
    slots = [(x_values[col], y_values[row]) for row in range(2) for col in range(columns)]
    # Center the one-row cases while retaining exactly separated components.
    if count <= 3:
        return [(x_values[col], y) for col in range(count)]
    return slots[:count]


def _draw_group(
    draw: ImageDraw.ImageDraw,
    box: tuple[int, int, int, int],
    y: int,
    count: int,
    color: str,
) -> None:
    x0, _, x1, _ = box
    draw.rounded_rectangle(
        (x0, y - BOX_HALF_HEIGHT, x1, y + BOX_HALF_HEIGHT),
        radius=12,
        fill=PANEL_FILL,
        outline=PANEL_OUTLINE,
        width=2,
    )
    for cx, cy in _dot_centers(int(count), box, y):
        draw.ellipse(
            (cx - DOT_RADIUS, cy - DOT_RADIUS, cx + DOT_RADIUS, cy + DOT_RADIUS),
            fill=color,
        )


def _centered_text(
    draw: ImageDraw.ImageDraw,
    xy: tuple[int, int],
    value: str,
    *,
    font: ImageFont.ImageFont,
    fill: str,
) -> None:
    draw.text(xy, value, font=font, fill=fill, anchor="mm")


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    _centered_text(
        draw,
        (WIDTH // 2, 25),
        "MATCH THE SIGNED DOT CHANGE",
        font=_font(20, bold=True),
        fill=TEXT,
    )
    _centered_text(draw, (122, 53), "LEFT", font=_font(13, bold=True), fill=MUTED_TEXT)
    _centered_text(draw, (598, 53), "RIGHT", font=_font(13, bold=True), fill=MUTED_TEXT)
    _centered_text(draw, (360, 67), "GRAY REFERENCE", font=_font(13, bold=True), fill=TEXT)

    reference = scene["reference_counts"]
    _draw_group(draw, LEFT_BOX, REFERENCE_Y, int(reference[0]), REFERENCE_COLOR)
    _draw_group(draw, RIGHT_BOX, REFERENCE_Y, int(reference[1]), REFERENCE_COLOR)
    draw.line((244, REFERENCE_Y, 476, REFERENCE_Y), fill="#9ca3af", width=2)
    draw.polygon(
        ((476, REFERENCE_Y), (464, REFERENCE_Y - 7), (464, REFERENCE_Y + 7)),
        fill="#9ca3af",
    )

    draw.line((28, 151, WIDTH - 28, 151), fill="#e5e7eb", width=2)
    _centered_text(draw, (360, 169), "COLORED CANDIDATE PAIRS", font=_font(12, bold=True), fill=MUTED_TEXT)

    pairs = scene["candidate_pairs"]
    order = scene["row_order"]
    if sorted(order) != sorted(COLORS) or set(pairs) != set(COLORS):
        raise ValueError("candidate colors or row order are incomplete")
    for y, color_name in zip(ROW_YS, order, strict=True):
        pair = pairs[color_name]
        color = COLORS[color_name]
        _draw_group(draw, LEFT_BOX, y, int(pair[0]), color)
        _draw_group(draw, RIGHT_BOX, y, int(pair[1]), color)
        _centered_text(draw, (360, y), color_name.upper(), font=_font(14, bold=True), fill=TEXT)
        draw.line((225, y, 310, y), fill="#e5e7eb", width=2)
        draw.line((410, y, 495, y), fill="#e5e7eb", width=2)

    return image
