"""Pixel-only inverse arm for the sandheap topple inset.

Receives one rasterised PNG and nothing else. It locates the framed block from
the dark-ink bounding box, recovers the internal rule lattice from dark row and
column projections, reads each cell's dot count and the blue drop marker by
colour segmentation plus connected components, counts the green tally squares,
then independently replays the declared toppling rule and reports the final dot
count of the drop cell.

No renderer constant, latent field, layout table or analytic answer is imported:
every geometric quantity below is measured from the image.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "sandheap_topple_inset-oracle-0.1.0"

DARK_MAX = 90
MIN_BLOB = 8


def _groups(flags: np.ndarray) -> list[tuple[int, int]]:
    """Maximal runs of True as (start, stop_exclusive) index pairs."""
    runs = []
    start = None
    for index, value in enumerate(flags):
        if value and start is None:
            start = index
        elif not value and start is not None:
            runs.append((start, index))
            start = None
    if start is not None:
        runs.append((start, len(flags)))
    return runs


def _components(mask: np.ndarray) -> int:
    """Count 4-connected True components of at least ``MIN_BLOB`` pixels."""
    seen = np.zeros(mask.shape, dtype=bool)
    total = 0
    height, width = mask.shape
    for y in range(height):
        for x in range(width):
            if not mask[y, x] or seen[y, x]:
                continue
            queue = deque([(y, x)])
            seen[y, x] = True
            size = 0
            while queue:
                cy, cx = queue.popleft()
                size += 1
                for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
            if size >= MIN_BLOB:
                total += 1
    return total


def decision_from_image(image: Image.Image) -> str:
    """Return the final dot count of the drop cell, read from pixels alone."""
    rgb = np.asarray(image.convert("RGB")).astype(np.int16)
    red, green, blue = rgb[:, :, 0], rgb[:, :, 1], rgb[:, :, 2]

    dark = (red < DARK_MAX) & (green < DARK_MAX) & (blue < DARK_MAX)
    if not dark.any():
        raise ValueError("no framed block found")
    ys, xs = np.nonzero(dark)
    y0, y1 = int(ys.min()), int(ys.max())
    x0, x1 = int(xs.min()), int(xs.max())
    block = dark[y0 : y1 + 1, x0 : x1 + 1]
    height, width = block.shape

    row_full = block.sum(axis=1) >= 0.85 * width
    row_runs = _groups(row_full)
    if len(row_runs) < 5:
        raise ValueError(f"expected five horizontal rules, found {len(row_runs)}")
    grid_runs = row_runs[:4]
    tally_run = row_runs[4]

    grid_top, grid_bottom = grid_runs[0][0], grid_runs[3][1]
    column_span = block[grid_top:grid_bottom, :]
    col_full = column_span.sum(axis=0) >= 0.85 * column_span.shape[0]
    col_runs = _groups(col_full)
    if len(col_runs) != 4:
        raise ValueError(f"expected four vertical rules, found {len(col_runs)}")

    pip = (red > 140) & (green < 110) & (blue < 110)
    ring = (blue > 140) & (red < 110) & (green < 110)
    tally = (green > 110) & (red < 110) & (blue < 140)
    pip = pip[y0 : y1 + 1, x0 : x1 + 1]
    ring = ring[y0 : y1 + 1, x0 : x1 + 1]
    tally = tally[y0 : y1 + 1, x0 : x1 + 1]

    cells = []
    drop_cell = None
    for row in range(3):
        counts = []
        top, bottom = grid_runs[row][1], grid_runs[row + 1][0]
        for col in range(3):
            left, right = col_runs[col][1], col_runs[col + 1][0]
            counts.append(_components(pip[top:bottom, left:right]))
            if ring[top:bottom, left:right].sum() >= MIN_BLOB:
                if drop_cell is not None:
                    raise ValueError("more than one drop marker")
                drop_cell = (row, col)
        cells.append(counts)
    if drop_cell is None:
        raise ValueError("no drop marker found")
    if any(count > 3 for row in cells for count in row):
        raise ValueError("cell holds more dots than the declared range")

    drops = _components(tally[grid_runs[3][1] : tally_run[0], :])
    if drops < 1:
        raise ValueError("no tally squares found")

    grid = [list(row) for row in cells]
    for _ in range(drops):
        grid[drop_cell[0]][drop_cell[1]] += 1
        guard = 0
        while True:
            hot = [(r, c) for r in range(3) for c in range(3) if grid[r][c] >= 4]
            if not hot:
                break
            guard += 1
            if guard > 500:
                raise ValueError("relaxation did not terminate")
            for r, c in hot:
                grid[r][c] -= 4
                for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < 3 and 0 <= nc < 3:
                        grid[nr][nc] += 1
    return str(grid[drop_cell[0]][drop_cell[1]])
