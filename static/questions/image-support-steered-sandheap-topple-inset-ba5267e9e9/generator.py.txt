"""Sandheap topple inset: abelian toppling replay inside one small framed block.

Latent scene z = {cells, ring, drops, pip_radius, origin, decor, canvas,
topple_order}. The visible block holds a 3x3 grid of cells (0-3 red dots each),
a blue outline marking the drop cell, and a bottom strip of green tally squares
giving the number of grains dropped on that cell. Grains are added one at a
time; any cell reaching four dots topples, losing four dots and sending one to
each orthogonal neighbour inside the grid (dots leaving the grid are lost).

The relaxation is abelian: the stable configuration is independent of the order
in which topples fire, so the declared replay has exactly one outcome. The
recorded ``topple_order`` is therefore a pure latent alias -- it never reaches
the raster and never reaches the analytic gold.

All decisive evidence lives inside one small connected block whose absolute
position is randomised; every mark outside the frame is pale grey decoration.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "sandheap_topple_inset-0.1.0"

GRID = 3
CELL = 34
LINE = 3
TALLY_H = 22
FRAME = 5
CANVAS = 760

INK_DARK = (18, 18, 20)
INK_PIP = (196, 28, 44)
INK_RING = (26, 66, 198)
INK_TALLY = (24, 132, 68)
PAPER = (255, 255, 255)
DECOR = (200, 200, 200)

# Fixed dot slots inside a cell (centres, cell-local pixels).
SLOTS = ((9, 9), (25, 9), (17, 25))
SLOT_PITCH = 16.0  # centre-to-centre distance between the two nearest slots
QUARANTINE_MARGIN = 3.0


def block_size() -> tuple[int, int]:
    """Outer pixel size of the framed block."""
    inner_w = GRID * CELL + (GRID + 1) * LINE
    inner_h = GRID * CELL + (GRID + 1) * LINE + TALLY_H + LINE
    return inner_w + 2 * (FRAME - LINE), inner_h + 2 * (FRAME - LINE)


def sample_scene(seed: int) -> dict:
    """Sample one latent scene deterministically from ``seed``."""
    rng = random.Random(seed * 7919 + 13)
    target = seed % 4  # round-robin answer target keeps classes balanced
    for _ in range(4000):
        cells = [[rng.randint(0, 3) for _ in range(GRID)] for _ in range(GRID)]
        ring = [rng.randrange(GRID), rng.randrange(GRID)]
        drops = rng.randint(1, 5)
        final, topples = _relax(cells, ring, drops)
        if topples < 1:
            continue
        if final[ring[0]][ring[1]] != target:
            continue
        break
    else:  # pragma: no cover - exhaustive search always succeeds in practice
        raise RuntimeError("scene sampling failed")

    pip_radius = 7 if rng.random() < 0.14 else rng.choice([4, 5, 6])
    bw, bh = block_size()
    origin = [rng.randint(12, CANVAS - bw - 12), rng.randint(12, CANVAS - bh - 12)]
    decor = _sample_decor(rng)
    order = list(range(9))
    rng.shuffle(order)
    return {
        "cells": cells,
        "ring": ring,
        "drops": drops,
        "pip_radius": pip_radius,
        "origin": origin,
        "decor": decor,
        "canvas": [CANVAS, CANVAS],
        "topple_order": order,
    }


def _sample_decor(rng: random.Random) -> list:
    """Pale grey, non-diagnostic marks scattered over the whole canvas."""
    marks = []
    for _ in range(rng.randint(7, 12)):
        kind = rng.choice(["rect", "ellipse", "line"])
        x = rng.randint(4, CANVAS - 70)
        y = rng.randint(4, CANVAS - 70)
        w = rng.randint(16, 64)
        h = rng.randint(16, 64)
        marks.append([kind, x, y, w, h, rng.choice([1, 2, 3])])
    return marks


def _relax(cells, ring, drops):
    """Add ``drops`` grains to ``ring`` and relax; returns (stable grid, topples)."""
    grid = [list(row) for row in cells]
    topples = 0
    for _ in range(drops):
        grid[ring[0]][ring[1]] += 1
        while True:
            hot = [(r, c) for r in range(GRID) for c in range(GRID) if grid[r][c] >= 4]
            if not hot:
                break
            for r, c in hot:
                grid[r][c] -= 4
                topples += 1
                for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < GRID and 0 <= nc < GRID:
                        grid[nr][nc] += 1
    return grid, topples


def analytic_gold(scene: dict) -> str:
    """Final dot count in the ringed drop cell after the declared replay."""
    cells = scene["cells"]
    ring = scene["ring"]
    drops = scene["drops"]
    final, _ = _relax(cells, ring, drops)
    return str(final[ring[0]][ring[1]])


def margin(scene: dict) -> float:
    """Smallest decisive pixel clearance: dot-to-dot gap or dot-to-rule gap."""
    radius = float(scene["pip_radius"])
    dot_gap = SLOT_PITCH - 2.0 * radius
    rule_gap = float(min(SLOTS[0])) - radius
    return round(min(dot_gap, rule_gap), 6)


def is_quarantined(scene: dict) -> bool:
    """Dots crowd their neighbours or the cell rule below reliable resolution."""
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Toppling order is a latent alias: abelian relaxation, and it is not drawn."""
    twin = dict(scene)
    twin["topple_order"] = list(reversed(scene["topple_order"]))
    rotated = dict(scene)
    order = list(scene["topple_order"])
    rotated["topple_order"] = order[3:] + order[:3]
    return [("topple_order_reversed", twin), ("topple_order_rotated", rotated)]


def render(scene: dict) -> Image.Image:
    """Rasterise the scene: grey decoration, then the framed evidence block."""
    width, height = scene["canvas"]
    image = Image.new("RGB", (width, height), PAPER)
    draw = ImageDraw.Draw(image)

    for kind, x, y, w, h, thick in scene["decor"]:
        if kind == "rect":
            draw.rectangle([x, y, x + w, y + h], outline=DECOR, width=thick)
        elif kind == "ellipse":
            draw.ellipse([x, y, x + w, y + h], outline=DECOR, width=thick)
        else:
            draw.line([x, y, x + w, y + h], fill=DECOR, width=thick)

    ox, oy = scene["origin"]
    bw, bh = block_size()
    draw.rectangle([ox - 6, oy - 6, ox + bw + 6, oy + bh + 6], fill=PAPER)

    pad = FRAME - LINE
    gx, gy = ox + pad, oy + pad
    grid_w = GRID * CELL + (GRID + 1) * LINE
    grid_h = GRID * CELL + (GRID + 1) * LINE
    block_h = grid_h + TALLY_H + LINE

    # Outer frame (thick) plus internal rules of the same ink.
    draw.rectangle(
        [ox, oy, ox + bw - 1, oy + bh - 1],
        outline=INK_DARK,
        width=FRAME,
    )
    for index in range(GRID + 1):
        y = gy + index * (CELL + LINE)
        draw.rectangle([gx, y, gx + grid_w - 1, y + LINE - 1], fill=INK_DARK)
    for index in range(GRID + 1):
        x = gx + index * (CELL + LINE)
        draw.rectangle([x, gy, x + LINE - 1, gy + grid_h - 1], fill=INK_DARK)
    y_tally = gy + block_h - LINE
    draw.rectangle([gx, y_tally, gx + grid_w - 1, y_tally + LINE - 1], fill=INK_DARK)

    radius = scene["pip_radius"]
    for row in range(GRID):
        for col in range(GRID):
            cx = gx + LINE + col * (CELL + LINE)
            cy = gy + LINE + row * (CELL + LINE)
            for slot in range(scene["cells"][row][col]):
                sx, sy = SLOTS[slot]
                draw.ellipse(
                    [cx + sx - radius, cy + sy - radius, cx + sx + radius, cy + sy + radius],
                    fill=INK_PIP,
                )
            if [row, col] == list(scene["ring"]):
                draw.rectangle(
                    [cx + 1, cy + 1, cx + CELL - 2, cy + CELL - 2],
                    outline=INK_RING,
                    width=2,
                )

    ty = gy + grid_h + LINE + 6
    for index in range(scene["drops"]):
        tx = gx + 6 + index * 16
        draw.rectangle([tx, ty, tx + 9, ty + 9], fill=INK_TALLY)
    return image
