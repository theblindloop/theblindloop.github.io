"""Independent image-only inverse for the serpentine raster world."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image, ImageDraw

TILE_SIZE = 240
MARKER_OFFSET = 18
SYMBOLS = ("sail", "hook", "rune", "comet")


def _components(mask: np.ndarray) -> list[tuple[int, float, float]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components = []
    for sy, sx in zip(*np.nonzero(mask)):
        if seen[sy, sx]:
            continue
        queue = deque([(int(sy), int(sx))])
        seen[sy, sx] = True
        count = 0
        sum_x = 0.0
        sum_y = 0.0
        while queue:
            y, x = queue.popleft()
            count += 1
            sum_x += x
            sum_y += y
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if count >= 30:
            components.append((count, sum_x / count, sum_y / count))
    return sorted(components, reverse=True)


def _locate_tile(rgb: np.ndarray) -> np.ndarray:
    spread = rgb.max(axis=2).astype(np.int16) - rgb.min(axis=2).astype(np.int16)
    mean = rgb.mean(axis=2)
    components = _components((spread > 32) & (mean < 235))
    if len(components) < 4:
        raise ValueError("four colored corner marks are required")
    marks = components[:4]
    xs = [item[1] for item in marks]
    ys = [item[2] for item in marks]
    x0 = int(round(min(xs) + MARKER_OFFSET))
    x1 = int(round(max(xs) - MARKER_OFFSET))
    y0 = int(round(min(ys) + MARKER_OFFSET))
    y1 = int(round(max(ys) - MARKER_OFFSET))
    if x1 - x0 != TILE_SIZE or y1 - y0 != TILE_SIZE:
        raise ValueError("colored marks do not define a 240 pixel square")
    tile = rgb[y0:y1, x0:x1].mean(axis=2).astype(np.float32)
    if tile.shape != (TILE_SIZE, TILE_SIZE):
        raise ValueError("marked square crop has the wrong shape")
    return tile


def _box_average(values: np.ndarray, radius: int = 4) -> np.ndarray:
    padded = np.pad(values, ((radius, radius), (radius, radius)), mode="edge")
    integral = np.pad(padded, ((1, 0), (1, 0)), mode="constant")
    integral = integral.cumsum(axis=0).cumsum(axis=1)
    size = 2 * radius + 1
    return (
        integral[size:, size:]
        - integral[:-size, size:]
        - integral[size:, :-size]
        + integral[:-size, :-size]
    ) / float(size * size)


def _glyph(symbol: str, dx: int = 0, dy: int = 0) -> np.ndarray:
    image = Image.new("L", (TILE_SIZE, TILE_SIZE), 0)
    draw = ImageDraw.Draw(image)
    width = 13

    def points(values: list[tuple[float, float]]) -> list[tuple[int, int]]:
        return [(int(round(x + dx)), int(round(y + dy))) for x, y in values]

    def line(values: list[tuple[float, float]]) -> None:
        draw.line(points(values), fill=255, width=width, joint="curve")

    if symbol == "sail":
        line([(119, 61), (119, 178)])
        line([(119, 69), (177, 139), (119, 139), (119, 69)])
        line([(89, 180), (164, 180)])
    elif symbol == "hook":
        line([(99, 60), (99, 145), (108, 164), (133, 177), (160, 174), (177, 155), (177, 125)])
        line([(99, 67), (139, 67)])
    elif symbol == "rune":
        line([(96, 71), (155, 71), (174, 94), (139, 119), (96, 119)])
        line([(96, 119), (128, 151), (174, 151)])
        line([(128, 151), (128, 178)])
    elif symbol == "comet":
        line([(84, 170), (169, 84)])
        line([(105, 149), (78, 149)])
        line([(120, 133), (82, 113)])
        line([(136, 117), (108, 87)])
        line([(151, 101), (142, 73)])
    else:
        raise ValueError(f"unknown symbol {symbol!r}")
    return np.asarray(image, dtype=np.float32) / 255.0


def _scores(restored_blur: np.ndarray) -> list[tuple[float, str]]:
    # Normalize away the unknown base and contrast, then compare a small grid of
    # translations. The average has already removed the fine carrier.
    high = float(np.percentile(restored_blur, 95))
    low = float(np.percentile(restored_blur, 5))
    scale = max(high - low, 1.0)
    observed = np.clip((high - restored_blur) / scale, 0.0, 1.0)
    observed = observed[36:204:2, 36:204:2]
    scores = []
    for symbol in SYMBOLS:
        best = float("inf")
        for dx in range(-16, 17, 2):
            for dy in range(-16, 17, 2):
                template = _glyph(symbol, dx, dy)[36:204:2, 36:204:2]
                error = float(np.mean((observed - template) ** 2))
                if error < best:
                    best = error
        scores.append((best, symbol))
    return sorted(scores)


def confidence_margin(image: Image.Image) -> float:
    rgb = np.asarray(image.convert("RGB"), dtype=np.float32)
    tile = _locate_tile(rgb)
    restored = tile.copy()
    restored[1::2] = restored[1::2, ::-1]
    blurred = _box_average(restored)
    if float(np.percentile(blurred, 95) - np.percentile(blurred, 5)) < 8.0:
        raise ValueError("restored raster has no measurable symbol contrast")
    scores = _scores(blurred)
    if len(scores) < 2:
        raise ValueError("not enough symbol templates")
    return float(scores[1][0] - scores[0][0])


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.float32)
    tile = _locate_tile(rgb)
    restored = tile.copy()
    restored[1::2] = restored[1::2, ::-1]
    blurred = _box_average(restored)
    if float(np.percentile(blurred, 95) - np.percentile(blurred, 5)) < 8.0:
        raise ValueError("restored raster has no measurable symbol contrast")
    scores = _scores(blurred)
    if not scores or len(scores) < 2 or scores[1][0] - scores[0][0] < 0.0015:
        raise ValueError("ambiguous restored symbol")
    return scores[0][1]
