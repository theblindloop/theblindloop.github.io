"""Deterministic renderer and analytic labels for the serpentine raster world."""

from __future__ import annotations

import math
from typing import Iterable

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 640
HEIGHT = 440
TILE_SIZE = 240
MARKER_OFFSET = 18
MARKER_RADIUS = 8
SYMBOLS = ("sail", "hook", "rune", "comet")

MARKER_PALETTES = (
    ((213, 58, 61), (39, 142, 207), (67, 166, 98), (221, 157, 42)),
    ((121, 76, 190), (218, 91, 48), (32, 158, 151), (198, 174, 39)),
    ((31, 137, 168), (210, 69, 103), (102, 155, 59), (224, 125, 42)),
    ((196, 72, 155), (44, 151, 92), (226, 132, 49), (55, 106, 196)),
)


def _transform_point(x: float, y: float, dx: float, dy: float) -> tuple[int, int]:
    return (int(round(x + dx)), int(round(y + dy)))


def _draw_glyph(symbol: str, dx: int, dy: int) -> np.ndarray:
    """Draw a broad, asymmetric glyph into the canonical square."""

    glyph = Image.new("L", (TILE_SIZE, TILE_SIZE), 0)
    draw = ImageDraw.Draw(glyph)
    width = 13

    def line(points: Iterable[tuple[float, float]], fill: int = 255) -> None:
        draw.line(
            [_transform_point(x, y, dx, dy) for x, y in points],
            fill=fill,
            width=width,
            joint="curve",
        )

    if symbol == "sail":
        line([(119, 61), (119, 178)])
        line([(119, 69), (177, 139), (119, 139), (119, 69)])
        line([(89, 180), (164, 180)])
    elif symbol == "hook":
        line([(99, 60), (99, 145), (108, 164), (133, 177), (160, 174), (177, 155), (177, 125)])
        line([(99, 67), (139, 67)])
    elif symbol == "rune":
        line([(96, 71), (155, 71), (174, 94), (139, 119), (96, 119)])
        line([(96, 119), (128, 151), (174, 151)])
        line([(128, 151), (128, 178)])
    elif symbol == "comet":
        line([(84, 170), (169, 84)])
        line([(105, 149), (78, 149)])
        line([(120, 133), (82, 113)])
        line([(136, 117), (108, 87)])
        line([(151, 101), (142, 73)])
    else:
        raise ValueError(f"unknown symbol {symbol!r}")
    return np.asarray(glyph, dtype=np.float32) / 255.0


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(int(seed))
    seed = int(seed)
    strength_levels = (38.0, 50.0, 62.0, 74.0)
    return {
        "seed": seed,
        "symbol": SYMBOLS[seed % len(SYMBOLS)],
        "tile_x": int(rng.integers(70, 280)),
        "tile_y": int(rng.integers(70, 150)),
        "carrier_phase": int(rng.integers(0, 8)),
        "carrier_amplitude": float(rng.integers(18, 26)),
        "signal_strength": strength_levels[(seed // len(SYMBOLS)) % len(strength_levels)],
        "base_level": int(rng.integers(182, 204)),
        "symbol_dx": int(rng.integers(-12, 13)),
        "symbol_dy": int(rng.integers(-12, 13)),
        "marker_palette_id": int(rng.integers(0, len(MARKER_PALETTES))),
    }


def _tile(scene: dict) -> np.ndarray:
    yy, xx = np.indices((TILE_SIZE, TILE_SIZE), dtype=np.float32)
    phase = (int(scene["carrier_phase"]) % 8) / 8.0
    carrier = float(scene["carrier_amplitude"]) * np.sin(
        2.0 * math.pi * (xx / 4.0 + yy / 6.0 + phase)
    )
    mask = _draw_glyph(str(scene["symbol"]), int(scene["symbol_dx"]), int(scene["symbol_dy"]))
    gray = float(scene["base_level"]) - float(scene["signal_strength"]) * mask + carrier
    return np.clip(np.rint(gray), 0, 255).astype(np.uint8)


def render(scene: dict) -> Image.Image:
    canvas = np.full((HEIGHT, WIDTH, 3), 247, dtype=np.uint8)
    tile = _tile(scene)
    x = int(scene["tile_x"])
    y = int(scene["tile_y"])
    canvas[y : y + TILE_SIZE, x : x + TILE_SIZE] = tile[:, :, None]

    # A serpentine scan stores each odd row in reverse order.
    canvas_tile = canvas[y : y + TILE_SIZE, x : x + TILE_SIZE].copy()
    canvas_tile[1::2] = canvas_tile[1::2, ::-1]
    canvas[y : y + TILE_SIZE, x : x + TILE_SIZE] = canvas_tile

    image = Image.fromarray(canvas, mode="RGB")
    draw = ImageDraw.Draw(image)
    palette = MARKER_PALETTES[int(scene["marker_palette_id"]) % len(MARKER_PALETTES)]
    centers = (
        (x - MARKER_OFFSET, y - MARKER_OFFSET),
        (x + TILE_SIZE + MARKER_OFFSET, y - MARKER_OFFSET),
        (x - MARKER_OFFSET, y + TILE_SIZE + MARKER_OFFSET),
        (x + TILE_SIZE + MARKER_OFFSET, y + TILE_SIZE + MARKER_OFFSET),
    )
    for (cx, cy), color in zip(centers, palette):
        draw.ellipse(
            (cx - MARKER_RADIUS, cy - MARKER_RADIUS, cx + MARKER_RADIUS, cy + MARKER_RADIUS),
            fill=color,
        )
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def margin(scene: dict) -> float:
    return float(scene["signal_strength"]) - float(scene["carrier_amplitude"]) - 18.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["carrier_phase"] = int(scene["carrier_phase"]) + 8
    return [("carrier_period_alias", twin)]
