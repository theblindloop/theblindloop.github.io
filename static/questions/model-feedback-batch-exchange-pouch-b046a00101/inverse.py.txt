"""Independent image-only inverse for batch exchange pouches."""

from __future__ import annotations

import numpy as np
from PIL import Image

TOKEN_COLORS = ((23, 137, 180), (151, 78, 163), (228, 132, 32), (55, 153, 93))
BOOTH_X = (24, 336, 648)
TRAY_X = (24, 258, 492, 726)


def _component_count(mask: np.ndarray) -> int | None:
    height, width = mask.shape
    remaining = set(int(value) for value in np.flatnonzero(mask))
    sizes: list[int] = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        size = 0
        while stack:
            value = stack.pop()
            y, x = divmod(value, width)
            size += 1
            for neighbor in (value - 1, value + 1, value - width, value + width):
                if neighbor in remaining:
                    ny, nx = divmod(neighbor, width)
                    if abs(nx - x) + abs(ny - y) == 1:
                        remaining.remove(neighbor)
                        stack.append(neighbor)
        if size >= 100:
            sizes.append(size)
    if any(not 150 <= size <= 230 for size in sizes):
        return None
    return len(sizes)


def _counts(arr: np.ndarray, box: tuple[int, int, int, int]) -> tuple[int, ...] | None:
    x0, y0, x1, y1 = box
    crop = arr[y0:y1, x0:x1]
    result: list[int] = []
    for color in TOKEN_COLORS:
        delta = np.max(np.abs(crop.astype(np.int16) - np.asarray(color, dtype=np.int16)), axis=2)
        count = _component_count(delta <= 2)
        if count is None:
            return None
        result.append(count)
    return tuple(result)


def _apply(start: tuple[int, ...], rules: list[tuple[int, int, int, int]]) -> tuple[int, ...] | None:
    counts = list(start)
    for source, needed, target, produced in rules:
        if source == target or needed < 1 or produced < 1:
            return None
        batches = counts[source] // needed
        counts[source] -= batches * needed
        counts[target] += batches * produced
    return tuple(counts)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.shape != (720, 960, 3):
        return "abstain"
    rules: list[tuple[int, int, int, int]] = []
    for x0 in BOOTH_X:
        left = _counts(arr, (x0 + 10, 88, x0 + 118, 188))
        right = _counts(arr, (x0 + 170, 88, x0 + 278, 188))
        if left is None or right is None:
            return "abstain"
        left_types = [index for index, count in enumerate(left) if count]
        right_types = [index for index, count in enumerate(right) if count]
        if len(left_types) != 1 or len(right_types) != 1:
            return "abstain"
        source, target = left_types[0], right_types[0]
        rules.append((source, left[source], target, right[target]))
    start = _counts(arr, (180, 246, 780, 344))
    trays = [_counts(arr, (x0 + 10, 444, x0 + 200, 688)) for x0 in TRAY_X]
    if start is None or any(counts is None for counts in trays) or not 10 <= sum(start) <= 15:
        return "abstain"
    final = _apply(start, rules)
    if final is None:
        return "abstain"
    matches = [index for index, counts in enumerate(trays) if counts == final]
    return f"tray-{chr(65 + matches[0])}" if len(matches) == 1 else "abstain"
