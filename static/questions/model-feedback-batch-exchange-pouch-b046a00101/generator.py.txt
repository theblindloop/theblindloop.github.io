"""Deterministic renderer for ordered, indivisible batch exchanges."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw, ImageFont

SIZE = (960, 720)
BACKGROUND = (247, 245, 239)
INK = (39, 43, 48)
PANEL = (255, 254, 250)
MUTED = (218, 215, 206)
ACCENT = (185, 91, 52)
TOKEN_COLORS = ((23, 137, 180), (151, 78, 163), (228, 132, 32), (55, 153, 93))
WEIGHTS = (15, 10, 4, 3)
RULES = ((0, 2, 1, 3), (1, 2, 2, 5), (2, 3, 3, 4))
DECISIONS = ("tray-A", "tray-B", "tray-C", "tray-D")
BOOTH_X = (24, 336, 648)
TRAY_X = (24, 258, 492, 726)
FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(FONT_PATH, size)


def _value(counts: tuple[int, ...] | list[int]) -> int:
    return sum(int(count) * weight for count, weight in zip(counts, WEIGHTS))


def _apply(start: tuple[int, ...] | list[int], order: tuple[int, ...] | list[int], *, once: bool = False, skip: int | None = None) -> tuple[int, ...]:
    counts = [int(value) for value in start]
    for stage, rule_index in enumerate(order):
        if stage == skip:
            continue
        source, needed, target, produced = RULES[int(rule_index)]
        batches = counts[source] // needed
        if once:
            batches = min(batches, 1)
        counts[source] -= batches * needed
        counts[target] += batches * produced
    return tuple(counts)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    for _ in range(500):
        start = tuple(rng.randint(1, 4) for _ in range(4))
        if not 10 <= sum(start) <= 15:
            continue
        order = list(range(3))
        rng.shuffle(order)
        correct = _apply(start, order)
        pool: list[tuple[int, ...]] = []
        schedules = list(itertools.permutations(range(3)))
        rng.shuffle(schedules)
        for alternate in schedules:
            pool.append(_apply(start, alternate))
        pool.append(_apply(start, order, once=True))
        pool.extend(_apply(start, order, skip=stage) for stage in range(3))
        unique: list[tuple[int, ...]] = []
        for counts in pool:
            if counts != correct and counts not in unique and _value(counts) == _value(start):
                unique.append(counts)
        if len(unique) < 3:
            continue
        rng.shuffle(unique)
        trays = unique[:3]
        correct_index = int(seed) % 4
        trays.insert(correct_index, correct)
        if max(max(counts) for counts in trays) > 40 or max(sum(counts) for counts in trays) > 46:
            continue
        return {
            "start_counts": list(start),
            "tray_counts": [list(counts) for counts in trays],
            "rule_order": order,
            "layout_variant": rng.randrange(16),
        }
    raise RuntimeError("unable to sample a valid batch-exchange scene")


def _validated(scene: dict) -> tuple[tuple[int, ...], tuple[tuple[int, ...], ...], tuple[int, ...], int]:
    start = tuple(int(value) for value in scene["start_counts"])
    trays = tuple(tuple(int(value) for value in counts) for counts in scene["tray_counts"])
    order = tuple(int(value) for value in scene["rule_order"])
    variant = int(scene["layout_variant"])
    if len(start) != 4 or len(trays) != 4 or any(len(counts) != 4 for counts in trays):
        raise ValueError("start and four trays require four token counts")
    if any(value < 0 or value > 46 for value in start) or any(value < 0 or value > 46 for counts in trays for value in counts):
        raise ValueError("token count out of range")
    if sorted(order) != [0, 1, 2] or not 0 <= variant < 16:
        raise ValueError("invalid rule order or layout")
    return start, trays, order, variant


def analytic_gold(scene: dict) -> str:
    try:
        start, trays, order, _ = _validated(scene)
        final = _apply(start, order)
        matches = [index for index, counts in enumerate(trays) if counts == final]
        return DECISIONS[matches[0]] if len(matches) == 1 else "abstain"
    except (KeyError, TypeError, ValueError, IndexError):
        return "abstain"


def margin(scene: dict) -> float:
    start, trays, order, _ = _validated(scene)
    final = _apply(start, order)
    differences = [sum(abs(a - b) for a, b in zip(counts, final)) for counts in trays]
    nonzero = [value for value in differences if value]
    return float(min(nonzero)) if differences.count(0) == 1 and nonzero else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        start, trays, _, _ = _validated(scene)
        return (
            analytic_gold(scene) == "abstain"
            or margin(scene) < 2.0
            or len(set(trays)) != 4
            or any(_value(counts) != _value(start) for counts in trays)
        )
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _tokens(draw: ImageDraw.ImageDraw, counts: tuple[int, ...] | list[int], box: tuple[int, int, int, int], columns: int, reverse: bool) -> None:
    x0, y0, x1, y1 = box
    types = [kind for kind, count in enumerate(counts) for _ in range(int(count))]
    if reverse:
        types.reverse()
    gap = 23
    rows = max(1, (len(types) + columns - 1) // columns)
    start_x = (x0 + x1) // 2 - (min(columns, max(1, len(types))) - 1) * gap // 2
    start_y = (y0 + y1) // 2 - (rows - 1) * gap // 2
    for index, kind in enumerate(types):
        cx = start_x + (index % columns) * gap
        cy = start_y + (index // columns) * gap
        draw.ellipse((cx - 8, cy - 8, cx + 8, cy + 8), fill=TOKEN_COLORS[kind], outline=INK, width=1)


def render(scene: dict) -> Image.Image:
    start, trays, order, variant = _validated(scene)
    image = Image.new("RGB", SIZE, BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font, label_font, small_font = _font(24), _font(18), _font(15)
    draw.text((24, 14), "VISIT BOOTHS 1 → 2 → 3", font=title_font, fill=INK)
    for stage, rule_index in enumerate(order):
        x0 = BOOTH_X[stage]
        draw.rounded_rectangle((x0, 52, x0 + 288, 196), radius=12, fill=PANEL, outline=INK, width=3)
        label = f"BOOTH {stage + 1}"
        draw.text((x0 + 12, 62), label, font=small_font, fill=INK)
        source, needed, target, produced = RULES[rule_index]
        left = [0, 0, 0, 0]
        right = [0, 0, 0, 0]
        left[source], right[target] = needed, produced
        _tokens(draw, left, (x0 + 16, 94, x0 + 112, 182), 3, bool((variant >> stage) & 1))
        _tokens(draw, right, (x0 + 176, 94, x0 + 272, 182), 3, not bool((variant >> stage) & 1))
        draw.line((x0 + 122, 136, x0 + 166, 136), fill=ACCENT, width=5)
        draw.polygon(((x0 + 166, 136), (x0 + 153, 126), (x0 + 153, 146)), fill=ACCENT)

    draw.text((24, 212), "START POUCH", font=label_font, fill=INK)
    draw.rounded_rectangle((24, 240, 936, 350), radius=12, fill=PANEL, outline=ACCENT, width=4)
    _tokens(draw, start, (190, 250, 770, 340), 12, bool(variant & 8))

    draw.text((24, 368), "FINAL POUCH?", font=label_font, fill=INK)
    for index, counts in enumerate(trays):
        x0 = TRAY_X[index]
        draw.rounded_rectangle((x0, 398, x0 + 210, 696), radius=12, fill=PANEL, outline=INK, width=3)
        label = f"tray {chr(65 + index)}"
        label_box = draw.textbbox((0, 0), label, font=small_font)
        draw.text((x0 + 105 - (label_box[2] - label_box[0]) // 2, 410), label, font=small_font, fill=INK)
        draw.line((x0 + 12, 438, x0 + 198, 438), fill=MUTED, width=2)
        _tokens(draw, counts, (x0 + 16, 450, x0 + 194, 682), 7, bool((variant + index) & 1))
    return image
