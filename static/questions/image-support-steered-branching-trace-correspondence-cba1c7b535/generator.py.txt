"""Compact branching-trace correspondence renderer and analytic arm."""

from __future__ import annotations

import copy
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

RENDERER_VERSION = "branching-trace-correspondence-0.1.0"
WIDTH, HEIGHT = 800, 600
SCALE = 4
PAPER = (248, 246, 240)
BOARD = (235, 241, 239)
CARD = (250, 250, 246)
BORDER = (188, 199, 196)
INK = (42, 54, 67)
TRACE = (24, 82, 114)
TRACE_WIDTH = 6
ARM_LENGTH = 25
BULGE_RADIUS = 14
BULGE_OFFSET = 7
PANEL_OFFSETS = (-62, 0, 62)
QUARANTINE_MARGIN = 0.5

# Direction order is up, right, down, left.  The perpendicular is the
# image-space left normal, so a sign records which side the interior bulge uses.
_DIRECTIONS = ((0, -1), (1, 0), (0, 1), (-1, 0))
_PERPENDICULARS = ((1, 0), (0, 1), (-1, 0), (0, -1))


def _font(size: int):
    for name in (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
        "/usr/share/fonts/truetype/liberation2/LiberationSans-Regular.ttf",
    ):
        if Path(name).is_file():
            return ImageFont.truetype(name, size=size * SCALE)
    return ImageFont.load_default()


def _clockwise_code(code: tuple[int, ...]) -> tuple[int, ...]:
    """Rotate a four-entry up/right/down/left code clockwise."""

    return (int(code[3]), int(code[0]), int(code[1]), int(code[2]))


def _record_by_slot(scene: dict) -> dict[int, dict]:
    return {int(record["slot"]): record for record in scene["candidate_traces"]}


def sample_scene(seed: int) -> dict:
    """Sample balanced codes, equal-ink candidates, and a randomized inset."""

    rng = random.Random(int(seed))
    reference = tuple(rng.choice((-1, 1)) for _ in range(4))
    # The candidate is drawn before the visible clockwise turn.  This is the
    # inverse of _clockwise_code(reference), so it is the exact match after the
    # declared transformation.
    matched = (reference[1], reference[2], reference[3], reference[0])
    wrong = list(matched)
    flip_count = 1 if int(seed) % 4 == 0 else 2 + rng.randrange(3)
    for index in rng.sample(range(4), flip_count):
        wrong[index] *= -1
    match_slot = 1 if rng.randrange(2) == 0 else 2
    other_slot = 3 - match_slot
    records = [
        {"slot": match_slot, "bulges": list(matched)},
        {"slot": other_slot, "bulges": wrong},
    ]
    rng.shuffle(records)
    return {
        "reference_bulges": list(reference),
        "candidate_traces": records,
        # All anchors stay inside a small central region.  Their variation
        # prevents absolute position from being a reliable answer cue.
        "anchor": [400 + rng.choice((-24, -12, 0, 12, 24)), 263 + rng.choice((-1, 0, 1))],
    }


def analytic_gold(scene: dict) -> str:
    """Return the slot whose turned code has minimum Hamming error."""

    reference = tuple(int(value) for value in scene["reference_bulges"])
    errors: dict[int, int] = {}
    for record in scene["candidate_traces"]:
        code = tuple(int(value) for value in record["bulges"])
        transformed = _clockwise_code(code)
        errors[int(record["slot"])] = sum(a != b for a, b in zip(reference, transformed))
    if set(errors) != {1, 2} or errors[1] == errors[2]:
        raise ValueError("candidate branch codes must have one unique nearest match")
    return "A" if errors[1] < errors[2] else "B"


def margin(scene: dict) -> float:
    """Return the runner-up gap between the two transformed Hamming errors."""

    reference = tuple(int(value) for value in scene["reference_bulges"])
    errors = []
    for record in scene["candidate_traces"]:
        code = _clockwise_code(tuple(int(value) for value in record["bulges"]))
        errors.append(sum(a != b for a, b in zip(reference, code)))
    if len(errors) != 2:
        raise ValueError("expected two candidate traces")
    return float(abs(errors[0] - errors[1]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def _scaled_point(point: tuple[float, float]) -> tuple[int, int]:
    return (int(round(point[0] * SCALE)), int(round(point[1] * SCALE)))


def _arm_points(cx: float, cy: float, arm: int, side: int) -> list[tuple[int, int]]:
    direction = _DIRECTIONS[arm]
    perpendicular = _PERPENDICULARS[arm]
    points = []
    for radius, offset in ((0, 0), (9, 0), (BULGE_RADIUS, side * BULGE_OFFSET), (20, side * BULGE_OFFSET), (ARM_LENGTH, 0)):
        points.append(
            _scaled_point(
                (
                    cx + direction[0] * radius + perpendicular[0] * offset,
                    cy + direction[1] * radius + perpendicular[1] * offset,
                )
            )
        )
    return points


def _draw_trace(draw: ImageDraw.ImageDraw, cx: float, cy: float, code: tuple[int, ...]) -> None:
    width = TRACE_WIDTH * SCALE
    for arm, side in enumerate(code):
        draw.line(_arm_points(cx, cy, arm, int(side)), fill=TRACE, width=width, joint="curve")
    radius = (TRACE_WIDTH * SCALE) // 2
    draw.ellipse(
        (int(cx * SCALE - radius), int(cy * SCALE - radius), int(cx * SCALE + radius), int(cy * SCALE + radius)),
        fill=TRACE,
    )


def _draw_label(draw: ImageDraw.ImageDraw, center_x: float, y: float, label: str, font) -> None:
    bbox = draw.textbbox((0, 0), label, font=font)
    width = bbox[2] - bbox[0]
    draw.text((int(center_x * SCALE - width / 2), int(y * SCALE)), label, font=font, fill=INK)


def render(scene: dict) -> Image.Image:
    """Render one antialiased page with one connected compact trace."""

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), PAPER)
    draw = ImageDraw.Draw(image)
    title_font = _font(28)
    subtitle_font = _font(16)
    label_font = _font(19)
    small_font = _font(14)
    anchor_x, anchor_y = (float(value) for value in scene["anchor"])
    centers = [(anchor_x + offset, anchor_y) for offset in PANEL_OFFSETS]

    draw.text((42 * SCALE, 36 * SCALE), "BRANCHING TRACE CORRESPONDENCE", font=title_font, fill=INK)
    draw.text(
        (44 * SCALE, 82 * SCALE),
        "Turn each candidate clockwise; compare the four interior bulges.",
        font=subtitle_font,
        fill=INK,
    )

    board_box = (anchor_x - 116, anchor_y - 78, anchor_x + 116, anchor_y + 78)
    draw.rounded_rectangle(tuple(int(v * SCALE) for v in board_box), radius=15 * SCALE, fill=BOARD, outline=BORDER, width=2 * SCALE)
    for center_x in (anchor_x - 62, anchor_x, anchor_x + 62):
        card_box = (center_x - 42, anchor_y - 57, center_x + 42, anchor_y + 57)
        draw.rounded_rectangle(tuple(int(v * SCALE) for v in card_box), radius=9 * SCALE, fill=CARD, outline=BORDER, width=1 * SCALE)
    _draw_label(draw, centers[0][0], anchor_y - 73, "R", label_font)
    _draw_label(draw, centers[1][0], anchor_y - 73, "A", label_font)
    _draw_label(draw, centers[2][0], anchor_y - 73, "B", label_font)

    # The bridge makes the decisive dark evidence one connected trace while
    # leaving the three local branch maps independently legible.
    draw.line(
        [_scaled_point((anchor_x - 96, anchor_y)), _scaled_point((anchor_x + 96, anchor_y))],
        fill=TRACE,
        width=TRACE_WIDTH * SCALE,
    )
    reference = tuple(int(value) for value in scene["reference_bulges"])
    _draw_trace(draw, centers[0][0], centers[0][1], reference)
    records = _record_by_slot(scene)
    _draw_trace(draw, centers[1][0], centers[1][1], tuple(int(value) for value in records[1]["bulges"]))
    _draw_trace(draw, centers[2][0], centers[2][1], tuple(int(value) for value in records[2]["bulges"]))
    draw.text(
        (int((anchor_x - 99) * SCALE), int((anchor_y + 88) * SCALE)),
        "same endpoints • same ink • follow the interior bulges",
        font=small_font,
        fill=INK,
    )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Exercise an order-only latent alias: visible slots remain fixed."""

    twin = copy.deepcopy(scene)
    twin["candidate_traces"] = list(reversed(twin["candidate_traces"]))
    return [("candidate_record_order", twin)]
