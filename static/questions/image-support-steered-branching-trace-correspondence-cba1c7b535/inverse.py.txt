"""Independent RGB-only inverse for the branching trace question.

The inverse locates the one long dark connected trace in the PNG, finds its
three tall junction columns, and samples the side of each interior excursion.
It never receives a scene record or imports the renderer.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

TRACE = np.asarray((24, 82, 114), dtype=np.float32)
TRACE_RADIUS = 44.0
ARM_RADIUS = 16
BULGE_OFFSET = 7
PATCH_RADIUS = 4


def _mask(image: Image.Image) -> np.ndarray:
    pixels = np.asarray(image.convert("RGB"), dtype=np.float32)
    distance = np.sqrt(np.sum((pixels - TRACE) ** 2, axis=2))
    return distance < TRACE_RADIUS


def _patch_count(mask: np.ndarray, x: int, y: int) -> int:
    y0 = max(0, y - PATCH_RADIUS)
    y1 = min(mask.shape[0], y + PATCH_RADIUS + 1)
    x0 = max(0, x - PATCH_RADIUS)
    x1 = min(mask.shape[1], x + PATCH_RADIUS + 1)
    return int(mask[y0:y1, x0:x1].sum())


def _horizontal_center(mask: np.ndarray) -> int:
    row_counts = mask.sum(axis=1)
    if not np.any(row_counts):
        raise ValueError("trace color is absent")
    # The bridge and all three horizontal arms share the center row and make
    # it the longest trace-colored row, independently of the inset anchor.
    return int(np.argmax(row_counts))


def _junction_centers(mask: np.ndarray, center_y: int) -> list[int]:
    height, width = mask.shape
    y0 = max(0, center_y - 45)
    y1 = min(height, center_y + 46)
    counts = mask[y0:y1].sum(axis=0).astype(np.float64)
    smooth = np.convolve(counts, np.ones(9, dtype=np.float64), mode="same")
    peaks: list[int] = []
    for index in sorted(range(width), key=lambda value: (-smooth[value], value)):
        if smooth[index] < 18.0:
            break
        if all(abs(index - prior) > 30 for prior in peaks):
            peaks.append(index)
        if len(peaks) == 3:
            break
    if len(peaks) != 3:
        ys, xs = np.where(mask)
        if len(xs) < 3:
            raise ValueError("cannot locate three branch junctions")
        left, right = int(xs.min()), int(xs.max())
        peaks = [left + 34, (left + right) // 2, right - 34]
    return sorted(int(value) for value in peaks)


def _bulge_code(mask: np.ndarray, center_x: int, center_y: int) -> tuple[int, ...]:
    samples = (
        ((center_x + BULGE_OFFSET, center_y - ARM_RADIUS), (center_x - BULGE_OFFSET, center_y - ARM_RADIUS)),
        ((center_x + ARM_RADIUS, center_y + BULGE_OFFSET), (center_x + ARM_RADIUS, center_y - BULGE_OFFSET)),
        ((center_x - BULGE_OFFSET, center_y + ARM_RADIUS), (center_x + BULGE_OFFSET, center_y + ARM_RADIUS)),
        ((center_x - ARM_RADIUS, center_y - BULGE_OFFSET), (center_x - ARM_RADIUS, center_y + BULGE_OFFSET)),
    )
    result = []
    for positive, negative in samples:
        positive_count = _patch_count(mask, *positive)
        negative_count = _patch_count(mask, *negative)
        if max(positive_count, negative_count) < 8 or abs(positive_count - negative_count) < 5:
            raise ValueError("interior bulge is not clearly rasterized")
        result.append(1 if positive_count > negative_count else -1)
    return tuple(result)


def _clockwise_code(code: tuple[int, ...]) -> tuple[int, ...]:
    return (code[3], code[0], code[1], code[2])


def decision_from_image(image: Image.Image) -> str:
    """Return A or B from the final raster only."""

    mask = _mask(image)
    if mask.ndim != 2 or mask.shape[1] < 700 or mask.shape[0] < 500:
        raise ValueError("unexpected raster dimensions")
    center_y = _horizontal_center(mask)
    centers = _junction_centers(mask, center_y)
    codes = [_bulge_code(mask, center_x, center_y) for center_x in centers]
    reference = codes[0]
    errors = []
    for code in codes[1:]:
        transformed = _clockwise_code(code)
        errors.append(sum(a != b for a, b in zip(reference, transformed)))
    if len(errors) != 2 or errors[0] == errors[1]:
        raise ValueError("pixel branch-code comparison is ambiguous")
    if not all(math.isfinite(float(value)) for value in errors):
        raise ValueError("non-finite branch-code error")
    return "A" if errors[0] < errors[1] else "B"
