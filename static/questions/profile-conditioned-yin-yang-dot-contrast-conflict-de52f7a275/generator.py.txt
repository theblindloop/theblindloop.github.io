"""Deterministic renderer and analytic arm for the yin-yang relation world."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageChops, ImageDraw


SCALE = 4
BACKGROUND = (248, 247, 243)
OUTER = (75, 82, 91)
TEAL = (19, 146, 154)
ACCENTS = {
    "coral": (220, 82, 76),
    "gold": (232, 177, 48),
    "violet": (127, 88, 190),
    "blue": (60, 116, 205),
}
PALETTES = {
    "ink": ((24, 29, 38), (239, 234, 218)),
    "soft": ((58, 65, 76), (215, 207, 184)),
    "boundary": ((113, 116, 121), (143, 146, 151)),
}


def sample_scene(seed: int) -> dict:
    """Return one member of an adjacent, nuisance-matched control pair."""

    pair = int(seed) // 2
    rng = random.Random(0x59A6_2026 ^ pair)
    radius = (148, 158, 168)[pair % 3]
    center = [256 + rng.randint(-22, 22), 254 + rng.randint(-20, 20)]
    positions = [(42, 42), (470, 44), (44, 468), (469, 469)]
    names = ["coral", "gold", "violet", "blue"]
    rng.shuffle(names)
    distractors = [
        {"center": [x, y], "color": color, "size": 9 + ((pair + i) % 3)}
        for i, ((x, y), color) in enumerate(zip(positions, names))
    ]
    return {
        "dot_relation": "opposite" if int(seed) % 2 == 0 else "matching",
        "center": center,
        "radius": radius,
        "phase_quarters": pair % 4,
        "palette_inverted": bool((pair // 4) % 2),
        "contrast_style": "soft" if (pair // 2) % 3 == 0 else "ink",
        "dot_radius": 21 + (pair % 3),
        "canvas_size": [512, 512],
        "distractors": distractors,
    }


def _symbol_patch(scene: dict) -> Image.Image:
    radius = int(scene["radius"]) * SCALE
    dot_radius = int(scene["dot_radius"]) * SCALE
    phase = int(scene["phase_quarters"]) % 4
    inverted = bool(scene["palette_inverted"])
    dark, light = PALETTES[str(scene["contrast_style"])]
    base, curve = (light, dark) if inverted else (dark, light)

    border = 6 * SCALE
    pad = 9 * SCALE
    side = 2 * (radius + pad) + 1
    center = radius + pad
    inner = radius - border
    bounds = (center - radius, center - radius, center + radius, center + radius)
    inner_bounds = (center - inner, center - inner, center + inner, center + inner)

    patch = Image.new("RGBA", (side, side), (0, 0, 0, 0))
    draw = ImageDraw.Draw(patch)
    draw.ellipse(bounds, fill=(*OUTER, 255))
    draw.ellipse(inner_bounds, fill=(*base, 255))

    disk_mask = Image.new("L", (side, side), 0)
    ImageDraw.Draw(disk_mask).ellipse(inner_bounds, fill=255)
    curve_mask = Image.new("L", (side, side), 0)
    mask_draw = ImageDraw.Draw(curve_mask)
    mask_draw.rectangle((center, center - inner, center + inner, center + inner), fill=255)
    half = inner // 2
    mask_draw.ellipse((center - half, center - inner, center + half, center), fill=255)
    mask_draw.ellipse((center - half, center, center + half, center + inner), fill=0)
    curve_mask = ImageChops.multiply(curve_mask, disk_mask)
    curve_layer = Image.new("RGBA", (side, side), (*curve, 255))
    patch.paste(curve_layer, (0, 0), curve_mask)

    top_surround, bottom_surround = curve, base
    if str(scene["dot_relation"]) == "opposite":
        top_fill, bottom_fill = bottom_surround, top_surround
    elif str(scene["dot_relation"]) == "matching":
        top_fill, bottom_fill = top_surround, bottom_surround
    else:
        raise ValueError("unknown dot_relation")
    ring = 5 * SCALE
    for cy, fill in ((center - half, top_fill), (center + half, bottom_fill)):
        draw.ellipse(
            (center - dot_radius - ring, cy - dot_radius - ring,
             center + dot_radius + ring, cy + dot_radius + ring),
            fill=(*TEAL, 255),
        )
        draw.ellipse(
            (center - dot_radius, cy - dot_radius,
             center + dot_radius, cy + dot_radius),
            fill=(*fill, 255),
        )

    if phase:
        patch = patch.rotate(-90 * phase, resample=Image.Resampling.NEAREST, expand=False)
    return patch


def render(scene: dict) -> Image.Image:
    width, height = (int(v) for v in scene["canvas_size"])
    cx, cy = (int(v) for v in scene["center"])
    _ = str(scene["dot_relation"])
    _ = int(scene["radius"])
    _ = int(scene["phase_quarters"])
    _ = bool(scene["palette_inverted"])
    _ = str(scene["contrast_style"])
    _ = int(scene["dot_radius"])
    canvas = Image.new("RGBA", (width * SCALE, height * SCALE), (*BACKGROUND, 255))
    patch = _symbol_patch(scene)
    left = cx * SCALE - patch.width // 2
    top = cy * SCALE - patch.height // 2
    canvas.alpha_composite(patch, (left, top))

    draw = ImageDraw.Draw(canvas)
    for item in scene["distractors"]:
        x, y = (int(v) * SCALE for v in item["center"])
        size = int(item["size"]) * SCALE
        color = ACCENTS[str(item["color"])]
        draw.polygon([(x, y - size), (x + size, y), (x, y + size), (x - size, y)], fill=(*color, 255))

    return canvas.convert("RGB").resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    relation = str(scene["dot_relation"])
    if relation not in {"opposite", "matching"}:
        raise ValueError("unknown dot_relation")
    return "yes" if relation == "opposite" else "no"


def margin(scene: dict) -> float:
    dark, light = PALETTES[str(scene["contrast_style"])]
    dark_luma = sum((299, 587, 114)[i] * dark[i] for i in range(3)) / 1000.0
    light_luma = sum((299, 587, 114)[i] * light[i] for i in range(3)) / 1000.0
    return round(abs(light_luma - dark_luma) / 2.0, 6)


def is_quarantined(scene: dict) -> bool:
    width, height = (int(v) for v in scene["canvas_size"])
    cx, cy = (int(v) for v in scene["center"])
    radius = int(scene["radius"])
    edge = min(cx - radius, cy - radius, width - cx - radius, height - cy - radius)
    return margin(scene) < 40.0 or radius < 120 or edge < 12


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    distractors = list(scene["distractors"])
    if len(distractors) < 2:
        return []
    twin = copy.deepcopy(scene)
    twin["distractors"] = list(reversed(distractors))
    return [("reverse_disjoint_distractor_paint_order", twin)]
