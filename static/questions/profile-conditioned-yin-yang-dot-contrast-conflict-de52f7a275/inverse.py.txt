"""Independent image-only inverse for dot-to-local-lobe color relations."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[list[tuple[int, int]]] = []
    for y, x in np.argwhere(mask):
        y = int(y)
        x = int(x)
        if seen[y, x]:
            continue
        seen[y, x] = True
        stack = [(y, x)]
        group: list[tuple[int, int]] = []
        while stack:
            yy, xx = stack.pop()
            group.append((yy, xx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    ny, nx = yy + dy, xx + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        groups.append(group)
    return groups


def _luminance(values: np.ndarray) -> np.ndarray:
    return values[..., 0] * 0.299 + values[..., 1] * 0.587 + values[..., 2] * 0.114


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.float32)
    red, green, blue = pixels[..., 0], pixels[..., 1], pixels[..., 2]
    teal = (red < 85) & (green > 100) & (blue > 105) & (np.abs(green - blue) < 55)

    rings = []
    for group in _components(teal):
        if len(group) < 120:
            continue
        ys = np.array([point[0] for point in group])
        xs = np.array([point[1] for point in group])
        box_w = int(xs.max() - xs.min() + 1)
        box_h = int(ys.max() - ys.min() + 1)
        if 38 <= box_w <= 72 and 38 <= box_h <= 72 and abs(box_w - box_h) <= 8:
            rings.append((float(xs.mean()), float(ys.mean()), (box_w + box_h) / 4.0))
    if len(rings) != 2:
        raise ValueError(f"expected two teal dot outlines, found {len(rings)}")

    relations = []
    height, width = red.shape
    yy, xx = np.ogrid[:height, :width]
    for cx, cy, outer_radius in rings:
        distance = np.sqrt((xx - cx) ** 2 + (yy - cy) ** 2)
        inside = distance <= outer_radius * 0.48
        around = (distance >= outer_radius * 1.28) & (distance <= outer_radius * 1.72)
        if int(inside.sum()) < 120 or int(around.sum()) < 300:
            raise ValueError("insufficient dot or surrounding-lobe evidence")
        fill_luma = float(np.median(_luminance(pixels[inside])))
        host_luma = float(np.median(_luminance(pixels[around])))
        contrast = abs(fill_luma - host_luma)
        if contrast > 70.0:
            relations.append("opposite")
        elif contrast < 24.0:
            relations.append("matching")
        else:
            raise ValueError("dot-to-lobe contrast is within the quarantine band")

    if relations == ["opposite", "opposite"]:
        return "yes"
    if relations == ["matching", "matching"]:
        return "no"
    raise ValueError("the two dot-to-lobe relations disagree")
