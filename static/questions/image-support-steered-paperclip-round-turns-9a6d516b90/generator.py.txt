"""Deterministic renderer for prior-conflicting paperclip turn counts.

Seeds form matched triples.  Within a triple every visible nuisance is shared;
only the number of large alternating semicircular end turns changes.  Three is
the familiar paperclip control, while two and four are matched counterfactuals.
"""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

CANVAS_SIZE = 512

BACKGROUNDS = (
    (249, 247, 241),
    (244, 249, 248),
    (249, 245, 249),
)
WIRE_COLORS = (
    (25, 94, 142),
    (146, 53, 73),
    (25, 119, 104),
)
PAPER_COLORS = (
    (224, 221, 212),
    (213, 224, 222),
    (225, 216, 225),
)
PROOF_COLORS = (
    (137, 137, 132),
    (127, 139, 137),
    (140, 129, 140),
)


def sample_scene(seed: int) -> dict:
    """Return one deterministic member of a style-matched 2/3/4 triple."""

    seed = int(seed)
    group = seed // 3
    rng = random.Random((group ^ 0x41C71F2D) * 1_000_003)
    return {
        "background_code": rng.randrange(len(BACKGROUNDS)),
        "canvas_size": CANVAS_SIZE,
        "decor_seed": rng.randrange(1, 2**31),
        "lead_extension": rng.randrange(20, 47),
        "offset_x": rng.randrange(-4, 5),
        "offset_y": rng.randrange(-6, 7),
        "paperclip_color_code": rng.randrange(len(WIRE_COLORS)),
        "run_span": rng.randrange(360, 383),
        "stroke_width": rng.choice((5, 6)),
        "turn_count": 2 + seed % 3,
        "turn_radius": rng.choice((14, 15, 16)),
    }


def _validate(scene: dict) -> None:
    if int(scene["canvas_size"]) != CANVAS_SIZE:
        raise ValueError("the public world uses a fixed 512px canvas")
    if int(scene["turn_count"]) not in (2, 3, 4):
        raise ValueError("turn_count must be 2, 3, or 4")
    if int(scene["turn_radius"]) not in (14, 15, 16):
        raise ValueError("turn_radius must be 14, 15, or 16 pixels")
    if int(scene["stroke_width"]) not in (5, 6):
        raise ValueError("stroke_width must be 5 or 6 pixels")
    if not 350 <= int(scene["run_span"]) <= 390:
        raise ValueError("run_span is outside the supported range")
    if not 15 <= int(scene["lead_extension"]) <= 52:
        raise ValueError("lead_extension is outside the supported range")
    if not -8 <= int(scene["offset_x"]) <= 8 or not -10 <= int(scene["offset_y"]) <= 10:
        raise ValueError("translation is outside the supported visible range")


def _geometry(scene: dict) -> tuple[int, int, list[int]]:
    """Return left/right tangency x positions and horizontal-run centers."""

    turns = int(scene["turn_count"])
    radius = int(scene["turn_radius"])
    span = int(scene["run_span"])
    center_x = CANVAS_SIZE // 2 + int(scene["offset_x"])
    x_left = center_x - span // 2
    x_right = x_left + span
    center_y = CANVAS_SIZE // 2 + int(scene["offset_y"])
    top = center_y - turns * radius
    ys = [top + index * 2 * radius for index in range(turns + 1)]
    return x_left, x_right, ys


def _wire_bounds(scene: dict) -> tuple[int, int, int, int]:
    x_left, x_right, ys = _geometry(scene)
    radius = int(scene["turn_radius"])
    half = (int(scene["stroke_width"]) + 1) // 2
    return (x_left - radius - half, ys[0] - half, x_right + radius + half, ys[-1] + half)


def _joint(draw: ImageDraw.ImageDraw, x: int, y: int, width: int, color: tuple[int, int, int]) -> None:
    radius = max(1, width // 2)
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)


def render(scene: dict) -> Image.Image:
    _validate(scene)
    background = BACKGROUNDS[int(scene["background_code"]) % len(BACKGROUNDS)]
    wire = WIRE_COLORS[int(scene["paperclip_color_code"]) % len(WIRE_COLORS)]
    paper = PAPER_COLORS[int(scene["background_code"]) % len(PAPER_COLORS)]
    proof = PROOF_COLORS[int(scene["background_code"]) % len(PROOF_COLORS)]
    image = Image.new("RGB", (CANVAS_SIZE, CANVAS_SIZE), background)
    draw = ImageDraw.Draw(image)

    # A pale page outline and remote pseudo-writing establish paperclip context.
    # Their randomized total ink is shared within matched triples and never uses
    # the wire palette seen by the independent inverse arm.
    draw.rounded_rectangle((25, 72, 486, 440), radius=18, outline=paper, width=3)
    decor = random.Random(int(scene["decor_seed"]))
    for band_y in (range(108, 183, 15), range(338, 413, 15)):
        for y in band_y:
            if decor.random() < 0.78:
                x0 = decor.randrange(57, 142)
                x1 = decor.randrange(275, 456)
                draw.line((x0, y, x1, y), fill=proof, width=decor.choice((2, 3)))
    for _ in range(decor.randrange(4, 22)):
        x = decor.randrange(45, 468)
        y = decor.choice((decor.randrange(88, 184), decor.randrange(334, 425)))
        radius = decor.randrange(2, 7)
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=proof)

    x_left, x_right, ys = _geometry(scene)
    turns = int(scene["turn_count"])
    radius = int(scene["turn_radius"])
    width = int(scene["stroke_width"])
    lead = int(scene["lead_extension"])

    # Successive horizontal runs are portions of one open wire.  Alternating
    # semicircles at opposite ends supply the only counted reversal events.
    for index, y in enumerate(ys):
        start, end = x_left, x_right
        if index == 0:
            start += lead
        if index == turns:
            if turns % 2 == 0:
                end -= lead
            else:
                start += lead
        draw.line((start, y, end, y), fill=wire, width=width)
        _joint(draw, start, y, width, wire)
        _joint(draw, end, y, width, wire)

    for index in range(turns):
        mid_y = (ys[index] + ys[index + 1]) // 2
        if index % 2 == 0:
            box = (x_right - radius, mid_y - radius, x_right + radius, mid_y + radius)
            draw.arc(box, start=-90, end=90, fill=wire, width=width)
            _joint(draw, x_right, ys[index], width, wire)
            _joint(draw, x_right, ys[index + 1], width, wire)
        else:
            box = (x_left - radius, mid_y - radius, x_left + radius, mid_y + radius)
            draw.arc(box, start=90, end=270, fill=wire, width=width)
            _joint(draw, x_left, ys[index], width, wire)
            _joint(draw, x_left, ys[index + 1], width, wire)
    return image


def analytic_gold(scene: dict) -> str:
    count = int(scene["turn_count"])
    if count not in (2, 3, 4):
        raise ValueError("turn_count must be 2, 3, or 4")
    return str(count)


def margin(scene: dict) -> float:
    """Visible clear gap between adjacent parallel wire runs, in pixels."""

    return float(2 * int(scene["turn_radius"]) - int(scene["stroke_width"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 14.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["paperclip_color_code"] = int(twin["paperclip_color_code"]) + len(WIRE_COLORS)
    return [("one_full_wire_palette_cycle", twin)]
