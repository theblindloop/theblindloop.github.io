"""Independent pixel-only inverse for paperclip rounded-turn count.

The inverse receives only the final RGB raster.  It segments a public wire-color
vocabulary, requires one intact wide serpentine component, and counts the full
horizontal crossings through a scene-relative central band.  A connected open
serpentine with N turns has exactly N+1 such runs.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    """Raised when the raster no longer exposes an unambiguous answer."""


_WIRE_COLORS = np.asarray(
    ((25, 94, 142), (146, 53, 73), (25, 119, 104)), dtype=np.int32
)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        queue: deque[tuple[int, int]] = deque([(y, x)])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        output.append(np.asarray(points, dtype=np.int32))
    return output


def _clusters(indices: np.ndarray) -> list[np.ndarray]:
    if len(indices) == 0:
        return []
    cuts = np.where(np.diff(indices) > 1)[0] + 1
    return [part for part in np.split(indices, cuts) if len(part)]


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int32)
    if array.shape != (512, 512, 3):
        raise OracleAbstention("expected a 512 by 512 RGB raster")

    delta = array[:, :, None, :] - _WIRE_COLORS[None, None, :, :]
    distances = np.sum(delta * delta, axis=3)
    mask = np.min(distances, axis=2) <= 12**2
    pieces = _components(mask)
    substantial = [piece for piece in pieces if len(piece) >= 100]
    speckle = sum(len(piece) for piece in pieces if len(piece) < 100)
    if speckle > 24 or len(substantial) != 1:
        raise OracleAbstention("wire-colored ink is fragmented or contaminated")

    component = substantial[0]
    ys, xs = component[:, 0], component[:, 1]
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    span_x = x1 - x0 + 1
    span_y = y1 - y0 + 1
    if not 380 <= span_x <= 430:
        raise OracleAbstention("wire does not have the declared distributed horizontal span")
    if not 55 <= span_y <= 140 or span_x / span_y < 2.7:
        raise OracleAbstention("wire support is not a thin wide paperclip route")
    if not 4_500 <= len(component) <= 13_500:
        raise OracleAbstention("wire ink area is outside the supported range")
    if x0 < 30 or x1 > 482 or y0 < 175 or y1 > 337:
        raise OracleAbstention("wire is clipped or displaced")

    # Use a scene-relative interior band, far from either end turn.  Every long
    # run fully crosses it, while no end arc can create an extra crossing.
    core0 = x0 + int(round(0.43 * span_x))
    core1 = x0 + int(round(0.57 * span_x))
    core_width = core1 - core0 + 1
    coverage = mask[:, core0 : core1 + 1].sum(axis=1)
    full_rows = np.flatnonzero(coverage >= 0.96 * core_width)
    bands = _clusters(full_rows)
    if any(not 5 <= len(band) <= 8 for band in bands):
        raise OracleAbstention("a horizontal run has unsupported thickness")
    run_count = len(bands)
    if run_count not in (3, 4, 5):
        raise OracleAbstention("could not recover three, four, or five full runs")

    centers = np.asarray([float(band.mean()) for band in bands])
    gaps = np.diff(centers)
    if len(gaps) and (float(gaps.min()) < 26.0 or float(gaps.max()) > 34.0):
        raise OracleAbstention("run spacing is ambiguous or a run is missing")

    # Both outer quarters must contain colored ink from large end arcs.  Together
    # with single-component connectivity this rejects disconnected parallel bars
    # and makes remote wire erasures cause abstention.
    left_arc = mask[y0 : y1 + 1, x0 : x0 + max(10, span_x // 10)]
    right_arc = mask[y0 : y1 + 1, x1 - max(10, span_x // 10) + 1 : x1 + 1]
    if int(left_arc.sum()) < 90 or int(right_arc.sum()) < 90:
        raise OracleAbstention("large alternating end turns are not visibly present")

    return str(run_count - 1)
