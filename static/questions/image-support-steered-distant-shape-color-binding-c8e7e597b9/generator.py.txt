"""Deterministic sampler and renderer for two distant binding maps."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw


WIDTH, HEIGHT = 960, 520
BACKGROUND = [248, 247, 242]
PANEL_FILL = [236, 239, 235]
PANEL_OUTLINE = [103, 111, 106]
COLORS = {
    "coral": [218, 62, 74],
    "teal": [17, 151, 143],
    "gold": [232, 163, 24],
}
SHAPES = ("circle", "square", "triangle")


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    panel_boxes = [[40, 38, 430, 482], [530, 38, 920, 482]]
    color_names = list(COLORS)
    left_colors = color_names[:]
    rng.shuffle(left_colors)
    if seed % 2 == 0:
        right_colors = left_colors[:]
    elif rng.randrange(2) == 0:
        right_colors = left_colors[1:] + left_colors[:1]
    else:
        right_colors = left_colors[-1:] + left_colors[:-1]

    objects: list[dict] = []
    for panel, (xlo, xhi), mapping in (
        (0, (105, 365), left_colors),
        (1, (595, 855), right_colors),
    ):
        slots = [0, 1, 2]
        rng.shuffle(slots)
        for shape_index, shape in enumerate(SHAPES):
            slot = slots[shape_index]
            cy = (137, 260, 383)[slot] + rng.randint(-13, 13)
            cx = rng.randint(xlo, xhi)
            size = rng.randint(34, 43)
            objects.append(
                {
                    "panel": panel,
                    "shape": shape,
                    "color": mapping[shape_index],
                    "rgb": COLORS[mapping[shape_index]],
                    "cx": cx,
                    "cy": cy,
                    "size": size,
                    "turn": rng.randrange(4),
                }
            )
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "panel_boxes": panel_boxes,
        "objects": objects,
        "background": BACKGROUND,
        "panel_fill": PANEL_FILL,
        "panel_outline": PANEL_OUTLINE,
    }


def _triangle_points(cx: int, cy: int, radius: int, turn: int) -> list[tuple[float, float]]:
    start = -math.pi / 2 + turn * math.pi / 2
    return [
        (cx + radius * math.cos(start + k * 2 * math.pi / 3),
         cy + radius * math.sin(start + k * 2 * math.pi / 3))
        for k in range(3)
    ]


def render(scene: dict):
    scale = 3
    image = Image.new(
        "RGB", (scene["width"] * scale, scene["height"] * scale), tuple(scene["background"])
    )
    draw = ImageDraw.Draw(image)
    for box in scene["panel_boxes"]:
        scaled = [value * scale for value in box]
        draw.rounded_rectangle(
            scaled,
            radius=22 * scale,
            fill=tuple(scene["panel_fill"]),
            outline=tuple(scene["panel_outline"]),
            width=3 * scale,
        )
    for obj in scene["objects"]:
        cx, cy, radius = obj["cx"] * scale, obj["cy"] * scale, obj["size"] * scale
        fill = tuple(obj["rgb"])
        if obj["shape"] == "circle":
            draw.ellipse([cx - radius, cy - radius, cx + radius, cy + radius], fill=fill)
        elif obj["shape"] == "square":
            draw.rectangle([cx - radius, cy - radius, cx + radius, cy + radius], fill=fill)
        elif obj["shape"] == "triangle":
            points = _triangle_points(cx, cy, radius, obj["turn"])
            draw.polygon(points, fill=fill)
        else:
            raise ValueError("unknown shape")
    return image.resize((scene["width"], scene["height"]), Image.Resampling.LANCZOS)


def _binding(scene: dict, panel: int) -> dict[str, str]:
    return {
        obj["shape"]: obj["color"]
        for obj in scene["objects"]
        if obj["panel"] == panel
    }


def analytic_gold(scene: dict) -> str:
    return "yes" if _binding(scene, 0) == _binding(scene, 1) else "no"


def margin(scene: dict) -> float:
    left, right = _binding(scene, 0), _binding(scene, 1)
    mismatch_count = sum(left[shape] != right[shape] for shape in SHAPES)
    return float(2 if mismatch_count == 0 else mismatch_count)


def is_quarantined(scene: dict) -> bool:
    # The sampler's large separation and size floors exclude ambiguous scenes.
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["objects"] = list(reversed(twin["objects"]))
    return [("reverse_disjoint_draw_order", twin)]
