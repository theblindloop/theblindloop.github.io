"""Independent inverse arm using only the final raster.

It finds the blank divide rather than using renderer coordinates, segments the
three fixed inks, classifies component geometry by bounding-box occupancy, and
compares the two reconstructed shape-to-color maps.
"""

from __future__ import annotations

import numpy as np


PROTOTYPES = np.asarray(
    [[218, 62, 74], [17, 151, 143], [232, 163, 24]], dtype=np.int16
)


def _shape(mask: np.ndarray) -> str:
    ys, xs = np.nonzero(mask)
    if len(xs) < 900:
        raise ValueError("colored component is missing or heavily damaged")
    width = int(xs.max() - xs.min() + 1)
    height = int(ys.max() - ys.min() + 1)
    if min(width, height) < 42 or max(width, height) > 98:
        raise ValueError("component scale is outside the declared visual regime")
    fill = len(xs) / float(width * height)
    if fill < 0.64:
        return "triangle"
    if fill < 0.89:
        return "circle"
    return "square"


def _map_for_side(labels: np.ndarray, colored: np.ndarray, side: np.ndarray) -> dict[str, int]:
    result: dict[str, int] = {}
    seen_colors: set[int] = set()
    for color_index in range(3):
        mask = (labels == color_index) & colored & side
        if int(mask.sum()) < 900:
            raise ValueError("each panel must contain one complete object of every color")
        shape = _shape(mask)
        if shape in result or color_index in seen_colors:
            raise ValueError("panel does not contain one object of every shape and color")
        result[shape] = color_index
        seen_colors.add(color_index)
    if set(result) != {"circle", "square", "triangle"}:
        raise ValueError("incomplete shape map")
    return result


def decision_from_image(image):
    # int32 avoids overflow when squared RGB differences exceed 32767.
    array = np.asarray(image.convert("RGB"), dtype=np.int32)
    distances = ((array[:, :, None, :] - PROTOTYPES[None, None, :, :]) ** 2).sum(axis=3)
    labels = distances.argmin(axis=2)
    colored = distances.min(axis=2) < 70 * 70
    if int(colored.sum()) < 7000:
        raise ValueError("too little colored evidence")

    occupied_columns = np.flatnonzero(colored.any(axis=0))
    if len(occupied_columns) < 2:
        raise ValueError("no colored components")
    gaps = np.diff(occupied_columns)
    split_at = int(np.argmax(gaps))
    if int(gaps[split_at]) < image.width * 0.10:
        raise ValueError("no wide blank divide between panels")
    split_x = (int(occupied_columns[split_at]) + int(occupied_columns[split_at + 1])) / 2.0
    xgrid = np.arange(image.width)[None, :]
    left = np.broadcast_to(xgrid < split_x, colored.shape)
    right = ~left
    left_map = _map_for_side(labels, colored, left)
    right_map = _map_for_side(labels, colored, right)
    return "yes" if left_map == right_map else "no"
