"""Deterministic sampler and renderer for constellation_total_tether."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

COLORS = (
    ("red", (210, 52, 62)),
    ("blue", (42, 103, 184)),
    ("green", (36, 145, 91)),
    ("orange", (226, 126, 33)),
)
BACKGROUND = (248, 247, 243)
DARK = (28, 34, 43)
DOT_COUNT = 28
QUARANTINE_GAP = 150.0


def _total_distance(points: list[list[float]], center: list[float]) -> float:
    return sum(math.hypot(float(p[0]) - center[0], float(p[1]) - center[1]) for p in points)


def _candidate_scores(scene: dict) -> list[tuple[float, str]]:
    return [
        (_total_distance(scene["points"], item["center"]), item["color"])
        for item in scene["candidates"]
    ]


def _candidate_layout(rng: random.Random, seed: int) -> tuple[list[list[float]], int, int]:
    """Four exchangeable rotated rectangle corners and two crossed slots."""
    base_x = rng.uniform(247.0, 265.0)
    base_y = rng.uniform(247.0, 265.0)
    half_x = rng.uniform(42.0, 49.0)
    half_y = rng.uniform(42.0, 49.0)
    angle = rng.uniform(0.0, 2.0 * math.pi)
    ca, sa = math.cos(angle), math.sin(angle)
    local = [(-half_x, -half_y), (half_x, -half_y), (half_x, half_y), (-half_x, half_y)]
    slots = [
        [round(base_x + ca * x - sa * y, 4), round(base_y + sa * x + ca * y, 4)]
        for x, y in local
    ]
    winner_slot = (int(seed) // 4) % 4
    # The outlier tail points toward an adjacent slot. This makes the arithmetic
    # mean favor that neighbor while the robust total-distance winner remains at
    # the core slot, so the world is not a centroid re-labeling.
    tail_slot = (winner_slot + 1) % 4
    if int(seed) % 7 == 0:
        # Put a different runner close to the winner, on the side away from the
        # tail. These are visible reduced-gap cases retained but quarantined.
        runner_slot = (winner_slot - 1) % 4
        wx, wy = slots[winner_slot]
        tx, ty = slots[tail_slot]
        length = math.hypot(tx - wx, ty - wy)
        slots[runner_slot] = [
            round(wx - 38.0 * (tx - wx) / length, 4),
            round(wy - 38.0 * (ty - wy) / length, 4),
        ]
    return slots, winner_slot, tail_slot


def _valid_point(
    point: tuple[float, float], existing: list[list[float]], candidate_centers: list[list[float]]
) -> bool:
    x, y = point
    return (
        28.0 <= x <= 484.0
        and 28.0 <= y <= 484.0
        and all(math.hypot(x - q[0], y - q[1]) >= 19.0 for q in existing)
        and all(math.hypot(x - c[0], y - c[1]) >= 28.0 for c in candidate_centers)
    )


def _make_points(
    rng: random.Random,
    target: list[float],
    tail_target: list[float],
    candidate_centers: list[list[float]],
    near_case: bool,
) -> list[list[float]]:
    """Build a skewed 20+8 cloud whose discrete total-distance winner is target."""
    dx, dy = tail_target[0] - target[0], tail_target[1] - target[1]
    norm = math.hypot(dx, dy)
    ux, uy = dx / norm, dy / norm
    vx, vy = -uy, ux
    target_index = candidate_centers.index(target)

    for _scene_try in range(1600):
        points: list[list[float]] = []
        # A broad core: every member matters, and there is no drawn contour or
        # local mark identifying its center.
        for _ in range(20):
            for _placement_try in range(700):
                angle = rng.uniform(0.0, 2.0 * math.pi)
                radius = math.sqrt(rng.uniform(35.0**2, 88.0**2))
                point = (target[0] + radius * math.cos(angle), target[1] + radius * math.sin(angle))
                if _valid_point(point, points, candidate_centers):
                    points.append([round(point[0], 4), round(point[1], 4)])
                    break
            else:
                break
        if len(points) != 20:
            continue
        # Eight distant points skew the coordinate mean toward the adjacent
        # candidate but stay below the 50% breakdown point of the distance sum.
        for _ in range(8):
            for _placement_try in range(900):
                along = rng.uniform(158.0, 218.0)
                across = rng.uniform(-94.0, 94.0)
                point = (
                    target[0] + along * ux + across * vx,
                    target[1] + along * uy + across * vy,
                )
                if _valid_point(point, points, candidate_centers):
                    points.append([round(point[0], 4), round(point[1], 4)])
                    break
            else:
                break
        if len(points) != DOT_COUNT:
            continue

        totals = [_total_distance(points, center) for center in candidate_centers]
        order = sorted(range(4), key=lambda index: totals[index])
        gap = totals[order[1]] - totals[order[0]]
        if order[0] != target_index or gap < (28.0 if near_case else 130.0):
            continue
        mean = [sum(p[axis] for p in points) / DOT_COUNT for axis in (0, 1)]
        mean_winner = min(
            range(4),
            key=lambda index: math.hypot(
                candidate_centers[index][0] - mean[0], candidate_centers[index][1] - mean[1]
            ),
        )
        if mean_winner == target_index:
            continue
        return points
    raise RuntimeError("could not sample a separated total-tether constellation")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    slots, winner_slot, tail_slot = _candidate_layout(rng, int(seed))
    points = _make_points(
        rng,
        slots[winner_slot],
        slots[tail_slot],
        slots,
        int(seed) % 7 == 0,
    )
    winner_color_index = int(seed) % len(COLORS)
    remaining_slots = [index for index in range(4) if index != winner_slot]
    rng.shuffle(remaining_slots)
    slot_for_color: dict[int, int] = {winner_color_index: winner_slot}
    for color_index, slot_index in zip(
        [index for index in range(4) if index != winner_color_index], remaining_slots
    ):
        slot_for_color[color_index] = slot_index
    candidates = [
        {"color": name, "rgb": list(rgb), "center": slots[slot_for_color[index]]}
        for index, (name, rgb) in enumerate(COLORS)
    ]
    return {
        "width": 512,
        "height": 512,
        "background": list(BACKGROUND),
        "dark_color": list(DARK),
        "point_radius": 7,
        "ring_radius": 13,
        "ring_width": 4,
        "points": points,
        "candidates": candidates,
    }


def render(scene: dict) -> Image.Image:
    scale = 4
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    radius = float(scene["point_radius"])
    for x, y in scene["points"]:
        draw.ellipse(
            ((x - radius) * scale, (y - radius) * scale, (x + radius) * scale, (y + radius) * scale),
            fill=tuple(scene["dark_color"]),
        )
    ring_radius = float(scene["ring_radius"])
    ring_width = int(scene["ring_width"])
    for item in scene["candidates"]:
        x, y = item["center"]
        draw.ellipse(
            ((x - ring_radius) * scale, (y - ring_radius) * scale,
             (x + ring_radius) * scale, (y + ring_radius) * scale),
            fill=tuple(scene["background"]),
            outline=tuple(item["rgb"]),
            width=ring_width * scale,
        )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    scores = sorted(_candidate_scores(scene))
    if len(scores) != 4 or scores[1][0] - scores[0][0] <= 0.0:
        raise ValueError("scene has no unique minimum-total-distance candidate")
    return scores[0][1]


def margin(scene: dict) -> float:
    scores = sorted(score for score, _ in _candidate_scores(scene))
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["points"] = list(reversed(twin["points"]))
    return [("reverse_dark_dot_draw_order", twin)]
