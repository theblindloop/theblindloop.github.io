"""Independent image-only parser and replay arm for the dynamic-link grammar."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

INK_LIMIT = 70
EDGE = (48, 105, 190)
QUERY = (218, 35, 178)
WHITE = (255, 255, 255)
LINK_COLORS = {(35, 157, 92), (24, 142, 118), (77, 151, 61)}
CUT_COLORS = {(218, 65, 58), (222, 79, 47), (200, 55, 91)}


def _mask_for(arr: np.ndarray, colors: set[tuple[int, int, int]]) -> np.ndarray:
    mask = np.zeros(arr.shape[:2], dtype=bool)
    for color in colors:
        mask |= np.all(arr == np.asarray(color, dtype=np.uint8), axis=2)
    return mask


def _components(mask: np.ndarray, minimum: int = 1) -> list[dict]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        points = []
        while queue:
            py, px = queue.popleft()
            points.append((py, px))
            for ny, nx in ((py - 1, px), (py + 1, px), (py, px - 1), (py, px + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            ys = [point[0] for point in points]
            xs = [point[1] for point in points]
            output.append({"area": len(points), "box": (min(xs), min(ys), max(xs), max(ys)), "cx": sum(xs) / len(xs), "cy": sum(ys) / len(ys)})
    return output


def _count_pips(black: np.ndarray, cx: float, cy: float) -> int:
    x0, x1 = int(round(cx)) - 16, int(round(cx)) + 17
    y0, y1 = int(round(cy)) - 16, int(round(cy)) + 17
    if x0 < 0 or y0 < 0 or x1 > black.shape[1] or y1 > black.shape[0]:
        raise ValueError("pip field touches raster edge")
    components = _components(black[y0:y1, x0:x1], minimum=25)
    pips = [item for item in components if 25 <= item["area"] <= 100]
    if len(pips) not in (1, 2, 3, 4):
        raise ValueError(f"invalid dot-number label with {len(pips)} dots")
    return len(pips)


def _parse_cards(black: np.ndarray) -> list[dict]:
    cards = []
    for item in _components(black, minimum=500):
        x0, y0, x1, y1 = item["box"]
        width, height = x1 - x0 + 1, y1 - y0 + 1
        if 120 <= width <= 135 and 210 <= height <= 230:
            cards.append(item)
    cards.sort(key=lambda item: item["cx"])
    if len(cards) != 6:
        raise ValueError(f"visual grammar damaged: found {len(cards)} update cards")
    return cards


def _has_edge(edge_mask: np.ndarray, one: dict, two: dict) -> bool:
    samples = []
    for t in np.linspace(0.18, 0.82, 100):
        x = int(round(one["cx"] * (1.0 - t) + two["cx"] * t))
        y = int(round(one["cy"] * (1.0 - t) + two["cy"] * t))
        y0, y1 = max(0, y - 2), min(edge_mask.shape[0], y + 3)
        x0, x1 = max(0, x - 2), min(edge_mask.shape[1], x + 3)
        samples.append(bool(edge_mask[y0:y1, x0:x1].any()))
    fraction = sum(samples) / len(samples)
    if 0.20 <= fraction <= 0.65:
        raise ValueError(f"ambiguous initial link evidence ({fraction:.2f})")
    return fraction > 0.65


def _connected(edges: set[tuple[int, int]], query: tuple[int, int]) -> bool:
    queue = deque([query[0]])
    seen = {query[0]}
    while queue:
        node = queue.popleft()
        if node == query[1]:
            return True
        for a, b in edges:
            other = b if node == a else a if node == b else None
            if other is not None and other not in seen:
                seen.add(other)
                queue.append(other)
    return False


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.ndim != 3 or arr.shape[0] < 500 or arr.shape[1] < 750:
        raise ValueError("image is too small")
    black = np.all(arr < INK_LIMIT, axis=2)
    white = _mask_for(arr, {WHITE})

    # The four large white station interiors are discovered by component area;
    # endpoint roundels on cards are smaller and therefore excluded.
    nodes = [item for item in _components(white, minimum=2000) if item["area"] > 2400]
    if len(nodes) != 4:
        raise ValueError(f"expected four station disks, found {len(nodes)}")
    numbered: dict[int, dict] = {}
    for node in nodes:
        number = _count_pips(black, node["cx"], node["cy"])
        if number in numbered:
            raise ValueError("duplicate station number")
        numbered[number] = node
    if set(numbered) != {1, 2, 3, 4}:
        raise ValueError("station numbering is incomplete")

    query_parts = _components(_mask_for(arr, {QUERY}), minimum=300)
    if len(query_parts) != 2:
        raise ValueError(f"expected two query rings, found {len(query_parts)}")
    query_nodes = []
    for ring in query_parts:
        distances = sorted((abs(ring["cx"] - item["cx"]) + abs(ring["cy"] - item["cy"]), number) for number, item in numbered.items())
        if distances[0][0] > 12 or distances[1][0] - distances[0][0] < 40:
            raise ValueError("query-ring binding is ambiguous")
        query_nodes.append(distances[0][1])
    if len(set(query_nodes)) != 2:
        raise ValueError("query rings bind the same station")

    edge_mask = _mask_for(arr, {EDGE})
    edges: set[tuple[int, int]] = set()
    for a in range(1, 5):
        for b in range(a + 1, 5):
            if _has_edge(edge_mask, numbered[a], numbered[b]):
                edges.add((a, b))
    if len(edges) != 3:
        raise ValueError(f"expected three initial links, found {len(edges)}")

    link_mask = _mask_for(arr, LINK_COLORS)
    cut_mask = _mask_for(arr, CUT_COLORS)
    for card in _parse_cards(black):
        x0, y0, x1, y1 = card["box"]
        crop_link = int(link_mask[y0:y1 + 1, x0:x1 + 1].sum())
        crop_cut = int(cut_mask[y0:y1 + 1, x0:x1 + 1].sum())
        if crop_link > 2500 and crop_cut < 50:
            op = "link"
        elif crop_cut > 2500 and crop_link < 50:
            op = "cut"
        else:
            raise ValueError("operation color is ambiguous")
        endpoint_y = y0 + 125
        endpoints = (_count_pips(black, x0 + 35, endpoint_y), _count_pips(black, x0 + 93, endpoint_y))
        if endpoints[0] == endpoints[1]:
            raise ValueError("self-link command is invalid")
        pair = tuple(sorted(endpoints))
        if op == "link":
            edges.add(pair)
        else:
            edges.discard(pair)
    query = tuple(sorted(query_nodes))
    return "yes" if _connected(edges, query) else "no"
