"""Deterministic sampler and renderer for dynamic undirected-link replay."""

from __future__ import annotations

import itertools
import random
from collections import deque

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 900, 640
BACKGROUND = (248, 247, 242)
INK = (28, 31, 36)
EDGE = (48, 105, 190)
QUERY = (218, 35, 178)
WHITE = (255, 255, 255)
OP_PALETTES = (
    {"link": (35, 157, 92), "cut": (218, 65, 58)},
    {"link": (24, 142, 118), "cut": (222, 79, 47)},
    {"link": (77, 151, 61), "cut": (200, 55, 91)},
)
ALL_EDGES = tuple(itertools.combinations((1, 2, 3, 4), 2))


def _edge(a: int, b: int) -> tuple[int, int]:
    return (a, b) if a < b else (b, a)


def _replay(initial_edges: list[list[int]] | list[tuple[int, int]], commands: list[dict]) -> tuple[set[tuple[int, int]], dict[str, int]]:
    edges = {_edge(int(pair[0]), int(pair[1])) for pair in initial_edges}
    stats = {"effective": 0, "noop": 0}
    for command in commands:
        pair = _edge(int(command["a"]), int(command["b"]))
        before = len(edges)
        if command["op"] == "link":
            edges.add(pair)
        else:
            edges.discard(pair)
        stats["effective" if len(edges) != before else "noop"] += 1
    return edges, stats


def _connected(edges: set[tuple[int, int]], query: tuple[int, int] | list[int]) -> bool:
    start, goal = int(query[0]), int(query[1])
    queue = deque([start])
    seen = {start}
    while queue:
        node = queue.popleft()
        if node == goal:
            return True
        for a, b in edges:
            other = b if node == a else a if node == b else None
            if other is not None and other not in seen:
                seen.add(other)
                queue.append(other)
    return False


def sample_scene(seed: int) -> dict:
    """Balance yes/no by seed parity while holding initial/final edge count fixed."""
    rng = random.Random(int(seed))
    target = int(seed) % 2 == 0
    palette_variant = rng.randrange(len(OP_PALETTES))
    shift = rng.choice((-10, -5, 0, 5, 10))
    decorations = [rng.randrange(3) for _ in range(6)]
    for _ in range(120000):
        initial = set(rng.sample(ALL_EDGES, 3))
        query = rng.choice(ALL_EDGES)
        ops = ["link"] * 3 + ["cut"] * 3
        rng.shuffle(ops)
        commands = [{"op": op, "a": pair[0], "b": pair[1]} for op, pair in zip(ops, (rng.choice(ALL_EDGES) for _ in range(6)), strict=True)]
        final, stats = _replay(sorted(initial), commands)
        if len(final) != 3 or stats["effective"] < 2 or stats["noop"] < 1:
            continue
        if _connected(initial, query) == target or _connected(final, query) != target:
            continue
        if _edge(*query) in final:
            continue
        return {
            "initial_edges": [list(pair) for pair in sorted(initial)],
            "query_nodes": list(query),
            "commands": commands,
            "palette_variant": palette_variant,
            "shift": shift,
            "decorations": decorations,
        }
    raise RuntimeError(f"could not sample answer stratum {target}")


def _pip_pattern(count: int) -> list[tuple[int, int]]:
    patterns = {
        1: [(0, 0)],
        2: [(-7, 0), (7, 0)],
        3: [(-7, 6), (7, 6), (0, -7)],
        4: [(-7, -7), (7, -7), (-7, 7), (7, 7)],
    }
    return patterns[count]


def _pips(draw: ImageDraw.ImageDraw, cx: int, cy: int, count: int, radius: int = 4) -> None:
    for dx, dy in _pip_pattern(count):
        draw.ellipse((cx + dx - radius, cy + dy - radius, cx + dx + radius, cy + dy + radius), fill=INK)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default(size=16)
    shift = int(scene["shift"])
    palette = OP_PALETTES[int(scene["palette_variant"])]
    centers = {
        1: (180 + shift, 105),
        2: (700 + shift, 105),
        3: (180 + shift, 300),
        4: (700 + shift, 300),
    }

    draw.text((28 + shift, 20), "START NETWORK", fill=INK, font=font)
    for a, b in scene["initial_edges"]:
        draw.line((centers[int(a)], centers[int(b)]), fill=EDGE, width=9)
    for node in scene["query_nodes"]:
        cx, cy = centers[int(node)]
        draw.ellipse((cx - 45, cy - 45, cx + 45, cy + 45), outline=QUERY, width=8)
    for number, (cx, cy) in centers.items():
        draw.ellipse((cx - 33, cy - 33, cx + 33, cy + 33), fill=WHITE, outline=INK, width=4)
        _pips(draw, cx, cy, number)

    draw.text((28 + shift, 354), "UPDATE CARDS: read left to right", fill=INK, font=font)
    for index, command in enumerate(scene["commands"]):
        left = 18 + shift + 146 * index
        top, right, bottom = 388, left + 128, 608
        draw.rectangle((left, top, right, bottom), fill=(238, 238, 234), outline=INK, width=4)
        draw.rectangle((left + 9, top + 9, right - 9, top + 53), fill=palette[command["op"]])
        cx = (left + right) // 2
        if command["op"] == "link":
            draw.rectangle((cx - 17, top + 28, cx + 17, top + 34), fill=WHITE)
            draw.rectangle((cx - 3, top + 17, cx + 3, top + 45), fill=WHITE)
        else:
            draw.rectangle((cx - 18, top + 28, cx + 18, top + 34), fill=WHITE)
        endpoint_y = top + 125
        endpoint_xs = (left + 35, left + 93)
        draw.line((endpoint_xs[0] + 24, endpoint_y, endpoint_xs[1] - 24, endpoint_y), fill=INK, width=4)
        for endpoint_x, node in zip(endpoint_xs, (int(command["a"]), int(command["b"])), strict=True):
            draw.ellipse((endpoint_x - 25, endpoint_y - 25, endpoint_x + 25, endpoint_y + 25), fill=WHITE, outline=INK, width=3)
            _pips(draw, endpoint_x, endpoint_y, node)
        for mark in range(int(scene["decorations"][index])):
            x = left + 49 + mark * 28
            draw.ellipse((x - 4, bottom - 24, x + 4, bottom - 16), fill=(180, 182, 180))
    return image


def analytic_gold(scene: dict) -> str:
    final, _ = _replay(scene["initial_edges"], scene["commands"])
    return "yes" if _connected(final, scene["query_nodes"]) else "no"


def margin(scene: dict) -> float:
    _ = (scene["initial_edges"], scene["query_nodes"], scene["commands"])
    return 14.0


def is_quarantined(scene: dict) -> bool:
    _ = (scene["initial_edges"], scene["query_nodes"], scene["commands"])
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
