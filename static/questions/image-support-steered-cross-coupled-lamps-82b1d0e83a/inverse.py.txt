"""Independent image-only inverse for the cross-coupled lamp diagram."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

_VIOLET = np.array((103, 79, 166), dtype=np.int16)
_GOLD = np.array((244, 181, 52), dtype=np.int16)
_CYAN = np.array((51, 190, 202), dtype=np.int16)
_CORAL = np.array((235, 111, 103), dtype=np.int16)
_NAVY = np.array((27, 42, 67), dtype=np.int16)


def _near(array: np.ndarray, color: np.ndarray, radius: int = 34) -> np.ndarray:
    delta = array.astype(np.int32) - color.astype(np.int32)
    return np.sum(delta * delta, axis=2) <= radius * radius


def _components(mask: np.ndarray, minimum: int) -> list[dict]:
    h, w = mask.shape
    seen = np.zeros((h, w), dtype=bool)
    output = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            pts = np.asarray(points)
            output.append({
                "area": len(points),
                "y0": int(pts[:, 0].min()), "y1": int(pts[:, 0].max()),
                "x0": int(pts[:, 1].min()), "x1": int(pts[:, 1].max()),
                "cy": float(pts[:, 0].mean()), "cx": float(pts[:, 1].mean()),
            })
    return output


def _replay(a: int, b: int, ops: list[int]) -> tuple[int, int]:
    for op in ops:
        if op == 0:
            if a:
                b ^= 1
        else:
            if b:
                a ^= 1
    return a, b


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    if array.ndim != 3 or array.shape[0] < 300 or array.shape[1] < 700:
        raise ValueError("unexpected or damaged canvas")

    # Recover exactly two violet lamp rings without using fixed coordinates.
    rings = []
    for comp in _components(_near(array, _VIOLET), 120):
        bw, bh = comp["x1"] - comp["x0"] + 1, comp["y1"] - comp["y0"] + 1
        if 25 <= bw <= 44 and 25 <= bh <= 44 and comp["area"] <= 900:
            rings.append(comp)
    if len(rings) != 2:
        raise ValueError(f"expected two intact lamp rings, found {len(rings)}")
    rings.sort(key=lambda item: item["cy"])
    if rings[1]["cy"] - rings[0]["cy"] < 30:
        raise ValueError("lamp order is unresolved")
    gold = _near(array, _GOLD)
    states = []
    for ring in rings:
        x0, x1 = ring["x0"] + 7, ring["x1"] - 6
        y0, y1 = ring["y0"] + 7, ring["y1"] - 6
        if x1 <= x0 or y1 <= y0:
            raise ValueError("damaged lamp interior")
        fraction = float(gold[y0:y1 + 1, x0:x1 + 1].mean())
        if fraction >= 0.55:
            states.append(1)
        elif fraction <= 0.08:
            states.append(0)
        else:
            raise ValueError("ambiguous lamp fill")

    # Colored card bodies locate three detached operations; direction comes
    # only from the dark arrow geometry, not the palette.
    card_mask = _near(array, _CYAN) | _near(array, _CORAL)
    cards = []
    for comp in _components(card_mask, 1800):
        bw, bh = comp["x1"] - comp["x0"] + 1, comp["y1"] - comp["y0"] + 1
        if 70 <= bw <= 105 and 48 <= bh <= 84 and comp["area"] >= 3000:
            cards.append(comp)
    if len(cards) != 3:
        raise ValueError(f"expected three intact operation cards, found {len(cards)}")
    cards.sort(key=lambda item: item["cx"])
    if min(cards[i + 1]["cx"] - cards[i]["cx"] for i in range(2)) < 120:
        raise ValueError("card order is unresolved")

    navy = _near(array, _NAVY, 38)
    ops = []
    for card in cards:
        pad = 8
        x0, x1 = card["x0"] + pad, card["x1"] - pad
        y0, y1 = card["y0"] + pad, card["y1"] - pad
        crop = navy[y0:y1 + 1, x0:x1 + 1]
        ys, xs = np.nonzero(crop)
        if len(ys) < 250 or (ys.max() - ys.min()) < 38:
            raise ValueError("operation arrow is missing or damaged")
        mid = (ys.min() + ys.max()) / 2.0
        upper = int(np.sum(ys < mid - 5))
        lower = int(np.sum(ys > mid + 5))
        if abs(lower - upper) < 70:
            raise ValueError("operation arrow direction is ambiguous")
        ops.append(0 if lower > upper else 1)

    a, b = _replay(states[0], states[1], ops)
    return ("on" if a else "off") + "-" + ("on" if b else "off")
