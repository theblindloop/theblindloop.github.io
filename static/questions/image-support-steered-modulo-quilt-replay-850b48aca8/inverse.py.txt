"""Independent PNG-only inverse for the modulo quilt."""

from __future__ import annotations

import numpy as np
from PIL import Image

ROWS = 6
COLS = 8
CELL = 48
_PALETTES = np.array((
    ((52, 184, 188), (232, 105, 119)),
    ((55, 150, 218), (232, 132, 62)),
    ((43, 171, 181), (226, 92, 96)),
    ((37, 173, 213), (224, 103, 132)),
), dtype=np.float64)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    if array.shape != (512, 512, 3):
        raise ValueError("invalid canvas")
    maximum = array.max(axis=2)
    minimum = array.min(axis=2)
    saturated = (maximum - minimum > 45) & (maximum < 245)
    flat_colors = _PALETTES.reshape(-1, 3)
    palette_distance = np.linalg.norm(array.astype(np.float64)[:, :, None, :] - flat_colors[None, None, :, :], axis=3)
    tile_ink = palette_distance.min(axis=2) < 18.0
    points = np.argwhere(tile_ink)
    if len(points) < 90000:
        raise ValueError("complete operation quilt not found")
    sy0, sx0 = points.min(axis=0)
    sy1, sx1 = points.max(axis=0) + 1

    # Saturated title/footer marks are tiny detached components. Determine the broad
    # board support from per-axis saturated coverage instead of exact coordinates.
    x_coverage = tile_ink.mean(axis=0)
    y_coverage = tile_ink.mean(axis=1)
    xs = np.flatnonzero(x_coverage > 0.40)
    ys = np.flatnonzero(y_coverage > 0.60)
    if not len(xs) or not len(ys):
        raise ValueError("board axes missing")
    # Tile fill begins immediately inside the four-pixel frame.
    left, right = int(xs.min()) - 4, int(xs.max()) + 5
    top, bottom = int(ys.min()) - 4, int(ys.max()) + 5
    if abs((right - left) - COLS * CELL) > 4 or abs((bottom - top) - ROWS * CELL) > 4:
        raise ValueError("invalid board geometry")

    observed = []
    for index in range(ROWS * COLS):
        row, column = divmod(index, COLS)
        x0 = left + column * CELL + 7
        x1 = left + (column + 1) * CELL - 7
        y0 = top + row * CELL + 7
        y1 = top + (row + 1) * CELL - 7
        patch = array[y0:y1, x0:x1].astype(np.float64)
        if patch.shape[:2] != (34, 34):
            raise ValueError("cropped tile")
        mean = patch.mean(axis=(0, 1))
        spread = np.sqrt(((patch - mean) ** 2).sum(axis=2)).mean()
        if spread > 8.0:
            raise ValueError("damaged or impure tile")
        observed.append(mean)

    means = np.asarray(observed)
    palette_scores = np.min(np.linalg.norm(means[:, None, :] - _PALETTES.reshape(-1, 3)[None, :, :], axis=2), axis=1)
    if float(palette_scores.max()) > 14.0:
        raise ValueError("unknown tile color")
    # Select the one palette pair that explains every tile, then decode its two public colors.
    fits = []
    for palette in _PALETTES:
        distances = np.linalg.norm(means[:, None, :] - palette[None, :, :], axis=2)
        fits.append(float(distances.min(axis=1).max()))
    palette_index = int(np.argmin(fits))
    if fits[palette_index] > 14.0:
        raise ValueError("inconsistent palette")
    distances = np.linalg.norm(means[:, None, :] - _PALETTES[palette_index][None, :, :], axis=2)
    if np.any(np.abs(distances[:, 0] - distances[:, 1]) < 25.0):
        raise ValueError("ambiguous tile")
    operations = distances.argmin(axis=1) + 1
    if not (18 <= int((operations == 1).sum()) <= 30):
        raise ValueError("implausibly imbalanced quilt")
    return str(int(operations.sum()) % 3)
