"""Deterministic sampler and renderer for a dense modulo-three quilt."""

from __future__ import annotations

import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

CANVAS = 512
ROWS = 6
COLS = 8
CELL = 48
PALETTES = (
    ((52, 184, 188), (232, 105, 119)),
    ((55, 150, 218), (232, 132, 62)),
    ((43, 171, 181), (226, 92, 96)),
    ((37, 173, 213), (224, 103, 132)),
)
ACCENTS = ((29, 37, 53), (40, 48, 63), (30, 45, 49))
FONT_PATH = Path("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf")


def _replay(operations: str) -> str:
    if len(operations) != ROWS * COLS or any(value not in "12" for value in operations):
        raise ValueError("operations must contain forty-eight 1/2 updates")
    state = 0
    for value in operations:
        state = (state + int(value)) % 3
    return str(state)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = int(seed) % 3
    for _ in range(10000):
        operations = "".join(rng.choice("12") for _ in range(ROWS * COLS))
        count_one = operations.count("1")
        if 18 <= count_one <= 30 and _replay(operations) == str(target):
            break
    else:
        raise RuntimeError("unable to sample balanced operation field")
    return {
        "operations": operations,
        "palette": rng.randrange(len(PALETTES)),
        "accent": rng.randrange(len(ACCENTS)),
        "board_shift": [rng.randint(-12, 12), rng.randint(-10, 10)],
    }


def analytic_gold(scene: dict) -> str:
    return _replay(str(scene["operations"]))


def margin(scene: dict) -> float:
    baseline = analytic_gold(scene)
    distances = []
    operations = str(scene["operations"])
    for index, value in enumerate(operations):
        changed = operations[:index] + ("2" if value == "1" else "1") + operations[index + 1 :]
        outcome = _replay(changed)
        distances.append(min((int(outcome) - int(baseline)) % 3, (int(baseline) - int(outcome)) % 3))
    return float(min(distances))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    operations = str(scene["operations"])
    _replay(operations)
    palette = PALETTES[int(scene["palette"])]
    accent = ACCENTS[int(scene["accent"])]
    shift_x, shift_y = (int(value) for value in scene["board_shift"])
    board_w, board_h = COLS * CELL, ROWS * CELL
    left = (CANVAS - board_w) // 2 + shift_x
    top = (CANVAS - board_h) // 2 + shift_y

    image = Image.new("RGB", (CANVAS, CANVAS), (250, 249, 245))
    draw = ImageDraw.Draw(image)
    title = ImageFont.truetype(str(FONT_PATH), 21)
    small = ImageFont.truetype(str(FONT_PATH), 13)
    draw.text((CANVAS // 2, 35), "MODULO-3 UPDATE QUILT", fill=accent, anchor="mm", font=title)
    draw.text((CANVAS // 2, 59), "START 0  •  READ ROWS  →", fill=accent, anchor="mm", font=small)

    for index, value in enumerate(operations):
        row, column = divmod(index, COLS)
        x0 = left + column * CELL
        y0 = top + row * CELL
        draw.rectangle((x0, y0, x0 + CELL, y0 + CELL), fill=palette[int(value) - 1])

    # A thick continuous lattice makes the rectangular program and its completeness visible.
    line = 4
    draw.rectangle((left, top, left + board_w, top + board_h), outline=accent, width=line)
    for column in range(1, COLS):
        draw.line((left + column * CELL, top, left + column * CELL, top + board_h), fill=accent, width=line)
    for row in range(1, ROWS):
        draw.line((left, top + row * CELL, left + board_w, top + row * CELL), fill=accent, width=line)
    draw.text((CANVAS // 2, 457), "COOL BLUE +1   •   WARM ORANGE +2   •   WRAP AFTER 2", fill=accent, anchor="mm", font=small)
    return image
