from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

PALETTE = {
    "background": (250, 250, 247),
    "navy": (24, 50, 74),
    "cyan": (25, 175, 196),
    "orange": (242, 165, 65),
    "lavender": (154, 120, 208),
    "gold": (214, 167, 44),
    "red": (217, 75, 75),
    "green": (47, 155, 98),
    "blue": (57, 119, 214),
}


def _mask(array: np.ndarray, color: tuple[int, int, int], tolerance: float = 18.0) -> np.ndarray:
    return np.linalg.norm(array.astype(np.float32) - np.asarray(color, dtype=np.float32), axis=2) <= tolerance


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result: list[np.ndarray] = []
    for sy, sx in np.argwhere(mask):
        if seen[sy, sx]:
            continue
        queue = deque([(int(sy), int(sx))])
        seen[sy, sx] = True
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y-1,x),(y+1,x),(y,x-1),(y,x+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            result.append(np.asarray(points, dtype=np.int32))
    return result


def _centroid_x(component: np.ndarray) -> float:
    return float(component[:, 1].mean())


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 200 or array.shape[1] < 600:
        raise ValueError("unsupported raster")

    # The largest navy component must contain an unbroken thin horizontal rail.
    navy_components = _components(_mask(array, PALETTE["navy"]), 80)
    if not navy_components:
        raise ValueError("carrier rail missing")
    carrier = max(navy_components, key=len)
    xs = carrier[:, 1]
    ys = carrier[:, 0]
    x0, x1 = int(xs.min()), int(xs.max())
    if x1 - x0 < 0.52 * array.shape[1] or x1 - x0 > 0.75 * array.shape[1]:
        raise ValueError("carrier rail span invalid")
    counts = np.bincount(xs - x0, minlength=x1 - x0 + 1)
    if len(counts) < 20 or int(counts[6:-6].min()) < 5:
        raise ValueError("carrier rail interrupted")
    rail_y = float(np.median(ys[(xs > x0 + 50) & (xs < x1 - 50)]))

    # Recover the gold station and the eight source beads purely by color components.
    gold = _components(_mask(array, PALETTE["gold"]), 80)
    if len(gold) != 1:
        raise ValueError("gold source station ambiguous")
    gold_x = _centroid_x(gold[0])
    bead_items: list[tuple[float, int]] = []
    for value, name in ((0, "cyan"), (1, "orange")):
        for component in _components(_mask(array, PALETTE[name]), 180):
            cy = float(component[:, 0].mean())
            if cy < rail_y:
                bead_items.append((_centroid_x(component), value))
    bead_items.sort()
    if len(bead_items) != 8 or bead_items[0][0] <= gold_x:
        raise ValueError("source bead string is incomplete")
    bits = [value for _, value in bead_items]
    runs: list[int] = []
    for value in bits:
        if not runs or value != bits[sum(runs) - 1]:
            runs.append(1)
        else:
            runs[-1] += 1
    if len(runs) != 4:
        raise ValueError("source does not expose four maximal runs")

    # Locate colored stations, then recover four unary stack columns after each.
    stations: list[tuple[float, str]] = []
    for name in ("red", "green", "blue"):
        components = _components(_mask(array, PALETTE[name]), 80)
        if len(components) != 1:
            raise ValueError(f"{name} station missing")
        stations.append((_centroid_x(components[0]), name))
    stations.sort()

    blocks = _components(_mask(array, PALETTE["lavender"]), 100)
    if len(blocks) != 24:
        raise ValueError("unary stack squares are incomplete")
    block_xs = sorted(_centroid_x(component) for component in blocks)
    unique_columns: list[list[float]] = []
    for x in block_xs:
        if not unique_columns or abs(x - float(np.mean(unique_columns[-1]))) > 3.0:
            unique_columns.append([x])
        else:
            unique_columns[-1].append(x)
    if len(unique_columns) != 12:
        raise ValueError("expected twelve stack columns")
    columns = [(float(np.mean(group)), len(group)) for group in unique_columns]
    codes: dict[str, list[int]] = {}
    for index, (sx, name) in enumerate(stations):
        end = stations[index + 1][0] if index + 1 < len(stations) else float(array.shape[1])
        values = [height for x, height in columns if sx < x < end]
        if len(values) != 4:
            raise ValueError(f"{name} station does not own four stacks")
        codes[name] = values
    matches = [name for name, values in codes.items() if values == runs]
    if len(matches) != 1:
        raise ValueError("pixel correspondence is not unique")
    return matches[0]
