from __future__ import annotations

import random
from PIL import Image, ImageDraw

BACKGROUND = "#FAFAF7"
NAVY = "#18324A"
CYAN = "#19AFC4"
ORANGE = "#F2A541"
LAVENDER = "#9A78D0"
TAG_RGB = {"red": "#D94B4B", "green": "#2F9B62", "blue": "#3977D6"}
ANSWERS = ("red", "green", "blue")


def _compositions() -> list[list[int]]:
    return [
        [a, b, c, 8 - a - b - c]
        for a in range(1, 5)
        for b in range(1, 5)
        for c in range(1, 5)
        if 1 <= 8 - a - b - c <= 4
    ]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    choices = _compositions()
    source = list(choices[rng.randrange(len(choices))])
    distractors = [c for c in choices if c != source and sum(abs(x - y) for x, y in zip(c, source)) >= 2]
    rng.shuffle(distractors)
    target_color = ANSWERS[int(seed) % 3]
    colors = list(ANSWERS)
    rng.shuffle(colors)
    codes: list[list[int]] = []
    di = 0
    for color in colors:
        if color == target_color:
            codes.append(list(source))
        else:
            while distractors[di] in codes:
                di += 1
            codes.append(list(distractors[di]))
            di += 1
    return {
        "source_runs": source,
        "start_bit": rng.randrange(2),
        "candidate_codes": codes,
        "candidate_colors": colors,
        "width": 1200,
        "height": 500,
        "rail_y": 286,
    }


def _source_bits(runs: list[int], start: int) -> list[int]:
    bits: list[int] = []
    value = int(start)
    for length in runs:
        bits.extend([value] * int(length))
        value = 1 - value
    return bits


def render(scene: dict) -> Image.Image:
    width, height, rail_y = int(scene["width"]), int(scene["height"]), int(scene["rail_y"])
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    # One unbroken carrier, deliberately confined to the central image region.
    draw.rounded_rectangle((220, rail_y - 4, 980, rail_y + 4), radius=4, fill=NAVY)
    # Gold source station and stem.
    draw.rectangle((237, rail_y + 4, 243, rail_y + 23), fill=NAVY)
    draw.ellipse((229, rail_y + 20, 251, rail_y + 42), fill="#D6A72C", outline=NAVY, width=2)
    bits = _source_bits(scene["source_runs"], scene["start_bit"])
    for i, bit in enumerate(bits):
        x = 270 + 24 * i
        draw.rectangle((x, rail_y - 29, x + 17, rail_y - 12), fill=CYAN if bit == 0 else ORANGE)
        draw.rectangle((x + 7, rail_y - 12, x + 10, rail_y - 4), fill=NAVY)

    starts = (500, 660, 820)
    for station_x, color, code in zip(starts, scene["candidate_colors"], scene["candidate_codes"], strict=True):
        draw.rectangle((station_x - 3, rail_y + 4, station_x + 3, rail_y + 23), fill=NAVY)
        draw.ellipse((station_x - 11, rail_y + 20, station_x + 11, rail_y + 42), fill=TAG_RGB[color], outline=NAVY, width=2)
        for j, stack_height in enumerate(code):
            x = station_x + 23 + j * 29
            for unit in range(int(stack_height)):
                y1 = rail_y - 12 - unit * 15
                draw.rectangle((x, y1 - 12, x + 12, y1), fill=LAVENDER)
            draw.rectangle((x + 5, rail_y - 12, x + 8, rail_y - 4), fill=NAVY)
    return image


def analytic_gold(scene: dict) -> str:
    target = list(scene["source_runs"])
    matches = [i for i, code in enumerate(scene["candidate_codes"]) if list(code) == target]
    if len(matches) != 1:
        raise ValueError("scene does not have one matching code")
    return str(scene["candidate_colors"][matches[0]])


def margin(scene: dict) -> float:
    target = list(scene["source_runs"])
    distances = [sum(abs(int(a) - int(b)) for a, b in zip(target, code)) for code in scene["candidate_codes"] if list(code) != target]
    return float(min(distances))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
