"""Independent pixel-only inverse arm."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

CENTERLINE = np.array((105, 163, 184), dtype=np.uint8)
GREEN = np.array((32, 166, 92), dtype=np.uint8)
RED = np.array((216, 68, 72), dtype=np.uint8)


def _centroid(rgb: np.ndarray, color: np.ndarray) -> np.ndarray:
    coords = np.argwhere(np.all(rgb == color, axis=2))
    if len(coords) < 20:
        raise ValueError("missing endpoint marker")
    return coords.mean(axis=0).astype(float)


def _largest_component(mask: np.ndarray) -> set[tuple[int, int]]:
    remaining = set(map(tuple, np.argwhere(mask)))
    largest: set[tuple[int, int]] = set()
    height, width = mask.shape
    while remaining:
        root = remaining.pop()
        component = {root}
        queue = [root]
        while queue:
            y, x = queue.pop()
            for node in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= node[0] < height and 0 <= node[1] < width and node in remaining:
                    remaining.remove(node)
                    component.add(node)
                    queue.append(node)
        if len(component) > len(largest):
            largest = component
    return largest


def _ordered_path(component: set[tuple[int, int]], start_marker: np.ndarray, finish_marker: np.ndarray) -> list[tuple[int, int]]:
    def neighbors(node: tuple[int, int]) -> list[tuple[int, int]]:
        y, x = node
        return [p for p in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)) if p in component]

    ends = [node for node in component if len(neighbors(node)) == 1]
    if len(component) < 500 or len(ends) != 2:
        raise ValueError("centerline is broken, branched, or too short")
    start = min(ends, key=lambda p: float(np.linalg.norm(np.asarray(p) - start_marker)))
    finish = ends[0] if ends[1] == start else ends[1]
    if np.linalg.norm(np.asarray(start) - start_marker) > 25 or np.linalg.norm(np.asarray(finish) - finish_marker) > 25:
        raise ValueError("centerline does not join both endpoint markers")
    ordered = [start]
    previous = None
    current = start
    while current != finish:
        choices = [node for node in neighbors(current) if node != previous]
        if len(choices) != 1:
            raise ValueError("ambiguous traversal")
        previous, current = current, choices[0]
        ordered.append(current)
    if len(ordered) != len(component):
        raise ValueError("unused centerline pixels")
    return ordered


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    start_marker = _centroid(rgb, GREEN)
    finish_marker = _centroid(rgb, RED)
    component = _largest_component(np.all(rgb == CENTERLINE, axis=2))
    path = _ordered_path(component, start_marker, finish_marker)
    center = (len(path) - 1) / 2.0
    before = np.asarray(path[int(center) - 7], dtype=float)
    after = np.asarray(path[int(center) + 8], dtype=float)
    if np.linalg.norm(after - before) < 12:
        raise ValueError("midpoint too close to a bend")
    before_distance = float(np.linalg.norm(before - finish_marker))
    after_distance = float(np.linalg.norm(after - finish_marker))
    if abs(before_distance - after_distance) < 2.0:
        raise ValueError("toward/away boundary is ambiguous")
    return "toward" if after_distance < before_distance else "away"
