"""Deterministic renderer and analytic arm for midroute_finish_approach."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

SIZE = 640
BACKGROUND = (248, 246, 238)
NAVY = (24, 45, 68)
CENTERLINE = (105, 163, 184)
GREEN = (32, 166, 92)
RED = (216, 68, 72)
INK = (42, 48, 55)


def _midpoint_info(points: list[list[int]]) -> tuple[tuple[float, float], tuple[float, float], float, float]:
    lengths = [math.dist(a, b) for a, b in zip(points, points[1:])]
    target = sum(lengths) / 2.0
    walked = 0.0
    for a, b, length in zip(points, points[1:], lengths):
        if walked + length >= target:
            offset = target - walked
            ux = (b[0] - a[0]) / length
            uy = (b[1] - a[1]) / length
            midpoint = (a[0] + ux * offset, a[1] + uy * offset)
            bend_clearance = min(offset, length - offset)
            finish = points[-1]
            projection = (finish[0] - midpoint[0]) * ux + (finish[1] - midpoint[1]) * uy
            return midpoint, (ux, uy), bend_clearance, projection
        walked += length
    raise ValueError("empty route")


def _transform(points: list[list[int]], code: int) -> list[list[int]]:
    def one(x: int, y: int) -> list[int]:
        if code == 0:
            return [x, y]
        if code == 1:
            return [SIZE - x, y]
        if code == 2:
            return [x, SIZE - y]
        if code == 3:
            return [SIZE - x, SIZE - y]
        if code == 4:
            return [y, x]
        if code == 5:
            return [SIZE - y, x]
        if code == 6:
            return [y, SIZE - x]
        return [SIZE - y, SIZE - x]

    return [one(x, y) for x, y in points]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    desired = "toward" if seed % 2 == 0 else "away"
    xs = [60, 145, 230, 315, 400, 490, 580]
    y_levels = [240, 320, 400]
    for _ in range(20000):
        route_ys = [rng.choice(y_levels)]
        while len(route_ys) < len(xs):
            choices = [value for value in y_levels if value != route_ys[-1]]
            route_ys.append(rng.choice(choices))
        points: list[list[int]] = [[xs[0], route_ys[0]]]
        for index in range(1, len(xs)):
            points.append([xs[index], route_ys[index - 1]])
            points.append([xs[index], route_ys[index]])
        _, _, bend_clearance, projection = _midpoint_info(points)
        label = "toward" if projection > 0 else "away"
        if label != desired or bend_clearance < 30 or abs(projection) < 65:
            continue
        transformed = _transform(points, rng.randrange(8))
        return {"points": transformed}
    raise RuntimeError("could not sample a stable route")


def render(scene: dict) -> Image.Image:
    points = [tuple(point) for point in scene["points"]]
    image = Image.new("RGB", (SIZE, SIZE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.line(points, fill=NAVY, width=11, joint="curve")
    draw.line(points, fill=CENTERLINE, width=1)
    radius = 13
    for point, color in ((points[0], GREEN), (points[-1], RED)):
        box = (point[0] - radius, point[1] - radius, point[0] + radius, point[1] + radius)
        draw.ellipse(box, fill=BACKGROUND, outline=color, width=6)
    draw.rounded_rectangle((174, 12, 466, 43), radius=8, fill=(237, 235, 226), outline=(205, 202, 192), width=1)
    draw.text((193, 22), "GREEN = START     RED = FINISH", fill=INK)
    return image


def analytic_gold(scene: dict) -> str:
    _, _, _, projection = _midpoint_info(scene["points"])
    return "toward" if projection > 0 else "away"


def margin(scene: dict) -> float:
    _, _, bend_clearance, projection = _midpoint_info(scene["points"])
    return float(min(bend_clearance, abs(projection)))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 24.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
