"""Renderer and analytic arm for the four-junction link-cover world."""

from __future__ import annotations

import itertools
import random
from copy import deepcopy

from PIL import Image, ImageDraw


ROWS = 3
COLS = 4
NODE_COUNT = ROWS * COLS
EDGE_COUNT = 11
COVER_LIMIT = 4

# Every template is a connected subgraph of the 3-by-4 grid.  Both classes
# have the same sorted degree sequence:
# (1,1,1,1,1,1,1,2,3,3,3,4).
_YES_TEMPLATES = (
    ((0, 1), (1, 2), (1, 5), (3, 7), (4, 5), (5, 6), (5, 9),
     (6, 7), (7, 11), (8, 9), (9, 10)),
    ((0, 4), (1, 2), (2, 3), (2, 6), (4, 5), (4, 8), (5, 6),
     (6, 7), (6, 10), (9, 10), (10, 11)),
)

_NO_TEMPLATES = (
    ((0, 1), (0, 4), (2, 6), (3, 7), (4, 5), (4, 8), (5, 6),
     (5, 9), (6, 7), (6, 10), (7, 11)),
    ((0, 1), (1, 2), (1, 5), (3, 7), (4, 5), (4, 8), (5, 6),
     (5, 9), (6, 7), (6, 10), (7, 11)),
    ((0, 1), (1, 2), (1, 5), (3, 7), (4, 5), (5, 6), (5, 9),
     (6, 7), (6, 10), (7, 11), (8, 9)),
    ((0, 1), (1, 5), (2, 6), (3, 7), (4, 5), (5, 6), (5, 9),
     (6, 7), (7, 11), (8, 9), (9, 10)),
)


def _minimum_vertex_cover_size(edges: list[list[int]] | tuple[tuple[int, int], ...]) -> int:
    """Return the exact minimum number of junctions touching all links."""

    normalized = [(int(a), int(b)) for a, b in edges]
    vertices = range(NODE_COUNT)
    for size in range(NODE_COUNT + 1):
        for chosen_tuple in itertools.combinations(vertices, size):
            chosen = set(chosen_tuple)
            if all(left in chosen or right in chosen for left, right in normalized):
                return size
    raise ValueError("no vertex cover exists")


def _layout(rotation: int, mirror: bool, offset: tuple[int, int]) -> list[list[int]]:
    cx, cy = 240, 196
    result: list[list[int]] = []
    for row in range(ROWS):
        for col in range(COLS):
            x = (col - 1.5) * 110
            y = (row - 1.0) * 110
            if mirror:
                x = -x
            for _ in range(rotation % 4):
                x, y = -y, x
            result.append([int(round(cx + x + offset[0])), int(round(cy + y + offset[1]))])
    return result


def sample_scene(seed: int) -> dict:
    """Sample paired, cue-controlled yes/no grid networks deterministically."""

    seed = int(seed)
    want_yes = seed % 2 == 0
    pair_rng = random.Random((seed // 2) ^ 0x4A17C09D)
    templates = _YES_TEMPLATES if want_yes else _NO_TEMPLATES
    edges = templates[pair_rng.randrange(len(templates))]
    rotation = pair_rng.randrange(4)
    mirror = bool(pair_rng.randrange(2))
    offset = (pair_rng.randint(-4, 4), pair_rng.randint(-4, 4))
    palette_index = pair_rng.randrange(2)
    node_color = ("#38BDF8", "#F59E0B")[palette_index]
    return {
        "nodes": _layout(rotation, mirror, offset),
        "edges": [[left, right] for left, right in edges],
        "colors": {
            "node": node_color,
            "link": "#263244",
            "background": "#F8FAFC",
        },
        "canvas": [480, 392],
        "link_width": 12,
        "node_radius": 17,
    }


def analytic_gold(scene: dict) -> str:
    return "yes" if _minimum_vertex_cover_size(scene["edges"]) <= COVER_LIMIT else "no"


def margin(scene: dict) -> float:
    """Integer cover-number distance between the two supported classes."""

    cover_size = _minimum_vertex_cover_size(scene["edges"])
    return float(abs(cover_size - (COVER_LIMIT + (0 if cover_size <= COVER_LIMIT else 1)))) + 1.0


def is_quarantined(scene: dict) -> bool:
    try:
        cover_size = _minimum_vertex_cover_size(scene["edges"])
        return not (
            len(scene["nodes"]) == NODE_COUNT
            and len(scene["edges"]) == EDGE_COUNT
            and cover_size in (4, 5)
            and int(scene["link_width"]) >= 10
            and int(scene["node_radius"]) >= 15
        )
    except Exception:
        return True


def render(scene: dict) -> Image.Image:
    width, height = scene["canvas"]
    colors = scene["colors"]
    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), colors["background"])
    draw = ImageDraw.Draw(image)
    points = [(x * scale, y * scale) for x, y in scene["nodes"]]
    for left, right in scene["edges"]:
        draw.line(
            [points[left], points[right]],
            fill=colors["link"],
            width=int(scene["link_width"]) * scale,
        )
    radius = int(scene["node_radius"]) * scale
    for x, y in points:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=colors["node"])
    return image.resize((width, height), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["edges"] = list(reversed(twin["edges"]))
    return [("reverse_edge_enumeration", twin)]
