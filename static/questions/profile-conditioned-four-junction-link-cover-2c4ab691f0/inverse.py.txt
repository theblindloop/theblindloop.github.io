"""Independent PNG-only inverse arm for the junction-cover decision."""

from __future__ import annotations

import itertools
from collections import deque

import numpy as np
from PIL import Image


_PALETTE = np.array(
    [
        (248, 250, 252),  # background
        (38, 50, 68),     # dark links
        (56, 189, 248),   # blue junction palette
        (245, 158, 11),   # amber junction palette
    ],
    dtype=np.int32,
)


def _components(mask: np.ndarray, *, eight: bool) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    neighbors = [(-1, 0), (1, 0), (0, -1), (0, 1)]
    if eight:
        neighbors += [(-1, -1), (-1, 1), (1, -1), (1, 1)]
    result: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = deque([(int(y), int(x))])
        pixels: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            pixels.append((cy, cx))
            for oy, ox in neighbors:
                ny, nx = cy + oy, cx + ox
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        result.append(np.asarray(pixels, dtype=np.int32))
    return result


def _recover_graph(image: Image.Image) -> list[set[int]]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    distances = ((arr[:, :, None, :] - _PALETTE[None, None, :, :]) ** 2).sum(axis=3)
    labels = distances.argmin(axis=2)
    node_parts = [part for part in _components(labels >= 2, eight=True) if len(part) >= 450]
    if len(node_parts) != 12:
        raise ValueError(f"expected 12 visible junctions, recovered {len(node_parts)}")
    centers = [
        (float(part[:, 1].mean()), float(part[:, 0].mean())) for part in node_parts
    ]
    link_parts = [part for part in _components(labels == 1, eight=True) if len(part) >= 250]
    if len(link_parts) != 11:
        raise ValueError(f"expected 11 complete dark links, recovered {len(link_parts)}")

    adjacency = [set() for _ in centers]
    for part in link_parts:
        xs = part[:, 1].astype(np.float64)
        ys = part[:, 0].astype(np.float64)
        endpoint_scores = []
        for index, (cx, cy) in enumerate(centers):
            minimum = float(np.min((xs - cx) ** 2 + (ys - cy) ** 2))
            endpoint_scores.append((minimum, index))
        endpoint_scores.sort()
        if endpoint_scores[1][0] > 27.0**2:
            raise ValueError("a dark link does not visibly terminate at two junctions")
        left, right = endpoint_scores[0][1], endpoint_scores[1][1]
        if left == right or right in adjacency[left]:
            raise ValueError("ambiguous or duplicate dark link recovered")
        adjacency[left].add(right)
        adjacency[right].add(left)
    return adjacency


def _minimum_cover_size(adjacency: list[set[int]]) -> int:
    edges = [(u, v) for u, neighbors in enumerate(adjacency) for v in neighbors if u < v]
    for size in range(len(adjacency) + 1):
        for chosen_tuple in itertools.combinations(range(len(adjacency)), size):
            chosen = set(chosen_tuple)
            if all(u in chosen or v in chosen for u, v in edges):
                return size
    raise ValueError("pixel graph has no vertex cover")


def decision_from_image(image: Image.Image) -> str:
    return "yes" if _minimum_cover_size(_recover_graph(image)) <= 4 else "no"
