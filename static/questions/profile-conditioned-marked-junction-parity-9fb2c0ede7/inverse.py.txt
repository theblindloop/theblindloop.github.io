"""Independent pixel-only inverse for the marked-junction parity world."""

from __future__ import annotations

from collections import deque
import hashlib

import numpy as np
from PIL import Image


ROUTE = np.array((37, 61, 93), dtype=np.float32)
GREEN = np.array((40, 165, 95), dtype=np.float32)
RED = np.array((216, 75, 75), dtype=np.float32)
MARKERS = (
    np.array((214, 146, 45), dtype=np.float32),
    np.array((136, 89, 181), dtype=np.float32),
)
_DECISION_CACHE: dict[str, str] = {}


def _close_to(arr: np.ndarray, color: np.ndarray, tolerance: float) -> np.ndarray:
    return np.linalg.norm(arr - color, axis=2) <= tolerance


def _components(mask: np.ndarray, minimum: int) -> list[tuple[float, float, int]]:
    """Find 8-connected colored components and return centroid x/y and area."""

    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components: list[tuple[float, float, int]] = []
    ys, xs = np.where(mask)
    for start_y, start_x in zip(ys.tolist(), xs.tolist()):
        if seen[start_y, start_x]:
            continue
        stack = [(start_y, start_x)]
        seen[start_y, start_x] = True
        total = 0
        sum_x = 0.0
        sum_y = 0.0
        while stack:
            y, x = stack.pop()
            total += 1
            sum_x += x
            sum_y += y
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not dx and not dy:
                        continue
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        if total >= minimum:
            components.append((sum_x / total, sum_y / total, total))
    components.sort(key=lambda item: item[2], reverse=True)
    return components


def _largest_centroid(mask: np.ndarray, what: str) -> tuple[float, float]:
    components = _components(mask, 20)
    if not components:
        raise ValueError(f"could not segment {what}")
    return components[0][0], components[0][1]


def _dilate(mask: np.ndarray, radius: int) -> np.ndarray:
    result = mask.copy()
    for dy in range(-radius, radius + 1):
        for dx in range(-radius, radius + 1):
            if dx * dx + dy * dy > radius * radius:
                continue
            shifted = np.zeros_like(mask)
            source_y0 = max(0, -dy)
            source_y1 = min(mask.shape[0], mask.shape[0] - dy)
            source_x0 = max(0, -dx)
            source_x1 = min(mask.shape[1], mask.shape[1] - dx)
            target_y0 = max(0, dy)
            target_y1 = target_y0 + (source_y1 - source_y0)
            target_x0 = max(0, dx)
            target_x1 = target_x0 + (source_x1 - source_x0)
            shifted[target_y0:target_y1, target_x0:target_x1] = mask[source_y0:source_y1, source_x0:source_x1]
            result |= shifted
    return result


def _route_pixels(arr: np.ndarray, start: tuple[float, float], finish: tuple[float, float]) -> np.ndarray:
    route_mask = _close_to(arr, ROUTE, 78.0)
    green_mask = _close_to(arr, GREEN, 75.0)
    red_mask = _close_to(arr, RED, 75.0)
    # The ring outline is intentionally drawn over the stroke.  A small
    # dilation bridges its antialiased four-to-five pixel interruption while
    # remaining far below the generous branch separation in sampled scenes.
    walkable = _dilate(route_mask | green_mask | red_mask, 5)
    start_x, start_y = round(start[0]), round(start[1])
    finish_x, finish_y = round(finish[0]), round(finish[1])
    if not (0 <= start_x < arr.shape[1] and 0 <= start_y < arr.shape[0]):
        raise ValueError("green centroid is outside image")
    if not (0 <= finish_x < arr.shape[1] and 0 <= finish_y < arr.shape[0]):
        raise ValueError("red centroid is outside image")
    if not walkable[start_y, start_x] or not walkable[finish_y, finish_x]:
        raise ValueError("terminal does not meet the route")

    distance = np.full(walkable.shape, -1, dtype=np.int32)
    distance[start_y, start_x] = 0
    queue: deque[tuple[int, int]] = deque([(start_y, start_x)])
    height, width = walkable.shape
    while queue:
        y, x = queue.popleft()
        if (x, y) == (finish_x, finish_y):
            break
        next_distance = int(distance[y, x]) + 1
        for dy in (-1, 0, 1):
            for dx in (-1, 0, 1):
                if not dx and not dy:
                    continue
                ny, nx = y + dy, x + dx
                if 0 <= ny < height and 0 <= nx < width and walkable[ny, nx] and distance[ny, nx] < 0:
                    distance[ny, nx] = next_distance
                    queue.append((ny, nx))
    if distance[finish_y, finish_x] < 0:
        raise ValueError("no connected green-to-red route in the raster")

    path = np.zeros_like(walkable, dtype=bool)
    y, x = finish_y, finish_x
    path[y, x] = True
    while (x, y) != (start_x, start_y):
        current_distance = distance[y, x]
        choices = []
        for dy in (-1, 0, 1):
            for dx in (-1, 0, 1):
                if not dx and not dy:
                    continue
                ny, nx = y + dy, x + dx
                if 0 <= ny < height and 0 <= nx < width and distance[ny, nx] == current_distance - 1:
                    choices.append((ny, nx))
        if not choices:
            raise ValueError("route backtrace is broken")
        y, x = choices[0]
        path[y, x] = True
    return path


def _distance_to_path(path: np.ndarray, center: tuple[float, float]) -> float:
    x, y = center
    radius = 30
    x0, x1 = max(0, round(x) - radius), min(path.shape[1], round(x) + radius + 1)
    y0, y1 = max(0, round(y) - radius), min(path.shape[0], round(y) + radius + 1)
    local_y, local_x = np.where(path[y0:y1, x0:x1])
    if len(local_x) == 0:
        return float("inf")
    dx = local_x + x0 - x
    dy = local_y + y0 - y
    return float(np.sqrt(np.square(dx) + np.square(dy)).min())


def _decision_uncached(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    green = _close_to(arr, GREEN, 75.0)
    red = _close_to(arr, RED, 75.0)
    start = _largest_centroid(green, "green terminal")
    finish = _largest_centroid(red, "red terminal")
    path = _route_pixels(arr, start, finish)

    marker_masks = [_close_to(arr, color, 58.0) for color in MARKERS]
    marker_mask = marker_masks[0] | marker_masks[1]
    marker_centers = [(x, y) for x, y, _ in _components(marker_mask, 150)]
    if len(marker_centers) != 5:
        raise ValueError(f"expected five ringed junctions, found {len(marker_centers)}")
    on_route = sum(_distance_to_path(path, center) <= 22.0 for center in marker_centers)
    return "even" if on_route % 2 == 0 else "odd"


def decision_from_image(image: Image.Image) -> str:
    """Recover only the parity decision from the final PNG."""

    key = hashlib.sha256(image.convert("RGB").tobytes()).hexdigest()
    if key not in _DECISION_CACHE:
        _DECISION_CACHE[key] = _decision_uncached(image)
        if len(_DECISION_CACHE) > 2048:
            _DECISION_CACHE.pop(next(iter(_DECISION_CACHE)))
    return _DECISION_CACHE[key]
