"""Renderer and analytic labels for the marked-junction parity world.

The picture is a small planar tree.  Five degree-three nodes are marked with
colored rings.  The green and red terminals are leaves of the same tree, so the
answer is obtained by tracing their unique graph path and taking the parity of
the marked junctions on it.
"""

from __future__ import annotations

from collections import deque
import math
import random

from PIL import Image, ImageColor, ImageDraw


RENDERER_VERSION = "marked-junction-parity-0.1.0"
WIDTH = 720
HEIGHT = 520
ROUTE_RGB = (37, 61, 93)
DISTRACTOR_RGB = (210, 216, 222)
JUNCTION_PALETTE = ("#D6922D", "#8859B5")


def _node_map(scene: dict) -> dict[str, dict]:
    return {node["id"]: node for node in scene["nodes"]}


def _graph_path(scene: dict) -> list[str]:
    """Return the unique node path from the green node to the red node."""

    nodes = scene["nodes"]
    edges = scene["edges"]
    node_map = {node["id"]: node for node in nodes}
    source = next(node["id"] for node in nodes if node["role"] == "source")
    target = next(node["id"] for node in nodes if node["role"] == "target")
    adjacency: dict[str, list[str]] = {node_id: [] for node_id in node_map}
    for left, right in edges:
        adjacency[left].append(right)
        adjacency[right].append(left)
    parent = {source: None}
    queue: deque[str] = deque([source])
    while queue:
        current = queue.popleft()
        if current == target:
            break
        for neighbor in adjacency[current]:
            if neighbor not in parent:
                parent[neighbor] = current
                queue.append(neighbor)
    if target not in parent:
        raise ValueError("terminals are disconnected")
    path: list[str] = []
    current: str | None = target
    while current is not None:
        path.append(current)
        current = parent[current]
    path.reverse()
    return path


def analytic_gold(scene: dict) -> str:
    """Answer from the visible graph description, without a stored answer bit."""

    path_ids = set(_graph_path(scene))
    marked_on_path = sum(
        node["id"] in path_ids and node["role"] == "junction" and node["ringed"]
        for node in scene["nodes"]
    )
    return "even" if marked_on_path % 2 == 0 else "odd"


def _point(node: dict) -> tuple[float, float]:
    return float(node["x"]), float(node["y"])


def _draw_scaled_line(draw: ImageDraw.ImageDraw, points: list[tuple[float, float]], width: int, scale: int, fill) -> None:
    draw.line(
        [(round(x * scale), round(y * scale)) for x, y in points],
        fill=fill,
        width=max(1, round(width * scale)),
        joint="curve",
    )


def _draw_scaled_ellipse(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    radius: float,
    scale: int,
    *,
    fill=None,
    outline=None,
    width: int = 1,
) -> None:
    x, y = center
    box = (
        round((x - radius) * scale),
        round((y - radius) * scale),
        round((x + radius) * scale),
        round((y + radius) * scale),
    )
    draw.ellipse(box, fill=fill, outline=outline, width=max(1, round(width * scale)))


def render(scene: dict) -> Image.Image:
    """Rasterize a scene.  No labels or metadata are embedded in the PNG."""

    scale = 3
    width = int(scene["width"])
    height = int(scene["height"])
    background = scene["background"]
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    route_color = ImageColor.getrgb(scene["route_color"])
    distractor_color = ImageColor.getrgb("#D2D8DE")

    for stroke in scene["distractors"]:
        _draw_scaled_line(draw, [tuple(point) for point in stroke], 3, scale, distractor_color)

    nodes = _node_map(scene)
    for left, right in scene["edges"]:
        _draw_scaled_line(draw, [_point(nodes[left]), _point(nodes[right])], int(scene["route_width"]), scale, route_color)

    for node in scene["nodes"]:
        if node["role"] == "junction":
            _draw_scaled_ellipse(
                draw,
                _point(node),
                float(scene["junction_radius"]),
                scale,
                outline=ImageColor.getrgb(scene["junction_color"]),
                width=int(scene["junction_width"]),
            )
        elif node["role"] in {"source", "target"}:
            _draw_scaled_ellipse(
                draw,
                _point(node),
                float(scene["terminal_radius"]),
                scale,
                fill=ImageColor.getrgb(node["color"]),
                outline=route_color,
                width=3,
            )
        else:
            _draw_scaled_ellipse(draw, _point(node), 4.5, scale, fill=route_color)

    if scale > 1:
        image = image.resize((width, height), Image.Resampling.LANCZOS)
    return image


def _transform_points(
    points: list[tuple[float, float]],
    *,
    width: int,
    height: int,
    angle: int,
    mirror: bool,
    scale_fraction: float,
    offset: tuple[float, float],
) -> list[tuple[float, float]]:
    cx, cy = 360.0, 260.0
    radians = math.radians(angle)
    cos_a, sin_a = math.cos(radians), math.sin(radians)
    raw: list[tuple[float, float]] = []
    for x, y in points:
        dx, dy = x - cx, y - cy
        if mirror:
            dx = -dx
        raw.append((dx * cos_a - dy * sin_a, dx * sin_a + dy * cos_a))
    min_x = min(x for x, _ in raw)
    max_x = max(x for x, _ in raw)
    min_y = min(y for _, y in raw)
    max_y = max(y for _, y in raw)
    fit = min((width - 96) / max(1.0, max_x - min_x), (height - 96) / max(1.0, max_y - min_y))
    fit *= scale_fraction
    center_x = (min_x + max_x) / 2.0
    center_y = (min_y + max_y) / 2.0
    return [
        (
            (x - center_x) * fit + width / 2.0 + offset[0],
            (y - center_y) * fit + height / 2.0 + offset[1],
        )
        for x, y in raw
    ]


def _make_topology(route_count: int, branch_sign: int, rng: random.Random) -> tuple[list[dict], list[list[str]], list[list[tuple[float, float]]]]:
    """Make a tree with ``route_count`` marked junctions on S--T path."""

    if route_count == 2:
        main_points = [(78.0, 268.0), (250.0, 276.0), (475.0, 262.0), (650.0, 270.0)]
    else:
        main_points = [
            (72.0, 268.0),
            (205.0, 278.0),
            (355.0, 258.0),
            (515.0, 281.0),
            (658.0, 268.0),
        ]
    off_count = 5 - route_count
    branch_y = main_points[1][1]
    if off_count == 3:
        off_points = [
            (175.0, branch_y + branch_sign * 118.0),
            (310.0, branch_y + branch_sign * 190.0),
            (485.0, branch_y + branch_sign * 158.0),
        ]
    else:
        off_points = [
            (175.0, branch_y + branch_sign * 122.0),
            (340.0, branch_y + branch_sign * 194.0),
        ]

    nodes: list[dict] = []
    nodes.append({"id": "S", "role": "source", "color": "#28A55F", "x": main_points[0][0], "y": main_points[0][1], "ringed": False})
    for index in range(route_count):
        x, y = main_points[index + 1]
        nodes.append({"id": f"J{index}", "role": "junction", "color": None, "x": x, "y": y, "ringed": True})
    nodes.append({"id": "T", "role": "target", "color": "#D84B4B", "x": main_points[-1][0], "y": main_points[-1][1], "ringed": False})
    for index, (x, y) in enumerate(off_points):
        nodes.append({"id": f"O{index}", "role": "junction", "color": None, "x": x, "y": y, "ringed": True})

    edges: list[list[str]] = []
    main_ids = ["S"] + [f"J{index}" for index in range(route_count)] + ["T"]
    edges.extend([[left, right] for left, right in zip(main_ids, main_ids[1:])])
    edges.append(["J0", "O0"])
    for index in range(off_count - 1):
        edges.append([f"O{index}", f"O{index + 1}"])

    leaf_index = 0
    # Each internal off-route junction gets one side leaf; the last gets two.
    for index, (x, y) in enumerate(off_points):
        leaf_positions: list[tuple[float, float]]
        if index < off_count - 1:
            if index == 0:
                leaf_positions = [(105.0, branch_y + branch_sign * 180.0)]
            else:
                leaf_positions = [(285.0, branch_y + branch_sign * 240.0)]
        else:
            if off_count == 3:
                leaf_positions = [
                    (410.0, branch_y + branch_sign * 246.0),
                    (590.0, branch_y + branch_sign * 210.0),
                ]
            else:
                leaf_positions = [
                    (275.0, branch_y + branch_sign * 242.0),
                    (455.0, branch_y + branch_sign * 246.0),
                ]
        for leaf_x, leaf_y in leaf_positions:
            leaf_id = f"L{leaf_index}"
            leaf_index += 1
            nodes.append({"id": leaf_id, "role": "leaf", "color": None, "x": leaf_x, "y": leaf_y, "ringed": False})
            edges.append([f"O{index}", leaf_id])

    # The non-first main junctions each have a downward/upward antenna, keeping
    # them degree three without putting another marked node on S--T.
    for index in range(1, route_count):
        x, y = main_points[index + 1]
        leaf_id = f"L{leaf_index}"
        leaf_index += 1
        nodes.append(
            {
                "id": leaf_id,
                "role": "leaf",
                "color": None,
                "x": x + rng.uniform(-14.0, 14.0),
                "y": y - branch_sign * rng.uniform(92.0, 116.0),
                "ringed": False,
            }
        )
        edges.append([f"J{index}", leaf_id])

    distractors = [
        [(92.0, 62.0), (180.0, 45.0), (252.0, 66.0)],
        [(506.0, 454.0), (594.0, 438.0), (674.0, 462.0)],
        [(568.0, 155.0), (650.0, 128.0), (688.0, 150.0)],
    ]
    return nodes, edges, distractors


def sample_scene(seed: int) -> dict:
    """Sample a balanced, comfortably separated scene from an integer seed."""

    rng = random.Random(int(seed))
    route_count = 2 + (int(seed) & 1)
    branch_sign = -1 if rng.randrange(2) == 0 else 1
    nodes, edges, distractors = _make_topology(route_count, branch_sign, rng)

    raw_points = [(float(node["x"]), float(node["y"])) for node in nodes]
    for stroke in distractors:
        raw_points.extend((float(x), float(y)) for x, y in stroke)
    angle = rng.choice((0, 90, 180, 270))
    mirror = bool(rng.randrange(2))
    scale_fraction = rng.uniform(0.93, 1.01)
    offset = (rng.uniform(-16.0, 16.0), rng.uniform(-12.0, 12.0))
    transformed = _transform_points(
        raw_points,
        width=WIDTH,
        height=HEIGHT,
        angle=angle,
        mirror=mirror,
        scale_fraction=scale_fraction,
        offset=offset,
    )
    for node, point in zip(nodes, transformed[: len(nodes)]):
        node["x"], node["y"] = round(point[0], 3), round(point[1], 3)
    cursor = len(nodes)
    transformed_strokes: list[list[list[float]]] = []
    for stroke in distractors:
        length = len(stroke)
        transformed_strokes.append(
            [[round(x, 3), round(y, 3)] for x, y in transformed[cursor : cursor + length]]
        )
        cursor += length

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": "white",
        "route_color": "#253D5D",
        "route_width": rng.choice((9, 10, 11)),
        "terminal_radius": round(rng.uniform(17.0, 21.0), 3),
        "junction_radius": round(rng.uniform(14.0, 18.0), 3),
        "junction_width": rng.choice((4, 5)),
        "junction_color": rng.choice(JUNCTION_PALETTE),
        "nodes": nodes,
        "edges": edges,
        "distractors": transformed_strokes,
    }


def _distance_point_segment(point: tuple[float, float], left: tuple[float, float], right: tuple[float, float]) -> float:
    px, py = point
    ax, ay = left
    bx, by = right
    dx, dy = bx - ax, by - ay
    denom = dx * dx + dy * dy
    if denom <= 1e-9:
        return math.dist(point, left)
    t = max(0.0, min(1.0, ((px - ax) * dx + (py - ay) * dy) / denom))
    return math.dist(point, (ax + t * dx, ay + t * dy))


def margin(scene: dict) -> float:
    """Return the smallest visible clearance separating route membership cases."""

    path = _graph_path(scene)
    path_set = set(path)
    nodes = _node_map(scene)
    path_segments = [
        (nodes[left], nodes[right])
        for left, right in scene["edges"]
        if left in path_set and right in path_set
    ]
    off_route = [
        node for node in scene["nodes"] if node["role"] == "junction" and node["id"] not in path_set
    ]
    if not off_route or not path_segments:
        return 0.0
    clearances = []
    for node in off_route:
        distance = min(
            _distance_point_segment(_point(node), _point(left), _point(right))
            for left, right in path_segments
        )
        clearances.append(distance - float(scene["junction_radius"]) - float(scene["route_width"]) / 2.0)
    return round(max(0.0, min(clearances)), 3)


def is_quarantined(scene: dict) -> bool:
    """Reject only scenes whose explicit ring/route clearance is too small."""

    try:
        return len(_graph_path(scene)) < 4 or margin(scene) < 28.0
    except (KeyError, StopIteration, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """There are no hidden, render-invariant aliases in this parameterization."""

    return []
