"""Independent PNG-only topology oracle for the paperclip loop."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "paperclip-loop-state-oracle-0.1.0"


def _foreground(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    if arr.ndim != 3 or arr.shape[2] != 3:
        raise ValueError("expected an RGB image")
    border = np.concatenate(
        (arr[0, :, :], arr[-1, :, :], arr[:, 0, :], arr[:, -1, :]), axis=0
    )
    background = np.median(border, axis=0)
    distance = np.sqrt(((arr - background) ** 2).sum(axis=2))
    mask = distance > 32.0
    if int(mask.sum()) < 500:
        raise ValueError("no substantial foreground component")
    return mask


def _components(mask: np.ndarray) -> list[np.ndarray]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y = int(y)
        x = int(x)
        if seen[y, x]:
            continue
        queue = [(y, x)]
        seen[y, x] = True
        pixels: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.pop()
            pixels.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dy or dx):
                        continue
                    ny = cy + dy
                    nx = cx + dx
                    if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        if len(pixels) >= 80:
            component = np.zeros_like(mask, dtype=bool)
            yy, xx = zip(*pixels)
            component[np.asarray(yy), np.asarray(xx)] = True
            components.append(component)
    return components


def _largest_component(mask: np.ndarray) -> np.ndarray:
    components = _components(mask)
    if not components:
        raise ValueError("foreground has no measurable component")
    return max(components, key=lambda item: int(item.sum()))


def _hole_areas(component: np.ndarray) -> list[int]:
    ys, xs = np.nonzero(component)
    if len(ys) == 0:
        return []
    pad = 2
    y0 = max(0, int(ys.min()) - pad)
    y1 = min(component.shape[0], int(ys.max()) + pad + 1)
    x0 = max(0, int(xs.min()) - pad)
    x1 = min(component.shape[1], int(xs.max()) + pad + 1)
    crop = component[y0:y1, x0:x1]
    empty = ~crop
    h, w = crop.shape
    outside = np.zeros_like(empty, dtype=bool)
    queue: deque[tuple[int, int]] = deque()
    for x in range(w):
        if empty[0, x] and not outside[0, x]:
            outside[0, x] = True
            queue.append((0, x))
        if empty[h - 1, x] and not outside[h - 1, x]:
            outside[h - 1, x] = True
            queue.append((h - 1, x))
    for y in range(h):
        if empty[y, 0] and not outside[y, 0]:
            outside[y, 0] = True
            queue.append((y, 0))
        if empty[y, w - 1] and not outside[y, w - 1]:
            outside[y, w - 1] = True
            queue.append((y, w - 1))
    while queue:
        y, x = queue.popleft()
        for dy, dx in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            ny = y + dy
            nx = x + dx
            if 0 <= ny < h and 0 <= nx < w and empty[ny, nx] and not outside[ny, nx]:
                outside[ny, nx] = True
                queue.append((ny, nx))
    holes = empty & ~outside
    hole_components = _components(holes)
    return sorted((int(item.sum()) for item in hole_components), reverse=True)


def decision_from_image(image: Image.Image) -> str:
    """Return ``open`` or ``closed`` from the raster alone."""

    component = _largest_component(_foreground(image))
    holes = _hole_areas(component)
    return "closed" if holes and holes[0] >= 1200 else "open"
