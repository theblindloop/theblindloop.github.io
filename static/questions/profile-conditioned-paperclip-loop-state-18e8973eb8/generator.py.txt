"""Deterministic renderer and analytic labels for the paperclip loop world."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

RENDERER_VERSION = "paperclip-loop-state-0.1.0"

INK_PALETTE: tuple[tuple[int, int, int], ...] = (
    (31, 49, 73),
    (47, 39, 78),
    (35, 78, 78),
    (86, 51, 49),
    (48, 61, 54),
)
BACKGROUND_PALETTE: tuple[tuple[int, int, int], ...] = (
    (250, 250, 248),
    (247, 249, 252),
    (253, 249, 243),
)
DEFAULT_SIZE = (512, 512)
QUARANTINE_MARGIN = 28.0


def sample_scene(seed: int) -> dict:
    """Sample one scene with state balanced by the generator's consecutive seeds."""

    seed = int(seed)
    # The nuisance RNG is independent of this parity so state is not tied to a
    # particular palette or geometry sequence in generated balanced runs.
    rng = random.Random((seed * 1_000_003) ^ 0x5EEDC0DE)
    state = "open" if seed % 2 else "closed"
    width, height = DEFAULT_SIZE
    center_x = rng.uniform(214.0, 298.0)
    center_y = rng.uniform(259.0, 287.0)
    loop_rx = rng.uniform(122.0, 145.0)
    loop_ry = rng.uniform(151.0, 178.0)
    stroke_width = rng.uniform(17.0, 22.0)
    # A small subset supplies a visible near-boundary open case for gallery and
    # quarantine coverage; its gap remains large enough to render unambiguously.
    if seed % 19 == 0:
        gap_angle = rng.uniform(0.31, 0.37)
    else:
        gap_angle = rng.uniform(0.72, 1.02)
    tail_angle = rng.uniform(-math.pi, math.pi)
    opening_angle = (tail_angle + math.pi + rng.uniform(-0.24, 0.24)) % (2.0 * math.pi)
    tail_depth = rng.uniform(0.23, 0.34)
    return {
        "state": state,
        "width": width,
        "height": height,
        "center_x": round(center_x, 5),
        "center_y": round(center_y, 5),
        "loop_rx": round(loop_rx, 5),
        "loop_ry": round(loop_ry, 5),
        "stroke_width": round(stroke_width, 5),
        "gap_angle": round(gap_angle, 6),
        "opening_angle": round(opening_angle, 6),
        "tail_angle": round(tail_angle, 6),
        "tail_depth": round(tail_depth, 6),
        "ink_color": list(rng.choice(INK_PALETTE)),
        "background_color": list(rng.choice(BACKGROUND_PALETTE)),
        "antialias": 4,
    }


def _loop_points(scene: dict, *, samples: int = 360) -> tuple[list[tuple[float, float]], bool]:
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    rx = float(scene["loop_rx"])
    ry = float(scene["loop_ry"])
    if scene["state"] == "closed":
        start = 0.0
        end = 2.0 * math.pi
        closed = True
    else:
        gap = float(scene["gap_angle"])
        opening = float(scene["opening_angle"])
        start = opening + gap / 2.0
        end = opening + 2.0 * math.pi - gap / 2.0
        closed = False
    points = []
    count = samples if closed else max(80, int(samples * (end - start) / (2.0 * math.pi)))
    for i in range(count + 1):
        angle = start + (end - start) * i / count
        points.append((cx + rx * math.cos(angle), cy + ry * math.sin(angle)))
    return points, closed


def _draw_round_polyline(
    draw: ImageDraw.ImageDraw,
    points: Iterable[tuple[float, float]],
    color: tuple[int, int, int],
    width: float,
    scale: int,
    *,
    closed: bool = False,
) -> None:
    raw = list(points)
    scaled = [(int(round(x * scale)), int(round(y * scale))) for x, y in raw]
    if closed:
        scaled.append(scaled[0])
    line_width = max(3, int(round(width * scale)))
    draw.line(scaled, fill=color, width=line_width, joint="curve")
    radius = line_width / 2.0
    caps = [scaled[0]] if closed else [scaled[0], scaled[-1]]
    for x, y in caps:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)


def _tail_points(scene: dict, samples: int = 80) -> list[tuple[float, float]]:
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    rx = float(scene["loop_rx"])
    ry = float(scene["loop_ry"])
    angle = float(scene["tail_angle"])
    depth = float(scene["tail_depth"])
    points = []
    for i in range(samples + 1):
        t = i / samples
        radial = 1.0 - (1.0 - depth) * t
        bend = 0.18 * math.sin(math.pi * t) + 0.08 * t
        a = angle + bend
        points.append((cx + rx * radial * math.cos(a), cy + ry * radial * math.sin(a)))
    return points


def render(scene: dict) -> Image.Image:
    """Render only from the public scene dictionary."""

    width = int(scene["width"])
    height = int(scene["height"])
    scale = int(scene["antialias"])
    if scale < 2:
        raise ValueError("antialias scale must be at least 2")
    background = tuple(int(v) for v in scene["background_color"])
    ink = tuple(int(v) for v in scene["ink_color"])
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    points, closed = _loop_points(scene)
    stroke = float(scene["stroke_width"])
    _draw_round_polyline(draw, points, ink, stroke, scale, closed=closed)
    # A short attached inner tail makes the loop read as a paperclip while not
    # closing or opening the answer-bearing outer boundary.
    _draw_round_polyline(draw, _tail_points(scene), ink, stroke * 0.82, scale)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """Closed-form label: the rendered state is the only answer-bearing field."""

    return str(scene["state"])


def _gap_clearance(scene: dict) -> float:
    rx = float(scene["loop_rx"])
    ry = float(scene["loop_ry"])
    gap = float(scene["gap_angle"])
    # Chord between the two opening endpoints on the ellipse.
    half = gap / 2.0
    chord = math.hypot(2.0 * rx * math.sin(half), 2.0 * ry * math.sin(half)) / math.sqrt(2.0)
    return max(0.0, chord - 1.65 * float(scene["stroke_width"]))


def margin(scene: dict) -> float:
    """Visible topological clearance in pixels, rounded for stable manifests."""

    if scene["state"] == "open":
        value = _gap_clearance(scene)
    else:
        value = min(float(scene["loop_rx"]), float(scene["loop_ry"])) - 1.8 * float(
            scene["stroke_width"]
        )
    return round(max(0.0, value), 3)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No pixel-identical latent aliases are claimed for this renderer."""

    return []
