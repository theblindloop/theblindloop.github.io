"""Deterministic sampler, renderer, and analytic arm for landmark atlases."""

from __future__ import annotations

import itertools
import math
import random

from PIL import Image, ImageDraw

CANVAS_WIDTH = 1024
CANVAS_HEIGHT = 1024
BACKGROUND = (255, 255, 255)
LANDMARK_RGB = (35, 38, 43)
REFERENCE_RGB = (105, 119, 132)
TAG_RGB = {
    "amber": (242, 177, 40),
    "cyan": (25, 174, 190),
    "violet": (146, 82, 184),
}
TAG_NAMES = tuple(TAG_RGB)
SHAPES = ("circle", "square", "triangle", "plus")

# These four centers keep the complete evidence world inside a medium-size canvas
# region while leaving the four decisive clusters mutually disconnected.
CLUSTER_CENTERS = ((300, 320), (703, 320), (300, 703), (703, 703))
TAG_DY = -70

BASE_PATTERNS = (
    ((-18, -15), (15, -19), (-13, 19), (20, 14)),
    ((-19, -12), (13, -18), (-15, 18), (19, 16)),
    ((-17, -19), (18, -12), (-19, 14), (15, 20)),
    ((-20, -14), (12, -19), (-14, 20), (19, 13)),
)


def _rotate(point: tuple[float, float], turns: int) -> tuple[float, float]:
    x, y = point
    for _ in range(turns % 4):
        x, y = -y, x
    return x, y


def _centered(layout: dict[str, list[int] | tuple[int, int]]) -> dict[str, tuple[float, float]]:
    xs = [float(layout[shape][0]) for shape in SHAPES]
    ys = [float(layout[shape][1]) for shape in SHAPES]
    mean_x = sum(xs) / len(xs)
    mean_y = sum(ys) / len(ys)
    return {
        shape: (float(layout[shape][0]) - mean_x, float(layout[shape][1]) - mean_y)
        for shape in SHAPES
    }


def _layout_distance(reference: dict, candidate: dict) -> float:
    ref = _centered(reference)
    cand = _centered(candidate)
    scores = []
    for turns in range(4):
        score = 0.0
        for shape in SHAPES:
            rx, ry = _rotate(ref[shape], turns)
            cx, cy = cand[shape]
            score += math.hypot(rx - cx, ry - cy)
        scores.append(score)
    return min(scores)


def sample_scene(seed: int) -> dict:
    """Sample balanced shape assignments, rotations, slots, and answer tags."""

    seed = int(seed)
    rng = random.Random(seed)
    coordinates = list(rng.choice(BASE_PATTERNS))
    rng.shuffle(coordinates)
    reference = {shape: list(point) for shape, point in zip(SHAPES, coordinates, strict=True)}
    turns = rng.randrange(4)
    correct = {
        shape: [int(value) for value in _rotate(tuple(reference[shape]), turns)]
        for shape in SHAPES
    }

    target_coordinates = [tuple(correct[shape]) for shape in SHAPES]
    distractors = []
    for permutation in itertools.permutations(target_coordinates):
        layout = {shape: list(point) for shape, point in zip(SHAPES, permutation, strict=True)}
        if _layout_distance(reference, layout) >= 12.0:
            distractors.append(layout)
    rng.shuffle(distractors)
    if len(distractors) < 2:
        raise RuntimeError("pattern does not admit two separated distractors")

    correct_slot = seed % 3
    candidates = distractors[:2]
    candidates.insert(correct_slot, correct)
    answer_tag = TAG_NAMES[(seed // 3) % 3]
    remaining_tags = [tag for tag in TAG_NAMES if tag != answer_tag]
    rng.shuffle(remaining_tags)
    candidate_tags = list(remaining_tags)
    candidate_tags.insert(correct_slot, answer_tag)

    return {
        "canvas_width": CANVAS_WIDTH,
        "canvas_height": CANVAS_HEIGHT,
        "offset_x": -rng.randrange(2),
        "offset_y": -rng.randrange(2),
        "reference_positions": reference,
        "candidate_positions": candidates,
        "candidate_tags": candidate_tags,
    }


def _draw_shape(draw: ImageDraw.ImageDraw, shape: str, center: tuple[int, int]) -> None:
    cx, cy = center
    if shape == "circle":
        draw.ellipse((cx - 9, cy - 9, cx + 9, cy + 9), fill=LANDMARK_RGB)
    elif shape == "square":
        draw.rectangle((cx - 8, cy - 8, cx + 8, cy + 8), fill=LANDMARK_RGB)
    elif shape == "triangle":
        draw.polygon(((cx, cy - 10), (cx - 10, cy + 10), (cx + 10, cy + 10)), fill=LANDMARK_RGB)
    elif shape == "plus":
        draw.rectangle((cx - 4, cy - 11, cx + 4, cy + 11), fill=LANDMARK_RGB)
        draw.rectangle((cx - 11, cy - 4, cx + 11, cy + 4), fill=LANDMARK_RGB)
    else:
        raise ValueError(f"unknown shape: {shape}")


def _draw_cluster(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    layout: dict,
    tag_color: tuple[int, int, int],
) -> None:
    cx, cy = center
    tag_y = cy + TAG_DY
    draw.rectangle((cx - 15, tag_y - 8, cx + 15, tag_y + 8), fill=tag_color)
    for shape in SHAPES:
        dx, dy = (int(value) for value in layout[shape])
        _draw_shape(draw, shape, (cx + dx, cy + dy))


def render(scene: dict) -> Image.Image:
    width = int(scene["canvas_width"])
    height = int(scene["canvas_height"])
    offset_x = int(scene["offset_x"])
    offset_y = int(scene["offset_y"])
    reference = scene["reference_positions"]
    candidates = scene["candidate_positions"]
    tags = [str(tag) for tag in scene["candidate_tags"]]
    if len(candidates) != 3 or len(tags) != 3:
        raise ValueError("scene must contain exactly three candidates and tags")
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    centers = [(x + offset_x, y + offset_y) for x, y in CLUSTER_CENTERS]
    _draw_cluster(draw, centers[0], reference, REFERENCE_RGB)
    for center, layout, tag in zip(centers[1:], candidates, tags, strict=True):
        _draw_cluster(draw, center, layout, TAG_RGB[tag])
    return image


def _candidate_distances(scene: dict) -> list[float]:
    return [
        _layout_distance(scene["reference_positions"], candidate)
        for candidate in scene["candidate_positions"]
    ]


def analytic_gold(scene: dict) -> str:
    distances = _candidate_distances(scene)
    matches = [index for index, value in enumerate(distances) if value < 1e-9]
    if len(matches) != 1:
        raise ValueError("scene does not contain exactly one quarter-turn match")
    return str(scene["candidate_tags"][matches[0]])


def margin(scene: dict) -> float:
    distances = sorted(_candidate_distances(scene))
    return float(distances[1] - distances[0]) if len(distances) >= 2 else 0.0


def is_quarantined(scene: dict) -> bool:
    distances = _candidate_distances(scene)
    return sum(value < 1e-9 for value in distances) != 1 or margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
