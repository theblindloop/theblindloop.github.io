"""Independent pixel-only inverse for the state-histogram world."""

from __future__ import annotations

from PIL import Image


LABELS = ("A", "B", "C", "D")
PANEL_COUNT = 6
TILE_COUNT = 4
STATE_COUNT = 3
TARGET_COUNTS = (3, 2, 1)
CANVAS = (1776, 800)
STATE_COLORS = ((52, 116, 205), (224, 108, 59), (52, 158, 94))
PANEL_X0 = 24
PANEL_TOP = 160
PANEL_WIDTH = 276
PANEL_GAP = 10
KEY_Y = 91
KEY_XS = (270, 490, 710)
ROW_TOP = 265
ROW_STEP = 103
MARKER_X_OFFSET = 190
MARKER_Y_OFFSET = 38


def _palette(frame: Image.Image) -> dict[tuple[int, int, int], int]:
    mapping: dict[tuple[int, int, int], int] = {}
    for state, x in enumerate(KEY_XS):
        color = frame.getpixel((x, KEY_Y))
        if color not in STATE_COLORS:
            raise ValueError("state key is missing or unreadable")
        if color in mapping:
            raise ValueError("state key reuses one color")
        mapping[color] = state
    return mapping


def _decode_states(frame: Image.Image, palette: dict[tuple[int, int, int], int]) -> list[list[int]]:
    table: list[list[int]] = []
    for panel in range(PANEL_COUNT):
        panel_x = PANEL_X0 + panel * (PANEL_WIDTH + PANEL_GAP)
        row: list[int] = []
        for tile in range(TILE_COUNT):
            x = panel_x + MARKER_X_OFFSET
            y = ROW_TOP + tile * ROW_STEP + MARKER_Y_OFFSET
            color = frame.getpixel((x, y))
            if color not in palette:
                raise ValueError(f"state marker P{panel + 1}/{LABELS[tile]} is unreadable")
            row.append(palette[color])
        table.append(row)
    return table


def decision_from_image(image: Image.Image) -> str:
    """Decode the visible key and twenty-four state markers from one PNG."""

    frame = image.convert("RGB")
    if frame.size != CANVAS:
        raise ValueError(f"unexpected raster size {frame.size}")
    palette = _palette(frame)
    states = _decode_states(frame, palette)
    scores: list[int] = []
    for tile in range(TILE_COUNT):
        counts = [0] * STATE_COUNT
        for panel in range(PANEL_COUNT):
            state = states[panel][tile]
            counts[state] += 1
        scores.append(sum(abs(counts[index] - TARGET_COUNTS[index]) for index in range(STATE_COUNT)))
    best = min(scores)
    winners = [index for index, score in enumerate(scores) if score == best]
    if len(winners) != 1:
        raise ValueError("image does not expose one unique histogram winner")
    return LABELS[winners[0]]
