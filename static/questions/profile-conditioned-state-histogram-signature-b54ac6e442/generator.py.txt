"""Deterministic renderer and analytic side for the state-histogram world."""

from __future__ import annotations

import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont


LABELS = ("A", "B", "C", "D")
PANEL_COUNT = 6
TILE_COUNT = 4
STATE_COUNT = 3
TARGET_COUNTS = (3, 2, 1)
CANVAS = (1776, 800)

BACKGROUND = (247, 249, 252)
CARD = (255, 255, 255)
INK = (35, 43, 56)
MUTED = (91, 103, 116)
OUTLINE = (166, 178, 191)
STATE_COLORS = ((52, 116, 205), (224, 108, 59), (52, 158, 94))
STATE_NAMES = ("LOW", "MID", "HIGH")
PANEL_TINTS = (
    (239, 245, 251),
    (247, 241, 235),
    (240, 248, 242),
    (246, 241, 249),
)

PANEL_X0 = 24
PANEL_TOP = 160
PANEL_WIDTH = 276
PANEL_HEIGHT = 600
PANEL_GAP = 10
KEY_Y = 91
KEY_XS = (270, 490, 710)
ROW_TOP = 265
ROW_STEP = 103
CELL_LEFT_INSET = 18
CELL_WIDTH = PANEL_WIDTH - 36
CELL_HEIGHT = 76
MARKER_X_OFFSET = 190
MARKER_Y_OFFSET = 38

_FONT_PATHS = (
    "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
    "/usr/share/fonts/dejavu/DejaVuSans.ttf",
)


def _font(size: int):
    for candidate in _FONT_PATHS:
        if Path(candidate).is_file():
            return ImageFont.truetype(candidate, size=size)
    return ImageFont.load_default()


def _centered_text(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font, fill):
    box = draw.textbbox((0, 0), text, font=font)
    width = box[2] - box[0]
    height = box[3] - box[1]
    draw.text((xy[0] - width / 2, xy[1] - height / 2 - box[1]), text, font=font, fill=fill)


def _state_color(scene: dict, state: int) -> tuple[int, int, int]:
    order = scene["state_palette_order"]
    return STATE_COLORS[int(order[int(state)])]


def _histogram_scores(scene: dict) -> list[int]:
    states = scene["states"]
    if len(states) != PANEL_COUNT or any(len(panel) != TILE_COUNT for panel in states):
        raise ValueError("states must be a six by four table")
    scores: list[int] = []
    for tile in range(TILE_COUNT):
        counts = [0] * STATE_COUNT
        for panel in range(PANEL_COUNT):
            value = int(states[panel][tile])
            if value not in range(STATE_COUNT):
                raise ValueError("state values must be LOW=0, MID=1, or HIGH=2")
            counts[value] += 1
        scores.append(sum(abs(counts[index] - TARGET_COUNTS[index]) for index in range(STATE_COUNT)))
    return scores


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene with exactly one 3-2-1 histogram winner."""

    seed = int(seed)
    logic_rng = random.Random(seed * 1000003 + 17)
    style_rng = random.Random((seed + 104729) * 1000033 + 31)
    winner = seed % TILE_COUNT
    states = [[0 for _ in range(TILE_COUNT)] for _ in range(PANEL_COUNT)]
    target: list[int] = []
    for state, count in enumerate(TARGET_COUNTS):
        target.extend([state] * count)
    logic_rng.shuffle(target)
    for panel, state in enumerate(target):
        states[panel][winner] = state

    for tile in range(TILE_COUNT):
        if tile == winner:
            continue
        for _ in range(1000):
            trajectory = [logic_rng.randrange(STATE_COUNT) for _ in range(PANEL_COUNT)]
            counts = [trajectory.count(state) for state in range(STATE_COUNT)]
            score = sum(abs(counts[index] - TARGET_COUNTS[index]) for index in range(STATE_COUNT))
            if score >= 2:
                break
        else:
            raise RuntimeError("could not sample a nonmatching histogram")
        for panel, state in enumerate(trajectory):
            states[panel][tile] = state

    state_palette_order = list(range(STATE_COUNT))
    style_rng.shuffle(state_palette_order)
    marker_radius = 5 if seed % 23 == 0 else style_rng.choice((12, 14, 16, 18))
    scene = {
        "states": states,
        "state_palette_order": state_palette_order,
        "marker_radius": marker_radius,
        "panel_tint": style_rng.randrange(len(PANEL_TINTS)),
    }
    if analytic_gold(scene) != LABELS[winner]:
        raise AssertionError("sampler failed to create the requested unique winner")
    return scene


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS, BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(25)
    section_font = _font(16)
    label_font = _font(22)
    small_font = _font(13)
    state_font = _font(14)

    draw.text((24, 20), "STATE HISTOGRAM CHECKPOINTS", font=title_font, fill=INK)
    draw.text(
        (24, 57),
        "Track each labeled tile across the six numbered panels and count its keyed states.",
        font=small_font,
        fill=MUTED,
    )
    draw.text((24, 78), "VISIBLE STATE KEY", font=section_font, fill=INK)
    for state, cx in enumerate(KEY_XS):
        draw.ellipse(
            (cx - 14, KEY_Y - 14, cx + 14, KEY_Y + 14),
            fill=_state_color(scene, state),
            outline=INK,
            width=1,
        )
        _centered_text(draw, (cx + 67, KEY_Y), f"{state}: {STATE_NAMES[state]}", state_font, INK)

    tint = PANEL_TINTS[int(scene["panel_tint"])]
    marker_radius = int(scene["marker_radius"])
    for panel in range(PANEL_COUNT):
        panel_x = PANEL_X0 + panel * (PANEL_WIDTH + PANEL_GAP)
        draw.rounded_rectangle(
            (panel_x, PANEL_TOP, panel_x + PANEL_WIDTH, PANEL_TOP + PANEL_HEIGHT),
            radius=12,
            fill=tint,
            outline=OUTLINE,
            width=2,
        )
        draw.text((panel_x + 16, PANEL_TOP + 14), f"P{panel + 1}", font=section_font, fill=INK)
        draw.line(
            (panel_x + 16, PANEL_TOP + 53, panel_x + PANEL_WIDTH - 16, PANEL_TOP + 53),
            fill=OUTLINE,
            width=1,
        )
        for tile, label in enumerate(LABELS):
            top = ROW_TOP + tile * ROW_STEP
            left = panel_x + CELL_LEFT_INSET
            state = int(scene["states"][panel][tile])
            draw.rounded_rectangle(
                (left, top, left + CELL_WIDTH, top + CELL_HEIGHT),
                radius=8,
                fill=CARD,
                outline=OUTLINE,
                width=1,
            )
            _centered_text(draw, (left + 42, top + CELL_HEIGHT / 2), label, label_font, INK)
            draw.line((left + 70, top + 13, left + 70, top + CELL_HEIGHT - 13), fill=OUTLINE, width=1)
            marker_x = panel_x + MARKER_X_OFFSET
            marker_y = top + MARKER_Y_OFFSET
            draw.ellipse(
                (marker_x - marker_radius, marker_y - marker_radius,
                 marker_x + marker_radius, marker_y + marker_radius),
                fill=_state_color(scene, state),
                outline=INK,
                width=1,
            )
    return image


def analytic_gold(scene: dict) -> str:
    scores = _histogram_scores(scene)
    best = min(scores)
    winners = [index for index, score in enumerate(scores) if score == best]
    if len(winners) != 1:
        raise ValueError("scene does not have a unique histogram winner")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    scores = sorted(_histogram_scores(scene))
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    scores = _histogram_scores(scene)
    return int(scene["marker_radius"]) < 8 or scores.count(min(scores)) != 1 or margin(scene) <= 0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases are claimed; all label inputs are visibly rendered."""

    return []
