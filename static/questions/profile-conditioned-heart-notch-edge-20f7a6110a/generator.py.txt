"""Deterministic filled-heart sampler and renderer."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

CANVAS = 512
SS = 4
EDGES = ("top", "right", "bottom", "left")
BACKGROUNDS = ("#f2eee7", "#e9f1f2", "#f2ecf3", "#edf2e8", "#f4efe1")
HEARTS = ("#d33f55", "#2f78c4", "#2f9a78", "#a24fb0", "#d87524")
BORDERS = ("#2b2830", "#26313a", "#332823", "#25332d")


def sample_scene(seed: int) -> dict:
    rng = random.Random((int(seed) ^ 0xA17E5C39) & 0xFFFFFFFFFFFFFFFF)
    return {
        "notch_edge": EDGES[int(seed) % 4],
        "center_x": rng.randint(235, 277),
        "center_y": rng.randint(235, 277),
        "scale": rng.randrange(102, 139, 4),
        "heart_color": rng.choice(HEARTS),
        "background_color": rng.choice(BACKGROUNDS),
        "border_color": rng.choice(BORDERS),
        "border_width": rng.randint(3, 6),
    }


def analytic_gold(scene: dict) -> str:
    edge = scene["notch_edge"]
    if edge not in EDGES:
        raise ValueError(f"invalid notch edge {edge!r}")
    return edge


def margin(scene: dict) -> float:
    """One-sided analytic notch prominence in output pixels."""
    return round(float(scene["scale"]) * 0.34, 3)


def is_quarantined(scene: dict) -> bool:
    scale = int(scene["scale"])
    cx, cy = int(scene["center_x"]), int(scene["center_y"])
    return scale < 96 or min(cx, cy, CANVAS - cx, CANVAS - cy) < scale + 12


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def rotate_counterfactual(scene: dict, quarter_turns: int = 1) -> dict:
    twin = deepcopy(scene)
    index = EDGES.index(analytic_gold(scene))
    twin["notch_edge"] = EDGES[(index + int(quarter_turns)) % 4]
    return twin


def _base_heart(scale: float) -> list[tuple[float, float]]:
    """Classic smooth heart, centered and normalized to a roughly square extent."""
    points = []
    for index in range(320):
        t = 2.0 * math.pi * index / 320.0
        x = 16.0 * math.sin(t) ** 3
        y = 13.0 * math.cos(t) - 5.0 * math.cos(2.0 * t) - 2.0 * math.cos(3.0 * t) - math.cos(4.0 * t)
        # Formula y spans approximately [-17, 11.9]. Center and invert for image coordinates.
        points.append((scale * x / 16.0, -scale * (y + 2.55) / 14.45))
    return points


def _oriented_points(scene: dict) -> list[tuple[int, int]]:
    edge = analytic_gold(scene)
    turns = EDGES.index(edge)
    angle = turns * math.pi / 2.0
    cos_a, sin_a = math.cos(angle), math.sin(angle)
    cx, cy = float(scene["center_x"]), float(scene["center_y"])
    result = []
    for x, y in _base_heart(float(scene["scale"])):
        rx = x * cos_a - y * sin_a
        ry = x * sin_a + y * cos_a
        result.append((round((cx + rx) * SS), round((cy + ry) * SS)))
    return result


def render(scene: dict) -> Image.Image:
    """Render one large familiar silhouette; all nuisance fields visibly contribute."""
    analytic_gold(scene)
    image = Image.new("RGB", (CANVAS * SS, CANVAS * SS), scene["background_color"])
    draw = ImageDraw.Draw(image)
    points = _oriented_points(scene)
    draw.polygon(points, fill=scene["heart_color"])
    draw.line(points + [points[0]], fill=scene["border_color"], width=int(scene["border_width"]) * SS, joint="curve")
    return image.resize((CANVAS, CANVAS), Image.Resampling.LANCZOS)
