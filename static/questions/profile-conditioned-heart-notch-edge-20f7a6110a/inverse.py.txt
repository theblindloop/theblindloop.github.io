"""Independent PNG-only inverse arm for the heart's inward notch orientation."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _largest_component(mask: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    best: list[tuple[int, int]] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        current: list[tuple[int, int]] = []
        while queue:
            yy, xx = queue.popleft()
            current.append((yy, xx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = yy + dy, xx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(current) > len(best):
            best = current
    if len(best) < 2500:
        raise ValueError("no large saturated silhouette detected")
    result = np.zeros_like(mask, dtype=bool)
    yy, xx = zip(*best)
    result[np.asarray(yy), np.asarray(xx)] = True
    return result


def _median_first(values: list[int]) -> float:
    if not values:
        raise ValueError("silhouette centerline evidence is missing")
    return float(np.median(np.asarray(values, dtype=np.float64)))


def measure_notch(image: Image.Image) -> dict:
    """Rank edge-facing indentation depths from the raster alone."""
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    if arr.ndim != 3 or min(arr.shape[:2]) < 96:
        raise ValueError("image is too small")
    colorfulness = arr.max(axis=2) - arr.min(axis=2)
    saturated = (colorfulness > 62) & (arr.max(axis=2) > 105)
    mask = _largest_component(saturated)
    ys, xs = np.nonzero(mask)
    x0, x1, y0, y1 = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    width, height = x1 - x0 + 1, y1 - y0 + 1
    if min(width, height) < 120 or max(width, height) > 340:
        raise ValueError("silhouette extent is implausible")

    # Use a narrow central strip. A heart's unique inward notch lets background
    # penetrate much farther from its facing edge than from the other edges.
    half_x = max(4, round(width * 0.035))
    half_y = max(4, round(height * 0.035))
    mid_x = int(round(float(np.median(xs))))
    mid_y = int(round(float(np.median(ys))))
    top_values: list[int] = []
    bottom_values: list[int] = []
    for x in range(max(x0, mid_x - half_x), min(x1, mid_x + half_x) + 1):
        found = np.flatnonzero(mask[y0 : y1 + 1, x])
        if found.size:
            top_values.append(int(found[0]))
            bottom_values.append(int(height - 1 - found[-1]))
    left_values: list[int] = []
    right_values: list[int] = []
    for y in range(max(y0, mid_y - half_y), min(y1, mid_y + half_y) + 1):
        found = np.flatnonzero(mask[y, x0 : x1 + 1])
        if found.size:
            left_values.append(int(found[0]))
            right_values.append(int(width - 1 - found[-1]))
    depths = {
        "top": _median_first(top_values),
        "right": _median_first(right_values),
        "bottom": _median_first(bottom_values),
        "left": _median_first(left_values),
    }
    ranked = sorted(depths.items(), key=lambda item: item[1], reverse=True)
    margin = float(ranked[0][1] - ranked[1][1])
    return {
        "depths": depths,
        "winner": ranked[0][0],
        "runner_up_margin": margin,
        "bbox": [x0, y0, x1, y1],
        "silhouette_pixels": int(mask.sum()),
    }


def decision_from_image(image: Image.Image) -> str:
    measurement = measure_notch(image)
    if measurement["runner_up_margin"] < 12.0:
        raise ValueError("notch orientation is absent or ambiguous")
    return str(measurement["winner"])
