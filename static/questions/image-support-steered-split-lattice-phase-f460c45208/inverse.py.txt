"""Independent image-only reconstruction of two dot-lattice phases."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        stack = [(sy, sx)]
        seen[sy, sx] = True
        points = []
        while stack:
            y, x = stack.pop()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= 20:
            output.append(np.asarray(points, dtype=np.float64))
    return output


def _levels(values: np.ndarray) -> np.ndarray:
    values = np.sort(np.asarray(values, dtype=np.float64))
    groups: list[list[float]] = [[float(values[0])]]
    for value in values[1:]:
        if float(value) - groups[-1][-1] > 3.0:
            groups.append([float(value)])
        else:
            groups[-1].append(float(value))
    return np.asarray([sum(group) / len(group) for group in groups])


def _period_and_center(points: np.ndarray) -> tuple[float, float, float]:
    xs = _levels(points[:, 0])
    ys = _levels(points[:, 1])
    if len(xs) != 4 or len(ys) != 4:
        raise ValueError("each patch must expose exactly four rows and four columns")
    gaps = np.concatenate((np.diff(xs), np.diff(ys)))
    period = float(np.median(gaps))
    if period < 18.0 or period > 35.0 or float(np.max(np.abs(gaps - period))) > 1.5:
        raise ValueError("dot rows and columns do not define a clean square lattice")
    return period, float(points[:, 0].mean()), float(points[:, 1].mean())


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    # All public ink palettes are saturated blue, while the card, border, and
    # background are nearly neutral. This rule is inferred from pixels only.
    blue = (array[:, :, 2] - array[:, :, 0] > 55) & (array[:, :, 2] - array[:, :, 1] > 24)
    components = _components(blue)
    if len(components) != 32:
        raise ValueError(f"expected 32 complete dot components, found {len(components)}")
    centers = np.asarray(
        [(float(component[:, 1].mean()), float(component[:, 0].mean())) for component in components]
    )
    order = np.argsort(centers[:, 0])
    sorted_x = centers[order, 0]
    gaps = np.diff(sorted_x)
    split = int(np.argmax(gaps)) + 1
    if split != 16 or float(gaps[split - 1]) < 80.0:
        raise ValueError("the two separated 16-dot patches cannot be recovered")
    left = centers[order[:split]]
    right = centers[order[split:]]
    p1, x1, y1 = _period_and_center(left)
    p2, x2, y2 = _period_and_center(right)
    period = (p1 + p2) / 2.0
    if abs(p1 - p2) > 1.0:
        raise ValueError("patch periods disagree")

    def residual(distance: float) -> float:
        return abs(distance - round(distance / period) * period)

    mismatch = max(residual(x2 - x1), residual(y2 - y1))
    return "yes" if mismatch <= period / 4.0 else "no"
