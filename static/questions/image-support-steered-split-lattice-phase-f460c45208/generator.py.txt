"""Deterministic latent sampler and renderer for split lattice phase."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw


INKS = ("#185bb4", "#087bad", "#3152ad", "#1768a8")
BACKGROUNDS = ("#ffffff", "#fbfcff", "#fffdf8")
CARDS = ("#f5f1e7", "#f2f4ed", "#f6f0f0")
QUESTION = (
    "The two separated 4 by 4 blue-dot patches each have equal horizontal and "
    "vertical spacing. Imagine extending their rows and columns through the blank "
    "gap. Do all 32 dots land on one common square lattice? Answer yes or no."
)


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    period = rng.randint(25, 27)
    mode = int(seed) % 6
    # Far classes dominate; modes 4/5 deliberately expose both sides near the
    # visible quarter-period boundary and are quarantined.
    mismatch_fraction = {0: 0.0, 1: 0.45, 2: 0.0, 3: 0.43, 4: 0.20, 5: 0.30}[mode]
    axis = rng.choice(("x", "y"))
    sign = rng.choice((-1.0, 1.0))
    common_x = rng.uniform(-0.18, 0.18) * period
    common_y = rng.uniform(-0.18, 0.18) * period
    delta = sign * mismatch_fraction * period
    left_x = common_x - (delta / 2.0 if axis == "x" else 0.0)
    right_x = common_x + (delta / 2.0 if axis == "x" else 0.0)
    left_y = common_y - (delta / 2.0 if axis == "y" else 0.0)
    right_y = common_y + (delta / 2.0 if axis == "y" else 0.0)
    return {
        "width": 640,
        "height": 640,
        "background": rng.choice(BACKGROUNDS),
        "ink": rng.choice(INKS),
        "card": rng.choice(CARDS),
        "period": period,
        "radius": rng.randint(5, 6),
        "center_x": 320.0,
        "center_y": 320.0,
        "global_dx": rng.randint(-6, 6),
        "global_dy": rng.randint(-14, 14),
        "left_offset_x": round(left_x, 5),
        "left_offset_y": round(left_y, 5),
        "right_offset_x": round(right_x, 5),
        "right_offset_y": round(right_y, 5),
    }


def _phase_residual(scene: dict[str, Any]) -> float:
    period = float(scene["period"])
    dx = float(scene["right_offset_x"]) - float(scene["left_offset_x"])
    dy = float(scene["right_offset_y"]) - float(scene["left_offset_y"])
    rx = abs(dx - round(dx / period) * period)
    ry = abs(dy - round(dy / period) * period)
    return max(rx, ry)


def analytic_gold(scene: dict[str, Any]) -> str:
    return "yes" if _phase_residual(scene) <= float(scene["period"]) / 4.0 else "no"


def margin(scene: dict[str, Any]) -> float:
    return abs(_phase_residual(scene) - float(scene["period"]) / 4.0)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) <= 2.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []


def render(scene: dict[str, Any]) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    background = str(scene["background"])
    ink = str(scene["ink"])
    card = str(scene["card"])
    period = float(scene["period"])
    radius = float(scene["radius"])
    center_x = float(scene["center_x"]) + float(scene["global_dx"])
    center_y = float(scene["center_y"]) + float(scene["global_dy"])
    left_offset_x = float(scene["left_offset_x"])
    left_offset_y = float(scene["left_offset_y"])
    right_offset_x = float(scene["right_offset_x"])
    right_offset_y = float(scene["right_offset_y"])

    scale = 4
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    card_box = (72 * scale, 150 * scale, (width - 72) * scale, (height - 150) * scale)
    draw.rounded_rectangle(
        card_box,
        radius=18 * scale,
        fill=card,
        outline="#c8c5bd",
        width=2 * scale,
    )

    centers = (
        (center_x - 6.0 * period + left_offset_x, center_y + left_offset_y),
        (center_x + 6.0 * period + right_offset_x, center_y + right_offset_y),
    )
    for patch_x, patch_y in centers:
        for row in range(4):
            for col in range(4):
                x = patch_x + (col - 1.5) * period
                y = patch_y + (row - 1.5) * period
                draw.ellipse(
                    (
                        (x - radius) * scale,
                        (y - radius) * scale,
                        (x + radius) * scale,
                        (y + radius) * scale,
                    ),
                    fill=ink,
                )
    return image.resize((width, height), Image.Resampling.LANCZOS)
