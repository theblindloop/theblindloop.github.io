"""Renderer and analytic arm for the four-card mosaic world.

The visible carrier is four disconnected perspective cards.  Each card carries
one quadrant of a 16 by 16 binary symbol; the question asks for four local
rectifications followed by a compass-position mosaic assembly.
"""

from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image, ImageDraw


WIDTH = 640
HEIGHT = 480
CARD_COUNT = 4
QUADRANT_SIZE = 8
MOSAIC_SIZE = 16
SYMBOLS = ("diamond", "cross")
INK_U0 = 0.18
INK_U1 = 0.82

PALETTES = (
    {"frame": (29, 57, 88), "ink": (226, 83, 66), "fiducial": (232, 165, 54)},
    {"frame": (42, 72, 57), "ink": (218, 76, 62), "fiducial": (236, 170, 53)},
    {"frame": (62, 47, 88), "ink": (230, 91, 63), "fiducial": (235, 171, 55)},
)
PAPERS = ((248, 246, 239), (243, 247, 243), (247, 243, 249))
CARD_PAPER = (239, 236, 224)


def _diamond() -> tuple[tuple[int, ...], ...]:
    rows = (
        "................",
        ".......##.......",
        "......####......",
        ".....##..##.....",
        "....##....##....",
        "...##......##...",
        "..##........##..",
        ".##..........##.",
        ".##..........##.",
        "..##........##..",
        "...##......##...",
        "....##....##....",
        ".....##..##.....",
        "......####......",
        ".......##.......",
        "................",
    )
    return tuple(tuple(int(char == "#") for char in row) for row in rows)


def _cross() -> tuple[tuple[int, ...], ...]:
    cells: set[tuple[int, int]] = set()
    for row in range(2, 14):
        cells.update({(row, 7), (row, 8)})
    for column in range(2, 14):
        cells.update({(7, column), (8, column)})
    cells.update({(2, 2), (2, 13), (13, 2), (13, 13)})
    cells.update({(1, 7), (1, 8), (14, 7), (14, 8)})
    return tuple(
        tuple(int((row, column) in cells) for column in range(MOSAIC_SIZE))
        for row in range(MOSAIC_SIZE)
    )


DIAMOND = _diamond()
CROSS = _cross()
TEMPLATES = {"diamond": DIAMOND, "cross": CROSS}


def _hamming(left: tuple[tuple[int, ...], ...], right: tuple[tuple[int, ...], ...]) -> int:
    return sum(a != b for row_a, row_b in zip(left, right) for a, b in zip(row_a, row_b))


def _homography(corners: list[tuple[float, float]]) -> np.ndarray:
    """Return the projective map from unit-square coordinates to the card."""

    source = ((0.0, 0.0), (1.0, 0.0), (1.0, 1.0), (0.0, 1.0))
    matrix: list[list[float]] = []
    target: list[float] = []
    for (u, v), (x, y) in zip(source, corners):
        matrix.append([u, v, 1.0, 0.0, 0.0, 0.0, -x * u, -x * v])
        target.append(x)
        matrix.append([0.0, 0.0, 0.0, u, v, 1.0, -y * u, -y * v])
        target.append(y)
    values = np.linalg.solve(np.asarray(matrix, dtype=float), np.asarray(target, dtype=float))
    return np.asarray(
        [
            [values[0], values[1], values[2]],
            [values[3], values[4], values[5]],
            [values[6], values[7], 1.0],
        ],
        dtype=float,
    )


def _map_point(h: np.ndarray, u: float, v: float) -> tuple[float, float]:
    value = h @ np.asarray([u, v, 1.0], dtype=float)
    return float(value[0] / value[2]), float(value[1] / value[2])


def sample_scene(seed: int) -> dict:
    """Sample balanced symbols with paired nuisance states for parity control."""

    value = int(seed)
    symbol = SYMBOLS[value & 1]
    # Adjacent scene seeds share nuisances while carrying opposite symbols.
    rng = random.Random((value // 2) * 1000003 + 739391)
    panels: list[dict] = []
    # The cards occupy one medium central block but leave at least one
    # ablation-grid column/row between the four decisive footprints.
    bases = ((142.0, 88.0), (363.0, 88.0), (142.0, 327.0), (363.0, 327.0))
    for index, (base_x, base_y) in enumerate(bases):
        panels.append(
            {
                "row": index // 2,
                "column": index % 2,
                "x": base_x + rng.uniform(-7.0, 7.0),
                "y": base_y + rng.uniform(-7.0, 7.0),
                "width": rng.uniform(64.0, 66.0),
                "height": rng.uniform(64.0, 66.0),
                "skew": [
                    rng.uniform(-2.5, 2.5),
                    rng.uniform(-2.5, 2.5),
                    rng.uniform(-2.5, 2.5),
                    rng.uniform(-2.5, 2.5),
                ],
            }
        )
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "panels": panels,
        "palette": rng.randrange(len(PALETTES)),
        "background": rng.randrange(len(PAPERS)),
        "symbol": symbol,
    }


def _panel_corners(panel: dict) -> list[tuple[float, float]]:
    x = float(panel["x"])
    y = float(panel["y"])
    width = float(panel["width"])
    height = float(panel["height"])
    skew = [float(value) for value in panel["skew"]]
    return [
        (x + skew[0], y + skew[1]),
        (x + width + skew[2], y + skew[1] * 0.35),
        (x + width + skew[3], y + height + skew[2] * 0.45),
        (x + skew[1] * 0.45, y + height + skew[3]),
    ]


def _polygon(h: np.ndarray, u0: float, v0: float, u1: float, v1: float) -> list[tuple[int, int]]:
    return [
        tuple(round(value) for value in _map_point(h, u0, v0)),
        tuple(round(value) for value in _map_point(h, u1, v0)),
        tuple(round(value) for value in _map_point(h, u1, v1)),
        tuple(round(value) for value in _map_point(h, u0, v1)),
    ]


def render(scene: dict) -> Image.Image:
    """Render four disconnected cards and their transformed raster fragments."""

    if (int(scene["width"]), int(scene["height"])) != (WIDTH, HEIGHT):
        raise ValueError("unsupported canvas size")
    symbol = str(scene["symbol"])
    if symbol not in TEMPLATES:
        raise ValueError(f"unknown symbol {symbol!r}")
    palette = PALETTES[int(scene["palette"]) % len(PALETTES)]
    paper = PAPERS[int(scene["background"]) % len(PAPERS)]
    image = Image.new("RGB", (WIDTH, HEIGHT), paper)
    draw = ImageDraw.Draw(image)
    template = TEMPLATES[symbol]
    for panel in scene["panels"]:
        corners = _panel_corners(panel)
        h = _homography(corners)
        draw.polygon([tuple(round(value) for value in point) for point in corners], fill=palette["frame"])
        draw.polygon(_polygon(h, 0.14, 0.14, 0.86, 0.86), fill=CARD_PAPER)
        row0 = int(panel["row"]) * QUADRANT_SIZE
        column0 = int(panel["column"]) * QUADRANT_SIZE
        for row in range(QUADRANT_SIZE):
            for column in range(QUADRANT_SIZE):
                if template[row0 + row][column0 + column]:
                    u0 = INK_U0 + (INK_U1 - INK_U0) * column / QUADRANT_SIZE
                    v0 = INK_U0 + (INK_U1 - INK_U0) * row / QUADRANT_SIZE
                    u1 = INK_U0 + (INK_U1 - INK_U0) * (column + 1) / QUADRANT_SIZE
                    v1 = INK_U0 + (INK_U1 - INK_U0) * (row + 1) / QUADRANT_SIZE
                    draw.polygon(_polygon(h, u0, v0, u1, v1), fill=palette["ink"])
        draw.line(
            [tuple(round(value) for value in point) for point in corners] + [tuple(round(value) for value in corners[0])],
            fill=palette["frame"],
            width=3,
            joint="curve",
        )
        for x, y in corners:
            radius = 6
            draw.ellipse((round(x) - radius, round(y) - radius, round(x) + radius, round(y) + radius), fill=palette["fiducial"])
    return image


def analytic_gold(scene: dict) -> str:
    symbol = str(scene["symbol"])
    if symbol not in TEMPLATES:
        raise ValueError(f"unknown symbol {symbol!r}")
    return symbol


def margin(scene: dict) -> float:
    symbol = str(scene["symbol"])
    template = TEMPLATES[symbol]
    other = TEMPLATES["cross" if symbol == "diamond" else "diamond"]
    return float(_hamming(template, other))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
