"""Deterministic station-network world for perfect-matching discovery."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 390
PANEL_CENTERS = ((160, 174), (480, 174), (800, 174))
LABELS = ("left", "center", "right")

# Every tree has eight vertices, seven edges, and degree multiset
# (4, 2, 2, 2, 1, 1, 1, 1). Only POSITIVE has a perfect matching.
POSITIVE = ((0, 1), (0, 2), (0, 3), (0, 4), (1, 5), (2, 6), (3, 7))
NEGATIVE_A = ((0, 1), (0, 4), (0, 5), (0, 6), (1, 2), (2, 3), (3, 7))
NEGATIVE_B = ((0, 1), (0, 2), (0, 4), (0, 5), (1, 6), (2, 3), (3, 7))

BASE_POSITIONS = {
    "positive": ((0, 0), (-51, -43), (55, -39), (10, 55), (-70, 31), (-91, -87), (91, -81), (25, 99)),
    "negative_a": ((-35, 0), (0, -39), (43, -14), (70, 31), (-82, -63), (-91, 0), (-76, 62), (88, 82)),
    "negative_b": ((-24, 0), (-60, -43), (22, -30), (59, 2), (-72, 39), (-9, 72), (-88, -87), (91, 45)),
}


def _canonical_edges(edges: Iterable[Iterable[int]]) -> tuple[tuple[int, int], ...]:
    return tuple((int(a), int(b)) for a, b in edges)


def _has_perfect_matching(edges: Iterable[Iterable[int]], n: int = 8) -> bool:
    adjacency = [set() for _ in range(n)]
    for a, b in _canonical_edges(edges):
        adjacency[a].add(b)
        adjacency[b].add(a)

    def pair(remaining: frozenset[int]) -> bool:
        if not remaining:
            return True
        first = min(remaining)
        return any(pair(remaining - {first, partner}) for partner in adjacency[first] & remaining)

    return pair(frozenset(range(n)))


def sample_scene(seed: int) -> dict:
    """Sample answer position, distractor order, and rigid panel embeddings."""
    rng = random.Random(int(seed))
    answer_index = int(seed) % 3
    negative_kinds = ["negative_a", "negative_b"]
    rng.shuffle(negative_kinds)
    kinds = []
    cursor = 0
    for index in range(3):
        if index == answer_index:
            kinds.append("positive")
        else:
            kinds.append(negative_kinds[cursor])
            cursor += 1
    sources = {"positive": POSITIVE, "negative_a": NEGATIVE_A, "negative_b": NEGATIVE_B}
    panels = []
    for kind in kinds:
        panels.append({
            "kind": kind,
            "edges": [list(edge) for edge in sources[kind]],
            "rotation": rng.randrange(4),
            "mirror": bool(rng.randrange(2)),
            "shift_x": rng.randrange(-4, 5),
            "shift_y": rng.randrange(-4, 5),
        })
    return {"panels": panels}


def _positions(panel: dict, center: tuple[int, int]) -> list[tuple[int, int]]:
    points = BASE_POSITIONS[str(panel["kind"])]
    turns = int(panel["rotation"]) % 4
    mirror = bool(panel["mirror"])
    output = []
    for raw_x, raw_y in points:
        x, y = (-raw_x if mirror else raw_x), raw_y
        for _ in range(turns):
            x, y = -y, x
        output.append((center[0] + x + int(panel["shift_x"]), center[1] + y + int(panel["shift_y"])))
    return output


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (248, 247, 242))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    for panel_index, panel in enumerate(scene["panels"]):
        cx, cy = PANEL_CENTERS[panel_index]
        draw.rounded_rectangle((cx - 142, 18, cx + 142, 334), radius=14, fill=(255, 255, 252), outline=(35, 40, 45), width=3)
        points = _positions(panel, (cx, cy))
        for a, b in panel["edges"]:
            draw.line((points[int(a)], points[int(b)]), fill=(104, 114, 124), width=10)
        for x, y in points:
            draw.ellipse((x - 14, y - 14, x + 14, y + 14), fill=(25, 178, 195), outline=(18, 68, 76), width=3)
        label = LABELS[panel_index]
        box = draw.textbbox((0, 0), label, font=font)
        draw.text((cx - (box[2] - box[0]) // 2, 351), label, fill=(25, 30, 35), font=font)
    return image


def analytic_gold(scene: dict) -> str:
    winners = [index for index, panel in enumerate(scene["panels"]) if _has_perfect_matching(panel["edges"])]
    if len(winners) != 1:
        raise ValueError(f"expected one perfectly matchable panel, found {winners}")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    values = [int(_has_perfect_matching(panel["edges"])) for panel in scene["panels"]]
    ordered = sorted(values, reverse=True)
    return float(ordered[0] - ordered[1])


def is_quarantined(scene: dict) -> bool:
    try:
        expected_degrees = [4, 2, 2, 2, 1, 1, 1, 1]
        for panel in scene["panels"]:
            if len(panel["edges"]) != 7:
                return True
            degrees = [0] * 8
            for a, b in panel["edges"]:
                degrees[int(a)] += 1
                degrees[int(b)] += 1
            if sorted(degrees, reverse=True) != expected_degrees:
                return True
        return margin(scene) < 1.0
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = {
        "panels": [
            {
                **panel,
                "edges": list(reversed(panel["edges"])),
            }
            for panel in scene["panels"]
        ]
    }
    return [("reverse_same_color_edge_draw_order", twin)]
