"""Independent raster-only inverse for perfect-matching panel selection."""

from __future__ import annotations

from collections import deque
from math import hypot

import numpy as np
from PIL import Image

LABELS = ("left", "center", "right")


def _components(mask: np.ndarray, minimum: int = 20) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        pixels = []
        while queue:
            yy, xx = queue.popleft()
            pixels.append((yy, xx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = yy + dy, xx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(pixels) >= minimum:
            ys = np.fromiter((p[0] for p in pixels), dtype=float)
            xs = np.fromiter((p[1] for p in pixels), dtype=float)
            output.append((float(xs.mean()), float(ys.mean()), len(pixels)))
    return output


def _point_segment_distance(point: tuple[float, float], a: tuple[float, float], b: tuple[float, float]) -> tuple[float, float]:
    px, py = point
    ax, ay = a
    bx, by = b
    dx, dy = bx - ax, by - ay
    denom = dx * dx + dy * dy
    if denom == 0:
        return hypot(px - ax, py - ay), 0.0
    t = ((px - ax) * dx + (py - ay) * dy) / denom
    closest = (ax + t * dx, ay + t * dy)
    return hypot(px - closest[0], py - closest[1]), t


def _line_score(link_mask: np.ndarray, first: tuple[float, float], second: tuple[float, float]) -> float:
    x0, y0 = first
    x1, y1 = second
    samples = max(24, int(hypot(x1 - x0, y1 - y0)))
    hits = 0
    for step in range(samples + 1):
        t = 0.20 + 0.60 * step / samples
        x = int(round(x0 + t * (x1 - x0)))
        y = int(round(y0 + t * (y1 - y0)))
        patch = link_mask[max(0, y - 2): y + 3, max(0, x - 2): x + 3]
        hits += int(bool(patch.any()))
    return hits / (samples + 1)


def _has_perfect_matching(adjacency: list[set[int]]) -> bool:
    def pair(remaining: frozenset[int]) -> bool:
        if not remaining:
            return True
        first = min(remaining)
        return any(pair(remaining - {first, partner}) for partner in adjacency[first] & remaining)

    return pair(frozenset(range(len(adjacency))))


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = pixels[..., 0], pixels[..., 1], pixels[..., 2]
    cyan = (red < 65) & (green > 130) & (green < 220) & (blue > 145) & (blue < 230)
    link = (red > 75) & (red < 135) & (green > 85) & (green < 145) & (blue > 90) & (blue < 155)
    stations = _components(cyan, minimum=180)
    if len(stations) != 24:
        raise ValueError(f"expected 24 visible stations, found {len(stations)}")
    panel_nodes: list[list[tuple[float, float]]] = [[], [], []]
    for x, y, _area in stations:
        panel_nodes[min(2, max(0, int(x * 3 // image.width)))].append((x, y))

    winners = []
    for panel_index, nodes in enumerate(panel_nodes):
        if len(nodes) != 8:
            raise ValueError(f"panel {panel_index} has {len(nodes)} stations")
        nodes.sort(key=lambda point: (point[1], point[0]))
        adjacency = [set() for _ in nodes]
        for a in range(8):
            for b in range(a + 1, 8):
                blocked_by_station = any(
                    0.12 < t < 0.88 and distance < 20.0
                    for c, node in enumerate(nodes) if c not in (a, b)
                    for distance, t in [_point_segment_distance(node, nodes[a], nodes[b])]
                )
                if not blocked_by_station and _line_score(link, nodes[a], nodes[b]) > 0.70:
                    adjacency[a].add(b)
                    adjacency[b].add(a)
        edge_count = sum(map(len, adjacency)) // 2
        degrees = sorted((len(neighbors) for neighbors in adjacency), reverse=True)
        if edge_count != 7 or degrees != [4, 2, 2, 2, 1, 1, 1, 1]:
            raise ValueError(f"panel {panel_index} recovered invalid graph: {edge_count} links, degrees {degrees}")
        if _has_perfect_matching(adjacency):
            winners.append(panel_index)
    if len(winners) != 1:
        raise ValueError(f"expected one perfectly matchable panel, found {winners}")
    return LABELS[winners[0]]
