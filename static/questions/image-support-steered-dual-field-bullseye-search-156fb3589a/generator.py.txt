"""Deterministic renderer and analytic arm for dual-field bullseye search."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

WIDTH = 800
HEIGHT = 800
BACKGROUND = [249, 247, 242]
COLORS = ("red", "blue", "green", "purple")
PALETTE = {
    "red": [210, 48, 64],
    "blue": [40, 92, 205],
    "green": [20, 142, 84],
    "purple": [128, 55, 176],
}
GOLD = [224, 169, 28]
BAR_LENGTH = 116.0
BAR_WIDTH = 24.0
TARGET_RADIUS = 62.0
Y_SLOTS = (128.0, 310.0, 492.0, 674.0)
FIELD_X = {"left": 166.0, "right": 634.0}
MIN_TANGENCY_CLEARANCE_PX = 22.0


def _line_distance(bar: dict, target: dict) -> float:
    """Perpendicular distance from the target center to a bar's infinite axis."""
    dx = float(target["cx"]) - float(bar["cx"])
    dy = float(target["cy"]) - float(bar["cy"])
    angle = float(bar["angle"])
    return abs(dx * math.sin(angle) - dy * math.cos(angle))


def _clearance(bar: dict, target: dict) -> float:
    return float(target["radius"]) - _line_distance(bar, target)


def _hit_sets(scene: dict) -> dict[str, set[str]]:
    bars = scene["bars"]
    target = scene["target"]
    if len(bars) != 2 * len(COLORS):
        raise ValueError("scene must have exactly eight bars")
    hits = {"left": set(), "right": set()}
    seen = {"left": set(), "right": set()}
    for bar in bars:
        side = str(bar["side"])
        color = str(bar["color"])
        if side not in hits or color not in COLORS or color in seen[side]:
            raise ValueError("each field must contain every color exactly once")
        seen[side].add(color)
        # Keep the analytic consequence visibly rooted at the two declared scene
        # fields. Inlining this tiny calculation also lets the static field audit
        # distinguish top-level scene inputs from keys of the nested bar records.
        dx = float(target["cx"]) - float(bar["cx"])
        dy = float(target["cy"]) - float(bar["cy"])
        angle = float(bar["angle"])
        distance = abs(dx * math.sin(angle) - dy * math.cos(angle))
        if float(target["radius"]) - distance > 0.0:
            hits[side].add(color)
    if any(colors != set(COLORS) for colors in seen.values()):
        raise ValueError("each field must contain every color exactly once")
    return hits


def tangency_clearance(scene: dict) -> float:
    return min(abs(_clearance(bar, scene["target"])) for bar in scene["bars"])


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target_color = COLORS[int(seed) % len(COLORS)]
    others = [color for color in COLORS if color != target_color]
    rng.shuffle(others)
    left_only, right_only, neither = others
    patterns = {
        target_color: (True, True),
        left_only: (True, False),
        right_only: (False, True),
        neither: (False, False),
    }

    target = {
        "cx": 400.0 + rng.randint(-10, 10),
        "cy": 400.0 + rng.randint(-10, 10),
        "radius": TARGET_RADIUS,
    }
    bars = []
    for side_index, side in enumerate(("left", "right")):
        order = list(COLORS)
        rng.shuffle(order)
        for slot, color in enumerate(order):
            cx = FIELD_X[side] + rng.randint(-10, 10)
            cy = Y_SLOTS[slot] + rng.randint(-9, 9)
            base = math.atan2(target["cy"] - cy, target["cx"] - cx)
            should_hit = patterns[color][side_index]
            if should_hit:
                delta = rng.uniform(-0.055, 0.055)
            else:
                delta = rng.choice((-1.0, 1.0)) * rng.uniform(0.38, 0.50)
            bars.append(
                {
                    "side": side,
                    "color": color,
                    "cx": float(cx),
                    "cy": float(cy),
                    "angle": float(base + delta),
                    "length": BAR_LENGTH,
                    "width": BAR_WIDTH,
                }
            )
    rng.shuffle(bars)
    scene = {
        "bars": bars,
        "target": target,
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
    }
    if is_quarantined(scene):
        raise RuntimeError(f"internal sampler produced a boundary scene for seed {seed}")
    return scene


def _rectangle_corners(bar: dict, scale: int) -> list[tuple[float, float]]:
    cx = float(bar["cx"]) * scale
    cy = float(bar["cy"]) * scale
    angle = float(bar["angle"])
    ux, uy = math.cos(angle), math.sin(angle)
    vx, vy = -uy, ux
    half_l = 0.5 * float(bar["length"]) * scale
    half_w = 0.5 * float(bar["width"]) * scale
    return [
        (cx + sx * half_l * ux + sy * half_w * vx, cy + sx * half_l * uy + sy * half_w * vy)
        for sx, sy in ((-1, -1), (-1, 1), (1, 1), (1, -1))
    ]


def render(scene: dict) -> Image.Image:
    scale = 4
    width = int(scene["width"])
    height = int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)

    target = scene["target"]
    tx = float(target["cx"]) * scale
    ty = float(target["cy"]) * scale
    radius = float(target["radius"]) * scale
    draw.ellipse((tx - radius, ty - radius, tx + radius, ty + radius), fill=tuple(GOLD))

    for bar in scene["bars"]:
        draw.polygon(_rectangle_corners(bar, scale), fill=tuple(PALETTE[str(bar["color"])]))

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    hits = _hit_sets(scene)
    winners = sorted(hits["left"] & hits["right"])
    if len(winners) != 1:
        raise ValueError("scene must have exactly one color hit in both fields")
    return winners[0]


def margin(scene: dict) -> float:
    hits = _hit_sets(scene)
    scores = sorted(
        (int(color in hits["left"]) + int(color in hits["right"]) for color in COLORS),
        reverse=True,
    )
    return float(scores[0] - scores[1]) if len(scores) > 1 else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        hits = _hit_sets(scene)
        counts = (len(hits["left"]), len(hits["right"]))
        return (
            counts != (2, 2)
            or margin(scene) < 1.0
            or tangency_clearance(scene) < MIN_TANGENCY_CLEARANCE_PX
        )
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["bars"] = list(reversed(twin["bars"]))
    return [("reverse_unordered_nonoverlapping_bar_records", twin)]
