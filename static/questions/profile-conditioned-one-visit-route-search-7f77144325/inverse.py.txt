from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[float, float, int]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        todo = deque([(int(y), int(x))])
        seen[y, x] = True
        xs: list[int] = []
        ys: list[int] = []
        while todo:
            cy, cx = todo.popleft()
            xs.append(cx)
            ys.append(cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    todo.append((ny, nx))
        if len(xs) >= 180:
            found.append((float(sum(xs)) / len(xs), float(sum(ys)) / len(ys), len(xs)))
    return found


def _cluster_axis(values: list[float]) -> list[float]:
    ordered = sorted(values)
    groups: list[list[float]] = []
    for value in ordered:
        if not groups or value - sum(groups[-1]) / len(groups[-1]) > 12:
            groups.append([value])
        else:
            groups[-1].append(value)
    if len(groups) != 3 or any(len(group) != 3 for group in groups):
        raise ValueError("each panel must expose a complete 3 by 3 circle lattice")
    return [sum(group) / len(group) for group in groups]


def _edge_present(dark: np.ndarray, a: tuple[float, float], b: tuple[float, float]) -> bool:
    ax, ay = a
    bx, by = b
    if abs(ay - by) < 12:
        y = int(round((ay + by) / 2))
        lo, hi = sorted((int(round(ax)), int(round(bx))))
        strip = dark[max(0, y - 3):y + 4, lo + 16:hi - 16]
    elif abs(ax - bx) < 12:
        x = int(round((ax + bx) / 2))
        lo, hi = sorted((int(round(ay)), int(round(by))))
        strip = dark[lo + 16:hi - 16, max(0, x - 3):x + 4]
    else:
        return False
    return bool(strip.size and strip.mean() > 0.55)


def _panel_graph(dark: np.ndarray, centers: list[tuple[float, float, int]]) -> list[set[int]]:
    xs = _cluster_axis([item[0] for item in centers])
    ys = _cluster_axis([item[1] for item in centers])
    points = [(x, y) for y in ys for x in xs]
    adj = [set() for _ in range(9)]
    for r in range(3):
        for c in range(2):
            a, b = 3 * r + c, 3 * r + c + 1
            if _edge_present(dark, points[a], points[b]):
                adj[a].add(b); adj[b].add(a)
    for r in range(2):
        for c in range(3):
            a, b = 3 * r + c, 3 * (r + 1) + c
            if _edge_present(dark, points[a], points[b]):
                adj[a].add(b); adj[b].add(a)
    edge_count = sum(map(len, adj)) // 2
    if edge_count != 9:
        raise ValueError(f"expected nine visible links, recovered {edge_count}")
    return adj


def _longest_path(adj: list[set[int]]) -> int:
    def visit(node: int, seen: frozenset[int]) -> int:
        best = len(seen)
        for nxt in adj[node]:
            if nxt not in seen:
                best = max(best, visit(nxt, seen | {nxt}))
        return best
    return max(visit(start, frozenset({start})) for start in range(9))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    spread = arr.max(axis=2) - arr.min(axis=2)
    colored = (spread > 65) & (arr.min(axis=2) < 205)
    centers = sorted(_components(colored), key=lambda item: item[0])
    if len(centers) != 36:
        raise ValueError(f"expected 36 visible colored circles, recovered {len(centers)}")
    groups = [centers[i:i + 9] for i in range(0, 36, 9)]
    if any(max(item[0] for item in groups[i]) >= min(item[0] for item in groups[i + 1])
           for i in range(3)):
        raise ValueError("panel circle groups overlap")
    dark = arr.max(axis=2) < 105
    scores = [_longest_path(_panel_graph(dark, group)) for group in groups]
    winners = [i for i, score in enumerate(scores) if score == 9]
    if len(winners) != 1:
        raise ValueError(f"expected one complete one-visit route, scores={scores}")
    if 9 - max(score for i, score in enumerate(scores) if i != winners[0]) < 1:
        raise ValueError("runner-up margin collapsed")
    return "ABCD"[winners[0]]
