from __future__ import annotations

import itertools
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1000, 380
GRID_EDGES = tuple(
    (3 * r + c, 3 * r + c + 1) for r in range(3) for c in range(2)
) + tuple((3 * r + c, 3 * (r + 1) + c) for r in range(2) for c in range(3))
DEGREE_SIGNATURE = (1, 1, 2, 2, 2, 2, 2, 3, 3)
NODE_PALETTE = ((34, 105, 210), (168, 62, 184), (0, 137, 123), (224, 92, 35))


def _adjacency(edges: Iterable[tuple[int, int]]) -> list[set[int]]:
    out = [set() for _ in range(9)]
    for a, b in edges:
        out[a].add(b)
        out[b].add(a)
    return out


def _connected(edges: Iterable[tuple[int, int]]) -> bool:
    adj = _adjacency(edges)
    seen = {0}
    todo = [0]
    while todo:
        for nxt in adj[todo.pop()]:
            if nxt not in seen:
                seen.add(nxt)
                todo.append(nxt)
    return len(seen) == 9


def _longest_path(edges: Iterable[tuple[int, int]]) -> int:
    adj = _adjacency(edges)

    def visit(node: int, seen: frozenset[int]) -> int:
        best = len(seen)
        for nxt in adj[node]:
            if nxt not in seen:
                best = max(best, visit(nxt, seen | {nxt}))
        return best

    return max(visit(start, frozenset({start})) for start in range(9))


def _catalog() -> tuple[list[list[list[int]]], list[list[list[int]]]]:
    winners: list[list[list[int]]] = []
    distractors: list[list[list[int]]] = []
    for edges in itertools.combinations(GRID_EDGES, 9):
        if not _connected(edges):
            continue
        degrees = tuple(sorted(len(v) for v in _adjacency(edges)))
        if degrees != DEGREE_SIGNATURE:
            continue
        encoded = [[a, b] for a, b in edges]
        (winners if _longest_path(edges) == 9 else distractors).append(encoded)
    return winners, distractors


WINNERS, DISTRACTORS = _catalog()


def _transform_vertex(v: int, transform: int) -> int:
    r, c = divmod(v, 3)
    if transform & 4:
        c = 2 - c
    for _ in range(transform & 3):
        r, c = c, 2 - r
    return 3 * r + c


def _transform_edges(edges: list[list[int]], transform: int) -> list[list[int]]:
    return sorted(
        [sorted((_transform_vertex(a, transform), _transform_vertex(b, transform))) for a, b in edges]
    )


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = rng.choice(WINNERS)
    hard = [graph for graph in DISTRACTORS if _longest_path(graph) == 8]
    chosen = rng.sample(hard, 3)
    panels = []
    for is_target, graph in [(True, target), *[(False, item) for item in chosen]]:
        panels.append({
            "edges": _transform_edges(graph, rng.randrange(8)),
            "is_target": is_target,
        })
    rng.shuffle(panels)
    return {"panels": panels, "node_color": list(rng.choice(NODE_PALETTE))}


def analytic_gold(scene: dict) -> str:
    winners = [i for i, panel in enumerate(scene["panels"]) if _longest_path(panel["edges"]) == 9]
    if len(winners) != 1:
        raise ValueError(f"expected one spanning-path panel, found {winners}")
    return "ABCD"[winners[0]]


def margin(scene: dict) -> float:
    scores = sorted((_longest_path(panel["edges"]) for panel in scene["panels"]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    scores = [_longest_path(panel["edges"]) for panel in scene["panels"]]
    return scores.count(9) != 1 or margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    scale = 2
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), (250, 250, 247))
    draw = ImageDraw.Draw(image)
    node_color = tuple(scene.get("node_color", NODE_PALETTE[0]))
    dark = (31, 35, 40)
    border = (154, 160, 166)
    panel_fill = (255, 255, 255)
    label_font = _font(24 * scale)
    title_font = _font(22 * scale)
    draw.text((WIDTH * scale // 2, 13 * scale), "VISIT ALL 9 CIRCLES ONCE", fill=dark,
              font=title_font, anchor="ma")
    for panel_index, panel in enumerate(scene["panels"]):
        x0 = 20 + panel_index * 240
        x1 = x0 + 220
        draw.rounded_rectangle((x0 * scale, 58 * scale, x1 * scale, 358 * scale),
                               radius=10 * scale, fill=panel_fill, outline=border, width=2 * scale)
        draw.text(((x0 + x1) * scale // 2, 74 * scale), "ABCD"[panel_index], fill=dark,
                  font=label_font, anchor="ma")
        points = {}
        for v in range(9):
            r, c = divmod(v, 3)
            points[v] = ((x0 + 45 + 65 * c) * scale, (125 + 92 * r) * scale)
        for a, b in panel["edges"]:
            draw.line((points[a], points[b]), fill=dark, width=7 * scale)
        radius = 12 * scale
        for cx, cy in points.values():
            draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius),
                         fill=node_color, outline=(255, 255, 255), width=2 * scale)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
