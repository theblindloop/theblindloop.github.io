"""Deterministic compact medallion renderer and analytic arm."""

from __future__ import annotations

import copy
import math
import random
from typing import Any

import numpy as np
from PIL import Image, ImageDraw

RENDERER_VERSION = "dual-signature-seal-search-0.1.0"
WIDTH = 768
HEIGHT = 768
SECTORS = ("NW", "NE", "SW", "SE")
CENTERS = {
    "NW": (192, 192),
    "NE": (576, 192),
    "SW": (192, 576),
    "SE": (576, 576),
}
BACKGROUND_PALETTES = (
    (242, 245, 248),
    (247, 241, 233),
    (239, 246, 241),
    (244, 240, 248),
)
INK_PALETTES = (
    (29, 57, 88),
    (117, 49, 43),
    (35, 91, 68),
    (79, 52, 115),
)
CARDINALS = ("N", "E", "S", "W")
LOCAL_SIZE = 180
LOCAL_CENTER = LOCAL_SIZE // 2


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced four-sector truth table with independent nuisance values."""

    rng = random.Random(int(seed))
    target = rng.choice(SECTORS)
    combinations = [(False, False), (True, False), (False, True), (True, True)]
    rng.shuffle(combinations)
    panels: list[dict[str, Any]] = []
    for sector, combination in zip(SECTORS, combinations, strict=True):
        has_hole, has_notch = combination
        base_x, base_y = CENTERS[sector]
        panels.append(
            {
                "sector": sector,
                "center": [base_x + rng.randint(-10, 10), base_y + rng.randint(-10, 10)],
                "has_hole": bool(has_hole),
                "has_notch": bool(has_notch),
                "notch_direction": rng.choice(CARDINALS),
                # These mode-specific dimensions keep foreground mass and
                # bounding boxes close without making the answer a size cue.
                "outer_radius": {("False", "False"): 66, ("True", "False"): 68, ("False", "True"): 68, ("True", "True"): 69}[(str(has_hole), str(has_notch))],
                "hole_radius": 17 if has_hole and not has_notch else 14 if has_hole else 0,
                "notch_depth": 28 if has_notch else 0,
                "notch_width": 24 if has_notch else 0,
            }
        )
    # Shuffle records independently of the fixed sector order. Rendering is
    # non-overlapping, so this is a declared pixel-identical latent alias.
    rng.shuffle(panels)
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(rng.choice(BACKGROUND_PALETTES)),
        "ink": list(rng.choice(INK_PALETTES)),
        "panels": panels,
    }


def _notch_polygon(direction: str, radius: int, depth: int, width: int) -> list[tuple[int, int]]:
    c = LOCAL_CENTER
    half = width // 2
    inner = radius - depth
    if direction == "E":
        return [(c + radius + 6, c - half), (c + radius + 6, c + half), (c + inner, c + half // 2), (c + inner, c - half // 2)]
    if direction == "W":
        return [(c - radius - 6, c - half), (c - radius - 6, c + half), (c - inner, c + half // 2), (c - inner, c - half // 2)]
    if direction == "S":
        return [(c - half, c + radius + 6), (c + half, c + radius + 6), (c + half // 2, c + inner), (c - half // 2, c + inner)]
    return [(c - half, c - radius - 6), (c + half, c - radius - 6), (c + half // 2, c - inner), (c - half // 2, c - inner)]


def _panel_mask(panel: dict[str, Any]) -> Image.Image:
    """Return a local binary mask; all ink is one connected component."""

    mask = Image.new("L", (LOCAL_SIZE, LOCAL_SIZE), 0)
    draw = ImageDraw.Draw(mask)
    c = LOCAL_CENTER
    radius = int(panel["outer_radius"])
    draw.ellipse((c - radius, c - radius, c + radius, c + radius), fill=255)
    if panel["has_hole"]:
        hole = int(panel["hole_radius"])
        draw.ellipse((c - hole, c - hole, c + hole, c + hole), fill=0)
    if panel["has_notch"]:
        draw.polygon(
            _notch_polygon(
                str(panel["notch_direction"]),
                radius,
                int(panel["notch_depth"]),
                int(panel["notch_width"]),
            ),
            fill=0,
        )
    return mask


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new(
        "RGB",
        (int(scene["width"]), int(scene["height"])),
        tuple(int(value) for value in scene["background"]),
    )
    ink = tuple(int(value) for value in scene["ink"])
    background = tuple(int(value) for value in scene["background"])
    for panel in scene["panels"]:
        mask = _panel_mask(panel)
        cx, cy = (int(value) for value in panel["center"])
        left = cx - LOCAL_CENTER
        top = cy - LOCAL_CENTER
        # Paste only the small local mask; this avoids any implicit antialiasing
        # and leaves a flat, independently segmentable RGB raster.
        solid = Image.new("RGB", (LOCAL_SIZE, LOCAL_SIZE), ink)
        # The local image starts as background; paste only the ink mask.
        image.paste(solid, (left, top), mask)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    winners = [
        str(panel["sector"])
        for panel in scene["panels"]
        if bool(panel["has_hole"]) and bool(panel["has_notch"])
    ]
    if len(winners) != 1:
        raise ValueError("scene must contain exactly one dual-signature medallion")
    return winners[0]


def margin(scene: dict[str, Any]) -> float:
    scores = sorted(
        (int(bool(panel["has_hole"])) + int(bool(panel["has_notch"])) for panel in scene["panels"]),
        reverse=True,
    )
    if len(scores) < 4 or scores[0] != 2 or scores[1] >= 2:
        raise ValueError("scene has no unique dual-signature winner")
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        return margin(scene) < 1.0
    except ValueError:
        return True


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    twin = copy.deepcopy(scene)
    twin["panels"] = list(reversed(twin["panels"]))
    return [("reorder_panel_records", twin)]


def rendered_ink_area(scene: dict[str, Any]) -> list[int]:
    """Diagnostic helper used only by local tests; not used by the pixel arm."""

    return [int(np.asarray(_panel_mask(panel), dtype=np.uint8).sum() // 255) for panel in scene["panels"]]
