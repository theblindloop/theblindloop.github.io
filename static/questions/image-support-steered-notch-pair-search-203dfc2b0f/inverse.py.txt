"""Independent image-only inverse for the paired-notch search world."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _colored_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    spread = array.max(axis=2) - array.min(axis=2)
    # All allowed tab palettes are strongly chromatic; frames, text, and the
    # off-white background are neutral. No renderer palette constants are used.
    return (spread > 62) & (array.max(axis=2) < 235)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        points = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= 900:
            output.append(np.asarray(points, dtype=np.int32))
    return output


def _decode_component(points: np.ndarray, mask: np.ndarray) -> tuple[float, int] | None:
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    height, width = int(y1 - y0 + 1), int(x1 - x0 + 1)
    if not (32 <= width <= 42 and 76 <= height <= 88 and 2300 <= len(points) <= 3300):
        return None
    crop = mask[y0 : y1 + 1, x0 : x1 + 1]
    # Interior background belongs to the circular cutout. Requiring its area
    # and compact span rejects clipping and destructive ablations.
    interior = ~crop[4:-4, 6:-6]
    ys, xs = np.nonzero(interior)
    if not (115 <= len(ys) <= 205):
        return None
    if int(ys.max() - ys.min() + 1) > 18 or int(xs.max() - xs.min() + 1) > 18:
        return None
    hole_y = float(y0 + 4 + ys.mean())
    relative = hole_y - float(y0)
    nominal = np.asarray((13.0, 31.0, 49.0, 67.0))
    distances = np.abs(nominal - relative)
    level = int(np.argmin(distances))
    if float(distances[level]) > 2.5:
        return None
    center_x = float(points[:, 1].mean())
    return center_x, level


def _pair(items: list[tuple[float, int]]) -> tuple[int, int] | None:
    if len(items) != 2:
        return None
    ordered = sorted(items)
    if ordered[1][0] - ordered[0][0] < 30:
        return None
    return ordered[0][1], ordered[1][1]


def decision_from_image(image: Image.Image) -> str:
    mask = _colored_mask(image)
    height, width = mask.shape
    decoded = []
    for points in _components(mask):
        item = _decode_component(points, mask)
        if item is not None:
            center_y = float(points[:, 0].mean())
            decoded.append((item[0], center_y, item[1]))

    # Regions are enumerated only by visible relational layout: the sample is
    # the top band, and A/B/C/D are the reading-order quadrants below it.
    sample_items = [(x, level) for x, y, level in decoded if y < height * 0.36]
    sample = _pair(sample_items)
    if sample is None:
        raise ValueError("sample pair is missing or ambiguous")

    slots: dict[str, list[tuple[float, int]]] = {key: [] for key in "ABCD"}
    for x, y, level in decoded:
        if y < height * 0.36:
            continue
        # The four labeled candidates occupy the four equal-width reading-order
        # regions. Relative quarters, not renderer coordinates, enumerate them.
        slot_index = min(3, max(0, int(4 * x / width)))
        label = "ABCD"[slot_index]
        slots[label].append((x, level))
    codes = {label: _pair(items) for label, items in slots.items()}
    matches = [
        label
        for label, code in codes.items()
        if code is not None and sum(code) == sum(sample)
    ]
    if len(matches) != 1:
        raise ValueError("raster does not expose one unique exact match")
    return matches[0]
