"""Deterministic renderer and analytic arm for paired-notch visual search."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 800, 640
BACKGROUND = (248, 247, 242)
PANEL = (229, 229, 222)
FRAME = (93, 94, 91)
TEXT = (47, 48, 46)
PALE_TEXT = (112, 113, 108)
PALETTES = (
    (18, 151, 179),
    (205, 72, 88),
    (112, 83, 181),
    (25, 151, 92),
)
LEVELS = (13, 31, 49, 67)
LABELS = ("A", "B", "C", "D")


def _code(value: Any) -> tuple[int, int]:
    if not isinstance(value, (list, tuple)) or len(value) != 2:
        raise ValueError("notch code must contain two levels")
    code = (int(value[0]), int(value[1]))
    if any(item not in range(4) for item in code):
        raise ValueError("notch level outside 0..3")
    return code


def _codes_for_gold(scene: dict) -> tuple[tuple[int, int], list[tuple[int, int]]]:
    sample = _code(scene["sample"])
    codes = [_code(value) for value in scene["codes"]]
    if len(codes) != 4:
        raise ValueError("exactly four candidates required")
    return sample, codes


def _visual_fields(scene: dict) -> tuple[int, int, tuple[int, int, int]]:
    shift_x = int(scene["shift_x"])
    shift_y = int(scene["shift_y"])
    palette_index = int(scene["palette_index"])
    if not -10 <= shift_x <= 10 or not -10 <= shift_y <= 10:
        raise ValueError("layout translation outside render bounds")
    if palette_index not in range(len(PALETTES)):
        raise ValueError("unknown palette")
    return shift_x, shift_y, PALETTES[palette_index]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x4E4F5443)
    valid_samples = [
        (left, right)
        for left in range(4)
        for right in range(4)
        if any(
            a != left and b != right and a + b == left + right
            for a in range(4)
            for b in range(4)
        )
    ]
    sample = rng.choice(valid_samples)
    answer_index = int(seed) % 4

    # Every candidate differs from the sample in both local notch positions.
    # Thus the matching total cannot be found by copying either single notch or
    # by exact-shape matching.
    alternate_pool = [
        (left, right)
        for left in range(4)
        for right in range(4)
        if left != sample[0] and right != sample[1]
    ]
    winner_pool = [code for code in alternate_pool if sum(code) == sum(sample)]
    distractor_pool = [code for code in alternate_pool if sum(code) != sum(sample)]
    winner = rng.choice(winner_pool)
    rng.shuffle(distractor_pool)
    distractors = distractor_pool[:3]
    codes: list[tuple[int, int]] = []
    next_distractor = 0
    for index in range(4):
        if index == answer_index:
            codes.append(winner)
        else:
            codes.append(distractors[next_distractor])
            next_distractor += 1
    return {
        "sample": list(sample),
        "codes": [list(code) for code in codes],
        "shift_x": rng.randint(-8, 8),
        "shift_y": rng.randint(-6, 6),
        "palette_index": (int(seed) // 4) % len(PALETTES),
    }


def analytic_gold(scene: dict) -> str:
    sample, codes = _codes_for_gold(scene)
    matches = [index for index, code in enumerate(codes) if sum(code) == sum(sample)]
    if len(matches) != 1:
        raise ValueError("scene must have one exact match")
    return LABELS[matches[0]]


def margin(scene: dict) -> float:
    sample, codes = _codes_for_gold(scene)
    distances = sorted(abs(sum(code) - sum(sample)) for code in codes)
    if not distances or distances[0] != 0 or (len(distances) > 1 and distances[1] == 0):
        return 0.0
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _draw_tab(draw: ImageDraw.ImageDraw, x: int, y: int, level: int, color: tuple[int, int, int]) -> None:
    draw.rounded_rectangle((x, y, x + 36, y + 82), radius=4, fill=color)
    cy = y + LEVELS[level]
    draw.ellipse((x + 11, cy - 7, x + 25, cy + 7), fill=BACKGROUND)


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    box: tuple[int, int, int, int],
    code: tuple[int, int],
    color: tuple[int, int, int],
    label: str | None,
) -> None:
    x0, y0, x1, y1 = box
    draw.rounded_rectangle(box, radius=10, fill=PANEL, outline=FRAME, width=3)
    tab_y = y0 + 19
    _draw_tab(draw, x0 + 42, tab_y, code[0], color)
    _draw_tab(draw, x0 + 92, tab_y, code[1], color)
    for offset in LEVELS:
        tick_y = tab_y + offset
        draw.line((x0 + 8, tick_y, x0 + 15, tick_y), fill=PALE_TEXT, width=2)
    if label is not None:
        font = ImageFont.load_default()
        center = (x0 + x1) // 2
        draw.rounded_rectangle((center - 13, y1 + 7, center + 13, y1 + 31), radius=5, fill=TEXT)
        bbox = draw.textbbox((0, 0), label, font=font)
        tw, th = bbox[2] - bbox[0], bbox[3] - bbox[1]
        draw.text((center - tw / 2, y1 + 19 - th / 2 - 1), label, fill=BACKGROUND, font=font)


def render(scene: dict) -> Image.Image:
    sample, codes = _codes_for_gold(scene)
    shift_x, shift_y, color = _visual_fields(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    title = "SAMPLE"
    title_box = draw.textbbox((0, 0), title, font=font)
    draw.text(((WIDTH - (title_box[2] - title_box[0])) / 2 + shift_x, 45 + shift_y), title, fill=TEXT, font=font)
    _draw_panel(draw, (315 + shift_x, 70 + shift_y, 485 + shift_x, 190 + shift_y), sample, color, None)

    subtitle = "ADD THE TWO HEIGHT NUMBERS - FIND THE SAME TOTAL"
    subtitle_box = draw.textbbox((0, 0), subtitle, font=font)
    draw.text(((WIDTH - (subtitle_box[2] - subtitle_box[0])) / 2 + shift_x, 227 + shift_y), subtitle, fill=PALE_TEXT, font=font)

    boxes = (
        (35, 360, 165, 480),
        (235, 360, 365, 480),
        (435, 360, 565, 480),
        (635, 360, 765, 480),
    )
    for code, label, box in zip(codes, LABELS, boxes):
        shifted = tuple(value + (shift_x if index % 2 == 0 else shift_y) for index, value in enumerate(box))
        _draw_panel(draw, shifted, code, color, label)
    return image
