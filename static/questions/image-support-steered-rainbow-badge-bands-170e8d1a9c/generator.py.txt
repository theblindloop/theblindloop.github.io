"""Deterministic scene sampler and forward renderer.

The badge area stays fixed within each matched quartet while its visible number
of chromatic partitions takes values 5, 6, 7, and 8.  The seven-band member is
the canonical-prior control; all members use exactly the same prompt.
"""

from __future__ import annotations

import random

import numpy as np
from PIL import Image

WIDTH = 512
HEIGHT = 512
OUTLINE = (34, 39, 48)
BACKGROUND_RGB = {
    "warm": (248, 247, 243),
    "cool": (244, 248, 249),
}
DISTRACTOR = (190, 195, 202)
PALETTE = {
    "red": (220, 49, 63),
    "orange": (239, 118, 43),
    "amber": (226, 172, 31),
    "lime": (139, 181, 48),
    "green": (36, 155, 91),
    "teal": (27, 157, 151),
    "cyan": (35, 146, 202),
    "blue": (50, 88, 187),
    "indigo": (85, 65, 171),
    "violet": (144, 70, 180),
    "magenta": (202, 61, 148),
    "coral": (225, 88, 105),
}


def sample_scene(seed: int) -> dict:
    """Return one member of a deterministic matched quartet.

    Seeds sharing ``seed // 4`` share every nuisance.  Only the visible band
    count and the corresponding prefix of a shuffled color order differ.
    """

    band_count = 5 + (int(seed) % 4)
    group = int(seed) // 4
    rng = random.Random((group + 0x9E3779B9) * 1_000_003)
    center_x = rng.randint(232, 280)
    center_y = rng.randint(232, 280)
    radius = rng.choice((90, 94, 98, 102, 106))
    angle = rng.choice((0, 90))
    outline_width = rng.choice((8, 10, 12))
    background = rng.choice(("warm", "cool"))
    color_order = list(PALETTE)
    rng.shuffle(color_order)

    distractors: list[list[int]] = []
    # Answer-independent pale dots sit well outside the badge. Their order is
    # immaterial, which is tested as a label-preserving distractor permutation.
    anchors = ((82, 94), (424, 91), (83, 419), (428, 421), (257, 55), (455, 258))
    chosen = rng.sample(anchors, 4)
    for x, y in chosen:
        distractors.append([x + rng.randint(-8, 8), y + rng.randint(-8, 8), rng.randint(5, 9)])

    return {
        "band_count": band_count,
        "center_x": center_x,
        "center_y": center_y,
        "radius": radius,
        "angle": angle,
        "colors": color_order[:band_count],
        "outline_width": outline_width,
        "distractors": distractors,
        "background": background,
    }


def _validate(scene: dict) -> None:
    count = int(scene["band_count"])
    if count not in (5, 6, 7, 8):
        raise ValueError("band_count must be 5, 6, 7, or 8")
    if len(scene["colors"]) != count or len(set(scene["colors"])) != count:
        raise ValueError("colors must contain one distinct entry per band")
    if any(name not in PALETTE for name in scene["colors"]):
        raise ValueError("unknown band color")
    if int(scene["angle"]) % 360 not in (0, 90, 180, 270):
        raise ValueError("angle must be an axis-aligned directed orientation")
    if scene["background"] not in BACKGROUND_RGB:
        raise ValueError("unknown background")


def render(scene: dict) -> Image.Image:
    _validate(scene)
    count = int(scene["band_count"])
    cx = int(scene["center_x"])
    cy = int(scene["center_y"])
    radius = int(scene["radius"])
    angle = int(scene["angle"]) % 360
    names = list(scene["colors"])
    border = int(scene["outline_width"])
    distractors = list(scene["distractors"])
    background = BACKGROUND_RGB[scene["background"]]
    if not (78 <= radius <= 116 and 6 <= border <= 14 and radius > border + 50):
        raise ValueError("badge geometry is outside the supported visible range")

    array = np.empty((HEIGHT, WIDTH, 3), dtype=np.uint8)
    array[:, :] = background
    yy, xx = np.ogrid[:HEIGHT, :WIDTH]

    for item in distractors:
        dx, dy, dot_radius = (int(item[0]), int(item[1]), int(item[2]))
        dot = (xx - dx) ** 2 + (yy - dy) ** 2 <= dot_radius**2
        array[dot] = DISTRACTOR

    outer = (xx - cx) ** 2 + (yy - cy) ** 2 <= radius**2
    array[outer] = OUTLINE

    inner_radius = radius - border
    inner = (xx - cx) ** 2 + (yy - cy) ** 2 <= inner_radius**2

    # Angles 180 and 270 are the opposite directed descriptions of the same
    # undirected stripe axis. Canonicalizing them while reversing the supplied
    # order makes the declared latent alias exactly pixel-identical.
    if angle in (180, 270):
        names.reverse()
    coordinate = xx - cx if angle in (0, 180) else yy - cy
    band_index = ((coordinate + inner_radius) * count) // (2 * inner_radius + 1)
    band_index = np.clip(band_index, 0, count - 1)
    for index, name in enumerate(names):
        array[inner & (band_index == index)] = PALETTE[name]

    return Image.fromarray(array, mode="RGB")


def analytic_gold(scene: dict) -> str:
    return str(int(scene["band_count"]))


def margin(scene: dict) -> float:
    inner_diameter = 2.0 * (int(scene["radius"]) - int(scene["outline_width"]))
    return inner_diameter / int(scene["band_count"]) - 10.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 3.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["angle"] = (int(scene["angle"]) + 180) % 360
    twin["colors"] = list(reversed(scene["colors"]))
    return [("reverse_directed_axis_and_color_order", twin)]
