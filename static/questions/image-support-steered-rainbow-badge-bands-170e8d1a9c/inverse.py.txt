"""Independent image-only inverse arm for the rainbow badge.

This module deliberately duplicates the public color vocabulary and contains
no import of forward-world, prompt, generation, verification, or gold code.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_COLORS = (
    (220, 49, 63),
    (239, 118, 43),
    (226, 172, 31),
    (139, 181, 48),
    (36, 155, 91),
    (27, 157, 151),
    (35, 146, 202),
    (50, 88, 187),
    (85, 65, 171),
    (144, 70, 180),
    (202, 61, 148),
    (225, 88, 105),
)
_DARK = (34, 39, 48)


def _component_count(mask: np.ndarray) -> int:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    components = 0
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        components += 1
        queue: deque[tuple[int, int]] = deque([(y, x)])
        seen[y, x] = True
        while queue:
            cy, cx = queue.popleft()
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
    return components


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 256 or array.shape[1] < 256:
        raise ValueError("raster is too small for a resolvable badge")

    masks = [np.all(array == np.asarray(color, dtype=np.uint8), axis=2) for color in _COLORS]
    areas = np.asarray([int(mask.sum()) for mask in masks])
    present = areas >= 220
    count = int(present.sum())
    if count not in (5, 6, 7, 8):
        raise ValueError("could not recover an allowed visible band count")
    if np.any((areas > 0) & (areas < 40)):
        raise ValueError("ambiguous chromatic fragments are present")

    chromatic = np.logical_or.reduce(masks)
    dark = np.all(array == np.asarray(_DARK, dtype=np.uint8), axis=2)
    badge = chromatic | dark
    if int(chromatic.sum()) < 16_000 or int(dark.sum()) < 2_000:
        raise ValueError("badge has insufficient filled area or outline support")
    if _component_count(badge) != 1:
        raise ValueError("outlined badge is not one connected component")
    for mask, is_present in zip(masks, present, strict=True):
        if is_present and _component_count(mask) != 1:
            raise ValueError("a counted band is interrupted")

    points = np.argwhere(badge)
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    height = int(y1 - y0 + 1)
    width = int(x1 - x0 + 1)
    if not (165 <= width <= 235 and 165 <= height <= 235 and abs(width - height) <= 4):
        raise ValueError("badge outline is not a medium circular region")

    center_y = (float(y0) + float(y1)) / 2.0
    center_x = (float(x0) + float(x1)) / 2.0
    radius_y = (height - 1) / 2.0
    radius_x = (width - 1) / 2.0
    yy, xx = np.ogrid[: array.shape[0], : array.shape[1]]
    expected = ((yy - center_y) / radius_y) ** 2 + ((xx - center_x) / radius_x) ** 2 <= 1.0
    expected_area = int(expected.sum())
    missing_fraction = float(np.count_nonzero(expected & ~badge)) / expected_area
    extra_fraction = float(np.count_nonzero(badge & ~expected)) / expected_area
    if missing_fraction > 0.0035 or extra_fraction > 0.0035:
        raise ValueError("badge area is erased, clipped, or non-circular")

    return str(count)
