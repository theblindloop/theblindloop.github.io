"""Independent pixel-only inverse for seam connector correspondence.

This module intentionally contains its own fixed public layout constants and
does not import renderer, prompts, generation, verification, or analytic gold.
It segments the four saturated palette colors, measures local edge positions in
the reference and candidate tiles, and compares their four seam signatures.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


PALETTE = np.asarray(
    [
        (225, 70, 75),
        (65, 105, 190),
        (70, 165, 95),
        (235, 155, 55),
    ],
    dtype=np.float32,
)
REF_TILE = (128, 140, 228, 240)
A_TILE = (442, 140, 542, 240)
B_TILE = (762, 140, 862, 240)
SIDES = ("top", "right", "bottom", "left")
# Only inspect the pixels immediately adjacent to a tile edge. A wider band
# would include a perpendicular stub near a corner and confuse its color with
# the intended endpoint.
EDGE_BAND = 10
MATCH_TOLERANCE = 0.08


def _palette_labels(arr: np.ndarray) -> np.ndarray:
    distances = np.sqrt(((arr[:, :, None, :] - PALETTE[None, None, :, :]) ** 2).sum(axis=3))
    labels = distances.argmin(axis=2)
    best = distances.min(axis=2)
    spread = arr.max(axis=2) - arr.min(axis=2)
    return np.where((best <= 90.0) & (spread >= 42.0), labels, -1)


def _edge_observation(labels: np.ndarray, rect: tuple[int, int, int, int], side: str, inward: bool) -> tuple[int, float]:
    x0, y0, x1, y1 = rect
    if side == "top":
        if inward:
            region = labels[y0 : y0 + EDGE_BAND + 1, x0 + 4 : x1 - 3]
        else:
            region = labels[y0 - EDGE_BAND : y0 + 1, x0 + 4 : x1 - 3]
        along_axis = 1
        origin = x0
        size = x1 - x0
    elif side == "bottom":
        if inward:
            region = labels[y1 - EDGE_BAND : y1 + 1, x0 + 4 : x1 - 3]
        else:
            region = labels[y1 : y1 + EDGE_BAND + 1, x0 + 4 : x1 - 3]
        along_axis = 1
        origin = x0
        size = x1 - x0
    elif side == "left":
        if inward:
            region = labels[y0 + 4 : y1 - 3, x0 : x0 + EDGE_BAND + 1]
        else:
            region = labels[y0 + 4 : y1 - 3, x0 - EDGE_BAND : x0 + 1]
        along_axis = 0
        origin = y0
        size = y1 - y0
    else:  # right
        if inward:
            region = labels[y0 + 4 : y1 - 3, x1 - EDGE_BAND : x1 + 1]
        else:
            region = labels[y0 + 4 : y1 - 3, x1 : x1 + EDGE_BAND + 1]
        along_axis = 0
        origin = y0
        size = y1 - y0

    valid = region >= 0
    if not np.any(valid):
        return (-1, 0.0)
    counts = np.bincount(region[valid].astype(np.int64), minlength=len(PALETTE))
    color = int(counts.argmax())
    coords = np.argwhere(region == color)
    if along_axis == 1:
        positions = coords[:, 1] + (x0 + 4)
    else:
        positions = coords[:, 0] + (y0 + 4)
    t = (float(np.median(positions)) - float(origin)) / float(size)
    return (color, t)


def _signature(labels: np.ndarray, rect: tuple[int, int, int, int], inward: bool) -> dict[str, tuple[int, float]]:
    return {side: _edge_observation(labels, rect, side, inward) for side in SIDES}


def _score(reference: dict[str, tuple[int, float]], candidate: dict[str, tuple[int, float]]) -> int:
    score = 0
    for side in SIDES:
        ref_color, ref_t = reference[side]
        cand_color, cand_t = candidate[side]
        if ref_color >= 0 and ref_color == cand_color and abs(ref_t - cand_t) <= MATCH_TOLERANCE:
            score += 1
    return score


def decision_from_image(image: Image.Image) -> str:
    """Return A/B using only the rasterized pixels."""

    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _palette_labels(arr)
    reference = _signature(labels, REF_TILE, inward=False)
    candidate_a = _signature(labels, A_TILE, inward=True)
    candidate_b = _signature(labels, B_TILE, inward=True)
    score_a = _score(reference, candidate_a)
    score_b = _score(reference, candidate_b)
    return "A" if score_a >= score_b else "B"
