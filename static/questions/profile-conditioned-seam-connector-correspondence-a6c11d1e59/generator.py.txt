"""Renderer and analytic label for seam-connector correspondence.

The reference panel exposes one colored stub on each side of a blank square.
Each candidate tile contains four inward stubs.  A candidate is correct exactly
when all four side/color/position signatures agree with the reference.  The
oracle deliberately re-measures those signatures from the PNG instead of
calling any of the helpers in this module.
"""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont


CANVAS = (1000, 430)
BACKGROUND = (255, 255, 255)
PALETTE = (
    (225, 70, 75),    # coral
    (65, 105, 190),   # blue
    (70, 165, 95),    # green
    (235, 155, 55),   # amber
)
REF_TILE = (128, 140, 228, 240)
A_TILE = (442, 140, 542, 240)
B_TILE = (762, 140, 862, 240)
TILE_SIZE = 100
STUB_LENGTH = 36
MATCH_TOLERANCE = 0.08
SIDES = ("top", "right", "bottom", "left")


def _point(rect: tuple[int, int, int, int], side: str, t: float) -> tuple[float, float]:
    x0, y0, x1, y1 = rect
    if side == "top":
        return (x0 + t * (x1 - x0), float(y0))
    if side == "right":
        return (float(x1), y0 + t * (y1 - y0))
    if side == "bottom":
        return (x0 + t * (x1 - x0), float(y1))
    if side == "left":
        return (float(x0), y0 + t * (y1 - y0))
    raise ValueError(f"unknown side {side!r}")


def _port_map(ports: Iterable[dict]) -> dict[str, dict]:
    return {str(port["side"]): port for port in ports}


def _match_count(reference_ports: list[dict], candidate_ports: list[dict]) -> int:
    """Count seam contacts that agree in side, color, and normalized position."""

    reference = _port_map(reference_ports)
    candidate = _port_map(candidate_ports)
    matches = 0
    for side in SIDES:
        left = reference.get(side)
        right = candidate.get(side)
        if left is None or right is None:
            continue
        if (
            int(left["color"]) == int(right["color"])
            and abs(float(left["t"]) - float(right["t"])) <= MATCH_TOLERANCE
        ):
            matches += 1
    return matches


def sample_scene(seed: int) -> dict:
    """Sample a balanced, rendered four-port correspondence scene."""

    rng = random.Random(int(seed))
    positions = [0.22, 0.36, 0.64, 0.78]
    rng.shuffle(positions)
    colors = list(range(len(PALETTE)))
    rng.shuffle(colors)
    reference_ports = [
        {"side": side, "t": positions[index], "color": colors[index]}
        for index, side in enumerate(SIDES)
    ]

    correct = [dict(port) for port in reference_ports]
    distractor = [dict(port) for port in reference_ports]
    mismatch_index = rng.randrange(len(SIDES))
    if rng.random() < 0.5:
        # A visibly shifted edge landmark; the shift is always beyond the
        # pixel comparison tolerance but preserves the number of stubs.
        old_t = float(distractor[mismatch_index]["t"])
        alternatives = [value for value in positions if abs(value - old_t) > 0.12]
        distractor[mismatch_index]["t"] = rng.choice(alternatives)
    else:
        # A color mismatch with exactly the same position and ink count.
        old_color = int(distractor[mismatch_index]["color"])
        distractor[mismatch_index]["color"] = (old_color + 1 + rng.randrange(3)) % len(PALETTE)

    if int(seed) % 2 == 0:
        candidate_a, candidate_b = correct, distractor
    else:
        candidate_a, candidate_b = distractor, correct

    # Keep the public color identities stable for the independent inverse. The
    # scene still counterbalances color as a cheap cue by permuting which
    # side receives each color; a palette-order alias would add no visual
    # information and would make the contract unnecessarily opaque.
    palette = list(PALETTE)
    return {
        "canvas": list(CANVAS),
        "background": list(BACKGROUND),
        "palette": [list(color) for color in palette],
        "reference_ports": reference_ports,
        "candidate_a": candidate_a,
        "candidate_b": candidate_b,
        "candidate_order": ["A", "B"],
        "stroke_width": 12,
    }


def _font(size: int):
    try:
        return ImageFont.load_default(size=size)
    except TypeError:  # Pillow versions with a fixed-size default font.
        return ImageFont.load_default()


def _scaled_box(box: tuple[float, float, float, float], scale: int) -> tuple[int, int, int, int]:
    return tuple(int(round(value * scale)) for value in box)  # type: ignore[return-value]


def _draw_panel(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], scale: int, title: str) -> None:
    draw.rectangle(_scaled_box(box, scale), fill=(248, 249, 251), outline=(150, 158, 170), width=2 * scale)
    x0, y0, x1, _ = box
    draw.text(
        ((x0 + x1) * scale // 2, 69 * scale),
        title,
        fill=(35, 40, 48),
        anchor="mm",
        font=_font(22 * scale),
    )


def _draw_tile(draw: ImageDraw.ImageDraw, rect: tuple[int, int, int, int], scale: int, fill: tuple[int, int, int]) -> None:
    draw.rectangle(_scaled_box(rect, scale), fill=fill, outline=(86, 96, 108), width=3 * scale)


def _draw_port(
    draw: ImageDraw.ImageDraw,
    rect: tuple[int, int, int, int],
    port: dict,
    palette: list[tuple[int, int, int]],
    scale: int,
    outward: bool,
    width: int,
) -> None:
    side = str(port["side"])
    start = _point(rect, side, float(port["t"]))
    sign = -1.0 if outward else 1.0
    if side == "top":
        end = (start[0], start[1] + sign * STUB_LENGTH)
    elif side == "bottom":
        end = (start[0], start[1] - sign * STUB_LENGTH)
    elif side == "left":
        end = (start[0] + sign * STUB_LENGTH, start[1])
    else:  # right
        end = (start[0] - sign * STUB_LENGTH, start[1])
    color = tuple(palette[int(port["color"])])
    draw.line(
        [(int(round(start[0] * scale)), int(round(start[1] * scale))),
         (int(round(end[0] * scale)), int(round(end[1] * scale)))],
        fill=color,
        width=width * scale,
    )


def render(scene: dict) -> Image.Image:
    """Render the fixed three-panel question world as an RGB PNG image."""

    canvas = tuple(int(value) for value in scene["canvas"])
    background = tuple(int(value) for value in scene["background"])
    palette = [tuple(int(value) for value in color) for color in scene["palette"]]
    width = int(scene["stroke_width"])
    scale = 3
    image = Image.new("RGB", (canvas[0] * scale, canvas[1] * scale), background)
    draw = ImageDraw.Draw(image)

    _draw_panel(draw, (28, 48, 328, 338), scale, "REFERENCE")
    _draw_panel(draw, (352, 48, 642, 338), scale, "CANDIDATE A")
    _draw_panel(draw, (672, 48, 962, 338), scale, "CANDIDATE B")

    # The surrounding reference board makes the missing central tile and its
    # four neighboring seam fragments explicit without encoding the answer.
    rx0, ry0, rx1, ry1 = REF_TILE
    outer = (rx0 - TILE_SIZE, ry0 - TILE_SIZE, rx1 + TILE_SIZE, ry1 + TILE_SIZE)
    draw.rectangle(_scaled_box(outer, scale), outline=(185, 191, 200), width=2 * scale)
    draw.line([(rx0 * scale, (ry0 - TILE_SIZE) * scale), (rx0 * scale, (ry1 + TILE_SIZE) * scale)], fill=(205, 210, 218), width=2 * scale)
    draw.line([(rx1 * scale, (ry0 - TILE_SIZE) * scale), (rx1 * scale, (ry1 + TILE_SIZE) * scale)], fill=(205, 210, 218), width=2 * scale)
    draw.line([((rx0 - TILE_SIZE) * scale, ry0 * scale), ((rx1 + TILE_SIZE) * scale, ry0 * scale)], fill=(205, 210, 218), width=2 * scale)
    draw.line([((rx0 - TILE_SIZE) * scale, ry1 * scale), ((rx1 + TILE_SIZE) * scale, ry1 * scale)], fill=(205, 210, 218), width=2 * scale)
    _draw_tile(draw, REF_TILE, scale, (255, 255, 255))
    # A small broken outline indicates the blank location, but gray pixels are
    # deliberately distinct from the four saturated seam colors.
    for start in range(0, TILE_SIZE, 16):
        end = min(start + 9, TILE_SIZE)
        draw.line([( (rx0 + start) * scale, ry0 * scale), ((rx0 + end) * scale, ry0 * scale)], fill=(115, 125, 138), width=2 * scale)
        draw.line([( (rx0 + start) * scale, ry1 * scale), ((rx0 + end) * scale, ry1 * scale)], fill=(115, 125, 138), width=2 * scale)
        draw.line([(rx0 * scale, (ry0 + start) * scale), (rx0 * scale, (ry0 + end) * scale)], fill=(115, 125, 138), width=2 * scale)
        draw.line([(rx1 * scale, (ry0 + start) * scale), (rx1 * scale, (ry0 + end) * scale)], fill=(115, 125, 138), width=2 * scale)
    for port in scene["reference_ports"]:
        _draw_port(draw, REF_TILE, port, palette, scale, outward=True, width=width)

    _draw_tile(draw, A_TILE, scale, (250, 251, 253))
    _draw_tile(draw, B_TILE, scale, (250, 251, 253))
    for port in scene["candidate_a"]:
        _draw_port(draw, A_TILE, port, palette, scale, outward=False, width=width)
    for port in scene["candidate_b"]:
        _draw_port(draw, B_TILE, port, palette, scale, outward=False, width=width)

    if scale > 1:
        image = image.resize(canvas, Image.Resampling.LANCZOS)
    return image


def analytic_gold(scene: dict) -> str:
    """Select the unique candidate with the larger visible seam score."""

    reference_ports = scene["reference_ports"]
    candidate_a = scene["candidate_a"]
    candidate_b = scene["candidate_b"]
    score_a = _match_count(reference_ports, candidate_a)
    score_b = _match_count(reference_ports, candidate_b)
    return "A" if score_a >= score_b else "B"


def margin(scene: dict) -> float:
    """Return the runner-up gap as a fraction of the four seam contacts."""

    reference_ports = scene["reference_ports"]
    candidate_a = scene["candidate_a"]
    candidate_b = scene["candidate_b"]
    score_a = _match_count(reference_ports, candidate_a)
    score_b = _match_count(reference_ports, candidate_b)
    return abs(score_a - score_b) / float(len(SIDES))


def is_quarantined(scene: dict) -> bool:
    """Quarantine tied correspondence scores as non-unique assignments."""

    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """This parametrization has no unrendered aliases; all labels are explicit."""

    return []
