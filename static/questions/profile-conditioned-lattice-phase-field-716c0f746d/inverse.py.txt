"""Independent pixel-only inverse for the periodic dot-field question."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

_N = 12
_COLORS = {
    "red": np.asarray((215, 58, 67), dtype=np.int16),
    "blue": np.asarray((48, 103, 202), dtype=np.int16),
    "green": np.asarray((42, 163, 103), dtype=np.int16),
}


def _components(mask: np.ndarray) -> list[np.ndarray]:
    coordinates = {tuple(value) for value in np.argwhere(mask)}
    components = []
    while coordinates:
        start = coordinates.pop()
        stack = [start]
        component = [start]
        while stack:
            row, col = stack.pop()
            for dr in (-1, 0, 1):
                for dc in (-1, 0, 1):
                    if dr == 0 and dc == 0:
                        continue
                    neighbor = (row + dr, col + dc)
                    if neighbor in coordinates:
                        coordinates.remove(neighbor)
                        stack.append(neighbor)
                        component.append(neighbor)
        if 20 <= len(component) <= 500:
            components.append(np.asarray(component, dtype=float))
    return components


def _dot_centers(rgb: np.ndarray, target: np.ndarray) -> np.ndarray:
    distance = np.max(np.abs(rgb.astype(np.int16) - target[None, None, :]), axis=2)
    mask = distance <= 4
    components = _components(mask)
    centers = []
    for component in components:
        row_mean, col_mean = np.mean(component, axis=0)
        centers.append((col_mean, row_mean))
    return np.asarray(centers, dtype=float)


def _one_dimensional_groups(values: np.ndarray, size: int) -> np.ndarray:
    ordered = np.sort(values)
    return np.asarray(
        [float(np.mean(ordered[start : start + size])) for start in range(0, len(ordered), size)],
        dtype=float,
    )


def _regularity(points: np.ndarray) -> float:
    if points.shape != (_N, 2):
        return 0.0
    centered = points - np.mean(points, axis=0)
    covariance = np.cov(centered, rowvar=False, bias=True)
    values, vectors = np.linalg.eigh(covariance)
    axis = vectors[:, int(np.argmax(values))]
    length = max(float(np.linalg.norm(axis)), 1e-9)
    axis = axis / length
    cross_axis = np.asarray((-axis[1], axis[0]), dtype=float)
    along = centered @ axis
    across = centered @ cross_axis
    columns = _one_dimensional_groups(along, 3)
    rows = _one_dimensional_groups(across, 4)
    if len(columns) != 4 or len(rows) != 3:
        return 0.0
    column_gap = np.diff(columns)
    row_gap = np.diff(rows)
    column_step = max(float(np.mean(np.abs(column_gap))), 1.0)
    row_step = max(float(np.mean(np.abs(row_gap))), 1.0)
    column_ids = np.argmin(np.abs(along[:, None] - columns[None, :]), axis=1)
    row_ids = np.argmin(np.abs(across[:, None] - rows[None, :]), axis=1)
    residual = float(
        np.sqrt(np.mean((along - columns[column_ids]) ** 2)) / column_step
        + np.sqrt(np.mean((across - rows[row_ids]) ** 2)) / row_step
    )
    spacing_error = float(
        np.std(np.abs(column_gap)) / column_step
        + np.std(np.abs(row_gap)) / row_step
    )
    occupied = len({(int(x), int(y)) for x, y in zip(column_ids, row_ids)}) / 12.0
    return float(occupied * math.exp(-2.8 * residual - 1.8 * spacing_error))


def decision_from_image(image: Image.Image) -> str:
    """Return only a palette label inferred from the final RGB raster."""

    rgb = np.asarray(image.convert("RGB"))
    scores = []
    for name, target in _COLORS.items():
        points = _dot_centers(rgb, target)
        scores.append((_regularity(points), name))
    scores.sort(key=lambda item: (-item[0], ("red", "blue", "green").index(item[1])))
    if not scores or scores[0][0] <= 0.0:
        return "ambiguous"
    return scores[0][1]
