"""Independent image-only oracle for paired constellation handedness."""

from __future__ import annotations

import itertools

import numpy as np
from PIL import Image


POINT_COUNT = 6
PERMUTATIONS = tuple(itertools.permutations(range(POINT_COUNT)))


def _ink_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    distance_from_white = np.max(255 - array, axis=2)
    return distance_from_white > 24


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    found: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        iy, ix = int(y), int(x)
        if seen[iy, ix]:
            continue
        stack = [(iy, ix)]
        seen[iy, ix] = True
        points: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        found.append(np.asarray(points, dtype=np.int32))
    return found


def _centers(mask: np.ndarray) -> np.ndarray:
    components = [item for item in _components(mask) if len(item) >= 80]
    if len(components) != 12:
        raise ValueError("expected twelve intact dot components")
    measurements = []
    for component in components:
        yx = component.astype(np.float64)
        area = float(len(component))
        if area < 80.0 or area > 600.0:
            raise ValueError("dot component has implausible raster area")
        measurements.append([float(yx[:, 1].mean()), float(yx[:, 0].mean())])
    return np.asarray(measurements, dtype=np.float64)


def _normalize(points: np.ndarray) -> np.ndarray:
    centered = points - points.mean(axis=0, keepdims=True)
    scale = float(np.sqrt(np.mean(np.sum(centered * centered, axis=1))))
    if scale < 2.0:
        raise ValueError("constellation extent is too small")
    return centered / scale


def _fit_residual(points_a: np.ndarray, points_b: np.ndarray, parity: int) -> float:
    matrix = points_a.T @ points_b
    u, _singular, vt = np.linalg.svd(matrix)
    uv_det = float(np.linalg.det(u @ vt))
    correction = np.diag((1.0, float(parity) * (1.0 if uv_det >= 0.0 else -1.0)))
    transform = u @ correction @ vt
    residual = points_a @ transform - points_b
    return float(np.sqrt(np.mean(np.sum(residual * residual, axis=1))))


def _registration(points_a: np.ndarray, points_b: np.ndarray) -> tuple[float, float]:
    a = _normalize(points_a)
    b = _normalize(points_b)
    proper = float("inf")
    mirror = float("inf")
    for permutation in PERMUTATIONS:
        candidate = b[list(permutation)]
        proper = min(proper, _fit_residual(a, candidate, 1))
        mirror = min(mirror, _fit_residual(a, candidate, -1))
    return proper, mirror


def decision_from_image(image: Image.Image) -> str:
    """Segment and register the two distant point sets using pixels only."""
    centers = _centers(_ink_mask(image))
    order = np.argsort(centers[:, 0], kind="mergesort")
    first = centers[order[:POINT_COUNT]]
    second = centers[order[POINT_COUNT:]]
    if float(first[:, 0].max()) + 90.0 >= float(second[:, 0].min()):
        raise ValueError("the two constellation islands are not separately spaced")
    proper, mirror = _registration(first, second)
    gap = abs(proper - mirror)
    if gap < 0.08:
        raise ValueError("proper and mirror registrations are too close")
    return "yes" if proper < mirror else "no"
