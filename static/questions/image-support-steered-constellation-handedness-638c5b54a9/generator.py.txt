"""Deterministic raster world for paired constellation handedness."""

from __future__ import annotations

import itertools
import math
import random
from typing import Any

import numpy as np
from PIL import Image, ImageDraw


CANVAS = (640, 384)
SUPERSAMPLE = 4
POINT_COUNT = 6
PERMUTATIONS = tuple(itertools.permutations(range(POINT_COUNT)))
PALETTE = ((38, 73, 96), (92, 57, 76), (42, 96, 76), (79, 67, 112), (118, 79, 35))


def _normalize(points: np.ndarray) -> np.ndarray:
    centered = points.astype(np.float64) - points.astype(np.float64).mean(axis=0, keepdims=True)
    scale = float(np.sqrt(np.mean(np.sum(centered * centered, axis=1))))
    if scale < 1e-9:
        raise ValueError("degenerate constellation")
    return centered / scale


def _fit_residual(points_a: np.ndarray, points_b: np.ndarray, parity: int) -> float:
    """Return normalized RMS residual for an orthogonal map of requested parity."""
    matrix = points_a.T @ points_b
    u, _singular, vt = np.linalg.svd(matrix)
    uv_det = float(np.linalg.det(u @ vt))
    correction = np.diag((1.0, float(parity) * (1.0 if uv_det >= 0.0 else -1.0)))
    transform = u @ correction @ vt
    residual = points_a @ transform - points_b
    return float(np.sqrt(np.mean(np.sum(residual * residual, axis=1))))


def _registration_residuals(points_a: np.ndarray, points_b: np.ndarray) -> tuple[float, float]:
    """Find the best proper and improper registrations over point correspondences."""
    a = _normalize(points_a)
    b = _normalize(points_b)
    proper = float("inf")
    mirror = float("inf")
    for permutation in PERMUTATIONS:
        candidate = b[list(permutation)]
        proper = min(proper, _fit_residual(a, candidate, 1))
        mirror = min(mirror, _fit_residual(a, candidate, -1))
    return proper, mirror


def _template_gap(template: list[list[float]]) -> float:
    points = np.asarray(template, dtype=np.float64)
    proper, mirror = _registration_residuals(points, points)
    return abs(mirror - proper)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample balanced, visibly asymmetric paired constellations."""
    rng = random.Random(int(seed))
    relation = "same" if int(seed) % 2 == 0 else "mirror"
    template: list[list[float]] | None = None
    for _attempt in range(160):
        candidate = [[rng.uniform(-1.08, 1.08), rng.uniform(-1.08, 1.08)] for _ in range(POINT_COUNT)]
        distances = [
            math.hypot(candidate[i][0] - candidate[j][0], candidate[i][1] - candidate[j][1])
            for i in range(POINT_COUNT)
            for j in range(i)
        ]
        if min(distances) < 0.52:
            continue
        gap = _template_gap(candidate)
        if gap >= 0.18:
            template = [[round(float(x), 6), round(float(y), 6)] for x, y in candidate]
            break
    if template is None:
        raise RuntimeError("could not sample an asymmetric constellation")

    # Keep both complete islands well inside the canvas while making their
    # jointly necessary support cross the outer QD grid cells.
    left = (rng.uniform(108.0, 132.0), rng.uniform(132.0, 252.0))
    right = (rng.uniform(508.0, 532.0), rng.uniform(132.0, 252.0))
    if rng.random() < 0.5:
        center_a, center_b = left, right
    else:
        center_a, center_b = right, left
    return {
        "relation": relation,
        "template": template,
        "center_a": [round(float(center_a[0]), 3), round(float(center_a[1]), 3)],
        "center_b": [round(float(center_b[0]), 3), round(float(center_b[1]), 3)],
        "scale_a": round(rng.uniform(47.0, 59.0), 3),
        "scale_b": round(rng.uniform(47.0, 59.0), 3),
        "angle_a": round(rng.uniform(-math.pi, math.pi), 6),
        "angle_b": round(rng.uniform(-math.pi, math.pi), 6),
        "radius": round(rng.uniform(8.0, 10.5), 3),
        "color": list(rng.choice(PALETTE)),
    }


def _cluster_points(scene: dict[str, Any], label: str) -> list[tuple[float, float]]:
    center = scene["center_a"] if label == "a" else scene["center_b"]
    scale = float(scene["scale_a"] if label == "a" else scene["scale_b"])
    angle = float(scene["angle_a"] if label == "a" else scene["angle_b"])
    mirror = label == "b" and scene["relation"] == "mirror"
    cosine = math.cos(angle)
    sine = math.sin(angle)
    result = []
    for raw_x, raw_y in scene["template"]:
        x, y = float(raw_x), float(raw_y)
        if mirror:
            y = -y
        rotated_x = cosine * x - sine * y
        rotated_y = sine * x + cosine * y
        result.append((float(center[0]) + scale * rotated_x, float(center[1]) + scale * rotated_y))
    return result


def render(scene: dict[str, Any]) -> Image.Image:
    """Render two far-separated, disconnected, same-colored dot sets."""
    width, height = CANVAS
    ss = SUPERSAMPLE
    image = Image.new("RGB", (width * ss, height * ss), "white")
    draw = ImageDraw.Draw(image)
    color = tuple(int(value) for value in scene["color"])
    radius = float(scene["radius"]) * ss
    for label in ("a", "b"):
        for x, y in _cluster_points(scene, label):
            cx, cy = x * ss, y * ss
            draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=color)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    """The visible relation is direct (same) versus reflected (mirror)."""
    return "yes" if scene["relation"] == "same" else "no"


def margin(scene: dict[str, Any]) -> float:
    """Distance between the proper and mirror registrations of the template."""
    return round(_template_gap(scene["template"]), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Exclude finite-resolution constellations with a weak handedness gap."""
    return margin(scene) < 0.08


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict]]:
    """No latent aliases are claimed for this injective scene parametrization."""
    return []
