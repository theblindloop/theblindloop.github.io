"""Analytic scene sampler and pixel renderer for Seeded Flood-Fill Reveal."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 768
HEIGHT = 768
GRID = 7
BORDER = 4
WALL_HALF = 2
TARGETS = ("plus", "tee", "arch")
SOURCE = (1, 3)

PALETTES = (
    {"background": (247, 244, 236), "cell": (225, 236, 239), "wall": (24, 38, 57)},
    {"background": (241, 246, 242), "cell": (236, 229, 218), "wall": (49, 29, 56)},
    {"background": (246, 242, 248), "cell": (221, 236, 225), "wall": (24, 54, 47)},
)
GOLD = (244, 174, 38)
DECOR = (166, 174, 181)


def _template(target: str) -> set[tuple[int, int]]:
    if target == "plus":
        return {(3, c) for c in range(7)} | {(r, 3) for r in range(7)}
    if target == "tee":
        return {(1, c) for c in range(7)} | {(r, 3) for r in range(7)}
    if target == "arch":
        return {(1, c) for c in range(1, 6)} | {
            (r, c) for r in range(2, 6) for c in (1, 5)
        }
    raise ValueError(f"unknown target {target!r}")


def _reachable_pattern(scene: dict) -> set[tuple[int, int]]:
    """Return the flooded silhouette, including an explicit boundary approach.

    Two outer plus-arm cells can move to the tee row.  This keeps thirteen
    connected cells but reduces the plus-versus-tee runner-up gap from 12 to 4.
    A third move (accepted for counterfactual probes, though not sampled) lands
    exactly on the visible tie and is therefore quarantined.
    """
    target = scene["target"]
    step = int(scene["blend_step"])
    cells = set(_template(target))
    if step:
        if target != "plus" or step not in (1, 2, 3):
            raise ValueError("blend_step is defined only for the plus boundary path")
        moves = (((3, 0), (1, 2)), ((3, 6), (1, 4)), ((3, 1), (1, 1)))
        for old, new in moves[:step]:
            cells.remove(old)
            cells.add(new)
    if len(cells) != 13 or SOURCE not in cells:
        raise AssertionError("reachable pattern contract changed")
    return cells


def _tree_edges(scene: dict) -> set[tuple[tuple[int, int], tuple[int, int]]]:
    """Return a fixed DFS spanning tree over the target's thirteen cells."""
    target = scene["target"]
    cells = _reachable_pattern(scene)
    order = {
        "plus": ((1, 0), (0, -1), (-1, 0), (0, 1)),
        "tee": ((0, -1), (0, 1), (1, 0), (-1, 0)),
        "arch": ((0, 1), (1, 0), (0, -1), (-1, 0)),
    }[target]
    visited = {SOURCE}
    stack = [SOURCE]
    edges: set[tuple[tuple[int, int], tuple[int, int]]] = set()
    while stack:
        current = stack[-1]
        unseen = []
        for dr, dc in order:
            nxt = (current[0] + dr, current[1] + dc)
            if nxt in cells and nxt not in visited:
                unseen.append(nxt)
        if not unseen:
            stack.pop()
            continue
        nxt = unseen[0]
        edge = tuple(sorted((current, nxt)))
        edges.add(edge)  # type: ignore[arg-type]
        visited.add(nxt)
        stack.append(nxt)
    if visited != cells or len(edges) != 12:
        raise AssertionError("template did not yield a spanning tree")
    return edges


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = TARGETS[int(seed) % len(TARGETS)]
    cell_size = rng.choice((14, 15, 16))
    panel_size = GRID * cell_size + 2 * BORDER
    panel_x = rng.randint(54, WIDTH - panel_size - 54)
    panel_y = rng.randint(54, HEIGHT - panel_size - 54)
    return {
        "target": target,
        "blend_step": 2 if target == "plus" and int(seed) % 18 == 9 else 0,
        "panel_x": panel_x,
        "panel_y": panel_y,
        "cell_size": cell_size,
        "palette_id": rng.randrange(len(PALETTES)),
        "decor_code": rng.randrange(8),
    }


def _draw_decorations(draw: ImageDraw.ImageDraw, code: int, panel_box: tuple[int, int, int, int]) -> None:
    """Draw answer-independent mid-gray marks well outside the focal panel."""
    x0, y0, x1, y1 = panel_box
    anchors = ((34, 50), (710, 48), (38, 704), (706, 708), (382, 35), (35, 382), (730, 382), (382, 730))
    for index in range(4):
        ax, ay = anchors[(code + index * 2) % len(anchors)]
        if x0 - 28 <= ax <= x1 + 28 and y0 - 28 <= ay <= y1 + 28:
            continue
        if (code + index) % 3 == 0:
            draw.ellipse((ax - 8, ay - 8, ax + 8, ay + 8), outline=DECOR, width=3)
        elif (code + index) % 3 == 1:
            draw.rectangle((ax - 7, ay - 7, ax + 7, ay + 7), outline=DECOR, width=3)
        else:
            draw.line((ax - 9, ay + 7, ax, ay - 8, ax + 9, ay + 7), fill=DECOR, width=3, joint="curve")


def render(scene: dict) -> Image.Image:
    target = scene["target"]
    blend_step = int(scene["blend_step"])
    panel_x = int(scene["panel_x"])
    panel_y = int(scene["panel_y"])
    cell = int(scene["cell_size"])
    palette = PALETTES[int(scene["palette_id"])]
    decor_code = int(scene["decor_code"])
    if target not in TARGETS or cell not in (14, 15, 16) or blend_step not in (0, 1, 2, 3):
        raise ValueError("unsupported scene")
    size = GRID * cell + 2 * BORDER
    panel_box = (panel_x, panel_y, panel_x + size - 1, panel_y + size - 1)
    image = Image.new("RGB", (WIDTH, HEIGHT), palette["background"])
    draw = ImageDraw.Draw(image)
    _draw_decorations(draw, decor_code, panel_box)

    # Start from a solid square and carve 49 pale cell interiors.
    draw.rectangle(panel_box, fill=palette["wall"])
    for row in range(GRID):
        for col in range(GRID):
            left = panel_x + BORDER + col * cell + WALL_HALF
            top = panel_y + BORDER + row * cell + WALL_HALF
            right = panel_x + BORDER + (col + 1) * cell - WALL_HALF - 1
            bottom = panel_y + BORDER + (row + 1) * cell - WALL_HALF - 1
            draw.rectangle((left, top, right, bottom), fill=palette["cell"])

    # Carve exactly twelve visible doorways: a spanning tree of the target.
    for a, b in _tree_edges(scene):
        ar, ac = a
        br, bc = b
        if ar == br:
            boundary = panel_x + BORDER + max(ac, bc) * cell
            cy = panel_y + BORDER + ar * cell + cell // 2
            draw.rectangle(
                (boundary - WALL_HALF, cy - cell // 4, boundary + WALL_HALF - 1, cy + cell // 4),
                fill=palette["cell"],
            )
        else:
            boundary = panel_y + BORDER + max(ar, br) * cell
            cx = panel_x + BORDER + ac * cell + cell // 2
            draw.rectangle(
                (cx - cell // 4, boundary - WALL_HALF, cx + cell // 4, boundary + WALL_HALF - 1),
                fill=palette["cell"],
            )

    sr, sc = SOURCE
    left = panel_x + BORDER + sc * cell + WALL_HALF + 1
    top = panel_y + BORDER + sr * cell + WALL_HALF + 1
    right = panel_x + BORDER + (sc + 1) * cell - WALL_HALF - 2
    bottom = panel_y + BORDER + (sr + 1) * cell - WALL_HALF - 2
    draw.rectangle((left, top, right, bottom), fill=GOLD)
    return image


def analytic_gold(scene: dict) -> str:
    distances = _template_distances(scene)
    if distances[1][0] - distances[0][0] <= 2:
        return "ambiguous"
    return distances[0][1]


def _template_distances(scene: dict) -> list[tuple[int, str]]:
    mask = _reachable_pattern(scene)
    return sorted((len(mask.symmetric_difference(_template(other))), other) for other in TARGETS)


def margin(scene: dict) -> float:
    distances = _template_distances(scene)
    return float(distances[1][0] - distances[0][0])


def is_quarantined(scene: dict) -> bool:
    return analytic_gold(scene) == "ambiguous" or margin(scene) <= 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
