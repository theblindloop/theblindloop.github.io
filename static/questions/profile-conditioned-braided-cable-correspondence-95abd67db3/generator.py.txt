"""Renderer and analytic arm for patterned-plug cable correspondence.

The reference contributes only a local 3x3 plug-face identity.  The matching
face in the cable bank starts one of four visually identical dark cables.  Five
banded layers contain straight vertical runs or explicit X crossovers; at an X
each cable continues along its diagonal, with a small background-colored bridge
gap making the crossing convention visible.  The reached socket's fill is the
decision.  Every plug has four black cells and every braid has four swaps.
"""

from __future__ import annotations

import itertools
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

WIDTH = 840
HEIGHT = 700
SUPERSAMPLE = 2
COLOR_NAMES = ("red", "green", "blue", "orange")
PAIR_LAYOUT = (((0, 1), (2, 3)), ((1, 2),), ((0, 1), (2, 3)), ((1, 2),), ((0, 1), (2, 3)))
LANE_X = (275.0, 440.0, 605.0, 770.0)
STAGE_TOP = 215.0
STAGE_HEIGHT = 64.0
PLUG_Y = 126.0
SOCKET_Y = 615.0

CANVAS = (245, 242, 236)
CARD = (255, 255, 253)
PLUG = (205, 217, 226)
EMPTY_CELL = (250, 250, 247)
INK = (38, 43, 49)
CABLE = (65, 73, 82)
OUTLINE = (132, 143, 151)
MAGENTA = (207, 43, 139)
STAGE_COLORS = ((232, 240, 247), (243, 234, 219))

PALETTES = (
    {"red": (210, 55, 65), "green": (38, 157, 91), "blue": (44, 106, 211), "orange": (234, 139, 30)},
    {"red": (195, 65, 79), "green": (31, 145, 104), "blue": (57, 112, 195), "orange": (220, 128, 43)},
    {"red": (225, 70, 50), "green": (54, 159, 77), "blue": (47, 119, 219), "orange": (236, 151, 45)},
)


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    filename = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    path = Path("/usr/share/fonts/truetype/dejavu") / filename
    try:
        return ImageFont.truetype(str(path), size=size)
    except OSError:
        return ImageFont.load_default()


def _pattern_key(pattern: list[list[int]]) -> frozenset[tuple[int, int]]:
    return frozenset((int(row), int(column)) for row, column in pattern)


def _apply_routes(routes: list[list[int]], start: int) -> int:
    position = int(start)
    for layer in routes:
        for left in layer:
            left = int(left)
            if position == left:
                position = left + 1
            elif position == left + 1:
                position = left
    return position


def _route_options(start: int, end: int) -> list[list[list[int]]]:
    options: list[list[list[int]]] = []
    bit_count = sum(len(pairs) for pairs in PAIR_LAYOUT)
    for bits in itertools.product((0, 1), repeat=bit_count):
        if sum(bits) != 4:
            continue
        cursor = 0
        routes: list[list[int]] = []
        for pairs in PAIR_LAYOUT:
            layer: list[int] = []
            for left, _ in pairs:
                if bits[cursor]:
                    layer.append(left)
                cursor += 1
            routes.append(layer)
        if _apply_routes(routes, start) == end:
            options.append(routes)
    return options


def _sample_patterns(rng: random.Random) -> tuple[list[list[int]], list[list[list[int]]], int]:
    cells = [(row, column) for row in range(3) for column in range(3)]
    reference = sorted(rng.sample(cells, 4))
    ref_set = set(reference)
    distractors: list[list[tuple[int, int]]] = []
    candidates = list(itertools.combinations(cells, 4))
    rng.shuffle(candidates)
    for candidate in candidates:
        chosen = set(candidate)
        if len(ref_set ^ chosen) < 2:
            continue
        if any(len(chosen ^ set(existing)) < 2 for existing in distractors):
            continue
        distractors.append(list(candidate))
        if len(distractors) == 3:
            break
    if len(distractors) != 3:
        raise RuntimeError("could not sample separated plug patterns")
    target_index = rng.randrange(4)
    patterns: list[list[tuple[int, int]]] = []
    iterator = iter(distractors)
    for index in range(4):
        patterns.append(list(reference) if index == target_index else next(iterator))
    return (
        [[int(row), int(column)] for row, column in reference],
        [[[int(row), int(column)] for row, column in pattern] for pattern in patterns],
        target_index,
    )


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0xB4A1D)
    reference, plug_patterns, target_index = _sample_patterns(rng)
    desired_end = rng.randrange(4)
    options = _route_options(target_index, desired_end)
    if not options:
        raise RuntimeError("crossover network cannot reach requested endpoint")
    routes = rng.choice(options)

    answer = COLOR_NAMES[int(seed) % 4]
    remaining_colors = [name for name in COLOR_NAMES if name != answer]
    rng.shuffle(remaining_colors)
    socket_colors: list[str | None] = [None, None, None, None]
    socket_colors[desired_end] = answer
    iterator = iter(remaining_colors)
    for index in range(4):
        if socket_colors[index] is None:
            socket_colors[index] = next(iterator)

    scene = {
        "width": WIDTH,
        "height": HEIGHT,
        "reference_pattern": reference,
        "plug_patterns": plug_patterns,
        "routes": routes,
        "socket_colors": [str(name) for name in socket_colors],
        "palette": {name: list(rgb) for name, rgb in rng.choice(PALETTES).items()},
        "cable_width": rng.choice((7, 8, 9)),
        "offset_x": rng.randint(-3, 3),
        "offset_y": rng.randint(-3, 3),
    }
    if is_quarantined(scene):
        raise RuntimeError("internal sampler produced a quarantined scene")
    return scene


def analytic_gold(scene: dict) -> str:
    reference_pattern = scene["reference_pattern"]
    plug_patterns = scene["plug_patterns"]
    routes = scene["routes"]
    socket_colors = scene["socket_colors"]
    reference = _pattern_key(reference_pattern)
    matches = [index for index, pattern in enumerate(plug_patterns) if _pattern_key(pattern) == reference]
    if len(matches) != 1:
        raise ValueError("reference pattern must match exactly one cable-bank plug")
    endpoint = _apply_routes(routes, matches[0])
    return str(socket_colors[endpoint])


def margin(scene: dict) -> float:
    reference = _pattern_key(scene["reference_pattern"])
    distances = sorted(len(reference ^ _pattern_key(pattern)) for pattern in scene["plug_patterns"])
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    try:
        reference = _pattern_key(scene["reference_pattern"])
        distances = sorted(len(reference ^ _pattern_key(pattern)) for pattern in scene["plug_patterns"])
        if len(distances) != 4 or distances[0] != 0 or distances[1] < 2:
            return True
        routes = scene["routes"]
        if len(routes) != 5 or sum(len(layer) for layer in routes) != 4:
            return True
        for layer, allowed in zip(routes, PAIR_LAYOUT, strict=True):
            allowed_left = {left for left, _ in allowed}
            if len(layer) != len(set(layer)) or not set(layer) <= allowed_left:
                return True
        endpoints = {_apply_routes(routes, start) for start in range(4)}
        return endpoints != {0, 1, 2, 3} or set(scene["socket_colors"]) != set(COLOR_NAMES)
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["reference_pattern"] = list(reversed(scene["reference_pattern"]))
    twin["plug_patterns"] = [list(reversed(pattern)) for pattern in scene["plug_patterns"]]
    twin["routes"] = [list(reversed(layer)) for layer in scene["routes"]]
    return [("reverse_unordered_cell_and_disjoint_swap_records", twin)]


def _draw_plug(draw: ImageDraw.ImageDraw, center: tuple[float, float], pattern: list[list[int]], scale: int, *, marked: bool = False) -> None:
    cx, cy = center
    def box(values: tuple[float, float, float, float]) -> tuple[int, int, int, int]:
        return tuple(int(round(value * scale)) for value in values)
    if marked:
        draw.rounded_rectangle(box((cx - 48, cy - 48, cx + 48, cy + 48)), radius=12 * scale, outline=MAGENTA, width=5 * scale)
    draw.rounded_rectangle(box((cx - 42, cy - 42, cx + 42, cy + 42)), radius=10 * scale, fill=PLUG, outline=OUTLINE, width=2 * scale)
    occupied = _pattern_key(pattern)
    for row in range(3):
        for column in range(3):
            x = cx + (column - 1) * 18
            y = cy + (row - 1) * 18
            fill = INK if (row, column) in occupied else EMPTY_CELL
            draw.rectangle(box((x - 7, y - 7, x + 7, y + 7)), fill=fill, outline=(155, 163, 168), width=1 * scale)


def render(scene: dict) -> Image.Image:
    scale = SUPERSAMPLE
    width, height = int(scene["width"]), int(scene["height"])
    ox, oy = float(scene.get("offset_x", 0)), float(scene.get("offset_y", 0))
    lane_x = [x + ox for x in LANE_X]
    image = Image.new("RGB", (width * scale, height * scale), CANVAS)
    draw = ImageDraw.Draw(image)

    def box(values: tuple[float, float, float, float]) -> tuple[int, int, int, int]:
        return tuple(int(round(value * scale)) for value in values)
    def point(values: tuple[float, float]) -> tuple[int, int]:
        return int(round(values[0] * scale)), int(round(values[1] * scale))

    draw.rounded_rectangle(box((18 + ox, 24 + oy, 207 + ox, 238 + oy)), radius=15 * scale, fill=CARD, outline=(187, 191, 193), width=2 * scale)
    draw.rounded_rectangle(box((220 + ox, 24 + oy, 826 + ox, 680 + oy)), radius=15 * scale, fill=CARD, outline=(187, 191, 193), width=2 * scale)
    heading = _font(15 * scale, bold=True)
    small = _font(12 * scale)
    tiny = _font(10 * scale, bold=True)
    draw.text(point((37 + ox, 43 + oy)), "REFERENCE", fill=(55, 60, 66), font=heading)
    draw.text(point((35 + ox, 204 + oy)), "match this exact face", fill=(86, 91, 96), font=small)
    draw.text(point((242 + ox, 43 + oy)), "CABLE BANK", fill=(55, 60, 66), font=heading)
    draw.text(point((385 + ox, 45 + oy)), "match face  •  trace straight through each X", fill=(82, 88, 94), font=small)

    stage_top = STAGE_TOP + oy
    for layer in range(5):
        y0 = stage_top + layer * STAGE_HEIGHT
        y1 = y0 + STAGE_HEIGHT
        fill = STAGE_COLORS[layer % 2]
        draw.rectangle(box((244 + ox, y0, 806 + ox, y1)), fill=fill)
        draw.text(point((249 + ox, y0 + 5)), str(layer + 1), fill=(118, 124, 130), font=tiny)

    cable_width = int(scene["cable_width"]) * scale
    routes = [set(int(value) for value in layer) for layer in scene["routes"]]
    for layer in range(5):
        y0 = stage_top + layer * STAGE_HEIGHT
        y1 = y0 + STAGE_HEIGHT
        destinations = list(range(4))
        for left in routes[layer]:
            destinations[left], destinations[left + 1] = destinations[left + 1], destinations[left]
        segments = [(lane_x[start], lane_x[destinations[start]]) for start in range(4)]
        for start_x, end_x in segments:
            draw.line((point((start_x, y0)), point((end_x, y1))), fill=CABLE, width=cable_width)
        for left in routes[layer]:
            x0, x1 = lane_x[left], lane_x[left + 1]
            cx, cy = (x0 + x1) / 2.0, (y0 + y1) / 2.0
            # A short halo and redraw make one cable visibly bridge the other.
            dx, dy = (x1 - x0) * 0.12, (y1 - y0) * 0.12
            fill = STAGE_COLORS[layer % 2]
            draw.line((point((cx - dx, cy - dy)), point((cx + dx, cy + dy))), fill=fill, width=cable_width + 8 * scale)
            draw.line((point((cx - dx, cy - dy)), point((cx + dx, cy + dy))), fill=CABLE, width=cable_width)

    for x in lane_x:
        draw.line((point((x, PLUG_Y + 42 + oy)), point((x, stage_top))), fill=CABLE, width=cable_width)
        draw.line((point((x, stage_top + 5 * STAGE_HEIGHT)), point((x, SOCKET_Y - 30 + oy))), fill=CABLE, width=cable_width)

    _draw_plug(draw, (112 + ox, 130 + oy), scene["reference_pattern"], scale, marked=True)
    for x, pattern in zip(lane_x, scene["plug_patterns"], strict=True):
        _draw_plug(draw, (x, PLUG_Y + oy), pattern, scale)

    palette = {name: tuple(int(channel) for channel in rgb) for name, rgb in scene["palette"].items()}
    socket_font = _font(11 * scale, bold=True)
    draw.text(point((242 + ox, 555 + oy)), "SOCKETS", fill=(82, 88, 94), font=socket_font)
    for x, color_name in zip(lane_x, scene["socket_colors"], strict=True):
        draw.ellipse(box((x - 30, SOCKET_Y - 30 + oy, x + 30, SOCKET_Y + 30 + oy)), fill=(239, 240, 238), outline=OUTLINE, width=3 * scale)
        draw.ellipse(box((x - 22, SOCKET_Y - 22 + oy, x + 22, SOCKET_Y + 22 + oy)), fill=palette[str(color_name)], outline=(64, 68, 72), width=2 * scale)

    return image.resize((width, height), Image.Resampling.LANCZOS)
