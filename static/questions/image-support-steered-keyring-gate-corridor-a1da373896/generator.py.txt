"""Renderer for the keyring_gate_corridor world.

Somewhere on the canvas sits one small lock made of a single thin corridor: an
unbranched, non-self-touching lattice route drawn with a constant stroke width.
A solid dark bead marks the end of the corridor where the walk starts.  Sitting
directly on the corridor are two kinds of coloured feature: a filled round key
disk, and a straight gate bar laid across the corridor.  Walking from the bead
along the corridor, every key disk that is met adds its colour to the ring of
keys already carried, and a gate bar can be passed only if a key of the gate's
own colour has already been met.  The decision is the colour of the first gate
that cannot be passed.

The decisive evidence is therefore the interior of one short thin connected
component: the corridor between the bead and the blocking gate, together with
the keys and gates lying on that stretch.  The order in which features are met
along the corridor is what decides the answer, so an endpoint alone is never
enough.  Everything past the blocking gate, and every grey mark outside the
lock, is inert.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "keyring_gate_corridor-renderer-0.1.0"

CANVAS = 448
BACKGROUND = (246, 246, 242)
INK = (28, 28, 34)
DECOR = (201, 201, 196)

KEY_COLOURS: dict[str, tuple[int, int, int]] = {
    "red": (198, 42, 42),
    "green": (32, 138, 66),
    "blue": (40, 84, 200),
    "purple": (142, 48, 168),
}
COLOUR_NAMES = ("red", "green", "blue", "purple")

GRID_SHAPES = ((3, 7), (7, 3))  # the route always lives in a narrow band
PITCH = 22
STROKE = 5           # corridor width in pixels (odd)
BEAD_R = 8           # start bead radius
KEY_R = 6            # key disk radius
GATE_HALF_LEN = 10   # gate bar reaches +-10px across the corridor
GATE_HALF_THICK = 1  # gate bar is 3px thick along the corridor

QUARANTINE_MARGIN = 14.0

_DIRS = ((1, 0), (-1, 0), (0, 1), (0, -1))


# ---------------------------------------------------------------- geometry ---
def _cell_px(scene: dict, cell: tuple[int, int]) -> tuple[int, int]:
    """Pixel centre of one lattice cell of the corridor."""
    return (scene["x0"] + PITCH * cell[0], scene["y0"] + PITCH * cell[1])


def _cells(scene: dict) -> list[tuple[int, int]]:
    return [tuple(cell) for cell in scene["cells"]]


def _features(scene: dict) -> list[dict]:
    """Features in the order in which the walk meets them."""
    return sorted(scene["features"], key=lambda f: f["at"])


def _along_extent(feature: dict) -> int:
    """Half-extent of a feature measured along the corridor, in pixels."""
    return KEY_R if feature["kind"] == "key" else GATE_HALF_THICK


def _replay(scene: dict) -> dict:
    """Walk the corridor from the bead and stop at the first unopened gate."""
    ring: list[str] = []
    for order, feature in enumerate(_features(scene)):
        if feature["kind"] == "key":
            if feature["colour"] not in ring:
                ring.append(feature["colour"])
        elif feature["colour"] not in ring:
            return {"blocked": feature["colour"], "order": order, "ring": list(ring)}
    return {"blocked": None, "order": None, "ring": ring}


def _bbox(scene: dict) -> tuple[int, int, int, int]:
    xs = [_cell_px(scene, cell)[0] for cell in _cells(scene)]
    ys = [_cell_px(scene, cell)[1] for cell in _cells(scene)]
    pad = max(BEAD_R, GATE_HALF_LEN) + 2
    return (min(xs) - pad, min(ys) - pad, max(xs) + pad, max(ys) + pad)


# ----------------------------------------------------------------- sampling ---
def _sample_path(rng: random.Random) -> list[tuple[int, int]] | None:
    """A self-avoiding, non-self-touching lattice route inside a narrow band."""
    cols, rows = rng.choice(GRID_SHAPES)
    target = rng.choice((10, 11, 12))
    cell = (rng.randrange(cols), rng.randrange(rows))
    path = [cell]
    used = {cell}
    for _ in range(target - 1):
        options = []
        for dx, dy in _DIRS:
            nxt = (path[-1][0] + dx, path[-1][1] + dy)
            if not (0 <= nxt[0] < cols and 0 <= nxt[1] < rows):
                continue
            if nxt in used:
                continue
            # non-self-touching: the new cell may only touch its predecessor
            touching = sum(
                1
                for ox, oy in _DIRS
                if (nxt[0] + ox, nxt[1] + oy) in used
            )
            if touching > 1:
                continue
            options.append(nxt)
        if not options:
            return None
        nxt = rng.choice(options)
        path.append(nxt)
        used.add(nxt)
    return path


def _straight_cells(path: list[tuple[int, int]]) -> list[int]:
    """Indices whose incoming and outgoing directions agree (skip cell 0/1)."""
    out = []
    for i in range(2, len(path) - 1):
        din = (path[i][0] - path[i - 1][0], path[i][1] - path[i - 1][1])
        dout = (path[i + 1][0] - path[i][0], path[i + 1][1] - path[i][1])
        if din == dout:
            out.append(i)
    return out


def _sample_sequence(rng: random.Random) -> tuple[list[dict], str]:
    """Feature sequence whose first unopened gate carries the sampled colour."""
    answer = rng.choice(COLOUR_NAMES)
    others = [c for c in COLOUR_NAMES if c != answer]
    rng.shuffle(others)
    passed = others[: rng.choice((0, 1, 1, 2))]

    head: list[dict] = []
    for colour in passed:
        head.append({"kind": "key", "colour": colour})
        head.append({"kind": "gate", "colour": colour})
    # decoy keys of colours whose gates never block the walk
    decoys = 1 if len(passed) == 2 else rng.choice((1, 1, 2))
    for _ in range(decoys):
        decoy = rng.choice(others)
        head.insert(rng.randrange(len(head) + 1), {"kind": "key", "colour": decoy})
    head.append({"kind": "gate", "colour": answer})

    # the blocking colour's own key exists, but only downstream of its gate
    tail: list[dict] = [{"kind": "key", "colour": answer}]
    # at least two gates are always painted, so the blocking one is a choice
    if not passed:
        tail.append({"kind": "gate", "colour": rng.choice(others)})
    elif rng.random() < 0.6:
        tail.append({"kind": rng.choice(("key", "gate")), "colour": rng.choice(others)})
    rng.shuffle(tail)
    return head + tail, answer


def sample_scene(seed: int) -> dict:
    """Sample one lock scene; every returned field is painted into the raster."""
    rng = random.Random(seed)
    for _ in range(400):
        path = _sample_path(rng)
        if path is None:
            continue
        # the route must stay a long thin band, not fill a compact block
        width = max(c[0] for c in path) - min(c[0] for c in path) + 1
        height = max(c[1] for c in path) - min(c[1] for c in path) + 1
        long_side, short_side = max(width, height), min(width, height)
        if long_side < 2 * short_side:
            continue
        sequence, answer = _sample_sequence(rng)
        slots = _straight_cells(path)
        if len(slots) < len(sequence):
            continue
        chosen = sorted(rng.sample(slots, len(sequence)))
        features = [
            {"at": chosen[i], "kind": item["kind"], "colour": item["colour"]}
            for i, item in enumerate(sequence)
        ]

        xs = [cell[0] for cell in path]
        ys = [cell[1] for cell in path]
        pad = max(BEAD_R, GATE_HALF_LEN) + 8
        span_x = (max(xs) - min(xs)) * PITCH
        span_y = (max(ys) - min(ys)) * PITCH
        x0 = rng.randrange(pad, CANVAS - span_x - pad) - min(xs) * PITCH
        y0 = rng.randrange(pad, CANVAS - span_y - pad) - min(ys) * PITCH

        scene = {
            "cells": [list(cell) for cell in path],
            "features": features,
            "x0": x0,
            "y0": y0,
            "decor": [],
        }
        if _replay(scene)["blocked"] != answer:
            continue
        if margin(scene) < QUARANTINE_MARGIN:
            continue
        scene["decor"] = _sample_decor(rng, scene)
        return scene
    raise RuntimeError(f"failed to sample a legal scene for seed {seed}")


def _sample_decor(rng: random.Random, scene: dict) -> list[dict]:
    """Inert grey marks, kept clear of the lock so they never touch it."""
    left, top, right, bottom = _bbox(scene)
    marks = []
    for _ in range(rng.randrange(4, 8)):
        for _try in range(80):
            w = rng.randrange(6, 22)
            h = rng.randrange(4, 18)
            x = rng.randrange(6, CANVAS - w - 6)
            y = rng.randrange(6, CANVAS - h - 6)
            if x + w + 14 < left or x - 14 > right or y + h + 14 < top or y - 14 > bottom:
                marks.append({"x": x, "y": y, "w": w, "h": h, "round": rng.random() < 0.4})
                break
    return marks


# ------------------------------------------------------------------ raster ---
def render(scene: dict) -> Image.Image:
    """Paint the scene; no anti-aliasing, so every colour class is exact."""
    image = Image.new("RGB", (CANVAS, CANVAS), BACKGROUND)
    draw = ImageDraw.Draw(image)

    for mark in scene["decor"]:
        box = [mark["x"], mark["y"], mark["x"] + mark["w"], mark["y"] + mark["h"]]
        if mark["round"]:
            draw.ellipse(box, fill=DECOR)
        else:
            draw.rectangle(box, fill=DECOR)

    cells = _cells(scene)
    half = STROKE // 2
    for a, b in zip(cells, cells[1:]):
        pa = _cell_px(scene, a)
        pb = _cell_px(scene, b)
        draw.rectangle(
            [
                min(pa[0], pb[0]) - half,
                min(pa[1], pb[1]) - half,
                max(pa[0], pb[0]) + half,
                max(pa[1], pb[1]) + half,
            ],
            fill=INK,
        )

    start = _cell_px(scene, cells[0])
    draw.ellipse(
        [start[0] - BEAD_R, start[1] - BEAD_R, start[0] + BEAD_R, start[1] + BEAD_R],
        fill=INK,
    )

    for feature in _features(scene):
        index = feature["at"]
        centre = _cell_px(scene, cells[index])
        colour = KEY_COLOURS[feature["colour"]]
        if feature["kind"] == "key":
            draw.ellipse(
                [
                    centre[0] - KEY_R,
                    centre[1] - KEY_R,
                    centre[0] + KEY_R,
                    centre[1] + KEY_R,
                ],
                fill=colour,
            )
        else:
            din = (
                cells[index][0] - cells[index - 1][0],
                cells[index][1] - cells[index - 1][1],
            )
            if din[0] != 0:  # corridor runs horizontally: bar stands vertically
                box = [
                    centre[0] - GATE_HALF_THICK,
                    centre[1] - GATE_HALF_LEN,
                    centre[0] + GATE_HALF_THICK,
                    centre[1] + GATE_HALF_LEN,
                ]
            else:
                box = [
                    centre[0] - GATE_HALF_LEN,
                    centre[1] - GATE_HALF_THICK,
                    centre[0] + GATE_HALF_LEN,
                    centre[1] + GATE_HALF_THICK,
                ]
            draw.rectangle(box, fill=colour)
    return image


# -------------------------------------------------------------------- gold ---
def analytic_gold(scene: dict) -> str:
    """Colour of the first gate the walk cannot pass."""
    blocked = _replay(scene)["blocked"]
    if blocked is None:
        raise ValueError("scene has no blocking gate")
    return blocked


def margin(scene: dict) -> float:
    """Smallest along-corridor clearance between consecutive painted marks."""
    features = _features(scene)
    gaps = []
    previous_index = 0
    previous_extent = BEAD_R
    for feature in features:
        span = (feature["at"] - previous_index) * PITCH
        gaps.append(float(span - previous_extent - _along_extent(feature)))
        previous_index = feature["at"]
        previous_extent = _along_extent(feature)
    return min(gaps) if gaps else 0.0


def is_quarantined(scene: dict) -> bool:
    """Marks closer than the quarantine clearance are illegible for the walk."""
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases: every sampled field is painted into the raster."""
    return []
