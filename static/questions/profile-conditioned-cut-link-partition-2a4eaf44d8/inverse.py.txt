"""Independent pixel-only inverse arm for `cut_link_partition`.

The module never sees a latent scene. It receives one rasterised PNG, maps every
pixel onto the public palette declared by the question contract, deletes the
violet link by dropping its pixels, labels the remaining ink with a run-based
union-find, and reports which of the three coloured dots end up carrying the
same ink label.

There is no graph model, no lattice, no link list and no geometry shared with
the renderer here: only colour distance, horizontal runs, and set membership.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "cut_link_partition-oracle-0.1.0"

# Public palette: white paper, grey wire, violet link, three dot colours.
PALETTE = (
    (255, 255, 255),
    (60, 64, 72),
    (150, 40, 190),
    (208, 32, 44),
    (36, 72, 200),
    (24, 132, 64),
)
BACKGROUND_INDEX = 0
WIRE_INDEX = 1
CUT_INDEX = 2
DOT_INDICES = {3: "red", 4: "blue", 5: "green"}
DOT_ORDER = ("red", "blue", "green")

BRIGHT_LEVEL = 218
MIN_DOT_PX = 240
MIN_INK_PX = 200


def _label_nearest(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    distances = np.stack(
        [((array - np.array(colour, dtype=np.float32)) ** 2).sum(axis=-1) for colour in PALETTE],
        axis=-1,
    )
    labels = distances.argmin(axis=-1).astype(np.int16)
    labels[array.min(axis=-1) >= BRIGHT_LEVEL] = BACKGROUND_INDEX
    return labels


def _components(mask: np.ndarray) -> np.ndarray:
    """Label 4-connected components of a boolean mask (0 means background)."""
    height, width = mask.shape
    runs: list[tuple[int, int, int]] = []
    row_runs: list[list[int]] = []
    parent: list[int] = [0]
    for y in range(height):
        row = mask[y].astype(np.int8)
        deltas = np.diff(np.concatenate(([np.int8(0)], row, [np.int8(0)])))
        starts = np.flatnonzero(deltas == 1)
        ends = np.flatnonzero(deltas == -1)
        indices = []
        for start, end in zip(starts.tolist(), ends.tolist()):
            parent.append(len(parent))
            runs.append((y, start, end))
            indices.append(len(runs))
        row_runs.append(indices)

    def find(node: int) -> int:
        while parent[node] != node:
            parent[node] = parent[parent[node]]
            node = parent[node]
        return node

    def union(left: int, right: int) -> None:
        left, right = find(left), find(right)
        if left != right:
            parent[max(left, right)] = min(left, right)

    for y in range(1, height):
        above = row_runs[y - 1]
        current = row_runs[y]
        if not above or not current:
            continue
        pointer = 0
        for index in current:
            _, start, end = runs[index - 1]
            while pointer < len(above) and runs[above[pointer] - 1][2] <= start:
                pointer += 1
            scan = pointer
            while scan < len(above) and runs[above[scan] - 1][1] < end:
                union(index, above[scan])
                scan += 1

    labels = np.zeros((height, width), dtype=np.int32)
    for index, (y, start, end) in enumerate(runs, start=1):
        labels[y, start:end] = find(index)
    return labels


def _largest_blob(mask: np.ndarray) -> np.ndarray | None:
    if not mask.any():
        return None
    labels = _components(mask)
    values, counts = np.unique(labels[mask], return_counts=True)
    best = int(values[counts.argmax()])
    if int(counts.max()) < MIN_DOT_PX:
        return None
    return labels == best


def decision_from_image(image: Image.Image) -> str:
    """Return the surviving-connection class from the final pixels alone."""
    labels = _label_nearest(image)
    dots: dict[str, np.ndarray] = {}
    for index, name in DOT_INDICES.items():
        blob = _largest_blob(labels == index)
        if blob is None:
            raise ValueError(f"no {name} dot of usable size was measured")
        dots[name] = blob

    ink = labels == WIRE_INDEX
    for blob in dots.values():
        ink |= blob
    ink_labels = _components(ink)
    sizes = np.bincount(ink_labels.reshape(-1))

    owners: dict[str, int] = {}
    for name in DOT_ORDER:
        values, counts = np.unique(ink_labels[dots[name]], return_counts=True)
        keep = [
            (int(value), int(count))
            for value, count in zip(values.tolist(), counts.tolist())
            if value and sizes[value] >= MIN_INK_PX
        ]
        if not keep:
            raise ValueError(f"the {name} dot sits on no measurable ink")
        keep.sort(key=lambda item: item[1], reverse=True)
        if len(keep) > 1 and keep[1][1] > 0.15 * keep[0][1]:
            raise ValueError(f"the {name} dot straddles several ink pieces")
        owners[name] = keep[0][0]

    together = [
        (first, second)
        for index, first in enumerate(DOT_ORDER)
        for second in DOT_ORDER[index + 1 :]
        if owners[first] == owners[second]
    ]
    if len(together) == 3:
        return "all_three"
    if len(together) != 1:
        raise ValueError("measured three mutually separated dots")
    first, second = together[0]
    return f"{first}_{second}"
