"""Independent pixel-only decision arm for the folded-cord bead world.

Input is one final RGB raster and nothing else. The arm segments the teal cord
ink and the three coloured bead squares, builds the connectivity of
cord-or-bead ink by flood filling, and reports the colour of the bead whose
connected cord is shared with neither of the other two beads. Any raster that
does not present exactly three single-blob beads in a clean two-plus-one
connectivity split raises, which is an abstention rather than a guess.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_CORD = (28, 82, 96)
_BEADS = {
    "red": (206, 42, 48),
    "blue": (44, 72, 208),
    "yellow": (232, 176, 20),
}
_TOL = 40.0
_MIN_BEAD_AREA = 100
_MAX_BEAD_AREA = 260


def _match(array: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    target = np.asarray(rgb, dtype=np.float64)
    return np.sqrt(((array - target) ** 2).sum(axis=2)) < _TOL


def _label(mask: np.ndarray) -> tuple[np.ndarray, int]:
    height, width = mask.shape
    labels = np.zeros((height, width), dtype=np.int32)
    count = 0
    for start_y in range(height):
        columns = np.nonzero(mask[start_y])[0]
        for raw_x in columns:
            start_x = int(raw_x)
            if labels[start_y, start_x]:
                continue
            count += 1
            queue = deque([(start_y, start_x)])
            labels[start_y, start_x] = count
            while queue:
                y, x = queue.popleft()
                for dy, dx in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < height and 0 <= nx < width:
                        if mask[ny, nx] and not labels[ny, nx]:
                            labels[ny, nx] = count
                            queue.append((ny, nx))
    return labels, count


def _bead_site(mask: np.ndarray, name: str) -> tuple[int, int]:
    area = int(mask.sum())
    if not area:
        raise ValueError(f"no {name} bead found")
    labels, count = _label(mask)
    sizes = [int((labels == index).sum()) for index in range(1, count + 1)]
    blobs = [size for size in sizes if size >= _MIN_BEAD_AREA]
    if len(blobs) != 1 or count != 1:
        raise ValueError(f"{name} bead is not a single blob")
    if not _MIN_BEAD_AREA <= area <= _MAX_BEAD_AREA:
        raise ValueError(f"{name} bead has an unexpected area {area}")
    ys, xs = np.nonzero(mask)
    height = int(ys.max() - ys.min() + 1)
    width = int(xs.max() - xs.min() + 1)
    if not (9 <= height <= 20 and 9 <= width <= 20) or area != height * width:
        raise ValueError(f"{name} bead is not a solid square")
    return int(round(ys.mean())), int(round(xs.mean()))


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.float64)
    if array.ndim != 3 or array.shape[2] != 3:
        raise ValueError("expected an RGB raster")

    cord = _match(array, _CORD)
    bead_masks = {name: _match(array, rgb) for name, rgb in _BEADS.items()}
    sites = {name: _bead_site(mask, name) for name, mask in bead_masks.items()}

    ink = cord.copy()
    for mask in bead_masks.values():
        ink |= mask
    if int(cord.sum()) < 4000:
        raise ValueError("too little cord ink to trace")

    labels, _ = _label(ink)
    owner = {}
    for name, (y, x) in sites.items():
        component = int(labels[y, x])
        if component == 0:
            raise ValueError(f"{name} bead is not attached to any ink")
        owner[name] = component

    names = sorted(owner)
    lone = [
        name
        for name in names
        if sum(1 for other in names if owner[other] == owner[name]) == 1
    ]
    shared = {owner[name] for name in names if name not in lone}
    if len(lone) != 1 or len(shared) != 1:
        raise ValueError(f"connectivity is not a two-plus-one split: {owner}")
    return lone[0]
