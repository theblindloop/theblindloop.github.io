"""Renderer and analytic gold for the folded-cord bead-partner world.

Every patch in a scene is one self-avoiding switchback stroke that has been cut
into three separate cords by two clear breaks, so the cords are interleaved but
can never cross. Exactly one patch carries three coloured beads printed on those
cords; two beads share one cord and the third sits alone. The analytic decision
is the colour of the lone bead.
"""

from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 560
HEIGHT = 560
BG = (255, 255, 255)
CORD = (28, 82, 96)
BEAD_COLORS = {
    "red": (206, 42, 48),
    "blue": (44, 72, 208),
    "yellow": (232, 176, 20),
}
COLOR_NAMES = ("red", "blue", "yellow")

CELL = 186
PATCH = 150
NROWS = 5
ROW_STEP = 30
INSET = 15
HW = 4  # cord half width; drawn thickness is 2 * HW + 1
BEAD_HW = 6  # bead half side; drawn bead is 2 * BEAD_HW + 1
LONG_LEN = PATCH - 2 * INSET
SEG_PERIOD = LONG_LEN + ROW_STEP
QUARANTINE_PX = 12.0


# --------------------------------------------------------------------------
# switchback geometry
# --------------------------------------------------------------------------
def _runs(patch: dict) -> tuple[list[dict], float]:
    """Return the ordered straight runs of one patch's switchback stroke."""

    segs = []
    left, right = INSET, INSET + LONG_LEN
    for index in range(NROWS):
        line = INSET + ROW_STEP * index
        if index % 2 == 0:
            segs.append((left, line, right, line))
        else:
            segs.append((right, line, left, line))
        if index < NROWS - 1:
            end = right if index % 2 == 0 else left
            segs.append((end, line, end, line + ROW_STEP))

    ox, oy = int(patch["ox"]), int(patch["oy"])
    runs: list[dict] = []
    cum = 0.0
    for x0, y0, x1, y1 in segs:
        if patch["orient"] == "v":
            x0, y0, x1, y1 = y0, x0, y1, x1
        p0 = (ox + x0, oy + y0)
        p1 = (ox + x1, oy + y1)
        length = float(abs(p1[0] - p0[0]) + abs(p1[1] - p0[1]))
        runs.append({"p0": p0, "p1": p1, "len": length, "cum": cum})
        cum += length
    return runs, cum


def _point_at(runs: list[dict], total: float, t: float) -> tuple[float, float]:
    t = min(max(float(t), 0.0), total)
    for run in runs:
        if t <= run["cum"] + run["len"] or run is runs[-1]:
            frac = (t - run["cum"]) / run["len"]
            frac = min(max(frac, 0.0), 1.0)
            x = run["p0"][0] + (run["p1"][0] - run["p0"][0]) * frac
            y = run["p0"][1] + (run["p1"][1] - run["p0"][1]) * frac
            return x, y
    raise AssertionError("unreachable")


def _arc_rects(runs: list[dict], total: float, t0: float, t1: float) -> list[tuple[int, int, int, int]]:
    rects = []
    for run in runs:
        lo = max(t0, run["cum"])
        hi = min(t1, run["cum"] + run["len"])
        if hi - lo <= 0.0:
            continue
        ax, ay = _point_at(runs, total, lo)
        bx, by = _point_at(runs, total, hi)
        rects.append(
            (
                int(round(min(ax, bx))) - HW,
                int(round(min(ay, by))) - HW,
                int(round(max(ax, bx))) + HW,
                int(round(max(ay, by))) + HW,
            )
        )
    return rects


def _arc_spans(patch: dict, total: float) -> list[tuple[float, float]]:
    (c1, g1), (c2, g2) = patch["cuts"]
    return [
        (0.0, c1 - g1 / 2.0),
        (c1 + g1 / 2.0, c2 - g2 / 2.0),
        (c2 + g2 / 2.0, total),
    ]


def _arc_samples(runs: list[dict], total: float, span: tuple[float, float]) -> list[tuple[float, float]]:
    lo, hi = span
    steps = max(2, int(hi - lo))
    return [_point_at(runs, total, lo + (hi - lo) * i / steps) for i in range(steps + 1)]


def bead_center(scene: dict, bead: dict) -> tuple[int, int]:
    runs, total = _runs(scene["main"])
    x, y = _point_at(runs, total, bead["t"])
    return int(round(x)), int(round(y))


def patch_rect(patch: dict) -> tuple[int, int, int, int]:
    ox, oy = int(patch["ox"]), int(patch["oy"])
    return ox, oy, ox + PATCH, oy + PATCH


def all_patches(scene: dict) -> list[dict]:
    return [scene["main"]] + list(scene["decoys"])


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------
def _sample_patch(rng: random.Random, cell: int, tight: bool) -> dict:
    row, col = divmod(cell, 3)
    ox = col * CELL + 18 + rng.randint(-6, 6)
    oy = row * CELL + 18 + rng.randint(-6, 6)
    first, second = sorted(rng.sample([1, 2, 3], 2))
    cuts = []
    for index, seg in enumerate((first, second)):
        centre = seg * SEG_PERIOD + 45.0 + rng.randint(0, 30)
        if tight and index == 0:
            gap = float(rng.randint(17, 21))
        else:
            gap = float(rng.randint(26, 40))
        cuts.append([round(centre, 1), gap])
    return {
        "cell": int(cell),
        "ox": int(ox),
        "oy": int(oy),
        "orient": "v" if rng.random() < 0.5 else "h",
        "cuts": cuts,
    }


def _safe_intervals(span: tuple[float, float]) -> list[tuple[float, float]]:
    lo, hi = span[0] + 30.0, span[1] - 30.0
    out = []
    for seg in range(NROWS):
        a = max(lo, seg * SEG_PERIOD + 20.0)
        b = min(hi, seg * SEG_PERIOD + LONG_LEN - 20.0)
        if b - a >= 1.0:
            out.append((a, b))
    return out


def _pick_t(rng: random.Random, intervals: list[tuple[float, float]]) -> float:
    weights = [b - a for a, b in intervals]
    total = sum(weights)
    draw = rng.random() * total
    for (a, b), weight in zip(intervals, weights):
        if draw <= weight:
            return round(a + (b - a) * (draw / weight), 1)
        draw -= weight
    a, b = intervals[-1]
    return round((a + b) / 2.0, 1)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) & 0x7FFFFFFF)
    tight = rng.random() < 0.18
    want_close_pair_shares_cord = rng.random() < 0.5

    for attempt in range(500):
        strict = attempt < 300
        cells = rng.sample(range(9), rng.choice([3, 3, 4]))
        main = _sample_patch(rng, cells[0], tight)
        decoys = [_sample_patch(rng, cell, False) for cell in cells[1:]]
        runs, total = _runs(main)
        spans = _arc_spans(main, total)

        pair_arc = rng.choice([0, 2])
        lone_arc = rng.choice([a for a in (0, 1, 2) if a != pair_arc])
        lone_color = rng.choice(COLOR_NAMES)
        pair_colors = [c for c in COLOR_NAMES if c != lone_color]
        rng.shuffle(pair_colors)

        pair_intervals = _safe_intervals(spans[pair_arc])
        lone_intervals = _safe_intervals(spans[lone_arc])
        if len(pair_intervals) < 2 or not lone_intervals:
            continue
        t_a = _pick_t(rng, pair_intervals)
        t_b = _pick_t(rng, pair_intervals)
        if abs(t_a - t_b) < 55.0:
            continue
        if int(t_a // SEG_PERIOD) == int(t_b // SEG_PERIOD):
            continue
        t_lone = _pick_t(rng, lone_intervals)

        beads = [
            {"color": pair_colors[0], "arc": pair_arc, "t": t_a},
            {"color": pair_colors[1], "arc": pair_arc, "t": t_b},
            {"color": lone_color, "arc": lone_arc, "t": t_lone},
        ]
        rng.shuffle(beads)
        scene = {
            "width": WIDTH,
            "height": HEIGHT,
            "main": main,
            "beads": beads,
            "decoys": decoys,
        }

        centers = [bead_center(scene, bead) for bead in beads]
        if min(
            max(abs(centers[i][0] - centers[j][0]), abs(centers[i][1] - centers[j][1]))
            for i in range(3)
            for j in range(i + 1, 3)
        ) < 26:
            continue

        best = None
        for i in range(3):
            for j in range(i + 1, 3):
                dist = math.dist(centers[i], centers[j])
                if best is None or dist < best[0]:
                    best = (dist, beads[i]["arc"] == beads[j]["arc"])
        if strict and best[1] != want_close_pair_shares_cord:
            continue
        if margin(scene) < 9.0:
            continue
        return scene

    raise RuntimeError(f"failed to sample a scene for seed {seed}")


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BG)
    draw = ImageDraw.Draw(image)
    for patch in all_patches(scene):
        runs, total = _runs(patch)
        for span in _arc_spans(patch, total):
            for rect in _arc_rects(runs, total, span[0], span[1]):
                draw.rectangle(rect, fill=CORD)
    for bead in scene["beads"]:
        cx, cy = bead_center(scene, bead)
        draw.rectangle(
            (cx - BEAD_HW, cy - BEAD_HW, cx + BEAD_HW, cy + BEAD_HW),
            fill=BEAD_COLORS[bead["color"]],
        )
    return image


# --------------------------------------------------------------------------
# gold, margin, quarantine, aliases
# --------------------------------------------------------------------------
def analytic_gold(scene: dict) -> str:
    arcs = [bead["arc"] for bead in scene["beads"]]
    lone = [bead["color"] for bead in scene["beads"] if arcs.count(bead["arc"]) == 1]
    if len(lone) != 1:
        raise ValueError("scene does not carry exactly one unshared bead")
    return lone[0]


def margin(scene: dict) -> float:
    main = scene["main"]
    runs, total = _runs(main)
    spans = _arc_spans(main, total)
    samples = [_arc_samples(runs, total, span) for span in spans]

    points = [np.asarray(group, dtype=np.float64) for group in samples]

    worst = float("inf")
    for i in range(3):
        for j in range(i + 1, 3):
            deltas = points[i][:, None, :] - points[j][None, :, :]
            worst = min(worst, float(np.hypot(deltas[..., 0], deltas[..., 1]).min()) - 2 * HW)

    centers = [bead_center(scene, bead) for bead in scene["beads"]]
    for bead, (cx, cy) in zip(scene["beads"], centers):
        for arc in (0, 1, 2):
            if arc == bead["arc"]:
                continue
            deltas = points[arc] - np.asarray((cx, cy), dtype=np.float64)
            worst = min(
                worst,
                float(np.hypot(deltas[:, 0], deltas[:, 1]).min()) - (BEAD_HW + HW),
            )
    for i in range(3):
        for j in range(i + 1, 3):
            cheb = max(
                abs(centers[i][0] - centers[j][0]),
                abs(centers[i][1] - centers[j][1]),
            )
            worst = min(worst, float(cheb) - (2 * BEAD_HW + 1))

    patches = all_patches(scene)
    for i in range(len(patches)):
        for j in range(i + 1, len(patches)):
            ax0, ay0, ax1, ay1 = patch_rect(patches[i])
            bx0, by0, bx1, by1 = patch_rect(patches[j])
            gap_x = max(bx0 - ax1, ax0 - bx1)
            gap_y = max(by0 - ay1, ay0 - by1)
            worst = min(worst, float(max(gap_x, gap_y)))
    return round(float(worst), 3)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical latent rewrites: bead list order and decoy list order."""

    transforms: list[tuple[str, dict]] = []

    flipped = dict(scene)
    flipped["beads"] = list(reversed(scene["beads"]))
    transforms.append(("reverse_bead_order", flipped))

    reversed_decoys = dict(scene)
    reversed_decoys["decoys"] = list(reversed(scene["decoys"]))
    transforms.append(("reverse_decoy_order", reversed_decoys))

    rotated = dict(scene)
    rotated["decoys"] = list(scene["decoys"][1:]) + list(scene["decoys"][:1])
    transforms.append(("rotate_decoy_order", rotated))
    return transforms
