"""Independent pixel-only inverse arm for contour_match_clutter_sector.

Receives only a PIL image and the public layout contract. It extracts the dark
reference polyline from the grey-bordered box at the top, then, for each of the
four grey-divided sectors, extracts the dark contour fragments (connected
components) and measures how much of the reference shape each fragment covers.
The sector whose fragment best covers the reference contour (an exact copy is
the winner) is returned. The same thickness and drawing conventions are not
read from the renderer: the oracle re-detects the reference and the components
directly from the raster.

Imports nothing from the renderer, prompts, generator, or verifier.
"""

from __future__ import annotations

import numpy as np

REF_X0, REF_Y0, REF_X1, REF_Y1 = 160, 20, 352, 150
GRID_Y, GRID_X, GRID_H = 160, 256, 336

_SECTORS = ("TL", "TR", "BL", "BR")
_SECTOR_REGIONS = {
    "TL": (0, GRID_Y, GRID_X, GRID_H),
    "TR": (GRID_X, GRID_Y, 512, GRID_H),
    "BL": (0, GRID_H, GRID_X, 512),
    "BR": (GRID_X, GRID_H, 512, 512),
}


def _bbox_mask(region: np.ndarray) -> np.ndarray | None:
    ys, xs = np.nonzero(region)
    if xs.size == 0:
        return None
    return region[ys.min(): ys.max() + 1, xs.min(): xs.max() + 1]


def _components(region: np.ndarray) -> list[np.ndarray]:
    ys, xs = np.nonzero(region.astype(bool))
    n = ys.size
    if n == 0:
        return []
    parent = np.arange(n)
    position = {(int(y), int(x)): i for i, (y, x) in enumerate(zip(ys, xs))}

    def find(a: int) -> int:
        while parent[a] != a:
            parent[a] = parent[parent[a]]
            a = parent[a]
        return a

    def union(a: int, b: int) -> None:
        ra = find(a)
        rb = find(b)
        if ra != rb:
            parent[rb] = ra

    for i in range(n):
        y = int(ys[i])
        x = int(xs[i])
        for dy in (-1, 0, 1):
            for dx in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                j = position.get((y + dy, x + dx))
                if j is not None:
                    union(i, j)

    root_map: dict[int, list[int]] = {}
    for i in range(n):
        root_map.setdefault(int(find(i)), []).append(i)

    comps = []
    for members in root_map.values():
        my = ys[members]
        mx = xs[members]
        ym, yy = int(my.min()), int(my.max())
        xm, xx = int(mx.min()), int(mx.max())
        mask = np.zeros((yy - ym + 1, xx - xm + 1), dtype=bool)
        mask[my - ym, mx - xm] = True
        comps.append(mask)
    return comps


def _coverage(ref: np.ndarray, other: np.ndarray) -> float:
    rh, rw = ref.shape
    oh, ow = other.shape
    h = min(rh, oh)
    w = min(rw, ow)
    ref_total = float(ref.sum())
    if ref_total == 0:
        return 0.0
    inter = float(np.logical_and(ref[:h, :w], other[:h, :w]).sum())
    return inter / ref_total


def decision_from_image(image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    dark = arr.sum(axis=2) < 150

    ref_mask = _bbox_mask(dark[REF_Y0:REF_Y1, REF_X0:REF_X1])
    if ref_mask is None:
        return "TL"

    best_sector = "TL"
    best_score = -1.0
    for sector in _SECTORS:
        x0, y0, x1, y1 = _SECTOR_REGIONS[sector]
        region = dark[y0:y1, x0:x1]
        score = 0.0
        for comp in _components(region):
            score = max(score, _coverage(ref_mask, comp))
        if score > best_score:
            best_score = score
            best_sector = sector
    return best_sector
