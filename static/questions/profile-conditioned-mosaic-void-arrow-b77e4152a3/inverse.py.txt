"""Independent image-only inverse arm for the mosaic cavity."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[np.ndarray] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        pixels = []
        while queue:
            yy, xx = queue.popleft()
            pixels.append((yy, xx))
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        groups.append(np.asarray(pixels, dtype=np.int32))
    return groups


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    near_white = (arr.min(axis=2) >= 242) & ((arr.max(axis=2) - arr.min(axis=2)) <= 14)
    height, width = near_white.shape
    enclosed = []
    for pts in _components(near_white):
        ys, xs = pts[:, 0], pts[:, 1]
        if xs.min() == 0 or ys.min() == 0 or xs.max() == width - 1 or ys.max() == height - 1:
            continue
        enclosed.append(pts)
    if not enclosed:
        raise ValueError("no enclosed near-white component")
    main = max(enclosed, key=len)
    if len(main) < 600:
        raise ValueError("largest empty component is too small")
    ys, xs = main[:, 0], main[:, 1]
    x0, x1 = int(xs.min()), int(xs.max())
    # Both endpoints are deliberately short vertical faces, not single-pixel vertices.
    def span_at(x: int) -> int:
        yy = ys[xs == x]
        return 0 if len(yy) == 0 else int(yy.max() - yy.min() + 1)
    left_span = max(span_at(x0 + d) for d in range(0, 2))
    right_span = max(span_at(x1 - d) for d in range(0, 2))
    if abs(left_span - right_span) <= 6:
        raise ValueError("ambiguous endpoint-span margin")
    return "right" if right_span < left_span else "left"
