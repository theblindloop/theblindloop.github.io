"""Deterministic colored-mosaic renderer with an arrow-shaped empty cavity."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 384, 320
TILE = 32
PALETTE = [
    (28, 93, 153),
    (215, 83, 65),
    (40, 148, 117),
    (231, 171, 48),
    (112, 79, 154),
    (32, 129, 139),
]
VOID = (250, 250, 246)
BORDER = (31, 37, 42)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    arrow = "right" if int(seed) % 2 == 0 else "left"
    shaft_height = rng.choice([44, 48, 52])
    tip_height = rng.choice([8, 16, 24, 32])
    center_x = 192 + rng.choice([-20, -10, 0, 10, 20])
    center_y = 160 + rng.choice([-18, -9, 0, 9, 18])
    # Corner cavities are answer-independent and never approach the main arrow.
    pool = [(48, 48), (336, 48), (48, 272), (336, 272), (80, 80), (304, 240)]
    rng.shuffle(pool)
    distractors = [[x, y, rng.choice([8, 10, 12])] for x, y in pool[:3]]
    return {
        "arrow": arrow,
        "center_x": center_x,
        "center_y": center_y,
        "shaft_length": rng.choice([92, 100, 108]),
        "head_length": rng.choice([52, 58, 64]),
        "shaft_height": shaft_height,
        "head_height": rng.choice([112, 120, 128]),
        "tip_height": tip_height,
        "tile_seed": rng.randrange(1_000_000_000),
        "palette_shift": rng.randrange(len(PALETTE)),
        "distractors": distractors,
    }


def _arrow_polygon(scene: dict) -> list[tuple[int, int]]:
    cx, cy = int(scene["center_x"]), int(scene["center_y"])
    shaft = int(scene["shaft_length"])
    head = int(scene["head_length"])
    sh = int(scene["shaft_height"])
    hh = int(scene["head_height"])
    th = int(scene["tip_height"])
    left = cx - (shaft + head) // 2
    base = left + shaft
    tip = base + head
    points = [
        (left, cy - sh // 2),
        (base, cy - sh // 2),
        (base, cy - hh // 2),
        (tip, cy - th // 2),
        (tip, cy + th // 2),
        (base, cy + hh // 2),
        (base, cy + sh // 2),
        (left, cy + sh // 2),
    ]
    if scene["arrow"] == "left":
        points = [(2 * cx - x, y) for x, y in points]
    return points


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BORDER)
    draw = ImageDraw.Draw(img)
    rng = random.Random(int(scene["tile_seed"]))
    shift = int(scene["palette_shift"])
    # A complete, answer-independent colored tile field; direction never controls color.
    for row, y in enumerate(range(16, HEIGHT - 16, TILE)):
        for col, x in enumerate(range(16, WIDTH - 16, TILE)):
            index = (rng.randrange(len(PALETTE)) + shift + row + 2 * col) % len(PALETTE)
            draw.rectangle((x, y, min(x + TILE - 1, WIDTH - 17), min(y + TILE - 1, HEIGHT - 17)), fill=PALETTE[index])
    draw.polygon(_arrow_polygon(scene), fill=VOID)
    for x, y, size in scene["distractors"]:
        half = int(size) // 2
        draw.rectangle((int(x) - half, int(y) - half, int(x) + half, int(y) + half), fill=VOID)
    return img


def analytic_gold(scene: dict) -> str:
    return str(scene["arrow"])


def margin(scene: dict) -> float:
    return float(abs(int(scene["shaft_height"]) - int(scene["tip_height"])))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 10.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
