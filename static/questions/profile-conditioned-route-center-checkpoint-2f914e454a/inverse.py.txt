"""Independent PNG-only inverse arm for minimax route centrality."""

from __future__ import annotations

from collections import deque
import math

import numpy as np
from PIL import Image

EDGE = np.array((38, 43, 52), dtype=np.int16)
JUNCTION = np.array((74, 145, 202), dtype=np.int16)
GRAY = np.array((126, 133, 142), dtype=np.int16)
CHECKPOINTS = {
    "orange": np.array((235, 139, 45), dtype=np.int16),
    "purple": np.array((139, 82, 190), dtype=np.int16),
    "blue": np.array((45, 108, 210), dtype=np.int16),
}


def _mask(array: np.ndarray, color: np.ndarray, tolerance: int = 10) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _components(mask: np.ndarray, minimum: int = 40) -> list[tuple[float, float]]:
    work = mask.copy()
    centers = []
    height, width = work.shape
    while work.any():
        y0, x0 = np.argwhere(work)[0]
        queue = deque([(int(y0), int(x0))])
        work[y0, x0] = False
        pixels = []
        while queue:
            y, x = queue.popleft()
            pixels.append((y, x))
            for yy, xx in ((y-1, x), (y+1, x), (y, x-1), (y, x+1)):
                if 0 <= yy < height and 0 <= xx < width and work[yy, xx]:
                    work[yy, xx] = False
                    queue.append((yy, xx))
        if len(pixels) >= minimum:
            centers.append((sum(x for y, x in pixels) / len(pixels), sum(y for y, x in pixels) / len(pixels)))
    return centers


def _supported_line(edge_mask: np.ndarray, p: tuple[float, float], q: tuple[float, float],
                    blockers: list[tuple[float, float]]) -> bool:
    dx, dy = q[0] - p[0], q[1] - p[1]
    length = math.hypot(dx, dy)
    if length < 45.0:
        return False
    for r in blockers:
        rx, ry = r[0] - p[0], r[1] - p[1]
        t = (rx * dx + ry * dy) / (length * length)
        if 0.10 < t < 0.90 and abs(rx * dy - ry * dx) / length < 19.0:
            return False
    usable = length - 44.0
    samples = max(12, int(usable))
    hits = 0
    height, width = edge_mask.shape
    for index in range(samples):
        distance = 22.0 + usable * index / max(1, samples - 1)
        x = p[0] + dx * distance / length
        y = p[1] + dy * distance / length
        xi, yi = int(round(x)), int(round(y))
        y0, y1 = max(0, yi-2), min(height, yi+3)
        x0, x1 = max(0, xi-2), min(width, xi+3)
        hits += bool(edge_mask[y0:y1, x0:x1].any())
    return hits / samples >= 0.90


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    if array.shape != (512, 512, 3):
        raise ValueError("expected a 512 by 512 RGB raster")
    edge_mask = _mask(array, EDGE)
    junctions = _components(_mask(array, JUNCTION))
    terminals = _components(_mask(array, GRAY))
    checkpoint_points = {}
    for name, rgb in CHECKPOINTS.items():
        parts = _components(_mask(array, rgb))
        if len(parts) != 1:
            raise ValueError(f"expected one {name} checkpoint")
        checkpoint_points[name] = parts[0]
    if len(junctions) != 4 or len(terminals) != 4:
        raise ValueError("expected four junctions and four gray terminals")

    labels = [*[f"j{i}" for i in range(4)], *[f"t{i}" for i in range(4)], *checkpoint_points]
    points = [*junctions, *terminals, *[checkpoint_points[name] for name in checkpoint_points]]
    graph = {label: [] for label in labels}
    for left in range(len(points)):
        for right in range(left + 1, len(points)):
            blockers = [points[k] for k in range(len(points)) if k not in (left, right)]
            if _supported_line(edge_mask, points[left], points[right], blockers):
                graph[labels[left]].append(labels[right])
                graph[labels[right]].append(labels[left])
    if sum(len(neighbors) for neighbors in graph.values()) != 20:
        raise ValueError(f"raster does not reconstruct as the expected ten-link tree: {graph}")

    scores = {}
    terminal_labels = {f"t{i}" for i in range(4)}
    for color in checkpoint_points:
        distances = {color: 0}
        queue = deque([color])
        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if neighbor not in distances:
                    distances[neighbor] = distances[node] + 1
                    queue.append(neighbor)
        if not terminal_labels <= distances.keys():
            raise ValueError("not every terminal is reachable")
        scores[color] = max(distances[node] for node in terminal_labels)
    best = min(scores.values())
    matches = [color for color, score in scores.items() if score == best]
    if len(matches) != 1:
        raise ValueError(f"raster has no unique minimax checkpoint: {scores}")
    return matches[0]
