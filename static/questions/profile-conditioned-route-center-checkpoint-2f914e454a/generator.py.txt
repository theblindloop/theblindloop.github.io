"""Deterministic raster world and analytic arm for minimax route centrality."""

from __future__ import annotations

import copy
import random
from collections import deque

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 512
BACKGROUND = (247, 248, 250)
EDGE = (38, 43, 52)
JUNCTION = (74, 145, 202)
GRAY = (126, 133, 142)
WHITE = (255, 255, 255)
CHECKPOINT_RGB = {
    "orange": (235, 139, 45),
    "purple": (139, 82, 190),
    "blue": (45, 108, 210),
}
CHECKPOINT_IDS = ("c0", "c1", "c2")


def _transform(x: int, y: int, transform_id: int) -> tuple[int, int]:
    """Apply one of eight square symmetries around the canvas center."""
    if transform_id >= 4:
        x = WIDTH - x
        transform_id -= 4
    for _ in range(transform_id):
        x, y = WIDTH - y, x
    return int(x), int(y)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    winner_slot = (int(seed) // 3) % 3
    left_arm = 4 - winner_slot

    spine = [
        (60, 110), (150, 110), (240, 110), (350, 110), (440, 110),
        (440, 250), (350, 250), (240, 250), (150, 250),
    ]
    transform_id = (5 * int(seed) + int(seed) // 7) % 8
    spine = [_transform(x, y, transform_id) for x, y in spine]
    branch_points = [
        _transform(240, 35, transform_id),
        _transform(350, 390, transform_id),
    ]

    checkpoint_at = {
        left_arm + offset: CHECKPOINT_IDS[offset]
        for offset in range(3)
    }
    nodes = []
    for index, (x, y) in enumerate(spine):
        if index in (0, 8):
            node_id, kind = f"terminal_{index}", "terminal"
        elif index in checkpoint_at:
            node_id, kind = checkpoint_at[index], "checkpoint"
        else:
            node_id, kind = f"junction_{index}", "junction"
        nodes.append({"id": node_id, "x": x, "y": y, "kind": kind})
    nodes.extend([
        {"id": "terminal_branch_a", "x": branch_points[0][0], "y": branch_points[0][1], "kind": "terminal"},
        {"id": "terminal_branch_b", "x": branch_points[1][0], "y": branch_points[1][1], "kind": "terminal"},
    ])

    spine_ids = [node["id"] for node in nodes[:9]]
    edges = [[spine_ids[i], spine_ids[i + 1]] for i in range(8)]
    edges.extend([[spine_ids[2], "terminal_branch_a"], [spine_ids[6], "terminal_branch_b"]])

    desired_color = ("orange", "purple", "blue")[int(seed) % 3]
    other_colors = [name for name in CHECKPOINT_RGB if name != desired_color]
    rng.shuffle(other_colors)
    checkpoint_colors = {}
    for slot, node_id in enumerate(CHECKPOINT_IDS):
        checkpoint_colors[node_id] = desired_color if slot == winner_slot else other_colors.pop()
    return {"nodes": nodes, "edges": edges, "checkpoint_colors": checkpoint_colors}


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    positions = {node["id"]: (int(node["x"]), int(node["y"])) for node in scene["nodes"]}
    for left, right in scene["edges"]:
        draw.line([positions[left], positions[right]], fill=EDGE, width=7)
    for node in scene["nodes"]:
        x, y = positions[node["id"]]
        kind = node["kind"]
        if kind == "terminal":
            radius = 15
            draw.rectangle((x-radius-3, y-radius-3, x+radius+3, y+radius+3), fill=WHITE)
            draw.rectangle((x-radius, y-radius, x+radius, y+radius), fill=GRAY)
        else:
            radius = 15 if kind == "checkpoint" else 10
            fill = CHECKPOINT_RGB[scene["checkpoint_colors"][node["id"]]] if kind == "checkpoint" else JUNCTION
            draw.ellipse((x-radius-3, y-radius-3, x+radius+3, y+radius+3), fill=WHITE)
            draw.ellipse((x-radius, y-radius, x+radius, y+radius), fill=fill)
    return image


def _scores(scene: dict) -> dict[str, int]:
    graph: dict[str, list[str]] = {node["id"]: [] for node in scene["nodes"]}
    for left, right in scene["edges"]:
        graph[left].append(right)
        graph[right].append(left)
    terminals = [node["id"] for node in scene["nodes"] if node["kind"] == "terminal"]
    scores = {}
    for checkpoint in scene["checkpoint_colors"]:
        distances = {checkpoint: 0}
        queue = deque([checkpoint])
        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if neighbor not in distances:
                    distances[neighbor] = distances[node] + 1
                    queue.append(neighbor)
        if any(terminal not in distances for terminal in terminals):
            raise ValueError("tree is disconnected")
        scores[checkpoint] = max(distances[terminal] for terminal in terminals)
    return scores


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    best = min(scores.values())
    matches = [node for node, score in scores.items() if score == best]
    if len(matches) != 1:
        raise ValueError("scene lacks a unique minimax checkpoint")
    return str(scene["checkpoint_colors"][matches[0]])


def margin(scene: dict) -> float:
    values = sorted(_scores(scene).values())
    return float(values[1] - values[0]) if len(values) >= 2 else 0.0


def is_quarantined(scene: dict) -> bool:
    nodes = scene["nodes"]
    for index, left in enumerate(nodes):
        for right in nodes[index + 1:]:
            dx = float(left["x"] - right["x"])
            dy = float(left["y"] - right["y"])
            if (dx * dx + dy * dy) ** 0.5 < 45.0:
                return True
    try:
        return margin(scene) < 1.0
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    rename = {node["id"]: f"v_{index}" for index, node in enumerate(twin["nodes"])}
    for node in twin["nodes"]:
        node["id"] = rename[node["id"]]
    twin["edges"] = [[rename[left], rename[right]] for left, right in twin["edges"]]
    twin["checkpoint_colors"] = {rename[node]: color for node, color in twin["checkpoint_colors"].items()}
    return [("global_node_id_rename", twin)]
