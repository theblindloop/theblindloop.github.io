"""Renderer and analytic model for the compulsory-route-checkpoint world."""

from __future__ import annotations

from collections import deque
import random

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 512
BACKGROUND = (248, 246, 240)
INK = (42, 47, 55)
GREEN = (32, 171, 96)
BLUE = (45, 104, 207)
ORANGE = (239, 137, 35)
PURPLE = (143, 73, 202)
PALETTE = {"orange": ORANGE, "purple": PURPLE}


def _reachable(edges: list[list[str]], start: str, destination: str, removed: str) -> bool:
    graph: dict[str, list[str]] = {}
    for left, right in edges:
        if removed in (left, right):
            continue
        graph.setdefault(left, []).append(right)
        graph.setdefault(right, []).append(left)
    pending = deque([start])
    seen = {start}
    while pending:
        node = pending.popleft()
        if node == destination:
            return True
        for nxt in graph.get(node, []):
            if nxt not in seen:
                seen.add(nxt)
                pending.append(nxt)
    return False


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    # Two degree-two backbone nodes flank a diamond. Only one is marked as the
    # compulsory checkpoint; the decoy is a degree-two node on either branch.
    base = {
        "s": (42, 256), "x": (91, 256), "a": (141, 256),
        "u1": (203, 154), "u2": (302, 154),
        "l1": (203, 358), "l2": (302, 358),
        "c": (364, 256), "y": (420, 256), "t": (470, 256),
    }
    # Small independent jitter changes lengths/angles without threatening clearance.
    jittered = {
        name: (x + rng.randint(-5, 5), y + rng.randint(-5, 5))
        for name, (x, y) in base.items()
    }
    quarter_turns = rng.randrange(4)
    reflect = bool(rng.randrange(2))

    def pose(point: tuple[int, int]) -> list[int]:
        x, y = point
        if reflect:
            y = HEIGHT - y
        for _ in range(quarter_turns):
            x, y = WIDTH - y, x
        return [x, y]

    nodes = {name: pose(point) for name, point in jittered.items()}
    edges = [
        ["s", "x"], ["x", "a"],
        ["a", "u1"], ["u1", "u2"], ["u2", "c"],
        ["a", "l1"], ["l1", "l2"], ["l2", "c"],
        ["c", "y"], ["y", "t"],
    ]
    compulsory_node = "x" if rng.randrange(2) == 0 else "y"
    decoy_node = rng.choice(["u1", "u2", "l1", "l2"])
    compulsory_color = "orange" if int(seed) % 2 == 0 else "purple"
    other_color = "purple" if compulsory_color == "orange" else "orange"
    checkpoints = {compulsory_color: compulsory_node, other_color: decoy_node}
    return {
        "nodes": nodes,
        "edges": edges,
        "start": "s",
        "destination": "t",
        "checkpoints": checkpoints,
    }


def render(scene: dict) -> Image.Image:
    # Direct accesses are intentional: these are exactly the rendered latent fields.
    nodes = scene["nodes"]
    edges = scene["edges"]
    start = scene["start"]
    destination = scene["destination"]
    checkpoints = scene["checkpoints"]
    scale = 4
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for left, right in edges:
        p = tuple(value * scale for value in nodes[left])
        q = tuple(value * scale for value in nodes[right])
        draw.line([p, q], fill=INK, width=8 * scale)
    # Small dark junction hubs make branch ownership unambiguous.
    for name in ("a", "c"):
        x, y = nodes[name]
        r = 6 * scale
        draw.ellipse((x * scale-r, y * scale-r, x * scale+r, y * scale+r), fill=INK)
    for color in ("orange", "purple"):
        x, y = nodes[checkpoints[color]]
        r = 14 * scale
        draw.ellipse((x * scale-r, y * scale-r, x * scale+r, y * scale+r),
                     fill=PALETTE[color], outline=(255, 255, 255), width=2 * scale)
    sx, sy = nodes[start]
    r = 15 * scale
    draw.ellipse((sx*scale-r, sy*scale-r, sx*scale+r, sy*scale+r),
                 fill=GREEN, outline=(255, 255, 255), width=2 * scale)
    # A smaller light center makes the green marker visibly a ring.
    ri = 6 * scale
    draw.ellipse((sx*scale-ri, sy*scale-ri, sx*scale+ri, sy*scale+ri), fill=BACKGROUND)
    tx, ty = nodes[destination]
    h = 14 * scale
    draw.rounded_rectangle((tx*scale-h, ty*scale-h, tx*scale+h, ty*scale+h),
                           radius=3*scale, fill=BLUE, outline=(255,255,255), width=2*scale)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    nodes = scene["nodes"]
    edges = scene["edges"]
    start = scene["start"]
    destination = scene["destination"]
    checkpoints = scene["checkpoints"]
    # Access nodes because graph-node existence is part of the rendered topology.
    if start not in nodes or destination not in nodes:
        raise ValueError("missing endpoint node")
    compulsory = [
        color for color, node in checkpoints.items()
        if not _reachable(edges, start, destination, node)
    ]
    if len(compulsory) != 1:
        raise ValueError(f"expected one compulsory checkpoint, got {compulsory}")
    return compulsory[0]


def margin(scene: dict) -> float:
    # Runner-up gap: after deleting the decoy, exactly two branch-node paths are
    # reduced to one, whereas deleting the winner leaves zero complete routes.
    return 1.0


def is_quarantined(scene: dict) -> bool:
    try:
        return analytic_gold(scene) not in {"orange", "purple"}
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # The representation is minimal; visual pose and color permutations change pixels.
    return []
