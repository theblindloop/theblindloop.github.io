"""Independent image-only inverse arm for route-network bottleneck detection."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

COLORS = {
    "green": np.array((32, 171, 96), dtype=np.int16),
    "blue": np.array((45, 104, 207), dtype=np.int16),
    "orange": np.array((239, 137, 35), dtype=np.int16),
    "purple": np.array((143, 73, 202), dtype=np.int16),
}


def _color_mask(array: np.ndarray, target: np.ndarray, radius: int = 55) -> np.ndarray:
    delta = array.astype(np.int32) - target.astype(np.int32)
    return np.sum(delta * delta, axis=2, dtype=np.int32) < radius * radius


def _center(mask: np.ndarray, name: str) -> tuple[int, int]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 80:
        raise ValueError(f"missing {name} marker")
    return int(round(float(xs.mean()))), int(round(float(ys.mean())))


def _disk(mask: np.ndarray, center: tuple[int, int], radius: int) -> None:
    x, y = center
    yy, xx = np.ogrid[:mask.shape[0], :mask.shape[1]]
    mask |= (xx - x) ** 2 + (yy - y) ** 2 <= radius * radius


def _connected(mask: np.ndarray, start: tuple[int, int], goal: tuple[int, int]) -> bool:
    height, width = mask.shape
    sx, sy = start
    gx, gy = goal
    if not mask[sy, sx] or not mask[gy, gx]:
        return False
    seen = np.zeros_like(mask, dtype=np.bool_)
    seen[sy, sx] = True
    pending = deque([(sx, sy)])
    while pending:
        x, y = pending.popleft()
        if (x, y) == (gx, gy):
            return True
        for nx, ny in ((x-1,y), (x+1,y), (x,y-1), (x,y+1)):
            if 0 <= nx < width and 0 <= ny < height and mask[ny, nx] and not seen[ny, nx]:
                seen[ny, nx] = True
                pending.append((nx, ny))
    return False


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    masks = {name: _color_mask(array, target) for name, target in COLORS.items()}
    centers = {name: _center(mask, name) for name, mask in masks.items()}
    # Neutral dark pixels only; saturated marker colors are deliberately excluded.
    maximum = array.max(axis=2)
    minimum = array.min(axis=2)
    dark = (maximum < 105) & ((maximum - minimum) < 35)
    results: dict[str, bool] = {}
    for removed in ("orange", "purple"):
        traversable = dark.copy()
        for name in ("green", "blue", "orange", "purple"):
            if name != removed:
                _disk(traversable, centers[name], 16)
        results[removed] = _connected(traversable, centers["green"], centers["blue"])
    compulsory = [color for color, still_connected in results.items() if not still_connected]
    if len(compulsory) != 1:
        raise ValueError(f"ambiguous pixel graph: reachability={results}")
    return compulsory[0]
