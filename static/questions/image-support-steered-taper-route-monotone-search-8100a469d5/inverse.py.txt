"""Pixel-only oracle for the monotone-taper ribbon-route world.

This module sees a single RGB image and the public question contract. It never
imports the renderer, prompts, generator, verifier or analytic gold, and it
never receives latent scene data.

Route: the modal colour is background; ink rows group into horizontal lanes;
inside a lane the number of ink pixels in a column is that route's thickness;
the largest rebound above a running minimum of that thickness profile scores the
lane; the smallest score wins and the runner-up sets the pixel margin.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "taper_route_monotone_search-oracle-0.1.0"

WORDS = ("first", "second", "third", "fourth")
INK_TOLERANCE = 24
LANE_MERGE_GAP = 20


def _ink_mask(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB")).astype(np.int16)
    flat = arr.reshape(-1, 3)
    keys = (flat[:, 0].astype(np.int32) << 16) + (flat[:, 1].astype(np.int32) << 8) + flat[:, 2]
    values, counts = np.unique(keys, return_counts=True)
    background = int(values[int(np.argmax(counts))])
    bg = np.array(
        [(background >> 16) & 255, (background >> 8) & 255, background & 255], dtype=np.int16
    )
    return (np.abs(arr - bg).max(axis=2) > INK_TOLERANCE)


def _lane_bands(mask: np.ndarray) -> list[tuple[int, int]]:
    rows = np.flatnonzero(mask.any(axis=1))
    if rows.size == 0:
        return []
    bands: list[list[int]] = [[int(rows[0]), int(rows[0])]]
    for row in rows[1:]:
        row = int(row)
        if row - bands[-1][1] <= LANE_MERGE_GAP:
            bands[-1][1] = row
        else:
            bands.append([row, row])
    return [(low, high) for low, high in bands]


def _max_rebound(thickness: np.ndarray) -> float:
    running = np.minimum.accumulate(thickness)
    return float(np.max(thickness - running))


def measure_from_image(image: Image.Image) -> dict | None:
    """Score every lane by its largest thickness rebound; report the winner."""

    mask = _ink_mask(image)
    bands = _lane_bands(mask)
    if not bands:
        return None
    scores: list[float] = []
    for low, high in bands:
        columns = mask[low : high + 1].sum(axis=0).astype(np.int32)
        filled = np.flatnonzero(columns)
        if filled.size < 8:
            scores.append(float("inf"))
            continue
        scores.append(_max_rebound(columns[filled[0] : filled[-1] + 1]))
    order = sorted(range(len(scores)), key=lambda i: (scores[i], i))
    best = order[0]
    if len(order) > 1 and np.isfinite(scores[order[1]]):
        pixel_margin = scores[order[1]] - scores[best]
    else:
        pixel_margin = 0.0
    return {
        "lane_count": len(bands),
        "lane_scores": scores,
        "winner_index": best,
        "pixel_margin": float(pixel_margin),
    }


def decision_from_image(image: Image.Image) -> str:
    """Ordinal position, from the top, of the only never-thickening ribbon."""

    report = measure_from_image(image)
    if report is None:
        return WORDS[0]
    index = int(report["winner_index"])
    return WORDS[min(index, len(WORDS) - 1)]
