"""Deterministic scene sampler, renderer, and analytic arm."""

from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image, ImageDraw

CANVAS = 384
SUPERSAMPLE = 4
LABELS = ("arch", "cup", "wave")

# Screen-coordinate templates.  Negative y is the left side of a rightward trip.
_TEMPLATES = {
    "arch": np.asarray([0.0, -1.0, -2.0, -1.0, 0.0]),
    "cup": np.asarray([0.0, 1.0, 2.0, 1.0, 0.0]),
    "wave": np.asarray([0.0, -1.0, 0.0, 1.0, 0.0]),
}


def _scores(offsets: list[float]) -> dict[str, float]:
    values = np.asarray(offsets, dtype=np.float64)
    if values.shape != (5,):
        raise ValueError("center_offsets must contain five values")
    baseline = np.linspace(values[0], values[-1], 5)
    residual = values - baseline
    scores: dict[str, float] = {}
    for name, template in _TEMPLATES.items():
        scale = max(0.0, float(np.dot(residual, template) / np.dot(template, template)))
        scores[name] = float(np.mean((residual - scale * template) ** 2))
    return scores


def _classify_offsets(offsets: list[float]) -> tuple[str, float]:
    ordered = sorted(_scores(offsets).items(), key=lambda item: (item[1], item[0]))
    return ordered[0][0], ordered[1][1] - ordered[0][1]


def sample_scene(seed: int) -> dict:
    """Sample one balanced compact scene from an integer seed."""

    rng = random.Random(int(seed))
    primary_index = int(seed) % len(LABELS)
    primary = LABELS[primary_index]
    competitor_index = (primary_index + 1 + rng.randrange(2)) % len(LABELS)
    competitor = LABELS[competitor_index]
    # A minority of examples approach the visible residual boundary while
    # retaining a comfortable raster margin.
    clarity = (1.0, 0.92, 0.84)[rng.randrange(3)]
    amplitude = 3.0
    offsets = (
        clarity * _TEMPLATES[primary]
        + (1.0 - clarity) * _TEMPLATES[competitor]
    ) * amplitude

    grid_centers = (72.0, 120.0, 168.0, 216.0, 264.0, 312.0)
    center_x = rng.choice(grid_centers)
    center_y = rng.choice(grid_centers)
    spacing = rng.choice((11.5, 12.0, 12.5))
    loop_radius = rng.choice((3.7, 3.9, 4.1))
    tilt = rng.choice((-0.25, -0.125, 0.0, 0.125, 0.25))

    distractors: list[list[float]] = []
    while len(distractors) < 7:
        x = rng.uniform(18.0, CANVAS - 18.0)
        y = rng.uniform(18.0, CANVAS - 18.0)
        if abs(x - center_x) < 58.0 and abs(y - center_y) < 38.0:
            continue
        length = rng.uniform(5.0, 10.0)
        angle = rng.uniform(0.0, math.tau)
        distractors.append(
            [
                round(x - 0.5 * length * math.cos(angle), 3),
                round(y - 0.5 * length * math.sin(angle), 3),
                round(x + 0.5 * length * math.cos(angle), 3),
                round(y + 0.5 * length * math.sin(angle), 3),
            ]
        )

    return {
        "width": CANVAS,
        "height": CANVAS,
        "center_x": center_x,
        "center_y": center_y,
        "spacing": spacing,
        "loop_radius": loop_radius,
        "tilt": tilt,
        "center_offsets": [round(float(value), 4) for value in offsets],
        "start_color": "#d64235",
        "end_color": "#e4aa18",
        "thread_color": "#263238",
        "background_color": "#faf9f4",
        "guide_color": "#c8d8df",
        "distractor_color": "#dbe4e7",
        "distractors": distractors,
    }


def _loop_centers(scene: dict) -> list[tuple[float, float]]:
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    spacing = float(scene["spacing"])
    tilt = float(scene["tilt"])
    offsets = scene["center_offsets"]
    return [
        (cx + (index - 2) * spacing, cy + tilt * (index - 2) + float(offsets[index]))
        for index in range(5)
    ]


def _unit(a: tuple[float, float], b: tuple[float, float]) -> tuple[float, float]:
    dx, dy = b[0] - a[0], b[1] - a[1]
    length = math.hypot(dx, dy)
    if length <= 0.0:
        raise ValueError("coincident carrier points")
    return dx / length, dy / length


def render(scene: dict) -> Image.Image:
    """Render the final RGB PNG world; no text or metadata carries the answer."""

    ss = SUPERSAMPLE
    width = int(scene["width"])
    height = int(scene["height"])
    image = Image.new("RGB", (width * ss, height * ss), scene["background_color"])
    draw = ImageDraw.Draw(image)

    def point(p: tuple[float, float]) -> tuple[float, float]:
        return p[0] * ss, p[1] * ss

    for x1, y1, x2, y2 in scene["distractors"]:
        draw.line(
            [(x1 * ss, y1 * ss), (x2 * ss, y2 * ss)],
            fill=scene["distractor_color"],
            width=1 * ss,
        )

    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    spacing = float(scene["spacing"])
    radius = float(scene["loop_radius"])
    tilt = float(scene["tilt"])
    centers = _loop_centers(scene)
    start = (cx - 3.0 * spacing, cy - 3.0 * tilt)
    end = (cx + 3.0 * spacing, cy + 3.0 * tilt)

    guide_box = [
        (cx - 3.45 * spacing) * ss,
        (cy - 15.0) * ss,
        (cx + 3.45 * spacing) * ss,
        (cy + 15.0) * ss,
    ]
    draw.rounded_rectangle(
        guide_box,
        radius=7 * ss,
        outline=scene["guide_color"],
        width=1 * ss,
    )

    thread_width = 2 * ss
    chain = [start, *centers, end]
    radii = [4.2, *([radius] * 5), 4.8]
    for index in range(len(chain) - 1):
        a, b = chain[index], chain[index + 1]
        ux, uy = _unit(a, b)
        # Sink half a supersampled pixel into each outline so the visible
        # carrier is connected while no connector crosses a loop interior.
        inset_a = max(0.0, radii[index] - 0.55)
        inset_b = max(0.0, radii[index + 1] - 0.55)
        p1 = (a[0] + ux * inset_a, a[1] + uy * inset_a)
        p2 = (b[0] - ux * inset_b, b[1] - uy * inset_b)
        draw.line([point(p1), point(p2)], fill=scene["thread_color"], width=thread_width)

    for x, y in centers:
        draw.ellipse(
            [
                (x - radius) * ss,
                (y - radius) * ss,
                (x + radius) * ss,
                (y + radius) * ss,
            ],
            outline=scene["thread_color"],
            width=thread_width,
        )

    start_radius = 4.2
    draw.ellipse(
        [
            (start[0] - start_radius) * ss,
            (start[1] - start_radius) * ss,
            (start[0] + start_radius) * ss,
            (start[1] + start_radius) * ss,
        ],
        fill=scene["start_color"],
    )
    ux, uy = _unit(start, end)
    nx, ny = -uy, ux
    tip = (end[0] + ux * 5.2, end[1] + uy * 5.2)
    rear_a = (end[0] - ux * 4.0 + nx * 4.2, end[1] - uy * 4.0 + ny * 4.2)
    rear_b = (end[0] - ux * 4.0 - nx * 4.2, end[1] - uy * 4.0 - ny * 4.2)
    draw.polygon([point(tip), point(rear_a), point(rear_b)], fill=scene["end_color"])

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return _classify_offsets(scene["center_offsets"])[0]


def margin(scene: dict) -> float:
    return float(_classify_offsets(scene["center_offsets"])[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.20


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
