"""Deterministic renderer and analytic arm for Portal Quotient Escape."""

from __future__ import annotations

from collections import Counter, deque
import copy
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1200, 800
N, CELL = 7, 42
BACKGROUND = (245, 242, 235)
CARD = (255, 255, 252)
INK = (42, 47, 57)
MUTED = (84, 91, 104)
GRID = (203, 209, 216)
CARD_EDGE = (91, 100, 114)
DECISIONS = ("A", "B", "C", "D")
TERMINALS = {"A": (0, 0), "B": (0, 6), "C": (6, 0), "D": (6, 6)}
START = (3, 3)
AUXILIARY = {"X": (1, 3), "Y": (5, 2), "Z": (4, 5)}
MARK_COLORS = {
    "S": (218, 65, 65), "A": (247, 174, 47), "B": (60, 132, 222),
    "C": (80, 170, 92), "D": (153, 88, 205),
}
PORTAL_COLORS = {
    "P": (22, 154, 181), "Q": (230, 103, 54),
    "R": (24, 143, 91), "T": (139, 74, 179),
}
PANEL_BOXES = ((20, 181, 386, 755), (417, 181, 783, 755), (814, 181, 1180, 755))
GRID_OFFSET = (36, 137)
FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(FONT_PATH, size)


def _canon(edge) -> tuple[tuple[int, int], tuple[int, int]]:
    return tuple(sorted((tuple(edge[0]), tuple(edge[1]))))


def _all_edges() -> list[tuple[tuple[int, int], tuple[int, int]]]:
    edges = []
    for r in range(N):
        for c in range(N):
            if c + 1 < N:
                edges.append(((r, c), (r, c + 1)))
            if r + 1 < N:
                edges.append(((r, c), (r + 1, c)))
    return edges


ALL_EDGES = _all_edges()


def _odd_walls(plans) -> set[tuple[tuple[int, int], tuple[int, int]]]:
    counts = Counter(_canon(edge) for plan in plans for edge in plan)
    return {edge for edge, count in counts.items() if count % 2}


def _plain_reachable(walls, start=START) -> set[tuple[int, int]]:
    walls = set(walls)
    seen = {start}
    queue = deque([start])
    while queue:
        r, c = queue.popleft()
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nxt = (r + dr, c + dc)
            if not (0 <= nxt[0] < N and 0 <= nxt[1] < N):
                continue
            if _canon(((r, c), nxt)) in walls or nxt in seen:
                continue
            seen.add(nxt)
            queue.append(nxt)
    return seen


def _portal_reachable(walls, portals, start=START) -> set[tuple[int, int]]:
    jumps = {}
    for cells in portals.values():
        a, b = (tuple(cells[0]), tuple(cells[1]))
        jumps[a] = b
        jumps[b] = a
    walls = set(walls)
    seen = {start}
    queue = deque([start])
    while queue:
        cell = queue.popleft()
        r, c = cell
        neighbors = []
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nxt = (r + dr, c + dc)
            if 0 <= nxt[0] < N and 0 <= nxt[1] < N and _canon((cell, nxt)) not in walls:
                neighbors.append(nxt)
        if cell in jumps:
            neighbors.append(jumps[cell])
        for nxt in neighbors:
            if nxt not in seen:
                seen.add(nxt)
                queue.append(nxt)
    return seen


def _partition(rng: random.Random) -> dict[tuple[int, int], str]:
    seeds = {START: "S", **{cell: label for label, cell in TERMINALS.items()},
             **{cell: label for label, cell in AUXILIARY.items()}}
    for _ in range(1200):
        owner = dict(seeds)
        while len(owner) < N * N:
            frontier = []
            for cell, label in owner.items():
                r, c = cell
                for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                    nxt = (r + dr, c + dc)
                    if 0 <= nxt[0] < N and 0 <= nxt[1] < N and nxt not in owner:
                        frontier.append((nxt, label))
            nxt, label = rng.choice(frontier)
            owner[nxt] = label
        sizes = Counter(owner.values())
        walls = [edge for edge in ALL_EDGES if owner[edge[0]] != owner[edge[1]]]
        if min(sizes.values()) >= 3 and 25 <= len(walls) <= 46:
            return owner
    raise RuntimeError("could not grow portal-ready regions")


def _choose_cell(rng: random.Random, owner, label: str, used: set) -> tuple[int, int]:
    fixed = {START, *TERMINALS.values()}
    choices = [cell for cell, value in owner.items() if value == label and cell not in used and cell not in fixed]
    if not choices:
        raise RuntimeError(f"no free cell in region {label}")
    cell = rng.choice(choices)
    used.add(cell)
    return cell


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = DECISIONS[int(seed) % len(DECISIONS)]
    owner = _partition(rng)
    final_walls = [edge for edge in ALL_EDGES if owner[edge[0]] != owner[edge[1]]]
    nonwalls = [edge for edge in ALL_EDGES if owner[edge[0]] == owner[edge[1]]]
    rng.shuffle(final_walls)
    rng.shuffle(nonwalls)
    decoys = nonwalls[:21]

    plans = [[], [], []]
    for index, edge in enumerate(final_walls):
        plans[index % 3].append(edge)
    pairs = [(0, 1), (1, 2), (0, 2)]
    rng.shuffle(pairs)
    for index, edge in enumerate(decoys):
        for plan_index in pairs[index % 3]:
            plans[plan_index].append(edge)
    rng.shuffle(plans)
    encoded_plans = []
    for plan in plans:
        rng.shuffle(plan)
        encoded_plans.append([
            [list(edge[1]), list(edge[0])] if rng.getrandbits(1)
            else [list(edge[0]), list(edge[1])]
            for edge in plan
        ])

    others = [label for label in DECISIONS if label != target]
    rng.shuffle(others)
    used = set()
    region_pairs = {
        "P": ("S", "X"),
        "Q": ("X", target),
        "R": (others[0], "Y"),
        "T": (others[1], "Z"),
    }
    portals = {
        symbol: [list(_choose_cell(rng, owner, left, used)),
                 list(_choose_cell(rng, owner, right, used))]
        for symbol, (left, right) in region_pairs.items()
    }
    scene = {"plans": encoded_plans, "portals": portals}
    assert not is_quarantined(scene)
    return scene


def analytic_gold(scene: dict) -> str:
    seen = _portal_reachable(_odd_walls(scene["plans"]), scene["portals"])
    matches = [label for label, cell in TERMINALS.items() if cell in seen]
    return matches[0] if len(matches) == 1 else "abstain"


def margin(scene: dict) -> float:
    return 1.0 if analytic_gold(scene) in DECISIONS else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        plans, portals = scene["plans"], scene["portals"]
        if len(plans) != 3 or set(portals) != set(PORTAL_COLORS):
            return True
        occurrences = [_canon(edge) for plan in plans for edge in plan]
        if any(edge not in ALL_EDGES for edge in occurrences):
            return True
        counts = Counter(occurrences)
        if not all(count in (1, 2) for count in counts.values()):
            return True
        if sum(count == 2 for count in counts.values()) < 18:
            return True
        if not all(18 <= len(plan) <= 32 for plan in plans):
            return True
        portal_cells = [tuple(cell) for cells in portals.values() for cell in cells]
        if any(len(cells) != 2 for cells in portals.values()) or len(set(portal_cells)) != 8:
            return True
        if any(not (0 <= r < N and 0 <= c < N) for r, c in portal_cells):
            return True
        if any(cell in {START, *TERMINALS.values()} for cell in portal_cells):
            return True
        plain = _plain_reachable(_odd_walls(plans))
        if any(cell in plain for cell in TERMINALS.values()):
            return True
        return analytic_gold(scene) not in DECISIONS
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    transformed = copy.deepcopy(scene)
    transformed["plans"] = [
        [[edge[1], edge[0]] for edge in plan] for plan in transformed["plans"]
    ]
    return [("reverse_all_stored_wall_endpoints", transformed)]


def _wall_line(edge, gx: int, gy: int) -> tuple[int, int, int, int]:
    (r1, c1), (r2, c2) = _canon(edge)
    if r1 == r2:
        x = gx + max(c1, c2) * CELL
        return (x, gy + r1 * CELL, x, gy + (r1 + 1) * CELL)
    y = gy + max(r1, r2) * CELL
    return (gx + c1 * CELL, y, gx + (c1 + 1) * CELL, y)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((WIDTH // 2, 25), "PORTAL QUOTIENT ESCAPE", fill=INK, font=_font(30), anchor="mm")
    draw.text((WIDTH // 2, 62), "same grid edge = same wall    •    matching letters = joined cells",
              fill=MUTED, font=_font(18), anchor="mm")
    draw.rounded_rectangle((171, 87, 1029, 148), radius=18, fill=(235, 238, 242),
                           outline=CARD_EDGE, width=3)
    draw.text((WIDTH // 2, 117), "shown once = WALL     shown twice = OPEN     portals JOIN",
              fill=INK, font=_font(21), anchor="mm")

    for panel_index, box in enumerate(PANEL_BOXES):
        x0, y0, x1, y1 = box
        draw.rounded_rectangle(box, radius=18, fill=CARD, outline=CARD_EDGE, width=4)
        draw.text(((x0 + x1) // 2, y0 + 35), f"PLAN {panel_index + 1}",
                  fill=INK, font=_font(22), anchor="mm")
        gx, gy = x0 + GRID_OFFSET[0], y0 + GRID_OFFSET[1]
        for k in range(N + 1):
            draw.line((gx + k * CELL, gy, gx + k * CELL, gy + N * CELL), fill=GRID, width=2)
            draw.line((gx, gy + k * CELL, gx + N * CELL, gy + k * CELL), fill=GRID, width=2)
        draw.rectangle((gx, gy, gx + N * CELL, gy + N * CELL), outline=CARD_EDGE, width=4)
        for edge in scene["plans"][panel_index]:
            draw.line(_wall_line(edge, gx, gy), fill=INK, width=10)

        for symbol, cells in scene["portals"].items():
            for r, c in cells:
                cx, cy = gx + c * CELL + CELL // 2, gy + r * CELL + CELL // 2
                draw.rounded_rectangle((cx - 13, cy - 13, cx + 13, cy + 13), radius=6,
                                       fill=PORTAL_COLORS[symbol], outline=INK, width=2)
                draw.text((cx, cy + 1), symbol, fill=(255, 255, 255), font=_font(15), anchor="mm")
        for label, (r, c) in {"S": START, **TERMINALS}.items():
            cx, cy = gx + c * CELL + CELL // 2, gy + r * CELL + CELL // 2
            draw.ellipse((cx - 15, cy - 15, cx + 15, cy + 15),
                         fill=MARK_COLORS[label], outline=INK, width=2)
            fill = (255, 255, 255) if label in {"S", "B", "D"} else (24, 27, 33)
            draw.text((cx, cy + 1), label, fill=fill, font=_font(16), anchor="mm")
        draw.text(((x0 + x1) // 2, y1 - 22), "S = START", fill=MUTED,
                  font=_font(15), anchor="mm")
    return image
