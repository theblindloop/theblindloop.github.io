"""Independent PNG-only inverse for Portal Quotient Escape."""

from __future__ import annotations

from collections import Counter, deque

import numpy as np
from PIL import Image

WIDTH, HEIGHT = 1200, 800
N, CELL = 7, 42
INK = np.array((42, 47, 57), dtype=np.int16)
GRID = np.array((203, 209, 216), dtype=np.int16)
PANEL_ORIGINS = ((20, 181), (417, 181), (814, 181))
GRID_OFFSET = (36, 137)
START = (3, 3)
TERMINALS = {"A": (0, 0), "B": (0, 6), "C": (6, 0), "D": (6, 6)}
MARK_COLORS = {
    "S": (218, 65, 65), "A": (247, 174, 47), "B": (60, 132, 222),
    "C": (80, 170, 92), "D": (153, 88, 205),
}
PORTAL_COLORS = {
    "P": (22, 154, 181), "Q": (230, 103, 54),
    "R": (24, 143, 91), "T": (139, 74, 179),
}


def _all_edges():
    out = []
    for r in range(N):
        for c in range(N):
            if c + 1 < N:
                out.append(((r, c), (r, c + 1)))
            if r + 1 < N:
                out.append(((r, c), (r + 1, c)))
    return out


ALL_EDGES = _all_edges()


def _color_count(arr: np.ndarray, cx: int, cy: int, color) -> int:
    patch = arr[cy - 11:cy + 12, cx - 11:cx + 12].astype(np.int16)
    ref = np.array(color, dtype=np.int16)
    return int((np.abs(patch - ref).max(axis=2) <= 3).sum())


def _segment_patch(arr: np.ndarray, gx: int, gy: int, edge):
    (r1, c1), (r2, c2) = edge
    if r1 == r2:
        x = gx + max(c1, c2) * CELL
        cy = gy + r1 * CELL + CELL // 2
        return arr[cy - 11:cy + 12, x - 4:x + 5]
    y = gy + max(r1, r2) * CELL
    cx = gx + c1 * CELL + CELL // 2
    return arr[y - 4:y + 5, cx - 11:cx + 12]


def _read_plan(arr: np.ndarray, x0: int, y0: int):
    gx, gy = x0 + GRID_OFFSET[0], y0 + GRID_OFFSET[1]
    for label, cell in {"S": START, **TERMINALS}.items():
        r, c = cell
        cx, cy = gx + c * CELL + CELL // 2, gy + r * CELL + CELL // 2
        if _color_count(arr, cx, cy, MARK_COLORS[label]) < 210:
            return None
    portals = {symbol: [] for symbol in PORTAL_COLORS}
    occupied = set()
    for r in range(N):
        for c in range(N):
            cx, cy = gx + c * CELL + CELL // 2, gy + r * CELL + CELL // 2
            hits = [symbol for symbol, color in PORTAL_COLORS.items()
                    if _color_count(arr, cx, cy, color) >= 165]
            if len(hits) > 1:
                return None
            if hits:
                portals[hits[0]].append((r, c))
                occupied.add((r, c))
    if any(len(cells) != 2 for cells in portals.values()) or len(occupied) != 8:
        return None

    walls = []
    for edge in ALL_EDGES:
        patch = _segment_patch(arr, gx, gy, edge).astype(np.int16)
        ink_fraction = float((np.abs(patch - INK).max(axis=2) <= 3).mean())
        grid_fraction = float((np.abs(patch - GRID).max(axis=2) <= 3).mean())
        if ink_fraction >= 0.69:
            walls.append(edge)
        elif grid_fraction < 0.12:
            return None
    return walls, portals


def decision_from_image(image: Image.Image) -> str:
    if image.size != (WIDTH, HEIGHT):
        return "abstain"
    arr = np.asarray(image.convert("RGB"))
    plans = []
    reference_portals = None
    for origin in PANEL_ORIGINS:
        result = _read_plan(arr, *origin)
        if result is None:
            return "abstain"
        plan, portals = result
        normalized = {symbol: tuple(sorted(cells)) for symbol, cells in portals.items()}
        if reference_portals is None:
            reference_portals = normalized
        elif normalized != reference_portals:
            return "abstain"
        plans.append(plan)

    counts = Counter(edge for plan in plans for edge in plan)
    if any(count not in (1, 2) for count in counts.values()):
        return "abstain"
    walls = {edge for edge, count in counts.items() if count % 2}
    jumps = {}
    for cells in reference_portals.values():
        a, b = cells
        jumps[a] = b
        jumps[b] = a
    seen = {START}
    queue = deque([START])
    while queue:
        cell = queue.popleft()
        r, c = cell
        neighbors = []
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nxt = (r + dr, c + dc)
            if 0 <= nxt[0] < N and 0 <= nxt[1] < N and tuple(sorted((cell, nxt))) not in walls:
                neighbors.append(nxt)
        if cell in jumps:
            neighbors.append(jumps[cell])
        for nxt in neighbors:
            if nxt not in seen:
                seen.add(nxt)
                queue.append(nxt)
    matches = [label for label, cell in TERMINALS.items() if cell in seen]
    return matches[0] if len(matches) == 1 else "abstain"
