"""Scene-free pixel arm for deflector_drop_route.

``decision_from_image`` receives one RGB image and nothing else. It imports no
renderer, prompt, generator, verifier or gold code, and it never sees a latent
scene. It recovers the chute grid from the drawn chute bar, the ball column from
the drawn disc, every deflector's column, level and direction from its own drawn
blob, and then replays the fall level by level.

Raises ValueError when the drawn apparatus is missing or malformed, so an erased
or corrupted board refuses instead of guessing.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "deflector_drop_route-oracle-0.1.0"

_CHUTE = (205, 205, 205)
_BALL = (30, 90, 220)
_RIGHT = (235, 120, 20)
_LEFT = (140, 40, 190)

_MIN_BLOB = 200          # px; a real glyph is far larger than any stray speck
_LEVEL_GAP = 30.0        # px; level pitch is 85, glyph y jitter is a few px


def _mask(arr: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    return (arr[:, :, 0] == rgb[0]) & (arr[:, :, 1] == rgb[1]) & (arr[:, :, 2] == rgb[2])


def _components(mask: np.ndarray, min_size: int) -> list[tuple[float, float, int]]:
    """Four-connected components as (centroid_x, centroid_y, size)."""
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out: list[tuple[float, float, int]] = []
    ys, xs = np.nonzero(mask)
    for sy, sx in zip(ys.tolist(), xs.tolist()):
        if seen[sy, sx]:
            continue
        q = deque([(sy, sx)])
        seen[sy, sx] = True
        px: list[int] = []
        py: list[int] = []
        while q:
            y, x = q.popleft()
            px.append(x)
            py.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    q.append((ny, nx))
        if len(px) >= min_size:
            out.append((float(np.mean(px)), float(np.mean(py)), len(px)))
    return out


def _chute_centres(mask: np.ndarray) -> list[float]:
    cols = np.nonzero(mask.any(axis=0))[0]
    if cols.size == 0:
        raise ValueError("no chute bar found")
    centres: list[float] = []
    run = [int(cols[0])]
    for x in cols[1:].tolist():
        if x - run[-1] <= 1:
            run.append(x)
        else:
            centres.append((run[0] + run[-1]) / 2.0)
            run = [x]
    centres.append((run[0] + run[-1]) / 2.0)
    return centres


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)

    centres = _chute_centres(_mask(arr, _CHUTE))
    if len(centres) < 3:
        raise ValueError(f"expected a row of chutes, found {len(centres)}")

    def column_of(x: float) -> int:
        return min(range(len(centres)), key=lambda i: abs(centres[i] - x))

    balls = _components(_mask(arr, _BALL), _MIN_BLOB)
    if len(balls) != 1:
        raise ValueError(f"expected exactly one ball, found {len(balls)}")
    col = column_of(balls[0][0])

    glyphs: list[tuple[float, float, int]] = []
    for rgb, direction in ((_RIGHT, 1), (_LEFT, -1)):
        for cx, cy, _ in _components(_mask(arr, rgb), _MIN_BLOB):
            glyphs.append((cy, cx, direction))
    if not glyphs:
        raise ValueError("no deflector arrows found")
    glyphs.sort()

    levels: list[list[tuple[float, int]]] = [[(glyphs[0][1], glyphs[0][2])]]
    prev_y = glyphs[0][0]
    for cy, cx, direction in glyphs[1:]:
        if cy - prev_y > _LEVEL_GAP:
            levels.append([])
        levels[-1].append((cx, direction))
        prev_y = cy

    pushes = 0
    for level in levels:
        for gx, direction in sorted(level):
            if column_of(gx) == col:
                nxt = col + direction
                if 0 <= nxt < len(centres):
                    col = nxt
                    pushes += 1
                break

    return f"{col + 1}|{pushes}"
