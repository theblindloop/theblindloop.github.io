"""Scene sampling, rendering and analytic gold for deflector_drop_route.

A ball is drawn at the top of a board of seven columns. Five levels of coloured
deflector arrows are drawn on the board. The ball falls level by level; on each
level only the arrow standing in the ball's own column has any effect, pushing
the ball one column in the arrow's direction. Every other arrow on that level is
inert for that fall. The answer is the numbered chute the ball lands in and how
many arrows actually pushed it.

The gold path reads the start column, every arrow's column, every arrow's level
and every arrow's direction back from the *drawn* geometry and colour, exactly
as the question states and exactly as the scene-free pixel arm does.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "deflector_drop_route-renderer-0.1.0"

# ---------------------------------------------------------------- geometry ---
CANVAS = (880, 620)
N_COLS = 7
N_LEVELS = 5
COL0_X = 110.0          # centre of column 1
COL_W = 110.0           # centre-to-centre column pitch
LEVEL0_Y = 150.0        # centre of level 1
LEVEL_H = 85.0          # centre-to-centre level pitch
BALL_Y = 74.0
BALL_R = 15
CHUTE_TOP = 542
CHUTE_BOT = 596
CHUTE_W = 96
ARROW_HALF_W = 31       # glyph half width
ARROW_HALF_H = 13       # glyph half height (head)
ARROW_SHAFT_H = 6       # shaft half height

# ------------------------------------------------------------------ colours --
BG = (255, 255, 255)
GUIDE = (232, 232, 232)
CHUTE = (205, 205, 205)
BALL = (30, 90, 220)
RIGHT_ARROW = (235, 120, 20)     # points right
LEFT_ARROW = (140, 40, 190)      # points left
ORDER_MARK = (150, 150, 150)     # the downward level-order arrow in the margin


def col_x(col: int) -> float:
    return COL0_X + COL_W * col


def level_y(level: int) -> float:
    return LEVEL0_Y + LEVEL_H * level


# ------------------------------------------------------------------ sampling --
def _drawn_ball_x(scene: dict) -> float:
    return col_x(scene["ball"]["col"]) + scene["ball"]["dx"]


def _drawn_arrow_xy(scene: dict, a: dict) -> tuple[float, float]:
    """Drawn centre of one arrow of ``scene``; all inputs are rendered fields."""
    del scene  # the arrow belongs to the scene; the board geometry is fixed
    return col_x(a["col"]) + a["dx"], level_y(a["level"]) + a["dy"]


def _nearest_col(x: float) -> int:
    return min(range(N_COLS), key=lambda c: abs(col_x(c) - x))


def _read_scene(scene: dict) -> tuple[int, list[list[dict]]]:
    """Recover start column and per-level arrows from drawn geometry only."""
    start = _nearest_col(_drawn_ball_x(scene))
    levels: list[list[dict]] = [[] for _ in range(N_LEVELS)]
    for a in scene["arrows"]:
        ax, ay = _drawn_arrow_xy(scene, a)
        lv = min(range(N_LEVELS), key=lambda r: abs(level_y(r) - ay))
        levels[lv].append({"col": _nearest_col(ax), "dir": a["dir"], "x": ax})
    for lv in levels:
        lv.sort(key=lambda d: d["x"])
    return start, levels


def _replay(start: int, levels: list[list[dict]]) -> tuple[int, int]:
    col = start
    pushes = 0
    for arrows in levels:
        for a in arrows:
            if a["col"] == col:
                col += a["dir"]
                pushes += 1
                break
    return col, pushes


def _sample_arrows(rng: random.Random, jitter: bool) -> list[dict] | None:
    arrows: list[dict] = []
    for lv in range(N_LEVELS):
        k = rng.choice([1, 2, 2, 2])
        cols = sorted(rng.sample(range(N_COLS), k))
        if k == 2 and cols[1] - cols[0] < 2:
            return None
        placed = []
        for c in cols:
            if c == 0:
                d = 1
            elif c == N_COLS - 1:
                d = -1
            else:
                d = rng.choice([-1, 1])
            placed.append((c, d))
        # no arrow on a level may push the ball into another arrow of that level
        occupied = {c for c, _ in placed}
        if any((c + d) in occupied for c, d in placed):
            return None
        for c, d in placed:
            arrows.append(
                {
                    "level": lv,
                    "col": c,
                    "dir": d,
                    "dx": round(rng.uniform(-6.0, 6.0), 3) if jitter else 0.0,
                    "dy": round(rng.uniform(-4.0, 4.0), 3) if jitter else 0.0,
                }
            )
    return arrows


def sample_scene(seed: int) -> dict:
    """Deterministically sample one scene.

    ``seed % 11`` selects an explicit near-boundary mode (7) or an explicit tight,
    quarantined mode (4); ``seed % 7`` is the counterbalancing target chute so
    that consecutive seeds sweep every answer class uniformly.
    """
    rng = random.Random(0xD00D ^ (seed * 2654435761 % (1 << 32)))
    mode = "tight" if seed % 11 == 4 else ("near" if seed % 11 == 7 else "plain")
    target_col = seed % 7
    target_pushes = 1 + (seed // 7) % 3

    if mode == "tight":
        clearance = rng.uniform(1.2, 5.2)
    elif mode == "near":
        clearance = rng.uniform(7.0, 11.0)
    else:
        clearance = rng.uniform(20.0, COL_W / 2 - 6.0)
    ball_dx = round((COL_W / 2 - clearance) * rng.choice([-1.0, 1.0]), 3)

    for attempt in range(9000):
        arrows = _sample_arrows(rng, jitter=True)
        if arrows is None:
            continue
        start = rng.randrange(N_COLS)
        scene = {
            "canvas": list(CANVAS),
            "n_cols": N_COLS,
            "n_levels": N_LEVELS,
            "mode": mode,
            "ball": {"col": start, "dx": ball_dx},
            "arrows": arrows,
        }
        s, levels = _read_scene(scene)
        final, pushes = _replay(s, levels)
        if final != target_col:
            continue
        if final == s:                       # the drawn ball column is never the answer
            continue
        if pushes < 1 or pushes >= len(arrows):   # some drawn arrow is always inert
            continue
        naive = max(0, min(N_COLS - 1, s + sum(a["dir"] for a in arrows)))
        if naive == final:                   # summing every drawn arrow never works
            continue
        if len(arrows) == pushes:
            continue
        if attempt < 6000 and pushes != target_pushes:
            continue
        return scene
    raise RuntimeError(f"scene sampling failed for seed {seed}")


# ------------------------------------------------------------------- render --
def _arrow_polygon(cx: float, cy: float, direction: int) -> list[tuple[float, float]]:
    hw, hh, sh = ARROW_HALF_W, ARROW_HALF_H, ARROW_SHAFT_H
    tail, tip = cx - hw * direction, cx + hw * direction
    neck = cx + (hw - 26) * direction
    return [
        (tail, cy - sh),
        (neck, cy - sh),
        (neck, cy - hh),
        (tip, cy),
        (neck, cy + hh),
        (neck, cy + sh),
        (tail, cy + sh),
    ]


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", tuple(scene["canvas"]), BG)
    d = ImageDraw.Draw(img)

    for c in range(N_COLS):
        x = col_x(c)
        d.line([(x, 52), (x, CHUTE_TOP - 8)], fill=GUIDE, width=3)

    # level-order marker in the left margin: the ball falls from top to bottom
    d.line([(46, 120), (46, 500)], fill=ORDER_MARK, width=5)
    d.polygon([(30, 494), (62, 494), (46, 524)], fill=ORDER_MARK)

    for c in range(N_COLS):
        x = col_x(c)
        d.rectangle([x - CHUTE_W / 2, CHUTE_TOP, x + CHUTE_W / 2, CHUTE_BOT], fill=CHUTE)

    for a in sorted(scene["arrows"], key=lambda a: (a["level"], a["col"])):
        ax, ay = _drawn_arrow_xy(scene, a)
        colour = RIGHT_ARROW if a["dir"] > 0 else LEFT_ARROW
        d.polygon(_arrow_polygon(ax, ay, a["dir"]), fill=colour)

    bx = _drawn_ball_x(scene)
    d.ellipse([bx - BALL_R, BALL_Y - BALL_R, bx + BALL_R, BALL_Y + BALL_R], fill=BALL)
    return img


# --------------------------------------------------------------------- gold --
def analytic_gold(scene: dict) -> str:
    start, levels = _read_scene(scene)
    final, pushes = _replay(start, levels)
    return f"{final + 1}|{pushes}"


def margin(scene: dict) -> float:
    """Least drawn clearance, in pixels, of any reading the answer depends on."""
    vals = [COL_W / 2 - abs(scene["ball"]["dx"])]
    for a in scene["arrows"]:
        vals.append(COL_W / 2 - abs(a["dx"]))
        vals.append(LEVEL_H / 2 - abs(a["dy"]))
    by_level: dict[int, list[float]] = {}
    for a in scene["arrows"]:
        ax, _ = _drawn_arrow_xy(scene, a)
        by_level.setdefault(a["level"], []).append(ax)
    for xs in by_level.values():
        xs.sort()
        for lo, hi in zip(xs, xs[1:]):
            vals.append(hi - lo - 2 * ARROW_HALF_W)
    return round(min(vals), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Arrows are an unordered bag: nothing in list order is drawn or replayed."""
    arrows = list(scene["arrows"])
    out: list[tuple[str, dict]] = []
    for name, perm in (
        ("arrows_reversed", arrows[::-1]),
        ("arrows_rotated", arrows[3:] + arrows[:3]),
        ("arrows_sorted_by_column", sorted(arrows, key=lambda a: (a["col"], a["level"]))),
    ):
        alt = dict(scene)
        alt["arrows"] = [dict(a) for a in perm]
        alt["ball"] = dict(scene["ball"])
        out.append((name, alt))
    return out
