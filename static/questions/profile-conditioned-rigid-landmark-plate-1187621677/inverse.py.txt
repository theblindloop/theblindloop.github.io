"""Independent image-only inverse arm for rigid landmark correspondence."""

from __future__ import annotations

import math
from collections import Counter, deque

from PIL import Image

def _colored_pixels(image: Image.Image) -> dict[tuple[int, int, int], set[tuple[int, int]]]:
    rgb = image.convert("RGB")
    width, _height = rgb.size
    counts = Counter(rgb.getdata())
    landmark_colors = {
        value
        for value, count in counts.items()
        if 400 <= count <= 1400 and max(value) - min(value) >= 65 and max(value) < 245
    }
    if len(landmark_colors) != 4:
        raise ValueError(f"expected four repeated saturated landmark colors, found {len(landmark_colors)}")
    groups = {value: set() for value in landmark_colors}
    for offset, value in enumerate(rgb.getdata()):
        if value in groups:
            groups[value].add((offset % width, offset // width))
    return groups


def _centers(colored: set[tuple[int, int]]) -> list[tuple[float, float]]:
    remaining = set(colored)
    found: list[tuple[float, float]] = []
    while remaining:
        start = remaining.pop()
        queue = deque([start])
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            px, py = queue.popleft()
            xs.append(px)
            ys.append(py)
            for neighbor in ((px - 1, py), (px + 1, py), (px, py - 1), (px, py + 1)):
                if neighbor in remaining:
                    remaining.remove(neighbor)
                    queue.append(neighbor)
        if len(xs) >= 80:
            found.append((sum(xs) / len(xs), sum(ys) / len(ys)))
    if len(found) != 4:
        raise ValueError(f"expected four components of a landmark color, found {len(found)}")
    return found


def _alignment_error(left: dict[object, tuple[float, float]], right: dict[object, tuple[float, float]]) -> float:
    identities = sorted(left)
    lp = [left[identity] for identity in identities]
    rp = [right[identity] for identity in identities]
    lcx = sum(p[0] for p in lp) / 4.0
    lcy = sum(p[1] for p in lp) / 4.0
    rcx = sum(p[0] for p in rp) / 4.0
    rcy = sum(p[1] for p in rp) / 4.0
    dot = cross = 0.0
    for p, q in zip(lp, rp):
        x, y = p[0] - lcx, p[1] - lcy
        u, v = q[0] - rcx, q[1] - rcy
        dot += x * u + y * v
        cross += x * v - y * u
    angle = math.atan2(cross, dot)
    c, s = math.cos(angle), math.sin(angle)
    residual = 0.0
    for p, q in zip(lp, rp):
        x, y = p[0] - lcx, p[1] - lcy
        u, v = q[0] - rcx, q[1] - rcy
        residual += (c * x - s * y - u) ** 2 + (s * x + c * y - v) ** 2
    return math.sqrt(residual / 4.0)


def decision_from_image(image: Image.Image) -> str:
    """Recover the answer from raster pixels and the public left-to-right option rule."""
    colored = _colored_pixels(image)
    by_color = {identity: _centers(pixels) for identity, pixels in colored.items()}
    panels: list[dict[object, tuple[float, float]]] = [dict() for _ in range(4)]
    for identity, points in by_color.items():
        reference = min(points, key=lambda point: point[1])
        options = sorted((point for point in points if point != reference), key=lambda point: point[0])
        if len(options) != 3:
            raise ValueError("could not separate reference from three options")
        panels[0][identity] = reference
        for index, point in enumerate(options, start=1):
            panels[index][identity] = point
    errors = [_alignment_error(panels[0], panel) for panel in panels[1:]]
    ranked = sorted(range(3), key=lambda index: errors[index])
    if errors[ranked[1]] - errors[ranked[0]] < 6.0:
        raise ValueError("ambiguous raster alignment")
    return "ABC"[ranked[0]]
