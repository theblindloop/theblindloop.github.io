"""Deterministic renderer and analytic arm for rigid landmark correspondence."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw, ImageFont

COLOR_ORDER = ("red", "blue", "green", "gold")
DEFAULT_PALETTE = {
    "red": "#d94b4b",
    "blue": "#3478c8",
    "green": "#39a866",
    "gold": "#e0a72f",
    "ink": "#26313a",
    "plate": "#f1f4f6",
    "background": "#ffffff",
}


def _rotate(points: dict[str, list[float]], angle: float) -> dict[str, list[float]]:
    c, s = math.cos(angle), math.sin(angle)
    return {
        name: [c * xy[0] - s * xy[1], s * xy[0] + c * xy[1]]
        for name, xy in points.items()
    }


def _translate(points: dict[str, list[float]], tx: float, ty: float) -> dict[str, list[float]]:
    return {name: [xy[0] + tx, xy[1] + ty] for name, xy in points.items()}


def sample_scene(seed: int) -> dict:
    """Sample one balanced, non-ambiguous four-landmark matching scene."""
    rng = random.Random(int(seed))
    base = {
        "red": [-57.0 + rng.uniform(-5, 5), -39.0 + rng.uniform(-4, 4)],
        "blue": [53.0 + rng.uniform(-5, 5), -29.0 + rng.uniform(-4, 4)],
        "green": [35.0 + rng.uniform(-5, 5), 47.0 + rng.uniform(-4, 4)],
        "gold": [-31.0 + rng.uniform(-5, 5), 19.0 + rng.uniform(-4, 4)],
    }
    reference = _rotate(base, rng.uniform(-math.pi, math.pi))

    match = _translate(_rotate(base, rng.uniform(-math.pi, math.pi)), rng.uniform(-5, 5), rng.uniform(-4, 4))
    reflected_base = {name: [-xy[0], xy[1]] for name, xy in base.items()}
    mirror = _translate(
        _rotate(reflected_base, rng.uniform(-math.pi, math.pi)),
        rng.uniform(-5, 5),
        rng.uniform(-4, 4),
    )
    swapped_base = copy.deepcopy(base)
    swapped_base["blue"], swapped_base["gold"] = swapped_base["gold"], swapped_base["blue"]
    swap = _translate(
        _rotate(swapped_base, rng.uniform(-math.pi, math.pi)),
        rng.uniform(-5, 5),
        rng.uniform(-4, 4),
    )

    order = ["match", "mirror", "swap"]
    rng.shuffle(order)
    return {
        "reference_points": reference,
        "candidate_points": {"match": match, "mirror": mirror, "swap": swap},
        "option_order": order,
        "panel_layout": {
            "reference": [360, 142],
            "options": [[130, 371], [360, 371], [590, 371]],
        },
        "palette": dict(DEFAULT_PALETTE),
        "canvas": [720, 520],
    }


def _alignment_error(left: dict[str, list[float]], right: dict[str, list[float]]) -> float:
    lp = [left[name] for name in COLOR_ORDER]
    rp = [right[name] for name in COLOR_ORDER]
    lcx = sum(p[0] for p in lp) / len(lp)
    lcy = sum(p[1] for p in lp) / len(lp)
    rcx = sum(p[0] for p in rp) / len(rp)
    rcy = sum(p[1] for p in rp) / len(rp)
    a = b = 0.0
    for p, q in zip(lp, rp):
        x, y = p[0] - lcx, p[1] - lcy
        u, v = q[0] - rcx, q[1] - rcy
        a += x * u + y * v
        b += x * v - y * u
    angle = math.atan2(b, a)
    c, s = math.cos(angle), math.sin(angle)
    total = 0.0
    for p, q in zip(lp, rp):
        x, y = p[0] - lcx, p[1] - lcy
        u, v = q[0] - rcx, q[1] - rcy
        dx = c * x - s * y - u
        dy = s * x + c * y - v
        total += dx * dx + dy * dy
    return math.sqrt(total / len(lp))


def _scores(scene: dict) -> list[float]:
    reference = scene["reference_points"]
    candidates = scene["candidate_points"]
    order = scene["option_order"]
    return [_alignment_error(reference, candidates[name]) for name in order]


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    return "ABC"[min(range(3), key=lambda index: scores[index])]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene))
    return scores[1] - scores[0]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def _draw_plate(
    draw: ImageDraw.ImageDraw,
    center: list[int],
    points: dict[str, list[float]],
    palette: dict[str, str],
    *,
    size: tuple[int, int],
) -> None:
    cx, cy = center
    pw, ph = size
    draw.rounded_rectangle(
        (cx - pw // 2, cy - ph // 2, cx + pw // 2, cy + ph // 2),
        radius=18,
        fill=palette["plate"],
        outline=palette["ink"],
        width=3,
    )
    for name in COLOR_ORDER:
        px = int(round(cx + points[name][0]))
        py = int(round(cy + points[name][1]))
        draw.ellipse((px - 13, py - 13, px + 13, py + 13), fill=palette["ink"])
        draw.ellipse((px - 9, py - 9, px + 9, py + 9), fill=palette[name])
        draw.ellipse((px - 3, py - 3, px + 3, py + 3), outline="#ffffff", width=2)


def render(scene: dict) -> Image.Image:
    canvas = scene["canvas"]
    palette = scene["palette"]
    layout = scene["panel_layout"]
    reference = scene["reference_points"]
    candidates = scene["candidate_points"]
    order = scene["option_order"]
    image = Image.new("RGB", tuple(canvas), palette["background"])
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    draw.text((329, 18), "REFERENCE", fill=palette["ink"], font=font)
    _draw_plate(draw, layout["reference"], reference, palette, size=(210, 150))
    draw.line((40, 252, 680, 252), fill="#b8c0c7", width=2)
    for index, kind in enumerate(order):
        center = layout["options"][index]
        _draw_plate(draw, center, candidates[kind], palette, size=(205, 150))
        label = "ABC"[index]
        draw.rounded_rectangle(
            (center[0] - 15, 471, center[0] + 15, 501),
            radius=6,
            fill=palette["ink"],
        )
        draw.text((center[0] - 3, 480), label, fill="#ffffff", font=font)
    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["palette"] = {
        name: value.upper() if isinstance(value, str) and value.startswith("#") else value
        for name, value in twin["palette"].items()
    }
    return [("hex_case_alias", twin)]
