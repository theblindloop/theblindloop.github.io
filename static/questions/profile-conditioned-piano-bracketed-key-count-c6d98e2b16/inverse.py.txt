"""Independent pixel-only inverse for the piano adjacency question.

The inverse does not import renderer state. It finds large components close to
the public light and dark key swatches, orders their centroids, and evaluates
the visible two-sided neighbor relation on that ordered raster list.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "piano-bracketed-key-count-oracle-0.1.0"

_LIGHT_SWATCHES = np.asarray(
    ((244, 222, 160), (205, 232, 220), (231, 208, 239)), dtype=np.float32
)
_DARK_SWATCHES = np.asarray(
    ((39, 53, 68), (55, 50, 67), (44, 66, 58)), dtype=np.float32
)
# Solid interiors are exact public swatch colors. A tight radius avoids
# mistaking antialiased frame pixels for the pale key class.
_SWATCH_TOLERANCE = 10.0
_MIN_COMPONENT_AREA = 500


def _swatch_mask(arr: np.ndarray, swatches: np.ndarray) -> np.ndarray:
    distances = np.linalg.norm(arr[..., None, :] - swatches[None, None, :, :], axis=-1)
    return distances.min(axis=-1) <= _SWATCH_TOLERANCE


def _components(mask: np.ndarray) -> list[tuple[np.ndarray, np.ndarray]]:
    """Return large 8-connected components as (ys, xs), using no scene data."""

    height, width = mask.shape
    remaining = mask.copy()
    found: list[tuple[np.ndarray, np.ndarray]] = []
    while remaining.any():
        y0, x0 = np.argwhere(remaining)[0]
        queue = deque([(int(y0), int(x0))])
        remaining[y0, x0] = False
        ys: list[int] = []
        xs: list[int] = []
        while queue:
            y, x = queue.popleft()
            ys.append(y)
            xs.append(x)
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dy or dx):
                        continue
                    yy = y + dy
                    xx = x + dx
                    if 0 <= yy < height and 0 <= xx < width and remaining[yy, xx]:
                        remaining[yy, xx] = False
                        queue.append((yy, xx))
        if len(xs) >= _MIN_COMPONENT_AREA:
            found.append((np.asarray(ys, dtype=np.float64), np.asarray(xs, dtype=np.float64)))
    return found


def _component_centers(components: list[tuple[np.ndarray, np.ndarray]]) -> list[tuple[float, float]]:
    return [(float(xs.mean()) + 0.5, float(ys.mean()) + 0.5) for ys, xs in components]


def decision_from_image(image: Image.Image) -> str:
    """Recover the bracket count from the final RGB raster alone."""

    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    light_components = _components(_swatch_mask(arr, _LIGHT_SWATCHES))
    dark_components = _components(_swatch_mask(arr, _DARK_SWATCHES))
    light_x = _component_centers(light_components)
    dark_x = _component_centers(dark_components)
    if len(light_x) + len(dark_x) < 3 or not light_x or not dark_x:
        raise ValueError("could not recover enough key components from the raster")

    ordered = [(x, "light") for x, _ in light_x] + [(x, "dark") for x, _ in dark_x]
    ordered.sort(key=lambda item: item[0])
    answer = sum(
        role == "dark"
        and index > 0
        and index + 1 < len(ordered)
        and ordered[index - 1][1] == "light"
        and ordered[index + 1][1] == "light"
        for index, (_, role) in enumerate(ordered)
    )
    return str(answer)
