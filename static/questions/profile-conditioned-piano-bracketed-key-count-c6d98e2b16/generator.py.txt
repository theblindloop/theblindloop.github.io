"""Deterministic renderer and analytic world for the piano adjacency question."""

from __future__ import annotations

import copy
import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1000
HEIGHT = 480
SUPERSAMPLE = 4

# These public swatches are duplicated by oracle.py rather than imported there.
LIGHT_SWATCHES = (
    (244, 222, 160),
    (205, 232, 220),
    (231, 208, 239),
)
DARK_SWATCHES = (
    (39, 53, 68),
    (55, 50, 67),
    (44, 66, 58),
)
BACKGROUND_SWATCHES = (
    (245, 249, 252),
    (250, 246, 239),
)
FRAME_FILL_SWATCHES = (
    (202, 208, 211),
    (213, 205, 198),
)
FRAME_OUTLINE = (88, 101, 110)
KEY_OUTLINE = (112, 121, 128)
TITLE_COLOR = (63, 78, 91)
BED_FILL = (159, 169, 176)


def _rgb(value: list[int] | tuple[int, int, int]) -> tuple[int, int, int]:
    return int(value[0]), int(value[1]), int(value[2])


def _role_sequence(rng: random.Random, n_slots: int, dark_count: int) -> list[str]:
    dark_positions = set(rng.sample(range(n_slots), dark_count))
    return ["dark" if index in dark_positions else "light" for index in range(n_slots)]


def _make_keys(
    roles: list[str],
    *,
    inner_left: float,
    inner_right: float,
    key_top: float,
    key_height: float,
    gap: float,
) -> list[dict[str, Any]]:
    pitch = (inner_right - inner_left) / len(roles)
    keys: list[dict[str, Any]] = []
    for index, role in enumerate(roles):
        x0 = inner_left + index * pitch + gap / 2.0
        x1 = inner_left + (index + 1) * pitch - gap / 2.0
        y0 = key_top
        y1 = key_top + key_height * (0.64 if role == "dark" else 1.0)
        keys.append(
            {
                "role": role,
                "x0": round(x0, 3),
                "x1": round(x1, 3),
                "y0": round(y0, 3),
                "y1": round(y1, 3),
            }
        )
    return keys


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one styled keyboard strip with an independently permuted key role list."""

    rng = random.Random(int(seed))
    n_slots = rng.randint(9, 16)
    dark_count = rng.randint(3, min(6, n_slots - 3))
    roles = _role_sequence(rng, n_slots, dark_count)

    frame_left = rng.uniform(58.0, 112.0)
    frame_right = WIDTH - rng.uniform(58.0, 112.0)
    frame_top = rng.uniform(150.0, 176.0)
    frame_bottom = rng.uniform(410.0, 438.0)
    inner_left = frame_left + rng.uniform(22.0, 30.0)
    inner_right = frame_right - rng.uniform(22.0, 30.0)
    key_top = frame_top + rng.uniform(33.0, 43.0)
    key_height = rng.uniform(154.0, 188.0)
    # The near-boundary band is explicit and appears occasionally in galleries;
    # all clear bands leave ample room for two distinct component masks.
    gap_band = rng.random()
    if gap_band < 0.12:
        gap = rng.uniform(1.5, 2.6)
    elif gap_band < 0.55:
        gap = rng.uniform(4.5, 7.0)
    else:
        gap = rng.uniform(8.0, 12.0)

    light_fill = rng.choice(LIGHT_SWATCHES)
    dark_fill = rng.choice(DARK_SWATCHES)
    background = rng.choice(BACKGROUND_SWATCHES)
    frame_fill = rng.choice(FRAME_FILL_SWATCHES)
    keys = _make_keys(
        roles,
        inner_left=inner_left,
        inner_right=inner_right,
        key_top=key_top,
        key_height=key_height,
        gap=gap,
    )
    return {
        "canvas_width": WIDTH,
        "canvas_height": HEIGHT,
        "background": list(background),
        "frame_fill": list(frame_fill),
        "frame_outline": list(FRAME_OUTLINE),
        "title": "PIANO",
        "title_color": list(TITLE_COLOR),
        "light_fill": list(light_fill),
        "dark_fill": list(dark_fill),
        "key_gap": round(gap, 6),
        "frame_left": round(frame_left, 3),
        "frame_right": round(frame_right, 3),
        "frame_top": round(frame_top, 3),
        "frame_bottom": round(frame_bottom, 3),
        "bed_height": 23,
        "keys": keys,
    }


def analytic_gold(scene: dict[str, Any]) -> str:
    """Count dark entries whose two immediate ordered neighbors are light."""

    roles = [str(key["role"]) for key in scene["keys"]]
    answer = sum(
        roles[index] == "dark"
        and roles[index - 1] == "light"
        and roles[index + 1] == "light"
        for index in range(1, len(roles) - 1)
    )
    return str(answer)


def margin(scene: dict[str, Any]) -> float:
    """Return the smallest visible key-to-key or frame clearance in pixels."""

    keys = scene["keys"]
    values = [float(keys[index + 1]["x0"]) - float(keys[index]["x1"]) for index in range(len(keys) - 1)]
    values.extend(
        [
            float(keys[0]["x0"]) - float(scene["frame_left"]) - 12.0,
            float(scene["frame_right"]) - float(keys[-1]["x1"]) - 12.0,
            float(keys[0]["y0"]) - float(scene["frame_top"]) - 12.0,
        ]
    )
    return round(min(values), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 3.0


def counterfactual_scene(scene: dict[str, Any]) -> dict[str, Any]:
    """Permute roles while retaining style, geometry, and the number of dark keys."""

    twin = copy.deepcopy(scene)
    roles = [str(key["role"]) for key in twin["keys"]]
    original = analytic_gold(scene)
    candidates: list[list[str]] = []
    for shift in range(1, len(roles)):
        candidates.append(roles[shift:] + roles[:shift])
    candidates.append(list(reversed(roles)))
    # A deterministic role permutation is enough: the visible key geometry and
    # dark-key area are unchanged, while the adjacency label can change.
    for replacement in candidates:
        if replacement == roles:
            continue
        trial = copy.deepcopy(twin)
        for key, role in zip(trial["keys"], replacement, strict=True):
            key["role"] = role
        if analytic_gold(trial) != original:
            return trial
    return twin


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only the visible key roles and declared styling."""

    ss = SUPERSAMPLE
    image = Image.new(
        "RGB",
        (int(scene["canvas_width"]) * ss, int(scene["canvas_height"]) * ss),
        _rgb(scene["background"]),
    )
    draw = ImageDraw.Draw(image)

    def box(values: list[float] | tuple[float, ...]) -> list[int]:
        return [round(float(value) * ss) for value in values]

    font = ImageFont.load_default()
    draw.text(
        (round(30 * ss), round(32 * ss)),
        str(scene["title"]),
        fill=_rgb(scene["title_color"]),
        font=font,
    )
    frame = [
        float(scene["frame_left"]),
        float(scene["frame_top"]),
        float(scene["frame_right"]),
        float(scene["frame_bottom"]),
    ]
    draw.rounded_rectangle(
        box(frame),
        radius=18 * ss,
        fill=_rgb(scene["frame_fill"]),
        outline=_rgb(scene["frame_outline"]),
        width=4 * ss,
    )
    for key in scene["keys"]:
        fill = scene["dark_fill"] if key["role"] == "dark" else scene["light_fill"]
        draw.rectangle(
            box([key["x0"], key["y0"], key["x1"], key["y1"]]),
            fill=_rgb(fill),
            outline=KEY_OUTLINE,
            width=2 * ss,
        )
    bed_top = float(scene["frame_bottom"]) - float(scene["bed_height"])
    draw.rectangle(
        box(
            [
                float(scene["frame_left"]) + 12.0,
                bed_top,
                float(scene["frame_right"]) - 12.0,
                float(scene["frame_bottom"]) - 10.0,
            ]
        ),
        fill=BED_FILL,
    )
    return image.resize((int(scene["canvas_width"]), int(scene["canvas_height"])), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No latent aliases are claimed; every label-bearing ordered role is rendered."""

    return []
