"""Deterministic analytic arm for nested-box hierarchy correspondence."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1000, 700
SCALE = 2
BACKGROUND = (247, 248, 250)
PANEL_FILL = (255, 255, 252)
INK = (28, 32, 39)
NEUTRAL = (145, 149, 157)
MAGENTA = (190, 50, 150)
DOT_RGB = {
    "red": (218, 68, 73),
    "green": (43, 157, 104),
    "blue": (54, 105, 204),
}
COLORS = ("red", "green", "blue")
BRANCH_KINDS = ("chain", "fork", "deepfork")
ROOT_BOXES = ((25, 100, 465, 675), (535, 100, 975, 675))
BRANCH_W, BRANCH_H = 120, 520


def _branch_code(kind: str) -> tuple:
    """Canonical unordered rooted-tree code, including the branch root."""
    if kind == "chain":
        return (((),),)
    if kind == "fork":
        return ((), ())
    if kind == "deepfork":
        return (((), ()),)
    raise ValueError(f"unknown branch kind {kind!r}")


def _depth_profile(code: tuple) -> tuple[int, ...]:
    current = [code]
    counts = []
    while current:
        counts.append(len(current))
        current = [child for node in current for child in node]
    return tuple(counts)


def _tree_distance(left: tuple, right: tuple) -> int:
    a, b = _depth_profile(left), _depth_profile(right)
    length = max(len(a), len(b))
    return sum(
        abs((a[index] if index < len(a) else 0) - (b[index] if index < len(b) else 0))
        for index in range(length)
    )


def sample_scene(seed: int) -> dict:
    """Sample balanced colors and independently rearranged hierarchy layouts."""
    seed = int(seed)
    rng = random.Random(seed)
    kinds = list(BRANCH_KINDS)
    marked_kind = kinds[(seed // 3) % len(kinds)]
    answer_color = COLORS[seed % len(COLORS)]

    remaining_colors = [color for color in COLORS if color != answer_color]
    rng.shuffle(remaining_colors)
    target_colors = {marked_kind: answer_color}
    for kind, color in zip((kind for kind in kinds if kind != marked_kind), remaining_colors):
        target_colors[kind] = color

    reference_order = kinds.copy()
    target_order = kinds.copy()
    rng.shuffle(reference_order)
    rng.shuffle(target_order)
    if target_order == reference_order:
        target_order = target_order[1:] + target_order[:1]

    layout_variants = {
        panel: {kind: rng.randrange(2) for kind in kinds}
        for panel in ("reference", "target")
    }
    draw_orders = {
        "reference": rng.sample(kinds, len(kinds)),
        "target": rng.sample(kinds, len(kinds)),
    }
    scene = {
        "branch_kinds": kinds,
        "reference_order": reference_order,
        "target_order": target_order,
        "marked_kind": marked_kind,
        "target_colors": target_colors,
        "layout_variants": layout_variants,
        "draw_orders": draw_orders,
    }
    if is_quarantined(scene):
        raise RuntimeError("sampler produced an invalid hierarchy")
    return scene


def analytic_gold(scene: dict) -> str:
    """Read the target color only after validating the marked branch identity."""
    kinds = list(scene["branch_kinds"])
    marked = scene["marked_kind"]
    colors = scene["target_colors"]
    if marked not in kinds or set(colors) != set(kinds):
        raise ValueError("marked branch or target color binding is malformed")
    answer = colors[marked]
    if answer not in COLORS:
        raise ValueError("target color is outside the answer set")
    return answer


def margin(scene: dict) -> float:
    """Minimum structural edit gap from the marked branch to a runner-up."""
    marked = scene["marked_kind"]
    kinds = list(scene["branch_kinds"])
    correct = _branch_code(marked)
    distances = [_tree_distance(correct, _branch_code(kind)) for kind in kinds if kind != marked]
    return float(min(distances)) if distances else 0.0


def is_quarantined(scene: dict) -> bool:
    """Reject ambiguous trees, invalid permutations, or unsafe layout state."""
    try:
        kinds = list(scene["branch_kinds"])
        if kinds != list(BRANCH_KINDS):
            return True
        if len({_branch_code(kind) for kind in kinds}) != len(kinds):
            return True
        for key in ("reference_order", "target_order"):
            if sorted(scene[key]) != sorted(kinds):
                return True
        if scene["marked_kind"] not in kinds:
            return True
        if set(scene["target_colors"]) != set(kinds):
            return True
        if sorted(scene["target_colors"].values()) != sorted(COLORS):
            return True
        for panel in ("reference", "target"):
            variants = scene["layout_variants"][panel]
            if set(variants) != set(kinds) or any(value not in (0, 1) for value in variants.values()):
                return True
            if sorted(scene["draw_orders"][panel]) != sorted(kinds):
                return True
        _ = analytic_gold(scene)
        return margin(scene) < 1.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse separated sibling draw order without changing pixels or gold."""
    twin = copy.deepcopy(scene)
    twin["draw_orders"]["reference"].reverse()
    twin["draw_orders"]["target"].reverse()
    return [("reverse_separated_sibling_draw_order", twin)]


def _sbox(box: tuple[int, int, int, int]) -> tuple[int, int, int, int]:
    return tuple(int(value * SCALE) for value in box)


def _outline(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], width: int = 4) -> None:
    draw.rectangle(_sbox(box), fill=PANEL_FILL, outline=INK, width=width * SCALE)


def _draw_branch(
    draw: ImageDraw.ImageDraw,
    box: tuple[int, int, int, int],
    kind: str,
    variant: int,
) -> None:
    x0, y0, x1, y1 = box
    _outline(draw, box)
    shift = 8 if variant else 0
    if kind == "chain":
        child = (x0 + 10 + shift, y0 + 62, x1 - 18 + shift, y1 - 28)
        _outline(draw, child)
        cx0, cy0, cx1, cy1 = child
        grand = (cx0 + 12 + (0 if variant else 6), cy0 + 62, cx1 - 12 + (0 if variant else 6), cy1 - 44)
        _outline(draw, grand)
    elif kind == "fork":
        inset = 10 + shift
        for top in (y0 + 72, y0 + 292):
            _outline(draw, (x0 + inset, top, x1 - (18 - shift), top + 158))
    elif kind == "deepfork":
        child = (x0 + 10 + shift, y0 + 60, x1 - (18 - shift), y1 - 28)
        _outline(draw, child)
        cx0, cy0, cx1, _ = child
        left_inset, right_inset = ((16, 12) if variant else (12, 16))
        for top in (cy0 + 68, cy0 + 250):
            _outline(draw, (cx0 + left_inset, top, cx1 - right_inset, top + 128))
    else:
        raise ValueError(f"unknown branch kind {kind!r}")


def _dot(draw: ImageDraw.ImageDraw, center: tuple[int, int], rgb: tuple[int, int, int]) -> None:
    x, y = center
    radius = 11
    draw.ellipse(_sbox((x - radius, y - radius, x + radius, y + radius)), fill=rgb, outline=(255, 255, 255), width=2 * SCALE)


def render(scene: dict) -> Image.Image:
    """Render all answer-bearing fields into an antialiased RGB image."""
    kinds = list(scene["branch_kinds"])
    reference_order = list(scene["reference_order"])
    target_order = list(scene["target_order"])
    marked = scene["marked_kind"]
    target_colors = scene["target_colors"]
    variants = scene["layout_variants"]
    draw_orders = scene["draw_orders"]
    if sorted(kinds) != sorted(BRANCH_KINDS):
        raise ValueError("unexpected branch set")

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default(size=25 * SCALE)
    draw.text((245 * SCALE, 48 * SCALE), "REFERENCE", fill=INK, anchor="mm", font=font)
    draw.text((755 * SCALE, 48 * SCALE), "TARGET", fill=INK, anchor="mm", font=font)

    panel_data = (
        ("reference", ROOT_BOXES[0], reference_order),
        ("target", ROOT_BOXES[1], target_order),
    )
    for panel, root, order in panel_data:
        _outline(draw, root, width=5)
        slot_for = {kind: index for index, kind in enumerate(order)}
        boxes = {}
        for kind in kinds:
            slot = slot_for[kind]
            x0 = root[0] + 18 + slot * 137
            boxes[kind] = (x0, root[1] + 28, x0 + BRANCH_W, root[1] + 28 + BRANCH_H)
        for kind in draw_orders[panel]:
            _draw_branch(draw, boxes[kind], kind, int(variants[panel][kind]))
        for kind in kinds:
            x0, y0, x1, _ = boxes[kind]
            color = NEUTRAL if panel == "reference" else DOT_RGB[target_colors[kind]]
            if panel == "reference" and kind == marked:
                color = MAGENTA
            _dot(draw, ((x0 + x1) // 2, y0 + 30), color)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
