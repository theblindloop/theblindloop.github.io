"""Render nine rotatable cards whose matching edge codes reveal an arrow."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 840, 660
BACKGROUND = (247, 249, 252)
PAPER = (255, 255, 255)
FRAME = (78, 91, 107)
INK = (18, 35, 58)
GUIDE = (198, 207, 217)
DISTRACTOR = (218, 225, 232)

# The public question gives these boundary-side meanings.
TOP = (232, 174, 35)
RIGHT = (225, 99, 45)
BOTTOM = (145, 78, 190)
LEFT = (30, 157, 177)
BOUNDARIES = (TOP, RIGHT, BOTTOM, LEFT)

LABELS = ("up", "right", "down", "left")
CONNECTOR_COLORS = (
    (193, 48, 90), (42, 116, 207), (42, 153, 92), (117, 87, 196),
    (206, 79, 43), (34, 137, 153), (181, 74, 151), (104, 129, 45),
    (67, 91, 181), (194, 111, 32), (35, 142, 118), (155, 68, 82),
)

CARD_SIZE = 96
CELL = 32
CARD_X = (28, 198, 368, 538, 708, 113, 283, 453, 623)
CARD_Y = (300, 300, 300, 300, 300, 505, 505, 505, 505)


def _blank() -> list[list[int]]:
    return [[0 for _ in range(9)] for _ in range(9)]


def _up() -> list[list[int]]:
    grid = _blank()
    for row, cols in (
        (1, (4,)),
        (2, (3, 4, 5)),
        (3, (2, 4, 6)),
        (4, (4,)),
        (5, (4,)),
        (6, (4,)),
        (7, (4,)),
    ):
        for col in cols:
            grid[row][col] = 1
    return grid


def _rotate_cw(grid: list[list[int]], turns: int) -> list[list[int]]:
    out = [row[:] for row in grid]
    for _ in range(turns % 4):
        out = [list(row) for row in zip(*out[::-1])]
    return out


def _rotate_edges_cw(edges: tuple[tuple[int, int, int], ...], turns: int) -> tuple[tuple[int, int, int], ...]:
    out = tuple(edges)
    for _ in range(turns % 4):
        out = (out[3], out[0], out[1], out[2])
    return out


def full_template(label: str) -> list[list[int]]:
    if label not in LABELS:
        raise ValueError(f"unknown glyph {label!r}")
    return _rotate_cw(_up(), LABELS.index(label))


def _hamming(a: list[list[int]], b: list[list[int]]) -> int:
    return sum(x != y for ar, br in zip(a, b, strict=True) for x, y in zip(ar, br, strict=True))


def sample_scene(seed: int) -> dict:
    """Counterbalance label against every nuisance in blocks of four seeds."""
    glyph = LABELS[int(seed) % 4]
    rng = random.Random((int(seed) // 4) * 104729 + 2801)
    source_order = list(range(9))
    draw_order = list(range(9))
    rng.shuffle(source_order)
    rng.shuffle(draw_order)
    turns = [rng.randrange(4) for _ in range(9)]
    connector_palette = [list(color) for color in CONNECTOR_COLORS]
    rng.shuffle(connector_palette)
    shift = [rng.randint(-6, 6), rng.randint(-6, 6)]
    anchors = ((55, 58), (785, 57), (80, 205), (760, 206), (225, 115), (615, 118))
    distractors = [
        [x + rng.randint(-8, 8), y + rng.randint(-8, 8), rng.choice((4, 5, 6))]
        for x, y in anchors
    ]
    distractor_order = list(range(len(distractors)))
    rng.shuffle(distractor_order)
    return {
        "glyph": glyph,
        "source_order": source_order,
        "turns": turns,
        "connector_palette": connector_palette,
        "draw_order": draw_order,
        "shift": shift,
        "distractors": distractors,
        "distractor_order": distractor_order,
    }


def _fragment(full: list[list[int]], piece: int) -> list[list[int]]:
    row0, col0 = 3 * (piece // 3), 3 * (piece % 3)
    return [row[col0:col0 + 3] for row in full[row0:row0 + 3]]


def _canonical_edges(piece: int, palette: list[list[int]]) -> tuple[tuple[int, int, int], ...]:
    row, col = divmod(piece, 3)
    top = TOP if row == 0 else tuple(palette[6 + (row - 1) * 3 + col])
    right = RIGHT if col == 2 else tuple(palette[row * 2 + col])
    bottom = BOTTOM if row == 2 else tuple(palette[6 + row * 3 + col])
    left = LEFT if col == 0 else tuple(palette[row * 2 + col - 1])
    return top, right, bottom, left


def _draw_connector(draw: ImageDraw.ImageDraw, x0: int, y0: int, side: int, color: tuple[int, int, int]) -> None:
    x1, y1 = x0 + CARD_SIZE, y0 + CARD_SIZE
    if side == 0:
        box = (x0 + 34, y0 - 14, x0 + 62, y0 + 2)
    elif side == 1:
        box = (x1 - 2, y0 + 34, x1 + 14, y0 + 62)
    elif side == 2:
        box = (x0 + 34, y1 - 2, x0 + 62, y1 + 14)
    else:
        box = (x0 - 14, y0 + 34, x0 + 2, y0 + 62)
    draw.rectangle(box, fill=color)


def _draw_card(draw: ImageDraw.ImageDraw, x0: int, y0: int, grid: list[list[int]], edges: tuple[tuple[int, int, int], ...]) -> None:
    draw.rectangle((x0, y0, x0 + CARD_SIZE, y0 + CARD_SIZE), fill=PAPER)
    draw.rectangle((x0, y0, x0 + CARD_SIZE, y0 + CARD_SIZE), outline=FRAME, width=4)
    for index in (1, 2):
        draw.line((x0 + index * CELL, y0, x0 + index * CELL, y0 + CARD_SIZE), fill=FRAME, width=3)
        draw.line((x0, y0 + index * CELL, x0 + CARD_SIZE, y0 + index * CELL), fill=FRAME, width=3)
    for row in range(3):
        for col in range(3):
            if grid[row][col]:
                cx, cy = x0 + col * CELL + 16, y0 + row * CELL + 16
                draw.rectangle((cx - 9, cy - 9, cx + 9, cy + 9), fill=INK)
    for side, color in enumerate(edges):
        _draw_connector(draw, x0, y0, side, color)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    dx, dy = (int(value) for value in scene["shift"])

    for index in scene["distractor_order"]:
        x, y, radius = scene["distractors"][index]
        x, y, radius = int(x) + dx, int(y) + dy, int(radius)
        draw.polygon(((x, y - radius), (x + radius, y), (x, y + radius), (x - radius, y)), fill=DISTRACTOR)

    # A neutral empty board declares the 3-by-3 output topology without encoding placement.
    bx, by, size = 330 + dx, 36 + dy, 180
    draw.rectangle((bx, by, bx + size, by + size), fill=PAPER, outline=GUIDE, width=4)
    for index in (1, 2):
        draw.line((bx + index * 60, by, bx + index * 60, by + size), fill=GUIDE, width=3)
        draw.line((bx, by + index * 60, bx + size, by + index * 60), fill=GUIDE, width=3)
    draw.line((bx, by - 10, bx + size, by - 10), fill=TOP, width=7)
    draw.line((bx + size + 10, by, bx + size + 10, by + size), fill=RIGHT, width=7)
    draw.line((bx, by + size + 10, bx + size, by + size + 10), fill=BOTTOM, width=7)
    draw.line((bx - 10, by, bx - 10, by + size), fill=LEFT, width=7)

    full = full_template(scene["glyph"])
    palette = scene["connector_palette"]
    for slot in scene["draw_order"]:
        piece = int(scene["source_order"][slot])
        turn = int(scene["turns"][piece])
        displayed = _rotate_cw(_fragment(full, piece), turn)
        edges = _rotate_edges_cw(_canonical_edges(piece, palette), turn)
        _draw_card(draw, CARD_X[slot] + dx, CARD_Y[slot] + dy, displayed, edges)
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["glyph"])


def margin(scene: dict) -> float:
    label = str(scene["glyph"])
    truth = full_template(label)
    return float(min(_hamming(truth, full_template(other)) for other in LABELS if other != label))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["draw_order"] = list(reversed(twin["draw_order"]))
    twin["distractor_order"] = list(reversed(twin["distractor_order"]))
    return [("reverse_disjoint_draw_orders", twin)]
