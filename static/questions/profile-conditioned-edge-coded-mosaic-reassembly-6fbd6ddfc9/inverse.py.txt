"""Independent pixel-only solver for the visible edge-matching puzzle."""

from __future__ import annotations

from collections import Counter

import numpy as np
from PIL import Image

_BACKGROUND = (247, 249, 252)
_PAPER = (255, 255, 255)
_FRAME = (78, 91, 107)
_INK = (18, 35, 58)
_TOP = (232, 174, 35)
_RIGHT = (225, 99, 45)
_BOTTOM = (145, 78, 190)
_LEFT = (30, 157, 177)
_BOUNDARIES = (_TOP, _RIGHT, _BOTTOM, _LEFT)
_LABELS = ("up", "right", "down", "left")


class OracleAbstention(ValueError):
    """Raised when visible evidence does not determine one answer."""


def _components(mask: np.ndarray, minimum: int) -> list[dict]:
    height, width = mask.shape
    remaining = set(int(value) for value in np.flatnonzero(mask))
    found: list[dict] = []
    while remaining:
        seed = remaining.pop()
        stack = [seed]
        area = 0
        min_x = max_x = seed % width
        min_y = max_y = seed // width
        sum_x = sum_y = 0
        while stack:
            value = stack.pop()
            y, x = divmod(value, width)
            area += 1
            sum_x += x
            sum_y += y
            min_x, max_x = min(min_x, x), max(max_x, x)
            min_y, max_y = min(min_y, y), max(max_y, y)
            for neighbor, allowed in (
                (value - 1, x > 0), (value + 1, x + 1 < width),
                (value - width, y > 0), (value + width, y + 1 < height),
            ):
                if allowed and neighbor in remaining:
                    remaining.remove(neighbor)
                    stack.append(neighbor)
        if area >= minimum:
            found.append({
                "area": area,
                "bbox": (min_x, min_y, max_x, max_y),
                "center": (sum_x / area, sum_y / area),
            })
    return found


def _rotate_cw(grid: list[list[int]], turns: int) -> list[list[int]]:
    out = [row[:] for row in grid]
    for _ in range(turns % 4):
        out = [list(row) for row in zip(*out[::-1])]
    return out


def _rotate_edges_cw(edges: tuple[tuple[int, int, int], ...], turns: int) -> tuple[tuple[int, int, int], ...]:
    out = tuple(edges)
    for _ in range(turns % 4):
        out = (out[3], out[0], out[1], out[2])
    return out


def _up() -> list[list[int]]:
    grid = [[0 for _ in range(9)] for _ in range(9)]
    for row, cols in (
        (1, (4,)), (2, (3, 4, 5)), (3, (2, 4, 6)),
        (4, (4,)), (5, (4,)), (6, (4,)), (7, (4,)),
    ):
        for col in cols:
            grid[row][col] = 1
    return grid


def _templates() -> dict[str, list[list[int]]]:
    base = _up()
    return {label: _rotate_cw(base, index) for index, label in enumerate(_LABELS)}


def _card_frames(arr: np.ndarray) -> list[dict]:
    mask = np.all(arr == np.asarray(_FRAME, dtype=np.uint8), axis=2)
    frames = _components(mask, minimum=650)
    square = []
    for item in frames:
        x0, y0, x1, y1 = item["bbox"]
        width, height = x1 - x0, y1 - y0
        if 90 <= width <= 102 and 90 <= height <= 102 and abs(width - height) <= 2:
            square.append(item)
    if len(square) != 9:
        raise OracleAbstention(f"expected nine visible card grids, found {len(square)}")
    square.sort(key=lambda item: (item["center"][1], item["center"][0]))
    return square


def _mode_color(patch: np.ndarray) -> tuple[int, int, int]:
    colors, counts = np.unique(patch.reshape(-1, 3), axis=0, return_counts=True)
    index = int(np.argmax(counts))
    if int(counts[index]) < int(0.85 * patch.shape[0] * patch.shape[1]):
        raise OracleAbstention("an edge bar has no stable visible color")
    color = tuple(int(value) for value in colors[index])
    if color in {_BACKGROUND, _PAPER, _FRAME, _INK}:
        raise OracleAbstention("an edge connector is missing or structural-colored")
    return color


def _read_card(arr: np.ndarray, frame: dict) -> dict:
    x0, y0, x1, y1 = frame["bbox"]
    width, height = x1 - x0, y1 - y0
    grid = []
    for row in range(3):
        values = []
        for col in range(3):
            cx = int(round(x0 + (col + 0.5) * width / 3.0))
            cy = int(round(y0 + (row + 0.5) * height / 3.0))
            patch = arr[cy - 4:cy + 5, cx - 4:cx + 5]
            ink = float(np.mean(np.all(patch == np.asarray(_INK, dtype=np.uint8), axis=2)))
            if ink > 0.85:
                values.append(1)
            elif ink < 0.05:
                values.append(0)
            else:
                raise OracleAbstention("a tile center has ambiguous ink")
        grid.append(values)
    cx, cy = (int(round(value)) for value in frame["center"])
    edge_centers = ((cx, y0 - 8), (x1 + 8, cy), (cx, y1 + 8), (x0 - 8, cy))
    edges = []
    for ex, ey in edge_centers:
        patch = arr[ey - 3:ey + 4, ex - 3:ex + 4]
        if patch.shape != (7, 7, 3):
            raise OracleAbstention("an edge connector falls outside the raster")
        edges.append(_mode_color(patch))
    return {"grid": grid, "edges": tuple(edges)}


def _solve(cards: list[dict]) -> list[dict]:
    variants = []
    for card in cards:
        variants.append([
            {
                "grid": _rotate_cw(card["grid"], turns),
                "edges": _rotate_edges_cw(card["edges"], turns),
            }
            for turns in range(4)
        ])

    placed: list[dict] = []
    used: set[int] = set()
    solutions: list[list[dict]] = []

    def legal(position: int, edges: tuple[tuple[int, int, int], ...]) -> bool:
        row, col = divmod(position, 3)
        expected_outer = (row == 0, col == 2, row == 2, col == 0)
        for side, is_outer in enumerate(expected_outer):
            if is_outer:
                if edges[side] != _BOUNDARIES[side]:
                    return False
            elif edges[side] in _BOUNDARIES:
                return False
        if row and edges[0] != placed[position - 3]["edges"][2]:
            return False
        if col and edges[3] != placed[position - 1]["edges"][1]:
            return False
        return True

    def visit(position: int) -> None:
        if len(solutions) > 1:
            return
        if position == 9:
            solutions.append([{"grid": [row[:] for row in item["grid"]], "edges": item["edges"]} for item in placed])
            return
        for card_index in range(9):
            if card_index in used:
                continue
            for variant in variants[card_index]:
                if not legal(position, variant["edges"]):
                    continue
                used.add(card_index)
                placed.append(variant)
                visit(position + 1)
                placed.pop()
                used.remove(card_index)

    visit(0)
    if len(solutions) != 1:
        raise OracleAbstention(f"edge constraints yield {len(solutions)} legal assemblies")
    return solutions[0]


def _assemble(solution: list[dict]) -> list[list[int]]:
    full = [[0 for _ in range(9)] for _ in range(9)]
    for position, card in enumerate(solution):
        row0, col0 = 3 * (position // 3), 3 * (position % 3)
        for row in range(3):
            for col in range(3):
                full[row0 + row][col0 + col] = card["grid"][row][col]
    return full


def _decode(image: Image.Image) -> tuple[str, float]:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    cards = [_read_card(arr, frame) for frame in _card_frames(arr)]
    assembled = _assemble(_solve(cards))
    templates = _templates()
    distances = {
        label: sum(
            value != templates[label][row][col]
            for row, values in enumerate(assembled)
            for col, value in enumerate(values)
        )
        for label in _LABELS
    }
    ranked = sorted(distances.items(), key=lambda item: (item[1], _LABELS.index(item[0])))
    best, best_distance = ranked[0]
    gap = float(ranked[1][1] - best_distance)
    if best_distance > 1 or gap < 8:
        raise OracleAbstention(f"arrow classification ambiguous: best={best_distance}, gap={gap}")
    return best, gap


def decision_with_margin(image: Image.Image) -> tuple[str, float]:
    """Recover the answer and runner-up Hamming margin from pixels alone."""
    return _decode(image)


def decision_from_image(image: Image.Image) -> str:
    return _decode(image)[0]
