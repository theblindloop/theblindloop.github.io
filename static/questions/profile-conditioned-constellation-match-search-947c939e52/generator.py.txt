"""Four-dot constellation search world."""

from __future__ import annotations

import itertools
import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 900, 560
LABELS = ("A", "B", "C", "D")
BLUE = (25, 95, 170)
INK = (31, 38, 48)
PANEL = (170, 177, 188)
ACCENT = (181, 55, 116)

BASE = ((-0.759, -0.596), (0.092, 0.445), (0.703, 0.623), (0.075, -0.599))
DISTRACTORS = (
    ((0.237, -0.105), (-0.789, -0.088), (-0.180, 0.605), (0.777, 0.621)),
    ((0.155, 0.556), (0.690, -0.452), (-0.620, -0.506), (-0.799, 0.478)),
    tuple((-x, y) for x, y in BASE),
    ((0.117, -0.180), (0.335, 0.515), (-0.324, 0.320), (-0.653, -0.560)),
    ((-0.460, -0.622), (-0.614, 0.369), (0.098, 0.285), (0.776, -0.508)),
)


def _transform(points: Iterable[tuple[float, float]], cx: float, cy: float,
               scale: float, angle: float) -> list[list[float]]:
    c, s = math.cos(angle), math.sin(angle)
    return [[cx + scale * (c*x - s*y), cy + scale * (s*x + c*y)] for x, y in points]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    winner = seed % 4
    reference = _transform(BASE, 450 + rng.uniform(-8, 8), 145 + rng.uniform(-5, 5),
                           rng.uniform(58, 67), rng.uniform(-math.pi, math.pi))
    choices = rng.sample(list(DISTRACTORS), 3)
    panels: list[list[list[float]]] = []
    d = 0
    for index, cx in enumerate((125, 340, 560, 775)):
        template = BASE if index == winner else choices[d]
        d += index != winner
        panels.append(_transform(template, cx + rng.uniform(-4, 4), 418 + rng.uniform(-4, 4),
                                 rng.uniform(52, 64), rng.uniform(-math.pi, math.pi)))
    return {"reference": reference, "panels": panels}


def _normalise(points: Iterable[Iterable[float]]) -> list[complex]:
    z = [complex(float(x), float(y)) for x, y in points]
    mean = sum(z) / len(z)
    z = [v - mean for v in z]
    norm = math.sqrt(sum(abs(v)**2 for v in z))
    return [v / norm for v in z]


def _match_error(left: Iterable[Iterable[float]], right: Iterable[Iterable[float]]) -> float:
    a = _normalise(left)
    b = _normalise(right)
    best = float("inf")
    for order in itertools.permutations(range(4)):
        bp = [b[i] for i in order]
        cross = sum(x.conjugate() * y for x, y in zip(bp, a))
        phase = cross / abs(cross) if abs(cross) else 1 + 0j
        err = math.sqrt(sum(abs(x * phase - y)**2 for x, y in zip(bp, a)))
        best = min(best, err)
    return best


def _scores(scene: dict) -> list[float]:
    reference = scene["reference"]
    panels = scene["panels"]
    return [_match_error(reference, points) for points in panels]


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    return LABELS[min(range(4), key=scores.__getitem__)]


def margin(scene: dict) -> float:
    ordered = sorted(_scores(scene))
    return round(ordered[1] - ordered[0], 9)


def is_quarantined(scene: dict) -> bool:
    scores = sorted(_scores(scene))
    return scores[0] > 0.035 or scores[1] - scores[0] < 0.12


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    path = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf" if bold else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
    try:
        return ImageFont.truetype(path, size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    reference = scene["reference"]
    panels = scene["panels"]
    scale = 3
    image = Image.new("RGB", (WIDTH*scale, HEIGHT*scale), "white")
    draw = ImageDraw.Draw(image)
    def rect(box, **kwargs):
        draw.rounded_rectangle(tuple(int(v*scale) for v in box), radius=12*scale, **kwargs)
    rect((330, 48, 570, 245), outline=ACCENT, width=3*scale)
    draw.text((450*scale, 24*scale), "REFERENCE", fill=ACCENT,
              font=_font(18*scale, True), anchor="mm")
    for index, cx in enumerate((125, 340, 560, 775)):
        rect((cx-92, 315, cx+92, 520), outline=PANEL, width=2*scale)
        draw.text((cx*scale, 535*scale), LABELS[index], fill=INK,
                  font=_font(22*scale, True), anchor="mm")
    for points in [reference, *panels]:
        for x, y in points:
            r = 10*scale
            draw.ellipse((int(x*scale)-r, int(y*scale)-r,
                          int(x*scale)+r, int(y*scale)+r), fill=BLUE)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
