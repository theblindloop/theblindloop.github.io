"""Independent PNG-only inverse arm for constellation matching."""

from __future__ import annotations

import itertools
import math
from collections import deque

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    seen = np.zeros(mask.shape, dtype=bool)
    found = []
    height, width = mask.shape
    for y, x in zip(*np.nonzero(mask & ~seen)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))]); seen[y, x] = True; pixels = []
        while queue:
            yy, xx = queue.popleft(); pixels.append((yy, xx))
            for dy, dx in ((-1,0),(1,0),(0,-1),(0,1)):
                ny, nx = yy+dy, xx+dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny,nx] and not seen[ny,nx]:
                    seen[ny,nx] = True; queue.append((ny,nx))
        if len(pixels) >= 120:
            ys = [p[0] for p in pixels]; xs = [p[1] for p in pixels]
            found.append((sum(xs)/len(xs), sum(ys)/len(ys), len(pixels)))
    return found


def _normalise(points):
    z = [complex(x, y) for x, y in points]
    mean = sum(z)/len(z); z = [v-mean for v in z]
    norm = math.sqrt(sum(abs(v)**2 for v in z))
    if norm < 1: raise ValueError("degenerate constellation")
    return [v/norm for v in z]


def _error(left, right):
    a, b = _normalise(left), _normalise(right)
    best = float("inf")
    for order in itertools.permutations(range(4)):
        bp = [b[i] for i in order]
        cross = sum(x.conjugate()*y for x,y in zip(bp,a))
        phase = cross/abs(cross) if abs(cross) else 1+0j
        best = min(best, math.sqrt(sum(abs(x*phase-y)**2 for x,y in zip(bp,a))))
    return best


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"))
    mask = (rgb[:,:,2] > 125) & (rgb[:,:,0] < 80) & (rgb[:,:,1] < 145)
    dots = _components(mask)
    if len(dots) != 20:
        raise ValueError(f"expected 20 visible blue dots, found {len(dots)}")
    dots.sort(key=lambda p: p[1])
    reference = [(x,y) for x,y,_ in dots[:4]]
    lower = sorted(dots[4:], key=lambda p: p[0])
    panels = [[(x,y) for x,y,_ in lower[i:i+4]] for i in range(0,16,4)]
    scores = [_error(reference, panel) for panel in panels]
    order = sorted(range(4), key=scores.__getitem__)
    if scores[order[0]] > 0.05 or scores[order[1]]-scores[order[0]] < 0.10:
        raise ValueError("ambiguous pixel registration")
    return LABELS[order[0]]
