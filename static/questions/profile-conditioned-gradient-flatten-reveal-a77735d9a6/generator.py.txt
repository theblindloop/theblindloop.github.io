"""gradient_flatten_reveal: planar-gradient flatten glyph reveal world.

Declared transform: flatten (remove) the best-fit linear brightness ramp that
runs across the image, then read the coarse diagonal-band glyph that the flat
field still carries. In the raw raster the band is masked by a strong left-to-
right brightness ramp plus per-cell noise, so no local patch shows it; only
subtracting the fitted planar gradient exposes the band. The band is the glyph:
rising (A) or falling (B).

Latent scene z = {noise, glyph, rampgain}. label = glyph.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RENDERER_VERSION = "gradient_flatten_reveal-0.1.0"

W = 512
H = 512
NGRID = 32          # cells per side
CELL = W // NGRID   # 16
BASE = 128.0
BIAS = 46.0
RAMPGAIN = 3.4      # brightness increase per cell column
NOISE_SIGMA = 38.0
BAND_WIDTH = 5      # half-width of the diagonal band in cell units
Q_TOL = 0.15


def band_masks() -> dict[str, np.ndarray]:
    """(NGRID, NGRID) bool masks: A rising '/', B falling '\\'."""
    r, c = np.indices((NGRID, NGRID))
    mask_a = np.abs((r + c) - (NGRID - 1)) <= BAND_WIDTH
    mask_b = np.abs(r - c) <= BAND_WIDTH
    return {"A": mask_a, "B": mask_b}


def build_scene(noise: int, glyph: str, rampgain: float = RAMPGAIN) -> dict:
    return {
        "noise": int(noise),
        "glyph": glyph,
        "rampgain": float(rampgain),
    }


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(int(seed) & 0xFFFFFFFF)
    glyph = "A" if rng.integers(0, 2) else "B"
    return build_scene(noise=int(seed), glyph=glyph)


def _glyph_mask(glyph: str) -> np.ndarray:
    return band_masks()[glyph]


def render_matrix(scene: dict) -> np.ndarray:
    """(NGRID, NGRID) mean-luminance of each lattice cell before ramp+noise."""
    n = NGRID
    r, c = np.indices((n, n))
    inside = _glyph_mask(scene["glyph"])
    sign = np.where(inside, 1.0, -1.0)
    bias = BIAS * sign
    nc2 = (n - 1) / 2.0
    ramp = float(scene["rampgain"]) * (c - nc2)
    rng = np.random.default_rng(int(scene["noise"]) & 0xFFFFFFFF)
    noise = rng.normal(0.0, NOISE_SIGMA, size=(n, n))
    return BASE + ramp + bias + noise


def render(scene: dict) -> Image.Image:
    cell = CELL
    mat = render_matrix(scene)
    arr = np.repeat(np.repeat(mat, cell, axis=0), cell, axis=1)
    arr = np.clip(arr, 0, 255)
    canvas = np.repeat(arr[:, :, None], 3, axis=2).astype(np.uint8)
    canvas = canvas[:W, :H]
    return Image.fromarray(canvas, "RGB")


def margin(scene: dict) -> float:
    chosen = _glyph_mask(scene["glyph"])
    other = _glyph_mask("B" if scene["glyph"] == "A" else "A")
    o_filled = int(other.sum())
    corr_other = float((chosen & other).sum()) / max(1, o_filled)
    return round(1.0 - corr_other, 6)


def analytic_gold(scene: dict) -> str:
    return scene["glyph"]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < Q_TOL


def latent_symmetries(scene: dict):
    """not_applicable: every latent field (the per-cell noise seed, the answer
    glyph, the ramp gain) directly changes the rendered pixels, so no distinct
    latent vector reproduces the same image. Declared empty."""
    return iter(())
