"""Independent pixel-only inverse arm for gradient_flatten_reveal.

Receives ONLY the final PNG. It:
  1. reads the mean luminance of every lattice cell;
  2. fits a planar (linear-in-row, linear-in-column) gradient by least squares;
  3. subtracts that fitted ramp, leaving a flat residual field;
  4. correlates the residual against the two declared diagonal-band templates
     A (rising) and B (falling);
  5. reports the template with the larger signed inside-vs-outside contrast.

No scene data, no analytic gold, no renderer imports. The residual is
polarity-invariant (relative inside/outside contrast), so a global brightness
inversion cannot flip the outcome.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "gradient_flatten_reveal-oracle-0.1.0"

W = 512
NGRID = 32
CELL = W // NGRID
BAND_WIDTH = 5


def _band_masks() -> dict[str, np.ndarray]:
    r, c = np.indices((NGRID, NGRID))
    mask_a = np.abs((r + c) - (NGRID - 1)) <= BAND_WIDTH
    mask_b = np.abs(r - c) <= BAND_WIDTH
    return {"A": mask_a, "B": mask_b}


def _cell_means(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("L"), dtype=np.float64)
    g = np.zeros((NGRID, NGRID), dtype=np.float64)
    for i in range(NGRID):
        for j in range(NGRID):
            cell = arr[i * CELL:(i + 1) * CELL, j * CELL:(j + 1) * CELL]
            g[i, j] = cell.mean()
    return g


def _flatten_planar(g: np.ndarray) -> np.ndarray:
    """Subtract the best-fit plane a + b*r + d*c from the cell grid."""
    r, c = np.indices(g.shape)
    ones = np.ones_like(g)
    A = np.stack([ones.ravel(), r.ravel(), c.ravel()], axis=1)
    coef, *_ = np.linalg.lstsq(A, g.ravel(), rcond=None)
    plane = (A @ coef).reshape(g.shape)
    return g - plane


def _contrast(res: np.ndarray, mask: np.ndarray) -> float:
    inside = res[mask].mean()
    outside = res[~mask].mean()
    return float(inside - outside)


def decision_from_image(image: Image.Image) -> str:
    g = _cell_means(image)
    res = _flatten_planar(g)
    masks = _band_masks()
    sa = _contrast(res, masks["A"])
    sb = _contrast(res, masks["B"])
    return "A" if abs(sa) >= abs(sb) else "B"


def reveal_margin(image: Image.Image) -> float:
    g = _cell_means(image)
    res = _flatten_planar(g)
    masks = _band_masks()
    sa = _contrast(res, masks["A"])
    sb = _contrast(res, masks["B"])
    return abs(abs(sa) - abs(sb))
