"""Independent pixel-only inverse oracle for the ladybug spot-count world.

Receives only the final PNG and re-counts the black spots on the red body
directly from the raster. It imports nothing from the renderer, prompts,
generator, verifier, or analytic-gold code, and accepts no latent scene data.

Method: segment the red oval (the body) and the near-black stain pixels. The
black pixels occur only in the stamped spots, so the number of black connected
components equals the number of spots. The colour thresholds below are
duplicated deliberately rather than imported, so a renderer/oracle disagreement
is exposed by the certification path instead of becoming a shared-code blind spot.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


def _black_mask(arr: np.ndarray) -> np.ndarray:
    r = arr[..., 0].astype(np.int16)
    g = arr[..., 1].astype(np.int16)
    b = arr[..., 2].astype(np.int16)
    return (r <= 70) & (g <= 70) & (b <= 70)


def _connected_components(mask: np.ndarray) -> int:
    h, w = mask.shape
    visited = np.zeros((h, w), dtype=bool)
    ys, xs = np.nonzero(mask)
    count = 0
    for k in range(xs.size):
        y = int(ys[k])
        x = int(xs[k])
        if visited[y, x]:
            continue
        count += 1
        stack = [(y, x)]
        visited[y, x] = True
        while stack:
            cy, cx = stack.pop()
            if cy > 0 and mask[cy - 1, cx] and not visited[cy - 1, cx]:
                visited[cy - 1, cx] = True
                stack.append((cy - 1, cx))
            if cy < h - 1 and mask[cy + 1, cx] and not visited[cy + 1, cx]:
                visited[cy + 1, cx] = True
                stack.append((cy + 1, cx))
            if cx > 0 and mask[cy, cx - 1] and not visited[cy, cx - 1]:
                visited[cy, cx - 1] = True
                stack.append((cy, cx - 1))
            if cx < w - 1 and mask[cy, cx + 1] and not visited[cy, cx + 1]:
                visited[cy, cx + 1] = True
                stack.append((cy, cx + 1))
    return count


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.int16)
    black = _black_mask(arr)
    if not np.any(black):
        return "0"
    return str(_connected_components(black))
