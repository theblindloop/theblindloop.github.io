"""Prior-conflict ladybug spot-count world renderer.

Latent state z = {n_spots, cx, cy, rx, ry, spots}. A single filled red oval
(the ladybug body) is drawn on a plain white canvas and n_spots black rounded
spots are stamped onto it. The closed-form gold is the scene's spot count:

    spot_count = n_spots            (a scene parameter, not a pixel measurement)

The canonical prior — that a ladybird has seven black spots — is recorded only
in the question narrative; it is never used as ground truth. The visible truth
is however many spots are actually stamped on the body.

Continuous verifiability margin (px): the minimum clearance of every spot from
its nearest neighbour spot edge and from the body boundary. It stays comfortably
above pixel scale across the sampled range and governs how unambiguously each
black stain is resolvable as a distinct blob.

Latent-alias mode: not_applicable. The gold is pinned by n_spots which is
rendered directly (each spot is a distinct stamped disk), so there is no latent
variable whose alias could fuzz the gold; different n_spots always yield a
different raster and there are no pixel-identical twins.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS_W = 384
CANVAS_H = 384
BACKGROUND = "white"
BODY_COLOR = (218, 34, 34)      # red oval
SPOT_COLOR = (0, 0, 0)          # near-black spots
SPOT_R = 12.0

N_SPOTS_RANGE = (3, 10)         # inclusive
RX_RANGE = (95.0, 112.0)
RY_RANGE = (72.0, 86.0)
CLEAR_TARGET = 8.0              # desired min clearance between spots and edge

RENDERER_VERSION = "prior_ladybug_spots-0.1.0"


def _ellipse_radius(cx: float, cy: float, rx: float, ry: float, x: float, y: float) -> float:
    """Distance from the ellipse centre to its boundary along the point's ray."""
    dx = x - cx
    dy = y - cy
    if abs(dx) < 1e-9 and abs(dy) < 1e-9:
        return min(rx, ry)
    angle = math.atan2(dy, dx)
    return (rx * ry) / math.hypot(ry * math.cos(angle), rx * math.sin(angle))


def _boundary_clearance(scene: dict[str, Any], x: float, y: float) -> float:
    cx = float(scene["cx"])
    cy = float(scene["cy"])
    rx = float(scene["rx"])
    ry = float(scene["ry"])
    dist = math.hypot(x - cx, y - cy)
    to_edge = _ellipse_radius(cx, cy, rx, ry, x, y) - dist
    return to_edge - SPOT_R


def _place_spots(
    rng: random.Random,
    cx: float,
    cy: float,
    sx: float,
    sy: float,
    n: int,
    clearance: float,
) -> list[list[float]]:
    """Place n spot centres inside the shrunk ellipse with pairwise clearance."""
    sep = 2.0 * SPOT_R + clearance
    spots: list[list[float]] = []
    attempts = 0
    while len(spots) < n:
        attempts += 1
        t = rng.random()
        ang = rng.uniform(0.0, 2.0 * math.pi)
        rad = math.sqrt(t)
        x = cx + rad * math.cos(ang) * sx
        y = cy + rad * math.sin(ang) * sy
        if all(math.hypot(x - px, y - py) >= sep for (px, py) in spots):
            spots.append([x, y])
        elif attempts > 200000 and len(spots) < n:
            # Guaranteed termination fallback: accept the candidate once the
            # rejection budget is exhausted; the margin/quarantine path flags it.
            spots.append([x, y])
    return spots


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(seed)
    n_spots = rng.randint(*N_SPOTS_RANGE)
    rx = rng.uniform(*RX_RANGE)
    ry = rng.uniform(*RY_RANGE)
    cx = rng.uniform(40.0 + rx, CANVAS_W - 40.0 - rx)
    cy = rng.uniform(40.0 + ry, CANVAS_H - 40.0 - ry)

    # Relax target clearance if the sampled size/count pack too tightly so that
    # exactly n_spots are always placed; the margin/quarantine path then reports
    # any genuinely ambiguous near-touching layout.
    clearance = CLEAR_TARGET
    spots: list[list[float]] = []
    while clearance > 2.0:
        sx = rx - SPOT_R - clearance
        sy = ry - SPOT_R - clearance
        if sx <= 0 or sy <= 0:
            clearance -= 1.0
            continue
        trial = _place_spots(rng, cx, cy, sx, sy, n_spots, clearance)
        # _place_spots with a tight packing may loop; guard the descent.
        if trial:
            spots = trial
            break
        clearance -= 0.5
    if not spots:
        sx = max(rx - SPOT_R - 3.0, 1.0)
        sy = max(ry - SPOT_R - 3.0, 1.0)
        spots = _place_spots(rng, cx, cy, sx, sy, n_spots, 3.0)

    return {
        "n_spots": n_spots,
        "cx": round(cx, 3),
        "cy": round(cy, 3),
        "rx": round(rx, 3),
        "ry": round(ry, 3),
        "spots": [[round(x, 3), round(y, 3)] for (x, y) in spots],
    }


def render(scene: dict[str, Any]) -> Image.Image:
    cx = float(scene["cx"])
    cy = float(scene["cy"])
    rx = float(scene["rx"])
    ry = float(scene["ry"])
    n_spots = int(scene["n_spots"])
    if len(scene["spots"]) != n_spots:
        raise ValueError("spot list length disagrees with n_spots")
    image = Image.new("RGB", (CANVAS_W, CANVAS_H), BACKGROUND)
    draw = ImageDraw.Draw(image)
    bbox = [
        round(cx - rx),
        round(cy - ry),
        round(cx + rx),
        round(cy + ry),
    ]
    draw.ellipse(bbox, fill=BODY_COLOR, outline=None)
    for (x, y) in scene["spots"]:
        draw.ellipse(
            [round(x - SPOT_R), round(y - SPOT_R), round(x + SPOT_R), round(y + SPOT_R)],
            fill=SPOT_COLOR,
            outline=None,
        )
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(int(scene["n_spots"]))


def margin(scene: dict[str, Any]) -> float:
    spots = scene["spots"]
    best = float("inf")
    for index, (x, y) in enumerate(spots):
        best = min(best, _boundary_clearance(scene, x, y))
        for (ox, oy) in spots[index + 1 :]:
            best = min(best, math.hypot(ox - x, oy - y) - 2.0 * SPOT_R)
    return round(max(best, 0.0), 3)


def is_quarantined(scene: dict[str, Any]) -> bool:
    # Spots become pixel-ambiguous only when two stains nearly touch or a stain
    # slides too close to the body rim. Inside the sampled range this is rare.
    return margin(scene) < 2.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # not_applicable: the gold is pinned by the rendered n_spots; there is no
    # declared pixel-identical alias to exercise.
    return []
