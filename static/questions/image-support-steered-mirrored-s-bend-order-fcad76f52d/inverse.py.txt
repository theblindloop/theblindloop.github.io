"""Independent image-only inverse arm for the two ordered S lobes."""

from __future__ import annotations

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    pass


def _blue_mask(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = arr[:, :, 0], arr[:, :, 1], arr[:, :, 2]
    return (blue - red > 42) & (blue - green > 18) & (blue > 105)


def _largest_component_fraction(mask: np.ndarray) -> tuple[int, float]:
    points = np.argwhere(mask)
    if not len(points):
        return 0, 0.0
    pending = {tuple(map(int, point)) for point in points}
    largest = 0
    count = 0
    height, width = mask.shape
    while pending:
        count += 1
        start = pending.pop()
        stack = [start]
        size = 1
        while stack:
            y, x = stack.pop()
            for neighbor in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                ny, nx = neighbor
                if 0 <= ny < height and 0 <= nx < width and neighbor in pending:
                    pending.remove(neighbor)
                    stack.append(neighbor)
                    size += 1
        largest = max(largest, size)
    return count, largest / len(points)


def decision_from_image(image: Image.Image) -> str:
    mask = _blue_mask(image)
    ys, xs = np.nonzero(mask)
    if len(xs) < 650:
        raise OracleAbstention("too little blue stroke remains")
    component_count, largest_fraction = _largest_component_fraction(mask)
    if component_count > 3 and largest_fraction < 0.96:
        raise OracleAbstention("blue stroke is materially disconnected")

    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    box_width, box_height = x1 - x0 + 1, y1 - y0 + 1
    if not (28 <= box_width <= 70 and 105 <= box_height <= 155):
        raise OracleAbstention("blue support has implausible S geometry")

    band_centers: list[float] = []
    for band in range(16):
        lo = y0 + box_height * band // 16
        hi = y0 + box_height * (band + 1) // 16
        by, bx = np.nonzero(mask[lo:hi, :])
        if len(bx) < 18:
            raise OracleAbstention("an ordered vertical section of the stroke is missing")
        band_centers.append(float(np.median(bx)))

    upper = float(np.mean(band_centers[2:6]))
    lower = float(np.mean(band_centers[10:14]))
    score = upper - lower
    if abs(score) < 10.0:
        raise OracleAbstention("lobe-order runner-up margin is too small")
    return "right_then_left" if score > 0.0 else "left_then_right"
