"""Deterministic renderer for a localized standard or mirrored S centerline."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS = 512
INKS = ("#1756B1", "#146F9E", "#3C45AD", "#075FA8")


def sample_scene(seed: int) -> dict[str, Any]:
    """Pair adjacent seeds: identical nuisance style, opposite visible order."""
    rng = random.Random(int(seed) // 2)
    return {
        "direction": "right_then_left" if int(seed) % 2 == 0 else "left_then_right",
        "center_x": rng.choice((96, 160, 224, 288, 352, 416)),
        "center_y": rng.choice((160, 224, 288, 352)),
        "amplitude": rng.uniform(13.5, 15.0),
        "height": rng.uniform(118.0, 136.0),
        "stroke": rng.uniform(8.0, 10.0),
        "ink": rng.choice(INKS),
        "width": CANVAS,
        "height_px": CANVAS,
        "background": "#FAFAF7",
    }


def _centerline(scene: dict[str, Any], scale: int = 1) -> list[tuple[float, float]]:
    sign = 1.0 if scene["direction"] == "right_then_left" else -1.0
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    amp = float(scene["amplitude"])
    height = float(scene["height"])
    points = []
    for index in range(161):
        t = index / 160.0
        x = cx + sign * amp * math.sin(2.0 * math.pi * t)
        y = cy - height / 2.0 + height * t
        points.append((x * scale, y * scale))
    return points


def render(scene: dict[str, Any]) -> Image.Image:
    scale = 4
    width = int(scene.get("width", CANVAS))
    height_px = int(scene.get("height_px", CANVAS))
    image = Image.new(
        "RGB",
        (width * scale, height_px * scale),
        scene.get("background", "#FAFAF7"),
    )
    draw = ImageDraw.Draw(image)
    draw.line(
        _centerline(scene, scale),
        fill=scene["ink"],
        width=max(1, round(float(scene["stroke"]) * scale)),
        joint="curve",
    )
    return image.resize((width, height_px), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(scene["direction"])


def margin(scene: dict[str, Any]) -> float:
    """Predicted separation of the two lobe centroids, in raster pixels."""
    return 4.0 * float(scene["amplitude"]) / math.pi


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 15.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
