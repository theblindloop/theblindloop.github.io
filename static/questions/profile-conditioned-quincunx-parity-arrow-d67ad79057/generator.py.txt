"""Deterministic sampler and raster renderer for quincunx parity filtering."""

from __future__ import annotations

import random
from typing import Iterable

import numpy as np
from PIL import Image, ImageDraw

CANVAS = (760, 650)
GRID_N = 13
RESULT_N = 11
SOURCE_INK = 84
ORIENTATIONS = ("up", "right", "down", "left")

BACKGROUND = (250, 248, 241)
LIGHT = (235, 232, 221)
GRID_LINE = (183, 187, 181)
FRAME = (25, 157, 179)
GOLD = (236, 174, 45)
KEY_LINE = (54, 61, 67)
DARK_PALETTES = ((29, 47, 68), (48, 43, 68), (43, 55, 50), (57, 42, 49))

_ARROW_UP = (
    "...........",
    ".....#.....",
    "....###....",
    "...#####...",
    "..#######..",
    ".....#.....",
    ".....#.....",
    ".....#.....",
    ".....#.....",
    "...........",
    "...........",
)


def _bits(rows: Iterable[str]) -> np.ndarray:
    return np.array([[ch == "#" for ch in row] for row in rows], dtype=bool)


def _templates() -> dict[str, np.ndarray]:
    up = _bits(_ARROW_UP)
    return {
        "up": up,
        "left": np.rot90(up, 1).copy(),
        "down": np.rot90(up, 2).copy(),
        "right": np.rot90(up, 3).copy(),
    }


def _parity(source: np.ndarray) -> np.ndarray:
    """Odd parity at quincunx sites: four corners plus center."""
    return (
        source[:-2, :-2]
        ^ source[:-2, 2:]
        ^ source[1:-1, 1:-1]
        ^ source[2:, :-2]
        ^ source[2:, 2:]
    )


def _preimage(target: np.ndarray, rng: random.Random) -> np.ndarray:
    """Solve the triangular GF(2) system after randomizing 48 border bits."""
    source = np.zeros((GRID_N, GRID_N), dtype=bool)
    for row in range(GRID_N):
        for col in range(GRID_N):
            if row >= RESULT_N or col >= RESULT_N:
                source[row, col] = bool(rng.getrandbits(1))
    for row in range(RESULT_N - 1, -1, -1):
        for col in range(RESULT_N - 1, -1, -1):
            source[row, col] = bool(
                target[row, col]
                ^ source[row, col + 2]
                ^ source[row + 1, col + 1]
                ^ source[row + 2, col]
                ^ source[row + 2, col + 2]
            )
    if not np.array_equal(_parity(source), target):
        raise RuntimeError("parity preimage construction failed")
    return source


def _classify(result: np.ndarray) -> tuple[str, float]:
    scores = {
        direction: int((result == template).sum())
        for direction, template in _templates().items()
    }
    ranked = sorted(scores.items(), key=lambda item: (-item[1], item[0]))
    return ranked[0][0], float(ranked[0][1] - ranked[1][1])


def _source(scene: dict) -> np.ndarray:
    rows = scene["source_rows"]
    if len(rows) != GRID_N or any(len(row) != GRID_N for row in rows):
        raise ValueError("source_rows must be a 13-by-13 binary raster")
    if any(ch not in "01" for row in rows for ch in row):
        raise ValueError("source_rows contains a non-binary cell")
    return np.array([[ch == "1" for ch in row] for row in rows], dtype=bool)


def sample_scene(seed: int) -> dict:
    direction = ORIENTATIONS[int(seed) % len(ORIENTATIONS)]
    rng = random.Random((int(seed) + 0x51A7) * 1_000_003)
    target = _templates()[direction]
    source = None
    for _ in range(20_000):
        proposed = _preimage(target, rng)
        if int(proposed.sum()) == SOURCE_INK:
            source = proposed
            break
    if source is None:
        raise RuntimeError("failed to find an ink-balanced parity preimage")
    cell_size = rng.choice((27, 28, 29, 30))
    origin_x = rng.randint(86, 104)
    origin_y = rng.randint(144, 164)
    dark_rgb = list(rng.choice(DARK_PALETTES))
    return {
        "source_rows": ["".join("1" if value else "0" for value in row) for row in source],
        "cell_size": cell_size,
        "origin_x": origin_x,
        "origin_y": origin_y,
        "dark_rgb": dark_rgb,
    }


def render(scene: dict) -> Image.Image:
    source = _source(scene)
    cell = int(scene["cell_size"])
    x0 = int(scene["origin_x"])
    y0 = int(scene["origin_y"])
    dark = tuple(int(value) for value in scene["dark_rgb"])
    if not 24 <= cell <= 34 or len(dark) != 3:
        raise ValueError("unsupported render style")
    image = Image.new("RGB", CANVAS, BACKGROUND)
    draw = ImageDraw.Draw(image)
    width = GRID_N * cell
    border = 7
    draw.rectangle(
        (x0 - border, y0 - border, x0 + width + border - 1, y0 + width + border - 1),
        fill=FRAME,
    )
    for row in range(GRID_N):
        for col in range(GRID_N):
            left, top = x0 + col * cell, y0 + row * cell
            fill = dark if source[row, col] else LIGHT
            draw.rectangle((left, top, left + cell - 1, top + cell - 1), fill=fill)
    for index in range(GRID_N + 1):
        pos = index * cell
        draw.line((x0 + pos, y0, x0 + pos, y0 + width), fill=GRID_LINE, width=1)
        draw.line((x0, y0 + pos, x0 + width, y0 + pos), fill=GRID_LINE, width=1)

    key_cell = 22
    key_x, key_y = x0 + width + 56, y0 + 74
    draw.rectangle(
        (key_x - 6, key_y - 6, key_x + 3 * key_cell + 5, key_y + 3 * key_cell + 5),
        fill=KEY_LINE,
    )
    quincunx = {(0, 0), (0, 2), (1, 1), (2, 0), (2, 2)}
    for row in range(3):
        for col in range(3):
            left, top = key_x + col * key_cell, key_y + row * key_cell
            fill = GOLD if (row, col) in quincunx else BACKGROUND
            draw.rectangle((left, top, left + key_cell - 1, top + key_cell - 1), fill=fill)
            draw.rectangle(
                (left, top, left + key_cell - 1, top + key_cell - 1),
                outline=KEY_LINE,
                width=2,
            )
    return image


def analytic_gold(scene: dict) -> str:
    return _classify(_parity(_source(scene)))[0]


def margin(scene: dict) -> float:
    return _classify(_parity(_source(scene)))[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 10.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
