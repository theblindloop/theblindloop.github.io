"""Deterministic scenes and raster for order-invariant colored-pair matching."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw, ImageFont

LABELS = ("A", "B", "C", "D")

# Five perceptually separated node colors.  Their mapping to latent color ids is
# permuted per scene, while the pixel oracle reasons directly over visible RGB.
BASE_PALETTE = (
    (204, 62, 72),   # red
    (47, 105, 190),  # blue
    (35, 145, 91),   # green
    (229, 160, 30),  # gold
    (139, 78, 172),  # purple
)

THEMES = (
    ((249, 248, 244), (239, 242, 241), (112, 122, 132), (43, 47, 53)),
    ((248, 247, 243), (242, 239, 235), (119, 121, 132), (46, 48, 55)),
    ((247, 249, 247), (237, 242, 239), (107, 123, 119), (42, 49, 48)),
)


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _all_five_cycles() -> tuple[tuple[tuple[int, int], ...], ...]:
    """The twelve undirected Hamilton cycles on five labeled colors."""
    cycles: set[tuple[tuple[int, int], ...]] = set()
    for tail in itertools.permutations(range(1, 5)):
        order = (0, *tail)
        edges = tuple(
            sorted(tuple(sorted((order[index], order[(index + 1) % 5])))
                   for index in range(5))
        )
        cycles.add(edges)
    return tuple(sorted(cycles))


ALL_CYCLES = _all_five_cycles()


def _signature(edges: list[list[int]] | tuple[tuple[int, int], ...]) -> tuple[tuple[int, int], ...]:
    """Validate and canonicalize five visible unordered color-pair relations."""
    if len(edges) != 5:
        raise ValueError("each region must show exactly five joined pairs")
    canonical: list[tuple[int, int]] = []
    for edge in edges:
        if len(edge) != 2:
            raise ValueError("each joined pair must have two endpoints")
        left, right = int(edge[0]), int(edge[1])
        if left == right or left not in range(5) or right not in range(5):
            raise ValueError("pair endpoints must be two distinct colors")
        canonical.append(tuple(sorted((left, right))))
    if len(set(canonical)) != 5:
        raise ValueError("pair identities must be unique")
    degree = [0] * 5
    for left, right in canonical:
        degree[left] += 1
        degree[right] += 1
    if degree != [2, 2, 2, 2, 2]:
        raise ValueError("each region must have identical degree-two color inventory")
    return tuple(sorted(canonical))


def _presented(signature: tuple[tuple[int, int], ...], rng: random.Random) -> list[list[int]]:
    """Choose visible token order and endpoint direction independently of identity."""
    edges = [list(edge) for edge in signature]
    rng.shuffle(edges)
    for edge in edges:
        if rng.randrange(2):
            edge.reverse()
    return edges


def _score(reference: list[list[int]], candidate: list[list[int]]) -> int:
    return len(set(_signature(reference)) & set(_signature(candidate)))


def sample_scene(seed: int) -> dict:
    """Sample balanced four-way search scenes; consecutive seeds cycle A/B/C/D."""
    value = int(seed)
    rng = random.Random(value ^ 0xC01A6E57)
    answer_index = value % 4

    reference_signature = rng.choice(ALL_CYCLES)
    near_cycles = [
        cycle for cycle in ALL_CYCLES
        if cycle != reference_signature
        and len(set(cycle) & set(reference_signature)) == 3
    ]
    if len(near_cycles) != 5:
        raise RuntimeError("five-cycle catalog lost its balanced near neighbors")
    distractor_signatures = rng.sample(near_cycles, 3)

    candidate_signatures: list[tuple[tuple[int, int], ...]] = []
    remaining = list(distractor_signatures)
    rng.shuffle(remaining)
    for index in range(4):
        candidate_signatures.append(
            reference_signature if index == answer_index else remaining.pop()
        )

    palette = list(BASE_PALETTE)
    rng.shuffle(palette)
    background, card, frame, line = THEMES[(value // 4) % len(THEMES)]
    scene = {
        "width": 1000,
        "height": 760,
        "reference_edges": _presented(reference_signature, rng),
        "candidate_edges": [_presented(signature, rng) for signature in candidate_signatures],
        "palette_rgb": [list(color) for color in palette],
        "background_rgb": list(background),
        "card_rgb": list(card),
        "frame_rgb": list(frame),
        "line_rgb": list(line),
        "node_radius": 12 + ((value // 12) % 2),
        "token_half_span": 36 + 2 * ((value // 24) % 3),
        "line_width": 6,
        "x_shift": rng.randint(-6, 6),
        "y_shift": rng.randint(-3, 3),
    }
    if margin(scene) < 2.0:
        raise RuntimeError("sampled scene violates the runner-up margin")
    return scene


def _draw_token(
    draw: ImageDraw.ImageDraw,
    edge: list[int],
    center: tuple[int, int],
    scene: dict,
    scale: int,
) -> None:
    cx, cy = center
    half_span = int(scene["token_half_span"])
    radius = int(scene["node_radius"])
    line = tuple(scene["line_rgb"])
    palette = [tuple(color) for color in scene["palette_rgb"]]
    draw.line(
        ((cx - half_span) * scale, cy * scale, (cx + half_span) * scale, cy * scale),
        fill=line,
        width=int(scene["line_width"]) * scale,
    )
    for x, color_id in ((cx - half_span, edge[0]), (cx + half_span, edge[1])):
        draw.ellipse(
            ((x - radius) * scale, (cy - radius) * scale,
             (x + radius) * scale, (cy + radius) * scale),
            fill=palette[int(color_id)],
            outline=(255, 255, 255),
            width=2 * scale,
        )


def render(scene: dict) -> Image.Image:
    """Render one reference above four equal-geometry labeled cards."""
    scale = 3
    width, height = int(scene["width"]), int(scene["height"])
    background = tuple(scene["background_rgb"])
    card = tuple(scene["card_rgb"])
    frame = tuple(scene["frame_rgb"])
    line = tuple(scene["line_rgb"])
    x_shift, y_shift = int(scene["x_shift"]), int(scene["y_shift"])

    reference = [list(edge) for edge in scene["reference_edges"]]
    candidates = [[list(edge) for edge in panel] for panel in scene["candidate_edges"]]
    _signature(reference)
    if len(candidates) != 4:
        raise ValueError("exactly four candidate cards are required")
    for panel in candidates:
        _signature(panel)

    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    title_color = line
    draw.text((width * scale // 2, 24 * scale), "MATCH THE FIVE COLOR PAIRS",
              fill=title_color, font=_font(24 * scale), anchor="mm")
    draw.text((width * scale // 2, 51 * scale),
              "PAIR ORDER AND LEFT–RIGHT ORDER DO NOT MATTER",
              fill=title_color, font=_font(14 * scale), anchor="mm")

    def card_box(box: tuple[int, int, int, int]) -> None:
        draw.rounded_rectangle(
            tuple(value * scale for value in box), radius=13 * scale,
            fill=card, outline=frame, width=3 * scale,
        )

    reference_box = (374 + x_shift, 70 + y_shift, 626 + x_shift, 320 + y_shift)
    card_box(reference_box)
    draw.text(((500 + x_shift) * scale, (94 + y_shift) * scale), "REFERENCE",
              fill=title_color, font=_font(15 * scale), anchor="mm")
    for row, edge in enumerate(reference):
        _draw_token(draw, edge, (500 + x_shift, 132 + y_shift + 39 * row), scene, scale)

    boxes = [
        (18 + 245 * index + x_shift, 350 + y_shift,
         247 + 245 * index + x_shift, 742 + y_shift)
        for index in range(4)
    ]
    for index, (panel, box) in enumerate(zip(candidates, boxes, strict=True)):
        card_box(box)
        center_x = (box[0] + box[2]) // 2
        draw.ellipse(
            ((center_x - 18) * scale, (374 + y_shift - 18) * scale,
             (center_x + 18) * scale, (374 + y_shift + 18) * scale),
            fill=(255, 255, 255), outline=frame, width=2 * scale,
        )
        draw.text((center_x * scale, (374 + y_shift) * scale), LABELS[index],
                  fill=title_color, font=_font(20 * scale), anchor="mm")
        for row, edge in enumerate(panel):
            _draw_token(draw, edge, (center_x, 426 + y_shift + 61 * row), scene, scale)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    scores = [_score(scene["reference_edges"], panel) for panel in scene["candidate_edges"]]
    winners = [index for index, score in enumerate(scores) if score == max(scores)]
    if len(winners) != 1 or scores[winners[0]] != 5:
        raise ValueError("scene lacks one unique exact colored-pair-set match")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    scores = sorted(
        (_score(scene["reference_edges"], panel) for panel in scene["candidate_edges"]),
        reverse=True,
    )
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    return (
        margin(scene) < 2.0
        or int(scene["node_radius"]) < 10
        or int(scene["token_half_span"]) < 28
        or int(scene["line_width"]) < 4
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
