"""Halo-correspondence world renderer (paper-semantic correspondence family).

A cross-panel attribute-binding task. A single ``reference`` disk (a solid
colored disk, no halo) is shown in the top panel. The bottom panel shows a
re-arranged row of ``candidate`` objects, each a solid fill disk surrounded by a
halo ring of a second color. Exactly one candidate has the same *fill* color as
the reference (it is the same object, re-arranged). The question asks: after
matching that object across the panels by its fill color, what is the color of
the halo bound to it?

The secondary attribute (halo) is present only in the candidate panel, so the
answer cannot be read from the reference panel alone; the pixel-only inverse arm
must first establish the cross-panel correspondence, then read the bound halo.

Latent state z (all exposed scene fields are rendered -- nothing hidden):

    ref_fill    : the reference disk's fill color (one of FILL_PALETTE)
    ref_cx, ref_cy : reference disk center
    cand_fills  : fill color of each candidate slot  (a permutation of fills)
    cand_halos  : halo color of each candidate slot (a permutation of halos)
    cand_centers : [x, y] of each candidate slot
    width, height : canvas

The analytic gold is derived from these rendered fields: match cand_fills to
ref_fill, then report cand_halos at that slot. Because every such field is drawn
into the raster, the gold is recoverable from the pixels alone.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "halo_correspondence-0.1.0"

BACKGROUND = "white"
DIVIDER = (70, 70, 70)

FILL_PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (255, 0, 0),
    "green": (0, 180, 0),
    "blue": (0, 0, 255),
    "orange": (255, 140, 0),
}

HALO_PALETTE: dict[str, tuple[int, int, int]] = {
    "cyan": (0, 200, 200),
    "magenta": (255, 0, 255),
    "yellow": (240, 230, 0),
    "purple": (160, 30, 220),
}

# Geometry (public visual spec; constant across scenes for counterbalance of ink).
WIDTH = 760
HEIGHT = 440
DIVIDER_Y = 230
DIVIDER_THICK = 3
FILL_R = 38
HALO_R = 58
REF_CENTER = (380, 115)
CAND_Y = 342
_SLOTS = {3: [160, 380, 600], 4: [150, 300, 450, 600]}

_COLOR_NAMES = {**FILL_PALETTE, **HALO_PALETTE}
_MARGIN_FLOOR = 90.0


def _color_dist(a: tuple[int, int, int], b: tuple[int, int, int]) -> float:
    return math.dist(a, b)


def sample_scene(seed: int) -> dict:
    """Sample a balanced, deterministic latent scene (dict)."""
    rng = random.Random(seed)
    n = rng.choice([3, 4])
    fills = rng.sample(list(FILL_PALETTE), n)
    ref_fill = rng.choice(fills)
    halos = rng.sample(list(HALO_PALETTE), n)
    pairs = list(zip(fills, halos))
    rng.shuffle(pairs)
    xs = _SLOTS[n]
    return {
        "ref_fill": ref_fill,
        "ref_cx": REF_CENTER[0],
        "ref_cy": REF_CENTER[1],
        "cand_fills": [p[0] for p in pairs],
        "cand_halos": [p[1] for p in pairs],
        "cand_centers": [[xs[i], CAND_Y] for i in range(n)],
        "width": WIDTH,
        "height": HEIGHT,
    }


def render(scene: dict) -> Image.Image:
    """Render I = R(z) deterministically at pixel resolution (crisp edges)."""
    w = int(scene["width"])
    h = int(scene["height"])
    img = Image.new("RGB", (w, h), BACKGROUND)
    draw = ImageDraw.Draw(img)
    # Reference panel: a solid fill disk with no halo.
    cx, cy = int(scene["ref_cx"]), int(scene["ref_cy"])
    draw.ellipse([cx - FILL_R, cy - FILL_R, cx + FILL_R, cy + FILL_R], fill=FILL_PALETTE[scene["ref_fill"]])
    # Candidate panel: each object is a halo ring of cand_halos with a fill disk
    # of cand_fills drawn on top.
    for (fx, fy), fill, halo in zip(scene["cand_centers"], scene["cand_fills"], scene["cand_halos"], strict=True):
        x, y = int(fx), int(fy)
        draw.ellipse([x - HALO_R, y - HALO_R, x + HALO_R, y + HALO_R], fill=HALO_PALETTE[halo])
        draw.ellipse([x - FILL_R, y - FILL_R, x + FILL_R, y + FILL_R], fill=FILL_PALETTE[fill])
    # Divider between the two panels (part of the visual spec).
    draw.rectangle([0, DIVIDER_Y, w, DIVIDER_Y + DIVIDER_THICK], fill=DIVIDER)
    return img


def _matched_index(scene: dict) -> int:
    return scene["cand_fills"].index(scene["ref_fill"])


def analytic_gold(scene: dict) -> str:
    """Decision: the halo color bound to the reference-matched candidate."""
    return scene["cand_halos"][_matched_index(scene)]


def margin(scene: dict) -> float:
    """Runner-up margin: nearest confusable alternative, across both the
    fill-matching step and the halo-answer step. Large -> unambiguous."""
    idx = _matched_index(scene)
    answer_halo = scene["cand_halos"][idx]
    d = []
    for i, h in enumerate(scene["cand_halos"]):
        if i != idx:
            d.append(_color_dist(HALO_PALETTE[answer_halo], HALO_PALETTE[h]))
    for f in scene["cand_fills"]:
        if f != scene["ref_fill"]:
            d.append(_color_dist(FILL_PALETTE[scene["ref_fill"]], FILL_PALETTE[f]))
    return float(min(d))


def is_quarantined(scene: dict) -> bool:
    """Quarantine only a genuinely ambiguous scene (never produced by the
    sampler, which keeps fills and halos distinct)."""
    return margin(scene) < _MARGIN_FLOOR


def latent_symmetries(scene: dict) -> list:
    """not_applicable: every label-influencing field is rendered, so the raster
    uniquely determines the gold; no pixel-identical transform is declared."""
    return []
