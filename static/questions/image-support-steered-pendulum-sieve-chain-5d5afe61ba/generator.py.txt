"""Deterministic Pillow renderer and analytic arm for a pendulum sieve chain."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 800
HEIGHT = 800
CELL = 100

COLOR_NAMES = ("red", "blue", "green", "orange", "purple", "cyan", "yellow", "pink")
PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (224, 76, 72),
    "blue": (66, 119, 220),
    "green": (78, 181, 104),
    "orange": (239, 145, 48),
    "purple": (147, 91, 199),
    "cyan": (44, 180, 190),
    "yellow": (235, 199, 54),
    "pink": (224, 107, 164),
}

_BACKGROUND = (255, 255, 255)
_INK = (32, 39, 51)
_START = (19, 133, 78)
_TURN = (176, 35, 68)
_DISTRACTOR = (229, 232, 237)


def survivor_index(count: int, start_side: str) -> int:
    """Return the original position surviving repeated alternating sweeps."""
    if count < 2:
        raise ValueError("a sieve chain needs at least two beads")
    if start_side not in {"left", "right"}:
        raise ValueError("start_side must be left or right")
    remaining = list(range(count))
    left_to_right = start_side == "left"
    while len(remaining) > 1:
        ordered = remaining if left_to_right else list(reversed(remaining))
        kept = ordered[::2]
        remaining = sorted(kept)
        left_to_right = not left_to_right
    return remaining[0]


def sample_scene(seed: int) -> dict:
    """Balance the answer color first, then randomize length, direction, and style."""
    seed = int(seed)
    answer_color = COLOR_NAMES[seed % len(COLOR_NAMES)]
    quotient = seed // len(COLOR_NAMES)
    bead_count = 5 + ((quotient + 3 * (seed % len(COLOR_NAMES))) % 4)
    start_side = "left" if (quotient + (seed % len(COLOR_NAMES))) % 2 == 0 else "right"
    answer_position = survivor_index(bead_count, start_side)

    content_rng = random.Random(seed * 1009 + 97)
    other_colors = [name for name in COLOR_NAMES if name != answer_color]
    content_rng.shuffle(other_colors)
    bead_colors = other_colors[: bead_count - 1]
    content_rng.shuffle(bead_colors)
    bead_colors.insert(answer_position, answer_color)

    style_rng = random.Random(seed * 7919 + 313)
    return {
        "bead_count": bead_count,
        "bead_colors": bead_colors,
        "start_side": start_side,
        "anchor_row": style_rng.randint(1, 6),
        "anchor_col": style_rng.randint(1, 6),
        "spacing": style_rng.choice((22, 23)),
        "bead_radius": style_rng.choice((7, 8)),
        "distractor_variant": style_rng.randrange(4),
    }


def _centers(scene: dict) -> tuple[list[int], int, int, int]:
    count = int(scene["bead_count"])
    spacing = int(scene["spacing"])
    cx = int(scene["anchor_col"]) * CELL + CELL // 2
    carrier_y = int(scene["anchor_row"]) * CELL + CELL // 2 - 7
    first_x = cx - round((count - 1) * spacing / 2)
    xs = [first_x + index * spacing for index in range(count)]
    return xs, carrier_y, xs[0] - 22, xs[-1] + 22


def _draw_distractors(draw: ImageDraw.ImageDraw, variant: int) -> None:
    layouts = (
        ((55, 55), (745, 745)),
        ((745, 55), (55, 745)),
        ((55, 55), (745, 55), (55, 745)),
        ((745, 745), (745, 55), (55, 745)),
    )
    for x, y in layouts[variant]:
        draw.arc((x - 7, y - 7, x + 7, y + 7), 20, 300, fill=_DISTRACTOR, width=2)


def _draw_triangle_cap(draw: ImageDraw.ImageDraw, x: int, y: int, points_right: bool) -> None:
    if points_right:
        points = ((x - 8, y - 9), (x - 8, y + 9), (x + 9, y))
    else:
        points = ((x + 8, y - 9), (x + 8, y + 9), (x - 9, y))
    draw.polygon(points, fill=_START, outline=_INK)
    draw.line((*points[0], *points[1]), fill=_INK, width=2)


def _draw_turn_cap(draw: ImageDraw.ImageDraw, x: int, y: int) -> None:
    draw.polygon(
        ((x, y - 9), (x + 8, y), (x, y + 9), (x - 8, y)),
        fill=_TURN,
        outline=_INK,
    )


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), _BACKGROUND)
    draw = ImageDraw.Draw(image)
    xs, carrier_y, left_cap_x, right_cap_x = _centers(scene)
    radius = int(scene["bead_radius"])
    bead_y = carrier_y + 13
    colors = list(scene["bead_colors"])

    _draw_distractors(draw, int(scene["distractor_variant"]))
    draw.line((left_cap_x, carrier_y, right_cap_x, carrier_y), fill=_INK, width=3)
    for x, name in zip(xs, colors, strict=True):
        draw.line((x, carrier_y, x, bead_y - radius), fill=_INK, width=2)
        draw.ellipse(
            (x - radius, bead_y - radius, x + radius, bead_y + radius),
            fill=PALETTE[name],
            outline=_INK,
            width=2,
        )

    if scene["start_side"] == "left":
        _draw_triangle_cap(draw, left_cap_x, carrier_y, True)
        _draw_turn_cap(draw, right_cap_x, carrier_y)
    elif scene["start_side"] == "right":
        _draw_turn_cap(draw, left_cap_x, carrier_y)
        _draw_triangle_cap(draw, right_cap_x, carrier_y, False)
    else:
        raise ValueError("invalid start_side")
    return image


def analytic_gold(scene: dict) -> str:
    colors = list(scene["bead_colors"])
    index = survivor_index(int(scene["bead_count"]), str(scene["start_side"]))
    return colors[index]


def margin(scene: dict) -> float:
    """Runner-up clearance from bead merging or interior-color loss, in pixels."""
    return float(
        min(
            int(scene["spacing"]) - 2 * int(scene["bead_radius"]),
            int(scene["bead_radius"]) - 3,
            8,
        )
    )


def is_quarantined(scene: dict) -> bool:
    count = int(scene.get("bead_count", 0))
    colors = list(scene.get("bead_colors", []))
    return (
        count not in range(5, 9)
        or len(colors) != count
        or len(set(colors)) != count
        or any(name not in COLOR_NAMES for name in colors)
        or scene.get("start_side") not in {"left", "right"}
        or int(scene.get("anchor_row", -1)) not in range(1, 7)
        or int(scene.get("anchor_col", -1)) not in range(1, 7)
        or int(scene.get("spacing", 0)) not in (22, 23)
        or int(scene.get("bead_radius", 0)) not in (7, 8)
        or int(scene.get("distractor_variant", -1)) not in range(4)
        or margin(scene) < 4.0
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
