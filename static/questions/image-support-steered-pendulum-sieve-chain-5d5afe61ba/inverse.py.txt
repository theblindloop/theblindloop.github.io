"""Independent PNG-only inverse for the pendulum sieve chain.

The inverse discovers the compact dark carrier, segments the public bead and cap
colors, recovers chain order and start side without fixed coordinates, and then
executes its own alternating destructive-filter replay.  It imports no candidate
module and receives no latent scene data.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (224, 76, 72),
    "blue": (66, 119, 220),
    "green": (78, 181, 104),
    "orange": (239, 145, 48),
    "purple": (147, 91, 199),
    "cyan": (44, 180, 190),
    "yellow": (235, 199, 54),
    "pink": (224, 107, 164),
}
_START = (19, 133, 78)
_TURN = (176, 35, 68)


def _components(mask: np.ndarray, minimum: int = 1) -> list[np.ndarray]:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if visited[y0, x0]:
            continue
        queue = deque([(y0, x0)])
        visited[y0, x0] = True
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            output.append(np.asarray(points, dtype=np.int32))
    return output


def _bounds(points: np.ndarray) -> tuple[int, int, int, int]:
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    return int(x0), int(y0), int(x1), int(y1)


def _exact_mask(array: np.ndarray, color: tuple[int, int, int], tolerance: int = 8) -> np.ndarray:
    delta = np.abs(array.astype(np.int16) - np.asarray(color, dtype=np.int16))
    return np.max(delta, axis=2) <= tolerance


def _single_cap(array: np.ndarray, color: tuple[int, int, int], name: str) -> tuple[float, float]:
    parts = _components(_exact_mask(array, color), minimum=35)
    if len(parts) != 1:
        raise ValueError(f"expected one intact {name} cap, found {len(parts)}")
    part = parts[0]
    x0, y0, x1, y1 = _bounds(part)
    if not (10 <= x1 - x0 + 1 <= 19 and 10 <= y1 - y0 + 1 <= 19):
        raise ValueError(f"{name} cap silhouette is malformed")
    y, x = part.mean(axis=0)
    return float(x), float(y)


def _read_beads(array: np.ndarray) -> list[tuple[float, float, str, int]]:
    beads: list[tuple[float, float, str, int]] = []
    for name, rgb in _PALETTE.items():
        parts = _components(_exact_mask(array, rgb), minimum=35)
        if len(parts) > 1:
            raise ValueError(f"duplicate or fragmented {name} bead")
        if not parts:
            continue
        part = parts[0]
        x0, y0, x1, y1 = _bounds(part)
        width, height = x1 - x0 + 1, y1 - y0 + 1
        area = len(part)
        if not (9 <= width <= 15 and 9 <= height <= 15 and 55 <= area <= 180):
            raise ValueError(f"{name} bead silhouette is malformed")
        y, x = part.mean(axis=0)
        beads.append((float(x), float(y), name, area))
    if not 5 <= len(beads) <= 8:
        raise ValueError(f"expected five to eight readable beads, found {len(beads)}")
    beads.sort()
    gaps = np.diff([item[0] for item in beads])
    if len(gaps) and (float(gaps.min()) < 20.0 or float(gaps.max()) > 25.0):
        raise ValueError("bead order has an ambiguous spacing")
    ys = np.asarray([item[1] for item in beads])
    if float(ys.max() - ys.min()) > 1.0:
        raise ValueError("beads do not form one ordered chain")
    return beads


def _validate_carrier(array: np.ndarray, beads: list[tuple[float, float, str, int]]) -> None:
    dark = np.max(array, axis=2) < 90
    parts = _components(dark, minimum=45)
    if not parts:
        raise ValueError("no intact connected carrier found")
    measured = []
    for part in parts:
        x0, y0, x1, y1 = _bounds(part)
        measured.append((x1 - x0 + 1, y1 - y0 + 1, len(part)))
    measured.sort(reverse=True)
    width, height, area = measured[0]
    bead_span = beads[-1][0] - beads[0][0]
    if not (bead_span + 24 <= width <= bead_span + 58):
        raise ValueError("carrier span is incomplete or inconsistent with beads")
    if not (22 <= height <= 38 and area >= 260):
        raise ValueError("carrier network is broken or malformed")


def _survivor_color(colors: list[str], start_left: bool) -> str:
    remaining = list(range(len(colors)))
    left_to_right = start_left
    while len(remaining) > 1:
        ordered = remaining if left_to_right else list(reversed(remaining))
        remaining = sorted(ordered[::2])
        left_to_right = not left_to_right
    return colors[remaining[0]]


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 64 or array.shape[1] < 64:
        raise ValueError("raster is too small")
    start_x, start_y = _single_cap(array, _START, "START")
    turn_x, turn_y = _single_cap(array, _TURN, "TURN")
    if abs(start_y - turn_y) > 1.0 or abs(start_x - turn_x) < 100.0:
        raise ValueError("chain cap direction is ambiguous")
    beads = _read_beads(array)
    if not (min(start_x, turn_x) < beads[0][0] < beads[-1][0] < max(start_x, turn_x)):
        raise ValueError("beads are not bracketed by the two caps")
    _validate_carrier(array, beads)
    colors = [item[2] for item in beads]
    return _survivor_color(colors, start_x < turn_x)
