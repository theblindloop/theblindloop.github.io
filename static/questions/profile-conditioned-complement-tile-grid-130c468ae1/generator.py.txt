"""Renderer and analytic arm for the complement-tiling grid search world.

A reference four-by-four grid with exactly eight filled cells sits alone in the
upper row.  Five candidate grids, each also carrying exactly eight filled cells,
sit in the lower row.  Exactly one candidate is the cell-by-cell complement of
the reference: it fills every cell the reference leaves empty and leaves empty
every cell the reference fills.  The decision is that candidate's position.
"""

from __future__ import annotations

import json
import random

from PIL import Image, ImageDraw

CANVAS_WIDTH = 820
CANVAS_HEIGHT = 410

CELLS = 4
N_CELLS = CELLS * CELLS
FILLED_PER_GRID = 8
N_CANDIDATES = 5

REF_CELL = 34
CAND_CELL = 30
FRAME = 3
GRID_LINE = 2

REF_TOP = 34
CAND_TOP = 248
CAND_GAP = 25
JITTER = 3

QUARANTINE_MARGIN_PERCENT = 9.0
MARGIN_FLOOR_PERCENT = 12.5

ANSWERS = tuple(f"grid-{index + 1}" for index in range(N_CANDIDATES))

# (background, structure, ink); every ink is far from its background in L1.
PALETTES = (
    ((246, 244, 238), (150, 150, 158), (38, 42, 58)),
    ((240, 246, 248), (140, 158, 166), (24, 58, 72)),
    ((248, 242, 242), (166, 146, 146), (72, 28, 34)),
    ((242, 246, 240), (146, 162, 142), (34, 60, 34)),
    ((244, 242, 250), (152, 148, 170), (46, 36, 78)),
)


def json_clone(scene: dict) -> dict:
    """Round-trip copy of a scene, matching what the manifest stores."""
    return json.loads(json.dumps(scene, sort_keys=True))


def _masks_equal(left, right) -> bool:
    return [bool(value) for value in left] == [bool(value) for value in right]


def _complement(mask) -> list:
    return [0 if bool(value) else 1 for value in mask]


def _swap_cells(mask, count: int, rng: random.Random) -> list:
    """Return a copy of ``mask`` with ``count`` filled/empty cells exchanged."""
    out = list(mask)
    filled = [i for i, v in enumerate(out) if bool(v)]
    empty = [i for i, v in enumerate(out) if not bool(v)]
    take_filled = rng.sample(filled, count)
    take_empty = rng.sample(empty, count)
    for i, j in zip(take_filled, take_empty):
        out[i], out[j] = 0, 1
    return out


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    palette = rng.randrange(len(PALETTES))

    reference = [0] * N_CELLS
    for index in rng.sample(range(N_CELLS), FILLED_PER_GRID):
        reference[index] = 1
    answer_mask = _complement(reference)

    base_k = rng.choice([1, 1, 2])
    lure_ks = [base_k] + [base_k + rng.randrange(2) for _ in range(2)]
    rng.shuffle(lure_ks)

    lures = []
    for count in lure_ks:
        for _ in range(64):
            candidate = _swap_cells(answer_mask, count, rng)
            taken = [answer_mask, reference] + lures
            if not any(_masks_equal(candidate, other) for other in taken):
                lures.append(candidate)
                break
        else:  # pragma: no cover - the 16-cell lattice always leaves room
            raise RuntimeError("could not build a distinct near-complement lure")

    answer_index = rng.randrange(N_CANDIDATES)
    copy_index = rng.choice([i for i in range(N_CANDIDATES) if i != answer_index])
    candidates = [None] * N_CANDIDATES
    candidates[answer_index] = list(answer_mask)
    candidates[copy_index] = list(reference)
    for slot in range(N_CANDIDATES):
        if candidates[slot] is None:
            candidates[slot] = lures.pop()

    jitter = [
        [rng.randint(-JITTER, JITTER), rng.randint(-JITTER, JITTER)]
        for _ in range(N_CANDIDATES)
    ]
    return {
        "width": CANVAS_WIDTH,
        "height": CANVAS_HEIGHT,
        "palette": palette,
        "reference": reference,
        "candidates": candidates,
        "reference_jitter": [rng.randint(-JITTER, JITTER), rng.randint(-JITTER, JITTER)],
        "candidate_jitter": jitter,
    }


def _conflicts(reference, candidate) -> int:
    """Cells where the overlay double-fills or leaves a hole."""
    return sum(
        1
        for ref, cand in zip(reference, candidate)
        if bool(ref) == bool(cand)
    )


def _scores(scene: dict) -> list:
    reference = scene["reference"]
    return [_conflicts(reference, mask) for mask in scene["candidates"]]


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    best = min(range(N_CANDIDATES), key=lambda index: scores[index])
    return ANSWERS[best]


def margin(scene: dict) -> float:
    """Runner-up conflict gap, as a percentage of the sixteen grid cells."""
    scores = sorted(_scores(scene))
    return round(100.0 * (scores[1] - scores[0]) / N_CELLS, 9)


def is_quarantined(scene: dict) -> bool:
    return float(margin(scene)) < QUARANTINE_MARGIN_PERCENT


def latent_symmetries(scene: dict) -> list:
    """Declared pixel-identical latent aliases that must preserve the answer."""
    period = dict(scene)
    period["palette"] = scene["palette"] + len(PALETTES)

    truthy = dict(scene)
    truthy["reference"] = [3 * int(bool(v)) for v in scene["reference"]]
    truthy["candidates"] = [
        [3 * int(bool(v)) for v in mask] for mask in scene["candidates"]
    ]
    return [("palette_period", period), ("mask_truthiness", truthy)]


def _draw_grid(draw, mask, left: int, top: int, cell: int, structure, ink) -> None:
    size = cell * CELLS
    for index, value in enumerate(mask):
        if not bool(value):
            continue
        row, column = divmod(index, CELLS)
        x0 = left + column * cell
        y0 = top + row * cell
        draw.rectangle([x0, y0, x0 + cell - 1, y0 + cell - 1], fill=ink)
    for step in range(1, CELLS):
        offset = step * cell
        draw.rectangle(
            [left + offset - GRID_LINE // 2, top,
             left + offset + GRID_LINE // 2 - 1, top + size - 1],
            fill=structure,
        )
        draw.rectangle(
            [left, top + offset - GRID_LINE // 2,
             left + size - 1, top + offset + GRID_LINE // 2 - 1],
            fill=structure,
        )
    draw.rectangle([left, top, left + size - 1, top + size - 1],
                   outline=structure, width=FRAME)


def render(scene: dict) -> Image.Image:
    background, structure, ink = PALETTES[scene["palette"] % len(PALETTES)]
    image = Image.new("RGB", (scene["width"], scene["height"]), background)
    draw = ImageDraw.Draw(image)

    ref_size = REF_CELL * CELLS
    ref_left = (scene["width"] - ref_size) // 2 + scene["reference_jitter"][0]
    ref_top = REF_TOP + scene["reference_jitter"][1]
    _draw_grid(draw, scene["reference"], ref_left, ref_top, REF_CELL, structure, ink)

    cand_size = CAND_CELL * CELLS
    row_width = N_CANDIDATES * cand_size + (N_CANDIDATES - 1) * CAND_GAP
    start = (scene["width"] - row_width) // 2
    for index, mask in enumerate(scene["candidates"]):
        dx, dy = scene["candidate_jitter"][index]
        left = start + index * (cand_size + CAND_GAP) + dx
        top = CAND_TOP + dy
        _draw_grid(draw, mask, left, top, CAND_CELL, structure, ink)
    return image
