"""Independent pixel-only graph recovery and residual topology decision."""
from __future__ import annotations

import math
from collections import defaultdict

import numpy as np

INK = np.array([31, 41, 55])
LINK = np.array([104, 116, 132])
COLORS = {
    "orange": np.array([230, 126, 34]),
    "green": np.array([24, 158, 105]),
    "purple": np.array([137, 82, 194]),
}


def _near(array, color, tolerance=30):
    return np.max(np.abs(array.astype(int) - color.astype(int)), axis=2) <= tolerance


def _components(mask, minimum=40):
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        points = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= minimum:
            result.append(points)
    return result


def _center(points):
    return (
        sum(point[1] for point in points) / len(points),
        sum(point[0] for point in points) / len(points),
    )


def _edge_color(array, p, q):
    x1, y1 = p
    x2, y2 = q
    distance = math.hypot(x2 - x1, y2 - y1)
    if distance < 65 or distance > 115:
        return None
    palette = [("gray", LINK), *COLORS.items()]
    votes = defaultdict(int)
    total = 0
    # The endpoint disks occupy about 16 pixels; inspect the visible link core.
    for step in range(19, max(20, int(distance) - 18)):
        t = step / distance
        x = int(round(x1 + t * (x2 - x1)))
        y = int(round(y1 + t * (y2 - y1)))
        patch = array[max(0, y - 2):y + 3, max(0, x - 2):x + 3]
        total += 1
        for name, rgb in palette:
            if patch.size and np.any(np.max(np.abs(patch.astype(int) - rgb), axis=2) <= 34):
                votes[name] += 1
    if total < 20:
        return None
    name, count = max(votes.items(), key=lambda item: item[1], default=(None, 0))
    return name if count / total > 0.88 else None


def _residual_stats(node_count, edges, removed_index):
    adjacency = [[] for _ in range(node_count)]
    for index, (u, v, _color) in enumerate(edges):
        if index != removed_index:
            adjacency[u].append(v)
            adjacency[v].append(u)
    seen = set()
    cyclic_components = 0
    components = 0
    for start in range(node_count):
        if start in seen:
            continue
        components += 1
        seen.add(start)
        stack = [start]
        vertices = 0
        degree_sum = 0
        while stack:
            node = stack.pop()
            vertices += 1
            degree_sum += len(adjacency[node])
            for nxt in adjacency[node]:
                if nxt not in seen:
                    seen.add(nxt)
                    stack.append(nxt)
        if degree_sum // 2 - vertices + 1 > 0:
            cyclic_components += 1
    return components, cyclic_components


def decision_from_image(image) -> str:
    array = np.asarray(image.convert("RGB"))
    station_components = _components(_near(array, INK), minimum=120)
    stations = [_center(component) for component in station_components]
    if len(stations) != 11:
        raise ValueError(f"expected 11 visible stations, found {len(stations)}")
    edges = []
    for first in range(len(stations)):
        for second in range(first + 1, len(stations)):
            color = _edge_color(array, stations[first], stations[second])
            if color is not None:
                edges.append((first, second, color))
    if len(edges) != 12:
        raise ValueError(f"expected 12 visible links, found {len(edges)}")
    candidate_indices = defaultdict(list)
    for index, (_u, _v, color) in enumerate(edges):
        if color != "gray":
            candidate_indices[color].append(index)
    if set(candidate_indices) != set(COLORS) or any(len(v) != 1 for v in candidate_indices.values()):
        raise ValueError(f"expected one link of each candidate color, found {dict(candidate_indices)}")
    outcomes = {
        color: _residual_stats(len(stations), edges, indices[0])
        for color, indices in candidate_indices.items()
    }
    winners = [color for color, outcome in outcomes.items() if outcome == (2, 2)]
    if len(winners) != 1:
        raise ValueError(f"pixel graph has non-unique answer {winners}: {outcomes}")
    return winners[0]
