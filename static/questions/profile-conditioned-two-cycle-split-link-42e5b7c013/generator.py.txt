"""Deterministic planar network renderer and independent analytic arm."""
from __future__ import annotations

import random
from collections import defaultdict

from PIL import Image, ImageDraw

CANVAS = 640
BG = (250, 250, 247)
INK = (31, 41, 55)
LINK = (104, 116, 132)
PALETTE = {
    "orange": (230, 126, 34),
    "green": (24, 158, 105),
    "purple": (137, 82, 194),
}
QUESTION = (
    "Gray and colored links join round stations. Imagine removing exactly one "
    "colored link. Which removal splits the network into two separate parts with "
    "a loop still present in both parts? Answer orange, green, or purple."
)

# Two square cycles are joined through k.  The three candidate roles are:
# target: l1-k separates one whole cycle from the other;
# tail: l3-t separates an acyclic tail; cycle: r1-r2 is not a cut edge.
BASE_STATIONS = {
    "l0": (110, 220), "l1": (200, 220), "l2": (200, 310), "l3": (110, 310),
    "k": (290, 220),
    "r0": (380, 220), "r1": (470, 220), "r2": (470, 310), "r3": (380, 310),
    "t": (110, 400), "u": (200, 400),
}
BASE_LINKS = [
    ("l0", "l1", "gray"), ("l1", "l2", "gray"),
    ("l2", "l3", "gray"), ("l3", "l0", "gray"),
    ("l1", "k", "target"), ("k", "r0", "gray"),
    ("r0", "r1", "gray"), ("r1", "r2", "cycle"),
    ("r2", "r3", "gray"), ("r3", "r0", "gray"),
    ("l3", "t", "tail"), ("t", "u", "gray"),
]


def _transform(point, turns, mirror, dx, dy):
    x, y = point
    if mirror:
        x = CANVAS - x
    for _ in range(turns):
        x, y = CANVAS - y, x
    return [x + dx, y + dy]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    colors = ["orange", "green", "purple"]
    target_color = colors[seed % 3]
    remaining = [c for c in colors if c != target_color]
    if rng.randrange(2):
        remaining.reverse()
    role_color = {"target": target_color, "tail": remaining[0], "cycle": remaining[1]}
    turns = rng.randrange(4)
    mirror = bool(rng.randrange(2))
    dx, dy = rng.randint(-18, 18), rng.randint(-18, 18)
    stations = [
        {"id": name, "center": _transform(point, turns, mirror, dx, dy)}
        for name, point in BASE_STATIONS.items()
    ]
    links = [
        {"u": u, "v": v, "color": role_color.get(role, "gray")}
        for u, v, role in BASE_LINKS
    ]
    rng.shuffle(stations)
    rng.shuffle(links)
    return {"canvas": CANVAS, "stations": stations, "links": links}


def _station_map(scene):
    return {row["id"]: tuple(row["center"]) for row in scene["stations"]}


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas"])
    scale = 3
    image = Image.new("RGB", (size * scale, size * scale), BG)
    draw = ImageDraw.Draw(image)
    stations = _station_map(scene)
    # Canonical drawing makes serialization order and edge orientation true aliases.
    links = sorted(scene["links"], key=lambda e: (min(e["u"], e["v"]), max(e["u"], e["v"])))
    for edge in links:
        u, v = sorted((edge["u"], edge["v"]))
        p = tuple(round(value * scale) for value in stations[u])
        q = tuple(round(value * scale) for value in stations[v])
        fill = LINK if edge["color"] == "gray" else PALETTE[edge["color"]]
        draw.line((p, q), fill=fill, width=10 * scale)
    radius = 14 * scale
    for name in sorted(stations):
        x, y = (round(value * scale) for value in stations[name])
        draw.ellipse(
            (x - radius, y - radius, x + radius, y + radius),
            fill=INK, outline=(255, 255, 255), width=2 * scale,
        )
    return image.resize((size, size), Image.Resampling.LANCZOS)


def _residual_stats(station_ids, links, removed_index):
    edges = [(e["u"], e["v"]) for i, e in enumerate(links) if i != removed_index]
    adjacency = defaultdict(list)
    for u, v in edges:
        adjacency[u].append(v)
        adjacency[v].append(u)
    seen = set()
    cyclic_flags = []
    for start in station_ids:
        if start in seen:
            continue
        stack = [start]
        seen.add(start)
        vertices = 0
        degree_sum = 0
        while stack:
            node = stack.pop()
            vertices += 1
            degree_sum += len(adjacency[node])
            for nxt in adjacency[node]:
                if nxt not in seen:
                    seen.add(nxt)
                    stack.append(nxt)
        edge_count = degree_sum // 2
        cyclic_flags.append(edge_count - vertices + 1 > 0)
    return len(cyclic_flags), sum(cyclic_flags)


def _candidate_outcomes(scene):
    ids = [row["id"] for row in scene["stations"]]
    outcomes = {}
    for index, edge in enumerate(scene["links"]):
        if edge["color"] != "gray":
            outcomes[edge["color"]] = _residual_stats(ids, scene["links"], index)
    return outcomes


def analytic_gold(scene: dict) -> str:
    outcomes = _candidate_outcomes(scene)
    winners = [color for color, outcome in outcomes.items() if outcome == (2, 2)]
    if len(winners) != 1:
        raise ValueError(f"expected one cyclic/cyclic split link, found {winners}: {outcomes}")
    return winners[0]


def margin(scene: dict) -> float:
    scores = []
    for outcome in _candidate_outcomes(scene).values():
        components, cyclic_components = outcome
        scores.append(-(abs(components - 2) + abs(cyclic_components - 2)))
    scores.sort(reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    try:
        outcomes = sorted(_candidate_outcomes(scene).values())
        return outcomes != [(1, 1), (2, 1), (2, 2)] or margin(scene) < 1.0
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = {
        "canvas": scene["canvas"],
        "stations": list(reversed(scene["stations"])),
        "links": [
            {"u": edge["v"], "v": edge["u"], "color": edge["color"]}
            for edge in reversed(scene["links"])
        ],
    }
    return [("reverse_unordered_storage_and_edge_orientation", twin)]
