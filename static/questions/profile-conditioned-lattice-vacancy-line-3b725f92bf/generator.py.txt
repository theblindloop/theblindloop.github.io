"""Lattice-vacancy world: which grid line do the missing dots trace?

Latent scene z = {canvas, palette, dot_radius, spacing, origin, rows, cols,
holes}. The image I = R(z) paints one identical dot at every site of an
axis-aligned square lattice EXCEPT at the sites listed in ``holes``.

Declared rule (stated verbatim in every prompt family): a *grid line* is a whole
row, a whole column, a diagonal running down to the right, or a diagonal running
up to the right. Exactly one grid line is missing more dots than any other grid
line, and the answer is that line's direction. The vacancies on the winning line
are never neighbours of each other, so no local blob marks the line: the answer
only exists in the global arrangement of the negative space.

    row      : sites with equal row index i
    column   : sites with equal column index j
    down     : sites with equal j - i  (down to the right)
    up       : sites with equal i + j  (up to the right)

Counterbalancing: the answer is drawn uniformly and independently of dot colour,
background colour, dot radius, lattice spacing, lattice size, lattice position,
hole count and total ink. Scattered decoy vacancies are always present, so the
count of missing dots never identifies the direction on its own.

Margin: m = (vacancies on the winning line) - (largest vacancy count on any
other grid line), an integer count difference. m == 1 scenes are the finite
boundary of the decision and are quarantined.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "lattice_vacancy_line-0.1.0"

CANVAS = 640
SS = 2  # supersampling factor for dot antialiasing

FAMILIES = ("row", "column", "down", "up")

ROW_CHOICES = (9, 10, 11)
COL_CHOICES = (9, 10, 11)
SPACING_CHOICES = (44, 48, 52)
RADIUS_CHOICES = (6, 7, 8)

MIN_LINE_LEN = 7  # the winning line must have at least this many sites
LINE_HOLES = (4, 5, 6)  # vacancies planted on the winning line
DECOY_HOLES = (3, 4, 5)  # scattered vacancies elsewhere
MIN_DOTS_PER_ROW = 3  # every row and column keeps at least this many dots
MIN_DOTS_PER_COL = 3

QUARANTINE_MARGIN = 2  # margin < 2  =>  quarantined
BOUNDARY_EVERY = 5  # one in five sampled scenes targets margin 1

DOT_COLORS = ("#1f3f7a", "#8c2130", "#2f6b34", "#6a3390", "#a5561b", "#1d6f78")
BG_COLORS = ("#ffffff", "#f6f3ea", "#eef2f7")


# --------------------------------------------------------------------------
# grid-line bookkeeping
# --------------------------------------------------------------------------
def _line_key(family: str, i: int, j: int) -> int:
    if family == "row":
        return i
    if family == "column":
        return j
    if family == "down":
        return j - i
    return i + j


def _line_sites(rows: int, cols: int, family: str, key: int) -> list[tuple[int, int]]:
    return [
        (i, j)
        for i in range(rows)
        for j in range(cols)
        if _line_key(family, i, j) == key
    ]


def _long_lines(rows: int, cols: int, family: str) -> list[int]:
    keys: dict[int, int] = {}
    for i in range(rows):
        for j in range(cols):
            key = _line_key(family, i, j)
            keys[key] = keys.get(key, 0) + 1
    return sorted(key for key, size in keys.items() if size >= MIN_LINE_LEN)


def line_counts(rows: int, cols: int, holes) -> dict:
    """Vacancy count of every grid line of every family."""
    counts: dict = {family: {} for family in FAMILIES}
    for i, j in holes:
        for family in FAMILIES:
            key = _line_key(family, i, j)
            counts[family][key] = counts[family].get(key, 0) + 1
    return counts


def _ranked(rows: int, cols: int, holes) -> list[tuple[int, str, int]]:
    counts = line_counts(rows, cols, holes)
    flat = [
        (count, family, key)
        for family in FAMILIES
        for key, count in counts[family].items()
    ]
    flat.sort(key=lambda item: (-item[0], item[1], item[2]))
    return flat


def _winner(scene: dict):
    rows, cols = scene["rows"], scene["cols"]
    holes = [tuple(hole) for hole in scene["holes"]]
    flat = _ranked(rows, cols, holes)
    if not flat:
        return None, 0
    best = flat[0]
    runner = flat[1][0] if len(flat) > 1 else 0
    ties = [item for item in flat if item[0] == best[0]]
    if len(ties) != 1:
        return None, 0
    return best, best[0] - runner


# --------------------------------------------------------------------------
# public interface
# --------------------------------------------------------------------------
def analytic_gold(scene: dict) -> str:
    """Direction of the unique grid line carrying the most vacancies."""
    best, _ = _winner(scene)
    if best is None:
        raise ValueError("scene has no unique maximal vacancy line")
    return best[1]


def margin(scene: dict) -> float:
    """Count difference between the winning line and the best rival line."""
    _, gap = _winner(scene)
    return float(gap)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict):
    """Declared pixel-identical latent aliases.

    ``holes`` is an unordered set of lattice sites: the renderer paints a dot at
    every site that is not listed, so any permutation of the list paints exactly
    the same pixels, and the winning-line statistic is permutation-invariant.
    Two nontrivial permutations are exercised. Every other latent field is
    painted directly (colours, radius, spacing, origin, lattice extent).
    """
    holes = scene["holes"]
    shift = max(1, len(holes) // 3)
    rotated = dict(scene, holes=holes[shift:] + holes[:shift])
    flipped = dict(scene, holes=list(reversed(holes)))
    return [("hole_order_rotation", rotated), ("hole_order_reversal", flipped)]


def _sites_ok(rows: int, cols: int, holes) -> bool:
    hole_set = set(holes)
    for i in range(rows):
        if cols - sum(1 for j in range(cols) if (i, j) in hole_set) < MIN_DOTS_PER_ROW:
            return False
    for j in range(cols):
        if rows - sum(1 for i in range(rows) if (i, j) in hole_set) < MIN_DOTS_PER_COL:
            return False
    return True


def _pick_spread(rng: random.Random, sites: list, count: int) -> list | None:
    """Choose ``count`` sites of a line so that no two chosen sites are adjacent."""
    for _ in range(200):
        chosen = sorted(rng.sample(range(len(sites)), count))
        if all(b - a >= 2 for a, b in zip(chosen, chosen[1:])):
            return [sites[index] for index in chosen]
    return None


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    boundary = seed % BOUNDARY_EVERY == 0
    # The answer is drawn once, before rejection sampling, so that unequal
    # acceptance rates between directions cannot bias the class balance.
    family = rng.choice(FAMILIES)
    for _ in range(4000):
        rows = rng.choice(ROW_CHOICES)
        cols = rng.choice(COL_CHOICES)
        keys = _long_lines(rows, cols, family)
        if not keys:
            continue
        key = rng.choice(keys)
        sites = _line_sites(rows, cols, family, key)
        sites.sort()
        planted = rng.choice(LINE_HOLES)
        if planted > (len(sites) + 1) // 2:
            continue
        line_holes = _pick_spread(rng, sites, planted)
        if line_holes is None:
            continue
        holes = list(line_holes)

        if boundary:
            # Plant a rival line one vacancy short of the winner: the scene then
            # sits on the finite decision boundary and is quarantined.
            rival_family = rng.choice([f for f in FAMILIES if f != family])
            rival_keys = _long_lines(rows, cols, rival_family)
            rng.shuffle(rival_keys)
            for rival_key in rival_keys:
                rival_sites = [
                    site
                    for site in _line_sites(rows, cols, rival_family, rival_key)
                    if site not in set(holes)
                ]
                rival_sites.sort()
                rival = _pick_spread(rng, rival_sites, planted - 1)
                if rival is not None:
                    holes.extend(rival)
                    break

        decoys = rng.choice(DECOY_HOLES)
        pool = [
            (i, j)
            for i in range(rows)
            for j in range(cols)
            if (i, j) not in set(holes)
        ]
        holes.extend(rng.sample(pool, min(decoys, len(pool))))
        rng.shuffle(holes)

        if not _sites_ok(rows, cols, holes):
            continue
        flat = _ranked(rows, cols, holes)
        best = flat[0]
        if best[1] != family or best[2] != key:
            continue
        if len([item for item in flat if item[0] == best[0]]) != 1:
            continue
        gap = best[0] - flat[1][0]
        if boundary and gap != 1:
            continue
        if not boundary and gap < 2:
            continue

        spacing = rng.choice(SPACING_CHOICES)
        radius = rng.choice(RADIUS_CHOICES)
        span_x = (cols - 1) * spacing
        span_y = (rows - 1) * spacing
        slack_x = (CANVAS - span_x) // 2 - radius - 6
        slack_y = (CANVAS - span_y) // 2 - radius - 6
        origin_x = (CANVAS - span_x) // 2 + rng.randint(-slack_x // 2, slack_x // 2)
        origin_y = (CANVAS - span_y) // 2 + rng.randint(-slack_y // 2, slack_y // 2)
        return {
            "width": CANVAS,
            "height": CANVAS,
            "bg_color": rng.choice(BG_COLORS),
            "dot_color": rng.choice(DOT_COLORS),
            "dot_radius": radius,
            "spacing": spacing,
            "origin_x": int(origin_x),
            "origin_y": int(origin_y),
            "rows": rows,
            "cols": cols,
            "holes": [[int(i), int(j)] for i, j in holes],
        }
    raise RuntimeError(f"scene sampling failed for seed {seed}")


def render(scene: dict) -> Image.Image:
    width, height = scene["width"], scene["height"]
    image = Image.new("RGB", (width * SS, height * SS), scene["bg_color"])
    draw = ImageDraw.Draw(image)
    radius = scene["dot_radius"] * SS
    spacing = scene["spacing"]
    holes = {tuple(hole) for hole in scene["holes"]}
    for i in range(scene["rows"]):
        for j in range(scene["cols"]):
            if (i, j) in holes:
                continue
            cx = (scene["origin_x"] + j * spacing) * SS
            cy = (scene["origin_y"] + i * spacing) * SS
            draw.ellipse(
                [cx - radius, cy - radius, cx + radius, cy + radius],
                fill=scene["dot_color"],
            )
    return image.resize((width, height), Image.LANCZOS)
