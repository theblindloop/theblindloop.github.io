"""Independent pixel-only inverse for critical doorway connectivity."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "critical_gate_connectivity-oracle-1.0.0"

WIDTH = 840
HEIGHT = 600
COLS = 5
ROWS = 4
GRID_X = 100
GRID_Y = 70
CELL_W = 120
CELL_H = 120
SCALE = 4
WALL_WIDTH = 14
DOOR_GAP = 72

WALL = np.array((37, 45, 57), dtype=np.int16)
MARKERS = {
    "red": np.array((218, 61, 70), dtype=np.int16),
    "blue": np.array((54, 106, 207), dtype=np.int16),
    "orange": np.array((235, 139, 38), dtype=np.int16),
}
GATES = {
    "cyan": np.array((24, 169, 183), dtype=np.int16),
    "violet": np.array((145, 76, 192), dtype=np.int16),
}


class OracleError(ValueError):
    """Raised when the raster does not expose one unambiguous answer."""


def _wall_mask(array: np.ndarray) -> np.ndarray:
    return (
        (array[:, :, 0] < 120)
        & (array[:, :, 1] < 130)
        & (array[:, :, 2] < 145)
    )


def _color_center(array: np.ndarray, colour: np.ndarray, name: str) -> tuple[float, float]:
    delta = array.astype(np.int32) - colour.astype(np.int32)
    mask = np.sum(delta * delta, axis=2) <= 65 * 65
    ys, xs = np.where(mask)
    if len(xs) < 250:
        raise OracleError(f"{name} mark is missing or too small")
    return float(xs.mean()), float(ys.mean())


def _gate_location(x: float, y: float) -> tuple[str, float, float]:
    candidates: list[tuple[float, str, float, float]] = []
    for boundary in range(1, COLS):
        gx = GRID_X + boundary * CELL_W
        for row in range(ROWS):
            gy = GRID_Y + (row + 0.5) * CELL_H
            candidates.append(((x - gx) ** 2 + (y - gy) ** 2, "v", gx, gy))
    for boundary in range(1, ROWS):
        gy = GRID_Y + boundary * CELL_H
        for col in range(COLS):
            gx = GRID_X + (col + 0.5) * CELL_W
            candidates.append(((x - gx) ** 2 + (y - gy) ** 2, "h", gx, gy))
    distance, axis, gx, gy = min(candidates)
    if distance > 18.0 * 18.0:
        raise OracleError(f"gate center ({x:.1f},{y:.1f}) is not on a visible doorway")
    return axis, gx, gy


def _coarse_outside(wall: np.ndarray) -> np.ndarray:
    height, width = wall.shape
    if (height % SCALE) or (width % SCALE):
        raise OracleError("unexpected image dimensions")
    coarse_wall = wall.reshape(height // SCALE, SCALE, width // SCALE, SCALE).any(axis=(1, 3))
    coarse_height, coarse_width = coarse_wall.shape
    seen = np.zeros_like(coarse_wall, dtype=bool)
    queue: deque[tuple[int, int]] = deque()
    for x in range(coarse_width):
        for y in (0, coarse_height - 1):
            if not coarse_wall[y, x] and not seen[y, x]:
                seen[y, x] = True
                queue.append((y, x))
    for y in range(coarse_height):
        for x in (0, coarse_width - 1):
            if not coarse_wall[y, x] and not seen[y, x]:
                seen[y, x] = True
                queue.append((y, x))
    while queue:
        y, x = queue.popleft()
        for next_y, next_x in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if (
                0 <= next_y < coarse_height
                and 0 <= next_x < coarse_width
                and not coarse_wall[next_y, next_x]
                and not seen[next_y, next_x]
            ):
                seen[next_y, next_x] = True
                queue.append((next_y, next_x))
    return seen


def _sealed_mask(wall: np.ndarray, axis: str, x: float, y: float) -> np.ndarray:
    sealed = wall.copy()
    ix = int(round(x))
    iy = int(round(y))
    along = DOOR_GAP // 2 + WALL_WIDTH // 2 + 4
    across = WALL_WIDTH // 2 + 3
    if axis == "v":
        sealed[max(0, iy - along):min(HEIGHT, iy + along + 1), max(0, ix - across):min(WIDTH, ix + across + 1)] = True
    elif axis == "h":
        sealed[max(0, iy - across):min(HEIGHT, iy + across + 1), max(0, ix - along):min(WIDTH, ix + along + 1)] = True
    else:
        raise OracleError(f"unknown gate axis {axis!r}")
    return sealed


def decision_from_image(image: Image.Image) -> str:
    if image.size != (WIDTH, HEIGHT):
        raise OracleError(f"unexpected image size {image.size}")
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    wall = _wall_mask(array)
    blue_x, blue_y = _color_center(array, MARKERS["blue"], "blue")
    blue_probe = (int(blue_y) // SCALE, int(blue_x) // SCALE)
    critical: list[str] = []
    for name in ("cyan", "violet"):
        gate_x, gate_y = _color_center(array, GATES[name], name)
        axis, snapped_x, snapped_y = _gate_location(gate_x, gate_y)
        outside = _coarse_outside(_sealed_mask(wall, axis, snapped_x, snapped_y))
        probe_y = min(outside.shape[0] - 1, max(0, blue_probe[0]))
        probe_x = min(outside.shape[1] - 1, max(0, blue_probe[1]))
        if not bool(outside[probe_y, probe_x]):
            critical.append(name)
    if len(critical) != 1:
        raise OracleError(f"expected one critical gate, found {critical}")
    return critical[0]
