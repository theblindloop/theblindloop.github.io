"""Renderer and analytic label for a white-space edge-sensitivity question.

The dark strokes are barriers.  The open internal doorways form a spanning tree
of rectangular white cells with one opening to the image exterior.  Two colored
rings identify two of those open doorways.  The label is the ring whose doorway
must be sealed to make the blue marker unreachable from the exterior.
"""

from __future__ import annotations

import math
import random
from collections import deque

from PIL import Image, ImageDraw

WIDTH = 840
HEIGHT = 600
COLS = 5
ROWS = 4
GRID_X = 100
GRID_Y = 70
CELL_W = 120
CELL_H = 120
GRID_RIGHT = GRID_X + COLS * CELL_W
GRID_BOTTOM = GRID_Y + ROWS * CELL_H
SCALE = 4

BACKGROUND = (249, 248, 244)
WALL = (37, 45, 57)
MARKER_COLORS = {
    "red": (218, 61, 70),
    "blue": (54, 106, 207),
    "orange": (235, 139, 38),
}
GATE_COLORS = {
    "cyan": (24, 169, 183),
    "violet": (145, 76, 192),
}

WALL_WIDTH = 14
DOOR_GAP = 72
MARKER_RADIUS = 21
GATE_RADIUS = 20

RENDERER_VERSION = "critical_gate_connectivity-renderer-1.0.0"
_OUTSIDE = COLS * ROWS
_ALL_V_DOORS = [(boundary, row) for boundary in range(1, COLS) for row in range(ROWS)]
_ALL_H_DOORS = [(col, boundary) for boundary in range(1, ROWS) for col in range(COLS)]
_ALL_OUTER_DOORS = (
    [("left", row) for row in range(ROWS)]
    + [("right", row) for row in range(ROWS)]
    + [("top", col) for col in range(COLS)]
    + [("bottom", col) for col in range(COLS)]
)


def _cell_id(col: int, row: int) -> int:
    return row * COLS + col


def _cell_from_id(index: int) -> tuple[int, int]:
    return index % COLS, index // COLS


def _cell_center(col: int, row: int) -> tuple[float, float]:
    return GRID_X + (col + 0.5) * CELL_W, GRID_Y + (row + 0.5) * CELL_H


def _edge_between(first: tuple[int, int], second: tuple[int, int]) -> tuple[str, int, int]:
    col1, row1 = first
    col2, row2 = second
    if row1 == row2 and abs(col1 - col2) == 1:
        return ("v", max(col1, col2), row1)
    if col1 == col2 and abs(row1 - row2) == 1:
        return ("h", col1, max(row1, row2))
    raise ValueError(f"non-neighbouring cells {first!r}, {second!r}")


def _edge_center(edge: tuple[str, int, int]) -> tuple[float, float]:
    axis, first, second = edge
    if axis == "v":
        return GRID_X + first * CELL_W, GRID_Y + (second + 0.5) * CELL_H
    if axis == "h":
        return GRID_X + (first + 0.5) * CELL_W, GRID_Y + second * CELL_H
    raise ValueError(f"unknown edge axis {axis!r}")


def _normalise(scene: dict) -> tuple[list[tuple[int, int]], list[tuple[int, int]], dict[str, list[int]], list[tuple[str, int, int]]]:
    vertical = sorted((int(item[0]), int(item[1])) for item in scene["v_doors"])
    horizontal = sorted((int(item[0]), int(item[1])) for item in scene["h_doors"])
    outer = {
        side: sorted(int(value) for value in scene["outer_doors"][side])
        for side in ("left", "right", "top", "bottom")
    }
    gates = [
        (str(item[0]), int(item[1]), int(item[2]))
        for item in scene["gate_edges"]
    ]
    return vertical, horizontal, outer, gates


def _adjacency(scene: dict, excluded: tuple[str, int, int] | None = None) -> dict[int, set[int]]:
    vertical, horizontal, outer, _ = _normalise(scene)
    graph: dict[int, set[int]] = {index: set() for index in range(_OUTSIDE + 1)}
    for boundary, row in vertical:
        edge = ("v", boundary, row)
        if edge == excluded:
            continue
        left = _cell_id(boundary - 1, row)
        right = _cell_id(boundary, row)
        graph[left].add(right)
        graph[right].add(left)
    for col, boundary in horizontal:
        edge = ("h", col, boundary)
        if edge == excluded:
            continue
        above = _cell_id(col, boundary - 1)
        below = _cell_id(col, boundary)
        graph[above].add(below)
        graph[below].add(above)
    for row in outer["left"]:
        cell = _cell_id(0, row)
        graph[_OUTSIDE].add(cell)
        graph[cell].add(_OUTSIDE)
    for row in outer["right"]:
        cell = _cell_id(COLS - 1, row)
        graph[_OUTSIDE].add(cell)
        graph[cell].add(_OUTSIDE)
    for col in outer["top"]:
        cell = _cell_id(col, 0)
        graph[_OUTSIDE].add(cell)
        graph[cell].add(_OUTSIDE)
    for col in outer["bottom"]:
        cell = _cell_id(col, ROWS - 1)
        graph[_OUTSIDE].add(cell)
        graph[cell].add(_OUTSIDE)
    return graph


def _outside_reachable(scene: dict, excluded: tuple[str, int, int] | None = None) -> set[int]:
    graph = _adjacency(scene, excluded)
    found = {_OUTSIDE}
    queue: deque[int] = deque([_OUTSIDE])
    while queue:
        node = queue.popleft()
        for neighbour in sorted(graph[node]):
            if neighbour not in found:
                found.add(neighbour)
                queue.append(neighbour)
    return found


def analytic_gold(scene: dict) -> str:
    """Select the gate whose closure disconnects the visible blue marker."""

    vertical, horizontal, outer, gates = _normalise(scene)
    _ = vertical, horizontal, outer  # make the rendered topology explicit here
    cells = [tuple(int(value) for value in position) for position in scene["marker_cells"]]
    marker_colors = [str(value) for value in scene["marker_colors"]]
    gate_colors = [str(value) for value in scene["gate_colors"]]
    if len(cells) != 3 or len(set(cells)) != 3:
        raise ValueError("expected three distinct marker cells")
    if len(gates) != 2 or len(gate_colors) != 2:
        raise ValueError("expected two marked gates")
    if "blue" not in marker_colors:
        raise ValueError("blue marker is required")
    blue_cell = _cell_id(*cells[marker_colors.index("blue")])
    critical: list[str] = []
    for edge, colour in zip(gates, gate_colors):
        if blue_cell not in _outside_reachable(scene, edge):
            critical.append(colour)
    if len(critical) != 1:
        raise ValueError(f"expected one critical gate, found {critical}")
    return critical[0]


def margin(scene: dict) -> float:
    """Return the smallest visible clearance relevant to the topology."""

    wall_width = float(scene["wall_width"])
    door_gap = float(scene["door_gap"])
    marker_radius = float(scene["marker_radius"])
    gate_radius = float(scene["gate_radius"])
    marker_clearance = min(
        CELL_W / 2.0 - marker_radius - wall_width / 2.0,
        CELL_H / 2.0 - marker_radius - wall_width / 2.0,
    )
    gate_clearance = door_gap / 2.0 - gate_radius
    return float(min(door_gap - wall_width, marker_clearance, gate_clearance))


def is_quarantined(scene: dict) -> bool:
    try:
        if margin(scene) < 14.0:
            return True
        vertical, horizontal, outer, gates = _normalise(scene)
        if len(vertical) + len(horizontal) != _OUTSIDE - 1:
            return True
        if any(item not in _ALL_V_DOORS for item in vertical) or len(set(vertical)) != len(vertical):
            return True
        if any(item not in _ALL_H_DOORS for item in horizontal) or len(set(horizontal)) != len(horizontal):
            return True
        for side, values in outer.items():
            limit = ROWS if side in {"left", "right"} else COLS
            if any(value < 0 or value >= limit for value in values) or len(set(values)) != len(values):
                return True
        if sum(len(values) for values in outer.values()) != 1:
            return True
        door_set = {("v", *item) for item in vertical} | {("h", *item) for item in horizontal}
        if len(gates) != 2 or len(set(gates)) != 2 or any(edge not in door_set for edge in gates):
            return True
        cells = [tuple(int(value) for value in position) for position in scene["marker_cells"]]
        if len(cells) != 3 or len(set(cells)) != 3 or any(
            col < 0 or col >= COLS or row < 0 or row >= ROWS for col, row in cells
        ):
            return True
        if set(str(value) for value in scene["marker_colors"]) != set(MARKER_COLORS):
            return True
        if set(str(value) for value in scene["gate_colors"]) != set(GATE_COLORS):
            return True
        if len(_outside_reachable(scene)) != _OUTSIDE + 1:
            return True
        analytic_gold(scene)
        return False
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent field is omitted from the visible scene or label."""

    return []


def _cell_neighbours(cell: tuple[int, int]) -> list[tuple[int, int]]:
    col, row = cell
    candidates = [(col - 1, row), (col + 1, row), (col, row - 1), (col, row + 1)]
    return [
        (next_col, next_row)
        for next_col, next_row in candidates
        if 0 <= next_col < COLS and 0 <= next_row < ROWS
    ]


def _random_tree(rng: random.Random, root: tuple[int, int]) -> list[tuple[str, int, int]]:
    visited = {root}
    stack = [root]
    edges: list[tuple[str, int, int]] = []
    while stack:
        current = stack[-1]
        choices = [cell for cell in _cell_neighbours(current) if cell not in visited]
        if not choices:
            stack.pop()
            continue
        next_cell = rng.choice(choices)
        visited.add(next_cell)
        stack.append(next_cell)
        edges.append(_edge_between(current, next_cell))
    return sorted(edges)


def _tree_parent(root: tuple[int, int], edges: list[tuple[str, int, int]]) -> tuple[dict[tuple[int, int], tuple[int, int] | None], dict[tuple[int, int], int]]:
    neighbours: dict[tuple[int, int], list[tuple[int, int]]] = {
        (col, row): [] for row in range(ROWS) for col in range(COLS)
    }
    for edge in edges:
        axis, first, second = edge
        if axis == "v":
            left, right = (first - 1, second), (first, second)
        else:
            left, right = (first, second - 1), (first, second)
        neighbours[left].append(right)
        neighbours[right].append(left)
    parent: dict[tuple[int, int], tuple[int, int] | None] = {root: None}
    depth: dict[tuple[int, int], int] = {root: 0}
    queue: deque[tuple[int, int]] = deque([root])
    while queue:
        node = queue.popleft()
        for neighbour in sorted(neighbours[node]):
            if neighbour not in parent:
                parent[neighbour] = node
                depth[neighbour] = depth[node] + 1
                queue.append(neighbour)
    return parent, depth


def _subtrees(parent: dict[tuple[int, int], tuple[int, int] | None]) -> dict[tuple[int, int], set[tuple[int, int]]]:
    children: dict[tuple[int, int], list[tuple[int, int]]] = {node: [] for node in parent}
    for node, ancestor in parent.items():
        if ancestor is not None:
            children[ancestor].append(node)
    output: dict[tuple[int, int], set[tuple[int, int]]] = {}

    def collect(node: tuple[int, int]) -> set[tuple[int, int]]:
        found = {node}
        for child in children[node]:
            found.update(collect(child))
        output[node] = found
        return found

    root = next(node for node, ancestor in parent.items() if ancestor is None)
    collect(root)
    return output


def _path_edges(target: tuple[int, int], parent: dict[tuple[int, int], tuple[int, int] | None]) -> list[tuple[str, int, int]]:
    result: list[tuple[str, int, int]] = []
    node = target
    while parent[node] is not None:
        ancestor = parent[node]
        assert ancestor is not None
        result.append(_edge_between(node, ancestor))
        node = ancestor
    return result


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) * 1009 + 7317)
    boundary_cells = [
        (col, row)
        for row in range(ROWS)
        for col in range(COLS)
        if col in {0, COLS - 1} or row in {0, ROWS - 1}
    ]
    all_cells = [(col, row) for row in range(ROWS) for col in range(COLS)]
    for _ in range(6000):
        root = rng.choice(boundary_cells)
        tree_edges = _random_tree(rng, root)
        if len(tree_edges) != _OUTSIDE - 1:
            continue
        parent, depth = _tree_parent(root, tree_edges)
        if len(parent) != _OUTSIDE:
            continue
        subtrees = _subtrees(parent)
        targets = [
            cell for cell in all_cells
            if depth[cell] >= 6
            and any(
                depth[child] >= 2 and depth[cell] - depth[child] >= 2
                for child, ancestor in parent.items()
                if ancestor is not None and child in _path_nodes(cell, parent)
            )
        ]
        if not targets:
            continue
        target = rng.choice(targets)
        path_edges = _path_edges(target, parent)
        path_edge_set = set(path_edges)
        path_nodes = _path_nodes(target, parent)
        a_options = [
            edge for edge in path_edges
            if _edge_child(edge, parent) in path_nodes
            and depth[_edge_child(edge, parent)] >= 2
            and depth[target] - depth[_edge_child(edge, parent)] >= 2
        ]
        b_options = []
        for child, ancestor in parent.items():
            if ancestor is None:
                continue
            edge = _edge_between(child, ancestor)
            if edge in path_edge_set:
                continue
            if depth[child] >= 2 and len(subtrees[child]) >= 2:
                b_options.append(edge)
        if not a_options or not b_options:
            continue
        gate_a = rng.choice(a_options)
        gate_b = rng.choice(b_options)
        if math.dist(_edge_center(gate_a), _edge_center(gate_b)) < 110.0:
            continue
        if math.dist(_edge_center(gate_a), _cell_center(*target)) < 170.0:
            continue
        b_child = _edge_child(gate_b, parent)
        red_options = sorted(subtrees[b_child] - {target})
        if not red_options:
            continue
        red = rng.choice(red_options)
        orange_options = [cell for cell in all_cells if cell not in {target, red}]
        orange = rng.choice(orange_options)

        outer = {side: [] for side in ("left", "right", "top", "bottom")}
        possible_sides = []
        col, row = root
        if col == 0:
            possible_sides.append("left")
        if col == COLS - 1:
            possible_sides.append("right")
        if row == 0:
            possible_sides.append("top")
        if row == ROWS - 1:
            possible_sides.append("bottom")
        side = rng.choice(possible_sides)
        outer[side].append(row if side in {"left", "right"} else col)

        marker_records = [(target, "blue"), (red, "red"), (orange, "orange")]
        rng.shuffle(marker_records)
        gate_edges = [gate_a, gate_b]
        gate_colors = ["cyan", "violet"]
        rng.shuffle(gate_colors)
        return {
            "v_doors": [list(edge[1:]) for edge in tree_edges if edge[0] == "v"],
            "h_doors": [list(edge[1:]) for edge in tree_edges if edge[0] == "h"],
            "outer_doors": {side: list(values) for side, values in sorted(outer.items())},
            "gate_edges": [list(edge) for edge in gate_edges],
            "gate_colors": gate_colors,
            "marker_cells": [list(cell) for cell, _ in marker_records],
            "marker_colors": [colour for _, colour in marker_records],
            "wall_width": WALL_WIDTH,
            "door_gap": DOOR_GAP,
            "marker_radius": MARKER_RADIUS,
            "gate_radius": GATE_RADIUS,
        }
    raise RuntimeError("could not sample a critical-gate maze")


def _path_nodes(target: tuple[int, int], parent: dict[tuple[int, int], tuple[int, int] | None]) -> set[tuple[int, int]]:
    nodes = {target}
    node = target
    while parent[node] is not None:
        ancestor = parent[node]
        assert ancestor is not None
        nodes.add(ancestor)
        node = ancestor
    return nodes


def _edge_child(edge: tuple[str, int, int], parent: dict[tuple[int, int], tuple[int, int] | None]) -> tuple[int, int]:
    axis, first, second = edge
    if axis == "v":
        endpoints = [(first - 1, second), (first, second)]
    else:
        endpoints = [(first, second - 1), (first, second)]
    for node in endpoints:
        ancestor = parent[node]
        if ancestor in endpoints and ancestor != node:
            return node
    raise ValueError(f"edge is not in the rooted tree: {edge!r}")


def _draw_wall(draw: ImageDraw.ImageDraw, p1: tuple[float, float], p2: tuple[float, float], width: int) -> None:
    draw.line(
        [(round(p1[0] * SCALE), round(p1[1] * SCALE)), (round(p2[0] * SCALE), round(p2[1] * SCALE))],
        fill=WALL,
        width=round(width * SCALE),
    )


def _draw_interval_with_door(
    draw: ImageDraw.ImageDraw,
    axis: str,
    fixed: float,
    start: float,
    end: float,
    door: bool,
    width: int,
    gap: int,
) -> None:
    if not door:
        p1 = (fixed, start) if axis == "vertical" else (start, fixed)
        p2 = (fixed, end) if axis == "vertical" else (end, fixed)
        _draw_wall(draw, p1, p2, width)
        return
    centre = (start + end) / 2.0
    half = gap / 2.0
    first_end = centre - half
    second_start = centre + half
    if first_end > start:
        p1 = (fixed, start) if axis == "vertical" else (start, fixed)
        p2 = (fixed, first_end) if axis == "vertical" else (first_end, fixed)
        _draw_wall(draw, p1, p2, width)
    if second_start < end:
        p1 = (fixed, second_start) if axis == "vertical" else (second_start, fixed)
        p2 = (fixed, end) if axis == "vertical" else (end, fixed)
        _draw_wall(draw, p1, p2, width)


def render(scene: dict) -> Image.Image:
    wall_width = int(scene["wall_width"])
    gap = int(scene["door_gap"])
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    vertical, horizontal, outer, _ = _normalise(scene)
    v_set = set(vertical)
    h_set = set(horizontal)

    for row in range(ROWS):
        y0 = GRID_Y + row * CELL_H
        y1 = y0 + CELL_H
        _draw_interval_with_door(draw, "vertical", GRID_X, y0, y1, row in outer["left"], wall_width, gap)
        _draw_interval_with_door(draw, "vertical", GRID_RIGHT, y0, y1, row in outer["right"], wall_width, gap)
    for col in range(COLS):
        x0 = GRID_X + col * CELL_W
        x1 = x0 + CELL_W
        _draw_interval_with_door(draw, "horizontal", GRID_Y, x0, x1, col in outer["top"], wall_width, gap)
        _draw_interval_with_door(draw, "horizontal", GRID_BOTTOM, x0, x1, col in outer["bottom"], wall_width, gap)

    for boundary in range(1, COLS):
        x = GRID_X + boundary * CELL_W
        for row in range(ROWS):
            y0 = GRID_Y + row * CELL_H
            _draw_interval_with_door(draw, "vertical", x, y0, y0 + CELL_H, (boundary, row) in v_set, wall_width, gap)
    for boundary in range(1, ROWS):
        y = GRID_Y + boundary * CELL_H
        for col in range(COLS):
            x0 = GRID_X + col * CELL_W
            _draw_interval_with_door(draw, "horizontal", y, x0, x0 + CELL_W, (col, boundary) in h_set, wall_width, gap)

    for edge, colour in zip(scene["gate_edges"], scene["gate_colors"]):
        cx, cy = _edge_center((str(edge[0]), int(edge[1]), int(edge[2])))
        radius = float(scene["gate_radius"])
        box = [
            round((cx - radius) * SCALE),
            round((cy - radius) * SCALE),
            round((cx + radius) * SCALE),
            round((cy + radius) * SCALE),
        ]
        draw.ellipse(box, fill=GATE_COLORS[str(colour)])
        inner = radius * 0.43
        draw.ellipse(
            [
                round((cx - inner) * SCALE),
                round((cy - inner) * SCALE),
                round((cx + inner) * SCALE),
                round((cy + inner) * SCALE),
            ],
            fill=BACKGROUND,
        )

    for position, colour in zip(scene["marker_cells"], scene["marker_colors"]):
        cx, cy = _cell_center(int(position[0]), int(position[1]))
        radius = float(scene["marker_radius"])
        draw.ellipse(
            [
                round((cx - radius) * SCALE),
                round((cy - radius) * SCALE),
                round((cx + radius) * SCALE),
                round((cy + radius) * SCALE),
            ],
            fill=MARKER_COLORS[str(colour)],
        )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
