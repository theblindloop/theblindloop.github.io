"""Inscribed-square band world: scene sampler, renderer and analytic gold.

One solid slate plate is drawn as a horizontally contiguous run of full-height
column strips, so the plate is a single connected, hole-free, area-like region of
medium span.  The decision recovers a two-dimensional clearance from that plate --
the side S of the largest axis-aligned square that fits entirely inside it -- and
places S in one of three bands cut by the plate's own width W at W/3 and W/2.

Both measured quantities come from the same region, so the decisive support is the
plate interior: no external ruler, legend or second object carries the answer.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "inscribed_square_band-0.1.0"

CANVAS_W = 640
CANVAS_H = 480

SHEET = (243, 241, 235)
PLATE = (74, 82, 96)      # the measured plate; unique in the palette
DECOY = (198, 199, 203)   # pale grey decoy body, declared decoration
DECO = (170, 171, 175)    # ticks and rings
FRAME = (60, 62, 68)

QUARANTINE_MARGIN = 3.0

# Plate envelope: roughly 40 per cent of the sheet in each direction, so the
# decisive support is one connected region of medium span that never crosses the
# image, and no tiny crop of the plate can settle the band on its own.
MIN_W, MAX_W = 225, 265
MIN_H, MAX_H = 180, 215
MIN_STRIP_W, MAX_STRIP_W = 20, 46
MIN_STRIP_H = 50
MIN_OVERLAP = 34
MIN_FILL, MAX_FILL = 0.58, 0.80

BANDS = ("less_than_third", "third_to_half", "more_than_half")


# --------------------------------------------------------------------------
# analytic gold (reads scene["plate"] only)
# --------------------------------------------------------------------------
def _strips(scene: dict) -> list[list[int]]:
    """Column strips of the plate, ordered left to right (order-independent)."""
    return sorted((list(rect) for rect in scene["plate"]["strips"]), key=lambda r: r[0])


def _clearance_and_width(scene: dict) -> tuple[int, int]:
    """(S, W): largest inscribed axis-aligned square side, and full plate width."""
    strips = _strips(scene)
    width = strips[-1][2] - strips[0][0] + 1
    best = 0
    for i in range(len(strips)):
        top = strips[i][1]
        bottom = strips[i][3]
        for j in range(i, len(strips)):
            top = max(top, strips[j][1])
            bottom = min(bottom, strips[j][3])
            height = bottom - top + 1
            if height <= 0:
                break
            span = strips[j][2] - strips[i][0] + 1
            side = height if height < span else span
            if side > best:
                best = side
    return best, width


def analytic_gold(scene: dict) -> str:
    """Band of S measured against the plate's own width W."""
    side, width = _clearance_and_width(scene)
    if 3 * side < width:
        return "less_than_third"
    if 2 * side > width:
        return "more_than_half"
    return "third_to_half"


def margin(scene: dict) -> float:
    """Pixel distance from S to the nearer of the two band boundaries W/3, W/2."""
    side, width = _clearance_and_width(scene)
    lower = abs(side - width / 3.0)
    upper = abs(side - width / 2.0)
    return round(lower if lower < upper else upper, 12)


def is_quarantined(scene: dict) -> bool:
    """S sits within a few pixels of a band boundary, so the band is unstable."""
    return bool(margin(scene) < QUARANTINE_MARGIN)


# --------------------------------------------------------------------------
# scene sampling
# --------------------------------------------------------------------------
def _try_plate(rng: random.Random) -> dict | None:
    """One candidate plate: contiguous full-height column strips, no holes."""
    total_w = rng.randint(MIN_W, MAX_W)
    total_h = rng.randint(MIN_H, MAX_H)
    origin_x = rng.randint(190, 300)
    origin_y = rng.randint(112, 150)

    widths: list[int] = []
    remaining = total_w
    while remaining > 0:
        if remaining <= MAX_STRIP_W:
            if remaining < MIN_STRIP_W:
                return None
            widths.append(remaining)
            break
        step = rng.randint(MIN_STRIP_W, MAX_STRIP_W)
        if remaining - step < MIN_STRIP_W:
            step = remaining - MIN_STRIP_W
            if step < MIN_STRIP_W:
                return None
        widths.append(step)
        remaining -= step
    if not 5 <= len(widths) <= 9:
        return None

    tops: list[int] = []
    bottoms: list[int] = []
    for _ in widths:
        height = rng.randint(MIN_STRIP_H, total_h)
        top = rng.randint(0, total_h - height)
        tops.append(top)
        bottoms.append(top + height - 1)
    if min(tops) != 0 or max(bottoms) != total_h - 1:
        return None
    for index in range(1, len(widths)):
        low = max(tops[index - 1], tops[index])
        high = min(bottoms[index - 1], bottoms[index])
        if high - low + 1 < MIN_OVERLAP:
            return None

    strips = []
    cursor = origin_x
    area = 0
    for width, top, bottom in zip(widths, tops, bottoms):
        strips.append([cursor, origin_y + top, cursor + width - 1, origin_y + bottom])
        area += width * (bottom - top + 1)
        cursor += width
    if not MIN_FILL <= area / float(total_w * total_h) <= MAX_FILL:
        return None
    return {"strips": strips}


def _sample_deco(rng: random.Random, plate: dict) -> dict:
    """Decoration confined to the sheet margins: never measured, never touching."""
    ticks = []
    for index in range(rng.randint(9, 14)):
        x = 46 + index * rng.randint(28, 34)
        if x > CANVAS_W - 46:
            break
        ticks.append([x, 34, x, 34 + rng.randint(10, 20)])
    rings = [
        [30, 396, 30 + rng.randint(20, 30), 396 + rng.randint(20, 30)],
        [CANVAS_W - 62, 60, CANVAS_W - 62 + rng.randint(20, 30), 60 + rng.randint(20, 30)],
    ]
    # A pale decoy body of comparable size, kept in the bottom margin strip so it
    # never touches the plate: colour, not size, selects the measured region.
    decoy_w = rng.randint(96, 150)
    decoy_h = rng.randint(60, 74)
    decoy_x = rng.choice([40, CANVAS_W - 40 - decoy_w])
    decoy_y = rng.randint(378, 456 - decoy_h)
    decoy = [decoy_x, decoy_y, decoy_x + decoy_w - 1, decoy_y + decoy_h - 1]
    return {"ticks": ticks, "rings": rings, "decoy": decoy}


def sample_scene(seed: int) -> dict:
    """Deterministic scene whose band is targeted for balance and boundary cover."""
    rng = random.Random(seed * 7919 + 13)
    target = BANDS[seed % 3]
    want_boundary = seed % 8 == 4

    fallback = None
    for _ in range(60000):
        plate = _try_plate(rng)
        if plate is None:
            continue
        probe = {"plate": plate}
        if analytic_gold(probe) != target:
            continue
        near = margin(probe) < QUARANTINE_MARGIN
        if near == want_boundary and (near or margin(probe) >= QUARANTINE_MARGIN + 1.0):
            return {
                "canvas_w": CANVAS_W,
                "canvas_h": CANVAS_H,
                "plate": plate,
                "deco": _sample_deco(rng, plate),
            }
        if fallback is None and not near:
            fallback = plate
    if fallback is None:
        raise RuntimeError(f"no plate found for seed {seed}")
    return {
        "canvas_w": CANVAS_W,
        "canvas_h": CANVAS_H,
        "plate": fallback,
        "deco": _sample_deco(rng, fallback),
    }


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def render(scene: dict) -> Image.Image:
    """Flat, antialiasing-free fills, so the mask equals the analytic region."""
    image = Image.new("RGB", (scene["canvas_w"], scene["canvas_h"]), SHEET)
    draw = ImageDraw.Draw(image)
    draw.rectangle([12, 12, scene["canvas_w"] - 13, scene["canvas_h"] - 13], outline=FRAME)

    deco = scene["deco"]
    draw.rectangle(deco["decoy"], fill=DECOY)
    for tick in deco["ticks"]:
        draw.line(tick, fill=DECO, width=1)
    for ring in deco["rings"]:
        draw.ellipse(ring, outline=DECO, width=2)
    for strip in scene["plate"]["strips"]:
        draw.rectangle(strip, fill=PLATE)
    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical relabellings of the scene that must preserve the gold."""
    twins: list[tuple[str, dict]] = []

    reversed_strips = dict(scene)
    reversed_strips["plate"] = {"strips": list(reversed(scene["plate"]["strips"]))}
    twins.append(("plate_strip_order_reversed", reversed_strips))

    rotated = dict(scene)
    strips = scene["plate"]["strips"]
    rotated["plate"] = {"strips": strips[2:] + strips[:2]}
    twins.append(("plate_strip_order_rotated", rotated))

    shuffled_deco = dict(scene)
    deco = dict(scene["deco"])
    deco["ticks"] = list(reversed(scene["deco"]["ticks"]))
    deco["rings"] = list(reversed(scene["deco"]["rings"]))
    shuffled_deco["deco"] = deco
    twins.append(("deco_draw_order_reversed", shuffled_deco))
    return twins
