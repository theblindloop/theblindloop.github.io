"""Image-only inverse arm for stack_op_replay.

Receives a rendered PNG and nothing else: no scene, no renderer, no gold, no
prompt or generator code. It re-derives the whole decision from pixels:

  1. segment ink (any non-white pixel) into connected components;
  2. keep components that carry chromatic (non-black) fill -- the bin walls and
     the order arrow are pure black and drop out;
  3. a component whose bounding box contains an interior white glyph is an
     operation card, otherwise it is a block of the initial stack;
  4. the glyph's aspect ratio separates a wide minus (remove) from a square-ish
     plus (add); a card's fill colour is classified to the nearest palette
     anchor, and so is every block's;
  5. blocks are ordered bottom to top by their drawn y, cards left to right by
     their drawn x, and the operations are replayed over the initial stack.

Returns ``"<top colour>|<final height>"`` or ``"abstain"`` when the evidence is
not decidable (no blocks, an empty stack mid-replay, unreadable glyphs, or a
fill colour that is not decisively nearest one anchor).
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "stack_op_replay-oracle-0.1.0"

# The public palette of the question. Only anchor identities are used; no scene
# parameters, coordinates, or renderer constants are read.
_ANCHORS: dict[str, tuple[int, int, int]] = {
    "red": (220, 30, 30),
    "blue": (30, 60, 220),
    "green": (20, 150, 60),
    "orange": (245, 150, 20),
    "purple": (140, 40, 180),
    "gray": (130, 130, 130),
}

_WHITE_MIN = 235   # a pixel is background/glyph white if every channel exceeds this
_BLACK_MAX = 70    # a pixel is outline black if every channel is below this
_MIN_AREA = 300    # ignore speckle
_GLYPH_MIN = 25    # interior white pixels needed to call a component a card
_ABSTAIN_CLEARANCE = 2.0


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """4-connected components of a boolean mask, as arrays of flat indices."""
    h, w = mask.shape
    flat = mask.reshape(-1)
    seen = np.zeros(flat.shape, dtype=bool)
    out = []
    for start in np.flatnonzero(flat):
        if seen[start]:
            continue
        stack = [int(start)]
        seen[start] = True
        comp = []
        while stack:
            p = stack.pop()
            comp.append(p)
            y, x = divmod(p, w)
            if x > 0 and flat[p - 1] and not seen[p - 1]:
                seen[p - 1] = True
                stack.append(p - 1)
            if x + 1 < w and flat[p + 1] and not seen[p + 1]:
                seen[p + 1] = True
                stack.append(p + 1)
            if y > 0 and flat[p - w] and not seen[p - w]:
                seen[p - w] = True
                stack.append(p - w)
            if y + 1 < h and flat[p + w] and not seen[p + w]:
                seen[p + w] = True
                stack.append(p + w)
        if len(comp) >= _MIN_AREA:
            out.append(np.array(comp, dtype=np.int64))
    return out


def _classify(color: np.ndarray) -> tuple[str, float]:
    """Nearest palette anchor and half the gap to the runner-up anchor."""
    ds = sorted((math.dist(color, v), k) for k, v in _ANCHORS.items())
    return ds[0][1], (ds[1][0] - ds[0][0]) / 2.0


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    h, w, _ = arr.shape
    lo = arr.min(axis=2)
    hi = arr.max(axis=2)
    ink = lo < _WHITE_MIN
    chromatic = ink & (hi >= _BLACK_MAX)
    white = lo > _WHITE_MIN

    blocks: list[tuple[int, str]] = []          # (y centre, colour name)
    cards: list[tuple[int, str, str]] = []      # (x, kind, colour name)

    for comp in _components(ink):
        ys, xs = np.divmod(comp, w)
        y0, y1, x0, x1 = int(ys.min()), int(ys.max()), int(xs.min()), int(xs.max())
        chrom = chromatic.reshape(-1)[comp]
        if chrom.sum() < _MIN_AREA // 3:
            continue  # pure-black furniture: bin walls, order arrow
        pix = arr.reshape(-1, 3)[comp[chrom]]
        fill = np.median(pix, axis=0)
        name, clearance = _classify(fill)
        if clearance < _ABSTAIN_CLEARANCE:
            return "abstain"

        # interior white pixels inside the bounding box => operation glyph
        inset = 5
        gy0, gy1 = min(y0 + inset, y1), max(y1 - inset, y0)
        gx0, gx1 = min(x0 + inset, x1), max(x1 - inset, x0)
        sub = white[gy0 : gy1 + 1, gx0 : gx1 + 1]
        if int(sub.sum()) >= _GLYPH_MIN:
            gys, gxs = np.nonzero(sub)
            gh = int(gys.max() - gys.min()) + 1
            gw = int(gxs.max() - gxs.min()) + 1
            if gh <= 0 or gw <= 0:
                return "abstain"
            ratio = gw / gh
            if ratio > 2.0:
                kind = "pop"
            elif ratio < 1.6:
                kind = "push"
            else:
                return "abstain"  # glyph is not decisively a plus or a minus
            cards.append((x0, kind, name))
        else:
            blocks.append(((y0 + y1) // 2, name))

    if not blocks or not cards:
        return "abstain"

    stack = [name for _, name in sorted(blocks, key=lambda b: -b[0])]
    for _, kind, name in sorted(cards, key=lambda c: c[0]):
        if kind == "push":
            if name == "gray":
                return "abstain"
            stack.append(name)
        else:
            if not stack:
                return "abstain"
            stack.pop()
    if not stack:
        return "abstain"
    return f"{stack[-1]}|{len(stack)}"
