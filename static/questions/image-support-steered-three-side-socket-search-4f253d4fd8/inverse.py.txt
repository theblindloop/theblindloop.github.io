"""Independent image-only inverse arm for three-sided socket search."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_CYAN = np.array((26, 164, 188), dtype=np.int16)
_AMBER = np.array((231, 158, 38), dtype=np.int16)
_LABELS = ("upper-left", "upper-right", "lower-left", "lower-right")


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    """Return (x0,y0,x1,y1,area) for 4-connected pixel components."""
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for sy, sx in zip(*np.nonzero(mask), strict=True):
        if seen[sy, sx]:
            continue
        queue = deque([(int(sy), int(sx))])
        seen[sy, sx] = True
        xs, ys = [], []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for dy, dx in ((-1, 0), (0, 1), (1, 0), (0, -1)):
                yy, xx = y + dy, x + dx
                if 0 <= yy < h and 0 <= xx < w and mask[yy, xx] and not seen[yy, xx]:
                    seen[yy, xx] = True
                    queue.append((yy, xx))
        output.append((min(xs), min(ys), max(xs) + 1, max(ys) + 1, len(xs)))
    return output


def _edge_neighbor_score(cyan: np.ndarray, box: tuple[int, int, int, int, int]) -> int:
    x0, y0, x1, y1, area = box
    cw, ch = x1 - x0, y1 - y0
    if not (12 <= cw <= 20 and 12 <= ch <= 20 and area >= 0.85 * cw * ch):
        return -1
    # Ignore the outermost pixel row when measuring the adjacent filled square;
    # this makes the test insensitive to edge raster conventions.
    inset = 2
    windows = (
        cyan[max(0, y0 - ch):y0, x0 + inset:x1 - inset],
        cyan[y0 + inset:y1 - inset, x1:min(cyan.shape[1], x1 + cw)],
        cyan[y1:min(cyan.shape[0], y1 + ch), x0 + inset:x1 - inset],
        cyan[y0 + inset:y1 - inset, max(0, x0 - cw):x0],
    )
    score = 0
    for window in windows:
        if window.size and float(window.mean()) >= 0.82:
            score += 1
    return score


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    cyan = np.max(np.abs(arr - _CYAN), axis=2) <= 12
    amber = np.max(np.abs(arr - _AMBER), axis=2) <= 12
    h, w = amber.shape
    scored: list[tuple[int, int]] = []
    for box in _components(amber):
        score = _edge_neighbor_score(cyan, box)
        if score < 0:
            continue
        x0, y0, x1, y1, _ = box
        cx, cy = (x0 + x1) / 2.0, (y0 + y1) / 2.0
        quadrant = (1 if cx >= w / 2 else 0) + (2 if cy >= h / 2 else 0)
        scored.append((quadrant, score))
    winners = [quadrant for quadrant, score in scored if score == 3]
    if len(winners) != 1:
        return "abstain"
    return _LABELS[winners[0]]
