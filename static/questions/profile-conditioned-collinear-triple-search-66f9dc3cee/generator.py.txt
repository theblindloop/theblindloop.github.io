"""Renderer and analytic checks for the collinear-triple search world.

The visible object is a four-panel array.  Every panel has six equal blue
markers; exactly one panel is sampled with a three-marker straight run.  The
analytic code measures the same declared quantity for every panel, rather than
using a stored target label.
"""

from __future__ import annotations

import itertools
import math
import random
from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "collinear-triple-search-renderer-0.1.0"
CANVAS = (960, 720)
PANEL_BOXES = (
    (40, 70, 450, 330),
    (510, 70, 920, 330),
    (40, 390, 450, 650),
    (510, 390, 920, 650),
)
PANEL_LABELS = ("A", "B", "C", "D")
PANEL_FILL = (246, 249, 252)
PANEL_BORDER = (178, 190, 201)
INK = (31, 86, 135)
TEXT = (25, 39, 53)
BACKGROUND = (255, 255, 255)
COLLINEAR_TOLERANCE = 3.0
MIN_RUNNER_UP_GAP = 8.0
NON_TARGET_RESIDUAL_FLOOR = 10.0
TARGET_OTHER_RESIDUAL_FLOOR = 4.0


def _triple_residual(points: list[list[float]] | tuple[tuple[float, float], ...]) -> float:
    """Maximum perpendicular residual to the least-squares line of three points."""

    mx = sum(float(p[0]) for p in points) / 3.0
    my = sum(float(p[1]) for p in points) / 3.0
    xx = yy = xy = 0.0
    for px, py in points:
        dx, dy = float(px) - mx, float(py) - my
        xx += dx * dx
        yy += dy * dy
        xy += dx * dy
    angle = 0.5 * math.atan2(2.0 * xy, xx - yy)
    normal = (-math.sin(angle), math.cos(angle))
    return max(
        abs((float(px) - mx) * normal[0] + (float(py) - my) * normal[1])
        for px, py in points
    )


def _panel_residuals(points: list[list[float]]) -> list[float]:
    return [
        _triple_residual([points[i], points[j], points[k]])
        for i, j, k in itertools.combinations(range(len(points)), 3)
    ]


def _panel_scores(panels: list[list[list[float]]]) -> list[float]:
    return [min(_panel_residuals(panel)) for panel in panels]


def _winner_and_gap(panels: list[list[list[float]]]) -> tuple[int, float, float]:
    scores = _panel_scores(panels)
    order = sorted(range(len(scores)), key=lambda index: (scores[index], index))
    return order[0], scores[order[0]], scores[order[1]] - scores[order[0]]


def _safe_bounds(box: tuple[int, int, int, int], radius: float) -> tuple[float, float, float, float]:
    x0, y0, x1, y1 = box
    return x0 + 58.0 + radius, x1 - 28.0 - radius, y0 + 76.0 + radius, y1 - 28.0 - radius


def _far_enough(point: list[float], others: list[list[float]], minimum: float) -> bool:
    return all(math.dist(point, other) >= minimum for other in others)


def _sample_panel(
    rng: random.Random,
    box: tuple[int, int, int, int],
    radius: float,
    target: bool,
) -> list[list[float]]:
    xmin, xmax, ymin, ymax = _safe_bounds(box, radius)
    separation = max(38.0, 3.2 * radius)

    for _ in range(12000):
        if target:
            angle = rng.uniform(0.0, math.pi)
            ux, uy = math.cos(angle), math.sin(angle)
            center = [rng.uniform(xmin + 76.0, xmax - 76.0), rng.uniform(ymin + 42.0, ymax - 42.0)]
            d_left = rng.uniform(55.0, 78.0)
            d_right = rng.uniform(55.0, 78.0)
            first = [center[0] - d_left * ux, center[1] - d_left * uy]
            middle = [center[0], center[1]]
            third = [center[0] + d_right * ux, center[1] + d_right * uy]
            points = [first, middle, third]
            if not all(xmin <= px <= xmax and ymin <= py <= ymax for px, py in points):
                continue
            if not all(_far_enough(point, points[:index], separation) for index, point in enumerate(points)):
                continue
        else:
            points = []

        fill_failed = False
        while len(points) < 6:
            placed = False
            for _candidate_attempt in range(1200):
                candidate = [rng.uniform(xmin, xmax), rng.uniform(ymin, ymax)]
                if _far_enough(candidate, points, separation):
                    points.append(candidate)
                    placed = True
                    break
            if not placed:
                fill_failed = True
                break
        if fill_failed:
            continue

        residuals = sorted(_panel_residuals(points))
        if target:
            if residuals[0] <= 0.001 and residuals[1] >= TARGET_OTHER_RESIDUAL_FLOOR:
                return points
        elif residuals[0] >= NON_TARGET_RESIDUAL_FLOOR:
            return points
    raise RuntimeError("could not sample a well-separated marker panel")


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; the seed modulo four selects the answer panel."""

    rng = random.Random(int(seed))
    target_panel = int(seed) % 4
    radius = rng.choice((11.0, 12.0, 13.0))
    panels = [
        _sample_panel(rng, box, radius, index == target_panel)
        for index, box in enumerate(PANEL_BOXES)
    ]
    scene = {
        "width": CANVAS[0],
        "height": CANVAS[1],
        "panels": panels,
        "marker_radius": radius,
        "ink": INK,
        "panel_fill": PANEL_FILL,
        "panel_border": PANEL_BORDER,
        "text": TEXT,
        "background": BACKGROUND,
    }
    winner, best, gap = _winner_and_gap(panels)
    if winner != target_panel or best > COLLINEAR_TOLERANCE or gap < MIN_RUNNER_UP_GAP:
        raise RuntimeError("sampler produced an invalid answerable scene")
    return scene


def _font(size: int) -> ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    """Render only the public scene fields into a deterministic RGB PNG."""

    supersample = 3
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width * supersample, height * supersample), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    panel_fill = tuple(scene["panel_fill"])
    panel_border = tuple(scene["panel_border"])
    ink = tuple(scene["ink"])
    text = tuple(scene["text"])
    radius = float(scene["marker_radius"])
    font = _font(25 * supersample)

    for index, (x0, y0, x1, y1) in enumerate(PANEL_BOXES):
        scaled_box = (x0 * supersample, y0 * supersample, x1 * supersample, y1 * supersample)
        draw.rounded_rectangle(
            scaled_box,
            radius=18 * supersample,
            fill=panel_fill,
            outline=panel_border,
            width=2 * supersample,
        )
        draw.text(
            ((x0 + 18) * supersample, (y0 + 13) * supersample),
            PANEL_LABELS[index],
            fill=text,
            font=font,
        )
        for px, py in scene["panels"][index]:
            draw.ellipse(
                (
                    (float(px) - radius) * supersample,
                    (float(py) - radius) * supersample,
                    (float(px) + radius) * supersample,
                    (float(py) + radius) * supersample,
                ),
                fill=ink,
            )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """Select the panel with the smallest best-triple residual."""

    winner, _best, _gap = _winner_and_gap(scene["panels"])
    return PANEL_LABELS[winner]


def margin(scene: dict) -> float:
    """Positive runner-up residual gap for the panel selection."""

    _winner, _best, gap = _winner_and_gap(scene["panels"])
    return float(gap)


def is_quarantined(scene: dict) -> bool:
    """Quarantine scenes too close to the visible residual boundary."""

    _winner, best, gap = _winner_and_gap(scene["panels"])
    return bool(best > COLLINEAR_TOLERANCE or gap < MIN_RUNNER_UP_GAP)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Point-list order is a render-invariant latent alias."""

    twin = dict(scene)
    twin["panels"] = [list(reversed([list(point) for point in panel])) for panel in scene["panels"]]
    return [("reverse_marker_enumeration", twin)]
