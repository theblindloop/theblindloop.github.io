"""Analytic sampler and Pillow renderer for visible-key checkpoint chains."""

from __future__ import annotations

import copy
import itertools
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 720
HEIGHT = 680
TERMINAL_COLORS = ("red", "blue", "orange")
SHAPES = ("circle", "square", "triangle")

PALETTES = (
    {
        "background": [247, 248, 245],
        "panel": [207, 212, 218],
        "link": [35, 39, 47],
        "green": [39, 166, 92],
        "purple": [132, 86, 204],
        "cyan": [27, 164, 181],
        "red": [222, 68, 75],
        "blue": [52, 111, 216],
        "orange": [235, 140, 38],
    },
    {
        "background": [250, 247, 242],
        "panel": [214, 207, 201],
        "link": [27, 33, 41],
        "green": [31, 145, 88],
        "purple": [151, 76, 187],
        "cyan": [18, 148, 169],
        "red": [195, 51, 68],
        "blue": [43, 127, 195],
        "orange": [218, 119, 35],
    },
)

_Y_PATTERNS = (
    (0, -35, 25, -24, 37, -17, 2),
    (2, 31, -29, 22, -37, 18, -1),
    (-3, -24, 38, 8, -34, 29, 1),
    (1, 26, -38, -5, 34, -27, 0),
)

_TOKEN_MULTISET = ("circle", "circle", "square", "square", "triangle")
_UNIQUE_SEQUENCES = tuple(sorted(set(itertools.permutations(_TOKEN_MULTISET))))


def _route_points(rng: random.Random, slot: int) -> list[list[int]]:
    xs = [58, 157, 257, 360, 463, 563, 662]
    if rng.random() < 0.5:
        xs.reverse()
    pattern = list(rng.choice(_Y_PATTERNS))
    if rng.random() < 0.5:
        pattern = [-value for value in pattern]
    base_y = 200 + 190 * slot
    points: list[list[int]] = []
    for index, (x, offset) in enumerate(zip(xs, pattern)):
        x_jitter = 0 if index in (0, 6) else rng.randint(-5, 5)
        y_jitter = rng.randint(-4, 4)
        points.append([x + x_jitter, base_y + offset + y_jitter])
    return points


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one balanced, JSON-serializable scene from an integer seed."""

    seed = int(seed)
    rng = random.Random(seed ^ 0x6B3A91D2)
    target_color = TERMINAL_COLORS[seed % len(TERMINAL_COLORS)]
    colors = list(TERMINAL_COLORS)
    rng.shuffle(colors)
    key_sequence = list(rng.choice(_UNIQUE_SEQUENCES))
    distractors = [
        list(sequence)
        for sequence in _UNIQUE_SEQUENCES
        if list(sequence) != key_sequence
        and sum(a == b for a, b in zip(sequence, key_sequence)) <= 3
    ]
    rng.shuffle(distractors)
    routes = []
    distractor_index = 0
    for slot, terminal_color in enumerate(colors):
        if terminal_color == target_color:
            sequence = list(key_sequence)
        else:
            sequence = list(distractors[distractor_index])
            distractor_index += 1
        routes.append(
            {
                "slot": slot,
                "terminal_color": terminal_color,
                "sequence": sequence,
                "points": _route_points(rng, slot),
            }
        )
    rng.shuffle(routes)
    palette = copy.deepcopy(PALETTES[rng.randrange(len(PALETTES))])
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(palette["background"]),
        "palette": palette,
        "key_sequence": key_sequence,
        "routes": routes,
    }


def _scaled_point(point: list[int], scale: int) -> tuple[int, int]:
    return int(point[0] * scale), int(point[1] * scale)


def _draw_disk(draw: ImageDraw.ImageDraw, center: list[int], radius: int, color, scale: int) -> None:
    cx, cy = _scaled_point(center, scale)
    r = radius * scale
    draw.ellipse((cx - r, cy - r, cx + r, cy + r), fill=tuple(color))


def _draw_checkpoint(
    draw: ImageDraw.ImageDraw,
    center: list[int],
    shape: str,
    color,
    scale: int,
) -> None:
    cx, cy = _scaled_point(center, scale)
    if shape == "circle":
        r = 15 * scale
        draw.ellipse((cx - r, cy - r, cx + r, cy + r), fill=tuple(color))
    elif shape == "square":
        r = 14 * scale
        draw.rectangle((cx - r, cy - r, cx + r, cy + r), fill=tuple(color))
    elif shape == "triangle":
        draw.polygon(
            ((cx, cy - 17 * scale), (cx - 17 * scale, cy + 14 * scale), (cx + 17 * scale, cy + 14 * scale)),
            fill=tuple(color),
        )
    else:
        raise ValueError(f"unknown checkpoint shape {shape!r}")


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only the supplied latent scene; no analytic answer is consulted."""

    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(scene["background"])
    palette = scene["palette"]
    routes = sorted(scene["routes"], key=lambda item: int(item["slot"]))
    scale = 4
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)

    draw.rounded_rectangle(
        (142 * scale, 12 * scale, 578 * scale, 98 * scale),
        radius=14 * scale,
        outline=tuple(palette["panel"]),
        width=2 * scale,
    )
    key_centers = [[230 + 65 * index, 55] for index in range(5)]
    for center, shape in zip(key_centers, scene["key_sequence"]):
        _draw_checkpoint(draw, center, shape, palette["cyan"], scale)

    for slot in range(3):
        top = (110 + 190 * slot) * scale
        bottom = (290 + 190 * slot) * scale
        draw.rounded_rectangle(
            (14 * scale, top, (width - 14) * scale, bottom),
            radius=16 * scale,
            outline=tuple(palette["panel"]),
            width=2 * scale,
        )

    for route in routes:
        points = [_scaled_point(point, scale) for point in route["points"]]
        draw.line(points, fill=tuple(palette["link"]), width=6 * scale, joint="curve")

    for route in routes:
        points = route["points"]
        _draw_disk(draw, points[0], 16, palette["green"], scale)
        for center, shape in zip(points[1:-1], route["sequence"]):
            _draw_checkpoint(draw, center, shape, palette["purple"], scale)
        _draw_disk(draw, points[-1], 16, palette[route["terminal_color"]], scale)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    key_sequence = list(scene["key_sequence"])
    winners = []
    for route in scene["routes"]:
        sequence = list(route["sequence"])
        if sequence == key_sequence:
            winners.append(str(route["terminal_color"]))
    if len(winners) != 1:
        raise ValueError(f"scene has {len(winners)} routes matching the key")
    return winners[0]


def margin(scene: dict[str, Any]) -> float:
    key_sequence = list(scene["key_sequence"])
    scores = []
    for route in scene["routes"]:
        sequence = list(route["sequence"])
        score = sum(left == right for left, right in zip(sequence, key_sequence))
        scores.append(score)
    scores.sort(reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        analytic_gold(scene)
    except ValueError:
        return True
    return margin(scene) < 2.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    twin = copy.deepcopy(scene)
    twin["routes"] = list(reversed(twin["routes"]))
    return [("reverse_route_record_order", twin)]
