"""Independent pixel-only inverse for first-entry timelines."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int = 1) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out: list[tuple[int, int, int, int, int]] = []
    for y, x in zip(*np.nonzero(mask & ~seen)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        points = []
        while queue:
            py, px = queue.popleft()
            points.append((py, px))
            for ny, nx in ((py - 1, px), (py + 1, px), (py, px - 1), (py, px + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            ys = [point[0] for point in points]
            xs = [point[1] for point in points]
            out.append((min(xs), min(ys), max(xs), max(ys), len(points)))
    return out


def _dot_count(array: np.ndarray, box: tuple[int, int, int, int, int]) -> int:
    x0, y0, x1, y1, _ = box
    crop = array[y0 : y1 + 1, x0 : x1 + 1]
    white = np.all(crop >= 248, axis=2)
    dots = []
    for component in _components(white, minimum=10):
        cx0, cy0, cx1, cy1, area = component
        if cx0 > 2 and cy0 > 2 and cx1 < crop.shape[1] - 3 and cy1 < crop.shape[0] - 3 and area < 100:
            dots.append(component)
    return len(dots)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    spread = array.max(axis=2) - array.min(axis=2)
    saturated = (spread > 45) & (array.max(axis=2) < 245) & (array.min(axis=2) < 210)
    cards = []
    for box in _components(saturated, minimum=500):
        x0, y0, x1, y1, area = box
        width, height = x1 - x0 + 1, y1 - y0 + 1
        if 28 <= width <= 46 and 25 <= height <= 42 and 500 <= area <= 1800:
            dots = _dot_count(array, box)
            if 1 <= dots <= 4:
                cards.append((box, dots))
    if len(cards) != 20:
        raise ValueError(f"abstain: expected 20 token cards, found {len(cards)}")

    shade = (
        (array[:, :, 0] >= 205) & (array[:, :, 0] <= 238)
        & (array[:, :, 1] >= 222) & (array[:, :, 1] <= 248)
        & (array[:, :, 2] >= 235) & (array[:, :, 2] <= 255)
        & ((array[:, :, 2] - array[:, :, 0]) >= 10)
    )
    zones = []
    for box in _components(shade, minimum=10000):
        x0, y0, x1, y1, area = box
        if x1 - x0 > 40 and y1 - y0 > 250:
            zones.append(box)
    zones.sort(key=lambda item: item[0])
    if len(zones) != 5:
        raise ValueError(f"abstain: expected 5 shaded zones, found {len(zones)}")

    states = [[None for _ in range(4)] for _ in range(5)]
    zone_centers = [(box[0] + box[2]) / 2 for box in zones]
    typical_width = float(np.median([box[2] - box[0] + 1 for box in zones]))
    for box, dots in cards:
        cx = (box[0] + box[2]) / 2
        panel = min(range(5), key=lambda index: abs(cx - zone_centers[index]))
        if abs(cx - zone_centers[panel]) > typical_width * 1.7:
            raise ValueError("abstain: token cannot be assigned to a panel")
        inside = zones[panel][0] <= cx <= zones[panel][2]
        if states[panel][dots - 1] is not None:
            raise ValueError("abstain: duplicate token identity in a panel")
        states[panel][dots - 1] = inside
    if any(value is None for panel in states for value in panel):
        raise ValueError("abstain: incomplete panel state")

    entries = []
    for token in range(4):
        hits = [step for step in range(1, 5) if states[step - 1][token] is False and states[step][token] is True]
        if hits:
            entries.append((min(hits), token))
    if not entries:
        raise ValueError("abstain: no entry transition")
    earliest = min(step for step, _ in entries)
    winners = [token for step, token in entries if step == earliest]
    if len(winners) != 1:
        raise ValueError("abstain: tied earliest entries")
    return str(winners[0] + 1)
