"""Deterministic Pillow renderer and analytic arm."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1000, 520
PANEL_COUNT, TOKEN_COUNT = 5, 4
PANEL_W, PANEL_H = 180, 390
PANEL_TOP = 76
PANEL_GAP = 14
LEFT = 26
ZONE_W = 70
CARD_W, CARD_H = 42, 40
ROW_Y = (139, 215, 291, 367)

PALETTES = (
    ((214, 73, 68), (37, 146, 99), (63, 111, 204), (221, 151, 45)),
    ((143, 83, 193), (22, 152, 168), (229, 91, 125), (92, 139, 51)),
    ((34, 133, 210), (230, 116, 37), (116, 82, 176), (25, 158, 111)),
)


def _entry_steps(states: list[list[bool]]) -> list[int | None]:
    steps: list[int | None] = []
    for token in range(TOKEN_COUNT):
        hits = [
            step
            for step in range(1, PANEL_COUNT)
            if not states[step - 1][token] and states[step][token]
        ]
        steps.append(min(hits) if hits else None)
    return steps


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    wanted = int(seed) % TOKEN_COUNT
    for _ in range(100000):
        states = [[bool(rng.getrandbits(1)) for _ in range(TOKEN_COUNT)] for _ in range(PANEL_COUNT)]
        if any(sum(panel) not in (1, 2, 3) for panel in states):
            continue
        steps = _entry_steps(states)
        finite = [(step, token) for token, step in enumerate(steps) if step is not None]
        if len(finite) < 2:
            continue
        earliest = min(step for step, _ in finite)
        winners = [token for step, token in finite if step == earliest]
        if winners != [wanted]:
            continue
        row_orders = []
        for _panel in range(PANEL_COUNT):
            order = list(range(TOKEN_COUNT))
            rng.shuffle(order)
            row_orders.append(order)
        colors = list(PALETTES[rng.randrange(len(PALETTES))])
        rng.shuffle(colors)
        return {
            "states": states,
            "row_orders": row_orders,
            "palette": [list(color) for color in colors],
        }
    raise RuntimeError("could not sample a valid timeline")


def analytic_gold(scene: dict) -> str:
    states = scene["states"]
    steps = _entry_steps(states)
    finite = [(step, token) for token, step in enumerate(steps) if step is not None]
    if not finite:
        raise ValueError("timeline has no entry")
    earliest = min(step for step, _ in finite)
    winners = [token for step, token in finite if step == earliest]
    if len(winners) != 1:
        raise ValueError("timeline has tied earliest entries")
    return str(winners[0] + 1)


def margin(scene: dict) -> float:
    steps = sorted(step for step in _entry_steps(scene["states"]) if step is not None)
    if len(steps) < 2:
        return 0.0
    return float(steps[1] - steps[0])


def is_quarantined(scene: dict) -> bool:
    try:
        analytic_gold(scene)
    except ValueError:
        return True
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _draw_token(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], color: tuple[int, int, int], dots: int) -> None:
    x0, y0, x1, y1 = box
    draw.rounded_rectangle(box, radius=7, fill=color, outline=(24, 29, 36), width=3)
    radius = 3
    spacing = 8
    total = (dots - 1) * spacing
    start = (x0 + x1) // 2 - total // 2
    cy = (y0 + y1) // 2
    for index in range(dots):
        cx = start + index * spacing
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=(255, 255, 255))


def render(scene: dict) -> Image.Image:
    states = scene["states"]
    row_orders = scene["row_orders"]
    palette = [tuple(value) for value in scene["palette"]]
    image = Image.new("RGB", (WIDTH, HEIGHT), (249, 248, 244))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    draw.text((LEFT, 24), "FIRST ENTRY TIMELINE", fill=(24, 29, 36), font=font)
    draw.text((LEFT, 45), "Read panels left to right. Dots identify tokens.", fill=(63, 70, 80), font=font)
    for panel in range(PANEL_COUNT):
        x0 = LEFT + panel * (PANEL_W + PANEL_GAP)
        x1 = x0 + PANEL_W
        zone_x = x1 - ZONE_W
        draw.rectangle((x0, PANEL_TOP, x1, PANEL_TOP + PANEL_H), fill=(255, 255, 255), outline=(31, 37, 45), width=3)
        draw.rectangle((zone_x, PANEL_TOP + 3, x1 - 3, PANEL_TOP + PANEL_H - 3), fill=(222, 237, 246))
        draw.line((zone_x, PANEL_TOP + 3, zone_x, PANEL_TOP + PANEL_H - 3), fill=(31, 93, 132), width=3)
        draw.text((x0 + 8, PANEL_TOP + 9), f"T{panel}", fill=(31, 37, 45), font=font)
        draw.text((zone_x + 25, PANEL_TOP + 9), "IN", fill=(31, 93, 132), font=font)
        if panel < PANEL_COUNT - 1:
            arrow_y = PANEL_TOP + PANEL_H // 2
            draw.line((x1 + 3, arrow_y, x1 + PANEL_GAP - 3, arrow_y), fill=(31, 37, 45), width=2)
            draw.polygon(((x1 + PANEL_GAP - 3, arrow_y), (x1 + PANEL_GAP - 8, arrow_y - 4), (x1 + PANEL_GAP - 8, arrow_y + 4)), fill=(31, 37, 45))
        for row, token in enumerate(row_orders[panel]):
            inside = bool(states[panel][token])
            cx = zone_x + ZONE_W // 2 if inside else x0 + (zone_x - x0) // 2
            cy = ROW_Y[row]
            box = (cx - CARD_W // 2, cy - CARD_H // 2, cx + CARD_W // 2, cy + CARD_H // 2)
            _draw_token(draw, box, palette[token], token + 1)
    draw.text((LEFT, 486), "OUTSIDE", fill=(70, 75, 82), font=font)
    draw.rectangle((96, 483, 112, 498), fill=(222, 237, 246), outline=(31, 93, 132))
    draw.text((118, 486), "SHADED = IN", fill=(31, 93, 132), font=font)
    return image
