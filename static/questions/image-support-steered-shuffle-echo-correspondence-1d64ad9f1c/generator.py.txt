"""Analytic sampler and Pillow renderer for Shuffle Echo Correspondence."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

CANVAS = 1200
GRID_CELL = CANVAS // 8
SHAPES = ("circle", "square", "triangle")
ANSWER_COLORS = ("amber", "cyan", "violet")
RGB = {
    "amber": (242, 169, 0),
    "cyan": (0, 166, 214),
    "violet": (139, 92, 246),
}
SLOT_OFFSETS = (-22, 0, 22)
FEATURE_CODES = {"triangle": 0.52, "circle": 0.79, "square": 1.0}


def _remaining(items: Iterable[str], chosen: str) -> list[str]:
    return [item for item in items if item != chosen]


def sample_scene(seed: int) -> dict:
    """Sample a balanced, translated scene from one integer seed.

    Consecutive seeds cross answer color with source slot. Shape and layout are
    varied independently enough that none is a reliable answer shortcut.
    """

    rng = random.Random(int(seed) * 104729 + 8191)
    answer = ANSWER_COLORS[int(seed) % 3]
    source_slot = (int(seed) // 3) % 3
    target_shape = SHAPES[(int(seed) // 9) % 3]

    other_shapes = _remaining(SHAPES, target_shape)
    rng.shuffle(other_shapes)
    before: list[str] = ["", "", ""]
    before[source_slot] = target_shape
    for index, shape in zip((i for i in range(3) if i != source_slot), other_shapes):
        before[index] = shape

    after_tail = _remaining(SHAPES, target_shape)
    rng.shuffle(after_tail)
    after = [target_shape, *after_tail]

    other_colors = _remaining(ANSWER_COLORS, answer)
    rng.shuffle(other_colors)
    query: list[str] = ["", "", ""]
    query[source_slot] = answer
    for index, color in zip((i for i in range(3) if i != source_slot), other_colors):
        query[index] = color

    # Panel centers occupy three non-adjacent cells in a 2-by-2 local span.
    # The base is randomized, while keeping all ink away from the canvas edge.
    return {
        "before_order": before,
        "after_order": after,
        "query_colors": query,
        "base_cell_x": rng.choice((1, 2, 3, 4)),
        "base_cell_y": rng.choice((1, 2, 3, 4)),
    }


def _panel_centers(scene: dict) -> tuple[tuple[int, int], tuple[int, int], tuple[int, int]]:
    x = (int(scene["base_cell_x"]) * GRID_CELL) + GRID_CELL // 2
    y = (int(scene["base_cell_y"]) * GRID_CELL) + GRID_CELL // 2
    return (x, y), (x + 2 * GRID_CELL, y), (x, y + 2 * GRID_CELL)


def _draw_shape(draw: ImageDraw.ImageDraw, shape: str, cx: int, cy: int) -> None:
    if shape == "circle":
        draw.ellipse((cx - 10, cy - 10, cx + 10, cy + 10), fill=(20, 24, 31))
    elif shape == "square":
        draw.rectangle((cx - 9, cy - 9, cx + 9, cy + 9), fill=(20, 24, 31))
    elif shape == "triangle":
        draw.polygon(((cx, cy - 11), (cx - 11, cy + 10), (cx + 11, cy + 10)), fill=(20, 24, 31))
    else:
        raise ValueError(f"unknown shape {shape!r}")


def _draw_card(draw: ImageDraw.ImageDraw, center: tuple[int, int], accent: tuple[int, int, int]) -> None:
    cx, cy = center
    draw.rounded_rectangle(
        (cx - 42, cy - 31, cx + 42, cy + 31),
        radius=10,
        fill=(250, 251, 252),
        outline=accent,
        width=3,
    )


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (CANVAS, CANVAS), (255, 255, 255))
    draw = ImageDraw.Draw(image)
    before_center, after_center, query_center = _panel_centers(scene)

    _draw_card(draw, before_center, (118, 151, 178))
    _draw_card(draw, after_center, (118, 151, 178))
    _draw_card(draw, query_center, (201, 158, 82))

    # A pale arrow makes the demonstrated rearrangement visible but is ignored
    # by the inverse arm; it is deliberately nondiagnostic.
    arrow_y = before_center[1]
    arrow_start = before_center[0] + 55
    arrow_end = after_center[0] - 55
    draw.line((arrow_start, arrow_y, arrow_end, arrow_y), fill=(176, 185, 194), width=5)
    draw.polygon(
        ((arrow_end, arrow_y), (arrow_end - 13, arrow_y - 9), (arrow_end - 13, arrow_y + 9)),
        fill=(176, 185, 194),
    )

    for offset, shape in zip(SLOT_OFFSETS, scene["before_order"]):
        _draw_shape(draw, shape, before_center[0] + offset, before_center[1])
    for offset, shape in zip(SLOT_OFFSETS, scene["after_order"]):
        _draw_shape(draw, shape, after_center[0] + offset, after_center[1])
    for offset, color in zip(SLOT_OFFSETS, scene["query_colors"]):
        cx, cy = query_center[0] + offset, query_center[1]
        draw.ellipse((cx - 10, cy - 10, cx + 10, cy + 10), fill=RGB[color])
    return image


def analytic_gold(scene: dict) -> str:
    target_shape = scene["after_order"][0]
    source_slot = scene["before_order"].index(target_shape)
    return str(scene["query_colors"][source_slot])


def margin(scene: dict) -> float:
    """Composition-min margin; shape separation is the bottleneck stage.

    Slot uniqueness and palette uniqueness each have a discrete gap of one,
    while the visible fill-ratio classifier has the smaller runner-up gap.
    """

    target = scene["after_order"][0]
    target_code = FEATURE_CODES[target]
    alternatives = [abs(target_code - FEATURE_CODES[item]) for item in SHAPES if item != target]
    return float(min(alternatives))


def is_quarantined(scene: dict) -> bool:
    try:
        before = tuple(scene["before_order"])
        after = tuple(scene["after_order"])
        colors = tuple(scene["query_colors"])
    except (KeyError, TypeError):
        return True
    return set(before) != set(SHAPES) or set(after) != set(SHAPES) or set(colors) != set(ANSWER_COLORS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
