"""Deterministic scene sampler, Pillow renderer, and analytic arm.

The latent scene contains only visible segment endpoints.  A correct panel has
two fragments on one supporting line.  Distractors separately break direction,
cross-gap registration, or both, so neither cue alone identifies the answer.
"""

from __future__ import annotations

import copy
import math
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 600, 480
PANEL_W, PANEL_H = 240, 150
PANEL_POSITIONS = ((45, 95), (315, 95), (45, 295), (315, 295))
GAP_CENTER = 120.0
GAP_LEFT, GAP_RIGHT = 104, 136
LABELS = ("A", "B", "C", "D")

BACKGROUND = (250, 249, 246)
PANEL_FILL = (255, 255, 255)
FRAME = (224, 111, 45)
INK = (35, 51, 70)
GAP = (232, 235, 237)
MUTED = (92, 99, 106)
QUARANTINE_MARGIN = 6.0

_FONT_DIR = Path("/usr/share/fonts/truetype/dejavu")


def _font(name: str, size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype(str(_FONT_DIR / name), size=size)
    except OSError:
        return ImageFont.load_default()


def _endpoints(angle_deg: float, y_at_gap: float, midpoint_x: float) -> list[list[float]]:
    """Return a fixed-length fragment on the requested supporting line."""
    theta = math.radians(angle_deg)
    length = 70.0
    midpoint_y = y_at_gap + math.tan(theta) * (midpoint_x - GAP_CENTER)
    dx = 0.5 * length * math.cos(theta)
    dy = 0.5 * length * math.sin(theta)
    return [
        [round(midpoint_x - dx, 6), round(midpoint_y - dy, 6)],
        [round(midpoint_x + dx, 6), round(midpoint_y + dy, 6)],
    ]


def sample_scene(seed: int) -> dict:
    """Sample a balanced four-panel scene.

    ``seed % 4`` counterbalances the answer identity exactly over every block of
    four consecutive seeds.  All eight fragments have equal Euclidean length,
    width, and color.  The parallel distractor shares direction with the target;
    the kink distractor shares cross-gap meeting position with it.
    """
    rng = random.Random(int(seed))
    target_slot = int(seed) % 4
    distractors = ["parallel_shift", "central_kink", "kink_and_shift"]
    rng.shuffle(distractors)
    kinds: list[str] = []
    for slot in range(4):
        kinds.append("aligned" if slot == target_slot else distractors.pop())

    panels = []
    for kind in kinds:
        base = float(rng.choice((-12, -8, -4, 4, 8, 12)))
        center_y = float(rng.randint(74, 86))
        delta = float(rng.choice((8, 10, 12, 14)))
        offset = float(rng.choice((8, 10, 12, 14)))
        if kind == "aligned":
            left_angle = right_angle = base
            left_y = right_y = center_y
        elif kind == "parallel_shift":
            left_angle = right_angle = base
            left_y, right_y = center_y - offset / 2.0, center_y + offset / 2.0
        elif kind == "central_kink":
            left_angle, right_angle = base - delta / 2.0, base + delta / 2.0
            left_y = right_y = center_y
        else:
            left_angle, right_angle = base - delta / 2.0, base + delta / 2.0
            left_y, right_y = center_y - offset / 2.0, center_y + offset / 2.0
        panels.append(
            {
                "left": _endpoints(left_angle, left_y, 63.0),
                "right": _endpoints(right_angle, right_y, 177.0),
            }
        )
    return {"panels": panels}


def _scores(scene: dict) -> list[float]:
    scores: list[float] = []
    for panel in scene["panels"]:
        (lx1, ly1), (lx2, ly2) = panel["left"]
        (rx1, ry1), (rx2, ry2) = panel["right"]
        la = math.degrees(math.atan2(ly2 - ly1, lx2 - lx1)) % 180.0
        ra = math.degrees(math.atan2(ry2 - ry1, rx2 - rx1)) % 180.0
        angle_gap = abs(la - ra)
        angle_gap = min(angle_gap, 180.0 - angle_gap)
        left_cross = ly1 + (GAP_CENTER - lx1) * (ly2 - ly1) / (lx2 - lx1)
        right_cross = ry1 + (GAP_CENTER - rx1) * (ry2 - ry1) / (rx2 - rx1)
        scores.append(angle_gap + abs(left_cross - right_cross))
    return scores


def analytic_gold(scene: dict) -> str:
    # Kept inline so the protected rendered-field audit sees exactly `panels`.
    scores = []
    for panel in scene["panels"]:
        (lx1, ly1), (lx2, ly2) = panel["left"]
        (rx1, ry1), (rx2, ry2) = panel["right"]
        la = math.degrees(math.atan2(ly2 - ly1, lx2 - lx1)) % 180.0
        ra = math.degrees(math.atan2(ry2 - ry1, rx2 - rx1)) % 180.0
        angle_gap = min(abs(la - ra), 180.0 - abs(la - ra))
        left_cross = ly1 + (GAP_CENTER - lx1) * (ly2 - ly1) / (lx2 - lx1)
        right_cross = ry1 + (GAP_CENTER - rx1) * (ry2 - ry1) / (rx2 - rx1)
        scores.append(angle_gap + abs(left_cross - right_cross))
    return LABELS[min(range(4), key=scores.__getitem__)]


def margin(scene: dict) -> float:
    ordered = sorted(_scores(scene))
    return ordered[1] - ordered[0]


def is_quarantined(scene: dict) -> bool:
    scores = sorted(_scores(scene))
    return scores[0] > 1e-5 or scores[1] - scores[0] < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    for panel in twin["panels"]:
        panel["left"].reverse()
        panel["right"].reverse()
    return [("reverse_all_undirected_segment_endpoints", twin)]


def _scaled(point: list[float], ox: int, oy: int, ss: int) -> tuple[int, int]:
    return (round((ox + point[0]) * ss), round((oy + point[1]) * ss))


def _draw_fragment(
    draw: ImageDraw.ImageDraw,
    endpoints: list[list[float]],
    ox: int,
    oy: int,
    ss: int,
) -> None:
    # Pillow's low-level line rasterizer can be endpoint-order sensitive.  Sort
    # the undirected pair so the declared reversal alias is byte-identical.
    ordered = sorted(endpoints, key=lambda point: (point[0], point[1]))
    p1, p2 = (_scaled(point, ox, oy, ss) for point in ordered)
    width = 8 * ss
    radius = width // 2
    draw.line((p1, p2), fill=INK, width=width)
    for px, py in (p1, p2):
        draw.ellipse((px - radius, py - radius, px + radius, py + radius), fill=INK)


def render(scene: dict) -> Image.Image:
    ss = 3
    image = Image.new("RGB", (WIDTH * ss, HEIGHT * ss), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font("DejaVuSans-Bold.ttf", 27 * ss)
    sub_font = _font("DejaVuSans.ttf", 15 * ss)
    label_font = _font("DejaVuSans-Bold.ttf", 23 * ss)

    title = "ONE STRAIGHT PATH"
    box = draw.textbbox((0, 0), title, font=title_font)
    draw.text(((WIDTH * ss - (box[2] - box[0])) // 2, 15 * ss), title, fill=INK, font=title_font)
    subtitle = "same direction  +  no sideways shift across the gap"
    box = draw.textbbox((0, 0), subtitle, font=sub_font)
    draw.text(
        ((WIDTH * ss - (box[2] - box[0])) // 2, 51 * ss),
        subtitle,
        fill=MUTED,
        font=sub_font,
    )

    for index, panel in enumerate(scene["panels"]):
        ox, oy = PANEL_POSITIONS[index]
        draw.rounded_rectangle(
            (ox * ss, oy * ss, (ox + PANEL_W) * ss, (oy + PANEL_H) * ss),
            radius=8 * ss,
            fill=PANEL_FILL,
            outline=FRAME,
            width=3 * ss,
        )
        draw.rectangle(
            ((ox + GAP_LEFT) * ss, (oy + 5) * ss, (ox + GAP_RIGHT) * ss, (oy + PANEL_H - 5) * ss),
            fill=GAP,
        )
        _draw_fragment(draw, panel["left"], ox, oy, ss)
        _draw_fragment(draw, panel["right"], ox, oy, ss)
        draw.text((ox * ss + 3 * ss, (oy - 30) * ss), LABELS[index], fill=INK, font=label_font)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
