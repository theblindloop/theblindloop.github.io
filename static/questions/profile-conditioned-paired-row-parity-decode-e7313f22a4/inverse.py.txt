"""Independent final-PNG-only inverse arm."""

from __future__ import annotations

import numpy as np
from PIL import Image

ROWS, COLS = 16, 10


def _template(parts: tuple[str, ...]) -> np.ndarray:
    atoms = {
        "top": {(1, c) for c in range(2, 8)},
        "mid": {(4, c) for c in range(2, 8)},
        "bottom": {(6, c) for c in range(2, 8)},
        "left": {(r, 1) for r in range(1, 7)},
        "right": {(r, 8) for r in range(1, 7)},
    }
    cells = set().union(*(atoms[p] for p in parts))
    return np.array([[int((r, c) in cells) for c in range(COLS)] for r in range(8)], dtype=np.uint8)


_REFERENCES = {
    "cup": _template(("left", "right", "bottom")),
    "cap": _template(("left", "right", "top")),
    "comb": _template(("top", "left", "mid")),
    "gate": _template(("left", "right", "mid")),
}


def _recover(image: Image.Image) -> tuple[str, float]:
    # int32 prevents overflow in the integer luminance calculation below.
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    # Cyan fiducial pixels are separated from every randomized cell palette.
    cyan = (arr[:, :, 0] < 60) & (arr[:, :, 1] > 135) & (arr[:, :, 1] < 195) & (arr[:, :, 2] > 150)
    ys, xs = np.nonzero(cyan)
    if len(xs) < 1000:
        return "abstain", 0.0
    left, right, top, bottom = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    # The 5-pixel frame surrounds the exact 240 by 384 cell field.
    x0, x1, y0, y1 = left + 5, right - 5, top + 5, bottom - 5
    if abs((x1 - x0 + 1) / COLS - 24) > 1 or abs((y1 - y0 + 1) / ROWS - 24) > 1:
        return "abstain", 0.0
    sampled = np.zeros((ROWS, COLS), dtype=np.uint8)
    for r in range(ROWS):
        for c in range(COLS):
            x = round(x0 + (c + 0.5) * (x1 - x0 + 1) / COLS)
            y = round(y0 + (r + 0.5) * (y1 - y0 + 1) / ROWS)
            red, green, blue = arr[y, x]
            sampled[r, c] = int((299 * red + 587 * green + 114 * blue) < 105000)
    decoded = sampled[0::2] ^ sampled[1::2]
    ranked = sorted((int(np.count_nonzero(decoded != ref)), name) for name, ref in _REFERENCES.items())
    gap = float(ranked[1][0] - ranked[0][0])
    if gap < 6:
        return "abstain", gap
    return ranked[0][1], gap


def decision_from_image(image: Image.Image) -> str:
    return _recover(image)[0]


def decision_with_margin(image: Image.Image) -> tuple[str, float]:
    return _recover(image)
