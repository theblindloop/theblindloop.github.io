"""Deterministic Pillow renderer and analytic arm."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 420, 500
ROWS, COLS, CELL = 16, 10, 24
FRAME = (18, 166, 186)
MARK = (218, 48, 132)

PALETTES = (
    ((28, 31, 38), (248, 246, 238)),
    ((25, 43, 68), (250, 247, 232)),
    ((56, 34, 27), (242, 248, 246)),
    ((48, 29, 61), (247, 244, 250)),
)


def _template(names: Iterable[str]) -> list[list[int]]:
    atoms = {
        "top": {(1, c) for c in range(2, 8)},
        "mid": {(4, c) for c in range(2, 8)},
        "bottom": {(6, c) for c in range(2, 8)},
        "left": {(r, 1) for r in range(1, 7)},
        "right": {(r, 8) for r in range(1, 7)},
    }
    cells = set().union(*(atoms[name] for name in names))
    return [[int((r, c) in cells) for c in range(COLS)] for r in range(8)]


TEMPLATES = {
    "cup": _template(("left", "right", "bottom")),
    "cap": _template(("left", "right", "top")),
    "comb": _template(("top", "left", "mid")),
    "gate": _template(("left", "right", "mid")),
}


def _decode(encoded: list[list[int]]) -> list[list[int]]:
    return [
        [int(encoded[2 * r][c]) ^ int(encoded[2 * r + 1][c]) for c in range(COLS)]
        for r in range(8)
    ]


def _scores(bitmap: list[list[int]]) -> list[tuple[int, str]]:
    return sorted(
        (
            sum(a != b for row_a, row_b in zip(bitmap, template) for a, b in zip(row_a, row_b)),
            name,
        )
        for name, template in TEMPLATES.items()
    )


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    name = tuple(TEMPLATES)[int(seed) % len(TEMPLATES)]
    target = TEMPLATES[name]
    zeros = [(r, c) for r in range(8) for c in range(COLS) if not target[r][c]]
    # Each glyph contains 18 cells. Setting exactly half of the 62 zero positions
    # to 11 makes every source image contain exactly 80 dark cells.
    double_dark = set(rng.sample(zeros, len(zeros) // 2))
    encoded = [[0 for _ in range(COLS)] for _ in range(ROWS)]
    for r in range(8):
        for c in range(COLS):
            if target[r][c]:
                carrier = rng.randrange(2)
                encoded[2 * r][c] = carrier
                encoded[2 * r + 1][c] = carrier ^ 1
            elif (r, c) in double_dark:
                encoded[2 * r][c] = encoded[2 * r + 1][c] = 1
    return {
        "encoded": encoded,
        "offset_x": rng.randint(-12, 12),
        "offset_y": rng.randint(-8, 8),
        "palette_index": rng.randrange(len(PALETTES)),
    }


def render(scene: dict) -> Image.Image:
    encoded = scene["encoded"]
    ox, oy = int(scene["offset_x"]), int(scene["offset_y"])
    dark, light = PALETTES[int(scene["palette_index"])]
    image = Image.new("RGB", (WIDTH, HEIGHT), (238, 240, 236))
    draw = ImageDraw.Draw(image)
    x0 = (WIDTH - COLS * CELL) // 2 + ox
    y0 = 62 + oy
    draw.rectangle((x0 - 5, y0 - 5, x0 + COLS * CELL + 5, y0 + ROWS * CELL + 5), fill=FRAME)
    for r in range(ROWS):
        for c in range(COLS):
            left, top = x0 + c * CELL, y0 + r * CELL
            color = dark if int(encoded[r][c]) else light
            draw.rectangle((left, top, left + CELL - 2, top + CELL - 2), fill=color)
    # A visible top-left marker pins row order and makes the starting pair explicit.
    draw.polygon(((x0 - 18, y0), (x0 - 7, y0 - 8), (x0 - 7, y0 + 8)), fill=MARK)
    draw.text((x0, 24), "PAIR ROWS, THEN COMPRESS", fill=(46, 53, 57))
    return image


def analytic_gold(scene: dict) -> str:
    bitmap = _decode(scene["encoded"])
    return _scores(bitmap)[0][1]


def margin(scene: dict) -> float:
    scores = _scores(_decode(scene["encoded"]))
    return float(scores[1][0] - scores[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
