"""Independent image-only inverse for paired stock orders."""
from __future__ import annotations
import itertools, numpy as np
from PIL import Image
TOKEN_COLORS=((24,137,181),(146,75,163),(228,130,30),(52,153,91)); STAR_COLOR=(238,185,45); RUN_COLOR=(72,79,89)
RECIPE_X=(22,336,650); ORDER_X=(15,202,389,576,763); ROW_Y=(470,545,620); LETTERS="ABCDE"
def _components(mask):
    h,w=mask.shape; remaining=set(int(v) for v in np.flatnonzero(mask)); sizes=[]
    while remaining:
        stack=[remaining.pop()]; size=0
        while stack:
            v=stack.pop(); y,x=divmod(v,w); size+=1
            for n in (v-1,v+1,v-w,v+w):
                if n in remaining:
                    ny,nx=divmod(n,w)
                    if abs(nx-x)+abs(ny-y)==1: remaining.remove(n); stack.append(n)
        if size>=70: sizes.append(size)
    return None if any(not 120<=s<=190 for s in sizes) else len(sizes)
def _count_color(arr,box,color):
    x0,y0,x1,y1=box; crop=arr[y0:y1,x0:x1]; delta=np.max(np.abs(crop.astype(np.int16)-np.asarray(color,dtype=np.int16)),axis=2)
    return _components(delta<=2)
def _counts(arr,box):
    values=tuple(_count_color(arr,box,c) for c in TOKEN_COLORS); return None if any(v is None for v in values) else values
def decision_from_image(image):
    arr=np.asarray(image.convert("RGB"),dtype=np.uint8)
    if arr.shape!=(720,960,3): return "abstain"
    recipes=[]
    for x0 in RECIPE_X:
        costs=_counts(arr,(x0+6,92,x0+152,194)); stars=_count_color(arr,(x0+196,92,x0+282,194),STAR_COLOR)
        if costs is None or stars is None or sum(v>0 for v in costs)!=2 or not 3<=stars<=7: return "abstain"
        recipes.append((costs,stars))
    stock=_counts(arr,(145,246,815,336))
    if stock is None or any(not 8<=x<=15 for x in stock): return "abstain"
    orders=[]
    for x0 in ORDER_X:
        plan=[]
        for cy in ROW_Y:
            value=_count_color(arr,(x0+42,cy-28,x0+174,cy+28),RUN_COLOR)
            if value is None or not 0<=value<=3: return "abstain"
            plan.append(value)
        orders.append(tuple(plan))
    uses=[tuple(sum(p[r]*recipes[r][0][c] for r in range(3)) for c in range(4)) for p in orders]
    rewards=[sum(p[r]*recipes[r][1] for r in range(3)) for p in orders]; scored=[]
    for a,b in itertools.combinations(range(5),2):
        if all(uses[a][c]+uses[b][c]<=stock[c] for c in range(4)): scored.append((rewards[a]+rewards[b],a,b))
    if not scored: return "abstain"
    top=max(x[0] for x in scored); winners=[(a,b) for score,a,b in scored if score==top]
    return f"pair-{LETTERS[winners[0][0]]}{LETTERS[winners[0][1]]}" if len(winners)==1 else "abstain"
