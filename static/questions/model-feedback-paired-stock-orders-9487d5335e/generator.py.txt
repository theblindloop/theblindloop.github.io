"""Deterministic renderer for choosing two orders against one shared stock."""
from __future__ import annotations
import itertools, random
from PIL import Image, ImageDraw, ImageFont

SIZE=(960,720); BACKGROUND=(246,244,238); PANEL=(255,254,250); INK=(38,42,48)
MUTED=(214,211,203); ACCENT=(178,76,54)
TOKEN_COLORS=((24,137,181),(146,75,163),(228,130,30),(52,153,91))
STAR_COLOR=(238,185,45); RUN_COLOR=(72,79,89)
RECIPE_COLORS=((208,73,73),(29,151,151),(128,92,184))
LETTERS="ABCDE"; PAIR_INDICES=tuple(itertools.combinations(range(5),2))
DECISIONS=tuple(f"pair-{LETTERS[a]}{LETTERS[b]}" for a,b in PAIR_INDICES)
RECIPE_X=(22,336,650); ORDER_X=(15,202,389,576,763)
FONT_PATH="/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"

def _font(size): return ImageFont.truetype(FONT_PATH,size)
def _usage(plan,recipes): return tuple(sum(int(plan[r])*int(recipes[r][0][c]) for r in range(3)) for c in range(4))
def _reward(plan,recipes): return sum(int(plan[r])*int(recipes[r][1]) for r in range(3))
def _pair_usage(a,b,recipes): return tuple(x+y for x,y in zip(_usage(a,recipes),_usage(b,recipes)))
def _pair_feasible(a,b,stock,recipes): return all(u<=s for u,s in zip(_pair_usage(a,b,recipes),stock))
def _pair_scores(orders,stock,recipes):
    return [(a,b,_reward(orders[a],recipes)+_reward(orders[b],recipes),_pair_feasible(orders[a],orders[b],stock,recipes)) for a,b in PAIR_INDICES]

def sample_scene(seed:int)->dict:
    rng=random.Random(int(seed))
    for _ in range(6000):
        links=[(0,1),(1,2),(2,3)]; rng.shuffle(links); recipes=[]
        for a,b in links:
            costs=[0,0,0,0]; costs[a],costs[b]=rng.randint(1,3),rng.randint(1,3)
            recipes.append((tuple(costs),rng.randint(3,7)))
        recipes=tuple(recipes); stock=tuple(rng.randint(8,15) for _ in range(4))
        pool=[p for p in itertools.product(range(4),repeat=3) if 2<=sum(p)<=7 and all(u<=s for u,s in zip(_usage(p,recipes),stock))]
        if len(pool)<8: continue
        orders=rng.sample(pool,5); scored=_pair_scores(orders,stock,recipes)
        feasible=sorted((score,a,b) for a,b,score,ok in scored if ok)
        if len(feasible)<3 or feasible[-1][0]-feasible[-2][0]<2: continue
        nominal_top=max(score for _,_,score,_ in scored)
        nominal=[(a,b) for a,b,score,_ in scored if score==nominal_top]
        if len(nominal)!=1 or _pair_feasible(orders[nominal[0][0]],orders[nominal[0][1]],stock,recipes): continue
        winner_old=feasible[-1][1:]; target=PAIR_INDICES[int(seed)%len(PAIR_INDICES)]
        remaining_old=[i for i in range(5) if i not in winner_old]; remaining_new=[i for i in range(5) if i not in target]
        rng.shuffle(remaining_old); rng.shuffle(remaining_new)
        old_for_new={target[0]:winner_old[0],target[1]:winner_old[1]}; old_for_new.update({n:o for n,o in zip(remaining_new,remaining_old)})
        shown=[orders[old_for_new[i]] for i in range(5)]
        return {"stock_counts":list(stock),"recipes":[{"costs":list(c),"stars":s} for c,s in recipes],"order_runs":[list(p) for p in shown],"layout_variant":rng.randrange(32)}
    raise RuntimeError("unable to sample a valid paired-stock scene")

def _validated(scene):
    stock=tuple(int(x) for x in scene["stock_counts"])
    recipes=tuple((tuple(int(x) for x in item["costs"]),int(item["stars"])) for item in scene["recipes"])
    orders=tuple(tuple(int(x) for x in p) for p in scene["order_runs"]); variant=int(scene["layout_variant"])
    if len(stock)!=4 or len(recipes)!=3 or len(orders)!=5: raise ValueError("wrong collection size")
    if any(len(c)!=4 or sum(v>0 for v in c)!=2 for c,_ in recipes): raise ValueError("bad recipe")
    if any(not 0<=x<=15 for x in stock) or any(not 0<=x<=3 for c,_ in recipes for x in c): raise ValueError("bad count")
    if any(not 3<=s<=7 for _,s in recipes) or any(len(p)!=3 or any(not 0<=x<=3 for x in p) for p in orders): raise ValueError("bad runs")
    if not 0<=variant<32: raise ValueError("bad variant")
    return stock,recipes,orders,variant

def analytic_gold(scene):
    try:
        stock,recipes,orders,_=_validated(scene)
        feasible=[(score,a,b) for a,b,score,ok in _pair_scores(orders,stock,recipes) if ok]
        if not feasible: return "abstain"
        top=max(x[0] for x in feasible); winners=[(a,b) for score,a,b in feasible if score==top]
        return f"pair-{LETTERS[winners[0][0]]}{LETTERS[winners[0][1]]}" if len(winners)==1 else "abstain"
    except (KeyError,TypeError,ValueError,IndexError): return "abstain"

def margin(scene):
    stock,recipes,orders,_=_validated(scene); scores=sorted((score for _,_,score,ok in _pair_scores(orders,stock,recipes) if ok),reverse=True)
    return float(scores[0]-scores[1]) if len(scores)>=2 and scores[0]>scores[1] else 0.0

def is_quarantined(scene):
    try:
        stock,recipes,orders,_=_validated(scene); scored=_pair_scores(orders,stock,recipes); feasible=[x for x in scored if x[3]]
        nominal=max(x[2] for x in scored); nominal_bad=all(not ok for _,_,score,ok in scored if score==nominal)
        individuals=all(all(u<=s for u,s in zip(_usage(p,recipes),stock)) for p in orders)
        return analytic_gold(scene)=="abstain" or margin(scene)<2 or len(feasible)<3 or len(set(orders))!=5 or not nominal_bad or not individuals
    except (KeyError,TypeError,ValueError): return True

def latent_symmetries(scene): return []

def _disks(draw,counts,box,columns,reverse,colors=TOKEN_COLORS,radius=7):
    x0,y0,x1,y1=box; kinds=[k for k,count in enumerate(counts) for _ in range(int(count))]
    if reverse: kinds.reverse()
    gap=20; rows=max(1,(len(kinds)+columns-1)//columns); shown_cols=min(columns,max(1,len(kinds)))
    sx=(x0+x1)//2-(shown_cols-1)*gap//2; sy=(y0+y1)//2-(rows-1)*gap//2
    for i,kind in enumerate(kinds):
        cx,cy=sx+(i%columns)*gap,sy+(i//columns)*gap
        draw.ellipse((cx-radius,cy-radius,cx+radius,cy+radius),fill=colors[kind],outline=INK,width=1)
def _same_disks(draw,count,box,columns,color,reverse=False): _disks(draw,[count],box,columns,reverse,colors=(color,),radius=7)

def render(scene):
    stock,recipes,orders,variant=_validated(scene); image=Image.new("RGB",SIZE,BACKGROUND); draw=ImageDraw.Draw(image)
    title,label,small=_font(23),_font(17),_font(14)
    draw.text((22,12),"RECIPES: TOKENS USED -> STARS EARNED",font=title,fill=INK)
    for i,((costs,stars),x0) in enumerate(zip(recipes,RECIPE_X)):
        draw.rounded_rectangle((x0,48,x0+288,202),radius=11,fill=PANEL,outline=RECIPE_COLORS[i],width=4)
        draw.text((x0+12,57),f"RECIPE {i+1}",font=small,fill=INK); _disks(draw,costs,(x0+8,96,x0+150,190),6,bool((variant>>i)&1))
        draw.line((x0+153,144,x0+194,144),fill=ACCENT,width=5); draw.polygon(((x0+194,144),(x0+181,134),(x0+181,154)),fill=ACCENT)
        _same_disks(draw,stars,(x0+198,96,x0+280,190),4,STAR_COLOR,bool((variant>>(i+3))&1))
    draw.text((22,216),"ONE SHARED START STOCK FOR THE TWO CHOSEN ORDERS",font=label,fill=INK)
    draw.rounded_rectangle((22,242,938,340),radius=11,fill=PANEL,outline=ACCENT,width=4); _disks(draw,stock,(150,248,810,334),14,bool(variant&16))
    draw.text((22,356),"CHOOSE TWO ORDERS  |  ONE MARK = ONE RECIPE RUN",font=label,fill=INK)
    for oi,(plan,x0) in enumerate(zip(orders,ORDER_X)):
        draw.rounded_rectangle((x0,386,x0+182,700),radius=11,fill=PANEL,outline=INK,width=3); draw.text((x0+55,398),f"ORDER {LETTERS[oi]}",font=small,fill=INK)
        draw.line((x0+9,430,x0+173,430),fill=MUTED,width=2)
        for ri,(runs,cy) in enumerate(zip(plan,(470,545,620))):
            draw.ellipse((x0+10,cy-14,x0+38,cy+14),fill=RECIPE_COLORS[ri],outline=INK,width=2)
            tw=draw.textbbox((0,0),str(ri+1),font=small)[2]; draw.text((x0+24-tw//2,cy-10),str(ri+1),font=small,fill=PANEL)
            _same_disks(draw,runs,(x0+45,cy-25,x0+171,cy+25),4,RUN_COLOR,bool((variant+oi+ri)&1))
            if ri<2: draw.line((x0+9,cy+36,x0+173,cy+36),fill=MUTED,width=1)
    return image
