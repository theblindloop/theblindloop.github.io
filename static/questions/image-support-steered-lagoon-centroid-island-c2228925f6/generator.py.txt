"""Deterministic sampler and renderer for Center of the Courtyard."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 384
HEIGHT = 384
CELL = 8
GRID_WIDTH = 20
GRID_HEIGHT = 15
BACKGROUND = (249, 248, 243)
PALE = (207, 210, 204)
PALE_EDGE = (187, 191, 184)
PALETTES = (
    (10, 107, 115),
    (24, 126, 123),
    (20, 101, 129),
)


def _base_regions(centroid_on_island: bool) -> tuple[set[tuple[int, int]], set[tuple[int, int]], set[tuple[int, int]]]:
    """Return outer sheet, island, and complete courtyard footprint cells."""

    all_cells = {(x, y) for x in range(GRID_WIDTH) for y in range(GRID_HEIGHT)}
    # Both footprints are 11 by 9.  The island stays fixed while the complete
    # footprint shifts three cells, so only full-region integration reveals
    # whether its centroid lands on the island.
    start_x = 3 if centroid_on_island else 6
    footprint = {(x, y) for x in range(start_x, start_x + 11) for y in range(3, 12)}
    island = {(x, y) for x in range(7, 10) for y in range(6, 9)}
    outer = all_cells - footprint
    return outer, island, footprint


def _transform_cells(
    cells: set[tuple[int, int]], rotation: int, reflected: bool
) -> tuple[set[tuple[int, int]], int, int]:
    width, height = GRID_WIDTH, GRID_HEIGHT
    output = set(cells)
    if reflected:
        output = {(width - 1 - x, y) for x, y in output}
    for _ in range(rotation % 4):
        output = {(height - 1 - y, x) for x, y in output}
        width, height = height, width
    return output, width, height


def _regions(scene: dict[str, Any]) -> tuple[set[tuple[int, int]], set[tuple[int, int]], set[tuple[int, int]], int, int]:
    outer, island, footprint = _base_regions(bool(scene["centroid_on_island"]))
    rotation = int(scene["rotation"])
    reflected = bool(scene["reflected"])
    outer_t, width, height = _transform_cells(outer, rotation, reflected)
    island_t, width_i, height_i = _transform_cells(island, rotation, reflected)
    footprint_t, width_f, height_f = _transform_cells(footprint, rotation, reflected)
    if (width, height) != (width_i, height_i) or (width, height) != (width_f, height_f):
        raise AssertionError("inconsistent transform dimensions")
    return outer_t, island_t, footprint_t, width, height


def _make_decorations(rng: random.Random) -> list[dict[str, int | str]]:
    anchors = ((31, 34), (351, 33), (31, 350), (351, 350), (192, 28))
    decorations: list[dict[str, int | str]] = []
    for index, (x, y) in enumerate(anchors):
        decorations.append(
            {
                "x": x + rng.randint(-5, 5),
                "y": y + rng.randint(-5, 5),
                "size": rng.randint(5, 8),
                "kind": "disk" if (index + rng.randrange(2)) % 2 else "square",
            }
        )
    return decorations


def sample_scene(seed: int) -> dict[str, Any]:
    """Adjacent seeds are an answer pair with identical nuisance variables."""

    pair = int(seed) // 2
    rng = random.Random(pair ^ 0x6A60C0DE)
    rotation = rng.randrange(4)
    reflected = bool(rng.randrange(2))
    grid_width, grid_height = (
        (GRID_WIDTH, GRID_HEIGHT) if rotation % 2 == 0 else (GRID_HEIGHT, GRID_WIDTH)
    )
    center_x = WIDTH // 2 + rng.randint(-3, 3)
    center_y = HEIGHT // 2 + rng.randint(-3, 3)
    return {
        "centroid_on_island": bool(int(seed) % 2),
        "origin_x": center_x - grid_width * CELL // 2,
        "origin_y": center_y - grid_height * CELL // 2,
        "rotation": rotation,
        "reflected": reflected,
        "palette": rng.randrange(len(PALETTES)),
        "decorations": _make_decorations(rng),
    }


def _draw_decorations(draw: ImageDraw.ImageDraw, scene: dict[str, Any]) -> None:
    for item in scene["decorations"]:
        x, y, size = int(item["x"]), int(item["y"]), int(item["size"])
        box = (x - size, y - size, x + size, y + size)
        if item["kind"] == "disk":
            draw.ellipse(box, fill=PALE, outline=PALE_EDGE, width=1)
        else:
            draw.rectangle(box, fill=PALE, outline=PALE_EDGE, width=1)


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_decorations(draw, scene)
    outer, island, _footprint, _width, _height = _regions(scene)
    teal = PALETTES[int(scene["palette"])]
    ox, oy = int(scene["origin_x"]), int(scene["origin_y"])
    for x, y in sorted(outer | island, key=lambda point: (point[1], point[0])):
        x0, y0 = ox + x * CELL, oy + y * CELL
        draw.rectangle((x0, y0, x0 + CELL - 1, y0 + CELL - 1), fill=teal)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    return "yes" if bool(scene["centroid_on_island"]) else "no"


def _centroid_clearance_cells(scene: dict[str, Any]) -> float:
    _outer, island, footprint, _width, _height = _regions(scene)
    cx = sum(x + 0.5 for x, _ in footprint) / len(footprint)
    cy = sum(y + 0.5 for _, y in footprint) / len(footprint)
    x0 = min(x for x, _ in island)
    x1 = max(x for x, _ in island) + 1.0
    y0 = min(y for _, y in island)
    y1 = max(y for _, y in island) + 1.0
    inside = x0 < cx < x1 and y0 < cy < y1
    if inside:
        return min(cx - x0, x1 - cx, cy - y0, y1 - cy)
    dx = max(x0 - cx, 0.0, cx - x1)
    dy = max(y0 - cy, 0.0, cy - y1)
    return -math.hypot(dx, dy)


def margin(scene: dict[str, Any]) -> float:
    """Signed centroid-to-island boundary clearance in pixels."""

    return round(_centroid_clearance_cells(scene) * CELL, 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        outer, island, footprint, grid_width, grid_height = _regions(scene)
        ox, oy = int(scene["origin_x"]), int(scene["origin_y"])
        palette = int(scene["palette"])
        decorations = scene["decorations"]
    except (KeyError, TypeError, ValueError):
        return True
    return bool(
        len(outer) != 201
        or len(island) != 9
        or len(footprint) != 99
        or not 0 <= palette < len(PALETTES)
        or not isinstance(decorations, list)
        or ox < 24
        or oy < 24
        or ox + grid_width * CELL > WIDTH - 24
        or oy + grid_height * CELL > HEIGHT - 24
        or abs(margin(scene)) < 8.0
    )


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
