"""Independent pixel-only inverse for Center of the Courtyard.

The inverse receives only a final PNG.  It segments teal ink, recovers the
outer component and detached island, fills across the island to reconstruct
the complete bounded courtyard footprint, computes that footprint's pixel-area
centroid, and tests whether the derived point is on the island.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

_TEALS = np.asarray(((10, 107, 115), (24, 126, 123), (20, 101, 129)), dtype=np.int16)
_EXPECTED_TOTAL = 210 * 8 * 8
_EXPECTED_OUTER = 201 * 8 * 8
_EXPECTED_ISLAND = 9 * 8 * 8
_EXPECTED_FOOTPRINT = 99 * 8 * 8
_OUTER_DIMS = {(160, 120), (120, 160)}
_FOOTPRINT_DIMS = {(88, 72), (72, 88)}
_MIN_CLEARANCE = 7.5


def _teal_mask(image: Image.Image) -> np.ndarray:
    if image.size != (384, 384):
        raise ValueError(f"unexpected canvas {image.size}")
    array = np.asarray(image.convert("RGB"), dtype=np.int32)
    palette = _TEALS.astype(np.int32)
    distances = ((array[:, :, None, :] - palette[None, None, :, :]) ** 2).sum(axis=3)
    return distances.min(axis=2) <= 30 * 30


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        seen[sy, sx] = True
        queue = [(sy, sx)]
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.pop()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        component = np.zeros_like(mask, dtype=bool)
        yy, xx = zip(*points)
        component[np.asarray(yy), np.asarray(xx)] = True
        output.append(component)
    return sorted(output, key=lambda item: int(item.sum()), reverse=True)


def _bounded_components(free: np.ndarray) -> list[np.ndarray]:
    height, width = free.shape
    seen = np.zeros_like(free, dtype=bool)
    bounded: list[np.ndarray] = []
    for sy, sx in np.argwhere(free):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        seen[sy, sx] = True
        queue = [(sy, sx)]
        points: list[tuple[int, int]] = []
        touches_edge = False
        while queue:
            y, x = queue.pop()
            points.append((y, x))
            touches_edge |= y == 0 or x == 0 or y == height - 1 or x == width - 1
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and free[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if not touches_edge:
            component = np.zeros_like(free, dtype=bool)
            yy, xx = zip(*points)
            component[np.asarray(yy), np.asarray(xx)] = True
            bounded.append(component)
    return bounded


def _extent(mask: np.ndarray) -> tuple[int, int, int, int]:
    ys, xs = np.nonzero(mask)
    if not len(xs):
        raise ValueError("empty component")
    return int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())


def decision_from_image(image: Image.Image) -> str:
    teal = _teal_mask(image)
    if int(teal.sum()) != _EXPECTED_TOTAL:
        raise ValueError(f"incomplete teal evidence: {int(teal.sum())} pixels")
    components = _components(teal)
    if len(components) != 2:
        raise ValueError(f"expected outer sheet and one island, found {len(components)} components")
    outer, island = components
    if int(outer.sum()) != _EXPECTED_OUTER or int(island.sum()) != _EXPECTED_ISLAND:
        raise ValueError("unexpected outer-sheet or island area")

    x0, y0, x1, y1 = _extent(outer)
    if (x1 - x0 + 1, y1 - y0 + 1) not in _OUTER_DIMS:
        raise ValueError("unexpected outer-sheet extent")
    ix0, iy0, ix1, iy1 = _extent(island)
    if (ix1 - ix0 + 1, iy1 - iy0 + 1) != (24, 24):
        raise ValueError("unexpected island extent")
    if not (x0 + 8 <= ix0 and ix1 <= x1 - 8 and y0 + 8 <= iy0 and iy1 <= y1 - 8):
        raise ValueError("island lacks a full visible separating gap")

    # Invert only the outer component in a padded crop.  Because the detached
    # island is absent from `outer`, it joins the surrounding white pixels here;
    # the sole bounded component is exactly the complete courtyard footprint
    # specified by the question (white space plus island).
    crop_outer = outer[y0 - 1 : y1 + 2, x0 - 1 : x1 + 2]
    bounded = _bounded_components(~crop_outer)
    if len(bounded) != 1:
        raise ValueError(f"expected one complete courtyard footprint, found {len(bounded)}")
    footprint = bounded[0]
    if int(footprint.sum()) != _EXPECTED_FOOTPRINT:
        raise ValueError(f"damaged courtyard footprint area {int(footprint.sum())}")
    fx0, fy0, fx1, fy1 = _extent(footprint)
    if (fx1 - fx0 + 1, fy1 - fy0 + 1) not in _FOOTPRINT_DIMS:
        raise ValueError("unexpected courtyard footprint extent")

    island_local = island[y0 - 1 : y1 + 2, x0 - 1 : x1 + 2]
    if np.any(island_local & ~footprint):
        raise ValueError("island is not nested in the courtyard footprint")
    py, px = np.nonzero(footprint)
    cx, cy = float(px.mean()), float(py.mean())
    probe_x, probe_y = int(math.floor(cx + 0.5)), int(math.floor(cy + 0.5))
    if not (0 <= probe_y < island_local.shape[0] and 0 <= probe_x < island_local.shape[1]):
        raise ValueError("derived centroid left the emblem crop")
    inside = bool(island_local[probe_y, probe_x])

    iy, ix = np.nonzero(island_local)
    left, right = float(ix.min()) - 0.5, float(ix.max()) + 0.5
    top, bottom = float(iy.min()) - 0.5, float(iy.max()) + 0.5
    if inside:
        clearance = min(cx - left, right - cx, cy - top, bottom - cy)
    else:
        dx = max(left - cx, 0.0, cx - right)
        dy = max(top - cy, 0.0, cy - bottom)
        clearance = math.hypot(dx, dy)
    if clearance < _MIN_CLEARANCE:
        raise ValueError(f"centroid/island boundary is ambiguous ({clearance:.2f}px)")
    return "yes" if inside else "no"
