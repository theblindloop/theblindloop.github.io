"""Renderer for the nearest-tick lean-count world.

A white measurement sheet carries two vertically separated rows of short
vertical ticks that never touch each other:

* a *reference lattice* of pure black ticks on a strictly uniform pitch, and
* a *gauge row* of pure red ticks, each one planted a small signed offset away
  from one of the black lattice ticks.

The decision is how many red gauge ticks sit to the **left** of the black
lattice tick that is nearest to them.  Every red tick contributes, and the black
tick nearest to a red one decides that red tick's lean, so the decisive evidence
is a set of disconnected marks distributed over one medium-sized band of the
sheet.  Grey and blue decoration is drawn away from that band and carries no
information.

Geometry is integer and antialiasing-free: every tick is an axis-aligned filled
rectangle of odd width, so its rendered centre column is exactly the sampled
centre and the pixel arm recovers the same numbers the analytic gold uses.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "nearest_tick_lean_count-renderer-0.1.0"

WIDTH = 800
HEIGHT = 400

TICK_WIDTH = 3  # odd, so the drawn centre column is the sampled centre
REF_TICK_HEIGHT = 30
GAUGE_TICK_HEIGHT = 22
ROW_GAP = 46  # white paper between the two rows

N_GAUGE = 6  # constant, so gauge-glyph count carries no answer signal
MIN_REF = 6
MAX_REF = 11
MIN_LATTICE_PX = 240  # the band is a substantial region of the sheet
MAX_LATTICE_PX = 460

BLACK = (0, 0, 0)
RED = (206, 30, 34)
GREY = (152, 152, 152)
BLUE = (118, 152, 208)
GREEN = (128, 176, 128)
WHITE = (255, 255, 255)

MIN_OFFSET_PX = 5  # smallest sampled |offset|, used only by near-boundary scenes
COMFORTABLE_OFFSET_PX = 7  # smallest |offset| in an ordinary scene
NEAREST_SLACK_PX = 7  # keeps |offset| away from the half-pitch ambiguity
QUARANTINE_MARGIN_PX = 6.0
SAMPLE_MARGIN_PX = 5.0

ANSWERS = ("1", "2", "3", "4")


def _pitches() -> tuple[int, ...]:
    return (32, 36, 40, 44)


def sample_scene(seed: int) -> dict:
    """Deterministically sample one answerable scene."""
    rng = random.Random(seed * 7919 + 13)
    for _ in range(400):
        pitch = rng.choice(_pitches())
        max_offset = pitch // 2 - NEAREST_SLACK_PX
        # Most scenes keep every tick comfortably clear of the lean boundary;
        # roughly a quarter are sampled near it and become the quarantined
        # near-boundary coverage of the gallery.
        near_boundary = rng.random() < 0.3
        min_offset = MIN_OFFSET_PX if near_boundary else COMFORTABLE_OFFSET_PX
        if max_offset < min_offset:
            continue

        # Which lattice ticks carry a gauge tick: strictly increasing indices
        # with gaps of one or two.  The first and last lattice tick always carry
        # one, so the lattice never extends past the gauge row and every black
        # tick is decisive - an interior one because removing it breaks the
        # uniform pitch, an end one because its own gauge tick then loses its
        # partner.
        index = 0
        indices = [index]
        while len(indices) < N_GAUGE:
            index += rng.choice((1, 1, 2))
            indices.append(index)
        n_ref = indices[-1] + 1
        if not MIN_REF <= n_ref <= MAX_REF:
            continue
        lattice_width = (n_ref - 1) * pitch
        if not MIN_LATTICE_PX <= lattice_width <= MAX_LATTICE_PX:
            continue

        n_left = rng.randint(1, 4)
        signs = [-1] * n_left + [1] * (N_GAUGE - n_left)
        rng.shuffle(signs)
        offsets = [sign * rng.randint(min_offset, max_offset) for sign in signs]

        lean_margin = min(
            min(abs(offset), pitch / 2.0 - abs(offset)) for offset in offsets
        )
        if lean_margin < SAMPLE_MARGIN_PX:
            continue

        origin_x = rng.randint(70, WIDTH - 70 - lattice_width)
        band_top = rng.randint(96, HEIGHT - 96 - REF_TICK_HEIGHT - ROW_GAP - GAUGE_TICK_HEIGHT)
        gauge_below = rng.choice((True, False))

        scene = {
            "width": WIDTH,
            "height": HEIGHT,
            "pitch": pitch,
            "n_ref": n_ref,
            "origin_x": origin_x,
            "band_top": band_top,
            "gauge_below": gauge_below,
            "gauge_indices": indices,
            "gauge_offsets": offsets,
            "decoration": _sample_decoration(rng, band_top),
        }
        return scene
    raise RuntimeError(f"no answerable scene for seed {seed}")


def _sample_decoration(rng: random.Random, band_top: int) -> list[dict]:
    """Non-decisive marks, kept clear of the two decisive rows."""
    band_bottom = band_top + REF_TICK_HEIGHT + ROW_GAP + GAUGE_TICK_HEIGHT
    zones = []
    if band_top - 30 > 20:
        zones.append((14, band_top - 30))
    if band_bottom + 30 < HEIGHT - 20:
        zones.append((band_bottom + 22, HEIGHT - 26))
    marks: list[dict] = []
    for _ in range(rng.randint(5, 9)):
        if not zones:
            break
        top, bottom = rng.choice(zones)
        if bottom - top < 12:
            continue
        marks.append(
            {
                "kind": rng.choice(("tick", "bar", "square")),
                "colour": rng.choice(("grey", "blue", "green")),
                "x": rng.randint(20, WIDTH - 60),
                "y": rng.randint(top, bottom - 10),
                "w": rng.randint(3, 26),
                "h": rng.randint(4, 14),
            }
        )
    return marks


# --------------------------------------------------------------------------
# geometry helpers shared by the analytic arm and the tests
# --------------------------------------------------------------------------


def reference_centres(scene: dict) -> list[int]:
    """Centre column of every black lattice tick, left to right."""
    return [scene["origin_x"] + i * scene["pitch"] for i in range(scene["n_ref"])]


def gauge_centres(scene: dict) -> list[int]:
    """Centre column of every red gauge tick, left to right."""
    centres = reference_centres(scene)
    return [
        centres[index] + offset
        for index, offset in zip(scene["gauge_indices"], scene["gauge_offsets"])
    ]


def _rows(scene: dict) -> tuple[tuple[int, int], tuple[int, int]]:
    """(top, height) of the reference row and of the gauge row."""
    top = scene["band_top"]
    if scene["gauge_below"]:
        ref = (top, REF_TICK_HEIGHT)
        gauge = (top + REF_TICK_HEIGHT + ROW_GAP, GAUGE_TICK_HEIGHT)
    else:
        gauge = (top, GAUGE_TICK_HEIGHT)
        ref = (top + GAUGE_TICK_HEIGHT + ROW_GAP, REF_TICK_HEIGHT)
    return ref, gauge


def decisive_boxes(scene: dict) -> list[list[int]]:
    """Bounding box of every decisive mark: all black and all red ticks."""
    (ref_top, ref_h), (gauge_top, gauge_h) = _rows(scene)
    half = TICK_WIDTH // 2
    boxes = [
        [x - half, ref_top, x + half, ref_top + ref_h - 1] for x in reference_centres(scene)
    ]
    boxes += [
        [x - half, gauge_top, x + half, gauge_top + gauge_h - 1] for x in gauge_centres(scene)
    ]
    return boxes


def support_box(scene: dict) -> list[int]:
    """Bounding box of the whole decisive band."""
    boxes = decisive_boxes(scene)
    return [
        min(b[0] for b in boxes),
        min(b[1] for b in boxes),
        max(b[2] for b in boxes),
        max(b[3] for b in boxes),
    ]


def _lean_flags(scene: dict) -> list[bool]:
    """True where a gauge tick sits left of its nearest lattice tick."""
    centres = reference_centres(scene)
    flags = []
    for gauge in gauge_centres(scene):
        nearest = min(centres, key=lambda c: (abs(c - gauge), c))
        flags.append(gauge < nearest)
    return flags


# --------------------------------------------------------------------------
# contract interface
# --------------------------------------------------------------------------


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (scene["width"], scene["height"]), WHITE)
    draw = ImageDraw.Draw(image)

    palette = {"grey": GREY, "blue": BLUE, "green": GREEN}
    for mark in scene["decoration"]:
        colour = palette[mark["colour"]]
        if mark["kind"] == "tick":
            draw.rectangle([mark["x"], mark["y"], mark["x"] + 2, mark["y"] + mark["h"]], fill=colour)
        elif mark["kind"] == "bar":
            draw.rectangle([mark["x"], mark["y"], mark["x"] + mark["w"], mark["y"] + 2], fill=colour)
        else:
            draw.rectangle(
                [mark["x"], mark["y"], mark["x"] + mark["h"], mark["y"] + mark["h"]], fill=colour
            )

    (ref_top, ref_h), (gauge_top, gauge_h) = _rows(scene)
    half = TICK_WIDTH // 2
    for x in reference_centres(scene):
        draw.rectangle([x - half, ref_top, x + half, ref_top + ref_h - 1], fill=BLACK)
    for x in gauge_centres(scene):
        draw.rectangle([x - half, gauge_top, x + half, gauge_top + gauge_h - 1], fill=RED)
    return image


def analytic_gold(scene: dict) -> str:
    return str(sum(_lean_flags(scene)))


def margin(scene: dict) -> float:
    """Weakest per-tick evidence, in pixels.

    For each gauge tick this is the smaller of the distance to its nearest
    lattice tick (how clearly it leans) and the distance from that offset to the
    half-pitch (how clearly that lattice tick is the nearest one).  The scene is
    as strong as its weakest gauge tick.
    """
    pitch = scene["pitch"]
    return float(
        min(min(abs(o), pitch / 2.0 - abs(o)) for o in scene["gauge_offsets"])
    )


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases: every sampled field draws ink."""
    return []


# --------------------------------------------------------------------------
# label-preserving variants exercised by the candidate tests
# --------------------------------------------------------------------------


def translated(scene: dict, dx: int, dy: int) -> dict:
    variant = dict(scene)
    variant["origin_x"] = scene["origin_x"] + dx
    variant["band_top"] = scene["band_top"] + dy
    variant["decoration"] = [dict(m) for m in scene["decoration"]]
    return variant


def flipped_rows(scene: dict) -> dict:
    """Swap which row is on top; the roles are carried by colour, not position."""
    variant = dict(scene)
    variant["gauge_below"] = not scene["gauge_below"]
    variant["decoration"] = [dict(m) for m in scene["decoration"]]
    return variant


def permuted_decoration(scene: dict) -> dict:
    variant = dict(scene)
    marks = [dict(m) for m in scene["decoration"]]
    variant["decoration"] = list(reversed(marks))
    return variant


def without_decoration(scene: dict) -> dict:
    variant = dict(scene)
    variant["decoration"] = []
    return variant
