"""Pixel-only inverse arm for the nearest-tick lean-count world.

Input is a single rasterised PNG.  No latent scene, renderer code, analytic
gold, manifest row, file name or metadata is available to this module; it
imports only numpy and Pillow.

Method
------
1. Segment two ink classes by colour alone: near-black ink (all channels dark
   and nearly equal) and saturated red ink (red channel dominant).
2. Recover each tick as a maximal run of consecutive image columns that contain
   ink of that class, and take its intensity-weighted centre column.  The ticks
   are vertical strokes separated by white paper, so column runs recover them
   without any assumption about how many there are.
3. For every red tick, find the nearest black tick centre and record whether the
   red tick lies to its left.
4. Report the number of left-leaning red ticks.

The arm abstains when the raster is not a legible instance of the public
contract: fewer than four black ticks, a number of red ticks other than the
declared six, a black lattice that is not on a single uniform pitch, a red tick
sitting exactly on a lattice tick, or a red tick whose offset is not clearly
inside the half-pitch, so that its nearest lattice tick is ambiguous or absent.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "nearest_tick_lean_count-oracle-0.1.0"

ABSTAIN = "abstain"
_MIN_RUN_PIXELS = 8  # a tick is a stroke, not a speck
_AMBIGUITY_SLACK_PX = 3.0
_DECLARED_GAUGE_TICKS = 6  # stated by the public question contract


def _masks(image: Image.Image) -> tuple[np.ndarray, np.ndarray]:
    rgb = np.asarray(image.convert("RGB")).astype(np.int16)
    red, green, blue = rgb[:, :, 0], rgb[:, :, 1], rgb[:, :, 2]
    hi = rgb.max(axis=2)
    lo = rgb.min(axis=2)
    black = (hi < 90) & ((hi - lo) < 40)
    red_ink = (red > 140) & (green < 100) & (blue < 100) & ((red - np.maximum(green, blue)) > 70)
    return black, red_ink


def _tick_centres(mask: np.ndarray) -> list[float]:
    """Centre column of every maximal run of ink-bearing columns."""
    counts = mask.sum(axis=0)
    occupied = counts > 0
    centres: list[float] = []
    start = None
    for column in range(mask.shape[1] + 1):
        here = bool(occupied[column]) if column < mask.shape[1] else False
        if here and start is None:
            start = column
        elif not here and start is not None:
            weights = counts[start:column].astype(np.float64)
            total = float(weights.sum())
            if total >= _MIN_RUN_PIXELS:
                columns = np.arange(start, column, dtype=np.float64)
                centres.append(float((columns * weights).sum() / total))
            start = None
    return centres


def _uniform_pitch(centres: list[float]) -> float | None:
    """Pitch of a lattice, or None when the black ticks are not uniform."""
    gaps = [b - a for a, b in zip(centres, centres[1:])]
    if not gaps:
        return None
    pitch = float(np.median(gaps))
    if pitch < 12.0:
        return None
    if max(abs(gap - pitch) for gap in gaps) > 1.5:
        return None
    return pitch


def decision_from_image(image: Image.Image) -> str:
    black, red_ink = _masks(image)
    reference = _tick_centres(black)
    gauge = _tick_centres(red_ink)
    # The public question declares the gauge row: exactly six red ticks. A
    # raster showing any other number is not a legible instance of the contract.
    if len(reference) < 4 or len(gauge) != _DECLARED_GAUGE_TICKS:
        return ABSTAIN

    pitch = _uniform_pitch(reference)
    if pitch is None:
        return ABSTAIN

    lattice = np.asarray(reference, dtype=np.float64)
    leaning = 0
    for centre in gauge:
        offsets = lattice - centre
        nearest = int(np.argmin(np.abs(offsets)))
        signed = float(offsets[nearest])
        # A legible gauge tick stands a little way from one lattice tick: its
        # offset must be clearly inside the half-pitch, otherwise the partner is
        # ambiguous or missing altogether.
        if abs(signed) > pitch / 2.0 - _AMBIGUITY_SLACK_PX:
            return ABSTAIN
        if abs(signed) < 1.0:
            return ABSTAIN
        if signed > 0:
            leaning += 1
    return str(leaning)
