"""Independent PNG-only inverse for component bounding-box solidification."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

GRID = 12


def _templates() -> dict[str, np.ndarray]:
    result = {}
    mask = np.zeros((GRID, GRID), bool); mask[3:5, 3:9] = 1; mask[8:10, 3:9] = 1; result["equals"] = mask
    mask = np.zeros((GRID, GRID), bool); mask[3:9, 3:5] = 1; mask[3:9, 8:10] = 1; result["pause"] = mask
    mask = np.zeros((GRID, GRID), bool); mask[2:5, 7:11] = 1; mask[7:10, 1:5] = 1; result["rising pair"] = mask
    mask = np.zeros((GRID, GRID), bool); mask[2:5, 1:5] = 1; mask[7:10, 7:11] = 1; result["falling pair"] = mask
    return result


TEMPLATES = _templates()


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    seen = np.zeros_like(mask, dtype=bool); groups = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]: continue
        seen[y0, x0] = True; queue = deque([(y0, x0)]); group = []
        while queue:
            y, x = queue.popleft(); group.append((y, x))
            for ny, nx in ((y-1, x), (y+1, x), (y, x-1), (y, x+1)):
                if 0 <= ny < mask.shape[0] and 0 <= nx < mask.shape[1] and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True; queue.append((ny, nx))
        groups.append(group)
    return groups


def _cyan(pixel: tuple[int, int, int]) -> bool:
    r, g, b = pixel
    return r < 10 and g > 120 and b > 145


def _bbox(image: Image.Image) -> tuple[int, int, int, int]:
    points = [(x, y) for y in range(image.height) for x in range(image.width) if _cyan(image.getpixel((x, y)))]
    if len(points) < 1000: raise ValueError("cyan grid frame missing")
    xs, ys = zip(*points)
    return min(xs), min(ys), max(xs), max(ys)


def decision_with_margin(image: Image.Image) -> tuple[str, float]:
    raster = image.convert("RGB")
    if raster.size != (512, 512): raise ValueError("unexpected raster size")
    fx0, fy0, fx1, fy1 = _bbox(raster)
    if abs((fx1-fx0) - (fy1-fy0)) > 2: raise ValueError("frame is not square")
    left, top = fx0 + 4, fy0 + 4
    span = (fx1 - fx0) - 8
    cell = round(span / GRID)
    if cell not in (28, 30, 32) or abs(span - GRID*cell) > 1: raise ValueError("invalid visible grid geometry")
    source = np.zeros((GRID, GRID), dtype=bool)
    for row in range(GRID):
        for col in range(GRID):
            cx, cy = left + col*cell + cell//2, top + row*cell + cell//2
            probes = [raster.getpixel((cx+dx, cy+dy)) for dx, dy in ((0,0),(-4,-4),(4,-4),(-4,4),(4,4))]
            source[row, col] = sum(max(p) < 100 for p in probes) >= 4
    groups = _components(source)
    if len(groups) != 2 or any(len(group) != 8 for group in groups):
        raise ValueError("expected two eight-cell side-connected source components")
    solid = np.zeros_like(source)
    for group in groups:
        ys = [p[0] for p in group]; xs = [p[1] for p in group]
        solid[min(ys):max(ys)+1, min(xs):max(xs)+1] = True
    scores = sorted((int(np.count_nonzero(solid ^ template)), label) for label, template in TEMPLATES.items())
    gap = scores[1][0] - scores[0][0]
    if scores[0][0] != 0 or gap < 8: raise ValueError(f"ambiguous solidified symbol: {scores[:2]}")
    return scores[0][1], float(gap)


def decision_from_image(image: Image.Image) -> str:
    return decision_with_margin(image)[0]
