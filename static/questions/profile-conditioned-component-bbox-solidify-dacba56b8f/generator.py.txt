"""Renderer and analytic arm for component bounding-box solidification."""

from __future__ import annotations

import copy
import random
from collections import deque

import numpy as np
from PIL import Image, ImageDraw

SYMBOLS = ("equals", "pause", "rising pair", "falling pair")
GRID = 12
CANVAS = 512


def _templates() -> dict[str, np.ndarray]:
    result: dict[str, np.ndarray] = {}
    mask = np.zeros((GRID, GRID), dtype=bool)
    mask[3:5, 3:9] = True; mask[8:10, 3:9] = True
    result["equals"] = mask
    mask = np.zeros((GRID, GRID), dtype=bool)
    mask[3:9, 3:5] = True; mask[3:9, 8:10] = True
    result["pause"] = mask
    mask = np.zeros((GRID, GRID), dtype=bool)
    mask[2:5, 7:11] = True; mask[7:10, 1:5] = True
    result["rising pair"] = mask
    mask = np.zeros((GRID, GRID), dtype=bool)
    mask[2:5, 1:5] = True; mask[7:10, 7:11] = True
    result["falling pair"] = mask
    assert {int(value.sum()) for value in result.values()} == {24}
    return result


TEMPLATES = _templates()
PALETTES = (
    {"paper": (250, 249, 246), "empty": (239, 241, 239), "ink": (23, 28, 35), "grid": (196, 203, 204), "frame": (0, 157, 181)},
    {"paper": (247, 250, 248), "empty": (240, 238, 242), "ink": (42, 25, 43), "grid": (202, 207, 205), "frame": (0, 143, 176)},
    {"paper": (251, 247, 244), "empty": (238, 240, 242), "ink": (19, 37, 38), "grid": (207, 201, 198), "frame": (0, 168, 166)},
)
DOT_COLORS = ((226, 92, 52), (75, 147, 88), (233, 179, 35), (102, 91, 181), (216, 59, 139), (55, 119, 190), (230, 132, 45), (96, 157, 151))
DOT_POSITIONS = ((24, 70), (24, 176), (24, 282), (24, 388), (488, 70), (488, 176), (488, 282), (488, 388))


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[list[tuple[int, int]]] = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]:
            continue
        seen[y0, x0] = True
        queue = deque([(y0, x0)])
        group: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft(); group.append((y, x))
            for ny, nx in ((y-1, x), (y+1, x), (y, x-1), (y, x+1)):
                if 0 <= ny < GRID and 0 <= nx < GRID and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True; queue.append((ny, nx))
        groups.append(group)
    return groups


def _sparse_component(y0: int, y1: int, x0: int, x1: int, variant: int) -> set[tuple[int, int]]:
    """Eight connected cells spanning a 2x6, 6x2, or 3x4 rectangle."""
    h, w = y1 - y0 + 1, x1 - x0 + 1
    if (h, w) == (2, 6):
        local = {(0, c) for c in range(4)} | {(1, c) for c in range(2, 6)}
        if variant % 2: local = {(r, w - 1 - c) for r, c in local}
    elif (h, w) == (6, 2):
        base = _sparse_component(0, 1, 0, 5, variant)
        local = {(c, r) for r, c in base}
    elif (h, w) == (3, 4):
        local = {(0, 0), (0, 1), (1, 0), (1, 1), (1, 2), (1, 3), (2, 2), (2, 3)}
        if variant % 2: local = {(r, w - 1 - c) for r, c in local}
    else:
        raise ValueError("unsupported template rectangle")
    return {(y0 + r, x0 + c) for r, c in local}


def _source_for(label: str, variants: list[int]) -> np.ndarray:
    target = TEMPLATES[label]
    source = np.zeros_like(target)
    for index, group in enumerate(_components(target)):
        ys = [p[0] for p in group]; xs = [p[1] for p in group]
        for y, x in _sparse_component(min(ys), max(ys), min(xs), max(xs), variants[index]):
            source[y, x] = True
    assert len(_components(source)) == 2 and int(source.sum()) == 16
    return source


def _solidify(source: np.ndarray) -> np.ndarray:
    result = np.zeros_like(source)
    for group in _components(source):
        ys = [p[0] for p in group]; xs = [p[1] for p in group]
        result[min(ys):max(ys)+1, min(xs):max(xs)+1] = True
    return result


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0xB0B5011D)
    target = SYMBOLS[int(seed) % len(SYMBOLS)]
    colors = list(range(len(DOT_COLORS))); rng.shuffle(colors)
    distractors = [[x, y, rng.randint(4, 7), color] for (x, y), color in zip(DOT_POSITIONS, colors)]
    return {
        "target": target,
        "source_cells": [list(p) for p in sorted(np.argwhere(_source_for(target, [rng.randrange(2), rng.randrange(2)])).tolist())],
        "frame_x": rng.randint(-8, 8),
        "frame_y": rng.randint(-8, 8),
        "cell_px": rng.choice((28, 30, 32)),
        "palette_variant": rng.randrange(len(PALETTES)),
        "distractors": distractors,
    }


def _source(scene: dict) -> np.ndarray:
    result = np.zeros((GRID, GRID), dtype=bool)
    for row, col in scene["source_cells"]:
        result[int(row), int(col)] = True
    return result


def render(scene: dict) -> Image.Image:
    source = _source(scene)
    if not np.array_equal(_solidify(source), TEMPLATES[str(scene["target"])]):
        raise ValueError("source components do not solidify to target")
    palette = PALETTES[int(scene["palette_variant"])]
    cell = int(scene["cell_px"]); span = GRID * cell
    left = (CANVAS - span) // 2 + int(scene["frame_x"])
    top = (CANVAS - span) // 2 + int(scene["frame_y"])
    image = Image.new("RGB", (CANVAS, CANVAS), palette["paper"])
    draw = ImageDraw.Draw(image)
    for x, y, radius, color in scene["distractors"]:
        draw.ellipse((x-radius, y-radius, x+radius, y+radius), fill=DOT_COLORS[int(color)])
    for row in range(GRID):
        for col in range(GRID):
            x0, y0 = left + col*cell, top + row*cell
            draw.rectangle((x0+2, y0+2, x0+cell-2, y0+cell-2), fill=palette["ink"] if source[row, col] else palette["empty"])
    for i in range(GRID + 1):
        draw.line((left+i*cell, top, left+i*cell, top+span), fill=palette["grid"], width=1)
        draw.line((left, top+i*cell, left+span, top+i*cell), fill=palette["grid"], width=1)
    draw.rectangle((left-4, top-4, left+span+4, top+span+4), outline=palette["frame"], width=4)
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["target"])


def margin(scene: dict) -> float:
    result = _solidify(_source(scene))
    scores = sorted(int(np.count_nonzero(result ^ template)) for template in TEMPLATES.values())
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    transformed = copy.deepcopy(scene)
    transformed["distractors"] = list(reversed(transformed["distractors"]))
    return [("reverse_nonoverlapping_distractor_draw_order", transformed)]
