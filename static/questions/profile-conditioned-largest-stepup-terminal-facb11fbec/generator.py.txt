"""Deterministic latent sampler, raster renderer, and analytic arm."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

WIDTH = 780
HEIGHT = 560
NODE_RADIUS = 17
LINK_WIDTH = 8
BACKGROUND = (248, 249, 252)
BLACK = (28, 31, 36)
SEPARATOR = (218, 222, 230)
COLORS = {
    "green": (20, 150, 82),
    "purple": (137, 70, 190),
    "red": (220, 55, 65),
    "blue": (45, 105, 210),
    "orange": (235, 140, 35),
}
TERMINALS = ("red", "blue", "orange")
QUARANTINE_GAP = 14.0

# Each template contains winner, runner-up, and low-score link sequences.
# All three sequences in a template have exactly the same total length.
_TEMPLATES = (
    (
        (120.0, 190.0, 145.0, 155.0),
        (140.0, 165.0, 135.0, 170.0),
        (160.0, 150.0, 145.0, 155.0),
    ),
    (
        (135.0, 195.0, 140.0, 160.0),
        (150.0, 175.0, 135.0, 170.0),
        (165.0, 150.0, 160.0, 155.0),
    ),
    (
        (125.0, 185.0, 150.0, 160.0),
        (145.0, 170.0, 135.0, 170.0),
        (160.0, 145.0, 165.0, 150.0),
    ),
    (
        (130.0, 184.0, 141.0, 155.0),
        (142.0, 176.0, 128.0, 164.0),
        (165.0, 145.0, 155.0, 145.0),
    ),
)


def _points_for_lengths(lengths: tuple[float, ...], center_x: float, mirror: bool, flip: bool):
    dy = 110.0
    steps = [math.sqrt(length * length - dy * dy) for length in lengths]
    direction = -1.0 if mirror else 1.0
    offsets = [0.0]
    for index, step in enumerate(steps):
        offsets.append(offsets[-1] + direction * (step if index % 2 == 0 else -step))
    shift = center_x - (min(offsets) + max(offsets)) / 2.0
    ys = [60.0 + 110.0 * index for index in range(5)]
    if flip:
        ys = [HEIGHT - value for value in ys]
    return [[round(value + shift, 3), round(y, 3)] for value, y in zip(offsets, ys)]


def sample_scene(seed: int) -> dict:
    """Sample a class-, panel-, orientation-, and template-balanced scene."""
    rng = random.Random(int(seed))
    template = _TEMPLATES[(int(seed) // 9) % len(_TEMPLATES)]
    scale = 0.985 + 0.03 * rng.random()
    sequences = [tuple(value * scale for value in sequence) for sequence in template]

    winner_color = TERMINALS[int(seed) % 3]
    winner_panel = (int(seed) // 3) % 3
    other_colors = [color for color in TERMINALS if color != winner_color]
    rng.shuffle(other_colors)
    other_roles = [1, 2]
    if int(seed) % 2:
        other_roles.reverse()

    role_for_panel = [None, None, None]
    role_for_panel[winner_panel] = 0
    open_panels = [index for index in range(3) if index != winner_panel]
    for panel, role in zip(open_panels, other_roles):
        role_for_panel[panel] = role

    color_for_role = {0: winner_color, other_roles[0]: other_colors[0], other_roles[1]: other_colors[1]}
    flip = bool((int(seed) // 2) % 2)
    paths = []
    for panel, role in enumerate(role_for_panel):
        center_x = 130.0 + 260.0 * panel + rng.uniform(-3.0, 3.0)
        mirror = bool((int(seed) + panel) % 2)
        points = _points_for_lengths(sequences[role], center_x, mirror, flip)
        terminal_color = color_for_role[role]
        nodes = []
        for index, (x, y) in enumerate(points):
            if index == 0:
                kind, color = "start", "green"
            elif index == len(points) - 1:
                kind, color = "terminal", terminal_color
            else:
                kind, color = "waypoint", "purple"
            nodes.append({"x": x, "y": y, "kind": kind, "color": color})
        paths.append({"nodes": nodes, "links": [[i, i + 1] for i in range(4)]})
    return {"width": WIDTH, "height": HEIGHT, "paths": paths}


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x in (260, 520):
        draw.line((x, 25, x, int(scene["height"]) - 25), fill=SEPARATOR, width=2)
    for path in scene["paths"]:
        nodes = path["nodes"]
        for left, right in path["links"]:
            a, b = nodes[left], nodes[right]
            draw.line((a["x"], a["y"], b["x"], b["y"]), fill=BLACK, width=LINK_WIDTH)
    for path in scene["paths"]:
        for node in path["nodes"]:
            x, y = node["x"], node["y"]
            draw.ellipse(
                (x - NODE_RADIUS, y - NODE_RADIUS, x + NODE_RADIUS, y + NODE_RADIUS),
                fill=COLORS[node["color"]],
            )
    return image


def analytic_gold(scene: dict) -> str:
    scored = []
    for path in scene["paths"]:
        nodes = path["nodes"]
        lengths = []
        for left, right in path["links"]:
            a, b = nodes[left], nodes[right]
            lengths.append(math.hypot(a["x"] - b["x"], a["y"] - b["y"]))
        score = max(lengths[index + 1] - lengths[index] for index in range(len(lengths) - 1))
        terminal = next(node["color"] for node in nodes if node["kind"] == "terminal")
        scored.append((score, terminal))
    scored.sort(reverse=True)
    return scored[0][1]


def margin(scene: dict) -> float:
    scores = []
    for path in scene["paths"]:
        nodes = path["nodes"]
        lengths = [
            math.hypot(nodes[left]["x"] - nodes[right]["x"], nodes[left]["y"] - nodes[right]["y"])
            for left, right in path["links"]
        ]
        scores.append(max(lengths[index + 1] - lengths[index] for index in range(len(lengths) - 1)))
    scores.sort(reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["paths"] = list(reversed(twin["paths"]))
    return [("reverse_disjoint_draw_order", twin)]
