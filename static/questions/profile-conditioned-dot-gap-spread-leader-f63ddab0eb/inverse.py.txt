"""Independent pixel-only inverse arm for colored dot rows."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

PALETTE = {
    "blue": (25, 118, 210),
    "orange": (230, 126, 34),
    "green": (46, 157, 85),
}


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    seen = np.zeros(mask.shape, dtype=bool)
    height, width = mask.shape
    found = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        points = []
        while queue:
            yy, xx = queue.popleft()
            points.append((yy, xx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = yy + dy, xx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= 80:
            ys = np.fromiter((p[0] for p in points), dtype=float)
            xs = np.fromiter((p[1] for p in points), dtype=float)
            found.append((float(xs.mean()), float(ys.mean()), len(points)))
    return found


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    scores = {}
    for name, rgb in PALETTE.items():
        delta = arr - np.asarray(rgb, dtype=np.int32)
        mask = np.sum(delta * delta, axis=2) < 75 * 75
        components = _components(mask)
        if len(components) != 5:
            raise ValueError(f"expected five {name} dots, found {len(components)}")
        xs = sorted(component[0] for component in components)
        gaps = [b - a for a, b in zip(xs, xs[1:])]
        scores[name] = max(gaps) - min(gaps)
    ordered = sorted(scores, key=scores.get, reverse=True)
    if scores[ordered[0]] - scores[ordered[1]] < 4.0:
        raise ValueError("pixel measurement is too close to the decision boundary")
    return ordered[0]
