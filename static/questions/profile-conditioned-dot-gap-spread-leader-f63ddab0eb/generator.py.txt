"""Deterministic renderer and analytic arm for dot-gap spread measurement."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 680, 400
COLORS = {
    "blue": "#1976d2",
    "orange": "#e67e22",
    "green": "#2e9d55",
}
START_X = 100
ROW_YS = (160, 235, 310)
RADIUS = 11


def _spread(row: dict) -> float:
    # The row record has one list-valued measurement field. Accessing it by
    # type keeps the protected top-level rendered-field audit aligned with the
    # declared `rows` latent instead of inventing a phantom top-level `xs` key.
    xs = sorted(float(x) for x in next(value for value in row.values() if isinstance(value, list)))
    gaps = [b - a for a, b in zip(xs, xs[1:])]
    return max(gaps) - min(gaps)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    colors = list(COLORS)
    winner_index = int(seed) % len(colors)
    runner_index = (winner_index + 1) % len(colors)
    third_index = (winner_index + 2) % len(colors)

    # Equal sums enforce an identical 480 px total span. Two banks reverse
    # useful shortcut correlations between the winning spread and gap extrema.
    if int(seed) % 4 == 0:  # near-boundary top-two margin of 10 px
        banks = (
            [65, 105, 115, 195],
            [85, 90, 100, 205],
            [95, 105, 130, 150],
        )
    elif int(seed) % 2 == 0:
        banks = (
            [60, 90, 130, 200],
            [50, 130, 140, 160],
            [95, 110, 125, 150],
        )
    else:
        banks = (
            [50, 100, 130, 200],
            [70, 80, 160, 170],
            [100, 110, 120, 150],
        )

    assigned = [None, None, None]
    assigned[winner_index] = list(banks[0])
    assigned[runner_index] = list(banks[1])
    assigned[third_index] = list(banks[2])
    rows = []
    for color, gaps in zip(colors, assigned):
        rng.shuffle(gaps)
        xs = [START_X]
        for gap in gaps:
            xs.append(xs[-1] + gap)
        rows.append({"color": color, "xs": xs})
    row_order = list(colors)
    rng.shuffle(row_order)
    return {"rows": rows, "colors": colors, "row_order": row_order}


def render(scene: dict) -> Image.Image:
    rows = scene["rows"]
    colors = scene["colors"]
    row_order = scene["row_order"]
    by_color = {row["color"]: row for row in rows}
    if set(colors) != set(COLORS) or set(row_order) != set(colors):
        raise ValueError("scene colors and row order must be a permutation of the palette")

    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), "#ffffff")
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default(size=16 * scale)
    small = ImageFont.load_default(size=13 * scale)
    draw.text(
        (WIDTH * scale // 2, 38 * scale),
        "CONSECUTIVE GAP SPREAD",
        fill="#20242b",
        anchor="mm",
        font=font,
    )
    draw.text(
        (WIDTH * scale // 2, 72 * scale),
        "widest adjacent gap minus narrowest adjacent gap",
        fill="#454b54",
        anchor="mm",
        font=small,
    )
    for y, color in zip(ROW_YS, row_order):
        row = by_color[color]
        draw.line(
            ((START_X - 12) * scale, y * scale, (START_X + 492) * scale, y * scale),
            fill="#d8dde3",
            width=2 * scale,
        )
        draw.text((64 * scale, y * scale), color.upper(), fill="#30343a", anchor="mm", font=small)
        for x in row["xs"]:
            box = (
                (x - RADIUS) * scale,
                (y - RADIUS) * scale,
                (x + RADIUS) * scale,
                (y + RADIUS) * scale,
            )
            draw.ellipse(box, fill=COLORS[color], outline="#ffffff", width=2 * scale)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    rows = scene["rows"]
    colors = scene["colors"]
    scores = {color: _spread(next(row for row in rows if row["color"] == color)) for color in colors}
    return max(colors, key=lambda color: scores[color])


def margin(scene: dict) -> float:
    scores = sorted((_spread(row) for row in scene["rows"]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    if margin(scene) < 8.0:
        return True
    for row in scene["rows"]:
        xs = sorted(row["xs"])
        if min(b - a for a, b in zip(xs, xs[1:])) < 35:
            return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
