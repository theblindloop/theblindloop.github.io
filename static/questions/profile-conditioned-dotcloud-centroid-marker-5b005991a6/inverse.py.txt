"""Independent pixel-only inverse arm for the dot-cloud centroid-marker world.

Recovers which colored dot sits at the emergent center of mass of the whole dot
constellation from the final PNG alone. It never sees the latent scene, analytic
gold, or renderer code. It segments each colored dot by palette, takes each
dot's centroid, computes the global centroid (mean) of all the dot centres, then
selects the color whose dot centre sits nearest that global balance point.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

DOT_COLORS = {
    "red": (225, 60, 60),
    "green": (60, 185, 85),
    "blue": (55, 95, 230),
    "purple": (180, 60, 205),
    "orange": (235, 140, 40),
    "teal": (30, 190, 195),
}
DOT_NAMES = ["red", "green", "blue", "purple", "orange", "teal"]

_TH = 40


def _mask(arr, rgb):
    r, g, b = arr[..., 0], arr[..., 1], arr[..., 2]
    return (
        (np.abs(r - rgb[0]) < _TH) & (np.abs(g - rgb[1]) < _TH) & (np.abs(b - rgb[2]) < _TH)
    )


def _dot_center(mask: np.ndarray):
    coords = np.argwhere(mask)
    if len(coords) < 20:
        return None
    return (float(coords[:, 1].mean()), float(coords[:, 0].mean()))


def decision_from_image(image: Image.Image) -> str:
    """The color name of the dot sitting on the whole constellation's centroid."""
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)

    centers = {}
    for color in DOT_NAMES:
        center = _dot_center(_mask(arr, DOT_COLORS[color]))
        if center is not None:
            centers[color] = center
    if not centers:
        return DOT_NAMES[0]

    gx = sum(cx for cx, _ in centers.values()) / len(centers)
    gy = sum(cy for _, cy in centers.values()) / len(centers)

    best = None
    best_distsq = None
    for color, (cx, cy) in centers.items():
        dsq = (cx - gx) * (cx - gx) + (cy - gy) * (cy - gy)
        if best is None or dsq < best_distsq:
            best = color
            best_distsq = dsq
    if best is None:
        return DOT_NAMES[0]
    return best
