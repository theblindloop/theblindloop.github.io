"""Dot-cloud centroid-marker world: which colored dot is at the constellation's center.

A scattered constellation of equal-sized colored dots is drawn on a white canvas.
Looking at the whole cloud, its emergent center of mass (the centroid of all the
dot centres) is a single global balance point. Exactly one colored dot is placed
on that centroid; every other dot is farther away. The decision is the color of
that centrally placed dot.

The global Gestalt structure is the constellation's barycenter/balance point, a
property that only arises from the whole set of marks together. The renderer owns
latent-scene sampling, rasterization, and the structural analytic gold. The
independent pixel-only inverse arm (``oracle.py``) recovers the same decision
from the final PNG alone: it segments every colored dot, takes each dot's
centroid, computes the global centroid of the whole constellation, and selects
the color whose dot sits nearest that balance point.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (255, 255, 255)

DOT_RADIUS = 10
_CX = WIDTH / 2.0
_CY = HEIGHT / 2.0

# Six distinct dot colors; each scene uses all six exactly once.
DOT_COLORS = {
    "red": (225, 60, 60),
    "green": (60, 185, 85),
    "blue": (55, 95, 230),
    "purple": (180, 60, 205),
    "orange": (235, 140, 40),
    "teal": (30, 190, 195),
}
DOT_NAMES = ["red", "green", "blue", "purple", "orange", "teal"]

N_DOTS = len(DOT_NAMES)          # 6 dots total
N_DISTRACTORS = N_DOTS - 1       # 5 distractor dots

# Distractors are sampled on a ring around the canvas centre.
_R_MIN = 105.0
_R_MAX = 165.0

# Quarantine band on the decision margin (px from the centroid to the nearest
# distractor dot; the true dot is at distance ~0).
QUAR_MIN = 40.0

# Deterministic rejection floor: keep the nearest distractor at least this far
# from the centroid so the pixel oracle never faces a degenerate ambiguity.
MARGIN_FLOOR = 25.0


def _unit(angle: float) -> tuple[float, float]:
    return (math.cos(angle), math.sin(angle))


def sample_scene(seed: int) -> dict:
    """Sample one latent scene deterministically from ``seed``.

    Five distractor dots are placed on a ring around the canvas centre; their
    mean position M is computed; the true dot (of color ``true_color``) is placed
    exactly at M. The global centroid of all six dots therefore equals M, so the
    true-colored dot sits precisely on the whole constellation's center of mass.
    A deterministic rejection loop keeps the nearest distractor a bounded margin
    from M so the pixel oracle never faces a degenerate ambiguity.
    """
    rng = np.random.default_rng(seed)

    colors = list(DOT_NAMES)
    rng.shuffle(colors)
    true_color = colors[0]
    distractor_colors = colors[1:]

    while True:
        distractors = []
        for _ in range(N_DISTRACTORS):
            angle = float(rng.uniform(0.0, 2.0 * math.pi))
            radius = float(rng.uniform(_R_MIN, _R_MAX))
            ux, uy = _unit(angle)
            distractors.append([_CX + radius * ux, _CY + radius * uy])
        mx = sum(p[0] for p in distractors) / N_DISTRACTORS
        my = sum(p[1] for p in distractors) / N_DISTRACTORS
        true_position = [float(mx), float(my)]
        nearest = min(math.hypot(px - mx, py - my) for (px, py) in distractors)
        if nearest >= MARGIN_FLOOR:
            break

    return {
        "true_color": true_color,
        "distractor_colors": distractor_colors,
        "distractors": distractors,
        "true_position": true_position,
    }


def render(scene: dict) -> Image.Image:
    """Rasterize the latent scene (a deterministic RGB PNG)."""
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    d = ImageDraw.Draw(img)

    # Distractor dots first, then the true dot on top at the centroid.
    for (px, py), color in zip(scene["distractors"], scene["distractor_colors"]):
        d.ellipse(
            [px - DOT_RADIUS, py - DOT_RADIUS, px + DOT_RADIUS, py + DOT_RADIUS],
            fill=DOT_COLORS[color],
        )
    tx, ty = scene["true_position"]
    d.ellipse(
        [tx - DOT_RADIUS, ty - DOT_RADIUS, tx + DOT_RADIUS, ty + DOT_RADIUS],
        fill=DOT_COLORS[scene["true_color"]],
    )
    return img


def _marker_center(mask: np.ndarray) -> tuple[float, float] | None:
    coords = np.argwhere(mask)
    if len(coords) < 20:
        return None
    return (float(coords[:, 1].mean()), float(coords[:, 0].mean()))


def analytic_gold(scene: dict) -> str:
    """Structural gold: the color name of the dot placed on the global centroid."""
    return scene["true_color"]


def margin(scene: dict) -> float:
    """Decision margin: distance from the centroid to the nearest distractor dot."""
    tx, ty = scene["true_position"]
    dists = [math.hypot(px - tx, py - ty) for (px, py) in scene["distractors"]]
    return float(min(dists))


def is_quarantined(scene: dict) -> bool:
    """Quarantine scenes whose nearest distractor comes too close to the centroid."""
    return margin(scene) < QUAR_MIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The scene is injectively rendered; no latent aliases are declared."""
    return []
