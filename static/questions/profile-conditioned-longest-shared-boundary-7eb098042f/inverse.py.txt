"""Independent pixel-only inverse arm for the longest-shared-boundary world.

receives ONLY the final PNG. It never imports or reads the renderer, the
analytic scene, prompts, or renderer state. It recovers the decision by:

1. segmenting the three colored filled regions by color;
2. for each colour pair, dilating both region masks by a small radius, so that
   a shared boundary becomes a broad overlap band whose pixel count is
   proportional to the arc length of the contact;
3. the pair that shares the LONGEST common boundary is the pair whose dilated
   masks have the LARGEST overlap pixel count.

The answer is the pair of coloured regions that shares the longest stretch of
common boundary.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "longest_shared_boundary-oracle-0.4.0"

_WHITE_THRESHOLD = 240
_COLOR_TOLERANCE = 95.0
_DILATE_RADIUS = 8
_PAIRS = [("red", "green"), ("red", "blue"), ("green", "blue")]
_RANK = {"red": 0, "green": 1, "blue": 2}
_COLORS = {
    "red": np.array([230.0, 45.0, 45.0]),
    "green": np.array([50.0, 185.0, 60.0]),
    "blue": np.array([50.0, 90.0, 230.0]),
}


def _color_mask(arr: np.ndarray, target: np.ndarray, tolerance: float) -> np.ndarray:
    diff = arr.astype(np.float32) - target
    return np.sqrt((diff * diff).sum(axis=-1)) < tolerance


def _dilate(mask: np.ndarray, radius: int) -> np.ndarray:
    out = np.zeros_like(mask)
    h, w = mask.shape
    for dy in range(-radius, radius + 1):
        for dx in range(-radius, radius + 1):
            if dx * dx + dy * dy > radius * radius:
                continue
            shifted = np.zeros_like(mask)
            src_y0 = max(0, dy)
            src_y1 = min(h, h + dy)
            src_x0 = max(0, dx)
            src_x1 = min(w, w + dx)
            dst_y0 = max(0, -dy)
            dst_y1 = dst_y0 + (src_y1 - src_y0)
            dst_x0 = max(0, -dx)
            dst_x1 = dst_x0 + (src_x1 - src_x0)
            patch = mask[src_y0:src_y1, src_x0:src_x1]
            out[dst_y0:dst_y1, dst_x0:dst_x1] |= patch
    return out


def _shared_pixels(arr: np.ndarray, c1: str, c2: str) -> int:
    m1 = _color_mask(arr, _COLORS[c1], _COLOR_TOLERANCE)
    m2 = _color_mask(arr, _COLORS[c2], _COLOR_TOLERANCE)
    if not m1.any() or not m2.any():
        return 0
    d1 = _dilate(m1, _DILATE_RADIUS)
    d2 = _dilate(m2, _DILATE_RADIUS)
    return int((d1 & d2).sum())


def decision_from_image(image: Image.Image) -> str:
    """The pair of coloured regions sharing the longest common boundary, from
    the final pixels alone."""
    arr = np.asarray(image.convert("RGB")).astype(np.float32)
    best = 0
    best_pair = _PAIRS[0]
    for c1, c2 in _PAIRS:
        count = _shared_pixels(arr, c1, c2)
        if count > best:
            best = count
            best_pair = "-".join(sorted((c1, c2), key=lambda c: _RANK[c]))
    return best_pair
