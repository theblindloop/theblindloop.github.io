"""Longest-shared-boundary world renderer.

Three colored filled regions (red, green, blue) are drawn on a white canvas.
One region is a wide base block near the bottom. The other two coloured regions
sit directly on top of the base block, side by side with a clear white gap
between them, so BOTH of them share a boundary with the base (their bottom edge
is flush with the base top edge). The two perches have independent widths and
independent heights, and the baseline widths are chosen so that one perch's
contact with the base is clearly LONGER than the other's.

Decision : which pair of colored regions (of red/green/blue) shares the LONGEST
common boundary (the longest stretch where two regions touch each other).

The plain dict scene keeps the AST reachability audit free of side effects.

    z = {
        "width": int,
        "height": int,
        "regions": [ {"x0": int, "y0": int, "x1": int, "y1": int, "color": str}, ... ],
    }
"""

from __future__ import annotations

import math
import random
from collections.abc import Iterator

from PIL import Image, ImageDraw

RENDERER_VERSION = "longest_shared_boundary-0.4.0"

WHITE = (255, 255, 255)
_COLOR_RGB = {"red": (230, 45, 45), "green": (50, 185, 60), "blue": (50, 90, 230)}

PAIRS = [("red", "green"), ("red", "blue"), ("green", "blue")]
_RANK = {"red": 0, "green": 1, "blue": 2}

# Canvas and layout constants.
_CANVAS = (600, 420)
_BASE = (40, 240, 560, 400)  # base block: x0,y0,x1,y1 (wide, bottom)
_LEFT_X0 = 60


def _pair_label(c1: str, c2: str) -> str:
    return "-".join(sorted((c1, c2), key=lambda c: _RANK[c]))


def sample_scene(rng: random.Random, width: int = _CANVAS[0], height: int = _CANVAS[1]) -> dict:
    """Sample a scene: one base block plus two same-bottom perches whose
    contacts with the base have clearly different lengths."""
    colors = ["red", "green", "blue"]
    base_color = rng.choice(colors)
    others = [c for c in colors if c != base_color]
    rng.shuffle(others)
    long_color, short_color = others[0], others[1]

    w_long = rng.randint(170, 230)
    w_short = rng.randint(70, 120)
    if w_long - w_short < 50:
        w_short = w_long - rng.randint(55, 70)
    gap = rng.randint(40, 70)
    h_long = rng.randint(80, 180)
    h_short = rng.randint(80, 180)

    if rng.random() < 0.5:
        left_color, right_color = long_color, short_color
        left_w, right_w = w_long, w_short
        left_h, right_h = h_long, h_short
    else:
        left_color, right_color = short_color, long_color
        left_w, right_w = w_short, w_long
        left_h, right_h = h_short, h_long

    left_x0 = _LEFT_X0
    right_x0 = left_x0 + left_w + gap
    base_x0, base_y0, base_x1, base_y1 = _BASE

    regions = [
        {"x0": base_x0, "y0": base_y0, "x1": base_x1, "y1": base_y1, "color": base_color},
        {
            "x0": left_x0,
            "y0": base_y0 - left_h,
            "x1": left_x0 + left_w,
            "y1": base_y0,
            "color": left_color,
        },
        {
            "x0": right_x0,
            "y0": base_y0 - right_h,
            "x1": right_x0 + right_w,
            "y1": base_y0,
            "color": right_color,
        },
    ]
    return {"width": width, "height": height, "regions": regions}


def _shared_boundary(a: dict, b: dict) -> float:
    """Length of the line segment along which two axis-aligned regions touch.

    Two regions touch along an edge when their projections overlap with positive
    length on one axis (the tangent direction) and their opposite extents are
    exactly flush on the other axis.
    """
    dx = min(a["x1"], b["x1"]) - max(a["x0"], b["x0"])
    dy = min(a["y1"], b["y1"]) - max(a["y0"], b["y0"])
    length = 0.0
    if dx > 0:
        # horizontal flush (a.y1 == b.y0 or a.y0 == b.y1): shared edge length = dx
        if a["y1"] == b["y0"] or a["y0"] == b["y1"]:
            length = max(length, float(dx))
    if dy > 0:
        if a["x1"] == b["x0"] or a["x0"] == b["x1"]:
            length = max(length, float(dy))
    return length


def analytic_gold(scene: dict) -> str:
    """The pair of colored regions sharing the longest common boundary."""
    regions = scene["regions"]
    best = 0.0
    best_pair = _pair_label(*PAIRS[0])
    for i in range(len(regions)):
        a = regions[i]
        for j in range(i + 1, len(regions)):
            b = regions[j]
            dx = min(a["x1"], b["x1"]) - max(a["x0"], b["x0"])
            dy = min(a["y1"], b["y1"]) - max(a["y0"], b["y0"])
            length = 0.0
            if dx > 0 and (a["y1"] == b["y0"] or a["y0"] == b["y1"]):
                length = float(dx)
            elif dy > 0 and (a["x1"] == b["x0"] or a["x0"] == b["x1"]):
                length = float(dy)
            if length > best:
                best = length
                best_pair = _pair_label(a["color"], b["color"])
    return best_pair


def _boundary_lengths(scene: dict) -> list[tuple[str, float]]:
    regions = scene["regions"]
    out = []
    for i in range(len(regions)):
        for j in range(i + 1, len(regions)):
            out.append(
                (_pair_label(regions[i]["color"], regions[j]["color"]),
                 _shared_boundary(regions[i], regions[j]))
            )
    return out


def margin(scene: dict) -> float:
    """Runner-up decision margin: winner shared-boundary length minus the
    second-longest shared-boundary length, in pixels."""
    lengths = sorted((v for _, v in _boundary_lengths(scene)), reverse=True)
    if len(lengths) < 2:
        return 0.0
    return float(lengths[0] - lengths[1])


def _perch_gap(scene: dict) -> float:
    """Horizontal gap between the two non-base (perch) regions."""
    regions = scene["regions"]
    perches = [r for r in regions if _perch_is_top(r, regions)]
    if len(perches) != 2:
        return float("inf")
    p, q = perches
    return float(max(p["x0"] - q["x1"], q["x0"] - p["x1"], 0))


def _perch_is_top(r: dict, regions: list[dict]) -> bool:
    base = next(iter(regions))
    # The base is the region with the largest area (the wide bottom block).
    a = (base["x1"] - base["x0"]) * (base["y1"] - base["y0"])
    b = (r["x1"] - r["x0"]) * (r["y1"] - r["y0"])
    return b < a


def is_quarantined(scene: dict) -> bool:
    """Quarantine scenes whose decision boundary is untrustworthy at raster
    scale: too-small a margin, perches that come close enough to be fused by
    dilation, or a perch that overhangs the base block."""
    if margin(scene) < 30.0:
        return True
    if _perch_gap(scene) < 36.0:
        return True
    regions = scene["regions"]
    base = next(iter(regions))
    for r in regions:
        if _perch_is_top(r, regions):
            if r["x1"] > base["x1"] - 15.0 or r["x0"] < base["x0"] + 15.0:
                return True
    return False


def render(scene: dict, supersample: int = 4) -> Image.Image:
    """Render R(z) deterministically as a pure function of the scene dict."""
    width = int(scene["width"])
    height = int(scene["height"])
    regions = scene["regions"]
    ss = supersample
    img = Image.new("RGB", (width * ss, height * ss), WHITE)
    draw = ImageDraw.Draw(img)
    for region in regions:
        x0 = int(round(region["x0"] * ss))
        y0 = int(round(region["y0"] * ss))
        x1 = int(round(region["x1"] * ss))
        y1 = int(round(region["y1"] * ss))
        draw.rectangle([x0, y0, x1, y1], fill=_COLOR_RGB[region["color"]])
    if ss > 1:
        img = img.resize((width, height), Image.LANCZOS)
    return img


def latent_symmetries(scene: dict) -> Iterator[dict]:
    """No nontrivial pixel-identical transform: every field directly redraws.

    Which color is the base and which perch is the longer contact are
    label-transforming relabels (they change the pixels) that the decision
    tracks exactly; declared not_applicable.
    """
    return iter(())
