"""Scene sampling, rasterisation, analytic gold, margin and quarantine.

World: a disc-and-bar network drawn on a jittered 4x4 lattice.  Exactly one
junction is circled by a black ring.  The decision is how many separate pieces
the rest of the network falls into once that ringed junction and every bar
touching it are erased.

Counterbalancing is structural: every scene has the same number of junctions,
the same number of bars, and a ringed junction of degree exactly four, so total
ink, junction count, bar count and the ringed junction's own neighbourhood are
constant across all three answer classes.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "hub_removal_pieces-renderer-0.1.0"

CANVAS = 640
COLS = ROWS = 4
SPACING = 150
ORIGIN = 95
JITTER = 34

NODE_RADIUS = 16
STROKE_WIDTH = 9
RING_RADIUS = 31
RING_WIDTH = 4

BACKGROUND = (255, 255, 255)
NODE_COLOR = (58, 58, 58)
STROKE_COLOR = (150, 150, 150)
RING_COLOR = (0, 0, 0)

N_NODES = COLS * ROWS
N_EDGES = 18
MIN_CLEARANCE = 48.0

DECISIONS = ("1", "2", "3")


def _cell(index: int) -> tuple[int, int]:
    return index % COLS, index // COLS


def _lattice_edges() -> list[tuple[int, int]]:
    edges = []
    for i in range(N_NODES):
        cx, cy = _cell(i)
        if cx + 1 < COLS:
            edges.append((i, i + 1))
        if cy + 1 < ROWS:
            edges.append((i, i + COLS))
    return edges


LATTICE_EDGES = _lattice_edges()
INTERIOR = [i for i in range(N_NODES) if 0 < _cell(i)[0] < COLS - 1 and 0 < _cell(i)[1] < ROWS - 1]


def _components(n: int, edges, removed: int | None = None) -> int:
    alive = [v for v in range(n) if v != removed]
    adj = {v: [] for v in alive}
    for a, b in edges:
        if a == removed or b == removed:
            continue
        adj[a].append(b)
        adj[b].append(a)
    seen: set[int] = set()
    count = 0
    for v in alive:
        if v in seen:
            continue
        count += 1
        stack = [v]
        seen.add(v)
        while stack:
            for w in adj[stack.pop()]:
                if w not in seen:
                    seen.add(w)
                    stack.append(w)
    return count


def _degrees(n: int, edges) -> dict:
    deg = {v: 0 for v in range(n)}
    for a, b in edges:
        deg[a] += 1
        deg[b] += 1
    return deg


def sample_scene(seed: int) -> dict:
    """Sample one scene.  The answer class is pinned by the seed itself so that
    consecutive seeds cycle through 1, 2 and 3 pieces in equal numbers."""
    rng = random.Random((seed * 1_000_003) ^ 0x5EED)
    target = 1 + (seed % 3)

    hub = None
    chosen = None
    for _ in range(20000):
        cand_hub = rng.choice(INTERIOR)
        hub_edges = [e for e in LATTICE_EDGES if cand_hub in e]
        if len(hub_edges) != 4:
            continue
        rest = [e for e in LATTICE_EDGES if cand_hub not in e]
        picked = rng.sample(rest, N_EDGES - 4)
        edges = hub_edges + picked
        if _components(N_NODES, edges) != 1:
            continue
        if _components(N_NODES, edges, removed=cand_hub) != target:
            continue
        hub, chosen = cand_hub, edges
        break
    if hub is None:  # pragma: no cover - the search always succeeds in practice
        raise RuntimeError(f"no scene found for seed {seed}")

    chosen = sorted(tuple(sorted(e)) for e in chosen)
    nodes = []
    for i in range(N_NODES):
        cx, cy = _cell(i)
        nodes.append(
            {
                "x": float(ORIGIN + cx * SPACING + rng.randint(-JITTER, JITTER)),
                "y": float(ORIGIN + cy * SPACING + rng.randint(-JITTER, JITTER)),
            }
        )

    return {
        "canvas": CANVAS,
        "nodes": nodes,
        "edges": [list(e) for e in chosen],
        "hub": hub,
        "node_radius": NODE_RADIUS,
        "stroke_width": STROKE_WIDTH,
        "ring_radius": RING_RADIUS,
        "background": list(BACKGROUND),
    }


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (scene["canvas"], scene["canvas"]), tuple(scene["background"]))
    d = ImageDraw.Draw(img)
    nodes = scene["nodes"]
    for a, b in scene["edges"]:
        d.line(
            [nodes[a]["x"], nodes[a]["y"], nodes[b]["x"], nodes[b]["y"]],
            fill=STROKE_COLOR,
            width=scene["stroke_width"],
        )
    r = scene["node_radius"]
    for node in nodes:
        d.ellipse(
            [node["x"] - r, node["y"] - r, node["x"] + r, node["y"] + r],
            fill=NODE_COLOR,
        )
    hub = nodes[scene["hub"]]
    rr = scene["ring_radius"]
    d.ellipse(
        [hub["x"] - rr, hub["y"] - rr, hub["x"] + rr, hub["y"] + rr],
        outline=RING_COLOR,
        width=RING_WIDTH,
    )
    return img


def analytic_gold(scene: dict) -> str:
    pieces = _components(len(scene["nodes"]), scene["edges"], removed=scene["hub"])
    return str(pieces)


def _point_segment_distance(px, py, ax, ay, bx, by) -> float:
    vx, vy = bx - ax, by - ay
    denom = vx * vx + vy * vy
    t = 0.0 if denom == 0 else max(0.0, min(1.0, ((px - ax) * vx + (py - ay) * vy) / denom))
    return math.hypot(px - (ax + t * vx), py - (ay + t * vy))


def margin(scene: dict) -> float:
    """Reading margin in pixels: the tightest separation anywhere in the raster
    between two structures the pixel arm must keep apart.  It is the runner-up
    margin for the decision because the count of pieces is read off the recovered
    graph, and only these separations can make that graph ambiguous."""
    nodes = scene["nodes"]
    r = scene["node_radius"]
    half = scene["stroke_width"] / 2.0
    worst = float(scene["canvas"])
    for i, a in enumerate(nodes):
        for j, b in enumerate(nodes):
            if j <= i:
                continue
            worst = min(worst, math.hypot(a["x"] - b["x"], a["y"] - b["y"]) - 2 * r)
    for i, node in enumerate(nodes):
        for a, b in scene["edges"]:
            if i in (a, b):
                continue
            gap = _point_segment_distance(
                node["x"], node["y"], nodes[a]["x"], nodes[a]["y"], nodes[b]["x"], nodes[b]["y"]
            )
            worst = min(worst, gap - r - half)
    hub = nodes[scene["hub"]]
    for i, node in enumerate(nodes):
        if i == scene["hub"]:
            continue
        d = math.hypot(hub["x"] - node["x"], hub["y"] - node["y"])
        worst = min(worst, abs(d - scene["ring_radius"]) - r)
    return round(float(worst), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < MIN_CLEARANCE


def latent_symmetries(scene: dict) -> list:
    """No latent aliases are declared: every scene field that the analytic gold
    reads (nodes, edges, hub) is painted in full into the raster."""
    return []
