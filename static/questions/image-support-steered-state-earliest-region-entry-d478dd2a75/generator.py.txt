"""Deterministic renderer and analytic gold for ordered region-entry cards."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "state-earliest-region-entry-0.1.0"
WIDTH = 768
HEIGHT = 512
SUPERSAMPLE = 4

PANEL_WIDTH = 68
PANEL_HEIGHT = 84
PANEL_X_STEP = 288
PANEL_Y_STEP = 220
PANEL_X_CHOICES = (198, 200, 202, 204)
PANEL_Y_CHOICES = (130, 132, 134)

BACKGROUND = (250, 252, 253)
PANEL_FILL = (236, 242, 245)
ZONE_FILL = (211, 226, 233)
ZONE_EDGE = (150, 171, 182)
PANEL_EDGE = (50, 69, 80)
TEXT = (45, 63, 73)
TOKEN_EDGE = (42, 58, 68)
TOKEN_PALETTE = (
    (221, 83, 86),
    (57, 132, 208),
    (66, 166, 111),
    (220, 158, 54),
)
ROW_LABELS = ("A", "B", "C", "D")

FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype(FONT_PATH, size)
    except OSError:
        return ImageFont.load_default()


def _box(values: tuple[float, ...], scale: int) -> list[int]:
    return [int(round(value * scale)) for value in values]


def _center_text(
    draw: ImageDraw.ImageDraw,
    text: str,
    center_x: float,
    top_y: float,
    font: ImageFont.FreeTypeFont | ImageFont.ImageFont,
    fill: tuple[int, int, int],
    scale: int,
) -> None:
    bounds = draw.textbbox((0, 0), text, font=font)
    width = bounds[2] - bounds[0]
    draw.text(
        (int(round(center_x * scale - width / 2)), int(round(top_y * scale))),
        text,
        fill=fill,
        font=font,
    )


def _entry_candidates(states: list[list[bool]]) -> list[tuple[int, int]]:
    candidates: list[tuple[int, int]] = []
    for label in range(4):
        if not states[3][label]:
            continue
        entries = [
            step
            for step in range(1, 4)
            if not states[step - 1][label] and states[step][label]
        ]
        if entries:
            candidates.append((min(entries), label))
    return sorted(candidates)


def _valid_states(value: Any) -> bool:
    return (
        isinstance(value, list)
        and len(value) == 4
        and all(
            isinstance(row, list)
            and len(row) == 4
            and all(isinstance(item, bool) for item in row)
            for row in value
        )
    )


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced scene with one unambiguous earliest surviving entry."""

    rng = random.Random(int(seed))
    target = (int(seed) * 5 + rng.randrange(4)) % 4
    target_transition = 1 + rng.randrange(3)
    states = [[False for _ in range(4)] for _ in range(4)]

    for card in range(4):
        states[card][target] = card >= target_transition

    others = [index for index in range(4) if index != target]
    rng.shuffle(others)
    first, second, third = others

    def track(entry: int | None, exit_at: int | None, initial: bool = False) -> list[bool]:
        values = [bool(initial)] * 4
        if entry is not None:
            for card in range(entry, 4):
                values[card] = True
        if exit_at is not None:
            for card in range(exit_at, 4):
                values[card] = False
        return values

    if target_transition == 1:
        tracks = {
            first: track(2, None),
            second: track(3, None),
            third: track(None, None, initial=rng.randrange(2) == 1),
        }
    elif target_transition == 2:
        tracks = {
            first: track(1, 3),
            second: track(3, None),
            third: track(None, None, initial=rng.randrange(2) == 1),
        }
    else:
        tracks = {
            first: track(1, 2),
            second: track(2, 3),
            third: track(None, None, initial=rng.randrange(2) == 1),
        }

    for label, values in tracks.items():
        for card, inside in enumerate(values):
            states[card][label] = inside

    palette = [tuple(color) for color in TOKEN_PALETTE]
    rng.shuffle(palette)
    return {
        "states": states,
        "palette": [list(color) for color in palette],
        "origin": [rng.choice(PANEL_X_CHOICES), rng.choice(PANEL_Y_CHOICES)],
    }


def _panel_origins(origin_x: float, origin_y: float) -> tuple[tuple[float, float], ...]:
    return (
        (origin_x, origin_y),
        (origin_x + PANEL_X_STEP, origin_y),
        (origin_x, origin_y + PANEL_Y_STEP),
        (origin_x + PANEL_X_STEP, origin_y + PANEL_Y_STEP),
    )


def render(scene: dict[str, Any]) -> Image.Image:
    """Render four separated snapshots with a visible shaded inside zone."""

    scale = SUPERSAMPLE
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    small_font = _font(8 * scale)
    row_font = _font(7 * scale)
    origin_x, origin_y = (float(value) for value in scene["origin"])
    palette = [tuple(int(value) for value in color) for color in scene["palette"]]
    states = scene["states"]

    for card_index, (left, top) in enumerate(_panel_origins(origin_x, origin_y), start=1):
        draw.rectangle(
            _box((left, top, left + PANEL_WIDTH, top + PANEL_HEIGHT), scale),
            fill=PANEL_FILL,
            outline=PANEL_EDGE,
            width=2 * scale,
        )
        zone_left = left + 35
        draw.rectangle(
            _box((zone_left, top + 18, left + PANEL_WIDTH - 5, top + PANEL_HEIGHT - 4), scale),
            fill=ZONE_FILL,
        )
        draw.line(
            [(int(round(zone_left * scale)), int(round((top + 18) * scale))),
             (int(round(zone_left * scale)), int(round((top + PANEL_HEIGHT - 4) * scale)))],
            fill=ZONE_EDGE,
            width=max(1, scale),
        )
        _center_text(draw, str(card_index), left + PANEL_WIDTH / 2, top + 3, small_font, TEXT, scale)

        for label, inside in enumerate(states[card_index - 1]):
            center_y = top + 23 + label * 14
            _center_text(draw, ROW_LABELS[label], left + 8, center_y - 5, row_font, TEXT, scale)
            center_x = left + (51 if inside else 21)
            color = palette[(label + card_index - 1) % len(palette)]
            draw.ellipse(
                _box((center_x - 5.5, center_y - 5.5, center_x + 5.5, center_y + 5.5), scale),
                fill=color,
                outline=TOKEN_EDGE,
                width=max(1, scale),
            )

    if scale > 1:
        image = image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the earliest entry among objects inside on the last card."""

    states = scene.get("states")
    if not _valid_states(states):
        raise ValueError("scene does not contain a four-by-four boolean state matrix")
    candidates = _entry_candidates(states)
    if not candidates:
        raise ValueError("scene has no final-inside object with an entry")
    return ROW_LABELS[candidates[0][1]]


def margin(scene: dict[str, Any]) -> float:
    """One-sided temporal margin: gap to the runner-up qualifying entry."""

    if is_quarantined(scene):
        return 0.0
    candidates = _entry_candidates(scene["states"])
    if len(candidates) < 2:
        return 1.0
    return float(candidates[1][0] - candidates[0][0])


def is_quarantined(scene: dict[str, Any]) -> bool:
    states = scene.get("states")
    if not _valid_states(states):
        return True
    candidates = _entry_candidates(states)
    if not candidates:
        return True
    return len([item for item in candidates if item[0] == candidates[0][0]]) != 1


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No latent label alias is claimed for this explicitly labeled encoding."""

    return []
