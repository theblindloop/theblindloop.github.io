"""Independent scene-free inverse arm for spoke-angle correspondence."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

PALETTE = {
    "red": np.array((213, 75, 75), dtype=np.int16),
    "blue": np.array((57, 118, 197), dtype=np.int16),
    "gold": np.array((214, 161, 29), dtype=np.int16),
    "purple": np.array((132, 85, 184), dtype=np.int16),
    "teal": np.array((0, 140, 145), dtype=np.int16),
    "dark": np.array((38, 50, 56), dtype=np.int16),
    "magenta": np.array((193, 47, 135), dtype=np.int16),
}
TAGS = ("red", "blue", "gold", "purple")


def _mask(arr: np.ndarray, name: str, tolerance: int = 12) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - PALETTE[name]), axis=2) <= tolerance


def _components(mask: np.ndarray) -> list[tuple[int, float, float, int, int, int, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    ys, xs = np.nonzero(mask)
    for sy, sx in zip(ys.tolist(), xs.tolist()):
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        area = sumx = sumy = 0
        minx = maxx = sx
        miny = maxy = sy
        while queue:
            y, x = queue.popleft()
            area += 1
            sumx += x
            sumy += y
            minx, maxx = min(minx, x), max(maxx, x)
            miny, maxy = min(miny, y), max(maxy, y)
            for ny in range(max(0, y - 1), min(h, y + 2)):
                for nx in range(max(0, x - 1), min(w, x + 2)):
                    if mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        output.append((area, sumx / area, sumy / area, minx, miny, maxx, maxy))
    return output


def _distance(a: tuple[float, float], b: tuple[float, float]) -> float:
    return math.hypot(a[0] - b[0], a[1] - b[1])


def _angle_signature(hub: tuple[float, float], satellites: list[tuple[float, float]]) -> float:
    nearby = sorted(satellites, key=lambda point: _distance(point, hub))
    nearby = [point for point in nearby if 45.0 <= _distance(point, hub) <= 82.0][:2]
    if len(nearby) != 2:
        raise ValueError("hub does not have exactly two nearby satellites")
    vectors = [(point[0] - hub[0], point[1] - hub[1]) for point in nearby]
    denominator = math.hypot(*vectors[0]) * math.hypot(*vectors[1])
    cosine = max(-1.0, min(1.0, (vectors[0][0] * vectors[1][0] + vectors[0][1] * vectors[1][1]) / denominator))
    return math.degrees(math.acos(cosine))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    if arr.ndim != 3 or arr.shape[0] < 600 or arr.shape[1] < 900:
        raise ValueError("unexpected raster")
    comps = {name: _components(_mask(arr, name)) for name in PALETTE}
    hubs = [(item[1], item[2]) for item in comps["dark"] if 1300 <= item[0] <= 2400]
    satellites = [(item[1], item[2]) for item in comps["teal"] if 300 <= item[0] <= 700]
    if len(hubs) != 8 or len(satellites) != 16:
        raise ValueError("expected eight hubs and sixteen satellites")

    tag_centers = {}
    for tag in TAGS:
        candidates = [(item[1], item[2]) for item in comps[tag] if 1200 <= item[0] <= 2400]
        if len(candidates) != 1:
            raise ValueError(f"expected one {tag} reference tag")
        tag_centers[tag] = candidates[0]

    reference_hubs = {}
    unused = set(range(len(hubs)))
    for tag, center in tag_centers.items():
        index = min(unused, key=lambda i: _distance(hubs[i], center))
        if _distance(hubs[index], center) > 145.0:
            raise ValueError("tag is not visibly bound to a hub")
        reference_hubs[tag] = hubs[index]
        unused.remove(index)
    target_hubs = [hubs[index] for index in unused]
    if len(target_hubs) != 4:
        raise ValueError("reference/target partition failed")

    rings = [(item[1], item[2]) for item in comps["magenta"] if 800 <= item[0] <= 1900]
    if len(rings) != 1:
        raise ValueError("expected one target marker ring")
    marked = min(target_hubs, key=lambda hub: _distance(hub, rings[0]))
    if _distance(marked, rings[0]) > 8.0:
        raise ValueError("marker does not center on one target hub")

    target_angle = _angle_signature(marked, satellites)
    differences = sorted(
        (abs(_angle_signature(hub, satellites) - target_angle), tag)
        for tag, hub in reference_hubs.items()
    )
    if differences[0][0] > 8.0 or differences[1][0] - differences[0][0] < 25.0:
        raise ValueError("pixel angles do not give a unique correspondence")
    return differences[0][1]
