"""Deterministic sampler, analytic arm, and Pillow raster renderer."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1000, 700
BG = "#f7f4ed"
CARD = "#fffefd"
DARK = "#263238"
GRAY = "#7a858a"
PALE = "#d8ddde"
TEAL = "#008c91"
MAGENTA = "#c12f87"
TAG_RGB = {
    "red": "#d54b4b",
    "blue": "#3976c5",
    "gold": "#d6a11d",
    "purple": "#8455b8",
}
TAG_NAMES = ("red", "blue", "gold", "purple")
GAPS = (45, 90, 135, 180)
XS = (140, 380, 620, 860)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    tag_gaps = list(GAPS)
    rng.shuffle(tag_gaps)
    reference = [
        {"tag": tag, "gap": gap, "rotation": rng.randrange(72) * 5}
        for tag, gap in zip(TAG_NAMES, tag_gaps)
    ]
    rng.shuffle(reference)
    for badge, x in zip(reference, XS):
        badge["x"] = x
        badge["y"] = 180

    answer_tag = TAG_NAMES[seed % len(TAG_NAMES)]
    answer_gap = next(item["gap"] for item in reference if item["tag"] == answer_tag)
    target_gaps = list(GAPS)
    rng.shuffle(target_gaps)
    target = [
        {
            "gap": gap,
            "rotation": rng.randrange(72) * 5,
            "marked": gap == answer_gap,
            "x": x,
            "y": 515,
        }
        for gap, x in zip(target_gaps, XS)
    ]
    scene = {"reference_badges": reference, "target_badges": target}
    if is_quarantined(scene):
        raise RuntimeError("sampler produced malformed correspondence")
    return scene


def analytic_gold(scene: dict) -> str:
    reference_badges = scene["reference_badges"]
    target_badges = scene["target_badges"]
    marked = [badge for badge in target_badges if badge["marked"]]
    if len(marked) != 1:
        raise ValueError("scene needs exactly one marked target")
    matches = [badge["tag"] for badge in reference_badges if badge["gap"] == marked[0]["gap"]]
    if len(matches) != 1:
        raise ValueError("marked angular signature must have one reference")
    return matches[0]


def margin(scene: dict) -> float:
    reference = scene["reference_badges"]
    target = scene["target_badges"]
    marked = [badge for badge in target if badge["marked"]]
    if len(marked) != 1:
        return 0.0
    distances = sorted(abs(int(badge["gap"]) - int(marked[0]["gap"])) for badge in reference)
    return float(distances[1] - distances[0]) if len(distances) >= 2 else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        reference = scene["reference_badges"]
        target = scene["target_badges"]
        if len(reference) != 4 or len(target) != 4:
            return True
        if {int(item["gap"]) for item in reference} != set(GAPS):
            return True
        if {int(item["gap"]) for item in target} != set(GAPS):
            return True
        if {item["tag"] for item in reference} != set(TAG_NAMES):
            return True
        if sum(bool(item["marked"]) for item in target) != 1:
            return True
        return margin(scene) < 40.0
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["reference_badges"].reverse()
    twin["target_badges"].reverse()
    return [("record_order_alias", twin)]


def _point(cx: int, cy: int, radius: int, degrees: float) -> tuple[int, int]:
    radians = math.radians(degrees)
    return round(cx + radius * math.cos(radians)), round(cy + radius * math.sin(radians))


def _draw_badge(draw: ImageDraw.ImageDraw, badge: dict, *, reference: bool) -> None:
    cx, cy = int(badge["x"]), int(badge["y"])
    gap = float(badge["gap"])
    rotation = float(badge["rotation"])
    directions = (rotation - gap / 2.0, rotation + gap / 2.0)
    satellites = [_point(cx, cy, 64, direction) for direction in directions]
    for sx, sy in satellites:
        draw.line((cx, cy, sx, sy), fill=GRAY, width=7)
    hub = [_point(cx, cy, 25, rotation + k * 45) for k in range(8)]
    draw.polygon(hub, fill=DARK)
    for sx, sy in satellites:
        draw.ellipse((sx - 12, sy - 12, sx + 12, sy + 12), fill=TEAL, outline=DARK, width=2)
    if not reference and bool(badge["marked"]):
        # Draw the marker last so its circular centroid remains an exact visual
        # locator regardless of spoke direction.
        draw.ellipse((cx - 39, cy - 39, cx + 39, cy + 39), outline=MAGENTA, width=7)
    if reference:
        color = TAG_RGB[badge["tag"]]
        draw.rounded_rectangle((cx - 35, cy + 89, cx + 35, cy + 119), radius=12, fill=color)


def render(scene: dict) -> Image.Image:
    reference_badges = scene["reference_badges"]
    target_badges = scene["target_badges"]
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    draw.text((500, 18), "REFERENCE", fill=DARK, font=_font(24, True), anchor="ma")
    draw.text((500, 350), "TARGET", fill=DARK, font=_font(24, True), anchor="ma")
    for x in XS:
        draw.rounded_rectangle((x - 105, 60, x + 105, 320), radius=14, fill=CARD, outline=PALE, width=3)
        draw.rounded_rectangle((x - 105, 390, x + 105, 650), radius=14, fill=CARD, outline=PALE, width=3)
    for badge in reference_badges:
        _draw_badge(draw, badge, reference=True)
    for badge in target_badges:
        _draw_badge(draw, badge, reference=False)
    return image
