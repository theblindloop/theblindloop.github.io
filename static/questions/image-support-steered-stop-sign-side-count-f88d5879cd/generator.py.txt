"""Deterministic renderer for visible STOP-sign side counting."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = HEIGHT = 512
SUPERSAMPLE = 4
SIDE_COUNTS = (6, 7, 8, 9, 10)
BACKGROUNDS = (
    (190, 214, 226),
    (219, 204, 183),
    (192, 218, 202),
    (214, 198, 221),
)
FACE = (194, 35, 45)
BORDER = (250, 248, 237)
TEXT = (255, 255, 247)


def _radius_for_area(side_count: int, target_area: float) -> float:
    """Circumradius for a regular n-gon with the requested analytic area."""
    return math.sqrt(2.0 * target_area / (side_count * math.sin(2.0 * math.pi / side_count)))


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed) ^ 0x57095A)
    side_count = SIDE_COUNTS[int(seed) % len(SIDE_COUNTS)]
    # Equal analytic area across answer identities blocks polygon ink area as a
    # monotone count proxy. The independent area jitter is shared by all classes.
    target_area = rng.uniform(139000.0, 147000.0)
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "center_x": rng.uniform(254.0, 258.0),
        "center_y": rng.uniform(254.0, 258.0),
        "target_area": target_area,
        "side_count": side_count,
        "rotation_phase": rng.randrange(3600),
        "face_rgb": list(FACE),
        "border_rgb": list(BORDER),
        "text_rgb": list(TEXT),
        "background_rgb": list(BACKGROUNDS[rng.randrange(len(BACKGROUNDS))]),
        "border_width": rng.randint(8, 12),
    }


def _vertices(scene: dict[str, Any], radius: float, scale: float) -> list[tuple[float, float]]:
    count = int(scene["side_count"])
    # Phase is deliberately canonicalized modulo a complete turn, enabling the
    # declared pixel-identical latent alias without floating-point drift.
    rotation = (int(scene["rotation_phase"]) % 3600) * (2.0 * math.pi / 3600.0)
    cx, cy = float(scene["center_x"]), float(scene["center_y"])
    return [
        ((cx + radius * math.cos(rotation + 2.0 * math.pi * i / count)) * scale,
         (cy + radius * math.sin(rotation + 2.0 * math.pi * i / count)) * scale)
        for i in range(count)
    ]


def render(scene: dict[str, Any]) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    count = int(scene["side_count"])
    radius = _radius_for_area(count, float(scene["target_area"]))
    ss = SUPERSAMPLE
    image = Image.new("RGB", (width * ss, height * ss), tuple(scene["background_rgb"]))
    draw = ImageDraw.Draw(image)
    border_width = int(scene["border_width"])
    draw.polygon(_vertices(scene, radius, ss), fill=tuple(scene["border_rgb"]))
    draw.polygon(_vertices(scene, radius - border_width, ss), fill=tuple(scene["face_rgb"]))

    font = ImageFont.truetype("DejaVuSans-Bold.ttf", 88 * ss)
    label = "STOP"
    box = draw.textbbox((0, 0), label, font=font, stroke_width=1 * ss)
    tw, th = box[2] - box[0], box[3] - box[1]
    x = float(scene["center_x"]) * ss - tw / 2 - box[0]
    y = float(scene["center_y"]) * ss - th / 2 - box[1]
    draw.text((x, y), label, fill=tuple(scene["text_rgb"]), font=font,
              stroke_width=1 * ss, stroke_fill=tuple(scene["face_rgb"]))
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(int(scene["side_count"]))


def margin(scene: dict[str, Any]) -> float:
    """Radial corner prominence over the midpoint of each straight side."""
    count = int(scene["side_count"])
    radius = _radius_for_area(count, float(scene["target_area"]))
    return radius * (1.0 - math.cos(math.pi / count))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 9.0 or int(scene["border_width"]) < 6


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    twin = dict(scene)
    twin["rotation_phase"] = int(scene["rotation_phase"]) + 3600
    return [("rotation_phase_plus_full_turn", twin)]
