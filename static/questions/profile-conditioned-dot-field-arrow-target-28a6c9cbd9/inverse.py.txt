"""Independent pixel-only inverse for the dot-field cavity world."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image, ImageFilter

PALETTE = {
    "red": np.array((216, 70, 70), dtype=np.int16),
    "blue": np.array((52, 120, 201), dtype=np.int16),
    "green": np.array((58, 157, 93), dtype=np.int16),
    "orange": np.array((229, 139, 42), dtype=np.int16),
}


def _color_centers(rgb: np.ndarray) -> dict[str, tuple[float, float]]:
    signed = rgb.astype(np.int16)
    centers = {}
    for name, target in PALETTE.items():
        mask = np.max(np.abs(signed - target), axis=2) <= 5
        yy, xx = np.nonzero(mask)
        if len(xx) < 30:
            raise ValueError(f"colored target {name} is absent")
        centers[name] = (float(xx.mean()), float(yy.mean()))
    return centers


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    centers = _color_centers(rgb)

    spread = rgb.max(axis=2).astype(np.int16) - rgb.min(axis=2).astype(np.int16)
    dark = (rgb.max(axis=2) < 70) & (spread < 18)
    yy, xx = np.nonzero(dark)
    if len(xx) < 5000:
        raise ValueError("dense neutral-dot field is absent")

    left, right = float(xx.min()), float(xx.max())
    top, bottom = float(yy.min()), float(yy.max())
    field_cx = (left + right) / 2.0
    field_cy = (top + bottom) / 2.0

    density_image = Image.fromarray((dark.astype(np.uint8) * 255), mode="L")
    density = np.asarray(density_image.filter(ImageFilter.GaussianBlur(radius=7)), dtype=np.float64)
    grid_y, grid_x = np.indices(density.shape)
    radial = np.hypot(grid_x - field_cx, grid_y - field_cy)
    interior = radial <= 115.0
    missing = np.clip((62.0 - density) / 62.0, 0.0, 1.0) * interior
    total = float(missing.sum())
    if total < 500.0:
        raise ValueError("no coherent low-density cavity")
    cavity_x = float((missing * grid_x).sum() / total)
    cavity_y = float((missing * grid_y).sum() / total)
    vx, vy = cavity_x - field_cx, cavity_y - field_cy
    if math.hypot(vx, vy) < 3.0:
        raise ValueError("cavity direction is ambiguous")

    best_color = None
    best_score = -float("inf")
    for color, (target_x, target_y) in centers.items():
        dx, dy = target_x - field_cx, target_y - field_cy
        length = math.hypot(dx, dy)
        if length <= 0:
            continue
        score = (vx * dx + vy * dy) / length
        if score > best_score:
            best_score = score
            best_color = color
    if best_color is None:
        raise ValueError("target binding failed")
    return best_color
