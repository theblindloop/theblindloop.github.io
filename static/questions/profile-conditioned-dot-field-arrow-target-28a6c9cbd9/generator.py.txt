"""Deterministic latent scene and Pillow renderer for a negative-space arrow."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 512
COLORS = ("red", "blue", "green", "orange")
RGB = {
    "red": (216, 70, 70),
    "blue": (52, 120, 201),
    "green": (58, 157, 93),
    "orange": (229, 139, 42),
}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x5A17C3)
    direction = int(seed) % 4  # north, east, south, west
    answer_color = COLORS[(int(seed) // 4) % 4]
    remaining = [color for color in COLORS if color != answer_color]
    rng.shuffle(remaining)
    color_order: list[str | None] = [None, None, None, None]
    color_order[direction] = answer_color
    for position, color in zip((i for i in range(4) if i != direction), remaining):
        color_order[position] = color
    return {
        "direction_index": direction,
        "color_order": color_order,
        "shift_x": rng.randint(-12, 12),
        "shift_y": rng.randint(-12, 12),
        "jitter_seed": rng.randrange(1, 2**31),
        "tip_length": rng.choice([68, 78, 88]),
        "tail_length": rng.choice([30, 33, 36]),
    }


def _rotate(x: float, y: float, quarter_turns: int) -> tuple[float, float]:
    # The base arrow points east. Direction indices are N, E, S, W.
    q = (int(quarter_turns) - 1) % 4
    for _ in range(q):
        x, y = -y, x
    return x, y


def _inside_base_arrow(x: float, y: float, tip: float, tail: float) -> bool:
    if -tail <= x <= 2 and abs(y) <= 23:
        return True
    if 0 <= x <= tip and abs(y) <= 50 * (1 - x / tip):
        return True
    return False


def _base_dots(scene: dict) -> Iterable[tuple[float, float]]:
    rng = random.Random(int(scene["jitter_seed"]))
    tip = float(scene["tip_length"])
    tail = float(scene["tail_length"])
    for row in range(-18, 19):
        for column in range(-18, 19):
            x = column * 9 + rng.uniform(-0.85, 0.85)
            y = row * 9 + rng.uniform(-0.85, 0.85)
            if not _inside_base_arrow(x, y, tip, tail):
                yield x, y


def render(scene: dict) -> Image.Image:
    direction = int(scene["direction_index"]) % 4
    color_order = list(scene["color_order"])
    cx = WIDTH / 2 + float(scene["shift_x"])
    cy = HEIGHT / 2 + float(scene["shift_y"])
    # Access dimensions here as well as in _base_dots so the visible field is
    # explicitly a consequence of every declared geometric latent.
    _ = (scene["jitter_seed"], scene["tip_length"], scene["tail_length"])

    image = Image.new("RGB", (WIDTH, HEIGHT), (250, 249, 246))
    draw = ImageDraw.Draw(image)
    edge = 171
    draw.rectangle((cx - edge, cy - edge, cx + edge, cy + edge), outline=(145, 149, 155), width=2)

    dot_color = (31, 35, 42)
    radius = 3
    for x, y in _base_dots(scene):
        rx, ry = _rotate(x, y, direction)
        px, py = cx + rx, cy + ry
        draw.ellipse((px - radius, py - radius, px + radius, py + radius), fill=dot_color)

    unit = ((0, -1), (1, 0), (0, 1), (-1, 0))
    target_radius = 12
    target_distance = 218
    for position, color in enumerate(color_order):
        ux, uy = unit[position]
        px, py = cx + ux * target_distance, cy + uy * target_distance
        draw.ellipse(
            (px - target_radius - 3, py - target_radius - 3, px + target_radius + 3, py + target_radius + 3),
            fill=(250, 249, 246),
            outline=(130, 134, 141),
            width=2,
        )
        draw.ellipse(
            (px - target_radius, py - target_radius, px + target_radius, py + target_radius),
            fill=RGB[str(color)],
        )
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["color_order"][int(scene["direction_index"]) % 4])


def margin(scene: dict) -> float:
    """Ideal cavity-centroid gap between the best and perpendicular directions."""
    tip = float(scene["tip_length"])
    tail = float(scene["tail_length"])
    shaft_area = (tail + 2.0) * 46.0
    shaft_center = (-tail + 2.0) / 2.0
    head_area = tip * 100.0 / 2.0
    head_center = tip / 3.0
    centroid = (shaft_area * shaft_center + head_area * head_center) / (shaft_area + head_area)
    return float(centroid)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["direction_index"] = int(scene["direction_index"]) + 4
    return [("direction_plus_full_turn", twin)]
