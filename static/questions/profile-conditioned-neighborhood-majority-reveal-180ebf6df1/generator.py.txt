"""Deterministic scene model and Pillow renderer for a local-majority reveal."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

N = 12
LABELS = ("ring", "tee", "chevron", "cup")
INKS = ("#191923", "#321b4a", "#123247", "#4a2020")
PAPERS = ("#fffdf8", "#f8fbff", "#fff9ee", "#f7fff9")
FRAME = "#00a9c4"
MIN_MARGIN = 8.0


def _zeros() -> list[list[int]]:
    return [[0 for _ in range(N)] for _ in range(N)]


def _core(label: str) -> list[list[int]]:
    g = _zeros()
    if label == "ring":
        for r in range(2, 10):
            for c in range(2, 10):
                if r in (2, 3, 8, 9) or c in (2, 3, 8, 9):
                    g[r][c] = 1
    elif label == "tee":
        for r in range(1, 3):
            for c in range(1, 11):
                g[r][c] = 1
        for r in range(1, 11):
            for c in range(5, 7):
                g[r][c] = 1
    elif label == "cup":
        for r in range(2, 9):
            for c in (2, 3, 8, 9):
                g[r][c] = 1
        for r in range(7, 9):
            for c in range(2, 10):
                g[r][c] = 1
    elif label == "chevron":
        for i in range(5):
            for dr in range(2):
                for dc in range(2):
                    g[1 + i + dr][1 + i + dc] = 1
                    g[1 + i + dr][9 - i + dc] = 1
    else:
        raise ValueError(f"unknown label {label!r}")
    return g


def _majority(grid: list[list[int]]) -> list[list[int]]:
    out = _zeros()
    for r in range(N):
        for c in range(N):
            votes = 0
            for rr in range(max(0, r - 1), min(N, r + 2)):
                for cc in range(max(0, c - 1), min(N, c + 2)):
                    votes += int(grid[rr][cc])
            out[r][c] = int(votes >= 5)
    return out


TEMPLATES = {name: _majority(_core(name)) for name in LABELS}


def _distance(a: Iterable[Iterable[int]], b: Iterable[Iterable[int]]) -> int:
    return sum(int(x != y) for ar, br in zip(a, b) for x, y in zip(ar, br))


def _scores(grid: list[list[int]]) -> list[tuple[int, str]]:
    transformed = _majority(grid)
    return sorted((_distance(transformed, TEMPLATES[name]), name) for name in LABELS)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    label = LABELS[int(seed) % len(LABELS)]
    base = _core(label)
    target = TEMPLATES[label]
    needed = 48 - sum(sum(row) for row in base)
    zeros = [(r, c) for r in range(N) for c in range(N) if not base[r][c]]
    # Add class-independent total ink while accepting only transform-neutral noise.
    accepted = None
    for _ in range(20000):
        rng.shuffle(zeros)
        trial = [row[:] for row in base]
        for r, c in zeros[:needed]:
            trial[r][c] = 1
        if _majority(trial) == target:
            accepted = trial
            break
    if accepted is None:
        raise RuntimeError("could not sample transform-neutral noise")
    cell_size = rng.choice((25, 26, 27, 28, 29))
    span = N * cell_size
    origin_x = (512 - span) // 2 + rng.randint(-13, 13)
    origin_y = (512 - span) // 2 + rng.randint(-13, 13)
    style = rng.randrange(len(INKS))
    return {
        "grid": accepted,
        "cell_size": cell_size,
        "origin_x": origin_x,
        "origin_y": origin_y,
        "ink": INKS[style],
        "paper": PAPERS[(style + int(seed) // 4) % len(PAPERS)],
    }


def render(scene: dict) -> Image.Image:
    grid = scene["grid"]
    cell = int(scene["cell_size"])
    x0 = int(scene["origin_x"])
    y0 = int(scene["origin_y"])
    ink = str(scene["ink"])
    paper = str(scene["paper"])
    image = Image.new("RGB", (512, 512), paper)
    draw = ImageDraw.Draw(image)
    span = N * cell
    draw.rectangle((x0 - 6, y0 - 6, x0 + span + 5, y0 + span + 5), outline=FRAME, width=4)
    grid_line = "#d8dde0"
    for r in range(N):
        for c in range(N):
            xa, ya = x0 + c * cell, y0 + r * cell
            draw.rectangle((xa, ya, xa + cell - 1, ya + cell - 1), outline=grid_line, width=1)
            if int(grid[r][c]):
                draw.rectangle((xa + 3, ya + 3, xa + cell - 4, ya + cell - 4), fill=ink)
    return image


def analytic_gold(scene: dict) -> str:
    ranked = _scores(scene["grid"])
    if ranked[0][0] != 0:
        raise ValueError("scene does not reveal an exact canonical glyph")
    return ranked[0][1]


def margin(scene: dict) -> float:
    ranked = _scores(scene["grid"])
    return float(ranked[1][0] - ranked[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < MIN_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
