"""Independent image-only inverse arm: locate frame, sample cells, filter, classify."""

from __future__ import annotations

from PIL import Image

N = 12
LABELS = ("ring", "tee", "chevron", "cup")


def _blank() -> list[list[int]]:
    return [[0] * N for _ in range(N)]


def _prototype(name: str) -> list[list[int]]:
    a = _blank()
    if name == "ring":
        for y in range(2, 10):
            for x in range(2, 10):
                a[y][x] = int(y in (2, 3, 8, 9) or x in (2, 3, 8, 9))
    elif name == "tee":
        for y in range(1, 3):
            for x in range(1, 11):
                a[y][x] = 1
        for y in range(1, 11):
            a[y][5] = a[y][6] = 1
    elif name == "cup":
        for y in range(2, 9):
            for x in (2, 3, 8, 9):
                a[y][x] = 1
        for y in range(7, 9):
            for x in range(2, 10):
                a[y][x] = 1
    elif name == "chevron":
        for j in range(5):
            for dy in (0, 1):
                for dx in (0, 1):
                    a[1 + j + dy][1 + j + dx] = 1
                    a[1 + j + dy][9 - j + dx] = 1
    return a


def _vote(a: list[list[int]]) -> list[list[int]]:
    b = _blank()
    for y in range(N):
        for x in range(N):
            count = 0
            for yy in range(max(0, y - 1), min(N, y + 2)):
                for xx in range(max(0, x - 1), min(N, x + 2)):
                    count += a[yy][xx]
            b[y][x] = int(count >= 5)
    return b


REFERENCE = {name: _vote(_prototype(name)) for name in LABELS}


def _frame_bbox(image: Image.Image) -> tuple[int, int, int, int]:
    im = image.convert("RGB")
    pts = []
    for y in range(im.height):
        for x in range(im.width):
            r, g, b = im.getpixel((x, y))
            if b > 150 and g > 115 and r < 45 and b > r + 100:
                pts.append((x, y))
    if len(pts) < 500:
        raise ValueError("cyan frame missing")
    xs, ys = zip(*pts)
    return min(xs), min(ys), max(xs), max(ys)


def _extract(image: Image.Image) -> list[list[int]]:
    im = image.convert("RGB")
    left, top, right, bottom = _frame_bbox(im)
    # Four-pixel frame begins six pixels before cell interior.
    x0, y0 = left + 6, top + 6
    span_x, span_y = right - left - 11, bottom - top - 11
    cell_x, cell_y = span_x / N, span_y / N
    if not (24.0 <= cell_x <= 30.0 and 24.0 <= cell_y <= 30.0):
        raise ValueError("invalid grid geometry")
    out = _blank()
    for y in range(N):
        for x in range(N):
            cx = int(round(x0 + (x + 0.5) * cell_x))
            cy = int(round(y0 + (y + 0.5) * cell_y))
            pixels = []
            for yy in range(cy - 3, cy + 4):
                for xx in range(cx - 3, cx + 4):
                    pixels.append(im.getpixel((xx, yy)))
            dark = sum(1 for r, g, b in pixels if r + g + b < 330)
            out[y][x] = int(dark >= 35)
    return out


def decision_from_image(image: Image.Image) -> str:
    observed = _vote(_extract(image))
    ranked = sorted(
        (sum(p != q for row, ref in zip(observed, REFERENCE[name]) for p, q in zip(row, ref)), name)
        for name in LABELS
    )
    if ranked[0][0] > 1 or ranked[1][0] - ranked[0][0] < 8:
        raise ValueError("ambiguous or corrupted revealed glyph")
    return ranked[0][1]
