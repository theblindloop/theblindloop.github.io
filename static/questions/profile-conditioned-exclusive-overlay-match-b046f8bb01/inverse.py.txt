"""Image-only inverse arm using color components and relative grid geometry."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

CYAN = np.array((20, 166, 190), dtype=np.int16)
ORANGE = np.array((238, 139, 45), dtype=np.int16)
TEAL = np.array((25, 145, 116), dtype=np.int16)
BORDER_COLORS = {
    "red": np.array((214, 58, 67), dtype=np.int16),
    "blue": np.array((48, 103, 190), dtype=np.int16),
    "gold": np.array((210, 153, 35), dtype=np.int16),
}


def _mask(rgb: np.ndarray, color: np.ndarray, tolerance: int = 8) -> np.ndarray:
    return np.max(np.abs(rgb.astype(np.int16) - color), axis=2) <= tolerance


def _components(mask: np.ndarray, minimum: int = 80) -> list[tuple[float, float, int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    found = []
    for y, x in zip(*np.nonzero(mask & ~seen)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        pixels = []
        while queue:
            cy, cx = queue.popleft()
            pixels.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(pixels) >= minimum:
            ys = [p[0] for p in pixels]
            xs = [p[1] for p in pixels]
            found.append((sum(xs) / len(xs), sum(ys) / len(ys), min(xs), min(ys), max(xs), max(ys), len(pixels)))
    return found


def _decode_grid(components: list[tuple[float, float, int, int, int, int, int]]) -> set[int]:
    if len(components) < 4:
        raise ValueError("too few filled cells")
    xs = [item[0] for item in components]
    ys = [item[1] for item in components]
    x0, x1 = min(xs), max(xs)
    y0, y1 = min(ys), max(ys)
    if x1 - x0 < 50 or y1 - y0 < 50:
        raise ValueError("filled cells do not span the grid")
    result = set()
    for x, y, *_ in components:
        col = int(round(3.0 * (x - x0) / (x1 - x0)))
        row = int(round(3.0 * (y - y0) / (y1 - y0)))
        if not (0 <= row < 4 and 0 <= col < 4):
            raise ValueError("cell lies outside reconstructed grid")
        result.add(row * 4 + col)
    if len(result) != len(components):
        raise ValueError("ambiguous cell reconstruction")
    return result


def _split_options(components: list[tuple[float, float, int, int, int, int, int]]) -> list[list[tuple]]:
    if not components:
        raise ValueError("no option tiles")
    ordered = sorted(components, key=lambda item: item[0])
    gaps = sorted(
        ((ordered[i + 1][0] - ordered[i][0], i) for i in range(len(ordered) - 1)),
        reverse=True,
    )
    if len(gaps) < 2 or gaps[1][0] < 80:
        raise ValueError("cannot separate three options")
    cuts = sorted([gaps[0][1] + 1, gaps[1][1] + 1])
    groups = [ordered[: cuts[0]], ordered[cuts[0] : cuts[1]], ordered[cuts[1] :]]
    if any(len(group) < 4 for group in groups):
        raise ValueError("an option has too few tiles")
    return groups


def _border_centers(rgb: np.ndarray) -> list[tuple[float, str]]:
    centers = []
    for name, color in BORDER_COLORS.items():
        components = _components(_mask(rgb, color), minimum=250)
        if not components:
            raise ValueError(f"missing {name} border")
        component = max(components, key=lambda item: item[-1])
        centers.append((component[0], name))
    return sorted(centers)


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"))
    a = _decode_grid(_components(_mask(rgb, CYAN)))
    b = _decode_grid(_components(_mask(rgb, ORANGE)))
    target = a ^ b
    option_groups = _split_options(_components(_mask(rgb, TEAL)))
    options = [_decode_grid(group) for group in option_groups]
    distances = [len(target ^ option) for option in options]
    best = min(distances)
    if best != 0 or distances.count(best) != 1:
        raise ValueError("no unique exact exclusive-overlay match")
    answer_index = distances.index(best)
    border_centers = _border_centers(rgb)
    option_center = sum(item[0] for item in option_groups[answer_index]) / len(option_groups[answer_index])
    return min(border_centers, key=lambda item: abs(item[0] - option_center))[1]
