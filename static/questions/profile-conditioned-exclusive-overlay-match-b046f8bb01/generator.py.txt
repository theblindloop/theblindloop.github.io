"""Deterministic sampler and Pillow renderer."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 920, 700
N = 4
BACKGROUND = (250, 249, 246)
INK = (43, 48, 55)
GRID = (105, 112, 120)
PANEL = (225, 228, 231)
CYAN = (20, 166, 190)
ORANGE = (238, 139, 45)
TEAL = (25, 145, 116)
BORDERS = {
    "red": (214, 58, 67),
    "blue": (48, 103, 190),
    "gold": (210, 153, 35),
}


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{name}", size)
    except OSError:
        return ImageFont.load_default()


def _edge_complete(bits: list[int]) -> bool:
    cells = set(bits)
    return (
        any(i < 4 for i in cells)
        and any(i >= 12 for i in cells)
        and any(i % 4 == 0 for i in cells)
        and any(i % 4 == 3 for i in cells)
    )


def _make_sources(rng: random.Random) -> tuple[list[int], list[int], list[int]]:
    universe = list(range(16))
    for _ in range(5000):
        a = sorted(rng.sample(universe, rng.randint(7, 10)))
        b = sorted(rng.sample(universe, rng.randint(7, 10)))
        target = sorted(set(a) ^ set(b))
        if 6 <= len(target) <= 10 and all(_edge_complete(x) for x in (a, b, target)):
            return a, b, target
    raise RuntimeError("could not sample a well-spread overlay")


def _swap_variant(target: list[int], rng: random.Random, used: set[tuple[int, ...]]) -> list[int]:
    full = set(range(16))
    source = set(target)
    for _ in range(1000):
        remove = rng.choice(sorted(source))
        add = rng.choice(sorted(full - source))
        candidate = sorted((source - {remove}) | {add})
        key = tuple(candidate)
        if key not in used and _edge_complete(candidate):
            used.add(key)
            return candidate
    raise RuntimeError("could not sample an equal-ink distractor")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    layer_a, layer_b, target = _make_sources(rng)
    used = {tuple(target)}
    d1 = _swap_variant(target, rng, used)
    d2 = _swap_variant(target, rng, used)
    answer_index = int(seed) % 3
    distractors = [d1, d2]
    options = []
    for index in range(3):
        options.append(target if index == answer_index else distractors.pop(0))
    shift = (int(seed) // 3) % 3
    base = ["red", "blue", "gold"]
    border_order = base[shift:] + base[:shift]
    return {
        "layer_a": layer_a,
        "layer_b": layer_b,
        "options": options,
        "border_order": border_order,
    }


def _draw_grid(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    filled: Iterable[int],
    fill_color: tuple[int, int, int],
    *,
    border_color: tuple[int, int, int] | None = None,
) -> None:
    cell = 34
    side = N * cell
    x0 = center[0] - side // 2
    y0 = center[1] - side // 2
    if border_color is not None:
        draw.rounded_rectangle(
            (x0 - 20, y0 - 20, x0 + side + 20, y0 + side + 20),
            radius=10,
            fill=(255, 255, 255),
            outline=border_color,
            width=7,
        )
    else:
        draw.rounded_rectangle(
            (x0 - 18, y0 - 18, x0 + side + 18, y0 + side + 18),
            radius=10,
            fill=(255, 255, 255),
            outline=PANEL,
            width=3,
        )
    for index in range(N * N):
        row, col = divmod(index, N)
        left = x0 + col * cell
        top = y0 + row * cell
        if index in filled:
            draw.rectangle((left + 5, top + 5, left + cell - 5, top + cell - 5), fill=fill_color)
    for k in range(N + 1):
        draw.line((x0 + k * cell, y0, x0 + k * cell, y0 + side), fill=GRID, width=2)
        draw.line((x0, y0 + k * cell, x0 + side, y0 + k * cell), fill=GRID, width=2)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(25, True)
    label_font = _font(19, True)
    small_font = _font(16, False)
    draw.text((WIDTH // 2, 22), "EXCLUSIVE OVERLAY", fill=INK, font=title_font, anchor="ma")
    draw.text(
        (WIDTH // 2, 58),
        "Fill a result cell when exactly ONE source layer is filled there",
        fill=INK,
        font=small_font,
        anchor="ma",
    )
    source_centers = [(310, 190), (610, 190)]
    for center, label, bits, color in zip(
        source_centers,
        ("LAYER 1", "LAYER 2"),
        (scene["layer_a"], scene["layer_b"]),
        (CYAN, ORANGE),
    ):
        draw.text((center[0], 100), label, fill=INK, font=label_font, anchor="ma")
        _draw_grid(draw, center, bits, color)
    draw.text((WIDTH // 2, 295), "+  ONE LAYER ONLY  =", fill=INK, font=label_font, anchor="ma")
    draw.line((80, 332, WIDTH - 80, 332), fill=PANEL, width=3)
    draw.text((WIDTH // 2, 352), "WHICH OPTION IS THE EXACT RESULT?", fill=INK, font=label_font, anchor="ma")
    option_centers = [(180, 500), (460, 500), (740, 500)]
    for index, center in enumerate(option_centers):
        border_name = scene["border_order"][index]
        _draw_grid(
            draw,
            center,
            scene["options"][index],
            TEAL,
            border_color=BORDERS[border_name],
        )
        draw.text((center[0], 602), border_name.upper(), fill=BORDERS[border_name], font=label_font, anchor="ma")
    draw.text(
        (WIDTH // 2, 657),
        "Both filled cancels  •  Both empty stays empty",
        fill=INK,
        font=small_font,
        anchor="ma",
    )
    return image


def analytic_gold(scene: dict) -> str:
    target = set(scene["layer_a"]) ^ set(scene["layer_b"])
    distances = [len(target ^ set(option)) for option in scene["options"]]
    best = min(distances)
    if best != 0 or distances.count(best) != 1:
        raise ValueError("scene has no unique exact overlay match")
    return str(scene["border_order"][distances.index(best)])


def _xor(a: Iterable[int], b: Iterable[int]) -> set[int]:
    return set(a) ^ set(b)


def margin(scene: dict) -> float:
    target = _xor(scene["layer_a"], scene["layer_b"])
    distances = sorted(len(target ^ set(option)) for option in scene["options"])
    if not distances:
        return 0.0
    return float(distances[1] - distances[0]) if len(distances) > 1 else 0.0


def is_quarantined(scene: dict) -> bool:
    target = _xor(scene["layer_a"], scene["layer_b"])
    distances = sorted(len(target ^ set(option)) for option in scene["options"])
    return len(distances) < 2 or distances[0] != 0 or distances.count(0) != 1 or margin(scene) < 2.0

    return []
