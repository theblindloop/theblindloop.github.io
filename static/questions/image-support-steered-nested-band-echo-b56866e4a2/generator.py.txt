"""Deterministic latent sampler, renderer, and analytic arm."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS = 512
PLAQUE_SIZE = 104
BACKGROUND = (250, 249, 246)
PLAQUE = (238, 231, 211)
NAVY = (20, 36, 64)
RING = (102, 120, 156)
TAG_COLORS = {
    "gold": (224, 164, 49),
    "red": (196, 64, 61),
    "green": (48, 142, 96),
    "blue": (55, 103, 190),
}
PANEL_OFFSETS = ((26, 24), (78, 24), (26, 77), (78, 77))
RADII = (5, 10, 15)
DOT_RADII = (2.5, 7.5, 12.5)
COMPOSITIONS = tuple(
    (a, b, 6 - a - b)
    for a in range(1, 3)
    for b in range(1, 5)
    if 1 <= 6 - a - b <= 4
)


def _angle_slots(rng: random.Random, count: int, band: int) -> list[int]:
    # Even spacing keeps equal-size dots visibly separate while a random phase
    # prevents angular position from leaking the depth signature.
    phase = rng.randrange(0, 360, 15)
    return sorted((round(index * 360 / count) + phase + band * 7) % 360 for index in range(count))


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    reference = list(rng.choice(COMPOSITIONS))
    alternatives = [list(item) for item in COMPOSITIONS if list(item) != reference]
    rng.shuffle(alternatives)
    answer = ("red", "green", "blue")[int(seed) % 3]
    wrong = alternatives[:2]
    rng.shuffle(wrong)
    candidate_counts: dict[str, list[int]] = {}
    wrong_index = 0
    for tag in ("red", "green", "blue"):
        if tag == answer:
            candidate_counts[tag] = list(reference)
        else:
            candidate_counts[tag] = wrong[wrong_index]
            wrong_index += 1

    tag_order = ["gold", "red", "green", "blue"]
    rng.shuffle(tag_order)
    dot_slots: dict[str, list[list[int]]] = {}
    for tag in tag_order:
        counts = reference if tag == "gold" else candidate_counts[tag]
        dot_slots[tag] = [_angle_slots(rng, count, band) for band, count in enumerate(counts)]

    centers = (160, 288, 416)
    plaque_center = [rng.choice(centers), rng.choice(centers)]
    return {
        "plaque_center": plaque_center,
        "reference_counts": reference,
        "candidate_counts": candidate_counts,
        "tag_order": tag_order,
        "dot_slots": dot_slots,
    }


def _counts_for(scene: dict[str, Any], tag: str) -> list[int]:
    if tag == "gold":
        return list(scene["reference_counts"])
    return list(scene["candidate_counts"][tag])


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (CANVAS, CANVAS), BACKGROUND)
    draw = ImageDraw.Draw(image)
    cx, cy = scene["plaque_center"]
    left, top = int(cx) - PLAQUE_SIZE // 2, int(cy) - PLAQUE_SIZE // 2
    right, bottom = left + PLAQUE_SIZE - 1, top + PLAQUE_SIZE - 1
    draw.rectangle((left, top, right, bottom), fill=NAVY)
    draw.rectangle((left + 2, top + 2, right - 2, bottom - 2), fill=PLAQUE)

    for index, tag in enumerate(scene["tag_order"]):
        ox, oy = PANEL_OFFSETS[index]
        pcx, pcy = left + ox, top + oy
        for radius in RADII:
            draw.ellipse(
                (pcx - radius, pcy - radius, pcx + radius, pcy + radius),
                outline=RING,
                width=2,
            )
        counts = _counts_for(scene, tag)
        angles_by_band = scene["dot_slots"][tag]
        for band, count in enumerate(counts):
            if len(angles_by_band[band]) != count:
                raise ValueError("dot slot count does not match visible count")
            radius = DOT_RADII[band]
            for degrees in angles_by_band[band]:
                angle = math.radians(float(degrees))
                x = int(round(pcx + radius * math.cos(angle)))
                y = int(round(pcy + radius * math.sin(angle)))
                draw.ellipse((x - 1, y - 1, x + 1, y + 1), fill=NAVY)
        draw.rectangle((pcx - 4, pcy + 18, pcx + 4, pcy + 23), fill=TAG_COLORS[tag])
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    reference = tuple(scene["reference_counts"])
    candidates = scene["candidate_counts"]
    matches = [
        tag
        for tag in scene["tag_order"]
        if tag != "gold" and tuple(candidates[tag]) == reference
    ]
    if len(matches) != 1:
        raise ValueError("scene does not contain one unique matching candidate")
    return matches[0]


def margin(scene: dict[str, Any]) -> float:
    reference = tuple(scene["reference_counts"])
    distances = []
    for tag in scene["tag_order"]:
        if tag == "gold" or tuple(scene["candidate_counts"][tag]) == reference:
            continue
        distances.append(sum(abs(a - b) for a, b in zip(reference, scene["candidate_counts"][tag])))
    return float(min(distances)) if distances else 0.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        answer = analytic_gold(scene)
    except (KeyError, TypeError, ValueError):
        return True
    return answer not in {"red", "green", "blue"} or margin(scene) < 2.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
