"""Independent pixel-only inverse arm for closed_contour_quadrant.

Receives only the final RGB image. It separates each of the four quadrant
cells, thresholds the dark outline ink, and decides for each cell whether the
contour is COMPLETELY CLOSED: a closed contour fully encloses a region of
background ink (the ring interior) that is not connected to the cell border
through background, whereas an open arc's gap connects its interior to the
outside so nothing is enclosed.

Detects the enclosed region by flood-filling all background reachable from the
cell border (4-connectivity) and checking whether any background pixel remains
unreached. Imports nothing from renderer/prompts/generate/verify.
"""

from __future__ import annotations

import numpy as np

SIZE = 480
MG = SIZE // 2  # 240
DARK_SUM = 300  # poly ink (25,25,25) sums to 75; divider (205,205,205) to 615

SECTORS = ("TL", "TR", "BL", "BR")
_CELL_X = {"TL": (0, MG), "TR": (MG, SIZE), "BL": (0, MG), "BR": (MG, SIZE)}
_CELL_Y = {"TL": (0, MG), "TR": (0, MG), "BL": (MG, SIZE), "BR": (MG, SIZE)}


def _cell_closed(dark: np.ndarray, y0: int, y1: int, x0: int, x1: int) -> bool:
    cell = dark[y0:y1, x0:x1]
    bg = ~cell
    h, w = cell.shape
    reach = np.zeros((h, w), dtype=bool)
    reach[0, :] |= bg[0, :]
    reach[h - 1, :] |= bg[h - 1, :]
    reach[:, 0] |= bg[:, 0]
    reach[:, w - 1] |= bg[:, w - 1]
    while True:
        prop = np.zeros_like(reach)
        prop[1:, :] |= reach[:-1, :]
        prop[:-1, :] |= reach[1:, :]
        prop[:, 1:] |= reach[:, :-1]
        prop[:, :-1] |= reach[:, 1:]
        prop &= bg
        if not (prop & ~reach).any():
            break
        reach |= prop
    enclosed = (bg & ~reach).any()
    return bool(enclosed)


def decision_from_image(image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    dark = arr.sum(axis=2) < DARK_SUM
    for sector in SECTORS:
        x0, x1 = _CELL_X[sector]
        y0, y1 = _CELL_Y[sector]
        if _cell_closed(dark, y0, y1, x0, x1):
            return sector
    return SECTORS[0]
