"""Closed-contour quadrant search world.

The image is a 2x2 grid of quadrants split by a light divider. Each quadrant
holds exactly one dark outline contour drawn as a circular ring. Exactly one
of the four contours is COMPLETELY CLOSED (a full ring with no gap); the other
three are OPEN arcs: the same ring with a visible wedge removed so a gap opens
between the inside and the outside of the contour.

The decision is the quadrant whose contour is completely closed. Every
attribute that decides the answer (which contour has zero gap) is drawn in the
raster; nothing else in the scene is the answer.
"""

from __future__ import annotations

import math

from PIL import Image, ImageDraw

SIZE = 480
BG = (255, 255, 255)
POLY = (25, 25, 25)
DIV = (205, 205, 205)
STROKE = 9
MG = SIZE // 2  # 240, split line

SECTORS = ("TL", "TR", "BL", "BR")
SECTOR_CENTERS = {
    "TL": (SIZE // 4, SIZE // 4),        # (120, 120)
    "TR": (3 * SIZE // 4, SIZE // 4),    # (360, 120)
    "BL": (SIZE // 4, 3 * SIZE // 4),    # (120, 360)
    "BR": (3 * SIZE // 4, 3 * SIZE // 4),# (360, 360)
}

MIN_GAP = 50.0   # degrees; open arcs always carry at least this gap
MAX_GAP = 120.0
_QUARANTINE_THRESH = 35.0


def sample_scene(seed: int) -> dict:
    import numpy as np

    rng = np.random.default_rng(seed)
    winner = SECTORS[int(rng.integers(0, len(SECTORS)))]
    gaps = {}
    gap_starts = {}
    for sector in SECTORS:
        if sector == winner:
            gaps[sector] = 0.0
        else:
            gaps[sector] = float(rng.uniform(MIN_GAP, MAX_GAP))
        gap_starts[sector] = float(rng.uniform(0.0, 360.0))
    radius = float(rng.uniform(52.0, 62.0))
    return {"gaps": gaps, "radius": radius, "gap_starts": gap_starts}


def _draw_contour(
    draw: ImageDraw.ImageDraw,
    cx: float,
    cy: float,
    r: float,
    w: int,
    gap: float,
    start: float,
) -> None:
    box = [cx - r, cy - r, cx + r, cy + r]
    draw.ellipse(box, outline=POLY, width=w)
    if gap <= 0:
        return
    pad = 5
    R = r + w / 2 + pad
    extra = 3.0
    a0 = math.radians(start - extra)
    a1 = math.radians(start + gap + extra)
    pts = [(cx, cy)]
    steps = max(8, int(abs(math.degrees(a1 - a0))))
    for i in range(steps + 1):
        t = a0 + (a1 - a0) * i / steps
        pts.append((cx + R * math.cos(t), cy + R * math.sin(t)))
    draw.polygon(pts, fill=BG)


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (SIZE, SIZE), BG)
    draw = ImageDraw.Draw(img)
    draw.line([MG, 0, MG, SIZE], fill=DIV, width=2)
    draw.line([0, MG, SIZE, MG], fill=DIV, width=2)
    r = scene["radius"]
    for sector in SECTORS:
        cx, cy = SECTOR_CENTERS[sector]
        _draw_contour(draw, cx, cy, r, STROKE, scene["gaps"][sector], scene["gap_starts"][sector])
    return img


def _winner(scene: dict) -> str:
    for sector in SECTORS:
        if scene["gaps"][sector] == 0:
            return sector
    raise RuntimeError("no closed contour found")


def analytic_gold(scene: dict) -> str:
    return _winner(scene)


def margin(scene: dict) -> float:
    gaps = [scene["gaps"][s] for s in SECTORS if scene["gaps"][s] > 0]
    if not gaps:
        return float(MAX_GAP)
    return float(min(gaps))


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < _QUARANTINE_THRESH)


def latent_symmetries(scene: dict) -> list:
    return []
