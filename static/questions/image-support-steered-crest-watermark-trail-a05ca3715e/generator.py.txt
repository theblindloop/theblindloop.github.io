"""Renderer and analytic gold for the crest_watermark_trail world.

One thin grey trail runs between a round bead and a square block inside a
medium-sized band of the canvas.  Coloured triangles sit directly on the trail:
an upward triangle raises a running count by one, a downward triangle lowers it
by one.  The count starts at zero at the bead.  The decision is the colour of
the triangle at which the running count reaches its single highest value, so the
answer is the argument of a maximum taken over the whole walk rather than a
final state or a first failure.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "crest_watermark_trail-renderer-0.1.0"

CANVAS = (640, 448)
BG = (247, 246, 242)
TRACK = (98, 100, 106)
TERMINAL = (26, 26, 30)
DECOR = (205, 205, 200)
COLOURS = {
    "red": (198, 52, 44),
    "green": (30, 128, 62),
    "blue": (44, 86, 196),
    "purple": (132, 52, 170),
}
NAMES = ("red", "green", "blue", "purple")

STROKE = 7
BEAD_R = 10
CAP_HALF = 7
TRI_UP = ((0, -8), (-9, 7), (9, 7))
TRI_DOWN = ((0, 8), (-9, -7), (9, -7))

TRAIL_WIDTH = 300
SEGMENTS = 12
MIN_MARK_GAP = 21.0
MIN_OFF_SEGMENT = 15.0
MIN_CLEARANCE = 24.0
QUARANTINE_CLEARANCE = 20.0


# ---------------------------------------------------------------- geometry ---
def _nodes(rng: random.Random) -> list[tuple[int, int]]:
    """Return a monotone zigzag polyline of SEGMENTS segments, width TRAIL_WIDTH."""
    steps = [rng.randint(22, 29) for _ in range(SEGMENTS)]
    total = sum(steps)
    steps[-1] += TRAIL_WIDTH - total
    if not 20 <= steps[-1] <= 31:
        return []
    amp = rng.randint(28, 34)
    nodes = []
    x = 0
    for index in range(SEGMENTS + 1):
        side = -1 if index % 2 == 0 else 1
        v = side * (amp // 2) + rng.randint(-2, 2)
        nodes.append((x, v))
        if index < SEGMENTS:
            x += steps[index]
    return nodes


def _point_segment_distance(px, py, ax, ay, bx, by) -> float:
    dx, dy = bx - ax, by - ay
    length = dx * dx + dy * dy
    if length == 0:
        return math.hypot(px - ax, py - ay)
    t = max(0.0, min(1.0, ((px - ax) * dx + (py - ay) * dy) / length))
    return math.hypot(px - (ax + t * dx), py - (ay + t * dy))


def _mark_centre(nodes, segment: int, t: float) -> tuple[int, int]:
    ax, ay = nodes[segment]
    bx, by = nodes[segment + 1]
    return (int(round(ax + t * (bx - ax))), int(round(ay + t * (by - ay))))


def _arc_positions(nodes, marks) -> list[float]:
    """Arc length from node 0 to each mark centre, in polyline order."""
    cumulative = [0.0]
    for index in range(len(nodes) - 1):
        ax, ay = nodes[index]
        bx, by = nodes[index + 1]
        cumulative.append(cumulative[-1] + math.hypot(bx - ax, by - ay))
    out = []
    for mark in marks:
        index = mark["segment"]
        ax, ay = nodes[index]
        bx, by = nodes[index + 1]
        out.append(cumulative[index] + mark["t"] * math.hypot(bx - ax, by - ay))
    return out


def _total_length(nodes) -> float:
    return sum(
        math.hypot(nodes[i + 1][0] - nodes[i][0], nodes[i + 1][1] - nodes[i][1])
        for i in range(len(nodes) - 1)
    )


# ------------------------------------------------------------------ replay ---
def _walk_order(scene: dict) -> list[dict]:
    """Marks in the order the walk meets them, starting at the bead."""
    marks = sorted(scene["marks"], key=lambda mark: (mark["segment"], mark["t"]))
    if scene["reversed"]:
        marks = list(reversed(marks))
    return marks


def _replay(scene: dict) -> dict:
    """Replay the running count and locate its single highest point."""
    order = _walk_order(scene)
    level = 0
    levels = []
    for mark in order:
        level += 1 if mark["kind"] == "up" else -1
        levels.append(level)
    peak = max(levels)
    crest_indexes = [i for i, value in enumerate(levels) if value == peak]
    crest = crest_indexes[0]
    other = [
        levels[i]
        for i in range(len(levels))
        if order[i]["colour"] != order[crest]["colour"]
    ]
    runner_up = max(other) if other else peak - 99
    return {
        "order": order,
        "levels": levels,
        "peak": peak,
        "crest": crest,
        "unique_peak": len(crest_indexes) == 1,
        "colour": order[crest]["colour"],
        "lead": peak - runner_up,
        "final": levels[-1],
    }


def _clearance(scene: dict) -> float:
    """Smallest along-trail gap between consecutive painted features, in pixels."""
    nodes = [tuple(node) for node in scene["nodes"]]
    marks = sorted(scene["marks"], key=lambda mark: (mark["segment"], mark["t"]))
    positions = _arc_positions(nodes, marks)
    edges = [0.0] + positions + [_total_length(nodes)]
    return min(edges[i + 1] - edges[i] for i in range(len(edges) - 1))


# ------------------------------------------------------------------ sample ---
def _feature_layout(rng: random.Random, nodes) -> list[dict] | None:
    count = rng.choice((9, 10))
    segments = sorted(rng.sample(range(1, SEGMENTS - 1), count))
    marks = []
    for segment in segments:
        t = 0.42 + 0.16 * rng.random()
        marks.append({"segment": segment, "t": round(t, 4)})
    centres = [_mark_centre(nodes, mark["segment"], mark["t"]) for mark in marks]
    for index, (cx, cy) in enumerate(centres):
        for other in range(index + 1, len(centres)):
            ox, oy = centres[other]
            if math.hypot(cx - ox, cy - oy) < MIN_MARK_GAP:
                return None
        for seg in range(len(nodes) - 1):
            if abs(seg - marks[index]["segment"]) <= 1:
                continue
            ax, ay = nodes[seg]
            bx, by = nodes[seg + 1]
            if _point_segment_distance(cx, cy, ax, ay, bx, by) < MIN_OFF_SEGMENT:
                return None
        for node in (nodes[0], nodes[-1]):
            if math.hypot(cx - node[0], cy - node[1]) < 24.0:
                return None
    return marks


def _counterbalanced(order: list[dict], replay: dict) -> bool:
    """Reject scenes whose answer is readable from a cheap local cue."""
    count = len(order)
    if not replay["unique_peak"]:
        return False
    crest = replay["crest"]
    if crest < 3 or crest > count - 4:
        return False
    if replay["peak"] < 2 or replay["lead"] not in (1, 2):
        return False
    kinds = [mark["kind"] for mark in order]
    colours = [mark["colour"] for mark in order]
    answer = colours[crest]
    if kinds[-1] != "down" or replay["final"] >= replay["peak"]:
        return False
    if "down" not in kinds[:crest] or "down" not in kinds[crest + 1 :]:
        return False
    ups = [index for index, kind in enumerate(kinds) if kind == "up"]
    if colours[ups[0]] == answer or colours[ups[-1]] == answer:
        return False
    if sum(1 for index in ups if colours[index] == answer) < 2:
        return False
    if colours[0] == answer or colours[-1] == answer:
        return False
    tally = {name: colours.count(name) for name in NAMES}
    if min(tally.values()) < 2:
        return False
    if tally[answer] > min(tally.values()) + 1:
        return False
    return True


def _decor(rng: random.Random, bbox) -> list[dict]:
    x0, y0, x1, y1 = bbox
    out = []
    for _ in range(rng.randint(6, 10)):
        for _attempt in range(40):
            cx = rng.randint(22, CANVAS[0] - 22)
            cy = rng.randint(22, CANVAS[1] - 22)
            if x0 - 26 <= cx <= x1 + 26 and y0 - 26 <= cy <= y1 + 26:
                continue
            if any(math.hypot(cx - item["cx"], cy - item["cy"]) < 34 for item in out):
                continue
            out.append(
                {
                    "cx": cx,
                    "cy": cy,
                    "kind": rng.choice(("up", "down", "bar")),
                }
            )
            break
    return out


def sample_scene(seed: int) -> dict:
    """Sample one scene; the target answer colour is balanced by seed."""
    rng = random.Random(seed * 7919 + 13)
    target = NAMES[seed % len(NAMES)]
    for _outer in range(400):
        nodes = _nodes(rng)
        if not nodes:
            continue
        marks = _feature_layout(rng, nodes)
        if marks is None:
            continue
        flipped = rng.random() < 0.5
        chosen = None
        for _inner in range(4000):
            kinds = [rng.choice(("up", "down")) for _ in marks]
            colours = [rng.choice(NAMES) for _ in marks]
            ordered = list(range(len(marks)))
            if flipped:
                ordered = list(reversed(ordered))
            for slot, index in enumerate(ordered):
                marks[index]["kind"] = kinds[slot]
                marks[index]["colour"] = colours[slot]
            probe = {"nodes": nodes, "marks": marks, "reversed": flipped}
            replay = _replay(probe)
            if replay["colour"] != target:
                continue
            if _counterbalanced(replay["order"], replay):
                chosen = [dict(mark) for mark in marks]
                break
        if chosen is None:
            continue
        xs = [node[0] for node in nodes]
        ys = [node[1] for node in nodes]
        pad_x, pad_y = 12, 14
        width = (max(xs) - min(xs)) + 2 * pad_x
        height = (max(ys) - min(ys)) + 2 * pad_y
        x0 = rng.randint(50, CANVAS[0] - width - 46)
        y0 = rng.randint(40, CANVAS[1] - height - 40)
        shift_x = x0 + pad_x - min(xs)
        shift_y = y0 + pad_y - min(ys)
        placed = [[node[0] + shift_x, node[1] + shift_y] for node in nodes]
        scene = {
            "nodes": placed,
            "marks": chosen,
            "reversed": flipped,
            "decor": [],
        }
        if _clearance(scene) < MIN_CLEARANCE:
            continue
        bbox = (x0, y0, x0 + width, y0 + height)
        scene["decor"] = _decor(rng, bbox)
        return scene
    raise RuntimeError(f"failed to sample a scene for seed {seed}")


# ------------------------------------------------------------------ render ---
def _triangle(cx: int, cy: int, kind: str) -> list[tuple[int, int]]:
    shape = TRI_UP if kind == "up" else TRI_DOWN
    return [(cx + dx, cy + dy) for dx, dy in shape]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS, BG)
    draw = ImageDraw.Draw(image)

    for item in scene["decor"]:
        cx, cy = item["cx"], item["cy"]
        if item["kind"] == "bar":
            draw.rectangle((cx - 10, cy - 3, cx + 10, cy + 3), fill=DECOR)
        else:
            draw.polygon(_triangle(cx, cy, item["kind"]), fill=DECOR)

    nodes = [tuple(node) for node in scene["nodes"]]
    for index in range(len(nodes) - 1):
        draw.line((nodes[index], nodes[index + 1]), fill=TRACK, width=STROKE)
    for node in nodes:
        draw.ellipse(
            (node[0] - STROKE // 2, node[1] - STROKE // 2,
             node[0] + STROKE // 2, node[1] + STROKE // 2),
            fill=TRACK,
        )

    start, end = (nodes[-1], nodes[0]) if scene["reversed"] else (nodes[0], nodes[-1])
    draw.ellipse(
        (start[0] - BEAD_R, start[1] - BEAD_R, start[0] + BEAD_R, start[1] + BEAD_R),
        fill=TERMINAL,
    )
    draw.rectangle(
        (end[0] - CAP_HALF, end[1] - CAP_HALF, end[0] + CAP_HALF, end[1] + CAP_HALF),
        fill=TERMINAL,
    )

    for mark in scene["marks"]:
        cx, cy = _mark_centre(nodes, mark["segment"], mark["t"])
        draw.polygon(_triangle(cx, cy, mark["kind"]), fill=COLOURS[mark["colour"]])
    return image


# -------------------------------------------------------------------- gold ---
def analytic_gold(scene: dict) -> str:
    return _replay(scene)["colour"]


def margin(scene: dict) -> float:
    replay = _replay(scene)
    return round(min(10.0 * replay["lead"], 0.5 * (_clearance(scene) - 14.0)), 6)


def is_quarantined(scene: dict) -> bool:
    replay = _replay(scene)
    return (not replay["unique_peak"]) or _clearance(scene) < QUARANTINE_CLEARANCE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases: every sampled field is painted into the raster."""
    return []
