"""Deterministic renderer and analytic reference for a long trace measurement."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 768
HEIGHT = 512
X0 = 36.0
X1 = 732.0
SEGMENTS = 15
GUIDE_INDICES = (5, 10)
SECTOR_BENDS = (range(1, 5), range(6, 10), range(11, 15))

TRACE_PALETTE = ((24, 116, 152), (174, 62, 80), (92, 72, 156), (22, 132, 94))
BACKGROUNDS = ((248, 249, 246), (246, 248, 250), (250, 247, 243))


def _centered_pattern(values: Iterable[float], drift: float) -> list[float]:
    values = list(values)
    average = sum(values) / len(values)
    return [drift + value - average for value in values]


def _make_slopes(rng: random.Random, winner: int, sector: int) -> list[float]:
    drift = rng.uniform(-0.16, 0.16)
    sign = -1.0 if rng.random() < 0.5 else 1.0
    if sector == winner:
        amplitude = rng.uniform(0.62, 0.78)
        raw = [amplitude, -amplitude, amplitude, -amplitude, amplitude]
    else:
        amplitude = rng.uniform(0.16, 0.23)
        raw = [amplitude, 0.58 * amplitude, 0.0, 0.58 * amplitude, amplitude]
    return _centered_pattern([sign * value for value in raw], drift)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    winner = int(seed) % 3
    variant = rng.randrange(3)
    trace_color = TRACE_PALETTE[rng.randrange(len(TRACE_PALETTE))]
    background = BACKGROUNDS[variant]
    frame_color = ((102, 112, 116), (88, 101, 108), (118, 105, 102))[variant]
    guide_color = ((190, 197, 198), (178, 190, 196), (201, 188, 184))[variant]
    stroke_width = 5 if variant != 2 else 6
    plot_y = 68 + (variant % 2) * 4
    plot_height = 378 - (variant % 2) * 8
    xs = [X0 + (X1 - X0) * index / SEGMENTS for index in range(SEGMENTS + 1)]
    y = rng.uniform(plot_y + 112.0, plot_y + plot_height - 112.0)
    points = [[round(xs[0], 3), round(y, 3)]]
    for sector in range(3):
        slopes = _make_slopes(rng, winner, sector)
        dx = (X1 - X0) / SEGMENTS
        for local in range(5):
            y += slopes[local] * dx
            # The construction has a bounded drift; this clamp only protects
            # against a future style change, and is never reached in the bins.
            y = max(plot_y + 30.0, min(plot_y + plot_height - 30.0, y))
            points.append([round(xs[sector * 5 + local + 1], 3), round(y, 3)])
    return {
        "vertices": points,
        "trace_color": list(trace_color),
        "background": list(background),
        "frame_color": list(frame_color),
        "guide_color": list(guide_color),
        "stroke_width": stroke_width,
        "plot_y": plot_y,
        "plot_height": plot_height,
        "variant": variant,
    }


def _sector_energies(scene: dict) -> list[float]:
    vertices = scene["vertices"]
    headings = []
    for left, right in zip(vertices[:-1], vertices[1:]):
        headings.append(math.atan2(float(right[1]) - float(left[1]), float(right[0]) - float(left[0])))
    return [
        sum(abs(headings[index] - headings[index - 1]) for index in bend_indices)
        for bend_indices in SECTOR_BENDS
    ]


def analytic_gold(scene: dict) -> str:
    energies = _sector_energies(scene)
    winner = max(range(3), key=lambda index: energies[index])
    return ("left", "middle", "right")[winner]


def margin(scene: dict) -> float:
    energies = sorted(_sector_energies(scene), reverse=True)
    return float(energies[0] - energies[1])


def is_quarantined(scene: dict) -> bool:
    vertices = scene["vertices"]
    if len(vertices) != SEGMENTS + 1:
        return True
    if margin(scene) < 0.32:
        return True
    for left, right in zip(vertices[:-1], vertices[1:]):
        if right[0] <= left[0] or not (25.0 < right[1] < HEIGHT - 25.0):
            return True
    return False


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    top = int(scene["plot_y"])
    bottom = int(scene["plot_y"] + scene["plot_height"])
    left = int(X0 - 18)
    right = int(X1 + 18)
    frame_width = 3 if scene["variant"] == 0 else 2
    draw.rectangle((left, top, right, bottom), outline=tuple(scene["frame_color"]), width=frame_width)
    vertices = scene["vertices"]
    for guide_index in GUIDE_INDICES:
        x = int(round(float(vertices[guide_index][0])))
        for y in range(top + 12, bottom - 8, 14):
            draw.line((x, y, x, min(y + 7, bottom - 1)), fill=tuple(scene["guide_color"]), width=2)
    draw.line(
        [(float(point[0]), float(point[1])) for point in vertices],
        fill=tuple(scene["trace_color"]),
        width=int(scene["stroke_width"]),
        joint="curve",
    )
    radius = max(3, int(scene["stroke_width"]) // 2)
    for point in (vertices[0], vertices[-1]):
        cx, cy = float(point[0]), float(point[1])
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=tuple(scene["trace_color"]))
    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
