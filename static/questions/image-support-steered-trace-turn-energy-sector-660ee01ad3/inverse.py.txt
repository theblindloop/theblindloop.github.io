"""Independent raster-only inverse for the trace turning-energy question."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

SEGMENTS = 15
SECTOR_BENDS = (range(1, 5), range(6, 10), range(11, 15))


def _largest_component(mask: np.ndarray) -> tuple[np.ndarray, int, int]:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    best: list[tuple[int, int]] = []
    components = 0
    for y, x in zip(*np.where(mask & ~visited)):
        y = int(y)
        x = int(x)
        if visited[y, x]:
            continue
        components += 1
        stack = [(y, x)]
        visited[y, x] = True
        current: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            current.append((cy, cx))
            for ny in range(max(0, cy - 1), min(height, cy + 2)):
                for nx in range(max(0, cx - 1), min(width, cx + 2)):
                    if mask[ny, nx] and not visited[ny, nx]:
                        visited[ny, nx] = True
                        stack.append((ny, nx))
        if len(current) > len(best):
            best = current
    result = np.zeros_like(mask, dtype=bool)
    if best:
        ys, xs = zip(*best)
        result[np.asarray(ys), np.asarray(xs)] = True
    return result, components, len(best)


def _trace_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    spread = array.max(axis=2) - array.min(axis=2)
    # The plot frame, guides, and paper are intentionally achromatic. The
    # saturated component is therefore the public, pixel-only trace signal.
    mask = spread > 35
    component, components, size = _largest_component(mask)
    if size < 900 or components > 1 and int(mask.sum()) - size > size * 0.12:
        raise ValueError("trace is incomplete or not a single connected component")
    return component


def _segment_slopes(mask: np.ndarray) -> list[float]:
    ys, xs = np.where(mask)
    if len(xs) < 900:
        raise ValueError("trace mask is too small")
    x_min = float(xs.min())
    x_max = float(xs.max())
    # The public plot exposes a long, complete route. Requiring nearly the
    # full expected span keeps endpoint erasures from being silently
    # renormalized into a shorter counterfeit route.
    if x_max - x_min < 650.0:
        raise ValueError("trace does not span the plot")
    centers: list[tuple[float, float]] = []
    for x in range(int(math.ceil(x_min)), int(math.floor(x_max)) + 1):
        column = np.where(mask[:, x])[0]
        if len(column):
            centers.append((float(x), float(np.median(column))))
    if len(centers) < SEGMENTS * 18:
        raise ValueError("trace centerline has too few columns")
    points = np.asarray(centers, dtype=float)
    slopes: list[float] = []
    for segment in range(SEGMENTS):
        start = x_min + (x_max - x_min) * segment / SEGMENTS
        end = x_min + (x_max - x_min) * (segment + 1) / SEGMENTS
        keep = (points[:, 0] >= start + 5.0) & (points[:, 0] <= end - 5.0)
        selected = points[keep]
        if len(selected) < 14:
            raise ValueError("one trace segment is unresolved")
        slope, _ = np.polyfit(selected[:, 0], selected[:, 1], 1)
        slopes.append(float(slope))
    return slopes


def decision_from_image(image: Image.Image) -> str:
    mask = _trace_mask(image)
    slopes = _segment_slopes(mask)
    headings = [math.atan(slope) for slope in slopes]
    energies = [
        sum(abs(headings[index] - headings[index - 1]) for index in bend_indices)
        for bend_indices in SECTOR_BENDS
    ]
    ranked = sorted(energies, reverse=True)
    if ranked[0] - ranked[1] < 0.22:
        raise ValueError("turning-energy comparison is ambiguous")
    return ("left", "middle", "right")[int(np.argmax(np.asarray(energies)))]
