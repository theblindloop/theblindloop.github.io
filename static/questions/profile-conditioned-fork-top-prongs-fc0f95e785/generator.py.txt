"""Deterministic renderer for one upright fork with visibly separated top prongs."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 384
HEIGHT = 384
SCALE = 4
PRONG_COUNTS = (3, 4, 5)
FILLS = (
    (36, 76, 132),
    (145, 48, 70),
    (35, 111, 88),
    (105, 61, 148),
    (157, 82, 25),
    (28, 102, 119),
)
BACKGROUNDS = (
    (250, 248, 241),
    (245, 249, 252),
    (250, 246, 250),
    (247, 249, 244),
)


def sample_scene(seed: int) -> dict:
    """Sample balanced prong identities with answer-independent style draws."""
    rng = random.Random(int(seed))
    prongs = PRONG_COUNTS[int(seed) % len(PRONG_COUNTS)]
    gap_width = rng.uniform(9.0, 13.0)
    head_width = rng.uniform(126.0, 146.0)
    prong_width = (head_width - (prongs - 1) * gap_width) / prongs
    return {
        "prongs": prongs,
        "center_x": rng.uniform(181.0, 203.0),
        "top_y": rng.uniform(47.0, 64.0),
        "prong_width": prong_width,
        "gap_width": gap_width,
        "prong_length": rng.uniform(108.0, 124.0),
        "neck_width": rng.uniform(38.0, 48.0),
        "handle_bottom": rng.uniform(332.0, 348.0),
        "fill_rgb": list(rng.choice(FILLS)),
        "background_rgb": list(rng.choice(BACKGROUNDS)),
    }


def _geometry(scene: dict) -> dict:
    n = int(scene["prongs"])
    cx = float(scene["center_x"]) % WIDTH
    top = float(scene["top_y"])
    tooth = float(scene["prong_width"])
    gap = float(scene["gap_width"])
    length = float(scene["prong_length"])
    neck = float(scene["neck_width"])
    bottom = float(scene["handle_bottom"])
    head_width = n * tooth + (n - 1) * gap
    left = cx - head_width / 2.0
    merge_y = top + length
    shoulder_y = merge_y + 62.0
    return {
        "n": n,
        "cx": cx,
        "top": top,
        "tooth": tooth,
        "gap": gap,
        "left": left,
        "head_width": head_width,
        "merge_y": merge_y,
        "shoulder_y": shoulder_y,
        "neck": neck,
        "bottom": bottom,
    }


def _q(value: float) -> int:
    return round(float(value) * SCALE)


def render(scene: dict) -> Image.Image:
    background = tuple(int(value) for value in scene["background_rgb"])
    fill = tuple(int(value) for value in scene["fill_rgb"])
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), background)
    draw = ImageDraw.Draw(image)
    g = _geometry(scene)

    tip_radius = max(2.0, min(5.0, g["tooth"] * 0.22))
    for index in range(g["n"]):
        x0 = g["left"] + index * (g["tooth"] + g["gap"])
        x1 = x0 + g["tooth"]
        draw.rounded_rectangle(
            (_q(x0), _q(g["top"]), _q(x1), _q(g["merge_y"] + 8.0)),
            radius=_q(tip_radius),
            fill=fill,
        )

    half_head = g["head_width"] / 2.0
    half_neck = g["neck"] / 2.0
    draw.polygon(
        [
            (_q(g["cx"] - half_head), _q(g["merge_y"])),
            (_q(g["cx"] + half_head), _q(g["merge_y"])),
            (_q(g["cx"] + half_neck), _q(g["shoulder_y"])),
            (_q(g["cx"] - half_neck), _q(g["shoulder_y"])),
        ],
        fill=fill,
    )
    draw.rounded_rectangle(
        (
            _q(g["cx"] - half_neck),
            _q(g["shoulder_y"] - 5.0),
            _q(g["cx"] + half_neck),
            _q(g["bottom"]),
        ),
        radius=_q(half_neck),
        fill=fill,
    )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(int(scene["prongs"]))


def margin(scene: dict) -> float:
    """Runner-up safety proxy: the smaller of solid-run width and slot width."""
    return min(float(scene["prong_width"]), float(scene["gap_width"]))


def is_quarantined(scene: dict) -> bool:
    g = _geometry(scene)
    horizontal = min(g["left"], WIDTH - (g["left"] + g["head_width"]))
    vertical = min(g["top"], HEIGHT - g["bottom"])
    contrast = sum(
        (float(a) - float(b)) ** 2
        for a, b in zip(scene["fill_rgb"], scene["background_rgb"])
    ) ** 0.5
    return margin(scene) < 8.0 or float(scene["prong_width"]) < 12.0 or min(horizontal, vertical) < 30.0 or contrast < 80.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["center_x"] = float(twin["center_x"]) + WIDTH
    return [("center_x_plus_one_canvas_period", twin)]
