"""Independent pixel-only inverse for the visible number of top fork prongs."""

from __future__ import annotations

from collections import Counter

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    pass


def _foreground_mask(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    height, width, _ = arr.shape
    edge = max(4, min(height, width) // 32)
    border = np.concatenate(
        [
            arr[:edge].reshape(-1, 3),
            arr[-edge:].reshape(-1, 3),
            arr[:, :edge].reshape(-1, 3),
            arr[:, -edge:].reshape(-1, 3),
        ],
        axis=0,
    )
    background = np.median(border, axis=0)
    mask = np.linalg.norm(arr - background, axis=2) > 42.0
    if int(mask.sum()) < 3500:
        raise OracleAbstention("no large utensil silhouette")
    ys, xs = np.nonzero(mask)
    if xs.min() <= 1 or ys.min() <= 1 or xs.max() >= width - 2 or ys.max() >= height - 2:
        raise OracleAbstention("foreground touches image boundary")
    return mask


def _solid_runs(row: np.ndarray, minimum_width: int) -> tuple[int, list[int]]:
    padded = np.pad(row.astype(np.int8), (1, 1))
    changes = np.diff(padded)
    starts = np.flatnonzero(changes == 1)
    ends = np.flatnonzero(changes == -1)
    widths = (ends - starts).tolist()
    kept = [int(width) for width in widths if width >= minimum_width]
    return len(kept), kept


def _measure(image: Image.Image) -> tuple[int, float]:
    mask = _foreground_mask(image)
    ys, xs = np.nonzero(mask)
    top = int(ys.min())
    bottom = int(ys.max())
    height = bottom - top + 1
    # The named top end occupies well over the upper quarter of the silhouette.
    # Sample a broad band derived from the raster bounding box, never a stored coordinate.
    start = top + max(4, int(round(height * 0.035)))
    stop = top + max(12, int(round(height * 0.29)))
    counts: list[int] = []
    minimum_width = max(5, int(round(mask.shape[1] * 0.018)))
    for y in range(start, min(stop, bottom + 1)):
        count, _ = _solid_runs(mask[y], minimum_width)
        if 1 <= count <= 7:
            counts.append(count)
    if len(counts) < 20:
        raise OracleAbstention("too few stable projection rows")
    histogram = Counter(counts)
    ranked = sorted(histogram.items(), key=lambda item: (-item[1], item[0]))
    winner, best = ranked[0]
    runner_up = ranked[1][1] if len(ranked) > 1 else 0
    decision_margin = float(best - runner_up)
    if winner not in (3, 4, 5) or best < 20 or decision_margin < 12 or best < runner_up * 1.8:
        raise OracleAbstention(
            f"ambiguous upper projection: winner={winner}, best={best}, runner_up={runner_up}"
        )
    return int(winner), decision_margin


def decision_from_image(image: Image.Image) -> str:
    decision, _ = _measure(image)
    return str(decision)
