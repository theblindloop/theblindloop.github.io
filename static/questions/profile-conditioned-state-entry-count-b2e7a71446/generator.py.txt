"""State-entry count: count band entries of a tracked token over ordered panels.

The image is a vertical timeline of PANELS=5 panels read top to bottom. Each
panel shows the same horizontal row of 6 slot boxes (slot 1 to slot 6, left to
right). A fixed shaded band always covers the two middle slots (slot 3 and slot
4) in every panel. A single royal-blue token sits in exactly one slot of each
panel; its slot gives that panel's "inside the band" state (true when the slot
is slot 3 or slot 4).

Reading the panels in order, the token's inside/outside state is tracked over
time. An "entry" happens whenever the token transitions from an outside slot to
an inside slot between two consecutive panels. The decision is the total number
of such entries across the whole timeline (a value in 0, 1, or 2).
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

N_SLOTS = 6
PANELS = 5

LEFT_M = 60
TOP_M = 70
SLOT_W = 110
SLOT_H = 90
GAP = 30
TOKEN_R = 22

WIDTH = LEFT_M + N_SLOTS * SLOT_W + (N_SLOTS - 1) * GAP + LEFT_M
HEIGHT = TOP_M + PANELS * SLOT_H + (PANELS - 1) * GAP + 60

BG = (255, 255, 255)
BORDER = (90, 90, 90)
BAND = (228, 228, 234)
TOKEN = (30, 60, 220)

BAND_SLOTS = (2, 3)  # slot 3 and slot 4 (0-indexed)

SLOT_LABELS = {i: f"slot {i + 1}" for i in range(N_SLOTS)}


def _col_center(i: int) -> int:
    return LEFT_M + SLOT_W // 2 + i * (SLOT_W + GAP)


def _col_x0(i: int) -> int:
    return _col_center(i) - SLOT_W // 2


def _panel_top(k: int) -> int:
    return TOP_M + k * (SLOT_H + GAP)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    positions = [rng.randrange(N_SLOTS) for _ in range(PANELS)]
    return {"positions": positions}


def _inside(slot: int) -> bool:
    return slot in BAND_SLOTS


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)
    positions = scene["positions"]

    for k in range(PANELS):
        top = _panel_top(k)
        for i in range(N_SLOTS):
            x0 = _col_x0(i)
            box = [x0, top, x0 + SLOT_W, top + SLOT_H]
            draw.rectangle(box, outline=BORDER, width=4)
            if i in BAND_SLOTS:
                inner = [x0 + 6, top + 6, x0 + SLOT_W - 6, top + SLOT_H - 6]
                draw.rectangle(inner, fill=BAND)
            cx = _col_center(i)
            cy = top + SLOT_H // 2
            if i == positions[k]:
                draw.ellipse(
                    [cx - TOKEN_R, cy - TOKEN_R, cx + TOKEN_R, cy + TOKEN_R],
                    fill=TOKEN,
                )

    label_y = TOP_M + PANELS * SLOT_H + (PANELS - 1) * GAP + 20
    for i in range(N_SLOTS):
        cx = _col_center(i)
        draw.text(
            (cx - 14, label_y),
            SLOT_LABELS[i],
            fill=(40, 40, 40),
        )
    return img


def analytic_gold(scene: dict) -> str:
    positions = scene["positions"]
    count = 0
    for k in range(PANELS - 1):
        if (not _inside(positions[k])) and _inside(positions[k + 1]):
            count += 1
    return str(count)


def margin(scene: dict) -> float:
    return 1.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    """No nontrivial render-invariant latent symmetry exists: the single latent
    field ``positions`` directly fixes the drawn blue token's slot in every
    panel, so any change to it moves a token and changes the raster. There is no
    color, label, or arrangement permutation that keeps the pixels identical
    while altering the band-entry count."""
    return []
