"""Independent pixel-only inverse arm for the scale-register world.

Receives only the final PNG. From the raster alone it:
  1. labels pixels against the fixed palette;
  2. locates the orange ring around the filled/dark cell in the left value
     track and maps its horizontal centre to a cell index, recovering the
     START value;
  3. reads the colour of each of the four operator disks from the operator
     area (green = DOUBLE, blue = HALVE, red = INVERT);
  4. replays the four operators in order from the recovered start and returns
     the final register value as a string.

No latent scene, renderer, prompt, or gold code is accessed.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

BG = (255, 255, 255)
BOX = (120, 120, 120)
VAL = (40, 40, 45)
ORANGE = (255, 140, 0)
GREEN = (45, 165, 60)
BLUE = (55, 85, 215)
RED = (205, 45, 45)

PALETTE = [BG, BOX, VAL, ORANGE, GREEN, BLUE, RED]
_BG, _BOX, _VAL, _ORANGE, _GREEN, _BLUE, _RED = range(7)

NSLOT = 4
KOPS = 4
VALUE_MIN = 1
VALUE_MAX = 4

LEFT_V = 60
CELL_W = 70
CELL_GAP = 10
OPS_X0 = 400
OP_W = 56
OP_GAP = 16
OP_Y = 100

OP_BY_LABEL = {_GREEN: "double", _BLUE: "halve", _RED: "invert"}


class UnreadableImageError(ValueError):
    """The raster does not expose the expected register structure."""


def _cell_center(i: int) -> int:
    return LEFT_V + CELL_W // 2 + i * (CELL_W + CELL_GAP)


def _op_center(j: int) -> int:
    return OPS_X0 + OP_W // 2 + j * (OP_W + OP_GAP)


def _label(arr: np.ndarray) -> np.ndarray:
    dists = np.stack(
        [np.linalg.norm(arr - np.array(c, dtype=np.float32), axis=-1) for c in PALETTE],
        axis=-1,
    )
    return dists.argmin(axis=-1)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label(arr)

    ring_ys, ring_xs = np.nonzero(labels == _ORANGE)
    if len(ring_xs) == 0:
        raise UnreadableImageError("no orange ring found")
    ring_cx = float(ring_xs.mean())
    centers = [_cell_center(i) for i in range(NSLOT)]
    idx = int(min(range(NSLOT), key=lambda i: abs(centers[i] - ring_cx)))
    start = idx + VALUE_MIN

    ops = []
    for j in range(KOPS):
        ox = _op_center(j)
        mask = (labels[:, ox] == _GREEN) | (labels[:, ox] == _BLUE) | (
            labels[:, ox] == _RED
        )
        ys = np.nonzero(mask)[0]
        if len(ys) == 0:
            raise UnreadableImageError(f"operator disk {j} not found")
        sample_label = int(np.bincount(labels[ys, ox]).argmax())
        ops.append(OP_BY_LABEL[sample_label])

    v = start
    for op in ops:
        if op == "double":
            v = min(VALUE_MAX, 2 * v)
        elif op == "halve":
            v = max(VALUE_MIN, v // 2)
        elif op == "invert":
            v = (VALUE_MAX + VALUE_MIN) - v
    return str(v)
