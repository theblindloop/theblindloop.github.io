"""Multiplicative scale-register state tracking.

A single integer register holds a value in {1, 2, 3, 4}. The leftmost panel
shows the START value: a horizontal track of four cells, where the cell whose
index equals the starting value is filled with a dark disk and surrounded by an
orange ring. To the right are exactly four operator panels, read left to right.
Each operator panel holds one colored disk glyph that declares a deterministic
re-scaling of the running value:

  GREEN disk  = DOUBLE : v := min(4, 2*v)
  BLUE  disk  = HALVE  : v := max(1, floor(v / 2))
  RED   disk  = INVERT : v := 5 - v          (complement on the value line)

The register value is never drawn after its start. The decision is the FINAL
value recovered by replaying the four declared operators in order from the
ringed start value. The multiplicative operators (and the complement) make the
final value an order-dependent nonlinear composition of the whole sequence that
cannot be recovered by summing any set of local cues or reading any single
frame.

The latent scene stores only ``start`` and ``ops``.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

NSLOT = 4
KOPS = 4
VALUE_MIN = 1
VALUE_MAX = 4

LEFT_V = 60
CELL_W = 70
CELL_GAP = 10
CELL_H = 110
TRACK_Y = 45

OPS_X0 = 400
OP_W = 56
OP_GAP = 16
OP_Y = 100

TOKEN_R = 18
RING_R = 26

WIDTH = 720
HEIGHT = 200

BG = (255, 255, 255)
BOX = (120, 120, 120)
VAL = (40, 40, 45)
ORANGE = (255, 140, 0)
GREEN = (45, 165, 60)
BLUE = (55, 85, 215)
RED = (205, 45, 45)

OP_COLORS = {"double": GREEN, "halve": BLUE, "invert": RED}
OP_NAMES = ["double", "halve", "invert"]
BINARY = ["double", "halve"]


def _cell_center(i: int) -> int:
    return LEFT_V + CELL_W // 2 + i * (CELL_W + CELL_GAP)


def _op_center(j: int) -> int:
    return OPS_X0 + OP_W // 2 + j * (OP_W + OP_GAP)


def final_value(scene: dict) -> int:
    """Replay the declared operators in order from the start value."""
    v = int(scene["start"])
    for op in scene["ops"]:
        if op == "double":
            v = min(VALUE_MAX, 2 * v)
        elif op == "halve":
            v = max(VALUE_MIN, v // 2)
        elif op == "invert":
            v = (VALUE_MAX + VALUE_MIN) - v
    return v


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 131 + 3)
    start = rng.randint(VALUE_MIN, VALUE_MAX)
    ops = [rng.choice(OP_NAMES) for _ in range(KOPS)]
    return {"start": start, "ops": ops}


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)

    start = int(scene["start"])
    idx = start - VALUE_MIN
    for i in range(NSLOT):
        cx = _cell_center(i)
        draw.rectangle(
            [cx - CELL_W // 2, TRACK_Y, cx + CELL_W // 2, TRACK_Y + CELL_H],
            outline=BOX,
            width=2,
        )
    cx = _cell_center(idx)
    cy = TRACK_Y + CELL_H // 2
    draw.ellipse([cx - TOKEN_R, cy - TOKEN_R, cx + TOKEN_R, cy + TOKEN_R], fill=VAL)
    draw.ellipse(
        [cx - RING_R, cy - RING_R, cx + RING_R, cy + RING_R], outline=ORANGE, width=4
    )

    for j, op in enumerate(scene["ops"]):
        ox = _op_center(j)
        draw.ellipse(
            [ox - 16, OP_Y - 16, ox + 16, OP_Y + 16], fill=OP_COLORS[op]
        )
    return img


def analytic_gold(scene: dict) -> str:
    return str(final_value(scene))


def margin(scene: dict) -> float:
    """Discrete selection margin.

    The reported answer is a selection of one register value from the discrete
    set {1, 2, 3, 4}. Because the register changes value by whole multiplicative
    or complement steps, any single-operator perturbation that changes the
    outcome lands on a *different* integer value, so the nearest competing
    answer sits at least one full discrete value away. Returning the register
    span as a constant positive margin keeps the boundary well separated.
    """
    return float(VALUE_MAX - VALUE_MIN + 1)


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    """No nontrivial render-invariant latent symmetry exists.

    ``start`` fixes exactly which cell is filled and orange-ringed (the drawn
    value), and ``ops`` fixes the color and order of the four operator disks.
    Any change to ``start`` moves the filled/ringed cell and changes the pixels;
    any change to an operator's element or its position changes a glyph's color
    or order and changes the raster; any permutation that does not change pixels
    cannot change the replayed gold because it is induced by exactly those
    fields. There is therefore no reparametrization yielding pixel-identical
    renders with a different gold, so no alias symmetry is declared.
    """
    return []
