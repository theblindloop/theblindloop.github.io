"""Independent image-only inverse arm for the capital-E spine-side world."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _glyph_mask(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = arr[..., 0], arr[..., 1], arr[..., 2]
    # Public visual rule: the glyph is the only strongly blue foreground region.
    return (blue - red > 75) & (blue - green > 24) & (blue > 150)


def decision_from_image(image: Image.Image) -> str:
    """Return the side whose glyph edge contains a continuous full-height run."""
    mask = _glyph_mask(image)
    ys, xs = np.nonzero(mask)
    if len(xs) < 1000:
        return "abstain"
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    if x1 - x0 < 80 or y1 - y0 < 100:
        return "abstain"
    edge = max(2, min(6, (x1 - x0 + 1) // 30))
    crop = mask[y0 : y1 + 1]
    left_score = int(crop[:, x0 : x0 + edge].sum())
    right_score = int(crop[:, x1 - edge + 1 : x1 + 1].sum())
    gap = abs(left_score - right_score)
    if gap < edge * 40:
        return "abstain"
    return "left" if left_score > right_score else "right"
