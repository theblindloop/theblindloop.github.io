"""Deterministic renderer for the capital-E spine-side world."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw


BLUE_PALETTES = (
    (24, 92, 214),
    (18, 124, 196),
    (42, 76, 202),
    (10, 112, 224),
)
BACKGROUNDS = ((250, 248, 242), (244, 248, 252), (252, 250, 246))
MARKERS = ((186, 154, 96), (164, 168, 174), (198, 132, 92))


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one balanced, fully rendered scene from an integer seed."""
    rng = random.Random(int(seed))
    canvas = 384
    width = rng.randrange(142, 190, 4)
    height = rng.randrange(188, 246, 4)
    stroke = rng.randrange(22, 34, 2)
    x = rng.randint(86, canvas - 86 - width)
    y = rng.randint(54, canvas - 54 - height)
    side = "left" if int(seed) % 2 == 0 else "right"
    marker_layout = []
    for _ in range(3):
        size = rng.randint(7, 11)
        # Markers remain outside a generous halo around the glyph.
        while True:
            mx = rng.randint(22, canvas - 22 - size)
            my = rng.randint(22, canvas - 22 - size)
            if mx + size < x - 24 or mx > x + width + 24 or my + size < y - 24 or my > y + height + 24:
                marker_layout.append([mx, my, size])
                break
    return {
        "spine_side": side,
        "canvas_size": canvas,
        "glyph_x": x,
        "glyph_y": y,
        "glyph_width": width,
        "glyph_height": height,
        "stroke_width": stroke,
        "middle_arm_fraction": rng.choice((0.66, 0.72, 0.78, 0.84)),
        "glyph_color": list(rng.choice(BLUE_PALETTES)),
        "background_color": list(rng.choice(BACKGROUNDS)),
        "marker_color": list(rng.choice(MARKERS)),
        "marker_layout": marker_layout,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    size = int(scene["canvas_size"])
    background = tuple(scene["background_color"])
    glyph_color = tuple(scene["glyph_color"])
    marker_color = tuple(scene["marker_color"])
    image = Image.new("RGB", (size, size), background)
    draw = ImageDraw.Draw(image)
    for mx, my, marker_size in scene["marker_layout"]:
        draw.rectangle((mx, my, mx + marker_size, my + marker_size), fill=marker_color)

    x = int(scene["glyph_x"])
    y = int(scene["glyph_y"])
    width = int(scene["glyph_width"])
    height = int(scene["glyph_height"])
    stroke = int(scene["stroke_width"])
    middle_fraction = float(scene["middle_arm_fraction"])
    side = scene["spine_side"]

    # Draw the one long spine and three plainly separated arms.
    if side == "left":
        draw.rectangle((x, y, x + stroke - 1, y + height - 1), fill=glyph_color)
        arm_boxes = (
            (x, y, x + width - 1, y + stroke - 1),
            (x, y + (height - stroke) // 2, x + int(width * middle_fraction) - 1, y + (height - stroke) // 2 + stroke - 1),
            (x, y + height - stroke, x + width - 1, y + height - 1),
        )
    elif side == "right":
        draw.rectangle((x + width - stroke, y, x + width - 1, y + height - 1), fill=glyph_color)
        middle_left = x + width - int(width * middle_fraction)
        arm_boxes = (
            (x, y, x + width - 1, y + stroke - 1),
            (middle_left, y + (height - stroke) // 2, x + width - 1, y + (height - stroke) // 2 + stroke - 1),
            (x, y + height - stroke, x + width - 1, y + height - 1),
        )
    else:
        raise ValueError(f"invalid spine_side: {side!r}")
    for box in arm_boxes:
        draw.rectangle(box, fill=glyph_color)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(scene["spine_side"])


def margin(scene: dict[str, Any]) -> float:
    """Runner-up gap: full-height edge ink minus the three arm bands, in px."""
    return float(int(scene["glyph_height"]) - 3 * int(scene["stroke_width"]))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 70.0 or int(scene["stroke_width"]) < 18


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # No distinct latent encoding is claimed to produce identical pixels.
    return []
