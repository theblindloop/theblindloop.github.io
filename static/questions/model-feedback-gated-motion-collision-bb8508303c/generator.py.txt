"""Deterministic renderer and analytic arm for a gated motion intervention."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 960
HEIGHT = 700
PANEL_RECTS = (
    (35, 100, 455, 375),
    (505, 100, 925, 375),
    (35, 400, 455, 675),
    (505, 400, 925, 675),
)

TOKEN_RADIUS = 22.0
RING_RADIUS = 44.0
TRIGGER_QUARANTINE_MARGIN = 12.0
COLLISION_QUARANTINE_MARGIN = 10.0
ARROW_GAP = 7.0
ARROW_WIDTH = 8
ARROW_HEAD_LENGTH = 17.0
ARROW_HEAD_WIDTH = 10.0

BACKGROUND = (238, 242, 246)
PANEL_FILL = (250, 251, 253)
FRAME = (112, 126, 140)
FRAME_ACCENT = (205, 213, 220)
BLUE_DISK = (42, 116, 214)
ORANGE_DISK = (232, 126, 37)
BLUE_ARROW = (29, 74, 139)
ORANGE_ARROW = (164, 78, 24)
ACTIVATION_RING = (19, 157, 158)
LABEL = (57, 70, 86)
WHITE = (255, 255, 255)

PANEL_LABELS = ("A", "B", "C", "D")
_BITMAPS = {
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "B": ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    "C": ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    "D": ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
}


def _distance(first: tuple[float, float], second: tuple[float, float]) -> float:
    return math.hypot(first[0] - second[0], first[1] - second[1])


def _point_in_rect(rect: tuple[int, int, int, int], point: tuple[float, float]) -> bool:
    x0, y0, x1, y1 = rect
    return x0 + 28.0 < point[0] < x1 - 28.0 and y0 + 28.0 < point[1] < y1 - 28.0


def _rounded(value: tuple[float, float]) -> tuple[float, float]:
    return round(float(value[0]), 3), round(float(value[1]), 3)


def _make_panel(
    rect: tuple[int, int, int, int], kind: str, rng: random.Random
) -> dict:
    """Create one panel from a visible causal configuration.

    The four configurations deliberately include both sides of the gate.  The
    ``kind`` is used only while sampling; it is not stored in the returned
    scene, so the trigger must be recovered from the rendered geometry.
    """

    x0, y0, _, _ = rect
    jx = rng.uniform(-7.0, 7.0)
    jy = rng.uniform(-7.0, 7.0)
    ring = (x0 + 215.0 + jx, y0 + 145.0 + jy)

    if kind == "winner":
        blue = (x0 + 100.0 + jx, y0 + 160.0 + jy)
        blue_target = (
            ring[0] + rng.uniform(-3.0, 3.0),
            ring[1] + rng.uniform(-3.0, 3.0),
        )
        orange = (
            x0 + 320.0 + rng.uniform(-7.0, 7.0),
            y0 + 110.0 + rng.uniform(-7.0, 7.0),
        )
        orange_target = (
            ring[0] + 28.0 + rng.uniform(-3.0, 3.0),
            ring[1] + rng.uniform(-3.0, 3.0),
        )
    elif kind == "trigger_miss":
        ring = (x0 + 215.0 + jx, y0 + 170.0 + jy)
        blue = (x0 + 100.0 + jx, y0 + 112.0 + jy)
        blue_target = (
            ring[0] + rng.uniform(-3.0, 3.0),
            ring[1] + rng.uniform(-3.0, 3.0),
        )
        orange = (
            x0 + 320.0 + rng.uniform(-7.0, 7.0),
            y0 + 110.0 + rng.uniform(-7.0, 7.0),
        )
        orange_target = (
            x0 + 350.0 + rng.uniform(-5.0, 5.0),
            y0 + 220.0 + rng.uniform(-5.0, 5.0),
        )
    elif kind == "no_trigger_would_hit":
        ring = (x0 + 220.0 + jx, y0 + 82.0 + jy)
        blue = (x0 + 100.0 + jx, y0 + 160.0 + jy)
        blue_target = (x0 + 145.0 + jx, y0 + 220.0 + jy)
        orange = (
            x0 + 320.0 + rng.uniform(-7.0, 7.0),
            y0 + 110.0 + rng.uniform(-7.0, 7.0),
        )
        orange_target = (
            blue_target[0] + rng.uniform(-3.0, 3.0),
            blue_target[1] + rng.uniform(-3.0, 3.0),
        )
    elif kind == "no_trigger_static_miss":
        ring = (x0 + 215.0 + jx, y0 + 215.0 + jy)
        blue = (x0 + 105.0 + jx, y0 + 108.0 + jy)
        blue_target = (x0 + 105.0 + jx, y0 + 195.0 + jy)
        orange = (
            x0 + 320.0 + rng.uniform(-7.0, 7.0),
            y0 + 120.0 + rng.uniform(-7.0, 7.0),
        )
        orange_target = (
            x0 + 365.0 + rng.uniform(-5.0, 5.0),
            y0 + 65.0 + rng.uniform(-5.0, 5.0),
        )
    else:
        raise ValueError(f"unknown panel kind {kind}")

    values = {
        "blue_x": blue[0],
        "blue_y": blue[1],
        "blue_target_x": blue_target[0],
        "blue_target_y": blue_target[1],
        "orange_x": orange[0],
        "orange_y": orange[1],
        "orange_target_x": orange_target[0],
        "orange_target_y": orange_target[1],
        "ring_x": ring[0],
        "ring_y": ring[1],
        "ring_radius": RING_RADIUS,
        "token_radius": TOKEN_RADIUS,
    }
    return {
        key: round(float(value), 3) if key not in {"token_radius", "ring_radius"} else float(value)
        for key, value in values.items()
    }


def sample_scene(seed: int) -> dict:
    """Sample four panels with exactly one final collision under the gate."""

    rng = random.Random(int(seed))
    answer_index = int(seed) % 4
    decoys = ["trigger_miss", "no_trigger_would_hit", "no_trigger_static_miss"]
    rng.shuffle(decoys)
    kinds = []
    decoy_index = 0
    for index in range(4):
        if index == answer_index:
            kinds.append("winner")
        else:
            kinds.append(decoys[decoy_index])
            decoy_index += 1
    return {
        "canvas_width": WIDTH,
        "canvas_height": HEIGHT,
        "panels": [_make_panel(rect, kind, rng) for rect, kind in zip(PANEL_RECTS, kinds)],
    }


def _draw_bitmap(draw: ImageDraw.ImageDraw, glyph: str, xy: tuple[int, int], scale: int = 4) -> None:
    x0, y0 = xy
    for row, bits in enumerate(_BITMAPS[glyph]):
        for col, bit in enumerate(bits):
            if bit == "1":
                draw.rectangle(
                    (
                        x0 + col * scale,
                        y0 + row * scale,
                        x0 + (col + 1) * scale - 1,
                        y0 + (row + 1) * scale - 1,
                    ),
                    fill=LABEL,
                )


def _draw_arrow(
    draw: ImageDraw.ImageDraw,
    start: tuple[float, float],
    end: tuple[float, float],
    color: tuple[int, int, int],
) -> None:
    dx = end[0] - start[0]
    dy = end[1] - start[1]
    length = math.hypot(dx, dy)
    if length <= ARROW_GAP + 1.0:
        raise ValueError("arrow is too short")
    ux, uy = dx / length, dy / length
    px, py = -uy, ux
    tail = (start[0] + ARROW_GAP * ux, start[1] + ARROW_GAP * uy)
    draw.line((tail, end), fill=color, width=ARROW_WIDTH)
    base = (end[0] - ARROW_HEAD_LENGTH * ux, end[1] - ARROW_HEAD_LENGTH * uy)
    left = (base[0] + ARROW_HEAD_WIDTH * px, base[1] + ARROW_HEAD_WIDTH * py)
    right = (base[0] - ARROW_HEAD_WIDTH * px, base[1] - ARROW_HEAD_WIDTH * py)
    draw.polygon((end, left, right), fill=color)


def _draw_disk(
    draw: ImageDraw.ImageDraw, center: tuple[float, float], color: tuple[int, int, int]
) -> None:
    x, y = center
    draw.ellipse(
        (
            round(x - TOKEN_RADIUS),
            round(y - TOKEN_RADIUS),
            round(x + TOKEN_RADIUS),
            round(y + TOKEN_RADIUS),
        ),
        fill=color,
        outline=WHITE,
        width=2,
    )


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rectangle((35, 42, 925, 45), fill=FRAME_ACCENT)
    for index, panel in enumerate(scene["panels"]):
        rect = PANEL_RECTS[index]
        x0, y0, x1, y1 = rect
        draw.rounded_rectangle(
            (x0, y0, x1, y1), radius=12, fill=PANEL_FILL, outline=FRAME, width=3
        )
        draw.rectangle(
            (x0 + 14, y0 + 14, x0 + 44, y0 + 44),
            fill=WHITE,
            outline=FRAME_ACCENT,
            width=1,
        )
        _draw_bitmap(draw, PANEL_LABELS[index], (x0 + 19, y0 + 17))

        ring_x = float(panel["ring_x"])
        ring_y = float(panel["ring_y"])
        ring_r = float(panel["ring_radius"])
        draw.ellipse(
            (
                round(ring_x - ring_r),
                round(ring_y - ring_r),
                round(ring_x + ring_r),
                round(ring_y + ring_r),
            ),
            outline=ACTIVATION_RING,
            width=6,
        )

        blue = (float(panel["blue_x"]), float(panel["blue_y"]))
        blue_target = (
            float(panel["blue_target_x"]),
            float(panel["blue_target_y"]),
        )
        orange = (float(panel["orange_x"]), float(panel["orange_y"]))
        orange_target = (
            float(panel["orange_target_x"]),
            float(panel["orange_target_y"]),
        )
        _draw_arrow(draw, blue, blue_target, BLUE_ARROW)
        _draw_arrow(draw, orange, orange_target, ORANGE_ARROW)
        _draw_disk(draw, blue, BLUE_DISK)
        _draw_disk(draw, orange, ORANGE_DISK)
    return image


def _trigger_gap(panel: dict) -> float:
    blue_target = (float(panel["blue_target_x"]), float(panel["blue_target_y"]))
    ring = (float(panel["ring_x"]), float(panel["ring_y"]))
    return _distance(blue_target, ring) - float(panel["ring_radius"])


def _final_gap(panel: dict) -> float:
    blue_target = (float(panel["blue_target_x"]), float(panel["blue_target_y"]))
    orange = (float(panel["orange_x"]), float(panel["orange_y"]))
    orange_target = (
        float(panel["orange_target_x"]),
        float(panel["orange_target_y"]),
    )
    final_orange = orange_target if _trigger_gap(panel) < 0.0 else orange
    return _distance(blue_target, final_orange) - 2.0 * float(panel["token_radius"])


def _current_gap(panel: dict) -> float:
    blue = (float(panel["blue_x"]), float(panel["blue_y"]))
    orange = (float(panel["orange_x"]), float(panel["orange_y"]))
    return _distance(blue, orange) - 2.0 * float(panel["token_radius"])


def analytic_gold(scene: dict) -> str:
    # Keep the analytic arm's scene access at the same top-level boundary as
    # render(): the protected field audit should see only the visible panels
    # collection.  Nested panel fields are deliberately read inline here.
    scores = []
    for panel in scene["panels"]:
        blue_target = (
            float(panel["blue_target_x"]),
            float(panel["blue_target_y"]),
        )
        ring = (float(panel["ring_x"]), float(panel["ring_y"]))
        trigger_gap = math.hypot(
            blue_target[0] - ring[0], blue_target[1] - ring[1]
        ) - float(panel["ring_radius"])
        orange = (float(panel["orange_x"]), float(panel["orange_y"]))
        orange_target = (
            float(panel["orange_target_x"]),
            float(panel["orange_target_y"]),
        )
        final_orange = orange_target if trigger_gap < 0.0 else orange
        scores.append(
            math.hypot(
                blue_target[0] - final_orange[0],
                blue_target[1] - final_orange[1],
            )
            - 2.0 * float(panel["token_radius"])
        )
    inside = [score < 0.0 for score in scores]
    if sum(inside) == 1:
        index = inside.index(True)
    else:
        index = min(range(len(scores)), key=lambda item: scores[item])
    return f"panel-{chr(65 + index)}"


def margin(scene: dict) -> float:
    panels = scene["panels"]
    values = []
    for panel in panels:
        values.extend(
            (
                abs(_trigger_gap(panel)),
                abs(_final_gap(panel)),
                _current_gap(panel),
            )
        )
    return float(min(values)) if values else 0.0


def is_quarantined(scene: dict) -> bool:
    panels = scene.get("panels", [])
    if len(panels) != len(PANEL_RECTS):
        return True
    try:
        scores = [_final_gap(panel) for panel in panels]
        trigger_gaps = [_trigger_gap(panel) for panel in panels]
        current_gaps = [_current_gap(panel) for panel in panels]
        endpoints = []
        for panel in panels:
            endpoints.extend(
                (
                    (float(panel["blue_x"]), float(panel["blue_y"])),
                    (float(panel["blue_target_x"]), float(panel["blue_target_y"])),
                    (float(panel["orange_x"]), float(panel["orange_y"])),
                    (float(panel["orange_target_x"]), float(panel["orange_target_y"])),
                    (float(panel["ring_x"]), float(panel["ring_y"])),
                )
            )
        for index, panel in enumerate(panels):
            rect = PANEL_RECTS[index]
            for point in (
                (float(panel["blue_x"]), float(panel["blue_y"])),
                (float(panel["blue_target_x"]), float(panel["blue_target_y"])),
                (float(panel["orange_x"]), float(panel["orange_y"])),
                (float(panel["orange_target_x"]), float(panel["orange_target_y"])),
                (float(panel["ring_x"]), float(panel["ring_y"])),
            ):
                if not _point_in_rect(rect, point):
                    return True
        return (
            sum(score < 0.0 for score in scores) != 1
            or min(abs(value) for value in trigger_gaps) < TRIGGER_QUARANTINE_MARGIN
            or min(abs(value) for value in scores) < COLLISION_QUARANTINE_MARGIN
            or min(current_gaps) < COLLISION_QUARANTINE_MARGIN
        )
    except (KeyError, TypeError, ValueError, ZeroDivisionError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
