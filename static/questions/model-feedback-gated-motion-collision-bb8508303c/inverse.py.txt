"""Independent image-only inverse for the gated motion question."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

WIDTH = 960
HEIGHT = 700
PANEL_RECTS = (
    (35, 100, 455, 375),
    (505, 100, 925, 375),
    (35, 400, 455, 675),
    (505, 400, 925, 675),
)

TOKEN_RADIUS = 22.0
BLUE_DISK = np.array([42, 116, 214], dtype=np.uint8)
ORANGE_DISK = np.array([232, 126, 37], dtype=np.uint8)
BLUE_ARROW = np.array([29, 74, 139], dtype=np.uint8)
ORANGE_ARROW = np.array([164, 78, 24], dtype=np.uint8)
ACTIVATION_RING = np.array([19, 157, 158], dtype=np.uint8)


def _mask(arr: np.ndarray, color: np.ndarray) -> np.ndarray:
    return np.all(arr == color, axis=2)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    coords = np.argwhere(mask)
    pending = {(int(row), int(col)) for row, col in coords}
    components: list[np.ndarray] = []
    while pending:
        start = pending.pop()
        stack = [start]
        found = [start]
        while stack:
            row, col = stack.pop()
            for drow in (-1, 0, 1):
                for dcol in (-1, 0, 1):
                    if drow == 0 and dcol == 0:
                        continue
                    neighbor = (row + drow, col + dcol)
                    if neighbor in pending:
                        pending.remove(neighbor)
                        stack.append(neighbor)
                        found.append(neighbor)
        components.append(np.asarray(found, dtype=float))
    return components


def _crop(mask: np.ndarray, rect: tuple[int, int, int, int]) -> np.ndarray:
    x0, y0, x1, y1 = rect
    return mask[y0 : y1 + 1, x0 : x1 + 1]


def _largest_component(
    arr: np.ndarray, rect: tuple[int, int, int, int], color: np.ndarray, minimum: int
) -> np.ndarray:
    components = [part for part in _components(_crop(_mask(arr, color), rect)) if len(part) >= minimum]
    if not components:
        raise ValueError("required colored component is missing")
    component = max(components, key=len).copy()
    x0, y0, _, _ = rect
    component[:, 0] += y0
    component[:, 1] += x0
    return component


def _center(component: np.ndarray) -> tuple[float, float]:
    return float(component[:, 1].mean()), float(component[:, 0].mean())


def _ring_geometry(
    arr: np.ndarray, rect: tuple[int, int, int, int]
) -> tuple[tuple[float, float], float]:
    points = np.argwhere(_crop(_mask(arr, ACTIVATION_RING), rect))
    if len(points) < 100:
        raise ValueError("activation ring is missing")
    x0, y0, _, _ = rect
    min_row, min_col = points.min(axis=0)
    max_row, max_col = points.max(axis=0)
    center = (
        x0 + (float(min_col) + float(max_col)) / 2.0,
        y0 + (float(min_row) + float(max_row)) / 2.0,
    )
    radius = ((float(max_col) - float(min_col)) + (float(max_row) - float(min_row))) / 4.0
    return center, radius


def _arrow_endpoint(
    arr: np.ndarray,
    rect: tuple[int, int, int, int],
    color: np.ndarray,
    start: tuple[float, float],
) -> tuple[float, float]:
    component = _largest_component(arr, rect, color, 80)
    xy = np.column_stack((component[:, 1], component[:, 0]))
    relative = xy - np.asarray(start, dtype=float)
    distances = np.linalg.norm(relative, axis=1)
    maximum = float(distances.max())
    tip_points = xy[distances >= maximum - 1.5]
    if len(tip_points) == 0 or maximum < 35.0:
        raise ValueError("motion arrow is missing or too short")
    tip = tip_points.mean(axis=0)
    return float(tip[0]), float(tip[1])


def _panel_score(arr: np.ndarray, index: int) -> float:
    rect = PANEL_RECTS[index]
    blue = _center(_largest_component(arr, rect, BLUE_DISK, 300))
    orange = _center(_largest_component(arr, rect, ORANGE_DISK, 300))
    blue_target = _arrow_endpoint(arr, rect, BLUE_ARROW, blue)
    orange_target = _arrow_endpoint(arr, rect, ORANGE_ARROW, orange)
    ring, ring_radius = _ring_geometry(arr, rect)
    trigger_gap = math.hypot(blue_target[0] - ring[0], blue_target[1] - ring[1]) - ring_radius
    final_orange = orange_target if trigger_gap < 0.0 else orange
    return math.hypot(blue_target[0] - final_orange[0], blue_target[1] - final_orange[1]) - 2.0 * TOKEN_RADIUS


def decision_from_image(image: Image.Image) -> str:
    """Return the unique panel whose conditional final disks overlap."""

    if image.size != (WIDTH, HEIGHT):
        raise ValueError(f"unexpected image size {image.size}")
    arr = np.asarray(image.convert("RGB"))
    scores = [_panel_score(arr, index) for index in range(len(PANEL_RECTS))]
    inside = [score < 0.0 for score in scores]
    if sum(inside) != 1:
        raise ValueError(f"expected one final collision, got scores {scores}")
    return f"panel-{chr(65 + inside.index(True))}"
