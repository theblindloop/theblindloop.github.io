"""Exclusion-matching binding world.

Five colored keys, one per row. Each row draws the black shapes that its key is
allowed to take. All five shapes (square, circle, triangle, plus, diamond) must be used
exactly once, so the drawn rows define a bipartite graph whose perfect matching
is unique. One shape is repeated at the bottom inside a black ring.

    answer = the color of the key that the unique matching gives the ringed shape

Nothing in the raster points at the answer: the ringed shape is drawn in at
least three rows, and the row that finally takes it is fixed only after other
rows have been forced and their shapes struck out of the remaining rows.

Latent state z is fully rendered: every field of the scene dict draws ink.
"""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "exclusion_matching_binding-0.1.0"

# ---------------------------------------------------------------- palette ---
WHITE = (255, 255, 255)
BLACK = (0, 0, 0)
GRAY = (150, 150, 150)

COLOR_RGB: dict[str, tuple[int, int, int]] = {
    "blue": (30, 90, 220),
    "green": (25, 150, 60),
    "orange": (240, 140, 20),
    "purple": (150, 45, 190),
    "red": (225, 35, 45),
}
COLOR_NAMES = ("blue", "green", "orange", "purple", "red")
SHAPES = ("square", "circle", "up_tri", "plus", "diamond")

# ----------------------------------------------------------------- layout ---
CANVAS_W = 1180
CANVAS_H = 1020
N_ROWS = 5

BAR_X0 = 60
BAR_W = 130
BAR_H = 70
SHAPE_X0 = 300
ROW_Y0 = 90
ROW_PITCH = 152

SEP_Y = 800
SEP_W = 4
SEP_X0 = 60
SEP_X1 = 1120

QUERY_CY = 900
QUERY_CX = 590
RING_GAP = 26
RING_W = 8

QUARANTINE_MARGIN = 11.0

_SIZES = tuple(range(86, 97, 2))     # 86..96, even
_GAPS = tuple(range(9, 46))          # intra-row shape spacing, the tight term
_JITTER = 6


# ------------------------------------------------------------ matchings ----
def _matchings(row_sets: list[list[str]]) -> list[tuple[str, ...]]:
    out = []
    for perm in itertools.permutations(SHAPES):
        if all(perm[r] in row_sets[r] for r in range(N_ROWS)):
            out.append(perm)
    return out


def _forced_order(row_sets: list[list[str]]) -> list[int]:
    """Order in which rows become forced by pure elimination."""
    remaining = {r: set(row_sets[r]) for r in range(N_ROWS)}
    order: list[int] = []
    while remaining:
        pick = None
        for r, opts in sorted(remaining.items()):
            if len(opts) == 1:
                pick = (r, next(iter(opts)))
                break
        if pick is None:
            counts: dict[str, list[int]] = {}
            for r, opts in remaining.items():
                for s in opts:
                    counts.setdefault(s, []).append(r)
            for s in SHAPES:
                if len(counts.get(s, [])) == 1:
                    pick = (counts[s][0], s)
                    break
        if pick is None:
            break
        r, s = pick
        order.append(r)
        del remaining[r]
        for opts in remaining.values():
            opts.discard(s)
    return order


def _greedy_rowwise(row_sets: list[list[str]]) -> dict[int, str]:
    """Naive top-down assignment: first canonical unused shape in each row."""
    used: set[str] = set()
    out: dict[int, str] = {}
    for r in range(N_ROWS):
        for s in SHAPES:
            if s in row_sets[r] and s not in used:
                used.add(s)
                out[r] = s
                break
    return out


# ------------------------------------------------------------- sampling ----
def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    for _ in range(30000):
        sizes = [rng.choice((1, 2, 3, 4)) for _ in range(N_ROWS)]
        if not 10 <= sum(sizes) <= 13:
            continue
        row_sets = [sorted(rng.sample(SHAPES, k)) for k in sizes]
        degree = {s: sum(1 for rs in row_sets if s in rs) for s in SHAPES}
        if min(degree.values()) == 0:
            continue
        found = _matchings(row_sets)
        if len(found) != 1:
            continue
        matching = found[0]
        order = _forced_order(row_sets)
        if len(order) != N_ROWS:
            continue
        greedy = _greedy_rowwise(row_sets)

        options = []
        for query in SHAPES:
            # the ringed shape must be drawn in at least three rows, so no row
            # claims it by sight
            if degree[query] < 3:
                continue
            answer_row = matching.index(query)
            if len(row_sets[answer_row]) < 2:
                continue
            rows_with_query = [r for r in range(N_ROWS) if query in row_sets[r]]
            # the topmost drawn row showing it must not be the answer
            if answer_row == rows_with_query[0]:
                continue
            # nor the uniquely shortest row showing it
            smallest = min(len(row_sets[r]) for r in rows_with_query)
            tied = [r for r in rows_with_query if len(row_sets[r]) == smallest]
            if len(tied) == 1 and tied[0] == answer_row:
                continue
            # nor the row a naive top-down greedy pass would give it to
            if greedy.get(answer_row) == query:
                continue
            # and it must be settled only after two other rows are forced
            if order.index(answer_row) < 2:
                continue
            options.append(query)
        if not options:
            continue
        query = rng.choice(options)

        for rs in row_sets:
            rng.shuffle(rs)
        key_colors = list(COLOR_NAMES)
        rng.shuffle(key_colors)

        return {
            "key_colors": key_colors,
            "row_sets": row_sets,
            "query": query,
            "row_gap": [rng.choice(_GAPS) for _ in range(N_ROWS)],
            "row_dx": [rng.randrange(-_JITTER, _JITTER + 1) for _ in range(N_ROWS)],
            "row_dy": [rng.randrange(-_JITTER, _JITTER + 1) for _ in range(N_ROWS)],
            "shape_sizes": [[rng.choice(_SIZES) for _ in rs] for rs in row_sets],
            "shape_dy": [[rng.randrange(-4, 5) for _ in rs] for rs in row_sets],
            "query_size": rng.choice(_SIZES),
            "query_dx": rng.randrange(-_JITTER, _JITTER + 1),
        }
    raise RuntimeError(f"no admissible scene for seed {seed}")


# --------------------------------------------------------------- geometry --
def _row_cy(scene: dict, r: int) -> int:
    return ROW_Y0 + r * ROW_PITCH + scene["row_dy"][r]


def _shape_boxes(scene: dict, r: int) -> list[tuple[int, int, int, int]]:
    """Inclusive bboxes of the black shapes drawn in row r."""
    boxes = []
    cursor = SHAPE_X0 + scene["row_dx"][r]
    cy = _row_cy(scene, r)
    for k, _ in enumerate(scene["row_sets"][r]):
        size = scene["shape_sizes"][r][k]
        y0 = cy + scene["shape_dy"][r][k] - size // 2
        boxes.append((cursor, y0, cursor + size - 1, y0 + size - 1))
        cursor += size + scene["row_gap"][r]
    return boxes


def _query_box(scene: dict) -> tuple[int, int, int, int]:
    size = scene["query_size"]
    x0 = QUERY_CX + scene["query_dx"] - size // 2
    y0 = QUERY_CY - size // 2
    return x0, y0, x0 + size - 1, y0 + size - 1


def _bar_box(scene: dict, r: int) -> tuple[int, int, int, int]:
    cy = _row_cy(scene, r)
    return BAR_X0, cy - BAR_H // 2, BAR_X0 + BAR_W - 1, cy - BAR_H // 2 + BAR_H - 1


# ---------------------------------------------------------------- drawing --
def _draw_shape(draw: ImageDraw.ImageDraw, shape: str, box) -> None:
    x0, y0, x1, y1 = box
    size = x1 - x0 + 1
    if shape == "square":
        draw.rectangle([x0, y0, x1, y1], fill=BLACK)
    elif shape == "circle":
        draw.ellipse([x0, y0, x1, y1], fill=BLACK)
    elif shape == "up_tri":
        draw.polygon([(x0 + size // 2, y0), (x1, y1), (x0, y1)], fill=BLACK)
    elif shape == "diamond":
        cx = x0 + size // 2
        cy = y0 + size // 2
        draw.polygon([(cx, y0), (x1, cy), (cx, y1), (x0, cy)], fill=BLACK)
    elif shape == "plus":
        arm = size // 3
        a0 = x0 + arm
        draw.rectangle([a0, y0, a0 + arm - 1, y1], fill=BLACK)
        draw.rectangle([x0, y0 + arm, x1, y0 + arm + arm - 1], fill=BLACK)
    else:  # pragma: no cover - guarded by sampling
        raise ValueError(shape)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (CANVAS_W, CANVAS_H), WHITE)
    draw = ImageDraw.Draw(image)

    for r in range(N_ROWS):
        draw.rectangle(list(_bar_box(scene, r)), fill=COLOR_RGB[scene["key_colors"][r]])
        for k, shape in enumerate(scene["row_sets"][r]):
            _draw_shape(draw, shape, _shape_boxes(scene, r)[k])

    draw.rectangle([SEP_X0, SEP_Y, SEP_X1, SEP_Y + SEP_W - 1], fill=GRAY)

    qx0, qy0, qx1, qy1 = _query_box(scene)
    _draw_shape(draw, scene["query"], (qx0, qy0, qx1, qy1))
    rx0 = qx0 - RING_GAP - RING_W
    ry0 = qy0 - RING_GAP - RING_W
    rx1 = qx1 + RING_GAP + RING_W
    ry1 = qy1 + RING_GAP + RING_W
    for i in range(RING_W):
        draw.rectangle([rx0 + i, ry0 + i, rx1 - i, ry1 - i], outline=BLACK)
    return image


# ------------------------------------------------------------------ gold ---
def _matching(scene: dict) -> tuple[str, ...]:
    found = _matchings([list(rs) for rs in scene["row_sets"]])
    if len(found) != 1:
        raise ValueError("scene does not have a unique perfect matching")
    return found[0]


def analytic_gold(scene: dict) -> str:
    matching = _matching(scene)
    row = matching.index(scene["query"])
    return scene["key_colors"][row]


def margin(scene: dict) -> float:
    clearances = [float(RING_GAP), float(CANVAS_W), float(CANVAS_H)]
    prev_bottom = None
    for r in range(N_ROWS):
        boxes = _shape_boxes(scene, r)
        bar = _bar_box(scene, r)
        clearances.append(float(boxes[0][0] - bar[2] - 1))
        for a, b in zip(boxes, boxes[1:]):
            clearances.append(float(b[0] - a[2] - 1))
        top = min(box[1] for box in boxes + [bar])
        bottom = max(box[3] for box in boxes + [bar])
        if prev_bottom is not None:
            clearances.append(float(top - prev_bottom - 1))
        prev_bottom = bottom
        clearances.append(float(top))
        clearances.append(float(CANVAS_W - 1 - max(box[2] for box in boxes)))
        for k, shape in enumerate(scene["row_sets"][r]):
            if shape == "plus":
                clearances.append(float(scene["shape_sizes"][r][k] // 3))
    clearances.append(float(SEP_Y - prev_bottom - 1))
    qbox = _query_box(scene)
    clearances.append(float(qbox[1] - RING_GAP - RING_W - (SEP_Y + SEP_W)))
    clearances.append(float(CANVAS_H - 1 - (qbox[3] + RING_GAP + RING_W)))
    if scene["query"] == "plus":
        clearances.append(float(scene["query_size"] // 3))
    return float(min(clearances))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Empty: every latent field draws ink, so rendering is injective."""
    return []
