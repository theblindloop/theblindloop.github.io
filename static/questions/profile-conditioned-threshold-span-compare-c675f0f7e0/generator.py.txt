"""Deterministic renderer and analytic reference for threshold-span scenes."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

RENDERER_VERSION = "threshold_span_compare-0.1.0"

WIDTH = 1000
HEIGHT = 600
PANEL_RECTS = ((55, 495), (505, 945))
PLOT_RECTS = ((85, 465), (535, 915))
PLOT_TOP = 130
PLOT_BOTTOM = 485
PLOT_HEIGHT = PLOT_BOTTOM - PLOT_TOP
CHECKPOINTS = 12
X_MARGIN = 18
QUARANTINE_MARGIN = 0.12
CHECKPOINT_CLEARANCE = 0.025

BACKGROUND = (255, 255, 255)
PANEL_FILL = (248, 249, 252)
GRID = (224, 228, 234)
AXIS = (54, 61, 72)
TEXT = (28, 34, 44)
SUBTEXT = (91, 98, 110)
REFERENCE = (55, 95, 155)
PALETTE = (
    (213, 66, 70),
    (225, 126, 35),
    (34, 149, 92),
    (150, 66, 178),
    (206, 48, 142),
    (20, 156, 150),
)


def _span(values: Iterable[float], threshold: float) -> float:
    """Return the normalized x-length above a horizontal level."""

    values = list(values)
    if len(values) != CHECKPOINTS:
        raise ValueError("each trace must have twelve checkpoint values")
    total = 0.0
    for left, right in zip(values[:-1], values[1:], strict=True):
        left_above = left > threshold
        right_above = right > threshold
        if left_above and right_above:
            fraction = 1.0
        elif not left_above and not right_above:
            fraction = 0.0
        else:
            crossing = (threshold - left) / (right - left)
            fraction = crossing if left_above else 1.0 - crossing
        total += fraction
    return total / (CHECKPOINTS - 1)


def _trace_is_safe(values: list[float], threshold: float) -> bool:
    span = _span(values, threshold)
    return (
        len(values) == CHECKPOINTS
        and min(values) > 0.13
        and max(values) < 0.87
        and min(abs(value - threshold) for value in values) >= CHECKPOINT_CLEARANCE
        and 0.10 < span < 0.90
    )


def _make_trace(rng: random.Random, threshold: float, phase: float) -> list[float]:
    values = []
    for index in range(CHECKPOINTS):
        wave = 0.215 * math.sin(phase + index * 0.78)
        secondary = 0.075 * math.cos(phase * 0.63 + index * 1.47)
        jitter = rng.uniform(-0.022, 0.022)
        values.append(threshold + wave + secondary + jitter)
    return values


def sample_scene(seed: int) -> dict:
    """Sample a balanced pair with a comfortable, visible span margin."""

    seed = int(seed)
    rng = random.Random(seed)
    threshold = 0.50 + rng.uniform(-0.045, 0.045)
    winner = "A" if seed % 2 == 0 else "B"
    chosen: tuple[list[float], list[float]] | None = None
    for _attempt in range(300):
        a = _make_trace(rng, threshold, rng.uniform(0.0, math.tau))
        b = _make_trace(rng, threshold, rng.uniform(0.0, math.tau))
        span_a = _span(a, threshold)
        span_b = _span(b, threshold)
        safe = _trace_is_safe(a, threshold) and _trace_is_safe(b, threshold)
        if safe and abs(span_a - span_b) >= QUARANTINE_MARGIN:
            chosen = (a, b) if winner == "A" else (b, a)
            break
    if chosen is None:
        raise RuntimeError(f"could not sample a safe threshold-span scene for seed {seed}")

    color_rng = random.Random((seed * 9176) ^ 0xA771CE)
    colors = [list(color) for color in color_rng.sample(PALETTE, 2)]
    return {
        "a_values": [round(value, 8) for value in chosen[0]],
        "b_values": [round(value, 8) for value in chosen[1]],
        "threshold": round(threshold, 8),
        "colors": colors,
        "style_token": seed % 11,
    }


def _x_positions(x0: int, x1: int) -> list[int]:
    left = x0 + X_MARGIN
    right = x1 - X_MARGIN
    return [round(left + i * (right - left) / (CHECKPOINTS - 1)) for i in range(CHECKPOINTS)]


def _y_from_value(value: float) -> int:
    return round(PLOT_BOTTOM - value * PLOT_HEIGHT)


def _draw_dashed_reference(draw: ImageDraw.ImageDraw, x0: int, x1: int, y: int) -> None:
    dash = 13
    gap = 8
    x = x0
    while x < x1:
        draw.line((x, y, min(x + dash, x1), y), fill=REFERENCE, width=2)
        x += dash + gap


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    label: str,
    x_panel: tuple[int, int],
    x_plot: tuple[int, int],
    values: list[float],
    threshold: float,
    color: list[int],
) -> None:
    panel_left, panel_right = x_panel
    x0, x1 = x_plot
    draw.rounded_rectangle(
        (panel_left, 96, panel_right, 525),
        radius=8,
        fill=PANEL_FILL,
        outline=(194, 200, 210),
        width=2,
    )
    draw.text((panel_left + 14, 106), label, fill=TEXT, font=ImageFont.load_default())
    for fraction in (0.25, 0.50, 0.75):
        y = _y_from_value(fraction)
        draw.line((x0, y, x1, y), fill=GRID, width=1)
    for x in _x_positions(x0, x1):
        draw.line((x, PLOT_TOP, x, PLOT_BOTTOM), fill=(239, 241, 245), width=1)
    draw.line((x0, PLOT_TOP, x0, PLOT_BOTTOM), fill=AXIS, width=2)
    draw.line((x0, PLOT_BOTTOM, x1, PLOT_BOTTOM), fill=AXIS, width=2)
    threshold_y = _y_from_value(threshold)
    _draw_dashed_reference(draw, x0, x1, threshold_y)
    points = [
        (x, _y_from_value(value))
        for x, value in zip(_x_positions(x0, x1), values, strict=True)
    ]
    draw.line(points, fill=tuple(color), width=6, joint="curve")
    for x, y in points:
        draw.ellipse((x - 5, y - 5, x + 5, y + 5), fill=tuple(color))


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    draw.text((55, 28), "Above-threshold span", fill=TEXT, font=font)
    draw.text(
        (55, 53),
        "Compare the horizontal width of each colored trace above the dashed reference",
        fill=SUBTEXT,
        font=font,
    )
    for index, (panel, plot, values) in enumerate(
        zip(PANEL_RECTS, PLOT_RECTS, (scene["a_values"], scene["b_values"]), strict=True)
    ):
        _draw_panel(
            draw,
            "A" if index == 0 else "B",
            panel,
            plot,
            values,
            scene["threshold"],
            scene["colors"][index],
        )
    return image


def analytic_gold(scene: dict) -> str:
    span_a = _span(scene["a_values"], scene["threshold"])
    span_b = _span(scene["b_values"], scene["threshold"])
    if span_a == span_b:
        raise ValueError("threshold spans tie")
    return "A" if span_a > span_b else "B"


def margin(scene: dict) -> float:
    return abs(
        _span(scene["a_values"], scene["threshold"])
        - _span(scene["b_values"], scene["threshold"])
    )


def is_quarantined(scene: dict) -> bool:
    try:
        values = list(scene["a_values"]) + list(scene["b_values"])
        threshold = float(scene["threshold"])
        return (
            len(scene["a_values"]) != CHECKPOINTS
            or len(scene["b_values"]) != CHECKPOINTS
            or margin(scene) < QUARANTINE_MARGIN
            or min(values) <= 0.13
            or max(values) >= 0.87
            or min(abs(value - threshold) for value in values) < CHECKPOINT_CLEARANCE
        )
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["style_token"] = (int(scene.get("style_token", 0)) + 1) % 11
    return [("nuisance_style_token_toggle", twin)]
