"""Independent inverse for the threshold-span question.

Only the final RGB raster is accepted.  The inverse recovers the reference
row, segments saturated trace ink, and estimates above-threshold occupancy.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH = 1000
HEIGHT = 600
PLOT_RECTS = ((85, 465), (535, 915))
PLOT_TOP = 130
PLOT_BOTTOM = 485
X_MARGIN = 18
REFERENCE = np.asarray((55, 95, 155), dtype=np.int16)


def _reference_row(arr: np.ndarray, x0: int, x1: int) -> float:
    roi = arr[100:510, x0 + 2 : x1 - 1].astype(np.int16)
    exactish = np.max(np.abs(roi - REFERENCE), axis=2) <= 4
    counts = exactish.sum(axis=1)
    candidates = np.flatnonzero(counts >= max(8, int(counts.max() * 0.45)))
    if not len(candidates):
        raise ValueError("dashed threshold is not recoverable")
    best = candidates[counts[candidates].argmax()]
    nearby = candidates[np.abs(candidates - best) <= 2]
    weights = counts[nearby].astype(np.float64)
    return float(np.average(nearby + 100, weights=weights))


def _trace_profile(arr: np.ndarray, x0: int, x1: int) -> tuple[np.ndarray, np.ndarray]:
    roi = arr[PLOT_TOP - 8 : PLOT_BOTTOM + 8, x0 + 3 : x1 - 2].astype(np.int16)
    spread = roi.max(axis=2) - roi.min(axis=2)
    reference_distance = np.max(np.abs(roi - REFERENCE), axis=2)
    # Trace colors are saturated and deliberately separated from the blue
    # reference ink.  Achromatic grid, axes, and text are excluded.
    mask = (spread >= 48) & (reference_distance > 20)
    left = X_MARGIN - 2
    right = (x1 - x0) - X_MARGIN + 2
    valid_x: list[float] = []
    valid_y: list[float] = []
    for local_x in range(max(0, left), min(mask.shape[1], right + 1)):
        ys = np.flatnonzero(mask[:, local_x]) + (PLOT_TOP - 8)
        if len(ys):
            valid_x.append(float(x0 + 3 + local_x))
            valid_y.append(float(np.median(ys)))
    if len(valid_x) < 0.70 * (right - left + 1):
        raise ValueError("colored trace is incomplete or unreadable")
    target_x = np.arange(x0 + X_MARGIN, x1 - X_MARGIN + 1, dtype=np.float64)
    target_y = np.interp(target_x, np.asarray(valid_x), np.asarray(valid_y))
    return target_x, target_y


def _above_span(arr: np.ndarray, x0: int, x1: int) -> float:
    threshold_y = _reference_row(arr, x0, x1)
    _, trace_y = _trace_profile(arr, x0, x1)
    return float(np.mean(trace_y < threshold_y))


def decision_from_image(image: Image.Image) -> str:
    """Return A/B from pixels alone, abstaining when raster evidence is close."""

    arr = np.asarray(image.convert("RGB"))
    if arr.shape != (HEIGHT, WIDTH, 3):
        raise ValueError("unexpected raster dimensions")
    spans = [_above_span(arr, x0, x1) for x0, x1 in PLOT_RECTS]
    difference = spans[0] - spans[1]
    if abs(difference) < 0.05:
        raise ValueError("measured above-threshold spans are too close to decide")
    return "A" if difference > 0 else "B"
