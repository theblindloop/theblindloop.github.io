"""Deterministic scene sampler and raster renderer for Double-Turn T Stamp."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 800
SIDES = ("top", "right", "bottom", "left")
ROLES = ("stamp", "turn_one", "turn_two")
PALETTE = (
    (19, 128, 145),
    (218, 92, 68),
    (121, 76, 172),
    (36, 112, 199),
    (205, 137, 24),
    (38, 145, 88),
)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    # Cycling the independent operands guarantees balanced marginal values even
    # for short consecutive galleries; nuisance variables remain RNG-driven.
    stamp_side = int(seed) % 4
    turn_one = 1 if (int(seed) // 4) % 2 == 0 else -1
    turn_two = 1 if (int(seed) // 8) % 2 == 0 else -1
    slot_order = list(ROLES)
    rng.shuffle(slot_order)
    colors = list(PALETTE)
    rng.shuffle(colors)
    base_choices = ((2, 2), (3, 2), (2, 3), (3, 3))
    cluster_x, cluster_y = rng.choice(base_choices)
    decorations = []
    for x, y in ((70, 90), (700, 90), (80, 700), (710, 700)):
        decorations.append([x + rng.randint(-18, 18), y + rng.randint(-18, 18), rng.randint(8, 15)])
    return {
        "stamp_side": stamp_side,
        "turn_one": turn_one,
        "turn_two": turn_two,
        "slot_order": slot_order,
        "cluster_x": cluster_x,
        "cluster_y": cluster_y,
        "palette": [list(color) for color in colors[:3]],
        "decorations": decorations,
    }


def analytic_gold(scene: dict) -> str:
    return SIDES[(int(scene["stamp_side"]) + int(scene["turn_one"]) + int(scene["turn_two"])) % 4]


def margin(scene: dict) -> float:
    # All states are exactly one quarter-turn apart; 1.0 is the runner-up gap in
    # units of discrete quarter turns.
    return 1.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["decorations"] = list(reversed(twin["decorations"]))
    return [("reverse_commutative_decorations", twin)]


def _draw_t(draw: ImageDraw.ImageDraw, cx: int, cy: int, side: int, color: tuple[int, int, int]) -> None:
    # Upright source: a wide crossbar at the top and a narrow central stem.
    points = set()
    for yy in range(-15, -8):
        for xx in range(-15, 16):
            points.add((xx, yy))
    for yy in range(-8, 16):
        for xx in range(-3, 4):
            points.add((xx, yy))
    for _ in range(side % 4):
        points = {(-y, x) for x, y in points}
    for x, y in points:
        draw.point((cx + x, cy + y), fill=color)


def _draw_stamp(draw: ImageDraw.ImageDraw, cx: int, cy: int, side: int, color: tuple[int, int, int]) -> None:
    draw.rectangle((cx - 23, cy - 23, cx + 23, cy + 23), outline=color, width=4)
    _draw_t(draw, cx, cy, side, color)


def _draw_turn(
    draw: ImageDraw.ImageDraw,
    cx: int,
    cy: int,
    direction: int,
    dots: int,
    color: tuple[int, int, int],
) -> None:
    draw.ellipse((cx - 23, cy - 23, cx + 23, cy + 23), outline=color, width=4)
    # Both arrows travel over the upper arc. Their head lies on the right for a
    # clockwise turn and on the left for a counterclockwise turn.
    angles = list(range(210, 331, 6))
    if direction < 0:
        angles.reverse()
    path = [
        (round(cx + 12 * math.cos(math.radians(a))), round(cy + 12 * math.sin(math.radians(a))))
        for a in angles
    ]
    draw.line(path, fill=color, width=4, joint="curve")
    hx, hy = path[-1]
    if direction > 0:
        head = [(hx + 5, hy + 5), (hx - 4, hy + 4), (hx + 2, hy - 4)]
    else:
        head = [(hx - 5, hy + 5), (hx + 4, hy + 4), (hx - 2, hy - 4)]
    draw.polygon(head, fill=color)
    dot_xs = (0,) if dots == 1 else (-6, 6)
    for dx in dot_xs:
        draw.ellipse((cx + dx - 2, cy + 11, cx + dx + 2, cy + 15), fill=color)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (250, 249, 246))
    draw = ImageDraw.Draw(image)
    # Harmless, remote, low-contrast marks: draw order is deliberately
    # commutative, supporting the declared latent alias.
    for x, y, radius in scene["decorations"]:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), outline=(215, 213, 207), width=2)

    bx = int(scene["cluster_x"])
    by = int(scene["cluster_y"])
    slots = ((bx, by), (bx + 2, by), (bx + 1, by + 2))
    role_to_color = {
        "stamp": tuple(scene["palette"][0]),
        "turn_one": tuple(scene["palette"][1]),
        "turn_two": tuple(scene["palette"][2]),
    }
    for slot, role in zip(slots, scene["slot_order"], strict=True):
        cx, cy = slot[0] * 100 + 50, slot[1] * 100 + 50
        color = role_to_color[role]
        if role == "stamp":
            _draw_stamp(draw, cx, cy, int(scene["stamp_side"]), color)
        elif role == "turn_one":
            _draw_turn(draw, cx, cy, int(scene["turn_one"]), 1, color)
        else:
            _draw_turn(draw, cx, cy, int(scene["turn_two"]), 2, color)
    return image
