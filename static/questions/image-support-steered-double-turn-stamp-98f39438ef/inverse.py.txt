"""Independent inverse arm using only pixels and the public visual contract."""

from __future__ import annotations

import numpy as np
from PIL import Image

_PALETTE = (
    (19, 128, 145),
    (218, 92, 68),
    (121, 76, 172),
    (36, 112, 199),
    (205, 137, 24),
    (38, 145, 88),
)
_SIDES = ("top", "right", "bottom", "left")


class PixelOracleError(ValueError):
    pass


def _components(mask: np.ndarray) -> list[np.ndarray]:
    points = {(int(y), int(x)) for y, x in np.argwhere(mask)}
    out = []
    while points:
        start = points.pop()
        stack = [start]
        group = [start]
        while stack:
            y, x = stack.pop()
            for q in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if q in points:
                    points.remove(q)
                    stack.append(q)
                    group.append(q)
        out.append(np.asarray(group, dtype=np.int32))
    return sorted(out, key=len, reverse=True)


def _bbox(comp: np.ndarray) -> tuple[int, int, int, int]:
    ys, xs = comp[:, 0], comp[:, 1]
    return int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())


def _parse_group(mask: np.ndarray) -> tuple[str, int]:
    comps = [c for c in _components(mask) if len(c) >= 9]
    if len(comps) < 2:
        raise PixelOracleError("required token components are missing")
    outer = comps[0]
    x0, y0, x1, y1 = _bbox(outer)
    if not (44 <= x1 - x0 + 1 <= 48 and 44 <= y1 - y0 + 1 <= 48 and len(outer) >= 450):
        raise PixelOracleError("token frame is incomplete")
    cx, cy = (x0 + x1) / 2.0, (y0 + y1) / 2.0
    corner_hits = sum(
        bool(mask[round(y), round(x)])
        for x, y in ((x0, y0), (x1, y0), (x1, y1), (x0, y1))
    )
    if corner_hits >= 3:
        # The second component is the filled T. Comparing occupancy in four
        # outer bands recovers its full-crossbar side.
        t = comps[1]
        if len(t) < 300:
            raise PixelOracleError("T stamp is incomplete")
        relx = t[:, 1] - cx
        rely = t[:, 0] - cy
        scores = (
            int(np.sum(rely <= -8)),
            int(np.sum(relx >= 8)),
            int(np.sum(rely >= 8)),
            int(np.sum(relx <= -8)),
        )
        order = sorted(scores, reverse=True)
        if order[0] - order[1] < 80:
            raise PixelOracleError("T orientation lacks a safe runner-up margin")
        return "stamp", int(np.argmax(scores))

    # Circular token: largest non-frame component is the arrow; tiny components
    # are ordinal dots. The arrowhead contributes a strong side-mass asymmetry.
    small = [c for c in comps[1:] if 15 <= len(c) <= 35]
    arrows = [c for c in comps[1:] if len(c) >= 70]
    if len(arrows) != 1 or len(small) not in (1, 2):
        raise PixelOracleError("turn role dots or arrow are incomplete")
    arrow = arrows[0]
    left = int(np.sum(arrow[:, 1] < cx))
    right = int(np.sum(arrow[:, 1] > cx))
    if abs(right - left) < 10:
        raise PixelOracleError("turn direction lacks a safe runner-up margin")
    direction = 1 if right > left else -1
    return ("turn_one" if len(small) == 1 else "turn_two"), direction


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    parsed = {}
    for color in _PALETTE:
        mask = np.all(arr == np.asarray(color, dtype=np.uint8), axis=2)
        if int(mask.sum()) < 20:
            continue
        role, value = _parse_group(mask)
        if role in parsed:
            raise PixelOracleError("duplicate visual role")
        parsed[role] = value
    if set(parsed) != {"stamp", "turn_one", "turn_two"}:
        raise PixelOracleError("all three disconnected visual roles are required")
    return _SIDES[(parsed["stamp"] + parsed["turn_one"] + parsed["turn_two"]) % 4]
