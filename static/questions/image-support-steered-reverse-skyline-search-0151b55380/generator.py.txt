"""Deterministic renderer and analytic arm for reverse-skyline search."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw

WIDTH = 640
HEIGHT = 560
BACKGROUND = (249, 248, 244)
GUIDE = (218, 219, 214)
COLORS = {
    "red": (210, 55, 65),
    "blue": (48, 105, 202),
    "green": (35, 151, 91),
    "purple": (139, 72, 186),
}
COLOR_NAMES = tuple(COLORS)
HEIGHT_LEVELS = (14, 26, 38, 50)
ROW_BASES = (118, 246, 374, 502)
LEFT_XS = (72, 100, 128, 156)
RIGHT_XS = (470, 498, 526, 554)
BAR_WIDTH = 14
ALLOWED_OFFSETS = ((0, 0), (8, 0), (0, 8), (8, 8))


def _permutation_with_matches(
    rng: random.Random,
    reference: list[int],
    desired_matches: int,
) -> list[int]:
    options = [
        list(option)
        for option in itertools.permutations(HEIGHT_LEVELS)
        if sum(a == b for a, b in zip(option, reference, strict=True)) == desired_matches
    ]
    if not options:
        raise RuntimeError(f"no permutation with {desired_matches} matches")
    return list(rng.choice(options))


def sample_scene(seed: int) -> dict:
    """Sample equal-ink skyline pairs with reversal scores 4, 2, 1, and 0."""
    seed = int(seed)
    rng = random.Random(seed)
    target = COLOR_NAMES[seed % len(COLOR_NAMES)]
    distractor_scores = [0, 1, 2]
    rng.shuffle(distractor_scores)
    score_for: dict[str, int] = {target: 4}
    for color, score in zip(
        (name for name in COLOR_NAMES if name != target),
        distractor_scores,
        strict=True,
    ):
        score_for[color] = score

    pairs = []
    for color in COLOR_NAMES:
        left = list(rng.sample(HEIGHT_LEVELS, len(HEIGHT_LEVELS)))
        reversed_left = list(reversed(left))
        right = (
            reversed_left
            if score_for[color] == 4
            else _permutation_with_matches(rng, reversed_left, score_for[color])
        )
        pairs.append({"color": color, "left": left, "right": right})

    rows = list(range(4))
    rng.shuffle(rows)
    return {
        "skyline_pairs": pairs,
        "row_for_color": {
            color: row for color, row in zip(COLOR_NAMES, rows, strict=True)
        },
        "offset": list(rng.choice(ALLOWED_OFFSETS)),
    }


def _reversal_scores(scene: dict) -> dict[str, int]:
    scores: dict[str, int] = {}
    for record in scene["skyline_pairs"]:
        scores[record["color"]] = sum(
            left_height == right_height
            for left_height, right_height in zip(
                reversed(record["left"]), record["right"], strict=True
            )
        )
    return scores


def analytic_gold(scene: dict) -> str:
    scores = _reversal_scores(scene)
    winners = [color for color, score in scores.items() if score == 4]
    if len(winners) != 1:
        raise ValueError(f"expected one exact reversal, found {len(winners)}")
    return winners[0]


def margin(scene: dict) -> float:
    values = sorted(_reversal_scores(scene).values(), reverse=True)
    if len(values) != 4:
        return 0.0
    return float(values[0] - values[1])


def is_quarantined(scene: dict) -> bool:
    records = scene.get("skyline_pairs", [])
    if len(records) != 4:
        return True
    colors = [record.get("color") for record in records]
    if set(colors) != set(COLOR_NAMES) or len(colors) != len(set(colors)):
        return True
    for record in records:
        for side in ("left", "right"):
            values = record.get(side, [])
            if len(values) != 4 or sorted(values) != list(HEIGHT_LEVELS):
                return True
    rows = scene.get("row_for_color", {})
    if set(rows) != set(COLOR_NAMES) or sorted(rows.values()) != list(range(4)):
        return True
    if tuple(scene.get("offset", ())) not in ALLOWED_OFFSETS:
        return True
    return sorted(_reversal_scores(scene).values()) != [0, 1, 2, 4] or margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Record order is a pixel-identical alias because colored bars never overlap."""
    twin = {
        "skyline_pairs": [
            {"color": record["color"], "left": list(record["left"]), "right": list(record["right"])}
            for record in reversed(scene["skyline_pairs"])
        ],
        "row_for_color": dict(scene["row_for_color"]),
        "offset": list(scene["offset"]),
    }
    return [("reverse_record_order", twin)]


def _draw_skyline(
    draw: ImageDraw.ImageDraw,
    xs: tuple[int, ...],
    base_y: int,
    heights: list[int],
    rgb: tuple[int, int, int],
    offset_x: int,
) -> None:
    for x, bar_height in zip(xs, heights, strict=True):
        x0 = x + offset_x
        draw.rectangle(
            (x0, base_y - int(bar_height), x0 + BAR_WIDTH - 1, base_y - 1),
            fill=rgb,
        )


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    offset_x, offset_y = (int(value) for value in scene["offset"])
    for record in scene["skyline_pairs"]:
        color = record["color"]
        base_y = ROW_BASES[int(scene["row_for_color"][color])] + offset_y
        draw.line(
            (LEFT_XS[0] - 8 + offset_x, base_y + 3,
             LEFT_XS[-1] + BAR_WIDTH + 7 + offset_x, base_y + 3),
            fill=GUIDE,
            width=2,
        )
        draw.line(
            (RIGHT_XS[0] - 8 + offset_x, base_y + 3,
             RIGHT_XS[-1] + BAR_WIDTH + 7 + offset_x, base_y + 3),
            fill=GUIDE,
            width=2,
        )
        _draw_skyline(draw, LEFT_XS, base_y, record["left"], COLORS[color], offset_x)
        _draw_skyline(draw, RIGHT_XS, base_y, record["right"], COLORS[color], offset_x)
    return image
