"""Independent pixel-only inverse for reverse-skyline pair search.

The inverse receives only a final raster. It segments the four public colors,
enumerates every connected bar, measures visible bounding-box heights, splits
the bars into distant left and right groups, orders each group by x position,
and requires one exact reversal with a runner-up match gap of at least two.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

PALETTE = {
    "red": np.array((210, 55, 65), dtype=np.int16),
    "blue": np.array((48, 105, 202), dtype=np.int16),
    "green": np.array((35, 151, 91), dtype=np.int16),
    "purple": np.array((139, 72, 186), dtype=np.int16),
}
EXPECTED_HEIGHTS = (14, 26, 38, 50)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= 100:
            output.append(np.asarray(points, dtype=np.int32))
    return output


def _decode_pair(components: list[np.ndarray]) -> tuple[list[int], list[int]] | None:
    """Return two visible height sequences, or None for damaged evidence."""
    if len(components) != 8:
        return None
    bars: list[tuple[float, int]] = []
    for points in components:
        ys, xs = points[:, 0], points[:, 1]
        width = int(xs.max() - xs.min() + 1)
        height = int(ys.max() - ys.min() + 1)
        area = int(len(points))
        if width != 14 or height not in EXPECTED_HEIGHTS or area != width * height:
            return None
        bars.append((float(xs.mean()), height))
    bars.sort(key=lambda item: item[0])
    left, right = bars[:4], bars[4:]
    if right[0][0] - left[-1][0] < 250.0:
        return None
    if any(group[index + 1][0] - group[index][0] < 20.0 for group in (left, right) for index in range(3)):
        return None
    left_heights = [height for _, height in left]
    right_heights = [height for _, height in right]
    if sorted(left_heights) != list(EXPECTED_HEIGHTS):
        return None
    if sorted(right_heights) != list(EXPECTED_HEIGHTS):
        return None
    return left_heights, right_heights


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8).astype(np.int16)
    decoded: dict[str, tuple[list[int], list[int]]] = {}
    for name, rgb in PALETTE.items():
        distance = np.linalg.norm(array - rgb, axis=2)
        pair = _decode_pair(_components(distance < 12.0))
        if pair is not None:
            decoded[name] = pair
    if len(decoded) < 2:
        raise ValueError("fewer than two complete color candidates")
    scores = {
        name: sum(
            left_height == right_height
            for left_height, right_height in zip(reversed(left), right, strict=True)
        )
        for name, (left, right) in decoded.items()
    }
    winners = [name for name, score in scores.items() if score == 4]
    if len(winners) != 1:
        raise ValueError("no unique exact-reversal winner")
    runners = [score for name, score in scores.items() if name != winners[0]]
    if not runners or 4 - max(runners) < 2:
        raise ValueError("exact-reversal winner lacks a runner-up margin")
    return winners[0]
