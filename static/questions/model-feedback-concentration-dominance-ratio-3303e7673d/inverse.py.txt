"""Independent PNG-only inverse arm for the concentration-dominance ratio world.

Recovers, from the raster alone, the per-colour weights (number of dark dots
beside each key swatch), identifies the heaviest colour (the one with three
dots), counts the coloured disks in the two boxes and splits them by colour, and
for each box forms the heavy mass and the light mass, then compares the two
heavy/light ratios by cross-multiplication.  Uses no renderer geometry
constants, scene parameters, prompt, or gold code; only chromatic masks,
connected components, and centroid grouping.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RGB = {
    "crimson": (204, 44, 64),
    "teal": (0, 140, 150),
    "amber": (232, 158, 32),
}
MID_X = 390
MIN_SWATCH = 150
MIN_DISK = 500
MIN_DOT = 40


def _color_mask(arr: np.ndarray, color: tuple, tol: int = 40) -> np.ndarray:
    r, g, b = color
    return (
        (np.abs(arr[:, :, 0] - r) <= tol)
        & (np.abs(arr[:, :, 1] - g) <= tol)
        & (np.abs(arr[:, :, 2] - b) <= tol)
    )


def _dark_mask(arr: np.ndarray) -> np.ndarray:
    chroma = arr.max(axis=2).astype(np.int16) - arr.min(axis=2).astype(np.int16)
    mean = arr.mean(axis=2).astype(np.int16)
    return (chroma <= 20) & (mean < 130)


def _components(mask: np.ndarray, minsize: int) -> list[dict]:
    pixels = set(zip(*np.nonzero(mask)))
    found = []
    while pixels:
        start = pixels.pop()
        stack = [start]
        cells = []
        while stack:
            y, x = stack.pop()
            cells.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if (ny, nx) in pixels:
                    pixels.remove((ny, nx))
                    stack.append((ny, nx))
        if len(cells) >= minsize:
            cy = sum(y for y, _ in cells) / len(cells)
            cx = sum(x for _, x in cells) / len(cells)
            found.append({"cx": cx, "cy": cy, "size": len(cells)})
    return found


def _read_weights(arr: np.ndarray) -> dict:
    weights = {}
    for color in RGB:
        mask = _color_mask(arr, RGB[color])
        mask[:40, :] = False
        mask[200:, :] = False
        comps = _components(mask, MIN_SWATCH)
        if not comps:
            raise ValueError(f"missing key swatch for {color}")
        comps.sort(key=lambda c: c["cx"])
        swatch = comps[0]
        cx = swatch["cx"]
        dark = _dark_mask(arr)
        window = np.zeros_like(dark)
        x0 = max(0, int(cx) + 25)
        x1 = int(cx) + 118
        window[80:160, x0:x1] = dark[80:160, x0:x1]
        dots = _components(window, MIN_DOT)
        w = len(dots)
        if w < 1 or w > 3:
            raise ValueError(f"invalid weight {w} for {color}")
        weights[color] = w
    return weights


def _per_side(arr: np.ndarray, weights: dict, heavy: str) -> tuple[list, list]:
    heavy_sum = [0, 0]
    light_sum = [0, 0]
    for color, w in weights.items():
        mask = _color_mask(arr, RGB[color])
        comps = _components(mask, MIN_DISK)
        for comp in comps:
            if comp["cy"] < 240:
                continue
            side = 0 if comp["cx"] < MID_X else 1
            if color == heavy:
                heavy_sum[side] += w
            else:
                light_sum[side] += w
    return heavy_sum, light_sum


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    weights = _read_weights(arr)
    heavy = max(weights, key=weights.__getitem__)
    heavy_sum, light_sum = _per_side(arr, weights, heavy)
    for side in (0, 1):
        if heavy_sum[side] == 0 or light_sum[side] == 0:
            raise ValueError("a box lacks heavy or light disks")
    # compare heavyA/lightA vs heavyB/lightB by cross-multiplication
    left = heavy_sum[0] * light_sum[1]
    right = heavy_sum[1] * light_sum[0]
    if left == right:
        raise ValueError("dominance ratios are indistinguishable")
    return "A" if left > right else "B"
