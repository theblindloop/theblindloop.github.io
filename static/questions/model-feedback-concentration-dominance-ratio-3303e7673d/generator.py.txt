"""Deterministic renderer and analytic arm for the concentration-dominance ratio world.

A key strip across the top shows three colour swatches; beside each swatch a row
of one, two, or three dark dots encodes that colour's weight (three dots is the
heaviest colour).  Down the card stand two framed boxes, A on the left and B on
the right, each holding a few coloured disks of these three colours.  The
operation is a quantitative relation on each box's two colour classes: a disk is
HEAVY when its colour is the heaviest colour (weight three), otherwise it is
LIGHT.  For each box compute the heavy mass = the weight of the box's heavy
disks and the light mass = the weight of the box's light disks, then take the
ratio heavy mass / light mass as the box's dominance (concentration).  Decide
which of the two boxes carries the larger dominance ratio.

The sampler always builds scenes in which the box with the LARGER dominance
ratio has the STRICTLY SMALLER heavy mass (and therefore also the smaller total
mass), so a model that just sums heavy disks, sums total weight, or compares
"heaviness" selects the wrong box.  Only the ratio of the two composed colour
classes decides the answer.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 780
HEIGHT = 840
COLORS = ["crimson", "teal", "amber"]
RGB = {
    "crimson": (204, 44, 64),
    "teal": (0, 140, 150),
    "amber": (232, 158, 32),
}
DARK = (45, 45, 45)
OUTLINE = (70, 70, 70)
BACKGROUND = (250, 250, 250)
SWATCH_XS = [130, 390, 650]
KEY_CY = 120
SWATCH_SIZE = 44
DOT_R = 9
DOT_SLOT = 26
DOT_X0_OFFSET = 14
DISK_R = 32
LEFT_CENTERS = [(155, 440), (155, 560), (275, 440), (275, 560)]
RIGHT_CENTERS = [(505, 440), (505, 560), (625, 440), (625, 560)]
MIN_GAP = 0.5
MAX_DISKS = 4


def _load_font(size: int):
    for path in (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf",
        "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
    ):
        try:
            return ImageFont.truetype(path, size)
        except OSError:
            continue
    return ImageFont.load_default()


def _heavy_color(weights: dict) -> str:
    return max(weights, key=weights.__getitem__)


def _class_sums(colors: list, weights: dict, heavy: str) -> tuple[int, int]:
    """Return (heavy mass, light mass) for a box given its disks."""
    heavy_mass = sum(int(weights[c]) for c in colors if c == heavy)
    light_mass = sum(int(weights[c]) for c in colors if c != heavy)
    return heavy_mass, light_mass


def _class_masses(scene: dict):
    weights = scene["weights"]
    heavy = _heavy_color(weights)
    return _class_sums(scene["box_a"], weights, heavy), _class_sums(
        scene["box_b"], weights, heavy
    )


def _ratios(scene: dict) -> tuple[float, float]:
    (ha, la), (hb, lb) = _class_masses(scene)
    return ha / la, hb / lb


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    for _ in range(20000):
        weights = [1, 2, 3]
        rng.shuffle(weights)
        weight_of = {color: int(w) for color, w in zip(COLORS, weights)}
        heavy = _heavy_color(weight_of)
        box_a = [rng.choice(COLORS) for _ in range(rng.choice([2, 3, 4]))]
        box_b = [rng.choice(COLORS) for _ in range(rng.choice([2, 3, 4]))]
        if heavy not in box_a or heavy not in box_b:
            continue
        if all(c == heavy for c in box_a) or all(c == heavy for c in box_b):
            continue
        (ha, la), (hb, lb) = _class_sums(box_a, weight_of, heavy), _class_sums(
            box_b, weight_of, heavy
        )
        ra = ha / la
        rb = hb / lb
        if abs(ra - rb) < MIN_GAP:
            continue
        if (ra > rb and not ha < hb) or (rb > ra and not hb < ha):
            continue
        return {
            "canvas_size": [WIDTH, HEIGHT],
            "weights": weight_of,
            "box_a": box_a,
            "box_b": box_b,
            "disk_r": DISK_R,
        }
    raise RuntimeError("could not sample a valid concentration-dominance scene")


def _draw_key(draw: ImageDraw.ImageDraw, weights: dict) -> None:
    for cx, color in zip(SWATCH_XS, COLORS):
        x0, y0 = cx - SWATCH_SIZE // 2, KEY_CY - SWATCH_SIZE // 2
        draw.rectangle(
            (x0, y0, x0 + SWATCH_SIZE, y0 + SWATCH_SIZE), fill=RGB[color], outline=OUTLINE
        )
        weight = int(weights[color])
        for k in range(weight):
            dot_cx = cx + SWATCH_SIZE // 2 + DOT_X0_OFFSET + k * DOT_SLOT
            draw.ellipse(
                (dot_cx - DOT_R, KEY_CY - DOT_R, dot_cx + DOT_R, KEY_CY + DOT_R), fill=DARK
            )


def _draw_box(draw: ImageDraw.ImageDraw, x0: int, x1: int) -> None:
    draw.rectangle((x0, 300, x1, 700), outline=OUTLINE, width=4)


def _draw_disks(draw: ImageDraw.ImageDraw, colors: list, centers: list) -> None:
    for color, (cx, cy) in zip(colors, centers):
        r = DISK_R
        draw.ellipse((cx - r, cy - r, cx + r, cy + r), fill=RGB[color], outline=OUTLINE)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_key(draw, scene["weights"])
    _draw_box(draw, 70, 360)
    _draw_box(draw, 420, 710)
    _draw_disks(draw, scene["box_a"], LEFT_CENTERS[: len(scene["box_a"])])
    _draw_disks(draw, scene["box_b"], RIGHT_CENTERS[: len(scene["box_b"])])
    font = _load_font(30)
    draw.text((205, 745), "A", font=font, fill=(40, 40, 40))
    draw.text((555, 745), "B", font=font, fill=(40, 40, 40))
    return image


def analytic_gold(scene: dict) -> str:
    weights = scene["weights"]
    heavy = _heavy_color(weights)
    ha, la = _class_sums(scene["box_a"], weights, heavy)
    hb, lb = _class_sums(scene["box_b"], weights, heavy)
    # compare ha/la vs hb/lb by cross-multiplication
    if ha * lb > hb * la:
        return "A"
    return "B"


def margin(scene: dict) -> float:
    ra, rb = _ratios(scene)
    return float(abs(ra - rb))


def is_quarantined(scene: dict) -> bool:
    try:
        weights = scene["weights"]
        if set(weights) != set(COLORS):
            return True
        if sorted(int(weights[c]) for c in COLORS) != [1, 2, 3]:
            return True
        if not (2 <= len(scene["box_a"]) <= MAX_DISKS and 2 <= len(scene["box_b"]) <= MAX_DISKS):
            return True
        if any(c not in COLORS for c in scene["box_a"] + scene["box_b"]):
            return True
        heavy = _heavy_color(weights)
        if heavy not in scene["box_a"] or heavy not in scene["box_b"]:
            return True
        if all(c == heavy for c in scene["box_a"]) or all(c == heavy for c in scene["box_b"]):
            return True
        if margin(scene) < MIN_GAP:
            return True
        (ha, la), (hb, lb) = _class_masses(scene)
        ra = ha / la
        rb = hb / lb
        if not ((ra > rb and ha < hb) or (rb > ra and hb < ha)):
            return True
        return False
    except (KeyError, TypeError, ZeroDivisionError):
        return True


def latent_symmetries(scene: dict) -> list:
    return []
