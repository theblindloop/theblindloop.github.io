"""Deterministic renderer and analytic label for a forked route.

The visible object is one dark connected Y-shaped route. Both terminal paths
share a trunk from the green start to a junction and then diverge through
several visible interior segments. The gold compares the two polyline
centerline distances, not endpoint displacement.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "forked_route_terminal-0.1.0"
WIDTH = 512
HEIGHT = 384
BACKGROUND = (255, 255, 255)
ROUTE_COLOR = (30, 48, 66)
GREEN_COLOR = (36, 190, 92)
AMBER_COLOR = (244, 166, 42)
VIOLET_COLOR = (126, 88, 224)
ROUTE_WIDTH = 8.0
MARKER_RADIUS = 14.0
QUARANTINE_GAP = 30.0


def _distance(a: list[float] | tuple[float, float], b: list[float] | tuple[float, float]) -> float:
    return math.hypot(float(a[0]) - float(b[0]), float(a[1]) - float(b[1]))


def _path_length(points: list[list[float]]) -> float:
    return sum(_distance(a, b) for a, b in zip(points, points[1:]))


def _common_prefix(rng: random.Random) -> list[list[float]]:
    x0 = rng.uniform(108.0, 132.0)
    cy = rng.uniform(178.0, 205.0)
    start = [x0, cy + rng.uniform(-4.0, 4.0)]
    p1 = [x0 + rng.uniform(44.0, 55.0), cy + rng.uniform(-11.0, 11.0)]
    p2 = [x0 + rng.uniform(91.0, 103.0), cy + rng.uniform(-9.0, 9.0)]
    junction = [x0 + rng.uniform(138.0, 151.0), cy + rng.uniform(-3.0, 3.0)]
    return [start, p1, p2, junction]


def _branch(
    prefix: list[list[float]],
    side: int,
    detour: float,
    rng: random.Random,
) -> list[list[float]]:
    junction = prefix[-1]
    x1 = rng.uniform(45.0, 51.0)
    x2 = rng.uniform(97.0, 103.0)
    x3 = rng.uniform(143.0, 151.0)
    y1 = rng.uniform(16.0, 21.0)
    y2 = rng.uniform(50.0, 56.0)
    y3 = rng.uniform(70.0, 77.0)
    points = [
        [junction[0] + x1, junction[1] + side * y1],
        [junction[0] + x2, junction[1] + side * (y2 + detour)],
        [junction[0] + x3, junction[1] + side * y3],
    ]
    return [*prefix, *points]


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced, fully visible route scene from one integer seed."""
    rng = random.Random(int(seed))
    prefix = _common_prefix(rng)
    # Color side is independent of which color receives the longer detour.
    amber_side = rng.choice((-1, 1))
    violet_side = -amber_side
    long_color = "amber" if int(seed) % 2 == 0 else "violet"
    near_band = int(seed) % 6 in (2, 5)
    detour = rng.uniform(34.0, 46.0) if near_band else rng.uniform(39.0, 61.0)
    amber_detour = detour if long_color == "amber" else 0.0
    violet_detour = detour if long_color == "violet" else 0.0
    amber_path = _branch(prefix, amber_side, amber_detour, rng)
    violet_path = _branch(prefix, violet_side, violet_detour, rng)
    # The two branch constructions use independent sub-jitters. Reject only a
    # pathological raster placement; ordinary label margins remain analytic.
    delta = _path_length(amber_path) - _path_length(violet_path)
    if (long_color == "amber" and delta < 0.0) or (long_color == "violet" and delta > 0.0):
        amber_path, violet_path = violet_path, amber_path
        # Swapping geometry also swaps visible terminal centers; the color-side
        # permutation remains independent of the label across seeds.
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
        "route_color": list(ROUTE_COLOR),
        "route_width": ROUTE_WIDTH,
        "marker_radius": MARKER_RADIUS,
        "green_color": list(GREEN_COLOR),
        "amber_color": list(AMBER_COLOR),
        "violet_color": list(VIOLET_COLOR),
        "amber_path": amber_path,
        "violet_path": violet_path,
    }


def _draw_path(
    draw: ImageDraw.ImageDraw,
    path: list[list[float]],
    color: tuple[int, int, int],
    width: int,
    scale: int,
) -> None:
    coords = [(round(point[0] * scale), round(point[1] * scale)) for point in path]
    draw.line(coords, fill=color, width=width, joint="curve")


def _draw_marker(
    draw: ImageDraw.ImageDraw,
    center: list[float],
    color: tuple[int, int, int],
    radius: float,
    scale: int,
) -> None:
    x, y = center
    box = (
        round((x - radius) * scale),
        round((y - radius) * scale),
        round((x + radius) * scale),
        round((y + radius) * scale),
    )
    draw.ellipse(box, fill=color)


def render(scene: dict[str, Any]) -> Image.Image:
    """Rasterize the route with fixed supersampling and no hidden payload."""
    scale = 4
    width = int(scene["width"])
    height = int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    route_color = tuple(scene["route_color"])
    route_width = max(1, round(float(scene["route_width"]) * scale))
    _draw_path(draw, scene["amber_path"], route_color, route_width, scale)
    _draw_path(draw, scene["violet_path"], route_color, route_width, scale)
    radius = float(scene["marker_radius"])
    _draw_marker(draw, scene["amber_path"][-1], tuple(scene["amber_color"]), radius, scale)
    _draw_marker(draw, scene["violet_path"][-1], tuple(scene["violet_color"]), radius, scale)
    _draw_marker(draw, scene["amber_path"][0], tuple(scene["green_color"]), radius, scale)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    amber_length = _path_length(scene["amber_path"])
    violet_length = _path_length(scene["violet_path"])
    return "amber" if amber_length >= violet_length else "violet"


def margin(scene: dict[str, Any]) -> float:
    return _path_length(scene["amber_path"]) - _path_length(scene["violet_path"])


def is_quarantined(scene: dict[str, Any]) -> bool:
    return abs(margin(scene)) < QUARANTINE_GAP


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
