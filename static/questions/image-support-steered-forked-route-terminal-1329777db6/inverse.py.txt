"""Independent image-only oracle for the forked route.

It segments the fixed dark route and the three fixed-color markers from the
final RGB image, then measures shortest centerline-like pixel paths from the
green marker to each terminal. It has no access to renderer state or latent
scene data.
"""

from __future__ import annotations

import heapq
import math

import numpy as np
from PIL import Image

ROUTE_RGB = np.asarray((30, 48, 66), dtype=np.float32)
GREEN_RGB = np.asarray((36, 190, 92), dtype=np.float32)
AMBER_RGB = np.asarray((244, 166, 42), dtype=np.float32)
VIOLET_RGB = np.asarray((126, 88, 224), dtype=np.float32)
COLOR_TOLERANCE = 62.0
ROUTE_TOLERANCE = 92.0
MARKER_SEARCH_RADIUS = 30.0


def _color_mask(array: np.ndarray, color: np.ndarray, tolerance: float) -> np.ndarray:
    return np.linalg.norm(array - color[None, None, :], axis=2) <= tolerance


def _center(mask: np.ndarray) -> tuple[float, float]:
    points = np.argwhere(mask)
    if len(points) < 20:
        raise ValueError("marker color is not recoverable from the image")
    # A single marker of each color is present, so its solid core centroid is
    # the most stable attachment reference under antialiasing.
    yx = points.astype(np.float64).mean(axis=0)
    return float(yx[1]), float(yx[0])


def _nearby_route(mask: np.ndarray, center: tuple[float, float]) -> np.ndarray:
    y, x = np.indices(mask.shape)
    local = mask & ((x - center[0]) ** 2 + (y - center[1]) ** 2 <= MARKER_SEARCH_RADIUS**2)
    points = np.argwhere(local)
    if len(points):
        return points
    points = np.argwhere(mask)
    if not len(points):
        raise ValueError("dark route is absent")
    distances = (points[:, 1] - center[0]) ** 2 + (points[:, 0] - center[1]) ** 2
    return points[np.argsort(distances)[:12]]


def _route_distances(mask: np.ndarray, source: tuple[float, float]) -> np.ndarray:
    """Dijkstra distance over the visible route pixels, seeded at a marker."""
    height, width = mask.shape
    distances = np.full((height, width), np.inf, dtype=np.float64)
    starts = _nearby_route(mask, source)
    heap: list[tuple[float, int, int]] = []
    for y, x in starts:
        cost = math.hypot(float(x) - source[0], float(y) - source[1])
        if cost < distances[y, x]:
            distances[y, x] = cost
            heapq.heappush(heap, (cost, int(y), int(x)))
    while heap:
        cost, y, x = heapq.heappop(heap)
        if cost != distances[y, x]:
            continue
        for dy in (-1, 0, 1):
            for dx in (-1, 0, 1):
                if not (dx or dy):
                    continue
                ny, nx = y + dy, x + dx
                if not (0 <= ny < height and 0 <= nx < width and mask[ny, nx]):
                    continue
                step = math.sqrt(2.0) if dx and dy else 1.0
                candidate = cost + step
                if candidate < distances[ny, nx]:
                    distances[ny, nx] = candidate
                    heapq.heappush(heap, (candidate, ny, nx))
    return distances


def _distance_to_marker(
    distances: np.ndarray,
    route_mask: np.ndarray,
    center: tuple[float, float],
) -> float:
    points = _nearby_route(route_mask, center)
    values = []
    for y, x in points:
        if np.isfinite(distances[y, x]):
            values.append(float(distances[y, x]) + math.hypot(float(x) - center[0], float(y) - center[1]))
    if not values:
        raise ValueError("marker is disconnected from the dark route")
    return min(values)


def decision_from_image(image: Image.Image) -> str:
    """Return the farther terminal using only the final rasterized PNG."""
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    green = _center(_color_mask(array, GREEN_RGB, COLOR_TOLERANCE))
    amber = _center(_color_mask(array, AMBER_RGB, COLOR_TOLERANCE))
    violet = _center(_color_mask(array, VIOLET_RGB, COLOR_TOLERANCE))
    route = _color_mask(array, ROUTE_RGB, ROUTE_TOLERANCE)
    if int(route.sum()) < 300:
        raise ValueError("insufficient connected dark route pixels")
    distances = _route_distances(route, green)
    amber_distance = _distance_to_marker(distances, route, amber)
    violet_distance = _distance_to_marker(distances, route, violet)
    return "amber" if amber_distance >= violet_distance else "violet"
