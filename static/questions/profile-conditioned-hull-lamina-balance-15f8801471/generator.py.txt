"""Deterministic latent sampler and raster renderer."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (250, 249, 246)
BLACK = (28, 31, 35)
RED = (220, 55, 58)
BLUE = (38, 105, 220)
DOT_RADIUS = 7
MARKER_RADIUS = 10
QUARANTINE_GAP = 12.0


def _cross(o: tuple[float, float], a: tuple[float, float], b: tuple[float, float]) -> float:
    return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])


def _convex_hull(points: list[list[int]] | list[tuple[float, float]]) -> list[tuple[float, float]]:
    pts = sorted({(float(x), float(y)) for x, y in points})
    if len(pts) < 3:
        raise ValueError("at least three distinct black points are required")
    lower: list[tuple[float, float]] = []
    for p in pts:
        while len(lower) >= 2 and _cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    upper: list[tuple[float, float]] = []
    for p in reversed(pts):
        while len(upper) >= 2 and _cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    hull = lower[:-1] + upper[:-1]
    if len(hull) < 3:
        raise ValueError("black points are collinear")
    return hull


def _polygon_centroid(poly: list[tuple[float, float]]) -> tuple[float, float]:
    twice_area = 0.0
    cx_num = 0.0
    cy_num = 0.0
    for p, q in zip(poly, poly[1:] + poly[:1]):
        cross = p[0] * q[1] - q[0] * p[1]
        twice_area += cross
        cx_num += (p[0] + q[0]) * cross
        cy_num += (p[1] + q[1]) * cross
    if abs(twice_area) < 1e-9:
        raise ValueError("degenerate hull")
    return cx_num / (3.0 * twice_area), cy_num / (3.0 * twice_area)


def _score(scene: dict) -> tuple[float, float]:
    center = _polygon_centroid(_convex_hull(scene["black_points"]))
    red = scene["red_marker"]
    blue = scene["blue_marker"]
    dr = math.hypot(red[0] - center[0], red[1] - center[1])
    db = math.hypot(blue[0] - center[0], blue[1] - center[1])
    return dr, db


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    center_x = rng.randint(225, 287)
    center_y = rng.randint(225, 287)
    radius_x = rng.randint(128, 178)
    radius_y = rng.randint(112, 168)
    rotation = rng.uniform(0.0, 2.0 * math.pi)
    count = 12 + 2 * (abs(int(seed)) % 3)
    points: list[list[int]] = []
    for index in range(count):
        theta = rotation + 2.0 * math.pi * index / count + rng.uniform(-0.055, 0.055)
        radial = rng.uniform(0.86, 1.0)
        x = round(center_x + radius_x * radial * math.cos(theta))
        y = round(center_y + radius_y * radial * math.sin(theta))
        points.append([x, y])

    # Shuffle representation order. Geometry and rendering are order-invariant.
    rng.shuffle(points)
    centroid = _polygon_centroid(_convex_hull(points))
    correct = [round(centroid[0]), round(centroid[1])]
    decoy_distance = rng.uniform(14.5, 19.0) if abs(int(seed)) % 7 == 0 else rng.uniform(48.0, 88.0)
    decoy = None
    for _ in range(100):
        theta = rng.uniform(0.0, 2.0 * math.pi)
        trial = [
            round(centroid[0] + decoy_distance * math.cos(theta)),
            round(centroid[1] + decoy_distance * math.sin(theta)),
        ]
        if (
            20 <= trial[0] < WIDTH - 20
            and 20 <= trial[1] < HEIGHT - 20
            and all(math.hypot(trial[0] - x, trial[1] - y) > 24 for x, y in points)
        ):
            decoy = trial
            break
    if decoy is None:
        raise RuntimeError("could not place distractor marker")

    if int(seed) % 2 == 0:
        red_marker, blue_marker = correct, decoy
    else:
        red_marker, blue_marker = decoy, correct
    return {
        "black_points": points,
        "red_marker": red_marker,
        "blue_marker": blue_marker,
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x, y in scene["black_points"]:
        draw.ellipse(
            (x - DOT_RADIUS, y - DOT_RADIUS, x + DOT_RADIUS, y + DOT_RADIUS),
            fill=BLACK,
        )
    red_marker = scene["red_marker"]
    blue_marker = scene["blue_marker"]
    for marker, color in ((red_marker, RED), (blue_marker, BLUE)):
        x, y = marker
        draw.ellipse(
            (x - MARKER_RADIUS, y - MARKER_RADIUS, x + MARKER_RADIUS, y + MARKER_RADIUS),
            fill=color,
        )
        draw.ellipse((x - 4, y - 4, x + 4, y + 4), fill=BACKGROUND)
    return image


def analytic_gold(scene: dict) -> str:
    red_distance, blue_distance = _score(scene)
    return "red" if red_distance < blue_distance else "blue"


def margin(scene: dict) -> float:
    red_distance, blue_distance = _score(scene)
    return abs(red_distance - blue_distance)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = {
        "black_points": list(reversed(scene["black_points"])),
        "red_marker": list(scene["red_marker"]),
        "blue_marker": list(scene["blue_marker"]),
    }
    return [("reverse_black_point_order", twin)]
