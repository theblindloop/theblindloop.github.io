"""Independent image-only inverse arm using only final RGB pixels."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

BLACK = np.array((28, 31, 35), dtype=np.uint8)
RED = np.array((220, 55, 58), dtype=np.uint8)
BLUE = np.array((38, 105, 220), dtype=np.uint8)


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[float, float, int]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            cy, cx = queue.popleft()
            xs.append(cx)
            ys.append(cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        found.append((sum(xs) / len(xs), sum(ys) / len(ys), len(xs)))
    return found


def _single_marker(array: np.ndarray, color: np.ndarray) -> tuple[float, float]:
    components = [item for item in _components(np.all(array == color, axis=2)) if item[2] >= 120]
    if len(components) != 1:
        raise ValueError("expected exactly one visible marker of each color")
    return components[0][0], components[0][1]


def _cross(o: tuple[float, float], a: tuple[float, float], b: tuple[float, float]) -> float:
    return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])


def _hull(points: list[tuple[float, float]]) -> list[tuple[float, float]]:
    pts = sorted(set(points))
    if len(pts) < 3:
        raise ValueError("not enough black dots")
    lower: list[tuple[float, float]] = []
    for p in pts:
        while len(lower) >= 2 and _cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    upper: list[tuple[float, float]] = []
    for p in reversed(pts):
        while len(upper) >= 2 and _cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    result = lower[:-1] + upper[:-1]
    if len(result) < 3:
        raise ValueError("degenerate black-dot hull")
    return result


def _area_centroid(poly: list[tuple[float, float]]) -> tuple[float, float]:
    area2 = cx = cy = 0.0
    for p, q in zip(poly, poly[1:] + poly[:1]):
        cross = p[0] * q[1] - q[0] * p[1]
        area2 += cross
        cx += (p[0] + q[0]) * cross
        cy += (p[1] + q[1]) * cross
    if abs(area2) < 1e-9:
        raise ValueError("degenerate hull area")
    return cx / (3.0 * area2), cy / (3.0 * area2)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    black = [item for item in _components(np.all(array == BLACK, axis=2)) if item[2] >= 120]
    if len(black) < 6:
        raise ValueError("insufficient global black-dot evidence")
    dot_centers = [(item[0], item[1]) for item in black]
    center = _area_centroid(_hull(dot_centers))
    red = _single_marker(array, RED)
    blue = _single_marker(array, BLUE)
    red_distance = math.hypot(red[0] - center[0], red[1] - center[1])
    blue_distance = math.hypot(blue[0] - center[0], blue[1] - center[1])
    if abs(red_distance - blue_distance) < 6.0:
        raise ValueError("ambiguous raster decision")
    return "red" if red_distance < blue_distance else "blue"
