"""Renderer for the twin comb ledger search world.

Two ledgers of four combs are pinned against opposite side margins of an
otherwise empty page. A comb is three vertical teeth of equal width standing on
one common baseline; its signature is the triple of tooth heights read left to
right. Exactly one comb in the left ledger carries the same signature as one
comb in the right ledger, and the answer is that left comb's row.

The decisive evidence is therefore two mutually disconnected marks that sit on
opposite sides of the page and are never joined by anything. Neither ledger
determines the answer on its own: every left comb is an equally plausible row
until the far ledger is read, and the sampler always plants a decoy pair that
agrees on two of its three teeth so that partial matching points elsewhere.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "twin_comb_ledger_search-renderer-0.1.0"

CANVAS_W = 768
CANVAS_H = 576

TOOTH_W = 14              # every tooth has this width, in pixels
TOOTH_GAP = 12            # horizontal gap between teeth of one comb
COMB_W = 3 * TOOTH_W + 2 * TOOTH_GAP
ROW_PITCH = 132           # vertical spacing between comb baselines
ROW_BASE = 130            # baseline of the top row before jitter
BASE_JITTER = 6
MIN_H = 34
MAX_H = 92
MARGIN_MIN = 44           # side margin range for either ledger
MARGIN_MAX = 74
SPECK = 3
CLEARANCE = 16            # specks stay this far away from every tooth

WORDS = ("first", "second", "third", "fourth")
QUARANTINE_MARGIN = 10.0

PALETTES = (
    ((246, 244, 238), (38, 42, 50)),
    ((238, 242, 246), (34, 46, 62)),
    ((243, 240, 232), (46, 40, 34)),
    ((240, 240, 244), (30, 52, 48)),
)


def _distance(left: list, right: list) -> int:
    """Chebyshev distance between two comb signatures, in pixels."""

    return max(abs(int(a) - int(b)) for a, b in zip(left, right))


def _fresh_signature(rng: random.Random, taken: list, floor: int) -> list | None:
    """A signature at least ``floor`` from every accepted signature."""

    for _ in range(500):
        pick = [rng.randint(MIN_H, MAX_H) for _ in range(3)]
        if all(_distance(pick, other) >= floor for other in taken):
            return pick
    return None


def _shift(source: list, index: int, amount: int) -> list | None:
    """Copy of ``source`` with one tooth moved by ``amount`` pixels."""

    for signed in (amount, -amount):
        value = int(source[index]) + signed
        if MIN_H <= value <= MAX_H:
            out = [int(v) for v in source]
            out[index] = value
            return out
    return None


def _gap_target(scene_seed: int, rng: random.Random) -> int:
    """Intended runner-up signature gap, cycled for near-boundary coverage."""

    phase = scene_seed % 12
    if phase == 5:
        return rng.randint(6, 9)          # below quarantine
    if phase in (2, 8):
        return rng.randint(10, 13)        # just above quarantine
    return rng.randint(16, 24)


def _signatures(scene: dict) -> tuple:
    left = [None] * 4
    right = [None] * 4
    for comb in scene["combs_left"]:
        left[int(comb["slot"])] = [int(v) for v in comb["heights"]]
    for comb in scene["combs_right"]:
        right[int(comb["slot"])] = [int(v) for v in comb["heights"]]
    if any(item is None for item in left + right):
        raise ValueError("scene does not fill both ledgers")
    return left, right


def _pairs(scene: dict) -> tuple:
    """Matching left/right rows and the smallest competing signature gap."""

    left, right = _signatures(scene)
    hits = [
        (i, j)
        for i in range(4)
        for j in range(4)
        if _distance(left[i], right[j]) == 0
    ]
    if len(hits) != 1:
        raise ValueError("scene does not have a unique twin comb pair")
    winner = hits[0]
    competing = [
        _distance(left[i], right[j])
        for i in range(4)
        for j in range(4)
        if (i, j) != winner
    ]
    return winner, float(min(competing))


def sample_scene(seed: int) -> dict:
    """Deterministic scene: two far-apart ledgers sharing exactly one signature."""

    rng = random.Random(seed * 7919 + 13)
    answer_row = seed % 4
    ink_rank_target = (seed + seed // 4) % 4
    palette = rng.randrange(len(PALETTES))

    for _ in range(4000):
        gap = _gap_target(seed, rng)
        floor = gap + 4

        winner_sig = [rng.randint(MIN_H, MAX_H) for _ in range(3)]
        taken = [winner_sig]
        left = [None] * 4
        left[answer_row] = winner_sig
        spare_rows = [row for row in range(4) if row != answer_row]
        ok = True
        for row in spare_rows:
            pick = _fresh_signature(rng, taken, floor)
            if pick is None:
                ok = False
                break
            taken.append(pick)
            left[row] = pick
        if not ok:
            continue

        sums = sorted(sum(sig) for sig in left)
        if sums.index(sum(winner_sig)) != ink_rank_target:
            continue

        twin_row = rng.randrange(4)
        right = [None] * 4
        right[twin_row] = [int(v) for v in winner_sig]
        free_rows = [row for row in range(4) if row != twin_row]
        rng.shuffle(free_rows)

        # Decoy A: agrees with a losing left comb on two of its three teeth.
        decoy_source = left[rng.choice(spare_rows)]
        decoy_a = _shift(decoy_source, rng.randrange(3), gap)
        # Decoy B: agrees with the winning left comb on two of its three teeth.
        decoy_b = _shift(winner_sig, rng.randrange(3), gap + 8)
        if decoy_a is None or decoy_b is None:
            continue
        right[free_rows[0]] = decoy_a
        right[free_rows[1]] = decoy_b
        filler = _fresh_signature(rng, taken + [decoy_a, decoy_b], floor)
        if filler is None:
            continue
        right[free_rows[2]] = filler

        scene = {
            "width": CANVAS_W,
            "height": CANVAS_H,
            "palette": palette,
            "left_x": rng.randint(MARGIN_MIN, MARGIN_MAX),
            "right_x": CANVAS_W - COMB_W - rng.randint(MARGIN_MIN, MARGIN_MAX),
            "combs_left": [
                {"slot": row, "heights": left[row], "dy": rng.randint(-BASE_JITTER, BASE_JITTER)}
                for row in range(4)
            ],
            "combs_right": [
                {"slot": row, "heights": right[row], "dy": rng.randint(-BASE_JITTER, BASE_JITTER)}
                for row in range(4)
            ],
            "specks": [],
        }
        try:
            (won_left, _), value = _pairs(scene)
        except ValueError:
            continue
        if won_left != answer_row or value != float(gap):
            continue
        _place_specks(scene, rng)
        return scene
    raise RuntimeError(f"failed to sample a twin comb ledger scene for seed {seed}")


def _place_specks(scene: dict, rng: random.Random) -> None:
    boxes = [box for box in tooth_boxes(scene)]
    specks: list = []
    wanted = rng.randint(10, 18)
    guard = 0
    while len(specks) < wanted and guard < 5000:
        guard += 1
        sx = rng.randint(8, CANVAS_W - 8 - SPECK)
        sy = rng.randint(8, CANVAS_H - 8 - SPECK)
        if any(
            x0 - CLEARANCE <= sx + SPECK and sx <= x1 + CLEARANCE
            and y0 - CLEARANCE <= sy + SPECK and sy <= y1 + CLEARANCE
            for x0, y0, x1, y1 in boxes
        ):
            continue
        if any(abs(sx - ox) < 4 * SPECK and abs(sy - oy) < 4 * SPECK for ox, oy in specks):
            continue
        specks.append([sx, sy])
    scene["specks"] = specks


def comb_boxes(scene: dict, comb: dict, side: str) -> list:
    """Pixel boxes of the three teeth of one comb."""

    x0 = int(scene["left_x"]) if side == "left" else int(scene["right_x"])
    baseline = ROW_BASE + int(comb["slot"]) * ROW_PITCH + int(comb["dy"])
    out = []
    for index, height in enumerate(comb["heights"]):
        left = x0 + index * (TOOTH_W + TOOTH_GAP)
        out.append((left, baseline - int(height) + 1, left + TOOTH_W - 1, baseline))
    return out


def tooth_boxes(scene: dict) -> list:
    """Every tooth box in the scene."""

    out = []
    for comb in scene["combs_left"]:
        out.extend(comb_boxes(scene, comb, "left"))
    for comb in scene["combs_right"]:
        out.extend(comb_boxes(scene, comb, "right"))
    return out


def analytic_gold(scene: dict) -> str:
    """Row of the left ledger whose signature is repeated in the right ledger."""

    (row, _), _ = _pairs(scene)
    return WORDS[row]


def margin(scene: dict) -> float:
    """Smallest competing signature gap between a left comb and a right comb."""

    _, value = _pairs(scene)
    return value


def is_quarantined(scene: dict) -> bool:
    """Near-boundary scenes where a decoy pair nearly repeats a signature."""

    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list:
    """Pixel-identical, answer-preserving rewrites of the latent scene."""

    out = []
    if len(scene["specks"]) > 1:
        flipped = dict(scene)
        flipped["specks"] = list(reversed(scene["specks"]))
        out.append(("speck_order_reversed", flipped))
    if len(scene["combs_left"]) > 1:
        rotated = dict(scene)
        rotated["combs_left"] = scene["combs_left"][1:] + scene["combs_left"][:1]
        out.append(("left_comb_record_order_rotated", rotated))
    if len(scene["combs_right"]) > 1:
        reversed_right = dict(scene)
        reversed_right["combs_right"] = list(reversed(scene["combs_right"]))
        out.append(("right_comb_record_order_reversed", reversed_right))
    return out


def render(scene: dict) -> Image.Image:
    """Rasterise the scene. Marks never overlap, so drawing order is irrelevant."""

    background, ink = PALETTES[int(scene["palette"]) % len(PALETTES)]
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), background)
    draw = ImageDraw.Draw(image)
    for sx, sy in scene["specks"]:
        draw.rectangle((sx, sy, sx + SPECK - 1, sy + SPECK - 1), fill=ink)
    for box in tooth_boxes(scene):
        draw.rectangle(box, fill=ink)
    return image
