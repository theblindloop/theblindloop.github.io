"""Deterministic renderer and analytic arm for alternating-link tracing."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
SUPERSAMPLE = 3
NODE_RADIUS = 11
INK = (35, 40, 48)
BACKGROUND = (255, 255, 255)
NEUTRAL = (218, 230, 240)
START = (38, 166, 91)
TERMINAL_RGB = {
    "red": (218, 62, 62),
    "blue": (52, 105, 205),
    "orange": (235, 142, 38),
}
COLORS = ("red", "blue", "orange")
TARGET = ("solid", "dashed", "solid", "dashed", "solid")
DISTRACTORS = (
    ("solid", "solid", "dashed", "dashed", "solid"),
    ("solid", "dashed", "dashed", "solid", "solid"),
)


def _transform_point(point: tuple[float, float], transform: int, dx: int, dy: int) -> list[int]:
    x, y = point
    x -= 256.0
    y -= 256.0
    if transform >= 4:
        x = -x
    for _ in range(transform % 4):
        x, y = -y, x
    return [int(round(x + 256.0 + dx)), int(round(y + 256.0 + dy))]


def sample_scene(seed: int) -> dict:
    """Sample a cue-balanced three-route fan from one integer seed."""
    rng = random.Random(int(seed) ^ 0x6A09E667)
    answer = COLORS[int(seed) % 3]
    target_slot = (int(seed) // 3) % 3
    transform = (int(seed) // 9) % 8
    dx = rng.randint(-3, 3)
    dy = rng.randint(-3, 3)

    start = (70.0, 256.0)
    radii = [78.0, 154.0, 230.0, 306.0, 382.0]
    base_wiggles = [10.0, -12.0, 14.0, -10.0, 0.0]
    # Shared perturbations preserve equal route length across the three slots.
    radii = [r + rng.uniform(-2.5, 2.5) for r in radii]
    wiggles = [w + rng.uniform(-2.5, 2.5) for w in base_wiggles]
    spread = rng.uniform(25.0, 28.0)
    angles = (-spread, 0.0, spread)

    other_colors = [color for color in COLORS if color != answer]
    rng.shuffle(other_colors)
    terminal_colors: list[str | None] = [None, None, None]
    terminal_colors[target_slot] = answer
    for slot, color in zip((i for i in range(3) if i != target_slot), other_colors):
        terminal_colors[slot] = color

    other_styles = list(DISTRACTORS)
    rng.shuffle(other_styles)
    styles_by_slot: list[tuple[str, ...] | None] = [None, None, None]
    styles_by_slot[target_slot] = TARGET
    for slot, styles in zip((i for i in range(3) if i != target_slot), other_styles):
        styles_by_slot[slot] = styles

    paths = []
    for slot, angle_deg in enumerate(angles):
        a = math.radians(angle_deg)
        ca, sa = math.cos(a), math.sin(a)
        points: list[list[int]] = [_transform_point(start, transform, dx, dy)]
        for radius, wiggle in zip(radii, wiggles):
            x = start[0] + radius * ca - wiggle * sa
            y = start[1] + radius * sa + wiggle * ca
            points.append(_transform_point((x, y), transform, dx, dy))
        paths.append(
            {
                "terminal": terminal_colors[slot],
                "points": points,
                "styles": list(styles_by_slot[slot] or ()),
            }
        )
    rng.shuffle(paths)
    return {"paths": paths}


def _draw_dashed(draw: ImageDraw.ImageDraw, a: tuple[int, int], b: tuple[int, int], width: int) -> None:
    dx = b[0] - a[0]
    dy = b[1] - a[1]
    length = math.hypot(dx, dy)
    ux, uy = dx / length, dy / length
    dash = 13 * SUPERSAMPLE
    gap = 9 * SUPERSAMPLE
    pos = 0.0
    total = length * SUPERSAMPLE
    while pos < total:
        end = min(pos + dash, total)
        draw.line(
            (
                a[0] * SUPERSAMPLE + ux * pos,
                a[1] * SUPERSAMPLE + uy * pos,
                a[0] * SUPERSAMPLE + ux * end,
                a[1] * SUPERSAMPLE + uy * end,
            ),
            fill=INK,
            width=width,
        )
        pos += dash + gap


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH * SUPERSAMPLE, HEIGHT * SUPERSAMPLE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    paths = scene["paths"]
    line_width = 7 * SUPERSAMPLE
    for path in paths:
        points = path["points"]
        for a, b, style in zip(points, points[1:], path["styles"]):
            if style == "solid":
                draw.line(
                    tuple(v * SUPERSAMPLE for point in (a, b) for v in point),
                    fill=INK,
                    width=line_width,
                )
            elif style == "dashed":
                _draw_dashed(draw, tuple(a), tuple(b), line_width)
            else:
                raise ValueError(f"unknown link style {style!r}")

    radius = NODE_RADIUS * SUPERSAMPLE
    # Draw each non-START node once; paths are disjoint away from START.
    for path in paths:
        for index, point in enumerate(path["points"][1:], start=1):
            fill = TERMINAL_RGB[path["terminal"]] if index == 5 else NEUTRAL
            x, y = point[0] * SUPERSAMPLE, point[1] * SUPERSAMPLE
            draw.ellipse(
                (x - radius, y - radius, x + radius, y + radius),
                fill=fill,
                outline=INK,
                width=3 * SUPERSAMPLE,
            )
    sx, sy = paths[0]["points"][0]
    sx *= SUPERSAMPLE
    sy *= SUPERSAMPLE
    draw.ellipse(
        (sx - radius, sy - radius, sx + radius, sy + radius),
        fill=START,
        outline=INK,
        width=3 * SUPERSAMPLE,
    )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    paths = scene["paths"]
    winners = [path["terminal"] for path in paths if tuple(path["styles"]) == TARGET]
    if len(winners) != 1:
        raise ValueError(f"expected one alternating route, found {len(winners)}")
    return str(winners[0])


def margin(scene: dict) -> float:
    paths = scene["paths"]
    errors = sorted(
        sum(actual != expected for actual, expected in zip(path["styles"], TARGET))
        for path in paths
    )
    return float(errors[1] - errors[0]) if len(errors) >= 2 else 0.0


def is_quarantined(scene: dict) -> bool:
    paths = scene["paths"]
    errors = [
        sum(actual != expected for actual, expected in zip(path["styles"], TARGET))
        for path in paths
    ]
    return len(errors) != 3 or errors.count(min(errors, default=0)) != 1 or margin(scene) <= 0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["paths"] = list(reversed(twin["paths"]))
    return [("reverse_route_storage_order", twin)]
