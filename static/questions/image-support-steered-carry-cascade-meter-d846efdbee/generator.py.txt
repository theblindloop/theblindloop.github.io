"""Renderer for the carry_cascade_meter world.

The picture holds one meter built from four separated tiles placed at the
corners of a square, with plain background between them:

* three lamp banks (top-left, top-right, bottom-left).  Each bank is a dark
  frame holding three round sockets in a vertical column; a socket is either
  lit (amber) or dark (pale).  The number of lit sockets is that bank's count,
  always 0, 1 or 2 before any tick is applied.
* one tick patch (bottom-right): a dark frame holding twelve small square
  slots, of which some are filled teal.  Each filled slot is one increment.

An increment adds one lit lamp to the ones bank (bottom-left).  A bank that
would be pushed past two lit lamps instead empties to zero and passes one
increment to the next bank up (bottom-left -> top-right -> top-left); the
top-left bank wraps to zero if it overflows in turn.  A crimson ring marks one
bank, and the decision is how many lamps that bank shows once every tick has
been applied.

Everything the analytic gold reads is painted: the lit sockets, the filled tick
slots, the crimson ring, the tile geometry and placement, and the inert grey
decoration.  `latent_symmetries` therefore returns an empty list and the
candidate declares `latent_alias.mode = not_applicable`.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "carry_cascade_meter-renderer-0.1.0"

CANVAS = 800
BACKGROUND = (246, 246, 242)
FRAME = (28, 30, 34)
OUTLINE = (122, 126, 132)
LAMP_ON = (240, 168, 32)
LAMP_OFF = (250, 250, 247)
TICK_ON = (44, 140, 176)
TICK_OFF = (250, 250, 247)
RING = (206, 42, 54)
DECOR = (188, 188, 184)

BASE = 3
SOCKETS = 3
TICK_SLOTS = 12
TICK_COLS = 4
TICK_ROWS = 3
MAX_TICKS = 10
FRAME_WIDTH = 3
PAD = 16
MIN_TILE_GAP = 202

_ANSWER_WORDS = ("zero", "one", "two")


# ---------------------------------------------------------------- sampling


def _final_digits(counts: list[int], ticks: int) -> list[int]:
    """Replay `ticks` increments over the three banks, most significant first."""
    total = (9 * counts[0] + 3 * counts[1] + counts[2] + ticks) % 27
    return [total // 9, (total // 3) % 3, total % 3]


def _reading(counts: list[int], ticks: int, ring: int) -> int:
    return _final_digits(counts, ticks)[ring]


def _sensitive(counts: list[int], ticks: int, ring: int) -> tuple[bool, ...]:
    """Which of the four tiles can flip the answer by one visible glyph."""
    base = _reading(counts, ticks, ring)
    flags = []
    for index in range(3):
        hit = False
        for delta in (-1, 1):
            value = counts[index] + delta
            if not 0 <= value < BASE:
                continue
            probe = list(counts)
            probe[index] = value
            if 9 * probe[0] + 3 * probe[1] + probe[2] + ticks > 26:
                continue
            hit = hit or _reading(probe, ticks, ring) != base
        flags.append(hit)
    tick_hit = False
    for delta in (-1, 1):
        value = ticks + delta
        if not 1 <= value <= MAX_TICKS:
            continue
        if 9 * counts[0] + 3 * counts[1] + counts[2] + value > 26:
            continue
        tick_hit = tick_hit or _reading(counts, value, ring) != base
    flags.append(tick_hit)
    return tuple(flags)


def _sample_state(rng: random.Random, target: int, ring: int) -> tuple[list[int], int]:
    """Rejection-sample a start state whose answer is `target` and whose
    separated tiles each carry decisive evidence."""
    fallback = None
    for _ in range(6000):
        counts = [rng.randrange(BASE) for _ in range(3)]
        ticks = rng.randint(1, MAX_TICKS)
        if 9 * counts[0] + 3 * counts[1] + counts[2] + ticks > 26:
            continue
        if _reading(counts, ticks, ring) != target:
            continue
        flags = _sensitive(counts, ticks, ring)
        needed = flags if ring == 0 else flags[1:]
        if all(needed):
            return counts, ticks
        if fallback is None and all(flags[1:]):
            fallback = (counts, ticks)
    if fallback is not None:
        return fallback
    raise RuntimeError("could not sample a decisive meter state")


def _lit_slots(rng: random.Random, count: int) -> list[int]:
    slots = [0] * SOCKETS
    for index in rng.sample(range(SOCKETS), count):
        slots[index] = 1
    return slots


def _decor(rng: random.Random, box: tuple[int, int, int, int]) -> list[dict]:
    """Inert light-grey marks, always clear of the meter's bounding box."""
    left, top, right, bottom = box
    marks: list[dict] = []
    attempts = 0
    wanted = rng.randint(7, 13)
    while len(marks) < wanted and attempts < 400:
        attempts += 1
        kind = rng.choice(("bar", "ring", "tick"))
        size = rng.randint(14, 46)
        x = rng.randint(12, CANVAS - 12 - size)
        y = rng.randint(12, CANVAS - 12 - size)
        if x + size + 14 > left and x - 14 < right and y + size + 14 > top and y - 14 < bottom:
            continue
        marks.append({"kind": kind, "x": x, "y": y, "size": size})
    return marks


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 104729)
    target = seed % BASE
    ring = 1 if (seed // BASE) % 4 == 3 else 0

    tier = seed % 8
    if tier == 0:
        ring_gap = 2
    elif tier in (1, 2):
        ring_gap = rng.choice((3, 4))
    else:
        ring_gap = rng.randint(5, 8)

    counts, ticks = _sample_state(rng, target, ring)

    tile_w = rng.randint(116, 124)
    tile_h = rng.randint(128, 136)
    gap_x = rng.randint(MIN_TILE_GAP, MIN_TILE_GAP + 12)
    gap_y = rng.randint(MIN_TILE_GAP, MIN_TILE_GAP + 12)
    span_w = 2 * tile_w + gap_x
    span_h = 2 * tile_h + gap_y
    x0 = (CANVAS - span_w) // 2 + rng.randint(-24, 24)
    y0 = (CANVAS - span_h) // 2 + rng.randint(-24, 24)

    socket_pitch = (tile_h - 2 * PAD) / SOCKETS
    socket_r = int((socket_pitch - rng.randint(6, 8)) / 2)
    tick_pitch = (tile_w - 2 * PAD) / TICK_COLS
    tick_side = int(tick_pitch - rng.randint(6, 7))

    scene = {
        "lit_slots": [_lit_slots(rng, value) for value in counts],
        "tick_slots": sorted(rng.sample(range(TICK_SLOTS), ticks)),
        "ring": ring,
        "x0": x0,
        "y0": y0,
        "tile_w": tile_w,
        "tile_h": tile_h,
        "gap_x": gap_x,
        "gap_y": gap_y,
        "socket_r": socket_r,
        "tick_side": tick_side,
        "ring_gap": ring_gap,
        "width": CANVAS,
        "height": CANVAS,
    }
    scene["decor"] = _decor(rng, _span_box(scene))
    return scene


# ---------------------------------------------------------------- geometry


def _span_box(scene: dict) -> tuple[int, int, int, int]:
    left = scene["x0"]
    top = scene["y0"]
    right = left + 2 * scene["tile_w"] + scene["gap_x"]
    bottom = top + 2 * scene["tile_h"] + scene["gap_y"]
    return left, top, right, bottom


def _tiles(scene: dict) -> list[tuple[int, int, int, int]]:
    """Tile boxes in fixed roles: top-left, top-right, bottom-left, ticks."""
    x0, y0 = scene["x0"], scene["y0"]
    tw, th = scene["tile_w"], scene["tile_h"]
    dx = tw + scene["gap_x"]
    dy = th + scene["gap_y"]
    return [
        (x0, y0, x0 + tw, y0 + th),
        (x0 + dx, y0, x0 + dx + tw, y0 + th),
        (x0, y0 + dy, x0 + tw, y0 + dy + th),
        (x0 + dx, y0 + dy, x0 + dx + tw, y0 + dy + th),
    ]


def _socket_centres(scene: dict, box: tuple[int, int, int, int]) -> list[tuple[float, float]]:
    left, top, right, bottom = box
    pitch = (bottom - top - 2 * PAD) / SOCKETS
    cx = (left + right) / 2.0
    return [(cx, top + PAD + (index + 0.5) * pitch) for index in range(SOCKETS)]


def _tick_boxes(scene: dict, box: tuple[int, int, int, int]) -> list[tuple[float, float]]:
    left, top, right, bottom = box
    pitch_x = (right - left - 2 * PAD) / TICK_COLS
    pitch_y = (bottom - top - 2 * PAD) / TICK_ROWS
    centres = []
    for index in range(TICK_SLOTS):
        row, col = divmod(index, TICK_COLS)
        centres.append(
            (left + PAD + (col + 0.5) * pitch_x, top + PAD + (row + 0.5) * pitch_y)
        )
    return centres


# ---------------------------------------------------------------- rendering


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(image)

    for mark in scene["decor"]:
        x, y, size = mark["x"], mark["y"], mark["size"]
        if mark["kind"] == "bar":
            draw.rectangle([x, y + size // 3, x + size, y + size // 3 + 5], fill=DECOR)
        elif mark["kind"] == "ring":
            draw.ellipse([x, y, x + size, y + size], outline=DECOR, width=3)
        else:
            draw.line([x, y + size, x + size, y], fill=DECOR, width=3)

    boxes = _tiles(scene)
    radius = scene["socket_r"]
    half = scene["tick_side"] / 2.0

    for index, box in enumerate(boxes):
        draw.rectangle(list(box), outline=FRAME, width=FRAME_WIDTH)
        if index < 3:
            for slot, (cx, cy) in enumerate(_socket_centres(scene, box)):
                fill = LAMP_ON if scene["lit_slots"][index][slot] else LAMP_OFF
                draw.ellipse(
                    [cx - radius, cy - radius, cx + radius, cy + radius],
                    fill=fill,
                    outline=OUTLINE,
                    width=2,
                )
        else:
            for slot, (cx, cy) in enumerate(_tick_boxes(scene, box)):
                fill = TICK_ON if slot in scene["tick_slots"] else TICK_OFF
                draw.rectangle(
                    [cx - half, cy - half, cx + half, cy + half],
                    fill=fill,
                    outline=OUTLINE,
                    width=2,
                )

    inset = FRAME_WIDTH + scene["ring_gap"]
    marked = boxes[scene["ring"]]
    draw.rectangle(
        [marked[0] + inset, marked[1] + inset, marked[2] - inset, marked[3] - inset],
        outline=RING,
        width=3,
    )
    return image


# ---------------------------------------------------------------- gold


def _counts(scene: dict) -> list[int]:
    return [sum(slots) for slots in scene["lit_slots"]]


def analytic_gold(scene: dict) -> str:
    counts = _counts(scene)
    ticks = len(scene["tick_slots"])
    return _ANSWER_WORDS[_reading(counts, ticks, scene["ring"])]


def margin(scene: dict) -> float:
    """Composite pixel margin: the smallest clearance or glyph half-extent whose
    loss would hide or merge one of the replay inputs."""
    boxes = _tiles(scene)
    pitch = (scene["tile_h"] - 2 * PAD) / SOCKETS
    socket_gap = pitch - 2 * scene["socket_r"]
    pitch_x = (scene["tile_w"] - 2 * PAD) / TICK_COLS
    pitch_y = (scene["tile_h"] - 2 * PAD) / TICK_ROWS
    tick_gap = min(pitch_x, pitch_y) - scene["tick_side"]
    ring_gap = float(scene["ring_gap"])
    inner_clear = PAD - FRAME_WIDTH - ring_gap
    left, top, right, bottom = _span_box(scene)
    edge_clear = min(left, top, scene["width"] - right, scene["height"] - bottom)
    return round(
        min(
            ring_gap,
            inner_clear,
            socket_gap,
            tick_gap,
            edge_clear / 8.0,
        ),
        4,
    )


def is_quarantined(scene: dict) -> bool:
    """Finite-resolution cases: the ring, the socket column or the tick lattice
    is too tight to be resolved with confidence."""
    return margin(scene) < 3.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases: every sampled field is painted (see candidate.json)."""
    return []
