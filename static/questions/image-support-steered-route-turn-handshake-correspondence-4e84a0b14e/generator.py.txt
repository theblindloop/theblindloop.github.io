"""Renderer for a connected route-turn handshake correspondence world.

The decisive object is one navy polyline that traverses three horizontally
separated fields.  Each field contains seven visible route stations and five
interior bends.  Candidate records are rendered by their explicit slot so
their list order is a harmless latent alias.
"""

from __future__ import annotations

import copy
import itertools
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

RENDERER_VERSION = "route-turn-handshake-correspondence-0.1.0"
WIDTH, HEIGHT = 960, 600
SCALE = 4
PAPER = (248, 246, 240)
INK = (28, 99, 132)
TEXT = (38, 55, 70)
GUIDE = (205, 214, 216)
SOFT = (230, 234, 232)

FIELD_X0 = (70, 360, 650)
FIELD_BASE = (230, 345, 460)
STATION_STEP = 32
STATION_COUNT = 7
TAIL_LENGTH = 24
SLOPE_DY = 8
SLOPE_VALUES = (-3, -2, -1, 1, 2, 3)
TURN_COUNT = STATION_COUNT - 2
QUARANTINE_MARGIN = 0.25


def _font(size: int):
    for name in (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
        "/usr/share/fonts/truetype/liberation2/LiberationSans-Regular.ttf",
    ):
        if Path(name).is_file():
            return ImageFont.truetype(name, size=size * SCALE)
    return ImageFont.load_default()


def _turn_signature(slopes: list[int] | tuple[int, ...]) -> tuple[int, ...]:
    if len(slopes) != STATION_COUNT - 1:
        raise ValueError("a route must have six station slopes")
    signature = []
    for left, right in zip(slopes[:-1], slopes[1:], strict=True):
        if right == left:
            raise ValueError("a route bend must be visibly nonzero")
        signature.append(1 if right > left else -1)
    return tuple(signature)


def _reverse_walk(signature: tuple[int, ...]) -> tuple[int, ...]:
    """Turn signs after walking the same polyline from right to left."""

    return tuple(-value for value in reversed(signature))


def _route_map(scene: dict) -> dict[str, list[int]]:
    return {
        str(record["slot"]): [int(value) for value in record["slopes"]]
        for record in scene["candidate_routes"]
    }


def sample_scene(seed: int) -> dict:
    """Sample balanced routes with equal edge-length multisets."""

    rng = random.Random(int(seed))
    reference = list(SLOPE_VALUES)
    rng.shuffle(reference)
    reference_signature = _turn_signature(reference)
    target_signature = _reverse_walk(reference_signature)
    permutations = [list(item) for item in itertools.permutations(SLOPE_VALUES)]
    matching = [item for item in permutations if _turn_signature(item) == target_signature]
    if not matching:
        raise RuntimeError("could not construct a reverse-walk match")
    matching_route = list(rng.choice(matching))

    wrong_pool = [
        item
        for item in permutations
        if _turn_signature(item) != target_signature
    ]
    if int(seed) % 7 == 0:
        # A visible near-boundary class: exactly one of five turn signs differs,
        # while the same slope multiset preserves route ink and edge lengths.
        near_pool = [
            item
            for item in wrong_pool
            if sum(
                a != b
                for a, b in zip(_reverse_walk(_turn_signature(item)), reference_signature, strict=True)
            )
            == 1
        ]
        wrong_pool = near_pool or wrong_pool
    else:
        # Keep the ordinary class comfortably away from the discrete boundary.
        max_distance = max(
            sum(
                a != b
                for a, b in zip(_reverse_walk(_turn_signature(item)), reference_signature, strict=True)
            )
            for item in wrong_pool
        )
        wrong_pool = [
            item
            for item in wrong_pool
            if sum(
                a != b
                for a, b in zip(_reverse_walk(_turn_signature(item)), reference_signature, strict=True)
            )
            == max_distance
        ]
    wrong_route = list(rng.choice(wrong_pool))
    matching_slot = "A" if rng.randrange(2) == 0 else "B"
    records = [
        {
            "slot": "A",
            "slopes": matching_route if matching_slot == "A" else wrong_route,
        },
        {
            "slot": "B",
            "slopes": matching_route if matching_slot == "B" else wrong_route,
        },
    ]
    rng.shuffle(records)
    return {
        "reference_slopes": [int(value) for value in reference],
        "candidate_routes": records,
    }


def analytic_gold(scene: dict) -> str:
    reference_signature = _turn_signature([int(value) for value in scene["reference_slopes"]])
    routes = _route_map(scene)
    scores = {
        slot: sum(
            left == right
            for left, right in zip(
                _reverse_walk(_turn_signature(routes[slot])), reference_signature, strict=True
            )
        )
        for slot in ("A", "B")
    }
    if scores["A"] == scores["B"]:
        raise ValueError("candidate turn handshakes are tied")
    return "A" if scores["A"] > scores["B"] else "B"


def margin(scene: dict) -> float:
    reference_signature = _turn_signature([int(value) for value in scene["reference_slopes"]])
    routes = _route_map(scene)
    scores = [
        sum(
            left == right
            for left, right in zip(
                _reverse_walk(_turn_signature(routes[slot])), reference_signature, strict=True
            )
        )
        for slot in ("A", "B")
    ]
    return abs(scores[0] - scores[1]) / float(TURN_COUNT)


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < QUARANTINE_MARGIN
    except (KeyError, TypeError, ValueError):
        return True


def _scaled_box(box: tuple[float, float, float, float]) -> tuple[int, int, int, int]:
    return tuple(int(round(value * SCALE)) for value in box)


def _station_points(x0: int, base: int, slopes: list[int]) -> list[tuple[float, float]]:
    points = [(float(x0), float(base))]
    y = float(base)
    for index, slope in enumerate(slopes, start=1):
        y += float(slope) * SLOPE_DY
        points.append((float(x0 + index * STATION_STEP), y))
    return points


def _draw_text(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font, fill) -> None:
    draw.text((int(xy[0] * SCALE), int(xy[1] * SCALE)), text, font=font, fill=fill)


def render(scene: dict) -> Image.Image:
    """Render a deterministic supersampled RGB raster."""

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), PAPER)
    draw = ImageDraw.Draw(image)
    title_font = _font(25)
    body_font = _font(15)
    label_font = _font(19)
    tiny_font = _font(11)

    _draw_text(draw, (40, 28), "ROUTE-TURN HANDSHAKE", title_font, TEXT)
    _draw_text(
        draw,
        (40, 68),
        "Trace each field left to right; walk candidates right to left before comparing bends.",
        body_font,
        TEXT,
    )
    draw.rounded_rectangle(
        _scaled_box((42, 110, 918, 558)),
        radius=14 * SCALE,
        outline=GUIDE,
        width=2 * SCALE,
    )

    field_labels = ("R  REFERENCE", "A  CANDIDATE", "B  CANDIDATE")
    for field_index, (x0, label) in enumerate(zip(FIELD_X0, field_labels, strict=True)):
        field_left = x0 - 25
        field_right = x0 + 6 * STATION_STEP + 25
        _draw_text(draw, (field_left, 126), label, label_font, TEXT)
        # Light station guides expose the seven measurement locations without
        # adding any pixels to the navy support used by the inverse.
        for station in range(STATION_COUNT):
            x = x0 + station * STATION_STEP
            draw.line(
                [(int(x * SCALE), 158 * SCALE), (int(x * SCALE), 535 * SCALE)],
                fill=SOFT,
                width=SCALE,
            )
        if field_index < 2:
            divider_x = x0 + 6 * STATION_STEP + 58
            draw.line(
                [(divider_x * SCALE, 120 * SCALE), (divider_x * SCALE, 548 * SCALE)],
                fill=GUIDE,
                width=SCALE,
            )

    routes = _route_map(scene)
    section_points = [
        _station_points(x0, base, routes.get(slot, [1, -1, 1, -1, 1, -1]))
        for x0, base, slot in (
            (FIELD_X0[0], FIELD_BASE[0], "R"),
            (FIELD_X0[1], FIELD_BASE[1], "A"),
            (FIELD_X0[2], FIELD_BASE[2], "B"),
        )
    ]
    section_points[0] = _station_points(FIELD_X0[0], FIELD_BASE[0], scene["reference_slopes"])

    route_points: list[tuple[float, float]] = list(section_points[0])
    route_points.extend(
        [
            (route_points[-1][0] + TAIL_LENGTH, route_points[-1][1]),
            (section_points[1][0][0] - TAIL_LENGTH, section_points[1][0][1]),
        ]
    )
    route_points.extend(section_points[1])
    route_points.extend(
        [
            (route_points[-1][0] + TAIL_LENGTH, route_points[-1][1]),
            (section_points[2][0][0] - TAIL_LENGTH, section_points[2][0][1]),
        ]
    )
    route_points.extend(section_points[2])
    draw.line(
        [(int(x * SCALE), int(y * SCALE)) for x, y in route_points],
        fill=INK,
        width=7 * SCALE,
        joint="curve",
    )
    # Rounded caps keep the global trace visibly continuous and do not encode
    # slot identity because the same cap is present at both global endpoints.
    radius = 3.5
    for x, y in (route_points[0], route_points[-1]):
        draw.ellipse(_scaled_box((x - radius, y - radius, x + radius, y + radius)), fill=INK)

    _draw_text(
        draw,
        (40, 570),
        "The navy route is continuous across all three fields; compare the five interior turns, not the connectors.",
        tiny_font,
        TEXT,
    )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Exercise the render-invariant permutation of slot-addressed records."""

    twin = copy.deepcopy(scene)
    twin["candidate_routes"] = list(reversed(twin["candidate_routes"]))
    return [("candidate_record_order", twin)]
