"""Independent RGB-only inverse for the connected route-turn question.

This module recovers seven route stations from the final raster, derives the
five signed bend orientations in each field, applies the public reverse-walk
rule, and chooses the higher whole-signature score.  It imports no renderer,
prompt, generator, verifier, or latent-scene code.
"""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

_INK = np.asarray((28, 99, 132), dtype=np.float32)
_FIELD_X0 = (70, 360, 650)
_FIELD_BASE = (230, 345, 460)
_STATION_STEP = 32
_STATION_COUNT = 7
_TURN_COUNT = 5
_COLOR_RADIUS = 72.0
_HALF_BAND = 72


def _route_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    distance = np.sqrt(np.sum((array - _INK) ** 2, axis=2))
    return distance < _COLOR_RADIUS


def _route_component(mask: np.ndarray) -> np.ndarray:
    """Keep the route only after proving it is one connected trace."""

    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components: list[list[tuple[int, int]]] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue: deque[tuple[int, int]] = deque([(y, x)])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dy or dx):
                        continue
                    ny, nx = cy + dy, cx + dx
                    if (
                        0 <= ny < height
                        and 0 <= nx < width
                        and mask[ny, nx]
                        and not seen[ny, nx]
                    ):
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        if len(points) >= 120:
            components.append(points)
    if len(components) != 1:
        raise ValueError("navy route is not one connected trace")
    component = np.zeros_like(mask, dtype=bool)
    ys, xs = zip(*components[0], strict=True)
    component[np.asarray(ys), np.asarray(xs)] = True
    return component


def _station_y(mask: np.ndarray, x: int, base: int) -> float:
    height, width = mask.shape
    left = max(0, x - 5)
    right = min(width, x + 6)
    top = max(0, base - _HALF_BAND)
    bottom = min(height, base + _HALF_BAND + 1)
    local = np.argwhere(mask[top:bottom, left:right])
    if len(local) < 12:
        raise ValueError("route station is missing or erased")
    y_values = local[:, 0].astype(np.float64) + float(top)
    observed = float(np.median(y_values))
    if not math.isfinite(observed):
        raise ValueError("route station measurement is non-finite")
    return observed


def _field_signature(mask: np.ndarray, x0: int, base: int) -> tuple[int, ...]:
    stations = [
        _station_y(mask, x0 + index * _STATION_STEP, base)
        for index in range(_STATION_COUNT)
    ]
    slopes = np.diff(np.asarray(stations, dtype=np.float64))
    signature = []
    for index in range(_TURN_COUNT):
        curvature = float(slopes[index + 1] - slopes[index])
        if abs(curvature) < 3.0:
            raise ValueError("route bend is ambiguous at raster resolution")
        signature.append(1 if curvature > 0 else -1)
    return tuple(signature)


def _reverse_walk(signature: tuple[int, ...]) -> tuple[int, ...]:
    return tuple(-value for value in reversed(signature))


def decision_from_image(image: Image.Image) -> str:
    """Return ``A`` or ``B`` from the final PNG alone."""

    mask = _route_component(_route_mask(image))
    if mask.ndim != 2 or mask.shape[1] < 900 or mask.shape[0] < 560:
        raise ValueError("unexpected raster dimensions")
    reference = _field_signature(mask, _FIELD_X0[0], _FIELD_BASE[0])
    candidate_a = _field_signature(mask, _FIELD_X0[1], _FIELD_BASE[1])
    candidate_b = _field_signature(mask, _FIELD_X0[2], _FIELD_BASE[2])
    scores = [
        sum(left == right for left, right in zip(_reverse_walk(candidate_a), reference, strict=True)),
        sum(left == right for left, right in zip(_reverse_walk(candidate_b), reference, strict=True)),
    ]
    if scores[0] == scores[1]:
        raise ValueError("candidate turn handshakes are tied")
    return "A" if scores[0] > scores[1] else "B"
