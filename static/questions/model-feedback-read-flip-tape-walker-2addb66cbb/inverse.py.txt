"""Independent image-only inverse for the tape-walker board."""

from __future__ import annotations

import numpy as np
from PIL import Image

EXPECTED_SIZE = (820, 650)
TAPE_WIDTH, TURNS = 8, 8
CELL, GAP, FIRST_X = 44, 8, 275
START_TOP = 70
OPTION_TOPS = (210, 292, 374, 456, 538)
INK = np.array((24, 59, 86), dtype=np.float32)
PAPER = np.array((248, 246, 238), dtype=np.float32)
GOLD = np.array((213, 141, 22), dtype=np.float32)


class OracleError(ValueError):
    pass


def _classify_cell(array: np.ndarray, column: int, top: int) -> int:
    left = FIRST_X + column * (CELL + GAP)
    patch = array[top + 12 : top + 32, left + 12 : left + 32].astype(np.float32)
    mean = patch.mean(axis=(0, 1))
    ink_distance = float(np.linalg.norm(mean - INK))
    paper_distance = float(np.linalg.norm(mean - PAPER))
    if ink_distance < 8.0 and ink_distance + 25.0 < paper_distance:
        return 1
    if paper_distance < 8.0 and paper_distance + 25.0 < ink_distance:
        return 0
    raise OracleError(f"cell {column} at y={top} has no clean binary fill")


def _read_pointer(array: np.ndarray, top: int) -> tuple[int, int]:
    counts: list[int] = []
    masks: list[np.ndarray] = []
    for column in range(TAPE_WIDTH):
        cx = FIRST_X + column * (CELL + GAP) + CELL // 2
        patch = array[top - 29 : top - 3, cx - 23 : cx + 24].astype(np.float32)
        mask = np.linalg.norm(patch - GOLD, axis=2) < 18.0
        masks.append(mask)
        counts.append(int(mask.sum()))
    active = [index for index, count in enumerate(counts) if count >= 100]
    if len(active) != 1 or any(count >= 22 for index, count in enumerate(counts) if index not in active):
        raise OracleError("missing or ambiguous gold pointer")
    pointer = active[0]
    mask = masks[pointer]
    xs = np.nonzero(mask)[1]
    centre = 23
    left_count = int((xs < centre - 2).sum())
    right_count = int((xs > centre + 2).sum())
    if right_count - left_count >= 45:
        direction = 1
    elif left_count - right_count >= 45:
        direction = -1
    else:
        raise OracleError("gold pointer facing is ambiguous")
    return pointer, direction


def _read_state(array: np.ndarray, top: int) -> dict:
    pointer, direction = _read_pointer(array, top)
    return {
        "bits": [_classify_cell(array, column, top) for column in range(TAPE_WIDTH)],
        "pointer": pointer,
        "direction": direction,
    }


def _validate_labels(array: np.ndarray) -> None:
    for top in OPTION_TOPS:
        cy = top + CELL // 2
        patch = array[cy - 18 : cy + 19, 135:172].astype(np.float32)
        navy_count = int((np.linalg.norm(patch - INK, axis=2) < 18.0).sum())
        white_count = int((patch.mean(axis=2) > 235).sum())
        if navy_count < 700 or white_count < 35:
            raise OracleError("an option label disk or glyph is missing")


def _advance(state: dict) -> dict:
    bits = list(state["bits"])
    pointer = int(state["pointer"])
    direction = int(state["direction"])
    old_value = bits[pointer]
    bits[pointer] = 0 if old_value else 1
    if old_value == 0:
        direction = -direction
        stride = 1
    else:
        stride = 2
    return {
        "bits": bits,
        "pointer": (pointer + stride * direction) % TAPE_WIDTH,
        "direction": direction,
    }


def _same(first: dict, second: dict) -> bool:
    return (
        first["bits"] == second["bits"]
        and first["pointer"] == second["pointer"]
        and first["direction"] == second["direction"]
    )


def decision_from_image(image: Image.Image) -> str:
    if image.size != EXPECTED_SIZE:
        raise OracleError(f"unexpected raster size {image.size}")
    array = np.asarray(image.convert("RGB"))
    _validate_labels(array)
    state = _read_state(array, START_TOP)
    options = [_read_state(array, top) for top in OPTION_TOPS]
    signatures = {(tuple(option["bits"]), option["pointer"], option["direction"]) for option in options}
    if len(signatures) != 5:
        raise OracleError("option states are not distinct")
    for _ in range(TURNS):
        state = _advance(state)
    matches = [index for index, option in enumerate(options) if _same(option, state)]
    if len(matches) != 1:
        raise OracleError("recovered state does not match exactly one option")
    return "ABCDE"[matches[0]]
