"""Deterministic sampler, renderer, and analytic tape-walker program."""

from __future__ import annotations

import random
from typing import Iterable, NamedTuple

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 820, 650
TAPE_WIDTH, TURNS = 8, 8
CELL, GAP, GRID_X = 44, 8, 275
START_TOP = 70
OPTION_TOPS = (210, 292, 374, 456, 538)
DARK = "#183b56"
LIGHT = "#f8f6ee"
BORDER = "#536879"
GOLD = "#d58d16"
BACKGROUND = "#ffffff"
FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"


class MachineState(NamedTuple):
    bits: tuple[int, ...]
    pointer: int
    direction: int


def _state(bits: Iterable[int], pointer: int, direction: int) -> MachineState:
    values = [int(value) for value in bits]
    if len(values) != TAPE_WIDTH or any(value not in (0, 1) for value in values):
        raise ValueError("a tape must contain eight binary cells")
    if not 0 <= int(pointer) < TAPE_WIDTH or int(direction) not in (-1, 1):
        raise ValueError("invalid pointer state")
    return MachineState(tuple(values), int(pointer), int(direction))


def _as_option(state: MachineState) -> dict:
    return {"bits": list(state.bits), "pointer": state.pointer, "direction": state.direction}


def _turn(state: MachineState) -> MachineState:
    current = _state(state.bits, state.pointer, state.direction)
    bits = list(current.bits)
    pointer = current.pointer
    direction = current.direction
    read = bits[pointer]
    bits[pointer] = 1 - read
    if read == 0:
        direction = -direction
        distance = 1
    else:
        distance = 2
    pointer = (pointer + direction * distance) % TAPE_WIDTH
    return _state(bits, pointer, direction)


def _evolve(state: MachineState, turns: int = TURNS) -> MachineState:
    result = _state(state.bits, state.pointer, state.direction)
    for _ in range(int(turns)):
        result = _turn(result)
    return result


def _wrong_post_flip(state: MachineState) -> MachineState:
    result = _state(state.bits, state.pointer, state.direction)
    for _ in range(TURNS):
        bits = list(result.bits)
        pointer = result.pointer
        direction = result.direction
        bits[pointer] = 1 - bits[pointer]
        if bits[pointer] == 0:
            direction = -direction
            distance = 1
        else:
            distance = 2
        result = _state(bits, (pointer + direction * distance) % TAPE_WIDTH, direction)
    return result


def _wrong_reverse_dark(state: MachineState) -> MachineState:
    result = _state(state.bits, state.pointer, state.direction)
    for _ in range(TURNS):
        bits = list(result.bits)
        pointer = result.pointer
        direction = result.direction
        read = bits[pointer]
        bits[pointer] = 1 - read
        if read == 1:
            direction = -direction
            distance = 1
        else:
            distance = 2
        result = _state(bits, (pointer + direction * distance) % TAPE_WIDTH, direction)
    return result


def _trace(state: MachineState) -> list[MachineState]:
    history = [_state(state.bits, state.pointer, state.direction)]
    for _ in range(TURNS):
        history.append(_turn(history[-1]))
    return history


def _valid_scene(scene: dict) -> tuple[MachineState, list[MachineState]]:
    required = {"start_bits", "start_pointer", "start_direction", "options"}
    if set(scene) != required:
        raise ValueError("scene fields do not match the declared label inputs")
    start = _state(scene["start_bits"], scene["start_pointer"], scene["start_direction"])
    options = [_state(option["bits"], option["pointer"], option["direction"]) for option in scene["options"]]
    if len(options) != 5 or len(set(options)) != 5:
        raise ValueError("five distinct composite options are required")
    return start, options


def _distance(first: MachineState, second: MachineState) -> int:
    return (
        sum(a != b for a, b in zip(first.bits, second.bits))
        + int(first.pointer != second.pointer)
        + int(first.direction != second.direction)
    )


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    for _ in range(1200):
        start = _state(
            [rng.randrange(2) for _ in range(TAPE_WIDTH)],
            rng.randrange(TAPE_WIDTH),
            rng.choice((-1, 1)),
        )
        history = _trace(start)
        visited = [state.pointer for state in history[:-1]]
        reads = [history[index].bits[history[index].pointer] for index in range(TURNS)]
        directions = {state.direction for state in history}
        wrap_used = any(
            abs(history[index + 1].pointer - history[index].pointer) > 2
            for index in range(TURNS)
        )
        target = history[-1]
        if not (
            2 <= sum(start.bits) <= 6
            and 2 <= sum(target.bits) <= 6
            and reads.count(0) >= 2
            and reads.count(1) >= 2
            and len(set(visited)) <= TURNS - 2
            and directions == {-1, 1}
            and wrap_used
        ):
            continue

        preferred = [history[-2], _evolve(start, TURNS + 1), _wrong_post_flip(start), _wrong_reverse_dark(start)]
        distractors: list[MachineState] = []
        for state in preferred:
            if state != target and state not in distractors:
                distractors.append(state)

        while len(distractors) < 4:
            bits = list(target.bits)
            pointer = target.pointer
            direction = target.direction
            mode = rng.randrange(3)
            if mode == 0:
                cell = rng.randrange(TAPE_WIDTH)
                bits[cell] ^= 1
            elif mode == 1:
                pointer = (pointer + rng.choice((1, 2, -1, -2))) % TAPE_WIDTH
            else:
                direction *= -1
            variant = _state(bits, pointer, direction)
            if variant != target and variant not in distractors:
                distractors.append(variant)

        distractors = distractors[:4]
        rng.shuffle(distractors)
        options = list(distractors)
        options.insert(int(seed) % 5, target)
        return {
            "start_bits": list(start.bits),
            "start_pointer": start.pointer,
            "start_direction": start.direction,
            "options": [_as_option(option) for option in options],
        }
    raise RuntimeError("could not sample a feedback-rich tape walk")


def analytic_gold(scene: dict) -> str:
    start, options = _valid_scene(scene)
    target = _evolve(start)
    matches = [index for index, option in enumerate(options) if option == target]
    if len(matches) != 1:
        raise ValueError("the final composite state must occur exactly once")
    return "ABCDE"[matches[0]]


def margin(scene: dict) -> float:
    start, options = _valid_scene(scene)
    target = _evolve(start)
    distances = [_distance(target, option) for option in options if option != target]
    return float(min(distances)) if distances else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        start, options = _valid_scene(scene)
        target = _evolve(start)
        return sum(option == target for option in options) != 1 or margin(scene) < 1.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    _valid_scene(scene)
    return []


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(FONT_PATH, size=size)


def _draw_pointer(draw: ImageDraw.ImageDraw, pointer: int, direction: int, top: int) -> None:
    cx = GRID_X + pointer * (CELL + GAP) + CELL // 2
    cy = top - 16
    if direction == 1:
        draw.line((cx - 18, cy, cx + 8, cy), fill=GOLD, width=6)
        draw.polygon(((cx + 19, cy), (cx + 7, cy - 10), (cx + 7, cy + 10)), fill=GOLD)
    else:
        draw.line((cx + 18, cy, cx - 8, cy), fill=GOLD, width=6)
        draw.polygon(((cx - 19, cy), (cx - 7, cy - 10), (cx - 7, cy + 10)), fill=GOLD)


def _draw_state(draw: ImageDraw.ImageDraw, state: MachineState, top: int, *, start: bool = False) -> None:
    checked = _state(state.bits, state.pointer, state.direction)
    for index, value in enumerate(checked.bits):
        left = GRID_X + index * (CELL + GAP)
        draw.rounded_rectangle(
            (left, top, left + CELL, top + CELL),
            radius=4,
            fill=DARK if value else LIGHT,
            outline=GOLD if start else BORDER,
            width=3,
        )
    _draw_pointer(draw, checked.pointer, checked.direction, top)


def render(scene: dict) -> Image.Image:
    start, options = _valid_scene(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((WIDTH // 2, 20), "GOLD ARROW = CURRENT CELL + FACING", fill=BORDER, font=_font(18), anchor="mm")
    draw.text((153, START_TOP + CELL / 2), "START", fill=GOLD, font=_font(25), anchor="mm")
    _draw_state(draw, start, START_TOP, start=True)
    draw.line((54, 145, WIDTH - 54, 145), fill="#d8dee3", width=2)
    draw.text((54, 168), "OPTIONS — complete state after 8 turns", fill=BORDER, font=_font(17), anchor="lm")
    for label, option, top in zip("ABCDE", options, OPTION_TOPS):
        centre = (153, top + CELL // 2)
        draw.ellipse((centre[0] - 22, centre[1] - 22, centre[0] + 22, centre[1] + 22), fill=DARK)
        draw.text(centre, label, fill="white", font=_font(25), anchor="mm")
        _draw_state(draw, option, top)
    return image
