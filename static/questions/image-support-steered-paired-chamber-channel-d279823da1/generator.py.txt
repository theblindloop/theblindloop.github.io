"""Renderer and analytic arm for the compact paired-chamber plaque."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
PALETTE = ("#176B87", "#7A3E9D", "#B45309", "#167A58")
BACKGROUNDS = ("#F7F3E8", "#F2F5F7", "#FFF8EE")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    radius = rng.randint(47, 53)
    chamber_radius = rng.randint(10, 12)
    # Keep a visible colored neck between the bridge and the two nuisance
    # chambers while retaining a thick, area-like connected opening.
    chamber_offset = radius - chamber_radius - 6
    bridge_half_width = chamber_radius + 2
    pad = radius + 20
    return {
        "orientation": "left-right" if int(seed) % 2 == 0 else "top-bottom",
        "cx": rng.randint(pad, WIDTH - pad),
        "cy": rng.randint(pad, HEIGHT - pad),
        "radius": radius,
        "chamber_radius": chamber_radius,
        "chamber_offset": chamber_offset,
        "bridge_half_width": bridge_half_width,
        "plaque_color": PALETTE[rng.randrange(len(PALETTE))],
        "background": BACKGROUNDS[rng.randrange(len(BACKGROUNDS))],
    }


def _ellipse(draw: ImageDraw.ImageDraw, cx: int, cy: int, r: int, fill: str, ss: int) -> None:
    draw.ellipse(((cx-r)*ss, (cy-r)*ss, (cx+r)*ss, (cy+r)*ss), fill=fill)


def render(scene: dict) -> Image.Image:
    ss = 4
    bg = scene["background"]
    img = Image.new("RGB", (WIDTH * ss, HEIGHT * ss), bg)
    draw = ImageDraw.Draw(img)
    cx, cy = int(scene["cx"]), int(scene["cy"])
    radius = int(scene["radius"])
    chamber_radius = int(scene["chamber_radius"])
    offset = int(scene["chamber_offset"])
    half_width = int(scene["bridge_half_width"])
    color = scene["plaque_color"]
    _ellipse(draw, cx, cy, radius, color, ss)
    centers = ((cx-offset, cy), (cx+offset, cy), (cx, cy-offset), (cx, cy+offset))
    for hx, hy in centers:
        _ellipse(draw, hx, hy, chamber_radius, bg, ss)
    if scene["orientation"] == "left-right":
        draw.rounded_rectangle(
            ((cx-offset)*ss, (cy-half_width)*ss, (cx+offset)*ss, (cy+half_width)*ss),
            radius=half_width*ss,
            fill=bg,
        )
    else:
        draw.rounded_rectangle(
            ((cx-half_width)*ss, (cy-offset)*ss, (cx+half_width)*ss, (cy+offset)*ss),
            radius=half_width*ss,
            fill=bg,
        )
    return img.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["orientation"])


def margin(scene: dict) -> float:
    # Approximate major-minus-minor extent of the connected opening.
    return float(2 * (int(scene["chamber_offset"]) + int(scene["chamber_radius"]) - int(scene["bridge_half_width"])))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 18.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
