"""Independent pixel-only inverse for the paired-chamber plaque."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int, bool]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        q = deque([(sy, sx)])
        seen[sy, sx] = True
        count = 0
        minx = maxx = sx
        miny = maxy = sy
        touches = False
        while q:
            y, x = q.popleft()
            count += 1
            minx, maxx = min(minx, x), max(maxx, x)
            miny, maxy = min(miny, y), max(maxy, y)
            touches = touches or x == 0 or y == 0 or x == w-1 or y == h-1
            for ny, nx in ((y-1,x),(y+1,x),(y,x-1),(y,x+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny,nx] and not seen[ny,nx]:
                    seen[ny,nx] = True
                    q.append((ny,nx))
        output.append((count, minx, miny, maxx, maxy, touches))
    return output


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    border = np.concatenate((arr[:4].reshape(-1,3), arr[-4:].reshape(-1,3), arr[:,:4].reshape(-1,3), arr[:,-4:].reshape(-1,3)))
    bg = np.median(border, axis=0)
    distance = np.linalg.norm(arr - bg, axis=2)
    ink = distance > 45.0
    ys, xs = np.nonzero(ink)
    if len(xs) < 800:
        raise ValueError("no substantial colored plaque")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    if x1-x0 < 70 or y1-y0 < 70:
        raise ValueError("plaque extent is incomplete")
    # Pixels close to the border-estimated background are pale void candidates.
    crop = distance[y0:y1+1, x0:x1+1] < 22.0
    enclosed = [c for c in _components(crop) if not c[5] and c[0] >= 120]
    if len(enclosed) < 3:
        raise ValueError("four-chamber organization is not recoverable")
    joined = max(enclosed, key=lambda c: c[0])
    _, minx, miny, maxx, maxy, _ = joined
    width = maxx-minx+1
    height = maxy-miny+1
    if max(width, height) < 45 or abs(width-height) < 18:
        raise ValueError("joined opening has no decisive axis")
    return "left-right" if width > height else "top-bottom"
