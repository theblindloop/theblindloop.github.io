"""Deterministic raster world for selecting a link on a visible cycle."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw


RENDERER_VERSION = "cycle_link_selection-0.1.0"
BACKGROUND = "#ffffff"
NETWORK_INK = "#25313b"
DECOR_INK = "#d8dee6"
COLORS = {
    "red": "#d94b48",
    "green": "#2b9d61",
    "blue": "#3d73c7",
}
SUPER = 4
QUARANTINE_SCALE = 0.99


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced compact graph with one labeled cycle edge."""

    rng = random.Random(int(seed))
    cycle_color = ("red", "green", "blue")[int(seed) % 3]
    tail_colors = [name for name in COLORS if name != cycle_color]
    rng.shuffle(tail_colors)
    return {
        "width": 512,
        "height": 512,
        # Choose several safe raster cells so absolute position is not a cue.
        "origin_x": rng.choice((192, 320)) + rng.choice((-2, -1, 0, 1, 2)),
        "origin_y": rng.choice((96, 224, 352)) + rng.choice((-2, -1, 0, 1, 2)),
        "cycle_color": cycle_color,
        "cycle_edge": rng.randrange(4),
        "tail_color_order": tail_colors,
        "scale": rng.choice((0.96, 1.00, 1.04)),
        "stroke_width": rng.choice((4, 5)),
        "node_radius": rng.choice((4.0, 4.5)),
        "decor_seed": rng.randrange(1, 2**31 - 1),
    }


def _point(scene: dict[str, Any], value: tuple[float, float]) -> tuple[int, int]:
    scale = float(scene["scale"])
    x = (float(scene["origin_x"]) + value[0] * scale) * SUPER
    y = (float(scene["origin_y"]) + value[1] * scale) * SUPER
    return (int(round(x)), int(round(y)))


def _nodes(scene: dict[str, Any]) -> dict[str, tuple[float, float]]:
    # A flat diamond keeps the decisive alternate route thin and elongated.
    return {
        "left": (-22.0, 0.0),
        "top": (0.0, -7.0),
        "right": (22.0, 0.0),
        "bottom": (0.0, 7.0),
        "outer_left": (-45.0, 0.0),
        "outer_right": (45.0, 0.0),
    }


def _edge_list() -> list[tuple[str, str]]:
    return [
        ("left", "top"),
        ("top", "right"),
        ("right", "bottom"),
        ("bottom", "left"),
    ]


def _draw_edge(
    draw: ImageDraw.ImageDraw,
    scene: dict[str, Any],
    nodes: dict[str, tuple[float, float]],
    edge: tuple[str, str],
    fill: str,
) -> None:
    width = max(1, int(round(float(scene["stroke_width"]) * SUPER)))
    draw.line(
        [_point(scene, nodes[edge[0]]), _point(scene, nodes[edge[1]])],
        fill=fill,
        width=width,
    )


def _draw_decoration(draw: ImageDraw.ImageDraw, scene: dict[str, Any]) -> None:
    """Add remote gray marks that carry no label or topology information."""

    rng = random.Random(int(scene["decor_seed"]))
    for base_x, base_y in ((48, 48), (464, 52), (52, 464), (462, 460)):
        x = base_x + rng.randint(-10, 10)
        y = base_y + rng.randint(-10, 10)
        radius = 2 * SUPER
        draw.ellipse(
            (x * SUPER - radius, y * SUPER - radius, x * SUPER + radius, y * SUPER + radius),
            fill=DECOR_INK,
        )
    for start, end in (((24, 152), (72, 152)), ((440, 336), (488, 336))):
        draw.line(
            [(start[0] * SUPER, start[1] * SUPER), (end[0] * SUPER, end[1] * SUPER)],
            fill=DECOR_INK,
            width=SUPER,
        )


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the graph, with the selected cycle edge replacing its dark stroke."""

    width = int(scene["width"])
    height = int(scene["height"])
    image = Image.new("RGB", (width * SUPER, height * SUPER), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_decoration(draw, scene)
    nodes = _nodes(scene)
    cycle_edges = _edge_list()
    chosen_cycle_edge = int(scene["cycle_edge"]) % len(cycle_edges)
    cycle_color = str(scene["cycle_color"])
    tail_left, tail_right = [str(value) for value in scene["tail_color_order"]]

    for index, edge in enumerate(cycle_edges):
        if index != chosen_cycle_edge:
            _draw_edge(draw, scene, nodes, edge, NETWORK_INK)
    _draw_edge(draw, scene, nodes, ("outer_left", "left"), COLORS[tail_left])
    _draw_edge(draw, scene, nodes, ("right", "outer_right"), COLORS[tail_right])
    _draw_edge(draw, scene, nodes, cycle_edges[chosen_cycle_edge], COLORS[cycle_color])

    radius = float(scene["node_radius"]) * SUPER
    for position in nodes.values():
        x, y = _point(scene, position)
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=NETWORK_INK)

    return image.resize((width, height), Image.Resampling.LANCZOS).convert("RGB")


def analytic_gold(scene: dict[str, Any]) -> str:
    """The unique cycle-bearing link is the explicit label input."""

    return str(scene["cycle_color"])


def margin(scene: dict[str, Any]) -> float:
    """Report the raster clearance supporting the topological decision."""

    return round(float(scene["scale"]) * 20.0 - 15.0, 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Exclude the smallest loop clearances from headline balance claims."""

    return float(scene["scale"]) < QUARANTINE_SCALE


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No intentionally pixel-identical latent aliases are declared."""

    del scene
    return []
