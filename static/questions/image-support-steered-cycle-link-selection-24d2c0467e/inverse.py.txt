"""Independent pixel-only oracle for the compact cycle-link glyph."""

from __future__ import annotations

import numpy as np
from PIL import Image


PALETTE = {
    "red": np.asarray([217, 75, 72], dtype=np.float32),
    "green": np.asarray([43, 157, 97], dtype=np.float32),
    "blue": np.asarray([61, 115, 199], dtype=np.float32),
}


def _components(mask: np.ndarray, minimum: int = 1) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    found: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y = int(y)
        x = int(x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    ny, nx = cy + dy, cx + dx
                    if (
                        0 <= ny < height
                        and 0 <= nx < width
                        and mask[ny, nx]
                        and not seen[ny, nx]
                    ):
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        if len(points) >= minimum:
            found.append(np.asarray(points, dtype=np.int32))
    return found


def _dilate(mask: np.ndarray, radius: int) -> np.ndarray:
    if radius <= 0:
        return mask.copy()
    height, width = mask.shape
    padded = np.pad(mask, radius, mode="constant")
    output = np.zeros_like(mask)
    for dy in range(2 * radius + 1):
        for dx in range(2 * radius + 1):
            output |= padded[dy : dy + height, dx : dx + width]
    return output


def _color_mask(array: np.ndarray) -> np.ndarray:
    values = array.astype(np.int16)
    high = values.max(axis=2)
    low = values.min(axis=2)
    return (high - low >= 48) & (high >= 105) & (low <= 242)


def _label_for_component(array: np.ndarray, component: np.ndarray) -> str:
    pixels = array[component[:, 0], component[:, 1]].astype(np.float32)
    observed = pixels.mean(axis=0)
    return min(PALETTE, key=lambda name: float(np.linalg.norm(observed - PALETTE[name])))


def _endpoints(component: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    points = component.astype(np.float32)
    centered = points - points.mean(axis=0, keepdims=True)
    covariance = centered.T @ centered
    values, vectors = np.linalg.eigh(covariance)
    axis = vectors[:, int(np.argmax(values))]
    projection = centered @ axis
    return points[int(np.argmin(projection))], points[int(np.argmax(projection))]


def _component_ids(mask: np.ndarray) -> np.ndarray:
    labels = np.full(mask.shape, -1, dtype=np.int32)
    for index, component in enumerate(_components(mask, minimum=1)):
        labels[component[:, 0], component[:, 1]] = int(index)
    return labels


def _nearby_labels(labels: np.ndarray, point: np.ndarray, radius: int = 9) -> set[int]:
    y, x = float(point[0]), float(point[1])
    height, width = labels.shape
    y0 = max(0, int(np.floor(y - radius)))
    y1 = min(height, int(np.ceil(y + radius + 1)))
    x0 = max(0, int(np.floor(x - radius)))
    x1 = min(width, int(np.ceil(x + radius + 1)))
    yy, xx = np.ogrid[y0:y1, x0:x1]
    disk = (yy - y) ** 2 + (xx - x) ** 2 <= float(radius * radius)
    values = labels[y0:y1, x0:x1][disk]
    return {int(value) for value in values if int(value) >= 0}


def _is_cycle_edge(array: np.ndarray, component: np.ndarray, all_colors: np.ndarray) -> bool:
    """Test endpoint connectivity after removing this colored stroke."""

    foreground = np.linalg.norm(array.astype(np.int16) - 255, axis=2) >= 22
    # The dark node disk is intentionally retained around each endpoint; a
    # larger dilation would erase the junction and make real alternate routes
    # look disconnected in the raster.
    removed = _dilate(all_colors, radius=3)
    labels = _component_ids(foreground & ~removed)
    first, second = _endpoints(component)
    left_labels = _nearby_labels(labels, first)
    right_labels = _nearby_labels(labels, second)
    return bool(left_labels & right_labels)


def decision_from_image(image: Image.Image) -> str:
    """Return the color of the sole highlighted link that belongs to a loop."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    color_mask = _color_mask(array)
    components = _components(color_mask, minimum=8)
    if len(components) != 3:
        raise ValueError(f"expected three colored links, found {len(components)}")
    labels: dict[str, np.ndarray] = {}
    for component in components:
        name = _label_for_component(array, component)
        if name in labels:
            raise ValueError("colored link labels are not unique")
        labels[name] = component

    cycle_names = []
    for name, component in labels.items():
        if _is_cycle_edge(array, component, color_mask & _component_mask(color_mask, component)):
            cycle_names.append(name)
    if len(cycle_names) != 1:
        raise ValueError(f"expected one cycle link, found {cycle_names}")
    return cycle_names[0]


def _component_mask(mask: np.ndarray, component: np.ndarray) -> np.ndarray:
    selected = np.zeros(mask.shape, dtype=bool)
    selected[component[:, 0], component[:, 1]] = True
    return selected
