"""Deterministic renderer and analytic arm for Pocket Elevator Replay."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (250, 249, 245)
VIOLET = (126, 78, 188)
ORANGE = (238, 126, 42)
CYAN = (24, 171, 196)
PALE = (229, 226, 218)
INK = (80, 78, 76)

LEVEL_NAMES = ("top", "middle", "bottom")
COMMANDS = ("up", "down")


def _step(level: int, command: str) -> int:
    if command == "up":
        return max(0, level - 1)
    if command == "down":
        return min(2, level + 1)
    raise ValueError(f"invalid command: {command}")


def _replay(start_level: int, commands: list[str]) -> int:
    level = start_level
    for command in commands:
        level = _step(level, command)
    return level


_BY_FINAL: dict[int, tuple[tuple[int, tuple[str, str]], ...]] = {
    0: (
        (0, ("up", "up")),
        (0, ("down", "up")),
        (1, ("up", "up")),
        (2, ("up", "up")),
    ),
    1: (
        (0, ("up", "down")),
        (1, ("up", "down")),
        (1, ("down", "up")),
        (2, ("down", "up")),
    ),
    2: (
        (0, ("down", "down")),
        (1, ("down", "down")),
        (2, ("up", "down")),
        (2, ("down", "down")),
    ),
}


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one balanced, unambiguous visible scene from an integer seed."""
    rng = random.Random(int(seed))
    target = int(seed) % 3
    start, commands = rng.choice(_BY_FINAL[target])
    return {
        "start_level": start,
        "commands": list(commands),
        "anchor_x": rng.choice((160, 224, 288, 352)),
        "anchor_y": rng.choice((96, 160, 224, 288, 352, 416)),
        "decoration_phase": rng.randrange(7),
    }


def _draw_arrow(draw: ImageDraw.ImageDraw, cx: int, cy: int, command: str) -> None:
    points = [
        (cx, cy - 12),
        (cx - 10, cy - 2),
        (cx - 4, cy - 2),
        (cx - 4, cy + 10),
        (cx + 4, cy + 10),
        (cx + 4, cy - 2),
        (cx + 10, cy - 2),
    ]
    if command == "down":
        points = [(x, 2 * cy - y) for x, y in points]
    draw.polygon(points, fill=CYAN)


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the complete scene. The analytic answer is never painted."""
    start = int(scene["start_level"])
    commands = list(scene["commands"])
    anchor_x = int(scene["anchor_x"])
    anchor_y = int(scene["anchor_y"])
    phase = int(scene["decoration_phase"])

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    # Remote, low-contrast texture is intentionally nondiagnostic.
    for index in range(11):
        x = 28 + ((index * 89 + phase * 31) % 456)
        y = 30 + ((index * 53 + phase * 47) % 452)
        if abs(x - anchor_x) < 115 and abs(y - anchor_y) < 75:
            continue
        draw.ellipse((x - 3, y - 3, x + 3, y + 3), outline=PALE, width=2)

    # Left disconnected component: three visible levels plus the starting bead.
    level_x = anchor_x - 64
    level_ys = (anchor_y - 24, anchor_y, anchor_y + 24)
    for y in level_ys:
        draw.ellipse((level_x - 8, y - 8, level_x + 8, y + 8), outline=VIOLET, width=4)
    bead_y = level_ys[start]
    draw.ellipse((level_x - 4, bead_y - 4, level_x + 4, bead_y + 4), fill=ORANGE)

    # Right disconnected component: two detached commands, visibly top-to-bottom.
    command_x = anchor_x + 64
    command_ys = (anchor_y - 21, anchor_y + 21)
    for y, command in zip(command_ys, commands, strict=True):
        _draw_arrow(draw, command_x, y, command)

    # Neutral brackets clarify grouping without connecting the decisive colors.
    draw.line((level_x - 16, anchor_y - 37, level_x - 16, anchor_y + 37), fill=INK, width=2)
    draw.line((command_x + 16, anchor_y - 37, command_x + 16, anchor_y + 37), fill=INK, width=2)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    start = int(scene["start_level"])
    commands = list(scene["commands"])
    return LEVEL_NAMES[_replay(start, commands)]


def margin(scene: dict[str, Any]) -> float:
    """Conservative discrete runner-up clearance in rendered pixels."""
    if is_quarantined(scene):
        return 0.0
    return 16.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        start = int(scene["start_level"])
        commands = list(scene["commands"])
        anchor_x = int(scene["anchor_x"])
        anchor_y = int(scene["anchor_y"])
        int(scene["decoration_phase"])
    except (KeyError, TypeError, ValueError):
        return True
    return not (
        start in (0, 1, 2)
        and len(commands) == 2
        and all(command in COMMANDS for command in commands)
        and anchor_x in (160, 224, 288, 352)
        and anchor_y in (96, 160, 224, 288, 352, 416)
    )


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
