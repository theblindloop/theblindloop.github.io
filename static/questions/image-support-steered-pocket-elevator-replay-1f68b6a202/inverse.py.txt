"""Independent inverse arm: recover and replay state from RGB pixels only."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            output.append(np.asarray(points, dtype=np.int32))
    return output


def _centroid(component: np.ndarray) -> tuple[float, float]:
    return float(component[:, 0].mean()), float(component[:, 1].mean())


def _arrow_direction(component: np.ndarray) -> str:
    ys = component[:, 0]
    y0, y1 = int(ys.min()), int(ys.max())
    if y1 - y0 < 15:
        raise ValueError("arrow glyph is too short")
    band = max(2, (y1 - y0 + 1) // 4)
    top = int(np.sum(ys < y0 + band))
    bottom = int(np.sum(ys > y1 - band))
    if abs(top - bottom) < 8:
        raise ValueError("arrow direction is ambiguous")
    # An up arrow has a narrow tip at the top and a broad stem band at the bottom.
    return "up" if top < bottom else "down"


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[:, :, 0], array[:, :, 1], array[:, :, 2]

    violet = (red >= 90) & (red <= 165) & (green >= 45) & (green <= 115) & (blue >= 145)
    orange = (red >= 205) & (green >= 85) & (green <= 165) & (blue <= 85)
    cyan = (red <= 65) & (green >= 135) & (blue >= 160)

    level_components = _components(violet, minimum=24)
    bead_components = _components(orange, minimum=20)
    arrow_components = _components(cyan, minimum=75)
    if len(level_components) != 3 or len(bead_components) != 1 or len(arrow_components) != 2:
        raise ValueError("required visible state components are missing or ambiguous")

    levels = sorted((_centroid(component) for component in level_components), key=lambda p: p[0])
    bead_y, bead_x = _centroid(bead_components[0])
    if max(abs(x - bead_x) for _, x in levels) > 5.0:
        raise ValueError("starting bead is not aligned to the level path")
    distances = [abs(y - bead_y) for y, _ in levels]
    ordered = sorted(distances)
    if ordered[0] > 4.5 or ordered[1] - ordered[0] < 12.0:
        raise ValueError("starting level cannot be resolved")
    level = int(np.argmin(distances))

    arrows = sorted(arrow_components, key=lambda component: _centroid(component)[0])
    commands = [_arrow_direction(component) for component in arrows]
    arrow_xs = [_centroid(component)[1] for component in arrows]
    if abs(arrow_xs[0] - arrow_xs[1]) > 4.0:
        raise ValueError("command order is not vertically aligned")
    if _centroid(arrows[1])[0] - _centroid(arrows[0])[0] < 24.0:
        raise ValueError("command order is not visibly separated")

    for command in commands:
        level = max(0, level - 1) if command == "up" else min(2, level + 1)
    return ("top", "middle", "bottom")[level]
