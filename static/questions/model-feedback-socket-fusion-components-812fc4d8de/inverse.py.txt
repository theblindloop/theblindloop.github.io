"""Independent RGB-only inverse arm for matching-socket component fusion."""

from __future__ import annotations

from collections import Counter

import numpy as np
from PIL import Image

WIDTH, HEIGHT = 1100, 760
INK = np.array((38, 44, 52), dtype=np.int16)
START_COLOR = np.array((33, 145, 87), dtype=np.int16)
TERMINAL_COLOR = np.array((224, 151, 32), dtype=np.int16)
COLORS = {
    "red": np.array((211, 62, 68), dtype=np.int16),
    "blue": np.array((48, 101, 204), dtype=np.int16),
    "orange": np.array((232, 122, 36), dtype=np.int16),
    "violet": np.array((137, 77, 185), dtype=np.int16),
    "teal": np.array((30, 154, 157), dtype=np.int16),
    "pink": np.array((207, 72, 145), dtype=np.int16),
}
IDENTITIES = tuple(f"{name}_{shape}" for name in COLORS for shape in ("circle", "square"))
CARD_CENTERS = tuple((115 + 217 * column, 285 + 305 * row) for row in range(2) for column in range(5))
SOCKET_OFFSETS = ((-52, 18), (52, 18), (0, 88))
MARKERS = {"start": 0, "A": 2, "B": 4, "C": 6, "D": 8}


class OracleAbstention(ValueError):
    pass


def _near(arr: np.ndarray, color: np.ndarray, threshold: float) -> np.ndarray:
    delta = arr.astype(np.int32) - color.astype(np.int32)
    return np.sum(delta * delta, axis=2, dtype=np.int64) < threshold * threshold


def _decode_socket(arr: np.ndarray, x: int, y: int) -> str | None:
    patch = arr[y - 22:y + 23, x - 22:x + 23]
    scores = {name: int(_near(patch, color, 52.0).sum()) for name, color in COLORS.items()}
    name, score = max(scores.items(), key=lambda item: item[1])
    if score < 500:
        return None
    color = COLORS[name]
    diagonal = sum(
        int(np.linalg.norm(arr[y + dy, x + dx].astype(np.int32) - color.astype(np.int32)) < 55.0)
        for dx, dy in ((-14, -14), (14, -14), (-14, 14), (14, 14))
    )
    shape = "square" if diagonal >= 3 else "circle"
    return f"{name}_{shape}"


class _Union:
    def __init__(self, n: int) -> None:
        self.parent = list(range(n))

    def find(self, value: int) -> int:
        while self.parent[value] != value:
            self.parent[value] = self.parent[self.parent[value]]
            value = self.parent[value]
        return value

    def join(self, left: int, right: int) -> None:
        a, b = self.find(left), self.find(right)
        if a != b:
            self.parent[max(a, b)] = min(a, b)


def decision_from_image(image: Image.Image) -> str:
    raster = image.convert("RGB")
    if raster.size != (WIDTH, HEIGHT):
        raise OracleAbstention(f"unexpected canvas {raster.size}")
    arr = np.asarray(raster)
    if int(_near(arr[165:710], INK, 60.0).sum()) < 15000:
        raise OracleAbstention("plate hubs or outlines are missing")
    sx, sy = CARD_CENTERS[MARKERS["start"]]
    if int(_near(arr[sy - 110:sy - 67, sx - 76:sx + 77], START_COLOR, 55.0).sum()) < 2600:
        raise OracleAbstention("START marker is missing")
    for label in ("A", "B", "C", "D"):
        tx, ty = CARD_CENTERS[MARKERS[label]]
        if int(_near(arr[ty - 115:ty - 58, tx - 29:tx + 30], TERMINAL_COLOR, 58.0).sum()) < 250:
            raise OracleAbstention(f"terminal {label} marker is missing")

    plates: list[list[str]] = []
    for cx, cy in CARD_CENTERS:
        decoded: list[str] = []
        for dx, dy in SOCKET_OFFSETS:
            identity = _decode_socket(arr, cx + dx, cy + dy)
            if identity is not None:
                decoded.append(identity)
        if len(decoded) not in (2, 3) or len(set(decoded)) != len(decoded):
            raise OracleAbstention(f"malformed plate sockets {decoded}")
        plates.append(decoded)
    counts = Counter(identity for plate in plates for identity in plate)
    if counts != Counter({identity: 2 for identity in IDENTITIES}):
        raise OracleAbstention(f"socket pair inventory is malformed: {dict(counts)}")
    locations: dict[str, list[int]] = {identity: [] for identity in IDENTITIES}
    for card, plate in enumerate(plates):
        for identity in plate:
            locations[identity].append(card)
    union = _Union(10)
    for identity, pair in locations.items():
        union.join(pair[0], pair[1])
    root = union.find(MARKERS["start"])
    answers = [label for label in ("A", "B", "C", "D") if union.find(MARKERS[label]) == root]
    groups = Counter(union.find(card) for card in range(10))
    if len(answers) != 1 or sorted(groups.values()) != [5, 5]:
        raise OracleAbstention(f"noncanonical connectivity: answers={answers}, groups={groups}")
    return answers[0]
