"""Deterministic matching-socket quotient-connectivity world."""

from __future__ import annotations

import random
from collections import Counter
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT, SCALE = 1100, 760, 2
BACKGROUND = (239, 241, 244)
PAPER = (255, 255, 255)
INK = (38, 44, 52)
MUTED = (176, 183, 191)
START_COLOR = (33, 145, 87)
TERMINAL_COLOR = (224, 151, 32)
BADGE_COLORS = {
    "red": (211, 62, 68),
    "blue": (48, 101, 204),
    "orange": (232, 122, 36),
    "violet": (137, 77, 185),
    "teal": (30, 154, 157),
    "pink": (207, 72, 145),
}
BADGE_ORDER = tuple(f"{name}_{shape}" for name in BADGE_COLORS for shape in ("circle", "square"))
CARD_CENTERS = tuple((115 + 217 * column, 285 + 305 * row) for row in range(2) for column in range(5))
SOCKET_OFFSETS = ((-52, 18), (52, 18), (0, 88))
MARKER_CARDS = {"start": 0, "A": 2, "B": 4, "C": 6, "D": 8}


class _Union:
    def __init__(self, n: int) -> None:
        self.parent = list(range(n))

    def find(self, value: int) -> int:
        while self.parent[value] != value:
            self.parent[value] = self.parent[self.parent[value]]
            value = self.parent[value]
        return value

    def join(self, left: int, right: int) -> None:
        a, b = self.find(left), self.find(right)
        if a != b:
            self.parent[max(a, b)] = min(a, b)


def _six_edges(cards: list[int], rng: random.Random) -> list[tuple[int, int]]:
    order = list(cards)
    rng.shuffle(order)
    return [(order[i], order[(i + 1) % 5]) for i in range(5)] + [(order[0], order[2])]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = ("A", "B", "C", "D")[int(seed) % 4]
    neutrals = [card for card in range(10) if card not in MARKER_CARDS.values()]
    rng.shuffle(neutrals)
    start_group = [MARKER_CARDS["start"], MARKER_CARDS[target], *neutrals[:3]]
    other_group = [card for card in range(10) if card not in start_group]
    edges = _six_edges(start_group, rng) + _six_edges(other_group, rng)
    identities = list(BADGE_ORDER)
    rng.shuffle(identities)
    plates: list[list[str]] = [[] for _ in range(10)]
    for identity, (left, right) in zip(identities, edges, strict=True):
        plates[left].append(identity)
        plates[right].append(identity)
    for plate in plates:
        rng.shuffle(plate)
    return {"plates": plates, "marker_cards": dict(MARKER_CARDS)}


def _decision(scene: dict) -> tuple[str, _Union]:
    plates = scene["plates"]
    markers = scene["marker_cards"]
    locations: dict[str, list[int]] = {}
    for card, identities in enumerate(plates):
        for identity in identities:
            locations.setdefault(str(identity), []).append(card)
    union = _Union(len(plates))
    for identity in BADGE_ORDER:
        pair = locations.get(identity, [])
        if len(pair) != 2:
            raise ValueError(f"socket {identity} occurs {len(pair)} times")
        union.join(pair[0], pair[1])
    root = union.find(int(markers["start"]))
    answers = [label for label in ("A", "B", "C", "D") if union.find(int(markers[label])) == root]
    if len(answers) != 1:
        raise ValueError(f"expected one connected terminal, got {answers}")
    return answers[0], union


def analytic_gold(scene: dict) -> str:
    return _decision(scene)[0]


def margin(scene: dict) -> float:
    _ = analytic_gold(scene)
    return 24.0


def is_quarantined(scene: dict) -> bool:
    try:
        if set(scene) != {"plates", "marker_cards"}:
            return True
        plates = scene["plates"]
        if not isinstance(plates, list) or len(plates) != 10:
            return True
        if any(len(plate) not in (2, 3) for plate in plates):
            return True
        counts = Counter(identity for plate in plates for identity in plate)
        if counts != Counter({identity: 2 for identity in BADGE_ORDER}):
            return True
        if scene["marker_cards"] != MARKER_CARDS:
            return True
        _, union = _decision(scene)
        groups = Counter(union.find(card) for card in range(10))
        return sorted(groups.values()) != [5, 5]
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["plates"] = [list(reversed(plate)) for plate in twin["plates"]]
    return [("reverse_unordered_socket_memberships", twin)]


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype("DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf", size * SCALE)


def _p(point: tuple[int, int]) -> tuple[int, int]:
    return point[0] * SCALE, point[1] * SCALE


def _b(box: tuple[int, int, int, int]) -> tuple[int, int, int, int]:
    return tuple(value * SCALE for value in box)  # type: ignore[return-value]


def _draw_socket(draw: ImageDraw.ImageDraw, center: tuple[int, int], identity: str) -> None:
    color_name, shape = identity.split("_", 1)
    x, y = center
    if shape == "circle":
        draw.ellipse(_b((x - 20, y - 20, x + 20, y + 20)), fill=BADGE_COLORS[color_name],
                     outline=INK, width=4 * SCALE)
    else:
        draw.rounded_rectangle(_b((x - 20, y - 20, x + 20, y + 20)), radius=3 * SCALE,
                               fill=BADGE_COLORS[color_name], outline=INK, width=4 * SCALE)


def render(scene: dict) -> Image.Image:
    if is_quarantined(scene):
        raise ValueError("cannot render quarantined scene")
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle(_b((18, 14, 1082, 132)), radius=16 * SCALE, fill=PAPER,
                           outline=MUTED, width=2 * SCALE)
    draw.text(_p((38, 29)), "MATCH SOCKETS", fill=INK, font=_font(27, True))
    draw.text(_p((38, 76)), "same color + same shape", fill=INK, font=_font(18))
    draw.text(_p((420, 43)), "each plate joins all sockets on it", fill=INK, font=_font(21, True))
    draw.text(_p((722, 82)), "match every pair across plates", fill=INK, font=_font(17))

    markers = scene["marker_cards"]
    reverse_markers = {int(card): label for label, card in markers.items()}
    for card, (cx, cy) in enumerate(CARD_CENTERS):
        draw.rounded_rectangle(_b((cx - 98, cy - 125, cx + 98, cy + 125)), radius=16 * SCALE,
                               fill=PAPER, outline=MUTED, width=3 * SCALE)
        marker = reverse_markers.get(card)
        if marker == "start":
            draw.rounded_rectangle(_b((cx - 73, cy - 106, cx + 73, cy - 70)), radius=14 * SCALE,
                                   fill=START_COLOR, outline=INK, width=2 * SCALE)
            draw.text(_p((cx - 48, cy - 102)), "START", fill=PAPER, font=_font(18, True))
        elif marker is not None:
            draw.ellipse(_b((cx - 24, cy - 111, cx + 24, cy - 63)), fill=PAPER,
                         outline=TERMINAL_COLOR, width=6 * SCALE)
            draw.text(_p((cx - 10, cy - 107)), marker, fill=INK, font=_font(25, True))
        else:
            draw.text(_p((cx - 27, cy - 102)), "PLATE", fill=MUTED, font=_font(13, True))

        identities = sorted(scene["plates"][card], key=BADGE_ORDER.index)
        positions = [(cx + dx, cy + dy) for dx, dy in SOCKET_OFFSETS[:len(identities)]]
        hub = (cx, cy + 46)
        for position in positions:
            draw.line([_p(hub), _p(position)], fill=INK, width=9 * SCALE)
        draw.ellipse(_b((hub[0] - 11, hub[1] - 11, hub[0] + 11, hub[1] + 11)), fill=INK)
        for position, identity in zip(positions, identities, strict=True):
            _draw_socket(draw, position, identity)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
