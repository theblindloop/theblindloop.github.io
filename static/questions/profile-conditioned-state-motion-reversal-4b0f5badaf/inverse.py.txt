"""Independent pixel-only inverse arm for the motion-reversal counter.

Receives only the final PNG. From the raster alone it:
  1. segments the four red marker disks by colour;
  2. locates each marker's centroid and groups them by horizontal panel;
  3. maps each marker to its site index on the track using the fixed world
     geometry (four panels of equal width, six evenly spaced sites);
  4. computes the three inter-panel moves and counts the direction reversals.

No latent scene, renderer, prompt, or gold code is accessed. The geometry
constants below restate the fixed visual layout of the rendered world only.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

MARKER_COLOR = (220, 40, 40)
MARGIN_LEFT = 60
PANEL_W = 220
SITES = 6
SITE_PAD = 16
SPACING = (PANEL_W - 2 * SITE_PAD) / (SITES - 1)


class UnreadableImageError(ValueError):
    """The raster does not expose the expected four-panel marker structure."""


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    visited = np.zeros_like(mask, dtype=bool)
    ys, xs = np.nonzero(mask)
    comps = []
    for y, x in zip(ys.tolist(), xs.tolist()):
        if visited[y, x]:
            continue
        points = []
        queue = deque([(y, x)])
        visited[y, x] = True
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dy == 0 and dx == 0:
                        continue
                    ny, nx = cy + dy, cx + dx
                    if (
                        0 <= ny < mask.shape[0]
                        and 0 <= nx < mask.shape[1]
                        and mask[ny, nx]
                        and not visited[ny, nx]
                    ):
                        visited[ny, nx] = True
                        queue.append((ny, nx))
        comps.append(points)
    return comps


def _centroid(points) -> tuple[float, float]:
    ys = [p[0] for p in points]
    xs = [p[1] for p in points]
    return float(np.mean(xs)), float(np.mean(ys))


def _mask_equal(arr: np.ndarray, color: tuple[int, int, int]) -> np.ndarray:
    r, g, b = color
    return (arr[:, :, 0] == r) & (arr[:, :, 1] == g) & (arr[:, :, 2] == b)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))

    comps = [comp for comp in _components(_mask_equal(arr, MARKER_COLOR)) if len(comp) >= 10]
    if len(comps) != 4:
        raise UnreadableImageError(f"expected 4 marker disks, found {len(comps)}")

    markers = sorted(_centroid(comp) for comp in comps)
    positions = []
    for cx, cy in markers:
        k = int((cx - MARGIN_LEFT) // PANEL_W)
        if not (0 <= k <= 3):
            raise UnreadableImageError(f"marker outside expected panels: x={cx:.1f}")
        left = MARGIN_LEFT + k * PANEL_W
        site = int(round((cx - (left + SITE_PAD)) / SPACING))
        if not (0 <= site < SITES):
            raise UnreadableImageError(f"marker site out of range: {site}")
        positions.append(site)

    moves = [positions[j + 1] - positions[j] for j in range(3)]
    if any(m == 0 for m in moves):
        raise UnreadableImageError("a stationary move makes direction ambiguous")
    count = sum(1 for j in range(2) if (moves[j] > 0) != (moves[j + 1] > 0))
    return str(count)
