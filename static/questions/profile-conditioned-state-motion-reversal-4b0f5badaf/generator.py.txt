"""State-motion reversal counter (state tracking).

The image shows four ordered panels P1, P2, P3, P4 arranged left to right.
Every panel shows the SAME horizontal track divided into six evenly spaced
sites (thin tick marks). A single red marker (a solid disk) sits on exactly one
site in each panel; the marker keeps its identity (the same token) as it moves
between panels. It is the ONLY red object in the image.

The marker moves to a (possibly different) site between each pair of consecutive
panels. A "reversal" occurs where the marker's direction of horizontal travel
changes sign: it moves right between P1->P2 then left between P2->P3 (or left
then right). With three moves (P1->P2, P2->P3, P3->P4) there are at most two
reversal opportunities.

The task is to count how many times the marker reverses its direction of travel
across the whole ordered sequence. The gold is a function of the whole ordered
position sequence, never of any single frame, and requires tracking the single
token by its position along the shared track across all four panels.

The latent scene stores ``positions`` (a length-4 list of site indices, one per
panel, in P1..P4 order) and ``sites`` (the number of sites on the track). The
answer is the number of sign changes among the three inter-panel moves.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 980
HEIGHT = 300

MARGIN_LEFT = 60
PANEL_W = 220
TRACK_Y = 190
SITES = 6
SITE_PAD = 16
SPACING = (PANEL_W - 2 * SITE_PAD) / (SITES - 1)

MARKER_R = 15

BG = (255, 255, 255)
TRACK_COLOR = (150, 150, 150)
LABEL_COLOR = (60, 60, 60)
MARKER_COLOR = (220, 40, 40)

PANEL_NAMES = ["P1", "P2", "P3", "P4"]


def _site_x(panel_left: int, site: int) -> int:
    return int(round(panel_left + SITE_PAD + site * SPACING))


def sample_scene(seed: int) -> dict:
    """Deterministic latent scene: 4 ordered panel positions on a 6-site track.

    Every consecutive move is nonzero (magnitude 1 or 2 sites), so the direction
    sign of each move is unambiguous. Positions are kept inside the track.
    """
    rng = random.Random(seed)
    for _ in range(20_000):
        positions = [rng.randrange(SITES)]
        valid = True
        for _ in range(3):
            step = rng.choice([-2, -1, 1, 2])
            nxt = positions[-1] + step
            if not (0 <= nxt < SITES):
                if 0 <= positions[-1] - step < SITES:
                    nxt = positions[-1] - step
                else:
                    valid = False
                    break
            positions.append(nxt)
        if valid:
            return {"positions": positions, "sites": SITES}
    raise RuntimeError("could not sample a valid position sequence")


def _moves(scene: dict) -> list[int]:
    positions = scene["positions"]
    return [positions[k + 1] - positions[k] for k in range(3)]


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)
    positions = scene["positions"]

    for k, name in enumerate(PANEL_NAMES):
        left = MARGIN_LEFT + k * PANEL_W
        first_x = _site_x(left, 0)
        last_x = _site_x(left, SITES - 1)
        draw.line([first_x, TRACK_Y, last_x, TRACK_Y], fill=TRACK_COLOR, width=3)
        for s in range(SITES):
            sx = _site_x(left, s)
            draw.line([sx, TRACK_Y - 9, sx, TRACK_Y + 9], fill=TRACK_COLOR, width=3)
        draw.text((left + PANEL_W // 2, 40), name, fill=LABEL_COLOR, anchor="mm")

        site = positions[k]
        cx = _site_x(left, site)
        draw.ellipse([cx - MARKER_R, TRACK_Y - MARKER_R, cx + MARKER_R, TRACK_Y + MARKER_R],
                     fill=MARKER_COLOR)
    return img


def analytic_gold(scene: dict) -> str:
    moves = _moves(scene)
    count = sum(1 for j in range(2) if (moves[j] > 0) != (moves[j + 1] > 0))
    return str(count)


def margin(scene: dict) -> float:
    """Verifiability margin: the smallest inter-panel move in pixels.

    A reversal decision is only reliable when every move is a clearly nonzero
    horizontal displacement; this margin is how far the least decisive move is
    from a stationary (direction-ambiguous) step. Moves are always >= 1 site.
    """
    moves = _moves(scene)
    return float(min(abs(m) for m in moves) * SPACING)


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    """No nontrivial render-invariant latent symmetry.

    ``positions`` directly determine the marker's drawn site in each of the
    four panels, and ``sites`` fixes the shared track geometry. Any
    reparametrization that moved the marker to a different site, changed the
    order of the panels, or changed the site count would change the raster, so
    the parametrization is minimal and there is no render-invariant alias.
    """
    return []
