"""Deterministic renderer for a two-route checkpoint-transition comparison."""

from __future__ import annotations

import itertools
import math
import random

from PIL import Image, ImageDraw

WIDTH = 1000
HEIGHT = 480
BG = (249, 248, 243)
PANEL = (239, 238, 231)
FRAME = (211, 210, 202)
DECOR = (222, 220, 211)

PALETTES = (
    {"route": (28, 45, 65), "start": (38, 112, 204), "finish": (229, 111, 43), "red": (207, 57, 64), "green": (39, 153, 91)},
    {"route": (36, 51, 72), "start": (44, 104, 194), "finish": (220, 101, 39), "red": (194, 51, 61), "green": (47, 141, 88)},
    {"route": (24, 40, 59), "start": (31, 122, 211), "finish": (236, 121, 49), "red": (216, 67, 69), "green": (32, 162, 96)},
)

# A roomy open rectangular spiral.  Its well-separated parallel runs make
# raster geodesics unambiguous while requiring actual tracing to order marks.
BASE_POINTS = (
    (30, 35), (285, 35), (285, 285), (70, 285), (70, 100),
    (230, 100), (230, 230), (130, 230), (130, 155), (185, 155),
)
BASE_FRACTIONS = (0.090, 0.266, 0.430, 0.570, 0.793, 0.935)


def _changes(word: tuple[str, ...] | list[str]) -> int:
    return sum(left != right for left, right in zip(word, word[1:]))


def _word_groups() -> dict[int, tuple[tuple[str, ...], ...]]:
    groups: dict[int, list[tuple[str, ...]]] = {value: [] for value in range(1, 6)}
    for red_positions in itertools.combinations(range(6), 3):
        red = set(red_positions)
        word = tuple("red" if index in red else "green" for index in range(6))
        groups[_changes(word)].append(word)
    return {key: tuple(values) for key, values in groups.items()}


WORDS_BY_CHANGES = _word_groups()


def _transform_route(mode: int, reverse: bool, origin_x: int, jitter_y: int) -> tuple[list[list[int]], list[float]]:
    points = list(BASE_POINTS)
    fractions = list(BASE_FRACTIONS)
    if reverse:
        points.reverse()
        fractions = [round(1.0 - value, 3) for value in reversed(fractions)]
    transformed: list[list[int]] = []
    rotation = mode % 4
    mirror = mode >= 4
    for raw_x, raw_y in points:
        x, y = raw_x, raw_y
        if mirror:
            x = 320 - x
        for _ in range(rotation):
            x, y = 320 - y, x
        transformed.append([origin_x + x, 80 + jitter_y + y])
    return transformed, fractions


def sample_scene(seed: int) -> dict:
    seed = int(seed)
    rng = random.Random(seed)

    # Every word contains exactly three red and three green marks.  Only their
    # traced adjacency structure varies.  Consecutive seeds balance answer side.
    lower_count = 1 + rng.randrange(4)
    higher_count = rng.randrange(lower_count + 1, 6)
    lower_word = rng.choice(WORDS_BY_CHANGES[lower_count])
    higher_word = rng.choice(WORDS_BY_CHANGES[higher_count])
    words = [higher_word, lower_word] if seed % 2 == 0 else [lower_word, higher_word]

    route_points: list[list[list[int]]] = []
    stations: list[list[float]] = []
    modes = [rng.randrange(8), rng.randrange(8)]
    reversals = [bool(rng.randrange(2)), bool(rng.randrange(2))]
    for index, origin_x in enumerate((20, 680)):
        points, fractions = _transform_route(
            modes[index], reversals[index], origin_x, rng.randrange(-5, 6)
        )
        route_points.append(points)
        stations.append(fractions)

    decorations: list[list[int]] = []
    while len(decorations) < 18:
        x = rng.randrange(15, WIDTH - 14)
        y = rng.randrange(18, HEIGHT - 17)
        if all(math.dist((x, y), point) > 27 for route in route_points for point in route):
            decorations.append([x, y])

    return {
        "canvas_size": [WIDTH, HEIGHT],
        "route_points": route_points,
        "checkpoint_stations": stations,
        "checkpoint_colors": [list(word) for word in words],
        "palette_variant": rng.randrange(len(PALETTES)),
        "decorations": decorations,
    }


def _polyline(points: list[list[int]]) -> tuple[list[tuple[float, float]], list[float], float]:
    output = [(float(x), float(y)) for x, y in points]
    lengths = [math.dist(left, right) for left, right in zip(output, output[1:])]
    return output, lengths, sum(lengths)


def _station(points: list[tuple[float, float]], lengths: list[float], total: float, fraction: float) -> tuple[float, float, float, float]:
    target = float(fraction) * total
    consumed = 0.0
    for (x0, y0), (x1, y1), length in zip(points, points[1:], lengths):
        if target <= consumed + length:
            local = (target - consumed) / length
            return (
                x0 + local * (x1 - x0),
                y0 + local * (y1 - y0),
                (x1 - x0) / length,
                (y1 - y0) / length,
            )
        consumed += length
    x0, y0 = points[-2]
    x1, y1 = points[-1]
    length = lengths[-1]
    return x1, y1, (x1 - x0) / length, (y1 - y0) / length


def _ring(draw: ImageDraw.ImageDraw, center: tuple[int, int], radius: int, color: tuple[int, int, int], width: int) -> None:
    x, y = center
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), outline=color, width=width)


def render(scene: dict) -> Image.Image:
    width, height = map(int, scene["canvas_size"])
    image = Image.new("RGB", (width, height), BG)
    draw = ImageDraw.Draw(image)
    palette = PALETTES[int(scene["palette_variant"])]

    draw.rounded_rectangle((10, 24, 380, 456), radius=20, fill=PANEL, outline=FRAME, width=2)
    draw.rounded_rectangle((620, 24, 990, 456), radius=20, fill=PANEL, outline=FRAME, width=2)
    for x, y in scene["decorations"]:
        draw.ellipse((x - 2, y - 2, x + 2, y + 2), fill=DECOR)

    for route_index, raw_points in enumerate(scene["route_points"]):
        points = [tuple(map(int, point)) for point in raw_points]
        draw.line(points, fill=palette["route"], width=9, joint="curve")
        metric, lengths, total = _polyline(raw_points)
        marker_records = []
        for index, (fraction, color_name) in enumerate(
            zip(scene["checkpoint_stations"][route_index], scene["checkpoint_colors"][route_index])
        ):
            x, y, tx, ty = _station(metric, lengths, total, float(fraction))
            side = 1.0 if index % 2 == 0 else -1.0
            cx = int(round(x + side * (-ty) * 15.0))
            cy = int(round(y + side * tx * 15.0))
            draw.line((int(round(x)), int(round(y)), cx, cy), fill=palette["route"], width=3)
            marker_records.append(((cx, cy), color_name))

        _ring(draw, points[0], 12, palette["start"], 4)
        ex, ey = points[-1]
        draw.ellipse((ex - 8, ey - 8, ex + 8, ey + 8), fill=palette["finish"])
        for (cx, cy), color_name in marker_records:
            draw.ellipse((cx - 7, cy - 7, cx + 7, cy + 7), fill=palette[color_name])
    return image


def analytic_gold(scene: dict) -> str:
    counts = [_changes(word) for word in scene["checkpoint_colors"]]
    if counts[0] == counts[1]:
        return "tie"
    return "left_more" if counts[0] > counts[1] else "right_more"


def margin(scene: dict) -> float:
    counts = [_changes(word) for word in scene["checkpoint_colors"]]
    return float(abs(counts[0] - counts[1]))


def is_quarantined(scene: dict) -> bool:
    try:
        words = scene["checkpoint_colors"]
        stations = scene["checkpoint_stations"]
        valid = (
            len(words) == 2
            and len(stations) == 2
            and all(len(word) == 6 and word.count("red") == 3 and word.count("green") == 3 for word in words)
            and all(len(values) == 6 and all(b - a > 0.08 for a, b in zip(values, values[1:])) for values in stations)
        )
        return not valid or margin(scene) <= 0.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
