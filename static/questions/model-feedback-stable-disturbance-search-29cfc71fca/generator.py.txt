"""Deterministic renderer and analytic model for stable disturbance search."""

from __future__ import annotations

import random
from pathlib import Path
from PIL import Image, ImageDraw, ImageFont

SIZE = (960, 720)
BACKGROUND = (247, 245, 238)
CARD = (255, 255, 252)
INK = (29, 38, 50)
BORDER = (104, 111, 120)
ONE = (30, 55, 78)
ZERO = (229, 232, 226)
XOR = (126, 73, 163)
AND = (28, 126, 125)
TARGET = (213, 126, 37)
INTERVENTION = (78, 104, 174)

INPUT_X = (10, 185, 360, 535)
GATE_X = (10, 245, 480)
GATE_Y = (160, 310)
OUTPUT_X = (10, 185, 360, 535)
OUTPUT_Y = 500
TARGET_X = 850
TARGET_Y = (108, 174, 240, 306)
ACTION_X = 742
ACTION_Y = (388, 462, 536, 610)

DECISIONS = ("card-A", "card-B", "card-C", "card-D")
GATE_PARENTS = (("I1", "I2"), ("I2", "I3"), ("I3", "I4"), ("G1", "G2"), ("G2", "G3"), ("G1", "G3"))
OUTPUT_PARENTS = (("G4", "G5"), ("G4", "G6"), ("G5", "G6"), ("G4", "G3"))

def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    for path in (Path("/usr/share/fonts/truetype/dejavu") / name, Path("/usr/share/fonts/dejavu") / name):
        if path.is_file():
            return ImageFont.truetype(str(path), size)
    return ImageFont.load_default()

def _apply(op: str, a: int, b: int) -> int:
    return int(bool(a) != bool(b)) if op == "XOR" else int(bool(a) and bool(b))

def evaluate(inputs: list[int], operations: list[str], disturbances: list[int], intervention: tuple[int, int] | None = None) -> tuple[list[int], list[int]]:
    current = list(inputs)
    if intervention is not None:
        current[int(intervention[0])] = int(intervention[1])
    values = {f"I{i+1}": value for i, value in enumerate(current)}
    gates: list[int] = []
    for index, (left, right) in enumerate(GATE_PARENTS):
        value = _apply(operations[index], values[left], values[right]) ^ int(disturbances[index])
        gates.append(value); values[f"G{index+1}"] = value
    outputs = [
        _apply(operations[6+i], values[left], values[right]) ^ int(disturbances[6+i])
        for i, (left, right) in enumerate(OUTPUT_PARENTS)
    ]
    return gates, outputs

def infer_disturbances(inputs: list[int], operations: list[str], gates: list[int], outputs: list[int]) -> list[int]:
    values = {f"I{i+1}": value for i, value in enumerate(inputs)}
    inferred: list[int] = []
    for index, (left, right) in enumerate(GATE_PARENTS):
        inferred.append(int(gates[index]) ^ _apply(operations[index], values[left], values[right]))
        values[f"G{index+1}"] = int(gates[index])
    for index, (left, right) in enumerate(OUTPUT_PARENTS):
        inferred.append(int(outputs[index]) ^ _apply(operations[6+index], values[left], values[right]))
    return inferred

def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x5A17D15); wanted = int(seed) % 4
    for _ in range(200000):
        inputs = [rng.randrange(2) for _ in range(4)]
        operations = [rng.choice(("XOR", "AND")) for _ in range(10)]
        disturbances = [rng.randrange(2) for _ in range(10)]
        gates, outputs = evaluate(inputs, operations, disturbances)
        interventions = [[i, 1 - inputs[i]] for i in range(4)]
        futures = [evaluate(inputs, operations, disturbances, tuple(action))[1] for action in interventions]
        target = futures[wanted]
        if sum(out == target for out in futures) != 1: continue
        if sum(a != b for a, b in zip(outputs, target)) < 2: continue
        nominal = [evaluate(inputs, operations, [0] * 10, tuple(action))[1] for action in interventions]
        nominal_hits = [i for i, out in enumerate(nominal) if out == target]
        if len(nominal_hits) != 1 or nominal_hits == [wanted]: continue
        return {
            "inputs": inputs, "operations": operations, "disturbances": disturbances,
            "current_gates": gates, "current_outputs": outputs,
            "interventions": interventions, "target_outputs": target,
        }
    raise RuntimeError("could not sample balanced stable-disturbance scene")

def _validate(scene: dict) -> tuple[list[int], list[str], list[int], list[int], list[int], list[list[int]], list[int]]:
    inputs = [int(v) for v in scene["inputs"]]; operations = [str(v) for v in scene["operations"]]
    disturbances = [int(v) for v in scene["disturbances"]]
    gates = [int(v) for v in scene["current_gates"]]; outputs = [int(v) for v in scene["current_outputs"]]
    interventions = [[int(a), int(b)] for a, b in scene["interventions"]]
    target = [int(v) for v in scene["target_outputs"]]
    if len(inputs) != 4 or len(operations) != 10 or len(disturbances) != 10 or len(gates) != 6 or len(outputs) != 4 or len(target) != 4: raise ValueError("wrong size")
    if any(v not in (0, 1) for v in inputs + disturbances + gates + outputs + target): raise ValueError("nonbinary")
    if any(op not in ("XOR", "AND") for op in operations): raise ValueError("bad operation")
    if interventions != [[i, 1-inputs[i]] for i in range(4)]: raise ValueError("bad actions")
    if evaluate(inputs, operations, disturbances) != (gates, outputs): raise ValueError("inconsistent current")
    return inputs, operations, disturbances, gates, outputs, interventions, target

def analytic_gold(scene: dict) -> str:
    try:
        inputs, operations, disturbances, _gates, _outputs, interventions, target = _validate(scene)
        hits = [i for i, action in enumerate(interventions) if evaluate(inputs, operations, disturbances, tuple(action))[1] == target]
        return DECISIONS[hits[0]] if len(hits) == 1 else "abstain"
    except (KeyError, TypeError, ValueError, IndexError):
        return "abstain"

def margin(scene: dict) -> float: return 1.0 if analytic_gold(scene) in DECISIONS else 0.0
def is_quarantined(scene: dict) -> bool: return analytic_gold(scene) == "abstain"
def latent_symmetries(scene: dict) -> list[tuple[str, dict]]: return []

def _center(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str, font: ImageFont.ImageFont, fill=INK) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0]-(box[2]-box[0])/2, xy[1]-(box[3]-box[1])/2-box[1]), text, font=font, fill=fill)

def _lamp(draw: ImageDraw.ImageDraw, center: tuple[int, int], value: int, radius: int = 20, outline=INK) -> None:
    x, y = center; draw.ellipse((x-radius, y-radius, x+radius, y+radius), fill=ONE if value else ZERO, outline=outline, width=3)
    _center(draw, center, str(value), _font(20, True), (255,255,255) if value else INK)

def _op(draw: ImageDraw.ImageDraw, box: tuple[int,int,int,int], op: str) -> None:
    draw.rounded_rectangle(box, radius=8, fill=XOR if op == "XOR" else AND)
    _center(draw, ((box[0]+box[2])//2, (box[1]+box[3])//2), op, _font(16, True), (255,255,255))

def render(scene: dict) -> Image.Image:
    inputs, operations, _disturbances, gates, outputs, interventions, target = _validate(scene)
    image = Image.new("RGB", SIZE, BACKGROUND); draw = ImageDraw.Draw(image)
    _center(draw, (350, 22), "CURRENT — infer each gate's hidden flip", _font(23, True))
    _center(draw, (350, 48), "shown result = named operation, then XOR its own stable hidden bit", _font(16))
    draw.line((720, 8, 720, 710), fill=BORDER, width=4)
    for i, (x, bit) in enumerate(zip(INPUT_X, inputs)):
        draw.rounded_rectangle((x,68,x+160,138), radius=11, fill=CARD, outline=BORDER, width=3)
        draw.text((x+12,78), f"SOURCE I{i+1}", font=_font(17, True), fill=INK); _lamp(draw, (x+128,103), bit, 20)
    for i, ((left, right), bit) in enumerate(zip(GATE_PARENTS, gates)):
        x, y = GATE_X[i%3], GATE_Y[i//3]
        draw.rounded_rectangle((x,y,x+220,y+122), radius=10, fill=CARD, outline=BORDER, width=3)
        draw.text((x+10,y+9), f"G{i+1}", font=_font(19, True), fill=INK)
        draw.text((x+12,y+58), left, font=_font(17, True), fill=INK); _op(draw,(x+54,y+48,x+119,y+88),operations[i])
        draw.text((x+128,y+58), right, font=_font(17, True), fill=INK); draw.text((x+165,y+59), "→", font=_font(18, True), fill=INK)
        _lamp(draw,(x+198,y+68),bit,18)
        draw.text((x+12,y+96),"CURRENT result",font=_font(13),fill=BORDER)
    for i, ((left, right), bit) in enumerate(zip(OUTPUT_PARENTS, outputs)):
        x, y = OUTPUT_X[i], OUTPUT_Y
        draw.rounded_rectangle((x,y,x+160,y+132),radius=10,fill=CARD,outline=BORDER,width=3)
        draw.text((x+10,y+9),f"OUTPUT {chr(65+i)}",font=_font(17,True),fill=INK)
        draw.text((x+10,y+60),left,font=_font(14,True),fill=INK); _op(draw,(x+42,y+48,x+105,y+87),operations[6+i])
        draw.text((x+111,y+60),right,font=_font(14,True),fill=INK); _lamp(draw,(x+80,y+108),bit,17)
    _center(draw,(840,28),"TEST",_font(25,True),INTERVENTION)
    _center(draw,(840,57),"TARGET lamps",_font(17,True),INK)
    for i, (y, bit) in enumerate(zip(TARGET_Y,target)):
        draw.text((755,y-11),f"{chr(65+i)}",font=_font(20,True),fill=INK); _lamp(draw,(TARGET_X,y),bit,21,outline=TARGET)
    _center(draw,(840,354),"TRY ONE CARD",_font(18,True),INK)
    for i, (y, action) in enumerate(zip(ACTION_Y,interventions)):
        draw.rounded_rectangle((ACTION_X,y,ACTION_X+196,y+61),radius=10,fill=CARD,outline=INTERVENTION,width=4)
        draw.text((754,y+17),f"{chr(65+i)}",font=_font(22,True),fill=INTERVENTION)
        draw.text((792,y+20),f"force I{action[0]+1} =",font=_font(17,True),fill=INK); _lamp(draw,(916,y+30),action[1],17,outline=INTERVENTION)
    return image
