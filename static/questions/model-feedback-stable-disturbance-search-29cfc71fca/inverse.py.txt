"""Independent fixed-layout pixel inverse for stable disturbance search."""

from __future__ import annotations
import numpy as np
from PIL import Image

BACKGROUND=(247,245,238); ONE=(30,55,78); ZERO=(229,232,226); XOR=(126,73,163); AND=(28,126,125)
INPUT_X=(10,185,360,535); GATE_X=(10,245,480); GATE_Y=(160,310); OUTPUT_X=(10,185,360,535); OUTPUT_Y=500
TARGET_X=850; TARGET_Y=(108,174,240,306); ACTION_X=742; ACTION_Y=(388,462,536,610)
DECISIONS=("card-A","card-B","card-C","card-D")
GATE_PARENTS=(("I1","I2"),("I2","I3"),("I3","I4"),("G1","G2"),("G2","G3"),("G1","G3"))
OUTPUT_PARENTS=(("G4","G5"),("G4","G6"),("G5","G6"),("G4","G3"))

def _near(pixel,color,tolerance=4): return bool(np.max(np.abs(pixel.astype(np.int16)-np.array(color,dtype=np.int16)))<=tolerance)
def _bit(arr,x,y):
    # Sample the unobscured right half of a lamp rather than its printed digit.
    pixel=arr[y,x+10]
    if _near(pixel,ONE): return 1
    if _near(pixel,ZERO): return 0
    return None
def _apply(op,a,b): return int(bool(a)!=bool(b)) if op=="XOR" else int(bool(a) and bool(b))

def _infer(inputs,operations,gates,outputs):
    values={f"I{i+1}":v for i,v in enumerate(inputs)}; disturbances=[]
    for i,(left,right) in enumerate(GATE_PARENTS):
        disturbances.append(gates[i]^_apply(operations[i],values[left],values[right])); values[f"G{i+1}"]=gates[i]
    for i,(left,right) in enumerate(OUTPUT_PARENTS): disturbances.append(outputs[i]^_apply(operations[6+i],values[left],values[right]))
    return disturbances

def _evaluate(inputs,operations,disturbances,action):
    src=list(inputs); src[action[0]]=action[1]; values={f"I{i+1}":v for i,v in enumerate(src)}
    for i,(left,right) in enumerate(GATE_PARENTS): values[f"G{i+1}"]=_apply(operations[i],values[left],values[right])^disturbances[i]
    return [_apply(operations[6+i],values[left],values[right])^disturbances[6+i] for i,(left,right) in enumerate(OUTPUT_PARENTS)]

def decision_from_image(image: Image.Image) -> str:
    arr=np.asarray(image.convert("RGB"),dtype=np.uint8)
    if arr.shape!=(720,960,3) or not _near(arr[710,950],BACKGROUND): return "abstain"
    inputs=[_bit(arr,x+128,103) for x in INPUT_X]
    if any(v is None for v in inputs): return "abstain"
    operations=[]; gates=[]
    for i in range(6):
        x,y=GATE_X[i%3],GATE_Y[i//3]; pixel=arr[y+52,x+58]
        if _near(pixel,XOR): operations.append("XOR")
        elif _near(pixel,AND): operations.append("AND")
        else: return "abstain"
        bit=_bit(arr,x+198,y+68)
        if bit is None: return "abstain"
        gates.append(bit)
    outputs=[]
    for x in OUTPUT_X:
        pixel=arr[OUTPUT_Y+52,x+46]
        if _near(pixel,XOR): operations.append("XOR")
        elif _near(pixel,AND): operations.append("AND")
        else: return "abstain"
        bit=_bit(arr,x+80,OUTPUT_Y+108)
        if bit is None: return "abstain"
        outputs.append(bit)
    target=[_bit(arr,TARGET_X,y) for y in TARGET_Y]
    actions=[[i,_bit(arr,916,y+30)] for i,y in enumerate(ACTION_Y)]
    if any(v is None for v in target) or any(v is None for _,v in actions): return "abstain"
    if any(value==inputs[index] for index,value in actions): return "abstain"
    disturbances=_infer(inputs,operations,gates,outputs)
    hits=[i for i,action in enumerate(actions) if _evaluate(inputs,operations,disturbances,action)==target]
    return DECISIONS[hits[0]] if len(hits)==1 else "abstain"
