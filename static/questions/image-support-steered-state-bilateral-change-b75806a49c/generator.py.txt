"""Bilateral state-change tracking world renderer.

A tracked binary signal is sampled twice at two temporally ordered records that
are spatially DISCONNECTED: an INITIAL record in the left-middle region and a
FINAL record in the right-middle region of the canvas. Each record is rendered
as a filled disk whose colour encodes the discrete state held at that time step:
blue = state OFF (0), orange = state ON (1). The two records are separated by a
wide white gap so that the decisive pixel support is two disconnected components
spanning a medium (regional) band of the image, never crossing the full canvas.

The analytic gold is the binary ANSWER to "did the tracked state change between
the INITIAL and FINAL records", i.e. whether the two records' states differ.
Every component (left record AND right record) contributes to the answer: both
must be read; the left state is sampled before the right state, so there is an
unambiguous time ordering defined by left-to-right position.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "state_bilateral_change-0.1.0"

CANVAS = 512
RADIUS = 26

# State colouring: blue = OFF (state 0), orange = ON (state 1).
COLOR_0 = (0, 90, 200)
COLOR_1 = (240, 130, 30)
OUTLINE = (50, 50, 50)

LEFT_CENTER = (150, 256)
RIGHT_CENTER = (362, 256)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    left_state = rng.choice((0, 1))
    right_state = rng.choice((0, 1))
    return {
        "left_state": left_state,
        "right_state": right_state,
    }


def _disk_color(state: int) -> tuple[int, int, int]:
    return COLOR_0 if state == 0 else COLOR_1


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (CANVAS, CANVAS), "white")
    draw = ImageDraw.Draw(img)
    # initial (left) and final (right) state records
    draw.ellipse(
        (
            LEFT_CENTER[0] - RADIUS,
            LEFT_CENTER[1] - RADIUS,
            LEFT_CENTER[0] + RADIUS,
            LEFT_CENTER[1] + RADIUS,
        ),
        fill=_disk_color(scene["left_state"]),
        outline=OUTLINE,
        width=1,
    )
    draw.ellipse(
        (
            RIGHT_CENTER[0] - RADIUS,
            RIGHT_CENTER[1] - RADIUS,
            RIGHT_CENTER[0] + RADIUS,
            RIGHT_CENTER[1] + RADIUS,
        ),
        fill=_disk_color(scene["right_state"]),
        outline=OUTLINE,
        width=1,
    )
    return img


def analytic_gold(scene: dict) -> str:
    change = int(scene["left_state"]) != int(scene["right_state"])
    return "yes" if change else "no"


def margin(scene: dict) -> float:
    """Surface gap between the two disconnected records (pus).

    This is the inter-record clearance that keeps the two decisive components
    cleanly separated and individually legible. It is constant and strictly
    positive across scenes because the records occupy fixed, non-touching
    positions; the discrete state QD per record does not change it.
    """
    dx = RIGHT_CENTER[0] - LEFT_CENTER[0]
    dy = RIGHT_CENTER[1] - LEFT_CENTER[1]
    center_dist = (dx * dx + dy * dy) ** 0.5
    return center_dist - 2 * RADIUS


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict):
    # not_applicable: no pixel-identical latent transform is declared; the two
    # records are distinguished only by their fixed spatial roles, and the time
    # ordering (left INITIAL -> right FINAL) is load-bearing.
    return []
