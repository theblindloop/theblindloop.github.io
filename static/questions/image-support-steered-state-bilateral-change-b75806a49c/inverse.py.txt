"""Independent pixel-only inverse arm for state_bilateral_change.

Receives ONLY a rasterized PNG. It segments the two state-colour masks (orange
= ON, blue = OFF), extracts connected components, and identifies the two record
disks by their horizontal order: the leftmost disk is the INITIAL (time-step 1)
record, the rightmost disk is the FINAL (time-step 2) record. It then binds each
disk's colour to a discrete state and reports whether the two states differ.

If either record is missing, merged, or out of its expected horizontal half, the
oracle refuses to decide (raises) instead of guessing. Destroying either
disconnected record therefore blocks or changes the decision, while destroying
empty canvas area changes nothing — the decisive support is exactly the two
disconnected regional components.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORANGE = (240, 130, 30)
BLUE = (0, 90, 200)
MIN_DISK_PX = 300


def _components(mask: np.ndarray):
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    comps = []
    ys_idx, xs_idx = np.nonzero(mask)
    for y, x in zip(ys_idx.tolist(), xs_idx.tolist()):
        if visited[y, x]:
            continue
        stack = [(y, x)]
        visited[y, x] = True
        ys = []
        xs = []
        while stack:
            cy, cx = stack.pop()
            ys.append(cy)
            xs.append(cx)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    stack.append((ny, nx))
        comps.append((np.asarray(ys, dtype=np.float64), np.asarray(xs, dtype=np.float64)))
    return comps


def _masks(image: Image.Image):
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    r = arr[..., 0]
    g = arr[..., 1]
    b = arr[..., 2]
    orange = (r > 170) & (g > 80) & (g < 210) & (b < 110)
    blue = (r < 80) & (b > 140) & (g < 160)
    return orange, blue


def decision_from_image(image: Image.Image) -> str:
    orange, blue = _masks(image)
    width = image.width
    disks = []  # (state, centroid_x)

    for comp in _components(orange):
        if len(comp[0]) >= MIN_DISK_PX:
            disks.append((1, float(np.mean(comp[1]))))
    for comp in _components(blue):
        if len(comp[0]) >= MIN_DISK_PX:
            disks.append((0, float(np.mean(comp[1]))))

    if len(disks) != 2:
        raise ValueError(
            f"expected exactly two state records, found {len(disks)}"
        )

    disks.sort(key=lambda item: item[1])
    left_state, right_state = disks[0][0], disks[1][0]

    mid = width / 2.0
    if not (disks[0][1] < mid < disks[1][1]):
        raise ValueError("records are not in the expected left/right halves")

    change = left_state != right_state
    return "yes" if change else "no"
