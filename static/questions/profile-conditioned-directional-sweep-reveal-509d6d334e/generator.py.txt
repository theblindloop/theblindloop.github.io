"""Deterministic renderer for the directional sweep reveal world."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image, ImageDraw


CANVAS = (420, 320)
SYMBOLS = ("arch", "diamond", "fork", "ring")
MARKER_PALETTE = (
    (220, 48, 66),
    (30, 142, 226),
    (24, 170, 102),
    (181, 70, 205),
    (224, 143, 32),
)
SWEEP_RADIUS = 12
SWEEP_SAMPLES = 2 * SWEEP_RADIUS + 1
CARRIER_PERIOD = 5


def _label_index(seed: int) -> int:
    """Use a separate deterministic stream so nuisance fields do not key the label."""
    value = (int(seed) + 0x9E3779B97F4A7C15) & ((1 << 64) - 1)
    value ^= value >> 30
    value = (value * 0xBF58476D1CE4E5B9) & ((1 << 64) - 1)
    value ^= value >> 27
    value = (value * 0x94D049BB133111EB) & ((1 << 64) - 1)
    value ^= value >> 31
    return int(value % len(SYMBOLS))


def sample_scene(seed: int) -> dict:
    """Sample one translated, marked raster with independent nuisance fields."""
    rng = np.random.default_rng(int(seed) & ((1 << 63) - 1))
    size = int(rng.choice((168, 184, 200)))
    tile_x = int(rng.integers(52, CANVAS[0] - size - 51))
    tile_y = int(rng.integers(46, CANVAS[1] - size - 45))
    return {
        "symbol": SYMBOLS[_label_index(seed)],
        "tile_x": tile_x,
        "tile_y": tile_y,
        "tile_size": size,
        "marker_color": [int(v) for v in rng.choice(MARKER_PALETTE)],
        "marker_size": int(rng.choice((9, 11, 13))),
        "phase_index": int(rng.integers(0, CARRIER_PERIOD)),
        "carrier_amplitude": int(rng.choice((82, 88, 94))),
        "signal_strength": int(rng.choice((46, 54, 62, 70, 78))),
        "base_level": int(rng.choice((142, 148, 154, 160))),
        "background": int(rng.choice((238, 243, 248))),
    }


def _segment_distance(
    x: np.ndarray,
    y: np.ndarray,
    start: tuple[float, float],
    end: tuple[float, float],
) -> np.ndarray:
    x0, y0 = start
    x1, y1 = end
    vx, vy = x1 - x0, y1 - y0
    denominator = vx * vx + vy * vy
    t = ((x - x0) * vx + (y - y0) * vy) / denominator
    t = np.clip(t, 0.0, 1.0)
    return np.hypot(x - (x0 + t * vx), y - (y0 + t * vy))


def _symbol_mask(symbol: str, size: int) -> np.ndarray:
    center = (size - 1) / 2.0
    yy, xx = np.indices((size, size), dtype=np.float32)
    x = (xx - center) / (size / 2.0)
    y = (yy - center) / (size / 2.0)
    if symbol == "ring":
        mask = (np.hypot(x, y) >= 0.34) & (np.hypot(x, y) < 0.47)
    elif symbol == "diamond":
        radius = np.abs(x) + np.abs(y)
        mask = (radius >= 0.49) & (radius < 0.61)
    elif symbol == "arch":
        uprights = (np.abs(np.abs(x) - 0.34) < 0.085) & (y >= -0.30) & (y <= 0.40)
        crown = (np.abs(y + 0.34) < 0.085) & (np.abs(x) <= 0.42)
        mask = uprights | crown
    elif symbol == "fork":
        left = _segment_distance(x, y, (-0.55, 0.38), (0.0, -0.24)) < 0.082
        right = _segment_distance(x, y, (0.55, 0.38), (0.0, -0.24)) < 0.082
        stem = _segment_distance(x, y, (0.0, -0.24), (0.0, 0.48)) < 0.082
        mask = left | right | stem
    else:
        raise ValueError(symbol)
    return mask & (np.abs(x) < 0.82) & (np.abs(y) < 0.82)


def render(scene: dict) -> Image.Image:
    """Render a carrier-dominated grayscale tile and four outside fiducials."""
    width, height = CANVAS
    image = np.full((height, width, 3), int(scene["background"]), dtype=np.uint8)
    size = int(scene["tile_size"])
    yy, xx = np.indices((size, size), dtype=np.float32)
    center = (size - 1) / 2.0
    normalized_x = (xx - center) / (size / 2.0)
    phase = int(scene["phase_index"]) % CARRIER_PERIOD
    carrier = float(scene["carrier_amplitude"]) * np.sin(
        2.0 * math.pi * (xx + phase) / float(CARRIER_PERIOD)
    )
    carrier *= (np.abs(normalized_x) < 0.84).astype(np.float32)
    mask = _symbol_mask(str(scene["symbol"]), size)
    tile = float(scene["base_level"]) + carrier - float(scene["signal_strength"]) * mask
    tile = np.clip(tile, 0, 255).astype(np.uint8)
    rgb_tile = np.repeat(tile[:, :, None], 3, axis=2)
    tx, ty = int(scene["tile_x"]), int(scene["tile_y"])
    image[ty : ty + size, tx : tx + size] = rgb_tile

    output = Image.fromarray(image, mode="RGB")
    draw = ImageDraw.Draw(output)
    half = int(scene["marker_size"]) // 2
    color = tuple(int(v) for v in scene["marker_color"])
    marker_centers = (
        (tx - 10, ty - 10),
        (tx + size + 9, ty - 10),
        (tx - 10, ty + size + 9),
        (tx + size + 9, ty + size + 9),
    )
    for cx, cy in marker_centers:
        draw.rectangle((cx - half, cy - half, cx + half, cy + half), fill=color)
    return output


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def margin(scene: dict) -> float:
    """One-sided transform-evidence margin in grayscale levels."""
    return float(scene["signal_strength"]) - 48.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["phase_index"] = int(scene["phase_index"]) + CARRIER_PERIOD
    return [("carrier_phase_period", twin)]
