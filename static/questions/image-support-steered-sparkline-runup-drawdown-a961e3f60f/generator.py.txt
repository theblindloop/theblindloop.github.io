"""Deterministic focal sparkline renderer and analytic decision arm."""

from __future__ import annotations

import math
from typing import Any

import numpy as np
from PIL import Image, ImageDraw

CANVAS_SIZE = 512
PALETTE = (
    (18, 126, 143),
    (122, 72, 171),
    (202, 78, 42),
    (24, 132, 82),
)


def _ordered_swings(levels: list[int] | tuple[int, ...]) -> tuple[float, float]:
    """Return (largest rise, largest fall) for chart values equal to -pixel_y."""

    values = [-float(level) for level in levels]
    lowest_so_far = values[0]
    highest_so_far = values[0]
    largest_rise = 0.0
    largest_fall = 0.0
    for value in values[1:]:
        largest_rise = max(largest_rise, value - lowest_so_far)
        largest_fall = max(largest_fall, highest_so_far - value)
        lowest_so_far = min(lowest_so_far, value)
        highest_so_far = max(highest_so_far, value)
    return largest_rise, largest_fall


def _sample_levels(rng: np.random.Generator) -> list[int]:
    """Sample a tied-endpoint path with a stable but nontrivial swing margin."""

    for _ in range(10_000):
        interior = [int(value) for value in rng.integers(-10, 11, size=7)]
        levels = [0, *interior, 0]
        if min(levels) > -8 or max(levels) < 8:
            continue
        if any(a == b for a, b in zip(levels, levels[1:])):
            continue
        rise, fall = _ordered_swings(levels)
        separation = abs(rise - fall)
        if 4.0 <= separation <= 10.0:
            return levels
    raise RuntimeError("could not sample a stable ordered-extrema trace")


def sample_scene(seed: int) -> dict[str, Any]:
    rng = np.random.default_rng(int(seed))
    levels = _sample_levels(rng)
    desired = "RISE" if int(seed) % 2 == 0 else "FALL"
    rise, fall = _ordered_swings(levels)
    observed = "RISE" if rise > fall else "FALL"
    if observed != desired:
        levels = list(reversed(levels))

    # Anchors are randomized among grid-cell interiors.  The whole trace remains
    # in one narrow row and about three adjacent high-resolution ablation cells.
    anchor_x = int(rng.choice((96, 160, 224, 288))) + int(rng.integers(-1, 2))
    anchor_y = int(rng.choice((160, 224, 288, 352))) + int(rng.integers(-1, 2))
    return {
        "canvas_size": CANVAS_SIZE,
        "anchor_x": anchor_x,
        "anchor_y": anchor_y,
        "x_step": int(rng.choice((13, 14))),
        "levels": levels,
        "stroke_width": int(rng.choice((4, 5))),
        "trace_rgb": list(PALETTE[int(rng.integers(0, len(PALETTE)))]),
    }


def render(scene: dict[str, Any]) -> Image.Image:
    canvas_size = int(scene["canvas_size"])
    anchor_x = int(scene["anchor_x"])
    anchor_y = int(scene["anchor_y"])
    x_step = int(scene["x_step"])
    levels = [int(value) for value in scene["levels"]]
    stroke_width = int(scene["stroke_width"])
    trace_rgb = tuple(int(value) for value in scene["trace_rgb"])

    image = Image.new("RGB", (canvas_size, canvas_size), (250, 250, 248))
    draw = ImageDraw.Draw(image)
    x_end = anchor_x + x_step * (len(levels) - 1)
    # A local neutral inset makes the chart-like role visible but is excluded by
    # the saturated-color oracle.  It carries no decision information.
    frame = (anchor_x - 10, anchor_y - 17, x_end + 10, anchor_y + 17)
    draw.rounded_rectangle(frame, radius=7, fill=(255, 255, 255), outline=(222, 225, 224), width=1)
    points = [
        (anchor_x + index * x_step, anchor_y + int(level))
        for index, level in enumerate(levels)
    ]
    draw.line(points, fill=trace_rgb, width=stroke_width, joint="curve")
    radius = stroke_width // 2
    for x, y in points[1:-1]:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=trace_rgb)
    return image


def margin(scene: dict[str, Any]) -> float:
    rise, fall = _ordered_swings([int(value) for value in scene["levels"]])
    return float(rise - fall)


def analytic_gold(scene: dict[str, Any]) -> str:
    rise, fall = _ordered_swings([int(value) for value in scene["levels"]])
    if math.isclose(rise, fall, abs_tol=1e-12):
        return "TIE"
    return "RISE" if rise > fall else "FALL"


def is_quarantined(scene: dict[str, Any]) -> bool:
    return abs(margin(scene)) <= 2.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
