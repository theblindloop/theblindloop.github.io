"""Independent image-only inverse arm for the focal sparkline world."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _largest_component(mask: np.ndarray) -> tuple[np.ndarray, int]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components: list[list[tuple[int, int]]] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue: deque[tuple[int, int]] = deque([(y, x)])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny in range(max(0, cy - 1), min(height, cy + 2)):
                for nx in range(max(0, cx - 1), min(width, cx + 2)):
                    if mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        components.append(points)
    if not components:
        raise ValueError("no colored trace is visible")
    components.sort(key=len, reverse=True)
    output = np.zeros_like(mask, dtype=bool)
    for y, x in components[0]:
        output[y, x] = True
    second_size = len(components[1]) if len(components) > 1 else 0
    return output, second_size


def _ordered_swings(values: np.ndarray) -> tuple[float, float]:
    lowest_so_far = float(values[0])
    highest_so_far = float(values[0])
    largest_rise = 0.0
    largest_fall = 0.0
    for value_raw in values[1:]:
        value = float(value_raw)
        largest_rise = max(largest_rise, value - lowest_so_far)
        largest_fall = max(largest_fall, highest_so_far - value)
        lowest_so_far = min(lowest_so_far, value)
        highest_so_far = max(highest_so_far, value)
    return largest_rise, largest_fall


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    channel_max = array.max(axis=2)
    channel_min = array.min(axis=2)
    # The chart frame and background are neutral.  Chroma segmentation therefore
    # finds the trace without knowing its palette or location.
    mask = (channel_max - channel_min >= 38) & (channel_min <= 205)
    component, second_size = _largest_component(mask)
    area = int(component.sum())
    if area < 250:
        raise ValueError("colored trace is too small or materially erased")
    if second_size >= max(12, area // 20):
        raise ValueError("colored trace is split into multiple components")

    ys, xs = np.nonzero(component)
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    width = x1 - x0 + 1
    height = y1 - y0 + 1
    if not 88 <= width <= 126 or not 12 <= height <= 32:
        raise ValueError("trace extent is outside the declared mini-trace regime")

    center_y = []
    for x in range(x0, x1 + 1):
        column = np.flatnonzero(component[:, x])
        if not len(column):
            raise ValueError("trace traversal has a missing horizontal slice")
        center_y.append(float(np.median(column)))
    # Chart values increase upward, opposite raster y.
    values = -np.asarray(center_y, dtype=np.float64)
    rise, fall = _ordered_swings(values)
    if abs(rise - fall) <= 1.5:
        raise ValueError("ordered swing comparison is raster-ambiguous")
    return "RISE" if rise > fall else "FALL"
