"""Independent image-only inverse for the rank-gain slopegraph.

The inverse discovers guide columns and legend colors from the raster. It never
receives a scene, palette, answer, filename, or renderer coordinate.
"""

from __future__ import annotations

from collections import Counter

from PIL import Image

LABELS = ("A", "B", "C", "D")


def _groups(values: list[int]) -> list[list[int]]:
    if not values:
        return []
    groups = [[values[0]]]
    for value in values[1:]:
        if value <= groups[-1][-1] + 1:
            groups[-1].append(value)
        else:
            groups.append([value])
    return groups


def _guide_columns(image: Image.Image) -> tuple[int, int]:
    pixels = image.load()
    candidates = []
    for x in range(image.width):
        count = sum(
            1
            for y in range(image.height)
            if 165 <= pixels[x, y][0] <= 195
            and abs(pixels[x, y][0] - pixels[x, y][1]) <= 2
            and abs(pixels[x, y][1] - pixels[x, y][2]) <= 2
        )
        if count >= 75:
            candidates.append(x)
    groups = [group for group in _groups(candidates) if len(group) >= 2]
    if len(groups) != 2:
        raise ValueError(f"expected two visible guide columns, found {len(groups)}")
    centers = [round(sum(group) / len(group)) for group in groups]
    if centers[1] - centers[0] < image.width * 0.25:
        raise ValueError("guide columns are not sufficiently separated")
    return centers[0], centers[1]


def _is_saturated(rgb: tuple[int, int, int]) -> bool:
    return max(rgb) - min(rgb) >= 65 and max(rgb) >= 105 and min(rgb) <= 175


def _legend_mapping(image: Image.Image, right_guide: int) -> dict[str, tuple[int, int, int]]:
    pixels = image.load()
    counts: Counter[tuple[int, int, int]] = Counter()
    ys: dict[tuple[int, int, int], list[int]] = {}
    cutoff = right_guide + max(35, image.width // 25)
    for y in range(image.height):
        for x in range(cutoff, image.width):
            rgb = pixels[x, y]
            if _is_saturated(rgb):
                counts[rgb] += 1
                ys.setdefault(rgb, []).append(y)
    colors = [rgb for rgb, count in counts.most_common(4) if count >= 180]
    if len(colors) != 4:
        raise ValueError("could not recover four substantial legend swatches")
    colors.sort(key=lambda rgb: sum(ys[rgb]) / len(ys[rgb]))
    row_centers = [sum(ys[rgb]) / len(ys[rgb]) for rgb in colors]
    if min(row_centers[i + 1] - row_centers[i] for i in range(3)) < 20:
        raise ValueError("legend swatch rows are not distinct")
    return dict(zip(LABELS, colors, strict=True))


def _endpoint_y(image: Image.Image, x_center: int, color: tuple[int, int, int]) -> float:
    pixels = image.load()
    points = []
    for x in range(max(0, x_center - 7), min(image.width, x_center + 8)):
        for y in range(image.height):
            if pixels[x, y] == color:
                points.append(y)
    if len(points) < 45:
        raise ValueError("endpoint marker is missing or materially erased")
    points.sort()
    return float(points[len(points) // 2])


def decision_from_image(image: Image.Image) -> str:
    """Return A-D using only recovered legend binding and endpoint order."""
    image = image.convert("RGB")
    before_x, after_x = _guide_columns(image)
    mapping = _legend_mapping(image, after_x)
    before_y = {label: _endpoint_y(image, before_x, color) for label, color in mapping.items()}
    after_y = {label: _endpoint_y(image, after_x, color) for label, color in mapping.items()}
    if min(abs(a - b) for i, a in enumerate(before_y.values()) for b in list(before_y.values())[i + 1 :]) < 14:
        raise ValueError("BEFORE ordering is raster-ambiguous")
    if min(abs(a - b) for i, a in enumerate(after_y.values()) for b in list(after_y.values())[i + 1 :]) < 14:
        raise ValueError("AFTER ordering is raster-ambiguous")
    before_order = sorted(LABELS, key=lambda label: (before_y[label], label))
    after_order = sorted(LABELS, key=lambda label: (after_y[label], label))
    before_rank = {label: rank for rank, label in enumerate(before_order, start=1)}
    after_rank = {label: rank for rank, label in enumerate(after_order, start=1)}
    gains = {label: before_rank[label] - after_rank[label] for label in LABELS}
    ordered = sorted(LABELS, key=lambda label: (-gains[label], label))
    if gains[ordered[0]] == gains[ordered[1]]:
        raise ValueError("pixel-recovered rank gain is tied")
    return ordered[0]
