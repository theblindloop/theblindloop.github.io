"""Deterministic renderer and analytic arm for a rank-gain slopegraph."""

from __future__ import annotations

import itertools
import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 760
HEIGHT = 500
SERIES_LABELS = ("A", "B", "C", "D")
GUIDE = (180, 180, 180)
INK = (37, 43, 52)
MUTED = (92, 101, 115)
BACKGROUND = (250, 251, 253)

# Each bank stays highly saturated, but none of the hues has a fixed series label.
PALETTE_BANKS = (
    ((205, 54, 67), (39, 112, 191), (32, 150, 103), (226, 126, 34)),
    ((128, 78, 190), (8, 151, 170), (221, 73, 132), (130, 145, 35)),
    ((225, 72, 45), (43, 126, 210), (21, 161, 124), (181, 84, 173)),
)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    # Pillow's bundled default font avoids any external asset or system-font dependency.
    return ImageFont.load_default(size=size)


def _ranks(scene: dict, key: str) -> dict[str, int]:
    ordered = sorted(scene["series"], key=lambda item: (-float(item[key]), item["label"]))
    return {item["label"]: rank for rank, item in enumerate(ordered, start=1)}


def _rank_gains(scene: dict) -> dict[str, int]:
    before = _ranks(scene, "before")
    after = _ranks(scene, "after")
    return {label: before[label] - after[label] for label in SERIES_LABELS}


def analytic_gold(scene: dict) -> str:
    """Series with maximal ordinal gain; only ``series`` can influence gold."""
    gains = _rank_gains(scene)
    ordered = sorted(gains, key=lambda label: (-gains[label], label))
    if gains[ordered[0]] == gains[ordered[1]]:
        raise ValueError("rank-gain tie has no analytic decision")
    return ordered[0]


def margin(scene: dict) -> float:
    """Runner-up decision margin in rank positions."""
    values = sorted(_rank_gains(scene).values(), reverse=True)
    return float(values[0] - values[1])


def _point_y(scene: dict, value: float) -> float:
    top = float(scene["plot_top"])
    bottom = float(scene["plot_bottom"])
    return bottom - float(value) * (bottom - top)


def is_quarantined(scene: dict) -> bool:
    span = float(scene["plot_bottom"] - scene["plot_top"])
    for key in ("before", "after"):
        values = sorted(float(item[key]) for item in scene["series"])
        if min((values[i + 1] - values[i]) * span for i in range(3)) < 18.0:
            return True
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """There are no pixel-identical latent aliases in this explicit scene."""
    return []


def _irregular_levels(rng: random.Random) -> list[float]:
    """Four highness values with large, nonuniform gaps."""
    for _ in range(1000):
        values = sorted((rng.uniform(0.08, 0.92) for _ in range(4)), reverse=True)
        gaps = [values[i] - values[i + 1] for i in range(3)]
        if min(gaps) >= 0.13 and max(gaps) - min(gaps) >= 0.045:
            return values
    raise RuntimeError("could not sample separated irregular levels")


def _winner(gains: dict[int, float]) -> int | None:
    ordered = sorted(gains, key=lambda index: (-gains[index], index))
    return ordered[0] if gains[ordered[0]] > gains[ordered[1]] else None


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene whose ordinal winner is not a raw-rise shortcut."""
    rng = random.Random(int(seed))
    desired = int(seed) % 4
    permutations = list(itertools.permutations(range(1, 5)))
    for _ in range(20000):
        before_ranks = rng.choice(permutations)
        after_ranks = rng.choice(permutations)
        ordinal = {i: before_ranks[i] - after_ranks[i] for i in range(4)}
        if _winner(ordinal) != desired or sorted(ordinal.values(), reverse=True)[0] - sorted(ordinal.values(), reverse=True)[1] != 1:
            continue

        before_levels = _irregular_levels(rng)
        after_levels = _irregular_levels(rng)
        before_values = {i: before_levels[before_ranks[i] - 1] for i in range(4)}
        after_values = {i: after_levels[after_ranks[i] - 1] for i in range(4)}
        raw_rise = {i: after_values[i] - before_values[i] for i in range(4)}
        if _winner(raw_rise) in (None, desired):
            continue
        x_before = rng.randint(150, 178)
        x_after = rng.randint(493, 525)
        plot_top = rng.randint(78, 92)
        plot_bottom = rng.randint(400, 420)
        pixel_length = {
            i: math.hypot(
                x_after - x_before,
                (after_values[i] - before_values[i]) * (plot_bottom - plot_top),
            )
            for i in range(4)
        }
        if _winner(pixel_length) in (None, desired):
            continue

        palette = list(rng.choice(PALETTE_BANKS))
        rng.shuffle(palette)
        draw_order = list(range(4))
        rng.shuffle(draw_order)
        scene = {
            "series": [
                {
                    "label": SERIES_LABELS[i],
                    "before": round(before_values[i], 6),
                    "after": round(after_values[i], 6),
                }
                for i in range(4)
            ],
            "palette": [list(color) for color in palette],
            "draw_order": draw_order,
            "x_before": x_before,
            "x_after": x_after,
            "plot_top": plot_top,
            "plot_bottom": plot_bottom,
        }
        if not is_quarantined(scene) and analytic_gold(scene) == SERIES_LABELS[desired]:
            return scene
    raise RuntimeError(f"could not sample rank-gain scene for seed {seed}")


def _dash(draw: ImageDraw.ImageDraw, x: int, y0: int, y1: int) -> None:
    for y in range(y0, y1 + 1, 16):
        draw.line((x, y, x, min(y + 9, y1)), fill=GUIDE, width=3)


def render(scene: dict) -> Image.Image:
    """Render a clean two-column slopegraph and an explicit top-to-bottom legend."""
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    x0 = int(scene["x_before"])
    x1 = int(scene["x_after"])
    top = int(scene["plot_top"])
    bottom = int(scene["plot_bottom"])

    draw.text((34, 20), "ORDINAL CHANGE", fill=INK, font=_font(24))
    draw.text((34, 49), "Compare ranks, not vertical distance", fill=MUTED, font=_font(15))
    _dash(draw, x0, top - 8, bottom + 8)
    _dash(draw, x1, top - 8, bottom + 8)
    draw.text((x0 - 32, bottom + 22), "BEFORE", fill=INK, font=_font(16))
    draw.text((x1 - 24, bottom + 22), "AFTER", fill=INK, font=_font(16))

    series = scene["series"]
    palette = [tuple(int(channel) for channel in color) for color in scene["palette"]]
    for index in scene["draw_order"]:
        item = series[index]
        color = palette[index]
        y0 = int(round(_point_y(scene, item["before"])))
        y1 = int(round(_point_y(scene, item["after"])))
        draw.line((x0, y0, x1, y1), fill=color, width=4)
    for index, item in enumerate(series):
        color = palette[index]
        y0 = int(round(_point_y(scene, item["before"])))
        y1 = int(round(_point_y(scene, item["after"])))
        for x, y in ((x0, y0), (x1, y1)):
            draw.ellipse((x - 9, y - 9, x + 9, y + 9), fill=color, outline=BACKGROUND, width=2)

    legend_x = 618
    legend_y = 124
    draw.rounded_rectangle((594, 88, 719, 278), radius=10, fill=(242, 245, 248), outline=(210, 216, 224), width=2)
    draw.text((616, 101), "SERIES", fill=INK, font=_font(15))
    for index, label in enumerate(SERIES_LABELS):
        y = legend_y + index * 36
        draw.rectangle((legend_x, y, legend_x + 21, y + 17), fill=palette[index])
        draw.text((650, y - 1), label, fill=INK, font=_font(18))
    return image
