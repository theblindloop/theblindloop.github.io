"""Deterministic raster renderer and analytic graph oracle for one candidate.

The scene contains four equal panels.  Each panel stores five node centres and
five undirected edge pairs.  The selected panel is the one whose graph has a
node of degree four.
"""

from __future__ import annotations

import copy
import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

CANVAS_SIZE = (960, 760)
PANEL_BOXES = (
    (35, 90, 465, 355),
    (495, 90, 925, 355),
    (35, 405, 465, 670),
    (495, 405, 925, 670),
)
LABELS = ("A", "B", "C", "D")
BACKGROUND = (255, 255, 255)
PANEL_BORDER = (184, 190, 200)
EDGE_COLOR = (66, 68, 72)
NODE_COLOR = (34, 104, 176)
NODE_EDGE = (20, 72, 132)
SUPER_SAMPLE = 4
EDGE_WIDTH = 7
NODE_RADIUS = 15

# One degree-four hub, followed by three five-edge graphs whose maximum degree
# is three or less.  Outer indices are in cyclic order, so every outer-to-outer
# edge is a short perimeter chord rather than a diameter through another node.
GRAPH_TEMPLATES = (
    ((0, 1), (0, 2), (0, 3), (0, 4), (1, 2)),
    ((0, 1), (1, 2), (2, 3), (3, 4), (4, 0)),
    ((0, 1), (0, 2), (0, 3), (1, 2), (3, 4)),
    ((0, 1), (0, 2), (1, 2), (2, 3), (3, 4)),
)


def _canonical_edges(edges: Iterable[Iterable[int]]) -> list[tuple[int, int]]:
    return sorted({tuple(sorted((int(edge[0]), int(edge[1])))) for edge in edges})


def _degree_scores(edge_lists: Iterable[Iterable[Iterable[int]]]) -> list[int]:
    scores: list[int] = []
    for edges in edge_lists:
        degrees = [0, 0, 0, 0, 0]
        for edge in edges:
            if len(edge) != 2:
                continue
            first, second = int(edge[0]), int(edge[1])
            if 0 <= first < 5 and 0 <= second < 5 and first != second:
                degrees[first] += 1
                degrees[second] += 1
        scores.append(max(degrees) if degrees else -1)
    return scores


def sample_scene(seed: int) -> dict:
    """Sample one balanced four-panel scene from an integer seed."""

    rng = random.Random(int(seed))
    target_panel = int(seed) % 4
    distractor_kinds = [1, 2, 3]
    rng.shuffle(distractor_kinds)
    kinds: list[int] = []
    distractor_index = 0
    for panel_index in range(4):
        if panel_index == target_panel:
            kinds.append(0)
        else:
            kinds.append(distractor_kinds[distractor_index])
            distractor_index += 1

    panels = []
    for panel_index, kind in enumerate(kinds):
        x0, y0, x1, y1 = PANEL_BOXES[panel_index]
        cx = (x0 + x1) / 2.0 + rng.uniform(-9.0, 9.0)
        cy = (y0 + y1) / 2.0 + rng.uniform(-8.0, 8.0)
        radius = rng.uniform(68.0, 78.0)
        rotation = rng.choice((0.0, math.pi / 2.0, math.pi, 3.0 * math.pi / 2.0))
        rotation += rng.uniform(-0.045, 0.045)
        direction = rng.choice((-1.0, 1.0))
        nodes = [[round(cx, 5), round(cy, 5)]]
        for outer_index in range(4):
            angle = rotation + direction * (math.pi / 2.0) * outer_index
            nodes.append(
                [
                    round(cx + radius * math.cos(angle), 5),
                    round(cy + radius * math.sin(angle), 5),
                ]
            )
        edges = [list(edge) for edge in GRAPH_TEMPLATES[kind]]
        rng.shuffle(edges)
        panels.append({"nodes": nodes, "edges": edges})

    return {"panels": panels}


def render(scene: dict) -> Image.Image:
    """Render only the scene's visible panels, nodes, and edge geometry."""

    scale = SUPER_SAMPLE
    image = Image.new("RGB", (CANVAS_SIZE[0] * scale, CANVAS_SIZE[1] * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)

    for panel_index, panel in enumerate(scene["panels"]):
        x0, y0, x1, y1 = PANEL_BOXES[panel_index]
        draw.rectangle(
            (x0 * scale, y0 * scale, x1 * scale, y1 * scale),
            outline=PANEL_BORDER,
            width=2 * scale,
        )
        nodes = panel["nodes"]
        for edge in panel["edges"]:
            first, second = int(edge[0]), int(edge[1])
            start = (round(nodes[first][0] * scale), round(nodes[first][1] * scale))
            end = (round(nodes[second][0] * scale), round(nodes[second][1] * scale))
            draw.line((start, end), fill=EDGE_COLOR, width=EDGE_WIDTH * scale)
        for node in nodes:
            cx, cy = float(node[0]) * scale, float(node[1]) * scale
            radius = NODE_RADIUS * scale
            draw.ellipse(
                (round(cx - radius), round(cy - radius), round(cx + radius), round(cy + radius)),
                fill=NODE_COLOR,
                outline=NODE_EDGE,
                width=2 * scale,
            )

    image = image.resize(CANVAS_SIZE, Image.Resampling.LANCZOS)
    # Labels are deliberately outside the network crop used by the inverse.
    label_draw = ImageDraw.Draw(image)
    for label, (x0, y0, _x1, _y1) in zip(LABELS, PANEL_BOXES):
        label_draw.text((x0 + 16, y0 + 12), label, fill=(20, 25, 32))
    return image


def analytic_gold(scene: dict) -> str:
    """Return the panel label whose visible graph has maximum degree four."""

    edge_lists = [panel["edges"] for panel in scene["panels"]]
    scores = _degree_scores(edge_lists)
    best = max(scores)
    winners = [index for index, score in enumerate(scores) if score == best]
    if len(winners) != 1 or best != 4:
        raise ValueError(f"scene is not uniquely hub-labelled: scores={scores}")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    """Return the maximum-degree runner-up gap in degree units."""

    edge_lists = [panel["edges"] for panel in scene["panels"]]
    scores = sorted(_degree_scores(edge_lists))
    if len(scores) < 2:
        return 0.0
    return float(scores[-1] - scores[-2])


def is_quarantined(scene: dict) -> bool:
    """Reject malformed or non-unique finite-resolution graph scenes."""

    panels = scene.get("panels") if isinstance(scene, dict) else None
    if not isinstance(panels, list) or len(panels) != 4:
        return True
    for panel in panels:
        if not isinstance(panel, dict):
            return True
        nodes = panel.get("nodes")
        edges = panel.get("edges")
        if not isinstance(nodes, list) or len(nodes) != 5:
            return True
        if not isinstance(edges, list) or len(edges) != 5:
            return True
        if any(
            not isinstance(node, list)
            or len(node) != 2
            or not all(isinstance(value, (int, float)) for value in node)
            for node in nodes
        ):
            return True
        if len(_canonical_edges(edges)) != 5:
            return True
        if any(
            len(edge) != 2
            or int(edge[0]) != edge[0]
            or int(edge[1]) != edge[1]
            or not (0 <= int(edge[0]) < 5 and 0 <= int(edge[1]) < 5)
            or int(edge[0]) == int(edge[1])
            for edge in edges
        ):
            return True
        for first in range(5):
            for second in range(first):
                if math.dist(nodes[first], nodes[second]) < 38.0:
                    return True
    edge_lists = [panel["edges"] for panel in panels]
    scores = _degree_scores(edge_lists)
    ordered = sorted(scores)
    return ordered[-1] != 4 or ordered[-1] - ordered[-2] < 1


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Exercise the edge-record-order alias: drawing same-color edges commutes."""

    twin = copy.deepcopy(scene)
    for panel in twin["panels"]:
        panel["edges"] = list(reversed(panel["edges"]))
    return [("reverse_edge_record_order", twin)]
