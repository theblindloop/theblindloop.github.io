"""Deterministic Pillow world for counting visible prongs on a trident."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

PALETTE = ("#1769aa", "#3155a6", "#0877b7", "#264f9e")
COUNTS = (2, 3, 4, 5)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced control/counterfactual scene from one integer seed."""
    rng = random.Random(int(seed))
    prong_count = COUNTS[int(seed) % len(COUNTS)]
    width, height = 640, 480
    # Every answer class occupies a similarly broad horizontal support; the
    # answer cannot be guessed from the icon's outer width.
    total_span = rng.randint(270, 320)
    spacing = total_span / (prong_count - 1)
    stroke_width = rng.randint(10, 12)
    total = spacing * (prong_count - 1)
    left_group = bool(rng.randrange(2))
    center_x = rng.randint(150, 190) if left_group else rng.randint(450, 490)
    center_x = max(42 + total // 2, min(width - 42 - total // 2, center_x))
    bottom_x = rng.randint(505, 550) if left_group else rng.randint(90, 135)
    top_y = rng.randint(32, 48)
    junction_y = rng.randint(255, 285)
    bottom_y = rng.randint(424, 448)
    color = PALETTE[rng.randrange(len(PALETTE))]
    return {
        "prong_count": prong_count,
        "width": width,
        "height": height,
        "center_x": center_x,
        "top_y": top_y,
        "junction_y": junction_y,
        "bottom_x": bottom_x,
        "bottom_y": bottom_y,
        "spacing": spacing,
        "stroke_width": stroke_width,
        "color": color,
    }


def _tip_xs(scene: dict[str, Any]) -> list[float]:
    count = int(scene["prong_count"])
    center_x = float(scene["center_x"])
    spacing = float(scene["spacing"])
    return [center_x + (index - (count - 1) / 2.0) * spacing for index in range(count)]


def render(scene: dict[str, Any]) -> Image.Image:
    """Draw one connected, thin branching trace with pointed upper terminals."""
    count = int(scene["prong_count"])
    width = int(scene["width"])
    height = int(scene["height"])
    center_x = float(scene["center_x"])
    top_y = float(scene["top_y"])
    junction_y = float(scene["junction_y"])
    bottom_x = float(scene["bottom_x"])
    bottom_y = float(scene["bottom_y"])
    spacing = float(scene["spacing"])
    stroke_width = int(scene["stroke_width"])
    color = str(scene["color"])
    if count not in COUNTS or width < 64 or height < 64 or spacing <= 0 or stroke_width <= 0:
        raise ValueError("unsupported scene")

    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), "white")
    draw = ImageDraw.Draw(image)
    sw = stroke_width * scale
    tip_y = top_y * scale
    neck_y = (top_y + 34) * scale
    shoulder_y = junction_y * scale
    jx, jy = center_x * scale, junction_y * scale

    # The lower stem is shared by every branch. A round bottom avoids creating
    # another upward point and provides a distant connectivity check.
    draw.line(
        [(bottom_x * scale, bottom_y * scale), (jx, jy)],
        fill=color,
        width=sw,
        joint="curve",
    )
    radius = sw / 2
    draw.ellipse(
        [bottom_x * scale - radius, bottom_y * scale - radius,
         bottom_x * scale + radius, bottom_y * scale + radius],
        fill=color,
    )

    tip_xs = _tip_xs(scene)
    # A long, thin shoulder connects all vertical prongs before meeting the
    # stem. This makes the visible structure a comb-like trident rather than a
    # filled fan: distant evidence remains joined by one pathlike component.
    draw.line(
        [(tip_xs[0] * scale, shoulder_y), (tip_xs[-1] * scale, shoulder_y)],
        fill=color,
        width=sw,
    )
    for x in tip_xs:
        sx = x * scale
        # A narrow pointed triangular cap is visibly distinct from the shaft.
        half = (stroke_width * 1.45) * scale
        draw.polygon([(sx, tip_y), (sx - half, neck_y), (sx + half, neck_y)], fill=color)
        draw.line([(sx, neck_y - scale), (sx, shoulder_y)], fill=color, width=sw)
        draw.ellipse(
            [sx - radius, shoulder_y - radius, sx + radius, shoulder_y + radius],
            fill=color,
        )

    draw.ellipse([jx - radius, jy - radius, jx + radius, jy + radius], fill=color)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(int(scene["prong_count"]))


def margin(scene: dict[str, Any]) -> float:
    """Runner-up margin: smallest tip separation after visible tip footprints."""
    spacing = float(scene["spacing"])
    stroke_width = float(scene["stroke_width"])
    side_clearance = min(_tip_xs(scene)[0], float(scene["width"]) - _tip_xs(scene)[-1])
    return float(min(spacing - 3.2 * stroke_width, side_clearance - 1.5 * stroke_width))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 20.0 or float(scene["top_y"]) < 20.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # The public scene is intentionally injective for declared fields.
    return []
