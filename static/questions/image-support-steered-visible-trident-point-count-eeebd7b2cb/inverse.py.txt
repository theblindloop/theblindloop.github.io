"""Independent image-only inverse arm for the connected trident trace."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    """The raster no longer exposes a stable connected trident."""


def _ink_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    # Colored trace is saturated and dark; white and antialias fringes are not.
    maximum = array.max(axis=2)
    minimum = array.min(axis=2)
    return ((maximum - minimum) > 35) & (array.mean(axis=2) < 225)


def _components(mask: np.ndarray, minimum: int = 1) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[np.ndarray] = []
    for y0, x0 in np.argwhere(mask):
        y, x = int(y0), int(x0)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            groups.append(np.asarray(points, dtype=np.int32))
    return groups


def decision_from_image(image: Image.Image) -> str:
    """Recover the count without coordinates, scene state, or a canonical prior."""
    mask = _ink_mask(image)
    components = _components(mask, minimum=12)
    if len(components) != 1:
        raise OracleAbstention("colored trace is absent or disconnected")
    points = components[0]
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    span_h = int(y1 - y0 + 1)
    span_w = int(x1 - x0 + 1)
    height, width = mask.shape
    if span_h < 0.68 * height or span_w < 0.20 * width:
        raise OracleAbstention("trace does not span a trident-like extent")
    if len(points) < 0.008 * width * height:
        raise OracleAbstention("trace is too thin or corrupted")

    # In the upper 13% of the trace height, each pointed cap is a separate
    # component. This band is derived from the foreground box, not fixed scene
    # coordinates. Its large vertical headroom tolerates antialiasing and style.
    cutoff = int(round(y0 + 0.13 * span_h))
    top = mask[int(y0) : cutoff + 1, int(x0) : int(x1) + 1]
    tip_groups = _components(top, minimum=max(20, int(0.00018 * width * height)))
    count = len(tip_groups)
    if count not in (2, 3, 4, 5):
        raise OracleAbstention(f"unstable visible tip count {count}")
    centers = sorted(float(group[:, 1].mean()) for group in tip_groups)
    if len(centers) > 1 and min(np.diff(centers)) < 0.075 * width:
        raise OracleAbstention("neighboring tips are not cleanly separated")

    # Both a shared junction below the shoulders and a distant lower stem must
    # remain visible; a crop of only the caps is insufficient.
    lower_start = int(round(y0 + 0.70 * span_h))
    lower = mask[lower_start : int(y1) + 1]
    if not lower.any():
        raise OracleAbstention("shared lower stem is missing")
    # The handle may lean left or right. Every occupied lower scanline must
    # still form one narrow run, rather than a second fan of branches.
    checked = 0
    for row in lower[:: max(1, lower.shape[0] // 12)]:
        xs = np.flatnonzero(row)
        if len(xs):
            checked += 1
            if int(xs.max() - xs.min() + 1) > 0.08 * width:
                raise OracleAbstention("lower handle has multiple or broad branches")
    if checked < 4:
        raise OracleAbstention("lower handle is not continuously visible")
    return str(count)
