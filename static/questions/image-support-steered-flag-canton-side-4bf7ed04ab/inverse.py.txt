"""Independent image-only inverse for the flag canton."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _largest_component(mask: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    best: list[tuple[int, int]] = []
    for yy, xx in np.argwhere(mask):
        y, x = int(yy), int(xx)
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = deque([(y, x)])
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) > len(best):
            best = points
    output = np.zeros_like(mask, dtype=bool)
    if best:
        yy, xx = zip(*best)
        output[np.asarray(yy), np.asarray(xx)] = True
    return output


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[..., 0], array[..., 1], array[..., 2]
    blue_mask = (blue > 95) & (blue > red + 38) & (blue > green + 24) & (red < 90)
    component = _largest_component(blue_mask)
    yy, xx = np.where(component)
    if len(xx) < 30000:
        raise ValueError("solid blue canton is missing or materially erased")
    x0, x1 = int(xx.min()), int(xx.max())
    y0, y1 = int(yy.min()), int(yy.max())
    box_area = (x1 - x0 + 1) * (y1 - y0 + 1)
    fill = len(xx) / float(box_area)
    if x1 - x0 < 175 or y1 - y0 < 160 or fill < 0.955:
        raise ValueError("blue canton is not a complete solid rectangle")

    # Recover the flag bounds from non-background, non-shadow chromatic ink.
    chroma = array.max(axis=2) - array.min(axis=2)
    foreground = (chroma > 22) & ~blue_mask
    fy, fx = np.where(foreground)
    if len(fx) < 12000:
        raise ValueError("striped flag field is missing")
    flag_mid = (float(fx.min()) + float(fx.max())) / 2.0
    canton_mid = float(xx.mean())
    separation = canton_mid - flag_mid
    if abs(separation) < 55.0:
        raise ValueError("canton side is ambiguous")
    return "left" if separation < 0.0 else "right"
