"""Deterministic Pillow renderer with closed-form scene gold."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw


CANVAS = (512, 512)
BLUE = (28, 73, 145)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    flag_w = rng.randrange(398, 431, 2)
    flag_h = rng.randrange(286, 313, 2)
    flag_x = (CANVAS[0] - flag_w) // 2 + rng.randint(-14, 14)
    flag_y = (CANVAS[1] - flag_h) // 2 + rng.randint(-14, 14)
    palettes = [
        ((184, 48, 58), (244, 235, 211)),
        ((165, 54, 66), (236, 228, 204)),
        ((196, 61, 48), (247, 239, 217)),
        ((174, 42, 53), (241, 232, 207)),
    ]
    red_rgb, light_rgb = palettes[rng.randrange(len(palettes))]
    return {
        "canton_side": "left" if int(seed) % 2 == 0 else "right",
        "flag_x": flag_x,
        "flag_y": flag_y,
        "flag_w": flag_w,
        "flag_h": flag_h,
        "canton_w": rng.randrange(202, 223, 2),
        "canton_h": rng.randrange(184, 207, 2),
        "stripe_count": rng.choice((7, 9, 11)),
        "stripe_phase": rng.randrange(2),
        "red_rgb": list(red_rgb),
        "light_rgb": list(light_rgb),
        "border_rgb": list(rng.choice(((43, 48, 55), (60, 56, 51), (35, 45, 55)))),
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS, (250, 248, 242))
    draw = ImageDraw.Draw(image)
    x = int(scene["flag_x"])
    y = int(scene["flag_y"])
    w = int(scene["flag_w"])
    h = int(scene["flag_h"])
    stripes = int(scene["stripe_count"])
    phase = int(scene["stripe_phase"])
    red = tuple(scene["red_rgb"])
    light = tuple(scene["light_rgb"])
    border = tuple(scene["border_rgb"])

    # A small neutral shadow gives the object a visible boundary without encoding its label.
    draw.rounded_rectangle((x + 7, y + 8, x + w + 7, y + h + 8), radius=7, fill=(215, 211, 202))
    for index in range(stripes):
        sy0 = y + (h * index) // stripes
        sy1 = y + (h * (index + 1)) // stripes - 1
        fill = red if (index + phase) % 2 == 0 else light
        draw.rectangle((x, sy0, x + w - 1, sy1), fill=fill)
    draw.rectangle((x, y, x + w - 1, y + h - 1), outline=border, width=5)

    cw = int(scene["canton_w"])
    ch = int(scene["canton_h"])
    side = scene["canton_side"]
    cx = x + 5 if side == "left" else x + w - 5 - cw
    cy = y + 5
    draw.rectangle((cx, cy, cx + cw - 1, cy + ch - 1), fill=BLUE)
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["canton_side"])


def margin(scene: dict) -> float:
    x = float(scene["flag_x"])
    w = float(scene["flag_w"])
    cw = float(scene["canton_w"])
    cx = x + 5.0 + cw / 2.0 if scene["canton_side"] == "left" else x + w - 5.0 - cw / 2.0
    return abs(cx - (x + w / 2.0))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 60.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
