def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced scene with the label independent of the answer."""

    rng = random.Random(int(seed))
    column_count = rng.choice(tuple(range(4, 11)))
    row_count = rng.choice(tuple(range(4, 9)))

    # Most scenes have generous cells; a minority deliberately approach the
    # finite-resolution boundary and are explicitly quarantined.
    near_boundary = rng.random() < 0.16
    if near_boundary:
        board_width = rng.uniform(190.0, 270.0)
        board_height = rng.uniform(155.0, 220.0)
    else:
        board_width = rng.uniform(400.0, 540.0)
        board_height = rng.uniform(255.0, 310.0)

    board_left = rng.uniform(54.0, WIDTH - board_width - 54.0)
    board_top = rng.uniform(145.0, HEIGHT - board_height - 26.0)
    style = STYLE_PALETTES[rng.randrange(len(STYLE_PALETTES))]
    border_width = rng.choice((7.0, 8.0, 9.0))

    return {
        "column_count": int(column_count),
        "row_count": int(row_count),
        "board_left": round(board_left, 3),
        "board_top": round(board_top, 3),
        "board_width": round(board_width, 3),
        "board_height": round(board_height, 3),
        "cell_a": list(style["cell_a"]),
        "cell_b": list(style["cell_b"]),
        "background": list(style["background"]),
        "outline": list(style["outline"]),
        "header_color": list(style["header"]),
        "accent_color": list(style["accent"]),
        "header_text": "CHECKERBOARD",
        "border_width": border_width,
    }


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _box(values: tuple[float, float, float, float]) -> tuple[float, ...]:
    return tuple(value * SUPERSAMPLE for value in values)


def render(scene: dict[str, Any]) -> Image.Image:
    """Render one labeled checkerboard as an RGB PNG-sized image."""

    ss = SUPERSAMPLE
    image = Image.new("RGB", (WIDTH * ss, HEIGHT * ss), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)

    left = float(scene["board_left"])
    top = float(scene["board_top"])
    width = float(scene["board_width"])
    height = float(scene["board_height"])
    columns = int(scene["column_count"])
    rows = int(scene["row_count"])
    border = float(scene["border_width"])
    outline = tuple(scene["outline"])
    cell_a = tuple(scene["cell_a"])
    cell_b = tuple(scene["cell_b"])

    # A quiet shadow and a solid outer frame provide a stable object boundary
    # without introducing any answer-bearing mark.
    draw.rounded_rectangle(
        _box((left + 5.0, top + 7.0, left + width + 5.0, top + height + 7.0)),
        radius=12 * ss,
        fill=(218, 222, 220),
    )
    draw.rounded_rectangle(
        _box((left, top, left + width, top + height)),
        radius=12 * ss,
        fill=outline,
    )

    inner_left = left + border
    inner_top = top + border
    inner_right = left + width - border
    inner_bottom = top + height - border
    cell_width = (inner_right - inner_left) / columns
    cell_height = (inner_bottom - inner_top) / rows

    for row in range(rows):
        for column in range(columns):
            x0 = inner_left + column * cell_width
            x1 = inner_left + (column + 1) * cell_width
            y0 = inner_top + row * cell_height
            y1 = inner_top + (row + 1) * cell_height
            fill = cell_a if (row + column) % 2 == 0 else cell_b
            draw.rectangle(_box((x0, y0, x1, y1)), fill=fill)

    grid_width = max(2, int(round(2.5 * ss)))
    for column in range(1, columns):
        x = inner_left + column * cell_width
        draw.line(
            (x * ss, inner_top * ss, x * ss, inner_bottom * ss),
            fill=outline,
            width=grid_width,
        )
    for row in range(1, rows):
        y = inner_top + row * cell_height
        draw.line(
            (inner_left * ss, y * ss, inner_right * ss, y * ss),
            fill=outline,
            width=grid_width,
        )

    center_x = left + width / 2.0
    label_y = max(42.0, top - 40.0)
    draw.text(
        (center_x * ss, label_y * ss),
        str(scene["header_text"]),
        fill=tuple(scene["header_color"]),
        anchor="mm",
        font=_font(27 * ss),
    )
    underline_y = label_y + 21.0
    draw.line(
        ((center_x - 54.0) * ss, underline_y * ss, (center_x + 54.0) * ss, underline_y * ss),
        fill=tuple(scene["accent_color"]),
        width=max(2, int(round(3.0 * ss))),
    )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    """The sole label input is the visible column cardinality."""

    return str(int(scene["column_count"]))


def margin(scene: dict[str, Any]) -> float:
    """Smallest cell extent after allowing for the visible grid rule."""

    width = (float(scene["board_width"]) - 2.0 * float(scene["border_width"])) / int(
        scene["column_count"]
    )
    height = (float(scene["board_height"]) - 2.0 * float(scene["border_width"])) / int(
        scene["row_count"]
    )
    return float(min(width, height) - 3.0)


def is_quarantined(scene: dict[str, Any]) -> bool:
    width = (float(scene["board_width"]) - 2.0 * float(scene["border_width"])) / int(
        scene["column_count"]
    )
    height = (float(scene["board_height"]) - 2.0 * float(scene["border_width"])) / int(
        scene["row_count"]
    )
    left = float(scene["board_left"])
    top = float(scene["board_top"])
    right = left + float(scene["board_width"])
    bottom = top + float(scene["board_height"])
    return (
        min(width, height) < 18.0
        or min(left, top, WIDTH - right, HEIGHT - bottom) < 12.0
    )


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No latent transform is pixel-identical under this non-periodic layout."""

    return []
