"""Independent image-only inverse for checkerboard column cardinality."""

from __future__ import annotations

import numpy as np
from PIL import Image

# Duplicated public render palette.  This module intentionally has no import or
# data path to renderer.py, prompts.py, generate.py, verify.py, or scene state.
CELL_COLORS = np.array(
    [
        (231, 241, 246),
        (55, 111, 145),
        (246, 232, 215),
        (178, 86, 46),
        (233, 239, 220),
        (88, 126, 57),
        (241, 228, 243),
        (126, 78, 139),
        (226, 240, 237),
        (42, 130, 121),
        (246, 236, 207),
        (184, 126, 39),
    ],
    dtype=np.float32,
)


def _cell_mask(image: Image.Image) -> np.ndarray:
    pixels = np.asarray(image.convert("RGB"), dtype=np.float32)
    distances = pixels[:, :, None, :] - CELL_COLORS[None, None, :, :]
    nearest = np.sqrt(np.sum(distances * distances, axis=-1)).min(axis=-1)
    # Exact interior pixels survive the fixed supersampled rasterization.  A
    # tight threshold keeps a different style's pale background from becoming a
    # false cell run.
    return nearest <= 10.0


def _runs(row: np.ndarray) -> list[tuple[int, int]]:
    padded = np.concatenate((np.array([False]), row, np.array([False])))
    changes = np.flatnonzero(padded[1:] != padded[:-1])
    return [
        (int(start), int(end))
        for start, end in zip(changes[0::2], changes[1::2])
        if int(end - start) >= 5
    ]


def decision_from_image(image: Image.Image) -> str:
    """Recover the number of vertical cell runs from the final raster alone."""

    mask = _cell_mask(image)
    best_total = 0
    best_count = 0
    for row in mask:
        runs = _runs(row)
        # A pale antialiased background can form one long false run.  A valid
        # checkerboard scanline has at least four separated vertical bands.
        if len(runs) < 4:
            continue
        total = sum(end - start for start, end in runs)
        if total > best_total or (total == best_total and len(runs) > best_count):
            best_total = total
            best_count = len(runs)
    if best_count < 4 or best_total < 40:
        raise ValueError("no stable checkerboard scanline was recovered")
    return str(best_count)
