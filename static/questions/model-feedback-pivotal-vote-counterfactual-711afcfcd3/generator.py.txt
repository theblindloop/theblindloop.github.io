"""Pivotal-vote world: which single hypothetical flip would change the summary.

The raster shows a two-stage majority aggregator drawn as a shallow tree. The
bottom row holds fifteen disks in five linked triples; each disk is either
filled black or hollow (white). Each of the five hollow middle circles is the
majority color of the three disks linked to it, and the hollow top circle is
the majority color of the five middle circles. No middle value and no top value
is ever drawn: the hollow circles are structure, not evidence.

Three of the fifteen bottom disks are ringed, one in amber, one in teal and one
in violet. Exactly one of those three is *pivotal*: flipping that single disk's
color (black to white, or white to black) and nothing else would change the top
circle's color. The other two ringed disks are inert, each for a different
reason: one flips its own middle circle but that middle circle is on the losing
side of the top vote, and one cannot move its middle circle at all. The
decision is the ring color of the pivotal disk.

Answering therefore requires evaluating the drawn world and three edited worlds
and reporting which edit is decisive; the quantity reported is a *difference
between worlds*, and none of the four worlds' aggregate values appear anywhere
in the raster.

The renderer owns scene sampling, rasterization and the analytic gold. The
independent pixel-only inverse arm (``oracle.py``) recovers the same ring color
from the final PNG alone.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 1480
HEIGHT = 780

GROUPS = 5
PER_GROUP = 3

LEAF_Y = 672
NODE_Y = 452
ROOT_Y = 232
CENTER_X = 740
GROUP_STEP = 300
LEAF_PITCH = 84

LEAF_R = 22
NODE_R = 30
ROOT_R = 38
OUTLINE_W = 6
RING_IN = 34
RING_OUT = 42

JITTER = 6
GAP_LO = 12
GAP_HI = 18

BACKGROUND = (255, 255, 255)
INK = (0, 0, 0)
EDGE_W = 5

RING_PALETTE = {
    "amber": (232, 150, 20),
    "teal": (0, 148, 148),
    "violet": (150, 60, 200),
}
RING_NAMES = ("amber", "teal", "violet")

QUARANTINE_PX = 11.0


# --------------------------------------------------------------------------
# mechanism
# --------------------------------------------------------------------------
def _majority(bits) -> int:
    return 1 if sum(bits) * 2 > len(bits) else 0


def _node_values(leaves) -> list:
    return [_majority(group) for group in leaves]


def _root_value(leaves) -> int:
    return _majority(_node_values(leaves))


def _flip(leaves, group: int, slot: int) -> list:
    edited = [list(g) for g in leaves]
    edited[group][slot] = 1 - edited[group][slot]
    return edited


def _is_pivotal(leaves, group: int, slot: int) -> bool:
    """True when flipping one bottom disk changes the top circle's color."""
    return _root_value(_flip(leaves, group, slot)) != _root_value(leaves)


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------
def _try_scene(rng: random.Random) -> dict | None:
    root = rng.randint(0, 1)
    order = list(range(GROUPS))
    rng.shuffle(order)
    winners = order[:3]           # middle circles agreeing with the top circle
    losers = order[3:]            # middle circles disagreeing with it

    pivot_group = rng.choice(winners)
    inert_group = rng.choice(losers)
    rest = [g for g in range(GROUPS) if g not in (pivot_group, inert_group)]
    third_group = rng.choice(rest)

    node_value = [root if g in winners else 1 - root for g in range(GROUPS)]
    leaves = []
    splits = []
    for g in range(GROUPS):
        value = node_value[g]
        if g in (pivot_group, inert_group):
            split = "two_one"
        elif g == third_group:
            split = rng.choice(("two_one", "unanimous"))
        else:
            split = rng.choice(("two_one", "unanimous"))
        splits.append(split)
        bits = [value] * PER_GROUP
        if split == "two_one":
            bits[rng.randrange(PER_GROUP)] = 1 - value
        leaves.append(bits)

    def majority_slots(g: int) -> list:
        return [s for s in range(PER_GROUP) if leaves[g][s] == node_value[g]]

    def minority_slots(g: int) -> list:
        return [s for s in range(PER_GROUP) if leaves[g][s] != node_value[g]]

    pivot_slot = rng.choice(majority_slots(pivot_group))
    inert_slot = rng.choice(majority_slots(inert_group))
    if splits[third_group] == "unanimous":
        third_slot = rng.randrange(PER_GROUP)
    else:
        third_slot = rng.choice(minority_slots(third_group))

    marked = [
        [pivot_group, pivot_slot],
        [inert_group, inert_slot],
        [third_group, third_slot],
    ]

    # the pivotal disk must be unique among the three ringed disks
    flags = [_is_pivotal(leaves, g, s) for g, s in marked]
    if flags != [True, False, False]:
        return None

    colors = list(RING_NAMES)
    rng.shuffle(colors)
    rings = [[marked[i][0], marked[i][1], colors[i]] for i in range(3)]
    rings.sort(key=lambda item: (item[0], item[1]))

    leaf_jit = [
        [[rng.randint(-JITTER, JITTER), rng.randint(-JITTER, JITTER)]
         for _ in range(PER_GROUP)]
        for _ in range(GROUPS)
    ]
    node_jit = [[rng.randint(-JITTER, JITTER), rng.randint(-JITTER, JITTER)]
                for _ in range(GROUPS)]
    root_jit = [rng.randint(-JITTER, JITTER), rng.randint(-JITTER, JITTER)]
    leaf_gap = [[rng.randint(GAP_LO, GAP_HI) for _ in range(PER_GROUP)]
                for _ in range(GROUPS)]
    node_gap = [rng.randint(GAP_LO, GAP_HI) for _ in range(GROUPS)]

    return {
        "leaves": leaves,
        "rings": rings,
        "leaf_jit": leaf_jit,
        "node_jit": node_jit,
        "root_jit": root_jit,
        "leaf_gap": leaf_gap,
        "node_gap": node_gap,
    }


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    for _ in range(400):
        scene = _try_scene(rng)
        if scene is not None:
            return scene
    raise RuntimeError("failed to sample a scene")


# --------------------------------------------------------------------------
# geometry
# --------------------------------------------------------------------------
def _group_x(g: int) -> float:
    return float(CENTER_X + (g - (GROUPS - 1) / 2.0) * GROUP_STEP)


def _leaf_center(scene: dict, g: int, s: int) -> tuple:
    dx, dy = scene["leaf_jit"][g][s]
    return (_group_x(g) + (s - 1) * LEAF_PITCH + dx, float(LEAF_Y + dy))


def _node_center(scene: dict, g: int) -> tuple:
    dx, dy = scene["node_jit"][g]
    return (_group_x(g) + dx, float(NODE_Y + dy))


def _root_center(scene: dict) -> tuple:
    dx, dy = scene["root_jit"]
    return (float(CENTER_X + dx), float(ROOT_Y + dy))


def _ring_lookup(scene: dict) -> dict:
    return {(g, s): color for g, s, color in scene["rings"]}


def _leaf_outer(scene: dict, g: int, s: int) -> float:
    return RING_OUT if (g, s) in _ring_lookup(scene) else LEAF_R


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def _circle(draw, cx, cy, r, fill, outline, width):
    draw.ellipse([cx - r, cy - r, cx + r, cy + r], fill=fill,
                 outline=outline, width=width)


def _shorten(p0, p1, gap0, gap1):
    """Trim a segment by ``gap0`` at its start and ``gap1`` at its end."""
    x0, y0 = p0
    x1, y1 = p1
    dx, dy = x1 - x0, y1 - y0
    length = (dx * dx + dy * dy) ** 0.5
    ux, uy = dx / length, dy / length
    return ((x0 + ux * gap0, y0 + uy * gap0), (x1 - ux * gap1, y1 - uy * gap1))


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    rings = _ring_lookup(scene)
    root = _root_center(scene)

    # links: leaf -> middle circle, middle circle -> top circle
    for g in range(GROUPS):
        node = _node_center(scene, g)
        for s in range(PER_GROUP):
            leaf = _leaf_center(scene, g, s)
            gap = scene["leaf_gap"][g][s]
            start_gap = (RING_OUT if (g, s) in rings else LEAF_R) + gap
            a, b = _shorten(leaf, node, start_gap, NODE_R + gap)
            draw.line([a, b], fill=INK, width=EDGE_W)
        gap = scene["node_gap"][g]
        a, b = _shorten(node, root, NODE_R + gap, ROOT_R + gap)
        draw.line([a, b], fill=INK, width=EDGE_W)

    # top and middle circles are hollow: they display no value
    _circle(draw, root[0], root[1], ROOT_R, BACKGROUND, INK, OUTLINE_W)
    for g in range(GROUPS):
        node = _node_center(scene, g)
        _circle(draw, node[0], node[1], NODE_R, BACKGROUND, INK, OUTLINE_W)

    # bottom disks: filled black = black, hollow = white
    for g in range(GROUPS):
        for s in range(PER_GROUP):
            cx, cy = _leaf_center(scene, g, s)
            if scene["leaves"][g][s] == 1:
                _circle(draw, cx, cy, LEAF_R, INK, INK, 1)
            else:
                _circle(draw, cx, cy, LEAF_R, BACKGROUND, INK, OUTLINE_W)
            if (g, s) in rings:
                color = RING_PALETTE[rings[(g, s)]]
                draw.ellipse([cx - RING_OUT, cy - RING_OUT,
                              cx + RING_OUT, cy + RING_OUT],
                             fill=None, outline=color,
                             width=RING_OUT - RING_IN)
    return image


# --------------------------------------------------------------------------
# gold, margin, quarantine
# --------------------------------------------------------------------------
def analytic_gold(scene: dict) -> str:
    leaves = [list(g) for g in scene["leaves"]]
    hits = [(g, s, color) for g, s, color in scene["rings"]
            if _is_pivotal(leaves, g, s)]
    if len(hits) != 1:
        raise ValueError("scene does not have a unique pivotal ringed disk")
    return hits[0][2]


def _clearances(scene: dict) -> list:
    values = []
    leaf_pts = []
    for g in range(GROUPS):
        for s in range(PER_GROUP):
            leaf_pts.append((_leaf_center(scene, g, s), _leaf_outer(scene, g, s)))
    for i in range(len(leaf_pts)):
        for j in range(i + 1, len(leaf_pts)):
            (x0, y0), r0 = leaf_pts[i]
            (x1, y1), r1 = leaf_pts[j]
            dist = ((x0 - x1) ** 2 + (y0 - y1) ** 2) ** 0.5
            values.append(dist - r0 - r1)
    for g, s, _color in scene["rings"]:
        values.append(float(RING_IN - LEAF_R))       # disk to its own ring
    for g in range(GROUPS):
        node = _node_center(scene, g)
        for s in range(PER_GROUP):
            values.append(float(scene["leaf_gap"][g][s]))
            leaf = _leaf_center(scene, g, s)
            values.append(abs(leaf[1] - node[1]) - LEAF_R - NODE_R)
        values.append(float(scene["node_gap"][g]))
        for h in range(g + 1, GROUPS):
            other = _node_center(scene, h)
            dist = ((node[0] - other[0]) ** 2 + (node[1] - other[1]) ** 2) ** 0.5
            values.append(dist - 2 * NODE_R)
    return values


def margin(scene: dict) -> float:
    return float(min(_clearances(scene)))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list:
    """No latent aliases: rendering is injective over sampled scenes."""
    return []
