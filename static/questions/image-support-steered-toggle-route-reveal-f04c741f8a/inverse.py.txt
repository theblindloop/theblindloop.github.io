from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

GLYPHS = {
    "X": ("10001", "01010", "00100", "01010", "10001"),
    "plus": ("00100", "00100", "11111", "00100", "00100"),
    "T": ("11111", "00100", "00100", "00100", "00100"),
    "L": ("10000", "10000", "10000", "10000", "11111"),
}
ORANGE = np.array((242, 139, 32), dtype=np.int32)
NAVY = np.array((36, 55, 99), dtype=np.int32)
GREEN = np.array((52, 168, 83), dtype=np.int32)
PINK = np.array((224, 92, 132), dtype=np.int32)


def _mask(arr: np.ndarray, color: np.ndarray, tolerance: int = 32) -> np.ndarray:
    delta = arr.astype(np.int32) - color
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        xs, ys = [], []
        while queue:
            cy, cx = queue.popleft()
            xs.append(cx)
            ys.append(cy)
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = cy + dy, cx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= 45:
            found.append((float(np.mean(xs)), float(np.mean(ys)), len(xs)))
    return found


def _template_bits(name: str) -> list[int]:
    return [int(value) for row in GLYPHS[name] for value in row]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    beads = []
    for color, value in ((ORANGE, 1), (NAVY, 0)):
        for x, y, area in _components(_mask(arr, color)):
            if 45 <= area <= 180:
                beads.append((x, y, value))
    beads.sort(key=lambda item: item[0])
    if len(beads) != 50:
        raise ValueError(f"expected 50 instruction beads, found {len(beads)}")
    if any(beads[i + 1][0] - beads[i][0] < 10 for i in range(49)):
        raise ValueError("instruction order is ambiguous")
    green_area = int(_mask(arr, GREEN).sum())
    pink_area = int(_mask(arr, PINK).sum())
    if max(green_area, pink_area) < 150 or min(green_area, pink_area) > 20:
        raise ValueError("START state is missing or ambiguous")
    state = 1 if green_area > pink_area else 0
    decoded = []
    for index in range(0, 50, 2):
        state ^= beads[index][2]
        state ^= beads[index + 1][2]
        decoded.append(state)
    scores = sorted((sum(a != b for a, b in zip(decoded, _template_bits(name))), name) for name in GLYPHS)
    if scores[0][0] != 0 or scores[1][0] - scores[0][0] < 4:
        raise ValueError("decoded raster is not an unambiguous supported symbol")
    return scores[0][1]
