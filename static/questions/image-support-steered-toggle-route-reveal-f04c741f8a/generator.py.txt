from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 1024, 512
GLYPHS = {
    "X": ("10001", "01010", "00100", "01010", "10001"),
    "plus": ("00100", "00100", "11111", "00100", "00100"),
    "T": ("11111", "00100", "00100", "00100", "00100"),
    "L": ("10000", "10000", "10000", "10000", "11111"),
}


def _bits(glyph: str) -> list[int]:
    return [int(value) for row in GLYPHS[glyph] for value in row]


def _instructions(scene: dict[str, Any]) -> tuple[int, list[int]]:
    desired = _bits(scene["glyph"])
    target = int(scene["target_flips"])
    initial = desired[-1] ^ (target & 1)
    state = initial
    pairs: list[list[int]] = []
    stable = []
    for index, value in enumerate(desired):
        change = state ^ value
        if change:
            pair = [1, 0]
            if int(scene["pair_orders"][index]):
                pair.reverse()
        else:
            pair = [0, 0]
            stable.append(index)
        pairs.append(pair)
        state = value
    base = sum(sum(pair) for pair in pairs)
    needed = (target - base) // 2
    choices = [int(value) for value in scene["double_cells"] if int(value) in stable]
    if target < base or (target - base) % 2 or len(choices) < needed:
        raise ValueError("invalid balanced instruction construction")
    for index in choices[:needed]:
        pairs[index] = [1, 1]
    flat = [value for pair in pairs for value in pair]
    if sum(flat) != target:
        raise ValueError("flip balancing failed")
    return initial, flat


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    glyph = tuple(GLYPHS)[int(seed) % 4]
    desired = _bits(glyph)
    target = 22 + rng.randrange(2)
    initial = desired[-1] ^ (target & 1)
    transitions = []
    state = initial
    for value in desired:
        transitions.append(state == value)
        state = value
    stable = [i for i, same in enumerate(transitions) if same]
    rng.shuffle(stable)
    return {
        "glyph": glyph,
        "phase": round(rng.uniform(-0.45, 0.45), 6),
        "amplitude": rng.randint(62, 78),
        "waves": rng.choice([1.5, 1.75, 2.0]),
        "target_flips": target,
        "pair_orders": [rng.randrange(2) for _ in range(25)],
        "double_cells": stable,
        "background": rng.choice(["ivory", "mist"]),
    }


def _route_points(scene: dict[str, Any]) -> list[tuple[float, float]]:
    points = []
    phase = float(scene["phase"])
    amp = float(scene["amplitude"])
    waves = float(scene["waves"])
    for i in range(50):
        x = 70.0 + i * (884.0 / 49.0)
        u = i / 49.0
        y = 256.0 + amp * math.sin(2.0 * math.pi * waves * u + phase)
        points.append((x, y))
    return points


def render(scene: dict[str, Any]) -> Image.Image:
    background = (249, 246, 236) if scene["background"] == "ivory" else (240, 246, 248)
    image = Image.new("RGB", (WIDTH, HEIGHT), background)
    draw = ImageDraw.Draw(image)
    points = _route_points(scene)
    initial, instructions = _instructions(scene)
    route = (20, 176, 190)
    draw.line(points, fill=route, width=4, joint="curve")
    sx, sy = points[0]
    start_fill = (52, 168, 83) if initial else (224, 92, 132)
    draw.polygon([(28, sy), (54, sy - 15), (54, sy + 15)], fill=start_fill, outline=(25, 45, 55), width=2)
    draw.line([(54, sy), points[0]], fill=route, width=4)
    ex, ey = points[-1]
    draw.ellipse((ex + 10, ey - 14, ex + 38, ey + 14), fill=background, outline=route, width=5)
    draw.line([points[-1], (ex + 10, ey)], fill=route, width=4)
    for (x, y), flip in zip(points, instructions):
        fill = (242, 139, 32) if flip else (36, 55, 99)
        draw.ellipse((x - 6, y - 6, x + 6, y + 6), fill=fill, outline=(255, 255, 255), width=1)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(scene["glyph"])


def margin(scene: dict[str, Any]) -> float:
    target = _bits(scene["glyph"])
    distances = sorted(sum(a != b for a, b in zip(target, _bits(name))) for name in GLYPHS)
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 4.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict]]:
    return []
