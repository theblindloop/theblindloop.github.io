"""Independent pixel-only inverse arm for the enclosure-depth-comparison world.

Receives ONLY the final PNG. It never imports or reads the renderer, the
analytic scene, prompts, or any renderer state. It recovers the decision by:

1. marking the green and red dot pixels and taking each dot's centroid;
2. taking the centroid of all black ring ink as the shared ring center;
3. labeling each black connected component (one per ring loop) and estimating
   that ring's radius as the mean distance of its ink from the shared center;
4. computing each dot's enclosure depth as the number of rings whose estimated
   radius exceeds the dot's distance from the shared center;
5. returning the color of the dot with the strictly greater depth.

The oracle thus recovers the nesting-depth topology of the network of closed
loops purely from the segmented pixels, without any renderer state.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "enclosure_depth_compare-oracle-0.4.0"

_GREEN = (0, 128, 0)
_RED = (230, 40, 40)
_DARK_MAX = 90


def _centroid(mask: np.ndarray) -> tuple[float, float] | None:
    ys, xs = np.nonzero(mask)
    if len(xs) == 0:
        return None
    return (float(xs.mean()), float(ys.mean()))


def _components(black: np.ndarray) -> list[np.ndarray]:
    """4-connected components of a boolean mask, returned as a list of masks."""
    h, w = black.shape
    labels = np.zeros_like(black, dtype=np.int32)
    current = 0
    for y in range(h):
        for x in range(w):
            if not black[y, x] or labels[y, x]:
                continue
            current += 1
            labels[y, x] = current
            stack = [(x, y)]
            while stack:
                cx, cy = stack.pop()
                for nx, ny in ((cx + 1, cy), (cx - 1, cy), (cx, cy + 1), (cx, cy - 1)):
                    if (
                        0 <= nx < w
                        and 0 <= ny < h
                        and black[ny, nx]
                        and labels[ny, nx] == 0
                    ):
                        labels[ny, nx] = current
                        stack.append((nx, ny))
    comps: list[np.ndarray] = []
    for label in range(1, current + 1):
        comps.append(labels == label)
    return comps


def _dot_mask(arr: np.ndarray, target: tuple[int, int, int]) -> np.ndarray:
    (tr, tg, tb) = target
    return (
        (arr[..., 0] == tr) & (arr[..., 1] == tg) & (arr[..., 2] == tb)
    )


def decision_from_image(image: Image.Image) -> str:
    """Which colored dot is enclosed by more closed loops, from pixels alone."""
    arr = np.asarray(image.convert("RGB")).astype(np.int16)

    dots: dict[str, tuple[float, float]] = {}
    for name, (r, g, b) in (("green", _GREEN), ("red", _RED)):
        c = _centroid(_dot_mask(arr, (r, g, b)))
        if c is not None:
            dots[name] = c
    if len(dots) < 2:
        return "green"

    black = (arr[..., 0] <= _DARK_MAX) & (arr[..., 1] <= _DARK_MAX) & (arr[..., 2] <= _DARK_MAX)
    center = _centroid(black)
    if center is None:
        return "green"

    radii: list[float] = []
    for comp in _components(black):
        ys, xs = np.nonzero(comp)
        dists = np.hypot(xs.astype(float) - center[0], ys.astype(float) - center[1])
        if len(dists):
            radii.append(float(dists.max()))
    if not radii:
        return "green"

    def depth(px: float, py: float) -> int:
        d = math.hypot(px - center[0], py - center[1])
        return sum(1 for r in radii if d < r)

    d_green = depth(dots["green"][0], dots["green"][1])
    d_red = depth(dots["red"][0], dots["red"][1])
    if d_green > d_red:
        return "green"
    if d_red > d_green:
        return "red"
    return "green"
