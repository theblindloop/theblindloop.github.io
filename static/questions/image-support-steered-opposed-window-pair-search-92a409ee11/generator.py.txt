"""Deterministic raster renderer for remote paired-window offset search."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 1200
HEIGHT = 900
QUANTITIES = 4
ROW_Y = (110, 310, 510, 710)
LEFT_X = 140
RIGHT_X = 1060
BADGE_HALF_WIDTH = 88
BADGE_HALF_HEIGHT = 68
WINDOW_RADIUS = 22
WINDOW_AMPLITUDES = (16, 32, 48, 64)
WINDOW_DY_VALUES = (-5, 0, 5)
BACKGROUNDS = ((250, 250, 247), (247, 250, 252), (249, 248, 252))
INK_COLORS = (
    (38, 74, 128),
    (142, 54, 65),
    (42, 116, 91),
    (116, 76, 151),
    (170, 94, 35),
    (34, 111, 132),
    (125, 55, 113),
    (74, 92, 49),
)


def _octagon(cx: int, cy: int) -> list[tuple[int, int]]:
    hw, hh = BADGE_HALF_WIDTH, BADGE_HALF_HEIGHT
    bevel = 22
    return [
        (cx - hw + bevel, cy - hh),
        (cx + hw - bevel, cy - hh),
        (cx + hw, cy - hh + bevel),
        (cx + hw, cy + hh - bevel),
        (cx + hw - bevel, cy + hh),
        (cx - hw + bevel, cy + hh),
        (cx - hw, cy + hh - bevel),
        (cx - hw, cy - hh + bevel),
    ]


def _draw_badge(draw: ImageDraw.ImageDraw, badge: dict, background: tuple[int, int, int]) -> None:
    cx = int(badge["x"])
    cy = int(badge["y"])
    draw.polygon(_octagon(cx, cy), fill=tuple(int(value) for value in badge["color"]))
    offset_x = int(badge["window_offset"])
    offset_y = int(badge["window_dy"])
    draw.ellipse(
        (
            cx + offset_x - WINDOW_RADIUS,
            cy + offset_y - WINDOW_RADIUS,
            cx + offset_x + WINDOW_RADIUS,
            cy + offset_y + WINDOW_RADIUS,
        ),
        fill=background,
    )


def _badge(*, x: int, y: int, color: tuple[int, int, int], offset: int, window_dy: int) -> dict:
    return {
        "x": int(x),
        "y": int(y),
        "color": [int(value) for value in color],
        "window_offset": int(offset),
        "window_dy": int(window_dy),
    }


def sample_scene(seed: int) -> dict:
    """Sample four balanced row pairs; the seed cycles the winning row."""

    rng = random.Random(int(seed))
    background = list(BACKGROUNDS[rng.randrange(len(BACKGROUNDS))])
    colors = list(INK_COLORS)
    rng.shuffle(colors)
    target_row = int(seed) % QUANTITIES
    amplitude = WINDOW_AMPLITUDES[rng.randrange(len(WINDOW_AMPLITUDES))]
    other_amplitudes = [value for value in WINDOW_AMPLITUDES if value != amplitude]
    rng.shuffle(other_amplitudes)
    target_sign = -1 if rng.randrange(2) == 0 else 1

    pair_offsets: list[tuple[int, int]] = [(target_sign * amplitude, -target_sign * amplitude)]
    # Every distractor has a nonzero signed sum: two same-sign pairs and one
    # unequal opposite-sign pair. Their row positions are independently mixed.
    a, b, c = other_amplitudes
    distractors = [(a, b), (-a, -b), (c, -a)]
    rng.shuffle(distractors)
    for row in range(QUANTITIES):
        if row != target_row:
            pair_offsets.insert(row, distractors.pop())

    badges: list[dict] = []
    for row, (left_offset, right_offset) in enumerate(pair_offsets):
        y = ROW_Y[row] + rng.randrange(-7, 8)
        left_y = y + rng.randrange(-3, 4)
        right_y = y + rng.randrange(-3, 4)
        left_x = LEFT_X + rng.randrange(-8, 9)
        right_x = RIGHT_X + rng.randrange(-8, 9)
        pair_colors = [colors[(2 * row) % len(colors)], colors[(2 * row + 1) % len(colors)]]
        if rng.randrange(2):
            pair_colors.reverse()
        badges.append(
            _badge(
                x=left_x,
                y=left_y,
                color=pair_colors[0],
                offset=left_offset,
                window_dy=rng.choice(WINDOW_DY_VALUES),
            )
        )
        badges.append(
            _badge(
                x=right_x,
                y=right_y,
                color=pair_colors[1],
                offset=right_offset,
                window_dy=rng.choice(WINDOW_DY_VALUES),
            )
        )

    rng.shuffle(badges)
    return {"badges": badges, "background": background}


def render(scene: dict) -> Image.Image:
    """Render only the visible badges and their enclosed background windows."""

    background = tuple(int(value) for value in scene["background"])
    image = Image.new("RGB", (WIDTH, HEIGHT), background)
    draw = ImageDraw.Draw(image)
    for badge in scene["badges"]:
        _draw_badge(draw, badge, background)
    return image


def _row_pairs(scene: dict) -> list[tuple[dict, dict]]:
    badges = list(scene["badges"])
    left = sorted((badge for badge in badges if int(badge["x"]) < WIDTH // 2), key=lambda item: int(item["y"]))
    right = sorted((badge for badge in badges if int(badge["x"]) >= WIDTH // 2), key=lambda item: int(item["y"]))
    if len(left) != QUANTITIES or len(right) != QUANTITIES:
        raise ValueError("scene does not contain four left and four right badges")
    return list(zip(left, right, strict=True))


def _scores(scene: dict) -> list[float]:
    return [
        float(abs(int(left["window_offset"]) + int(right["window_offset"])))
        for left, right in _row_pairs(scene)
    ]


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    best = min(scores)
    winners = [index for index, score in enumerate(scores) if score == best]
    if len(winners) != 1 or best != 0:
        raise ValueError("scene has no unique equal-and-opposite offset row")
    return str(winners[0] + 1)


def margin(scene: dict) -> float:
    try:
        scores = sorted(_scores(scene))
    except (KeyError, TypeError, ValueError):
        return 0.0
    if len(scores) < 2:
        return 0.0
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    try:
        scores = _scores(scene)
    except (KeyError, TypeError, ValueError):
        return True
    if len(scores) != QUANTITIES or min(scores) != 0:
        return True
    return margin(scene) < 8.0 or sum(score == 0 for score in scores) != 1


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Expose a nontrivial record-order alias with identical pixels."""

    twin = dict(scene)
    twin["badges"] = list(reversed(scene["badges"]))
    return [("reverse_badge_record_order", twin)]
