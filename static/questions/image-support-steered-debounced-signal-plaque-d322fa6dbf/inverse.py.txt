"""Independent pixel-only inverse for the debounced signal plaque."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

CYAN = np.array((0, 183, 199), dtype=np.int16)
CORAL = np.array((242, 100, 90), dtype=np.int16)
GOLD = np.array((245, 184, 46), dtype=np.int16)
DARK = np.array((37, 52, 70), dtype=np.int16)
LIGHT = np.array((245, 244, 238), dtype=np.int16)


def _mask(array: np.ndarray, color: np.ndarray, tolerance: float = 18.0) -> np.ndarray:
    delta = array.astype(np.int32) - color.astype(np.int32)
    return np.sum(delta * delta, axis=2, dtype=np.int32) <= tolerance * tolerance


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for sy, sx in np.argwhere(mask):
        if seen[sy, sx]:
            continue
        queue = deque([(int(sy), int(sx))])
        seen[sy, sx] = True
        xs, ys = [], []
        while queue:
            y, x = queue.popleft()
            xs.append(x); ys.append(y)
            for ny, nx in ((y-1,x),(y+1,x),(y,x-1),(y,x+1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny,nx] and not seen[ny,nx]:
                    seen[ny,nx] = True
                    queue.append((ny,nx))
        output.append((len(xs), min(xs), min(ys), max(xs), max(ys)))
    return output


def _solid_tiles(array: np.ndarray, color: np.ndarray, label: str) -> list[tuple[float, float, bool]]:
    items = []
    for area, x0, y0, x1, y1 in _components(_mask(array, color)):
        if area < 20:
            continue
        width, height = x1 - x0 + 1, y1 - y0 + 1
        if not (22 <= width <= 26 and 22 <= height <= 26 and 500 <= area <= 650):
            raise ValueError(f"damaged {label} tile")
        items.append(((x0+x1)/2, (y0+y1)/2, label == "cyan"))
    return items


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 64 or array.shape[1] < 64:
        raise ValueError("invalid raster")
    signals = _solid_tiles(array, CYAN, "cyan") + _solid_tiles(array, CORAL, "coral")
    if len(signals) != 5:
        raise ValueError("expected exactly five signal tiles")
    golds = [item for item in _components(_mask(array, GOLD)) if item[0] >= 20]
    if len(golds) != 1:
        raise ValueError("expected one START frame")
    area, x0, y0, x1, y1 = golds[0]
    if not (350 <= area <= 450 and 22 <= x1-x0+1 <= 26 and 22 <= y1-y0+1 <= 26):
        raise ValueError("damaged START frame")
    start = ((x0+x1)/2, (y0+y1)/2)
    all_centers = [(start[0], start[1], None)] + signals
    ordered = sorted(all_centers, key=lambda item: (round(item[1] / 20), item[0]))
    if ordered[0][2] is not None:
        raise ValueError("START must be first in row order")
    xs = sorted({round(item[0]) for item in all_centers})
    ys = sorted({round(item[1]) for item in all_centers})
    if len(xs) != 3 or len(ys) != 2:
        raise ValueError("plaque grid is incomplete")
    if not all(26 <= b-a <= 30 for a,b in zip(xs,xs[1:])) or not (26 <= ys[1]-ys[0] <= 30):
        raise ValueError("plaque spacing is ambiguous")
    # Classify the START center against the two declared fills with a runner-up gap.
    patch = array[int(start[1])-4:int(start[1])+5, int(start[0])-4:int(start[0])+5]
    mean = patch.astype(np.float64).mean(axis=(0,1))
    d_dark = float(np.linalg.norm(mean - DARK))
    d_light = float(np.linalg.norm(mean - LIGHT))
    if abs(d_dark - d_light) < 80:
        raise ValueError("ambiguous START fill")
    output = d_dark < d_light
    previous = None
    for _, _, request in ordered[1:]:
        if previous is not None and request == previous:
            output = bool(request)
        previous = request
    return "yes" if output else "no"
