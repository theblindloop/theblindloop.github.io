"""Deterministic renderer for checkpoint span comparison."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 768, 512
SCALE = 3
DEFAULT_PALETTE = {
    "background": [248, 246, 238],
    "route": [29, 49, 73],
    "cyan": [0, 172, 193],
    "orange": [232, 118, 35],
    "start": [42, 157, 89],
    "finish": [205, 62, 62],
}


def _lengths(points):
    return [math.hypot(b[0] - a[0], b[1] - a[1]) for a, b in zip(points, points[1:])]


def _total(points):
    return sum(_lengths(points))


def _point_at(points, distance):
    remaining = float(distance)
    for a, b, length in zip(points, points[1:], _lengths(points)):
        if remaining <= length:
            t = remaining / length if length else 0.0
            return (a[0] + t * (b[0] - a[0]), a[1] + t * (b[1] - a[1]))
        remaining -= length
    return tuple(points[-1])


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    offset = [rng.randint(-5, 5), rng.randint(-4, 4)]
    # A single diagonal folded trace spans the image without filling its box.
    points = []
    for index in range(9):
        x = 68 + index * 79 + rng.randint(-7, 7)
        base_y = 58 + index * 49
        wiggle = 0 if index in (0, 8) else (34 if index % 2 else -34)
        y = base_y + wiggle + rng.randint(-7, 7)
        points.append([x, y])

    route_total = _total(points)
    # The wider interval is alternated deterministically, while exact placements vary.
    wide_a = 0.08 + rng.uniform(0.00, 0.035)
    wide_b = 0.88 + rng.uniform(-0.025, 0.025)
    narrow_a = 0.29 + rng.uniform(-0.025, 0.025)
    narrow_b = 0.61 + rng.uniform(-0.025, 0.025)
    wide = sorted([wide_a * route_total, wide_b * route_total])
    narrow = sorted([narrow_a * route_total, narrow_b * route_total])
    if int(seed) % 2 == 0:
        distances = wide + narrow
        colors = ["cyan", "cyan", "orange", "orange"]
    else:
        distances = narrow + wide
        colors = ["cyan", "cyan", "orange", "orange"]
    return {
        "points": points,
        "checkpoint_distances": [round(v, 6) for v in distances],
        "checkpoint_colors": colors,
        "palette": deepcopy(DEFAULT_PALETTE),
        "offset": offset,
    }


def _shifted_points(scene):
    dx, dy = scene["offset"]
    return [[p[0] + dx, p[1] + dy] for p in scene["points"]]


def render(scene: dict) -> Image.Image:
    palette = scene["palette"]
    points = _shifted_points(scene)
    distances = scene["checkpoint_distances"]
    colors = scene["checkpoint_colors"]
    bg = tuple(palette["background"])
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), bg)
    draw = ImageDraw.Draw(image)

    scaled = [(round(x * SCALE), round(y * SCALE)) for x, y in points]
    draw.line(scaled, fill=tuple(palette["route"]), width=8 * SCALE, joint="curve")

    def ring(center, color, radius=13, width=5):
        x, y = center
        box = [
            round((x - radius) * SCALE), round((y - radius) * SCALE),
            round((x + radius) * SCALE), round((y + radius) * SCALE),
        ]
        draw.ellipse(box, outline=tuple(color), width=width * SCALE)

    for distance, color in zip(distances, colors):
        ring(_point_at(points, distance), palette[color])
    ring(points[0], palette["start"], radius=15, width=6)
    ring(points[-1], palette["finish"], radius=15, width=6)
    # Restore an uninterrupted, thin centerline through every hollow ring.
    draw.line(scaled, fill=tuple(palette["route"]), width=5 * SCALE, joint="curve")

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    points = scene["points"]
    distances = scene["checkpoint_distances"]
    colors = scene["checkpoint_colors"]
    # Touch points so the analytic label explicitly depends on the rendered route geometry.
    if _total(points) <= max(distances):
        raise ValueError("checkpoint lies beyond route")
    grouped = {"cyan": [], "orange": []}
    for distance, color in zip(distances, colors):
        grouped[color].append(float(distance))
    spans = {color: abs(values[1] - values[0]) for color, values in grouped.items()}
    return "cyan" if spans["cyan"] > spans["orange"] else "orange"


def margin(scene: dict) -> float:
    distances = scene["checkpoint_distances"]
    colors = scene["checkpoint_colors"]
    grouped = {"cyan": [], "orange": []}
    for distance, color in zip(distances, colors):
        grouped[color].append(float(distance))
    return abs(abs(grouped["cyan"][1] - grouped["cyan"][0]) - abs(grouped["orange"][1] - grouped["orange"][0]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 30.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    # Move latent points while compensating in the separately rendered offset.
    twin["points"] = [[p[0] + 1, p[1] - 1] for p in scene["points"]]
    twin["offset"] = [scene["offset"][0] - 1, scene["offset"][1] + 1]
    return [("compensated_coordinate_translation", twin)]
