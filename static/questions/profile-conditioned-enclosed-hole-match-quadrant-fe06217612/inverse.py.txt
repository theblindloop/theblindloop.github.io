"""Independent pixel-only inverse arm for enclosed_hole_match_quadrant.

Receives only a final PNG (and the public question contract). It thresholds the
dark glyph ink, flood-fills the white background from the image border, and
gathers the bright pixels that are NOT connected to that background: these are
the enclosed holes of the glyphs. It counts the reference glyph's enclosed holes
in the top strip and every quadrant glyph's enclosed holes by where each hole's
centroid falls, then reports the single quadrant whose hole count equals the
reference count, or "NONE" when none or several quadrants match.

It imports nothing from the renderer, prompts, generator, or verifier, and it
accepts no latent scene data.
"""

from __future__ import annotations

import numpy as np

REF_BOTTOM = 135
MID_X = 256
MID_Y = 320
DARK_THRESH = 128
WHITE_THRESH = 128
MIN_COMP = 30

QUAD_NAMES = {"tl": "top-left", "tr": "top-right", "bl": "bottom-left", "br": "bottom-right"}


def _flood_background(white: np.ndarray) -> np.ndarray:
    h, w = white.shape
    bg = np.zeros_like(white)
    bg[0, :] = white[0, :]
    bg[h - 1, :] = white[h - 1, :]
    bg[:, 0] = white[:, 0]
    bg[:, w - 1] = white[:, w - 1]
    bg &= white
    while bg.any():
        grown = (
            np.roll(bg, 1, axis=0)
            | np.roll(bg, -1, axis=0)
            | np.roll(bg, 1, axis=1)
            | np.roll(bg, -1, axis=1)
        )
        grown &= white
        new = grown & ~bg
        bg |= new
        if not new.any():
            break
    return bg


def _components(mask: np.ndarray) -> list[tuple[np.ndarray, np.ndarray]]:
    ys, xs = np.nonzero(mask)
    n = len(xs)
    if n == 0:
        return []
    parent = list(range(n))

    def find(a: int) -> int:
        while parent[a] != a:
            parent[a] = parent[parent[a]]
            a = parent[a]
        return a

    idx = {}
    for i in range(n):
        idx[(int(ys[i]), int(xs[i]))] = i
    for i in range(n):
        y = int(ys[i])
        x = int(xs[i])
        for dy in (-1, 0, 1):
            for dx in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                j = idx.get((y + dy, x + dx))
                if j is not None:
                    ra, rb = find(i), find(j)
                    if ra != rb:
                        parent[ra] = rb
    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(i)
    return [(ys[list(ids)], xs[list(ids)]) for ids in groups.values()]


def decision_from_image(image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    lum = arr.mean(axis=2)
    white = lum > WHITE_THRESH
    bg = _flood_background(white)
    enclosed = white & ~bg

    ref = 0
    counts = {"tl": 0, "tr": 0, "bl": 0, "br": 0}
    for ys, xs in _components(enclosed):
        if len(xs) < MIN_COMP:
            continue
        cx = float(xs.mean())
        cy = float(ys.mean())
        if cy < REF_BOTTOM:
            ref += 1
            continue
        if cy < MID_Y:
            region = "tl" if cx < MID_X else "tr"
        else:
            region = "bl" if cx < MID_X else "br"
        counts[region] += 1

    matches = [r for r in counts if counts[r] == ref]
    if len(matches) == 1:
        return QUAD_NAMES[matches[0]]
    return "NONE"
