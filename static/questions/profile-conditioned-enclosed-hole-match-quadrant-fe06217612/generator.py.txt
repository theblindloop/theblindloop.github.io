"""Enclosed-hole reference-matching search world.

A dark reference glyph sits in a box at the top of the image. A glyph is a filled
dark disk that contains some number of bright "holes": small background-coloured
disks punched fully inside the dark ink, so each hole is a bright region that is
completely enclosed by surrounding dark ink (not connected to the outside).

Below, the image is split into four quadrants (top-left, top-right, bottom-left,
bottom-right), and each quadrant holds one such dark glyph with some number of
enclosed holes. Exactly one quadrant's glyph has the SAME number of enclosed
holes as the reference glyph; the other three each carry a different hole count.

The question asks which quadrant contains the glyph whose enclosed-hole count
matches the reference glyph. The analytic decision counts the enclosed holes of
the reference (0..3) and of every quadrant glyph, then reports the single
quadrant whose count equals the reference count.

Every glyph is the same size, colour and ink weight, and every glyph's hole
count is a small integer, so the answer cannot be recovered from size, colour,
total ink, position, or a single-feature salience cue: only the count of each
glyph's enclosed bright regions, compared with the reference, determines the
decision. Rendering is fully deterministic given the scene.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image, ImageDraw

SIZE = 512
BG = (255, 255, 255)
INK = (30, 30, 30)
BOX = (205, 205, 205)

REF_BOX = (140, 8, 372, 120)
REF_CENTER = (256.0, 64.0)
REF_R = 34.0
REF_HOLE_R = 8.0

MID_X = 256
MID_Y = 320
MAIN_TOP = 130

QUAD_CENTERS = {
    "tl": (128.0, 215.0),
    "tr": (384.0, 215.0),
    "bl": (128.0, 415.0),
    "br": (384.0, 415.0),
}
GLYPH_R = 48.0
GLYPH_HOLE_R = 11.0

QUADRANTS = ("tl", "tr", "bl", "br")
QUAD_NAMES = {"tl": "top-left", "tr": "top-right", "bl": "bottom-left", "br": "bottom-right"}

QUARANTINE_THRESH = 1.0


def _place_holes(cx: float, cy: float, radius: float, k: int, hr: float) -> list[list[float]]:
    if k <= 0:
        return []
    holes = []
    for i in range(k):
        rh = 0.42 * radius if k > 1 else 0.0
        ang = 2.0 * math.pi * i / k
        hx = cx + rh * math.cos(ang)
        hy = cy + rh * math.sin(ang)
        holes.append([float(hx), float(hy), hr])
    return holes


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(seed)
    ref_count = int(rng.integers(0, 4))
    winner = QUADRANTS[int(rng.integers(0, len(QUADRANTS)))]
    remaining = [v for v in (0, 1, 2, 3) if v != ref_count]
    rng.shuffle(remaining)
    others = [q for q in QUADRANTS if q != winner]
    counts = {q: None for q in QUADRANTS}
    for q, c in zip(others, remaining):
        counts[q] = c
    counts[winner] = ref_count

    quads = {}
    for q in QUADRANTS:
        cx, cy = QUAD_CENTERS[q]
        quads[q] = {
            "cx": cx,
            "cy": cy,
            "r": GLYPH_R,
            "holes": _place_holes(cx, cy, GLYPH_R, counts[q], GLYPH_HOLE_R),
        }
    ref_cx, ref_cy = REF_CENTER
    ref = {
        "cx": ref_cx,
        "cy": ref_cy,
        "r": REF_R,
        "holes": _place_holes(ref_cx, ref_cy, REF_R, ref_count, REF_HOLE_R),
    }
    return {"ref": ref, "quads": quads}


def _draw_glyph(draw: ImageDraw.ImageDraw, spec: dict) -> None:
    cx, cy, r = spec["cx"], spec["cy"], spec["r"]
    draw.ellipse([cx - r, cy - r, cx + r, cy + r], fill=INK)
    for hx, hy, hr in spec["holes"]:
        draw.ellipse([hx - hr, hy - hr, hx + hr, hy + hr], fill=BG)


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (SIZE, SIZE), BG)
    draw = ImageDraw.Draw(img)
    x0, y0, x1, y1 = REF_BOX
    draw.rectangle([x0, y0, x1, y1], outline=BOX, width=2)
    draw.line([(MID_X, MAIN_TOP), (MID_X, SIZE)], fill=BOX, width=2)
    draw.line([(0, MID_Y), (SIZE, MID_Y)], fill=BOX, width=2)
    _draw_glyph(draw, scene["ref"])
    for q in QUADRANTS:
        _draw_glyph(draw, scene["quads"][q])
    return img


def _count(scene: dict, key: str) -> int:
    return len(scene["quads"][key]["holes"])


def analytic_gold(scene: dict) -> str:
    ref_count = len(scene["ref"]["holes"])
    for q in QUADRANTS:
        if len(scene["quads"][q]["holes"]) == ref_count:
            return QUAD_NAMES[q]
    return QUAD_NAMES["br"]


def margin(scene: dict) -> float:
    ref_count = len(scene["ref"]["holes"])
    winner = analytic_gold(scene)
    diffs = [
        abs(_count(scene, q) - ref_count) for q in QUADRANTS if QUAD_NAMES[q] != winner
    ]
    return float(min(diffs)) if diffs else 0.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_THRESH


def latent_symmetries(scene: dict) -> list:
    return []
