"""Render a marked room partition whose answer is a dual-graph predicate."""

from __future__ import annotations

import copy
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 400
HEIGHT = 400
BACKGROUND = [248, 247, 241]
BLACK = [28, 30, 34]
CYAN = [0, 151, 181]
ORANGE = [228, 105, 25]
FRAME = [68, 68, 332, 332]
GRID = 4
WALL = 11
DOT_R = 10
MIN_MARGIN = 10


def _neighbors(cell: tuple[int, int]) -> set[tuple[int, int]]:
    row, col = cell
    return {
        (nr, nc)
        for nr, nc in ((row - 1, col), (row + 1, col), (row, col - 1), (row, col + 1))
        if 0 <= nr < GRID and 0 <= nc < GRID
    }


def _has_common(pair: list[list[int]]) -> bool:
    a, b = (tuple(pair[0]), tuple(pair[1]))
    return bool(_neighbors(a) & _neighbors(b))


def _transform_cell(cell: tuple[int, int], transform: int) -> tuple[int, int]:
    row, col = cell
    if transform >= 4:
        col = GRID - 1 - col
    for _ in range(transform % 4):
        row, col = col, GRID - 1 - row
    return row, col


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    cells = [(r, c) for r in range(GRID) for c in range(GRID)]
    winner_pairs = []
    loser_pairs = []
    for i, a in enumerate(cells):
        for b in cells[i + 1:]:
            pair = [list(a), list(b)]
            (winner_pairs if _has_common(pair) else loser_pairs).append((a, b))
    # Select disjoint pairs with all markers well separated from one another.
    choices = []
    for win in winner_pairs:
        for lose in loser_pairs:
            if len(set(win + lose)) < 4:
                continue
            # Ensure the combined evidence spans a regional, not canvas-wide, support.
            if max(abs(a[0]-b[0]) + abs(a[1]-b[1]) for a in win for b in lose) < 2:
                continue
            choices.append((win, lose))
    win, lose = rng.choice(choices)
    cyan_wins = int(seed) % 2 == 0
    transform = rng.randrange(8)
    cyan = win if cyan_wins else lose
    orange = lose if cyan_wins else win
    cyan_cells = sorted([list(_transform_cell(cell, transform)) for cell in cyan])
    orange_cells = sorted([list(_transform_cell(cell, transform)) for cell in orange])
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
        "wall_color": list(BLACK),
        "cyan": list(CYAN),
        "orange": list(ORANGE),
        "frame": list(FRAME),
        "rows": GRID,
        "cols": GRID,
        "cyan_cells": cyan_cells,
        "orange_cells": orange_cells,
        "transform": transform,
    }


def _bounds(scene: dict[str, Any]) -> tuple[list[int], list[int]]:
    x0, y0, x1, y1 = scene["frame"]
    xs = [round(x0 + (x1 - x0) * i / int(scene["cols"])) for i in range(int(scene["cols"]) + 1)]
    ys = [round(y0 + (y1 - y0) * i / int(scene["rows"])) for i in range(int(scene["rows"]) + 1)]
    return xs, ys


def _center(scene: dict[str, Any], cell: list[int]) -> tuple[int, int]:
    xs, ys = _bounds(scene)
    row, col = cell
    return (xs[col] + xs[col + 1]) // 2, (ys[row] + ys[row + 1]) // 2


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    xs, ys = _bounds(scene)
    ink = tuple(scene["wall_color"])
    half = WALL // 2
    for x in xs:
        draw.rectangle([x - half, ys[0] - half, x + half, ys[-1] + half], fill=ink)
    for y in ys:
        draw.rectangle([xs[0] - half, y - half, xs[-1] + half, y + half], fill=ink)
    for cell in scene["cyan_cells"]:
        x, y = _center(scene, cell)
        draw.ellipse([x - DOT_R, y - DOT_R, x + DOT_R, y + DOT_R], fill=tuple(scene["cyan"]))
    for cell in scene["orange_cells"]:
        x, y = _center(scene, cell)
        draw.ellipse([x - DOT_R, y - DOT_R, x + DOT_R, y + DOT_R], fill=tuple(scene["orange"]))
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    outcomes = {"cyan": _has_common(scene["cyan_cells"]), "orange": _has_common(scene["orange_cells"])}
    winners = [color for color, value in outcomes.items() if value]
    if len(winners) != 1:
        raise ValueError("exactly one marker pair must have a common neighbor")
    return winners[0]


def margin(scene: dict[str, Any]) -> float:
    return float(MIN_MARGIN)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < MIN_MARGIN


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    twin = copy.deepcopy(scene)
    twin["cyan_cells"] = list(reversed(twin["cyan_cells"]))
    twin["orange_cells"] = list(reversed(twin["orange_cells"]))
    return [("reverse_pair_storage", twin)]
