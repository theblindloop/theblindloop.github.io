"""Independent pixel-only inverse arm for Whole-Trace Regression Trend.

The oracle accepts only a final RGB image. It segments the sole saturated
component, checks that this thin component is intact and spans the chart, takes
the median colored-pixel height in each occupied x-column, and independently
fits a least-squares line. No renderer state, source data, manifest, coordinate
constant, filename, metadata, or analytic answer is available here.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


def _trace_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    maximum = array.max(axis=2)
    minimum = array.min(axis=2)
    chroma = maximum - minimum
    return (chroma >= 62) & (maximum >= 105) & (minimum <= 150)


def _component_size(mask: np.ndarray, start: tuple[int, int]) -> int:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    stack = [start]
    seen[start] = True
    count = 0
    while stack:
        y, x = stack.pop()
        count += 1
        for ny, nx in (
            (y - 1, x),
            (y + 1, x),
            (y, x - 1),
            (y, x + 1),
            (y - 1, x - 1),
            (y - 1, x + 1),
            (y + 1, x - 1),
            (y + 1, x + 1),
        ):
            if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                seen[ny, nx] = True
                stack.append((ny, nx))
    return count


def _column_centers(image: Image.Image) -> np.ndarray:
    mask = _trace_mask(image)
    points = np.argwhere(mask)
    if len(points) < image.width * 4:
        raise ValueError("colored trace is missing or materially erased")
    first = tuple(int(value) for value in points[0])
    if _component_size(mask, first) != int(mask.sum()):
        raise ValueError("colored trace is severed or has extra components")

    ys, xs = np.nonzero(mask)
    x0, x1 = int(xs.min()), int(xs.max())
    span = x1 - x0 + 1
    if span < image.width * 0.88:
        raise ValueError("colored trace does not span the plot")
    if int(ys.max() - ys.min() + 1) < image.height * 0.19:
        raise ValueError("colored trace has implausibly little vertical variation")

    centers = []
    for x in range(x0, x1 + 1):
        column = np.flatnonzero(mask[:, x])
        if not len(column):
            raise ValueError("colored trace has a missing interior column")
        if len(column) > 28:
            raise ValueError("colored component is not a thin trace")
        centers.append(float(np.median(column)))
    values = np.asarray(centers, dtype=np.float64)
    if len(values) < image.width * 0.88:
        raise ValueError("too few trace columns remain")
    return values


def decision_from_image(image: Image.Image) -> str:
    values = _column_centers(image)
    x = np.arange(len(values), dtype=np.float64)
    x -= x.mean()
    y = values - values.mean()
    denominator = float(np.dot(x, x))
    if denominator <= 0.0:
        raise ValueError("pixel regression is undefined")
    slope = float(np.dot(x, y) / denominator)
    endpoint_equivalent_change = abs(slope) * (len(values) - 1)
    if endpoint_equivalent_change <= 5.0:
        raise ValueError("pixel-measured trend is too close to a tie")
    return "rise" if slope < 0.0 else "fall"
