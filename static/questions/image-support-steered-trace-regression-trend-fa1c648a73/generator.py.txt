"""Deterministic renderer and analytic arm for Whole-Trace Regression Trend."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 800
HEIGHT = 520
QUARANTINE_PX = 12.0
TRACE_PALETTE = (
    (13, 132, 148),
    (43, 116, 181),
    (126, 76, 176),
    (205, 92, 45),
)
PAPER_PALETTE = (
    (249, 248, 244),
    (247, 249, 248),
    (250, 247, 245),
)
INK = (42, 47, 52)
FRAME = (143, 149, 151)
GRID = (219, 220, 216)


def _font(size: int, *, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _trend_slope(scene: dict[str, Any]) -> float:
    values = [float(value) for value in scene["trace_y"]]
    if len(values) < 32:
        raise ValueError("trace has too few column samples")
    center_x = (len(values) - 1) / 2.0
    mean_y = sum(values) / len(values)
    numerator = sum((index - center_x) * (value - mean_y) for index, value in enumerate(values))
    denominator = sum((index - center_x) ** 2 for index in range(len(values)))
    if denominator <= 0:
        raise ValueError("trace regression is undefined")
    return numerator / denominator


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample cue-balanced reflection pairs with equal-height endpoints.

    Seeds 2k and 2k+1 share every item of chart furniture and the same
    unsigned waveform. The odd member reflects all trace heights about the
    chart midpoint, exchanging the regression sign without changing endpoint
    height, color, placement, count, span, or total colored ink.
    """

    seed = int(seed)
    group = seed // 2
    answer_index = seed % 2
    rng = random.Random(0x51_0FE_2026 + group * 1_000_003)

    chart_left = 27 + (group % 5 - 2) * 3
    chart_top = 130 + (group % 3 - 1) * 5
    chart_width = 746
    chart_height = 300
    trace_count = chart_width - 15
    middle = chart_top + chart_height // 2

    primary = 58 + rng.randint(0, 12)
    harmonic = rng.randint(-13, 13)
    bowl = rng.randint(-18, 18)
    ripple = rng.randint(-10, 10)
    base: list[int] = []
    for index in range(trace_count):
        t = index / (trace_count - 1)
        value = (
            primary * math.sin(2.0 * math.pi * t)
            + harmonic * math.sin(4.0 * math.pi * t)
            + bowl * (math.cos(2.0 * math.pi * t) - 1.0)
            + ripple * (math.cos(4.0 * math.pi * t) - 1.0)
        )
        base.append(int(round(middle + value)))
    base[0] = middle
    base[-1] = middle
    trace_y = base if answer_index == 0 else [2 * middle - value for value in base]

    scene = {
        "width": WIDTH,
        "height": HEIGHT,
        "chart_left": chart_left,
        "chart_top": chart_top,
        "chart_width": chart_width,
        "chart_height": chart_height,
        "trace_y": trace_y,
        "trace_rgb": list(TRACE_PALETTE[group % len(TRACE_PALETTE)]),
        "paper_rgb": list(PAPER_PALETTE[group % len(PAPER_PALETTE)]),
    }
    if margin(scene) <= QUARANTINE_PX:
        raise AssertionError("sampler produced an ambiguous trend")
    return scene


def analytic_gold(scene: dict[str, Any]) -> str:
    slope = _trend_slope(scene)
    if slope == 0.0:
        raise ValueError("least-squares trend is tied")
    # Raster y increases downward, so a negative y-slope visibly rises.
    return "rise" if slope < 0.0 else "fall"


def margin(scene: dict[str, Any]) -> float:
    values = scene["trace_y"]
    return float(abs(_trend_slope(scene)) * (len(values) - 1))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) <= QUARANTINE_PX


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []


def _validate_scene(scene: dict[str, Any]) -> tuple[int, int, int, int, list[int]]:
    width = int(scene["width"])
    height = int(scene["height"])
    left = int(scene["chart_left"])
    top = int(scene["chart_top"])
    chart_width = int(scene["chart_width"])
    chart_height = int(scene["chart_height"])
    values = [int(value) for value in scene["trace_y"]]
    trace = tuple(int(value) for value in scene["trace_rgb"])
    paper = tuple(int(value) for value in scene["paper_rgb"])
    if (width, height) != (WIDTH, HEIGHT):
        raise ValueError("unexpected canvas size")
    if not (18 <= left <= 42 and 120 <= top <= 145):
        raise ValueError("chart translation is outside its visible range")
    if chart_width != 746 or chart_height != 300 or len(values) != chart_width - 15:
        raise ValueError("unexpected chart or trace dimensions")
    if left + chart_width >= width - 15 or top + chart_height >= height - 60:
        raise ValueError("chart does not fit the canvas")
    if min(values) < top + 18 or max(values) > top + chart_height - 18:
        raise ValueError("trace leaves the chart interior")
    if values[0] != values[-1]:
        raise ValueError("trace endpoints must share a visible height")
    if any(not 0 <= channel <= 255 for color in (trace, paper) for channel in color):
        raise ValueError("RGB channel is outside range")
    return left, top, chart_width, chart_height, values


def render(scene: dict[str, Any]) -> Image.Image:
    left, top, chart_width, chart_height, values = _validate_scene(scene)
    width = int(scene["width"])
    height = int(scene["height"])
    trace = tuple(int(value) for value in scene["trace_rgb"])
    paper = tuple(int(value) for value in scene["paper_rgb"])
    right = left + chart_width
    bottom = top + chart_height
    start_x = left + 8

    image = Image.new("RGB", (width, height), paper)
    draw = ImageDraw.Draw(image)
    draw.text((left, 25), "WHOLE-TRACE TREND", fill=INK, font=_font(25, bold=True))
    draw.text(
        (left, 62),
        "Fit all x-columns equally  •  endpoints may be level",
        fill=INK,
        font=_font(16),
    )
    draw.rectangle((left, top, right, bottom), outline=FRAME, width=2)
    for fraction in (0.25, 0.5, 0.75):
        x = int(round(left + fraction * chart_width))
        y = int(round(top + fraction * chart_height))
        draw.line((x, top + 2, x, bottom - 2), fill=GRID, width=1)
        draw.line((left + 2, y, right - 2, y), fill=GRID, width=1)
    draw.line((left + 10, bottom - 14, right - 10, bottom - 14), fill=FRAME, width=2)
    draw.polygon(
        ((right - 10, bottom - 14), (right - 20, bottom - 20), (right - 20, bottom - 8)),
        fill=FRAME,
    )
    draw.text((right - 8, bottom + 12), "x", fill=INK, font=_font(15), anchor="ra")
    points = [(start_x + index, value) for index, value in enumerate(values)]
    draw.line(points, fill=trace, width=7, joint="curve")
    draw.ellipse((start_x - 4, values[0] - 4, start_x + 4, values[0] + 4), fill=trace)
    end_x = start_x + len(values) - 1
    draw.ellipse((end_x - 4, values[-1] - 4, end_x + 4, values[-1] + 4), fill=trace)
    draw.text(
        (left, bottom + 25),
        "RISE means the fitted line goes upward toward the right.",
        fill=INK,
        font=_font(15),
    )
    return image
