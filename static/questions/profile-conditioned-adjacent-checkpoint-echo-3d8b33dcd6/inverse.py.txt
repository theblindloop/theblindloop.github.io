"""Independent pixel-only inverse arm.

It segments visible colors, reconstructs the connected dark cable, orders each
checkpoint by graph distance from the green start, and tests adjacent equality.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int = 20) -> list[np.ndarray]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        q = deque([(int(y), int(x))])
        seen[y, x] = True
        pts = []
        while q:
            cy, cx = q.popleft()
            pts.append((cy, cx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = cy + dy, cx + dx
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    q.append((ny, nx))
        if len(pts) >= minimum:
            found.append(np.asarray(pts, dtype=np.int32))
    return found


def _centers(mask: np.ndarray) -> list[tuple[float, float]]:
    return [(float(c[:, 0].mean()), float(c[:, 1].mean())) for c in _components(mask)]


def _nearest(mask: np.ndarray, center: tuple[float, float], radius: int = 24) -> tuple[int, int]:
    cy, cx = center
    y0, y1 = max(0, int(cy) - radius), min(mask.shape[0], int(cy) + radius + 1)
    x0, x1 = max(0, int(cx) - radius), min(mask.shape[1], int(cx) + radius + 1)
    ys, xs = np.nonzero(mask[y0:y1, x0:x1])
    if not len(ys):
        raise ValueError("marker is not attached to the dark cable")
    ys, xs = ys + y0, xs + x0
    k = np.argmin((ys - cy) ** 2 + (xs - cx) ** 2)
    return int(ys[k]), int(xs[k])


def _distances(mask: np.ndarray, start: tuple[int, int]) -> np.ndarray:
    h, w = mask.shape
    dist = np.full((h, w), -1, dtype=np.int32)
    dist[start] = 0
    q = deque([start])
    while q:
        y, x = q.popleft()
        nd = int(dist[y, x]) + 1
        for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            ny, nx = y + dy, x + dx
            if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and dist[ny, nx] < 0:
                dist[ny, nx] = nd
                q.append((ny, nx))
    return dist


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    r, g, b = arr[..., 0], arr[..., 1], arr[..., 2]
    dark = (r < 105) & (g < 105) & (b < 115)
    green = (g > 120) & (g > r * 1.35) & (g > b * 1.25)
    orange = (r > 170) & (g > 85) & (g < 180) & (b < 90) & (r > g * 1.25)
    purple = (b > 135) & (r > 85) & (g < 125) & (b > g * 1.25)

    starts = _centers(green)
    orange_centers = _centers(orange)
    purple_centers = _centers(purple)
    if len(starts) != 1 or len(orange_centers) != 3 or len(purple_centers) != 3:
        raise ValueError("expected one green start and six checkpoint rings (three per color)")
    start = _nearest(dark, starts[0], 20)
    dist = _distances(dark, start)
    ordered = []
    for color, centers in (("orange", orange_centers), ("purple", purple_centers)):
        for center in centers:
            y, x = _nearest(dark, center)
            d = int(dist[y, x])
            if d < 0:
                raise ValueError("checkpoint is disconnected from the green-started cable")
            ordered.append((d, color))
    ordered.sort()
    gaps = [b[0] - a[0] for a, b in zip(ordered, ordered[1:])]
    if not gaps or min(gaps) < 24:
        raise ValueError("checkpoint order is pixel-ambiguous")
    colors = [color for _, color in ordered]
    return "yes" if any(a == b for a, b in zip(colors, colors[1:])) else "no"
