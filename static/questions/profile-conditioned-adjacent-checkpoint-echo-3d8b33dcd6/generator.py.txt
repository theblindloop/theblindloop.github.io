"""Renderer and analytic arm for the adjacent-checkpoint echo world."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw


PALETTES = (
    {"background": (248, 247, 242), "dark": (35, 40, 48), "green": (35, 174, 92),
     "red": (214, 54, 64), "orange": (238, 133, 35), "purple": (139, 77, 199)},
    {"background": (252, 250, 246), "dark": (28, 36, 45), "green": (28, 160, 88),
     "red": (224, 61, 68), "orange": (230, 118, 26), "purple": (151, 73, 207)},
    {"background": (246, 248, 245), "dark": (42, 43, 51), "green": (40, 181, 96),
     "red": (205, 48, 60), "orange": (242, 145, 39), "purple": (130, 69, 190)},
)


def _base_points(rng: random.Random) -> list[list[int]]:
    # A non-self-crossing cable that repeatedly reverses horizontal direction.
    j = lambda n: rng.randint(-n, n)
    pts = [
        [58 + j(4), 72 + j(9)], [183 + j(7), 72 + j(9)],
        [183 + j(7), 154 + j(8)], [92 + j(6), 154 + j(8)],
        [92 + j(6), 270 + j(8)], [286 + j(8), 270 + j(8)],
        [286 + j(8), 118 + j(8)], [408 + j(7), 118 + j(8)],
        [408 + j(7), 304 + j(7)], [463 + j(4), 304 + j(7)],
    ]
    # Restore exact orthogonality after independent jitter.
    for i in range(1, len(pts)):
        if i % 2:
            pts[i][1] = pts[i - 1][1]
        else:
            pts[i][0] = pts[i - 1][0]
    return pts


def _transform_points(points: list[list[int]], mode: int) -> list[list[int]]:
    if mode == 0:
        return points
    if mode == 1:  # vertical reflection
        return [[x, 383 - y] for x, y in points]
    if mode == 2:  # 180-degree rotation: green start moves to the right
        return [[511 - x, 383 - y] for x, y in points]
    return [[511 - x, y] for x, y in points]  # horizontal reflection


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    points = _transform_points(_base_points(rng), seed % 4)
    eligible = list(range(9))
    # Six ordered checkpoints, distributed across the route with equal total ink.
    checkpoint_segments = sorted(rng.sample(eligible, 6))
    if seed % 2:
        # YES: exactly three of each color, with at least one adjacent echo.
        colors = ["orange"] * 3 + ["purple"] * 3
        while True:
            rng.shuffle(colors)
            if any(a == b for a, b in zip(colors, colors[1:])):
                break
    else:
        # NO: strict alternation, also exactly three of each color.
        first = rng.choice(["orange", "purple"])
        second = "purple" if first == "orange" else "orange"
        colors = [first if i % 2 == 0 else second for i in range(6)]
    return {
        "width": 512,
        "height": 384,
        "points": points,
        "checkpoint_segments": checkpoint_segments,
        "checkpoint_colors": colors,
        "palette": deepcopy(PALETTES[rng.randrange(len(PALETTES))]),
    }


def _midpoint(a: list[int], b: list[int]) -> tuple[float, float]:
    return ((a[0] + b[0]) / 2.0, (a[1] + b[1]) / 2.0)


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    points = [tuple(p) for p in scene["points"]]
    segments = [int(i) for i in scene["checkpoint_segments"]]
    colors = list(scene["checkpoint_colors"])
    palette = {key: tuple(value) for key, value in scene["palette"].items()}
    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), palette["background"])
    draw = ImageDraw.Draw(image)
    spoints = [(x * scale, y * scale) for x, y in points]

    def disk(cx: float, cy: float, radius: int, fill: tuple[int, int, int]) -> None:
        draw.ellipse(((cx - radius) * scale, (cy - radius) * scale,
                      (cx + radius) * scale, (cy + radius) * scale), fill=fill)

    # End markers are painted first; the cable is then drawn into their centers.
    sx, sy = points[0]
    disk(sx, sy, 17, palette["green"])
    disk(sx, sy, 10, palette["background"])

    # End: red square with a dark cable center.
    ex, ey = points[-1]
    draw.rectangle(((ex - 16) * scale, (ey - 16) * scale,
                    (ex + 16) * scale, (ey + 16) * scale), fill=palette["red"])
    draw.rectangle(((ex - 9) * scale, (ey - 9) * scale,
                    (ex + 9) * scale, (ey + 9) * scale), fill=palette["background"])

    draw.line(spoints, fill=palette["dark"], width=9 * scale, joint="curve")

    # Each checkpoint is a colored ring tethered just beside the cable, leaving
    # the centerline itself uninterrupted. Alternating tether side avoids overlap.
    for order, (segment, color) in enumerate(zip(segments, colors)):
        cx, cy = _midpoint(scene["points"][segment], scene["points"][segment + 1])
        a, b = points[segment], points[segment + 1]
        dx, dy = b[0] - a[0], b[1] - a[1]
        length = max(1.0, math.hypot(dx, dy))
        side = -1 if order % 2 else 1
        nx, ny = side * (-dy / length), side * (dx / length)
        mx, my = cx + nx * 18, cy + ny * 18
        draw.line(((cx * scale, cy * scale), (mx * scale, my * scale)),
                  fill=palette["dark"], width=3 * scale)
        disk(mx, my, 10, palette[color])
        disk(mx, my, 4, palette["background"])

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    colors = list(scene["checkpoint_colors"])
    return "yes" if any(a == b for a, b in zip(colors, colors[1:])) else "no"


def _checkpoint_positions(scene: dict) -> list[float]:
    points = scene["points"]
    cumulative = [0.0]
    for a, b in zip(points, points[1:]):
        cumulative.append(cumulative[-1] + math.dist(a, b))
    return [
        cumulative[i] + 0.5 * math.dist(points[i], points[i + 1])
        for i in scene["checkpoint_segments"]
    ]


def margin(scene: dict) -> float:
    positions = _checkpoint_positions(scene)
    # Runner-up margin: smallest along-cable gap whose ordering would have to
    # collapse before a different checkpoint could become adjacent.
    return round(min(b - a for a, b in zip(positions, positions[1:])), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 36.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
