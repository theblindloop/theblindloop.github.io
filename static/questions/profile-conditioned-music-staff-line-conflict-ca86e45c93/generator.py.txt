"""Deterministic analytic arm and supersampled music-staff renderer."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS = 512
SUPERSAMPLE = 4
QUARANTINE_GAP_PX = 6.0
BACKGROUNDS = (
    (250, 248, 242),
    (246, 249, 252),
    (251, 247, 250),
    (247, 251, 246),
)
STAFF_INKS = (
    (28, 48, 69),
    (65, 39, 68),
    (35, 63, 47),
    (66, 48, 34),
)
NOTE_INKS = (
    (27, 91, 145),
    (143, 48, 61),
    (43, 112, 81),
    (102, 58, 139),
)


def _style(seed: int, tight: bool = False) -> dict[str, Any]:
    rng = random.Random(int(seed))
    background = rng.choice(BACKGROUNDS)
    staff = rng.choice(STAFF_INKS)
    accents = [value for value in NOTE_INKS if value != staff]
    left_x = rng.randint(48, 82)
    right_x = rng.randint(426, 468)
    center_y = rng.randint(238, 274)
    line_gap = rng.randint(31, 43)
    line_width = rng.randint(3, 6)
    if tight:
        line_gap = 18
        line_width = 7
    barline_x = right_x - rng.randint(18, 34)
    positions = sorted(rng.sample(range(left_x + 54, barline_x - 34), rng.randint(4, 7)))
    note_specs = []
    for index, x in enumerate(positions):
        note_specs.append(
            {
                "x": x,
                "step": rng.choice((-3, -2, -1, 0, 1, 2, 3)),
                "radius_x": rng.randint(10, 15),
                "radius_y": rng.randint(7, 11),
                "stem_direction": -1 if (index + rng.randint(0, 1)) % 2 == 0 else 1,
                "stem_length": rng.randint(34, 56),
                "stem_width": rng.randint(3, 5),
            }
        )
    return {
        "width": CANVAS,
        "height": CANVAS,
        "background_rgb": list(background),
        "ink_rgb": list(staff),
        "accent_rgb": list(rng.choice(accents)),
        "left_x": left_x,
        "right_x": right_x,
        "center_y": center_y,
        "line_gap": line_gap,
        "line_width": line_width,
        "note_specs": note_specs,
        "barline_x": barline_x,
        "barline_width": rng.randint(3, 6),
    }


def matched_scene(seed: int, line_count: int, tight: bool = False) -> dict[str, Any]:
    """Return one style with a chosen visible line count for matched pairs."""
    if int(line_count) not in (4, 5, 6):
        raise ValueError("line_count must be 4, 5, or 6")
    scene = _style(int(seed), tight=tight)
    scene["line_count"] = int(line_count)
    return scene


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample deterministic scenes balanced over four, five, and six lines."""
    line_count = (4, 5, 6)[int(seed) % 3]
    return matched_scene(int(seed) * 104729 + 43, line_count, tight=(int(seed) % 19 == 0))


def _line_positions(scene: dict) -> list[float]:
    count = int(scene["line_count"])
    center = float(scene["center_y"])
    gap = float(scene["line_gap"])
    return [center + (index - (count - 1) / 2.0) * gap for index in range(count)]


def render(scene: dict) -> Image.Image:
    """Render staff lines, noteheads and one explicitly ignorable barline."""
    scale = SUPERSAMPLE
    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(int(v) for v in scene["background_rgb"])
    ink = tuple(int(v) for v in scene["ink_rgb"])
    accent = tuple(int(v) for v in scene["accent_rgb"])
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)

    def box(values: tuple[float, float, float, float]) -> tuple[int, int, int, int]:
        return tuple(int(round(value * scale)) for value in values)

    left = float(scene["left_x"])
    right = float(scene["right_x"])
    line_width = float(scene["line_width"])
    line_positions = _line_positions(scene)
    for y in line_positions:
        draw.rounded_rectangle(
            box((left, y - line_width / 2.0, right, y + line_width / 2.0)),
            radius=max(1, int(round(line_width * scale / 2.0))),
            fill=ink,
        )

    # The barline is vertical and visibly much shorter than every staff line.
    bar_x = float(scene["barline_x"])
    bar_width = float(scene["barline_width"])
    draw.rectangle(
        box(
            (
                bar_x - bar_width / 2.0,
                line_positions[0] - line_width / 2.0,
                bar_x + bar_width / 2.0,
                line_positions[-1] + line_width / 2.0,
            )
        ),
        fill=ink,
    )

    gap = float(scene["line_gap"])
    center = float(scene["center_y"])
    for spec in scene["note_specs"]:
        x = float(spec["x"])
        y = center + float(spec["step"]) * gap / 2.0
        rx = float(spec["radius_x"])
        ry = float(spec["radius_y"])
        direction = int(spec["stem_direction"])
        stem_length = float(spec["stem_length"])
        stem_width = float(spec["stem_width"])
        draw.ellipse(box((x - rx, y - ry, x + rx, y + ry)), fill=accent)
        stem_x = x + rx - stem_width / 2.0 if direction < 0 else x - rx + stem_width / 2.0
        stem_end = y - stem_length if direction < 0 else y + stem_length
        draw.rectangle(
            box(
                (
                    stem_x - stem_width / 2.0,
                    min(y, stem_end),
                    stem_x + stem_width / 2.0,
                    max(y, stem_end),
                )
            ),
            fill=accent,
        )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "yes" if int(scene["line_count"]) == 5 else "no"


def margin(scene: dict) -> float:
    """Discrete distance in line count to the nearest opposite decision."""
    count = int(scene["line_count"])
    opposite = (4, 6) if count == 5 else (5,)
    return float(min(abs(count - other) for other in opposite))


def geometric_margin(scene: dict) -> float:
    return float(scene["line_gap"]) - float(scene["line_width"])


def is_quarantined(scene: dict) -> bool:
    length = float(scene["right_x"]) - float(scene["left_x"])
    return geometric_margin(scene) < QUARANTINE_GAP_PX or length < 280.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # No pixel-identical latent aliases are intentionally introduced.
    return []
