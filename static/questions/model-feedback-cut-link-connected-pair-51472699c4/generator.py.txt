"""Deterministic graph sampler and renderer for edge-deletion connectivity."""

from __future__ import annotations

from collections import deque
import copy
import random

from PIL import Image, ImageDraw

WIDTH = 640
HEIGHT = 640
BACKGROUND = (250, 248, 244)
INK = (43, 48, 56)
JOINT = (218, 230, 241)
CUT = (205, 47, 63)
COLORS = {
    "amber": (232, 156, 37),
    "cyan": (31, 163, 190),
    "violet": (132, 79, 188),
}
COLOR_NAMES = ("amber", "cyan", "violet")


def _edge(a: str, b: str) -> list[str]:
    return sorted((a, b))


def _grid_edges() -> list[list[str]]:
    edges: list[list[str]] = []
    for row in range(5):
        for col in range(5):
            here = f"n{row}{col}"
            if col < 4:
                edges.append(_edge(here, f"n{row}{col + 1}"))
            if row < 4:
                edges.append(_edge(here, f"n{row + 1}{col}"))
    return edges


def _components(node_ids: list[str], edges: list[list[str]], omitted: list[str] | None = None) -> list[set[str]]:
    adjacency = {node_id: [] for node_id in node_ids}
    skip = tuple(omitted) if omitted is not None else None
    for a, b in edges:
        if skip is not None and tuple(sorted((a, b))) == skip:
            continue
        adjacency[a].append(b)
        adjacency[b].append(a)
    remaining = set(node_ids)
    parts: list[set[str]] = []
    while remaining:
        start = min(remaining)
        reached = {start}
        queue = deque([start])
        while queue:
            here = queue.popleft()
            for neighbor in adjacency[here]:
                if neighbor not in reached:
                    reached.add(neighbor)
                    queue.append(neighbor)
        parts.append(reached)
        remaining -= reached
    return parts


def sample_scene(seed: int) -> dict:
    """Sample a balanced 5x5 network with one marked bridge and internal cycles."""

    rng = random.Random(int(seed))
    node_ids = [f"n{row}{col}" for row in range(5) for col in range(5)]
    nodes = [
        {"id": f"n{row}{col}", "x": 100 + 110 * col, "y": 100 + 110 * row}
        for row in range(5) for col in range(5)
    ]

    candidates = _grid_edges()
    rng.shuffle(candidates)
    parent = {node_id: node_id for node_id in node_ids}

    def find(x: str) -> str:
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    tree: list[list[str]] = []
    for a, b in candidates:
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[rb] = ra
            tree.append(_edge(a, b))
        if len(tree) == 24:
            break

    viable: list[tuple[list[str], list[set[str]]]] = []
    for edge in tree:
        parts = _components(node_ids, tree, edge)
        if len(parts) == 2 and min(map(len, parts)) >= 8:
            viable.append((edge, sorted(parts, key=lambda part: min(part))))
    if not viable:
        raise RuntimeError("sampler produced no balanced bridge")
    cut_edge, parts = viable[rng.randrange(len(viable))]

    tree_set = {tuple(edge) for edge in tree}
    same_side_extras = [
        edge for edge in _grid_edges()
        if tuple(edge) not in tree_set
        and any(edge[0] in part and edge[1] in part for part in parts)
    ]
    rng.shuffle(same_side_extras)
    if len(same_side_extras) < 6:
        raise RuntimeError("sampler produced too few internal cycle edges")
    edges = tree + same_side_extras[:6]

    answer_color = COLOR_NAMES[int(seed) % 3]
    answer_side = (int(seed) // 3) % 2
    available = [sorted(part - set(cut_edge)) for part in parts]
    rng.shuffle(available[0])
    rng.shuffle(available[1])
    terminal_pairs: dict[str, list[str]] = {}
    terminal_pairs[answer_color] = sorted([available[answer_side].pop(), available[answer_side].pop()])
    for color in COLOR_NAMES:
        if color != answer_color:
            terminal_pairs[color] = sorted([available[0].pop(), available[1].pop()])

    rng.shuffle(nodes)
    rng.shuffle(edges)
    # Rendering uses canonical order, so these record permutations are aliases.
    return {
        "nodes": nodes,
        "edges": edges,
        "cut_edge": cut_edge,
        "terminal_pairs": terminal_pairs,
    }


def _answer_and_parts(scene: dict) -> tuple[str, list[set[str]]]:
    node_ids = [str(node["id"]) for node in scene["nodes"]]
    edges = [list(map(str, edge)) for edge in scene["edges"]]
    cut_edge = list(map(str, scene["cut_edge"]))
    parts = _components(node_ids, edges, cut_edge)
    connected: list[str] = []
    for color, pair in scene["terminal_pairs"].items():
        if len(pair) != 2:
            raise ValueError("every terminal color must occur twice")
        if any(pair[0] in part and pair[1] in part for part in parts):
            connected.append(str(color))
    if len(parts) != 2 or len(connected) != 1:
        raise ValueError("expected two post-cut components and one connected pair")
    return connected[0], parts


def analytic_gold(scene: dict) -> str:
    answer, _ = _answer_and_parts(scene)
    if answer not in COLORS:
        raise ValueError("unknown answer color")
    return answer


def margin(scene: dict) -> float:
    """One discrete edge operation separates this valid world from its boundary."""

    _answer_and_parts(scene)
    return 1.0


def is_quarantined(scene: dict) -> bool:
    try:
        node_ids = [str(node["id"]) for node in scene["nodes"]]
        edges = [list(map(str, edge)) for edge in scene["edges"]]
        cut = _edge(*map(str, scene["cut_edge"]))
        if len(node_ids) != 25 or len(set(node_ids)) != 25 or len(edges) != 30:
            return True
        if cut not in [_edge(*edge) for edge in edges]:
            return True
        if len(_components(node_ids, edges)) != 1:
            return True
        _, parts = _answer_and_parts(scene)
        if min(map(len, parts)) < 8:
            return True
        flat = [node for pair in scene["terminal_pairs"].values() for node in pair]
        if set(scene["terminal_pairs"]) != set(COLOR_NAMES) or len(flat) != 6 or len(set(flat)) != 6:
            return True
        if set(flat) & set(cut):
            return True
        return False
    except (KeyError, TypeError, ValueError):
        return True


def render(scene: dict) -> Image.Image:
    scale = 4
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    node_map = {str(node["id"]): node for node in scene["nodes"]}
    terminal_color = {
        node_id: color for color, pair in scene["terminal_pairs"].items() for node_id in pair
    }
    for a_id, b_id in sorted((_edge(*edge) for edge in scene["edges"])):
        a, b = node_map[a_id], node_map[b_id]
        draw.line(
            [(a["x"] * scale, a["y"] * scale), (b["x"] * scale, b["y"] * scale)],
            fill=INK, width=7 * scale,
        )
    for node_id in sorted(node_map):
        node = node_map[node_id]
        color = COLORS.get(terminal_color.get(node_id, ""), JOINT)
        radius = 17 if node_id in terminal_color else 12
        box = [
            (node["x"] - radius) * scale, (node["y"] - radius) * scale,
            (node["x"] + radius) * scale, (node["y"] + radius) * scale,
        ]
        draw.ellipse(box, fill=color, outline=INK, width=3 * scale)

    a, b = (node_map[node_id] for node_id in scene["cut_edge"])
    mx, my = (a["x"] + b["x"]) / 2.0, (a["y"] + b["y"]) / 2.0
    arm = 15
    draw.line([((mx - arm) * scale, (my - arm) * scale), ((mx + arm) * scale, (my + arm) * scale)], fill=BACKGROUND, width=13 * scale)
    draw.line([((mx - arm) * scale, (my + arm) * scale), ((mx + arm) * scale, (my - arm) * scale)], fill=BACKGROUND, width=13 * scale)
    draw.line([((mx - arm) * scale, (my - arm) * scale), ((mx + arm) * scale, (my + arm) * scale)], fill=CUT, width=7 * scale)
    draw.line([((mx - arm) * scale, (my + arm) * scale), ((mx + arm) * scale, (my - arm) * scale)], fill=CUT, width=7 * scale)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["nodes"] = list(reversed(twin["nodes"]))
    twin["edges"] = list(reversed(twin["edges"]))
    twin["terminal_pairs"] = {
        color: list(reversed(pair))
        for color, pair in reversed(list(twin["terminal_pairs"].items()))
    }
    return [("reverse_record_orders", twin)]
