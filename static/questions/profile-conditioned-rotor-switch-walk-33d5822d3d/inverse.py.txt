"""Independent PNG-only inverse for the self-flipping rotor walk."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_STATE_COLORS = np.array(((35, 116, 190), (0, 145, 126), (112, 78, 170)), dtype=np.int16)
_START_COLOR = np.array((218, 54, 53), dtype=np.int16)
_STEP_COLOR = np.array((235, 132, 35), dtype=np.int16)
_PIP_COLOR = np.array((25, 28, 32), dtype=np.int16)


def _near_any(arr: np.ndarray, colors: np.ndarray, tolerance: float) -> np.ndarray:
    delta = arr.astype(np.int32)[:, :, None, :] - colors.astype(np.int32)[None, None, :, :]
    return np.min(np.sum(delta * delta, axis=3), axis=2) <= tolerance * tolerance


def _near(arr: np.ndarray, color: np.ndarray, tolerance: float) -> np.ndarray:
    delta = arr.astype(np.int32) - color.astype(np.int32)
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _components(mask: np.ndarray, min_area: int) -> list[tuple[int, float, float]]:
    """Return (area, cx, cy) for 4-connected components above min_area."""
    ys, xs = np.nonzero(mask)
    remaining = set(zip(ys.tolist(), xs.tolist()))
    found: list[tuple[int, float, float]] = []
    while remaining:
        first = remaining.pop()
        queue = deque([first])
        count = 0
        sx = 0.0
        sy = 0.0
        while queue:
            y, x = queue.popleft()
            count += 1
            sx += x
            sy += y
            for neighbor in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if neighbor in remaining:
                    remaining.remove(neighbor)
                    queue.append(neighbor)
        if count >= min_area:
            found.append((count, sx / count, sy / count))
    return found


def _replay(states: dict[int, bool], start: int, steps: int) -> int:
    local = dict(states)
    position = start
    for _ in range(steps):
        clockwise = local[position]
        old = position
        position = 1 + ((position - 1 + (1 if clockwise else -1)) % 4)
        local[old] = not local[old]
    return position


def decision_from_image(image: Image.Image) -> str:
    """Recover station states, identities, start, and step count from RGB pixels."""
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.shape[0] < 500 or arr.shape[1] < 700:
        raise ValueError(f"unsupported raster size {arr.shape[:2]}")

    state_mask = _near_any(arr, _STATE_COLORS, 28.0)
    station_components = _components(state_mask, 900)
    if len(station_components) != 4:
        raise ValueError(f"expected four state glyphs, found {len(station_components)}")
    station_components.sort(key=lambda item: (item[2], item[1]))

    pip_mask = _near(arr, _PIP_COLOR, 32.0)
    pips = [item for item in _components(pip_mask, 35) if item[0] <= 150]
    if len(pips) != 10:
        raise ValueError(f"expected ten isolated identity pips, found {len(pips)}")

    pip_counts = [0, 0, 0, 0]
    for _, px, py in pips:
        distances = [(px - cx) ** 2 + (py - cy) ** 2 for _, cx, cy in station_components]
        index = int(np.argmin(distances))
        if distances[index] > 110.0**2:
            raise ValueError("identity pip is not bound to a station")
        pip_counts[index] += 1
    if sorted(pip_counts) != [1, 2, 3, 4]:
        raise ValueError(f"station pip identities are not unique: {pip_counts}")

    states: dict[int, bool] = {}
    centers_by_label: dict[int, tuple[float, float]] = {}
    for component, label in zip(station_components, pip_counts, strict=True):
        area, cx, cy = component
        if 2500 <= area < 2850:
            raise ValueError(f"ambiguous filled/ring area {area}")
        states[label] = area >= 2850
        centers_by_label[label] = (cx, cy)

    start_components = _components(_near(arr, _START_COLOR, 30.0), 300)
    if len(start_components) != 1:
        raise ValueError(f"expected one start ring, found {len(start_components)}")
    _, sx, sy = start_components[0]
    start_distances = {
        label: (sx - center[0]) ** 2 + (sy - center[1]) ** 2
        for label, center in centers_by_label.items()
    }
    start = min(start_distances, key=start_distances.get)
    ordered = sorted(start_distances.values())
    if ordered[1] - ordered[0] < 10000:
        raise ValueError("start ring is not uniquely bound")

    step_components = _components(_near(arr, _STEP_COLOR, 30.0), 300)
    steps = len(step_components)
    if not 6 <= steps <= 9:
        raise ValueError(f"expected six to nine STEP cards, found {steps}")

    final = _replay(states, start, steps)
    return "ABCD"[final - 1]
