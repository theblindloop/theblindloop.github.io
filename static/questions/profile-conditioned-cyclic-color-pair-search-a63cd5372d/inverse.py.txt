"""Independent image-only inverse for cyclic color-pair search.

The inverse does not import the renderer or any scene-bearing module.  It
segments saturated connected components, groups the disk centroids into the
four visible panel quadrants, orders every panel's six components by angle, and
compares only equality of their measured component colors.
"""

from __future__ import annotations

import math
from collections import Counter, deque

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")
EXPECTED_SIZE = (960, 700)
MIN_COMPONENT_AREA = 500


def _is_ink(pixel: tuple[int, int, int]) -> bool:
    red, green, blue = pixel
    return max(pixel) - min(pixel) >= 55 and max(pixel) <= 235


def _components(image: Image.Image) -> list[dict]:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    width, height = image.size
    spread = array.max(axis=2).astype(np.int16) - array.min(axis=2).astype(np.int16)
    ys, xs = np.nonzero((spread >= 55) & (array.max(axis=2) <= 235))
    remaining = set(zip(xs.tolist(), ys.tolist()))
    components: list[dict] = []
    while remaining:
        start = remaining.pop()
        queue = deque([start])
        coordinates = [start]
        while queue:
            px, py = queue.popleft()
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not dx and not dy:
                        continue
                    neighbor = (px + dx, py + dy)
                    if neighbor in remaining:
                        remaining.remove(neighbor)
                        coordinates.append(neighbor)
                        queue.append(neighbor)
        area = len(coordinates)
        if area >= MIN_COMPONENT_AREA:
            xs_component = [point[0] for point in coordinates]
            ys_component = [point[1] for point in coordinates]
            colors = Counter(tuple(int(channel) for channel in array[y, x]) for x, y in coordinates)
            color, _ = colors.most_common(1)[0]
            components.append(
                {
                    "area": area,
                    "x": sum(xs_component) / area,
                    "y": sum(ys_component) / area,
                    "color": color,
                }
            )
    return components


def _panel_index(x: float, y: float, width: int, height: int) -> int:
    return (0 if y < height / 2 else 2) + (0 if x < width / 2 else 1)


def _panel_score(components: list[dict]) -> int:
    if len(components) != 6:
        raise ValueError("each panel must contain six disk components")
    center_x = sum(item["x"] for item in components) / 6.0
    center_y = sum(item["y"] for item in components) / 6.0
    ordered = sorted(
        components,
        key=lambda item: math.atan2(item["y"] - center_y, item["x"] - center_x),
    )
    colors = [item["color"] for item in ordered]
    return sum(colors[index] == colors[(index + 1) % 6] for index in range(6))


def decision_from_image(image: Image.Image) -> str:
    """Return a panel label, or raise ValueError when the raster is ambiguous."""

    if image.size != EXPECTED_SIZE:
        raise ValueError(f"expected a {EXPECTED_SIZE} raster")
    components = _components(image.convert("RGB"))
    if len(components) != 24:
        raise ValueError(f"expected 24 colored disk components, found {len(components)}")
    grouped: list[list[dict]] = [[], [], [], []]
    for component in components:
        grouped[_panel_index(component["x"], component["y"], image.width, image.height)].append(component)
    if any(len(group) != 6 for group in grouped):
        raise ValueError("colored components do not form four six-disk panels")
    scores = [_panel_score(group) for group in grouped]
    ordered = sorted(scores, reverse=True)
    if ordered[0] <= ordered[1] or ordered[0] < 1:
        raise ValueError("cyclic color-pair evidence is tied or absent")
    return LABELS[scores.index(ordered[0])]
