"""Deterministic renderer and analytic scorer for cyclic color-pair search."""

from __future__ import annotations

import itertools
import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

CANVAS_SIZE = (960, 700)
LABELS = ("A", "B", "C", "D")
PANEL_BOXES = (
    (40, 70, 440, 330),
    (520, 70, 920, 330),
    (40, 370, 440, 630),
    (520, 370, 920, 630),
)
BASE_PALETTE = ((220, 70, 70), (55, 125, 220), (55, 170, 105))
PANEL_FILL = (249, 249, 246)
PANEL_BORDER = (78, 84, 91)
TEXT_COLOR = (24, 28, 32)

# Every panel has two instances of each abstract color token.  These two
# target patterns have one and two cyclic equal-color edges respectively.
TARGET_PATTERNS = (
    (0, 0, 1, 2, 1, 2),
    (0, 0, 1, 2, 2, 1),
)


def _adjacency_score(tokens: Iterable[int]) -> int:
    values = tuple(int(value) for value in tokens)
    if len(values) != 6:
        raise ValueError("a ring must contain exactly six color tokens")
    return sum(values[index] == values[(index + 1) % 6] for index in range(6))


def _distractor_patterns() -> tuple[tuple[int, ...], ...]:
    values = sorted(
        {
            tuple(permutation)
            for permutation in itertools.permutations((0, 0, 1, 1, 2, 2))
            if _adjacency_score(permutation) == 0
        }
    )
    if not values:
        raise AssertionError("distractor construction produced no patterns")
    return tuple(values)


DISTRACTOR_PATTERNS = _distractor_patterns()


def sample_scene(seed: int) -> dict:
    """Sample one balanced scene; target identity is independent of nuisances."""

    selector = random.Random(int(seed) ^ 0x51ED270B)
    nuisance = random.Random(int(seed) ^ 0xC0FFEE11)
    target_index = selector.randrange(4)
    target_pattern = TARGET_PATTERNS[nuisance.randrange(len(TARGET_PATTERNS))]

    palette = list(BASE_PALETTE)
    nuisance.shuffle(palette)
    panels = []
    for index in range(4):
        if index == target_index:
            tokens = target_pattern
        else:
            tokens = DISTRACTOR_PATTERNS[nuisance.randrange(len(DISTRACTOR_PATTERNS))]
        panels.append(
            {
                "label": LABELS[index],
                "tokens": list(tokens),
                # Rotating the ring makes the first displayed dot irrelevant.
                "phase": nuisance.randrange(6),
            }
        )

    # These nuisance values are rendered identically in all four panels.  They
    # change scale across scenes but cannot carry panel identity.
    geometry = {
        "panel_boxes": [list(box) for box in PANEL_BOXES],
        "ring_radius": nuisance.choice((78, 82, 86)),
        "dot_radius": nuisance.choice((19, 20, 21)),
    }
    return {
        "panels": panels,
        "palette": [list(rgb) for rgb in palette],
        "geometry": geometry,
    }


def _panel_center(box: Iterable[int]) -> tuple[int, int]:
    left, top, right, bottom = (int(value) for value in box)
    return ((left + right) // 2, (top + bottom) // 2 + 10)


def _dot_centers(panel: dict, box: Iterable[int], ring_radius: int) -> list[tuple[int, int]]:
    cx, cy = _panel_center(box)
    phase = int(panel["phase"]) % 6
    centers = []
    for slot in range(6):
        angle = -math.pi / 2 + (slot + phase) * (math.pi / 3)
        centers.append(
            (int(round(cx + ring_radius * math.cos(angle))), int(round(cy + ring_radius * math.sin(angle))))
        )
    return centers


def render(scene: dict) -> Image.Image:
    geometry = scene["geometry"]
    boxes = geometry["panel_boxes"]
    ring_radius = int(geometry["ring_radius"])
    dot_radius = int(geometry["dot_radius"])
    palette = [tuple(int(channel) for channel in rgb) for rgb in scene["palette"]]
    if len(scene["panels"]) != 4 or len(boxes) != 4 or len(palette) != 3:
        raise ValueError("scene has an invalid panel, geometry, or palette count")

    image = Image.new("RGB", CANVAS_SIZE, "white")
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    for panel, box in zip(scene["panels"], boxes):
        left, top, right, bottom = (int(value) for value in box)
        draw.rounded_rectangle((left, top, right, bottom), radius=14, fill=PANEL_FILL, outline=PANEL_BORDER, width=3)
        draw.text((left + 16, top + 13), str(panel["label"]), fill=TEXT_COLOR, font=font)
        centers = _dot_centers(panel, box, ring_radius)
        tokens = panel["tokens"]
        if len(tokens) != 6:
            raise ValueError("each panel must have six tokens")
        phase = int(panel["phase"]) % 6
        for slot, center in enumerate(centers):
            token = int(tokens[(slot - phase) % 6])
            color = palette[token]
            x, y = center
            draw.ellipse((x - dot_radius, y - dot_radius, x + dot_radius, y + dot_radius), fill=color)
    return image


def _scores(scene: dict) -> list[int]:
    panels = scene["panels"]
    if len(panels) != 4:
        raise ValueError("scene must contain four panels")
    return [_adjacency_score(panel["tokens"]) for panel in panels]


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    ordered = sorted(scores, reverse=True)
    if ordered[0] <= ordered[1]:
        raise ValueError("scene has no unique cyclic adjacency winner")
    return LABELS[scores.index(ordered[0])]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene), reverse=True)
    if len(scores) < 2:
        return 0.0
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    try:
        scores = _scores(scene)
    except (KeyError, TypeError, ValueError):
        return True
    return len(scores) != 4 or margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # There are no latent fields that can be changed while preserving every
    # final pixel; the alias mode is therefore explicitly not_applicable.
    return []
