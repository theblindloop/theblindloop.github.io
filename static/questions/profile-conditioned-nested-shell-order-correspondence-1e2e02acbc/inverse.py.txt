"""Independent raster inverse for the nested-shell correspondence question.

Only the finished PNG reaches this module. It finds exact-color shell pixels,
fits one circle per color in each panel, orders fitted radii from large to
small, and compares the resulting color chains.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


PANEL_BOXES = {
    "reference": (24, 54, 386, 506),
    "A": (446, 54, 808, 506),
    "B": (870, 54, 1232, 506),
}
SHELL_PALETTE = {
    "red": (217, 63, 57),
    "blue": (43, 96, 190),
    "green": (40, 151, 83),
    "purple": (139, 76, 174),
}
EXPECTED_SHELLS = 4
MIN_COMPONENT_PIXELS = 180
MAX_CENTER_SPREAD = 5.5


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    visited = np.zeros(mask.shape, dtype=bool)
    output: list[np.ndarray] = []
    ys, xs = np.nonzero(mask)
    for start_y, start_x in zip(ys.tolist(), xs.tolist(), strict=True):
        if visited[start_y, start_x]:
            continue
        stack = [(start_y, start_x)]
        visited[start_y, start_x] = True
        points: list[tuple[int, int]] = []
        while stack:
            y, x = stack.pop()
            points.append((y, x))
            for next_y, next_x in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if (
                    0 <= next_y < height
                    and 0 <= next_x < width
                    and mask[next_y, next_x]
                    and not visited[next_y, next_x]
                ):
                    visited[next_y, next_x] = True
                    stack.append((next_y, next_x))
        output.append(np.asarray(points, dtype=np.float64))
    return output


def _fit_circle(component: np.ndarray) -> tuple[float, float, float]:
    """Algebraic least-squares circle fit on (row, column) component pixels."""

    if len(component) < MIN_COMPONENT_PIXELS:
        raise ValueError("shell component is too small")
    y = component[:, 0] + 0.5
    x = component[:, 1] + 0.5
    matrix = np.column_stack((2.0 * x, 2.0 * y, np.ones(len(x))))
    target = x * x + y * y
    solution, _, rank, _ = np.linalg.lstsq(matrix, target, rcond=None)
    if int(rank) < 3:
        raise ValueError("shell pixels do not define a circle")
    cx, cy, constant = (float(value) for value in solution)
    radius_sq = cx * cx + cy * cy + constant
    if radius_sq <= 0.0:
        raise ValueError("invalid fitted radius")
    return cx, cy, float(np.sqrt(radius_sq))


def _crop(array: np.ndarray, panel: str) -> tuple[np.ndarray, int, int]:
    left, top, right, bottom = PANEL_BOXES[panel]
    x0, y0 = left + 8, top + 28
    return array[y0 : bottom - 8, x0 : right - 8], x0, y0


def _shell_measurements(image: Image.Image, panel: str) -> list[tuple[str, float, float, float]]:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    crop, x0, y0 = _crop(array, panel)
    measurements: list[tuple[str, float, float, float]] = []
    for color, rgb in SHELL_PALETTE.items():
        mask = np.all(crop == np.asarray(rgb, dtype=np.uint8), axis=2)
        components = [component for component in _components(mask) if len(component) >= MIN_COMPONENT_PIXELS]
        if len(components) != 1:
            raise ValueError(f"{panel}: expected one connected shell for {color}, found {len(components)}")
        cx, cy, radius = _fit_circle(components[0])
        measurements.append((color, cx + x0, cy + y0, radius))
    if len(measurements) != EXPECTED_SHELLS:
        raise ValueError(f"{panel}: incomplete shell palette")
    center_x = float(np.mean([item[1] for item in measurements]))
    center_y = float(np.mean([item[2] for item in measurements]))
    spread = max(float(np.hypot(item[1] - center_x, item[2] - center_y)) for item in measurements)
    if spread > MAX_CENTER_SPREAD:
        raise ValueError(f"{panel}: shell centers disagree by {spread:.2f}px")
    return sorted(measurements, key=lambda item: item[3], reverse=True)


def _sequence(image: Image.Image, panel: str) -> tuple[str, ...]:
    return tuple(item[0] for item in _shell_measurements(image, panel))


def _mismatch(reference: tuple[str, ...], candidate: tuple[str, ...]) -> int:
    if len(reference) != len(candidate):
        return max(len(reference), len(candidate))
    return sum(left != right for left, right in zip(reference, candidate, strict=True))


def decision_from_image(image: Image.Image) -> str:
    """Return A or B from shell pixels in the final raster alone."""

    reference = _sequence(image, "reference")
    candidate_a = _sequence(image, "A")
    candidate_b = _sequence(image, "B")
    scores = (_mismatch(reference, candidate_a), _mismatch(reference, candidate_b))
    if scores.count(0) != 1 or scores[0] == scores[1]:
        raise ValueError(f"ambiguous or incomplete shell correspondence: scores={scores}")
    return "A" if scores[0] == 0 else "B"
