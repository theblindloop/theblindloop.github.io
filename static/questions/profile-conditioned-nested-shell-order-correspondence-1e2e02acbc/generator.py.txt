"""Deterministic RGB renderer and analytic label for nested shell order."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "nested_shell_order_correspondence-0.1.0"
CANVAS_SIZE = (1260, 560)
SUPERSAMPLE = 4
BACKGROUND = (247, 248, 251)
PANEL_FILL = (255, 255, 255)
PANEL_OUTLINE = (194, 198, 207)
PANEL_BOXES = {
    "reference": (24, 54, 386, 506),
    "A": (446, 54, 808, 506),
    "B": (870, 54, 1232, 506),
}
PANEL_CENTERS = {"reference": (205.0, 300.0), "A": (627.0, 300.0), "B": (1051.0, 300.0)}
SHELL_COLORS = ("red", "blue", "green", "purple")
SHELL_PALETTE = {
    "red": (217, 63, 57),
    "blue": (43, 96, 190),
    "green": (40, 151, 83),
    "purple": (139, 76, 174),
}
BASE_RADII = (58.0, 84.0, 110.0, 136.0)  # outer to inner order is sorted descending
NORMAL_SCALES = (0.92, 0.98, 1.04, 1.10)
NORMAL_STROKES = (10.0, 12.0, 14.0)


def _font(size: int):
    try:
        return ImageFont.load_default(size=size)
    except TypeError:
        return ImageFont.load_default()


def _swap_nonterminal(order: list[str], rng: random.Random) -> list[str]:
    """Make a decoy that differs at two middle ranks, never at only a cue rank."""

    output = list(order)
    left = rng.choice((0, 1))
    right = left + 1
    output[left], output[right] = output[right], output[left]
    return output


def _object(order: list[str], center: tuple[float, float], scale: float, stroke: float) -> dict:
    return {
        "order": list(order),
        "center": [round(float(center[0]), 3), round(float(center[1]), 3)],
        "scale": round(float(scale), 3),
        "stroke": round(float(stroke), 3),
    }


def sample_scene(seed: int) -> dict:
    """Sample equal-statistic candidate panels with a balanced exact order match."""

    rng = random.Random(int(seed))
    colors = list(SHELL_COLORS)
    rng.shuffle(colors)
    reference_order = list(colors)
    wrong_order = _swap_nonterminal(reference_order, rng)

    ref_cx, ref_cy = PANEL_CENTERS["reference"]
    reference = _object(
        reference_order,
        (ref_cx + rng.uniform(-12.0, 12.0), ref_cy + rng.uniform(-10.0, 10.0)),
        rng.choice(NORMAL_SCALES),
        rng.choice(NORMAL_STROKES),
    )

    # Candidate A/B have identical geometric statistics. Only the order permutation
    # and balanced left/right placement decide the answer.
    candidate_scale = 0.86 if int(seed) % 7 == 0 else rng.choice(NORMAL_SCALES)
    candidate_stroke = 18.0 if int(seed) % 7 == 0 else rng.choice(NORMAL_STROKES)
    acx, acy = PANEL_CENTERS["A"]
    bcx, bcy = PANEL_CENTERS["B"]
    candidate_a_center = (acx + rng.uniform(-15.0, 15.0), acy + rng.uniform(-12.0, 12.0))
    candidate_b_center = (bcx + rng.uniform(-15.0, 15.0), bcy + rng.uniform(-12.0, 12.0))
    correct_is_a = int(seed) % 2 == 0
    correct = _object(reference_order, candidate_a_center if correct_is_a else candidate_b_center, candidate_scale, candidate_stroke)
    wrong = _object(wrong_order, candidate_b_center if correct_is_a else candidate_a_center, candidate_scale, candidate_stroke)
    if correct_is_a:
        candidate_a, candidate_b = correct, wrong
    else:
        candidate_a, candidate_b = wrong, correct
    return {
        "reference": reference,
        "candidate_a": candidate_a,
        "candidate_b": candidate_b,
    }


def _radii_for(obj: dict) -> tuple[float, ...]:
    scale = float(obj["scale"])
    # The latent order is explicitly outside-to-inside, so pair its first
    # color with the largest radius.
    return tuple(radius * scale for radius in sorted(BASE_RADII, reverse=True))


def _sequence_mismatch(reference: list[str], candidate: list[str]) -> int:
    if len(reference) != len(candidate):
        return max(len(reference), len(candidate))
    return sum(left != right for left, right in zip(reference, candidate, strict=True))


def _scores(scene: dict) -> tuple[int, int]:
    reference = list(scene["reference"]["order"])
    return (
        _sequence_mismatch(reference, list(scene["candidate_a"]["order"])),
        _sequence_mismatch(reference, list(scene["candidate_b"]["order"])),
    )


def _min_visible_gap(obj: dict) -> float:
    radii = sorted(_radii_for(obj), reverse=True)
    stroke = float(obj["stroke"])
    return min(radii[index] - radii[index + 1] - stroke for index in range(len(radii) - 1))


def render(scene: dict) -> Image.Image:
    """Render only public fields of the supplied scene into a final RGB raster."""

    width, height = CANVAS_SIZE
    scale = SUPERSAMPLE
    image = Image.new("RGB", (width * scale, height * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    labels = {"reference": "REFERENCE", "A": "A", "B": "B"}
    for panel, box in PANEL_BOXES.items():
        left, top, right, bottom = box
        draw.rounded_rectangle(
            (left * scale, top * scale, right * scale, bottom * scale),
            radius=14 * scale,
            fill=PANEL_FILL,
            outline=PANEL_OUTLINE,
            width=2 * scale,
        )
        draw.text(
            ((left + right) * scale / 2.0, (top + 21) * scale),
            labels[panel],
            fill=(54, 59, 70),
            anchor="mm",
            font=_font(18 * scale),
        )

    # Keep the three label-bearing accesses explicit: the v0.4 latent-field
    # audit must be able to see that every analytic input is rendered.
    public_objects = (
        ("reference", scene["reference"]),
        ("A", scene["candidate_a"]),
        ("B", scene["candidate_b"]),
    )
    for _panel, obj in public_objects:
        cx, cy = (float(value) for value in obj["center"])
        stroke = max(3, round(float(obj["stroke"]) * scale))
        for color, radius in zip(obj["order"], _radii_for(obj), strict=True):
            r = radius * scale
            draw.ellipse(
                (round(cx * scale - r), round(cy * scale - r), round(cx * scale + r), round(cy * scale + r)),
                outline=SHELL_PALETTE[str(color)],
                width=stroke,
            )
    return image.resize((width, height), Image.Resampling.LANCZOS).convert("RGB")


def analytic_gold(scene: dict) -> str:
    """Select the unique candidate whose complete visible color chain matches."""

    scores = _scores(scene)
    if scores.count(0) != 1 or scores[0] == scores[1]:
        raise ValueError(f"ambiguous shell correspondence: scores={scores}")
    return "A" if scores[0] == 0 else "B"


def margin(scene: dict) -> float:
    """Runner-up gap in discrete shell-rank mismatches."""

    scores = _scores(scene)
    return float(abs(scores[0] - scores[1]))


def is_quarantined(scene: dict) -> bool:
    """Quarantine ambiguous assignments and near-touching finite-resolution shells."""

    try:
        scores = _scores(scene)
        if scores.count(0) != 1 or scores[0] == scores[1] or margin(scene) < 1.0:
            return True
        objects = (scene["reference"], scene["candidate_a"], scene["candidate_b"])
        return min(_min_visible_gap(obj) for obj in objects) < 6.0
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """All label-bearing shell fields are rendered; no latent alias is declared."""

    return []
