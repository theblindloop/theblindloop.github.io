"""Independent pixel-only inverse for closed color-domino chain search."""

from __future__ import annotations

from collections import deque
import itertools

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")
PALETTE = np.array(
    ((209, 73, 91), (48, 102, 190), (42, 157, 111), (233, 162, 59)),
    dtype=np.float64,
)


def _components(mask: np.ndarray) -> list[tuple[int, float, float, int, int]]:
    """Return area, centroid and extent for each 8-connected mask component."""
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[int, float, float, int, int]] = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        area = sum_x = sum_y = 0
        x0 = x1 = sx
        y0 = y1 = sy
        while queue:
            y, x = queue.popleft()
            area += 1
            sum_x += x
            sum_y += y
            x0, x1 = min(x0, x), max(x1, x)
            y0, y1 = min(y0, y), max(y1, y)
            for yy in range(max(0, y - 1), min(height, y + 2)):
                for xx in range(max(0, x - 1), min(width, x + 2)):
                    if mask[yy, xx] and not seen[yy, xx]:
                        seen[yy, xx] = True
                        queue.append((yy, xx))
        found.append((area, sum_x / area, sum_y / area, x1 - x0 + 1, y1 - y0 + 1))
    return found


def _endpoint_components(array: np.ndarray) -> list[tuple[float, float, int]]:
    rgb = array.astype(np.float64)
    distances = np.linalg.norm(rgb[:, :, None, :] - PALETTE[None, None, :, :], axis=3)
    nearest = distances.argmin(axis=2)
    minimum = distances.min(axis=2)
    endpoints: list[tuple[float, float, int]] = []
    for color in range(4):
        mask = (nearest == color) & (minimum < 38.0)
        for area, cx, cy, width, height in _components(mask):
            if 750 <= area <= 1500 and 33 <= width <= 45 and 33 <= height <= 45:
                endpoints.append((cx, cy, color))
    if len(endpoints) != 32:
        raise ValueError(f"expected 32 visible colored ends, recovered {len(endpoints)}")
    return endpoints


def _rows(endpoints: list[tuple[float, float, int]]) -> list[list[list[int]]]:
    ordered = sorted(endpoints, key=lambda item: item[1])
    groups: list[list[tuple[float, float, int]]] = []
    for endpoint in ordered:
        if not groups or endpoint[1] - np.mean([item[1] for item in groups[-1]]) > 35:
            groups.append([endpoint])
        else:
            groups[-1].append(endpoint)
    if len(groups) != 4 or any(len(group) != 8 for group in groups):
        raise ValueError(f"could not recover four rows of eight ends: {[len(g) for g in groups]}")

    panels: list[list[list[int]]] = []
    for group in groups:
        if max(item[1] for item in group) - min(item[1] for item in group) > 14:
            raise ValueError("a domino row is vertically ambiguous")
        by_x = sorted(group, key=lambda item: item[0])
        if any(by_x[i + 1][0] - by_x[i][0] < 28 for i in range(7)):
            raise ValueError("colored ends overlap horizontally")
        colors = [int(item[2]) for item in by_x]
        if [colors.count(value) for value in range(4)] != [2, 2, 2, 2]:
            raise ValueError(f"row endpoint inventory is corrupted: {colors}")
        panels.append([[colors[i], colors[i + 1]] for i in range(0, 8, 2)])
    return panels


def _best_score(panel: list[list[int]]) -> int:
    best = -1
    for order in itertools.permutations(range(4)):
        for flips in itertools.product((0, 1), repeat=4):
            chain = []
            for index, flip in zip(order, flips):
                a, b = panel[index]
                chain.append((b, a) if flip else (a, b))
            score = sum(chain[i][1] == chain[(i + 1) % 4][0] for i in range(4))
            best = max(best, int(score))
    return best


def decision_from_image(image: Image.Image) -> str:
    """Enumerate visible rows, recover tile pairs, and solve each circular search."""
    array = np.asarray(image.convert("RGB"))
    if array.ndim != 3 or array.shape[0] < 300 or array.shape[1] < 500:
        raise ValueError("image is too small for four domino rows")
    panels = _rows(_endpoint_components(array))
    scores = [_best_score(panel) for panel in panels]
    order = sorted(range(4), key=lambda index: scores[index], reverse=True)
    if scores[order[0]] != 4 or scores[order[0]] - scores[order[1]] < 2:
        raise ValueError(f"no unique closed-chain winner with margin two: {scores}")
    return LABELS[order[0]]
