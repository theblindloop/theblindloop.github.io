"""Deterministic renderer for closed color-domino chain search."""

from __future__ import annotations

import copy
import itertools
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 840
HEIGHT = 680
SCALE = 2
LABELS = ("A", "B", "C", "D")
VISIBLE_COLORS = ("#D1495B", "#3066BE", "#2A9D6F", "#E9A23B")
BACKGROUND = "#F7F9FB"
INK = "#263238"
CARD_FILL = "#FFFFFF"
ROW_CENTERS = (100, 255, 410, 565)
CARD_X = (112, 278, 444, 610)
CARD_WIDTH = 148
CARD_HEIGHT = 72


def _best_closed_score(panel: list[list[int]]) -> int:
    """Maximum number of equal-color joins in a circular use of all tiles."""
    if len(panel) != 4 or any(len(tile) != 2 for tile in panel):
        raise ValueError("each panel must contain four two-ended dominoes")
    best = -1
    for order in itertools.permutations(range(4)):
        for flips in itertools.product((0, 1), repeat=4):
            oriented = []
            for index, flip in zip(order, flips):
                left, right = map(int, panel[index])
                oriented.append((right, left) if flip else (left, right))
            score = sum(
                oriented[index][1] == oriented[(index + 1) % 4][0]
                for index in range(4)
            )
            best = max(best, int(score))
    return best


def _panel_scores(scene: dict) -> list[int]:
    return [_best_closed_score(panel) for panel in scene["panels"]]


def sample_scene(seed: int) -> dict:
    """Make one four-row scene with equal endpoint inventories in every row."""
    rng = random.Random(int(seed))
    winner = int(seed) % 4

    cycle = list(range(4))
    rng.shuffle(cycle)
    valid = [[cycle[i], cycle[(i + 1) % 4]] for i in range(4)]
    invalid = [
        [[0, 1], [0, 1], [2, 3], [2, 3]],
        [[0, 2], [0, 2], [1, 3], [1, 3]],
        [[0, 3], [0, 3], [1, 2], [1, 2]],
    ]
    rng.shuffle(invalid)

    panels: list[list[list[int]]] = []
    distractor_index = 0
    for panel_index in range(4):
        tiles = copy.deepcopy(valid if panel_index == winner else invalid[distractor_index])
        if panel_index != winner:
            distractor_index += 1
        rng.shuffle(tiles)
        for tile in tiles:
            if rng.randrange(2):
                tile.reverse()
        panels.append(tiles)

    palette = list(VISIBLE_COLORS)
    rng.shuffle(palette)
    scene = {
        "panels": panels,
        "palette": palette,
        "row_offsets": [[rng.randint(-5, 5), rng.randint(-4, 4)] for _ in range(4)],
        "tile_offsets": [
            [[rng.randint(-2, 2), rng.randint(-3, 3)] for _ in range(4)]
            for _ in range(4)
        ],
        "disk_radius": rng.randint(18, 21),
    }
    assert sorted(_panel_scores(scene)) == [2, 2, 2, 4]
    for panel in panels:
        endpoints = [value for tile in panel for value in tile]
        assert [endpoints.count(value) for value in range(4)] == [2, 2, 2, 2]
    return scene


def analytic_gold(scene: dict) -> str:
    scores = _panel_scores(scene)
    order = sorted(range(4), key=lambda index: scores[index], reverse=True)
    if scores[order[0]] != 4 or scores[order[0]] - scores[order[1]] < 2:
        raise ValueError(f"scene lacks a unique closed chain: {scores}")
    return LABELS[order[0]]


def margin(scene: dict) -> float:
    scores = sorted(_panel_scores(scene), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    scores = _panel_scores(scene)
    return scores.count(4) != 1 or margin(scene) < 2.0


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size
    )


def render(scene: dict) -> Image.Image:
    """Render only the declared visual state; analytic gold is never consulted."""
    panels = scene["panels"]
    palette = scene["palette"]
    row_offsets = scene["row_offsets"]
    tile_offsets = scene["tile_offsets"]
    radius = int(scene["disk_radius"])
    if len(panels) != 4 or len(palette) != 4:
        raise ValueError("scene must have four panels and four palette entries")

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    label_font = _font(30 * SCALE)
    for row, panel in enumerate(panels):
        if len(panel) != 4:
            raise ValueError("each row must contain four dominoes")
        row_dx, row_dy = map(int, row_offsets[row])
        cy = (ROW_CENTERS[row] + row_dy) * SCALE
        frame = (
            (76 + row_dx) * SCALE,
            cy - 55 * SCALE,
            (792 + row_dx) * SCALE,
            cy + 55 * SCALE,
        )
        draw.rounded_rectangle(
            frame,
            radius=16 * SCALE,
            fill="#EEF2F5",
            outline="#9AA7B0",
            width=2 * SCALE,
        )
        draw.text(
            ((29 + row_dx) * SCALE, cy - 20 * SCALE),
            LABELS[row],
            fill=INK,
            font=label_font,
        )

        for tile_index, tile in enumerate(panel):
            if len(tile) != 2:
                raise ValueError("each domino needs two ends")
            tile_dx, tile_dy = map(int, tile_offsets[row][tile_index])
            x0 = (CARD_X[tile_index] + row_dx + tile_dx) * SCALE
            y0 = cy - CARD_HEIGHT * SCALE / 2 + tile_dy * SCALE
            x1 = x0 + CARD_WIDTH * SCALE
            y1 = y0 + CARD_HEIGHT * SCALE
            draw.rounded_rectangle(
                (x0, y0, x1, y1),
                radius=11 * SCALE,
                fill=CARD_FILL,
                outline=INK,
                width=3 * SCALE,
            )
            mid = (x0 + x1) / 2
            draw.line((mid, y0 + 5 * SCALE, mid, y1 - 5 * SCALE), fill=INK, width=2 * SCALE)
            for side, token in enumerate(tile):
                cx = x0 + (CARD_WIDTH * (0.25 if side == 0 else 0.75)) * SCALE
                endpoint_cy = (y0 + y1) / 2
                rr = radius * SCALE
                draw.ellipse(
                    (cx - rr, endpoint_cy - rr, cx + rr, endpoint_cy + rr),
                    fill=palette[int(token)],
                )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Relabel latent tokens while preserving every rendered endpoint color."""
    permutation = (2, 0, 3, 1)  # old token -> new token
    twin = copy.deepcopy(scene)
    twin["panels"] = [
        [[permutation[int(value)] for value in tile] for tile in panel]
        for panel in scene["panels"]
    ]
    new_palette = [""] * 4
    for old, new in enumerate(permutation):
        new_palette[new] = scene["palette"][old]
    twin["palette"] = new_palette
    return [("global_token_relabel", twin)]
