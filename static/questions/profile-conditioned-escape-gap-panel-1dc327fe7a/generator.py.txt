"""Escape-gap panel world.

Latent scene z = {width, height, palette, radius, ring_width, dot_radius, panels}.

Four square panels sit side by side. Each panel holds one broken ring: a circle
outline of fixed radius with a single angular opening, plus one small filled dot
outside the ring. A dot can leave its panel's ring only if the direction from the
ring centre to the dot falls inside that ring's opening. Exactly one panel is
aimed that way; in every other panel the dot's bearing misses the opening.

The decision is the panel whose opening faces its dot. The margin is the smallest
angular clearance in the image, in degrees: for the answer panel the angle from
the dot bearing to the nearer opening edge, and for every other panel the angle
by which its dot bearing misses its opening.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "escape_gap_panel-0.1.0"

WIDTH = 720
HEIGHT = 200
PANEL_COUNT = 4
PANEL_WIDTH = WIDTH // PANEL_COUNT
PANEL_CY = HEIGHT / 2.0
RADIUS = 48
RING_WIDTH = 7
DOT_RADIUS = 8
SUPERSAMPLE = 3

MIN_INSIDE_CLEARANCE = 8.0
MIN_OUTSIDE_CLEARANCE = 14.0
QUARANTINE_CLEARANCE = 10.0

GAP_MIN = 34.0
GAP_MAX = 52.0
DIST_MIN = 66.0
DIST_MAX = 76.0

PANELS = ("panel_1", "panel_2", "panel_3", "panel_4")

PALETTES = (
    {"background": (255, 255, 255), "ring": (26, 26, 30), "dot": (30, 28, 28), "frame": (176, 176, 176)},
    {"background": (247, 245, 239), "ring": (18, 32, 46), "dot": (44, 20, 22), "frame": (170, 172, 174)},
    {"background": (242, 246, 250), "ring": (42, 22, 24), "dot": (20, 34, 30), "frame": (168, 176, 184)},
    {"background": (249, 250, 243), "ring": (22, 42, 28), "dot": (24, 24, 52), "frame": (180, 176, 166)},
)


def _panel_index(cx: float) -> int:
    return min(PANEL_COUNT - 1, max(0, int(cx // PANEL_WIDTH)))


def _angular_distance(first: float, second: float) -> float:
    """Absolute angular separation of two bearings, in [0, 180] degrees."""
    delta = (first - second) % 360.0
    if delta > 180.0:
        delta = 360.0 - delta
    return delta


def _signed_clearances(scene: dict) -> dict:
    """Per panel: bearing miss minus half opening. Negative means aimed out."""
    result: dict = {}
    for panel in scene["panels"]:
        miss = _angular_distance(panel["dot_deg"], panel["gap_deg"])
        result[PANELS[_panel_index(panel["cx"])]] = miss - panel["gap_width"] / 2.0
    return result


def analytic_gold(scene: dict) -> str:
    """Panel whose ring opening faces its own dot."""
    clearances = _signed_clearances(scene)
    if len(clearances) != PANEL_COUNT:
        raise ValueError("scene does not populate every panel exactly once")
    aimed = [name for name, value in clearances.items() if value < 0.0]
    if len(aimed) != 1:
        raise ValueError(f"scene has no unique aimed panel: {sorted(aimed)}")
    return aimed[0]


def margin(scene: dict) -> float:
    """Smallest angular clearance anywhere in the image, in degrees."""
    clearances = _signed_clearances(scene)
    if not clearances:
        raise ValueError("scene has no panel")
    return round(min(abs(value) for value in clearances.values()), 6)


def is_quarantined(scene: dict) -> bool:
    """Near-boundary scenes whose tightest clearance is hard to resolve."""
    return bool(margin(scene) < QUARANTINE_CLEARANCE)


def latent_symmetries(scene: dict):
    """Storage aliases: panel record order and full-turn angle representatives."""
    reordered = dict(scene, panels=[scene["panels"][i] for i in (2, 0, 3, 1)])
    reversed_order = dict(scene, panels=list(reversed(scene["panels"])))
    wound = dict(
        scene,
        panels=[
            dict(panel, gap_deg=panel["gap_deg"] + 360.0, dot_deg=panel["dot_deg"] - 360.0)
            for panel in scene["panels"]
        ],
    )
    return [
        ("panel_record_order", reordered),
        ("panel_record_reversal", reversed_order),
        ("angle_period_representative", wound),
    ]


def _spread(rng: random.Random, low: float, high: float, count: int, apart: float) -> list:
    """Distinct half-degree values in [low, high] separated by at least apart."""
    for _ in range(200):
        steps = sorted(rng.sample(range(int((high - low) * 2) + 1), count))
        values = [low + step / 2.0 for step in steps]
        if all(values[i + 1] - values[i] >= apart for i in range(count - 1)):
            rng.shuffle(values)
            return values
    raise RuntimeError("could not spread values")


def sample_scene(seed: int) -> dict:
    rng = random.Random(repr(("escape_gap_panel", seed)))
    target = seed % PANEL_COUNT
    gap_rank = (seed // PANEL_COUNT) % PANEL_COUNT
    dist_rank = (seed // PANEL_COUNT + 2 * (seed % PANEL_COUNT)) % PANEL_COUNT
    palette_index = rng.randrange(len(PALETTES))
    for _ in range(4000):
        widths = sorted(_spread(rng, GAP_MIN, GAP_MAX, PANEL_COUNT, 3.0))
        distances = sorted(_spread(rng, DIST_MIN, DIST_MAX, PANEL_COUNT, 2.0))
        others = [i for i in range(PANEL_COUNT) if i != target]
        width_pool = [w for j, w in enumerate(widths) if j != gap_rank]
        dist_pool = [d for j, d in enumerate(distances) if j != dist_rank]
        rng.shuffle(width_pool)
        rng.shuffle(dist_pool)
        assigned_width = {target: widths[gap_rank]}
        assigned_dist = {target: distances[dist_rank]}
        for slot, index in enumerate(others):
            assigned_width[index] = width_pool[slot]
            assigned_dist[index] = dist_pool[slot]
        panels = []
        for index in range(PANEL_COUNT):
            gap_width = assigned_width[index]
            gap_deg = rng.randrange(720) / 2.0
            side = rng.choice((-1.0, 1.0))
            if index == target:
                ceiling = min(24.0, gap_width / 2.0)
                if ceiling < MIN_INSIDE_CLEARANCE:
                    break
                steps = int((ceiling - MIN_INSIDE_CLEARANCE) * 2)
                clearance = MIN_INSIDE_CLEARANCE + rng.randint(0, steps) / 2.0
                offset = gap_width / 2.0 - clearance
            else:
                clearance = MIN_OUTSIDE_CLEARANCE + rng.randrange(0, 261) / 2.0
                offset = gap_width / 2.0 + clearance
            panels.append(
                {
                    "cx": float(index * PANEL_WIDTH + PANEL_WIDTH / 2.0),
                    "cy": PANEL_CY,
                    "gap_deg": gap_deg,
                    "gap_width": gap_width,
                    "dot_deg": (gap_deg + side * offset) % 360.0,
                    "dot_distance": assigned_dist[index],
                }
            )
        if len(panels) != PANEL_COUNT:
            continue
        rng.shuffle(panels)
        scene = {
            "width": WIDTH,
            "height": HEIGHT,
            "palette": palette_index,
            "radius": RADIUS,
            "ring_width": RING_WIDTH,
            "dot_radius": DOT_RADIUS,
            "panels": panels,
        }
        try:
            if analytic_gold(scene) != PANELS[target]:
                continue
        except ValueError:
            continue
        if margin(scene) < MIN_INSIDE_CLEARANCE:
            continue
        return scene
    raise RuntimeError(f"could not sample a valid scene for seed {seed}")


def render(scene: dict) -> Image.Image:
    palette = PALETTES[scene["palette"] % len(PALETTES)]
    width = scene["width"]
    height = scene["height"]
    radius = scene["radius"]
    ring_width = scene["ring_width"]
    dot_radius = scene["dot_radius"]
    ss = SUPERSAMPLE
    image = Image.new("RGB", (width * ss, height * ss), palette["background"])
    draw = ImageDraw.Draw(image)
    line = max(1, ss // 2)

    draw.rectangle(
        [(0, 0), (width * ss - 1, height * ss - 1)], outline=palette["frame"], width=line
    )
    for index in range(1, PANEL_COUNT):
        x = index * PANEL_WIDTH * ss
        draw.line([(x, 0), (x, height * ss)], fill=palette["frame"], width=line)

    for panel in scene["panels"]:
        cx = panel["cx"] * ss
        cy = panel["cy"] * ss
        gap_deg = panel["gap_deg"] % 360.0
        half = panel["gap_width"] / 2.0
        box = [
            (cx - radius * ss, cy - radius * ss),
            (cx + radius * ss, cy + radius * ss),
        ]
        draw.arc(box, gap_deg + half, gap_deg - half + 360.0, fill=palette["ring"], width=ring_width * ss)
        theta = math.radians(panel["dot_deg"] % 360.0)
        dx = cx + panel["dot_distance"] * ss * math.cos(theta)
        dy = cy + panel["dot_distance"] * ss * math.sin(theta)
        draw.ellipse(
            [
                (dx - dot_radius * ss, dy - dot_radius * ss),
                (dx + dot_radius * ss, dy + dot_radius * ss),
            ],
            fill=palette["dot"],
        )
    return image.resize((width, height), Image.LANCZOS)
