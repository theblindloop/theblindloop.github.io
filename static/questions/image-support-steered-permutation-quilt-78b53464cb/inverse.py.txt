"""Independent image-only inverse arm for the permutation quilt."""

from __future__ import annotations

import itertools

import numpy as np
from PIL import Image

N = 4
PERMS = tuple(itertools.permutations(range(N)))
PALETTE = {
    "navy": np.array((30, 43, 66), dtype=np.int16),
    "teal": np.array((48, 166, 166), dtype=np.int16),
    "cream": np.array((240, 229, 194), dtype=np.int16),
    "reference": np.array((205, 214, 226), dtype=np.int16),
    "ruby": np.array((210, 65, 75), dtype=np.int16),
    "green": np.array((45, 170, 105), dtype=np.int16),
    "blue": np.array((55, 120, 220), dtype=np.int16),
    "gold": np.array((235, 170, 45), dtype=np.int16),
}


def _mask(array: np.ndarray, color: np.ndarray, tolerance: int = 12) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _bbox(mask: np.ndarray) -> tuple[int, int, int, int]:
    points = np.argwhere(mask)
    if len(points) < 100:
        raise ValueError("required frame is missing")
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    return int(x0), int(y0), int(x1), int(y1)


def _canonical(matrix) -> tuple[int, ...]:
    return min(
        tuple(matrix[rp[r]][cp[c]] for r in range(N) for c in range(N))
        for rp in PERMS
        for cp in PERMS
    )


def _read_matrix(array: np.ndarray, frame_color: np.ndarray) -> list[list[int]]:
    frame_mask = _mask(array, frame_color)
    x0, y0, x1, y1 = _bbox(frame_mask)
    width, height = x1 - x0 + 1, y1 - y0 + 1
    if not (130 <= width <= 225 and 130 <= height <= 225):
        raise ValueError("frame geometry is corrupted")
    border = int(round(min(width, height) * (0.075 if width > 180 else 0.095)))
    inner_x0, inner_y0 = x0 + border, y0 + border
    inner_x1, inner_y1 = x1 - border, y1 - border
    cell_w = (inner_x1 - inner_x0 + 1) / N
    cell_h = (inner_y1 - inner_y0 + 1) / N
    matrix: list[list[int]] = []
    for row in range(N):
        values = []
        for col in range(N):
            cx = int(round(inner_x0 + (col + 0.5) * cell_w))
            cy = int(round(inner_y0 + (row + 0.5) * cell_h))
            patch = array[max(0, cy - 3):cy + 4, max(0, cx - 3):cx + 4]
            mean = patch.reshape(-1, 3).mean(axis=0)
            teal_distance = float(np.linalg.norm(mean - PALETTE["teal"]))
            cream_distance = float(np.linalg.norm(mean - PALETTE["cream"]))
            if min(teal_distance, cream_distance) > 55:
                raise ValueError("tile center is corrupted")
            values.append(1 if teal_distance < cream_distance else 0)
        matrix.append(values)
    if sum(map(sum, matrix)) != 8:
        raise ValueError("mosaic occupancy is not balanced")
    return matrix


def _validate_connected_quilt(array: np.ndarray) -> None:
    # The displayed world is one solid, area-like quilt. Every local sector must
    # remain present before any correspondence is attempted.
    foreground = np.linalg.norm(array.astype(np.int16) - 255, axis=2) > 35
    points = np.argwhere(foreground)
    if len(points) < array.shape[0] * array.shape[1] * 0.70:
        raise ValueError("continuous quilt is incomplete")
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    if (x1 - x0 + 1) < array.shape[1] * 0.88 or (y1 - y0 + 1) < array.shape[0] * 0.88:
        raise ValueError("quilt no longer spans the image")
    for gy in range(8):
        sy0 = int(y0 + (y1 - y0 + 1) * gy / 8)
        sy1 = int(y0 + (y1 - y0 + 1) * (gy + 1) / 8)
        for gx in range(8):
            sx0 = int(x0 + (x1 - x0 + 1) * gx / 8)
            sx1 = int(x0 + (x1 - x0 + 1) * (gx + 1) / 8)
            if float(foreground[sy0:sy1, sx0:sx1].mean()) < 0.93:
                raise ValueError("a quilt sector is erased")


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or min(array.shape[:2]) < 256:
        raise ValueError("unsupported raster")
    _validate_connected_quilt(array)
    reference = _read_matrix(array, PALETTE["reference"])
    target = _canonical(reference)
    matches = []
    for label in ("ruby", "green", "blue", "gold"):
        candidate = _read_matrix(array, PALETTE[label])
        if _canonical(candidate) == target:
            matches.append(label)
    if len(matches) != 1:
        raise ValueError("image has no unique matching frame")
    return matches[0]
