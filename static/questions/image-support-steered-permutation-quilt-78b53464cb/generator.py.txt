"""Renderer and analytic arm for a row/column-permutation correspondence quilt."""

from __future__ import annotations

import itertools
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH = HEIGHT = 768
N = 4
FRAME_COLORS = ("ruby", "green", "blue", "gold")
RGB = {
    "navy": (30, 43, 66),
    "teal": (48, 166, 166),
    "cream": (240, 229, 194),
    "reference": (205, 214, 226),
    "ruby": (210, 65, 75),
    "green": (45, 170, 105),
    "blue": (55, 120, 220),
    "gold": (235, 170, 45),
}
SLOTS = {
    "nw": (86, 382),
    "ne": (534, 382),
    "sw": (86, 588),
    "se": (534, 588),
}
PERMS = tuple(itertools.permutations(range(N)))


def _matrix_tuple(matrix: Iterable[Iterable[int]]) -> tuple[tuple[int, ...], ...]:
    return tuple(tuple(int(value) for value in row) for row in matrix)


def _transform(matrix, row_perm, col_perm):
    source = _matrix_tuple(matrix)
    return tuple(tuple(source[row_perm[r]][col_perm[c]] for c in range(N)) for r in range(N))


def _canonical(matrix) -> tuple[int, ...]:
    return min(
        tuple(value for row in _transform(matrix, rp, cp) for value in row)
        for rp in PERMS
        for cp in PERMS
    )


def _orbit_distance(left, right) -> int:
    a = _matrix_tuple(left)
    return min(
        sum(a[r][c] != transformed[r][c] for r in range(N) for c in range(N))
        for rp in PERMS
        for cp in PERMS
        for transformed in (_transform(right, rp, cp),)
    )


def _balanced_matrix(rng: random.Random):
    values = [0] * 8 + [1] * 8
    rng.shuffle(values)
    return tuple(tuple(values[r * N + c] for c in range(N)) for r in range(N))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    while True:
        reference = _balanced_matrix(rng)
        # Avoid highly symmetric references with tiny equivalence orbits.
        orbit = {_transform(reference, rp, cp) for rp in PERMS for cp in PERMS}
        if len(orbit) >= 72:
            break

    correct = _transform(reference, rng.choice(PERMS), rng.choice(PERMS))
    distractors = []
    seen = {_canonical(reference)}
    while len(distractors) < 3:
        candidate = _balanced_matrix(rng)
        key = _canonical(candidate)
        if key in seen or _orbit_distance(reference, candidate) < 4:
            continue
        seen.add(key)
        distractors.append(candidate)

    correct_index = int(seed) % 4
    candidates = list(distractors)
    candidates.insert(correct_index, correct)

    desired_answer = FRAME_COLORS[(int(seed) // 4) % 4]
    frame_colors = list(FRAME_COLORS)
    rng.shuffle(frame_colors)
    swap_index = frame_colors.index(desired_answer)
    frame_colors[correct_index], frame_colors[swap_index] = (
        frame_colors[swap_index],
        frame_colors[correct_index],
    )
    slots = list(SLOTS)
    rng.shuffle(slots)
    return {
        "reference": [list(row) for row in reference],
        "candidates": [[list(row) for row in matrix] for matrix in candidates],
        "frame_colors": frame_colors,
        "candidate_slots": slots,
        "width": WIDTH,
        "height": HEIGHT,
    }


def _draw_matrix(draw: ImageDraw.ImageDraw, matrix, box, frame_rgb, border: int) -> None:
    x0, y0, x1, y1 = box
    draw.rectangle(box, fill=frame_rgb)
    inner = (x0 + border, y0 + border, x1 - border, y1 - border)
    ix0, iy0, ix1, iy1 = inner
    cell = (ix1 - ix0 + 1) // N
    for row in range(N):
        for col in range(N):
            left = ix0 + col * cell
            top = iy0 + row * cell
            right = ix0 + (col + 1) * cell - 1
            bottom = iy0 + (row + 1) * cell - 1
            color = RGB["teal"] if int(matrix[row][col]) else RGB["cream"]
            draw.rectangle((left, top, right, bottom), fill=color)
            draw.rectangle((left, top, right, bottom), outline=RGB["navy"], width=3)


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    reference = scene["reference"]
    candidates = scene["candidates"]
    frame_colors = scene["frame_colors"]
    candidate_slots = scene["candidate_slots"]

    image = Image.new("RGB", (width, height), "white")
    draw = ImageDraw.Draw(image)
    draw.rectangle((28, 28, width - 29, height - 29), fill=RGB["navy"])
    font = ImageFont.load_default()
    draw.text((348, 47), "REFERENCE", fill=RGB["cream"], font=font)
    _draw_matrix(draw, reference, (276, 72, 491, 287), RGB["reference"], 16)
    draw.text((301, 326), "REARRANGE WHOLE ROWS + WHOLE COLUMNS", fill=RGB["cream"], font=font)
    for matrix, frame, slot in zip(candidates, frame_colors, candidate_slots, strict=True):
        x0, y0 = SLOTS[slot]
        _draw_matrix(draw, matrix, (x0, y0, x0 + 147, y0 + 147), RGB[frame], 14)
        draw.text((x0 + 50, y0 - 20), frame.upper(), fill=RGB["cream"], font=font)
    return image


def analytic_gold(scene: dict) -> str:
    reference = scene["reference"]
    candidates = scene["candidates"]
    frame_colors = scene["frame_colors"]
    target = _canonical(reference)
    matches = [index for index, candidate in enumerate(candidates) if _canonical(candidate) == target]
    if len(matches) != 1:
        raise ValueError("scene does not have one unique correspondence")
    return str(frame_colors[matches[0]])


def margin(scene: dict) -> float:
    reference = scene["reference"]
    candidates = scene["candidates"]
    distances = sorted(_orbit_distance(reference, candidate) for candidate in candidates)
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    try:
        distances = [_orbit_distance(scene["reference"], item) for item in scene["candidates"]]
    except Exception:
        return True
    return distances.count(0) != 1 or sorted(distances)[1] < 4


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
