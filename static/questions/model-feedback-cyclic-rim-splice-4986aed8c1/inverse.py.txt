"""Independent PNG-only inverse for Cyclic Rim Splice."""
from __future__ import annotations
from collections import deque
import numpy as np
from PIL import Image

WIDTH,HEIGHT=1100,800; N,CELL=7,68; GRID_ORIGIN=(62,231); INK=np.array((42,47,57),dtype=np.int16)
START=(3,3); TERMINALS={"A":(0,0),"B":(0,6),"C":(6,0),"D":(6,6)}
DIRECTIONS=((-1,0),(0,1),(1,0),(0,-1))
MARK_COLORS={"S":(215,70,65),"A":(247,174,47),"B":(45,119,204),"C":(76,166,88),"D":(150,82,191)}
PORTAL_COLORS={"P":(25,151,171),"Q":(146,79,190)}
RIM_COLORS={"red":(224,68,75),"gold":(244,177,43),"green":(50,166,101),"blue":(54,122,214)}

def _color_count(patch,color):
    ref=np.array(color,dtype=np.int16); return int((np.abs(patch.astype(np.int16)-ref).max(axis=2)<=3).sum())
def _center(cell):
    r,c=cell; gx,gy=GRID_ORIGIN; return gx+c*CELL+CELL//2,gy+r*CELL+CELL//2
def _read_rims(arr,cx,cy):
    patches=(arr[cy-22:cy-15,cx-8:cx+9],arr[cy-8:cy+9,cx+15:cx+22],arr[cy+15:cy+22,cx-8:cx+9],arr[cy-8:cy+9,cx-22:cx-15])
    result=[]
    for patch in patches:
        counts={tag:_color_count(patch,color) for tag,color in RIM_COLORS.items()}; tag,count=max(counts.items(),key=lambda x:x[1])
        if count<80 or sum(v>=35 for v in counts.values())!=1: return None
        result.append(tag)
    return result if sorted(result)==sorted(RIM_COLORS) else None

def decision_from_image(image:Image.Image)->str:
    if image.size!=(WIDTH,HEIGHT): return "abstain"
    arr=np.asarray(image.convert("RGB"))
    for label,cell in {"S":START,**TERMINALS}.items():
        cx,cy=_center(cell)
        if _color_count(arr[cy-13:cy+14,cx-13:cx+14],MARK_COLORS[label])<350: return "abstain"
    portals={symbol:[] for symbol in PORTAL_COLORS}; blocked=set()
    for r in range(N):
        for c in range(N):
            cx,cy=_center((r,c)); patch=arr[cy-9:cy+10,cx-9:cx+10]
            hits=[symbol for symbol,color in PORTAL_COLORS.items() if _color_count(patch,color)>=140]
            if len(hits)>1: return "abstain"
            if hits:
                rims=_read_rims(arr,cx,cy)
                if rims is None: return "abstain"
                portals[hits[0]].append({"cell":(r,c),"rims":tuple(rims)}); blocked.add((r,c))
    if any(len(v)!=2 for v in portals.values()) or len(blocked)!=4: return "abstain"
    for entries in portals.values(): entries.sort(key=lambda e:e["cell"])
    graph={x:set() for r in range(N) for c in range(N) if (x:=(r,c)) not in blocked}
    for r in range(N):
        for c in range(N):
            if (r,c) in blocked: continue
            for dr,dc in ((0,1),(1,0)):
                nxt=(r+dr,c+dc)
                if not (0<=nxt[0]<N and 0<=nxt[1]<N) or nxt in blocked: continue
                x1,y1=_center((r,c)); x2,y2=_center(nxt)
                patch=arr[y1-4:y1+5,min(x1,x2)+22:max(x1,x2)-21] if r==nxt[0] else arr[min(y1,y2)+22:max(y1,y2)-21,x1-4:x1+5]
                frac=float((np.abs(patch.astype(np.int16)-INK).max(axis=2)<=3).mean())
                if frac>=.72: graph[(r,c)].add(nxt); graph[nxt].add((r,c))
                elif frac>.08: return "abstain"
    for entries in portals.values():
        maps=[]
        for e in entries:
            r,c=e["cell"]; maps.append({tag:(r+DIRECTIONS[i][0],c+DIRECTIONS[i][1]) for i,tag in enumerate(e["rims"])})
        for tag in RIM_COLORS:
            a,b=maps[0][tag],maps[1][tag]
            if a not in graph or b not in graph: return "abstain"
            graph[a].add(b); graph[b].add(a)
    seen={START}; q=deque([START])
    while q:
        a=q.popleft()
        for b in graph[a]:
            if b not in seen: seen.add(b); q.append(b)
    hits=[k for k,v in TERMINALS.items() if v in seen]
    return hits[0] if len(hits)==1 else "abstain"
