"""Deterministic analytic and visual world for Cyclic Rim Splice."""
from __future__ import annotations
from collections import deque
import copy, random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1100, 800
N, CELL = 7, 68
BACKGROUND=(245,242,235); CARD=(255,255,252); INK=(42,47,57); MUTED=(84,91,104); GRID=(218,222,227); CARD_EDGE=(91,100,114)
DECISIONS=("A","B","C","D")
TERMINALS={"A":(0,0),"B":(0,6),"C":(6,0),"D":(6,6)}; START=(3,3)
PORTAL_CELLS={"P":((1,2),(5,4)),"Q":((1,4),(5,2))}
DIRECTIONS=((-1,0),(0,1),(1,0),(0,-1)); RIM_NAMES=("red","gold","green","blue")
RIM_COLORS={"red":(224,68,75),"gold":(244,177,43),"green":(50,166,101),"blue":(54,122,214)}
PORTAL_COLORS={"P":(25,151,171),"Q":(146,79,190)}
MARK_COLORS={"S":(215,70,65),"A":(247,174,47),"B":(45,119,204),"C":(76,166,88),"D":(150,82,191)}
GRID_ORIGIN=(62,231); FONT_PATH="/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"

def _font(size): return ImageFont.truetype(FONT_PATH,size)
def _canon(edge): return tuple(sorted((tuple(edge[0]),tuple(edge[1]))))
def _all_edges():
    blocked={x for xs in PORTAL_CELLS.values() for x in xs}; out=[]
    for r in range(N):
        for c in range(N):
            if (r,c) in blocked: continue
            for dr,dc in ((0,1),(1,0)):
                q=(r+dr,c+dc)
                if 0<=q[0]<N and 0<=q[1]<N and q not in blocked: out.append(((r,c),q))
    return out
ALL_EDGES=_all_edges()

def _portal_map(portals):
    links=[]
    for entries in portals.values():
        maps=[]
        for e in entries:
            r,c=e["cell"]; maps.append({tag:(r+DIRECTIONS[i][0],c+DIRECTIONS[i][1]) for i,tag in enumerate(e["rims"])})
        links.extend((maps[0][tag],maps[1][tag]) for tag in RIM_NAMES)
    return links

def _reachable(tracks,portals=None,hub=False):
    blocked={x for xs in PORTAL_CELLS.values() for x in xs}
    graph={x:set() for r in range(N) for c in range(N) if (x:=(r,c)) not in blocked}
    for a,b in map(_canon,tracks): graph[a].add(b); graph[b].add(a)
    if portals:
        if hub:
            for entries in portals.values():
                mouths=[]
                for e in entries:
                    r,c=e["cell"]; mouths.extend((r+dr,c+dc) for dr,dc in DIRECTIONS)
                mouths=[x for x in mouths if x in graph]
                for a in mouths: graph[a].update(x for x in mouths if x!=a)
        else:
            for a,b in _portal_map(portals):
                if a in graph and b in graph: graph[a].add(b); graph[b].add(a)
    seen={START}; q=deque([START])
    while q:
        a=q.popleft()
        for b in graph[a]:
            if b not in seen: seen.add(b); q.append(b)
    return seen
def _hits(seen): return [k for k,v in TERMINALS.items() if v in seen]

def _partition(rng):
    blocked={x for xs in PORTAL_CELLS.values() for x in xs}
    owner={START:"S",**{v:k for k,v in TERMINALS.items()},(0,3):"X",(6,3):"Y",(3,0):"Z",(3,6):"W"}
    while len(owner)<N*N-len(blocked):
        frontier=[]
        for a,label in owner.items():
            for dr,dc in DIRECTIONS:
                b=(a[0]+dr,a[1]+dc)
                if 0<=b[0]<N and 0<=b[1]<N and b not in blocked and b not in owner: frontier.append((b,label))
        if not frontier: raise RuntimeError("partition stalled")
        b,label=rng.choice(frontier); owner[b]=label
    return owner

def sample_scene(seed):
    rng=random.Random(int(seed)); target=DECISIONS[int(seed)%4]
    for _ in range(10000):
        owner=_partition(rng); tracks=[_canon(e) for e in ALL_EDGES if owner[e[0]]==owner[e[1]]]
        if not 18<=len(tracks)<=40: continue
        portals={}
        for symbol,cells in PORTAL_CELLS.items():
            entries=[]
            for cell in cells:
                rims=list(RIM_NAMES); rng.shuffle(rims); entries.append({"cell":list(cell),"rims":rims})
            portals[symbol]=entries
        if _hits(_reachable(tracks)): continue
        exact=_hits(_reachable(tracks,portals)); hub=_hits(_reachable(tracks,portals,True))
        if exact!=[target] or len(hub)<2 or target not in hub: continue
        rng.shuffle(tracks)
        encoded=[[list(e[1]),list(e[0])] if rng.getrandbits(1) else [list(e[0]),list(e[1])] for e in tracks]
        scene={"tracks":encoded,"portals":portals}
        if not is_quarantined(scene): return scene
    raise RuntimeError(f"could not sample accepted scene for seed {seed}")

def analytic_gold(scene):
    hits=_hits(_reachable(scene["tracks"],scene["portals"])); return hits[0] if len(hits)==1 else "abstain"
def margin(scene): return 1.0 if analytic_gold(scene) in DECISIONS else 0.0
def is_quarantined(scene):
    try:
        if set(scene)!={"tracks","portals"} or set(scene["portals"])!=set(PORTAL_CELLS): return True
        tracks=[_canon(e) for e in scene["tracks"]]
        if len(tracks)!=len(set(tracks)) or any(e not in ALL_EDGES for e in tracks): return True
        for symbol,entries in scene["portals"].items():
            if len(entries)!=2 or tuple(tuple(e["cell"]) for e in entries)!=PORTAL_CELLS[symbol]: return True
            if any(sorted(e["rims"])!=sorted(RIM_NAMES) for e in entries): return True
        if _hits(_reachable(tracks)): return True
        exact=_hits(_reachable(tracks,scene["portals"])); hub=_hits(_reachable(tracks,scene["portals"],True))
        return len(exact)!=1 or len(hub)<2 or exact[0] not in hub
    except Exception: return True
def latent_symmetries(scene):
    out=copy.deepcopy(scene); out["tracks"]=list(reversed(out["tracks"]))
    return [("reverse_stored_track_order",out)]

def _center(cell):
    r,c=cell; gx,gy=GRID_ORIGIN; return gx+c*CELL+CELL//2,gy+r*CELL+CELL//2
def _draw_portal(draw,cx,cy,symbol,rims):
    draw.rounded_rectangle((cx-24,cy-24,cx+24,cy+24),radius=6,fill=(236,238,241),outline=INK,width=3)
    boxes=((cx-10,cy-24,cx+10,cy-13),(cx+13,cy-10,cx+24,cy+10),(cx-10,cy+13,cx+10,cy+24),(cx-24,cy-10,cx-13,cy+10))
    for tag,box in zip(rims,boxes): draw.rectangle(box,fill=RIM_COLORS[tag],outline=INK,width=1)
    draw.ellipse((cx-10,cy-10,cx+10,cy+10),fill=PORTAL_COLORS[symbol],outline=INK,width=1)
    draw.text((cx,cy),symbol,fill="white",font=_font(14),anchor="mm")

    draw.text((843,628),"Every mark is part of the map.",fill=MUTED,font=_font(14),anchor="mm")
    return image
