"""Renderer and analytic gold for the colored open-box correspondence world.

The only answer-bearing property of a record is ``color`` paired with ``gap``.
The remaining record fields are rendered nuisance geometry.  Three panels use
the same five colors and the same multiset of gap directions; one candidate
preserves the color-to-gap map and the other permutes it.
"""

from __future__ import annotations

import copy
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

RENDERER_VERSION = "colored_gap_direction_correspondence-0.1.0"
WIDTH = 960
HEIGHT = 360
SUPERSAMPLE = 4
GAPS = ("UP", "RIGHT", "DOWN", "LEFT")
COLORS = ("red", "blue", "green", "purple", "orange")
PALETTE = {
    "red": (220, 50, 47),
    "blue": (37, 99, 235),
    "green": (22, 163, 74),
    "purple": (147, 51, 234),
    "orange": (234, 132, 15),
}
PANEL_ORIGINS = {
    "reference": (20, 52),
    "candidate_a": (340, 52),
    "candidate_b": (660, 52),
}
PANEL_SIZE = (280, 280)
SLOT_CENTERS = ((55, 92), (140, 92), (225, 92), (95, 205), (180, 205))
PANEL_TITLES = {"reference": "REFERENCE", "candidate_a": "A", "candidate_b": "B"}


def _font(scale: int = 1) -> ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", 16 * scale)
    except OSError:
        return ImageFont.load_default()


def _signature(panel: list[dict]) -> dict[str, str]:
    return {str(record["color"]): str(record["gap"]) for record in panel}


def _mismatch(reference: dict[str, str], panel: dict[str, str]) -> int:
    return sum(panel.get(color) != gap for color, gap in reference.items())


def _rectangles_for_gap(x: int, y: int, size: int, stroke: int, gap: str) -> list[tuple[int, int, int, int]]:
    """Return the three filled bars making an open square outline."""

    t = max(4, min(stroke, size // 3))
    last = x + size - 1
    bottom = y + size - 1
    bars = {
        "UP": [
            (x, y, x + t - 1, bottom),
            (last - t + 1, y, last, bottom),
            (x, bottom - t + 1, last, bottom),
        ],
        "RIGHT": [
            (x, y, last, y + t - 1),
            (x, bottom - t + 1, last, bottom),
            (x, y, x + t - 1, bottom),
        ],
        "DOWN": [
            (x, y, last, y + t - 1),
            (last - t + 1, y, last, bottom),
            (x, y, x + t - 1, bottom),
        ],
        "LEFT": [
            (x, y, last, y + t - 1),
            (x, bottom - t + 1, last, bottom),
            (last - t + 1, y, last, bottom),
        ],
    }
    return bars[gap]


def _draw_panel(draw: ImageDraw.ImageDraw, key: str, records: list[dict], scale: int) -> None:
    ox, oy = PANEL_ORIGINS[key]
    pw, ph = PANEL_SIZE
    draw.rectangle((ox * scale, oy * scale, (ox + pw) * scale, (oy + ph) * scale), fill=(248, 250, 252), outline=(203, 213, 225), width=2 * scale)
    for record in records:
        x = ox + int(record["x"])
        y = oy + int(record["y"])
        size = int(record["size"])
        stroke = int(record["stroke"])
        ink = PALETTE[str(record["color"])]
        for left, top, right, bottom in _rectangles_for_gap(x, y, size, stroke, str(record["gap"])):
            draw.rectangle((left * scale, top * scale, (right + 1) * scale - 1, (bottom + 1) * scale - 1), fill=ink)


def render(scene: dict) -> Image.Image:
    """Rasterize the three visible panels from a JSON-compatible scene."""

    image = Image.new("RGB", (WIDTH * SUPERSAMPLE, HEIGHT * SUPERSAMPLE), (255, 255, 255))
    draw = ImageDraw.Draw(image)
    font = _font(SUPERSAMPLE)
    for key in ("reference", "candidate_a", "candidate_b"):
        ox, oy = PANEL_ORIGINS[key]
        title = PANEL_TITLES[key]
        bbox = draw.textbbox((0, 0), title, font=font)
        title_w = bbox[2] - bbox[0]
        title_x = (ox + PANEL_SIZE[0] / 2) * SUPERSAMPLE - title_w / 2
        draw.text((int(title_x), 22 * SUPERSAMPLE), title, fill=(30, 41, 59), font=font)
    # Keep these accesses explicit so the protected rendered-field audit can
    # verify that every analytic label input is also visible in the raster.
    _draw_panel(draw, "reference", scene["reference"], SUPERSAMPLE)
    _draw_panel(draw, "candidate_a", scene["candidate_a"], SUPERSAMPLE)
    _draw_panel(draw, "candidate_b", scene["candidate_b"], SUPERSAMPLE)
    image = image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
    return image


def _panel_records(rng: random.Random, mapping: dict[str, str], geometries: list[tuple[int, int]]) -> list[dict]:
    slots = list(range(len(SLOT_CENTERS)))
    rng.shuffle(slots)
    geometry_order = list(range(len(geometries)))
    rng.shuffle(geometry_order)
    records = []
    for color in COLORS:
        slot = slots[len(records)]
        cx, cy = SLOT_CENTERS[slot]
        size, stroke = geometries[geometry_order[len(records)]]
        records.append(
            {
                "color": color,
                "gap": mapping[color],
                "x": int(round(cx - size / 2)),
                "y": int(round(cy - size / 2)),
                "size": int(size),
                "stroke": int(stroke),
            }
        )
    rng.shuffle(records)
    return records


def sample_scene(seed: int) -> dict:
    """Sample a balanced, answer-counterbalanced scene from one integer seed."""

    rng = random.Random(int(seed))
    gaps = [GAPS[index % len(GAPS)] for index in range(len(COLORS))]
    rng.shuffle(gaps)
    reference_mapping = dict(zip(COLORS, gaps))
    wrong_gaps = gaps[1:] + gaps[:1]
    wrong_mapping = dict(zip(COLORS, wrong_gaps))
    match_a = int(seed) % 2 == 0

    # A shared geometry multiset equalizes coarse ink and size cues.  Each panel
    # independently reassigns those geometries and slots to colors.
    geometries = [(rng.randint(50, 60), rng.randint(7, 9)) for _ in COLORS]
    reference = _panel_records(rng, reference_mapping, geometries)
    if match_a:
        candidate_a = _panel_records(rng, reference_mapping, geometries)
        candidate_b = _panel_records(rng, wrong_mapping, geometries)
    else:
        candidate_a = _panel_records(rng, wrong_mapping, geometries)
        candidate_b = _panel_records(rng, reference_mapping, geometries)
    return {
        "reference": reference,
        "candidate_a": candidate_a,
        "candidate_b": candidate_b,
    }


def analytic_gold(scene: dict) -> str:
    """Compare complete visible color-to-gap signatures and return ``A``/``B``."""

    reference = _signature(scene["reference"])
    score_a = _mismatch(reference, _signature(scene["candidate_a"]))
    score_b = _mismatch(reference, _signature(scene["candidate_b"]))
    if score_a == score_b:
        return "AMBIGUOUS"
    return "A" if score_a < score_b else "B"


def margin(scene: dict) -> float:
    """Return the positive gap between the two candidate mismatch scores."""

    reference = _signature(scene["reference"])
    score_a = _mismatch(reference, _signature(scene["candidate_a"]))
    score_b = _mismatch(reference, _signature(scene["candidate_b"]))
    return float(abs(score_a - score_b))


def _valid_panel(panel: list[dict]) -> bool:
    if len(panel) != len(COLORS):
        return False
    if {record.get("color") for record in panel} != set(COLORS):
        return False
    if any(record.get("gap") not in GAPS for record in panel):
        return False
    boxes = []
    for record in panel:
        try:
            x = int(record["x"])
            y = int(record["y"])
            size = int(record["size"])
            stroke = int(record["stroke"])
        except (KeyError, TypeError, ValueError):
            return False
        if x < 10 or y < 35 or x + size > PANEL_SIZE[0] - 10 or y + size > PANEL_SIZE[1] - 10:
            return False
        if not (42 <= size <= 70 and 4 <= stroke <= size // 3):
            return False
        boxes.append((x, y, x + size, y + size))
    for index, left in enumerate(boxes):
        for right in boxes[index + 1 :]:
            if min(left[2], right[2]) - max(left[0], right[0]) > 3 and min(left[3], right[3]) - max(left[1], right[1]) > 3:
                return False
    return True


def is_quarantined(scene: dict) -> bool:
    """Reject malformed, ambiguous, clipped, overlapping, or tied scenes."""

    panels = (scene.get("reference"), scene.get("candidate_a"), scene.get("candidate_b"))
    if any(not isinstance(panel, list) or not _valid_panel(panel) for panel in panels):
        return True
    reference = _signature(scene["reference"])
    score_a = _mismatch(reference, _signature(scene["candidate_a"]))
    score_b = _mismatch(reference, _signature(scene["candidate_b"]))
    return score_a == score_b


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Exercise the record-order permutation alias; rendering is non-overlapping."""

    twin = copy.deepcopy(scene)
    for key in ("reference", "candidate_a", "candidate_b"):
        twin[key] = list(reversed(twin[key]))
    return [("reverse_unordered_panel_records", twin)]
