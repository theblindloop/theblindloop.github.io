"""Independent pixel-only inverse arm for the seam-patch-completion world.

Receives ONLY the final rasterized PNG (and the public visual spec duplicated
below). It never imports the renderer, prompt, generator, verifier, or
analytic-gold code, and never reads scene data.

Public visual spec (the oracle's own specification):

  * a top band holds colored horizontal contour lines whose middle is blank
    (a window at x in [WX0, WX1]); each color appears as a segment on the left
    half and a segment on the right half of the band;
  * a horizontal divider separates the top band from a row of three framed
    candidate patches A, B, C; each patch is a rectangle holding one proposed
    horizontal segment per color;
  * the task: identify the patch A/B/C that would continue every contour
    across the seam (each color's proposed segment sits at the vertical
    position that continues that color's reference line).

The oracle recovers, from pixels alone: each reference color's line y, each
patch's proposed y per color, the per-color continuity, and finally the unique
patch label that continues every contour.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "seam_patch_completion-oracle-0.1.0"

BACKGROUND = (255, 255, 255)
PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (255, 0, 0),
    "green": (0, 180, 0),
    "blue": (0, 0, 255),
    "orange": (255, 140, 0),
    "purple": (150, 30, 220),
}

PATCH_CX = [250, 430, 610]
R_LO = 70
R_HI = 230
P_LO = 330
P_HI = 490
WX0 = 375
WX1 = 485
X_LEFT = 30
WIN = 110
tolerance_px = 4.0
CONTINUITY_TOL = 4.0


def _mapped(ref_y: float) -> float:
    return P_LO + (ref_y - R_LO)


def _nearest_names(pixels: np.ndarray) -> np.ndarray:
    names = list(PALETTE)
    rgbs = np.array([PALETTE[n] for n in names], dtype=np.float32)
    d = np.linalg.norm(pixels[..., None, :] - rgbs[None, None, :, :], axis=-1)
    return np.array(names)[d.argmin(axis=-1)]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.float32)
    h, _w = arr.shape[:2]
    if h < 100:
        raise ValueError("image too small to analyze")

    bg = (arr[..., 0] > 200) & (arr[..., 1] > 200) & (arr[..., 2] > 200)
    dark = arr.max(axis=2) < 110
    ink = ~(bg | dark)

    names = list(PALETTE)

    # --- Reference line y per color, from the left half of the top band. ---
    ref_y: dict[str, float] = {}
    left_region = arr[R_LO:R_HI, X_LEFT:WX0 - 4]
    left_ink = ink[R_LO:R_HI, X_LEFT:WX0 - 4]
    if not left_ink.any():
        raise ValueError("no reference contours found")
    lbl = _nearest_names(left_region)
    ys_grid = np.arange(R_LO, R_HI)[:, None] * np.ones(
        (1, left_region.shape[1]), dtype=np.float32
    )
    for name in names:
        m = (lbl == name) & left_ink
        if m.any():
            ref_y[name] = float(ys_grid[m].mean())

    # --- Per-patch proposed y per color. ---
    patch_y: list[dict[str, float]] = []
    for cx in PATCH_CX:
        x0 = cx - WIN // 2
        x1 = cx + WIN // 2
        region = arr[P_LO:P_HI, x0:x1]
        reg_ink = ink[P_LO:P_HI, x0:x1]
        lbl = _nearest_names(region)
        ys_grid = np.arange(P_LO, P_HI)[:, None] * np.ones(
            (1, region.shape[1]), dtype=np.float32
        )
        row: dict[str, float] = {}
        for name in names:
            m = (lbl == name) & reg_ink
            if m.any():
                row[name] = float(ys_grid[m].mean())
        patch_y.append(row)

    # --- Continuity: does a patch continue every color that the reference has? ---
    present = [name for name in names if name in ref_y]
    if not present:
        raise ValueError("no contour colors recoverable")
    candidates: list[int] = []
    for p, row in enumerate(patch_y):
        ok = True
        for name in present:
            if name not in row:
                ok = False
                break
            if abs(row[name] - _mapped(ref_y[name])) > CONTINUITY_TOL:
                ok = False
                break
        if ok:
            candidates.append(p)
    if len(candidates) != 1:
        raise ValueError(f"expected exactly one continuing patch, found {candidates}")
    return "ABC"[candidates[0]]
