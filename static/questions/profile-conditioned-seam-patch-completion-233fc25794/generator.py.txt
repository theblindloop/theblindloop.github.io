"""Seam-patch-completion world renderer (paper-semantic correspondence family).

A cross-panel structural-completion task. The top panel shows a band of colored
horizontal contour lines whose run is interrupted in the middle by a blank
vertical ``window``: on the left of the window and on the right of the window
each color's contour is visible, but the middle is missing. Below the divider a
row of three candidate ``patches`` (A, B, C) each proposes a way to fill the
window. Exactly one patch makes *every* colored contour continue continuously
across the seam: for each color it places that color's segment at the vertical
position that continues that color's left and right fragments. The question
asks which candidate completes the window so every contour continues across the
seam.

Latent scene fields (every field below is drawn into the raster):

    colors   : the K contour color names (order is the correspondence key)
    ref_ys   : for each color, the reference line's y position (top band)
    patch_ys : for each candidate, for each color, the proposed continuation
               y position within that patch (bottom band)

The correct patch is the one whose proposed positions continue every contour;
it is recoverable from the pixels alone and not stored as a hidden label.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "seam_patch_completion-0.1.0"

BACKGROUND = "white"
DIVIDER = (30, 30, 30)
PATCH_BORDER = (40, 40, 40)
PATCH_FILL = (248, 248, 248)

PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (255, 0, 0),
    "green": (0, 180, 0),
    "blue": (0, 0, 255),
    "orange": (255, 140, 0),
    "purple": (150, 30, 220),
}

# Public geometry (constant; part of the visual spec).
W = 860
H = 560
K_LINES = 3
X_LEFT = 30
X_RIGHT = 830
WIN = 110
WX0 = 375
WX1 = 485
R_LO = 70
R_HI = 230
P_LO = 330
P_HI = 490
DIV = 300
DIV_THICK = 4
PATCH_CX = [250, 430, 610]
PATCH_CY = (P_LO + P_HI) // 2
STROKE = 6
MARGIN_MIN = 14.0
QUARANTINE_FLOOR = 10.0
TOL = 3

_NAMES = list(PALETTE)


def _mapped(ref_y: float) -> float:
    """Translate a reference-band y into the same-relative patch-band y."""
    return P_LO + (ref_y - R_LO)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    colors = rng.sample(_NAMES, K_LINES)
    # Distinct reference y positions with a clear minimum gap.
    ref_ys: list[float] = []
    for _ in range(K_LINES):
        for _attempt in range(1000):
            y = rng.uniform(R_LO + 14, R_HI - 14)
            if all(abs(y - existing) >= 22 for existing in ref_ys):
                ref_ys.append(y)
                break
        else:
            ref_ys.append(R_LO + 14 + 25 * len(ref_ys))
    correct = rng.randrange(3)
    correct_local = [_mapped(y) for y in ref_ys]
    patch_ys: list[list[float]] = []
    for p in range(3):
        if p == correct:
            patch_ys.append(list(correct_local))
        else:
            local = list(correct_local)
            nd = rng.randint(1, K_LINES)
            displaced = rng.sample(range(K_LINES), nd)
            for idx in displaced:
                # Move far enough to be clearly non-continuing.
                for _attempt in range(2000):
                    target = rng.uniform(P_LO + 14, P_HI - 14)
                    if abs(target - correct_local[idx]) >= MARGIN_MIN:
                        local[idx] = target
                        break
                else:
                    local[idx] = correct_local[idx] + MARGIN_MIN + 1
                # clamp to a drawable range
                local[idx] = max(P_LO + 8, min(P_HI - 8, local[idx]))
            patch_ys.append(local)
    return {
        "colors": colors,
        "ref_ys": [round(y, 3) for y in ref_ys],
        "patch_ys": [[round(v, 3) for v in row] for row in patch_ys],
    }


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (W, H), BACKGROUND)
    draw = ImageDraw.Draw(img)
    colors = scene["colors"]
    ref_ys = scene["ref_ys"]
    patch_ys = scene["patch_ys"]

    # Reference contour lines: left and right halves around the blank window.
    for y_c, color in zip(ref_ys, colors, strict=True):
        y = int(round(y_c))
        fill = PALETTE[color]
        draw.line([X_LEFT, y, WX0 - 2, y], fill=fill, width=STROKE)
        draw.line([WX1 + 2, y, X_RIGHT, y], fill=fill, width=STROKE)

    # Divider between the reference band and the candidate patches.
    draw.rectangle([0, DIV, W, DIV + DIV_THICK], fill=DIVIDER)
    # Light reference-band background frame for the top panel.
    draw.rectangle([X_LEFT - 8, R_LO - 8, X_RIGHT + 8, R_HI + 8],
                   outline=(150, 150, 150), width=2)

    # Wait: draw the frame behind lines would cover them; draw frame after lines
    # only as an outline is fine since it is just a thin border.
    draw.rectangle([X_LEFT - 8, R_LO - 8, X_RIGHT + 8, R_HI + 8],
                   outline=(150, 150, 150), width=2)
    # Window edges (subtle vertical marks delimiting the missing span).
    draw.line([WX0, R_LO, WX0, R_HI], fill=(120, 120, 120), width=2)
    draw.line([WX1, R_LO, WX1, R_HI], fill=(120, 120, 120), width=2)

    # Candidate patches: a framed panel per slot with proposed segments.
    for p, cx in enumerate(PATCH_CX):
        x0 = cx - WIN // 2
        x1 = cx + WIN // 2
        draw.rectangle([x0, P_LO, x1, P_HI], fill=PATCH_FILL,
                       outline=PATCH_BORDER, width=2)
        for idx, color in enumerate(colors):
            y = int(round(patch_ys[p][idx]))
            draw.line([x0 + 2, y, x1 - 2, y], fill=PALETTE[color], width=STROKE)
        # Slot letter label.
        draw.text((cx - 4, P_HI + 8), "ABC"[p], fill=(60, 60, 60))
    return img


def _correct_index(scene: dict) -> int:
    ref_ys = scene["ref_ys"]
    patch_ys = scene["patch_ys"]
    correct_local = [_mapped(y) for y in ref_ys]
    for p in range(len(patch_ys)):
        if all(abs(patch_ys[p][idx] - correct_local[idx]) <= TOL
               for idx in range(len(ref_ys))):
            return p
    raise ValueError("no patch continues every contour")


def analytic_gold(scene: dict) -> str:
    """Decision: the label of the candidate patch that completes the window."""
    return "ABC"[_correct_index(scene)]


def margin(scene: dict) -> float:
    """Runner-up margin: the minimum displacement of any distractor segment from
    the position that would continue its contour. Large -> unambiguous."""
    ref_ys = scene["ref_ys"]
    patch_ys = scene["patch_ys"]
    correct_local = [_mapped(y) for y in ref_ys]
    best = float("inf")
    for p in range(len(patch_ys)):
        if p == _correct_index(scene):
            continue
        for idx in range(len(ref_ys)):
            d = abs(patch_ys[p][idx] - correct_local[idx])
            if d > TOL:
                best = min(best, d)
    return float(best) if best < float("inf") else QUARANTINE_FLOOR


def is_quarantined(scene: dict) -> bool:
    """Quarantine a scene whose runner-up is uncomfortably close to continuing."""
    return margin(scene) < QUARANTINE_FLOOR


def latent_symmetries(scene: dict) -> list:
    """not_applicable: every label-influencing field is rendered, so the raster
    uniquely determines the completion; no pixel-identical transform is declared."""
    return []
