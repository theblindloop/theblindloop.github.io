"""Sharpest-corner selection world: which colored dot marks the most acute vertex.

Five colored dots are drawn on a white canvas at the five corners of one
irregular convex five-sided polygon (pentagon) that the dots together outline.
Exactly one corner is noticeably sharper (its interior angle is clearly the
smallest, most acute) than the other four. The decision is the color of the dot
that sits at that sharpest corner.

The global Gestalt structure is the closed convex outline the five dots reveal
as a group; the decision selects the single vertex with the most acute interior
angle of that recovered polygon. The analytic gold owns latent-scene sampling,
rasterization, and the structural gold. The independent pixel-only inverse arm
(``oracle.py``) recovers the same decision from the final PNG alone: it locates
each colored dot, orders them around their shared centroid into the outline,
measures every interior angle, and selects the color of the most acute corner.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (255, 255, 255)

DOT_RADIUS = 11

# Five marker colors; one dot per vertex of the outlined pentagon.
MARKER_COLORS = {
    "red": (230, 50, 50),
    "green": (50, 190, 90),
    "blue": (50, 90, 230),
    "purple": (170, 60, 200),
    "orange": (240, 140, 30),
}
MARKER_NAMES = ["red", "green", "blue", "purple", "orange"]

# Sampled polygon center is kept central so every corner dot stays inside the
# canvas with a clear margin to every border.
_CX_MIN = 195
_CX_MAX = 317
_R_MIN = 130.0
_R_MAX = 150.0

# Two adjacent-arc gaps around the sharp vertex are made large so its interior
# angle is clearly the most acute; the other three gaps share the remainder.
_BIG_MIN = 95.0
_BIG_MAX = 118.0

# Quarantine band on the angular margin (deg): gap between the sharpest corner
# interior angle and the second-sharpest. Scenes below this are ambiguous.
QUAR_MIN = 14.0


def interior_angles(pts):
    """Interior angles (degrees) at each vertex of a polygon ordered around it."""
    n = len(pts)
    angs = []
    for i in range(n):
        a = pts[i]
        b = pts[(i + 1) % n]
        c0 = pts[(i - 1) % n]
        v1 = (a[0] - c0[0], a[1] - c0[1])
        v2 = (b[0] - a[0], b[1] - a[1])
        d1 = math.hypot(*v1)
        d2 = math.hypot(*v2)
        if d1 == 0 or d2 == 0:
            angs.append(180.0)
            continue
        cos = -(v1[0] * v2[0] + v1[1] * v2[1]) / (d1 * d2)
        cos = max(-1.0, min(1.0, cos))
        angs.append(math.degrees(math.acos(cos)))
    return angs


def angular_order(pts):
    """Return vertex indices ordered counterclockwise around their centroid."""
    cx0 = sum(p[0] for p in pts) / len(pts)
    cy0 = sum(p[1] for p in pts) / len(pts)
    return sorted(range(len(pts)), key=lambda i: math.atan2(pts[i][1] - cy0, pts[i][0] - cx0))


def _convex(v):
    n = len(v)
    sgn = 0
    for i in range(n):
        a = v[i]
        b = v[(i + 1) % n]
        c = v[(i + 2) % n]
        cr = (b[0] - a[0]) * (c[1] - b[1]) - (b[1] - a[1]) * (c[0] - b[0])
        if sgn == 0:
            sgn = 1 if cr > 0 else -1
        elif cr * sgn <= 0:
            return False
    return True


def _sep_ok(pts, minimum=42.0):
    for i in range(len(pts)):
        for j in range(i + 1, len(pts)):
            if math.hypot(pts[i][0] - pts[j][0], pts[i][1] - pts[j][1]) < minimum:
                return False
    return True


def sample_scene(seed: int) -> dict:
    """Sample one latent scene deterministically from ``seed``."""
    rng = np.random.default_rng(seed)
    for _attempt in range(4000):
        cx = float(rng.uniform(_CX_MIN, _CX_MAX))
        cy = float(rng.uniform(_CX_MIN, _CX_MAX))
        R = float(rng.uniform(_R_MIN, _R_MAX))
        sharp = int(rng.integers(0, 5))
        big1 = float(rng.uniform(_BIG_MIN, _BIG_MAX))
        big2 = float(rng.uniform(_BIG_MIN, _BIG_MAX))
        rest = (360.0 - big1 - big2) / 3.0
        gaps = [rest, rest, rest, rest, rest]
        gaps[sharp] = big1
        gaps[(sharp + 4) % 5] = big2
        rot = float(rng.uniform(0.0, 360.0))
        va = []
        cur = rot
        for i in range(5):
            va.append(cur)
            cur += gaps[i]
        pts = [
            [cx + R * math.cos(math.radians(va[i])), cy + R * math.sin(math.radians(va[i]))]
            for i in range(5)
        ]
        if not _convex(pts):
            continue
        order = angular_order(pts)
        ordered = [pts[i] for i in order]
        angs = sorted(interior_angles(ordered))
        if angs[1] - angs[0] < QUAR_MIN:
            continue
        pad = 42.0
        if not all(pad <= p[0] <= WIDTH - pad and pad <= p[1] <= HEIGHT - pad for p in pts):
            continue
        if not _sep_ok(pts):
            continue
        colors = list(MARKER_NAMES)
        rng.shuffle(colors)
        return {
            "vertices": [[float(x), float(y)] for (x, y) in pts],
            "colors": colors,
        }
    raise RuntimeError("could not sample a valid sharpest-corner scene")


def render(scene: dict) -> Image.Image:
    """Rasterize the latent scene (a deterministic RGB PNG)."""
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    d = ImageDraw.Draw(img)
    for (vx, vy), color in zip(scene["vertices"], scene["colors"]):
        d.ellipse(
            [vx - DOT_RADIUS, vy - DOT_RADIUS, vx + DOT_RADIUS, vy + DOT_RADIUS],
            fill=MARKER_COLORS[color],
        )
    return img


def _sharp_index(vertices):
    """Original vertex index of the most acute corner of the outlined pentagon."""
    order = angular_order(vertices)
    ordered = [vertices[i] for i in order]
    angs = interior_angles(ordered)
    pos = int(angs.index(min(angs)))
    return order[pos]


def analytic_gold(scene: dict) -> str:
    """Structural gold: the color of the dot at the sharpest corner."""
    k = _sharp_index(scene["vertices"])
    return scene["colors"][k]


def margin(scene: dict) -> float:
    """Decision margin (deg): second-sharpest minus sharpest interior angle."""
    order = angular_order(scene["vertices"])
    ordered = [scene["vertices"][i] for i in order]
    angs = sorted(interior_angles(ordered))
    return float(angs[1] - angs[0])


def is_quarantined(scene: dict) -> bool:
    """Quarantine scenes whose sharpest corner margin is too small."""
    return margin(scene) < QUAR_MIN


def latent_symmetries(scene: dict) -> list:
    """The scene is injectively rendered; no latent aliases are declared."""
    return []
