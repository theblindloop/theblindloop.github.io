"""Independent pixel-only inverse arm for the sharpest-corner world.

Recovers which colored dot marks the sharpest (most acute) corner of the
irregular pentagon that the five colored dots outline, from the final PNG alone.
It never sees the latent scene, the analytic gold, or the renderer code. It
segments each color dot by palette, takes each dot's centroid, orders the five
centroids around their shared centroid into the outline, measures every interior
angle, and selects the color of the most acute vertex.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

BACKGROUND = (255, 255, 255)
MARKER_COLORS = {
    "red": (230, 50, 50),
    "green": (50, 190, 90),
    "blue": (50, 90, 230),
    "purple": (170, 60, 200),
    "orange": (240, 140, 30),
}
MARKER_NAMES = ["red", "green", "blue", "purple", "orange"]

_TH = 40
_MIN_PIXELS = 40


def _mask(arr, rgb):
    r = arr[..., 0]
    g = arr[..., 1]
    b = arr[..., 2]
    return (np.abs(r - rgb[0]) < _TH) & (np.abs(g - rgb[1]) < _TH) & (np.abs(b - rgb[2]) < _TH)


def _centroid(mask):
    coords = np.argwhere(mask)
    if len(coords) < _MIN_PIXELS:
        return None
    return (float(coords[:, 1].mean()), float(coords[:, 0].mean()))


def _angular_order(pts):
    cx0 = sum(p[0] for p in pts) / len(pts)
    cy0 = sum(p[1] for p in pts) / len(pts)
    return sorted(range(len(pts)), key=lambda i: math.atan2(pts[i][1] - cy0, pts[i][0] - cx0))


def _interior_angles(pts):
    n = len(pts)
    angs = []
    for i in range(n):
        a = pts[i]
        b = pts[(i + 1) % n]
        c0 = pts[(i - 1) % n]
        v1 = (a[0] - c0[0], a[1] - c0[1])
        v2 = (b[0] - a[0], b[1] - a[1])
        d1 = math.hypot(*v1)
        d2 = math.hypot(*v2)
        if d1 == 0 or d2 == 0:
            angs.append(180.0)
            continue
        cos = -(v1[0] * v2[0] + v1[1] * v2[1]) / (d1 * d2)
        cos = max(-1.0, min(1.0, cos))
        angs.append(math.degrees(math.acos(cos)))
    return angs


def decision_from_image(image: Image.Image) -> str:
    """The color of the dot at the sharpest corner of the outlined pentagon."""
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)

    centers = []
    colors = []
    for color in MARKER_NAMES:
        c = _centroid(_mask(arr, MARKER_COLORS[color]))
        if c is None:
            continue
        centers.append(c)
        colors.append(color)
    if len(centers) < 5:
        return MARKER_NAMES[0]

    order = _angular_order(centers)
    ordered = [centers[i] for i in order]
    angs = _interior_angles(ordered)
    pos = int(angs.index(min(angs)))
    return colors[order[pos]]
