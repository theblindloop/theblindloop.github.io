from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw


WIDTH = 512
HEIGHT = 512
BACKGROUND = (248, 248, 244)
PALETTES = (
    ((32, 72, 118), (244, 203, 116)),
    ((91, 50, 86), (245, 210, 132)),
    ((25, 92, 91), (246, 196, 126)),
    ((76, 67, 44), (240, 192, 142)),
)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    return {
        "answer_class": "cycle" if int(seed) % 2 == 0 else "tree",
        "cx": rng.randint(105, 407),
        "cy": rng.randint(105, 407),
        "rotation": rng.choice([0, 90, 180, 270]),
        "scale": rng.choice([0.92, 0.96, 1.0]),
        "palette_index": rng.randrange(len(PALETTES)),
        "variant": rng.randrange(4),
    }


def _turn(point: tuple[float, float], quarter_degrees: int) -> tuple[float, float]:
    x, y = point
    turns = (int(quarter_degrees) // 90) % 4
    for _ in range(turns):
        x, y = -y, x
    return x, y


def _map(scene: dict, point: tuple[float, float]) -> tuple[int, int]:
    x, y = _turn(point, scene["rotation"])
    s = float(scene["scale"])
    return round(scene["cx"] + s * x), round(scene["cy"] + s * y)


def _network_segments(scene: dict) -> list[tuple[tuple[float, float], tuple[float, float]]]:
    variant = int(scene["variant"])
    wobble = (-3, -1, 1, 3)[variant]
    if scene["answer_class"] == "cycle":
        # One rectangular cycle plus four short terminal stems.  The four
        # terminals are held constant across classes; only cycle rank changes.
        corners = [(-25, -23), (25, -23), (25, 23), (-25, 23)]
        segments = [
            (corners[0], corners[1]),
            (corners[1], corners[2]),
            (corners[2], corners[3]),
            (corners[3], corners[0]),
            ((0, -23), (wobble, -42)),
            ((25, 0), (42, -wobble)),
            ((0, 23), (-wobble, 42)),
            ((-25, 0), (-42, wobble)),
        ]
    else:
        # A loop-free four-terminal tree with comparable span and channel
        # thickness.  Offset junctions prevent a fixed plus-sign template.
        a = (-9 + wobble, -5)
        b = (11 + wobble, 7)
        segments = [
            (a, b),
            (a, (-31, -32)),
            (a, (-39, 25)),
            (b, (34, -29)),
            (b, (38, 31)),
        ]
    return segments


def render(scene: dict) -> Image.Image:
    # Explicit reads keep every declared latent visually accountable.
    answer_class = scene["answer_class"]
    cx = scene["cx"]
    cy = scene["cy"]
    rotation = scene["rotation"]
    scale = scene["scale"]
    palette_index = scene["palette_index"]
    variant = scene["variant"]
    _ = (answer_class, cx, cy, rotation, scale, variant)

    plaque, channel = PALETTES[int(palette_index)]
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    radius = round(55 * float(scale))
    corner = max(10, round(13 * float(scale)))
    draw.rounded_rectangle(
        (int(cx) - radius, int(cy) - radius, int(cx) + radius, int(cy) + radius),
        radius=corner,
        fill=plaque,
    )
    width = max(9, round(12 * float(scale)))
    terminal_radius = max(5, round(width * 0.54))
    segments = _network_segments(scene)
    for start, end in segments:
        draw.line([_map(scene, start), _map(scene, end)], fill=channel, width=width)
    # Repaint graph vertices and the four degree-one terminals as rounded nodes,
    # eliminating line-join cracks without changing graph topology.
    points = {point for segment in segments for point in segment}
    for point in points:
        x, y = _map(scene, point)
        draw.ellipse(
            (x - terminal_radius, y - terminal_radius, x + terminal_radius, y + terminal_radius),
            fill=channel,
        )
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["answer_class"])


def margin(scene: dict) -> float:
    # Minimum nominal half-width of the channel, comfortably above a pixel.
    return 5.0 * float(scene["scale"])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
