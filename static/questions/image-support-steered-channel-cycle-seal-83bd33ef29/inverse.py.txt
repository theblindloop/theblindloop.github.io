from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        output.append(np.asarray(points, dtype=np.int32))
    return output


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    # Recover the two inks by broad photometric roles, not by a known palette.
    # The outer border supplies the image's background estimate.  Every sampled
    # plaque is dark, while every channel is bright but chromatically separated
    # from that background.  This also makes destructive white ablations visible
    # as missing evidence rather than as a newly invented color cluster.
    border = np.concatenate((array[0], array[-1], array[:, 0], array[:, -1]), axis=0)
    background = np.median(border.astype(np.float32), axis=0)
    lightness = array.astype(np.float32).mean(axis=2)
    distance = np.linalg.norm(array.astype(np.float32) - background, axis=2)
    plaque = lightness < 155.0
    channel = (lightness >= 155.0) & (distance > 35.0)

    channel_components = _components(channel)
    if len(channel_components) != 1 or len(channel_components[0]) < 750:
        raise ValueError("cream channel is absent, too small, or disconnected")
    ys, xs = channel_components[0][:, 0], channel_components[0][:, 1]
    if int(xs.max() - xs.min()) < 62 or int(ys.max() - ys.min()) < 62:
        raise ValueError("channel span is incomplete")

    substantial_plaque = [part for part in _components(plaque) if len(part) >= 90]
    if not substantial_plaque or max(len(part) for part in substantial_plaque) < 4000:
        raise ValueError("outer plaque is absent or badly damaged")
    islands = [part for part in substantial_plaque if len(part) < 1600]
    if len(islands) == 1:
        return "cycle"
    if len(islands) == 0 and len(substantial_plaque) == 1:
        return "tree"
    raise ValueError("plaque topology is ambiguous")
