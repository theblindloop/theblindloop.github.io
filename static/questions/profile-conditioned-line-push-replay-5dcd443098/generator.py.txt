"""Deterministic sampler and renderer for collision-propagating line pushes."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 820, 520
BACKGROUND = (248, 247, 242)
INK = (30, 32, 36)
QUERY = (218, 35, 178)
PALETTES = (
    {"a": (220, 58, 48), "b": (35, 116, 210), "c": (32, 162, 92)},
    {"a": (235, 118, 32), "b": (44, 138, 194), "c": (103, 78, 191)},
    {"a": (202, 54, 126), "b": (19, 151, 151), "c": (102, 139, 35)},
)


def _replay(positions: dict[str, int], commands: list[dict[str, str]]) -> tuple[dict[str, int], dict[str, int]]:
    pos = {name: int(slot) for name, slot in positions.items()}
    stats = {"direct": 0, "chain": 0, "blocked": 0}
    for command in commands:
        name = command["crate"]
        step = 1 if command["direction"] == "right" else -1
        occupied = {slot: crate for crate, slot in pos.items()}
        cursor = pos[name] + step
        chain = [name]
        while 1 <= cursor <= 4 and cursor in occupied:
            chain.append(occupied[cursor])
            cursor += step
        if not 1 <= cursor <= 4:
            stats["blocked"] += 1
            continue
        for crate in reversed(chain):
            pos[crate] += step
        stats["direct" if len(chain) == 1 else "chain"] += 1
    return pos, stats


def sample_scene(seed: int) -> dict:
    """Balance the final queried bay by seed modulo four."""
    rng = random.Random(int(seed))
    target = int(seed) % 4 + 1
    palette_variant = rng.randrange(len(PALETTES))
    shift = rng.choice((-9, -4, 0, 4, 9))
    decorations = [rng.randrange(3) for _ in range(6)]
    names = ("a", "b", "c")
    for _ in range(80000):
        occupied = rng.sample((1, 2, 3, 4), 3)
        rng.shuffle(occupied)
        positions = dict(zip(names, occupied))
        query = rng.choice(names)
        commands = [
            {"crate": rng.choice(names), "direction": rng.choice(("left", "right"))}
            for _index in range(6)
        ]
        final, stats = _replay(positions, commands)
        query_moves = sum(
            1 for before, after in zip(
                [positions] + [_replay(positions, commands[:i])[0] for i in range(1, 6)],
                [_replay(positions, commands[:i])[0] for i in range(1, 7)],
            ) if before[query] != after[query]
        )
        if final[query] == target and stats["chain"] >= 1 and stats["blocked"] >= 1 and query_moves >= 1:
            return {
                "positions": positions,
                "query": query,
                "commands": commands,
                "palette_variant": palette_variant,
                "shift": shift,
                "decorations": decorations,
            }
    raise RuntimeError(f"could not sample answer stratum {target}")


def _dot_row(draw: ImageDraw.ImageDraw, center_x: int, y: int, count: int) -> None:
    start = center_x - 12 * (count - 1) / 2
    for index in range(count):
        x = int(round(start + 12 * index))
        draw.ellipse((x - 4, y - 4, x + 4, y + 4), fill=INK)


def _arrow(draw: ImageDraw.ImageDraw, cx: int, cy: int, direction: str) -> None:
    sign = 1 if direction == "right" else -1
    draw.rectangle((cx - 25, cy - 6, cx + 25, cy + 6), fill=INK)
    tip = cx + sign * 36
    base = cx + sign * 10
    draw.polygon(((tip, cy), (base, cy - 22), (base, cy + 22)), fill=INK)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default(size=15)
    palette = PALETTES[int(scene["palette_variant"])]
    shift = int(scene["shift"])

    draw.text((34 + shift, 22), "START: bays 1 to 4", fill=INK, font=font)
    bay_lefts = [45 + shift + 190 * index for index in range(4)]
    bays = []
    for number, left in enumerate(bay_lefts, start=1):
        box = (left, 68, left + 135, 212)
        bays.append(box)
        draw.rectangle(box, outline=INK, width=4)
        _dot_row(draw, left + 67, 91, number)

    for name, slot in scene["positions"].items():
        box = bays[int(slot) - 1]
        cx = (box[0] + box[2]) // 2
        cy = 157
        draw.ellipse((cx - 27, cy - 27, cx + 27, cy + 27), fill=palette[name])
        if name == scene["query"]:
            draw.ellipse((cx - 38, cy - 38, cx + 38, cy + 38), outline=QUERY, width=7)

    draw.text((34 + shift, 244), "PUSH CARDS: read left to right", fill=INK, font=font)
    for index, command in enumerate(scene["commands"]):
        left = 24 + shift + 130 * index
        top, right, bottom = 278, left + 108, 475
        draw.rectangle((left, top, right, bottom), outline=INK, width=4)
        cx = (left + right) // 2
        draw.ellipse((cx - 38, 323, cx + 38, 399), fill=palette[command["crate"]])
        _arrow(draw, cx, 361, command["direction"])
        for mark in range(int(scene["decorations"][index])):
            x = left + 24 + mark * 18
            draw.polygon(((x, 444), (x + 5, 449), (x, 454), (x - 5, 449)), fill=(180, 182, 180))
    return image


def analytic_gold(scene: dict) -> str:
    final, _ = _replay(scene["positions"], scene["commands"])
    return str(final[scene["query"]])


def margin(scene: dict) -> float:
    _ = (scene["positions"], scene["query"], scene["commands"])
    return 12.0


def is_quarantined(scene: dict) -> bool:
    _ = (scene["positions"], scene["query"], scene["commands"])
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
