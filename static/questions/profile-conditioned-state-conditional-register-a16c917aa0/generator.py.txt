"""Value-conditional three-state register accumulator (state tracking).

The image shows a single three-valued register carried through K=5 ordered
panels (left to right). Each panel draws the register's current value as a
colored disk at the panel center: dark gray = 0, teal = 1, magenta = 2. An
orange ring around the leftmost token marks the start of the tracked chain.
Between each pair of consecutive panels sits an operator glyph:

  * a GREEN circle  -> add one to the current value (modulo 3);
  * a BLUE square   -> subtract one from the current value (modulo 3);
  * a RED triangle  -> conditional: if the current value is 0 it becomes 1,
                       otherwise it becomes 2.

The latent scene stores only ``values`` (each panel's running register value,
already consistent with the operators) and ``ops`` (the operator kinds drawn
between the panels). The decision is the final value, which is a
state-dependent, non-additive composition of the whole ordered sequence:
replayed from the ring-marked start value through every operator in order.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 850
HEIGHT = 200
PANEL_W = 120
PANEL_H = 150
GAP = 50
MARGIN_LEFT = 30
PITCH = PANEL_W + GAP
TOKEN_R = 24
GLYPH_R = 14
HALF_H = 75
CENTER_Y = HEIGHT // 2

BG = (255, 255, 255)
BORDER = (200, 200, 200)
RING_MARK = (240, 150, 20)
GREEN = (0, 150, 40)
BLUE = (30, 60, 210)
RED = (210, 40, 40)

VALUE_COLORS = {
    0: (105, 105, 105),
    1: (0, 150, 120),
    2: (190, 0, 160),
}
VALUE_NAMES = {0: "0", 1: "1", 2: "2"}

PANEL_COUNT = 5


def _panel_center(index: int) -> tuple[int, int]:
    return (MARGIN_LEFT + PITCH * index + PANEL_W // 2, CENTER_Y)


def _glyph_center(index: int) -> tuple[int, int]:
    left = MARGIN_LEFT + PITCH * index + PANEL_W
    return (left + GAP // 2, CENTER_Y)


def apply_op(value: int, op: str) -> int:
    """Deterministic register transition for a single operator."""
    if op == "inc":
        return (value + 1) % 3
    if op == "dec":
        return (value - 1) % 3
    if op == "cond":
        return 1 if value == 0 else 2
    raise ValueError(f"unknown operator {op!r}")


def sample_scene(seed: int) -> dict:
    """Deterministic latent scene from an integer seed."""
    rng = random.Random(seed)
    initial = rng.randint(0, 2)
    ops = [rng.choice(["inc", "dec", "cond"]) for _ in range(PANEL_COUNT - 1)]
    values = [initial]
    value = initial
    for op in ops:
        value = apply_op(value, op)
        values.append(value)
    return {"values": values, "ops": ops}


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)
    values = scene["values"]
    ops = scene["ops"]

    for i in range(PANEL_COUNT):
        left = MARGIN_LEFT + PITCH * i
        top = CENTER_Y - HALF_H
        draw.rectangle([left, top, left + PANEL_W, top + PANEL_H], outline=BORDER, width=3)
        cx, cy = _panel_center(i)
        box = [cx - TOKEN_R, cy - TOKEN_R, cx + TOKEN_R, cy + TOKEN_R]
        draw.ellipse(box, fill=VALUE_COLORS[values[i]])

    cx, cy = _panel_center(0)
    draw.ellipse(
        [cx - TOKEN_R - 7, cy - TOKEN_R - 7, cx + TOKEN_R + 7, cy + TOKEN_R + 7],
        outline=RING_MARK,
        width=4,
    )

    for i, op in enumerate(ops):
        gx, gy = _glyph_center(i)
        if op == "inc":
            draw.ellipse([gx - GLYPH_R, gy - GLYPH_R, gx + GLYPH_R, gy + GLYPH_R], fill=GREEN)
        elif op == "dec":
            draw.rectangle(
                [gx - GLYPH_R, gy - GLYPH_R, gx + GLYPH_R, gy + GLYPH_R], fill=BLUE
            )
        else:
            draw.polygon(
                [(gx, gy - GLYPH_R), (gx - GLYPH_R, gy + GLYPH_R), (gx + GLYPH_R, gy + GLYPH_R)],
                fill=RED,
            )
    return img


def analytic_gold(scene: dict) -> str:
    value = scene["values"][0]
    for op in scene["ops"]:
        value = apply_op(value, op)
    return VALUE_NAMES[value]


def margin(scene: dict) -> float:
    return 10.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    """No nontrivial render-invariant latent symmetry: every latent field
    (``values``, ``ops``) directly changes the pixels -- ``values`` controls the
    drawn value-colored tokens and ``ops`` controls the drawn operator glyphs.
    Any other permutation of panel slots would reorder the image and the ordered
    sequence, changing the pixels."""
    return []
