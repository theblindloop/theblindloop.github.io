"""Scene-free inverse arm for the visible rainbow-band count."""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageColor

# Public visual vocabulary, intentionally duplicated rather than imported.
_PALETTE = np.array(
    [
        ImageColor.getrgb(value)
        for value in (
            "#d62828", "#f77f00", "#fcbf49", "#2a9d3f", "#00a6a6",
            "#1976d2", "#5138a5", "#9c27b0", "#e83e8c",
        )
    ],
    dtype=np.float32,
)
_MIN_RUN = 6


def _rgb_saturation(arr: np.ndarray) -> np.ndarray:
    high = arr.max(axis=2)
    low = arr.min(axis=2)
    return (high - low) / np.maximum(high, 1.0)


def _runs(values: list[int]) -> list[tuple[int, int]]:
    runs: list[tuple[int, int]] = []
    for value in values:
        if value < 0:
            continue
        if runs and runs[-1][0] == value:
            runs[-1] = (value, runs[-1][1] + 1)
        else:
            runs.append((value, 1))
    return runs


def measure(image: Image.Image) -> tuple[int, float]:
    """Return (band count, runner-up pixel margin) from raster pixels only."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    sat = _rgb_saturation(arr)
    bright = arr.max(axis=2)
    colored = (sat > 0.42) & (bright > 90)
    ys, xs = np.nonzero(colored)
    if len(xs) < 1000:
        raise ValueError("no measurable rainbow-colored region")

    # The rainbow center is inferred from its saturated support, never assumed.
    center_x = int(round((float(xs.min()) + float(xs.max())) / 2.0))
    x0, x1 = max(0, center_x - 2), min(arr.shape[1], center_x + 3)
    radial = np.median(arr[:, x0:x1, :], axis=1)
    distances = np.linalg.norm(radial[:, None, :] - _PALETTE[None, :, :], axis=2)
    labels = distances.argmin(axis=1)
    nearest = distances.min(axis=1)
    labels = np.where(nearest < 92.0, labels, -1)

    # Only substantial consecutive color runs are bands. LANCZOS transition
    # pixels can form 1-2 px nearest-palette islands, which are discarded.
    significant = [(label, length) for label, length in _runs(labels.tolist()) if length >= _MIN_RUN]
    if not significant:
        raise ValueError("no substantial colored arc bands")
    lengths = [length for _, length in significant]
    if len(significant) < 4 or len(significant) > 9:
        raise ValueError(f"implausible rainbow band count {len(significant)}")
    return len(significant), float(min(lengths))


def decision_from_image(image: Image.Image) -> str:
    count, _ = measure(image)
    return str(count)
