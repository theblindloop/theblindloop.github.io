"""Deterministic Pillow renderer for a prior-conflicting rainbow-band count."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 512
PALETTE = (
    "#d62828",  # red
    "#f77f00",  # orange
    "#fcbf49",  # gold
    "#2a9d3f",  # green
    "#00a6a6",  # teal
    "#1976d2",  # blue
    "#5138a5",  # indigo
    "#9c27b0",  # violet
    "#e83e8c",  # pink
)
BACKGROUND = "#eaf7ff"


def sample_scene(seed: int) -> dict:
    """Sample balanced counts 4..9; seven is the familiar-prior control."""
    rng = random.Random(int(seed) ^ 0x5A17B0)
    count = 4 + (int(seed) % 6)
    colors = list(PALETTE)
    rng.shuffle(colors)
    outer_radius = rng.randint(174, 194)
    total_thickness = rng.randint(116, 126)
    center_x = rng.randint(244, 268)
    baseline_y = rng.randint(348, 372)
    return {
        "band_count": count,
        "palette": colors,
        "center_x": center_x,
        "baseline_y": baseline_y,
        "outer_radius": outer_radius,
        "total_thickness": total_thickness,
        "cloud_variant": rng.randrange(4),
        "width": WIDTH,
        "height": HEIGHT,
    }


def _cloud(draw: ImageDraw.ImageDraw, x: float, y: float, scale: float) -> None:
    fill = "#ffffff"
    outline = "#b8d7e8"
    blobs = [
        (x - 42 * scale, y - 4 * scale, 35 * scale),
        (x - 9 * scale, y - 22 * scale, 43 * scale),
        (x + 30 * scale, y - 6 * scale, 34 * scale),
    ]
    for cx, cy, radius in blobs:
        draw.ellipse(
            [cx - radius, cy - radius, cx + radius, cy + radius],
            fill=fill,
            outline=outline,
            width=max(2, round(2 * scale)),
        )
    draw.rounded_rectangle(
        [x - 70 * scale, y - 8 * scale, x + 66 * scale, y + 34 * scale],
        radius=18 * scale,
        fill=fill,
        outline=outline,
        width=max(2, round(2 * scale)),
    )


def render(scene: dict) -> Image.Image:
    """Render nested, touching semicircular bands with constant total colored area."""
    scale = 4
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    cx = float(scene["center_x"]) * scale
    cy = float(scene["baseline_y"]) * scale
    outer = float(scene["outer_radius"]) * scale
    total = float(scene["total_thickness"]) * scale
    count = int(scene["band_count"])
    colors = list(scene["palette"][:count])

    # Draw outer-to-inner nested half-disks. Later half-disks expose equal-width
    # annular bands; the final background half-disk opens the rainbow's center.
    for index, color in enumerate(colors):
        radius = outer - index * total / count
        draw.pieslice(
            [cx - radius, cy - radius, cx + radius, cy + radius],
            start=180,
            end=360,
            fill=color,
        )
    inner = outer - total
    draw.pieslice(
        [cx - inner, cy - inner, cx + inner, cy + inner],
        start=180,
        end=360,
        fill=BACKGROUND,
    )

    # Clouds establish the familiar rainbow visual world while leaving the top
    # radial evidence untouched. Variant changes cloud scale/order only.
    variant = int(scene["cloud_variant"])
    cloud_scale = (0.76, 0.84, 0.92, 1.0)[variant]
    _cloud(draw, cx - outer + 12 * scale, cy - 3 * scale, cloud_scale * scale)
    _cloud(draw, cx + outer - 12 * scale, cy - 3 * scale, (1.02 - variant * 0.04) * scale)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(int(scene["band_count"]))


def margin(scene: dict) -> float:
    """Runner-up margin: ideal radial pixels supporting the thinnest band."""
    return float(scene["total_thickness"]) / int(scene["band_count"])


def is_quarantined(scene: dict) -> bool:
    per_band = margin(scene)
    outer = float(scene["outer_radius"])
    cx = float(scene["center_x"])
    cy = float(scene["baseline_y"])
    width, height = int(scene["width"]), int(scene["height"])
    clipped = cx - outer < 4 or cx + outer > width - 4 or cy - outer < 4 or cy > height - 4
    return per_band < 12.0 or clipped


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Exercise a real JSON numeric-representation alias (int versus float)."""
    twin = dict(scene)
    twin["center_x"] = float(scene["center_x"])
    twin["baseline_y"] = float(scene["baseline_y"])
    twin["outer_radius"] = float(scene["outer_radius"])
    twin["total_thickness"] = float(scene["total_thickness"])
    return [("equivalent_numeric_representation", twin)]
