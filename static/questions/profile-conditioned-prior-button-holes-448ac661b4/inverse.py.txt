"""Pixel-only inverse arm: count the button holes from the raster alone.

This module receives only the rendered PNG and never imports renderer,
prompts, generate, verify, or analytic-gold code, and it accepts no latent
scene data. It classifies each pixel to a fixed two-colour world palette by
nearest Euclidean colour, then counts the number of 4-connected
background-coloured components that are fully enclosed inside the button body
(those not connected to the image border). Each such enclosed blob is one hole.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

# Fixed world palette (must match renderer constants).
BACKGROUND = (252, 252, 250)
BUTTON_COLOR = (120, 150, 190)

_PALETTE = [BACKGROUND, BUTTON_COLOR]
_BG_LABEL = 0


def _label_pixels(arr: np.ndarray) -> np.ndarray:
    dists = np.stack(
        [np.linalg.norm(arr - np.array(c, dtype=np.float32), axis=-1) for c in _PALETTE],
        axis=-1,
    )
    return dists.argmin(axis=-1)


def _count_components(mask: np.ndarray) -> int:
    """Count 4-connected components of the boolean mask (BFS flood fill)."""
    h, w = mask.shape
    visited = np.zeros((h, w), dtype=bool)
    count = 0
    ys, xs = np.nonzero(mask)
    for y, x in zip(ys.tolist(), xs.tolist()):
        if visited[y, x]:
            continue
        count += 1
        stack = [(y, x)]
        visited[y, x] = True
        while stack:
            cy, cx = stack.pop()
            for ny, nx in (
                (cy - 1, cx),
                (cy + 1, cx),
                (cy, cx - 1),
                (cy, cx + 1),
            ):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    stack.append((ny, nx))
    return count


def decision_from_image(image: Image.Image) -> str:
    """Return the counted number of enclosed button holes as a string."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label_pixels(arr)
    bg = labels == _BG_LABEL
    h, w = bg.shape

    # Flood-fill the outer background connected to the image border.
    outer = np.zeros((h, w), dtype=bool)
    stack = []
    for y in range(h):
        for x in (0, w - 1):
            if bg[y, x] and not outer[y, x]:
                outer[y, x] = True
                stack.append((y, x))
    for y in (0, h - 1):
        for x in range(w):
            if bg[y, x] and not outer[y, x]:
                outer[y, x] = True
                stack.append((y, x))
    while stack:
        cy, cx = stack.pop()
        for ny, nx in (
            (cy - 1, cx),
            (cy + 1, cx),
            (cy, cx - 1),
            (cy, cx + 1),
        ):
            if 0 <= ny < h and 0 <= nx < w and bg[ny, nx] and not outer[ny, nx]:
                outer[ny, nx] = True
                stack.append((ny, nx))

    holes = bg & ~outer
    return str(_count_components(holes))
