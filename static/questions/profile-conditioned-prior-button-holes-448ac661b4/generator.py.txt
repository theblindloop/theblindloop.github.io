"""Prior-conflict button world: count the holes punched in a round button.

The latent scene places a round button on a plain background and punches a
given number ``k`` of circular holes (cutouts that reveal the background colour)
in a symmetric cross/ring layout on the button face. A familiar canonical prior
asserts that a classic shirt button is sewn with FOUR holes; the counterfactual
renders a non-canonical hole count and the decision must be read from the
enclosed background blobs in the pixels, never from the prior. The pixel oracle
counts the background-coloured disks that are fully enclosed inside the button
body (i.e. background components not connected to the image border).

Latent alias mode is ``not_applicable``: every latent field (the hole count k,
button radius, cutout radius, centre, and canvas) is directly reflected in the
raster, so the rendering is injective in the latent scene and there is no
render-invariant alias. The canonical-prior name "button" is a
matched-control/counterfactual manipulation of the *question*, not of any
latent field.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

CANVAS = (460, 460)
BACKGROUND = (252, 252, 250)
BUTTON_COLOR = (120, 150, 190)  # denim-blue button body

# Hole layout distance from the button centre (px).
_HOLE_D = 55.0
# Supported off-quarantine hole counts. The canonical prior is 4; 2, 3 and 5
# are the counterfactuals that contradict it.
SUPPORTED_K = (2, 3, 4, 5)

_MIN_HOLE_R = 6.0    # a hole smaller than this is not a visible cutout
_MIN_EDGE_CLR = 8.0  # a hole this close to the rim is ambiguous
_MIN_HOLE_GAP = 8.0  # two holes this close may merge and be miscounted


def _hole_positions(k: int, d: float) -> list[tuple[float, float]]:
    """Symmetric, non-overlapping hole centres for a given count k."""
    if k == 2:
        return [(-d, 0.0), (d, 0.0)]
    if k == 3:
        return [(0.0, -d), (-d, d), (d, d)]
    if k == 4:
        return [(-d, -d), (d, -d), (-d, d), (d, d)]
    if k == 5:
        return [(0.0, 0.0), (-d, 0.0), (d, 0.0), (0.0, -d), (0.0, d)]
    return []


def sample_scene(seed: int) -> dict:
    """Deterministic scene dict from an integer seed."""
    rng = random.Random(seed)
    k = SUPPORTED_K[rng.randrange(len(SUPPORTED_K))]
    r = rng.uniform(148.0, 162.0)
    rh = rng.uniform(12.0, 15.0)
    return {
        "k": k,
        "r": round(r, 3),
        "rh": round(rh, 3),
        "cx": CANVAS[0] / 2.0,
        "cy": CANVAS[1] / 2.0,
        "canvas": list(CANVAS),
    }


def render(scene: dict) -> Image.Image:
    """Render the round button with k punched holes to an RGB image."""
    w, h = scene["canvas"]
    cx, cy = float(scene["cx"]), float(scene["cy"])
    r = float(scene["r"])
    rh = float(scene["rh"])
    img = Image.new("RGB", (int(w), int(h)), BACKGROUND)
    draw = ImageDraw.Draw(img)
    # Button body (solid disk).
    draw.ellipse([cx - r, cy - r, cx + r, cy + r], fill=BUTTON_COLOR)
    # Punch holes by revealing the background colour inside the body.
    for hx, hy in _hole_positions(int(scene["k"]), _HOLE_D):
        draw.ellipse(
            [cx + hx - rh, cy + hy - rh, cx + hx + rh, cy + hy + rh],
            fill=BACKGROUND,
        )
    return img


def analytic_gold(scene: dict) -> str:
    """Decision = the visible number of holes, as a string."""
    return str(scene["k"])


def margin(scene: dict) -> float:
    """Verifiability margin = the smallest visible clearance around any hole.

    Computed as the minimum of the hole size, the rim clearance, and the
    gap between neighbouring holes.
    """
    r = float(scene["r"])
    rh = float(scene["rh"])
    centres = _hole_positions(int(scene["k"]), _HOLE_D)
    if not centres:
        return 0.0
    max_radius = max(math.hypot(hx, hy) for hx, hy in centres)
    rim_clearance = r - max_radius - rh
    inter = float("inf")
    for i in range(len(centres)):
        for j in range(i + 1, len(centres)):
            gap = math.hypot(
                centres[i][0] - centres[j][0], centres[i][1] - centres[j][1]
            ) - 2.0 * rh
            inter = min(inter, gap)
    return float(min(rh, rim_clearance, inter))


def is_quarantined(scene: dict) -> bool:
    """Quarantine ambiguous resolutions (too-thin holes or tiny clearances)."""
    r = float(scene["r"])
    rh = float(scene["rh"])
    centres = _hole_positions(int(scene["k"]), _HOLE_D)
    if not centres or rh < _MIN_HOLE_R:
        return True
    max_radius = max(math.hypot(hx, hy) for hx, hy in centres)
    if r - max_radius - rh < _MIN_EDGE_CLR:
        return True
    for i in range(len(centres)):
        for j in range(i + 1, len(centres)):
            gap = math.hypot(
                centres[i][0] - centres[j][0], centres[i][1] - centres[j][1]
            ) - 2.0 * rh
            if gap < _MIN_HOLE_GAP:
                return True
    return False


def latent_symmetries(scene: dict):
    """not_applicable: empty, the latent rendering is injective."""
    return []
