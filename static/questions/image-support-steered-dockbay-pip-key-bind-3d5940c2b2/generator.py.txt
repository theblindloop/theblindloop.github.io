"""Renderer for the dock-bay pip-key binding world.

One cream board carries seven small slate-outlined boxes placed on the cells of
a checkerboard inside one medium square block of the canvas, plus purely
decorative black dots and coloured bars on the open board.

Each box holds between one and six small black square dots: that count is the
box's key. Four boxes (BAYS) additionally carry a solid coloured band across
the top of their inside; three boxes (CRATES) carry no band. Every crate key is
also a bay key, so each crate docks into the bay with the same key and exactly
one bay is left unmatched. The decision is that unmatched bay's band colour.

The decisive evidence is therefore the dot cluster of every one of the seven
boxes plus the band of the winning bay: seven separated components spread over
one medium block of the canvas. Removing any single box breaks the matching.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "dockbay_pip_key_bind-0.1.0"

CANVAS = 960
GRID = 8
CELL = CANVAS // GRID
BLOCK = 4  # the checkerboard lives in a BLOCK x BLOCK cell square

BOX = 46
OUTLINE = 3
BOX_INSET = (CELL - BOX) // 2  # 37 px: keeps every box clear of neighbouring
# ablation windows, which reach at most CELL // 4 = 30 px into a neighbour cell

PIP = 5
PIP_X0 = 4
PIP_Y0 = 16
PIP_DX = 10
PIP_DY = 13
PIP_COLS = 4
PIP_ROWS = 2

BAND_TOP = 3
BAND_HEIGHT = 10

DOT = 5
BAR_W = 22
BAR_H = 8
BOARD_MARGIN = 16
CLUTTER_CLEARANCE = 26

BOARD = (240, 234, 220)
SLATE = (58, 70, 86)
INK = (24, 26, 30)
COLORS: dict[str, tuple[int, int, int]] = {
    "red": (198, 56, 48),
    "blue": (72, 76, 208),
    "green": (44, 152, 66),
    "purple": (132, 54, 158),
}
COLOR_NAMES = ("red", "blue", "green", "purple")

N_BAYS = 4
N_CRATES = 3
KEY_POOL = (1, 2, 3, 4, 5, 6)
QUARANTINE_MARGIN = 3.0
TIGHT_EVERY = 5  # deterministic near-boundary coverage


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------
def _block_cells(rng: random.Random) -> tuple[list[list[int]], int, int]:
    row0 = rng.randrange(0, GRID - BLOCK + 1)
    col0 = rng.randrange(0, GRID - BLOCK + 1)
    parity = rng.randrange(2)
    cells = [
        [dr, dc]
        for dr in range(BLOCK)
        for dc in range(BLOCK)
        if (dr + dc) % 2 == parity
    ]
    # keep the two block-diagonal extremes so the support span stays regional
    if parity == 0:
        protected = {(0, 0), (BLOCK - 1, BLOCK - 1)}
    else:
        protected = {(0, BLOCK - 1), (BLOCK - 1, 0)}
    droppable = [cell for cell in cells if tuple(cell) not in protected]
    dropped = rng.choice(droppable)
    kept = [cell for cell in cells if cell != dropped]
    return [[row0 + dr, col0 + dc] for dr, dc in kept], row0, col0


def _pip_slots(rng: random.Random, key: int, tight: bool) -> list[list[int]]:
    slots = [[i, j] for j in range(PIP_ROWS) for i in range(PIP_COLS)]
    chosen = rng.sample(slots, key)
    placed: list[list[int]] = []
    for i, j in chosen:
        placed.append(
            [
                PIP_X0 + PIP_DX * i + rng.randint(-1, 1),
                PIP_Y0 + PIP_DY * j + rng.randint(-1, 1),
            ]
        )
    if tight and len(placed) >= 2:
        # squeeze one horizontally adjacent pair towards a 1 px gap
        order = sorted(range(len(placed)), key=lambda k: (placed[k][1], placed[k][0]))
        for a, b in zip(order, order[1:]):
            pa, pb = placed[a], placed[b]
            if abs(pa[1] - pb[1]) <= 1 and 0 < pb[0] - pa[0] <= PIP_DX + 2:
                pb[0] = pa[0] + PIP + 1
                pb[1] = pa[1]
                break
    return placed


def _box_origin(cell: list[int]) -> tuple[int, int]:
    return cell[1] * CELL + BOX_INSET, cell[0] * CELL + BOX_INSET


def _clear_of_boxes(x0: int, y0: int, w: int, h: int, boxes: list[dict]) -> bool:
    for box in boxes:
        bx, by = _box_origin(box["cell"])
        gap_x = max(bx - (x0 + w), x0 - (bx + BOX))
        gap_y = max(by - (y0 + h), y0 - (by + BOX))
        if max(gap_x, gap_y) < CLUTTER_CLEARANCE:
            return False
    return True


def _decoration(rng: random.Random, boxes: list[dict]) -> list[dict]:
    items: list[dict] = []
    wanted = [("dot", DOT, DOT)] * rng.randint(10, 16)
    wanted += [("bar", BAR_W, BAR_H)] * rng.randint(4, 7)
    for kind, w, h in wanted:
        for _ in range(80):
            x0 = rng.randint(BOARD_MARGIN, CANVAS - BOARD_MARGIN - w)
            y0 = rng.randint(BOARD_MARGIN, CANVAS - BOARD_MARGIN - h)
            if not _clear_of_boxes(x0, y0, w, h, boxes):
                continue
            if any(
                max(
                    max(other["x"] - (x0 + w), x0 - (other["x"] + other["w"])),
                    max(other["y"] - (y0 + h), y0 - (other["y"] + other["h"])),
                )
                < 8
                for other in items
            ):
                continue
            items.append(
                {
                    "kind": kind,
                    "x": x0,
                    "y": y0,
                    "w": w,
                    "h": h,
                    "color": rng.choice(COLOR_NAMES) if kind == "bar" else "ink",
                }
            )
            break
    return items


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    tight = seed % TIGHT_EVERY == 0
    cells, row0, col0 = _block_cells(rng)
    rng.shuffle(cells)

    roles = ["bay"] * N_BAYS + ["crate"] * N_CRATES
    rng.shuffle(roles)

    bay_keys = rng.sample(KEY_POOL, N_BAYS)
    unmatched_slot = rng.randrange(N_BAYS)
    crate_keys = [key for index, key in enumerate(bay_keys) if index != unmatched_slot]
    rng.shuffle(crate_keys)

    # the answer colour is cycled from the seed so every class appears, while
    # geometry, keys, positions and clutter stay independently sampled
    target = COLOR_NAMES[seed % len(COLOR_NAMES)]
    bay_colors = [name for name in COLOR_NAMES if name != target]
    rng.shuffle(bay_colors)
    bay_colors.insert(unmatched_slot, target)

    boxes: list[dict] = []
    bay_index = 0
    crate_index = 0
    for cell, role in zip(cells, roles):
        if role == "bay":
            key = bay_keys[bay_index]
            color = bay_colors[bay_index]
            bay_index += 1
        else:
            key = crate_keys[crate_index]
            color = ""
            crate_index += 1
        boxes.append(
            {
                "cell": cell,
                "role": role,
                "key": key,
                "color": color,
                "pips": _pip_slots(rng, key, tight),
            }
        )

    boxes.sort(key=lambda box: (box["cell"][0], box["cell"][1]))
    scene = {
        "canvas": CANVAS,
        "block": [row0, col0],
        "boxes": boxes,
        "decoration": _decoration(rng, boxes),
        "tight": tight,
    }
    return scene


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (scene["canvas"], scene["canvas"]), BOARD)
    draw = ImageDraw.Draw(image)

    for item in sorted(
        scene["decoration"], key=lambda d: (d["y"], d["x"], d["kind"], d["color"])
    ):
        fill = INK if item["kind"] == "dot" else COLORS[item["color"]]
        draw.rectangle(
            [item["x"], item["y"], item["x"] + item["w"] - 1, item["y"] + item["h"] - 1],
            fill=fill,
        )

    for box in sorted(scene["boxes"], key=lambda b: (b["cell"][0], b["cell"][1])):
        x0, y0 = _box_origin(box["cell"])
        draw.rectangle([x0, y0, x0 + BOX - 1, y0 + BOX - 1], fill=BOARD, outline=SLATE, width=OUTLINE)
        if box["role"] == "bay":
            bx = x0 + OUTLINE
            by = y0 + BAND_TOP
            draw.rectangle(
                [bx, by, x0 + BOX - 1 - OUTLINE, by + BAND_HEIGHT - 1],
                fill=COLORS[box["color"]],
            )
        for px, py in sorted(box["pips"], key=lambda p: (p[1], p[0])):
            draw.rectangle([x0 + px, y0 + py, x0 + px + PIP - 1, y0 + py + PIP - 1], fill=INK)
    return image


# --------------------------------------------------------------------------
# analytic gold, margin, quarantine
# --------------------------------------------------------------------------
def _bays_and_crates(scene: dict) -> tuple[list[dict], list[dict]]:
    bays = [box for box in scene["boxes"] if box["role"] == "bay"]
    crates = [box for box in scene["boxes"] if box["role"] == "crate"]
    return bays, crates


def analytic_gold(scene: dict) -> str:
    bays, crates = _bays_and_crates(scene)
    crate_keys = [crate["key"] for crate in crates]
    unmatched = [bay for bay in bays if bay["key"] not in crate_keys]
    if len(bays) != N_BAYS or len(crates) != N_CRATES:
        raise ValueError("scene does not carry four bays and three crates")
    if len({bay["key"] for bay in bays}) != N_BAYS or len(set(crate_keys)) != N_CRATES:
        raise ValueError("keys are not distinct within their role")
    if not set(crate_keys) <= {bay["key"] for bay in bays}:
        raise ValueError("a crate key has no bay")
    if len(unmatched) != 1:
        raise ValueError("the unmatched bay is not unique")
    return str(unmatched[0]["color"])


def _pip_clearance(box: dict) -> float:
    pips = box["pips"]
    best = float(BOX)
    for index, (ax, ay) in enumerate(pips):
        for bx, by in pips[index + 1 :]:
            gap = max(abs(ax - bx) - PIP, abs(ay - by) - PIP)
            best = min(best, float(gap))
    return best


def margin(scene: dict) -> float:
    boxes = scene["boxes"]
    values = [_pip_clearance(box) for box in boxes]

    for index, box in enumerate(boxes):
        ax, ay = _box_origin(box["cell"])
        for other in boxes[index + 1 :]:
            bx, by = _box_origin(other["cell"])
            values.append(
                float(max(max(bx - (ax + BOX), ax - (bx + BOX)), max(by - (ay + BOX), ay - (by + BOX))))
            )
        for item in scene["decoration"]:
            values.append(
                float(
                    max(
                        max(item["x"] - (ax + BOX), ax - (item["x"] + item["w"])),
                        max(item["y"] - (ay + BOX), ay - (item["y"] + item["h"])),
                    )
                )
            )
    return float(min(values))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Stored box, pip and decoration lists are consumed as sets.

    `render` sorts each of them into a canonical drawing order and the drawn
    shapes are pairwise disjoint, so reversing any stored order re-draws a
    byte-identical raster and cannot move the analytic gold.
    """
    reversed_boxes = []
    for box in scene["boxes"]:
        twin = dict(box)
        twin["pips"] = list(reversed(box["pips"]))
        reversed_boxes.append(twin)
    return [
        ("decoration_order", {**scene, "decoration": list(reversed(scene["decoration"]))}),
        ("box_order", {**scene, "boxes": list(reversed(scene["boxes"]))}),
        ("pip_order", {**scene, "boxes": reversed_boxes}),
    ]
