"""Pixel-only inverse arm for the dock-bay pip-key binding world.

The oracle receives one rasterized PNG plus the public question contract and
nothing else. It imports no renderer, prompt, generator, verifier or gold code
and never sees a latent scene.

Pipeline:

1. quantize every pixel to the declared flat palette (board, slate outline,
   ink, and the four band colours); too much unreadable colour is refused;
2. label 4-connected slate components and keep the ones that are hollow
   rectangular rings clear of the image border - those are the boxes;
3. inside each box, measure the coloured band (a box with one band colour is a
   BAY, a box with none is a CRATE) and label the ink dots to read its key;
4. check the declared combinatorics: distinct bay keys, distinct crate keys,
   every crate key present among the bays, one crate fewer than bays, and
   exactly one bay left without a crate;
5. answer with that bay's band colour.

Any structural violation raises OracleAbstention, so a damaged, ablated or
ambiguous raster is refused rather than guessed. Ink dots and coloured bars
lying on the open board are never inside a box and so never counted.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

BOARD = (240, 234, 220)
SLATE = (58, 70, 86)
INK = (24, 26, 30)
BAND_COLORS: tuple[tuple[str, tuple[int, int, int]], ...] = (
    ("red", (198, 56, 48)),
    ("blue", (72, 76, 208)),
    ("green", (44, 152, 66)),
    ("purple", (132, 54, 158)),
)
PALETTE: tuple[tuple[str, tuple[int, int, int]], ...] = (
    ("board", BOARD),
    ("slate", SLATE),
    ("ink", INK),
) + BAND_COLORS

TOLERANCE = 40
MAX_UNKNOWN = 200

MIN_BOX_SIDE = 20
MAX_BOX_SIDE = 200
MAX_RING_FILL = 0.60
BORDER_CLEARANCE = 4
MIN_BOX_COUNT = 3

MIN_BAND_AREA = 120
MAX_BAND_STRAY = 8
MIN_DOT_AREA = 9
MAX_DOT_AREA = 40
MAX_DOT_SIDE = 9
MAX_KEY = 12


class OracleAbstention(Exception):
    """Raised when the image does not present the declared structure."""


def _quantize(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int32)
    flat = array.reshape(-1, 3)
    best = np.zeros(flat.shape[0], dtype=np.int32)
    best_distance = np.full(flat.shape[0], 1 << 30, dtype=np.int64)
    for index, (_, rgb) in enumerate(PALETTE):
        delta = flat - np.asarray(rgb, dtype=np.int32)
        distance = (delta * delta).sum(axis=1)
        better = distance < best_distance
        best_distance = np.where(better, distance, best_distance)
        best = np.where(better, index, best)
    unknown = best_distance > TOLERANCE * TOLERANCE
    if int(unknown.sum()) > MAX_UNKNOWN:
        raise OracleAbstention("too many pixels are outside the declared palette")
    best = np.where(unknown, 0, best)
    return best.reshape(array.shape[:2])


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """4-connected components of a boolean mask, as arrays of (y, x) points."""
    height, width = mask.shape
    labels = np.zeros((height, width), dtype=np.int32)
    out: list[list[tuple[int, int]]] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if labels[y, x]:
            continue
        out.append([])
        current = len(out)
        labels[y, x] = current
        stack = [(y, x)]
        while stack:
            cy, cx = stack.pop()
            out[-1].append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not labels[ny, nx]:
                    labels[ny, nx] = current
                    stack.append((ny, nx))
    return [np.asarray(points, dtype=np.int32) for points in out]


def _index(name: str) -> int:
    for position, (label, _) in enumerate(PALETTE):
        if label == name:
            return position
    raise KeyError(name)


def _boxes(labels: np.ndarray) -> list[tuple[int, int, int, int]]:
    height, width = labels.shape
    found: list[tuple[int, int, int, int]] = []
    for points in _components(labels == _index("slate")):
        y0, x0 = int(points[:, 0].min()), int(points[:, 1].min())
        y1, x1 = int(points[:, 0].max()), int(points[:, 1].max())
        side_y, side_x = y1 - y0 + 1, x1 - x0 + 1
        if side_x < MIN_BOX_SIDE or side_y < MIN_BOX_SIDE:
            continue
        if side_x > MAX_BOX_SIDE or side_y > MAX_BOX_SIDE:
            raise OracleAbstention("a slate component is not a small box outline")
        if (
            y0 < BORDER_CLEARANCE
            or x0 < BORDER_CLEARANCE
            or y1 >= height - BORDER_CLEARANCE
            or x1 >= width - BORDER_CLEARANCE
        ):
            raise OracleAbstention("a box outline touches the image border")
        if len(points) > MAX_RING_FILL * side_x * side_y:
            raise OracleAbstention("a slate component is solid, not a box outline")
        if labels[(y0 + y1) // 2, (x0 + x1) // 2] == _index("slate"):
            raise OracleAbstention("a box outline has no open inside")
        found.append((y0, x0, y1, x1))
    if len(found) < MIN_BOX_COUNT:
        raise OracleAbstention("fewer than three boxes are visible")
    return sorted(found)


def _thickness(labels: np.ndarray, box: tuple[int, int, int, int]) -> int:
    y0, x0, y1, x1 = box
    column = (x0 + x1) // 2
    slate = _index("slate")
    depth = 0
    while y0 + depth <= y1 and labels[y0 + depth, column] == slate:
        depth += 1
    if depth < 1 or depth > 8:
        raise OracleAbstention("a box outline has an unreadable thickness")
    return depth


def _read_box(labels: np.ndarray, box: tuple[int, int, int, int]) -> tuple[str, int]:
    y0, x0, y1, x1 = box
    depth = _thickness(labels, box)
    inside = labels[y0 + depth : y1 + 1 - depth, x0 + depth : x1 + 1 - depth]
    if inside.size == 0:
        raise OracleAbstention("a box has no readable inside")
    if int((inside == _index("slate")).sum()) > 0:
        raise OracleAbstention("slate intrudes into a box inside")

    band = ""
    for name, _ in BAND_COLORS:
        area = int((inside == _index(name)).sum())
        if area >= MIN_BAND_AREA:
            if band:
                raise OracleAbstention("a box carries two band colours")
            band = name
        elif area > MAX_BAND_STRAY:
            raise OracleAbstention("a box carries an unreadable partial band")

    key = 0
    for points in _components(inside == _index("ink")):
        span_y = int(points[:, 0].max() - points[:, 0].min() + 1)
        span_x = int(points[:, 1].max() - points[:, 1].min() + 1)
        if span_y > MAX_DOT_SIDE or span_x > MAX_DOT_SIDE:
            raise OracleAbstention("an ink blob inside a box is not a single dot")
        if not MIN_DOT_AREA <= len(points) <= MAX_DOT_AREA:
            raise OracleAbstention("an ink blob inside a box is not dot sized")
        key += 1
    if key == 0:
        raise OracleAbstention("a box carries no dots")
    if key > MAX_KEY:
        raise OracleAbstention("a box carries an implausible number of dots")
    return band, key


def decision_from_image(image: Image.Image) -> str:
    labels = _quantize(image)
    bays: list[tuple[str, int]] = []
    crates: list[int] = []
    for box in _boxes(labels):
        band, key = _read_box(labels, box)
        if band:
            bays.append((band, key))
        else:
            crates.append(key)

    if len(bays) < 2:
        raise OracleAbstention("fewer than two bays are visible")
    if len(crates) != len(bays) - 1:
        raise OracleAbstention("the crate count is not one less than the bay count")
    bay_keys = [key for _, key in bays]
    if len(set(bay_keys)) != len(bay_keys):
        raise OracleAbstention("two bays share a key")
    if len(set(crates)) != len(crates):
        raise OracleAbstention("two crates share a key")
    if len({name for name, _ in bays}) != len(bays):
        raise OracleAbstention("two bays share a band colour")
    if not set(crates) <= set(bay_keys):
        raise OracleAbstention("a crate key has no bay")
    unmatched = [name for name, key in bays if key not in crates]
    if len(unmatched) != 1:
        raise OracleAbstention("the unmatched bay is not unique")
    return unmatched[0]
