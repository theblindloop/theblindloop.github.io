"""Renderer and analytic label for color-labeled Delaunay-face correspondence.

Each panel contains six separate colored disks.  The informative structure is
the set of three-color faces in the Delaunay triangulation of the disk centers.
The image renderer and the inverse oracle intentionally implement the geometry
independently.
"""

from __future__ import annotations

import itertools
import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "delaunay-face-correspondence-0.1.0"
WIDTH, HEIGHT = 1080, 390
PANEL_CENTERS = ((180, 224), (540, 224), (900, 224))
PANEL_BOXES = ((28, 96, 332, 354), (388, 96, 692, 354), (748, 96, 1052, 354))
PANEL_X_RANGES = ((28, 332), (388, 692), (748, 1052))
PANEL_Y_RANGE = (96, 354)
COLORS = {
    "red": (221, 55, 48),
    "blue": (45, 112, 184),
    "green": (43, 153, 91),
    "orange": (235, 137, 45),
    "purple": (133, 79, 177),
    "teal": (31, 157, 157),
}
COLOR_ORDER = tuple(COLORS)
BACKGROUND = (250, 250, 247)
INK = (35, 39, 44)

# These are deliberately non-symmetric, well-separated six-point templates.
# Their generic position gives the raster inverse comfortable circumcircle
# margins while the sampler still varies the visible geometry.
_TEMPLATES = (
    ((-86.0, -42.0), (-28.0, -82.0), (65.0, -67.0), (-78.0, 33.0), (-4.0, 16.0), (73.0, 63.0)),
    ((-79.0, -62.0), (8.0, -86.0), (78.0, -25.0), (-72.0, 37.0), (0.0, 13.0), (60.0, 75.0)),
    ((-88.0, -17.0), (-48.0, -77.0), (43.0, -74.0), (-71.0, 67.0), (11.0, 23.0), (80.0, 52.0)),
    ((-76.0, -76.0), (2.0, -51.0), (82.0, -58.0), (-84.0, 30.0), (-20.0, 75.0), (70.0, 48.0)),
    ((-91.0, -52.0), (-33.0, -78.0), (72.0, -39.0), (-66.0, 50.0), (7.0, 9.0), (76.0, 76.0)),
    ((-83.0, -67.0), (-6.0, -88.0), (79.0, -45.0), (-82.0, 19.0), (-13.0, 57.0), (71.0, 70.0)),
)


def _orient(a: tuple[float, float], b: tuple[float, float], c: tuple[float, float]) -> float:
    return (b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0])


def _circumcircle(
    a: tuple[float, float], b: tuple[float, float], c: tuple[float, float]
) -> tuple[float, float, float] | None:
    determinant = 2.0 * _orient(a, b, c)
    if abs(determinant) < 1e-8:
        return None
    aa = a[0] * a[0] + a[1] * a[1]
    bb = b[0] * b[0] + b[1] * b[1]
    cc = c[0] * c[0] + c[1] * c[1]
    ux = (aa * (b[1] - c[1]) + bb * (c[1] - a[1]) + cc * (a[1] - b[1])) / determinant
    uy = (aa * (c[0] - b[0]) + bb * (a[0] - c[0]) + cc * (b[0] - a[0])) / determinant
    radius_sq = (ux - a[0]) ** 2 + (uy - a[1]) ** 2
    return ux, uy, radius_sq


def _delaunay_faces(points: list[dict]) -> frozenset[tuple[str, str, str]]:
    """Return the unoriented color triples with empty circumcircles."""
    positions = [(float(point["x"]), float(point["y"])) for point in points]
    faces: set[tuple[str, str, str]] = set()
    for i, j, k in itertools.combinations(range(len(points)), 3):
        circle = _circumcircle(positions[i], positions[j], positions[k])
        if circle is None:
            continue
        ux, uy, radius_sq = circle
        empty = True
        for index, (x, y) in enumerate(positions):
            if index in (i, j, k):
                continue
            if (x - ux) ** 2 + (y - uy) ** 2 < radius_sq - 1e-7:
                empty = False
                break
        if empty:
            faces.add(tuple(sorted((points[i]["color"], points[j]["color"], points[k]["color"]))))
    return frozenset(faces)


def _minimum_separation(points: list[dict]) -> float:
    positions = [(float(point["x"]), float(point["y"])) for point in points]
    return min(math.dist(left, right) for left, right in itertools.combinations(positions, 2))


def _minimum_circle_gap(points: list[dict]) -> float:
    """Smallest absolute radius gap between a triple's circle and other points."""
    positions = [(float(point["x"]), float(point["y"])) for point in points]
    gaps = []
    for i, j, k in itertools.combinations(range(len(points)), 3):
        circle = _circumcircle(positions[i], positions[j], positions[k])
        if circle is None:
            return 0.0
        ux, uy, radius_sq = circle
        radius = math.sqrt(radius_sq)
        for index, (x, y) in enumerate(positions):
            if index not in (i, j, k):
                gaps.append(abs(math.dist((ux, uy), (x, y)) - radius))
    return min(gaps) if gaps else 0.0


def _minimum_triangle_area(points: list[dict]) -> float:
    positions = [(float(point["x"]), float(point["y"])) for point in points]
    return min(abs(_orient(positions[i], positions[j], positions[k])) / 2.0 for i, j, k in itertools.combinations(range(len(points)), 3))


def _panel_is_safe(panel: dict) -> bool:
    try:
        points = panel["points"]
        if len(points) != 6 or {point["color"] for point in points} != set(COLOR_ORDER):
            return False
        return (
            _minimum_separation(points) >= 32.0
            and _minimum_circle_gap(points) >= 1.3
            and _minimum_triangle_area(points) >= 150.0
            and len(_delaunay_faces(points)) >= 4
        )
    except (KeyError, TypeError, ValueError, ZeroDivisionError):
        return False


def _rotate_scale(points: list[dict], angle: float, scale: float, reflection: float) -> list[dict]:
    ca, sa = math.cos(angle), math.sin(angle)
    transformed = []
    for point in points:
        px = float(point["x"]) * reflection
        py = float(point["y"])
        transformed.append(
            {
                "x": scale * (ca * px - sa * py),
                "y": scale * (sa * px + ca * py),
                "color": point["color"],
            }
        )
    return transformed


def _wrong_color_permutation(points: list[dict], rng: random.Random, reference_faces: frozenset) -> list[dict]:
    for _ in range(200):
        indices = list(range(len(points)))
        rng.shuffle(indices)
        if indices == list(range(len(points))):
            continue
        wrong = [dict(point) for point in points]
        for index, source in enumerate(indices):
            wrong[index]["color"] = points[source]["color"]
        if _delaunay_faces(wrong) != reference_faces:
            return wrong
    raise RuntimeError("template has no face-changing color permutation")


def sample_scene(seed: int) -> dict:
    """Sample a balanced, finite-resolution-safe scene."""
    rng = random.Random(int(seed))
    for _ in range(5000):
        template = rng.choice(_TEMPLATES)
        angle = rng.uniform(0.0, 2.0 * math.pi)
        scale = rng.uniform(0.96, 1.05)
        ca, sa = math.cos(angle), math.sin(angle)
        reference_points = []
        colors = list(COLOR_ORDER)
        rng.shuffle(colors)
        for (x, y), color in zip(template, colors):
            reference_points.append(
                {
                    "x": scale * (ca * x - sa * y),
                    "y": scale * (sa * x + ca * y),
                    "color": color,
                }
            )
        reference = {"points": reference_points, "radius": rng.randint(9, 13)}
        if not _panel_is_safe(reference):
            continue
        reference_faces = _delaunay_faces(reference_points)

        matching = _rotate_scale(
            reference_points,
            rng.uniform(0.0, 2.0 * math.pi),
            rng.uniform(0.88, 1.12),
            -1.0 if rng.randrange(2) else 1.0,
        )
        wrong_base = _rotate_scale(
            reference_points,
            rng.uniform(0.0, 2.0 * math.pi),
            rng.uniform(0.88, 1.12),
            -1.0 if rng.randrange(2) else 1.0,
        )
        wrong = _wrong_color_permutation(wrong_base, rng, reference_faces)
        candidate_a = {"points": matching, "radius": rng.randint(9, 13)}
        candidate_b = {"points": wrong, "radius": rng.randint(9, 13)}
        if rng.randrange(2):
            candidate_a, candidate_b = candidate_b, candidate_a
        scene = {"reference": reference, "candidate_a": candidate_a, "candidate_b": candidate_b}
        if not all(_panel_is_safe(scene[key]) for key in ("reference", "candidate_a", "candidate_b")):
            continue
        if _face_error(reference_faces, _delaunay_faces(candidate_a["points"])) == _face_error(
            reference_faces, _delaunay_faces(candidate_b["points"])
        ):
            continue
        return scene
    raise RuntimeError("could not sample a finite-resolution-safe Delaunay constellation")


def _face_error(reference: frozenset, candidate: frozenset) -> int:
    return len(reference.symmetric_difference(candidate))


def _draw_panel(draw: ImageDraw.ImageDraw, panel: dict, center: tuple[int, int], label: str, title: str) -> None:
    cx, cy = center
    for point in panel["points"]:
        x = int(round(cx + float(point["x"])))
        y = int(round(cy + float(point["y"])))
        radius = int(panel["radius"])
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=COLORS[point["color"]])
    draw.text((cx, 77), title, fill=INK, anchor="ma")
    draw.text((cx, 374), label, fill=INK, anchor="ma")


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((WIDTH // 2, 28), "COLOR-LABELED EMPTY-CIRCUMCIRCLE TRIANGLES", fill=INK, anchor="ma")
    for box in PANEL_BOXES:
        draw.rectangle(box, outline=INK, width=2)
    _draw_panel(draw, scene["reference"], PANEL_CENTERS[0], "REFERENCE", "REFERENCE")
    _draw_panel(draw, scene["candidate_a"], PANEL_CENTERS[1], "A", "CANDIDATE")
    _draw_panel(draw, scene["candidate_b"], PANEL_CENTERS[2], "B", "CANDIDATE")
    return image


def analytic_gold(scene: dict) -> str:
    reference = scene["reference"]
    candidate_a = scene["candidate_a"]
    candidate_b = scene["candidate_b"]
    reference_faces = _delaunay_faces(reference["points"])
    error_a = _face_error(reference_faces, _delaunay_faces(candidate_a["points"]))
    error_b = _face_error(reference_faces, _delaunay_faces(candidate_b["points"]))
    if error_a == error_b:
        raise ValueError("candidate Delaunay face errors are tied")
    return "A" if error_a < error_b else "B"


def margin(scene: dict) -> float:
    reference_faces = _delaunay_faces(scene["reference"]["points"])
    errors = [
        _face_error(reference_faces, _delaunay_faces(scene[key]["points"]))
        for key in ("candidate_a", "candidate_b")
    ]
    return float(abs(errors[0] - errors[1]))


def is_quarantined(scene: dict) -> bool:
    try:
        if not all(_panel_is_safe(scene[key]) for key in ("reference", "candidate_a", "candidate_b")):
            return True
        return margin(scene) < 1.0
    except (KeyError, TypeError, ValueError, ZeroDivisionError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
