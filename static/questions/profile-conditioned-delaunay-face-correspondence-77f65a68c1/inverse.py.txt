"""Independent image-only inverse for Delaunay-face correspondence."""

from __future__ import annotations

import itertools
import math

import numpy as np
from PIL import Image

WIDTH, HEIGHT = 1080, 390
PANEL_X_RANGES = ((28, 332), (388, 692), (748, 1052))
PANEL_Y_RANGE = (96, 354)
COLORS = (
    ("red", (221, 55, 48)),
    ("blue", (45, 112, 184)),
    ("green", (43, 153, 91)),
    ("orange", (235, 137, 45)),
    ("purple", (133, 79, 177)),
    ("teal", (31, 157, 157)),
)


def _component_centroid(mask: np.ndarray) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 20:
        raise ValueError("missing colored point component")
    remaining = set(zip(xs.tolist(), ys.tolist()))
    best: list[tuple[int, int]] = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        component = [start]
        while stack:
            x, y = stack.pop()
            for neighbor in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
                if neighbor in remaining:
                    remaining.remove(neighbor)
                    stack.append(neighbor)
                    component.append(neighbor)
        if len(component) > len(best):
            best = component
    if len(best) < 100:
        raise ValueError("colored point component is too small")
    return (sum(x for x, _ in best) / len(best), sum(y for _, y in best) / len(best))


def _panel_points(array: np.ndarray, x_range: tuple[int, int]) -> dict[str, tuple[float, float]]:
    x0, x1 = x_range
    y0, y1 = PANEL_Y_RANGE
    panel = array[y0:y1, x0:x1]
    points = {}
    for name, rgb in COLORS:
        mask = np.all(panel == np.asarray(rgb, dtype=np.uint8), axis=2)
        x, y = _component_centroid(mask)
        points[name] = (x + x0, y + y0)
    return points


def _orient(a: tuple[float, float], b: tuple[float, float], c: tuple[float, float]) -> float:
    return (b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0])


def _circle(a: tuple[float, float], b: tuple[float, float], c: tuple[float, float]) -> tuple[float, float, float] | None:
    determinant = 2.0 * _orient(a, b, c)
    if abs(determinant) < 1e-7:
        return None
    aa = a[0] * a[0] + a[1] * a[1]
    bb = b[0] * b[0] + b[1] * b[1]
    cc = c[0] * c[0] + c[1] * c[1]
    ux = (aa * (b[1] - c[1]) + bb * (c[1] - a[1]) + cc * (a[1] - b[1])) / determinant
    uy = (aa * (c[0] - b[0]) + bb * (a[0] - c[0]) + cc * (b[0] - a[0])) / determinant
    return ux, uy, (ux - a[0]) ** 2 + (uy - a[1]) ** 2


def _faces(points: dict[str, tuple[float, float]]) -> frozenset[tuple[str, str, str]]:
    names = tuple(name for name, _ in COLORS)
    result: set[tuple[str, str, str]] = set()
    for triple in itertools.combinations(names, 3):
        circle = _circle(*(points[name] for name in triple))
        if circle is None:
            raise ValueError("collinear pixel points")
        ux, uy, radius_sq = circle
        if all(
            (points[name][0] - ux) ** 2 + (points[name][1] - uy) ** 2 >= radius_sq - 0.35
            for name in names
            if name not in triple
        ):
            result.add(tuple(sorted(triple)))
    return frozenset(result)


def _error(reference: frozenset, candidate: frozenset) -> int:
    return len(reference.symmetric_difference(candidate))


def decision_from_image(image: Image.Image) -> str:
    """Recover the answer from the PNG only; no scene or renderer is accepted."""
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.shape[:2] != (HEIGHT, WIDTH):
        raise ValueError(f"unexpected raster size {array.shape[:2]}")
    reference = _faces(_panel_points(array, PANEL_X_RANGES[0]))
    errors = [
        _error(reference, _faces(_panel_points(array, PANEL_X_RANGES[index])))
        for index in (1, 2)
    ]
    if errors[0] == errors[1]:
        raise ValueError("pixel candidates have tied Delaunay face errors")
    return "A" if errors[0] < errors[1] else "B"
