"""Render a note whose visible folded corner may conflict with the dog-ear prior."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
CORNERS = ("top-left", "top-right", "bottom-left", "bottom-right")
PALETTES = (
    ("#dceeff", "#e24a33", "#5e7184"),
    ("#fff0cf", "#3267c8", "#806b51"),
    ("#e8f4df", "#a13db8", "#5b7656"),
    ("#f3e4ef", "#168a78", "#795c72"),
)


def sample_scene(seed: int) -> dict:
    """Return a deterministic scene; consecutive seeds balance all four labels."""
    rng = random.Random(int(seed))
    corner = CORNERS[int(seed) % 4]
    paper_color, fold_color, line_color = PALETTES[(int(seed) // 4) % len(PALETTES)]
    width = rng.randrange(264, 333, 8)
    height = rng.randrange(226, 299, 8)
    x0 = rng.randint(55, WIDTH - 55 - width)
    y0 = rng.randint(55, HEIGHT - 55 - height)
    fold_size = rng.randrange(46, 79, 4)
    rule_count = rng.randint(3, 6)
    markers = []
    # Small neutral binder dots are deliberately far from every page corner.
    for index in range(rng.randint(2, 5)):
        markers.append(
            {
                "x_frac": round(0.20 + 0.13 * index, 3),
                "y_frac": round(rng.uniform(0.35, 0.72), 3),
                "radius": rng.randint(3, 5),
            }
        )
    return {
        "corner": corner,
        "note_box": [x0, y0, x0 + width, y0 + height],
        "fold_size": fold_size,
        "paper_color": paper_color,
        "fold_color": fold_color,
        "line_color": line_color,
        "rule_count": rule_count,
        "marker_layout": markers,
        "prior_condition": "control" if corner == "top-right" else "counterfactual",
        "seed": int(seed),
    }


def _fold_points(scene: dict) -> list[tuple[int, int]]:
    x0, y0, x1, y1 = scene["note_box"]
    size = int(scene["fold_size"])
    corner = scene["corner"]
    if corner == "top-left":
        return [(x0, y0), (x0 + size, y0), (x0, y0 + size)]
    if corner == "top-right":
        return [(x1, y0), (x1 - size, y0), (x1, y0 + size)]
    if corner == "bottom-left":
        return [(x0, y1), (x0 + size, y1), (x0, y1 - size)]
    return [(x1, y1), (x1 - size, y1), (x1, y1 - size)]


def render(scene: dict) -> Image.Image:
    """Draw the scene with ordinary Pillow primitives at a fixed resolution."""
    image = Image.new("RGB", (WIDTH, HEIGHT), "#fafafa")
    draw = ImageDraw.Draw(image)
    x0, y0, x1, y1 = scene["note_box"]
    draw.rectangle((x0, y0, x1, y1), fill=scene["paper_color"], outline="#26313a", width=5)

    count = int(scene["rule_count"])
    for i in range(count):
        y = y0 + int((i + 1) * (y1 - y0) / (count + 1))
        draw.line((x0 + 38, y, x1 - 38, y), fill=scene["line_color"], width=3)

    # Sorting makes marker list order a real pixel-identical latent alias.
    for marker in sorted(scene["marker_layout"], key=lambda item: (item["x_frac"], item["y_frac"])):
        x = x0 + int(float(marker["x_frac"]) * (x1 - x0))
        y = y0 + int(float(marker["y_frac"]) * (y1 - y0))
        r = int(marker["radius"])
        draw.ellipse((x - r, y - r, x + r, y + r), fill="#555b61")

    points = _fold_points(scene)
    draw.polygon(points, fill=scene["fold_color"], outline="#26313a")
    # The diagonal crease makes the fold semantics explicit in the pixels.
    draw.line((points[1], points[2]), fill="#26313a", width=4)
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["corner"])


def margin(scene: dict) -> float:
    """Conservative fold-area margin over the largest plausible distractor."""
    fold_area = float(scene["fold_size"]) ** 2 / 2.0
    marker_area = max((3.15 * float(m["radius"]) ** 2 for m in scene["marker_layout"]), default=0.0)
    return fold_area - marker_area


def is_quarantined(scene: dict) -> bool:
    x0, y0, x1, y1 = scene["note_box"]
    return (
        int(scene["fold_size"]) < 40
        or int(scene["fold_size"]) * 2 >= min(x1 - x0, y1 - y0)
        or margin(scene) < 700.0
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reorder the set-like marker records without changing any pixels."""
    twin = deepcopy(scene)
    twin["marker_layout"] = list(reversed(twin["marker_layout"]))
    return [("reverse_marker_record_order", twin)]


def rotate_clockwise(scene: dict) -> dict:
    """Declared label-transforming image rotation used by the candidate tests."""
    mapping = {
        "top-left": "top-right",
        "top-right": "bottom-right",
        "bottom-right": "bottom-left",
        "bottom-left": "top-left",
    }
    x0, y0, x1, y1 = scene["note_box"]
    twin = deepcopy(scene)
    twin["corner"] = mapping[scene["corner"]]
    twin["note_box"] = [WIDTH - 1 - y1, x0, WIDTH - 1 - y0, x1]
    # Fractions rotate within the note; widths/heights interchange automatically.
    for marker in twin["marker_layout"]:
        old_x, old_y = marker["x_frac"], marker["y_frac"]
        marker["x_frac"], marker["y_frac"] = round(1.0 - old_y, 3), round(old_x, 3)
    return twin
