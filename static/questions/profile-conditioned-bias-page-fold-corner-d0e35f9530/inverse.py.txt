"""Independent pixel-only inverse for the folded-note corner decision."""

from __future__ import annotations

import colorsys

from PIL import Image


ANSWERS = ("top-left", "top-right", "bottom-left", "bottom-right")


def _is_target(rgb: tuple[int, int, int]) -> bool:
    r, g, b = rgb
    _, saturation, value = colorsys.rgb_to_hsv(r / 255.0, g / 255.0, b / 255.0)
    return saturation > 0.42 and 0.25 < value < 0.93


def decision_from_image(image: Image.Image) -> str:
    """Find the dominant saturated patch, then name its image quadrant."""
    rgb = image.convert("RGB")
    width, height = rgb.size
    coords = [
        (x, y)
        for y in range(height)
        for x in range(width)
        if _is_target(rgb.getpixel((x, y)))
    ]
    if len(coords) < 400:
        return "abstain"
    xs = [point[0] for point in coords]
    ys = [point[1] for point in coords]
    cx = sum(xs) / len(xs)
    cy = sum(ys) / len(ys)

    # Recover the note bounds independently from all pixels unlike the background.
    foreground = []
    for y in range(height):
        for x in range(width):
            r, g, b = rgb.getpixel((x, y))
            if min(r, g, b) < 242:
                foreground.append((x, y))
    if len(foreground) < 2000:
        return "abstain"
    left = min(p[0] for p in foreground)
    right = max(p[0] for p in foreground)
    top = min(p[1] for p in foreground)
    bottom = max(p[1] for p in foreground)
    mid_x = (left + right) / 2.0
    mid_y = (top + bottom) / 2.0

    horizontal = "left" if cx < mid_x else "right"
    vertical = "top" if cy < mid_y else "bottom"
    decision = f"{vertical}-{horizontal}"
    # The target must be meaningfully nearer its claimed corner than the midlines.
    if abs(cx - mid_x) < 0.18 * (right - left) or abs(cy - mid_y) < 0.18 * (bottom - top):
        return "abstain"
    return decision if decision in ANSWERS else "abstain"
