"""Independent image-only inverse arm for the paired histogram world."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_BLUE = np.array((31, 119, 180), dtype=np.int16)
_ORANGE = np.array((230, 111, 28), dtype=np.int16)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Return 4-connected component coordinates without layout assumptions."""
    work = mask.copy()
    found: list[np.ndarray] = []
    height, width = work.shape
    while work.any():
        y0, x0 = np.argwhere(work)[0]
        queue = deque([(int(y0), int(x0))])
        work[y0, x0] = False
        coords: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            coords.append((y, x))
            for yy, xx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= yy < height and 0 <= xx < width and work[yy, xx]:
                    work[yy, xx] = False
                    queue.append((yy, xx))
        if len(coords) >= 100:
            found.append(np.asarray(coords, dtype=np.int32))
    return found


def _bar_frequencies(array: np.ndarray, target: np.ndarray) -> list[float]:
    distance = np.max(np.abs(array.astype(np.int16) - target), axis=2)
    components = _components(distance <= 2)
    if len(components) != 6:
        raise ValueError(f"expected six colored bars, found {len(components)}")
    measured = []
    for component in components:
        ys, xs = component[:, 0], component[:, 1]
        box_width = int(xs.max() - xs.min() + 1)
        box_height = int(ys.max() - ys.min() + 1)
        fill = len(component) / float(box_width * box_height)
        if box_width < 10 or box_height < 20 or fill < 0.97:
            raise ValueError("colored component is not a solid histogram bar")
        measured.append((float(xs.mean()), float(len(component))))
    measured.sort()
    centers = [item[0] for item in measured]
    if any(b - a < 8 for a, b in zip(centers, centers[1:])):
        raise ValueError("histogram bars are not distinctly ordered")
    return [item[1] for item in measured]


def _weighted_variance(frequencies: list[float]) -> float:
    total = sum(frequencies)
    if total <= 0:
        raise ValueError("empty histogram")
    mean = sum((index + 1) * value for index, value in enumerate(frequencies)) / total
    return sum(value * ((index + 1) - mean) ** 2 for index, value in enumerate(frequencies)) / total


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    blue = _weighted_variance(_bar_frequencies(array, _BLUE))
    orange = _weighted_variance(_bar_frequencies(array, _ORANGE))
    if abs(blue - orange) < 0.02:
        raise ValueError("pixel variance comparison is ambiguous")
    return "blue" if blue > orange else "orange"
