"""Deterministic paired-histogram world with analytic weighted variance gold."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

BLUE = (31, 119, 180)
ORANGE = (230, 111, 28)
INK = (35, 39, 45)
GRID = (220, 224, 229)
BACKGROUND = (250, 250, 248)


def _variance(heights: list[int]) -> float:
    total = float(sum(heights))
    mean = sum((index + 1) * value for index, value in enumerate(heights)) / total
    return sum(value * ((index + 1) - mean) ** 2 for index, value in enumerate(heights)) / total


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample equal-total histograms with answer identity balanced by seed parity."""
    rng = random.Random(int(seed))
    desired = "blue" if int(seed) % 2 == 0 else "orange"
    # Equal total ink defeats a cheap total-area cue. Rejection produces broad,
    # non-template shapes and a comfortably measurable variance gap.
    for _ in range(20000):
        blue = [rng.randint(2, 10) for _ in range(6)]
        total = sum(blue)
        orange = [rng.randint(2, 10) for _ in range(5)]
        orange.append(total - sum(orange))
        if not 2 <= orange[-1] <= 10:
            continue
        rng.shuffle(orange)
        gap = _variance(blue) - _variance(orange)
        if 0.38 <= abs(gap) <= 2.8 and ((gap > 0) == (desired == "blue")):
            return {
                "blue_heights": blue,
                "orange_heights": orange,
                "width": 480,
                "height": 320,
            }
    raise RuntimeError("could not sample a separated equal-total scene")


def render(scene: dict[str, Any]) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    baseline = height - 56
    top = 54
    panel_lefts = (37, width // 2 + 27)
    panel_width = 176
    bar_width = 22
    gap = 6
    scale = 17

    for y in range(baseline - 2 * scale, top - 1, -2 * scale):
        for left in panel_lefts:
            draw.line((left, y, left + panel_width, y), fill=GRID, width=1)

    blue_heights = scene["blue_heights"]
    orange_heights = scene["orange_heights"]
    for left, values, color, label in (
        (panel_lefts[0], blue_heights, BLUE, "BLUE"),
        (panel_lefts[1], orange_heights, ORANGE, "ORANGE"),
    ):
        draw.line((left, top, left, baseline), fill=INK, width=2)
        draw.line((left, baseline, left + panel_width, baseline), fill=INK, width=2)
        for index, value in enumerate(values):
            x0 = left + 7 + index * (bar_width + gap)
            x1 = x0 + bar_width - 1
            y0 = baseline - int(value) * scale
            draw.rectangle((x0, y0, x1, baseline - 2), fill=color)
            draw.text((x0 + 7, baseline + 7), str(index + 1), fill=INK)
        font = ImageFont.load_default()
        box = draw.textbbox((0, 0), label, font=font)
        tx = left + (panel_width - (box[2] - box[0])) // 2
        draw.text((tx, 24), label, fill=INK, font=font)
    draw.text((166, height - 24), "BIN POSITION", fill=INK)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    blue = _variance(scene["blue_heights"])
    orange = _variance(scene["orange_heights"])
    return "blue" if blue > orange else "orange"


def margin(scene: dict[str, Any]) -> float:
    return abs(_variance(scene["blue_heights"]) - _variance(scene["orange_heights"]))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 0.18


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
