"""Sampler and renderer for three synchronous conditional-flip rounds."""

from __future__ import annotations

import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

SIZE = (900, 650)
BACKGROUND = (247, 246, 241)
INK = (31, 38, 48)
CARD = (255, 255, 253)
BORDER = (102, 110, 121)
ON = (28, 49, 72)
OFF = (226, 230, 225)
ROUND_FILL = (239, 242, 244)
ARROW = (51, 59, 70)
COLORS = ((202, 67, 64), (44, 112, 181), (42, 142, 96), (137, 83, 168))
LETTERS = ("A", "B", "C", "D")
DECISIONS = ("lamp-A", "lamp-B", "lamp-C", "lamp-D")
START_X = (45, 260, 475, 690)
ROUND_Y = (225, 365, 505)
RULE_X = (155, 395, 635)


def _font(size: int) -> ImageFont.ImageFont:
    path = Path("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf")
    return ImageFont.truetype(str(path), size=size) if path.exists() else ImageFont.load_default()


def evolve(initial_state: list[int], round_rules: list[list[list[int]]]) -> list[int]:
    if len(initial_state) != 4 or len(round_rules) != 3:
        raise ValueError("four bits and three rounds required")
    state = [int(value) for value in initial_state]
    if any(value not in (0, 1) for value in state):
        raise ValueError("lamp states must be binary")
    for rules in round_rules:
        if len(rules) != 3:
            raise ValueError("three rules per round required")
        old = state[:]
        new = state[:]
        for pair in rules:
            if len(pair) != 2:
                raise ValueError("rule must be a source-target pair")
            source, target = map(int, pair)
            if source == target or not (0 <= source < 4 and 0 <= target < 4):
                raise ValueError("invalid rule binding")
            if old[source]:
                new[target] ^= 1
        state = new
    return state


def _sequential(initial_state: list[int], round_rules: list[list[list[int]]]) -> list[int]:
    state = list(initial_state)
    for rules in round_rules:
        for source, target in rules:
            if state[source]:
                state[target] ^= 1
    return state


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x53594E43)
    target = int(seed) % 4
    directed = [(source, dest) for source in range(4) for dest in range(4) if source != dest]
    for _ in range(100000):
        initial = [rng.randrange(2) for _ in range(4)]
        if sum(initial) in (0, 4):
            continue
        rounds: list[list[list[int]]] = []
        chain_rounds = 0
        for _round in range(3):
            rules = [list(pair) for pair in rng.sample(directed, 3)]
            if any(rules[i][1] == rules[j][0] for i in range(3) for j in range(i + 1, 3)):
                chain_rounds += 1
            rounds.append(rules)
        final = evolve(initial, rounds)
        if final != [int(index == target) for index in range(4)]:
            continue
        if chain_rounds < 2 or _sequential(initial, rounds) == final:
            continue
        state = initial[:]
        active_counts: list[int] = []
        for rules in rounds:
            active_counts.append(sum(state[source] for source, _target in rules))
            state = _one_round(state, rules)
        if all(count > 0 for count in active_counts):
            return {"initial_state": initial, "round_rules": rounds}
    raise RuntimeError("could not sample a balanced synchronous scene")


def _one_round(state: list[int], rules: list[list[int]]) -> list[int]:
    old = state[:]
    new = state[:]
    for source, target in rules:
        if old[source]:
            new[target] ^= 1
    return new


def _validate(scene: dict) -> tuple[list[int], list[list[list[int]]]]:
    initial = [int(value) for value in scene["initial_state"]]
    rounds = [[[int(source), int(target)] for source, target in rules] for rules in scene["round_rules"]]
    evolve(initial, rounds)
    return initial, rounds


def analytic_gold(scene: dict) -> str:
    try:
        initial, rounds = _validate(scene)
        final = evolve(initial, rounds)
        on = [index for index, value in enumerate(final) if value]
        return DECISIONS[on[0]] if len(on) == 1 else "abstain"
    except (KeyError, TypeError, ValueError, IndexError):
        return "abstain"


def margin(scene: dict) -> float:
    return 1.0 if analytic_gold(scene) in DECISIONS else 0.0


def is_quarantined(scene: dict) -> bool:
    return analytic_gold(scene) == "abstain"


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _center(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str,
            font: ImageFont.ImageFont, fill: tuple[int, int, int]) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2 - box[1]),
              text, font=font, fill=fill)


def render(scene: dict) -> Image.Image:
    initial, rounds = _validate(scene)
    image = Image.new("RGB", SIZE, BACKGROUND)
    draw = ImageDraw.Draw(image)
    _center(draw, (450, 28), "SIMULTANEOUS FLIP ROUNDS", _font(25), INK)
    _center(draw, (450, 60), "ON source flips target  •  read all sources before updating", _font(17), BORDER)

    for index, (x, value) in enumerate(zip(START_X, initial)):
        draw.rounded_rectangle((x, 84, x + 165, 194), radius=14, fill=CARD, outline=COLORS[index], width=4)
        _center(draw, (x + 82, 105), f"LAMP {LETTERS[index]}", _font(18), COLORS[index])
        draw.ellipse((x + 57, 121, x + 107, 171), fill=ON if value else OFF, outline=INK, width=3)
        _center(draw, (x + 82, 184), "ON" if value else "OFF", _font(15), INK)

    for round_index, (y, rules) in enumerate(zip(ROUND_Y, rounds)):
        draw.rounded_rectangle((25, y, 875, y + 112), radius=14, fill=ROUND_FILL, outline=BORDER, width=3)
        _center(draw, (85, y + 38), f"ROUND {round_index + 1}", _font(19), INK)
        _center(draw, (85, y + 75), "same old state", _font(12), BORDER)
        for slot, ((source, target), x) in enumerate(zip(rules, RULE_X)):
            draw.rounded_rectangle((x, y + 14, x + 210, y + 98), radius=12, fill=CARD, outline=BORDER, width=2)
            draw.ellipse((x + 20, y + 28, x + 78, y + 86), fill=COLORS[source], outline=INK, width=2)
            _center(draw, (x + 49, y + 57), LETTERS[source], _font(22), (255, 255, 255))
            _center(draw, (x + 106, y + 57), "→", _font(31), ARROW)
            draw.rounded_rectangle((x + 135, y + 28, x + 193, y + 86), radius=8,
                                   fill=COLORS[target], outline=INK, width=2)
            _center(draw, (x + 164, y + 57), LETTERS[target], _font(22), (255, 255, 255))
    return image
