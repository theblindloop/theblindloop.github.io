"""Independent fixed-layout pixel inverse for synchronous flip rounds."""

from __future__ import annotations

import numpy as np
from PIL import Image

BACKGROUND = (247, 246, 241)
ON = (28, 49, 72)
OFF = (226, 230, 225)
COLORS = ((202, 67, 64), (44, 112, 181), (42, 142, 96), (137, 83, 168))
START_X = (45, 260, 475, 690)
ROUND_Y = (225, 365, 505)
RULE_X = (155, 395, 635)
DECISIONS = ("lamp-A", "lamp-B", "lamp-C", "lamp-D")


def _near(pixel: np.ndarray, color: tuple[int, int, int], tolerance: int = 4) -> bool:
    return bool(np.max(np.abs(pixel.astype(np.int16) - np.array(color, dtype=np.int16))) <= tolerance)


def _which_color(pixel: np.ndarray) -> int | None:
    hits = [index for index, color in enumerate(COLORS) if _near(pixel, color)]
    return hits[0] if len(hits) == 1 else None


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.shape != (650, 900, 3) or not _near(arr[640, 890], BACKGROUND):
        return "abstain"

    state: list[int] = []
    for x in START_X:
        pixel = arr[146, x + 82]
        if _near(pixel, ON):
            state.append(1)
        elif _near(pixel, OFF):
            state.append(0)
        else:
            return "abstain"

    rounds: list[list[tuple[int, int]]] = []
    for y in ROUND_Y:
        rules: list[tuple[int, int]] = []
        for x in RULE_X:
            source = _which_color(arr[y + 57, x + 29])
            target = _which_color(arr[y + 57, x + 184])
            if source is None or target is None or source == target:
                return "abstain"
            rules.append((source, target))
        rounds.append(rules)

    for rules in rounds:
        old = state[:]
        new = state[:]
        for source, target in rules:
            if old[source]:
                new[target] ^= 1
        state = new
    on = [index for index, value in enumerate(state) if value]
    return DECISIONS[on[0]] if len(on) == 1 else "abstain"
