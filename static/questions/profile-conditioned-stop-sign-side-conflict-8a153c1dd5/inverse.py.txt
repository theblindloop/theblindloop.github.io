"""Independent image-only inverse arm for the sign's visible top support.

This module deliberately contains its own public color and geometry policy and
does not import renderer, prompt, generation, verification, scene, or gold code.
Only the final RGB raster enters ``decision_from_image``.
"""

from __future__ import annotations

from dataclasses import asdict, dataclass

import numpy as np
from PIL import Image

_PROTOTYPES = {"corner": 0.045, "flat": 0.405}
MIN_RUNNER_UP_MARGIN = 0.12
MAX_BEST_RESIDUAL = 0.15


class PixelOracleError(ValueError):
    """Raised when the raster does not expose an unambiguous red top support."""


@dataclass(frozen=True)
class PixelMeasurement:
    decision: str
    support_fraction: float
    runner_up_margin: float
    best_residual: float
    red_pixel_count: int
    red_bbox: tuple[int, int, int, int]

    def to_dict(self) -> dict:
        return asdict(self)


def _red_mask(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    red = arr[:, :, 0]
    green = arr[:, :, 1]
    blue = arr[:, :, 2]
    # All allowed reds satisfy these broad relational thresholds.  Relational
    # channel tests tolerate antialiasing and do not depend on one renderer RGB.
    return (red >= 120) & (red - green >= 55) & (red - blue >= 45)


def measure_from_image(image: Image.Image) -> PixelMeasurement:
    """Measure the top support and the best-vs-runner-up decision gap."""

    mask = _red_mask(image)
    ys, xs = np.nonzero(mask)
    if len(xs) < 5_000:
        raise PixelOracleError("too little red ink to expose a sign boundary")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    width = x1 - x0 + 1
    height = y1 - y0 + 1
    if width < 120 or height < 120:
        raise PixelOracleError("red component is too small for top-support recovery")
    if x0 <= 2 or y0 <= 2 or x1 >= mask.shape[1] - 3 or y1 >= mask.shape[0] - 3:
        raise PixelOracleError("red boundary touches the raster edge")
    aspect = width / height
    if not 0.82 <= aspect <= 1.22:
        raise PixelOracleError("red silhouette is not a roughly isotropic sign")

    # Examine a shallow, scale-normalized band immediately below the first red
    # row.  A flat top contributes a long support interval on every such row;
    # a vertex-up octagon contributes a narrow wedge whose width grows slowly.
    band_depth = max(4, int(round(height * 0.025)))
    spans: list[int] = []
    for y in range(y0, min(y1 + 1, y0 + band_depth)):
        row_x = np.flatnonzero(mask[y])
        if len(row_x):
            spans.append(int(row_x.max() - row_x.min() + 1))
    if not spans:
        raise PixelOracleError("top support contains no measurable red rows")
    support_fraction = max(spans) / width

    ranked = sorted(
        (abs(support_fraction - prototype), label)
        for label, prototype in _PROTOTYPES.items()
    )
    best_residual, decision = ranked[0]
    runner_up_margin = ranked[1][0] - best_residual
    if best_residual > MAX_BEST_RESIDUAL or runner_up_margin < MIN_RUNNER_UP_MARGIN:
        raise PixelOracleError(
            "ambiguous top support: "
            f"fraction={support_fraction:.4f}, best_residual={best_residual:.4f}, "
            f"runner_up_margin={runner_up_margin:.4f}"
        )
    return PixelMeasurement(
        decision=decision,
        support_fraction=round(float(support_fraction), 8),
        runner_up_margin=round(float(runner_up_margin), 8),
        best_residual=round(float(best_residual), 8),
        red_pixel_count=int(len(xs)),
        red_bbox=(x0, y0, x1, y1),
    )


def decision_from_image(image: Image.Image) -> str:
    """Return ``flat`` or ``corner`` using final image pixels only."""

    return measure_from_image(image).decision
