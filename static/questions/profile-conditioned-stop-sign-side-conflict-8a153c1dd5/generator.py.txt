"""Deterministic renderer and analytic arm for the STOP-sign top-support world."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 512
HEIGHT = 512
SUPERSAMPLE = 4
TOP_SHAPES = ("flat", "corner")
MIN_HALF_SUPPORT_PX = 24.0

_RED_PALETTE = (
    (190, 38, 48),
    (206, 43, 50),
    (220, 48, 54),
    (183, 35, 55),
)
_BACKGROUNDS = (
    (241, 243, 246),
    (246, 242, 232),
    (232, 240, 238),
    (237, 234, 244),
)


def sample_scene(seed: int) -> dict[str, Any]:
    """Return one member of a deterministic matched flat/corner pair.

    Consecutive seeds share every nuisance.  Even seeds show the canonical
    flat-side-up octagon; odd seeds rotate that same octagon so a corner is at
    the top.  The wording and upright STOP lettering are identical.
    """

    seed = int(seed)
    pair_seed = seed // 2
    rng = random.Random((pair_seed * 1_000_003) ^ 0x5A17C0DE)
    radius = round(rng.uniform(118.0, 158.0), 3)
    center_x = round(rng.uniform(230.0, 282.0), 3)
    center_y = round(rng.uniform(207.0, 235.0), 3)
    return {
        "top_shape": TOP_SHAPES[seed % 2],
        "center_x": center_x,
        "center_y": center_y,
        "radius": radius,
        "red_rgb": list(rng.choice(_RED_PALETTE)),
        "background_rgb": list(rng.choice(_BACKGROUNDS)),
        "border_width": rng.choice((5, 6, 7)),
    }


def _vertices(scene: dict[str, Any], radius_scale: float = 1.0) -> list[tuple[float, float]]:
    cx = float(scene["center_x"]) * SUPERSAMPLE
    cy = float(scene["center_y"]) * SUPERSAMPLE
    radius = float(scene["radius"]) * float(radius_scale) * SUPERSAMPLE
    # Screen angles increase clockwise.  A vertex at -90 degrees makes a
    # pointed top; a -22.5-degree phase shift makes the adjacent pair share the
    # minimum y coordinate and therefore forms a horizontal top edge.
    phase = -math.pi / 2 if scene["top_shape"] == "corner" else -math.pi / 2 - math.pi / 8
    return [
        (
            cx + radius * math.cos(phase + index * math.pi / 4),
            cy + radius * math.sin(phase + index * math.pi / 4),
        )
        for index in range(8)
    ]


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf", size=size)
    except OSError:
        return ImageFont.load_default(size=size)


def render(scene: dict[str, Any]) -> Image.Image:
    """Render a clean sign scene as the final 512x512 RGB raster."""

    background = tuple(int(value) for value in scene["background_rgb"])
    red = tuple(int(value) for value in scene["red_rgb"])
    image = Image.new("RGB", (WIDTH * SUPERSAMPLE, HEIGHT * SUPERSAMPLE), background)
    draw = ImageDraw.Draw(image)
    cx = float(scene["center_x"]) * SUPERSAMPLE
    cy = float(scene["center_y"]) * SUPERSAMPLE
    radius = float(scene["radius"]) * SUPERSAMPLE

    # Neutral pole is drawn first and remains visually behind the red sign.
    post_half = max(8.0 * SUPERSAMPLE, radius * 0.075)
    post_top = cy + radius * 0.54
    draw.rounded_rectangle(
        (cx - post_half, post_top, cx + post_half, HEIGHT * SUPERSAMPLE + 12),
        radius=int(3 * SUPERSAMPLE),
        fill=(108, 113, 119),
        outline=(74, 79, 85),
        width=2 * SUPERSAMPLE,
    )

    outer = _vertices(scene)
    draw.polygon(outer, fill=red)

    # A pale inset border reinforces the sign reading without changing the red
    # outer support used by the image-only inverse arm.
    inset_scale = max(0.72, 1.0 - (13.0 + int(scene["border_width"])) / float(scene["radius"]))
    inset = _vertices(scene, inset_scale)
    border_color = (253, 249, 239)
    border_width = int(scene["border_width"]) * SUPERSAMPLE
    draw.line(inset + [inset[0]], fill=border_color, width=border_width, joint="curve")

    font_size = max(40, int(float(scene["radius"]) * 0.47)) * SUPERSAMPLE
    draw.text(
        (cx, cy + 2 * SUPERSAMPLE),
        "STOP",
        fill=(255, 252, 243),
        font=_font(font_size),
        anchor="mm",
        stroke_width=max(1, SUPERSAMPLE // 2),
        stroke_fill=(255, 252, 243),
    )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the label from the sole permitted gold-bearing field."""

    value = str(scene["top_shape"])
    if value not in TOP_SHAPES:
        raise ValueError(f"invalid top_shape {value!r}")
    return value


def margin(scene: dict[str, Any]) -> float:
    """One-sided pixel headroom of the octagon's top feature.

    ``r*sin(pi/8)`` is the half-span from the top feature axis to either
    adjacent octagon vertex.  Subtracting the public 24-pixel visibility floor
    expresses how far the feature lies above the finite-resolution quarantine.
    """

    return float(scene["radius"]) * math.sin(math.pi / 8) - MIN_HALF_SUPPORT_PX


def is_quarantined(scene: dict[str, Any]) -> bool:
    if margin(scene) <= 0.0:
        return True
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    radius = float(scene["radius"])
    return cx - radius < 8 or cx + radius > WIDTH - 8 or cy - radius < 8 or cy + radius > HEIGHT - 8


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No nontrivial pixel-identical latent aliases are claimed."""

    return []
