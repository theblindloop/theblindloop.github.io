"""Deterministic Pillow renderer for a stable move-to-front list replay."""

from __future__ import annotations

import copy
import random
from PIL import Image, ImageDraw, ImageFont

SHAPES = ("circle", "square", "triangle", "diamond")
COLORS = ((214, 68, 74), (39, 126, 209), (39, 165, 104), (231, 145, 38))
PAPER = {"paper_a": (250, 248, 242), "paper_b": (250, 248, 242)}


def _font(size: int, bold: bool = False):
    path = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf" if bold else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
    return ImageFont.truetype(path, size)


def _final_order(scene: dict) -> list[str]:
    order = list(scene["initial_order"])
    for moved in scene["moves"]:
        order.remove(moved)
        order.insert(0, moved)
    return order


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    initial = list(SHAPES)
    rng.shuffle(initial)
    moves = [rng.choice(SHAPES) for _ in range(5)]
    # Force genuine nonlocal work and include a repeated identity without fixing the answer.
    if len(set(moves)) < 3:
        moves[0:3] = rng.sample(list(SHAPES), 3)
    desired = SHAPES[int(seed) % len(SHAPES)]
    work = list(initial)
    for moved in moves:
        work.remove(moved)
        work.insert(0, moved)
    query = work.index(desired)
    color_order = list(range(4))
    rng.shuffle(color_order)
    return {
        "initial_order": initial,
        "moves": moves,
        "query_position": query,
        "palette": {"paper": "paper_a", "color_order": color_order},
        "width": 1000,
        "height": 640,
    }


def _draw_shape(draw: ImageDraw.ImageDraw, shape: str, center: tuple[int, int], color, radius: int = 31) -> None:
    x, y = center
    box = (x - radius, y - radius, x + radius, y + radius)
    if shape == "circle":
        draw.ellipse(box, fill=color, outline=(30, 34, 40), width=3)
    elif shape == "square":
        draw.rectangle(box, fill=color, outline=(30, 34, 40), width=3)
    elif shape == "triangle":
        draw.polygon([(x, y - radius), (x - radius, y + radius), (x + radius, y + radius)], fill=color, outline=(30, 34, 40), width=3)
    elif shape == "diamond":
        draw.polygon([(x, y - radius), (x - radius, y), (x, y + radius), (x + radius, y)], fill=color, outline=(30, 34, 40), width=3)
    else:
        raise ValueError(f"unknown shape {shape}")


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    palette = scene["palette"]
    image = Image.new("RGB", (width, height), PAPER[palette["paper"]])
    draw = ImageDraw.Draw(image)
    title = _font(30, True)
    small = _font(19, True)
    draw.text((40, 24), "MOVE-TO-FRONT", font=title, fill=(28, 32, 38))
    draw.text((40, 72), "START ORDER  •  FRONT → BACK", font=small, fill=(55, 60, 66))

    xs = (170, 390, 610, 830)
    y = 190
    query = int(scene["query_position"])
    # The cyan ring marks a position, not an identity; it encloses the current start token.
    qx = xs[query]
    draw.rounded_rectangle((qx - 58, y - 58, qx + 58, y + 58), radius=15, outline=(0, 174, 230), width=7)
    draw.text((qx - 34, y + 68), "QUERY", font=_font(15, True), fill=(0, 110, 160))
    colors = {shape: COLORS[palette["color_order"][i]] for i, shape in enumerate(SHAPES)}
    for index, shape in enumerate(scene["initial_order"]):
        _draw_shape(draw, shape, (xs[index], y), colors[shape], 31)
        draw.text((xs[index] - 6, 245), str(index + 1), font=_font(16, True), fill=(65, 68, 72))

    draw.line((75, 294, 925, 294), fill=(150, 154, 158), width=2)
    draw.text((40, 318), "MOVE CARDS  •  READ LEFT → RIGHT", font=small, fill=(55, 60, 66))
    card_xs = (105, 300, 495, 690, 885)
    for index, (cx, shape) in enumerate(zip(card_xs, scene["moves"]), start=1):
        draw.rounded_rectangle((cx - 76, 365, cx + 76, 515), radius=13, fill=(255, 255, 255), outline=(74, 79, 86), width=3)
        draw.text((cx - 33, 378), f"{index}  MOVE", font=_font(14, True), fill=(48, 52, 58))
        _draw_shape(draw, shape, (cx, 458), colors[shape], 27)
    draw.text((40, 558), "Rule: remove the pictured token and insert it at FRONT.", font=_font(19), fill=(55, 60, 66))
    draw.text((40, 588), "All other tokens keep their relative order.", font=_font(19), fill=(55, 60, 66))
    return image


def analytic_gold(scene: dict) -> str:
    return _final_order(scene)[int(scene["query_position"])]


def margin(scene: dict) -> float:
    # Runner-up separation: minimum fixed icon gap/outline clearance in source pixels.
    _ = (scene["initial_order"], scene["moves"], scene["query_position"])
    return 24.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["palette"]["paper"] = "paper_b" if scene["palette"]["paper"] == "paper_a" else "paper_a"
    return [("paper_name_alias", twin)]
