"""Independent image-only inverse for Move-to-Front Replay.

The parser uses only raster regions, connected components, shape occupancy and
visible left-to-right order. It receives no latent scene or renderer module.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


def _components(mask: np.ndarray):
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y0, x0 in zip(*np.nonzero(mask & ~seen)):
        if seen[y0, x0]:
            continue
        stack = [(int(y0), int(x0))]
        seen[y0, x0] = True
        pts = []
        while stack:
            y, x = stack.pop()
            pts.append((y, x))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                yy, xx = y + dy, x + dx
                if 0 <= yy < h and 0 <= xx < w and mask[yy, xx] and not seen[yy, xx]:
                    seen[yy, xx] = True
                    stack.append((yy, xx))
        if len(pts) >= 500:
            ys = np.fromiter((p[0] for p in pts), dtype=int)
            xs = np.fromiter((p[1] for p in pts), dtype=int)
            found.append((len(pts), int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max()), float(xs.mean()), float(ys.mean())))
    return found


def _shape(component) -> str:
    area, x0, y0, x1, y1, cx, cy = component
    width = x1 - x0 + 1
    height = y1 - y0 + 1
    occupancy = area / float(width * height)
    rel_y = (cy - y0) / max(1.0, height - 1)
    if occupancy > 0.88:
        return "square"
    if occupancy > 0.68:
        return "circle"
    if rel_y > 0.54:
        return "triangle"
    return "diamond"


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    if rgb.shape[1] < 500 or rgb.shape[0] < 300:
        raise ValueError("raster too small")
    high = rgb.max(axis=2)
    low = rgb.min(axis=2)
    saturated = (high - low > 55) & (high < 250) & (low < 235)
    cyan = (rgb[:, :, 2] > 220) & (rgb[:, :, 1] > 140) & (rgb[:, :, 0] < 70)
    token_mask = saturated & ~cyan

    height = rgb.shape[0]
    initial_components = [c for c in _components(token_mask[: int(height * 0.46), :]) if c[0] > 900 and c[2] > 100]
    initial_components.sort(key=lambda c: c[5])
    if len(initial_components) != 4:
        raise ValueError(f"expected four initial tokens, found {len(initial_components)}")
    initial = [_shape(c) for c in initial_components]
    if set(initial) != {"circle", "square", "triangle", "diamond"}:
        raise ValueError("initial identities are unresolved")

    card_offset = int(height * 0.54)
    card_components = [c for c in _components(token_mask[card_offset:, :]) if c[0] > 650]
    card_components.sort(key=lambda c: c[5])
    if len(card_components) != 5:
        raise ValueError(f"expected five move icons, found {len(card_components)}")
    moves = [_shape(c) for c in card_components]
    if any(item not in initial for item in moves):
        raise ValueError("move identity is unresolved")

    cyan_components = _components(cyan[: int(height * 0.46), :])
    if len(cyan_components) != 1:
        raise ValueError("query ring is missing or ambiguous")
    query_x = cyan_components[0][5]
    centers = [c[5] for c in initial_components]
    distances = [abs(x - query_x) for x in centers]
    query = int(np.argmin(distances))
    ordered = list(initial)
    for moved in moves:
        ordered.remove(moved)
        ordered.insert(0, moved)
    return ordered[query]
