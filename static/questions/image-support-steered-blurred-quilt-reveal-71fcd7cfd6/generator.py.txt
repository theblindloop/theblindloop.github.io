"""Renderer for a balanced halftone quilt revealed by spatial averaging."""

from __future__ import annotations

import random

import numpy as np
from PIL import Image

CANVAS = 512
GRID = 10
BORDER = 8
SYMBOLS = ("X", "plus", "diamond", "ring")
PALETTES = (
    ((28, 67, 91), (239, 196, 99)),
    ((72, 45, 92), (184, 226, 183)),
    ((25, 83, 58), (239, 176, 210)),
)
FRAME = (111, 62, 174)
BACKGROUND = (247, 245, 239)


def _glyph(name: str) -> np.ndarray:
    """Return one of four equal-ink (20-cell) low-frequency templates."""
    out = np.zeros((GRID, GRID), dtype=bool)
    if name == "X":
        for y in range(GRID):
            out[y, y] = True
            out[y, GRID - 1 - y] = True
    elif name == "plus":
        out[:, 4] = True
        out[4, :] = True
        out[5, 5] = True
    elif name == "diamond":
        for y in range(GRID):
            for x in range(GRID):
                out[y, x] = abs(x - 4.5) + abs(y - 4.5) == 5
    elif name == "ring":
        out[2, 2:8] = True
        out[7, 2:8] = True
        out[2:8, 2] = True
        out[2:8, 7] = True
    else:
        raise ValueError(f"unknown symbol: {name}")
    if int(out.sum()) != 20:
        raise AssertionError("templates must have equal low-frequency ink")
    return out


def sample_scene(seed: int) -> dict:
    """Sample balanced answer identity and answer-independent nuisances."""
    mixer = random.Random((int(seed) ^ 0x6A09E667) & 0xFFFFFFFF)
    return {
        "symbol": SYMBOLS[int(seed) % len(SYMBOLS)],
        "texture_seed": mixer.randrange(1, 2**31),
        "palette_index": (int(seed) // 4) % len(PALETTES),
        "offset_x": mixer.randrange(-4, 5),
        "offset_y": mixer.randrange(-4, 5),
        "cell_px": 16 if ((int(seed) // 8) % 2 == 0) else 20,
        "strength": 1 + ((int(seed) // 4) % 2),
    }


def render(scene: dict) -> Image.Image:
    """Render an opaque two-color micro-check field inside a visible fiducial."""
    symbol = str(scene["symbol"])
    texture_seed = int(scene["texture_seed"])
    palette_index = int(scene["palette_index"])
    offset_x = int(scene["offset_x"])
    offset_y = int(scene["offset_y"])
    cell_px = int(scene["cell_px"])
    strength = int(scene["strength"])
    if cell_px not in (16, 20) or strength not in (1, 2):
        raise ValueError("unsupported cell size or reveal strength")
    dark, light = PALETTES[palette_index % len(PALETTES)]
    inner = GRID * cell_px
    outer = inner + 2 * BORDER
    left = (CANVAS - outer) // 2 + offset_x
    top = (CANVAS - outer) // 2 + offset_y
    arr = np.full((CANVAS, CANVAS, 3), BACKGROUND, dtype=np.uint8)
    arr[top : top + outer, left : left + outer] = FRAME
    glyph = _glyph(symbol)
    micro = cell_px // 4
    for gy in range(GRID):
        for gx in range(GRID):
            local = random.Random(texture_seed + 1009 * gy + 9176 * gx)
            order = list(range(16))
            local.shuffle(order)
            dark_count = 8 + strength if glyph[gy, gx] else 8 - strength
            dark_slots = set(order[:dark_count])
            y0 = top + BORDER + gy * cell_px
            x0 = left + BORDER + gx * cell_px
            for slot in range(16):
                my, mx = divmod(slot, 4)
                color = dark if slot in dark_slots else light
                arr[
                    y0 + my * micro : y0 + (my + 1) * micro,
                    x0 + mx * micro : x0 + (mx + 1) * micro,
                ] = color
    return Image.fromarray(arr, mode="RGB")


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def margin(scene: dict) -> float:
    # All four templates have pairwise Hamming distance 32. Strength is the
    # visible separation between on/off macro-cell averages.
    return float(32 * int(scene["strength"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 48.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
