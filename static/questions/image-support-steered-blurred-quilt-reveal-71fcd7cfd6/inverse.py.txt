"""Independent pixel-only inverse for the framed low-pass quilt."""

from __future__ import annotations

import numpy as np
from PIL import Image

_GRID = 10
_BORDER = 8

# These public answer-shape templates are intentionally encoded independently
# of renderer construction code.
_PATTERNS = {
    "X": (
        "#........#", ".#......#.", "..#....#..", "...#..#...", "....##....",
        "....##....", "...#..#...", "..#....#..", ".#......#.", "#........#",
    ),
    "plus": (
        "....#.....", "....#.....", "....#.....", "....#.....", "##########",
        "....##....", "....#.....", "....#.....", "....#.....", "....#.....",
    ),
    "diamond": (
        "....##....", "...#..#...", "..#....#..", ".#......#.", "#........#",
        "#........#", ".#......#.", "..#....#..", "...#..#...", "....##....",
    ),
    "ring": (
        "..........", "..........", "..######..", "..#....#..", "..#....#..",
        "..#....#..", "..#....#..", "..######..", "..........", "..........",
    ),
}


def _template(rows: tuple[str, ...]) -> np.ndarray:
    return np.asarray([[char == "#" for char in row] for row in rows], dtype=bool)


def _measure(image: Image.Image) -> tuple[np.ndarray, float]:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.ndim != 3 or arr.shape[0] < 128 or arr.shape[1] < 128:
        raise ValueError("image is too small for the declared transform")

    # The violet frame is the visible fiducial; coordinates are recovered from
    # pixels, never supplied by scene state or assumed from canvas position.
    r, g, b = arr[..., 0], arr[..., 1], arr[..., 2]
    violet = (r >= 90) & (r <= 135) & (g >= 42) & (g <= 82) & (b >= 150) & (b <= 195)
    points = np.argwhere(violet)
    if len(points) < 2000:
        raise ValueError("missing violet fiducial frame")
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    height, width = int(y1 - y0 + 1), int(x1 - x0 + 1)
    if width != height or (width - 2 * _BORDER) % _GRID or width not in (176, 216):
        raise ValueError("ambiguous framed-field geometry")
    inner_size = width - 2 * _BORDER
    expected_frame = width * width - inner_size * inner_size
    frame_crop = violet[y0 : y1 + 1, x0 : x1 + 1]
    if int(frame_crop.sum()) < int(0.97 * expected_frame):
        raise ValueError("fiducial frame is materially damaged")

    field = arr[
        y0 + _BORDER : y1 - _BORDER + 1,
        x0 + _BORDER : x1 - _BORDER + 1,
    ]
    colors, inverse, counts = np.unique(field.reshape(-1, 3), axis=0, return_inverse=True, return_counts=True)
    if len(colors) != 2 or int(counts.min()) < field.shape[0] * field.shape[1] // 5:
        raise ValueError("framed texture is incomplete or has ambiguous colors")
    luminance = colors @ np.asarray([0.2126, 0.7152, 0.0722])
    dark_index = int(np.argmin(luminance))
    dark = (inverse.reshape(field.shape[:2]) == dark_index).astype(np.float64)
    cell = inner_size // _GRID
    averages = dark.reshape(_GRID, cell, _GRID, cell).mean(axis=(1, 3))
    contrast = float(np.median(np.abs(averages - 0.5)))
    if contrast < 0.045:
        raise ValueError("low-pass reveal is below the finite-resolution boundary")
    return averages, contrast


def decision_with_margin(image: Image.Image) -> tuple[str, float]:
    averages, _ = _measure(image)
    centered = averages - averages.mean()
    scores = {}
    for name, rows in _PATTERNS.items():
        sign = np.where(_template(rows), 1.0, -1.0)
        scores[name] = float(np.sum(centered * sign))
    ordered = sorted(scores.items(), key=lambda item: (-item[1], item[0]))
    margin = ordered[0][1] - ordered[1][1]
    if margin < 1.0:
        raise ValueError("runner-up correlation margin is ambiguous")
    return ordered[0][0], float(margin)


def decision_from_image(image: Image.Image) -> str:
    return decision_with_margin(image)[0]
