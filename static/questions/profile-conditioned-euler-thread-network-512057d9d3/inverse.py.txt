"""Independent pixel-only inverse for colored Euler-trail networks."""

from __future__ import annotations

from collections import deque
import math

from PIL import Image


LABELS = ("red", "blue", "green")
NODE_DARKNESS = 125


def _is_node(pixel: tuple[int, int, int]) -> bool:
    r, g, b = pixel
    return r < NODE_DARKNESS and g < NODE_DARKNESS and b < NODE_DARKNESS and max(pixel) - min(pixel) < 24


def _label(pixel: tuple[int, int, int]) -> str | None:
    r, g, b = (float(value) for value in pixel)
    brightest = max(r, g, b)
    darkest = min(r, g, b)
    if brightest - darkest < 45 or brightest > 245:
        return None
    # Hue sectors are intentionally broad enough for all declared palette
    # styles while excluding the neutral background and waypoint disks.
    if r > g * 1.35 and r > b * 1.20:
        return "red"
    if b > r * 1.20 and b > g * 1.03:
        return "blue"
    if g > r * 1.30 and g > b * 1.04:
        return "green"
    return None


def _components(mask: list[bool], width: int, height: int, minimum: int) -> list[tuple[float, float]]:
    seen = bytearray(width * height)
    centers = []
    for index, active in enumerate(mask):
        if not active or seen[index]:
            continue
        queue = deque([index])
        seen[index] = 1
        count = 0
        total_x = 0
        total_y = 0
        while queue:
            current = queue.popleft()
            x = current % width
            y = current // width
            count += 1
            total_x += x
            total_y += y
            for nx, ny in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
                if 0 <= nx < width and 0 <= ny < height:
                    neighbor = ny * width + nx
                    if mask[neighbor] and not seen[neighbor]:
                        seen[neighbor] = 1
                        queue.append(neighbor)
        if count >= minimum:
            centers.append((total_x / count, total_y / count))
    return centers


def _near_label(rgb, width: int, height: int, x: float, y: float, label: str, radius: int = 2) -> bool:
    for yy in range(max(0, int(round(y)) - radius), min(height, int(round(y)) + radius + 1)):
        for xx in range(max(0, int(round(x)) - radius), min(width, int(round(x)) + radius + 1)):
            if _label(rgb[yy * width + xx]) == label:
                return True
    return False


def _edge_support(rgb, width: int, height: int, left: tuple[float, float], right: tuple[float, float], label: str) -> float:
    distance = math.hypot(right[0] - left[0], right[1] - left[1])
    samples = max(18, int(distance / 5.0))
    hits = 0
    for step in range(1, samples):
        fraction = step / samples
        x = left[0] * (1.0 - fraction) + right[0] * fraction
        y = left[1] * (1.0 - fraction) + right[1] * fraction
        if _near_label(rgb, width, height, x, y, label, radius=2):
            hits += 1
    return hits / max(1, samples - 1)


def _connected(adjacency: list[list[int]]) -> bool:
    reached = {0}
    pending = [0]
    while pending:
        node = pending.pop()
        for neighbor, present in enumerate(adjacency[node]):
            if present and neighbor not in reached:
                reached.add(neighbor)
                pending.append(neighbor)
    return len(reached) == len(adjacency)


def _network_is_euler(rgb, width: int, height: int, centers: list[tuple[float, float]], label: str) -> bool:
    adjacency = [[0 for _ in centers] for _ in centers]
    for left in range(len(centers)):
        for right in range(left + 1, len(centers)):
            if _edge_support(rgb, width, height, centers[left], centers[right], label) >= 0.70:
                adjacency[left][right] = adjacency[right][left] = 1
    if not _connected(adjacency):
        return False
    odd = sum(sum(row) % 2 for row in adjacency)
    return odd in (0, 2)


def decision_from_image(image: Image.Image) -> str:
    rgb_image = image.convert("RGB")
    width, height = rgb_image.size
    flat = list(rgb_image.getdata())
    node_mask = [_is_node(pixel) for pixel in flat]
    centers = _components(node_mask, width, height, minimum=45)
    if len(centers) != 18:
        raise ValueError(f"expected 18 visible waypoint components, found {len(centers)}")
    centers.sort(key=lambda point: (point[1], point[0]))
    lanes = [sorted(centers[index:index + 6]) for index in range(0, 18, 6)]
    label_pixels = {label: [index for index, pixel in enumerate(flat) if _label(pixel) == label] for label in LABELS}
    lane_medians = [sum(point[1] for point in lane) / 6.0 for lane in lanes]
    euler_labels = []
    for label in LABELS:
        indices = label_pixels[label]
        if len(indices) < 40:
            raise ValueError(f"missing colored network for {label}")
        median_y = sorted(index // width for index in indices)[len(indices) // 2]
        lane_index = min(range(3), key=lambda index: abs(lane_medians[index] - median_y))
        if _network_is_euler(flat, width, height, lanes[lane_index], label):
            euler_labels.append(label)
    if len(euler_labels) != 1:
        raise ValueError(f"ambiguous Euler-trail reconstruction: {euler_labels}")
    return euler_labels[0]
