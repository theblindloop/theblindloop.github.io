"""Deterministic analytic renderer for a visibly barred route-tree edge."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT, SCALE = 640, 480, 3
TERMINAL_NAMES = ("orange", "purple", "blue")
COLORS = {
    "background": (250, 249, 246),
    "dark": (35, 41, 49),
    "gray": (137, 145, 154),
    "green": (38, 158, 86),
    "orange": (235, 126, 30),
    "purple": (143, 70, 190),
    "blue": (38, 109, 216),
    "red": (218, 48, 55),
}

# The three leaf edges are deliberately separated and comparable in length.
BASE_NODES = [
    [62, 240], [148, 240], [238, 132], [238, 348],
    [352, 132], [352, 348], [508, 70], [536, 210], [512, 414],
]
BASE_EDGES = [[0, 1], [1, 2], [1, 3], [2, 4], [3, 5], [4, 6], [4, 7], [5, 8]]
LEAF_EDGES = ([4, 6], [4, 7], [5, 8])
TERMINAL_NODES = (6, 7, 8)


def _transform_point(point: list[int], transform: int) -> list[int]:
    x, y = point
    if transform == 1:
        return [WIDTH - x, y]
    if transform == 2:
        return [x, HEIGHT - y]
    if transform == 3:
        return [WIDTH - x, HEIGHT - y]
    return [x, y]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    desired = TERMINAL_NAMES[int(seed) % len(TERMINAL_NAMES)]
    colors = list(TERMINAL_NAMES)
    rng.shuffle(colors)
    target_slot = colors.index(desired)
    transform = (int(seed) // 3) % 4
    nodes = [_transform_point(point, transform) for point in BASE_NODES]
    # Small global translation is answer-independent and keeps generous margins.
    dx, dy = rng.choice((-8, 0, 8)), rng.choice((-7, 0, 7))
    nodes = [[x + dx, y + dy] for x, y in nodes]
    edges = [list(edge) for edge in BASE_EDGES]
    return {
        "nodes": nodes,
        "edges": edges,
        "start_node": 0,
        "terminal_nodes": list(TERMINAL_NODES),
        "terminal_colors": colors,
        "barred_edge": list(LEAF_EDGES[target_slot]),
        "transform": transform,
        "width": WIDTH,
        "height": HEIGHT,
    }


def _points(scene: dict, edge: list[int]) -> list[tuple[int, int]]:
    return [tuple(int(v) * SCALE for v in scene["nodes"][index]) for index in edge]


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width * SCALE, height * SCALE), COLORS["background"])
    draw = ImageDraw.Draw(image)
    for edge in scene["edges"]:
        draw.line(_points(scene, edge), fill=COLORS["dark"], width=9 * SCALE)

    # Gray dots explicitly declare every graph junction.
    degrees = [0] * len(scene["nodes"])
    for a, b in scene["edges"]:
        degrees[int(a)] += 1
        degrees[int(b)] += 1
    for index, degree in enumerate(degrees):
        if degree >= 2 and index != int(scene["start_node"]):
            x, y = scene["nodes"][index]
            draw.ellipse(((x - 8) * SCALE, (y - 8) * SCALE, (x + 8) * SCALE, (y + 8) * SCALE),
                         fill=COLORS["gray"], outline=COLORS["dark"], width=2 * SCALE)

    sx, sy = scene["nodes"][int(scene["start_node"])]
    draw.ellipse(((sx - 15) * SCALE, (sy - 15) * SCALE, (sx + 15) * SCALE, (sy + 15) * SCALE),
                 fill=COLORS["green"], outline=COLORS["dark"], width=2 * SCALE)
    draw.ellipse(((sx - 7) * SCALE, (sy - 7) * SCALE, (sx + 7) * SCALE, (sy + 7) * SCALE),
                 fill=COLORS["background"])

    for node, name in zip(scene["terminal_nodes"], scene["terminal_colors"]):
        x, y = scene["nodes"][int(node)]
        draw.rectangle(((x - 15) * SCALE, (y - 15) * SCALE, (x + 15) * SCALE, (y + 15) * SCALE),
                       fill=COLORS[str(name)], outline=COLORS["dark"], width=2 * SCALE)

    # Two separated red strokes cross and visibly close exactly one edge.
    a, b = [int(v) for v in scene["barred_edge"]]
    (x1, y1), (x2, y2) = scene["nodes"][a], scene["nodes"][b]
    dx, dy = x2 - x1, y2 - y1
    length = math.hypot(dx, dy)
    ux, uy = dx / length, dy / length
    px, py = -uy, ux
    mx, my = (x1 + x2) / 2.0, (y1 + y2) / 2.0
    for offset in (-8.0, 8.0):
        cx, cy = mx + offset * ux, my + offset * uy
        p1 = (int(round((cx - 16 * px) * SCALE)), int(round((cy - 16 * py) * SCALE)))
        p2 = (int(round((cx + 16 * px) * SCALE)), int(round((cy + 16 * py) * SCALE)))
        draw.line((p1, p2), fill=COLORS["red"], width=6 * SCALE)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def _reachable(scene: dict, removed: frozenset[int]) -> set[int]:
    adjacency = {index: [] for index in range(len(scene["nodes"]))}
    for a, b in scene["edges"]:
        if frozenset((int(a), int(b))) == removed:
            continue
        adjacency[int(a)].append(int(b))
        adjacency[int(b)].append(int(a))
    seen = {int(scene["start_node"])}
    stack = list(seen)
    while stack:
        node = stack.pop()
        for neighbor in adjacency[node]:
            if neighbor not in seen:
                seen.add(neighbor)
                stack.append(neighbor)
    return seen


def analytic_gold(scene: dict) -> str:
    removed = frozenset(int(v) for v in scene["barred_edge"])
    reachable = _reachable(scene, removed)
    lost = [name for node, name in zip(scene["terminal_nodes"], scene["terminal_colors"])
            if int(node) not in reachable]
    if len(lost) != 1:
        raise ValueError("closure does not disconnect exactly one terminal")
    return str(lost[0])


def margin(scene: dict) -> float:
    a, b = [int(v) for v in scene["barred_edge"]]
    x1, y1 = scene["nodes"][a]
    x2, y2 = scene["nodes"][b]
    half_edge = 0.5 * math.hypot(x2 - x1, y2 - y1)
    # Runner-up safety margin: endpoint clearance after closure footprint.
    return float(half_edge - 28.0)


def is_quarantined(scene: dict) -> bool:
    try:
        analytic_gold(scene)
    except (ValueError, KeyError, TypeError):
        return True
    return margin(scene) < 18.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    transformed = dict(scene)
    transformed["terminal_nodes"] = list(scene["terminal_nodes"][1:]) + [scene["terminal_nodes"][0]]
    transformed["terminal_colors"] = list(scene["terminal_colors"][1:]) + [scene["terminal_colors"][0]]
    return [("cycle_parallel_terminal_bindings", transformed)]
