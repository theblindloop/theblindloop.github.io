"""Independent inverse arm using only pixels from the final PNG."""

from __future__ import annotations

from collections import Counter, deque

import numpy as np
from PIL import Image

PALETTE = {
    "dark": (35, 41, 49), "gray": (137, 145, 154), "green": (38, 158, 86),
    "orange": (235, 126, 30), "purple": (143, 70, 190), "blue": (38, 109, 216),
    "red": (218, 48, 55),
}


def _mask(array: np.ndarray, rgb: tuple[int, int, int], radius: float = 54.0) -> np.ndarray:
    delta = array.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2, dtype=np.int64) <= radius * radius


def _components(mask: np.ndarray, minimum: int = 12) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    result: list[list[tuple[int, int]]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        pixels: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            pixels.append((cy, cx))
            for ny, nx in ((cy-1, cx), (cy+1, cx), (cy, cx-1), (cy, cx+1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(pixels) >= minimum:
            result.append(pixels)
    return result


def _labels(mask: np.ndarray) -> np.ndarray:
    labels = np.zeros(mask.shape, dtype=np.int16)
    height, width = mask.shape
    label = 0
    for y, x in zip(*np.nonzero(mask)):
        if labels[y, x]:
            continue
        label += 1
        queue = deque([(int(y), int(x))])
        labels[y, x] = label
        while queue:
            cy, cx = queue.popleft()
            for ny, nx in ((cy-1, cx), (cy+1, cx), (cy, cx-1), (cy, cx+1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and labels[ny, nx] == 0:
                    labels[ny, nx] = label
                    queue.append((ny, nx))
    return labels


def _owner(object_mask: np.ndarray, labels: np.ndarray) -> int:
    counts = Counter(int(value) for value in labels[object_mask] if value > 0)
    if not counts:
        raise ValueError("marker is detached from the visible network")
    return counts.most_common(1)[0][0]


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    if array.ndim != 3 or array.shape[0] < 300 or array.shape[1] < 400:
        raise ValueError("image is too small")
    red_components = _components(_mask(array, PALETTE["red"]), minimum=25)
    if len(red_components) != 2:
        raise ValueError("expected exactly two visible red closure bars")

    object_masks = {name: _mask(array, PALETTE[name]) for name in ("green", "orange", "purple", "blue")}
    for name, mask in object_masks.items():
        if len(_components(mask, minimum=80)) != 1:
            raise ValueError(f"expected one visible {name} marker")

    # Red closure pixels are intentionally absent: the raster itself now encodes
    # the counterfactual cut, while all route, junction, start, and terminal inks
    # form the graph whose components are measured.
    network = np.zeros(array.shape[:2], dtype=bool)
    for name in ("dark", "gray", "green", "orange", "purple", "blue"):
        network |= _mask(array, PALETTE[name])
    labels = _labels(network)
    start_label = _owner(object_masks["green"], labels)
    lost = [name for name in ("orange", "purple", "blue")
            if _owner(object_masks[name], labels) != start_label]
    if len(lost) != 1:
        raise ValueError("closure does not visibly isolate exactly one terminal")
    return lost[0]
