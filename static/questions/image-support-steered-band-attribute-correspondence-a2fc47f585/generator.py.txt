"""Connected spanning-plaque attribute-correspondence world.

One large connected area-like plaque spans nearly the whole image. At two
widely separated locations (left and right inside that same plaque) sit two
coloured blocks. The question is a correspondence decision: do the two distant
blocks of the single connected region carry the SAME attribute colour?

Latent state z = {color_left, color_right} plus the fixed geometry constants
below. Every latent field is directly painted into the image and read by the
analytic gold, so the parametrisation is minimal / injective: there is no
nontrivial latent symmetry (declared ``not_applicable``).

Decision: same = (color_left == color_right).
"""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw

RENDERER_VERSION = "band_attribute_correspondence-0.1.0"

CANVAS = 512
PLAQUE_MARGIN = 2
FRAME_WIDTH = 2

WHITE = (255, 255, 255)
PLAQUE = (234, 234, 234)
FRAME = (150, 150, 150)

PALETTE: dict[str, tuple[int, int, int]] = {
    "coral": (232, 93, 77),
    "teal": (31, 158, 141),
    "violet": (106, 81, 176),
}
PALETTE_NAMES = tuple(PALETTE)

BLOCK_RADIUS = 34
LEFT_CENTER = (150, 256)
RIGHT_CENTER = (362, 256)


def _plaque_rect() -> tuple[int, int, int, int]:
    return (
        PLAQUE_MARGIN,
        PLAQUE_MARGIN,
        CANVAS - PLAQUE_MARGIN,
        CANVAS - PLAQUE_MARGIN,
    )


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    color_left = rng.choice(PALETTE_NAMES)
    color_right = rng.choice(PALETTE_NAMES)
    return {"color_left": color_left, "color_right": color_right}


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (CANVAS, CANVAS), WHITE)
    draw = ImageDraw.Draw(img)
    draw.rectangle(_plaque_rect(), fill=PLAQUE, outline=FRAME, width=FRAME_WIDTH)
    for center, color in (
        (LEFT_CENTER, scene["color_left"]),
        (RIGHT_CENTER, scene["color_right"]),
    ):
        cx, cy = center
        draw.ellipse(
            [cx - BLOCK_RADIUS, cy - BLOCK_RADIUS, cx + BLOCK_RADIUS, cy + BLOCK_RADIUS],
            fill=PALETTE[color],
        )
    return img


def analytic_gold(scene: dict) -> str:
    return "yes" if scene["color_left"] == scene["color_right"] else "no"


def margin(scene: dict) -> float:
    a = np.array(PALETTE[scene["color_left"]], dtype=float)
    b = np.array(PALETTE[scene["color_right"]], dtype=float)
    return float(np.linalg.norm(a - b))


def is_quarantined(scene: dict) -> bool:
    # The palette uses well-separated, pairwise-distinct colours and blocks are
    # solid-filled discs, so colour identity is a crisp finite-resolution event.
    # Equal-colour scenes (margin 0) are a legitimate class, not an ambiguous
    # boundary, so no sampled scene is quarantined.
    return False


def latent_symmetries(scene: dict) -> list:
    return []


def scene_targets(scene: dict) -> dict:
    return {
        "color_left": scene["color_left"],
        "color_right": scene["color_right"],
    }
