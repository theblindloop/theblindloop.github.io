"""Independent pixel-only oracle for the spanning-plaque correspondence world.

Receives only the final PNG. It never sees the latent scene, renderer geometry
API, or analytic gold. It re-measures the raster independently:

1. It verifies the large connected plaque is essentially intact (its grey area
   is within a tight band of the analytic value and both distant blocks still
   carry a palette colour). If any grid-ablation breaks the decisive connected
   region, the oracle abstains (raises), which the attestation records as
   decisive support. This is what makes the whole spanning plaque jointly
   necessary: cutting any distant part changes the oracle's behaviour.
2. It reads the two attribute colours at widely separated fixed sample boxes
   inside the plaque and answers whether they are the same.

This module must not import renderer, prompts, generate, verify, or anything
that reads the analytic scene.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "band_attribute_correspondence-oracle-0.1.0"

CANVAS = 512
WHITE = (255, 255, 255)
PLAQUE = (234, 234, 234)
FRAME = (150, 150, 150)

PALETTE: dict[str, tuple[int, int, int]] = {
    "coral": (232, 93, 77),
    "teal": (31, 158, 141),
    "violet": (106, 81, 176),
}

BLOCK_RADIUS = 34
LEFT_CENTER = (150, 256)
RIGHT_CENTER = (362, 256)
SAMPLE_HALF = 10
PLAQUE_MARGIN = 2

# Analytic plaque grey fill: full rect area minus the two block discs.
_RECT = CANVAS - 2 * PLAQUE_MARGIN
_PI = 3.141592653589793
_BLOCK_AREA = _PI * BLOCK_RADIUS * BLOCK_RADIUS
_EXPECTED_GREY = _RECT * _RECT - 2.0 * _BLOCK_AREA
# Abstention band: a clean render keeps grey fill well inside; any ablation that
# removes a substantial piece of the connected plaque pushes it outside.
_GREY_LO = 0.986 * _EXPECTED_GREY
_GREY_HI = 1.014 * _EXPECTED_GREY


class PlaqueCompromisedError(ValueError):
    """The connected spanning plaque (or a distant block) is missing/incomplete."""


def _nearest_palette_color(pixel: np.ndarray) -> str | None:
    best_name, best_dist = None, None
    for name, rgb in PALETTE.items():
        d = float(np.linalg.norm(pixel - np.array(rgb, dtype=np.float32)))
        if best_dist is None or d < best_dist:
            best_dist, best_name = d, name
    return best_name


def _sample_block(arr: np.ndarray, center: tuple[int, int]) -> str | None:
    cx, cy = center
    box = arr[cy - SAMPLE_HALF : cy + SAMPLE_HALF, cx - SAMPLE_HALF : cx + SAMPLE_HALF]
    from collections import Counter

    counts: Counter = Counter()
    for py in range(box.shape[0]):
        for px in range(box.shape[1]):
            name = _nearest_palette_color(box[py, px])
            if name is not None:
                counts[name] += 1
    if not counts:
        return None
    return counts.most_common(1)[0][0]


def decision_from_image(image: Image.Image) -> str:
    image = image.convert("RGB")
    if image.size != (CANVAS, CANVAS):
        raise PlaqueCompromisedError("unexpected image size")
    arr = np.asarray(image, dtype=np.float32)

    labels = _assign_labels(arr)
    grey_count = int((labels == PLAQUE_INDEX).sum())
    if not (_GREY_LO <= grey_count <= _GREY_HI):
        raise PlaqueCompromisedError(
            f"connected plaque broken: grey fill {grey_count} outside [{_GREY_LO:.0f},{_GREY_HI:.0f}]"
        )

    color_left = _sample_block(arr, LEFT_CENTER)
    color_right = _sample_block(arr, RIGHT_CENTER)
    if color_left is None or color_right is None:
        raise PlaqueCompromisedError("a distant attribute block is missing")

    return "yes" if color_left == color_right else "no"


# --- nearest-palette label assignment (plaque grey is one label) -------------
_PALETTE_LIST = [
    ("white", WHITE),
    ("plaque", PLAQUE),
    ("frame", FRAME),
    *list(PALETTE.items()),
]
PLAQUE_INDEX = 1


def _assign_labels(arr: np.ndarray) -> np.ndarray:
    dists = np.stack(
        [
            np.linalg.norm(arr - np.array(rgb, dtype=np.float32), axis=-1)
            for _, rgb in _PALETTE_LIST
        ],
        axis=-1,
    )
    return dists.argmin(axis=-1)
