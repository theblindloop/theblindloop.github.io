"""Independent PNG-only inverse arm: segment nodes/links and solve the cut."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE = {
    "orange": (242, 142, 43),
    "purple": (143, 99, 199),
    "blue": (78, 121, 167),
    "hub": (77, 182, 172),
    "terminal": (122, 132, 145),
    "dark": (44, 51, 59),
}


def _mask(arr: np.ndarray, rgb: tuple[int, int, int], radius: float = 35.0) -> np.ndarray:
    delta = arr.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2, dtype=np.int32) <= radius * radius


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    work = mask.copy()
    h, w = work.shape
    found: list[np.ndarray] = []
    for y0, x0 in zip(*np.nonzero(work)):
        if not work[y0, x0]:
            continue
        work[y0, x0] = False
        stack = [(int(y0), int(x0))]
        points = []
        while stack:
            y, x = stack.pop()
            points.append((y, x))
            for yy, xx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= yy < h and 0 <= xx < w and work[yy, xx]:
                    work[yy, xx] = False
                    stack.append((yy, xx))
        if len(points) >= minimum:
            found.append(np.asarray(points, dtype=np.float64))
    return found


def _centers(arr: np.ndarray, name: str, expected: int) -> list[tuple[float, float, str]]:
    comps = _components(_mask(arr, PALETTE[name], 38.0), 100)
    if len(comps) != expected:
        raise ValueError(f"expected {expected} {name} markers, found {len(comps)}")
    return [(float(c[:, 1].mean()), float(c[:, 0].mean()), name) for c in comps]


def _graph_from_pixels(arr: np.ndarray):
    nodes = []
    for name in ("orange", "purple", "blue"):
        nodes.extend(_centers(arr, name, 1))
    nodes.extend(_centers(arr, "hub", 4))
    nodes.extend(_centers(arr, "terminal", 6))

    links = _components(_mask(arr, PALETTE["dark"], 30.0), 40)
    if len(links) != 12:
        raise ValueError(f"expected 12 separated dark links, found {len(links)}")
    adjacency = {i: set() for i in range(len(nodes))}
    xy = np.asarray([(n[0], n[1]) for n in nodes])
    for component in links:
        pts = component[:, ::-1]
        # Minimum squared pixel distance from each marker center to this link.
        distances = np.min(np.sum((pts[:, None, :] - xy[None, :, :]) ** 2, axis=2), axis=0)
        order = np.argsort(distances)
        a, b = int(order[0]), int(order[1])
        if distances[b] > 24.0 ** 2:
            raise ValueError("a dark link has no second visible endpoint")
        adjacency[a].add(b)
        adjacency[b].add(a)
    if sum(len(v) for v in adjacency.values()) != 24:
        raise ValueError("link endpoint association is not a twelve-edge tree")
    return nodes, adjacency


def _terminal_counts_after(adjacency: dict[int, set[int]], removed: int, terminals: set[int]) -> list[int]:
    unseen = set(adjacency) - {removed}
    counts = []
    while unseen:
        start = next(iter(unseen))
        unseen.remove(start)
        queue = deque([start])
        component = {start}
        while queue:
            node = queue.popleft()
            for nxt in adjacency[node] - {removed}:
                if nxt in unseen:
                    unseen.remove(nxt)
                    component.add(nxt)
                    queue.append(nxt)
        count = len(component & terminals)
        if count:
            counts.append(count)
    return sorted(counts)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    nodes, adjacency = _graph_from_pixels(arr)
    terminals = {i for i, node in enumerate(nodes) if node[2] == "terminal"}
    winners = []
    for index, node in enumerate(nodes):
        if node[2] in {"orange", "purple", "blue"}:
            if _terminal_counts_after(adjacency, index, terminals) == [3, 3]:
                winners.append(node[2])
    if len(winners) != 1:
        raise ValueError(f"expected one balanced checkpoint, found {winners}")
    return winners[0]
