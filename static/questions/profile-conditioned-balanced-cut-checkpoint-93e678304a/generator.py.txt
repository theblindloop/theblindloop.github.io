"""Deterministic raster world for a balanced vertex cut in a visible tree."""

from __future__ import annotations

import itertools
import random
from copy import deepcopy

from PIL import Image, ImageDraw

SIZE = 512
SS = 3
BG = (248, 247, 243)
DARK = (44, 51, 59)
HUB = (77, 182, 172)
TERMINAL = (122, 132, 145)
COLORS = {
    "orange": (242, 142, 43),
    "purple": (143, 99, 199),
    "blue": (78, 121, 167),
}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    balanced_slot = int(seed) % 3
    patterns = ([3, 1, 1, 1], [1, 2, 2, 1], [1, 1, 1, 3])
    colors = list(COLORS)
    rng.shuffle(colors)
    return {
        "leaf_counts": list(patterns[balanced_slot]),
        "checkpoint_colors": colors,
        "rotation": rng.randrange(4),
    }


def _scores(scene: dict) -> list[int]:
    counts = [int(v) for v in scene["leaf_counts"]]
    total = sum(counts)
    return [abs(2 * sum(counts[: i + 1]) - total) for i in range(3)]


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    winner = min(range(3), key=scores.__getitem__)
    if scores.count(scores[winner]) != 1:
        raise ValueError("balanced checkpoint is not unique")
    return str(scene["checkpoint_colors"][winner])


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene))
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    counts = [int(v) for v in scene["leaf_counts"]]
    colors = list(scene["checkpoint_colors"])
    scores = _scores(scene) if len(counts) == 4 and all(v > 0 for v in counts) else []
    return not (
        sum(counts) == 6
        and sorted(colors) == sorted(COLORS)
        and len(scores) == 3
        and scores.count(0) == 1
        and sorted(scores)[1] - sorted(scores)[0] >= 2
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["leaf_counts"] = list(reversed(scene["leaf_counts"]))
    twin["checkpoint_colors"] = list(reversed(scene["checkpoint_colors"]))
    twin["rotation"] = (int(scene["rotation"]) + 2) % 4
    return [("reverse_representation_half_turn", twin)]


def _rotate(point: tuple[int, int], turns: int) -> tuple[int, int]:
    x, y = point
    for _ in range(turns % 4):
        x, y = SIZE - x, y
        x, y = y, x
    return x, y


def _leaf_offsets(zone: int, count: int) -> list[tuple[int, int]]:
    if count == 3 and zone == 0:
        return [(-68, -70), (-86, 0), (-68, 70)]
    if count == 3 and zone == 3:
        return [(68, 70), (86, 0), (68, -70)]
    if count == 2:
        return [(0, -76), (0, 76)]
    if count == 1 and zone == 0:
        return [(-82, 0)]
    if count == 1 and zone == 3:
        return [(82, 0)]
    if count == 1:
        return [(0, -76 if zone % 2 else 76)]
    raise ValueError("unsupported leaf-count layout")


def _ellipse(draw: ImageDraw.ImageDraw, center: tuple[int, int], radius: int, fill, outline=None, width=1):
    x, y = center
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=fill, outline=outline, width=width)


def render(scene: dict) -> Image.Image:
    counts = [int(v) for v in scene["leaf_counts"]]
    colors = [str(v) for v in scene["checkpoint_colors"]]
    turns = int(scene["rotation"]) % 4
    if len(counts) != 4 or len(colors) != 3:
        raise ValueError("malformed scene")

    hubs0 = [(104, 256), (205, 256), (307, 256), (408, 256)]
    checks0 = [(154, 256), (256, 256), (358, 256)]
    leaves0: list[tuple[int, int]] = []
    leaf_edges0: list[tuple[tuple[int, int], tuple[int, int]]] = []
    for zone, (hub, count) in enumerate(zip(hubs0, counts)):
        for dx, dy in _leaf_offsets(zone, count):
            leaf = (hub[0] + dx, hub[1] + dy)
            leaves0.append(leaf)
            leaf_edges0.append((hub, leaf))

    hubs = [_rotate(p, turns) for p in hubs0]
    checks = [_rotate(p, turns) for p in checks0]
    leaves = [_rotate(p, turns) for p in leaves0]
    spine0 = list(itertools.chain.from_iterable(
        ((hubs0[i], checks0[i]), (checks0[i], hubs0[i + 1])) for i in range(3)
    ))
    edges = [(_rotate(a, turns), _rotate(b, turns)) for a, b in spine0 + leaf_edges0]

    img = Image.new("RGB", (SIZE * SS, SIZE * SS), BG)
    draw = ImageDraw.Draw(img)
    def sc(p):
        return (p[0] * SS, p[1] * SS)
    for a, b in edges:
        draw.line((sc(a), sc(b)), fill=DARK, width=7 * SS)

    for p in hubs:
        _ellipse(draw, sc(p), 10 * SS, HUB, outline=BG, width=2 * SS)
    for p, name in zip(checks, colors):
        _ellipse(draw, sc(p), 15 * SS, COLORS[name], outline=BG, width=3 * SS)
    for x, y in leaves:
        x *= SS
        y *= SS
        r = 11 * SS
        draw.rectangle((x - r, y - r, x + r, y + r), fill=TERMINAL, outline=BG, width=3 * SS)

    return img.resize((SIZE, SIZE), Image.Resampling.LANCZOS)
