"""Independent PNG-only inverse for visible colored traffic-signal lamps."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    pass


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    """Return area and bounding box for every 4-connected True component."""
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=np.uint8)
    output: list[tuple[int, int, int, int, int]] = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = 1
        area = 0
        min_x = max_x = sx
        min_y = max_y = sy
        while queue:
            y, x = queue.popleft()
            area += 1
            min_x, max_x = min(min_x, x), max(max_x, x)
            min_y, max_y = min(min_y, y), max(max_y, y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = 1
                    queue.append((ny, nx))
        output.append((area, min_x, min_y, max_x, max_y))
    return output


def decision_from_image(image: Image.Image) -> str:
    """Count large saturated near-circular components without scene information."""
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    maximum = array.max(axis=2)
    minimum = array.min(axis=2)
    saturation_span = maximum - minimum
    # Lamps are the only strongly chromatic regions; antialias pixels are included.
    mask = (saturation_span >= 52) & (maximum >= 125) & (minimum <= 205)
    disks = 0
    for area, x0, y0, x1, y1 in _components(mask):
        width = x1 - x0 + 1
        height = y1 - y0 + 1
        if area < 2600 or width < 65 or height < 65:
            continue
        aspect = width / height
        fill = area / float(width * height)
        if 0.82 <= aspect <= 1.22 and 0.62 <= fill <= 0.88:
            disks += 1
    if disks not in (2, 3):
        raise OracleAbstention(f"expected two or three visible disk-like lamps, found {disks}")
    return str(disks)
