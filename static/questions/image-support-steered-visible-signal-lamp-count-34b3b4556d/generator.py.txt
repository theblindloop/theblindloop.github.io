"""Deterministic latent sampler and raster renderer for traffic-signal lamp count."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 768
HEIGHT = 512
SUPERSAMPLE = 3
PALETTES = (
    ((222, 52, 62), (246, 177, 45), (45, 174, 96)),
    ((45, 118, 220), (232, 76, 137), (241, 156, 45)),
    ((120, 74, 205), (32, 171, 165), (231, 82, 60)),
    ((218, 65, 50), (47, 148, 207), (238, 173, 44)),
)


def sample_scene(seed: int) -> dict:
    """Return one balanced, fully visible control/counterfactual scene."""
    rng = random.Random(int(seed))
    lamp_count = 2 + (int(seed) & 1)
    return {
        "lamp_count": lamp_count,
        "center_x": rng.randint(424, 440),
        "center_y": rng.randint(216, 232),
        "scale": rng.choice([0.94, 0.97, 1.0, 1.03, 1.06]),
        "lamp_radius": rng.randint(43, 50),
        "palette": rng.randrange(len(PALETTES)),
        "color_offset": rng.randrange(3),
        "cover_style": rng.randrange(3),
        "bolt_layout": rng.randrange(4),
    }


def _ellipse(draw: ImageDraw.ImageDraw, box: tuple[float, float, float, float], **kwargs) -> None:
    draw.ellipse(tuple(round(v * SUPERSAMPLE) for v in box), **kwargs)


def _rect(draw: ImageDraw.ImageDraw, box: tuple[float, float, float, float], **kwargs) -> None:
    draw.rounded_rectangle(
        tuple(round(v * SUPERSAMPLE) for v in box),
        radius=round(14 * SUPERSAMPLE),
        **kwargs,
    )


def render(scene: dict) -> Image.Image:
    """Render a neutral horizontal signal; chromatic disks are the only lamps."""
    lamp_count = int(scene["lamp_count"])
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    scale = float(scene["scale"])
    radius = float(scene["lamp_radius"]) * scale
    palette_index = int(scene["palette"])
    color_offset = int(scene["color_offset"])
    cover_style = int(scene["cover_style"])
    bolt_layout = int(scene["bolt_layout"])

    image = Image.new("RGB", (WIDTH * SUPERSAMPLE, HEIGHT * SUPERSAMPLE), (247, 246, 242))
    draw = ImageDraw.Draw(image)

    # A neutral icon world: pavement shadow, post, and a single signal housing.
    draw.ellipse(
        (180 * SUPERSAMPLE, 443 * SUPERSAMPLE, 676 * SUPERSAMPLE, 477 * SUPERSAMPLE),
        fill=(223, 222, 217),
    )
    post_w = 25 * scale
    draw.rounded_rectangle(
        (
            round((cx - post_w / 2) * SUPERSAMPLE),
            round((cy + 87 * scale) * SUPERSAMPLE),
            round((cx + post_w / 2) * SUPERSAMPLE),
            round(452 * SUPERSAMPLE),
        ),
        radius=round(6 * SUPERSAMPLE),
        fill=(93, 99, 104),
        outline=(49, 54, 58),
        width=round(3 * SUPERSAMPLE),
    )
    housing = (
        cx - 286 * scale,
        cy - 88 * scale,
        cx + 286 * scale,
        cy + 88 * scale,
    )
    _rect(draw, housing, fill=(66, 71, 76), outline=(31, 34, 37), width=4 * SUPERSAMPLE)
    inner = (
        cx - 272 * scale,
        cy - 74 * scale,
        cx + 272 * scale,
        cy + 74 * scale,
    )
    _rect(draw, inner, fill=(178, 182, 181), outline=(215, 217, 213), width=2 * SUPERSAMPLE)

    positions = (cx - 192 * scale, cx, cx + 192 * scale)
    active = (0, 2) if lamp_count == 2 else (0, 1, 2)
    palette = PALETTES[palette_index % len(PALETTES)]
    for index, px in enumerate(positions):
        bezel_r = radius + 9 * scale
        _ellipse(
            draw,
            (px - bezel_r, cy - bezel_r, px + bezel_r, cy + bezel_r),
            fill=(42, 46, 49),
            outline=(25, 28, 30),
            width=2 * SUPERSAMPLE,
        )
        if index in active:
            color = palette[(index + color_offset) % 3]
            _ellipse(
                draw,
                (px - radius, cy - radius, px + radius, cy + radius),
                fill=color,
                outline=tuple(max(0, value - 45) for value in color),
                width=2 * SUPERSAMPLE,
            )
            # Small neutral reflection remains a hole in the same colored component.
            highlight = 7 * scale
            _ellipse(
                draw,
                (
                    px - radius * 0.44 - highlight,
                    cy - radius * 0.42 - highlight,
                    px - radius * 0.44 + highlight,
                    cy - radius * 0.42 + highlight,
                ),
                fill=(241, 241, 234),
            )
        else:
            cover_r = radius - 2 * scale
            _ellipse(
                draw,
                (px - cover_r, cy - cover_r, px + cover_r, cy + cover_r),
                fill=(126, 132, 132),
                outline=(89, 94, 95),
                width=2 * SUPERSAMPLE,
            )
            if cover_style == 0:
                draw.line(
                    ((px - 18 * scale) * SUPERSAMPLE, cy * SUPERSAMPLE,
                     (px + 18 * scale) * SUPERSAMPLE, cy * SUPERSAMPLE),
                    fill=(91, 96, 97), width=4 * SUPERSAMPLE,
                )
            elif cover_style == 1:
                draw.line(
                    (px * SUPERSAMPLE, (cy - 18 * scale) * SUPERSAMPLE,
                     px * SUPERSAMPLE, (cy + 18 * scale) * SUPERSAMPLE),
                    fill=(91, 96, 97), width=4 * SUPERSAMPLE,
                )
            else:
                for angle in (0.0, math.pi):
                    bx = px + math.cos(angle) * 19 * scale
                    by = cy + math.sin(angle) * 19 * scale
                    _ellipse(draw, (bx - 4, by - 4, bx + 4, by + 4), fill=(82, 87, 88))

    # Gray housing bolts are varied independently and are explicitly non-lamps.
    bolt_sets = (
        ((-252, -56), (252, 56)),
        ((-252, 56), (252, -56)),
        ((-252, -56), (252, -56)),
        ((-252, 56), (252, 56)),
    )
    for dx, dy in bolt_sets[bolt_layout % len(bolt_sets)]:
        bx, by = cx + dx * scale, cy + dy * scale
        _ellipse(draw, (bx - 5, by - 5, bx + 5, by + 5), fill=(43, 47, 49))

    image = image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
    return image


def analytic_gold(scene: dict) -> str:
    return str(int(scene["lamp_count"]))


def margin(scene: dict) -> float:
    """Conservative runner-up clearance in rendered pixels."""
    radius = float(scene["lamp_radius"]) * float(scene["scale"])
    separation_gap = 192.0 * float(scene["scale"]) - 2.0 * radius
    area_radius_clearance = radius - 28.0
    housing_clearance = 74.0 * float(scene["scale"]) - radius
    return round(min(separation_gap, area_radius_clearance, housing_clearance), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 10.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
