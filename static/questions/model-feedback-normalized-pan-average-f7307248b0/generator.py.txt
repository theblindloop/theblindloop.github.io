"""Deterministic renderer and analytic arm for the normalized-pan average world.

Two framed boxes, A on the left and B on the right, hold coloured disks; the two
boxes need not hold the same number of disks.  A key strip across the top shows
three colour swatches; beside each swatch a row of one, two, or three dark dots
encodes that colour's weight.  The operation is a quantitative relation: for
each box compute total weight = sum over that box's disks of (the disk's colour
weight, read from the key), then divide that total by the box's disk count to
give the box's average weight per disk, and decide which of the two composed
averages is larger.

The sampler always builds scenes in which the raw disk-count comparison and the
raw weighted-total comparison are both opposite to the average comparison, so a
box can hold more disks AND the larger raw total yet still have the smaller
average weight (a dilute, spread-thin pan).  Only the averaged, size-normalized
relation carries the answer.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 780
HEIGHT = 840
COLORS = ["crimson", "teal", "amber"]
RGB = {
    "crimson": (204, 44, 64),
    "teal": (0, 140, 150),
    "amber": (232, 158, 32),
}
DARK = (45, 45, 45)
OUTLINE = (70, 70, 70)
BACKGROUND = (250, 250, 250)
SWATCH_XS = [130, 390, 650]
KEY_CY = 120
SWATCH_SIZE = 44
DOT_R = 9
DOT_SLOT = 26
DOT_X0_OFFSET = 14
DISK_R = 32
LEFT_CENTERS = [(155, 440), (155, 560), (275, 440), (275, 560)]
RIGHT_CENTERS = [(505, 440), (505, 560), (625, 440), (625, 560)]
MIN_GAP = 1.0
MAX_DISKS = 4


def _load_font(size: int):
    for path in (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf",
        "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
    ):
        try:
            return ImageFont.truetype(path, size)
        except OSError:
            continue
    return ImageFont.load_default()


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    weights = [1, 2, 3]
    rng.shuffle(weights)
    weight_of = {color: int(weight) for color, weight in zip(COLORS, weights)}
    last = None
    for _ in range(6000):
        n_a = rng.choice([2, 3, 4])
        n_b = rng.choice([2, 3, 4])
        if n_a == n_b:
            continue
        box_a = [rng.choice(COLORS) for _ in range(n_a)]
        box_b = [rng.choice(COLORS) for _ in range(n_b)]
        total_a = sum(int(weight_of[c]) for c in box_a)
        total_b = sum(int(weight_of[c]) for c in box_b)
        avg_a = total_a / n_a
        avg_b = total_b / n_b
        if abs(avg_a - avg_b) < MIN_GAP:
            continue
        cs = (n_a > n_b) - (n_a < n_b)
        ts = (total_a > total_b) - (total_a < total_b)
        avs = (avg_a > avg_b) - (avg_a < avg_b)
        if cs == 0 or ts == 0 or avs == 0:
            continue
        if avs == cs or avs == ts:
            continue
        return {
            "canvas_size": [WIDTH, HEIGHT],
            "weights": {c: int(weight_of[c]) for c in COLORS},
            "box_a": box_a,
            "box_b": box_b,
            "disk_r": DISK_R,
        }
    raise RuntimeError("could not sample a valid normalized-pan scene")


def _totals(scene: dict) -> tuple[int, int]:
    weight = scene["weights"]
    return sum(int(weight[c]) for c in scene["box_a"]), sum(
        int(weight[c]) for c in scene["box_b"]
    )


def _averages(scene: dict) -> tuple[float, float]:
    total_a, total_b = _totals(scene)
    return total_a / len(scene["box_a"]), total_b / len(scene["box_b"])


def _draw_key(draw: ImageDraw.ImageDraw, weights: dict) -> None:
    for cx, color in zip(SWATCH_XS, COLORS):
        x0, y0 = cx - SWATCH_SIZE // 2, KEY_CY - SWATCH_SIZE // 2
        draw.rectangle(
            (x0, y0, x0 + SWATCH_SIZE, y0 + SWATCH_SIZE), fill=RGB[color], outline=OUTLINE
        )
        weight = int(weights[color])
        for k in range(weight):
            dot_cx = cx + SWATCH_SIZE // 2 + DOT_X0_OFFSET + k * DOT_SLOT
            draw.ellipse(
                (dot_cx - DOT_R, KEY_CY - DOT_R, dot_cx + DOT_R, KEY_CY + DOT_R), fill=DARK
            )


def _draw_box(draw: ImageDraw.ImageDraw, x0: int, x1: int) -> None:
    draw.rectangle((x0, 300, x1, 700), outline=OUTLINE, width=4)


def _draw_disks(draw: ImageDraw.ImageDraw, colors: list, centers: list) -> None:
    for color, (cx, cy) in zip(colors, centers):
        r = DISK_R
        draw.ellipse((cx - r, cy - r, cx + r, cy + r), fill=RGB[color], outline=OUTLINE)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_key(draw, scene["weights"])
    _draw_box(draw, 70, 360)
    _draw_box(draw, 420, 710)
    _draw_disks(draw, scene["box_a"], LEFT_CENTERS[: len(scene["box_a"])])
    _draw_disks(draw, scene["box_b"], RIGHT_CENTERS[: len(scene["box_b"])])
    font = _load_font(30)
    draw.text((205, 745), "A", font=font, fill=(40, 40, 40))
    draw.text((555, 745), "B", font=font, fill=(40, 40, 40))
    return image


def analytic_gold(scene: dict) -> str:
    avg_a, avg_b = _averages(scene)
    return "A" if avg_a > avg_b else "B"


def margin(scene: dict) -> float:
    avg_a, avg_b = _averages(scene)
    return float(abs(avg_a - avg_b))


def is_quarantined(scene: dict) -> bool:
    try:
        weight = scene["weights"]
        if set(weight) != set(COLORS):
            return True
        if sorted(int(weight[c]) for c in COLORS) != [1, 2, 3]:
            return True
        if not (2 <= len(scene["box_a"]) <= MAX_DISKS and 2 <= len(scene["box_b"]) <= MAX_DISKS):
            return True
        if len(scene["box_a"]) == len(scene["box_b"]):
            return True
        if any(c not in COLORS for c in scene["box_a"] + scene["box_b"]):
            return True
        avg_a, avg_b = _averages(scene)
        if abs(avg_a - avg_b) < MIN_GAP:
            return True
        total_a, total_b = _totals(scene)
        cs = (len(scene["box_a"]) > len(scene["box_b"])) - (len(scene["box_a"]) < len(scene["box_b"]))
        ts = (total_a > total_b) - (total_a < total_b)
        avs = (avg_a > avg_b) - (avg_a < avg_b)
        if cs == 0 or ts == 0 or avs == 0:
            return True
        if avs == cs or avs == ts:
            return True
        return False
    except (KeyError, TypeError):
        return True


def latent_symmetries(scene: dict) -> list:
    return []
