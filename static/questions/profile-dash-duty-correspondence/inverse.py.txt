"""Independent pixel-only inverse arm for the dash-to-gap duty-ratio
correspondence world.

Receives ONLY the final rasterized PNG and the public question contract (the
visual spec). It never imports the renderer, prompt, generator, verifier, or
analytic-gold code, and never reads scene data.

Public visual spec (this file's own specification):

  * the image contains four horizontal dashed bars, each a strip of solid
    rectangular dashes separated by gaps; the top bar is the gray reference,
    the other three are red, green and blue candidates;
  * each bar has a dash-to-gap ratio (length of one solid dash divided by the
    empty gap to its right): short/equal/long;
  * the task: decide which candidate color has the same dash-to-gap ratio as
    the gray reference bar, reading the ratio from pixels alone.

The oracle recovers each bar's pattern by color, measures dash and gap pixel
runs along the bar's central row, forms the dash/gap ratio, and selects the
color whose ratio equals the gray reference's.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "dash_duty_correspondence-oracle-0.1.0"

_PALETTE = {
    "gray": (120, 120, 120),
    "red": (225, 45, 45),
    "green": (0, 170, 85),
    "blue": (45, 65, 225),
}
_WHITE = np.array([255.0, 255.0, 255.0])
_WHITE_DIST = 25.0
_COLOR_DIST = 60.0
_MIN_PIX = 30
_TOL_LOG = 0.35


def _duty_ratio(pts: np.ndarray) -> float:
    ys = pts[:, 0]
    xs = pts[:, 1]
    if len(pts) < _MIN_PIX:
        raise ValueError("color cluster has too few pixels")
    row = int(round(float(np.median(ys))))
    row_xs = np.sort(xs[ys == row])
    if row_xs.size == 0:
        raise ValueError("no pixels on the measurement row")
    starts = [row_xs[0]]
    ends = []
    for i in range(1, row_xs.size):
        if row_xs[i] - row_xs[i - 1] > 1:
            ends.append(row_xs[i - 1])
            starts.append(row_xs[i])
    ends.append(row_xs[-1])
    dashes = [end - start + 1 for start, end in zip(starts, ends)]
    if len(dashes) < 2:
        raise ValueError("not enough dashes to measure a ratio")
    gaps = [starts[k + 1] - ends[k] - 1 for k in range(len(dashes) - 1)]
    if not gaps:
        raise ValueError("no gaps measured")
    D = float(np.median(dashes))
    G = float(np.median(gaps))
    if G <= 0:
        raise ValueError("degenerate gap")
    return D / G


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.float32)
    white = np.linalg.norm(arr - _WHITE, axis=-1) < _WHITE_DIST

    assign = np.full(arr.shape[:2], -1, dtype=int)
    best = np.full(arr.shape[:2], np.inf)
    names = list(_PALETTE)
    for i, name in enumerate(names):
        dist = np.linalg.norm(arr - np.array(_PALETTE[name]), axis=-1)
        closer = dist < best
        best[closer] = dist[closer]
        assign[closer] = i
    assign[white] = -1

    ratios: dict[str, float] = {}
    for i, name in enumerate(names):
        pts = np.argwhere(assign == i)
        if len(pts) < _MIN_PIX:
            continue
        ratios[name] = _duty_ratio(pts)

    if "gray" not in ratios:
        raise ValueError("reference gray bar not found")
    ref_ratio = ratios["gray"]
    matched = None
    for name in ("red", "green", "blue"):
        if name not in ratios:
            continue
        if abs(np.log2(ratios[name] / ref_ratio)) < _TOL_LOG:
            if matched is not None:
                raise ValueError("ambiguous: multiple candidates match the reference ratio")
            matched = name
    if matched is None:
        raise ValueError("no candidate reproduced the reference dash-to-gap ratio")
    return matched


if __name__ == "__main__":
    import sys

    img = Image.open(sys.argv[1]).convert("RGB")
    print(decision_from_image(img))
