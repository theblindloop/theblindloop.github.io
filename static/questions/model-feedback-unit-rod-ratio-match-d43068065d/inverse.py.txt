"""Scene-free pixel oracle for the unit-rod ratio-match world.

Receives only the rendered PNG. It segments the two flat ink colours, splits
the canvas into the framed reference strip and the numbered-panel strip by
row, pairs each panel's bar with the rod that lies to its right, forms the
dimensionless quotient bar-height / rod-length for every panel and for the
reference, and reports the panel whose quotient equals the reference quotient.

It imports no renderer, prompt, generator, verifier or gold code and never
sees latent scene data.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_BAR = (38, 50, 78)
_ROD = (230, 126, 34)
_SPLIT_ROW = 316  # everything above belongs to the framed reference strip


def _mask(arr: np.ndarray, colour: tuple[int, int, int]) -> np.ndarray:
    return (
        (arr[:, :, 0] == colour[0])
        & (arr[:, :, 1] == colour[1])
        & (arr[:, :, 2] == colour[2])
    )


def _x_runs(mask: np.ndarray) -> list[tuple[int, int]]:
    cols = mask.any(axis=0)
    runs: list[tuple[int, int]] = []
    start = None
    for x, on in enumerate(cols):
        if on and start is None:
            start = x
        elif not on and start is not None:
            runs.append((start, x - 1))
            start = None
    if start is not None:
        runs.append((start, len(cols) - 1))
    return runs


def _heights(mask: np.ndarray) -> list[int]:
    out = []
    for x0, x1 in _x_runs(mask):
        rows = mask[:, x0 : x1 + 1].any(axis=1)
        idx = np.flatnonzero(rows)
        out.append(int(idx[-1] - idx[0] + 1))
    return out


def _widths(mask: np.ndarray) -> list[int]:
    return [x1 - x0 + 1 for x0, x1 in _x_runs(mask)]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    bars = _mask(arr, _BAR)
    rods = _mask(arr, _ROD)

    ref_bar = _heights(bars[:_SPLIT_ROW])
    ref_rod = _widths(rods[:_SPLIT_ROW])
    if len(ref_bar) != 1 or len(ref_rod) != 1:
        return "panel-0"
    target = ref_bar[0] / ref_rod[0]

    panel_bars = _heights(bars[_SPLIT_ROW:])
    panel_rods = _widths(rods[_SPLIT_ROW:])
    if len(panel_bars) != len(panel_rods) or not panel_bars:
        return "panel-0"

    ratios = [h / s for h, s in zip(panel_bars, panel_rods)]
    best = min(range(len(ratios)), key=lambda i: abs(ratios[i] - target))
    return f"panel-{best + 1}"
