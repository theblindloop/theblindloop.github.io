"""Unit-rod ratio match world.

Six panels, each holding one vertical dark bar and one horizontal orange rod.
The rod is that panel's private unit of length: the panel's reading is the
dimensionless quotient bar-height / rod-length, an exact small integer. The
framed reference panel at the top carries its own bar and rod, hence its own
reading. Exactly one of the five numbered panels reads the same number of its
own rods as the reference reads of its own rod.

Nothing about absolute size answers the question: the sampler forces the bar
closest in pixels to the reference bar, the rod closest in pixels to the
reference rod, the tallest bar and the largest reading all to belong to
non-answer panels. Only the quotient identifies the panel.

All ink is axis-aligned flat-filled rectangles, so the raster is
byte-deterministic and free of anti-aliasing.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 1560
HEIGHT = 872

WHITE = (255, 255, 255)
BAR = (38, 50, 78)
ROD = (230, 126, 34)
REF_FRAME = (190, 30, 140)
PANEL_FRAME = (120, 124, 132)

# reference (framed) panel
REF_BOX = (556, 18, 1004, 300)          # x0, y0, x1, y1
REF_FRAME_W = 7
REF_BASE_Y = 278                        # baseline the reference bar stands on
REF_BAR_X = 596
REF_BAR_W = 46
REF_ROD_X = 706
REF_ROD_Y = 92                          # top edge of the reference rod
ROD_T = 26                              # rod thickness (both strips)

# numbered panels
N_PANELS = 5
PANEL_Y0 = 332
PANEL_Y1 = 852
PANEL_X0 = 26
PANEL_W = 292
PANEL_GAP = 12
PANEL_FRAME_W = 4
BASE_Y = 806                            # baseline every numbered bar stands on
BAR_W = 46

MAX_BAR = 400
MAX_REF_BAR = 240
MIN_ROD = 34
MAX_ROD = 100
MIN_REF_ROD = 30
MAX_REF_ROD = 78
QUARANTINE_MARGIN = 40.0


def _panel_x(i: int) -> int:
    return PANEL_X0 + i * (PANEL_W + PANEL_GAP)


def _readings(scene: dict) -> list[int]:
    return [h // s for h, s in zip(scene["bars"], scene["rods"])]


def _ref_reading(scene: dict) -> int:
    return scene["ref_bar"] // scene["ref_rod"]


def _answer_index(scene: dict) -> int:
    target = _ref_reading(scene)
    hits = [i for i, v in enumerate(_readings(scene)) if v == target]
    return hits[0]


def _try_scene(rng: random.Random) -> dict | None:
    target = rng.randint(3, 8)

    ref_rod = rng.randint(MIN_REF_ROD, min(MAX_REF_ROD, MAX_REF_BAR // target))
    ref_bar = target * ref_rod

    ans = rng.randrange(N_PANELS)

    values: list[int] = [0] * N_PANELS
    rods: list[int] = [0] * N_PANELS
    pool = [v for v in range(2, 10) if v != target]
    rng.shuffle(pool)
    others = pool[: N_PANELS - 1]
    if not any(v > target for v in others) or not any(v < target for v in others):
        return None
    it = iter(others)
    for i in range(N_PANELS):
        values[i] = target if i == ans else next(it)

    for i, v in enumerate(values):
        hi = min(MAX_ROD, MAX_BAR // v)
        if hi < MIN_ROD:
            return None
        rods[i] = rng.randint(MIN_ROD, hi)
    bars = [v * s for v, s in zip(values, rods)]

    # the answer panel must not simply look like the reference panel
    if abs(rods[ans] - ref_rod) < 12 or abs(bars[ans] - ref_bar) < 60:
        return None
    # bar heights must be separable from each other
    order = sorted(bars)
    if any(b - a < 16 for a, b in zip(order, order[1:])):
        return None
    # every absolute-size shortcut must point somewhere else
    if max(range(N_PANELS), key=lambda i: bars[i]) == ans:
        return None
    if max(range(N_PANELS), key=lambda i: values[i]) == ans:
        return None
    if min(range(N_PANELS), key=lambda i: abs(bars[i] - ref_bar)) == ans:
        return None
    if min(range(N_PANELS), key=lambda i: abs(rods[i] - ref_rod)) == ans:
        return None
    bar_x = [rng.randint(_panel_x(i) + 28, _panel_x(i) + 118) for i in range(N_PANELS)]
    rod_x = [
        rng.randint(_panel_x(i) + 174, _panel_x(i) + PANEL_W - 18 - rods[i])
        for i in range(N_PANELS)
    ]
    rod_y = [rng.randint(PANEL_Y0 + 40, BASE_Y - ROD_T - 40) for _ in range(N_PANELS)]

    paint_order = list(range(N_PANELS))
    rng.shuffle(paint_order)

    return {
        "ref_bar": ref_bar,
        "ref_rod": ref_rod,
        "bars": bars,
        "rods": rods,
        "bar_x": bar_x,
        "rod_x": rod_x,
        "rod_y": rod_y,
        "paint_order": paint_order,
    }


def sample_scene(seed: int) -> dict:
    for attempt in range(4000):
        rng = random.Random((seed * 1_000_003 + attempt * 97 + 11) & 0xFFFFFFFF)
        scene = _try_scene(rng)
        if scene is not None:
            return scene
    raise RuntimeError(f"no admissible scene for seed {seed}")


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), WHITE)
    draw = ImageDraw.Draw(image)

    x0, y0, x1, y1 = REF_BOX
    draw.rectangle([x0, y0, x1, y1], outline=REF_FRAME, width=REF_FRAME_W)
    draw.rectangle(
        [REF_BAR_X, REF_BASE_Y - scene["ref_bar"], REF_BAR_X + REF_BAR_W - 1, REF_BASE_Y - 1],
        fill=BAR,
    )
    draw.rectangle(
        [REF_ROD_X, REF_ROD_Y, REF_ROD_X + scene["ref_rod"] - 1, REF_ROD_Y + ROD_T - 1],
        fill=ROD,
    )

    for i in scene["paint_order"]:
        px = _panel_x(i)
        draw.rectangle(
            [px, PANEL_Y0, px + PANEL_W, PANEL_Y1], outline=PANEL_FRAME, width=PANEL_FRAME_W
        )
        bx = scene["bar_x"][i]
        draw.rectangle(
            [bx, BASE_Y - scene["bars"][i], bx + BAR_W - 1, BASE_Y - 1], fill=BAR
        )
        rx, ry = scene["rod_x"][i], scene["rod_y"][i]
        draw.rectangle([rx, ry, rx + scene["rods"][i] - 1, ry + ROD_T - 1], fill=ROD)

    return image


def analytic_gold(scene: dict) -> str:
    return f"panel-{_answer_index(scene) + 1}"


def margin(scene: dict) -> float:
    """Smallest bar-height change in pixels that would create a second match."""
    target = _ref_reading(scene)
    ans = _answer_index(scene)
    return float(
        min(
            abs(scene["bars"][i] - target * scene["rods"][i])
            for i in range(N_PANELS)
            if i != ans
        )
    )


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Panels never overlap, so the paint order is photometrically invisible."""
    order = list(scene["paint_order"])
    out: list[tuple[str, dict]] = []
    rotated = dict(scene)
    rotated["paint_order"] = order[1:] + order[:1]
    out.append(("rotate_paint_order", rotated))
    reversed_ = dict(scene)
    reversed_["paint_order"] = list(reversed(order))
    out.append(("reverse_paint_order", reversed_))
    return out
