"""Analytic sampler and deterministic Pillow renderer."""

from __future__ import annotations

import itertools
import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

W, H = 1000, 900
BG = (247, 244, 236)
CARD = (255, 255, 252)
INK = (42, 48, 56)
MUTED = (105, 112, 120)
CYAN = (20, 166, 184)
AMBER = (230, 146, 24)
LABELS = ("A", "B", "C", "D")
POSITIONS = ((90, 80), (520, 80), (90, 530), (520, 530))
CARD_W, CARD_H = 390, 340


def _relation_code(beads: list[int]) -> list[int]:
    """1 is same/cyan, 0 is different/amber."""
    return [int(beads[i] == beads[i + 1]) for i in range(6)]


def _balanced_bead_rows() -> list[list[int]]:
    rows = []
    for ones in itertools.combinations(range(7), 4):
        row = [int(i in ones) for i in range(7)]
        if sum(_relation_code(row)) == 3:
            rows.append(row)
    return rows


BEAD_ROWS = _balanced_bead_rows()
CODE_ROWS = [list(bits) for bits in itertools.product((0, 1), repeat=6) if sum(bits) == 3]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    winner = rng.randrange(4)
    cards = []
    bead_pool = BEAD_ROWS[:]
    rng.shuffle(bead_pool)
    for i in range(4):
        beads = bead_pool[i]
        truth = _relation_code(beads)
        if i == winner:
            code = truth[:]
        else:
            choices = [c for c in CODE_ROWS if sum(a != b for a, b in zip(c, truth)) >= 2]
            code = rng.choice(choices)[:]
        cards.append({"beads": beads[:], "code": code})
    return {"cards": cards, "reverse_render": False}


def _scores(scene: dict) -> list[int]:
    return [sum(a == b for a, b in zip(_relation_code(card["beads"]), card["code"])) for card in scene["cards"]]


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    best = max(scores)
    winners = [i for i, value in enumerate(scores) if value == best]
    if len(winners) != 1:
        raise ValueError("non-unique winner")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    scores = _scores(scene)
    return scores.count(max(scores)) != 1 or margin(scene) < 2.0 or max(scores) != 6


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    for card in twin["cards"]:
        card["beads"].reverse()
        card["code"].reverse()
    twin["reverse_render"] = not bool(scene["reverse_render"])
    return [("reverse_sequences_and_render_axis", twin)]


def _font(size: int):
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (W, H), BG)
    draw = ImageDraw.Draw(image)
    draw.text((W // 2, 34), "FIND THE CORRECT TRANSITION CODE", anchor="mm", font=_font(25), fill=INK)
    draw.text((W // 2, 68), "cyan = SAME     amber = DIFFERENT", anchor="mm", font=_font(17), fill=MUTED)
    reverse = bool(scene["reverse_render"])
    for idx, ((x0, y0), card) in enumerate(zip(POSITIONS, scene["cards"])):
        draw.rounded_rectangle((x0, y0, x0 + CARD_W, y0 + CARD_H), radius=16, fill=CARD, outline=INK, width=4)
        draw.text((x0 + 22, y0 + 24), LABELS[idx], anchor="mm", font=_font(27), fill=INK)
        beads = list(reversed(card["beads"])) if reverse else card["beads"]
        code = list(reversed(card["code"])) if reverse else card["code"]
        sx, top_y, size, gap = x0 + 31, y0 + 50, 28, 22
        for j, bit in enumerate(beads):
            xx = sx + j * (size + gap)
            draw.ellipse((xx, top_y, xx + size, top_y + size), fill=CYAN if bit else AMBER, outline=INK, width=2)
        tile_y = y0 + 305
        for j, bit in enumerate(code):
            xx = sx + (size + gap) // 2 + j * (size + gap)
            draw.rounded_rectangle((xx, tile_y, xx + size, tile_y + size), radius=4, fill=CYAN if bit else AMBER, outline=INK, width=2)
    return image
