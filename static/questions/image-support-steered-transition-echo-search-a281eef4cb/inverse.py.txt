"""Independent inverse arm using only raster pixels and public colors."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

CYAN = np.array((20, 166, 184), dtype=np.int16)
AMBER = np.array((230, 146, 24), dtype=np.int16)
LABELS = ("A", "B", "C", "D")


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    seen = np.zeros(mask.shape, dtype=bool)
    out = []
    h, w = mask.shape
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        q = deque([(int(y), int(x))])
        seen[y, x] = True
        xs, ys = [], []
        while q:
            yy, xx = q.popleft()
            xs.append(xx); ys.append(yy)
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = yy + dy, xx + dx
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    q.append((ny, nx))
        if len(xs) >= 250:
            out.append((sum(xs) / len(xs), sum(ys) / len(ys), len(xs)))
    return out


def _colored_objects(image: Image.Image) -> list[tuple[float, float, int]]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    dc = np.max(np.abs(arr - CYAN), axis=2)
    da = np.max(np.abs(arr - AMBER), axis=2)
    cyan = _components(dc <= 8)
    amber = _components(da <= 8)
    return [(x, y, 1) for x, y, _ in cyan] + [(x, y, 0) for x, y, _ in amber]


def _cluster_axis(values: list[float], tolerance: float) -> list[list[int]]:
    order = sorted(range(len(values)), key=values.__getitem__)
    groups: list[list[int]] = []
    for idx in order:
        if not groups or abs(values[idx] - sum(values[j] for j in groups[-1]) / len(groups[-1])) > tolerance:
            groups.append([idx])
        else:
            groups[-1].append(idx)
    return groups


def decision_from_image(image: Image.Image) -> str:
    objects = _colored_objects(image)
    if len(objects) < 13:
        raise ValueError("too few visible tokens")
    width, height = image.size
    quadrant_objects = [[], [], [], []]
    for obj in objects:
        q = (2 if obj[1] >= height / 2 else 0) + (1 if obj[0] >= width / 2 else 0)
        quadrant_objects[q].append(obj)
    complete_scores: dict[int, int] = {}
    for q, tokens in enumerate(quadrant_objects):
        if len(tokens) != 13:
            continue
        ygroups = _cluster_axis([o[1] for o in tokens], 80.0)
        if len(ygroups) != 2:
            continue
        rows = [sorted((tokens[i] for i in group), key=lambda o: o[0]) for group in ygroups]
        upper, lower = rows
        if len(upper) != 7 or len(lower) != 6:
            continue
        truth = [int(upper[k][2] == upper[k + 1][2]) for k in range(6)]
        complete_scores[q] = sum(truth[k] == lower[k][2] for k in range(6))
    winners = [q for q, score in complete_scores.items() if score == 6]
    if len(winners) != 1:
        raise ValueError("no unique intact valid card")
    return LABELS[winners[0]]
