"""Independent pixel-only inverse for broad-bridge island search.

This module deliberately does not import any candidate module.  It finds the
visible card and panel slots from pixels, segments cyan/teal ink, enumerates
connected components, and requires exactly one component spanning both lobes.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_PANEL_NAMES = ("upper-left", "upper-right", "lower-left", "lower-right")
_PATTERN_OFFSETS = ((24, 28), (164, 28), (24, 128), (164, 128))
_PATTERN_W = 112
_PATTERN_H = 80


def _card_bbox(array: np.ndarray) -> tuple[int, int, int, int]:
    # The warm neutral card is distinct from white, dark frames, and saturated ink.
    hi = array.max(axis=2)
    lo = array.min(axis=2)
    total = array.astype(np.int16).sum(axis=2)
    mask = (lo >= 225) & (hi <= 250) & ((hi - lo) <= 14) & (total < 740)
    ys, xs = np.nonzero(mask)
    if len(xs) < 16_000:
        raise ValueError("card is missing or materially corrupted")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    if x1 - x0 < 288 or y1 - y0 < 216:
        raise ValueError("card extent is not recoverable")
    return x0, y0, x1, y1


def _cyan_mask(array: np.ndarray) -> np.ndarray:
    r = array[:, :, 0].astype(np.int16)
    g = array[:, :, 1].astype(np.int16)
    b = array[:, :, 2].astype(np.int16)
    return (r < 80) & (g > 115) & (b > 105) & ((g + b - 2 * r) > 170)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    seen = np.zeros(mask.shape, dtype=bool)
    output: list[np.ndarray] = []
    height, width = mask.shape
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= 32:
            output.append(np.asarray(points, dtype=np.int16))
    return output


def _span_score(mask: np.ndarray) -> float:
    components = _components(mask)
    if not components:
        return 0.0
    return float(max(int(points[:, 1].max()) - int(points[:, 1].min()) + 1 for points in components))


def _solid_core_fraction(mask: np.ndarray) -> float:
    # The broad bridge plus one cell of each lobe forms a visible 48x48 solid core.
    # Requiring this area to survive is the raster counterpart of a broad, not
    # hairline, connection and gives destructive evidence erasure a clear effect.
    return float(mask[16:64, 32:80].mean())


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    x0, y0, _, _ = _card_bbox(array)
    cyan = _cyan_mask(array)
    scores = []
    core_fractions = []
    for dx, dy in _PATTERN_OFFSETS:
        crop = cyan[y0 + dy : y0 + dy + _PATTERN_H, x0 + dx : x0 + dx + _PATTERN_W]
        if crop.shape != (_PATTERN_H, _PATTERN_W):
            raise ValueError("panel crop is incomplete")
        scores.append(_span_score(crop))
        core_fractions.append(_solid_core_fraction(crop))
    order = sorted(range(4), key=lambda index: scores[index], reverse=True)
    winner, runner = order[0], order[1]
    if scores[winner] < 88.0 or scores[winner] - scores[runner] < 16.0:
        raise ValueError(f"no unique broad spanning island: scores={scores}")
    if core_fractions[winner] < 0.97:
        raise ValueError(f"winning bridge core is incomplete: fill={core_fractions[winner]:.3f}")
    return _PANEL_NAMES[winner]
