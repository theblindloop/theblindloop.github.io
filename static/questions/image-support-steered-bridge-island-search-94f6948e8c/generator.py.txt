"""Deterministic latent scene and raster renderer for broad-bridge search."""

from __future__ import annotations

import copy
import random
from typing import Any

from PIL import Image, ImageDraw

PANEL_NAMES = ("upper-left", "upper-right", "lower-left", "lower-right")
PALETTES = (
    (24, 174, 190),
    (26, 154, 181),
    (28, 164, 146),
)
BACKGROUND = (255, 255, 255)
CARD = (244, 242, 235)
FRAME = (42, 49, 62)
CELL = 8
PATTERN_W = 14 * CELL
PATTERN_H = 10 * CELL
CARD_W = 300
CARD_H = 228
PATTERN_OFFSETS = ((24, 28), (164, 28), (24, 128), (164, 128))
FRAME_OFFSETS = ((18, 22), (158, 22), (18, 122), (158, 122))


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    # Consecutive generator seeds guarantee exact four-way answer balance.
    target_panel = int(seed) % 4
    origin = [rng.randint(40, 172), rng.randint(58, 224)]
    # Each entry permutes which harmless outer corner cells are removed.
    variants = [rng.randrange(4) for _ in range(4)]
    return {
        "width": 512,
        "height": 512,
        "origin": origin,
        "target_panel": target_panel,
        "variants": variants,
        "palette": list(PALETTES[rng.randrange(len(PALETTES))]),
        "nonce": rng.randrange(1_000_000),
    }


def _panel_name(index: int) -> str:
    return PANEL_NAMES[int(index)]


def analytic_gold(scene: dict[str, Any]) -> str:
    return _panel_name(scene["target_panel"])


def margin(scene: dict[str, Any]) -> float:
    # A spanning component is 112 px wide; the widest disconnected lobe is 64 px.
    return 48.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 24.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    twin = copy.deepcopy(scene)
    twin["nonce"] = int(twin.get("nonce", 0)) + 1
    return [("increment_render_inert_nonce", twin)]


def _removed_cells(variant: int) -> set[tuple[int, int]]:
    choices = (
        {(x, y) for x in (0, 1, 2, 11, 12, 13) for y in (0, 9)},
        {(x, y) for x in (2, 3, 4, 9, 10, 11) for y in (0, 9)},
        {(x, y) for x in (0, 1, 2, 11, 12, 13) for y in (1, 8)},
        {(x, y) for x in (2, 3, 4, 9, 10, 11) for y in (1, 8)},
    )
    return choices[int(variant) % len(choices)]


def _occupied_cells(scene: dict[str, Any], panel: int) -> set[tuple[int, int]]:
    cells = {(x, y) for x in range(5) for y in range(10)}
    cells |= {(x, y) for x in range(9, 14) for y in range(10)}
    cells -= _removed_cells(int(scene["variants"][panel]))
    if panel == int(scene["target_panel"]):
        # A thick, connected, area-like 4x6 bridge: the only decisive cue.
        cells |= {(x, y) for x in range(5, 9) for y in range(2, 8)}
    else:
        # Equal ink, but two vertically separated inward tabs never touch.
        cells |= {(x, y) for x in range(5, 8) for y in range(0, 4)}
        cells |= {(x, y) for x in range(6, 9) for y in range(6, 10)}
    return cells


def render(scene: dict[str, Any]) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    ox, oy = map(int, scene["origin"])
    cyan = tuple(int(v) for v in scene["palette"])
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rectangle((ox, oy, ox + CARD_W - 1, oy + CARD_H - 1), fill=CARD)
    for panel, ((fx, fy), (px, py)) in enumerate(zip(FRAME_OFFSETS, PATTERN_OFFSETS)):
        draw.rectangle(
            (ox + fx, oy + fy, ox + fx + 123, oy + fy + 91),
            outline=FRAME,
            width=2,
        )
        for cx, cy in _occupied_cells(scene, panel):
            x0 = ox + px + cx * CELL
            y0 = oy + py + cy * CELL
            draw.rectangle((x0, y0, x0 + CELL - 1, y0 + CELL - 1), fill=cyan)
    return image
