"""Independent inverse arm: recover turn budgets from raster pixels only."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> int:
    height, width = mask.shape
    seen: set[tuple[int, int]] = set()
    count = 0
    for yy, xx in np.argwhere(mask):
        start = (int(yy), int(xx))
        if start in seen:
            continue
        count += 1
        stack = [start]
        seen.add(start)
        while stack:
            y, x = stack.pop()
            for point in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                py, px = point
                if 0 <= py < height and 0 <= px < width and mask[py, px] and point not in seen:
                    seen.add(point)
                    stack.append(point)
    return count


def _trace_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[..., 0], array[..., 1], array[..., 2]
    # All legal trace palettes are blue-green; gold marks and gray/white backgrounds fail this.
    return (blue > red + 28) & (green > red + 18) & (blue > 70) & (green > 65)


def decision_from_image(image: Image.Image) -> str:
    mask = _trace_mask(image)
    if int(mask.sum()) < 350:
        raise ValueError("trace is absent or too damaged")
    if _components(mask) != 1:
        raise ValueError("trace is disconnected")
    ys, xs = np.nonzero(mask)
    if int(xs.max() - xs.min()) < 90:
        raise ValueError("trace horizontal span is incomplete")

    # The rendered path has eight equal-width runs. Infer endpoint center positions
    # from its antialiased butt extent, then fit each run away from its bends.
    left = float(xs.min()) + 2.0
    right = float(xs.max()) - 2.0
    step = (right - left) / 8.0
    if not 11.0 <= step <= 15.0:
        raise ValueError("unexpected run spacing")
    headings: list[float] = []
    for index in range(8):
        low = left + (index + 0.22) * step
        high = left + (index + 0.78) * step
        select = mask & (np.arange(mask.shape[1])[None, :] >= low) & (
            np.arange(mask.shape[1])[None, :] <= high
        )
        sy, sx = np.nonzero(select)
        if len(sx) < 18 or float(np.ptp(sx)) < 4.0:
            raise ValueError("a required interior run is missing")
        centered_x = sx.astype(np.float64) - float(np.mean(sx))
        denom = float(centered_x @ centered_x)
        if denom <= 0:
            raise ValueError("degenerate line fit")
        slope = float(centered_x @ (sy.astype(np.float64) - float(np.mean(sy))) / denom)
        headings.append(math.degrees(math.atan(slope)))
    turns = [abs(headings[index] - headings[index - 1]) for index in range(1, 8)]
    before = sum(turns[:3])
    after = sum(turns[4:])
    if abs(before - after) < 3.0:
        raise ValueError("pixel turn totals are ambiguous")
    return "BEFORE" if before > after else "AFTER"
