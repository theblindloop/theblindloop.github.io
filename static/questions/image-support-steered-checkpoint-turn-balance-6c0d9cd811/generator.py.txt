"""Deterministic latent sampler and renderer for Checkpoint Turn Balance."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
RUN_COUNT = 8
QUARANTINE_DEG = 5.0
TRACE_PALETTE = ("#087e8b", "#006d77", "#147d92")


def _turn_totals(scene: dict[str, Any]) -> tuple[float, float]:
    dys = [float(value) for value in scene["dys"]]
    headings = [math.degrees(math.atan2(value, 13.0)) for value in dys]
    turns = [abs(headings[index] - headings[index - 1]) for index in range(1, RUN_COUNT)]
    return sum(turns[:3]), sum(turns[4:])


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    desired = "BEFORE" if int(seed) % 2 == 0 else "AFTER"
    near_boundary = int(seed) % 6 == 0
    values = (-6, -4, -2, 0, 2, 4, 6)
    for _ in range(50000):
        dys = [rng.choice(values) for _ in range(RUN_COUNT)]
        # Every interior join must be a visibly nonzero bend; no hidden run boundaries.
        if any(dys[index] == dys[index - 1] for index in range(1, RUN_COUNT)):
            continue
        cumulative = [0]
        for rise in dys:
            cumulative.append(cumulative[-1] + rise)
        if max(cumulative) - min(cumulative) > 20:
            continue
        probe = {"dys": dys}
        before, after = _turn_totals(probe)
        decision = "BEFORE" if before > after else "AFTER"
        gap = abs(before - after)
        if decision != desired:
            continue
        if near_boundary and not (22.0 <= gap <= 30.0):
            continue
        if not near_boundary and not (34.0 <= gap <= 75.0):
            continue
        anchor_y = rng.choice((96, 224, 352, 416))
        centered_y0 = anchor_y - (min(cumulative) + max(cumulative)) / 2.0
        return {
            "width": WIDTH,
            "height": HEIGHT,
            "x0": rng.choice((90, 218, 346)),
            "y0": centered_y0,
            "dx": 13,
            "dys": dys,
            "trace_color": rng.choice(TRACE_PALETTE),
            "marker_color": "#d49b00",
            "background": "#ffffff",
        }
    raise RuntimeError(f"could not sample a balanced scene for seed {seed}")


def render(scene: dict[str, Any]) -> Image.Image:
    scale = 4
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), scene["background"])
    draw = ImageDraw.Draw(image)
    x = float(scene["x0"])
    y = float(scene["y0"])
    points = [(x, y)]
    for rise in scene["dys"]:
        x += float(scene["dx"])
        y += float(rise)
        points.append((x, y))
    scaled = [(round(px * scale), round(py * scale)) for px, py in points]
    draw.line(scaled, fill=scene["trace_color"], width=5 * scale, joint="curve")
    # Two small gold marks bracket the central vertex without covering trace pixels.
    checkpoint_x, checkpoint_y = scaled[4]
    draw.line(
        [(checkpoint_x, checkpoint_y - 12 * scale), (checkpoint_x, checkpoint_y - 8 * scale)],
        fill=scene["marker_color"],
        width=3 * scale,
    )
    draw.line(
        [(checkpoint_x, checkpoint_y + 8 * scale), (checkpoint_x, checkpoint_y + 12 * scale)],
        fill=scene["marker_color"],
        width=3 * scale,
    )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    before, after = _turn_totals(scene)
    return "BEFORE" if before > after else "AFTER"


def margin(scene: dict[str, Any]) -> float:
    before, after = _turn_totals(scene)
    return abs(before - after)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < QUARANTINE_DEG


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
