"""Deterministic latent sampler and renderer for Ribbon Link Correspondence."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw

BACKGROUND = (250, 248, 242)
ROUTE = (55, 62, 70)
LANDMARK = (18, 21, 25)
MAGENTA = (210, 48, 142)
ANSWER_RGB = {
    "amber": (234, 153, 30),
    "cyan": (24, 166, 188),
    "violet": (132, 82, 190),
}
SHAPES = ("circle", "square", "diamond", "triangle")
ANSWERS = ("amber", "cyan", "violet")
SLOT_STEP = 74
ROW_GAP = 48


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    top_order = rng.sample(list(SHAPES), len(SHAPES))
    query_index = rng.randrange(3)
    target = frozenset(top_order[query_index : query_index + 2])
    choices = [
        list(order)
        for order in itertools.permutations(SHAPES)
        if any(frozenset(order[i : i + 2]) == target for i in range(3))
        and list(order) != top_order
    ]
    bottom_order = rng.choice(choices)
    match_index = next(
        i for i in range(3) if frozenset(bottom_order[i : i + 2]) == target
    )
    desired = ANSWERS[int(seed) % len(ANSWERS)]
    remaining = [name for name in ANSWERS if name != desired]
    rng.shuffle(remaining)
    bottom_colors = [""] * 3
    bottom_colors[match_index] = desired
    for index in range(3):
        if not bottom_colors[index]:
            bottom_colors[index] = remaining.pop()
    return {
        "canvas_width": 640,
        "canvas_height": 480,
        "left": 185 + rng.choice((-8, 0, 8)),
        "top_y": 216 + rng.choice((-2, 0, 2)),
        "top_order": top_order,
        "bottom_order": bottom_order,
        "query_index": query_index,
        "bottom_colors": bottom_colors,
    }


def _draw_shape(draw: ImageDraw.ImageDraw, name: str, cx: int, cy: int) -> None:
    if name == "circle":
        draw.ellipse((cx - 15, cy - 15, cx + 15, cy + 15), fill=LANDMARK)
    elif name == "square":
        draw.rectangle((cx - 13, cy - 13, cx + 13, cy + 13), fill=LANDMARK)
    elif name == "diamond":
        draw.polygon(
            ((cx, cy - 16), (cx + 16, cy), (cx, cy + 16), (cx - 16, cy)),
            fill=LANDMARK,
        )
    elif name == "triangle":
        draw.polygon(
            ((cx, cy - 16), (cx + 17, cy + 14), (cx - 17, cy + 14)),
            fill=LANDMARK,
        )
    else:
        raise ValueError(f"unknown shape {name!r}")


def render(scene: dict) -> Image.Image:
    width = int(scene["canvas_width"])
    height = int(scene["canvas_height"])
    left = int(scene["left"])
    top_y = int(scene["top_y"])
    top_order = list(scene["top_order"])
    bottom_order = list(scene["bottom_order"])
    query_index = int(scene["query_index"])
    bottom_colors = list(scene["bottom_colors"])
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    bottom_y = top_y + ROW_GAP
    xs = [left + index * SLOT_STEP for index in range(4)]
    start_x = left - 28
    turn_x = xs[-1] + 30

    # One visibly connected return ribbon. All overlays touch this backbone.
    draw.line(
        ((start_x, top_y), (turn_x, top_y), (turn_x, bottom_y), (start_x, bottom_y)),
        fill=ROUTE,
        width=7,
        joint="curve",
    )
    qx0, qx1 = xs[query_index], xs[query_index + 1]
    draw.line(((qx0 + 13, top_y), (qx1 - 13, top_y)), fill=MAGENTA, width=9)
    for index, color_name in enumerate(bottom_colors):
        draw.line(
            ((xs[index] + 13, bottom_y), (xs[index + 1] - 13, bottom_y)),
            fill=ANSWER_RGB[color_name],
            width=9,
        )
    for x, name in zip(xs, top_order, strict=True):
        _draw_shape(draw, name, x, top_y)
    for x, name in zip(xs, bottom_order, strict=True):
        _draw_shape(draw, name, x, bottom_y)
    return image


def _label_parts(scene: dict) -> tuple[list[str], list[str], int, list[str]]:
    return (
        list(scene["top_order"]),
        list(scene["bottom_order"]),
        int(scene["query_index"]),
        list(scene["bottom_colors"]),
    )


def _matching_indices(
    top_order: list[str], bottom_order: list[str], query_index: int
) -> list[int]:
    if query_index not in range(3):
        return []
    target = frozenset(top_order[query_index : query_index + 2])
    return [
        index
        for index in range(3)
        if frozenset(bottom_order[index : index + 2]) == target
    ]


def analytic_gold(scene: dict) -> str:
    top_order, bottom_order, query_index, bottom_colors = _label_parts(scene)
    if (
        sorted(top_order) != sorted(SHAPES)
        or sorted(bottom_order) != sorted(SHAPES)
        or sorted(bottom_colors) != sorted(ANSWERS)
    ):
        raise ValueError("scene is not a pair of landmark/color permutations")
    matches = _matching_indices(top_order, bottom_order, query_index)
    if len(matches) != 1:
        raise ValueError("marked adjacency has no unique lower correspondence")
    return bottom_colors[matches[0]]


def margin(scene: dict) -> float:
    top_order, bottom_order, query_index, bottom_colors = _label_parts(scene)
    if sorted(bottom_colors) != sorted(ANSWERS) or query_index not in range(3):
        return 0.0
    target = frozenset(top_order[query_index : query_index + 2])
    scores = sorted(
        (len(target.intersection(bottom_order[index : index + 2])) for index in range(3)),
        reverse=True,
    )
    return float(scores[0] - scores[1]) if len(scores) >= 2 else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        top_order, bottom_order, query_index, bottom_colors = _label_parts(scene)
        valid = (
            sorted(top_order) == sorted(SHAPES)
            and sorted(bottom_order) == sorted(SHAPES)
            and sorted(bottom_colors) == sorted(ANSWERS)
            and len(_matching_indices(top_order, bottom_order, query_index)) == 1
            and margin(scene) > 0.0
        )
        return not valid
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
