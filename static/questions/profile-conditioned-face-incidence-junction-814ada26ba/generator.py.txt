"""Deterministic planar-wall renderer and analytic bounded-face label.

The scene contains one small planar graph.  Three degree-three vertices are
marked with visible colors.  In the fixed topology, one mark is on two bounded
faces, one is on one bounded face, and one lies only on the outside face.  The
answer is the color assigned to the two-face vertex.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw


RENDERER_VERSION = "face_incidence_junction-renderer-0.1.0"
WIDTH, HEIGHT = 960, 640
BACKGROUND = (248, 247, 242)
EDGE_COLOR = (92, 100, 112)
NODE_COLOR = (31, 36, 45)
MARK_COLORS: dict[str, tuple[int, int, int]] = {
    "red": (218, 60, 65),
    "green": (43, 155, 91),
    "blue": (59, 103, 199),
}
NODE_RADIUS = 17
LINE_WIDTH = 7
LINK_GAP = 6

# A square with a diagonal gives two bounded triangular faces.  The branch
# through node 5 is a tree, so it touches only the unbounded face.  Nodes 0, 1,
# and 5 all have degree three and are the only colored vertices.
BASE_POINTS: tuple[tuple[float, float], ...] = (
    (0.0, 0.0),   # 0: A, two bounded faces
    (1.0, 0.0),   # 1: B, one bounded face plus a leaf
    (1.0, 1.0),   # 2: C
    (0.0, 1.0),   # 3: D
    (1.90, -0.10), # 4: leaf at B
    (-1.0, 1.0),  # 5: X, tree junction
    (-1.86, 1.55),# 6: tree leaf
    (-1.86, 0.45),# 7: tree leaf
)
BASE_EDGES: tuple[tuple[int, int], ...] = (
    (0, 1), (1, 2), (2, 3), (3, 0), (0, 2),
    (1, 4), (3, 5), (5, 6), (5, 7),
)
MARKED_NODES = (0, 1, 5)


def _xy(node: list[Any]) -> tuple[float, float]:
    return float(node[0]), float(node[1])


def _face_walks(scene: dict[str, Any]) -> list[tuple[list[int], float]]:
    """Traverse every directed half-edge and return its boundary walk/area."""

    nodes = scene["nodes"]
    edges = scene["edges"]
    adjacency: list[list[int]] = [[] for _ in nodes]
    for raw_edge in edges:
        first, second = int(raw_edge[0]), int(raw_edge[1])
        adjacency[first].append(second)
        adjacency[second].append(first)
    for index, neighbors in enumerate(adjacency):
        px, py = _xy(nodes[index])
        neighbors.sort(
            key=lambda other: math.atan2(
                _xy(nodes[other])[1] - py,
                _xy(nodes[other])[0] - px,
            )
        )

    directed = [(first, second) for first, second in edges for first, second in ((int(first), int(second)), (int(second), int(first)))]
    visited: set[tuple[int, int]] = set()
    faces: list[tuple[list[int], float]] = []
    for start in directed:
        if start in visited:
            continue
        current = start
        walk: list[int] = []
        while current not in visited:
            visited.add(current)
            first, second = current
            walk.append(first)
            neighbors = adjacency[second]
            position = neighbors.index(first)
            # With counter-clockwise neighbor order, taking the previous ray
            # keeps the face on the left.  Reflection only reverses signs; the
            # outside face is identified below by absolute area.
            next_node = neighbors[(position - 1) % len(neighbors)]
            current = (second, next_node)
        area = 0.0
        if len(walk) >= 3:
            for index, first in enumerate(walk):
                second = walk[(index + 1) % len(walk)]
                x1, y1 = _xy(nodes[first])
                x2, y2 = _xy(nodes[second])
                area += x1 * y2 - x2 * y1
            area *= 0.5
        faces.append((walk, area))
    return faces


def _face_counts(scene: dict[str, Any]) -> dict[str, int]:
    faces = _face_walks(scene)
    if not faces:
        raise ValueError("graph has no faces")
    outside = max(range(len(faces)), key=lambda index: abs(faces[index][1]))
    bounded = [set(walk) for index, (walk, area) in enumerate(faces) if index != outside and abs(area) > 1e-6]
    counts: dict[str, int] = {}
    for index, node in enumerate(scene["nodes"]):
        mark = str(node[2])
        if mark in MARK_COLORS:
            counts[mark] = sum(index in face for face in bounded)
    return counts


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a pose and a random color assignment for the fixed visible graph."""

    rng = random.Random(int(seed))
    scale = rng.choice((90.0, 96.0, 102.0, 108.0))
    quarter_turns = rng.randrange(4)
    mirror = bool(rng.randrange(2))
    center_x = 480.0 + rng.randint(-28, 28)
    center_y = 320.0 + rng.randint(-22, 22)
    min_x = min(point[0] for point in BASE_POINTS)
    max_x = max(point[0] for point in BASE_POINTS)
    min_y = min(point[1] for point in BASE_POINTS)
    max_y = max(point[1] for point in BASE_POINTS)
    base_center_x = (min_x + max_x) / 2.0
    base_center_y = (min_y + max_y) / 2.0
    angle = quarter_turns * math.pi / 2.0
    cosine, sine = math.cos(angle), math.sin(angle)

    points: list[tuple[int, int]] = []
    for raw_x, raw_y in BASE_POINTS:
        x = (raw_x - base_center_x) * scale
        y = (raw_y - base_center_y) * scale
        if mirror:
            x = -x
        rotated_x = cosine * x - sine * y
        rotated_y = sine * x + cosine * y
        # Small visible perturbations prevent the graph from being a single
        # pixel-template while leaving wide clearance from all boundaries.
        jitter_x = rng.randint(-2, 2)
        jitter_y = rng.randint(-2, 2)
        points.append((int(round(center_x + rotated_x)) + jitter_x, int(round(center_y + rotated_y)) + jitter_y))

    colors = list(MARK_COLORS)
    rng.shuffle(colors)
    mark_for_node = {role: colors[position] for position, role in enumerate(MARKED_NODES)}
    nodes = [[point[0], point[1], mark_for_node.get(index, "none")] for index, point in enumerate(points)]
    edges = [[first, second] for first, second in BASE_EDGES]
    return {"nodes": nodes, "edges": edges}


def _orientation(first: tuple[float, float], second: tuple[float, float], third: tuple[float, float]) -> float:
    return (second[0] - first[0]) * (third[1] - first[1]) - (second[1] - first[1]) * (third[0] - first[0])


def _segments_intersect(
    first: tuple[float, float],
    second: tuple[float, float],
    third: tuple[float, float],
    fourth: tuple[float, float],
) -> bool:
    eps = 1e-7
    a = _orientation(first, second, third)
    b = _orientation(first, second, fourth)
    c = _orientation(third, fourth, first)
    d = _orientation(third, fourth, second)
    return ((a > eps and b < -eps) or (a < -eps and b > eps)) and ((c > eps and d < -eps) or (c < -eps and d > eps))


def _point_segment_distance(point: tuple[float, float], first: tuple[float, float], second: tuple[float, float]) -> float:
    dx, dy = second[0] - first[0], second[1] - first[1]
    length_sq = dx * dx + dy * dy
    if length_sq <= 1e-9:
        return math.dist(point, first)
    parameter = ((point[0] - first[0]) * dx + (point[1] - first[1]) * dy) / length_sq
    parameter = max(0.0, min(1.0, parameter))
    closest = (first[0] + parameter * dx, first[1] + parameter * dy)
    return math.dist(point, closest)


def _geometry_is_clear(scene: dict[str, Any]) -> bool:
    nodes = scene.get("nodes")
    edges = scene.get("edges")
    if not isinstance(nodes, list) or not isinstance(edges, list) or len(nodes) != 8 or len(edges) != 9:
        return False
    marks = [str(node[2]) for node in nodes if isinstance(node, list) and len(node) == 3]
    if marks.count("red") != 1 or marks.count("green") != 1 or marks.count("blue") != 1 or marks.count("none") != 5:
        return False
    if any(index >= len(nodes) or len(adjacency) != 3 for index, adjacency in ((role, [edge for edge in edges if role in edge]) for role in MARKED_NODES)):
        return False
    points = [_xy(node) for node in nodes]
    if any(x < 30 or x > WIDTH - 30 or y < 30 or y > HEIGHT - 30 for x, y in points):
        return False
    for first in range(len(points)):
        for second in range(first):
            if math.dist(points[first], points[second]) < 2 * NODE_RADIUS + 12:
                return False
    segments: list[tuple[int, int, tuple[float, float], tuple[float, float]]] = []
    normalized_edges: set[tuple[int, int]] = set()
    for raw_edge in edges:
        if not isinstance(raw_edge, list) or len(raw_edge) != 2:
            return False
        first, second = int(raw_edge[0]), int(raw_edge[1])
        if not (0 <= first < len(points) and 0 <= second < len(points)) or first == second:
            return False
        key = tuple(sorted((first, second)))
        if key in normalized_edges:
            return False
        normalized_edges.add(key)
        if math.dist(points[first], points[second]) < 2 * (NODE_RADIUS + LINK_GAP) + 20:
            return False
        segments.append((first, second, points[first], points[second]))
    for index, (first, second, start, end) in enumerate(segments):
        for other_first, other_second, other_start, other_end in segments[:index]:
            if {first, second} & {other_first, other_second}:
                continue
            if _segments_intersect(start, end, other_start, other_end):
                return False
        for node_index, point in enumerate(points):
            if node_index not in {first, second} and _point_segment_distance(point, start, end) < NODE_RADIUS + 7:
                return False
    degrees = [0] * len(points)
    for first, second in normalized_edges:
        degrees[first] += 1
        degrees[second] += 1
    if any(degrees[role] != 3 for role in MARKED_NODES):
        return False
    try:
        counts = _face_counts(scene)
    except (KeyError, TypeError, ValueError, IndexError):
        return False
    return sorted(counts.values()) == [0, 1, 2]


def analytic_gold(scene: dict[str, Any]) -> str:
    counts = _face_counts(scene)
    if sorted(counts.values()) != [0, 1, 2]:
        raise ValueError(f"expected face incidences 0, 1, 2; got {counts}")
    return next(color for color, count in counts.items() if count == 2)


def margin(scene: dict[str, Any]) -> float:
    counts = _face_counts(scene)
    ordered = sorted(counts.values(), reverse=True)
    if len(ordered) < 2:
        return 0.0
    return float(ordered[0] - ordered[1])


def is_quarantined(scene: dict[str, Any]) -> bool:
    return not _geometry_is_clear(scene)


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only the visible graph; no analytic labels are painted."""

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    nodes = scene["nodes"]
    for raw_edge in scene["edges"]:
        first, second = int(raw_edge[0]), int(raw_edge[1])
        x1, y1 = _xy(nodes[first])
        x2, y2 = _xy(nodes[second])
        dx, dy = x2 - x1, y2 - y1
        length = math.hypot(dx, dy)
        ux, uy = dx / length, dy / length
        start = (int(round(x1 + ux * (NODE_RADIUS + LINK_GAP))), int(round(y1 + uy * (NODE_RADIUS + LINK_GAP))))
        end = (int(round(x2 - ux * (NODE_RADIUS + LINK_GAP))), int(round(y2 - uy * (NODE_RADIUS + LINK_GAP))))
        draw.line((start, end), fill=EDGE_COLOR, width=LINE_WIDTH)
    for node in nodes:
        x, y = _xy(node)
        color = MARK_COLORS.get(str(node[2]), NODE_COLOR)
        draw.ellipse((int(round(x)) - NODE_RADIUS, int(round(y)) - NODE_RADIUS, int(round(x)) + NODE_RADIUS, int(round(y)) + NODE_RADIUS), fill=color)
    return image


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """There are no hidden render-invariant latent fields."""

    return []
