"""Deterministic renderer for a visible clock-numeral winding conflict."""

from __future__ import annotations

import hashlib
import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 600
HEIGHT = 600
LABELS = ("12", "3", "6", "9")
ANSWERS = ("clockwise", "counterclockwise")
BADGE_FILL = (255, 249, 226)

# Public, deliberately legible 5-by-7 block numerals. Each pattern occupies its
# complete grid bounding box, allowing an independent raster decoder to recover
# the printed symbols without renderer coordinates or latent state.
GLYPHS = {
    "1": ("00100", "01100", "10100", "00100", "00100", "00100", "11111"),
    "2": ("11110", "00001", "00001", "11110", "10000", "10000", "11111"),
    "3": ("11110", "00001", "00001", "01110", "00001", "00001", "11110"),
    "6": ("01111", "10000", "10000", "11110", "10001", "10001", "01110"),
    "9": ("01110", "10001", "10001", "01111", "00001", "00001", "11110"),
}

DIAL_COLORS = ((196, 222, 238), (211, 230, 204), (239, 216, 196), (221, 211, 239))
INK_COLORS = ((23, 52, 105), (94, 29, 56), (18, 78, 61), (92, 55, 12))
BACKGROUNDS = ((237, 242, 247), (244, 239, 232), (235, 243, 237))


def _rng(pair_seed: int, channel: str) -> random.Random:
    payload = f"clock-winding-v1|{int(pair_seed)}|{channel}".encode("utf-8")
    return random.Random(int.from_bytes(hashlib.sha256(payload).digest()[:8], "big"))


def sample_scene(seed: int) -> dict:
    """Return matched control/counterfactual pairs for consecutive seeds."""
    seed = int(seed)
    pair_seed = seed // 2
    style = _rng(pair_seed, "style")
    paint_order = list(range(len(LABELS)))
    _rng(pair_seed, "paint").shuffle(paint_order)
    return {
        "winding": ANSWERS[seed % 2],
        "center_x": style.randint(292, 308),
        "center_y": style.randint(292, 308),
        "radius": style.choice((218, 222, 226)),
        "rotation": style.uniform(-24.0, 24.0),
        "dial_color": list(style.choice(DIAL_COLORS)),
        "ink_color": list(style.choice(INK_COLORS)),
        "background": list(style.choice(BACKGROUNDS)),
        "glyph_cell": style.choice((6, 7)),
        "paint_order": paint_order,
    }


def label_centers(scene: dict) -> list[tuple[int, int]]:
    """Visible label centers, exposed for metamorphic tests but not the oracle."""
    cx = int(scene["center_x"])
    cy = int(scene["center_y"])
    orbit = int(round(float(scene["radius"]) * 0.66))
    direction = 1.0 if scene["winding"] == "clockwise" else -1.0
    rotation = float(scene["rotation"])
    centers: list[tuple[int, int]] = []
    for index in range(len(LABELS)):
        angle = math.radians(-90.0 + rotation + direction * 90.0 * index)
        centers.append((round(cx + orbit * math.cos(angle)), round(cy + orbit * math.sin(angle))))
    return centers


def _draw_glyph(
    draw: ImageDraw.ImageDraw, label: str, center: tuple[int, int], cell: int, ink: tuple[int, int, int]
) -> None:
    digits = list(label)
    width_cells = 5 * len(digits) + max(0, len(digits) - 1)
    x0 = center[0] - (width_cells * cell) // 2
    y0 = center[1] - (7 * cell) // 2
    for digit_index, digit in enumerate(digits):
        offset = digit_index * 6 * cell
        for row, bits in enumerate(GLYPHS[digit]):
            for column, bit in enumerate(bits):
                if bit == "1":
                    left = x0 + offset + column * cell
                    top = y0 + row * cell
                    draw.rectangle((left, top, left + cell - 1, top + cell - 1), fill=ink)


def render(scene: dict) -> Image.Image:
    winding = str(scene["winding"])
    if winding not in ANSWERS:
        raise ValueError(f"unsupported winding {winding!r}")
    cx, cy = int(scene["center_x"]), int(scene["center_y"])
    radius = int(scene["radius"])
    rotation = float(scene["rotation"])
    dial_color = tuple(int(v) for v in scene["dial_color"])
    ink_color = tuple(int(v) for v in scene["ink_color"])
    background = tuple(int(v) for v in scene["background"])
    cell = int(scene["glyph_cell"])
    order = [int(v) for v in scene["paint_order"]]
    if sorted(order) != list(range(len(LABELS))):
        raise ValueError("paint_order must be a permutation of four labels")

    image = Image.new("RGB", (WIDTH, HEIGHT), background)
    draw = ImageDraw.Draw(image)
    outline = (61, 66, 72)
    draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=dial_color, outline=outline, width=6)

    # Neutral ticks establish a visible dial frame without encoding the answer.
    for tick in range(24):
        angle = math.radians(-90.0 + rotation + 15.0 * tick)
        outer = radius - 11
        inner = radius - (25 if tick % 2 == 0 else 19)
        p0 = (round(cx + inner * math.cos(angle)), round(cy + inner * math.sin(angle)))
        p1 = (round(cx + outer * math.cos(angle)), round(cy + outer * math.sin(angle)))
        draw.line((p0, p1), fill=outline, width=4 if tick % 2 == 0 else 3)
    draw.ellipse((cx - 9, cy - 9, cx + 9, cy + 9), fill=outline)

    centers = label_centers(scene)
    badge_radius = 52
    for index in order:
        bx, by = centers[index]
        draw.ellipse(
            (bx - badge_radius, by - badge_radius, bx + badge_radius, by + badge_radius),
            fill=BADGE_FILL,
            outline=outline,
            width=4,
        )
        _draw_glyph(draw, LABELS[index], (bx, by), cell, ink_color)
    return image


def analytic_gold(scene: dict) -> str:
    winding = str(scene["winding"])
    if winding not in ANSWERS:
        raise ValueError(f"unsupported winding {winding!r}")
    return winding


def margin(scene: dict) -> float:
    """Angular distance in degrees to either shorter-arc direction boundary."""
    del scene
    return 90.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["paint_order"] = list(reversed(scene["paint_order"]))
    return [("reverse_disjoint_badge_paint_order", twin)]
