"""Independent inverse arm for disconnected zipper cards; pixels only."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

BACKGROUND = np.array((250, 250, 247), dtype=np.float32)
INK = np.array((31, 38, 48), dtype=np.float32)
PATTERN_COLORS = np.array(
    ((48, 105, 178), (218, 82, 78), (71, 157, 101), (235, 188, 62)),
    dtype=np.float32,
)
TAG_NAMES = ("amber", "cyan", "violet")
TAG_COLORS = np.array(
    ((244, 137, 32), (35, 187, 207), (151, 83, 190)), dtype=np.float32
)


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    """Return 4-connected component boxes as left, top, right, bottom, area."""
    work = mask.copy()
    height, width = work.shape
    components: list[tuple[int, int, int, int, int]] = []
    for start_y, start_x in np.argwhere(work):
        y0, x0 = int(start_y), int(start_x)
        if not work[y0, x0]:
            continue
        work[y0, x0] = False
        queue = deque(((x0, y0),))
        left = right = x0
        top = bottom = y0
        area = 0
        while queue:
            x, y = queue.popleft()
            area += 1
            left, right = min(left, x), max(right, x)
            top, bottom = min(top, y), max(bottom, y)
            for nx, ny in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
                if 0 <= nx < width and 0 <= ny < height and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((nx, ny))
        components.append((left, top, right + 1, bottom + 1, area))
    return components


def _frame_boxes(array: np.ndarray, color: np.ndarray) -> list[tuple[int, int, int, int]]:
    mask = np.linalg.norm(array.astype(np.float32) - color, axis=2) < 18.0
    boxes = []
    for left, top, right, bottom, area in _components(mask):
        width, height = right - left, bottom - top
        if width >= 260 and height >= 125 and area >= 5000:
            boxes.append((left, top, right, bottom))
    return boxes


def _read_sequence(
    array: np.ndarray, box: tuple[int, int, int, int], length: int
) -> list[int]:
    left, top, right, bottom = box
    width, height = right - left, bottom - top
    if width < 260 or height < 125 or not 1.75 < width / height < 2.25:
        raise ValueError("a card frame has invalid geometry")
    values: list[int] = []
    gap_fraction = 0.023
    cell_fraction = (0.90 - gap_fraction * (length - 1)) / length
    for index in range(length):
        # Central patches are located relative to the recovered frame, never by
        # renderer coordinates. They avoid both the frame and inter-cell gaps.
        center = 0.05 + index * (cell_fraction + gap_fraction) + cell_fraction / 2
        x0 = round(left + (center - 0.31 * cell_fraction) * width)
        x1 = round(left + (center + 0.31 * cell_fraction) * width)
        y0 = round(top + 0.28 * height)
        y1 = round(top + 0.72 * height)
        patch = array[y0:y1, x0:x1].astype(np.float32).reshape(-1, 3)
        if len(patch) < 300:
            raise ValueError("a tile patch is too small")
        distances = np.linalg.norm(patch[:, None, :] - PATTERN_COLORS[None, :, :], axis=2)
        nearest = distances.argmin(axis=1)
        counts = np.bincount(nearest, minlength=4)
        winner = int(counts.argmax())
        purity = float(counts[winner] / len(patch))
        close = float(np.mean(distances[:, winner] < 18.0))
        if purity < 0.94 or close < 0.94:
            raise ValueError(f"tile {index} is erased, mixed, or off-palette")
        values.append(winner)
    return values


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.shape[0] < 500 or array.shape[1] < 850:
        raise ValueError("canvas is too small for the five-card layout")

    source_boxes = sorted(_frame_boxes(array, INK))
    if len(source_boxes) != 2:
        raise ValueError(f"expected two disconnected source cards, found {len(source_boxes)}")
    source_boxes.sort(key=lambda box: box[0])
    if max(box[1] for box in source_boxes) >= array.shape[0] * 0.5:
        raise ValueError("source cards are not in the upper field")
    source_a = _read_sequence(array, source_boxes[0], 4)
    source_b = _read_sequence(array, source_boxes[1], 4)
    if sorted(source_a) != list(range(4)) or sorted(source_b) != list(range(4)):
        raise ValueError("each source must visibly contain each pattern color once")

    candidates: list[tuple[str, list[int], tuple[int, int, int, int]]] = []
    for tag, color in zip(TAG_NAMES, TAG_COLORS, strict=True):
        boxes = _frame_boxes(array, color)
        if len(boxes) != 1:
            raise ValueError(f"expected one disconnected {tag} frame, found {len(boxes)}")
        box = boxes[0]
        if box[1] <= array.shape[0] * 0.5:
            raise ValueError(f"{tag} candidate is not in the lower field")
        sequence = _read_sequence(array, box, 8)
        if sorted(sequence) != [0, 0, 1, 1, 2, 2, 3, 3]:
            raise ValueError(f"{tag} candidate does not have the balanced histogram")
        candidates.append((tag, sequence, box))

    if len({box[0] for _, _, box in candidates}) != 3:
        raise ValueError("candidate cards are not spatially distinct")
    target = [value for pair in zip(source_a, source_b, strict=True) for value in pair]
    matches = [tag for tag, sequence, _ in candidates if sequence == target]
    if len(matches) != 1:
        raise ValueError(f"expected one pixel-level zipper match, found {len(matches)}")

    wrong = [sequence for _, sequence, _ in candidates if sequence != target]
    a_correct = sorted(sequence[0::2] == source_a for sequence in wrong)
    b_correct = sorted(sequence[1::2] == source_b for sequence in wrong)
    if a_correct != [False, True] or b_correct != [False, True]:
        raise ValueError("candidate evidence does not expose both half-correct alternatives")
    return matches[0]
