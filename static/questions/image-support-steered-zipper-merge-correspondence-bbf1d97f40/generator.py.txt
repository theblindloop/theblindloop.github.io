"""Deterministic sampling and rendering for the zipper-merge world."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 1024
HEIGHT = 640
BACKGROUND = (250, 250, 247)
INK = (31, 38, 48)
PATTERN_COLORS = (
    (48, 105, 178),
    (218, 82, 78),
    (71, 157, 101),
    (235, 188, 62),
)
TAG_COLORS = {
    "amber": (244, 137, 32),
    "cyan": (35, 187, 207),
    "violet": (151, 83, 190),
}
TAG_NAMES = ("amber", "cyan", "violet")

CARD_W = 300
CARD_H = 150
SOURCE_Y = 72
SOURCE_XS = (32, 692)
CANDIDATE_Y = 374
CANDIDATE_XS = (32, 362, 692)
FRAME = 8
INNER_PAD_X = 15
INNER_PAD_Y = 22
CELL_GAP = 7


def _shuffle_permutation(rng: random.Random) -> list[int]:
    values = list(range(4))
    rng.shuffle(values)
    return values


def _rank_derangement(values: list[int], rng: random.Random) -> list[int]:
    """Rotate ranks so every position changes while the histogram is fixed."""
    shift = rng.choice((1, 2, 3))
    return values[shift:] + values[:shift]


def _zip(source_a: list[int], source_b: list[int]) -> list[int]:
    return [value for pair in zip(source_a, source_b, strict=True) for value in pair]


def _hamming(left: list[int], right: list[int]) -> int:
    return sum(a != b for a, b in zip(left, right, strict=True))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    source_a = _shuffle_permutation(rng)
    source_b = _shuffle_permutation(rng)
    target = _zip(source_a, source_b)

    # Each distractor is exactly right for one remote source and wrong at all
    # four ranks from the other. Thus either source alone leaves two candidates.
    wrong_b = _zip(source_a, _rank_derangement(source_b, rng))
    wrong_a = _zip(_rank_derangement(source_a, rng), source_b)

    candidate_tags = list(TAG_NAMES)
    rng.shuffle(candidate_tags)
    desired_tag = TAG_NAMES[int(seed) % len(TAG_NAMES)]
    gold_index = candidate_tags.index(desired_tag)
    candidates = [wrong_b, wrong_a]
    candidates.insert(gold_index, target)
    return {
        "source_a": source_a,
        "source_b": source_b,
        "candidates": candidates,
        "candidate_tags": candidate_tags,
    }


def analytic_gold(scene: dict) -> str:
    target = _zip(scene["source_a"], scene["source_b"])
    matches = [i for i, candidate in enumerate(scene["candidates"]) if candidate == target]
    if len(matches) != 1:
        raise ValueError(f"expected one exact zipper merge, found {len(matches)}")
    return scene["candidate_tags"][matches[0]]


def margin(scene: dict) -> float:
    target = _zip(scene["source_a"], scene["source_b"])
    distances = sorted(_hamming(target, candidate) for candidate in scene["candidates"])
    if len(distances) != 3 or distances[0] != 0:
        return 0.0
    return float(distances[1])


def is_quarantined(scene: dict) -> bool:
    try:
        source_a = scene["source_a"]
        source_b = scene["source_b"]
        candidates = scene["candidates"]
        tags = scene["candidate_tags"]
        if (
            sorted(source_a) != list(range(4))
            or sorted(source_b) != list(range(4))
            or len(candidates) != 3
            or any(len(candidate) != 8 for candidate in candidates)
            or any(sorted(candidate) != [0, 0, 1, 1, 2, 2, 3, 3] for candidate in candidates)
            or sorted(tags) != sorted(TAG_NAMES)
        ):
            return True
        target = _zip(source_a, source_b)
        matches = [candidate == target for candidate in candidates]
        if sum(matches) != 1 or margin(scene) < 4.0:
            return True
        wrong = [candidate for candidate, match in zip(candidates, matches, strict=True) if not match]
        a_correct = [candidate[0::2] == source_a for candidate in wrong]
        b_correct = [candidate[1::2] == source_b for candidate in wrong]
        return sorted(a_correct) != [False, True] or sorted(b_correct) != [False, True]
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _draw_card(
    draw: ImageDraw.ImageDraw,
    x: int,
    y: int,
    values: list[int],
    frame_color: tuple[int, int, int],
) -> None:
    draw.rectangle((x, y, x + CARD_W - 1, y + CARD_H - 1), fill=frame_color)
    draw.rectangle(
        (x + FRAME, y + FRAME, x + CARD_W - FRAME - 1, y + CARD_H - FRAME - 1),
        fill=BACKGROUND,
    )
    left = x + INNER_PAD_X
    right = x + CARD_W - INNER_PAD_X
    top = y + INNER_PAD_Y
    bottom = y + CARD_H - INNER_PAD_Y
    available = right - left - CELL_GAP * (len(values) - 1)
    cell_width = available / len(values)
    for index, value in enumerate(values):
        x0 = round(left + index * (cell_width + CELL_GAP))
        x1 = round(left + (index + 1) * cell_width + index * CELL_GAP) - 1
        draw.rectangle((x0, top, x1, bottom - 1), fill=PATTERN_COLORS[value])


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_card(draw, SOURCE_XS[0], SOURCE_Y, scene["source_a"], INK)
    _draw_card(draw, SOURCE_XS[1], SOURCE_Y, scene["source_b"], INK)
    for x, values, tag in zip(
        CANDIDATE_XS, scene["candidates"], scene["candidate_tags"], strict=True
    ):
        _draw_card(draw, x, CANDIDATE_Y, values, TAG_COLORS[tag])
    return image
