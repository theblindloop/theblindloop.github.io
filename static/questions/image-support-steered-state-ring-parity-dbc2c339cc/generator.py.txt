"""Compact state-ring toggle world (question-world@0.4.0).

One small connected annulus (the "state ring") sits at a randomized position on
an otherwise plain canvas. The ring is split into four equal sectors (top,
right, bottom, left); each sector is filled either blue (tracked state starts or
stays OFF) or orange (a flip). A small tick dot just above the ring declares the
initial tracked state: blue dot = OFF, orange dot = ON. Traverse the four sectors
clockwise starting from the sector just after the tick; every orange sector flips
the tracked state and every blue sector leaves it unchanged. The tracked state
finishes ON when start_state XOR (number of orange sectors mod 2) is true.

Distant blue/orange decoy dots elsewhere on the canvas are nondiagnostic: the
answer depends only on the one compact ring and its attached tick.

The decisive pixel-oracle support is therefore a single small connected
area-like component (the ring plus its tick) at a randomized absolute position;
remote same-colour marks never influence the decision (support_scale=focal,
support_shape=compact).
"""

from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image, ImageDraw

RENDERER_VERSION = "state_ring_parity-0.1.0"

BLUE = (0, 90, 210)
ORANGE = (255, 150, 30)
BACKGROUND = (255, 255, 255)

CANVAS = 448
R_OUTER = 50.0
R_INNER = 26.0
TICK_OFFSET = 13.0
TICK_R = 6.0
DECOY_R = 6.0
SUPERSAMPLE = 4

# Sector order being read clockwise from the tick: top -> right -> bottom -> left.
_SECTOR_ANGLE_DEG = {
    "top": (-135.0, -45.0, -90.0),      # (start angle, end angle, sample angle)
    "right": (-45.0, 45.0, 0.0),
    "bottom": (45.0, 135.0, 90.0),
    "left": (135.0, 225.0, 180.0),
}
_SECTOR_ORDER = ["top", "right", "bottom", "left"]

COLOR_RGB = {"blue": BLUE, "orange": ORANGE}


def _tick_position(scene: dict) -> tuple[float, float]:
    cx, cy = scene["center"]
    return (float(cx), float(cy) - (R_OUTER + TICK_OFFSET))


def _wedge_points(
    cx: float, cy: float, a0_deg: float, a1_deg: float, r_out: float, r_in: float
) -> list[tuple[float, float]]:
    pts = []
    n = 40
    for i in range(n + 1):
        a = a0_deg + (a1_deg - a0_deg) * i / n
        rad = math.radians(a)
        pts.append((cx + r_out * math.cos(rad), cy + r_out * math.sin(rad)))
    for i in range(n + 1):
        a = a1_deg - (a1_deg - a0_deg) * i / n
        rad = math.radians(a)
        pts.append((cx + r_in * math.cos(rad), cy + r_in * math.sin(rad)))
    return pts


def sample_scene(seed: int) -> dict:
    """Deterministically sample one scene from an integer seed."""
    rng = random.Random(seed)
    w = CANVAS
    lo, hi = R_OUTER + TICK_OFFSET + 24, w - (R_OUTER + TICK_OFFSET + 24)
    cx = round(rng.uniform(lo, hi))
    cy = round(rng.uniform(lo, hi))

    sector_colors = [rng.choice(["blue", "orange"]) for _ in range(4)]
    start_state = rng.choice(["blue", "orange"])

    tick = _tick_position({"center": (cx, cy)})
    decoys = []
    for _ in range(3):
        for _attempt in range(2000):
            dx = round(rng.uniform(24, w - 24))
            dy = round(rng.uniform(24, w - 24))
            d_center = math.hypot(dx - cx, dy - cy)
            d_tick = math.hypot(dx - tick[0], dy - tick[1])
            if d_center > 150 and d_tick > 70:
                decoys.append([dx, dy, rng.choice(["blue", "orange"])])
                break

    return {
        "seed": seed,
        "center": [int(cx), int(cy)],
        "r_outer": R_OUTER,
        "r_inner": R_INNER,
        "sector_colors": list(sector_colors),
        "start_state": start_state,
        "decoy_points": decoys,
    }


def render(scene: dict) -> Image.Image:
    ss = SUPERSAMPLE
    big = Image.new("RGB", (CANVAS * ss, CANVAS * ss), BACKGROUND)
    draw = ImageDraw.Draw(big)
    cx, cy = scene["center"]
    r_out = scene["r_outer"] * ss
    r_in = scene["r_inner"] * ss
    cx_ss, cy_ss = cx * ss, cy * ss

    for name, color in zip(_SECTOR_ORDER, scene["sector_colors"], strict=True):
        a0, a1, _sample = _SECTOR_ANGLE_DEG[name]
        pts = _wedge_points(cx_ss, cy_ss, a0, a1, r_out, r_in)
        draw.polygon(pts, fill=COLOR_RGB[color])

    tick_x, tick_y = _tick_position(scene)
    tr = TICK_R * ss
    draw.ellipse(
        [tick_x * ss - tr, tick_y * ss - tr, tick_x * ss + tr, tick_y * ss + tr],
        fill=COLOR_RGB[scene["start_state"]],
    )

    for x, y, color in scene["decoy_points"]:
        draw.ellipse(
            [x * ss - DECOY_R * ss, y * ss - DECOY_R * ss, x * ss + DECOY_R * ss, y * ss + DECOY_R * ss],
            fill=COLOR_RGB[color],
        )

    return big.resize((CANVAS, CANVAS), Image.LANCZOS)


def _orange_count(scene: dict) -> int:
    return sum(1 for c in scene["sector_colors"] if c == "orange")


def analytic_gold(scene: dict) -> str:
    start_on = scene["start_state"] == "orange"
    parity = _orange_count(scene) % 2
    final_on = bool(start_on) ^ bool(parity)
    return "yes" if final_on else "no"


def margin(scene: dict) -> float:
    return float(_orange_count(scene))


def is_quarantined(scene: dict) -> bool:
    cx, cy = scene["center"]
    tick = _tick_position(scene)
    for x, y, _color in scene["decoy_points"]:
        if math.hypot(x - cx, y - cy) < 150:
            return True
        if math.hypot(x - tick[0], y - tick[1]) < 70:
            return True
    return False


def latent_symmetries(scene: dict):
    return []


if __name__ == "__main__":
    for s in (0, 1, 2):
        sc = sample_scene(s)
        img = render(sc)
        print(s, sc["start_state"], sc["sector_colors"], analytic_gold(sc), img.size)
