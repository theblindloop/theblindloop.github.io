"""Independent pixel-only inverse arm for the compact state-ring world.

Reads only the final PNG. No scene, renderer, prompt, generator, or gold code is
imported. It locates the single largest connected blue-orange component (the
state ring), infers its centre and radii, reads the four sector colours at the
cardinal mid-ring sampling points, reads the tick dot colour just above the ring
(initial state), and reports whether the tracked state finishes ON.

Ablating the ring/tick region (or removing the ring entirely) makes this arm
abstain (raise) or change its decision, which is exactly the compact focal
support the world is designed around; remote decoy dots are ignored.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "state_ring_parity-oracle-0.1.0"

COLORS = np.array([[0, 90, 210], [255, 150, 30], [255, 255, 255]], dtype=np.float32)
BLUE = (0, 90, 210)
ORANGE = (255, 150, 30)

_MIN_RING_PX = 500
_TICK_SEARCH = 16


def _label(arr: np.ndarray) -> np.ndarray:
    d = np.linalg.norm(arr[:, :, None, :] - COLORS[None, None, :, :], axis=3)
    return d.argmin(axis=2)


def _largest_component(labels: np.ndarray) -> np.ndarray:
    colored = labels < 2
    h, w = colored.shape
    visited = np.zeros((h, w), dtype=bool)
    best_mask = np.zeros((h, w), dtype=bool)
    best_count = 0
    ys, xs = np.nonzero(colored)
    order = np.argsort(ys * w + xs)
    for idx in order:
        py = int(ys[idx])
        px = int(xs[idx])
        if visited[py, px]:
            continue
        count = 0
        stack = [(py, px)]
        visited[py, px] = True
        mask = np.zeros((h, w), dtype=bool)
        while stack:
            cy, cx = stack.pop()
            mask[cy, cx] = True
            count += 1
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < h and 0 <= nx < w and colored[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    stack.append((ny, nx))
        if count > best_count:
            best_count = count
            best_mask = mask
    return best_mask


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    h, w = arr.shape[:2]
    labels = _label(arr)
    mask = _largest_component(labels)
    count = int(mask.sum())
    if count < _MIN_RING_PX:
        raise ValueError("state ring not found (insufficient coloured area)")

    ys, xs = np.nonzero(mask)
    cy = float(ys.mean())
    cx = float(xs.mean())
    dists = np.hypot(ys - cy, xs - cx)
    r_o = float(dists.max())
    r_i = float(dists.min())
    r_mid = (r_o + r_i) / 2.0

    sector_colors = []
    for a_deg in (-90.0, 0.0, 90.0, 180.0):
        a = math.radians(a_deg)
        sx = int(round(cx + r_mid * math.cos(a)))
        sy = int(round(cy + r_mid * math.sin(a)))
        if not (0 <= sx < w and 0 <= sy < h):
            raise ValueError("sector sample out of bounds")
        c = int(labels[sy, sx])
        if c == 0:
            sector_colors.append("blue")
        elif c == 1:
            sector_colors.append("orange")
        else:
            raise ValueError("sector sample landed on background")

    found_start = None
    ty0 = int(round(cy - (r_o + 13.0)))
    tx0 = int(round(cx))
    for dy in range(-_TICK_SEARCH, _TICK_SEARCH + 1):
        for dx in range(-_TICK_SEARCH, _TICK_SEARCH + 1):
            ty = ty0 + dy
            tx = tx0 + dx
            if not (0 <= ty < h and 0 <= tx < w):
                continue
            if not (labels[ty, tx] < 2):
                continue
            found_start = "blue" if int(labels[ty, tx]) == 0 else "orange"
            break
        if found_start is not None:
            break
    if found_start is None:
        raise ValueError("state-ring tick not found")

    start_on = found_start == "orange"
    parity = sum(1 for c in sector_colors if c == "orange") % 2
    final_on = bool(start_on) ^ bool(parity)
    return "yes" if final_on else "no"
