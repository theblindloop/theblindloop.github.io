from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _large_components(mask: np.ndarray) -> list[int]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    sizes: list[int] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        size = 0
        while queue:
            cy, cx = queue.popleft()
            size += 1
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if size >= 80:
            sizes.append(size)
    return sizes


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[:, :, 0], array[:, :, 1], array[:, :, 2]
    cyan = (blue > 145) & (green > 135) & (red < 75) & ((blue - red) > 90)
    violet = (red > 105) & (blue > 135) & (green < 115) & ((blue - green) > 45)
    trace = cyan | violet
    if int(trace.sum()) < 5000:
        raise ValueError("colored trace is missing or too short")
    components = _large_components(trace)
    if len(components) != 1:
        raise ValueError("colored trace is not one continuous component")

    occupied_columns = np.flatnonzero(trace.any(axis=0))
    if len(occupied_columns) < 2:
        raise ValueError("trace extent is unresolved")
    left, right = int(occupied_columns[0]), int(occupied_columns[-1])
    if right - left < int(0.78 * array.shape[1]):
        raise ValueError("trace does not span the image")
    gaps = np.diff(occupied_columns)
    if len(gaps) and int(gaps.max()) > 3:
        raise ValueError("trace has an erased interval")

    operations: list[int] = []
    extent = right - left + 1
    for index in range(9):
        # Use the observed bounding extent, and trim junction pixels where two
        # colors meet. No renderer coordinate or latent state is available here.
        x0 = left + (extent * index) // 9 + 8
        x1 = left + (extent * (index + 1)) // 9 - 8
        if x1 <= x0:
            raise ValueError("span is too narrow")
        cyan_count = int(cyan[:, x0:x1].sum())
        violet_count = int(violet[:, x0:x1].sum())
        total = cyan_count + violet_count
        if total < 250 or abs(cyan_count - violet_count) < 0.55 * total:
            raise ValueError("span color is ambiguous")
        operations.append(1 if cyan_count > violet_count else -1)

    balance = 0
    checkpoints = []
    for index, operation in enumerate(operations, start=1):
        balance += operation
        if index >= 7:
            checkpoints.append(abs(balance))
    ranked = sorted((value, index) for index, value in enumerate(checkpoints))
    if ranked[0][0] == ranked[1][0]:
        raise ValueError("checkpoint decision is tied")
    return ("A", "B", "C")[ranked[0][1]]
