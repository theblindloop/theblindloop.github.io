from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 360
CYAN = ((22, 177, 207), (18, 161, 193))
VIOLET = ((145, 76, 190), (130, 68, 176))
INK = (38, 43, 51)
GOLD = (225, 166, 35)
LABELS = ("A", "B", "C")


def _checkpoint_values(scene: dict) -> tuple[int, int, int]:
    balance = 0
    values = []
    for index, operation in enumerate(scene["ops"], start=1):
        balance += int(operation)
        if index >= 7:
            values.append(balance)
    return tuple(values)


def _winner_and_gap(scene: dict) -> tuple[str, float]:
    magnitudes = [abs(value) for value in _checkpoint_values(scene)]
    ranked = sorted((value, index) for index, value in enumerate(magnitudes))
    winner = LABELS[ranked[0][1]]
    return winner, float(ranked[1][0] - ranked[0][0])


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = LABELS[int(seed) % len(LABELS)]
    possibilities = list(itertools.product((-1, 1), repeat=9))
    rng.shuffle(possibilities)
    chosen = None
    for operations in possibilities:
        probe = {"ops": list(operations)}
        winner, gap = _winner_and_gap(probe)
        if winner == target and gap >= 1.0:
            chosen = list(operations)
            break
    if chosen is None:
        raise RuntimeError("could not construct a balanced trace")
    levels = (153, 180, 207)
    ys = [rng.choice(levels)]
    for _ in range(9):
        choices = [value for value in levels if abs(value - ys[-1]) <= 27]
        ys.append(rng.choice(choices))
    return {
        "ops": chosen,
        "ys": ys,
        "style_variant": int(seed) % 2,
    }


def analytic_gold(scene: dict) -> str:
    return _winner_and_gap(scene)[0]


def margin(scene: dict) -> float:
    return _winner_and_gap(scene)[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    variant = int(scene["style_variant"])
    background = (249, 247, 241) if variant == 0 else (245, 248, 247)
    image = Image.new("RGB", (WIDTH, HEIGHT), background)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    draw.text((32, 24), "RUNNING BALANCE TRACE", fill=INK, font=font)
    draw.text((32, 45), "cyan +1    violet -1", fill=(75, 80, 87), font=font)
    draw.line((45, 82, 915, 82), fill=(212, 210, 202), width=2)

    xs = [65 + 92 * index for index in range(10)]
    ys = [int(value) for value in scene["ys"]]
    operations = [int(value) for value in scene["ops"]]
    cyan, violet = CYAN[variant], VIOLET[variant]
    x0, y0 = xs[0], ys[0]
    draw.ellipse((x0 - 15, y0 - 15, x0 + 15, y0 + 15), outline=GOLD, width=5)
    for index, operation in enumerate(operations):
        color = cyan if operation == 1 else violet
        draw.line((xs[index], ys[index], xs[index + 1], ys[index + 1]), fill=color, width=12)

    # Round origin and directional arrow are neutral so the operation parser only
    # sees the colored connected trace itself.
    draw.text((x0 - 21, y0 + 23), "START 0", fill=INK, font=font)
    xe, ye = xs[-1], ys[-1]
    draw.polygon(((xe + 23, ye), (xe + 2, ye - 14), (xe + 2, ye + 14)), fill=INK)

    for label, vertex_index in zip(LABELS, (7, 8, 9), strict=True):
        x, y = xs[vertex_index], ys[vertex_index]
        label_y = y - 45 if vertex_index % 2 else y + 31
        connector_end = y - 12 if label_y < y else y + 12
        connector_start = label_y + 16 if label_y < y else label_y - 5
        draw.line((x, connector_start, x, connector_end), fill=INK, width=2)
        draw.rounded_rectangle((x - 12, label_y - 4, x + 12, label_y + 15), radius=4, fill=background, outline=INK, width=2)
        draw.text((x - 3, label_y), label, fill=INK, font=font)

    draw.text((273, 312), "Choose the checkpoint whose balance is closest to zero.", fill=(75, 80, 87), font=font)
    return image
