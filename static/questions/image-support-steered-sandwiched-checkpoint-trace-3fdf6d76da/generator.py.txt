"""Deterministic latent sampler and raster renderer for a local sequence motif trace."""

from __future__ import annotations

import itertools
import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 640
HEIGHT = 480

_ROUTE_PALETTES = ((23, 43, 70), (31, 48, 76), (39, 42, 68))
_BACKGROUNDS = ((248, 247, 242), (247, 249, 246), (250, 247, 245))
_STARTS = ((35, 105, 210), (43, 113, 218), (31, 98, 201))
_FINISHES = ((238, 132, 42), (232, 122, 35), (242, 143, 50))
_REDS = ((210, 55, 65), (202, 48, 62), (218, 62, 72))
_GREENS = ((35, 158, 92), (42, 166, 98), (30, 148, 86))


def _motif_indices(colors: list[str]) -> list[int]:
    return [
        index
        for index in range(1, len(colors) - 1)
        if colors[index - 1] == colors[index + 1] != colors[index]
    ]


def _word_pool(answer: str) -> tuple[tuple[str, ...], ...]:
    words = []
    for bits in itertools.product(("red", "green"), repeat=8):
        colors = list(bits)
        hits = _motif_indices(colors)
        if colors.count("red") == 4 and len(hits) == 1 and colors[hits[0]] == answer:
            words.append(bits)
    return tuple(words)


_WORDS = {answer: _word_pool(answer) for answer in ("red", "green")}


def _transform_points(points: list[tuple[int, int]], mode: int) -> list[list[int]]:
    transformed = []
    for x, y in points:
        if mode & 1:
            x = WIDTH - x
        if mode & 2:
            y = HEIGHT - y
        transformed.append([int(x), int(y)])
    return transformed


def _polyline_lengths(
    points: Iterable[Iterable[float]],
) -> tuple[list[tuple[float, float]], list[float], float]:
    pts = [(float(point[0]), float(point[1])) for point in points]
    lengths = [math.hypot(b[0] - a[0], b[1] - a[1]) for a, b in zip(pts, pts[1:])]
    return pts, lengths, sum(lengths)


def sample_scene(seed: int) -> dict:
    """Sample a class-balanced route; consecutive integer seeds alternate answers."""

    rng = random.Random(int(seed))
    dx = rng.randint(-13, 13)
    dy = rng.randint(-9, 9)
    base = [
        (160, 318),
        (220, 214),
        (300, 270),
        (360, 158),
        (438, 214),
        (486, 118),
    ]
    jittered = [
        (x + dx + rng.randint(-4, 4), y + dy + rng.randint(-4, 4)) for x, y in base
    ]
    points = _transform_points(jittered, rng.randrange(4))
    if rng.randrange(2):
        points.reverse()

    metric_points, segment_lengths, total = _polyline_lengths(points)
    schedule = ((0, 0.38), (0, 0.76), (1, 0.55), (2, 0.30), (2, 0.70), (3, 0.55), (4, 0.24), (4, 0.62))
    consumed = [0.0]
    for length in segment_lengths:
        consumed.append(consumed[-1] + length)
    checkpoint_indices = []
    for segment, local in schedule:
        local += rng.uniform(-0.012, 0.012)
        checkpoint_indices.append(
            round((consumed[segment] + local * segment_lengths[segment]) / total, 5)
        )

    answer = "red" if int(seed) % 2 == 0 else "green"
    colors = list(rng.choice(_WORDS[answer]))
    palette_index = rng.randrange(len(_ROUTE_PALETTES))
    return {
        "route_points": points,
        "checkpoint_indices": checkpoint_indices,
        "checkpoint_colors": colors,
        "route_color": list(_ROUTE_PALETTES[palette_index]),
        "background_color": list(_BACKGROUNDS[palette_index]),
        "start_color": list(_STARTS[palette_index]),
        "finish_color": list(_FINISHES[palette_index]),
        "red_color": list(_REDS[palette_index]),
        "green_color": list(_GREENS[palette_index]),
        "stroke_width": 9,
        "marker_radius": 8,
    }


def _station(
    points: list[tuple[float, float]],
    lengths: list[float],
    total: float,
    fraction: float,
) -> tuple[float, float, float, float]:
    target = max(0.0, min(1.0, float(fraction))) * total
    consumed = 0.0
    for (x0, y0), (x1, y1), length in zip(points, points[1:], lengths):
        if target <= consumed + length:
            local = (target - consumed) / length
            x = x0 + local * (x1 - x0)
            y = y0 + local * (y1 - y0)
            return x, y, (x1 - x0) / length, (y1 - y0) / length
        consumed += length
    x0, y0 = points[-2]
    x1, y1 = points[-1]
    length = lengths[-1]
    return x1, y1, (x1 - x0) / length, (y1 - y0) / length


def render(scene: dict) -> Image.Image:
    """Draw one continuous regional route and eight route-adjacent rings."""

    route_points = scene["route_points"]
    checkpoint_indices = scene["checkpoint_indices"]
    checkpoint_colors = scene["checkpoint_colors"]
    route_color = tuple(scene["route_color"])
    background = tuple(scene["background_color"])
    start_color = tuple(scene["start_color"])
    finish_color = tuple(scene["finish_color"])
    red_color = tuple(scene["red_color"])
    green_color = tuple(scene["green_color"])
    stroke_width = int(scene["stroke_width"])
    marker_radius = int(scene["marker_radius"])

    image = Image.new("RGB", (WIDTH, HEIGHT), background)
    draw = ImageDraw.Draw(image)
    points = [(int(point[0]), int(point[1])) for point in route_points]
    draw.line(points, fill=route_color, width=stroke_width, joint="curve")

    metric_points, lengths, total = _polyline_lengths(route_points)
    for index, (fraction, name) in enumerate(zip(checkpoint_indices, checkpoint_colors)):
        x, y, tx, ty = _station(metric_points, lengths, total, fraction)
        side = 1.0 if index % 2 == 0 else -1.0
        cx = x + side * (-ty) * 14.0
        cy = y + side * tx * 14.0
        color = red_color if name == "red" else green_color
        draw.ellipse(
            (
                round(cx - marker_radius),
                round(cy - marker_radius),
                round(cx + marker_radius),
                round(cy + marker_radius),
            ),
            fill=color,
        )

    for center, color in ((points[0], start_color), (points[-1], finish_color)):
        x, y = center
        draw.ellipse((x - 16, y - 16, x + 16, y + 16), outline=color, width=5)
    return image


def analytic_gold(scene: dict) -> str:
    colors = scene["checkpoint_colors"]
    hits = _motif_indices(colors)
    if len(hits) != 1:
        raise ValueError("scene does not contain exactly one sandwiched checkpoint")
    return colors[hits[0]]


def margin(scene: dict) -> float:
    """Top-versus-runner-up gap among the six discrete ABA motif scores."""

    colors = scene["checkpoint_colors"]
    scores = [
        float(colors[index - 1] == colors[index + 1] != colors[index])
        for index in range(1, len(colors) - 1)
    ]
    scores.sort(reverse=True)
    return scores[0] - scores[1]


def is_quarantined(scene: dict) -> bool:
    colors = list(scene["checkpoint_colors"])
    fractions = [float(value) for value in scene["checkpoint_indices"]]
    points, lengths, _ = _polyline_lengths(scene["route_points"])
    return not (
        len(points) == 6
        and min(lengths, default=0.0) >= 80.0
        and len(colors) == 8
        and colors.count("red") == colors.count("green") == 4
        and len(_motif_indices(colors)) == 1
        and len(fractions) == 8
        and 0.06 <= fractions[0]
        and fractions[-1] <= 0.94
        and all(b - a >= 0.055 for a, b in zip(fractions, fractions[1:]))
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
