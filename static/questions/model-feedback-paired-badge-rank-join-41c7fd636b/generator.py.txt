def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    candidates = (
        Path("/usr/share/fonts/truetype/dejavu") / name,
        Path("/usr/share/fonts/dejavu") / name,
    )
    for path in candidates:
        if path.is_file():
            return ImageFont.truetype(str(path), size)
    return ImageFont.load_default()


def _draw_bitmap(
    draw: ImageDraw.ImageDraw,
    left: int,
    top: int,
    bitmap: tuple[str, ...],
    *,
    fill: tuple[int, int, int] = INK,
) -> None:
    for row_index, row in enumerate(bitmap):
        for col_index, value in enumerate(row):
            if value == "1":
                x0 = left + col_index * CELL
                y0 = top + row_index * CELL
                draw.rectangle((x0, y0, x0 + CELL - 1, y0 + CELL - 1), fill=fill)


def _draw_icon_box(
    draw: ImageDraw.ImageDraw,
    left: int,
    top: int,
    bitmap: tuple[str, ...],
    *,
    outline: tuple[int, int, int] = INK,
    fill: tuple[int, int, int] = (242, 246, 249),
    width: int = 3,
) -> None:
    draw.rounded_rectangle(
        (left, top, left + ICON_SIZE - 1, top + ICON_SIZE - 1),
        radius=12,
        fill=fill,
        outline=outline,
        width=width,
    )
    _draw_bitmap(draw, left + ICON_INSET, top + ICON_INSET, bitmap)


def _draw_arrow(draw: ImageDraw.ImageDraw, x0: int, y: int, x1: int) -> None:
    draw.line((x0, y, x1 - 13, y), fill=MUTED, width=4)
    draw.polygon(((x1 - 13, y - 9), (x1, y), (x1 - 13, y + 9)), fill=MUTED)


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    box: tuple[int, int, int, int],
    title: str,
    subtitle: str,
) -> None:
    x, y, w, h = box
    draw.rounded_rectangle((x, y, x + w, y + h), radius=18, fill=PANEL, outline=LINE, width=3)
    draw.text((x + 22, y + 18), title, font=_font(25, True), fill=INK)
    draw.text((x + 22, y + 54), subtitle, font=_font(15), fill=MUTED)


def _valid_pair_rows(rows: object, second_limit: int) -> bool:
    if not isinstance(rows, list) or len(rows) != IDENTITY_COUNT:
        return False
    for row in rows:
        if not isinstance(row, list) or len(row) != 2:
            return False
        if any(not isinstance(value, int) or isinstance(value, bool) for value in row):
            return False
        if not 0 <= row[0] < IDENTITY_COUNT or not 0 <= row[1] < second_limit:
            return False
    return True


def _legal_scene_values(query_badges: object, key_one: object, key_two: object) -> bool:
    if (
        not isinstance(query_badges, list)
        or len(query_badges) != 2
        or any(not isinstance(value, int) or isinstance(value, bool) for value in query_badges)
        or len(set(query_badges)) != 2
        or any(not 0 <= value < IDENTITY_COUNT for value in query_badges)
    ):
        return False
    if not _valid_pair_rows(key_one, IDENTITY_COUNT) or not _valid_pair_rows(key_two, IDENTITY_COUNT):
        return False
    if sorted(row[0] for row in key_one) != list(range(IDENTITY_COUNT)):
        return False
    if sorted(row[1] for row in key_one) != list(range(IDENTITY_COUNT)):
        return False
    if sorted(row[0] for row in key_two) != list(range(IDENTITY_COUNT)):
        return False
    if sorted(row[1] for row in key_two) != list(range(IDENTITY_COUNT)):
        return False
    first = {badge: ribbon for badge, ribbon in key_one}
    second = {ribbon: rank for ribbon, rank in key_two}
    return second[first[query_badges[0]]] != second[first[query_badges[1]]]


def sample_scene(seed: int) -> dict:
    """Sample two query badges and two independently permuted visible tables."""

    rng = random.Random(int(seed))
    query_badges = rng.sample(range(IDENTITY_COUNT), 2)

    badge_order = list(range(IDENTITY_COUNT))
    rng.shuffle(badge_order)
    badge_to_ribbon = list(range(IDENTITY_COUNT))
    rng.shuffle(badge_to_ribbon)
    key_one = [[badge, badge_to_ribbon[badge]] for badge in badge_order]

    ribbon_order = list(range(IDENTITY_COUNT))
    rng.shuffle(ribbon_order)
    ribbon_to_rank = list(range(IDENTITY_COUNT))
    rng.shuffle(ribbon_to_rank)
    key_two = [[ribbon, ribbon_to_rank[ribbon]] for ribbon in ribbon_order]
    return {
        "query_badges": query_badges,
        "key_one": key_one,
        "key_two": key_two,
    }


def analytic_gold(scene: dict) -> str:
    """Compare the two ranks reached by the two visible query badges."""

    query_badges = scene["query_badges"]
    key_one = scene["key_one"]
    key_two = scene["key_two"]
    if not _legal_scene_values(query_badges, key_one, key_two):
        raise ValueError("scene is not a legal paired rank join")
    badge_to_ribbon = {badge: ribbon for badge, ribbon in key_one}
    ribbon_to_rank = {ribbon: rank for ribbon, rank in key_two}
    first_rank = ribbon_to_rank[badge_to_ribbon[query_badges[0]]]
    second_rank = ribbon_to_rank[badge_to_ribbon[query_badges[1]]]
    return ANSWERS[0] if first_rank < second_rank else ANSWERS[1]


def margin(scene: dict) -> float:
    query_badges = scene["query_badges"]
    key_one = scene["key_one"]
    key_two = scene["key_two"]
    if not _legal_scene_values(query_badges, key_one, key_two):
        return 0.0
    badge_to_ribbon = {badge: ribbon for badge, ribbon in key_one}
    ribbon_to_rank = {ribbon: rank for ribbon, rank in key_two}
    first_rank = ribbon_to_rank[badge_to_ribbon[query_badges[0]]]
    second_rank = ribbon_to_rank[badge_to_ribbon[query_badges[1]]]
    return float(abs(first_rank - second_rank))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _draw_rank_meter(draw: ImageDraw.ImageDraw, left: int, top: int, rank: int) -> None:
    for index in range(IDENTITY_COUNT):
        x0 = left + index * (RANK_SEG_W + RANK_GAP)
        fill = RANK_FILL if index <= rank else RANK_EMPTY
        draw.rounded_rectangle(
            (x0, top, x0 + RANK_SEG_W - 1, top + RANK_H - 1),
            radius=5,
            fill=fill,
            outline=LINE,
            width=2,
        )


def render(scene: dict) -> Image.Image:
    """Render the paired query, shuffled join key, and shuffled rank key."""

    query_badges = scene["query_badges"]
    key_one = scene["key_one"]
    key_two = scene["key_two"]
    if not _legal_scene_values(query_badges, key_one, key_two):
        raise ValueError("cannot render an invalid paired rank scene")

    image = Image.new("RGB", (WIDTH, HEIGHT), PAPER)
    draw = ImageDraw.Draw(image)
    draw.text((34, 25), "PAIRED RANK JOIN", font=_font(31, True), fill=INK)
    draw.text(
        (34, 66),
        "Carry both query identities through two shuffled tables.",
        font=_font(16),
        fill=MUTED,
    )

    _draw_panel(draw, (QUERY_X, PANEL_Y, QUERY_W, PANEL_H), "QUERY", "compare P and Q")
    _draw_panel(draw, (PANEL_A_X, PANEL_Y, PANEL_A_W, PANEL_H), "KEY A", "badge  →  ribbon")
    _draw_panel(draw, (PANEL_B_X, PANEL_Y, PANEL_B_W, PANEL_H), "KEY B", "ribbon  →  rank")

    draw.text((A_BADGE_X, 179), "BADGE", font=_font(13, True), fill=MUTED)
    draw.text((A_RIBBON_X, 179), "RIBBON", font=_font(13, True), fill=MUTED)
    draw.text((B_RIBBON_X, 179), "RIBBON", font=_font(13, True), fill=MUTED)
    draw.text((B_RANK_X, 179), "RANK: 1 low  →  6 high", font=_font(13, True), fill=MUTED)

    for label, badge, top in zip(("P", "Q"), query_badges, QUERY_TOPS):
        draw.rounded_rectangle(
            (64, top - 23, 286, top + ICON_SIZE + 22),
            radius=15,
            fill=(255, 252, 242),
            outline=GOLD,
            width=5,
        )
        draw.text((79, top + 30), label, font=_font(27, True), fill=GOLD)
        _draw_icon_box(
            draw,
            QUERY_ICON_X,
            top,
            BADGE_BITMAPS[badge],
            outline=GOLD,
            fill=(255, 255, 255),
            width=4,
        )

    draw.line((72, 661, 280, 661), fill=LINE, width=2)
    draw.text((69, 690), "Find P and Q", font=_font(16, True), fill=INK)
    draw.text((69, 719), "separately in KEY A.", font=_font(16), fill=MUTED)
    draw.text((69, 766), "Then compare their", font=_font(16), fill=MUTED)
    draw.text((69, 795), "ranks from KEY B.", font=_font(16), fill=MUTED)
    draw.text((69, 857), "smaller meter = earlier rank", font=_font(14), fill=MUTED)

    for row_index, (badge, ribbon) in enumerate(key_one):
        top = ROW_TOP + row_index * ROW_H
        draw.rounded_rectangle(
            (PANEL_A_X + 17, top - 8, PANEL_A_X + PANEL_A_W - 17, top + 95),
            radius=13,
            fill=ROW,
            outline=(225, 231, 236),
            width=2,
        )
        _draw_icon_box(draw, A_BADGE_X, top, BADGE_BITMAPS[badge])
        _draw_arrow(draw, A_BADGE_X + ICON_SIZE + 16, top + 43, A_RIBBON_X - 14)
        _draw_icon_box(draw, A_RIBBON_X, top, RIBBON_BITMAPS[ribbon], fill=(245, 243, 250))

    for row_index, (ribbon, rank) in enumerate(key_two):
        top = ROW_TOP + row_index * ROW_H
        draw.rounded_rectangle(
            (PANEL_B_X + 17, top - 8, PANEL_B_X + PANEL_B_W - 17, top + 95),
            radius=13,
            fill=ROW,
            outline=(225, 231, 236),
            width=2,
        )
        _draw_icon_box(draw, B_RIBBON_X, top, RIBBON_BITMAPS[ribbon], fill=(245, 243, 250))
        _draw_arrow(draw, B_RIBBON_X + ICON_SIZE + 16, top + 43, B_RANK_X - 14)
        _draw_rank_meter(draw, B_RANK_X, top + 26, rank)
        draw.text(
            (B_RANK_X + RANK_TOTAL_W + 16, top + 34),
            str(rank + 1),
            font=_font(18, True),
            fill=INK,
        )

    draw.text((PANEL_A_X + 22, 895), "Rows are independently shuffled.", font=_font(14), fill=MUTED)
    draw.text((PANEL_B_X + 22, 895), "Read rank only after matching ribbon.", font=_font(14), fill=MUTED)
    return image
