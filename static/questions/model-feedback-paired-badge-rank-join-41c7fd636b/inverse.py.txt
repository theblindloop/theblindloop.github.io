"""Independent raster-only inverse for the paired badge rank join board."""

from __future__ import annotations

from PIL import Image


IDENTITY_COUNT = 6
ANSWERS = ("P", "Q")
INK = (24, 42, 61)
RANK_FILL = (54, 122, 174)
BADGES = (
    ("0111110", "1100011", "1000001", "1000001", "1000001", "1100011", "0111110"),
    ("0001000", "0001000", "0001000", "1111111", "0001000", "0001000", "0001000"),
    ("1111111", "0111110", "0011100", "0001000", "0001000", "0001000", "0001000"),
    ("1000001", "0100010", "0010100", "0001000", "0010100", "0100010", "1000001"),
    ("1010101", "0000000", "1111111", "0000000", "1010101", "0000000", "1111111"),
    ("0011100", "0111110", "1111111", "0011100", "0011100", "0011100", "0011100"),
)
RIBBONS = (
    ("0011100", "0100010", "1000001", "1000001", "1000001", "0100010", "0011100"),
    ("0001000", "0011100", "0101010", "1000001", "1111111", "0001000", "0001000"),
    ("1111111", "1000001", "1010101", "1000001", "1010101", "1000001", "1111111"),
    ("1000001", "0100010", "0010100", "0001000", "0010100", "0100010", "1000001"),
    ("1111111", "1000000", "1011110", "1000001", "1011110", "1000000", "1111111"),
    ("1000001", "1100011", "0110110", "0001000", "0110110", "1100011", "1000001"),
)

ROW_TOP = 205
ROW_H = 112
ICON_INSET = 9
CELL = 10
QUERY_ICON_X = 128
QUERY_TOPS = (274, 494)
A_BADGE_X, A_RIBBON_X = 402, 795
B_RIBBON_X, B_RANK_X = 1042, 1295
RANK_SEG_W, RANK_GAP, RANK_H = 22, 7, 36


def _dark(pixel: tuple[int, int, int]) -> bool:
    return sum(pixel) < 500


def _read_bitmap(
    image: Image.Image,
    left: int,
    top: int,
    templates: tuple[tuple[str, ...], ...],
) -> int:
    observed = []
    for row in range(7):
        values = []
        for col in range(7):
            pixel = image.getpixel(
                (left + ICON_INSET + col * CELL + 5, top + ICON_INSET + row * CELL + 5)
            )
            values.append("1" if _dark(pixel) else "0")
        observed.append("".join(values))
    scores = [sum(a != b for a, b in zip(observed, template)) for template in templates]
    best = min(range(len(scores)), key=scores.__getitem__)
    if scores[best] != 0:
        raise ValueError("icon bitmap is not legible")
    return best


def _read_rank(image: Image.Image, left: int, top: int) -> int:
    values = []
    for index in range(IDENTITY_COUNT):
        x = left + index * (RANK_SEG_W + RANK_GAP) + RANK_SEG_W // 2
        y = top + RANK_H // 2
        pixel = image.getpixel((x, y))
        values.append(pixel == RANK_FILL)
    count = sum(values)
    if count < 1 or any(not value for value in values[:count]) or any(values[count:]):
        raise ValueError("rank meter is not a filled prefix")
    return count - 1


def decision_from_image(image: Image.Image) -> str:
    """Recover both badge -> ribbon -> rank paths using PNG pixels only."""

    image = image.convert("RGB")
    query = [
        _read_bitmap(image, QUERY_ICON_X, top, BADGES)
        for top in QUERY_TOPS
    ]
    if len(set(query)) != 2:
        raise ValueError("query badges are not distinct")

    badge_to_ribbon = {}
    for row in range(IDENTITY_COUNT):
        top = ROW_TOP + row * ROW_H
        badge = _read_bitmap(image, A_BADGE_X, top, BADGES)
        ribbon = _read_bitmap(image, A_RIBBON_X, top, RIBBONS)
        if badge in badge_to_ribbon:
            raise ValueError("duplicate badge in KEY A")
        badge_to_ribbon[badge] = ribbon

    ribbon_to_rank = {}
    for row in range(IDENTITY_COUNT):
        top = ROW_TOP + row * ROW_H
        ribbon = _read_bitmap(image, B_RIBBON_X, top, RIBBONS)
        rank = _read_rank(image, B_RANK_X, top + 26)
        if ribbon in ribbon_to_rank:
            raise ValueError("duplicate ribbon in KEY B")
        ribbon_to_rank[ribbon] = rank

    if set(badge_to_ribbon) != set(range(IDENTITY_COUNT)):
        raise ValueError("incomplete badge key")
    if set(ribbon_to_rank) != set(range(IDENTITY_COUNT)):
        raise ValueError("incomplete ribbon key")
    if set(ribbon_to_rank.values()) != set(range(IDENTITY_COUNT)):
        raise ValueError("rank values are not a permutation")

    first_rank = ribbon_to_rank[badge_to_ribbon[query[0]]]
    second_rank = ribbon_to_rank[badge_to_ribbon[query[1]]]
    if first_rank == second_rank:
        raise ValueError("queried ranks tie")
    return ANSWERS[0] if first_rank < second_rank else ANSWERS[1]
