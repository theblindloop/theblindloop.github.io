"""Scene-free pixel inverse for alternating-sides search.

The oracle discovers saturated connected components, distinguishes long axes
from compact dot components by visible shape, groups dots to their nearest
axis, and computes above/below transition counts. It imports no world code.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int, float, float]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y0, x0 in zip(*np.nonzero(mask & ~seen), strict=True):
        if seen[y0, x0]:
            continue
        queue = deque([(int(y0), int(x0))])
        seen[y0, x0] = True
        pixels: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            pixels.append((y, x))
            for yy, xx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= yy < h and 0 <= xx < w and mask[yy, xx] and not seen[yy, xx]:
                    seen[yy, xx] = True
                    queue.append((yy, xx))
        if len(pixels) < 20:
            continue
        ys = np.fromiter((p[0] for p in pixels), dtype=float)
        xs = np.fromiter((p[1] for p in pixels), dtype=float)
        output.append((int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max()), len(pixels), float(xs.mean()), float(ys.mean())))
    return output


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    chroma = arr.max(axis=2) - arr.min(axis=2)
    saturated = (chroma > 55) & (arr.min(axis=2) < 210)
    comps = _components(saturated)
    axes = []
    dots = []
    for x0, y0, x1, y1, area, cx, cy in comps:
        width, height = x1 - x0 + 1, y1 - y0 + 1
        if width > 130 and height <= 10 and area > 500:
            axes.append((cx, cy, width))
        elif 16 <= width <= 24 and 16 <= height <= 24 and 240 <= area <= 420:
            dots.append((cx, cy))
    if len(axes) != 4 or len(dots) != 24:
        raise ValueError(f"abstain: expected 4 axes and 24 dots, found {len(axes)} and {len(dots)}")
    axes.sort(key=lambda item: (item[1], item[0]))
    groups: list[list[tuple[float, float]]] = [[] for _ in axes]
    for dot in dots:
        distances = [((dot[0] - ax[0]) / ax[2]) ** 2 + ((dot[1] - ax[1]) / 90.0) ** 2 for ax in axes]
        groups[int(np.argmin(distances))].append(dot)
    scores = []
    for points, (_, axis_y, _) in zip(groups, axes, strict=True):
        if len(points) != 6:
            raise ValueError("abstain: a panel does not contain six dots")
        points.sort()
        sides = []
        for _, cy in points:
            delta = cy - axis_y
            if abs(delta) < 18:
                raise ValueError("abstain: dot too close to axis")
            sides.append(1 if delta > 0 else -1)
        if sides.count(-1) != 3 or sides.count(1) != 3:
            raise ValueError("abstain: panel lacks balanced side counts")
        scores.append(sum(a != b for a, b in zip(sides, sides[1:])))
    winners = [index for index, score in enumerate(scores) if score == 5]
    if len(winners) != 1:
        raise ValueError(f"abstain: expected one fully alternating panel, scores={scores}")
    runner_up = sorted(scores, reverse=True)[1]
    if 5 - runner_up < 1:
        raise ValueError("abstain: insufficient runner-up gap")
    return LABELS[winners[0]]
