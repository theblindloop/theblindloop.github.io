"""Independent image-only inverse arm for XOR register replay."""

from __future__ import annotations

import numpy as np
from PIL import Image

TOKEN_COLORS = ((211, 62, 72), (42, 119, 211), (35, 164, 101), (231, 137, 31))
STATE_COLORS = {True: (246, 192, 35), False: (128, 83, 181)}


def _near(rgb: np.ndarray, color: tuple[int, int, int], radius: int = 12) -> np.ndarray:
    delta = rgb.astype(np.int32) - np.asarray(color, dtype=np.int32)
    return np.sum(delta * delta, axis=2) <= radius * radius


def _components(mask: np.ndarray, minimum: int = 120):
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y0, x0 in zip(*np.nonzero(mask)):
        if seen[y0, x0]:
            continue
        stack = [(int(y0), int(x0))]
        seen[y0, x0] = True
        points = []
        while stack:
            y, x = stack.pop()
            points.append((y, x))
            for dy, dx in ((-1,0),(1,0),(0,-1),(0,1)):
                yy, xx = y+dy, x+dx
                if 0 <= yy < h and 0 <= xx < w and mask[yy,xx] and not seen[yy,xx]:
                    seen[yy,xx] = True
                    stack.append((yy,xx))
        if len(points) >= minimum:
            ys = np.fromiter((p[0] for p in points), dtype=int)
            xs = np.fromiter((p[1] for p in points), dtype=int)
            found.append((len(points), int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max()), float(xs.mean()), float(ys.mean())))
    return found


def _shape(component) -> str:
    area, x0, y0, x1, y1, cx, cy = component
    width, height = x1-x0+1, y1-y0+1
    occupancy = area / float(width * height)
    relative_y = (cy-y0) / max(1.0, height-1)
    if occupancy > 0.90:
        return "square"
    if occupancy > 0.69:
        return "circle"
    if relative_y > 0.55:
        return "triangle"
    return "diamond"


def _token_components(rgb: np.ndarray):
    mask = np.zeros(rgb.shape[:2], dtype=bool)
    for color in TOKEN_COLORS:
        mask |= _near(rgb, color)
    return _components(mask)


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"))
    if rgb.shape[0] < 600 or rgb.shape[1] < 800:
        raise ValueError("raster is too small")
    tokens = _token_components(rgb)
    initial = sorted([c for c in tokens if 110 < c[6] < 200 and c[0] > 500], key=lambda c: c[5])
    cards = sorted([c for c in tokens if 390 < c[6] < 490 and 180 < c[0] < 1300], key=lambda c: c[5])
    query = [c for c in tokens if c[6] > 570 and c[0] > 450]
    if len(initial) != 4 or len(cards) != 15 or len(query) != 1:
        raise ValueError(f"unreadable token structure: {len(initial)}/{len(cards)}/{len(query)}")
    identities = [_shape(c) for c in initial]
    if set(identities) != {"circle", "square", "triangle", "diamond"}:
        raise ValueError("initial identities are ambiguous")

    states = []
    for value, color in STATE_COLORS.items():
        states.extend((c[5], value) for c in _components(_near(rgb, color), 400) if 190 < c[6] < 260)
    states.sort()
    if len(states) != 4 or any(abs(initial[i][5] - states[i][0]) > 25 for i in range(4)):
        raise ValueError("initial states are ambiguous")
    values = {identities[i]: states[i][1] for i in range(4)}

    # Visible card gaps separate the five cards; within each card, x-order is
    # source, source, destination.
    groups = []
    for item in cards:
        if not groups or item[5] - groups[-1][-1][5] > 90:
            groups.append([item])
        else:
            groups[-1].append(item)
    if len(groups) != 5 or any(len(group) != 3 for group in groups):
        raise ValueError("card grouping is ambiguous")
    for group in groups:
        left, right, destination = [_shape(c) for c in group]
        if len({left, right, destination}) != 3:
            raise ValueError("card operands are unresolved")
        values[destination] = values[left] != values[right]
    result = values[_shape(query[0])]
    return "ON" if result else "OFF"
