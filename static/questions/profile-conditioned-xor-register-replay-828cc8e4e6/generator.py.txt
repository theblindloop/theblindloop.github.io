"""Deterministic renderer and analytic replay for XOR register assignments."""

from __future__ import annotations

import copy
import random
from PIL import Image, ImageDraw, ImageFont

SHAPES = ("circle", "square", "triangle", "diamond")
TOKEN_COLORS = ((211, 62, 72), (42, 119, 211), (35, 164, 101), (231, 137, 31))
PAPER = {"paper_a": (248, 249, 246), "paper_b": (248, 249, 246)}
INK = (31, 38, 48)
ON = (246, 192, 35)
OFF = (128, 83, 181)
CYAN = (0, 169, 218)


def _font(size: int, bold: bool = False):
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{name}", size)


def _replay(initial: list[bool], operations: list[list[int]]) -> list[bool]:
    values = list(initial)
    for left, right, destination in operations:
        result = values[left] != values[right]
        values[destination] = result
    return values


def sample_scene(seed: int) -> dict:
    desired = bool(int(seed) % 2 == 0)
    for attempt in range(20000):
        rng = random.Random(int(seed) * 10009 + attempt * 131)
        initial = [bool(rng.randrange(2)) for _ in SHAPES]
        if sum(initial) not in (1, 2, 3):
            continue
        operations = []
        for _ in range(5):
            left, right, destination = rng.sample(range(4), 3)
            operations.append([left, right, destination])
        query = rng.randrange(4)
        destinations = [op[2] for op in operations]
        # Require a write to the query, repeated writes, and at least two
        # write-then-read dependencies so order cannot be discarded.
        dependencies = sum(
            op[2] in later[:2]
            for index, op in enumerate(operations[:-1])
            for later in operations[index + 1:index + 2]
        )
        if query not in destinations or len(set(destinations)) == len(destinations) or dependencies < 2:
            continue
        if _replay(initial, operations)[query] != desired:
            continue
        color_order = list(range(4))
        rng.shuffle(color_order)
        return {
            "initial": initial,
            "operations": operations,
            "query": query,
            "palette": {"paper": "paper_a", "color_order": color_order},
            "width": 1000,
            "height": 700,
        }
    raise RuntimeError("could not sample balanced XOR scene")


def analytic_gold(scene: dict) -> str:
    return "ON" if _replay(scene["initial"], scene["operations"])[int(scene["query"])] else "OFF"


def margin(scene: dict) -> float:
    _ = (scene["initial"], scene["operations"], scene["query"])
    return 22.0


def is_quarantined(scene: dict) -> bool:
    try:
        valid_ops = all(
            len(op) == 3 and len(set(op)) == 3 and all(0 <= int(v) < 4 for v in op)
            for op in scene["operations"]
        )
        return not (
            len(scene["initial"]) == 4
            and len(scene["operations"]) == 5
            and 0 <= int(scene["query"]) < 4
            and valid_ops
            and margin(scene) >= 12.0
        )
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    current = scene["palette"]["paper"]
    twin["palette"]["paper"] = "paper_b" if current == "paper_a" else "paper_a"
    return [("paper_name_alias", twin)]


def _draw_shape(draw: ImageDraw.ImageDraw, shape: str, center: tuple[int, int], color, radius: int) -> None:
    x, y = center
    box = (x - radius, y - radius, x + radius, y + radius)
    if shape == "circle":
        draw.ellipse(box, fill=color, outline=INK, width=3)
    elif shape == "square":
        draw.rectangle(box, fill=color, outline=INK, width=3)
    elif shape == "triangle":
        draw.polygon([(x, y-radius), (x-radius, y+radius), (x+radius, y+radius)], fill=color, outline=INK)
    elif shape == "diamond":
        draw.polygon([(x, y-radius), (x-radius, y), (x, y+radius), (x+radius, y)], fill=color, outline=INK)
    else:
        raise ValueError(shape)


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width, height), PAPER[scene["palette"]["paper"]])
    draw = ImageDraw.Draw(image)
    title, label, small = _font(28, True), _font(17, True), _font(14, True)
    colors = {shape: TOKEN_COLORS[scene["palette"]["color_order"][i]] for i, shape in enumerate(SHAPES)}

    draw.text((38, 22), "XOR REGISTER REPLAY", font=title, fill=INK)
    draw.rounded_rectangle((30, 70, 970, 278), radius=18, fill=(255,255,255), outline=INK, width=3)
    draw.text((50, 87), "STARTING REGISTERS", font=label, fill=INK)
    xs = (150, 385, 620, 855)
    for identity, x in enumerate(xs):
        shape = SHAPES[identity]
        _draw_shape(draw, shape, (x, 158), colors[shape], 28)
        state = bool(scene["initial"][identity])
        draw.ellipse((x-20, 208, x+20, 248), fill=ON if state else OFF, outline=INK, width=3)
        draw.text((x-16, 253), "ON" if state else "OFF", font=_font(13, True), fill=INK)

    draw.rounded_rectangle((30, 296, 970, 542), radius=18, fill=(255,255,255), outline=INK, width=3)
    draw.text((50, 312), "XOR CARDS  •  READ LEFT → RIGHT", font=label, fill=INK)
    card_centers = (110, 305, 500, 695, 890)
    for number, (cx, operation) in enumerate(zip(card_centers, scene["operations"]), start=1):
        left, right, destination = operation
        draw.rounded_rectangle((cx-83, 355, cx+83, 515), radius=13, fill=(244,247,250), outline=(91,99,110), width=3)
        draw.text((cx-66, 367), f"{number}", font=small, fill=(75,82,92))
        for offset, identity in ((-50, left), (-4, right), (54, destination)):
            shape = SHAPES[identity]
            _draw_shape(draw, shape, (cx+offset, 445), colors[shape], 18)
        draw.text((cx-36, 399), "XOR", font=_font(12, True), fill=(31,105,175))
        draw.line((cx+18, 445, cx+31, 445), fill=INK, width=3)
        draw.polygon([(cx+35,445),(cx+27,439),(cx+27,451)], fill=INK)

    draw.rounded_rectangle((260, 562, 740, 677), radius=17, fill=(255,255,255), outline=CYAN, width=5)
    draw.text((304, 603), "QUERY", font=label, fill=(0,105,145))
    query_shape = SHAPES[int(scene["query"])]
    _draw_shape(draw, query_shape, (630, 619), colors[query_shape], 27)
    return image
