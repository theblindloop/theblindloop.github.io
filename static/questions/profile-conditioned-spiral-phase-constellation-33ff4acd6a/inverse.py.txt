"""Independent PNG-only inverse for spiral phase."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


_PALETTE = {
    "red": np.array((204.0, 65.0, 61.0)),
    "blue": np.array((49.0, 96.0, 181.0)),
    "green": np.array((39.0, 151.0, 92.0)),
}
_MIN_COMPONENT = 20
_PALETTE_NAMES = tuple(_PALETTE)
_PALETTE_ARRAY = np.stack([_PALETTE[name] for name in _PALETTE_NAMES])


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    """Return centroids of solid-color components without renderer geometry."""

    remaining = mask.copy()
    height, width = remaining.shape
    components = []
    for y, x in zip(*np.nonzero(remaining)):
        if not remaining[y, x]:
            continue
        stack = [(int(y), int(x))]
        remaining[y, x] = False
        count = 0
        sum_x = 0.0
        sum_y = 0.0
        while stack:
            cy, cx = stack.pop()
            count += 1
            sum_x += cx
            sum_y += cy
            for dy in (-1, 0, 1):
                ny = cy + dy
                if ny < 0 or ny >= height:
                    continue
                for dx in (-1, 0, 1):
                    nx = cx + dx
                    if nx < 0 or nx >= width or not remaining[ny, nx]:
                        continue
                    remaining[ny, nx] = False
                    stack.append((ny, nx))
        if count >= _MIN_COMPONENT:
            components.append((sum_x / count, sum_y / count, count))
    return components


def _score(centers: list[tuple[float, float]]) -> float:
    if len(centers) < 4:
        return -1.0
    coords = np.asarray(centers, dtype=np.float64)
    center = coords.mean(axis=0)
    delta = coords - center
    angles = np.mod(np.arctan2(delta[:, 1], delta[:, 0]), 2.0 * math.pi)
    radii = np.sqrt((delta * delta).sum(axis=1))
    order = np.argsort(angles, kind="mergesort")
    ordered = radii[order]
    ranks = np.arange(ordered.size, dtype=np.float64)
    rank_delta = ranks - ranks.mean()
    rank_energy = float((rank_delta * rank_delta).sum())
    best = -1.0
    for cut in range(ordered.size):
        rotated = np.concatenate((ordered[cut:], ordered[:cut]))
        radius_delta = rotated - rotated.mean()
        denom = math.sqrt(rank_energy * float((radius_delta * radius_delta).sum()))
        if denom > 1e-9:
            best = max(best, float((rank_delta * radius_delta).sum() / denom))
    return best


def decision_from_image(image: Image.Image) -> str:
    """Classify the highest radius-versus-angle correlation from the final PNG."""

    if not isinstance(image, Image.Image):
        raise TypeError("decision_from_image expects a PIL image")
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    scores = {}
    distances = np.sqrt(((array[:, :, None, :] - _PALETTE_ARRAY[None, None, :, :]) ** 2).sum(axis=3))
    nearest = np.argmin(distances, axis=2)
    for index, name in enumerate(_PALETTE_NAMES):
        mask = (nearest == index) & (distances[:, :, index] < 115.0)
        centers = _components(mask)
        scores[name] = _score([(x, y) for x, y, _area in centers])
    return max(scores, key=lambda name: (scores[name], name))
