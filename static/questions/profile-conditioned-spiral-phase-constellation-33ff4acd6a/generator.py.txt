"""Deterministic renderer for a global polar-phase constellation question."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw


RENDERER_VERSION = "spiral_phase_constellation-0.1.0"
CANVAS = (720, 420)
SUPERSAMPLE = 4
BACKGROUND = (248, 248, 244)
PALETTE = {
    "red": (204, 65, 61),
    "blue": (49, 96, 181),
    "green": (39, 151, 92),
}
COLORS = tuple(PALETTE)
POINT_COUNT = 14

# These permutations preserve the exact radial multiset while breaking the
# radius-versus-angle pairing.  They are deliberately different from one
# another so no single decoy ordering is itself a fixed local template.
_PERMUTATIONS = (
    (0, 7, 2, 10, 4, 12, 1, 9, 6, 13, 3, 11, 5, 8),
    (8, 1, 11, 3, 13, 5, 0, 7, 2, 10, 4, 12, 6, 9),
)


def _mean_point(points: list[list[float]]) -> tuple[float, float]:
    return (
        sum(float(point[0]) for point in points) / len(points),
        sum(float(point[1]) for point in points) / len(points),
    )


def _spiral_score(points: list[list[float]]) -> float:
    """Best signed correlation after choosing the unknown spiral cut angle."""

    if len(points) < 4:
        return -1.0
    cx, cy = _mean_point(points)
    polar = []
    for x, y in points:
        dx = float(x) - cx
        dy = float(y) - cy
        polar.append((math.atan2(dy, dx) % (2.0 * math.pi), math.hypot(dx, dy)))
    polar.sort(key=lambda item: item[0])
    radii = [item[1] for item in polar]
    ranks = list(range(len(radii)))
    mean_rank = sum(ranks) / len(ranks)
    rank_energy = sum((rank - mean_rank) ** 2 for rank in ranks)
    best = -1.0
    for cut in range(len(radii)):
        rotated = radii[cut:] + radii[:cut]
        mean_radius = sum(rotated) / len(rotated)
        covariance = sum(
            (rank - mean_rank) * (radius - mean_radius)
            for rank, radius in zip(ranks, rotated)
        )
        radius_energy = sum((radius - mean_radius) ** 2 for radius in rotated)
        denominator = math.sqrt(rank_energy * radius_energy)
        if denominator > 1e-8:
            best = max(best, covariance / denominator)
    return best


def _make_points(
    center: tuple[float, float],
    radii: list[float],
    rotation: float,
    rng: random.Random,
    permutation: tuple[int, ...] | None,
) -> list[list[float]]:
    angles = [
        rotation
        + (2.0 * math.pi * index / POINT_COUNT)
        + rng.uniform(-0.018, 0.018)
        for index in range(POINT_COUNT)
    ]
    values = radii if permutation is None else [radii[index] for index in permutation]
    raw = [
        [
            float(center[0]) + values[index] * math.cos(angles[index]),
            float(center[1]) + values[index] * math.sin(angles[index]),
        ]
        for index in range(POINT_COUNT)
    ]
    # Recenter each constellation so the visible centroid is the latent polar
    # center.  This makes the global evidence recoverable without renderer data.
    mx, my = _mean_point(raw)
    return [
        [round(x + float(center[0]) - mx, 3), round(y + float(center[1]) - my, 3)]
        for x, y in raw
    ]


def sample_scene(seed: int) -> dict:
    """Sample one balanced scene with one coherent spiral and two shuffles."""

    rng = random.Random((seed + 7919) * 1009)
    target_color = COLORS[seed % len(COLORS)]
    slot_centers = [(120.0, 210.0), (360.0, 210.0), (600.0, 210.0)]
    rng.shuffle(slot_centers)
    radii = sorted(
        54.0 + 58.0 * index / (POINT_COUNT - 1) + rng.uniform(-1.8, 1.8)
        for index in range(POINT_COUNT)
    )
    dot_radius = round(rng.choice((6.2, 6.6, 7.0)), 2)
    points: dict[str, list[list[float]]] = {}
    decoy_index = 0
    for color, center in zip(COLORS, slot_centers):
        permutation = None if color == target_color else _PERMUTATIONS[decoy_index]
        if permutation is not None:
            decoy_index += 1
        points[color] = _make_points(
            center,
            radii,
            rng.uniform(0.0, 2.0 * math.pi),
            rng,
            permutation,
        )
    return {
        "target_color": target_color,
        "points": points,
        "dot_radius": dot_radius,
    }


def render(scene: dict) -> Image.Image:
    """Render equal-footprint colored dots; list order has no visual meaning."""

    target_color = str(scene["target_color"])
    if target_color not in PALETTE:
        raise ValueError(f"unknown target color {target_color!r}")
    points_by_color = scene["points"]
    radius = float(scene["dot_radius"]) * SUPERSAMPLE
    image = Image.new(
        "RGB", (CANVAS[0] * SUPERSAMPLE, CANVAS[1] * SUPERSAMPLE), BACKGROUND
    )
    draw = ImageDraw.Draw(image)
    # The target-first paint order is intentionally pixel-invariant: groups do
    # not overlap, and this makes target_color a rendered latent field rather
    # than an analytic-only hidden label.
    paint_order = [target_color] + [color for color in COLORS if color != target_color]
    for color in paint_order:
        for x, y in points_by_color[color]:
            px = float(x) * SUPERSAMPLE
            py = float(y) * SUPERSAMPLE
            draw.ellipse(
                (px - radius, py - radius, px + radius, py + radius),
                fill=PALETTE[color],
            )
    return image.resize(CANVAS, Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["target_color"])


def margin(scene: dict) -> float:
    scores = sorted((_spiral_score(scene["points"][color]) for color in COLORS), reverse=True)
    return round(scores[0] - scores[1], 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.22 or any(
        len(scene["points"].get(color, [])) < 10 for color in COLORS
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["points"] = {
        color: [list(point) for point in reversed(scene["points"][color])]
        for color in COLORS
    }
    return [("reverse_same_colour_point_lists", twin)]
