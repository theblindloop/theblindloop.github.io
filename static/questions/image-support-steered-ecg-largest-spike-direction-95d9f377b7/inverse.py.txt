"""Independent image-only inverse for the largest ECG excursion."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _component_sizes(mask: np.ndarray) -> list[int]:
    seen = np.zeros(mask.shape, dtype=bool)
    sizes: list[int] = []
    height, width = mask.shape
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        size = 0
        while queue:
            cy, cx = queue.popleft()
            size += 1
            for ny in range(max(0, cy - 1), min(height, cy + 2)):
                for nx in range(max(0, cx - 1), min(width, cx + 2)):
                    if mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        sizes.append(size)
    return sorted(sizes, reverse=True)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    chroma = arr.max(axis=2) - arr.min(axis=2)
    mask = (chroma >= 58) & (arr.min(axis=2) <= 205)
    if int(mask.sum()) < 380:
        raise ValueError("colored ECG trace missing or severely erased")
    components = [size for size in _component_sizes(mask) if size >= 20]
    if len(components) != 1:
        raise ValueError("ECG trace is not one connected visible component")
    ys, xs = np.nonzero(mask)
    xmin, xmax = int(xs.min()), int(xs.max())
    if xmax - xmin < 125:
        raise ValueError("ECG trace is too short")
    span = xmax - xmin + 1
    end_mask = (xs <= xmin + 0.22 * span) | (xs >= xmax - 0.22 * span)
    baseline = float(np.median(ys[end_mask]))
    upward = baseline - float(ys.min())
    downward = float(ys.max()) - baseline
    if max(upward, downward) < 12.0:
        raise ValueError("dominant ECG excursion is missing")
    if abs(upward - downward) < 8.0:
        raise ValueError("ECG excursion direction is ambiguous")
    return "yes" if upward > downward else "no"
