"""Deterministic scene sampler and renderer for a compact ECG spike."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

PALETTE = ("#176B87", "#7C3AED", "#C2415D", "#087E8B")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0xEC6217)
    spike_up = int(seed) % 2 == 0
    main_height = rng.randint(16, 18)
    counter_height = rng.randint(4, 5)
    band_center = rng.choice((120, 168, 216, 264))
    baseline_shift = (main_height - counter_height) // 2
    return {
        "spike_up": spike_up,
        "trace_color": PALETTE[rng.randrange(len(PALETTE))],
        "background": "#FAF9F5",
        "width": 512,
        "height": 384,
        "left": rng.randint(105, 245),
        "baseline_y": band_center + (baseline_shift if spike_up else -baseline_shift),
        "trace_width": rng.randint(154, 168),
        "main_height": main_height,
        "counter_height": counter_height,
        "line_width": rng.choice((4, 5)),
    }


def _path(scene: dict, scale: int = 1) -> list[tuple[int, int]]:
    left = int(scene["left"])
    base = int(scene["baseline_y"])
    width = int(scene["trace_width"])
    main = int(scene["main_height"])
    counter = int(scene["counter_height"])
    sign = 1 if bool(scene["spike_up"]) else -1
    fractions = (0.0, 0.38, 0.46, 0.53, 0.60, 0.68, 1.0)
    offsets = (0, 0, sign * counter, -sign * main, sign * counter, 0, 0)
    return [
        ((left + round(width * frac)) * scale, (base + offset) * scale)
        for frac, offset in zip(fractions, offsets, strict=True)
    ]


def render(scene: dict) -> Image.Image:
    scale = 4
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), scene["background"])
    draw = ImageDraw.Draw(image)
    left = int(scene["left"])
    base = int(scene["baseline_y"])
    trace_width = int(scene["trace_width"])
    # An achromatic instrument badge supplies context but is ignored by the color oracle.
    sign = 1 if bool(scene["spike_up"]) else -1
    band_center = base - sign * ((int(scene["main_height"]) - int(scene["counter_height"])) // 2)
    pad_x, pad_y = 24, 34
    draw.rounded_rectangle(
        ((left - pad_x) * scale, (band_center - pad_y) * scale,
         (left + trace_width + pad_x) * scale, (band_center + pad_y) * scale),
        radius=13 * scale, fill="#F1F0EC", outline="#D7D5CE", width=2 * scale,
    )
    draw.line(
        ((left - 12) * scale, base * scale,
         (left + trace_width + 12) * scale, base * scale),
        fill="#DFDDD6", width=1 * scale,
    )
    points = _path(scene, scale)
    draw.line(points, fill=scene["trace_color"], width=int(scene["line_width"]) * scale, joint="curve")
    radius = int(scene["line_width"]) * scale // 2
    for x, y in (points[0], points[-1]):
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=scene["trace_color"])
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "yes" if bool(scene["spike_up"]) else "no"


def margin(scene: dict) -> float:
    return float(int(scene["main_height"]) - int(scene["counter_height"]))


def is_quarantined(scene: dict) -> bool:
    try:
        points = _path(scene)
        width, height = int(scene["width"]), int(scene["height"])
        if margin(scene) < 8.0 or int(scene["line_width"]) < 4:
            return True
        return any(x < 20 or x >= width - 20 or y < 20 or y >= height - 20 for x, y in points)
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["trace_color"] = str(scene["trace_color"]).lower()
    return [("hex_case_alias", twin)]
